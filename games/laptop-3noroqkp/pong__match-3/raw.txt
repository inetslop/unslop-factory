let audioCtx = null;
function initAudio() {
    if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();

function playSound(freq, dur, type, vol) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = type || 'square';
    osc.frequency.value = freq;
    gain.gain.setValueAtTime(vol || 0.15, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + dur);
}
function sfxBounce() { playSound(220, 0.08, 'square', 0.1); }
function sfxMatch() { playSound(440, 0.15, 'sine', 0.2); playSound(660, 0.15, 'sine', 0.15); }

// Game constants
const COLS = 7, ROWS = 8;
const COLORS = ['#ff4466', '#44aaff', '#44ff88', '#ffdd44', '#cc66ff'];

function getGridLayout() {
    const gridW = Math.min(W * 0.55, 420);
    const gridH = Math.min(H * 0.75, 480);
    const cellW = gridW / COLS;
    const cellH = gridH / ROWS;
    const gx = W * 0.32;

    return { gx, gy, cellW, cellH, gridW, gridH };
}

function initGrid() {
    grid = [];
    for (let r = 0; r < ROWS; r++) {
        grid[r] = [];
        for (let c = 0; c < COLS; c++) {
            grid[r][c] = Math.floor(Math.random() * COLORS.length);
        }
    }
    // Remove initial matches
    let hasMatch = true;
    while (hasMatch) {
        hasMatch = false;
        for (let r = 0; r < ROWS; r++) {
            for (let c = 0; c < COLS; c++) {
                if (c + 2 < COLS && grid[r][c] === grid[r][c+1] && grid[r][c] === grid[r][c+2]) {
                    grid[r][c+2] = (grid[r][c] + 1 + Math.floor(Math.random()*4)) % COLORS.length;
                    hasMatch = true;
                }
                if (r + 2 < ROWS && grid[r][c] === grid[r+1][c] && grid[r][c] === grid[r+2][c]) {
                    grid[r+2][c] = (grid[r][c] + 1 + Math.floor(Math.random()*4)) % COLORS.length;
                    hasMatch = true;
                }
            }
        }
    }
}

// Match logic
function checkMatches() {
    let matched = new Set();
    // Horizontal
    for (let r = 0; r < ROWS; r++) {
        for (let c = 0; c < COLS - 2; c++) {
            if (grid[r][c] !== -1 && grid[r][c] === grid[r][c+1] && grid[r][c] === grid[r][c+2]) {
                matched.add(r * COLS + c);
                matched.add(r * COLS + c + 1);
                matched.add(r * COLS + c + 2);
            }
        }
    }
    // Vertical
    for (let c = 0; c < COLS; c++) {
        for (let r = 0; r < ROWS - 2; r++) {
            if (grid[r][c] !== -1 && grid[r][c] === grid[r+1][c] && grid[r][c] === grid[r+2][c]) {
                matched.add(r * COLS + c);
                matched.add((r+1) * COLS + c);
                matched.add((r+2) * COLS + c);
            }
        }
    }
    return matched;
}

function clearMatches(matched) {
    if (matched.size === 0) return false;
    comboCounter++;
    const points = matched.size * 10 * comboCounter;
    score += points;
    
    const layout = getGridLayout();
    floatingTexts.push({ x: layout.gx + layout.gridW/2, y: layout.gy + layout.gridH/2, text: '+' + points, life: 1, vy: -60 });
    
    sfxMatch();
    
    for (const idx of matched) {
        const r = Math.floor(idx / COLS);
        const c = idx % COLS;
        grid[r][c] = -1;
        // Spawn particles
        const px = layout.gx + c * layout.cellW + layout.cellW/2;
        const py = layout.gy + r * layout.cellH + layout.cellH/2;
        for (let i = 0; i < 5; i++) {
            particles.push({
                x: px, y: py,
                vx: (Math.random()-0.5) * 200,
                vy: (Math.random()-0.5) * 200,
                life: 0.6,
                color: COLORS[grid[r][c] >= 0 ? grid[r][c] : Math.floor(Math.random()*5)]
            });
        }
    }
    
    // Gravity
    for (let c = 0; c < COLS; c++) {
        let write = ROWS - 1;
        for (let r = ROWS - 1; r >= 0; r--) {
            if (grid[r][c] !== -1) {
                grid[write][c] = grid[r][c];
                if (write !== r) grid[r][c] = -1;
                write--;
            }
        }
        for (let r = write; r >= 0; r--) {
            grid[r][c] = Math.floor(Math.random() * COLORS.length);
        }
    }
    
    // Check for cascades
    setTimeout(() => {
        const moreMatches = checkMatches();
        if (moreMatches.size > 0) clearMatches(moreMatches);
        else comboCounter = 0;
    }, 200);
    
    return true;
}

function swapGems(r, c) {
    const neighbors = [];
    if (r > 0) neighbors.push([r-1, c]);
    if (r < ROWS-1) neighbors.push([r+1, c]);
    if (c > 0) neighbors.push([r, c-1]);
    if (c < COLS-1) neighbors.push([r, c+1]);
    
    if (neighbors.length === 0) return;
    const [nr, nc] = neighbors[Math.floor(Math.random() * neighbors.length)];
    
    const tmp = grid[r][c];
    grid[r][c] = grid[nr][nc];
    grid[nr][nc] = tmp;
}

function ballHitsGrid() {
    const layout = getGridLayout();
    const r = Math.floor((ball.y - layout.gy) / layout.cellH);
    const c = Math.floor((ball.x - layout.gx) / layout.cellW);
    
    if (r >= 0 && r < ROWS && c >= 0 && c < COLS) {
        const cellIdx = r * COLS + c;
        if (cellIdx !== lastInteractionCell && interactionCooldown <= 0) {
            lastInteractionCell = cellIdx;
            interactionCooldown = 0.5;
            swapGems(r, c);
            const matches = checkMatches();
            if (matches.size > 0) clearMatches(matches);
        }
    }
}

function update(dt) {
    if (state !== 'playing') return;
    
    timer -= dt;
    if (timer <= 0) {
        timer = 0;
        state = 'gameover';
        sfxGameOver();
        return;
    }
    
    // Paddle
    if (keys['KeyW'] || keys['ArrowUp']) paddle.y -= paddle.speed * dt;
    if (keys['KeyS'] || keys['ArrowDown']) paddle.y += paddle.speed * dt;
    paddle.y = Math.max(0, Math.min(H - paddle.h, paddle.y));
    
    // Ball physics
    ball.x += ball.vx * dt;
    ball.y += ball.vy * dt;
    
    // Top/bottom bounce
    if (ball.y < 15) { ball.y = 15; ball.vy = Math.abs(ball.vy); sfxBounce(); }
    if (ball.y > H - 15) { ball.y = H - 15; ball.vy = -Math.abs(ball.vy); sfxBounce(); }
    
    // Left paddle bounce
    const px = W * 0.08;
    if (ball.x < px + 8 && ball.vx < 0) {
        if (ball.y > paddle.y && ball.y < paddle.y + paddle.h) {
            ball.x = px + 8;
            ball.vx = Math.abs(ball.vx);
            // Deflect based on where it hits the paddle
            const hitPos = (ball.y - paddle.y) / paddle.h;
            ball.vy += (hitPos - 0.5) * 200;
            sfxPaddle();
        } else {
            // Ball missed the paddle - reset from left
            ball.x = W * 0.15;
            ball.y = H / 2;
            ball.vx = 300 + Math.random() * 100;
            ball.vy = (Math.random() - 0.5) * 200;
        }
    }
    
    // Right wall bounce
    if (ball.x > W - 20) {
        ball.x = W - 20;
        ball.vx = -Math.abs(ball.vx);
        sfxBounce();
    }
    
    // Interaction cooldown
    interactionCooldown -= dt;
    
    // Ball-gem interaction
    ballHitsGrid();
    
    // Particles
    for (let i = particles.length - 1; i >= 0; i--) {
        particles[i].x += particles[i].vx * dt;
        particles[i].y += particles[i].vy * dt;
        particles[i].life -= dt;
        if (particles[i].life <= 0) particles.splice(i, 1);
    }
    
    // Floating texts
    for (let i = floatingTexts.length - 1; i >= 0; i--) {
        floatingTexts[i].y += floatingTexts[i].vy * dt;
        floatingTexts[i].life -= dt;
        if (floatingTexts[i].life <= 0) floatingTexts.splice(i, 1);
    }
}
