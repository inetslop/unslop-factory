  [0,-1]  // W
];
const OPP = [2,3,0,1]; // opposite direction index

function rotMask(m){ return ((m << 1) | (m >> 3)) & 7; }
// Rotate bitmask right by 90°: N(1)->E(2)->S(4)->W(8)... wait let me use bits properly.
// Bit 0=N, bit 1=E, bit 2=S, bit 3=W
function rotateMask(m){ return ((m << 1) | (m >> 3)) & 15; }

function playTone(freq, dur, type, vol){
  if(!audioCtx || muted) return;
  const o = audioCtx.createOscillator();
  const g = audioCtx.createGain();

  o.frequency.value = freq;
  g.gain.setValueAtTime(vol||0.15, audioCtx.currentTime);
  g.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + (dur||0.1));
  o.connect(g); g.connect(audioCtx.destination);
  o.start(); o.stop(audioCtx.currentTime+(dur||0.1));
}

function generatePuzzle(c,r){
  // Generate spanning tree using randomized DFS
  const visited = Array.from({length:r},()=>Array(c).fill(false));
  const edges = []; // [r,c] -> set of connected dirs (bitmask)
  const conn = Array.from({length:r},()=>Array.from({length:c},()=>0));

  function dfs(cr,cc){
    visited[cr][cc]=true;
    let shuffled=[0,1,2,3].sort(()=>Math.random()-0.5);
    for(const d of shuffled){
      const nr=cr+DIRS[d][0], nc=cc+DIRS[d][1];
      if(nr>=0&&nr<r&&nc>=0&&nc<c&&!visited[nr][nc]){
        conn[cr][cc] |= (1<<d);
        conn[nr][nc] |= (1<<OPP[d]);
        edges.push([cr,cc,d]);
        dfs(nr,nc);
      }
    }
  }

  // Start from a random corner or edge cell
  const startR = Math.floor(Math.random()*r);
  const startC = Math.floor(Math.random()*c);
  sourceR = startR; sourceC = startC;
  dfs(startR, startC);

  // Build grid with solution rotations = 0 (since we generated them directly)
  const solGrid = [];
  for(let i=0;i<r;i++){
    solGrid[i]=[];
    for(let j=0;j<c;j++){
      solGrid[i][j] = { connections: conn[i][j], rotation: 0 };
    }
  }

  // Shuffle rotations (1-3 extra rotations per tile, at least ensure not all solved)
  const shuffled = [];
  for(let i=0;i<r;i++){
    shuffled[i]=[];
    for(let j=0;j<c;j++){
      let extra = Math.floor(Math.random()*4);
      // Ensure it's not the solution (at least some tiles are wrong)
      if(extra===0 && Math.random()<0.5) extra = 1+Math.floor(Math.random()*3);
      shuffled[i][j] = {
        connections: rotateMaskN(solGrid[i][j].connections, extra),
        rotation: extra
      };
    }
  }

  // Make sure puzzle isn't already solved
  if(isSolved(shuffled, c, r)){
    // Rotate a random non-source tile
    let tr = Math.floor(Math.random()*r), tc = Math.floor(Math.random()*c);
    if(tr===startR&&tc===startC){tr=(tr+1)%r;tc=(tc+1)%c;}
    shuffled[tr][tc].rotation = (shuffled[tr][tc].rotation + 1) % 4;
    shuffled[tr][tc].connections = rotateMaskN(solGrid[tr][tc].connections, shuffled[tr][tc].rotation);
  }

  return shuffled;
}

function rotateMaskN(m, times){
  for(let i=0;i<times;i++) m = rotateMask(m);
  return m;
}

function isSolved(g, c, r){
  // BFS from source
  const visited = Array.from({length:r},()=>Array(c).fill(false));
  const queue = [[sourceR, sourceC]];
  visited[sourceR][sourceC]=true;
  while(queue.length){
    const [cr,cc]=queue.shift();
    for(let d=0;d<4;d++){
      if(!(g[cr][cc].connections & (1<<d))) continue;
      const nr=cr+DIRS[d][0], nc=cc+DIRS[d][1];
      if(nr<0||nr>=r||nc<0||nc>=c) continue;
      if(visited[nr][nc]) continue;
      // Check neighbor connects back
      if(g[nr][nc].connections & (1<<OPP[d])){
        visited[nr][nc]=true;
        queue.push([nr,nc]);
      }
    }
  }
  let count=0;
  for(let i=0;i<r;i++)for(let j=0;j<c;j++)if(visited[i][j])count++;
  return count === r*c;
}

function getWater(g, c, r){
  const visited = Array.from({length:r},()=>Array(c).fill(false));
  const queue = [[sourceR, sourceC]];
  visited[sourceR][sourceC]=true;
  while(queue.length){
    const [cr,cc]=queue.shift();
    for(let d=0;d<4;d++){
      if(!(g[cr][cc].connections & (1<<d))) continue;
      const nr=cr+DIRS[d][0], nc=cc+DIRS[d][1];
      if(nr<0||nr>=r||nc<0||nc>=c) continue;
      if(visited[nr][nc]) continue;
      if(g[nr][nc].connections & (1<<OPP[d])){
        visited[nr][nc]=true;
        queue.push([nr,nc]);
      }
    }
  }
  return visited;
}

function resize(){
  const maxW = Math.min(window.innerWidth - 40, 500);

function draw(){
  const w = canvas.width, h = canvas.height;
  ctx.fillStyle = '#16213e';
  ctx.fillRect(0,0,w,h);

  const water = solved ? null : getWater(grid, cols, rows);
  if(!water) for(let i=0;i<rows;i++)for(let j=0;j<cols;j++) water[i][j]=true;

  // Draw grid lines
  ctx.strokeStyle = 'rgba(255,255,255,0.08)';
  ctx.lineWidth = 1;
  for(let i=0;i<=cols;i++){ctx.beginPath();ctx.moveTo(i*tileSize,0);ctx.lineTo(i*tileSize,h);ctx.stroke();}
  for(let j=0;j<=rows;j++){ctx.beginPath();ctx.moveTo(0,j*tileSize);ctx.lineTo(w,j*tileSize);ctx.stroke();}

  // Draw tiles
  for(let r=0;r<rows;r++){
    for(let c=0;c<cols;c++){
      const x = c*tileSize, y = r*tileSize;
      const hasWater = water[r][c];
      const isSource = (r===sourceR && c===sourceC);

      // Tile background
      ctx.fillStyle = isSource ? '#1b5e20' : 'rgba(255,255,255,0.03)';
      ctx.fillRect(x+1,y+1,tileSize-2,tileSize-2);

      if(isSource){
        // Draw source icon
        ctx.fillStyle = '#4caf50';
        ctx.font = `bold ${tileSize*0.5}px sans-serif`;
        ctx.textAlign='center';ctx.textBaseline='middle';
        ctx.fillText('⛲', x+tileSize/2, y+tileSize/2);
      }

      // Draw pipe connections
      const conn = grid[r][c].connections;
      const pipeColor = hasWater ? '#4fc3f7' : '#546e7a';
      const pipeW = Math.max(tileSize*0.18, 6);

      ctx.strokeStyle = pipeColor;
      ctx.lineWidth = pipeW;
      ctx.lineCap = 'round';
      ctx.shadowColor = hasWater ? 'rgba(79,195,247,0.4)' : 'transparent';
      ctx.shadowBlur = hasWater ? 8 : 0;

      const cx = x + tileSize/2, cy = y + tileSize/2;
      const half = tileSize/2;

      // Draw each direction connection as a line from center to edge
      for(let d=0;d<4;d++){
        if(conn & (1<<d)){
          ctx.beginPath();
          ctx.moveTo(cx, cy);
          const dir = DIRS[d];
          ctx.lineTo(cx + dir[0]*half, cy + dir[1]*half);
          ctx.stroke();
        }
      }

      // Center dot

        ctx.arc(cx, cy, pipeW*0.6, 0, Math.PI*2);

        ctx.fill();

      ctx.shadowBlur = 0;

    ctx.lineWidth = 3;

  }
}
