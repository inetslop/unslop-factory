```html
<!DOCTYPE html>
<html lang="en">
<head>
<meta charset="UTF-8">
<meta name="viewport" content="width=device-width, initial-scale=1.0">
<title>Gravity Cards</title>
<style>
* { margin: 0; padding: 0; box-sizing: border-box; }
html, body { width: 100%; height: 100%; overflow: hidden; background: #1a1a2e; }
canvas { display: block; width: 100vw; height: 100vh; }
</style>

<canvas id="game"></canvas>
<script>
const canvas = document.getElementById('game');
const ctx = canvas.getContext('2d');

let W, H;
function resize() {
    W = canvas.width = window.innerWidth;
    H = canvas.height = window.innerHeight;
}
window.addEventListener('resize', resize);
resize();

// Audio
let audioCtx = null;
function initAudio() {
    if (!audioCtx) {
        audioCtx = new (window.AudioContext || window.webkitAudioContext)();
    }
}
function playSound(freq, type, duration, volume) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = type;
    osc.frequency.value = freq;
    gain.gain.setValueAtTime(volume || 0.15, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + duration);
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + duration);
}

// Game state
let gameState = 'start'; // start, playing, gameover
let score = 0;
let lives = 3;
let highScore = 0;
try { highScore = parseInt(localStorage.getItem('gravityCardsHigh') || '0'); } catch(e) {}
let tilt = 0;
let targetTilt = 0;
let cards = [];
let particles = [];
let spawnTimer = 0;
let spawnInterval = 120;
let speedMultiplier = 1;
let frameCount = 0;
let shakeTime = 0;

const COLORS = [
    { name: 'red', value: '#e74c3c' },
    { name: 'blue', value: '#3498db' },
    { name: 'green', value: '#2ecc71' },
    { name: 'yellow', value: '#f1c40f' }
];

const SYMBOLS = ['♥', '♦', '♣', '♠'];

function getSlots() {
    const slotW = W / COLORS.length;
    return COLORS.map((c, i) => ({
        x: i * slotW,
        y: H - 80,
        w: slotW,
        h: 70,
        color: c.value,
        name: c.name
    }));
}

function spawnCard() {
    const colorIdx = Math.floor(Math.random() * COLORS.length);
    const cardW = 50;
    const cardH = 70;
    const x = 60 + Math.random() * (W - 120);
    cards.push({
        x: x,
        y: -cardH,
        w: cardW,
        h: cardH,
        vx: 0,
        vy: 0,
        colorIdx: colorIdx,
        symbol: SYMBOLS[Math.floor(Math.random() * SYMBOLS.length)],
        rotation: 0,
        rotVel: 0,
        alive: true
    });
}

function spawnParticles(x, y, color, count) {
    for (let i = 0; i < count; i++) {
        const angle = Math.random() * Math.PI * 2;
        const speed = 2 + Math.random() * 4;
        particles.push({
            x: x, y: y,
            vx: Math.cos(angle) * speed,
            vy: Math.sin(angle) * speed - 2,
            life: 1,
            decay: 0.02 + Math.random() * 0.03,
            color: color,
            size: 3 + Math.random() * 5
        });
    }
}

function resetGame() {
    score = 0;
    lives = 3;
    cards = [];
    particles = [];
    tilt = 0;
    targetTilt = 0;
    spawnTimer = 0;
    spawnInterval = 120;
    speedMultiplier = 1;
    frameCount = 0;
    shakeTime = 0;
}

function startGame() {
    resetGame();
    gameState = 'playing';
}

// Input
let keys = {};
window.addEventListener('keydown', (e) => {
    keys[e.key] = true;
    if (!audioCtx) initAudio();
    
    if (gameState === 'start' && (e.key === 'Enter' || e.key === ' ')) {
        startGame();
        return;
    }
    if (gameState === 'gameover' && (e.key === 'Enter' || e.key === ' ')) {
        startGame();
        return;
    }
});
window.addEventListener('keyup', (e) => { keys[e.key] = false; });

// Update
function update() {
    if (gameState !== 'playing') return;
    
    frameCount++;
    speedMultiplier = 1 + Math.floor(frameCount / 500) * 0.3;
    spawnInterval = Math.max(40, 120 - Math.floor(frameCount / 300) * 5);
    
    // Tilt control
    if (keys['ArrowLeft'] || keys['a'] || keys['A']) targetTilt = -0.6;
    else if (keys['ArrowRight'] || keys['d'] || keys['D']) targetTilt = 0.6;
    else targetTilt = 0;
    
    tilt += (targetTilt - tilt) * 0.15;
    
    // Spawn cards
    spawnTimer++;
    if (spawnTimer >= spawnInterval) {
        spawnTimer = 0;
        spawnCard();
    }
    
    const slots = getSlots();
    const gravity = 0.25 * speedMultiplier;
    const tiltForce = Math.sin(tilt) * 0.8 * speedMultiplier;
    
    for (let i = cards.length - 1; i >= 0; i--) {
        const c = cards[i];
        if (!c.alive) { cards.splice(i, 1); continue; }
        
        c.vy += gravity;
        c.vx += tiltForce;
        c.vx *= 0.98; // friction
        c.x += c.vx;
        c.y += c.vy;
        c.rotVel = tiltForce * 0.3;
        c.rotation += c.rotVel;
        
        // Bounce off walls
        if (c.x < 0) { c.x = 0; c.vx *= -0.5; }
        if (c.x + c.w > W) { c.x = W - c.w; c.vx *= -0.5; }
        
        // Check slot collision
        for (let s = 0; s < slots.length; s++) {
            const sl = slots[s];
            if (c.y + c.h >= sl.y && c.x + c.w > sl.x && c.x < sl.x + sl.w) {
                c.alive = false;
                cards.splice(i, 1);
                
                if (s === c.colorIdx) {
                    // Correct!
                    score += Math.floor(10 * speedMultiplier);
                    playSound(523 + s * 100, 'sine', 0.3, 0.2);
                    spawnParticles(c.x + c.w/2, sl.y, COLORS[c.colorIdx].value, 15);
                } else {
                    // Wrong!
                    lives--;
                    shakeTime = 15;
                    playSound(150, 'sawtooth', 0.4, 0.3);
                    spawnParticles(c.x + c.w/2, sl.y, '#ff0000', 10);
                }
                
                if (lives <= 0) {
                    gameState = 'gameover';
                    playSound(100, 'sawtooth', 0.8, 0.4);
                    try { 
                        if (score > highScore) {
                            highScore = score;
                            localStorage.setItem('gravityCardsHigh', highScore);
                        }
                    } catch(e) {}
                }
                break;
            }
        }
        
        // Fell off bottom without hitting slot
        if (c.y > H + 50) {
            c.alive = false;
            cards.splice(i, 1);
        }
    }
    
    // Update particles
    for (let i = particles.length - 1; i >= 0; i--) {
        const p = particles[i];
        p.x += p.vx;
        p.y += p.vy;
        p.vy += 0.3;
        p.life -= p.decay;
        if (p.life <= 0) particles.splice(i, 1);
    }
    
    // Shake decay
    if (shakeTime > 0) shakeTime--;
}

// Draw
function drawCard(x, y, w, h, colorIdx, symbol, rotation) {
    ctx.save();
    ctx.translate(x + w/2, y + h/2);
    ctx.rotate(rotation);
    
    // Card body
    ctx.fillStyle = '#fff';
    ctx.strokeStyle = COLORS[colorIdx].value;
    ctx.lineWidth = 3;
    roundRect(ctx, -w/2, -h/2, w, h, 8);
    ctx.fill();
    ctx.stroke();
    
    // Color band
    ctx.fillStyle = COLORS[colorIdx].value;
    ctx.globalAlpha = 0.3;
    roundRect(ctx, -w/2 + 4, -h/2 + 4, w - 8, h - 8, 5);
    ctx.fill();
    ctx.globalAlpha = 1;
    
    // Symbol
    ctx.fillStyle = COLORS[colorIdx].value;
    ctx.font = 'bold 28px serif';
    ctx.textAlign = 'center';
    ctx.textBaseline = 'middle';
    ctx.fillText(symbol, 0, 0);
    
    ctx.restore();
}

function roundRect(ctx, x, y, w, h, r) {
    ctx.beginPath();
    ctx.moveTo(x + r, y);
    ctx.lineTo(x + w - r, y);
    ctx.quadraticCurveTo(x + w, y, x + w, y + r);
    ctx.lineTo(x + w, y + h - r);
    ctx.quadraticCurveTo(x + w, y + h, x + w - r, y + h);
    ctx.lineTo(x + r, y + h);
    ctx.quadraticCurveTo(x, y + h, x, y + h - r);
    ctx.lineTo(x, y + r);
    ctx.quadraticCurveTo(x, y, x + r, y);
    ctx.closePath();
}

function draw() {
    ctx.save();
    
    // Screen shake
    if (shakeTime > 0) {
        const sx = (Math.random() - 0.5) * shakeTime;
        const sy = (Math.random() - 0.5) * shakeTime;
        ctx.translate(sx, sy);
    }
    
    // Background
    const bgGrad = ctx.createLinearGradient(0, 0, 0, H);
    bgGrad.addColorStop(0, '#1a1a2e');
    bgGrad.addColorStop(1, '#16213e');
    ctx.fillStyle = bgGrad;
    ctx.fillRect(-20, -20, W + 40, H + 40);
    
    if (gameState === 'start') {
        drawStartScreen();
    } else if (gameState === 'playing' || gameState === 'gameover') {
        drawGame();
        if (gameState === 'gameover') drawGameOver();
    }
    
    ctx.restore();
}

function drawGame() {
    const slots = getSlots();
    
    // Draw tilt indicator (subtle lines)
    ctx.save();
    ctx.globalAlpha = 0.1;
    ctx.strokeStyle = '#fff';
    ctx.lineWidth = 1;
    for (let i = 0; i < 5; i++) {
        const y = 100 + i * 80;
        ctx.beginPath();
        ctx.moveTo(0, y);
        ctx.lineTo(W, y + Math.sin(tilt) * 30);
        ctx.stroke();
    }
    ctx.restore();
    
    // Draw slots
    for (let s = 0; s < slots.length; s++) {
        const sl = slots[s];
        ctx.save();
        ctx.globalAlpha = 0.4 + Math.sin(frameCount * 0.05) * 0.1;
        ctx.fillStyle = COLORS[s].value;
        roundRect(ctx, sl.x, sl.y, sl.w, sl.h, 8);
        ctx.fill();
        ctx.globalAlpha = 1;
        
        // Slot label
        ctx.fillStyle = '#fff';
        ctx.font = 'bold 20px sans-serif';
        ctx.textAlign = 'center';
        ctx.textBaseline = 'middle';
        ctx.fillText(COLORS[s].name.toUpperCase(), sl.x + sl.w/2, sl.y + sl.h/2);
        
        // Arrow indicator
        ctx.font = '16px sans-serif';
        ctx.globalAlpha = 0.7;
        ctx.fillText('▼', sl.x + sl.w/2, sl.y + 15);
        ctx.restore();
    }
    
    // Draw cards
    for (const c of cards) {
        drawCard(c.x, c.y, c.w, c.h, c.colorIdx, c.symbol, c.rotation);
    }
    
    // Draw particles
    for (const p of particles) {
        ctx.save();
        ctx.globalAlpha = p.life;
        ctx.fillStyle = p.color;
        ctx.beginPath();
        ctx.arc(p.x, p.y, p.size * p.life, 0, Math.PI * 2);
        ctx.fill();
        ctx.restore();
    }
    
    // HUD
    ctx.fillStyle = '#fff';
    ctx.font = 'bold 24px sans-serif';
    ctx.textAlign = 'left';
    ctx.textBaseline = 'top';
    ctx.fillText(`Score: ${score}`, 20, 20);
    ctx.fillText(`Lives: ${'❤️'.repeat(Math.max(0, lives))}`, 20, 50);
    
    // High score
    ctx.textAlign = 'right';
    ctx.font = '16px sans-serif';
    ctx.fillStyle = '#aaa';
    ctx.fillText(`Best: ${highScore}`, W - 20, 20);
    
    // Tilt indicator
    ctx.textAlign = 'center';
    ctx.font = '14px sans-serif';
    ctx.fillStyle = '#888';
    const tiltDeg = Math.round(tilt * 57.3);
    ctx.fillText(`Tilt: ${tiltDeg}°`, W/2, 20);
    
    // Speed indicator
    ctx.textAlign = 'left';
    ctx.font = '14px sans-serif';
    ctx.fillStyle = Math.floor(speedMultiplier) > 1 ? '#f39c12' : '#888';
    ctx.fillText(`Speed: x${speedMultiplier.toFixed(1)}`, 20, 75);
}

function drawStartScreen() {
    // Title
    ctx.fillStyle = '#fff';
    ctx.font = 'bold 64px sans-serif';
    ctx.textAlign = 'center';
    ctx.textBaseline = 'middle';
    ctx.fillText('GRAVITY CARDS', W/2, H/2 - 100);
    
    // Subtitle
    ctx.font = '24px sans-serif';
    ctx.fillStyle = '#aaa';
    ctx.fillText('Guide cards to their matching slots!', W/2, H/2 - 50);
    
    // Instructions
    ctx.font = '18px sans-serif';
    ctx.fillStyle = '#ccc';
    const instructions = [
        '← / A  : Tilt Left',
        '→ / D  : Tilt Right',
        '',
        'Match card colors to bottom slots',
        'Wrong slot = lose a life!',
        'Speed increases over time'
    ];
    instructions.forEach((line, i) => {
        ctx.fillText(line, W/2, H/2 + 10 + i * 28);
    });
    
    // Start prompt
    const pulse = 0.5 + Math.sin(Date.now() * 0.004) * 0.5;
    ctx.globalAlpha = pulse;
    ctx.font = 'bold 28px sans-serif';
    ctx.fillStyle = '#fff';
    ctx.fillText('Press ENTER or SPACE to Start', W/2, H/2 + 220);
    ctx.globalAlpha = 1;
    
    // Draw decorative cards
    for (let i = 0; i < 4; i++) {
        const dx = W/2 - 90 + i * 60;
        const dy = H/2 + 150 + Math.sin(Date.now() * 0.003 + i) * 10;
        drawCard(dx, dy, 40, 55, i, SYMBOLS[i], Math.sin(Date.now() * 0.002 + i) * 0.2);
    }
}

function drawGameOver() {
    ctx.fillStyle = 'rgba(0,0,0,0.7)';
    ctx.fillRect(0, 0, W, H);
    
    ctx.fillStyle = '#e74c3c';
    ctx.font = 'bold 56px sans-serif';
    ctx.textAlign = 'center';
    ctx.textBaseline = 'middle';
    ctx.fillText('GAME OVER', W/2, H/2 - 60);
    
    ctx.fillStyle = '#fff';
    ctx.font = '28px sans-serif';
    ctx.fillText(`Final Score: ${score}`, W/2, H/2);
    
    const isNewHigh = score >= highScore && score > 0;
    if (isNewHigh) {
        ctx.fillStyle = '#f1c40f';
        ctx.font = 'bold 24px sans-serif';
        ctx.fillText('★ NEW HIGH SCORE! ★', W/2, H/2 + 45);
    } else {
        ctx.fillStyle = '#aaa';
        ctx.font = '20px sans-serif';
        ctx.fillText(`Best: ${highScore}`, W/2, H/2 + 45);
    }
    
    const pulse = 0.5 + Math.sin(Date.now() * 0.004) * 0.5;
    ctx.globalAlpha = pulse;
    ctx.fillStyle = '#fff';
    ctx.font = '22px sans-serif';
    ctx.fillText('Press ENTER or SPACE to Play Again', W/2, H/2 + 100);
    ctx.globalAlpha = 1;
}

// Game loop
function gameLoop() {
    update();
    draw();
    requestAnimationFrame(gameLoop);
}

gameLoop();
</script>
```
