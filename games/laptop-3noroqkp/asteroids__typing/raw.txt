<script>
(function(){

// Audio
let audioCtx = null;
function initAudio(){ if(!audioCtx) audioCtx = new (window.AudioContext||webkitAudioContext)(); }
function playLaser(){
if(!audioCtx) return;
const o = audioCtx.createOscillator(); const g = audioCtx.createGain();
o.type='sine'; o.frequency.value=880; g.gain.setValueAtTime(0.15,audioCtx.currentTime);
g.gain.exponentialRampToValueAtTime(0.001,audioCtx.currentTime+0.1);
o.connect(g); g.connect(audioCtx.destination); o.start(); o.stop(audioCtx.currentTime+0.1);
}
function playExplosion(){
if(!audioCtx) return;
const o = audioCtx.createOscillator(); const g = audioCtx.createGain();
o.type='sawtooth'; o.frequency.setValueAtTime(200,audioCtx.currentTime);
o.frequency.exponentialRampToValueAtTime(60,audioCtx.currentTime+0.4);
g.gain.setValueAtTime(0.3,audioCtx.currentTime); g.gain.exponentialRampToValueAtTime(0.001,audioCtx.currentTime+0.4);
o.connect(g); g.connect(audioCtx.destination); o.start(); o.stop(audioCtx.currentTime+0.4);
}
function playDamage(){
if(!audioCtx) return;
const o = audioCtx.createOscillator(); const g = audioCtx.createGain();
o.type='square'; o.frequency.value=80; g.gain.setValueAtTime(0.25,audioCtx.currentTime);
g.gain.exponentialRampToValueAtTime(0.001,audioCtx.currentTime+0.3);
o.connect(g); g.connect(audioCtx.destination); o.start(); o.stop(audioCtx.currentTime+0.3);
}

// Words by difficulty
const words = {
short: ['AT','GO','UP','IN','IT','TO','OR','AN','ON','AS','BE','DO','IF','NO','OF','SO','WE','HE','ME','MY','BY','AM','IS','US','OH','HA','AA','OK','AW','UM','EH'],
med: ['THE','AND','FOR','ARE','BUT','NOT','YOU','ALL','CAN','HAD','HER','WAS','ONE','OUR','OUT','DAY','GET','HAS','HIM','HIS','HOW','MAN','NEW','NOW','OLD','SEE','TWO','WILL','WON','WHO','WHY','MAY','SHE','USE','WAY','YOU','PUT','BIG','TOP','RUN'],
long: ['GAME','WORD','STAR','SHIP','CODE','TYPE','FAST','LIVE','FIRE','GLOW','NEON','VOID','PUSH','DRAG','SCAN','LOCK','BEAM','NODE','BYTE','DATA','LINK','PORT','GRID','FLUX','CORE','HEX','DIAL','TUNE','SYNC','RAIL','BOLT','WARP','PLAS','CHIM'],
xl: ['ASTEROID','LASER','QUANTUM','NEBULA','COSMIC','GALAXY','PHOTON','CIPHER','SIGNAL','VECTOR','ORBITAL','PULSAR']
};

let stars = [];
function genStars(){ stars=[]; for(let i=0;i<150;i++){ stars.push({x:Math.random()*W,y:Math.random()*H,s:Math.random()*2+0.5,b:Math.random()*0.7+0.3,tw:Math.random()*0.02+0.005}); } }
genStars();

let game = {
running: false, score: 0, level: 1, health: 6, maxHealth: 6,
asteroids: [], particles: [], shipX: 0, shipY: 0,
targetAsteroid: null, typedLen: 0, spawnTimer: 0, spawnInterval: 2000,
speedMult: 1, shakeTime: 0, shakeMag: 0
};

function pickWord(){
const lvl = game.level;
let pool;
if(lvl<=2) pool = words.short.concat(words.med);
else if(lvl<=4) pool = words.med.concat(words.long);
else if(lvl<=6) pool = words.long;
else pool = words.long.concat(words.xl);
return pool[Math.floor(Math.random()*pool.length)];
}

function spawnAsteroid(){
const word = pickWord();
const r = Math.max(28, 50 - game.level*3);
const x = r + Math.random()*(W-2*r);
const y = -r;
const speed = (0.4 + game.level*0.15) * game.speedMult * (H/700);
game.asteroids.push({x, y, r, word, speed, vx: (Math.random()-0.5)*0.3, rot: Math.random()*Math.PI*2, rotSpeed:(Math.random()-0.5)*0.01});
}

function spawnParticles(x,y,color,count){
for(let i=0;i<count;i++){
const a=Math.random()*Math.PI*2; const sp=Math.random()*4+1;
game.particles.push({x,y,vx:Math.cos(a)*sp,vy:Math.sin(a)*sp,life:1,color,r:Math.random()*4+1});
}
}

function resetGame(){
game.score=0; game.level=1; game.health=6; game.maxHealth=6;
game.asteroids=[]; game.particles=[]; game.typedLen=0; game.targetAsteroid=null;
game.spawnTimer=0; game.spawnInterval=2000; game.speedMult=1; game.shakeTime=0;
updateHUD();
}

function updateHUD(){
hudScore.textContent='SCORE: '+game.score;
hudLevel.textContent='LEVEL: '+game.level;
const bars = Math.max(0,Math.round(game.health/game.maxHealth*6));
hudHealth.textContent='HULL: '+'█'.repeat(bars)+'░'.repeat(6-bars);
}

function updateTargetDisplay(){
if(!game.targetAsteroid){ targetDisplay.innerHTML=''; return; }
const w = game.targetAsteroid.word;
const t = Math.min(game.typedLen, w.length);
const typed = w.substring(0,t);
const remaining = w.substring(t);
targetDisplay.innerHTML = '<span class="typed">'+typed+'</span><span class="remaining">'+remaining+'</span>';
}

function getTarget(){
// Closest to bottom (highest y)
let best=null, bestY=-Infinity;
for(const a of game.asteroids){ if(a.y>bestY){bestY=a.y;best=a;} }
return best;
}

function handleKey(e){
if(!game.running) return;
const key = e.key.toUpperCase();
if(key.length!==1 || !/[A-Z]/.test(key)) return;
e.preventDefault();
initAudio();

if(!game.targetAsteroid){ game.targetAsteroid = getTarget(); if(!game.targetAsteroid) return; }
const a = game.targetAsteroid;
const expected = a.word[game.typedLen];
if(key===expected){
game.typedLen++;
playLaser();
spawnParticles(game.shipX, game.shipY-20, '#4af', 3);
if(game.typedLen>=a.word.length){
// Destroy!
game.score += a.word.length*10*game.level;
playExplosion();
spawnParticles(a.x,a.y,'#f80',20);
spawnParticles(a.x,a.y,'#ff4',15);
const idx = game.asteroids.indexOf(a);
if(idx>=0) game.asteroids.splice(idx,1);
game.targetAsteroid=null; game.typedLen=0;
checkLevelUp();
}
}else{
// Wrong key - reset progress on this asteroid
game.typedLen=0;
game.shakeTime=8; game.shakeMag=3;
playDamage();
}
updateHUD(); updateTargetDisplay();
}

function checkLevelUp(){
const threshold = game.level * 100;
if(game.score >= (game.level)*200){
game.level++;
game.spawnInterval = Math.max(800, 2000 - game.level*150);
game.speedMult = 1 + game.level*0.1;
}
}

function update(dt){
if(!game.running) return;
game.shakeTime = Math.max(0, game.shakeTime-1);

// Spawn
game.spawnTimer += dt;
if(game.spawnTimer >= game.spawnInterval){
game.spawnTimer = 0;
spawnAsteroid();
}

// Update asteroids
for(let i=game.asteroids.length-1;i>=0;i--){
const a = game.asteroids[i];
a.y += a.speed * dt*0.06;
a.x += a.vx;
a.rot += a.rotSpeed;
if(a.x < a.r) { a.x=a.r; a.vx=Math.abs(a.vx); }
if(a.x > W-a.r){ a.x=W-a.r; a.vx=-Math.abs(a.vx); }
// Hit ship zone
if(a.y + a.r > H - 60){
game.asteroids.splice(i,1);
game.health--;
playDamage();
game.shakeTime=15; game.shakeMag=8;
spawnParticles(a.x,a.y,'#f44',25);
if(game.targetAsteroid===a){ game.targetAsteroid=null; game.typedLen=0; }
updateHUD(); updateTargetDisplay();
if(game.health<=0) gameOver();
}
}

// Auto-target: if no target, lock to nearest
if(!game.targetAsteroid && game.asteroids.length>0){
game.targetAsteroid = getTarget();
}
// If target is gone, re-lock
if(game.targetAsteroid && !game.asteroids.includes(game.targetAsteroid)){
game.typedLen=0; game.targetAsteroid=null;
if(game.asteroids.length>0) game.targetAsteroid=getTarget();
updateTargetDisplay();
}

// Particles
for(let i=game.particles.length-1;i>=0;i--){
const p=game.particles[i];
p.x+=p.vx; p.y+=p.vy; p.life-=0.03;
if(p.life<=0) game.particles.splice(i,1);
}

// Stars twinkle
for(const s of stars){ s.b += s.tw; if(s.b>1||s.b<0.2) s.tw=-s.tw; }
}

function draw(){
ctx.save();
if(game.shakeTime>0){
const sx=(Math.random()-0.5)*game.shakeMag;
const sy=(Math.random()-0.5)*game.shakeMag;
ctx.translate(sx,sy);
}

// Background
const grad = ctx.createLinearGradient(0,0,0,H);
grad.addColorStop(0,'#050510'); grad.addColorStop(1,'#0a0a2a');
ctx.fillStyle=grad; ctx.fillRect(-10,-10,W+20,H+20);

// Stars
for(const s of stars){
ctx.globalAlpha = 0.3+s.b*0.7;
ctx.fillStyle='#fff';
ctx.fillRect(s.x,s.y,s.s,s.s);
}
ctx.globalAlpha=1;

// Asteroids
for(const a of game.asteroids){
const isTarget = (a===game.targetAsteroid);
ctx.save();
ctx.translate(a.x,a.y);
ctx.rotate(a.rot);

// Body
ctx.beginPath();
const sides=7;
for(let i=0;i<sides;i++){
const ang=i/sides*Math.PI*2;
const rr=a.r*(0.85+Math.sin(i*2.3)*0.15);
if(i===0) ctx.moveTo(Math.cos(ang)*rr,Math.sin(ang)*rr);
else ctx.lineTo(Math.cos(ang)*rr,Math.sin(ang)*rr);
}
ctx.closePath();
const ag = ctx.createRadialGradient(0,0,a.r*0.2,0,0,a.r);
ag.addColorStop(0,isTarget?'#6af':'#876');
ag.addColorStop(1,'#333');
ctx.fillStyle=ag; ctx.fill();
ctx.strokeStyle=isTarget?'#4af':'#555'; ctx.lineWidth=2; ctx.stroke();

// Craters
ctx.fillStyle='rgba(0,0,0,0.3)';
ctx.beginPath(); ctx.arc(a.r*0.2,-a.r*0.1,a.r*0.15,0,Math.PI*2); ctx.fill();
ctx.beginPath(); ctx.arc(-a.r*0.2,a.r*0.2,a.r*0.12,0,Math.PI*2); ctx.fill();

// Word
ctx.rotate(-a.rot);
ctx.font='bold '+Math.max(12,Math.floor(a.r*0.45))+'px Courier New';
ctx.textAlign='center'; ctx.textBaseline='middle';
const w=a.word; const t=Math.min(game.typedLen,w.length);
if(isTarget){
for(let i=0;i<w.length;i++){
ctx.fillStyle=i<t?'#4f8':'#fff';
const cw = ctx.measureText(w[i]).width;
// Draw letter by letter centered
}
// Simpler: draw whole word with color split
const totalW = ctx.measureText(w).width;
let cx = -totalW/2;
for(let i=0;i<w.length;i++){
ctx.fillStyle=i<t?'#4f8':'#fff';
ctx.fillText(w[i],cx+ctx.measureText(w[i]).width/2, 0);
cx+=ctx.measureText(w[i]).width;
}
}else{
ctx.fillStyle='#ddd';
ctx.fillText(w,0,0);
}

ctx.restore();
}

// Target glow ring
if(game.targetAsteroid){
const ta=game.targetAsteroid;
ctx.strokeStyle='rgba(100,180,255,0.5)';
ctx.lineWidth=3;
ctx.beginPath(); ctx.arc(ta.x,ta.y,ta.r+6,0,Math.PI*2); ctx.stroke();
}

// Ship
const sx=W/2, sy=H-40;
game.shipX=sx; game.shipY=sy;
ctx.save();
ctx.translate(sx,sy);
// Engine glow
const eg=ctx.createRadialGradient(0,15,2,0,15,15);
eg.addColorStop(0,'rgba(100,180,255,0.8)'); eg.addColorStop(1,'rgba(100,180,255,0)');
ctx.fillStyle=eg;
ctx.beginPath(); ctx.arc(0,15,15,0,Math.PI*2); ctx.fill();
// Body
ctx.fillStyle='#4af';
ctx.beginPath(); ctx.moveTo(0,-20); ctx.lineTo(-18,15); ctx.lineTo(-8,10); ctx.lineTo(0,15); ctx.lineTo(8,10); ctx.lineTo(18,15); ctx.closePath(); ctx.fill();
ctx.strokeStyle='#8df'; ctx.lineWidth=1.5; ctx.stroke();
// Cockpit
ctx.fillStyle='#aef';
ctx.beginPath(); ctx.arc(0,-5,5,0,Math.PI*2); ctx.fill();

if(game.targetAsteroid && game.typedLen>0){

for(const p of game.particles){
ctx.globalAlpha=p.life;

}

// Danger line
ctx.strokeStyle='rgba(255,60,60,0.2)';
ctx.lineWidth=1;
ctx.setLineDash([8,8]);
ctx.beginPath(); ctx.moveTo(0,H-60); ctx.lineTo(W,H-60); ctx.stroke();

}

let lastTime=0;
function loop(t){
const dt = Math.min(t-lastTime,50); lastTime=t;
update(dt); draw();
requestAnimationFrame(loop);
}

function gameOver(){
