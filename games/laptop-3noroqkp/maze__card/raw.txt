// Maze generation (recursive backtracker)
function generateMaze(size) {
    let grid = [];
    for (let y = 0; y < size; y++) {
        grid[y] = [];
        for (let x = 0; x < size; x++) {
            grid[y][x] = { walls: [true, true, true, true], visited: false }; // top, right, bottom, left
        }
    }
    
    let stack = [];
    let cx = 0, cy = 0;
    grid[cy][cx].visited = true;
    stack.push({ x: cx, y: cy });
    
    while (stack.length > 0) {
        let current = stack[stack.length - 1];
        let neighbors = [];
        let dirs = [
            { dx: 0, dy: -1, wall: 0, opp: 2 }, // up
            { dx: 1, dy: 0, wall: 1, opp: 3 },  // right
            { dx: 0, dy: 1, wall: 2, opp: 0 }, // down
            { dx: -1, dy: 0, wall: 3, opp: 1 } // left
        ];
        
        for (let d of dirs) {
            let nx = current.x + d.dx;
            let ny = current.y + d.dy;
            if (nx >= 0 && nx < size && ny >= 0 && ny < size && !grid[ny][nx].visited) {
                neighbors.push({ ...d, nx, ny });
            }
        }
        
        if (neighbors.length === 0) {
            stack.pop();
        } else {
            let n = neighbors[Math.floor(Math.random() * neighbors.length)];
            grid[current.y][current.x].walls[n.wall] = false;
            grid[n.ny][n.nx].walls[n.opp] = false;
            grid[n.ny][n.nx].visited = true;
            stack.push({ x: n.nx, y: n.ny });
        }
    }
    return grid;
}

function generateDeck() {
    let deck = [];
    for (let i = 0; i < 8; i++) deck.push('UP');
    for (let i = 0; i < 8; i++) deck.push('DOWN');
    for (let i = 0;  i < 8; i++) deck.push('LEFT');
    for (let i = 0; i < 8; i++) deck.push('RIGHT');
    for (let i = 0; i < 4; i++) deck.push('WILD');
    for (let i = 0; i < 3; i++) deck.push('BREAK');
    for (let i = 0; i < 2; i++) deck.push('TELEPORT');
    // Shuffle
    for (let i = deck.length - 1; i > 0; i--) {
        let j = Math.floor(Math.random() * (i + 1));
        [deck[i], deck[j]] = [deck[j], deck[i]];
    }
    return deck;
}

    if (deck.length === 0) {

                [deck[i], deck[j]] = [deck[j], deck[i]];

}

function canMove(x, y, dx, dy) {
    let wallIdx;
    if (dy === -1) wallIdx = 0;
    else if (dx === 1) wallIdx = 1;
    else if (dy === 1) wallIdx = 2;
    else if (dx === -1) wallIdx = 3;
    else return false;
    
    let nx = x + dx, ny = y + dy;
    if (nx < 0 || nx >= MAZE_SIZE || ny < 0 || ny >= MAZE_SIZE) return false;
    return !maze[y][x].walls[wallIdx];
}

            vy: (Math.random() - 0.5) * 4,

            color: color
        });
    }
}

function pickBestDirection() {
    let dirs = [{dx:0,dy:-1},{dx:1,dy:0},{dx:0,dy:1},{dx:-1,dy:0}];
    let best = null;
    let bestScore = -999;
    
    for (let d of dirs) {
        if (!canMove(playerPos.x, playerPos.y, d.dx, d.dy)) continue;
        let nx = playerPos.x + d.dx;
        let ny = playerPos.y + d.dy;
        let dist = Math.abs(nx - exitPos.x) + Math.abs(ny - exitPos.y);
        // Count open directions from new position
        let openDirs = 0;
        for (let dd of dirs) {
            if (canMove(nx, ny, dd.dx, dd.dy)) openDirs++;
        }
        let s = -dist + openDirs * 0.5;
        if (s > bestScore) {
            bestScore = s;
            best = d;
        }
    }
    return best;
}

    let cardH = cardAreaH * 0.8;
    let totalCardW = hand.length * (cardW + 10) - 10;
    let startX = (W - totalCardW) / 2;
    let startY = H - cardAreaH - 5;
    
    for (let i = 0; i < hand.length; i++) {
        let cx = startX + i * (cardW + 10);
        let cy = startY;
        let isSelected = (i === selectedCard) || (breakTarget === i);
        
        // Card background
        ctx.fillStyle = isSelected ? 'rgba(99,102,241,0.3)' : 'rgba(55,65,81,0.8)';
        ctx.strokeStyle = isSelected ? '#818cf8' : '#4c5ca8';
        ctx.lineWidth = 2;
        roundRect(ctx, cx, cy, cardW, cardH, 8);
        ctx.fill();
        ctx.stroke();
        
        // Card content
        let type = hand[i];
        let ct = CARD_TYPES[type];
        ctx.fillStyle = ct.color;
        ctx.font = `bold ${cardH * 0.4}px sans-serif`;
        ctx.textAlign = 'center';
        ctx.textBaseline = 'middle';
        ctx.fillText(ct.symbol, cx + cardW / 2, cy + cardH * 0.35);
        
        ctx.fillStyle = '#e2e8f0';
        ctx.font = `${cardH * 0.18}px sans-serif`;
        ctx.fillText(ct.name, cx + cardW / 2, cy + cardH * 0.7);
        
        // Number label
        ctx.fillStyle = '#6b7280';
        ctx.font = `${cardH * 0.15}px sans-serif`;
        ctx.fillText(`${i + 1}`, cx + cardW / 2, cy + cardH * 0.9);
    }
    
    // End screens
    if (gameState === 'won' || gameState === 'lost') {
        ctx.fillStyle = 'rgba(26,26,46,0.85)';
        ctx.fillRect(0, 0, W, H);
        
        ctx.textAlign = 'center';
        ctx.textBaseline = 'middle';
        
        if (gameState === 'won') {
            ctx.fillStyle = '#4ade80';
            ctx.font = `bold ${W * 0.06}px sans-serif`;
            ctx.fillText('🎉 MAZE SOLVED!', W / 2, H * 0.35);
            ctx.fillStyle = '#e2e8f0';
            ctx.font = `${W * 0.03}px sans-serif`;
            ctx.fillText(`Score: ${score}`, W / 2, H * 0.45);
            ctx.fillText(`Turns remaining: ${turnsLeft}`, W / 2, H * 0.52);
        } else {
            ctx.fillStyle = '#f87171';
            ctx.font = `bold ${W * 0.06}px sans-serif`;
            ctx.fillText('💀 OUT OF TURNS', W / 2, H * 0.35);
            ctx.fillStyle = '#e2e8f0';
            ctx.font = `${W * 0.03}px sans-serif`;
            ctx.fillText('The maze claims another soul...', W / 2, H * 0.45);
        }
        
        ctx.fillStyle = '#a5b4fc';
        ctx.font = `${W * 0.025}px sans-serif`;
        ctx.fillText('Press SPACE or R to play again', W / 2, H * 0.6);
    }
    
    animFrame++;
}

function roundRect(ctx, x, y, w, h, r) {
    ctx.beginPath();
    ctx.moveTo(x + r, y);
    ctx.lineTo(x + w - r, y);
    ctx.quadraticCurveTo(x + w, y, x + w, y + r);
    ctx.lineTo(x + w, y + h - r);
    ctx.quadraticCurveTo(x + w, y + h, x + w - r, y + h);
    ctx.lineTo(x + r, y + h);
    ctx.quadraticCurveTo(x, y + h, x, y + h - r);
    ctx.lineTo(x, y + r);
    ctx.quadraticCurveTo(x, y, x + r, y);
    ctx.closePath();
}

// Input handling
function getCardAtPoint(mx, my) {
    let cardAreaH = H * 0.22;
    let cardW = Math.min(70, W * 0.1);
    let cardH = cardAreaH * 0.8;
    let totalCardW = hand.length * (cardW + 10) - 10;
    let startX = (W - totalCardW) / 2;
    let startY = H - cardAreaH - 5;
    
    for (let i = 0; i < hand.length; i++) {
        let cx = startX + i * (cardW + 10);
        if (mx >= cx && mx <= cx + cardW && my >= startY && my <= startY + cardH) {
            return i;
        }
    }
    return -1;
}

canvas.addEventListener('click', (e) => {
    initAudio();
    let rect = canvas.getBoundingClientRect();
    let mx = e.clientX - rect.left;

window.addEventListener('keydown', (e) => {
    initAudio();
    
    if (gameState === 'start') {
        if (e.code === 'Space' || e.code === 'Enter') {
            document.getElementById('startScreen').style.display = 'none';
            gameState = 'playing';
        }
        return;
    }
    
    if (gameState === 'won' || gameState === 'lost') {
        if (e.code === 'Space' || e.code === 'KeyR') {
            initGame();
            gameState = 'playing';
        }
        return;
    }
    
    if (breakTarget !== null) {
        // Direction selection for break
        let dirs = {
            'KeyW': { dx: 0, dy: -1, wall: 0 },
            'ArrowUp': { dx: 0, dy: -1, wall: 0 },
            'KeyD': { dx: 1, dy: 0, wall: 1 },
            'ArrowRight': { dx: 1, dy: 0, wall: 1 },
            'KeyS': { dx: 0, dy: 1, wall: 2 },
            'ArrowDown': { dx: 0, dy: 1, wall: 2 },
            'KeyA': { dx: -1, dy: 0, wall: 3 },
            'ArrowLeft': { dx: -1, dy: 0, wall: 3 }
        };
        
        if (dirs[e.code]) {
            let d = dirs[e.code];
            let nx = playerPos.x + d.dx;
            let ny = playerPos.y + d.dy;
            if (nx >= 0 && nx < MAZE_SIZE && ny >= 0 && ny < MAZE_SIZE && maze[playerPos.y][playerPos.x].walls[d.wall]) {
                maze[playerPos.y][playerPos.x].walls[d.wall] = false;
                // Also break the opposite wall of neighbor
                let oppWall = (d.wall + 2) % 4;
                maze[ny][nx].walls[oppWall] = false;
                addParticles(playerPos.x, playerPos.y, '#fb923c', 10);
                playSound('break');
                
                // Remove card and proceed
                let idx = breakTarget;
                discardPile.push(hand[idx]);
                hand.splice(idx, 1);
                let newCard = drawCard();
                if (newCard) hand.push(newCard);
                
                turnsLeft--;
                if (turnsLeft <= 0) {
                    gameState = 'lost';
                    playSound('lose');
                }
            } else {
                message = 'No wall there!';
                messageTimer = 40;
                playSound('error');
            }
            breakTarget = null;
        }
        return;
    }
    
    // Number keys to play cards
    let num = parseInt(e.key);
    if (num >= 1 && num <= 5 && num <= hand.length) {
        playCard(num - 1);
    }
});
