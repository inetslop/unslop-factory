<script>
const canvas = document.getElementById('game');
const ctx = canvas.getContext('2d');

let W, H;
function resize() {
    W = window.innerWidth;
    H = window.innerHeight;
    canvas.width = W;
    canvas.height = H;
}
window.addEventListener('resize', resize);
resize();

// Audio
let audioCtx = null;
function initAudio() { if (!audioCtx) audioCtx = new (window.AudioContext || webkitAudioContext)(); }
function playSound(freq, type, dur, vol) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = type;
    osc.frequency.value = freq;
    gain.gain.setValueAtTime(vol || 0.15, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
    osc.connect(gain); gain.connect(audioCtx.destination);
    osc.start(); osc.stop(audioCtx.currentTime + dur);
}
function sfxShoot() { playSound(800, 'square', 0.05, 0.08); }
function sfxExplode() { playSound(150, 'sawtooth', 0.3, 0.2); }
function sfxPlace() { playSound(400, 'triangle', 0.1, 0.15); }
function sfxWave() { playSound(200, 'square', 0.5, 0.1); }

// Game State
let state = 'START'; // START, PLAYING, GAMEOVER
let score = 0;
let wave = 0;
let baseHP = 100;
let maxBaseHP = 100;
let enemies = [];
let projectiles = [];
let particles = [];
let towers = [];
let towerSlots = [];
let selectedTowerType = 0;
let frameCount = 0;
let shakeTimer = 0;
let waveDelay = 0;
let stars = [];

// Tower Types
const TOWER_TYPES = [
    { name: 'Cannon', color: '#4fc3f7', dmg: 20, rate: 60, projSpeed: 8, pierce: false, icon: '●' },
    { name: 'Rapid', color: '#ff9800', dmg: 8, rate: 15, projSpeed: 12, pierce: false, icon: '≡' },
    { name: 'Laser', color: '#e040fb', dmg: 35, rate: 90, projSpeed: 6, pierce: true, icon: '◆' }
];

function initStars() {
    stars = [];
    for (let i = 0; i < 80; i++) {
        stars.push({ x: Math.random() * W, y: Math.random() * H, s: Math.random() * 2 + 0.5, speed: Math.random() * 0.3 + 0.1 });
    }
}

function initSlots() {
    towerSlots = [];
    const slotCount = 5;
    const margin = W * 0.12;
    for (let i = 0; i < slotCount; i++) {
        const x = margin + (W - 2 * margin) * (i / (slotCount - 1));
        towerSlots.push({ x: x, y: H - 50, occupied: false, tower: null });
    }
}

function spawnWave() {
    wave++;
    sfxWave();
    const cols = Math.min(6 + wave, 10);
    const rows = Math.min(3 + Math.floor(wave / 2), 6);
    const spacingX = Math.max(W * 0.06, 50);
    const spacingY = Math.max(H * 0.07, 40);
    const startX = W - 80;
    const startY = H * 0.15;
    const hp = 20 + wave * 5;
    const speed = (0.3 + wave * 0.05) * (W / 800);

    for (let r = 0; r < rows; r++) {
        for (let c = 0; c < cols; c++) {
            enemies.push({
                x: startX + c * spacingX,
                y: startY + r * spacingY,
                baseX: startX + c * spacingX,
                baseY: startY + r * spacingY,
                hp: hp, maxHp: hp,
                row: r, col: c,
                type: (r % 3 === 0) ? 2 : (r % 3 === 1) ? 1 : 0,
                speed: speed,
                alive: true,
                flash: 0
            });
        }
    }
}

function resetGame() {
    score = 0; wave = 0; baseHP = maxBaseHP;
    enemies = []; projectiles = []; particles = []; towers = [];
    frameCount = 0; shakeTimer = 0; waveDelay = 60;
    initSlots();
    state = 'PLAYING';
}

function drawEnemy(e) {
    const s = 12 + e.type * 3;
    ctx.fillStyle = e.flash > 0 ? '#fff' : getEnemyColor(e.type);
    // Classic invader pixel art (simplified)
    const px = e.x, py = e.y;
    if (e.type === 0) {
        // Small crab
        ctx.fillRect(px - 4, py - 8, 8, 4);
        ctx.fillRect(px - 8, py - 4, 16, 4);
        ctx.fillRect(px - 12, py, 24, 4);
        ctx.fillRect(px - 8, py + 4, 4, 4);
        ctx.fillRect(px + 4, py + 4, 4, 4);
    } else if (e.type === 1) {
        // Medium squid
        ctx.fillRect(px - 6, py - 8, 12, 4);
        ctx.fillRect(px - 10, py - 4, 20, 4);
        ctx.fillRect(px - 14, py, 28, 4);
        ctx.fillRect(px - 10, py + 4, 4, 4);
        ctx.fillRect(px + 6, py + 4, 4, 4);
    } else {
        // Big octopus
        ctx.fillRect(px - 8, py - 12, 16, 4);
        ctx.fillRect(px - 12, py - 8, 24, 8);
        ctx.fillRect(px - 16, py, 32, 8);
        ctx.fillRect(px - 12, py + 8, 8, 4);
        ctx.fillRect(px + 4, py + 8, 8, 4);
    }
    // HP bar
    if (e.hp < e.maxHp) {
        const bw = 20;
        ctx.fillStyle = '#333';
        ctx.fillRect(px - bw/2, py - 16, bw, 3);
        ctx.fillStyle = '#f44360';
        ctx.fillRect(px - bw/2, py - 16, bw * (e.hp / e.maxHp), 3);
    }
}

function getEnemyColor(type) {
    const colors = ['#76ff03', '#ff9800', '#f44360'];
    return colors[type];
}

function drawTower(t) {
    const x = t.x, y = t.y;
    const type = TOWER_TYPES[t.type];
    ctx.fillStyle = type.color;
    // Base
    ctx.fillRect(x - 18, y - 5, 36, 20);
    // Barrel
    if (t.angle !== undefined) {
        ctx.save();
        ctx.translate(x, y);
        ctx.rotate(t.angle || 0);
        ctx.fillRect(0, -3, 24, 6);
        ctx.restore();
    } else {
        ctx.fillRect(x - 4, y - 25, 8, 20);
    }
    // Glow
    ctx.shadowColor = type.color;
    ctx.shadowBlur = 10;
    ctx.fillRect(x - 14, y - 3, 28, 4);
    ctx.shadowBlur = 0;
}

function drawProjectile(p) {
    ctx.fillStyle = p.color;
    ctx.shadowColor = p.color;
    ctx.shadowBlur = 6;
    if (p.pierce) {
        ctx.fillRect(p.x - 2, p.y - 8, 4, 16);
    } else {
        ctx.beginPath();
        ctx.arc(p.x, p.y, 3, 0, Math.PI * 2);
        ctx.fill();
    }
    ctx.shadowBlur = 0;
}

function spawnParticles(x, y, color, count) {
    for (let i = 0; i < count; i++) {
        const angle = Math.random() * Math.PI * 2;
        const speed = Math.random() * 3 + 1;
        particles.push({
            x: x, y: y,
            vx: Math.cos(angle) * speed,
            vy: Math.sin(angle) * speed,
            life: 30 + Math.random() * 20,
            color: color,
            size: Math.random() * 4 + 1
        });
    }
}

function update() {
    frameCount++;

    // Stars
    for (let s of stars) {
        s.x -= s.speed;
        if (s.x < 0) { s.x = W; s.y = Math.random() * H; }
    }

    if (state !== 'PLAYING') return;

    // Wave spawning
    if (enemies.length === 0 && waveDelay > 0) {
        waveDelay--;
        if (waveDelay <= 0) spawnWave();
    } else if (enemies.length === 0 && waveDelay <= 0) {
        waveDelay = 90;
    }

    // Enemy movement
    const moveSpeed = enemies[0] ? enemies[0].speed : 0.5;
    let needStepDown = false;
    for (let e of enemies) {
        if (!e.alive) continue;
        e.x -= moveSpeed;
        if (e.flash > 0) e.flash--;
        // Check if needs to step down
        if (e.row >= 5 && frameCount % 120 === 0) needStepDown = true;
    }

    // Enemies reaching base
    for (let e of enemies) {
        if (!e.alive) continue;
        if (e.x < W * 0.08) {
            e.alive = false;
            baseHP -= 15;
            shakeTimer = 10;
            spawnParticles(e.x, e.y, '#f44360', 15);
            sfxExplode();
            if (baseHP <= 0) {
                baseHP = 0;
                state = 'GAMEOVER';
            }
        }
    }

    // Tower shooting
    for (let t of towers) {
        t.cooldown--;
        if (t.cooldown > 0) continue;
        // Find target
        let target = null, minDist = Infinity;
        for (let e of enemies) {
            if (!e.alive) continue;
            const dx = e.x - t.x, dy = e.y - t.y;
            const dist = Math.sqrt(dx * dx + dy * dy);
            if (dist < minDist && e.x > W * 0.08) {
                minDist = dist;
                target = e;
            }
        }
        if (target) {
            t.cooldown = TOWER_TYPES[t.type].rate;
            const angle = Math.atan2(target.y - t.y, target.x - t.x);
            t.angle = angle;
            const type = TOWER_TYPES[t.type];
            projectiles.push({
                x: t.x + Math.cos(angle) * 20,
                y: t.y + Math.sin(angle) * 20,
                vx: Math.cos(angle) * type.projSpeed,
                vy: Math.sin(angle) * type.projSpeed,
                dmg: type.dmg,
                pierce: type.pierce,
                color: type.color,
                life: 60
            });
            sfxShoot();
        }
    }

    // Projectiles
    for (let p of projectiles) {
        p.x += p.vx;
        p.y += p.vy;
        p.life--;
        if (p.life <= 0 || p.x < 0 || p.x > W || p.y < 0 || p.y > H) {
            p.dead = true;
            continue;
        }
        // Hit detection
        for (let e of enemies) {
            if (!e.alive) continue;
            const dx = p.x - e.x, dy = p.y - e.y;
            if (Math.abs(dx) < 15 && Math.abs(dy) < 15) {
                e.hp -= p.dmg;
                e.flash = 4;
                if (!p.pierce) p.dead = true;
                else p.life = Math.min(p.life, 20);
                if (e.hp <= 0) {
                    e.alive = false;
                    score += 10 * (e.type + 1);
                    spawnParticles(e.x, e.y, getEnemyColor(e.type), 12);
                    sfxExplode();
                }
                break;
            }
        }
    }

    projectiles = projectiles.filter(p => !p.dead && p.life > 0);
    enemies = enemies.filter(e => e.alive);

    // Particles
    for (let p of particles) {
        p.x += p.vx;
        p.y += p.vy;
        p.vy += 0.1;
        p.life--;
    }
    particles = particles.filter(p => p.life > 0);

    // Shake
    if (shakeTimer > 0) shakeTimer--;
}

function draw() {
    ctx.save();
    // Screen shake
    if (shakeTimer > 0) {
        const sx = (Math.random() - 0.5) * shakeTimer;
        const sy = (Math.random() - 0.5) * shakeTimer;
        ctx.translate(sx, sy);
    }

    // Background
    ctx.fillStyle = '#0a0a1a';
    ctx.fillRect(0, 0, W, H);

    // Stars
    ctx.fillStyle = '#fff';
    for (let s of stars) {
        ctx.globalAlpha = 0.3 + s.s * 0.2;
        ctx.fillRect(s.x, s.y, s.s, s.s);
    }
    ctx.globalAlpha = 1;

    if (state === 'START') {
        drawStartScreen();
    } else if (state === 'PLAYING' || state === 'GAMEOVER') {
        drawGame();
    }

    ctx.restore();
}

function drawStartScreen() {
    // Title
    ctx.fillStyle = '#76ff03';
    ctx.font = `bold ${Math.min(W * 0.05, 48)}px monospace`;
    ctx.textAlign = 'center';
    ctx.fillText('SPACE INVADERS', W / 2, H * 0.2);
    ctx.fillStyle = '#ff9800';
    ctx.font = `bold ${Math.min(W * 0.04, 36)}px monospace`;
    ctx.fillText('TOWER DEFENSE', W / 2, H * 0.27);

    // Instructions
    ctx.fillStyle = '#ccc';
    ctx.font = `${Math.min(W * 0.02, 18)}px monospace`;
    const lines = [
        'Place towers to defend your base from invading aliens!',
        '',
        '[1] Select Cannon  [2] Select Rapid  [3] Select Laser',
        '[Click] Place tower on a slot',
        '[R] Restart / New Game',
        '',
        'Enemies march toward your base from the right.',
        'If they reach the left edge, your base takes damage!',
        '',
        'Tower Types:',
        '  ● Cannon - Balanced damage & rate',
        '  ≡ Rapid - Low damage, very fast fire',
        '  ◆ Laser - High damage, slow, pierces through'
    ];
    lines.forEach((line, i) => {
        ctx.fillText(line, W / 2, H * 0.38 + i * Math.min(H * 0.035, 24));
    });

    // Blinking prompt
    if (Math.floor(frameCount / 30) % 2 === 0) {
        ctx.fillStyle = '#fff';
        ctx.font = `bold ${Math.min(W * 0.025, 22)}px monospace`;
        ctx.fillText('[ SPACE ] or [ ENTER ] to Start', W / 2, H * 0.85);
    }

    // Decorative invaders
    const t = frameCount * 0.02;
    for (let i = 0; i < 3; i++) {
        const ex = W * 0.3 + i * W * 0.2;
        const ey = H * 0.65 + Math.sin(t + i) * 10;
        ctx.save();
        ctx.translate(ex, ey);
        const fake = { x: 0, y: 0, type: i, flash: 0 };
        drawEnemy(fake);
        ctx.restore();
    }
}

function drawGame() {
    // Base zone indicator
    ctx.fillStyle = 'rgba(244, 67, 96, 0.1)';
    ctx.fillRect(0, 0, W * 0.08, H);
    ctx.strokeStyle = '#f44360';
    ctx.lineWidth = 2;
    ctx.strokeRect(0, 0, W * 0.08, H);

    // Tower slots
    for (let s of towerSlots) {
        if (!s.occupied) {
            ctx.strokeStyle = 'rgba(100,200,255,0.4)';
            ctx.lineWidth = 2;
            ctx.setLineDash([5, 5]);
            ctx.strokeRect(s.x - 20, s.y - 25, 40, 30);
            ctx.setLineDash([]);
            ctx.fillStyle = 'rgba(100,200,255,0.3)';
            ctx.font = '12px monospace';
            ctx.textAlign = 'center';
            ctx.fillText('T', s.x, s.y - 5);
        }
    }

    // Towers
    for (let t of towers) drawTower(t);

    // Enemies
    for (let e of enemies) if (e.alive) drawEnemy(e);

    // Projectiles
    for (let p of projectiles) drawProjectile(p);

    // Particles
    for (let p of particles) {
        ctx.globalAlpha = p.life / 50;
        ctx.fillStyle = p.color;
        ctx.fillRect(p.x - p.size/2, p.y - p.size/2, p.size, p.size);
    }
    ctx.globalAlpha = 1;

    // HUD
    ctx.textAlign = 'left';
    ctx.font = `bold ${Math.min(W * 0.02, 18)}px monospace`;
    ctx.fillStyle = '#fff';
    ctx.fillText(`WAVE: ${wave}`, 20, 30);
    ctx.fillText(`SCORE: ${score}`, 20, 55);

    // HP Bar
    const barW = W * 0.25;
    ctx.fillStyle = '#333';
    ctx.fillRect(W - barW - 40, 15, barW, 20);
    ctx.fillStyle = baseHP > 50 ? '#76ff03' : baseHP > 25 ? '#ff9800' : '#f44360';
    ctx.fillRect(W - barW - 40, 15, barW * (baseHP / maxBaseHP), 20);
    ctx.strokeStyle = '#fff';
    ctx.strokeRect(W - barW - 40, 15, barW, 20);
    ctx.fillStyle = '#fff';
    ctx.fillText(`BASE: ${Math.ceil(baseHP)}%`, W - barW - 35, 30);

    // Tower selector
    ctx.textAlign = 'center';
    const selX = W / 2;
    ctx.font = `14px monospace`;
    for (let i = 0; i < 3; i++) {
        const tx = selX + (i - 1) * 140;
        const ty = H - 20;
        const tType = TOWER_TYPES[i];
        ctx.fillStyle = selectedTowerType === i ? tType.color : '#555';
        ctx.fillText(`[${i+1}] ${tType.name}`, tx, ty);
    }

    // Game Over overlay
    if (state === 'GAMEOVER') {
        ctx.fillStyle = 'rgba(0,0,0,0.7)';
        ctx.fillRect(0, 0, W, H);
        ctx.textAlign = 'center';
        ctx.fillStyle = '#f44360';
        ctx.font = `bold ${Math.min(W * 0.05, 48)}px monospace`;
        ctx.fillText('BASE DESTROYED', W / 2, H * 0.35);
        ctx.fillStyle = '#fff';
        ctx.font = `${Math.min(W * 0.03, 24)}px monospace`;
        ctx.fillText(`Final Score: ${score}  |  Waves Survived: ${wave}`, W / 2, H * 0.45);
        if (Math.floor(frameCount / 30) % 2 === 0) {
            ctx.fillStyle = '#76ff03';
            ctx.fillText('[ R ] to Restart', W / 2, H * 0.6);
        }
    }
}

// Input
window.addEventListener('keydown', (e) => {
    initAudio();
    if (state === 'START' && (e.code === 'Space' || e.code === 'Enter')) {
        resetGame();
        return;
    }
    if (state === 'GAMEOVER' && e.code === 'KeyR') {
        resetGame();
        return;
    }
    if (state !== 'PLAYING') return;

    if (e.code === 'Digit1') selectedTowerType = 0;
    if (e.code === 'Digit2') selectedTowerType = 1;
    if (e.code === 'Digit3') selectedTowerType = 2;
    if (e.code === 'KeyR') resetGame();
});

canvas.addEventListener('click', (e) => {
    initAudio();
    if (state !== 'PLAYING') return;

    const rect = canvas.getBoundingClientRect();
    const mx = e.clientX - rect.left;
    const my = e.clientY - rect.top;

    // Check tower slots
    for (let s of towerSlots) {
        const dx = mx - s.x, dy = my - s.y;
        if (Math.abs(dx) < 30 && Math.abs(dy) < 40) {
            if (!s.occupied) {
                s.occupied = true;
                const t = { x: s.x, y: s.y, type: selectedTowerType, cooldown: 0, angle: -Math.PI / 2 };
                towers.push(t);
                s.tower = t;
                sfxPlace();
            }
            return;
        }
    }
});

// Game Loop
function loop() {
    update();
    draw();
    requestAnimationFrame(loop);
}

initStars();
initSlots();
loop();
</script>
</body>
