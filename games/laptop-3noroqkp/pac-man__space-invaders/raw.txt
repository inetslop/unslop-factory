<!DOCTYPE html>
<html lang="en">
<head>

    <meta name="viewport" content="width=device-width, initial-scale=1.0">

    <style>

<script>
(function() {
    const canvas = document.getElementById('game');
    const ctx = canvas.getContext('2d');

    let W, H, scale;
    function resize() {
        W = window.innerWidth;
        H = window.innerHeight;
        canvas.width = W;
        canvas.height = H;
        scale = Math.min(W, H) / 600;
    }
    window.addEventListener('resize', resize);
    resize();

    // Audio
    let audioCtx = null;
    function initAudio() {
        if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
    }
    function playSound(type) {
        if (!audioCtx) return;
        const osc = audioCtx.createOscillator();
        const gain = audioCtx.createGain();
        osc.connect(gain);
        gain.connect(audioCtx.destination);
        const now = audioCtx.currentTime;
        switch(type) {
            case 'shoot':
                osc.type = 'square';
                osc.frequency.setValueAtTime(800, now);
                osc.frequency.exponentialRampToValueAtTime(200, now + 0.1);
                gain.gain.setValueAtTime(0.15, now);
                gain.gain.exponentialRampToValueAtTime(0.001, now + 0.1);
                osc.start(now);
                osc.stop(now + 0.1);
                break;
            case 'explosion':
                osc.type = 'sawtooth';
                osc.frequency.setValueAtTime(150, now);
                osc.frequency.exponentialRampToValueAtTime(30, now + 0.3);
                gain.gain.setValueAtTime(0.2, now);
                gain.gain.exponentialRampToValueAtTime(0.001, now + 0.3);
                osc.start(now);
                osc.stop(now + 0.3);
                break;
            case 'waka':
                osc.type = 'square';
                osc.frequency.setValueAtTime(500, now);
                osc.frequency.exponentialRampToValueAtTime(1000, now + 0.08);
                gain.gain.setValueAtTime(0.12, now);
                gain.gain.exponentialRampToValueAtTime(0.001, now + 0.1);
                osc.start(now);
                osc.stop(now + 0.1);
                break;
            case 'powerup':
                osc.type = 'sine';
                osc.frequency.setValueAtTime(400, now);
                osc.frequency.linearRampToValueAtTime(800, now + 0.15);
                gain.gain.setValueAtTime(0.15, now);
                gain.gain.exponentialRampToValueAtTime(0.001, now + 0.2);
                osc.start(now);
                osc.stop(now + 0.2);
                break;
            case 'hit':
                osc.type = 'sawtooth';
                osc.frequency.setValueAtTime(80, now);
                gain.gain.setValueAtTime(0.3, now);
                gain.gain.exponentialRampToValueAtTime(0.001, now + 0.2);
                osc.start(now);
                osc.stop(now + 0.2);
                break;
        }
    }

    // Game state
    let gameState = 'start'; // start, playing, gameover
    let score = 0;
    let lives = 3;
    let highScore = 0;
    try { highScore = parseInt(localStorage.getItem('pacSpaceHigh') || '0'); } catch(e) {}
    let frame = 0;
    let level = 1;

    // Pac-Man
    let pacman = { x: 0, y: 0, w: 30, h: 30, mouthAngle: 0, speed: 4, invincible: 0 };

    // Bullets (Pac-Man's shots)
    let bullets = [];
    let bulletCooldown = 0;

    // Aliens
    let aliens = [];
    let alienDirection = 1;
    let alienSpeed = 1.5;
    let alienMoveTimer = 0;
    let alienShootTimer = 0;

    // Enemy bullets
    let enemyBullets = [];

    // Power-ups (dots)
    let powerUps = [];
    let powerUpTimer = 0;

    // Particles
    let particles = [];

    // Stars background
    let stars = [];
    function initStars() {
        stars = [];
        for (let i = 0; i < 80; i++) {
            stars.push({ x: Math.random() * W, y: Math.random() * H, size: Math.random() * 2 + 0.5, speed: Math.random() * 0.3 + 0.1, twinkle: Math.random() * Math.PI * 2 });
        }
    }

    function initAliens() {
        aliens = [];
        const cols = 10;
        const rows = 5;
        const spacingX = 45 * scale;
        const spacingY = 35 * scale;
        const startX = (W - cols * spacingX) / 2 + spacingX / 2;
        const startY = 80 * scale;
        for (let r = 0; r < rows; r++) {
            for (let c = 0; c < cols; c++) {
                aliens.push({
                    x: startX + c * spacingX,
                    y: startY + r * spacingY,
                    w: 28 * scale,
                    h: 24 * scale,
                    alive: true,
                    type: r < 2 ? 0 : (r < 4 ? 1 : 2),
                    animFrame: Math.floor(Math.random() * 2)
                });
            }
        }
    }

    function resetGame() {
        score = 0;
        lives = 3;
        level = 1;
        pacman.x = W / 2;
        pacman.y = H - 60 * scale;
        pacman.invincible = 0;
        bullets = [];
        enemyBullets = [];
        powerUps = [];
        particles = [];
        bulletCooldown = 0;
        alienDirection = 1;
        alienSpeed = 1.5 + level * 0.3;
        initAliens();
    }

    // Input
    let keys = {};
    window.addEventListener('keydown', function(e) {
        keys[e.code] = true;
        if (e.code === 'Space') e.preventDefault();
        initAudio();
        if (gameState === 'start' && (e.code === 'Space' || e.code === 'Enter')) {
            gameState = 'playing';
            resetGame();
        } else if (gameState === 'gameover' && (e.code === 'Space' || e.code === 'Enter' || e.code === 'KeyR')) {
            gameState = 'playing';
            resetGame();
        }
    });
    window.addEventListener('keyup', function(e) { keys[e.code] = false; });

    // Drawing functions
    function drawPacman(x, y, size, mouthAngle) {
        ctx.save();
        ctx.translate(x, y);
        ctx.fillStyle = '#FFE135';
        ctx.beginPath();
        const gap = Math.abs(mouthAngle) * 0.8;
        if (gap < 0.1) gap = 0.1;
        ctx.arc(0, 0, size / 2, gap, Math.PI * 2 - gap);
        ctx.lineTo(0, 0);
        ctx.closePath();
        ctx.fill();
        // Eyes
        ctx.fillStyle = '#FFF';
        ctx.beginPath();
        ctx.arc(-size * 0.15, -size * 0.2, size * 0.12, 0, Math.PI * 2);
        ctx.fill();
        ctx.restore();
    }

    function drawAlien(alien) {
        const { x, y, w, h, type, animFrame } = alien;
        ctx.save();
        ctx.translate(x, y);
        const colors = ['#FF0044', '#00FF88', '#FFAA00'];
        const color = colors[type];
        ctx.fillStyle = color;

        // Pixel art alien (3 frames animation)
        const s = w / 8;
        const f = animFrame % 2;

        if (type === 0) {
            // Top row - squid-like
            ctx.fillRect(2*s, 1*s, 4*s, 1*s);
            ctx.fillRect(1*s, 2*s, 6*s, 1*s);
            ctx.fillRect(1*s, 3*s, 2*s, 1*s);
            ctx.fillRect(5*s, 3*s, 2*s, 1*s);
            ctx.fillRect(0*s, 4*s, 8*s, 1*s);
            ctx.fillRect(1*s, 5*s, 6*s, 1*s);
            if (f === 0) {
                ctx.fillRect(2*s, 6*s, 2*s, 1*s);
                ctx.fillRect(4*s, 6*s, 2*s, 1*s);
            } else {
                ctx.fillRect(1*s, 6*s, 2*s, 1*s);
                ctx.fillRect(5*s, 6*s, 2*s, 1*s);
            }
        } else if (type === 1) {
            // Middle - crab-like
            ctx.fillRect(2*s, 1*s, 2*s, 1*s);
            ctx.fillRect(4*s, 1*s, 2*s, 1*s);
            ctx.fillRect(1*s, 2*s, 6*s, 1*s);
            ctx.fillRect(1*s, 3*s, 6*s, 1*s);
            ctx.fillRect(0*s, 3*s, 1*s, 1*s);
            ctx.fillRect(7*s, 3*s, 1*s, 1*s);
            ctx.fillRect(0*s, 4*s, 8*s, 1*s);
            ctx.fillRect(2*s, 5*s, 2*s, 1*s);
            ctx.fillRect(4*s, 5*s, 2*s, 1*s);
            if (f === 0) {
                ctx.fillRect(0*s, 6*s, 2*s, 1*s);
                ctx.fillRect(6*s, 6*s, 2*s, 1*s);
            } else {
                ctx.fillRect(2*s, 6*s, 2*s, 1*s);
                ctx.fillRect(4*s, 6*s, 2*s, 1*s);
            }
        } else {
            // Bottom - octopus-like
            ctx.fillRect(2*s, 0*s, 4*s, 1*s);
            ctx.fillRect(1*s, 1*s, 6*s, 1*s);
            ctx.fillRect(0*s, 2*s, 8*s, 1*s);
            ctx.fillRect(0*s, 3*s, 8*s, 1*s);
            ctx.fillRect(1*s, 4*s, 6*s, 1*s);
            ctx.fillRect(0*s, 5*s, 2*s, 1*s);
            ctx.fillRect(6*s, 5*s, 2*s, 1*s);
            if (f === 0) {
                ctx.fillRect(1*s, 6*s, 2*s, 1*s);
                ctx.fillRect(5*s, 6*s, 2*s, 1*s);
            } else {
                ctx.fillRect(0*s, 6*s, 2*s, 1*s);
                ctx.fillRect(6*s, 6*s, 2*s, 1*s);
            }
        }
        ctx.restore();
    }

    function drawPowerUp(p) {
        ctx.save();
        ctx.translate(p.x, p.y);
        const pulse = Math.sin(frame * 0.15) * 2;
        if (p.type === 0) {
            // Cherry (double shot)
            ctx.fillStyle = '#FF3366';
            ctx.beginPath();
            ctx.arc(-4, 4 + pulse * 0.3, 5, 0, Math.PI * 2);
            ctx.fill();
            ctx.beginPath();
            ctx.arc(4, 4 + pulse * 0.3, 5, 0, Math.PI * 2);
            ctx.fill();
            ctx.strokeStyle = '#44AA00';
            ctx.lineWidth = 2;
            ctx.beginPath();
            ctx.moveTo(-4, -1);
            ctx.quadraticCurveTo(0, -8, 0, -12);
            ctx.stroke();
        } else {
            // WPELLET (speed)
            ctx.fillStyle = '#FFD700';
            ctx.font = `bold ${14 * scale}px monospace`;
            ctx.textAlign = 'center';
            ctx.fillText('★', 0, 5 + pulse * 0.3);
        }
        ctx.restore();
    }

    function drawParticle(p) {
        ctx.save();
        ctx.globalAlpha = p.life / p.maxLife;
        ctx.fillStyle = p.color;
        ctx.fillRect(p.x - p.size/2, p.y - p.size/2, p.size, p.size);
        ctx.restore();
    }

    function spawnExplosion(x, y, color, count) {
        for (let i = 0; i < count; i++) {
            const angle = Math.random() * Math.PI * 2;
            const speed = Math.random() * 3 + 1;
            particles.push({
                x: x, y: y,
                vx: Math.cos(angle) * speed,
                vy: Math.sin(angle) * speed,
                size: Math.random() * 4 + 2,
                color: color,
                life: 30 + Math.random() * 20,
                maxLife: 50
            });
        }
    }

    // Update
    function update() {
        frame++;

        if (gameState !== 'playing') return;

        // Pac-Man movement
        let moveX = 0;
        if (keys['ArrowLeft'] || keys['KeyA']) moveX = -1;
        if (keys['ArrowRight'] || keys['KeyD']) moveX = 1;
        pacman.x += moveX * pacman.speed * scale / 2;
        pacman.x = Math.max(20, Math.min(W - 20, pacman.x));
        pacman.mouthAngle += moveX !== 0 ? 0.15 : 0.05;

        if (pacman.invincible > 0) pacman.invincible--;

        // Shooting
        if (bulletCooldown > 0) bulletCooldown--;
        if ((keys['Space'] || keys['KeyW'] || keys['ArrowUp']) && bulletCooldown <= 0) {
            bullets.push({ x: pacman.x, y: pacman.y - 20 * scale, vy: -8 * scale, w: 4, h: 12 });
            bulletCooldown = 15;
            playSound('shoot');
        }

        // Update bullets
        for (let i = bullets.length - 1; i >= 0; i--) {
            bullets[i].y += bullets[i].vy;
            if (bullets[i].y < -20) bullets.splice(i, 1);
        }

        // Aliens movement
        alienMoveTimer++;
        const moveInterval = Math.max(3, 20 - level * 2);
        if (alienMoveTimer >= moveInterval) {
            alienMoveTimer = 0;
            for (let a of aliens) {
                if (!a.alive) continue;
                a.x += alienDirection * alienSpeed * scale / 2;
            }
            // Check bounds
            let minX = W, maxX = 0;
            for (let a of aliens) {
                if (a.alive) {
                    minX = Math.min(minX, a.x - a.w/2);
                    maxX = Math.max(maxX, a.x + a.w/2);
                }
            }
            if (minX < 10 || maxX > W - 10) {
                alienDirection *= -1;
                for (let a of aliens) {
                    if (a.alive) { a.y += 15 * scale / moveInterval; a.x += alienDirection * 20 * scale / moveInterval; }
                }
            }
        }

        // Animate aliens
        if (frame % 15 === 0) {
            for (let a of aliens) { if (a.alive) a.animFrame++; }
        }

        // Alien shooting
        alienShootTimer++;
        const shootInterval = Math.max(30, 80 - level * 10);
        if (alienShootTimer >= shootInterval) {
            alienShootTimer = 0;
            const aliveAliens = aliens.filter(a => a.alive);
            if (aliveAliens.length > 0) {
                const shooter = aliveAliens[Math.floor(Math.random() * aliveAliens.length)];
                enemyBullets.push({ x: shooter.x, y: shooter.y + 15, vy: 3 * scale, w: 4, h: 12 });
            }
        }

        // Update enemy bullets
        for (let i = enemyBullets.length - 1; i >= 0; i--) {
            enemyBullets[i].y += enemyBullets[i].vy;
            if (enemyBullets[i].y > H + 20) { enemyBullets.splice(i, 1); continue; }
            // Hit Pac-Man
            if (pacman.invincible <= 0) {
                const dx = Math.abs(enemyBullets[i].x - pacman.x);
                const dy = Math.abs(enemyBullets[i].y - pacman.y);
                if (dx < 20 && dy < 20) {
                    enemyBullets.splice(i, 1);
                    lives--;
                    playSound('hit');
                    spawnExplosion(pacman.x, pacman.y, '#FFE135', 15);
                    if (lives <= 0) {
                        gameState = 'gameover';
                        try { 
                            if (score > highScore) { highScore = score; localStorage.setItem('pacSpaceHigh', highScore); }
                        } catch(e) {}
                    } else {
                        pacman.invincible = 90;
                    }
                }
            }
        }

        // Bullet vs Alien collision
        for (let i = bullets.length - 1; i >= 0; i--) {
            const b = bullets[i];
            for (let j = aliens.length - 1; j >= 0; j--) {
                const a = aliens[j];
                if (!a.alive) continue;
                if (Math.abs(b.x - a.x) < a.w/2 + 4 && Math.abs(b.y - a.y) < a.h/2 + 6) {
                    a.alive = false;
                    bullets.splice(i, 1);
                    score += (a.type === 0 ? 30 : a.type === 1 ? 20 : 10) * level;
                    playSound('explosion');
                    const colors = ['#FF0044', '#00FF88', '#FFAA00'];
                    spawnExplosion(a.x, a.y, colors[a.type], 12);
                    break;
                }
            }
        }

        // Bullet vs PowerUp
        for (let i = bullets.length - 1; i >= 0; i--) {
            const b = bullets[i];
            for (let j = powerUps.length - 1; j >= 0; j--) {
                const p = powerUps[j];
                if (Math.abs(b.x - p.x) < 20 && Math.abs(b.y - p.y) < 20) {
                    bullets.splice(i, 1);
                    powerUps.splice(j, 1);
                    playSound('powerup');
                    score += 50;
                    if (p.type === 0) { /* double shot handled by score */ }
                    else { pacman.speed = Math.min(6, pacman.speed + 0.5); }
                    spawnExplosion(p.x, p.y, '#FFD700', 8);
                }
            }
        }

        // PowerUp timer
        powerUpTimer++;
        if (powerUpTimer > 180) {
            powerUpTimer = 0;
            const type = Math.random() < 0.5 ? 0 : 1;
            powerUps.push({ x: Math.random() * (W - 80) + 40, y: H * 0.3 + Math.random() * H * 0.3, type: type });
        }

        // Check level clear
        const aliveCount = aliens.filter(a => a.alive).length;
        if (aliveCount === 0) {
            level++;
            alienSpeed = 1.5 + level * 0.3;
            initAliens();
            score += 100;
        }

        // Check aliens reaching bottom
        for (let a of aliens) {
            if (a.alive && a.y > pacman.y - 30) {
                lives = 0;
                gameState = 'gameover';
                try { 
                    if (score > highScore) { highScore = score; localStorage.setItem('pacSpaceHigh', highScore); }
                } catch(e) {}
            }
        }

        // Update particles
        for (let i = particles.length - 1; i >= 0; i--) {
            const p = particles[i];
            p.x += p.vx;
            p.y += p.vy;
            p.vy += 0.1;
            p.life--;
            if (p.life <= 0) particles.splice(i, 1);
        }

        // Stars
        for (let s of stars) {
            s.twinkle += 0.05;
        }
    }

    function draw() {
        ctx.fillStyle = '#0A0A1A';
        ctx.fillRect(0, 0, W, H);

        // Stars
        for (let s of stars) {
            const alpha = 0.3 + Math.sin(s.twinkle) * 0.3;
            ctx.globalAlpha = alpha;
            ctx.fillStyle = '#FFF';
            ctx.fillRect(s.x, s.y, s.size, s.size);
        }
        ctx.globalAlpha = 1;

        if (gameState === 'start') {
            // Title
            const titleSize = Math.min(W / 8, 60);
            ctx.font = `bold ${titleSize}px monospace`;
            ctx.textAlign = 'center';
            ctx.fillStyle = '#FFE135';
            ctx.fillText('PAC-MAN', W/2, H * 0.25);
            ctx.fillStyle = '#FF4466';
            ctx.fillText('SPACE INVADERS', W/2, H * 0.25 + titleSize);

            // Pac-Man animation on start screen
            const sm = Math.sin(frame * 0.1) * 0.8;
            drawPacman(W/2 - 150, H * 0.5, 40, sm);
            // Alien
            ctx.save();
            ctx.translate(W/2 + 150, H * 0.5);
            ctx.scale(2, 2);
            const fakeAlien = { x:0, y:0, w:30, h:26, type:0, animFrame: Math.floor(frame/15)%2 };
            drawAlien({x:0,y:0,w:30,h:26,type:0,alive:true,animFrame:Math.floor(frame/15)%2});
            ctx.restore();

            // Controls
            const ctrlSize = Math.min(W / 25, 18);
            ctx.font = `${ctrlSize}px monospace`;
            ctx.fillStyle = '#AAA';
            ctx.fillText('← → or A/D : Move', W/2, H * 0.65);
            ctx.fillText('SPACE / ↑ / W : Shoot', W/2, H * 0.7);
            ctx.fillText('', W/2, H * 0.78);
            const blink = Math.sin(frame * 0.1) > 0;
            if (blink) {
                ctx.fillStyle = '#FFE135';
                ctx.font = `bold ${ctrlSize + 4}px monospace`;
                ctx.fillText('PRESS SPACE TO START', W/2, H * 0.85);
            }
            if (highScore > 0) {
                ctx.font = `${ctrlSize}px monospace`;
                ctx.fillStyle = '#FFAA00';
                ctx.fillText(`HIGH SCORE: ${highScore}`, W/2, H * 0.92);
            }
        } else if (gameState === 'playing') {
            // Draw power-ups
            for (let p of powerUps) drawPowerUp(p);

            // Draw aliens
            for (let a of aliens) { if (a.alive) drawAlien(a); }

            // Draw bullets
            ctx.fillStyle = '#FFFF44';
            for (let b of bullets) {
                ctx.fillRect(b.x - 2, b.y - 6, 4, 12);
                // Glow
                ctx.globalAlpha = 0.3;
                ctx.fillRect(b.x - 4, b.y - 8, 8, 16);
                ctx.globalAlpha = 1;
            }

            // Draw enemy bullets
            ctx.fillStyle = '#FF4444';
            for (let b of enemyBullets) {
                ctx.fillRect(b.x - 2, b.y - 6, 4, 12);
                ctx.globalAlpha = 0.3;
                ctx.fillRect(b.x - 4, b.y - 8, 8, 16);
                ctx.globalAlpha = 1;
            }

            // Draw Pac-Man
            if (pacman.invincible <= 0 || Math.floor(pacman.invincible / 5) % 2 === 0) {
                drawPacman(pacman.x, pacman.y, pacman.w * scale, pacman.mouthAngle);
            }

            // Particles
            for (let p of particles) drawParticle(p);

            // HUD
            const hudSize = Math.min(W / 30, 16);
            ctx.font = `bold ${hudSize}px monospace`;
            ctx.textAlign = 'left';
            ctx.fillStyle = '#FFE135';
            ctx.fillText(`SCORE: ${score}`, 20, 30);
            ctx.fillStyle = '#FF4444';
            for (let i = 0; i < lives; i++) {
                drawPacman(150 + i * 30, 25, 20, 0.3);
            }
            ctx.textAlign = 'right';
            ctx.fillStyle = '#44FF88';
            ctx.fillText(`LEVEL ${level}`, W - 20, 30);
            if (highScore > 0) {
                ctx.fillStyle = '#888';
                ctx.font = `${hudSize * 0.8}px monospace`;
                ctx.fillText(`HI: ${highScore}`, W - 20, 50);
            }
        } else if (gameState === 'gameover') {
            // Still draw the game state behind
            for (let a of aliens) { if (a.alive) drawAlien(a); }
            for (let p of particles) drawParticle(p);

            ctx.font = `bold ${Math.min(W/10, 50)}px monospace`;
            ctx.textAlign = 'center';
            ctx.fillStyle = '#FF4444';
            ctx.fillText('GAME OVER', W/2, H * 0.35);

            ctx.font = `${Math.min(W/25, 20)}px monospace`;
            ctx.fillStyle = '#FFE135';
            ctx.fillText(`FINAL SCORE: ${score}`, W/2, H * 0.48);
            ctx.fillStyle = '#FFAA00';
            ctx.fillText(`HIGH SCORE: ${highScore}`, W/2, H * 0.55);

            const blink = Math.sin(frame * 0.1) > 0;
            if (blink) {
                ctx.fillStyle = '#44FF88';
                ctx.fillText('PRESS SPACE OR R TO RESTART', W/2, H * 0.7);
            }
        }
    }

    function gameLoop() {
        update();
        draw();
        requestAnimationFrame(gameLoop);
    }
