<script>
const canvas = document.getElementById('game');
const ctx = canvas.getContext('2d');

let W, H;
function resize() {
    W = canvas.width = window.innerWidth;
    H = canvas.height = window.innerHeight;
}
window.addEventListener('resize', resize);
resize();

// Audio
let audioCtx = null;
function initAudio() {
    if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
}
function playSound(type) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    
    if (type === 'shoot') {
        osc.type = 'square';
        osc.frequency.setValueAtTime(800, audioCtx.currentTime);
        osc.frequency.exponentialRampToValueAtTime(200, audioCtx.currentTime + 0.1);
        gain.gain.setValueAtTime(0.15, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.1);
        osc.start(audioCtx.currentTime);
        osc.stop(audioCtx.currentTime + 0.1);
    } else if (type === 'explosion') {
        osc.type = 'sawtooth';
        osc.frequency.setValueAtTime(200, audioCtx.currentTime);
        osc.frequency.exponentialRampToValueAtTime(50, audioCtx.currentTime + 0.3);
        gain.gain.setValueAtTime(0.2, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.3);
        osc.start(audioCtx.currentTime);
        osc.stop(audioCtx.currentTime + 0.3);
    } else if (type === 'playerHit') {
        osc.type = 'sawtooth';
        osc.frequency.setValueAtTime(400, audioCtx.currentTime);
        osc.frequency.exponentialRampToValueAtTime(80, audioCtx.currentTime + 0.5);
        gain.gain.setValueAtTime(0.3, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.5);
        osc.start(audioCtx.currentTime);
        osc.stop(audioCtx.currentTime + 0.5);
    } else if (type === 'wave') {
        osc.type = 'sine';
        osc.frequency.setValueAtTime(440, audioCtx.currentTime);
        osc.frequency.exponentialRampToValueAtTime(880, audioCtx.currentTime + 0.2);
        gain.gain.setValueAtTime(0.15, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.3);
        osc.start(audioCtx.currentTime);
        osc.stop(audioCtx.currentTime + 0.3);
    }
}

// Game State
let gameState = 'start'; // start, playing, gameover
let score = 0;
let highScore = 0;
let wave = 1;
let playerLives = 3;

try {
    highScore = parseInt(localStorage.getItem('siTopDownHigh') || '0');
} catch(e) { highScore = 0; }

// Stars
const stars = [];
for (let i = 0; i < 150; i++) {
    stars.push({ x: Math.random(), y: Math.random(), s: Math.random() * 2 + 0.5, sp: Math.random() * 0.3 + 0.1 });
}

// Player
let player = { x: 0, y: 0, w: 40, h: 30, speed: 6 };
let bullets = [];
let enemies = [];
let enemyBullets = [];
let particles = [];
let shieldBlocks = [];

function initGame() {
    score = 0;
    wave = 1;
    playerLives = 3;
    bullets = [];
    enemyBullets = [];
    particles = [];
    player.x = W / 2;
    player.y = H - 60;
    spawnWave();
}

function spawnWave() {
    enemies = [];
    const cols = 10;
    const rows = 5;
    const spacingX = Math.min(50, (W * 0.7) / cols);
    const spacingY = 40;
    const startX = W / 2 - (cols - 1) * spacingX / 2;
    const startY = 80 + wave * 10;

    for (let r = 0; r < rows; r++) {
        for (let c = 0; c < cols; c++) {
            enemies.push({
                x: startX + c * spacingX,
                y: startY + r * spacingY,
                w: 30, h: 25,
                row: r, col: c,
                hp: r === 0 ? 1 : (r < 3 ? 2 : 3),
                type: r,
                alive: true
            });
        }
    }
}

function spawnShield() {
    shieldBlocks = [];
    const shieldY = H - 120;
    const segW = 8;
    const segH = 6;
    for (let row = 0; row < 4; row++) {
        for (let col = 0; col < Math.floor(W / segW); col++) {
            if (col > W * 0.3 / segW && col < W * 0.7 / segW) continue;
            shieldBlocks.push({ x: col * segW, y: shieldY + row * segH, w: segW, h: segH, alive: true });
        }
    }
}

function createParticles(x, y, color, count) {
    for (let i = 0; i < count; i++) {
        const angle = Math.random() * Math.PI * 2;
        const speed = Math.random() * 4 + 1;
        particles.push({
            x, y,
            vx: Math.cos(angle) * speed,
            vy: Math.sin(angle) * speed,
            life: 1,
            decay: Math.random() * 0.03 + 0.02,
            color,
            size: Math.random() * 4 + 2
        });
    }
}

// Input
const keys = {};
window.addEventListener('keydown', (e) => {
    keys[e.code] = true;
    initAudio();
    if (gameState === 'start' && (e.code === 'Space' || e.code === 'Enter')) {
        gameState = 'playing';
        initGame();
        spawnShield();
    } else if (gameState === 'gameover' && (e.code === 'KeyR' || e.code === 'Enter')) {
        gameState = 'playing';
        initGame();
        spawnShield();
    }
});
window.addEventListener('keyup', (e) => { keys[e.code] = false; });

// Enemy movement
let enemyDirX = 1;
let enemySpeed = 2;
let enemyMoveTimer = 0;
let enemyShootTimer = 0;

function updateEnemies() {
    const alive = enemies.filter(e => e.alive);
    if (alive.length === 0) {
        wave++;
        spawnWave();
        playSound('wave');
        return;
    }

    const speedMult = 1 + wave * 0.3;
    enemySpeed = (1.5 + wave * 0.5) * speedMult;

    let minX = Infinity, maxX = -Infinity;
    for (const e of alive) {
        if (e.x < minX) minX = e.x;
        if (e.x > maxX) maxX = e.x;
    }

    if ((enemyDirX > 0 && maxX + enemySpeed >= W - 20) || (enemyDirX < 0 && minX - enemySpeed <= 20)) {
        enemyDirX *= -1;
        for (const e of alive) e.y += 20;
    }

    for (const e of alive) {
        e.x += enemySpeed * enemyDirX;
    }

    // Enemy shooting
    enemyShootTimer++;
    const shootInterval = Math.max(30, 80 - wave * 5);
    if (enemyShootTimer >= shootInterval) {
        enemyShootTimer = 0;
        const shooters = alive.filter(e => e.row === 4 || Math.random() < 0.3);
        if (shooters.length > 0) {
            const shooter = shooters[Math.floor(Math.random() * shooters.length)];
            enemyBullets.push({ x: shooter.x, y: shooter.y + 15, vy: 4 + wave * 0.3, vx: 0 });
        }
    }

    // Check if enemies reached player level
    for (const e of alive) {
        if (e.y > H - 80) {
            gameOver();
            return;
        }
    }
}

function updateBullets() {
    for (let i = bullets.length - 1; i >= 0; i--) {
        bullets[i].y -= 10;
        if (bullets[i].y < -10) bullets.splice(i, 1);
    }
    for (let i = enemyBullets.length - 1; i >= 0; i--) {
        enemyBullets[i].y += enemyBullets[i].vy;
        if (enemyBullets[i].y > H + 10) enemyBullets.splice(i, 1);
    }
}

function checkCollisions() {
    // Bullets vs enemies
    for (let i = bullets.length - 1; i >= 0; i--) {
        const b = bullets[i];
        let hit = false;
        for (const e of enemies) {
            if (!e.alive) continue;
            if (b.x > e.x - e.w/2 && b.x < e.x + e.w/2 && b.y > e.y - e.h/2 && b.y < e.y + e.h/2) {
                e.hp--;
                hit = true;
                if (e.hp <= 0) {
                    e.alive = false;
                    score += (5 - e.row) * 10 + wave * 5;
                    createParticles(e.x, e.y, getEnemyColor(e.type), 12);
                    playSound('explosion');
                }
                break;
            }
        }
        if (hit) bullets.splice(i, 1);

        // Bullets vs shields
        for (const s of shieldBlocks) {
            if (!s.alive) continue;
            if (b.x > s.x && b.x < s.x + s.w && b.y > s.y && b.y < s.y + s.h) {
                s.alive = false;
                bullets.splice(i, 1);
                hit = true;
                break;
            }
        }
    }

    // Enemy bullets vs player
    for (let i = enemyBullets.length - 1; i >= 0; i--) {
        const b = enemyBullets[i];
        if (b.x > player.x - player.w/2 && b.x < player.x + player.w/2 &&
            b.y > player.y - player.h/2 && b.y < player.y + player.h/2) {
            enemyBullets.splice(i, 1);
            playerLives--;
            createParticles(player.x, player.y, '#4ff', 20);
            playSound('playerHit');
            if (playerLives <= 0) gameOver();
        }

        // Enemy bullets vs shields
        for (const s of shieldBlocks) {
            if (!s.alive) continue;
            if (b.x > s.x && b.x < s.x + s.w && b.y > s.y && b.y < s.y + s.h) {
                s.alive = false;
                enemyBullets.splice(i, 1);
                break;
            }
        }
    }

    // Enemies vs shields (when they reach shield level)
    for (const e of enemies) {
        if (!e.alive) continue;
        for (const s of shieldBlocks) {
            if (!s.alive) continue;
            if (e.x > s.x - 10 && e.x < s.x + s.w + 10 && e.y > s.y - 5 && e.y < s.y + s.h + 5) {
                s.alive = false;
            }
        }
    }
}

function getEnemyColor(type) {
    const colors = ['#f44', '#fa4', '#ff2', '#4f8', '#4ff'];
    return colors[type] || '#fff';
}

function gameOver() {
    gameState = 'gameover';
    if (score > highScore) {
        highScore = score;
        try { localStorage.setItem('siTopDownHigh', highScore.toString()); } catch(e) {}
    }
}

function updateParticles() {
    for (let i = particles.length - 1; i >= 0; i--) {
        const p = particles[i];
        p.x += p.vx;
        p.y += p.vy;
        p.life -= p.decay;
        if (p.life <= 0) particles.splice(i, 1);
    }
}

function updateStars() {
    for (const s of stars) {
        s.y += s.sp * 0.005;
        if (s.y > 1) { s.y = 0; s.x = Math.random(); }
    }
}

// Rendering
function drawStars() {
    ctx.fillStyle = '#fff';
    for (const s of stars) {
        ctx.globalAlpha = 0.5 + Math.sin(Date.now() * 0.003 + s.x * 10) * 0.3;
        ctx.fillRect(s.x * W, s.y * H, s.s, s.s);
    }
    ctx.globalAlpha = 1;
}

function drawPlayer() {
    const px = player.x, py = player.y;
    // Top-down ship (viewed from above)
    ctx.save();
    ctx.translate(px, py);
    
    // Main body
    ctx.fillStyle = '#4af';
    ctx.beginPath();
    ctx.moveTo(0, -18);
    ctx.lineTo(-8, -5);
    ctx.lineTo(-15, 12);
    ctx.lineTo(-6, 8);
    ctx.lineTo(0, 15);
    ctx.lineTo(6, 8);
    ctx.lineTo(15, 12);
    ctx.lineTo(8, -5);
    ctx.closePath();
    ctx.fill();
    
    // Cockpit
    ctx.fillStyle = '#8df';
    ctx.beginPath();
    ctx.ellipse(0, -4, 4, 7, 0, 0, Math.PI * 2);
    ctx.fill();
    
    // Engine glow
    ctx.fillStyle = `rgba(100, 200, 255, ${0.5 + Math.sin(Date.now() * 0.01) * 0.3})`;
    ctx.beginPath();
    ctx.ellipse(0, 16, 5, 3, 0, 0, Math.PI * 2);
    ctx.fill();
    
    ctx.restore();
}

function drawEnemy(e) {
    if (!e.alive) return;
    const color = getEnemyColor(e.type);
    const pulse = Math.sin(Date.now() * 0.005 + e.x * 0.1) * 2;
    
    ctx.save();
    ctx.translate(e.x, e.y);
    ctx.fillStyle = color;
    ctx.strokeStyle = color;
    ctx.lineWidth = 2;
    
    // Top-down alien (viewed from above - different shapes by type)
    if (e.type === 0) {
        // Small alien - diamond with spikes
        ctx.beginPath();
        ctx.moveTo(0, -10 + pulse);
        ctx.lineTo(-8, 0);
        ctx.lineTo(0, 10);
        ctx.lineTo(8, 0);
        ctx.closePath();
        ctx.fill();
        // Eyes
        ctx.fillStyle = '#fff';
        ctx.fillRect(-4, -3, 2, 2);
        ctx.fillRect(2, -3, 2, 2);
    } else if (e.type === 1) {
        // Medium alien - hexagonal
        ctx.beginPath();
        for (let i = 0; i < 6; i++) {
            const a = (i / 6) * Math.PI * 2;
            const r = 12 + pulse;
            if (i === 0) ctx.moveTo(Math.cos(a) * r, Math.sin(a) * r);
            else ctx.lineTo(Math.cos(a) * r, Math.sin(a) * r);
        }
        ctx.closePath();
        ctx.fill();
        ctx.fillStyle = '#fff';
        ctx.fillRect(-5, -4, 3, 3);
        ctx.fillRect(2, -4, 3, 3);
    } else if (e.type === 2) {
        // Large alien - octagonal with wings
        ctx.beginPath();
        for (let i = 0; i < 8; i++) {
            const a = (i / 8) * Math.PI * 2 + 0.39;
            const r = 14 + pulse;
            if (i === 0) ctx.moveTo(Math.cos(a) * r, Math.sin(a) * r);
            else ctx.lineTo(Math.cos(a) * r, Math.sin(a) * r);
        }
        ctx.closePath();
        ctx.fill();
        // Wings
        ctx.beginPath();
        ctx.moveTo(-14, 0); ctx.lineTo(-20, -8);
        ctx.moveTo(14, 0); ctx.lineTo(20, -8);
        ctx.stroke();
        ctx.fillStyle = '#fff';
        ctx.fillRect(-6, -5, 3, 3);
        ctx.fillRect(3, -5, 3, 3);
    } else {
        // Boss alien - large with crown
        ctx.beginPath();
        for (let i = 0; i < 10; i++) {
            const a = (i / 10) * Math.PI * 2;
            const r = (i % 2 === 0 ? 16 : 12) + pulse;
            if (i === 0) ctx.moveTo(Math.cos(a) * r, Math.sin(a) * r);
            else ctx.lineTo(Math.cos(a) * r, Math.sin(a) * r);
        }
        ctx.closePath();
        ctx.fill();
        // Crown spikes
        ctx.beginPath();
        for (let i = -3; i <= 3; i++) {
            ctx.moveTo(i * 4, -14 + pulse);
            ctx.lineTo(i * 4, -20 + pulse);
        }
        ctx.stroke();
        ctx.fillStyle = '#fff';
        ctx.fillRect(-7, -5, 3, 3);
        ctx.fillRect(4, -5, 3, 3);
    }
    
    ctx.restore();
}

function drawBullets() {
    ctx.fillStyle = '#ff4';
    for (const b of bullets) {
        ctx.fillRect(b.x - 2, b.y - 8, 4, 12);
        ctx.fillStyle = 'rgba(255, 255, 100, 0.3)';
        ctx.fillRect(b.x - 4, b.y - 8, 8, 12);
        ctx.fillStyle = '#ff4';
    }
    
    for (const b of enemyBullets) {
        ctx.fillStyle = '#f4f';
        ctx.beginPath();
        ctx.arc(b.x, b.y, 4, 0, Math.PI * 2);
        ctx.fill();
        ctx.fillStyle = 'rgba(255, 100, 255, 0.3)';
        ctx.beginPath();
        ctx.arc(b.x, b.y, 7, 0, Math.PI * 2);
        ctx.fill();
    }
}

function drawShields() {
    ctx.fillStyle = '#2a8';
    for (const s of shieldBlocks) {
        if (s.alive) ctx.fillRect(s.x, s.y, s.w - 1, s.h - 1);
    }
}

function drawParticles() {
    for (const p of particles) {
        ctx.globalAlpha = p.life;
        ctx.fillStyle = p.color;
        ctx.fillRect(p.x - p.size/2, p.y - p.size/2, p.size, p.size);
    }
    ctx.globalAlpha = 1;
}

function drawHUD() {
    ctx.fillStyle = '#fff';
    ctx.font = 'bold 16px monospace';
    ctx.textAlign = 'left';
    ctx.fillText(`SCORE: ${score}`, 20, 30);
    ctx.fillText(`WAVE: ${wave}`, 20, 52);
    
    // Lives
    ctx.textAlign = 'right';
    for (let i = 0; i < playerLives; i++) {
        ctx.fillStyle = '#4af';
        const lx = W - 30 - i * 30;
        ctx.beginPath();
        ctx.moveTo(lx, 25);
        ctx.lineTo(lx - 8, 40);
        ctx.lineTo(lx + 8, 40);
        ctx.closePath();
        ctx.fill();
    }
    
    ctx.textAlign = 'center';
    ctx.fillStyle = '#888';
    ctx.font = '12px monospace';
    ctx.fillText(`HI: ${highScore}`, W / 2, 25);
}

function drawStartScreen() {
    ctx.fillStyle = '#0a0a1a';
    ctx.fillRect(0, 0, W, H);
    drawStars();
    
    ctx.textAlign = 'center';
    ctx.fillStyle = '#4af';
    ctx.font = 'bold 48px monospace';
    ctx.fillText('SPACE INVADERS', W/2, H/3 - 20);
    ctx.font = 'bold 24px monospace';
    ctx.fillStyle = '#f44';
    ctx.fillText('TOP-DOWN', W/2, H/3 + 20);
    
    ctx.fillStyle = '#fff';
    ctx.font = '16px monospace';
    ctx.fillText('← → or A/D : Move', W/2, H/2 + 20);
    ctx.fillText('SPACE : Shoot', W/2, H/2 + 50);
    ctx.fillText('R : Restart (on game over)', W/2, H/2 + 80);
    
    ctx.fillStyle = `rgba(255,255,255,${0.5 + Math.sin(Date.now() * 0.005) * 0.4})`;
    ctx.font = '18px monospace';
    ctx.fillText('PRESS SPACE TO START', W/2, H * 0.75);
    
    if (highScore > 0) {
        ctx.fillStyle = '#ff4';
        ctx.font = '14px monospace';
        ctx.fillText(`HIGH SCORE: ${highScore}`, W/2, H * 0.85);
    }
}

function drawGameOver() {
    ctx.textAlign = 'center';
    ctx.fillStyle = 'rgba(0,0,0,0.7)';
    ctx.fillRect(0, 0, W, H);
    
    ctx.fillStyle = '#f44';
    ctx.font = 'bold 48px monospace';
    ctx.fillText('GAME OVER', W/2, H/3);
    
    ctx.fillStyle = '#fff';
    ctx.font = '24px monospace';
    ctx.fillText(`FINAL SCORE: ${score}`, W/2, H/3 + 50);
    ctx.fillText(`WAVE REACHED: ${wave}`, W/2, H/3 + 90);
    
    if (score >= highScore && score > 0) {
        ctx.fillStyle = '#ff4';
        ctx.font = '20px monospace';
        ctx.fillText('★ NEW HIGH SCORE! ★', W/2, H/3 + 130);
    }
    
    ctx.fillStyle = `rgba(255,255,255,${0.5 + Math.sin(Date.now() * 0.005) * 0.4})`;
    ctx.font = '18px monospace';
    ctx.fillText('PRESS R TO RESTART', W/2, H * 0.7);
}

// Main loop
let lastTime = 0;
function gameLoop(time) {
    const dt = time - lastTime;
    lastTime = time;
    
    if (gameState === 'start') {
        updateStars();
        drawStartScreen();
    } else if (gameState === 'playing') {
        // Update player
        if (keys['ArrowLeft'] || keys['KeyA']) player.x -= player.speed;
        if (keys['ArrowRight'] || keys['KeyD']) player.x += player.speed;
        player.x = Math.max(25, Math.min(W - 25, player.x));
        
        if (keys['Space'] && bullets.length < 3) {
            // Auto-fire with cooldown managed by bullet limit
            if (!keys['_justFired'] || time % 10 === 0) {
                bullets.push({ x: player.x, y: player.y - 20 });
                playSound('shoot');
            }
        }
        
        updateEnemies();
        updateBullets();
        checkCollisions();
        updateParticles();
        updateStars();
        
        // Draw
        ctx.fillStyle = '#0a0a1a';
        ctx.fillRect(0, 0, W, H);
        drawStars();
        drawShields();
        for (const e of enemies) drawEnemy(e);
        drawBullets();
        drawPlayer();
        drawParticles();
        drawHUD();
    } else if (gameState === 'gameover') {
        updateStars();
        ctx.fillStyle = '#0a0a1a';
        ctx.fillRect(0, 0, W, H);
        drawStars();
        for (const e of enemies) if (e.alive) drawEnemy(e);
        drawParticles();
        drawHUD();
        drawGameOver();
    }
    
    requestAnimationFrame(gameLoop);
}

requestAnimationFrame(gameLoop);
</script>
</body>
</html>
