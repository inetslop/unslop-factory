```html
<!DOCTYPE html>
<html lang="en">
<head>
<meta charset="UTF-8">
<meta name="viewport" content="width=device-width, initial-scale=1.0">
<title>Breakout Endless</title>
<style>
* { margin: 0; padding: 0; box-sizing: border-box; }
html, body { width: 100%; height: 100%; overflow: hidden; background: #0a0a1a; }
canvas { display: block; }
</style>
</head>
<body>
<canvas id="game"></canvas>
<script>
(function() {
const canvas = document.getElementById('game');
const ctx = canvas.getContext('2d');

let W, H;
function resize() {
W = window.innerWidth;
H = window.innerHeight;
canvas.width = W;
canvas.height = H;
}
window.addEventListener('resize', resize);
resize();

// Audio
let audioCtx = null;
function initAudio() {
if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
}
function playSound(type) {
if (!audioCtx) return;
const osc = audioCtx.createOscillator();
const gain = audioCtx.createGain();
osc.connect(gain);
gain.connect(audioCtx.destination);
const now = audioCtx.currentTime;
switch(type) {
case 'paddle':
osc.type = 'square'; osc.frequency.setValueAtTime(220, now);
osc.frequency.exponentialRampToValueAtTime(180, now + 0.08);
gain.gain.setValueAtTime(0.15, now); gain.gain.exponentialRampToValueAtTime(0.001, now + 0.1);
osc.start(now); osc.stop(now + 0.1); break;
case 'brick':
osc.type = 'square'; osc.frequency.setValueAtTime(440, now);
osc.frequency.exponentialRampToValueAtTime(880, now + 0.05);
gain.gain.setValueAtTime(0.2, now); gain.gain.exponentialRampToValueAtTime(0.001, now + 0.12);
osc.start(now); osc.stop(now + 0.12); break;
case 'wall':
osc.type = 'sine'; osc.frequency.setValueAtTime(300, now);
gain.gain.setValueAtTime(0.08, now); gain.gain.exponentialRampToValueAtTime(0.001, now + 0.05);
osc.start(now); osc.stop(now + 0.05); break;
case 'life':
osc.type = 'sawtooth'; osc.frequency.setValueAtTime(200, now);
osc.frequency.exponentialRampToValueAtTime(60, now + 0.4);
gain.gain.setValueAtTime(0.2, now); gain.gain.exponentialRampToValueAtTime(0.001, now + 0.4);
osc.start(now); osc.stop(now + 0.4); break;
case 'wave':
osc.type = 'sine'; osc.frequency.setValueAtTime(440, now);
osc.frequency.exponentialRampToValueAtTime(880, now + 0.3);
gain.gain.setValueAtTime(0.15, now); gain.gain.exponentialRampToValueAtTime(0.001, now + 0.4);
osc.start(now); osc.stop(now + 0.4); break;
case 'powerup':
osc.type = 'sine'; osc.frequency.setValueAtTime(600, now);
osc.frequency.exponentialRampToValueAtTime(1200, now + 0.15);
gain.gain.setValueAtTime(0.15, now); gain.gain.exponentialRampToValueAtTime(0.001, now + 0.2);
osc.start(now); osc.stop(now + 0.2); break;
}
}

// Game state
let state = 'start'; // start, playing, gameover, paused
let score = 0;
let lives = 3;
let wave = 1;
let highScore = 0;
try { highScore = parseInt(localStorage.getItem('breakoutEndlessHigh') || '0'); } catch(e) {}

// Paddle
let paddle = { w: 120, h: 16, x: 0, y: 0 };
let targetPaddleW = 120;
let wideTimer = 0;

// Ball
let ball = { x: 0, y: 0, r: 8, vx: 0, vy: 0, speed: 5 };
let balls = [];
let stuckBall = true;

// Bricks
let bricks = [];
const BRICK_H = 24;
const BRICK_GAP = 3;
const SIDE_PAD = 20;

// Particles
let particles = [];

// Power-ups
let powerups = [];
const POWERUP_TYPES = ['wide', 'multi', 'slow'];

function getBrickCols() {
return Math.floor((W - SIDE_PAD * 2) / (60 + BRICK_GAP));
}

function spawnWave(w) {
bricks = [];
const rows = Math.min(5 + w, 12);
const cols = getBrickCols();
const brickW = (W - SIDE_PAD * 2) / cols;
for (let r = 0; r < rows; r++) {
for (let c = 0; c < cols; c++) {
const hp = 1 + Math.floor(w / 3);
if (w > 5 && (r + c) % 3 === 0) hp += 1;
bricks.push({ x: SIDE_PAD + c * (brickW + BRICK_GAP), y: 40 + r * (BRICK_H + BRICK_GAP), w: brickW, h: BRICK_H, hp: hp, maxHp: hp });
}
}
ball.speed = 5 + w * 0.4;
paddle.w = targetPaddleW;
wideTimer = 0;
balls = [];
stuckBall = true;
resetBall();
}

function resetBall() {
const b = { x: W / 2, y: H - 60, r: 8, vx: 0, vy: 0, speed: ball.speed };
balls = [b];
stuckBall = true;
}

function launchBall() {
if (!stuckBall) return;
const b = balls[0];
const angle = -Math.PI / 2 + (Math.random() - 0.5) * 0.6;
b.vx = Math.cos(angle) * b.speed;
b.vy = Math.sin(angle) * b.speed;
stuckBall = false;
}

function spawnParticles(x, y, color, count) {
for (let i = 0; i < count; i++) {
const a = Math.random() * Math.PI * 2;
const s = 1 + Math.random() * 3;
particles.push({ x, y, vx: Math.cos(a) * s, vy: Math.sin(a) * s - 1, life: 1, color });
}
}

function spawnPowerup(x, y) {
if (Math.random() < 0.25) {
const type = POWERUP_TYPES[Math.floor(Math.random() * POWERUP_TYPES.length)];
powerups.push({ x, y, vy: 2, type, r: 12 });
}
}

function updatePowerup(pu) {
switch(pu.type) {
case 'wide': targetPaddleW = Math.min(180, paddle.w + 30); wideTimer = 600; break;
case 'multi': {
const b = balls[0];
if (b && balls.length < 5) {
for (let i = 0; i < 2; i++) {
const a = Math.random() * Math.PI * 2;
balls.push({ x: b.x, y: b.y, r: 8, vx: Math.cos(a) * b.speed * 0.7, vy: -Math.abs(b.vy) * 0.5, speed: b.speed });
}
} break; }
case 'slow': balls.forEach(b => { b.vx *= 0.8; b.vy *= 0.8; b.speed *= 0.85; }); break;
}
}

function getBrickColor(hp) {
const colors = ['#ff4444', '#ff8844', '#ffcc44', '#44ff44', '#44ccff', '#cc44ff'];
return colors[Math.min(hp - 1, colors.length - 1)];
}

// Input
let mouseX = W / 2;
let keysDown = {};

window.addEventListener('mousemove', e => { mouseX = e.clientX; });
window.addEventListener('touchmove', e => { if (e.touches.length) mouseX = e.touches[0].clientX; }, { passive: true });
window.addEventListener('touchstart', e => { initAudio(); if (state === 'playing' && stuckBall) launchBall(); else if (state === 'start') startGame(); else if (state === 'gameover') restart(); mouseX = e.touches[0].clientX; });

window.addEventListener('keydown', e => {
initAudio();
keysDown[e.code] = true;
if (e.code === 'Space' || e.code === 'ArrowUp') {
if (state === 'start') startGame();
else if (state === 'playing' && stuckBall) launchBall();
else if (state === 'gameover') restart();
}
if (e.code === 'KeyR' && state === 'gameover') restart();
});
window.addEventListener('keyup', e => { keysDown[e.code] = false; });

canvas.addEventListener('click', () => {
initAudio();
if (state === 'start') startGame();
else if (state === 'playing' && stuckBall) launchBall();
else if (state === 'gameover') restart();
});

function startGame() {
score = 0; lives = 3; wave = 1;
targetPaddleW = 120; wideTimer = 0;
paddle.w = 120; paddle.y = H - 50; paddle.x = W / 2 - 60;
spawnWave(1);
state = 'playing';
}

function restart() {
startGame();
}

function loseLife() {
lives--;
playSound('life');
if (lives <= 0) {
state = 'gameover';
try { if (score > highScore) { highScore = score; localStorage.setItem('breakoutEndlessHigh', String(highScore)); } } catch(e) {}
} else {
resetBall();
}
}

function nextWave() {
wave++;
playSound('wave');
spawnWave(wave);
}

// Update
function update() {
if (state !== 'playing') return;

// Paddle
const pSpeed = 8;
if (keysDown['ArrowLeft']) paddle.x -= pSpeed;
if (keysDown['ArrowRight']) paddle.x += pSpeed;
paddle.x += (mouseX - paddle.x - paddle.w / 2) * 0.3;
paddle.x = Math.max(0, Math.min(W - paddle.w, paddle.x));

// Wide timer
if (wideTimer > 0) { wideTimer--; if (wideTimer <= 0) targetPaddleW = 120; }
paddle.w += (targetPaddleW - paddle.w) * 0.05;

// Ball
for (let i = balls.length - 1; i >= 0; i--) {
const b = balls[i];
if (stuckBall) { b.x = paddle.x + paddle.w / 2; b.y = paddle.y - b.r - 2; continue; }

b.x += b.vx;
b.y += b.vy;

// Walls
if (b.x - b.r <= 0) { b.x = b.r; b.vx = Math.abs(b.vx); playSound('wall'); }
if (b.x + b.r >= W) { b.x = W - b.r; b.vx = -Math.abs(b.vx); playSound('wall'); }
if (b.y - b.r <= 0) { b.y = b.r; b.vy = Math.abs(b.vy); playSound('wall'); }

// Paddle collision
if (b.vy > 0 && b.y + b.r >= paddle.y && b.y + b.r <= paddle.y + paddle.h + 5 && b.x >= paddle.x - b.r && b.x <= paddle.x + paddle.w + b.r) {
const hitPos = (b.x - paddle.x) / paddle.w;
const angle = (hitPos - 0.5) * Math.PI * 0.7;
const speed = Math.sqrt(b.vx * b.vx + b.vy * b.vy);
b.vx = Math.sin(angle) * speed;
b.vy = -Math.cos(angle) * speed;
b.y = paddle.y - b.r;
playSound('paddle');
}

// Brick collision
for (let j = bricks.length - 1; j >= 0; j--) {
const br = bricks[j];
if (b.x + b.r > br.x && b.x - b.r < br.x + br.w && b.y + b.r > br.y && b.y - b.r < br.y + br.h) {
// Determine side
const overlapL = b.x + b.r - br.x;
const overlapR = br.x + br.w - (b.x - b.r);
const overlapT = b.y + b.r - br.y;
const overlapB = br.y + br.h - (b.y - b.r);
const minOverlap = Math.min(overlapL, overlapR, overlapT, overlapB);

if (minOverlap === overlapT || minOverlap === overlapB) { b.vy = -b.vy; }
else { b.vx = -b.vx; }

br.hp--;
playSound('brick');
spawnParticles(b.x, b.y, getBrickColor(br.hp), 5);

if (br.hp <= 0) {
score += 10 * br.maxHp + wave * 2;
spawnPowerup(br.x + br.w / 2, br.y + br.h / 2);
bricks.splice(j, 1);
}

// Check wave clear
if (bricks.length === 0) { nextWave(); break; }
break;
}
}

// Remove ball if off screen bottom
const dead = balls.filter(b => b.y > H + 20).length;
if (dead > 0) {
balls = balls.filter(b => b.y <= H + 20);
loseLife();
if (state !== 'playing') return;
}

// Powerups
for (let i = powerups.length - 1; i >= 0; i--) {
const pu = powerups[i];
pu.y += pu.vy;
// Paddle catch
if (pu.y + pu.r > paddle.y && pu.y - pu.r < paddle.y + paddle.h && pu.x > paddle.x && pu.x < paddle.x + paddle.w) {
updatePowerup(pu);
playSound('powerup');
powerups.splice(i, 1);
continue;
}
if (pu.y > H + 20) powerups.splice(i, 1);
}

// Particles
for (let i = particles.length - 1; i >= 0; i--) {
const p = particles[i];
p.x += p.vx; p.y += p.vy;
p.vy += 0.15;
p.life -= 0.02;
if (p.life <= 0) particles.splice(i, 1);
}
}

// Draw
function draw() {
ctx.fillStyle = '#0a0a1a';
ctx.fillRect(0, 0, W, H);

// Stars background
for (let i = 0; i < 50; i++) {
const sx = (i * 137.5) % W;
const sy = (i * 97.3) % (H - 100);
ctx.fillStyle = `rgba(255,255,255,${0.3 + (i % 5) * 0.1})`;
ctx.fillRect(sx, sy, 1.5, 1.5);
}

if (state === 'start') {
drawStartScreen();
return;
}

// Bricks
for (const br of bricks) {
const color = getBrickColor(br.hp);
ctx.fillStyle = color;
ctx.fillRect(br.x, br.y, br.w, br.h);
ctx.fillStyle = 'rgba(255,255,255,0.15)';
ctx.fillRect(br.x, br.y, br.w, 3);
}

// Powerups
for (const pu of powerups) {
ctx.beginPath();
ctx.arc(pu.x, pu.y, pu.r, 0, Math.PI * 2);
ctx.fillStyle = pu.type === 'wide' ? '#44ff44' : pu.type === 'multi' ? '#44ccff' : '#ffcc44';
ctx.fill();
ctx.fillStyle = '#fff';
ctx.font = 'bold 10px monospace';
ctx.textAlign = 'center';
const label = pu.type === 'wide' ? 'W' : pu.type === 'multi' ? 'M' : 'S';
ctx.fillText(label, pu.x, pu.y + 4);
}

// Paddle
ctx.fillStyle = '#ffffff';
ctx.shadowColor = '#44ccff';
ctx.shadowBlur = 10;
ctx.fillRect(paddle.x, paddle.y, paddle.w, paddle.h);
ctx.shadowBlur = 0;

// Balls
for (const b of balls) {
ctx.beginPath();
ctx.arc(b.x, b.y, b.r, 0, Math.PI * 2);
ctx.fillStyle = '#ffffff';
ctx.shadowColor = '#44ccff';
ctx.shadowBlur = 8;
ctx.fill();
ctx.shadowBlur = 0;
}

// Particles
for (const p of particles) {
ctx.globalAlpha = p.life;
ctx.fillStyle = p.color;
ctx.fillRect(p.x - 2, p.y - 2, 4, 4);
ctx.globalAlpha = 1;
}

// HUD
drawHUD();

if (state === 'gameover') drawGameOver();
}

function drawHUD() {
ctx.fillStyle = '#fff';
ctx.font = 'bold 16px monospace';
ctx.textAlign = 'left';
ctx.fillText(`SCORE: ${score}`, 20, 25);
ctx.textAlign = 'center';
ctx.fillText(`WAVE: ${wave}`, W / 2, 25);
ctx.textAlign = 'right';
ctx.fillText(`LIVES: ${'❤'.repeat(lives)}`, W - 20, 25);
}

function drawStartScreen() {
ctx.fillStyle = '#fff';
ctx.textAlign = 'center';
ctx.font = 'bold 48px monospace';
ctx.shadowColor = '#44ccff'; ctx.shadowBlur = 20;
ctx.fillText('BREAKOUT', W / 2, H / 3);
ctx.font = 'bold 36px monospace';
ctx.fillStyle = '#44ccff';
ctx.fillText('ENDLESS', W / 2, H / 3 + 50);
ctx.shadowBlur = 0;

ctx.font = '18px monospace';
ctx.fillStyle = '#aaa';
const lines = [
'Move: Mouse / Arrow Keys',
'Launch: Click / Space',
'',
'Clear all bricks to advance waves!',
'Difficulty increases each wave.',
'',
'Power-ups: W=Wide  M=Multi  S=Slow'
];
lines.forEach((line, i) => ctx.fillText(line, W / 2, H / 2 + 30 + i * 28));

if (highScore > 0) {
ctx.fillStyle = '#ffcc44';
ctx.font = '16px monospace';
ctx.fillText(`HIGH SCORE: ${highScore}`, W / 2, H - 60);
}

ctx.fillStyle = '#fff';
ctx.font = 'bold 22px monospace';
ctx.fillText('Click or Press SPACE to Start', W / 2, H - 30);
}

function drawGameOver() {
ctx.fillStyle = 'rgba(0,0,0,0.7)';
ctx.fillRect(0, 0, W, H);
ctx.fillStyle = '#ff4444';
ctx.textAlign = 'center';
ctx.font = 'bold 48px monospace';
ctx.fillText('GAME OVER', W / 2, H / 3);
ctx.fillStyle = '#fff';
ctx.font = '24px monospace';
ctx.fillText(`Final Score: ${score}`, W / 2, H / 3 + 50);
ctx.fillText(`Wave Reached: ${wave}`, W / 2, H / 3 + 85);
if (highScore > 0) {
ctx.fillStyle = '#ffcc44';
ctx.font = '18px monospace';
ctx.fillText(`High Score: ${highScore}`, W / 2, H / 3 + 120);
}
ctx.fillStyle = '#aaa';
ctx.font = '20px monospace';
ctx.fillText('Click or Press R to Restart', W / 2, H - 40);
}

// Game loop
let lastTime = 0;
function gameLoop(time) {
const dt = time - lastTime;
lastTime = time;
update();
draw();
requestAnimationFrame(gameLoop);
}
requestAnimationFrame(gameLoop);
})();
</script>
</body>
```
