<script>
const canvas = document.getElementById('canvas');
const ctx = canvas.getContext('2d');
const scoreDisplay = document.getElementById('score-display');
const turnIndicator = document.getElementById('turn-indicator');
const startScreen = document.getElementById('start-screen');
const gameOverScreen = document.getElementById('game-over-screen');
const winnerText = document.getElementById('winner-text');
const finalScore = document.getElementById('final-score');

const COLS = 15;
const ROWS = 9;
const PADDLE_HEIGHT = 3;
const WIN_SCORE = 5;

let cellSize, offsetX, offsetY;
let state = 'start'; // start, playing, animating, gameover
let scores = [0, 0];
let ball = { x: 7, y: 4, dx: 1, dy: 0 };
let paddles = [{ y: 3 }, { y: 3 }]; // stores row index of paddle center
let activePlayer = 0; // 0 or 1, whose side the ball is heading toward
let turnPhase = 'player'; // player's turn to move paddle, then ball moves
let animTimer = null;
let audioCtx = null;

function resize() {
  const container = document.getElementById('game-container');
  const maxW = Math.min(container.clientWidth - 20, 800);
  const maxH = Math.min(container.clientHeight - 140, 560);
  cellSize = Math.floor(Math.min(maxW / COLS, maxH / ROWS));
  canvas.width = cellSize * COLS;
  canvas.height = cellSize * ROWS;
  offsetX = (canvas.width) / 2 - (cellSize * COLS) / 2;
  offsetY = (canvas.height) / 2 - (cellSize * ROWS) / 2;
  draw();
}

function initAudio() {
  if (!audioCtx) {
    audioCtx = new (window.AudioContext || window.webkitAudioContext)();
  }
}

function playSound(freq, duration, type) {
  if (!audioCtx) return;
  const osc = audioCtx.createOscillator();
  const gain = audioCtx.createGain();
  osc.type = type || 'square';
  osc.frequency.value = freq;
  gain.gain.setValueAtTime(0.15, audioCtx.currentTime);
  gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + duration);
  osc.connect(gain);
  gain.connect(audioCtx.destination);
  osc.start();
  osc.stop(audioCtx.currentTime + duration);
}

function draw() {
  ctx.fillStyle = '#0a0a1a';
  ctx.fillRect(0, 0, canvas.width, canvas.height);
  
  // Draw grid
  ctx.strokeStyle = 'rgba(50,60,80,0.3)';
  ctx.lineWidth = 1;
  for (let i = 0; i <= COLS; i++) {
    ctx.beginPath();
    ctx.moveTo(i * cellSize, 0);
    ctx.lineTo(i * cellSize, canvas.height);
    ctx.stroke();
  }
  for (let j = 0; j <= ROWS; j++) {
    ctx.beginPath();
    ctx.moveTo(0, j * cellSize);
    ctx.lineTo(canvas.width, j * cellSize);
    ctx.stroke();
  }
  
  // Center line
  ctx.strokeStyle = 'rgba(100,150,255,0.3)';
  ctx.setLineDash([cellSize * 0.3, cellSize * 0.3]);
  ctx.beginPath();
    ctx.moveTo(COLS / 2 * cellSize, 0);
    ctx.lineTo(COLS / 2 * cellSize, canvas.height);
    ctx.stroke();
  ctx.setLineDash([]);
  
  if (state === 'playing' || state === 'animating') {
    // Draw paddles
    for (let p = 0; p < 2; p++) {
      const col = p === 0 ? 1 : COLS - 2;
      ctx.fillStyle = p === 0 ? '#4af' : '#f66';
      for (let r = 0; r < PADDLE_HEIGHT; r++) {
        const row = paddles[p].y - Math.floor(PADDLE_HEIGHT / 2) + r;
        if (row >= 0 && row < ROWS) {
          ctx.fillRect(col * cellSize + 3, row * cellSize + 3, cellSize - 6, cellSize - 6);
        }
      }
    }
    
    // Draw ball
    ctx.fillStyle = '#ff0';
    ctx.shadowColor = '#ff0';
    ctx.shadowBlur = 10;
    ctx.fillRect(ball.x * cellSize + 3, ball.y * cellSize + 3, cellSize - 6, cellSize - 6);
    ctx.shadowBlur = 0;
  }
}

function startGame() {
  scores = [0, 0];
  paddles = [{ y: Math.floor(ROWS / 2) }, { y: Math.floor(ROWS / 2) }];
  resetBall();
  state = 'playing';
  turnPhase = 'player';
  startScreen.classList.add('hidden');
  gameOverScreen.classList.add('hidden');
  updateUI();
  draw();
}

function resetBall() {
  ball.x = Math.floor(COLS / 2);
  ball.y = Math.floor(ROWS / 2);
  const dir = Math.random() > 0.5 ? 1 : -1;
  const dyOpts = [-1, 0, 1];
  ball.dx = dir;
  ball.dy = dyOpts[Math.floor(Math.random() * 3)];
  if (ball.dy === 0) ball.dy = Math.random() > 0.5 ? 1 : -1;
  activePlayer = ball.dx > 0 ? 1 : 0;
}

function updateUI() {
  scoreDisplay.textContent = `P1: ${scores[0]}  |  P2: ${scores[1]}`;
  if (state === 'playing') {
    const names = ['PLAYER 1', 'PLAYER 2'];
    const colors = ['#4af', '#f66'];
    turnIndicator.innerHTML = `<span style="color:${colors[activePlayer]}">${names[activePlayer]}'S TURN</span><br><small>Move: ${activePlayer === 0 ? 'W/S' : '↑/↓'} | Stay: ${activePlayer === 0 ? 'Space' : 'Enter'}</small>`;
  } else {
    turnIndicator.textContent = '';
  }
}

function movePaddle(player, dir) {
  const newY = paddles[player].y + (dir === 'up' ? -1 : 1);
  if (newY >= Math.floor(PADDLE_HEIGHT / 2) && newY <= ROWS - 1 - Math.floor(PADVLE_HEIGHT / 2)) {
    // bounds check: paddle center must allow all cells to be in range
    const top = newY - Math.floor(PADDLE_HEIGHT / 2);
    const bottom = top + PADDLE_HEIGHT - 1;
    if (top >= 0 && bottom < ROWS) {
      paddles[player].y = newY;
      playSound(200, 0.05, 'sine');
    }
  }
}

function advanceBall() {
  ball.x += ball.dx;
  
  // Bounce off top/bottom
  if (ball.y + ball.dy < 0 || ball.y + ball.dy >= ROWS) {
    ball.dy = -ball.dy;
  }
  ball.y += ball.dy;
  
  // Check paddle hit or score
  const targetPlayer = ball.dx > 0 ? 1 : 0; // player whose side the ball is heading to
  const targetCol = targetPlayer === 0 ? 1 : COLS - 2;
  
  if (ball.x <= 1 || ball.x >= COLS - 2) {
    if (ball.dx < 0 && ball.x <= 1) {
      // Ball reached left paddle
      const paddleRow = paddles[0].y;
      const top = paddleRow - Math.floor(PADDLE_HEIGHT / 2);
      const bottom = top + PADDLE_HEIGHT - 1;
      if (ball.y >= top && ball.y <= bottom) {
        // Bounce!
        ball.dx = 1;
        playSound(400, 0.1, 'square');
        activePlayer = 1;
      } else {
        // Score for player 2
        scores[1]++;
        playSound(150, 0.3, 'sawtooth');
        checkWin();
        return;
      }
    } else if (ball.dx > 0 && ball.x >= COLS - 2) {
      // Ball reached right paddle
      const paddleRow = paddles[1].y;
      const top = paddleRow - Math.floor(PADDLE_HEIGHT / 2);
      const bottom = top + PADDLE_HEIGHT - 1;
      if (ball.y >= top && ball.y <= bottom) {
        ball.dx = -1;
        playSound(400, 0.1, 'square');
        activePlayer = 0;
      } else {
        scores[0]++;
        playSound(150, 0.3, 'sawtooth');
        checkWin();
        return;
      }
    }
  }
  
  // Determine whose turn it is (the player the ball is heading toward)
  activePlayer = ball.dx > 0 ? 1 : 0;
  state = 'playing';
  updateUI();
  draw();
}

function checkWin() {
  if (scores[0] >= WIN_SCORE || scores[1] >= WIN_SCORE) {
    state = 'gameover';
    const winner = scores[0] >= WIN_SCORE ? 'PLAYER 1' : 'PLAYER 2';
    winnerText.textContent = `${winner} WINS!`;
    finalScore.textContent = `Final Score: ${scores[0]} - ${scores[1]}`;
    gameOverScreen.classList.remove('hidden');
    playSound(300, 0.5, 'sine');
    setTimeout(() => playSound(400, 0.5, 'sine'), 200);
    setTimeout(() => playSound(500, 0.8, 'sine'), 400);
  } else {
    resetBall();
    state = 'playing';
  }
  updateUI();
  draw();
}

function handleInput(player, action) {
  if (state !== 'playing' || activePlayer !== player) return;
  
  if (action === 'up') movePaddle(player, 'up');
  else if (action === 'down') movePaddle(player, 'down');
  else if (action === 'stay') { /* no movement */ }
  
  // After player acts, advance ball
  state = 'animating';
  turnIndicator.textContent = 'Ball moving...';
  
  setTimeout(() => {
    advanceBall();
  }, 300);
}

window.addEventListener('keydown', (e) => {
  if (state === 'playing' && turnPhase === 'player') {
    // Player 1: W/S + Space
    if (e.code === 'KeyW') { e.preventDefault(); initAudio(); handleInput(0, 'up'); }
    else if (e.code === 'KeyS') { e.preventDefault(); initAudio(); handleInput(0, 'down'); }
    else if (e.code === 'Space') { e.preventDefault(); initAudio(); handleInput(0, 'stay'); }
    
    // Player 2: Arrows + Enter
    else if (e.code === 'ArrowUp') { e.preventDefault(); initAudio(); handleInput(1, 'up'); }
    else if (e.code === 'ArrowDown') { e.preventDefault(); initAudio(); handleInput(1, 'down'); }
    else if (e.code === 'Enter') { e.preventDefault(); initAudio(); handleInput(1, 'stay'); }
  }
  
  // Start / Restart
  if (state === 'start' && (e.code === 'Space' || e.code === 'Enter')) {
    initAudio();
    startGame();
  }
  if (state === 'gameover' && (e.code === 'Space' || e.code === 'Enter')) {
    startGame();
  }
});

document.getElementById('start-btn').addEventListener('click', () => {
  initAudio();
  startGame();
});

document.getElementById('restart-btn').addEventListener('click', () => {
  startGame();
});

window.addEventListener('resize', resize);
resize();
draw();
</script>
