(function(){
let audioCtx = null;
function initAudio(){ if(!audioCtx) audioCtx = new (window.AudioContext||webkitAudioContext)(); }
function playClick(){
  if(!audioCtx) return;
  let o = audioCtx.createOscillator(), g = audioCtx.createGain();
  o.type='sine'; o.frequency.value=400; g.gain.setValueAtTime(0.15,audioCtx.currentTime);
  g.gain.exponentialRampToValueAtTime(0.001,audioCtx.currentTime+0.1);
  o.connect(g); g.connect(audioCtx.destination); o.start(); o.stop(audioCtx.currentTime+0.1);
}
function playWin(){
  if(!audioCtx) return;
  let notes=[523,659,784,1047];
  notes.forEach((f,i)=>{
    let o=audioCtx.createOscillator(),g=audioCtx.createGain();
    o.type='sine';o.frequency.value=f;
    g.gain.setValueAtTime(0.2,audioCtx.currentTime+i*0.15);
    g.gain.exponentialRampToValueAtTime(0.001,audioCtx.currentTime+i*0.15+0.3);
    o.connect(g);g.connect(audioCtx.destination);
    o.start(audioCtx.currentTime+i*0.15);o.stop(audioCtx.currentTime+i*0.15+0.3);
  });
}
function playError(){
  if(!audioCtx) return;

let board=[], gridSize=3, emptyPos=null, moves=0, timerInterval=null, seconds=0;
let level=1, gameActive=false, solved=false;
const totalCards = ()=> gridSize*gridSize - 1;

function getCardColor(n){
  const h = (n*360/totalCards())%360;
  return `hsl(${h},55%,42%)`;
}

function createSolvedBoard(){
  let b=[]; for(let i=1;i<totalCards();i++) b.push(i); b.push(0); return b;
}

function shuffleBoard(b){
  // Perform valid random moves to ensure solvability
  let e = b.indexOf(0);
  for(let i=0;i<200*gridSize;i++){
    let neighbors=getNeighbors(e);
    let pick=neighbors[Math.floor(Math.random()*neighbors.length)];
    b[e]=b[pick]; b[pick]=0; e=pick;
  }
  // Check not already solved
  if(isSolved(b)){ shuffleBoard(b); return; }
}

function getNeighbors(pos){
  let r=Math.floor(pos/gridSize), c=pos%gridSize, n=[];
  if(r>0) n.push(pos-gridSize);
  if(r<gridSize-1) n.push(pos+gridSize);
  if(c>0) n.push(pos-1);
  if(c<gridSize-1) n.push(pos+1);
  return n;
}

function isSolved(b){
  for(let i=0;i<b.length-1;i++){ if(b[i]!==i+1) return false; }
  return b[b.length-1]===0;
}

function render(){
  const boardEl=document.getElementById('board');
  const maxDim = Math.min(window.innerWidth - 40, window.innerHeight - 140);
  const cellSize = Math.floor((maxDim - (gridSize-1)*6) / gridSize);
  boardEl.style.gridTemplateColumns=`repeat(${gridSize},${cellSize}px)`;
  boardEl.innerHTML='';
  for(let i=0;i<board.length;i++){
    let card=document.createElement('div');
    card.className='card'+(board[i]===0?' empty':'');
    card.style.width=cellSize+'px'; card.style.height=cellSize+'px';
    if(board[i]!==0){
      card.textContent=board[i];
      card.style.background=getCardColor(board[i]);
      card.style.fontSize=Math.floor(cellSize*0.35)+'px';
      card.style.boxShadow=`0 4px 12px rgba(0,0,0,0.4), inset 0 1px 0 rgba(255,255,255,0.2)`;
    }
    card.dataset.pos=i;
    boardEl.appendChild(card);
  }
  document.getElementById('moveCount').textContent=moves;
  document.getElementById('levelDisplay').textContent=level;
}

function startTimer(){
  clearInterval(timerInterval); seconds=0;
  updateTimer();
  timerInterval=setInterval(()=>{ seconds++; updateTimer(); },1000);
}
function stopTimer(){ clearInterval(timerInterval); }
function updateTimer(){
  let m=Math.floor(seconds/60), s=seconds%60;
  document.getElementById('timeCount').textContent=m+':'+(s<10?'0':'')+s;
}

function startGame(size){
  initAudio();
  gridSize=size; level=size-2;
  board=createSolvedBoard(); shuffleBoard(board);
  emptyPos=board.indexOf(0); moves=0; solved=false; gameActive=true;
  document.getElementById('overlay').classList.add('hidden');
  render(); startTimer();
}

function tryMove(pos){
  if(!gameActive||solved) return;
  let r=Math.floor(pos/gridSize), c=pos%gridSize;
  let er=Math.floor(emptyPos/gridSize), ec=emptyPos%gridSize;
  if(Math.abs(r-er)+Math.abs(c-ec)!==1){ playError(); return; }
  board[emptyPos]=board[pos]; board[pos]=0; emptyPos=pos; moves++;
  playClick(); render(); checkWin();
}

function checkWin(){
  if(isSolved(board)){
    solved=true; gameActive=false; stopTimer(); playWin();
    setTimeout(showWin,400);
  }
}

  document.getElementById('nextBtn').onclick=()=>startGame(nextLevel);

}

// Confetti
let confettiParts=[];
function launchConfetti(){
  const canvas=document.getElementById('confetti-canvas');
  canvas.width=window.innerWidth; canvas.height=window.innerHeight;
  const ctx=canvas.getContext('2d');
  confettiParts=[];
  for(let i=0;i<120;i++){
    confettiParts.push({x:Math.random()*canvas.width,y:-20,vx:(Math.random()-0.5)*4,vy:Math.random()*3+2,
      color:`hsl(${Math.random()*360},80%,60%)`,size:Math.random()*8+4,rot:Math.random()*Math.PI*2,rv:(Math.random()-0.5)*0.2});
  }
  animateConfetti(ctx,canvas);
}
function animateConfetti(ctx,canvas){
  ctx.clearRect(0,0,canvas.width,canvas.height);
  let alive=false;
  confettiParts.forEach(p=>{
    p.x+=p.vx;p.y+=p.vy;p.vy+=0.1;p.rot+=p.rv;
    if(p.y<canvas.height+20){alive=true;}
    ctx.save();ctx.translate(p.x,p.y);ctx.rotate(p.rot);
    ctx.fillStyle=p.color;ctx.fillRect(-p.size/2,-p.size/2,p.size,p.size*0.6);ctx.restore();
  });
  if(alive) requestAnimationFrame(()=>animateConfetti(ctx,canvas));
  else ctx.clearRect(0,0,canvas.width,canvas.height);
}

  else return;
  let nr=er+dr, nc=ec+dc;

});
