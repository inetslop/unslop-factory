<script>
const canvas = document.getElementById('c');
const ctx = canvas.getContext('2d');

let W, H;
function resize() {
    W = window.innerWidth;
    H = window.innerHeight;
    canvas.width = W;
    canvas.height = H;
}
resize();
window.addEventListener('resize', resize);

// Audio
let audioCtx = null;
function initAudio() {
    if (!audioCtx) {
        audioCtx = new (window.AudioContext || window.webkitAudioContext)();
    }
}

function playLaser() {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = 'sawtooth';
    osc.frequency.setValueAtTime(880, audioCtx.currentTime);
    osc.frequency.exponentialRampToValueAtTime(220, audioCtx.currentTime + 0.1);
    gain.gain.setValueAtTime(0.3, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.15);
    osc.connect(gain).connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + 0.15);
}

function playExplosion(size) {
    if (!audioCtx) return;
    const bufferSize = audioCtx.sampleRate * 0.3;
    const buffer = audioCtx.createBuffer(1, bufferSize, audioCtx.sampleRate);
    const data = buffer.getChannelData(0);
    for (let i = 0; i < bufferSize; i++) {
        data[i] = (Math.random() * 2 - 1) * Math.pow(1 - i / bufferSize, size === 3 ? 2 : 3);
    }
    const source = audioCtx.createBufferSource();
    const gain = audioCtx.createGain();
    gain.gain.setValueAtTime(0.4, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.3);
    source.buffer = buffer;
    source.connect(gain).connect(audioCtx.destination);
    source.start();
}

function playThrust() {
    if (!audioCtx) return;
    const bufferSize = audioCtx.sampleRate * 0.05;
    const buffer = audioCtx.createBuffer(1, bufferSize, audioCtx.sampleRate);
    const data = buffer.getChannelData(0);
    for (let i = 0; i < bufferSize; i++) {
        data[i] = (Math.random() * 2 - 1) * 0.5;
    }
    const source = audioCtx.createBufferSource();
    const gain = audioCtx.createGain();
    gain.gain.setValueAtTime(0.08, audioCtx.currentTime);
    source.buffer = buffer;
    source.connect(gain).connect(audioCtx.destination);
    source.start();
}

// Game state
let gameState = 'start'; // start, playing, gameover
let score = 0;
let lives = 3;
let highScore = 0;
try {
    highScore = parseInt(localStorage.getItem('asteroids_high') || '0');
} catch(e) {}

// Ship
let ship = {};
let bullets = [];
let asteroids = [];
let particles = [];
let thrustParticles = [];

const keys = {};
window.addEventListener('keydown', e => {
    keys[e.code] = true;
    if (e.code === 'Space') e.preventDefault();
    if (e.code === 'Enter' || e.code === 'Space') {
        if (gameState === 'start') startGame();
        else if (gameState === 'gameover' && e.code === 'Enter') startGame();
    }
});
window.addEventListener('keyup', e => { keys[e.code] = false; });

function resetShip() {
    ship = { x: W / 2, y: H / 2, angle: -Math.PI / 2, vx: 0, vy: 0 };
}

function startGame() {
    initAudio();
    gameState = 'playing';
    score = 0;
    lives = 3;
    bullets = [];
    asteroids = [];
    particles = [];
    thrustParticles = [];
    resetShip();
    spawnAsteroid(3);
    spawnAsteroid(3);
    spawnAsteroid(2);
}

function spawnAsteroid(size, x, y) {
    let ax, ay;
    if (x !== undefined) {
        ax = x; ay = y;
    } else {
        const side = Math.floor(Math.random() * 4);
        if (side === 0) { ax = -20; ay = Math.random() * H; }
        else if (side === 1) { ax = W + 20; ay = Math.random() * H; }
        else if (side === 2) { ax = Math.random() * W; ay = -20; }
        else { ax = Math.random() * W; ay = H + 20; }
    }
    const speed = (4 - size) * 0.5 + Math.random() * 1;
    const angle = Math.random() * Math.PI * 2;
    const radius = size === 3 ? 35 : size === 2 ? 20 : 10;
    
    // Generate jagged shape
    const verts = [];
    const numVerts = 8 + Math.floor(Math.random() * 4);
    for (let i = 0; i < numVerts; i++) {
        const a = (i / numVerts) * Math.PI * 2;
        const r = radius * (0.7 + Math.random() * 0.3);
        verts.push({ x: Math.cos(a) * r, y: Math.sin(a) * r });
    }

    asteroids.push({
        x: ax, y: ay, size, radius,
        vx: Math.cos(angle) * speed, vy: Math.sin(angle) * speed,
        rot: 0, rotSpeed: (Math.random() - 0.5) * 0.03,
        verts
    });
}

function spawnParticles(x, y, count, color) {
    for (let i = 0; i < count; i++) {
        const a = Math.random() * Math.PI * 2;
        const s = Math.random() * 3 + 1;
        particles.push({ x, y, vx: Math.cos(a) * s, vy: Math.sin(a) * s, life: 40 + Math.random() * 20, color: color || '#fff' });
    }
}

function wrap(obj) {
    const r = obj.radius || 15;
    if (obj.x < -r) obj.x = W + r;
    if (obj.x > W + r) obj.x = -r;
    if (obj.y < -r) obj.y = H + r;
    if (obj.y > H + r) obj.y = -r;
}

function dist(a, b) {
    return Math.sqrt((a.x - b.x) ** 2 + (a.y - b.y) ** 2);
}

let fireCooldown = 0;
let invincible = 0;
let waveTimer = 0;

function update() {
    if (gameState !== 'playing') return;

    // Ship controls
    if (keys['ArrowLeft'] || keys['KeyA']) ship.angle -= 0.07;
    if (keys['ArrowRight'] || keys['KeyD']) ship.angle += 0.07;
    if (keys['ArrowUp'] || keys['KeyW']) {
        ship.vx += Math.cos(ship.angle) * 0.15;
        ship.vy += Math.sin(ship.angle) * 0.15;
        if (Math.random() < 0.3) {
            playThrust();
            thrustParticles.push({
                x: ship.x - Math.cos(ship.angle) * 12,
                y: ship.y - Math.sin(ship.angle) * 12,
                life: 15, maxLife: 15
            });
        }
    }

    // Friction
    ship.vx *= 0.98;
    ship.vy *= 0.98;
    ship.x += ship.vx;
    ship.y += ship.vy;
    wrap(ship);

    // Fire
    if (fireCooldown > 0) fireCooldown--;
    if (keys['Space'] && fireCooldown <= 0) {
        bullets.push({ x: ship.x + Math.cos(ship.angle) * 12, y: ship.y + Math.sin(ship.angle) * 12, vx: Math.cos(ship.angle) * 8 + ship.vx, vy: Math.sin(ship.angle) * 8 + ship.vy, life: 60 });
        fireCooldown = 10;
        playLaser();
    }

    // Invincibility timer
    if (invincible > 0) invincible--;

    // Update bullets
    for (let i = bullets.length - 1; i >= 0; i--) {
        const b = bullets[i];
        b.x += b.vx;
        b.y += b.vy;
        b.life--;
        wrap(b);
        if (b.life <= 0) bullets.splice(i, 1);
    }

    // Update asteroids
    for (let i = asteroids.length - 1; i >= 0; i--) {
        const a = asteroids[i];
        a.x += a.vx;
        a.y += a.vy;
        a.rot += a.rotSpeed;
        wrap(a);

        // Bullet collision
        for (let j = bullets.length - 1; j >= 0; j--) {
            const b = bullets[j];
            if (dist(a, b) < a.radius + 3) {
                bullets.splice(j, 1);
                score += a.size === 3 ? 20 : a.size === 2 ? 50 : 100;
                playExplosion(a.size);
                spawnParticles(a.x, a.y, a.size * 5, '#ff8844');
                if (a.size > 1) {
                    for (let k = 0; k < 2; k++) spawnAsteroid(a.size - 1, a.x, a.y);
                }
                asteroids.splice(i, 1);
                break;
            }
        }
    }

    // Ship-asteroid collision
    if (invincible <= 0) {
        for (let i = asteroids.length - 1; i >= 0; i--) {
            const a = asteroids[i];
            if (dist(ship, a) < a.radius + 10) {
                lives--;
                playExplosion(3);
                spawnParticles(ship.x, ship.y, 20, '#44ff88');
                invincible = 120;
                resetShip();
                if (lives <= 0) {
                    gameState = 'gameover';
                    try { localStorage.setItem('asteroids_high', highScore); } catch(e) {}
                }
            }
        }
    }

    // Update particles
    for (let i = particles.length - 1; i >= 0; i--) {
        const p = particles[i];
        p.x += p.vx;
        p.y += p.vy;
        p.life--;
        if (p.life <= 0) particles.splice(i, 1);
    }

    // Update thrust particles
    for (let i = thrustParticles.length - 1; i >= 0; i--) {
        const p = thrustParticles[i];
        p.x -= Math.cos(ship.angle) * 2;
        p.y -= Math.sin(ship.angle) * 2;
        p.life--;
        if (p.life <= 0) thrustParticles.splice(i, 1);
    }

    // Spawn more asteroids over time
    waveTimer++;
    const maxAsteroids = 5 + Math.floor(score / 200);
    if (asteroids.length < maxAsteroids && waveTimer % 300 === 0) {
        spawnAsteroid(3);
    }

    // High score update
    if (score > highScore) highScore = score;
}

function draw() {
    ctx.fillStyle = '#000';
    ctx.fillRect(0, 0, W, H);

    if (gameState === 'start') {
        drawStartScreen();
        return;
    }

    // Stars background
    drawStars();

    // Particles
    for (const p of particles) {
        ctx.globalAlpha = p.life / 60;
        ctx.fillStyle = p.color;
        ctx.fillRect(p.x - 2, p.y - 2, 4, 4);
    }
    ctx.globalAlpha = 1;

    // Thrust particles
    for (const p of thrustParticles) {
        const alpha = p.life / p.maxLife;
        ctx.globalAlpha = alpha;
        ctx.fillStyle = '#ff6600';
        ctx.fillRect(p.x - 2, p.y - 2, 4, 4);
    }
    ctx.globalAlpha = 1;

    // Bullets
    ctx.fillStyle = '#fff';
    for (const b of bullets) {
        ctx.beginPath();
        ctx.arc(b.x, b.y, 3, 0, Math.PI * 2);
        ctx.fill();
    }

    // Asteroids
    ctx.strokeStyle = '#aaa';
    ctx.lineWidth = 2;
    for (const a of asteroids) {
        ctx.save();
        ctx.translate(a.x, a.y);
        ctx.rotate(a.rot);
        ctx.beginPath();
        ctx.moveTo(a.verts[0].x, a.verts[0].y);
        for (let i = 1; i < a.verts.length; i++) {
            ctx.lineTo(a.verts[i].x, a.verts[i].y);
        }
        ctx.closePath();
        ctx.stroke();
        ctx.restore();
    }

    // Ship (blinking when invincible)
    if (invincible <= 0 || Math.floor(invincible / 5) % 2 === 0) {
        ctx.save();
        ctx.translate(ship.x, ship.y);
        ctx.rotate(ship.angle);
        ctx.strokeStyle = '#4f8';
        ctx.lineWidth = 2;
        ctx.beginPath();
        ctx.moveTo(14, 0);
        ctx.lineTo(-10, -9);
        ctx.lineTo(-6, 0);
        ctx.lineTo(-10, 9);
        ctx.closePath();
        ctx.stroke();
        ctx.restore();
    }

    // HUD
    ctx.fillStyle = '#fff';
    ctx.font = '18px monospace';
    ctx.textAlign = 'left';
    ctx.fillText('SCORE: ' + score, 20, 30);
    ctx.fillText('HIGH: ' + highScore, 20, 55);

    // Lives
    for (let i = 0; i < lives; i++) {
        ctx.save();
        ctx.translate(W - 40 - i * 30, 30);
        ctx.rotate(-Math.PI / 2);
        ctx.strokeStyle = '#4f8';
        ctx.lineWidth = 1.5;
        ctx.beginPath();
        ctx.moveTo(10, 0);
        ctx.lineTo(-7, -6);
        ctx.lineTo(-5, 0);
        ctx.lineTo(-7, 6);
        ctx.closePath();
        ctx.stroke();
        ctx.restore();
    }

    if (gameState === 'gameover') {
        drawGameOver();
    }
}

let starField = [];
function initStars() {
    for (let i = 0; i < 100; i++) {
        starField.push({ x: Math.random(), y: Math.random(), s: Math.random() * 2 + 0.5, b: Math.random() });
    }
}
initStars();

function drawStars() {
    ctx.fillStyle = '#fff';
    for (const s of starField) {
        ctx.globalAlpha = 0.3 + s.b * 0.7;
        ctx.fillRect(s.x * W, s.y * H, s.s, s.s);
    }
    ctx.globalAlpha = 1;
}

function drawStartScreen() {
    drawStars();
    ctx.fillStyle = '#fff';
    ctx.font = 'bold 48px monospace';
    ctx.textAlign = 'center';
    ctx.fillText('ASTEROIDS', W / 2, H / 3);

    ctx.font = '18px monospace';
    ctx.fillStyle = '#aaa';
    const lines = [
        '← → or A/D - Rotate',
        '↑ or W - Thrust',
        'SPACE - Fire',
        '',
        'Press ENTER to Start'
    ];
    for (let i = 0; i < lines.length; i++) {
        ctx.fillText(lines[i], W / 2, H / 3 + 60 + i * 28);
    }

    if (highScore > 0) {
        ctx.fillStyle = '#ff0';
        ctx.font = '16px monospace';
        ctx.fillText('HIGH SCORE: ' + highScore, W / 2, H / 3 + 60 + lines.length * 28 + 20);
    }

    // Decorative asteroids
    const t = Date.now() * 0.001;
    for (let i = 0; i < 5; i++) {
        const x = W / 2 + Math.cos(t + i * 1.3) * (100 + i * 40);
        const y = H / 2 + Math.sin(t * 0.7 + i * 2.1) * (80 + i * 30);
        ctx.save();
        ctx.translate(x, y);
        ctx.rotate(t * 0.5 + i);
        ctx.strokeStyle = '#666';
        ctx.lineWidth = 1.5;
        ctx.beginPath();
        for (let v = 0; v < 7; v++) {
            const a = (v / 7) * Math.PI * 2;
            const r = 12 + Math.sin(v * 3.7) * 4;
            if (v === 0) ctx.moveTo(Math.cos(a) * r, Math.sin(a) * r);
            else ctx.lineTo(Math.cos(a) * r, Math.sin(a) * r);
        }
        ctx.closePath();
        ctx.stroke();
        ctx.restore();
    }
}

function drawGameOver() {
    ctx.fillStyle = 'rgba(0,0,0,0.6)';
    ctx.fillRect(0, 0, W, H);
    ctx.fillStyle = '#f44';
    ctx.font = 'bold 42px monospace';
    ctx.textAlign = 'center';
    ctx.fillText('GAME OVER', W / 2, H / 3);

    ctx.fillStyle = '#fff';
    ctx.font = '20px monospace';
    ctx.fillText('Score: ' + score, W / 2, H / 3 + 50);
    ctx.fillText('High: ' + highScore, W / 2, H / 3 + 80);

    ctx.fillStyle = '#aaa';
    ctx.font = '16px monospace';
    ctx.fillText('Press ENTER to Play Again', W / 2, H / 3 + 140);
}

function gameLoop() {
    update();
    draw();
    requestAnimationFrame(gameLoop);
}
gameLoop();
</script>
