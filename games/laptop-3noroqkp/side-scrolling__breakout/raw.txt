```html
<!DOCTYPE html>
<html lang="en">
<head>
<meta charset="UTF-8">
<title>Side-Scroll Breakout</title>
<style>
* { margin: 0; padding: 0; box-sizing: border-box; }
html, body { width: 100%; height: 100%; overflow: hidden; background: #0a0a1a; }
canvas { display: block; width: 100%; height: 100%; }
</style>

<body>
<canvas id="game"></canvas>
<script>
(function() {
const canvas = document.getElementById('game');
const ctx = canvas.getContext('2d');

let W, H;
function resize() {
W = window.innerWidth;
H = window.innerHeight;
canvas.width = W;
canvas.height = H;
}
window.addEventListener('resize', resize);
resize();

// Audio
let audioCtx = null;
function initAudio() {
if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
}
function playSound(freq, dur, type, vol) {
if (!audioCtx) return;
const osc = audioCtx.createOscillator();
const gain = audioCtx.createGain();
osc.type = type || 'square';
osc.frequency.value = freq;
gain.gain.setValueAtTime(vol || 0.15, audioCtx.currentTime);
gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
osc.connect(gain);
gain.connect(audioCtx.destination);
osc.start();
osc.stop(audioCtx.currentTime + dur);
}

// Game constants
const WORLD_WIDTH = 3200;
const PADDLE_H = 18;
const BALL_R = 7;
const BRICK_H = 28;
const BRICK_W = 56;
const BRICK_PAD = 4;
const ROWS = 5;
const COLS = Math.floor((WORLD_WIDTH - 100) / (BRICK_W + BRICK_PAD));

// State
let gameState = 'start'; // start, playing, ballReady, gameOver, win
let score = 0;
let lives = 3;
let ball = { x: 0, y: 0, vx: 0, vy: 0, speed: 6 };
let paddle = { w: 100, x: 0, targetX: 0 };
let cameraX = 0;
let bricks = [];
let particles = [];
let trail = [];
let level = 1;

function resetBall() {
paddle.x = WORLD_WIDTH / 2 - paddle.w / 2;
ball.x = paddle.x + paddle.w / 2;
ball.y = H - PADDLE_H - BALL_R - 4;
ball.vx = 0;
ball.vy = 0;
gameState = 'ballReady';
cameraX = Math.max(0, Math.min(WORLD_WIDTH - W, ball.x - W / 2));
}

function initBricks() {
bricks = [];
const startX = 50;
const startY = 60;
for (let r = 0; r < ROWS; r++) {
for (let c = 0; c < COLS; c++) {
if (Math.random() < 0.12 && level > 1) continue; // some gaps in later levels
const x = startX + c * (BRICK_W + BRICK_PAD);
const y = startY + r * (BRICK_H + BRICK_PAD);
bricks.push({
x: x, y: y, w: BRICK_W, h: BRICK_H,
alive: true,
hp: r < 1 ? 2 : 1,
color: getBrickColor(r)
});
}
}
}

function getBrickColor(row) {
const colors = [
['#ff4444', '#cc2222'],
['#ff8833', '#cc6622'],
['#ffdd33', '#ccaa22'],
['#44dd44', '#22aa22'],
['#4488ff', '#2266cc']
];
return colors[row % colors.length];
}

function startGame() {
score = 0;
lives = 3;
level = 1;
initBricks();
resetBall();
particles = [];
trail = [];
}

function nextLevel() {
level++;
ball.speed += 0.5;
initBricks();
resetBall();
}

// Input
let keys = {};
window.addEventListener('keydown', function(e) {
keys[e.code] = true;
if (gameState === 'start') {
initAudio();
startGame();
} else if (gameState === 'ballReady' && (e.code === 'Space' || e.code === 'ArrowUp')) {
launchBall();
} else if ((e.code === 'KeyR') && (gameState === 'gameOver' || gameState === 'win')) {
startGame();
}
});
window.addEventListener('keyup', function(e) { keys[e.code] = false; });
canvas.addEventListener('click', function() {
initAudio();
if (gameState === 'start') startGame();
else if (gameState === 'ballReady') launchBall();
else if (gameState === 'gameOver' || gameState === 'win') startGame();
});

function launchBall() {
const angle = -Math.PI / 3 + Math.random() * (Math.PI / 3);
ball.vx = Math.cos(angle) * ball.speed;
ball.vy = -Math.sin(angle) * ball.speed;
gameState = 'playing';
}

// Particles
function spawnParticles(x, y, color) {
for (let i = 0; i < 8; i++) {
const angle = Math.random() * Math.PI * 2;
const speed = 1 + Math.random() * 3;
particles.push({
x: x, y: y,
vx: Math.cos(angle) * speed,
vy: Math.sin(angle) * speed - 1,
life: 1,
color: color,
size: 2 + Math.random() * 3
});
}
}

// Update
function update(dt) {
if (gameState !== 'playing' && gameState !== 'ballReady') return;

// Paddle movement
const paddleSpeed = 8;
if (keys['ArrowLeft'] || keys['KeyA']) paddle.x -= paddleSpeed;
if (keys['ArrowRight'] || keys['KeyD']) paddle.x += paddleSpeed;

// Clamp paddle to world bounds, but also ensure it's within camera view
const minPaddleX = Math.max(0, cameraX);
const maxPaddleX = Math.min(WORLD_WIDTH - paddle.w, cameraX + W);
paddle.x = Math.max(minPaddleX, Math.min(maxPaddleX, paddle.x));

if (gameState === 'ballReady') {
ball.x = paddle.x + paddle.w / 2;
ball.y = H - PADDLE_H - BALL_R - 4;
updateCamera();
return;
}

// Ball physics
ball.x += ball.vx;
ball.y += ball.vy;

// Trail
trail.push({ x: ball.x, y: ball.y, life: 1 });
if (trail.length > 12) trail.shift();

// Walls
if (ball.x < BALL_R) { ball.x = BALL_R; ball.vx = Math.abs(ball.vx); playSound(200, 0.1); }
if (ball.x > WORLD_WIDTH - BALL_R) { ball.x = WORLD_WIDTH - BALL_R; ball.vx = -Math.abs(ball.vx); playSound(200, 0.1); }
if (ball.y < BALL_R) { ball.y = BALL_R; ball.vy = Math.abs(ball.vy); playSound(300, 0.1); }

// Paddle collision
const paddleY = H - PADDLE_H;
if (ball.vy > 0 && ball.y + BALL_R >= paddleY && ball.y + BALL_R <= paddleY + PADDLE_H + 5) {
if (ball.x >= paddle.x - BALL_R && ball.x <= paddle.x + paddle.w + BALL_R) {
const hitPos = (ball.x - paddle.x) / paddle.w;
const angle = (hitPos - 0.5) * Math.PI * 0.7;
ball.vx = Math.sin(angle) * ball.speed;
ball.vy = -Math.cos(angle) * ball.speed;
ball.y = paddleY - BALL_R;
playSound(400, 0.1);
}
}

// Ball falls
if (ball.y > H + 50) {
lives--;
playSound(150, 0.3, 'sawtooth', 0.2);
if (lives <= 0) {
gameState = 'gameOver';
} else {
resetBall();
}
return;
}

// Brick collisions
for (let i = 0; i < bricks.length; i++) {
const b = bricks[i];
if (!b.alive) continue;
if (ball.x + BALL_R > b.x && ball.x - BALL_R < b.x + b.w &&
ball.y + BALL_R > b.y && ball.y - BALL_R < b.y + b.h) {
// Determine side
const overlapLeft = ball.x + BALL_R - b.x;
const overlapRight = b.x + b.w - (ball.x - BALL_R);
const overlapTop = ball.y + BALL_R - b.y;
const overlapBottom = b.y + b.h - (ball.y - BALL_R);
const minOverlap = Math.min(overlapLeft, overlapRight, overlapTop, overlapBottom);

if (minOverlap === overlapLeft || minOverlap === overlapRight) {
ball.vx = -ball.vx;
} else {
ball.vy = -ball.vy;
}

b.hp--;
if (b.hp <= 0) {
b.alive = false;
score += (ROWS - 0) * 10;
spawnParticles(b.x + b.w / 2, b.y + b.h / 2, b.color[0]);
playSound(500 + Math.random() * 200, 0.15);
} else {
playSound(350, 0.08);
}

// Check win
let remaining = 0;
for (let j = 0; j < bricks.length; j++) if (bricks[j].alive) remaining++;
if (remaining === 0) {
gameState = 'win';
playSound(800, 0.4, 'sine', 0.2);
}
break;
}

updateCamera();

// Particles
for (let i = particles.length - 1; i >= 0; i--) {
const p = particles[i];
p.x += p.vx;
p.y += p.vy;
p.vy += 0.15;
p.life -= 0.03;
if (p.life <= 0) particles.splice(i, 1);
}

// Trail fade
for (let i = 0; i < trail.length; i++) trail[i].life -= 0.08;
}

function updateCamera() {
const targetX = ball.x - W / 2;
cameraX += (targetX - cameraX) * 0.08;
cameraX = Math.max(0, Math.min(WORLD_WIDTH - W, cameraX));
}

// Render
function render() {
ctx.fillStyle = '#0a0a1a';
ctx.fillRect(0, 0, W, H);

const cx = cameraX;

// Background grid
ctx.strokeStyle = 'rgba(40,40,80,0.3)';
ctx.lineWidth = 1;
for (let x = 0; x < WORLD_WIDTH; x += 60) {
const sx = x - cx;
if (sx > -60 && sx < W + 60) {
ctx.beginPath();
ctx.moveTo(sx, 0);
ctx.lineTo(sx, H);
ctx.stroke();
}
}
for (let y = 0; y < H; y += 60) {
ctx.beginPath();
ctx.moveTo(0, y);
ctx.lineTo(W, y);
ctx.stroke();
}

// Bricks
for (const b of bricks) {
if (!b.alive) continue;
const sx = b.x - cx;
if (sx + b.w < 0 || sx > W) continue;
ctx.fillStyle = b.color[0];
ctx.fillRect(sx, b.y, b.w, b.h);
ctx.fillStyle = 'rgba(255,255,255,0.2)';
ctx.fillRect(sx, b.y, b.w, 4);
if (b.hp > 1) {
ctx.strokeStyle = 'rgba(255,255,255,0.5)';
ctx.lineWidth = 1;
ctx.strokeRect(sx + 2, b.y + 2, b.w - 4, b.h - 4);
}
}

// Trail
for (const t of trail) {
if (t.life <= 0) continue;
const sx = t.x - cx;
ctx.fillStyle = `rgba(100,200,255,${t.life * 0.3})`;
ctx.beginPath();
ctx.arc(sx, t.y, BALL_R * t.life, 0, Math.PI * 2);
ctx.fill();
}

// Ball
const bx = ball.x - cx;
ctx.fillStyle = '#ffffff';
ctx.shadowColor = '#44aaff';
ctx.shadowBlur = 12;
ctx.beginPath();
ctx.arc(bx, ball.y, BALL_R, 0, Math.PI * 2);
ctx.fill();
ctx.shadowBlur = 0;

// Paddle
const px = paddle.x - cx;
const py = H - PADDLE_H;
const grad = ctx.createLinearGradient(px, py, px, py + PADDLE_H);
grad.addColorStop(0, '#44aaff');
grad.addColorStop(1, '#2266cc');
ctx.fillStyle = grad;
ctx.beginPath();
ctx.roundRect(px, py, paddle.w, PADDLE_H, 8);
ctx.fill();

// Particles
for (const p of particles) {
const sx = p.x - cx;
ctx.fillStyle = `rgba(${hexToRgb(p.color)},${p.life})`;
ctx.fillRect(sx - p.size / 2, p.y - p.size / 2, p.size, p.size);
}

// HUD
ctx.fillStyle = 'rgba(0,0,0,0.5)';
ctx.fillRect(0, 0, W, 36);
ctx.font = 'bold 16px monospace';
ctx.fillStyle = '#fff';
ctx.textAlign = 'left';
ctx.fillText(`SCORE: ${score}`, 16, 24);
ctx.fillText(`LIVES: ${'❤'.repeat(lives)}`, W / 2 - 50, 24);
ctx.textAlign = 'right';
ctx.fillText(`LEVEL: ${level}`, W - 16, 24);

// World position indicator
const posPct = (ball.x / WORLD_WIDTH * 100).toFixed(0);
ctx.textAlign = 'center';
ctx.fillStyle = 'rgba(255,255,255,0.5)';
ctx.font = '12px monospace';
ctx.fillText(`◀ ${posPct}% ▶`, W / 2, H - 10);

// Overlays
if (gameState === 'start') {
drawOverlay('SIDE-SCROLL BREAKOUT', [
'← → or A/D : Move paddle',
'Space / Click : Launch ball',
'R : Restart after game over',
'',
'The world scrolls with the ball!',
'Clear all bricks to advance.']);
} else if (gameState === 'ballReady') {
ctx.font = '16px monospace';
ctx.fillStyle = '#aaf';
ctx.textAlign = 'center';
ctx.fillText('Press SPACE or CLICK to launch', W / 2, H / 2 + 40);
} else if (gameState === 'gameOver') {
drawOverlay('GAME OVER', [`Final Score: ${score}`, 'Press R or Click to restart']);
} else if (gameState === 'win') {
if (level >= 3) {
drawOverlay('YOU WIN!', [`Final Score: ${score}`, 'Press R or Click to play again']);
} else {
ctx.fillStyle = 'rgba(0,0,0,0.7)';
ctx.fillRect(0, H / 2 - 40, W, 80);
ctx.font = 'bold 24px monospace';
ctx.fillStyle = '#4f4';
ctx.textAlign = 'center';
ctx.fillText(`LEVEL ${level} COMPLETE!`, W / 2, H / 2);
ctx.font = '16px monospace';
ctx.fillStyle = '#fff';
ctx.fillText('Press SPACE for next level', W / 2, H / 2 + 30);
if (keys['Space']) {
nextLevel();
}
}
}
}

function drawOverlay(title, lines) {
ctx.fillStyle = 'rgba(0,0,0,0.8)';
ctx.fillRect(0, 0, W, H);
ctx.font = 'bold 36px monospace';
ctx.fillStyle = '#4af';
ctx.textAlign = 'center';
ctx.fillText(title, W / 2, H / 3);
ctx.font = '18px monospace';
ctx.fillStyle = '#ccc';
lines.forEach((line, i) => {
ctx.fillText(line, W / 2, H / 3 + 40 + i * 28);
});
}

function hexToRgb(hex) {
const r = parseInt(hex.slice(1, 3), 16);
const g = parseInt(hex.slice(3, 5), 16);
const b = parseInt(hex.slice(5, 7), 16);
return `${r},${g},${b}`;
}

// Handle space for next level
window.addEventListener('keydown', function(e) {
if (gameState === 'win' && level < 3 && e.code === 'Space') {
nextLevel();
}
});

// Game loop
let lastTime = 0;
function gameLoop(time) {
const dt = Math.min((time - lastTime) / 16.67, 3);
lastTime = time;
update(dt);
render();
requestAnimationFrame(gameLoop);
}
requestAnimationFrame(gameLoop);

})();
</script>
```
