<script>
const canvas = document.getElementById('game');
const ctx = canvas.getContext('2d');

let W, H;
function resize() {
    W = canvas.width = window.innerWidth;
    H = canvas.height = window.innerHeight;
}
window.addEventListener('resize', resize);
resize();

// Audio
let audioCtx = null;
function initAudio() {
    if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
}
function playSound(type) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    if (type === 'shoot') {
        osc.type = 'square';
        osc.frequency.setValueAtTime(400, audioCtx.currentTime);
        osc.frequency.exponentialRampToValueAtTime(200, audioCtx.currentTime + 0.1);
        gain.gain.setValueAtTime(0.15, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.1);
        osc.start(); osc.stop(audioCtx.currentTime + 0.1);
    } else if (type === 'hit') {
        osc.type = 'sawtooth';
        osc.frequency.setValueAtTime(200, audioCtx.currentTime);
        osc.frequency.exponentialRampToValueAtTime(80, audioCtx.currentTime + 0.15);
        gain.gain.setValueAtTime(0.2, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.15);
        osc.start(); osc.stop(audioCtx.currentTime + 0.15);
    } else if (type === 'death') {
        osc.type = 'sawtooth';
        osc.frequency.setValueAtTime(300, audioCtx.currentTime);
        osc.frequency.exponentialRampToValueAtTime(50, audioCtx.currentTime + 0.4);
        gain.gain.setValueAtTime(0.3, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.4);
        osc.start(); osc.stop(audioCtx.currentTime + 0.4);
    } else if (type === 'win') {
        osc.type = 'sine';
        osc.frequency.setValueAtTime(523, audioCtx.currentTime);
        gain.gain.setValueAtTime(0.2, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.5);
        osc.start(); osc.stop(audioCtx.currentTime + 0.5);
    }
}

// Game State
const STATE = { MENU: 0, PLAYING: 1, ROUND_END: 2, GAME_OVER: 3 };
let gameState = STATE.MENU;
let score1 = 0, score2 = 0;
const WIN_SCORE = 3;
let roundTimer = 0;
let shakeTime = 0;
let shakeMag = 0;

// Input
const keys = {};
window.addEventListener('keydown', e => {
    keys[e.code] = true;
    initAudio();
    if (gameState === STATE.MENU && (e.code === 'Space' || e.code === 'Enter')) {
        startGame();
    } else if (gameState === STATE.GAME_OVER && (e.code === 'Space' || e.code === 'Enter')) {
        resetToMenu();
    } else if (gameState === STATE.ROUND_END && (e.code === 'Space' || e.code === 'Enter')) {
        startRound();
    }
});
window.addEventListener('keyup', e => { keys[e.code] = false; });

// Particles
let particles = [];
function spawnParticles(x, y, color, count) {
    for (let i = 0; i < count; i++) {
        const angle = Math.random() * Math.PI * 2;
        const speed = 1 + Math.random() * 4;
        particles.push({
            x, y,
            vx: Math.cos(angle) * speed,
            vy: Math.sin(angle) * speed,
            life: 30 + Math.random() * 20,
            maxLife: 50,
            color,
            size: 2 + Math.random() * 3
        });
    }
}

// Obstacles
let obstacles = [];
function generateObstacles() {
    obstacles = [];
    const margin = 80;
    const count = 5 + Math.floor(Math.random() * 4);
    for (let i = 0; i < count; i++) {
        const w = 60 + Math.random() * 100;
        const h = 60 + Math.random() * 100;
        let ox, oy, tries = 0;
        do {
            ox = margin + Math.random() * (W - margin * 2 - w);
            oy = margin + Math.random() * (H - margin * 2 - h);
            tries++;
        } while (tries < 50 && obstacles.some(o => {
            const dx = Math.abs((ox + w/2) - (o.x + o.w/2));
            const dy = Math.abs((oy + h/2) - (o.y + o.h/2));
            return dx < 150 && dy < 150;
        }));
        obstacles.push({ x: ox, y: oy, w, h });
    }
}

// Players
class Player {
    constructor(id, x, y, color) {
        this.id = id;
        this.x = x; this.y = y;
        this.radius = 18;
        this.color = color;
        this.health = 100;
        this.maxHealth = 100;
        this.cooldown = 0;
        this.dead = false;
        this.angle = id === 1 ? 0 : Math.PI;
    }
}

let players = [];
let bullets = [];

function startGame() {
    score1 = 0; score2 = 0;
    startRound();
}

function resetToMenu() {
    gameState = STATE.MENU;
}

function startRound() {
    generateObstacles();
    const p1x = W * 0.2, p1y = H * 0.5;
    const p2x = W * 0.8, p2y = H * 0.5;
    players = [
        new Player(1, p1x, p1y, '#ff4444'),
        new Player(2, p2x, p2y, '#4488ff')
    ];
    bullets = [];
    particles = [];
    gameState = STATE.PLAYING;
}

function shake(mag, time) {
    shakeMag = mag;
    shakeTime = time;
}

// Collision helpers
function circleRect(cx, cy, r, rx, ry, rw, rh) {
    const dx = cx - (rx + rw/2);
    const dy = cy - (ry + rh/2);
    const closestX = Math.max(rx, Math.min(cx, rx + rw));
    const closestY = Math.max(ry, Math.min(cy, ry + rh));
    const dist = Math.hypot(cx - closestX, cy - closestY);
    return dist < r;
}

function circleCircle(x1, y1, r1, x2, y2, r2) {
    return Math.hypot(x1 - x2, y1 - y2) < r1 + r2;
}

function update() {
    if (gameState !== STATE.PLAYING) return;

    // Update players
    for (const p of players) {
        if (p.dead) continue;
        let dx = 0, dy = 0;
        if (p.id === 1) {
            if (keys['KeyW']) dy = -1;
            if (keys['KeyS']) dy = 1;
            if (keys['KeyA']) dx = -1;
            if (keys['KeyD']) dx = 1;
            p.angle = Math.atan2(dy, dx) || p.angle;
        } else {
            if (keys['ArrowUp']) dy = -1;
            if (keys['ArrowDown']) dy = 1;
            if (keys['ArrowLeft']) dx = -1;
            if (keys['ArrowRight']) dx = 1;
            p.angle = Math.atan2(dy, dx) || p.angle;
        }

        const speed = 4;
        if (dx !== 0 && dy !== 0) { dx *= 0.707; dy *= 0.707; }
        const nx = p.x + dx * speed;
        const ny = p.y + dy * speed;

        // Obstacle collision
        let canMoveX = true, canMoveY = true;
        for (const o of obstacles) {
            if (circleRect(nx, p.y, p.radius, o.x, o.y, o.w, o.h)) canMoveX = false;
            if (circleRect(p.x, ny, p.radius, o.x, o.y, o.w, o.h)) canMoveY = false;
        }
        if (canMoveX) p.x += dx * speed;
        if (canMoveY) p.y += dy * speed;

        // Bounds
        p.x = Math.max(p.radius, Math.min(W - p.radius, p.x));
        p.y = Math.max(p.radius, Math.min(H - p.radius, p.y));

        // Shooting
        if (p.cooldown > 0) p.cooldown--;
        const shootKey = p.id === 1 ? 'Space' : 'Enter';
        if (keys[shootKey] && p.cooldown <= 0) {
            const bx = p.x + Math.cos(p.angle) * p.radius;
            const by = p.y + Math.sin(p.angle) * p.radius;
            bullets.push({ x: bx, y: by, vx: Math.cos(p.angle) * 8, vy: Math.sin(p.angle) * 8, owner: p.id, life: 120 });
            p.cooldown = 15;
            playSound('shoot');
        }
    }

    // Update bullets
    for (let i = bullets.length - 1; i >= 0; i--) {
        const b = bullets[i];
        b.x += b.vx;
        b.y += b.vy;
        b.life--;
        if (b.life <= 0 || b.x < -20 || b.x > W + 20 || b.y < -20 || b.y > H + 20) {
            bullets.splice(i, 1);
            continue;
        }

        // Obstacle hit
        let hitWall = false;
        for (const o of obstacles) {
            if (b.x > o.x && b.x < o.x + o.w && b.y > o.y && b.y < o.y + o.h) {
                hitWall = true;
                break;
            }
        }
        if (hitWall) {
            spawnParticles(b.x, b.y, '#ffaa00', 5);
            bullets.splice(i, 1);
            continue;
        }

        // Player hit
        for (const p of players) {
            if (p.dead || p.id === b.owner) continue;
            if (circleCircle(b.x, b.y, 4, p.x, p.y, p.radius)) {
                p.health -= 25;
                spawnParticles(b.x, b.y, p.color, 10);
                playSound('hit');
                shake(5, 8);
                if (p.health <= 0) {
                    p.dead = true;
                    spawnParticles(p.x, p.y, p.color, 30);
                    playSound('death');
                    shake(10, 20);
                    endRound(b.owner);
                }
                bullets.splice(i, 1);
                break;
            }
        }
    }

    // Update particles
    for (let i = particles.length - 1; i >= 0; i--) {
        const p = particles[i];
        p.x += p.vx;
        p.y += p.vy;
        p.vx *= 0.95;
        p.vy *= 0.95;
        p.life--;
        if (p.life <= 0) particles.splice(i, 1);
    }

    // Shake
    if (shakeTime > 0) shakeTime--;
}

function endRound(winnerId) {
    gameState = STATE.ROUND_END;
    roundTimer = 120;
    if (winnerId === 1) score1++; else score2++;
    playSound('win');
}

function draw() {
    ctx.save();
    if (shakeTime > 0) {
        const sx = (Math.random() - 0.5) * shakeMag;
        const sy = (Math.random() - 0.5) * shakeMag;
        ctx.translate(sx, sy);
    }

    // Background
    ctx.fillStyle = '#1a1a2e';
    ctx.fillRect(-10, -10, W + 20, H + 20);

    // Grid
    ctx.strokeStyle = 'rgba(255,255,255,0.03)';
    ctx.lineWidth = 1;
    for (let x = 0; x < W; x += 40) {
        ctx.beginPath(); ctx.moveTo(x, 0); ctx.lineTo(x, H); ctx.stroke();
    }
    for (let y = 0; y < H; y += 40) {
        ctx.beginPath(); ctx.moveTo(0, y); ctx.lineTo(W, y); ctx.stroke();
    }

    if (gameState === STATE.MENU) {
        drawMenu();
        ctx.restore();
        return;
    }

    // Obstacles
    for (const o of obstacles) {
        ctx.fillStyle = '#3a3a5c';
        ctx.fillRect(o.x, o.y, o.w, o.h);
        ctx.strokeStyle = '#5a5a8c';
        ctx.lineWidth = 2;
        ctx.strokeRect(o.x, o.y, o.w, o.h);
    }

    // Bullets
    for (const b of bullets) {
        const color = b.owner === 1 ? '#ff6666' : '#66aaff';
        ctx.beginPath();
        ctx.arc(b.x, b.y, 4, 0, Math.PI * 2);
        ctx.fillStyle = color;
        ctx.fill();
        ctx.beginPath();
        ctx.arc(b.x, b.y, 7, 0, Math.PI * 2);
        ctx.strokeStyle = color;
        ctx.lineWidth = 1;
        ctx.globalAlpha = 0.4;
        ctx.stroke();
        ctx.globalAlpha = 1;
    }

    // Players
    for (const p of players) {
        if (p.dead) continue;
        // Body
        ctx.beginPath();
        ctx.arc(p.x, p.y, p.radius, 0, Math.PI * 2);
        ctx.fillStyle = p.color;
        ctx.fill();
        ctx.strokeStyle = '#fff';
        ctx.lineWidth = 2;
        ctx.stroke();

        // Gun direction
        const gx = p.x + Math.cos(p.angle) * (p.radius + 8);
        const gy = p.y + Math.sin(p.angle) * (p.radius + 8);
        ctx.beginPath();
        ctx.moveTo(p.x + Math.cos(p.angle) * p.radius, p.y + Math.sin(p.angle) * p.radius);
        ctx.lineTo(gx, gy);
        ctx.strokeStyle = '#fff';
        ctx.lineWidth = 4;
        ctx.stroke();

        // Health bar above player
        const bw = 40;
        ctx.fillStyle = 'rgba(0,0,0,0.5)';
        ctx.fillRect(p.x - bw/2, p.y - p.radius - 15, bw, 6);
        ctx.fillStyle = p.health > 50 ? '#4f4' : p.health > 25 ? '#ff4' : '#f44';
        ctx.fillRect(p.x - bw/2, p.y - p.radius - 15, bw * (p.health / p.maxHealth), 6);
    }

    // Particles
    for (const p of particles) {
        ctx.globalAlpha = p.life / p.maxLife;
        ctx.fillStyle = p.color;
        ctx.fillRect(p.x - p.size/2, p.y - p.size/2, p.size, p.size);
    }
    ctx.globalAlpha = 1;

    // HUD
    if (gameState === STATE.PLAYING || gameState === STATE.ROUND_END) {
        ctx.font = 'bold 16px monospace';
        ctx.fillStyle = '#ff4444';
        ctx.textAlign = 'left';
        ctx.fillText('P1: ' + score1, 20, 30);
        ctx.fillStyle = '#4488ff';
        ctx.textAlign = 'right';
        ctx.fillText('P2: ' + score2, W - 20, 30);
        ctx.fillStyle = '#fff';
        ctx.textAlign = 'center';
        ctx.font = '12px monospace';
        ctx.fillText('First to ' + WIN_SCORE, W/2, 30);

        // Score dots
        for (let i = 0; i < WIN_SCORE; i++) {
            ctx.beginPath();
            ctx.arc(W/2 - 40 + i * 20, 50, 6, 0, Math.PI * 2);
            ctx.fillStyle = i < score1 ? '#ff4444' : 'rgba(255,255,255,0.2)';
            ctx.fill();
            ctx.beginPath();
            ctx.arc(W/2 + 40 - i * 20, 50, 6, 0, Math.PI * 2);
            ctx.fillStyle = i < score2 ? '#4488ff' : 'rgba(255,255,255,0.2)';
            ctx.fill();
        }
    }

    // Round End overlay
    if (gameState === STATE.ROUND_END) {
        ctx.fillStyle = 'rgba(0,0,0,0.6)';
        ctx.fillRect(0, 0, W, H);
        const winner = score1 > score2 ? 'PLAYER 1' : 'PLAYER 2';
        const wcolor = score1 > score2 ? '#ff4444' : '#4488ff';
        ctx.font = 'bold 36px monospace';
        ctx.fillStyle = wcolor;
        ctx.textAlign = 'center';
        ctx.fillText(winner + ' WINS THE ROUND!', W/2, H/2);
        ctx.font = '18px monospace';
        ctx.fillStyle = '#fff';
        ctx.fillText('Press SPACE or ENTER to continue', W/2, H/2 + 50);
    }

    // Game Over
    if (gameState === STATE.GAME_OVER) {
        ctx.fillStyle = 'rgba(0,0,0,0.7)';
        ctx.fillRect(0, 0, W, H);
        const winner = score1 > score2 ? 'PLAYER 1' : 'PLAYER 2';
        const wcolor = score1 > score2 ? '#ff4444' : '#4488ff';
        ctx.font = 'bold 48px monospace';
        ctx.fillStyle = wcolor;
        ctx.textAlign = 'center';
        ctx.fillText(winner + ' WINS!', W/2, H/2 - 20);
        ctx.font = '20px monospace';
        ctx.fillStyle = '#fff';
        ctx.fillText('Press SPACE or ENTER to play again', W/2, H/2 + 40);
    }

    ctx.restore();
}

function drawMenu() {
    ctx.textAlign = 'center';
    ctx.font = 'bold 52px monospace';
    ctx.fillStyle = '#fff';
    ctx.fillText('ARENA CLASH', W/2, H/2 - 80);

    ctx.font = '18px monospace';
    ctx.fillStyle = '#aaa';
    ctx.fillText('A two-player top-down shooter', W/2, H/2 - 40);

    ctx.font = '16px monospace';
    ctx.fillStyle = '#ff4444';
    ctx.fillText('PLAYER 1: WASD + SPACE to shoot', W/2, H/2 + 20);
    ctx.fillStyle = '#4488ff';
    ctx.fillText('PLAYER 2: ARROWS + ENTER to shoot', W/2, H/2 + 50);

    ctx.fillStyle = '#fff';
    ctx.font = '14px monospace';
    ctx.fillText('First to ' + WIN_SCORE + ' round wins! Avoid the obstacles.', W/2, H/2 + 100);

    const pulse = Math.sin(Date.now() / 300) * 0.5 + 0.5;
    ctx.globalAlpha = pulse;
    ctx.font = 'bold 22px monospace';
    ctx.fillStyle = '#fff';
    ctx.fillText('Press SPACE or ENTER to start', W/2, H/2 + 160);
    ctx.globalAlpha = 1;
}

function checkRoundEnd() {
    if (players[0].dead || players[1].dead) return;
}

function gameLoop() {
    update();

    if (gameState === STATE.ROUND_END) {
        roundTimer--;
        if (roundTimer <= 0) {
            if (score1 >= WIN_SCORE || score2 >= WIN_SCORE) {
                gameState = STATE.GAME_OVER;
            } else {
                startRound();
            }
        }
    }

    draw();
    requestAnimationFrame(gameLoop);
}

gameLoop();
</script>
