const canvas = document.getElementById('c');
const ctx = canvas.getContext('2d');

let W, H, SCALE;
function resize() {
  W = window.innerWidth;
  H = window.innerHeight;
  canvas.width = W;
  canvas.height = H;
  SCALE = Math.min(W / 800, H / 600);
}
window.addEventListener('resize', resize);
resize();

// Game constants
const COLS = 16, ROWS = 12;
let cellSize;
let gridX, gridY;

function calcGrid() {
  cellSize = Math.floor(Math.min(W / (COLS + 4), H / (ROWS + 4))) * SCALE;
  gridX = Math.floor((W - COLS * cellSize) / 2);
  gridY = Math.floor((H - ROWS * cellSize) / 2);
}

// Game state
let gameState = 'start'; // start, place, execute, levelComplete, gameOver, win
let level = 1;
let score = 0;
let bestScore = 0;
try { bestScore = parseInt(localStorage.getItem('bulletHelmBest')) || 0; } catch(e) {}

// Player
let player = { x: 0, y: 0, px: 0, py: 0, alive: true };
// Barriers
let barriers = []; // {x, y} grid positions
let maxBarriers = 3;
// Bullets
let bullets = [];
let bulletPattern = [];
let waveTimer = 0;
let waveDuration = 0;
let exitTile = { x: 0, y: 0 };
let particles = [];
let shakeTime = 0;
let shakeMag = 0;
let msgTimer = 0;
let msgText = '';
let phaseShifts = 2;
let phaseTimer = 0;
let invulnTimer = 0;

// Input
let keys = {};
window.addEventListener('keydown', e => {
  keys[e.key] = true;
  if (gameState === 'start' && (e.key === 'Enter' || e.key === ' ')) startGame();
  if (gameState === 'gameOver' && (e.key === 'Enter' || e.key === ' ')) restartGame();
  if (gameState === 'levelComplete' && (e.key === 'Enter' || e.key === ' ')) nextLevel();
  if (gameState === 'place' && e.key === ' ') startWave();
  if (gameState === 'execute' && e.key === ' ') usePhaseShift();
  e.preventDefault();
});
window.addEventListener('keyup', e => { keys[e.key] = false; });

// Mouse for placement
let mouseX = 0, mouseY = 0;
canvas.addEventListener('mousemove', e => {
  const r = canvas.getBoundingClientRect();
  mouseX = e.clientX - r.left;
  mouseY = e.clientY - r.top;
});
canvas.addEventListener('click', e => {
  if (gameState === 'place') placeBarrier();
});

function startGame() {
  level = 1;
  score = 0;
  initLevel();
}

function restartGame() {
  level = 1;
  score = 0;
  gameState = 'place';
  initLevel();
}

function initLevel() {
  calcGrid();
  barriers = [];
  bullets = [];
  particles = [];
  player.alive = true;
  player.x = 2;
  player.y = Math.floor(ROWS / 2);
  player.px = player.x * cellSize + gridX + cellSize/2;
  player.py = player.y * cellSize + gridY + cellSize/2;
  phaseShifts = 2;
  invulnTimer = 0;
  
  // Exit tile
  exitTile.x = COLS - 3;
  exitTile.y = Math.floor(ROWS / 2);
  
  // Generate bullet pattern
  generatePattern();
  gameState = 'place';
}

function nextLevel() {
  level++;
  if (level > 8) {
    gameState = 'win';
    try { localStorage.setItem('bulletHelmBest', Math.max(bestScore, score).toString()); } catch(e) {}
    return;
  }
  initLevel();
}

function generatePattern() {
  bulletPattern = [];
  waveDuration = 180 + level * 30; // frames
  
  const types = ['sweep', 'dualSweep', 'spiral', 'diagonal', 'radial', 'zigzag', 'converge', 'chaos'];
  const type = types[(level - 1) % types.length];
  
  let speed = 0.3 + level * 0.05; // cells per frame
  
  switch(type) {
    case 'sweep':
      for (let r = 0; r < ROWS; r += Math.max(1, 2 - Math.floor(level/3))) {
        bulletPattern.push({ x: -1, y: r + 0.5, vx: speed, vy: 0, life: waveDuration });
      }
      break;
    case 'dualSweep':
      for (let r = 0; r < ROWS; r += Math.max(1, 3 - level)) {
        bulletPattern.push({ x: -1, y: r + 0.5, vx: speed, vy: 0, life: waveDuration });
        bulletPattern.push({ x: COLS + 1, y: r + 0.5, vx: -speed, vy: 0, life: waveDuration });
      }
      break;
    case 'spiral':
      for (let i = 0; i < 20 + level * 3; i++) {
        const angle = (i / (20 + level * 3)) * Math.PI * 4;
        bulletPattern.push({
          x: COLS/2, y: ROWS/2,
          vx: Math.cos(angle) * speed * 1.5,
          vy: Math.sin(angle) * speed * 1.5,
          life: waveDuration, delay: i * 3
        });
      }
      break;
    case 'diagonal':
      for (let i = 0; i < 8 + level * 2; i++) {
        bulletPattern.push({ x: -1, y: i * (ROWS / (8 + level*2)), vx: speed * 0.7, vy: speed * 0.3, life: waveDuration });
        bulletPattern.push({ x: COLS+1, y: ROWS - i * (ROWS / (8 + level*2)), vx: -speed * 0.7, vy: speed * 0.3, life: waveDuration });
      }
      break;
    case 'radial':
      for (let i = 0; i < 16 + level * 2; i++) {
        const angle = (i / (16 + level*2)) * Math.PI * 2;
        bulletPattern.push({
          x: COLS/2, y: ROWS/2,
          vx: Math.cos(angle) * speed,
          vy: Math.sin(angle) * speed,
          life: waveDuration, delay: i * 2
        });
      }
      break;
    case 'zigzag':
      for (let r = 0; r < ROWS; r += 2) {
        for (let c = 0; c < COLS; c += 3) {
          bulletPattern.push({ x: c, y: -1, vx: 0, vy: speed * 0.5, life: waveDuration, delay: c * 4 });
        }
      }
      break;
    case 'converge':
      for (let i = 0; i < 24 + level * 2; i++) {
        const angle = (i / (24 + level*2)) * Math.PI * 2;
        bulletPattern.push({
          x: COLS/2 + Math.cos(angle) * 8, y: ROWS/2 + Math.sin(angle) * 6,
          vx: -Math.cos(angle) * speed * 0.8,
          vy: -Math.sin(angle) * speed * 0.8,
          life: waveDuration, delay: i
        });
      }
      break;
    case 'chaos':
      for (let i = 0; i < 12 + level * 3; i++) {
        const sx = Math.random() < 0.5 ? -1 : COLS+1;
        bulletPattern.push({ x: sx, y: Math.random() * ROWS, vx: (sx < 0 ? 1 : -1) * speed * (0.5 + Math.random()*0.5), vy: (Math.random()-0.5)*speed*0.3, life: waveDuration });
      }
      break;
  }
}

function placeBarrier() {
  if (barriers.length >= maxBarriers) return;
  
  const gx = Math.floor((mouseX - gridX) / cellSize);
  const gy = Math.floor((mouseY - gridY) / cellSize);
  
  if (gx < 0 || gx >= COLS || gy < 0 || gy >= ROWS) return;
  if (player.x === gx && player.y === gy) return;
  if (exitTile.x === gx && exitTile.y === gy) return;
  if (barriers.some(b => b.x === gx && b.y === gy)) return;
  
  barriers.push({ x: gx, y: gy });
}

function startWave() {
  gameState = 'execute';
  waveTimer = 0;
  bullets = [];
  
  // Initialize bullets from pattern
  for (let p of bulletPattern) {
    let bx = p.x * cellSize + gridX + cellSize/2;
    let by = p.y * cellSize + gridY + cellSize/2;
    bullets.push({
      x: bx, y: by,
      vx: p.vx * cellSize, vy: p.vy * cellSize,
      life: p.life || waveDuration,
      delay: p.delay || 0,
      active: (p.delay || 0) <= 0,
      radius: Math.max(3, cellSize * 0.2)
    });
  }
}

function usePhaseShift() {
  if (phaseShifts > 0 && invulnTimer <= 0) {
    phaseShifts--;
    invulnTimer = 45; // frames of invulnerability
    for (let i = 0; i < 12; i++) {
      particles.push({ x: player.px, y: player.py, vx: (Math.random()-0.5)*3, vy: (Math.random()-0.5)*3, life: 20, color: '#4ff' });
    }
  }
}

function isBarrier(gx, gy) {
  return barriers.some(b => b.x === gx && b.y === gy);
}

function update() {
  if (gameState !== 'execute') return;
  
  waveTimer++;
  
  // Move player
  let dx = 0, dy = 0;
  if (keys['w'] || keys['ArrowUp']) dy = -1;
  if (keys['s'] || keys['ArrowDown']) dy = 1;
  if (keys['a'] || keys['ArrowLeft']) dx = -1;
  if (keys['d'] || keys['ArrowRight']) dx = 1;
  
  if (dx !== 0 || dy !== 0) {
    let speed = cellSize * 0.04;
    player.px += dx * speed;
    player.py += dy * speed;
    
    // Clamp to grid
    player.px = Math.max(gridX + cellSize*0.3, Math.min(gridX + COLS*cellSize - cellSize*0.3, player.px));
    player.py = Math.max(gridY + cellSize*0.3, Math.min(gridY + ROWS*cellSize - cellSize*0.3, player.py));
    
    // Barriers block player
    let pgx = Math.floor((player.px - gridX) / cellSize);
    let pgy = Math.floor((player.py - gridY) / cellSize);
    if (isBarrier(pgx, pgy)) {
      player.px -= dx * speed;
      player.py -= dy * speed;
    }
  }
  
  // Invulnerability timer
  if (invulnTimer > 0) invulnTimer--;
  
  // Update bullets
  for (let b of bullets) {
    if (!b.active) {
      if (b.delay > 0) b.delay--;
      else b.active = true;
      continue;
    }
    b.x += b.vx;
    b.y += b.vy;
    b.life--;
    
    // Barrier collision
    let bgx = Math.floor((b.x - gridX) / cellSize);
    let bgy = Math.floor((b.y - gridY) / cellSize);
    if (isBarrier(bgx, bgy)) {
      b.life = 0;
      for (let i = 0; i < 4; i++) {
        particles.push({ x: b.x, y: b.y, vx: (Math.random()-0.5)*2, vy: (Math.random()-0.5)*2, life: 15, color: '#fa0' });
      }
    }
    
    // Player collision
    if (b.life > 0 && invulnTimer <= 0) {
      let d = Math.hypot(b.x - player.px, b.y - player.py);
      if (d < b.radius + cellSize * 0.15) {
        killPlayer();
        return;
      }
    }
    
    // Remove off-screen bullets
    if (b.x < gridX - cellSize*2 || b.x > gridX + COLS*cellSize + cellSize*2 ||
        b.y < gridY - cellSize*2 || b.y > gridY + ROWS*cellSize + cellSize*2) {
      b.life = 0;
    }
  }
  
  bullets = bullets.filter(b => b.life > 0);
  
  // Check exit
  let pgx = Math.floor((player.px - gridX) / cellSize);
  let pgy = Math.floor((player.py - gridY) / cellSize);
  if (pgx === exitTile.x && pgy === exitTile.y) {
    levelComplete();
    return;
  }
  
  // Check wave complete
  if (waveTimer >= waveDuration + 60) {
    // Survived the wave - bonus points
    score += 10 * level;
    levelComplete();
  }
  
  // Update particles
  for (let p of particles) {
    p.x += p.vx;
    p.y += p.vy;
    p.life--;
  }
  particles = particles.filter(p => p.life > 0);
}

function killPlayer() {
  player.alive = false;
  shakeTime = 20;
  shakeMag = 5;
  for (let i = 0; i < 30; i++) {
    particles.push({ x: player.px, y: player.py, vx: (Math.random()-0.5)*6, vy: (Math.random()-0.5)*6, life: 40, color: Math.random() < 0.5 ? '#f44' : '#ff8' });
  }
  gameState = 'gameOver';
  try {
    if (score > bestScore) {
      bestScore = score;
      localStorage.setItem('bulletHelmBest', bestScore.toString());
    }
  } catch(e) {}
}

function levelComplete() {
  score += 50 * level + Math.max(0, waveDuration - waveTimer);
  gameState = 'levelComplete';
  msgTimer = 60;
  msgText = `Level ${level} Complete!  +${50 * level}`;
}

function draw() {
  calcGrid();
  
  ctx.fillStyle = '#0a0a12';
  ctx.fillRect(0, 0, W, H);
  
  let ox = 0, oy = 0;
  if (shakeTime > 0) {
    shakeTime--;
    ox = (Math.random()-0.5) * shakeMag;
    oy = (Math.random()-0.5) * shakeMag;
  }
  
  ctx.save();
  ctx.translate(ox, oy);
  
  // Draw grid
  ctx.strokeStyle = '#1a1a2e';
  ctx.lineWidth = 1;
  for (let c = 0; c <= COLS; c++) {
    ctx.beginPath();
    ctx.moveTo(gridX + c * cellSize, gridY);
    ctx.lineTo(gridX + c * cellSize, gridY + ROWS * cellSize);
    ctx.stroke();
  }
  for (let r = 0; r <= ROWS; r++) {
    ctx.beginPath();
    ctx.moveTo(gridX, gridY + r * cellSize);
    ctx.lineTo(gridX + COLS * cellSize, gridY + r * cellSize);
    ctx.stroke();
  }
  
  // Draw exit tile
  let ex = gridX + exitTile.x * cellSize;
  let ey = gridY + exitTile.y * cellSize;
  let pulse = Math.sin(Date.now() * 0.005) * 0.3 + 0.7;
  ctx.fillStyle = `rgba(0, 255, 180, ${pulse * 0.3})`;
  ctx.fillRect(ex + 4, ey + 4, cellSize - 8, cellSize - 8);
  ctx.strokeStyle = `rgba(0, 255, 180, ${pulse})`;
  ctx.lineWidth = 2;
  ctx.strokeRect(ex + 6, ey + 6, cellSize - 12, cellSize - 12);
  
  // Draw barriers
  for (let b of barriers) {
    let bx = gridX + b.x * cellSize;
    let by = gridY + b.y * cellSize;
    ctx.fillStyle = '#4a6fa5';
    ctx.fillRect(bx + 2, by + 2, cellSize - 4, cellSize - 4);
    ctx.strokeStyle = '#7ab8ff';
    ctx.lineWidth = 2;
    ctx.strokeRect(bx + 2, by + 2, cellSize - 4, cellSize - 4);
    // Cross pattern
    ctx.beginPath();
    ctx.moveTo(bx + cellSize*0.3, by + cellSize*0.3);
    ctx.lineTo(bx + cellSize*0.7, by + cellSize*0.7);
    ctx.moveTo(bx + cellSize*0.7, by + cellSize*0.3);
    ctx.lineTo(bx + cellSize*0.3, by + cellSize*0.7);
    ctx.stroke();
  }
  
  // Draw bullets
  for (let b of bullets) {
    if (!b.active) continue;
    let gradient = ctx.createRadialGradient(b.x, b.y, 0, b.x, b.y, b.radius * 1.5);
    gradient.addColorStop(0, '#fff');
    gradient.addColorStop(0.4, '#ff6644');
    gradient.addColorStop(1, 'rgba(255, 68, 68, 0)');
    ctx.fillStyle = gradient;
    ctx.beginPath();
    ctx.arc(b.x, b.y, b.radius * 1.5, 0, Math.PI * 2);
    ctx.fill();
    
    ctx.fillStyle = '#fff';
    ctx.beginPath();
    ctx.arc(b.x, b.y, b.radius * 0.5, 0, Math.PI * 2);
    ctx.fill();
  }
  
  // Draw player
  if (player.alive) {
    let pColor = invulnTimer > 0 ? '#4ff' : '#fff';
    ctx.fillStyle = pColor;
    ctx.beginPath();
    ctx.arc(player.px, player.py, cellSize * 0.25, 0, Math.PI * 2);
    ctx.fill();
    
    // Glow
    let pg = ctx.createRadialGradient(player.px, player.py, 0, player.px, player.py, cellSize * 0.4);
    pg.addColorStop(0, 'rgba(100, 200, 255, 0.3)');
    pg.addColorStop(1, 'rgba(100, 200, 255, 0)');
    ctx.fillStyle = pg;
    ctx.beginPath();
    ctx.arc(player.px, player.py, cellSize * 0.4, 0, Math.PI * 2);
    ctx.fill();
  }
  
  // Draw particles
  for (let p of particles) {
    ctx.globalAlpha = p.life / 30;
    ctx.fillStyle = p.color;
    ctx.fillRect(p.x - 2, p.y - 2, 4, 4);
    ctx.globalAlpha = 1;
  }
  
  // HUD
  ctx.restore();
  
  if (gameState === 'place') {
    drawPlaceHUD();
  } else if (gameState === 'execute') {
    drawExecuteHUD();
  } else if (gameState === 'start') {
    drawStartScreen();
  } else if (gameState === 'gameOver') {
    drawGameOver();
  } else if (gameState === 'levelComplete') {
    drawLevelComplete();
  } else if (gameState === 'win') {
    drawWin();
  }
}

function drawPlaceHUD() {
  ctx.fillStyle = 'rgba(0,0,0,0.7)';
  ctx.fillRect(0, 0, W, 40);
  ctx.fillStyle = '#fff';
  ctx.font = `${Math.max(14, 16 * SCALE)}px monospace`;
  ctx.textAlign = 'left';
  ctx.fillText(`LEVEL ${level} - PLACEMENT PHASE`, 20, 26);
  ctx.textAlign = 'right';
  ctx.fillStyle = '#7ab8ff';
  ctx.fillText(`Barriers: ${barriers.length}/${maxBarriers}`, W - 20, 26);
  
  // Instructions at bottom
  ctx.textAlign = 'center';
  ctx.fillStyle = '#aaa';
  ctx.font = `${Math.max(12, 14 * SCALE)}px monospace`;
  ctx.fillText('Click grid to place barriers | SPACE to start wave', W/2, H - 20);
  
  // Pattern preview indicator
  ctx.fillStyle = '#f84';
  ctx.fillText('⚠ Bullets incoming - plan your defenses!', W/2, H - 50);
}

function drawExecuteHUD() {
  ctx.fillStyle = 'rgba(0,0,0,0.7)';
  ctx.fillRect(0, 0, W, 40);
  ctx.fillStyle = '#fff';
  ctx.font = `${Math.max(14, 16 * SCALE)}px monospace`;
  ctx.textAlign = 'left';
  ctx.fillText(`LEVEL ${level}`, 20, 26);
  
  ctx.textAlign = 'center';
  let timeLeft = Math.max(0, waveDuration + 60 - waveTimer);
  ctx.fillStyle = timeLeft < 60 ? '#f44' : '#aaa';
  ctx.fillText(`Time: ${Math.ceil(timeLeft/60)}s`, W/2, 26);
  
  ctx.textAlign = 'right';

  ctx.font = `bold ${Math.max(24, 48 * SCALE)}px monospace`;
  ctx.fillText('BULLET HELM', W/2, H * 0.25);
  
  ctx.font = `${Math.max(12, 16 * SCALE)}px monospace`;
  ctx.fillStyle = '#aaa';
  let lines = [
    'A bullet-hell puzzle game',
    '',
    'Each level: Place barriers to block bullet patterns,',
    'then survive and reach the EXIT tile.',
    '',
    'WASD / Arrows - Move',
    'SPACE - Start wave / Phase Shift (invulnerability)',
    'Click - Place barriers during setup',
    '',
    `Best Score: ${bestScore}`,
    '',
    'Press ENTER to start'
  ];
  lines.forEach((line, i) => {
    if (i === 0 || i >= 3 && i <= 8) ctx.fillStyle = '#ddd';
    else ctx.fillStyle = '#888';
    ctx.fillText(line, W/2, H * 0.4 + i * Math.max(20, 24 * SCALE));
  });
  
  // Animated bullets in background
  let t = Date.now() * 0.001;
  for (let i = 0; i < 20; i++) {
    let bx = W/2 + Math.cos(t + i * 0.5) * (100 + i * 10);
    let by = H * 0.6 + Math.sin(t * 1.3 + i * 0.7) * (80 + i * 5);
    ctx.fillStyle = `rgba(255, 100, 68, ${0.3 + Math.sin(t + i) * 0.2})`;
    ctx.beginPath();
    ctx.arc(bx, by, 3 + Math.sin(t*2+i)*1, 0, Math.PI*2);
    ctx.fill();
  }
}
