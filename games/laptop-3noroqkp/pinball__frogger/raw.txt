let audioCtx = null;
function initAudio() {
    if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
}
function playSound(freq, dur, type, vol) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = type || 'sine';
    osc.frequency.value = freq;
    gain.gain.setValueAtTime(vol || 0.3, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + dur);
}

// Game State
let state = 'start'; // start, playing, gameover

function update(dt) {
    if (state !== 'playing') return;
    
    layoutFlippers();
    layoutBumpers();
    layoutWalls();
    layoutSlings();
    
    const tb = getTableBounds();
    
    // Launcher
    if (launcherActive) {
        launcherPower += dt * 2;
        if (keys['Space'] && !ballInPlay) {
            launchBall();
        }
        if (launcherPower > 1) { launcherPower = 0; }
    }
    
    // Flippers
    const flipSpeed = 0.3;
    flippers.left.currentAngle = flippers.left.currentAngle !== undefined ? flippers.left.currentAngle : flippers.left.restAngle;
    flippers.right.currentAngle = flippers.right.currentAngle !== undefined ? flippers.right.currentAngle : flippers.right.restAngle;
    
    if (keys['ArrowLeft'] || keys['KeyA']) {
        if (flippers.left.currentAngle > flippers.left.activeAngle) {
            flippers.left.currentAngle -= flipSpeed;
        }
    } else {
        if (flippers.left.currentAngle < flippers.left.restAngle) {
            flippers.left.currentAngle += flipSpeed * 0.7;
        }
    }
    
    if (keys['ArrowRight'] || keys['KeyD']) {
        if (flippers.right.currentAngle < flippers.right.activeAngle) {
            flippers.right.currentAngle += flipSpeed;
        }
    } else {
        if (flippers.right.currentAngle > flippers.right.restAngle) {
            flippers.right.currentAngle -= flipSpeed * 0.7;
        }
    }
    
    // Ball physics
    if (!ballInPlay) return;
    
    const gravity = 400 * scale;
    ball.vy += gravity * dt;
    ball.x += ball.vx * dt;
    ball.y += ball.vy * dt;
    
    // Bumper collisions
    for (let b of bumpers) {
        const dx = ball.x - b.x;
        const dy = ball.y - b.y;
        const dist = Math.sqrt(dx * dx + dy * dy);
        const minDist = ball.r + b.r;
        if (dist < minDist) {
            const nx = dx / dist;
            const ny = dy / dist;
            ball.x = b.x + nx * minDist;
            ball.y = b.y + ny * minDist;
            const dot = ball.vx * nx + ball.vy * ny;
            ball.vx -= 2 * dot * nx;
            ball.vy -= 2 * dot * ny;
            // Bounce boost
            ball.vx += nx * 5 * scale;
            ball.vy += ny * 5 * scale;
            score += 100 * combo;
            combo = Math.min(combo + 1, 5);
            b.hit = true;
            playSound(600, 0.15, 'square', 0.3);
        }
    }
    
    // Wall collisions
    for (let w of walls) {
        if (w.x2 === w.x1) {
            // Vertical wall
            const wx = w.x1;
            if (Math.abs(ball.x - wx) < ball.r && ball.y > Math.min(w.y1, w.y2) && ball.y < Math.max(w.y1, w.y2)) {
                if (ball.vx * (ball.x - wx) < 0) {
                    ball.x = wx + (ball.x > wx ? ball.r : -ball.r);
                    ball.vx *= -0.8;
                    playSound(300, 0.1, 'sine', 0.2);
                }
            }
        }
    }
    
    // Top wall
    if (ball.y - ball.r < tb.oy) {
        ball.y = tb.oy + ball.r;
        ball.vy = Math.abs(ball.vy) * 0.8;
        playSound(300, 0.1, 'sine', 0.2);
    }
    
    // Slings collisions
    for (let s of slings) {
        const dx = s.x2 - s.x1;
        const dy = s.y2 - s.y1;
        const len = Math.sqrt(dx * dx + dy * dy);
        const nx = -dy / len;
        const ny = dx / len;
        
        const px = ball.x - s.x1;
        const py = ball.y - s.y1;
        const proj = px * (dx / len) + py * (dy / len);
        const perp = px * nx + py * ny;
        
        if (proj > 0 && proj < len && Math.abs(perp) < ball.r) {
            const side = perp > 0 ? 1 : -1;
            ball.y += side * (ball.r - Math.abs(perp)) * ny * -1;
            ball.x += side * (ball.r - Math.abs(perp)) * nx * -1;
            
            const vDotN = ball.vx * nx + ball.vy * ny;
            if (vDotN > 0) {
                ball.vx -= 2 * vDotN * nx;
                ball.vy -= 2 * vDotN * ny;
                ball.vx *= 0.85;
                ball.vy *= 0.85;
                playSound(350, 0.1, 'sine', 0.2);
            }
        }
    }
    
    // Flipper collisions (capsule shape)
    for (let key of ['left', 'right']) {
        const f = flippers[key];
        const angle = f.currentAngle;
        const endX = f.pivotX + Math.cos(angle) * f.len * scale;
        const endY = f.pivotY + Math.sin(angle) * f.len * scale;
        
        // Point-to-segment distance
        const dx = endX - f.pivotX;
        const dy = endY - f.pivotY;
        const len2 = dx * dx + dy * dy;
        let t = ((ball.x - f.pivotX) * dx + (ball.y - f.pivotY) * dy) / len2;
        t = Math.max(0, Math.min(1, t));
        const cx = f.pivotX + t * dx;
        const cy = f.pivotY + t * dy;
        
        const bdx = ball.x - cx;
        const bdy = ball.y - cy;
        const dist = Math.sqrt(bdx * bdx + bdy * bdy);
        const minDist = ball.r + 6 * scale;
        
        if (dist < minDist) {
            const nx = bdx / dist;
            const ny = bdy / dist;
            ball.x = cx + nx * minDist;
            ball.y = cy + ny * minDist;
            
            // Add flipper velocity
            const fActive = (key === 'left' && keys['ArrowLeft']) || (key === 'right' && keys['ArrowRight']);
            if (fActive) {
                const boost = 8 * scale;
                ball.vx += nx * boost;
                ball.vy += ny * boost - 3 * scale;
            } else {
                const dot = ball.vx * nx + ball.vy * ny;
                if (dot < 0) {
                    ball.vx -= 2 * dot * nx * 0.7;
                    ball.vy -= 2 * dot * ny * 0.7;
                }
            }
        }
    }
    
    // Out of bounds (bottom)
    if (ball.y > tb.oy + tb.h + 50 * scale) {
        loseBall();
    }
    
    // Speed limit
    const maxSpeed = 600 * scale;
    const speed = Math.sqrt(ball.vx * ball.vx + ball.vy * ball.vy);
    if (speed > maxSpeed) {
        ball.vx = ball.vx / speed * maxSpeed;
        ball.vy = ball.vy / speed * maxSpeed;
    }
    
    // Bumper hit reset
    for (let b of bumpers) {
        if (b.hit && Math.random() < 0.1) b.hit = false;
    }
}

let lastTime = 0;
