```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Puzzle Asteroids</title>
    <style>
        * { margin: 0; padding: 0; box-sizing: border-box; }
        html, body { width: 100%; height: 100%; overflow: hidden; background: #000; }
        canvas { display: block; width: 100vw; height: 100vh; }
    </style>
</head>
<body>
<canvas id="game"></canvas>
<script>
(function() {
    const canvas = document.getElementById('game');
    const ctx = canvas.getContext('2d');

    let W, H;
    function resize() {
        W = window.innerWidth;
        H = window.innerHeight;
        canvas.width = W;
        canvas.height = H;
    }
    resize();
    window.addEventListener('resize', resize);

    // Audio
    let audioCtx = null;
    function initAudio() {
        if (!audioCtx) audioCtx = new (window.AudioContext || webkitAudioContext)();
    }
    function playSound(type) {
        if (!audioCtx) return;
        const osc = audioCtx.createOscillator();
        const gain = audioCtx.createGain();
        osc.connect(gain);
        gain.connect(audioCtx.destination);
        if (type === 'shoot') {
            osc.type = 'sawtooth';
            osc.frequency.setValueAtTime(400, audioCtx.currentTime);
            osc.frequency.exponentialRampToValueAtTime(100, audioCtx.currentTime + 0.15);
            gain.gain.setValueAtTime(0.15, audioCtx.currentTime);
            gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.15);
            osc.start();
            osc.stop(audioCtx.currentTime + 0.15);
        } else if (type === 'correct') {
            osc.type = 'sine';
            osc.frequency.setValueAtTime(600, audioCtx.currentTime);
            osc.frequency.exponentialRampToValueAtTime(900, audioCtx.currentTime + 0.2);
            gain.gain.setValueAtTime(0.2, audioCtx.currentTime);
            gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.3);
            osc.start();
            osc.stop(audioCtx.currentTime + 0.3);
        } else if (type === 'wrong') {
            osc.type = 'square';
            osc.frequency.setValueAtTime(200, audioCtx.currentTime);
            osc.frequency.exponentialRampToValueAtTime(80, audioCtx.currentTime + 0.3);
            gain.gain.setValueAtTime(0.15, audioCtx.currentTime);
            gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.3);
            osc.start();
            osc.stop(audioCtx.currentTime + 0.3);
        } else if (type === 'explosion') {
            osc.type = 'sawtooth';
            osc.frequency.setValueAtTime(150, audioCtx.currentTime);
            osc.frequency.exponentialRampToValueAtTime(30, audioCtx.currentTime + 0.4);
            gain.gain.setValueAtTime(0.25, audioCtx.currentTime);
            gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.4);
            osc.start();
            osc.stop(audioCtx.currentTime + 0.4);
        } else if (type === 'gameover') {
            osc.type = 'sine';
            osc.frequency.setValueAtTime(400, audioCtx.currentTime);
            osc.frequency.exponentialRampToValueAtTime(100, audioCtx.currentTime + 1);
            gain.gain.setValueAtTime(0.2, audioCtx.currentTime);
            gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 1);
            osc.start();
            osc.stop(audioCtx.currentTime + 1);
        } else if (type === 'wave') {
            osc.type = 'sine';
            osc.frequency.setValueAtTime(500, audioCtx.currentTime);
            osc.frequency.exponentialRampToValueAtTime(800, audioCtx.currentTime + 0.3);
            gain.gain.setValueAtTime(0.15, audioCtx.currentTime);
            gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.4);
            osc.start();
            osc.stop(audioCtx.currentTime + 0.4);
        }
    }

    // Game state
    let gameState = 'start'; // start, playing, gameover
    let score = 0;
    let lives = 3;
    let wave = 1;
    let targetNumber = 1;
    let maxTarget = 1;
    let ship = null;
    let bullets = [];
    let asteroids = [];
    let particles = [];
    let stars = [];
    let messageTimer = 0;
    let messageText = '';
    let invincibleTimer = 0;

    // Stars background
    function initStars() {
        stars = [];
        for (let i = 0; i < 150; i++) {
            stars.push({
                x: Math.random() * W,
                y: Math.random() * H,
                size: Math.random() * 2 + 0.5,
                brightness: Math.random() * 0.7 + 0.3
            });
        }
    }
    initStars();

    // Input
    let keys = {};
    window.addEventListener('keydown', function(e) {
        keys[e.code] = true;
        if (gameState === 'start' && (e.code === 'Space' || e.code === 'Enter')) {
            initAudio();
            startGame();
        }
        if (gameState === 'gameover' && (e.code === 'Space' || e.code === 'Enter')) {
            startGame();
        }
        if (gameState === 'playing' && e.code === 'KeyR') {
            startGame();
        }
    });
    window.addEventListener('keyup', function(e) {
        keys[e.code] = false;
    });

    function startGame() {
        gameState = 'playing';
        score = 0;
        lives = 3;
        wave = 1;
        ship = { x: W / 2, y: H / 2, angle: -Math.PI / 2, vx: 0, vy: 0 };
        bullets = [];
        particles = [];
        invincibleTimer = 2;
        nextWave();
    }

    function nextWave() {
        targetNumber = 1;
        let count = Math.min(3 + wave, 8);
        maxTarget = count;
        asteroids = [];
        let nums = [];
        for (let i = 0; i < count; i++) nums.push(i + 1);
        // Shuffle
        for (let i = nums.length - 1; i > 0; i--) {
            const j = Math.floor(Math.random() * (i + 1));
            [nums[i], nums[j]] = [nums[j], nums[i]];
        }
        for (let i = 0; i < count; i++) {
            let x, y;
            do {
                x = Math.random() * W;
                y = Math.random() * H;
            } while (ship && Math.hypot(x - ship.x, y - ship.y) < 150);
            asteroids.push(createAsteroid(x, y, nums[i], getRadius(count, i)));
        }
        messageText = 'WAVE ' + wave + ' — Destroy in order: 1, 2, 3...';
        messageTimer = 2.5;
        playSound('wave');
    }

    function getRadius(count, idx) {
        let base = Math.min(W, H) * 0.04;
        if (count <= 5) return base + 15;
        if (idx < count / 2) return base + 10;
        return base;
    }

    function createAsteroid(x, y, num, radius) {
        let verts = [];
        let n = Math.floor(Math.random() * 4) + 7;
        for (let i = 0; i < n; i++) {
            let angle = (i / n) * Math.PI * 2;
            let r = radius * (0.75 + Math.random() * 0.3);
            verts.push({ x: Math.cos(angle) * r, y: Math.sin(angle) * r });
        }
        let speed = (0.3 + wave * 0.1) * (Math.min(W, H) / 800);
        let dir = Math.random() * Math.PI * 2;
        return {
            x, y, num, radius, verts,
            vx: Math.cos(dir) * speed,
            vy: Math.sin(dir) * speed,
            rot: Math.random() * Math.PI * 2,
            rotSpeed: (Math.random() - 0.5) * 0.02,
            active: true
        };
    }

    function spawnParticles(x, y, count, color) {
        for (let i = 0; i < count; i++) {
            let angle = Math.random() * Math.PI * 2;
            let speed = Math.random() * 3 + 1;
            particles.push({
                x, y,
                vx: Math.cos(angle) * speed,
                vy: Math.sin(angle) * speed,
                life: 1,
                decay: 0.02 + Math.random() * 0.02,
                size: Math.random() * 3 + 1,
                color: color || '#fff'
            });
        }
    }

    function wrap(obj) {
        if (obj.x < -50) obj.x = W + 50;
        if (obj.x > W + 50) obj.x = -50;
        if (obj.y < -50) obj.y = H + 50;
        if (obj.y > H + 50) obj.y = -50;
    }

    function dist(a, b) {
        return Math.hypot(a.x - b.x, a.y - b.y);
    }

    let shootCooldown = 0;
    let lastTime = 0;

    function update(dt) {
        if (gameState !== 'playing') return;

        // Ship controls
        if (keys['ArrowLeft'] || keys['KeyA']) ship.angle -= 4 * dt;
        if (keys['ArrowRight'] || keys['KeyD']) ship.angle += 4 * dt;
        if (keys['ArrowUp'] || keys['KeyW']) {
            ship.vx += Math.cos(ship.angle) * 200 * dt;
            ship.vy += Math.sin(ship.angle) * 200 * dt;
        }

        // Friction
        ship.vx *= 0.98;
        ship.vy *= 0.98;
        ship.x += ship.vx;
        ship.y += ship.vy;
        wrap(ship);

        // Shooting
        shootCooldown -= dt;
        if ((keys['Space'] || keys['KeyZ']) && shootCooldown <= 0) {
            let bx = ship.x + Math.cos(ship.angle) * 15;
            let by = ship.y + Math.sin(ship.angle) * 15;
            bullets.push({ x: bx, y: by, vx: Math.cos(ship.angle) * 8 + ship.vx * 0.3, vy: Math.sin(ship.angle) * 8 + ship.vy * 0.3, life: 2 });
            shootCooldown = 0.3;
            playSound('shoot');
        }

        // Update bullets
        for (let i = bullets.length - 1; i >= 0; i--) {
            let b = bullets[i];
            b.x += b.vx;
            b.y += b.vy;
            b.life -= dt;
            wrap(b);
            if (b.life <= 0) bullets.splice(i, 1);
        }

        // Update asteroids
        for (let a of asteroids) {
            if (!a.active) continue;
            a.x += a.vx;
            a.y += a.vy;
            a.rot += a.rotSpeed;
            wrap(a);
        }

        // Bullet-asteroid collision
        for (let i = bullets.length - 1; i >= 0; i--) {
            let b = bullets[i];
            for (let a of asteroids) {
                if (!a.active) continue;
                if (dist(b, a) < a.radius + 5) {
                    bullets.splice(i, 1);
                    if (a.num === targetNumber) {
                        // Correct!
                        a.active = false;
                        playSound('correct');
                        score += 100 * wave;
                        spawnParticles(a.x, a.y, 20, '#4fff4f');
                        targetNumber++;
                        if (targetNumber > maxTarget) {
                            // Wave complete!
                            setTimeout(function() {
                                wave++;
                                nextWave();
                            }, 1500);
                            messageText = 'WAVE COMPLETE!';
                            messageTimer = 2;
                            playSound('explosion');
                        }
                    } else {
                        // Wrong asteroid - penalty
                        playSound('wrong');
                        score = Math.max(0, score - 50);
                        spawnParticles(b.x, b.y, 10, '#ff4f4f');
                        messageText = 'WRONG! Need: ' + targetNumber;
                        messageTimer = 1.2;
                    }
                    break;
                }
            }
        }

        // Ship-asteroid collision
        if (invincibleTimer > 0) {
            invincibleTimer -= dt;
        } else {
            for (let a of asteroids) {
                if (!a.active) continue;
                if (dist(ship, a) < a.radius + 12) {
                    lives--;
                    playSound('explosion');
                    spawnParticles(ship.x, ship.y, 30, '#ff8844');
                    if (lives <= 0) {
                        gameState = 'gameover';
                        playSound('gameover');
                    } else {
                        ship.x = W / 2;
                        ship.y = H / 2;
                        ship.vx = 0;
                        ship.vy = 0;
                        ship.angle = -Math.PI / 2;
                        invincibleTimer = 2.5;
                    }
                    break;
                }
            }
        }

        // Update particles
        for (let i = particles.length - 1; i >= 0; i--) {
            let p = particles[i];
            p.x += p.vx;
            p.y += p.vy;
            p.life -= p.decay;
            if (p.life <= 0) particles.splice(i, 1);
        }

        // Message timer
        if (messageTimer > 0) messageTimer -= dt;
    }

    function draw() {
        ctx.fillStyle = '#0a0a1a';
        ctx.fillRect(0, 0, W, H);

        // Stars
        for (let s of stars) {
            ctx.fillStyle = 'rgba(255,255,255,' + s.brightness + ')';
            ctx.fillRect(s.x, s.y, s.size, s.size);
        }

        if (gameState === 'start') {
            drawStartScreen();
            return;
        }

        // Particles
        for (let p of particles) {
            ctx.fillStyle = p.color || '#fff';
            ctx.globalAlpha = p.life;
            ctx.beginPath();
            ctx.arc(p.x, p.y, p.size * p.life, 0, Math.PI * 2);
            ctx.fill();
        }
        ctx.globalAlpha = 1;

        // Asteroids
        for (let a of asteroids) {
            if (!a.active) continue;
            let isTarget = (a.num === targetNumber);
            ctx.save();
            ctx.translate(a.x, a.y);
            ctx.rotate(a.rot);
            ctx.beginPath();
            for (let i = 0; i < a.verts.length; i++) {
                if (i === 0) ctx.moveTo(a.verts[i].x, a.verts[i].y);
                else ctx.lineTo(a.verts[i].x, a.verts[i].y);
            }
            ctx.closePath();
            ctx.fillStyle = isTarget ? '#1a3a1a' : '#2a2a3a';
            ctx.fill();
            ctx.strokeStyle = isTarget ? '#4fff4f' : '#666';
            ctx.lineWidth = isTarget ? 3 : 1.5;
            ctx.stroke();

            // Number
            ctx.rotate(-a.rot);
            ctx.fillStyle = isTarget ? '#4fff4f' : '#aaa';
            ctx.font = 'bold ' + Math.floor(a.radius * 0.8) + 'px monospace';
            ctx.textAlign = 'center';
            ctx.textBaseline = 'middle';
            ctx.fillText(a.num, 0, 2);
            ctx.restore();

            // Glow for target
            if (isTarget) {
                ctx.save();
                ctx.globalAlpha = 0.3 + Math.sin(Date.now() * 0.005) * 0.15;
                ctx.beginPath();
                ctx.arc(a.x, a.y, a.radius + 8, 0, Math.PI * 2);
                ctx.strokeStyle = '#4fff4f';
                ctx.lineWidth = 2;
                ctx.stroke();
                ctx.restore();
            }
        }

        // Bullets
        for (let b of bullets) {
            ctx.fillStyle = '#ff0';
            ctx.beginPath();
            ctx.arc(b.x, b.y, 3, 0, Math.PI * 2);
            ctx.fill();
        }

        // Ship
        if (gameState === 'playing') {
            let blink = invincibleTimer > 0 && Math.floor(Date.now() / 150) % 2 === 0;
            if (!blink) {
                ctx.save();
                ctx.translate(ship.x, ship.y);
                ctx.rotate(ship.angle);
                ctx.beginPath();
                ctx.moveTo(15, 0);
                ctx.lineTo(-10, -10);
                ctx.lineTo(-6, 0);
                ctx.lineTo(-10, 10);
                ctx.closePath();
                ctx.fillStyle = '#fff';
                ctx.fill();
                ctx.strokeStyle = '#8cf';
                ctx.lineWidth = 2;
                ctx.stroke();
                // Thrust
                if (keys['ArrowUp'] || keys['KeyW']) {
                    ctx.beginPath();
                    ctx.moveTo(-6, -5);
                    ctx.lineTo(-14, 0);
                    ctx.lineTo(-6, 5);
                    ctx.fillStyle = '#f80';
                    ctx.fill();
                }
                ctx.restore();
            }
        }

        // HUD
        if (gameState === 'playing') {
            ctx.fillStyle = '#fff';
            ctx.font = '18px monospace';
            ctx.textAlign = 'left';
            ctx.fillText('SCORE: ' + score, 20, 30);
            ctx.fillText('WAVE: ' + wave, 20, 55);
            ctx.fillText('TARGET: ' + targetNumber, 20, 80);
            // Lives
            for (let i = 0; i < lives; i++) {
                ctx.save();
                ctx.translate(W - 40 - i * 35, 35);
                ctx.rotate(-Math.PI / 2);
                ctx.beginPath();
                ctx.moveTo(12, 0);
                ctx.lineTo(-8, -8);
                ctx.lineTo(-5, 0);
                ctx.lineTo(-8, 8);
                ctx.closePath();
                ctx.fillStyle = '#fff';
                ctx.fill();
                ctx.restore();
            }
        }

        // Message
        if (messageTimer > 0) {
            ctx.fillStyle = 'rgba(255,255,255,' + Math.min(messageTimer, 1) + ')';
            ctx.font = 'bold 24px monospace';
            ctx.textAlign = 'center';
            ctx.fillText(messageText, W / 2, H - 60);
        }

        // Game Over
        if (gameState === 'gameover') {
            ctx.fillStyle = 'rgba(0,0,0,0.6)';
            ctx.fillRect(0, 0, W, H);
            ctx.fillStyle = '#ff4444';
            ctx.font = 'bold 48px monospace';
            ctx.textAlign = 'center';
            ctx.fillText('GAME OVER', W / 2, H / 2 - 40);
            ctx.fillStyle = '#fff';
            ctx.font = '24px monospace';
            ctx.fillText('Score: ' + score, W / 2, H / 2 + 10);
            ctx.fillText('Wave reached: ' + wave, W / 2, H / 2 + 45);
            ctx.fillStyle = '#8cf';
            ctx.font = '18px monospace';
            ctx.fillText('Press SPACE or ENTER to restart', W / 2, H / 2 + 100);
        }
    }

    function drawStartScreen() {
        ctx.fillStyle = '#fff';
        ctx.font = 'bold 42px monospace';
        ctx.textAlign = 'center';
        ctx.fillText('PUZZLE ASTEROIDS', W / 2, H / 3);

        ctx.font = '18px monospace';
        ctx.fillStyle = '#aaa';
        let lines = [
            'Destroy asteroids IN NUMERICAL ORDER!',
            '',
            '⬆ / W — Thrust',
            '⬅ / A — Rotate Left',
            '➡ / D — Rotate Right',
            'SPACE / Z — Shoot',
            'R — Restart',
            '',
            'The glowing green asteroid shows the next target.',
            'Shoot wrong ones and lose points!',
            '',
            'Press SPACE or ENTER to Start'
        ];
        for (let i = 0; i < lines.length; i++) {
            ctx.fillText(lines[i], W / 2, H / 3 + 60 + i * 28);
        }

        // Decorative asteroids
        let t = Date.now() * 0.001;
        for (let i = 0; i < 5; i++) {
            let x = W / 2 + Math.cos(t + i * 1.3) * (W * 0.3);
            let y = H / 2 + Math.sin(t * 0.7 + i * 1.3) * (H * 0.3);
            ctx.save();
            ctx.translate(x, y);
            ctx.rotate(t + i);
            ctx.beginPath();
            for (let j = 0; j < 6; j++) {
                let a2 = (j / 6) * Math.PI * 2;
                let r = 15 + Math.sin(a2 * 3) * 5;
                if (j === 0) ctx.moveTo(Math.cos(a2) * r, Math.sin(a2) * r);
                else ctx.lineTo(Math.cos(a2) * r, Math.sin(a2) * r);
            }
            ctx.closePath();
            ctx.strokeStyle = '#4a4a6a';
            ctx.lineWidth = 1.5;
            ctx.stroke();
            ctx.restore();
        }
    }

    function gameLoop(time) {
        let dt = Math.min((time - lastTime) / 1000, 0.05);
        lastTime = time;
        update(dt);
        draw();
        requestAnimationFrame(gameLoop);
    }
    requestAnimationFrame(gameLoop);

})();
</script>
</body>
</html>
```
