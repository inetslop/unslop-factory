function tryMove(playerIdx, dx, dy) {
    const p = players[playerIdx];
    const nx = p.x + dx;
    const ny = p.y + dy;

    // Can't walk into wall or other player
    if (isWall(nx, ny)) return;
    if (hasPlayerAt(nx, ny, playerIdx)) return;

    const boxIdx = hasBoxAt(nx, ny);
    if (boxIdx !== -1) {
        // Try to push box
        const bx = nx + dx;
        const by = ny + dy;
        // Can't push into wall, another box, or a player
        if (isWall(bx, by)) return;
        if (hasBoxAt(bx, by) !== -1) return;
        if (players[0].x === bx && players[0].y === by) return;
        if (players[1].x === bx && players[1].y === by) return;
        // Push the box
        boxes[boxIdx].x = bx;
        boxes[boxIdx].y = by;
    }

    p.x = nx;
    p.y = ny;
    moves++;
    playMoveSound();
    checkWin();
}

            ctx.fillText('Press ENTER for next level', W/2, H * 0.55);
        } else {
            ctx.fillText('YOU BEAT ALL LEVELS!', W/2, H * 0.35);
            ctx.fillStyle = '#e9c46a';
            ctx.font = `${Math.min(W*0.035, 28)}px 'Segoe UI', sans-serif`;
            ctx.fillText('🏆 Cooperative Victory! 🏆', W/2, H * 0.45);
            ctx.fillStyle = '#ccd6f6';
            ctx.font = `${Math.min(W*0.03, 24)}px 'Segoe UI', sans-serif`;
            ctx.fillText(`Total: ${moves} moves`, W/2, H * 0.53);
            ctx.fillText('Press ENTER to play again', W/2, H * 0.61);
        }
    }
}

function drawParticles() {
    particles.forEach(p => {
        ctx.globalAlpha = p.life;
        ctx.fillStyle = p.color;
        ctx.beginPath();
        ctx.arc(p.x, p.y, 4, 0, Math.PI*2);
        ctx.fill();
    });
    ctx.globalAlpha = 1;
}

// Input
window.addEventListener('keydown', (e) => {
    if (!audioCtx) initAudio();

    if (gameState === 'start') {
        if (e.key === 'Enter' || e.key === ' ') {
            gameState = 'playing';
            loadLevel(currentLevel);
            playStartSound();
        }
        return;
    }

    if (gameState === 'won') {
        if (e.key === 'Enter' || e.key === ' ') {
            if (currentLevel < LEVELS.length - 1) {
                currentLevel++;
                loadLevel(currentLevel);
            } else {
                currentLevel = 0;
                loadLevel(currentLevel);
            }
            gameState = 'playing';
            particles = [];
        }
        return;
    }

    if (gameState === 'playing') {
        // Player 1: WASD
        switch(e.key) {
            case 'w': case 'W': tryMove(0, 0, -1); break;
            case 'a': case 'A': tryMove(0, -1, 0); break;
            case 's': case 'S': tryMove(0, 0, 1); break;
            case 'd': case 'D': tryMove(0, 1, 0); break;
        }

        // Player 2: Arrow keys
        switch(e.key) {
            case 'ArrowUp': e.preventDefault(); tryMove(1, 0, -1); break;
            case 'ArrowDown': e.preventDefault(); tryMove(1, 0, 1); break;
            case 'ArrowLeft': e.preventDefault(); tryMove(1, -1, 0); break;
            case 'ArrowRight': e.preventDefault(); tryMove(1, 1, 0); break;
        }

        // Restart
        if (e.key === 'r' || e.key === 'R') {
            loadLevel(currentLevel);
        }
    }
});

// Game loop
function gameLoop() {
    draw();
    requestAnimationFrame(gameLoop);
}

// Init
window.addEventListener('resize', resize);
resize();
loadLevel(0);
gameState = 'start';
gameLoop();
