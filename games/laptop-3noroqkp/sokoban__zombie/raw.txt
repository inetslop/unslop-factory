function move(dx,dy){
  const nx=player.x+dx, ny=player.y+dy;
  if(isWall(nx,ny)) return;
  const box = boxAt(nx,ny);
  if(box){
    const bx=nx+dx, by=ny+dy;
    if(isWall(bx,by)||boxAt(bx,by)){ playSound('fail'); return; }

      addParticles(bx,by);
    }

    playSound('move');

  player.x=nx; player.y=ny;
  moves++;
  checkWin();
}

  }
}

          ctx.strokeRect(x+1,y+1,cs-2,cs-2);
        } else {
          ctx.fillStyle=(r+c)%2===0?'#151520':'#121218';
          ctx.fillRect(x,y,cs,cs);
        }
      }
    }

    // Draw targets
    for(const t of targets){
      drawTarget(off.x+t.x*cs+cs/2, off.y+t.y*cs+cs/2, cs, t);
    }

    // Draw boxes (zombies)
    for(const b of boxes){
      const done=onTarget(b.x,b.y);
      drawZombieFace(off.x+b.x*cs+cs/2, off.y+b.y*cs+cs/2, cs, done);
      if(done){
        ctx.fillStyle=`hsla(120,70%,40%,${0.3+Math.sin(t*4)*0.1})`;
        ctx.fillRect(off.x+b.x*cs,off.y+b.y*cs,cs,cs);
      }
    }

    // Draw player
    drawPlayer(off.x+player.x*cs+cs/2, off.y+player.y*cs+cs/2, cs);

    // Particles
    for(let i=particles.length-1;i>=0;i--){
      const p=particles[i];
      p.x+=p.vx;p.y+=p.vy;p.vy+=0.05;p.life-=0.03;
      if(p.life<=0){particles.splice(i,1);continue;}
      ctx.fillStyle=p.color;ctx.globalAlpha=p.life;
      ctx.fillRect(p.x-2,p.y-2,4,4);
      ctx.globalAlpha=1;
    }

    // HUD
    const doneCount=boxes.filter(b=>onTarget(b.x,b.y)).length;
    ctx.textAlign='left';
    ctx.fillStyle='#8f8';
    ctx.font=`${Math.min(W*0.025,16)}px Courier New`;
    ctx.fillText(`Level ${currentLevel+1}: ${LEVELS[currentLevel].name}`,off.x, off.y-20);
    ctx.fillText(`Zombies contained: ${doneCount}/${boxes.length}   Moves: ${moves}   Pushes: ${pushes}`,off.x, off.y-45);

    if(state==='levelComplete'){
      ctx.fillStyle='rgba(0,0,0,0.7)';ctx.fillRect(0,0,W,H);
      ctx.textAlign='center';
      ctx.fillStyle='#4a9e4a';
      ctx.font=`bold ${Math.min(W*0.06,36)}px Courier New`;
      ctx.fillText('LEVEL CLEARED!',W/2,H*0.4);
      ctx.fillStyle='#8f8';
      ctx.font=`${Math.min(W*0.03,18)}px Courier New`;
      ctx.fillText(`Moves: ${moves}  |  Pushes: ${pushes}`,W/2,H*0.5);
      ctx.fillStyle=`hsla(120,60%,${50+Math.sin(t*3)*20}%,1)`;
      ctx.font=`bold ${Math.min(W*0.035,22)}px Courier New`;
      if(currentLevel<LEVELS.length-1) ctx.fillText('[ PRESS N FOR NEXT LEVEL ]',W/2,H*0.65);
      else ctx.fillText('[ ALL ZOMBIES CONTAINED! ]',W/2,H*0.65);
    }
  }
  else if(state==='allComplete'){
    // This is handled via levelComplete on last level
  }

  requestAnimationFrame(draw);
}

  initAudio();
  if(state==='start'){
    state='playing';
    currentLevel=0;
    loadLevel(0);
    e.preventDefault();
    return;
  }
  if(state==='levelComplete'){
    if(e.key==='n'||e.key==='N'||e.key===' '){
      if(currentLevel<LEVELS.length-1){
        currentLevel++;
        loadLevel(currentLevel);
        state='playing';
      } else {
        // All complete - back to start
        state='start';
      }
    }
    if(e.key==='r'||e.key==='R'){loadLevel(currentLevel);state='playing';}
    e.preventDefault();
    return;
  }
  if(state==='playing'){
    let dx=0,dy=0;
    if(e.key==='ArrowUp'||e.key==='w'||e.key==='W') dy=-1;
    else if(e.key==='ArrowDown'||e.key==='s'||e.key==='S') dy=1;
    else if(e.key==='ArrowLeft'||e.key==='a'||e.key==='A') dx=-1;
    else if(e.key==='ArrowRight'||e.key==='d'||e.key==='D') dx=1;
    else if(e.key==='r'||e.key==='R'){loadLevel(currentLevel);e.preventDefault();return;}
    else return;
    move(dx,dy);
    e.preventDefault();
  }
});
