<script>
const canvas = document.getElementById('c');
const ctx = canvas.getContext('2d');

let W, H;
function resize() {
    W = window.innerWidth;
    H = window.innerHeight;
    canvas.width = W;
    canvas.height = H;
}
window.addEventListener('resize', resize);
resize();

// Audio
let audioCtx = null;
function initAudio() {
    if (!audioCtx) {
        try { audioCtx = new (window.AudioContext || window.webkitAudioContext)(); } catch(e) {}
    }
}
function playTone(freq, dur, vol) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = 'square';
    osc.frequency.value = freq;
    gain.gain.setValueAtTime(vol || 0.1, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + dur);
}
function sfxPaddle() { playTone(220, 0.08, 0.15); }
function sfxBlock() { playTone(440, 0.06, 0.2); }
function sfxWall() { playTone(330, 0.04, 0.1); }
function sfxWin() { for(let i=0;i<5;i++) setTimeout(()=>playTone(440+i*110,0.15,0.2),i*80); }

// Game state
const ROWS = 5, COLS = 8;
const BLOCK_COLORS = ['#ff4444','#ff8833','#ffcc33','#44dd44','#44aaff'];

let gameState = 'start'; // start, playing, won
let winner = null;
let frameCount = 0;

class Player {
    constructor(id) {
        this.id = id;
        this.blocksRemaining = ROWS * COLS;
        this.reset();
    }
    reset() {
        this.blocksRemaining = ROWS * COLS;
        this.blocks = [];
        for (let r = 0; r < ROWS; r++) {
            for (let c = 0; c < COLS; c++) {
                this.blocks.push({ row: r, col: c, alive: true });
            }
        }
        this.ballLaunched = false;
    }
}

let players = [new Player(0), new Player(1)];
let paddles = [{}, {}];
let balls = [{}, {}];

function setupGame() {
    for (let i = 0; i < 2; i++) {
        const halfW = W / 2;
        players[i].reset();
        paddles[i] = {
            x: halfW / 2 - (halfW * 0.15),
            w: halfW * 0.3,
            h: Math.max(12, H * 0.02)
        };
        balls[i] = {
            x: halfW / 2,
            y: H * 0.75,
            r: Math.max(6, W * 0.008),
            dx: (i === 0 ? 1 : -1) * 3.5,
            dy: -4,
            speed: Math.max(3.5, H * 0.005)
        };
    }
}

function getBlockRect(playerId, row, col) {
    const halfW = W / 2;
    const offsetX = playerId * halfW;
    const blockAreaTop = H * 0.06;
    const blockAreaBottom = H * 0.32;
    const margin = Math.max(10, halfW * 0.04);
    const availW = halfW - margin * 2;
    const availH = blockAreaBottom - blockAreaTop;
    const bw = availW / COLS;
    const bh = availH / ROWS;
    const gap = Math.max(2, W * 0.003);
    return {
        x: offsetX + margin + col * bw + gap/2,
        y: blockAreaTop + row * bh + gap/2,
        w: bw - gap,
        h: bh - gap
    };
}

function getHalfBounds(i) {
    const halfW = W / 2;
    return { left: i * halfW, right: (i+1) * halfW, top: 0, bottom: H };
}

const keys = {};
window.addEventListener('keydown', e => {
    keys[e.code] = true;
    initAudio();
    if (gameState === 'start') {
        if (e.code === 'Space' || e.code === 'Enter') {
            startGame();
        }
    } else if (gameState === 'playing') {
        if (e.code === 'KeyR') restartGame();
        // Launch balls
        if (e.code === 'Space') launchBall(0);
        if (e.code === 'Enter') launchBall(1);
    } else if (gameState === 'won') {
        if (e.code === 'Space' || e.code === 'Enter' || e.code === 'KeyR') restartGame();
    }
});
window.addEventListener('keyup', e => { keys[e.code] = false; });

function startGame() {
    gameState = 'playing';
    winner = null;
    players[0].reset();
    players[1].reset();
    setupGame();
}
function restartGame() {
    gameState = 'start';
    winner = null;
    players[0].reset();
    players[1].reset();
    setupGame();
}

function launchBall(i) {
    if (gameState !== 'playing') return;
    if (!players[i].ballLaunched) {
        players[i].ballLaunched = true;
        const b = balls[i];
        const halfW = W / 2;
        b.x = i * halfW + halfW / 2;
        b.y = H * 0.75;
        b.dx = (Math.random() - 0.5) * 2 || 1;
        b.dy = -b.speed;
        const mag = Math.sqrt(b.dx * b.dx + b.dy * b.dy);
        if (mag === 0) { b.dx = 1; b.dy = -1; }
    }
}

function update(i) {
    if (gameState !== 'playing') return;
    const halfW = W / 2;
    const offsetX = i * halfW;
    const p = players[i];
    const paddle = paddles[i];
    const ball = balls[i];

    // Paddle movement
    const pSpeed = Math.max(6, W * 0.01);
    if (i === 0) {
        if (keys['KeyA']) paddle.x -= pSpeed;
        if (keys['KeyD']) paddle.x += pSpeed;
    } else {
        if (keys['ArrowLeft']) paddle.x -= pSpeed;
        if (keys['ArrowRight']) paddle.x += pSpeed;
    }
    paddle.x = Math.max(offsetX + 5, Math.min(offsetX + halfW - paddle.w - 5, paddle.x));

    if (!p.ballLaunched) return;

    // Ball movement
    ball.x += ball.dx;
    ball.y += ball.dy;

    const left = offsetX;
    const right = offsetX + halfW;
    const top = 0;
    const bottom = H;

    // Wall collisions
    if (ball.x - ball.r <= left) { ball.x = left + ball.r; ball.dx = Math.abs(ball.dx); sfxWall(); }
    if (ball.x + ball.r >= right) { ball.x = right - ball.r; ball.dx = -Math.abs(ball.dx); sfxWall(); }
    if (ball.y - ball.r <= top) { ball.y = top + ball.r; ball.dy = Math.abs(ball.dy); sfxWall(); }

    // Paddle collision
    const paddleY = bottom - paddle.h - 10;
    if (ball.dy > 0 && ball.y + ball.r >= paddleY && ball.y + ball.r <= paddleY + paddle.h + ball.r) {
        if (ball.x >= paddle.x && ball.x <= paddle.x + paddle.w) {
            ball.y = paddleY - ball.r;
            const hitPos = (ball.x - paddle.x) / paddle.w;
            ball.dx = (hitPos - 0.5) * ball.speed * 1.2;
            ball.dy = -Math.sqrt(ball.speed * ball.speed - ball.dx * ball.dx);
            if (Math.abs(ball.dy) < ball.speed * 0.3) ball.dy = -ball.speed * 0.6;
            sfxPaddle();
        }
    }

    // Block collisions
    for (let b of p.blocks) {
        if (!b.alive) continue;
        const rect = getBlockRect(i, b.row, b.col);
        if (ball.x + ball.r > rect.x && ball.x - ball.r < rect.x + rect.w &&
            ball.y + ball.r > rect.y && ball.y - ball.r < rect.y + rect.h) {
            b.alive = false;
            p.blocksRemaining--;
            sfxBlock();

            // Determine bounce direction
            const overlapL = (ball.x + ball.r) - rect.x;
            const overlapR = (rect.x + rect.w) - (ball.x - ball.r);
            const overlapT = (ball.y + ball.r) - rect.y;
            const overlapB = (rect.y + rect.h) - (ball.y - ball.r);
            const minOverlap = Math.min(overlapL, overlapR, overlapT, overlapB);

            if (minOverlap === overlapT || minOverlap === overlapB) {
                ball.dy = -ball.dy;
            } else {
                ball.dx = -ball.dx;
            }

            // Normalize speed
            const mag = Math.sqrt(ball.dx * ball.dx + ball.dy * ball.dy);
            const targetSpeed = Math.max(3.5, H * 0.005);
            if (mag > 0) {
                ball.dx = (ball.dx / mag) * targetSpeed;
                ball.dy = (ball.dy / mag) * targetSpeed;
            }

            break; // one block per frame
        }
    }

    // Ball lost (below screen)
    if (ball.y - ball.r > bottom) {
        p.ballLaunched = false;
        const halfW2 = W / 2;
        ball.x = i * halfW2 + halfW2 / 2;
        ball.y = H * 0.75;
        ball.dx = (Math.random() - 0.5) * 2 || 1;
        ball.dy = -ball.speed;
    }

    // Win check
    if (p.blocksRemaining <= 0 && gameState === 'playing') {
        gameState = 'won';
        winner = i;
        sfxWin();
    }
}

function drawBlock(x, y, w, h, color) {
    ctx.fillStyle = color;
    ctx.beginPath();
    ctx.roundRect(x, y, w, h, 3);
    ctx.fill();
    // Highlight
    ctx.fillStyle = 'rgba(255,255,255,0.15)';
    ctx.fillRect(x + 2, y + 2, w - 4, h * 0.4);
}

function draw() {
    ctx.clearRect(0, 0, W, H);

    if (gameState === 'start') {
        drawStartScreen();
        return;
    }

    const halfW = W / 2;

    for (let i = 0; i < 2; i++) {
        const offsetX = i * halfW;

        // Background
        ctx.fillStyle = '#0a0a1a';
        ctx.fillRect(offsetX, 0, halfW, H);

        // Blocks
        const p = players[i];
        for (let b of p.blocks) {
            if (!b.alive) continue;
            const rect = getBlockRect(i, b.row, b.col);
            drawBlock(rect.x, rect.y, rect.w, rect.h, BLOCK_COLORS[b.row]);
        }

        // Paddle
        const paddleY = H - paddles[i].h - 10;
        ctx.fillStyle = i === 0 ? '#44ddff' : '#ff66aa';
        ctx.beginPath();
        ctx.roundRect(paddles[i].x, paddleY, paddles[i].w, paddles[i].h, 5);
        ctx.fill();

        // Ball
        const ball = balls[i];
        if (p.ballLaunched || true) {
            ctx.fillStyle = '#ffffff';
            ctx.shadowColor = '#ffffff';
            ctx.shadowBlur = 8;
            ctx.beginPath();
            ctx.arc(ball.x, ball.y, ball.r, 0, Math.PI * 2);
            ctx.fill();
            ctx.shadowBlur = 0;
        }

        // Score
        ctx.fillStyle = i === 0 ? '#44ddff' : '#ff66aa';
        ctx.font = `bold ${Math.max(14, W*0.018)}px monospace`;
        ctx.textAlign = 'left';
        ctx.fillText(`P${i+1}: ${p.blocksRemaining}`, offsetX + 15, H * 0.035);

        // Launch hint
        if (!p.ballLaunched) {
            ctx.fillStyle = 'rgba(255,255,255,0.6)';
            ctx.font = `${Math.max(12, W*0.012)}px monospace`;
            ctx.textAlign = 'center';
            const keyHint = i === 0 ? 'SPACE' : 'ENTER';
            ctx.fillText(`Press ${keyHint} to launch`, offsetX + halfW/2, H * 0.82);
        }
    }

    // Divider
    ctx.strokeStyle = '#333';
    ctx.lineWidth = 2;
    ctx.setLineDash([8, 6]);
    ctx.beginPath();
    ctx.moveTo(halfW, 0);
    ctx.lineTo(halfW, H);
    ctx.stroke();
    ctx.setLineDash([]);

    // Won screen
    if (gameState === 'won') {
        ctx.fillStyle = 'rgba(0,0,0,0.7)';
        ctx.fillRect(0, 0, W, H);
        ctx.fillStyle = winner === 0 ? '#44ddff' : '#ff66aa';
        ctx.font = `bold ${Math.max(28, W*0.04)}px monospace`;
        ctx.textAlign = 'center';
        ctx.fillText(`PLAYER ${winner + 1} WINS!`, W/2, H/2 - 20);
        ctx.fillStyle = '#ffffff';
        ctx.font = `${Math.max(14, W*0.018)}px monospace`;
        ctx.fillText('Press SPACE or ENTER to play again', W/2, H/2 + 30);
    }
}

function drawStartScreen() {
    ctx.fillStyle = '#0a0a1a';
    ctx.fillRect(0, 0, W, H);

    const cx = W / 2;
    const cy = H / 2;

    // Title
    ctx.fillStyle = '#ffffff';
    ctx.font = `bold ${Math.max(32, W*0.05)}px monospace`;
    ctx.textAlign = 'center';
    ctx.fillText('BREAKOUT DUEL', cx, cy - H * 0.18);

    // Subtitle
    ctx.fillStyle = '#888';
    ctx.font = `${Math.max(13, W*0.015)}px monospace`;
    ctx.fillText('Two-player Breakout race — first to clear all blocks wins!', cx, cy - H * 0.12);

    // Controls
    const lineH = Math.max(24, H * 0.035);
    let y = cy - H * 0.02;

    ctx.fillStyle = '#44ddff';
    ctx.font = `bold ${Math.max(16, W*0.02)}px monospace`;
    ctx.fillText('PLAYER 1', cx - W * 0.15, y);
    ctx.fillStyle = '#ffffff';
    ctx.font = `${Math.max(13, W*0.014)}px monospace`;
    ctx.fillText('A / D — Move Paddle', cx - W * 0.15, y + lineH);
    ctx.fillText('SPACE — Launch Ball', cx - W * 0.15, y + lineH * 2);

    ctx.fillStyle = '#ff66aa';
    ctx.font = `bold ${Math.max(16, W*0.02)}px monospace`;
    ctx.fillText('PLAYER 2', cx + W * 0.15, y);
    ctx.fillStyle = '#ffffff';
    ctx.font = `${Math.max(13, W*0.014)}px monospace`;
    ctx.fillText('← / → — Move Paddle', cx + W * 0.15, y + lineH);
    ctx.fillText('ENTER — Launch Ball', cx + W * 0.15, y + lineH * 2);

    // Start prompt
    ctx.fillStyle = `rgba(255,255,255,${0.5 + 0.5 * Math.sin(frameCount * 0.05)})`;
    ctx.font = `bold ${Math.max(18, W*0.025)}px monospace`;
    ctx.fillText('Press SPACE to Start', cx, cy + H * 0.15);

    // R hint
    ctx.fillStyle = '#555';
    ctx.font = `${Math.max(11, W*0.011)}px monospace`;
    ctx.fillText('R — Restart at any time', cx, cy + H * 0.2);
}

let lastTime = 0;
function gameLoop(timestamp) {
    frameCount++;
    update(0);
    update(1);
    draw();
    requestAnimationFrame(gameLoop);
}

setupGame();
requestAnimationFrame(gameLoop);
</script>
