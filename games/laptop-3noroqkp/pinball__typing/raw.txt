let audioCtx = null;
function initAudio() { if (!audioCtx) audioCtx = new (window.AudioContext || webkitAudioContext)(); }
function playSound(freq, dur, type='sine', vol=0.3) {
    if (!audioCtx) return;
    const o = audioCtx.createOscillator();
    const g = audioCtx.createGain();
    o.type = type; o.frequency.value = freq;
    g.gain.setValueAtTime(vol, audioCtx.currentTime);
    g.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
    o.connect(g); g.connect(audioCtx.destination);
    o.start(); o.stop(audioCtx.currentTime + dur);
}

// Game state
let gameState = 'start'; // start, playing, gameover
let score = 0;
let lives = 3;
let combo = 0;
let highScore = 0;
try { highScore = parseInt(localStorage.getItem('typingPinballHigh') || '0'); } catch(e) {}

// Words
const WORDS = ['WORD','TIME','FIRE','STAR','GLOW','NEON','BYTE','CODE','DATA','HEX','PIXEL','RUSH','JOLT','FLUX','NOVA','PULSE','DRIFT','SURGE','BLAZE','QUICK','RAPID','SWIFT','BRIGHT','SHARP','DEEP','WIDE','TALL','SMALL','FAST','SLOW','BIG','TINY','LOUD','QUIET','COLD','WARM','DARK','LIGHT','SOFT','HARD','NEW','OLD','TRUE','FREE','MIX','FLOW'];

const BALL_R = 12;
const GRAVITY = 0.35;

// Ball

// Flippers
const LEFT_FLIPPER = { x1: 200, y1: 800, x2: 350, y2: 790, angle: 0.4, angVel: 0, targetAngle: 0.4 };
const RIGHT_FLIPPER = { x1: 450, y1: 790, x2: 600, y2: 800, angle: -0.4, angVel: 0, targetAngle: -0.4 };

// Targets
let targets = [];
let activeTarget = null; // { word, progress, timer, maxTime }
let particles = [];
let floatTexts = [];

function spawnTarget() {
    const word = WORDS[Math.floor(Math.random() * WORDS.length)];
    const x = TABLE.x + 80 + Math.random() * (TABLE.w - 160);
    const y = TABLE.y + 60 + Math.random() * (TABLE.h - 250);
    targets.push({ word, x, y, r: 35, pulse: 0 });
}

function resetBall() {
    ball.x = 400; ball.y = 750; ball.vx = 0; ball.vy = 0;
    ball.active = false; ball.launched = false;
}

function startGame() {
    score = 0; lives = 3; combo = 0;
    targets = []; activeTarget = null;
    particles = []; floatTexts = [];
    resetBall();
    for (let i = 0; i < 3; i++) spawnTarget();
    gameState = 'playing';
}

function launchBall() {
    if (!ball.active) {
        ball.active = true; ball.launched = true;
        ball.vx = (Math.random() - 0.5) * 4;
        ball.vy = -12;
        playSound(200, 0.2, 'square', 0.2);
    }
}

function loseLife() {
    lives--;
    combo = 0;
    if (activeTarget) { activeTarget = null; }
    floatTexts.push({ text: 'MISS!', x: ball.x, y: ball.y, life: 60, color: '#ff4444' });
    playSound(150, 0.4, 'sawtooth', 0.3);
    if (lives <= 0) {
        gameState = 'gameover';
        try { if (score > highScore) { highScore = score; localStorage.setItem('typingPinballHigh', highScore); } } catch(e) {}
    } else {
        setTimeout(resetBall, 800);
    }
}

function typeLetter(ch) {
    if (!activeTarget || !activeTarget.word) return;
    const nextChar = activeTarget.word[activeTarget.progress].toLowerCase();
    if (ch.toLowerCase() === nextChar) {
        activeTarget.progress++;
        playSound(400 + activeTarget.progress * 80, 0.1, 'sine', 0.2);
        if (activeTarget.progress >= activeTarget.word.length) {
            // Success!

            score += pts;

                particles.push({ x: activeTarget.x, y: activeTarget.y, vx: Math.cos(a) * (2+Math.random()*3), vy: Math.sin(a) * (2+Math.random()*3), life: 40, color: '#44ff88' });
            }

// Physics
function updatePhysics() {
    if (!ball.active) return;

    ball.vy += GRAVITY;
    ball.x += ball.vx;
    ball.y += ball.vy;

    // Wall collisions
    if (ball.x - BALL_R < TABLE.x) { ball.x = TABLE.x + BALL_R; ball.vx = Math.abs(ball.vx) * 0.8; playSound(300, 0.1, 'square', 0.1); }
    if (ball.x + BALL_R > TABLE.x + TABLE.w) { ball.x = TABLE.x + TABLE.w - BALL_R; ball.vx = -Math.abs(ball.vx) * 0.8; playSound(300, 0.1, 'square', 0.1); }
    if (ball.y - BALL_R < TABLE.y) { ball.y = TABLE.y + BALL_R; ball.vy = Math.abs(ball.vy) * 0.8; playSound(300, 0.1, 'square', 0.1); }

    // Top arch
    const cx = TABLE.x + TABLE.w / 2, cy = TABLE.y;
    const dx = ball.x - cx, dy = ball.y - cy;
    const dist = Math.sqrt(dx*dx + dy*dy);
    if (dist < BALL_R + 40 && ball.y < TABLE.y + 80) {
        // skip arch for simplicity, use flat top wall
    }

    // Bumpers
    const bumpers = [
        { x: 250, y: 350, r: 30 },
        { x: 550, y: 350, r: 30 },
        { x: 400, y: 200, r: 25 }
    ];
    for (const b of bumpers) {
        const dx = ball.x - b.x, dy = ball.y - b.y;
        const dist = Math.sqrt(dx*dx + dy*dy);
        if (dist < BALL_R + b.r) {
            const nx = dx / dist, ny = dy / dist;
            ball.x = b.x + nx * (BALL_R + b.r);
            ball.y = b.y + ny * (BALL_R + b.r);
            const dot = ball.vx * nx + ball.vy * ny;
            ball.vx -= 2 * dot * nx;
            ball.vy -= 2 * dot * ny;
            ball.vx *= 1.2; ball.vy *= 1.2;
            playSound(500, 0.15, 'square', 0.2);
            for (let i = 0; i < 5; i++) {
                const a = Math.random() * Math.PI * 2;
                particles.push({ x: ball.x, y: ball.y, vx: Math.cos(a)*3, vy: Math.sin(a)*3, life: 20, color: '#ffff44' });
            }
        }
    }

    // Target collision
    for (const t of targets) {
        const dx = ball.x - t.x, dy = ball.y - t.y;
        const dist = Math.sqrt(dx*dx + dy*dy);
        if (dist < BALL_R + t.r && !activeTarget) {
            activeTarget = { word: t.word, progress: 0, timer: t.word.length * 60, maxTime: t.word.length * 60 };
            playSound(700, 0.15, 'sine', 0.3);
        } else if (dist < BALL_R + t.r && activeTarget === undefined) {
            // bounce off
        }
    }

    // Flipper collision
    function flipperCollide(flipper, isLeft) {
        const fx = isLeft ? LEFT_FLIPPER.x1 : RIGHT_FLIPPER.x1;
        const fy = isLeft ? LEFT_FLIPPER.y1 : RIGHT_FLIPPER.y1;
        const angle = flipper.angle;
        const len = 150;
        const tipX = fx + Math.cos(angle) * (isLeft ? len : -len);
        const tipY = fy + Math.sin(angle) * (isLeft ? len : -len);

        // Line segment vs circle
        const ax = ball.x, ay = ball.y;
        const bx = fx, by = fy;
        const cx2 = tipX, cy2 = tipY;
        const ab = Math.sqrt((cx2-bx)**2 + (cy2-by)**2);
        let t2 = ((ax-bx)*(cx2-bx) + (ay-by)*(cy2-by)) / (ab*ab);
        t2 = Math.max(0, Math.min(1, t2));
        const px = bx + t2*(cx2-bx), py = by + t2*(cy2-by);
        const dist = Math.sqrt((ax-px)**2 + (ay-py)**2);
        if (dist < BALL_R + 8) {
            const nx = (ax - px) / dist, ny = (ay - py) / dist;
            ball.x = px + nx * (BALL_R + 8);
            ball.y = py + ny * (BALL_R + 8);
            const dot = ball.vx * nx + ball.vy * ny;
            if (dot < 0) {
                ball.vx -= 2 * dot * nx;
                ball.vy -= 2 * dot * ny;
            }
            // Add flipper impulse
            const fImpulse = 8;
            ball.vy -= fImpulse;
            ball.vx += (isLeft ? 1 : -1) * 2;
            playSound(250, 0.1, 'square', 0.2);
        }
    }
    flipperCollide(LEFT_FLIPPER, true);
    flipperCollide(RIGHT_FLIPPER, false);

    // Ball out of table (bottom)
    if (ball.y > TABLE.y + TABLE.h + 50) {
        ball.active = false;
        loseLife();
    }

    // Speed cap
    const spd = Math.sqrt(ball.vx*ball.vx + ball.vy*ball.vy);
    if (spd > 15) { ball.vx *= 15/spd; ball.vy *= 15/spd; }
}

function updateFlippers() {
    LEFT_FLIPPER.angle += (LEFT_FLIPPER.targetAngle - LEFT_FLIPPER.angle) * 0.3;
    RIGHT_FLIPPER.angle += (RIGHT_FLIPPER.targetAngle - RIGHT_FLIPPER.angle) * 0.3;
}

function updateActiveTarget() {
    if (!activeTarget) return;
    activeTarget.timer--;
    if (activeTarget.timer <= 0) {
        loseLife();
    }
}

function updateParticles() {
    for (let i = particles.length - 1; i >= 0; i--) {
        const p = particles[i];
        p.x += p.vx; p.y += p.vy;
        p.vy += 0.1;
        p.life--;
        if (p.life <= 0) particles.splice(i, 1);
    }
}

function updateFloatTexts() {
    for (let i = floatTexts.length - 1; i >= 0; i--) {
        floatTexts[i].y -= 1;
        floatTexts[i].life--;
        if (floatTexts[i].life <= 0) floatTexts.splice(i, 1);
    }
}

function updateTargets() {
    for (const t of targets) t.pulse += 0.05;
}

    }

        ctx.moveTo(fx, fy);
        ctx.lineTo(tx, ty);
        ctx.strokeStyle = '#ff8844';
        ctx.lineWidth = 12;
        ctx.lineCap = 'round';
        ctx.stroke();
        ctx.strokeStyle = '#ffcc88';
        ctx.lineWidth = 6;
        ctx.stroke();

    drawFlipper(LEFT_FLIPPER, true);
    drawFlipper(RIGHT_FLIPPER, false);

    // Ball
    if (ball.active) {
        const bg = ctx.createRadialGradient(ball.x - 3, ball.y - 3, 2, ball.x, ball.y, BALL_R);
        bg.addColorStop(0, '#fff');
        bg.addColorStop(1, '#88f');
        ctx.fillStyle = bg;
        ctx.beginPath();
        ctx.arc(ball.x, ball.y, BALL_R, 0, Math.PI * 2);
        ctx.fill();

    // Particles

        ctx.fillStyle = p.color;

let lastTime = 0;
function gameLoop(time) {
    const dt = time - lastTime;
    lastTime = time;

    if (gameState === 'start') {
        drawStartScreen();

        updatePhysics();
        updateFlippers();
