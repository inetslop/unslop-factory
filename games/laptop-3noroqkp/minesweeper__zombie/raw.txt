<script>
let cols = 9, rows = 10;
let grid = [];
let zombies = 10;
let flags = 0;
let revealed = 0;
let gameOver = false;
let timerInterval = null;
let seconds = 0;
let audioCtx = null;

function initAudio() {
  if (!audioCtx) {
    audioCtx = new (window.AudioContext || window.webkitAudioContext)();
  }
}

function playTone(freq, duration, type, volume) {
  if (!audioCtx) return;
  const osc = audioCtx.createOscillator();
  const gain = audioCtx.createGain();
  osc.type = type || 'square';
  osc.frequency.value = freq;
  gain.gain.setValueAtTime(volume || 0.15, audioCtx.currentTime);
  gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + duration);
  osc.connect(gain);
  gain.connect(audioCtx.destination);
  osc.start();
  osc.stop(audioCtx.currentTime + duration);
}

function playClick() { initAudio(); playTone(400, 0.05, 'square', 0.1); }
function playFlag() { initAudio(); playTone(600, 0.08, 'triangle', 0.12); }
function playDeath() {
  initAudio();
  playTone(150, 0.4, 'sawtooth', 0.3);
  setTimeout(() => playTone(100, 0.6, 'sawtooth', 0.25), 150);
}
function playWin() {
  initAudio();
  [523, 659, 784, 1047].forEach((f, i) => setTimeout(() => playTone(f, 0.2, 'square', 0.12), i * 120));
}

function showStart() {
  document.getElementById('game-screen').style.display = 'none';
  document.getElementById('end-screen').style.display = 'none';
  document.getElementById('start-screen').style.display = 'flex';
}

function startGame(c, r) {
  cols = c; rows = r || c;
  zombies = Math.floor(c * r * 0.16);
  if (zombies < 1) zombies = 1;
  grid = [];
  flags = 0; revealed = 0; gameOver = false; seconds = 0;
  
  document.getElementById('start-screen').style.display = 'none';
  document.getElementById('end-screen').style.display = 'none';
  document.getElementById('game-screen').style.display = 'flex';
  
  for (let i = 0; i < rows; i++) {
    grid[i] = [];
    for (let j = 0; j < cols; j++) {
      grid[i][j] = { zombie: false, revealed: false, flagged: false, adj: 0 };
    }
  }
  
  placeZombies();
  buildGrid();
  updateHUD();
  
  if (timerInterval) clearInterval(timerInterval);
  timerInterval = setInterval(() => {
    if (!gameOver) { seconds++; document.getElementById('timer').textContent = '⏱ ' + seconds + 's'; }
  }, 1000);
}

function placeZombies() {
  for (let i = 0; i < zombies; i++) {
    let r, c;
    do {
      r = Math.floor(Math.random() * rows);
      c = Math.floor(Math.random() * cols);
    } while (grid[r][c].zombie);
    grid[r][c].zombie = true;
  }
  for (let i = 0; i < rows; i++) {
    for (let j = 0; j < cols; j++) {
      if (!grid[i][j].zombie) {
        let adj = 0;
        for (let di = -1; di <= 1; di++) {
          for (let dj = -1; dj <= 1; dj++) {
            if (di === 0 && dj === 0) continue;
            let ni = i + di, nj = j + dj;
            if (ni >= 0 && ni < rows && nj >= 0 && nj < cols && grid[ni][nj].zombie) adj++;
          }
        }
        grid[i][j].adj = adj;
      }
    }
  }
}

function buildGrid() {
  const container = document.getElementById('grid-container');
  container.innerHTML = '';
  container.style.gridTemplateColumns = `repeat(${cols}, 1fr)`;
  
  for (let i = 0; i < rows; i++) {
    for (let j = 0; j < cols; j++) {
      const cell = document.createElement('div');
      cell.className = 'cell hidden';
      cell.dataset.i = i;
      cell.dataset.j = j;
      cell.addEventListener('click', () => handleClick(i, j));
      cell.addEventListener('contextmenu', (e) => { e.preventDefault(); handleFlag(i, j); });
      container.appendChild(cell);
    }
  }
}

function getCellEl(i, j) {
  return document.getElementById('grid-container').children[i * cols + j];
}

function handleClick(i, j) {
  if (gameOver) return;
  const cell = grid[i][j];
  if (cell.flagged || cell.revealed) return;
  
  playClick();
  cell.revealed = true;
  revealed++;
  
  const el = getCellEl(i, j);
  el.className = 'cell revealed';
  
  if (cell.zombie) {
    loseGame(i, j);
    return;
  }
  
  // Flood fill for zero-adjacent cells
  if (cell.adj === 0) {
    floodReveal(i, j);
  }
  
  checkWin();
}

function floodReveal(si, sj) {
  const stack = [[si, sj]];
  while (stack.length > 0) {
    const [i, j] = stack.pop();
    for (let di = -1; di <= 1; di++) {
      for (let dj = -1; dj <= 1; dj++) {
        if (di === 0 && dj === 0) continue;
        let ni = i + di, nj = j + dj;
        if (ni < 0 || ni >= rows || nj < 0 || nj >= cols) continue;
        const nc = grid[ni][nj];
        if (nc.flagged || nc.revealed) continue;
        nc.revealed = true;
        revealed++;
        const nel = getCellEl(ni, nj);
        if (nc.zombie) {
          // Shouldn't happen in flood but safety
        } else {
          nel.className = 'cell revealed';
          if (nc.adj > 0) {
            nel.textContent = nc.adj;
            nel.classList.add('n' + nc.adj);
          }
        }
        if (!nc.zombie && nc.adj === 0) {
          stack.push([ni, nj]);
        }
      }
    }
  }
  // Update all revealed number cells
  for (let ii = 0; ii < rows; ii++) {
    for (let jj = 0; jj < cols; jj++) {
      const cc = grid[ii][jj];
      if (cc.revealed && !cc.zombie) {
        const el = getCellEl(ii, jj);
        if (el.textContent === '' && cc.adj > 0) {
          el.textContent = cc.adj;
          el.classList.add('n' + cc.adj);
        }
      }
    }
  }
}

function handleFlag(i, j) {
  if (gameOver) return;
  const cell = grid[i][j];
  if (cell.revealed) return;
  
  playFlag();
  cell.flagged = !cell.flagged;
  flags += cell.flagged ? 1 : -1;
  
  const el = getCellEl(i, j);
  if (cell.flagged) {
    el.className = 'cell hidden flagged';
    el.textContent = '🚩';
  } else {
    el.className = 'cell hidden';
    el.textContent = '';
  }
  updateHUD();
}

function loseGame(hi, hj) {
  gameOver = true;
  clearInterval(timerInterval);
  
  playDeath();
  
  const hitEl = getCellEl(hi, hj);
  hitEl.className = 'cell revealed zombie';
  hitEl.textContent = '🧟';
  
  // Reveal all zombies
  for (let i = 0; i < rows; i++) {
    for (let j = 0; j < cols; j++) {
      const cell = grid[i][j];
      const el = getCellEl(i, j);
      if (cell.zombie) {
        el.className = 'cell revealed zombie';
        el.textContent = '🧟';
      } else if (cell.flagged) {
        // Wrong flag - show skull
        if (!cell.revealed && !cell.zombie) {
          el.className = 'cell revealed';
          el.textContent = '💀';
        }
      } else if (!cell.revealed) {
        el.style.opacity = '0.5';
      }
    }
  }
  
  document.getElementById('grid-container').classList.add('shake');
  setTimeout(() => document.getElementById('grid-container').classList.remove('shake'), 500);
  
  setTimeout(() => showEnd(false), 1200);
}

function checkWin() {
  const safeCells = rows * cols - zombies;
  if (revealed >= safeCells) {
    gameOver = true;
    clearInterval(timerInterval);
    playWin();
    // Flag all remaining zombies
    for (let i = 0; i < rows; i++) {
      for (let j = 0; j < cols; j++) {
        if (grid[i][j].zombie && !grid[i][j].flagged) {
          const el = getCellEl(i, j);
          el.className = 'cell hidden flagged';
          el.textContent = '🚩';
          flags++;
        }
      }
    }
    updateHUD();
    setTimeout(() => showEnd(true), 800);
  }
}

function showEnd(won) {
  document.getElementById('game-screen').style.display = 'none';
  document.getElementById('end-screen').style.display = 'flex';
  
  const titleEl = document.getElementById('end-title');
  const msgEl = document.getElementById('end-msg');
  const statsEl = document.getElementById('end-stats');
  
  if (won) {
    titleEl.textContent = '🏆 AREA CLEARED!';
    titleEl.style.color = '#4caf50';
    msgEl.textContent = 'Safe zone established. The horde is contained.';
  } else {
    titleEl.textContent = '☠ INFECTED ☠';
    titleEl.style.color = '#ff5252';
    msgEl.textContent = 'You were eaten by the zombies. The horde grows...';
  }
  
  statsEl.textContent = `Time: ${seconds}s | Revealed: ${revealed}/${rows * cols - zombies}`;
}

function updateHUD() {
  document.getElementById('flag-count').textContent = '🚩 ' + flags + ' / ' + zombies;
  document.getElementById('timer').textContent = '⏱ ' + seconds + 's';
}

// Keyboard support
window.addEventListener('keydown', (e) => {
  if (e.key === 'r' || e.key === 'R') {
    if (document.getElementById('game-screen').style.display === 'flex' || document.getElementById('end-screen').style.display === 'flex') {
      startGame(cols, rows);
    }
  }
});

// Prevent context menu on the page
document.addEventListener('contextmenu', (e) => e.preventDefault());

// Resize handling
window.addEventListener('resize', () => {
  // Grid uses clamp() so it auto-scales
});
</script>
