function generateWalls() {
  walls = [];
  const patterns = [
    // Central cross
    [[4,1],[4,2],[4,3],[4,5],[4,6],[4,7],
     [1,4],[2,4],[3,4],[5,4],[6,4],[7,4]],

}

function initGame() {
  score = 0;
  movesLeft = 30;
  combo = 0;
  selected = null;
  animating = false;

        grid[r][c] = -1; // wall

        let col;
        do {

        grid[r][c] = col;

    }
  }

function wouldMatch(row, col, color) {
  if (col >= 2) {
    if (grid[row] && grid[row][col-1] === color && grid[row] && grid[row][col-2] === color) return true;
  }
  if (row >= 2) {
    if (grid[row-1] && grid[row-1][col] === color && grid[row-2] && grid[row-2][col] === color) return true;
  }
  return false;
}

function isWall(r, c) {
  if (r < 0 || r >= ROWS || c < 0 || c >= COLS) return true;
  return grid[r][c] === -1;
}

function findMatches() {
  let matches = [];
  // Horizontal
  for (let r = 0; r < ROWS; r++) {
    for (let c = 0; c < COLS; c++) {
      if (isWall(r, c)) continue;
      const col = grid[r][c];
      let count = 1;
      while (c + count < COLS && !isWall(r, c+count) && grid[r][c+count] === col) count++;
      if (count >= 3) {
        for (let i = 0; i < count; i++) matches.push([r, c+i]);
      }
      c += count - 1;
    }
  }
  // Vertical
  for (let c = 0; c < COLS; c++) {
    for (let r = 0; r < ROWS; r++) {
      if (isWall(r, c)) continue;
      const col = grid[r][c];
      let count = 1;
      while (r + count < ROWS && !isWall(r+count, c) && grid[r+count][c] === col) count++;
      if (count >= 3) {
        for (let i = 0; i < count; i++) matches.push([r+i, c]);
      }
      r += count - 1;
    }
  }
  // Deduplicate
  const set = new Set(matches.map(m => m[0] + ',' + m[1]));
  return Array.from(set).map(s => s.split(',').map(Number));
}

function canSwap(r1, c1, r2, c2) {
  if (isWall(r1, c1) || isWall(r2, c2)) return false;
  const dr = Math.abs(r1 - r2), dc = Math.abs(c1 - c2);
  if (dr + dc !== 1) return false;
  // No wall between (they're adjacent so no wall cell between them)
  return true;
}

function swap(r1, c1, r2, c2) {
  const t = grid[r1][c1];
  grid[r1][c1] = grid[r2][c2];
  grid[r2][c2] = t;
}

  clearingTiles = matches.map(m => ({ r: m[0], c: m[1], progress: 0 }));
  combo = 1;
  const pts = matches.length * 10 * combo;
  score += pts;
  
  for (let m of matches) {
    const x = offsetX + m[1] * cellSize + cellSize/2;
    const y = offsetY + m[0] * cellSize + cellSize/2;
    for (let i = 0; i < 5; i++) {

        life: 30, maxLife: 30,
        color: COLORS[grid[m[0]][m[1]]]
      });
    }
  }
  
  shakeTime = matches.length > 5 ? 8 : 4;
  
  setTimeout(() => {
    for (let m of clearingTiles) {
      grid[m.r][m.c] = -2; // mark for removal
    }

      const newMatches = findMatches();
      if (newMatches.length > 0) {
        combo++;

}

function applyGravity() {
  for (let c = 0; c < COLS; c++) {
    let writeRow = ROWS - 1;
    for (let r = ROWS - 1; r >= 0; r--) {
      if (isWall(r, c)) continue;
      if (writeRow !== r) {
        grid[writeRow][c] = grid[r][c];
        grid[r][c] = -2;
      }
      writeRow--;
    }
  }
}

function fillEmpty() {
  for (let c = 0; c < COLS; c++) {
    for (let r = 0; r < ROWS; r++) {
      if (grid[r][c] === -2 && !isWall(r, c)) {
        grid[r][c] = Math.floor(Math.random() * COLORS.length);
      }
    }
  }
}

function endGame() {
  gameState = 'gameover';
  try { 
    if (score > highScore) { highScore = score; localStorage.setItem('mazeMatch3High', String(highScore)); }

function getCellFromPos(x, y) {
  const c = Math.floor((x - offsetX) / cellSize);
  const r = Math.floor((y - offsetY) / cellSize);
  if (r < 0 || r >= ROWS || c < 0 || c >= COLS) return null;
  if (isWall(r, c)) return null;
  return { r, c };
}

  ctx.fillRect(0, 0, W, H);

    sy = (Math.random()-0.5) * shakeTime;
    shakeTime--;
  }
  
  ctx.save();
  ctx.translate(sx, sy);
  
  if (gameState === 'start') {
    drawStartScreen();
  } else if (gameState === 'playing' || gameState === 'gameover') {
    drawGame();
  }
  
  ctx.restore();
}

function drawGem(cx, cy, s) {
  const r = s * 0.5;
  ctx.beginPath();
  for (let i = 0; i < 6; i++) {
    const angle = (i / 6) * Math.PI * 2 - Math.PI/2;
    const px = cx + Math.cos(angle) * r;
    const py = cy + Math.sin(angle) * r;
    if (i === 0) ctx.moveTo(px, py);
    else ctx.lineTo(px, py);
  }
  ctx.closePath();
  ctx.fill();
  
  // Inner highlight
  ctx.fillStyle = 'rgba(255,255,255,0.3)';
  ctx.beginPath();
  ctx.arc(cx - r*0.2, cy - r*0.2, r*0.3, 0, Math.PI*2);
  ctx.fill();
}

function update() {
  // Update particles
  for (let i = particles.length - 1; i >= 0; i--) {
    const p = particles[i];
    p.x += p.vx;
    p.y += p.vy;
    p.vy += 0.2;
    p.life--;
    if (p.life <= 0) particles.splice(i, 1);
  }
  
  // Update clearing animation
  for (let ct of clearingTiles) {
    ct.progress = Math.min(ct.progress + 0.08, 1);
  }
}

function gameLoop() {
  update();
  draw();
  requestAnimationFrame(gameLoop);
}

  if (!mouseDown || !dragStart || gameState !== 'playing' || animating) return;

  const x = e.clientX - rect.left;
  const y = e.clientY - rect.top;
  const cell = getCellFromPos(x, y);
  
  if (cell && dragStart) {
    const dr = Math.abs(cell.r - dragStart.r);
    const dc = Math.abs(cell.c - dragStart.c);
    
    if (dr + dc === 1) {
      trySwap(dragStart.r, dragStart.c, cell.r, cell.c);
      mouseDown = false;
      dragStart = null;
    }
  }
});

canvas.addEventListener('mouseup', (e) => {
  if (!mouseDown || gameState !== 'playing' || animating) { 
    mouseDown = false; 
    return; 
  }
  
  const rect = canvas.getBoundingClientRect();
  const x = e.clientX - rect.left;
  const y = e.clientY - rect.top;
  const cell = getCellFromPos(x, y);
  
  if (!cell) { 
    mouseDown = false; 
    dragStart = null; 
    return; 
  }
  
  if (selected && selected.r === cell.r && selected.c === cell.c) {
    selected = null;
  } else if (selected) {
    const dr = Math.abs(cell.r - selected.r);
    const dc = Math.abs(cell.c - selected.c);
    if (dr + dc === 1) {
      trySwap(selected.r, selected.c, cell.r, cell.c);
    } else {
      selected = cell;
    }
  } else {
    selected = cell;
  }
  
  mouseDown = false;
  dragStart = null;
});

function trySwap(r1, c1, r2, c2) {
  if (!canSwap(r1, c1, r2, c2)) return;
  
  swap(r1, c1, r2, c2);
  
  const matches = findMatches();
  if (matches.length === 0) {
    // Invalid - swap back
    swap(r1, c1, r2, c2);
    selected = null;
    return;
  }
  
  movesLeft--;
  selected = null;
  animating = true;
  startClearAnimation(matches);
}

// Keyboard restart
window.addEventListener('keydown', (e) => {
  if (e.key === 'r' || e.key === 'R') {
    initGame();
    gameState = 'playing';
  }
  if (e.key === 'Escape') {
    gameState = 'start';
  }
});
