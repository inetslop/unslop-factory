<script>
(function(){
const canvas = document.getElementById('board');
const ctx = canvas.getContext('2d');
const COLS = 10, ROWS = 20, CELL = 30;
canvas.width = COLS * CELL;
canvas.height = ROWS * CELL;

let audioCtx = null;
function initAudio() { if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)(); }
function playSound(freq, dur, type) {
if (!audioCtx) return;
const osc = audioCtx.createOscillator();
const gain = audioCtx.createGain();
osc.type = type || 'square';
osc.frequency.value = freq;
gain.gain.setValueAtTime(0.15, audioCtx.currentTime);
gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
osc.connect(gain).connect(audioCtx.destination);
osc.start();
osc.stop(audioCtx.currentTime + dur);
}

const SHAPES = [
[{x:0,y:0},{x:1,y:0},{x:2,y:0},{x:3,y:0}],
[{x:1,y:0},{x:0,y:1},{x:2,y:1},{x:1,y:2}],
[{x:0,y:0},{x:1,y:0},{x:1,y:1},{x:2,y:1}],
[{x:1,y:0},{x:0,y:1},{x:0,y:2},{x:1,y:2}],
[{x:0,y:0},{x:1,y:0},{x:0,y:1},{x:0,y:2}],
[{x:0,y:0},{x:1,y:0},{x:1,y:1},{x:2,y:1}],
[{x:1,y:0},{x:0,y:0},{x:0,y:1},{x:1,y:1}]
];
const COLORS = ['#ff6b6b','#4ecdc4','#ffd93d','#6c5ce7','#a8e6cf','#ff9ff3','#f9f871'];

let grid, piece, nextPiece, score, lines, shards, gameOver;
let dropInterval, lastDrop, animId;
let clickPower = 1, lineMult = 1, autoClickLevel = 0, autoPlayLevel = 0;
let shopItems = [];
let autoClickTimer = null;
let aiTimer = null;
let nextPreviewCanvas;

function initGame() {
grid = Array.from({length: ROWS}, () => Array(COLS).fill(null));
score = 0; lines = 0; shards = 0; gameOver = false;
clickPower = 1; lineMult = 1; autoClickLevel = 0; autoPlayLevel = 0;
shopItems = [
{name:'Auto Play', desc:'AI helps play', baseCost:50, level:0, maxLevel:5},
{name:'Line Mult', desc:'More shards/line', baseCost:30, level:0, maxLevel:10},
{name:'Click Power', desc:'+shards/click', baseCost:20, level:0, maxLevel:10},
{name:'Auto Click', desc:'Auto clicks/s', baseCost:80, level:0, maxLevel:5}
];
spawnPiece();
updateUI();
buildShop();
clearIntervals();
}

function spawnPiece() {
const idx = Math.floor(Math.random() * SHAPES.length);
piece = { shape: SHAPES[idx], color: COLORS[idx], x: 3, y: 0 };
nextPreviewCanvas = piece;
}

function rotate(shape) {
return shape[0].y === 0 ? 
[{x:shape[0].y,y:-shape[0].x},{x:shape[1].y,y:-shape[1].x},{x:shape[2].y,y:-shape[2].x},{x:shape[3].y,y:-shape[3].x}] : shape;
}

function valid(x, y, shape) {
for (let c of shape) {
let nx = x + c.x, ny = y + c.y;
if (nx < 0 || nx >= COLS || ny >= ROWS) return false;
if (ny >= 0 && grid[ny][nx]) return false;
}
return true;
}

function merge() {
for (let c of piece.shape) {
let x = piece.x + c.x, y = piece.y + c.y;
if (y >= 0) grid[y][x] = piece.color;
}
}

function clearLines() {
let cleared = 0;
for (let r = ROWS - 1; r >= 0; r--) {
if (grid[r].every(c => c)) {
grid.splice(r, 1);
grid.unshift(Array(COLS).fill(null));
cleared++;
r++;
}
}
if (cleared > 0) {
let pts = [0, 40, 100, 300, 1200][cleared] * (1 + Math.floor(lines/10));
score += pts;
lines += cleared;
shards += cleared * 10 * lineMult;
playSound(440 + cleared*100, 0.2, 'sine');
updateUI();
}
}

function drop() {
if (valid(piece.x, piece.y + 1, piece.shape)) {
piece.y++;
} else {
merge();
clearLines();
if (!valid(piece.x, piece.y, piece.shape)) {
gameOver = true;
showGameOver();
return;
}
spawnPiece();
}
}

function hardDrop() {
while (valid(piece.x, piece.y + 1, piece.shape)) piece.y++;
merge();
clearLines();
score += 2;
if (!valid(piece.x, piece.y, piece.shape)) { gameOver = true; showGameOver(); return; }
spawnPiece();
playSound(200, 0.1, 'sawtooth');
}

function rotatePiece() {
let rotated = piece.shape.map(c => ({x: -c.y, y: c.x}));
if (valid(piece.x, piece.y, rotated)) {
piece.shape = rotated;
playSound(300, 0.05, 'triangle');
} else if (valid(piece.x + 1, piece.y, rotated)) {
piece.x++;
piece.shape = rotated;
} else if (valid(piece.x - 1, piece.y, rotated)) {
piece.x--;
piece.shape = rotated;
}
}

function moveLeft() { if (valid(piece.x-1, piece.y, piece.shape)) piece.x--; }
function moveRight() { if (valid(piece.x+1, piece.y, piece.shape)) piece.x++; }

function drawCell(x, y, color) {
ctx.fillStyle = color;
ctx.fillRect(x * CELL + 1, y * CELL + 1, CELL - 2, CELL - 2);
ctx.fillStyle = 'rgba(255,255,255,0.2)';
ctx.fillRect(x * CELL + 1, y * CELL + 1, CELL - 2, 4);
}

function draw() {
ctx.fillStyle = '#0d1117';
ctx.fillRect(0, 0, canvas.width, canvas.height);
// grid lines
ctx.strokeStyle = 'rgba(79,195,247,0.05)';
for (let i = 0; i <= COLS; i++) { ctx.beginPath(); ctx.moveTo(i*CELL,0); ctx.lineTo(i*CELL,canvas.height); ctx.stroke(); }
for (let i = 0; i <= ROWS; i++) { ctx.beginPath(); ctx.moveTo(0,i*CELL); ctx.lineTo(canvas.width,i*CELL); ctx.stroke(); }
// filled cells
for (let r = 0; r < ROWS; r++) for (let c = 0; c < COLS; c++) if (grid[r][c]) drawCell(c, r, grid[r][c]);
// ghost
if (piece && !gameOver) {
let gy = piece.y;
while (valid(piece.x, gy+1, piece.shape)) gy++;
for (let c of piece.shape) {
let x = piece.x + c.x, y = gy + c.y;
if (y >= 0) { ctx.strokeStyle = piece.color; ctx.lineWidth = 1; ctx.strokeRect(x*CELL+2, y*CELL+2, CELL-4, CELL-4); }
}
// current piece
for (let c of piece.shape) {
let x = piece.x + c.x, y = piece.y + c.y;
if (y >= 0) drawCell(x, y, piece.color);
}
}
}

function gameLoop(time) {
if (gameOver) return;
let interval = Math.max(100, 800 - lines * 20);
if (time - lastDrop > interval) drop();
draw();
animId = requestAnimationFrame(gameLoop);
}

function showGameOver() {
document.getElementById('game-over').style.display = 'block';
document.getElementById('final-score').textContent = score;
document.getElementById('final-lines').textContent = lines;
playSound(150, 0.5, 'sawtooth');
clearIntervals();
}

function clearIntervals() {
if (autoClickTimer) clearInterval(autoClickTimer);
if (aiTimer) clearInterval(aiTimer);
}

function updateUI() {
document.getElementById('shards-display').textContent = shards;
document.getElementById('score-display').textContent = `Score: ${score} | Lines: ${lines}`;
document.getElementById('click-power-label').textContent = `+${clickPower}/click`;
updateShopButtons();
}

function buildShop() {
const list = document.getElementById('shop-list');
list.innerHTML = '';
shopItems.forEach((item, i) => {
let div = document.createElement('div');
div.className = 'shop-item';
let cost = getCost(i);
let levelText = item.level > 0 ? `Lv.${item.level}` : '';
div.innerHTML = `<span class="name">${item.name} ${levelText}</span><span class="cost">💎${cost}</span>`;
let btn = document.createElement('button');
btn.className = 'buy-btn';
btn.textContent = 'BUY';
btn.onclick = () => buyUpgrade(i);
div.appendChild(btn);
list.appendChild(div);
});
}

function getCost(i) {
return Math.floor(shopItems[i].baseCost * Math.pow(1.5, shopItems[i].level));
}

function buyUpgrade(i) {
let item = shopItems[i];
if (item.level >= item.maxLevel) return;
let cost = getCost(i);
if (shards < cost) return;
shards -= cost;
item.level++;
applyUpgrades();
playSound(600, 0.15, 'sine');
updateUI();
buildShop();
}

function applyUpgrades() {
lineMult = 1 + shopItems[1].level;
clickPower = 1 + shopItems[2].level;
autoClickLevel = shopItems[3].level;
autoPlayLevel = shopItems[0].level;
clearIntervals();
if (autoClickLevel > 0) {
autoClickTimer = setInterval(() => {
shards += clickPower * autoClickLevel;
updateUI();
}, 1000);
}
if (autoPlayLevel > 0 && !gameOver) {
let speed = Math.max(300, 1500 - autoPlayLevel * 250);
aiTimer = setInterval(() => aiStep(), speed);
}
}

function updateShopButtons() {
let btns = document.querySelectorAll('.buy-btn');
btns.forEach((btn, i) => {
let item = shopItems[i];
btn.disabled = shards < getCost(i) || item.level >= item.maxLevel;
});
}

function aiStep() {
if (gameOver || !piece) return;
// Find best x position
let bestX = 0, bestScore = -1;
for (let x = 0; x <= COLS - 4; x++) {
if (!valid(x, piece.y, piece.shape)) continue;
let s = 0;
for (let c of piece.shape) {
let ny = piece.y + c.y + 1;
while (ny < ROWS && !grid[ny][x + c.x]) ny++;
s += Math.abs(c.y) - ny * 0.5;
}
if (s > bestScore) { bestScore = s; bestX = x; }
}
// Move to best position
let dx = bestX - piece.x;
for (let i = 0; i < Math.abs(dx); i++) {
if (dx > 0 && valid(piece.x+1, piece.y, piece.shape)) piece.x++;
else if (dx < 0 && valid(piece.x-1, piece.y, piece.shape)) piece.x--;
}
// Rotate a few times
for (let r = 0; r < autoPlayLevel % 2 + 1; r++) rotatePiece();
// Hard drop occasionally
if (autoPlayLevel >= 3) hardDrop();
}

function clickHandler() {
initAudio();
shards += clickPower;
playSound(500 + Math.random()*200, 0.08, 'sine');
updateUI();
}

// Input
window.addEventListener('keydown', (e) => {
if (!audioCtx && (e.key === ' ' || e.key === 'Enter')) initAudio();
if (gameOver) return;
switch(e.code) {
case 'ArrowLeft': e.preventDefault(); moveLeft(); break;
case 'ArrowRight': e.preventDefault(); moveRight(); break;
case 'ArrowUp': e.preventDefault(); rotatePiece(); break;
case 'ArrowDown': e.preventDefault(); drop(); score++; updateUI(); break;
case 'Space': e.preventDefault(); hardDrop(); break;
}
});

// Start
document.getElementById('start-btn').addEventListener('click', () => {
initAudio();
document.getElementById('start-screen').style.display = 'none';
document.getElementById('game-area').style.display = 'flex';
initGame();
lastDrop = performance.now();
animId = requestAnimationFrame(gameLoop);
playSound(440, 0.1, 'sine');
});

document.getElementById('click-btn').addEventListener('click', clickHandler);
document.getElementById('restart-btn').addEventListener('click', () => {
document.getElementById('game-over').style.display = 'none';
initGame();
lastDrop = performance.now();
animId = requestAnimationFrame(gameLoop);
});

// Handle resize
function resize() {
const container = document.getElementById('game-container');
const maxH = window.innerHeight - 20;
const scale = Math.min(1, maxH / 640);
canvas.style.width = (COLS * CELL * scale) + 'px';
canvas.style.height = (ROWS * CELL * scale) + 'px';
document.getElementById('side-panel').style.transform = `scale(${scale})`;
document.getElementById('side-panel').style.transformOrigin = 'top left';
}
window.addEventListener('resize', resize);
resize();
})();
</script>
</body>
</html>
