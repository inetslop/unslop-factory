function playSound(freq, dur, type) {
  if (!audioCtx) return;
  const osc = audioCtx.createOscillator();
  const gain = audioCtx.createGain();
  osc.type = type || 'square';
  osc.frequency.value = freq;
  gain.gain.setValueAtTime(0.1, audioCtx.currentTime);
  gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
  osc.connect(gain);
  gain.connect(audioCtx.destination);
  osc.start();
  osc.stop(audioCtx.currentTime + dur);
}

function initGame() {

const DIRS = { up: [-1, 0], down: [1, 0], left: [0, -1], right: [0, 1] };
const OPPOSITE = { up: 'down', down: 'up', left: 'right', right: 'left' };

function canMove(row, col) {
  if (row < 0 || row >= ROWS || col < 0 || col >= COLS) return false;
  return maze[row][col] !== 1;
}

function moveEntity(ent, speedMult) {
  const nextRow = ent.row + DIRS[ent.dir][0];
  const nextCol = ent.col + DIRS[ent.dir][1];
  
  if (canMove(nextRow, nextCol)) {
    ent.row = nextRow;
    ent.col = nextCol;
    return true;
  }
  return false;
}

function tryTurn(ent) {
  const nr = ent.row + DIRS[ent.nextDir][0];
  const nc = ent.col + DIRS[ent.nextDir][1];
  if (canMove(nr, nc)) {
    ent.dir = ent.nextDir;
    return true;
  }
  return false;
}

function ghostAI(ghost, idx) {
  const frightened = frightenedTimer > 0;
  
  // Try to continue current direction first
  const cr = ghost.row + DIRS[ghost.dir][0];
  const cc = ghost.col + DIRS[ghost.dir][1];
  
  if (canMove(cr, cc) && !frightened) {
    // Prefer continuing straight unless it leads to dead end or we should turn
    if (Math.random() < 0.7) return;
  }
  
  const options = [];
  for (const d of ['up', 'down', 'left', 'right']) {
    if (d === OPPOSITE[ghost.dir]) continue;
    const nr = ghost.row + DIRS[d][0];
    const nc = ghost.col + DIRS[d][1];
    if (canMove(nr, nc)) options.push(d);
  }
  
  if (options.length === 0) {
    // Dead end - reverse
    ghost.dir = OPPOSITE[ghost.dir];
    return;
  }
  
  if (frightened) {
    ghost.dir = options[Math.floor(Math.random() * options.length)];
  } else {
    // Chase nearest player
    let bestDir = options[0];
    let bestDist = Infinity;
    
    for (const p of players) {
      if (lives[players.indexOf(p)] <= 0) continue;
      const dist = Math.abs(ghost.row - p.row) + Math.abs(ghost.col - p.col);
      if (dist < bestDist) {
        bestDist = dist;
        for (const d of options) {
          const nr = ghost.row + DIRS[d][0];
          const nc = ghost.col + DIRS[d][1];
          const d2 = Math.abs(nr - p.row) + Math.abs(nc - p.col);
          if (d2 < bestDist) { bestDir = d; bestDist = d2; }
        }
      }
    }
    
    // 30% chance to go random for unpredictability
    if (Math.random() < 0.3) {
      ghost.dir = options[Math.floor(Math.random() * options.length)];
    } else {
      ghost.dir = bestDir;
    }
  }
}

function update() {
  if (gameState !== 'playing') return;
  
  animFrame++;
  
  if (frightenedTimer > 0) frightenedTimer--;
  
  // Move players (every frame for smoothness, but grid-based)
  const playerSpeed = 1; // one cell per tick effectively
  
  for (let i = 0; i < 2; i++) {
    if (lives[i] <= 0) continue;
    const p = players[i];
    
    if (animFrame % 4 === 0) { // Player moves every 4 frames
      tryTurn(p);
      moveEntity(p, 1);
      
      // Check pellet collection
      const key = p.row + ',' + p.col;
      if (pellets[key]) {
        delete pellets[key];
        scores[i] += 10;
        playSound(440, 0.05, 'sine');
      } else if (powerPellets[key]) {
        delete powerPellets[key];
        scores[i] += 50;
        frightenedTimer = frightDuration;
        playSound(220, 0.2, 'sawtooth');
      }
    }
  }
  
  // Move ghosts
  const ghostSpeed = frightenedTimer > 0 ? 3 : 4; // frames between moves
  
  for (let i = 0; i < ghosts.length; i++) {
    const g = ghosts[i];
    
    if (animFrame % ghostSpeed === 0) {
      ghostAI(g, i);
      const nr = g.row + DIRS[g.dir][0];
      const nc = g.col + DIRS[g.dir][1];
      if (canMove(nr, nc)) {
        g.row = nr;
        g.col = nc;
      } else {
        // Stuck - reverse
        g.dir = OPPOSITE[g.dir];
      }
    }
    
    // Check collision with players
    for (let pi = 0; pi < 2; pi++) {
      if (lives[pi] <= 0) continue;
      const p = players[pi];
      if (g.row === p.row && g.col === p.col) {
        if (frightenedTimer > 0) {
          // Eat ghost
          scores[pi] += 200;
          playSound(880, 0.3, 'square');
          resetPositions();
          break;
        } else {
          lives[pi]--;
          playSound(150, 0.5, 'sawtooth');
          if (lives[pi] <= 0) {
            // Check if both players are dead
            if (lives[0] <= 0 && lives[1] <= 0) {
              gameState = 'over';
            } else {
              resetPositions();
            }
          } else {
            resetPositions();
          }
        }
      }
    }
  }
  
  // Check win condition
  const remaining = Object.keys(pellets).length + Object.keys(powerPellets).length;
  if (remaining === 0) {
    gameState = 'win';
  }
}

        ctx.strokeStyle = '#42A5F5';

        ctx.fillStyle = '#FFD54F';
        ctx.beginPath();
        ctx.arc(x + cellSize/2, y + cellSize/2, cellSize * 0.12, 0, Math.PI*2);

    ctx.arc(x + r*0.3, y - r*0.1, r*0.1, 0, Math.PI*2);
    ctx.fill();
  }
  
  // Draw players
  const playerColors = ['#FFEB3B', '#E91E63'];
  for (let i = 0; i < 2; i++) {

      case 'left': startAngle = Math.PI + mouthAngle; endAngle = Math.PI - mouthAngle; break;

      case 'down': startAngle = Math.PI/2 + mouthAngle; endAngle = Math.PI/2 - mouthAngle; break;

    ctx.closePath();
    ctx.fill();
  }

  ctx.fillStyle = '#FFEB3B';
  ctx.font = `bold ${Math.max(14, cellSize * 0.5)}px monospace`;

  ctx.fillText(`P2: ${scores[1]}`, 10, 40);

  } else if (gameState === 'over') {
    drawOverScreen();
  } else if (gameState === 'win') {
    drawWinScreen();
  }
}
