    if (!audioCtx) audioCtx = new (window.AudioContext || webkitAudioContext)();

function playSound(freq, duration, type, vol) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = type || 'square';
    osc.frequency.value = freq;
    gain.gain.setValueAtTime(vol || 0.1, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + duration);
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + duration);
}
function sfxJump() { playSound(400, 0.15, 'square', 0.08); }

// Physics
function updatePlayer(p) {
    if (p.dead) {
        p.respawnTimer--;
        if (p.respawnTimer <= 0) {
            p.dead = false;
            p.x = p.id === 1 ? Math.max(camera.x + 50, camera.x + W/2 - 200) : Math.max(camera.x + 80, camera.x + W/2 - 170);
            p.y = H - 200;
            p.vx = 0;
            p.vy = 0;
        }
        return;
    }

    if (p.hurtTimer > 0) { p.hurtTimer--; return; }

    let left = false, right = false, jump = false;
    if (p.id === 1) {
        left = keys['KeyA'];
        right = keys['KeyD'];
        jump = keys['KeyW'];
    } else {
        left = keys['ArrowLeft'];
        right = keys['ArrowRight'];
        jump = keys['ArrowUp'];
    }

    if (left) p.vx = -5;
    else if (right) p.vx = 5;
    else p.vx *= 0.8;

    if (jump && p.onGround) {
        p.vy = -12;
        p.onGround = false;
        sfxJump();
    }

    p.vy += GRAVITY;
    if (p.vy > 15) p.vy = 15;

    // Move X
    p.x += p.vx;
    if (p.x < 0) p.x = 0;
    if (p.x + p.w > LEVEL_WIDTH) p.x = LEVEL_WIDTH - p.w;

    // Collide X with platforms
    for (let plat of platforms) {
        if (p.x + p.w > plat.x && p.x < plat.x + plat.w && p.y + p.h > plat.y && p.y < plat.y + plat.h) {
            if (p.vx > 0) p.x = plat.x - p.w;
            else if (p.vx < 0) p.x = plat.x + plat.w;
        }
    }

    // Move Y
    p.y += p.vy;
    p.onGround = false;

    for (let plat of platforms) {
        if (p.x + p.w > plat.x && p.x < plat.x + plat.w && p.y + p.h > plat.y && p.y < plat.y + plat.h) {
            if (p.vy > 0) {
                p.y = plat.y - p.h;
                p.vy = 0;
                p.onGround = true;
            } else if (p.vy < 0) {
                p.y = plat.y + plat.h;
                p.vy = 0;
            }
        }
    }

    // Fall off screen
    if (p.y > H + 50) {
        killPlayer(p);
    }

    // Spike collisions
    for (let s of spikes) {
        if (p.x + p.w > s.x && p.x < s.x + s.w && p.y + p.h > s.y && p.y < s.y + s.h) {
            killPlayer(p);
            break;
        }
    }
    for (let ms of movingSpikes) {
        if (p.x + p.w > ms.x && p.x < ms.x + ms.w && p.y + p.h > ms.y && p.y < ms.y + ms.h) {
            killPlayer(p);
            break;
        }
    }

    // Gem collection
    for (let g of gems) {
        if (!g.collected && Math.abs(p.x + p.w/2 - g.x) < 30 && Math.abs(p.y + p.h/2 - g.y) < 30) {
            g.collected = true;
            p.gemsCollected++;
            sfxCollect();
        }
    }

    // Goal
    if (!goalFlag.collected && p.x + p.w > goalFlag.x && p.x < goalFlag.x + 40 && Math.abs(p.y - goalFlag.y) < 60) {
        goalFlag.collected = true;
        winGame(p);
    }
}

function killPlayer(p) {
    if (p.dead) return;
    p.dead = true;
    p.respawnTimer = 90;
    sfxHurt();
}

function winGame(p) {
    winner = p.id;
    gameState = 'finished';
    winTimer = 0;
    sfxWin();
}

function updateMovingSpikes() {
    for (let ms of movingSpikes) {
        ms.x += ms.speed * ms.dir;
        if (ms.x > ms.startX + ms.range) { ms.dir = -1; }
        if (ms.x < ms.startX - ms.range) { ms.dir = 1; }
    }
}

function updateCamera() {
    let midX = (player1.x + player2.x) / 2;
    let midY = (player1.y + player2.y) / 2;
    let dist = Math.abs(player1.x - player2.x);

    // Zoom based on distance
    let targetZoom = 1;
    if (dist > W * 0.7) {
        targetZoom = Math.max(0.5, W * 0.7 / dist);
    }

    camera.zoom += (targetZoom - camera.zoom) * 0.05;
    camera.x += (midX - W / (2 * camera.zoom) - camera.x) * 0.1;
    camera.y += ((midY - H / 2) - camera.y) * 0.08;

    // Clamp
    camera.x = Math.max(0, Math.min(LEVEL_WIDTH - W / camera.zoom, camera.x));
    camera.y = Math.max(-100, Math.min(H - H / camera.zoom + 60, camera.y));
}

function drawEndScreen() {

    // Darken

    ctx.fillRect(0, 0, W, H);

    const p2Gems = player2.gemsCollected;

    // Determine final winner considering gems
    let finalWinner;
    if (p1Gems > p2Gems) finalWinner = 1;
    else if (p2Gems > p1Gems) finalWinner = 2;
    else finalWinner = winner; // tiebreaker: who reached first

    ctx.fillStyle = wColor;
    ctx.font = 'bold 42px monospace';
    ctx.textAlign = 'center';
    ctx.fillText('PLAYER ' + finalWinner + ' WINS!', W/2, H/2 - 60);

    // Scores
    ctx.fillStyle = '#ffd700';
    ctx.font = '20px monospace';
    const total1 = p1Gems * 100 + (finalWinner === 1 ? 500 : 0);
    const total2 = p2Gems * 100 + (finalWinner === 2 ? 500 : 0);
    ctx.fillText('P1: ' + p1Gems + ' gems (' + total1 + ' pts)', W/2 - 150, H/2);
    ctx.fillText('P2: ' + p2Gems + ' gems (' + total2 + ' pts)', W/2 + 150, H/2);

    if (p1Gems === p2Gems) {
        ctx.fillStyle = '#aaa';
        ctx.font = '14px monospace';
        ctx.fillText('(Tie broken by finish order)', W/2, H/2 + 30);
    }

    const pulse = Math.sin(winTimer * 0.08) * 0.4 + 0.6;
    ctx.globalAlpha = pulse;
    ctx.fillStyle = '#fff';
    ctx.font = '18px monospace';
    ctx.fillText('Press R or ENTER to Play Again', W/2, H/2 + 80);
    ctx.globalAlpha = 1;
}

// Main loop
function gameLoop() {
    frameCount++;

    if (gameState === 'start') {
        drawStartScreen();
    } else if (gameState === 'playing') {
        updatePlayer(player1);
        updatePlayer(player2);
        updateMovingSpikes();
        updateCamera();
        drawBackground();
        drawWorld();
        drawHUD();
    } else if (gameState === 'finished') {
        // Still render the world
        drawBackground();
        drawWorld();
        drawHUD();
        drawEndScreen();
    }

    requestAnimationFrame(gameLoop);
}

resetGame();
gameLoop();
