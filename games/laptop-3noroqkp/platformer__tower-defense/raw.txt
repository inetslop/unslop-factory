<!DOCTYPE html>
<html lang="en">

<style>

<script>
const canvas = document.getElementById('game');
const ctx = canvas.getContext('2d');

let W, H;
function resize() {
    W = canvas.width = window.innerWidth;
    H = canvas.height = window.innerHeight;
}
window.addEventListener('resize', resize);
resize();

// Audio
let audioCtx = null;
function initAudio() {
    if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
}
function playSound(freq, dur, type, vol) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = type || 'square';
    osc.frequency.value = freq;
    gain.gain.setValueAtTime(vol || 0.1, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + dur);
}

// Game State
const STATE = { START: 0, PLAYING: 1, GAME_OVER: 2 };
let gameState = STATE.START;

// Input
const keys = {};
window.addEventListener('keydown', e => {
    keys[e.code] = true;
    if (gameState === STATE.START && (e.code === 'Space' || e.code === 'Enter')) {
        initAudio();
        startGame();
    }
    if (gameState === STATE.GAME_OVER && (e.code === 'Space' || e.code === 'Enter')) {
        startGame();
    }
});
window.addEventListener('keyup', e => { keys[e.code] = false; });

// Game variables
let player, platforms, towers, enemies, bullets, particles, gold, lives, wave, score, waveTimer, spawnTimer, screenShake, time, goldRegenTimer;

const GRAVITY = 0.6;
const PLAYER_SPEED = 4;
const JUMP_FORCE = -12;
const TOWER_COST = 50;
const TOWER_RANGE = 300;
const TOWER_COOLDOWN = 30;
const TOWER_DAMAGE = 20;
const ENEMY_BASE_HP = 60;
const KILL_REWARD = 15;

function startGame() {
    gameState = STATE.PLAYING;
    player = { x: 80, y: 0, vx: 0, vy: 0, w: 28, h: 36, onGround: false, facing: 1 };
    platforms = generatePlatforms();
    towers = [];
    enemies = [];
    bullets = [];
    particles = [];
    gold = 100;
    lives = 5;
    wave = 1;
    score = 0;
    waveTimer = 0;
    spawnTimer = 0;
    screenShake = 0;
    time = 0;
    goldRegenTimer = 0;
    player.y = H - 80;
}

function generatePlatforms() {
    const plats = [];
    // Ground
    plats.push({ x: 0, y: H - 50, w: W * 0.3, h: 20 });
    plats.push({ x: W * 0.4, y: H - 50, w: W * 0.6, h: 20 });
    // Floating platforms
    const configs = [
        { x: W * 0.15, y: H - 140, w: 120 },
        { x: W * 0.35, y: H - 180, w: 100 },
        { x: W * 0.55, y: H - 130, w: 140 },
        { x: W * 0.72, y: H - 200, w: 110 },
        { x: W * 0.85, y: H - 120, w: 90 },
        { x: W * 0.25, y: H - 260, w: 100 },
        { x: W * 0.6, y: H - 280, w: 130 },
    ];
    for (const c of configs) {
        plats.push({ x: c.x, y: c.y, w: c.w, h: 14 });
    }
    return plats;
}

function spawnEnemy() {
    const hp = ENEMY_BASE_HP + wave * 15;
    const speed = 0.8 + wave * 0.1;
    const y = H - 50 - (Math.random() < 0.3 ? 90 : 0);
    enemies.push({ x: W + 20, y: y, w: 30, h: 36, hp: hp, maxHp: hp, speed: speed, onGround: false, vy: 0, type: Math.floor(Math.random() * 3) });
}

function spawnParticle(x, y, color, count) {
    for (let i = 0; i < count; i++) {
        const angle = Math.random() * Math.PI * 2;
        const speed = 1 + Math.random() * 3;
        particles.push({ x, y, vx: Math.cos(angle) * speed, vy: Math.sin(angle) * speed - 2, life: 30 + Math.random() * 20, color, size: 2 + Math.random() * 4 });
    }
}

function update() {
    if (gameState !== STATE.PLAYING) return;
    time++;
    goldRegenTimer++;
    if (goldRegenTimer >= 180) { gold += 5; goldRegenTimer = 0; }

    // Wave management
    waveTimer++;
    const enemiesNeeded = 3 + wave * 2;
    spawnTimer++;
    const spawnInterval = Math.max(60, 120 - wave * 5);
    if (spawnTimer >= spawnInterval && enemies.length < enemiesNeeded) {
        spawnEnemy();
        spawnTimer = 0;
    }
    if (enemies.length === 0 && waveTimer > 300) {
        wave++;
        waveTimer = 0;
        gold += 30;
    }

    // Player physics
    player.vx = 0;
    if (keys['KeyA'] || keys['ArrowLeft']) { player.vx = -PLAYER_SPEED; player.facing = -1; }
    if (keys['KeyD'] || keys['ArrowRight']) { player.vx = PLAYER_SPEED; player.facing = 1; }
    if ((keys['KeyW'] || keys['Space'] || keys['ArrowUp']) && player.onGround) {
        player.vy = JUMP_FORCE;
        player.onGround = false;
        playSound(200, 0.1, 'sine', 0.05);
    }

    player.vy += GRAVITY;
    player.x += player.vx;
    player.y += player.vy;

    // Platform collision for player
    player.onGround = false;
    for (const p of platforms) {
        if (player.x + player.w > p.x && player.x < p.x + p.w) {
            if (player.vy >= 0 && player.y + player.h <= p.y + 5 && player.y + player.h + player.vy >= p.y) {
                player.y = p.y - player.h;
                player.vy = 0;
                player.onGround = true;
            }
        }
    }

    // Ground collision (bottom)
    if (player.y + player.h > H) {
        player.y = H - player.h;
        player.vy = 0;
        player.onGround = true;
    }
    player.x = Math.max(0, Math.min(W - player.w, player.x));

    // Place tower
    if (keys['KeyE'] || keys['KeyF']) {
        if (gold >= TOWER_COST && player.onGround) {
            const tx = player.facing === 1 ? player.x + player.w : player.x - 20;
            let canPlace = true;
            for (const t of towers) {
                if (Math.abs(t.x - tx) < 30) canPlace = false;
            }
            // Check if on platform
            let onPlat = false;
            for (const p of platforms) {
                if (tx + 20 > p.x && tx < p.x + p.w && Math.abs(player.y + player.h - p.y) < 10) onPlat = true;
            }
            if (canPlace && onPlat) {
                towers.push({ x: tx, y: player.y + player.h - 30, cooldown: 0, angle: 0 });
                gold -= TOWER_COST;
                playSound(400, 0.15, 'square', 0.08);
                spawnParticle(tx + 10, player.y + player.h, '#4ff', 5);
            } else {
                playSound(150, 0.2, 'sawtooth', 0.05);
            }
        }
    }

    // Update towers
    for (const t of towers) {
        t.cooldown--;
        if (t.cooldown <= 0) {
            let target = null;
            let dist = TOWER_RANGE;
            for (const e of enemies) {
                const d = Math.hypot(e.x - t.x, e.y - t.y);
                if (d < dist) { dist = d; target = e; }
            }
            if (target) {
                const angle = Math.atan2(target.y - t.y, target.x - t.x);
                t.angle = angle;
                bullets.push({ x: t.x + 10, y: t.y, vx: Math.cos(angle) * 8, vy: Math.sin(angle) * 8, damage: TOWER_DAMAGE });
                t.cooldown = TOWER_COOLDOWN;
                playSound(600 + Math.random() * 200, 0.05, 'square', 0.03);
            }
        }
    }

    // Update bullets
    for (let i = bullets.length - 1; i >= 0; i--) {
        const b = bullets[i];
        b.x += b.vx;
        b.y += b.vy;
        if (b.x < 0 || b.x > W || b.y < 0 || b.y > H) { bullets.splice(i, 1); continue; }
        for (let j = enemies.length - 1; j >= 0; j--) {
            const e = enemies[j];
            if (b.x > e.x && b.x < e.x + e.w && b.y > e.y && b.y < e.y + e.h) {
                e.hp -= b.damage;
                spawnParticle(b.x, b.y, '#ff0', 3);
                bullets.splice(i, 1);
                if (e.hp <= 0) {
                    spawnParticle(e.x + e.w/2, e.y + e.h/2, '#f44', 10);
                    screenShake = 5;
                    gold += KILL_REWARD;
                    score += 10 * wave;
                    enemies.splice(j, 1);
                    playSound(80, 0.3, 'sawtooth', 0.1);
                }
                break;
            }
        }
    }

    // Update enemies
    for (let i = enemies.length - 1; i >= 0; i--) {
        const e = enemies[i];
        e.x -= e.speed;
        e.vy += GRAVITY;
        e.y += e.vy;

        // Enemy ground collision
        for (const p of platforms) {
            if (e.x + e.w > p.x && e.x < p.x + p.w) {
                if (e.vy >= 0 && e.y + e.h <= p.y + 5 && e.y + e.h + e.vy >= p.y) {
                    e.y = p.y - e.h;
                    e.vy = 0;
                }
            }
        }
        if (e.y + e.h > H) { e.y = H - e.h; e.vy = 0; }

        // Enemy reaches left
        if (e.x <= 0) {
            enemies.splice(i, 1);
            lives--;
            screenShake = 8;
            spawnParticle(20, e.y, '#f00', 8);
            playSound(60, 0.4, 'sawtooth', 0.15);
            if (lives <= 0) {
                gameState = STATE.GAME_OVER;
            }
        }

        // Enemy touches player
        if (e.x + e.w > player.x && e.x < player.x + player.w && e.y + e.h > player.y && e.y < player.y + player.h) {
            lives--;
            screenShake = 6;
            enemies.splice(i, 1);
            spawnParticle(player.x + player.w/2, player.y + player.h/2, '#f80', 8);
            playSound(100, 0.3, 'sawtooth', 0.12);
            if (lives <= 0) gameState = STATE.GAME_OVER;
        }
    }

    // Update particles
    for (let i = particles.length - 1; i >= 0; i--) {
        const p = particles[i];
        p.x += p.vx;
        p.y += p.vy;
        p.vy += 0.2;
        p.life--;
        if (p.life <= 0) particles.splice(i, 1);
    }

    // Screen shake decay
    if (screenShake > 0) screenShake *= 0.85;
}

function draw() {
    ctx.save();
    if (screenShake > 0.5) {
        ctx.translate((Math.random() - 0.5) * screenShake, (Math.random() - 0.5) * screenShake);
    }

    // Background
    const grad = ctx.createLinearGradient(0, 0, 0, H);
    grad.addColorStop(0, '#0a0a2e');
    grad.addColorStop(1, '#1a1a3e');
    ctx.fillStyle = grad;
    ctx.fillRect(-10, -10, W + 20, H + 20);

    // Stars
    for (let i = 0; i < 50; i++) {
        const sx = (i * 137.7 + 50) % W;
        const sy = (i * 97.3 + 20) % (H * 0.6);
        const brightness = 0.3 + Math.sin(time * 0.02 + i) * 0.2;
        ctx.fillStyle = `rgba(255,255,255,${brightness})`;
        ctx.fillRect(sx, sy, 2, 2);
    }

    if (gameState === STATE.START) {
        drawStartScreen();
        ctx.restore();
        return;
    }

    // Platforms
    for (const p of platforms) {
        ctx.fillStyle = '#2a2a5e';
        ctx.fillRect(p.x, p.y, p.w, p.h);
        ctx.fillStyle = '#4a4a9e';
        ctx.fillRect(p.x, p.y, p.w, 3);
    }

    // Towers
    for (const t of towers) {
        ctx.fillStyle = '#2266aa';
        ctx.fillRect(t.x, t.y, 20, 30);
        ctx.fillStyle = '#44aaff';
        ctx.fillRect(t.x + 2, t.y + 2, 16, 8);
        // Barrel
        ctx.save();
        ctx.translate(t.x + 10, t.y + 5);
        ctx.rotate(t.angle);
        ctx.fillStyle = '#aaddff';
        ctx.fillRect(0, -3, 18, 6);
        ctx.restore();
    }

    // Player
    ctx.fillStyle = '#4f4';
    ctx.fillRect(player.x, player.y, player.w, player.h);
    ctx.fillStyle = '#8f8';
    ctx.fillRect(player.x + 4, player.y + 4, 20, 12);
    // Eyes
    ctx.fillStyle = '#fff';
    const eyeX = player.facing === 1 ? player.x + 16 : player.x + 6;
    ctx.fillRect(eyeX, player.y + 8, 5, 5);
    // Gun arm
    ctx.fillStyle = '#8f8';
    if (player.facing === 1) {
        ctx.fillRect(player.x + player.w, player.y + 20, 10, 4);
    } else {
        ctx.fillRect(player.x - 10, player.y + 20, 10, 4);
    }

    // Enemies
    for (const e of enemies) {
        const colors = ['#f44', '#f84', '#f4a'];
        ctx.fillStyle = colors[e.type % 3];
        ctx.fillRect(e.x, e.y, e.w, e.h);
        ctx.fillStyle = '#fff';
        ctx.fillRect(e.x + 4, e.y + 6, 6, 6);
        ctx.fillRect(e.x + e.w - 10, e.y + 6, 6, 6);
        // HP bar
        const hpRatio = e.hp / e.maxHp;
        ctx.fillStyle = '#300';
        ctx.fillRect(e.x, e.y - 8, e.w, 4);
        ctx.fillStyle = `hsl(${hpRatio * 120}, 100%, 50%)`;
        ctx.fillRect(e.x, e.y - 8, e.w * hpRatio, 4);
    }

    // Bullets
    for (const b of bullets) {
        ctx.fillStyle = '#ff0';
        ctx.beginPath();
        ctx.arc(b.x, b.y, 3, 0, Math.PI * 2);
        ctx.fill();
    }

    // Particles
    for (const p of particles) {
        const alpha = p.life / 50;
        ctx.fillStyle = p.color + Math.floor(alpha * 255).toString(16).padStart(2, '0');
        ctx.fillRect(p.x - p.size/2, p.y - p.size/2, p.size, p.size);
    }

    // HUD
    ctx.fillStyle = 'rgba(0,0,0,0.5)';
    ctx.fillRect(10, 10, 220, 90);
    ctx.font = '14px monospace';
    ctx.fillStyle = '#ff0';
    ctx.fillText(`Gold: ${Math.floor(gold)} (Cost: ${TOWER_COST})`, 20, 30);
    ctx.fillStyle = '#f44';
    ctx.fillText(`Lives: ${lives}`, 20, 50);
    ctx.fillStyle = '#4ff';
    ctx.fillText(`Wave: ${wave}`, 20, 70);
    ctx.fillStyle = '#fff';
    ctx.fillText(`Score: ${score}`, 20, 90);

    // Base indicator
    ctx.fillStyle = '#f0f';
    ctx.font = 'bold 16px monospace';
    ctx.fillText('BASE', 5, H - 30);
    ctx.fillStyle = `rgba(255,0,255,${0.3 + Math.sin(time * 0.1) * 0.2})`;
    ctx.fillRect(0, 0, 4, H);

    if (gameState === STATE.GAME_OVER) {
        drawGameOver();
    }

    ctx.restore();
}

function drawStartScreen() {
    ctx.fillStyle = '#fff';
    ctx.font = 'bold 36px monospace';
    ctx.textAlign = 'center';
    ctx.fillText('TOWER PLATFORM DEFENSE', W/2, H/4);
    ctx.font = '18px monospace';
    ctx.fillStyle = '#aaf';
    ctx.fillText('Run & jump across platforms to place turrets!', W/2, H/2 - 60);
    ctx.fillText('Enemies attack from the right. Defend your base!', W/2, H/2 - 35);
    ctx.font = '16px monospace';
    ctx.fillStyle = '#8f8';
    ctx.fillText('[A/D or ←→] Move   [W/Space] Jump', W/2, H/2 + 20);
    ctx.fillText('[E or F] Place Tower (costs gold)', W/2, H/2 + 45);
    ctx.fillText('Kill enemies to earn gold. Gold regenerates over time.', W/2, H/2 + 70);
    ctx.fillStyle = '#ff0';
    ctx.font = 'bold 20px monospace';
    const blink = Math.sin(Date.now() * 0.005) > 0;
    if (blink) ctx.fillText('PRESS SPACE TO START', W/2, H * 0.75);
    ctx.textAlign = 'left';
}

function drawGameOver() {
    ctx.fillStyle = 'rgba(0,0,0,0.7)';
    ctx.fillRect(0, 0, W, H);
    ctx.textAlign = 'center';
    ctx.font = 'bold 40px monospace';
    ctx.fillStyle = '#f44';
    ctx.fillText('GAME OVER', W/2, H/3);
    ctx.font = '20px monospace';
    ctx.fillStyle = '#fff';
    ctx.fillText(`Final Score: ${score}`, W/2, H/2 - 20);
    ctx.fillText(`Waves Survived: ${wave}`, W/2, H/2 + 15);
    ctx.fillStyle = '#ff0';
    const blink = Math.sin(Date.now() * 0.005) > 0;
    if (blink) ctx.fillText('PRESS SPACE TO RESTART', W/2, H * 0.7);
    ctx.textAlign = 'left';
}

function gameLoop() {
    update();
    draw();
    requestAnimationFrame(gameLoop);
}

gameLoop();
</script>
