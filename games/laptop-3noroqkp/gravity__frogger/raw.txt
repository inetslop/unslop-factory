const canvas = document.getElementById('game');
const ctx = canvas.getContext('2d');

let W, H, CELL;
function resize() {
  const aspect = 13 / 8;
  let w = window.innerWidth;
  let h = window.innerHeight;
  if (w / h > aspect) { w = h * aspect; }
  else { h = w / aspect; }
  W = Math.floor(w);
  H = Math.floor(h);
  canvas.width = W;
  canvas.height = H;
  CELL = Math.floor(W / 13);
}
resize();
window.addEventListener('resize', resize);

const COLS = 13, ROWS = 6;
const GAME_STATE = { MENU: 0, PLAYING: 1, DEAD: 2, WIN: 3 };
let state = GAME_STATE.MENU;
let frog = { col: 6, row: 5 };
let score = 0, lives = 3, level = 1;
let gravityDir = null; // 'up','down','left','right'
let gravityTimer = 0;
const GRAVITY_INTERVAL = 4000;
let gravityFlash = 0;

// Cars
let cars = [];
// Turtles
let turtles = [];
// Particles
let particles = [];

function initLevel() {
  cars = [];
  turtles = [];
  // Road rows: 2,3,4 (rows 1-3 in 0-indexed for road)
  const roadSpeeds = [2.5, -3, 3.5, -2];
  const roadRows = [2, 3, 4, 5];
  for (let r = 0; r < 4; r++) {
    const row = roadRows[r];
    const speed = roadSpeeds[r] * (1 + level * 0.2);
    const count = 3 + Math.floor(level / 2);
    for (let i = 0; i < count; i++) {
      cars.push({
        col: Math.floor(Math.random() * COLS),
        row: row,
        speed: speed,
        w: 1.8,
        color: `hsl(${Math.random()*360},70%,55%)`
      });
    }
  }
  // Turtles in row 1
  const tCount = 4;
  for (let i = 0; i < tCount; i++) {
    turtles.push({
      col: Math.floor(Math.random() * COLS),
      row: 1,
      speed: 1.5 + level * 0.3,
      w: 1.2
    });
  }
  // Home pads in row 0
  homePads = [];
  for (let c = 0; c < COLS; c += 2) {
    if (c % 2 === 0) homePads.push(c);
  }
}

let homePads = [];
initLevel();

function startGame() {
  state = GAME_STATE.PLAYING;
  frog = { col: 6, row: 5 };
  score = 0; lives = 3; level = 1;
  gravityDir = null;
  gravityTimer = GRAVITY_INTERVAL;
  initLevel();
}

function resetFrog() {
  frog = { col: 6, row: 5 };
}

// Audio
let audioCtx = null;
function playSound(freq, dur, type) {
  if (!audioCtx) return;
  const osc = audioCtx.createOscillator();
  const gain = audioCtx.createGain();
  osc.type = type || 'square';
  osc.frequency.value = freq;
  gain.gain.setValueAtTime(0.15, audioCtx.currentTime);
  gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
  osc.connect(gain);
  gain.connect(audioCtx.destination);
  osc.start();
  osc.stop(audioCtx.currentTime + dur);
}

function initAudio() {
  if (!audioCtx) {
    audioCtx = new (window.AudioContext || window.webkitAudioContext)();
  }
}

document.addEventListener('keydown', (e) => {
  initAudio();
  if (state === GAME_STATE.MENU) {
    startGame();
    return;
  }
  if (state === GAME_STATE.DEAD || state === GAME_STATE.WIN) {
    if (e.key === 'r' || e.key === 'R') startGame();
    return;
  }
  if (state !== GAME_STATE.PLAYING) return;

  let dx = 0, dy = 0;
  if (e.key === 'ArrowUp') dy = -1;
  else if (e.key === 'ArrowDown') dy = 1;
  else if (e.key === 'ArrowLeft') dx = -1;
  else if (e.key === 'ArrowRight') dx = 1;
  else return;

  e.preventDefault();
  moveFrog(dx, dy);
});

function moveFrog(dx, dy) {
  let nc = frog.col + dx;
  let nr = frog.row + dy;

  if (nr < 0 || nr >= ROWS || nc < 0 || nc >= COLS) return;

  // Check river row - must be on turtle
  if (nr === 1) {
    let onTurtle = false;
    for (let t of turtles) {
      if (t.row === 1 && Math.abs(t.col - nc) < t.w / 2 + 0.3) {
        onTurtle = true;
        break;
      }
    }
    if (!onTurtle) {
      drownFrog();
      return;
    }
  }

  frog.col = nc;
  frog.row = nr;
  playSound(200 + Math.random() * 100, 0.1, 'square');

  // Check home
  if (nr === 0) {
    let onPad = false;
    for (let p of homePads) {
      if (nc === p) onPad = true;
    }
    if (onPad) {
      winLevel();
      return;
    } else {
      // Off pad in row 0 = death
      drownFrog();
      return;
    }
  }
}

function drownFrog() {
  playSound(150, 0.3, 'sawtooth');
  spawnParticles(frog.col, frog.row, '#44ff88', 8);
  lives--;
  if (lives <= 0) {
    state = GAME_STATE.DEAD;
  } else {
    resetFrog();
  }
}

function winLevel() {
  playSound(600, 0.2, 'sine');
  setTimeout(() => playSound(800, 0.2, 'sine'), 100);
  score += 50 * level;
  spawnParticles(frog.col, frog.row, '#ffff44', 15);
  level++;
  initLevel();
  resetFrog();
}

function spawnParticles(col, row, color, count) {
  for (let i = 0; i < count; i++) {
    particles.push({
      x: (col + 0.5) * CELL,
      y: (row + 0.5) * CELL,
      vx: (Math.random() - 0.5) * 4,
      vy: (Math.random() - 0.5) * 4,
      life: 1,
      color: color
    });
  }
}

function update(dt) {
  if (state !== GAME_STATE.PLAYING) return;

  // Update cars
  for (let c of cars) {
    c.col += c.speed * dt / 1000;
    if (c.col > COLS + 1) c.col = -2;
    if (c.col < -2) c.col = COLS + 1;
  }

  // Update turtles
  for (let t of turtles) {
    t.col += t.speed * dt / 1000;
    if (t.col > COLS + 1) t.col = -2;
    if (t.col < -2) t.col = COLS + 1;
  }

  // Gravity mechanic
  gravityTimer -= dt;
  if (gravityTimer <= 0) {
    const dirs = ['up', 'down', 'left', 'right'];
    gravityDir = dirs[Math.floor(Math.random() * 4)];
    gravityTimer = GRAVITY_INTERVAL - level * 300;
    if (gravityTimer < 2000) gravityTimer = 2000;
    gravityFlash = 1;
    playSound(300, 0.15, 'triangle');
  }
  gravityFlash = Math.max(0, gravityFlash - dt / 300);

  // Apply gravity pull every ~800ms if frog doesn't move against it
  applyGravity(dt);

  // Check collisions with cars
  for (let c of cars) {
    const dc = Math.abs(c.col - frog.col);
    const dr = Math.abs(c.row - frog.row);
    if (dc < c.w / 2 + 0.3 && dr < 0.6) {
      playSound(100, 0.4, 'sawtooth');
      spawnParticles(frog.col, frog.row, '#ff4444', 10);
      lives--;
      if (lives <= 0) state = GAME_STATE.DEAD;
      else resetFrog();
      return;
    }
  }

  // Update particles
  for (let i = particles.length - 1; i >= 0; i--) {
    const p = particles[i];
    p.x += p.vx;
    p.y += p.vy;
    p.vy += 0.15;
    p.life -= dt / 800;
    if (p.life <= 0) particles.splice(i, 1);
  }
}

let gravityAccum = 0;
function applyGravity(dt) {
  if (!gravityDir) return;
  gravityAccum += dt;
  if (gravityAccum >= 900 - level * 50) {
    gravityAccum = 0;
    let dx = 0, dy = 0;
    if (gravityDir === 'up') dy = -1;
    else if (gravityDir === 'down') dy = 1;
    else if (gravityDir === 'left') dx = -1;
    else if (gravityDir === 'right') dx = 1;

    let nc = frog.col + dx;
    let nr = frog.row + dy;

    // Only pull if valid and not blocked
    if (nr >= 0 && nr < ROWS && nc >= 0 && nc < COLS) {
      if (nr === 1) {
        let onTurtle = false;
        for (let t of turtles) {
          if (t.row === 1 && Math.abs(t.col - nc) < t.w / 2 + 0.3) onTurtle = true;
        }
        if (!onTurtle) { drownFrog(); return; }
      }
      frog.col = nc;
      frog.row = nr;
      playSound(120, 0.08, 'sine');

      if (nr === 0) {
        let onPad = false;
        for (let p of homePads) if (nc === p) onPad = true;
        if (!onPad) { drownFrog(); return; }
      }
    }
  }
}

function draw() {
  ctx.fillStyle = '#0a0a1a';
  ctx.fillRect(0, 0, W, H);

  const ox = (W - COLS * CELL) / 2;
  const oy = (H - ROWS * CELL) / 2;

  // Draw rows
  for (let r = 0; r < ROWS; r++) {
    let color;
    if (r === 0) color = '#1a3a1a';
    else if (r === 1) color = '#0a2a4a';
    else color = '#1a1a1a';

    ctx.fillStyle = color;
    ctx.fillRect(ox, oy + r * CELL, COLS * CELL, CELL);

    // Row dividers
    ctx.strokeStyle = 'rgba(255,255,255,0.1)';
    ctx.lineWidth = 1;
    for (let c = 0; c <= COLS; c++) {
      ctx.beginPath();
      ctx.moveTo(ox + c * CELL, oy + r * CELL);
      ctx.lineTo(ox + c * CELL, oy + (r + 1) * CELL);
      ctx.stroke();
    }
  }

  // Home pads
  for (let p of homePads) {
    ctx.fillStyle = '#2a5a2a';
    ctx.beginPath();
    ctx.ellipse(ox + (p + 0.5) * CELL, oy + 0.5 * CELL, CELL * 0.35, CELL * 0.25, 0, 0, Math.PI * 2);
    ctx.fill();
    ctx.strokeStyle = '#4a8a4a';
    ctx.lineWidth = 2;
    ctx.stroke();
  }

  // Turtles
  for (let t of turtles) {
    drawTurtle(ox + (t.col + 0.5) * CELL, oy + 1.5 * CELL);
  }

  // Cars
  for (let c of cars) {
    const cx = ox + (c.col + 0.5) * CELL;
    const cy = oy + (c.row + 0.5) * CELL;
    ctx.fillStyle = c.color;
    ctx.fillRect(cx - CELL * 0.4, cy - CELL * 0.25, CELL * 0.8, CELL * 0.5);
    // Windows
    ctx.fillStyle = 'rgba(200,230,255,0.6)';
    ctx.fillRect(cx - CELL * 0.15, cy - CELL * 0.15, CELL * 0.3, CELL * 0.3);
    // Wheels
    ctx.fillStyle = '#222';
    ctx.beginPath();
    ctx.arc(cx - CELL * 0.25, cy + CELL * 0.25, CELL * 0.12, 0, Math.PI * 2);
    ctx.fill();
    ctx.beginPath();
    ctx.arc(cx + CELL * 0.25, cy + CELL * 0.25, CELL * 0.12, 0, Math.PI * 2);
    ctx.fill();
  }

  // Frog
  if (state === GAME_STATE.PLAYING) {
    drawFrog(ox + (frog.col + 0.5) * CELL, oy + (frog.row + 0.5) * CELL);
  }

  // Particles
  for (let p of particles) {
    ctx.globalAlpha = p.life;
    ctx.fillStyle = p.color;
    ctx.fillRect(p.x - 3, p.y - 3, 6, 6);
    ctx.globalAlpha = 1;
  }

  // UI
  drawUI(ox, oy);

  if (state === GAME_STATE.MENU) drawMenu();
  if (state === GAME_STATE.DEAD) drawGameOver();
  if (state === GAME_STATE.WIN) drawWin();
}

function drawFrog(x, y) {
  ctx.fillStyle = '#44ff88';
  ctx.beginPath();
  ctx.ellipse(x, y, CELL * 0.35, CELL * 0.3, 0, 0, Math.PI * 2);
  ctx.fill();
  // Eyes
  ctx.fillStyle = '#fff';
  ctx.beginPath();
  ctx.arc(x - CELL * 0.12, y - CELL * 0.15, CELL * 0.1, 0, Math.PI * 2);
  ctx.fill();
  ctx.beginPath();
  ctx.arc(x + CELL * 0.12, y - CELL * 0.15, CELL * 0.1, 0, Math.PI * 2);
  ctx.fill();
  // Pupils
  ctx.fillStyle = '#000';
  ctx.beginPath();
  ctx.arc(x - CELL * 0.12, y - CELL * 0.13, CELL * 0.05, 0, Math.PI * 2);
  ctx.fill();
  ctx.beginPath();
  ctx.arc(x + CELL * 0.12, y - CELL * 0.13, CELL * 0.05, 0, Math.PI * 2);
  ctx.fill();
}

function drawTurtle(x, y) {
  ctx.fillStyle = '#3a6a3a';
  ctx.beginPath();
  ctx.ellipse(x, y, CELL * 0.3, CELL * 0.25, 0, 0, Math.PI * 2);
  ctx.fill();
  // Shell pattern
  ctx.strokeStyle = '#2a4a2a';
  ctx.lineWidth = 1;
  ctx.beginPath();
  ctx.moveTo(x - CELL * 0.15, y);
  ctx.lineTo(x + CELL * 0.15, y);
  ctx.stroke();
  // Head
  ctx.fillStyle = '#4a8a4a';
  ctx.beginPath();
  ctx.arc(x + CELL * 0.3, y, CELL * 0.1, 0, Math.PI * 2);
  ctx.fill();
}

function drawUI(ox, oy) {
  // Score
  ctx.fillStyle = '#fff';
  ctx.font = `${Math.floor(CELL * 0.45)}px monospace`;
  ctx.textAlign = 'left';
  ctx.fillText(`SCORE: ${score}`, ox + 5, oy - CELL * 0.3);

  // Lives
  ctx.fillStyle = '#44ff88';
  for (let i = 0; i < lives; i++) {
    ctx.beginPath();
    ctx.arc(ox + COLS * CELL - 20 - i * 25, oy - CELL * 0.3 + 5, 8, 0, Math.PI * 2);
    ctx.fill();
  }

  // Level
  ctx.fillStyle = '#aaa';
  ctx.textAlign = 'center';
  ctx.fillText(`LEVEL ${level}`, ox + (COLS * CELL) / 2, oy - CELL * 0.3);

  // Gravity indicator
  if (gravityDir && state === GAME_STATE.PLAYING) {
    const gx = ox + COLS * CELL - 30;
    const gy = oy + ROWS * CELL / 2;
    ctx.save();
    ctx.translate(gx, gy);

    // Flash effect
    if (gravityFlash > 0) {
      ctx.globalAlpha = gravityFlash;
      ctx.fillStyle = '#ff4444';
      ctx.beginPath();
      ctx.arc(0, 0, CELL * 0.5, 0, Math.PI * 2);
      ctx.fill();
      ctx.globalAlpha = 1;
    }

    ctx.strokeStyle = '#ff6666';
    ctx.lineWidth = 3;
    ctx.beginPath();
    if (gravityDir === 'up') { ctx.moveTo(0, 15); ctx.lineTo(0, -15); ctx.moveTo(-8, -5); ctx.lineTo(0, -15); ctx.lineTo(8, -5); }
    else if (gravityDir === 'down') { ctx.moveTo(0, -15); ctx.lineTo(0, 15); ctx.moveTo(-8, 5); ctx.lineTo(0, 15); ctx.lineTo(8, 5); }
    else if (gravityDir === 'left') { ctx.moveTo(15, 0); ctx.lineTo(-15, 0); ctx.moveTo(-5, -8); ctx.lineTo(-15, 0); ctx.lineTo(-5, 8); }
    else if (gravityDir === 'right') { ctx.moveTo(-15, 0); ctx.lineTo(15, 0); ctx.moveTo(5, -8); ctx.lineTo(15, 0); ctx.lineTo(5, 8); }
    ctx.stroke();

    ctx.fillStyle = '#ff6666';
    ctx.font = `${Math.floor(CELL * 0.25)}px monospace`;
    ctx.textAlign = 'center';
    ctx.fillText('GRAV', 0, 30);
    ctx.restore();
  }
}

function drawMenu() {
  ctx.fillStyle = 'rgba(0,0,0,0.8)';
  ctx.fillRect(0, 0, W, H);

  const cx = W / 2;
  const cy = H / 2;

  ctx.fillStyle = '#44ff88';
  ctx.font = `bold ${Math.floor(H * 0.08)}px monospace`;
  ctx.textAlign = 'center';
  ctx.fillText('FROGGER', cx, cy - H * 0.15);
  ctx.fillStyle = '#ff6666';
  ctx.font = `bold ${Math.floor(H * 0.06)}px monospace`;
  ctx.fillText('GRAVITY', cx, cy - H * 0.08);

  ctx.fillStyle = '#fff';
  ctx.font = `${Math.floor(H * 0.03)}px monospace`;
  ctx.fillText('Cross the road! Avoid cars & water.', cx, cy + H * 0.02);
  ctx.fillText('BUT watch out for GRAVITY PULLS!', cx, cy + H * 0.06);
  ctx.fillText('', cx, cy + H * 0.1);
  ctx.fillStyle = '#aaa';
  ctx.font = `${Math.floor(H * 0.025)}px monospace`;
  ctx.fillText('Arrow Keys = Move', cx, cy + H * 0.1);
  ctx.fillText('Ride turtles in the river', cx, cy + H * 0.13);
  ctx.fillText('Reach home pads (row 0) to advance', cx, cy + H * 0.16);
  ctx.fillText('Gravity will pull you every few seconds!', cx, cy + H * 0.2);

  ctx.fillStyle = '#44ff88';
  ctx.font = `${Math.floor(H * 0.03)}px monospace`;
  const blink = Math.sin(Date.now() / 400) > 0;
  if (blink) ctx.fillText('PRESS ANY KEY TO START', cx, cy + H * 0.28);

  // Draw a little frog on menu
  drawFrog(cx, cy - H * 0.22);
}

function drawGameOver() {
  ctx.fillStyle = 'rgba(0,0,0,0.7)';
  ctx.fillRect(0, 0, W, H);
  const cx = W / 2, cy = H / 2;
  ctx.fillStyle = '#ff4444';
  ctx.font = `bold ${Math.floor(H * 0.08)}px monospace`;
  ctx.textAlign = 'center';
  ctx.fillText('GAME OVER', cx, cy - H * 0.05);
  ctx.fillStyle = '#fff';
  ctx.font = `${Math.floor(H * 0.04)}px monospace`;
  ctx.fillText(`Final Score: ${score}`, cx, cy + H * 0.05);
  ctx.fillText('Press R to Restart', cx, cy + H * 0.12);
}

function drawWin() {
  ctx.fillStyle = 'rgba(0,0,0,0.7)';
  ctx.fillRect(0, 0, W, H);
  const cx = W / 2, cy = H / 2;
  ctx.fillStyle = '#44ff88';
  ctx.font = `bold ${Math.floor(H * 0.07)}px monospace`;
  ctx.textAlign = 'center';
  ctx.fillText('YOU WIN!', cx, cy - H * 0.05);
  ctx.fillStyle = '#fff';
  ctx.font = `${Math.floor(H * 0.04)}px monospace`;
  ctx.fillText(`Score: ${score}`, cx, cy + H * 0.05);
  ctx.fillText('Press R to Play Again', cx, cy + H * 0.12);
}

let lastTime = 0;
function gameLoop(time) {
  const dt = Math.min(time - lastTime, 50);
  lastTime = time;
  update(dt);
  draw();
  requestAnimationFrame(gameLoop);
}
requestAnimationFrame(gameLoop);
