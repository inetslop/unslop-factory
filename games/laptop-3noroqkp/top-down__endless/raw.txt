```html
<!DOCTYPE html>
<html lang="en">
<head>
<meta charset="UTF-8">
<meta name="viewport" content="width=device-width, initial-scale=1.0">
<title>Endless Highway</title>
<style>
* { margin: 0; padding: 0; box-sizing: border-box; }
html, body { width: 100%; height: 100%; overflow: hidden; background: #1a1a2e; }
canvas { display: block; width: 100%; height: 100%; }

<body>
<canvas id="game"></canvas>
<script>
const canvas = document.getElementById('game');
const ctx = canvas.getContext('2d');

let W, H, scale;
const LOGICAL_W = 800;
const ROAD_LEFT = 150;
const ROAD_RIGHT = 650;
const ROAD_W = ROAD_RIGHT - ROAD_LEFT;
const LANE_COUNT = 4;
const LANE_W = ROAD_W / LANE_COUNT;

function resize() {
    W = window.innerWidth;
    H = window.innerHeight;
    canvas.width = W;
    canvas.height = H;
    scale = Math.min(W / LOGICAL_W, H / 900);
}
window.addEventListener('resize', resize);
resize();

// Audio
let audioCtx = null;
function initAudio() {
    if (!audioCtx) {
        audioCtx = new (window.AudioContext || window.webkitAudioContext)();
    }
}
function playSound(type) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    if (type === 'crash') {
        osc.type = 'sawtooth';
        osc.frequency.setValueAtTime(150, audioCtx.currentTime);
        osc.frequency.exponentialRampToValueAtTime(30, audioCtx.currentTime + 0.5);
        gain.gain.setValueAtTime(0.4, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.5);
        osc.start();
        osc.stop(audioCtx.currentTime + 0.5);
    } else if (type === 'score') {
        osc.type = 'sine';
        osc.frequency.setValueAtTime(800, audioCtx.currentTime);
        osc.frequency.exponentialRampToValueAtTime(1200, audioCtx.currentTime + 0.1);
        gain.gain.setValueAtTime(0.15, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.15);
        osc.start();
        osc.stop(audioCtx.currentTime + 0.15);
    } else if (type === 'milestone') {
        osc.type = 'sine';
        osc.frequency.setValueAtTime(440, audioCtx.currentTime);
        gain.gain.setValueAtTime(0.2, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.3);
        osc.start();
        osc.stop(audioCtx.currentTime + 0.3);
    }
}

// Game state
let state = 'start'; // start, playing, gameover
let player, obstacles, particles, roadOffset, score, speed, maxSpeed, distance;
let trees = [];
let lastMilestone = 0;
let shakeTime = 0;
let shakeIntensity = 0;

const keys = {};
window.addEventListener('keydown', e => {
    keys[e.code] = true;
    if (state === 'start' && (e.code === 'Space' || e.code === 'Enter')) startGame();
    else if (state === 'gameover' && (e.code === 'Space' || e.code === 'Enter' || e.code === 'KeyR')) startGame();
    initAudio();
});
window.addEventListener('keyup', e => { keys[e.code] = false; });

function startGame() {
    state = 'playing';
    score = 0;
    distance = 0;
    speed = 4;
    maxSpeed = 18;
    roadOffset = 0;
    lastMilestone = 0;
    shakeTime = 0;
    player = { x: LOGICAL_W / 2, y: 750, w: 36, h: 64, vx: 0 };
    obstacles = [];
    particles = [];
    trees = [];
    for (let i = 0; i < 12; i++) {
        trees.push({ x: Math.random() < 0.5 ? 30 + Math.random() * 90 : 680 + Math.random() * 90, y: Math.random() * 900 - 100 });
    }
}

function spawnObstacle() {
    const lane = Math.floor(Math.random() * LANE_COUNT);
    const x = ROAD_LEFT + lane * LANE_W + LANE_W / 2;
    const type = Math.random();
    let obs;
    if (type < 0.6) {
        obs = { x, y: -80, w: 34, h: 60, vx: 0, type: 'car', color: `hsl(${Math.floor(Math.random()*360)},70%,50%)` };
    } else if (type < 0.8) {
        obs = { x, y: -40, w: 20, h: 20, vx: 0, type: 'cone' };
    } else {
        obs = { x, y: -100, w: 50, h: 80, vx: 0, type: 'truck', color: `hsl(${Math.floor(Math.random()*360)},40%,40%)` };
    }
    obs.lane = lane;
    obstacles.push(obs);
}

function crash() {
    state = 'gameover';
    shakeTime = 0.5;
    shakeIntensity = 12;
    playSound('crash');
    for (let i = 0; i < 30; i++) {
        const angle = Math.random() * Math.PI * 2;
        const spd = 2 + Math.random() * 6;
        particles.push({ x: player.x, y: player.y, vx: Math.cos(angle) * spd, vy: Math.sin(angle) * spd - 2, life: 1, color: Math.random() < 0.5 ? '#ff4444' : '#ffaa00', size: 3 + Math.random() * 5 });
    }
}

function update(dt) {
    if (state !== 'playing') return;
    
    // Increase speed over time
    speed = Math.min(maxSpeed, speed + dt * 0.3);
    distance += speed * dt * 10;
    score = Math.floor(distance / 10);
    
    // Milestone
    if (score - lastMilestone >= 500) {
        lastMilestone = score;
        playSound('milestone');
    }
    
    // Player movement
    const accel = 800;
    const friction = 0.85;
    if (keys['ArrowLeft'] || keys['KeyA']) player.vx -= accel * dt;
    if (keys['ArrowRight'] || keys['KeyD']) player.vx += accel * dt;
    if (keys['ArrowUp'] || keys['KeyW']) player.y -= 300 * dt;
    if (keys['ArrowDown'] || keys['KeyS']) player.y += 300 * dt;
    player.vx *= friction;
    player.x += player.vx;
    player.y = Math.max(100, Math.min(H / scale - 50, player.y));
    
    // Clamp to road
    const halfW = player.w / 2;
    if (player.x - halfW < ROAD_LEFT) { player.x = ROAD_LEFT + halfW; player.vx = 0; }
    if (player.x + halfW > ROAD_RIGHT) { player.x = ROAD_RIGHT - halfW; player.vx = 0; }
    
    // Road scroll
    roadOffset += speed * dt * 60;
    
    // Trees
    trees.forEach(t => { t.y += speed * dt * 40; if (t.y > H / scale + 50) { t.y = -50; t.x = Math.random() < 0.5 ? 30 + Math.random() * 90 : 680 + Math.random() * 90; } });
    
    // Spawn obstacles
    const spawnRate = Math.max(0.3, 1.2 - speed * 0.04);
    if (Math.random() < dt / spawnRate) spawnObstacle();
    
    // Update obstacles
    for (let i = obstacles.length - 1; i >= 0; i--) {
        const o = obstacles[i];
        o.y += speed * dt * 50;
        if (o.type === 'car') o.x += Math.sin(o.y * 0.02) * 0.3;
        if (o.y > H / scale + 100) { obstacles.splice(i, 1); continue; }
        
        // Collision
        const dx = Math.abs(player.x - o.x);
        const dy = Math.abs(player.y - o.y);
        const cw = (player.w + o.w) / 2 * 0.8;
        const ch = (player.h + o.h) / 2 * 0.8;
        if (dx < cw && dy < ch) { crash(); return; }
    }
    
    // Particles
    for (let i = particles.length - 1; i >= 0; i--) {
        const p = particles[i];
        p.x += p.vx; p.y += p.vy;
        p.life -= dt * 2;
        if (p.life <= 0) particles.splice(i, 1);
    }
}

function drawCar(x, y, w, h, color, isPlayer) {
    ctx.save();
    ctx.translate(x, y);
    
    // Body
    const r = 8;
    ctx.fillStyle = color;
    ctx.beginPath();
    ctx.roundRect(-w/2, -h/2, w, h, r);
    ctx.fill();
    
    // Windshield
    ctx.fillStyle = isPlayer ? '#44ccff' : '#335566';
    ctx.fillRect(-w/2 + 5, -h/2 + 8, w - 10, 14);
    
    // Rear window
    ctx.fillStyle = isPlayer ? '#44ccff' : '#335566';
    ctx.fillRect(-w/2 + 5, h/2 - 20, w - 10, 12);
    
    // Wheels
    ctx.fillStyle = '#222';
    ctx.fillRect(-w/2 - 3, -h/2 + 8, 5, 14);
    ctx.fillRect(w/2 - 2, -h/2 + 8, 5, 14);
    ctx.fillRect(-w/2 - 3, h/2 - 22, 5, 14);
    ctx.fillRect(w/2 - 2, h/2 - 22, 5, 14);
    
    // Headlights (player)
    if (isPlayer) {
        ctx.fillStyle = '#ffff88';
        ctx.fillRect(-w/2 + 3, -h/2, 6, 4);
        ctx.fillRect(w/2 - 9, -h/2, 6, 4);
    } else {
        // Taillights for enemies (they face down)
        ctx.fillStyle = '#ff3333';
        ctx.fillRect(-w/2 + 3, h/2 - 4, 6, 4);
        ctx.fillRect(w/2 - 9, h/2 - 4, 6, 4);
    }
    
    ctx.restore();
}

function drawCone(x, y) {
    ctx.save();
    ctx.translate(x, y);
    ctx.fillStyle = '#ff6600';
    ctx.beginPath();
    ctx.moveTo(0, -12);
    ctx.lineTo(-8, 10);
    ctx.lineTo(8, 10);
    ctx.closePath();
    ctx.fill();
    ctx.fillStyle = '#fff';
    ctx.fillRect(-6, -2, 12, 4);
    ctx.restore();
}

function drawTruck(x, y, w, h, color) {
    ctx.save();
    ctx.translate(x, y);
    ctx.fillStyle = color;
    ctx.beginPath();
    ctx.roundRect(-w/2, -h/2, w, h, 6);
    ctx.fill();
    // Cab
    ctx.fillStyle = '#8899aa';
    ctx.fillRect(-w/2 + 4, -h/2 + 5, w - 8, 18);
    ctx.fillStyle = '#335566';
    ctx.fillRect(-w/2 + 8, -h/2 + 8, w - 16, 10);
    // Cargo area lines
    ctx.strokeStyle = 'rgba(255,255,255,0.2)';
    ctx.lineWidth = 1;
    for (let i = -h/2 + 30; i < h/2 - 10; i += 15) {
        ctx.beginPath(); ctx.moveTo(-w/2 + 4, i); ctx.lineTo(w/2 - 4, i); ctx.stroke();
    }
    // Taillights
    ctx.fillStyle = '#ff3333';
    ctx.fillRect(-w/2 + 3, h/2 - 5, 8, 5);
    ctx.fillRect(w/2 - 11, h/2 - 5, 8, 5);
    ctx.restore();
}

function drawTree(x, y) {
    ctx.fillStyle = '#2d5a1b';
    ctx.beginPath();
    ctx.arc(x, y, 14, 0, Math.PI * 2);
    ctx.fill();
    ctx.fillStyle = '#3d7a2b';
    ctx.beginPath();
    ctx.arc(x - 3, y - 3, 8, 0, Math.PI * 2);
    ctx.fill();
}

function render() {
    const sw = W;
    const sh = H;
    
    ctx.save();
    ctx.clearRect(0, 0, sw, sh);
    
    // Apply shake
    let sx = 0, sy = 0;
    if (shakeTime > 0) {
        sx = (Math.random() - 0.5) * shakeIntensity * shakeTime * 2;
        sy = (Math.random() - 0.5) * shakeIntensity * shakeTime * 2;
    }
    
    ctx.save();
    ctx.translate(sw/2 + sx, sh/2 + sy);
    ctx.scale(scale, scale);
    ctx.translate(-LOGICAL_W/2, -450);
    
    // Grass
    ctx.fillStyle = '#2a5e1a';
    ctx.fillRect(0, 0, LOGICAL_W, 900);
    
    // Trees
    trees.forEach(t => drawTree(t.x, t.y));
    
    // Road
    ctx.fillStyle = '#3a3a3a';
    ctx.fillRect(ROAD_LEFT, 0, ROAD_W, 900);
    
    // Road edges
    ctx.fillStyle = '#fff';
    ctx.fillRect(ROAD_LEFT - 4, 0, 4, 900);
    ctx.fillRect(ROAD_RIGHT, 0, 4, 900);
    
    // Lane markings
    ctx.strokeStyle = '#ccc';
    ctx.lineWidth = 3;
    ctx.setLineDash([30, 25]);
    const lineOffset = roadOffset % 55;
    for (let i = 1; i < LANE_COUNT; i++) {
        const lx = ROAD_LEFT + i * LANE_W;
        ctx.beginPath();
        ctx.moveTo(lx, -55 + lineOffset);
        ctx.lineTo(lx, 900 + lineOffset);
        ctx.stroke();
    }
    ctx.setLineDash([]);
    
    // Obstacles
    obstacles.forEach(o => {
        if (o.type === 'car') drawCar(o.x, o.y, o.w, o.h, o.color, false);
        else if (o.type === 'cone') drawCone(o.x, o.y);
        else if (o.type === 'truck') drawTruck(o.x, o.y, o.w, o.h, o.color);
    });
    
    // Player
    if (state === 'playing' || state === 'start') {
        const px = state === 'start' ? LOGICAL_W / 2 : player.x;
        const py = state === 'start' ? 750 : player.y;
        drawCar(px, py, 36, 64, '#2196f3', true);
    }
    
    // Particles
    particles.forEach(p => {
        ctx.globalAlpha = p.life;
        ctx.fillStyle = p.color;
        ctx.beginPath();
        ctx.arc(p.x, p.y, p.size * p.life, 0, Math.PI * 2);
        ctx.fill();
        ctx.globalAlpha = 1;
    });
    
    // HUD
    if (state === 'playing') {
        ctx.fillStyle = '#fff';
        ctx.font = 'bold 24px monospace';
        ctx.textAlign = 'left';
        ctx.fillText(`Score: ${score}`, 20, 35);
        ctx.fillText(`Speed: ${Math.floor(speed * 10)} km/h`, 20, 65);
    }
    
    // Start screen
    if (state === 'start') {
        ctx.fillStyle = 'rgba(0,0,0,0.7)';
        ctx.fillRect(0, 0, LOGICAL_W, 900);
        ctx.fillStyle = '#fff';
        ctx.font = 'bold 48px monospace';
        ctx.textAlign = 'center';
        ctx.fillText('ENDLESS HIGHWAY', LOGICAL_W/2, 300);
        ctx.font = '20px monospace';
        ctx.fillStyle = '#aaa';
        ctx.fillText('Dodge traffic. Survive as long as you can.', LOGICAL_W/2, 370);
        ctx.fillStyle = '#fff';
        ctx.fillText('[← →] or [A D] — Move', LOGICAL_W/2, 430);
        ctx.fillText('[↑ ↓] or [W S] — Nudge forward/back', LOGICAL_W/2, 465);
        ctx.fillStyle = '#ffcc00';
        ctx.font = 'bold 24px monospace';
        const blink = Math.sin(Date.now() * 0.005) > 0;
        if (blink) ctx.fillText('Press SPACE to Start', LOGICAL_W/2, 550);
    }
    
    // Game over
    if (state === 'gameover') {
        ctx.fillStyle = 'rgba(0,0,0,0.6)';
        ctx.fillRect(0, 0, LOGICAL_W, 900);
        ctx.fillStyle = '#ff4444';
        ctx.font = 'bold 52px monospace';
        ctx.textAlign = 'center';
        ctx.fillText('CRASHED!', LOGICAL_W/2, 350);
        ctx.fillStyle = '#fff';
        ctx.font = '28px monospace';
        ctx.fillText(`Final Score: ${score}`, LOGICAL_W/2, 420);
        ctx.fillStyle = '#aaa';
        ctx.font = '18px monospace';
        let best = 0;
        try { best = parseInt(localStorage.getItem('eh_best') || '0'); } catch(e) {}
        if (score > best) {
            try { localStorage.setItem('eh_best', score); } catch(e) {}
            ctx.fillStyle = '#ffcc00';
            ctx.fillText('★ NEW HIGH SCORE! ★', LOGICAL_W/2, 470);
        } else {
            ctx.fillText(`Best: ${best}`, LOGICAL_W/2, 470);
        }
        ctx.fillStyle = '#fff';
        ctx.font = '20px monospace';
        const blink2 = Math.sin(Date.now() * 0.005) > 0;
        if (blink2) ctx.fillText('Press SPACE or R to Restart', LOGICAL_W/2, 540);
    }
    
    ctx.restore();
    ctx.restore();
}

let lastTime = 0;
function loop(time) {
    const dt = Math.min((time - lastTime) / 1000, 0.05);
    lastTime = time;
    
    if (shakeTime > 0) shakeTime -= dt;
    
    update(dt);
    render();
    
    requestAnimationFrame(loop);
}
requestAnimationFrame(loop);
</script>
</body>
```
