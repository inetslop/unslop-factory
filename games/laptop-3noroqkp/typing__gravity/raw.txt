<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">

    <title>Gravity Typing</title>
    <style>

<script>
const canvas = document.getElementById('game');
const ctx = canvas.getContext('2d');

let W, H;
function resize() {
    W = window.innerWidth;
    H = window.innerHeight;
    canvas.width = W;
    canvas.height = H;
}
window.addEventListener('resize', resize);
resize();

// Audio
let audioCtx = null;
function initAudio() {
    if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
}
function playTone(freq, dur, type, vol) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = type || 'sine';
    osc.frequency.value = freq;
    gain.gain.setValueAtTime(vol || 0.1, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + dur);
}

// Game state
let gameState = 'start'; // start, playing, gameover
let score = 0;
let highScore = 0;
let lives = 3;
let words = [];
let particles = [];
let shakeTime = 0;
let shakeMag = 0;
let spawnTimer = 0;
let difficulty = 1;
let timeAlive = 0;
let typedFlash = 0;
let comboCount = 0;

try {
    highScore = parseInt(localStorage.getItem('gravityTypingHigh') || '0') || 0;
} catch(e) {}

const WORDS_LIST = [
    "GRAVITY","TYPING","QUICK","SMART","BRIGHT","SWIFT","MAGNET","ORBIT",
    "NEBULA","COSMIC","PLASMA","FUSION","GALAXY","METEOR","COMET","ASTRO",
    "LUNAR","SOLAR","PHOTON","QUANTUM","MATRIX","VECTOR","CIPHER","ENCODE",
    "DATA","STREAM","PULSE","WAVE","FLUX","TENSOR","NEXUS","VERTEX",
    "PRISM","ECHO","GHOST","SHADOW","EMBER","SPARK","THUNDER","LIGHTNING"
];

function randWord() {
    return WORDS_LIST[Math.floor(Math.random() * WORDS_LIST.length)];
}

function spawnWord() {
    const word = randWord();
    const fontSize = Math.max(24, Math.min(36, W / 20));
    const maxW = W * 0.7;
    let size = fontSize;
    while (word.length * size * 0.6 > maxW && size > 18) size -= 2;
    
    const x = W * 0.2 + Math.random() * W * 0.6 - word.length * size * 0.3;
    words.push({
        text: word,
        x: Math.max(10, x),
        y: -size,
        vy: 0,
        size: size,
        progress: 0,
        life: 1,
        hue: Math.floor(Math.random() * 360)
    });
}

function spawnParticles(x, y, count, hue) {
    for (let i = 0; i < count; i++) {
        const angle = Math.random() * Math.PI * 2;
        const speed = 1 + Math.random() * 4;
        particles.push({
            x, y,
            vx: Math.cos(angle) * speed,
            vy: Math.sin(angle) * speed - 2,
            life: 1,
            size: 2 + Math.random() * 3,
            hue
        });
    }
}

function resetGame() {
    words = [];
    particles = [];
    score = 0;
    lives = 3;
    difficulty = 1;
    timeAlive = 0;
    spawnTimer = 0;
    comboCount = 0;
    gameState = 'playing';
}

function gameOver() {
    gameState = 'gameover';
    shakeTime = 20;
    shakeMag = 8;
    try {
        if (score > highScore) {
            highScore = score;
            localStorage.setItem('gravityTypingHigh', highScore.toString());
        }
    } catch(e) {}
}

// Input
let typedChar = '';
window.addEventListener('keydown', (e) => {
    initAudio();
    
    if (gameState === 'start') {
        resetGame();
        return;
    }
    if (gameState === 'gameover' && e.key === ' ') {
        resetGame();
        return;
    }
    if (gameState !== 'playing') return;
    
    const key = e.key.toUpperCase();
    if (key.length === 1) {
        typedFlash = 5;
        handleTyping(key);
    }
});

function handleTyping(key) {
    // Find the lowest word with progress that matches
    let bestWord = null;
    let bestY = -Infinity;
    
    for (const w of words) {
        if (w.progress < w.text.length && w.y > bestY) {
            if (w.text[w.progress] === key) {
                bestWord = w;
                bestY = w.y;
            }
        }
    }
    
    if (bestWord) {
        bestWord.progress++;
        comboCount++;
        const pts = 10 * Math.floor(difficulty / 2 + 1);
        score += pts;
        
        playTone(400 + bestWord.progress * 50, 0.1, 'sine', 0.08);
        
        // Particle burst at the progress position
        const px = bestWord.x + bestWord.progress * bestWord.size * 0.6;
        const py = bestWord.y + bestWord.size / 2;
        spawnParticles(px, py, 5, bestWord.hue);
        
        if (bestWord.progress >= bestWord.text.length) {
            // Word complete!
            score += 50 * Math.floor(difficulty / 2 + 1);
            playTone(800, 0.2, 'sine', 0.15);
            playTone(1200, 0.3, 'sine', 0.1);
            spawnParticles(bestWord.x + bestWord.text.length * bestWord.size * 0.3, bestWord.y, 20, bestWord.hue);
            comboCount = 0;
            words.splice(words.indexOf(bestWord), 1);
        }
    } else {
        // Wrong key
        typedFlash = 8;
        playTone(150, 0.1, 'square', 0.05);
    }
}

// Update
function update(dt) {
    if (gameState !== 'playing') return;
    
    timeAlive += dt;
    difficulty = 1 + Math.floor(timeAlive / 15) * 0.5;
    
    // Spawn words
    const spawnInterval = Math.max(1.2, 3 - timeAlive * 0.03);
    spawnTimer += dt;
    if (spawnTimer >= spawnInterval) {
        spawnTimer = 0;
        spawnWord();
    }
    
    // Update words with gravity
    const gravity = 80 + difficulty * 20;
    for (let i = words.length - 1; i >= 0; i--) {
        const w = words[i];
        w.vy += gravity * dt;
        w.y += w.vy * dt;
        
        // Check if word reached bottom
        if (w.y + w.size > H - 20) {
            lives--;
            shakeTime = 10;
            shakeMag = 5;
            playTone(100, 0.3, 'sawtooth', 0.1);
            spawnParticles(w.x + w.text.length * w.size * 0.3, H - 20, 15, 0);
            words.splice(i, 1);
            
            if (lives <= 0) {
                gameOver();
            }
        }
    }
    
    // Update particles
    for (let i = particles.length - 1; i >= 0; i--) {
        const p = particles[i];
        p.x += p.vx;
        p.y += p.vy;
        p.vy += 0.15;
        p.life -= dt * 2;
        if (p.life <= 0) particles.splice(i, 1);
    }
    
    // Shake decay
    if (shakeTime > 0) shakeTime--;
}

// Draw
function draw() {
    ctx.save();
    
    // Screen shake
    if (shakeTime > 0) {
        const sx = (Math.random() - 0.5) * shakeMag;
        const sy = (Math.random() - 0.5) * shakeMag;
        ctx.translate(sx, sy);
    }
    
    // Background
    ctx.fillStyle = '#0a0a1a';
    ctx.fillRect(-10, -10, W + 20, H + 20);
    
    // Grid lines (subtle)
    ctx.strokeStyle = 'rgba(50, 60, 120, 0.15)';
    ctx.lineWidth = 1;
    for (let y = 0; y < H; y += 40) {
        ctx.beginPath();
        ctx.moveTo(0, y);
        ctx.lineTo(W, y);
        ctx.stroke();
    }
    
    if (gameState === 'start') {
        drawStartScreen();
    } else if (gameState === 'playing') {
        drawGame();
    } else if (gameState === 'gameover') {
        drawGameOver();
    }
    
    ctx.restore();
}

function drawStartScreen() {
    // Title
    const pulse = 0.8 + Math.sin(Date.now() * 0.003) * 0.2;
    ctx.font = `bold ${Math.min(64, W/12)}px Courier New`;
    ctx.textAlign = 'center';
    ctx.fillStyle = `rgba(100, 200, 255, ${pulse})`;
    ctx.shadowColor = '#4af';
    ctx.shadowBlur = 20;
    ctx.fillText('GRAVITY TYPING', W/2, H * 0.3);
    ctx.shadowBlur = 0;
    
    // Subtitle
    ctx.font = `${Math.min(20, W/30)}px Courier New`;
    ctx.fillStyle = 'rgba(180, 200, 255, 0.8)';
    ctx.fillText('Words fall with gravity. Type them before they hit the ground!', W/2, H * 0.42);
    
    // Controls
    ctx.font = `${Math.min(16, W/35)}px Courier New`;
    ctx.fillStyle = 'rgba(150, 170, 220, 0.7)';
    ctx.fillText('TYPE the falling words letter by letter', W/2, H * 0.55);
    ctx.fillText('Complete a word to score points', W/2, H * 0.60);
    ctx.fillText('3 lives — don\'t let words hit the floor!', W/2, H * 0.65);
    
    // High score
    if (highScore > 0) {
        ctx.fillStyle = 'rgba(255, 220, 100, 0.8)';
        ctx.fillText(`HIGH SCORE: ${highScore}`, W/2, H * 0.74);
    }
    
    // Start prompt
    const blink = Math.sin(Date.now() * 0.005) > 0;
    if (blink) {
        ctx.font = `bold ${Math.min(24, W/25)}px Courier New`;
        ctx.fillStyle = '#fff';
        ctx.fillText('PRESS ANY KEY TO START', W/2, H * 0.85);
    }
}

function drawGame() {
    // Bottom danger line
    const dangerY = H - 20;
    ctx.strokeStyle = 'rgba(255, 50, 50, 0.6)';
    ctx.lineWidth = 3;
    ctx.setLineDash([10, 5]);
    ctx.beginPath();
    ctx.moveTo(0, dangerY);
    ctx.lineTo(W, dangerY);
    ctx.stroke();
    ctx.setLineDash([]);
    
    // Words
    for (const w of words) {
        const charW = w.size * 0.6;
        
        for (let i = 0; i < w.text.length; i++) {
            const cx = w.x + i * charW;
            const cy = w.y + w.size / 2;
            
            let color, glow;
            if (i < w.progress) {
                // Already typed - green/glowing
                color = `hsl(${w.hue}, 80%, 60%)`;
                glow = 15;
            } else if (i === w.progress) {
                // Next to type - bright white
                color = '#fff';
                glow = 20;
            } else {
                // Untyped - dimmer
                color = `hsl(${w.hue}, 40%, 50%)`;
                glow = 5;
            }
            
            ctx.font = `bold ${w.size}px Courier New`;
            ctx.textAlign = 'center';
            ctx.fillStyle = color;
            ctx.shadowColor = color;
            ctx.shadowBlur = glow;
            ctx.fillText(w.text[i], cx + charW/2, cy + w.size/3);
        }
        ctx.shadowBlur = 0;
        
        // Progress bar under word
        const barW = w.text.length * charW;
        const barH = 4;
        const barX = w.x;
        const barY = w.y + w.size + 5;
        ctx.fillStyle = 'rgba(255,255,255,0.2)';
        ctx.fillRect(barX, barY, barW, barH);
        ctx.fillStyle = `hsl(${w.hue}, 80%, 60%)`;
        ctx.fillRect(barX, barY, barW * (w.progress / w.text.length), barH);
    }
    
    // Particles
    for (const p of particles) {
        ctx.globalAlpha = p.life;
        ctx.fillStyle = `hsl(${p.hue}, 80%, 70%)`;
        ctx.shadowColor = `hsl(${p.hue}, 80%, 70%)`;
        ctx.shadowBlur = 5;
        ctx.fillRect(p.x - p.size/2, p.y - p.size/2, p.size * p.life, p.size * p.life);
    }
    ctx.globalAlpha = 1;
    ctx.shadowBlur = 0;
    
    // HUD
    ctx.font = `bold ${Math.min(24, W/25)}px Courier New`;
    ctx.textAlign = 'left';
    ctx.fillStyle = '#4af';
    ctx.fillText(`SCORE: ${score}`, 20, 35);
    
    // Lives
    for (let i = 0; i < lives; i++) {
        ctx.fillStyle = '#f44';
        ctx.shadowColor = '#f44';
        ctx.shadowBlur = 8;
        ctx.font = '20px Courier New';
        ctx.fillText('♥', W - 80 + i * 30, 35);
    }
    ctx.shadowBlur = 0;
    
    // Combo
    if (comboCount > 3) {
        ctx.textAlign = 'center';
        ctx.fillStyle = '#ff0';
        ctx.font = `bold ${Math.min(18, W/30)}px Courier New`;
        ctx.fillText(`COMBO x${comboCount}`, W/2, 35);
    }
    
    // Difficulty indicator
    ctx.textAlign = 'right';
    ctx.fillStyle = 'rgba(255,255,255,0.5)';
    ctx.font = '14px Courier New';
    ctx.fillText(`SPEED: ${Math.floor(difficulty * 10)}%`, W - 20, H - 35);
}

function drawGameOver() {
    // Darken
    ctx.fillStyle = 'rgba(10, 10, 26, 0.8)';
    ctx.fillRect(0, 0, W, H);
    
    ctx.textAlign = 'center';
    ctx.font = `bold ${Math.min(48, W/10)}px Courier New`;
    ctx.fillStyle = '#f44';
    ctx.shadowColor = '#f44';
    ctx.shadowBlur = 20;
    ctx.fillText('GAME OVER', W/2, H * 0.3);
    ctx.shadowBlur = 0;
    
    ctx.font = `${Math.min(28, W/20)}px Courier New`;
    ctx.fillStyle = '#fff';
    ctx.fillText(`SCORE: ${score}`, W/2, H * 0.45);
    
    if (score >= highScore && score > 0) {
        ctx.fillStyle = '#ff0';
        ctx.font = `bold ${Math.min(22, W/25)}px Courier New`;
        ctx.fillText('NEW HIGH SCORE!', W/2, H * 0.53);
    } else {
        ctx.fillStyle = 'rgba(255,220,100,0.7)';
        ctx.font = `${Math.min(18, W/30)}px Courier New`;
        ctx.fillText(`BEST: ${highScore}`, W/2, H * 0.53);
    }
    
    const blink = Math.sin(Date.now() * 0.005) > 0;
    if (blink) {
        ctx.font = `bold ${Math.min(20, W/28)}px Courier New`;
        ctx.fillStyle = '#4af';
        ctx.fillText('PRESS SPACE TO RESTART', W/2, H * 0.7);
    }
}

// Game loop
let lastTime = 0;
function gameLoop(timestamp) {
    const dt = Math.min((timestamp - lastTime) / 1000, 0.05);
    lastTime = timestamp;
    
    update(dt);
    draw();
    
    requestAnimationFrame(gameLoop);
}

requestAnimationFrame(gameLoop);
</script>
