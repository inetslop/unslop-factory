function playSound(freq, dur, type, vol) {
  if (!audioCtx) return;
  const osc = audioCtx.createOscillator();
  const gain = audioCtx.createGain();
  osc.type = type || 'square';
  osc.frequency.value = freq;
  gain.gain.setValueAtTime(vol || 0.1, audioCtx.currentTime);
  gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
  osc.connect(gain);
  gain.connect(audioCtx.destination);
  osc.start();
  osc.stop(audioCtx.currentTime + dur);
}

function getRotated(shape, rot) {
  const rows = shape.length;
  const cols = shape[0].length;
  let r = shape.map(r => [...r]);
  for (let i = 0; i < rot % 4; i++) {
    // Rotate clockwise
    let nr = [];
    for (let c = 0; c < r[0].length; c++) {
      let col = [];
      for (let rr = r.length - 1; rr >= 0; rr--) col.push(r[rr][c]);
      nr.push(col);
    }
    r = nr;
  }
  return r;
}

function spawnPiece() {
  const shapeId = current ? next : nextFromBag();

function collides(shape, px, py) {
  for (let r = 0; r < shape.length; r++) {
    for (let c = 0; c < shape[r].length; c++) {
      if (shape[r][c]) {
        let nx = px + c, ny = py + r;
        if (nx < 0 || nx >= COLS || ny >= ROWS) return true;
        if (ny >= 0 && board[ny][nx]) return true;
      }
    }
  }
  return false;
}

function lockPiece() {
  let shape = getRotated(SHAPES[current.shape], current.rot);

    for (let c = 0; c < shape[r].length; c++) {
      if (shape[r][c]) {
        let ny = current.y + r, nx = current.x + c;
        if (ny >= 0) board[ny][nx] = current.shape;
      }

  // Check line clears

function move(dx) {
  if (!current || gameOver || !gameRunning) return;
  let shape = getRotated(SHAPES[current.shape], current.rot);
  if (!collides(shape, current.x + dx, current.y)) {
    current.x += dx;
    sMove();
  }
}

function rotate() {
  if (!current || gameOver || !gameRunning) return;
  let newRot = (current.rot + 1) % 4;
  let shape = getRotated(SHAPES[current.shape], newRot);
  // Simple wall kicks
  let kicks = [0, -1, 1, -2, 2];
  for (let k of kicks) {
    if (!collides(shape, current.x + k, current.y)) {
      current.rot = newRot;
      current.x += k;
      sRotate();
      return;
    }
    // Try up
    if (!collides(shape, current.x + k, current.y - 1)) {
      current.rot = newRot;
      current.x += k;
      current.y -= 1;
      sRotate();
      return;
    }
  }
}

function hardDrop() {
  if (!current || gameOver || !gameRunning) return;
  let shape = getRotated(SHAPES[current.shape], current.rot);
  while (!collides(shape, current.x, current.y + 1)) {
    current.y++;
  }
  score += 2 * (ROWS - current.y);
  sDrop();
  lockPiece();
}

function getGhostY() {
  if (!current) return ROWS;
  let shape = getRotated(SHAPES[current.shape], current.rot);
  let y = current.y;
  while (!collides(shape, current.x, y + 1)) y++;
  return y;
}

function drawAsteroidBlock(x, y, size, colorIdx, alpha) {
  let c = COLORS[colorIdx];
  let a = alpha || 1;
  // Base rock shape - slightly irregular
  let pad = size * 0.06;
  ctx.globalAlpha = a;
  ctx.fillStyle = c;
  roundRect(x+pad, y+pad, size-pad*2, size-pad*2, size*0.15);
  ctx.fill();

  // Darker border
  ctx.strokeStyle = darken(c, 0.4);
  ctx.lineWidth = 1.5;
  roundRect(x+pad, y+pad, size-pad*2, size-pad*2, size*0.15);
  ctx.stroke();

  // Craters
  let seed = colorIdx * 7 + 3;
  for (let i = 0; i < 3; i++) {
    let sx = x + pad + (hash(seed+i*13) % 100)/100 * (size - pad*2 - size*0.3);
    let sy = y + pad + (hash(seed+i*29+7) % 100)/100 * (size - pad*2 - size*0.3);
    let sr = size * (0.08 + hash(seed+i*41) % 50 / 500);
    ctx.fillStyle = darken(c, 0.3);
    ctx.beginPath();
    ctx.arc(sx + size*0.15, sy + size*0.15, sr, 0, Math.PI*2);
    ctx.fill();
  }

  // Highlight
  ctx.fillStyle = lighten(c, 0.2);
  ctx.globalAlpha = a * 0.3;
  ctx.beginPath();
  ctx.arc(x + size*0.35, y + size*0.35, size*0.15, 0, Math.PI*2);
  ctx.fill();

  ctx.globalAlpha = 1;
}

function roundRect(x, y, w, h, r) {
  ctx.beginPath();
  ctx.moveTo(x+r, y);
  ctx.lineTo(x+w-r, y);
  ctx.quadraticCurveTo(x+w, y, x+w, y+r);
  ctx.lineTo(x+w, y+h-r);
  ctx.quadraticCurveTo(x+w, y+h, x+w-r, y+h);
  ctx.lineTo(x+r, y+h);
  ctx.quadraticCurveTo(x, y+h, x, y+h-r);
  ctx.lineTo(x, y+r);
  ctx.quadraticCurveTo(x, y, x+r, y);
  ctx.closePath();
}

function hash(n) { return (n * 16807 + 12345) % 2147483647; }

function darken(hex, amt) {
  let r = parseInt(hex.slice(1,3),16), g = parseInt(hex.slice(3,5),16), b = parseInt(hex.slice(5,7),16);
  r = Math.floor(r * (1-amt)); g = Math.floor(g*(1-amt)); b = Math.floor(b*(1-amt));
  return `rgb(${r},${g},${b})`;
}
function lighten(hex, amt) {
  let r = parseInt(hex.slice(1,3),16), g = parseInt(hex.slice(3,5),16), b = parseInt(hex.slice(5,7),16);
  r = Math.min(255, Math.floor(r + (255-r)*amt));
  g = Math.min(255, Math.floor(g + (255-g)*amt));
  b = Math.min(255, Math.floor(b + (255-b)*amt));
  return `rgb(${r},${g},${b})`;
}

      ctx.fillStyle = `rgba(255,255,200,${0.3 + Math.sin(clearTimer/50)*0.3})`;

        if (shape[r][c]) {
          let gx = boardX + (current.x + c) * cellSize;

  }

function update(dt) {
  if (!gameRunning || gameOver) return;
  if (clearing) {
    clearTimer -= dt;
    if (clearTimer <= 0) processClear();
    return;
  }
  accumulator += dt;
  while (accumulator >= dropInterval) {
    accumulator -= dropInterval;
    let shape = getRotated(SHAPES[current.shape], current.rot);
    if (!collides(shape, current.x, current.y + 1)) {
      current.y++;
    } else {
      lockPiece();
    }
  }
  // Update particles
  for (let p of particles) {
    p.x += p.vx;
    p.y += p.vy;
    p.vy += 0.15;
    p.life -= 0.02;
  }
  particles = particles.filter(p => p.life > 0);
}

function render() {
  drawBackground();
  calcLayout();
  if (!started) {
    drawStartScreen();
  } else {
    drawBoard();
    drawInfo();
    if (gameOver) drawGameOver();
  }
}

let lastFrame = 0;
function gameLoop(timestamp) {
  let dt = timestamp - lastFrame;
  lastFrame = timestamp;
  if (dt > 100) dt = 16;
  update(dt);
  render();
  requestAnimationFrame(gameLoop);
}

// Input
window.addEventListener('keydown', function(e) {
  initAudio();
  if (e.key === 'Enter' || e.key === ' ') {
    if (!started) {
      started = true;
      gameRunning = true;
      resetGame();
      calcLayout();
      e.preventDefault();
      return;
    }
    if (gameOver) {
      gameRunning = true;
      resetGame();
      calcLayout();
      e.preventDefault();
      return;
    }
  }

  if (!started || gameOver) return;

  if (e.key === 'ArrowLeft') { move(-1); e.preventDefault(); }
  else if (e.key === 'ArrowRight') { move(1); e.preventDefault(); }
  else if (e.key === 'ArrowDown') {
    let shape = getRotated(SHAPES[current.shape], current.rot);
    if (!collides(shape, current.x, current.y + 1)) {
      current.y++;
      score += 1;
      sMove();
    }
    e.preventDefault();
  }
  else if (e.key === 'ArrowUp' || e.key === 'z' || e.key === 'Z') { rotate(); e.preventDefault(); }
  else if (e.key === 'x' || e.key === 'X') {
    // CCW rotation
    let newRot = (current.rot + 3) % 4;
    let shape = getRotated(SHAPES[current.shape], newRot);
    let kicks = [0, -1, 1, -2, 2];
    for (let k of kicks) {
      if (!collides(shape, current.x + k, current.y)) {
        current.rot = newRot;
        current.x += k;
        sRotate();
        break;
      }
      if (!collides(shape, current.x + k, current.y - 1)) {
        current.rot = newRot;
        current.x += k;
        current.y -= 1;
        sRotate();
        break;
      }
    }
    e.preventDefault();
  }
  else if (e.key === ' ') { hardDrop(); e.preventDefault(); }
});

  else if (gameOver) { gameRunning = true; resetGame(); calcLayout(); }
});

// Start
