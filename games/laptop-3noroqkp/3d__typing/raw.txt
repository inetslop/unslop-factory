function initAudio(){

function playTypeSound(correct){
if(!audioCtx)return;
const osc=audioCtx.createOscillator();
const gain=audioCtx.createGain();
osc.connect(gain);gain.connect(audioCtx.destination);
osc.type='sine';
osc.frequency.value=correct?600+Math.random()*200:150;
gain.gain.setValueAtTime(0.1,audioCtx.currentTime);
gain.gain.exponentialRampToValueAtTime(0.001,audioCtx.currentTime+0.1);
osc.start();osc.stop(audioCtx.currentTime+0.1);
}

function playDestroySound(){
if(!audioCtx)return;
const osc=audioCtx.createOscillator();
const gain=audioCtx.createGain();
osc.connect(gain);gain.connect(audioCtx.destination);
osc.type='square';
osc.frequency.setValueAtTime(400,audioCtx.currentTime);
osc.frequency.exponentialRampToValueAtTime(800,audioCtx.currentTime+0.15);
gain.gain.setValueAtTime(0.15,audioCtx.currentTime);
gain.gain.exponentialRampToValueAtTime(0.001,audioCtx.currentTime+0.2);
osc.start();osc.stop(audioCtx.currentTime+0.2);
}

function playHitSound(){
if(!audioCtx)return;
const osc=audioCtx.createOscillator();
const gain=audioCtx.createGain();
osc.connect(gain);gain.connect(audioCtx.destination);
osc.type='sawtooth';
osc.frequency.setValueAtTime(200,audioCtx.currentTime);
osc.frequency.exponentialRampToValueAtTime(50,audioCtx.currentTime+0.3);
gain.gain.setValueAtTime(0.2,audioCtx.currentTime);
gain.gain.exponentialRampToValueAtTime(0.001,audioCtx.currentTime+0.3);
osc.start();osc.stop(audioCtx.currentTime+0.3);
}

function getWordForLevel(lvl){

if(lvl<=2)return w.length<=6;
if(lvl<=4)return w.length<=8;
return true;
});
return pool[Math.floor(Math.random()*pool.length)];
}

function spawnWord(){

scene.appendChild(el);

}

if(w&&w.alive){
w.el.classList.add('active');

function destroyWord(w, byPlayer){
if(!w.alive)return;
w.alive=false;
w.destroying=true;
const el=w.el;
el.style.transition='transform 0.3s, opacity 0.3s';
el.style.opacity='0';

score+=w.text.length*10*(combo+1);

playDestroySound();

function spawnParticles(w){
const rect=scene.getBoundingClientRect();
for(let i=0;i<8;i++){
const p=document.createElement('div');
p.className='particle';
p.textContent=['*','+','!','#'][Math.floor(Math.random()*4)];

const speed=50+Math.random()*100;
scene.appendChild(p);
let px=0,pz=-900,py=0;
let vx=Math.cos(angle)*speed,vz=0,vy=Math.sin(angle)*speed*0.5;
let life=600;

px+=vx*0.016;pz+=vz*0.016;py+=vy*0.016;

}

}

function getNearestUnfinishedWord(){
let best=null,bestZ=-99999;
for(const w of words){
if(w.alive&&!w.destroying&&w.typedCount<w.text.length){
if(w.z>bestZ){bestZ=w.z;best=w;}
}
}
return best;
}

function handleKey(e){
if(state!=='playing')return;
initAudio();
const key=e.key;
if(key==='Backspace'){
typedChars=typedChars.slice(0,-1);
if(activeWord&&activeWord.typedCount>0){
activeWord.typedCount--;
updateWordDisplay(activeWord);
}
updateInputBar();
return;
}
if(key.length!==1)return;
if(!activeWord){
setWordActive(getNearestUnfinishedWord());
if(!activeWord)return;
}
const expected=activeWord.text[activeWord.typedCount];
const pressed=key.toUpperCase();
if(pressed===expected){
activeWord.typedCount++;
typedChars+=key;
playTypeSound(true);
updateWordDisplay(activeWord);
if(activeWord.typedCount>=activeWord.text.length){
destroyWord(activeWord,true);
}
}else{
playTypeSound(false);

},300);
typedChars='';
updateInputBar();
}
}

spawnWord();
}

w.el.style.color= w===activeWord ? '#ffcc00' : (w.z>-400?'#ff6644':'#4488ff');

destroyWord(w,false);
}
}

setWordActive(getNearestUnfinishedWord());
}

words=[];
}
}
