<script>
const canvas = document.getElementById('c');
const ctx = canvas.getContext('2d');

let W, H;
function resize() {
    W = window.innerWidth;
    H = window.innerHeight;
    canvas.width = W;
    canvas.height = H;
}
resize();
window.addEventListener('resize', resize);

// Audio
let audioCtx = null;
function initAudio() {
    if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
}
function playSound(freq, dur, type, vol) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = type || 'square';
    osc.frequency.value = freq;
    gain.gain.value = vol || 0.1;
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + dur);
}

// Game state
const STATE = { MENU: 0, PLAYING: 1, GAMEOVER: 2 };
let state = STATE.MENU;
let scoreLeft = 0, scoreRight = 0;
const WIN_SCORE = 7;

// Paddles
let leftPaddle = { y: 0, h: 80, speed: 5 };
let rightPaddle = { y: 0, h: 80, speed: 4.5 };
const paddleW = 12;
const paddleOffset = 30;

// Ball
let ball = { x: 0, y: 0, vx: 0, vy: 0, r: 8, speed: 6 };

// Typing
let currentWord = '';
let typedChars = [];
let wordIndex = 0;
let wordCompleteTimer = 0;
let powerBoost = 0;

const words = [
    'THE', 'AND', 'FOR', 'TOP', 'CUP', 'MAP', 'JOT',
    'GAME', 'WORD', 'PING', 'PONG', 'FAST', 'TYPE',
    'BOLT', 'GLOW', 'NEON', 'FLUX', 'GRID', 'BYTE',
    'CODE', 'DATA', 'LINK', 'NODE', 'SYNC', 'TAP',
    'PIXL', 'RAIL', 'SWIFT', 'QUANTUM', 'CIPHER',
    'VECTOR', 'SIGNAL', 'PROXY', 'CACHE', 'DEBUG',
    'KERNEL', 'MODULE', 'PACKET', 'RENDER', 'SPRITE',
    'ALGORITHM', 'FUNCTION', 'INTERFACE',
    'COMPILATION', 'REFERENCE', 'SIMULATION',
    'TRANSMISSION', 'ARCHITECTURE'
];

function getWord() {
    if (wordIndex < words.length) return words[wordIndex++];
    // Random longer word for endless difficulty
    const prefixes = ['NEURO', 'CYBER', 'QUANTUM', 'HYPER', 'ULTRA', 'MEGA', 'OMNI'];
    const suffixes = ['BYTE', 'FLUX', 'GRID', 'NODE', 'PULSE', 'SHIFT', 'STREAM', 'LINK', 'CORE'];
    let w = prefixes[Math.floor(Math.random()*prefixes.length)] + suffixes[Math.floor(Math.random()*suffixes.length)];
    return w;
}

function nextWord() {
    currentWord = getWord();
    typedChars = [];
}

function resetBall(direction) {
    ball.x = W / 2;
    ball.y = H / 2;
    const angle = (Math.random() - 0.5) * Math.PI / 3;
    ball.vx = direction * Math.cos(angle);
    ball.vy = Math.sin(angle);
    const spd = 5 + scoreLeft + scoreRight;
    ball.speed = Math.min(spd, 12);
}

function resetGame() {
    scoreLeft = 0;
    scoreRight = 0;
    wordIndex = 0;
    powerBoost = 0;
    leftPaddle.y = H / 2 - leftPaddle.h / 2;
    rightPaddle.y = H / 2 - rightPaddle.h / 2;
    nextWord();
    resetBall(1);
}

// Particles
let particles = [];
function spawnParticles(x, y, color, count) {
    for (let i = 0; i < count; i++) {
        const angle = Math.random() * Math.PI * 2;
        const spd = 1 + Math.random() * 4;
        particles.push({ x, y, vx: Math.cos(angle)*spd, vy: Math.sin(angle)*spd, life: 30 + Math.random()*20, color });
    }
}

// Input
let keysDown = {};
window.addEventListener('keydown', (e) => {
    if (state === STATE.MENU) {
        initAudio();
        state = STATE.PLAYING;
        resetGame();
        playSound(440, 0.15, 'sine', 0.15);
        return;
    }
    if (state === STATE.GAMEOVER) {
        state = STATE.MENU;
        return;
    }
    if (state !== STATE.PLAYING) return;
    if (e.key.length === 1 && e.key !== ' ') {
        handleTyping(e.key);
    }
});

function handleTyping(key) {
    const expected = currentWord[typedChars.length].toUpperCase();
    const pressed = key.toUpperCase();
    if (pressed === expected) {
        typedChars.push(pressed);
        playSound(600 + typedChars.length * 50, 0.08, 'sine', 0.1);
        // Move paddle up
        leftPaddle.y -= 3;
        if (typedChars.length === currentWord.length) {
            // Word complete!
            powerBoost = 60;
            scoreLeft++;
            playSound(880, 0.2, 'sine', 0.2);
            playSound(1100, 0.3, 'sine', 0.15);
            spawnParticles(leftPaddle.x + paddleW, leftPaddle.y + leftPaddle.h/2, '#4fff4f', 15);
            nextWord();
            if (scoreLeft >= WIN_SCORE) {
                state = STATE.GAMEOVER;
                playSound(440, 0.5, 'sine', 0.2);
            }
        }
    } else {
        // Wrong key - small penalty
        leftPaddle.y += 1;
        playSound(150, 0.1, 'square', 0.08);
    }
}

// Update
let lastTime = 0;
function update(dt) {
    if (state !== STATE.PLAYING) return;

    // Power boost decay
    if (powerBoost > 0) powerBoost--;

    // Left paddle bounds
    leftPaddle.y = Math.max(0, Math.min(H - leftPaddle.h, leftPaddle.y));

    // Right paddle AI
    const targetY = ball.y;
    const diff = targetY - (rightPaddle.y + rightPaddle.h / 2);
    const aiSpeed = rightPaddle.speed + scoreRight * 0.3;
    if (Math.abs(diff) > 5) {
        rightPaddle.y += Math.sign(diff) * aiSpeed;
    }
    rightPaddle.y = Math.max(0, Math.min(H - rightPaddle.h, rightPaddle.y));

    // Ball move
    ball.x += ball.vx * (ball.speed / 5);
    ball.y += ball.vy * (ball.speed / 5);

    // Ball bounce top/bottom
    if (ball.y - ball.r < 0) { ball.y = ball.r; ball.vy *= -1; playSound(300, 0.05, 'square', 0.08); }
    if (ball.y + ball.r > H) { ball.y = H - ball.r; ball.vy *= -1; playSound(300, 0.05, 'square', 0.08); }

    // Left paddle collision
    const lx = paddleOffset + paddleW;
    if (ball.x - ball.r < lx && ball.x - ball.r > paddleOffset - 10 &&
        ball.y > leftPaddle.y && ball.y < leftPaddle.y + leftPaddle.h && ball.vx < 0) {
        ball.x = lx + ball.r;
        const hitPos = (ball.y - leftPaddle.y) / leftPaddle.h - 0.5;
        ball.vy += hitPos * 3;
        ball.vx = Math.abs(ball.vx);
        // Normalize
        const mag = Math.sqrt(ball.vx*ball.vx + ball.vy*ball.vy);
        ball.vx = (ball.vx / mag) * 1;
        ball.vy = (ball.vy / mag) * 1;
        playSound(500, 0.1, 'square', 0.12);
        spawnParticles(lx, ball.y, '#4f4fff', 8);
    }

    // Right paddle collision
    const rx = W - paddleOffset - paddleW;
    if (ball.x + ball.r > rx && ball.x + ball.r < rx + 15 &&
        ball.y > rightPaddle.y && ball.y < rightPaddle.y + rightPaddle.h && ball.vx > 0) {
        ball.x = rx - ball.r;
        const hitPos = (ball.y - rightPaddle.y) / rightPaddle.h - 0.5;
        ball.vy += hitPos * 3;
        ball.vx = -Math.abs(ball.vx);
        const mag = Math.sqrt(ball.vx*ball.vx + ball.vy*ball.vy);
        ball.vx = (ball.vx / mag) * 1;
        ball.vy = (ball.vy / mag) * 1;
        playSound(400, 0.1, 'square', 0.12);
        spawnParticles(rx, ball.y, '#ff4f4f', 8);
    }

    // Score - ball passes left
    if (ball.x + ball.r < 0) {
        scoreRight++;
        playSound(200, 0.3, 'sawtooth', 0.15);
        spawnParticles(10, ball.y, '#ff4f4f', 20);
        if (scoreRight >= WIN_SCORE) { state = STATE.GAMEOVER; }
        else resetBall(1);
    }

    // Score - ball passes right
    if (ball.x - ball.r > W) {
        scoreLeft++;
        playSound(700, 0.3, 'sine', 0.2);
        spawnParticles(W-10, ball.y, '#4fff4f', 20);
        if (scoreLeft >= WIN_SCORE) { state = STATE.GAMEOVER; }
        else resetBall(-1);
    }

    // Particles
    for (let i = particles.length - 1; i >= 0; i--) {
        const p = particles[i];
        p.x += p.vx;
        p.y += p.vy;
        p.life--;
        if (p.life <= 0) particles.splice(i, 1);
    }
}

// Render
function render() {
    ctx.fillStyle = '#0a0a1a';
    ctx.fillRect(0, 0, W, H);

    // Center line
    ctx.setLineDash([10, 15]);
    ctx.strokeStyle = 'rgba(255,255,255,0.1)';
    ctx.lineWidth = 2;
    ctx.beginPath();
    ctx.moveTo(W/2, 0);
    ctx.lineTo(W/2, H);
    ctx.stroke();
    ctx.setLineDash([]);

    if (state === STATE.MENU) {
        renderMenu();
        return;
    }

    // Paddles
    ctx.fillStyle = powerBoost > 0 ? '#4fff4f' : '#4f4fff';
    ctx.shadowColor = powerBoost > 0 ? '#4fff4f' : '#4f4fff';
    ctx.shadowBlur = 15;
    ctx.fillRect(paddleOffset, leftPaddle.y, paddleW, leftPaddle.h);

    ctx.fillStyle = '#ff4f4f';
    ctx.shadowColor = '#ff4f4f';
    ctx.fillRect(W - paddleOffset - paddleW, rightPaddle.y, paddleW, rightPaddle.h);

    // Ball
    ctx.shadowColor = '#fff';
    ctx.shadowBlur = 20;
    ctx.fillStyle = '#fff';
    ctx.beginPath();
    ctx.arc(ball.x, ball.y, ball.r, 0, Math.PI * 2);
    ctx.fill();
    ctx.shadowBlur = 0;

    // Particles
    for (const p of particles) {
        ctx.globalAlpha = p.life / 50;
        ctx.fillStyle = p.color;
        ctx.fillRect(p.x - 2, p.y - 2, 4, 4);
    }
    ctx.globalAlpha = 1;

    // Scores
    ctx.font = 'bold 48px monospace';
    ctx.textAlign = 'center';
    ctx.fillStyle = 'rgba(79,79,255,0.6)';
    ctx.fillText(scoreLeft, W/2 - 80, 70);
    ctx.fillStyle = 'rgba(255,79,79,0.6)';
    ctx.fillText(scoreRight, W/2 + 80, 70);

    // Word display
    if (state === STATE.PLAYING) {
        const wordSize = Math.min(36, Math.max(16, 400 / currentWord.length));
        ctx.font = `bold ${wordSize}px monospace`;
        ctx.textAlign = 'center';

        // Background bar
        const wordWidth = currentWord.length * wordSize * 0.65 + 40;
        ctx.fillStyle = 'rgba(0,0,0,0.5)';
        ctx.fillRect(W/2 - wordWidth/2, H - 90, wordWidth, wordSize + 20);

        // Draw each character
        const startX = W/2 - (currentWord.length * wordSize * 0.325);
        for (let i = 0; i < currentWord.length; i++) {
            let color;
            if (i < typedChars.length) color = '#4fff4f';
            else if (i === typedChars.length) color = '#ffff4f';
            else color = 'rgba(255,255,255,0.3)';

            ctx.fillStyle = color;
            ctx.shadowColor = i === typedChars.length ? '#ffff4f' : 'transparent';
            ctx.shadowBlur = i === typedChars.length ? 10 : 0;
            ctx.fillText(currentWord[i], startX + i * wordSize * 0.65, H - 55);
        }
        ctx.shadowBlur = 0;

        // Power boost indicator
        if (powerBoost > 0) {
            ctx.font = '14px monospace';
            ctx.fillStyle = '#4fff4f';
            ctx.fillText('POWER!', W/2, H - 100);
        }
    }

    if (state === STATE.GAMEOVER) {
        renderGameOver();
    }
}

function renderMenu() {
    ctx.textAlign = 'center';
    ctx.fillStyle = '#fff';
    ctx.font = 'bold 52px monospace';
    ctx.shadowColor = '#4f4fff';
    ctx.shadowBlur = 20;
    ctx.fillText('TYPING PONG', W/2, H/2 - 60);
    ctx.shadowBlur = 0;

    ctx.font = '18px monospace';
    ctx.fillStyle = 'rgba(255,255,255,0.7)';
    ctx.fillText('Type words to move your paddle up', W/2, H/2 - 10);
    ctx.fillText('Complete words to score points!', W/2, H/2 + 20);
    ctx.fillText('', W/2, H/2 + 50);
    ctx.font = '16px monospace';
    ctx.fillStyle = 'rgba(255,255,255,0.5)';
    ctx.fillText('First to 7 points wins', W/2, H/2 + 50);

    // Blinking prompt
    if (Math.floor(Date.now() / 600) % 2 === 0) {
        ctx.font = '20px monospace';
        ctx.fillStyle = '#4fff4f';
        ctx.fillText('Press any key to start', W/2, H/2 + 110);
    }

    // Decorative ball
    const t = Date.now() / 1000;
    ctx.beginPath();
    ctx.arc(W/2 + Math.sin(t*2)*50, H/2 + 160 + Math.cos(t*3)*20, 8, 0, Math.PI*2);
    ctx.fillStyle = '#fff';
    ctx.shadowColor = '#fff';
    ctx.shadowBlur = 15;
    ctx.fill();
    ctx.shadowBlur = 0;
}

function renderGameOver() {
    ctx.fillStyle = 'rgba(0,0,0,0.7)';
    ctx.fillRect(0, 0, W, H);

    const won = scoreLeft >= WIN_SCORE;
    ctx.textAlign = 'center';
    ctx.font = 'bold 48px monospace';
    ctx.fillStyle = won ? '#4fff4f' : '#ff4f4f';
    ctx.shadowColor = won ? '#4fff4f' : '#ff4f4f';
    ctx.shadowBlur = 20;
    ctx.fillText(won ? 'YOU WIN!' : 'DEFEAT', W/2, H/2 - 30);
    ctx.shadowBlur = 0;

    ctx.font = '24px monospace';
    ctx.fillStyle = '#fff';
    ctx.fillText(`${scoreLeft} - ${scoreRight}`, W/2, H/2 + 20);

    if (Math.floor(Date.now() / 600) % 2 === 0) {
        ctx.font = '18px monospace';
        ctx.fillStyle = 'rgba(255,255,255,0.7)';
        ctx.fillText('Press any key to return to menu', W/2, H/2 + 80);
    }

    // Try high score
    try {
        let best = parseInt(localStorage.getItem('typingPongBest')) || 0;
        const total = scoreLeft + scoreRight;
        if (total > best) localStorage.setItem('typingPongBest', total);
    } catch(e) {}
}

// Game loop
function loop(time) {
    const dt = Math.min((time - lastTime) / 16.67, 3);
    lastTime = time;
    update(dt);
    render();
    requestAnimationFrame(loop);
}

// Init positions
leftPaddle.y = H/2 - leftPaddle.h/2;
rightPaddle.y = H/2 - rightPaddle.h/2;
ball.x = W/2; ball.y = H/2;
requestAnimationFrame(loop);
</script>
