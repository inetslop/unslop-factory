<script>
const canvas = document.getElementById('c');
const ctx = canvas.getContext('2d');

let W, H;
function resize() {
  W = window.innerWidth;
  H = window.innerHeight;
  canvas.width = W;
  canvas.height = H;
}
window.addEventListener('resize', resize);
resize();

// Game constants
const GRID = 12;
const MINE_DENSITY = 0.28;
const ALERT_INCREMENT = 10;
const ALERT_MAX = 100;
const VISION_RADIUS = 2.5;

// Game state
let gameState = 'start'; // start, playing, dead, caught, win
let grid = [];
let playerX = 0, playerY = 0;
let alertLevel = 0;
let visited = [];
let exitX, exitY;
let animTimer = 0;
let deathTimer = 0;
let particles = [];
let shakeTime = 0;
let shakeIntensity = 0;

// Audio
let audioCtx = null;
function initAudio() {
  if (!audioCtx) {
    audioCtx = new (window.AudioContext || window.webkitAudioContext)();
  }
}

function playTone(freq, duration, type = 'sine', vol = 0.15) {
  if (!audioCtx) return;
  const osc = audioCtx.createOscillator();
  const gain = audioCtx.createGain();
  osc.type = type;
  osc.frequency.value = freq;
  gain.gain.setValueAtTime(vol, audioCtx.currentTime);
  gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + duration);
  osc.connect(gain);
  gain.connect(audioCtx.destination);
  osc.start();
  osc.stop(audioCtx.currentTime + duration);
}

function playStep() { playTone(220, 0.08, 'triangle', 0.08); }
function playAlert() { playTone(440, 0.15, 'sawtooth', 0.06); }
function playDeath() { playTone(100, 0.5, 'sawtooth', 0.2); playTone(80, 0.7, 'square', 0.15); }
function playCaught() { playTone(600, 0.3, 'square', 0.1); playTone(500, 0.4, 'square', 0.08); }
function playWin() { playTone(440, 0.15, 'sine', 0.12); setTimeout(()=>playTone(550, 0.15, 'sine', 0.12), 150); setTimeout(()=>playTone(660, 0.3, 'sine', 0.15), 300); }
function playStart() { playTone(330, 0.1, 'triangle', 0.1); }

function initGame() {
  grid = [];
  visited = [];
  particles = [];
  playerX = 0;
  playerY = 0;
  exitX = GRID - 1;
  exitY = GRID - 1;
  alertLevel = 0;
  deathTimer = 0;

  // Generate grid with mines
  for (let y = 0; y < GRID; y++) {
    grid[y] = [];
    visited[y] = [];
    for (let x = 0; x < GRID; x++) {
      grid[y][x] = false;
      visited[y][x] = false;
    }
  }

  // Place mines (avoid start and exit areas)
  let placed = 0;
  const targetMines = Math.floor(GRID * GRID * MINE_DENSITY);
  while (placed < targetMines) {
    const rx = Math.floor(Math.random() * GRID);
    const ry = Math.floor(Math.random() * GRID);
    if (!grid[ry][rx]) {
      // Keep start and exit clear
      if (rx === 0 && ry === 0) continue;
      if (Math.abs(rx) + Math.abs(ry) <= 1) continue;
      if (rx === exitX && ry === exitY) continue;
      grid[ry][rx] = true;
      placed++;
    }
  }

  // Mark start as visited
  visited[0][0] = true;
}

function getAdjacentMineCount(x, y) {
  let count = 0;
  for (let dy = -1; dy <= 1; dy++) {
    for (let dx = -1; dx <= 1; dx++) {
      if (dx === 0 && dy === 0) continue;
      const nx = x + dx, ny = y + dy;
      if (nx >= 0 && nx < GRID && ny >= 0 && ny < GRID && grid[ny][nx]) count++;
    }
  }
  return count;
}

function isAdjacentToMine(x, y) {
  for (let dy = -1; dy <= 1; dy++) {
    for (let dx = -1; dx <= 1; dx++) {
      if (dx === 0 && dy === 0) continue;
      const nx = x + dx, ny = y + dy;
      if (nx >= 0 && nx < GRID && ny >= 0 && ny < GRID && grid[ny][nx]) return true;
    }
  }
  return false;
}

function movePlayer(dx, dy) {
  const nx = playerX + dx;
  const ny = playerY + dy;
  if (nx < 0 || nx >= GRID || ny < 0 || ny >= GRID) return;

  playStep();
  playerX = nx;
  playerY = ny;
  visited[ny][nx] = true;

  // Check mine
  if (grid[ny][nx]) {
    gameState = 'dead';
    deathTimer = 0;
    shakeTime = 0.5;
    shakeIntensity = 8;
    playDeath();
    spawnExplosion(nx, ny);
    return;
  }

  // Check alert
  if (isAdjacentToMine(nx, ny)) {
    alertLevel += ALERT_INCREMENT;
    playAlert();
    if (alertLevel >= ALERT_MAX) {
      gameState = 'caught';
      deathTimer = 0;
      shakeTime = 0.3;
      shakeIntensity = 5;
      playCaught();
      return;
    }
  }

  // Check win
  if (nx === exitX && ny === exitY) {
    gameState = 'win';
    deathTimer = 0;
    playWin();
    spawnWinParticles();
  }
}

function spawnExplosion(x, y) {
  const cellSize = getCellSize();
  const px = getGridX() + (x + 0.5) * cellSize;
  const py = getGridY() + (y + 0.5) * cellSize;
  for (let i = 0; i < 20; i++) {
    const angle = Math.random() * Math.PI * 2;
    const speed = 1 + Math.random() * 3;
    particles.push({ x: px, y: py, vx: Math.cos(angle) * speed, vy: Math.sin(angle) * speed, life: 1, color: `hsl(${Math.random()*30}, 100%, ${50+Math.random()*30}%)` });
  }
}

function spawnWinParticles() {
  const cellSize = getCellSize();
  const px = getGridX() + (exitX + 0.5) * cellSize;
  const py = getGridY() + (exitY + 0.5) * cellSize;
  for (let i = 0; i < 30; i++) {
    const angle = Math.random() * Math.PI * 2;
    const speed = 1 + Math.random() * 2;
    particles.push({ x: px, y: py, vx: Math.cos(angle) * speed, vy: Math.sin(angle) * speed - 1, life: 1, color: `hsl(${140+Math.random()*60}, 80%, ${50+Math.random()*30}%)` });
  }
}

function getCellSize() {
  const gridPixel = Math.min(W * 0.7, H * 0.7);
  return gridPixel / GRID;
}

function getGridX() {
  return (W - GRID * getCellSize()) / 2;
}

function getGridY() {
  return (H - GRID * getCellSize()) / 2 + 20;
}

// Input
let keysDown = {};
window.addEventListener('keydown', (e) => {
  if (!audioCtx) initAudio();
  keysDown[e.key] = true;

  if (gameState === 'start') {
    if (['w','a','s','d','ArrowUp','ArrowDown','ArrowLeft','ArrowRight','Enter',' '].includes(e.key)) {
      gameState = 'playing';
      initGame();
      playStart();
    }
  } else if (gameState === 'playing') {
    if (e.key === 'w' || e.key === 'ArrowUp') movePlayer(0, -1);
    else if (e.key === 's' || e.key === 'ArrowDown') movePlayer(0, 1);
    else if (e.key === 'a' || e.key === 'ArrowLeft') movePlayer(-1, 0);
    else if (e.key === 'd' || e.key === 'ArrowRight') movePlayer(1, 0);
  } else if (gameState === 'dead' || gameState === 'caught' || gameState === 'win') {
    if (e.key === 'r' || e.key === 'R' || e.key === 'Enter' || e.key === ' ') {
      initGame();
      gameState = 'playing';
      playStart();
    }
  }

  if (e.key === 'Escape') {
    if (gameState !== 'start') {
      gameState = 'start';
    }
  }
});
window.addEventListener('keyup', (e) => { keysDown[e.key] = false; });

// Rendering
function draw() {
  ctx.fillStyle = '#0a0e17';
  ctx.fillRect(0, 0, W, H);

  if (gameState === 'start') {
    drawStartScreen();
    return;
  }

  // Shake
  let offsetX = 0, offsetY = 0;
  if (shakeTime > 0) {
    offsetX = (Math.random() - 0.5) * shakeIntensity * (shakeTime / 0.5);
    offsetY = (Math.random() - 0.5) * shakeIntensity * (shakeTime / 0.5);
  }

  ctx.save();
  ctx.translate(offsetX, offsetY);

  const cellSize = getCellSize();
  const gx = getGridX();
  const gy = getGridY();

  // Draw grid
  for (let y = 0; y < GRID; y++) {
    for (let x = 0; x < GRID; x++) {
      const cx = gx + x * cellSize;
      const cy = gy + y * cellSize;

      const distToPlayer = Math.sqrt((x - playerX) ** 2 + (y - playerY) ** 2);
      const inVision = distToPlayer <= VISION_RADIUS;
      const isVisited = visited[y][x];

      // Cell background
      if (isVisited || inVision) {
        let alpha = Math.max(0, 1 - distToPlayer / (VISION_RADIUS + 1));
        ctx.fillStyle = `rgba(20, 30, 50, ${alpha * 0.8})`;
        ctx.fillRect(cx + 1, cy + 1, cellSize - 2, cellSize - 2);

        // Border
        ctx.strokeStyle = `rgba(40, 70, 120, ${alpha * 0.5})`;
        ctx.lineWidth = 1;
        ctx.strokeRect(cx + 1, cy + 1, cellSize - 2, cellSize - 2);

        // Mine count for visited cells
        if (isVisited && !grid[y][x]) {
          const count = getAdjacentMineCount(x, y);
          if (count > 0) {
            ctx.fillStyle = `rgba(100, 200, 255, ${alpha})`;
            ctx.font = `bold ${Math.floor(cellSize * 0.4)}px 'Courier New', monospace`;
            ctx.textAlign = 'center';
            ctx.textBaseline = 'middle';
            ctx.fillText(count.toString(), cx + cellSize / 2, cy + cellSize / 2);
          }
        }

        // Reveal mines on death
        if (gameState === 'dead' && grid[y][x]) {
          ctx.fillStyle = '#ff3030';
          ctx.font = `bold ${Math.floor(cellSize * 0.5)}px 'Courier New', monospace`;
          ctx.textAlign = 'center';
          ctx.textBaseline = 'middle';
          ctx.fillText('✕', cx + cellSize / 2, cy + cellSize / 2);
        }

        // Show all mines on death
        if (gameState === 'dead') {
          for (let my = 0; my < GRID; my++) {
            for (let mx = 0; mx < GRID; mx++) {
              if (grid[my][mx] && !(mx === playerX && my === playerY)) {
                ctx.fillStyle = 'rgba(255, 60, 60, 0.3)';
                ctx.font = `${Math.floor(cellSize * 0.4)}px 'Courier New', monospace`;
                ctx.textAlign = 'center';
                ctx.textBaseline = 'middle';
                ctx.fillText('•', gx + mx * cellSize + cellSize / 2, gy + my * cellSize + cellSize / 2);
              }
            }
          }
        }
      } else {
        // Fog
        ctx.fillStyle = 'rgba(8, 12, 20, 0.95)';
        ctx.fillRect(cx + 1, cy + 1, cellSize - 2, cellSize - 2);
        ctx.strokeStyle = 'rgba(20, 35, 60, 0.3)';
        ctx.lineWidth = 1;
        ctx.strokeRect(cx + 1, cy + 1, cellSize - 2, cellSize - 2);
      }

      // Exit marker
      if (x === exitX && y === exitY && inVision) {
        const pulse = Math.sin(animTimer * 3) * 0.3 + 0.7;
        ctx.fillStyle = `rgba(50, 255, 100, ${pulse * 0.4})`;
        ctx.fillRect(cx + cellSize * 0.2, cy + cellSize * 0.2, cellSize * 0.6, cellSize * 0.6);
        ctx.strokeStyle = `rgba(50, 255, 100, ${pulse})`;
        ctx.lineWidth = 2;
        ctx.strokeRect(cx + cellSize * 0.2, cy + cellSize * 0.2, cellSize * 0.6, cellSize * 0.6);
      }
    }
  }

  // Draw player
  if (gameState === 'playing' || gameState === 'win') {
    const px = gx + (playerX + 0.5) * cellSize;
    const py = gy + (playerY + 0.5) * cellSize;
    const r = cellSize * 0.3;

    // Glow
    const gradient = ctx.createRadialGradient(px, py, 0, px, py, r * 2);
    gradient.addColorStop(0, 'rgba(80, 255, 140, 0.3)');
    gradient.addColorStop(1, 'rgba(80, 255, 140, 0)');
    ctx.fillStyle = gradient;
    ctx.fillRect(px - r * 2, py - r * 2, r * 4, r * 4);

    // Body
    ctx.beginPath();
    ctx.arc(px, py, r, 0, Math.PI * 2);
    ctx.fillStyle = '#50ff8c';
    ctx.fill();
    ctx.strokeStyle = '#aaffcc';
    ctx.lineWidth = 2;
    ctx.stroke();

    // Eye
    ctx.beginPath();
    ctx.arc(px, py - r * 0.15, r * 0.25, 0, Math.PI * 2);
    ctx.fillStyle = '#fff';
    ctx.fill();
  }

  if (gameState === 'dead') {
    // X mark on player
    const px = gx + (playerX + 0.5) * cellSize;
    const py = gy + (playerY + 0.5) * cellSize;
    const r = cellSize * 0.25;
    ctx.strokeStyle = '#ff4444';
    ctx.lineWidth = 3;
    ctx.beginPath();
    ctx.moveTo(px - r, py - r); ctx.lineTo(px + r, py + r);
    ctx.moveTo(px + r, py - r); ctx.lineTo(px - r, py + r);
    ctx.stroke();
  }

  // Particles
  for (let i = particles.length - 1; i >= 0; i--) {
    const p = particles[i];
    p.x += p.vx;
    p.y += p.vy;
    p.vy += 0.05;
    p.life -= 0.02;
    if (p.life <= 0) { particles.splice(i, 1); continue; }
    ctx.fillStyle = p.color;
    ctx.globalAlpha = p.life;
    ctx.beginPath();
    ctx.arc(p.x, p.y, 3 * p.life, 0, Math.PI * 2);
    ctx.fill();
    ctx.globalAlpha = 1;
  }

  // Alert bar
  const barW = GRID * cellSize;
  const barH = 24;
  const barX = gx;
  const barY = gy - 50;

  ctx.fillStyle = 'rgba(10, 15, 25, 0.9)';
  ctx.fillRect(barX, barY, barW, barH);
  ctx.strokeStyle = 'rgba(100, 180, 255, 0.4)';
  ctx.lineWidth = 1;
  ctx.strokeRect(barX, barY, barW, barH);

  const alertPct = alertLevel / ALERT_MAX;
  const alertColor = alertPct < 0.5 ? '#44ff88' : alertPct < 0.8 ? '#ffcc00' : '#ff3344';
  ctx.fillStyle = alertColor;
  ctx.globalAlpha = 0.7 + Math.sin(animTimer * 5) * 0.1 * alertPct;
  ctx.fillRect(barX + 2, barY + 2, (barW - 4) * alertPct, barH - 4);
  ctx.globalAlpha = 1;

  ctx.fillStyle = '#aaddff';
  ctx.font = `bold ${Math.floor(barH * 0.5)}px 'Courier New', monospace`;
  ctx.textAlign = 'left';
  ctx.textBaseline = 'middle';
  ctx.fillText('ALERT', barX + 8, barY + barH / 2);

  ctx.textAlign = 'right';
  ctx.fillText(Math.floor(alertLevel) + '%', barX + barW - 8, barY + barH / 2);

  // Status messages
  if (gameState === 'dead') {
    ctx.fillStyle = 'rgba(10, 15, 25, 0.85)';
    ctx.fillRect(0, 0, W, H);
    drawEndScreen('DETONATED', '#ff4444', 'You stepped on a mine.');
  } else if (gameState === 'caught') {
    ctx.fillStyle = 'rgba(10, 15, 25, 0.85)';
    ctx.fillRect(0, 0, W, H);
    drawEndScreen('DETECTED', '#ffcc00', 'Your alert level reached maximum.');
  } else if (gameState === 'win') {
    ctx.fillStyle = 'rgba(10, 15, 25, 0.85)';
    ctx.fillRect(0, 0, W, H);
    drawEndScreen('EXTRACTION SUCCESS', '#44ff88', 'You reached the exit undetected!');
  }

  ctx.restore();
}

function drawEndScreen(title, color, subtitle) {
  ctx.textAlign = 'center';
  ctx.textBaseline = 'middle';

  const titleSize = Math.min(W * 0.05, 42);
  ctx.font = `bold ${titleSize}px 'Courier New', monospace`;
  ctx.fillStyle = color;
  ctx.fillText(title, W / 2, H * 0.35);

  ctx.font = `${Math.min(W * 0.022, 18)}px 'Courier New', monospace`;
  ctx.fillStyle = '#aaccdd';
  ctx.fillText(subtitle, W / 2, H * 0.45);

  if (gameState === 'win') {
    ctx.font = `${Math.min(W * 0.018, 16)}px 'Courier New', monospace`;
    ctx.fillStyle = '#6688aa';
    ctx.fillText(`Final Alert Level: ${Math.floor(alertLevel)}%`, W / 2, H * 0.53);
  }

  if (Math.floor(animTimer * 2) % 2 === 0) {
    ctx.font = `bold ${Math.min(W * 0.025, 20)}px 'Courier New', monospace`;
    ctx.fillStyle = '#88bbdd';
    ctx.fillText('PRESS R OR ENTER TO RESTART', W / 2, H * 0.65);
  }
}

// Game loop
let lastTime = 0;
function gameLoop(time) {
  const dt = Math.min((time - lastTime) / 1000, 0.05);
  lastTime = time;
  animTimer += dt;

  if (shakeTime > 0) shakeTime -= dt;

  draw();
  requestAnimationFrame(gameLoop);
}

initGame();
requestAnimationFrame(gameLoop);
