function playSound(type) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.connect(gain);

    switch(type) {

            osc.type = 'square';
            osc.frequency.setValueAtTime(300, audioCtx.currentTime);
            osc.frequency.linearRampToValueAtTime(500, audioCtx.currentTime + 0.1);
            gain.gain.setValueAtTime(0.15, audioCtx.currentTime);
            gain.gain.linearRampToValueAtTime(0, audioCtx.currentTime + 0.15);
            osc.start(); osc.stop(audioCtx.currentTime + 0.15);
            break;
        case 'collect':
            osc.type = 'sine';
            osc.frequency.setValueAtTime(600, audioCtx.currentTime);
            osc.frequency.linearRampToValueAtTime(900, audioCtx.currentTime + 0.1);
            gain.gain.setValueAtTime(0.2, audioCtx.currentTime);

function loadLevel(idx) {
    const data = levels[idx].data;
    levelH = data.length;
    levelW = data[0].length;
    tiles = [];
    plates = [];
    doors = [];
    keys_ = [];
    crates = [];
    spikes = [];
    goal = null;

    for (let r = 0; r < levelH; r++) {
        for (let c = 0; c < data[r].length; c++) {
            const ch = data[r][c];
            if (ch === '#') tiles.push({ x: c, y: r, type: 'wall' });
            else if (ch === 'P') player = { x: c * TILE + 4, y: r * TILE + 4, vx: 0, vy: 0, w: 24, h: 26, onGround: false, facing: 1 };
            else if (ch === 'D') doors.push({ x: c, y: r, open: false });
            else if (ch === 'G') goal = { x: c * TILE + 4, y: r * TILE + 4, w: 24, h: 26 };
            else if (ch === 'B') plates.push({ x: c, y: r, pressed: false });
            else if (ch === 'K') keys_.push({ x: c * TILE + 4, y: r * TILE + 4, w: 16, h: 16, collected: false });
            else if (ch === 'C') crates.push({ x: c * TILE + 4, y: r * TILE + 4, vx: 0, vy: 0, w: 28, h: 28, onGround: false });
            else if (ch === 'S') spikes.push({ x: c, y: r });
        }
    }
    player.vx = 0;
    player.vy = 0;
}

function getTileAt(px, py) {
    const c = Math.floor(px / TILE);
    const r = Math.floor(py / TILE);
    if (r < 0 || r >= levelH) return '#';
    if (c < 0 || c >= levelW) return '#';
    return levels[currentLevel].data[r][c];
}

function isSolid(px, py) {
    const ch = getTileAt(px, py);
    return ch === '#' || (doors.some(d => d.x * TILE === Math.floor(px/TILE)*TILE && d.y === Math.floor(py/TILE) && !d.open));
}

function isDoorAt(px, py) {
    const c = Math.floor(px / TILE);
    const r = Math.floor(py / TILE);
    return doors.find(d => d.x === c && d.y === r && !d.open);
}

function moveEntity(ent) {
    ent.vy += 0.6; // gravity
    if (ent.vy > 12) ent.vy = 12;

    // Horizontal
    ent.x += ent.vx;
    let c, r;
    // Check horizontal collisions
    for (let i = 0; i < 3; i++) {
        const testX = ent.x + (i === 0 ? 0 : i === 1 ? ent.w : 0);
        const testY = ent.y + (i === 1 ? 2 : 8);
        if (isSolid(testX, testY) || isDoorAt(testX, testY)) {
            if (ent.vx > 0) ent.x = Math.floor((testX - 1) / TILE) * TILE - ent.w;
            else if (ent.vx < 0) ent.x = Math.floor(testX / TILE + 1) * TILE;
            ent.vx = 0;
        }
    }

    // Vertical
    ent.y += ent.vy;
    ent.onGround = false;
    for (let i = 0; i < 3; i++) {
        const testX = ent.x + (i === 0 ? 2 : i === 1 ? ent.w - 2 : Math.floor(ent.w/2));
        const testY = ent.y + (ent.vy > 0 ? ent.h : 0);
        if (isSolid(testX, testY)) {
            if (ent.vy > 0) {
                ent.y = Math.floor(testY / TILE) * TILE - ent.h;
                ent.vy = 0;
                ent.onGround = true;
            } else if (ent.vy < 0) {
                ent.y = Math.floor(testY / TILE + 1) * TILE;
                ent.vy = 0;
            }
        }
    }

    // Bounds check
    if (ent.y > levelH * TILE + 100) return true; // fell off

    // Spike check
    for (const s of spikes) {
        const sx = s.x * TILE, sy = s.y * TILE;
        if (rectOverlap(ent.x, ent.y, ent.w, ent.h, sx + 4, sy + 16, TILE - 8, TILE - 16)) return 'spike';
    }

    // Plate check
    for (const p of plates) {
        const px = p.x * TILE, py = p.y * TILE;
        if (!p.pressed && rectOverlap(ent.x, ent.y, ent.w, ent.h, px, py + TILE - 8, TILE, 8)) {
            p.pressed = true;
            playSound('plate');
            // Open connected doors
            for (const d of doors) d.open = true;
            playSound('door');
        }
    }

    return false;
}

function rectOverlap(x1, y1, w1, h1, x2, y2, w2, h2) {
    return x1 < x2 + w2 && x1 + w1 > x2 && y1 < y2 + h2 && y1 + h1 > y2;
}

function spawnParticles(x, y, color, count) {
    for (let i = 0; i < count; i++) {
        particles.push({
            x: x + Math.random() * 20 - 10,
            y: y + Math.random() * 20 - 10,
            vx: (Math.random() - 0.5) * 4,
            vy: (Math.random() - 0.5) * 4 - 2,
            life: 30 + Math.random() * 20,
            color: color
        });
    }
}

function drawTile(x, y, ch) {

function update() {
    time++;

    if (gameState === 'start') {
        if (keys['Enter'] || keys['Space']) {
            initAudio();
            gameState = 'playing';
            currentLevel = 0;
            loadLevel(0);
            camX = player.x - W/2;
            camY = player.y - H/2;
        }
        return;
    }

    if (gameState === 'levelComplete') {
        transTimer++;
        if ((keys['Enter'] || keys['Space']) && transTimer > 30) {
            currentLevel++;
            if (currentLevel >= levels.length) {
                gameState = 'gameComplete';
            } else {
                loadLevel(currentLevel);
                camX = player.x - W/2;
                camY = player.y - H/2;
                gameState = 'playing';
            }
        }
        return;
    }

    if (gameState === 'gameComplete') {
        if (keys['KeyR']) {
            currentLevel = 0;
            loadLevel(0);
            camX = player.x - W/2;
            camY = player.y - H/2;
            gameState = 'playing';
        }
        return;
    }

    if (gameState === 'dead') {
        deathTimer--;
        if (deathTimer <= 0) {
            loadLevel(currentLevel);
            camX = player.x - W/2;
            camY = player.y - H/2;
            gameState = 'playing';
        }
        return;
    }

    // Playing state
    if (keys['KeyR']) {
        loadLevel(currentLevel);
        camX = player.x - W/2;
        camY = player.y - H/2;
        return;
    }

    // Player input
    let moveSpeed = 3.5;
    if (keys['ArrowLeft'] || keys['KeyA']) {
        player.vx = -moveSpeed;
        player.facing = -1;
    } else if (keys['ArrowRight'] || keys['KeyD']) {
        player.vx = moveSpeed;
        player.facing = 1;
    } else {
        player.vx *= 0.7;
        if (Math.abs(player.vx) < 0.3) player.vx = 0;
    }

    if ((keys['ArrowUp'] || keys['KeyW'] || keys['Space']) && player.onGround) {
        player.vy = -10;
        player.onGround = false;
        playSound('jump');
    }

    // Move player
    const result = moveEntity(player);
    if (result === true) {
        gameState = 'dead';
        deathTimer = 30;
        playSound('death');
        spawnParticles(player.x, player.y, '255,100,100', 15);
        return;
    } else if (result === 'spike') {
        gameState = 'dead';
        deathTimer = 30;
        playSound('death');
        spawnParticles(player.x, player.y, '255,100,100', 15);
        return;
    }

    // Crate physics
    for (const c of crates) {
        const oldX = c.x;
        moveEntity(c);
        if (c.onGround) {
            // Check if crate is on a plate
            for (const p of plates) {
                const px = p.x * TILE, py = p.y * TILE;
                if (!p.pressed && rectOverlap(c.x, c.y, c.w, c.h, px, py + TILE - 8, TILE, 8)) {
                    p.pressed = true;
                    playSound('plate');
                    for (const d of doors) d.open = true;
                    playSound('door');
                }
            }
        }
    }

    // Player pushes crates
    for (const c of crates) {
        if (rectOverlap(player.x, player.y, player.w, player.h, c.x, c.y, c.w, c.h)) {
            const pushDir = player.facing;
            c.vx = pushDir * 2;
        }
    }

    // Key collection
    for (const k of keys_) {
        if (!k.collected && rectOverlap(player.x, player.y, player.w, player.h, k.x, k.y, k.w, k.h)) {
            k.collected = true;
            playSound('collect');
            spawnParticles(k.x + 8, k.y + 8, '255,221,0', 10);
        }
    }

    // Goal check
    if (goal && rectOverlap(player.x, player.y, player.w, player.h, goal.x, goal.y, goal.w, goal.h)) {
        playSound('win');
        spawnParticles(goal.x + 12, goal.y + 12, '0,255,100', 20);
        if (currentLevel < levels.length - 1) {
            gameState = 'levelComplete';
            transTimer = 0;
        } else {
            gameState = 'gameComplete';
        }
        return;
    }

    // Update particles
    for (let i = particles.length - 1; i >= 0; i--) {
        const p = particles[i];
        p.x += p.vx;
        p.y += p.vy;
        p.vy += 0.2;
        p.life--;
        if (p.life <= 0) particles.splice(i, 1);
    }

    updateCamera();
}

function draw() {
    ctx.clearRect(0, 0, W, H);

    if (gameState === 'start') {
        drawStartScreen();
        return;
    }

    // Background
    const grad = ctx.createLinearGradient(0, 0, 0, H);
    grad.addColorStop(0, '#1a1a3e');
    grad.addColorStop(1, '#0f0f23');
    ctx.fillStyle = grad;
    ctx.fillRect(0, 0, W, H);

    // Draw tiles
    const startCol = Math.floor(camX / TILE) - 1;
    const endCol = Math.ceil((camX + W) / TILE) + 1;
    const startRow = Math.floor(camY / TILE) - 1;
    const endRow = Math.ceil((camY + H) / TILE) + 1;

    for (let r = startRow; r <= endRow; r++) {
        for (let c = startCol; c <= endCol; c++) {
            if (r < 0 || r >= levelH || c < 0 || c >= levelW) continue;
            const ch = levels[currentLevel].data[r][c];
            if (ch !== '.') drawTile(c, r, ch);
        }
    }

    // Draw crates
    for (const c of crates) drawCrate(c);

    // Draw player
    drawPlayer();

    // Draw particles
    drawParticles();

    // UI
    drawUI();

    if (gameState === 'dead') drawDeath();
    if (gameState === 'levelComplete') drawLevelComplete();
    if (gameState === 'gameComplete') drawGameComplete();
}

function gameLoop() {
    update();
    draw();
    requestAnimationFrame(gameLoop);
}

loadLevel(0);
player.x = 100; player.y = 100; // default until start
camX = 0; camY = 0;
gameLoop();
