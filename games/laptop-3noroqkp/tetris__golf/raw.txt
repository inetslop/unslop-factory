const ctx = canvas.getContext('2d');

let W, H, scale;
let audioCtx = null;

function resize() {
    const maxW = window.innerWidth;
    const maxH = window.innerHeight;
    const ratio = 700 / 900;
    if (maxW / maxH > ratio) {
        canvas.height = maxH;
        canvas.width = Math.floor(maxH * ratio);
    } else {
        canvas.width = maxW;

}

// Audio
function initAudio() {
    if (!audioCtx) audioCtx = new (window.AudioContext || webkitAudioContext)();
}
function playSound(type) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    if (type === 'move') {
        osc.type = 'sine';
        osc.frequency.value = 200;
        gain.gain.setValueAtTime(0.1, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.05);
        osc.start(); osc.stop(audioCtx.currentTime + 0.05);
    } else if (type === 'rotate') {
        osc.type = 'sine';
        osc.frequency.value = 300;
        gain.gain.setValueAtTime(0.12, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.08);
        osc.start(); osc.stop(audioCtx.currentTime + 0.08);
    } else if (type === 'drop') {
        osc.type = 'triangle';
        osc.frequency.value = 150;
        gain.gain.setValueAtTime(0.2, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.1);
        osc.start(); osc.stop(audioCtx.currentTime + 0.1);
    } else if (type === 'sink') {
        osc.type = 'sine';
        osc.frequency.setValueAtTime(600, audioCtx.currentTime);
        osc.frequency.exponentialRampToValueAtTime(800, audioCtx.currentTime + 0.15);
        gain.gain.setValueAtTime(0.25, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.3);
        osc.start(); osc.stop(audioCtx.currentTime + 0.3);
    } else if (type === 'hole') {
        osc.type = 'sine';
        osc.frequency.setValueAtTime(400, audioCtx.currentTime);
        osc.frequency.exponentialRampToValueAtTime(600, audioCtx.currentTime + 0.1);
        osc.frequency.exponentialRampToValueAtTime(800, audioCtx.currentTime + 0.2);
        gain.gain.setValueAtTime(0.3, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.5);
        osc.start(); osc.stop(audioCtx.currentTime + 0.5);
    } else if (type === 'gameover') {
        osc.type = 'sawtooth';
        osc.frequency.setValueAtTime(300, audioCtx.currentTime);
        osc.frequency.exponentialRampToValueAtTime(100, audioCtx.currentTime + 0.5);
        gain.gain.setValueAtTime(0.2, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.6);
        osc.start(); osc.stop(audioCtx.currentTime + 0.6);
    }
}

// Game state
const COLS = 10;
const ROWS = 20;
let cellSize, boardX, boardY;
let grid, currentPiece, nextPiece, score, linesCleared, strokes, hole, par;
let gameRunning, gameOver, showStart, showHoleComplete;
let dropInterval, lastDrop, lockDelay;
let particles = [];
let flagAnimation = 0;

const PIECES = [
    { shape: [[1,1,1,1]], color: '#ffffff' },
    { shape: [[1,1],[1,1]], color: '#fff8dc' },
    { shape: [[1,1,1],[0,1,0]], color: '#f0f0f0' },
    { shape: [[1,0,1],[1,1,1]], color: '#e8e8e8' },
    { shape: [[1,1,0],[0,1,1]], color: '#ffffff' },
    { shape: [[0,1,1],[1,1,0]], color: '#f5f5dc' },
    { shape: [[1,1],[1,0],[1,0]], color: '#fffaaa' }
];

function resetGame() {
    grid = Array.from({length: ROWS}, () => Array(COLS).fill(null));
    score = 0; linesCleared = 0; strokes = 0; hole = 1; par = 8;
    gameOver = false; showHoleComplete = false;
    nextPiece = randomPiece();
    spawnPiece();
    particles = [];
}

function randomPiece() {
    return Math.floor(Math.random() * PIECES.length);
}

function spawnPiece() {
    const p = PIECES[nextPiece];
    currentPiece = {
        type: nextPiece,
        shape: p.shape.map(r => [...r]),
        x: Math.floor(COLS / 2) - Math.floor(p.shape[0].length / 2),
        y: 0,
        color: p.color
    };
    nextPiece = randomPiece();
    if (collides(currentPiece.x, currentPiece.y, currentPiece.shape)) {
        gameOver = true;
        playSound('gameover');
    }
}

function collides(x, y, shape) {
    for (let r = 0; r < shape.length; r++) {
        for (let c = 0; c < shape[r].length; c++) {
            if (shape[r][c]) {
                let nx = x + c, ny = y + r;
                if (nx < 0 || nx >= COLS || ny >= ROWS) return true;
                if (ny >= 0 && grid[ny][nx]) return true;
            }
        }
    }
    return false;
}

function rotate() {
    let s = currentPiece.shape;
    let rotated = [];
    for (let r = 0; r < s.length; r++) {
        for (let c = 0; c < s[r].length; c++) {
            if (s[r][c]) rotated.push([r, c]);
        }
    }
    let newShape = [];
    let maxR = Math.max(...rotated.map(p => p[1]));
    let maxC = Math.max(...rotated.map(p => p[0]));
    for (let r = 0; r <= maxR; r++) {
        let row = [];
        for (let c = 0; c <= maxC; c++) {
            row.push(rotated.some(p => p[1] === r && p[0] === c) ? 1 : 0);
        }
        newShape.push(row);
    }
    let nx = currentPiece.x, ny = currentPiece.y;
    if (collides(nx, ny, newShape)) {
        // Wall kick
        if (!collides(nx + 1, ny, newShape)) nx += 1;
        else if (!collides(nx - 1, ny, newShape)) nx -= 1;
        else return;
    }
    currentPiece.shape = newShape;
    currentPiece.x = nx;
    playSound('rotate');
}

function lockPiece() {
    for (let r = 0; r < currentPiece.shape.length; r++) {
        for (let c = 0; c < currentPiece.shape[r].length; c++) {
            if (currentPiece.shape[r][c]) {
                let gy = currentPiece.y + r, gx = currentPiece.x + c;
                if (gy >= 0) grid[gy][gx] = currentPiece.color;
            }
        }
    }
    strokes++;
    playSound('drop');
    clearLines();
    spawnPiece();
}

function clearLines() {
    let cleared = 0;
    for (let r = ROWS - 1; r >= 0; r--) {
        if (grid[r].every(c => c !== null)) {
            grid.splice(r, 1);
            grid.unshift(Array(COLS).fill(null));
            cleared++;
            playSound('sink');
            // Particles
            for (let c = 0; c < COLS; c++) {
                particles.push({ x: boardX + c * cellSize + cellSize/2, y: boardY + r * cellSize + cellSize/2, vx: (Math.random()-0.5)*4, vy: -Math.random()*3-1, life: 30, color: '#fff' });
            }
        }
    }
    if (cleared > 0) {
        linesCleared += cleared;
        score += [0, 100, 300, 500, 800][cleared] * hole;
        flagAnimation = 20;
        if (linesCleared >= par) {
            showHoleComplete = true;
            playSound('hole');
            let diff = strokes - par;
            let label = '';
            if (diff <= -3) label = 'EAGLE!';
            else if (diff === -2) label = 'BIRDIE!';
            else if (diff === -1) label = 'ALBATROSS!';
            else if (diff === 0) label = 'PAR';
            else if (diff <= 3) label = 'BOGEY';
            else label = 'DOUBLE BOGEY';
            showHoleCompleteLabel = label;
        }
    }
}

let showHoleCompleteLabel = '';

function nextHole() {
    hole++;
    par = Math.min(8 + (hole - 1) * 2, 20);
    linesCleared = 0;
    strokes = 0;
    showHoleComplete = false;
}

// Drawing
function drawBoard() {
    boardX = Math.floor((W - COLS * cellSize) / 2);
    boardY = Math.floor(H / 2 - (ROWS * cellSize) / 2);

    // Background
    ctx.fillStyle = '#1a5c1a';
    ctx.fillRect(0, 0, W, H);

    // Board background - golf green
    let grad = ctx.createLinearGradient(boardX, boardY, boardX, boardY + ROWS * cellSize);
    grad.addColorStop(0, '#2d7a2d');
    grad.addColorStop(1, '#1e5c1e');
    ctx.fillStyle = grad;
    ctx.fillRect(boardX - 4*scale, boardY - 4*scale, COLS * cellSize + 8*scale, ROWS * cellSize + 8*scale);

    // Grid lines
    ctx.strokeStyle = 'rgba(255,255,255,0.08)';
    ctx.lineWidth = 1;
    for (let r = 0; r <= ROWS; r++) {
        ctx.beginPath();
        ctx.moveTo(boardX, boardY + r * cellSize);
        ctx.lineTo(boardX + COLS * cellSize, boardY + r * cellSize);
        ctx.stroke();
    }
    for (let c = 0; c <= COLS; c++) {
        ctx.beginPath();
        ctx.moveTo(boardX + c * cellSize, boardY);
        ctx.lineTo(boardX + c * cellSize, boardY + ROWS * cellSize);
        ctx.stroke();
    }

    // Filled cells (golf balls)
    for (let r = 0; r < ROWS; r++) {
        for (let c = 0; c < COLS; c++) {
            if (grid[r][c]) drawGolfBall(boardX + c * cellSize, boardY + r * cellSize, cellSize);
        }
    }

    // Current piece
    if (currentPiece && gameRunning) {
        for (let r = 0; r < currentPiece.shape.length; r++) {
            for (let c = 0; c < currentPiece.shape[r].length; c++) {
                if (currentPiece.shape[r][c]) {
                    let px = boardX + (currentPiece.x + c) * cellSize;
                    let py = boardY + (currentPiece.y + r) * cellSize;
                    drawGolfBall(px, py, cellSize);
                }
            }
        }
        // Ghost piece
        let ghostY = currentPiece.y;
        while (!collides(currentPiece.x, ghostY + 1, currentPiece.shape)) ghostY++;
        ctx.globalAlpha = 0.3;
        for (let r = 0; r < currentPiece.shape.length; r++) {
            for (let c = 0; c < currentPiece.shape[r].length; c++) {
                if (currentPiece.shape[r][c]) {
                    let px = boardX + (currentPiece.x + c) * cellSize;
                    let py = boardY + (ghostY + r) * cellSize;
                    drawGolfBall(px, py, cellSize);
                }
            }
        }
        ctx.globalAlpha = 1;
    }

    // Particles
    particles = particles.filter(p => p.life > 0);
    particles.forEach(p => {
        p.x += p.vx; p.y += p.vy; p.vy += 0.2; p.life--;
        ctx.globalAlpha = p.life / 30;
        ctx.fillStyle = p.color;
        ctx.beginPath();
        ctx.arc(p.x, p.y, 3 * scale, 0, Math.PI * 2);
        ctx.fill();
    });
    ctx.globalAlpha = 1;

    // Flag animation on sink
    if (flagAnimation > 0) {
        flagAnimation--;
        let fx = boardX + COLS * cellSize / 2;
        let fy = boardY - 20 * scale;
        ctx.strokeStyle = '#fff';
        ctx.lineWidth = 2 * scale;
        ctx.beginPath();
        ctx.moveTo(fx, fy);
        ctx.lineTo(fx, fy - 30 * scale);
        ctx.stroke();
        ctx.fillStyle = '#ff4444';
        ctx.beginPath();
        ctx.moveTo(fx, fy - 30 * scale);
        ctx.lineTo(fx + 20 * scale, fy - 25 * scale);
        ctx.lineTo(fx, fy - 20 * scale);
        ctx.fill();
    }

    // HUD
    drawHUD();
}

function drawGolfBall(x, y, size) {
    let cx = x + size/2, cy = y + size/2;
    let r = size * 0.38;
    // Ball body
    ctx.fillStyle = '#fff';
    ctx.beginPath();
    ctx.arc(cx, cy, r, 0, Math.PI * 2);
    ctx.fill();
    // Dimples
    ctx.fillStyle = 'rgba(0,0,0,0.15)';
    let dimples = [
        [0, -0.5], [0.3, -0.3], [-0.3, -0.3],
        [0, 0], [0.4, 0], [-0.4, 0],
        [0.2, 0.4], [-0.2, 0.4]
    ];
    dimples.forEach(d => {
        ctx.beginPath();
        ctx.arc(cx + d[0]*r*0.6, cy + d[1]*r*0.6, r*0.12, 0, Math.PI*2);
        ctx.fill();
    });
    // Highlight
    ctx.fillStyle = 'rgba(255,255,255,0.4)';
    ctx.beginPath();
    ctx.arc(cx - r*0.3, cy - r*0.3, r*0.25, 0, Math.PI*2);
    ctx.fill();
}

function draw() {
    cellSize = Math.floor(Math.min((W - 150*scale) / COLS, (H - 80*scale) / ROWS));

    if (showStart) {
        drawStart();
    } else {
        drawBoard();
        if (gameOver) drawGameOver();
        if (showHoleComplete) drawHoleComplete();
    }
}

let lastTime = 0;
function gameLoop(time) {
    let dt = time - lastTime;
    lastTime = time;

    if (gameRunning && !gameOver && !showHoleComplete) {
        dropInterval = Math.max(300 - (hole - 1) * 25, 80);
        if (dt > dropInterval) {
            lastDrop = time;
            if (!collides(currentPiece.x, currentPiece.y + 1, currentPiece.shape)) {
                currentPiece.y++;
            } else {
                lockPiece();
            }
        }
    }

    draw();
    requestAnimationFrame(gameLoop);
}

function handleKey(e) {
    if (!audioCtx) initAudio();

    if (showStart) {
        if (e.key === 'Enter') {
            showStart = false;
            gameRunning = true;
            resetGame();
        }
        return;
    }

    if (gameOver) {
        if (e.key === 'Enter') {
            gameOver = false;
            gameRunning = true;
            resetGame();
        }
        return;
    }

    if (showHoleComplete) {
        if (e.key === 'Enter') {
            nextHole();
        }
        return;
    }

    if (!gameRunning || !currentPiece) return;

    switch (e.key) {
        case 'ArrowLeft':
            if (!collides(currentPiece.x - 1, currentPiece.y, currentPiece.shape)) {
                currentPiece.x--;
                playSound('move');
            }
            e.preventDefault();
            break;
        case 'ArrowRight':
            if (!collides(currentPiece.x + 1, currentPiece.y, currentPiece.shape)) {
                currentPiece.x++;
                playSound('move');
            }
            e.preventDefault();
            break;
        case 'ArrowUp':
            rotate();
            e.preventDefault();
            break;
        case 'ArrowDown':
            if (!collides(currentPiece.x, currentPiece.y + 1, currentPiece.shape)) {
                currentPiece.y++;
                score += 1;
            } else {
                lockPiece();
            }
            e.preventDefault();
            break;
        case ' ':
            while (!collides(currentPiece.x, currentPiece.y + 1, currentPiece.shape)) {
                currentPiece.y++;
                score += 2;
            }
            lockPiece();
            e.preventDefault();
            break;
    }
}

window.addEventListener('keydown', handleKey);

    if (showStart || gameOver || showHoleComplete) {

    if (!gameRunning || !currentPiece) return;
    let t = e.changedTouches[0];
    let dx = t.clientX - touchStartX;
    let dy = t.clientY - touchStartY;
    if (Math.abs(dx) > Math.abs(dy)) {
        if (dx > 20 && !collides(currentPiece.x + 1, currentPiece.y, currentPiece.shape)) { currentPiece.x++; }
        else if (dx < -20 && !collides(currentPiece.x - 1, currentPiece.y, currentPiece.shape)) { currentPiece.x--; }
    } else if (dy > 60) {
        while (!collides(currentPiece.x, currentPiece.y + 1, currentPiece.shape)) { currentPiece.y++; score += 2; }
        lockPiece();
    } else if (dy < -30) {
        rotate();
    } else if (Math.abs(dx) < 10 && Math.abs(dy) < 10) {
        // tap = hard drop
        while (!collides(currentPiece.x, currentPiece.y + 1, currentPiece.shape)) { currentPiece.y++; score += 2; }
        lockPiece();
    }
});
