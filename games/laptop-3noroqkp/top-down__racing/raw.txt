  if (!audioCtx) return;

  const gain = audioCtx.createGain();
  osc.type = 'sawtooth';
  osc.frequency.value = 80 + speed * 2;

class Car {
  constructor(x, y, angle, color) {
    this.x = x; this.y = y;
    this.angle = angle;
    this.speed = 0;
    this.maxSpeed = 8;
    this.accel = 0.15;
    this.brake = 0.3;
    this.friction = 0.97;
    this.offTrackFriction = 0.92;
    this.steerSpeed = 0.04;
    this.color = color;
    this.lap = 0;
    this.lastCheckpoint = -1;
    this.checkpoints = new Set();
    this.finishTime = null;
    this.isPlayer = false;
  }

  update(keys, isOffTrack) {
    const accel = keys.up ? this.accel : 0;
    const brake = keys.down ? this.brake : 0;
    const handbrake = keys.space ? 0.94 : 1;

    if (accel) this.speed += accel;
    if (brake) this.speed -= brake;
    this.speed *= isOffTrack ? this.offTrackFriction : this.friction;
    this.speed *= handbrake;

    if (this.speed > this.maxSpeed) this.speed = this.maxSpeed;
    if (this.speed < -2) this.speed = -2;
    if (Math.abs(this.speed) < 0.05 && !accel && !brake) this.speed = 0;

    const steerFactor = Math.min(Math.abs(this.speed) / 3, 1);
    if (keys.left && this.speed > 0.1) this.angle -= this.steerSpeed * steerFactor;
    else if (keys.left && this.speed < -0.1) this.angle += this.steerSpeed * steerFactor * 0.5;
    if (keys.right && this.speed > 0.1) this.angle += this.steerSpeed * steerFactor;
    else if (keys.right && this.speed < -0.1) this.angle -= this.steerSpeed * steerFactor * 0.5;

    this.x += Math.cos(this.angle) * this.speed;
    this.y += Math.sin(this.angle) * this.speed;
  }

  updateAI(target, speedMult) {
    const dx = target.x - this.x;
    const dy = target.y - this.y;
    const dist = Math.sqrt(dx*dx + dy*dy);
    const targetAngle = Math.atan2(dy, dx);
    let angleDiff = targetAngle - this.angle;
    while (angleDiff > Math.PI) angleDiff -= 2*Math.PI;
    while (angleDiff < -Math.PI) angleDiff += 2*Math.PI;

    if (Math.abs(angleDiff) > 0.1) {
      this.angle += Math.sign(angleDiff) * this.steerSpeed * 0.8;
    }

    const targetSpeed = this.maxSpeed * speedMult * (dist < 60 ? 0.7 : 1);
    if (this.speed < targetSpeed) this.speed += this.accel * 0.8;
    else this.speed *= 0.95;

    this.x += Math.cos(this.angle) * this.speed;
    this.y += Math.sin(this.angle) * this.speed;
  }

  draw(ctx, camX, camY) {
    ctx.save();
    ctx.translate(this.x - camX, this.y - camY);

    ctx.fillRect(-14, -7, 4, 14);
    ctx.restore();
  }
}

function isOnTrack(x, y) {
  for (let i = 0; i < trackPoints.length; i++) {
    const a = trackPoints[i];
    const b = trackPoints[(i+1) % trackPoints.length];
    const d = pointToSegmentDist(x, y, a, b);
    if (d < TRACK_WIDTH / 2 + 10) return true;
  }
  return false;
}

function pointToSegmentDist(px, py, ax, ay, bx, by) {
  const dx = bx - ax, dy = by - ay;
  const len2 = dx*dx + dy*dy;
  let t = ((px-ax)*dx + (py-ay)*dy) / len2;
  t = Math.max(0, Math.min(1, t));
  const cx = ax + t*dx, cy = ay + t*dy;
  return Math.sqrt((px-cx)**2 + (py-cy)**2);
}

function getCheckpointIndex(car) {
  let bestDist = Infinity, bestIdx = -1;
  for (let i = 0; i < trackPoints.length; i++) {
    const dx = car.x - trackPoints[i].x;
    const dy = car.y - trackPoints[i].y;
    const d = dx*dx + dy*dy;
    if (d < bestDist) { bestDist = d; bestIdx = i; }
  }
  return bestIdx;
}

function checkLap(car) {
  const idx = getCheckpointIndex(car);
  // Checkpoint at index 0 is the start/finish
  car.checkpoints.add(idx);
  if (idx === 0 && car.checkpoints.size >= trackPoints.length * 0.7) {
    car.lap++;
    car.checkpoints.clear();
    if (car.isPlayer && car.lap >= TOTAL_LAPS) {
      car.finishTime = raceTime;
    }
  }
}

function drawTrack(camX, camY) {

  ctx.fillStyle = '#2d5a27';

  }
  ctx.closePath();
  ctx.strokeStyle = '#444';
  ctx.lineWidth = TRACK_WIDTH + 8;
  ctx.stroke();

  // Road
  ctx.beginPath();
  ctx.moveTo(trackPoints[0].x - camX, trackPoints[0].y - camY);
  for (let i = 1; i <= trackPoints.length; i++) {
    const p = trackPoints[i % trackPoints.length];
    ctx.lineTo(p.x - camX, p.y - camY);
  }
  ctx.closePath();
  ctx.strokeStyle = '#555';
  ctx.lineWidth = TRACK_WIDTH;
  ctx.stroke();

  // Center dashes
  ctx.setLineDash([20, 20]);
  ctx.beginPath();
  ctx.moveTo(trackPoints[0].x - camX, trackPoints[0].y - camY);
  for (let i = 1; i <= trackPoints.length; i++) {
    const p = trackPoints[i % trackPoints.length];
    ctx.lineTo(p.x - camX, p.y - camY);
  }
  ctx.closePath();
  ctx.strokeStyle = '#fff';
  ctx.lineWidth = 2;
  ctx.stroke();
  ctx.setLineDash([]);

  // Start/Finish line
  const sp = trackPoints[0];
  const np = trackPoints[1];
  const a = Math.atan2(np.y - sp.y, np.x - sp.x) + Math.PI/2;
  ctx.save();
  ctx.translate(sp.x - camX, sp.y - camY);
  ctx.rotate(a);
  for (let i = 0; i < TRACK_WIDTH/8; i++) {
    ctx.fillStyle = i % 2 === 0 ? '#fff' : '#000';
    ctx.fillRect(-TRACK_WIDTH/2 + i*(TRACK_WIDTH/8), -4, TRACK_WIDTH/8, 8);
  }
  ctx.restore();

  // Curbs (red/white on edges)
  for (let i = 0; i < trackPoints.length; i++) {
    const p1 = trackPoints[i];
    const p2 = trackPoints[(i+1) % trackPoints.length];
    const dx = p2.x - p1.x, dy = p2.y - p1.y;
    const len = Math.sqrt(dx*dx + dy*dy);
    const nx = -dy/len * TRACK_WIDTH/2;
    const ny = dx/len * TRACK_WIDTH/2;

    ctx.beginPath();
    ctx.moveTo(p1.x - camX + nx, p1.y - camY + ny);
    ctx.lineTo(p2.x - camX + nx, p2.y - camY + ny);
    ctx.strokeStyle = i % 2 === 0 ? '#cc0000' : '#fff';
    ctx.lineWidth = 5;
    ctx.stroke();

    ctx.beginPath();
    ctx.moveTo(p1.x - camX - nx, p1.y - camY - ny);
    ctx.lineTo(p2.x - camX - nx, p2.y - camY - ny);
    ctx.strokeStyle = i % 2 === 0 ? '#fff' : '#cc0000';
    ctx.lineWidth = 5;
    ctx.stroke();
  }
}

function getPlayerPosition() {
  let pos = 1;
  const playerProgress = getCarProgress(player);
  for (const ai of aiCars) {
    if (getCarProgress(ai) > playerProgress) pos++;
  }
  return pos;
}

function getCarProgress(car) {
  return car.lap * trackPoints.length + getCheckpointIndex(car);
}

// Main loop
let lastTime = 0;
function gameLoop(time) {
  const dt = Math.min((time - lastTime) / 16.67, 3);
  lastTime = time;

  if (gameState === 'countdown') {
    countdownTimer -= dt * 0.025;
    if (countdownTimer <= 0) gameState = 'racing';
  }

  if (gameState === 'racing' || gameState === 'finished') {
    raceTime += dt * 0.0167;
  }

  // Update player
  if (player && (gameState === 'racing')) {
    const off = !isOnTrack(player.x, player.y);
    player.update(keys, off);
    checkLap(player);
    if (player.lap >= TOTAL_LAPS && gameState === 'racing') {
      gameState = 'finished';
    }

    // Engine sound
    if (Math.abs(player.speed) > 0.5 && Math.random() < 0.3) {
      playEngine(Math.abs(player.speed));
    }
  }

  // Update AI
  if (aiCars && gameState === 'racing') {
    aiCars.forEach((ai, i) => {
      const idx = getCheckpointIndex(ai);
      const nextIdx = (idx + 1) % trackPoints.length;
      const target = trackPoints[nextIdx];
      const speedMult = [0.85, 0.9, 0.93, 0.96][i];
      ai.updateAI(target, speedMult);
      checkLap(ai);

      // Keep AI on track roughly
      if (!isOnTrack(ai.x, ai.y)) {
        const nearest = getNearestPoint(ai);
        ai.x += (nearest.x - ai.x) * 0.05;
        ai.y += (nearest.y - ai.y) * 0.05;
      }
    });
  }

  // Camera
  if (player) {
    camX = player.x - W/2;
    camY = player.y - H/2;
  }

  // Render
  ctx.clearRect(0, 0, W, H);
  drawTrack(camX, camY);

  // Draw AI cars
  if (aiCars) aiCars.forEach(ai => ai.draw(ctx, camX, camY));
  // Draw player
  if (player) player.draw(ctx, camX, camY);

  // HUD
  if (gameState === 'racing' || gameState === 'finished') {
    drawHUD();
    drawMinimap();
    drawOffTrackWarning();
  }

  if (gameState === 'countdown') drawCountdown();
  if (gameState === 'finished') drawFinish();

function getNearestPoint(car) {
  let best = trackPoints[0], bd = Infinity;
  for (const p of trackPoints) {
    const d = (car.x-p.x)**2 + (car.y-p.y)**2;
    if (d < bd) { bd = d; best = p; }
