const canvas = document.getElementById('game');
const ctx = canvas.getContext('2d');
const overlay = document.getElementById('overlay');
const hud = document.getElementById('hud');
const scoreEl = document.getElementById('score');
const comboEl = document.getElementById('combo');
const startBtn = document.getElementById('startBtn');

let W, H, cellSize, gridX, gridY, cols = 8, rows = 8;
let gems = [];
let selected = null;
let animating = false;
let score = 0;
let comboCount = 1;
let audioCtx = null;
let particles = [];
let gemAnims = [];

const COLORS = [
{ fill: '#e74c3c', glow: 'rgba(231,76,60,0.5)' },
{ fill: '#3498db', glow: 'rgba(52,152,219,0.5)' },
{ fill: '#2ecc71', glow: 'rgba(46,204,113,0.5)' },
{ fill: '#f1c40f', glow: 'rgba(241,196,15,0.5)' },
{ fill: '#9b59b6', glow: 'rgba(155,89,182,0.5)' },
{ fill: '#e67e22', glow: 'rgba(230,126,34,0.5)' }
];

function resize() {
W = window.innerWidth; H = window.innerHeight;
canvas.width = W; canvas.height = H;
cellSize = Math.min((W - 60) / cols, (H - 120) / rows);
gridX = (W - cols * cellSize) / 2;
gridY = (H - rows * cellSize) / 2 + 30;
}
window.addEventListener('resize', resize);

function initAudio() {
if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
}

function playSound(type) {
if (!audioCtx) return;
const osc = audioCtx.createOscillator();
const gain = audioCtx.createGain();
osc.connect(gain);
gain.connect(audioCtx.destination);
if (type === 'swap') {
osc.type = 'sine'; osc.frequency.setValueAtTime(400, audioCtx.currentTime);
osc.frequency.linearRampToValueAtTime(600, audioCtx.currentTime + 0.1);
gain.gain.setValueAtTime(0.2, audioCtx.currentTime);
gain.gain.linearRampToValueAtTime(0, audioCtx.currentTime + 0.15);
osc.start(); osc.stop(audioCtx.currentTime + 0.15);
} else if (type === 'match') {
osc.type = 'triangle'; osc.frequency.setValueAtTime(500, audioCtx.currentTime);
osc.frequency.linearRampToValueAtTime(800, audioCtx.currentTime + 0.2);
gain.gain.setValueAtTime(0.3, audioCtx.currentTime);
gain.gain.linearRampToValueAtTime(0, audioCtx.currentTime + 0.25);
osc.start(); osc.stop(audioCtx.currentTime + 0.25);
} else if (type === 'cascade') {
osc.type = 'sine'; osc.frequency.setValueAtTime(600, audioCtx.currentTime);
osc.frequency.linearRampToValueAtTime(1200, audioCtx.currentTime + 0.3);
gain.gain.setValueAtTime(0.25, audioCtx.currentTime);
gain.gain.linearRampToValueAtTime(0, audioCtx.currentTime + 0.3);
osc.start(); osc.stop(audioCtx.currentTime + 0.3);
} else if (type === 'invalid') {
osc.type = 'sawtooth'; osc.frequency.setValueAtTime(200, audioCtx.currentTime);
gain.gain.setValueAtTime(0.15, audioCtx.currentTime);
gain.gain.linearRampToValueAtTime(0, audioCtx.currentTime + 0.1);
osc.start(); osc.stop(audioCtx.currentTime + 0.1);
}
}

function randomGem() { return Math.floor(Math.random() * COLORS.length); }

function initGrid() {
gems = [];
for (let r = 0; r < rows; r++) {
gems[r] = [];
for (let c = 0; c < cols; c++) {
let g;
do { g = randomGem(); } while (wouldMatch(r, c, g));
gems[r][c] = g;
}
}
}

function wouldMatch(r, c, color) {
if (c >= 2 && gems[r] && gems[r][c-1] === color && gems[r][c-2] === color) return true;
if (r >= 2 && gems[r-1] && gems[r-1][c] === color && gems[r-2] && gems[r-2][c] === color) return true;
return false;
}

function findMatches() {
let matches = new Set();
// Horizontal
for (let r = 0; r < rows; r++) {
for (let c = 0; c < cols - 2; c++) {
if (gems[r][c] !== null && gems[r][c] === gems[r][c+1] && gems[r][c] === gems[r][c+2]) {
matches.add(r + ',' + c); matches.add(r + ',' + (c+1)); matches.add(r + ',' + (c+2));
}
}
}
// Vertical
for (let r = 0; r < rows - 2; r++) {
for (let c = 0; c < cols; c++) {
if (gems[r][c] !== null && gems[r][c] === gems[r+1][c] && gems[r][c] === gems[r+2][c]) {
matches.add(r + ',' + c); matches.add((r+1) + ',' + c); matches.add((r+2) + ',' + c);
}
}
}
return [...matches].map(s => s.split(',').map(Number));
}

function applyGravity() {
let moved = false;
for (let c = 0; c < cols; c++) {
for (let r = rows - 1; r >= 0; r--) {
if (gems[r][c] === null) {
for (let above = r - 1; above >= 0; above--) {
if (gems[above][c] !== null) {
let below = above + 1;
while (below < rows && gems[below][c] === null) below++;
gems[below][c] = gems[above][c];
gems[above][c] = null;
moved = true;
break;
}
}
}
}
// Fill from top
for (let c = 0; c < cols; c++) {
for (let r = rows - 1; r >= 0; r--) {
if (gems[r][c] === null) {
let emptyCount = 0;
for (let r2 = r; r2 >= 0; r2--) if (gems[r2][c] === null) emptyCount++;
gems[r][c] = randomGem();
moved = true;
}
}
}
}
return moved;
}

function spawnParticles(r, c, colorIdx) {
const cx = gridX + c * cellSize + cellSize / 2;
const cy = gridY + r * cellSize + cellSize / 2;
for (let i = 0; i < 6; i++) {
const angle = Math.random() * Math.PI * 2;
const speed = 2 + Math.random() * 4;
particles.push({ x: cx, y: cy, vx: Math.cos(angle) * speed, vy: Math.sin(angle) * speed - 2, life: 1, color: COLORS[colorIdx].fill });
}
}

function drawGem(x, y, size, colorIdx, scale = 1) {
const s = size * 0.85 * scale;
const cx = x + size/2, cy = y + size/2;
ctx.save();
ctx.translate(cx, cy);
ctx.shadowColor = COLORS[colorIdx].glow;
ctx.shadowBlur = 10 * scale;
ctx.fillStyle = COLORS[colorIdx].fill;

// Draw gem shape (rounded diamond)
const r2 = s / 2;
ctx.beginPath();
ctx.moveTo(0, -r2);
ctx.quadraticCurveTo(r2, -r2, r2, 0);
ctx.quadraticCurveTo(r2, r2, 0, r2);
ctx.quadraticCurveTo(-r2, r2, -r2, 0);
ctx.quadraticCurveTo(-r2, -r2, 0, -r2);
ctx.closePath();
ctx.fill();

// Highlight
ctx.fillStyle = 'rgba(255,255,255,0.3)';
ctx.beginPath();
ctx.arc(-s*0.15, -s*0.15, s*0.2, 0, Math.PI*2);
ctx.fill();
ctx.restore();
}

function render() {
ctx.clearRect(0, 0, W, H);

// Background grid
for (let r = 0; r < rows; r++) {
for (let c = 0; c < cols; c++) {
const x = gridX + c * cellSize;
const y = gridY + r * cellSize;
ctx.fillStyle = (r + c) % 2 === 0 ? 'rgba(255,255,255,0.03)' : 'rgba(255,255,255,0.06)';
ctx.fillRect(x, y, cellSize, cellSize);
}
}

// Grid border
ctx.strokeStyle = 'rgba(255,255,255,0.1)';
ctx.lineWidth = 2;
ctx.strokeRect(gridX, gridY, cols * cellSize, rows * cellSize);

// Gems
for (let r = 0; r < rows; r++) {
for (let c = 0; c < cols; c++) {
if (gems[r][c] !== null) {
const x = gridX + c * cellSize;
const y = gridY + r * cellSize;
let scale = 1;
if (selected && selected.r === r && selected.c === c) scale = 1.15;
drawGem(x, y, cellSize, gems[r][c], scale);
}
}
}

// Selection highlight
if (selected) {
ctx.strokeStyle = '#fff';
ctx.lineWidth = 3;
ctx.shadowColor = '#fff';
ctx.shadowBlur = 8;
ctx.strokeRect(gridX + selected.c * cellSize + 2, gridY + selected.r * cellSize + 2, cellSize - 4, cellSize - 4);
ctx.shadowBlur = 0;
}

// Particles
for (let i = particles.length - 1; i >= 0; i--) {
const p = particles[i];
p.x += p.vx;
p.y += p.vy;
p.vy += 0.3;
p.life -= 0.025;
if (p.life <= 0) { particles.splice(i, 1); continue; }
ctx.globalAlpha = p.life;
ctx.fillStyle = p.color;
ctx.beginPath();
ctx.arc(p.x, p.y, 4 * p.life, 0, Math.PI * 2);
ctx.fill();
ctx.globalAlpha = 1;
}

requestAnimationFrame(render);
}

function swap(r1, c1, r2, c2) {
let temp = gems[r1][c1];
gems[r1][c1] = gems[r2][c2];
gems[r2][c2] = temp;
}

async function resolveBoard() {
comboCount = 1;
let hasMatches = true;
while (hasMatches) {
const matches = findMatches();
if (matches.length === 0) { hasMatches = false; break; }

const points = matches.length * 10 * comboCount;
score += points;
scoreEl.textContent = score;
comboEl.textContent = 'x' + comboCount;

playSound(comboCount > 1 ? 'cascade' : 'match');

for (const [r, c] of matches) {
spawnParticles(r, c, gems[r][c]);
gems[r][c] = null;
}

await delay(300);
applyGravity();
comboCount++;
if (comboCount > 1) playSound('cascade');
await delay(200);
}
animating = false;
}

function delay(ms) { return new Promise(res => setTimeout(res, ms)); }

async function trySwap(r1, c1, r2, c2) {
animating = true;
swap(r1, c1, r2, c2);
playSound('swap');
await delay(200);

const matches = findMatches();
if (matches.length === 0) {
// Swap back
swap(r1, c1, r2, c2);
playSound('invalid');
await delay(200);
animating = false;
} else {
await resolveBoard();
}
selected = null;
}

function getCellFromPos(x, y) {
const c = Math.floor((x - gridX) / cellSize);
const r = Math.floor((y - gridY) / cellSize);
if (r >= 0 && r < rows && c >= 0 && c < cols) return { r, c };
return null;
}

canvas.addEventListener('click', (e) => {
initAudio();
if (animating || gems.length === 0) return;
const cell = getCellFromPos(e.clientX, e.clientY);
if (!cell) { selected = null; return; }

if (!selected) {
selected = cell;
} else {
const dr = Math.abs(cell.r - selected.r);
const dc = Math.abs(cell.c - selected.c);
if ((dr === 1 && dc === 0) || (dr === 0 && dc === 1)) {
trySwap(selected.r, selected.c, cell.r, cell.c);
} else {
selected = cell;
}
}
});

function startGame() {
initAudio();
score = 0;
comboCount = 1;
particles = [];
selected = null;
animating = false;
scoreEl.textContent = '0';
comboEl.textContent = 'x1';
initGrid();
hud.style.display = 'flex';
overlay.classList.add('hidden');
}

startBtn.addEventListener('click', startGame);
window.addEventListener('keydown', (e) => {
if (e.key === 'r' || e.key === 'R') {
if (gems.length > 0) startGame();
}
if (e.key === 'Enter' && gems.length === 0) startGame();
});

resize();
requestAnimationFrame(render);
</script>
</body>
