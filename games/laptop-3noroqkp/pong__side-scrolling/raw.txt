const canvas = document.getElementById('game');
const ctx = canvas.getContext('2d');

let W, H;
function resize() {
    W = canvas.width = window.innerWidth;
    H = canvas.height = window.innerHeight;
}
window.addEventListener('resize', resize);

// Audio
let audioCtx = null;
function initAudio() {
    if (!audioCtx) audioCtx = new (window.AudioContext || webkitAudioContext)();
}
function playSound(freq, dur, type, vol) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = type || 'square';
    osc.frequency.value = freq;
    gain.gain.setValueAtTime(vol || 0.15, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + dur);
}

// Game state
let state = 'start'; // start, playing, gameover
let scorePlayer = 0;
let scoreAI = 0;
const WIN_SCORE = 11;

// Paddles
let paddlePlayer = { x: 0, w: 120, h: 14 };
let paddleAI = { x: 0, w: 120, h: 14 };
let ball = { x: 0, y: 0, vx: 0, vy: 0, r: 8, speed: 7 };

// Parallax layers
let parallaxLayers = [];
function initParallax() {
    parallaxLayers = [
        { stars: [], speed: 0.3, size: 1 },
        { stars: [], speed: 0.8, size: 2 },
        { stars: [], speed: 1.5, size: 3 }
    ];
    for (let l = 0; l < 3; l++) {
        const count = 30 + l * 20;
        for (let i = 0; i < count; i++) {
            parallaxLayers[l].stars.push({
                x: Math.random() * W,
                y: Math.random() * H,
                twinkle: Math.random() * Math.PI * 2
            });
        }
    }
}
initParallax();

// Input
let keys = {};
window.addEventListener('keydown', e => {
    keys[e.code] = true;
    if (state === 'start' && (e.code === 'Space' || e.code === 'Enter')) {
        initAudio();
        startGame();
    }
    if (state === 'gameover' && (e.code === 'Space' || e.code === 'Enter' || e.code === 'KeyR')) {
        startGame();
    }

window.addEventListener('keyup', e => { keys[e.code] = false; });

function startGame() {
    state = 'playing';
    scorePlayer = 0;
    scoreAI = 0;
    resetBall();
}

function resetBall() {
    ball.x = W / 2;
    ball.y = H / 2;
    const angle = (Math.random() - 0.5) * Math.PI / 3;
    const dir = Math.random() < 0.5 ? 1 : -1;
    ball.vx = Math.cos(angle) * ball.speed * dir;
    ball.vy = Math.sin(angle) * ball.speed;
    paddlePlayer.x = W / 2 - paddlePlayer.w / 2;
    paddleAI.x = W / 2 - paddleAI.w / 2;
}

function serveBall(direction) {
    ball.x = W / 2;
    ball.y = H / 2;
    const angle = (Math.random() - 0.5) * Math.PI / 4;
    ball.vx = Math.cos(angle) * ball.speed * direction;
    ball.vy = Math.sin(angle) * ball.speed;
}

function update() {
    if (state !== 'playing') return;

    // Player paddle
    const pSpeed = 8;
    if (keys['ArrowLeft'] || keys['KeyA']) paddlePlayer.x -= pSpeed;
    if (keys['ArrowRight'] || keys['KeyD']) paddlePlayer.x += pSpeed;
    paddlePlayer.x = Math.max(0, Math.min(W - paddlePlayer.w, paddlePlayer.x));

    // AI paddle
    const aiTarget = ball.x;
    const aiSpeed = 5.5 + scoreAI * 0.2;
    const aiCenter = paddleAI.x + paddleAI.w / 2;
    const diff = aiTarget - aiCenter;
    if (Math.abs(diff) > 10) {
        paddleAI.x += Math.sign(diff) * Math.min(aiSpeed, Math.abs(diff));
    }
    paddleAI.x = Math.max(0, Math.min(W - paddleAI.w, paddleAI.x));

    // Ball movement
    ball.x += ball.vx;
    ball.y += ball.vy;

    // Bounce off left/right walls
    if (ball.x - ball.r < 0) { ball.x = ball.r; ball.vx = Math.abs(ball.vx); playSound(220, 0.1, 'square', 0.1); }
    if (ball.x + ball.r > W) { ball.x = W - ball.r; ball.vx = -Math.abs(ball.vx); playSound(220, 0.1, 'square', 0.1); }

    // Paddle collision - top paddle (AI)
    const topY = 30;
    if (ball.vy < 0 && ball.y - ball.r < topY + paddleAI.h && ball.y - ball.r > topY - 10) {
        if (ball.x > paddleAI.x && ball.x < paddleAI.x + paddleAI.w) {
            ball.y = topY + paddleAI.h + ball.r;
            ball.vy = Math.abs(ball.vy);
            // Add spin based on hit position
            const hitPos = (ball.x - (paddleAI.x + paddleAI.w / 2)) / (paddleAI.w / 2);
            ball.vx += hitPos * 3;
            playSound(440, 0.15, 'square', 0.2);
        }
    }

    // Paddle collision - bottom paddle (Player)
    const botY = H - 30;
    if (ball.vy > 0 && ball.y + ball.r > botY - paddlePlayer.h && ball.y + ball.r < botY + 10) {
        if (ball.x > paddlePlayer.x && ball.x < paddlePlayer.x + paddlePlayer.w) {
            ball.y = botY - paddlePlayer.h - ball.r;
            ball.vy = -Math.abs(ball.vy);
            const hitPos = (ball.x - (paddlePlayer.x + paddlePlayer.w / 2)) / (paddlePlayer.w / 2);
            ball.vx += hitPos * 3;
            playSound(440, 0.15, 'square', 0.2);
        }
    }

    // Score detection
    if (ball.y - ball.r < 0) {
        scorePlayer++;
        playSound(880, 0.3, 'sine', 0.3);
        if (scorePlayer >= WIN_SCORE) { state = 'gameover'; }
        else serveBall(-1);
    }
    if (ball.y + ball.r > H) {
        scoreAI++;
        playSound(220, 0.4, 'sawtooth', 0.3);
        if (scoreAI >= WIN_SCORE) { state = 'gameover'; }

    // Speed increase
    ball.speed += 0.01;
}

// Particles
let particles = [];
function spawnParticles(x, y, color, count) {
    for (let i = 0; i < count; i++) {
        const angle = Math.random() * Math.PI * 2;
        const speed = 1 + Math.random() * 4;
        particles.push({
            x, y,
            vx: Math.cos(angle) * speed,
            vy: Math.sin(angle) * speed,
            life: 1,
            color
        });
    }
}

function updateParticles() {
    for (let i = particles.length - 1; i >= 0; i--) {
        const p = particles[i];
        p.x += p.vx;
        p.y += p.vy;
        p.life -= 0.02;
        if (p.life <= 0) particles.splice(i, 1);
    }
}

function updateParallax() {
    for (let l = 0; l < parallaxLayers.length; l++) {
        const layer = parallaxLayers[l];
        for (const star of layer.stars) {
            star.x -= layer.speed;
            if (star.x < -5) { star.x = W + 5; star.y = Math.random() * H; }
            star.twinkle += 0.02;
        }
    }
}

function draw() {
    ctx.fillStyle = '#0a0a1a';
    ctx.fillRect(0, 0, W, H);

    // Parallax stars
    for (let l = 0; l < parallaxLayers.length; l++) {
        const layer = parallaxLayers[l];
        for (const star of layer.stars) {
            const alpha = 0.3 + 0.4 * Math.sin(star.twinkle);
            ctx.fillStyle = `rgba(150, 200, 255, ${alpha})`;
            ctx.beginPath();
            ctx.arc(star.x, star.y, layer.size * 0.5, 0, Math.PI * 2);
            ctx.fill();
        }
    }

    // Grid lines (side-scrolling effect)
    ctx.strokeStyle = 'rgba(50, 80, 140, 0.15)';
    ctx.lineWidth = 1;
    const gridOffset = (Date.now() * 0.05) % 80;
    for (let x = -gridOffset; x < W + 80; x += 80) {
        ctx.beginPath();
        ctx.moveTo(x, 0);
        ctx.lineTo(x, H);
        ctx.stroke();
    }
    for (let y = 0; y < H; y += 80) {
        ctx.beginPath();
        ctx.moveTo(0, y);
        ctx.lineTo(W, y);
        ctx.stroke();
    }

    if (state === 'start') {
        drawStartScreen();
    } else if (state === 'playing' || state === 'gameover') {
        // Center line
        ctx.strokeStyle = 'rgba(100, 180, 255, 0.3)';
        ctx.lineWidth = 2;
        ctx.setLineDash([10, 15]);
        ctx.beginPath();
        ctx.moveTo(W / 2, 0);
        ctx.lineTo(W / 2, H);
        ctx.stroke();
        ctx.setLineDash([]);

        // AI Paddle (top)
        ctx.shadowBlur = 15;
        ctx.shadowColor = '#ff4466';
        ctx.fillStyle = '#ff4466';
        roundRect(paddleAI.x, 30, paddleAI.w, paddleAI.h, 7);

        // Player Paddle (bottom)
        ctx.shadowColor = '#44aaff';
        ctx.fillStyle = '#44aaff';
        roundRect(paddlePlayer.x, H - 30 - paddlePlayer.h, paddlePlayer.w, paddlePlayer.h, 7);
        ctx.shadowBlur = 0;

        // Ball
        if (state === 'playing') {
            ctx.shadowBlur = 20;
            ctx.shadowColor = '#ffffff';
            ctx.fillStyle = '#ffffff';
            ctx.beginPath();
            ctx.arc(ball.x, ball.y, ball.r, 0, Math.PI * 2);
            ctx.fill();
            // Trail
            for (let i = 1; i <= 5; i++) {
                const alpha = 0.3 / i;
                ctx.fillStyle = `rgba(255, 255, 255, ${alpha})`;
                ctx.beginPath();
                ctx.arc(ball.x - ball.vx * i, ball.y - ball.vy * i, ball.r * (1 - i * 0.12), 0, Math.PI * 2);
                ctx.fill();
            }
            ctx.shadowBlur = 0;
        }

        // Particles
        for (const p of particles) {
            ctx.fillStyle = `rgba(${p.color}, ${p.life})`;
            ctx.beginPath();
            ctx.arc(p.x, p.y, 3 * p.life, 0, Math.PI * 2);
            ctx.fill();
        }

        // Score
        ctx.font = 'bold 48px monospace';
        ctx.textAlign = 'center';
        ctx.fillStyle = '#ff4466';
        ctx.fillText(scoreAI, W / 2, 80);
        ctx.fillStyle = '#44aaff';
        ctx.fillText(scorePlayer, W / 2, H - 50);

        if (state === 'gameover') {
            drawGameOver();
        }
    }
}

function roundRect(x, y, w, h, r) {
    ctx.beginPath();
    ctx.moveTo(x + r, y);
    ctx.lineTo(x + w - r, y);
    ctx.arcTo(x + w, y, x + w, y + h, r);
    ctx.lineTo(x + w, y + h);
    ctx.arcTo(x, w, y + h, x, y + h, r);
    ctx.lineTo(x, y + h);
    ctx.arcTo(x, y + h, x, y, r);
    ctx.lineTo(x, y);
    ctx.arcTo(x, y, x + r, y, r);
    ctx.closePath();
    ctx.fill();
}
