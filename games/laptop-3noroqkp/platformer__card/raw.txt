const canvas = document.getElementById('game');

let W, H;

// Audio
let audioCtx = null;
function initAudio() {
    if (!audioCtx) audioCtx = new (window.AudioContext || webkitAudioContext)();
}
function playSound(type, freq, dur) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = type;
    osc.frequency.value = freq;
    gain.gain.setValueAtTime(0.15, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + dur);
}

// Game state
let gameState = 'START'; // START, PLAYING, WIN, LOSE
let score = 0;
let bestScore = 0;
try { bestScore = parseInt(localStorage.getItem('cardPlatformerBest')) || 0; } catch(e) {}

// Input
const keys = {};
window.addEventListener('keydown', e => {
    keys[e.code] = true;
    if (gameState === 'START') { initAudio(); startGame(); }
    else if (gameState === 'WIN' || gameState === 'LOSE') { if (e.code === 'KeyR') startGame(); }
});
window.addEventListener('keyup', e => { keys[e.code] = false; });

// Particles
let particles = [];
function spawnParticles(x, y, color, count, speed) {
    for (let i = 0; i < count; i++) {
        const angle = Math.random() * Math.PI * 2;
        const spd = Math.random() * speed;
        particles.push({ x, y, vx: Math.cos(angle)*spd, vy: Math.sin(angle)*spd - 2, life: 1, color, size: Math.random()*4+2 });
    }
}

// Card definitions
const CARD_TYPES = [
    { name: 'JUMP', color: '#4fc3f7', icon: '↑', desc: 'Mega Jump' },
    { name: 'DASH', color: '#ff9800', icon: '→', desc: 'Quick Dash' },
    { name: 'FIRE', color: '#f44336', icon: '●', desc: 'Fireball' },
    { name: 'SHIELD', color: '#66bb6a', icon: '◇', desc: 'Shield 3s' },
    { name: 'SPRING', color: '#ab47bc', icon: '∿', desc: 'Spring Step' },
    { name: 'BOLT', color: '#fff176', icon: '⚡', desc: 'Lightning' }
];

// Level
let platforms = [];
let enemies = [];
let projectiles = [];
let flag = null;

function buildLevel() {
    platforms = [];
    enemies = [];
    projectiles = [];
    
    const ph = 20;
    // Ground
    platforms.push({ x: 0, y: H - 60, w: W * 0.3, h: ph });
    platforms.push({ x: W * 0.4, y: H - 60, w: W * 0.3, h: ph });
    platforms.push({ x: W * 0.75, y: H - 60, w: W * 0.25, h: ph });
    
    // Floating platforms
    const spacing = Math.min(W, 800) / 12;
    for (let i = 0; i < 8; i++) {
        const px = i * spacing + Math.random() * 40 - 20;
        const py = H - 60 - (80 + Math.random() * 120);
        const pw = 80 + Math.random() * 80;
        platforms.push({ x: px, y: py, w: pw, h: ph });
    }
    
    // Enemies on some platforms
    for (let i = 2; i < platforms.length; i += 2) {
        const p = platforms[i];
        enemies.push({
            x: p.x + p.w/2 - 15, y: p.y - 30, w: 30, h: 30,
            vx: 1.5, dir: 1, hp: 1, alive: true,
            minX: p.x, maxX: p.x + p.w - 30, type: Math.floor(Math.random()*2)
        });
    }
    
    // Flag at end
    flag = { x: W - 80, y: H - 60 - 60, w: 20, h: 60 };
}

// Player
let player;
function resetPlayer() {
    player = {
        x: 50, y: H - 100, w: 30, h: 40,
        vx: 0, vy: 0, onGround: false,
        jumps: 0, maxJumps: 2,
        springCharges: 0, shieldTimer: 0, dashTimer: 0, dashDir: 0,
        invincible: 0, hurtTimer: 0
    };
}

// Card system
let deck = [], hand = [], discard = [];

function buildDeck() {
    deck = [];
    for (let i = 0; i < 12; i++) {
        deck.push({ type: Math.floor(Math.random() * CARD_TYPES.length), used: false });
    }
    // Ensure at least one of each in deck start
    for (let i = 0; i < CARD_TYPES.length; i++) {
        if (!deck.some(c => c.type === i)) deck[i] = { type: i, used: false };
    }
    shuffle(deck);
    hand = [];
    discard = [];
    drawCards(4);
}

function shuffle(arr) {
    for (let i = arr.length - 1; i > 0; i--) {
        const j = Math.floor(Math.random() * (i + 1));
        [arr[i], arr[j]] = [arr[j], arr[i]];
    }
}

function drawCards(n) {
    while (n > 0 && hand.length < 4) {
        if (deck.length === 0) {
            deck = [...discard];
            discard = [];
            shuffle(deck);
        }
        if (deck.length > 0) {
            hand.push(deck.pop());
            n--;
        } else break;
    }
}

function playCard(index) {
    if (index < 0 || index >= hand.length) return;
    const card = hand[index];
    const type = card.type;
    
    switch(type) {
        case 0: // JUMP
            player.vy = -18;
            player.jumps = 2;
            spawnParticles(player.x + player.w/2, player.y + player.h, '#4fc3f7', 15, 5);
            playSound('sine', 400, 0.15);
            break;
        case 1: // DASH
            const dir = keys['ArrowLeft'] || keys['KeyA'] ? -1 : 1;
            player.dashTimer = 12;
            player.dashDir = dir;
            spawnParticles(player.x + player.w/2, player.y + player.h/2, '#ff9800', 10, 4);
            playSound('square', 200, 0.1);
            break;
        case 2: // FIRE
            const fDir = keys['ArrowLeft'] || keys['KeyA'] ? -1 : 1;
            projectiles.push({ x: player.x + player.w/2, y: player.y + player.h/2, vx: fDir * 10, vy: 0, w: 16, h: 16, life: 80 });
            spawnParticles(player.x + player.w/2, player.y + player.h/2, '#f44336', 5, 3);
            playSound('sawtooth', 150, 0.2);
            break;
        case 3: // SHIELD
            player.shieldTimer = 180;
            spawnParticles(player.x + player.w/2, player.y + player.h/2, '#66bb6a', 12, 4);
            playSound('sine', 600, 0.3);
            break;
        case 4: // SPRING
            player.springCharges = 5;
            spawnParticles(player.x + player.w/2, player.y + player.h, '#ab47bc', 8, 3);
            playSound('triangle', 300, 0.15);
            break;
        case 5: // BOLT
            for (let e of enemies) {
                if (e.alive && Math.abs(e.x - player.x) < 200) {
                    e.hp -= 1;
                    spawnParticles(e.x + e.w/2, e.y + e.h/2, '#fff176', 20, 6);
                    playSound('square', 800, 0.3);
                }
            }
            // Visual bolt
            const boltTarget = enemies.find(e => e.alive && Math.abs(e.x - player.x) < 200);
            if (boltTarget) {
                for (let t = 0; t < 10; t++) {
                    particles.push({ x: boltTarget.x + boltTarget.w/2, y: boltTarget.y + boltTarget.h/2 - 50 + t*6, vx: (Math.random()-0.5)*3, vy: 0, life: 1, color: '#fff176', size: 3 });
                }
            }
            break;
    }
    
    hand.splice(index, 1);
    discard.push(card);
    playSound('triangle', 250 + type * 50, 0.1);
}

// Game functions
function startGame() {
    gameState = 'PLAYING';
    score = 0;
    resetPlayer();
    buildLevel();
    buildDeck();
    particles = [];
}

function update() {
    if (gameState !== 'PLAYING') return;
    
    // Player movement
    const moveSpeed = 4;
    if (player.dashTimer > 0) {
        player.vx = player.dashDir * 12;
        player.dashTimer--;
    } else {
        if (keys['ArrowLeft'] || keys['KeyA']) player.vx = -moveSpeed;
        else if (keys['ArrowRight'] || keys['KeyD']) player.vx = moveSpeed;
        else player.vx *= 0.8;
    }
    
    // Gravity
    player.vy += 0.6;
    
    // Move
    player.x += player.vx;
    player.y += player.vy;
    
    // Platform collision
    player.onGround = false;
    for (const p of platforms) {
        if (player.x + player.w > p.x && player.x < p.x + p.w) {
            if (player.vy >= 0 && player.y + player.h >= p.y && player.y + player.h - player.vy <= p.y + 5) {
                player.y = p.y - player.h;
                player.vy = 0;
                player.onGround = true;
                player.jumps = 2;
                
                if (player.springCharges > 0) {
                    player.vy = -14;
                    player.springCharges--;
                    spawnParticles(player.x + player.w/2, player.y + player.h, '#ab47bc', 8, 4);
                    playSound('sine', 500, 0.1);
                }
            }
        }
    }
    
    // Bounds
    if (player.x < 0) player.x = 0;
    if (player.x + player.w > W) player.x = W - player.w;
    if (player.y > H + 50) {
        // Fell off
        player.hurtTimer++;
        if (player.shieldTimer > 0) {
            player.shieldTimer--;
            player.y = H - 120;
            player.vy = 0;
            player.x = 50;
        } else {
            gameState = 'LOSE';
            saveScore();
        }
    }
    
    // Timers
    if (player.shieldTimer > 0) player.shieldTimer--;
    if (player.invincible > 0) player.invincible--;
    if (player.hurtTimer > 0) player.hurtTimer--;
    
    // Enemies
    for (const e of enemies) {
        if (!e.alive) continue;
        e.x += e.vx * e.dir;
        if (e.x <= e.minX || e.x >= e.maxX) e.dir *= -1;
        
        // Enemy-player collision
        if (player.invincible <= 0 && player.shieldTimer <= 0 &&
            player.x + player.w > e.x && player.x < e.x + e.w &&
            player.y + player.h > e.y && player.y < e.y + e.h) {
            player.vy = -8;
            player.vx = (player.x < e.x) ? -5 : 5;
            player.invincible = 60;
            spawnParticles(player.x + player.w/2, player.y + player.h/2, '#ff5722', 10, 4);
            playSound('sawtooth', 100, 0.3);
            
            if (player.shieldTimer > 0) { /* blocked */ }
            else {
                player.hurtTimer++;
                if (player.hurtTimer >= 3) { gameState = 'LOSE'; saveScore(); }
            }
        }
    }
    
    // Projectiles
    for (let i = projectiles.length - 1; i >= 0; i--) {
        const p = projectiles[i];
        p.x += p.vx;
        p.life--;
        if (p.life <= 0 || p.x < 0 || p.x > W) { projectiles.splice(i, 1); continue; }
        
        for (const e of enemies) {
            if (!e.alive) continue;
            if (p.x + p.w > e.x && p.x < e.x + e.w && p.y + p.h > e.y && p.y < e.y + e.h) {
                e.hp--;
                spawnParticles(e.x + e.w/2, e.y + e.h/2, '#f44336', 15, 5);
                playSound('square', 100, 0.2);
                if (e.hp <= 0) { e.alive = false; score += 100; }
                projectiles.splice(i, 1);
            }
        }
    }
    
    // Flag check
    if (flag && player.x + player.w > flag.x && player.x < flag.x + flag.w &&
        player.y + player.h > flag.y && player.y < flag.y + flag.h) {
        score += 500;
        gameState = 'WIN';
        saveScore();
        playSound('sine', 800, 0.5);
    }
    
    // Particles
    for (let i = particles.length - 1; i >= 0; i--) {
        const p = particles[i];
        p.x += p.vx;
        p.y += p.vy;
        p.vy += 0.2;
        p.life -= 0.02;
        if (p.life <= 0) particles.splice(i, 1);
    }
}

function saveScore() {
    if (score > bestScore) {
        bestScore = score;
        try { localStorage.setItem('cardPlatformerBest', String(bestScore)); } catch(e) {}
    }
}

// Rendering
function draw() {
    // Background
    const grad = ctx.createLinearGradient(0, 0, 0, H);
    grad.addColorStop(0, '#16213e');
    grad.addColorStop(1, '#0f3460');
    ctx.fillStyle = grad;
    ctx.fillRect(0, 0, W, H);
    
    // Stars
    for (let i = 0; i < 50; i++) {
        const sx = (i * 137.5) % W;
        const sy = (i * 97.3) % (H * 0.6);
        ctx.fillStyle = `rgba(255,255,255,${0.3 + Math.sin(Date.now()/1000 + i) * 0.2})`;
        ctx.fillRect(sx, sy, 2, 2);
    }
    
    if (gameState === 'START') {
        drawStartScreen();
        return;
    }
    
    // Platforms
    for (const p of platforms) {
        ctx.fillStyle = '#4a90d9';
        ctx.fillRect(p.x, p.y, p.w, p.h);
        ctx.fillStyle = '#6ab0ff';
        ctx.fillRect(p.x, p.y, p.w, 4);
    }
    
    // Flag
    if (flag) {
        ctx.fillStyle = '#ffd700';
        ctx.fillRect(flag.x + 5, flag.y, 4, flag.h);
        ctx.fillStyle = '#ff6b6b';
        const wave = Math.sin(Date.now()/300) * 5;
        ctx.beginPath();
        ctx.moveTo(flag.x + 9, flag.y);
        ctx.lineTo(flag.x + 35 + wave, flag.y + 10);
        ctx.lineTo(flag.x + 9, flag.y + 20);
        ctx.fill();
    }
    
    // Enemies
    for (const e of enemies) {
        if (!e.alive) continue;
        const ec = e.type === 0 ? '#ff5722' : '#9c27b0';
        ctx.fillStyle = ec;
        ctx.fillRect(e.x, e.y, e.w, e.h);
        // Eyes
        ctx.fillStyle = '#fff';
        ctx.fillRect(e.x + 5, e.y + 8, 6, 6);
        ctx.fillRect(e.x + 19, e.y + 8, 6, 6);
        ctx.fillStyle = '#000';
        ctx.fillRect(e.x + 7, e.y + 10, 3, 3);
        ctx.fillRect(e.x + 21, e.y + 10, 3, 3);
    }
    
    // Projectiles
    for (const p of projectiles) {
        ctx.fillStyle = '#ff5722';
        ctx.beginPath();
        ctx.arc(p.x + p.w/2, p.y + p.h/2, p.w/2, 0, Math.PI*2);
        ctx.fill();
        ctx.fillStyle = '#ffeb3b';
        ctx.beginPath();
        ctx.arc(p.x + p.w/2, p.y + p.h/2, p.w/4, 0, Math.PI*2);
        ctx.fill();
    }
    
    // Player
    if (player.invincible <= 0 || Math.floor(Date.now()/100) % 2 === 0) {
        ctx.fillStyle = player.shieldTimer > 0 ? '#66bb6a' : '#e94560';
        ctx.fillRect(player.x, player.y, player.w, player.h);
        // Face
        ctx.fillStyle = '#fff';
        ctx.fillRect(player.x + 5, player.y + 8, 7, 7);
        ctx.fillRect(player.x + 18, player.y + 8, 7, 7);
        ctx.fillStyle = '#000';
        ctx.fillRect(player.x + 7, player.y + 10, 4, 4);
        ctx.fillRect(player.x + 20, player.y + 10, 4, 4);
        
        // Shield glow
        if (player.shieldTimer > 0) {
            ctx.strokeStyle = `rgba(102,187,106,${0.3 + Math.sin(Date.now()/200)*0.2})`;
            ctx.lineWidth = 3;
            ctx.beginPath();
            ctx.arc(player.x + player.w/2, player.y + player.h/2, 25, 0, Math.PI*2);
            ctx.stroke();
        }
    }
    
    // Particles
    for (const p of particles) {
        ctx.globalAlpha = p.life;
        ctx.fillStyle = p.color;
        ctx.fillRect(p.x, p.y, p.size, p.size);
    }
    ctx.globalAlpha = 1;
    
    // Card hand
    drawCards();
    
    // HUD
    ctx.fillStyle = 'rgba(0,0,0,0.5)';
    ctx.fillRect(10, 10, 200, 80);
    ctx.fillStyle = '#fff';
    ctx.font = '14px monospace';
    ctx.fillText(`Score: ${score}`, 20, 30);
    ctx.fillText(`Best: ${bestScore}`, 20, 50);
    ctx.fillText(`Deck: ${deck.length} | Discard: ${discard.length}`, 20, 70);
    
    if (gameState === 'WIN' || gameState === 'LOSE') {
        ctx.fillStyle = 'rgba(0,0,0,0.7)';
        ctx.fillRect(0, 0, W, H);
        ctx.fillStyle = gameState === 'WIN' ? '#4fc3f7' : '#f44336';
        ctx.font = 'bold 48px monospace';
        ctx.textAlign = 'center';
        ctx.fillText(gameState === 'WIN' ? 'VICTORY!' : 'GAME OVER', W/2, H/2 - 40);
        ctx.fillStyle = '#fff';
        ctx.font = '24px monospace';
        ctx.fillText(`Score: ${score}`, W/2, H/2 + 10);
        ctx.fillText('Press R to Restart', W/2, H/2 + 50);
        ctx.textAlign = 'left';
    }
}

function drawCards() {
    const cardW = Math.min(100, W/6);
    const cardH = Math.min(140, H/4);
    const totalW = hand.length * (cardW + 15) - 15;
    const startX = (W - totalW) / 2;
    const startY = H - cardH - 15;
    
    for (let i = 0; i < hand.length; i++) {
        const c = hand[i];
        const ct = CARD_TYPES[c.type];
        const x = startX + i * (cardW + 15);
        const y = startY;
        
        // Card bg
        ctx.fillStyle = 'rgba(20,20,40,0.9)';
        ctx.fillRect(x, y, cardW, cardH);
        ctx.strokeStyle = ct.color;
        ctx.lineWidth = 2;
        ctx.strokeRect(x, y, cardW, cardH);
        
        // Icon
        ctx.fillStyle = ct.color;
        ctx.font = `bold ${cardH/3}px monospace`;
        ctx.textAlign = 'center';
        ctx.fillText(ct.icon, x + cardW/2, y + cardH/2 + 10);
        
        // Name
        ctx.font = `${Math.min(12, cardW/8)}px monospace`;
        ctx.fillText(ct.name, x + cardW/2, y + cardH - 20);
        
        // Key hint
        ctx.fillStyle = '#aaa';
        ctx.font = `10px monospace`;
        ctx.fillText(`[${i+1}]`, x + cardW/2, y + cardH - 6);
    }
    ctx.textAlign = 'left';
}

function drawStartScreen() {
    ctx.fillStyle = '#fff';
    ctx.font = 'bold 42px monospace';
    ctx.textAlign = 'center';
    ctx.fillText('CARD PLATFORMER', W/2, H/3);
    
    ctx.font = '18px monospace';
    ctx.fillStyle = '#aaa';
    const lines = [
        'Arrow Keys / WASD - Move & Jump',
        '1-4 - Play Cards from Hand',
        '',
        'Use cards to fight enemies & reach the flag!',
        '',
        'CARDS: JUMP | DASH | FIRE | SHIELD | SPRING | BOLT',
        '',
        'Press any key to start'
    ];
    lines.forEach((l, i) => {
        ctx.fillText(l, W/2, H/2 + 20 + i * 28);
    });
    ctx.textAlign = 'left';
}

    if (gameState !== 'PLAYING') return;

        e.preventDefault();
        if (player.jumps > 0) {

            player.jumps--;
            playSound('sine', 300, 0.1);

        }
    }
});

// Game loop
let lastTime = 0;
function loop(time) {
    const dt = time - lastTime;
    lastTime = time;
    
    update();
    draw();
    
    requestAnimationFrame(loop);
}
