function playSound(freq, dur, type, vol) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = type || 'sine';
    osc.frequency.value = freq;
    gain.gain.setValueAtTime(vol || 0.15, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + dur);
}

// Particles
function spawnParticles(x, y, color, count) {
    for (let i = 0; i < count; i++) {
        const angle = Math.random() * Math.PI * 2;
        const speed = 50 + Math.random() * 150;
        particles.push({
            x, y,
            vx: Math.cos(angle) * speed,
            vy: Math.sin(angle) * speed,
            life: 0.5 + Math.random() * 0.5,
            maxLife: 0.5 + Math.random() * 0.5,
            color, size: 2 + Math.random() * 3
        });
    }
}

function updateParticles(dt) {
    for (let i = particles.length - 1; i >= 0; i--) {
        const p = particles[i];
        p.x += p.vx * dt;
        p.y += p.vy * dt;
        p.life -= dt;
        if (p.life <= 0) particles.splice(i, 1);
    }
}

// Collision
function collideWalls() {
    const lvl = levels[currentLevel];
    const r = ball.r;

    // Boundary walls
    if (ball.x - r < 0) { ball.x = r; ball.vx = Math.abs(ball.vx) * 0.7; playBounce(); }
    if (ball.x + r > 800) { ball.x = 800 - r; ball.vx = -Math.abs(ball.vx) * 0.7; playBounce(); }
    if (ball.y - r < 0) { ball.y = r; ball.vy = Math.abs(ball.vy) * 0.7; playBounce(); }
    if (ball.y + r > 600) { ball.y = 600 - r; ball.vy = -Math.abs(ball.vy) * 0.7; playBounce(); }

    // Level walls
    for (const w of lvl.walls) {
        const closestX = Math.max(w.x, Math.min(ball.x, w.x + w.w));
        const closestY = Math.max(w.y, Math.min(ball.y, w.y + w.h));
        const dx = ball.x - closestX;
        const dy = ball.y - closestY;
        const dist = Math.sqrt(dx * dx + dy * dy);

        if (dist < r) {
            const nx = dx / dist;
            const ny = dy / dist;
            const overlap = r - dist;
            ball.x += nx * overlap;
            ball.y += ny * overlap;
            const dot = ball.vx * nx + ball.vy * ny;
            if (dot < 0) {
                ball.vx -= 2 * dot * nx * 0.7;
                ball.vy -= 2 * dot * ny * 0.7;
                playBounce();
                spawnParticles(ball.x - nx * r, ball.y - ny * r, '#ff6644', 3);
            }
        }
    }

    // Spikes
    for (const s of lvl.spikes) {
        const closestX = Math.max(s.x, Math.min(ball.x, s.x + s.w));
        const closestY = Math.max(s.y, Math.min(ball.y, s.y + s.h));
        const dx = ball.x - closestX;
        const dy = ball.y - closestY;
        if (dx * dx + dy * dy < r * r) {
            die();
            return;
        }
    }

    // Goal
    const g = lvl.goal;
    if (ball.x > g.x && ball.x < g.x + g.w && ball.y > g.y && ball.y < g.y + g.h) {
        winLevel();
    }
}

// Update
let lastTime = performance.now();
function update() {
    const now = performance.now();
    let dt = (now - lastTime) / 1000;
    lastTime = now;
    if (dt > 0.05) dt = 0.05;

    if (gameState === 'playing') {
        levelTime += dt;
        ball.vx += gravityDir.x * GRAVITY_STRENGTH * dt;
        ball.vy += gravityDir.y * GRAVITY_STRENGTH * dt;

        // Speed cap
        const speed = Math.sqrt(ball.vx * ball.vx + ball.vy * ball.vy);
        if (speed > 500) {
            ball.vx = (ball.vx / speed) * 500;
            ball.vy = (ball.vy / speed) * 500;
        }

        ball.x += ball.vx * dt;
        ball.y += ball.vy * dt;

        collideWalls();
        updateParticles(dt);
    } else if (gameState === 'dead') {
        deathTimer += dt;
        updateParticles(dt);
        if (deathTimer > 1) {
            loadLevel(currentLevel);
            gameState = 'playing';
        }
    } else if (gameState === 'levelComplete') {
        levelTransition += dt;
        updateParticles(dt);
        if (levelTransition > 1.5) {
            currentLevel++;
            if (currentLevel >= levels.length) {
                gameState = 'gameComplete';
            } else {
                loadLevel(currentLevel);
                gameState = 'playing';
            }
        }
    }

    // Shake
    if (shakeTime > 0) shakeTime -= dt;

    draw();
    requestAnimationFrame(update);
}

    const spd = Math.sqrt(ball.vx * ball.vx + ball.vy * ball.vy);
    ctx.fillStyle = spd > 300 ? '#ff6644' : '#8899aa';
    ctx.fillText(`Speed: ${spd.toFixed(0)}`, 20, 96);
}

function drawDeath() {
    ctx.fillStyle = `rgba(255, 50, 50, ${Math.min(deathTimer * 2, 0.3)})`;
    ctx.fillRect(0, 0, 800, 600);

    if (deathTimer > 0.3) {
        ctx.fillStyle = '#ff4444';
        ctx.font = 'bold 36px monospace';
        ctx.textAlign = 'center';
        ctx.fillText('OUCH!', 400, 300);
    }
}

function drawLevelComplete() {
    const alpha = Math.min(levelTransition * 2, 1);
    ctx.fillStyle = `rgba(68, 255, 136, ${alpha * 0.1})`;
    ctx.fillRect(0, 0, 800, 600);

    ctx.globalAlpha = alpha;
    ctx.fillStyle = '#44ff88';
    ctx.font = 'bold 36px monospace';
    ctx.textAlign = 'center';
    ctx.fillText('LEVEL COMPLETE!', 400, 280);
    ctx.font = '18px monospace';
    ctx.fillStyle = '#aaffcc';
    ctx.fillText(`Time: ${levelTime.toFixed(1)}s`, 400, 320);

    if (currentLevel < levels.length - 1) {
        ctx.fillStyle = '#ffffff';
        ctx.font = '16px monospace';
        const pulse = 0.5 + Math.sin(performance.now() / 400) * 0.5;
        ctx.globalAlpha = alpha * pulse;
        ctx.fillText('Press SPACE for next level', 400, 380);
    } else {
        ctx.fillStyle = '#ffffff';
        ctx.font = '16px monospace';
        const pulse = 0.5 + Math.sin(performance.now() / 400) * 0.5;
        ctx.globalAlpha = alpha * pulse;
        ctx.fillText('Press SPACE to continue', 400, 380);
    }
    ctx.globalAlpha = 1;
}

function drawGameComplete() {
    ctx.fillStyle = 'rgba(10, 10, 26, 0.8)';
    ctx.fillRect(0, 0, 800, 600);

    ctx.fillStyle = '#44ff88';
    ctx.font = 'bold 42px monospace';
    ctx.textAlign = 'center';
    ctx.fillText('YOU WIN!', 400, 250);

    ctx.font = '20px monospace';
    ctx.fillStyle = '#aaffcc';
    ctx.fillText(`Total Time: ${totalTime.toFixed(1)}s`, 400, 310);

    // Confetti-like particles
    const t = performance.now() / 1000;
    for (let i = 0; i < 20; i++) {
        const px = 400 + Math.sin(t * 2 + i * 0.7) * (50 + i * 10);
        const py = 300 + Math.cos(t * 1.5 + i * 1.1) * (30 + i * 5);
        ctx.beginPath();
        ctx.arc(px, py, 3, 0, Math.PI * 2);
        ctx.fillStyle = `hsl(${(i * 36 + t * 50) % 360}, 80%, 60%)`;
        ctx.fill();
    }

    ctx.fillStyle = '#ffffff';
    ctx.font = '18px monospace';
    const pulse = 0.5 + Math.sin(performance.now() / 400) * 0.5;
    ctx.globalAlpha = pulse;
    ctx.fillText('Press SPACE to play again', 400, 400);
    ctx.globalAlpha = 1;

    // Track total time for game complete
    totalTime = levelTime;
}
