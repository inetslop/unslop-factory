const canvas = document.getElementById('game');
const ctx = canvas.getContext('2d');

let W, H;
function resize() {
    W = window.innerWidth;
    H = window.innerHeight;
    canvas.width = W;
    canvas.height = H;
}
window.addEventListener('resize', resize);
resize();

// Audio
let audioCtx = null;
function initAudio() {
    if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
}
function playSound(freq, type, duration, vol) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = type;
    osc.frequency.value = freq;
    gain.gain.setValueAtTime(vol || 0.15, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + duration);
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + duration);
}

// Game state
let gameState = 'start'; // start, playing, gameover
let score = 0;
let highScore = 0;
try { highScore = parseInt(localStorage.getItem('tds_high') || '0'); } catch(e) {}
let lives = 3;
let frameCount = 0;
let scrollSpeed = 2.5;

// Player
const player = { x: 150, y: 0, w: 28, h: 40, vy: 0, invincible: 0, dashCooldown: 0 };

// Arrays
let bullets = [];
let enemies = [];
let particles = [];
let collectibles = [];
let bgElements = [];
let stars = [];

// Input
const keys = {};
window.addEventListener('keydown', e => {
    keys[e.code] = true;
    if (gameState === 'start' && (e.code === 'Space' || e.code === 'Enter')) startGame();
    if (gameState === 'gameover' && (e.code === 'KeyR' || e.code === 'Enter')) startGame();
    if (gameState === 'playing') {
        if (e.code === 'Space') shoot();
        if (e.code === 'ShiftLeft' || e.code === 'ShiftRight') dash();
    }
});
window.addEventListener('keyup', e => { keys[e.code] = false; });

function startGame() {
    initAudio();
    gameState = 'playing';
    score = 0;
    lives = 3;
    frameCount = 0;
    scrollSpeed = 2.5;
    player.y = H / 2;
    player.vy = 0;
    player.invincible = 0;
    player.dashCooldown = 0;
    bullets = [];
    enemies = [];
    particles = [];
    collectibles = [];
    bgElements = [];
    initBg();
}

function initBg() {
    // Top-down terrain elements
    for (let i = 0; i < 30; i++) {
        bgElements.push({
            x: Math.random() * W * 2,
            y: Math.random() * H,
            type: Math.floor(Math.random() * 5), // 0=tree,1=rock,2=water,3=path,4=bush
            size: 15 + Math.random() * 30,
            speed: 0.3 + Math.random() * 0.7
        });
    }
}

function shoot() {
    if (frameCount % 8 === 0) return;
    bullets.push({ x: player.x + player.w, y: player.y - 4, vx: 12, vy: 0, w: 12, h: 6 });
    playSound(880, 'square', 0.08, 0.1);
}

function dash() {
    if (player.dashCooldown <= 0) {
        player.invincible = 20;
        player.dashCooldown = 40;
        playSound(220, 'sawtooth', 0.15, 0.1);
        for (let i = 0; i < 8; i++) {
            particles.push({ x: player.x, y: player.y, vx: -3 + Math.random() * 6, vy: -2 + Math.random() * 4, life: 20, color: '#7df' });
        }
    }
}

function spawnEnemy() {
    const type = Math.floor(Math.random() * 3);
    const ey = 50 + Math.random() * (H - 100);
    if (type === 0) {
        enemies.push({ x: W + 40, y: ey, vx: -(2 + Math.random() * 2), vy: (Math.random() - 0.5) * 2, w: 36, h: 36, type: 0, hp: 1, phase: Math.random() * Math.PI * 2 });
    } else if (type === 1) {
        enemies.push({ x: W + 40, y: ey, vx: -(1.5 + Math.random()), vy: 0, w: 44, h: 44, type: 1, hp: 3, phase: 0 });
    } else {
        enemies.push({ x: W + 40, y: ey, vx: -(3 + Math.random() * 2), vy: (Math.random() - 0.5) * 3, w: 28, h: 28, type: 2, hp: 1, phase: Math.random() * Math.PI * 2 });
    }
}

function spawnCollectible() {
    const cy = 60 + Math.random() * (H - 120);
    collectibles.push({ x: W + 30, y: cy, vx: -scrollSpeed, vy: 0, w: 20, h: 20, phase: 0 });
}

function update() {
    frameCount++;
    
    // Scroll speed increases over time
    scrollSpeed = 2.5 + Math.floor(score / 500) * 0.5;
    
    // Player movement
    if (keys['ArrowUp'] || keys['KeyW']) player.vy -= 0.8;
    if (keys['ArrowDown'] || keys['KeyS']) player.vy += 0.8;
    if (keys['ArrowLeft'] || keys['KeyA']) player.x -= 3;
    if (keys['ArrowRight'] || keys['KeyD']) player.x += 3;
    
    player.x = Math.max(20, Math.min(W * 0.5, player.x));
    player.vy *= 0.85;
    player.y += player.vy;
    player.y = Math.max(20, Math.min(H - 40, player.y));
    
    if (player.invincible > 0) player.invincible--;
    if (player.dashCooldown > 0) player.dashCooldown--;
    
    // Spawn enemies
    if (frameCount % Math.max(30, 60 - Math.floor(score / 200)) === 0) spawnEnemy();
    if (frameCount % 90 === 0) spawnCollectible();
    
    // Update bullets
    for (let i = bullets.length - 1; i >= 0; i--) {
        bullets[i].x += bullets[i].vx;
        if (bullets[i].x > W + 20) bullets.splice(i, 1);
    }
    
    // Update enemies
    for (let i = enemies.length - 1; i >= 0; i--) {
        const e = enemies[i];
        e.x += e.vx - scrollSpeed * 0.3;
        e.phase += 0.05;
        if (e.type === 0) e.y += Math.sin(e.phase * 2) * 1.5;
        if (e.type === 2) e.y += Math.cos(e.phase * 3) * 2;
        
        if (e.x < -60) { enemies.splice(i, 1); continue; }
        
        // Bullet collision
        for (let j = bullets.length - 1; j >= 0; j--) {
            const b = bullets[j];
            if (b.x < e.x + e.w && b.x + b.w > e.x && b.y < e.y + e.h && b.y + b.h > e.y) {
                e.hp--;
                bullets.splice(j, 1);
                playSound(440, 'square', 0.1, 0.12);
                for (let k = 0; k < 5; k++) {
                    particles.push({ x: e.x + e.w/2, y: e.y + e.h/2, vx: (Math.random()-0.5)*4, vy: (Math.random()-0.5)*4, life: 15, color: '#f80' });
                }
                if (e.hp <= 0) {
                    score += e.type === 1 ? 30 : 15;
                    for (let k = 0; k < 10; k++) {
                        particles.push({ x: e.x + e.w/2, y: e.y + e.h/2, vx: (Math.random()-0.5)*6, vy: (Math.random()-0.5)*6, life: 25, color: '#ff0' });
                    }
                    playSound(200, 'sawtooth', 0.2, 0.15);
                    enemies.splice(i, 1);
                }
                break;
            }
        }
        
        // Player collision
        if (player.invincible <= 0 && i < enemies.length) {
            const e2 = enemies[i];
            if (player.x < e2.x + e2.w && player.x + player.w > e2.x && player.y < e2.y + e2.h && player.y + player.h > e2.y) {
                lives--;
                player.invincible = 90;
                playSound(150, 'sawtooth', 0.3, 0.2);
                for (let k = 0; k < 12; k++) {
                    particles.push({ x: player.x + player.w/2, y: player.y + player.h/2, vx: (Math.random()-0.5)*5, vy: (Math.random()-0.5)*5, life: 20, color: '#f44' });
                }
                enemies.splice(i, 1);
                if (lives <= 0) {
                    gameState = 'gameover';
                    try {
                        if (score > highScore) { highScore = score; localStorage.setItem('tds_high', String(highScore)); }
                    } catch(e2) {}
                }
            }
        }
    }
    
    // Update collectibles
    for (let i = collectibles.length - 1; i >= 0; i--) {
        const c = collectibles[i];
        c.x += c.vx;
        c.phase += 0.1;
        if (c.x < -30) { collectibles.splice(i, 1); continue; }
        
        if (player.x < c.x + c.w && player.x + player.w > c.x && player.y < c.y + c.h && player.y + player.h > c.y) {
            score += 10;
            playSound(1200, 'sine', 0.15, 0.1);
            for (let k = 0; k < 6; k++) {
                particles.push({ x: c.x + c.w/2, y: c.y + c.h/2, vx: (Math.random()-0.5)*3, vy: (Math.random()-0.5)*3, life: 15, color: '#ff0' });
            }
            collectibles.splice(i, 1);
        }
    }
    
    // Update particles
    for (let i = particles.length - 1; i >= 0; i--) {
        const p = particles[i];
        p.x += p.vx;
        p.y += p.vy;
        p.life--;
        if (p.life <= 0) particles.splice(i, 1);
    }
    
    // Update bg elements
    for (const el of bgElements) {
        el.x -= scrollSpeed * el.speed;
        if (el.x < -60) {
            el.x = W + 40 + Math.random() * 100;
            el.y = Math.random() * H;
            el.type = Math.floor(Math.random() * 5);
            el.size = 15 + Math.random() * 30;
        }
    }
    
    // Score from distance
    if (frameCount % 10 === 0) score++;
}

function drawTopDown(x, y, size, type) {
    ctx.save();
    switch(type) {
        case 0: // Tree (top-down)
            ctx.beginPath();
            ctx.arc(x + size/2, y + size/2, size * 0.45, 0, Math.PI * 2);
            ctx.fillStyle = '#2d6b3f';
            ctx.fill();
            ctx.beginPath();
            ctx.arc(x + size/2, y + size/2, size * 0.3, 0, Math.PI * 2);
            ctx.fillStyle = '#3a8c52';
            ctx.fill();
            ctx.beginPath();
            ctx.arc(x + size/2 - 3, y + size/2 - 3, size * 0.15, 0, Math.PI * 2);
            ctx.fillStyle = '#4caf50';
            ctx.fill();
            break;
        case 1: // Rock
            ctx.beginPath();
            ctx.moveTo(x, y + size/2);
            ctx.lineTo(x + size*0.3, y);
            ctx.lineTo(x + size*0.7, y);
            ctx.lineTo(x + size, y + size/2);
            ctx.lineTo(x + size*0.7, y + size);
            ctx.lineTo(x + size*0.3, y + size);
            ctx.closePath();
            ctx.fillStyle = '#666';
            ctx.fill();
            ctx.strokeStyle = '#555';
            ctx.stroke();
            break;
        case 2: // Water (top-down)
            ctx.beginPath();
            ctx.ellipse(x + size/2, y + size/2, size*0.5, size*0.35, 0, 0, Math.PI * 2);
            ctx.fillStyle = 'rgba(30,100,180,0.4)';
            ctx.fill();
            ctx.strokeStyle = 'rgba(100,180,255,0.3)';
            ctx.stroke();
            break;
        case 3: // Path/dirt
            ctx.fillStyle = 'rgba(139,109,60,0.3)';
            ctx.fillRect(x, y, size * 1.5, size * 0.6);
            break;
        case 4: // Bush
            for (let i = 0; i < 3; i++) {
                ctx.beginPath();
                ctx.arc(x + size*0.2 + i*size*0.25, y + size*0.4, size*0.2, 0, Math.PI * 2);
                ctx.fillStyle = '#4a7c3f';
                ctx.fill();
            }
            break;
    }
    ctx.restore();
}

function drawPlayer() {
    if (player.invincible > 0 && Math.floor(frameCount / 4) % 2 === 0) return;
    
    const px = player.x, py = player.y;
    // Body (top-down - oval/circle seen from above)
    ctx.beginPath();
    ctx.ellipse(px + player.w/2, py + player.h/2, player.w * 0.4, player.h * 0.45, 0, 0, Math.PI * 2);
    ctx.fillStyle = '#e74c3c';
    ctx.fill();
    ctx.strokeStyle = '#c0392b';
    ctx.lineWidth = 2;
    ctx.stroke();
    
    // Top highlight
    ctx.beginPath();
    ctx.ellipse(px + player.w/2 - 4, py + player.h/2 - 6, player.w * 0.15, player.h * 0.15, 0, 0, Math.PI * 2);
    ctx.fillStyle = '#ff6b6b';
    ctx.fill();
    
    // Direction indicator (top-down - pointing right)
    ctx.beginPath();
    ctx.moveTo(px + player.w - 4, py + player.h/2);
    ctx.lineTo(px + player.w + 8, py + player.h/2);
    ctx.strokeStyle = '#fff';
    ctx.lineWidth = 3;
    ctx.stroke();
    
    // Engine glow
    if (frameCount % 6 < 3) {
        ctx.beginPath();
        ctx.arc(px + 4, py + player.h/2, 5, 0, Math.PI * 2);
        ctx.fillStyle = 'rgba(100,200,255,0.6)';
        ctx.fill();
    }
}

function drawEnemy(e) {
    const cx = e.x + e.w/2, cy = e.y + e.h/2;
    
    if (e.type === 0) { // Small bug (top-down)
        ctx.beginPath();
        ctx.ellipse(cx, cy, e.w*0.35, e.h*0.45, 0, 0, Math.PI * 2);
        ctx.fillStyle = '#8e44ad';
        ctx.fill();
        // Legs
        for (let i = -1; i <= 1; i += 2) {
            for (let j = -1; j <= 1; j++) {
                ctx.beginPath();
                ctx.moveTo(cx + i*e.w*0.25, cy + j*e.h*0.2);
                ctx.lineTo(cx + i*e.w*0.4, cy + j*e.h*0.35);
                ctx.strokeStyle = '#6c3483';
                ctx.lineWidth = 2;
                ctx.stroke();
            }
        }
        // Eye
        ctx.beginPath();
        ctx.arc(cx - 5, cy, 4, 0, Math.PI * 2);
        ctx.fillStyle = '#ff0';
        ctx.fill();
    } else if (e.type === 1) { // Large spider/creature (top-down)
        ctx.beginPath();
        ctx.arc(cx, cy, e.w*0.4, 0, Math.PI * 2);
        ctx.fillStyle = '#2c3e50';
        ctx.fill();
        // Legs (spider-like)
        for (let i = 0; i < 8; i++) {
            const angle = (i / 8) * Math.PI * 2 + e.phase * 0.5;
            const lx = cx + Math.cos(angle) * e.w * 0.35;
            const ly = cy + Math.sin(angle) * e.h * 0.35;
            ctx.beginPath();
            ctx.moveTo(cx, cy);
            ctx.lineTo(lx, ly);
            ctx.strokeStyle = '#1a252f';
            ctx.lineWidth = 3;
            ctx.stroke();
        }
        // Eyes (multiple)
        for (let i = -1; i <= 1; i += 2) {
            ctx.beginPath();
            ctx.arc(cx + i*6, cy - 4, 3, 0, Math.PI * 2);
            ctx.fillStyle = '#f44336';
            ctx.fill();
        }
    } else { // Fast dart (top-down)
        ctx.beginPath();
        ctx.ellipse(cx, cy, e.w*0.5, e.h*0.3, -0.3, 0, Math.PI * 2);
        ctx.fillStyle = '#ff9800';
        ctx.fill();
        ctx.strokeStyle = '#e65100';
        ctx.stroke();
        // Trail
        ctx.beginPath();
        ctx.moveTo(cx - e.w*0.4, cy);
        ctx.lineTo(cx - e.w*0.7, cy + 3);
        ctx.strokeStyle = 'rgba(255,152,0,0.4)';
        ctx.lineWidth = 2;
        ctx.stroke();
    }
}

function draw() {
    // Background - top-down ground
    const grad = ctx.createLinearGradient(0, 0, W, 0);
    grad.addColorStop(0, '#3d5c3a');
    grad.addColorStop(1, '#2e4a2b');
    ctx.fillStyle = grad;
    ctx.fillRect(0, 0, W, H);
    
    // Ground texture lines (top-down grass/terrain)
    ctx.strokeStyle = 'rgba(60,100,50,0.3)';
    ctx.lineWidth = 1;
    for (let i = 0; i < W; i += 40) {
        const offset = -(frameCount * scrollSpeed * 0.3) % 40;
        ctx.beginPath();
        ctx.moveTo(i + offset, 0);
        ctx.lineTo(i + offset - 10, H);
        ctx.stroke();
    }
    
    // Draw bg elements (terrain)
    for (const el of bgElements) {
        drawTopDown(el.x, el.y, el.size, el.type);
    }
    
    // Draw collectibles
    for (const c of collectibles) {
        const bob = Math.sin(c.phase) * 4;
        ctx.save();
        ctx.translate(c.x + c.w/2, c.y + c.h/2 + bob);
        ctx.rotate(c.phase * 0.5);
        // Star shape
        ctx.beginPath();
        for (let i = 0; i < 5; i++) {
            const angle = (i / 5) * Math.PI * 2 - Math.PI/2;
            const r = i % 2 === 0 ? 10 : 5;
            const px2 = Math.cos(angle * 2.4) * r;
            const py2 = Math.sin(angle * 2.4) * r;
            if (i === 0) ctx.moveTo(px2, py2);
            else ctx.lineTo(Math.cos((i/5)*Math.PI*2 - Math.PI/2) * (i%2?5:10), Math.sin((i/5)*Math.PI*2 - Math.PI/2) * (i%2?5:10));
        }
        ctx.closePath();
        // Simpler star
        ctx.beginPath();
        for (let i = 0; i < 5; i++) {
            const outerAngle = (i * 4 * Math.PI / 5) - Math.PI/2;
            const innerAngle = outerAngle + Math.PI/5;
            if (i === 0) ctx.moveTo(Math.cos(outerAngle)*10, Math.sin(outerAngle)*10);
            else {
                ctx.lineTo(Math.cos(outerAngle)*10, Math.sin(outerAngle)*10);
                ctx.lineTo(Math.cos(innerAngle)*5, Math.sin(innerAngle)*5);
            }
        }
        ctx.closePath();
        ctx.fillStyle = '#ffd700';
        ctx.fill();
        ctx.strokeStyle = '#ff8f00';
        ctx.lineWidth = 1;
        ctx.stroke();
        ctx.restore();
    }
    
    // Draw enemies
    for (const e of enemies) drawEnemy(e);
    
    // Draw bullets
    for (const b of bullets) {
        ctx.beginPath();
        ctx.ellipse(b.x + b.w/2, b.y + b.h/2, b.w*0.5, b.h*0.4, 0, 0, Math.PI * 2);
        ctx.fillStyle = '#7df';
        ctx.fill();
        // Glow
        ctx.beginPath();
        ctx.arc(b.x + b.w/2, b.y + b.h/2, 8, 0, Math.PI * 2);
        ctx.fillStyle = 'rgba(100,220,255,0.3)';
        ctx.fill();
    }
    
    // Draw player
    drawPlayer();
    
    // Draw particles
    for (const p of particles) {
        ctx.beginPath();
        ctx.arc(p.x, p.y, p.life * 0.4, 0, Math.PI * 2);
        ctx.fillStyle = p.color;
        ctx.globalAlpha = p.life / 25;
        ctx.fill();
        ctx.globalAlpha = 1;
    }
    
    // HUD
    ctx.fillStyle = 'rgba(0,0,0,0.5)';
    ctx.fillRect(10, 10, 200, 80);
    ctx.fillStyle = '#fff';
    ctx.font = 'bold 16px monospace';
    ctx.textAlign = 'left';
    ctx.fillText(`SCORE: ${score}`, 20, 35);
    ctx.fillText(`HI: ${highScore}`, 20, 55);
    
    // Lives (top-down hearts)
    for (let i = 0; i < lives; i++) {
        ctx.beginPath();
        ctx.arc(30 + i * 25, 72, 8, 0, Math.PI * 2);
        ctx.fillStyle = '#e74c3c';
        ctx.fill();
    }
    
    // Speed indicator
    ctx.fillStyle = 'rgba(0,0,0,0.3)';
    ctx.fillRect(W - 150, 10, 140, 25);
    ctx.fillStyle = '#7df';
    ctx.font = '12px monospace';
    ctx.fillText(`SPEED: ${scrollSpeed.toFixed(1)}x`, W - 140, 27);
}

function drawStartScreen() {
    const grad = ctx.createLinearGradient(0, 0, W, H);
    grad.addColorStop(0, '#1a1a2e');
    grad.addColorStop(1, '#16213e');
    ctx.fillStyle = grad;
    ctx.fillRect(0, 0, W, H);
    
    // Decorative top-down elements
    for (let i = 0; i < 15; i++) {
        const x = (i * 137 + frameCount * 0.2) % W;

function drawGameOver() {
    ctx.fillStyle = 'rgba(0,0,0,0.7)';
    ctx.fillRect(0, 0, W, H);
    
    ctx.fillStyle = '#f44336';
    ctx.font = 'bold 52px monospace';
    ctx.textAlign = 'center';
    ctx.fillText('GAME OVER', W/2, H/3);
    
    ctx.fillStyle = '#fff';
    ctx.font = '24px monospace';
    ctx.fillText(`Final Score: ${score}`, W/2, H/2 - 10);
    ctx.fillText(`High Score: ${highScore}`, W/2, H/2 + 30);
    
    ctx.fillStyle = '#ffd700';
    ctx.font = '20px monospace';
    const blink = Math.sin(frameCount * 0.05) > 0;
    if (blink) ctx.fillText('[ PRESS R OR ENTER TO RESTART ]', W/2, H * 0.75);
}

function loop() {
    frameCount++;
    
    if (gameState === 'playing') {
        update();
        draw();
    } else if (gameState === 'start') {
        drawStartScreen();
    } else if (gameState === 'gameover') {
        draw();
        drawGameOver();
    }
    
    requestAnimationFrame(loop);
}

loop();
