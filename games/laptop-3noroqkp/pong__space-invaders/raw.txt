// Audio
let audioCtx = null;
function initAudio() {
    if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
}
function playSound(type) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    
    if (type === 'shoot') {
        osc.type = 'square';
        osc.frequency.setValueAtTime(400, audioCtx.currentTime);
        osc.frequency.exponentialRampToValueAtTime(800, audioCtx.currentTime + 0.1);
        gain.gain.setValueAtTime(0.15, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.15);
        osc.start();
        osc.stop(audioCtx.currentTime + 0.15);
    } else if (type === 'explosion') {
        osc.type = 'sawtooth';
        osc.frequency.setValueAtTime(200, audioCtx.currentTime);
        osc.frequency.exponentialRampToValueAtTime(50, audioCtx.currentTime + 0.3);
        gain.gain.setValueAtTime(0.2, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.3);
        osc.start();
        osc.stop(audioCtx.currentTime + 0.3);
    } else if (type === 'wave') {
        osc.type = 'sine';
        osc.frequency.setValueAtTime(440, audioCtx.currentTime);
        osc.frequency.setValueAtTime(550, audioCtx.currentTime + 0.1);
        osc.frequency.setValueAtTime(660, audioCtx.currentTime + 0.2);
        gain.gain.setValueAtTime(0.15, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.4);
        osc.start();
        osc.stop(audioCtx.currentTime + 0.4);
    } else if (type === 'gameover') {
        osc.type = 'sawtooth';
        osc.frequency.setValueAtTime(300, audioCtx.currentTime);
        osc.frequency.exponentialRampToValueAtTime(50, audioCtx.currentTime + 1);
        gain.gain.setValueAtTime(0.2, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 1);
        osc.start();
        osc.stop(audioCtx.currentTime + 1);
    }
}

    stars = [];

function spawnAliens() {
    aliens = [];
    const rows = Math.min(3 + wave, 6);
    const cols = Math.min(5 + wave, 10);
    const alienW = 32;
    const alienH = 24;
    const spacingX = 50;
    const spacingY = 40;
    const startX = W / 2 - (cols * spacingX) / 2 + spacingX / 2;
    const startY = 80;
    
    for (let r = 0; r < rows; r++) {
        for (let c = 0; c < cols; c++) {
            let type = 0;

            else if (r === 1) type = 1;
            aliens.push({
                x: startX + c * spacingX,

        ctx.fillStyle = color;

function drawPaddle(x, y, w, h) {
    // Glow
    ctx.shadowColor = '#00ffff';
    ctx.shadowBlur = 15;
    ctx.fillStyle = '#00ffff';
    ctx.fillRect(x, y, w, h);
    // Inner
    ctx.shadowBlur = 0;

    ctx.fillRect(x + 4, y + 3, w - 8, h - 6);
}

function drawBall(x, y, r) {
    ctx.shadowColor = '#ffff00';
    ctx.shadowBlur = 12;
    ctx.fillStyle = '#ffff00';
    ctx.beginPath();
    ctx.arc(x, y, r, 0, Math.PI * 2);
    ctx.fill();
    ctx.shadowBlur = 0;
    ctx.fillStyle = '#ffffff';
    ctx.beginPath();
    ctx.arc(x - r*0.3, y - r*0.3, r*0.4, 0, Math.PI * 2);
    ctx.fill();
}

function drawStarfield() {
    for (let star of stars) {
        star.twinkle += 0.02;
        const alpha = 0.5 + Math.sin(star.twinkle) * 0.4;
        ctx.fillStyle = `rgba(255,255,255,${alpha})`;
        ctx.fillRect(star.x, star.y, star.size, star.size);
    }
}

function drawParticles() {
    for (let i = particles.length - 1; i >= 0; i--) {
        let p = particles[i];
        p.x += p.vx;
        p.y += p.vy;
        p.life--;
        if (p.life <= 0) {
            particles.splice(i, 1);
            continue;
        }
        const alpha = p.life / p.maxLife;
        ctx.fillStyle = `rgba(${p.r},${p.g},${p.b},${alpha})`;
        ctx.fillRect(p.x, p.y, p.size, p.size);
    }
}

function spawnExplosion(x, y, color) {
    const colors = [
        [255, 100, 100], [255, 200, 50], [100, 255, 100],
        [255, 150, 0], [200, 100, 255]
    ];
    const c = colors[color % colors.length];
    for (let i = 0; i < 12; i++) {
        const angle = Math.random() * Math.PI * 2;
        const speed = Math.random() * 4 + 1;
        particles.push({
            x: x, y: y,
            vx: Math.cos(angle) * speed,
            vy: Math.sin(angle) * speed,
            life: 30, maxLife: 30,
            size: Math.random() * 4 + 2,
            r: c[0], g: c[1], b: c[2]
        });
    }
}

// Game update
function update() {
    if (gameState !== 'playing') return;
    
    frameCount++;
    
    // Paddle movement
    if (keys['ArrowLeft'] || keys['KeyA']) paddle.x -= paddle.speed;
    if (keys['ArrowRight'] || keys['KeyD']) paddle.x += paddle.speed;
    paddle.x = Math.max(0, Math.min(W - paddle.w, paddle.x));
    
    // Ball auto-fire
    ballTimer++;
    if (!ball.active && ballTimer > 30) {
        ball.active = true;
        ball.x = paddle.x + paddle.w / 2;
        ball.y = H - 40;
        ball.vx = (Math.random() - 0.5) * 3;
        ball.vy = -10;
        playSound('shoot');
    }
    
    // Ball movement
    if (ball.active) {
        ball.x += ball.vx;
        ball.y += ball.vy;
        
        // Bounce off walls
        if (ball.x < ball.r) { ball.x = ball.r; ball.vx *= -1; }
        if (ball.x > W - ball.r) { ball.x = W - ball.r; ball.vx *= -1; }
        
        // Ball off top
        if (ball.y < -20) {
            ball.active = false;
            ballTimer = 0;
        }
        
        // Check collision with aliens
        for (let alien of aliens) {
            if (!alien.alive) continue;
            if (ball.x > alien.x && ball.x < alien.x + alien.w &&
                ball.y > alien.y && ball.y < alien.y + alien.h) {
                alien.alive = false;
                const pts = (3 - alien.type) * 10 + wave * 5;
                score += pts;
                spawnExplosion(alien.x + alien.w/2, alien.y + alien.h/2, alien.type);
                playSound('explosion');
                // Ball bounces slightly
                ball.vy = -Math.abs(ball.vy) * 0.9;
                ball.vx += (Math.random() - 0.5) * 2;
            }
        }
    }
    
    // Alien movement
    alienDescendTimer++;
    const aliveAliens = aliens.filter(a => a.alive);
    
    if (aliveAliens.length > 0) {
        let minX = Infinity, maxX = -Infinity;
        for (let a of aliveAliens) {
            if (a.x < minX) minX = a.x;
            if (a.x + a.w > maxX) maxX = a.x + a.w;
        }
        
        const hitLeft = alienDir === 1 && maxX >= W - 10;
        const hitRight = alienDir === -1 && minX <= 10;
        
        if (hitLeft || hitRight) {
            alienDir *= -1;
            for (let a of aliveAliens) a.y += 20; // Drop down
        } else {
            for (let a of aliveAliens) a.x += alienDir * alienSpeed;
        }
        
        // Frame animation
        if (alienDescendTimer % 30 === 0) {
            for (let a of aliveAliens) a.frame = 1 - a.frame;
        }
    }
    
    // Check if aliens reached bottom
    for (let a of aliveAliens) {
        if (a.y + a.h >= H - 60) {
            gameState = 'gameover';
            playSound('gameover');
            try { localStorage.setItem('pongInvadersHigh', Math.max(highScore, score).toString()); } catch(e) {}
            highScore = Math.max(highScore, score);
        }
    }
    
    // Check wave clear
    if (aliveAliens.length === 0) {
        wave++;
        playSound('wave');
        spawnAliens();
        ball.active = false;
        ballTimer = 0;
    }
}

function render() {
    ctx.fillStyle = '#0a0a1a';
    ctx.fillRect(0, 0, W, H);
    
    if (gameState === 'start') {
        drawStartScreen();
        return;
    }
    
    drawStarfield();
    drawParticles();
    
    // Draw aliens
    for (let alien of aliens) {
        if (!alien.alive) continue;
        drawPixelAlien(alien.x, alien.y, alien.w, alien.h, alien.type, alien.frame);
    }
    
    // Draw paddle
    drawPaddle(paddle.x, H - 40, paddle.w, paddle.h);
    
    // Draw ball
    if (ball.active) drawBall(ball.x, ball.y, ball.r);
    
    // Danger zone line
    ctx.strokeStyle = 'rgba(255,0,0,0.3)';
    ctx.setLineDash([10, 10]);
    ctx.beginPath();
    ctx.moveTo(0, H - 60);
    ctx.lineTo(W, H - 60);
    ctx.stroke();
    ctx.setLineDash([]);
    
    drawHUD();
    
    if (gameState === 'gameover') {
        drawGameOver();
    }
}

function gameLoop() {
    update();
    render();
    requestAnimationFrame(gameLoop);
}

gameLoop();
