<script>
const canvas = document.getElementById('game');
const ctx = canvas.getContext('2d');

let W, H;
function resize() {
    W = canvas.width = window.innerWidth;
    H = canvas.height = window.innerHeight;
}
window.addEventListener('resize', resize);
resize();

// Audio
let audioCtx = null;
function initAudio() {
    if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
}
function playSound(freq, type, duration, vol) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = type;
    osc.frequency.setValueAtTime(freq, audioCtx.currentTime);
    osc.frequency.exponentialRampToValueAtTime(freq * 0.5, audioCtx.currentTime + duration);
    gain.gain.setValueAtTime(vol, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + duration);
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + duration);
}
function playDestroy(size) {
    const base = size === 'large' ? 120 : size === 'medium' ? 200 : 350;
    playSound(base, 'sawtooth', 0.15, 0.3);
}
function playCombo(combo) {
    playSound(400 + combo * 50, 'square', 0.1, 0.2);
}
function playLifeLost() {
    playSound(80, 'sawtooth', 0.4, 0.4);
}
function playGameOver() {
    playSound(60, 'triangle', 0.8, 0.5);
}

// Game state
let gameState = 'start'; // start, playing, gameover
let score = 0;
let highScore = 0;
let lives = 3;
let combo = 0;
let lastClickTime = 0;
let asteroids = [];
let particles = [];
let floatTexts = [];
let spawnTimer = 0;
let gameTime = 0;
let stars = [];

try { highScore = parseInt(localStorage.getItem('asteroidsHighScore') || '0'); } catch(e) { highScore = 0; }

// Generate starfield
function genStars() {
    stars = [];
    for (let i = 0; i < 150; i++) {
        stars.push({ x: Math.random() * W, y: Math.random() * H, s: Math.random() * 2 + 0.5, b: Math.random() });
    }
}
genStars();

// Asteroid class
class Asteroid {
    constructor(x, y, size) {
        this.size = size; // 'large', 'medium', 'small'
        const sizes = { large: 40, medium: 25, small: 15 };
        this.radius = sizes[size];
        this.x = x || (this.radius + Math.random() * (W - this.radius * 2));
        this.y = y || (-this.radius);
        const speeds = { large: 60, medium: 90, small: 130 };
        this.vy = speeds[size] * (1 + gameTime * 0.05);
        this.vx = (Math.random() - 0.5) * 40;
        this.rotation = Math.random() * Math.PI * 2;
        this.rotSpeed = (Math.random() - 0.5) * 3;
        this.points = { large: 10, medium: 25, small: 75 }[size];
        // Generate rocky shape
        this.vertices = [];
        const verts = size === 'large' ? 10 : size === 'medium' ? 8 : 6;
        for (let i = 0; i < verts; i++) {
            const angle = (i / verts) * Math.PI * 2;
            const r = this.radius * (0.75 + Math.random() * 0.25);
            this.vertices.push({ x: Math.cos(angle) * r, y: Math.sin(angle) * r });
        }
    }
    update(dt) {
        this.x += this.vx * dt;
        this.y += this.vy * dt;
        this.rotation += this.rotSpeed * dt;
        if (this.x < this.radius) { this.x = this.radius; this.vx *= -1; }
        if (this.x > W - this.radius) { this.x = W - this.radius; this.vx *= -1; }
    }
    draw(ctx) {
        ctx.save();
        ctx.translate(this.x, this.y);
        ctx.rotate(this.rotation);
        ctx.beginPath();
        ctx.moveTo(this.vertices[0].x, this.vertices[0].y);
        for (let i = 1; i < this.vertices.length; i++) {
            ctx.lineTo(this.vertices[i].x, this.vertices[i].y);
        }
        ctx.closePath();
        const colors = { large: '#8B7355', medium: '#A0926B', small: '#C4B084' };
        ctx.fillStyle = colors[this.size];
        ctx.fill();
        ctx.strokeStyle = '#5a4a3a';
        ctx.lineWidth = 2;
        ctx.stroke();
        // Craters
        ctx.beginPath();
        for (let i = 0; i < 3; i++) {
            const cx = (Math.random() - 0.5) * this.radius * 0.6;
            const cy = (Math.random() - 0.5) * this.radius * 0.6;
            ctx.moveTo(cx + 3, cy);
            ctx.arc(cx, cy, 3, 0, Math.PI * 2);
        }
        ctx.fillStyle = 'rgba(0,0,0,0.2)';
        ctx.fill();
        ctx.restore();
    }
}

// Particle
class Particle {
    constructor(x, y, color) {
        this.x = x;
        this.y = y;
        const angle = Math.random() * Math.PI * 2;
        const speed = 50 + Math.random() * 150;
        this.vx = Math.cos(angle) * speed;
        this.vy = Math.sin(angle) * speed;
        this.life = 1;
        this.decay = 1.5 + Math.random() * 2;
        this.size = 2 + Math.random() * 4;
        this.color = color;
    }
    update(dt) {
        this.x += this.vx * dt;
        this.y += this.vy * dt;
        this.vy += 100 * dt;
        this.life -= this.decay * dt;
    }
    draw(ctx) {
        if (this.life <= 0) return;
        ctx.globalAlpha = this.life;
        ctx.fillStyle = this.color;
        ctx.beginPath();
        ctx.arc(this.x, this.y, this.size * this.life, 0, Math.PI * 2);
        ctx.fill();
        ctx.globalAlpha = 1;
    }
}

// Float text
class FloatText {
    constructor(x, y, text, color) {
        this.x = x;
        this.y = y;
        this.text = text;
        this.color = color;
        this.life = 1;
    }
    update(dt) {
        this.y -= 50 * dt;
        this.life -= 1.2 * dt;
    }
    draw(ctx) {
        if (this.life <= 0) return;
        ctx.globalAlpha = this.life;
        ctx.fillStyle = this.color;
        ctx.font = 'bold 18px Courier New';
        ctx.textAlign = 'center';
        ctx.fillText(this.text, this.x, this.y);
        ctx.globalAlpha = 1;
    }
}

function spawnAsteroid() {
    const rand = Math.random();
    let size;
    if (rand < 0.5) size = 'large';
    else if (rand < 0.8) size = 'medium';
    else size = 'small';
    asteroids.push(new Asteroid(null, null, size));
}

function getSpawnInterval() {
    return Math.max(0.3, 1.2 - gameTime * 0.02);
}

function spawnParticles(x, y, count, color) {
    for (let i = 0; i < count; i++) {
        particles.push(new Particle(x, y, color));
    }
}

function destroyAsteroid(index) {
    const a = asteroids[index];
    playDestroy(a.size);
    
    const comboMult = Math.min(combo, 10);
    const pts = a.points * comboMult;
    score += pts;
    
    floatTexts.push(new FloatText(a.x, a.y, `+${pts}`, '#FFD700'));
    if (combo > 2) {
        floatTexts.push(new FloatText(a.x, a.y - 25, `x${comboMult} COMBO!`, '#FF6B35'));
        playCombo(combo);
    }
    
    const colors = ['{ large: "#C4A870", medium: "#D4B884", small: "#E8D4A0" }'];
    const pColor = a.size === 'large' ? '#C4A870' : a.size === 'medium' ? '#D4B884' : '#E8D4A0';
    spawnParticles(a.x, a.y, a.size === 'large' ? 15 : a.size === 'medium' ? 10 : 6, pColor);
    
    // Split
    if (a.size === 'large') {
        asteroids.splice(index, 1);
        asteroids.push(new Asteroid(a.x - 20, a.y, 'medium'));
        asteroids.push(new Asteroid(a.x + 20, a.y, 'medium'));
    } else if (a.size === 'medium') {
        asteroids.splice(index, 1);
        asteroids.push(new Asteroid(a.x - 15, a.y, 'small'));
        asteroids.push(new Asteroid(a.x + 15, a.y, 'small'));
    } else {
        asteroids.splice(index, 1);
    }
}

function handleClick(x, y) {
    if (gameState !== 'playing') return;
    
    const now = performance.now();
    if (now - lastClickTime < 500) combo++;
    else combo = 1;
    lastClickTime = now;
    
    let hit = false;
    for (let i = asteroids.length - 1; i >= 0; i--) {
        const a = asteroids[i];
        const dx = x - a.x;
        const dy = y - a.y;
        if (dx * dx + dy * dy <= (a.radius + 10) * (a.radius + 10)) {
            destroyAsteroid(i);
            hit = true;
            break;
        }
    }
    
    if (!hit) {
        combo = 0;
    }
}

function resetGame() {
    score = 0;
    lives = 3;
    combo = 0;
    lastClickTime = 0;
    asteroids = [];
    particles = [];
    floatTexts = [];
    spawnTimer = 0;
    gameTime = 0;
}

function gameOver() {
    gameState = 'gameover';
    playGameOver();
    try { 
        if (score > highScore) {
            highScore = score;
            localStorage.setItem('asteroidsHighScore', String(highScore));
        }
    } catch(e) {}
}

// Input
canvas.addEventListener('click', (e) => {
    initAudio();
    const rect = canvas.getBoundingClientRect();
    const x = e.clientX - rect.left;
    const y = e.clientY - rect.top;
    
    if (gameState === 'start' || gameState === 'gameover') {
        resetGame();
        gameState = 'playing';
    } else {
        handleClick(x, y);
    }
});

canvas.addEventListener('touchstart', (e) => {
    e.preventDefault();
    initAudio();
    const rect = canvas.getBoundingClientRect();
    const touch = e.touches[0];
    const x = touch.clientX - rect.left;
    const y = touch.clientY - rect.top;
    
    if (gameState === 'start' || gameState === 'gameover') {
        resetGame();
        gameState = 'playing';
    } else {
        handleClick(x, y);
    }
});

window.addEventListener('keydown', (e) => {
    if (e.key === ' ' || e.key === 'Enter') {
        initAudio();
        if (gameState === 'start' || gameState === 'gameover') {
            resetGame();
            gameState = 'playing';
        }
    }
});

// Drawing helpers
function drawStars() {
    for (const s of stars) {
        const twinkle = 0.5 + 0.5 * Math.sin(gameTime * 2 + s.b * 10);
        ctx.globalAlpha = 0.3 + twinkle * 0.7;
        ctx.fillStyle = '#fff';
        ctx.beginPath();
        ctx.arc(s.x, s.y, s.s * 0.5, 0, Math.PI * 2);
        ctx.fill();
    }
    ctx.globalAlpha = 1;
}

function drawHUD() {
    // Score
    ctx.fillStyle = '#fff';
    ctx.font = 'bold 24px Courier New';
    ctx.textAlign = 'left';
    ctx.fillText(`SCORE: ${score}`, 20, 35);
    
    // High score
    ctx.font = '16px Courier New';
    ctx.fillStyle = '#aaa';
    ctx.fillText(`HI: ${highScore}`, 20, 58);
    
    // Lives
    ctx.textAlign = 'right';
    ctx.font = 'bold 24px Courier New';
    ctx.fillStyle = '#FF4444';
    let hearts = '';
    for (let i = 0; i < lives; i++) hearts += '❤';
    ctx.fillText(hearts, W - 20, 35);
    
    // Combo
    if (combo > 1) {
        ctx.textAlign = 'center';
        ctx.fillStyle = '#FFD700';
        ctx.font = 'bold 20px Courier New';
        ctx.fillText(`COMBO x${Math.min(combo, 10)}`, W / 2, 40);
    }
}

function drawStartScreen() {
    ctx.fillStyle = '#0a0a1a';
    ctx.fillRect(0, 0, W, H);
    drawStars();
    
    ctx.textAlign = 'center';
    ctx.fillStyle = '#E8D4A0';
    ctx.font = 'bold 48px Courier New';
    ctx.fillText('ASTEROIDS CLICKER', W / 2, H / 3);
    
    ctx.font = '18px Courier New';
    ctx.fillStyle = '#aaa';
    const lines = [
        'Click asteroids to destroy them!',
        '',
        'Large rocks split into medium,',
        'medium splits into small.',
        '',
        'Build combos by clicking fast!',
        'Don\'t let them reach the bottom!',
        '',
        'Lives: 3'
    ];
    lines.forEach((line, i) => {
        ctx.fillText(line, W / 2, H / 3 + 60 + i * 28);
    });
    
    ctx.fillStyle = '#FFD700';
    ctx.font = 'bold 22px Courier New';
    const pulse = 0.5 + 0.5 * Math.sin(performance.now() / 500);
    ctx.globalAlpha = pulse;
    ctx.fillText('[ CLICK TO START ]', W / 2, H - 60);
    ctx.globalAlpha = 1;
}

function drawGameOverScreen() {
    ctx.fillStyle = 'rgba(0,0,0,0.7)';
    ctx.fillRect(0, 0, W, H);
    
    ctx.textAlign = 'center';
    ctx.fillStyle = '#FF4444';
    ctx.font = 'bold 48px Courier New';
    ctx.fillText('GAME OVER', W / 2, H / 3);
    
    ctx.fillStyle = '#fff';
    ctx.font = 'bold 28px Courier New';
    ctx.fillText(`Score: ${score}`, W / 2, H / 3 + 50);
    
    if (score >= highScore && score > 0) {
        ctx.fillStyle = '#FFD700';
        ctx.font = 'bold 22px Courier New';
        ctx.fillText('NEW HIGH SCORE!', W / 2, H / 3 + 90);
    } else {
        ctx.fillStyle = '#aaa';
        ctx.font = '18px Courier New';
        ctx.fillText(`Best: ${highScore}`, W / 2, H / 3 + 90);
    }
    
    ctx.fillStyle = '#FFD700';
    ctx.font = 'bold 20px Courier New';
    const pulse = 0.5 + 0.5 * Math.sin(performance.now() / 500);
    ctx.globalAlpha = pulse;
    ctx.fillText('[ CLICK TO RESTART ]', W / 2, H - 60);
    ctx.globalAlpha = 1;
}

// Main loop
let lastTime = performance.now();

function gameLoop(now) {
    const dt = Math.min((now - lastTime) / 1000, 0.05);
    lastTime = now;
    
    if (gameState === 'start') {
        drawStartScreen();
    } else if (gameState === 'playing') {
        gameTime += dt;
        
        // Spawn
        spawnTimer -= dt;
        if (spawnTimer <= 0) {
            spawnAsteroid();
            spawnTimer = getSpawnInterval();
        }
        
        // Update asteroids
        for (let i = asteroids.length - 1; i >= 0; i--) {
            asteroids[i].update(dt);
            if (asteroids[i].y > H + asteroids[i].radius) {
                asteroids.splice(i, 1);
                lives--;
                playLifeLost();
                spawnParticles(W / 2, H - 10, 8, '#FF4444');
                if (lives <= 0) gameOver();
            }
        }
        
        // Update particles
        for (let i = particles.length - 1; i >= 0; i--) {
            particles[i].update(dt);
            if (particles[i].life <= 0) particles.splice(i, 1);
        }
        
        // Update float texts
        for (let i = floatTexts.length - 1; i >= 0; i--) {
            floatTexts[i].update(dt);
            if (floatTexts[i].life <= 0) floatTexts.splice(i, 1);
        }
        
        // Draw
        ctx.fillStyle = '#0a0a1a';
        ctx.fillRect(0, 0, W, H);
        drawStars();
        
        for (const a of asteroids) a.draw(ctx);
        for (const p of particles) p.draw(ctx);
        for (const f of floatTexts) f.draw(ctx);
        
        drawHUD();
    } else if (gameState === 'gameover') {
        // Still draw the scene behind
        ctx.fillStyle = '#0a0a1a';
        ctx.fillRect(0, 0, W, H);
        drawStars();
        for (const a of asteroids) a.draw(ctx);
        for (const p of particles) p.draw(ctx);
        drawHUD();
        drawGameOverScreen();
        
        // Update particles even in gameover
        for (let i = particles.length - 1; i >= 0; i--) {
            particles[i].update(dt);
            if (particles[i].life <= 0) particles.splice(i, 1);
        }
    }
    
    requestAnimationFrame(gameLoop);
}

requestAnimationFrame(gameLoop);
</script>
