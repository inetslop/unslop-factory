```html
<!DOCTYPE html>
<html lang="en">
<head>
<meta charset="UTF-8">
<meta name="viewport" content="width=device-width, initial-scale=1.0">
<title>Snake</title>
<style>
* { margin: 0; padding: 0; box-sizing: border-box; }
html, body { width: 100%; height: 100%; overflow: hidden; background: #0a0a0f; font-family: 'Courier New', monospace; }
canvas { display: block; width: 100vw; height: 100vh; }
</style>
</head>
<body>
<canvas id="game"></canvas>
<script>
(function() {
    const canvas = document.getElementById('game');
    const ctx = canvas.getContext('2d');

    let W, H, cellSize, gridW, gridH;
    let snake, dir, nextDir, food, score, highScore, speed, lastMove, running, dead, particles, shake;
    let audioCtx = null;
    let state = 'start'; // start, play, dead

    try { highScore = parseInt(localStorage.getItem('snakeHighScore')) || 0; } catch(e) { highScore = 0; }

    function saveHigh() { try { localStorage.setItem('snakeHighScore', highScore); } catch(e){} }

    function resize() {
        W = window.innerWidth; H = window.innerHeight;
        canvas.width = W; canvas.height = H;
        cellSize = Math.floor(Math.min(W, H) / 28);
        gridW = Math.floor(W / cellSize);
        gridH = Math.floor(H / cellSize);
    }
    window.addEventListener('resize', resize);
    resize();

    function initAudio() {
        if (!audioCtx) {
            audioCtx = new (window.AudioContext || window.webkitAudioContext)();
        }
    }

    function beep(freq, dur, type, vol) {
        if (!audioCtx) return;
        const osc = audioCtx.createOscillator();
        const gain = audioCtx.createGain();
        osc.type = type || 'square';
        osc.frequency.value = freq;
        gain.gain.setValueAtTime(vol || 0.15, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
        osc.connect(gain);
        gain.connect(audioCtx.destination);
        osc.start();
        osc.stop(audioCtx.currentTime + dur);
    }

    function playEat() { beep(600, 0.1, 'square', 0.12); beep(800, 0.08, 'square', 0.1); }
    function playDeath() { beep(200, 0.3, 'sawtooth', 0.2); beep(150, 0.4, 'square', 0.15); }
    function playStart() { beep(440, 0.1, 'square', 0.1); beep(660, 0.1, 'square', 0.1); }

    function resetGame() {
        const cx = Math.floor(gridW / 2), cy = Math.floor(gridH / 2);
        snake = [{x: cx, y: cy}, {x: cx - 1, y: cy}, {x: cx - 2, y: cy}];
        dir = {x: 1, y: 0};
        nextDir = {x: 1, y: 0};
        score = 0;
        speed = 8;
        lastMove = performance.now();
        running = true;
        dead = false;
        particles = [];
        shake = 0;
        placeFood();
    }

    function placeFood() {
        let fx, fy;
        do {
            fx = Math.floor(Math.random() * gridW);
            fy = Math.floor(Math.random() * gridH);
        } while (snake.some(s => s.x === fx && s.y === fy));
        food = {x: fx, y: fy};
    }

    function addParticles(x, y, color) {
        for (let i = 0; i < 8; i++) {
            const angle = Math.random() * Math.PI * 2;
            const spd = 1 + Math.random() * 3;
            particles.push({
                x: x * cellSize + cellSize/2,
                y: y * cellSize + cellSize/2,
                vx: Math.cos(angle) * spd,
                vy: Math.sin(angle) * spd,
                life: 1,
                color: color || '#4f4'
            });
        }
    }

    function updateParticles(dt) {
        for (let i = particles.length - 1; i >= 0; i--) {
            const p = particles[i];
            p.x += p.vx * cellSize * dt;
            p.y += p.vy * cellSize * dt;
            p.life -= dt * 3;
            if (p.life <= 0) particles.splice(i, 1);
        }
    }

    function handleInput() {
        const key = document.key;
        if (key === 'ArrowUp' || key === 'w' || key === 'W') {
            if (dir.x !== 0) nextDir = {x: 0, y: -1};
        } else if (key === 'ArrowDown' || key === 's' || key === 'S') {
            if (dir.x !== 0) nextDir = {x: 0, y: 1};
        } else if (key === 'ArrowLeft' || key === 'a' || key === 'A') {
            if (dir.y !== 0) nextDir = {x: -1, y: 0};
        } else if (key === 'ArrowRight' || key === 'd' || key === 'D') {
            if (dir.y !== 0) nextDir = {x: 1, y: 0};
        } else if (key === ' ' || key === 'Enter') {
            if (state === 'start' || state === 'dead') {
                initAudio();
                resetGame();
                state = 'play';
                playStart();
            }
        }
    }

    window.addEventListener('keydown', function(e) {
        document.key = e.key;
        if (state !== 'play' && (e.key === ' ' || e.key === 'Enter')) {
            initAudio();
            resetGame();
            state = 'play';
            playStart();
        }
        handleInput();
        if (['ArrowUp','ArrowDown','ArrowLeft','ArrowRight',' '].includes(e.key)) e.preventDefault();
    });

    function update(dt) {
        if (!running || dead) return;
        const interval = 1000 / speed;
        lastMove += dt * 1000;
        while (lastMove >= interval) {
            lastMove -= interval;
            dir = nextDir;
            const head = {x: snake[0].x + dir.x, y: snake[0].y + dir.y};

            if (head.x < 0 || head.x >= gridW || head.y < 0 || head.y >= gridH) {
                dead = true;
                shake = 1;
                playDeath();
                if (score > highScore) { highScore = score; saveHigh(); }
                state = 'dead';
                return;
            }

            for (let i = 0; i < snake.length; i++) {
                if (snake[i].x === head.x && snake[i].y === head.y) {
                    dead = true;
                    shake = 1;
                    playDeath();
                    if (score > highScore) { highScore = score; saveHigh(); }
                    state = 'dead';
                    return;
                }
            }

            snake.unshift(head);

            if (head.x === food.x && head.y === food.y) {
                score++;
                speed = 8 + score * 0.5;
                addParticles(head.x, head.y, '#4f4');
                playEat();
                placeFood();
            } else {
                snake.pop();
            }
        }
    }

    function drawGrid() {
        ctx.fillStyle = '#0a0a12';
        ctx.fillRect(0, 0, W, H);
        ctx.strokeStyle = 'rgba(30,30,50,0.5)';
        ctx.lineWidth = 1;
        for (let x = 0; x <= gridW; x++) {
            ctx.beginPath(); ctx.moveTo(x * cellSize, 0); ctx.lineTo(x * cellSize, H); ctx.stroke();
        }
        for (let y = 0; y <= gridH; y++) {
            ctx.beginPath(); ctx.moveTo(0, y * cellSize); ctx.lineTo(W, y * cellSize); ctx.stroke();
        }
    }

    function drawFood() {
        const fx = food.x * cellSize + cellSize / 2;
        const fy = food.y * cellSize + cellSize / 2;
        const pulse = 1 + Math.sin(performance.now() * 0.005) * 0.15;
        ctx.save();
        ctx.translate(fx, fy);
        ctx.scale(pulse, pulse);
        ctx.fillStyle = '#ff3366';
        ctx.shadowColor = '#ff3366';
        ctx.shadowBlur = 12;
        ctx.beginPath();
        ctx.arc(0, 0, cellSize * 0.35, 0, Math.PI * 2);
        ctx.fill();
        ctx.restore();
    }

    function drawSnake() {
        for (let i = snake.length - 1; i >= 0; i--) {
            const s = snake[i];
            const t = i / snake.length;
            const r = Math.floor(50 + t * 20);
            const g = Math.floor(200 - t * 80);
            const b = Math.floor(50 + t * 30);
            ctx.fillStyle = `rgb(${r},${g},${b})`;

            if (i === 0) {
                ctx.shadowColor = '#4f4';
                ctx.shadowBlur = 10;
            } else {
                ctx.shadowBlur = 0;
            }

            const pad = cellSize * 0.08;
            ctx.beginPath();
            ctx.roundRect(s.x * cellSize + pad, s.y * cellSize + pad, cellSize - pad * 2, cellSize - pad * 2, cellSize * 0.15);
            ctx.fill();
        }
        ctx.shadowBlur = 0;

        // Eyes on head
        const head = snake[0];
        const hx = head.x * cellSize + cellSize / 2;
        const hy = head.y * cellSize + cellSize / 2;
        const ex = dir.x * cellSize * 0.15;
        const ey = dir.y * cellSize * 0.15;
        ctx.fillStyle = '#fff';
        ctx.beginPath();
        ctx.arc(hx + ex - dir.y * cellSize * 0.2, hy + ey - dir.x * cellSize * 0.2, cellSize * 0.1, 0, Math.PI * 2);
        ctx.fill();
        ctx.beginPath();
        ctx.arc(hx + ex + dir.y * cellSize * 0.2, hy + ey + dir.x * cellSize * 0.2, cellSize * 0.1, 0, Math.PI * 2);
        ctx.fill();
    }

    function drawParticles() {
        for (const p of particles) {
            ctx.globalAlpha = p.life;
            ctx.fillStyle = p.color;
            ctx.beginPath();
            ctx.arc(p.x, p.y, cellSize * 0.15 * p.life, 0, Math.PI * 2);
            ctx.fill();
        }
        ctx.globalAlpha = 1;
    }

    function drawHUD() {
        ctx.fillStyle = '#fff';
        ctx.font = `bold ${Math.min(24, cellSize)}px 'Courier New', monospace`;
        ctx.textAlign = 'left';
        ctx.fillText(`SCORE: ${score}`, 16, 30);
        ctx.textAlign = 'right';
        ctx.fillText(`BEST: ${highScore}`, W - 16, 30);
    }

    function drawStartScreen() {
        ctx.fillStyle = '#0a0a12';
        ctx.fillRect(0, 0, W, H);

        const cx = W / 2, cy = H / 2;
        ctx.textAlign = 'center';

        // Title
        ctx.font = `bold ${Math.min(64, W * 0.1)}px 'Courier New', monospace`;
        ctx.fillStyle = '#4f4';
        ctx.shadowColor = '#4f4';
        ctx.shadowBlur = 20;
        ctx.fillText('SNAKE', cx, cy - 80);
        ctx.shadowBlur = 0;

        // Controls
        ctx.font = `${Math.min(18, W * 0.03)}px 'Courier New', monospace`;
        ctx.fillStyle = '#aaa';
        ctx.fillText('Arrow Keys / WASD to move', cx, cy - 20);
        ctx.fillText('Eat the red food. Don\'t hit walls or yourself!', cx, cy + 15);
        ctx.fillText('', cx, cy + 40);

        // Blinking prompt
        if (Math.floor(performance.now() / 600) % 2 === 0) {
            ctx.fillStyle = '#fff';
            ctx.font = `bold ${Math.min(22, W * 0.035)}px 'Courier New', monospace`;
            ctx.fillText('Press SPACE or ENTER to start', cx, cy + 80);
        }

        if (highScore > 0) {
            ctx.font = `${Math.min(16, W * 0.025)}px 'Courier New', monospace`;
            ctx.fillStyle = '#ff3366';
            ctx.fillText(`High Score: ${highScore}`, cx, cy + 120);
        }

        // Decorative snake
        const t = performance.now() * 0.002;
        for (let i = 0; i < 12; i++) {
            const sx = cx - 80 + i * 14;
            const sy = cy - 130 + Math.sin(t + i * 0.5) * 8;
            ctx.fillStyle = `rgba(79,255,79,${1 - i/12})`;
            ctx.beginPath();
            ctx.arc(sx, sy, 4, 0, Math.PI * 2);
            ctx.fill();
        }
    }

    function drawDeadScreen() {
        ctx.fillStyle = 'rgba(10,10,15,0.7)';
        ctx.fillRect(0, 0, W, H);

        const cx = W / 2, cy = H / 2;
        ctx.textAlign = 'center';

        ctx.font = `bold ${Math.min(48, W * 0.08)}px 'Courier New', monospace`;
        ctx.fillStyle = '#ff3366';
        ctx.shadowColor = '#ff3366';
        ctx.shadowBlur = 15;
        ctx.fillText('GAME OVER', cx, cy - 40);
        ctx.shadowBlur = 0;

        ctx.font = `${Math.min(24, W * 0.04)}px 'Courier New', monospace`;
        ctx.fillStyle = '#fff';
        ctx.fillText(`Score: ${score}`, cx, cy + 10);

        if (score >= highScore && score > 0) {
            ctx.fillStyle = '#ff3366';
            ctx.font = `bold ${Math.min(20, W * 0.03)}px 'Courier New', monospace`;
            ctx.fillText('NEW HIGH SCORE!', cx, cy + 45);
        }

        if (Math.floor(performance.now() / 600) % 2 === 0) {
            ctx.fillStyle = '#aaa';
            ctx.font = `${Math.min(18, W * 0.03)}px 'Courier New', monospace`;
            ctx.fillText('Press SPACE to play again', cx, cy + 90);
        }
    }

    let lastTime = performance.now();

    function loop(now) {
        const dt = Math.min((now - lastTime) / 1000, 0.1);
        lastTime = now;

        if (state === 'start') {
            drawStartScreen();
        } else if (state === 'play') {
            update(dt);
            updateParticles(dt);

            let ox = 0, oy = 0;
            if (shake > 0) {
                shake -= dt * 3;
                if (shake < 0) shake = 0;
                const sx = (Math.random() - 0.5) * shake * 10;
                const sy = (Math.random() - 0.5) * shake * 10;
                ox = sx; oy = sy;
            }

            ctx.save();
            ctx.translate(ox, oy);
            drawGrid();
            drawFood();
            drawSnake();
            drawParticles();
            drawHUD();
            ctx.restore();
        } else if (state === 'dead') {
            updateParticles(dt);
            let ox = 0, oy = 0;
            if (shake > 0) {
                shake -= dt * 3;
                if (shake < 0) shake = 0;
                const sx = (Math.random() - 0.5) * shake * 10;
                const sy = (Math.random() - 0.5) * shake * 10;
                ox = sx; oy = sy;
            }

            ctx.save();
            ctx.translate(ox, oy);
            drawGrid();
            drawFood();
            drawSnake();
            drawParticles();
            ctx.restore();

            drawDeadScreen();
        }

        requestAnimationFrame(loop);
    }

    // Polyfill roundRect if needed
    if (!CanvasRenderingContext2D.prototype.roundRect) {
        CanvasRenderingContext2D.prototype.roundRect = function(x, y, w, h, r) {
            if (typeof r === 'number') r = [r, r, r, r];
            const [tl, tr, br, bl] = Array.isArray(r) ? r : [r, r, r, r];
            this.moveTo(x + tl, y);
            this.arcTo(x + w, y, x + w, y + h, tr);
            this.arcTo(x + w, y + h, x, y + h, br);
            this.arcTo(x, y + h, x, y, bl);
            this.arcTo(x, y, x + w, y, tl);
            this.closePath();
        };
    }

    requestAnimationFrame(loop);
})();
</script>
</body>
</html>
```
