<script>
const canvas = document.getElementById('c');
const ctx = canvas.getContext('2d');

let W, H;
function resize() {
    W = window.innerWidth;
    H = window.innerHeight;
    canvas.width = W;
    canvas.height = H;
}
window.addEventListener('resize', resize);
resize();

// Audio
let audioCtx = null;
function initAudio() {
    if (!audioCtx) {
        audioCtx = new (window.AudioContext || window.webkitAudioContext)();
    }
}
function playSound(freq, type, duration, vol) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = type;
    osc.frequency.value = freq;
    gain.gain.setValueAtTime(vol || 0.15, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + duration);
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + duration);
}
function playCoin() { playSound(880, 'sine', 0.15, 0.2); setTimeout(() => playSound(1100, 'sine', 0.1, 0.15), 60); }
function playFlip() { playSound(300, 'triangle', 0.08, 0.1); }
function playDeath() { playSound(150, 'sawtooth', 0.4, 0.2); setTimeout(() => playSound(100, 'sawtooth', 0.5, 0.15), 100); }

// Game state
let state = 'start'; // start, playing, dead
let ball, gravityDir, worldY, speed, score, coins, particles, obstacles, coinItems, stars;
let highScore = 0;
try { highScore = parseInt(localStorage.getItem('gravityFallHigh') || '0'); } catch(e) {}
let time = 0;
let shakeTime = 0;

function init() {
    ball = { x: W / 2, y: H * 0.35, vy: 0, r: 14 };
    gravityDir = 1;
    worldY = 0;
    speed = 2.5;
    score = 0;
    coins = 0;
    particles = [];
    obstacles = [];
    coinItems = [];
    stars = [];
    for (let i = 0; i < 60; i++) {
        stars.push({ x: Math.random() * W, y: Math.random() * H, s: Math.random() * 2 + 0.5, sp: Math.random() * 0.5 + 0.2 });
    }
    // Initial obstacles
    for (let i = 0; i < 5; i++) {
        spawnObstacle(i * 300 - 100);
    }
}

function spawnObstacle(offsetY) {
    const gapSize = Math.max(90, 160 - speed * 8);
    const gapCenter = 80 + Math.random() * (H - 160);
    const type = Math.random() <  0.5 ? 'top' : 'bottom';
    let obs;
    if (type === 'top') {
        obs = { x: 0, y: (offsetY || worldY + H), w: W, h: gapCenter - gapSize / 2, isTop: true, passed: false };
        // Add spikes on bottom edge
        obs.spikes = [];
        const spikeCount = Math.floor(obs.h / 30);
        for (let i = 0; i < spikeCount; i++) {
            obs.spikes.push({ x: (i + 0.5) * (W / spikeCount), y: obs.y + obs.h });
        }
    } else {
        obs = { x: 0, y: gapCenter + gapSize / 2, w: W, h: H - (gapCenter + gapSize / 2), isTop: false, passed: false };
        obs.spikes = [];
        const spikeCount = Math.floor(obs.h / 30);
        for (let i = 0; i < spikeCount; i++) {
            obs.spikes.push({ x: (i + 0.5) * (W / spikeCount), y: obs.y });
        }
    }
    // Add coin in gap
    if (Math.random() < 0.7) {
        coinItems.push({ x: W * 0.2 + Math.random() * W * 0.6, y: gapCenter, r: 10, collected: false });
    }
    obstacles.push(obs);
}

function flipGravity() {
    if (state !== 'playing') return;
    gravityDir *= -1;
    playFlip();
    // Particle burst
    for (let i = 0; i < 6; i++) {
        const a = Math.random() * Math.PI * 2;
        particles.push({ x: ball.x, y: ball.y, vx: Math.cos(a) * 3, vy: Math.sin(a) * 3, life: 20, color: '#4fc3f7' });
    }
}

function startGame() {
    initAudio();
    init();
    state = 'playing';
}

function die() {
    if (state !== 'playing') return;
    state = 'dead';
    playDeath();
    shakeTime = 15;
    // Explosion particles
    for (let i = 0; i < 30; i++) {
        const a = Math.random() * Math.PI * 2;
        const sp = Math.random() * 6 + 2;
        particles.push({ x: ball.x, y: ball.y, vx: Math.cos(a) * sp, vy: Math.sin(a) * sp, life: 40, color: Math.random() < 0.5 ? '#ff5252' : '#ffab40' });
    }
    if (score > highScore) {
        highScore = score;
        try { localStorage.setItem('gravityFallHigh', '' + highScore); } catch(e) {}
    }
}

function update() {
    time++;
    if (state !== 'playing') return;

    speed += 0.003;
    const g = 0.45 * gravityDir;
    ball.vy += g;
    ball.y += ball.vy;
    worldY -= speed;

    score = Math.floor(-worldY / 10);

    // Spawn obstacles
    if (obstacles.length < 8) {
        const lastObs = obstacles[obstacles.length - 1];
        if (!lastObs || (lastObs.y + lastObs.h < worldY + H)) {
            spawnObstacle();
        }
    }

    // Remove far obstacles
    while (obstacles.length > 0 && obstacles[0].y + obstacles[0].h < worldY - 100) {
        obstacles.shift();
    }
    while (coinItems.length > 0 && coinItems[0].y < worldY - 100) {
        coinItems.shift();
    }

    // Collision with obstacles
    for (let obs of obstacles) {
        const oy = obs.y;
        if (!obs.isTop) {
            // Bottom obstacle: ball hits if ball.y + r > oy and ball is within gap x range (full width)
            if (ball.y + ball.r > oy && ball.y - ball.r < oy + obs.h) {
                die(); return;
            }
        } else {
            if (ball.y - ball.r < oy + obs.h && ball.y + ball.r > oy) {
                die(); return;
            }
        }
    }

    // Coin collection
    for (let c of coinItems) {
        if (c.collected) continue;
        const dx = ball.x - c.x;
        const dy = ball.y - c.y;
        if (dx * dx + dy * dy < (ball.r + c.r) * (ball.r + c.r)) {
            c.collected = true;
            coins++;
            score += 5;
            playCoin();
            for (let i = 0; i < 8; i++) {
                const a = Math.random() * Math.PI * 2;
                particles.push({ x: c.x, y: c.y, vx: Math.cos(a) * 3, vy: Math.sin(a) * 3, life: 15, color: '#ffd740' });
            }
        }
    }

    // Particles
    for (let i = particles.length - 1; i >= 0; i--) {
        const p = particles[i];
        p.x += p.vx;
        p.y += p.vy;
        p.life--;
        if (p.life <= 0) particles.splice(i, 1);
    }

    // Stars
    for (let s of stars) {
        s.y += s.sp * speed * 0.3;
        if (s.y > H) { s.y = -5; s.x = Math.random() * W; }
    }
}

function render() {
    ctx.save();
    if (shakeTime > 0) {
        shakeTime--;
        const sx = (Math.random() - 0.5) * 8;
        const sy = (Math.random() - 0.5) * 8;
        ctx.translate(sx, sy);
    }

    // Background gradient
    const grad = ctx.createLinearGradient(0, 0, 0, H);
    grad.addColorStop(0, '#0d1b2a');
    grad.addColorStop(1, '#1b2838');
    ctx.fillStyle = grad;
    ctx.fillRect(-10, -10, W + 20, H + 20);

    // Stars
    ctx.fillStyle = 'rgba(255,255,255,0.6)';
    for (let s of stars) {
        ctx.beginPath();
        ctx.arc(s.x, s.y, s.s, 0, Math.PI * 2);
        ctx.fill();
    }

    if (state === 'start') {
        drawStartScreen();
        ctx.restore();
        return;
    }

    // Obstacles
    for (let obs of obstacles) {
        const oy = obs.y + worldY * 0; // worldY offset is handled by rendering position
        // Actually we track worldY as a scroll value. Let me adjust: obstacles have absolute y, we render at obs.y (already in world space), and we offset by... 
        // Let me reconsider: I'll just use obs.y directly and the ball moves relative.
        // Actually let me simplify: obstacles move down screen as world scrolls.
        const renderY = obs.y;
        
        ctx.fillStyle = '#263238';
        if (obs.isTop) {
            ctx.fillRect(0, renderY, W, obs.h);
            // Spikes
            ctx.fillStyle = '#ef5350';
            for (let sp of obs.spikes) {
                ctx.beginPath();
                ctx.moveTo(sp.x - 12, sp.y + worldY * 0);
                ctx.lineTo(sp.x, sp.y + 18);
                ctx.lineTo(sp.x + 12, sp.y);
                ctx.closePath();
                ctx.fill();
            }
        } else {
            ctx.fillRect(0, renderY, W, obs.h);
            ctx.fillStyle = '#ef5350';
            for (let sp of obs.spikes) {
                ctx.beginPath();
                ctx.moveTo(sp.x - 12, sp.y);
                ctx.lineTo(sp.x, sp.y - 18);
                ctx.lineTo(sp.x + 12, sp.y);
                ctx.closePath();
                ctx.fill();
            }
        }
    }

    // Coins
    for (let c of coinItems) {
        if (c.collected) continue;
        const pulse = Math.sin(time * 0.1) * 2;
        ctx.beginPath();
        ctx.arc(c.x, c.y, c.r + pulse * 0.3, 0, Math.PI * 2);
        ctx.fillStyle = '#ffd740';
        ctx.fill();
        ctx.beginPath();
        ctx.arc(c.x, c.y, c.r - 3, 0, Math.PI * 2);
        ctx.fillStyle = '#fff9c4';
        ctx.fill();
    }

    // Particles
    for (let p of particles) {
        const alpha = p.life / 30;
        ctx.globalAlpha = Math.min(1, alpha);
        ctx.beginPath();
        ctx.arc(p.x, p.y, 3, 0, Math.PI * 2);
        ctx.fillStyle = p.color;
        ctx.fill();
    }
    ctx.globalAlpha = 1;

    // Ball
    if (state === 'playing') {
        // Trail
        for (let i = 1; i <= 5; i++) {
            const alpha = 0.3 / i;
            ctx.beginPath();
            ctx.arc(ball.x, ball.y - ball.vy * i * 0.8, ball.r * (1 - i * 0.12), 0, Math.PI * 2);
            ctx.fillStyle = `rgba(79,195,243,${alpha})`;
            ctx.fill();
        }

        // Ball body
        const bGrad = ctx.createRadialGradient(ball.x - 4, ball.y - 4, 0, ball.x, ball.y, ball.r);
        bGrad.addColorStop(0, '#81d4fa');
        bGrad.addColorStop(1, '#0288d1');
        ctx.beginPath();
        ctx.arc(ball.x, ball.y, ball.r, 0, Math.PI * 2);
        ctx.fillStyle = bGrad;
        ctx.fill();

        // Glow
        ctx.shadowBlur = 15;
        ctx.shadowColor = '#4fc3f7';
        ctx.beginPath();
        ctx.arc(ball.x, ball.y, ball.r, 0, Math.PI * 2);
        ctx.stroke();
        ctx.shadowBlur = 0;

        // Gravity indicator
        const arrowY = ball.y + (gravityDir === 1 ? 30 : -30);
        ctx.beginPath();
        ctx.moveTo(ball.x, arrowY);
        ctx.lineTo(ball.x - 6, arrowY - gravityDir * 8);
        ctx.lineTo(ball.x + 6, arrowY - gravityDir * 8);
        ctx.closePath();
        ctx.fillStyle = 'rgba(255,255,255,0.5)';
        ctx.fill();
    }

    // HUD
    ctx.font = 'bold 24px monospace';
    ctx.fillStyle = '#fff';
    ctx.textAlign = 'left';
    ctx.fillText('SCORE: ' + score, 20, 36);
    ctx.font = '16px monospace';
    ctx.fillStyle = '#ffd740';
    ctx.fillText('◆ ' + coins, 20, 60);
    ctx.textAlign = 'right';
    ctx.font = '14px monospace';
    ctx.fillStyle = 'rgba(255,255,255,0.5)';
    ctx.fillText('BEST: ' + highScore, W - 20, 36);

    if (state === 'dead') {
        drawDeathScreen();
    }

    ctx.restore();
}

function drawStartScreen() {
    ctx.textAlign = 'center';
    ctx.font = 'bold 48px monospace';
    const t = Math.sin(time * 0.05) * 3;
    ctx.fillStyle = '#4fc3f7';
    ctx.shadowBlur = 20;
    ctx.shadowColor = '#4fc3f7';
    ctx.fillText('GRAVITY FALL', W / 2, H / 2 - 60 + t);
    ctx.shadowBlur = 0;

    ctx.font = '18px monospace';
    ctx.fillStyle = 'rgba(255,255,255,0.8)';
    ctx.fillText('Click / Tap / SPACE to flip gravity', W / 2, H / 2 + 10);
    ctx.fillText('Dodge the walls. Collect coins.', W / 2, H / 2 + 40);
    ctx.fillText('Speed increases over time!', W / 2, H / 2 + 70);

    const blink = Math.sin(time * 0.1) > 0;
    if (blink) {
        ctx.font = 'bold 22px monospace';
        ctx.fillStyle = '#fff';
        ctx.fillText('TAP TO START', W / 2, H / 2 + 130);
    }

    if (highScore > 0) {
        ctx.font = '16px monospace';
        ctx.fillStyle = 'rgba(255,255,255,0.5)';
        ctx.fillText('Best: ' + highScore, W / 2, H / 2 + 170);
    }
}

function drawDeathScreen() {
    ctx.textAlign = 'center';
    ctx.fillStyle = 'rgba(0,0,0,0.6)';
    ctx.fillRect(0, 0, W, H);

    ctx.font = 'bold 42px monospace';
    ctx.fillStyle = '#ef5350';
    ctx.fillText('GAME OVER', W / 2, H / 2 - 40);

    ctx.font = '24px monospace';
    ctx.fillStyle = '#fff';
    ctx.fillText('Score: ' + score, W / 2, H / 2 + 10);
    ctx.fillStyle = '#ffd740';
    ctx.fillText('Coins: ' + coins, W / 2, H / 2 + 45);

    if (score >= highScore && score > 0) {
        ctx.font = '18px monospace';
        ctx.fillStyle = '#69f0ae';
        ctx.fillText('★ NEW HIGH SCORE! ★', W / 2, H / 2 + 80);
    }

    const blink = Math.sin(time * 0.1) > 0;
    if (blink) {
        ctx.font = 'bold 20px monospace';
        ctx.fillStyle = '#fff';
        ctx.fillText('TAP TO RESTART', W / 2, H / 2 + 130);
    }
}

let inputReady = false;
function handleInput() {
    initAudio();
    if (!inputReady) { inputReady = true; return; }
    if (state === 'start') {
        startGame();
    } else if (state === 'dead') {
        if (shakeTime > 5) return; // prevent instant restart
        startGame();
    } else if (state === 'playing') {
        flipGravity();
    }
}

window.addEventListener('keydown', (e) => {
    if (e.code === 'Space' || e.code === 'ArrowUp' || e.code === 'ArrowDown') {
        e.preventDefault();
        handleInput();
    }
});
canvas.addEventListener('click', () => { initAudio(); handleInput(); });
canvas.addEventListener('touchstart', (e) => { e.preventDefault(); initAudio(); handleInput(); });

function loop() {
    update();
    render();
    requestAnimationFrame(loop);
}

init();
loop();
</script>
