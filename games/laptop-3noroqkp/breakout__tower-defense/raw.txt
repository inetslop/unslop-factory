<script>
(function(){
const canvas = document.getElementById('game');
const ctx = canvas.getContext('2d');

let W, H;
function resize() {
    W = window.innerWidth;
    H = window.innerHeight;
    canvas.width = W;
    canvas.height = H;
}
window.addEventListener('resize', resize);
resize();

// Audio
let audioCtx = null;
function initAudio() {
    if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
}
function playSound(freq, dur, type, vol) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = type || 'square';
    osc.frequency.value = freq;
    gain.gain.setValueAtTime(vol || 0.15, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + dur);
}

// Game state
let state = 'start'; // start, playing, gameover, win
let score = 0;
let baseHP = 100;
let maxBaseHP = 100;
let wave = 1;
let maxWaves = 15;
let balls = [];
let bricks = [];
let enemies = [];
let projectiles = [];
let particles = [];
let powerups = [];
let paddle = { x: 0, w: 120, h: 18, speed: 8 };
let fireCooldown = 0;
let fireRate = 30; // frames between shots
let waveTimer = 0;
let enemiesToSpawn = 0;
let spawnTimer = 0;
let waveActive = false;
let shieldCount = 0;
let multiBallTimer = 0;
let rapidFireTimer = 0;
let expandTimer = 0;
let shakeTimer = 0;
let comboCounter = 0;

const keys = {};
window.addEventListener('keydown', e => { keys[e.code] = true; if(state==='start') startGame(); });
window.addEventListener('keyup', e => { keys[e.code] = false; });
canvas.addEventListener('click', () => {
    initAudio();
    if (state === 'start') startGame();
    else if (state === 'gameover' || state === 'win') startGame();
});

function startGame() {
    state = 'playing';
    score = 0;
    baseHP = maxBaseHP;
    wave = 1;
    balls = [{x: W/2, y: H - 80, vx: 4, vy: -5, r: 8, onPaddle: true}];
    paddle.x = W / 2;
    paddle.w = 120;
    fireCooldown = 0;
    fireRate = 30;
    shieldCount = 0;
    multiBallTimer = 0;
    rapidFireTimer = 0;
    expandTimer = 0;
    comboCounter = 0;
    enemies = [];
    projectiles = [];
    particles = [];
    powerups = [];
    bricks = [];
    generateBricks();
    startWave();
}

function generateBricks() {
    bricks = [];
    const rows = 5;
    const cols = Math.floor(W / 60);
    const bw = W / cols;
    const bh = 28;
    const offset = (W - cols * bw) / 2;
    for (let r = 0; r < rows; r++) {
        for (let c = 0; c < cols; c++) {
            const hp = r < 2 ? 3 : r < 4 ? 2 : 1;
            bricks.push({
                x: offset + c * bw + 2,
                y: 60 + r * (bh + 4) + 2,
                w: bw - 4,
                h: bh,
                hp: hp,
                maxHp: hp,
                color: getBrickColor(hp),
                alive: true
            });
        }
    }
}

function getBrickColor(hp) {
    if (hp >= 3) return '#ff4444';
    if (hp === 2) return '#ffaa00';
    return '#44ff88';
}

function startWave() {
    waveActive = true;
    enemiesToSpawn = 3 + wave * 2;
    spawnTimer = 0;
    waveTimer = 0;
}

function spawnEnemy() {
    const types = ['scout', 'grunt', 'tank'];
    let type;
    if (wave % 5 === 0 && enemiesToSpawn === 1) {
        type = 'boss';
    } else {
        const r = Math.random();
        if (r < 0.4) type = 'scout';
        else if (r < 0.75) type = 'grunt';
        else type = 'tank';
    }
    
    const x = 80 + Math.random() * (W - 160);
    const baseY = -40;
    
    let e = { x, y: baseY, type, vx: 0, vy: 0, shooting: false };
    
    switch(type) {
        case 'scout':
            e.r = 12; e.hp = 1; e.maxHp = 1; e.speed = 2.5; e.color = '#ff6699'; e.score = 10;
            break;
        case 'grunt':
            e.r = 18; e.hp = 3; e.maxHp = 3; e.speed = 1.2; e.color = '#ffaa44'; e.score = 25;
            break;
        case 'tank':
            e.r = 26; e.hp = 6; e.maxHp = 6; e.speed = 0.7; e.color = '#8855ff'; e.score = 50;
            break;
        case 'boss':
            e.r = 45; e.hp = 30 + wave * 5; e.maxHp = e.hp; e.speed = 0.4; e.color = '#ff2222'; e.score = 200;
            e.shooting = true;
            break;
    }
    enemies.push(e);
}

function fireProjectile() {
    if (fireCooldown <= 0) {
        projectiles.push({ x: paddle.x, y: H - 40, vx: 0, vy: -12, r: 5 });
        fireCooldown = rapidFireTimer > 0 ? Math.floor(fireRate / 2) : fireRate;
        playSound(880, 0.08, 'square', 0.1);
    }
}

function spawnParticles(x, y, color, count) {
    for (let i = 0; i < count; i++) {
        const angle = Math.random() * Math.PI * 2;
        const speed = 1 + Math.random() * 4;
        particles.push({
            x, y,
            vx: Math.cos(angle) * speed,
            vy: Math.sin(angle) * speed,
            life: 30 + Math.random() * 20,
            maxLife: 50,
            color,
            r: 2 + Math.random() * 3
        });
    }
}

function spawnPowerup(x, y) {
    const types = ['multi', 'expand', 'rapid', 'shield'];
    const type = types[Math.floor(Math.random() * types.length)];
    powerups.push({ x, y, vy: 2, r: 12, type, life: 300 });
}

function update() {
    if (state !== 'playing') return;
    
    // Paddle movement
    if (keys['ArrowLeft'] || keys['KeyA']) paddle.x -= paddle.speed;
    if (keys['ArrowRight'] || keys['KeyD']) paddle.x += paddle.speed;
    paddle.x = Math.max(paddle.w/2, Math.min(W - paddle.w/2, paddle.x));
    
    // Fire
    if (keys['Space'] || keys['KeyW']) fireProjectile();
    
    if (fireCooldown > 0) fireCooldown--;
    if (multiBallTimer > 0) multiBallTimer--;
    if (rapidFireTimer > 0) rapidFireTimer--;
    if (expandTimer > 0) { expandTimer--; if (expandTimer === 0) paddle.w = Math.max(120, paddle.w - 2); }
    
    // Ball physics
    for (let i = balls.length - 1; i >= 0; i--) {
        const b = balls[i];
        if (b.onPaddle) {
            b.x = paddle.x;
            b.y = H - 50;
            continue;
        }
        b.x += b.vx;
        b.y += b.vy;
        
        // Wall bounce
        if (b.x < b.r) { b.x = b.r; b.vx = Math.abs(b.vx); playSound(300, 0.05, 'sine', 0.08); }
        if (b.x > W - b.r) { b.x = W - b.r; b.vx = -Math.abs(b.vx); playSound(300, 0.05, 'sine', 0.08); }
        if (b.y < b.r) { b.y = b.r; b.vy = Math.abs(b.vy); playSound(300, 0.05, 'sine', 0.08); }
        
        // Paddle bounce
        if (b.vy > 0 && b.y + b.r >= H - 40 && b.y - b.r <= H - 40 + paddle.h && Math.abs(b.x - paddle.x) <= paddle.w/2 + b.r) {
            b.y = H - 40 - b.r;
            const hitPos = (b.x - paddle.x) / (paddle.w/2);
            b.vx = hitPos * 6;
            b.vy = -Math.abs(b.vy);
            playSound(440, 0.08, 'sine', 0.12);
        }
        
        // Brick collision
        for (let j = bricks.length - 1; j >= 0; j--) {
            const br = bricks[j];
            if (!br.alive) continue;
            if (b.x + b.r > br.x && b.x - b.r < br.x + br.w && b.y + b.r > br.y && b.y - b.r < br.y + br.h) {
                br.hp--;
                comboCounter++;
                score += 5 * Math.min(comboCounter, 10);
                if (br.hp <= 0) {
                    br.alive = false;
                    spawnParticles(br.x + br.w/2, br.y + br.h/2, br.color, 8);
                    playSound(600, 0.1, 'square', 0.15);
                    if (Math.random() < 0.15) spawnPowerup(br.x + br.w/2, br.y + br.h/2);
                } else {
                    b.vy *= -1;
                    playSound(500, 0.06, 'square', 0.1);
                }
                break;
            }
        }
        
        // Ball lost
        if (b.y > H + 20) {
            balls.splice(i, 1);
            comboCounter = 0;
            if (balls.length === 0) {
                // New ball on paddle
                balls.push({x: paddle.x, y: H - 50, vx: 4, vy: -5, r: 8, onPaddle: true});
            }
        }
    }
    
    // Launch ball if on paddle
    for (let b of balls) {
        if (b.onPaddle && keys['Space']) {
            b.onPaddle = false;
            b.vy = -7;
            b.vx = (Math.random() - 0.5) * 4;
        }
    }
    
    // Spawn enemies
    if (waveActive) {
        waveTimer++;
        spawnTimer++;
        const interval = Math.max(30, 80 - wave * 5);
        if (spawnTimer >= interval && enemiesToSpawn > 0) {
            spawnEnemy();
            enemiesToSpawn--;
            spawnTimer = 0;
        }
        if (enemiesToSpawn <= 0 && enemies.length === 0) {
            waveActive = false;
            if (wave >= maxWaves) {
                state = 'win';
                return;
            }
            wave++;
            setTimeout(() => { if(state==='playing') startWave(); }, 1500);
        }
    }
    
    // Update enemies
    for (let i = enemies.length - 1; i >= 0; i--) {
        const e = enemies[i];
        e.y += e.speed;
        e.x += Math.sin(e.y * 0.02) * 0.5;
        
        // Boss shooting
        if (e.shooting && waveTimer % 60 === 0) {
            const angle = Math.atan2(H - 40 - e.y, paddle.x - e.x);
            projectiles.push({ x: e.x, y: e.y + e.r, vx: Math.cos(angle) * 3, vy: Math.sin(angle) * 3, r: 6, enemy: true });
        }
        
        // Reached base
        if (e.y + e.r > H - 20) {
            let dmg = 10;
            if (e.type === 'boss') dmg = 30;
            if (shieldCount > 0) { shieldCount--; spawnParticles(e.x, e.y, '#44aaff', 15); }
            else baseHP -= dmg;
            enemies.splice(i, 1);
            shakeTimer = 15;
            playSound(150, 0.3, 'sawtooth', 0.2);
            if (baseHP <= 0) { state = 'gameover'; return; }
            continue;
        }
        
        // Projectile hit enemy
        for (let j = projectiles.length - 1; j >= 0; j--) {
            const p = projectiles[j];
            if (p.enemy) continue;
            const dx = e.x - p.x, dy = e.y - p.y;
            if (dx*dx + dy*dy < (e.r + p.r) * (e.r + p.r)) {
                e.hp--;
                projectiles.splice(j, 1);
                spawnParticles(p.x, p.y, '#ffaa00', 4);
                playSound(700, 0.05, 'square', 0.08);
                if (e.hp <= 0) {
                    score += e.score;
                    spawnParticles(e.x, e.y, e.color, 12);
                    playSound(300, 0.15, 'sawtooth', 0.15);
                    enemies.splice(i, 1);
                }
                break;
            }
        }
        
        // Ball hits enemy
        for (let b of balls) {
            const dx = e.x - b.x, dy = e.y - b.y;
            if (dx*dx + dy*dy < (e.r + b.r) * (e.r + b.r)) {
                e.hp -= 2;
                b.vy *= -1;
                spawnParticles(b.x, b.y, '#ffffff', 5);
                playSound(500, 0.08, 'sine', 0.1);
                if (e.hp <= 0) {
                    score += e.score;
                    spawnParticles(e.x, e.y, e.color, 12);
                    enemies.splice(i, 1);
                    break;
                }
            }
        }
    }
    
    // Update projectiles
    for (let i = projectiles.length - 1; i >= 0; i--) {
        const p = projectiles[i];
        p.x += p.vx;
        p.y += p.vy;
        if (p.y < -20 || p.y > H + 20 || p.x < -20 || p.x > W + 20) {
            projectiles.splice(i, 1);
        }
    }
    
    // Update particles
    for (let i = particles.length - 1; i >= 0; i--) {
        const p = particles[i];
        p.x += p.vx;
        p.y += p.vy;
        p.vy += 0.1;
        p.life--;
        if (p.life <= 0) particles.splice(i, 1);
    }
    
    // Update powerups
    for (let i = powerups.length - 1; i >= 0; i--) {
        const pu = powerups[i];
        pu.y += pu.vy;
        pu.life--;
        if (pu.life <= 0 || pu.y > H) { powerups.splice(i, 1); continue; }
        
        // Collect with paddle
        if (Math.abs(pu.x - paddle.x) < paddle.w/2 + pu.r && Math.abs(pu.y - (H-40)) < 30) {
            applyPowerup(pu.type);
            spawnParticles(pu.x, pu.y, '#ffffff', 8);
            playSound(1200, 0.15, 'sine', 0.15);
            powerups.splice(i, 1);
        }
    }
    
    if (shakeTimer > 0) shakeTimer--;
}

function applyPowerup(type) {
    switch(type) {
        case 'multi':
            const b = balls[0] || {x: paddle.x, y: H-50, vx: 3, vy: -6, r: 8};
            if (!b.onPaddle) {
                for (let i = 0; i < 2; i++) {
                    const angle = Math.atan2(b.vy, b.vx) + (i === 0 ? 0.5 : -0.5);
                    balls.push({x: b.x, y: b.y, vx: Math.cos(angle)*6, vy: Math.sin(angle)*6, r: 8});
                }
            }
            break;
        case 'expand':
            paddle.w = Math.min(200, paddle.w + 30);
            expandTimer = 600;
            break;
        case 'rapid':
            rapidFireTimer = 600;
            break;
        case 'shield':
            shieldCount++;
            break;
    }
}

function draw() {
    ctx.save();
    
    // Screen shake
    if (shakeTimer > 0) {
        const sx = (Math.random() - 0.5) * shakeTimer;
        const sy = (Math.random() - 0.5) * shakeTimer;
        ctx.translate(sx, sy);
    }
    
    // Background
    ctx.fillStyle = '#0a0a1a';
    ctx.fillRect(-10, -10, W+20, H+20);
    
    // Grid lines
    ctx.strokeStyle = 'rgba(30, 40, 80, 0.3)';
    ctx.lineWidth = 1;
    for (let x = 0; x < W; x += 50) {
        ctx.beginPath(); ctx.moveTo(x, 0); ctx.lineTo(x, H); ctx.stroke();
    }
    for (let y = 0; y < H; y += 50) {
        ctx.beginPath(); ctx.moveTo(0, y); ctx.lineTo(W, y); ctx.stroke();
    }
    
    if (state === 'start') {
        drawStartScreen();
        ctx.restore();
        return;
    }
    
    // Base line
    ctx.fillStyle = baseHP > 50 ? '#22cc88' : baseHP > 25 ? '#ffaa00' : '#ff3333';
    ctx.fillRect(0, H - 15, W, 4);
    
    // HP bar
    const hpW = Math.min(W * 0.6, 400);
    const hpX = (W - hpW) / 2;
    ctx.fillStyle = '#1a1a2a';
    ctx.fillRect(hpX, H - 12, hpW, 8);
    ctx.fillStyle = baseHP > 50 ? '#22cc88' : baseHP > 25 ? '#ffaa00' : '#ff3333';
    ctx.fillRect(hpX, H - 12, hpW * (baseHP / maxBaseHP), 8);
    
    // Bricks
    for (const br of bricks) {
        if (!br.alive) continue;
        ctx.fillStyle = br.color;
        ctx.globalAlpha = 0.4 + 0.6 * (br.hp / br.maxHp);
        ctx.fillRect(br.x, br.y, br.w, br.h);
        ctx.globalAlpha = 1;
        ctx.strokeStyle = 'rgba(255,255,255,0.3)';
        ctx.strokeRect(br.x, br.y, br.w, br.h);
    }
    
    // Enemies
    for (const e of enemies) {
        ctx.fillStyle = e.color;
        ctx.beginPath();
        if (e.type === 'boss') {
            // Draw boss as hexagon
            for (let i = 0; i < 6; i++) {
                const angle = (i / 6) * Math.PI * 2;
                const px = e.x + Math.cos(angle) * e.r;
                const py = e.y + Math.sin(angle) * e.r;
                if (i === 0) ctx.moveTo(px, py); else ctx.lineTo(px, py);
            }
            ctx.closePath();
        } else {
            ctx.arc(e.x, e.y, e.r, 0, Math.PI * 2);
        }
        ctx.fill();
        
        // HP bar for enemies
        if (e.hp < e.maxHp) {
            const bw = e.r * 2;
            ctx.fillStyle = '#333';
            ctx.fillRect(e.x - bw/2, e.y - e.r - 8, bw, 4);
            ctx.fillStyle = '#ff4444';
            ctx.fillRect(e.x - bw/2, e.y - e.r - 8, bw * (e.hp / e.maxHp), 4);
        }
    }
    
    // Projectiles
    for (const p of projectiles) {
        ctx.fillStyle = p.enemy ? '#ff6666' : '#ffff44';
        ctx.beginPath();
        ctx.arc(p.x, p.y, p.r, 0, Math.PI * 2);
        ctx.fill();
        // Trail
        ctx.strokeStyle = p.enemy ? 'rgba(255,100,100,0.4)' : 'rgba(255,255,100,0.4)';
        ctx.lineWidth = 3;
        ctx.beginPath();
        ctx.moveTo(p.x, p.y);
        ctx.lineTo(p.x - p.vx * 2, p.y - p.vy * 2);
        ctx.stroke();
    }
    
    // Balls
    for (const b of balls) {
        ctx.fillStyle = '#ffffff';
        ctx.shadowColor = '#aaccff';
        ctx.shadowBlur = 10;
        ctx.beginPath();
        ctx.arc(b.x, b.y, b.r, 0, Math.PI * 2);
        ctx.fill();
        ctx.shadowBlur = 0;
    }
    
    // Paddle
    const py = H - 40;
    const pw = paddle.w;
    ctx.fillStyle = rapidFireTimer > 0 ? '#ffaa00' : '#44ccff';
    ctx.shadowColor = '#44ccff';
    ctx.shadowBlur = expandTimer > 0 ? 15 : 8;
    ctx.beginPath();
    ctx.roundRect(paddle.x - pw/2, py - paddle.h/2, pw, paddle.h, 5);
    ctx.fill();
    ctx.shadowBlur = 0;
    
    // Shield indicator
    if (shieldCount > 0) {
        ctx.strokeStyle = 'rgba(68,170,255,0.5)';
        ctx.lineWidth = 3;
        ctx.beginPath();
        ctx.arc(paddle.x, py, pw/2 + 10, -Math.PI * 0.8, Math.PI * 0.8);
        ctx.stroke();
    }
    
    // Powerups
    for (const pu of powerups) {
        ctx.fillStyle = getPowerupColor(pu.type);
        ctx.beginPath();
        ctx.arc(pu.x, pu.y, pu.r, 0, Math.PI * 2);
        ctx.fill();
        ctx.fillStyle = '#fff';
        ctx.font = 'bold 12px monospace';
        ctx.textAlign = 'center';
        ctx.fillText(getPowerupLabel(pu.type), pu.x, pu.y + 4);
    }
    
    // Particles
    for (const p of particles) {
        ctx.globalAlpha = p.life / p.maxLife;
        ctx.fillStyle = p.color;
        ctx.beginPath();
        ctx.arc(p.x, p.y, p.r * (p.life / p.maxLife), 0, Math.PI * 2);
        ctx.fill();
    }
    ctx.globalAlpha = 1;
    
    // HUD
    ctx.fillStyle = '#ffffff';
    ctx.font = 'bold 16px monospace';
    ctx.textAlign = 'left';
    ctx.fillText(`SCORE: ${score}`, 15, 25);
    ctx.textAlign = 'center';
    ctx.fillText(`WAVE ${wave}/${maxWaves}`, W/2, 25);
    ctx.textAlign = 'right';
    ctx.fillText(`HP: ${Math.max(0, baseHP)}`, W - 15, 25);
    
    if (state === 'gameover') drawGameOver();
    if (state === 'win') drawWinScreen();
    
    ctx.restore();
}

function getPowerupColor(type) {
    switch(type) {
        case 'multi': return '#ff66cc';
        case 'expand': return '#66ff66';
        case 'rapid': return '#ffaa00';
        case 'shield': return '#44aaff';
    }
}

function getPowerupLabel(type) {
    switch(type) {
        case 'multi': return 'M';
        case 'expand': return 'E';
        case 'rapid': return 'R';
        case 'shield': return 'S';
    }
}

function drawStartScreen() {
    ctx.fillStyle = '#ffffff';
    ctx.font = 'bold 48px monospace';

    CanvasRenderingContext2D.prototype.roundRect = function(x, y, w, h, r) {
        this.moveTo(x + r, y);
        this.lineTo(x + w - r, y);
        this.arcTo(x + w, y, x + w, y + r, r);
        this.lineTo(x + w, y + h - r);
        this.arcTo(x + w, y + h, x + w - r, y + h, r);
        this.lineTo(x + r, y + h);
        this.arcTo(x, y + h, x, y + h - r, r);
        this.lineTo(x, y + r);
        this.arcTo(x, y, x + r, y, r);
        this.closePath();
    };
