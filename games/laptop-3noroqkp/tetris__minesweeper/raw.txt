(function(){

const maxW=window.innerWidth, maxH=window.innerHeight;
const gameW=COLS*CELL+160, gameH=ROWS*CELL+40;

W=Math.floor(gameW*scale); H=Math.floor(gameH*scale);
cvs.width=W; cvs.height=H;
document.getElementById('container').style.width=W+'px';
document.getElementById('container').style.height=H+'px';
}

// Audio
let audioCtx=null;
function initAudio(){ if(!audioCtx) audioCtx=new(window.AudioContext||window.webkitAudioContext)(); }
function playSound(freq,dur,type,vol){
if(!audioCtx)return;
const o=audioCtx.createOscillator(), g=audioCtx.createGain();
o.type=type||'square'; o.frequency.value=freq;
g.gain.value=vol||0.15; g.gain.exponentialRampToValueAtTime(0.001,audioCtx.currentTime+dur);
o.connect(g); g.connect(audioCtx.destination);
o.start(); o.stop(audioCtx.currentTime+dur);
}
function sfxMove(){playSound(200,0.05,'square',0.08);}
function sfxRotate(){playSound(400,0.08,'triangle',0.1);}
function sfxDrop(){playSound(150,0.1,'square',0.15);}
function sfxClear(){for(let i=0;i<3;i++)setTimeout(()=>playSound(600+i*200,0.15,'sine',0.12),i*80);}
function sfxBoom(){if(!audioCtx)return;const o=audioCtx.createOscillator(),g=audioCtx.createGain();o.type='sawtooth';o.frequency.setValueAtTime(100,audioCtx.currentTime);o.frequency.exponentialRampToValueAtTime(30,audioCtx.currentTime+0.5);g.gain.value=0.4;g.gain.exponentialRampToValueAtTime(0.001,audioCtx.currentTime+0.6);o.connect(g);g.connect(audioCtx.destination);o.start();o.stop(audioCtx.currentTime+0.6);}
function sfxLevelUp(){for(let i=0;i<5;i++)setTimeout(()=>playSound(400+i*100,0.1,'sine',0.1),i*70);}

// Game state
let grid=[], mines=[], numbers=[];
let piece=null, nextPiece=null;
let score=0, lines=0, level=1, gameOver=false, paused=false, started=false;
let dropTimer=0, lastTime=0;
let particles=[], explosions=[];
let shakeTime=0, shakeMag=0;

const SHAPES=[
{cells:[[0,0],[1,0],[2,0],[3,0]],name:'I',color:'#00e5ff'},
{cells:[[0,0],[1,0],[0,1],[1,1]],name:'O',color:'#ffe500'},
{cells:[[0,0],[1,0],[2,0],[1,1]],name:'T',color:'#b366ff'},
{cells:[[1,0],[2,0],[0,1],[1,1]],name:'S',color:'#00e573'},
{cells:[[0,0],[1,0],[1,1],[2,1]],name:'Z',color:'#ff4444'},
{cells:[[0,0],[0,1],[0,2],[1,1]],name:'J',color:'#4a9eff'},
{cells:[[1,0],[1,1],[1,2],[0,1]],name:'L',color:'#ff8c00'}
];

function randomPiece(){
const s=SHAPES[Math.floor(Math.random()*SHAPES.length)];
return {shape:s, x:Math.floor(COLS/2)-1, y:-2, rot:0};
}

function rotate(cells,rot){
let c=cells.map(r=>[r[0],r[1]]);
for(let i=0;i<((rot%4)+4)%4;i++){c=c.map(([x,y])=>[-y,x+1]);}
const mx=Math.min(...c.map(p=>p[0])), my=Math.min(...c.map(p=>p[1]));
return c.map(([x,y])=>[x-mx,y-my]);
}

function getPieceCells(p){
if(!p)return[];
const r=p.rot%4;
let cells=rotate(p.shape.cells,r);
return cells.map(([cx,cy])=>[p.x+cx,p.y+cy]);
}

function valid(x,y,cells,ignoreMines){
for(const[cx,cy]of cells){
const gx=x+cx, gy=y+cy;
if(gx<0||gx>=COLS||gy>=ROWS)return false;
if(gy>=0&&grid[gy][gx])return false;
}
return true;
}

function validNoMine(x,y,cells){
for(const[cx,cy]of cells){
const gx=x+cx, gy=y+cy;
if(gy>=0&&mines.includes(gy*COLS+gx))return false;
}
return true;
}

function spawnMines(count){
mines=[];
let placed=0;
while(placed<count){
const r=Math.floor(Math.random()*(ROWS-5)); // avoid top 5 rows
const c=Math.floor(Math.random()*COLS);
const idx=r*COLS+c;
if(!mines.includes(idx)&&!grid[r][c]){mines.push(idx);placed++;}
}
updateNumbers();
}

function updateNumbers(){
numbers=[];
for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++){
if(grid[r][c]){
let n=0;
for(let dr=-1;dr<=1;dr++)for(let dc=-1;dc<=1;dc++){
if(dr===0&&dc===0)continue;
const nr=r+dr,nc=c+dc;
if(nr>=0&&nr<ROWS&&nc>=0&&nc<COLS&&mines.includes(nr*COLS+nc))n++;
}
if(n>0)numbers.push({r,c,n});
}
}
}

function newGame(){
grid=Array.from({length:ROWS},()=>Array(COLS).fill(0));
mines=[]; numbers=[]; particles=[]; explosions=[];
score=0; lines=0; level=1; gameOver=false; paused=false;
piece=randomPiece(); nextPiece=randomPiece();
spawnMines(3+level);
dropTimer=0; lastTime=performance.now();
}

function lockPiece(){
const cells=getPieceCells(piece);
let onMine=false;
for(const[cx,cy]of cells){
const gx=piece.x+cx, gy=piece.y+cy;
if(gy>=0&&gy<ROWS&&gx>=0&&gx<COLS){
grid[gy][gx]=1;
if(mines.includes(gy*COLS+gx))onMine=true;
}
}
updateNumbers();
if(onMine){triggerExplosion();return;}
sfxDrop();
checkLines();
piece=randomPiece();
if(!valid(0,0,getPieceCells(piece)))gameOver=true;
}

function checkLines(){
let cleared=0;
for(let r=ROWS-1;r>=0;r--){
if(grid[r].every(v=>v===1)){
cleared++;
sfxClear();
for(let c=0;c<COLS;c++){
for(let i=0;i<3;i++)particles.push({x:(c*CELL+CELL/2)*scale,y:(r*CELL+CELL/2)*scale,vx:(Math.random()-0.5)*4,vy:(Math.random()-1)*3,life:30,color:'#fff',size:3});
}
grid.splice(r,1);
grid.unshift(Array(COLS).fill(0));
}
}
if(cleared>0){
score+=[0,100,300,500,800][cleared]*level;
lines+=cleared;
let nl=Math.floor(lines/10)+1;
if(nl>level){level=nl;sfxLevelUp();spawnMines(2);}
}
}

function triggerExplosion(){
gameOver=true; sfxBoom(); shakeTime=30; shakeMag=8;
const cells=getPieceCells(piece);
for(const[cx,cy]of cells){
const gx=piece.x+cx, gy=piece.y+cy;
if(gy>=0&&gy<ROWS&&gx>=0&&gx<COLS){
explosions.push({x:gx,y:gy,timer:60});
for(let i=0;i<8;i++)particles.push({x:(gx*CELL+CELL/2)*scale,y:(gy*CELL+CELL/2)*scale,vx:(Math.random()-0.5)*8,vy:(Math.random()-0.5)*8,life:40,color:Math.random()>0.5?'#ff4444':'#ffaa00',size:4});
}
}
setTimeout(showGameOver,1200);
}

function showGameOver(){
document.getElementById('go-stats').textContent=`Score: ${score} | Lines: ${lines} | Level: ${level}`;
document.getElementById('gameover-screen').classList.remove('hidden');
}

function hardDrop(){
while(valid(piece.x,piece.y+1,getPieceCells(piece))){piece.y++;}
lockPiece();
}

function softDrop(){
if(valid(piece.x,piece.y+1,getPieceCells(piece)))piece.y++;
else lockPiece();
}

function move(dx){
const nx=piece.x+dx;
if(valid(nx,piece.y,getPieceCells(piece))){piece.x=nx;sfxMove();}
}

function rotatePiece(){
const nr=(piece.rot+1)%4;
const test={...piece,rot:nr};
const cells=getPieceCells(test);
for(const ox of[0,-1,1,-2,2]){
if(valid(piece.x+ox,piece.y,cells)){piece.x+=ox;piece.rot=nr;sfxRotate();break;}
if(valid(piece.x+ox,piece.y-1,cells)){piece.x+=ox;piece.y--;piece.rot=nr;sfxRotate();break;}
}
}

// Rendering
function drawCell(x,y,color,alpha){
ctx.globalAlpha=alpha||1;
const px=x*CELL*scale, py=y*CELL*scale, s=CELL*scale;
ctx.fillStyle=color;
ctx.fillRect(px+1,py+1,s-2,s-2);
ctx.strokeStyle='rgba(255,255,255,0.2)';
ctx.strokeRect(px+1,py+1,s-2,s-2);
ctx.globalAlpha=1;
}

function drawMine(x,y,size){
const px=(x*CELL+CELL/2)*scale, py=(y*CELL+CELL/2)*scale;
const r=size*scale*0.3;
ctx.fillStyle='#ff2222';
ctx.beginPath();ctx.arc(px,py,r,0,Math.PI*2);ctx.fill();
// spikes
for(let a=0;a<8;a++){
const ang=a*Math.PI/4;
ctx.beginPath();
ctx.moveTo(px+Math.cos(ang)*r*0.5, py+Math.sin(ang)*r*0.5);
ctx.lineTo(px+Math.cos(ang)*r*1.2, py+Math.sin(ang)*r*1.2);
ctx.strokeStyle='#ff2222';ctx.lineWidth=2;ctx.stroke();
}
}

function render(){
const ox=(COLS*CELL*scale)/2-((COLS*CELL*scale)/2), oy=0;
ctx.save();
if(shakeTime>0){
shakeTime--;
ctx.translate((Math.random()-0.5)*shakeMag*scale,(Math.random()-0.5)*shakeMag*scale);
}

// Background
ctx.fillStyle='#0a0a1a';
ctx.fillRect(0,0,W,H);

// Grid
const gx=20*scale, gy=20*scale;
for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++){
const px=gx+c*CELL*scale, py=gy+r*CELL*scale;
ctx.strokeStyle='rgba(74,158,255,0.1)';
ctx.strokeRect(px,py,CELL*scale,CELL*scale);
}

// Placed blocks
for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++){
if(grid[r][c]){
drawCell(c,r,'#4a9eff',0.7);
}
}

// Mine explosions
explosions.forEach(e=>{
e.timer--;
const t=e.timer/60;
const px=gx+(e.x*CELL+CELL/2)*scale, py=gy+(e.y*CELL+CELL/2)*scale;
ctx.globalAlpha=t;
ctx.fillStyle='#ff4444';
ctx.beginPath();ctx.arc(px,py,CELL*scale*(1-t)*0.8,0,Math.PI*2);ctx.fill();
ctx.globalAlpha=1;
});
explosions=explosions.filter(e=>e.timer>0);

// Numbers (minesweeper style)
numbers.forEach(n=>{
const px=gx+(n.c*CELL+CELL/2)*scale, py=gy+(n.r*CELL+CELL/2)*scale;
ctx.fillStyle='#ffdd44';
ctx.font=`bold ${12*scale}px Courier New`;
ctx.textAlign='center';ctx.textBaseline='middle';
ctx.fillText(n.n,px,py);
});

// Current piece
if(piece&&!gameOver){
const cells=getPieceCells(piece);
for(const[cx,cy]of cells){
const gx2=piece.x+cx, gy2=piece.y+cy;
if(gy2>=0)drawCell(gx2,gy2,piece.shape.color,0.9);
}
// Ghost
let ghostY=piece.y;
while(valid(piece.x,ghostY+1,getPieceCells(piece)))ghostY++;
const gcells=getPieceCells({...piece,y:ghostY});
for(const[cx,cy]of gcells){
const gx2=piece.x+cx, gy2=ghostY+cy;
if(gy2>=0)drawCell(gx2,gy2,piece.shape.color,0.2);
}
}

// Particles
particles.forEach(p=>{
p.x+=p.vx*scale; p.y+=p.vy*scale; p.vy+=0.3*scale; p.life--;
ctx.globalAlpha=p.life/40;
ctx.fillStyle=p.color;
ctx.fillRect(p.x-p.size,p.y-p.size,p.size*2,p.size*2);
});
ctx.globalAlpha=1;
particles=particles.filter(p=>p.life>0);

// Side panel
const px2=gx+COLS*CELL*scale+15*scale;
ctx.fillStyle='#4a9eff';
ctx.font=`bold ${14*scale}px Courier New`;
ctx.textAlign='left';ctx.textBaseline='top';
ctx.fillText(`SCORE: ${score}`,px2,gy);
ctx.fillText(`LINES: ${lines}`,px2,gy+25*scale);
ctx.fillText(`LEVEL: ${level}`,px2,gy+50*scale);

// Next piece
ctx.fillText('NEXT:',px2,gy+90*scale);
if(nextPiece){
const r=nextPiece.rot%4;

drawCellAt(px2+(cx-mx)*CELL*scale+CELL*scale, gy+110*scale+cy*CELL*scale, nextPiece.shape.color, 0.8);

}

}

function drawCellAt(x,y,color,alpha){
ctx.globalAlpha=alpha;
const s=CELL*scale;
ctx.fillStyle=color;
ctx.fillRect(x+1,y+1,s-2,s-2);
ctx.strokeStyle='rgba(255,255,255,0.2)';
ctx.strokeRect(x+1,y+1,s-2,s-2);
ctx.globalAlpha=1;
}

function gameLoop(time){
if(!started)return;
const dt=time-lastTime; lastTime=time;
if(!paused&&!gameOver){
dropTimer+=dt;
const interval=Math.max(100,500-level*30);
if(dropTimer>interval&&piece){
if(valid(piece.x,piece.y+1,getPieceCells(piece)))piece.y++;
else lockPiece();
dropTimer=0;
}
}
render();
requestAnimationFrame(gameLoop);
}

// Input
document.addEventListener('keydown',e=>{
initAudio();
if(!started)return;
if(e.code==='KeyR'){newGame();return;}
if(e.code==='KeyP'&&!gameOver){paused=!paused;document.getElementById('pause-screen').classList.toggle('hidden',!paused);return;}
if(gameOver||paused)return;
switch(e.code){
case 'ArrowLeft':move(-1);break;
case 'ArrowRight':move(1);break;
case 'ArrowDown':softDrop();dropTimer=0;break;
case 'ArrowUp':rotatePiece();break;
case 'Space':e.preventDefault();hardDrop();dropTimer=0;break;
}
});

document.getElementById('btn-start').addEventListener('click',()=>{
initAudio();
started=true;
document.getElementById('start-screen').classList.add('hidden');
newGame();
requestAnimationFrame(gameLoop);
});

document.getElementById('gameover-screen').classList.add('hidden');
newGame();
});
