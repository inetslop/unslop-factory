// Audio
let audioCtx = null;
function initAudio() {
if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
}
function playBoostSound() {
if (!audioCtx) return;
const osc = audioCtx.createOscillator();
const gain = audioCtx.createGain();
osc.connect(gain); gain.connect(audioCtx.destination);
osc.type = 'sawtooth'; osc.frequency.setValueAtTime(200, audioCtx.currentTime);
osc.frequency.exponentialRampToValueAtTime(600, audioCtx.currentTime + 0.1);
gain.gain.setValueAtTime(0.15, audioCtx.currentTime);
gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.15);
osc.start(); osc.stop(audioCtx.currentTime + 0.15);
}
function playFinishSound() {
if (!audioCtx) return;
const notes = [523, 659, 784, 1047];
notes.forEach((f, i) => {
const osc = audioCtx.createOscillator();
const gain = audioCtx.createGain();
osc.connect(gain); gain.connect(audioCtx.destination);
osc.type = 'sine'; osc.frequency.value = f;
gain.gain.setValueAtTime(0.2, audioCtx.currentTime + i * 0.15);
gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + i * 0.15 + 0.3);
osc.start(audioCtx.currentTime + i * 0.15); osc.stop(audioCtx.currentTime + i * 0.15 + 0.3);
});
}
function playCrashSound() {
if (!audioCtx) return;
const osc = audioCtx.createOscillator();
const gain = audioCtx.createGain();
osc.connect(gain); gain.connect(audioCtx.destination);
osc.type = 'square'; osc.frequency.setValueAtTime(150, audioCtx.currentTime);
osc.frequency.exponentialRampToValueAtTime(50, audioCtx.currentTime + 0.3);
gain.gain.setValueAtTime(0.3, audioCtx.currentTime);
gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.4);
osc.start(); osc.stop(audioCtx.currentTime + 0.4);
}

ctx.fillStyle = groundGrad;
ctx.fillRect(0, topY, W, H - topY);

// Road
ctx.beginPath();

const roadGrad = ctx.createLinearGradient(0, topY, 0, H);

ctx.strokeStyle = '#ff0';

ctx.beginPath(); ctx.moveTo(topRight, topY); ctx.lineTo(botRight, botY); ctx.stroke();

// Lane markings
const laneW_top = (topRight - topLeft) / LANE_COUNT;
const laneW_bot = (botRight - botLeft) / LANE_COUNT;
for (let l = 1; l < LANE_COUNT; l++) {
const tlx = topLeft + laneW_top * l;
const blx = botLeft + laneW_bot * l;
// Dashed lines
const dashLen = 30;
const gap = 20;
const totalDist = H - topY;
const scrollOffset = (roadOffset * 0.5) % (dashLen + gap);
for (let d = 0; d < totalDist; d += dashLen + gap) {
const y1 = topY + d + scrollOffset;
if (y1 > H || y1 + dashLen < topY) continue;
const t1 = Math.max(0, (y1 - topY)) / totalDist;
const t2 = Math.min(1, (y1 + dashLen - topY)) / totalDist;
const x1 = topLeft + (topRight - topLeft) * l * 0.4 + (tlx - topLeft) * t1;
const x2 = topLeft + (topRight - topLeft) * l * 0.4 + (tlx - topLeft) * t2;
// Simplified: just draw perspective-correct dashes
const px1 = topLeft + (botLeft - topLeft) * t1 + (l / LANE_COUNT) * (botRight - botLeft);
const py1 = topY + (H - topY) * t1;
const px2 = topLeft + (botLeft - topLeft) * t2 + (l / LANE_COUNT) * (botRight - botLeft);
const py2 = topY + (H - topY) * t2;
if (py1 < py2 && py2 > topY) {
ctx.strokeStyle = 'rgba(255,255,255,0.6)';
ctx.lineWidth = 3;
ctx.beginPath(); ctx.moveTo(px1, py1); ctx.lineTo(px2, py2); ctx.stroke();
}
}
}

const cxT = (topLeft + topRight) / 2;
const cxBot = (botLeft + botRight) / 2;
ctx.strokeStyle = 'rgba(255,255,0,0.4)';

ctx.fillStyle = '#fff';

const fx = botLeft + (botRight - botLeft) * (i / Math.floor(roadW/15));
ctx.fillRect(fx, fy, 12, 8);
}
}
}

function drawCar(x, y, w, h, color, isPlayer) {
// Shadow
ctx.fillStyle = 'rgba(0,0,0,0.4)';

bodyGrad.addColorStop(0, color);

ctx.fillStyle = bodyGrad;
ctx.beginPath();

ctx.fill();

// Windshield

ctx.fill();
}
}

function shadeColor(color, percent) {
const num = parseInt(color.replace('#',''), 16);
const r = Math.min(255, Math.max(0, (num >> 16) + percent));
const g = Math.min(255, Math.max(0, ((num >> 8) & 0xff) + percent));
const b = Math.min(255, Math.max(0, (num & 0xff) + percent));
return `rgb(${r},${g},${b})`;
}

function drawParticles() {

// Update
let lastTime = 0;
function update(dt) {
if (gameState !== 'playing') return;

raceTime += dt;
timeLeft -= dt;
if (timeLeft <= 0) { endRace(false); return; }

// Player speed decay
player.speed *= Math.pow(0.95, dt * 60);
if (player.speed < 0.1) player.speed = 0;

// Move player
player.dist += player.speed * dt * 8;
roadOffset += player.speed * dt * 40;

// Lane movement
const targetX = getLaneX(player.lane);
player.x += (targetX - player.x) * Math.min(1, dt * 8);

// Check finish
if (player.dist >= raceDistance) { endRace(true); return; }

// Update opponents
opponents.forEach(o => {
if (!o.finished) {
o.dist += o.speed * dt * 8;
if (o.dist >= raceDistance) o.finished = true;
}
// Lane switching AI
if (Math.random() < 0.005 && !o.finished) {
const newLane = Math.max(0, Math.min(LANE_COUNT-1, o.lane + (Math.random()>0.5?1:-1)));
o.lane = newLane;
o.targetX = getLaneX(newLane);
}
o.x += (o.targetX - o.x) * Math.min(1, dt * 4);
});

// Particles
particles.forEach(p => { p.x += p.vx; p.y += p.vy; p.life -= dt * 2; });
particles = particles.filter(p => p.life > 0);
}

function render() {
ctx.clearRect(0, 0, W, H);
drawRoad();

// Draw opponents (far to near by distance)
const sortedOpps = [...opponents].sort((a,b) => b.dist - a.dist);
sortedOpps.forEach(o => {
if (o.finished && o.dist > player.dist + 100) return;
const progress = o.dist / raceDistance;
const y = H * 0.82 - progress * (H * 0.65);
if (y < ROAD_TOP || y > H) return;
const scale = Math.max(0.3, 1 - (y - ROAD_TOP) / (H - ROAD_TOP) * 0.7);
drawCar(o.x, y, 40*scale, 50*scale, o.color, false);
});

// Draw player
const py = H * 0.82;
drawCar(player.x, py, 44, 55, '#ff6b35', true);

drawParticles();
drawHUD();
}

function gameLoop(timestamp) {
const dt = Math.min((timestamp - lastTime) / 1000, 0.05);
lastTime = timestamp;
update(dt);
render();
requestAnimationFrame(gameLoop);
}

// Nitro key
