const canvas = document.getElementById('game');
const ctx = canvas.getContext('2d');

let W, H;
function resize() {
    W = canvas.width = window.innerWidth;
    H = canvas.height = window.innerHeight;
}
window.addEventListener('resize', resize);
resize();

// Audio
let audioCtx = null;
function initAudio() {
    if (!audioCtx) audioCtx = new (window.AudioContext || webkitAudioContext)();
}
function playSound(type) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    if (type === 'hit') {
        osc.type = 'sine';
        osc.frequency.setValueAtTime(220, audioCtx.currentTime);
        osc.frequency.exponentialRampToValueAtTime(110, audioCtx.currentTime + 0.1);
        gain.gain.setValueAtTime(0.3, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.15);
        osc.start(audioCtx.currentTime);
        osc.stop(audioCtx.currentTime + 0.15);
    } else if (type === 'sink') {
        osc.type = 'sine';
        osc.frequency.setValueAtTime(440, audioCtx.currentTime);
        osc.frequency.exponentialRampToValueAtTime(880, audioCtx.currentTime + 0.3);
        gain.gain.setValueAtTime(0.4, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.4);
        osc.start(audioCtx.currentTime);
        osc.stop(audioCtx.currentTime + 0.4);
    } else if (type === 'bounce') {
        osc.type = 'triangle';
        osc.frequency.setValueAtTime(150, audioCtx.currentTime);
        gain.gain.setValueAtTime(0.2, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.08);
        osc.start(audioCtx.currentTime);
        osc.stop(audioCtx.currentTime + 0.08);
    } else if (type === 'par') {
        osc.type = 'sine';
        osc.frequency.setValueAtTime(330, audioCtx.currentTime);
        gain.gain.setValueAtTime(0.2, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.3);
        osc.start(audioCtx.currentTime);
        osc.stop(audioCtx.currentTime + 0.3);
    }
}

// Game State
let gameState = 'start'; // start, playing, levelComplete, gameOver
let currentLevel = 0;
let strokes = 0;
let totalStrokes = 0;
let animFrame = null;
let time = 0;

// Ball
let ball = { x: 0, y: 0, vx: 0, vy: 0, r: 12 };
let isAiming = false;
let aimStart = { x: 0, y: 0 };
let aimEnd = { x: 0, y: 0 };

// Level data
const levels = [
    {
        name: "First Putt",
        ballStart: { x: 0.2, y: 0.5 },
        holes: [{ x: 0.8, y: 0.5 }],
        walls: [],
        slopes: [],
        movers: []
    },
    {

        ballStart: { x: 0.15, y: 0.5 },

        slopes: [],
        movers: []
    },
    {
        name: "Slope City",
        ballStart: { x: 0.15, y: 0.8 },
        holes: [{ x: 0.85, y: 0.2 }],
        walls: [],
        slopes: [{ x: 0.4, y: 0.3, w: 0.2, h: 0.4, fx: 0, fy: -100 }],
        movers: []
    },
    {
        name: "Moving Parts",
        ballStart: { x: 0.15, y: 0.5 },
        holes: [{ x: 0.85, y: 0.5 }],
        walls: [],
        slopes: [],
        movers: [{ x: 0.45, y: 0.3, w: 0.08, h: 0.4, speed: 1, axis: 'y', range: 0.2 }]
    },
    {
        name: "The Gauntlet",
        ballStart: { x: 0.1, y: 0.5 },
        holes: [{ x: 0.9, y: 0.5 }],
        walls: [{ x: 0.3, y: 0.1, w: 0.08, h: 0.4 }, { x: 0.6, y: 0.5, w: 0.08, h: 0.4 }],
        slopes: [{ x: 0.45, y: 0.6, w: 0.15, h: 0.3, fx: -50, fy: 0 }],
        movers: []
    },
    {
        name: "Final Challenge",
        ballStart: { x: 0.1, y: 0.9 },
        holes: [{ x: 0.9, y: 0.1 }],
        walls: [{ x: 0.35, y: 0.4, w: 0.12, h: 0.3 }, { x: 0.6, y: 0.1, w: 0.12, h: 0.3 }],
        slopes: [{ x: 0.5, y: 0.6, w: 0.2, h: 0.3, fx: 80, fy: -50 }],
        movers: [{ x: 0.4, y: 0.7, w: 0.08, h: 0.25, speed: 1.5, axis: 'x', range: 0.15 }]
    }
];

function loadLevel(idx) {
    const lvl = levels[idx];
    ball.x = lvl.ballStart.x * W;
    ball.y = lvl.ballStart.y * H;
    ball.vx = 0;
    ball.vy = 0;
    strokes = 0;
}

function getMoverPos(mover) {
    let mx = mover.x * W;
    let my = mover.y * H;
    if (mover.axis === 'x') {
        mx += Math.sin(time * mover.speed) * mover.range * W;
    } else {
        my += Math.sin(time * mover.speed) * mover.range * H;
    }
    return { x: mx, y: my, w: mover.w * W, h: mover.h * H };
}

function isBallMoving() {
    return Math.abs(ball.vx) > 0.5 || Math.abs(ball.vy) > 0.5;
}

function update(dt) {
    if (gameState !== 'playing') return;
    time += dt;

    // Apply slopes
    const lvl = levels[currentLevel];
    for (const s of lvl.slopes) {
        const sx = s.x * W, sy = s.y * H, sw = s.w * W, sh = s.h * H;
        if (ball.x > sx && ball.x < sx + sw && ball.y > sy && ball.y < sy + sh) {
            ball.vx += s.fx * dt;
            ball.vy += s.fy * dt;
        }
    }

    // Friction
    const friction = 0.985;
    ball.vx *= friction;
    ball.vy *= friction;

    if (Math.abs(ball.vx) < 1 && Math.abs(ball.vy) < 1) {
        ball.vx = 0;
        ball.vy = 0;
    }

    // Move ball
    ball.x += ball.vx * dt;
    ball.y += ball.vy * dt;

    // Wall collisions (static)
    for (const w of lvl.walls) {
        const wx = w.x * W, wy = w.y * H, ww = w.w * W, wh = w.h * H;
        resolveWallCollision(wx, wy, ww, wh);
    }

    // Mover collisions
    for (const m of lvl.movers) {
        const mp = getMoverPos(m);
        resolveWallCollision(mp.x, mp.y, mp.w, mp.h);
    }

    // Boundary collision
    if (ball.x - ball.r < 0) { ball.x = ball.r; ball.vx *= -0.7; playSound('bounce'); }
    if (ball.x + ball.r > W) { ball.x = W - ball.r; ball.vx *= -0.7; playSound('bounce'); }
    if (ball.y - ball.r < 0) { ball.y = ball.r; ball.vy *= -0.7; playSound('bounce'); }
    if (ball.y + ball.r > H) { ball.y = H - ball.r; ball.vy *= -0.7; playSound('bounce'); }

    // Hole capture
    const speed = Math.sqrt(ball.vx * ball.vx + ball.vy * ball.vy);
    for (const h of lvl.holes) {
        const hx = h.x * W, hy = h.y * H;
        const dx = ball.x - hx, dy = ball.y - hy;
        const dist = Math.sqrt(dx * dx + dy * dy);
        if (dist < 25 && speed < 200) {
            // Sink!
            playSound('sink');
            gameState = 'levelComplete';
            return;
        }
        // Gravity pull toward hole
        if (dist < 35 && dist > 0) {
            const force = 30 / (dist * dist) * 100;
            ball.vx -= (dx / dist) * force * dt;
            ball.vy -= (dy / dist) * force * dt;
        }
    }
}

function resolveWallCollision(wx, wy, ww, wh) {
    const bx = ball.x, by = ball.y, br = ball.r;
    const closestX = Math.max(wx, Math.min(bx, wx + ww));
    const closestY = Math.max(wy, Math.min(by, wy + wh));
    const dx = bx - closestX, dy = by - closestY;
    const dist = Math.sqrt(dx * dx + dy * dy);
    if (dist < br) {
        const overlap = br - dist;
        const nx = dx / (dist || 1), ny = dy / (dist || 1);
        ball.x += nx * overlap;
        ball.y += ny * overlap;
        const dot = ball.vx * nx + ball.vy * ny;
        ball.vx -= 2 * dot * nx * 0.8;
        ball.vy -= 2 * dot * ny * 0.8;
        playSound('bounce');
    }
}

function draw() {
    const lvl = levels[currentLevel];
    // Background - green
    ctx.fillStyle = '#2d7a3e';
    ctx.fillRect(0, 0, W, H);

    // Grid lines for texture
    ctx.strokeStyle = 'rgba(255,255,255,0.03)';
    ctx.lineWidth = 1;
    for (let i = 0; i < W; i += 40) {
        ctx.beginPath(); ctx.moveTo(i, 0); ctx.lineTo(i, H); ctx.stroke();
    }
    for (let i = 0; i < H; i += 40) {
        ctx.beginPath(); ctx.moveTo(0, i); ctx.lineTo(W, i); ctx.stroke();
    }

    // Slopes
    for (const s of lvl.slopes) {
        const sx = s.x * W, sy = s.y * H, sw = s.w * W, sh = s.h * H;
        ctx.fillStyle = 'rgba(255, 200, 0, 0.15)';
        ctx.fillRect(sx, sy, sw, sh);
        // Arrows
        const ax = s.fx > 0 ? 1 : -1;
        const ay = s.fy > 0 ? 1 : -1;
        for (let i = 0; i < sw; i += 30) {
            for (let j = 0; j < sh; j += 30) {
                const cx2 = sx + i + 15, cy2 = sy + j + 15;
                ctx.strokeStyle = 'rgba(255, 200, 0, 0.4)';
                ctx.lineWidth = 2;
                ctx.beginPath();
                ctx.moveTo(cx2 - ax * 8, cy2);
                ctx.lineTo(cx2 + ax * 8, cy2);
                ctx.stroke();
                if (ay !== 0) {
                    ctx.beginPath();
                    ctx.moveTo(cx2, cy2 - ay * 8);
                    ctx.lineTo(cx2, cy2 + ay * 8);
                    ctx.stroke();
                }
            }
        }
    }

    // Walls
    for (const w of lvl.walls) {
        const wx = w.x * W, wy = w.y * H, ww = w.w * W, wh = w.h * H;
        ctx.fillStyle = '#5a3e2b';
        ctx.fillRect(wx, wy, ww, wh);
        ctx.strokeStyle = '#3d2a1a';
        ctx.lineWidth = 3;
        ctx.strokeRect(wx, wy, ww, wh);
    }

    // Movers
    for (const m of lvl.movers) {
        const mp = getMoverPos(m);
        ctx.fillStyle = '#8b4513';
        ctx.fillRect(mp.x, mp.y, mp.w, mp.h);
        ctx.strokeStyle = '#5a2d0c';
        ctx.lineWidth = 3;
        ctx.strokeRect(mp.x, mp.y, mp.w, mp.h);
    }

    // Holes
    for (const h of lvl.holes) {
        const hx = h.x * W, hy = h.y * H;
        ctx.beginPath();
        ctx.arc(hx, hy, 22, 0, Math.PI * 2);
        ctx.fillStyle = '#111';
        ctx.fill();
        ctx.beginPath();
        ctx.arc(hx, hy, 25, 0, Math.PI * 2);
        ctx.strokeStyle = 'rgba(255,255,255,0.3)';
        ctx.lineWidth = 2;
        ctx.stroke();
    }

    // Aim line
    if (isAiming) {
        const dx = aimStart.x - aimEnd.x;
        const dy = aimStart.y - aimEnd.y;
        const len = Math.sqrt(dx * dx + dy * dy);
        const power = Math.min(len, 200);

        // Direction arrow
        ctx.strokeStyle = 'rgba(255, 255, 0, 0.8)';
        ctx.lineWidth = 3;
        ctx.setLineDash([8, 6]);
        ctx.beginPath();
        ctx.moveTo(ball.x, ball.y);
        ctx.lineTo(ball.x + (dx / len) * power * 0.5, ball.y + (dy / len) * power * 0.5);
        ctx.stroke();
        ctx.setLineDash([]);

        // Power indicator
        const pw = power / 200;
        ctx.fillStyle = `rgba(${Math.floor(255 * pw)}, ${Math.floor(255 * (1 - pw))}, 0, 0.8)`;
        ctx.fillRect(ball.x - 30, ball.y + 25, 60 * pw, 8);
        ctx.strokeStyle = 'rgba(255,255,255,0.5)';
        ctx.lineWidth = 1;
        ctx.strokeRect(ball.x - 30, ball.y + 25, 60, 8);
    }

    // Ball
    if (gameState === 'playing' || gameState === 'levelComplete') {
        ctx.beginPath();
        ctx.arc(ball.x, ball.y, ball.r, 0, Math.PI * 2);
        const grad = ctx.createRadialGradient(ball.x - 3, ball.y - 3, 2, ball.x, ball.y, ball.r);
        grad.addColorStop(0, '#fff');
        grad.addColorStop(1, '#ddd');
        ctx.fillStyle = grad;
        ctx.fill();
        ctx.strokeStyle = '#999';
        ctx.lineWidth = 1.5;
        ctx.stroke();

        // Shadow
        ctx.beginPath();
        ctx.ellipse(ball.x + 3, ball.y + 4, ball.r * 0.8, ball.r * 0.4, 0, 0, Math.PI * 2);
        ctx.fillStyle = 'rgba(0,0,0,0.2)';
        ctx.fill();
    }

    // HUD
    ctx.fillStyle = 'rgba(0,0,0,0.6)';
    ctx.fillRect(10, 10, 200, 70);
    ctx.fillStyle = '#fff';
    ctx.font = 'bold 18px Segoe UI';
    ctx.fillText(`Level ${currentLevel + 1}: ${lvl.name}`, 20, 35);
    ctx.font = '16px Segoe UI';
    ctx.fillText(`Strokes: ${strokes}`, 20, 60);

    // Overlays
    if (gameState === 'start') drawStartScreen();
    else if (gameState === 'levelComplete') drawLevelComplete();
    else if (gameState === 'gameOver') drawGameOver();
}

function drawStartScreen() {
    ctx.fillStyle = 'rgba(0, 0, 0, 0.7)';
    ctx.fillRect(0, 0, W, H);

    ctx.fillStyle = '#fff';
    ctx.font = 'bold 48px Segoe UI';
    ctx.textAlign = 'center';

    // Bounce animation
    const bounceY = Math.sin(time * 3) * 10;
    ctx.font = '24px Segoe UI';
    ctx.fillStyle = `rgba(255, 255, 100, ${0.5 + Math.sin(time * 2) * 0.3})`;
    ctx.fillText('Click to Start', W / 2, H / 2 + 80 + bounceY);

    ctx.textAlign = 'left';

function drawLevelComplete() {
    ctx.fillStyle = 'rgba(0, 0, 0, 0.6)';
    ctx.fillRect(0, 0, W, H);

    ctx.fillStyle = '#4eff4e';
    ctx.font = 'bold 42px Segoe UI';
    ctx.textAlign = 'center';
    ctx.fillText('🎉 Hole in One... wait, Sunk!', W / 2, H / 3);

    ctx.fillStyle = '#fff';
    ctx.font = '24px Segoe UI';
    ctx.fillText(`Strokes: ${strokes}`, W / 2, H / 3 + 50);

    if (currentLevel < levels.length - 1) {
        const bounceY = Math.sin(time * 3) * 8;
        ctx.fillStyle = '#ffdd44';
        ctx.font = '22px Segoe UI';
        ctx.fillText(`Click for Level ${currentLevel + 2}`, W / 2, H / 2 + 60 + bounceY);
    } else {
        gameState = 'gameOver';
    }

    ctx.textAlign = 'left';
}

function drawGameOver() {
    ctx.fillStyle = 'rgba(0, 0, 0, 0.7)';
    ctx.fillRect(0, 0, W, H);

    ctx.fillStyle = '#ffd700';
    ctx.font = 'bold 42px Segoe UI';
    ctx.textAlign = 'center';
    ctx.fillText('🏆 COMPLETE!', W / 2, H / 3);

    ctx.fillStyle = '#fff';
    ctx.font = '24px Segoe UI';
    ctx.fillText(`Total Strokes: ${totalStrokes}`, W / 2, H / 3 + 50);

    const rating = totalStrokes <= levels.length * 2 ? 'MVP ⭐⭐⭐' : totalStrokes <= levels.length * 3.5 ? 'Pro ⭐⭐' : 'Amateur ⭐';
    ctx.fillText(rating, W / 2, H / 3 + 90);

    const bounceY = Math.sin(time * 3) * 8;
    ctx.fillStyle = '#ffdd44';
    ctx.font = '22px Segoe UI';
    ctx.fillText('Click to Play Again', W / 2, H / 2 + 60 + bounceY);

    ctx.textAlign = 'left';
}

function gameLoop(timestamp) {
    const dt = Math.min((timestamp - lastTime) / 1000, 0.05);
    lastTime = timestamp;
    time += dt;

    update(dt);
    draw();
    animFrame = requestAnimationFrame(gameLoop);
}

let lastTime = 0;

// Input
function getPos(e) {
    const rect = canvas.getBoundingClientRect();
    const x = (e.clientX || (e.touches && e.touches[0].clientX)) - rect.left;
    const y = (e.clientY || (e.touches && e.touches[0].clientY)) - rect.top;
    return { x, y };
}

function onDown(e) {
    initAudio();
    const pos = getPos(e);

    if (gameState === 'start') {
        gameState = 'playing';
        currentLevel = 0;
        totalStrokes = 0;
        loadLevel(0);
        return;
    } else if (gameState === 'levelComplete') {
        if (currentLevel < levels.length - 1) {
            currentLevel++;
            loadLevel(currentLevel);
            gameState = 'playing';
        } else {
            gameState = 'gameOver';
        }
        return;
    } else if (gameState === 'gameOver') {
        gameState = 'start';
        return;
    }

    if (gameState === 'playing' && !isBallMoving()) {
        const dx = pos.x - ball.x, dy = pos.y - ball.y;
        if (Math.sqrt(dx * dx + dy * dy) < 80) {
            isAiming = true;
            aimStart = { x: ball.x, y: ball.y };
            aimEnd = pos;
        }
    }
}

function onMove(e) {
    if (!isAiming) return;
    const pos = getPos(e);
    aimEnd = pos;
}

function onUp(e) {
    if (!isAiming) return;
    isAiming = false;
    const dx = aimStart.x - aimEnd.x;
    const dy = aimStart.y - aimEnd.y;
    const len = Math.sqrt(dx * dx + dy * dy);
    if (len < 10) return;

    const power = Math.min(len, 200);
    ball.vx = (dx / len) * power * 3;
    ball.vy = (dy / len) * power * 3;
    strokes++;
    totalStrokes++;
    playSound('hit');
}

window.addEventListener('mousedown', onDown);
window.addEventListener('mousemove', onMove);
window.addEventListener('mouseup', onUp);
window.addEventListener('touchstart', e => { e.preventDefault(); onDown(e.touches[0]); }, { passive: false });
window.addEventListener('touchmove', e => { e.preventDefault(); onMove(e.touches[0]); }, { passive: false });
window.addEventListener('touchend', e => { e.preventDefault(); onUp(e); }, { passive: false });

// Keyboard restart
window.addEventListener('keydown', e => {
    if (e.key === 'r' || e.key === 'R') {
        if (gameState === 'playing') {
            loadLevel(currentLevel);
        } else if (gameState === 'start') {
            gameState = 'playing';
            currentLevel = 0;
            totalStrokes = 0;
            loadLevel(0);
        }
    }
    if (e.key === 'Enter' || e.key === ' ') {
        if (gameState === 'start') {
            initAudio();
            gameState = 'playing';
            currentLevel = 0;
            totalStrokes = 0;
            loadLevel(0);
        } else if (gameState === 'levelComplete') {
            if (currentLevel < levels.length - 1) {
                currentLevel++;
                loadLevel(currentLevel);
                gameState = 'playing';
            }
        } else if (gameState === 'gameOver') {
            gameState = 'start';
        }
    }
});

// Start loop
loadLevel(0);
animFrame = requestAnimationFrame(gameLoop);
</script>
