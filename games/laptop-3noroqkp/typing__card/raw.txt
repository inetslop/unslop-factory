const canvas = document.getElementById('c');
const ctx = canvas.getContext('2d');

let W, H;
function resize() {
  W = canvas.width = window.innerWidth;
  H = canvas.height = window.innerHeight;
}
window.addEventListener('resize', resize);
resize();

const WORDS = [
  "fire","wind","star","moon","tree","lake","wolf","bird","fish","rock",
  "sand","rain","snow","frost","ember","storm","shadow","crystal","thunder",
  "dawn","dusk","gale","peak","vale","grove","cove","heath","moor","clay",
  "moss","fern","reed","thorn","briar","rowan","yew","oak","elm","ash",
  "loam","peat","sedge","hawthorn","lichen","mist","gold","dark","light",
  "flame","raven","falcon","heron","otter","badger","lynx","wren","lark",
  "swift","robin","finch","dove","eagle","osprey","heron","stork","crane"
];

const COLORS = ["red","blue","green","yellow"];
const COLOR_HEX = { red:"#e74c3c", blue:"#3498db", green:"#2ecc71", yellow:"#f1c40f" };
const WILD_COLOR = "#2c3e50";

let gameState = 'start'; // start, playing, won, lost
let playerHand = [];
let aiHand = [];
let discardPile = [];
let drawPile = [];
let currentColor = "red";
let input = "";
let turn = 'player'; // player or ai
let aiTimer = 0;
let aiTyping = "";
let particles = [];
let msg = "";
let msgTimer = 0;
let score = { wins: 0, losses: 0 };
let shakeTime = 0;

function shuffle(arr) {
  for (let i = arr.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    [arr[i], arr[j]] = [arr[j], arr[i]];
  }
  return arr;
}

function randomWord() { return WORDS[Math.floor(Math.random() * WORDS.length)]; }

function makeCard() {
  if (Math.random() < 0.08) return { word: "*", color: "wild" };
  return { word: randomWord(), color: COLORS[Math.floor(Math.random() * 4)] };
}

function initGame() {
  drawPile = [];
  for (let i = 0; i < 60; i++) drawPile.push(makeCard());
  shuffle(drawPile);
  
  playerHand = [];
  aiHand = [];
  for (let i = 0; i < 7; i++) {
    playerHand.push(drawPile.pop());
    aiHand.push(drawPile.pop());
  }
  
  discardPile = [drawPile.pop()];
  currentColor = discardPile[discardPile.length - 1].color === "wild" ? "red" : discardPile[discardPile.length - 1].color;
  
  input = "";
  turn = 'player';
  aiTimer = 0;
  aiTyping = "";
  particles = [];
  msg = "";
  msgTimer = 0;
  shakeTime = 0;
}

function drawPileCard() {
  if (drawPile.length === 0) {
    // reshuffle discard into draw
    let top = discardPile.pop();
    while (discardPile.length > 0) drawPile.push(discardPile.pop());
    shuffle(drawPile);
  }
  return drawPile.pop();
}

function canPlay(card, color) {
  if (card.color === "wild") return true;
  return card.color === color;
}

function playCard(hand, index) {
  const card = hand.splice(index, 1)[0];
  discardPile.push(card);
  if (card.color !== "wild") currentColor = card.color;
  addParticles(W/2, H/2, card.color === "wild" ? "#9b59b6" : COLOR_HEX[card.color]);
  return card;
}

function checkWin() {
  if (playerHand.length === 0) { gameState = 'won'; score.wins++; }
  if (aiHand.length === 0) { gameState = 'lost'; score.losses++; }
}

function addParticles(x, y, color) {
  for (let i = 0; i < 12; i++) {
    const angle = Math.random() * Math.PI * 2;
    const speed = 2 + Math.random() * 4;
    particles.push({ x, y, vx: Math.cos(angle)*speed, vy: Math.sin(angle)*speed - 2, life: 1, color });
  }
}

function showMsg(text) { msg = text; msgTimer = 60; }

// Audio
let audioCtx = null;
function initAudio() {
  if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
}
function playSound(freq, dur, type) {
  if (!audioCtx) return;
  const osc = audioCtx.createOscillator();
  const gain = audioCtx.createGain();
  osc.type = type || 'sine';
  osc.frequency.value = freq;
  gain.gain.setValueAtTime(0.15, audioCtx.currentTime);
  gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
  osc.connect(gain);
  gain.connect(audioCtx.destination);
  osc.start();
  osc.stop(audioCtx.currentTime + dur);
}

function keydown(e) {
  if (gameState === 'start') {
    initAudio();
    gameState = 'playing';
    initGame();
    return;
  }
  if (gameState === 'won' || gameState === 'lost') {
    if (e.key === 'Enter' || e.key === ' ') {
      gameState = 'playing';
      initGame();
    }
    return;
  }
  if (gameState !== 'playing') return;
  if (turn !== 'player') return;
  
  if (e.key === 'Backspace') { input = input.slice(0, -1); e.preventDefault(); return; }
  if (e.key === 'Enter') { e.preventDefault(); return; }
  if (e.key.length === 1 && /[a-z*]/i.test(e.key)) {
    const ch = e.key.toLowerCase();
    if (ch === '*') input += '*';
    else input += ch;
    e.preventDefault();
    tryPlayInput();
  }
}

window.addEventListener('keydown', keydown);

function tryPlayInput() {
  for (let i = 0; i < playerHand.length; i++) {
    if (playerHand[i].word.toLowerCase() === input && canPlay(playerHand[i], currentColor)) {
      playCard(playerHand, i);
      input = "";
      playSound(520, 0.15, 'triangle');
      checkWin();
      if (gameState === 'playing') {
        turn = 'ai';
        aiTimer = 90; // frames (~1.5s at 60fps)
      }
      return;
    }
  }
  // No match - flash error
  shakeTime = 8;
  playSound(150, 0.1, 'square');
}

function aiTurn() {
  if (gameState !== 'playing') return;
  
  // AI tries to play a card
  let played = false;
  for (let i = 0; i < aiHand.length; i++) {
    if (canPlay(aiHand[i], currentColor)) {
      playCard(aiHand, i);
      played = true;
      playSound(400, 0.12, 'triangle');
      break;
    }
  }
  
  if (!played) {
    // Draw a card
    const c = drawPileCard();
    aiHand.push(c);
    showMsg("AI draws a card");
  }
  
  checkWin();
  if (gameState === 'playing') {
    turn = 'player';
    input = "";
  }
}

function update() {
  if (msgTimer > 0) msgTimer--;
  if (shakeTime > 0) shakeTime--;
  
  // Update particles
  for (let i = particles.length - 1; i >= 0; i--) {
    const p = particles[i];
    p.x += p.vx;
    p.y += p.vy;
    p.vy += 0.15;
    p.life -= 0.02;
    if (p.life <= 0) particles.splice(i, 1);
  }
  
  if (gameState !== 'playing') return;
  
  if (turn === 'ai') {
    aiTimer--;
    if (aiTimer <= 0) aiTurn();
  }
}

function drawCard(x, y, w, h, card, faceUp) {
  ctx.save();
  
  const color = card.color === "wild" ? WILD_COLOR : COLOR_HEX[card.color];
  
  // Card background
  ctx.fillStyle = '#fff';
  roundRect(x-2, y-2, w+4, h+4, 8);
  ctx.fill();
  
  ctx.fillStyle = color;
  roundRect(x, y, w, h, 6);
  ctx.fill();
  
  if (faceUp) {
    // Word
    ctx.fillStyle = '#fff';
    ctx.font = `bold ${Math.min(14, w/8)}px monospace`;
    ctx.textAlign = 'center';
    ctx.textBaseline = 'middle';
    
    const word = card.word === "*" ? "*" : card.word;
    // Fit text
    let fontSize = Math.min(14, (w-10) / (word.length * 0.6));
    ctx.font = `bold ${fontSize}px monospace`;
    ctx.fillText(word, x + w/2, y + h/2);
    
    // Color label
    if (card.color !== "wild") {
      ctx.font = `${Math.max(8, h/5)}px sans-serif`;
      ctx.fillStyle = 'rgba(255,255,255,0.7)';
      const labels = { red:"R", blue:"B", green:"G", yellow:"Y" };
      ctx.fillText(labels[card.color], x + w - 12, y + 12);
    } else {
      ctx.font = `bold ${h/3}px sans-serif`;
      ctx.fillStyle = '#f1c40f';
      ctx.textAlign = 'center';
      ctx.fillText("★", x + w/2, y + h - 12);
    }
  } else {
    // Card back pattern
    ctx.fillStyle = 'rgba(255,255,255,0.1)';
    for (let i = 0; i < 3; i++) {
      roundRect(x + 8, y + 8 + i * (h/3), w - 16, h/4, 3);
      ctx.fill();
    }
  }
  
  ctx.restore();
}

function roundRect(x, y, w, h, r) {
  ctx.beginPath();
  ctx.moveTo(x+r, y);
  ctx.lineTo(x+w-r, y);
  ctx.quadraticCurveTo(x+w, y, x+w, y+r);
  ctx.lineTo(x+w, y+h-r);
  ctx.quadraticCurveTo(x+w, y+h, x+w-r, y+h);
  ctx.lineTo(x+r, y+h);
  ctx.quadraticCurveTo(x, y+h, x, y+h-r);
  ctx.lineTo(x, y+r);
  ctx.quadraticCurveTo(x, y, x+r, y);
  ctx.closePath();
}

function draw() {
  ctx.fillStyle = '#1a1a2e';
  ctx.fillRect(0, 0, W, H);
  
  if (gameState === 'start') {
    drawStartScreen();
    return;
  }
  
  if (gameState === 'won' || gameState === 'lost') {
    drawEndScreen();
    return;
  }
  
  // Shake effect
  let sx = 0, sy = 0;
  if (shakeTime > 0) {
    sx = (Math.random()-0.5) * shakeTime;
    sy = (Math.random()-0.5) * shakeTime;
  }
  
  ctx.save();
  ctx.translate(sx, sy);
  
  const cardW = Math.min(90, W/12);
  const cardH = cardW * 1.4;
  
  // AI hand (top)
  const aiStartX = W/2 - (aiHand.length * (cardW + 8)) / 2;
  ctx.fillStyle = '#fff';
  ctx.font = 'bold 14px sans-serif';
  ctx.textAlign = 'left';
  ctx.fillText(`AI: ${aiHand.length} cards`, 20, 30);
  
  for (let i = 0; i < aiHand.length; i++) {
    drawCard(aiStartX + i * (cardW + 8), 45, cardW, cardH, aiHand[i], false);
  }
  
  // Discard pile (right)
  const dispX = W - cardW - 30;
  const dispY = H/2 - cardH/2;
  
  if (discardPile.length > 0) {
    drawCard(dispX, dispY, cardW, cardH, discardPile[discardPile.length-1], true);
    // Glow for current color
    ctx.strokeStyle = currentColor === 'wild' ? '#9b59b6' : COLOR_HEX[currentColor];
    ctx.lineWidth = 3;
    ctx.globalAlpha = 0.5 + Math.sin(Date.now()/300)*0.3;
    roundRect(dispX-4, dispY-4, cardW+8, cardH+8, 10);
    ctx.stroke();
    ctx.globalAlpha = 1;
  }
  
  // Current color indicator
  ctx.fillStyle = currentColor === 'wild' ? '#9b59b6' : COLOR_HEX[currentColor];
  ctx.font = 'bold 16px sans-serif';
  ctx.textAlign = 'center';
  const colorName = currentColor.charAt(0).toUpperCase() + currentColor.slice(1);
  ctx.fillText(`Need: ${colorName}`, dispX + cardW/2, dispY - 15);
  
  // Draw pile count (bottom right)
  ctx.fillStyle = '#aaa';
  ctx.font = '12px sans-serif';
  ctx.textAlign = 'right';
  ctx.fillText(`Draw: ${drawPile.length}`, W - 30, H - 20);
  
  // Player hand (bottom)
  const pStartX = W/2 - (playerHand.length * (cardW + 10)) / 2;
  const pY = H - cardH - 50;
  
  for (let i = 0; i < playerHand.length; i++) {
    const cx = pStartX + i * (cardW + 10);
    const highlight = canPlay(playerHand[i], currentColor);
    if (highlight) {
      ctx.shadowColor = COLOR_HEX[currentColor] === WILD_COLOR ? '#9b59b6' : (playerHand[i].color === 'wild' ? '#9b59b6' : COLOR_HEX[playerHand[i].color]);
      ctx.shadowBlur = 10;
    }
    drawCard(cx, pY, cardW, cardH, playerHand[i], true);
    ctx.shadowBlur = 0;
  }
  
  // Input area
  const inputY = H - 25;
  ctx.fillStyle = '#2a2a4e';
  roundRect(W/2 - 150, inputY - 18, 300, 32, 6);
  ctx.fill();
  
  ctx.fillStyle = turn === 'player' ? '#fff' : '#666';
  ctx.font = 'bold 18px monospace';
  ctx.textAlign = 'center';
  ctx.fillText(input + (turn === 'player' ? '▌' : ''), W/2, inputY + 2);
  
  // Turn indicator
  ctx.fillStyle = turn === 'player' ? '#2ecc71' : '#e74c3c';
  ctx.font = 'bold 13px sans-serif';
  ctx.textAlign = 'center';
  ctx.fillText(turn === 'player' ? "YOUR TURN" : "AI THINKING...", W/2, inputY - 30);
  
  // Message
  if (msgTimer > 0 && msg) {
    ctx.fillStyle = `rgba(255,255,255,${msgTimer/60})`;
    ctx.font = 'bold 18px sans-serif';
    ctx.textAlign = 'center';
    ctx.fillText(msg, W/2, H/2);
  }
  
  // Particles
  for (const p of particles) {
    ctx.globalAlpha = p.life;
    ctx.fillStyle = p.color;
    ctx.beginPath();
    ctx.arc(p.x, p.y, 4 * p.life, 0, Math.PI*2);
    ctx.fill();
  }
  ctx.globalAlpha = 1;
  
  // Score (top center)
  ctx.fillStyle = '#aaa';
  ctx.font = '12px sans-serif';
  ctx.textAlign = 'center';
  ctx.fillText(`W:${score.wins} L:${score.losses}`, W/2, 20);
  
  ctx.restore();
}

function drawStartScreen() {
  ctx.fillStyle = '#fff';
  ctx.font = `bold ${Math.min(48, W/12)}px sans-serif`;
  ctx.textAlign = 'center';
  ctx.fillText("TYPE CARD", W/2, H/3);
  
  ctx.font = `${Math.min(16, W/30)}px sans-serif`;
  ctx.fillStyle = '#ccc';
  const lines = [
    "Match the current color by typing words on your cards.",
    "",
    "• Type a word that's on one of your cards",
    "• The card's color must match the required color",
    "• ★ Wild cards work with any color",
    "• First to empty their hand wins!",
    "",
    "Press any key to start"
  ];
  lines.forEach((line, i) => {
    ctx.fillText(line, W/2, H/3 + 50 + i * 28);
  });
}

function drawEndScreen() {
  const won = gameState === 'won';
  ctx.fillStyle = won ? '#2ecc71' : '#e74c3c';
  ctx.font = `bold ${Math.min(48, W/12)}px sans-serif`;
  ctx.textAlign = 'center';
  ctx.fillText(won ? "YOU WIN!" : "AI WINS", W/2, H/3);
  
  ctx.fillStyle = '#ccc';
  ctx.font = `${Math.min(16, W/30)}px sans-serif`;
  ctx.fillText(`Score: ${score.wins}W - ${score.losses}L`, W/2, H/3 + 50);
  ctx.fillText("Press Enter to play again", W/2, H/3 + 90);
  
  // Particles on win
  if (won && Math.random() < 0.3) {
    addParticles(Math.random()*W, H/4, `hsl(${Math.random()*360},80%,60%)`);
  }
}

function loop() {
  update();
  draw();
  requestAnimationFrame(loop);
}

initGame();
gameState = 'start';
loop();
</script>
