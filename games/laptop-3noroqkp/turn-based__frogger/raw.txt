<!DOCTYPE html>

<script>
const canvas = document.getElementById('game');
const ctx = canvas.getContext('2d');

let cellSize, boardX, boardY, boardW, boardH;
let gameState = 'start'; // start, playing, win, lose
let frog = { x: 6, y: 8 };
let turn = 0;
let score = 0;
let highScore = 0;
let deathTimer = 0;
let winAnimation = 0;

const COLS = 13;
const ROWS = 9;

// Define lanes
const LANE_TYPES = {
    LILY: 'lily',
    CROC: 'croc',
    SAFE: 'safe',
    TRAFFIC_R: 'traffic_r',
    TRAFFIC_L: 'traffic_l'
};

// Row definitions
const rowTypes = [
    LANE_TYPES.LILY,      // 0 - goal
    LANE_TYPES.CROC,     // 1
    LANE_TYPES.SAFE,     // 2
    LANE_TYPES.TRAFFIC_R,// 3
    LANE_TYPES.TRAFFIC_L,// 4
    LANE_TYPES.SAFE,     // 5
    LANE_TYPES.CROC,     // 6
    LANE_TYPES.TRAFFIC_R,// 7
    LANE_TYPES.SAFE      // 8 - start
];

// Lily pad positions (goal)
const lilyPads = [1, 3, 5, 7, 9, 11];

// Cars per lane
let cars = [];
// Crocs
let crocs = [];

function initGame() {
    frog = { x: 6, y: 8 };
    turn = 0;
    score = 0;
    deathTimer = 0;
    winAnimation = 0;
    
    // Initialize cars
    // Row 3: moving right, 2 cars spaced out
    cars = [
        { row: 3, x: -1, speed: 1 },
        { row: 3, x: 5, speed: 1 },
        { row: 4, x: 12, speed: -1 },
        { row: 4, x: 6, speed: -1 },
        { row: 7, x: -2, speed: 1 }
    ];
    
    // Initialize crocs
    crocs = [
        { row: 1, x: 2, dir: 1 },
        { row: 1, x: 9, dir: -1 },
        { row: 6, x: 5, dir: 1 }
    ];
}

function moveHazards() {
    // Move cars
    for (let car of cars) {
        car.x += car.speed;
        if (car.speed > 0 && car.x >= COLS) car.x = -2;
        if (car.speed < 0 && car.x < -1) car.x = COLS + 1;
    }
    // Move crocs (they bounce off walls)
    for (let c of crocs) {
        c.x += c.dir;
        if (c.x <= 0 || c.x >= COLS - 1) c.dir *= -1;
    }
}

function checkCollision() {
    // Check car collision
    for (let car of cars) {
        if (car.row === frog.y && Math.abs(car.x - frog.x) < 1.5) return true;
    }
    // Check croc collision
    for (let c of crocs) {
        if (c.row === frog.y && Math.abs(c.x - frog.x) < 1.2) return true;
    }
    // Check if on water without lily pad or safe row
    const rt = rowTypes[frog.y];
    if (rt === LANE_TYPES.CROC) {
        // Must be on a croc? No, must NOT be on croc but it's water... 
        // In turn-based, you can stand in water safely as long as no croc hits you
        // Actually let's say: if you're on a CROC row and not adjacent to safe, you're fine
        // Water is survivable, only crocs kill
    }
    return false;
}

function checkWin() {
    if (frog.y === 0 && lilyPads.includes(frog.x)) return true;
    return false;
}

function playerMove(dx, dy) {
    if (gameState !== 'playing') return;
    
    let nx = frog.x + dx;
    let ny = frog.y + dy;
    
    // Bounds check
    if (nx < 0 || nx >= COLS || ny < 0 || ny >= ROWS) return;
    
    frog.x = nx;
    frog.y = ny;
    turn++;
    
    // Move hazards
    moveHazards();
    
    // Check collision
    if (checkCollision()) {
        gameState = 'lose';
        deathTimer = 60;
        return;
    }
    
    // Check win
    if (checkWin()) {
        gameState = 'win';
        score = Math.max(0, 100 - turn * 5) + Math.floor(Math.random() * 20);
        if (score > highScore) highScore = score;
        try { localStorage.setItem('frogger_high', highScore); } catch(e) {}
        winAnimation = 0;
        return;
    }
}

function resize() {
    const w = window.innerWidth;
    const h = window.innerHeight;
    canvas.width = w;
    canvas.height = h;
    
    cellSize = Math.min(Math.floor(w / (COLS + 2)), Math.floor((h - 100) / (ROWS + 2)));
    boardW = COLS * cellSize;
    boardH = ROWS * cellSize;
    boardX = Math.floor((w - boardW) / 2);
    boardY = Math.floor((h - boardH) / 2) + 20;
}

window.addEventListener('resize', resize);
resize();

// Load high score
try { highScore = parseInt(localStorage.getItem('frogger_high')) || 0; } catch(e) {}

// Input
window.addEventListener('keydown', (e) => {
    if (gameState === 'start') {
        initGame();
        gameState = 'playing';
        return;
    }
    if (gameState === 'lose' || gameState === 'win') {
        initGame();
        gameState = 'playing';
        return;
    }
    if (gameState !== 'playing') return;
    
    switch(e.key) {
        case 'ArrowUp': case 'w': case 'W': playerMove(0, -1); break;
        case 'ArrowDown': case 's': case 'S': playerMove(0, 1); break;
        case 'ArrowLeft': case 'a': case 'A': playerMove(-1, 0); break;
        case 'ArrowRight': case 'd': case 'D': playerMove(1, 0); break;
    }
    e.preventDefault();
});

function drawCell(x, y, color) {
    ctx.fillStyle = color;
    ctx.fillRect(boardX + x * cellSize, boardY + y * cellSize, cellSize, cellSize);
}

function drawBoard() {
    // Draw rows based on type
    for (let y = 0; y < ROWS; y++) {
        const rt = rowTypes[y];
        if (rt === LANE_TYPES.LILY) {
            for (let x = 0; x < COLS; x++) {
                if (lilyPads.includes(x)) {
                    drawCell(x, y, '#2d8a4e');
                    // Draw lily pad
                    ctx.fillStyle = '#1a5c30';
                    ctx.beginPath();
                    ctx.arc(boardX + x * cellSize + cellSize/2, boardY + y * cellSize + cellSize/2, cellSize * 0.35, 0, Math.PI * 2);
                    ctx.fill();
                    ctx.fillStyle = '#4caf50';
                    ctx.beginPath();
                    ctx.arc(boardX + x * cellSize + cellSize/2, boardY + y * cellSize + cellSize/2, cellSize * 0.25, 0, Math.PI * 2);
                    ctx.fill();
                } else {
                    drawCell(x, y, '#1565a8');
                }
            }
        } else if (rt === LANE_TYPES.CROC) {
            for (let x = 0; x < COLS; x++) {
                drawCell(x, y, '#1976d2');
                // Water ripples
                ctx.strokeStyle = 'rgba(255,255,255,0.1)';
                ctx.lineWidth = 1;
                for (let i = 0; i < 3; i++) {
                    const rx = boardX + x * cellSize + cellSize * 0.2 + i * cellSize * 0.3;
                    const ry = boardY + y * cellSize + cellSize * 0.5 + Math.sin(turn * 0.1 + i) * 2;
                    ctx.beginPath();
                    ctx.arc(rx, ry, cellSize * 0.1, 0, Math.PI);
                    ctx.stroke();
                }
            }
        } else if (rt === LANE_TYPES.SAFE) {
            for (let x = 0; x < COLS; x++) {
                drawCell(x, y, '#4a7c2f');
            }
        } else { // Traffic
            for (let x = 0; x < COLS; x++) {
                drawCell(x, y, '#3a3a3a');
                // Road markings
                if (x % 2 === 0) {
                    ctx.fillStyle = '#ffeb3b';
                    ctx.fillRect(boardX + x * cellSize + cellSize/2 - 2, boardY + y * cellSize + cellSize/2, 4, 3);
                }
            }
        }
    }
    
    // Draw cars
    for (let car of cars) {
        if (car.x < -1 || car.x >= COLS) continue;
        const cx = boardX + car.x * cellSize;
        const cy = boardY + car.row * cellSize;
        
        // Car body
        ctx.fillStyle = car.speed > 0 ? '#e53935' : '#1e88e5';
        ctx.fillRect(cx + cellSize * 0.1, cy + cellSize * 0.2, cellSize * 0.8, cellSize * 0.6);
        // Car top
        ctx.fillStyle = car.speed > 0 ? '#c62828' : '#1565c0';
        ctx.fillRect(cx + cellSize * 0.25, cy + cellSize * 0.3, cellSize * 0.5, cellSize * 0.4);
        // Windows
        ctx.fillStyle = '#90caf9';
        ctx.fillRect(cx + cellSize * 0.3, cy + cellSize * 0.32, cellSize * 0.15, cellSize * 0.2);
        ctx.fillRect(cx + cellSize * 0.55, cy + cellSize * 0.32, cellSize * 0.15, cellSize * 0.2);
    }
    
    // Draw crocs
    for (let c of crocs) {
        const cx = boardX + c.x * cellSize;
        const cy = boardY + c.row * cellSize;
        
        ctx.fillStyle = '#5d8a3c';
        ctx.beginPath();
        ctx.ellipse(cx + cellSize/2, cy + cellSize/2, cellSize * 0.4, cellSize * 0.25, 0, 0, Math.PI * 2);
        ctx.fill();
        // Head
        const headX = c.dir > 0 ? cx + cellSize * 0.7 : cx + cellSize * 0.3;
        ctx.fillStyle = '#4a7c2f';
        ctx.beginPath();
        ctx.arc(headX, cy + cellSize/2, cellSize * 0.18, 0, Math.PI * 2);
        ctx.fill();
        // Eyes
        ctx.fillStyle = '#ffeb3b';
        ctx.beginPath();
        ctx.arc(headX, cy + cellSize * 0.4, cellSize * 0.06, 0, Math.PI * 2);
        ctx.fill();
    }
    
    // Draw frog
    if (gameState === 'playing' || gameState === 'win') {
        const fx = boardX + frog.x * cellSize;
        const fy = boardY + frog.y * cellSize;
        
        ctx.fillStyle = '#4caf50';
        ctx.beginPath();
        ctx.arc(fx + cellSize/2, fy + cellSize/2, cellSize * 0.35, 0, Math.PI * 2);
        ctx.fill();
        // Eyes
        ctx.fillStyle = '#fff';
        ctx.beginPath();
        ctx.arc(fx + cellSize * 0.38, fy + cellSize * 0.35, cellSize * 0.1, 0, Math.PI * 2);
        ctx.arc(fx + cellSize * 0.62, fy + cellSize * 0.35, cellSize * 0.1, 0, Math.PI * 2);
        ctx.fill();
        ctx.fillStyle = '#000';
        ctx.beginPath();
        ctx.arc(fx + cellSize * 0.38, fy + cellSize * 0.35, cellSize * 0.05, 0, Math.PI * 2);
        ctx.arc(fx + cellSize * 0.62, fy + cellSize * 0.35, cellSize * 0.05, 0, Math.PI * 2);
        ctx.fill();
    }
    
    if (gameState === 'lose') {
        // X eyes
        const fx = boardX + frog.x * cellSize;
        const fy = boardY + frog.y * cellSize;
        ctx.fillStyle = '#4caf50';
        ctx.beginPath();
        ctx.arc(fx + cellSize/2, fy + cellSize/2, cellSize * 0.35, 0, Math.PI * 2);
        ctx.fill();
        ctx.strokeStyle = '#000';
        ctx.lineWidth = 2;
        const ex = fx + cellSize * 0.38, ey = fy + cellSize * 0.35;
        ctx.beginPath();
        ctx.moveTo(ex - 4, ey - 4); ctx.lineTo(ex + 4, ey + 4);
        ctx.moveTo(ex + 4, ey - 4); ctx.lineTo(ex - 4, ey + 4);
        ctx.stroke();
        const ex2 = fx + cellSize * 0.62;
        ctx.beginPath();
        ctx.moveTo(ex2 - 4, ey - 4); ctx.lineTo(ex2 + 4, ey + 4);
        ctx.moveTo(ex2 + 4, ey - 4); ctx.lineTo(ex2 - 4, ey + 4);
        ctx.stroke();
    }
}

function drawHUD() {
    ctx.fillStyle = 'rgba(0,0,0,0.7)';
    ctx.fillRect(0, 0, canvas.width, 40);
    
    ctx.fillStyle = '#fff';
    ctx.font = `bold ${Math.min(18, cellSize * 0.6)}px monospace`;
    ctx.textAlign = 'left';
    ctx.fillText(`TURN: ${turn}`, 20, 27);
    ctx.textAlign = 'center';
    ctx.fillText(`HIGH SCORE: ${highScore}`, canvas.width / 2, 27);
    ctx.textAlign = 'right';
    ctx.fillText('FROGGER: TURN-BASED', canvas.width - 20, 27);
}

function drawStartScreen() {
    ctx.fillStyle = '#1a1a2e';
    ctx.fillRect(0, 0, canvas.width, canvas.height);
    
    const cx = canvas.width / 2;
    const cy = canvas.height / 2;
    
    ctx.fillStyle = '#4caf50';
    ctx.font = `bold ${Math.min(48, cellSize * 1.5)}px monospace`;
    ctx.textAlign = 'center';
    ctx.fillText('FROGGER', cx, cy - 80);
    
    ctx.fillStyle = '#fff';
    ctx.font = `${Math.min(20, cellSize * 0.7)}px monospace`;
    ctx.fillText('TURN-BASED EDITION', cx, cy - 50);
    
    ctx.fillStyle = '#aaa';
    ctx.font = `${Math.min(16, cellSize * 0.55)}px monospace`;
    ctx.fillText('Guide the frog to a lily pad in the top row.', cx, cy + 10);
    ctx.fillText('Each turn: you move, then all hazards advance.', cx, cy + 40);
    ctx.fillText('', cx, cy + 60);
    ctx.fillStyle = '#8bc34a';
    ctx.fillText('Controls: Arrow Keys / WASD', cx, cy + 70);
    ctx.fillStyle = '#ffeb3b';
    ctx.fillText('Avoid cars and crocodiles!', cx, cy + 100);
    ctx.fillText('Reach a lily pad (row 0) to win.', cx, cy + 130);
    
    // Blinking prompt
    if (Math.floor(Date.now() / 500) % 2 === 0) {
        ctx.fillStyle = '#fff';
        ctx.font = `bold ${Math.min(22, cellSize * 0.8)}px monospace`;
        ctx.fillText('PRESS ANY KEY TO START', cx, cy + 190);
    }
    
    if (highScore > 0) {
        ctx.fillStyle = '#ffeb3b';
        ctx.font = `${Math.min(16, cellSize * 0.5)}px monospace`;
        ctx.fillText(`High Score: ${highScore}`, cx, cy + 230);
    }
}

function drawEndScreen(win) {
    // Dim overlay
    ctx.fillStyle = 'rgba(0,0,0,0.6)';
    ctx.fillRect(0, 0, canvas.width, canvas.height);
    
    const cx = canvas.width / 2;
    const cy = canvas.height / 2;
    
    if (win) {
        winAnimation += 0.05;
        ctx.fillStyle = '#4caf50';
        ctx.font = `bold ${Math.min(48, cellSize * 1.5)}px monospace`;
        ctx.textAlign = 'center';
        const bounce = Math.sin(winAnimation * 3) * 10;
        ctx.fillText('FROG CROSSED!', cx, cy - 40 + bounce);
        
        ctx.fillStyle = '#fff';
        ctx.font = `${Math.min(24, cellSize * 0.8)}px monospace`;
        ctx.fillText(`Turns: ${turn}`, cx, cy + 10);
        ctx.fillText(`Score: ${score}`, cx, cy + 50);
        
        if (score >= highScore && score > 0) {
            ctx.fillStyle = '#ffeb3b';
            ctx.font = `bold ${Math.min(20, cellSize * 0.7)}px monospace`;
            ctx.fillText('NEW HIGH SCORE!', cx, cy + 90);
        }
    } else {
        ctx.fillStyle = '#e53935';
        ctx.font = `bold ${Math.min(48, cellSize * 1.5)}px monospace`;
        ctx.textAlign = 'center';
        ctx.fillText('SPLAT!', cx, cy - 40);
        
        ctx.fillStyle = '#fff';
        ctx.font = `${Math.min(20, cellSize * 0.7)}px monospace`;
        ctx.fillText(`You survived ${turn} turns.`, cx, cy + 10);
    }
    
    if (Math.floor(Date.now() / 500) % 2 === 0) {
        ctx.fillStyle = '#fff';
        ctx.font = `${Math.min(18, cellSize * 0.6)}px monospace`;
        ctx.fillText('PRESS ANY KEY TO RESTART', cx, cy + 140);
    }
}

function render() {
    ctx.clearRect(0, 0, canvas.width, canvas.height);
    
    if (gameState === 'start') {
        drawStartScreen();
    } else {
        // Background
        ctx.fillStyle = '#263238';
        ctx.fillRect(0, 0, canvas.width, canvas.height);
        
        drawBoard();
        drawHUD();
        
        if (gameState === 'lose') {
            deathTimer--;
            if (deathTimer <= 0) gameState = 'end_lose';
        }
        if (gameState === 'win' || gameState === 'end_win') {
            // handled by win flag
        }
        
        if (gameState === 'end_lose') drawEndScreen(false);
        if (gameState === 'end_win') drawEndScreen(true);
    }
    
    requestAnimationFrame(render);
}

// Handle end states for restart
window.addEventListener('keydown', (e) => {
    if (gameState === 'end_lose' || gameState === 'end_win') {
        initGame();
        gameState = 'playing';
    }
}, true);

// Fix: consolidate state handling
function update() {
    if (gameState === 'lose' && deathTimer <= 0) {
        gameState = 'end_lose';
    }
}

setInterval(update, 50);

initGame();
requestAnimationFrame(render);
</script>
