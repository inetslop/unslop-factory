const canvas = document.getElementById('game');
const ctx = canvas.getContext('2d');

let W, H, cellW, cellH;
const COLS = 8;
const ROWS = 10;

function resize() {
    W = window.innerWidth;
    H = window.innerHeight;
    canvas.width = W;
    canvas.height = H;
    cellW = W / (COLS + 2);
    cellH = H / (ROWS + 2);
}
window.addEventListener('resize', resize);
resize();

// Game state
let gameState = 'start'; // start, playing, gameover, win
let score = 0;
let highScore = 0;
let lives = 3;
let frogCol = 3;
let frogRow = ROWS - 1;
let frogAlive = true;
let frogTimer = 0;
let safeZones = [];
let obstacles = [];
let particles = [];
let stars = [];
let scorePopups = [];

try { highScore = parseInt(localStorage.getItem('spaceFroggerHigh') || '0'); } catch(e) {}

function initStars() {
    stars = [];
    for (let i = 0; i < 80; i++) {
        stars.push({ x: Math.random() * W, y: Math.random() * H, size: Math.random() * 2 + 0.5, speed: Math.random() * 0.3 + 0.1 });
    }
}

function initGame() {
    score = 0;
    lives = 3;
    frogCol = 3;
    frogRow = ROWS - 1;
    frogAlive = true;
    frogTimer = 0;
    particles = [];
    scorePopups = [];
    
    // Safe zones at top (row 0) - positions: 0,2,4,6 (4 zones)
    safeZones = [0, 2, 4, 6].map(c => ({ col: c, filled: false }));
    
    // Obstacle lanes
    obstacles = [];
    const laneConfigs = [
        { row: 1, speed: 1.5, dir: 1, type: 'alien', size: 1 },
        { row: 2, speed: 0.8, dir: -1, type: 'ship', size: 1 },
        { row: 3, speed: 2.2, dir: 1, type: 'ufo', size: 1 },
        { row: 4, speed: 1.2, dir: -1, type: 'alien', size: 1 },
        { row: 5, speed: 0.6, dir: 1, type: 'ship', size: 2 },
        { row: 6, speed: 1.8, dir: -1, type: 'ufo', size: 1 },
        { row: 7, speed: 1.0, dir: 1, type: 'alien', size: 1 },
        { row: 8, speed: 2.5, dir: -1, type: 'ship', size: 1 },
    ];
    
    laneConfigs.forEach(cfg => {
        let count = cfg.type === 'ship' ? (cfg.size > 1 ? 3 : 4) : 5;
        for (let i = 0; i < count; i++) {
            obstacles.push({
                row: cfg.row,
                x: Math.random() * (W - cellW),
                speed: cfg.speed * (W / 800),
                dir: cfg.dir,
                type: cfg.type,
                size: cfg.size
            });
        }
    });
}

function drawStars() {
    ctx.fillStyle = '#0a0a1a';
    ctx.fillRect(0, 0, W, H);
    stars.forEach(s => {
        s.y += s.speed;
        if (s.y > H) { s.y = 0; s.x = Math.random() * W; }
        ctx.fillStyle = `rgba(255,255,255,${0.3 + s.size * 0.2})`;
        ctx.fillRect(s.x, s.y, s.size, s.size);
    });
}

function drawAlien(x, y, size) {
    const s = cellW * 0.6 * size;
    ctx.fillStyle = '#ff4444';
    // Body
    ctx.beginPath();
    ctx.arc(x + s/2, y + s*0.4, s*0.35, 0, Math.PI*2);
    ctx.fill();
    // Eyes
    ctx.fillStyle = '#ffff00';
    ctx.beginPath();
    ctx.arc(x + s*0.3, y + s*0.35, s*0.1, 0, Math.PI*2);
    ctx.arc(x + s*0.7, y + s*0.35, s*0.1, 0, Math.PI*2);
    ctx.fill();
    // Legs
    ctx.strokeStyle = '#ff4444';
    ctx.lineWidth = 2;
    ctx.beginPath();
    ctx.moveTo(x + s*0.2, y + s*0.7);
    ctx.lineTo(x + s*0.1, y + s*0.95);
    ctx.moveTo(x + s*0.8, y + s*0.7);
    ctx.lineTo(x + s*0.9, y + s*0.95);
    ctx.stroke();
}

function drawShip(x, y, size) {
    const s = cellW * 0.6 * size;
    ctx.fillStyle = '#4488ff';
    // Hull
    ctx.beginPath();
    ctx.moveTo(x + s*0.5, y);
    ctx.lineTo(x + s, y + s*0.7);
    ctx.lineTo(x + s*0.3, y + s*0.7);
    ctx.closePath();
    ctx.fill();
    // Window
    ctx.fillStyle = '#aaddff';
    ctx.beginPath();
    ctx.arc(x + s*0.55, y + s*0.4, s*0.15, 0, Math.PI*2);
    ctx.fill();
}

function drawUFO(x, y, size) {
    const s = cellW * 0.6 * size;
    ctx.fillStyle = '#88ff44';
    // Disc
    ctx.beginPath();
    ctx.ellipse(x + s/2, y + s*0.5, s*0.5, s*0.25, 0, 0, Math.PI*2);
    ctx.fill();
    // Dome
    ctx.fillStyle = '#ccffaa';
    ctx.beginPath();
    ctx.arc(x + s/2, y + s*0.35, s*0.2, 0, Math.PI);
    ctx.fill();
    // Lights
    for (let i = 0; i < 4; i++) {
        ctx.fillStyle = `hsl(${(Date.now()/20 + i*90) % 360}, 100%, 70%)`;
        ctx.beginPath();
        ctx.arc(x + s*(0.15 + i*0.23), y + s*0.5, s*0.06, 0, Math.PI*2);
        ctx.fill();
    }
}

function drawFrog(x, y) {
    const s = cellW * 0.7;
    // Body
    ctx.fillStyle = '#44ff44';
    ctx.beginPath();
    ctx.arc(x + s/2, y + s*0.55, s*0.35, 0, Math.PI*2);
    ctx.fill();
    // Head
    ctx.beginPath();
    ctx.arc(x + s/2, y + s*0.3, s*0.25, 0, Math.PI*2);
    ctx.fill();
    // Eyes
    ctx.fillStyle = '#ffffff';
    ctx.beginPath();
    ctx.arc(x + s*0.38, y + s*0.25, s*0.1, 0, Math.PI*2);
    ctx.arc(x + s*0.62, y + s*0.25, s*0.1, 0, Math.PI*2);
    ctx.fill();
    ctx.fillStyle = '#000000';
    ctx.beginPath();
    ctx.arc(x + s*0.38, y + s*0.27, s*0.05, 0, Math.PI*2);
    ctx.arc(x + s*0.62, y + s*0.27, s*0.05, 0, Math.PI*2);
    ctx.fill();
    // Legs
    ctx.fillStyle = '#33cc33';
    ctx.beginPath();
    ctx.ellipse(x + s*0.25, y + s*0.75, s*0.12, s*0.08, 0, 0, Math.PI*2);
    ctx.ellipse(x + s*0.75, y + s*0.75, s*0.12, s*0.08, 0, 0, Math.PI*2);
    ctx.fill();
}

function drawSafeZone(z, x, y) {
    const s = cellW * 0.8;
    if (z.filled) {
        ctx.fillStyle = 'rgba(255,215,0,0.3)';
        ctx.fillRect(x + s*0.1, y + s*0.1, s*0.8, s*0.8);
        ctx.fillStyle = '#ffd700';
        ctx.font = `bold ${s*0.4}px monospace`;
        ctx.textAlign = 'center';
        ctx.fillText('★', x + s/2, y + s*0.65);
    } else {
        ctx.strokeStyle = 'rgba(255,215,0,0.6)';
        ctx.lineWidth = 2;
        ctx.strokeRect(x + s*0.1, y + s*0.1, s*0.8, s*0.8);
        ctx.fillStyle = 'rgba(255,215,0,0.2)';
        ctx.font = `${s*0.3}px monospace`;
        ctx.textAlign = 'center';
        ctx.fillText('HOME', x + s/2, y + s*0.6);
    }
}

function drawGrid() {
    const ox = cellW;
    const oy = cellH;
    
    // Lane backgrounds
    for (let r = 1; r < ROWS - 1; r++) {
        const y = oy + r * cellH;
        if (r % 2 === 0) {
            ctx.fillStyle = 'rgba(30,30,60,0.4)';
            ctx.fillRect(ox, y, COLS * cellW, cellH);
        } else {
            ctx.fillStyle = 'rgba(10,20,50,0.3)';
            ctx.fillRect(ox, y, COLS * cellW, cellH);
        }
    }
    
    // Bottom (start area)
    const by = oy + (ROWS - 1) * cellH;
    ctx.fillStyle = 'rgba(20,40,20,0.5)';
    ctx.fillRect(ox, by, COLS * cellW, cellH);
    
    // Top safe zone area
    const ty = oy;
    ctx.fillStyle = 'rgba(40,30,10,0.5)';
    ctx.fillRect(ox, ty, COLS * cellW, cellH);
    
    // Safe zones
    safeZones.forEach(z => drawSafeZone(z, ox + z.col * cellW, ty));
}

function drawObstacles() {
    const oy = cellH;
    obstacles.forEach(o => {
        const y = oy + o.row * cellH;
        if (o.type === 'alien') drawAlien(o.x - cellW*0.3, y, o.size);
        else if (o.type === 'ship') drawShip(o.x - cellW*0.3, y, o.size);
        else if (o.type === 'ufo') drawUFO(o.x - cellW*0.3, y, o.size);
    });
}

function drawFrog() {
    if (!frogAlive) return;
    const ox = cellW + frogCol * cellW;
    const oy = cellH + frogRow * cellH;
    drawFrog(ox, oy);
}

function drawParticles() {
    particles.forEach(p => {
        ctx.fillStyle = p.color;
        ctx.globalAlpha = p.life / p.maxLife;
        ctx.fillRect(p.x - p.size/2, p.y - p.size/2, p.size, p.size);
        ctx.globalAlpha = 1;
    });
}

function drawScorePopups() {
    scorePopups.forEach(sp => {
        ctx.fillStyle = sp.color;
        ctx.globalAlpha = sp.life / sp.maxLife;
        ctx.font = `bold ${sp.size}px monospace`;
        ctx.textAlign = 'center';
        ctx.fillText(sp.text, sp.x, sp.y);
        ctx.globalAlpha = 1;
    });
}

    ctx.fillStyle = '#ffffff';

function explode(x, y, color) {
    for (let i = 0; i < 12; i++) {
        const angle = Math.random() * Math.PI * 2;
        const speed = Math.random() * 3 + 1;
        particles.push({
            x: x, y: y,
            vx: Math.cos(angle) * speed,
            vy: Math.sin(angle) * speed,
            size: Math.random() * 6 + 2,
            color: color,
            life: 30, maxLife: 30
        });
    }
}

function updateObstacles(dt) {
    const margin = cellW;
    obstacles.forEach(o => {
        o.x += o.speed * o.dir * dt * 60;
        if (o.dir > 0 && o.x > W + cellW) o.x = margin - cellW * 0.5;
        if (o.dir < 0 && o.x < margin - cellW * 2) o.x = W + cellW * 0.5;
    });
}

function checkCollision() {
    const fx = cellW + frogCol * cellW + cellW * 0.15;
    const fy = cellH + frogRow * cellH + cellH * 0.15;
    const fw = cellW * 0.7;
    const fh = cellH * 0.7;
    
    for (let o of obstacles) {
        if (o.row !== frogRow) continue;
        const ox = o.x - cellW * 0.3 + cellW * 0.15;
        const oy2 = cellH + o.row * cellH + cellH * 0.15;
        const ow = cellW * 0.7;
        const oh = cellH * 0.7;
        
        if (fx < ox + ow && fx + fw > ox && fy < oy2 + oh && fy + fh > oy2) {
            return true;
        }
    }
    return false;
}

function checkSafeZone() {
    if (frogRow === 0) {
        for (let z of safeZones) {
            if (z.col === frogCol && !z.filled) {
                z.filled = true;
                score += 1000;
                scorePopups.push({ x: cellW + frogCol * cellW + cellW/2, y: cellH, text: '+1000', color: '#ffd700', size: 24, life: 60, maxLife: 60 });
                return true;
            }
        }
    }
    return false;
}

function checkAllZonesFilled() {
    return safeZones.every(z => z.filled);
}

function moveFrog(dr, dc) {
    if (!frogAlive) return;
    let nr = frogRow + dr;
    let nc = frogCol + dc;
    
    if (nc < 0 || nc >= COLS) return; // Don't go off screen sides
    if (nr < 0 || nr >= ROWS) return;
    
    frogRow = nr;
    frogCol = nc;
    
    if (checkSafeZone()) {
        if (checkAllZonesFilled()) {
            gameState = 'win';
            saveHighScore();
        } else {
            frogRow = ROWS - 1;
            frogCol = 3;
        }
    }
    
    // Check collision after move
    setTimeout(() => {
        if (checkCollision()) {
            killFrog();
        }
    }, 50);
}

function killFrog() {
    if (!frogAlive) return;
    frogAlive = false;
    const fx = cellW + frogCol * cellW + cellW/2;
    const fy = cellH + frogRow * cellH + cellH/2;
    explode(fx, fy, '#44ff44');
    lives--;
    
    if (lives <= 0) {
        gameState = 'gameover';
        saveHighScore();
    } else {
        setTimeout(() => {
            frogRow = ROWS - 1;
            frogCol = 3;
            frogAlive = true;
        }, 800);
    }
}

function saveHighScore() {
    if (score > highScore) {
        highScore = score;
        try { localStorage.setItem('spaceFroggerHigh', highScore.toString()); } catch(e) {}
    }
}

let lastTime = 0;
function gameLoop(time) {
    const dt = Math.min((time - lastTime) / 16.67, 3);
    lastTime = time;
    
    if (gameState === 'start') {
        drawStartScreen();
    } else if (gameState === 'playing') {
        updateObstacles(dt);
        
        // Update particles
        for (let i = particles.length - 1; i >= 0; i--) {
            const p = particles[i];
            p.x += p.vx;
            p.y += p.vy;
            p.life--;
            if (p.life <= 0) particles.splice(i, 1);
        }
        
        // Update score popups
        for (let i = scorePopups.length - 1; i >= 0; i--) {
            const sp = scorePopups[i];
            sp.y -= 1;
            sp.life--;
            if (sp.life <= 0) scorePopups.splice(i, 1);
        }
        
        // Continuous collision check
        if (frogAlive && frogRow >= 1 && frogRow <= ROWS - 2) {
            const fx = cellW + frogCol * cellW;
            const fy = cellH + frogRow * cellH;
            for (let o of obstacles) {
                if (o.row === frogRow) {
                    const ox = o.x - cellW * 0.3;
                    if (Math.abs((fx + cellW/2) - (ox + cellW*0.5)) < cellW * 0.4) {
                        killFrog();
                        break;
                    }
                }
            }
        }
        
        drawStars();
        drawGrid();
        drawObstacles();
        drawFrog();
        drawParticles();
        drawScorePopups();
        drawHUD();
    } else if (gameState === 'gameover') {
        drawGameOver();
    } else if (gameState === 'win') {
        drawWinScreen();
    }
    
    requestAnimationFrame(gameLoop);
}

// Input
window.addEventListener('keydown', e => {
    if (e.code === 'Space' || e.code === 'Enter') {
        e.preventDefault();
        if (gameState === 'start') {
            gameState = 'playing';
            initGame();
            initStars();
        } else if (gameState === 'gameover') {
            gameState = 'start';
        } else if (gameState === 'win') {
            // Next level - increase speeds
            obstacles.forEach(o => o.speed *= 1.2);
            gameState = 'playing';
            frogRow = ROWS - 1;
            frogCol = 3;
            frogAlive = true;
        }
    }
    
    if (gameState !== 'playing' || !frogAlive) return;
    
    switch(e.code) {
        case 'ArrowUp': e.preventDefault(); moveFrog(-1, 0); break;
        case 'ArrowDown': e.preventDefault(); moveFrog(1, 0); break;
        case 'ArrowLeft': e.preventDefault(); moveFrog(0, -1); break;
        case 'ArrowRight': e.preventDefault(); moveFrog(0, 1); break;
    }
