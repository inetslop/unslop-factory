// Audio
let audioCtx = null;
function initAudio() {
    if (!audioCtx) audioCtx = new (window.AudioContext || webkitAudioContext)();
}
function playSound(type) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    if (type === 'shoot') {
        osc.type = 'square';
        osc.frequency.setValueAtTime(800, audioCtx.currentTime);
        osc.frequency.exponentialRampToValueAtTime(200, audioCtx.currentTime + 0.1);
        gain.gain.setValueAtTime(0.15, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.1);
        osc.start(); osc.stop(audioCtx.currentTime + 0.1);
    } else if (type === 'explosion') {
        osc.type = 'sawtooth';
        osc.frequency.setValueAtTime(200, audioCtx.currentTime);
        osc.frequency.exponentialRampToValueAtTime(50, audioCtx.currentTime + 0.3);
        gain.gain.setValueAtTime(0.25, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.3);
        osc.start(); osc.stop(audioCtx.currentTime + 0.3);
    } else if (type === 'hurt') {
        osc.type = 'sawtooth';
        osc.frequency.setValueAtTime(300, audioCtx.currentTime);
        osc.frequency.exponentialRampToValueAtTime(80, audioCtx.currentTime + 0.2);
        gain.gain.setValueAtTime(0.2, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.2);
        osc.start(); osc.stop(audioCtx.currentTime + 0.2);
    } else if (type === 'jump') {
        osc.type = 'sine';
        osc.frequency.setValueAtTime(300, audioCtx.currentTime);
        osc.frequency.exponentialRampToValueAtTime(600, audioCtx.currentTime + 0.1);
        gain.gain.setValueAtTime(0.1, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.15);
        osc.start(); osc.stop(audioCtx.currentTime + 0.15);
    } else if (type === 'wave') {
        osc.type = 'square';
        osc.frequency.setValueAtTime(400, audioCtx.currentTime);
        osc.frequency.setValueAtTime(500, audioCtx.currentTime + 0.1);
        osc.frequency.setValueAtTime(600, audioCtx.currentTime + 0.2);
        gain.gain.setValueAtTime(0.15, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.3);
        osc.start(); osc.stop(audioCtx.currentTime + 0.3);
    } else if (type === 'powerup') {
        osc.type = 'sine';
        osc.frequency.setValueAtTime(500, audioCtx.currentTime);
        osc.frequency.exponentialRampToValueAtTime(1200, audioCtx.currentTime + 0.2);
        gain.gain.setValueAtTime(0.2, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.25);
        osc.start(); osc.stop(audioCtx.currentTime + 0.25);
    }
}

function resetPlayer() {
    player = {
        x: W / 2, y: H - 60, w: 24, h: 32,
        vx: 0, vy: 0, hp: 5, maxHp: 5,

    };

// Platforms
let platforms = [];
function generatePlatforms() {
    platforms = [
        { x: 0, y: H - 20, w: W, h: 20 }, // ground
        { x: W * 0.1, y: H - 100, w: W * 0.2, h: 14 },
        { x: W * 0.7, y: H - 100, w: W * 0.2, h: 14 },
        { x: W * 0.35, y: H - 180, w: W * 0.3, h: 14 },
        { x: W * 0.05, y: H - 260, w: W * 0.18, h: 14 },
        { x: W * 0.75, y: H - 260, w: W * 0.2, h: 14 },

    ];

// Enemies
let enemies = [];
let enemyFormationTimer = 0;
let enemyMoveDir = 1;
let enemySpeed = 1;
let enemyDescendAmount = 0;

function spawnWave(w) {
    enemies = [];
    const rows = Math.min(3 + w, 6);
    const cols = Math.min(5 + w, 8);
    const spacingX = Math.max(40, W / (cols + 2));
    const spacingY = 35;
    const startX = W / 2 - (cols * spacingX) / 2;
    const startY = 60;
    
    for (let r = 0; r < rows; r++) {
        for (let c = 0; c < cols; c++) {
            let hp = 1 + Math.floor(w / 3);
            if (r === 0) hp += 2; // front row tougher
            enemies.push({
                x: startX + c * spacingX, y: startY + r * spacingY,
                w: 28, h: 24, hp: hp, maxHp: hp,
                row: r, col: c, alive: true,
                type: r === 0 ? 'tank' : (r < 3 ? 'soldier' : 'fly')
            });
        }
    }
    enemyMoveDir = 1;
    enemyFormationTimer = 0;
    enemySpeed = 0.5 + w * 0.15;
}

// Particles
let particles = [];
function spawnParticles(x, y, color, count) {
    for (let i = 0; i < count; i++) {
        const angle = Math.random() * Math.PI * 2;
        const speed = 1 + Math.random() * 3;
        particles.push({
            x: x, y: y, vx: Math.cos(angle) * speed, vy: Math.sin(angle) * speed - 1,
            life: 30 + Math.random() * 20, maxLife: 50, color: color
        });
    }
}

// Stars background
let stars = [];
function generateStars() {
    stars = [];
    for (let i = 0; i < 80; i++) {
        stars.push({ x: Math.random() * W, y: Math.random() * H, size: Math.random() * 2 + 0.5, speed: Math.random() * 0.3 + 0.1 });
    }
}

function startGame() {
    gameState = 'playing';
    score = 0;
    wave = 1;
    frameCount = 0;
    playerBullets = [];
    enemyBullets = [];
    particles = [];
    resetPlayer();
    generatePlatforms();
    spawnWave(1);
    generateStars();
}

// Game update
function update() {
    if (gameState !== 'playing') return;
    frameCount++;
    
    // Player movement
    const speed = 4;
    if (keys['ArrowLeft'] || keys['KeyA']) { player.vx = -speed; player.facing = 'left'; }
    else if (keys['ArrowRight'] || keys['KeyD']) { player.vx = speed; player.facing = 'right'; }
    else player.vx = 0;
    
    if ((keys['ArrowUp'] || keys['KeyW'] || keys['Space']) && player.onGround) {
        player.vy = -12;
        player.onGround = false;
        playSound('jump');
    }
    
    // Gravity
    player.vy += 0.6;
    if (player.vy > 15) player.vy = 15;
    
    player.x += player.vx;
    player.y += player.vy;
    
    // Platform collision
    player.onGround = false;
    for (let p of platforms) {
        if (player.x + player.w > p.x && player.x < p.x + p.w) {
            if (player.vy >= 0 && player.y + player.h >= p.y && player.y + player.h <= p.y + p.h + player.vy + 2) {
                player.y = p.y - player.h;
                player.vy = 0;
                player.onGround = true;
            }
        }
    }
    
    // Bounds
    if (player.x < 0) player.x = 0;
    if (player.x + player.w > W) player.x = W - player.w;
    if (player.y + player.h > H - 20) { player.y = H - 20 - player.h; player.vy = 0; player.onGround = true; }
    
    // Invincibility timer
    if (player.invincible > 0) player.invincible--;
    
    // Shooting
    if (player.shootCooldown > 0) player.shootCooldown--;
    if ((keys['KeyX'] || keys['KeyZ'] || keys['KeyF']) && player.shootCooldown <= 0) {
        let dir = 'up';
        if (player.facing === 'left' || player.facing === 'right') {
            if (!keys['ArrowUp'] && !keys['KeyW'] && !keys['Space']) dir = player.facing;
        }
        
        const bx = player.x + player.w / 2 - 3;
        const by = player.y - (dir === 'up' ? player.h : 0);
        let bvx = 0, bvy = 0;
        if (dir === 'up') { bvy = -8; }
        else if (dir === 'left') { bvx = -7; by = player.y + player.h / 2 - 3; }
        else if (dir === 'right') { bvx = 7; by = player.y + player.h / 2 - 3; }
        
        playerBullets.push({ x: bx, y: by, w: 6, h: 12, vx: bvx, vy: bvy });
        if (player.doubleShot) {
            playerBullets.push({ x: bx + (dir === 'left' ? 8 : -8), y: by, w: 6, h: 12, vx: bvx, vy: bvy });
        }
        player.shootCooldown = dir === 'up' ? 12 : 15;
        playSound('shoot');
    }
    
    // Update player bullets
    for (let i = playerBullets.length - 1; i >= 0; i--) {
        const b = playerBullets[i];
        b.x += b.vx;
        b.y += b.vy;
        if (b.y < -20 || b.x < -20 || b.x > W + 20) playerBullets.splice(i, 1);
    }
    
    // Enemy formation movement
    enemyFormationTimer++;
    const aliveEnemies = enemies.filter(e => e.alive);
    if (aliveEnemies.length > 0 && enemyFormationTimer % Math.max(3, 20 - wave * 2) === 0) {
        let edgeX = 0;
        for (let e of aliveEnemies) {
            if (e.x > edgeX) edgeX = e.x + e.w;
        }
        const minX = Math.min(...aliveEnemies.map(e => e.x));
        
        if ((enemyMoveDir === 1 && edgeX > W - 40) || (enemyMoveDir === -1 && minX < 20)) {
            enemyMoveDir *= -1;
            for (let e of enemies) { if (e.alive) e.y += 20; }
        }
        
        const moveAmount = enemySpeed * (aliveEnemies.length > 10 ? 1 : 1.5);
        for (let e of enemies) {
            if (e.alive) e.x += moveAmount * enemyMoveDir;
        }
    }
    
    // Enemy shooting
    if (frameCount % Math.max(20, 60 - wave * 5) === 0 && aliveEnemies.length > 0) {
        const shooters = aliveEnemies.filter(e => e.row >= 2);
        if (shooters.length > 0) {
            const shooter = shooters[Math.floor(Math.random() * shooters.length)];
            const spread = (Math.random() - 0.5) * 3;
            enemyBullets.push({
                x: shooter.x + shooter.w / 2, y: shooter.y + shooter.h,
                w: 6, h: 10, vx: spread, vy: 4 + wave * 0.3
            });
        }
    }
    
    // Update enemy bullets
    for (let i = enemyBullets.length - 1; i >= 0; i--) {
        const b = enemyBullets[i];
        b.x += b.vx;
        b.y += b.vy;
        if (b.y > H || b.x < -20 || b.x > W + 20) { enemyBullets.splice(i, 1); continue; }
        
        // Hit player
        if (player.invincible <= 0 &&
            b.x < player.x + player.w && b.x + b.w > player.x &&
            b.y < player.y + player.h && b.y + b.h > player.y) {
            enemyBullets.splice(i, 1);
            player.hp--;
            player.invincible = 60;
            playSound('hurt');
            spawnParticles(player.x + player.w/2, player.y + player.h/2, '#ff4444', 8);
            if (player.hp <= 0) {
                gameState = 'gameover';
                try {
                    if (score > highScore) { highScore = score; localStorage.setItem('platformInvadersHigh', highScore); }
                } catch(e) {}
            }
        }
    }
    
    // Bullet-enemy collision
    for (let i = playerBullets.length - 1; i >= 0; i--) {
        const b = playerBullets[i];
        for (let j = enemies.length - 1; j >= 0; j--) {
            const e = enemies[j];
            if (!e.alive) continue;
            if (b.x < e.x + e.w && b.x + b.w > e.x && b.y < e.y + e.h && b.y + b.h > e.y) {
                playerBullets.splice(i, 1);
                e.hp--;
                if (e.hp <= 0) {
                    e.alive = false;
                    score += e.type === 'tank' ? 30 : e.type === 'soldier' ? 20 : 10;
                    playSound('explosion');
                    spawnParticles(e.x + e.w/2, e.y + e.h/2, '#44ff88', 12);
                } else {
                    playSound('shoot');
                }
                break;
            }
        }
    }
    
    // Check wave clear
    if (aliveEnemies.length === 0) {
        if (wave >= maxWaves) {
            gameState = 'win';
            try {
                if (score > highScore) { highScore = score; localStorage.setItem('platformInvadersHigh', highScore); }
            } catch(e) {}
        } else {
            wave++;
            playSound('wave');
            spawnWave(wave);
            player.hp = Math.min(player.hp + 1, player.maxHp);
            if (wave % 3 === 0 && !player.doubleShot) {
                player.doubleShot = true;
                playSound('powerup');
            }
        }
    }
    
    // Check enemies reached bottom
    for (let e of enemies) {
        if (e.alive && e.y + e.h > H - 40) {
            player.hp = 0;
            gameState = 'gameover';
            try {
                if (score > highScore) { highScore = score; localStorage.setItem('platformInvadersHigh', highScore); }
            } catch(e2) {}
        }
    }
    
    // Update particles
    for (let i = particles.length - 1; i >= 0; i--) {
        const p = particles[i];
        p.x += p.vx;
        p.y += p.vy;
        p.vy += 0.1;
        p.life--;
        if (p.life <= 0) particles.splice(i, 1);
    }
    
    // Update stars
    for (let s of stars) {
        s.x -= s.speed;
        if (s.x < 0) { s.x = W; s.y = Math.random() * H; }
    }
}

// Render
function drawPixelRect(x, y, w, h, color) {
    ctx.fillStyle = color;
    ctx.fillRect(Math.floor(x), Math.floor(y), w, h);
}

function drawPlayer() {
    if (player.invincible > 0 && Math.floor(frameCount / 4) % 2 === 0) return;
    
    const px = Math.floor(player.x), py = Math.floor(player.y);
    // Body
    ctx.fillStyle = '#4488ff';
    ctx.fillRect(px + 4, py + 12, 16, 16);
    // Head
    ctx.fillStyle = '#ffcc88';
    ctx.fillRect(px + 6, py, 12, 12);
    // Visor
    ctx.fillStyle = '#00ffff';
    ctx.fillRect(px + 7, py + 3, 10, 5);
    // Legs
    ctx.fillStyle = '#3366cc';
    const legOffset = player.onGround && player.vx !== 0 ? (Math.floor(frameCount / 8) % 2 === 0 ? 2 : -2) : 0;
    ctx.fillRect(px + 4, py + 28, 6, 4 + legOffset);
    ctx.fillRect(px + 14, py + 28, 6, 4 - legOffset);
    // Gun
    ctx.fillStyle = '#888';
    if (player.facing === 'up') {
        ctx.fillRect(px + 10, py - 4, 4, 8);
    } else if (player.facing === 'left') {
        ctx.fillRect(px - 6, py + 18, 10, 4);
    } else {
        ctx.fillRect(px + 20, py + 18, 10, 4);
    }
}

function drawEnemy(e) {
    const ex = Math.floor(e.x), ey = Math.floor(e.y);
    const color = e.type === 'tank' ? '#ff6644' : e.type === 'soldier' ? '#44ff88' : '#ffcc44';
    
    // Body
    ctx.fillStyle = color;
    if (e.type === 'tank') {
        ctx.fillRect(ex + 2, ey + 4, 24, 16);
        ctx.fillRect(ex, ey + 8, 28, 10);
        // Eyes
        ctx.fillStyle = '#fff';
        ctx.fillRect(ex + 5, ey + 7, 5, 5);
        ctx.fillRect(ex + 18, ey + 7, 5, 5);
        ctx.fillStyle = '#f00';
        ctx.fillRect(ex + 6, ey + 8, 3, 3);
        ctx.fillRect(ex + 19, ey + 8, 3, 3);
    } else if (e.type === 'soldier') {
        ctx.fillRect(ex + 4, ey + 2, 20, 20);
        ctx.fillRect(ex, ey + 6, 4, 12);
        ctx.fillRect(ex + 24, ey + 6, 4, 12);
        // Eyes
        ctx.fillStyle = '#fff';
        ctx.fillRect(ex + 7, ey + 6, 5, 5);
        ctx.fillRect(ex + 16, ey + 6, 5, 5);
    } else {
        ctx.fillRect(ex + 2, ey + 4, 24, 18);
        // Wings
        ctx.fillStyle = '#ffaa00';
        const wingFlap = Math.floor(frameCount / 6) % 2 === 0 ? 3 : -3;
        ctx.fillRect(ex - 2, ey + 8 + wingFlap, 6, 4);
        ctx.fillRect(ex + 24, ey + 8 + wingFlap, 6, 4);
        // Eyes
        ctx.fillStyle = '#fff';
        ctx.fillRect(ex + 8, ey + 7, 4, 4);
        ctx.fillRect(ex + 16, ey + 7, 4, 4);
    }
    
    // HP bar
    if (e.hp < e.maxHp) {
        ctx.fillStyle = '#333';
        ctx.fillRect(ex, ey - 5, e.w, 3);
        ctx.fillStyle = '#f44';
        ctx.fillRect(ex, ey - 5, e.w * (e.hp / e.maxHp), 3);
    }
}

    ctx.fillStyle = '#00ffff';
    for (let b of playerBullets) {
        ctx.fillRect(b.x, b.y, b.w, b.h);
    }
    
    // Enemy bullets
    ctx.fillStyle = '#ff4444';
    for (let b of enemyBullets) {
        ctx.fillRect(b.x, b.y, b.w, b.h);
    }
    
    // Particles
    for (let p of particles) {
        ctx.globalAlpha = p.life / p.maxLife;
        ctx.fillStyle = p.color;
        ctx.fillRect(p.x, p.y, 4, 4);
    }
    ctx.globalAlpha = 1;
    
    // HUD
    ctx.fillStyle = '#fff';
    ctx.font = '16px monospace';
    ctx.textAlign = 'left';
    ctx.fillText('SCORE: ' + score, 15, 25);
    ctx.fillText('WAVE: ' + wave + '/' + maxWaves, 15, 48);
    
    // HP bar
    ctx.fillStyle = '#333';
    ctx.fillRect(15, H - 45, 100, 16);
    ctx.fillStyle = player.hp > 2 ? '#44ff88' : '#ff4444';
    ctx.fillRect(15, H - 45, 100 * (player.hp / player.maxHp), 16);
    ctx.strokeStyle = '#fff';
    ctx.strokeRect(15, H - 45, 100, 16);
    ctx.fillStyle = '#fff';
    ctx.font = '12px monospace';
    ctx.fillText('HP', 55, H - 32);
    
    if (player.doubleShot) {
        ctx.fillStyle = '#ffcc44';
        ctx.fillText('DOUBLE SHOT', 130, H - 32);
    }
}

function drawPlatforms() {
    for (let i = 0; i < platforms.length; i++) {
        const p = platforms[i];
        if (i === 0) { // ground
            ctx.fillStyle = '#1a3a5a';
            ctx.fillRect(p.x, p.y, p.w, p.h);
            ctx.fillStyle = '#2a5a8a';
            for (let x = 0; x < W; x += 20) {
                ctx.fillRect(x, p.y, 18, 3);
            }
        } else {
            ctx.fillStyle = '#2a4a6a';
            ctx.fillRect(p.x, p.y, p.w, p.h);
            ctx.fillStyle = '#4a7aaa';
            ctx.fillRect(p.x, p.y, p.w, 3);
        }
    }
}

function gameLoop() {
    update();
    draw();
    requestAnimationFrame(gameLoop);
}

generateStars();
gameLoop();
