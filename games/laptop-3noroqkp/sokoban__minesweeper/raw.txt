function initAudio() {
    if (!audioCtx) {
        audioCtx = new (window.AudioContext || window.webkitAudioContext)();
    }
}

function playTone(freq, duration, type='square', volume=0.1) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = type;
    osc.frequency.value = freq;
    gain.gain.setValueAtTime(volume, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + duration);
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + duration);
}

function playMove() { playTone(220, 0.08, 'square', 0.05); }
function playPush() { playTone(180, 0.1, 'triangle', 0.08); }
function playWin() { playTone(440, 0.15, 'sine', 0.1); setTimeout(()=>playTone(660,0.15,'sine',0.1),150); setTimeout(()=>playTone(880,0.2,'sine',0.1),300); }
function playDead() { playTone(150, 0.3, 'sawtooth', 0.12); setTimeout(()=>playTone(100,0.4,'sawtooth',0.12),100); }

function parseLevel(idx) {
    const lvl = levels[idx];
    gridH = lvl.length;
    gridW = Math.max(...lvl.map(r => r.length));
    
    player = {x:0, y:0};
    boxes = [];
    targets = [];
    mines = [];
    walls = [];
    numbers = Array.from({length:gridH}, () => Array(gridW).fill(0));
    deadMines = [];
    moves = 0;

    for (let y = 0; y < gridH; y++) {
        for (let x = 0; x < gridW; x++) {
            const ch = lvl[y][x] || '#';
            if (ch === '#' || ch === undefined) {
                walls.push({x, y});
            }
            if (ch === '@') player = {x, y};
            if (ch === '$') boxes.push({x, y});
            if (ch === '*') targets.push({x, y});
            if (ch === 'M') mines.push({x, y});
        }
    }

    // Compute numbers: for each non-wall, non-mine cell, count adjacent mines
    for (let y = 0; y < gridH; y++) {
        for (let x = 0; x < gridW; x++) {
            const isWall = walls.some(w => w.x===x && w.y===y);
            if (isWall) continue;
            let count = 0;
            for (let dy = -1; dy <= 1; dy++) {
                for (let dx = -1; dx <= 1; dx++) {
                    if (dx === 0 && dy === 0) continue;
                    const nx = x+dx, ny = y+dy;
                    if (nx>=0 && nx<gridW && ny>=0 && ny<gridH) {
                        if (mines.some(m => m.x===nx && m.y===ny)) count++;
                    }
                }
            }
            numbers[y][x] = count;
        }
    }
}

function isWall(x, y) {
    return walls.some(w => w.x === x && w.y === y);
}

function isMine(x, y) {
    return mines.some(m => m.x === x && m.y === y);
}

function boxAt(x, y) {
    return boxes.find(b => b.x === x && b.y === y);
}

function canMove(x, y) {
    if (x < 0 || x >= gridW || y < 0 || y >= gridH) return false;
    return !isWall(x, y);
}

function tryMove(dx, dy) {
    initAudio();
    const nx = player.x + dx;
    const ny = player.y + dy;
    
    if (!canMove(nx, ny)) return;

    const box = boxAt(nx, ny);
    
    if (box) {
        const bx = nx + dx;
        const by = ny + dy;
        // Can push only if target cell is empty (not wall, not another box)
        if (!canMove(bx, by)) return;
        if (boxAt(bx, by)) return;
        
        // Move box
        box.x = bx;
        box.y = by;
        // Move player
        player.x = nx;
        player.y = ny;
        moves++;
        playPush();
    } else {
        // Check for mine
        if (isMine(nx, ny)) {
            deadMines.push({x: nx, y: ny});
            gameState = 'dead';
            playDead();
            return;
        }
        player.x = nx;
        player.y = ny;
        moves++;
        playMove();
    }

    checkWin();
}

function checkWin() {
    const allDone = boxes.every(b => targets.some(t => t.x === b.x && t.y === b.y));
    if (allDone) {
        gameState = 'won';
        score += Math.max(0, 100 - moves * 2);
        playWin();
        setTimeout(() => {
            if (currentLevel >= levels.length - 1) {
                gameState = 'gameover';
            } else {
                currentLevel++;
                parseLevel(currentLevel);
                gameState = 'playing';
            }
        }, 1500);
    }
}

    const doneCount = boxes.filter(b => targets.some(t=>t.x===b.x&&t.y===b.y)).length;
    ctx.fillText(`Boxes: ${doneCount}/${boxes.length}  Moves: ${moves}  Score: ${score}`, offsetX + 200, 30);
}

function drawMine(px, py, revealed) {
    const cx = px + cellSize/2, cy = py + cellSize/2;
    const r = cellSize * 0.3;
    ctx.fillStyle = revealed ? '#4a4a4a' : '#333';
    ctx.beginPath();
    ctx.arc(cx, cy, r, 0, Math.PI*2);
    ctx.fill();
    // Spikes
    ctx.strokeStyle = revealed ? '#666' : '#555';
    ctx.lineWidth = 2;
    for (let i = 0; i < 8; i++) {
        const angle = (i / 8) * Math.PI * 2;
        ctx.beginPath();
        ctx.moveTo(cx + Math.cos(angle)*r*0.5, cy + Math.sin(angle)*r*0.5);
        ctx.lineTo(cx + Math.cos(angle)*r, cy + Math.sin(angle)*r);
        ctx.stroke();
    }
}

function render() {
    ctx.clearRect(0, 0, W, H);
    
    if (gameState === 'start') {
        drawStartScreen();
    } else {
        ctx.fillStyle = '#1a1a2e';
        ctx.fillRect(0, 0, W, H);
        drawGrid();
        
        if (gameState === 'dead') drawDeadScreen();
        if (gameState === 'won') drawWonScreen();
        if (gameState === 'gameover') drawGameOverScreen();
    }

    requestAnimationFrame(render);
}

function startGame() {
    currentLevel = 0;
    score = 0;
    parseLevel(0);
    gameState = 'playing';
}

function restartLevel() {
    parseLevel(currentLevel);
    gameState = 'playing';
}

window.addEventListener('keydown', (e) => {
    if (gameState === 'start') {
        if (e.key === 'Enter' || e.key === ' ') {
            initAudio();
            startGame();
        }
        return;
    }

    if (gameState === 'gameover') {
        if (e.key === 'Enter' || e.key === ' ') {
            gameState = 'start';
        }
        return;
    }

    if (gameState === 'dead') {
        if (e.key.toLowerCase() === 'r') restartLevel();
        return;
    }

    if (gameState === 'won') return; // auto-advances

    if (gameState !== 'playing') return;

    let dx = 0, dy = 0;
    switch(e.key) {
        case 'ArrowUp': case 'w': case 'W': dy = -1; break;
        case 'ArrowDown': case 's': case 'S': dy = 1; break;
        case 'ArrowLeft': case 'a': case 'A': dx = -1; break;
        case 'ArrowRight': case 'd': case 'D': dx = 1; break;
        case 'r': case 'R': e.preventDefault(); restartLevel(); return;
        default: return;
    }
    e.preventDefault();
    tryMove(dx, dy);
});

// Touch/click support for mobile
let touchStartX = 0, touchStartY = 0;
canvas.addEventListener('touchstart', (e) => {
    initAudio();
    const t = e.touches[0];
    touchStartX = t.clientX;
    touchStartY = t.clientY;
});
canvas.addEventListener('touchend', (e) => {
    if (gameState === 'start' || gameState === 'gameover') { 
        if(gameState==='start') startGame(); else gameState='start';
        return; 
    }
    const t = e.changedTouches[0];
    const dx = t.clientX - touchStartX;
    const dy = t.clientY - touchStartY;
    if (Math.abs(dx) > 20 || Math.abs(dy) > 20) {
        if (Math.abs(dx) > Math.abs(dy)) {
            tryMove(dx > 0 ? 1 : -1, 0);
        } else {
            tryMove(0, dy > 0 ? 1 : -1);
        }
    }
});

// Start rendering
render();
