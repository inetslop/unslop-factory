<script>
const canvas = document.getElementById('game');
const ctx = canvas.getContext('2d');

let W, H;
function resize() {
  W = window.innerWidth;
  H = window.innerHeight;
  canvas.width = W;
  canvas.height = H;
}
window.addEventListener('resize', resize);
resize();

// Audio
let audioCtx = null;
function initAudio() {
  if (!audioCtx) {
    audioCtx = new (window.AudioContext || window.webkitAudioContext)();
  }
}
function playSound(freq, duration, type = 'square', volume = 0.15) {
  if (!audioCtx) return;
  const osc = audioCtx.createOscillator();
  const gain = audioCtx.createGain();
  osc.type = type;
  osc.frequency.value = freq;
  gain.gain.setValueAtTime(volume, audioCtx.currentTime);
  gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + duration);
  osc.connect(gain);
  gain.connect(audioCtx.destination);
  osc.start();
  osc.stop(audioCtx.currentTime + duration);
}

// Game state
const STATE = { MENU: 0, PLAYING: 1, POINT_SCORED: 2, GAME_OVER: 3 };
let gameState = STATE.MENU;

// Paddle settings
const PADDLE_W = 14;
const PADDLE_H_RATIO = 0.2;
const BALL_R_RATIO = 0.015;
const WIN_SCORE = 5;

let paddle1, paddle2, ball;
let score1 = 0, score2 = 0;
let strokesThisHole = 0;
let lastGolfTerm = '';
let showTimer = 0;
let particles = [];
let time = 0;

function initGame() {
  const pH = H * PADDLE_H_RATIO;
  const br = Math.max(W, H) * BALL_R_RATIO;
  paddle1 = { x: 30, y: H / 2 - pH / 2, w: PADDLE_W, h: pH, vy: 0 };
  paddle2 = { x: W - 30 - PADDLE_W, y: H / 2 - pH / 2, w: PADDLE_W, h: pH, vy: 0 };
  ball = { x: W / 2, y: H / 2, r: br, vx: 0, vy: 0, spin: 0 };
  score1 = 0; score2 = 0;
  strokesThisHole = 0;
  lastGolfTerm = '';
  showTimer = 0;
  particles = [];
}

function serveBall(direction) {
  const speed = Math.min(W, H) * 0.007;
  ball.vx = speed * direction;
  ball.vy = (Math.random() - 0.5) * speed * 0.6;
  ball.spin = 0;
  strokesThisHole = 0;
}

function getGolfTerm(strokes) {
  if (strokes === 0) return 'HOLE IN ONE!';
  if (strokes === 1) return 'ACE!';
  if (strokes <= 3) return 'BIRDIE!';
  if (strokes <= 5) return 'PAR!';
  return 'BOGEY';
}

function addParticles(x, y, color, count) {
  for (let i = 0; i < count; i++) {
    const angle = Math.random() * Math.PI * 2;
    const speed = Math.random() * 4 + 1;
    particles.push({
      x, y,
      vx: Math.cos(angle) * speed,
      vy: Math.sin(angle) * speed,
      life: 1,
      decay: 0.02 + Math.random() * 0.03,
      size: Math.random() * 4 + 2,
      color
    });
  }
}

function scorePoint(scorer) {
  if (scorer === 1) score1++;
  else score2++;
  lastGolfTerm = getGolfTerm(strokesThisHole);
  showTimer = 90;
  gameState = STATE.POINT_SCORED;

  const isWin = (scorer === 1 && score1 >= WIN_SCORE) || (scorer === 2 && score2 >= WIN_SCORE);
  if (isWin) {
    setTimeout(() => { gameState = STATE.GAME_OVER; }, 2000);
  } else {
    setTimeout(() => {
      gameState = STATE.PLAYING;
      serveBall(scorer === 1 ? 1 : -1);
    }, 1500);
  }

  playSound(scorer === 1 ? 440 : 330, 0.3, 'sine', 0.2);
  addParticles(ball.x, ball.y, scorer === 1 ? '#ffdd44' : '#66aaff', 20);
}

// Input
const keys = {};
window.addEventListener('keydown', (e) => {
  keys[e.code] = true;
  if (gameState === STATE.MENU || gameState === STATE.GAME_OVER) {
    initAudio();
    initGame();
    gameState = STATE.PLAYING;
    serveBall(Math.random() < 0.5 ? 1 : -1);
  }
});
window.addEventListener('keyup', (e) => { keys[e.code] = false; });

// Update
function update(dt) {
  if (gameState !== STATE.PLAYING && gameState !== STATE.POINT_SCORED) return;

  const paddleSpeed = H * 0.008;

  // Paddle 1: W/S
  if (keys['KeyW']) paddle1.y -= paddleSpeed;
  if (keys['KeyS']) paddle1.y += paddleSpeed;
  // Paddle 2: Arrow Up/Down
  if (keys['ArrowUp']) paddle2.y -= paddleSpeed;
  if (keys['ArrowDown']) paddle2.y += paddleSpeed;

  // Clamp paddles
  paddle1.y = Math.max(0, Math.min(H - paddle1.h, paddle1.y));
  paddle2.y = Math.max(0, Math.min(H - paddle2.h, paddle2.y));

  if (gameState !== STATE.PLAYING) return;

  // Ball spin effect (curve)
  ball.vy += ball.spin * 0.3;
  ball.spin *= 0.98;

  // Move ball
  ball.x += ball.vx;
  ball.y += ball.vy;

  // Top/bottom bounce
  if (ball.y - ball.r < 0) { ball.y = ball.r; ball.vy = Math.abs(ball.vy); playSound(200, 0.1, 'triangle', 0.1); }
  if (ball.y + ball.r > H) { ball.y = H - ball.r; ball.vy = -Math.abs(ball.vy); playSound(200, 0.1, 'triangle', 0.1); }

  // Paddle collision
  const p1Right = paddle1.x + paddle1.w;
  if (ball.vx < 0 && ball.x - ball.r <= p1Right && ball.x - ball.r >= paddle1.x - 5 &&
      ball.y >= paddle1.y && ball.y <= paddle1.y + paddle1.h) {
    const hitPos = (ball.y - paddle1.y) / paddle1.h;
    ball.vx = Math.abs(ball.vx) * 1.02;
    ball.vy = (hitPos - 0.5) * Math.min(W, H) * 0.012;
    ball.spin = (hitPos - 0.5) * 3;
    strokesThisHole++;
    ball.x = p1Right + ball.r;
    playSound(600, 0.1, 'square', 0.12);
    addParticles(ball.x, ball.y, '#ffffff', 5);
  }

  const p2Left = paddle2.x;
  if (ball.vx > 0 && ball.x + ball.r >= p2Left && ball.x + ball.r <= p2Left + paddle2.w + 5 &&
      ball.y >= paddle2.y && ball.y <= paddle2.y + paddle2.h) {
    const hitPos = (ball.y - paddle2.y) / paddle2.h;
    ball.vx = -Math.abs(ball.vx) * 1.02;
    ball.vy = (hitPos - 0.5) * Math.min(W, H) * 0.012;
    ball.spin = -(hitPos - 0.5) * 3;
    strokesThisHole++;
    ball.x = p2Left - ball.r;
    playSound(600, 0.1, 'square', 0.12);
    addParticles(ball.x, ball.y, '#ffffff', 5);
  }

  // Score detection
  if (ball.x + ball.r < -20) scorePoint(2);
  if (ball.x - ball.r > W + 20) scorePoint(1);

  // Update particles
  for (let i = particles.length - 1; i >= 0; i--) {
    const p = particles[i];
    p.x += p.vx;
    p.y += p.vy;
    p.life -= p.decay;
    if (p.life <= 0) particles.splice(i, 1);
  }

  if (showTimer > 0) showTimer--;
}

// Draw
function drawBackground() {
  // Golf course green gradient
  const grad = ctx.createLinearGradient(0, 0, 0, H);
  grad.addColorStop(0, '#2d6b1e');
  grad.addColorStop(0.5, '#3a8c25');
  grad.addColorStop(1, '#2d6b1e');
  ctx.fillStyle = grad;
  ctx.fillRect(0, 0, W, H);

  // Mowing stripes
  const stripeW = W / 12;
  for (let i = 0; i < 12; i += 2) {
    ctx.fillStyle = 'rgba(255,255,255,0.03)';
    ctx.fillRect(i * stripeW, 0, stripeW, H);
  }

  // Center line (sand bunker)
  const dashLen = 15;
  ctx.strokeStyle = '#c4a35a';
  ctx.lineWidth = 3;
  ctx.setLineDash([dashLen, dashLen * 0.7]);
  ctx.beginPath();
  ctx.moveTo(W / 2, 0);
  ctx.lineTo(W / 2, H);
  ctx.stroke();
  ctx.setLineDash([]);

  // Bunker dots
  ctx.fillStyle = 'rgba(196, 163, 90, 0.4)';
  for (let y = 10; y < H; y += 30) {
    ctx.beginPath();
    ctx.arc(W / 2 + (Math.sin(y * 0.1) * 5), y, 2, 0, Math.PI * 2);
    ctx.fill();
  }
}

function drawPaddle(paddle, isLeft) {
  // Wooden club look
  const grad = ctx.createLinearGradient(paddle.x, 0, paddle.x + paddle.w, 0);
  grad.addColorStop(0, '#8B5E3C');
  grad.addColorStop(0.5, '#D2A679');
  grad.addColorStop(1, '#8B5E3C');
  ctx.fillStyle = grad;

  // Rounded paddle
  const r = 4;
  ctx.beginPath();
  ctx.roundRect(paddle.x, paddle.y, paddle.w, paddle.h, r);
  ctx.fill();
  ctx.strokeStyle = '#5c3a1e';
  ctx.lineWidth = 2;
  ctx.stroke();

  // Grip lines
  ctx.strokeStyle = 'rgba(92, 58, 30, 0.4)';
  ctx.lineWidth = 1;
  for (let y = paddle.y + 10; y < paddle.y + paddle.h - 10; y += 12) {
    ctx.beginPath();
    ctx.moveTo(paddle.x + 3, y);
    ctx.lineTo(paddle.x + paddle.w - 3, y);
    ctx.stroke();
  }
}

function drawBall() {
  // Shadow
  ctx.fillStyle = 'rgba(0,0,0,0.2)';
  ctx.beginPath();
  ctx.ellipse(ball.x + 3, ball.y + 4, ball.r, ball.r * 0.6, 0, 0, Math.PI * 2);
  ctx.fill();

  // White ball
  const grad = ctx.createRadialGradient(ball.x - ball.r * 0.3, ball.y - ball.r * 0.3, 0, ball.x, ball.y, ball.r);
  grad.addColorStop(0, '#ffffff');
  grad.addColorStop(1, '#cccccc');
  ctx.fillStyle = grad;
  ctx.beginPath();
  ctx.arc(ball.x, ball.y, ball.r, 0, Math.PI * 2);
  ctx.fill();

  // Dimples
  ctx.fillStyle = 'rgba(180,180,180,0.5)';
  const dr = ball.r * 0.15;
  for (let a = 0; a < 6; a++) {
    const angle = (a / 6) * Math.PI * 2 + time * 0.02;
    const dx = Math.cos(angle) * ball.r * 0.4;
    const dy = Math.sin(angle) * ball.r * 0.4;
    ctx.beginPath();
    ctx.arc(ball.x + dx, ball.y + dy, dr, 0, Math.PI * 2);
    ctx.fill();
  }
}

function drawParticles() {
  for (const p of particles) {
    ctx.globalAlpha = p.life;
    ctx.fillStyle = p.color;
    ctx.beginPath();
    ctx.arc(p.x, p.y, p.size * p.life, 0, Math.PI * 2);
    ctx.fill();
  }
  ctx.globalAlpha = 1;
}

function drawScore() {
  const holeNum = Math.min(score1 + score2 + 1, WIN_SCORE);
  ctx.font = `bold ${Math.min(W, H) * 0.04}px 'Segoe UI', sans-serif`;
  ctx.textAlign = 'center';
  ctx.fillStyle = 'rgba(255,255,255,0.9)';

  // Hole indicator
  ctx.font = `bold ${Math.min(W, H) * 0.025}px 'Segoe UI', sans-serif`;
  ctx.fillText(`HOLE ${holeNum} / ${WIN_SCORE}`, W / 2, Math.min(W, H) * 0.04);

  // Scores
  ctx.font = `bold ${Math.min(W, H) * 0.06}px 'Segoe UI', sans-serif`;
  ctx.fillStyle = '#ffdd44';
  ctx.fillText(score1, W / 2 - 50, Math.min(W, H) * 0.09);
  ctx.fillStyle = '#66aaff';
  ctx.fillText(score2, W / 2 + 50, Math.min(W, H) * 0.09);

  // Player labels
  ctx.font = `${Math.min(W, H) * 0.018}px 'Segoe UI', sans-serif`;
  ctx.fillStyle = 'rgba(255,255,255,0.6)';
  ctx.fillText('PLAYER 1 (W/S)', W / 2 - 50, Math.min(W, H) * 0.12);
  ctx.fillText('PLAYER 2 (↑/↓)', W / 2 + 50, Math.min(W, H) * 0.12);

  // Golf term popup
  if (showTimer > 0 && lastGolfTerm) {
    const alpha = Math.min(1, showTimer / 30);
    ctx.globalAlpha = alpha;
    ctx.font = `bold ${Math.min(W, H) * 0.05}px 'Segoe UI', sans-serif`;
    ctx.fillStyle = '#ffdd44';
    ctx.textAlign = 'center';
    const scale = showTimer > 70 ? (90 - showTimer) / 20 : 1;
    ctx.save();
    ctx.translate(W / 2, H / 2);
    ctx.scale(scale, scale);
    ctx.fillText(lastGolfTerm, 0, 0);
    ctx.font = `${Math.min(W, H) * 0.02}px 'Segoe UI', sans-serif`;
    ctx.fillStyle = '#fff';
    ctx.fillText(`${strokesThisHole} stroke${strokesThisHole !== 1 ? 's' : ''}`, 0, Math.min(W,H)*0.035);
    ctx.restore();
    ctx.globalAlpha = 1;
  }
}

function drawMenu() {
  drawBackground();
  ctx.textAlign = 'center';

  // Title
  const titleSize = Math.min(W, H) * 0.08;
  ctx.font = `bold ${titleSize}px 'Segoe UI', sans-serif`;
  ctx.fillStyle = '#ffdd44';
  ctx.fillText('⛳ GOLF PONG', W / 2, H * 0.3);

  // Subtitle
  ctx.font = `${Math.min(W, H) * 0.025}px 'Segoe UI', sans-serif`;
  ctx.fillStyle = '#ffffff';
  ctx.fillText('A Pong game with a golf twist!', W / 2, H * 0.4);

  // Controls
  ctx.font = `${Math.min(W, H) * 0.02}px 'Segoe UI', sans-serif`;
  ctx.fillStyle = 'rgba(255,255,255,0.8)';
  ctx.fillText('Player 1: W / S keys', W / 2, H * 0.52);
  ctx.fillText('Player 2: ↑ / ↓ Arrow Keys', W / 2, H * 0.57);
  ctx.fillText(`First to ${WIN_SCORE} holes wins!`, W / 2, H * 0.64);

  // Spin info
  ctx.font = `${Math.min(W, H) * 0.016}px 'Segoe UI', sans-serif`;
  ctx.fillStyle = 'rgba(255,255,255,0.5)';
  ctx.fillText('Hit off-center to add spin and curve the ball!', W / 2, H * 0.72);

  // Start prompt
  const pulse = Math.sin(time * 0.05) * 0.3 + 0.7;
  ctx.globalAlpha = pulse;
  ctx.font = `bold ${Math.min(W, H) * 0.03}px 'Segoe UI', sans-serif`;
  ctx.fillStyle = '#ffffff';
  ctx.fillText('Press SPACE to Start', W / 2, H * 0.85);
  ctx.globalAlpha = 1;

  // Decorative golf ball
  ctx.beginPath();
  ctx.arc(W / 2, H * 0.2, Math.min(W,H) * 0.03, 0, Math.PI * 2);
  ctx.fillStyle = '#fff';
  ctx.fill();
}

function drawGameOver() {
  drawBackground();
  drawScore();

  ctx.textAlign = 'center';
  const winner = score1 > score2 ? 'Player 1' : 'Player 2';
  const color = score1 > score2 ? '#ffdd44' : '#66aaff';

  ctx.font = `bold ${Math.min(W, H) * 0.07}px 'Segoe UI', sans-serif`;
  ctx.fillStyle = color;
  ctx.fillText(`${winner} Wins!`, W / 2, H * 0.35);

  ctx.font = `${Math.min(W, H) * 0.03}px 'Segoe UI', sans-serif`;
  ctx.fillStyle = '#fff';
  ctx.fillText(`Final Score: ${score1} - ${score2}`, W / 2, H * 0.45);

  const pulse = Math.sin(time * 0.05) * 0.3 + 0.7;
  ctx.globalAlpha = pulse;
  ctx.font = `bold ${Math.min(W, H) * 0.025}px 'Segoe UI', sans-serif`;
  ctx.fillStyle = '#ffffff';
  ctx.fillText('Press SPACE to Play Again', W / 2, H * 0.6);
  ctx.globalAlpha = 1;
}

// Main loop
let lastTime = 0;
function gameLoop(timestamp) {
  time++;
  const dt = timestamp - lastTime;
  lastTime = timestamp;

  if (gameState === STATE.MENU) {
    drawMenu();
  } else if (gameState === STATE.GAME_OVER) {
    update(dt);
    drawGameOver();
    drawParticles();
  } else {
    update(dt);
    drawBackground();
    drawPaddle(paddle1, true);
    drawPaddle(paddle2, false);
    if (gameState === STATE.PLAYING) drawBall();
    drawParticles();
    drawScore();
  }

  requestAnimationFrame(gameLoop);
}

initGame();
requestAnimationFrame(gameLoop);
</script>
