// Audio
let audioCtx = null;
function initAudio() {
    if (!audioCtx) {
        audioCtx = new (window.AudioContext || window.webkitAudioContext)();
    }
}
function playBeep(freq, dur, type, vol) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = type || 'square';
    osc.frequency.value = freq;
    gain.gain.value = vol || 0.1;
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + dur);
}
function sfxJump() { playBeep(440, 0.1, 'square', 0.08); }
function sfxDie() { playBeep(120, 0.3, 'sawtooth', 0.15); }
function sfxWin() { playBeep(660, 0.15, 'square', 0.1); setTimeout(()=>playBeep(880,0.2,'square',0.1),150); }

// Game constants
const ROWS = 8;
const COLS = 9;
let cellSize, gridW, gridH, offsetX, offsetY;

function computeLayout() {
    const halfW = W / 2;
    const maxCell = Math.min(halfW - 20, H * 0.7);
    cellSize = Math.floor(maxCell / COLS);
    if (cellSize > 60) cellSize = 60;
    gridW = cellSize * COLS;
    gridH = cellSize * ROWS;
    offsetX = (halfW - gridW) / 2;
    offsetY = Math.max(10, (H - gridH - 40) / 2);
}

// Game state
let gameState = 'start'; // start, playing, gameover
let players = [null, null];
let frameCount = 0;

class Player {
    constructor(id, side) {
        this.id = id;
        this.side = side; // 0 left, 1 right
        this.col = Math.floor(COLS / 2);
        this.row = ROWS - 1;
        this.lives = 3;
        this.score = 0;
        this.wins = 0;
        this.alive = true;
        this.yRiding = null; // which log
        this.crossedRoads = new Set();
        this.homeTimer = 0;
    }
    reset() {
        this.col = Math.floor(COLS / 2);
        this.row = ROWS - 1;
        this.lives = 3;
        this.score = 0;
        this.wins = 0;
        this.alive = true;
        this.yRiding = null;
        this.crossedRoads = new Set();
        this.homeTimer = 0;
    }
}

// Car/traffic system
let cars = []; // {side, row, col(float), dir, speed, type}
let logs = []; // {side, row, col(float), dir, speed, width}

function spawnCars(side) {
    for (let r = 0; r < ROWS; r++) {
        if (isRiverRow(r)) continue;
        if (r === 0 || r === ROWS - 1) continue; // home and start are safe
        const count = 2 + Math.floor(Math.random() * 2);
        for (let i = 0; i < count; i++) {
            const col = (Math.random() * COLS * 1.5) | 0;
            const dir = Math.random() > 0.5 ? 1 : -1;
            const speed = 0.8 + Math.random() * 0.6;
            cars.push({ side, row: r, col: col * cellSize / 2, dir, speed, type: Math.floor(Math.random() * 3) });
        }
    }
}

function spawnLogs(side) {
    for (let r = 0; r < ROWS; r++) {
        if (!isRiverRow(r)) continue;
        const count = 2 + Math.floor(Math.random() * 2);
        for (let i = 0; i < count; i++) {
            const col = (Math.random() * COLS) | 0;
            const dir = Math.random() > 0.5 ? 1 : -1;
            const speed = 0.5 + Math.random() * 0.4;
            const width = 2 + Math.floor(Math.random() * 2); // in cells
            logs.push({ side, row: r, col: col * cellSize / 2, dir, speed, width });
        }
    }
}

function isRiverRow(r) { return r >= 3 && r <= 4; }
function isHomeRow(r) { return r === 0; }
function isStartRow(r) { return r === ROWS - 1; }

function resetLevel() {
    cars = [];
    logs = [];
    spawnCars(0); spawnCars(1);
    spawnLogs(0); spawnLogs(1);
}

function startGame() {
    players[0] = new Player(0, 0);
    players[1] = new Player(1, 1);
    resetLevel();
    gameState = 'playing';
}

// Input
const keys = {};
window.addEventListener('keydown', (e) => {
    keys[e.key] = true;
    initAudio();
    if (gameState === 'start' || gameState === 'gameover') {
        if (e.key === 'Enter' || e.key === ' ') startGame();
    }
    if (gameState === 'playing') handleInput(e.key);
});
window.addEventListener('keyup', (e) => { keys[e.key] = false; });

function handleInput(key) {
    // P1: WASD
    if (players[0] && players[0].alive) {
        if (key === 'w' || key === 'W') tryMove(0, 0, -1);
        if (key === 's' || key === 'S') tryMove(0, 0, 1);
        if (key === 'a' || key === 'A') tryMove(0, -1, 0);
        if (key === 'd' || key === 'D') tryMove(0, 1, 0);
    }
    // P2: Arrow keys
    if (players[1] && players[1].alive) {
        if (key === 'ArrowUp') { e.preventDefault(); tryMove(1, 0, -1); }
        if (key === 'ArrowDown') { e.preventDefault(); tryMove(1, 0, 1); }
        if (key === 'ArrowLeft') { e.preventDefault(); tryMove(1, -1, 0); }
        if (key === 'ArrowRight') { e.preventDefault(); tryMove(1, 1, 0); }
    }
}

function tryMove(pid, dx, dy) {
    const p = players[pid];
    if (!p || !p.alive) return;
    let nc = p.col + dx;
    let nr = p.row + dy;
    if (nc < 0 || nc >= COLS || nr < 0 || nr >= ROWS) return;
    p.col = nc;
    p.row = nr;
    p.yRiding = null;
    sfxJump();

    // Check home
    if (nr === 0) {
        p.homeTimer++;
        if (p.homeTimer >= 30) {
            p.wins++;
            p.score += 200;
            sfxWin();
            if (p.wins >= 5) {
                gameState = 'gameover';
            } else {
                p.col = Math.floor(COLS / 2);
                p.row = ROWS - 1;
                p.homeTimer = 0;
                p.crossedRoads.clear();
            }
        }
    }

    // Score for crossing road rows
    if (nr >= 1 && nr <= 2 || nr >= 5 && nr <= 6) {
        const rowKey = pid * 10 + nr;
        if (!p.crossedRoads.has(rowKey)) {
            p.crossedRoads.add(rowKey);
            p.score += 50;
        }
    }
}

function update() {
    if (gameState !== 'playing') return;
    frameCount++;

    // Update cars
    for (let c of cars) {
        c.col += c.speed * c.dir * cellSize * 0.5;
        if (c.dir > 0 && c.col > COLS * cellSize + cellSize * 2) c.col = -cellSize * 2;
        if (c.dir < 0 && c.col < -cellSize * 2) c.col = COLS * cellSize + cellSize * 2;
    }

    // Update logs
    for (let l of logs) {
        l.col += l.speed * l.dir * cellSize * 0.3;
        if (l.dir > 0 && l.col > COLS * cellSize + cellSize * 4) l.col = -cellSize * l.width - cellSize * 2;
        if (l.dir < 0 && l.col < -cellSize * l.width - cellSize * 2) l.col = COLS * cellSize + cellSize * 2;
    }

    // Check players
    for (let pid = 0; pid < 2; pid++) {
        const p = players[pid];
        if (!p || !p.alive) continue;

        if (isRiverRow(p.row)) {
            // Check if on a log
            let onLog = false;
            for (let l of logs) {
                if (l.side !== pid || l.row !== p.row) continue;
                const logLeft = Math.floor(l.col / cellSize);
                const logRight = logLeft + l.width - 1;
                if (p.col >= logLeft && p.col <= logRight) {
                    onLog = true;
                    // Move player with log
                    p.yRiding = l;
                    break;
                }
            }
            if (!onLog) {
                killPlayer(pid);
                continue;
            }
            // If riding a log, shift position
            for (let l of logs) {
                if (l.side === pid && l.row === p.row) {
                    const logLeft = Math.floor(l.col / cellSize);
                    const prevLogLeft = Math.floor((l.col - l.speed * l.dir * cellSize * 0.3) / cellSize);
                    // Simple: just check if player is still on log
                }
            }
        } else if (!isHomeRow(p.row) && !isStartRow(p.row)) {
            // Check car collision
            for (let c of cars) {
                if (c.side !== pid || c.row !== p.row) continue;
                const carCol = Math.floor(c.col / cellSize);
                const carRight = carCol + 1;
                if (p.col >= carCol && p.col <= carRight) {
                    killPlayer(pid);
                    break;
                }
            }
        }
    }
}

function killPlayer(pid) {
    const p = players[pid];
    p.alive = false;
    p.lives--;
    sfxDie();
    if (p.lives <= 0) {
        // Player loses
    } else {
        setTimeout(() => {
            if (gameState !== 'playing') return;
            p.alive = true;
            p.col = Math.floor(COLS / 2);
            p.row = ROWS - 1;
            p.yRiding = null;
            p.homeTimer = 0;
        }, 1500);
    }
}

// Rendering
function drawFrog(x, y, s, color) {
    ctx.fillStyle = color;
    // Body
    ctx.beginPath();
    ctx.ellipse(x + s/2, y + s/2 + s*0.05, s*0.4, s*0.35, 0, 0, Math.PI*2);
    ctx.fill();
    // Head
    ctx.beginPath();
    ctx.ellipse(x + s/2, y + s*0.2, s*0.3, s*0.25, 0, 0, Math.PI*2);
    ctx.fill();
    // Eyes
    ctx.fillStyle = '#fff';
    ctx.beginPath();
    ctx.arc(x + s*0.3, y + s*0.15, s*0.1, 0, Math.PI*2);
    ctx.fill();
    ctx.beginPath();
    ctx.arc(x + s*0.7, y + s*0.15, s*0.1, 0, Math.PI*2);
    ctx.fill();
    ctx.fillStyle = '#000';
    ctx.beginPath();
    ctx.arc(x + s*0.3, y + s*0.15, s*0.05, 0, Math.PI*2);
    ctx.fill();
    ctx.beginPath();
    ctx.arc(x + s*0.7, y + s*0.15, s*0.05, 0, Math.PI*2);
    ctx.fill();
    // Legs
    ctx.fillStyle = color;
    ctx.fillRect(x + s*0.1, y + s*0.7, s*0.15, s*0.2);
    ctx.fillRect(x + s*0.75, y + s*0.7, s*0.15, s*0.2);
}

function drawCar(x, y, s, type) {
    const colors = ['#e74c3c', '#3498db', '#f39c12'];
    ctx.fillStyle = colors[type % 3];
    // Car body
    ctx.fillRect(x, y + s*0.25, s, s*0.5);
    // Roof
    ctx.fillStyle = 'rgba(255,255,255,0.3)';
    ctx.fillRect(x + s*0.2, y + s*0.15, s*0.6, s*0.2);
    // Wheels
    ctx.fillStyle = '#222';
    ctx.beginPath();
    ctx.arc(x + s*0.2, y + s*0.8, s*0.15, 0, Math.PI*2);
    ctx.fill();
    ctx.beginPath();
    ctx.arc(x + s*0.8, y + s*0.8, s*0.15, 0, Math.PI*2);
    ctx.fill();
}

function draw() {
    computeLayout();
    ctx.fillStyle = '#1a1a2e';
    ctx.fillRect(0, 0, W, H);

    if (gameState === 'start') {
        drawStartScreen();
        return;
    }

    // Draw both player areas
    for (let side = 0; side < 2; side++) {
        const ox = side * (W / 2) + offsetX;
        const oy = offsetY;

        // Draw grid
        for (let r = 0; r < ROWS; r++) {
            const ry = oy + r * cellSize;
            if (isHomeRow(r)) {
                ctx.fillStyle = '#2d5a27';
                ctx.fillRect(ox, ry, gridW, cellSize);
                // Home spots
                for (let c = 0; c < COLS; c += 2) {
                    ctx.fillStyle = '#1a3d18';
                    ctx.beginPath();
                    ctx.arc(ox + c * cellSize + cellSize/2, ry + cellSize/2, cellSize*0.3, 0, Math.PI*2);
                    ctx.fill();
                }
            } else if (isRiverRow(r)) {
                ctx.fillStyle = '#1565c0';
                ctx.fillRect(ox, ry, gridW, cellSize);
                // Water animation
                ctx.strokeStyle = 'rgba(255,255,255,0.2)';
                for (let i = 0; i < COLS * 2; i++) {
                    const wx = ox + (i * cellSize * 0.5 + frameCount * 0.3) % gridW;
                    ctx.beginPath();
                    ctx.moveTo(wx, ry + cellSize * 0.3);
                    ctx.lineTo(wx + cellSize * 0.3, ry + cellSize * 0.5);
                    ctx.stroke();
                }
            } else if (isStartRow(r)) {
                ctx.fillStyle = '#2d5a27';
                ctx.fillRect(ox, ry, gridW, cellSize);
            } else {
                ctx.fillStyle = '#333';
                ctx.fillRect(ox, ry, gridW, cellSize);
                // Road markings
                ctx.strokeStyle = '#ffd700';
                ctx.setLineDash([cellSize * 0.2, cellSize * 0.2]);
                ctx.beginPath();
                ctx.moveTo(ox, ry + cellSize/2);
                ctx.lineTo(ox + gridW, ry + cellSize/2);
                ctx.stroke();
                ctx.setLineDash([]);
            }
            // Grid lines
            ctx.strokeStyle = 'rgba(255,255,255,0.1)';
            for (let c = 0; c <= COLS; c++) {
                ctx.beginPath();
                ctx.moveTo(ox + c * cellSize, ry);
                ctx.lineTo(ox + c * cellSize, ry + cellSize);
                ctx.stroke();
            }
        }

        // Draw logs
        for (let l of logs) {
            if (l.side !== side) continue;
            const lx = ox + l.col;
            const ly = oy + l.row * cellSize + cellSize * 0.2;
            const lw = l.width * cellSize;
            const lh = cellSize * 0.6;
            ctx.fillStyle = '#8b4513';
            ctx.fillRect(lx, ly, lw, lh);
            ctx.fillStyle = '#a0522d';
            ctx.fillRect(lx + 2, ly + 2, lw - 4, lh - 4);
            // Log rings
            ctx.strokeStyle = '#6b3410';
            ctx.beginPath();
            ctx.arc(lx + cellSize*0.5, ly + lh/2, lh*0.3, 0, Math.PI*2);
            ctx.stroke();
        }

        // Draw cars
        for (let c of cars) {
            if (c.side !== side) continue;
            const cx = ox + c.col;
            const cy = oy + c.row * cellSize;
            drawCar(cx, cy, cellSize, c.type);
        }

        // Draw player
        const p = players[side];
        if (p && p.alive) {
            const px = ox + p.col * cellSize;
            const py = oy + p.row * cellSize;
            drawFrog(px, py, cellSize, side === 0 ? '#4caf50' : '#8bc34a');
        }

        // Player HUD
        ctx.fillStyle = '#fff';
        ctx.font = `bold ${Math.min(cellSize*0.6, 20)}px monospace`;
        const hudX = side === 0 ? W * 0.25 : W * 0.75;
        ctx.textAlign = 'center';
        ctx.fillText(`P${side+1}`, hudX, oy - cellSize * 0.3);

        // Score and lives
        ctx.font = `${Math.min(cellSize*0.45, 16)}px monospace`;
        const scoreStr = `Score: ${p.score}  |  Wins: ${p.wins}/5  |  Lives: ${'❤'.repeat(Math.max(0,p.lives))}`;
        ctx.fillText(scoreStr, hudX, oy + gridH + cellSize * 0.5);

        // Controls hint
        ctx.fillStyle = 'rgba(255,255,255,0.4)';
        ctx.font = `${Math.min(cellSize*0.35, 12)}px monospace`;
        const ctrlText = side === 0 ? 'WASD' : 'Arrow Keys';
        ctx.fillText(ctrlText, hudX, oy + gridH + cellSize * 0.9);

        // Divider
        if (side === 0) {
            ctx.strokeStyle = '#fff';
            ctx.lineWidth = 2;
            ctx.setLineDash([5, 5]);
            ctx.beginPath();
            ctx.moveTo(W/2, oy - cellSize);
            ctx.lineTo(W/2, oy + gridH + cellSize);
            ctx.stroke();
            ctx.setLineDash([]);
        }
    }

    if (gameState === 'gameover') {
        drawGameOver();
    }
}

function drawStartScreen() {

    ctx.fillStyle = '#1a1a2e';
    ctx.fillRect(0, 0, W, H);

    ctx.textAlign = 'center';
    ctx.fillStyle = '#4caf50';
    ctx.font = `bold ${Math.min(W*0.08, 60)}px monospace`;
    ctx.fillText('FROGGER', W/2, H * 0.2);
    ctx.fillStyle = '#fff';
    ctx.font = `bold ${Math.min(W*0.04, 30)}px monospace`;
    ctx.fillText('TWO PLAYER', W/2, H * 0.28);

    // Frogs
    drawFrog(W/2 - cellSize*1.5, H*0.35, cellSize, '#4caf50');
    drawFrog(W/2 + cellSize*0.5, H*0.35, cellSize, '#8bc34a');

    ctx.fillStyle = '#ccc';
    ctx.font = `${Math.min(W*0.025, 18)}px monospace`;
    const lines = [
        'PLAYER 1 (Left):  W A S D',
        'PLAYER 2 (Right): Arrow Keys',
        '',
        'Cross the road and river to reach HOME!',',
        'Avoid cars and water. Ride logs across the river.',
        'First to win 5 times wins the match!',
        '',
        'Press ENTER or SPACE to Start'
    ];
    lines.forEach((line, i) => {
        ctx.fillText(line, W/2, H * 0.45 + i * cellSize * 0.6);
    });

    // Animated frog bounce
    const bounce = Math.sin(frameCount * 0.1) * 5;
    drawFrog(W/2 - cellSize/2, H * 0.85 + bounce, cellSize, '#4caf50');
}

function drawGameOver() {
    ctx.fillStyle = 'rgba(0,0,0,0.7)';
    ctx.fillRect(0, 0, W, H);

    let winner = '';
    if (players[0].wins >= 5) winner = 'PLAYER 1 WINS!';
    else if (players[1].wins >= 5) winner = 'PLAYER 2 WINS!';
    else {
        // Determine who has more wins, or both lost lives
        if (players[0].lives <= 0 && players[1].lives <= 0) winner = "IT'S A TIE!";
        else if (players[0].lives <= 0) winner = 'PLAYER 2 WINS!';
        else if (players[1].lives <= 0) winner = 'PLAYER 1 WINS!';
        else winner = `P${players[0].wins >= players[1].wins ? 1 : 2} LEADS ${Math.max(players[0].wins, players[1].wins)}-${Math.min(players[0].wins, players[1].wins)}`;
    }

    ctx.textAlign = 'center';
    ctx.fillStyle = '#ffd700';
    ctx.font = `bold ${Math.min(W*0.05, 48)}px monospace`;
    ctx.fillText('GAME OVER', W/2, H * 0.3);
    ctx.fillStyle = '#fff';
    ctx.font = `bold ${Math.min(W*0.035, 32)}px monospace`;
    ctx.fillText(winner, W/2, H * 0.42);

    ctx.fillStyle = '#ccc';
    ctx.font = `${Math.min(W*0.025, 20)}px monospace`;
    ctx.fillText(`P1: ${players[0].wins} wins | P2: ${players[1].wins} wins`, W/2, H * 0.52);

    ctx.fillStyle = '#4caf50';
    ctx.font = `${Math.min(W*0.025, 20)}px monospace`;
    const pulse = Math.sin(frameCount * 0.1) * 0.3 + 0.7;
    ctx.globalAlpha = pulse;
    ctx.fillText('Press ENTER to Play Again', W/2, H * 0.65);
    ctx.globalAlpha = 1;

    // Check for restart
    if (keys['Enter'] || keys[' ']) {
        startGame();
    }
}

// Main loop
function gameLoop() {
    frameCount++;
    update();
    draw();
    requestAnimationFrame(gameLoop);
}

computeLayout();
gameLoop();
