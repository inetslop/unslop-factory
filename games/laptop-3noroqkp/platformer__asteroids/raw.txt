function generateAsteroidVertices(w, h) {
    const verts = [];
    const segments = Math.floor(Math.random() * 4) + 6;
    for (let i = 0; i < segments; i++) {
        const angle = (i / segments) * Math.PI * 2;
        const rx = w / 2 + (Math.random() - 0.5) * w * 0.3;
        const ry = h / 2 + (Math.random() - 0.5) * h * 0.4;
        verts.push({ x: Math.cos(angle) * rx, y: Math.sin(angle) * ry });
    }
    return verts;
}

let time = 0;

function update(dt) {
    if (gameState !== 'PLAYING') return;
    
    time += dt;
    
    // Player input
    const moveSpeed = 5;
    const gravity = 0.6;
    const jumpForce = -13;
    
    if (keys['KeyA'] || keys['ArrowLeft']) {
        player.vx = -moveSpeed;
        player.facing = -1;
    } else if (keys['KeyD'] || keys['ArrowRight']) {
        player.vx = moveSpeed;
        player.facing = 1;
    } else {
        player.vx *= 0.8;
    }
    
    // Jump
    if ((keys['Space'] || keys['KeyW'] || keys['ArrowUp']) && player.jumpCount < 2) {
        if (player.vy > -5) {
            player.vy = jumpForce;
            player.jumpCount++;
            playSound('jump');
            spawnParticles(player.x + player.w/2, player.y + player.h, '#88ccff', 5);
        }
    }
    
    // Gravity
    player.vy += gravity;
    if (player.vy > 15) player.vy = 15;
    
    // Move
    player.x += player.vx;
    player.y += player.vy;
    
    // Platform collision
    player.onGround = false;
    for (let p of platforms) {
        if (p.type === 'goal') continue;
        
        const px = player.x, py = player.y, pw = player.w, ph = player.h;
        
        if (px + pw > p.x && px < p.x + p.w) {
            // Top collision (landing)
            if (player.vy >= 0 && py + ph <= p.y + 10 && py + ph + player.vy >= p.y) {
                player.y = p.y - ph;
                player.vy = 0;
                player.onGround = true;
                player.jumpCount = 0;
            }
            // Bottom collision (hitting head)
            if (player.vy < 0 && py >= p.y + p.h - 10 && py - Math.abs(player.vy) <= p.y + p.h) {
                player.y = p.y + p.h;
                player.vy = 0;
            }
        }
        
        // Side collisions
        if (py + ph > p.y && py < p.y + p.h) {
            if (player.vx > 0 && px + pw <= p.x + 10 && px + pw + player.vx >= p.x) {
                player.x = p.x - pw;
                player.vx = 0;
            } else if (player.vx < 0 && px >= p.x + p.w - 10 && px + player.vx <= p.x + p.w) {
                player.x = p.x + p.w;
                player.vx = 0;
            }
        }
    }
    
    // Fall off screen
    if (player.y > H + 50) {
        hitPlayer();
    }
    
    // Boundary
    if (player.x < -20) player.x = -20;
    if (player.x > levelWidth) player.x = levelWidth;
    
    // Crystal collection
    for (let c of crystals) {
        if (c.collected) continue;
        const dx = (player.x + player.w/2) - c.x;
        const dy = (player.y + player.h/2) - c.y;
        if (Math.abs(dx) < 30 && Math.abs(dy) < 30) {
            c.collected = true;
            crystalsCollected++;
            score += 50;
            playSound('collect');
            spawnParticles(c.x, c.y, '#ffdd44', 10);
        }
    }
    
    // Enemy collision
    if (invincibleTimer > 0) {
        invincibleTimer -= dt;
    } else {
        for (let e of enemies) {
            const dx = (player.x + player.w/2) - (e.x);
            const dy = (player.y + player.h/2) - (e.y);
            if (Math.abs(dx) < (player.w/2 + e.w/2) && Math.abs(dy) < (player.h/2 + e.h/2)) {
                // Stomp check
                if (player.vy > 0 && player.y + player.h < e.y + e.h * 0.3) {
                    e.vx = 0; e.vy = 0; e.stomped = true;
                    score += 100;
                    playSound('hit');
                    spawnParticles(e.x, e.y, '#ff6644', 12);
                } else {
                    hitPlayer();
                }
            }
        }
    }
    
    // Update enemies
    for (let e of enemies) {
        if (e.stomped) continue;
        e.x += e.vx;
        e.y += e.vy;
        e.rotation += e.rotSpeed;
        
        if (e.x < e.minX || e.x > e.maxX) e.vx *= -1;
        if (e.y < e.minY || e.y > e.maxY) e.vy *= -1;
    }
    
    // Goal check
    const goal = platforms.find(p => p.type === 'goal');
    if (goal) {
        if (player.x + player.w > goal.x && player.x < goal.x + goal.w &&
            player.y + player.h > goal.y && player.y < goal.y + goal.h) {
            gameState = 'WIN';
            score += 500;
            playSound('win');
        }
    }
    
    // Update particles
    for (let i = particles.length - 1; i >= 0; i--) {
        const p = particles[i];
        p.x += p.vx;
        p.y += p.vy;
        p.vy += 0.2;
        p.life -= p.decay;
        if (p.life <= 0) particles.splice(i, 1);
    }
    
    // Camera
    const targetX = player.x - W * 0.4;
    const targetY = player.y - H * 0.5;
    camera.x += (targetX - camera.x) * 0.08;
    camera.y += (targetY - camera.y) * 0.08;
    if (camera.x < 0) camera.x = 0;
    if (camera.x > levelWidth - W) camera.x = levelWidth - W;
    if (camera.y < -100) camera.y = -100;
    if (camera.y > H - 200) camera.y = H - 200;
    
    // Shake
    if (shakeTimer > 0) shakeTimer -= dt;
}

    // Crystals
    ctx.fillStyle = '#ffdd44';
    ctx.fillText(`◆ ${crystalsCollected}/${totalCrystals}`, 20, 60);
    
    // Lives
    ctx.fillStyle = '#ff6666';
    let livesStr = '';
    for (let i = 0; i < lives; i++) livesStr += '♥ ';
    ctx.fillText(livesStr, W - 120, 35);
    
    // Goal indicator
    if (goalX > camera.x) {
        const gx = goalX - camera.x;
        ctx.fillStyle = 'rgba(0,255,150,0.6)';

    if (Math.floor(time * 2) % 2 === 0) {

function render() {
    ctx.save();
    
    // Shake
    if (shakeTimer > 0) {
        const sx = (Math.random() - 0.5) * 8;
        const sy = (Math.random() - 0.5) * 8;
        ctx.translate(sx, sy);
    }
    
    if (gameState === 'START') {
        drawStartScreen();
        ctx.restore();
        return;
    }
    
    // Background gradient
    const grad = ctx.createLinearGradient(0, 0, 0, H);
    grad.addColorStop(0, '#0a0a2a');
    grad.addColorStop(1, '#1a1a3a');
    ctx.fillStyle = grad;
    ctx.fillRect(0, 0, W, H);
    
    drawStars();
    
    // Draw platforms
    for (let p of platforms) {
        if (p.x + p.w > camera.x - 50 && p.x < camera.x + W + 50) {
            drawPlatform(p);
        }
    }
    
    // Crystals
    for (let c of crystals) {
        if (!c.collected && Math.abs(c.x - camera.x - W/2) < W) {
            drawCrystal(c);
        }
    }
    
    // Enemies
    for (let e of enemies) {
        if (Math.abs(e.x - camera.x - W/2) < W) {
            drawEnemy(e);
        }
    }
    
    // Player
    drawPlayer();
    
    // Particles
    drawParticles();
    
    // HUD
    drawHUD();
    
    ctx.restore();
    
    if (gameState === 'GAMEOVER') drawGameOver();
    if (gameState === 'WIN') drawWin();
}

// Main Loop
let lastTimestamp = 0;
function gameLoop(timestamp) {
    const dt = Math.min((timestamp - lastTimestamp) / 16.67, 3);
    lastTimestamp = timestamp;
    
    update(dt / 60 * 16.67);
    render();
    
    requestAnimationFrame(gameLoop);
}

// Init
generateStars();
requestAnimationFrame(gameLoop);
