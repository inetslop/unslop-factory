let audioCtx = null;
let engineOsc = null;
let engineGain = null;
function initAudio() {
    if (audioCtx) return;
    audioCtx = new (window.AudioContext || window.webkitAudioContext)();
    engineOsc = audioCtx.createOscillator();
    engineOsc.type = 'sawtooth';
    engineOsc.frequency.value = 80;
    engineGain = audioCtx.createGain();
    engineGain.gain.value = 0;
    const filter = audioCtx.createBiquadFilter();
    filter.type = 'lowpass';
    filter.frequency.value = 400;
    engineOsc.connect(filter);
    filter.connect(engineGain);
    engineGain.connect(audioCtx.destination);
    engineOsc.start();
}

function playEngine(speed) {
    if (!engineOsc) return;
    const freq = 60 + speed * 0.4;
    engineOsc.frequency.setTargetAtTime(freq, audioCtx.currentTime, 0.1);
    engineGain.gain.setTargetAtTime(0.05 + speed * 0.0003, audioCtx.currentTime, 0.1);
}

function stopEngine() {
    if (engineGain) engineGain.gain.setTargetAtTime(0, audioCtx.currentTime, 0.2);
}

function playCrash() {
    if (!audioCtx) return;
    const buf = audioCtx.createBuffer(1, 4410, 44100);
    const data = buf.getChannelData(0);
    for (let i = 0; i < 4410; i++) data[i] = (Math.random() * 2 - 1) * Math.exp(-i / 500);
    const src = audioCtx.createBufferSource();
    src.buffer = buf;
    const g = audioCtx.createGain();
    g.gain.value = 0.3;
    src.connect(g);
    g.connect(audioCtx.destination);
    src.start();
}

function buildTrack() {
    segments = [];
    const addRoad = (enter, hold, leave, curve, y) => {
        for (let i = 0; i < enter; i++) addSegment(curve * (i / enter), y);
        for (let i = 0; i < hold; i++) addSegment(curve, y);
        for (let i = 0; i < leave; i++) addSegment(curve * (1 - i / leave), y);
    };
    const addStraight = (n) => addRoad(n, n, n, 0, 0);
    const addCurve = (n, c) => addRoad(n, n, n, c, 0);
    const addHill = (n, h) => {
        for (let i = 0; i < n; i++) segments.push({ curve: 0, y: Math.sin(i / n * Math.PI) * h });
    };

    addStraight(50);
    addCurve(60, 2);
    addStraight(40);
    addHill(40, 30);
    addCurve(50, -3);
    addStraight(30);
    addHill(50, -20);
    addCurve(70, 4);
    addStraight(60);
    addHill(30, 40);
    addCurve(40, -2);

// Rendering
function project(p, camX, camY, camZ) {
    const dz = p.z - camZ;
    const scale = CAM_DEPTH / dz;
    return {
        x: (p.x - camX) * scale + W / 2,
        y: H / 2 - (p.y - camY) * scale,
        w: ROAD_WIDTH * scale,
        scale: scale
    };
}

function renderRoad() {
    const baseSegment = Math.floor(position / SEGMENT_LENGTH);
    const basePercent = (position % SEGMENT_LENGTH) / SEGMENT_LENGTH;

    // Sky gradient
    const skyGrad = ctx.createLinearGradient(0, 0, 0, H * 0.5);
    skyGrad.addColorStop(0, '#1a0a3e');
    skyGrad.addColorStop(1, '#4a90d9');
    ctx.fillStyle = skyGrad;
    ctx.fillRect(0, 0, W, H);

    // Stars
    for (let i = 0; i < 50; i++) {
        const sx = (i * 137.5) % W;
        const sy = (i * 73.3) % (H * 0.3);
        ctx.fillStyle = `rgba(255,255,255,${0.3 + (i % 3) * 0.3})`;
        ctx.fillRect(sx, sy, 2, 2);
    }

    let maxy = H;
    let x = 0;
    let dx = -(segments[baseSegment % trackLength].curve * basePercent);

    const projected = [];

    for (let n = 0; n < DRAW_DISTANCE; n++) {
        const segIdx = (baseSegment + n) % trackLength;
        const seg = segments[segIdx];
        const zWorld = (n - basePercent) * SEGMENT_LENGTH;
        if (zWorld <= 0) continue;

        const camX = playerX * ROAD_WIDTH;
        const p = project(
            { x: x, y: seg.y * CAM_HEIGHT / 50, z: zWorld },
            camX, CAM_HEIGHT, 0
        );

        if (p.w > 0 && p.y < maxy) {
            projected.push({ n, segIdx, p, x, dx });
            maxy = p.y;
        }

        x += dx;
        dx += seg.curve * SEGMENT_LENGTH / ROAD_WIDTH * 50;
    }

    // Draw from far to near
    for (let i = projected.length - 1; i >= 0; i--) {
        const { p, segIdx, n } = projected[i];
        const next = i > 0 ? projected[i - 1] : null;
        if (!next) continue;

        // Fog
        const fogAmt = Math.exp(-n / (DRAW_DISTANCE / FOG_DENSITY));

        // Grass
        ctx.fillStyle = lerpColor('#2d5a1b', '#1a0a3e', 1 - fogAmt);
        ctx.fillRect(0, p.y, W, next.p.y - p.y + 1);

        // Road
        const roadL1 = p.x - p.w / 2;
        const roadR1 = p.x + p.w / 2;
        const roadL2 = next.p.x - next.p.w / 2;
        const roadR2 = next.p.x + next.p.w / 2;

        // Road surface
        ctx.fillStyle = lerpColor('#444', '#1a0a3e', 1 - fogAmt);
        ctx.beginPath();
        ctx.moveTo(roadL1, p.y);
        ctx.lineTo(roadR1, p.y);
        ctx.lineTo(roadR2, next.p.y);
        ctx.lineTo(roadL2, next.p.y);
        ctx.closePath();
        ctx.fill();

        // Road edges (rumble)
        const edgeW1 = p.w * 0.05;
        const edgeW2 = next.p.w * 0.05;
        ctx.fillStyle = lerpColor((segIdx % 2 === 0) ? '#fff' : '#c00', '#1a0a3e', 1 - fogAmt);
        // Left edge
        ctx.beginPath();
        ctx.moveTo(roadL1, p.y);
        ctx.lineTo(roadL1 + edgeW1, p.y);
        ctx.lineTo(roadL2 + edgeW2, next.p.y);
        ctx.lineTo(roadL2, next.p.y);
        ctx.closePath();
        ctx.fill();
        // Right edge
        ctx.beginPath();
        ctx.moveTo(roadR1 - edgeW1, p.y);
        ctx.lineTo(roadR1, p.y);
        ctx.lineTo(roadR2, next.p.y);
        ctx.lineTo(roadR2 - edgeW2, next.p.y);
        ctx.closePath();
        ctx.fill();

        // Center line
        if (segIdx % 3 < 2) {
            const cl1 = p.x - p.w * 0.01;
            const cr1 = p.x + p.w * 0.01;
            const cl2 = next.p.x - next.p.w * 0.01;
            const cr2 = next.p.x + next.p.w * 0.01;
            ctx.fillStyle = lerpColor('#ff0', '#1a0a3e', 1 - fogAmt);
            ctx.beginPath();
            ctx.moveTo(cl1, p.y);
            ctx.lineTo(cr1, p.y);
            ctx.lineTo(cr2, next.p.y);
            ctx.lineTo(cl2, next.p.y);
            ctx.closePath();
            ctx.fill();
        }

        // Sprites
        if (seg.sprite) {
            const sp = seg.sprite;
            const spX = p.x + sp.offset * p.w / 2;
            const spScale = p.scale * 50;
            drawTree(spX, p.y, spScale, fogAmt);
        }
    }

    // Draw traffic
    for (let t of traffic) {
        let relZ = t.z - position;
        if (relZ < 0) relZ += trackLength * SEGMENT_LENGTH;
        const n = Math.floor(relZ / SEGMENT_LENGTH);
        if (n >= DRAW_DISTANCE || n <= 0) continue;

        const segIdx = (baseSegment + n) % trackLength;
        const proj = projected.find(pp => pp.n === n);
        if (!proj) continue;

        const p = proj.p;
        const carX = p.x + t.x * p.w / 2;
        const carW = p.w * 0.15;
        const carH = carW * 0.7;

        ctx.fillStyle = lerpColor(t.color, '#1a0a3e', 1 - Math.exp(-n / (DRAW_DISTANCE / FOG_DENSITY)));
        ctx.fillRect(carX - carW / 2, p.y - carH, carW, carH);
        // Windshield
        ctx.fillStyle = lerpColor('#87ceeb', '#1a0a3e', 1 - Math.exp(-n / (DRAW_DISTANCE / FOG_DENSITY)));
        ctx.fillRect(carX - carW * 0.3, p.y - carH * 0.85, carW * 0.6, carH * 0.3);
    }

    // Draw player car

function drawTree(x, y, scale, fog) {

    ctx.lineTo(x + scale * 0.4, y - scale * 0.4);
    ctx.lineTo(x, y - scale * 1.2);
    ctx.closePath();
    ctx.fill();

    ctx.fillStyle = `rgba(50, 100, 35, ${a})`;
    ctx.beginPath();
    ctx.moveTo(x - scale * 0.35, y - scale * 0.6);
    ctx.lineTo(x + scale * 0.35, y - scale * 0.6);
    ctx.lineTo(x, y - scale * 1.4);
    ctx.closePath();
    ctx.fill();
}

function drawPlayerCar() {

function roundRect(x, y, w, h, r) {
    ctx.beginPath();
    ctx.moveTo(x + r, y);
    ctx.lineTo(x + w - r, y);
    ctx.quadraticCurveTo(x + w, y, x + w, y + r);
    ctx.lineTo(x + w, y + h - r);
    ctx.quadraticCurveTo(x + w, y + h, x + w - r, y + h);
    ctx.lineTo(x + r, y + h);
    ctx.quadraticCurveTo(x, y + h, x, y + h - r);
    ctx.quadraticCurveTo(x, y, x + r, y);
    ctx.closePath();
    ctx.fill();
}

function lerpColor(c1, c2, t) {
    const p = parseInt;
    let r1, g1, b1, r2, g2, b2;
    if (c1.startsWith('#')) {
        r1 = p(c1.substr(1, 2), 16); g1 = p(c1.substr(3, 2), 16); b1 = p(c1.substr(5, 2), 16);
    } else return c1;
    if (c2.startsWith('#')) {
        r2 = p(c2.substr(1, 2), 16); g2 = p(c2.substr(3, 2), 16); b2 = p(c2.substr(5, 2), 16);
    } else return c1;
    const r = Math.round(r1 + (r2 - r1) * t);
    const g = Math.round(g1 + (g2 - g1) * t);
    const b = Math.round(b1 + (b2 - b1) * t);
    return `rgb(${r},${g},${b})`;
}

function update(dt) {
    if (gameState !== STATE.PLAYING) return;

    const baseSeg = segments[Math.floor(position / SEGMENT_LENGTH) % trackLength];

    // Speed control
    if (keys['ArrowUp'] || keys['KeyW']) {
        speed += ACCEL * dt * 60;
    } else if (keys['ArrowDown'] || keys['KeyS']) {
        speed += BRAKE * dt * 60;
    } else {
        speed += DECEL * dt * 60;
    }

    // Off-road deceleration
    if (Math.abs(playerX) > 1.0) {
        speed += OFF_DECEL * dt * 60;
        // Bump effect
        playerY = Math.sin(Date.now() * 0.02) * 5;
    } else {
        playerY = 0;
    }

    speed = Math.max(0, Math.min(speed, MAX_SPEED));

    // Steer
    const steerAmt = 0.03 * (speed / MAX_SPEED + 0.3);
    if (keys['ArrowLeft'] || keys['KeyA']) playerX -= steerAmt * dt * 60;
    if (keys['ArrowRight'] || keys['KeyD']) playerX += steerAmt * dt * 60;

    // Centripetal force from curves
    playerX -= baseSeg.curve * (speed / MAX_SPEED) * 0.015 * dt * 60;

    playerX = Math.max(-2, Math.min(2, playerX));

    // Move forward
    position += speed * dt;
    if (position >= trackLength * SEGMENT_LENGTH) position -= trackLength * SEGMENT_LENGTH;

    // Update traffic
    for (let t of traffic) {
        t.z += t.speed * dt;
        if (t.z >= trackLength * SEGMENT_LENGTH) t.z -= trackLength * SEGMENT_LENGTH;

        // Collision check
        let relZ = t.z - position;
        if (relZ < 0) relZ += trackLength * SEGMENT_LENGTH;
        if (relZ < SEGMENT_LENGTH * 2 && Math.abs(t.x - playerX) < 0.35) {
            // Crash!
            gameState = STATE.GAMEOVER;
            playCrash();
            stopEngine();
            try { localStorage.setItem('turboRushHigh', String(Math.floor(position / SEGMENT_LENGTH))); } catch(e) {}
        }
    }

    playEngine(speed);
}

    ctx.fillText(`DISTANCE: ${dist}m`, W / 2, H * 0.5);

    let best = 0;
    try { best = parseInt(localStorage.getItem('turboRushHigh') || '0'); } catch(e) {}
    if (dist >= best) {
        ctx.fillStyle = '#ffcc00';
        ctx.fillText('NEW RECORD!', W / 2, H * 0.58);
    }

    const pulse = 0.5 + Math.sin(Date.now() * 0.003) * 0.5;
    ctx.fillStyle = `rgba(255,255,100,${pulse})`;
    ctx.font = `${Math.min(W * 0.03, 24)}px monospace`;
    ctx.fillText('PRESS SPACE TO RESTART', W / 2, H * 0.75);
}

// Main loop
let lastTime = 0;
function gameLoop(time) {
    const dt = Math.min((time - lastTime) / 1000, 0.05);
    lastTime = time;

    if (gameState === STATE.MENU) {
        renderMenu();
    } else if (gameState === STATE.PLAYING) {
        update(dt);
        renderRoad();
        renderHUD();
    } else if (gameState === STATE.GAMEOVER) {
        renderRoad();
        renderGameOver();
    }

    requestAnimationFrame(gameLoop);
}

requestAnimationFrame(gameLoop);
