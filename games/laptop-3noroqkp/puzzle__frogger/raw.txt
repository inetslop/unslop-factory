```html
<!DOCTYPE html>
<html lang="en">
<head>
<meta charset="UTF-8">
<meta name="viewport" content="width=device-width, initial-scale=1.0">
<title>Frogger Puzzle</title>
<style>
* { margin: 0; padding: 0; box-sizing: border-box; }

#overlay {
position: absolute; top: 0; left: 0; width: 100%; height: 100%;
display: flex; flex-direction: column; align-items: center; justify-content: center;

}
#overlay h1 { color: #4a4; font-size: clamp(24px, 5vw, 48px); margin-bottom: 10px; text-shadow: 0 0 20px #4a4; }
#overlay p { color: #aaa; font-size: clamp(12px, 2vw, 16px); margin: 4px 0; }
#overlay .btn {
margin-top: 20px; padding: 12px 32px; background: #2a5; color: #fff; border: none;

}

.hidden { display: none !important; }

<body>
<canvas id="game"></canvas>
<div id="overlay">
<h1>FROGGER PUZZLE</h1>
<p>Navigate the frog from bottom to lily pads.</p>
<p>Move: Arrow Keys / WASD</p>
<p>All hazards move each turn — plan your path!</p>
<p>Reach a lily pad to win. Avoid cars & water.</p>
<button class="btn" id="startBtn">START GAME</button>
</div>

<script>
const canvas = document.getElementById('game');
const ctx = canvas.getContext('2d');
const overlay = document.getElementById('overlay');
const startBtn = document.getElementById('startBtn');

const COLS = 13;
const ROWS = 14;
let cellSize, offsetX, offsetY;
let gameState = 'menu'; // menu, playing, won, lost
let frog = { x: 6, y: 12 };
let moveCount = 0;
let animFrame = 0;
let audioCtx = null;

// Grid types: 0=road, 1=river, 2=safe(alley), 3=goal(lily), 4=start
const gridType = [];
for (let r = 0; r < ROWS; r++) {
gridType[r] = [];
for (let c = 0; c < COLS; c++) {
if (r === 0) gridType[r][c] = 3;
else if (r >= 1 && r <= 3) gridType[r][c] = 1;
else if (r === 4) gridType[r][c] = 2;
else if (r >= 5 && r <= 8) gridType[r][c] = 0;
else if (r === 9) gridType[r][c] = 2;
else gridType[r][c] = 4;
}
}

// Lily pad positions (row 0)
const lilyPads = [1, 3, 5, 7, 9, 11];

// Cars: {col, row, dir(1 or -1), speed}
let cars = [];
// Logs: {col, row, dir, speed}
let logs = [];

function initHazards() {
cars = [];
logs = [];

// Road rows: 5,6,7,8
const roadConfigs = [
{ row: 5, count: 3, dir: 1, speed: 1 },
{ row: 6, count: 2, dir: -1, speed: 1 },
{ row: 7, count: 3, dir: 1, speed: 2 },
{ row: 8, count: 2, dir: -1, speed: 1 }
];

roadConfigs.forEach(cfg => {
for (let i = 0; i < cfg.count; i++) {
const col = Math.floor((i * COLS) / cfg.count);
cars.push({ col, row: cfg.row, dir: cfg.dir, speed: cfg.speed });
}
});

// River rows: 1,2,3
const riverConfigs = [
{ row: 1, count: 3, dir: -1, speed: 1 },
{ row: 2, count: 4, dir: 1, speed: 1 },
{ row: 3, count: 3, dir: -1, speed: 2 }
];

riverConfigs.forEach(cfg => {
for (let i = 0; i < cfg.count; i++) {
const col = Math.floor((i * COLS) / cfg.count);
logs.push({ col, row: cfg.row, dir: cfg.dir, speed: cfg.speed });
}
});
}

function resize() {
const w = window.innerWidth;
const h = window.innerHeight;
canvas.width = w;
canvas.height = h;
cellSize = Math.min(Math.floor(w / COLS), Math.floor(h / ROWS));
offsetX = Math.floor((w - cellSize * COLS) / 2);
offsetY = Math.floor((h - cellSize * ROWS) / 2);
}

window.addEventListener('resize', resize);
resize();

function playTone(freq, dur, type) {
if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
const osc = audioCtx.createOscillator();
const gain = audioCtx.createGain();
osc.type = type || 'square';
osc.frequency.value = freq;
gain.gain.setValueAtTime(0.15, audioCtx.currentTime);
gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
osc.connect(gain);
gain.connect(audioCtx.destination);
osc.start();
osc.stop(audioCtx.currentTime + dur);
}

function playMove() { playTone(220, 0.08, 'square'); }
function playWin() { playTone(440, 0.15); setTimeout(() => playTone(660, 0.15), 150); setTimeout(() => playTone(880, 0.3), 300); }
function playLose() { playTone(150, 0.3, 'sawtooth'); }

function moveHazard(h, dir, speed) {
h.col += h.dir * speed;
if (h.dir > 0 && h.col >= COLS - 1) h.col = 0;
if (h.dir < 0 && h.col <= 0) h.col = COLS - 1;
}

function moveAll() {
cars.forEach(c => moveHazard(c, null, c.speed));
logs.forEach(l => moveHazard(l, null, l.speed));
}

function checkWin() {
return frog.y === 0 && lilyPads.includes(frog.x);
}

function checkDeath() {
// In river without log
if (frog.y >= 1 && frog.y <= 3) {
const onLog = logs.some(l => l.row === frog.y && l.col === frog.x);
if (!onLog) return true;
}
// On road with car
if (frog.y >= 5 && frog.y <= 8) {
const hitCar = cars.some(c => c.row === frog.y && c.col === frog.x);
if (hitCar) return true;
}
return false;
}

function tryMove(dx, dy) {
let nx = frog.x + dx;
let ny = frog.y + dy;
if (nx < 0 || nx >= COLS || ny < 0 || ny >= ROWS) return;

frog.x = nx;
frog.y = ny;
moveCount++;
playMove();

// Move hazards after frog moves
moveAll();

animFrame = 15;

if (checkWin()) {
gameState = 'won';
playWin();
} else if (checkDeath()) {
gameState = 'lost';
playLose();
}
}

function resetGame() {
frog = { x: 6, y: 12 };
moveCount = 0;
animFrame = 0;
initHazards();
gameState = 'playing';
}

startBtn.addEventListener('click', () => {
if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
overlay.classList.add('hidden');
resetGame();
});

document.addEventListener('keydown', (e) => {
if (gameState === 'menu') return;

if (gameState === 'won' || gameState === 'lost') {
if (e.key === 'r' || e.key === 'R' || e.key === ' ') {
resetGame();
}
return;
}

if (gameState !== 'playing') return;

let dx = 0, dy = 0;
switch (e.key) {
case 'ArrowUp': case 'w': case 'W': dy = -1; break;
case 'ArrowDown': case 's': case 'S': dy = 1; break;
case 'ArrowLeft': case 'a': case 'A': dx = -1; break;
case 'ArrowRight': case 'd': case 'D': dx = 1; break;
default: return;
}
e.preventDefault();
tryMove(dx, dy);
});

// Touch controls
let touchStartX = 0, touchStartY = 0;
canvas.addEventListener('touchstart', (e) => {
const t = e.touches[0];
touchStartX = t.clientX;
touchStartY = t.clientY;
});
canvas.addEventListener('touchend', (e) => {
if (gameState !== 'playing') return;
const t = e.changedTouches[0];
const dx = t.clientX - touchStartX;
const dy = t.clientY - touchStartY;
if (Math.abs(dx) > Math.abs(dy)) {
tryMove(Math.sign(dx), 0);
} else {
tryMove(0, Math.sign(dy));
}
});

function drawCell(x, y, type) {
const px = offsetX + x * cellSize;
const py = offsetY + y * cellSize;
const s = cellSize;

if (type === 3) { // lily pad row background
ctx.fillStyle = '#0a2a4a';
} else if (type === 1) { // river
ctx.fillStyle = '#0d3b6e';
} else if (type === 0) { // road
ctx.fillStyle = '#2d2d2d';
} else if (type === 2) { // alley
ctx.fillStyle = '#1a4a1a';
} else { // start
ctx.fillStyle = '#1a3a1a';
}
ctx.fillRect(px, py, s, s);

// Grid lines
ctx.strokeStyle = 'rgba(0,0,0,0.2)';
ctx.lineWidth = 1;
ctx.strokeRect(px, py, s, s);

// Road markings
if (type === 0 && y % 2 === 0) {
ctx.fillStyle = '#555';
for (let i = 0; i < COLS; i++) {
const cx = offsetX + i * cellSize + s / 2;
ctx.fillRect(cx - 2, py + s / 2 - 1, 4, 2);
}
}

// Water ripples
if (type === 1) {
ctx.fillStyle = 'rgba(100,180,255,0.1)';
for (let i = 0; i < 3; i++) {
const rx = px + Math.random() * s;
const ry = py + Math.random() * s;
ctx.beginPath();
ctx.arc(rx, ry, 2, 0, Math.PI * 2);
ctx.fill();
}
}
}

function drawLilyPad(x) {
const px = offsetX + x * cellSize;
const py = offsetY + 0 * cellSize;
const s = cellSize;
ctx.fillStyle = '#2d8a2d';
ctx.beginPath();
ctx.arc(px + s/2, py + s/2, s * 0.4, 0, Math.PI * 2);
ctx.fill();
// Notch
ctx.fillStyle = '#0d3b6e';
ctx.beginPath();
ctx.moveTo(px + s/2, py + s/2);
ctx.lineTo(px + s/2 + s*0.15, py + s*0.15);
ctx.lineTo(px + s/2 - s*0.15, py + s*0.15);
ctx.fill();
}

function drawCar(car) {
const px = offsetX + car.col * cellSize;
const py = offsetY + car.row * cellSize;
const s = cellSize;
const pad = 2;
ctx.fillStyle = '#cc3333';
ctx.fillRect(px + pad, py + pad, s - pad*2, s - pad*2);
// Windshield
ctx.fillStyle = '#87CEEB';
ctx.fillRect(px + pad + 3, py + pad + 3, s - pad*2 - 6, (s - pad*2) * 0.3);
}

function drawLog(log) {
const px = offsetX + log.col * cellSize;
const py = offsetY + log.row * cellSize;
const s = cellSize;
const pad = 3;
ctx.fillStyle = '#8B5E3C';
ctx.fillRect(px + pad, py + pad, s - pad*2, s - pad*2);
// Wood grain
ctx.strokeStyle = '#6B4226';
ctx.lineWidth = 1;
for (let i = 0; i < 3; i++) {
const ly = py + pad + (s - pad*2) * (i+1) / 4;
ctx.beginPath();
ctx.moveTo(px + pad + 2, ly);
ctx.lineTo(px + s - pad - 2, ly);
ctx.stroke();
}
}

function drawFrog() {
const px = offsetX + frog.x * cellSize;
const py = offsetY + frog.y * cellSize;
const s = cellSize;
const cx = px + s/2;
const cy = py + s/2;

// Body
ctx.fillStyle = '#32CD32';
ctx.beginPath();
ctx.arc(cx, cy, s * 0.35, 0, Math.PI * 2);
ctx.fill();

// Eyes
ctx.fillStyle = '#fff';
ctx.beginPath();
ctx.arc(cx - s*0.15, cy - s*0.2, s*0.1, 0, Math.PI * 2);
ctx.fill();
ctx.beginPath();
ctx.arc(cx + s*0.15, cy - s*0.2, s*0.1, 0, Math.PI * 2);
ctx.fill();

// Pupils
ctx.fillStyle = '#000';
ctx.beginPath();
ctx.arc(cx - s*0.15, cy - s*0.22, s*0.05, 0, Math.PI * 2);
ctx.fill();
ctx.beginPath();
ctx.arc(cx + s*0.15, cy - s*0.22, s*0.05, 0, Math.PI * 2);
ctx.fill();

// Legs
ctx.fillStyle = '#228B22';
ctx.fillRect(cx - s*0.35, cy + s*0.15, s*0.2, s*0.2);
ctx.fillRect(cx + s*0.15, cy + s*0.15, s*0.2, s*0.2);
}

function drawHUD() {
ctx.fillStyle = '#fff';
ctx.font = `${Math.max(12, cellSize * 0.4)}px monospace`;
ctx.textAlign = 'left';
ctx.fillText(`Moves: ${moveCount}`, offsetX + 5, offsetY - 8);

if (gameState === 'won') {
ctx.fillStyle = '#4f4';
ctx.font = `bold ${Math.max(20, cellSize * 1.2)}px monospace`;
ctx.textAlign = 'center';
ctx.fillText('🎉 YOU WIN! 🎉', canvas.width/2, canvas.height/2 - 20);
ctx.fillStyle = '#fff';
ctx.font = `${Math.max(14, cellSize * 0.5)}px monospace`;
ctx.fillText(`Solved in ${moveCount} moves`, canvas.width/2, canvas.height/2 + 20);
ctx.fillText('Press R or Space to restart', canvas.width/2, canvas.height/2 + 50);
}

if (gameState === 'lost') {
ctx.fillStyle = '#f44';
ctx.font = `bold ${Math.max(20, cellSize * 1.2)}px monospace`;
ctx.textAlign = 'center';
ctx.fillText('💀 SPLAT! 💀', canvas.width/2, canvas.height/2 - 20);
ctx.fillStyle = '#fff';
ctx.font = `${Math.max(14, cellSize * 0.5)}px monospace`;
ctx.fillText('Press R or Space to restart', canvas.width/2, canvas.height/2 + 20);
}
}

function drawArrowIndicator() {
if (gameState !== 'playing') return;
// Show which row the frog is in as a hint
const fx = offsetX + frog.x * cellSize + cellSize/2;
const fy = offsetY + frog.y * cellSize + cellSize/2;
ctx.fillStyle = 'rgba(255,255,255,0.3)';
ctx.font = `${cellSize * 0.3}px monospace`;
ctx.textAlign = 'center';
}

function render() {
ctx.clearRect(0, 0, canvas.width, canvas.height);
ctx.fillStyle = '#1a1a2e';
ctx.fillRect(0, 0, canvas.width, canvas.height);

// Draw grid
for (let r = 0; r < ROWS; r++) {
for (let c = 0; c < COLS; c++) {
drawCell(c, r, gridType[r][c]);
}
}

// Lily pads
lilyPads.forEach(x => drawLilyPad(x));

// Logs
logs.forEach(drawLog);

// Cars
cars.forEach(drawCar);

// Frog
if (gameState === 'playing' || gameState === 'won') {
drawFrog();
} else if (gameState === 'lost') {
// Draw X eyes
const px = offsetX + frog.x * cellSize;
const py = offsetY + frog.y * cellSize;
const s = cellSize;
ctx.fillStyle = '#32CD32';
ctx.beginPath();
ctx.arc(px + s/2, py + s/2, s*0.35, 0, Math.PI * 2);
ctx.fill();
ctx.strokeStyle = '#000';
ctx.lineWidth = 2;
const ex = px + s/2 - s*0.15, ey = py + s/2 - s*0.2;
ctx.beginPath(); ctx.moveTo(ex-4,ey-4); ctx.lineTo(ex+4,ey+4); ctx.stroke();
ctx.beginPath(); ctx.moveTo(ex+4,ey-4); ctx.lineTo(ex-4,ey+4); ctx.stroke();
const ex2 = px + s/2 + s*0.15;
ctx.beginPath(); ctx.moveTo(ex2-4,ey-4); ctx.lineTo(ex2+4,ey+4); ctx.stroke();
ctx.beginPath(); ctx.moveTo(ex2+4,ey-4); ctx.lineTo(ex2-4,ey+4); ctx.stroke();
}

drawHUD();
}

function gameLoop() {
if (gameState === 'playing' && animFrame > 0) animFrame--;
render();
requestAnimationFrame(gameLoop);
}

initHazards();
gameLoop();
</script>
</body>
</html>
```
