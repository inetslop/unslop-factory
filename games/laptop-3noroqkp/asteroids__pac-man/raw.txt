// Audio
let audioCtx = null;
function initAudio() {
if (!audioCtx) audioCtx = new (window.AudioContext || webkitAudioContext)();
}
function playSound(type) {
if (!audioCtx) return;
const osc = audioCtx.createOscillator();
const gain = audioCtx.createGain();
osc.connect(gain);
gain.connect(audioCtx.destination);
switch(type) {
case 'dot':
osc.type = 'sine'; osc.frequency.value = 600 + Math.random() * 200;
gain.gain.setValueAtTime(0.1, audioCtx.currentTime);
gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.1);
osc.start(); osc.stop(audioCtx.currentTime + 0.1); break;
case 'pellet':
osc.type = 'sine'; osc.frequency.value = 400;
gain.gain.setValueAtTime(0.2, audioCtx.currentTime);
gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.3);
osc.start(); osc.stop(audioCtx.currentTime + 0.3); break;
case 'shoot':
osc.type = 'square'; osc.frequency.value = 200;
gain.gain.setValueAtTime(0.1, audioCtx.currentTime);
gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.1);
osc.start(); osc.stop(audioCtx.currentTime + 0.1); break;
case 'explode':
osc.type = 'sawtooth'; osc.frequency.value = 80;
gain.gain.setValueAtTime(0.3, audioCtx.currentTime);
gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.4);
osc.start(); osc.stop(audioCtx.currentTime + 0.4); break;
case 'hit':
osc.type = 'sawtooth'; osc.frequency.value = 60;
gain.gain.setValueAtTime(0.4, audioCtx.currentTime);
gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.5);
osc.start(); osc.stop(audioCtx.currentTime + 0.5); break;
case 'ghost':
osc.type = 'square'; osc.frequency.value = 300;
gain.gain.setValueAtTime(0.2, audioCtx.currentTime);
gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.2);
osc.start(); osc.stop(audioCtx.currentTime + 0.2); break;
}
}

initAudio();
});

// Player
let player = null;
function createPlayer() {
return { x: W/2, y: H/2, angle: -Math.PI/2, vx: 0, vy: 0, thrusting: false, invincible: 0, radius: 15 };
}

// Bullets
let bullets = [];
function shootBullet() {
if (!player || player.invincible > 0) return;
const speed = 8;
bullets.push({ x: player.x + Math.cos(player.angle)*20, y: player.y + Math.sin(player.angle)*20, vx: Math.cos(player.angle)*speed + player.vx*0.5, vy: Math.sin(player.angle)*speed + player.vy*0.5, life: 60 });
playSound('shoot');
}

// Dots & Pellets
let dots = [];
let pellets = [];
function spawnDots() {
dots = [];
pellets = [];
const spacing = 80;
for (let x = spacing/2; x < W - spacing/2; x += spacing) {
for (let y = spacing/2; y < H - spacing/2; y += spacing) {
if (Math.random() < 0.75) {
dots.push({ x: x + (Math.random()-0.5)*30, y: y + (Math.random()-0.5)*30, eaten: false });
}
}
}
// Pellets at random spots
for (let i = 0; i < 4; i++) {
const dx = Math.random() * W;
const dy = Math.random() * H;
pellets.push({ x: dx, y: dy, eaten: false, pulse: 0 });
}
}

// Asteroids
let asteroids = [];
function createAsteroid(x, y, size) {
const speed = (1 + Math.random() * 2) * (size === 3 ? 1 : size === 2 ? 0.7 : 0.4);
const angle = Math.random() * Math.PI * 2;
return { x, y, vx: Math.cos(angle)*speed, vy: Math.sin(angle)*speed, size, radius: size * 20 + 10, verts: generateVerts(size), rot: 0, rotSpeed: (Math.random()-0.5)*0.05 };
}
function generateVerts(size) {
const n = 8 + Math.floor(Math.random() * 4);
const v = [];
for (let i = 0; i < n; i++) {
const a = (i/n) * Math.PI * 2;
const r = 0.7 + Math.random() * 0.3;
v.push({ angle: a, dist: r });
}
return v;
}
function spawnAsteroids(count) {
for (let i = 0; i < count; i++) {
let x, y;
do { x = Math.random() * W; y = Math.random() * H; } while (player && dist(x,y,player.x,player.y) < 150);
asteroids.push(createAsteroid(x, y, 3));
}
}

// Enemies (Ghost ships)
let enemies = [];
const enemyColors = ['#ff4444', '#44ff44', '#4488ff', '#ffff44', '#ff44ff'];
function spawnEnemies(count) {
for (let i = 0; i < count; i++) {
let x, y;
do { x = Math.random() * W; y = Math.random() * H; } while (player && dist(x,y,player.x,player.y) < 250);
enemies.push({ x, y, vx: (Math.random()-0.5)*3, vy: (Math.random()-0.5)*3, angle: Math.random()*Math.PI*2, radius: 18, color: enemyColors[i % enemyColors.length], vulnerable: false, vulnTimer: 0, speed: 1.5 + level * 0.3 });
}
}

// Particles
let particles = [];
function spawnParticles(x, y, color, count) {
for (let i = 0; i < count; i++) {
const a = Math.random() * Math.PI * 2;
const s = Math.random() * 4 + 1;
particles.push({ x, y, vx: Math.cos(a)*s, vy: Math.sin(a)*s, life: 30 + Math.random()*20, color });
}
}

// Stars background
let stars = [];

stars = [];
for (let i = 0; i < 150; i++) {
stars.push({ x: Math.random()*W, y: Math.random()*H, size: Math.random()*1.5+0.5, brightness: Math.random() });
}
}

// Utilities
function dist(x1,y1,x2,y2) { const dx=x2-x1, dy=y2-y1; return Math.sqrt(dx*dx+dy*dy); }
function wrap(obj) {
if (obj.x < -50) obj.x = W + 50;
if (obj.x > W + 50) obj.x = -50;
if (obj.y < -50) obj.y = H + 50;
if (obj.y > H + 50) obj.y = -50;
}

// Game functions

state = 'PLAYING';
score = 0; lives = 3; level = 1;
startLevel();

function startLevel() {
player = createPlayer();
bullets = [];
asteroids = [];
enemies = [];
particles = [];
spawnDots();
spawnAsteroids(3 + level);
spawnEnemies(Math.min(2 + Math.floor(level/1), 5));
generateStars();
}

function nextLevel() {
level++;
state = 'LEVEL_TRANSITION';
transitionTimer = 90;
}

function loseLife() {
lives--;
shakeTimer = 20;
playSound('hit');
if (lives <= 0) { state = 'GAME_OVER'; }
else { player.x = W/2; player.y = H/2; player.angle = -Math.PI/2; player.vx = 0; player.vy = 0; player.invincible = 120; }
}

// Update
function update() {
frameCount++;

if (state === 'LEVEL_TRANSITION') {
transitionTimer--;
if (transitionTimer <= 0) { startLevel(); state = 'PLAYING'; }
return;
}
if (state !== 'PLAYING') return;

// Player movement
if (player) {
if (keys['ArrowLeft'] || keys['KeyA']) player.angle -= 0.08;
if (keys['ArrowRight'] || keys['KeyD']) player.angle += 0.08;
if (keys['ArrowUp'] || keys['KeyW']) {
player.vx += Math.cos(player.angle) * 0.3;
player.vy += Math.sin(player.angle) * 0.3;
player.thrusting = true;
} else player.thrusting = false;

// Friction
player.vx *= 0.98;
player.vy *= 0.98;
const maxSpeed = 5;
const spd = Math.sqrt(player.vx*player.vx + player.vy*player.vy);
if (spd > maxSpeed) { player.vx = player.vx/spd*maxSpeed; player.vy = player.vy/spd*maxSpeed; }

player.x += player.vx;
player.y += player.vy;
wrap(player);
if (player.invincible > 0) player.invincible--;

// Shoot
if (keys['Space'] && frameCount % 12 === 0) shootBullet();
}

// Bullets
for (let i = bullets.length-1; i >= 0; i--) {
const b = bullets[i];
b.x += b.vx; b.y += b.vy; wrap(b);
b.life--;
if (b.life <= 0) { bullets.splice(i,1); continue; }

// Bullet hits asteroid
let hitAsteroid = false;
for (let j = asteroids.length-1; j >= 0; j--) {
const a = asteroids[j];
if (dist(b.x,b.y,a.x,a.y) < a.radius + 5) {
hitAsteroid = true;
score += a.size * 10;
playSound('explode');
spawnParticles(a.x, a.y, '#aaa', 12);
asteroids.splice(j,1);
if (a.size > 1) {
for (let k = 0; k < 2; k++) asteroids.push(createAsteroid(a.x + (Math.random()-0.5)*20, a.y + (Math.random()-0.5)*20, a.size-1));
}
break;
}
}
if (hitAsteroid) { bullets.splice(i,1); continue; }

// Bullet hits enemy
let hitEnemy = false;
for (let j = enemies.length-1; j >= 0; j--) {
const e = enemies[j];
if (dist(b.x,b.y,e.x,e.y) < e.radius + 5) {
if (e.vulnerable || true) { // Can always shoot ghosts but vulnerable gives bonus
score += e.vulnerable ? 200 : 100;
playSound('ghost');
spawnParticles(e.x, e.y, e.color, 15);
enemies.splice(j,1);
hitEnemy = true;
break;
}
}
if (hitEnemy) { bullets.splice(i,1); continue; }
}

// Dots collection
if (player) {
for (let i = dots.length-1; i >= 0; i--) {
const d = dots[i];
if (!d.eaten && dist(player.x, player.y, d.x, d.y) < 20) {
d.eaten = true;
score += 10;
playSound('dot');
dots.splice(i,1);
}
}
for (let i = pellets.length-1; i >= 0; i--) {
const p = pellets[i];
if (!p.eaten && dist(player.x, player.y, p.x, p.y) < 25) {
p.eaten = true;
score += 50;
playSound('pellet');
// Make enemies vulnerable
enemies.forEach(e => { e.vulnerable = true; e.vulnTimer = 300; });
pellets.splice(i,1);
}
}

// Check level complete
if (dots.length === 0 && pellets.length === 0) {
nextLevel();
return;
}
}

// Asteroids update
for (const a of asteroids) { a.x += a.vx; a.y += a.vy; wrap(a); a.rot += a.rotSpeed; }

// Enemies update
for (let i = enemies.length-1; i >= 0; i--) {
const e = enemies[i];
if (e.vulnTimer > 0) { e.vulnTimer--; if (e.vulnTimer <= 0) e.vulnerable = false; }

if (player && player.invincible <= 0) {
const dx = player.x - e.x, dy = player.y - e.y;
const d = Math.sqrt(dx*dx+dy*dy) || 1;
// Chase with some randomness
const chaseForce = e.vulnerable ? -0.08 : 0.05;
e.vx += (dx/d) * chaseForce + (Math.random()-0.5)*0.1;
e.vy += (dy/d) * chaseForce + (Math.random()-0.5)*0.1;
const spd = Math.sqrt(e.vx*e.vx+e.vy*e.vy);
if (spd > e.speed) { e.vx = e.vx/spd*e.speed; e.vy = e.vy/spd*e.speed; }
} else {
e.vx += (Math.random()-0.5)*0.1;
e.vy += (Math.random()-0.5)*0.1;
const spd = Math.sqrt(e.vx*e.vx+e.vy*e.vy);
if (spd > e.speed) { e.vx = e.vx/spd*e.speed; e.vy = e.vy/spd*e.speed; }
}
e.x += e.vx; e.y += e.vy; wrap(e);
e.angle = Math.atan2(e.vy, e.vx);

// Enemy catches player
if (player && player.invincible <= 0 && dist(player.x,player.y,e.x,e.y) < e.radius + player.radius - 5) {
loseLife();
spawnParticles(player.x, player.y, '#fff', 20);
}
}

// Asteroid hits player
if (player && player.invincible <= 0) {
for (let i = asteroids.length-1; i >= 0; i--) {
const a = asteroids[i];
if (dist(player.x, player.y, a.x, a.y) < a.radius + player.radius - 5) {
loseLife();
spawnParticles(a.x, a.y, '#aaa', 12);
asteroids.splice(i,1);
if (a.size > 1) {
for (let k = 0; k < 2; k++) asteroids.push(createAsteroid(a.x + (Math.random()-0.5)*20, a.y + (Math.random()-0.5)*20, a.size-1));
}
break;
}
}
}

// Particles
for (let i = particles.length-1; i >= 0; i--) {
const p = particles[i];
p.x += p.vx; p.y += p.vy; p.life--;
if (p.life <= 0) particles.splice(i,1);
}

// Pellet pulse
for (const p of pellets) p.pulse += 0.1;
}

// Stars

}

ctx.fill();

ctx.shadowBlur = 0;
ctx.beginPath(); ctx.arc(-5, -6, 4, 0, Math.PI*2); ctx.fill();
ctx.beginPath(); ctx.arc(5, -6, 4, 0, Math.PI*2); ctx.fill();
ctx.fillStyle = '#000';
const lookX = Math.cos(e.angle)*2;
const lookY = Math.sin(e.angle)*2;
ctx.beginPath(); ctx.arc(-5+lookX, -6+lookY, 2, 0, Math.PI*2); ctx.fill();
ctx.beginPath(); ctx.arc(5+lookX, -6+lookY, 2, 0, Math.PI*2); ctx.fill();

}

// Bullets
for (const b of bullets) {
ctx.fillStyle = '#ff4444';
ctx.shadowColor = '#ff4444';
ctx.shadowBlur = 5;
ctx.beginPath();
ctx.arc(b.x, b.y, 3, 0, Math.PI*2);
ctx.fill();
ctx.shadowBlur = 0;
}

// Player
if (player) {
const blink = player.invincible > 0 && Math.floor(frameCount/5) % 2 === 0;
if (!blink) {
ctx.save();
ctx.translate(player.x, player.y);
ctx.rotate(player.angle);
ctx.strokeStyle = '#fff';
ctx.lineWidth = 2;
ctx.shadowColor = '#44ff44';
ctx.shadowBlur = 8;

ctx.lineTo(-6, 0);
ctx.lineTo(-10, 10);

ctx.stroke();

// Thrust flame
if (player.thrusting) {
ctx.strokeStyle = '#ff8800';
ctx.beginPath();
ctx.moveTo(-10, -4);
ctx.lineTo(-18 + Math.random()*5, 0);
ctx.lineTo(-10, 4);
ctx.stroke();
}
ctx.shadowBlur = 0;
ctx.restore();
}
}

// Particles
for (const p of particles) {
const a = p.life / 50;
ctx.fillStyle = `rgba(${p.color === '#fff' ? '255,255,255' : p.color === '#aaa' ? '170,170,170' : '255,100,100'},${a})`;
ctx.fillRect(p.x-1, p.y-1, 3, 3);
}

// HUD
ctx.fillStyle = '#fff';
ctx.font = 'bold 16px monospace';
ctx.textAlign = 'left';
ctx.fillText('SCORE: ' + score, 20, 30);
ctx.fillText('LEVEL: ' + level, 20, 55);
ctx.fillText('LIVES: ' + '❤'.repeat(Math.max(0,lives)), 20, 80);

// Dots remaining

}
