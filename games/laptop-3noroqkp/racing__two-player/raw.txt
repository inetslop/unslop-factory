const LAPS_TO_WIN = 3;
const TRACK_WIDTH_RATIO = 0.13;
const ACCEL = 400;
const MAX_SPEED = 320;
const FRICTION = 150;
const REVERSE_ACCEL = 250;
const MAX_REVERSE = -100;
const TURN_RATE = 3.8;
const OFF_TRACK_DRAG = 500;

let gameState = 'start';
let countdown = 0;
let winner = null;
let keys = {};
let lastTime = 0;

const trackCX = () => W / 2;
const trackCY = () => H / 2;
const trackA = () => Math.min(W, H) * 0.38;
const trackB = () => Math.min(W, H) * 0.38 * 0.65;
const trackW = () => Math.min(W, H) * TRACK_WIDTH_RATIO;

class Car {
    constructor(x, y, angle, color, glowColor) {
        this.x = x;
        this.y = y;
        this.angle = angle;
        this.speed = 0;
        this.color = color;
        this.glowColor = glowColor;
        this.laps = 0;
        this.prevAngle = angle;
        this.accumulatedAngle = 0;
        this.finished = false;
    }

    update(dt, up, down, left, right) {
        if (this.finished) return;

        const onTrack = isOnTrack(this.x, this.y);

        if (up) {
            this.speed += ACCEL * dt;
        } else if (down) {
            if (this.speed > 0) {
                this.speed -= REVERSE_ACCEL * 1.5 * dt;
            } else {
                this.speed -= REVERSE_ACCEL * dt;
            }
        } else {
            if (this.speed > 0) {
                this.speed = Math.max(0, this.speed - FRICTION * dt);
            } else if (this.speed < 0) {
                this.speed = Math.min(0, this.speed + FRICTION * dt);
            }
        }

        if (!onTrack) {
            const drag = OFF_TRACK_DRAG * dt;
            if (this.speed > 0) this.speed = Math.max(0, this.speed - drag);
            else if (this.speed < 0) this.speed = Math.min(0, this.speed + drag);
        }

        this.speed = Math.max(MAX_REVERSE, Math.min(MAX_SPEED, this.speed));

        const turnFactor = Math.min(1, Math.abs(this.speed) / 60);
        if (left && Math.abs(this.speed) > 5) {
            this.angle -= TURN_RATE * turnFactor * dt;
        } else if (right && Math.abs(this.speed) > 5) {
            this.angle += TURN_RATE * turnFactor * dt;
        }

        this.x += Math.cos(this.angle) * this.speed * dt;
        this.y += Math.sin(this.angle) * this.speed * dt;

        // Keep in bounds
        const margin = 30;
        this.x = Math.max(margin, Math.min(W - margin, this.x));
        this.y = Math.max(margin, Math.min(H - margin, this.y));

        // Lap tracking
        const cx = trackCX(), cy = trackCY();
        const currentAngle = Math.atan2(this.y - cy, this.x - cx);
        let delta = currentAngle - this.prevAngle;
        if (delta > Math.PI) delta -= 2 * Math.PI;
        if (delta < -Math.PI) delta += 2 * Math.PI;
        this.accumulatedAngle += delta;
        this.prevAngle = currentAngle;

        if (Math.abs(this.accumulatedAngle) >= 2 * Math.PI * 0.95) {
            this.laps++;
            this.accumulatedAngle = 0;
            if (this.laps >= LAPS_TO_WIN) {
                this.finished = true;
            }
        }
    }

    draw() {
        ctx.save();
        ctx.translate(this.x, this.y);
        ctx.rotate(this.angle);

        ctx.fillRect(-16, -8, 4, 4);
        ctx.fillRect(-16, 4, 4, 4);

        ctx.restore();
    }
}

function isOnTrack(px, py) {
    const cx = trackCX(), cy = trackCY();
    const a = trackA(), b = trackB(), w = trackW();
    const aOuter = a + w / 2;
    const bOuter = b + w / 2;
    const aInner = Math.max(10, a - w / 2);
    const bInner = Math.max(10, b - w / 2);

    const outerVal = ((px - cx) / aOuter) ** 2 + ((py - cy) / bOuter) ** 2;
    const innerVal = ((px - cx) / aInner) ** 2 + ((py - cy) / bInner) ** 2;

    return outerVal < 1 && innerVal > 1;
}

    if (Math.floor(Date.now() / 500) % 2 === 0) {
        ctx.fillStyle = '#ffffff';
        ctx.font = 'bold 24px monospace';
        ctx.fillText('Press SPACE to Start', W / 2, H * 0.9);
    }

    ctx.textAlign = 'left';
}

function drawCountdown() {
    ctx.fillStyle = 'rgba(0,0,0,0.3)';
    ctx.fillRect(0, 0, W, H);

    const num = Math.ceil(countdown);
    if (num > 0) {
        ctx.fillStyle = '#ffffff';
        ctx.font = 'bold 120px monospace';
        ctx.textAlign = 'center';
        ctx.fillText(num.toString(), W / 2, H / 2 + 40);
    } else {
        ctx.fillStyle = '#00ff88';
        ctx.font = 'bold 80px monospace';
        ctx.textAlign = 'center';
        ctx.fillText('GO!', W / 2, H / 2 + 30);
    }
    ctx.textAlign = 'left';
}

function drawWinScreen() {
    ctx.fillStyle = 'rgba(0,0,0,0.7)';
    ctx.fillRect(0, 0, W, H);

    const winColor = winner === 1 ? '#e63946' : '#457b9d';
    ctx.fillStyle = winColor;
    ctx.font = 'bold 60px monospace';
    ctx.textAlign = 'center';
    ctx.fillText(`PLAYER ${winner} WINS!`, W / 2, H / 2 - 20);

    ctx.fillStyle = '#ffffff';
    ctx.font = '24px monospace';
    if (Math.floor(Date.now() / 500) % 2 === 0) {
        ctx.fillText('Press R to Race Again', W / 2, H / 2 + 50);
    }

    ctx.textAlign = 'left';
}

function drawParticles() {
    // Simple speed lines behind fast cars
    if (Math.abs(car1.speed) > 200) {
        for (let i = 0; i < 3; i++) {
            const ox = -Math.cos(car1.angle) * (20 + i * 8);
            const oy = -Math.sin(car1.angle) * (20 + i * 8);
            ctx.fillStyle = `rgba(230,57,70,${0.3 - i * 0.08})`;
            ctx.beginPath();
            ctx.arc(car1.x + ox, car1.y + oy, 3, 0, Math.PI * 2);
            ctx.fill();
        }
    }
    if (Math.abs(car2.speed) > 200) {
        for (let i = 0; i < 3; i++) {
            const ox = -Math.cos(car2.angle) * (20 + i * 8);
            const oy = -Math.sin(car2.angle) * (20 + i * 8);
            ctx.fillStyle = `rgba(69,123,157,${0.3 - i * 0.08})`;
            ctx.beginPath();
            ctx.arc(car2.x + ox, car2.y + oy, 3, 0, Math.PI * 2);
            ctx.fill();
        }
    }
}

function gameLoop(timestamp) {
    const dt = Math.min((timestamp - lastTime) / 1000, 0.05);
    lastTime = timestamp;

    if (gameState === 'start') {
        drawStartScreen();
        requestAnimationFrame(gameLoop);
        return;
    }

    if (gameState === 'countdown') {
        countdown -= dt;
        drawTrack();
        car1.draw();
        car2.draw();
        drawHUD();
        drawCountdown();
        if (countdown <= -0.5) {
            gameState = 'playing';
        }
        requestAnimationFrame(gameLoop);
        return;
    }

    if (gameState === 'finished') {
        drawTrack();
        car1.draw();
        car2.draw();
        drawHUD();
        drawWinScreen();
        requestAnimationFrame(gameLoop);
        return;
    }

    // Playing state
    car1.update(dt, keys['w'] || keys['W'], keys['s'] || keys['S'], keys['a'] || keys['A'], keys['d'] || keys['D']);
    car2.update(dt, keys['ArrowUp'], keys['ArrowDown'], keys['ArrowLeft'], keys['ArrowRight']);

    // Check win
    if (car1.finished && !winner) {
        winner = 1;
        gameState = 'finished';
    } else if (car2.finished && !winner) {
        winner = 2;
        gameState = 'finished';
    }

    // Draw
    drawTrack();
    car1.draw();
    car2.draw();
    drawParticles();
    drawHUD();

    requestAnimationFrame(gameLoop);
}

function startGame() {
    initGame();
    winner = null;
    countdown = 3;
    gameState = 'countdown';
}

// Audio - simple beeps on countdown
let audioCtx = null;
function playBeep(freq, duration) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.frequency.value = freq;
    gain.gain.value = 0.15;
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + duration);
    osc.start();
    osc.stop(audioCtx.currentTime + duration);
}

let lastCountNum = 4;

function checkCountdownAudio() {
    if (gameState === 'countdown') {
        const num = Math.ceil(countdown);
        if (num !== lastCountNum && num >= 1) {
            playBeep(440, 0.15);
            lastCountNum = num;
        }
        if (num <= 0 && lastCountNum > 0) {
            playBeep(880, 0.3);
            lastCountNum = 0;
        }
    }
}

// Input
window.addEventListener('keydown', (e) => {
    keys[e.key] = true;
    e.preventDefault();

    if (!audioCtx && (gameState === 'start' || gameState === 'countdown')) {
        audioCtx = new (window.AudioContext || window.webkitAudioContext)();
    }

    if (gameState === 'start' && (e.key === ' ' || e.key === 'Enter')) {
        startGame();
    } else if (gameState === 'finished' && (e.key === 'r' || e.key === 'R')) {
        startGame();
    }
});

window.addEventListener('keyup', (e) => {
    keys[e.key] = false;
});

// Prevent arrow keys from scrolling
window.addEventListener('keydown', (e) => {
    if (['ArrowUp','ArrowDown','ArrowLeft','ArrowRight',' '].includes(e.key)) {
        e.preventDefault();
    }
});

// Start
initGame();
requestAnimationFrame(gameLoop);
