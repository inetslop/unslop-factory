```html
<!DOCTYPE html>
<html lang="en">
<head>
<meta charset="UTF-8">
<meta name="viewport" content="width=device-width, initial-scale=1.0">
<title>Tower Racer</title>
<style>
* { margin: 0; padding: 0; box-sizing: border-box; }
html, body { width: 100%; height: 100%; overflow: hidden; background: #0a0a1a; }
canvas { display: block; width: 100vw; height: 100vh; }
</style>

<body>
<canvas id="game"></canvas>
<script>
const canvas = document.getElementById('game');
const ctx = canvas.getContext('2d');

let W, H;
function resize() {
    W = canvas.width = window.innerWidth;
    H = canvas.height = window.innerHeight;
}
window.addEventListener('resize', resize);
resize();

// Audio
let audioCtx = null;
function initAudio() {
    if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
}
function playSound(freq, dur, type, vol) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = type || 'square';
    osc.frequency.value = freq;
    gain.gain.setValueAtTime(vol || 0.1, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + dur);
}

// Game state
let gameState = 'start'; // start, playing, gameover
let score = 0;
let highScore = 0;
try { highScore = parseInt(localStorage.getItem('towerRacerHigh') || '0'); } catch(e) {}

let player = {};
let enemies = [];
let turrets = [];
let bullets = [];
let particles = [];
let laneMarkers = [];
let towerEnergy = 100;
let maxTowerEnergy = 100;
let energyRegen = 0.5;
let distance = 0;
let waveTimer = 0;
let waveNum = 0;
let screenShake = 0;
let gameTime = 0;

const keys = {};
window.addEventListener('keydown', e => {
    keys[e.code] = true;
    if (gameState === 'start' && (e.code === 'Space' || e.code === 'Enter')) startGame();
    else if (gameState === 'gameover' && (e.code === 'KeyR' || e.code === 'Enter')) startGame();
    if (e.code === 'Space' && gameState === 'playing') deployTurret();
});
window.addEventListener('keyup', e => { keys[e.code] = false; });

function getTrackBounds() {
    const trackW = Math.min(W * 0.6, 500);
    return { left: (W - trackW) / 2, right: (W + trackW) / 2, w: trackW };
}

function startGame() {
    initAudio();
    gameState = 'playing';
    score = 0;
    distance = 0;
    waveTimer = 0;
    waveNum = 0;
    towerEnergy = 100;
    screenShake = 0;
    gameTime = 0;
    enemies = [];
    turrets = [];
    bullets = [];
    particles = [];
    laneMarkers = [];
    const tb = getTrackBounds();
    player = { x: W / 2, y: H - 80, vx: 0, hp: 100, maxHp: 100, speed: 4 };
    for (let i = 0; i < 20; i++) {
        laneMarkers.push({ y: i * 60 });
    }
}

function deployTurret() {
    const tb = getTrackBounds();
    const cost = 30;
    if (towerEnergy < cost) return;
    towerEnergy -= cost;
    // Alternate sides based on which side has fewer turrets nearby
    let leftCount = 0, rightCount = 0;
    for (let t of turrets) {
        if (Math.abs(t.y - player.y) < 150) {
            if (t.side === 'left') leftCount++; else rightCount++;
        }
    }
    const side = leftCount <= rightCount ? 'left' : 'right';
    const tx = side === 'left' ? tb.left + 20 : tb.right - 20;
    turrets.push({ x: tx, y: player.y, side, hp: 5, shots: 15, fireTimer: 0 });
    playSound(440, 0.1, 'triangle', 0.15);
    spawnParticles(tx, player.y, '#0ff', 8);
}

function spawnParticles(x, y, color, count) {
    for (let i = 0; i < count; i++) {
        const a = Math.random() * Math.PI * 2;
        const s = 1 + Math.random() * 3;
        particles.push({ x, y, vx: Math.cos(a) * s, vy: Math.sin(a) * s, life: 30, color });
    }
}

function spawnEnemy() {
    const tb = getTrackBounds();
    const laneW = tb.w / 3;
    const lane = Math.floor(Math.random() * 3);
    const ex = tb.left + laneW * (lane + 0.5) + (Math.random() - 0.5) * laneW * 0.4;
    const speed = 1.5 + waveNum * 0.15 + Math.random();
    enemies.push({ x: ex, y: -30, vy: speed, hp: 2 + Math.floor(waveNum / 3), maxHp: 2 + Math.floor(waveNum / 3), size: 14 });
}

function update() {
    if (gameState !== 'playing') return;
    gameTime += 1 / 60;
    const tb = getTrackBounds();

    // Player movement
    let ax = 0;
    if (keys['ArrowLeft'] || keys['KeyA']) ax -= 1;
    if (keys['ArrowRight'] || keys['KeyD']) ax += 1;
    player.vx += ax * 0.8;
    player.vx *= 0.85;
    player.x += player.vx;
    player.x = Math.max(tb.left + 20, Math.min(tb.right - 20, player.x));

    // Distance & score
    distance += 1;
    score = Math.floor(distance / 10) + waveNum * 50;

    // Energy regen
    towerEnergy = Math.min(maxTowerEnergy, towerEnergy + energyRegen);

    // Lane markers scroll
    for (let m of laneMarkers) {
        m.y += player.speed;
        if (m.y > H + 30) m.y -= 20 * 60;
    }

    // Waves
    waveTimer++;
    const spawnInterval = Math.max(40, 120 - waveNum * 8);
    if (waveTimer >= spawnInterval) {
        waveTimer = 0;
        waveNum++;
        const count = 1 + Math.floor(waveNum / 5);
        for (let i = 0; i < count; i++) setTimeout(() => spawnEnemy(), i * 200);
    }

    // Enemies
    for (let i = enemies.length - 1; i >= 0; i--) {
        const e = enemies[i];
        e.y += e.vy;
        // Move toward player slightly
        e.x += (player.x - e.x) * 0.005;

        if (e.y > H + 50) { enemies.splice(i, 1); continue; }

        // Check collision with player
        const dx = e.x - player.x, dy = e.y - player.y;
        if (Math.abs(dx) < 25 && Math.abs(dy) < 25) {
            player.hp -= 20;
            screenShake = 10;
            playSound(100, 0.3, 'sawtooth', 0.2);
            spawnParticles(player.x, player.y, '#f44', 12);
            enemies.splice(i, 1);
            if (player.hp <= 0) { endGame(); return; }
        }
    }

    // Turrets
    for (let i = turrets.length - 1; i >= 0; i--) {
        const t = turrets[i];
        t.y += player.speed * 0.3;
        if (t.shots <= 0) { turrets.splice(i, 1); continue; }
        if (t.y > H + 50) { turrets.splice(i, 1); continue; }

        t.fireTimer++;
        if (t.fireTimer >= 25) {
            t.fireTimer = 0;
            // Find nearest enemy in range
            let target = null, bestDist = 300;
            for (let e of enemies) {
                const d = Math.hypot(e.x - t.x, e.y - t.y);
                if (d < bestDist && e.y > t.y - 50) { bestDist = d; target = e; }
            }
            if (target) {
                t.shots--;
                const a = Math.atan2(target.y - t.y, target.x - t.x);
                bullets.push({ x: t.x, y: t.y, vx: Math.cos(a) * 8, vy: Math.sin(a) * 8, life: 40 });
                playSound(800, 0.05, 'square', 0.05);
            }
        }
    }

    // Bullets
    for (let i = bullets.length - 1; i >= 0; i--) {
        const b = bullets[i];
        b.x += b.vx;
        b.y += b.vy;
        b.life--;
        if (b.life <= 0) { bullets.splice(i, 1); continue; }

        for (let j = enemies.length - 1; j >= 0; j--) {
            const e = enemies[j];
            if (Math.abs(b.x - e.x) < e.size && Math.abs(b.y - e.y) < e.size) {
                e.hp--;
                spawnParticles(e.x, e.y, '#ff0', 4);
                playSound(600, 0.05, 'square', 0.08);
                if (e.hp <= 0) {
                    score += 10;
                    spawnParticles(e.x, e.y, '#f80', 10);
                    enemies.splice(j, 1);
                }
                bullets.splice(i, 1);
                break;
            }
        }
    }

    // Particles
    for (let i = particles.length - 1; i >= 0; i--) {
        const p = particles[i];
        p.x += p.vx;
        p.y += p.vy;
        p.life--;
        if (p.life <= 0) particles.splice(i, 1);
    }

    // Screen shake decay
    if (screenShake > 0) screenShake *= 0.85;
}

function endGame() {
    gameState = 'gameover';
    try {
        if (score > highScore) {
            highScore = score;
            localStorage.setItem('towerRacerHigh', highScore);
        }
    } catch(e) {}
}

function draw() {
    ctx.save();
    if (screenShake > 0.5) {
        ctx.translate((Math.random() - 0.5) * screenShake, (Math.random() - 0.5) * screenShake);
    }

    // Background
    ctx.fillStyle = '#0a0a1a';
    ctx.fillRect(-20, -20, W + 40, H + 40);

    const tb = getTrackBounds();

    if (gameState === 'start') {
        drawStartScreen();
        ctx.restore();
        return;
    }

    // Track
    ctx.fillStyle = '#1a1a2e';
    ctx.fillRect(tb.left, 0, tb.w, H);

    // Track borders
    ctx.strokeStyle = '#0ff';
    ctx.lineWidth = 3;
    ctx.shadowColor = '#0ff';
    ctx.shadowBlur = 10;
    ctx.beginPath();
    ctx.moveTo(tb.left, 0); ctx.lineTo(tb.left, H);
    ctx.stroke();
    ctx.beginPath();
    ctx.moveTo(tb.right, 0); ctx.lineTo(tb.right, H);
    ctx.stroke();

    // Lane markers
    ctx.shadowBlur = 0;
    ctx.strokeStyle = 'rgba(255,255,255,0.3)';
    ctx.lineWidth = 2;
    const laneW = tb.w / 3;
    for (let l = 1; l < 3; l++) {
        const lx = tb.left + laneW * l;
        for (let m of laneMarkers) {
            ctx.beginPath();
            ctx.moveTo(lx, m.y);
            ctx.lineTo(lx, m.y + 25);
            ctx.stroke();
        }
    }

    // Turrets
    for (let t of turrets) {
        ctx.fillStyle = t.side === 'left' ? '#0ff' : '#f0f';
        ctx.shadowColor = t.side === 'left' ? '#0ff' : '#f0f';
        ctx.shadowBlur = 8;
        ctx.beginPath();
        ctx.arc(t.x, t.y, 8, 0, Math.PI * 2);
        ctx.fill();
        // Ammo indicator
        ctx.fillStyle = '#fff';
        ctx.font = '10px monospace';
        ctx.textAlign = 'center';
        ctx.fillText(Math.ceil(t.shots / 5), t.x, t.y + 3);
    }

    // Enemies
    for (let e of enemies) {
        const pulse = 1 + Math.sin(gameTime * 8) * 0.1;
        ctx.fillStyle = '#f44';
        ctx.shadowColor = '#f44';
        ctx.shadowBlur = 12;
        ctx.beginPath();
        ctx.arc(e.x, e.y, e.size * pulse, 0, Math.PI * 2);
        ctx.fill();
        // HP bar
        if (e.hp < e.maxHp) {
            ctx.fillStyle = '#400';
            ctx.fillRect(e.x - 12, e.y - 22, 24, 4);
            ctx.fillStyle = '#f44';
            ctx.fillRect(e.x - 12, e.y - 22, 24 * (e.hp / e.maxHp), 4);
        }
    }

    // Bullets
    ctx.shadowBlur = 0;
    for (let b of bullets) {
        ctx.fillStyle = '#ff0';
        ctx.shadowColor = '#ff0';
        ctx.shadowBlur = 6;
        ctx.beginPath();
        ctx.arc(b.x, b.y, 4, 0, Math.PI * 2);
        ctx.fill();
    }

    // Player car
    ctx.shadowColor = '#0f0';
    ctx.shadowBlur = 15;
    ctx.fillStyle = '#0f0';
    ctx.beginPath();
    ctx.moveTo(player.x, player.y - 20);
    ctx.lineTo(player.x - 14, player.y + 16);
    ctx.lineTo(player.x + 14, player.y + 16);
    ctx.closePath();
    ctx.fill();
    // Cockpit
    ctx.fillStyle = '#ff0';
    ctx.shadowBlur = 5;
    ctx.beginPath();
    ctx.arc(player.x, player.y, 5, 0, Math.PI * 2);
    ctx.fill();

    // Particles
    for (let p of particles) {
        ctx.fillStyle = p.color;
        ctx.globalAlpha = p.life / 30;
        ctx.beginPath();
        ctx.arc(p.x, p.y, 3, 0, Math.PI * 2);
        ctx.fill();
    }
    ctx.globalAlpha = 1;

    // HUD
    ctx.shadowBlur = 0;
    ctx.fillStyle = '#fff';
    ctx.font = 'bold 18px monospace';
    ctx.textAlign = 'left';
    ctx.fillText(`SCORE: ${score}`, 20, 30);
    ctx.fillText(`WAVE: ${waveNum}`, 20, 55);
    ctx.fillText(`HP: ${Math.max(0, player.hp)}`, 20, 80);

    // Energy bar
    ctx.fillStyle = '#111';
    ctx.fillRect(W - 220, 20, 200, 24);
    const eRatio = towerEnergy / maxTowerEnergy;
    ctx.fillStyle = eRatio > 0.3 ? '#0ff' : '#f80';
    ctx.fillRect(W - 218, 22, 196 * eRatio, 20);
    ctx.fillStyle = '#fff';
    ctx.font = '12px monospace';
    ctx.textAlign = 'center';
    ctx.fillText('TOWER ENERGY [SPACE]', W - 120, 35);

    // HP bar
    ctx.fillStyle = '#111';
    ctx.fillRect(20, 92, 150, 16);
    ctx.fillStyle = player.hp > 30 ? '#0f0' : '#f44';
    ctx.fillRect(22, 94, 148 * (player.hp / player.maxHp), 12);

    // Game over
    if (gameState === 'gameover') {
        drawGameOver();
    }

    ctx.restore();
}

function drawStartScreen() {
    ctx.fillStyle = '#0ff';
    ctx.font = 'bold 48px monospace';
    ctx.textAlign = 'center';
    ctx.shadowColor = '#0ff';
    ctx.shadowBlur = 20;
    ctx.fillText('TOWER RACER', W / 2, H / 3);

    ctx.shadowBlur = 0;
    ctx.fillStyle = '#fff';
    ctx.font = '18px monospace';
    const lines = [
        'Steer: ← → or A D',
        'Deploy Turret: SPACE (costs energy)',
        '',
        'Turrets auto-fire at enemies ahead.',
        'Don\'t let enemies reach your car!',
        '',
        `HIGH SCORE: ${highScore}`,
        '',
        '[SPACE] to Start'
    ];
    lines.forEach((line, i) => {
        ctx.fillText(line, W / 2, H / 3 + 60 + i * 28);
    });
}

function drawGameOver() {
    ctx.fillStyle = 'rgba(0,0,0,0.7)';
    ctx.fillRect(0, 0, W, H);
    ctx.fillStyle = '#f44';
    ctx.font = 'bold 42px monospace';
    ctx.textAlign = 'center';
    ctx.fillText('GAME OVER', W / 2, H / 3);
    ctx.fillStyle = '#fff';
    ctx.font = '22px monospace';
    ctx.fillText(`Score: ${score}`, W / 2, H / 3 + 50);
    ctx.fillText(`High Score: ${highScore}`, W / 2, H / 3 + 85);
    ctx.fillStyle = '#0ff';
    ctx.font = '18px monospace';
    ctx.fillText('[R] or [ENTER] to Restart', W / 2, H / 3 + 140);
}

let lastTime = 0;
function loop(time) {
    requestAnimationFrame(loop);
    update();
    draw();
}
requestAnimationFrame(loop);
</script>
</body>
```
