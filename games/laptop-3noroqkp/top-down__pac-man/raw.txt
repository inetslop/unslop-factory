function isWall(x, y) {
    if (x < 0 || x >= COLS) return false; // tunnel
    if (y < 0 || y >= ROWS) return true;
    return gameGrid[y][x] === 1;
}

function canMove(x, y, dir) {
    const nx = x + (dir === 'left' ? -1 : dir === 'right' ? 1 : 0);
    const ny = y + (dir === 'up' ? -1 : dir === 'down' ? 1 : 0);
    if (nx < 0 || nx >= COLS) return true; // tunnel wrap
    if (ny < 0 || ny >= ROWS) return false;
    return !isWall(nx, ny);
}

function moveEntity(ent, speed) {
    ent.progress += speed;
    while (ent.progress >= 1) {
        ent.progress -= 1;
        ent.x = ent.tx; ent.y = ent.ty;
        if (ent.x < 0) ent.x = COLS - 1;
        if (ent.x >= COLS) ent.x = 0;
    }
    // Interpolated position
    const dx = ent.tx - ent.x, dy = ent.ty - ent.y;
    ent.px = ent.x + dx * ent.progress;
    ent.py = ent.y + dy * ent.progress;
}

function stepEntity(ent, speed) {
    if (ent.progress < 1) {
        moveEntity(ent, speed);
        return;
    }
    // At a grid intersection, decide next move
    let moved = false;
    const dirs = ['up', 'down', 'left', 'right'];
    
    if (ent === pac) {
        if (canMove(ent.x, ent.y, ent.nextDir)) {
            ent.dir = ent.nextDir;
        }
        if (canMove(ent.x, ent.y, ent.dir)) {
            ent.tx = ent.x + (ent.dir === 'left' ? -1 : ent.dir === 'right' ? 1 : 0);
            ent.ty = ent.y + (ent.dir === 'up' ? -1 : ent.dir === 'down' ? 1 : 0);
            if (ent.tx < 0) ent.tx = COLS - 1;
            if (ent.tx >= COLS) ent.tx = 0;
            moved = true;
        }
    } else {
        const nextDirs = [];
        for (const d of dirs) {
            // Don't reverse
            const rev = d === 'up' ? 'down' : d === 'down' ? 'up' : d === 'left' ? 'right' : 'left';
            if (d !== rev && canMove(ent.x, ent.y, d)) nextDirs.push(d);
        }
        if (nextDirs.length > 0) {
            let chosenDir;
            if (ent.eaten) {
                // Go to ghost house
                chosenDir = choosePathToHouse(ent);
            } else if (ghostFrightened && !ent.inHouse) {
                // Flee randomly
                chosenDir = nextDirs[Math.floor(Math.random() * nextDirs.length)];
            } else {
                chosenDir = chooseChaseDir(ent, nextDirs);
            }
            ent.dir = chosenDir;
        } else if (nextDirs.length === 0) {
            // Must reverse
            const rev = ent.dir === 'up' ? 'down' : ent.dir === 'down' ? 'up' : ent.dir === 'left' ? 'right' : 'left';
            ent.dir = rev;
        }
        if (canMove(ent.x, ent.y, ent.dir)) {
            ent.tx = ent.x + (ent.dir === 'left' ? -1 : ent.dir === 'right' ? 1 : 0);
            ent.ty = ent.y + (ent.dir === 'up' ? -1 : ent.dir === 'down' ? 1 : 0);
            if (ent.tx < 0) ent.tx = COLS - 1;
            if (ent.tx >= COLS) ent.tx = 0;
            moved = true;
        }
    }
    
    if (moved) {
        moveEntity(ent, speed);
        // If fully moved, snap to target
        if (ent.progress >= 1) {
            ent.x = ent.tx; ent.y = ent.ty;
            if (ent.x < 0) ent.x = COLS - 1;
            if (ent.x >= COLS) ent.x = 0;
            ent.px = ent.x; ent.py = ent.y;
            ent.progress = 0;
        } else {
            const dx2 = ent.tx - ent.x, dy2 = ent.ty - ent.y;
            ent.px = ent.x + dx2 * ent.progress;
            ent.py = ent.y + dy2 * ent.progress;
        }
    }
}

function chooseChaseDir(ghost, nextDirs) {
    if (nextDirs.length === 0) return ghost.dir;
    
    let targetX, targetY;
    switch (ghost.type) {
        case 0: // Blinky - chase directly
            targetX = pac.x; targetY = pac.y;
            break;
        case 1: // Pinky - aim ahead of Pac-Man
            const pdx = pac.dir === 'left' ? -2 : pac.dir === 'right' ? 2 : 0;
            const pdy = pac.dir === 'up' ? -2 : pac.dir === 'down' ? 2 : 0;
            targetX = pac.x + pdx; targetY = pac.y + pdy;
            break;
        case 2: // Inky - random-ish with some chase
            if (Math.random() < 0.3) {
                targetX = Math.floor(Math.random() * COLS);
                targetY = Math.floor(Math.random() * ROWS);
            } else {
                targetX = pac.x; targetY = pac.y;
            }
            break;
        case 3: // Clyde - slow, sometimes random
            if (Math.random() < 0.2) {
                return nextDirs[Math.floor(Math.random() * nextDirs.length)];
            }
            targetX = pac.x; targetY = pac.y;
            break;
    }
    
    let bestDir = nextDirs[0], bestDist = Infinity;
    for (const d of nextDirs) {
        const nx = ghost.x + (d === 'left' ? -1 : d === 'right' ? 1 : 0);
        const ny = ghost.y + (d === 'up' ? -1 : d === 'down' ? 1 : 0);
        const dist = Math.abs(nx - targetX) + Math.abs(ny - targetY);
        if (dist < bestDist) { bestDist = dist; bestDir = d; }
    }
    return bestDir;
}

function choosePathToHouse(ghost) {
    // Navigate to ghost house entrance (9, 9)
    const dirs = ['up', 'down', 'left', 'right'];
    let bestDir = 'down', bestDist = Infinity;
    for (const d of dirs) {
        if (!canMove(ghost.x, ghost.y, d)) continue;
        const nx = ghost.x + (d === 'left' ? -1 : d === 'right' ? 1 : 0);
        const ny = ghost.y + (d === 'up' ? -1 : d === 'down' ? 1 : 0);
        const dist = Math.abs(nx - 9) + Math.abs(ny - 9);
        if (dist < bestDist) { bestDist = dist; bestDir = d; }
    }
    return bestDir;
}

function checkCollision() {
    for (const g of ghosts) {
        if (g.eaten) continue;
        const dx = Math.abs(pac.px - g.px), dy = Math.abs(pac.py - g.py);
        if (dx < 0.6 && dy < 0.6) {
            if (ghostFrightened) {
                g.eaten = true;
                g.eyeMode = false;
                score += 200 * level;
            } else {
                return 'death';
            }
        }
    }
    return null;
}

function eatPellet() {
    const gx = Math.floor(pac.px), gy = Math.floor(pac.py);
    if (gx < 0 || gx >= COLS || gy < 0 || gy >= ROWS) return;
    if (gameGrid[gy][gx] === 0) {
        gameGrid[gy][gx] = -1; // eaten
        score += 10 * level;
        pelletsRemaining--;
        checkWin();
    } else if (gameGrid[gy][gx] === 2) {
        gameGrid[gy][gx] = -1;
        score += 50 * level;
        pelletsRemaining--;
        powerTimer = 360; // ~6 seconds at 60fps
        ghostFrightened = true;
        checkWin();
    }
}

function update() {
    if (gameState !== 'playing' && gameState !== 'dead') return;
    
    if (gameState === 'dead') {
        deathAnim++;
        if (deathAnim > 90) {
            lives--;
            if (lives <= 0) {
                gameState = 'gameOver';
            } else {
                resetPositions();
                gameState = 'playing';
            }
        }
        return;
    }
    
    animFrame++;
    
    if (powerTimer > 0) {
        powerTimer--;
        if (powerTimer <= 0) ghostFrightened = false;
    }
    
    // Move Pac-Man
    stepEntity(pac, 0.15);
    
    // Eat pellet
    eatPellet();
    
    // Move ghosts
    for (const g of ghosts) {
        if (g.inHouse) {
            g.houseTimer--;
            if (g.houseTimer <= 0) {
                g.inHouse = false;
                g.x = 9; g.y = 9; g.px = 9; g.py = 9;
                g.tx = 9; g.ty = 9; g.progress = 0;
                g.dir = Math.random() < 0.5 ? 'left' : 'right';
            }
            continue;
        }
        
        if (g.eaten) {
            // Navigate back to house
            stepEntity(g, 1.2);
            if (g.x === 9 && g.y === 9 || g.x === 7 && g.y === 9 || g.x === 11 && g.y === 9) {
                if (Math.abs(g.px - 9) < 0.5 && Math.abs(g.py - 9) < 0.5) {
                    g.eaten = false;
                    g.inHouse = true;
                    g.houseTimer = 30 + Math.random() * 30;
                }
            }
            continue;
        }
        
        const speed = ghostFrightened ? 0.1 : (0.12 * (1 + level * 0.05));
        stepEntity(g, Math.min(speed, 0.14));
    }
    
    // Check collision
    const result = checkCollision();
    if (result === 'death') {
        gameState = 'dead';
        deathAnim = 0;
    }
}

// Rendering
function drawMaze() {
    for (let r = 0; r < ROWS; r++) {
        for (let c = 0; c < COLS; c++) {
            const x = gridOffsetX + c * cellSize;
            const y = gridOffsetY + r * cellSize;
            
            if (gameGrid[r][c] === 1) {
                ctx.fillStyle = '#2121DE';
                ctx.fillRect(x, y, cellSize, cellSize);
                // Inner glow
                ctx.fillStyle = '#3838FF';
                const pad = Math.floor(cellSize * 0.15);
                ctx.fillRect(x + pad, y + pad, cellSize - pad * 2, cellSize - pad * 2);
            }
            
            if (gameGrid[r][c] === 0) {
                // Pellet
                ctx.fillStyle = '#FFB8AE';
                const pr = cellSize * 0.12;
                ctx.beginPath();
                ctx.arc(x + cellSize / 2, y + cellSize / 2, pr, 0, Math.PI * 2);
                ctx.fill();
            } else if (gameGrid[r][c] === 2) {
                // Power pellet - pulse
                const pulse = 1 + 0.3 * Math.sin(animFrame * 0.15);
                ctx.fillStyle = '#FFB8AE';
                ctx.beginPath();
                ctx.arc(x + cellSize / 2, y + cellSize / 2, cellSize * 0.25 * pulse, 0, Math.PI * 2);
                ctx.fill();
            }
            
            if (gameGrid[r][c] === 3) {
                // Ghost house - dark
                ctx.fillStyle = '#111';
                ctx.fillRect(x, y, cellSize, cellSize);
            }
        }
    }
}

function drawPac() {
    const x = gridOffsetX + pac.px * cellSize;
    const y = gridOffsetY + pac.py * cellSize;
    const r = cellSize * 0.4;
    
    if (gameState === 'dead') {
        // Death animation - expanding mouth
        const openAngle = Math.min(deathAnim / 30, Math.PI);
        ctx.fillStyle = '#FFE55C';
        ctx.beginPath();
        ctx.arc(x + cellSize/2, y + cellSize/2, r, openAngle, Math.PI * 2 - openAngle);
        ctx.lineTo(x + cellSize/2, y + cellSize/2);
        ctx.fill();
        return;
    }
    
    let mouthAngle = 0.3;
    const mouthOpen = (Math.sin(animFrame * 0.3) + 1) * 0.5 * mouthAngle;
    
    let startA, endA;
    switch (pac.dir) {
        case 'up': startA = -Math.PI/2 + mouthOpen; endA = -Math.PI/2 - mouthOpen + Math.PI; break;
        case 'down': startA = Math.PI/2 + mouthOpen; endA = Math.PI/2 - mouthOpen + Math.PI; break;
        case 'left': startA = Math.PI + mouthOpen; endA = Math.PI - mouthOpen + Math.PI; break;
        case 'right': startA = -mouthOpen; endA = mouthOpen + Math.PI; break;
    }
    
    ctx.fillStyle = '#FFE55C';
    ctx.beginPath();
    ctx.moveTo(x + cellSize/2, y + cellSize/2);
    ctx.arc(x + cellSize/2, y + cellSize/2, r, startA - Math.PI, endA - Math.PI);
    ctx.closePath();
    ctx.fill();
}

function drawGhost(g) {
    const x = gridOffsetX + g.px * cellSize;
    const y = gridOffsetY + g.py * cellSize;
    const r = cellSize * 0.4;
    
    let color;
    if (g.eaten && !g.inHouse) {
        // Just eyes when eaten
        ctx.fillStyle = '#fff';
        drawEyes(x, y, r);
        return;
    } else if (ghostFrightened) {
        const flash = Math.floor(powerTimer / 20) % 2 === 0;
        color = flash ? '#2121DE' : '#3838FF';
    } else {
        const colors = ['#FF0000', '#FFB8AE', '#00FFFF', '#FFB847'];
        color = colors[g.type];
    }
    
    // Body
    ctx.fillStyle = color;
    ctx.beginPath();
    ctx.arc(x + cellSize/2, y + cellSize/2 - r*0.1, r, Math.PI, 0);
    // Wavy bottom
    const waveY = y + cellSize/2 + r * 0.6;
    ctx.lineTo(x + cellSize/2 + r, waveY);
    const waveH = r * 0.3;
    for (let i = 0; i < 4; i++) {
        const wx = x + cellSize/2 + r - (i + 1) * (r * 2 / 4);
        const wy = waveY - (i % 2 === 0 ? 0 : waveH);
        ctx.lineTo(wx, wy);
    }
    ctx.closePath();
    ctx.fill();
    
    // Eyes
    if (!ghostFrightened) {
        ctx.fillStyle = '#fff';
        drawEyes(x, y, r);
    } else {
        // Frightened face
        ctx.fillStyle = '#FF0000';
        const ex = x + cellSize/2;
        const ey = y + cellSize/2 - r*0.1;
        const er = r * 0.2;
        ctx.beginPath(); ctx.arc(ex - r*0.3, ey, er, 0, Math.PI*2); ctx.fill();
        ctx.beginPath(); ctx.arc(ex + r*0.3, ey, er, 0, Math.PI*2); ctx.fill();
    }
}

function drawEyes(x, y, r) {
    const ex = x + cellSize/2;
    const ey = y + cellSize/2 - r * 0.15;
    const er = r * 0.25;
    
    ctx.fillStyle = '#fff';
    ctx.beginPath(); ctx.arc(ex - r*0.3, ey, er, 0, Math.PI*2); ctx.fill();
    ctx.beginPath(); ctx.arc(ex + r*0.3, ey, er, 0, Math.PI*2); ctx.fill();
    
    // Pupils based on direction
    const px = (pac.px - 0) < 0 ? -1 : 1;
    const py = (pac.py - 0) < 0 ? -1 : 1;
    ctx.fillStyle = '#00F';
    ctx.beginPath(); ctx.arc(ex - r*0.3 + px * er*0.4, ey + py * er*0.4, er*0.5, 0, Math.PI*2); ctx.fill();
    ctx.beginPath(); ctx.arc(ex + r*0.3 + px * er*0.4, ey + py * er*0.4, er*0.5, 0, Math.PI*2); ctx.fill();
}

// Audio (simple beeps on first input)
let audioCtx = null;
function initAudio() {
    if (!audioCtx) {
        try { audioCtx = new (window.AudioContext || window.webkitAudioContext)(); } catch(e) {}
    }
}

function playBeep(freq, duration, type) {
    if (!audioCtx) return;
    try {
        const osc = audioCtx.createOscillator();
        const gain = audioCtx.createGain();
        osc.type = type || 'square';
        osc.frequency.value = freq;
        gain.gain.setValueAtTime(0.1, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + duration);
        osc.connect(gain);
        gain.connect(audioCtx.destination);
        osc.start();
        osc.stop(audioCtx.currentTime + duration);
    } catch(e) {}
}

let lastKeySound = 0;
window.addEventListener('keydown', e => {
    initAudio();
    const now = Date.now();
    if (now - lastKeySound > 100) {
        playBeep(440, 0.05);
        lastKeySound = now;
    }
});
