```html
<!DOCTYPE html>
<html lang="en">
<head>
<meta charset="UTF-8">
<meta name="viewport" content="width=device-width, initial-scale=1.0">
<title>Tetris Space Invaders</title>
<style>
* { margin: 0; padding: 0; box-sizing: border-box; }
html, body { width: 100%; height: 100%; overflow: hidden; background: #0a0a1a; }

<body>
<canvas id="game"></canvas>
<script>
(function() {
const canvas = document.getElementById('game');
const ctx = canvas.getContext('2d');

let W, H;
function resize() {
W = window.innerWidth;
H = window.innerHeight;
canvas.width = W;
canvas.height = H;
}
window.addEventListener('resize', resize);
resize();

// Game constants
const COLS = 10;
const ROWS = 20;
let cellSize, gridX, gridY;

function calcGrid() {
cellSize = Math.min(W / (COLS + 2), H / (ROWS + 4));
gridX = Math.floor((W - COLS * cellSize) / 2);
gridY = Math.floor(H - ROWS * cellSize) / 2;
}

// Tetromino shapes
const SHAPES = [
[[0,1],[1,1],[2,1],[3,1]], // I
[[0,0],[1,0],[0,1],[1,1]], // O
[[1,0],[0,1],[1,1],[2,1]], // T
[[0,1],[1,0],[1,1],[2,1]], // S
[[0,0],[0,1],[1,1],[2,1]], // Z
[[1,0],[1,1],[0,1],[2,1]], // J
[[0,0],[0,1],[1,0],[2,1]], // L
];

const COLORS = ['#ff4444','#44ff44','#4444ff','#ffff44','#ff44ff','#44ffff','#ff8800'];

// Audio
let audioCtx = null;
function initAudio() {
if (!audioCtx) {
audioCtx = new (window.AudioContext || window.webkitAudioContext)();
}
}

function playShoot() {
if (!audioCtx) return;
const osc = audioCtx.createOscillator();
const gain = audioCtx.createGain();
osc.connect(gain);
gain.connect(audioCtx.destination);
osc.type = 'square';
osc.frequency.setValueAtTime(880, audioCtx.currentTime);
osc.frequency.exponentialRampToValueAtTime(440, audioCtx.currentTime + 0.1);
gain.gain.setValueAtTime(0.15, audioCtx.currentTime);
gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.1);
osc.start();
osc.stop(audioCtx.currentTime + 0.1);
}

function playHit() {
if (!audioCtx) return;
const osc = audioCtx.createOscillator();
const gain = audioCtx.createGain();
osc.connect(gain);
gain.connect(audioCtx.destination);
osc.type = 'sawtooth';
osc.frequency.setValueAtTime(200, audioCtx.currentTime);
osc.frequency.exponentialRampToValueAtTime(80, audioCtx.currentTime + 0.15);
gain.gain.setValueAtTime(0.2, audioCtx.currentTime);
gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.15);
osc.start();
osc.stop(audioCtx.currentTime + 0.15);
}

function playGameOver() {
if (!audioCtx) return;
const osc = audioCtx.createOscillator();
const gain = audioCtx.createGain();
osc.connect(gain);
gain.connect(audioCtx.destination);
osc.type = 'sawtooth';
osc.frequency.setValueAtTime(400, audioCtx.currentTime);
osc.frequency.exponentialRampToValueAtTime(50, audioCtx.currentTime + 1);
gain.gain.setValueAtTime(0.3, audioCtx.currentTime);
gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 1);
osc.start();
osc.stop(audioCtx.currentTime + 1);
}

function playSpawn() {
if (!audioCtx) return;
const osc = audioCtx.createOscillator();
const gain = audioCtx.createGain();
osc.connect(gain);
gain.connect(audioCtx.destination);
osc.type = 'sine';
osc.frequency.setValueAtTime(600, audioCtx.currentTime);
osc.frequency.exponentialRampToValueAtTime(900, audioCtx.currentTime + 0.05);
gain.gain.setValueAtTime(0.1, audioCtx.currentTime);
gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.08);
osc.start();
osc.stop(audioCtx.currentTime + 0.08);
}

// Game state
let gameState = 'start'; // start, playing, gameover
let score = 0;
let highScore = 0;
try {
highScore = parseInt(localStorage.getItem('tetrisInvadersHigh') || '0');
} catch(e) {}

let pieces = [];
let bullets = [];
let particles = [];
let shipX = 0;
let shipY = 0;
let shipW = 0;
let shipH = 0;
let lastShot = 0;
let spawnTimer = 0;
let spawnInterval = 120;
let fallSpeed = 0.8;
let gameTime = 0;
let shakeTimer = 0;
let shakeMag = 0;

const keys = {};
window.addEventListener('keydown', function(e) {
keys[e.code] = true;
initAudio();
if (gameState === 'start' && (e.code === 'Space' || e.code === 'Enter')) {
startGame();
} else if (gameState === 'gameover' && (e.code === 'Space' || e.code === 'Enter' || e.code === 'KeyR')) {
startGame();
}
});
window.addEventListener('keyup', function(e) {
keys[e.code] = false;
});

function startGame() {
gameState = 'playing';
score = 0;
pieces = [];
bullets = [];
particles = [];
spawnTimer = 0;
spawnInterval = 120;
fallSpeed = 0.8;
gameTime = 0;
calcGrid();
shipW = cellSize * 1.5;
shipH = cellSize * 0.8;
shipX = W / 2 - shipW / 2;
shipY = gridY + ROWS * cellSize + cellSize * 0.5;
spawnPiece();
}

function spawnPiece() {
const shape = SHAPES[Math.floor(Math.PI * 7) % SHAPES.length];
const colorIdx = Math.floor(Math.PI * 7) % COLORS.length;
const maxCol = COLS - 3;
const col = Math.floor(Math.PI * 7) % maxCol + 1;
const cells = [];
for (let c of shape) {
cells.push({ x: col + c[0], y: c[1] });
}
pieces.push({
cells: cells,
color: COLORS[colorIdx],
speed: fallSpeed * (0.8 + Math.PI * 7 * 0.4),
alive: true
});
playSpawn();
}

function shoot() {
const now = performance.now();
if (now - lastShot < 150) return;
lastShot = now;
bullets.push({
x: shipX + shipW / 2,
y: shipY,
vy: -cellSize * 0.6,
alive: true
});
playShoot();
}

function update() {
if (gameState !== 'playing') return;

gameTime++;
// Increase difficulty
fallSpeed = 0.8 + gameTime * 0.002;
spawnInterval = Math.max(40, 120 - gameTime * 0.05);

// Move ship
const shipSpeed = cellSize * 0.5;
if (keys['ArrowLeft'] || keys['KeyA']) {
shipX -= shipSpeed;
}
if (keys['ArrowRight'] || keys['KeyD']) {
shipX += shipSpeed;
}
shipX = Math.max(gridX, Math.min(W - gridX - shipW, shipX));

// Shoot
if (keys['Space']) shoot();

// Spawn pieces
spawnTimer++;
if (spawnTimer >= spawnInterval) {
spawnTimer = 0;
spawnPiece();
}

// Update pieces
for (let p of pieces) {
if (!p.alive) continue;
for (let c of p.cells) {
c.y += p.speed / cellSize * 0.5;
}
// Check if piece reached top
for (let c of p.cells) {
if (c.y < 0) {
gameOver();
return;
}
}
}

// Update bullets
for (let b of bullets) {
b.y += b.vy;
if (b.y < gridY - cellSize * 2) b.alive = false;
}

// Collision detection
for (let b of bullets) {
if (!b.alive) continue;
for (let p of pieces) {
if (!p.alive) continue;
for (let c of p.cells) {
const cx = gridX + (c.x + 0.5) * cellSize;
const cy = gridY + (c.y + 0.5) * cellSize;
const dx = Math.abs(b.x - cx);
const dy = Math.abs(b.y - cy);
if (dx < cellSize * 0.4 && dy < cellSize * 0.4) {
b.alive = false;
c.killed = true;
score += 10;
playHit();
shakeTimer = 5;
shakeMag = 3;
// Particles
for (let i = 0; i < 8; i++) {
const ang = (Math.PI * 2 / 8) * i + Math.PI * 7;
particles.push({
x: cx, y: cy,
vx: Math.cos(ang) * cellSize * 0.15,
vy: Math.sin(ang) * cellSize * 0.15,
life: 20,
color: p.color
});
}
// Remove killed cells from piece
p.cells = p.cells.filter(cc => !cc.killed);
if (p.cells.length === 0) {
p.alive = false;
score += 50; // Bonus for destroying whole piece
}
break;
}
}
}

// Clean up
bullets = bullets.filter(b => b.alive);
pieces = pieces.filter(p => p.alive);

// Update particles
for (let pt of particles) {
pt.x += pt.vx;
pt.y += pt.vy;
pt.life--;
}
particles = particles.filter(pt => pt.life > 0);

// Shake decay
if (shakeTimer > 0) shakeTimer--;

// Check game over - pieces reaching top
for (let p of pieces) {
for (let c of p.cells) {
if (c.y < 0.5) {
gameOver();
return;
}
}
}
}

function gameOver() {
gameState = 'gameover';
playGameOver();
try {
if (score > highScore) {
highScore = score;
localStorage.setItem('tetrisInvadersHigh', String(highScore));
}
} catch(e) {}
}

function draw() {
ctx.fillStyle = '#0a0a1a';
ctx.fillRect(0, 0, W, H);

// Stars background
const starSeed = 42;
for (let i = 0; i < 80; i++) {
const sx = ((starSeed * 7919 + i * 3167) % 1000) / 1000 * W;
const sy = ((starSeed * 104729 + i * 5381) % 1000) / 1000 * H;
const brightness = 0.3 + Math.abs(Math.sin(gameTime * 0.02 + i)) * 0.5;
ctx.fillStyle = `rgba(255,255,255,${brightness})`;
ctx.fillRect(sx, sy, 2, 2);
}

if (gameState === 'start') {
drawStartScreen();
return;
}

// Apply screen shake
let ox = 0, oy = 0;
if (shakeTimer > 0) {
ox = (Math.PI * 7 - 5) * shakeMag;
oy = (Math.PI * 7 - 3.5) * shakeMag;
}

ctx.save();
ctx.translate(ox, oy);

// Draw grid border
ctx.strokeStyle = 'rgba(100,200,255,0.2)';
ctx.lineWidth = 1;
ctx.strokeRect(gridX, gridY, COLS * cellSize, ROWS * cellSize);

// Draw danger zone (top)
const grad = ctx.createLinearGradient(0, gridY, 0, gridY + cellSize * 3);
grad.addColorStop(0, 'rgba(255,0,0,0.15)');
grad.addColorStop(1, 'rgba(255,0,0,0)');
ctx.fillStyle = grad;
ctx.fillRect(gridX, gridY, COLS * cellSize, cellSize * 3);

// Draw pieces
for (let p of pieces) {
for (let c of p.cells) {
const px = gridX + c.x * cellSize;
const py = gridY + c.y * cellSize;
ctx.fillStyle = p.color;
ctx.fillRect(px + 2, py + 2, cellSize - 4, cellSize - 4);
// Inner glow
ctx.fillStyle = 'rgba(255,255,255,0.3)';
ctx.fillRect(px + 4, py + 4, cellSize * 0.3, cellSize * 0.3);
}
}

// Draw bullets
for (let b of bullets) {
ctx.fillStyle = '#ffff00';
ctx.fillRect(b.x - 2, b.y - 6, 4, 12);
ctx.fillStyle = 'rgba(255,255,100,0.5)';
ctx.fillRect(b.x - 4, b.y - 8, 8, 16);
}

// Draw particles
for (let pt of particles) {
const alpha = pt.life / 20;
ctx.fillStyle = pt.color.replace(')', `)`).replace('#', 'rgba(').replace(/'/g, '');
ctx.globalAlpha = alpha;
ctx.fillRect(pt.x - 3, pt.y - 3, 6, 6);
ctx.globalAlpha = 1;
}

// Draw ship
drawShip(shipX, shipY, shipW, shipH);

// HUD
ctx.fillStyle = '#fff';
ctx.font = `bold ${Math.floor(cellSize * 0.7)}px monospace`;
ctx.textAlign = 'left';
ctx.fillText(`SCORE: ${score}`, gridX + 10, gridY - cellSize * 0.5);
ctx.textAlign = 'right';
ctx.fillText(`BEST: ${highScore}`, W - gridX - 10, gridY - cellSize * 0.5);

// Speed indicator
const speedPct = Math.min(100, Math.floor((fallSpeed - 0.8) / 2 * 100));
ctx.textAlign = 'center';
ctx.fillStyle = speedPct > 60 ? '#ff4444' : '#44ff44';
ctx.fillText(`THREAT: ${speedPct}%`, W / 2, gridY - cellSize * 0.5);

ctx.restore();

if (gameState === 'gameover') {
drawGameOver();
}
}

function drawShip(x, y, w, h) {
const cx = x + w / 2;
const cy = y + h / 2;

// Body
ctx.fillStyle = '#44ccff';
ctx.beginPath();
ctx.moveTo(cx, y);
ctx.lineTo(x, y + h);
ctx.lineTo(x + w, y + h);
ctx.closePath();
ctx.fill();

// Cockpit
ctx.fillStyle = '#aaffff';
ctx.fillRect(cx - w * 0.15, cy - h * 0.2, w * 0.3, h * 0.4);

// Engine glow
const flicker = Math.abs(Math.sin(performance.now() * 0.02));
ctx.fillStyle = `rgba(255,${100 + flicker * 155},0,${0.5 + flicker * 0.5})`;
ctx.beginPath();
ctx.moveTo(cx - w * 0.3, y + h);
ctx.lineTo(cx, y + h + h * 0.4);
ctx.lineTo(cx + w * 0.3, y + h);
ctx.closePath();
ctx.fill();
}

function drawStartScreen() {
calcGrid();

// Title
ctx.textAlign = 'center';
const titleSize = Math.min(W * 0.08, 60);
ctx.font = `bold ${titleSize}px monospace`;

// Glow effect
ctx.shadowColor = '#44ccff';
ctx.shadowBlur = 20;
ctx.fillStyle = '#44ccff';
ctx.fillText('TETRIS', W / 2, H * 0.25);
ctx.fillStyle = '#ff4444';
ctx.shadowColor = '#ff4444';
ctx.fillText('SPACE INVADERS', W / 2, H * 0.25 + titleSize);
ctx.shadowBlur = 0;

// Subtitle
ctx.font = `${Math.min(W * 0.03, 24)}px monospace`;
ctx.fillStyle = '#aaa';
ctx.fillText('Destroy falling tetrominoes before they reach the top!', W / 2, H * 0.4);

// Controls
ctx.font = `${Math.min(W * 0.025, 20)}px monospace`;
ctx.fillStyle = '#88ccff';
const cy = H * 0.5;
const lines = [
'← → or A/D : Move ship',
'SPACE : Shoot',
'',
'Bonus for destroying entire pieces!',
'Speed increases over time...'
];
lines.forEach((line, i) => {
ctx.fillText(line, W / 2, cy + i * 30);
});

// Start prompt
const pulse = 0.5 + Math.abs(Math.sin(performance.now() * 0.003)) * 0.5;
ctx.globalAlpha = pulse;
ctx.fillStyle = '#fff';
ctx.font = `bold ${Math.min(W * 0.03, 28)}px monospace`;
ctx.fillText('PRESS SPACE TO START', W / 2, H * 0.8);
ctx.globalAlpha = 1;

// High score
ctx.fillStyle = '#ffff44';
ctx.font = `${Math.min(W * 0.025, 20)}px monospace`;
ctx.fillText(`HIGH SCORE: ${highScore}`, W / 2, H * 0.9);

// Decorative tetrominoes
const decoColors = ['#ff4444','#44ff44','#4444ff','#ffff44'];
for (let i = 0; i < 4; i++) {
const dx = W * 0.15 + i * W * 0.2;
const dy = H * 0.65 + Math.sin(performance.now() * 0.002 + i) * 10;
ctx.fillStyle = decoColors[i];
for (let j = 0; j < 4; j++) {
const shape = SHAPES[i][j];
ctx.fillRect(dx + shape[0] * 20, dy + shape[1] * 20, 18, 18);
}
}
}

function drawGameOver() {
ctx.fillStyle = 'rgba(0,0,0,0.7)';
ctx.fillRect(0, 0, W, H);

ctx.textAlign = 'center';
ctx.font = `bold ${Math.min(W * 0.07, 50)}px monospace`;
ctx.shadowColor = '#ff4444';
ctx.shadowBlur = 15;
ctx.fillStyle = '#ff4444';
ctx.fillText('GAME OVER', W / 2, H * 0.3);
ctx.shadowBlur = 0;

ctx.font = `${Math.min(W * 0.035, 30)}px monospace`;
ctx.fillStyle = '#fff';
ctx.fillText(`FINAL SCORE: ${score}`, W / 2, H * 0.45);

if (score >= highScore && score > 0) {
ctx.fillStyle = '#ffff44';
ctx.fillText('NEW HIGH SCORE!', W / 2, H * 0.53);
}

const pulse = 0.5 + Math.abs(Math.sin(performance.now() * 0.003)) * 0.5;
ctx.globalAlpha = pulse;
ctx.fillStyle = '#aaa';
ctx.font = `${Math.min(W * 0.03, 24)}px monospace`;
ctx.fillText('PRESS SPACE TO RESTART', W / 2, H * 0.7);
ctx.globalAlpha = 1;
}

// Main loop
let lastTime = 0;
function gameLoop(time) {
const dt = time - lastTime;
lastTime = time;

if (gameState === 'playing') {
update();
}
draw();
requestAnimationFrame(gameLoop);
}

calcGrid();
shipW = cellSize * 1.5;
shipH = cellSize * 0.8;
shipX = W / 2 - shipW / 2;
shipY = gridY + ROWS * cellSize + cellSize * 0.5;
requestAnimationFrame(gameLoop);

})();
</script>
</body>
```
