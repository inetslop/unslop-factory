function playSound(freq, dur, type='sine', vol=0.3) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = type;
    osc.frequency.value = freq;
    gain.gain.setValueAtTime(vol, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
    osc.connect(gain); gain.connect(audioCtx.destination);
    osc.start(); osc.stop(audioCtx.currentTime + dur);
}

function initGrid() {
    grid = [];
    for (let r = 0; r < ROWS; r++) {
        grid[r] = [];
        for (let c = 0; c < COLS; c++) {

        let r = Math.floor(Math.random() * ROWS);

}

function removeInitialMatches() {
    for (let r = 0; r < ROWS; r++) {
        for (let c = 0; c < COLS; c++) {
            if (!grid[r][c].isMine) {
                // Check right

                // Check down
                if (r < ROWS - 2 && grid[r][c].color === grid[r+1][c].color && grid[r+2][c].color === grid[r][c].color) {

                }

        }
    }
}

function hasMove() {
    for (let r = 0; r < ROWS; r++) {
        for (let c = 0; c < COLS; c++) {
            if (grid[r][c].isMine) continue;
            // Try swap right
            if (c < COLS-1 && !grid[r][c+1].isMine) {
                [grid[r][c], grid[r][c+1]] = [grid[r][c+1], grid[r][c]];
                if (checkMatch(r,c) || checkMatch(r,c+1)) { [grid[r][c], grid[r][c+1]] = [grid[r][c+1], grid[r][c]]; return true; }
                [grid[r][c], grid[r][c+1]] = [grid[r][c+1], grid[r][c]];
            }
            // Try swap down
            if (r < ROWS-1 && !grid[r+1][c].isMine) {
                [grid[r][c], grid[r+1][c]] = [grid[r+1][c], grid[r][c]];
                if (checkMatch(r,c) || checkMatch(r+1,c)) { [grid[r][c], grid[r+1][c]] = [grid[r+1][c], grid[r][c]]; return true; }
                [grid[r][c], grid[r+1][c]] = [grid[r+1][c], grid[r][c]];
            }
        }
    }
    return false;
}

function checkMatch(r, c) {
    const color = grid[r][c].color;
    if (grid[r][c].isMine) return 0;
    // Check horizontal
    let count = 1;
    for (let i = c-1; i >= 0 && grid[r][i].color === color && !grid[r][i].isMine; i--) count++;
    for (let i = c+1; i < COLS && grid[r][i].color === color && !grid[r][i].isMine; i++) count++;
    if (count >= 3) return count;
    // Check vertical
    count = 1;
    for (let i = r-1; i >= 0 && grid[i][c].color === color && !grid[i][c].isMine; i--) count++;
    for (let i = r+1; i < ROWS && grid[i][c].color === color && !grid[i][c].isMine; i++) count++;
    if (count >= 3) return count;
    return 0;
}

function findMatches() {
    let matched = [];
    for (let r = 0; r < ROWS; r++) {
        for (let c = 0; c < COLS; c++) {
            if (grid[r][c].isMine) continue;
            const color = grid[r][c].color;
            // Horizontal
            let count = 1;
            while (c + count < COLS && grid[r][c+count].color === color && !grid[r][c+count].isMine) count++;
            if (count >= 3) for (let i = 0; i < count; i++) matched.push([r, c+i]);
            // Vertical
            count = 1;
            while (r + count < ROWS && grid[r+count][c].color === color && !grid[r+count][c].isMine) count++;
            if (count >= 3) for (let i = 0; i < count; i++) matched.push([r+i, c]);
        }
    }
    // Deduplicate
    const seen = new Set();
    return matched.filter(([r,c]) => { const k=r*COLS+c; if(seen.has(k))return false; seen.add(k); return true; });
}

function clearMatches(matches) {
    for (const [r,c] of matches) {
        grid[r][c].color = -1;
        score += 10;
        spawnParticles(r, c);
    }
    scoreDisplay.textContent = 'Score: ' + score;
}

function applyGravity() {
    let moved = false;
    for (let c = 0; c < COLS; c++) {
        for (let r = ROWS-1; r >= 0; r--) {
            if (grid[r][c].color === -1 && !grid[r][c].isMine) {
                // Find nearest gem above
                let src = r - 1;
                while (src >= 0 && grid[src][c].color === -1) src--;
                if (src >= 0) {
                    grid[r][c] = grid[src][c];
                    grid[src][c] = { color: Math.floor(Math.random()*4), isMine: false, revealed: false };
                    moved = true;
                } else {
                    // Top row - new gem (small chance of mine)
                    const isM = Math.random() < 0.03;
                    grid[r][c] = { color: Math.floor(Math.random()*4), isMine: isM, revealed: false };
                    moved = true;
                }
            }
        }
    }
    return moved;
}

function spawnParticles(r, c) {
    const cx = gridX + (c+0.5)*cellSize;
    const cy = gridY + (r+0.5)*cellSize;
    const col = GEM_COLORS[grid[r][c].color >= 0 ? grid[r][c].color : 0];
    for (let i = 0; i < 6; i++) {
        particles.push({ x:cx, y:cy, vx:(Math.random()-0.5)*200, vy:(Math.random()-0.5)*200-50, life:1, color:col });
    }
}

function spawnExplosion(r, c) {
    const cx = gridX + (c+0.5)*cellSize;
    const cy = gridY + (r+0.5)*cellSize;
    for (let i = 0; i < 20; i++) {
        const angle = Math.random()*Math.PI*2;
        const speed = 100+Math.random()*200;
        particles.push({ x:cx, y:cy, vx:Math.cos(angle)*speed, vy:Math.sin(angle)*speed, life:1, color: Math.random()<0.5?'#ff6633':'#ffcc00' });
    }
}

function processBoard() {

                setTimeout(() => { applyGravity(); animating = false; checkGameState(); }, 200);
            } else {
                animating = false;
                checkGameState();
            }
        }, 150);
    } else {
        animating = false;
        checkGameState();
    }
}

function checkGameState() {
    if (!hasMove()) {
        // Shuffle non-mine tiles
        let colors = [];
        for (let r=0;r<ROWS;r++) for(let c=0;c<COLS;c++) if(!grid[r][c].isMine) colors.push(grid[r][c].color);
        for (let i=colors.length-1;i>0;i--){const j=Math.floor(Math.random()*(i+1));[colors[i],colors[j]]=[colors[j],colors[i]];}
        let idx=0;
        for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++){if(!grid[r][c].isMine)grid[r][c].color=colors[idx++];}
        removeInitialMatches();
    }
}

function swap(r1,c1,r2,c2) {
    // Check for mines
    if (grid[r1][c1].isMine || grid[r2][c2].isMine) {
        // MINE TRIGGERED!
        if (grid[r1][c1].isMine) { grid[r1][c1].revealed = true; spawnExplosion(r1,c1); }
        if (grid[r2][c2].isMine) { grid[r2][c2].revealed = true; spawnExplosion(r2,c2); }
        playBoom();
        lives--;
        updateLives();
        animating = true;
        setTimeout(() => {
            // Remove mine - make it empty and apply gravity
            for (let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++){if(grid[r][c].revealed){grid[r][c].color=-1;}}
            animating = false;
            if (lives <= 0) gameOver();
            else processBoard();
        }, 500);
    } else {
        // Normal swap

        playSwap();
        animating = true;
        setTimeout(() => { processBoard(); }, 100);
    }
}

function drawGem(x, y, size, color, alpha=1) {
    ctx.save();
    ctx.globalAlpha = alpha;
    const pad = size * 0.15;
    const s = size - pad*2;
    
    if (color === 0) { // Red - diamond
        ctx.fillStyle = '#ff4444';
        ctx.beginPath();
        ctx.moveTo(x+size/2, y+pad);
        ctx.lineTo(x+size-pad, y+size/2);
        ctx.lineTo(x+size/2, y+size-pad);
        ctx.lineTo(x+pad, y+size/2);
        ctx.closePath(); ctx.fill();
        ctx.fillStyle = 'rgba(255,255,255,0.3)';
        ctx.beginPath();
        ctx.moveTo(x+size/2, y+pad);
        ctx.lineTo(x+size-pad*1.5, y+size/2);
        ctx.lineTo(x+size/2, y+size/2-4);
        ctx.closePath(); ctx.fill();
    } else if (color === 1) { // Blue - circle
        ctx.fillStyle = '#4488ff';
        ctx.beginPath();
        ctx.arc(x+size/2, y+size/2, s/2, 0, Math.PI*2);
        ctx.fill();
        ctx.fillStyle = 'rgba(255,255,255,0.3)';
        ctx.beginPath();
        ctx.arc(x+size/2-4, y+size/2-4, s/4, 0, Math.PI*2);
        ctx.fill();
    } else if (color === 2) { // Green - hexagon
        ctx.fillStyle = '#44dd88';
        ctx.beginPath();
        for(let i=0;i<6;i++){const a=i*Math.PI/3-Math.PI/6;ctx.lineTo(x+size/2+Math.cos(a)*s/2, y+size/2+Math.sin(a)*s/2);}
        ctx.closePath(); ctx.fill();
        ctx.fillStyle = 'rgba(255,255,255,0.2)';
        ctx.beginPath();
        for(let i=0;i<6;i++){const a=i*Math.PI/3-Math.PI/6;ctx.lineTo(x+size/2+Math.cos(a)*s/4, y+size/2+Math.sin(a)*s/4);}
        ctx.closePath(); ctx.fill();
    } else if (color === 3) { // Yellow - star
        ctx.fillStyle = '#ffcc33';
        ctx.beginPath();
        for(let i=0;i<5;i++){const a=i*Math.PI*2/5-Math.PI/2;const r1=s/2,r2=s/4;
        if(i===0)ctx.moveTo(x+size/2+Math.cos(a)*r1,y+size/2+Math.sin(a)*r1);
        else{ctx.lineTo(x+size/2+Math.cos(a)*r1,y+size/2+Math.sin(a)*r1);const a2=a+Math.PI/5;ctx.lineTo(x+size/2+Math.cos(a2)*r2,y+size/2+Math.sin(a2)*r2);}
        }
        ctx.closePath(); ctx.fill();
    }
    ctx.restore();
}

function draw() {

    // Background
    ctx.fillStyle = '#1a1a2e';

    let ox = 0, oy = 0;
    if (shakeTime > 0) {
        ox = (Math.random()-0.5)*shakeIntensity;
        oy = (Math.random()-0.5)*shakeIntensity;
    }
    
    ctx.save();
    ctx.translate(ox, oy);
    
    // Grid background
    ctx.fillStyle = 'rgba(255,255,255,0.03)';
    ctx.fillRect(gridX-4, gridY-4, COLS*cellSize+8, ROWS*cellSize+8);
    
    // Cells
    for (let r = 0; r < ROWS; r++) {
        for (let c = 0; c < COLS; c++) {
            const x = gridX + c * cellSize, y = gridY + r * cellSize;
            
            // Cell background
            ctx.fillStyle = (r+c)%2===0 ? 'rgba(40,40,80,0.6)' : 'rgba(35,35,70,0.6)';
            ctx.fillRect(x+1, y+1, cellSize-2, cellSize-2);
            
            if (grid[r][c].revealed) {
                drawMine(x, y, cellSize);
            } else if (grid[r][c].color >= 0) {
                drawGem(x, y, cellSize, grid[r][c].color);
            }
            
            // Selection highlight
            if (selected && selected[0]===r && selected[1]===c) {
                ctx.strokeStyle = '#ffd700';
                ctx.lineWidth = 3;
                ctx.strokeRect(x+2, y+2, cellSize-4, cellSize-4);
            }
        }
    }
    
    // Particles
    for (let i = particles.length-1; i >= 0; i--) {
        const p = particles[i];
        ctx.fillStyle = p.color;
        ctx.globalAlpha = p.life;
        ctx.fillRect(p.x-3, p.y-3, 6, 6);
        ctx.globalAlpha = 1;
    }
    
    ctx.restore();
}

let lastTime = 0;
function loop(time) {
    const dt = Math.min((time - lastTime) / 1000, 0.05);
    lastTime = time;
    
    if (shakeTime > 0) shakeTime -= dt;
    
    // Update particles
    for (let i = particles.length-1; i >= 0; i--) {
        const p = particles[i];
        p.x += p.vx * dt;
        p.y += p.vy * dt;
        p.vy += 300 * dt;
        p.life -= dt * 2;
        if (p.life <= 0) particles.splice(i, 1);
    }
    
    draw();
    requestAnimationFrame(loop);
}

startBtn.addEventListener('click', startGame);
requestAnimationFrame(loop);
