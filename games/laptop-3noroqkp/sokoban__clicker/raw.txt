function tryAutoSolve() {
// Find a box not on target and try to push it toward nearest target

const [dx, dy] = findPushDir(b);

function findPushDir(box) {
// Simple heuristic: push toward nearest target
let best = null;
let bestDist = Infinity;
for (const t of targets) {
const dist = Math.abs(t.x - box.x) + Math.abs(t.y - box.y);
if (dist < bestDist && dist > 0) {
bestDist = dist;
best = t;
}
}
if (!best) return [0,0];
let dx = 0, dy = 0;
if (best.x > box.x) dx = 1; else if (best.x < box.x) dx = -1;
else if (best.y > box.y) dy = 1; else if (best.y < box.y) dy = -1;
return [dx, dy];
}

}
}

function playClick() {
if (!audioCtx) return;

const gain = audioCtx.createGain();

osc.connect(gain).connect(audioCtx.destination);
osc.start();
osc.stop(audioCtx.currentTime + 0.1);
}
