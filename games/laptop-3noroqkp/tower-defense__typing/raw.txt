const canvas = document.getElementById('game');
const ctx = canvas.getContext('2d');

let W, H;
function resize() {
    W = window.innerWidth;
    H = window.innerHeight;
    canvas.width = W;
    canvas.height = H;
}
window.addEventListener('resize', resize);
resize();

// Audio
let audioCtx = null;
function initAudio() {
    if (!audioCtx) audioCtx = new (window.AudioContext || webkitAudioContext)();
}
function playSound(type) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    switch(type) {
        case 'key':
            osc.type = 'sine';
            osc.frequency.setValueAtTime(600, audioCtx.currentTime);
            osc.frequency.exponentialRampToValueAtTime(800, audioCtx.currentTime + 0.05);
            gain.gain.setValueAtTime(0.1, audioCtx.currentTime);
            gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.08);
            osc.start(); osc.stop(audioCtx.currentTime + 0.08);
            break;
        case 'kill':
            osc.type = 'sawtooth';
            osc.frequency.setValueAtTime(200, audioCtx.currentTime);
            osc.frequency.exponentialRampToValueAtTime(50, audioCtx.currentTime + 0.3);
            gain.gain.setValueAtTime(0.3, audioCtx.currentTime);
            gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.3);
            osc.start(); osc.stop(audioCtx.currentTime + 0.3);
            break;
        case 'damage':
            osc.type = 'square';
            osc.frequency.setValueAtTime(150, audioCtx.currentTime);
            osc.frequency.exponentialRampToValueAtTime(80, audioCtx.currentTime + 0.2);
            gain.gain.setValueAtTime(0.4, audioCtx.currentTime);
            gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.25);
            osc.start(); osc.stop(audioCtx.currentTime + 0.25);
            break;
        case 'wave':
            osc.type = 'sine';
            osc.frequency.setValueAtTime(400, audioCtx.currentTime);
            osc.frequency.exponentialRampToValueAtTime(800, audioCtx.currentTime + 0.2);
            gain.gain.setValueAtTime(0.15, audioCtx.currentTime);
            gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.3);
            osc.start(); osc.stop(audioCtx.currentTime + 0.3);
            break;
    }
}

// Word lists by difficulty
const wordLists = [
    ['cat','dog','run','sit','bat','hit','top','wet','red','big','map','box','cup','fun','gap','jet','kit','lot','net','nut'],
    ['jump','swim','code','type','game','tower','defend','attack','battle','shield','arrow','blade','sword','magic','power','thrust','strike','crash','burst','flame'],
    ['warrior','defender','protector','champion','guardian','sentinel','vanguard','bastion','rampart','bulwark','stronghold','citadel','phalanx','legionnaire','cavalryman','archer','musketeer','gunner','sapper','engineer'],
    ['commander-in-chief','fortress-bastion','unbreakable-shield','irreducible-vanguard','impregnable-rampart','indomitable-sentinel','invincible-guardian','insurmountable-wall','indestructible-tower']
];

// Game state
let state = 'menu'; // menu, playing, gameover
let score = 0;
let highScore = 0;
try { highScore = parseInt(localStorage.getItem('typeDefenseHigh') || '0'); } catch(e) {}
let health = 10;
let maxHealth = 10;
let wave = 0;
let enemies = [];
let particles = [];
let shakeTime = 0;
let shakeMag = 0;
let typedInput = '';
let matchedEnemy = null;
let lastKeyTime = 0;
let waveTimer = 0;
let spawnTimer = 0;
let gameTime = 0;
let combo = 0;
let lastKillTime = 0;

// Lanes
const NUM_LANES = 4;
function getLaneY(lane) {
    const topMargin = 80;
    const bottomMargin = 120;
    const laneH = (H - topMargin - bottomMargin) / NUM_LANES;
    return topMargin + laneH * lane + laneH / 2;
}

class Enemy {
    constructor(word, lane) {
        this.word = word;
        this.lane = lane;
        this.x = W + 60;
        this.y = getLaneY(lane);
        this.speed = 40 + wave * 3 + Math.random() * 20;
        this.size = Math.min(35, 25 + word.length * 0.3);
        this.alive = true;
        this.hitFlash = 0;
        this.color = `hsl(${180 + Math.random() * 60}, 70%, 55%)`;
    }
    update(dt) {
        this.x -= this.speed * dt;
        if (this.hitFlash > 0) this.hitFlash -= dt;
        if (this.x < 30) {
            this.alive = false;
            health--;
            shakeTime = 0.3;
            shakeMag = 8;
            playSound('damage');
            spawnParticles(20, 40, 15, '#ff3333');
        }
    }
    draw() {
        const s = this.size;
        ctx.save();
        // Body - hexagon
        ctx.beginPath();
        for (let i = 0; i < 6; i++) {
            const a = Math.PI / 3 * i - Math.PI / 6;
            const px = this.x + s * Math.cos(a);
            const py = this.y + s * Math.sin(a);
            if (i === 0) ctx.moveTo(px, py); else ctx.lineTo(px, py);
        }
        ctx.closePath();
        ctx.fillStyle = this.hitFlash > 0 ? '#ffffff' : this.color;
        ctx.fill();
        ctx.strokeStyle = 'rgba(255,255,255,0.4)';
        ctx.lineWidth = 2;
        ctx.stroke();

        // Word above
        const fontSize = Math.max(12, Math.min(20, 30 - wordLenFactor(this.word.length)));
        ctx.font = `bold ${fontSize}px monospace`;
        ctx.textAlign = 'center';
        ctx.fillStyle = '#ffffff';
        ctx.fillText(this.word, this.x, this.y - s - 8);

        // Glow if matched
        if (matchedEnemy === this) {
            ctx.shadowColor = '#00ffcc';
            ctx.shadowBlur = 15;
            ctx.strokeStyle = '#00ffcc';
            ctx.lineWidth = 3;
            ctx.stroke();
            ctx.shadowBlur = 0;
        }
        ctx.restore();
    }
}

function wordLenFactor(len) { return len > 6 ? (len - 6) * 1.5 : 0; }

function spawnParticles(x, y, count, color) {
    for (let i = 0; i < count; i++) {
        const angle = Math.random() * Math.PI * 2;
        const speed = 50 + Math.random() * 150;
        particles.push({
            x, y,
            vx: Math.cos(angle) * speed,
            vy: Math.sin(angle) * speed,
            life: 0.5 + Math.random() * 0.5,
            maxLife: 0.5 + Math.random() * 0.5,
            color,
            size: 2 + Math.random() * 4
        });
    }
}

function spawnEnemy() {
    const diffLevel = Math.min(3, Math.floor(wave / 3));
    const list = wordLists[diffLevel];
    const word = list[Math.floor(Math.random() * list.length)];
    // Avoid same word appearing too many times
    const laneCount = enemies.filter(e => e.lane === -1).length;
    let lane = Math.floor(Math.random() * NUM_LANES);
    enemies.push(new Enemy(word, lane));
}

function startWave() {
    wave++;
    playSound('wave');
    spawnTimer = 0;
}

function startGame() {
    state = 'playing';
    score = 0;
    health = maxHealth;
    wave = 0;
    enemies = [];
    particles = [];
    typedInput = '';
    matchedEnemy = null;
    combo = 0;
    gameTime = 0;
    startWave();
}

function killEnemy(enemy) {
    enemy.alive = false;
    const now = performance.now() / 1000;
    if (now - lastKillTime < 2) combo++; else combo = 0;
    lastKillTime = now;
    const pts = enemy.word.length * 10 * (1 + combo * 0.5);
    score += Math.floor(pts);
    playSound('kill');
    spawnParticles(enemy.x, enemy.y, 25, '#00ffcc');
    spawnParticles(enemy.x, enemy.y, 10, '#ffffff');
    if (matchedEnemy === enemy) { matchedEnemy = null; typedInput = ''; }
}

function checkTyping() {
    // Find matching enemy
    matchedEnemy = null;
    for (const e of enemies) {
        if (!e.alive) continue;
        if (e.word.startsWith(typedInput)) {
            matchedEnemy = e;
            break;
        }
    }
}

function handleKey(e) {
    if (state === 'menu') {
        initAudio();
        startGame();
        return;
    }
    if (state === 'gameover') {
        if (e.key === 'r' || e.key === 'R' || e.key === 'Enter') {
            startGame();
        }
        return;
    }
    if (state !== 'playing') return;

    if (e.key.length === 1 && /[a-z]/i.test(e.key)) {
        const ch = e.key.toLowerCase();
        initAudio();
        playSound('key');
        typedInput += ch;
        checkTyping();
        // Check for exact match
        for (const en of enemies) {
            if (!en.alive) continue;
            if (typedInput === en.word) {
                killEnemy(en);
                break;
            }
        }
    } else if (e.key === 'Backspace') {
        typedInput = '';
        checkTyping();
    } else if (e.key === 'Escape') {
        typedInput = '';
        checkTyping();
    }
}

window.addEventListener('keydown', handleKey);

// Drawing helpers
function drawBackground() {
    // Dark gradient
    const grad = ctx.createLinearGradient(0, 0, 0, H);
    grad.addColorStop(0, '#0a0a2a');
    grad.addColorStop(1, '#1a0a2a');
    ctx.fillStyle = grad;
    ctx.fillRect(0, 0, W, H);

    // Grid lines
    ctx.strokeStyle = 'rgba(100, 150, 255, 0.08)';
    ctx.lineWidth = 1;
    for (let i = 0; i < NUM_LANES; i++) {
        const y = getLaneY(i);
        ctx.beginPath();
        ctx.moveTo(60, y);
        ctx.lineTo(W - 20, y);
        ctx.stroke();
    }

    // Base (left side)
    drawBase();
}

function drawBase() {
    const baseX = 30;
    const baseY = H / 2;
    const baseH = (H - 200) / NUM_LANES * NUM_LANES;

    // Tower shape
    ctx.save();
    ctx.fillStyle = 'rgba(100, 200, 255, 0.15)';
    ctx.fillRect(baseX - 10, baseY - baseH / 2, 20, baseH);

    // Glow
    const glowIntensity = health / maxHealth;
    ctx.shadowColor = `rgba(100, 200, 255, ${glowIntensity * 0.8})`;
    ctx.shadowBlur = 20;
    ctx.strokeStyle = `rgba(100, 200, 255, ${0.3 + glowIntensity * 0.5})`;
    ctx.lineWidth = 3;
    ctx.strokeRect(baseX - 10, baseY - baseH / 2, 20, baseH);
    ctx.shadowBlur = 0;
    ctx.restore();
}

function drawHUD() {
    // Health bar
    const hbX = W / 2 - 150;
    const hbY = 20;
    const hbW = 300;
    const hbH = 20;
    ctx.fillStyle = 'rgba(0,0,0,0.5)';
    ctx.fillRect(hbX, hbY, hbW, hbH);
    const hpRatio = health / maxHealth;
    const hpColor = hpRatio > 0.5 ? '#44ff88' : hpRatio > 0.25 ? '#ffaa33' : '#ff4444';
    ctx.fillStyle = hpColor;
    ctx.fillRect(hbX, hbY, hbW * hpRatio, hbH);
    ctx.strokeStyle = 'rgba(255,255,255,0.3)';
    ctx.strokeRect(hbX, hbY, hbW, hbH);

    // Health text
    ctx.font = '14px monospace';
    ctx.textAlign = 'center';
    ctx.fillStyle = '#fff';
    ctx.fillText(`${health} / ${maxHealth}`, W / 2, hbY + 15);

    // Score
    ctx.textAlign = 'left';
    ctx.font = 'bold 18px monospace';
    ctx.fillStyle = '#ffffff';
    ctx.fillText(`SCORE: ${score}`, 20, 30);

    // Wave
    ctx.textAlign = 'right';
    ctx.fillText(`WAVE ${wave}`, W - 20, 30);

    // Combo
    if (combo > 1) {
        ctx.textAlign = 'center';
        ctx.font = 'bold 24px monospace';
        ctx.fillStyle = '#ffcc00';
        ctx.fillText(`x${combo} COMBO!`, W / 2, 65);
    }

    // Typed input display
    if (typedInput) {
        const inputW = typedInput.length * 14 + 20;
        const inX = W / 2 - inputW / 2;
        const inY = H - 60;
        ctx.fillStyle = 'rgba(0,0,0,0.7)';
        ctx.fillRect(inX, inY, inputW, 35);
        ctx.strokeStyle = matchedEnemy ? '#00ffcc' : 'rgba(255,100,100,0.5)';
        ctx.lineWidth = 2;
        ctx.strokeRect(inX, inY, inputW, 35);
        ctx.font = 'bold 20px monospace';
        ctx.textAlign = 'center';
        ctx.fillStyle = matchedEnemy ? '#00ffcc' : '#ff6666';
        ctx.fillText(typedInput, W / 2, inY + 25);
    }

    // Hint
    if (!typedInput) {
        ctx.font = '13px monospace';
        ctx.textAlign = 'center';
        ctx.fillStyle = 'rgba(255,255,255,0.4)';
        ctx.fillText('Type the words to destroy enemies!', W / 2, H - 25);
    }
}

function drawParticles() {
    for (const p of particles) {
        const alpha = p.life / p.maxLife;
        ctx.globalAlpha = alpha;
        ctx.fillStyle = p.color;
        ctx.beginPath();
        ctx.arc(p.x, p.y, p.size * alpha, 0, Math.PI * 2);
        ctx.fill();
    }
    ctx.globalAlpha = 1;
}

function drawMenu() {
    drawBackground();

    // Title
    ctx.save();
    ctx.shadowColor = '#00ffcc';
    ctx.shadowBlur = 30;
    ctx.font = `bold ${Math.min(72, W * 0.08)}px monospace`;
    ctx.textAlign = 'center';
    ctx.fillStyle = '#ffffff';
    ctx.fillText('TYPE DEFENSE', W / 2, H * 0.3);
    ctx.shadowBlur = 0;

    // Subtitle
    ctx.font = `${Math.min(18, W * 0.03)}px monospace`;
    ctx.fillStyle = 'rgba(255,255,255,0.7)';
    ctx.fillText('A Tower Defense Where You Type to Fight', W / 2, H * 0.3 + 40);

    // Instructions
    const lines = [
        'Enemies march toward your base with words above them.',
        'Type a word exactly as shown to destroy that enemy.',
        'If enemies reach your base, you lose health.',
        'Waves get harder - longer words, faster enemies!',
        '',
        `HIGH SCORE: ${highScore}`
    ];
    ctx.font = `${Math.min(15, W * 0.025)}px monospace`;
    ctx.fillStyle = 'rgba(200,230,255,0.8)';
    lines.forEach((line, i) => {
        ctx.fillText(line, W / 2, H * 0.45 + i * 28);
    });

    // Start prompt
    const pulse = 0.5 + 0.5 * Math.sin(performance.now() / 500);
    ctx.font = `bold ${Math.min(22, W * 0.035)}px monospace`;

    ctx.fillText('Press any key to start', W / 2, H * 0.8);
    ctx.restore();

function drawGameOver() {
    // Dim overlay
    ctx.fillStyle = 'rgba(0,0,0,0.7)';
    ctx.fillRect(0, 0, W, H);

    ctx.save();
    ctx.shadowColor = '#ff3333';
    ctx.shadowBlur = 20;
    ctx.font = `bold ${Math.min(60, W * 0.07)}px monospace`;
    ctx.textAlign = 'center';
    ctx.fillStyle = '#ff4444';
    ctx.fillText('BASE DESTROYED', W / 2, H * 0.3);
    ctx.shadowBlur = 0;

    ctx.font = `${Math.min(24, W * 0.03)}px monospace`;
    ctx.fillStyle = '#ffffff';
    ctx.fillText(`Final Score: ${score}`, W / 2, H * 0.42);
    ctx.fillText(`Waves Survived: ${wave}`, W / 2, H * 0.48);

    if (score >= highScore && score > 0) {
        ctx.fillStyle = '#ffcc00';
        ctx.fillText('NEW HIGH SCORE!', W / 2, H * 0.55);
    } else {
        ctx.fillStyle = 'rgba(255,255,255,0.6)';
        ctx.fillText(`High Score: ${highScore}`, W / 2, H * 0.55);
    }

    const pulse = 0.5 + 0.5 * Math.sin(performance.now() / 500);
    ctx.font = `bold ${Math.min(20, W * 0.03)}px monospace`;
    ctx.fillStyle = `rgba(0, 255, 204, ${0.5 + pulse * 0.5})`;
    ctx.fillText('Press R or Enter to restart', W / 2, H * 0.7);
    ctx.restore();
}

// Game loop
let lastTime = performance.now();
function gameLoop(now) {
    const dt = Math.min((now - lastTime) / 1000, 0.05);
    lastTime = now;

    // Screen shake
    ctx.save();
    if (shakeTime > 0) {
        shakeTime -= dt;
        const sx = (Math.random() - 0.5) * shakeMag * (shakeTime / 0.3);
        const sy = (Math.random() - 0.5) * shakeMag * (shakeTime / 0.3);
        ctx.translate(sx, sy);
    }

    if (state === 'menu') {
        drawMenu();
    } else if (state === 'playing') {
        gameTime += dt;

        // Spawn logic
        spawnTimer -= dt;
        const maxEnemies = Math.min(5 + wave, 12);
        const enemiesAlive = enemies.filter(e => e.alive).length;
        if (spawnTimer <= 0 && enemiesAlive < maxEnemies) {
            spawnEnemy();
            spawnTimer = Math.max(1.5 - wave * 0.1, 0.6) + Math.random() * 0.5;
        }

        // Check wave progression
        if (enemies.filter(e => e.alive).length === 0 && enemies.length > 0) {
            waveTimer += dt;
            if (waveTimer > 1.5) {
                startWave();
                waveTimer = 0;
            }
        }
        // Auto advance wave if no enemies for a while
        if (enemies.filter(e => e.alive).length === 0 && enemies.length === 0) {
            waveTimer += dt;
            if (waveTimer > 1.0) {
                startWave();
                waveTimer = 0;
            }
        }

        // Update enemies
        for (const e of enemies) {
            if (e.alive) e.update(dt);
        }
        enemies = enemies.filter(e => e.alive || e.x > -50);

        // Check game over
        if (health <= 0) {
            state = 'gameover';
            try {
                if (score > highScore) {
                    highScore = score;
                    localStorage.setItem('typeDefenseHigh', String(highScore));
                }
            } catch(e2) {}
        }

        // Update particles
        for (const p of particles) {
            p.x += p.vx * dt;
            p.y += p.vy * dt;
            p.vy += 200 * dt; // gravity
            p.life -= dt;
        }
        particles = particles.filter(p => p.life > 0);

        // Draw
        drawBackground();
        for (const e of enemies) { if (e.alive) e.draw(); }
        drawParticles();
        drawHUD();
    } else if (state === 'gameover') {
        drawBackground();
        for (const e of enemies) { if (e.alive) e.draw(); }
        drawParticles();
        drawGameOver();
    }

    ctx.restore();
    requestAnimationFrame(gameLoop);
}

requestAnimationFrame(gameLoop);
</script>
