// Audio
let audioCtx = null;
function initAudio() {
    if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
}
function playSound(type) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    if (type === 'click') {
        osc.type = 'sine';
        osc.frequency.setValueAtTime(800, audioCtx.currentTime);
        osc.frequency.exponentialRampToValueAtTime(1200, audioCtx.currentTime + 0.05);
        gain.gain.setValueAtTime(0.3, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.15);
        osc.start(audioCtx.currentTime);
        osc.stop(audioCtx.currentTime + 0.15);
    } else if (type === 'gold') {
        osc.type = 'sine';
        osc.frequency.setValueAtTime(1000, audioCtx.currentTime);
        osc.frequency.exponentialRampToValueAtTime(2000, audioCtx.currentTime + 0.1);
        gain.gain.setValueAtTime(0.4, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.3);
        osc.start(audioCtx.currentTime);
        osc.stop(audioCtx.currentTime + 0.3);
    } else if (type === 'bomb') {
        osc.type = 'sawtooth';
        osc.frequency.setValueAtTime(200, audioCtx.currentTime);
        osc.frequency.exponentialRampToValueAtTime(50, audioCtx.currentTime + 0.3);
        gain.gain.setValueAtTime(0.4, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.3);
        osc.start(audioCtx.currentTime);
        osc.stop(audioCtx.currentTime + 0.3);
    } else if (type === 'countdown') {
        osc.type = 'square';
        osc.frequency.setValueAtTime(440, audioCtx.currentTime);
        gain.gain.setValueAtTime(0.2, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.15);
        osc.start(audioCtx.currentTime);
        osc.stop(audioCtx.currentTime + 0.15);
    } else if (type === 'gameover') {
        osc.type = 'sine';
        osc.frequency.setValueAtTime(600, audioCtx.currentTime);
        osc.frequency.exponentialRampToValueAtTime(200, audioCtx.currentTime + 0.5);
        gain.gain.setValueAtTime(0.3, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.5);
        osc.start(audioCtx.currentTime);
        osc.stop(audioCtx.currentTime + 0.5);
    } else if (type === 'start') {
        osc.type = 'sine';
        osc.frequency.setValueAtTime(400, audioCtx.currentTime);
        osc.frequency.exponentialRampToValueAtTime(800, audioCtx.currentTime + 0.2);
        gain.gain.setValueAtTime(0.3, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.3);
        osc.start(audioCtx.currentTime);
        osc.stop(audioCtx.currentTime + 0.3);

}

// Game state
let gameState = 'start'; // start, playing, gameover
let timeLeft = 30;
let lastTime = 0;
let items = [];
let particles = [];
let floatTexts = [];
let spawnTimer = 0;
let countdownBeep = 0;

const GAME_DURATION = 30;

// Player data
let p1 = { score: 0, combo: 0, maxCombo: 0 };
let p2 = { score: 0, combo: 0, maxCombo: 0 };

function resetGame() {
    timeLeft = GAME_DURATION;
    items = [];
    particles = [];
    floatTexts = [];
    spawnTimer = 0;
    countdownBeep = 0;
    p1 = { score: 0, combo: 0, maxCombo: 0 };
    p2 = { score: 0, combo: 0, maxCombo: 0 };
}

function spawnItem() {
    const side = Math.random() < 0.5 ? 1 : 2;
    const typeRoll = Math.random();
    let type, points, color, radius;
    
    if (typeRoll < 0.6) {
        type = 'normal';
        points = 1;
        color = side === 1 ? '#4fc3f7' : '#ff8a65';
        radius = 22;
    } else if (typeRoll < 0.85) {
        type = 'gold';
        points = 3;
        color = '#ffd700';
        radius = 18;
    } else {
        type = 'bomb';
        points = -2;
        color = '#f44336';
        radius = 20;
    }

    const margin = 80;
    let x, y;
    if (side === 1) {
        x = margin + Math.random() * (W / 2 - margin);
    } else {
        x = W / 2 + margin + Math.random() * (W / 2 - margin);
    }
    y = margin + Math.random() * (H - margin * 2);

    items.push({ x, y, radius, type, points, color, side, life: 4, pulse: 0 });
}

function createParticles(x, y, color, count) {
    for (let i = 0; i < count; i++) {
        const angle = Math.random() * Math.PI * 2;
        const speed = 2 + Math.random() * 4;
        particles.push({
            x, y,
            vx: Math.cos(angle) * speed,
            vy: Math.sin(angle) * speed,
            life: 0.5 + Math.random() * 0.3,
            maxLife: 0.8,
            color,
            radius: 3 + Math.random() * 4
        });
    }
}

function addFloatText(x, y, text, color) {
    floatTexts.push({ x, y, text, color, life: 1, vy: -2 });
}

// Input
let mouseX = 0, mouseY = 0;
canvas.addEventListener('mousemove', (e) => {
    const rect = canvas.getBoundingClientRect();
    mouseX = e.clientX - rect.left;
    mouseY = e.clientY - rect.top;
});

canvas.addEventListener('click', (e) => {
    initAudio();
    const rect = canvas.getBoundingClientRect();
    const x = e.clientX - rect.left;
    const y = e.clientY - rect.top;

    if (gameState === 'start') {
        startGame();
        return;
    }
    if (gameState === 'gameover') {
        // Check restart button
        const bx = W / 2, by = H / 2 + 80, bw = 200, bh = 50;
        if (x >= bx - bw/2 && x <= bx + bw/2 && y >= by - bh/2 && y <= by + bh/2) {
            startGame();
        }
        return;
    }

    if (gameState === 'playing') {
        let clicked = false;
        for (let i = items.length - 1; i >= 0; i--) {
            const item = items[i];
            const dx = x - item.x;
            const dy = y - item.y;
            if (dx * dx + dy * dy <= item.radius * item.radius) {
                clicked = true;
                let player;
                if (item.side === 1) player = p1;
                else player = p2;

                if (item.type === 'bomb') {
                    player.score += item.points;
                    player.combo = 0;
                    addFloatText(item.x, item.y, `${item.points}`, '#f44336');
                    playSound('bomb');
                } else {
                    player.combo++;
                    if (player.combo > player.maxCombo) player.maxCombo = player.combo;
                    const comboBonus = Math.floor(player.combo / 5);
                    const pts = item.points + comboBonus;
                    player.score += pts;
                    addFloatText(item.x, item.y, `+${pts}`, item.type === 'gold' ? '#ffd700' : (item.side === 1 ? '#4fc3f7' : '#ff8a65'));
                    playSound(item.type === 'gold' ? 'gold' : 'click');
                }

                createParticles(item.x, item.y, item.color, item.type === 'bomb' ? 12 : 8);
                items.splice(i, 1);
                break;
            }
        }
    }
});

// Keyboard for restart
window.addEventListener('keydown', (e) => {
    if (e.key === 'r' || e.key === 'R') {
        if (gameState === 'gameover' || gameState === 'start') {
            initAudio();
            startGame();
        }
    }
    if (e.key === 'Escape' && gameState === 'playing') {
        gameState = 'start';
    }
});

function startGame() {
    resetGame();
    gameState = 'playing';
    playSound('start');
}

function update(dt) {
    if (gameState !== 'playing') return;

    timeLeft -= dt;
    if (timeLeft <= 0) {
        timeLeft = 0;
        gameState = 'gameover';
        playSound('gameover');
        return;
    }

    // Countdown beep
    const sec = Math.ceil(timeLeft);
    if (sec !== countdownBeep && sec <= 5 && sec > 0) {
        countdownBeep = sec;
        playSound('countdown');
    }

    // Spawn items
    spawnTimer -= dt;
    if (spawnTimer <= 0) {
        const maxItems = 8;
        if (items.length < maxItems) {
            spawnItem();
        }
        // Increase spawn rate as game progresses
        const progress = 1 - timeLeft / GAME_DURATION;
        spawnTimer = Math.max(0.4, 1.2 - progress * 0.6);
    }

    // Update items
    for (let i = items.length - 1; i >= 0; i--) {
        items[i].life -= dt;
        items[i].pulse += dt * 5;
        if (items[i].life <= 0) {
            items.splice(i, 1);
        }
    }

    // Update particles
    for (let i = particles.length - 1; i >= 0; i--) {
        const p = particles[i];
        p.x += p.vx;
        p.y += p.vy;
        p.vy += 5 * dt;
        p.life -= dt;
        if (p.life <= 0) particles.splice(i, 1);
    }

    // Update float texts
    for (let i = floatTexts.length - 1; i >= 0; i--) {
        const ft = floatTexts[i];
        ft.y += ft.vy;
        ft.life -= dt * 1.5;
        if (ft.life <= 0) floatTexts.splice(i, 1);
    }
}

function draw() {
    ctx.fillStyle = '#1a1a2e';
    ctx.fillRect(0, 0, W, H);

    if (gameState === 'start') {
        drawStartScreen();
    } else if (gameState === 'playing') {
        drawGame();
    } else if (gameState === 'gameover') {
        drawGame();
        drawGameOver();
    }
}

function drawStartScreen() {
    // Title
    ctx.fillStyle = '#ffffff';
    ctx.font = `bold ${Math.min(72, W * 0.08)}px Arial`;
    ctx.textAlign = 'center';
    ctx.fillText('CLICK BATTLE', W / 2, H * 0.25);

    // Subtitle
    ctx.font = `${Math.min(24, W * 0.03)}px Arial`;
    ctx.fillStyle = '#aaa';
    ctx.fillText('A two-player clicker duel!', W / 2, H * 0.35);

    // Instructions
    ctx.font = `${Math.min(20, W * 0.025)}px Arial`;
    ctx.fillStyle = '#ccc';
    const lines = [
        'Player 1 (Left) vs Player 2 (Right)',
        '',
        'Click items to score points!',
        'Blue/Green circles = +1 point',
        'Gold stars = +3 points',
        'Red bombs = -2 points',
        'Build combos for bonus points!',
        '',
        `First to ${GAME_DURATION} seconds wins!`,
    ];
    let y = H * 0.45;
    lines.forEach((line, i) => {
        ctx.fillText(line, W / 2, y);
        y += Math.min(32, H * 0.04);
    });

    // Start prompt
    const pulse = 0.7 + Math.sin(Date.now() * 0.005) * 0.3;
    ctx.globalAlpha = pulse;
    ctx.font = `bold ${Math.min(28, W * 0.035)}px Arial`;
    ctx.fillStyle = '#4fc3f7';
    ctx.fillText('Click to Start!', W / 2, H * 0.85);
    ctx.globalAlpha = 1;

    // Decorative items
    const t = Date.now() * 0.003;
    for (let i = 0; i < 6; i++) {
        const angle = t + i * Math.PI / 3;
        const rx = W * 0.35;
        const ry = H * 0.15;
        const cx = W / 2 + Math.cos(angle) * rx;
        const cy = H * 0.25 + Math.sin(angle) * ry * 0.3;
        ctx.beginPath();
        ctx.arc(cx, cy, 8, 0, Math.PI * 2);
        ctx.fillStyle = i % 3 === 0 ? '#ffd700' : (i < 3 ? '#4fc3f7' : '#ff8a65');
        ctx.fill();
    }
}

function drawGame() {
    // Divider
    ctx.strokeStyle = 'rgba(255,255,255,0.2)';
    ctx.lineWidth = 2;
    ctx.setLineDash([10, 10]);
    ctx.beginPath();
    ctx.moveTo(W / 2, 60);
    ctx.lineTo(W / 2, H);
    ctx.stroke();
    ctx.setLineDash([]);

    // Top bar background
    ctx.fillStyle = 'rgba(0,0,0,0.5)';
    ctx.fillRect(0, 0, W, 60);

    // Timer
    const timerColor = timeLeft <= 5 ? '#f44336' : '#ffffff';
    ctx.fillStyle = timerColor;
    ctx.font = `bold 28px Arial`;
    ctx.textAlign = 'center';
    ctx.fillText(Math.ceil(timeLeft) + '', W / 2, 40);

    // Scores
    ctx.textAlign = 'left';
    ctx.fillStyle = '#4fc3f7';
    ctx.font = `bold 24px Arial`;
    ctx.fillText(`P1: ${p1.score}`, 20, 40);
    if (p1.combo >= 5) {
        ctx.font = `16px Arial`;
        ctx.fillStyle = '#ffd700';
        ctx.fillText(`x${p1.combo} combo!`, 20, 58);
    }

    ctx.textAlign = 'right';
    ctx.fillStyle = '#ff8a65';
    ctx.font = `bold 24px Arial`;
    ctx.fillText(`P2: ${p2.score}`, W - 20, 40);
    if (p2.combo >= 5) {
        ctx.font = `16px Arial`;
        ctx.fillStyle = '#ffd700';
        ctx.fillText(`${p2.combo}x combo!`, W - 20, 58);
    }

    // Items
    items.forEach(item => {
        const alpha = item.life < 1 ? item.life : 1;
        ctx.globalAlpha = alpha;
        const pulseScale = 1 + Math.sin(item.pulse) * 0.1;
        const r = item.radius * pulseScale;

        if (item.type === 'gold') {
            drawStar(item.x, item.y, r, item.color);
        } else if (item.type === 'bomb') {
            drawBomb(item.x, item.y, r, item.color);
        } else {
            ctx.beginPath();
            ctx.arc(item.x, item.y, r, 0, Math.PI * 2);
            ctx.fillStyle = item.color;
            ctx.fill();
            ctx.strokeStyle = 'rgba(255,255,255,0.5)';
            ctx.lineWidth = 2;
            ctx.stroke();
        }

        // Glow
        ctx.shadowBlur = 15;
        ctx.shadowColor = item.color;
        ctx.beginPath();
        ctx.arc(item.x, item.y, r * 0.3, 0, Math.PI * 2);
        ctx.fillStyle = 'rgba(255,255,255,0.3)';
        ctx.fill();
        ctx.shadowBlur = 0;

        ctx.globalAlpha = 1;
    });

    // Particles
    particles.forEach(p => {
        const alpha = p.life / p.maxLife;
        ctx.globalAlpha = alpha;
        ctx.beginPath();
        ctx.arc(p.x, p.y, p.radius * alpha, 0, Math.PI * 2);
        ctx.fillStyle = p.color;
        ctx.fill();
    });
    ctx.globalAlpha = 1;

    // Float texts
    floatTexts.forEach(ft => {
        ctx.globalAlpha = ft.life;
        ctx.font = `bold 20px Arial`;
        ctx.textAlign = 'center';
        ctx.fillStyle = ft.color;
        ctx.fillText(ft.text, ft.x, ft.y);
    });
    ctx.globalAlpha = 1;
}

function drawStar(x, y, r, color) {
    ctx.beginPath();
    for (let i = 0; i < 5; i++) {
        const angle = (i * 4 * Math.PI) / 5 - Math.PI / 2;
        const px = x + Math.cos(angle) * r;
        const py = y + Math.sin(angle) * r;
        if (i === 0) ctx.moveTo(px, py);
        else ctx.lineTo(px, py);
    }
    ctx.closePath();
    ctx.fillStyle = color;
    ctx.fill();
    ctx.strokeStyle = 'rgba(255,255,255,0.6)';
    ctx.lineWidth = 2;
    ctx.stroke();
}

function drawBomb(x, y, r, color) {
    ctx.beginPath();
    ctx.arc(x, y, r, 0, Math.PI * 2);
    ctx.fillStyle = color;
    ctx.fill();
    // Fuse
    ctx.beginPath();
    ctx.moveTo(x + r * 0.5, y - r * 0.7);
    ctx.quadraticCurveTo(x + r * 0.8, y - r * 1.2, x + r * 0.3, y - r * 1.1);
    ctx.strokeStyle = '#8b4513';
    ctx.lineWidth = 3;
    ctx.stroke();
    // Skull-like X
    ctx.font = `bold ${r}px Arial`;
    ctx.textAlign = 'center';
    ctx.textBaseline = 'middle';
    ctx.fillStyle = '#fff';
    ctx.fillText('✕', x, y + 2);
}

function drawGameOver() {
    // Overlay
    ctx.fillStyle = 'rgba(0,0,0,0.7)';
    ctx.fillRect(0, 0, W, H);

    ctx.textAlign = 'center';
    ctx.fillStyle = '#ffffff';
    ctx.font = `bold ${Math.min(48, W * 0.06)}px Arial`;
    ctx.fillText('GAME OVER', W / 2, H * 0.25);

    // Winner
    let winnerText;
    if (p1.score > p2.score) winnerText = '🏆 Player 1 Wins!';
    else if (p2.score > p1.score) winnerText = '🏆 Player 2 Wins!';
    else winnerText = "It's a Tie!";

    ctx.font = `bold ${Math.min(36, W * 0.045)}px Arial`;
    const winColor = p1.score > p2.score ? '#4fc3f7' : p2.score > p1.score ? '#ff8a65' : '#fff';
    ctx.fillStyle = winColor;
    ctx.fillText(winnerText, W / 2, H * 0.38);

    // Final scores
    ctx.font = `bold ${Math.min(28, W * 0.035)}px Arial`;
    ctx.fillStyle = '#4fc3f7';
    ctx.fillText(`P1: ${p1.score} pts (Max Combo: ${p1.maxCombo})`, W / 2 - 150, H * 0.48);
    ctx.fillStyle = '#ff8a65';
    ctx.fillText(`P2: ${p2.score} pts (Max Combo: ${p2.maxCombo})`, W / 2 + 150, H * 0.48);

    // Restart button
    const bx = W / 2, by = H / 2 + 80, bw = 200, bh = 50;
    ctx.fillStyle = '#4fc3f7';
    roundRect(bx - bw/2, by - bh/2, bw, bh, 10);
    ctx.fill();
    ctx.fillStyle = '#000';
    ctx.font = `bold 22px Arial`;
    ctx.fillText('PLAY AGAIN', bx, by + 7);

    // Hint
    ctx.fillStyle = '#888';
    ctx.font = `16px Arial`;
    ctx.fillText('(or press R)', W / 2, H / 2 + 130);
}

function roundRect(x, y, w, h, r) {
    ctx.beginPath();
    ctx.moveTo(x + r, y);
    ctx.lineTo(x + w - r, y);
    ctx.quadraticCurveTo(x + w, y, x + w, y + r);
    ctx.lineTo(x + w, y + h - r);
    ctx.quadraticCurveTo(x + w, y + h, x + w - r, y + h);
    ctx.lineTo(x + r, y + h);
    ctx.quadraticCurveTo(x, y + h, x, y + h - r);
    ctx.lineTo(x, y + r);
    ctx.quadraticCurveTo(x, y, x + r, y);
    ctx.closePath();
}

// Game loop
function gameLoop(timestamp) {
    const dt = Math.min((timestamp - lastTime) / 1000, 0.1);
    lastTime = timestamp;
    update(dt);
    draw();
    requestAnimationFrame(gameLoop);
}
requestAnimationFrame(gameLoop);
</script>
</body>
