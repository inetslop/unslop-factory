const TILE = 40;
const GRAVITY = 0.6;
const JUMP_VEL = -12;
const MOVE_SPEED = 4;
const FRICTION = 0.85;

let gameState = 'start'; // start, playing, won, allwon
let currentLevel = 0;

function playSound(type) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    
    if (type === 'jump') {
        osc.type = 'sine';
        osc.frequency.setValueAtTime(300, audioCtx.currentTime);
        osc.frequency.linearRampToValueAtTime(600, audioCtx.currentTime + 0.1);
        gain.gain.setValueAtTime(0.2, audioCtx.currentTime);
        gain.gain.linearRampToValueAtTime(0, audioCtx.currentTime + 0.15);
        osc.start();
        osc.stop(audioCtx.currentTime + 0.15);
    } else if (type === 'push') {
        osc.type = 'square';
        osc.frequency.setValueAtTime(150, audioCtx.currentTime);
        gain.gain.setValueAtTime(0.15, audioCtx.currentTime);
        gain.gain.linearRampToValueAtTime(0, audioCtx.currentTime + 0.1);
        osc.start();
        osc.stop(audioCtx.currentTime + 0.1);
    } else if (type === 'place') {
        osc.type = 'sine';
        osc.frequency.setValueAtTime(500, audioCtx.currentTime);
        osc.frequency.linearRampToValueAtTime(800, audioCtx.currentTime + 0.2);
        gain.gain.setValueAtTime(0.25, audioCtx.currentTime);
        gain.gain.linearRampToValueAtTime(0, audioCtx.currentTime + 0.3);
        osc.start();
        osc.stop(audioCtx.currentTime + 0.3);
    } else if (type === 'win') {
        osc.type = 'sine';
        const notes = [523, 659, 784, 1047];
        notes.forEach((f, i) => {
            osc.frequency.setValueAtTime(f, audioCtx.currentTime + i * 0.12);
        });
        gain.gain.setValueAtTime(0.3, audioCtx.currentTime);
        gain.gain.linearRampToValueAtTime(0, audioCtx.currentTime + 0.5);
        osc.start();
        osc.stop(audioCtx.currentTime + 0.5);
    } else if (type === 'land') {
        osc.type = 'triangle';
        osc.frequency.setValueAtTime(100, audioCtx.currentTime);
        gain.gain.setValueAtTime(0.1, audioCtx.currentTime);
        gain.gain.linearRampToValueAtTime(0, audioCtx.currentTime + 0.08);
        osc.start();
        osc.stop(audioCtx.currentTime + 0.08);
    }
}

function loadLevel(idx) {
    if (idx >= levels.length) idx = 0;
    currentLevel = idx;
    const lvl = levels[idx];
    grid = [];
    boxes = [];
    targets = [];
    particles = [];
    wonLevel = false;
    winTimer = 0;

    for (let r = 0; r < lvl.length; r++) {
        grid[r] = [];
        for (let c = 0; c < lvl[r].length; c++) {
            const ch = lvl[r][c];
            if (ch === '#') grid[r][c] = 1;
            else grid[r][c] = 0;
            
            if (ch === 'P') { player.x = c * TILE + 4; player.y = r * TILE + 4; }
            if (ch === 'B') boxes.push({ x: c * TILE + 4, y: r * TILE + 4, onTarget: false });
            if (ch === 'T') targets.push({ x: c * TILE, y: r * TILE });
        }
    }
    
    player.vx = 0;
    player.vy = 0;
    player.onGround = false;
}

function isSolid(col, row) {
    if (row < 0 || row >= grid.length) return true;
    if (col < 0 || col >= (grid[0] ? grid[0].length : 0)) return true;
    return grid[row][col] === 1;
}

function isBoxAt(px, py) {
    for (let b of boxes) {
        const bc = Math.floor((px + player.w/2) / TILE);
        const br = Math.floor((py + player.h/2) / TILE);
        const bb = Math.floor((b.x + 16) / TILE);
        const bb2 = Math.floor((b.y + 16) / TILE);
        if (bc === bb && br === bb2) return b;
    }
    return null;
}

function boxAt(col, row) {
    for (let b of boxes) {
        const bc = Math.floor((b.x + 16) / TILE);
        const br = Math.floor((b.y + 16) / TILE);
        if (bc === col && br === row) return b;
    }
    return null;
}

function update() {
    if (gameState !== 'playing' && gameState !== 'won') return;
    
    if (wonLevel) {
        winTimer++;
        if (winTimer > 90) {
            currentLevel++;
            if (currentLevel >= levels.length) {
                gameState = 'allwon';
            } else {
                loadLevel(currentLevel);
                gameState = 'playing';
            }
        }
        return;
    }

    // Player physics
    if (keys['ArrowLeft'] || keys['KeyA']) { player.vx -= 0.8; player.facing = -1; }
    if (keys['ArrowRight'] || keys['KeyD']) { player.vx += 0.8; player.facing = 1; }
    if ((keys['ArrowUp'] || keys['KeyW'] || keys['Space']) && player.onGround) {
        player.vy = JUMP_VEL;
        player.onGround = false;
        playSound('jump');
    }

    player.vx *= FRICTION;
    if (Math.abs(player.vx) > 6) player.vx = Math.sign(player.vx) * 6;
    player.vy += GRAVITY;
    if (player.vy > 15) player.vy = 15;

    // Move X
    let newX = player.x + player.vx;
    let colL = Math.floor(newX / TILE);
    let colR = Math.floor((newX + player.w - 1) / TILE);
    let rowT = Math.floor(player.y / TILE);
    let rowB = Math.floor((player.y + player.h - 1) / TILE);

    if (player.vx > 0) {
        let col = Math.floor((newX + player.w) / TILE);
        for (let r = rowT; r <= rowB; r++) {
            if (isSolid(col, r)) {
                newX = col * TILE - player.w;
                // Check box push
                const box = boxAt(col, Math.floor((player.y + player.h/2) / TILE));
                if (box) {
                    const bc = Math.floor((box.x + 16) / TILE);
                    const nextCol = bc + 1;
                    if (!isSolid(nextCol, bc !== -1 ? Math.floor((box.y+16)/TILE) : rowT) && !boxAt(nextCol, Math.floor((box.y+16)/TILE))) {
                        box.x += TILE;
                        playSound('push');
                    }
                }
                player.vx = 0;
                break;
            }
        }
    } else if (player.vx < 0) {
        let col = Math.floor(newX / TILE);
        for (let r = rowT; r <= rowB; r++) {
            if (isSolid(col, r)) {
                newX = (col + 1) * TILE;
                const box = boxAt(col, Math.floor((player.y + player.h/2) / TILE));
                if (box) {
                    const bc = Math.floor((box.x + 16) / TILE);
                    const nextCol = bc - 1;
                    if (!isSolid(nextCol, Math.floor((box.y+16)/TILE)) && !boxAt(nextCol, Math.floor((box.y+16)/TILE))) {
                        box.x -= TILE;
                        playSound('push');
                    }
                }
                player.vx = 0;
                break;
            }
        }
    }
    player.x = newX;

    // Move Y
    let newY = player.y + player.vy;
    if (player.vy > 0) {
        let row = Math.floor((newY + player.h) / TILE);
        let colC = Math.floor((player.x + player.w/2) / TILE);
        if (isSolid(colC, row)) {
            newY = row * TILE - player.h;
            if (!player.onGround) playSound('land');
            player.vy = 0;
            player.onGround = true;
        } else {
            player.onGround = false;
        }
    } else if (player.vy < 0) {
        let row = Math.floor(newY / TILE);
        let colC = Math.floor((player.x + player.w/2) / TILE);
        if (isSolid(colC, row)) {
            newY = (row + 1) * TILE;
            player.vy = 0;
        }
    }
    player.y = newY;

    // Update boxes - check targets and gravity
    for (let b of boxes) {
        b.onTarget = false;
        const bc = Math.floor((b.x + 16) / TILE);
        const br = Math.floor((b.y + 16) / TILE);
        for (let t of targets) {
            if (Math.floor(t.x / TILE) === bc && Math.floor(t.y / TILE) === br) {
                b.onTarget = true;
            }
        }
        // Box gravity - keep boxes on ground
        const boxRow = Math.floor((b.y + 36) / TILE);
        if (!isSolid(Math.floor((b.x+16)/TILE), boxRow)) {
            b.y += GRAVITY * 2;
        } else {
            const snapY = (boxRow - 1) * TILE + 4;
            if (Math.abs(b.y - snapY) < 5) b.y = snapY;
        }
    }

    // Check win
    if (boxes.length > 0 && boxes.every(b => b.onTarget)) {
        wonLevel = true;
        gameState = 'won';
        playSound('win');
        spawnWinParticles();
    }
}

function spawnWinParticles() {
    for (let i = 0; i < 30; i++) {
        particles.push({
            x: player.x + player.w/2,
            y: player.y,
            vx: (Math.random() - 0.5) * 8,
            vy: -Math.random() * 10 - 3,
            life: 60 + Math.random() * 30,
            color: `hsl(${Math.random() * 360}, 100%, 60%)`
        });
    }
}

function updateParticles() {
    for (let p of particles) {
        p.x += p.vx;
        p.y += p.vy;
        p.vy += 0.3;
        p.life--;
    }
    particles = particles.filter(p => p.life > 0);
}

    ctx.translate(offX, offY);

    if (gameState === 'playing' && e.code === 'KeyR') {
        loadLevel(currentLevel);
    }
    
    // Prevent scrolling
    if (['Space', 'ArrowUp', 'ArrowDown', 'ArrowLeft', 'ArrowRight'].includes(e.code)) {
        e.preventDefault();
    }
});

window.addEventListener('keyup', (e) => {
    keys[e.code] = false;
});

function gameLoop() {
    update();
    updateParticles();
    draw();
    requestAnimationFrame(gameLoop);
}

loadLevel(0);
gameLoop();
