function playSound(type) {
  if (!audioCtx) return;
  const now = audioCtx.currentTime;
  const osc = audioCtx.createOscillator();
  const gain = audioCtx.createGain();
  osc.connect(gain);
  gain.connect(audioCtx.destination);

  switch(type) {
    case 'swap':
      osc.type = 'sine';
      osc.frequency.setValueAtTime(400, now);
      osc.frequency.linearRampToValueAtTime(600, now + 0.1);
      gain.gain.setValueAtTime(0.2, now);
      gain.gain.linearRampToValueAtTime(0, now + 0.15);
      osc.start(now); osc.stop(now + 0.15);
      break;
    case 'invalid':
      osc.type = 'square';
      osc.frequency.setValueAtTime(150, now);
      gain.gain.setValueAtTime(0.15, now);
      gain.gain.linearRampToValueAtTime(0, now + 0.2);
      osc.start(now); osc.stop(now + 0.2);
      break;
    case 'match':
      osc.type = 'sawtooth';
      osc.frequency.setValueAtTime(300, now);
      osc.frequency.linearRampToValueAtTime(80, now + 0.2);
      gain.gain.setValueAtTime(0.25, now);
      genGainLinear(gain, 0, now + 0.25);
      osc.start(now); osc.stop(now + 0.25);
      break;
    case 'row':
      osc.type = 'square';
      osc.frequency.setValueAtTime(200, now);
      osc.frequency.linearRampToValueAtTime(100, now + 0.3);
      gain.gain.setValueAtTime(0.15, now);
      gain.gain.linearRampToValueAtTime(0, now + 0.3);
      osc.start(now); osc.stop(now + 0.3);

      gain.gain.setValueAtTime(0.3, now);

      osc.start(now); osc.stop(now + 1);
      break;
  }

function initGrid() {
  grid = [];

    grid[r] = [];

function randomType() { return Math.floor(Math.random() * TYPES); }

function hasMatch() {
  return findMatches().length > 0;
}

function findMatches() {
  const matches = new Set();
  // Horizontal
  for (let r = 0; r < ROWS; r++) {
    for (let c = 0; c < COLS - 2; c++) {
      if (grid[r][c] !== -1 && grid[r][c] === grid[r][c+1] && grid[r][c] === grid[r][c+2]) {
        matches.add(r * COLS + c);
        matches.add(r * COLS + c + 1);
        matches.add(r * COLS + c + 2);
      }
    }
  }
  // Vertical
  for (let r = 0; r < ROWS - 2; r++) {
    for (let c = 0; c < COLS; c++) {
      if (grid[r][c] !== -1 && grid[r][c] === grid[r+1][c] && grid[r][c] === grid[r+2][c]) {
        matches.add(r * COLS + c);
        matches.add((r+1) * COLS + c);
        matches.add((r+2) * COLS + c);
      }
    }
  }
  const result = [];
  matches.forEach(idx => {
    result.push({ r: Math.floor(idx / COLS), c: idx % COLS });
  });
  return result;
}

}

function resolveMatches() {
  const matches = findMatches();
  if (matches.length === 0) {
    resolving = false;
    combo = 0;
    return;
  }

  combo++;
  score += matches.length * 10 * combo;

    grid[cell.r][cell.c] = -1;
  }

  // Apply gravity
  applyGravity();

  // Check again after delay
  setTimeout(() => resolveMatches(), 200);
}

function applyGravity() {
  for (let c = 0; c < COLS; c++) {
    // Move non-empty cells down
    let writeRow = ROWS - 1;
    for (let r = ROWS - 1; r >= 0; r--) {
      if (grid[r][c] !== -1) {
        grid[writeRow][c] = grid[r][c];
        if (writeRow !== r) grid[r][c] = -1;
        writeRow--;
      }
    }
    // Fill top with new aliens
    for (let r = writeRow; r >= 0; r--) {
      grid[r][c] = randomType();
    }
  }
}

function addNewRow() {
  // Shift everything down and add new row at top
  for (let r = ROWS - 1; r > 0; r--) {
    for (let c = 0; c < COLS; c++) {
      grid[r][c] = grid[r-1][c];
    }
  }
  // New top row
  for (let c = 0; c < COLS; c++) {
    grid[0][c] = randomType();
  }

  playSound('row');
  shakeMag = 3;
  shakeTime = 0.3;

  // Check game over
  if (!hasValidMove()) {
    endGame();
  }
}

function hasValidMove() {
  for (let r = 0; r < ROWS; r++) {
    for (let c = 0; c < COLS; c++) {
      // Try swap right
      if (c < COLS - 1) {
        swap(r, c, r, c + 1);
        if (hasMatch()) { swap(r, c, r, c + 1); return true; }
        swap(r, c, r, c + 1);
      }
      // Try swap down
      if (r < ROWS - 1) {
        swap(r, c, r + 1, c);
        if (hasMatch()) { swap(r, c, r + 1, c); return true; }
        swap(r, c, r + 1, c);
      }
    }
  }
  return false;
}

function swap(r1, c1, r2, c2) {
  const tmp = grid[r1][c1];
  grid[r1][c1] = grid[r2][c2];
  grid[r2][c2] = tmp;
}

function handleClick(e) {
  if (resolving) return;
  const x = e.clientX - gridX;
  const y = e.clientY - gridY;
  const c = Math.floor(x / cellSize);
  const r = Math.floor(y / cellSize);

  if (c < 0 || c >= COLS || r < 0 || r >= ROWS) { selected = null; return; }

  if (!selected) {
    selected = { r, c };
  } else {
    // Check if adjacent
    const dr = Math.abs(selected.r - r);
    const dc = Math.abs(selected.c - c);
    if ((dr === 1 && dc === 0) || (dr === 0 && dc === 1)) {
      // Attempt swap
      playSound('swap');
      swap(selected.r, selected.c, r, c);
      if (hasMatch()) {
        resolving = true;
        combo = 0;
        resolveMatches();
      } else {
        playSound('invalid');
        swap(selected.r, selected.c, r, c); // swap back
      }
      selected = null;
    } else if (selected.r === r && selected.c === c) {
      selected = null;
    } else {
      selected = { r, c };
    }
  }
}

// Rendering
function drawAlien(type, x, y, size, alpha = 1) {
  if (type < 0 || type >= TYPES) return;
  const pattern = alienPatterns[type];
  const px = size / 10; // pixel size within the cell
  const py = size / 8;
  const offsetX = x + (size - 10 * px) / 2;
  const offsetY = y + (size - 8 * py) / 2;

  ctx.globalAlpha = alpha;
  ctx.fillStyle = alienColors[type];

  // Glow
  ctx.shadowColor = alienGlow[type];
  ctx.shadowBlur = size * 0.15;

  for (let row = 0; row < 8; row++) {
    for (let col = 0; col < 10; col++) {
      if (pattern[row][col]) {
        ctx.fillRect(offsetX + col * px, offsetY + row * py, px + 0.5, py + 0.5);
      }
    }
  }
  ctx.shadowBlur = 0;
  ctx.globalAlpha = 1;
}

function drawBackground() {
  ctx.fillStyle = '#0a0a1a';
  ctx.fillRect(0, 0, W, H);

  // Stars
  const t = performance.now() * 0.001;
  for (const star of stars) {
    const sx = ((star.x + t * star.speed * 0.01) % 1) * W;
    const sy = star.y * H;
    const brightness = 0.5 + 0.5 * Math.sin(t * 2 + star.x * 10);
    ctx.fillStyle = `rgba(255,255,255,${brightness * 0.7})`;
    ctx.fillRect(sx, sy, star.s, star.s);
  }
}

function drawGrid() {

  for (let r = 0; r <= ROWS; r++) {
    ctx.beginPath();
    ctx.moveTo(gridX, gridY + r * cellSize);
    ctx.lineTo(gridX + COLS * cellSize, gridY + r * cellSize);
    ctx.stroke();
  }
  for (let c = 0; c <= COLS; c++) {
    ctx.beginPath();
    ctx.moveTo(gridX + c * cellSize, gridY);
    ctx.lineTo(gridX + c * cellSize, gridY + ROWS * cellSize);
    ctx.stroke();
  }

  // Aliens
  for (let r = 0; r < ROWS; r++) {
    for (let c = 0; c < COLS; c++) {
      if (grid[r][c] !== -1) {
        drawAlien(grid[r][c], gridX + c * cellSize, gridY + r * cellSize, cellSize);
      }
    }
  }

  // Selection highlight
  if (selected && !resolving) {
    const sx = gridX + selected.c * cellSize;
    const sy = gridY + selected.r * cellSize;
    ctx.strokeStyle = '#ffffff';
    ctx.lineWidth = 3;
    ctx.shadowColor = '#00ffff';
    ctx.shadowBlur = 10;
    ctx.strokeRect(sx + 2, sy + 2, cellSize - 4, cellSize - 4);
    ctx.shadowBlur = 0;
  }
}

function drawParticles() {
  for (const p of particles) {
    ctx.globalAlpha = p.life;
    ctx.fillStyle = p.color;
    ctx.fillRect(p.x - p.size/2, p.y - p.size/2, p.size, p.size);
  }
  ctx.globalAlpha = 1;
}

function drawHUD() {
  // Score
  ctx.font = `bold ${Math.min(24, cellSize * 0.6)}px monospace`;
  ctx.fillStyle = '#00ffff';
  ctx.textAlign = 'left';
  ctx.fillText(`SCORE: ${score}`, gridX, 30);

  ctx.textAlign = 'right';
  ctx.fillStyle = '#ffff00';
  ctx.fillText(`HI: ${highScore}`, gridX + COLS * cellSize, 30);

  // Combo indicator
  if (combo > 1 && resolving) {
    ctx.textAlign = 'center';
    ctx.font = `bold ${cellSize * 0.7}px monospace`;
    ctx.fillStyle = `hsl(${(performance.now() * 0.5) % 360}, 100%, 60%)`;
    ctx.fillText(`COMBO x${combo}!`, W/2, gridY - 10);
  }

  // Row timer bar
  const progress = rowTimer / rowInterval;
  const barW = COLS * cellSize;
  const barH = 6;
  ctx.fillStyle = 'rgba(255, 50, 50, 0.3)';
  ctx.fillRect(gridX, gridY - 14, barW, barH);
  ctx.fillStyle = progress > 0.7 ? '#ff4444' : '#44ff44';
  ctx.fillRect(gridX, gridY - 14, barW * (1 - progress), barH);

  // Warning flash
  if (progress > 0.8) {
    const flash = Math.sin(performance.now() * 0.01) * 0.5 + 0.5;
    ctx.fillStyle = `rgba(255, 0, 0, ${flash * 0.3})`;
    ctx.fillRect(gridX - 4, gridY - 18, barW + 8, barH + 4);
  }
}

// Game loop
function update(dt) {
  if (gameState !== 'PLAYING') return;

  gameTime += dt;
  rowTimer += dt;

  // Decrease interval over time
  const currentInterval = Math.max(6, rowInterval - Math.floor(gameTime / 30));

  if (rowTimer >= currentInterval) {
    rowTimer = 0;
    addNewRow();
  }

  // Update particles
  for (let i = particles.length - 1; i >= 0; i--) {
    const p = particles[i];
    p.x += p.vx;
    p.y += p.vy;
    p.vy += 0.1;
    p.life -= p.decay;
    if (p.life <= 0) particles.splice(i, 1);
  }

  // Shake decay
  if (shakeTime > 0) shakeTime -= dt;
}

function render() {
  ctx.save();

  // Screen shake
  if (shakeTime > 0) {
    const sx = (Math.random() - 0.5) * shakeMag;
    const sy = (Math.random() - 0.5) * shakeMag;
    ctx.translate(sx, sy);
  }

  if (gameState === 'START') {
    drawStartScreen();
  } else {
    drawBackground();
    drawGrid();
    drawParticles();
    drawHUD();
    if (gameState === 'GAMEOVER') drawGameOver();
  }

  ctx.restore();
}

function gameLoop(timestamp) {
  const dt = Math.min((timestamp - lastTime) / 1000, 0.1);
  lastTime = timestamp;

  update(dt);
  render();
  requestAnimationFrame(gameLoop);
}
