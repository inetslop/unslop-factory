<script>
const canvas = document.getElementById('game');
const ctx = canvas.getContext('2d');

let W, H, CELL, COLS, ROWS;
let audioCtx = null;

function resize() {
    W = window.innerWidth;
    H = window.innerHeight;
    canvas.width = W;
    canvas.height = H;
    const maxCell = Math.floor(Math.min(W / 20, H / 15));
    CELL = Math.max(20, maxCell);
    COLS = Math.floor(W / CELL);
    ROWS = Math.floor(H / CELL);
}
window.addEventListener('resize', resize);
resize();

// Audio
function initAudio() {
    if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
}

function playSound(type) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    
    if (type === 'eat') {
        osc.type = 'square';
        osc.frequency.setValueAtTime(600, audioCtx.currentTime);
        osc.frequency.exponentialRampToValueAtTime(1200, audioCtx.currentTime + 0.05);
        gain.gain.setValueAtTime(0.15, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.1);
        osc.start();
        osc.stop(audioCtx.currentTime + 0.1);
    } else if (type === 'hit') {
        osc.type = 'sawtooth';
        osc.frequency.setValueAtTime(200, audioCtx.currentTime);
        osc.frequency.exponentialRampToValueAtTime(50, audioCtx.currentTime + 0.3);
        gain.gain.setValueAtTime(0.3, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.4);
        osc.start();
        osc.stop(audioCtx.currentTime + 0.4);
    } else if (type === 'shoot') {
        osc.type = 'square';
        osc.frequency.setValueAtTime(800, audioCtx.currentTime);
        osc.frequency.exponentialRampToValueAtTime(200, audioCtx.currentTime + 0.05);
        gain.gain.setValueAtTime(0.05, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.08);
        osc.start();
        osc.stop(audioCtx.currentTime + 0.08);
    } else if (type === 'score') {
        osc.type = 'sine';
        osc.frequency.setValueAtTime(440, audioCtx.currentTime);
        osc.frequency.setValueAtTime(660, audioCtx.currentTime + 0.1);
        gain.gain.setValueAtTime(0.1, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.3);
        osc.start();
        osc.stop(audioCtx.currentTime + 0.3);
    } else if (type === 'gameover') {
        osc.type = 'sawtooth';
        osc.frequency.setValueAtTime(400, audioCtx.currentTime);
        osc.frequency.exponentialRampToValueAtTime(60, audioCtx.currentTime + 1);
        gain.gain.setValueAtTime(0.25, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 1.2);
        osc.start();
        osc.stop(audioCtx.currentTime + 1.2);
    }
}

// Game state
let state = 'start'; // start, playing, gameover
let snake, dir, nextDir, food, bullets, particles, score, highScore = 0;
let moveTimer, moveInterval, spawnTimer, patternTimer, patternIndex, difficulty;
let shootCooldown, canShoot;

function resetGame() {
    const startX = Math.floor(COLS / 2);
    const startY = Math.floor(ROWS / 2);
    snake = [{ x: startX, y: startY }, { x: startX - 1, y: startY }];
    dir = { x: 1, y: 0 };
    nextDir = { x: 1, y: 0 };
    food = spawnFood();
    bullets = [];
    particles = [];
    score = 0;
    moveTimer = 0;
    moveInterval = 8;
    spawnTimer = 0;
    patternTimer = 0;
    patternIndex = 0;
    difficulty = 0;
    shootCooldown = 0;
    canShoot = false;
}

function spawnFood() {
    let fx, fy;
    do {
        fx = Math.floor(Math.random() * COLS);
        fy = Math.floor(Math.random() * ROWS);
    } while (snake.some(s => s.x === fx && s.y === fy));
    return { x: fx, y: fy };
}

function spawnPattern(type) {
    const cx = W / 2;
    const cy = H / 2;
    
    if (type === 'radial') {
        const count = 8 + Math.floor(difficulty / 5);
        const angle = Math.random() * Math.PI * 2;
        for (let i = 0; i < count; i++) {
            const a = angle + (i / count) * Math.PI * 2;
            const speed = 1.5 + difficulty * 0.1;
            bullets.push({ x: cx, y: cy, vx: Math.cos(a) * speed, vy: Math.sin(a) * speed, r: 4 });
        }
    } else if (type === 'spiral') {
        const count = 5 + Math.floor(difficulty / 3);
        for (let i = 0; i < count; i++) {
            const a = patternTimer * 0.5 + i * 0.3;
            const speed = 1.2 + difficulty * 0.08;
            bullets.push({ x: cx, y: -10, vx: Math.cos(a) * speed, vy: Math.sin(a) * speed + 0.5, r: 4 });
        }
    } else if (type === 'aimed') {
        const head = snake[0];
        const hx = head.x * CELL + CELL / 2;
        const hy = head.y * CELL + CELL / 2;
        const count = 3 + Math.floor(difficulty / 4);
        for (let i = 0; i < count; i++) {
            const spread = (i - (count - 1) / 2) * 0.15;
            const angle = Math.atan2(hy - 0, hx - W / 2) + spread;
            const speed = 2 + difficulty * 0.1;
            bullets.push({ x: W / 2, y: -10, vx: Math.cos(angle) * speed, vy: Math.sin(angle) * speed, r: 5 });
        }
    } else if (type === 'wall') {
        const gapX = Math.random() * (W - 80) + 40;
        for (let x = 20; x < W - 20; x += 30) {
            if (Math.abs(x - gapX) > 50) {
                bullets.push({ x: x, y: -10, vx: 0, vy: 1.5 + difficulty * 0.05, r: 4 });
            }
        }
    } else if (type === 'cross') {
        const count = 3;
        for (let i = 0; i < count; i++) {
            bullets.push({ x: -10, y: H / 2, vx: 2 + difficulty * 0.1, vy: (i - 1) * 1.5, r: 4 });
            bullets.push({ x: W + 10, y: H / 2, vx: -(2 + difficulty * 0.1), vy: (i - 1) * 1.5, r: 4 });
        }
    } else if (type === 'rain') {
        const count = 3 + Math.floor(difficulty / 3);
        for (let i = 0; i < count; i++) {
            bullets.push({ x: Math.random() * W, y: -10, vx: (Math.random() - 0.5) * 2, vy: 1 + difficulty * 0.08, r: 4 });
        }
    }
}

function shootBullet() {
    if (!canShoot || shootCooldown > 0) return;
    const head = snake[0];
    const hx = head.x * CELL + CELL / 2;
    const hy = head.y * CELL + CELL / 2;
    bullets.push({ x: hx, y: hy, vx: dir.x * 5, vy: dir.y * 5, r: 3, friendly: true });
    shootCooldown = 15;
    playSound('shoot');
}

function update(dt) {
    if (state !== 'playing') return;
    
    difficulty += dt * 0.01;
    
    // Move snake
    moveTimer += dt;
    const interval = Math.max(3, moveInterval - difficulty * 0.3);
    if (moveTimer >= interval) {
        moveTimer = 0;
        dir = nextDir;
        const head = snake[0];
        const nx = head.x + dir.x;
        const ny = head.y + dir.y;
        
        // Check wall collision
        if (nx < 0 || nx >= COLS || ny < 0 || ny >= ROWS) {
            gameOver();
            return;
        }
        // Check self collision
        for (let i = 1; i < snake.length; i++) {
            if (snake[i].x === nx && snake[i].y === ny) {
                gameOver();
                return;
            }
        }
        
        snake.unshift({ x: nx, y: ny });
        
        // Check food
        if (nx === food.x && ny === food.y) {
            score += 10;
            playSound('eat');
            spawnParticles(nx * CELL + CELL/2, ny * CELL + CELL/2, '#ff0', 8);
            food = spawnFood();
            canShoot = true;
        } else {
            snake.pop();
        }
        
        // Score for survival
        score += 1;
    }
    
    // Shoot cooldown
    if (shootCooldown > 0) shootCooldown -= dt / 16;
    
    // Spawn patterns
    spawnTimer += dt;
    patternTimer += dt * 0.05;
    const spawnInterval = Math.max(30, 80 - difficulty * 2);
    if (spawnTimer >= spawnInterval) {
        spawnTimer = 0;
        const types = ['radial', 'spiral', 'aimed', 'wall', 'cross', 'rain'];
        const maxPattern = Math.min(types.length, 2 + Math.floor(difficulty / 3));
        const type = types[Math.floor(Math.random() * maxPattern)];
        spawnPattern(type);
    }
    
    // Update bullets
    for (let i = bullets.length - 1; i >= 0; i--) {
        const b = bullets[i];
        b.x += b.vx * dt / 16;
        b.y += b.vy * dt / 16;
        
        if (b.x < -20 || b.x > W + 20 || b.y < -30 || b.y > H + 20) {
            bullets.splice(i, 1);
            continue;
        }
        
        // Collision with snake
        const head = snake[0];
        const hx = head.x * CELL + CELL / 2;
        const hy = head.y * CELL + CELL / 2;
        
        for (let s = 0; s < snake.length; s++) {
            const sx = snake[s].x * CELL + CELL / 2;
            const sy = snake[s].y * CELL + CELL / 2;
            const dx = b.x - sx;
            const dy = b.y - sy;
            const dist = Math.sqrt(dx * dx + dy * dy);
            
            if (dist < (b.r + CELL * 0.3)) {
                if (b.friendly) {
                    // Friendly bullet destroys enemy bullets nearby
                    for (let j = bullets.length - 1; j >= 0; j--) {
                        if (j === i) continue;
                        const eb = bullets[j];
                        const exd = b.x - eb.x;
                        const eyd = b.y - eb.y;
                        if (Math.sqrt(exd*exd + eyd*eyd) < 30) {
                            spawnParticles(eb.x, eb.y, '#f80', 4);
                            bullets.splice(j, 1);
                            score += 5;
                        }
                    }
                    bullets.splice(i, 1);
                    break;
                } else {
                    gameOver();
                    return;
                }
            }
        }
    }
    
    // Update particles
    for (let i = particles.length - 1; i >= 0; i--) {
        const p = particles[i];
        p.x += p.vx * dt / 16;
        p.y += p.vy * dt / 16;
        p.life -= dt / 16;
        if (p.life <= 0) particles.splice(i, 1);
    }
}

function spawnParticles(x, y, color, count) {
    for (let i = 0; i < count; i++) {
        const a = Math.random() * Math.PI * 2;
        const speed = 1 + Math.random() * 3;
        particles.push({ x, y, vx: Math.cos(a) * speed, vy: Math.sin(a) * speed, life: 20 + Math.random() * 15, color });
    }
}

function gameOver() {
    state = 'gameover';
    if (score > highScore) {
        highScore = score;
        try { localStorage.setItem('bhs_high', highScore); } catch(e) {}
    }
    playSound('gameover');
    spawnParticles(snake[0].x * CELL + CELL/2, snake[0].y * CELL + CELL/2, '#f00', 20);
}

function draw() {
    ctx.fillStyle = '#0a0a1a';
    ctx.fillRect(0, 0, W, H);
    
    // Grid (subtle)
    ctx.strokeStyle = 'rgba(30,30,60,0.5)';
    ctx.lineWidth = 1;
    for (let x = 0; x <= COLS; x++) {
        ctx.beginPath();
        ctx.moveTo(x * CELL, 0);
        ctx.lineTo(x * CELL, ROWS * CELL);
        ctx.stroke();
    }
    for (let y = 0; y <= ROWS; y++) {
        ctx.beginPath();
        ctx.moveTo(0, y * CELL);
        ctx.lineTo(COLS * CELL, y * CELL);
        ctx.stroke();
    }
    
    if (state === 'start') {
        drawStartScreen();
    } else {
        // Food
        const fx = food.x * CELL + CELL / 2;
        const fy = food.y * CELL + CELL / 2;
        ctx.fillStyle = '#ff3366';
        ctx.shadowColor = '#ff3366';
        ctx.shadowBlur = 10;
        ctx.beginPath();
        ctx.arc(fx, fy, CELL * 0.35, 0, Math.PI * 2);
        ctx.fill();
        ctx.shadowBlur = 0;
        
        // Snake
        for (let i = snake.length - 1; i >= 0; i--) {
            const s = snake[i];
            const sx = s.x * CELL + CELL / 2;
            const sy = s.y * CELL + CELL / 2;
            const t = i / snake.length;
            const r = Math.floor(50 + (1 - t) * 100);
            const g = Math.floor(200 + (1 - t) * 55);
            ctx.fillStyle = `rgb(${r},${g},50)`;
            if (i === 0) {
                ctx.shadowColor = '#0f0';
                ctx.shadowBlur = 8;
            } else {
                ctx.shadowBlur = 0;
            }
            ctx.beginPath();
            ctx.arc(sx, sy, CELL * 0.4, 0, Math.PI * 2);
            ctx.fill();
        }
        ctx.shadowBlur = 0;
        
        // Bullets
        for (const b of bullets) {
            if (b.friendly) {
                ctx.fillStyle = '#0ff';
                ctx.shadowColor = '#0ff';
            } else {
                ctx.fillStyle = '#f44';
                ctx.shadowColor = '#f44';
            }
            ctx.shadowBlur = 6;
            ctx.beginPath();
            ctx.arc(b.x, b.y, b.r, 0, Math.PI * 2);
            ctx.fill();
        }
        ctx.shadowBlur = 0;
        
        // Particles
        for (const p of particles) {
            ctx.globalAlpha = Math.min(1, p.life / 20);
            ctx.fillStyle = p.color;
            ctx.fillRect(p.x - 2, p.y - 2, 4, 4);
        }
        ctx.globalAlpha = 1;
        
        // HUD
        ctx.fillStyle = '#fff';
        ctx.font = 'bold 16px monospace';
        ctx.textAlign = 'left';
        ctx.fillText(`SCORE: ${score}`, 15, 25);
        ctx.fillText(`HI: ${highScore}`, 15, 45);
        ctx.fillText(`LEN: ${snake.length}`, 15, 65);
        if (canShoot) {
            ctx.fillStyle = '#0ff';
            ctx.fillText('[SPACE] SHOOT', 15, 85);
        }
        
        // Difficulty indicator
        ctx.textAlign = 'right';
        ctx.fillStyle = difficulty > 10 ? '#f44' : '#888';
        const diffText = `THREAT: ${Math.floor(difficulty)}`;
        ctx.fillText(diffText, W - 15, 25);
        
        if (state === 'gameover') {
            drawGameOver();
        }
    }
}

function drawStartScreen() {
    ctx.textAlign = 'center';
    ctx.fillStyle = '#0f0';
    ctx.shadowColor = '#0f0';
    ctx.shadowBlur = 20;
    ctx.font = `bold ${Math.min(48, W/15)}px monospace`;
    ctx.fillText('BULLET HELL SNAKE', W/2, H/3);
    ctx.shadowBlur = 0;
    
    ctx.fillStyle = '#aaa';
    ctx.font = '16px monospace';
    const lines = [
        'ARROW KEYS - Move snake',
        'SPACE - Shoot (unlocked after eating)',
        '',
        'Eat food to grow and survive.',
        'Dodge the bullet patterns!',
        'Don\'t hit walls or your own tail.',
        '',
        'Press ANY KEY to start'
    ];
    lines.forEach((line, i) => {
        ctx.fillText(line, W/2, H/2 + i * 24);
    });
    
    // Animated bullets in background
    const t = Date.now() * 0.001;
    for (let i = 0; i < 20; i++) {
        const bx = (Math.sin(t + i * 0.7) * 0.5 + 0.5) * W;
        const by = (Math.cos(t * 0.7 + i * 1.3) * 0.5 + 0.5) * H;
        ctx.fillStyle = `rgba(255,68,68,${0.3 + Math.sin(t + i) * 0.2})`;
        ctx.beginPath();
        ctx.arc(bx, by, 4, 0, Math.PI * 2);
        ctx.fill();
    }
}

function drawGameOver() {
    ctx.fillStyle = 'rgba(0,0,0,0.7)';
    ctx.fillRect(0, 0, W, H);
    
    ctx.textAlign = 'center';
    ctx.fillStyle = '#f44';
    ctx.shadowColor = '#f44';
    ctx.shadowBlur = 15;
    ctx.font = `bold ${Math.min(42, W/12)}px monospace`;
    ctx.fillText('GAME OVER', W/2, H/3);
    ctx.shadowBlur = 0;
    
    ctx.fillStyle = '#fff';
    ctx.font = '20px monospace';
    ctx.fillText(`Score: ${score}`, W/2, H/3 + 45);
    ctx.fillText(`High Score: ${highScore}`, W/2, H/3 + 75);
    
    ctx.fillStyle = '#8f8';
    ctx.font = '16px monospace';
    ctx.fillText('Press R to restart', W/2, H/3 + 120);
}

let lastTime = 0;
function gameLoop(time) {
    const dt = Math.min(time - lastTime, 50);
    lastTime = time;
    
    if (state === 'playing') {
        update(dt);
    } else {
        // Update particles even in gameover
        for (let i = particles.length - 1; i >= 0; i--) {
            const p = particles[i];
            p.x += p.vx * dt / 16;
            p.y += p.vy * dt / 16;
            p.life -= dt / 16;
            if (p.life <= 0) particles.splice(i, 1);
        }
    }
    
    draw();
    requestAnimationFrame(gameLoop);
}

// Input
let keys = {};
window.addEventListener('keydown', (e) => {
    keys[e.key] = true;
    
    if (state === 'start') {
        initAudio();
        resetGame();
        state = 'playing';
        return;
    }
    
    if (state === 'gameover' && (e.key === 'r' || e.key === 'R')) {
        resetGame();
        state = 'playing';
        return;
    }
    
    if (state !== 'playing') return;
    
    if (e.key === 'ArrowUp' || e.key === 'w' || e.key === 'W') {
        if (dir.x !== 0) nextDir = { x: 0, y: -1 };
    } else if (e.key === 'ArrowDown' || e.key === 's' || e.key === 'S') {
        if (dir.x !== 0) nextDir = { x: 0, y: 1 };
    } else if (e.key === 'ArrowLeft' || e.key === 'a' || e.key === 'A') {
        if (dir.y !== 0) nextDir = { x: -1, y: 0 };
    } else if (e.key === 'ArrowRight' || e.key === 'd' || e.key === 'D') {
        if (dir.y !== 0) nextDir = { x: 1, y: 0 };
    } else if (e.key === ' ') {
        e.preventDefault();
        shootBullet();
    }
    
    // Prevent scrolling
    if (['ArrowUp','ArrowDown','ArrowLeft','ArrowRight',' '].includes(e.key)) {
        e.preventDefault();
    }
});

window.addEventListener('keyup', (e) => {
    keys[e.key] = false;
});

// Load high score
try {
    const saved = localStorage.getItem('bhs_high');
    if (saved) highScore = parseInt(saved) || 0;
} catch(e) {}

resetGame();
state = 'start';
requestAnimationFrame(gameLoop);
</script>
