<script>
(function() {
  const SYMBOLS = ['🔥','💧','⭐','🌙','⚡','🍀','🎵','💎'];
  let cards = [];
  let flipped = [];
  let moves = 0;
  let pairsFound = 0;
  let seconds = 0;
  let timerInterval = null;
  let gameActive = false;
  let audioCtx = null;

  const board = document.getElementById('board');
  const startScreen = document.getElementById('startScreen');
  const hud = document.getElementById('hud');
  const restartBtn = document.getElementById('restartBtn');
  const winScreen = document.getElementById('winScreen');
  const moveCountEl = document.getElementById('moveCount');
  const timeCountEl = document.getElementById('timeCount');
  const pairCountEl = document.getElementById('pairCount');
  const startBtn = document.getElementById('startBtn');
  const playAgainBtn = document.getElementById('playAgainBtn');
  const particlesCanvas = document.getElementById('particlesCanvas');
  const pCtx = particlesCanvas.getContext('2d');

  let particles = [];
  let animFrame = null;

  function initAudio() {
    if (!audioCtx) {
      audioCtx = new (window.AudioContext || window.webkitAudioContext)();
    }
  }

  function playFlipSound() {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.type = 'sine';
    osc.frequency.setValueAtTime(400, audioCtx.currentTime);
    osc.frequency.exponentialRampToValueAtTime(800, audioCtx.currentTime + 0.1);
    gain.gain.setValueAtTime(0.3, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.15);
    osc.start(audioCtx.currentTime);
    osc.stop(audioCtx.currentTime + 0.15);
  }

  function playMatchSound() {
    if (!audioCtx) return;
    const notes = [523, 659, 784];
    notes.forEach((freq, i) => {
      const osc = audioCtx.createOscillator();
      const gain = audioCtx.createGain();
      osc.connect(gain);
      gain.connect(audioCtx.destination);
      osc.type = 'sine';
      osc.frequency.setValueAtTime(freq, audioCtx.currentTime + i * 0.08);
      gain.gain.setValueAtTime(0.25, audioCtx.currentTime + i * 0.08);
      gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.3 + i * 0.08);
      osc.start(audioCtx.currentTime + i * 0.08);
      osc.stop(audioCtx.currentTime + 0.4 + i * 0.08);
    });
  }

  function playMismatchSound() {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.type = 'sawtooth';
    osc.frequency.setValueAtTime(200, audioCtx.currentTime);
    osc.frequency.exponentialRampToValueAtTime(100, audioCtx.currentTime + 0.2);
    gain.gain.setValueAtTime(0.15, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.25);
    osc.start(audioCtx.currentTime);
    osc.stop(audioCtx.currentTime + 0.25);
  }

  function playWinSound() {
    if (!audioCtx) return;
    const melody = [523, 659, 784, 1047, 784, 1047];
    melody.forEach((freq, i) => {
      const osc = audioCtx.createOscillator();
      const gain = audioCtx.createGain();
      osc.connect(gain);
      gain.connect(audioCtx.destination);
      osc.type = 'sine';
      osc.frequency.setValueAtTime(freq, audioCtx.currentTime + i * 0.12);
      gain.gain.setValueAtTime(0.3, audioCtx.currentTime + i * 0.12);
      gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.4 + i * 0.12);
      osc.start(audioCtx.currentTime + i * 0.12);
      osc.stop(audioCtx.currentTime + 0.5 + i * 0.12);
    });
  }

  function formatTime(s) {
    const m = Math.floor(s / 60);
    const sec = s % 60;
    return m + ':' + (sec < 10 ? '0' : '') + sec;
  }

  function startTimer() {
    clearInterval(timerInterval);
    timerInterval = setInterval(() => {
      seconds++;
      timeCountEl.textContent = formatTime(seconds);
    }, 1000);
  }

  function resizeCanvas() {
    particlesCanvas.width = window.innerWidth;
    particlesCanvas.height = window.innerHeight;
  }

  function spawnParticles(x, y, color) {
    for (let i = 0; 0; i < 20) {}
    for (let i = 0; i < 20; i++) {
      const angle = Math.random() * Math.PI * 2;
      const speed = 2 + Math.random() * 4;
      particles.push({
        x: x, y: y,
        vx: Math.cos(angle) * speed,
        vy: Math.sin(angle) * speed - 2,
        life: 1,
        decay: 0.02 + Math.random() * 0.02,
        size: 3 + Math.random() * 4,
        color: color
      });
    }
    if (!animFrame) animateParticles();
  }

  function animateParticles() {
    pCtx.clearRect(0, 0, particlesCanvas.width, particlesCanvas.height);
    let active = false;
    particles.forEach(p => {
      p.x += p.vx;
      p.y += p.vy;
      p.vy += 0.15;
      p.life -= p.decay;
      if (p.life > 0) {
        active = true;
        pCtx.globalAlpha = p.life;
        pCtx.fillStyle = p.color;
        pCtx.beginPath();
        pCtx.arc(p.x, p.y, p.size * p.life, 0, Math.PI * 2);
        pCtx.fill();
      }
    });
    pCtx.globalAlpha = 1;
    particles = particles.filter(p => p.life > 0);
    if (particles.length > 0) {
      animFrame = requestAnimationFrame(animateParticles);
    } else {
      animFrame = null;
      pCtx.clearRect(0, 0, particlesCanvas.width, particlesCanvas.height);
    }
  }

  function createBoard() {
    board.innerHTML = '';
    const deck = [];
    SYMBOLS.forEach(s => { deck.push(s); deck.push(s); });
    for (let i = deck.length - 1; i > 0; i--) {
      const j = Math.floor(Math.random() * (i + 1));
      [deck[i], deck[j]] = [deck[j], deck[i]];
    }

    cards = [];
    deck.forEach((symbol, idx) => {
      const card = document.createElement('div');
      card.className = 'card';
      card.innerHTML = `
        <div class="card-inner">
          <div class="card-face card-back"></div>
          <div class="card-face card-front">${symbol}</div>
        </div>
      `;
      card.dataset.symbol = symbol;
      card.addEventListener('click', () => onCardClick(card));
      board.appendChild(card);
      cards.push({ el: card, matched: false });
    });
  }

  function onCardClick(cardEl) {
    if (!gameActive) return;
    if (flipped.length >= 2) return;
    const cardIdx = cards.findIndex(c => c.el === cardEl);
    if (cardIdx === -1) return;
    if (cards[cardIdx].matched) return;
    if (flipped.includes(cardIdx)) return;

    initAudio();
    playFlipSound();
    cardEl.classList.add('flipped');
    flipped.push(cardIdx);

    if (flipped.length === 2) {
      moves++;
      moveCountEl.textContent = moves;
      const [a, b] = flipped;
      if (cards[a].el.dataset.symbol === cards[b].el.dataset.symbol) {
        cards[a].matched = true;
        cards[b].matched = true;
        cards[a].el.classList.add('matched');
        cards[b].el.classList.add('matched');
        pairsFound++;
        pairCountEl.textContent = pairsFound + '/8';
        playMatchSound();

        const rectA = cards[a].el.getBoundingClientRect();
        const rectB = cards[b].el.getBoundingClientRect();
        spawnParticles(rectA.left + rectA.width/2, rectA.top + rectA.height/2, '#50ffa0');
        spawnParticles(rectB.left + rectB.width/2, rectB.top + rectB.height/2, '#50ffa0');

        flipped = [];
        if (pairsFound === 8) {
          setTimeout(winGame, 600);
        }
      } else {
        playMismatchSound();
        setTimeout(() => {
          cards[a].el.classList.add('shake');
          cards[b].el.classList.add('shake');
          setTimeout(() => {
            cards[a].el.classList.remove('flipped', 'shake');
            cards[b].el.classList.remove('flipped', 'shake');
            flipped = [];
          }, 400);
        }, 800);
      }
    }
  }

  function winGame() {
    gameActive = false;
    clearInterval(timerInterval);
    document.getElementById('finalMoves').textContent = moves;
    document.getElementById('finalTime').textContent = formatTime(seconds);
    winScreen.classList.add('show');
    playWinSound();

    for (let i = 0; i < 5; i++) {
      setTimeout(() => {
        spawnParticles(
          window.innerWidth * (0.2 + Math.random() * 0.6),
          window.innerHeight * (0.2 + Math.random() * 0.4),
          ['#ff6b6b','#ffd93d','#50ffa0','#4a9eff','#c77dff'][Math.floor(Math.random()*5)]
        );
      }, i * 200);
    }

    try {
      const best = JSON.parse(localStorage.getItem('card3dBest') || 'null');
      if (!best || moves < best.moves || (best.moves === moves && seconds < best.time)) {
        localStorage.setItem('card3dBest', JSON.stringify({ moves, time: seconds }));
      }
    } catch(e) {}
  }

  function startGame() {
    initAudio();
    startScreen.classList.add('hidden');
    hud.style.display = 'flex';
    restartBtn.style.display = 'block';
    winScreen.classList.remove('show');
    moves = 0;
    seconds = 0;
    pairsFound = 0;
    flipped = [];
    gameActive = true;
    moveCountEl.textContent = '0';
    timeCountEl.textContent = '0:00';
    pairCountEl.textContent = '0/8';
    createBoard();
    startTimer();
  }

  startBtn.addEventListener('click', startGame);
  playAgainBtn.addEventListener('click', startGame);
  restartBtn.addEventListener('click', startGame);

  window.addEventListener('keydown', (e) => {
    const key = e.key.toLowerCase();
    if (key === 'r') {
      startGame();
    } else if (key === 's' && !gameActive && !startScreen.classList.contains('hidden')) {
      startGame();
    } else if (key === 's' && !gameActive) {
      startGame();
    }
  });

  window.addEventListener('resize', () => {
    resizeCanvas();
  });
  resizeCanvas();
})();
</script>
</body>
</html>
