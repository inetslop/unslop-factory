function wouldCauseMatch(r, c, v) {

return false;
}

function findMatches() {

function applyGravity() {
let moved = false;
for (let c = 0; c < COLS; c++) {
let writeRow = ROWS - 1;
for (let r = ROWS - 1; r >= 0; r--) {
if (grid[r][c] !== null) {
if (writeRow !== r) {
grid[writeRow][c] = grid[r][c];
grid[r][c] = null;
moved = true;
}
writeRow--;
}
}
// Fill top with new gems
for (let r = writeRow; r >= 0; r--) {
grid[r][c] = Math.floor(Math.random() * NUM_COLORS);
moved = true;
}
}
return moved;
}

function spawnParticles(x, y, color) {
const col = COLORS[color || 0];
for (let i = 0; i < 6; i++) {
const angle = Math.random() * Math.PI * 2;
const speed = 1 + Math.random() * 3;
particles.push({ x, y, vx: Math.cos(angle)*speed, vy: Math.sin(angle)*speed - 1, life: 1, color: col });
}
}

function updateParticles(dt) {
for (let i = particles.length - 1; i >= 0; i--) {
const p = particles[i];
p.x += p.vx; p.y += p.vy; p.vy += 0.15;
p.life -= dt * 2;
if (p.life <= 0) particles.splice(i, 1);
}
}

function drawGem(x, y, size, colorIdx, alpha) {

const r = Math.min(size * 0.35, 24);
const cx = x + size/2, cy = y + size/2;
ctx.save();
ctx.globalAlpha = alpha || 1;

// Gem body - hexagonal shape
ctx.beginPath();
for (let i = 0; i < 6; i++) {
const a = (Math.PI / 3) * i - Math.PI / 6;
const px = cx + r * Math.cos(a);
const py = cy + r * Math.sin(a);
if (i === 0) ctx.moveTo(px, py); else ctx.lineTo(px, py);
}
ctx.closePath();

// Gradient fill
const grad = ctx.createRadialGradient(cx - r*0.3, cy - r*0.3, 0, cx, cy, r);
grad.addColorStop(0, `rgba(${col.r+60},${col.g+60},${col.b+60},1)`);
grad.addColorStop(0.7, `rgba(${col.r},${col.g},${col.b},1)`);
grad.addColorStop(1, `rgba(${Math.max(0,col.r-80)},${Math.max(0,col.g-80)},${Math.max(0,col.b-80)},1)`);
ctx.fillStyle = grad;
ctx.fill();

// Highlight
ctx.beginPath();
ctx.arc(cx - r*0.2, cy - r*0.3, r*0.25, 0, Math.PI*2);
ctx.fillStyle = `rgba(255,255,255,0.4)`;
ctx.fill();

// Border

const py = cy + r * Math.sin(a);

ctx.restore();

ctx.fill();
ctx.restore();
});

function isAdjacent(a, b) {
return Math.abs(a.r - b.r) + Math.abs(a.c - b.c) === 1;
}

function handleInput(r, c) {
if (animating || !gameStarted) return;
if (r < 0 || r >= ROWS || c < 0 || c >= COLS) return;

if (!selected) {
selected = { r, c };
} else if (selected.r === r && selected.c === c) {
selected = null;
} else if (isAdjacent(selected, { r, c })) {
attemptSwap(selected, { r, c });
selected = null;
} else {
selected = { r, c };
}
}

function attemptSwap(a, b) {
// Animate swap
swapAnim = { fromR: a.r, fromC: a.c, toR: b.r, toC: b.c, t: 0, duration: 200, done: false };
animating = true;
playSwapSound();

const startTime = performance.now();
function animateSwap(now) {
const elapsed = now - startTime;
swapAnim.t = Math.min(elapsed / swapAnim.duration, 1);
if (swapAnim.t >= 1) {
// Perform actual swap
const tmp = grid[a.r][a.c];
grid[a.r][a.c] = grid[b.r][b.c];
grid[b.r][b.c] = tmp;
swapAnim = null;

// Check if valid match
if (hasMatchAt(a.r, a.c) || hasMatchAt(b.r, b.c)) {
resolveBoard();
} else {
// Swap back
const tmp2 = grid[a.r][a.c];
grid[a.r][a.c] = grid[b.r][b.c];
grid[b.r][b.c] = tmp2;
animating = false;
}
return;
}
requestAnimationFrame(animateSwap);
}
requestAnimationFrame(animateSwap);
}

function resolveBoard() {
cascadeCount = 0;
resolveStep();
}

function resolveStep() {
const matched = findMatches();
if (matched.size === 0) {
animating = false;
return;
}
removeMatches(matched);
applyGravity();
// Brief delay then check again
setTimeout(() => {
if (!gameStarted) return;
cascadeCount++;
playCascadeSound(cascadeCount);
resolveStep();
}, 300);
}

function getCellFromPos(x, y) {

const r = Math.floor((y - boardY) / cellSize);
if (r >= 0 && r < ROWS && c >= 0 && c < COLS) return { r, c };
return null;
}
