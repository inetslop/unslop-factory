const canvas = document.getElementById('game');
const ctx = canvas.getContext('2d');

let W, H;

// Audio
let audioCtx = null;
function initAudio() {
    if (!audioCtx) {
        audioCtx = new (window.AudioContext || window.webkitAudioContext)();
    }
}
function playSound(type) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    if (type === 'thrust') {
        osc.type = 'sawtooth';
        osc.frequency.value = 80;
        gain.gain.setValueAtTime(0.05, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.1);
        osc.start();
        osc.stop(audioCtx.currentTime + 0.1);
    } else if (type === 'alert') {
        osc.type = 'square';
        osc.frequency.value = 440;
        gain.gain.setValueAtTime(0.1, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.3);
        osc.start();
        osc.stop(audioCtx.currentTime + 0.3);
    } else if (type === 'explosion') {
        osc.type = 'sawtooth';
        osc.frequency.value = 60;
        gain.gain.setValueAtTime(0.2, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.5);
        osc.start();
        osc.stop(audioCtx.currentTime + 0.5);
    } else if (type === 'win') {
        osc.type = 'sine';
        osc.frequency.value = 660;
        gain.gain.setValueAtTime(0.15, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.8);
        osc.start();
        osc.stop(audioCtx.currentTime + 0.8);
    }
}

// Game state
let gameState = 'start'; // start, playing, won, lost
let keys = {};
let frameCount = 0;

window.addEventListener('keydown', e => {
    keys[e.code] = true;
    initAudio();
    if (gameState === 'start' && (e.code === 'Space' || e.code === 'Enter')) startGame();
    if ((gameState === 'won' || gameState === 'lost') && (e.code === 'Space' || e.code === 'Enter')) startGame();
});
window.addEventListener('keyup', e => { keys[e.code] = false; });

// Stars
let stars = [];
function genStars() {
    stars = [];
    for (let i = 0; i < 200; i++) {
        stars.push({ x: Math.random() * W, y: Math.random() * H, s: Math.random() * 1.5 + 0.5, b: Math.random() });
    }
}

// Game objects
let player, asteroids, enemies, extractionPoint;
let stealthMeter, heatSignature, gameTime;

function startGame() {
    gameState = 'playing';
    genStars();
    stealthMeter = 0;
    heatSignature = 0;
    gameTime = 0;
    
    // Player
    player = {
        x: W * 0.1,
        y: H * 0.5,
        vx: 0,
        vy: 0,
        angle: -Math.PI / 2,
        thrusting: false,
        alive: true
    };

    // Extraction point
    extractionPoint = { x: W * 0.9, y: H * 0.5, radius: 40 };

    // Asteroids
    asteroids = [];
    const numAsteroids = 18;
    for (let i = 0; i < numAsteroids; i++) {
        let ax, ay, tries = 0;
        do {
            ax = W * 0.25 + Math.random() * W * 0.6;
            ay = H * 0.1 + Math.random() * H * 0.8;
            tries++;
        } while (tries < 50 && asteroids.some(a => dist(ax, ay, a.x, a.y) < a.radius * 2));
        
        const radius = 25 + Math.random() * 40;
        const verts = [];
        const numVerts = 6 + Math.floor(Math.random() * 4);
        for (let v = 0; v < numVerts; v++) {
            const a = (v / numVerts) * Math.PI * 2;
            const r = radius * (0.7 + Math.random() * 0.3);
            verts.push({ x: Math.cos(a) * r, y: Math.sin(a) * r });
        }
        asteroids.push({ x: ax, y: ay, radius, verts, rot: Math.random() * Math.PI * 2, rotSpeed: (Math.random() - 0.5) * 0.01 });
    }

    // Enemies
    enemies = [];
    const numEnemies = 4;
    for (let i = 0; i < numEnemies; i++) {
        const ex = W * 0.35 + Math.random() * W * 0.45;
        const ey = H * 0.15 + Math.random() * H * 0.7;
        enemies.push({
            x: ex, y: ey,
            angle: Math.random() * Math.PI * 2,
            patrolAngle: Math.random() * Math.PI * 2,
            speed: 0.3 + Math.random() * 0.4,
            radarRange: 180 + Math.random() * 80,
            radarArc: Math.PI / 3.5,
            detecting: false,
            firing: false,
            fireTimer: 0
        });
    }
}

function dist(x1, y1, x2, y2) {
    return Math.sqrt((x2 - x1) * (x2 - x1) + (y2 - y1) * (y2 - y1));
}

function pointInPolygon(px, py, cx, cy, verts, rot) {
    let inside = false;
    const n = verts.length;
    for (let i = 0; i < n; i++) {
        const j = (i + 1) % n;
        const vi = rotatePoint(verts[i], rot);
        const vj = rotatePoint(verts[j], rot);
        const xi = cx + vi.x, yi = cy + vi.y;
        const xj = cx + vj.x, yj = cy + vj.y;
        if (((yi > py) !== (yj > py)) && (px < (xj - xi) * (py - yi) / (yj - yi) + xi)) {
            inside = !inside;
        }
    }
    return inside;
}

function rotatePoint(p, a) {
    const c = Math.cos(a), s = Math.sin(a);
    return { x: p.x * c - p.y * s, y: p.x * s + p.y * c };
}

function lineIntersectsAsteroid(x1, y1, x2, y2, ast) {
    // Check if line segment from (x1,y1) to (x2,y2) intersects the asteroid polygon
    // Simplified: check against bounding circle first
    const dx = x2 - x1, dy = y2 - y1;
    const len = Math.sqrt(dx * dx + dy * dy);
    if (len === 0) return false;
    
    // Check distance from asteroid center to line segment
    const cx = ast.x, cy = ast.y;
    const t = Math.max(0, Math.min(1, ((cx - x1) * dx + (cy - y1) * dy) / (len * len)));
    const px = x1 + t * dx, py = y1 + t * dy;
    const d = dist(px, py, cx, cy);
    
    if (d < ast.radius) {
        // More precise: check if line passes through polygon
        return true;
    }
    return false;
}

function hasLineOfSight(x1, y1, x2, y2) {
    for (const ast of asteroids) {
        const dx = x2 - x1, dy = y2 - y1;
        const len = Math.sqrt(dx * dx + dy * dy);
        if (len === 0) continue;
        
        // Check distance from asteroid center to line segment
        const cx = ast.x, cy = ast.y;
        const t = Math.max(0, Math.min(1, ((cx - x1) * dx + (cy - y1) * dy) / (len * len)));
        const px = x1 + t * dx, py = y1 + t * dy;
        const d = dist(px, py, cx, cy);
        
        if (d < ast.radius * 0.7) return false; // blocked!
    }
    return true;
}

function update() {
    if (gameState !== 'playing') return;
    
    gameTime++;
    
    // Player movement
    const thrustPower = 0.15;
    const friction = 0.985;
    
    let turning = false;
    if (keys['ArrowLeft'] || keys['KeyA']) {
        player.angle -= 0.06;
        turning = true;
    }
    if (keys['ArrowRight'] || keys['KeyD']) {
        player.angle += 0.06;
        turning = true;
    }
    
    player.thrusting = false;
    if (keys['ArrowUp'] || keys['KeyW']) {
        player.vx += Math.cos(player.angle) * thrustPower;
        player.vy += Math.sin(player.angle) * thrustPower;
        player.thrusting = true;
    }
    
    // Brake
    if (keys['ArrowDown'] || keys['KeyS']) {
        player.vx *= 0.92;
        player.vy *= 0.92;
    }
    
    player.vx *= friction;
    player.vy *= friction;
    player.x += player.vx;
    player.y += player.vy;
    
    // Wrap around screen
    if (player.x < 0) player.x = W;
    if (player.x > W) player.x = 0;
    if (player.y < 0) player.y = H;
    if (player.y > H) player.y = 0;
    
    // Heat signature
    const speed = Math.sqrt(player.vx * player.vx + player.vy * player.vy);
    heatSignature = Math.min(1, speed / 3);
    
    // Stealth detection
    let detected = false;
    for (const enemy of enemies) {
        // Update enemy patrol
        enemy.patrolAngle += (Math.random() - 0.5) * 0.02;
        enemy.x += Math.cos(enemy.patrolAngle) * enemy.speed;
        enemy.y += Math.sin(enemy.patrolAngle) * enemy.speed;
        
        // Wrap
        if (enemy.x < -50) enemy.x = W + 50;
        if (enemy.x > W + 50) enemy.x = -50;
        if (enemy.y < -50) enemy.y = H + 50;
        if (enemy.y > H + 50) enemy.y = -50;
        
        // Face movement direction
        const targetAngle = enemy.patrolAngle;
        let da = targetAngle - enemy.angle;
        while (da > Math.PI) da -= Math.PI * 2;
        while (da < -Math.PI) da += Math.PI * 2;
        enemy.angle += da * 0.05;
        
        // Check if player is in radar cone
        const d = dist(enemy.x, enemy.y, player.x, player.y);
        const angleToPlayer = Math.atan2(player.y - enemy.y, player.x - enemy.x);
        let radarDiff = angleToPlayer - enemy.angle;
        while (radarDiff > Math.PI) radarDiff -= Math.PI * 2;
        while (radarDiff < -Math.PI) radarDiff += Math.PI * 2;
        
        if (d < enemy.radarRange && Math.abs(radarDiff) < enemy.radarArc / 2) {
            if (hasLineOfSight(enemy.x, enemy.y, player.x, player.y)) {
                detected = true;
                enemy.detecting = true;
                
                // Stealth increases based on distance and heat
                const proximityFactor = 1 - d / enemy.radarRange;
                const heatFactor = 0.3 + heatSignature * 0.7;
                stealthMeter += (0.5 + proximityFactor * 1.5) * heatFactor * 0.02;
            } else {
                enemy.detecting = false;
            }
        } else {
            enemy.detecting = false;
        }
    }
    
    // Stealth decay when not detected
    if (!detected) {
        stealthMeter -= 0.015;
    }
    stealthMeter = Math.max(0, Math.min(1, stealthMeter));
    
    // If stealth maxed out, enemies fire
    if (stealthMeter >= 1) {
        for (const enemy of enemies) {
            const d = dist(enemy.x, enemy.y, player.x, player.y);
            if (d < enemy.radarRange * 1.5 && !enemy.firing) {
                enemy.firing = true;
                playSound('explosion');
            }
        }
        // Player is lost
        gameState = 'lost';
        return;
    }
    
    // Check win condition
    if (dist(player.x, player.y, extractionPoint.x, extractionPoint.y) < extractionPoint.radius) {
        gameState = 'won';
        playSound('win');
    }
}

function draw() {
    ctx.fillStyle = '#0a0a12';
    ctx.fillRect(0, 0, W, H);
    
    // Stars
    for (const s of stars) {
        const twinkle = 0.5 + 0.5 * Math.sin(frameCount * 0.02 + s.b * 10);
        ctx.fillStyle = `rgba(200, 220, 255, ${twinkle * 0.6})`;
        ctx.fillRect(s.x, s.y, s.s, s.s);
    }
    
    if (gameState === 'start') {
        drawStartScreen();
        return;
    }
    
    // Extraction point
    ctx.strokeStyle = '#4fff4f';
    ctx.lineWidth = 2;
    ctx.setLineDash([5, 5]);
    ctx.beginPath();
    ctx.arc(extractionPoint.x, extractionPoint.y, extractionPoint.radius, 0, Math.PI * 2);
    ctx.stroke();
    ctx.setLineDash([]);
    ctx.fillStyle = 'rgba(79, 255, 79, 0.1)';
    ctx.fill();
    ctx.fillStyle = '#4fff4f';
    ctx.font = '12px monospace';
    ctx.textAlign = 'center';
    ctx.fillText('EXTRACTION', extractionPoint.x, extractionPoint.y + 4);
    
    // Asteroids
    for (const ast of asteroids) {
        ctx.strokeStyle = '#556677';
        ctx.lineWidth = 1.5;
        ctx.beginPath();
        const n = ast.verts.length;
        for (let i = 0; i < n; i++) {
            const p = rotatePoint(ast.verts[i], ast.rot);
            if (i === 0) ctx.moveTo(p.x, p.y);
            else ctx.lineTo(p.x, p.y);
        }
        ctx.closePath();
        ctx.stroke();
        ctx.fillStyle = 'rgba(40, 50, 60, 0.3)';
        ctx.fill();
    }
    
    // Enemies
    for (const enemy of enemies) {
        // Radar cone
        ctx.save();
        ctx.translate(enemy.x, enemy.y);
        ctx.rotate(enemy.angle);
        
        const gradient = ctx.createRadialGradient(0, 0, 0, 0, 0, enemy.radarRange);
        if (enemy.detecting) {
            gradient.addColorStop(0, 'rgba(255, 100, 50, 0.15)');
            gradient.addColorStop(1, 'rgba(255, 100, 50, 0)');
        } else {
            gradient.addColorStop(0, 'rgba(255, 50, 50, 0.08)');
            gradient.addColorStop(1, 'rgba(255, 50, 50, 0)');
        }
        
        ctx.fillStyle = gradient;
        ctx.beginPath();
        ctx.moveTo(0, 0);
        ctx.arc(0, 0, enemy.radarRange, -enemy.radarArc / 2, enemy.radarArc / 2);
        ctx.closePath();
        ctx.fill();
        
        // Enemy ship (diamond shape)
        ctx.strokeStyle = '#ff4444';
        ctx.lineWidth = 1.5;
        ctx.beginPath();
        ctx.moveTo(12, 0);
        ctx.lineTo(-8, -7);
        ctx.lineTo(-4, 0);
        ctx.lineTo(-8, 7);
        ctx.closePath();
        ctx.stroke();
        
        ctx.restore();
        
        // Detection indicator
        if (enemy.detecting) {
            ctx.fillStyle = '#ff6644';
            ctx.font = '10px monospace';
            ctx.textAlign = 'center';
            ctx.fillText('!!', enemy.x, enemy.y - 15);
        }
    }
    
    // Player
    if (player.alive && gameState === 'playing') {
        ctx.save();
        ctx.translate(player.x, player.y);
        ctx.rotate(player.angle);
        
        ctx.strokeStyle = '#44ff88';
        ctx.lineWidth = 2;
        ctx.beginPath();
        ctx.moveTo(14, 0);
        ctx.lineTo(-10, -9);
        ctx.lineTo(-6, 0);
        ctx.lineTo(-10, 9);
        ctx.closePath();
        ctx.stroke();
        
        // Thrust flame
        if (player.thrusting) {
            ctx.strokeStyle = '#ffaa33';
            ctx.lineWidth = 1.5;
            const flicker = Math.random() * 4;
            ctx.beginPath();
            ctx.moveTo(-8, -3);
            ctx.lineTo(-16 - flicker, 0);
            ctx.lineTo(-8, 3);
            ctx.stroke();
        }
        
        ctx.restore();
    }
    
    // HUD
    drawHUD();
    
    if (gameState === 'won') drawEndScreen('MISSION COMPLETE', '#4fff4f');
    if (gameState === 'lost') drawEndScreen('DETECTED & DESTROYED', '#ff4444');
}

function drawHUD() {
    // Stealth meter
    const barW = 200, barH = 16;
    const bx = W / 2 - barW / 2, by = 20;
    
    ctx.fillStyle = 'rgba(0,0,0,0.5)';
    ctx.fillRect(bx - 4, by - 4, barW + 8, barH + 8);
    
    const fillColor = stealthMeter < 0.5 ? '#44ff88' : stealthMeter < 0.8 ? '#ffaa33' : '#ff4444';
    ctx.fillStyle = 'rgba(40, 40, 60, 0.8)';
    ctx.fillRect(bx, by, barW, barH);
    ctx.fillStyle = fillColor;
    ctx.fillRect(bx, by, barW * stealthMeter, barH);
    
    ctx.strokeStyle = '#8899aa';
    ctx.lineWidth = 1;
    ctx.strokeRect(bx, by, barW, barH);
    
    ctx.fillStyle = '#ccddee';
    ctx.font = '11px monospace';
    ctx.textAlign = 'center';
    ctx.fillText('STEALTH', W / 2, by + barH + 14);
    
    // Heat indicator
    const hx = bx + barW + 20, hy = by;
    ctx.fillStyle = '#ccddee';
    ctx.font = '11px monospace';
    ctx.textAlign = 'left';
    ctx.fillText('HEAT: ' + Math.round(heatSignature * 100) + '%', hx, hy + barH - 2);
    
    // Speed
    const spd = Math.sqrt(player.vx * player.vx + player.vy * player.vy).toFixed(1);
    ctx.fillText('VEL: ' + spd, hx, hy + barH + 14);
}

function drawStartScreen() {
    ctx.fillStyle = '#aaddff';
    ctx.font = 'bold 36px monospace';
    ctx.textAlign = 'center';
    ctx.fillText('ASTEROIDS: STEALTH', W / 2, H * 0.25);
    
    ctx.font = '14px monospace';
    ctx.fillStyle = '#8899aa';
    const lines = [
        'Navigate through the asteroid field to reach EXTRACTION.',
        '',
        'ENEMIES patrol with radar cones.',
        'Stay in their line of sight and your STEALTH meter rises.',
        'Using engines increases your HEAT SIGNATURE (more detectable).',
        'Asteroids block enemy line of sight - use them for cover!',
        '',
        'If stealth reaches 100%, you are detected and destroyed.',
    ];
    lines.forEach((line, i) => {
        ctx.fillText(line, W / 2, H * 0.35 + i * 22);
    });
    
    ctx.fillStyle = '#44ff88';
    ctx.font = '16px monospace';
    const pulse = 0.5 + 0.5 * Math.sin(frameCount * 0.05);
    ctx.globalAlpha = pulse;
    ctx.fillText('PRESS SPACE TO BEGIN', W / 2, H * 0.78);
    ctx.globalAlpha = 1;
    
    ctx.fillStyle = '#667788';
    ctx.font = '12px monospace';
    ctx.fillText('W/↑: Thrust | A/D or ←/→: Turn | S/↓: Brake', W / 2, H * 0.85);
}

function drawEndScreen(text, color) {
    ctx.fillStyle = 'rgba(0,0,0,0.7)';
    ctx.fillRect(0, 0, W, H);
    
    ctx.fillStyle = color;
    ctx.font = 'bold 32px monospace';
    ctx.textAlign = 'center';
    ctx.fillText(text, W / 2, H * 0.4);
    
    ctx.fillStyle = '#8899aa';
    ctx.font = '16px monospace';
    const pulse = 0.5 + 0.5 * Math.sin(frameCount * 0.05);
    ctx.globalAlpha = pulse;
    ctx.fillText('PRESS SPACE TO RESTART', W / 2, H * 0.6);
    ctx.globalAlpha = 1;
}

function gameLoop() {
    frameCount++;
    update();
    draw();
    requestAnimationFrame(gameLoop);
}

genStars();
gameLoop();
