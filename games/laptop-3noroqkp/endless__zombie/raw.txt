const canvas = document.getElementById('game');
const ctx = canvas.getContext('2d');

let W, H;
function resize() {
    W = canvas.width = window.innerWidth;
    H = canvas.height = window.innerHeight;

window.addEventListener('resize', resize);

// Audio
let audioCtx = null;
function initAudio() {
    if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
}
function playSound(type) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    
    if (type === 'shoot') {
        osc.type = 'square';
        osc.frequency.setValueAtTime(200, audioCtx.currentTime);
        osc.frequency.exponentialRampToValueAtTime(80, audioCtx.currentTime + 0.1);
        gain.gain.setValueAtTime(0.3, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.1);
        osc.start(audioCtx.currentTime);
        osc.stop(audioCtx.currentTime + 0.1);
    } else if (type === 'hit') {
        osc.type = 'sawtooth';
        osc.frequency.setValueAtTime(150, audioCtx.currentTime);
        osc.frequency.exponentialRampToValueAtTime(50, audioCtx.currentTime + 0.2);
        gain.gain.setValueAtTime(0.4, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.2);
        osc.start(audioCtx.currentTime);
        osc.stop(audioCtx.currentTime + 0.2);
    } else if (type === 'kill') {
        osc.type = 'triangle';
        osc.frequency.setValueAtTime(400, audioCtx.currentTime);
        osc.frequency.exponentialRampToValueAtTime(600, audioCtx.currentTime + 0.15);
        gain.gain.setValueAtTime(0.25, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.15);
        osc.start(audioCtx.currentTime);
        osc.stop(audioCtx.currentTime + 0.15);
    } else if (type === 'jump') {
        osc.type = 'sine';
        osc.frequency.setValueAtTime(300, audioCtx.currentTime);
        osc.frequency.exponentialRampToValueAtTime(600, audioCtx.currentTime + 0.1);
        gain.gain.setValueAtTime(0.2, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.1);
        osc.start(audioCtx.currentTime);
        osc.stop(audioCtx.currentTime + 0.1);
    } else if (type === 'gameover') {
        osc.type = 'sawtooth';
        osc.frequency.setValueAtTime(300, audioCtx.currentTime);
        osc.frequency.exponentialRampToValueAtTime(50, audioCtx.currentTime + 0.8);
        gain.gain.setValueAtTime(0.4, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.8);
        osc.start(audioCtx.currentTime);
        osc.stop(audioCtx.currentTime + 0.8);
    } else if (type === 'powerup') {
        osc.type = 'sine';
        osc.frequency.setValueAtTime(500, audioCtx.currentTime);
        osc.frequency.exponentialRampToValueAtTime(1000, audioCtx.currentTime + 0.2);
        gain.gain.setValueAtTime(0.3, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.2);
        osc.start(audioCtx.currentTime);
        osc.stop(audioCtx.currentTime + 0.2);
    }
}

// Game State
let gameState = 'start'; // start, playing, gameover
let score = 0;
let highScore = 0;
try { highScore = parseInt(localStorage.getItem('zombieEndlessHigh') || '0'); } catch(e) {}
let frameCount = 0;
let speed = 4;
let baseSpeed = 4;

// Player
const player = {
    x: 0, y: 0, w: 40, h: 50,
    vy: 0, grounded: true,
    health: 3, maxHealth: 3,
    invincible: 0,
    shooting: false, shootCooldown: 0,
    ducking: false,
    animFrame: 0
};

// Arrays
let zombies = [];
let bullets = [];
let particles = [];
let powerups = [];
let groundObstacles = [];
let bgBuildings = [];
let bgClouds = [];
let stars = [];

const GROUND_Y_RATIO = 0.8;
function groundY() { return H * GROUND_Y_RATIO; }

// Input
const keys = {};
window.addEventListener('keydown', e => {
    keys[e.code] = true;
    if (gameState === 'start' && (e.code === 'Space' || e.code === 'Enter')) startGame();
    else if (gameState === 'gameover' && (e.code === 'Space' || e.code === 'Enter' || e.code === 'KeyR')) startGame();
});
window.addEventListener('keyup', e => { keys[e.code] = false; });

function initBackground() {
    bgBuildings = [];
    for (let i = 0; i < 12; i++) {
        bgBuildings.push({
            x: Math.random() * W,
            w: 60 + Math.random() * 100,
            h: 80 + Math.random() * 150,
            color: `hsl(${240 + Math.random()*30}, ${10+Math.random()*10}%, ${15+Math.random()*10}%)`
        });
    }
    bgClouds = [];
    for (let i = 0; i < 8; i++) {
        bgClouds.push({ x: Math.random() * W, y: 30 + Math.random() * H * 0.2, w: 60 + Math.random() * 80 });
    }
    stars = [];
    for (let i = 0; i < 50; i++) {
        stars.push({ x: Math.random() * W, y: Math.random() * H * 0.4, size: Math.random() * 2 });
    }
}

function startGame() {
    initAudio();
    gameState = 'playing';
    score = 0;
    speed = baseSpeed;
    frameCount = 0;
    player.x = W * 0.15;
    player.y = groundY() - player.h;
    player.vy = 0;
    player.grounded = true;
    player.health = 3;
    player.invincible = 0;
    player.shootCooldown = 0;
    player.ducking = false;
    zombies = [];
    bullets = [];
    particles = [];
    powerups = [];
    groundObstacles = [];
    initBackground();
}

function spawnZombie() {
    const type = Math.random();
    let z = { x: W + 50, y: 0, w: 35, h: 50, vx: -speed * 0.7, hp: 1, type: 'walker', animFrame: 0 };
    
    if (type < 0.4) {
        z.type = 'walker';
        z.y = groundY() - z.h;
        z.vx = -speed * 0.6;
        z.hp = 1;
    } else if (type < 0.7) {
        z.type = 'runner';
        z.y = groundY() - z.h;
        z.vx = -speed * 1.2;
        z.hp = 1;
        z.w = 30;
        z.h = 45;
    } else if (type < 0.85) {
        z.type = 'crawler';
        z.y = groundY() - 25;
        z.vx = -speed * 0.9;
        z.hp = 1;
        z.w = 45;
        z.h = 25;
    } else if (type < 0.95) {
        z.type = 'tank';
        z.y = groundY() - z.h;
        z.vx = -speed * 0.4;
        z.hp = 3;
        z.w = 50;
        z.h = 60;
    } else {
        z.type = 'flier';
        z.y = groundY() - player.h - 30 - Math.random() * 40;
        z.vx = -speed * 0.8;
        z.hp = 1;
        z.w = 35;
        z.h = 30;
    }
    
    zombies.push(z);
}

function spawnPowerup() {
    const type = Math.random() < 0.7 ? 'health' : 'rapid';
    powerups.push({
        x: W + 30,
        y: groundY() - 60 - Math.random() * 40,
        w: 25, h: 25, type, collected: false
    });
}

function spawnParticles(x, y, color, count) {
    for (let i = 0; i < count; i++) {
        particles.push({
            x, y,
            vx: (Math.random() - 0.5) * 6,
            vy: (Math.random() - 0.5) * 6 - 2,
            life: 30 + Math.random() * 20,
            maxLife: 50,
            color, size: 3 + Math.random() * 4
        });
    }
}

function update() {
    if (gameState !== 'playing') return;
    
    frameCount++;
    score += speed * 0.1;
    speed = baseSpeed + Math.floor(score / 100) * 0.3;
    
    // Player physics
    if (!player.grounded) {
        player.vy += 0.7;
        player.y += player.vy;
        if (player.y >= groundY() - player.h) {
            player.y = groundY() - player.h;
            player.vy = 0;
            player.grounded = true;
        }
    }
    
    // Jump
    if ((keys['Space'] || keys['ArrowUp'] || keys['KeyW']) && player.grounded) {
        player.vy = -14;
        player.grounded = false;
        playSound('jump');
    }
    
    // Duck
    player.ducking = (keys['ArrowDown'] || keys['KeyS']);
    if (player.ducking && player.grounded) {
        player.h = 30;
        player.y = groundY() - player.h;
    } else {
        player.h = 50;
        if (player.grounded) player.y = groundY() - player.h;
    }
    
    // Shoot
    if ((keys['KeyZ'] || keys['KeyX'] || keys['KeyF']) && player.shootCooldown <= 0) {
        bullets.push({ x: player.x + player.w, y: player.y + player.h/2 - 5, w: 8, h: 4, vx: 12 });
        player.shootCooldown = 12;
        playSound('shoot');
    }
    if (player.shootCooldown > 0) player.shootCooldown--;
    
    // Spawn zombies
    const spawnRate = Math.max(30, 80 - Math.floor(score / 50));
    if (frameCount % spawnRate === 0) spawnZombie();
    
    // Spawn powerups
    if (frameCount % 600 === 0) spawnPowerup();
    
    // Update bullets
    for (let i = bullets.length - 1; i >= 0; i--) {
        bullets[i].x += bullets[i].vx;
        if (bullets[i].x > W + 20) bullets.splice(i, 1);
    }
    
    // Update zombies
    for (let i = zombies.length - 1; i >= 0; i--) {
        const z = zombies[i];
        z.x += z.vx;
        z.animFrame++;
        
        if (z.type === 'flier') {
            z.y += Math.sin(z.animFrame * 0.1) * 1.5;
        }
        
        // Bullet collision
        for (let j = bullets.length - 1; j >= 0; j--) {
            const b = bullets[j];
            if (b.x < z.x + z.w && b.x + b.w > z.x && b.y < z.y + z.h && b.y + b.h > z.y) {
                z.hp--;
                bullets.splice(j, 1);
                spawnParticles(z.x + z.w/2, z.y + z.h/2, '#4a7', 5);
                playSound('hit');
                if (z.hp <= 0) {
                    zombies.splice(i, 1);
                    score += 10;
                    spawnParticles(z.x + z.w/2, z.y + z.h/2, '#f44', 10);
                    playSound('kill');
                }
                break;
            }
        }
        
        // Player collision
        if (player.invincible <= 0) {
            if (player.x < z.x + z.w && player.x + player.w > z.x &&
                player.y < z.y + z.h && player.y + player.h > z.y) {
                player.health--;
                player.invincible = 60;
                spawnParticles(player.x + player.w/2, player.y + player.h/2, '#f44', 8);
                playSound('hit');
                if (player.health <= 0) {
                    gameState = 'gameover';
                    playSound('gameover');
                    try { localStorage.setItem('zombieEndlessHigh', Math.floor(score)); } catch(e) {}
                }
            }
        }
        
        if (z.x < -100) zombies.splice(i, 1);
    }
    
    // Update powerups
    for (let i = powerups.length - 1; i >= 0; i--) {
        const p = powerups[i];
        p.x -= speed;
        if (p.x < -50) { powerups.splice(i, 1); continue; }
        
        if (player.x < p.x + p.w && player.x + player.w > p.x &&
            player.y < p.y + p.h && player.y + player.h > p.y) {
            if (p.type === 'health' && player.health < player.maxHealth) {
                player.health++;
            } else if (p.type === 'rapid') {
                player.shootCooldown = 0; // rapid fire burst
                for (let k = 0; k < 5; k++) {
                    bullets.push({ x: player.x + player.w, y: player.y + player.h/2 - 5, w: 8, h: 4, vx: 12 + k * 2 });
                }
            }
            playSound('powerup');
            spawnParticles(p.x, p.y, '#ff0', 8);
            powerups.splice(i, 1);
        }
    }
    
    // Update particles
    for (let i = particles.length - 1; i >= 0; i--) {
        const p = particles[i];
        p.x += p.vx;
        p.y += p.vy;
        p.vy += 0.3;
        p.life--;
        if (p.life <= 0) particles.splice(i, 1);
    }
    
    // Background scroll
    for (let b of bgBuildings) { b.x -= speed * 0.2; if (b.x + b.w < 0) b.x = W + Math.random() * 100; }
    for (let c of bgClouds) { c.x -= speed * 0.1; if (c.x + c.w < 0) c.x = W + Math.random() * 200; }
    
    if (player.invincible > 0) player.invincible--;
    player.animFrame++;
}

function drawPlayer() {
    ctx.save();
    if (player.invincible > 0 && Math.floor(player.invincible / 5) % 2 === 0) {
        ctx.globalAlpha = 0.4;
    }
    
    const px = player.x, py = player.y;
    const legAnim = Math.sin(player.animFrame * 0.3) * (player.grounded ? 8 : 3);
    
    // Body
    ctx.fillStyle = '#2196F3';
    ctx.fillRect(px + 10, py + 15, 20, player.h - 30);
    
    // Head
    ctx.fillStyle = '#ffccbc';
    ctx.beginPath();
    ctx.arc(px + 20, py + 10, 10, 0, Math.PI * 2);
    ctx.fill();
    
    // Eyes
    ctx.fillStyle = '#333';
    ctx.fillRect(px + 18, py + 7, 3, 3);
    ctx.fillRect(px + 25, py + 7, 3, 3);
    
    // Legs
    ctx.fillStyle = '#1565C0';
    ctx.save();
    ctx.translate(px + 14, py + player.h - 15);
    ctx.rotate(legAnim * 0.08);
    ctx.fillRect(-3, 0, 6, 15);
    ctx.restore();
    ctx.save();
    ctx.translate(px + 28, py + player.h - 15);
    ctx.rotate(-legAnim * 0.08);
    ctx.fillRect(-3, 0, 6, 15);
    ctx.restore();
    
    // Gun arm
    ctx.fillStyle = '#333';
    ctx.fillRect(px + player.w - 5, py + 20, 15, 4);
    
    ctx.restore();
}

function drawZombie(z) {
    ctx.save();
    const anim = Math.sin(z.animFrame * 0.15) * 3;
    
    if (z.type === 'walker' || z.type === 'runner') {
        // Body
        ctx.fillStyle = z.type === 'runner' ? '#7b2d2d' : '#4a7a4a';
        ctx.fillRect(z.x + 5, z.y + 12, z.w - 10, z.h - 25);
        
        // Head
        ctx.fillStyle = '#8faa8f';
        ctx.beginPath();
        ctx.arc(z.x + z.w/2, z.y + 10, 9, 0, Math.PI * 2);
        ctx.fill();
        
        // Eyes (glowing)
        ctx.fillStyle = z.type === 'runner' ? '#ff4444' : '#ffff00';
        ctx.fillRect(z.x + z.w/2 - 5, z.y + 7, 3, 3);
        ctx.fillRect(z.x + z.w/2 + 3, z.y + 7, 3, 3);
        
        // Arms (reaching)
        ctx.fillStyle = '#8faa8f';
        ctx.save();
        ctx.translate(z.x + 5, z.y + 20);
        ctx.rotate(anim * 0.1);
        ctx.fillRect(-15, -2, 15, 4);
        ctx.restore();
        
        // Legs
        ctx.fillStyle = '#3a5a3a';
        ctx.save();
        ctx.translate(z.x + z.w/2 - 6, z.y + z.h - 12);
        ctx.rotate(anim * 0.15);
        ctx.fillRect(-3, 0, 5, 12);
        ctx.restore();
        ctx.save();
        ctx.translate(z.x + z.w/2 + 6, z.y + z.h - 12);
        ctx.rotate(-anim * 0.15);
        ctx.fillRect(-3, 0, 5, 12);
        ctx.restore();
        
    } else if (z.type === 'crawler') {
        // Crawler - low and wide
        ctx.fillStyle = '#5a7a5a';
        ctx.fillRect(z.x, z.y + 5, z.w, z.h - 5);
        // Head at front
        ctx.fillStyle = '#8faa8f';
        ctx.beginPath();
        ctx.arc(z.x + 5, z.y + z.h/2, 8, 0, Math.PI * 2);
        ctx.fill();
        ctx.fillStyle = '#ffff00';
        ctx.fillRect(z.x + 1, z.y + z.h/2 - 3, 3, 3);
        // Crawling legs
        for (let i = 0; i < 4; i++) {
            const lx = z.x + 10 + i * 10;
            ctx.fillStyle = '#4a6a4a';
            ctx.fillRect(lx, z.y + z.h - 5, 3, 8);
        }
        
    } else if (z.type === 'tank') {
        // Big tank zombie
        ctx.fillStyle = '#3d5c3d';
        ctx.fillRect(z.x, z.y + 10, z.w, z.h - 20);
        ctx.fillStyle = '#6b8f6b';
        ctx.beginPath();
        ctx.arc(z.x + z.w/2, z.y + 12, 14, 0, Math.PI * 2);
        ctx.fill();
        ctx.fillStyle = '#ff2222';
        ctx.fillRect(z.x + z.w/2 - 8, z.y + 8, 5, 4);
        ctx.fillRect(z.x + z.w/2 + 4, z.y + 8, 5, 4);
        // HP bar
        ctx.fillStyle = '#333';
        ctx.fillRect(z.x, z.y - 10, z.w, 6);
        ctx.fillStyle = '#f44';
        ctx.fillRect(z.x, z.y - 10, z.w * (z.hp / 3), 6);
        
    } else if (z.type === 'flier') {
        // Flying bat/zombie
        const wingFlap = Math.sin(z.animFrame * 0.4) * 10;
        ctx.fillStyle = '#5a3a6b';
        ctx.beginPath();
        ctx.arc(z.x + z.w/2, z.y + z.h/2, 10, 0, Math.PI * 2);
        ctx.fill();
        // Wings
        ctx.fillStyle = '#4a2a5a';
        ctx.save();
        ctx.translate(z.x + z.w/2 - 5, z.y + z.h/2);
        ctx.rotate(-wingFlap * 0.05);
        ctx.fillRect(-20, -3, 20, 6);
        ctx.restore();
        ctx.save();
        ctx.translate(z.x + z.w/2 + 5, z.y + z.h/2);
        ctx.rotate(wingFlap * 0.05);
        ctx.fillRect(0, -3, 20, 6);
        ctx.restore();
        // Eyes
        ctx.fillStyle = '#ff0';
        ctx.fillRect(z.x + z.w/2 - 4, z.y + z.h/2 - 2, 3, 3);
        ctx.fillRect(z.x + z.w/2 + 2, z.y + z.h/2 - 2, 3, 3);
    }
    
    ctx.restore();
}

function drawBackground() {
    // Sky gradient
    const grad = ctx.createLinearGradient(0, 0, 0, H);
    grad.addColorStop(0, '#0a0a1a');
    grad.addColorStop(0.5, '#1a1a3e');
    grad.addColorStop(1, '#2a2a4e');
    ctx.fillStyle = grad;
    ctx.fillRect(0, 0, W, H);
    
    // Stars
    ctx.fillStyle = '#fff';
    for (let s of stars) {
        ctx.globalAlpha = 0.5 + Math.sin(frameCount * 0.02 + s.x) * 0.3;
        ctx.fillRect(s.x, s.y, s.size, s.size);
    }
    ctx.globalAlpha = 1;
    
    // Moon
    ctx.fillStyle = '#e8e4d4';
    ctx.beginPath();
    ctx.arc(W * 0.85, H * 0.12, 30, 0, Math.PI * 2);
    ctx.fill();
    ctx.fillStyle = '#0a0a1a';
    ctx.beginPath();
    ctx.arc(W * 0.85 - 10, H * 0.12 - 5, 25, 0, Math.PI * 2);
    ctx.fill();
    
    // Clouds
    ctx.fillStyle = 'rgba(100, 100, 140, 0.3)';
    for (let c of bgClouds) {
        ctx.beginPath();
        ctx.ellipse(c.x, c.y, c.w/2, 15, 0, 0, Math.PI * 2);
        ctx.fill();
    }
    
    // Buildings silhouette
    for (let b of bgBuildings) {
        ctx.fillStyle = b.color;
        ctx.fillRect(b.x, groundY() - b.h, b.w, b.h);
        // Windows
        ctx.fillStyle = 'rgba(255, 200, 50, 0.3)';
        for (let wy = 0; wy < b.h - 20; wy += 20) {
            for (let wx = 10; wx < b.w - 10; wx += 20) {
                if (Math.sin(b.x * 7 + wx * 3 + wy) > 0.3) {
                    ctx.fillRect(b.x + wx, groundY() - b.h + 15 + wy, 8, 10);
                }
            }
        }
    }
    
    // Ground
    ctx.fillStyle = '#2a2a2a';
    ctx.fillRect(0, groundY(), W, H - groundY());
    ctx.fillStyle = '#3a3a3a';
    ctx.fillRect(0, groundY(), W, 4);
    
    // Road markings
    ctx.fillStyle = '#555';
    for (let i = 0; i < W + speed * frameCount % 60; i += 60) {
        ctx.fillRect((i - (frameCount * speed) % 60), groundY() + 20, 30, 4);
    }
}

function drawHUD() {
    // Score
    ctx.fillStyle = '#fff';
    ctx.font = 'bold 24px monospace';
    ctx.textAlign = 'left';
    ctx.fillText(`SCORE: ${Math.floor(score)}`, 20, 35);
    
    ctx.font = '14px monospace';
    ctx.fillStyle = '#aaa';
    ctx.fillText(`HI: ${Math.max(highScore, Math.floor(score))}`, 20, 55);
    
    // Health hearts
    for (let i = 0; i < player.maxHealth; i++) {
        ctx.fillStyle = i < player.health ? '#f44' : '#444';
        ctx.font = '20px monospace';
        ctx.fillText(i < player.health ? '♥' : '♡', W - 100 + i * 30, 35);
    }
    
    // Speed indicator
    ctx.fillStyle = '#ff0';
    ctx.font = '12px monospace';
    ctx.textAlign = 'right';
    ctx.fillText(`SPEED: ${speed.toFixed(1)}x`, W - 20, H - 20);
}

function drawStartScreen() {
    drawBackground();
    
    ctx.fillStyle = 'rgba(0,0,0,0.7)';
    ctx.fillRect(0, 0, W, H);
    
    // Title
    ctx.textAlign = 'center';
    ctx.fillStyle = '#f44';
    ctx.font = `bold ${Math.min(W*0.06, 60)}px monospace`;
    ctx.fillText('ZOMBIE ENDLESS', W/2, H * 0.25);
    
    ctx.fillStyle = '#fff';
    ctx.font = `${Math.min(W*0.03, 24)}px monospace`;
    ctx.fillText('RUNNER', W/2, H * 0.33);
    
    // Instructions
    ctx.font = `${Math.min(W*0.025, 18)}px monospace`;
    ctx.fillStyle = '#ccc';
    ctx.fillText('↑ / W / SPACE - Jump', W/2, H * 0.5);
    ctx.fillText('↓ / S - Duck', W/2, H * 0.55);
    ctx.fillText('Z / X / F - Shoot', W/2, H * 0.6);
    ctx.fillText('Avoid zombies! Collect power-ups!', W/2, H * 0.68);
    
    // Blinking start prompt
    if (Math.floor(Date.now() / 500) % 2 === 0) {
        ctx.fillStyle = '#ff0';
        ctx.font = `bold ${Math.min(W*0.03, 22)}px monospace`;
        ctx.fillText('PRESS SPACE OR ENTER TO START', W/2, H * 0.8);
    }
    
    // High score
    if (highScore > 0) {
        ctx.fillStyle = '#aaa';
        ctx.font = '16px monospace';
        ctx.fillText(`High Score: ${highScore}`, W/2, H * 0.9);
    }
}

function drawGameOver() {
    ctx.fillStyle = 'rgba(0,0,0,0.8)';
    ctx.fillRect(0, 0, W, H);
    
    ctx.textAlign = 'center';
    ctx.fillStyle = '#f44';
    ctx.font = `bold ${Math.min(W*0.06, 50)}px monospace`;
    ctx.fillText('YOU DIED', W/2, H * 0.3);
    
    ctx.fillStyle = '#fff';
    ctx.font = `${Math.min(W*0.04, 28)}px monospace`;
    ctx.fillText(`Score: ${Math.floor(score)}`, W/2, H * 0.45);
    
    if (Math.floor(score) >= highScore && score > 0) {
        ctx.fillStyle = '#ff0';
        ctx.font = '20px monospace';
        ctx.fillText('NEW HIGH SCORE!', W/2, H * 0.55);
    } else {
        ctx.fillStyle = '#aaa';
        ctx.font = '18px monospace';
        ctx.fillText(`Best: ${highScore}`, W/2, H * 0.55);
    }
    
    if (Math.floor(Date.now() / 500) % 2 === 0) {
        ctx.fillStyle = '#ff0';
        ctx.font = `bold ${Math.min(W*0.03, 20)}px monospace`;
        ctx.fillText('PRESS SPACE / R TO RESTART', W/2, H * 0.7);
    }
}

function render() {
    if (gameState === 'start') {
        drawStartScreen();
        return;
    }
    
    drawBackground();
    
    // Draw powerups
    for (let p of powerups) {
        ctx.fillStyle = p.type === 'health' ? '#f44' : '#ff0';
        ctx.beginPath();
        ctx.arc(p.x + p.w/2, p.y + p.h/2, 12, 0, Math.PI * 2);
        ctx.fill();
        ctx.fillStyle = '#fff';
        ctx.font = 'bold 14px monospace';
        ctx.textAlign = 'center';
        ctx.fillText(p.type === 'health' ? '+' : '⚡', p.x + p.w/2, p.y + p.h/2 + 5);
    }
    
    // Draw zombies
    for (let z of zombies) drawZombie(z);
    
    // Draw bullets
    ctx.fillStyle = '#ff0';
    for (let b of bullets) {
        ctx.fillRect(b.x, b.y, b.w, b.h);
        ctx.fillStyle = 'rgba(255, 200, 0, 0.3)';
        ctx.fillRect(b.x - 4, b.y - 1, b.w + 4, b.h + 2);
        ctx.fillStyle = '#ff0';
    }
    
    // Draw player
    drawPlayer();
    
    // Draw particles
    for (let p of particles) {
        ctx.globalAlpha = p.life / p.maxLife;
        ctx.fillStyle = p.color;
        ctx.fillRect(p.x, p.y, p.size, p.size);
    }
    ctx.globalAlpha = 1;
    
    // HUD
    drawHUD();
    
    if (gameState === 'gameover') drawGameOver();
}

function gameLoop() {
    update();
    render();
    requestAnimationFrame(gameLoop);
}

initBackground();
gameLoop();
</script>
