function playSound(freq, duration, type, vol) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = type || 'sine';
    osc.frequency.value = freq;
    gain.gain.setValueAtTime(vol || 0.3, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + duration);
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + duration);
}

// Collision helpers
function pointToLineDist(px, py, x1, y1, x2, y2) {
    const dx = x2 - x1;
    const dy = y2 - y1;
    const len2 = dx * dx + dy * dy;
    if (len2 === 0) return Math.hypot(px - x1, py - y1);
    let t = ((px - x1) * dx + (py - y1) * dy) / len2;
    t = Math.max(0, Math.min(1, t));
    const cx = x1 + t * dx;
    const cy = y1 + t * dy;
    return { dist: Math.hypot(px - cx, py - cy), nx: (cx - px) / (Math.hypot(px-cx,py-cy)||1), ny: (cy - py) / (Math.hypot(px-cx,py-cy)||1), t };
}

function collideWall(wall) {
    const res = pointToLineDist(ball.x, ball.y, wall.x1, wall.y1, wall.x2, wall.y2);
    if (res.dist < ball.r) {
        // Push ball out
        const overlap = ball.r - res.dist;
        ball.x += res.nx * overlap;
        ball.y += res.ny * overlap;
        // Reflect velocity
        const dot = ball.vx * res.nx + ball.vy * res.ny;
        if (dot > 0) {
            ball.vx -= 2 * dot * res.nx * BOUNCE_DAMPING;
            ball.vy -= 2 * dot * res.ny * BOUNCE_DAMPING;
            playSound(300 + Math.random() * 200, 0.08, 'triangle', 0.2);
        }
    }
}

function collideBumper(b) {
    const dx = ball.x - b.x;
    const dy = ball.y - b.y;
    const dist = Math.hypot(dx, dy);
    if (dist < ball.r + b.r) {
        const nx = dx / (dist || 1);
        const ny = dy / (dist || 1);
        const overlap = ball.r + b.r - dist;
        ball.x += nx * overlap;
        ball.y += ny * overlap;

        const speed = Math.hypot(ball.vx, ball.vy);
        ball.vx = nx * (speed + 8);
        ball.vy = ny * (speed + 8);

        if (!b.hit) {
            b.hit = true;
            b.hitTimer = 15;
            addScore(50, b.x, b.y);
            playSound(400, 0.2, 'square', 0.3);
            for (let i = 0; i < 6; i++) {
                particles.push({
                    x: b.x + nx * b.r, y: b.y + ny * b.r,
                    vx: (Math.random() - 0.5) * 6,
                    vy: (Math.random() - 0.5) * 6,
                    life: 25, maxLife: 25, color: '#ffdd44'
                });
            }
        }
    }
}

function addScore(pts, x, y) {
    score += pts;
    popups.push({ x: x, y: y, text: '+' + pts, life: 40, maxLife: 40 });
}

function spawnParticles(x, y, color, count) {
    for (let i = 0; i < count; i++) {
        const angle = Math.random() * Math.PI * 2;
        const speed = 2 + Math.random() * 4;
        particles.push({
            x: x, y: y,
            vx: Math.cos(angle) * speed,
            vy: Math.sin(angle) * speed,
            life: 30, maxLife: 30, color: color
        });
    }
}

function update() {
    if (gameState !== 'playing') return;

    // Kick cooldown
    if (kickCooldown > 0) kickCooldown--;

    // Physics
    ball.vy += GRAVITY;
    ball.vx *= DAMPING;
    ball.vy *= DAMPING;
    ball.x += ball.vx;
    ball.y += ball.vy;

    // Collide with walls
    for (const w of walls) {
        collideWall(w);
    }

    // Moving walls
    const time = Date.now() * 0.001;
    for (const mw of movingWalls) {
        mw.angle = Math.sin(time * mw.speed * 60 + mw.offset) * 0.8;
        const len = mw.len;
        const rad = mw.angle;
        const dx = Math.cos(rad) * len;
        const dy = Math.sin(rad) * len;
        const wall = { x1: mw.x - dx/2, y1: mw.y - dy/2, x2: mw.x + dx/2, y2: mw.y + dy/2 };
        collideWall(wall);
    }

    // Bumpers
    for (const b of bumpers) {
        if (b.hit && b.hitTimer > 0) {
            b.hitTimer--;
            if (b.hitTimer <= 0) b.hit = false;
        }
        collideBumper(b);
    }

    // Camera follows ball
    const targetCamX = ball.x - W * 0.3;
    camX += (targetCamX - camX) * 0.05;
    if (camX < 0) camX = 0;

    // Check game over
    if (ball.y > H + 50 || ball.x < -100) {
        gameState = 'gameover';
        if (score > highScore) {
            highScore = score;
            try { localStorage.setItem('sideScrollPinballHigh', highScore.toString()); } catch(e) {}
        }
        playSound(150, 0.5, 'sawtooth', 0.3);
    }

    // Particles
    for (let i = particles.length - 1; i >= 0; i--) {
        particles[i].x += particles[i].vx;
        particles[i].y += particles[i].vy;
        particles[i].vy += 0.2;
        particles[i].life--;
        if (particles[i].life <= 0) particles.splice(i, 1);
    }

    // Popups
    for (let i = popups.length - 1; i >= 0; i--) {
        popups[i].y -= 1;
        popups[i].life--;
        if (popups[i].life <= 0) popups.splice(i, 1);
    }
}

function draw() {
    ctx.fillStyle = '#0a0a1a';
    ctx.fillRect(0, 0, W, H);

    if (gameState === 'start') {
        drawStartScreen();

    // Background grid
    ctx.strokeStyle = 'rgba(50, 60, 100, 0.3)';

    const startX = Math.floor(camX / gridSize) * gridSize;
    for (let x = startX; x < camX + W + gridSize; x += gridSize) {
        ctx.beginPath(); ctx.moveTo(x, 0); ctx.lineTo(x, H); ctx.stroke();
    }
    for (let y = 0; y < H; y += gridSize) {
        ctx.beginPath(); ctx.moveTo(0, y); ctx.lineTo(W, y); ctx.stroke();
    }

    // Draw walls
    ctx.lineCap = 'round';
    ctx.lineWidth = 6;
    ctx.strokeStyle = '#4a9eff';
    ctx.shadowColor = '#4a9eff';
    ctx.shadowBlur = 8;
    for (const w of walls) {
        ctx.beginPath();
        ctx.moveTo(w.x1, w.y1);
        ctx.lineTo(w.x2, w.y2);
        ctx.stroke();
    }

    // Moving walls
    ctx.strokeStyle = '#ff6b35';
    ctx.shadowColor = '#ff6b35';
    for (const mw of movingWalls) {
        const rad = mw.angle;
        const dx = Math.cos(rad) * mw.len / 2;
        const dy = Math.sin(rad) * mw.len / 2;
        ctx.beginPath();
        ctx.moveTo(mw.x - dx, mw.y - dy);
        ctx.lineTo(mw.x + dx, mw.y + dy);
        ctx.stroke();
    }

    // Draw bumpers
    for (const b of bumpers) {
        const scale = b.hit ? 1.2 : 1;
        ctx.beginPath();
        ctx.arc(b.x, b.y, b.r * scale, 0, Math.PI * 2);
        if (b.hit) {
            ctx.fillStyle = '#ffdd44';
            ctx.shadowColor = '#ffdd44';
        } else {
            ctx.fillStyle = '#ff4466';
            ctx.shadowColor = '#ff4466';
        }
        ctx.fill();
        ctx.strokeStyle = 'rgba(255,255,255,0.5)';
        ctx.lineWidth = 3;
        ctx.stroke();

        // Inner circle

        ctx.arc(b.x, b.y, b.r * 0.5 * scale, 0, Math.PI * 2);
        ctx.fillStyle = 'rgba(255,255,255,0.3)';
        ctx.fill();
    }

    // Draw ball
    ctx.shadowColor = '#ffffff';
    ctx.shadowBlur = 12;
    ctx.beginPath();
    ctx.arc(ball.x, ball.y, ball.r, 0, Math.PI * 2);
    const grad = ctx.createRadialGradient(ball.x - 3, ball.y - 3, 0, ball.x, ball.y, ball.r);
    grad.addColorStop(0, '#ffffff');
    grad.addColorStop(1, '#aaaacc');
    ctx.fillStyle = grad;
    ctx.fill();

    // Particles
    for (const p of particles) {
        const alpha = p.life / p.maxLife;
        ctx.globalAlpha = alpha;
        ctx.fillStyle = p.color;
        ctx.beginPath();
        ctx.arc(p.x, p.y, 3, 0, Math.PI * 2);
        ctx.fill();
    }
    ctx.globalAlpha = 1;

    // Score popups
    for (const p of popups) {
        const alpha = p.life / p.maxLife;
        ctx.globalAlpha = alpha;
        ctx.fillStyle = '#ffdd44';
        ctx.font = 'bold 20px monospace';
        ctx.textAlign = 'center';
        ctx.fillText(p.text, p.x, p.y);
    }
    ctx.globalAlpha = 1;

    ctx.restore();

    // HUD
    ctx.shadowBlur = 0;
    ctx.fillStyle = '#ffffff';
    ctx.font = 'bold 24px monospace';
    ctx.textAlign = 'left';
    ctx.fillText('SCORE: ' + score, 20, 40);
    ctx.font = '16px monospace';
    ctx.fillStyle = '#888';
    ctx.fillText('HIGH: ' + highScore, 20, 65);

    // Kick indicator
    if (kickCooldown > 0) {
        ctx.fillStyle = 'rgba(255,107,53,' + (kickCooldown / 20) + ')';
        ctx.font = '14px monospace';
        ctx.textAlign = 'center';
        ctx.fillText('COOLDOWN', ball.x - camX, ball.y + 40);
    } else {
        ctx.fillStyle = '#4a9eff';
        ctx.font = '14px monospace';
        ctx.textAlign = 'center';
        ctx.fillText('[SPACE] KICK', W / 2, H - 50);
    }

        drawGameOver();
    }
}
