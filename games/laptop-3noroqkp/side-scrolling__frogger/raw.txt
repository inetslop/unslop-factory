const canvas = document.getElementById('game');
const ctx = canvas.getContext('2d');

let W, H, CELL, COLS = 13, ROWS = 9;
let scale = 1;

function resize() {
    const maxW = window.innerWidth;
    const maxH = window.innerHeight;
    scale = Math.min(maxW / (COLS * 64), maxH / (ROWS * 64)) * 0.95;
    W = COLS * 64 * scale;
    H = ROWS * 64 * scale;
    canvas.width = W;
    canvas.height = H;
    CELL = 64 * scale;
}
window.addEventListener('resize', resize);
resize();

// Audio
let audioCtx = null;
function initAudio() {
    if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
}
function playSound(type) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    gain.gain.setValueAtTime(0.15, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.2);
    if (type === 'move') { osc.type = 'square'; osc.frequency.value = 440; }
    else if (type === 'splash') { osc.type = 'sawtooth'; osc.frequency.value = 150; gain.gain.setValueAtTime(0.2, audioCtx.currentTime); gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.4); }
    else if (type === 'crash') { osc.type = 'sawtooth'; osc.frequency.value = 100; gain.gain.setValueAtTime(0.25, audioCtx.currentTime); gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.3); }
    else if (type === 'score') { osc.type = 'square'; osc.frequency.value = 880; }
    else if (type === 'home') { osc.type = 'triangle'; osc.frequency.value = 660; gain.gain.setValueAtTime(0.2, audioCtx.currentTime); gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.5); }
    else if (type === 'life') { osc.type = 'triangle'; osc.frequency.value = 330; gain.gain.setValueAtTime(0.2, audioCtx.currentTime); gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.6); }
    osc.start();
    osc.stop(audioCtx.currentTime + 0.5);
}

// Game state
let gameState = 'START'; // START, PLAYING, ROUND_CLEAR, GAME_OVER
let score = 0;
let lives = 3;
let highScore = 0;
try { highScore = parseInt(localStorage.getItem('frogger_high') || '0'); } catch(e) {}
let frogCol = 6, frogRow = 8;
let frogAnimX = 6 * CELL, frogAnimY = 8 * CELL;
let moveTimer = 0;
const MOVE_DELAY = 120;
let traffic = [];
let water = [];
let particles = [];
let roundClearTimer = 0;
let gameTime = 0;

function initTraffic() {
    traffic = [];
    // Rows 5-8 (4 lanes)
    const configs = [
        { row: 5, type: 'car', speed: 1.2, dir: 1, count: 3 },
        { row: 6, type: 'truck', speed: 0.7, dir: -1, count: 2 },
        { row: 7, type: 'car', speed: 1.5, dir: -1, count: 4 },
        { row: 8, type: 'bus', speed: 0.9, dir: 1, count: 2 }
    ];
    for (const cfg of configs) {
        const entities = [];
        const spacing = Math.floor(COLS * CELL / cfg.count);
        for (let i = 0; i < cfg.count; i++) {
            const w = cfg.type === 'truck' ? 2.5 : cfg.type === 'bus' ? 3 : 1.8;
            entities.push({ x: i * spacing + Math.random() * spacing * 0.5, w: w * CELL });
        }
        traffic.push({ ...cfg, entities });
    }
}

function initWater() {
    water = [];
    // Rows 1-3 (3 lanes)
    const configs = [
        { row: 1, type: 'turtle', speed: 0.8, dir: -1, count: 4, still: false },
        { row: 2, type: 'log', speed: 0.5, dir: 1, count: 3, still: true },
        { row: 3, type: 'turtle', speed: 1.0, dir: 1, count: 3, still: false }
    ];
    for (const cfg of configs) {
        const entities = [];
        const spacing = Math.floor(COLS * CELL / cfg.count);
        for (let i = 0; i < cfg.count; i++) {
            const w = cfg.type === 'log' ? 2 : 1.5;
            entities.push({ x: i * spacing + Math.random() * spacing * 0.3, w: w * CELL });
        }
        water.push({ ...cfg, entities });
    }
}

function resetFrog() {
    frogCol = 6;
    frogRow = 8;
    frogAnimX = 6 * CELL;
    frogAnimY = 8 * CELL;
    moveTimer = 0;
}

function startGame() {
    score = 0;
    lives = 3;
    initTraffic();
    initWater();
    resetFrog();
    gameState = 'PLAYING';
}

function addParticles(x, y, color, count) {
    for (let i = 0; i < count; i++) {
        particles.push({
            x: x + CELL/2, y: y + CELL/2,
            vx: (Math.random()-0.5)*3*scale,
            vy: (Math.random()-0.5)*3*scale - 1*scale,
            life: 1, color
        });
    }
}

// Input
let keys = {};
window.addEventListener('keydown', (e) => {
    initAudio();
    keys[e.key] = true;
    if (gameState === 'START') {
        if (e.key === ' ' || e.key === 'Enter') startGame();
    } else if (gameState === 'GAME_OVER') {
        if (e.key === ' ' || e.key === 'Enter') startGame();
    } else if (gameState === 'PLAYING') {
        handleMove(e.key);
    }
});
window.addEventListener('keyup', (e) => { keys[e.key] = false; });

function handleMove(key) {
    if (moveTimer > 0) return;
    let dx = 0, dy = 0;
    if (key === 'ArrowUp' || key === 'w') dy = -1;
    else if (key === 'ArrowDown' || key === 's') dy = 1;
    else if (key === 'ArrowLeft' || key === 'a') dx = -1;
    else if (key === 'ArrowRight' || key === 'd') dx = 1;
    else return;

    let nc = frogCol + dx, nr = frogRow + dy;
    if (nc < 0 || nc >= COLS || nr < 0 || nr >= ROWS) return;

    // Check home row
    if (nr === 0) {
        if ([2,4,6,8,10].includes(nc)) {
            // Made it home!
            playSound('home');
            score += 50 + Math.floor(score / 10);
            addParticles(nc * CELL, nr * CELL, '#4fff4f', 15);
            if (lives <= 0) {
                gameState = 'GAME_OVER';
                try { localStorage.setItem('frogger_high', highScore.toString()); } catch(e) {}
            } else {
                gameState = 'ROUND_CLEAR';
                roundClearTimer = 60;
            }
            return;
        } else {
            nr = 1; // Bounce back
        }
    }

    frogCol = nc;
    frogRow = nr;
    moveTimer = MOVE_DELAY;
    playSound('move');

    // Check collisions
    if (nr >= 5 && nr <= 8) {
        // Traffic row
        const lane = traffic[nr - 5];
        for (const ent of lane.entities) {
            const ex = ((ent.x % (COLS * CELL + 200*scale)) + COLS*CELL) % (COLS*CELL) - 100*scale;
            if (Math.abs((nc*CELL + CELL/2) - (ex + ent.w/2)) < (CELL + ent.w)/2 * 0.7) {
                // Hit by vehicle!
                playSound('crash');
                addParticles(nc*CELL, nr*CELL, '#ff4444', 12);
                lives--;
                if (lives <= 0) {
                    gameState = 'GAME_OVER';
                    highScore = Math.max(highScore, score);
                    try { localStorage.setItem('frogger_high', highScore.toString()); } catch(e) {}
                } else {
                    playSound('life');
                    resetFrog();
                }
                return;
            }
        }
    } else if (nr >= 1 && nr <= 3) {
        // Water row
        const lane = water[nr - 1];
        let onEntity = false;
        for (const ent of lane.entities) {
            const ex = ((ent.x % (COLS * CELL + 200*scale)) + COLS*CELL) % (COLS*CELL) - 100*scale;
            if (Math.abs((nc*CELL + CELL/2) - (ex + ent.w/2)) < (CELL + ent.w)/2 * 0.7) {
                onEntity = true;
                break;
            }
        }
        if (!onEntity) {
            playSound('splash');
            addParticles(nc*CELL, nr*CELL, '#44aaff', 10);
            lives--;
            if (lives <= 0) {
                gameState = 'GAME_OVER';
                highScore = Math.max(highScore, score);
                try { localStorage.setItem('frogger_high', highScore.toString()); } catch(e) {}
            } else {
                playSound('life');
                resetFrog();
            }
        }
    }
}

// Update
function update(dt) {
    if (gameState !== 'PLAYING' && gameState !== 'ROUND_CLEAR') return;

    gameTime += dt;
    if (moveTimer > 0) moveTimer -= dt;

    // Animate frog
    const targetX = frogCol * CELL;
    const targetY = frogRow * CELL;
    frogAnimX += (targetX - frogAnimX) * 0.25;
    frogAnimY += (targetY - frogAnimY) * 0.25;

    // Update traffic
    for (const lane of traffic) {
        for (const ent of lane.entities) {
            ent.x += lane.speed * lane.dir * scale * dt * 0.06;
            const totalW = COLS * CELL + 200*scale;
            if (lane.dir > 0 && ent.x > totalW - 100*scale) ent.x = -ent.w;
            if (lane.dir < 0 && ent.x < -ent.w - 100*scale) ent.x = COLS * CELL;
        }
    }

    // Update water
    for (const lane of water) {
        if (lane.still) continue;
        for (const ent of lane.entities) {
            ent.x += lane.speed * lane.dir * scale * dt * 0.04;
            const totalW = COLS * CELL + 200*scale;
            if (lane.dir > 0 && ent.x > totalW - 100*scale) ent.x = -ent.w;
            if (lane.dir < 0 && ent.x < -ent.w - 100*scale) ent.x = COLS * CELL;
        }
    }

    // Particles
    for (let i = particles.length - 1; i >= 0; i--) {
        const p = particles[i];
        p.x += p.vx;
        p.y += p.vy;
        p.vy += 0.1 * scale;
        p.life -= dt * 0.02;
        if (p.life <= 0) particles.splice(i, 1);
    }

    // Round clear
    if (gameState === 'ROUND_CLEAR') {
        roundClearTimer -= dt;
        if (roundClearTimer <= 0) {
            resetFrog();
            gameState = 'PLAYING';
        }
    }
}

// Draw
function draw() {
    ctx.clearRect(0, 0, W, H);

    // Background
    for (let r = 0; r < ROWS; r++) {
        if (r === 0) {
            ctx.fillStyle = '#2a1a4a';
        } else if (r >= 1 && r <= 3) {
            ctx.fillStyle = '#1a3a5a';
        } else if (r === 4) {
            ctx.fillStyle = '#1a4a1a';
        } else {
            ctx.fillStyle = '#2a2a2a';
        }
        ctx.fillRect(0, r * CELL, W, CELL);
    }

    // Home pads
    const padCols = [2, 4, 6, 8, 10];
    for (const c of padCols) {
        drawPaddle(c * CELL, 0);
    }

    // Water entities
    for (let i = 0; i < water.length; i++) {
        const lane = water[i];
        for (const ent of lane.entities) {
            let ex = ((ent.x % (COLS * CELL + 200*scale)) + COLS*CELL) % (COLS*CELL) - 100*scale;
            drawWaterEntity(ex, lane.row * CELL, lane.type);
        }
    }

    // Traffic entities
    for (const lane of traffic) {
        for (const ent of lane.entities) {
            let ex = ((ent.x % (COLS * CELL + 200*scale)) + COLS*CELL) % (COLS*CELL) - 100*scale;
            drawVehicle(ex, lane.row * CELL, lane.type);
        }
    }

    // Median line
    ctx.strokeStyle = '#4a8a4a';
    ctx.lineWidth = 2 * scale;
    ctx.setLineDash([8*scale, 6*scale]);
    ctx.beginPath();
    ctx.moveTo(0, 4.5 * CELL);
    ctx.lineTo(W, 4.5 * CELL);
    ctx.stroke();
    ctx.setLineDash([]);

    // Frog
    if (gameState === 'PLAYING' || gameState === 'ROUND_CLEAR') {
        drawFrog(frogAnimX, frogAnimY);
    }

    // Particles
    for (const p of particles) {
        ctx.globalAlpha = p.life;
        ctx.fillStyle = p.color;
        ctx.fillRect(p.x - 3*scale, p.y - 3*scale, 6*scale, 6*scale);
    }
    ctx.globalAlpha = 1;

    // HUD
    drawHUD();

    // Overlays
    if (gameState === 'START') drawStartScreen();
    else if (gameState === 'GAME_OVER') drawGameOver();
    else if (gameState === 'ROUND_CLEAR') drawRoundClear();
}

function drawPaddle(x, y) {
    const s = CELL;
    ctx.fillStyle = '#3a7a3a';
    ctx.beginPath();
    ctx.ellipse(x + s/2, y + s*0.6, s*0.4, s*0.25, 0, 0, Math.PI*2);
    ctx.fill();
    ctx.fillStyle = '#4a9a4a';
    ctx.beginPath();
    ctx.ellipse(x + s/2, y + s*0.55, s*0.35, s*0.2, 0, 0, Math.PI*2);
    ctx.fill();
}

function drawWaterEntity(x, y, type) {
    const s = CELL;
    if (type === 'turtle') {
        ctx.fillStyle = '#3a7a3a';
        ctx.beginPath();
        ctx.ellipse(x + s/2, y + s*0.55, s*0.4, s*0.3, 0, 0, Math.PI*2);
        ctx.fill();
        // Head
        ctx.fillStyle = '#5aaa5a';
        ctx.beginPath();
        ctx.arc(x + s*0.75, y + s*0.5, s*0.12, 0, Math.PI*2);
        ctx.fill();
    } else {
        // Log
        ctx.fillStyle = '#6a4a2a';
        ctx.fillRect(x, y + s*0.35, s * 1.8, s * 0.3);
        ctx.fillStyle = '#7a5a3a';
        ctx.fillRect(x + 4*scale, y + s*0.38, s * 1.7, s * 0.24);
    }
}

function drawVehicle(x, y, type) {
    const s = CELL;
    if (type === 'car') {
        ctx.fillStyle = '#cc3333';
        ctx.fillRect(x, y + s*0.35, s*1.2, s*0.4);
        ctx.fillStyle = '#ff5555';
        ctx.fillRect(x + 4*scale, y + s*0.25, s*1.1, s*0.2);
        // Windows
        ctx.fillStyle = '#aaddff';
        ctx.fillRect(x + s*0.15, y + s*0.28, s*0.3, s*0.15);
        ctx.fillRect(x + s*0.6, y + s*0.28, s*0.3, s*0.15);
    } else if (type === 'truck') {
        ctx.fillStyle = '#cc9933';
        ctx.fillRect(x, y + s*0.3, s*1.8, s*0.45);
        ctx.fillStyle = '#ffbb44';
        ctx.fillRect(x + 4*scale, y + s*0.25, s*1.7, s*0.25);
    } else {
        // Bus
        ctx.fillStyle = '#3366cc';
        ctx.fillRect(x, y + s*0.25, s*2.2, s*0.5);
        ctx.fillStyle = '#4488ff';
        ctx.fillRect(x + 4*scale, y + s*0.2, s*2.1, s*0.3);
        // Windows
        ctx.fillStyle = '#aaddff';
        for (let i = 0; i < 3; i++) {
            ctx.fillRect(x + s*(0.2 + i*0.65), y + s*0.25, s*0.4, s*0.15);
        }
    }
}

function drawFrog(x, y) {
    const s = CELL;
    // Body
    ctx.fillStyle = '#33cc33';
    ctx.beginPath();
    ctx.arc(x + s/2, y + s*0.55, s*0.35, 0, Math.PI*2);
    ctx.fill();
    // Head
    ctx.fillStyle = '#44ee44';
    ctx.beginPath();
    ctx.arc(x + s/2, y + s*0.4, s*0.28, 0, Math.PI*2);
    ctx.fill();
    // Eyes
    ctx.fillStyle = '#ffffff';
    ctx.beginPath();
    ctx.arc(x + s*0.35, y + s*0.35, s*0.1, 0, Math.PI*2);
    ctx.arc(x + s*0.65, y + s*0.35, s*0.1, 0, Math.PI*2);
    ctx.fill();
    ctx.fillStyle = '#000';
    ctx.beginPath();
    ctx.arc(x + s*0.37, y + s*0.35, s*0.05, 0, Math.PI*2);
    ctx.arc(x + s*0.63, y + s*0.35, s*0.05, 0, Math.PI*2);
    ctx.fill();
}

function drawHUD() {
    const fs = Math.max(12, 14*scale);
    ctx.font = `bold ${fs}px monospace`;
    ctx.fillStyle = '#ffffff';
    ctx.textAlign = 'left';
    ctx.fillText(`SCORE: ${score}`, 10*scale, fs + 5);
    ctx.textAlign = 'right';
    ctx.fillText(`HI: ${highScore}`, W - 10*scale, fs + 5);

    // Lives as frogs
    for (let i = 0; i < lives; i++) {
        ctx.fillStyle = '#33cc33';
        ctx.beginPath();
        ctx.arc(10*scale + 40*scale + i*25*scale, fs/2 + 5, 6*scale, 0, Math.PI*2);
        ctx.fill();
    }
}

function drawStartScreen() {
    ctx.fillStyle = 'rgba(0,0,0,0.7)';
    ctx.fillRect(0, 0, W, H);

    const fs = Math.max(16, 24*scale);
    const fsSmall = Math.max(12, 16*scale);
    ctx.textAlign = 'center';
    ctx.fillStyle = '#33cc33';
    ctx.font = `bold ${fs}px monospace`;
    ctx.fillText('FROGGER', W/2, H*0.25);

    ctx.fillStyle = '#ffffff';
    ctx.font = `${fsSmall}px monospace`;
    ctx.fillText('Guide the frog across traffic and water', W/2, H*0.4);
    ctx.fillText('to reach the lily pads at the top!', W/2, H*0.4 + fsSmall*1.5);

    ctx.fillStyle = '#aaaaff';
    ctx.font = `${fsSmall}px monospace`;
    ctx.fillText('Arrow Keys / WASD to move', W/2, H*0.6);
    ctx.fillText('Avoid cars and water - ride turtles & logs!', W/2, H*0.6 + fsSmall*1.5);

    const pulse = Math.sin(gameTime * 0.005) * 0.3 + 0.7;
    ctx.globalAlpha = pulse;
    ctx.fillStyle = '#ffff44';
    ctx.font = `bold ${fsSmall}px monospace`;
    ctx.fillText('Press SPACE or ENTER to Start', W/2, H*0.8);
    ctx.globalAlpha = 1;

    if (highScore > 0) {
        ctx.fillStyle = '#ffaa44';
        ctx.font = `${fsSmall}px monospace`;
        ctx.fillText(`High Score: ${highScore}`, W/2, H*0.9);
    }
}

function drawGameOver() {
    ctx.fillStyle = 'rgba(0,0,0,0.7)';
    ctx.fillRect(0, 0, W, H);

    const fs = Math.max(16, 24*scale);
    const fsSmall = Math.max(12, 16*scale);
    ctx.textAlign = 'center';
    ctx.fillStyle = '#ff4444';
    ctx.font = `bold ${fs}px monospace`;
    ctx.fillText('GAME OVER', W/2, H*0.35);

    ctx.fillStyle = '#ffffff';
    ctx.font = `${fsSmall}px monospace`;
    ctx.fillText(`Final Score: ${score}`, W/2, H*0.5);

    if (score >= highScore && score > 0) {
        ctx.fillStyle = '#ffff44';
        ctx.fillText('NEW HIGH SCORE!', W/2, H*0.6);
    }

    const pulse = Math.sin(gameTime * 0.005) * 0.3 + 0.7;
    ctx.globalAlpha = pulse;
    ctx.fillStyle = '#aaaaff';
    ctx.fillText('Press SPACE to Play Again', W/2, H*0.75);
    ctx.globalAlpha = 1;
}

function drawRoundClear() {
    const fsSmall = Math.max(12, 16*scale);
    ctx.fillStyle = 'rgba(0,0,50,0.3)';
    ctx.fillRect(0, 0, W, H);
    ctx.textAlign = 'center';
    ctx.fillStyle = '#44ff44';
    ctx.font = `bold ${fsSmall}px monospace`;
    ctx.fillText('HOME! +50', W/2, H*0.5);
}

// Main loop
let lastTime = 0;
function loop(time) {
    const dt = Math.min(time - lastTime, 50);
    lastTime = time;
    update(dt);
    draw();
    requestAnimationFrame(loop);
}
requestAnimationFrame(loop);
