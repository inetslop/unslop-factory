```html
<!DOCTYPE html>
<html lang="en">
<head>
<meta charset="UTF-8">
<meta name="viewport" content="width=device-width, initial-scale=1.0">
<title>Breakout Maze</title>
<style>
* { margin: 0; padding: 0; box-sizing: border-box; }
html, body { width: 100%; height: 100%; overflow: hidden; background: #0a0a1a; }
canvas { display: block; width: 100vw; height: 100vh; }
</style>
</head>
<body>
<canvas id="game"></canvas>
<script>
(function() {
const canvas = document.getElementById('game');
const ctx = canvas.getContext('2d');

let W, H;
function resize() {
W = window.innerWidth;
H = window.innerHeight;
canvas.width = W;
canvas.height = H;
}
window.addEventListener('resize', resize);
resize();

// Audio
let audioCtx = null;
function initAudio() {
if (!audioCtx) {
audioCtx = new (window.AudioContext || window.webkitAudioContext)();
}
}
function playSound(freq, type, duration, volume) {
if (!audioCtx) return;
const osc = audioCtx.createOscillator();
const gain = audioCtx.createGain();
osc.type = type;
osc.frequency.value = freq;
gain.gain.setValueAtTime(volume || 0.15, audioCtx.currentTime);
gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + (duration || 0.1));
osc.connect(gain);
gain.connect(audioCtx.destination);
osc.start();
osc.stop(audioCtx.currentTime + (duration || 0.1));
}

// Game state
let gameState = 'start'; // start, playing, win, lose
let score = 0;
let lives = 3;
let highScore = 0;
try {
highScore = parseInt(localStorage.getItem('breakoutMazeHigh') || '0');
} catch(e) {}

// Paddle
let paddle = { w: 120, h: 18, x: 0, y: 0, speed: 10 };

// Ball
let ball = { x: 0, y: 0, r: 8, dx: 0, dy: 0, speed: 6, stuck: true };

// Bricks
let bricks = [];
const BRICK_ROWS = 5;
const BRICK_COLS = 12;
const BRICK_H = 24;
const BRICK_GAP = 3;
const BRICK_TOP = 50;

// Maze walls
let mazeWalls = [];

// Particles
let particles = [];

// Keys
let keys = {};
window.addEventListener('keydown', e => {
keys[e.code] = true;
if (gameState === 'start' && (e.code === 'Space' || e.code === 'Enter')) {
initAudio();
startGame();
}
if ((gameState === 'win' || gameState === 'lose') && e.code === 'KeyR') {
startGame();
}
});
window.addEventListener('keyup', e => { keys[e.code] = false; });

// Mouse
let mouseX = W / 2;
canvas.addEventListener('mousemove', e => { mouseX = e.clientX; });
canvas.addEventListener('click', () => {
initAudio();
if (gameState === 'start') startGame();
else if (gameState === 'lose' || gameState === 'win') startGame();
else if (ball.stuck) launchBall();
});

function generateMazeWalls() {
mazeWalls = [];
const mazeTop = H * 0.35;
const mazeBottom = H * 0.7;
const wallCount = 8 + Math.floor(Math.random() * 5);
const used = [];

for (let i = 0; i < wallCount; i++) {
let attempts = 0;
while (attempts < 50) {
attempts++;
const isH = Math.random() > 0.4;
let x, y, w, h;
if (isH) {
w = 80 + Math.random() * 120;
h = 12;
x = Math.random() * (W - w);
y = mazeTop + Math.random() * (mazeBottom - mazeTop);
} else {
w = 12;
h = 80 + Math.random() * 120;
x = Math.random() * (W - w);
y = mazeTop + Math.random() * (mazeBottom - mazeTop - h));
if (y < mazeTop) y = mazeTop;
}
// Check overlap
let overlap = false;
for (let j = 0; j < used.length; j++) {
const u = used[j];
if (x < u.x + u.w && x + w > u.x && y < u.y + u.h && y + h > u.y) {
overlap = true; break;
}
}
if (!overlap) {
used.push({x, y, w, h});
mazeWalls.push({x, y, w, h, color: `hsl(${180 + Math.random()*60}, 70%, 50%)`});
break;
}
}
}
}

function generateBricks() {
bricks = [];
const totalW = W - 40;
const brickW = (totalW - (BRICK_COLS - 1) * BRICK_GAP) / BRICK_COLS;
for (let r = 0; r < BRICK_ROWS; r++) {
for (let c = 0; c < BRICK_COLS; c++) {
bricks.push({
x: 20 + c * (brickW + BRICK_GAP),
y: BRICK_TOP + r * (BRICK_H + BRICK_GAP),
w: brickW,
h: BRICK_H,
alive: true,
color: `hsl(${r * 50}, 80%, ${55 + r * 5}%)`,
points: (BRICK_ROWS - r) * 10
});
}
}
}

function startGame() {
gameState = 'playing';
score = 0;
lives = 3;
paddle.w = 120;
paddle.y = H - 50;
paddle.x = W / 2 - paddle.w / 2;
ball.stuck = true;
ball.x = W / 2;
ball.y = H - 80;
ball.speed = 6 + Math.random() * 2;
resetBall();
generateBricks();
generateMazeWalls();
particles = [];
}

function resetBall() {
ball.stuck = true;
ball.x = W / 2;
ball.y = H - 80;
const angle = (-Math.PI/3) * (Math.random() > 0.5 ? 1 : -1);
ball.dx = Math.cos(angle) * ball.speed;
ball.dy = Math.sin(angle) * ball.speed;
}

function launchBall() {
if (!ball.stuck) return;
ball.stuck = false;
playSound(220, 'square', 0.1, 0.1);
}

function spawnParticles(x, y, color, count) {
for (let i = 0; i < (count || 8); i++) {
const angle = Math.random() * Math.PI * 2;
const speed = 2 + Math.random() * 4;
particles.push({
x, y,
vx: Math.cos(angle) * speed,
vy: Math.sin(angle) * speed,
life: 30 + Math.random() * 20,
maxLife: 50,
color,
size: 3 + Math.random() * 4
});
}
}

function update() {
if (gameState !== 'playing') return;

// Paddle
if (keys['ArrowLeft'] || keys['KeyA']) paddle.x -= paddle.speed;
if (keys['ArrowRight'] || keys['KeyD']) paddle.x += paddle.speed;
paddle.x = Math.max(0, Math.min(W - paddle.w, paddle.x));
if (!keys['ArrowLeft'] && !keys['ArrowRight'] && !keys['KeyA'] && !keys['KeyD']) {
const target = mouseX - paddle.w / 2;
paddle.x += (target - paddle.x) * 0.3;
}

// Ball
if (ball.stuck) {
ball.x = W / 2;
ball.y = H - 80;
return;
}

ball.x += ball.dx;
ball.y += ball.dy;

// Wall collisions
if (ball.x - ball.r < 0) { ball.x = ball.r; ball.dx = -ball.dx; playSound(300, 'square', 0.05, 0.08); }
if (ball.x + ball.r > W) { ball.x = W - ball.r; ball.dx = -ball.dx; playSound(300, 'square', 0.05, 0.08); }
if (ball.y - ball.r < 0) { ball.y = ball.r; ball.dy = -ball.dy; playSound(300, 'square', 0.05, 0.08); }

// Bottom - lose life
if (ball.y + ball.r > H) {
lives--;
playSound(100, 'sawtooth', 0.4, 0.2);
spawnParticles(ball.x, ball.y, '#ff4444', 15);
if (lives <= 0) {
gameState = 'lose';
try {
if (score > highScore) {
highScore = score;
localStorage.setItem('breakoutMazeHigh', highScore.toString());
}
} catch(e) {}
return;
}
resetBall();
}

// Paddle collision
if (ball.dy > 0 && ball.y + ball.r >= paddle.y && ball.y + ball.r <= paddle.y + paddle.h + 5 &&
ball.x >= paddle.x && ball.x <= paddle.x + paddle.w) {
const hitPos = (ball.x - paddle.x) / paddle.w;
const angle = (hitPos - 0.5) * Math.PI * 0.7;
ball.dx = Math.sin(angle) * ball.speed;
ball.dy = -Math.cos(angle) * ball.speed;
ball.y = paddle.y - ball.r;
playSound(440, 'square', 0.08, 0.12);
}

// Maze wall collisions
for (let mw of mazeWalls) {
if (circleRectCollide(ball.x, ball.y, ball.r, mw.x, mw.y, mw.w, mw.h)) {
resolveBallRect(mw);
playSound(500, 'triangle', 0.06, 0.1);
}
}

// Brick collisions
for (let b of bricks) {
if (!b.alive) continue;
if (circleRectCollide(ball.x, ball.y, ball.r, b.x, b.y, b.w, b.h)) {
b.alive = false;
score += b.points;
playSound(600 + Math.random() * 200, 'square', 0.15, 0.15);
spawnParticles(ball.x, ball.y, b.color, 12);
resolveBallRect(b);
// Check win
let remaining = 0;
for (let rb of bricks) if (rb.alive) remaining++;
if (remaining === 0) {
gameState = 'win';
try {
if (score > highScore) {
highScore = score;
localStorage.setItem('breakoutMazeHigh', highScore.toString());
}
} catch(e) {}
}
break;
}
}

// Particles
for (let i = particles.length - 1; i >= 0; i--) {
const p = particles[i];
p.x += p.vx;
p.y += p.vy;
p.vy += 0.2;
p.life--;
if (p.life <= 0) particles.splice(i, 1);
}
}

function circleRectCollide(cx, cy, cr, rx, ry, rw, rh) {
const closestX = Math.max(rx, Math.min(cx, rx + rw));
const closestY = Math.max(ry, Math.min(cy, ry + rh));
const dx = cx - closestX;
const dy = cy - closestY;
return (dx * dx + dy * dy) < (cr * cr);
}

function resolveBallRect(rect) {
const cx = ball.x, cy = ball.y, r = ball.r;
const rx = rect.x, ry = rect.y, rw = rect.w, rh = rect.h;
const closestX = Math.max(rx, Math.min(cx, rx + rw));
const closestY = Math.max(ry, Math.min(cy, ry + rh));
const dx = cx - closestX;
const dy = cy - closestY;
if (Math.abs(dx) > Math.abs(dy)) {
ball.dx = -ball.dx;
ball.x = dx > 0 ? rx + rw + r : rx - r;
} else {
ball.dy = -ball.dy;
ball.y = dy > 0 ? ry + rh + r : ry - r;
}
}

function draw() {
ctx.fillStyle = '#0a0a1a';
ctx.fillRect(0, 0, W, H);

// Background grid
ctx.strokeStyle = 'rgba(50, 60, 100, 0.3)';
ctx.lineWidth = 1;
const gridSize = 50;
for (let x = 0; x < W; x += gridSize) {
ctx.beginPath(); ctx.moveTo(x, 0); ctx.lineTo(x, H); ctx.stroke();
}
for (let y = 0; y < H; y += gridSize) {
ctx.beginPath(); ctx.moveTo(0, y); ctx.lineTo(W, y); ctx.stroke();
}

if (gameState === 'start') {
drawStartScreen();
return;
}

// Maze walls
for (let mw of mazeWalls) {
ctx.fillStyle = mw.color;
ctx.shadowColor = mw.color;
ctx.shadowBlur = 8;
ctx.fillRect(mw.x, mw.y, mw.w, mw.h);
ctx.shadowBlur = 0;
ctx.strokeStyle = 'rgba(255,255,255,0.3)';
ctx.strokeRect(mw.x, mw.y, mw.w, mw.h);
}

// Bricks
for (let b of bricks) {
if (!b.alive) continue;
ctx.fillStyle = b.color;
ctx.shadowColor = b.color;
ctx.shadowBlur = 5;
ctx.fillRect(b.x, b.y, b.w, b.h);
ctx.shadowBlur = 0;
ctx.strokeStyle = 'rgba(255,255,255,0.4)';
ctx.lineWidth = 1;
ctx.strokeRect(b.x, b.y, b.w, b.h);
}

// Paddle
const grad = ctx.createLinearGradient(paddle.x, paddle.y, paddle.x, paddle.y + paddle.h);
grad.addColorStop(0, '#44aaff');
grad.addColorStop(1, '#2266cc');
ctx.fillStyle = grad;
ctx.shadowColor = '#44aaff';
ctx.shadowBlur = 10;
ctx.fillRect(paddle.x, paddle.y, paddle.w, paddle.h);
ctx.shadowBlur = 0;

// Ball
if (!ball.stuck || gameState === 'playing') {
ctx.beginPath();
ctx.arc(ball.x, ball.y, ball.r, 0, Math.PI * 2);
const ballGrad = ctx.createRadialGradient(ball.x - 2, ball.y - 2, 0, ball.x, ball.y, ball.r);
ballGrad.addColorStop(0, '#ffffff');
ballGrad.addColorStop(1, '#ffaa33');
ctx.fillStyle = ballGrad;
ctx.shadowColor = '#ffaa33';
ctx.shadowBlur = 12;
ctx.fill();
ctx.shadowBlur = 0;
}

// Particles
for (let p of particles) {
const alpha = p.life / p.maxLife;
ctx.globalAlpha = alpha;
ctx.fillStyle = p.color;
ctx.fillRect(p.x - p.size/2, p.y - p.size/2, p.size, p.size);
}
ctx.globalAlpha = 1;

// HUD
ctx.fillStyle = '#fff';
ctx.font = 'bold 20px monospace';
ctx.textAlign = 'left';
ctx.fillText(`SCORE: ${score}`, 20, 30);
ctx.textAlign = 'right';
ctx.fillText(`LIVES: ${'❤'.repeat(lives)}`, W - 20, 30);
ctx.textAlign = 'center';
ctx.fillStyle = 'rgba(255,255,255,0.5)';
ctx.font = '14px monospace';
ctx.fillText(`HIGH: ${Math.max(highScore, score)}`, W/2, 30);

if (gameState === 'win') {
drawEndScreen('MAZE CLEARED!', '#44ff88');
} else if (gameState === 'lose') {
drawEndScreen('GAME OVER', '#ff4444');
}
}

function drawStartScreen() {
ctx.fillStyle = '#fff';
ctx.textAlign = 'center';
ctx.font = 'bold 48px monospace';
ctx.shadowColor = '#44aaff';
ctx.shadowBlur = 20;
ctx.fillText('BREAKOUT MAZE', W/2, H/3);
ctx.shadowBlur = 0;

ctx.font = '18px monospace';
ctx.fillStyle = 'rgba(255,255,255,0.8)';
ctx.fillText('Navigate the ball through maze walls to destroy all bricks!', W/2, H/3 + 50);
ctx.fillText('', W/2, H/3 + 90);
ctx.fillStyle = 'rgba(150,200,255,0.9)';
ctx.fillText('← → / A D : Move Paddle', W/2, H/3 + 80);
ctx.fillText('MOUSE : Move Paddle', W/2, H/3 + 110);
ctx.fillText('CLICK / SPACE : Launch Ball', W/2, H/3 + 140);

ctx.fillStyle = `rgba(255,255,255,${0.5 + Math.sin(Date.now()*0.005)*0.3})`;
ctx.font = 'bold 24px monospace';
ctx.fillText('CLICK TO START', W/2, H/2 + 40);

if (highScore > 0) {
ctx.fillStyle = 'rgba(255,220,100,0.7)';
ctx.font = '16px monospace';
ctx.fillText(`HIGH SCORE: ${highScore}`, W/2, H/2 + 80);
}
}

function drawEndScreen(text, color) {
ctx.fillStyle = 'rgba(0,0,0,0.7)';
ctx.fillRect(0, 0, W, H);
ctx.textAlign = 'center';
ctx.font = 'bold 52px monospace';
ctx.shadowColor = color;
ctx.shadowBlur = 15;
ctx.fillStyle = color;
ctx.fillText(text, W/2, H/3);
ctx.shadowBlur = 0;

ctx.fillStyle = '#fff';
ctx.font = '28px monospace';
ctx.fillText(`Score: ${score}`, W/2, H/3 + 60);
ctx.font = '18px monospace';
ctx.fillStyle = 'rgba(255,255,255,0.6)';
ctx.fillText(`High Score: ${Math.max(highScore, score)}`, W/2, H/3 + 100);

ctx.fillStyle = `rgba(255,255,255,${0.5 + Math.sin(Date.now()*0.005)*0.3})`;
ctx.font = 'bold 22px monospace';
ctx.fillText('CLICK or PRESS R TO RESTART', W/2, H/2 + 60);
}

function loop() {
update();
draw();
requestAnimationFrame(loop);
}

window.addEventListener('keydown', e => {
if (e.code === 'Space' && gameState === 'playing' && ball.stuck) {
initAudio();
launchBall();
}
});

loop();
})();
</script>
</body>
</html>
```
