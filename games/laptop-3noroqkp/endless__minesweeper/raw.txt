  for (let dx = -1; dx <= 1; dx++)
    for (let dy = -1; dy <= 1; dy++) {
      const n = getCell(x + dx, y + dy);
      if (n.state === 'revealed' || n.state === 'flagged') return true;
    }
  return false;
}

function drawCell(cell, ox, oy) {
  const sx = ox + cell.x * tileW;
  const sy = oy + cell.y * tileH;
  const pad = 2;
  const w = tileW - pad * 2;
  const h = tileH - pad * 2;

  if (cell.state === 'hidden') {
    // Hidden tile
    ctx.fillStyle = '#1e2a3a';
    ctx.fillRect(sx + pad, sy + pad, w, h);
    ctx.strokeStyle = '#2a3a4a';
    ctx.lineWidth = 1;
    ctx.strokeRect(sx + pad + 0.5, sy + pad + 0.5, w, h);
    // Subtle gradient
    const g = ctx.createLinearGradient(sx+pad, sy+pad, sx+pad, sy+pad+h);
    g.addColorStop(0, 'rgba(255,255,255,0.04)');
    g.addColorStop(1, 'rgba(0,0,0,0.08)');
    ctx.fillStyle = g;
    ctx.fillRect(sx + pad, sy + pad, w, h);
  } else if (cell.state === 'flagged') {
    ctx.fillStyle = '#2a3a4a';
    ctx.fillRect(sx + pad, sy + pad, w, h);
    // Draw flag
    const fx = sx + tileW/2;
    const fy = sy + tileH/2;
    ctx.strokeStyle = '#ff8a65';
    ctx.lineWidth = 2;
    ctx.beginPath();
    ctx.moveTo(fx - 4, fy + 10);
    ctx.lineTo(fx - 4, fy - 10);
    ctx.stroke();
    ctx.fillStyle = '#ff5252';
    ctx.beginPath();
    ctx.moveTo(fx - 4, fy - 10);
    ctx.lineTo(fx + 6, fy - 5);
    ctx.lineTo(fx - 4, fy);
    ctx.fill();
  } else if (cell.state === 'revealed') {
    if (isMineAt(cell.x, cell.y)) {
      // Revealed mine (game over)
      ctx.fillStyle = '#b71c1c';
      ctx.fillRect(sx + pad, sy + pad, w, h);
      drawMine(sx + tileW/2, sy + tileH/2);
    } else {
      ctx.fillStyle = cell.number === 0 ? '#141e28' : '#1a2535';
      ctx.fillRect(sx + pad, sy + pad, w, h);
      if (cell.number > 0) {
        ctx.fillStyle = NUMBER_COLORS[cell.number];
        ctx.font = `bold ${tileW * 0.5}px 'Segoe UI', sans-serif`;
        ctx.textAlign = 'center';
        ctx.textBaseline = 'middle';
        ctx.fillText(cell.number, sx + tileW/2, sy + tileH/2 + 1);
      }
    }
  }
}

function drawMine(cx, cy) {
  const r = tileW * 0.25;
  ctx.fillStyle = '#fff';
  ctx.beginPath();
  ctx.arc(cx, cy, r, 0, Math.PI * 2);
  ctx.fill();
  // Spikes
  ctx.strokeStyle = '#fff';
  ctx.lineWidth = 2;
  for (let i = 0; i < 4; i++) {
    const a = i * Math.PI / 2;
    ctx.beginPath();
    ctx.moveTo(cx - Math.cos(a) * r, cy - Math.sin(a) * r);
    ctx.lineTo(cx + Math.cos(a) * r, cy + Math.sin(a) * r);
    ctx.stroke();
  }
  // Diagonals
  const d = r * 0.7;
  ctx.beginPath();
  ctx.moveTo(cx - d, cy - d); ctx.lineTo(cx + d, cy + d);
  ctx.moveTo(cx + d, cy - d); ctx.lineTo(cx - d, cy + d);
  ctx.stroke();
}

function drawHover(ox, oy) {
  const cell = getCell(hoverX, hoverY);
  if (cell.state !== 'hidden') return;
  const sx = ox + hoverX * tileW;
  const sy = oy + hoverY * tileH;
  ctx.fillStyle = 'rgba(100,180,255,0.15)';
  ctx.fillRect(sx + 2, sy + 2, tileW - 4, tileH - 4);
  ctx.strokeStyle = 'rgba(100,180,255,0.5)';
  ctx.lineWidth = 1;
  ctx.strokeRect(sx + 2.5, sy + 2.5, tileW - 5, tileH - 5);
}

function formatTime(s) {
  const m = Math.floor(s / 60);
  const sec = s % 60;
  return (m > 0 ? m + ':' : '') + (sec < 10 ? '0' : '') + sec + 's';
}
