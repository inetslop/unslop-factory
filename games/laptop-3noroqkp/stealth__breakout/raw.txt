let dpr = 1, cw = 0, ch = 0, scale = 1, offX = 0, offY = 0;

function resize(){
  cw = window.innerWidth; ch = window.innerHeight;
  dpr = Math.min(window.devicePixelRatio || 1, 2);
  canvas.width = Math.floor(cw*dpr);
  canvas.height = Math.floor(ch*dpr);
  canvas.style.width = cw+'px'; canvas.style.height = ch+'px';

let audioCtx = null;
function initAudio(){ if(!audioCtx){ try{ audioCtx = new (window.AudioContext||window.webkitAudioContext)(); }catch(e){} } if(audioCtx && audioCtx.state==='suspended') audioCtx.resume(); }
function beep(freq, dur, type, gain, slideTo){
  if(!audioCtx) return;
  const o = audioCtx.createOscillator(), g = audioCtx.createGain();
  o.type = type||'square'; o.frequency.value = freq;
  if(slideTo) o.frequency.exponentialRampToValueAtTime(slideTo, audioCtx.currentTime + dur);
  g.gain.setValueAtTime(gain||0.12, audioCtx.currentTime);
  g.gain.exponentialRampToValueAtTime(0.0001, audioCtx.currentTime + dur);
  o.connect(g); g.connect(audioCtx.destination);
  o.start(); o.stop(audioCtx.currentTime + dur);
}

// ---- Game state ----
const S = { start:'start', playing:'playing', win:'win', lose:'lose' };
let state = S.start;

let paddle, ball, bricks, bullets, scanner, hearts, score, wave, stealth, invuln, time, lastTs;

function reset(){
  paddle = { x: W/2, y: H-46, w: 120, h: 14, vx:0 };
  ball   = { x: W/2, y: H-70, r: 7, vx:0, vy:0, stuck:true, speed:380 };
  bullets = [];
  hearts = 3; score = 0; wave = 1; stealth = 100; invuln = 0; time = 0;

}

function makeBricks(w){
  const cols = 8, rows = 5;
  const margin = 24, top = 64, gap = 6, bh = 24;
  const bw = (W - margin*2 - (cols-1)*gap)/cols;
  bricks = [];
  for(let r=0;r<rows;r++){
    for(let c=0;c<cols;c++){
      let type='normal';
      // guards appear in lower rows and grow with wave
      const guardChance = Math.min(0.15 + w*0.06, 0.5);
      const roll = Math.random();
      if(r>=2 && roll < guardChance) type='guard';
      else if(roll < guardChance + 0.18) type='armor';
      bricks.push({ x: margin + c*(bw+gap), y: top + r*(bh+gap), w:bw, h:bh, hp: type==='armor'?2:1, type, alive:true, fireCd: 1 + Math.random()*3 });
    }
  }
}

function handleKey(k){
  if(state===S.start){
    if(k===' '||k==='Enter'){ startGame(); }
    return;
  }
  if(k==='r'||k==='R'){ reset(); state=S.playing; ball.stuck=true; ball.x=paddle.x; ball.y=paddle.y-paddle.h-10; return; }
  if(state===S.lose){
    if(k===' '||k==='Enter'){ reset(); state=S.playing; ball.stuck=true; ball.x=paddle.x; ball.y=paddle.y-paddle.h-10; }
    return;
  }
  if(state===S.win){
    // next wave on space / enter
    if(k===' '||k==='Enter'){ nextWave(); }
    return;
  }
  if(state===S.playing && k===' '){ launchBall(); }
}

function startGame(){ reset(); state=S.playing; ball.stuck=true; ball.x=paddle.x; ball.y=paddle.y-paddle.h-10; sfx.launch&&0; }

function nextWave(){ wave++; makeBricks(wave); bullets=[]; ball.speed = 380 + (wave-1)*24; ball.stuck=true; ball.x=paddle.x; ball.y=paddle.y-paddle.h-10; state=S.playing; invuln=1.2; }

function launchBall(){
  if(!ball.stuck) return;
  ball.stuck=false;
  const a = -Math.PI/2 + (Math.random()*0.6 - 0.3);
  ball.vx = Math.cos(a)*ball.speed;
  ball.vy = Math.sin(a)*ball.speed;
  sfx.launch();
}

function loseHeart(){
  if(invuln>0) return;
  hearts--; invuln = 1.0; sfx.hurt();
  if(hearts<=0){ state=S.lose; saveHigh(); sfx.lose(); }
}

// ---- Update ----
function update(dt){
  time += dt;
  // stealth meter
  const cloaked = (keys['Shift']||keys['ShiftLeft']||keys['ShiftRight']) && state===S.playing;
  if(state!==S.playing) return;

  if(cloaked && stealth>0){ stealth = Math.max(0, stealth - 42*dt); }
  else { stealth = Math.min(100, stealth + 26*dt); }
  const canCloak = cloaked && stealth>0.5;

  // paddle width depends on cloak
  paddle.w = canCloak ? 72 : 120;

  // paddle movement (keys velocity)
  let mv = 0;
  if(keys['ArrowLeft']) mv -= 1;
  if(keys['ArrowRight']) mv += 1;
  if(mv!==0){ paddle.x += mv*560*dt; pointerX=null; }
  else if(pointerX!==null){ const t=pointerX; paddle.x += (t - paddle.x)*Math.min(1, dt*22); }
  paddle.x = Math.max(paddle.w/2+8, Math.min(W-8-paddle.w/2, paddle.x));

  // scanner motion + blink
  const per = 5.2; const tt=(time % per)/per;
  if(tt<0.5) scanner.x = 30 + (W-60)*(tt*2); else scanner.x = 30 + (W-60)*(1-(tt-0.5)*2);
  scanner.blinkPhase = (time*1.4)%1;
  scanner.on = scanner.blinkPhase < 0.72;

  // invulnerability timer
  if(invuln>0) invuln=Math.max(0,invuln-dt);

  // guard bricks fire bullets
  for(const b of bricks){
    if(b.alive && b.type==='guard'){
      b.fireCd -= dt;
      if(b.fireCd<=0){
        b.fireCd = 2.4 + Math.random()*3.5 - Math.min(wave*0.15,1.5);
        bullets.push({ x:b.x+b.w/2, y:b.y+b.h, vy:230+wave*12, r:4 });
      }
    }
  }

  // ball physics
  if(ball.stuck){ ball.x=paddle.x; ball.y=paddle.y-paddle.h-10; }
  else {
    ball.x += ball.vx*dt; ball.y += ball.vy*dt;
    // walls
    if(ball.x-ball.r<4){ ball.x=ball.r+4; ball.vx=Math.abs(ball.vx); sfx.wall(); }
    if(ball.x+ball.r>W-4){ ball.x=W-4-ball.r; ball.vx=-Math.abs(ball.vx); sfx.wall(); }
    if(ball.y-ball.r<4){ ball.y=ball.r+4; ball.vy=Math.abs(ball.vy); sfx.wall(); }

    // paddle collision (only when not falling off)
    if(ball.vy>0 && ball.y+ball.r>=paddle.y && ball.y-ball.r<=paddle.y+paddle.h){
      const within = Math.abs(ball.x - paddle.x) <= paddle.w/2 + ball.r;
      if(within){
        const rel = (ball.x - paddle.x)/(paddle.w/2);
        const ang = (-Math.PI/2) + rel*1.05; // up to ~60deg
        ball.vx = Math.cos(ang)*ball.speed;
        ball.vy = Math.sin(ang)*ball.speed;
        if(ball.vy>-40) ball.vy=-40;
        sfx.bounce();
      }
    }

    // bricks
    for(const b of bricks){
      if(!b.alive) continue;
      if(hitBrick(b)){ 
        b.hp--;
        if(b.hp<=0){
          b.alive=false;
          score += b.type==='guard'?40 : b.type==='armor'?25 : 10;
          sfx[b.type==='armor'?'armor':(b.type==='guard'?'guard':'brick')]();
        } else { sfx.armor(); }
      }
    }

    // ball off bottom
    if(ball.y-ball.r>H+20){ loseHeart(); if(state===S.playing){ ball.stuck=true; ball.x=paddle.x; ball.y=paddle.y-paddle.h-10; } }
  }

  // bullets
  for(const bl of bullets){ bl.y += bl.vy*dt; }
  bullets = bullets.filter(b=>{
    if(b.y>H+20) return false;
    if(!canCloak && invuln<=0){
      if(Math.abs(b.x-paddle.x)<=paddle.w/2+b.r && b.y>=paddle.y-b.r && b.y<=paddle.y+paddle.h+b.r){ loseHeart(); return false; }
    }
    return true;
  });

  // scanner damage
  if(scanner.on && !canCloak && invuln<=0){
    const beamW=8;
    if(Math.abs(scanner.x - paddle.x) <= (paddle.w/2 + beamW/2)){ loseHeart(); }
  }

  // win check
  if(bricks.every(b=>!b.alive)){
    state=S.win; highScore=Math.max(highScore,score); saveHigh(); sfx.win();
  }
}

function hitBrick(b){
  const cx=ball.x, cy=ball.y, r=ball.r;
  const nx = Math.max(b.x, Math.min(cx,b.x+b.w));
  const ny = Math.max(b.y, Math.min(cy,b.y+b.h));
  const dx=cx-nx, dy=cy-ny;
  if(dx*dx+dy*dy > r*r) return false;
  // bounce by side
  const fromLeft = cx < b.x + b.w/2;
  const fromTop  = cy < b.y + b.h/2;
  if(fromTop===fromLeft){ ball.vx=-ball.vx; } else { ball.vy=-ball.vy; }
  return true;
}

  if(state===S.playing){ drawPlayfield(); }
  else { drawPlayfield(true); drawOverlay(); }
}

      // eye
      ctx.fillStyle='#fff'; ctx.beginPath(); ctx.arc(b.x+b.w/2, b.y+b.h/2, 4,0,7); ctx.fill();
      ctx.fillStyle='#7a1020'; ctx.beginPath(); ctx.arc(b.x+b.w/2, b.y+b.h/2, 2,0,7); ctx.fill();
    }
    if(b.type==='armor' && b.hp>1){
      ctx.strokeStyle='rgba(255,255,255,0.5)'; ctx.lineWidth=1; 
      for(let i=1;i<3;i++){ const yy=b.y+i*(b.h/3); ctx.beginPath(); ctx.moveTo(b.x+4,yy); ctx.lineTo(b.x+b.w-4,yy); ctx.stroke(); }
    }
  }

  // scanner beam
  if(state===S.playing && scanner.on){
    const grd=ctx.createLinearGradient(scanner.x-10,0,scanner.x+10,0);
    grd.addColorStop(0,'rgba(255,60,80,0)');
    grd.addColorStop(0.5,'rgba(255,60,80,0.9)');
    grd.addColorStop(1,'rgba(255,60,80,0)');
    ctx.fillStyle=grd; ctx.fillRect(scanner.x-10,4,20,H-8);
    // emitters
    ctx.fillStyle='#ff3c50';
    ctx.beginPath(); ctx.arc(scanner.x,10,7,0,7); ctx.fill();
    ctx.beginPath(); ctx.arc(scanner.x,H-10,7,0,7); ctx.fill();
  } else if(state===S.playing){
    // standby faint line
    ctx.fillStyle='rgba(255,60,80,0.12)'; ctx.fillRect(scanner.x-3,4,6,H-8);
  }

  // bullets
  for(const b of bullets){
    ctx.fillStyle='#ffd23f'; ctx.beginPath(); ctx.arc(b.x,b.y,b.r,0,7); ctx.fill();
    ctx.fillStyle='rgba(255,210,63,0.4)'; ctx.beginPath(); ctx.arc(b.x,b.y-b.vy*0.02,b.r*1.6,0,7); ctx.fill();
  }

  // ball
  if(ball && state!==S.start){
    ctx.save();
    ctx.shadowColor='rgba(150,240,255,0.9)'; ctx.shadowBlur=16;
    const bg=ctx.createRadialGradient(ball.x-2,ball.y-2,1,ball.x,ball.y,ball.r);
    bg.addColorStop(0,'#eaffff'); bg.addColorStop(1,'#37b6ff');
    ctx.fillStyle=bg; ctx.beginPath(); ctx.arc(ball.x,ball.y,ball.r,0,7); ctx.fill();
    ctx.restore();
  }

  // paddle / cloak
  if(paddle){
    const canCloak=(keys['Shift']||keys['ShiftLeft']||keys['ShiftRight']) && state===S.playing;
    let alpha = invuln>0 ? (Math.floor(time*20)%2?0.4:1) : 1;
    ctx.save();
    if(canCloak){ ctx.globalAlpha=0.5; ctx.shadowColor='rgba(180,120,255,0.9)'; ctx.shadowBlur=22; }
    else { ctx.shadowColor='rgba(90,220,255,0.7)'; ctx.shadowBlur=12; }
    const pg=ctx.createLinearGradient(0,paddle.y,0,paddle.y+paddle.h);
    if(canCloak){ pg.addColorStop(0,'#c9a6ff'); pg.addColorStop(1,'#5b3fb0'); }
    else { pg.addColorStop(0,'#dff4ff'); pg.addColorStop(1,'#2f8fd0'); }
    ctx.fillStyle=pg; roundRect(paddle.x-paddle.w/2,paddle.y,paddle.w,paddle.h,6); ctx.fill();
    ctx.restore();
  }

  // HUD
  drawHUD();
}

  ctx.fillStyle='rgba(255,255,255,0.5)'; ctx.textAlign='left'; ctx.font='700 11px system-ui, sans-serif';

function shield(x,y){
  ctx.beginPath();
  ctx.moveTo(x,y-6); ctx.lineTo(x+7,y-3); ctx.lineTo(x+7,y+3);
  ctx.quadraticCurveTo(x+7,y+9,x,y+12); ctx.quadraticCurveTo(x-7,y+9,x-7,y+3);
  ctx.lineTo(x-7,y-3); ctx.closePath(); ctx.fill();
}

function roundRect(x,y,w,h,r){ 
  if(w< r*2) r=w/2; if(h<r*2) r=h/2;
  ctx.beginPath();
  ctx.moveTo(x+r,y);
  ctx.arcTo(x+w,y,x+w,y+h,r);
  ctx.arcTo(x+w,y+h,x,y+h,r);
  ctx.arcTo(x,y+h,x,y,r);
  ctx.arcTo(x,y,x+w,y,r);
  ctx.closePath();
}
