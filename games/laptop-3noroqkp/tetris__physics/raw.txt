function initAudio() {
if (!audioCtx) {
try { audioCtx = new (window.AudioContext || webkitAudioContext)(); } catch(e) {}
}
}
function playSound(freq, dur, type, vol) {
if (!audioCtx) return;
const osc = audioCtx.createOscillator();
const gain = audioCtx.createGain();
osc.type = type || 'square';
osc.frequency.value = freq;
gain.gain.setValueAtTime(vol || 0.15, audioCtx.currentTime);
gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
osc.connect(gain).connect(audioCtx.destination);
osc.start();
osc.stop(audioCtx.currentTime + dur);
}
function playDrop() { playSound(80, 0.15, 'sine', 0.3); }
function playRotate() { playSound(300, 0.06, 'square', 0.08); }
function playMove() { playSound(200, 0.04, 'square', 0.05); }
function playClear() { for(let i=0;i<4;i++) setTimeout(()=>playSound(400+i*100,0.1,'square',0.12),i*50); }
function playGameOver() { playSound(150, 0.5, 'sawtooth', 0.2); }

// Tetrominoes
const SHAPES = [
[], // empty
[[0,0],[1,0],[2,0],[3,0]], // I
[[0,0],[1,0],[0,1],[1,1]], // O
[[1,0],[0,1],[1,1],[2,1]], // T
[[1,0],[0,1],[1,1],[1,2]], // S
[[0,0],[1,0],[1,1],[2,1]], // Z
[[0,0],[0,1],[1,1],[0,2]], // J
[[1,0],[1,1],[0,2],[1,2]], // L
];
const COLORS = [
'', '#00f5ff', '#ffee00', '#ff69b4', '#ffa500', '#ff3333', '#3388ff', '#ff8800'
];
const GHOST_COLORS = [
'', 'rgba(0,245,255,0.3)', 'rgba(255,238,0,0.3)', 'rgba(255,105,180,0.3)',
'rgba(255,165,0,0.3)', 'rgba(255,51,51,0.3)', 'rgba(51,136,255,0.3)', 'rgba(255,136,0,0.3)'
];

function getShape(type) {
return SHAPES[type].map(c => [c[0], c[1]]);
}

// Game state
let state = 'START'; // START, PLAYING, GAMEOVER
let grid = [];
let currentPiece = null;
let nextPieces = [];
let score = 0, lines = 0, level = 1;
let dropTimer = 0;
let lastTime = 0;
let particles = [];
let screenShake = { x: 0, y: 0, intensity: 0 };
let clearAnimations = [];

function initGrid() {
grid = Array.from({ length: ROWS }, () => Array(COLS).fill(0));
}

function createPiece(type) {
const cells = getShape(type);
const cx = Math.floor((COLS - 4) / 2);
return {
type, cells: cells.map(c => [c[0], c[1]]),
x: cx, y: 0,
vx: 0, vy: 0, // physics velocity
angularVel: 0, rotAnim: 0,
bouncing: false, bounceVy: 0,
settled: false
};
}

function randomBag() {
let bag = [1,2,3,4,5,6,7];
for (let i = bag.length - 1; i > 0; i--) {
const j = Math.floor(Math.random() * (i + 1));
[bag[i], bag[j]] = [bag[j], bag[i]];
}
return bag;
}

function getNextPiece() {
if (nextPieces.length < 5) {
let newBag = randomBag();
nextPieces = nextPieces.concat(newBag);
}
return nextPieces.shift();
}

function spawnPiece() {
const type = getNextPiece();
currentPiece = createPiece(type);
currentPiece.vy = 0;
currentPiece.vx = 0;
if (collides(currentPiece)) {
state = 'GAMEOVER';
playGameOver();
}
}

function getCells(piece) {
return piece.cells.map(c => [piece.x + c[0], piece.y + c[1]]);
}

function collides(piece, offX, offY) {
const ox = offX || 0, oy = offY || 0;
for (let [cx, cy] of getCells(piece)) {
const nx = cx + ox, ny = cy + oy;
if (nx < 0 || nx >= COLS || ny >= ROWS) return true;
if (ny < 0) continue;
if (grid[ny][nx]) return true;
}
return false;
}

function rotate(piece, dir) {
const cells = getCells(piece);
let cx = 0, cy = 0;
for (let [x, y] of cells) { cx += x; cy += y; }
cx /= 4; cy /= 4;
let newCells = cells.map(([x, y]) => {
const dx = x - cx, dy = y - cy;
let rx, ry;
if (dir === 1) { rx = -dy; ry = dx; }
else { rx = dy; ry = -dx; }
return [Math.round(cx + rx), Math.round(cy + ry)];
});
// Normalize
let minX = Infinity, minY = Infinity;
for (let [x,y] of newCells) { if(x<minX)minX=x; if(y<minY)minY=y; }
newCells = newCells.map(([x,y]) => [x-minX, y-minY]);

// Wall kicks
const kicks = [[0,0],[1,0],[-1,0],[0,-1],[1,-1],[-1,-1]];
for (let [kx, ky] of kicks) {
let testPiece = {...piece, cells: newCells};
if (!collides(testPiece, kx, ky)) {
// Check if this would overlap settled blocks
let valid = true;
let testX = piece.x + kx, testY = piece.y + ky;
for (let [cx2, cy2] of newCells) {
const gx = testX + cx2, gy = testY + cy2;
if (gy >= 0 && gy < ROWS && gx >= 0 && gx < COLS && grid[gy][gx]) { valid = false; break; }
}
if (valid) {
piece.cells = newCells;
piece.x += kx;
piece.y += ky;
piece.rotAnim = 1;
playRotate();
return true;
}
}
}
return false;
}

function getGhostY(piece) {
let py = piece.y;
while (!collides({...piece, y: py + 1}, 0, 0)) py++;
return py;
}

function lockPiece() {
for (let [cx, cy] of getCells(currentPiece)) {
if (cy >= 0 && cy < ROWS && cx >= 0 && cx < COLS) {
grid[cy][cx] = currentPiece.type;
}
}
checkLines();
playDrop();
screenShake.intensity = 3;
spawnParticles(currentPiece);
setTimeout(spawnPiece, 100);
currentPiece.settled = true;
}

function checkLines() {
let cleared = [];
for (let r = ROWS - 1; r >= 0; r--) {
if (grid[r].every(c => c !== 0)) cleared.push(r);
}
if (cleared.length > 0) {
playClear();
clearAnimations.push({ rows: [...cleared], timer: 0 });
for (let r of cleared) {
for (let c = 0; c < COLS; c++) grid[r][c] = 0;
}
// Remove cleared rows and add empty at top
for (let i = 0; i < cleared.length; i++) {
grid.splice(cleared[i], 1);
grid.unshift(Array(COLS).fill(0));
}
lines += cleared.length;
let points = [0, 40, 100, 300, 1200][cleared.length];
score += points * level;
level = Math.floor(lines / 10) + 1;
}
}

function spawnParticles(piece) {
for (let [cx, cy] of getCells(piece)) {
if (cy >= 0 && cy < ROWS) {
for (let i = 0; i < 3; i++) {
particles.push({
x: (piece.x + cx) * CELL + CELL/2,
y: (piece.y + cy) * CELL + CELL/2,
vx: (Math.random()-0.5)*80,
vy: -Math.random()*60,
life: 0.5,
color: COLORS[piece.type]
});
}
}
}
}

function gravity() {
return 300 + level * 40; // px/s^2
}

function update(dt) {
if (state !== 'PLAYING' || !currentPiece || currentPiece.settled) return;

// Physics: gravity
let g = gravity();
currentPiece.vy += g * dt;
if (currentPiece.vy > 800) currentPiece.vy = 800; // terminal velocity

// Horizontal inertia decay
currentPiece.vx *= Math.pow(0.05, dt);
if (Math.abs(currentPiece.vx) < 1) currentPiece.vx = 0;

// Apply movement
let newY = currentPiece.y + currentPiece.vy * dt / CELL;
let newX = currentPiece.x + currentPiece.vx * dt / CELL;

// Try horizontal
if (newX !== currentPiece.x) {
let dx = Math.sign(newX - currentPiece.x);
if (!collides(currentPiece, dx, 0)) {
currentPiece.x += dx;
currentPiece.vx *= 0.5;
} else {
currentPiece.vx = 0;
}
}

// Try vertical (with bounce physics)
let targetY = currentPiece.y + Math.round(currentPiece.vy * dt / CELL);
while (targetY !== currentPiece.y) {
let step = Math.sign(targetY - currentPiece.y);
if (!collides(currentPiece, 0, step)) {
currentPiece.y += step;
} else {
// Landing!
if (currentPiece.vy > 50) {
currentPiece.bouncing = true;
currentPiece.bounceVy = -currentPiece.vy * 0.25; // restitution
screenShake.intensity = Math.min(currentPiece.vy / 30, 5);
} else {
lockPiece();
return;
}
break;
}
targetY = currentPiece.y + Math.sign(targetY - currentPiece.y);
}

// Bounce physics
if (currentPiece.bouncing) {
currentPiece.bounceVy += g * dt;
let by = currentPiece.bounceVy * dt / CELL;
if (by >= 0 || collides(currentPiece, 0, Math.round(by))) {
currentPiece.bouncing = false;
currentPiece.vy = 0;
lockPiece();
return;
}
currentPiece.y += Math.round(by);
}

// Rotation animation decay
if (currentPiece.rotAnim > 0) {
currentPiece.rotAnim -= dt * 8;
if (currentPiece.rotAnim < 0) currentPiece.rotAnim = 0;
}
}

function updateParticles(dt) {
for (let p of particles) {
p.x += p.vx * dt;
p.y += p.vy * dt;
p.vy += 200 * dt;
p.life -= dt;
}
particles = particles.filter(p => p.life > 0);

screenShake.intensity *= Math.pow(0.01, dt);
if (screenShake.intensity < 0.1) screenShake.intensity = 0;
screenShake.x = (Math.random() - 0.5) * screenShake.intensity;
screenShake.y = (Math.random() - 0.5) * screenShake.intensity;

for (let ca of clearAnimations) {
ca.timer += dt;
}
clearAnimations = clearAnimations.filter(ca => ca.timer < 0.4);
}

function render() {
ctx.save();
ctx.translate(screenShake.x, screenShake.y);

// Background
ctx.fillStyle = '#0a0a0f';
ctx.fillRect(-10, -10, W + 20, H + 20);

let ox = (W - COLS * CELL) / 2;
let oy = Math.max(10, (H - ROWS * CELL) / 2);

if (state === 'START') {
renderStartScreen(ox, oy);
ctx.restore();
return;
}

// Grid border
ctx.strokeStyle = '#333';
ctx.lineWidth = 2;
ctx.strokeRect(ox - 4, oy - 4, COLS * CELL + 8, ROWS * CELL + 8);

// Grid cells
for (let r = 0; r < ROWS; r++) {
for (let c = 0; c < COLS; c++) {
if (grid[r][c]) {
drawCell(ox + c * CELL, oy + r * CELL, COLORS[grid[r][c]], 1);
}
}
}

// Clear animations
for (let ca of clearAnimations) {
let alpha = 1 - ca.timer / 0.4;
ctx.fillStyle = `rgba(255,255,255,${alpha * 0.8})`;
for (let r of ca.rows) {
ctx.fillRect(ox, oy + r * CELL, COLS * CELL, CELL);
}
}

// Ghost piece
if (currentPiece && !currentPiece.settled) {
let ghostY = getGhostY(currentPiece);
let ghostCells = currentPiece.cells.map(([cx, cy]) => [currentPiece.x + cx, ghostY + cy]);
for (let [gx, gy] of ghostCells) {
if (gy >= 0 && gy < ROWS) {
drawCell(ox + gx * CELL, oy + gy * CELL, GHOST_COLORS[currentPiece.type], 0.4);
}
}
}

// Current piece with physics effects
if (currentPiece && !currentPiece.settled) {
let rotOffset = currentPiece.rotAnim * 0.1;
for (let [cx, cy] of getCells(currentPiece)) {
if (cy >= 0 && cy < ROWS) {
let px = ox + cx * CELL;
let py = oy + cy * CELL;
let squish = currentPiece.bouncing ? 0.9 : 1;
ctx.save();
ctx.translate(px + CELL/2, py + CELL/2);

ctx.rotate(rotOffset * 0.1);
drawCell(-CELL/2, -CELL/2, COLORS[currentPiece.type], 1);
ctx.restore();
}
}
}

// Particles
for (let p of particles) {
ctx.globalAlpha = p.life / 0.5;
ctx.fillStyle = p.color;
ctx.fillRect(ox + p.x - 3, oy + p.y - 3, 6, 6);
}
ctx.globalAlpha = 1;

// Side panels
renderSidePanel(ox, oy);

if (state === 'GAMEOVER') {
renderGameOver(ox, oy);
}

ctx.restore();
}

function drawCell(x, y, color, alpha) {
ctx.fillStyle = color;
ctx.globalAlpha = alpha;
let pad = 2;
ctx.fillRect(x + pad, y + pad, CELL - pad * 2, CELL - pad * 2);
// Highlight
ctx.fillStyle = 'rgba(255,255,255,0.3)';

// Shadow
ctx.fillStyle = 'rgba(0,0,0,0.3)';
ctx.fillRect(x + pad, y + CELL - pad - 4, CELL - pad * 2, 4);
ctx.globalAlpha = 1;
}

function startGame() {
initGrid();
score = 0; lines = 0; level = 1;
particles = [];
clearAnimations = [];
screenShake.intensity = 0;
nextPieces = randomBag();
state = 'PLAYING';
spawnPiece();
}

// Input
let keys = {};
window.addEventListener('keydown', (e) => {
initAudio();
if (keys[e.code]) return;
keys[e.code] = true;

if (state === 'START') {
if (e.code === 'Enter' || e.code === 'Space') startGame();
return;
}

if (state === 'GAMEOVER') {
if (e.code === 'KeyR') startGame();
return;
}

if (state !== 'PLAYING' || !currentPiece || currentPiece.settled) return;

switch (e.code) {
case 'ArrowLeft':
if (!collides(currentPiece, -1, 0)) { currentPiece.x--; currentPiece.vx = -60; playMove(); }
break;
case 'ArrowRight':
if (!collides(currentPiece, 1, 0)) { currentPiece.x++; currentPiece.vx = 60; playMove(); }
break;
case 'ArrowDown':
currentPiece.vy = Math.max(currentPiece.vy, 400); // soft drop boost
break;
case 'Space': {
// Hard drop with physics - fast fall
let hardDrop = true;
while (!collides(currentPiece, 0, 1)) {
currentPiece.y++;
}
currentPiece.vy = 800;
lockPiece();
hardDrop = false;
break;
}
case 'ArrowUp':
case 'KeyZ':
rotate(currentPiece, 1);
break;
case 'KeyX':
rotate(currentPiece, -1);
break;
}
e.preventDefault();
});

window.addEventListener('keyup', (e) => { keys[e.code] = false; });

// Game loop
function gameLoop(timestamp) {
let dt = Math.min((timestamp - lastTime) / 1000, 0.05);
lastTime = timestamp;

if (state === 'PLAYING') update(dt);
updateParticles(dt);
render();
requestAnimationFrame(gameLoop);
}

initGrid();
nextPieces = randomBag();
requestAnimationFrame(gameLoop);

})();
