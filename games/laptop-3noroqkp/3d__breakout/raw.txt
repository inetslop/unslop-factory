// Audio
let audioCtx = null;
function initAudio() {
    if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
}
function playSound(freq, duration, type, vol) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = type || 'square';
    osc.frequency.value = freq;
    gain.gain.setValueAtTime(vol || 0.15, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + (duration || 0.1));
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + (duration || 0.1));
}

// Game state
const STATE = { MENU: 0, PLAYING: 1, PAUSED: 2, GAMEOVER: 3, WIN: 4 };

// 3D Projection
const CAM = { x: 0, y: -80, z: 500 };
const FOCAL = 600;

function project(x, y, z) {
    const dx = x - CAM.x;
    const dy = y - CAM.y;
    const dz = z - CAM.z;
    const scale = FOCAL / (FOCAL + dz * 0.5);
    return {
        x: W/2 + dx * scale,
        y: H/2 + dy * scale,
        scale: scale
    };
}

// Game objects

const FIELD_H = 400;

const BRICK_W = 52;

const BRICK_GAP = 4;
const BRICK_DEPTH = 20;

function toFieldX(x) { return (x - W/2) / (W * 0.7); }
function toFieldY(y) { return (y - H/2) / (H * 0.65); }

function fieldToScreen(fx, fy, fz) {

function createBricks() {
    bricks = [];
    const patterns = [
        (r, c) => true,
        (r, c) => (r + c) % 2 === 0 || r < 2,
        (r, c) => r !== 2 || c % 3 !== 1,
        (r, c) => c !== 4 && c !== 5 || r < 3,
    ];
    const pattern = patterns[(level - 1) % patterns.length];
    for (let r = 0; r < BRICK_ROWS; r++) {
        for (let c = 0; c < BRICK_COLS; c++) {
            if (pattern(r, c)) {

                const fy = (r - (BRICK_ROWS-1)/2) * (BRICK_H + BRICK_GAP) - 30;
                bricks.push({
                    x: fx, y: fy, w: BRICK_W, h: BRICK_H, depth: BRICK_DEPTH,
                    hp: level >= 3 ? 2 : 1, maxHp: level >= 3 ? 2 : 1,
                    color: getBrickColor(r),
                    alive: true,
                    breakAnim: 0
                });

        }
    }

// Update
function update(dt) {
    if (gameState !== STATE.PLAYING) return;

    // Paddle movement
    const targetX = toFieldX(mouseX);
    paddle.x += (targetX - paddle.x) * 0.15;
    paddle.x = Math.max(-FIELD_W/2 + 60, Math.min(FIELD_W/2 - 60, paddle.x));

    if (keys['ArrowLeft']) paddle.x -= 8;
    if (keys['ArrowRight']) paddle.x += 8;

    if (!ball.active) return;

    // Ball physics
    ball.x += ball.vx;
    ball.y += ball.vy;

    // Wall collisions
    if (ball.x - ball.r < -FIELD_W/2) {
        ball.x = -FIELD_W/2 + ball.r;
        ball.vx = Math.abs(ball.vx);
        playSound(200, 0.1, 'sine', 0.1);
    }
    if (ball.x + ball.r > FIELD_W/2) {
        ball.x = FIELD_W/2 - ball.r;
        ball.vx = -Math.abs(ball.vx);
        playSound(200, 0.1, 'sine', 0.1);
    }
    if (ball.y - ball.r < -FIELD_H/2) {
        ball.y = -FIELD_H/2 + ball.r;
        ball.vy = Math.abs(ball.vy);
        playSound(200, 0.1, 'sine', 0.1);
    }

    // Paddle collision
    const py = paddle.y - paddle.h/2;
    if (ball.vy > 0 && ball.y + ball.r > py && ball.y - ball.r < py + paddle.h) {
        if (Math.abs(ball.x - paddle.x) < paddle.w/2 + ball.r) {
            ball.y = py - ball.r;
            const hitPos = (ball.x - paddle.x) / (paddle.w/2);
            ball.vx = hitPos * 6;
            ball.vy = -Math.abs(ball.vy);
            playSound(400, 0.1, 'triangle', 0.15);
        }
    }

    // Brick collisions
    for (let b of bricks) {
        if (!b.alive) continue;
        if (ball.x + ball.r > b.x - b.w/2 && ball.x - ball.r < b.x + b.w/2 &&
            ball.y + ball.r > b.y - b.h/2 && ball.y - ball.r < b.y + b.h/2) {

            b.hp--;
            if (b.hp <= 0) {
                b.alive = false;
                b.breakAnim = 1;
                score += 10 * level;
                shakeTime = 0.15;
                shakeIntensity = 3;
                const sp = fieldToScreen(b.x, b.y, 0);
                spawnParticles(sp.x, sp.y, b.color, 8);
            } else {
                b.breakAnim = 0.5;
                score += 5;
            }

            // Determine bounce direction
            const overlapX = ball.r - Math.abs(ball.x - b.x);
            const overlapY = ball.r - Math.abs(ball.y - b.y);
            if (overlapX < overlapY) {
                ball.vx = ball.x > b.x ? Math.abs(ball.vx) : -Math.abs(ball.vx);
            } else {
                ball.vy = ball.y > b.y ? Math.abs(ball.vy) : -Math.abs(ball.vy);
            }

            playSound(600, 0.15, 'square', 0.2);
            break;
        }
    }

    // Ball lost
    if (ball.y + ball.r > FIELD_H/2) {
        lives--;
        playSound(150, 0.4, 'sawtooth', 0.2);
        shakeTime = 0.3;
        shakeIntensity = 5;
        if (lives <= 0) {
            gameState = STATE.GAMEOVER;
            try {
                if (score > highScore) {
                    highScore = score;
                    localStorage.setItem('breakout3d_high', String(highScore));
                }
            } catch(e) {}
        } else {
            resetBall();
        }
    }

    // Level complete
    if (bricks.every(b => !b.alive)) {
        gameState = STATE.WIN;
        playSound(800, 0.3, 'sine', 0.2);
        setTimeout(() => playSound(1000, 0.3, 'sine', 0.2), 150);
        setTimeout(() => playSound(1200, 0.4, 'sine', 0.2), 300);
    }

    // Update particles
    for (let i = particles.length - 1; i >= 0; i--) {
        const p = particles[i];
        p.x += p.vx;
        p.y += p.vy;
        p.vy += 5 * dt;
        p.life -= dt * 2;
        if (p.life <= 0) particles.splice(i, 1);
    }

    // Shake decay
    if (shakeTime > 0) shakeTime -= dt;
}

function drawWalls() {

function drawBrick(b) {
    const sc = b.breakAnim > 0 ? 1 + b.breakAnim * 0.3 : 1;
    if (b.breakAnim > 0) b.breakAnim -= 0.05;

    const cx = fieldToScreen(b.x, b.y, 0);
    const hw = (b.w / 2) * sc * 0.7;
    const hh = (b.h / 2) * sc * 0.4;
    const depth = b.depth * 0.5;

    // Top face
    ctx.fillStyle = lighten(b.color, 30);
    ctx.beginPath();
    ctx.moveTo(cx.x - hw + depth*0.3, cx.y - hh - depth);
    ctx.lineTo(cx.x + hw + depth*0.3, cx.y - hh - depth);
    ctx.lineTo(cx.x + hw, cx.y - hh);
    ctx.lineTo(cx.x - hw, cx.y - hh);
    ctx.closePath();
    ctx.fill();

    // Front face
    const grad = ctx.createLinearGradient(cx.x - hw, cx.y - hh, cx.x - hw, cx.y + hh);
    grad.addColorStop(0, lighten(b.color, 10));
    grad.addColorStop(1, darken(b.color, 20));
    ctx.fillStyle = b.hp < b.maxHp ? darken(b.color, 40) : grad;
    ctx.fillRect(cx.x - hw, cx.y - hh, hw * 2, hh * 2);

    // Right face
    ctx.fillStyle = darken(b.color, 30);
    ctx.beginPath();
    ctx.moveTo(cx.x + hw, cx.y - hh);
    ctx.lineTo(cx.x + hw + depth*0.3, cx.y - hh - depth);
    ctx.lineTo(cx.x + hw + depth*0.3, cx.y + hh - depth);
    ctx.lineTo(cx.x + hw, cx.y + hh);
    ctx.closePath();
    ctx.fill();

    // Highlight
    ctx.fillStyle = 'rgba(255,255,255,0.15)';
    ctx.fillRect(cx.x - hw + 3, cx.y - hh + 2, hw * 2 - 6, 4);

    // Crack for damaged
    if (b.hp < b.maxHp && b.alive) {
        ctx.strokeStyle = 'rgba(0,0,0,0.5)';
        ctx.lineWidth = 1;
        ctx.beginPath();
        ctx.moveTo(cx.x - hw*0.3, cx.y - hh);
        ctx.lineTo(cx.x + hw*0.2, cx.y);
        ctx.lineTo(cx.x - hw*0.1, cx.y + hh);
        ctx.stroke();
    }

    // Outline
    ctx.strokeStyle = darken(b.color, 50);
    ctx.lineWidth = 1;
    ctx.strokeRect(cx.x - hw, cx.y - hh, hw * 2, hh * 2);
}

function drawPaddle() {
    const p = fieldToScreen(paddle.x, paddle.y, 0);
    const hw = paddle.w / 2 * 0.7;
    const hh = paddle.h / 2 * 0.5;
    const depth = 12;

    // Shadow
    ctx.fillStyle = 'rgba(0,0,0,0.3)';
    ctx.beginPath();
    ctx.ellipse(p.x, p.y + hh + 8, hw * 0.9, 6, 0, 0, Math.PI * 2);
    ctx.fill();

    // Top face
    const grad = ctx.createLinearGradient(p.x - hw, p.y - hh, p.x + hw, p.y - hh);
    grad.addColorStop(0, '#4488ff');
    grad.addColorStop(0.5, '#66aaff');
    grad.addColorStop(1, '#4488ff');
    ctx.fillStyle = grad;
    ctx.beginPath();
    ctx.moveTo(p.x - hw + depth*0.2, p.y - hh - depth);
    ctx.lineTo(p.x + hw + depth*0.2, p.y - hh - depth);
    ctx.lineTo(p.x + hw, p.y - hh);
    ctx.lineTo(p.x - hw, p.y - hh);
    ctx.closePath();
    ctx.fill();

    // Front face
    const fGrad = ctx.createLinearGradient(p.x, p.y - hh, p.x, p.y + hh);
    fGrad.addColorStop(0, '#5599ff');
    fGrad.addColorStop(1, '#2255aa');
    ctx.fillStyle = fGrad;
    ctx.fillRect(p.x - hw, p.y - hh, hw * 2, hh * 2);

    // Right face
    ctx.fillStyle = '#3366cc';
    ctx.beginPath();
    ctx.moveTo(p.x + hw, p.y - hh);
    ctx.lineTo(p.x + hw + depth*0.2, p.y - hh - depth);
    ctx.lineTo(p.x + hw + depth*0.2, p.y + hh - depth);
    ctx.lineTo(p.x + hw, p.y + hh);
    ctx.closePath();
    ctx.fill();

    // Glow line on top
    ctx.strokeStyle = 'rgba(150, 220, 255, 0.8)';
    ctx.lineWidth = 2;
    ctx.beginPath();
    ctx.moveTo(p.x - hw + 4, p.y - hh);
    ctx.lineTo(p.x + hw - 4, p.y - hh);
    ctx.stroke();
}

function drawBall() {
    const p = fieldToScreen(ball.x, ball.y, 0);
    const r = ball.r * 0.7;

    // Shadow
    ctx.fillStyle = 'rgba(0,0,0,0.2)';
    ctx.beginPath();
    ctx.ellipse(p.x, p.y + r + 4, r * 0.8, r * 0.3, 0, 0, Math.PI * 2);
    ctx.fill();

    // Ball gradient
    const grad = ctx.createRadialGradient(p.x - r*0.3, p.y - r*0.3, r*0.1, p.x, p.y, r);
    grad.addColorStop(0, '#ffffff');
    grad.addColorStop(0.5, '#aaddff');
    grad.addColorStop(1, '#4488cc');
    ctx.fillStyle = grad;
    ctx.beginPath();
    ctx.arc(p.x, p.y, r, 0, Math.PI * 2);
    ctx.fill();

    // Specular highlight
    ctx.fillStyle = 'rgba(255,255,255,0.7)';
    ctx.beginPath();
    ctx.arc(p.x - r*0.3, p.y - r*0.3, r*0.25, 0, Math.PI * 2);
    ctx.fill();

    // Trail
    const trailLen = 5;
    for (let i = 1; i <= trailLen; i++) {
        const tx = p.x - ball.vx * i * 3;
        const ty = p.y - ball.vy * i * 3;
        ctx.globalAlpha = 0.15 / i;
        ctx.fillStyle = '#88ccff';
        ctx.beginPath();
        ctx.arc(tx, ty, r * (1 - i / trailLen), 0, Math.PI * 2);
        ctx.fill();
    }
    ctx.globalAlpha = 1;
}

function drawHUD() {
    ctx.fillStyle = 'rgba(0,0,0,0.5)';

        ctx.font = '18px monospace';

    }
}

function lighten(color, amt) {
    return `rgb(${Math.min(255, color[0]+amt)},${Math.min(255,color[1]+amt)},${Math.min(255,color[2]+amt)})`;
}
function darken(color, amt) {
    return `rgb(${Math.max(0, color[0]-amt)},${Math.max(0,color[1]-amt)},${Math.max(0,color[2]-amt)})`;
}

// Game loop
let lastTime = 0;
function gameLoop(time) {
    const dt = Math.min((time - lastTime) / 1000, 0.05);
    lastTime = time;

    update(dt);
    render();
    requestAnimationFrame(gameLoop);
}
