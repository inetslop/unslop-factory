function playSound(freq, dur, type, vol) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = type || 'sine';
    osc.frequency.value = freq;
    gain.gain.setValueAtTime(vol || 0.3, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + dur);
}

function getLevelWalls(level) {
    const walls = [];

    walls.push({ x: 0, y: H - t, w: W, h: t });

    const lw = levels[level];
    for (const w of lw.walls) {
        walls.push({
            x: w.x * W,
            y: w.y * H,
            w: Math.max(w.w * W, 10),
            h: Math.max(w.h * H, 10)

    return walls;

// Particles
function spawnParticles(x, y, color, count) {
    for (let i = 0; i < count; i++) {
        const angle = Math.random() * Math.PI * 2;
        const speed = 1 + Math.random() * 4;
        particles.push({
            x, y,
            vx: Math.cos(angle) * speed,
            vy: Math.sin(angle) * speed,
            life: 1,
            decay: 0.02 + Math.random() * 0.03,
            color,
            size: 2 + Math.random() * 4
        });
    }
}

function updateParticles() {
    for (let i = particles.length - 1; i >= 0; i--) {
        const p = particles[i];
        p.x += p.vx;
        p.y += p.vy;
        p.vy += 0.1;
        p.life -= p.decay;
        if (p.life <= 0) particles.splice(i, 1);
    }
}

function drawParticles() {
    for (const p of particles) {
        ctx.globalAlpha = p.life;
        ctx.fillStyle = p.color;
        ctx.beginPath();
        ctx.arc(p.x, p.y, p.size * p.life, 0, Math.PI * 2);
        ctx.fill();
    }
    ctx.globalAlpha = 1;
}

// Physics
function updatePhysics() {
    // Input force
    if (keys['ArrowLeft'] || keys['KeyA']) ball.vx -= FORCE;
    if (keys['ArrowRight'] || keys['KeyD']) ball.vx += FORCE;
    if (keys['ArrowUp'] || keys['KeyW']) ball.vy -= FORCE;
    if (keys['ArrowDown'] || keys['KeyS']) ball.vy += FORCE;

    // Gravity
    ball.vy += GRAVITY * 0.3;

    // Friction
    ball.vx *= FRICTION;
    ball.vy *= FRICTION;

    // Speed cap
    const maxSpeed = 12;
    const speed = Math.sqrt(ball.vx * ball.vx + ball.vy * ball.vy);
    if (speed > maxSpeed) {
        ball.vx = (ball.vx / speed) * maxSpeed;
        ball.vy = (ball.vy / speed) * maxSpeed;
    }

    // Move
    ball.x += ball.vx;
    ball.y += ball.vy;

    // Wall collisions
    const walls = getLevelWalls(currentLevel);
    for (const wall of walls) {
        const closestX = Math.max(wall.x, Math.min(ball.x, wall.x + wall.w));
        const closestY = Math.max(wall.y, Math.min(ball.y, wall.y + wall.h));
        const dx = ball.x - closestX;
        const dy = ball.y - closestY;
        const dist = Math.sqrt(dx * dx + dy * dy);

        if (dist < ball.r) {
            const nx = dx / (dist || 1);
            const ny = dy / (dist || 1);
            const overlap = ball.r - dist;
            ball.x += nx * overlap;
            ball.y += ny * overlap;

            const dot = ball.vx * nx + ball.vy * ny;
            if (dot < 0) {
                ball.vx -= 2 * dot * nx * BOUNCE;
                ball.vy -= 2 * dot * ny * BOUNCE;
                if (Math.abs(dot) > 1) {
                    playSound(150 + Math.random() * 50, 0.1, 'square', 0.1);
                    spawnParticles(ball.x, ball.y, '#ff6b35', 3);
                }
            }
        }
    }

    // Goal check
    const lw = levels[currentLevel];
    const gx = lw.goal.x * W;
    const gy = lw.goal.y * H;
    const gr = 20;
    const gdx = ball.x - gx;
    const gdy = ball.y - gy;
    if (Math.sqrt(gdx * gdx + gdy * gdy) < ball.r + gr) {
        gameState = STATE.LEVEL_COMPLETE;
        const levelTime = (Date.now() - levelStartTime) / 1000;
        totalScore += Math.max(0, Math.floor(1000 - levelTime * 50));
        spawnParticles(gx, gy, '#4dff4d', 30);
        playSound(523, 0.3, 'sine', 0.4);
        setTimeout(() => playSound(659, 0.3, 'sine', 0.4), 100);
        setTimeout(() => playSound(784, 0.5, 'sine', 0.4), 200);
    }

    timer = (Date.now() - levelStartTime) / 1000;
}
