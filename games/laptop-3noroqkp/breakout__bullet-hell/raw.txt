<script>
const canvas = document.getElementById('game');
const ctx = canvas.getContext('2d');

let W, H;
function resize() {
    W = canvas.width = window.innerWidth;
    H = canvas.height = window.innerHeight;
}
window.addEventListener('resize', resize);
resize();

// Audio
let audioCtx = null;
function initAudio() {
    if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
}
function playSound(freq, dur, type, vol) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = type || 'square';
    osc.frequency.value = freq;
    gain.gain.setValueAtTime(vol || 0.1, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + dur);
}

// Game state
let state = 'start'; // start, playing, gameover, waveTransition
let score = 0;
let lives = 3;
let wave = 1;
let shakeX = 0, shakeY = 0, shakeMag = 0;
let particles = [];
let playerBullets = [];
let enemyBullets = [];
let bricks = [];
let popups = [];
let fireRate = 0;
let waveTimer = 0;
let transitionText = '';

// Paddle
const paddle = { x: 0, w: 80, h: 12, y: 0 };

// Input
let keys = {};
let mouseX = 0;
window.addEventListener('keydown', e => { keys[e.code] = true; if (state === 'start') startGame(); if (state === 'gameover' && e.code === 'KeyR') startGame(); });
window.addEventListener('keyup', e => { keys[e.code] = false; });
canvas.addEventListener('mousemove', e => { mouseX = e.clientX / window.innerWidth * W; });

function startGame() {
    initAudio();
    state = 'playing';
    score = 0;
    lives = 3;
    wave = 1;
    playerBullets = [];
    enemyBullets = [];
    bricks = [];
    particles = [];
    popups = [];
    paddle.x = W / 2 - paddle.w / 2;
    paddle.y = H - 40;
    spawnWave();
}

function spawnWave() {
    bricks = [];
    const patterns = ['grid', 'diamond', 'spiral', 'fortress'];
    const pattern = patterns[(wave - 1) % patterns.length];
    
    if (pattern === 'grid') {
        const cols = 8, rows = Math.min(4 + wave, 8);
        for (let r = 0; r < rows; r++) {
            for (let c = 0; c < cols; c++) {
                bricks.push(createBrick(
                    W * 0.15 + c * (W * 0.7 / cols),
                    60 + r * 35,
                    W * 0.7 / cols - 8, 28,
                    getBrickType(r, c)
                ));
            }
        }
    } else if (pattern === 'diamond') {
        const size = Math.min(3 + wave, 6);
        for (let r = 0; r < size * 2 - 1; r++) {
            const offset = Math.abs(r - (size - 1));
            const count = size - offset;
            for (let c = 0; c < count; c++) {
                const x = W / 2 + (c - (count - 1) / 2) * 45 + offset * 45 - offset;
                bricks.push(createBrick(x, 60 + r * 38, 38, 30, getBrickType(r, c)));
            }
        }
    } else if (pattern === 'spiral') {
        const count = Math.min(20 + wave * 5, 40);
        for (let i = 0; i < count; i++) {
            const angle = (i / count) * Math.PI * 2 * 1.5;
            const radius = 80 + i * 3;
            const x = W / 2 + Math.cos(angle) * radius - 19;
            const y = 80 + Math.sin(angle) * radius * 0.6;
            if (y > 30 && y < H * 0.4 && x > 30 && x < W - 60) {
                bricks.push(createBrick(x, y, 36, 28, getBrickType(i, i)));
            }
        }
    } else if (pattern === 'fortress') {
        const layers = Math.min(3 + wave, 5);
        for (let l = 0; l < layers; l++) {
            const w_ = W * (0.6 - l * 0.08);
            const h_ = W_ * 0.12;
            const x = (W - w_) / 2;
            const y = 50 + l * 40;
            bricks.push(createBrick(x, y, w_, 32, 'wall'));
        }
    }
}

function getBrickType(r, c) {
    if (r === 0 && c % 3 === 0) return 'shooter';
    if (r === 1 && c % 4 === 0) return 'spreader';
    if (r === 2 && c % 5 === 0) return 'spiral';
    return 'normal';
}

function createBrick(x, y, w, h, type) {
    const hp = type === 'wall' ? 8 : (type === 'shooter' ? 3 : type === 'spreader' ? 2 : 1);
    return { x, y, w, h, hp, maxHp: hp, type, fireTimer: Math.random() * 60, angle: 0, dead: false };
}

function update(dt) {
    if (state !== 'playing') return;
    
    // Paddle movement
    const speed = 500;
    if (keys['ArrowLeft'] || keys['KeyA']) paddle.x -= speed * dt;
    if (keys['ArrowRight'] || keys['KeyD']) paddle.x += speed * dt;
    if (Math.abs(mouseX - W/2) > 10 || keys['ArrowLeft'] || keys['ArrowRight']) {
        // Use mouse position if it's not centered (mouse is being used)
    }
    paddle.x = Math.max(0, Math.min(W - paddle.w, paddle.x));
    
    // Auto-fire
    fireRate += dt;
    const rate = 0.12;
    if (fireRate >= rate) {
        fireRate = 0;
        playerBullets.push({ x: paddle.x + paddle.w / 2 - 8, y: paddle.y, vy: -600, r: 3 });
        playerBullets.push({ x: paddle.x + paddle.w / 2 + 8, y: paddle.y, vy: -600, r: 3 });
        if (wave > 2) {
            playerBullets.push({ x: paddle.x + paddle.w / 2, y: paddle.y - 10, vy: -550, vx: -40, r: 3 });
            playerBullets.push({ x: paddle.x + paddle.w / 2, y: paddle.y - 10, vy: -550, vx: 40, r: 3 });
        }
        playSound(800, 0.05, 'square', 0.03);
    }
    
    // Update player bullets
    for (let i = playerBullets.length - 1; i >= 0; i--) {
        const b = playerBullets[i];
        b.y += b.vy * dt;
        if (b.vx) b.x += b.vx * dt;
        if (b.y < -20 || b.x < -20 || b.x > W + 20) {
            playerBullets.splice(i, 1);
            continue;
        }
        // Hit bricks
        let hit = false;
        for (let j = bricks.length - 1; j >= 0; j--) {
            const br = bricks[j];
            if (br.dead) continue;
            if (b.x + b.r > br.x && b.x - b.r < br.x + br.w && b.y + b.r > br.y && b.y - b.r < br.y + br.h) {
                br.hp--;
                hit = true;
                if (br.hp <= 0) {
                    br.dead = true;
                    bricks.splice(j, 1);
                    score += 10 * wave;
                    spawnExplosion(br.x + br.w/2, br.y + br.h/2, '#ff6644');
                    // Drop bullets on death (bullet hell!)
                    const dropCount = 3 + Math.floor(wave / 2);
                    for (let k = 0; k < dropCount; k++) {
                        const angle = (k / dropCount) * Math.PI * 2;
                        enemyBullets.push({ x: br.x + br.w/2, y: br.y + br.h/2, vx: Math.cos(angle) * 150, vy: Math.sin(angle) * 150 + 50, r: 4 });
                    }
                    playSound(200, 0.15, 'sawtooth', 0.08);
                } else {
                    playSound(400, 0.05, 'square', 0.04);
                }
                break;
            }
        }
        if (hit) playerBullets.splice(i, 1);
    }
    
    // Update bricks (they fire back!)
    for (let i = 0; i < bricks.length; i++) {
        const br = bricks[i];
        if (br.dead) continue;
        br.fireTimer -= dt;
        br.angle += dt * 2;
        
        if (br.fireTimer <= 0) {
            const fireInterval = br.type === 'shooter' ? 1.5 : br.type === 'spreader' ? 2 : br.type === 'spiral' ? 0.8 : 3;
            br.fireTimer = fireInterval + Math.random() * 0.5;
            
            const cx = br.x + br.w / 2;
            const cy = br.y + br.h / 2;
            
            if (br.type === 'shooter') {
                // Aimed shot at player
                const dx = paddle.x + paddle.w/2 - cx;
                const dy = paddle.y - cy;
                const dist = Math.sqrt(dx*dx + dy*dy) || 1;
                enemyBullets.push({ x: cx, y: cy, vx: (dx/dist) * 250, vy: (dy/dist) * 250, r: 5 });
            } else if (br.type === 'spreader') {
                // Spread shot downward
                for (let a = -1; a <= 1; a++) {
                    enemyBullets.push({ x: cx, y: cy + br.h/2, vx: a * 80, vy: 180, r: 4 });
                }
            } else if (br.type === 'spiral') {
                // Rotating shot
                const angle = br.angle;
                enemyBullets.push({ x: cx, y: cy, vx: Math.cos(angle) * 200, vy: Math.sin(angle) * 200 + 80, r: 4 });
            } else {
                // Normal - slow drift down
                if (Math.random() < 0.3) {
                    enemyBullets.push({ x: cx, y: cy + br.h/2, vx: (Math.random()-0.5)*60, vy: 120, r: 4 });
                }
            }
        }
    }
    
    // Update enemy bullets
    for (let i = enemyBullets.length - 1; i >= 0; i--) {
        const b = enemyBullets[i];
        b.x += b.vx * dt;
        b.y += b.vy * dt;
        
        if (b.y > H + 30 || b.y < -50 || b.x < -50 || b.x > W + 50) {
            enemyBullets.splice(i, 1);
            continue;
        }
        
        // Hit paddle
        if (b.x + b.r > paddle.x && b.x - b.r < paddle.x + paddle.w && b.y + b.r > paddle.y && b.y - b.r < paddle.y + paddle.h) {
            enemyBullets.splice(i, 1);
            lives--;
            shakeMag = 8;
            spawnExplosion(paddle.x + paddle.w/2, paddle.y, '#ff0044');
            playSound(100, 0.3, 'sawtooth', 0.15);
            if (lives <= 0) {
                state = 'gameover';
            }
        }
    }
    
    // Check wave clear
    if (bricks.length === 0 && state === 'playing') {
        wave++;
        score += 100 * wave;
        state = 'waveTransition';
        waveTimer = 2;
        transitionText = `WAVE ${wave}`;
        playSound(600, 0.3, 'sine', 0.1);
    }
    
    // Update particles
    for (let i = particles.length - 1; i >= 0; i--) {
        const p = particles[i];
        p.x += p.vx * dt;
        p.y += p.vy * dt;
        p.life -= dt;
        if (p.life <= 0) particles.splice(i, 1);
    }
    
    // Update popups
    for (let i = popups.length - 1; i >= 0; i--) {
        const p = popups[i];
        p.y -= 40 * dt;
        p.life -= dt;
        if (p.life <= 0) popups.splice(i, 1);
    }
    
    // Screen shake decay
    if (shakeMag > 0) {
        shakeX = (Math.random() - 0.5) * shakeMag;
        shakeY = (Math.random() - 0.5) * shakeMag;
        shakeMag *= 0.85;
        if (shakeMag < 0.5) { shakeMag = 0; shakeX = 0; shakeY = 0; }
    }
}

function spawnExplosion(x, y, color) {
    for (let i = 0; i < 12; i++) {
        const angle = Math.random() * Math.PI * 2;
        const speed = 50 + Math.random() * 200;
        particles.push({ x, y, vx: Math.cos(angle)*speed, vy: Math.sin(angle)*speed, life: 0.3 + Math.random()*0.4, color });
    }
}

function render() {
    ctx.save();
    ctx.translate(shakeX, shakeY);
    
    // Background
    ctx.fillStyle = '#0a0a1a';
    ctx.fillRect(-10, -10, W + 20, H + 20);
    
    // Grid lines (subtle)
    ctx.strokeStyle = 'rgba(30, 40, 80, 0.3)';
    ctx.lineWidth = 1;
    for (let x = 0; x < W; x += 50) {
        ctx.beginPath(); ctx.moveTo(x, 0); ctx.lineTo(x, H); ctx.stroke();
    }
    for (let y = 0; y < H; y += 50) {
        ctx.beginPath(); ctx.moveTo(0, y); ctx.lineTo(W, y); ctx.stroke();
    }
    
    if (state === 'start') {
        renderStartScreen();
        ctx.restore();
        return;
    }
    
    if (state === 'gameover') {
        renderGameOver();
        ctx.restore();
        return;
    }
    
    // Bricks
    for (const br of bricks) {
        let color;
        switch(br.type) {
            case 'shooter': color = '#ff4466'; break;
            case 'spreader': color = '#ffaa22'; break;
            case 'spiral': color = '#aa44ff'; break;
            case 'wall': color = '#44aaff'; break;
            default: color = '#44ff88'; break;
        }
        const alpha = 0.3 + 0.7 * (br.hp / br.maxHp);
        ctx.fillStyle = color;
        ctx.globalAlpha = alpha;
        ctx.fillRect(br.x, br.y, br.w, br.h);
        ctx.globalAlpha = 1;
        ctx.strokeStyle = color;
        ctx.lineWidth = 1;
        ctx.strokeRect(br.x, br.y, br.w, br.h);
        
        // HP bar for multi-hp bricks
        if (br.maxHp > 1) {
            const barW = br.w * (br.hp / br.maxHp);
            ctx.fillStyle = '#fff';
            ctx.fillRect(br.x, br.y + br.h - 3, barW, 2);
        }
    }
    
    // Player bullets
    ctx.fillStyle = '#00ffcc';
    for (const b of playerBullets) {
        ctx.beginPath();
        ctx.arc(b.x, b.y, b.r, 0, Math.PI * 2);
        ctx.fill();
    }
    
    // Enemy bullets
    for (const b of enemyBullets) {
        ctx.fillStyle = '#ff3366';
        ctx.beginPath();
        ctx.arc(b.x, b.y, b.r, 0, Math.PI * 2);
        ctx.fill();
        ctx.fillStyle = '#ffaaaa';
        ctx.beginPath();
        ctx.arc(b.x, b.y, b.r * 0.5, 0, Math.PI * 2);
        ctx.fill();
    }
    
    // Paddle
    const grad = ctx.createLinearGradient(paddle.x, paddle.y, paddle.x, paddle.y + paddle.h);
    grad.addColorStop(0, '#00ffcc');
    grad.addColorStop(1, '#0066aa');
    ctx.fillStyle = grad;
    ctx.fillRect(paddle.x, paddle.y, paddle.w, paddle.h);
    ctx.strokeStyle = '#00ffff';
    ctx.lineWidth = 1;
    ctx.strokeRect(paddle.x, paddle.y, paddle.w, paddle.h);
    
    // Particles
    for (const p of particles) {
        ctx.globalAlpha = p.life * 2;
        ctx.fillStyle = p.color;
        ctx.fillRect(p.x - 2, p.y - 2, 4, 4);
    }
    ctx.globalAlpha = 1;
    
    // Popups
    for (const p of popups) {
        ctx.globalAlpha = p.life * 2;
        ctx.fillStyle = '#fff';
        ctx.font = 'bold 16px monospace';
        ctx.textAlign = 'center';
        ctx.fillText(p.text, p.x, p.y);
    }
    ctx.globalAlpha = 1;
    
    // HUD
    ctx.fillStyle = '#fff';
    ctx.font = 'bold 18px monospace';
    ctx.textAlign = 'left';
    ctx.fillText(`SCORE: ${score}`, 20, 30);
    ctx.fillText(`WAVE: ${wave}`, 20, 55);
    
    // Lives
    for (let i = 0; i < lives; i++) {
        ctx.fillStyle = '#ff4466';
        ctx.fillRect(W - 80 + i * 25, 20, 18, 12);
    }
    
    // Wave transition
    if (state === 'waveTransition') {
        ctx.fillStyle = `rgba(255,255,255,${Math.min(waveTimer * 0.3, 0.8)})`;
        ctx.font = 'bold 48px monospace';
        ctx.textAlign = 'center';
        ctx.fillText(transitionText, W/2, H/2);
    }
    
    ctx.restore();
}

function renderStartScreen() {
    ctx.fillStyle = '#fff';
    ctx.font = 'bold 42px monospace';
    ctx.textAlign = 'center';
    ctx.fillText('BREAKOUT BULLET-HELL', W/2, H/3);
    
    ctx.font = '18px monospace';
    ctx.fillStyle = '#aaccff';
    ctx.fillText('← → or A/D to move paddle', W/2, H/2 - 20);
    ctx.fillText('Auto-fires at bricks', W/2, H/2 + 10);
    ctx.fillText('Bricks shoot back in patterns!', W/2, H/2 + 40);
    ctx.fillText('Dodge the bullets. Clear all waves.', W/2, H/2 + 70);
    
    ctx.fillStyle = '#ff6644';
    ctx.font = 'bold 24px monospace';
    const pulse = Math.sin(Date.now() * 0.005) * 0.3 + 0.7;
    ctx.globalAlpha = pulse;
    ctx.fillText('PRESS ANY KEY TO START', W/2, H * 0.75);
    ctx.globalAlpha = 1;
    
    // Legend
    ctx.font = '14px monospace';
    ctx.fillStyle = '#ff4466';
    ctx.fillText('■ SHOOTER (aimed)', W/2 - 150, H * 0.85);
    ctx.fillStyle = '#ffaa22';
    ctx.fillText('■ SPREADER', W/2, H * 0.85);
    ctx.fillStyle = '#aa44ff';
    ctx.fillText('■ SPIRAL', W/2 + 150, H * 0.85);
    ctx.fillStyle = '#44ff88';
    ctx.fillText('■ NORMAL', W/2 - 150, H * 0.9);
}

function renderGameOver() {
    ctx.fillStyle = '#fff';
    ctx.font = 'bold 48px monospace';
    ctx.textAlign = 'center';
    ctx.fillText('GAME OVER', W/2, H/3);
    
    ctx.font = '24px monospace';
    ctx.fillStyle = '#aaccff';
    ctx.fillText(`FINAL SCORE: ${score}`, W/2, H/2 - 20);
    ctx.fillText(`WAVES CLEARED: ${wave - 1}`, W/2, H/2 + 20);
    
    ctx.font = 'bold 20px monospace';
    const pulse = Math.sin(Date.now() * 0.005) * 0.3 + 0.7;
    ctx.globalAlpha = pulse;
    ctx.fillStyle = '#ff6644';
    ctx.fillText('PRESS R TO RESTART', W/2, H * 0.7);
    ctx.globalAlpha = 1;
}

let lastTime = 0;
function gameLoop(time) {
    const dt = Math.min((time - lastTime) / 1000, 0.05);
    lastTime = time;
    
    if (state === 'waveTransition') {
        waveTimer -= dt;
        if (waveTimer <= 0) {
            state = 'playing';
            spawnWave();
        }
    }
    
    update(dt);
    render();
    requestAnimationFrame(gameLoop);
}
requestAnimationFrame(gameLoop);
</script>
</body>
