const canvas = document.getElementById('game');
const ctx = canvas.getContext('2d');

let W, H;
function resize() {
    W = window.innerWidth;
    H = window.innerHeight;
    canvas.width = W;
    canvas.height = H;
}
window.addEventListener('resize', resize);
resize();

// Audio
let audioCtx = null;
function initAudio() {
    if (!audioCtx) {
        audioCtx = new (window.AudioContext || window.webkitAudioContext)();
    }
}
function playSound(freq, type, duration, vol) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = type;
    osc.frequency.value = freq;
    gain.gain.setValueAtTime(vol || 0.15, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + (duration || 0.1));
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + (duration || 0.1));
}

// Game state
const GRID = 20;
let snake, food, score, highScore, gameState, direction, nextDir, particles, tickTimer, lastTime, speed;

try {
    highScore = parseInt(localStorage.getItem('snake3d_high') || '0');
} catch(e) { highScore = 0; }

function saveHigh() {
    try { localStorage.setItem('snake3d_high', String(highScore)); } catch(e) {}
}

function reset() {
    snake = [{x: Math.floor(GRID/2), y: Math.floor(GRID/2)}];
    direction = {x: 0, y: -1};
    nextDir = {x: 0, y: -1};
    score = 0;
    speed = 8;
    tickTimer = 0;
    lastTime = performance.now();
    spawnFood();
    particles = [];
}

function spawnFood() {
    let fx, fy;
    do {
        fx = Math.floor(Math.random() * GRID);
        fy = Math.floor(Math.random() * GRID);
    } while (snake.some(s => s.x === fx && s.y === fy));
    food = {x: fx, y: fy};
}

function startGame() {
    reset();
    gameState = 'playing';
}

// Isometric projection
const TILE_W = 36;
const TILE_H = 18;
const CUBE_H = 14;

function isoX(gx, gy) { return (gx - gy) * TILE_W / 2; }
function isoY(gx, gy) { return (gx + gy) * TILE_H / 2; }

function getOrigin() {
    const ox = W / 2;
    const oy = H / 2 - (GRID * TILE_H / 4);
    return {ox, oy};
}

function drawCube(gx, gy, topColor, leftColor, rightColor, scale) {
    scale = scale || 1;
    const {ox, oy} = getOrigin();
    const sx = ox + isoX(gx, gy);
    const sy = oy + isoY(gx, gy);
    const hw = TILE_W / 2 * scale;
    const hh = TILE_H / 2 * scale;
    const ch = CUBE_H * scale;

    // Top face
    ctx.beginPath();
    ctx.moveTo(sx, sy - hh);
    ctx.lineTo(sx + hw, sy);
    ctx.lineTo(sx, sy + hh);
    ctx.lineTo(sx - hw, sy);
    ctx.closePath();
    ctx.fillStyle = topColor;
    ctx.fill();

    // Left face
    ctx.beginPath();
    ctx.moveTo(sx - hw, sy);
    ctx.lineTo(sx, sy + hh);
    ctx.lineTo(sx, sy + hh + ch);
    ctx.lineTo(sx - hw, sy + ch);
    ctx.closePath();
    ctx.fillStyle = leftColor;
    ctx.fill();

    // Right face
    ctx.beginPath();
    ctx.moveTo(sx + hw, sy);
    ctx.lineTo(sx, sy + hh);
    ctx.lineTo(sx, sy + hh + ch);
    ctx.lineTo(sx + hw, sy + ch);
    ctx.closePath();
    ctx.fillStyle = rightColor;
    ctx.fill();

    // Subtle top highlight
    ctx.beginPath();
    ctx.moveTo(sx, sy - hh * 0.6);
    ctx.lineTo(sx + hw * 0.5, sy - hh * 0.2);
    ctx.lineTo(sx, sy + hh * 0.1);
    ctx.lineTo(sx - hw * 0.5, sy - hh * 0.2);
    ctx.closePath();
    ctx.fillStyle = 'rgba(255,255,255,0.1)';
    ctx.fill();
}

function drawGround() {
    const {ox, oy} = getOrigin();
    for (let gy = 0; gy < GRID; gy++) {
        for (let gx = 0; gx < GRID; gx++) {
            const sx = ox + isoX(gx, gy);
            const sy = oy + isoY(gx, gy);
            const hw = TILE_W / 2;
            const hh = TILE_H / 2;
            const shade = (gx + gy) % 2 === 0 ? '#2a2a4a' : '#252540';
            ctx.beginPath();
            ctx.moveTo(sx, sy - hh);
            ctx.lineTo(sx + hw, sy);
            ctx.lineTo(sx, sy + hh);
            ctx.lineTo(sx - hw, sy);
            ctx.closePath();
            ctx.fillStyle = shade;
            ctx.fill();
            ctx.strokeStyle = 'rgba(100,100,180,0.15)';
            ctx.lineWidth = 0.5;
            ctx.stroke();
        }
    }

    // Border walls (3D blocks around the grid)
    for (let i = -1; i <= GRID; i++) {
        // Top wall
        if (i >= 0 && i < GRID) {
            drawCube(i, -1, '#3a3a5e', '#2a2a48', '#1e1e38', 0.6);
            drawCube(-1, i, '#3a3a5e', '#2a2a48', '#1e1e38', 0.6);
        }
    }
}

function drawFood() {
    const time = performance.now() * 0.003;
    const bounce = Math.sin(time) * 3;
    const {ox, oy} = getOrigin();
    const sx = ox + isoX(food.x, food.y);
    const sy = oy + isoY(food.x, food.y) - bounce;

    // Glow
    ctx.beginPath();
    ctx.arc(sx, sy - 4, 12, 0, Math.PI * 2);
    ctx.fillStyle = 'rgba(255,50,50,0.2)';
    ctx.fill();

    // Apple-like shape
    ctx.beginPath();
    ctx.arc(sx, sy - 6, 8, 0, Math.PI * 2);
    const grad = ctx.createRadialGradient(sx - 2, sy - 8, 1, sx, sy - 6, 8);
    grad.addColorStop(0, '#ff6666');
    grad.addColorStop(1, '#cc2222');
    ctx.fillStyle = grad;
    ctx.fill();

    // Stem
    ctx.beginPath();
    ctx.moveTo(sx, sy - 14);
    ctx.lineTo(sx + 3, sy - 18);
    ctx.strokeStyle = '#4a7c3f';
    ctx.lineWidth = 2;
    ctx.stroke();

    // Leaf
    ctx.beginPath();
    ctx.ellipse(sx + 5, sy - 17, 4, 2, 0.3, 0, Math.PI * 2);
    ctx.fillStyle = '#5a9c4f';
    ctx.fill();
}

function drawSnake() {
    // Draw from tail to head for proper layering
    for (let i = snake.length - 1; i >= 0; i--) {
        const seg = snake[i];
        const t = i / Math.max(snake.length, 1);
        let topCol, leftCol, rightCol;

        if (i === 0) {
            // Head - bright green
            topCol = '#4dff4d';
            leftCol = '#2a9922';
            rightCol = '#1a7715';
        } else {
            const r = Math.floor(60 + t * 30);
            const g = Math.floor(200 - t * 80);
            topCol = `rgb(${r},${g},40)`;
            leftCol = `rgb(${Math.floor(r*0.7)},${Math.floor(g*0.6)},30)`;
            rightCol = `rgb(${Math.floor(r*0.5)},${Math.floor(g*0.45)},25)`;
        }

        const scale = 1 - t * 0.15;
        drawCube(seg.x, seg.y, topCol, leftCol, rightCol, scale);
    }

    // Eyes on head
    const head = snake[0];
    const {ox, oy} = getOrigin();
    const hsx = ox + isoX(head.x, head.y);
    const hsy = oy + isoY(head.x, head.y) - CUBE_H;

    const eyeOff1 = {x: direction.y * 3, y: -direction.x * 3};
    const eyeOff2 = {x: -direction.y * 3, y: direction.x * 3};

    ctx.beginPath();
    ctx.arc(hsx + eyeOff1.x, hsy - 2, 2.5, 0, Math.PI * 2);
    ctx.fillStyle = '#fff';
    ctx.fill();
    ctx.beginPath();
    ctx.arc(hsx + eyeOff1.x, hsy - 2, 1.2, 0, Math.PI * 2);
    ctx.fillStyle = '#111';
    ctx.fill();

    ctx.beginPath();
    ctx.arc(hsx + eyeOff2.x, hsy - 2, 2.5, 0, Math.PI * 2);
    ctx.fillStyle = '#fff';
    ctx.fill();
    ctx.beginPath();
    ctx.arc(hsx + eyeOff2.x, hsy - 2, 1.2, 0, Math.PI * 2);
    ctx.fillStyle = '#111';
    ctx.fill();
}

function drawParticles() {
    for (let p of particles) {
        ctx.beginPath();
        ctx.arc(p.x, p.y, p.r, 0, Math.PI * 2);
        ctx.fillStyle = `rgba(${p.cr},${p.cg},${p.cb},${p.life})`;
        ctx.fill();
    }
}

function spawnParticles(gx, gy) {
    const {ox, oy} = getOrigin();
    const sx = ox + isoX(gx, gy);
    const sy = oy + isoY(gx, gy);
    for (let i = 0; i < 12; i++) {
        const angle = Math.random() * Math.PI * 2;
        const spd = 1 + Math.random() * 3;
        particles.push({
            x: sx, y: sy - 5,
            vx: Math.cos(angle) * spd,
            vy: Math.sin(angle) * spd - 2,
            r: 2 + Math.random() * 3,
            life: 1,
            cr: 80 + Math.floor(Math.random()*175),
            cg: 200 + Math.floor(Math.random()*55),
            cb: 40
        });
    }
}

function updateParticles(dt) {
    for (let p of particles) {
        p.x += p.vx;
        p.y += p.vy;
        p.vy += 0.15;
        p.life -= dt * 2;
    }
    particles = particles.filter(p => p.life > 0);
}

function tick() {
    direction = nextDir;
    const head = snake[0];
    const nx = head.x + direction.x;
    const ny = head.y + direction.y;

    if (nx < 0 || nx >= GRID || ny < 0 || ny >= GRID) {
        return gameOver();
    }
    for (let i = 1; i < snake.length; i++) {
        if (snake[i].x === nx && snake[i].y === ny) {
            return gameOver();
        }
    }

    const newHead = {x: nx, y: ny};
    snake.unshift(newHead);

    if (nx === food.x && ny === food.y) {
        score += 10;
        speed = 8 + Math.floor(score / 50);
        if (score > highScore) {
            highScore = score;
            saveHigh();
        }
        spawnParticles(nx, ny);
        playSound(440, 'sine', 0.15, 0.2);
        playSound(660, 'sine', 0.1, 0.15);
        spawnFood();
    } else {
        snake.pop();
    }
}

function gameOver() {
    gameState = 'over';
    playSound(150, 'square', 0.4, 0.2);
    if (score > highScore) {
        highScore = score;
        saveHigh();
    }
}

function drawUI() {
    ctx.fillStyle = 'rgba(0,0,0,0.5)';
    ctx.fillRect(0, 0, W, 36);
    ctx.fillStyle = '#4dff4d';
    ctx.font = 'bold 18px monospace';
    ctx.textAlign = 'left';
    ctx.fillText(`SCORE: ${score}`, 20, 24);
    ctx.fillStyle = '#aaa';
    ctx.textAlign = 'right';
    ctx.fillText(`BEST: ${highScore}`, W - 20, 24);
}

function drawStartScreen() {
    ctx.fillStyle = '#1a1a2e';
    ctx.fillRect(0, 0, W, H);

    // Draw decorative ground
    for (let gy = 0; gy < GRID; gy++) {
        for (let gx = 0; gx < GRID; gx++) {
            const sx = W/2 + isoX(gx - GRID/2, gy - GRID/2);
            const sy = H/2 + isoY(gx - GRID/2, gy - GRID/2) - 20;
            const hw = TILE_W / 2;
            const hh = TILE_H / 2;
            ctx.beginPath();
            ctx.moveTo(sx, sy - hh);
            ctx.lineTo(sx + hw, sy);
            ctx.lineTo(sx, sy + hh);
            ctx.lineTo(sx - hw, sy);
            ctx.closePath();
            ctx.fillStyle = (gx+gy)%2===0 ? '#2a2a4a' : '#252540';
            ctx.fill();
        }
    }

    // Title cubes
    const time = performance.now() * 0.002;
    for (let i = 0; i < 5; i++) {
        const gx = -2 + i;
        const gy = 0;
        const t = Math.sin(time + i * 0.5) * 3;
        drawCube(gx, gy, '#4dff4d', '#2a9922', '#1a7715', 1);
    }

    ctx.textAlign = 'center';
    ctx.fillStyle = '#4dff4d';
    ctx.font = 'bold 48px monospace';
    ctx.fillText('3D SNAKE', W/2, H/2 - 80);

    ctx.fillStyle = '#aaa';
    ctx.font = '16px monospace';

    ctx.fillText('Eat apples. Don\'t hit walls or yourself!', W/2, H/2 + 45);

    const pulse = Math.abs(Math.sin(time * 2)) * 0.5 + 0.5;
    ctx.fillStyle = `rgba(77,255,77,${pulse})`;
    ctx.font = 'bold 20px monospace';
    ctx.fillText('Press SPACE to Start', W/2, H/2 + 100);

    if (highScore > 0) {
        ctx.fillStyle = '#888';
        ctx.font = '14px monospace';
        ctx.fillText(`High Score: ${highScore}`, W/2, H/2 + 140);
    }

function drawGameOver() {
    ctx.fillStyle = 'rgba(0,0,0,0.7)';
    ctx.fillRect(0, 0, W, H);
    ctx.textAlign = 'center';
    ctx.fillStyle = '#ff4444';
    ctx.font = 'bold 42px monospace';
    ctx.fillText('GAME OVER', W/2, H/2 - 30);
    ctx.fillStyle = '#fff';
    ctx.font = '22px monospace';
    ctx.fillText(`Score: ${score}`, W/2, H/2 + 15);
    if (score >= highScore && score > 0) {
        ctx.fillStyle = '#ffd700';
        ctx.font = '18px monospace';
        ctx.fillText('★ NEW HIGH SCORE! ★', W/2, H/2 + 45);
    }
    const pulse = Math.abs(Math.sin(performance.now() * 0.003)) * 0.5 + 0.5;
    ctx.fillStyle = `rgba(255,255,255,${pulse})`;
    ctx.font = '18px monospace';
    ctx.fillText('Press SPACE to Restart', W/2, H/2 + 90);
}

function draw() {
    ctx.fillStyle = '#1a1a2e';
    ctx.fillRect(0, 0, W, H);

    if (gameState === 'start') {
        drawStartScreen();
        return;
    }

    drawGround();
    drawFood();
    drawSnake();
    drawParticles();
    drawUI();

    if (gameState === 'over') {
        drawGameOver();
    }
}

let gameState = 'start';
reset();

function gameLoop(now) {
    const dt = Math.min((now - lastTime) / 1000, 0.1);
    lastTime = now;

    if (gameState === 'playing') {
        tickTimer += dt * speed;
        while (tickTimer >= 1) {
            tick();
            tickTimer -= 1;
            if (gameState !== 'playing') break;
        }
        updateParticles(dt);
    } else {
        updateParticles(dt);
    }

    draw();
    requestAnimationFrame(gameLoop);
}

requestAnimationFrame(gameLoop);

// Input
window.addEventListener('keydown', (e) => {
    if (!audioCtx) initAudio();

    if (gameState === 'start' || gameState === 'over') {
        if (e.code === 'Space' || e.code === 'Enter') {
            startGame();
            playSound(330, 'sine', 0.1, 0.15);
        }
        return;
    }

    if (gameState !== 'playing') return;

    let d = null;
    switch(e.code) {
        case 'ArrowUp': case 'KeyW': d = {x:0, y:-1}; break;
        case 'ArrowDown': case 'KeyS': d = {x:0, y:1}; break;
        case 'ArrowLeft': case 'KeyA': d = {x:-1, y:0}; break;
        case 'ArrowRight': case 'KeyD': d = {x:1, y:0}; break;
    }
    if (d) {
        e.preventDefault();
        if (d.x !== 0 || d.y !== 0) {
            // Prevent reversing
            if (!(d.x === -direction.x && d.y === -direction.y)) {
                nextDir = d;
            }
        }
    }
});

// Touch controls for mobile
let touchStartX, touchStartY;
window.addEventListener('touchstart', (e) => {
    if (!audioCtx) initAudio();
    const t = e.touches[0];
    touchStartX = t.clientX;
    touchStartY = t.clientY;
    if (gameState === 'start' || gameState === 'over') {
        startGame();
    }
});
window.addEventListener('touchend', (e) => {
    if (!touchStartX) return;
    const t = e.changedTouches[0];
    const dx = t.clientX - touchStartX;
    const dy = t.clientY - touchStartY;
    if (Math.abs(dx) > 20 || Math.abs(dy) > 20) {
        let d;
        if (Math.abs(dx) > Math.abs(dy)) {
            d = dx > 0 ? {x:1,y:0} : {x:-1,y:0};
        } else {
            d = dy > 0 ? {x:0,y:1} : {x:0,y:-1};
        }
        if (!(d.x === -direction.x && d.y === -direction.y)) {
            nextDir = d;
        }
    }
});
