```html
<!DOCTYPE html>
<html lang="en">
<head>

<style>

<canvas id="game"></canvas>
<script>
const canvas = document.getElementById('game');
const ctx = canvas.getContext('2d');

let W, H;
function resize() {
    W = window.innerWidth;
    H = window.innerHeight;
    canvas.width = W;
    canvas.height = H;
}
window.addEventListener('resize', resize);
resize();

// Audio
let audioCtx = null;
function initAudio() {
    if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
}
function playSound(freq, dur, type, vol) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = type || 'square';
    osc.frequency.value = freq;
    gain.gain.setValueAtTime(vol || 0.15, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + dur);
}
function playType() { playSound(800, 0.05, 'square', 0.1); }
function playCorrect() { playSound(600, 0.1, 'sine', 0.2); setTimeout(()=>playSound(900,0.1,'sine',0.15),50); }
function playWrong() { playSound(150, 0.3, 'sawtooth', 0.2); }
function playCombo(n) { playSound(400+n*100, 0.15, 'sine', 0.2); }

// Game state
let state = 'start'; // start, playing, gameover
let score = 0, combo = 0, maxCombo = 0, lives = 3;
let words = [];
let particles = [];
let shakeTime = 0, shakeMag = 0;
let typeBuffer = '';
let spawnTimer = 0;
let difficulty = 1;
let highScore = 0;

try { highScore = parseInt(localStorage.getItem('typeStormHigh') || '0'); } catch(e) {}

// Word list
const wordList = [
    'cat','dog','run','big','sky','fun','top','jet','map','nap',
    'box','cup','dip','end','fly','gap','hat','ink','jam','key',
    'leg','mix','nut','owl','pig','rip','saw','tin','use','van',
    'word','type','game','code','test','play','fire','star','moon','wind',
    'storm','power','light','sound','speed','force','magic','ghost','sword','shield',
    'dragon','castle','knight','battle','quest','frozen','crystal','temple','forest','ocean',
    'thunder','lightning','explosion','velocity','momentum','fragment','circuit','photon','plasma','quantum'
];

function getWordsForLevel(lvl) {
    if (lvl <= 2) return wordList.slice(0, 48);
    if (lvl <= 5) return wordList.slice(0, 96);
    return wordList;
}

function randomWord() {
    const list = getWordsForLevel(difficulty);
    return list[Math.floor(Math.random() * list.length)];
}

function spawnWord() {
    const text = randomWord();
    const fontSize = Math.max(18, 28 - difficulty * 1.5);
    const maxW = text.length * fontSize * 0.6;
    const minW = 60;
    const x = minW + Math.random() * (W - minW - maxW);
    const y = 60 + Math.random() * (H - 120);
    const speed = (0.3 + difficulty * 0.15) * (Math.random() > 0.5 ? 1 : -1);
    const timerMax = Math.max(2, 6 - difficulty * 0.4);
    words.push({
        text, x: Math.min(x, W - maxW), y, 
        vx: speed, vy: (Math.random()-0.5)*0.3,
        fontSize, timer: timerMax, timerMax,
        active: true, alpha: 1, scale: 1, spawnAnim: 0
    });
}

function spawnParticles(x, y, color, count) {
    for (let i = 0; i < (count || 12); i++) {
        const angle = Math.random() * Math.PI * 2;
        const speed = 2 + Math.random() * 4;
        particles.push({
            x, y, vx: Math.cos(angle)*speed, vy: Math.sin(angle)*speed - 1,
            life: 1, decay: 0.02 + Math.random()*0.03,
            size: 2 + Math.random()*4, color
        });
    }
}

function resetGame() {
    score = 0; combo = 0; maxCombo = 0; lives = 3;
    words = []; particles = []; typeBuffer = '';
    spawnTimer = 0; difficulty = 1; shakeTime = 0;
    state = 'playing';
}

function inputHandler() {
    if (!audioCtx) initAudio();
}

window.addEventListener('keydown', (e) => {
    inputHandler();
    if (state === 'start') { resetGame(); return; }
    if (state === 'gameover') { 
        if (e.key === 'Enter' || e.key === ' ') resetGame();
        return; 
    }
    if (state !== 'playing') return;
    
    if (e.key === 'Backspace') { typeBuffer = typeBuffer.slice(0, -1); return; }
    if (e.key === 'Tab') { e.preventDefault(); typeBuffer = ''; return; }
    if (e.key.length === 1 && /[a-z]/i.test(e.key)) {
        typeBuffer += e.key.toLowerCase();
        playType();
        checkWords();
    }
});

function checkWords() {
    for (let i = words.length - 1; i >= 0; i--) {
        const w = words[i];
        if (!w.active) continue;
        if (w.text === typeBuffer) {
            // Correct!
            w.active = false;
            combo++;
            maxCombo = Math.max(maxCombo, combo);
            const pts = 10 * Math.min(combo, 10);
            score += pts;
            playCorrect();
            if (combo > 2) playCombo(combo);
            spawnParticles(w.x + w.text.length*w.fontSize*0.3, w.y + w.fontSize/2, '#4f8', 15);
            typeBuffer = '';
            words.splice(i, 1);
            return;
        }
    }
    // Check if buffer is valid prefix of any word
    let valid = false;
    for (const w of words) {
        if (!w.active) continue;
        if (w.text.startsWith(typeBuffer)) { valid = true; break; }
    }
    if (!valid && typeBuffer.length > 0) {
        // Wrong character
        playWrong();
        combo = 0;
        typeBuffer = '';
    }
}

function update(dt) {
    if (state !== 'playing') return;
    
    difficulty = 1 + Math.floor(score / 100);
    const spawnInterval = Math.max(0.8, 3 - difficulty * 0.25);
    const maxWords = Math.min(4 + Math.floor(difficulty/2), 8);
    
    spawnTimer += dt;
    if (spawnTimer >= spawnInterval && words.length < maxWords) {
        spawnWord();
        spawnTimer = 0;
    }
    
    for (let i = words.length - 1; i >= 0; i--) {
        const w = words[i];
        if (!w.active) continue;
        w.spawnAnim = Math.min(w.spawnAnim + dt * 5, 1);
        w.scale = 0.5 + 0.5 * w.spawnAnim;
        w.x += w.vx;
        w.y += w.vy;
        
        const maxW = w.text.length * w.fontSize * 0.6;
        if (w.x < 20) { w.x = 20; w.vx = Math.abs(w.vx); }
        if (w.x > W - maxW - 20) { w.x = W - maxW - 20; w.vx = -Math.abs(w.vx); }
        if (w.y < 50) { w.y = 50; w.vy = Math.abs(w.vy); }
        if (w.y > H - 80) { w.y = H - 80; w.vy = -Math.abs(w.vy); }
        
        w.timer -= dt;
        if (w.timer <= 0) {
            // Word expired
            spawnParticles(w.x + maxW/2, w.y + w.fontSize/2, '#f44', 10);
            words.splice(i, 1);
            lives--;
            combo = 0;
            shakeTime = 0.3;
            shakeMag = 8;
            playWrong();
            if (lives <= 0) {
                state = 'gameover';
                try { 
                    if (score > highScore) { highScore = score; localStorage.setItem('typeStormHigh', highScore); }
                } catch(e) {}
            }
        }
    }
    
    for (let i = particles.length - 1; i >= 0; i--) {
        const p = particles[i];
        p.x += p.vx; p.y += p.vy;
        p.vy += 0.15;
        p.life -= p.decay;
        if (p.life <= 0) particles.splice(i, 1);
    }
    
    shakeTime -= dt;
}

function draw() {
    ctx.save();
    if (shakeTime > 0) {
        const s = shakeMag * (shakeTime / 0.3);
        ctx.translate((Math.random()-0.5)*s, (Math.random()-0.5)*s);
    }
    
    // Background
    ctx.fillStyle = '#0a0a1a';
    ctx.fillRect(-20, -20, W+40, H+40);
    
    // Grid lines
    ctx.strokeStyle = 'rgba(30, 40, 80, 0.3)';
    ctx.lineWidth = 1;
    for (let x = 0; x < W; x += 60) {
        ctx.beginPath(); ctx.moveTo(x, 0); ctx.lineTo(x, H); ctx.stroke();
    }
    for (let y = 0; y < H; y += 60) {
        ctx.beginPath(); ctx.moveTo(0, y); ctx.lineTo(W, y); ctx.stroke();
    }
    
    if (state === 'start') {
        drawStartScreen();
    } else if (state === 'playing') {
        drawGame();
    } else if (state === 'gameover') {
        drawGameOver();
    }
    
    // Particles
    for (const p of particles) {
        ctx.globalAlpha = p.life;
        ctx.fillStyle = p.color;
        ctx.fillRect(p.x - p.size/2, p.y - p.size/2, p.size, p.size);
    }
    ctx.globalAlpha = 1;
    
    ctx.restore();
}

function drawStartScreen() {
    const cx = W/2, cy = H/2;
    
    ctx.fillStyle = '#4af';
    ctx.font = `bold ${Math.min(72, W*0.08)}px Courier New`;
    ctx.textAlign = 'center';
    ctx.fillText('TYPE STORM', cx, cy - 100);
    
    ctx.fillStyle = '#8cf';
    ctx.font = `${Math.min(20, W*0.03)}px Courier New`;
    ctx.fillText('Words are falling from the sky!', cx, cy - 40);
    ctx.fillText('Type them before their timer runs out.', cx, cy - 15);
    
    ctx.fillStyle = '#aaa';
    ctx.font = `${Math.min(18, W*0.025)}px Courier New`;
    ctx.fillText('Type the word to destroy it', cx, cy + 40);
    ctx.fillText('Chain correct answers for combo multiplier!', cx, cy + 70);
    ctx.fillText('Lose a life if a word expires', cx, cy + 100);
    
    ctx.fillStyle = `rgba(255,255,255, ${0.5 + 0.5*Math.sin(Date.now()*0.003)})`;
    ctx.font = `bold ${Math.min(24, W*0.035)}px Courier New`;
    ctx.fillText('PRESS ANY KEY TO START', cx, cy + 160);
    
    if (highScore > 0) {
        ctx.fillStyle = '#fa0';
        ctx.font = `${Math.min(18, W*0.025)}px Courier New`;
        ctx.fillText(`HIGH SCORE: ${highScore}`, cx, cy + 210);
    }
}

function drawGame() {
    // Words
    for (const w of words) {
        if (!w.active) continue;
        const fs = Math.floor(w.fontSize * w.scale);
        ctx.font = `bold ${fs}px Courier New`;
        ctx.textAlign = 'left';
        
        const timerRatio = w.timer / w.timerMax;
        const r = Math.floor(100 + (1-timerRatio)*155);
        const g = Math.floor(200 * timerRatio);
        const color = `rgb(${r},${g},100)`;
        
        ctx.fillStyle = color;
        ctx.fillText(w.text, w.x, w.y + fs/3);
        
        // Timer bar
        const barW = w.text.length * fs * 0.6;
        ctx.fillStyle = 'rgba(255,255,255,0.1)';
        ctx.fillRect(w.x, w.y + fs*0.4, barW, 3);
        ctx.fillStyle = color;
        ctx.fillRect(w.x, w.y + fs*0.4, barW * timerRatio, 3);
        
        // Highlight typed portion
        if (typeBuffer && w.text.startsWith(typeBuffer)) {
            const hlW = typeBuffer.length * fs * 0.6;
            ctx.fillStyle = 'rgba(100,255,200,0.4)';
            ctx.fillRect(w.x - 2, w.y - 2, hlW, fs + 4);
        }
    }
    
    // Type buffer display
    if (typeBuffer.length > 0) {
        ctx.fillStyle = '#4f8';
        ctx.font = `bold ${Math.min(24, W*0.03)}px Courier New`;
        ctx.textAlign = 'center';
        ctx.fillText(typeBuffer, W/2, H - 30);
    }
    
    // HUD
    ctx.textAlign = 'left';
    ctx.fillStyle = '#fff';
    ctx.font = `bold ${Math.min(28, W*0.035)}px Courier New`;
    ctx.fillText(`SCORE: ${score}`, 20, 35);
    
    // Combo
    if (combo > 1) {
        const comboScale = 1 + Math.sin(Date.now()*0.01) * 0.1;
        ctx.fillStyle = '#ff0';
        ctx.font = `bold ${Math.min(24, W*0.03)*comboScale}px Courier New`;
        ctx.fillText(`x${combo} COMBO!`, 20, 70);
    }
    
    // Lives
    ctx.textAlign = 'right';
    for (let i = 0; i < 3; i++) {
        ctx.fillStyle = i < lives ? '#f44' : '#333';
        ctx.font = `${Math.min(28, W*0.035)}px Courier New`;
        ctx.fillText(i < lives ? '♥' : '♡', W - 20 - i * 35, 35);
    }
    
    // Difficulty level
    ctx.fillStyle = '#888';
    ctx.font = `${Math.min(16, W*0.02)}px Courier New`;
    ctx.fillText(`LVL ${difficulty}`, W - 20, 65);
}

function drawGameOver() {
    const cx = W/2, cy = H/2;
    
    ctx.fillStyle = '#f44';
    ctx.font = `bold ${Math.min(64, W*0.07)}px Courier New`;
    ctx.textAlign = 'center';
    ctx.fillText('GAME OVER', cx, cy - 80);
    
    ctx.fillStyle = '#fff';
    ctx.font = `${Math.min(28, W*0.035)}px Courier New`;
    ctx.fillText(`SCORE: ${score}`, cx, cy - 20);
    ctx.fillText(`MAX COMBO: x${maxCombo}`, cx, cy + 20);
    
    if (score >= highScore && score > 0) {
        ctx.fillStyle = '#ff0';
        ctx.font = `bold ${Math.min(24, W*0.03)}px Courier New`;
        ctx.fillText('★ NEW HIGH SCORE! ★', cx, cy + 65);
    } else if (highScore > 0) {
        ctx.fillStyle = '#fa0';
        ctx.font = `${Math.min(20, W*0.028)}px Courier New`;
        ctx.fillText(`HIGH SCORE: ${highScore}`, cx, cy + 65);
    }
    
    ctx.fillStyle = `rgba(255,255,255, ${0.5 + 0.5*Math.sin(Date.now()*0.003)})`;
    ctx.font = `${Math.min(22, W*0.028)}px Courier New`;
    ctx.fillText('PRESS ENTER TO RESTART', cx, cy + 130);
}

// Game loop
let lastTime = 0;
function loop(time) {
    const dt = Math.min((time - lastTime) / 1000, 0.05);
    lastTime = time;
    
    update(dt);
    draw();
    requestAnimationFrame(loop);
}
requestAnimationFrame(loop);
</script>
</body>
```
