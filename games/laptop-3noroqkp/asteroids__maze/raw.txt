const canvas = document.getElementById('game');
const ctx = canvas.getContext('2d');

let W, H, SCALE;
function resize() {
    const r = Math.min(window.innerWidth / 800, window.innerHeight / 600);
    W = 800; H = 600;

// Audio
let audioCtx = null;
function initAudio() {
    if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
}
function playSound(freq, type, duration, vol) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = type;
    osc.frequency.value = freq;
    gain.gain.setValueAtTime(vol || 0.3, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + duration);
    osc.connect(gain); gain.connect(audioCtx.destination);
    osc.start(); osc.stop(audioCtx.currentTime + duration);
}
function sfxShoot() { playSound(800, 'square', 0.1, 0.2); }
function sfxHit() { playSound(200, 'sawtooth', 0.3, 0.4); }
function sfxExplode() { playSound(100, 'sawtooth', 0.5, 0.5); playSound(60, 'square', 0.4, 0.3); }
function sfxLevelUp() { playSound(440, 'square', 0.2, 0.3); setTimeout(() => playSound(660, 'square', 0.2, 0.3), 150); setTimeout(() => playSound(880, 'square', 0.3, 0.3), 300); }
function sfxDie() { playSound(150, 'sawtooth', 0.6, 0.5); }

// Game state
const GRID = 21; // maze grid size (odd number)
const CELL = 28;
let gameState = 'start'; // start, playing, gameover, levelup
let score = 0, lives = 3, level = 1;
let highScore = 0;
try { highScore = parseInt(localStorage.getItem('mazeAsteroidsHigh') || '0'); } catch(e) {}

// Maze generation
let maze = [];
function generateMaze() {
    // Use recursive backtracker for a proper maze, then add some extra openings
    let grid = Array.from({length: GRID}, () => Array(GRID).fill(1));
    let stack = [[1,1]];
    grid[1][1] = 0;
    const dirs = [[0,-2],[0,2],[-2,0],[2,0]];
    
    while (stack.length > 0) {
        let [cx, cy] = stack[stack.length - 1];
        let shuffled = dirs.slice().sort(() => Math.random() - 0.5);
        let moved = false;
        for (let d of shuffled) {
            let nx = cx + d[0], ny = cy + d[1];
            if (nx >= 1 && nx < GRID-1 && ny >= 1 && ny < GRID-1 && grid[nx][ny] === 1) {
                grid[nx][ny] = 0;
                grid[cx + d[0]/2][cy + d[1]/2] = 0;
                stack.push([nx, ny]);
                moved = true;
                break;
            }
        }
        if (!moved) stack.pop();
    }
    
    // Add extra openings to create loops (more interesting gameplay)
    let extraOpenings = 8 + level * 2;
    for (let i = 0; i < extraOpenings; i++) {
        let rx = Math.floor(Math.random() * (GRID-2)) + 1;
        let ry = Math.floor(Math.random() * (GRID-2)) + 1;
        if (grid[rx][ry] === 1) {
            // Only open if it connects two passages
            let h = grid[rx][ry-1] === 0 && grid[rx][ry+1] === 0;
            let v = grid[rx-1][ry] === 0 && grid[rx+1][ry] === 0;
            if (h || v) grid[rx][ry] = 0;
        }
    }
    
    // Ensure center is open for player start
    let cx = Math.floor(GRID/2), cy = Math.floor(GRID/2);
    grid[cx][cy] = 0;
    
    return grid;
}

// Player
let player = { x: 0, y: 0, dir: 0, cooldown: 0 };
const DIRS = [[1,0],[0,1],[-1,0],[0,-1]]; // right, down, left, up
const ARROWS = ['→','↓','←','↑'];

// Asteroids
let asteroids = [];
function spawnAsteroid() {
    let type = Math.random();
    let size = type < 0.5 ? 2 : (type < 0.8 ? 1.5 : 1);
    let angle = Math.random() * Math.PI * 2;
    let speed = (1 + level * 0.3) * (Math.random() * 0.5 + 0.7);
    
    // Find a valid open cell to spawn in (not too close to player)
    let attempts = 0;
    while (attempts < 50) {
        attempts++;
        let gx = Math.floor(Math.random() * GRID);
        let gy = Math.floor(Math.random() * GRID);
        if (maze[gx][gy] === 0) {
            let dx = gx - player.x, dy = gy - player.y;
            if (dx*dx + dy*dy > 25) break;
        }
    }
    
    asteroids.push({
        x: gx * CELL + CELL/2,
        y: gy * CELL + CELL/2,
        vx: Math.cos(angle) * speed,
        vy: Math.sin(angle) * speed,
        size: size,
        hp: size > 1.5 ? 3 : (size > 1 ? 2 : 1),
        angle: Math.random() * Math.PI * 2,
        spin: (Math.random() - 0.5) * 0.05
    });
}

// Bullets
let bullets = [];

// Particles
let particles = [];

function initLevel() {
    maze = generateMaze();
    player.x = Math.floor(GRID/2);
    player.y = Math.floor(GRID/2);
    player.dir = 0;
    player.cooldown = 0;
    asteroids = [];
    bullets = [];
    particles = [];
    
    let numAsteroids = 3 + level * 2;
    for (let i = 0; i < numAsteroids; i++) spawnAsteroid();
}

function isWall(gx, gy) {
    if (gx < 0 || gx >= GRID || gy < 0 || gy >= GRID) return true;
    return maze[gx][gy] === 1;
}

function movePlayer(dx, dy) {
    let nx = player.x + dx, ny = player.y + dy;
    if (!isWall(nx, ny)) {
        player.x = nx;
        player.y = ny;
        // Set direction
        if (dx === 1) player.dir = 0;
        else if (dx === -1) player.dir = 2;
        else if (dy === 1) player.dir = 1;
        else if (dy === -1) player.dir = 3;
    }
}

function shoot() {
    if (player.cooldown > 0) return;
    player.cooldown = 15;
    let dx = DIRS[player.dir][0], dy = DIRS[player.dir][1];
    bullets.push({ x: player.x * CELL + CELL/2, y: player.y * CELL + CELL/2, vx: dx * 4, vy: dy * 4 });
    sfxShoot();
}

function explode(x, y, count, color) {
    for (let i = 0; i < count; i++) {
        let angle = Math.random() * Math.PI * 2;
        let speed = Math.random() * 3 + 1;
        particles.push({ x, y, vx: Math.cos(angle)*speed, vy: Math.sin(angle)*speed, life: 30+Math.random()*20, color: color || '#ff6644' });
    }
}

function update() {
    if (gameState !== 'playing') return;
    
    // Player cooldown
    if (player.cooldown > 0) player.cooldown--;
    
    // Spawn asteroids periodically
    let spawnRate = Math.max(180 - level * 15, 60);
    if (frameCount % spawnRate === 0 && asteroids.length < 8 + level * 2) {
        spawnAsteroid();
    }
    
    // Update bullets
    for (let i = bullets.length - 1; i >= 0; i--) {
        let b = bullets[i];
        b.x += b.vx;
        b.y += b.vy;
        
        // Check wall collision
        let gx = Math.floor(b.x / CELL), gy = Math.floor(b.y / CELL);
        if (isWall(gx, gy)) {
            bullets.splice(i, 1);
            continue;
        }
        if (b.x < 0 || b.x > GRID*CELL || b.y < 0 || b.y > GRID*CELL) {
            bullets.splice(i, 1);
            continue;
        }
        
        // Check asteroid collision
        for (let j = asteroids.length - 1; j >= 0; j--) {
            let a = asteroids[j];
            let dx = b.x - a.x, dy = b.y - a.y;
            let dist = Math.sqrt(dx*dx + dy*dy);
            if (dist < a.size * CELL * 0.4) {
                bullets.splice(i, 1);
                a.hp--;
                sfxHit();
                if (a.hp <= 0) {
                    score += Math.floor(50 / a.size);
                    explode(a.x, a.y, 12, '#ff6644');
                    sfxExplode();
                    asteroids.splice(j, 1);
                } else {
                    // Split into smaller
                    a.size = Math.max(1, a.size - 0.5);
                    explode(a.x, a.y, 6, '#ffaa33');
                }
                break;
            }
        }
    }
    
    // Update asteroids
    for (let i = asteroids.length - 1; i >= 0; i--) {
        let a = asteroids[i];
        a.x += a.vx;
        a.y += a.vy;
        a.angle += a.spin;
        
        // Wall bounce
        let gx = Math.floor(a.x / CELL), gy = Math.floor(a.y / CELL);
        if (isWall(gx, gy)) {
            // Bounce off wall - reverse the component that hit
            let pgx = Math.floor((a.x - a.vx) / CELL), pgy = Math.floor((a.y - a.vy) / CELL);
            if (pgx !== gx) a.vx *= -1;
            else a.vy *= -1;
            a.x -= a.vx * 2;
            a.y -= a.vy * 2;
        }
        // Boundary bounce
        if (a.x < 0 || a.x > GRID*CELL) { a.vx *= -1; a.x = Math.max(0, Math.min(GRID*CELL, a.x)); }
        if (a.y < 0 || a.y > GRID*CELL) { a.vy *= -1; a.y = Math.max(0, Math.min(GRID*CELL, a.y)); }
        
        // Player collision
        let pdx = a.x - (player.x * CELL + CELL/2);
        let pdy = a.y - (player.y * CELL + CELL/2);
        if (Math.sqrt(pdx*pdx+pdy*pdy) < (a.size * 0.35 + 0.3) * CELL) {
            lives--;
            sfxDie();
            explode(player.x * CELL + CELL/2, player.y * CELL + CELL/2, 20, '#44ff88');
            if (lives <= 0) {
                gameState = 'gameover';
                try { 
                    if (score > highScore) { highScore = score; localStorage.setItem('mazeAsteroidsHigh', highScore); }
                } catch(e) {}
            } else {
                // Invulnerability frames - just knock asteroid away
                a.vx *= -1; a.vy *= -1;
            }
        }
    }
    
    // Update particles
    for (let i = particles.length - 1; i >= 0; i--) {
        let p = particles[i];
        p.x += p.vx; p.y += p.vy;
        p.vx *= 0.95; p.vy *= 0.95;
        p.life--;
        if (p.life <= 0) particles.splice(i, 1);
    }
    
    // Level up check
    let threshold = level * 200;
    if (score >= threshold && score >= (level-1) * 200 + 200) {
        if (level === 1 && score >= 200) { level++; sfxLevelUp(); gameState = 'levelup'; setTimeout(() => { initLevel(); gameState = 'playing'; }, 1500); }
        else if (score >= level * 200) { level++; sfxLevelUp(); gameState = 'levelup'; setTimeout(() => { initLevel(); gameState = 'playing'; }, 1500); }
    }
}

// Rendering
function drawMaze() {
    for (let x = 0; x < GRID; x++) {
        for (let y = 0; y < GRID; y++) {
            if (maze[x][y] === 1) {
                ctx.fillStyle = '#2a2a4a';
                ctx.fillRect(x * CELL, y * CELL, CELL, CELL);
                ctx.strokeStyle = '#3a3a6a';
                ctx.strokeRect(x*CELL+1, y*CELL+1, CELL-2, CELL-2);
            } else {
                // Floor glow
                ctx.fillStyle = '#0d0d20';
                ctx.fillRect(x * CELL, y * CELL, CELL, CELL);
            }
        }
    }
}

function drawPlayer() {
    let px = player.x * CELL + CELL/2;
    let py = player.y * CELL + CELL/2;
    ctx.save();
    ctx.translate(px, py);
    ctx.rotate(player.dir * Math.PI / 2);
    
    // Ship shape
    ctx.fillStyle = '#44ff88';
    ctx.beginPath();
    ctx.moveTo(10, 0);
    ctx.lineTo(-7, -6);
    ctx.lineTo(-4, 0);
    ctx.lineTo(-7, 6);
    ctx.closePath();
    ctx.fill();
    
    // Engine glow
    ctx.fillStyle = '#aaffcc';
    ctx.beginPath();
    ctx.arc(-5, 0, 3, 0, Math.PI*2);
    ctx.fill();
    
    ctx.restore();
}

function drawAsteroid(a) {
    let r = a.size * CELL * 0.4;
    ctx.save();
    ctx.translate(a.x, a.y);
    ctx.rotate(a.angle);
    
    // Rocky shape
    ctx.fillStyle = '#8b7355';
    ctx.strokeStyle = '#a0926b';
    ctx.lineWidth = 1.5;
    ctx.beginPath();
    for (let i = 0; i < 8; i++) {
        let angle = (i / 8) * Math.PI * 2;
        let rr = r * (0.7 + Math.sin(i*3.7 + a.size)*0.3);
        if (i === 0) ctx.moveTo(Math.cos(angle)*rr, Math.sin(angle)*rr);
        else ctx.lineTo(Math.cos(angle)*rr, Math.sin(angle)*rr);
    }
    ctx.closePath();
    ctx.fill();
    ctx.stroke();
    
    // Craters
    ctx.fillStyle = '#6b5335';
    ctx.beginPath();
    ctx.arc(-r*0.2, -r*0.1, r*0.2, 0, Math.PI*2);
    ctx.fill();
    ctx.beginPath();
    ctx.arc(r*0.25, r*0.2, r*0.15, 0, Math.PI*2);
    ctx.fill();
    
    ctx.restore();
}

function drawBullets() {
    ctx.fillStyle = '#ffee44';
    for (let b of bullets) {
        ctx.beginPath();
        ctx.arc(b.x, b.y, 3, 0, Math.PI*2);
        ctx.fill();
        // Trail
        ctx.fillStyle = 'rgba(255,238,68,0.4)';
        ctx.beginPath();
        ctx.arc(b.x - b.vx, b.y - b.vy, 2, 0, Math.PI*2);
        ctx.fill();
        ctx.fillStyle = '#ffee44';
    }
}

function drawParticles() {
    for (let p of particles) {
        let alpha = p.life / 50;
        ctx.fillStyle = `rgba(${parseInt(p.color.slice(1,3),16)},${parseInt(p.color.slice(3,5),16)},${parseInt(p.color.slice(5,7),16)},${alpha})`;
        ctx.fillRect(p.x - 2, p.y - 2, 4, 4);
    }
}

function drawHUD() {
    ctx.fillStyle = 'rgba(0,0,0,0.7)';
    ctx.fillRect(0, 0, W, 36);
    
    ctx.font = '14px "Courier New"';
    ctx.fillStyle = '#fff';

    ctx.fillText(`SCORE: ${score}`, 15, 24);
    ctx.textAlign = 'center';
    ctx.fillText(`LEVEL: ${level}`, W/2, 24);
    ctx.textAlign = 'right';
    ctx.fillText(`LIVES: ${'❤'.repeat(Math.max(0,lives))}  HI: ${Math.max(highScore, score)}`, W-15, 24);
}

    ctx.fillStyle = '#0a0a1a';
    ctx.fillRect(0, 0, W, H);
    
    // Title
    ctx.font = 'bold 48px "Courier New"';
    ctx.textAlign = 'center';

            if(j===0) ctx.moveTo(Math.cos(a2)*r2, Math.sin(a2)*r2);
            else ctx.lineTo(Math.cos(a2)*r2, Math.sin(a2)*r2);
        }
        ctx.closePath(); ctx.fill();
        ctx.restore();
    }
    
    let pulse = Math.sin(t*3) * 0.5 + 0.5;
    ctx.font = '20px "Courier New"';
    ctx.fillStyle = `rgba(68,255,136,${pulse})`;
    ctx.fillText('PRESS ENTER TO START', W/2, 540);

function drawGameOver() {
    ctx.fillStyle = 'rgba(0,0,0,0.8)';
    ctx.fillRect(0, 0, W, H);
    ctx.font = 'bold 42px "Courier New"';
    ctx.textAlign = 'center';
    ctx.fillStyle = '#ff4444';

    ctx.fillStyle = `rgba(68,255,136,${pulse})`;
    ctx.fillText('PRESS ENTER TO RESTART', W/2, H/2 + 110);
}

function drawLevelUp() {
    ctx.fillStyle = 'rgba(0,0,0,0.6)';
    ctx.fillRect(0, 0, W, H);
    ctx.font = 'bold 36px "Courier New"';
    ctx.textAlign = 'center';
    ctx.fillStyle = '#44ff88';
    ctx.fillText(`LEVEL ${level}`, W/2, H/2);
    ctx.font = '18px "Courier New"';

// Input
let keys = {};
window.addEventListener('keydown', (e) => {
    keys[e.key] = true;
    initAudio();
    
    if (gameState === 'start' && (e.key === 'Enter' || e.key === ' ')) {
        gameState = 'playing';
        score = 0; lives = 3; level = 1;
        initLevel();
    } else if (gameState === 'gameover' && (e.key === 'Enter' || e.key === ' ')) {
        gameState = 'playing';
        score = 0; lives = 3; level = 1;
        initLevel();
    }
    
    if (gameState !== 'playing') return;
    
    if (e.key === 'ArrowRight' || e.key === 'd') movePlayer(1, 0);
    else if (e.key === 'ArrowLeft' || e.key === 'a') movePlayer(-1, 0);
    else if (e.key === 'ArrowUp' || e.key === 'w') movePlayer(0, -1);
    else if (e.key === 'ArrowDown' || e.key === 's') movePlayer(0, 1);
    
    if (e.key === ' ') shoot();
});
window.addEventListener('keyup', (e) => { keys[e.key] = false; });

// Prevent scrolling
window.addEventListener('keydown', (e) => {
    if (['ArrowUp','ArrowDown','ArrowLeft','ArrowRight',' '].includes(e.key)) e.preventDefault();
});

let frameCount = 0;
function render() {
    ctx.clearRect(0, 0, W, H);
    
    if (gameState === 'start') {
        drawStartScreen();
    } else if (gameState === 'playing' || gameState === 'levelup' || gameState === 'gameover') {
        // Background
        ctx.fillStyle = '#0a0a1a';
        ctx.fillRect(0, 0, W, H);
        
        drawMaze();
        drawBullets();
        for (let a of asteroids) drawAsteroid(a);
        if (gameState !== 'gameover') drawPlayer();
        drawParticles();
        drawHUD();
        
        if (gameState === 'gameover') drawGameOver();
        else if (gameState === 'levelup') drawLevelUp();
    }
    
    frameCount++;
    update();
    requestAnimationFrame(render);
}

initLevel();
render();
