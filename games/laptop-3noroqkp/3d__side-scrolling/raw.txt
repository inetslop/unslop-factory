const canvas = document.getElementById('game');
const ctx = canvas.getContext('2d');

let W, H;
function resize() {
    W = canvas.width = window.innerWidth;
    H = canvas.height = window.innerHeight;
}
window.addEventListener('resize', resize);
resize();

// Audio
let audioCtx = null;
function initAudio() {
    if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
}
function playSound(freq, type, duration, vol) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = type;
    osc.frequency.value = freq;
    gain.gain.setValueAtTime(vol || 0.15, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + (duration || 0.2));
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + (duration || 0.2));
}
function playJump() { playSound(300, 'square', 0.15, 0.1); }
function playCoin() { playSound(800, 'sine', 0.1, 0.12); setTimeout(() => playSound(1200, 'sine', 0.15, 0.1), 50); }
function playHit() { playSound(100, 'sawtooth', 0.3, 0.2); }
function playScore() { playSound(600, 'triangle', 0.08, 0.08); }

// Game state
let state = 'start'; // start, playing, gameover
let score = 0;
let best = 0;
try { best = parseInt(localStorage.getItem('neonRushBest')) || 0; } catch(e) {}
let speed = 4;
let distance = 0;
let frameCount = 0;

// Player
const LANES = [-1, 0, 1];
let player = { lane: 1, x: 0, y: 0, vy: 0, jumping: false, groundY: 0 };
let targetLaneX = 0;

// World objects
let obstacles = [];
let coins = [];
let particles = [];
let roadLines = [];

function resetGame() {
    score = 0;
    speed = 4;
    distance = 0;
    frameCount = 0;
    player.lane = 1;
    player.x = 0;
    player.y = 0;
    player.vy = 0;
    player.jumping = false;
    targetLaneX = 0;
    obstacles = [];
    coins = [];
    particles = [];
    roadLines = [];
    for (let i = 0; i < 30; i++) {
        roadLines.push({ z: i * 2 });
    }
}

// Perspective projection
const FOCAL = 200;
const HORIZON_Y = 0.35;

function project(x, y, z) {
    const scale = FOCAL / (z + FOCAL);
    const sx = W / 2 + x * scale * W * 0.3;
    const sy = H * HORIZON_Y + (y - H * 0.5) * scale;
    return { x: sx, y: sy, scale: scale };
}

function getLaneX(lane) {
    return (lane - 1) * W * 0.22;
}

function spawnObstacle() {
    const type = Math.random() < 0.6 ? 'barrier' : 'wall';
    let lane = Math.floor(Math.random() * 3);
    if (type === 'wall') {
        // Wall blocks 1-2 lanes
        const extraLane = Math.random() < 0.5 ? (lane + 1) % 3 : -1;
        obstacles.push({ type: 'wall', lane, extraLane, z: 80 });
    } else {
        obstacles.push({ type: 'barrier', lane, z: 80 });
    }
}

function spawnCoin() {
    const lane = Math.floor(Math.random() * 3);
    const count = 3 + Math.floor(Math.random() * 4);
    for (let i = 0; i < count; i++) {
        coins.push({ lane, z: 80 + i * 1.5, collected: false });
    }
}

function spawnParticles(x, y, color, count) {
    for (let i = 0; i < (count || 5); i++) {
        particles.push({
            x, y,
            vx: (Math.random() - 0.5) * 6,
            vy: (Math.random() - 0.5) * 6 - 2,
            life: 1,
            color: color || '#fff'
        });
    }
}

// Input
let keys = {};
window.addEventListener('keydown', (e) => {
    keys[e.code] = true;
    if (state === 'start') {
        initAudio();
        state = 'playing';
        resetGame();
        return;
    }
    if (state === 'gameover' && e.code === 'KeyR') {
        state = 'playing';
        resetGame();
        return;
    }
    if (state !== 'playing') return;

    if (e.code === 'ArrowLeft' || e.code === 'KeyA') {
        player.lane = Math.max(0, player.lane - 1);
        playJump();
    }
    if (e.code === 'ArrowRight' || e.code === 'KeyD') {
        player.lane = Math.min(2, player.lane + 1);
        playJump();
    }
    if ((e.code === 'Space' || e.code === 'ArrowUp' || e.code === 'KeyW') && !player.jumping) {
        player.vy = -8;
        player.jumping = true;
        playJump();
    }
});
window.addEventListener('keyup', (e) => { keys[e.code] = false; });

// Touch support
let touchStartX = 0, touchStartY = 0;
canvas.addEventListener('touchstart', (e) => {
    e.preventDefault();
    initAudio();
    if (state === 'start') { state = 'playing'; resetGame(); return; }
    if (state === 'gameover') { state = 'playing'; resetGame(); return; }
    const t = e.touches[0];
    touchStartX = t.clientX;
    touchStartY = t.clientY;
});
canvas.addEventListener('touchmove', (e) => {
    e.preventDefault();

    e.preventDefault();
    const t = e.changedTouches[0];
    const dx = t.clientX - touchStartX;
    const dy = t.clientY - touchStartY;
    if (state !== 'playing') return;
    if (Math.abs(dx) > Math.abs(dy)) {
        if (dx > 30) player.lane = Math.min(2, player.lane + 1);
        else if (dx < -30) player.lane = Math.max(0, player.lane - 1);
    } else if (dy < -30) {
        if (!player.jumping) { player.vy = -8; player.jumping = true; playJump(); }
    }
});

// Update
function update() {
    if (state !== 'playing') return;
    frameCount++;
    distance += speed * 0.1;
    score = Math.floor(distance) + coinsCollected;
    speed = 4 + Math.min(distance * 0.02, 8);

    // Player physics
    player.vy += 0.6;
    player.y += player.vy;
    if (player.y > 0) { player.y = 0; player.jumping = false; player.vy = 0; }

    targetLaneX = getLaneX(player.lane);

    // Spawn
    if (frameCount % Math.max(30, Math.floor(80 - speed * 3)) === 0) spawnObstacle();
    if (frameCount % 120 === 0) spawnCoin();

    // Move objects
    for (let i = obstacles.length - 1; i >= 0; i--) {
        obstacles[i].z -= speed * 0.5;
        if (obstacles[i].z < 1) {
            // Check collision
            const ob = obstacles[i];
            if (!player.jumping && player.y > -20) {
                if (ob.type === 'barrier' && ob.lane === player.lane) {
                    gameOver(); return;
                }
                if (ob.type === 'wall') {
                    const lanes = [ob.lane];
                    if (ob.extraLane >= 0) lanes.push(ob.extraLane);
                    if (lanes.includes(player.lane)) { gameOver(); return; }
                }
            }
            obstacles.splice(i, 1);
        }
    }

    for (let i = coins.length - 1; i >= 0; i--) {
        coins[i].z -= speed * 0.5;
        if (!coins[i].collected && Math.abs(coins[i].z - 2) < 3 && coins[i].lane === player.lane && player.y > -40) {
            coins[i].collected = true;
            score += 10;
            playCoin();
            const p = project(getLaneX(player.lane), -20, 5);
            spawnParticles(p.x, p.y, '#ffd700', 8);
        }
        if (coins[i].z < 1) coins.splice(i, 1);
    }

    // Road lines
    for (let i = roadLines.length - 1; i >= 0; i--) {
        roadLines[i].z -= speed * 0.5;
        if (roadLines[i].z < 0) roadLines[i].z += 60;
    }

    // Particles
    for (let i = particles.length - 1; i >= 0; i--) {
        particles[i].x += particles[i].vx;
        particles[i].y += particles[i].vy;
        particles[i].life -= 0.03;
        if (particles[i].life <= 0) particles.splice(i, 1);
    }
}

let coinsCollected = 0;
function gameOver() {
    state = 'gameover';
    playHit();
    try {
        if (score > best) { best = score; localStorage.setItem('neonRushBest', best); }
    } catch(e) {}
    const p = project(getLaneX(player.lane), 0, 5);
    spawnParticles(p.x, p.y, '#ff3333', 20);
}

// Render
function render() {
    ctx.fillStyle = '#0a0a1a';
    ctx.fillRect(0, 0, W, H);

    // Sky gradient
    const skyGrad = ctx.createLinearGradient(0, 0, 0, H * HORIZON_Y);
    skyGrad.addColorStop(0, '#050510');
    skyGrad.addColorStop(1, '#1a1a3a');
    ctx.fillStyle = skyGrad;
    ctx.fillRect(0, 0, W, H * HORIZON_Y);

    // Stars
    for (let i = 0; i < 50; i++) {
        const sx = ((i * 137 + frameCount * 0.02) % W);
        const sy = (i * 97) % (H * HORIZON_Y * 0.8);
        ctx.fillStyle = `rgba(255,255,255,${0.3 + (i % 3) * 0.2})`;
        ctx.fillRect(sx, sy, 1.5, 1.5);
    }

    // Ground
    const groundGrad = ctx.createLinearGradient(0, H * HORIZON_Y, 0, H);
    groundGrad.addColorStop(0, '#1a1a2e');
    groundGrad.addColorStop(1, '#0f0f1f');
    ctx.fillStyle = groundGrad;
    ctx.fillRect(0, H * HORIZON_Y, W, H - H * HORIZON_Y);

    // Road
    drawRoad();

    // Collect and sort renderables by z (far to near)
    let renderables = [];
    for (const ob of obstacles) {
        renderables.push({ type: 'obstacle', data: ob, z: ob.z });
    }
    for (const c of coins) {
        if (!c.collected) renderables.push({ type: 'coin', data: c, z: c.z });
    }
    renderables.sort((a, b) => b.z - a.z);

    // Draw world objects
    for (const r of renderables) {
        if (r.type === 'obstacle') drawObstacle(r.data);
        else drawCoin(r.data);
    }

    // Draw player
    drawPlayer();

    // Particles
    for (const p of particles) {
        ctx.globalAlpha = p.life;
        ctx.fillStyle = p.color;
        ctx.fillRect(p.x - 3, p.y - 3, 6, 6);
        ctx.globalAlpha = 1;
    }

    // HUD
    if (state === 'playing') {
        drawHUD();
    }

    // Overlays
    if (state === 'start') drawStartScreen();
    if (state === 'gameover') drawGameOver();
}

function drawRoad() {
    // Road edges and lanes with perspective
    for (let i = 0; i < roadLines.length; i++) {
        const z = roadLines[i].z;
        const p = project(0, 0, z);
        const halfW = W * 0.28 * p.scale;

        // Road surface lines
        ctx.strokeStyle = `rgba(0, 255, 200, ${Math.min(p.scale * 0.5, 0.3)})`;
        ctx.lineWidth = 1;
        ctx.beginPath();
        ctx.moveTo(W / 2 - halfW, p.y);
        ctx.lineTo(W / 2 + halfW, p.y);
        ctx.stroke();

        // Lane dividers
        for (let l = 0; l < 2; l++) {
            const lx = (l === 0 ? -1 : 1) * W * 0.11 * p.scale;
            ctx.strokeStyle = `rgba(0, 255, 200, ${Math.min(p.scale * 0.4, 0.25)})`;
            ctx.beginPath();
            ctx.moveTo(W / 2 + lx - W * 0.03 * p.scale, p.y);
            ctx.lineTo(W / 2 + lx + W * 0.03 * p.scale, p.y);
            ctx.stroke();
        }
    }

    // Road edges (continuous)
    for (let z = 1; z < 80; z += 2) {
        const p = project(0, 0, z);
        const halfW = W * 0.28 * p.scale;
        ctx.strokeStyle = `rgba(0, 255, 200, ${Math.min((1 - z / 80) * 0.6, 0.4)})`;
        ctx.lineWidth = 2;
        ctx.beginPath();
        ctx.moveTo(W / 2 - halfW, p.y);
        ctx.lineTo(W / 2 - halfW, project(0, 0, z + 2).y);
        ctx.stroke();
        ctx.beginPath();
        ctx.moveTo(W / 2 + halfW, p.y);
        ctx.lineTo(W / 2 + halfW, project(0, 0, z + 2).y);
        ctx.stroke();
    }
}

function drawObstacle(ob) {
    const p = project(getLaneX(ob.lane), 0, ob.z);
    const s = p.scale;
    const w = W * 0.18 * s;
    const h = H * 0.12 * s;

    if (ob.type === 'barrier') {
        // Jump-over barrier
        ctx.fillStyle = `rgba(255, 50, 50, ${Math.min(s * 1.5, 1)})`;
        ctx.strokeStyle = '#ff3333';
        ctx.lineWidth = 2;
        const bx = p.x - w / 2;
        const by = p.y - h;
        ctx.fillRect(bx, by, w, h * 0.4);
        ctx.strokeRect(bx, by, w, h * 0.4);

        // Glow
        ctx.shadowColor = '#ff3333';
        ctx.shadowBlur = 10 * s;
        ctx.strokeRect(bx, by, w, h * 0.4);
        ctx.shadowBlur = 0;
    } else {
        // Wall - taller, blocks lane(s)
        const lanes = [ob.lane];
        if (ob.extraLane >= 0) lanes.push(ob.extraLane);
        for (const l of lanes) {
            const lp = project(getLaneX(l), 0, ob.z);
            const lw = W * 0.16 * s;
            const lh = H * 0.25 * s;
            ctx.fillStyle = `rgba(50, 100, 255, ${Math.min(s * 1.5, 0.9)})`;
            ctx.strokeStyle = '#4488ff';
            ctx.lineWidth = 2;
            const wx = lp.x - lw / 2;
            const wy = lp.y - lh;
            ctx.fillRect(wx, wy, lw, lh);
            ctx.strokeRect(wx, wy, lw, lh);

            // Inner detail
            ctx.strokeStyle = 'rgba(100, 150, 255, 0.5)';
            ctx.strokeRect(wx + lw * 0.15, wy + lh * 0.15, lw * 0.7, lh * 0.7);
        }
    }
}

function drawCoin(c) {
    const p = project(getLaneX(c.lane), -30, c.z);
    const s = p.scale;
    const r = 12 * Math.max(s, 0.1);
    ctx.fillStyle = '#ffd700';
    ctx.strokeStyle = '#ffaa00';
    ctx.lineWidth = 2;
    ctx.beginPath();
    ctx.ellipse(p.x, p.y, r, r * 0.6, 0, 0, Math.PI * 2);
    ctx.fill();
    ctx.stroke();

    // Shine
    ctx.fillStyle = 'rgba(255,255,200,0.5)';
    ctx.beginPath();
    ctx.ellipse(p.x - r * 0.3, p.y - r * 0.2, r * 0.3, r * 0.2, 0, 0, Math.PI * 2);
    ctx.fill();
}

function drawPlayer() {
    const px = W / 2 + targetLaneX * 0.8;
    const py = H * 0.72 + player.y * 3;
    const s = 1;

    // Shadow
    ctx.fillStyle = 'rgba(0,255,200,0.2)';
    ctx.beginPath();
    ctx.ellipse(px, H * 0.72 + 15, 25, 8, 0, 0, Math.PI * 2);
    ctx.fill();

    // Body (neon character)
    ctx.fillStyle = '#00ffcc';
    ctx.strokeStyle = '#00ffcc';
    ctx.lineWidth = 3;
    ctx.shadowColor = '#00ffcc';
    ctx.shadowBlur = 15;

    // Head
    ctx.beginPath();
    ctx.arc(px, py - 40, 12, 0, Math.PI * 2);
    ctx.fill();

    // Body
    ctx.beginPath();
    ctx.moveTo(px, py - 28);
    ctx.lineTo(px, py - 5);
    ctx.stroke();

    // Arms
    const armSwing = Math.sin(frameCount * 0.3) * (player.jumping ? 0 : 1);
    ctx.beginPath();
    ctx.moveTo(px, py - 22);
    ctx.lineTo(px - 15 * armSwing, py - 12 + armSwing * 5);
    ctx.stroke();
    ctx.beginPath();
    ctx.moveTo(px, py - 22);
    ctx.lineTo(px + 15 * armSwing, py - 12 - armSwing * 5);
    ctx.stroke();

    // Legs
    const legSwing = Math.sin(frameCount * 0.3) * (player.jumping ? 0 : 1);
    ctx.beginPath();
    ctx.moveTo(px, py - 5);
    ctx.lineTo(px - 10 * legSwing, py + 12);
    ctx.stroke();
    ctx.beginPath();
    ctx.moveTo(px, py - 5);
    ctx.lineTo(px + 10 * legSwing, py + 12);
    ctx.stroke();

    ctx.shadowBlur = 0;
}

function drawHUD() {
    ctx.fillStyle = 'rgba(0, 255, 200, 0.9)';
    ctx.font = `bold ${Math.min(W * 0.04, 32)}px monospace`;
    ctx.textAlign = 'left';
    ctx.fillText(`SCORE: ${score}`, 20, 40);
    ctx.font = `${Math.min(W * 0.025, 20)}px monospace`;
    ctx.fillStyle = 'rgba(0, 255, 200, 0.6)';
    ctx.fillText(`BEST: ${best}`, 20, 70);
    ctx.textAlign = 'right';
    ctx.fillStyle = 'rgba(255, 200, 0, 0.8)';
    ctx.fillText(`${Math.floor(speed * 10)} km/h`, W - 20, 40);
}

function drawStartScreen() {
    ctx.fillStyle = 'rgba(5, 5, 20, 0.85)';
    ctx.fillRect(0, 0, W, H);

    const cx = W / 2;
    const cy = H / 2;

    // Title
    ctx.textAlign = 'center';
    ctx.font = `bold ${Math.min(W * 0.08, 72)}px monospace`;
    ctx.fillStyle = '#00ffcc';
    ctx.shadowColor = '#00ffcc';
    ctx.shadowBlur = 20;
    ctx.fillText('NEON RUSH', cx, cy - 80);
    ctx.shadowBlur = 0;

    // Subtitle
    ctx.font = `${Math.min(W * 0.03, 24)}px monospace`;
    ctx.fillStyle = '#ff66aa';
    ctx.fillText('3D ENDLESS RUNNER', cx, cy - 40);

    // Controls
    ctx.font = `${Math.min(W * 0.022, 18)}px monospace`;
    ctx.fillStyle = 'rgba(255,255,255,0.7)';
    const lines = [
        '← → or A/D : Switch Lane',
        'SPACE / ↑ / W : Jump',
        '',
        'Dodge barriers & walls',
        'Collect coins for bonus points!',
        '',
        'Press any key to start'
    ];
    lines.forEach((line, i) => {
        ctx.fillText(line, cx, cy + 20 + i * 28);
    });

    // Best score
    if (best > 0) {
        ctx.fillStyle = '#ffd700';
        ctx.fillText(`Best: ${best}`, cx, cy + 20 + lines.length * 28 + 10);
    }

    // Animated border
    const t = Date.now() * 0.003;
    ctx.strokeStyle = `rgba(0, 255, 200, ${0.3 + Math.sin(t) * 0.2})`;
    ctx.lineWidth = 2;
    ctx.strokeRect(cx - 220, cy - 120, 440, 380);
}

function drawGameOver() {
    ctx.fillStyle = 'rgba(5, 5, 20, 0.8)';
    ctx.fillRect(0, 0, W, H);

    const cx = W / 2;
    const cy = H / 2;

    ctx.textAlign = 'center';
    ctx.font = `bold ${Math.min(W * 0.07, 64)}px monospace`;
    ctx.fillStyle = '#ff3333';
    ctx.shadowColor = '#ff3333';
    ctx.shadowBlur = 15;
    ctx.fillText('GAME OVER', cx, cy - 50);
    ctx.shadowBlur = 0;

    ctx.font = `${Math.min(W * 0.04, 36)}px monospace`;
    ctx.fillStyle = '#fff';
    ctx.fillText(`Score: ${score}`, cx, cy + 10);

    ctx.font = `${Math.min(W * 0.028, 24)}px monospace`;
    ctx.fillStyle = '#ffd700';
    ctx.fillText(`Best: ${best}`, cx, cy + 50);

    ctx.fillStyle = 'rgba(255,255,255,0.6)';
    ctx.font = `${Math.min(W * 0.024, 20)}px monospace`;
    ctx.fillText('Press R to restart', cx, cy + 100);
}

// Main loop
function loop() {
    update();
    render();
    requestAnimationFrame(loop);
}
resetGame();
loop();
</script>
