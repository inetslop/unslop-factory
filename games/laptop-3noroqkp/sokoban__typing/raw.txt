<script>
(function(){
  const LEVELS = [
    // Level 1: Push right x2
    ["#######",
     "#     #",
     "# @$* #",
     "#     #",
     "#######"],
    // Level 2: Push up x1
    ["#######",
     "# *   #",
     "# $   #",
     "# @   #",
     "#######"],
    // Level 3: Push right (two separate boxes)
    ["########",
     "#     #",
     "# @$* #",
     "#   $ #",
     "# *   #",
     "#     #",
     "########"],
    // Level 4: Two boxes push up
    ["########",
     "# * * #",
     "# $ $ #",
     "# @   #",
     "#     #",
     "########"],
    // Level 5: Mixed pushes
    ["#########",
     "#       #",
     "#  $  * #",
     "#  @    #",
     "#  $  * #",
     "#       #",
     "#########"]
  ];

  let currentLevel = 0;
  let totalMoves = 0;
  let grid = [];
  let playerPos = {r:0,c:0};
  let boxes = [];
  let targets = [];
  let levelSolved = false;
  let audioCtx = null;

  const gridEl = document.getElementById('grid-container');
  const inputEl = document.getElementById('cmd-input');
  const hintEl = document.getElementById('hint');
  const levelDisp = document.getElementById('level-display');
  const movesDisp = document.getElementById('moves-display');

  function initAudio() {
    if (!audioCtx) {
      audioCtx = new (window.AudioContext || window.webkitAudioContext)();
    }
  }

  function playTone(freq, dur, type='sine', vol=0.15) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = type;
    osc.frequency.value = freq;
    gain.gain.setValueAtTime(vol, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + dur);
  }

  function sfxMove() { playTone(300, 0.08, 'square', 0.1); }
  function sfxPush() { playTone(200, 0.12, 'triangle', 0.15); }
  function sfxWall() { playTone(100, 0.15, 'sawtooth', 0.1); }
  function sfxLevelUp() {
    playTone(440, 0.15, 'sine', 0.2);
    setTimeout(()=>playTone(550, 0.15, 'sine', 0.2), 150);
    setTimeout(()=>playTone(660, 0.2, 'sine', 0.2), 300);
  }
  function sfxWin() {
    [440,550,660,880].forEach((f,i)=>setTimeout(()=>playTone(f,0.3,'sine',0.2), i*200));
  }

  function loadLevel(idx) {
    const raw = LEVELS[idx];
    grid = [];
    boxes = [];
    targets = [];
    for (let r = 0; r < raw.length; r++) {
      grid[r] = [];
      for (let c = 0; c < raw[r].length; c++) {
        const ch = raw[r][c];
        if (ch === '@' || ch === '+') { playerPos = {r, c}; grid[r][c] = '.'; }
        else if (ch === '$' || ch === '%') { boxes.push({r, c}); grid[r][c] = '.'; }
        else if (ch === '*') { targets.push({r, c}); grid[r][c] = '.'; }
        else if (ch === '#') { grid[r][c] = '#'; }
        else { grid[r][c] = '.'; }
      }
    }
    levelSolved = false;
    renderGrid();
  }

  function isBoxAt(r, c) {
    return boxes.some(b => b.r === r && b.c === c);
  }

  function isTargetAt(r, c) {
    return targets.some(t => t.r === r && t.c === c);
  }

  function renderGrid() {
    const rows = grid.length;
    const cols = grid[0].length;
    gridEl.style.gridTemplateColumns = `repeat(${cols}, 1fr)`;
    gridEl.innerHTML = '';
    for (let r = 0; r < rows; r++) {
      for (let c = 0; c < cols; c++) {
        const cell = document.createElement('div');
        cell.className = 'cell';
        if (grid[r][c] === '#') {
          cell.classList.add('wall');
        } else {
          cell.classList.add('floor');
          if (isTargetAt(r,c) && isBoxAt(r,c)) {
            cell.classList.remove('floor');
            cell.classList.add('box-target');
            cell.textContent = '★';
          } else if (isTargetAt(r,c)) {
            cell.classList.remove('floor');
            cell.classList.add('target');
            cell.textContent = '★';
          } else if (isBoxAt(r,c)) {
            cell.classList.remove('floor');
            cell.classList.add('box');
            cell.textContent = '■';
          }
        }
        if (playerPos.r === r && playerPos.c === c) {
          cell.className = 'cell player';
          cell.textContent = '◆';
        }
        gridEl.appendChild(cell);
      }
    }
  }

  function tryMove(dr, dc) {
    if (levelSolved) return;
    const nr = playerPos.r + dr;
    const nc = playerPos.c + dc;
    if (nr < 0 || nr >= grid.length || nc < 0 || nc >= grid[0].length) return;
    if (grid[nr][nc] === '#') { sfxWall(); showHint('BUMP!'); return; }

    if (isBoxAt(nr, nc)) {
      const br = nr + dr;
      const bc = nc + dc;
      if (br < 0 || br >= grid.length || bc < 0 || bc >= grid[0].length) { sfxWall(); showHint('BLOCKED!'); return; }
      if (grid[br][bc] === '#') { sfxWall(); showHint('BLOCKED!'); return; }
      if (isBoxAt(br, bc)) { sfxWall(); showHint('BOXES CAN\'T STACK!'); return; }
      // Push box
      const box = boxes.find(b => b.r === nr && b.c === nc);
      box.r = br;
      box.c = bc;
      playerPos = {r: nr, c: nc};
      sfxPush();
    } else {
      playerPos = {r: nr, c: nc};
      sfxMove();
    }

    totalMoves++;
    movesDisp.textContent = 'MOVES: ' + totalMoves;
    renderGrid();
    checkWin();
  }

  function checkWin() {
    if (boxes.every(b => isTargetAt(b.r, b.c))) {
      levelSolved = true;
      sfxLevelUp();
      showHint('✓ LEVEL COMPLETE!');
      gridEl.classList.add('level-complete-flash');
      setTimeout(() => {
        gridEl.classList.remove('level-complete-flash');
        currentLevel++;
        if (currentLevel >= LEVELS.length) {
          showWinScreen();
        } else {
          loadLevel(currentLevel);
          levelDisp.textContent = `LEVEL ${currentLevel+1}/${LEVELS.length}`;
          hintEl.textContent = '';
        }
      }, 1200);
    }
  }

  function showHint(msg) {
    hintEl.textContent = msg;
    setTimeout(() => { if (hintEl.textContent === msg) hintEl.textContent = ''; }, 1500);
  }

  function processCommand(cmd) {
    const c = cmd.trim().toUpperCase();
    let dr = 0, dc = 0;
    if (c === 'NORTH' || c === 'N') { dr = -1; }
    else if (c === 'SOUTH' || c === 'S') { dr = 1; }
    else if (c === 'EAST' || c === 'E') { dc = 1; }
    else if (c === 'WEST' || c === 'W') { dc = -1; }
    else { showHint('INVALID COMMAND'); return; }
    tryMove(dr, dc);
  }

  function showWinScreen() {
    document.getElementById('game-screen').classList.remove('active');
    document.getElementById('win-screen').classList.add('active');
    document.getElementById('final-stats').textContent = `All levels cleared in ${totalMoves} moves!`;
    sfxWin();
  }

  function startGame() {
    initAudio();
    currentLevel = 0;
    totalMoves = 0;
    document.getElementById('start-screen').classList.remove('active');
    document.getElementById('game-screen').classList.add('active');
    levelDisp.textContent = `LEVEL 1/${LEVELS.length}`;
    movesDisp.textContent = 'MOVES: 0';
    loadLevel(0);
    inputEl.value = '';
    inputEl.focus();
  }

  document.getElementById('btn-start').addEventListener('click', startGame);
  document.getElementById('btn-restart').addEventListener('click', startGame);

  inputEl.addEventListener('keydown', (e) => {
    if (e.key === 'Enter') {
      e.preventDefault();
      processCommand(inputEl.value);
      inputEl.value = '';
    }
  });

  // Allow single letter shortcuts without Enter
  window.addEventListener('keydown', (e) => {
    const gameActive = document.getElementById('game-screen').classList.contains('active');
    if (!gameActive || levelSolved) return;
    if (document.activeElement === inputEl && e.key.length === 1) {
      // Let the input handle it normally, but also allow single-key shortcuts
    }
  });

  // Handle resize
  window.addEventListener('resize', () => { renderGrid(); });

})();
</script>
</body>
</html>
