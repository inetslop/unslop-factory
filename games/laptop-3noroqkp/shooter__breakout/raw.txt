<script>
(function() {
const canvas = document.getElementById('game');
const ctx = canvas.getContext('2d');

let W, H;
function resize() {
W = window.innerWidth;
H = window.innerHeight;

// Audio
let audioCtx = null;
function initAudio() {
if (!audioCtx) {
audioCtx = new (window.AudioContext || window.webkitAudioContext)();
}
}
function playSound(type) {
if (!audioCtx) return;
const osc = audioCtx.createOscillator();
const gain = audioCtx.createGain();
osc.connect(gain);
gain.connect(audioCtx.destination);
switch(type) {
case 'shoot':
osc.type = 'square';
osc.frequency.setValueAtTime(400, audioCtx.currentTime);
osc.frequency.exponentialRampToValueAtTime(800, audioCtx.currentTime + 0.05);
gain.gain.setValueAtTime(0.15, audioCtx.currentTime);
gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.1);
osc.start();
osc.stop(audioCtx.currentTime + 0.1);
break;
case 'hit':
osc.type = 'sawtooth';
osc.frequency.setValueAtTime(200, audioCtx.currentTime);
osc.frequency.exponentialRampToValueAtTime(50, audioCtx.currentTime + 0.15);
gain.gain.setValueAtTime(0.3, audioCtx.currentTime);
gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.15);
osc.start();
osc.stop(audioCtx.currentTime + 0.15);
break;
case 'brick':
osc.type = 'triangle';
osc.frequency.setValueAtTime(600, audioCtx.currentTime);
osc.frequency.exponentialRampToValueAtTime(200, audioCtx.currentTime + 0.2);
gain.gain.setValueAtTime(0.25, audioCtx.currentTime);
gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.2);
osc.start();
osc.stop(audioCtx.currentTime + 0.2);
break;
case 'powerup':
osc.type = 'sine';
osc.frequency.setValueAtTime(400, audioCtx.currentTime);
osc.frequency.linearRampToValueAtTime(800, audioCtx.currentTime + 0.15);
gain.gain.setValueAtTime(0.2, audioCtx.currentTime);
gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.3);
osc.start();
osc.stop(audioCtx.currentTime + 0.3);
break;
case 'explosion':
osc.type = 'sawtooth';
osc.frequency.setValueAtTime(150, audioCtx.currentTime);
osc.frequency.exponentialRampToValueAtTime(30, audioCtx.currentTime + 0.4);
gain.gain.setValueAtTime(0.4, audioCtx.currentTime);
gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.4);
osc.start();
osc.stop(audioCtx.currentTime + 0.4);
break;
case 'lose':
osc.type = 'sawtooth';
osc.frequency.setValueAtTime(300, audioCtx.currentTime);
osc.frequency.exponentialRampToValueAtTime(80, audioCtx.currentTime + 0.5);
gain.gain.setValueAtTime(0.3, audioCtx.currentTime);
gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.5);
osc.start();
osc.stop(audioCtx.currentTime + 0.5);
break;
case 'win':
osc.type = 'sine';
osc.frequency.setValueAtTime(400, audioCtx.currentTime);
osc.frequency.linearRampToValueAtTime(800, audioCtx.currentTime + 0.2);
gain.gain.setValueAtTime(0.3, audioCtx.currentTime);
gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.5);
osc.start();
osc.stop(audioCtx.currentTime + 0.5);
break;
}
}

// Game State
const STATE = { START: 0, PLAYING: 1, GAME_OVER: 2, WIN: 3 };
let state = STATE.START;
let score = 0;
let lives = 3;
let level = 1;
let screenShake = 0;

// Player
let player = { x: 0, y: 0, w: 50, h: 30, speed: 8, shootCooldown: 0, shootRate: 200, powerLevel: 0, shield: false };
let bullets = [];
let enemyBullets = [];
let bricks = [];
let particles = [];
let powerUps = [];
let enemies = [];
let stars = [];

// Input
let keys = {};
window.addEventListener('keydown', e => {
keys[e.code] = true;
initAudio();
if (state === STATE.START && (e.code === 'Space' || e.code === 'Enter')) startGame();
else if ((state === STATE.GAME_OVER || state === STATE.WIN) && e.code === 'KeyR') startGame();
});
window.addEventListener('keyup', e => { keys[e.code] = false; });

// Stars background
function initStars() {
stars = [];
for (let i = 0; i < 100; i++) {
stars.push({ x: Math.random() * W, y: Math.random() * H, size: Math.random() * 2 + 0.5, speed: Math.random() * 0.3 + 0.1 });
}
}
initStars();

// Brick colors by level
function getBrickColor(row, col) {
const colors = [
['#ff4466','#ff6688','#ff4466','#ff6688'],
['#44ff66','#66ff88','#44ff66','#66ff88'],
['#4488ff','#66aaff','#4488ff','#66aaff'],
['#ffaa22','#ffcc44','#ffaa22','#ffcc44'],
['#aa44ff','#cc66ff','#aa44ff','#cc66ff']
];
return colors[row % 5][col % 4];
}

function createBricks() {
bricks = [];
const rows = Math.min(3 + level, 8);
const cols = Math.min(8 + level * 2, 14);
const bw = Math.min(60, (W - 80) / cols);
const bh = 25;
const offsetX = (W - cols * bw) / 2;
for (let r = 0; r < rows; r++) {
for (let c = 0; c < cols; c++) {
if (Math.random() > 0.15) {
const hp = r < 2 ? 3 : (r < 4 ? 2 : 1);
bricks.push({
x: offsetX + c * bw,
y: 60 + r * (bh + 4),
w: bw - 4,
h: bh,
hp: hp,
maxHp: hp,
color: getBrickColor(r, c)
});
}
}
}
// Enemy turrets
enemies = [];
const numEnemies = Math.min(1 + level, 5);
for (let i = 0; i < numEnemies; i++) {
enemies.push({
x: (W / (numEnemies + 1)) * (i + 1) - 20,
y: H * 0.35 + Math.random() * H * 0.15,
w: 40, h: 25,
hp: 2 + level,
maxHp: 2 + level,
shootTimer: Math.random() * 2000 + 1000,
moveDir: Math.random() > 0.5 ? 1 : -1,
speed: 1 + level * 0.3
});
}
}

function startGame() {
state = STATE.PLAYING;
score = 0;
lives = 3;
level = 1;
player.x = W / 2;
player.y = H - 60;
player.shootCooldown = 0;
player.powerLevel = 0;
player.shield = false;
player.shootRate = 200;
bullets = [];
enemyBullets = [];
particles = [];
powerUps = [];
createBricks();
}

function spawnParticles(x, y, color, count) {
for (let i = 0; i < count; i++) {
const angle = Math.random() * Math.PI * 2;
const speed = Math.random() * 4 + 1;
particles.push({
x: x, y: y,
vx: Math.cos(angle) * speed,
vy: Math.sin(angle) * speed,
life: 1,
decay: Math.random() * 0.03 + 0.02,
color: color,
size: Math.random() * 4 + 2
});
}
}

function spawnPowerUp(x, y) {
const types = ['rapid', 'spread', 'shield'];
const type = types[Math.floor(Math.random() * types.length)];
powerUps.push({ x: x, y: y, w: 20, h: 20, vy: 2, type: type });
}

function shoot() {
if (player.shootCooldown > 0) return;
player.shootCooldown = player.shootRate;
const bx = player.x;
const by = player.y - 15;
bullets.push({ x: bx, y: by, vx: 0, vy: -12, w: 4, h: 16 });
if (player.powerLevel >= 1) {
bullets.push({ x: bx - 8, y: by, vx: -1.5, vy: -12, w: 4, h: 16 });
bullets.push({ x: bx + 8, y: by, vx: 1.5, vy: -12, w: 4, h: 16 });
}
if (player.powerLevel >= 2) {
bullets.push({ x: bx - 14, y: by, vx: -3, vy: -10, w: 4, h: 16 });
bullets.push({ x: bx + 14, y: by, vx: 3, vy: -10, w: 4, h: 16 });
}
playSound('shoot');
}

function update(dt) {
if (state !== STATE.PLAYING) return;

// Player movement
if (keys['ArrowLeft'] || keys['KeyA']) player.x -= player.speed;
if (keys['ArrowRight'] || keys['KeyD']) player.x += player.speed;
player.x = Math.max(30, Math.min(W - 30, player.x));

// Shooting
if (keys['Space'] || keys['KeyW'] || keys['ArrowUp']) shoot();

// Cooldown
if (player.shootCooldown > 0) player.shootCooldown -= dt;

// Bullets
for (let i = bullets.length - 1; i >= 0; i--) {
const b = bullets[i];
b.x += b.vx;
b.y += b.vy;
if (b.y < -20 || b.x < -20 || b.x > W + 20) bullets.splice(i, 1);
}

// Enemy bullets
for (let i = enemyBullets.length - 1; i >= 0; i--) {
const b = enemyBullets[i];
b.x += b.vx;
b.y += b.vy;
if (b.y > H + 20 || b.y < -20 || b.x < -20 || b.x > W + 20) {
enemyBullets.splice(i, 1);
continue;
}
// Hit player
if (!player.shield && Math.abs(b.x - player.x) < 25 && Math.abs(b.y - player.y) < 20) {
enemyBullets.splice(i, 1);
lives--;
screenShake = 10;
playSound('explosion');
spawnParticles(player.x, player.y, '#ff4444', 20);
if (lives <= 0) {
state = STATE.GAME_OVER;
playSound('lose');
}
}
}

// Bricks
for (let i = bricks.length - 1; i >= 0; i--) {
const br = bricks[i];
for (let j = bullets.length - 1; j >= 0; j--) {
const b = bullets[j];
if (b.x + b.w/2 > br.x && b.x - b.w/2 < br.x + br.w &&
b.y + b.h/2 > br.y && b.y - b.h/2 < br.y + br.h) {
br.hp--;
bullets.splice(j, 1);
playSound('hit');
screenShake = 3;
if (br.hp <= 0) {
score += 10 * level;
playSound('brick');
spawnParticles(br.x + br.w/2, br.y + br.h/2, br.color, 15);
if (Math.random() < 0.2) spawnPowerUp(br.x + br.w/2, br.y);
bricks.splice(i, 1);
} else {
spawnParticles(b.x, b.y, '#ffffff', 3);
}
break;
}
}

// Enemies
for (let i = enemies.length - 1; i >= 0; i--) {
const e = enemies[i];
e.x += e.moveDir * e.speed;
if (e.x < 20 || e.x > W - 60) e.moveDir *= -1;

// Enemy shooting
e.shootTimer -= dt;
if (e.shootTimer <= 0) {
e.shootTimer = Math.random() * 2000 + 800;
const angle = Math.atan2(player.y - e.y, player.x - e.x);
enemyBullets.push({
x: e.x + e.w/2, y: e.y + e.h,
vx: Math.cos(angle) * 5, vy: Math.sin(angle) * 5,
w: 6, h: 6
});
}

// Bullets hit enemies
for (let j = bullets.length - 1; j >= 0; j--) {
const b = bullets[j];
if (b.x + b.w/2 > e.x && b.x - b.w/2 < e.x + e.w &&
b.y + b.h/2 > e.y && b.y - b.h/2 < e.y + e.h) {
e.hp--;
bullets.splice(j, 1);
playSound('hit');
screenShake = 3;
spawnParticles(b.x, b.y, '#ffaa00', 5);
if (e.hp <= 0) {
score += 50 * level;
playSound('explosion');
spawnParticles(e.x + e.w/2, e.y + e.h/2, '#ffaa00', 25);
enemies.splice(i, 1);
}
break;
}
}
}

// Power-ups
for (let i = powerUps.length - 1; i >= 0; i--) {
const p = powerUps[i];
p.y += p.vy;
if (p.y > H + 30) { powerUps.splice(i, 1); continue; }
if (Math.abs(p.x - player.x) < 40 && Math.abs(p.y - player.y) < 40) {
playSound('powerup');
switch(p.type) {
case 'rapid': player.shootRate = Math.max(80, player.shootRate * 0.6); break;
case 'spread': player.powerLevel = Math.min(2, player.powerLevel + 1); break;
case 'shield': player.shield = true; break;
}
spawnParticles(p.x, p.y, '#44ffff', 10);
powerUps.splice(i, 1);
}
}

// Particles
for (let i = particles.length - 1; i >= 0; i--) {
const p = particles[i];
p.x += p.vx;
p.y += p.vy;
p.vy += 0.1;
p.life -= p.decay;
if (p.life <= 0) particles.splice(i, 1);
}

// Screen shake decay
if (screenShake > 0) screenShake *= 0.85;
if (screenShake < 0.5) screenShake = 0;

// Check win
if (bricks.length === 0 && enemies.length === 0) {
level++;
createBricks();
playSound('win');
}

// Stars
for (let s of stars) {
s.y += s.speed;
if (s.y > H) { s.y = 0; s.x = Math.random() * W; }
}
}

function draw() {
ctx.save();
if (screenShake > 0) {
ctx.translate(
(Math.random() - 0.5) * screenShake,
(Math.random() - 0.5) * screenShake
);
}

// Background
const grad = ctx.createLinearGradient(0, 0, 0, H);
grad.addColorStop(0, '#0a0a2e');
grad.addColorStop(1, '#1a0a3e');
ctx.fillStyle = grad;
ctx.fillRect(-10, -10, W + 20, H + 20);

// Stars
ctx.fillStyle = '#ffffff';
for (let s of stars) {
ctx.globalAlpha = 0.5 + Math.sin(Date.now() * 0.001 + s.x) * 0.3;
ctx.fillRect(s.x, s.y, s.size, s.size);
}
ctx.globalAlpha = 1;

if (state === STATE.START) {
drawStartScreen();
} else {
drawGame();
if (state === STATE.GAME_OVER) drawGameOver();
if (state === STATE.WIN) drawWin();
}

ctx.restore();
}

function drawStartScreen() {
ctx.textAlign = 'center';
ctx.fillStyle = '#44ffff';
ctx.font = `bold ${Math.min(60, W * 0.08)}px monospace`;
ctx.fillText('BREAKOUT SHOOTER', W/2, H * 0.3);

ctx.fillStyle = '#ffffff';
ctx.font = `${Math.min(18, W * 0.03)}px monospace`;
ctx.fillText('Move: A/D or Arrow Keys', W/2, H * 0.45);
ctx.fillText('Shoot: Space / W / Up Arrow', W/2, H * 0.5);
ctx.fillText('Collect power-ups to upgrade your cannon!', W/2, H * 0.57);
ctx.fillText('Destroy all bricks and enemies to advance levels.', W/2, H * 0.64);

const pulse = Math.sin(Date.now() * 0.005) * 0.5 + 0.5;
ctx.globalAlpha = pulse;
ctx.fillStyle = '#ffaa00';
ctx.font = `bold ${Math.min(24, W * 0.04)}px monospace`;
ctx.fillText('PRESS SPACE TO START', W/2, H * 0.78);
ctx.globalAlpha = 1;
}

function drawGame() {
// Bricks
for (let br of bricks) {
const alpha = br.hp / br.maxHp;
ctx.fillStyle = br.color;
ctx.globalAlpha = 0.5 + alpha * 0.5;
ctx.fillRect(br.x, br.y, br.w, br.h);
ctx.globalAlpha = 1;
// Brick border
ctx.strokeStyle = '#ffffff';
ctx.lineWidth = 1;
ctx.strokeRect(br.x, br.y, br.w, br.h);
}

// Enemies
for (let e of enemies) {
ctx.fillStyle = '#ff6644';
ctx.fillRect(e.x, e.y, e.w, e.h);
ctx.strokeStyle = '#ffaa00';
ctx.lineWidth = 2;
ctx.strokeRect(e.x, e.y, e.w, e.h);
// HP bar
const hpW = (e.w - 6) * (e.hp / e.maxHp);
ctx.fillStyle = '#ff2222';
ctx.fillRect(e.x + 3, e.y - 8, hpW, 4);
ctx.strokeStyle = '#ffffff';
ctx.lineWidth = 1;
ctx.strokeRect(e.x + 3, e.y - 8, e.w - 6, 4);
}

// Bullets
for (let b of bullets) {
ctx.fillStyle = '#44ffff';
ctx.shadowColor = '#44ffff';
ctx.shadowBlur = 8;
ctx.fillRect(b.x - b.w/2, b.y - b.h/2, b.w, b.h);
ctx.shadowBlur = 0;
}

// Enemy bullets
for (let b of enemyBullets) {
ctx.fillStyle = '#ff4444';
ctx.shadowColor = '#ff4444';
ctx.shadowBlur = 6;
ctx.beginPath();
ctx.arc(b.x, b.y, 5, 0, Math.PI * 2);
ctx.fill();
ctx.shadowBlur = 0;
}

// Power-ups
for (let p of powerUps) {
const colors = { rapid: '#ffaa00', spread: '#44ff66', shield: '#4488ff' };
const icons = { rapid: 'R', spread: 'S', shield: 'O' };
ctx.fillStyle = colors[p.type];
ctx.shadowColor = colors[p.type];
ctx.shadowBlur = 10;
ctx.fillRect(p.x - p.w/2, p.y - p.h/2, p.w, p.h);
ctx.shadowBlur = 0;
ctx.fillStyle = '#000';
ctx.font = 'bold 14px monospace';
ctx.textAlign = 'center';
ctx.fillText(icons[p.type], p.x, p.y + 5);
}

// Player cannon
const px = player.x;
const py = player.y;
ctx.fillStyle = player.shield ? '#4488ff' : '#44ffff';
ctx.shadowColor = player.shield ? '#4488ff' : '#44ffff';
ctx.shadowBlur = 12;
// Cannon body
ctx.beginPath();
ctx.moveTo(px, py - 20);
ctx.lineTo(px + 20, py + 15);
ctx.lineTo(px - 20, py + 15);
ctx.closePath();
ctx.fill();
// Barrel
ctx.fillRect(px - 4, py - 30, 8, 20);
ctx.shadowBlur = 0;

// Shield effect
if (player.shield) {
ctx.strokeStyle = '#4488ff';
ctx.lineWidth = 2;
ctx.globalAlpha = 0.5 + Math.sin(Date.now() * 0.01) * 0.3;
ctx.beginPath();
ctx.arc(px, py, 35, 0, Math.PI * 2);
ctx.stroke();
ctx.globalAlpha = 1;
}

// Particles
for (let p of particles) {
ctx.globalAlpha = p.life;
ctx.fillStyle = p.color;
ctx.fillRect(p.x - p.size/2, p.y - p.size/2, p.size, p.size);
}
ctx.globalAlpha = 1;

// HUD
ctx.textAlign = 'left';
ctx.fillStyle = '#ffffff';
ctx.font = 'bold 16px monospace';
ctx.fillText(`SCORE: ${score}`, 20, 30);
ctx.fillText(`LIVES: ${'♥'.repeat(Math.max(0, lives))}`, 20, 55);
ctx.textAlign = 'right';
ctx.fillText(`LEVEL: ${level}`, W - 20, 30);
}

function drawGameOver() {
ctx.fillStyle = 'rgba(0,0,0,0.7)';
ctx.fillRect(0, 0, W, H);
ctx.textAlign = 'center';
ctx.fillStyle = '#ff4444';
ctx.font = `bold ${Math.min(50, W * 0.07)}px monospace`;
ctx.fillText('GAME OVER', W/2, H * 0.4);
ctx.fillStyle = '#ffffff';
ctx.font = `${Math.min(24, W * 0.04)}px monospace`;
ctx.fillText(`Final Score: ${score}`, W/2, H * 0.52);
ctx.fillText('Press R to Restart', W/2, H * 0.62);
}

function drawWin() {
ctx.fillStyle = 'rgba(0,0,0,0.7)';
ctx.fillRect(0, 0, W, H);
ctx.textAlign = 'center';
ctx.fillStyle = '#44ff66';
ctx.font = `bold ${Math.min(50, W * 0.07)}px monospace`;
ctx.fillText('LEVEL CLEAR!', W/2, H * 0.4);
}

// Game loop
let lastTime = 0;
function gameLoop(time) {
const dt = Math.min(time - lastTime, 50);
lastTime = time;
update(dt);
draw();
requestAnimationFrame(gameLoop);
}
requestAnimationFrame(gameLoop);

})();
</script>
</body>
</html>
