(function(){

// Audio
let audioCtx = null;
function initAudio(){ if(!audioCtx) audioCtx = new (window.AudioContext||window.webkitAudioContext)(); }
function playTone(freq, dur, type='square', vol=0.1){
if(!audioCtx) return;
const o = audioCtx.createOscillator(); const g = audioCtx.createGain();
o.type = type; o.frequency.value = freq; g.gain.value = vol;
g.gain.connect(audioCtx.destination); o.connect(g);
o.start(); g.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
o.stop(audioCtx.currentTime + dur);
}
function sfxMove(){ playTone(400, 0.08, 'square', 0.06); }
function sfxPush(){ playTone(250, 0.12, 'triangle', 0.08); }
function sfxPlace(){ playTone(800, 0.15,'sine',0.1); setTimeout(()=>playTone(1200,0.2,'sine',0.08),80); }
function sfxWin(){ for(let i=0;i<6;i++) setTimeout(()=>playTone(400+i*150,0.2,'square',0.07),i*100); }

// Levels: # wall, @ player, $ pod, . pad, + player on pad, * pod on pad

let level = 0, moves = 0, score = 0;

let won = false, animating = false;

let particles = [];
let ghosts = [];

function parseLevel(lv){
grid = lv.map(r => r.split(''));
pods = []; pads = [];
playerX=0; playerY=0; won=false; moves=0; history=[]; animating=false;
for(let y=0;y<grid.length;y++) for(let x=0;x<grid[y].length;x++){
const c = grid[y][x];
if(c==='@'||c==='+'){ playerX=x; playerY=y; }
if(c==='$'||c==='*'){ pods.push({x,y}); }
if(c==='.'||c==='+'||c==='*'){ pads.push({x,y}); }
}
cols = grid[0].length; rows = grid.length;
computeSize();
initGhosts();
}

function computeSize(){
const maxW = window.innerWidth * 0.92;
const maxH = window.innerHeight * 0.85;
cellSize = Math.floor(Math.min(maxW/cols, maxH/rows));
cellSize = Math.max(cellSize, 24);
W = cols * cellSize + cellSize*2;
H = rows * cellSize + cellSize*2;
canvas.width = W; canvas.height = H;
}

function initGhosts(){
ghosts = [];
const colors = ['#ff0000','#00ffff','#ffb8ff','#ff8000'];
for(let i=0;i<4;i++){
ghosts.push({ x: Math.random()*W, y: Math.random()*H, vx:(Math.random()-0.5)*30, vy:(Math.random()-0.5)*30, c: colors[i] });
}
}

function resize(){ computeSize(); }
window.addEventListener('resize', resize);

function isWall(x,y){ return x<0||y<0||x>=cols||y>=rows||grid[y][x]==='#'; }
function getPodAt(x,y){ return pods.find(p=>p.x===x&&p.y===y); }
function isPad(x,y){ return pads.some(p=>p.x===x&&p.y===y); }

function canMove(dx,dy){
const nx=playerX+dx, ny=playerY+dy;
if(isWall(nx,ny)) return null;
const pod = getPodAt(nx,ny);
if(pod){
const px=nx+dx, py=ny+dy;
if(isWall(px,py)||getPodAt(px,py)) return null;
return { type:'push', nx, ny, px, py };
}
return { type:'move', nx, ny };
}

function doMove(dx,dy){
if(animating || won) return;
const m = canMove(dx,dy);
if(!m) return;
initAudio();
history.push({x:playerX,y:playerY,pods:pods.map(p=>({...p}))});
moves++; hMov.textContent=moves; score+=10; hScore.textContent=score;

if(m.type==='move'){
animFromX=playerX; animFromY=playerY; animToX=m.nx; animToY=m.ny; animT=0; animating=true;
sfxMove();
}else{
animFromX=playerX; animFromY=playerY; animToX=m.nx; animToY=m.ny; animT=0; animating=true;
podAnim = { active:true, fromX:m.nx, fromY:m.ny, toX:m.px, toY:m.py, t:0 };
sfxPush();
}
}

function undo(){
if(history.length===0 || animating) return;
const h = history.pop();
playerX=h.x; playerY=h.y; pods=h.pods.map(p=>({...p}));
moves++; hMov.textContent=moves; score=Math.max(0,score-5); hScore.textContent=score;
}

function checkWin(){
return pods.every(p=>isPad(p.x,p.y));
}

function spawnParticles(x,y){
for(let i=0;i<12;i++){
const a = Math.random()*Math.PI*2, sp = 60+Math.random()*80;
particles.push({ x: x*cellSize+cellSize/2, y: y*cellSize+cellSize/2, vx:Math.cos(a)*sp, vy:Math.sin(a)*sp, life:1, c:`hsl(${180+Math.random()*60},100%,60%)` });
}
}

function update(dt){
gameTime += dt;
if(animating){
animT += dt/0.15;
if(animT>=1){
animT=1; animating=false;
playerX=animToX; playerY=animToY;
if(podAnim.active){
const pod = getPodAt(podAnim.fromX,podAnim.fromY);
if(pod){ pod.x=podAnim.toX; pod.y=podAnim.toY; }
podAnim.active=false;
sfxPlace();
spawnParticles(podAnim.toX, podAnim.toY);
}
if(checkWin()){ won=true; score+=100; hScore.textContent=score; sfxWin(); setTimeout(()=>{ showOverlay('LEVEL COMPLETE!','Press N for next level or R to restart'); },600); }
}
}else if(podAnim.active){
podAnim.t += dt/0.15;
if(podAnim.t>=1){ podAnim.t=1; podAnim.active=false; }
}

for(let p of particles){ p.x+=p.vx*dt; p.y+=p.vy*dt; p.life-=dt*2; if(p.life<0) p.life=0; }
particles = particles.filter(p=>p.life>0);

for(let g of ghosts){
g.x += g.vx*dt; g.y += g.vy*dt;
if(g.x<-40||g.x>W+40) g.vx*=-1;
if(g.y<-40||g.y>H+40) g.vy*=-1;
}
}

function drawGhost(g,alpha){
const s = cellSize*0.35;
ctx.globalAlpha = alpha;
ctx.fillStyle = g.c;
ctx.beginPath();
ctx.arc(g.x, g.y-s*0.2, s, Math.PI, 0);
ctx.lineTo(g.x+s, g.y+s*0.8);
for(let i=0;i<4;i++){ const bx=g.x+s - (i+1)*s/4; ctx.quadraticCurveTo(bx+s/8, g.y+s*0.5, bx, g.y+s*0.8); }
ctx.closePath(); ctx.fill();
// eyes
ctx.fillStyle='#fff';
ctx.beginPath(); ctx.arc(g.x-s*0.3,g.y-s*0.2,s*0.25,0,Math.PI*2); ctx.fill();
ctx.beginPath(); ctx.arc(g.x+s*0.3,g.y-s*0.2,s*0.25,0,Math.PI*2); ctx.fill();
ctx.fillStyle='#00f';
ctx.beginPath(); ctx.arc(g.x-s*0.25,g.y-s*0.1,s*0.12,0,Math.PI*2); ctx.fill();
ctx.beginPath(); ctx.arc(g.x+s*0.35,g.y-s*0.1,s*0.12,0,Math.PI*2); ctx.fill();
ctx.globalAlpha=1;
}

function drawPacMan(px,py,mouthAngle){
const cx = px*cellSize + cellSize/2;
const cy = py*cellSize + cellSize/2;
const r = cellSize*0.38;
ctx.fillStyle='#ffd700';
ctx.beginPath();
ctx.arc(cx,cy,r, mouthAngle, -mouthAngle+Math.PI*2);
ctx.lineTo(cx,cy); ctx.closePath(); ctx.fill();
// glow
ctx.shadowColor='#ffd700'; ctx.shadowBlur=8;
ctx.strokeStyle='rgba(255,215,0,0.4)'; ctx.lineWidth=2;
ctx.beginPath(); ctx.arc(cx,cy,r,mouthAngle,-mouthAngle+Math.PI*2); ctx.lineTo(cx,cy); ctx.closePath(); ctx.stroke();
ctx.shadowBlur=0;
}

function drawPod(px,py){
const cx = px*cellSize + cellSize/2;
const cy = py*cellSize + cellSize/2;
const r = cellSize*0.32;
const pulse = 1+Math.sin(gameTime*4)*0.08;
ctx.fillStyle='#0ff';
ctx.shadowColor='#0ff'; ctx.shadowBlur=12*pulse;
ctx.beginPath(); ctx.arc(cx,cy,r*pulse,0,Math.PI*2); ctx.fill();
ctx.shadowBlur=0;
ctx.fillStyle='rgba(255,255,255,0.5)';
ctx.beginPath(); ctx.arc(cx-r*0.2,cy-r*0.2,r*0.3,0,Math.PI*2); ctx.fill();
}

function drawPad(px,py){
const cx = px*cellSize + cellSize/2;
const cy = py*cellSize + cellSize/2;
const r = cellSize*0.28;
const pulse = 1+Math.sin(gameTime*3)*0.15;
ctx.strokeStyle='#f0f'; ctx.lineWidth=2;
ctx.shadowColor='#f0f'; ctx.shadowBlur=10*pulse;
ctx.beginPath(); ctx.arc(cx,cy,r*pulse,0,Math.PI*2); ctx.stroke();
ctx.shadowBlur=0;
ctx.fillStyle='rgba(255,0,255,0.1)';
ctx.beginPath(); ctx.arc(cx,cy,r*pulse,0,Math.PI*2); ctx.fill();
}

function draw(){
ctx.clearRect(0,0,W,H);
// floor
for(let y=0;y<rows;y++) for(let x=0;x<cols;x++){
if(grid[y][x]==='#') continue;
const px = x*cellSize+cellSize, py=y*cellSize+cellSize;
ctx.fillStyle='#1a1a3e';
ctx.fillRect(x*cellSize+cellSize,y*cellSize+cellSize,cellSize,cellSize);
// dots
ctx.fillStyle='rgba(255,255,200,0.2)';
ctx.beginPath(); ctx.arc(px+cellSize/2,py+cellSize/2,2,0,Math.PI*2); ctx.fill();
}
// walls
for(let y=0;y<rows;y++) for(let x=0;x<cols;x++){
if(grid[y][x]!=='#') continue;
const wx=x*cellSize+cellSize, wy=y*cellSize+cellSize;
ctx.fillStyle='#1a0a4e';
ctx.fillRect(wx,wy,cellSize,cellSize);
ctx.strokeStyle='#4433cc'; ctx.lineWidth=2;
ctx.strokeRect(wx+3,wy+3,cellSize-6,cellSize-6);
ctx.fillStyle='rgba(100,50,255,0.3)';
ctx.fillRect(wx+6,wy+6,cellSize-12,cellSize-12);
}
// pads
for(const p of pads) drawPad(p.x,p.y);
// pods
const podOffset = {};
if(animating && m_type==='push'){
const t=animT;
podOffset[`${podAnim.fromX}_${podAnim.fromY}`]={x:podAnim.fromX+(podAnim.toX-podAnim.fromX)*t, y:podAnim.fromY+(podAnim.toY-podAnim.fromY)*t};
}
for(const p of pods){
const off = podOffset[`${p.x}_${p.y}`];
if(off) drawPod(off.x,off.y); else drawPod(p.x,p.y);
}
// player
let ppX=playerX, ppY=playerY;
if(animating){ const t=animT; ppX=animFromX+(animToX-animFromX)*t; ppY=animFromY+(animToY-animFromY)*t; }
const mouth = 0.3+Math.abs(Math.sin(gameTime*5))*0.4;
drawPacMan(ppX,ppY,mouth);
// particles
for(const p of particles){ ctx.globalAlpha=p.life; ctx.fillStyle=p.c; ctx.fillRect(p.x-2,p.y-2,4,4); }
ctx.globalAlpha=1;
}

let m_type='';
function render(t){
const dt = Math.min((t - lastTime)/1000, 0.05); lastTime=t;
if(animating && !m_type) m_type = podAnim.active?'push':'move';
update(dt);
draw();
animId = requestAnimationFrame(render);
}

}
