// Audio
function initAudio() {
  if (!audioCtx) {
    audioCtx = new (window.AudioContext || window.webkitAudioContext)();
  }
}

function playSound(freq, duration, type, volume) {
  if (!audioCtx) return;
  const osc = audioCtx.createOscillator();
  const gain = audioCtx.createGain();
  osc.type = type || 'square';
  osc.frequency.value = freq;
  gain.gain.setValueAtTime(volume || 0.15, audioCtx.currentTime);
  gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + duration);
  osc.connect(gain);
  gain.connect(audioCtx.destination);
  osc.start();
  osc.stop(audioCtx.currentTime + duration);
}

function playMove() { playSound(220, 0.1, 'square', 0.1); }
function playPush() { playSound(330, 0.15, 'triangle', 0.15); }
function playTarget() { playSound(660, 0.3, 'sine', 0.2); playSound(880, 0.3, 'sine', 0.15); }
function playComplete() { 
  for (let i = 0; i < 5; i++) setTimeout(() => playSound(440 + i * 110, 0.2, 'sine', 0.2), i * 80); 
}

// Level generation
function generateLevel(lvl) {
  const baseSize = Math.min(5 + Math.floor(lvl / 2), 12);
  const w = baseSize + (lvl > 3 ? 1 : 0);
  const h = baseSize + (lvl > 5 ? 1 : 0);
  
  let attempts = 0;
  while (attempts < 100) {
    attempts++;
    let g = generateGrid(w, h, lvl);
    if (g && validateLevel(g)) {
      return g;
    }
  }
  // Fallback: simple open room
  let g = [];
  for (let y = 0; y < h; y++) {
    g.push([]);
    for (let x = 0; x < w; x++) {
      g[y].push((x === 0 || x === w-1 || y === 0 || y === h-1) ? 1 : 0);
    }
  }
  let boxes = Math.min(2 + Math.floor(lvl / 3), 4);
  for (let i = 0; i < boxes; i++) {
    g[1][1 + i] = 2; // targets along top

  for (let i = 0; i < boxes; i++) {
    g[h-2][1 + i] = 4; // boxes near bottom
  }
  g[h-2][0] = 5; // player
  return g;
}

function generateGrid(w, h, lvl) {
  let g = [];
  for (let y = 0; y < h; y++) {
    g.push([]);
    for (let x = 0; x < w; x++) {
      g[y].push((x === 0 || x === w-1 || y === 0 || y === h-1) ? 1 : 0);
    }
  }
  
  // Add internal walls
  const wallCount = Math.min(lvl, 6);
  for (let i = 0; i < wallCount; i++) {
    let wx = 1 + Math.floor(Math.random() * (w - 2));
    let wy = 1 + Math.floor(Math.random() * (h - 2));
    if (g[wy][wx] === 0) g[wy][wx] = 1;
  }
  
  // Ensure connectivity and place entities
  const numBoxes = Math.min(1 + Math.floor(lvl / 2), 5);
  
  // Place targets in corners/edges (easy to push into)
  let targets = [];
  let candidates = [];
  for (let y = 1; y < h - 1; y++) {
    for (let x = 1; x < w - 1; x++) {
      if (g[y][x] === 0) {
        // Prefer positions adjacent to walls
        let wallAdj = 0;
        if (y > 0 && g[y-1][x] === 1) wallAdj++;
        if (y < h-1 && g[y+1][x] === 1) wallAdj++;
        if (x > 0 && g[y][x-1] === 1) wallAdj++;
        if (x < w-1 && g[y][x+1] === 1) wallAdj++;
        candidates.push({ x, y, wallAdj });
      }
    }
  }
  
  // Sort by wall adjacency (prefer spots against walls)
  candidates.sort((a, b) => b.wallAdj - a.wallAdj);
  
  for (let i = 0; i < numBoxes && i < candidates.length; i++) {
    targets.push(candidates[i]);
    g[candidates[i].y][candidates[i].x] = 2;
  }
  
  // Place boxes in open areas away from targets
  let boxes = [];
  let boxCandidates = [];
  for (let y = 1; y < h - 1; y++) {
    for (let x = 1; x < w - 1; x++) {
      if (g[y][x] === 0) {
        let minDist = Infinity;
        for (let t of targets) {
          let d = Math.abs(x - t.x) + Math.abs(y - t.y);
          minDist = Math.min(minDist, d);
        }
        boxCandidates.push({ x, y, dist: minDist });
      }
    }
  }
  boxCandidates.sort((a, b) => b.dist - a.dist);
  
  for (let i = 0; i < numBoxes && i < boxCandidates.length; i++) {
    boxes.push(boxCandidates[i]);
    g[boxCandidates[i].y][boxCandidates[i].x] = 4;
  }
  
  // Place player
  let px, py;
  for (let y = 1; y < h - 1; y++) {
    for (let x = 1; x < w - 1; x++) {
      if (g[y][x] === 0) {
        // Check player can reach a box
        let canReachBox = false;
        for (let b of boxes) {
          if (bfs(g, x, y, b.x, b.y)) { canReachBox = true; break; }
        }
        if (canReachBox) { px = x; py = y; break; }
      }
    }
    if (px !== undefined) break;
  }
  
  if (px === undefined) {
    // Just find any open spot
    for (let y = 1; y < h - 1; y++) {
      for (let x = 1; x < w - 1; x++) {
        if (g[y][x] === 0) { px = x; py = y; break; }
      }
      if (px !== undefined) break;
    }
  }
  
  if (px === undefined) return null;
  g[py][px] = 5;
  
  return g;
}

function bfs(g, sx, sy, tx, ty) {
  let h = g.length, w = g[0].length;
  let visited = Array.from({length: h}, () => Array(w).fill(false));
  let queue = [[sx, sy]];
  visited[sy][sx] = true;
  
  while (queue.length > 0) {
    let [cx, cy] = queue.shift();
    if (cx === tx && cy === ty) return true;
    
    const dirs = [[0,-1],[0,1],[-1,0],[1,0]];
    for (let [dx, dy] of dirs) {
      let nx = cx + dx, ny = cy + dy;
      if (nx >= 0 && nx < w && ny >= 0 && ny < h && !visited[ny][nx]) {
        if (g[ny][nx] !== 1 && g[ny][nx] !== 4) {
          visited[ny][nx] = true;
          queue.push([nx, ny]);
        }
      }
    }
  }
  return false;
}

function validateLevel(g) {
  // Check all boxes can potentially reach targets (simple heuristic)
  let h = g.length, w = g[0].length;
  for (let y = 1; y < h - 1; y++) {
    for (let x = 1; x < w - 1; x++) {
      if (g[y][x] === 4) {
        // Check box isn't stuck in corner with no push direction
        let above = g[y-1] && g[y-1][x] !== 1;
        let below = g[y+1] && g[y+1][x] !== 1;
        let left = g[y][x-1] !== 1;
        let right = g[y][x+1] !== 1;
        if (!above && !below && !left && !right) return false;
      }
    }
  }
  return true;
}

function startLevel(lvl) {
  level = lvl;
  moves = 0;
  grid = generateLevel(lvl);
  
  let h = grid.length, w = grid[0].length;
  for (let y = 0; y < h; y++) {
    for (let x = 0; x < w; x++) {
      if (grid[y][x] === 5) { playerX = x; playerY = y; }
    }
  }
  
  cellSize = Math.min(
    (W - 40) / w, 
    (H - 120) / h, 
    60
  );
}

function checkComplete() {
  for (let row of grid) {
    for (let cell of row) {
      if (cell === 4) return false; // box not on target
    }
  }
  return true;
}

function move(dx, dy) {
  if (gameState !== STATE.PLAYING) return;
  initAudio();
  
  let nx = playerX + dx;
  let ny = playerY + dy;
  let h = grid.length, w = grid[0].length;
  
  if (nx < 0 || nx >= w || ny < 0 || ny >= h) return;
  if (grid[ny][nx] === 1) return; // wall
  
  if (grid[ny][nx] === 4 || grid[ny][nx] === 3) {
    // Try to push box
    let bx = nx + dx, by = ny + dy;
    if (bx < 0 || bx >= w || by < 0 || by >= h) return;
    if (grid[by][bx] === 1 || grid[by][bx] === 4 || grid[by][bx] === 3) return;
    
    // Push!
    let target = (grid[by][bx] === 2);
    grid[playerY][playerX] = 0;
    grid[ny][nx] = (target ? 6 : 4); // box on floor or target
    grid[by][bx] = 5; // player at old box pos
    playerX = nx; playerY = ny;
    moves++; totalMoves++;
    
    if (target) { playTarget(); }
    else { playPush(); }
    
    // Check complete
    setTimeout(() => {
      if (checkComplete()) {
        gameState = STATE.COMPLETE;
        completeTimer = 0;
        playComplete();
        spawnParticles();
      }
    }, 50);
  } else if (grid[ny][nx] === 0 || grid[ny][nx] === 2) {
    // Move to empty or target
    grid[playerY][playerX] = 0;
    grid[ny][nx] = 5;
    playerX = nx; playerY = ny;
    moves++; totalMoves++;
    playMove();
  }
}

// Particles
function spawnParticles() {
  let h = grid.length, w = grid[0].length;
  for (let y = 0; y < h; y++) {
    for (let x = 0; x < w; x++) {
      if (grid[y][x] === 6) { // box on target
        let cx = getCellCenter(x, y);
        for (let i = 0; i < 8; i++) {
          let angle = Math.random() * Math.PI * 2;
          let speed = 1 + Math.random() * 3;
          particles.push({
            x: cx.x, y: cx.y,
            vx: Math.cos(angle) * speed,
            vy: Math.sin(angle) * speed - 1,
            life: 60,
            color: `hsl(${40 + Math.random() * 30}, 100%, ${50 + Math.random() * 30}%)`
          });
        }
      }
    }
  }
}

function getCellCenter(gx, gy) {
  let w = grid[0].length, h = grid.length;
  let ox = (W - w * cellSize) / 2;
  let oy = (H - h * cellSize) / 2 + 30;
  return { x: ox + gx * cellSize + cellSize/2, y: oy + gy * cellSize + cellSize/2 };
}

function updateParticles() {
  for (let p of particles) {
    p.x += p.vx;
    p.y += p.vy;
    p.vy += 0.1;
    p.life--;
  }
  particles = particles.filter(p => p.life > 0);
}

}

function drawTarget(cx, cy) {
  let s = cellSize * 0.5;
  let x = cx + cellSize/2, y = cy + cellSize/2;
  ctx.fillStyle = 'rgba(255, 215, 0, 0.3)';
  ctx.beginPath();
  ctx.arc(x, y, s/2, 0, Math.PI * 2);
  ctx.fill();
  
  // Star shape
  ctx.fillStyle = '#ffd700';
  ctx.font = `${s}px monospace`;
  ctx.textAlign = 'center';
  ctx.textBaseline = 'middle';
  ctx.fillText('★', x, y + 2);
}

function drawBox(cx, cy, onTarget) {
  let pad = cellSize * 0.1;
  let s = cellSize - pad * 2;
  ctx.fillStyle = onTarget ? '#4a9' : '#c8752c';
  ctx.fillRect(cx + pad, cy + pad, s, s);
  ctx.strokeStyle = onTarget ? '#6fd' : '#e8a040';
  ctx.lineWidth = 2;
  ctx.strokeRect(cx + pad + 3, cy + pad + 3, s - 6, s - 6);
  
  // Box detail
  ctx.fillStyle = 'rgba(255,255,255,0.1)';
  ctx.fillRect(cx + pad + 4, cy + pad + 4, s/3, s/3);
}

function drawPlayer(cx, cy) {
  let x = cx + cellSize/2, y = cy + cellSize/2;
  let r = cellSize * 0.35;
  
  // Glow
  ctx.shadowColor = '#4fc3f7';
  ctx.shadowBlur = 10;
  
  // Body
  ctx.fillStyle = '#4fc3f7';
  ctx.beginPath();
  ctx.arc(x, y + r*0.2, r * 0.8, 0, Math.PI * 2);
  ctx.fill();
  
  // Face
  ctx.shadowBlur = 0;
  ctx.fillStyle = '#fff';
  let eyeR = cellSize * 0.06;
  ctx.beginPath(); ctx.arc(x - r*0.3, y - r*0.15, eyeR, 0, Math.PI*2); ctx.fill();
  ctx.beginPath(); ctx.arc(x + r*0.3, y - r*0.15, eyeR, 0, Math.PI*2); ctx.fill();
  
  // Smile
  ctx.strokeStyle = '#fff';
  ctx.lineWidth = 2;
  ctx.beginPath();
  ctx.arc(x, y + r*0.15, r*0.3, 0.2, Math.PI - 0.2);
  ctx.stroke();
}

      case 'ArrowLeft': case 'KeyA': move(-1, 0); break;
      case 'ArrowRight': case 'KeyD': move(1, 0); break;
      case 'KeyR': startLevel(level); break;
    }
    e.preventDefault();
  }
});

// Game loop
function loop() {
  animTimer++;
  updateParticles();
  draw();
  requestAnimationFrame(loop);
}

loop();
