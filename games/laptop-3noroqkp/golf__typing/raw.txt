function initAudio() {

function playTone(freq, dur, type = 'sine', vol = 0.3) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = type;
    osc.frequency.value = freq;
    gain.gain.setValueAtTime(vol, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + dur);
}
function playTyping() { playTone(600 + Math.random() * 200, 0.05, 'sine', 0.15); }
function playError() { playTone(150, 0.2, 'square', 0.2); }
function playHoleIn() { 
    playTone(523, 0.15, 'sine', 0.3); 
    setTimeout(() => playTone(659, 0.15, 'sine', 0.3), 100);

function handleKey(e) {
    initAudio();
    
    if (gameState === 'start') {
        if (e.key === 'Enter' || e.key === ' ') startGame();
        return;
    }
    
    if (gameState === 'holeComplete') {
        if (e.key === 'Enter' || e.key === ' ') {
            nextHole();
        }
        return;
    }
    
    if (gameState === 'gameOver') {
        if (e.key === 'Enter' || e.key === ' ') startGame();
        return;
    }
    
    if (gameState !== 'playing') return;
    
    const word = getWord();
    if (!word) return;
    
    const expectedChar = word[currentCharIndex];
    const typedChar = e.key.toLowerCase();
    
    if (typedChar === expectedChar) {

        currentCharIndex++;
        typingFeedback = 'correct';
        feedbackTimer = 0.3;
        
        // Move ball forward
        const progress = currentCharIndex / word.length;

            // Word complete!
            strokes++;

            ballTargetX = flagX;

            if (strokes < par) {

                birdieText = "Par";

                birdieText = `Bogey (+${strokes - par})`;

            // Particles
            for (let i = 0; i < 20; i++) {
                particles.push({
                    x: flagX, y: ballY,

                    life: 1,

        feedbackTimer = 0.5;
        flashTimer = 0.3;

    }

function drawFlag(x, y, time) {
    // Pole
    ctx.strokeStyle = '#FFFFFF';
    ctx.lineWidth = 3;
    ctx.beginPath();
    ctx.moveTo(x, y + 30);
    ctx.lineTo(x, y - 20);
    ctx.stroke();
    
    // Flag
    const wave = Math.sin(time * 0.005) * 3;
    ctx.fillStyle = '#FF4444';
    ctx.beginPath();
    ctx.moveTo(x, y - 20);
    ctx.lineTo(x + 20 + wave, y - 12);
    ctx.lineTo(x, y - 4);
    ctx.closePath();
    ctx.fill();
    
    // Hole
    ctx.fillStyle = '#1a1a1a';
    ctx.beginPath();
    ctx.ellipse(x, y + 30, 8, 5, 0, 0, Math.PI * 2);
    ctx.fill();
}

function roundRect(x, y, w, h, r) {
    ctx.beginPath();
    ctx.moveTo(x + r, y);
    ctx.lineTo(x + w - r, y);
    ctx.quadraticCurveTo(x + w, y, x + w, y + r);
    ctx.lineTo(x + w, y + h - r);
    ctx.quadraticCurveTo(x + w, y + h, x + w - r, y + h);
    ctx.lineTo(x + r, y + h);
    ctx.quadraticCurveTo(x, y + h, x, y + h - r);
    ctx.lineTo(x, y + r);
    ctx.quadraticCurveTo(x, y, x + r, y);
    ctx.closePath();
    ctx.fill();
}

let lastTime = 0;
function gameLoop(time) {
    const dt = Math.min((time - lastTime) / 16.67, 3);
    lastTime = time;
    
    // Animate ball
    ballX += (ballTargetX - ballX) * 0.12;
    
    // Update particles
    for (let i = particles.length - 1; i >= 0; i--) {
        particles[i].x += particles[i].vx * dt;
        particles[i].y += particles[i].vy * dt;
        particles[i].vy += 0.3 * dt;
        particles[i].life -= 0.02 * dt;
        if (particles[i].life <= 0) particles.splice(i, 1);
    }
    
    // Timers
    if (flashTimer > 0) flashTimer -= 0.016 * dt;
    if (feedbackTimer > 0) feedbackTimer -= 0.016 * dt;
    if (gameState === 'holeComplete') holeAnimTimer += 0.016 * dt;
    
    // Clear
    ctx.clearRect(0, 0, W, H);
    
    drawScene(time);
    
    requestAnimationFrame(gameLoop);
}

requestAnimationFrame(gameLoop);
