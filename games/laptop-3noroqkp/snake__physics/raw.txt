<!DOCTYPE html>
<html>
<head>
<meta charset="UTF-8">
<title>Physics Snake</title>
<style>
* { margin: 0; padding: 0; box-sizing: border-box; }

<script>
const canvas = document.getElementById('game');
const ctx = canvas.getContext('2d');

let W, H;
function resize() {
    W = window.innerWidth;
    H = window.innerHeight;
    canvas.width = W;
    canvas.height = H;
}
window.addEventListener('resize', resize);
resize();

// Audio
let audioCtx = null;
function initAudio() {
    if (!audioCtx) {
        audioCtx = new (window.AudioContext || window.webkitAudioContext)();
    }
}
function playEatSound() {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.type = 'sine';
    osc.frequency.setValueAtTime(600, audioCtx.currentTime);
    osc.frequency.exponentialRampToValueAtTime(1200, audioCtx.currentTime + 0.1);
    gain.gain.setValueAtTime(0.3, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.15);
    osc.start(audioCtx.currentTime);
    osc.stop(audioCtx.currentTime + 0.15);
}
function playDeathSound() {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.type = 'sawtooth';
    osc.frequency.setValueAtTime(300, audioCtx.currentTime);
    osc.frequency.exponentialRampToValueAtTime(60, audioCtx.currentTime + 0.5);
    gain.gain.setValueAtTime(0.4, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.5);
    osc.start(audioCtx.currentTime);
    osc.stop(audioCtx.currentTime + 0.5);
}

// Game state
const SEGMENT_RADIUS = 8;
const CONSTRAINT_DIST = 14;
const GRAVITY = 0.35;
const DAMPING = 0.98;
const HEAD_FORCE = 1.8;
const INITIAL_SEGMENTS = 12;

let segments = [];
let food = null;
let score = 0;
let highScore = 0;
let gameState = 'start'; // start, playing, dead
let particles = [];
let time = 0;

try {
    highScore = parseInt(localStorage.getItem('physicsSnakeHigh') || '0');
} catch(e) { highScore = 0; }

function saveHighScore() {
    try {
        localStorage.setItem('physicsSnakeHigh', highScore.toString());
    } catch(e) {}
}

function initSegments() {
    segments = [];
    const startX = W / 2;
    const startY = H / 3;
    for (let i = 0; i < INITIAL_SEGMENTS; i++) {
        segments.push({
            x: startX,
            y: startY - i * CONSTRAINT_DIST,
            px: startX,
            py: startY - i * CONSTRAINT_DIST
        });
    }
}

function spawnFood() {
    const margin = 50;
    food = {
        x: margin + Math.random() * (W - margin * 2),
        y: margin + Math.random() * (H - margin * 2),
        radius: 12,
        pulse: 0
    };
}

function spawnParticles(x, y, color) {
    for (let i = 0; i < 15; i++) {
        const angle = Math.random() * Math.PI * 2;
        const speed = 2 + Math.random() * 4;
        particles.push({
            x: x, y: y,
            vx: Math.cos(angle) * speed,
            vy: Math.sin(angle) * speed - 2,
            life: 1,
            color: color
        });
    }
}

function startGame() {
    initSegments();
    score = 0;
    particles = [];
    spawnFood();
    gameState = 'playing';
}

// Input
const keys = {};
window.addEventListener('keydown', (e) => {
    keys[e.code] = true;
    if (gameState === 'start' || gameState === 'dead') {
        initAudio();
        startGame();
    }
});
window.addEventListener('keyup', (e) => {
    keys[e.code] = false;
});

// Physics update
function updatePhysics() {
    // Apply forces to head
    const head = segments[0];
    let fx = 0, fy = 0;
    if (keys['ArrowUp'] || keys['KeyW']) { fy -= HEAD_FORCE; }
    if (keys['ArrowDown'] || keys['KeyS']) { fy += HEAD_FORCE; }
    if (keys['ArrowLeft'] || keys['KeyA']) { fx -= HEAD_FORCE; }
    if (keys['ArrowRight'] || keys['KeyD']) { fx += HEAD_FORCE; }

    // Verlet integration for all segments
    for (let i = 0; i < segments.length; i++) {
        const seg = segments[i];
        let vx = (seg.x - seg.px) * DAMPING;
        let vy = (seg.y - seg.py) * DAMPING;

        if (i === 0) {
            vx += fx;
            vy += fy;
        }

        seg.px = seg.x;
        seg.py = seg.y;
        seg.x += vx;
        seg.y += vy + GRAVITY;

        // Wall collisions
        if (seg.x < SEGMENT_RADIUS) { seg.x = SEGMENT_RADIUS; seg.px = seg.x + (seg.x - seg.px) * 0.5; }
        if (seg.x > W - SEGMENT_RADIUS) { seg.x = W - SEGMENT_RADIUS; seg.px = seg.x + (seg.x - seg.px) * 0.5; }
        if (seg.y < SEGMENT_RADIUS) { seg.y = SEGMENT_RADIUS; seg.py = seg.y + (seg.y - seg.py) * 0.5; }
        if (seg.y > H - SEGMENT_RADIUS) { seg.y = H - SEGMENT_RADIUS; seg.py = seg.y + (seg.y - seg.py) * 0.5; }
    }

    // Constraint solving (multiple iterations for stability)
    for (let iter = 0; iter < 8; iter++) {
        for (let i = 0; i < segments.length - 1; i++) {
            const a = segments[i];
            const b = segments[i + 1];
            let dx = b.x - a.x;
            let dy = b.y - a.y;
            let dist = Math.sqrt(dx * dx + dy * dy);
            if (dist === 0) continue;
            const diff = (dist - CONSTRAINT_DIST) / dist;
            const ox = dx * diff * 0.5;
            const oy = dy * diff * 0.5;

            if (i > 0) { a.x += ox; a.y += oy; }
            b.x -= ox; b.y -= oy;
        }
    }

    // Check food collision
    if (food) {
        const hx = segments[0].x;
        const hy = segments[0].y;
        const fdx = hx - food.x;
        const fdy = hy - food.y;
        if (Math.sqrt(fdx * fdx + fdy * fdy) < SEGMENT_RADIUS + food.radius) {
            // Eat!
            playEatSound();
            score++;
            spawnParticles(food.x, food.y, '#ff6b35');
            // Add segments to tail
            const lastSeg = segments[segments.length - 1];
            for (let i = 0; i < 2; i++) {
                segments.push({
                    x: lastSeg.x,
                    y: lastSeg.y,
                    px: lastSeg.x,
                    py: lastSeg.y
                });
            }
            food = null;
            spawnFood();
        }
    }

    // Check self-collision (head vs body, skip first 6 segments)
    if (segments.length > 8) {
        const hx = segments[0].x;
        const hy = segments[0].y;
        for (let i = 6; i < segments.length; i++) {
            const dx = hx - segments[i].x;
            const dy = hy - segments[i].y;
            if (Math.sqrt(dx * dx + dy * dy) < SEGMENT_RADIUS * 1.5) {
                // Death!
                playDeathSound();
                gameState = 'dead';
                spawnParticles(hx, hy, '#ff0040');
                if (score > highScore) {
                    highScore = score;
                    saveHighScore();
                }
                break;
            }
        }
    }

    // Update particles
    for (let i = particles.length - 1; i >= 0; i--) {
        const p = particles[i];
        p.x += p.vx;
        p.y += p.vy;
        p.vy += 0.2;
        p.life -= 0.03;
        if (p.life <= 0) particles.splice(i, 1);
    }

    // Food pulse
    if (food) food.pulse += 0.1;
}

// Render
function draw() {
    ctx.fillStyle = '#0f0f1a';
    ctx.fillRect(0, 0, W, H);

    // Draw subtle grid
    ctx.strokeStyle = 'rgba(255,255,255,0.03)';
    ctx.lineWidth = 1;
    const gridSize = 50;
    for (let x = 0; x < W; x += gridSize) {
        ctx.beginPath(); ctx.moveTo(x, 0); ctx.lineTo(x, H); ctx.stroke();
    }
    for (let y = 0; y < H; y += gridSize) {
        ctx.beginPath(); ctx.moveTo(0, y); ctx.lineTo(W, y); ctx.stroke();
    }

    // Draw food
    if (food && gameState === 'playing') {
        const pulse = Math.sin(food.pulse) * 3;
        const r = food.radius + pulse;
        const grad = ctx.createRadialGradient(food.x, food.y, 0, food.x, food.y, r);
        grad.addColorStop(0, '#ff6b35');
        grad.addColorStop(0.7, '#ff4500');
        grad.addColorStop(1, 'rgba(255,69,0,0)');
        ctx.fillStyle = grad;
        ctx.beginPath();
        ctx.arc(food.x, food.y, r + 5, 0, Math.PI * 2);
        ctx.fill();

        ctx.fillStyle = '#ffaa35';
        ctx.beginPath();
        ctx.arc(food.x, food.y, r - 4, 0, Math.PI * 2);
        ctx.fill();
    }

    // Draw particles
    for (const p of particles) {
        ctx.globalAlpha = p.life;
        ctx.fillStyle = p.color;
        ctx.beginPath();
        ctx.arc(p.x, p.y, 3 * p.life, 0, Math.PI * 2);
        ctx.fill();
    }
    ctx.globalAlpha = 1;

    // Draw snake segments (from tail to head)
    if (segments.length > 0) {
        for (let i = segments.length - 1; i >= 0; i--) {
            const seg = segments[i];
            const t = i / segments.length;
            const r = Math.floor(50 + t * 80);
            const g = Math.floor(200 - t * 100);
            const b = Math.floor(50 + t * 50);

            // Glow
            ctx.fillStyle = `rgba(${r},${g},${b},0.3)`;
            ctx.beginPath();
            ctx.arc(seg.x, seg.y, SEGMENT_RADIUS + 3, 0, Math.PI * 2);
            ctx.fill();

            ctx.fillStyle = `rgb(${r},${g},${b})`;
            ctx.beginPath();
            ctx.arc(seg.x, seg.y, SEGMENT_RADIUS, 0, Math.PI * 2);
            ctx.fill();

            // Highlight
            ctx.fillStyle = `rgba(255,255,255,${0.1 + t * 0.1})`;
            ctx.beginPath();
            ctx.arc(seg.x - 2, seg.y - 2, SEGMENT_RADIUS * 0.4, 0, Math.PI * 2);
            ctx.fill();
        }

        // Draw eyes on head
        const head = segments[0];
        let nextX = segments[1].x - head.x;
        let nextY = segments[1].y - head.y;
        const len = Math.sqrt(nextX * nextX + nextY * nextY) || 1;
        const nx = nextX / len;
        const ny = nextY / len;

        // Eye positions (perpendicular to direction)
        const perpX = -ny;
        const perpY = nx;
        const eyeOffset = 4;
        const eyeR = 3.5;

        ctx.fillStyle = 'white';
        ctx.beginPath();
        ctx.arc(head.x + perpX * eyeOffset + nx * 3, head.y + perpY * eyeOffset + ny * 3, eyeR, 0, Math.PI * 2);
        ctx.fill();
        ctx.beginPath();
        ctx.arc(head.x - perpX * eyeOffset + nx * 3, head.y - perpY * eyeOffset + ny * 3, eyeR, 0, Math.PI * 2);
        ctx.fill();

        // Pupils
        ctx.fillStyle = '#1a1a2e';
        ctx.beginPath();
        ctx.arc(head.x + perpX * eyeOffset + nx * 4.5, head.y + perpY * eyeOffset + ny * 4.5, 2, 0, Math.PI * 2);
        ctx.fill();
        ctx.beginPath();
        ctx.arc(head.x - perpX * eyeOffset + nx * 4.5, head.y - perpY * eyeOffset + ny * 4.5, 2, 0, Math.PI * 2);
        ctx.fill();
    }

    // UI
    if (gameState === 'playing') {
        ctx.fillStyle = 'rgba(255,255,255,0.9)';
        ctx.font = 'bold 24px monospace';
        ctx.textAlign = 'left';
        ctx.fillText(`Score: ${score}`, 20, 35);
        ctx.font = '14px monospace';
        ctx.fillStyle = 'rgba(255,255,255,0.5)';
        ctx.fillText(`Best: ${highScore}`, 20, 55);
    }

    if (gameState === 'start') {
        drawStartScreen();
    }

    if (gameState === 'dead') {
        drawDeathScreen();
    }
}

function drawStartScreen() {
    ctx.fillStyle = 'rgba(0,0,0,0.7)';
    ctx.fillRect(0, 0, W, H);

    ctx.textAlign = 'center';
    ctx.fillStyle = '#4dff4d';
    ctx.font = 'bold 52px monospace';
    ctx.fillText('PHYSICS SNAKE', W / 2, H / 2 - 80);

    ctx.fillStyle = 'rgba(255,255,255,0.8)';
    ctx.font = '18px monospace';
    ctx.fillText('Steer your snake to eat food and grow!', W / 2, H / 2 - 30);
    ctx.fillText('Don\'t let your head touch your body!', W / 2, H / 2 + 5);

    ctx.fillStyle = 'rgba(255,255,255,0.6)';
    ctx.font = '16px monospace';
    ctx.fillText('Arrow Keys / WASD to move', W / 2, H / 2 + 50);

    const blink = Math.sin(time * 3) > 0;
    if (blink) {
        ctx.fillStyle = '#ffaa35';
        ctx.font = 'bold 20px monospace';

    }

function drawDeathScreen() {
    ctx.fillStyle = 'rgba(0,0,0,0.6)';
    ctx.fillRect(0, 0, W, H);

    ctx.textAlign = 'center';
    ctx.fillStyle = '#ff0040';
    ctx.font = 'bold 48px monospace';
    ctx.fillText('SNAKE DOWN!', W / 2, H / 2 - 50);

    ctx.fillStyle = 'rgba(255,255,255,0.9)';
    ctx.font = '24px monospace';
    ctx.fillText(`Score: ${score}`, W / 2, H / 2 + 10);

    if (score >= highScore && score > 0) {
        ctx.fillStyle = '#ffaa35';
        ctx.font = '20px monospace';
        ctx.fillText('NEW HIGH SCORE!', W / 2, H / 2 + 45);
    }

    const blink = Math.sin(time * 3) > 0;
    if (blink) {
        ctx.fillStyle = 'rgba(255,255,255,0.7)';
        ctx.font = '18px monospace';
        ctx.fillText('Press any key to restart', W / 2, H / 2 + 90);
    }
}

// Game loop
let lastTime = 0;
function gameLoop(timestamp) {
    time += 0.016;
    if (gameState === 'playing') {
        updatePhysics();
    } else {
        // Update particles even when not playing
        for (let i = particles.length - 1; i >= 0; i--) {
            const p = particles[i];
            p.x += p.vx;
            p.y += p.vy;
            p.vy += 0.2;
            p.life -= 0.03;
            if (p.life <= 0) particles.splice(i, 1);
        }
    }
    draw();
    requestAnimationFrame(gameLoop);
}

initSegments();
spawnFood();
requestAnimationFrame(gameLoop);
</script>
