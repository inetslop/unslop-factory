let audioCtx = null;
function initAudio() {
    if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();

function playSound(type) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.connect(gain); gain.connect(audioCtx.destination);
    if (type === 'shoot') { osc.frequency.value = 400; osc.type = 'square'; gain.gain.setValueAtTime(0.1, audioCtx.currentTime); gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime+0.1); }
    else if (type === 'explosion') { osc.frequency.value = 80; osc.type = 'sawtooth'; gain.gain.setValueAtTime(0.2, audioCtx.currentTime); gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime+0.3); }
    else if (type === 'death') { osc.frequency.value = 200; osc.type = 'sine'; gain.gain.setValueAtTime(0.15, audioCtx.currentTime); gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime+0.4); }
    else if (type === 'place') { osc.frequency.value = 600; osc.type = 'sine'; gain.gain.setValueAtTime(0.1, audioCtx.currentTime); gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime+0.15); }
    else if (type === 'zombie') { osc.frequency.value = 150; osc.type = 'sawtooth'; gain.gain.setValueAtTime(0.1, audioCtx.currentTime); gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime+0.2); }
    osc.start(); osc.stop(audioCtx.currentTime + 0.5);
}

function buildPath() {
    const margin = 60;
    const cols = Math.floor((W - margin*2) / gridSize);
    const rows = Math.floor((H - margin*2 - 80) / gridSize);
    path = [];
    // Create a winding path
    let cx = 0, cy = Math.floor(rows/2);
    const visited = new Set();
    while (cx < cols && path.length < 30) {
        const key = cx+','+cy;
        if (!visited.has(key)) {
            path.push({ x: margin + cx*gridSize + gridSize/2, y: margin + cy*gridSize + gridSize/2 });
            visited.add(key);
        }
        // Move right or change direction
        if (cx < cols - 1) cx++;
        else break;
    }
    // Better path: snake pattern
    path = [];
    const startY = Math.floor(rows * 0.3);
    const endY = Math.floor(rows * 0.7);
    for (let i = 0; i < cols; i++) {
        let y;
        if (i < Math.floor(cols*0.2)) y = startY;
        else if (i < Math.floor(cols*0.5)) y = Math.floor(rows/2);
        else if (i < Math.floor(cols*0.8)) y = endY;
        else y = Math.floor(rows/2);
        path.push({ x: margin + i*gridSize + gridSize/2, y: margin + y*gridSize + gridSize/2 });
    }
}

function isOnPath(gx, gy) {
    const px = 60 + gx * gridSize + gridSize/2;
    const py = 60 + gy * gridSize + gridSize/2;
    for (let p of path) {
        if (Math.abs(px - p.x) < gridSize*0.8 && Math.abs(py - p.y) < gridSize*0.8) return true;
    }
    // Check segments
    for (let i = 0; i < path.length-1; i++) {
        const a = path[i], b = path[i+1];
        if (px >= Math.min(a.x,b.x)-gridSize*0.5 && px <= Math.max(a.x,b.x)+gridSize*0.5 &&
            py >= Math.min(a.y,b.y)-gridSize*0.5 && py <= Math.max(a.y,b.y)+gridSize*0.5) return true;
    }
    return false;
}

function getTowerType(type) {
    const types = [
        { name: 'Gun', cost: 50, range: 120, damage: 15, fireRate: 8, color: '#4ecca3', projSpeed: 6 },
        { name: 'Cannon', cost: 100, range: 100, damage: 40, fireRate: 25, splash: 50, color: '#e94560', projSpeed: 4 },
        { name: 'Ice', cost: 75, range: 130, damage: 5, fireRate: 12, slow: 0.4, color: '#7ec8e3', projSpeed: 5 }
    ];
    return types[type];
}

        const tType = getTowerType(selectedTower);
        if (money >= tType.cost) {
            const gx = Math.floor((mx - 60) / gridSize);
            const gy = Math.floor((my - 60) / gridSize);
            const px = 60 + gx * gridSize + gridSize/2;
            const py = 60 + gy * gridSize + gridSize/2;
            
            if (!isOnPath(gx, gy) && my > 80) {
                // Check no tower already there
                let canPlace = true;
                for (let t of towers) if (Math.abs(t.x - px) < gridSize*0.8 && Math.abs(t.y - py) < gridSize*0.8) canPlace = false;
                
                if (canPlace) {
                    money -= tType.cost;
                    towers.push({ x: px, y: py, type: selectedTower, angle: 0, cooldown: 0 });
                    playSound('place');
                    updateHUD();
                }
            }
        }
    }
});

function spawnZombie() {
    if (path.length < 2) return;
    const startX = path[0].x - gridSize*2;
    const startY = path[0].y + (Math.random()-0.5)*gridSize;
    
    let hp = 60 + wave * 15;
    let speed = 0.8 + Math.random() * 0.4 + wave * 0.05;
    let type = 'normal';
    
    if (wave > 3 && Math.random() < 0.2) {
        type = 'fast'; speed *= 1.8; hp *= 0.6;
    } else if (wave > 5 && Math.random() < 0.15) {
        type = 'tank'; hp *= 3; speed *= 0.6;
    }
    
    zombies.push({
        x: startX, y: startY, pathIndex: 0, progress: 0,
        hp, maxHp: hp, speed, type, slowTimer: 0, angle: 0
    });
}

function updateZombies(dt) {
    for (let i = zombies.length - 1; i >= 0; i--) {
        const z = zombies[i];
        let spd = z.speed * (z.slowTimer > 0 ? 0.4 : 1);
        if (z.slowTimer > 0) z.slowTimer -= dt;
        
        if (z.pathIndex >= path.length - 1) {
            // Reached end
            lives--;
            zombies.splice(i, 1);
            playSound('zombie');
            updateHUD();
            if (lives <= 0) gameOver();
            continue;
        }
        
        const target = path[z.pathIndex + 1];
        const dx = target.x - z.x, dy = target.y - z.y;
        const dist = Math.sqrt(dx*dx + dy*dy);
        z.angle = Math.atan2(dy, dx);
        
        if (dist > 5) {
            z.x += (dx/dist) * spd * dt * 60;
            z.y += (dy/dist) * spd * dt * 60;
        } else {
            z.pathIndex++;
        }
    }
}

function updateTowers(dt) {
    for (let t of towers) {
        if (t.cooldown > 0) { t.cooldown -= dt; continue; }
        
        const type = getTowerType(t.type);
        let target = null, bestDist = Infinity;
        
        for (let z of zombies) {
            const dx = z.x - t.x, dy = z.y - t.y;
            const dist = Math.sqrt(dx*dx + dy*dy);
            if (dist < type.range && dist < bestDist) { bestDist = dist; target = z; }
        }
        
        if (target) {
            t.angle = Math.atan2(target.y - t.y, target.x - t.x);
            t.cooldown = type.fireRate / 60;
            
            projectiles.push({
                x: t.x, y: t.y, tx: target.x, ty: target.y,
                target, speed: type.projSpeed, damage: type.damage,
                type: t.type, splash: type.splash || 0, slow: type.slow || 0
            });
            
            if (t.type === 0) playSound('shoot');
        }
    }
}

function updateProjectiles(dt) {
    for (let i = projectiles.length - 1; i >= 0; i--) {
        const p = projectiles[i];
        
        // Track target
        if (p.target && p.target.hp > 0) {
            p.tx = p.target.x; p.ty = p.target.y;
        }
        
        const dx = p.tx - p.x, dy = p.ty - p.y;
        const dist = Math.sqrt(dx*dx + dy*dy);
        
        if (dist < 8) {
            // Hit!
            if (p.type === 1 && p.splash > 0) {
                playSound('explosion');
                for (let z of zombies) {
                    const dx2 = z.x - p.tx, dy2 = z.y - p.ty;
                    if (Math.sqrt(dx2*dx2+dy2*dy2) < p.splash) {
                        z.hp -= p.damage;
                    }
                }
                for (let j = 0; j < 8; j++) {
                    const a = Math.random()*Math.PI*2;
                    particles.push({ x: p.tx, y: p.ty, vx: Math.cos(a)*3, vy: Math.sin(a)*3, life: 0.5, color: '#ff6633' });
                }
            } else {
                if (p.target && p.target.hp > 0) {
                    p.target.hp -= p.damage;
                    if (p.type === 2 && p.slow) p.target.slowTimer = 2;
                    playSound('shoot');
                }
                for (let j = 0; j < 3; j++) {
                    const a = Math.random()*Math.PI*2;
                    particles.push({ x: p.tx, y: p.ty, vx: Math.cos(a)*2, vy: Math.sin(a)*2, life: 0.3, color: p.type===2?'#7ec8e3':'#4ecca3' });
                }
            }
            projectiles.splice(i, 1);
        } else {
            p.x += (dx/dist) * p.speed * dt * 60;
            p.y += (dy/dist) * p.speed * dt * 60;
        }
    }
    
    // Remove dead zombies
    for (let i = zombies.length - 1; i >= 0; i--) {
        if (zombies[i].hp <= 0) {
            money += 10 + wave * 2;
            score += 10 + wave * 5;
            playSound('death');
            for (let j = 0; j < 6; j++) {
                const a = Math.random()*Math.PI*2;
                particles.push({ x: zombies[i].x, y: zombies[i].y, vx: Math.cos(a)*2, vy: Math.sin(a)*2, life: 0.5, color: '#4a7' });
            }
            zombies.splice(i, 1);
            updateHUD();
        }
    }
}

function updateParticles(dt) {
    for (let i = particles.length - 1; i >= 0; i--) {
        const p = particles[i];
        p.x += p.vx * dt * 60;
        p.y += p.vy * dt * 60;
        p.life -= dt;
        if (p.life <= 0) particles.splice(i, 1);
    }
}

function draw() {
    ctx.fillStyle = '#1a2332';
    ctx.fillRect(0, 0, W, H);
    
    // Draw grid
    ctx.strokeStyle = 'rgba(255,255,255,0.03)';
    ctx.lineWidth = 1;
    for (let x = 60; x < W; x += gridSize) { ctx.beginPath(); ctx.moveTo(x,80); ctx.lineTo(x,H-80); ctx.stroke(); }
    for (let y = 80; y < H-80; y += gridSize) { ctx.beginPath(); ctx.moveTo(60,y); ctx.lineTo(W-60,y); ctx.stroke(); }
    
    // Draw path
    if (path.length > 1) {
        ctx.strokeStyle = '#3d5a3a';
        ctx.lineWidth = gridSize * 0.8;
        ctx.lineCap = 'round'; ctx.lineJoin = 'round';
        ctx.beginPath();
        ctx.moveTo(path[0].x, path[0].y);
        for (let i = 1; i < path.length; i++) ctx.lineTo(path[i].x, path[i].y);
        ctx.stroke();
        
        ctx.strokeStyle = '#2a4a28';
        ctx.lineWidth = gridSize * 0.6;
        ctx.beginPath();
        ctx.moveTo(path[0].x, path[0].y);
        for (let i = 1; i < path.length; i++) ctx.lineTo(path[i].x, path[i].y);
        ctx.stroke();
        
        // Path direction arrows
        ctx.fillStyle = 'rgba(255,255,255,0.1)';
        for (let i = 0; i < path.length - 1; i += 3) {
            const a = path[i], b = path[i+1];
            const mx = (a.x+b.x)/2, my = (a.y+b.y)/2;
            const ang = Math.atan2(b.y-a.y, b.x-a.x);
            ctx.save();
            ctx.translate(mx, my); ctx.rotate(ang);
            ctx.beginPath(); ctx.moveTo(5,0); ctx.lineTo(-5,-4); ctx.lineTo(-5,4); ctx.fill();
            ctx.restore();
        }
    }
    
    // Draw base
    if (path.length > 0) {
        const end = path[path.length-1];
        ctx.fillStyle = '#e94560';
        ctx.beginPath(); ctx.arc(end.x, end.y, 20, 0, Math.PI*2); ctx.fill();
        ctx.fillStyle = '#fff';
        ctx.font = '16px sans-serif'; ctx.textAlign = 'center';
        ctx.fillText('🏠', end.x, end.y + 5);
    }
    
    // Draw towers
    for (let t of towers) {
        const type = getTowerType(t.type);
        ctx.fillStyle = type.color;
        ctx.beginPath(); ctx.arc(t.x, t.y, 16, 0, Math.PI*2); ctx.fill();
        ctx.strokeStyle = '#fff'; ctx.lineWidth = 2;
        ctx.beginPath(); ctx.arc(t.x, t.y, 16, 0, Math.PI*2); ctx.stroke();
        
        // Barrel
        ctx.save();
        ctx.translate(t.x, t.y); ctx.rotate(t.angle);
        ctx.fillStyle = '#333';
        ctx.fillRect(0, -3, 20, 6);
        ctx.restore();
        
        // Range indicator when hovering
        if (hoveredPath && Math.abs(hoveredPath.x - t.x) < 20 && Math.abs(hoveredPath.y - t.y) < 20) {
            ctx.strokeStyle = type.color; ctx.globalAlpha = 0.3;
            ctx.beginPath(); ctx.arc(t.x, t.y, type.range, 0, Math.PI*2); ctx.stroke();
            ctx.globalAlpha = 1;
        }
    }
    
    // Draw zombies
    for (let z of zombies) {
        const size = z.type === 'tank' ? 18 : 14;
        
        // Body
        ctx.fillStyle = z.type === 'fast' ? '#ff9944' : z.type === 'tank' ? '#8855cc' : '#66cc44';
        ctx.beginPath(); ctx.arc(z.x, z.y, size, 0, Math.PI*2); ctx.fill();
        
        // Eyes
        ctx.fillStyle = '#ff0000';
        const ex = Math.cos(z.angle) * 5, ey = Math.sin(z.angle) * 5;
        ctx.beginPath(); ctx.arc(z.x + ex - 3, z.y + ey, 2, 0, Math.PI*2); ctx.fill();
        ctx.beginPath(); ctx.arc(z.x + ex + 3, z.y + ey, 2, 0, Math.PI*2); ctx.fill();
        
        // Slow effect
        if (z.slowTimer > 0) {
            ctx.strokeStyle = '#7ec8e3'; ctx.lineWidth = 2;
            ctx.beginPath(); ctx.arc(z.x, z.y, size + 4, 0, Math.PI*2); ctx.stroke();
        }
        
        // Health bar
        const hw = size * 2;
        ctx.fillStyle = '#333';
        ctx.fillRect(z.x - hw/2, z.y - size - 10, hw, 5);
        ctx.fillStyle = z.hp/z.maxHp > 0.5 ? '#4ecca3' : '#e94560';
        ctx.fillRect(z.x - hw/2, z.y - size - 10, hw * (z.hp/z.maxHp), 5);
    }
    
    // Draw projectiles
    for (let p of projectiles) {
        ctx.fillStyle = p.type === 1 ? '#ff6633' : p.type === 2 ? '#7ec8e3' : '#4ecca3';
        ctx.beginPath(); ctx.arc(p.x, p.y, 4, 0, Math.PI*2); ctx.fill();
    }
    
    // Draw particles
    for (let p of particles) {
        ctx.globalAlpha = p.life * 2;
        ctx.fillStyle = p.color;
        ctx.beginPath(); ctx.arc(p.x, p.y, 3, 0, Math.PI*2); ctx.fill();
        ctx.globalAlpha = 1;
    }
    
    // Placement preview
    if (selectedTower >= 0 && hoveredPath && gameState === 'playing') {
        const gx = Math.floor((hoveredPath.x - 60) / gridSize);
        const gy = Math.floor((hoveredPath.y - 60) / gridSize);
        const px = 60 + gx * gridSize + gridSize/2;
        const py = 60 + gy * gridSize + gridSize/2;
        const type = getTowerType(selectedTower);
        
        ctx.globalAlpha = 0.5;
        ctx.fillStyle = type.color;
        ctx.beginPath(); ctx.arc(px, py, 14, 0, Math.PI*2); ctx.fill();
        ctx.strokeStyle = type.color;
        ctx.beginPath(); ctx.arc(px, py, type.range, 0, Math.PI*2); ctx.stroke();
        ctx.globalAlpha = 1;
    }
}

let lastTime = 0;
function gameLoop(time) {
    const dt = Math.min((time - lastTime) / 1000, 0.05);
    lastTime = time;
    
    if (gameState === 'playing') {
        // Spawn zombies
        waveTimer += dt;
        spawnTimer -= dt;
        if (spawnTimer <= 0 && waveZombiesLeft > 0) {
            spawnZombie();
            waveZombiesLeft--;
            spawnTimer = Math.max(0.3, 1.5 - wave * 0.05);
        }
        
        if (waveZombiesLeft === 0 && zombies.length === 0) {
            money += 50 + wave * 10;
            updateHUD();
            setTimeout(nextWave, 2000);
            waveZombiesLeft = -1; // Prevent double trigger
        }
        
        updateZombies(dt);
        updateTowers(dt);
        updateProjectiles(dt);
        updateParticles(dt);
    }
    
    draw();
    requestAnimationFrame(gameLoop);
}

requestAnimationFrame(gameLoop);
