const ctx = canvas.getContext('2d');

// Audio
let audioCtx = null;
function initAudio() {
    if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
}
function playSound(freq, dur, type, vol) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = type || 'square';
    osc.frequency.value = freq;
    gain.gain.setValueAtTime(vol || 0.15, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + dur);
}

// Game Constants
const TILE = 48;
const GRAVITY = 0.5;
const JUMP_VEL = -12;
const MOVE_SPEED = 3;

// Game State
let gameState = 'menu'; // menu, playing, win, lose
let turn = 0;
let animFrame = 0;
let animTimer = 0;
let shakeTimer = 0;
let shakeX = 0, shakeY = 0;
let particles = [];
let damagePopups = [];

// Level Data
const LEVEL_WIDTH = 40; // in tiles
const terrain = [];
function buildTerrain() {
    terrain.length = 0;
    for (let x = 0; x < LEVEL_WIDTH; x++) {
        let h = 5;
        if (x >= 8 && x <= 12) h = 7;
        if (x >= 14 && x <= 18) h = 4;
        if (x >= 20 && x <= 24) h = 8;
        if (x >= 26 && x <= 30) h = 5;
        if (x >= 32 && x <= 36) h = 7;
        if (x >= 37 && x <= 39) h = 4;
        terrain[x] = h;
    }
}

// Player
let player = {};
function resetPlayer() {
    player = {
        x: 2 * TILE, y: 0, vy: 0,
        hp: 50, maxHp: 50, atk: 12, def: 3,
        facing: 1, onGround: false,
        actionUsed: false, attackAnim: 0, hitFlash: 0
    };
    player.y = (terrain[2] - 1) * TILE;
}

// Enemies
let enemies = [];
function buildEnemies() {
    enemies = [
        { type: 'slime', x: 8*TILE, hp: 20, maxHp: 20, atk: 6, def: 0, facing: -1, hitFlash: 0, alive: true },
        { type: 'slime', x: 14*TILE, hp: 20, maxHp: 20, atk: 6, def: 0, facing: -1, hitFlash: 0, alive: true },
        { type: 'skeleton', x: 20*TILE, hp: 35, maxHp: 35, atk: 10, def: 2, facing: -1, hitFlash: 0, alive: true },
        { type: 'skeleton', x: 26*TILE, hp: 35, maxHp: 35, atk: 10, def: 2, facing: -1, hitFlash: 0, alive: true },
        { type: 'dragon', x: 35*TILE, hp: 80, maxHp: 80, atk: 18, def: 4, facing: -1, hitFlash: 0, alive: true }
    ];
}

function getGroundY(tileX) {
    const tx = Math.floor(tileX / TILE);
    const clamped = Math.max(0, Math.min(LEVEL_WIDTH - 1, tx));
    return terrain[clamped] * TILE;
}

// Camera
let cameraX = 0;
function updateCamera() {
    const targetX = player.x - canvas.width / 2 + 100;
    cameraX += (targetX - cameraX) * 0.08;
    cameraX = Math.max(0, Math.min(cameraX, LEVEL_WIDTH * TILE - canvas.width));
}

// Particles
function spawnParticles(x, y, color, count) {
    for (let i = 0; i < count; i++) {
        particles.push({
            x: x + (Math.random()-0.5)*20,
            y: y + (Math.random()-0.5)*20,
            vx: (Math.random()-0.5)*4,
            vy: -Math.random()*3 - 1,
            life: 30 + Math.random()*20,
            color: color
        });
    }
}

function spawnDamagePopup(x, y, text, color) {
    damagePopups.push({ x, y, text, color, life: 45 });
}

// Input
let keys = {};
window.addEventListener('keydown', (e) => {
    keys[e.code] = true;
    if (gameState === 'menu') {
        initAudio();
        startGame();
    } else if (gameState === 'playing' && !player.actionUsed) {
        handleAction(e.code);
    } else if ((gameState === 'win' || gameState === 'lose') && e.code === 'KeyR') {
        startGame();
    }
});
window.addEventListener('keyup', (e) => { keys[e.code] = false; });

function handleAction(code) {
    player.actionUsed = true;
    turn++;
    
    if (code === 'ArrowLeft') {
        movePlayer(-MOVE_SPEED);
        playSound(200, 0.1, 'triangle', 0.1);
    } else if (code === 'ArrowRight') {
        movePlayer(MOVE_SPEED);
        playSound(200, 0.1, 'triangle', 0.1);
    } else if (code === 'ArrowUp') {
        jumpPlayer();
        playSound(400, 0.15, 'sine', 0.15);
    } else if (code === 'Space' || code === 'KeyZ') {
        playerAttack();
    } else if (code === 'KeyS' || code === 'ArrowDown') {
        // Wait / Defend
        player.defending = true;
        playSound(300, 0.1, 'sine', 0.1);
    }
    
    setTimeout(() => {
        player.actionUsed = false;
        enemyTurn();
        checkWinLose();
    }, 400);
}

function movePlayer(dx) {
    player.x += dx * TILE / MOVE_SPEED;
    player.facing = dx > 0 ? 1 : -1;
    player.x = Math.max(0, Math.min(player.x, (LEVEL_WIDTH-1) * TILE));
    // Snap to ground
    const gy = getGroundY(player.x);
    if (player.y < gy) {
        player.y += 2;
    } else {
        player.y = gy;
        player.onGround = true;
    }
}

function jumpPlayer() {
    if (player.onGround) {
        player.vy = JUMP_VEL;
        player.onGround = false;
    }
}

function playerAttack() {
    const range = TILE * 1.8;
    let hit = false;
    for (let e of enemies) {
        if (!e.alive) continue;
        const dx = Math.abs(e.x - player.x);
        if (dx < range && ((player.facing > 0 && e.x > player.x) || (player.facing < 0 && e.x < player.x))) {
            const dmg = Math.max(1, player.atk - e.def + Math.floor(Math.random()*4));
            e.hp -= dmg;
            e.hitFlash = 15;
            hit = true;
            spawnParticles(e.x, e.y - 20, '#ff4444', 8);
            spawnDamagePopup(e.x, e.y - 40, '-' + dmg, '#ff4444');
            playSound(150, 0.2, 'sawtooth', 0.2);
            shakeTimer = 8;
            if (e.hp <= 0) {
                e.alive = false;
                spawnParticles(e.x, e.y - 20, '#ffaa00', 15);
                playSound(100, 0.4, 'sawtooth', 0.3);
            }
        }
    }
    if (!hit) {
        // whiff
        playSound(250, 0.1, 'sine', 0.08);
    }
    player.attackAnim = 15;
}

function enemyTurn() {
    for (let e of enemies) {
        if (!e.alive) continue;
        const dx = player.x - e.x;
        const dist = Math.abs(dx);
        
        // Face player
        e.facing = dx > 0 ? 1 : -1;
        
        // Check ground level
        const eTileX = Math.floor(e.x / TILE);
        const pTileX = Math.floor(player.x / TILE);
        
        if (dist < TILE * 1.5) {
            // Attack player
            let dmg = Math.max(1, e.atk - player.def + Math.floor(Math.random()*3));
            if (player.defending) dmg = Math.floor(dmg / 2);
            player.hp -= dmg;
            player.hitFlash = 15;
            spawnParticles(player.x, player.y - 20, '#ff6644', 6);
            spawnDamagePopup(player.x, player.y - 40, '-' + dmg, '#ff8866');
            playSound(120, 0.2, 'square', 0.2);
            shakeTimer = 6;
        } else if (dist > TILE * 1.5 && dist < TILE * 4) {
            // Move toward player
            const moveDir = dx > 0 ? 1 : -1;
            e.x += moveDir * TILE * 0.7;
            e.x = Math.max(0, Math.min(e.x, (LEVEL_WIDTH-1)*TILE));
        } else if (dist >= TILE * 4) {
            // Far away, slow approach
            const moveDir = dx > 0 ? 1 : -1;
            e.x += moveDir * TILE * 0.3;
        }
        
        // Snap to ground
        const gy = getGroundY(e.x);
        if (e.y < gy) {
            e.y += 2;
        } else {
            e.y = gy;
        }
    }
}

function checkWinLose() {
    if (player.hp <= 0) {
        player.hp = 0;
        gameState = 'lose';
        playSound(80, 0.6, 'sawtooth', 0.3);
    }
    if (!enemies.some(e => e.alive)) {
        gameState = 'win';
        playSound(523, 0.3, 'sine', 0.2);
        setTimeout(() => playSound(659, 0.3, 'sine', 0.2), 150);
        setTimeout(() => playSound(784, 0.5, 'sine', 0.2), 300);
    }
}

function startGame() {
    buildTerrain();
    resetPlayer();
    buildEnemies();
    cameraX = 0;
    turn = 0;
    particles = [];
    damagePopups = [];
    player.defending = false;
    gameState = 'playing';
}

// Rendering
function drawBackground() {
    // Sky gradient
    const grad = ctx.createLinearGradient(0, 0, 0, canvas.height);
    grad.addColorStop(0, '#1a1a3e');
    grad.addColorStop(0.5, '#2d2d5e');
    grad.addColorStop(1, '#4a3b6b');
    ctx.fillStyle = grad;
    ctx.fillRect(0, 0, canvas.width, canvas.height);
    
    // Stars
    ctx.fillStyle = 'rgba(255,255,255,0.6)';
    for (let i = 0; i < 50; i++) {
        const sx = ((i * 137 + 50) % (canvas.width + 200)) - 100 - (cameraX * 0.1 % (canvas.width + 200));
        const sy = (i * 97) % (canvas.height * 0.4);
        const ss = 1 + (i % 3) * 0.5;
        ctx.fillRect(((sx % canvas.width) + canvas.width) % canvas.width, sy, ss, ss);
    }
    
    // Mountains (parallax)
    drawMountains(0.2, '#2a2a4a', 0.3);
    drawMountains(0.4, '#3a3a5a', 0.6);
}

function drawMountains(speed, color, heightFactor) {
    ctx.fillStyle = color;
    const offset = -cameraX * speed;
    for (let i = 0; i < 12; i++) {
        const mx = ((i * 300 + 50) % (LEVEL_WIDTH * TILE)) - 200 + offset;
        const mh = 80 + (i * 47 % 120);
        const mw = 150 + (i * 31 % 100);
        ctx.beginPath();
        ctx.moveTo(mx, canvas.height);
        ctx.lineTo(mx + mw/2, canvas.height - mh * heightFactor);
        ctx.lineTo(mx + mw, canvas.height);
        ctx.closePath();
        ctx.fill();
    }
}

function drawTerrain() {
    const startX = Math.max(0, Math.floor(cameraX / TILE) - 1);
    const endX = Math.min(LEVEL_WIDTH, Math.ceil((cameraX + canvas.width) / TILE) + 1);
    
    for (let x = startX; x < endX; x++) {
        const h = terrain[x];
        const screenX = x * TILE - cameraX;
        const groundY = h * TILE;
        
        // Ground block
        ctx.fillStyle = '#4a6741';
        ctx.fillRect(screenX, canvas.height - 0, TILE, (canvas.height - groundY));
        
        // Top layer
        ctx.fillStyle = '#5d8a3c';
        ctx.fillRect(screenX, canvas.height - 0, TILE, 8);
        
        // Texture dots
        ctx.fillStyle = '#3d6b32';
        for (let ty = 1; ty < (canvas.height - groundY) / TILE; ty++) {
            for (let tx = 0; tx < 2; tx++) {
                if ((x * 7 + ty * 13 + tx * 3) % 5 === 0) {
                    ctx.fillRect(screenX + 8 + tx*20, groundY - canvas.height + ty*TILE + 12, 4, 4);
                }
            }
        }
        
        // Side edges
        if (x > 0 && terrain[x] !== terrain[x-1]) {
            const diff = terrain[x] - terrain[x-1];
            if (diff > 0) {
                ctx.fillStyle = '#3a5230';
                ctx.fillRect(screenX, canvas.height - diff*TILE, 4, diff * TILE);
            } else if (diff < 0) {
                ctx.fillStyle = '#3a5230';
                ctx.fillRect(screenX + TILE - 4, canvas.height, 4, -diff * TILE);
            }
        }
    }
}

function drawPlayer() {
    const px = player.x - cameraX;
    const py = player.y - canvas.height + 20;
    
    // Hit flash
    if (player.hitFlash > 0) {
        ctx.globalAlpha = player.hitFlash % 4 < 2 ? 0.5 : 1;
    }
    
    // Body
    const bodyColor = player.defending ? '#4488ff' : '#e8d44d';
    ctx.fillStyle = bodyColor;
    ctx.fillRect(px - 12, py - 36, 24, 36);
    
    // Head
    ctx.fillStyle = '#ffd700';
    ctx.fillRect(px - 10, py - 50, 20, 16);
    
    // Eyes
    ctx.fillStyle = '#222';
    const eyeOff = player.facing * 4;
    ctx.fillRect(px + eyeOff - 3, py - 46, 3, 4);
    
    // Sword (attack anim)
    if (player.attackAnim > 0) {
        const swing = Math.sin((15 - player.attackAnim) / 15 * Math.PI);
        ctx.fillStyle = '#ccc';
        ctx.save();
        ctx.translate(px + player.facing * 14, py - 28);
        ctx.rotate(player.facing * (-0.5 + swing * 1.2));
        ctx.fillRect(0, -2, 30, 4);
        ctx.restore();
    } else {
        // Idle sword
        ctx.fillStyle = '#aaa';
        ctx.save();
        ctx.translate(px + player.facing * 14, py - 24);
        ctx.rotate(player.facing * 0.3);
        ctx.fillRect(0, -2, 22, 3);
        ctx.restore();
    }
    
    // Shield when defending
    if (player.defending) {
        ctx.fillStyle = 'rgba(100,150,255,0.4)';
        ctx.beginPath();
        ctx.arc(px, py - 28, 22, 0, Math.PI * 2);
        ctx.fill();
    }
    
    ctx.globalAlpha = 1;
}

function drawEnemy(e) {
    if (!e.alive) return;
    const ex = e.x - cameraX;
    const ey = e.y - canvas.height + 20;
    
    if (e.hitFlash > 0) {
        ctx.globalAlpha = e.hitFlash % 4 < 2 ? 0.5 : 1;
    }
    
    if (e.type === 'slime') {
        // Blob body
        const squish = Math.sin(animFrame * 0.1) * 3;
        ctx.fillStyle = '#5cb85c';
        ctx.beginPath();
        ctx.ellipse(ex, ey - 14 + squish/2, 16, 14 - squish/2, 0, 0, Math.PI*2);
        ctx.fill();
        // Eyes
        ctx.fillStyle = '#fff';
        ctx.fillRect(ex - 8, ey - 22, 5, 5);
        ctx.fillRect(ex + 3, ey - 22, 5, 5);
        ctx.fillStyle = '#000';
        ctx.fillRect(ex - 6, ey - 20, 3, 3);
        ctx.fillRect(ex + 5, ey - 20, 3, 3);
    } else if (e.type === 'skeleton') {
        // Body
        ctx.fillStyle = '#ddd';
        ctx.fillRect(ex - 10, ey - 34, 20, 34);
        // Skull
        ctx.fillStyle = '#eee';
        ctx.fillRect(ex - 9, ey - 48, 18, 15);
        // Eyes
        ctx.fillStyle = '#f00';
        ctx.fillRect(ex - 6, ey - 44, 4, 4);
        ctx.fillRect(ex + 2, ey - 44, 4, 4);
        // Sword
        ctx.fillStyle = '#999';
        ctx.save();
        ctx.translate(ex + e.facing * 12, ey - 26);
        ctx.rotate(e.facing * 0.4);
        ctx.fillRect(0, -1, 24, 3);
        ctx.restore();
    } else if (e.type === 'dragon') {
        // Big body
        const breathe = Math.sin(animFrame * 0.05) * 4;
        ctx.fillStyle = '#8b2500';
        ctx.beginPath();
        ctx.ellipse(ex, ey - 30 + breathe/2, 28, 24 - breathe/2, 0, 0, Math.PI*2);
        ctx.fill();
        // Head
        ctx.fillStyle = '#a52a1a';
        ctx.fillRect(ex - 16, ey - 56, 32, 20);
        // Eyes
        ctx.fillStyle = '#ff0';
        ctx.fillRect(ex - 10, ey - 52, 5, 5);
        ctx.fillRect(ex + 5, ey - 52, 5, 5);
        // Wings
        ctx.fillStyle = 'rgba(139,37,0,0.6)';
        ctx.beginPath();
        ctx.moveTo(ex - 20, ey - 45);
        ctx.lineTo(ex - 45, ey - 70 + breathe);
        ctx.lineTo(ex - 25, ey - 35);
        ctx.fill();
        ctx.beginPath();
        ctx.moveTo(ex + 20, ey - 45);
        ctx.lineTo(ex + 45, ey - 70 + breathe);
        ctx.lineTo(ex + 25, ey - 35);
        ctx.fill();
        // Claws
        ctx.fillStyle = '#6b1a00';
        ctx.fillRect(ex - 8, ey - 6, 6, 8);
        ctx.fillRect(ex + 4, ey - 6, 6, 8);
    }
    
    // HP bar
    const hpW = 30;
    const hpRatio = e.hp / e.maxHp;
    ctx.fillStyle = '#333';
    ctx.fillRect(ex - hpW/2, ey - (e.type === 'dragon' ? 65 : 55), hpW, 5);
    ctx.fillStyle = hpRatio > 0.5 ? '#4f4' : hpRatio > 0.25 ? '#ff4' : '#f44';
    ctx.fillRect(ex - hpW/2, ey - (e.type === 'dragon' ? 65 : 55), hpW * hpRatio, 5);
    
    ctx.globalAlpha = 1;
}

function drawParticles() {
    for (let p of particles) {
        ctx.fillStyle = p.color;
        ctx.globalAlpha = p.life / 40;
        ctx.fillRect(p.x - cameraX - 3, p.y - canvas.height + 20 - 3, 6, 6);
    }
    ctx.globalAlpha = 1;
}

function drawDamagePopups() {
    for (let d of damagePopups) {
        ctx.fillStyle = d.color;
        ctx.font = 'bold 18px monospace';
        ctx.textAlign = 'center';
        const alpha = d.life / 45;
        ctx.globalAlpha = alpha;
        ctx.fillText(d.text, d.x - cameraX, d.y - canvas.height + 20 - (45 - d.life));
    }
    ctx.globalAlpha = 1;
}

function drawEndScreen(won) {
    ctx.fillStyle = 'rgba(0,0,0,0.7)';
    ctx.fillRect(0, 0, canvas.width, canvas.height);
    
    ctx.textAlign = 'center';
    if (won) {
        ctx.fillStyle = '#ffd700';
        ctx.font = 'bold 42px monospace';
        ctx.fillText('VICTORY!', canvas.width/2, canvas.height/2 - 30);
        ctx.fillStyle = '#fff';
        ctx.font = '18px monospace';
        ctx.fillText(`Cleared in ${turn} turns`, canvas.width/2, canvas.height/2 + 20);
    } else {
        ctx.fillStyle = '#ff4444';
        ctx.font = 'bold 42px monospace';
        ctx.fillText('DEFEATED', canvas.width/2, canvas.height/2 - 30);
        ctx.fillStyle = '#fff';
        ctx.font = '18px monospace';
        ctx.fillText(`Fell on turn ${turn}`, canvas.width/2, canvas.height/2 + 20);
    }
    
    ctx.fillStyle = '#aaa';
    ctx.font = '16px monospace';
    const pulse = Math.sin(Date.now() * 0.003) * 0.3 + 0.7;
    ctx.globalAlpha = pulse;
    ctx.fillText('Press R to Restart', canvas.width/2, canvas.height/2 + 70);
    ctx.globalAlpha = 1;
}

// Main Loop
function update() {
    animFrame++;
    
    // Gravity for player
    if (!player.onGround) {
        player.vy += GRAVITY;
        player.y += player.vy;
        const gy = getGroundY(player.x);
        if (player.y >= gy) {
            player.y = gy;
            player.vy = 0;
            player.onGround = true;
        }
    }
    
    // Update particles
    for (let i = particles.length - 1; i >= 0; i--) {
        const p = particles[i];
        p.x += p.vx;
        p.y += p.vy;
        p.vy += 0.2;
        p.life--;
        if (p.life <= 0) particles.splice(i, 1);
    }
    
    // Update damage popups
    for (let i = damagePopups.length - 1; i >= 0; i--) {
        damagePopups[i].life--;
        if (damagePopups[i].life <= 0) damagePopups.splice(i, 1);
    }
    
    // Timers
    if (player.attackAnim > 0) player.attackAnim--;
    if (player.hitFlash > 0) player.hitFlash--;
    for (let e of enemies) if (e.hitFlash > 0) e.hitFlash--;
    
    // Shake
    if (shakeTimer > 0) {
        shakeTimer--;
        shakeX = (Math.random()-0.5) * 4;
        shakeY = (Math.random()-0.5) * 4;
    } else {
        shakeX = 0;
        shakeY = 0;
    }
    
    updateCamera();
}

function render() {
    ctx.save();
    ctx.translate(shakeX, shakeY);
    
    if (gameState === 'menu') {
        drawMenu();
    } else {
        drawBackground();
        drawTerrain();
        
        // Draw enemies
        for (let e of enemies) drawEnemy(e);
        
        drawPlayer();
        drawParticles();
        drawDamagePopups();
        drawHUD();
        
        if (gameState === 'win') drawEndScreen(true);
        if (gameState === 'lose') drawEndScreen(false);
    }
    
    ctx.restore();
}

function gameLoop() {
    update();
    render();
    requestAnimationFrame(gameLoop);
}

// Init
buildTerrain();
resetPlayer();
buildEnemies();
gameLoop();
</script>
