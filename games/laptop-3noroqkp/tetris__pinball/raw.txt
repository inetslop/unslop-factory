const canvas = document.getElementById('c');
const ctx = canvas.getContext('2d');

let W, H;
function resize() {
    W = canvas.width = window.innerWidth;
    H = canvas.height = window.innerHeight;
}
window.addEventListener('resize', resize);
resize();

// Game constants
const BALL_RADIUS = 10;
const GRAVITY = 0.35;
const BOUNCE_DAMPING = 0.82;
const FLIPPER_FORCE = 14;
const GRID_COLS = 8;
const GRID_CELL = 36;
const MAX_LIVES = 3;

let gameState = 'START'; // START, PLAYING, GAME_OVER
let score = 0;
let lives = MAX_LIVES;
let combo = 0;
let balls = [];
let grid = [];
let flippers = {};
let bumpers = [];
let particles = [];
let messages = [];
let audioCtx = null;

const COLORS = ['#ff3366','#33ff66','#3366ff','#ffcc33','#ff6633','#cc33ff','#33cccc','#ff33cc'];

function initGrid() {
    grid = Array.from({length: 20}, () => Array(GRID_COLS).fill(null));
}

function initBumpers() {
    const scale = Math.min(W, H) / 800;
    bumpers = [
        { x: W * 0.35, y: H * 0.45, r: 22 * scale },
        { x: W * 0.65, y: H * 0.45, r: 22 * scale },
        { x: W * 0.5, y: H * 0.3, r: 18 * scale },
    ];
}

function initFlippers() {
    const fw = Math.min(W * 0.28, 140);
    const fy = H * 0.82;
    flippers.left = { pivotX: W * 0.5 - fw, pivotY: fy, angle: 0.4, restAngle: 0.4, activeAngle: -0.3, active: false };
    flippers.right = { pivotX: W * 0.5 + fw, pivotY: fy, angle: -0.4, restAngle: -0.4, activeAngle: 0.3, active: false };
}

function spawnBall() {
    const x = W * 0.5;
    const y = H * 0.12;
    balls.push({ x, y, vx: (Math.random() - 0.5) * 4, vy: 0, r: BALL_RADIUS, trail: [] });
}

function resetGame() {
    score = 0;
    lives = MAX_LIVES;
    combo = 0;
    balls = [];
    particles = [];
    messages = [];
    initGrid();
    initBumpers();
    initFlippers();
    spawnBall();
}

// Audio
function playSound(freq, dur, type) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = type || 'square';
    osc.frequency.value = freq;
    gain.gain.setValueAtTime(0.15, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + dur);
}

function ensureAudio() {
    if (!audioCtx) {
        try { audioCtx = new (window.AudioContext || window.webkitAudioContext)(); } catch(e) {}
    }
}

// Input
const keys = {};
window.addEventListener('keydown', e => {
    keys[e.code] = true;
    ensureAudio();
    if (gameState === 'START' && (e.code === 'Space' || e.code === 'Enter')) {
        resetGame();
        gameState = 'PLAYING';
    } else if (gameState === 'GAME_OVER' && (e.code === 'Space' || e.code === 'Enter')) {
        resetGame();
        gameState = 'PLAYING';
    }
});
window.addEventListener('keyup', e => { keys[e.code] = false; });

// Particles
function spawnParticles(x, y, color, count) {
    for (let i = 0; i < count; i++) {
        const a = Math.random() * Math.PI * 2;
        const s = 1 + Math.random() * 4;
        particles.push({ x, y, vx: Math.cos(a)*s, vy: Math.sin(a)*s, life: 30+Math.random()*30, color });
    }
}

function addMessage(text, x, y) {
    messages.push({ text, x, y, life: 60 });
}

// Physics update
function updatePhysics() {
    // Flippers
    if (keys['ArrowLeft'] || keys['KeyA']) flippers.left.active = true;
    else flippers.left.active = false;
    if (keys['ArrowRight'] || keys['KeyD']) flippers.right.active = true;
    else flippers.right.active = false;

    // Smooth flipper animation
    const lTarget = flippers.left.active ? flippers.left.activeAngle : flippers.left.restAngle;
    flippers.left.angle += (lTarget - flippers.left.angle) * 0.4;
    const rTarget = flippers.right.active ? flippers.right.activeAngle : flippers.right.restAngle;
    flippers.right.angle += (rTarget - flippers.right.angle) * 0.4;

    for (let bi = balls.length - 1; bi >= 0; bi--) {
        const b = balls[bi];
        b.vy += GRAVITY;
        b.x += b.vx;
        b.y += b.vy;

        // Trail
        b.trail.push({x: b.x, y: b.y});
        if (b.trail.length > 12) b.trail.shift();

        // Wall collisions
        const playLeft = W * 0.08;
        const playRight = W * 0.92;
        const topY = H * 0.15;
        const bottomY = H * 0.95;

        if (b.x - b.r < playLeft) { b.x = playLeft + b.r; b.vx = Math.abs(b.vx) * BOUNCE_DAMPING; }
        if (b.x + b.r > playRight) { b.x = playRight - b.r; b.vx = -Math.abs(b.vx) * BOUNCE_DAMPING; }
        if (b.y - b.r < topY) { b.y = topY + b.r; b.vy = Math.abs(b.vy) * BOUNCE_DAMPING; }

        // Bumper collisions
        for (const bp of bumpers) {
            const dx = b.x - bp.x;
            const dy = b.y - bp.y;
            const dist = Math.sqrt(dx*dx + dy*dy);
            if (dist < b.r + bp.r) {
                const nx = dx / dist, ny = dy / dist;
                b.x = bp.x + nx * (b.r + bp.r);
                b.y = bp.y + ny * (b.r + bp.r);
                const dot = b.vx * nx + b.vy * ny;
                b.vx -= 2 * dot * nx;
                b.vy -= 2 * dot * ny;
                b.vx *= 1.05;
                b.vy *= 1.05;
                score += 50;
                playSound(440 + Math.random() * 200, 0.1);
                spawnParticles(b.x, b.y, '#ffcc33', 6);
            }
        }

        // Flipper collisions
        for (const key of ['left', 'right']) {
            const f = flippers[key];
            const fw = Math.min(W * 0.28, 140);
            const tipX = f.pivotX + Math.cos(f.angle) * fw;
            const tipY = f.pivotY + Math.sin(f.angle) * fw;

            // Ball vs line segment
            const t = Math.max(0, Math.min(1, ((b.x - f.pivotX)*(tipX - f.pivotX) + (b.y - f.pivotY)*(tipY - f.pivotY)) / (fw*fw)));
            const cx = f.pivotX + t * (tipX - f.pivotX);
            const cy = f.pivotY + t * (tipY - f.pivotY);
            const dx = b.x - cx, dy = b.y - cy;
            const dist = Math.sqrt(dx*dx + dy*dy);
            if (dist < b.r + 8) {
                const nx = dx / (dist || 1), ny = dy / (dist || 1);
                b.x = cx + nx * (b.r + 8);
                b.y = cy + ny * (b.r + 8);
                if (f.active) {
                    b.vy = -FLIPPER_FORCE;
                    b.vx += (key === 'left' ? -1 : 1) * 3;
                    score += 10;
                    playSound(200, 0.05);
                } else {
                    const dot = b.vx * nx + b.vy * ny;
                    if (dot < 0) {
                        b.vx -= 2 * dot * nx;
                        b.vy -= 2 * dot * ny;
                    }
                }
            }
        }

        // Top zone: ball enters grid
        const gridTop = H * 0.15;
        const gridLeft = W/2 - (GRID_COLS * GRID_CELL) / 2;
        if (b.y < gridTop + 30 && b.vy > 0) {
            // Place in grid
            let col = Math.floor((b.x - gridLeft) / GRID_CELL);
            col = Math.max(0, Math.min(GRID_COLS - 1, col));
            placeInGrid(col);
            balls.splice(bi, 1);
            spawnBall();
            continue;
        }

        // Ball falls off bottom
        if (b.y > H + 50) {
            balls.splice(bi, 1);
            lives--;
            combo = 0;
            playSound(100, 0.3, 'sawtooth');
            if (lives <= 0) {
                gameState = 'GAME_OVER';
            } else {
                spawnBall();
            }
        }
    }

    // Update particles
    for (let i = particles.length - 1; i >= 0; i--) {
        const p = particles[i];
        p.x += p.vx;
        p.y += p.vy;
        p.vy += 0.1;
        p.life--;
        if (p.life <= 0) particles.splice(i, 1);
    }

    // Update messages
    for (let i = messages.length - 1; i >= 0; i--) {
        messages[i].y -= 1;
        messages[i].life--;
        if (messages[i].life <= 0) messages.splice(i, 1);
    }
}

function placeInGrid(col) {
    // Find lowest empty row in this column
    for (let r = 19; r >= 0; r--) {
        if (!grid[r][col]) {
            grid[r][col] = COLORS[Math.floor(Math.random() * COLORS.length)];
            spawnParticles(W/2 + col * GRID_CELL - (GRID_COLS*GRID_CELL)/2 + GRID_CELL/2, H * 0.15 + r * GRID_CELL, '#ffffff', 8);
            playSound(600, 0.1);
            break;
        }
    }

    // Check for line clears
    let cleared = 0;
    for (let r = 0; r < 20; r++) {
        if (grid[r].every(c => c !== null)) {
            cleared++;
            grid[r] = Array(GRID_COLS).fill(null);
        }
    }

    if (cleared > 0) {
        combo++;
        const pts = cleared * 500 * combo;
        score += pts;
        addMessage(cleared + ' LINE' + (cleared>1?'S':'') + ' CLEAR! +' + pts, W/2, H*0.4);
        playSound(800, 0.3);
        playSound(1000, 0.3);
        spawnParticles(W/2, H*0.3, '#ffcc33', 20);
    } else {
        combo = 0;
    }

    // Check overflow
    for (let c = 0; c < GRID_COLS; c++) {
        if (grid[0][c] !== null) {
            gameState = 'GAME_OVER';
            return;
        }
    }
}

// Rendering
function draw() {
    ctx.fillStyle = '#0a0a1a';
    ctx.fillRect(0, 0, W, H);

    if (gameState === 'START') {
        drawStartScreen();
        return;
    }

    // Playfield walls
    const playLeft = W * 0.08;
    const playRight = W * 0.92;
    const topY = H * 0.15;

    ctx.strokeStyle = '#334';
    ctx.lineWidth = 3;
    ctx.strokeRect(playLeft, topY, playRight - playLeft, H * 0.8);

    // Grid
    const gridLeft = W/2 - (GRID_COLS * GRID_CELL) / 2;
    for (let r = 0; r < 12; r++) {
        for (let c = 0; c < GRID_COLS; c++) {
            const x = gridLeft + c * GRID_CELL;
            const y = topY + r * GRID_CELL;
            if (grid[r] && grid[r][c]) {
                ctx.fillStyle = grid[r][c];
                ctx.fillRect(x + 1, y + 1, GRID_CELL - 2, GRID_CELL - 2);
                ctx.strokeStyle = 'rgba(255,255,255,0.3)';
                ctx.strokeRect(x + 1, y + 1, GRID_CELL - 2, GRID_CELL - 2);
            } else {
                ctx.strokeStyle = 'rgba(255,255,255,0.05)';
                ctx.strokeRect(x, y, GRID_CELL, GRID_CELL);
            }
        }
    }

    // Bumpers
    for (const bp of bumpers) {
        ctx.beginPath();
        ctx.arc(bp.x, bp.y, bp.r, 0, Math.PI * 2);
        const grad = ctx.createRadialGradient(bp.x, bp.y, 0, bp.x, bp.y, bp.r);
        grad.addColorStop(0, '#ffcc33');
        grad.addColorStop(1, '#ff6600');
        ctx.fillStyle = grad;
        ctx.fill();
        ctx.strokeStyle = '#fff';
        ctx.lineWidth = 2;
        ctx.stroke();
    }

    // Flippers
    const fw = Math.min(W * 0.28, 140);
    for (const key of ['left', 'right']) {
        const f = flippers[key];
        const tipX = f.pivotX + Math.cos(f.angle) * fw;
        const tipY = f.pivotY + Math.sin(f.angle) * fw;
        ctx.beginPath();
        ctx.moveTo(f.pivotX, f.pivotY);
        ctx.lineTo(tipX, tipY);
        ctx.strokeStyle = key === 'left' ? '#ff3366' : '#33ff66';
        ctx.lineWidth = 10;
        ctx.lineCap = 'round';
        ctx.stroke();
        // Pivot glow
        ctx.beginPath();
        ctx.arc(f.pivotX, f.pivotY, 8, 0, Math.PI*2);
        ctx.fillStyle = '#fff';
        ctx.fill();
    }

    // Balls with trail
    for (const b of balls) {
        for (let i = 0; i < b.trail.length; i++) {
            const t = b.trail[i];
            const alpha = (i / b.trail.length) * 0.5;
            ctx.beginPath();
            ctx.arc(t.x, t.y, b.r * (i / b.trail.length), 0, Math.PI*2);
            ctx.fillStyle = `rgba(0, 200, 255, ${alpha})`;
            ctx.fill();
        }
        ctx.beginPath();
        ctx.arc(b.x, b.y, b.r, 0, Math.PI*2);
        const bg = ctx.createRadialGradient(b.x-3, b.y-3, 0, b.x, b.y, b.r);
        bg.addColorStop(0, '#ffffff');
        bg.addColorStop(1, '#00ccff');
        ctx.fillStyle = bg;
        ctx.fill();
    }

    // Particles
    for (const p of particles) {
        ctx.globalAlpha = p.life / 60;
        ctx.fillStyle = p.color;
        ctx.fillRect(p.x - 2, p.y - 2, 4, 4);
    }
    ctx.globalAlpha = 1;

    // Messages
    for (const m of messages) {
        ctx.globalAlpha = m.life / 60;
        ctx.fillStyle = '#ffcc33';
        ctx.font = 'bold 24px monospace';
        ctx.textAlign = 'center';
        ctx.fillText(m.text, m.x, m.y);
    }
    ctx.globalAlpha = 1;

    // HUD
    ctx.fillStyle = '#fff';
    ctx.font = 'bold 18px monospace';
    ctx.textAlign = 'left';
    ctx.fillText('SCORE: ' + score, 20, 30);
    ctx.fillText('LIVES: ' + '❤'.repeat(lives), 20, 55);
    if (combo > 1) {
        ctx.fillStyle = '#ffcc33';
        ctx.fillText('COMBO x' + combo, 20, 80);
    }

    // Game over overlay
    if (gameState === 'GAME_OVER') {
        ctx.fillStyle = 'rgba(0,0,0,0.7)';
        ctx.fillRect(0, 0, W, H);
        ctx.fillStyle = '#ff3366';
        ctx.font = 'bold 48px monospace';
        ctx.textAlign = 'center';
        ctx.fillText('GAME OVER', W/2, H/2 - 40);
        ctx.fillStyle = '#fff';
        ctx.font = '24px monospace';
        ctx.fillText('Score: ' + score, W/2, H/2 + 10);
        ctx.font = '18px monospace';
        ctx.fillText('Press SPACE to restart', W/2, H/2 + 60);
    }
}

    ctx.fillRect(0, 0, W, H);

    // Title
    ctx.fillStyle = '#00ccff';
    ctx.font = 'bold 52px monospace';

    ctx.fillText('PINBALL', W/2, H*0.25);
    ctx.fillStyle = '#ffcc33';
    ctx.fillText('TETRIS', W/2, H*0.25 + 60);

    // Description
    ctx.fillStyle = '#aaa';
    ctx.font = '18px monospace';
    const lines = [
        'Keep the ball bouncing with pinball flippers!',
        'When the ball enters the top grid zone, it locks in.',
        'Fill complete rows to clear them (Tetris style).',
        'If the grid overflows or you lose all lives, game over!',
        '',
        'CONTROLS:',
        '← / A  = Left Flipper',
        '→ / D  = Right Flipper',
        '',
        'Score: Bumpers=50  Flippers=10  Lines=500x combo',
    ];
    lines.forEach((l, i) => {
        ctx.fillText(l, W/2, H*0.4 + i * 28);
    });

    // Blinking prompt
    if (Math.floor(Date.now() / 500) % 2 === 0) {
        ctx.fillStyle = '#fff';
        ctx.font = 'bold 22px monospace';
        ctx.fillText('PRESS SPACE TO START', W/2, H * 0.88);
    }

    // Decorative balls
    const t = Date.now() / 1000;
    for (let i = 0; i < 5; i++) {
        const bx = W/2 + Math.sin(t + i * 1.3) * W * 0.3;
        const by = H * 0.6 + Math.cos(t * 0.7 + i) * 30;
        ctx.beginPath();
        ctx.arc(bx, by, 5, 0, Math.PI*2);
        ctx.fillStyle = COLORS[i % COLORS.length];
        ctx.fill();
    }
}

// Main loop
let lastTime = 0;
function gameLoop(timestamp) {
    const dt = timestamp - lastTime;
    lastTime = timestamp;

    if (gameState === 'PLAYING') {
        updatePhysics();
    }
    draw();
    requestAnimationFrame(gameLoop);
}
