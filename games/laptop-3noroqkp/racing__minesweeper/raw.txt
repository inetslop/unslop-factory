<body>
    <div id="game-container">
        <canvas id="canvas"></canvas>
        <div id="hud" class="hidden">
            <div class="hud-item"><span class="hud-label">SCORE</span><span class="hud-value" id="score-display">0</span></div>
            <div class="hud-item"><span class="hud-label">TIME</span><div id="timer-bar"><div id="timer-fill"></div></div></div>
            <div class="hud-item"><span class="hud-label">BEST</span><span class="hud-value" id="best-display">0</span></div>
        </div>
    </div>
    <script>
    (function() {
        const canvas = document.getElementById('canvas');
        const ctx = canvas.getContext('2d');
        const hud = document.getElementById('hud');
        const scoreDisplay = document.getElementById('score-display');
        const bestDisplay = document.getElementById('best-display');
        const timerFill = document.getElementById('timer-fill');

        let W, H;
        function resize() {
            W = window.innerWidth;
            H = window.innerHeight;
            canvas.width = W;
            canvas.height = H;
        }
        window.addEventListener('resize', resize);
        resize();

        // Game state
        const GRID_W = 16;
        const GRID_H = 12;
        const MINE_COUNT = 35;
        const TIME_LIMIT = 45;

        let game = {
            state: 'start', // start, playing, over
            grid: [],
            cursorX: 0,
            cursorY: 0,
            mines: 0,
            score: 0,
            timeLeft: TIME_LIMIT,
            lastTime: 0,
            particles: [],
            shake: 0,
            bestScore: 0
        };

        // Load best score
        try {
            const saved = localStorage.getItem('racingMinesweeperBest');
            if (saved) game.bestScore = parseInt(saved);
        } catch(e) {}
        bestDisplay.textContent = game.bestScore;

        // Audio
        let audioCtx = null;
        function initAudio() {
            if (!audioCtx) {
                audioCtx = new (window.AudioContext || window.webkitAudioContext)();
            }
        }
        function playSound(type) {
            if (!audioCtx) return;
            const osc = audioCtx.createOscillator();
            const gain = audioCtx.createGain();
            osc.connect(gain);
            gain.connect(audioCtx.destination);
            
            if (type === 'reveal') {
                osc.type = 'sine';
                osc.frequency.setValueAtTime(400, audioCtx.currentTime);
                osc.frequency.linearRampToValueAtTime(600, audioCtx.currentTime + 0.1);
                gain.gain.setValueAtTime(0.3, audioCtx.currentTime);
                gain.gain.linearRampToValueAtTime(0, audioCtx.currentTime + 0.15);
                osc.start(audioCtx.currentTime);
                osc.stop(audioCtx.currentTime + 0.15);
            } else if (type === 'mine') {
                osc.type = 'sawtooth';
                osc.frequency.setValueAtTime(200, audioCtx.currentTime);
                osc.frequency.linearRampToValueAtTime(50, audioCtx.currentTime + 0.5);
                gain.gain.setValueAtTime(0.5, audioCtx.currentTime);
                gain.gain.linearRampToValueAtTime(0, audioCtx.currentTime + 0.5);
                osc.start(audioCtx.currentTime);
                osc.stop(audioCtx.currentTime + 0.5);
            } else if (type === 'move') {
                osc.type = 'square';
                osc.frequency.setValueAtTime(200, audioCtx.currentTime);
                gain.gain.setValueAtTime(0.1, audioCtx.currentTime);
                gain.gain.linearRampToValueAtTime(0, audioCtx.currentTime + 0.05);
                osc.start(audioCtx.currentTime);
                osc.stop(audioCtx.currentTime + 0.05);
            } else if (type === 'win') {
                osc.type = 'sine';
                osc.frequency.setValueAtTime(400, audioCtx.currentTime);
                osc.frequency.linearRampToValueAtTime(800, audioCtx.currentTime + 0.3);
                gain.gain.setValueAtTime(0.3, audioCtx.currentTime);
                gain.gain.linearRampToValueAtTime(0, audioCtx.currentTime + 0.4);
                osc.start(audioCtx.currentTime);
                osc.stop(audioCtx.currentTime + 0.4);
            }
        }

        function getCellSize() {
            const maxW = W * 0.8;
            const maxH = (H - 100) * 0.75;
            return Math.floor(Math.min(maxW / GRID_W, maxH / GRID_H));
        }

        function getGridOrigin() {
            const cs = getCellSize();
            return {
                x: (W - cs * GRID_W) / 2,
                y: (H - cs * GRID_H) / 2 + 20
            };
        }

        function initGame() {
            game.grid = [];
            for (let y = 0; y < GRID_H; y++) {
                game.grid[y] = [];
                for (let x = 0; x < GRID_W; x++) {
                    game.grid[y][x] = { mine: false, revealed: false, neighbors: 0 };
                }
            }
            // Place mines (not under cursor start)
            let placed = 0;
            while (placed < MINE_COUNT) {
                const x = Math.floor(Math.random() * GRID_W);
                const y = Math.floor(Math.random() * GRID_H);
                if (!game.grid[y][x].mine && !(x === game.cursorX && y === game.cursorY)) {
                    game.grid[y][x].mine = true;
                    placed++;
                }
            }
            // Count neighbors
            for (let y = 0; y < GRID_H; y++) {
                for (let x = 0; x < GRID_W; x++) {
                    if (!game.grid[y][x].mine) {
                        let n = 0;
                        for (let dy = -1; dy <= 1; dy++) {
                            for (let dx = -1; dx <= 1; dx++) {
                                if (dy === 0 && dx === 0) continue;
                                const ny = y + dy, nx = x + dx;
                                if (ny >= 0 && ny < GRID_H && nx >= 0 && nx < GRID_W && game.grid[ny][nx].mine) n++;
                            }
                        }
                        game.grid[y][x].neighbors = n;
                    }
                }
            }
            game.score = 0;
            game.timeLeft = TIME_LIMIT;
            game.particles = [];
            game.shake = 0;
            scoreDisplay.textContent = '0';
        }

        function startGame() {
            initAudio();
            initGame();
            game.state = 'playing';
            hud.classList.remove('hidden');
            game.lastTime = performance.now();
        }

        function endGame(won) {
            game.state = 'over';
            if (won) playSound('win');
            else { playSound('mine'); game.shake = 15; }
            
            if (game.score > game.bestScore) {
                game.bestScore = game.score;
                try { localStorage.setItem('racingMinesweeperBest', String(game.bestScore)); } catch(e) {}
            }
            bestDisplay.textContent = game.bestScore;
        }

        function revealCell(x, y) {
            const cell = game.grid[y][x];
            if (cell.revealed) return;
            
            cell.revealed = true;
            
            if (cell.mine) {
                endGame(false);
                // Reveal all mines
                for (let gy = 0; gy < GRID_H; gy++)
                    for (let gx = 0; gx < GRID_W; gx++)
                        if (game.grid[gy][gx].mine) game.grid[gy][gx].revealed = true;
                return;
            }
            
            playSound('reveal');
            game.score += cell.neighbors === 0 ? 15 : 10 + cell.neighbors * 2;
            scoreDisplay.textContent = game.score;
            
            // Spawn particles
            const cs = getCellSize();
            const origin = getGridOrigin();
            const px = origin.x + x * cs + cs/2;
            const py = origin.y + y * cs + cs/2;
            for (let i = 0; i < 5; i++) {
                game.particles.push({
                    x: px, y: py,
                    vx: (Math.random()-0.5)*3,
                    vy: (Math.random()-0.5)*3 - 1,
                    life: 1,
                    color: `hsl(${40 + Math.random()*20}, 100%, 60%)`
                });
            }

            // Check win
            let allRevealed = true;
            for (let gy = 0; gy < GRID_H; gy++)
                for (let gx = 0; gx < GRID_W; gx++)
                    if (!game.grid[gy][gx].revealed) { allRevealed = false; break; }
            if (allRevealed) endGame(true);

            // Time bonus
            game.timeLeft += 1.5;
            if (game.timeLeft > TIME_LIMIT) game.timeLeft = TIME_LIMIT;
        }

        function moveCursor(dx, dy) {
            const nx = game.cursorX + dx;
            const ny = game.cursorY + dy;
            if (nx >= 0 && nx < GRID_W && ny >= 0 && ny < GRID_H) {
                game.cursorX = nx;
                game.cursorY = ny;
                playSound('move');
            }
        }

        // Input
        window.addEventListener('keydown', function(e) {
            if (game.state === 'start') {
                if (e.key === ' ' || e.key === 'Enter') startGame();
            } else if (game.state === 'playing') {
                initAudio();
                switch(e.key) {
                    case 'ArrowUp': e.preventDefault(); moveCursor(0, -1); break;
                    case 'ArrowDown': e.preventDefault(); moveCursor(0, 1); break;
                    case 'ArrowLeft': e.preventDefault(); moveCursor(-1, 0); break;
                    case 'ArrowRight': e.preventDefault(); moveCursor(1, 0); break;
                    case ' ': e.preventDefault(); revealCell(game.cursorX, game.cursorY); break;
                    case 'Enter': e.preventDefault(); revealCell(game.cursorX, game.cursorY); break;
                }
            } else if (game.state === 'over') {
                if (e.key === ' ' || e.key === 'Enter' || e.key === 'r') startGame();
            }
        });

        // Drawing
        function drawCheckeredBg() {
            const size = 40;
            for (let y = 0; y < Math.ceil(H/size); y++) {
                for (let x = 0; x < Math.ceil(W/size); x++) {
                    ctx.fillStyle = (x+y) % 2 === 0 ? '#0a0a1a' : '#0e0e22';
                    ctx.fillRect(x*size, y*size, size, size);
                }
            }
        }

        function drawStartScreen() {
            drawCheckeredBg();
            
            // Title
            ctx.save();
            ctx.textAlign = 'center';
            ctx.font = `bold ${Math.min(W*0.06, 60)}px Courier New`;
            ctx.fillStyle = '#ff6600';
            ctx.shadowColor = '#ff6600';
            ctx.shadowBlur = 20;
            ctx.fillText('RACING', W/2, H*0.25);
            ctx.font = `bold ${Math.min(W*0.08, 80)}px Courier New`;
            ctx.fillStyle = '#ffcc00';
            ctx.shadowColor = '#ffcc00';
            ctx.fillText('MINESWEEPER', W/2, H*0.35);
            ctx.restore();

            // Instructions
            ctx.save();
            ctx.textAlign = 'center';
            ctx.font = `${Math.min(W*0.025, 20)}px Courier New`;
            ctx.fillStyle = '#aaa';
            const lines = [
                'Navigate the minefield and reveal safe cells!',
                '',
                '[Arrow Keys] Move cursor',
                '[Space / Enter] Reveal cell',
                '',
                'Clear safe cells before time runs out!',
                'More neighbors on a cell = more points!',
                'Each reveal adds bonus time!'
            ];
            lines.forEach((line, i) => {
                ctx.fillText(line, W/2, H*0.5 + i * 30);
            });
            
            // Blinking start prompt
            if (Math.floor(performance.now() / 500) % 2 === 0) {
                ctx.font = `bold ${Math.min(W*0.03, 28)}px Courier New`;
                ctx.fillStyle = '#fff';
                ctx.fillText('Press SPACE to RACE!', W/2, H*0.85);
            }
            
            // Best score
            if (game.bestScore > 0) {
                ctx.font = `${Math.min(W*0.02, 18)}px Courier New`;
                ctx.fillStyle = '#ff6600';
                ctx.fillText(`Best: ${game.bestScore}`, W/2, H*0.92);
            }
            ctx.restore();
        }

        function drawGrid() {
            const cs = getCellSize();
            const origin = getGridOrigin();
            
            for (let y = 0; y < GRID_H; y++) {
                for (let x = 0; x < GRID_W; x++) {
                    const cx = origin.x + x * cs;
                    const cy = origin.y + y * cs;
                    const cell = game.grid[y][x];
                    
                    // Cell background
                    if (!cell.revealed) {
                        ctx.fillStyle = '#1a1a3a';
                        ctx.fillRect(cx, cy, cs, cs);
                        ctx.strokeStyle = '#2a2a5a';
                        ctx.strokeRect(cx+0.5, cy+0.5, cs-1, cs-1);
                    } else if (cell.mine) {
                        ctx.fillStyle = '#4a0000';
                        ctx.fillRect(cx, cy, cs, cs);
                        // Draw mine
                        ctx.fillStyle = '#ff3333';
                        ctx.beginPath();
                        ctx.arc(cx+cs/2, cy+cs/2, cs*0.3, 0, Math.PI*2);
                        ctx.fill();
                        // Spikes
                        for (let a = 0; a < 8; a++) {
                            const angle = a * Math.PI / 4;
                            ctx.beginPath();
                            ctx.moveTo(cx+cs/2, cy+cs/2);
                            ctx.lineTo(cx+cs/2 + Math.cos(angle)*cs*0.35, cy+cs/2 + Math.sin(angle)*cs*0.35);
                            ctx.strokeStyle = '#ff3333';
                            ctx.lineWidth = 2;
                            ctx.stroke();
                        }
                    } else {
                        // Safe revealed cell - checker pattern
                        const isCheck = (x+y) % 2 === 0;
                        ctx.fillStyle = isCheck ? '#151530' : '#1a1a3a';
                        ctx.fillRect(cx, cy, cs, cs);
                        
                        if (cell.neighbors > 0) {
                            ctx.font = `bold ${cs * 0.6}px Courier New`;
                            ctx.textAlign = 'center';
                            ctx.textBaseline = 'middle';
                            const colors = ['#ff6600','#ff9900','#ffcc00','#ffee00','#aaff00','#66ff00','#33ff66'];
                            ctx.fillStyle = colors[Math.min(cell.neighbors - 1, 7)];
                            ctx.fillText(String(cell.neighbors), cx+cs/2, cy+cs/2);
                        }
                    }
                    
                    // Cursor
                    if (x === game.cursorX && y === game.cursorY && !cell.revealed) {
                        ctx.strokeStyle = '#00ffff';
                        ctx.lineWidth = 3;
                        ctx.shadowColor = '#00ffff';
                        ctx.shadowBlur = 10;
                        ctx.strokeRect(cx+2, cy+2, cs-4, cs-4);
                        ctx.shadowBlur = 0;
                        
                        // Arrow indicator
                        ctx.fillStyle = 'rgba(0,255,255,0.3)';
                        ctx.fillRect(cx+2, cy+2, cs-4, cs-4);
                    }
                    
                    // Grid lines
                    ctx.strokeStyle = '#2a2a5a';
                    ctx.lineWidth = 1;
                    ctx.strokeRect(cx, cy, cs, cs);
                }
            }
        }

        function drawParticles() {
            game.particles = game.particles.filter(p => p.life > 0);
            game.particles.forEach(p => {
                p.x += p.vx;
                p.y += p.vy;
                p.vy += 0.1;
                p.life -= 0.03;
                ctx.globalAlpha = p.life;
                ctx.fillStyle = p.color;
                ctx.fillRect(p.x - 3, p.y - 3, 6, 6);
            });
            ctx.globalAlpha = 1;
        }

        function drawOverScreen() {
            // Darken
            ctx.fillStyle = 'rgba(0,0,0,0.7)';
            ctx.fillRect(0, 0, W, H);
            
            ctx.save();
            ctx.textAlign = 'center';
            
            const won = game.timeLeft > 0 && !game.grid[game.cursorY][game.cursorX].mine;
            // Actually check if we cleared all
            let allClear = true;
            for (let y = 0; y < GRID_H; y++)
                for (let x = 0; x < GRID_W; x++)
                    if (!game.grid[y][x].revealed) { allClear = false; break; }
            
            ctx.font = `bold ${Math.min(W*0.05, 50)}px Courier New`;
            ctx.fillStyle = allClear ? '#33ff66' : '#ff3333';
            ctx.shadowColor = allClear ? '#33ff66' : '#ff3333';
            ctx.shadowBlur = 20;
            ctx.fillText(allClear ? 'FINISH!' : 'WRECKED!', W/2, H*0.3);
            
            ctx.font = `${Math.min(W*0.03, 28)}px Courier New`;
            ctx.fillStyle = '#fff';
            ctx.shadowBlur = 0;
            ctx.fillText(`Score: ${game.score}`, W/2, H*0.45);
            ctx.fillText(`Best: ${game.bestScore}`, W/2, H*0.53);
            
            if (Math.floor(performance.now() / 500) % 2 === 0) {
                ctx.font = `bold ${Math.min(W*0.025, 24)}px Courier New`;
                ctx.fillStyle = '#ffcc00';
                ctx.fillText('Press SPACE to Race Again!', W/2, H*0.7);
            }
            ctx.restore();
        }

        function drawSpeedLines() {
            if (game.state !== 'playing') return;
            const t = performance.now() / 1000;
            ctx.save();
            ctx.globalAlpha = 0.3;
            for (let i = 0; i < 8; i++) {
                const y = ((t * 200 + i * H/8) % H);
                ctx.strokeStyle = `hsl(${(i*40)%360}, 100%, 50%)`;
                ctx.lineWidth = 2;
                ctx.beginPath();
                ctx.moveTo(0, y);
                ctx.lineTo(W, y + 20);
                ctx.stroke();
            }
            ctx.restore();
        }

        function update(dt) {
            if (game.state === 'playing') {
                game.timeLeft -= dt;
                if (game.timeLeft <= 0) {
                    game.timeLeft = 0;
                    endGame(false);
                }
                timerFill.style.width = `${(game.timeLeft / TIME_LIMIT) * 100}%`;
                
                // Timer color
                const ratio = game.timeLeft / TIME_LIMIT;
                if (ratio < 0.25) timerFill.style.background = '#ff3333';
                else if (ratio < 0.5) timerFill.style.background = 'linear-gradient(90deg, #ff6600, #ffcc00)';
                else timerFill.style.background = 'linear-gradient(90deg, #33ff66, #ffcc00)';
            }
            
            if (game.shake > 0) {
                game.shake -= dt * 30;
            }
        }

        function render() {
            ctx.save();
            
            // Screen shake
            if (game.shake > 0) {
                const sx = (Math.random()-0.5) * game.shake;
                const sy = (Math.random()-0.5) * game.shake;
                ctx.translate(sx, sy);
            }
            
            if (game.state === 'start') {
                drawStartScreen();
            } else {
                drawCheckeredBg();
                drawSpeedLines();
                drawGrid();
                drawParticles();
                if (game.state === 'over') drawOverScreen();
            }
            
            ctx.restore();
        }

        function gameLoop(timestamp) {
            const dt = Math.min((timestamp - game.lastTime) / 1000, 0.1);
            game.lastTime = timestamp;
            update(dt);
            render();
            requestAnimationFrame(gameLoop);
        }

        game.lastTime = performance.now();
        initGame();
        game.state = 'start';
        requestAnimationFrame(gameLoop);
    })();
    </script>
</body>
</html>
