function createDeck() {
  const suits = [
    { symbol: '♠', color: 'black' },
    { symbol: '♥', color: 'red' },
    { symbol: '♦', color: 'red' },
    { symbol: '♣', color: 'black' }
  ];
  const ranks = [
    { label: '2', val: 2 }, { label: '3', val: 3 }, { label: '4', val: 4 },
    { label: '5', val: 5 }, { label: '6', val: 6 }, { label: '7', val: 7 },
    { label: '8', val: 8 }, { label: '9', val: 9 }, { label: '10', val: 10 },
    { label: 'J', val: 11 }, { label: 'Q', val: 12 }, { label: 'K', val: 13 },
    { label: 'A', val: 14 }
  ];
  let d = [];
  for (let s of suits) {
    for (let r of ranks) {
      d.push({ rank: r.label, value: r.val, suit: s.symbol, color: s.color });
    }
  }
  // Shuffle
  for (let i = d.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    [d[i], d[j]] = [d[j], d[i]];
  }
  return d;
}

function resolveTrick2() {
  phase = 'revealed';
  const v1 = playedCards.p1 ? playedCards.p1.value : -1;

    leader = 'p1';
