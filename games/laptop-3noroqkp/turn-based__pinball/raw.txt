    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = 'sine';

function distToSeg(px, py, x1, y1, x2, y2) {
    const dx = x2 - x1, dy = y2 - y1;
    const len2 = dx*dx + dy*dy;
    let t = ((px-x1)*dx + (py-y1)*dy) / len2;
    t = Math.max(0, Math.min(1, t));
    const cx = x1 + t*dx, cy = y1 + t*dy;
    return { dist: Math.hypot(px-cx, py-cy), nx: (cx-px)/Math.hypot(px-cx,py-cy)||0, ny: (cy-py)/Math.hypot(px-cx,py-cy)||0, cx, cy };
}

function updatePhysics(dt) {
    const s = scale;
    ball.vy += 500 * s * dt;
    ball.x += ball.vx * dt;
    ball.y += ball.vy * dt;

    // Trail
    ballTrail.push({x: ball.x, y: ball.y});
    if (ballTrail.length > 20) ballTrail.shift();

    // Wall collisions
    for (const w of walls) {
        const d = distToSeg(ball.x, ball.y, w.x1, w.y1, w.x2, w.y2);
        if (d.dist < ball.r) {
            const nx = d.nx, ny = d.ny;
            const dot = ball.vx * nx + ball.vy * ny;
            if (dot < 0) {
                ball.vx -= 2 * dot * nx * 0.85;
                ball.vy -= 2 * dot * ny * 0.85;
                ball.x += nx * (ball.r - d.dist);
                ball.y += ny * (ball.r - d.dist);
                playBounce(300, 0.15);
            }
        }
    }

    // Bumper collisions
    for (const b of bumpers) {
        const dx = ball.x - b.x;
        const dy = ball.y - b.y;
        const dist = Math.hypot(dx, dy);
        if (dist < ball.r + b.r) {
            const nx = dx / dist, ny = dy / dist;
            const overlap = ball.r + b.r - dist;
            ball.x += nx * overlap;
            ball.y += ny * overlap;
            const dot = ball.vx * nx + ball.vy * ny;
            if (dot < 0) {
                ball.vx -= 2 * dot * nx;
                ball.vy -= 2 * dot * ny;
                // Bumper boost
                ball.vx += nx * 80 * s;
                ball.vy += ny * 80 * s;
                bumpersHitThisTurn++;
                score += b.points;
                playBounce(500 + Math.random()*200, 0.4);
                spawnParticles(b.x, b.y, b.color);
                shakeTime = 0.1;
            }
        }
    }

    // Check drain (ball falls below table)
    if (ball.y > tableY + tableH + 20*s) {
        return 'drain';
    }

    // Check ball stopped in upper area (safe zone = y < 60% of table)
    const speed = Math.hypot(ball.vx, ball.vy);
    if (speed < 10 * s && ball.y < tableY + tableH * 0.7) {
        return 'rest';
    }

    // Ball too slow overall
    if (speed < 5 * s) {
        ball.vy += 5; // nudge to prevent infinite rest
    }

    // Timeout safety
    return null;
}

function spawnParticles(x, y, color) {
    for (let i = 0; i < 8; i++) {
        const a = Math.random() * Math.PI * 2;
        const sp = (1 + Math.random()*3) * scale;
        particles.push({ x, y, vx: Math.cos(a)*sp, vy: Math.sin(a)*sp, life: 0.5, color });
    }
}

function updateParticles(dt) {
    for (let i = particles.length - 1; i >= 0; i--) {
        const p = particles[i];
        p.x += p.vx * dt;
        p.y += p.vy * dt;
        p.life -= dt;
        if (p.life <= 0) particles.splice(i, 1);
    }
}

let lastTime = 0;
function gameLoop(time) {
    const dt = Math.min((time - lastTime) / 1000, 0.033);
    lastTime = time;

    // Shake
    let sx = 0, sy = 0;
    if (shakeTime > 0) {
        shakeTime -= dt;
        sx = (Math.random() - 0.5) * 6 * scale;
        sy = (Math.random() - 0.5) * 6 * scale;
    }

    ctx.save();
    ctx.translate(sx, sy);

    if (state === STATE.MENU) {
        drawMenu();
    } else if (state === STATE.GAME_OVER) {
        drawTable();
        drawBall();
        drawHUD();
        drawGameOver();
    } else {
        drawTable();

        if (state === STATE.AIMING) {
            drawAimArrow();
            drawBall();
        } else if (state === STATE.PLAYING) {
            playTime += dt;
            const result = updatePhysics(dt);
            updateParticles(dt);
            drawBall();
            drawParticles();

            if (result === 'drain') {
                lives--;
                playDrain();
                resultText = 'DRAIN! Ball lost.';
                state = STATE.RESULT;
                resultTimer = 1.5;
                if (lives <= 0) {
                    setTimeout(() => {
                        if (score > highScore) {
                            highScore = score;
                            try { localStorage.setItem('pinball_hs', String(highScore)); } catch(e) {}
                        }
                        state = STATE.GAME_OVER;
                    }, 1500);
                }
            } else if (result === 'rest') {
                playScore();
                resultText = `SAVED! ${bumpersHitThisTurn} bumper(s) hit.`;
                state = STATE.RESULT;
                resultTimer = 1.2;
                turn++;
            }

            // Safety timeout
            if (playTime > 8) {
                resultText = 'Ball settled.';
                state = STATE.RESULT;
                resultTimer = 1;
                turn++;
            }
        } else if (state === STATE.RESULT) {
            drawBall();
            drawParticles();
            updateParticles(dt);
            drawResult();

            resultTimer -= dt;
            if (resultTimer <= 0) {
                resetBall();
                state = STATE.AIMING;
            }
        }

        drawHUD();
    }

    ctx.restore();
    requestAnimationFrame(gameLoop);
}

    initAudio();

    if (state === STATE.MENU || state === STATE.GAME_OVER) {
        if (e.key === 'Enter') startGame();
        return;
    }

    if (state === STATE.AIMING) {
        if (e.key === 'ArrowLeft' || e.key === 'a') angle = Math.max(angle - 3, -80);
        if (e.key === 'ArrowRight' || e.key === 'd') angle = Math.min(angle + 3, 80);
        if (e.key === 'ArrowUp' || e.key === 'w') power = Math.min(power + 2, 100);
        if (e.key === 'ArrowDown' || e.key === 's') power = Math.max(power - 2, 10);
        if (e.key === ' ') { launchBall(); playTime = 0; }
    }

    if (state === STATE.PLAYING) {
        // Nudge! Slight push in aim direction
        const rad = (angle - 90) * Math.PI / 180;
        ball.vx += Math.cos(rad) * 50 * scale;
        ball.vy += Math.sin(rad) * 50 * scale;
    }
});

        const mx = e.clientX - rect.left;

        const s = scale;
        if (mx > tableX + tableW - 100*s && my > tableY + tableH - 60*s) {
            launchBall();
            playTime = 0;
        }
    }

function updateAimFromMouse(e) {
    const rect = canvas.getBoundingClientRect();
    const mx = e.clientX - rect.left;
    const my = e.clientY - rect.top;
    const dx = mx - ball.x;
    const dy = my - ball.y;
    let a = Math.atan2(dy, dx) * 180 / Math.PI + 90;
    angle = Math.max(-80, Math.min(80, a));
    power = Math.min(100, Math.max(10, Math.hypot(dx, dy) / (scale * 2)));
}

    if (dragging && state === STATE.AIMING) { dragging = false; }
