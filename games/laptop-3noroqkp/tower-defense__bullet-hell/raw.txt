Let me go with the wpIdx approach:
```javascript
function updateEnemy(e, dt) {
  const wp = waypoints[e.wpIdx];
  const dx = wp[0] - e.x;
  const dy = wp[1] - e.y;
  const dist = Math.sqrt(dx*dx + dy*dy);
  if (dist < 2) {
    e.wpIdx++;
    if (e.wpIdx >= waypoints.length) {
      // Enemy reached the end
      lives -= e.livesCost || 1;
      e.dead = true;
      return;
    }
  } else {
    const move = e.speed * dt;
    e.x += (dx/dist) * move;
    e.y += (dy/dist) * move;
  }
}
```

    if(!wp){enemies.splice(i,1);continue;}
    const dx=wp[0]-e.x,dy=wp[1]-e.y;
    const dist=Math.sqrt(dx*dx+dy*dy);
    const mv=e.spd*dt;
    if(dist<=mv){
      e.pathDist+=dist;
      e.wpi++;
      if(e.wpi>=WAYPOINTS.length){

        if(lives<=0){gs='gameover';saveBest();sfx('over');}

      e.x+=dx/dist*mv;e.y+=dy/dist*mv;e.pathDist+=mv;
    }

    t.cd-=dt*1000;
