function playSound(freq, dur, type = 'square', vol = 0.15) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = type;
    osc.frequency.value = freq;
    gain.gain.setValueAtTime(vol, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + dur);
}

function initGrid() {
    grid = [];
    for (let r = 0; r < gridSize; r++) {
        grid[r] = [];
        for (let c = 0; c < gridSize; c++) {
            let t;
            do { t = Math.floor(Math.random() * 5); } while (wouldMatch(r, c, t));
            grid[r][c] = t;
        }
    }
}

function wouldMatch(r, c, t) {
    if (c >= 2 && grid[r][c-1] === t && grid[r][c-2] === t) return true;
    if (r >= 2 && grid[r-1][c] === t && grid[r-2][c] === t) return true;
    return false;
}

function findMatches() {
    let matches = [];
    // Horizontal
    for (let r = 0; r < gridSize; r++) {
        for (let c = 0; c < gridSize - 2; c++) {
            let t = grid[r][c];
            if (t === null) continue;
            let count = 1;
            while (c + count < gridSize && grid[r][c + count] === t) count++;
            if (count >= 3) {
                for (let i = 0; i < count; i++) matches.push({ r, c: c + i });
                c += count - 1;
            }
        }
    }
    // Vertical
    for (let c = 0; c < gridSize; c++) {
        for (let r = 0; r < gridSize - 2; r++) {
            let t = grid[r][c];
            if (t === null) continue;
            let count = 1;
            while (r + count < gridSize && grid[r + count][c] === t) count++;
            if (count >= 3) {
                for (let i = 0; i < count; i++) matches.push({ r: r + i, c });
                r += count - 1;
            }
        }
    }
    // Deduplicate
    let unique = [];
    let seen = new Set();
    for (let m of matches) {
        let key = m.r * gridSize + m.c;
        if (!seen.has(key)) { seen.add(key); unique.push(m); }
    }
    return unique;
}

function removeMatches(matches) {
    for (let m of matches) {
        grid[m.r][m.c] = null;
        spawnMatchParticles(m.r, m.c);
        firePlayerBullets(m.r, m.c);
    }
}

function applyGravity() {
    let moved = false;
    for (let c = 0; c < gridSize; c++) {
        for (let r = gridSize - 1; r >= 0; r--) {
            if (grid[r][c] === null) {
                for (let r2 = r - 1; r2 >= 0; r2--) {
                    if (grid[r2][c] !== null) {
                        grid[r][c] = grid[r2][c];
                        grid[r2][c] = null;
                        moved = true;
                        break;
                    }
                }
            }
        }
    }
    // Fill top
    for (let c = 0; c < gridSize; c++) {
        for (let r = 0; r < gridSize; r++) {
            if (grid[r][c] === null) {
                grid[r][c] = Math.floor(Math.random() * 5);
                moved = true;
            }
        }
    }
    return moved;
}

function processCascade() {
    animating = false;
    let matches = findMatches();
    if (matches.length > 0) {
        combo++;
        let pts = matches.length * 10 * combo;
        score += pts;
        playMatch();
        screenShake = Math.min(screenShake + matches.length, 8);
        removeMatches(matches);
        applyGravity();
        animating = true;
    } else {
        combo = 0;
        animating = false;
    }
}

function trySwap(r1, c1, r2, c2) {
    if (Math.abs(r1 - r2) + Math.abs(c1 - c2) !== 1) return;
    let temp = grid[r1][c1];
    grid[r1][c1] = grid[r2][c2];
    grid[r2][c2] = temp;
    if (findMatches().length === 0) {
        // Swap back
        grid[r2][c2] = grid[r1][c1];
        grid[r1][c1] = temp;
    } else {
        playSwap();
        animating = true;
    }
}

function firePlayerBullets(r, c) {
    let x = gridX + (c + 0.5) * cellSize;
    let y = gridY + (r + 0.5) * cellSize;
    playShoot();

            vy: Math.sin(angle + i * 0.3) * 8,
            life: 120,

        });

}

function spawnMatchParticles(r, c) {
    let x = gridX + (c + 0.5) * cellSize;
    let y = gridY + (r + 0.5) * cellSize;
    let color = GEM_COLORS[grid[r][c] || 0];
    for (let i = 0; i < 6; i++) {
        let a = Math.random() * Math.PI * 2;
        let sp = 1 + Math.random() * 3;
        particles.push({ x, y, vx: Math.cos(a) * sp, vy: Math.sin(a) * sp - 2, life: 30, color });
    }
}

function spawnEnemy(type, x) {
    let hpVal = type === 0 ? 1 : type === 1 ? 3 : 5;
    enemies.push({
        type, x: x || Math.random() * (W - 80) + 40,
        y: -30,
        vy: 0.3 + wave * 0.05,
        hp: hpVal, maxHp: hpVal,
        shootTimer: Math.floor(Math.random() * 60),
        angle: 0,
        size: type === 2 ? 24 : 18
    });
}

function enemyShoot(e) {
    let bx = e.x;
    let by = e.y + e.size / 2;
    if (e.type === 0) {
        // Aimed at random grid position
        let tx = gridX + Math.random() * cellSize * gridSize;
        let ty = gridY + Math.random() * cellSize * gridSize;
        let a = Math.atan2(ty - by, tx - bx);
        enemyBullets.push({ x: bx, y: by, vx: Math.cos(a) * 3, vy: Math.sin(a) * 3, life: 200 });
    } else if (e.type === 1) {
        // Spread of 5
        for (let i = -2; i <= 2; i++) {
            let a = Math.PI / 2 + i * 0.3;
            enemyBullets.push({ x: bx, y: by, vx: Math.cos(a) * 2.5, vy: Math.sin(a) * 2.5, life: 200 });
        }
    } else {
        // Spiral burst
        for (let i = 0; i < 8; i++) {
            let a = e.angle + i * (Math.PI * 2 / 8);
            enemyBullets.push({ x: bx, y: by, vx: Math.cos(a) * 2, vy: Math.sin(a) * 2, life: 180 });
        }
    }
}

function update(dt) {
    time += dt;

    if (state !== 'playing') return;

    // Screen shake decay
    screenShake *= 0.9;
    flashAlpha *= 0.95;

    // Wave system
    waveTimer++;
    if (waveTimer > 600) { // ~10 seconds per wave at 60fps
        wave++;
        waveTimer = 0;
        maxHp = Math.min(5 + Math.floor(wave / 3), 15);
        hp = Math.min(hp, maxHp);
    }

    // Spawn enemies
    spawnTimer--;
    if (spawnTimer <= 0) {
        let types = wave < 2 ? [0] : wave < 4 ? [0, 1] : [0, 1, 2];
        let type = types[Math.floor(Math.random() * types.length)];
        spawnEnemy(type);
        spawnTimer = Math.max(60 - wave * 5, 20);
    }

    // Update enemies
    for (let e of enemies) {
        e.y += e.vy;
        e.angle += 0.03;
        e.shootTimer--;
        if (e.shootTimer <= 0) {
            enemyShoot(e);
            e.shootTimer = Math.max(40 - wave * 2, 15);
        }
    }

    // Update player bullets
    for (let i = playerBullets.length - 1; i >= 0; i--) {
        let b = playerBullets[i];
        b.x += b.vx;
        b.y += b.vy;
        b.life--;
        if (b.life <= 0 || b.y < -20) { playerBullets.splice(i, 1); continue; }

        // Hit enemies
        for (let j = enemies.length - 1; j >= 0; j--) {
            let e = enemies[j];
            let dx = b.x - e.x, dy = b.y - e.y;
            if (Math.abs(dx) < e.size && Math.abs(dy) < e.size) {
                e.hp--;
                playerBullets.splice(i, 1);
                // Hit particles
                for (let k = 0; k < 4; k++) {
                    let a = Math.random() * Math.PI * 2;
                    particles.push({ x: b.x, y: b.y, vx: Math.cos(a) * 3, vy: Math.sin(a) * 3, life: 20, color: GEM_COLORS[0] });
                }
                if (e.hp <= 0) {
                    score += 50 + wave * 10;
                    playEnemyDie();
                    for (let k = 0; k < 10; k++) {
                        let a = Math.random() * Math.PI * 2;
                        particles.push({ x: e.x, y: e.y, vx: Math.cos(a) * 4, vy: Math.sin(a) * 4, life: 30, color: { r: 255, g: 100, b: 50 } });
                    }
                    enemies.splice(j, 1);
                }
                break;
            }
        }
    }

    // Update enemy bullets
    for (let i = enemyBullets.length - 1; i >= 0; i--) {
        let b = enemyBullets[i];
        b.x += b.vx;
        b.y += b.vy;
        b.life--;
        if (b.life <= 0 || b.x < -20 || b.x > W + 20) { enemyBullets.splice(i, 1); continue; }

        // Hit grid gems
        let gc = Math.floor((b.x - gridX) / cellSize);
        let gr = Math.floor((b.y - gridY) / cellSize);
        if (gc >= 0 && gc < gridSize && gr >= 0 && gr < gridSize) {
            // Check if it's actually hitting a gem area
            let inCell = true;
            if (inCell && grid[gr][gc] !== null) {
                // Destroy the gem!
                grid[gr][gc] = null;
                applyGravity();
                for (let k = 0; k < 5; k++) {
                    let a = Math.random() * Math.PI * 2;
                    particles.push({ x: b.x, y: b.y, vx: Math.cos(a) * 3, vy: Math.sin(a) * 3, life: 20, color: { r: 255, g: 50, b: 50 } });
                }
                playHurt();
                screenShake = 4;
                enemyBullets.splice(i, 1);
                continue;
            }
        }

        // Passed bottom - player takes damage
        if (b.y > H + 20) {
            hp--;
            flashAlpha = 0.3;
            playHurt();
            screenShake = 6;
            enemyBullets.splice(i, 1);
            if (hp <= 0) {
                state = 'gameover';
                try { localStorage.setItem('bhm_high', String(highScore)); } catch(e) {}
            }
        }
    }

    // Update particles
    for (let i = particles.length - 1; i >= 0; i--) {
        let p = particles[i];
        p.x += p.vx;
        p.y += p.vy;
        p.vy += 0.2;
        p.life--;
        if (p.life <= 0) particles.splice(i, 1);
    }

    // Update high score
    if (score > highScore) highScore = score;
}

function drawGem(x, y, size, type, alpha = 1) {
    let c = GEM_COLORS[type];
    ctx.globalAlpha = alpha;
    ctx.fillStyle = `rgb(${c.r},${c.g},${c.b})`;
    let r = size * 0.35;
    // Draw gem shape (diamond)
    ctx.beginPath();
    ctx.moveTo(x, y - r);
    ctx.lineTo(x + r, y);
    ctx.lineTo(x, y + r);
    ctx.lineTo(x - r, y);
    ctx.closePath();
    ctx.fill();
    // Highlight
    ctx.fillStyle = `rgba(255,255,255,0.3)`;
    ctx.beginPath();
    ctx.moveTo(x, y - r * 0.6);
    ctx.lineTo(x + r * 0.4, y - r * 0.2);
    ctx.lineTo(x, y + r * 0.1);
    ctx.closePath();
    ctx.fill();
    ctx.globalAlpha = 1;
}

            }
        }

        );

        ctx.save();
        ctx.translate(e.x, e.y);
        let s = e.size;
        if (e.type === 0) {
            // Diamond enemy
            ctx.fillStyle = '#ff4444';
            ctx.beginPath();
            ctx.moveTo(0, -s);
            ctx.lineTo(s, 0);
            ctx.lineTo(0, s);
            ctx.lineTo(-s, 0);
            ctx.closePath();
            ctx.fill();

            ctx.beginPath();
            ctx.moveTo(0, -s * 0.5);

            ctx.closePath();
            ctx.fill();

            ctx.beginPath();
            for (let i = 0; i < 6; i++) {
                let a = i * Math.PI / 3 - Math.PI / 2;
                let px = Math.cos(a) * s, py = Math.sin(a) * s;
                if (i === 0) ctx.moveTo(px, py); else ctx.lineTo(px, py);
            }
            ctx.closePath();
            ctx.fill();
            ctx.fillStyle = '#88bbff';
            ctx.beginPath();
            ctx.arc(0, 0, s * 0.4, 0, Math.PI * 2);
            ctx.fill();
        } else {
            // Spinning star enemy
            ctx.rotate(e.angle);
            ctx.fillStyle = '#ff44ff';
            ctx.beginPath();
            for (let i = 0; i < 8; i++) {
                let a = i * Math.PI / 4;
                let r2 = i % 2 === 0 ? s : s * 0.5;
                let px = Math.cos(a) * r2, py = Math.sin(a) * r2;
                if (i === 0) ctx.moveTo(px, py); else ctx.lineTo(px, py);
            }
            ctx.closePath();
            ctx.fill();
        }
        ctx.restore();

        // HP bar
        let hpw = e.size * 1.5;
        ctx.fillStyle = 'rgba(0,0,0,0.5)';
        ctx.fillRect(e.x - hpw / 2, e.y - e.size - 8, hpw, 4);

    }

        ctx.arc(b.x, b.y, 4, 0, Math.PI * 2);
        ctx.fill();
        // Trail
        ctx.fillStyle = `rgba(${b.color.r},${b.color.g},${b.color.b},0.3)`;
        ctx.beginPath();
        ctx.arc(b.x - b.vx, b.y - b.vy, 3, 0, Math.PI * 2);
        ctx.fill();
    }

        ctx.fillStyle = '#ff6644';
        ctx.beginPath();
        ctx.arc(b.x, b.y, 5, 0, Math.PI * 2);
        ctx.fill();
        ctx.fillStyle = '#ffaa88';
        ctx.beginPath();
        ctx.arc(b.x, b.y, 2.5, 0, Math.PI * 2);
        ctx.fill();
    }

    // Particles

        ctx.font = '12px monospace';
        ctx.textAlign = 'right';
        ctx.fillText(`HP ${hp}/${maxHp}`, W - 25, 34);

        // High score
        ctx.textAlign = 'center';
        ctx.fillStyle = '#aaa';
        ctx.font = '12px monospace';
        ctx.fillText(`HI: ${highScore}`, W / 2, 30);
    }

    // Flash effect
    if (flashAlpha > 0.01) {
        ctx.fillStyle = `rgba(255, 0, 0, ${flashAlpha})`;
        ctx.fillRect(0, 0, W, H);
    }

    // Game over overlay
    if (state === 'gameover') {
        ctx.fillStyle = 'rgba(0, 0, 0, 0.7)';
        ctx.fillRect(0, 0, W, H);
        ctx.fillStyle = '#ff4444';
        ctx.font = 'bold 48px monospace';
        ctx.textAlign = 'center';
        ctx.fillText('GAME OVER', W / 2, H / 2 - 40);
        ctx.fillStyle = '#fff';
        ctx.font = '24px monospace';
        ctx.fillText(`Score: ${score}`, W / 2, H / 2 + 10);
        ctx.fillText(`Best: ${highScore}`, W / 2, H / 2 + 45);
        ctx.fillStyle = '#88f';
        ctx.font = '16px monospace';
        ctx.fillText('Press SPACE or click to restart', W / 2, H / 2 + 90);
    }

    ctx.restore();
}

let dragStart = null;

function getGridPos(x, y) {
    let c = Math.floor((x - gridX) / cellSize);
    let r = Math.floor((y - gridY) / cellSize);
    if (r >= 0 && r < gridSize && c >= 0 && c < gridSize) return { r, c };
    return null;
}

canvas.addEventListener('mousedown', (e) => {
    initAudio();
    let rect = canvas.getBoundingClientRect();
    let mx = e.clientX - rect.left;
    let my = e.clientY - rect.top;

    if (state === 'title' || state === 'gameover') {
        startGame();
        return;
    }

    if (animating) return;

    let pos = getGridPos(mx, my);
    if (!pos) return;

    if (selected) {
        if (Math.abs(selected.r - pos.r) + Math.abs(selected.c - pos.c) === 1) {
            trySwap(selected.r, selected.c, pos.r, pos.c);
            selected = null;
        } else {
            selected = pos;
        }
    } else {
        selected = pos;
    }
});

    if (state !== 'playing' || animating) return;

    let mx = e.clientX - rect.left;

let lastFrame = performance.now();
function gameLoop(now) {
    let dt = Math.min((now - lastFrame) / 16.67, 3);
    lastFrame = now;

    update(dt);

    // Handle animation (cascade)
    if (animating && state === 'playing') {
        processCascade();
    }

    draw();
    requestAnimationFrame(gameLoop);
}
