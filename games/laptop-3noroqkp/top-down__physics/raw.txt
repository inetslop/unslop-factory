function playSound(type) {

function resize() {
    W = window.innerWidth;
    H = window.innerHeight;
    canvas.width = W;
    canvas.height = H;
    
    // Table takes up most of the space with some margin
    const margin = Math.min(W, H) * 0.08;
    tableW = W - margin * 2;
    tableH = H - margin * 2;
    tableX = margin;
    tableY = margin;
    
    // Keep aspect ratio ~2:1 for the table
    if (tableW > tableH * 2) {
        tableW = tableH * 2;
        tableX = (W - tableW) / 2;
    } else if (tableH > tableW * 0.8) {
        tableH = tableW * 0.8;
        tableY = (H - tableH) / 2;
    }
    
    scale = Math.min(tableW, tableH) / 400;
}

    const r = BALL_R * scale + 4;
    pockets = [

        { x: tableX + tableW / 2, y: tableY + r * 0.5 },
        { x: tableX + tableW - r, y: tableY + r },
        { x: tableX + r, y: tableY + tableH - r },
        { x: tableX + tableW / 2, y: tableY + tableH - r * 0.5 },
        { x: tableX + tableW - r, y: tableY + tableH - r }
    ];
}

function initLevel(lvl) {
    const cfg = LEVELS[lvl - 1];
    balls = [];
    particles = [];
    pocketAnim = [];
    
    // Cue ball at bottom center

        x: tableX + tableW / 2,

        mass: 1,
        color: BALL_COLORS[0],
        isCue: true,
        active: true,
        pocketAnim: 0

    balls.push(cueBall);
    
    // Object balls
    cfg.balls.forEach((pos, i) => {
        balls.push({
            x: tableX + pos[0] * tableW,
            y: tableY + pos[1] * tableH,
            vx: 0, vy: 0,
            r: BALL_R * scale,
            mass: 1,
            color: BALL_COLORS[i + 1],
            isCue: false,
            active: true,
            pocketAnim: 0

    score = 0;

        gameState = 'gameover';

    }

function spawnParticles(x, y, color, count) {
    for (let i = 0; i < count; i++) {
        const angle = Math.random() * Math.PI * 2;
        const speed = 1 + Math.random() * 3;
        particles.push({
            x, y,
            vx: Math.cos(angle) * speed,
            vy: Math.sin(angle) * speed,
            life: 1,
            color: color,
            r: 2 + Math.random() * 3
        });
    }
}

function updatePhysics() {
    if (gameState !== 'playing') return;
    
    let anyMoving = false;
    
    for (let b of balls) {
        if (!b.active) continue;
        
        if (Math.abs(b.vx) > STOP_SPEED || Math.abs(b.vy) > STOP_SPEED) {
            anyMoving = true;
            b.x += b.vx;
            b.y += b.vy;
            b.vx *= FRICTION;
            b.vy *= FRICTION;
            
            if (Math.abs(b.vx) < STOP_SPEED && Math.abs(b.vy) < STOP_SPEED) {
                b.vx = 0;
                b.vy = 0;
            }
        } else {
            b.vx = 0;
            b.vy = 0;
        }
        
        // Wall collisions
        if (b.x - b.r < tableX) {
            b.x = tableX + b.r;
            b.vx = Math.abs(b.vx) * WALL_RESTITUTION;
            if (Math.abs(b.vx) > 1) playSound('wall');
        }
        if (b.x + b.r > tableX + tableW) {
            b.x = tableX + tableW - b.r;
            b.vx = -Math.abs(b.vx) * WALL_RESTITUTION;
            if (Math.abs(b.vx) > 1) playSound('wall');
        }
        if (b.y - b.r < tableY) {
            b.y = tableY + b.r;
            b.vy = Math.abs(b.vy) * WALL_RESTITUTION;
            if (Math.abs(b.vy) > 1) playSound('wall');
        }
        if (b.y + b.r > tableY + tableH) {
            b.y = tableY + tableH - b.r;
            b.vy = -Math.abs(b.vy) * WALL_RESTITUTION;
            if (Math.abs(b.vy) > 1) playSound('wall');
        }
    }
    
    // Ball-ball collisions
    for (let i = 0; i < balls.length; i++) {
        for (let j = i + 1; j < balls.length; j++) {
            const a = balls[i], b = balls[j];
            if (!a.active || !b.active) continue;
            
            const dx = b.x - a.x;
            const dy = b.y - a.y;
            const dist = Math.sqrt(dx * dx + dy * dy);
            const minDist = a.r + b.r;
            
            if (dist < minDist && dist > 0) {
                // Separate
                const overlap = minDist - dist;
                const nx = dx / dist, ny = dy / dist;
                a.x -= nx * overlap * 0.5;
                a.y -= ny * overlap * 0.5;
                b.x += nx * overlap * 0.5;
                b.y += ny * overlap * 0.5;
                
                // Elastic collision
                const dvx = a.vx - b.vx;
                const dvy = a.vy - b.vy;
                const dot = dvx * nx + dvy * ny;
                
                if (dot > 0) {
                    const impulse = dot / (a.mass + b.mass);
                    a.vx -= impulse * b.mass * nx;
                    a.vy -= impulse * b.mass * ny;
                    b.vx += impulse * a.mass * nx;
                    b.vy += impulse * a.mass * ny;
                    
                    const speed = Math.sqrt(dvx * dvx + dvy * dvy);
                    if (speed > 1) {
                        playSound('hit');
                        spawnParticles((a.x + b.x) / 2, (a.y + b.y) / 2, '#fff', 3);
                    }
                }
            }
        }
    }
    
    // Pocket detection
    for (let b of balls) {
        if (!b.active) continue;
        for (let p of pockets) {
            const dx = b.x - p.x;
            const dy = b.y - p.y;
            const dist = Math.sqrt(dx * dx + dy * dy);
            if (dist < POCKET_R * scale * 0.6) {
                b.active = false;
                b.pocketAnim = 1;
                b.vx = 0;
                b.vy = 0;
                pocketAnim.push({ x: p.x, y: p.y, life: 1, color: b.color });
                playSound('pocket');
                spawnParticles(p.x, p.y, b.color, 12);
                
                if (!b.isCue) {
                    score += 10;
                } else {
                    // Cue ball pocketed - respawn
                    setTimeout(() => {
                        b.active = true;
                        b.pocketAnim = 0;
                        b.x = tableX + tableW / 2;
                        b.y = tableY + tableH * 0.8;
                        b.vx = 0;
                        b.vy = 0;
                    }, 500);
                }
            }
        }
    }
    
    // Update particles
    for (let i = particles.length - 1; i >= 0; i--) {
        const p = particles[i];
        p.x += p.vx;
        p.y += p.vy;
        p.vx *= 0.95;
        p.vy *= 0.95;
        p.life -= 0.03;
        if (p.life <= 0) particles.splice(i, 1);
    }
    
    // Update pocket animations
    for (let i = pocketAnim.length - 1; i >= 0; i--) {
        pocketAnim[i].life -= 0.02;
        if (pocketAnim[i].life <= 0) pocketAnim.splice(i, 1);
    }
    
    // Check win/lose
    if (!anyMoving && gameState === 'playing') {
        const objectBallsLeft = balls.filter(b => !b.isCue && b.active).length;
        
        if (objectBallsLeft === 0) {
            gameState = 'won';
            playSound('win');
            return;
        }
        
        if (shotsLeft <= 0) {
            gameState = 'lost';
            playSound('lose');
        }
    }
}

function shoot() {
    if (!aimStart || !cueBall || !cueBall.active) return;
    
    const dx = aimStart.x - mousePos.x;
    const dy = aimStart.y - mousePos.y;
    const power = Math.min(Math.sqrt(dx * dx + dy * dy) / 10, MAX_POWER);
    
    if (power < 0.5) return;
    
    const angle = Math.atan2(dy, dx);
    cueBall.vx = Math.cos(angle) * power * scale * 0.8;
    cueBall.vy = Math.sin(angle) * power * scale * 0.8;
    
    shotsLeft--;
    playSound('shoot');
    aimStart = null;
    mousePos = null;
}

function render() {
    ctx.fillStyle = '#0a0a1a';
    ctx.fillRect(0, 0, W, H);
    
    // Screen shake
    if (shakeTime > 0) {
        shakeTime -= 0.05;
        const sx = (Math.random() - 0.5) * shakeIntensity;
        const sy = (Math.random() - 0.5) * shakeIntensity;
        ctx.save();
        ctx.translate(sx, sy);
    }
    
    // Table background
    ctx.fillStyle = '#1a4a2a';
    ctx.fillRect(tableX, tableY, tableW, tableH);
    
    // Table border (rails)
    ctx.strokeStyle = '#2a6a3a';
    ctx.lineWidth = 6;
    ctx.strokeRect(tableX - 3, tableY - 3, tableW + 6, tableH + 6);
    
    ctx.strokeStyle = '#0ff';
    ctx.lineWidth = 1;
    ctx.globalAlpha = 0.3;
    ctx.strokeRect(tableX - 1, tableY - 1, tableW + 2, tableH + 2);
    ctx.globalAlpha = 1;
    
    // Pockets
    for (let p of pockets) {
        ctx.beginPath();
        ctx.arc(p.x, p.y, POCKET_R * scale * 0.6, 0, Math.PI * 2);
        ctx.fillStyle = '#0a0a0a';
        ctx.fill();
        ctx.strokeStyle = '#333';
        ctx.lineWidth = 2;
        ctx.stroke();
    }
    
    // Pocket animations (balls sinking)
    for (let pa of pocketAnim) {
        ctx.globalAlpha = pa.life;
        ctx.beginPath();
        ctx.arc(pa.x, pa.y, pa.life * BALL_R * scale, 0, Math.PI * 2);
        ctx.fillStyle = pa.color;
        ctx.fill();
        ctx.globalAlpha = 1;
    }
    
    // Particles
    for (let p of particles) {
        ctx.globalAlpha = p.life;
        ctx.beginPath();
        ctx.arc(p.x, p.y, p.r * p.life, 0, Math.PI * 2);
        ctx.fillStyle = p.color;
        ctx.fill();
        ctx.globalAlpha = 1;
    }
    
    // Balls
    for (let b of balls) {
        if (!b.active) continue;
        
        // Shadow
        ctx.beginPath();
        ctx.arc(b.x + 2, b.y + 2, b.r, 0, Math.PI * 2);
        ctx.fillStyle = 'rgba(0,0,0,0.3)';
        ctx.fill();
        
        // Ball body
        const grad = ctx.createRadialGradient(b.x - b.r * 0.3, b.y - b.r * 0.3, 0, b.x, b.y, b.r);
        grad.addColorStop(0, lightenColor(b.color, 40));
        grad.addColorStop(1, b.color);
        ctx.beginPath();
        ctx.arc(b.x, b.y, b.r, 0, Math.PI * 2);
        ctx.fillStyle = grad;
        ctx.fill();
        
        // Highlight
        ctx.beginPath();
        ctx.arc(b.x - b.r * 0.25, b.y - b.r * 0.25, b.r * 0.35, 0, Math.PI * 2);
        ctx.fillStyle = 'rgba(255,255,255,0.4)';
        ctx.fill();
        
        // Glow for object balls
        if (!b.isCue) {
            ctx.beginPath();
            ctx.arc(b.x, b.y, b.r + 2, 0, Math.PI * 2);
            ctx.strokeStyle = b.color;
            ctx.globalAlpha = 0.3;
            ctx.lineWidth = 1;
            ctx.stroke();
            ctx.globalAlpha = 1;
        }
    }
    
    // Aim line
    if (aimStart && mousePos && cueBall && cueBall.active) {
        const dx = aimStart.x - mousePos.x;
        const dy = aimStart.y - mousePos.y;
        const len = Math.sqrt(dx * dx + dy * dy);
        
        if (len > 5) {
            const angle = Math.atan2(dy, dx);
            const lineLen = Math.min(len, MAX_POWER * 10);
            
            // Dashed aim line
            ctx.setLineDash([5, 5]);
            ctx.beginPath();
            ctx.moveTo(cueBall.x, cueBall.y);
            ctx.lineTo(
                cueBall.x + Math.cos(angle) * lineLen,
                cueBall.y + Math.sin(angle) * lineLen
            );
            ctx.strokeStyle = 'rgba(255,255,255,0.6)';
            ctx.lineWidth = 2;
            ctx.stroke();
            ctx.setLineDash([]);
            
            // Power indicator
            const power = Math.min(len / 10, MAX_POWER) / MAX_POWER;
            ctx.beginPath();
            ctx.arc(cueBall.x, cueBall.y, BALL_R * scale + 5, -Math.PI / 2, -Math.PI / 2 + power * Math.PI * 2);
            ctx.strokeStyle = power > 0.8 ? '#ff4444' : power > 0.5 ? '#ffaa00' : '#44ff44';
            ctx.lineWidth = 3;
            ctx.stroke();
            
            // Power text
            ctx.font = `${12 * scale}px monospace`;
            ctx.fillStyle = '#fff';
            ctx.textAlign = 'center';
            ctx.fillText(`${Math.round(power * 100)}%`, cueBall.x, cueBall.y - BALL_R * scale - 12);
        }
    }
    
    // HUD
    ctx.font = `bold ${16 * Math.min(scale, 1.5)}px monospace`;
    ctx.textAlign = 'left';
    ctx.fillStyle = '#fff';
    ctx.fillText(`LEVEL ${level}`, tableX + 10, tableY - 12);
    ctx.fillText(`SCORE: ${score}`, tableX + tableW / 2 - 50, tableY - 12);
    
    const objectBallsLeft = balls.filter(b => !b.isCue && b.active).length;
    ctx.textAlign = 'right';
    ctx.fillText(`BALLS: ${objectBallsLeft}`, tableX + tableW - 80, tableY - 12);
    ctx.fillText(`SHOTS: ${shotsLeft}`, tableX + tableW - 10, tableY - 12);
    
    // Shots indicator dots
    const dotSize = 6;
    for (let i = 0; i < LEVELS[level - 1].shots; i++) {
        ctx.beginPath();
        ctx.arc(tableX + tableW / 2 - 50 + i * (dotSize + 4), tableY + tableH + 15, dotSize / 2, 0, Math.PI * 2);
        ctx.fillStyle = i < shotsLeft ? '#ffaa00' : '#333';
        ctx.fill();
    }
    
    if (shakeTime > 0) {
        ctx.restore();
    }
    
    // Overlays
    if (gameState === 'start') {
        drawStartScreen();
    } else if (gameState === 'won') {
        drawWinScreen();
    } else if (gameState === 'lost') {
        drawLoseScreen();
    } else if (gameState === 'gameover') {
        drawGameOverScreen();
    }
}

function lightenColor(hex, amt) {
    let r = parseInt(hex.slice(1, 3), 16);
    let g = parseInt(hex.slice(3, 5), 16);
    let b = parseInt(hex.slice(5, 7), 16);
    r = Math.min(255, r + amt);
    g = Math.min(255, g + amt);
    b = Math.min(255, b + amt);
    return `rgb(${r},${g},${b})`;
}

canvas.addEventListener('mousedown', (e) => {
    initAudio();

    const mx = e.clientX - rect.left;
    const my = e.clientY - rect.top;
    
    if (gameState === 'start') {
        startGame();
        return;
    }
    if (gameState === 'won') {
        nextLevel();
        return;
    }
    if (gameState === 'lost') {
        initLevel(level);
        return;

        startGame();
        return;

    if (gameState === 'playing' && cueBall && cueBall.active) {
        const allStopped = balls.every(b => Math.abs(b.vx) < 0.1 && Math.abs(b.vy) < 0.1);
        if (!allStopped) return;
        
        const dx = mx - cueBall.x;
        const dy = my - cueBall.y;
        const dist = Math.sqrt(dx * dx + dy * dy);
        
        // Allow clicking anywhere to start aiming (more forgiving)
        aimStart = { x: mx, y: my };
        mousePos = { x: mx, y: my };
    }
});
