function dist(x1,y1,x2,y2) { return Math.sqrt((x2-x1)**2+(y2-y1)**2); }

// Enemy class
class Enemy {
    constructor(hp, speed, type) {
        this.pathPixels = getPathPixels();
        this.hp = hp; this.maxHp = hp;
        this.speed = speed;
        this.type = type || 'normal'; // normal, fast, tank
        this.progress = 0;
        this.alive = true;
        this.slowTimer = 0;
        this.size = type === 'tank' ? cellSize*0.35 : cellSize*0.25;
    }
    update(dt) {
        if (!this.alive) return;
        let spd = this.speed * (this.slowTimer > 0 ? 0.5 : 1);
        if (this.slowTimer > 0) this.slowTimer -= dt;
        this.progress += spd * dt * cellSize;
        // Move along path
        let totalDist = 0;
        for (let i = 1; i < this.pathPixels.length; i++) {
            totalDist += dist(this.pathPixels[i-1].x, this.pathPixels[i-1].y, this.pathPixels[i].x, this.pathPixels[i].y);
        }
        if (this.progress >= totalDist) {
            this.alive = false;
            lives--;
            return;
        }
        let d = 0;
        for (let i = 1; i < this.pathPixels.length; i++) {
            let segLen = dist(this.pathPixels[i-1].x, this.pathPixels[i-1].y, this.pathPixels[i].x, this.pathPixels[i].y);
            if (d + segLen >= this.progress) {
                let t = (this.progress - d) / segLen;
                this.x = this.pathPixels[i-1].x + (this.pathPixels[i].x - this.pathPixels[i-1].x) * t;
                this.y = this.pathPixels[i-1].y + (this.pathPixels[i].y - this.pathPixels[i-1].y) * t;
                break;
            }
            d += segLen;
        }
        if (this.x === undefined) { this.x = this.pathPixels[0].x; this.y = this.pathPixels[0].y; }
    }
    draw() {
        if (!this.alive) return;
        ctx.save();
        ctx.translate(this.x, this.y);
        // Body
        let color = this.type === 'tank' ? '#9b59b6' : this.type === 'fast' ? '#f39c12' : '#e74c3c';
        ctx.fillStyle = color;
        ctx.beginPath();
        ctx.arc(0, 0, this.size, 0, Math.PI*2);

        ctx.fillRect(-bw/2, -this.size-10, bw * (this.hp/this.maxHp), 5);
        // Slow effect

            ctx.strokeStyle = 'rgba(168,230,207,0.6)';
            ctx.lineWidth = 2;
            ctx.beginPath();
            ctx.arc(0, 0, this.size+4, 0, Math.PI*2);
            ctx.stroke();

    }
}

// Tower class
class Tower {
    constructor(gx, gy, type) {
        this.gx = gx; this.gy = gy;
        let p = getPixelPos(gx, gy);
        this.x = p.x; this.y = p.y;
        this.type = type;
        this.tdata = towerTypes[type];
        this.cooldown = 0;
        this.angle = 0;
    }
    update(dt) {
        this.cooldown -= dt;
        if (this.cooldown <= 0) {
            // Find target
            let best = null, bestProg = -1;
            for (let e of enemies) {
                if (!e.alive) continue;
                let d = dist(this.x, this.y, e.x, e.y);
                if (d <= this.tdata.range * cellSize && e.progress > bestProg) {
                    best = e; bestProg = e.progress;
                }
            }
            if (best) {
                this.angle = Math.atan2(best.y - this.y, best.x - this.x);
                let proj = new Projectile(this.x, this.y, best, this.tdata);
                projectiles.push(proj);
                this.cooldown = this.tdata.fireRate;
                playSound('shoot');
            }
        }
    }
    draw() {
        ctx.save();
        ctx.translate(this.x, this.y);
        // Base
        ctx.fillStyle = '#2d3436';
        ctx.beginPath();
        ctx.arc(0, 0, cellSize*0.35, 0, Math.PI*2);
        ctx.fill();

            ctx.stroke();
        }
    }
}

// Projectile class
class Projectile {
    constructor(x, y, target, tdata) {
        this.x = x; this.y = y;
        this.target = target;
        this.tdata = tdata;
        this.speed = tdata.projSpeed * cellSize;
        this.alive = true;
    }
    update(dt) {
        if (!this.target || !this.target.alive) { this.alive = false; return; }
        let dx = this.target.x - this.x, dy = this.target.y - this.y;
        let d = Math.sqrt(dx*dx+dy*dy);
        if (d < cellSize * 0.2) {
            this.hit();
            return;
        }
        this.x += (dx/d) * this.speed * dt;
        this.y += (dy/d) * this.speed * dt;
    }
    hit() {
        this.alive = false;
        if (this.tdata.splash > 0) {
            for (let e of enemies) {
                if (!e.alive) continue;
                if (dist(this.x, this.y, e.x, e.y) < cellSize * this.tdata.splash) {
                    e.hp -= this.tdata.damage;
                    if (this.tdata.slow) e.slowTimer = 2;
                    if (e.hp <= 0) { killEnemy(e); }
                }
            }
        } else {
            this.target.hp -= this.tdata.damage;
            if (this.tdata.slow) this.target.slowTimer = 2;
            if (this.target.hp <= 0 && this.target.alive) killEnemy(this.target);
        }
        playSound('hit');
        // Particles
        for (let i = 0; i < 4; i++) {
            particles.push({ x: this.x, y: this.y, vx: (Math.random()-0.5)*100, vy: (Math.random()-0.5)*100, life: 0.3, color: this.tdata.color });
        }
    }
    draw() {
        ctx.fillStyle = this.tdata.color;
        ctx.beginPath();
        ctx.arc(this.x, this.y, cellSize*0.08, 0, Math.PI*2);
        ctx.fill();
    }
}

function killEnemy(e) {
    e.alive = false;
    money += e.type === 'tank' ? 30 : e.type === 'fast' ? 15 : 20;
    playSound('kill');
    for (let i = 0; i < 8; i++) {
        particles.push({ x: e.x, y: e.y, vx: (Math.random()-0.5)*150, vy: (Math.random()-0.5)*150, life: 0.4, color: '#fff' });
    }
}

function startWave() {
    wave++;
    waveActive = true;
    playSound('wave');
    let count = 3 + wave * 2;
    spawnQueue = [];
    for (let i = 0; i < count; i++) {
        let type = 'normal';
        if (wave >= 3 && i % 4 === 2) type = 'fast';
        if (wave >= 5 && i % 5 === 4) type = 'tank';
        let hp = (20 + wave * 10) * (type === 'tank' ? 3 : type === 'fast' ? 0.7 : 1);
        let speed = (1.5 + wave * 0.1) * (type === 'fast' ? 1.8 : type === 'tank' ? 0.6 : 1);
        spawnQueue.push({ hp: Math.floor(hp), speed, type });
    }
    spawnTimer = 0;
}

function updateGame(dt) {
    if (lives <= 0) { gameState = 'gameover'; return; }

    // Wave spawning
    if (!waveActive && nextWaveIn > 0) {
        nextWaveIn -= dt;
        if (nextWaveIn <= 0) startWave();
    }
    if (waveActive) {
        if (spawnQueue.length > 0) {
            spawnTimer -= dt;
            if (spawnTimer <= 0) {
                let s = spawnQueue.shift();
                enemies.push(new Enemy(s.hp, s.speed, s.type));
                spawnTimer = Math.max(0.4, 1.2 - wave * 0.05);
            }
        } else if (enemies.every(e => !e.alive)) {
            waveActive = false;
            nextWaveIn = 3;
            money += 30 + wave * 5;
        }
    }

    // Update enemies
    for (let e of enemies) e.update(dt);
    enemies = enemies.filter(e => e.alive);

    // Update towers
    for (let t of towers) t.update(dt);

    // Update projectiles
    for (let p of projectiles) p.update(dt);
    projectiles = projectiles.filter(p => p.alive);

    // Update particles
    for (let p of particles) { p.x += p.vx*dt; p.y += p.vy*dt; p.life -= dt; }
    particles = particles.filter(p => p.life > 0);
}

    ctx.textAlign = 'center';
    ctx.fillStyle = '#e74c3c';
    ctx.font = `bold ${Math.floor(Math.min(W,H)*0.08)}px monospace`;
    ctx.fillText('GAME OVER', W/2, H*0.4);
    ctx.fillStyle = '#fff';
    ctx.font = `${Math.floor(Math.min(W,H)*0.04)}px monospace`;
    ctx.fillText(`You survived ${wave} waves!`, W/2, H*0.52);
    ctx.fillStyle = '#aaa';
    ctx.font = `${Math.floor(Math.min(W,H)*0.03)}px monospace`;
    let pulse = 0.5 + Math.sin(Date.now()*0.003)*0.5;
    ctx.globalAlpha = pulse;
    ctx.fillText('Press R or SPACE to restart', W/2, H*0.65);
    ctx.globalAlpha = 1;
}

function resetGame() {
    money = 150; lives = 20; wave = 0;
    enemies = []; towers = []; projectiles = []; particles = [];
    spawnQueue = []; waveActive = false; nextWaveIn = 3;
    gameState = 'playing';
}

// Input
let mouseX = 0, mouseY = 0;
window.addEventListener('mousemove', e => { mouseX = e.clientX; mouseY = e.clientY; });

window.addEventListener('click', e => {
    initAudio();
    if (gameState === 'start') { resetGame(); return; }
    if (gameState === 'gameover') { resetGame(); return; }
    if (gameState !== 'playing') return;

    // Check tower spots
    for (let s of towerSpots) {
        let p = getPixelPos(s.x, s.y);
        if (dist(e.clientX, e.clientY, p.x, p.y) < cellSize * 0.4) {
            let occupied = towers.some(t => t.gx === s.x && t.gy === s.y);
            if (!occupied && money >= towerTypes[selectedTower].cost) {
                money -= towerTypes[selectedTower].cost;
                towers.push(new Tower(s.x, s.y, selectedTower));
                playSound('place');
            }
        }
    }
});

window.addEventListener('keydown', e => {
    initAudio();
    if (e.key === '1') selectedTower = 0;
    else if (e.key === '2') selectedTower = 1;
    else if (e.key === '3') selectedTower = 2;
    else if (e.key === ' ') { e.preventDefault(); if (gameState === 'start' || gameState === 'gameover') resetGame(); }
    else if (e.key === 'r' || e.key === 'R') resetGame();
});

// Main loop
let lastTime = 0;
function gameLoop(time) {
    let dt = Math.min((time - lastTime) / 1000, 0.05);
    lastTime = time;

    if (gameState === 'start') drawStartScreen();
    else if (gameState === 'playing') { updateGame(dt); drawGrid(); }
    else if (gameState === 'gameover') { drawGrid(); drawGameOver(); }

    requestAnimationFrame(gameLoop);
}
requestAnimationFrame(gameLoop);
