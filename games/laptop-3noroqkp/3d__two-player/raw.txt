// Audio
let audioCtx = null;
function initAudio() {
    if (!audioCtx) audioCtx = new (window.AudioContext || webkitAudioContext)();
}
function playSound(freq, dur, type) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = type || 'square';
    osc.frequency.value = freq;
    gain.gain.setValueAtTime(0.15, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + dur);
}

const camX = 0, camY = 18, camZ = -14, camDist = 28;
const FOV = 500;

function project(x, y, z) {
    let dx = x - camX;
    let dy = y - camY;
    let dz = z - camZ;
    let scale = FOV / (dz + camDist);
    return {
        sx: W / 2 + dx * scale,
        sy: H / 2 - dy * scale,
        scale: scale
    };
}

const GRAVITY = -18;
const MOVE_SPEED = 6;
const DASH_FORCE = 7;
const FRICTION = 0.85;
const JUMP_VEL = 9;

function updatePlayer(p, up, down, left, right, jump, dash) {
    if (!p.alive) return;

    // Movement
    let ax = 0, az = 0;
    if (keys[up]) az += MOVE_SPEED;
    if (keys[down]) az -= MOVE_SPEED;
    if (keys[left]) ax -= MOVE_SPEED;
    if (keys[right]) ax += MOVE_SPEED;

    p.vx += ax * 0.15;
    p.vz += az * 0.15;

    // Jump
    if (keys[jump] && p.onGround) {
        p.vy = JUMP_VEL;
        p.onGround = false;
        playSound(200, 0.1, 'sine');
    }

    // Dash/Push
    if (keys[dash] && p.dashCooldown <= 0) {
        let dx = p2.x - p1.x, dz = p2.z - p1.z;
        let len = Math.sqrt(dx * dx + dz * dz);
        if (len > 0.01) {
            dx /= len; dz /= len;
        } else { dx = 0; dz = 1; }
        p.vx += dx * DASH_FORCE;
        p.vz += dz * DASH_FORCE;
        p.dashCooldown = 25;
        p.dashTimer = 8;
        playSound(400, 0.15, 'sawtooth');
        for (let i = 0; i < 8; i++) {
            particles.push({
                x: p.x + dx * 0.5, y: p.y + 0.5, z: p.z + dz * 0.5,
                vx: dx * (2 + Math.random() * 2), vy: Math.random() * 3, vz: dz * (2 + Math.random() * 2),
                life: 30, color: p.color
            });
        }
    }

    // Friction
    p.vx *= FRICTION;
    p.vz *= FRICTION;
    if (p.dashCooldown > 0) p.dashCooldown--;

    // Gravity
    p.vy += GRAVITY * 0.016;
    p.x += p.vx * 0.16;
    p.y += p.vy * 0.16;
    p.z += p.vz * 0.16;

    // Ground
    if (p.y <= 0) {
        p.y = 0;
        p.vy = 0;
        p.onGround = true;
    }

    // Platform bounds - check if falling off
    let halfW = PLAT_SIZE + 1;
    if (Math.abs(p.x) > halfW || Math.abs(p.z) > halfW) {
        p.alive = false;
        p.fallAnim = 0;
        playSound(150, 0.4, 'sawtooth');
        for (let i = 0; i < 20; i++) {
            particles.push({
                x: p.x, y: p.y + Math.random(), z: p.z,
                vx: (Math.random() - 0.5) * 4, vy: Math.random() * 5, vz: (Math.random() - 0.5) * 4,
                life: 60, color: p.color
            });
        }
    }

    // Collision with other player (push apart)
    let other = p.id === 1 ? p2 : p1;
    if (other.alive) {
        let dx = p.x - other.x, dz = p.z - other.z;
        let dist = Math.sqrt(dx * dx + dz * dz);
        let minDist = p.w;
        if (dist < minDist && dist > 0.01) {
            let nx = dx / dist, nz = dz / dist;
            let overlap = minDist - dist;
            p.x += nx * overlap * 0.5;
            p.z += nz * overlap * 0.5;
            other.x -= nx * overlap * 0.5;
            other.z -= nz * overlap * 0.5;
        }
    }

    if (p.dashTimer > 0) p.dashTimer--;
}

function updateParticles() {
    for (let i = particles.length - 1; i >= 0; i--) {
        let pt = particles[i];
        pt.x += pt.vx * 0.05;
        pt.y += pt.vy * 0.05;
        pt.z += pt.vz * 0.05;
        pt.vy -= 0.3;
        pt.life--;
        if (pt.life <= 0) particles.splice(i, 1);
    }
}

function drawPlatform() {
    // Draw platform as a series of quads
    let gridStep = 2;
    for (let gz = -PLAT_SIZE; gz < PLAT_SIZE; gz += gridStep) {
        for (let gx = -PLAT_SIZE; gx < PLAT_SIZE; gx += gridStep) {
            let x1 = gx, z1 = gz, x2 = gx + gridStep, z2 = gz + gridStep;
            let c1 = project(x1, 0, z1);
            let c2 = project(x2, 0, z1);
            let c3 = project(x2, 0, z2);
            let c4 = project(x1, 0, z2);

            // Depth-based color
            let depth = (gz + PLAT_SIZE) / (PLAT_SIZE * 2);
            let r = Math.floor(30 + depth * 20);
            let g = Math.floor(35 + depth * 15);
            let b = Math.floor(50 + depth * 25);

            ctx.fillStyle = `rgb(${r},${g},${b})`;
            ctx.beginPath();
            ctx.moveTo(c1.sx, c1.sy);
            ctx.lineTo(c2.sx, c2.sy);
            ctx.lineTo(c3.sx, c3.sy);
            ctx.lineTo(c4.sx, c4.sy);
            ctx.closePath();
            ctx.fill();

            // Grid lines
            ctx.strokeStyle = 'rgba(100,140,255,0.15)';
            ctx.lineWidth = 1;
            ctx.stroke();
        }
    }

    // Platform edges (glowing border)
    let corners = [
        project(-PLAT_SIZE, 0, -PLAT_SIZE),
        project(PLAT_SIZE, 0, -PLAT_SIZE),
        project(PLAT_SIZE, 0, PLAT_SIZE),
        project(-PLAT_SIZE, 0, PLAT_SIZE)
    ];
    ctx.strokeStyle = 'rgba(100,180,255,0.6)';
    ctx.lineWidth = 3;
    ctx.beginPath();
    ctx.moveTo(corners[0].sx, corners[0].sy);

    ctx.closePath();
    ctx.stroke();

    // Platform sides (thickness)
    let platHeight = -2;
    let edges = [
        [[-PLAT_SIZE, -PLAT_SIZE], [PLAT_SIZE, -PLAT_SIZE]],
        [[PLAT_SIZE, -PLAT_SIZE], [PLAT_SIZE, PLAT_SIZE]],
        [[PLAT_SIZE, PLAT_SIZE], [-PLAT_SIZE, PLAT_SIZE]],
        [[-PLAT_SIZE, PLAT_SIZE], [-PLAT_SIZE, -PLAT_SIZE]]
    ];
    for (let e of edges) {
        let t1 = project(e[0][0], 0, e[0][1]);
        let t2 = project(e[1][0], 0, e[1][1]);
        let b1 = project(e[0][0], platHeight, e[0][1]);
        let b2 = project(e[1][0], platHeight, e[1][1]);
        ctx.fillStyle = 'rgba(20,25,40,0.8)';

        ctx.moveTo(t1.sx, t1.sy);
        ctx.lineTo(t2.sx, t2.sy);
        ctx.lineTo(b2.sx, b2.sy);
        ctx.lineTo(b1.sx, b1.sy);
        ctx.closePath();
        ctx.fill();
    }
}

function drawBox(x, y, z, w, h, d, color, glowColor) {
    let hw = w / 2, hh = h / 2, hd = d / 2;
    // 8 corners
    let verts = [
        project(x - hw, y + h, z - hd),
        project(x + hw, y + h, z - hd),
        project(x + hw, y + h, z + hd),
        project(x - hw, y + h, z + hd),
        project(x - hw, y, z - hd),
        project(x + hw, y, z - hd),
        project(x + hw, y, z + hd),
        project(x - hw, y, z + hd)
    ];

    // Faces: [top, bottom, front, back, left, right]
    let faces = [
        { idx: [0,1,2,3], shade: 1.0 },   // top
        { idx: [4,5,6,7], shade: 0.4 },   // bottom
        { idx: [4,7,3,0], shade: 0.7 },   // front (z-)
        { idx: [5,1,2,6], shade: 0.7 },   // back (z+)
        { idx: [4,0,3,7], shade: 0.85 },  // left (x-)
        { idx: [5,6,2,1], shade: 0.85 }   // right (x+)
    ];

    // Sort faces by average depth
    faces.sort((a, b) => {
        let da = a.idx.reduce((s, i) => s + verts[i].scale, 0);
        let db = b.idx.reduce((s, i) => s + verts[i].scale, 0);
        return da - db;
    });

    // Parse base color
    let r = parseInt(color.slice(1,3), 16), g = parseInt(color.slice(3,5), 16), b = parseInt(color.slice(5,7), 16);

    for (let f of faces) {
        ctx.fillStyle = `rgb(${Math.floor(r * f.shade)},${Math.floor(g * f.shade)},${Math.floor(b * f.shade)})`;
        ctx.beginPath();
        ctx.moveTo(verts[f.idx[0]].sx, verts[f.idx[0]].sy);
        for (let i = 1; i < 4; i++) ctx.lineTo(verts[f.idx[i]].sx, verts[f.idx[i]].sy);
        ctx.closePath();
        ctx.fill();
    }

    // Glow
    let center = project(x, y + h/2, z);
    let glowR = center.scale * 1.5;
    let grad = ctx.createRadialGradient(center.sx, center.sy, 0, center.sx, center.sy, glowR);
    grad.addColorStop(0, glowColor);
    grad.addColorStop(1, 'transparent');
    ctx.fillStyle = grad;
    ctx.beginPath();
    ctx.arc(center.sx, center.sy, glowR, 0, Math.PI * 2);
    ctx.fill();

    // Shadow on platform
    let shadow = project(x, 0, z);
    let shR = shadow.scale * hw;
    ctx.fillStyle = 'rgba(0,0,0,0.3)';
    ctx.beginPath();
    ctx.ellipse(shadow.sx, shadow.sy, shR, shR * 0.4, 0, 0, Math.PI * 2);
    ctx.fill();
}

function drawPlayer(p) {
    if (!p.alive) return;
    let squash = p.onGround ? 1 : 0.85 + (p.vy > 0 ? 0.15 : -0.1);
    let w = p.w * (2 - squash);
    let h = p.h * squash;

    drawBox(p.x, p.y, p.z, w, h, p.d, p.color, p.glowColor);

    // Face (simple eyes)
    let faceY = p.y + h * 0.75;
    let eyeL = project(p.x - 0.25, faceY, p.z - p.d/2 - 0.1);
    let eyeR = project(p.x + 0.25, faceY, p.z - p.d/2 - 0.1);
    ctx.fillStyle = '#fff';
    let er = Math.max(2, eyeL.scale * 0.15);
    ctx.beginPath(); ctx.arc(eyeL.sx, eyeL.sy, er, 0, Math.PI*2); ctx.fill();
    ctx.beginPath(); ctx.arc(eyeR.sx, eyeR.sy, er, 0, Math.PI*2); ctx.fill();

    // Dash effect
    if (p.dashTimer > 0) {
        let c = project(p.x, p.y + h/2, p.z);
        for (let i = 0; i < 3; i++) {
            ctx.strokeStyle = p.glowColor;
            ctx.lineWidth = 2;
            let r = (8 - p.dashTimer) * c.scale * 0.5 + er;
            ctx.beginPath();
            ctx.arc(c.sx, c.sy, r, 0, Math.PI * 2);
            ctx.stroke();
        }
    }
}

function drawParticles() {
    for (let pt of particles) {
        let p = project(pt.x, pt.y, pt.z);
        let alpha = pt.life / 60;
        let r = Math.max(1, p.scale * 0.2 * alpha);
        ctx.fillStyle = pt.color;
        ctx.globalAlpha = alpha;
        ctx.beginPath();
        ctx.arc(p.sx, p.sy, r, 0, Math.PI * 2);
        ctx.fill();
    }
    ctx.globalAlpha = 1;
}

}

let lastTime = 0;
function gameLoop(time) {
    let dt = Math.min((time - lastTime) / 16.67, 3);
    lastTime = time;

    if (gameState === 'playing') {
        updatePlayer(p1, 'KeyW', 'KeyS', 'KeyA', 'KeyD', 'Space', 'KeyF');
        updatePlayer(p2, 'ArrowUp', 'ArrowDown', 'ArrowLeft', 'ArrowRight', 'Enter', 'Slash');
        updateParticles();

        // Check for round end
        if (!p1.alive || !p2.alive) {
            if (p1.alive && !p2.alive) score1++;
            else if (!p1.alive && p2.alive) score2++;
            else { /* both fell - no score */ }

            if (score1 >= WINS_NEEDED || score2 >= WINS_NEEDED) {
                winner = score1 >= WINS_NEEDED ? 1 : 2;
                gameState = 'gameOver';
            } else {
                gameState = 'roundEnd';
            }
        }
    }

    render();
    requestAnimationFrame(gameLoop);
}
