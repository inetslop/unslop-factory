(function() {
const canvas = document.getElementById('game');
const ctx = canvas.getContext('2d');

// Audio
let audioCtx = null;
function initAudio() {
if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
}
function playSound(freq, dur, type, vol) {
if (!audioCtx) return;
const osc = audioCtx.createOscillator();
const gain = audioCtx.createGain();
osc.type = type || 'square';
osc.frequency.value = freq;
gain.gain.setValueAtTime(vol || 0.15, audioCtx.currentTime);
gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
osc.connect(gain); gain.connect(audioCtx.destination);
osc.start(); osc.stop(audioCtx.currentTime + dur);
}
function sfxJump() { playSound(280, 0.15, 'square', 0.1); }
function sfxCoin() { playSound(880, 0.1, 'sine', 0.12); setTimeout(() => playSound(1320, 0.15, 'sine', 0.1), 60); }
function sfxBuy() { playSound(440, 0.1, 'triangle', 0.15); setTimeout(() => playSound(660, 0.12, 'triangle', 0.12), 80); }
function sfxClick() { playSound(600, 0.05, 'sine', 0.08); }

// Game State
let gameState = 'start'; // start, playing, gameover
let keys = {};
let lastTime = 0;
let gameTime = 0;

// Player
const player = {
x: 0, y: 0, vx: 0, vy: 0, w: 28, h: 36,
onGround: false, jumps: 0, maxJumps: 1,
speed: 4, jumpPower: -12, gravity: 0.5
};

// Upgrades
const upgrades = {
jump: { level: 0, cost: 50, name: 'Jump Power', desc: '+1 jump force' },
speed: { level: 0, cost: 80, name: 'Speed', desc: '+0.5 move speed' },
autoClick: { level: 0, cost: 120, name: 'Auto Mine', desc: '+1 coin/sec auto' },
multiplier: { level: 0, cost: 200, name: 'Multiplier', desc: 'x1.5 all coins' },
doubleJump: { level: 0, max: 1, cost: 500, name: 'Double Jump', desc: 'Jump in air!' }
};

let coins = 0;
let totalEarned = 0;
let coinsPerSecond = 0;
let mineClickPower = 1;

// Platforms
let platforms = [];
function generatePlatforms() {
platforms = [];
const pw = W * 0.7;
const px = (W - pw) / 2;
// Ground
platforms.push({ x: px, y: H - 40, w: pw, h: 20 });
// Floating platforms
for (let i = 0; i < 6; i++) {
const py = H - 120 - i * 85;
const pw2 = Math.min(120 + Math.random() * 100, W * 0.3);
const px2 = Math.random() * (W - pw2);
platforms.push({ x: px2, y: py, w: pw2, h: 16 });
}
// Side platforms
for (let i = 0; i < 4; i++) {
const py = H - 150 - i * 120;
const side = Math.random() > 0.5 ? 0 : W - 100;
platforms.push({ x: side, y: py, w: 80 + Math.random() * 40, h: 16 });
}
}

// Floating coins (clickable)
let floatingCoins = [];
let coinSpawnTimer = 0;
function spawnCoin() {
const c = {
x: 50 + Math.random() * (W - 100),
y: 80 + Math.random() * (H - 250),
r: 14, life: 6, value: 5 + Math.floor(Math.random() * 10),
vx: (Math.random() - 0.5) * 0.5, vy: 0, collected: false
};
floatingCoins.push(c);
}

// Particles
let particles = [];
function addParticle(x, y, color, count) {
for (let i = 0; i < count; i++) {
particles.push({ x, y, vx: (Math.random()-0.5)*6, vy: (Math.random()-0.5)*6 - 2, life: 1, color });
}
}

// Floating text
let floatTexts = [];
function addFloatText(x, y, text, color) {
floatTexts.push({ x, y, text, color: color || '#FFD700', life: 1 });
}

// Mine button
const mineBtn = { x: 0, y: 0, w: 160, h: 60 };

function resetGame() {
player.x = W / 2 - player.w / 2;
player.y = H - 80;
player.vx = 0; player.vy = 0;
player.onGround = false; player.jumps = 0;
coins = 0; totalEarned = 0; coinsPerSecond = 0; mineClickPower = 1;
floatingCoins = []; particles = []; floatTexts = [];
coinSpawnTimer = 0; gameTime = 0;
for (let k in upgrades) upgrades[k].level = 0;
generatePlatforms();
}

// Input
window.addEventListener('keydown', e => {
keys[e.code] = true;
if (gameState === 'start' && (e.code === 'Space' || e.code === 'Enter')) {
initAudio(); gameState = 'playing'; resetGame();
}
if (gameState === 'playing' && e.code === 'KeyR') { resetGame(); }
});
window.addEventListener('keyup', e => { keys[e.code] = false; });

canvas.addEventListener('click', e => {
const rect = canvas.getBoundingClientRect();
const mx = e.clientX - rect.left;
const my = e.clientY - rect.top;

if (gameState === 'start') {
initAudio(); gameState = 'playing'; resetGame(); return;
}
if (gameState !== 'playing') return;

// Check mine button
if (mx >= mineBtn.x && mx <= mineBtn.x + mineBtn.w && my >= mineBtn.y && my <= mineBtn.y + mineBtn.h) {
const earned = Math.floor(mineClickPower * getMultiplier());
coins += earned; totalEarned += earned;
addFloatText(mx, my - 20, '+' + earned);
addParticle(mx, my, '#FFD700', 4);
sfxClick();
return;
}

// Check floating coins
for (let i = floatingCoins.length - 1; i >= 0; i--) {
const c = floatingCoins[i];
const dx = mx - c.x, dy = my - c.y;
if (dx*dx + dy*dy < (c.r+10)*(c.r+10)) {
const val = Math.floor(c.value * getMultiplier());
coins += val; totalEarned += val;
addFloatText(c.x, c.y, '+' + val);
addParticle(c.x, c.y, '#FFD700', 6);
sfxCoin();
floatingCoins.splice(i, 1);
}
}

// Check upgrade buttons
for (let i = 0; i < upgradeButtons.length; i++) {
const b = upgradeButtons[i];
if (mx >= b.x && mx <= b.x + b.w && my >= b.y && my <= b.y + b.h) {
buyUpgrade(i);
}
}
});

function getMultiplier() {
return Math.pow(1.5, upgrades.multiplier.level);
}

// Upgrade buttons (recalculated each frame for layout)
let upgradeButtons = [];

function buyUpgrade(idx) {
const keysArr = Object.keys(upgrades);
const key = keysArr[idx];
const up = upgrades[key];
if (up.max && up.level >= up.max) return;
if (coins < getUpgradeCost(key)) return;
coins -= getUpgradeCost(key);
up.level++;
sfxBuy();
addFloatText(W/2, H/2 - 50, 'UPGRADED!', '#4FC3F7');

// Apply upgrades
player.jumpPower = -(12 + up.jump ? 0 : 0) - (12 + upgrades.jump.level * 2);
player.speed = 4 + upgrades.speed.level * 0.5;
player.maxJumps = 1 + upgrades.doubleJump.level;
coinsPerSecond = upgrades.autoClick.level * 2;
}

function getUpgradeCost(key) {
const up = upgrades[key];
return Math.floor(up.cost * Math.pow(1.6, up.level));
}

// Main loop
function update(dt) {
if (gameState !== 'playing') return;
gameTime += dt;

// Auto income
if (coinsPerSecond > 0) {
const autoEarn = coinsPerSecond * getMultiplier() * dt;
coins += autoEarn; totalEarned += autoEarn;
}

// Player movement
let moveX = 0;
if (keys['ArrowLeft'] || keys['KeyA']) moveX = -1;
if (keys['ArrowRight'] || keys['KeyD']) moveX = 1;
player.vx = moveX * player.speed;

// Jump
if ((keys['ArrowUp'] || keys['KeyW'] || keys['Space']) && player.jumps < player.maxJumps) {
if (!player._jumpHeld) {
player.vy = player.jumpPower;
player.jumps++;
player.onGround = false;
sfxJump();
player._jumpHeld = true;
}
} else if (keys['ArrowUp'] || keys['KeyW'] || keys['Space']) {
player._jumpHeld = true;
} else {
player._jumpHeld = false;
}

// Gravity
player.vy += player.gravity;
player.x += player.vx;
player.y += player.vy;

// Platform collision
player.onGround = false;
for (const p of platforms) {
if (player.x + player.w > p.x && player.x < p.x + p.w) {
if (player.vy >= 0 && player.y + player.h <= p.y + 10 && player.y + player.h + player.vy >= p.y) {
player.y = p.y - player.h;
player.vy = 0;
player.onGround = true;
player.jumps = 0;
}
}
}

// Walls
if (player.x < 0) player.x = 0;
if (player.x + player.w > W) player.x = W - player.w;
// Floor
if (player.y + player.h > H - 20) {
player.y = H - 20 - player.h;
player.vy = 0;
player.onGround = true;
player.jumps = 0;
}
// Ceiling
if (player.y < 0) { player.y = 0; player.vy = 0; }

// Spawn coins
coinSpawnTimer += dt;
if (coinSpawnTimer > 2 && floatingCoins.length < 8) {
coinSpawnTimer = 0;
spawnCoin();
}

// Update floating coins
for (let i = floatingCoins.length - 1; i >= 0; i--) {
const c = floatingCoins[i];
c.x += c.vx;
c.life -= dt;
if (c.life <= 0) { floatingCoins.splice(i, 1); continue; }

// Player touch collect
if (player.x + player.w > c.x - c.r && player.x < c.x + c.r &&
    player.y + player.h > c.y - c.r && player.y < c.y + c.r) {
const val = Math.floor(c.value * getMultiplier());
coins += val; totalEarned += val;
addFloatText(c.x, c.y, '+' + val);
addParticle(c.x, c.y, '#FFD700', 5);
sfxCoin();
floatingCoins.splice(i, 1);
}
}

// Particles
for (let i = particles.length - 1; i >= 0; i--) {
const p = particles[i];
p.x += p.vx; p.y += p.vy; p.vy += 0.3;
p.life -= dt * 2;
if (p.life <= 0) particles.splice(i, 1);
}

// Float texts
for (let i = floatTexts.length - 1; i >= 0; i--) {
floatTexts[i].y -= 40 * dt;
floatTexts[i].life -= dt * 1.5;
if (floatTexts[i].life <= 0) floatTexts.splice(i, 1);
}

// Update mine button position
mineBtn.x = W / 2 - mineBtn.w / 2;
mineBtn.y = H - mineBtn.h - 15;
}

function draw() {
// Background gradient
const grad = ctx.createLinearGradient(0, 0, 0, H);
grad.addColorStop(0, '#0f0c29');
grad.addColorStop(1, '#1a1a3e');
ctx.fillStyle = grad;
ctx.fillRect(0, 0, W, H);

// Stars
ctx.fillStyle = 'rgba(255,255,255,0.4)';
for (let i = 0; i < 50; i++) {
const sx = (i * 137 + 50) % W;
const sy = (i * 97 + 20) % (H * 0.6);
ctx.fillRect(sx, sy, 2, 2);
}

if (gameState === 'start') {
drawStartScreen();
return;
}

// Platforms
for (const p of platforms) {
ctx.fillStyle = '#4a90d9';
ctx.fillRect(p.x, p.y, p.w, p.h);
ctx.fillStyle = '#5ba3e8';
ctx.fillRect(p.x, p.y, p.w, 4);
}

// Floating coins
for (const c of floatingCoins) {
const alpha = Math.min(1, c.life / 1.5);
ctx.globalAlpha = alpha;
ctx.fillStyle = '#FFD700';
ctx.beginPath(); ctx.arc(c.x, c.y, c.r, 0, Math.PI*2); ctx.fill();
ctx.fillStyle = '#FFF8DC';
ctx.beginPath(); ctx.arc(c.x - 3, c.y - 3, c.r * 0.4, 0, Math.PI*2); ctx.fill();
ctx.fillStyle = '#FFA500';
ctx.font = 'bold 12px monospace';
ctx.textAlign = 'center';
ctx.fillText(c.value, c.x, c.y + 4);
ctx.globalAlpha = 1;
}

// Player
ctx.fillStyle = '#FF6B35';
ctx.fillRect(player.x, player.y, player.w, player.h);
// Eyes
ctx.fillStyle = '#FFF';
ctx.fillRect(player.x + 5, player.y + 8, 7, 7);
ctx.fillRect(player.x + 16, player.y + 8, 7, 7);
ctx.fillStyle = '#000';
const lookX = player.vx > 0 ? 4 : player.vx < 0 ? 2 : 3;
ctx.fillRect(player.x + 5 + lookX, player.y + 11, 3, 3);
ctx.fillRect(player.x + 16 + lookX, player.y + 11, 3, 3);
// Legs animation
const legOffset = Math.abs(Math.sin(gameTime * 10)) * (Math.abs(player.vx) > 0 ? 4 : 0);
ctx.fillStyle = '#CC4400';
ctx.fillRect(player.x + 4, player.y + player.h - 6, 8, 6 + legOffset);
ctx.fillRect(player.x + player.w - 12, player.y + player.h - 6, 8, 6 - legOffset);

// Particles
for (const p of particles) {
ctx.globalAlpha = p.life;
ctx.fillStyle = p.color;
ctx.fillRect(p.x - 3, p.y - 3, 6, 6);
}
ctx.globalAlpha = 1;

// Float texts
for (const ft of floatTexts) {
ctx.globalAlpha = Math.min(1, ft.life * 2);
ctx.fillStyle = ft.color;
ctx.font = 'bold 18px monospace';
ctx.textAlign = 'center';
ctx.fillText(ft.text, ft.x, ft.y);
}
ctx.globalAlpha = 1;

// Mine button
const mbGrad = ctx.createLinearGradient(mineBtn.x, mineBtn.y, mineBtn.x, mineBtn.y + mineBtn.h);
mbGrad.addColorStop(0, '#4a90d9');
mbGrad.addColorStop(1, '#2c5f8a');
ctx.fillStyle = mbGrad;
roundRect(ctx, mineBtn.x, mineBtn.y, mineBtn.w, mineBtn.h, 12);
ctx.fill();
ctx.strokeStyle = '#6ab4ff';
ctx.lineWidth = 2;
roundRect(ctx, mineBtn.x, mineBtn.y, mineBtn.w, mineBtn.h, 12);
ctx.stroke();
ctx.fillStyle = '#FFF';
ctx.font = 'bold 16px monospace';
ctx.textAlign = 'center';
ctx.fillText('⛏ MINE', W/2, mineBtn.y + 30);
ctx.font = '11px monospace';
ctx.fillText('+' + Math.floor(mineClickPower * getMultiplier()), W/2, mineBtn.y + 48);

// UI - Top bar
ctx.fillStyle = 'rgba(0,0,0,0.6)';
ctx.fillRect(0, 0, W, 50);
ctx.fillStyle = '#FFD700';
ctx.font = 'bold 20px monospace';
ctx.textAlign = 'left';
ctx.fillText('\u{1FA99} ' + Math.floor(coins), 15, 32);
ctx.fillStyle = '#aaa';
ctx.font = '13px monospace';
ctx.fillText('Total: ' + Math.floor(totalEarned) + ' | Auto: ' + coinsPerSecond * getMultiplier() + '/s', 180, 32);

// Upgrade panel (right side)
const panelX = W - 240;
const panelY = 60;
ctx.fillStyle = 'rgba(0,0,0,0.7)';
roundRect(ctx, panelX, panelY, 230, upgrades.length * 52 + 10, 8);
ctx.fill();

upgradeButtons = [];
const keysArr = Object.keys(upgrades);
for (let i = 0; i < keysArr.length; i++) {
const key = keysArr[i];
const up = upgrades[key];
const by = panelY + 5 + i * 52;
const bw = 220, bh = 48;
const bx = panelX + 5;
upgradeButtons.push({ x: bx, y: by, w: bw, h: bh });

const canBuy = coins >= getUpgradeCost(key) && (!up.max || up.level < up.max);
ctx.fillStyle = canBuy ? 'rgba(74,144,217,0.3)' : 'rgba(80,80,80,0.3)';
roundRect(ctx, bx, by, bw, bh, 6);
ctx.fill();

ctx.fillStyle = canBuy ? '#FFF' : '#888';
ctx.font = 'bold 12px monospace';
ctx.textAlign = 'left';
const maxed = up.max && up.level >= up.max;
ctx.fillText((maxed ? '\u2713 ' : '') + up.name + (up.max ? '' : ' Lv' + up.level), bx + 8, by + 16);
ctx.font = '10px monospace';
ctx.fillStyle = canBuy ? '#aaa' : '#666';
ctx.fillText(maxed ? 'MAXED' : '$' + getUpgradeCost(key) + ' - ' + up.desc, bx + 8, by + 32);
}

// Controls hint
ctx.fillStyle = 'rgba(0,0,0,0.5)';
ctx.fillRect(0, H - 25, W, 25);
ctx.fillStyle = '#888';
ctx.font = '11px monospace';
ctx.textAlign = 'center';
ctx.fillText('←→/AD: Move | ↑/W/Space: Jump | Click coins & mine! | R: Restart', W/2, H - 7);
}

function roundRect(ctx, x, y, w, h, r) {
ctx.beginPath();
ctx.moveTo(x + r, y);
ctx.lineTo(x + w - r, y); ctx.quadraticCurveTo(x+w, y, x+w, y+r);
ctx.lineTo(x+w, y+h-r); ctx.quadraticCurveTo(x+w, y+h, x+w-r, y+h);
ctx.lineTo(x+r, y+h); ctx.quadraticCurveTo(x, y+h, x, y+h-r);
ctx.lineTo(x, y+r); ctx.quadraticCurveTo(x, y, x+r, y);
ctx.closePath();
}

function drawStartScreen() {
ctx.fillStyle = '#FFF';
ctx.font = 'bold 48px monospace';
ctx.textAlign = 'center';
ctx.fillText('COIN CLIMB', W/2, H/3);
ctx.font = '18px monospace';
ctx.fillStyle = '#FFD700';
ctx.fillText('\u{1FA99} The Clicker Platformer \u{1FA99}', W/2, H/3 + 45);

ctx.fillStyle = '#ccc';
ctx.font = '16px monospace';
const lines = [
'Move: ← → or A D',
'Jump: ↑ / W / Space',
'',
'Click floating coins for bonus cash!',
'Spam the MINE button for steady income.',
'Buy upgrades to go further and faster.',
'',
'Collect coins. Climb high. Get rich.'
];
lines.forEach((l, i) => ctx.fillText(l, W/2, H/2 + 20 + i * 28));

ctx.fillStyle = '#4a90d9';
ctx.font = 'bold 22px monospace';
const pulse = Math.abs(Math.sin(gameTime * 3)) * 0.5 + 0.5;
ctx.globalAlpha = pulse;
ctx.fillText('[ CLICK TO START ]', W/2, H - 80);
ctx.globalAlpha = 1;
}

function loop(timestamp) {
const dt = Math.min((timestamp - lastTime) / 1000, 0.05);
lastTime = timestamp;
update(dt);
draw();
requestAnimationFrame(loop);
}

generatePlatforms();
requestAnimationFrame(loop);
})();
