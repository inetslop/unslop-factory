let audioCtx = null;
function initAudio() {
    if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();

function playSound(freq, type, duration, vol) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = type;
    osc.frequency.value = freq;
    gain.gain.setValueAtTime(vol || 0.15, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + duration);
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + duration);
}
function sfxJump() { playSound(400, 'square', 0.15, 0.1); }
function sfxCoin() { playSound(800, 'sine', 0.1, 0.12); playSound(1200, 'sine', 0.15, 0.08); }

function sfxWin() { playSound(600, 'sine', 0.2, 0.1); setTimeout(()=>playSound(800,'sine',0.2,0.1),150); setTimeout(()=>playSound(1000,'sine',0.3,0.1),300); }

let player = {};

function parseLevel(idx) {
    const lvl = levels[idx];
    const map = lvl.map;
    coins = [];
    enemies = [];
    walls = [];
    for (let y = 0; y < map.length; y++) {
        for (let x = 0; x < map[y].length; x++) {
            const c = map[y][x];
            if (c === 'C') coins.push({ x, y, collected: false, bob: Math.random() * Math.PI * 2 });
            if (c === 'E') enemies.push({ x, y, dir: [1,0], speed: 0.8 + idx * 0.3, path: null });
        }
    }
    // Parse walls (##)
    for (let y = 0; y < map.length; y++) {
        for (let x = 0; x < map[y].length; x++) {
            if (map[y][x] === '#') walls.push({ x, y });
        }
    }
    // Setup enemy paths - they patrol in lines
    enemies.forEach(e => {
        let dx = e.x, dy = e.y;
        const path = [];
        while (dx >= 0 && dx < map[0].length) {
            if (map[dy][dx] === '.' || map[dy][dx] === 'E') path.push({ x: dx, y: dy });
            else break;
            dx += e.dir[0];
        }
        e.path = path;
        e.pi = 0;
    });

    player = { x: lvl.start.x, y: lvl.start.y, vx: 0, vy: 0, jumping: false, jumpY: 0 };
}

function startGame() {
    score = 0;
    lives = 3;
    level = 0;
    parseLevel(level);
    gameState = 'playing';

function respawn() {
    lives--;
    if (lives <= 0) { gameState = 'gameover'; return; }
    const lvl = levels[level];
    player.x = lvl.start.x;
    player.y = lvl.start.y;
    player.jumping = false;
    player.jumpY = 0;
    invincibleTimer = 2;
}

function tryJump() {
    if (player.jumping) return;
    const lvl = levels[level];
    const map = lvl.map;
    // Check if adjacent tile is a gap or wall - can jump over
    const dirs = [[0,-1],[0,1],[-1,0],[1,0]];
    let canJump = false;
    for (const d of dirs) {
        const nx = player.x + d[0], ny = player.y + d[1];
        if (ny >= 0 && ny < map.length && nx >= 0 && nx < map[0].length) {
            if (map[ny][nx] === '#' || map[ny][nx] === 'G') canJump = true;
        }
    }
    if (canJump) {
        player.jumping = true;
        player.jumpY = 0.5;
        sfxJump();
        spawnParticles(player.x, player.y, '#4fc3f7', 5);
    }
}

function spawnParticles(x, y, color, count) {
    for (let i = 0; i < count; i++) {
        particles.push({
            x: x + 0.5, y: y + 0.5,
            vx: (Math.random() - 0.5) * 3,
            vy: (Math.random() - 0.5) * 3,
            life: 0.5 + Math.random() * 0.3,
            color: color,
            size: 2 + Math.random() * 3
        });
    }
}

function isWall(x, y) {
    const lvl = levels[level];
    const map = lvl.map;
    if (y < 0 || y >= map.length || x < 0 || x >= map[0].length) return true;
    return map[y][x] === '#' || map[y][x] === 'G';
}

function isGap(x, y) {
    const lvl = levels[level];
    const map = lvl.map;
    if (y < 0 || y >= map.length || x < 0 || x >= map[0].length) return true;
    return map[y][x] === '#';
}

function update(dt) {
    if (gameState !== 'playing') return;
    
    time += dt;
    invincibleTimer = Math.max(0, invincibleTimer - dt);
    
    // Player movement
    let mx = 0, my = 0;
    if (keys['KeyW'] || keys['ArrowUp']) my = -1;
    if (keys['KeyS'] || keys['ArrowDown']) my = 1;
    if (keys['KeyA'] || keys['ArrowLeft']) mx = -1;
    if (keys['KeyD'] || keys['ArrowRight']) mx = 1;
    
    if (player.jumping) {
        player.jumpY -= dt * 3;
        if (player.jumpY <= 0) {
            player.jumpY = 0;
            player.jumping = false;
            sfxLand();
        }
    }
    
    const speed = 5;
    let nx = player.x + mx * speed * dt;
    let ny = player.y + my * speed * dt;
    
    if (player.jumping) {
        // Can move freely while jumping
        player.x = Math.max(0, Math.min(levels[level].map[0].length - 1, nx));
        player.y = Math.max(0, Math.min(levels[level].map.length - 1, ny));
    } else {
        if (!isWall(Math.round(nx), player.y) && !isGap(Math.round(nx), player.y)) {
            player.x = nx;
        }
        if (!isWall(player.x, Math.round(ny)) && !isGap(player.x, Math.round(ny))) {
            player.y = ny;
        }
    }
    
    // Clamp
    player.x = Math.max(0, Math.min(levels[level].map[0].length - 1, player.x));
    player.y = Math.max(0, Math.min(levels[level].map.length - 1, player.y));
    
    // Collect coins
    coins.forEach(c => {
        if (!c.collected) {
            const dx = player.x - c.x;
            const dy = player.y - c.y;
            if (Math.sqrt(dx*dx + dy*dy) < 0.6) {
                c.collected = true;
                score += 10;
                sfxCoin();
                spawnParticles(c.x, c.y, '#ffd700', 8);
            }
        }
    });
    
    // Update enemies
    enemies.forEach(e => {
        if (!e.path || e.path.length < 2) return;
        const t = time * e.speed;
        const idx = Math.floor(t) % e.path.length;
        const nextIdx = (idx + 1) % e.path.length;
        const frac = t - Math.floor(t);
        const p0 = e.path[idx];
        const p1 = e.path[nextIdx];
        e.x = p0.x + (p1.x - p0.x) * frac;
        e.y = p0.y + (p1.y - p0.y) * frac;
        
        // Check collision with player
        if (invincibleTimer <= 0) {
            const dx = player.x - e.x;
            const dy = player.y - e.y;
            if (Math.sqrt(dx*dx + dy*dy) < 0.5 && !player.jumping) {
                sfxHurt();
                spawnParticles(player.x, player.y, '#ff4444', 10);
                gameState = 'dead';
            }
        }
    });
    
    // Check finish
    const fin = levels[level].finish;
    const fdx = player.x - fin.x;
    const fdy = player.y - fin.y;
    if (Math.sqrt(fdx*fdx + fdy*fdy) < 0.5) {
        score += 50;
        sfxWin();
        level++;
        if (level >= levels.length) {
            gameState = 'win';
        } else {
            parseLevel(level);
        }
    }
    
    // Update particles
    particles.forEach(p => {
        p.x += p.vx * dt;
        p.y += p.vy * dt;
        p.life -= dt;
    });
    particles = particles.filter(p => p.life > 0);
}

function getTileSize() {
    const lvl = levels[level];
    const cols = lvl.map[0].length;
    const rows = lvl.map.length;
    return Math.min((W - 40) / cols, (H - 80) / rows);
}

function getOffset() {
    const lvl = levels[level];
    const ts = getTileSize();
    const cols = lvl.map[0].length;
    const rows = lvl.map.length;
    return {
        x: (W - cols * ts) / 2,
        y: (H - rows * ts) / 2 + 20
    };
}

    ctx.fillText('Reach the flag to advance levels', W/2, H/2 + 80);
    
    // Animated prompt
    const pulse = Math.sin(time * 3) * 0.3 + 0.7;
    ctx.globalAlpha = pulse;
    ctx.fillStyle = '#4fc3f7';
    ctx.font = '18px monospace';
    ctx.fillText('Press SPACE to Start', W/2, H/2 + 150);
    ctx.globalAlpha = 1;

    for (let i = 0; i < 20; i++) {
        const px = (Math.sin(time * 0.5 + i * 1.3) * 0.5 + 0.5) * W;
        const py = (Math.cos(time * 0.3 + i * 0.7) * 0.5 + 0.5) * H;
        ctx.fillStyle = `rgba(79,195,247,${0.1 + Math.sin(time + i) * 0.05})`;
        ctx.beginPath();
        ctx.arc(px, py, 3, 0, Math.PI * 2);
        ctx.fill();
    }
}

// Game Loop
let lastTime = 0;
function gameLoop(timestamp) {
    const dt = Math.min((timestamp - lastTime) / 1000, 0.05);
    lastTime = timestamp;
    
    if (gameState === 'playing') update(dt);
    else time += dt;
    
    render();
    requestAnimationFrame(gameLoop);
}
