// Audio
let audioCtx = null;
function initAudio() {
    if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();

function playSound(freq, duration, type, vol) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = type || 'square';
    osc.frequency.value = freq;
    gain.gain.setValueAtTime(vol || 0.1, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + (duration || 0.1));
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + (duration || 0.1));
}
function sBreak() { playSound(440, 0.08, 'square', 0.08); }
function sBounce() { playSound(220, 0.05, 'triangle', 0.06); }
function sWall() { playSound(330, 0.04, 'sine', 0.05); }
function sPaddle() { playSound(180, 0.06, 'triangle', 0.07); }
function sClear() { for(let i=0;i<3;i++) setTimeout(()=>playSound(440+i*220,0.1,'square',0.08),i*50); }
function sLose() { playSound(110, 0.5, 'sawtooth', 0.1); }

// Game constants
const COLS = 10;
const ROWS = 8;
const TETRO_COLORS = {
    I: '#00e5ff', O: '#ffe04d', T: '#b44dff',
    S: '#4dff7c', Z: '#ff4d6a', J: '#4d8bff', L: '#ff9f4d'
};

// Game state

let particles = [];

function cellSize() { return Math.min(W / (COLS + 2), H / (ROWS + 6)); }
function gridOffsetX() { return (W - COLS * cellSize()) / 2; }
function gridOffsetY() { return Math.max(30, H * 0.08); }

function initLevel(lvl) {
    bricks = [];
    let cs = cellSize();
    let ox = gridOffsetX();
    let oy = gridOffsetY();

    // Generate tetromino patterns
    let layout = generateLayout(lvl);
    for (let r = 0; r < ROWS; r++) {
        for (let c = 0; c < COLS; c++) {
            if (layout[r][c] !== null) {
                bricks.push({
                    col: c, row: r,
                    x: ox + c * cs, y: oy + r * cs,
                    w: cs, h: cs,
                    hp: Math.min(3, 1 + Math.floor(lvl / 2)),
                    maxHp: Math.min(3, 1 + Math.floor(lvl / 2)),
                    color: layout[r][c],
                    alive: true
                });
            }
        }
    }

    paddle.w = Math.max(80, cs * 2.5);
    paddle.h = Math.max(14, cs * 0.35);
    paddle.x = (W - paddle.w) / 2;
    paddle.y = H - Math.max(40, H * 0.06) - paddle.h;

    ball.r = Math.max(6, cs * 0.18);
    ball.speed = Math.min(W * 0.003, Math.max(5, W / 120)) * (1 + lvl * 0.08);
    resetBall();
}

function generateLayout(lvl) {
    let grid = Array.from({length: ROWS}, () => Array(COLS).fill(null));
    let pieces = ['I','O','T','S','Z','J','L'];

    // Place tetromino shapes in the grid
    let placements = [
        // I horizontal at top
        {type:'I', dir:'h', r:0, c:3},
        {type:'O', r:0, c:8},
        {type:'T', r:1, c:2},
        {type:'J', r:1, c:6},
        {type:'L', r:2, c:0},
        {type:'S', r:2, c:4},
        {type:'Z', r:3, c:7},
        {type:'T', r:3, c:1},
        {type:'I', dir:'v', r:0, c:9},
        {type:'O', r:4, c:2},
        {type:'L', r:4, c:5},
        {type:'S', r:5, c:8},
        {type:'J', r:5, c:1},
        {type:'Z', r:6, c:4},
        {type:'O', r:6, c:7},
        {type:'T', r:7, c:0},
    ];

    // Shuffle based on level
    let shuffled = [...placements];
    for (let i = shuffled.length - 1; i > 0; i--) {
        let j = Math.floor(Math.random() * (i + 1));
        [shuffled[i], shuffled[j]] = [shuffled[j], shuffled[i]];
    }

    let colorMap = {};
    let pIdx = 0;
    for (let r = 0; r < ROWS; r++) {
        for (let c = 0; c < COLS; c++) {
            if (shuffled[pIdx]) {
                let p = shuffled[pIdx];
                pIdx++;
                let cells = getTetrominoCells(p.type, p.dir, r, c);
                for (let cell of cells) {
                    if (cell.r >= 0 && cell.r < ROWS && cell.c >= 0 && cell.c < COLS) {
                        grid[cell.r][cell.c] = TETRO_COLORS[p.type];
                    }
                }
            }
        }
    }

    // Fill remaining with random colors if not full
    for (let r = 0; r < ROWS; r++) {
        for (let c = 0; c < COLS; c++) {
            if (!grid[r][c] && Math.random() < 0.3) {
                let col = pieces[Math.floor(Math.random() * pieces.length)];
                grid[r][c] = TETRO_COLORS[col];
            }
        }
    }

    return grid;
}

function getTetrominoCells(type, dir, r, c) {
    let cells = [];
    switch(type) {
        case 'I':
            if (dir === 'v') { for(let i=0;i<4;i++) cells.push({r:r+i,c:c}); }
            else { for(let i=0;i<4;i++) cells.push({r:r,c:c+i}); }
            break;
        case 'O':
            cells.push({r:r,c:c},{r:r,c:c+1},{r:r+1,c:c},{r:r+1,c:c+1});
            break;
        case 'T':
            cells.push({r:r,c:c},{r:r,c:c+1},{r:r,c:c+2},{r:r+1,c:c+1});
            break;
        case 'S':
            cells.push({r:r,c:c+1},{r:r,c:c+2},{r:r+1,c:c},{r:r+1,c:c+1});
            break;
        case 'Z':
            cells.push({r:r,c:c},{r:r,c:c+1},{r:r+1,c:c+1},{r:r+1,c:c+2});
            break;
        case 'J':
            cells.push({r:r,c:c},{r:r+1,c:c},{r:r+1,c:c+1},{r:r+1,c:c+2});
            break;
        case 'L':
            cells.push({r:r,c:c+2},{r:r+1,c:c},{r:r+1,c:c+1},{r:r+1,c:c+2});
            break;
    }
    return cells;
}

function resetBall() {
    ball.x = paddle.x + paddle.w / 2;
    ball.y = paddle.y - ball.r - 4;
    let angle = -Math.PI / 4 + Math.random() * Math.PI / 2;
    ball.dx = Math.cos(angle) * ball.speed;
    ball.dy = -Math.abs(Math.sin(angle)) * ball.speed;
}

// Particles
function spawnParticles(x, y, color, count) {
    for (let i = 0; i < count; i++) {
        let angle = Math.random() * Math.PI * 2;
        let spd = 1 + Math.random() * 3;
        particles.push({
            x: x, y: y,
            vx: Math.cos(angle) * spd,
            vy: Math.sin(angle) * spd - 1,
            life: 1,
            color: color,
            size: 2 + Math.random() * 3
        });
    }
}

function updateParticles(dt) {
    for (let p of particles) {
        p.x += p.vx;
        p.y += p.vy;
        p.vy += 0.15;
        p.life -= dt * 2;
    }
    particles = particles.filter(p => p.life > 0);
}

// Update
function update(dt) {
    if (state !== 'PLAYING' && state !== 'BALL_HELD') return;

    // Paddle movement
    let paddleSpeed = W * 0.7;
    if (keys['ArrowLeft'] || keys['KeyA']) paddle.x -= paddleSpeed * dt;
    if (keys['ArrowRight'] || keys['KeyD']) paddle.x += paddleSpeed * dt;
    paddle.x = Math.max(0, Math.min(W - paddle.w, paddle.x));

    if (state === 'BALL_HELD') {
        ball.x = paddle.x + paddle.w / 2;
        ball.y = paddle.y - ball.r - 4;
        return;
    }

    // Ball movement
    ball.x += ball.dx * dt * 60;
    ball.y += ball.dy * dt * 60;

    // Wall collisions
    if (ball.x - ball.r <= 0) { ball.x = ball.r; ball.dx = Math.abs(ball.dx); sWall(); }
    if (ball.x + ball.r >= W) { ball.x = W - ball.r; ball.dx = -Math.abs(ball.dx); sWall(); }
    if (ball.y - ball.r <= 0) { ball.y = ball.r; ball.dy = Math.abs(ball.dy); sWall(); }

    // Paddle collision
    if (ball.dy > 0 &&
        ball.y + ball.r >= paddle.y &&
        ball.y + ball.r <= paddle.y + paddle.h + 10 &&
        ball.x >= paddle.x &&
        ball.x <= paddle.x + paddle.w) {
        let hitPos = (ball.x - paddle.x) / paddle.w;
        let angle = (hitPos - 0.5) * Math.PI * 0.7;
        ball.dx = Math.sin(angle) * ball.speed;
        ball.dy = -Math.cos(angle) * ball.speed;
        ball.y = paddle.y - ball.r;
        sPaddle();
    }

    // Ball lost
    if (ball.y - ball.r > H) {
        lives--;
        sLose();
        if (lives <= 0) {
            state = 'GAME_OVER';
            try {
                if (score > highScore) {
                    highScore = score;
                    localStorage.setItem('breakoutTetrisHigh', String(highScore));
                }
            } catch(e){}
        } else {
            resetBall();
            state = 'BALL_HELD';
        }
        return;
    }

    // Brick collisions
    for (let b of bricks) {
        if (!b.alive) continue;
        if (ball.x + ball.r > b.x && ball.x - ball.r < b.x + b.w &&
            ball.y + ball.r > b.y && ball.y - ball.r < b.y + b.h) {

            // Determine bounce direction
            let overlapL = (ball.x + ball.r) - b.x;
            let overlapR = (b.x + b.w) - (ball.x - ball.r);
            let overlapT = (ball.y + ball.r) - b.y;
            let overlapB = (b.y + b.h) - (ball.y - ball.r);
            let minOverlap = Math.min(overlapL, overlapR, overlapT, overlapB);

            if (minOverlap === overlapT || minOverlap === overlapB) {
                ball.dy = -ball.dy;
            } else {
                ball.dx = -ball.dx;
            }

            b.hp--;
            sBreak();
            spawnParticles(ball.x, ball.y, b.color, 5);

            if (b.hp <= 0) {
                b.alive = false;
                score += 10 * level;
                spawnParticles(b.x + b.w/2, b.y + b.h/2, b.color, 10);
            } else {
                score += 5;
            }

            // Check level clear
            if (bricks.every(br => !br.alive)) {
                state = 'LEVEL_CLEAR';
                sClear();
                score += 100 * level;
            }
            break;
        }
    }
}

// Draw
function draw() {
    ctx.fillStyle = '#0a0a1a';
    ctx.fillRect(0, 0, W, H);

    // Subtle grid background
    ctx.strokeStyle = 'rgba(255,255,255,0.03)';
    ctx.lineWidth = 1;
    let cs = cellSize();
    let ox = gridOffsetX();
    let oy = gridOffsetY();
    for (let r = 0; r < ROWS + 2; r++) {
        for (let c = 0; c < COLS; c++) {
            ctx.strokeRect(ox + c * cs, oy + r * cs, cs, cs);
        }
    }

    // Bricks
    for (let b of bricks) {
        if (!b.alive) continue;
        let ratio = b.hp / b.maxHp;
        ctx.globalAlpha = 0.4 + ratio * 0.6;
        ctx.fillStyle = b.color;
        let pad = 2;
        roundRect(ctx, b.x + pad, b.y + pad, b.w - pad*2, b.h - pad*2, 3);
        ctx.fill();
        // Highlight
        ctx.globalAlpha = 0.3;
        ctx.fillStyle = '#fff';
        roundRect(ctx, b.x + pad + 2, b.y + pad + 2, b.w - pad*4, (b.h - pad*4)*0.35, 2);
        ctx.fill();
        ctx.globalAlpha = 1;
    }

    // Paddle
    let grad = ctx.createLinearGradient(paddle.x, paddle.y, paddle.x, paddle.y + paddle.h);
    grad.addColorStop(0, '#4d8bff');
    grad.addColorStop(1, '#2a5dbf');
    ctx.fillStyle = grad;
    roundRect(ctx, paddle.x, paddle.y, paddle.w, paddle.h, 4);
    ctx.fill();
    ctx.strokeStyle = 'rgba(255,255,255,0.3)';
    ctx.lineWidth = 1;
    ctx.stroke();

    // Ball
    if (state === 'PLAYING' || state === 'BALL_HELD') {
        ctx.beginPath();
        ctx.arc(ball.x, ball.y, ball.r, 0, Math.PI * 2);
        let bgrad = ctx.createRadialGradient(ball.x - ball.r*0.3, ball.y - ball.r*0.3, 0, ball.x, ball.y, ball.r);
        bgrad.addColorStop(0, '#fff');
        bgrad.addColorStop(1, '#4d8bff');
        ctx.fillStyle = bgrad;
        ctx.fill();
    }

    // Particles
    for (let p of particles) {
        ctx.globalAlpha = p.life;
        ctx.fillStyle = p.color;
        ctx.fillRect(p.x - p.size/2, p.y - p.size/2, p.size, p.size);
    }
    ctx.globalAlpha = 1;

    // HUD
    ctx.font = `bold ${Math.max(14, W*0.018)}px monospace`;
    ctx.fillStyle = '#fff';
    ctx.textAlign = 'left';
    ctx.fillText(`SCORE: ${score}`, 20, 25);
    ctx.textAlign = 'right';
    ctx.fillText(`LIVES: ${'❤'.repeat(lives)}`, W - 20, 25);
    ctx.textAlign = 'center';
    ctx.fillText(`LEVEL ${level}`, W/2, 25);

    // Overlays
    if (state === 'START') drawStart();
    else if (state === 'GAME_OVER') drawGameOver();
    else if (state === 'LEVEL_CLEAR') drawLevelClear();
    else if (state === 'BALL_HELD') {
        ctx.font = `${Math.max(12, W*0.014)}px monospace`;
        ctx.fillStyle = 'rgba(255,255,255,0.6)';
        ctx.textAlign = 'center';
        ctx.fillText('Press SPACE to launch', W/2, H - 30);
    }
}

    ctx.textAlign = 'center';

function roundRect(ctx, x, y, w, h, r) {
    ctx.beginPath();
    ctx.moveTo(x + r, y);
    ctx.lineTo(x + w - r, y);
    ctx.quadraticCurveTo(x + w, y, x + w, y + r);
    ctx.lineTo(x + w, y + h - r);
    ctx.quadraticCurveTo(x + w, y + h, x + w - r, y + h);
    ctx.lineTo(x + r, y + h);
    ctx.quadraticCurveTo(x, y + h, x, y + h - r);
    ctx.lineTo(x, y + r);
    ctx.quadraticCurveTo(x, y, x + r, y);
    ctx.closePath();
}

// Game loop
let lastTime = 0;
function gameLoop(timestamp) {
    let dt = Math.min((timestamp - lastTime) / 1000, 0.05);
    lastTime = timestamp;

    update(dt);
    updateParticles(dt);
    draw();

    requestAnimationFrame(gameLoop);
}
