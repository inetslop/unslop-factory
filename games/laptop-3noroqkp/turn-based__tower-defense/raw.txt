<script>
const canvas = document.getElementById('canvas');
const ctx = canvas.getContext('2d');
const COLS = 14, ROWS = 9;
let TILE = 50;
let gameActive = false, animating = false, turnPhase = 'player';
let gold = 150, lives = 20, wave = 0, maxWave = 10;
let grid = [], path = [], enemies = [], towers = [];
let selectedTowerType = null, selectedTower = null;
let spawnQueue = [], enemyIdCounter = 0;

const PATH_COORDS = [
[0,4],[1,4],[2,4],[3,4],[3,3],[3,2],[4,2],[5,2],[6,2],[6,3],[6,4],[6,5],[6,6],[7,6],[8,6],[9,6],[9,5],[9,4],[9,3],[9,2],[10,2],[11,2],[12,2],[13,2]
];

const TOWER_DEFS = {
cannon: { name:'Cannon', cost:50, damage:15, range:3, cooldown:1, color:'#e94560', projColor:'#ff6b6b' },
sniper: { name:'Sniper', cost:100, damage:40, range:5, cooldown:2, color:'#4a9eff', projColor:'#87ceeb' },
frost: { name:'Frost', cost:75, damage:5, range:3, cooldown:1, slow:true, color:'#00d4ff', projColor:'#80ffff' }
};

const ENEMY_TYPES = [
{ name:'Grunt', hp:30, speed:1, reward:20, color:'#e94560', radius:14 },
{ name:'Runner', hp:20, speed:2, reward:30, color:'#f39c12', radius:12 },
{ name:'Tank', hp:80, speed:1, reward:50, color:'#8e44ad', radius:16 },
{ name:'Boss', hp:200, speed:1, reward:100, color:'#c0392b', radius:20 }
];

const WAVES = [
[{type:0,count:5}],
[{type:0,count:8}],
[{type:0,count:5},{type:1,count:3}],
[{type:0,count:4},{type:1,count:5}],
[{type:2,count:3},{type:0,count:4}],
[{type:1,count:6},{type:2,count:2}],
[{type:2,count:4},{type:1,count:4}],
[{type:0,count:8},{type:2,count:3},{type:1,count:3}],
[{type:2,count:5},{type:1,count:5},{type:0,count:5}],
[{type:3,count:2},{type:2,count:3},{type:1,count:4}]
];

function resize() {
const w = window.innerWidth, h = window.innerHeight;
const topH = 50, botH = 60, padding = 10;
const availW = w - padding*2, availH = h - topH - botH - padding*2;
TILE = Math.floor(Math.min(availW/COLS, availH/ROWS));
canvas.width = COLS * TILE;
canvas.height = ROWS * TILE;
canvas.style.marginTop = (topH/2)+'px';
document.getElementById('topBar').style.display='flex';
document.getElementById('bottomBar').style.display='flex';
draw();
}

function initGame() {
gold=150; lives=20; wave=0; enemies=[]; towers=[]; spawnQueue=[]; enemyIdCounter=0;
grid = Array.from({length:ROWS},()=>Array(COLS).fill(null));
path = PATH_COORDS.map(p=>({x:p[0],y:p[1]}));
updateUI(); draw();
}

function startWave() {
wave++;
spawnQueue = [];
const waveDef = WAVES[wave-1];
let delay=0;
for(const g of waveDef){
for(let i=0;i<g.count;i++){
spawnQueue.push({type:g.type, delay: delay++});
}
}
showWaveMsg('Wave '+wave);
updateUI();
}

function showWaveMsg(txt) {
const el = document.getElementById('waveMsg');
el.textContent = txt; el.style.opacity='1';
setTimeout(()=>{el.style.opacity='0';},1500);
}

function showMsg(txt){
const el = document.getElementById('msg');
el.textContent=txt; el.style.opacity='1';
setTimeout(()=>{el.style.opacity='0';},1200);
}

function updateUI() {
document.getElementById('livesDisplay').textContent=lives;
document.getElementById('goldDisplay').textContent=gold;
document.getElementById('waveDisplay').textContent=wave+'/'+maxWave;
document.querySelectorAll('.towerBtn').forEach(b=>{
const t=b.dataset.type; if(!t)return;
const cost=TOWER_DEFS[t].cost;
b.classList.toggle('disabled', gold<cost);
});
}

function draw() {
ctx.clearRect(0,0,canvas.width,canvas.height);
for(let r=0;r<ROWS;r++) for(let c=0;c<COLS;c++){
const x=c*TILE,y=r*TILE;
if(isPath(c,r)){ ctx.fillStyle='#2d3436'; }
else { ctx.fillStyle=(r+c)%2?'#1e2a3a':'#1a2535'; }
ctx.fillRect(x,y,TILE,TILE);
ctx.strokeStyle='rgba(255,255,255,0.05)'; ctx.strokeRect(x,y,TILE,TILE);
}
// Draw path arrows
for(let i=0;i<path.length-1;i++){
const a=path[i],b=path[i+1];
ctx.fillStyle='rgba(255,255,255,0.1)';
const cx=(a.x+0.5)*TILE,cy=(a.y+0.5)*TILE;
ctx.beginPath(); ctx.arc(cx,cy,TILE*0.15,0,Math.PI*2); ctx.fill();
}
// Towers
for(const t of towers) drawTower(t);
// Enemies
for(const e of enemies) drawEnemy(e);
// Selected tower range
if(selectedTower){
const def=TOWER_DEFS[selectedTower.type];
ctx.strokeStyle='rgba(255,215,0,0.3)'; ctx.lineWidth=2;
ctx.beginPath(); ctx.arc((selectedTower.x+0.5)*TILE,(selectedTower.y+0.5)*TILE,def.range*TILE,0,Math.PI*2); ctx.stroke();
}
}

function isPath(c,r){ return PATH_COORDS.some(p=>p[0]===c&&p[1]===r); }

function drawTower(t){
const x=t.x*TILE,y=t.y*TILE;
const def=TOWER_DEFS[t.type];
ctx.fillStyle=def.color; ctx.globalAlpha=0.3;
ctx.fillRect(x+2,y+2,TILE-4,TILE-4);
ctx.globalAlpha=1;
ctx.beginPath(); ctx.arc(x+TILE/2,y+TILE/2,TILE*0.3,0,Math.PI*2);
ctx.fillStyle=def.color; ctx.fill();
ctx.strokeStyle='#fff'; ctx.lineWidth=2; ctx.stroke();
// Level indicator
if(t.level>1){
ctx.fillStyle='#ffd700'; ctx.font='bold 10px sans-serif';
ctx.textAlign='center'; ctx.fillText('★'.repeat(t.level-1), x+TILE/2, y+TILE-4);
}
// Cooldown indicator
if(t.cooldownLeft>0){
ctx.fillStyle='rgba(255,100,100,0.7)'; ctx.font='9px sans-serif'; ctx.textAlign='center';
ctx.fillText('⏳'+t.cooldownLeft, x+TILE/2, y+12);
}
}

function drawEnemy(e){
const px=(e.x+0.5)*TILE, py=(e.y+0.5)*TILE;
const def=ENEMY_TYPES[e.type];
ctx.beginPath(); ctx.arc(px,py,def.radius,0,Math.PI*2);
ctx.fillStyle=def.color; ctx.fill();
ctx.strokeStyle='#fff'; ctx.lineWidth=1.5; ctx.stroke();
// HP bar
const hpPct=e.hp/e.maxHp;
ctx.fillStyle='#333'; ctx.fillRect(px-15,py-def.radius-8,30,5);
ctx.fillStyle=hpPct>0.5?'#2ecc71':hpPct>0.25?'#f39c12':'#e74560';
ctx.fillRect(px-15,py-def.radius-8,30*hpPct,5);
// Slow indicator
if(e.slowTimer>0){ ctx.fillStyle='#00d4ff'; ctx.font='10px sans-serif'; ctx.textAlign='center'; ctx.fillText('❄',px,py+3); }
}

function placeTower(type,c,r){
if(gold<TOWER_DEFS[type].cost) return showMsg('Not enough gold!');
if(grid[r][c]||isPath(c,r)) return;
gold-=TOWER_DEFS[type].cost;
grid[r][c]={type,x:c,y:r,level:1,damage:TOWER_DEFS[type].damage,range:TOWER_DEFS[type].range,cooldownLeft:0,maxCooldown:TOWER_DEFS[type].cooldown};
towers.push(grid[r][c]);
updateUI(); draw();
}

function upgradeTower(t){
const cost=50*t.level;
if(gold<cost) return showMsg('Need '+cost+'g to upgrade!');
gold-=cost; t.level++; t.damage+=Math.floor(TOWER_DEFS[t.type].damage*0.4);
t.range=Math.min(t.range+ (t.level%2===0?1:0), 7);
updateUI(); draw(); showMsg('Upgraded!');
}

function sellTower(t){
const refund=Math.floor(TOWER_DEFS[t.type].cost*0.6*t.level);
gold+=refund;
grid[t.y][t.x]=null;
towers=towers.filter(x=>x!==t);
selectedTower=null; document.getElementById('infoPanel').style.display='none';
updateUI(); draw(); showMsg('Sold for '+refund+'g');
}

function endTurn(){
if(turnPhase!=='player') return;
turnPhase='enemy';
document.getElementById('turnIndicator').textContent='Enemy Turn...';
processEnemies();
processTowers();
checkWaveEnd();
if(lives<=0) return gameOver(false);
turnPhase='player';
document.getElementById('turnIndicator').textContent='Your Turn';
updateUI(); draw();
}

function processEnemies(){
const active = enemies.filter(e=>e.alive);
// Spawn new
for(const s of spawnQueue){
if(s.delay===0 && !enemies.find(e=>e.spawnId===s.type+'-'+enemyIdCounter)){
spawnEnemy(s.type);
}
s.delay--;
}
// Move enemies
for(const e of active){
if(!e.alive) continue;
let spd=e.speed;
if(e.slowTimer>0){ spd=Math.max(1,spd-1); e.slowTimer--; }
const steps=spd;
for(let s=0;s<steps;s++){
const idx=path.findIndex(p=>p.x===e.x&&p.y===e.y);
if(idx>=0 && idx<path.length-1){
const np=path[idx+1];
e.x=np.x; e.y=np.y;
} else if(idx===path.length-1){
lives--; e.alive=false;
updateUI();
if(lives<=0) return;
}
}
}
enemies=enemies.filter(e=>e.alive);
}

function processTowers(){
for(const t of towers){
if(t.cooldownLeft>0){ t.cooldownLeft--; continue; }
const target=findTarget(t);
if(target){
t.cooldownLeft=t.maxCooldown;
target.hp-=t.damage;
if(TOWER_DEFS[t.type].slow && target.slowTimer<=0) target.slowTimer=2;
// Damage flash
target.hitFlash=3;
if(target.hp<=0){
target.alive=false;
gold+=ENEMY_TYPES[target.type].reward;
showMsg('+'+ENEMY_TYPES[target.type].reward+'g!');
}
updateUI();
}
}
enemies=enemies.filter(e=>e.alive);
}

function findTarget(t){
let best=null, bestDist=Infinity;
for(const e of enemies){
if(!e.alive) continue;
const dx=Math.abs(e.x-t.x), dy=Math.abs(e.y-t.y);
const dist=Math.max(dx,dy); // Chebyshev
if(dist<=t.range && dist<bestDist){ best=e; bestDist=dist; }
}
return best;
}

function spawnEnemy(type){
const def=ENEMY_TYPES[type];
const hpScale=1+wave*0.3;
enemies.push({id:enemyIdCounter++,type,x:path[0].x,y:path[0].y,hp:Math.floor(def.hp*hpScale),maxHp:Math.floor(def.hp*hpScale),speed:def.speed,alive:true,slowTimer:0,hitFlash:0});
}

function checkWaveEnd(){
if(wave<maxWave && spawnQueue.length===0 && enemies.length===0){
gold+=30+wave*10;
updateUI(); showMsg('Wave Complete! +'+(30+wave*10)+'g');
if(wave>=maxWave) gameOver(true);
}
}

function gameOver(win){
gameActive=false;
const ov=document.getElementById('overlay');
ov.style.display='flex';
document.getElementById('overlayContent').innerHTML=
`<h1>${win?'🏆 Victory!':'💀 Defeat!'}</h1><p>${win?'You defended the kingdom through all 10 waves!':'The enemies have breached your defenses.'}</p><button id="startBtn" onclick="location.reload()">Play Again</button>`;
}

// Input
let mouseX=0,mouseY=0;
canvas.addEventListener('mousemove',e=>{
const r=canvas.getBoundingClientRect();
mouseX=Math.floor((e.clientX-r.left)/TILE);
mouseY=Math.floor((e.clientY-r.top)/TILE);
});

canvas.addEventListener('click',e=>{
if(!gameActive) return;
const r=canvas.getBoundingClientRect();
const c=Math.floor((e.clientX-r.left)/TILE);
const rr=Math.floor((e.clientY-r.top)/TILE);
if(c<0||c>=COLS||rr<0||rr>=ROWS) return;

if(selectedTowerType && !grid[rr][c] && !isPath(c,rr)){
placeTower(selectedTowerType,c,rr);
selectedTowerType=null;
document.querySelectorAll('.towerBtn').forEach(b=>b.classList.remove('selected'));
} else if(grid[rr][c]){
selectedTower=grid[rr][c];
showInfoPanel();
} else {
selectedTower=null;
document.getElementById('infoPanel').style.display='none';
draw();
}
});

function showInfoPanel(){
const t=selectedTower; if(!t) return;
const def=TOWER_DEFS[t.type];
const upCost=50*t.level;
const sellVal=Math.floor(def.cost*0.6*t.level);
document.getElementById('infoPanel').style.display='block';
document.getElementById('infoPanel').innerHTML=`<h4>${def.name} Lv.${t.level}</h4><p>DMG: ${t.damage}</p><p>Range: ${t.range}</p><p>CD: ${t.maxCooldown}</p><br><button id="upBtn">Upgrade (${upCost}g)</button><button id="sellBtn">Sell (+${sellVal}g)</button>`;
document.getElementById('upBtn').onclick=()=>upgradeTower(t);
document.getElementById('sellBtn').onclick=()=>{ sellTower(t); };
}

document.querySelectorAll('.towerBtn').forEach(btn=>{
btn.addEventListener('click',()=>{
const t=btn.dataset.type; if(!t) return;
if(gold<TOWER_DEFS[t].cost) return showMsg('Not enough gold!');
selectedTowerType = selectedTowerType===t ? null : t;
document.querySelectorAll('.towerBtn').forEach(b=>b.classList.remove('selected'));
if(selectedTowerType) btn.classList.add('selected');
});
});

document.getElementById('endTurnBtn').addEventListener('click',()=>{ if(gameActive) endTurn(); });

window.addEventListener('keydown',e=>{
if(!gameActive) return;
const k=e.key.toLowerCase();
if(k==='1') selectTowerType('cannon');
else if(k==='2') selectTowerType('sniper');
else if(k==='3') selectTowerType('frost');
else if(k==='e') endTurn();
else if(k==='u' && selectedTower) upgradeTower(selectedTower);
else if(k==='s' && selectedTower) sellTower(selectedTower);
});

function selectTowerType(t){
if(gold<TOWER_DEFS[t].cost) return showMsg('Not enough gold!');
selectedTowerType=t;
document.querySelectorAll('.towerBtn').forEach(b=>{
b.classList.toggle('selected', b.dataset.type===t);
});
}

document.getElementById('startBtn').addEventListener('click',()=>{
document.getElementById('overlay').style.display='none';
gameActive=true;
initGame();
startWave();
});

window.addEventListener('resize',resize);
resize();
draw();
</script>
