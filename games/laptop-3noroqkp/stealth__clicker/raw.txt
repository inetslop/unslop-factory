function isWall(x, y) {
if (x < 0 || x >= COLS || y < 0 || y >= ROWS) return true;
return walls.some(w => w.x === x && w.y === y);
}

function bfsPath(sx, sy, ex, ey) {
const queue = [[sx, sy]];
const visited = {};
const parent = {};
visited[sx + ',' + sy] = true;
while (queue.length > 0) {
const [cx, cy] = queue.shift();
if (cx === ex && cy === ey) {
const path = [];
let cur = [ex, ey];
while (cur[0] !== sx || cur[1] !== sy) {
path.unshift(cur);
const p = parent[cur[0] + ',' + cur[1]];
cur = p;
}
return path;
}
const dirs = [[0,-1],[0,1],[-1,0],[1,0]];
for (const d of dirs) {
const nx = cx + d[0], ny = cy + d[1];
const key = nx + ',' + ny;
if (!visited[key] && !isWall(nx, ny)) {
visited[key] = true;
parent[nx + ',' + ny] = [cx, cy];
queue.push([nx, ny]);
}
}
}
return null;
}

function guardCanSee(guard, px, py) {
// Check if player is in line of sight from guard
const dx = px - guard.px;
const dy = py - guard.py;
const dist = Math.sqrt(dx*dx + dy*dy);
if (dist > TILE * 4.5) return false;
// Direction check
const angle = Math.atan2(dy, dx);
const facingAngle = guard.dir === 1 ? 0 : Math.PI; // simplified: face horizontal based on path direction
// Actually let's make guards see in all directions but have limited range
return dist < TILE * 4.5;
}

function isAdjacentToShadow(px, py) {
const gx = Math.floor(px / TILE);
const gy = Math.floor(py / TILE);
for (const s of shadows) {
if (Math.abs(s.x - gx) <= 1 && Math.abs(s.y - gy) <= 1) return true;
}
return false;
}

function update() {
frames++;
if (gameState !== 'playing') return;

// Player movement
if (!player.moving && pathQueue.length > 0) {
const next = pathQueue[0];
if (next.x !== player.gx || next.y !== player.gy) {
pathQueue.shift();
player.moving = true;
player.tx = next.x * TILE + TILE/2;
player.ty = next.y * TILE + TILE/2;
}
}

if (player.moving) {
const speed = 0.15;
const dx = player.tx - player.px;
const dy = player.ty - player.py;
const dist = Math.sqrt(dx*dx + dy*dy);
if (dist < TILE * speed) {
player.px = player.tx;
player.py = player.ty;
player.moving = false;
player.gx = Math.floor(player.px / TILE);
player.gy = Math.floor(player.py / TILE);
playSound('step');
} else {
player.px += (dx / dist) * TILE * speed;
player.py += (dy / dist) * TILE * speed;
}
}

// Check shadow hiding
player.hidden = isAdjacentToShadow(player.px, player.py);

// Check objective
if (!hasItem && objective && Math.abs(player.gx - objective.x) <= 1 && Math.abs(player.gy - objective.y) <= 1) {
hasItem = true;
playSound('steal');
}

// Check exit
if (hasItem && exitPos && player.gx === exitPos.x && player.gy === exitPos.y) {
gameState = 'levelComplete';
playSound('escape');
return;
}

// Guard movement
let seen = false;
for (const guard of guards) {
const nextPos = guard.path[guard.pathIdx];
const targetX = nextPos[0] * TILE + TILE/2;
const targetY = nextPos[1] * TILE + TILE/2;
const dx = targetX - guard.px;
const dy = targetY - guard.py;
const dist = Math.sqrt(dx*dx + dy*dy);

if (dist < TILE * 0.5) {
guard.gx = nextPos[0];
guard.gy = nextPos[1];
guard.pathIdx = (guard.pathIdx + 1) % guard.path.length;
} else {
const speed = 1 + level * 0.3;
guard.px += (dx / dist) * speed;
guard.py += (dy / dist) * speed;
}

// Check if guard sees player
if (!player.hidden && guardCanSee(guard, player.px, player.py)) {
seen = true;
}
}

if (seen) {
alarmTimer++;
if (alarmTimer > 30) {
gameState = 'caught';
playSound('alarm');
}
} else {
alarmTimer = Math.max(0, alarmTimer - 1);
}
}
