<script>
(function() {
    const canvas = document.getElementById('gameCanvas');
    const ctx = canvas.getContext('2d');
    const startScreen = document.getElementById('start-screen');
    const startBtn = document.getElementById('start-btn');
    const shopPanel = document.getElementById('shop-panel');
    const upgradesList = document.getElementById('upgrades-list');
    const scoreDisplay = document.getElementById('score-display');
    const statClick = document.getElementById('stat-click');
    const statAuto = document.getElementById('stat-auto');
    const statMult = document.getElementById('stat-mult');
    const restartBtn = document.getElementById('restart-btn');
    const clickHint = document.getElementById('click-hint');

    let W, H;
    let gameStarted = false;
    let lastTime = 0;
    let audioCtx = null;

    // Game state
    let score = 0;
    let clickPower = 1;
    let autoRate = 0;
    let multiplier = 1;
    let totalClicks = 0;

    // Ball state
    let ballX, ballY, ballVX, ballVY;
    let paddleL, paddleR;
    let particles = [];
    let floatTexts = [];
    let glowPulse = 0;
    let autoTimer = 0;

    // Upgrades
    const upgrades = [
        { name: "Stronger Paddle", desc: "+1 per click", baseCost: 10, costMult: 1.5, level: 0, effect: () => { clickPower += 1; } },
        { name: "Auto-Bot", desc: "+1/s auto bounce", baseCost: 25, costMult: 1.8, level: 0, effect: () => { autoRate += 1; } },
        { name: "Speed Boost", desc: "x2 multiplier", baseCost: 100, costMult: 3, level: 0, effect: () => { multiplier *= 2; } },
        { name: "Twin Paddle", desc: "+5 per click", baseCost: 200, costMult: 2, level: 0, effect: () => { clickPower += 5; } },
        { name: "Swarm Bots", desc: "+10/s auto", baseCost: 500, costMult: 2.2, level: 0, effect: () => { autoRate += 10; } },
        { name: "Quantum Pong", desc: "x5 multiplier", baseCost: 2000, costMult: 4, level: 0, effect: () => { multiplier *= 5; } },
        { name: "Plasma Paddle", desc: "+25 per click", baseCost: 5000, costMult: 2.5, level: 0, effect: () => { clickPower += 25; } },
    ];

    function getUpgradeCost(u) {
        return Math.floor(u.baseCost * Math.pow(u.costMult, u.level));
    }

    function resize() {
        W = window.innerWidth;
        H = window.innerHeight;
        canvas.width = W;
        canvas.height = H;
        ballX = W / 2;
        paddleL = W * 0.15;
        paddleR = W * 0.85;
    }

    function initBall() {
        const courtH = H * 0.7;
        const top = H * 0.15;
        ballX = W / 2;
        ballY = top + Math.random() * courtH;
        ballVX = (Math.random() > 0.5 ? 1 : -1) * 3;
        ballVY = (Math.random() - 0.5) * 4;
    }

    function playBeep(freq, dur) {
        if (!audioCtx) return;
        const osc = audioCtx.createOscillator();
        const gain = audioCtx.createGain();
        osc.connect(gain);
        gain.connect(audioCtx.destination);
        osc.type = 'square';
        osc.frequency.value = freq;
        gain.gain.setValueAtTime(0.08, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
        osc.start();
        osc.stop(audioCtx.currentTime + dur);
    }

    function spawnParticles(x, y, count) {
        for (let i = 0; i < count; i++) {
            const angle = Math.random() * Math.PI * 2;
            const speed = 1 + Math.random() * 4;
            particles.push({
                x, y,
                vx: Math.cos(angle) * speed,
                vy: Math.sin(angle) * speed,
                life: 1,
                size: 2 + Math.random() * 3
            });
        }
    }

    function spawnFloatText(x, y, text) {
        floatTexts.push({ x, y, text, life: 1, vy: -1.5 });
    }

    function handleClick(e) {
        if (!gameStarted) return;
        const rect = canvas.getBoundingClientRect();
        const mx = e.clientX - rect.left;
        const my = e.clientY - rect.top;

        // Only register clicks in left half (player side)
        if (mx < W * 0.5) {
            const pts = Math.floor(clickPower * multiplier);
            score += pts;
            totalClicks++;
            playBeep(880, 0.08);
            spawnParticles(paddleL, my, 8);
            spawnFloatText(mx, my - 20, '+' + formatNum(pts));
            glowPulse = 1;

            // Nudge ball toward player side for visual
            if (ballX > W * 0.4) {
                ballVX = -(3 + Math.random() * 2);
            }
        }
    }

    function formatNum(n) {
        if (n >= 1e9) return (n / 1e9).toFixed(1) + 'B';
        if (n >= 1e6) return (n / 1e3).toFixed(1) + 'M';
        if (n >= 1e4) return (n / 1e3).toFixed(1) + 'K';
        return Math.floor(n).toString();
    }

    function buyUpgrade(idx) {
        const u = upgrades[idx];
        const cost = getUpgradeCost(u);
        if (score >= cost) {
            score -= cost;
            u.level++;
            u.effect();
            playBeep(1200, 0.1);
            renderUpgrades();
            updateStats();
        }
    }

    function renderUpgrades() {
        upgradesList.innerHTML = '';
        upgrades.forEach((u, i) => {
            const cost = getUpgradeCost(u);
            const btn = document.createElement('button');
            btn.className = 'upgrade-btn';
            btn.disabled = score < cost;
            btn.innerHTML = `<span>${u.name} <span class="cost">${formatNum(cost)}</span></span><span class="desc">${u.desc} | Lvl ${u.level}</span>`;
            btn.addEventListener('click', () => buyUpgrade(i));
            upgradesList.appendChild(btn);
        });
    }

    function updateStats() {
        scoreDisplay.textContent = formatNum(score);
        statClick.textContent = '+' + formatNum(Math.floor(clickPower * multiplier)) + '/click';
        statAuto.textContent = Math.floor(autoRate * multiplier) + '/s auto';
        statMult.textContent = 'x' + (multiplier >= 100 ? multiplier : multiplier % 1 === 0 ? multiplier : multiplier.toFixed(1));
    }

    function update(dt) {
        // Auto income
        if (autoRate > 0) {
            autoTimer += dt;
            const interval = Math.max(0.1, 1 / Math.max(autoRate, 1));
            while (autoTimer >= interval) {
                autoTimer -= interval;
                const pts = Math.floor((autoRate / Math.max(autoRate, 1)) * multiplier);
                score += Math.max(1, Math.floor(autoRate * multiplier * dt));
            }
        }

        // Ball physics
        const courtTop = H * 0.15;
        const courtBottom = H * 0.85;
        ballX += ballVX;
        ballY += ballVY;

        if (ballY < courtTop + 5 || ballY > courtBottom - 5) {
            ballVY *= -1;
            playBeep(440, 0.05);
        }

        if (ballX < paddleL - 8) {
            ballVX = Math.abs(ballVX) * 1.02;
            spawnParticles(paddleL, ballY, 5);
            playBeep(660, 0.06);
        }
        if (ballX > paddleR + 8) {
            ballVX = -Math.abs(ballVX) * 1.02;
            spawnParticles(paddleR, ballY, 5);
            playBeep(660, 0.06);
        }

        if (ballX < -20) { ballX = W + 20; ballVX = Math.abs(ballVX); }
        if (ballX > W + 20) { ballX = -20; ballVX = -Math.abs(ballVX); }

        // Particles
        for (let i = particles.length - 1; i >= 0; i--) {
            const p = particles[i];
            p.x += p.vx;
            p.y += p.vy;
            p.life -= dt * 2;
            if (p.life <= 0) particles.splice(i, 1);
        }

        // Float texts
        for (let i = floatTexts.length - 1; i >= 0; i--) {
            const f = floatTexts[i];
            f.y += f.vy;
            f.life -= dt * 1.5;
            if (f.life <= 0) floatTexts.splice(i, 1);
        }

        // Glow
        glowPulse = Math.max(0, glowPulse - dt * 3);

        updateStats();
    }

    function draw() {
        ctx.fillStyle = '#0a0a0a';
        ctx.fillRect(0, 0, W, H);

        // Court lines
        const courtTop = H * 0.15;
        const courtBottom = H * 0.85;

        // Center dashed line
        ctx.strokeStyle = 'rgba(76,255,68,0.3)';
        ctx.lineWidth = 2;
        ctx.setLineDash([10, 15]);
        ctx.beginPath();
        ctx.moveTo(W / 2, courtTop);
        ctx.lineTo(W / 2, courtBottom);
        ctx.stroke();
        ctx.setLineDash([]);

        // Court borders
        ctx.strokeStyle = 'rgba(76,255,68,0.2)';
        ctx.lineWidth = 1;
        ctx.beginPath();
        ctx.moveTo(W * 0.05, courtTop);
        ctx.lineTo(W * 0.95, courtTop);
        ctx.moveTo(W * 0.05, courtBottom);
        ctx.lineTo(W * 0.95, courtBottom);
        ctx.stroke();

        // Paddles
        const paddleH = H * 0.12;
        const paddleW = 8;

        // Left paddle (player)
        const glowColor = `rgba(76,255,68,${0.6 + glowPulse * 0.4})`;
        ctx.shadowBlur = 15 + glowPulse * 20;
        ctx.shadowColor = '#4f4';
        ctx.fillStyle = glowColor;
        ctx.fillRect(paddleL - paddleW / 2, ballY - paddleH / 2, paddleW, paddleH);

        // Right paddle (AI)
        ctx.shadowBlur = 10;
        ctx.fillStyle = 'rgba(76,255,68,0.6)';
        const aiPaddleY = courtTop + ((ballY - courtTop) / (courtBottom - courtTop)) * (courtBottom - courtTop);
        ctx.fillRect(paddleR - paddleW / 2, ballY - paddleH / 2, paddleW, paddleH);

        // Ball
        ctx.shadowBlur = 15;
        ctx.shadowColor = '#fff';
        ctx.fillStyle = '#fff';
        ctx.beginPath();
        ctx.arc(ballX, ballY, 6, 0, Math.PI * 2);
        ctx.fill();

        ctx.shadowBlur = 0;

        // Particles
        particles.forEach(p => {
            ctx.globalAlpha = p.life;
            ctx.fillStyle = '#4f4';
            ctx.fillRect(p.x - p.size / 2, p.y - p.size / 2, p.size, p.size);
        });
        ctx.globalAlpha = 1;

        // Float texts
        floatTexts.forEach(f => {
            ctx.globalAlpha = f.life;
            ctx.fillStyle = '#4f4';
            ctx.font = 'bold 16px Courier New';
            ctx.textAlign = 'center';
            ctx.fillText(f.text, f.x, f.y);
        });
        ctx.globalAlpha = 1;

        // Scanline effect (subtle)
        ctx.fillStyle = 'rgba(0,0,0,0.03)';
        for (let y = 0; y < H; y += 4) {
            ctx.fillRect(0, y, W, 2);
        }
    }

    function gameLoop(timestamp) {
        if (!gameStarted) return;
        const dt = Math.min((timestamp - lastTime) / 1000, 0.1);
        lastTime = timestamp;
        update(dt);
        draw();
        requestAnimationFrame(gameLoop);
    }

    function startGame() {
        if (gameStarted) return;
        gameStarted = true;
        startScreen.style.display = 'none';
        shopPanel.style.display = 'block';
        restartBtn.style.display = 'block';
        clickHint.style.display = 'block';
        setTimeout(() => { clickHint.style.display = 'none'; }, 5000);

        if (!audioCtx) {
            audioCtx = new (window.AudioContext || window.webkitAudioContext)();
        }

        resize();
        initBall();
        renderUpgrades();
        updateStats();
        lastTime = performance.now();
        requestAnimationFrame(gameLoop);
    }

    function restartGame() {
        score = 0;
        clickPower = 1;
        autoRate = 0;
        multiplier = 1;
        totalClicks = 0;
        particles = [];
        floatTexts = [];
        glowPulse = 0;
        autoTimer = 0;
        upgrades.forEach(u => u.level = 0);
        initBall();
        renderUpgrades();
        updateStats();
    }

    // Events
    window.addEventListener('resize', resize);
    canvas.addEventListener('click', handleClick);
    startBtn.addEventListener('click', startGame);
    restartBtn.addEventListener('click', restartGame);

    // Keyboard support
    window.addEventListener('keydown', (e) => {
        if (!gameStarted && (e.key === ' ' || e.key === 'Enter')) {
            startGame();
        }
    });

    // Prevent scrolling
    window.addEventListener('keydown', (e) => {
        if (e.key === ' ') e.preventDefault();
    });

    resize();
    initBall();
    draw();
})();
</script>
</body>
