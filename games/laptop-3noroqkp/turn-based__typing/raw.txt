<script>
(function(){
let audioCtx = null;
function initAudio() { if (!audioCtx) audioCtx = new (window.AudioContext || webkitAudioContext)(); }
function playTone(freq, dur, type, vol) {
  if (!audioCtx) return;
  const o = audioCtx.createOscillator();
  const g = audioCtx.createGain();
  o.type = type || 'square';
  o.frequency.value = freq;
  g.gain.value = vol || 0.1;
  g.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
  o.connect(g); g.connect(audioCtx.destination);
  o.start(); o.stop(audioCtx.currentTime + dur);
}
function sfxKeystroke() { playTone(800 + Math.random()*200, 0.05, 'square', 0.03); }
function sfxHit() { playTone(100, 0.3, 'sawtooth', 0.2); playTone(200, 0.2, 'square', 0.15); }
function sfxEnemyHit() { playTone(60, 0.4, 'sawtooth', 0.25); }
function sfxWin() { for(let i=0;i<4;i++) setTimeout(()=>playTone(400+i*100,0.3,'square',0.15),i*150); }
function sfxLose() { for(let i=0;i<4;i++) setTimeout(()=>playTone(300-i*60,0.4,'sawtooth',0.2),i*200); }

const words = {
  easy: ["cat","dog","run","top","red","sun","sky","big","hot","map","pen","box","hat","leg","bed","net","car","bus","fan","jam"],
  medium: ["word","play","game","type","code","data","test","idea","time","fire","star","moon","tree","bird","fish","rock","wind","rain","snow","cloud"],
  hard: ["player","dragon","warrior","monster","champion","keyboard","alphabet","thunder","frozen","crystal","shadow","flame","storm","blade","sword","shield","castle","battle","victory","glory"]
};

let state = 'start';
let playerHp, enemyHp, maxHp, round, maxRounds;
let currentWord, targetIdx, timerInterval, timerEnd, timerDuration;
let enemyAnimInterval, enemySuccessChance;
let totalDealt, totalTaken, roundsWon;

const $ = id => document.getElementById(id);

function startGame() {
  initAudio();
  playerHp = 100; enemyHp = 100; maxHp = 100;
  round = 1; maxRounds = 10;
  totalDealt = 0; totalTaken = 0; roundsWon = 0;
  showScreen('battleScreen');
  updateHP();
  startPlayerTurn();
}

function showScreen(id) {
  document.querySelectorAll('.screen').forEach(s => s.classList.remove('active'));
  $(id).classList.add('active');
}

function updateHP() {
  const pe = Math.max(0, playerHp), ee = Math.max(0, enemyHp);
  $('playerHpBar').style.width = (pe/maxHp*100)+'%';
  $('enemyHpBar').style.width = (ee/maxHp*100)+'%';
  $('playerHpText').textContent = pe+'/'+maxHp;
  $('enemyHpText').textContent = ee+'/'+maxHp;
}

function getWord() {
  const tier = round <= 3 ? 'easy' : round <= 7 ? 'medium' : 'hard';
  const list = words[tier];
  return list[Math.floor(Math.random()*list.length)];
}

function startPlayerTurn() {
  state = 'playerTurn';
  $('turnIndicator').textContent = "⚔️ Your Turn!";
  $('enemyAnimWord').textContent = '';
  currentWord = getWord();
  targetIdx = 0;
  renderWord();
  timerDuration = Math.max(3, currentWord.length * 0.8);
  timerEnd = performance.now() + timerDuration * 1000;
  $('timerFill').style.width = '100%';
  $('timerFill').style.background = '#ff0';
  clearInterval(timerInterval);
  timerInterval = setInterval(updateTimer, 50);
}

function renderWord() {
  const disp = $('wordDisplay');
  disp.innerHTML = '';
  for (let i = 0; i < currentWord.length; i++) {
    const span = document.createElement('span');
    span.className = 'word-char';
    span.textContent = currentWord[i];
    if (i === targetIdx) span.classList.add('next');
    disp.appendChild(span);
  }
}

function updateTimer() {
  const elapsed = (performance.now() - timerEnd) / 1000;
  const remaining = Math.max(0, timerDuration + elapsed);
  const pct = Math.max(0, remaining / timerDuration * 100);
  $('timerFill').style.width = pct + '%';
  if (pct < 30) $('timerFill').style.background = '#f44';
  else if (pct < 60) $('timerFill').style.background = '#fa0';
  else $('timerFill').style.background = '#ff0';
  if (remaining <= 0) { playerFail(); }
}

function playerType(char) {
  if (state !== 'playerTurn') return;
  sfxKeystroke();
  const expected = currentWord[targetIdx];
  const chars = $('wordDisplay').children;
  if (char === expected) {
    chars[targetIdx].classList.remove('next');
    chars[targetIdx].classList.add('correct');
    targetIdx++;
    if (targetIdx >= currentWord.length) { playerSuccess(); return; }
    if (chars[targetIdx]) { chars[targetIdx-1].classList.remove('next'); chars[targetIdx].classList.add('next'); }
  } else {
    if (chars[targetIdx]) { chars[targetIdx].classList.add('incorrect'); setTimeout(()=>chars[targetIdx].classList.remove('incorrect'),300); }
  }
}

function playerSuccess() {
  clearInterval(timerInterval);
  state = 'resolving';
  const dmg = 12 + Math.floor(Math.random()*8);
  enemyHp -= dmg;
  totalDealt += dmg;
  roundsWon++;
  $('turnIndicator').textContent = "✅ Hit! -" + dmg + " HP";
  $('turnIndicator').style.color = '#4f8';
  sfxHit();
  showDmg('enemy', '-'+dmg);
  updateHP();
  if (enemyHp <= 0) { setTimeout(()=>endGame(true), 1200); return; }
  setTimeout(startEnemyTurn, 1200);
}

function playerFail() {
  clearInterval(timerInterval);
  state = 'resolving';
  const dmg = 8 + Math.floor(Math.random()*6);
  playerHp -= dmg;
  totalTaken += dmg;
  $('turnIndicator').textContent = "❌ Missed! -" + dmg + " HP";
  $('turnIndicator').style.color = '#f44';
  sfxEnemyHit();
  showDmg('player', '-'+dmg);
  updateHP();
  if (playerHp <= 0) { setTimeout(()=>endGame(false), 1200); return; }
  setTimeout(startEnemyTurn, 1200);
}

function startEnemyTurn() {
  state = 'enemyTurn';
  $('turnIndicator').textContent = "🐉 Dragon's Turn...";
  $('turnIndicator').style.color = '#f80';
  const enemyWord = getWord();
  enemySuccessChance = 0.5 + round * 0.04;
  let idx = 0;
  $('enemyAnimWord').textContent = '';
  clearInterval(enemyAnimInterval);
  const speed = Math.max(120, 300 - round*15);
  enemyAnimInterval = setInterval(() => {
    if (idx < enemyWord.length) {
      $('enemyAnimWord').textContent += enemyWord[idx];
      idx++;
      playTone(200+Math.random()*100, 0.04, 'square', 0.02);
    } else {
      clearInterval(enemyAnimInterval);
      setTimeout(resolveEnemyTurn, 600);
    }
  }, speed);
}

function resolveEnemyTurn() {
  if (Math.random() < enemySuccessChance) {
    const dmg = 8 + Math.floor(Math.random()*7);
    playerHp -= dmg;
    totalTaken += dmg;
    $('turnIndicator').textContent = "🔥 Dragon hits! -" + dmg + " HP";
    $('turnIndicator').style.color = '#f44';
    sfxEnemyHit();
    showDmg('player', '-'+dmg);
  } else {
    $('turnIndicator').textContent = "💨 Dragon fumbles!";
    $('turnIndicator').style.color = '#8f8';
    playTone(150,0.2,'triangle',0.1);
  }
  updateHP();
  if (playerHp <= 0) { setTimeout(()=>endGame(false), 1000); return; }
  round++;
  $('roundInfo').textContent = 'Round '+round+'/'+maxRounds;
  setTimeout(startPlayerTurn, 1200);
}

function showDmg(target, text) {
  const el = document.createElement('div');
  el.className = 'dmg-popup ' + (target==='enemy'?'dmg-dealt':'dmg-taken');
  el.textContent = text;
  el.style.left = (target==='enemy'?'20%':'60%');
  el.style.top = target==='enemy'?'15%':'75%';
  $('game').appendChild(el);
  setTimeout(()=>el.remove(), 1000);
}

function endGame(won) {
  state = 'ended';
  clearInterval(timerInterval); clearInterval(enemyAnimInterval);
  showScreen('endScreen');
  const title = $('resultTitle');
  if (won) { title.textContent = '🏆 VICTORY!'; title.className = 'win-text'; sfxWin(); }
  else { title.textContent = '💀 DEFEATED'; title.className = 'lose-text'; sfxLose(); }
  $('resultStats').innerHTML =
    'Rounds Won: '+roundsWon+'<br>' +
    'Damage Dealt: '+totalDealt+'<br>' +
    'Damage Taken: '+totalTaken+'<br>' +
    'Final Round: '+Math.min(round, maxRounds)+'/'+maxRounds;
}

function resize() { /* CSS handles this */ }
window.addEventListener('resize', resize);

window.addEventListener('keydown', e => {
  if (state === 'start') {
    if (e.code === 'Space' || e.code === 'Enter') startGame();
    return;
  }
  if (state === 'ended') {
    if (e.code === 'Space' || e.code === 'Enter') startGame();
    return;
  }
  if (state === 'playerTurn') {
    if (e.key.length === 1 && /[a-z]/i.test(e.key)) playerType(e.key.toLowerCase());
  }
});

$('restartBtn').addEventListener('click', () => { initAudio(); startGame(); });

try { localStorage.getItem('x'); } catch(e){}
})();
</script>
