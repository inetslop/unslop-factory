```html
<!DOCTYPE html>
<html>
<head>
<meta charset="UTF-8">
<title>Endless Pac-Man</title>
<style>
* { margin: 0; padding: 0; box-sizing: border-box; }
body { background: #000; overflow: hidden; width: 100vw; height: 100vh; display: flex; align-items: center; justify-content: center; }
canvas { display: block; }
</style>
</head>
<body>
<canvas id="game"></canvas>
<script>
(function() {
const canvas = document.getElementById('game');
const ctx = canvas.getContext('2d');

let W, H;
function resize() {
W = window.innerWidth;
H = window.innerHeight;
canvas.width = W;
canvas.height = H;
}
resize();
window.addEventListener('resize', resize);

// Audio
let audioCtx = null;
function initAudio() {
if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
}
function playSound(freq, type, duration, volume) {
if (!audioCtx) return;
const osc = audioCtx.createOscillator();
const gain = audioCtx.createGain();
osc.type = type;
osc.frequency.value = freq;
gain.gain.setValueAtTime(volume || 0.1, audioCtx.currentTime);
gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + duration);
osc.connect(gain);
gain.connect(audioCtx.destination);
osc.start();
osc.stop(audioCtx.currentTime + duration);
}
function playWaka() {
playSound(400, 'square', 0.08, 0.05);
}
function playPowerUp() {
playSound(200, 'sawtooth', 0.3, 0.1);
setTimeout(() => playSound(300, 'sawtooth', 0.3, 0.1), 100);
}
function playEatGhost() {
playSound(600, 'square', 0.15, 0.12);
setTimeout(() => playSound(800, 'square', 0.15, 0.12), 80);
}
function playDeath() {
playSound(300, 'sawtooth', 0.4, 0.15);
setTimeout(() => playSound(200, 'sawtooth', 0.4, 0.15), 150);
}
function playGameOver() {
playSound(200, 'sawtooth', 0.6, 0.15);
setTimeout(() => playSound(150, 'sawtooth', 0.8, 0.15), 300);
}

// Game state
let gameState = 'start'; // start, playing, gameover
let score = 0;
let highScore = 0;
let lives = 3;
let speed = 4;
let distance = 0;
let time = 0;
let frameCount = 0;

try {
highScore = parseInt(localStorage.getItem('endlessPacHigh') || '0');
} catch(e) {}

// Pac-Man
const pac = {
x: 150, y: 0, lane: 1, targetLane: 1,
radius: 20, mouthAngle: 0, mouthDir: 1
};

// Lanes
function getLaneY(lane) {
return H * 0.3 + lane * (H * 0.2);
}

// Dots
let dots = [];
let powerPellets = [];
let ghosts = [];
let particles = [];
let popups = [];

function spawnDot() {
const lane = Math.floor(Math.random() * 3);
dots.push({ x: W + 20, y: getLaneY(lane), radius: 5 });
}

function spawnPowerPellet() {
const lane = Math.floor(Math.random() * 3);
powerPellets.push({ x: W + 20, y: getLaneY(lane), radius: 12, pulse: 0 });
}

function spawnGhost() {
const lane = Math.floor(Math.random() * 3);
const colors = ['#ff0000', '#00ffff', '#ff8c00', '#ff69b4'];
ghosts.push({
x: W + 50, y: getLaneY(lane), lane: lane, targetLane: lane,
radius: 18, color: colors[Math.floor(Math.random() * colors.length)],
speed: speed * (0.6 + Math.random() * 0.4),
eaten: false, eatTimer: 0, wobble: Math.random() * Math.PI * 2
});
}

function spawnParticles(x, y, color, count) {
for (let i = 0; i < count; i++) {
const angle = Math.random() * Math.PI * 2;
const spd = 1 + Math.random() * 4;
particles.push({ x, y, vx: Math.cos(angle) * spd, vy: Math.sin(angle) * spd, life: 30, color });
}
}

function spawnPopup(x, y, text, color) {
popups.push({ x, y, text, color, life: 60 });
}

function resetGame() {
score = 0;
lives = 3;
speed = 4;
distance = 0;
time = 0;
frameCount = 0;
pac.lane = 1;
pac.targetLane = 1;
pac.y = getLaneY(1);
dots = [];
powerPellets = [];
ghosts = [];
particles = [];
popups = [];
// Initial dots
for (let i = 0; i < 20; i++) {
const lane = Math.floor(Math.random() * 3);
dots.push({ x: W + 100 + i * 60, y: getLaneY(lane), radius: 5 });
}
// Initial power pellet
spawnPowerPellet();
gameState = 'playing';
}

// Input
window.addEventListener('keydown', function(e) {
initAudio();
if (gameState === 'start') {
resetGame();
return;
}
if (gameState === 'gameover') {
if (e.key === ' ' || e.key === 'Enter' || e.key === 'r' || e.key === 'R') {
resetGame();
}
return;
}
if (gameState === 'playing') {
if (e.key === 'ArrowUp' || e.key === 'w' || e.key === 'W') {
pac.targetLane = Math.max(0, pac.targetLane - 1);
playSound(500, 'sine', 0.05, 0.05);
}
if (e.key === 'ArrowDown' || e.key === 's' || e.key === 'S') {
pac.targetLane = Math.min(2, pac.targetLane + 1);
playSound(500, 'sine', 0.05, 0.05);
}
}
});

// Touch support
let touchStartY = null;
canvas.addEventListener('touchstart', function(e) {
initAudio();
if (gameState === 'start') { resetGame(); return; }
if (gameState === 'gameover') { resetGame(); return; }
touchStartY = e.touches[0].clientY;
});
canvas.addEventListener('touchend', function(e) {
if (gameState !== 'playing' || touchStartY === null) return;
const endY = e.changedTouches[0].clientY;
const diff = endY - touchStartY;
if (Math.abs(diff) > 20) {
if (diff < 0 && pac.targetLane > 0) pac.targetLane--;
else if (diff > 0 && pac.targetLane < 2) pac.targetLane++;
}
touchStartY = null;
});

// Update
function update() {
if (gameState !== 'playing') return;

time++;
frameCount++;
distance += speed;
speed = 4 + Math.floor(distance / 500) * 0.3;

// Pac-Man movement
const targetY = getLaneY(pac.targetLane);
pac.y += (targetY - pac.y) * 0.15;
pac.mouthAngle += 0.12 * pac.mouthDir;
if (pac.mouthAngle > 0.4 || pac.mouthAngle < -0.1) pac.mouthDir *= -1;

// Spawn dots
if (frameCount % Math.max(15, 30 - Math.floor(distance / 200)) === 0) spawnDot();
if (frameCount % 800 === 0) spawnPowerPellet();
if (frameCount % Math.max(40, 80 - Math.floor(distance / 100)) === 0) spawnGhost();

// Move dots
for (let i = dots.length - 1; i >= 0; i--) {
dots[i].x -= speed;
if (dots[i].x < -20) { dots.splice(i, 1); continue; }
// Collect
const dx = dots[i].x - pac.x;
const dy = dots[i].y - pac.y;
if (Math.sqrt(dx * dx + dy * dy) < pac.radius + dots[i].radius) {
score += 10;
spawnParticles(dots[i].x, dots[i].y, '#ffff00', 5);
playWaka();
dots.splice(i, 1);
}
}

// Move power pellets
for (let i = powerPellets.length - 1; i >= 0; i--) {
powerPellets[i].x -= speed;
powerPellets[i].pulse += 0.1;
if (powerPellets[i].x < -20) { powerPellets.splice(i, 1); continue; }
const dx = powerPellets[i].x - pac.x;
const dy = powerPellets[i].y - pac.y;
if (Math.sqrt(dx * dx + dy * dy) < pac.radius + powerPellets[i].radius) {
score += 50;
spawnPopup(pac.x, pac.y - 30, '+50', '#ff0');
playPowerUp();
// Make all ghosts vulnerable
for (let g of ghosts) { g.eaten = true; g.eatTimer = 180; }
powerPellets.splice(i, 1);
}
}

// Move ghosts
for (let i = ghosts.length - 1; i >= 0; i--) {
const g = ghosts[i];
g.x -= speed * (g.eaten ? 0.3 : 1) + (g.eaten ? 0 : g.speed * 0.5);
g.wobble += 0.08;

if (!g.eaten) {
// Chase pac-man
const targetLane = pac.lane;
if (frameCount % 30 === 0) g.targetLane = targetLane + Math.floor(Math.random() * 2) - 1;
g.targetLane = Math.max(0, Math.min(2, g.targetLane));
g.y += (getLaneY(g.targetLane) - g.y) * 0.05;
}

if (g.eaten) {
g.eatTimer--;
if (g.eatTimer <= 0) g.eaten = false;
}

// Remove off-screen
if (g.x < -50) { ghosts.splice(i, 1); continue; }

// Collision with pac-man
const dx = g.x - pac.x;
const dy = g.y - pac.y;
const dist = Math.sqrt(dx * dx + dy * dy);
if (dist < pac.radius + g.radius - 5) {
if (g.eaten) {
score += 100;
spawnPopup(g.x, g.y - 20, '+100', '#0f0');
playEatGhost();
spawnParticles(g.x, g.y, g.color, 10);
ghosts.splice(i, 1);
} else {
lives--;
playDeath();
spawnParticles(pac.x, pac.y, '#ffff00', 15);
if (lives <= 0) {
gameState = 'gameover';
playGameOver();
try { localStorage.setItem('endlessPacHigh', String(highScore)); } catch(e) {}
} else {
// Reset position slightly
pac.targetLane = 1;
pac.y = getLaneY(1);
ghosts = ghosts.filter(gg => gg.x > pac.x + 100);
}
}
}
}

// Particles
for (let i = particles.length - 1; i >= 0; i--) {
particles[i].x += particles[i].vx;
particles[i].y += particles[i].vy;
particles[i].life--;
if (particles[i].life <= 0) particles.splice(i, 1);
}

// Popups
for (let i = popups.length - 1; i >= 0; i--) {
popups[i].y -= 0.5;
popups[i].life--;
if (popups[i].life <= 0) popups.splice(i, 1);
}

// High score update
if (score > highScore) highScore = score;
}

// Draw
function draw() {
ctx.fillStyle = '#0a0a1a';
ctx.fillRect(0, 0, W, H);

if (gameState === 'start') {
drawStartScreen();
return;
}

// Draw lanes
for (let i = 0; i < 3; i++) {
const y = getLaneY(i);
ctx.strokeStyle = '#1a1a3a';
ctx.lineWidth = 2;
ctx.beginPath();
ctx.moveTo(0, y - 30);
ctx.lineTo(W, y - 30);
ctx.stroke();
ctx.beginPath();
ctx.moveTo(0, y + 30);
ctx.lineTo(W, y + 30);
ctx.stroke();
}

// Draw dots
for (const d of dots) {
ctx.fillStyle = '#ffff00';
ctx.beginPath();
ctx.arc(d.x, d.y, d.radius, 0, Math.PI * 2);
ctx.fill();
}

// Draw power pellets
for (const p of powerPellets) {
const s = p.radius + Math.sin(p.pulse) * 4;
ctx.fillStyle = '#ffff00';
ctx.beginPath();
ctx.arc(p.x, p.y, s, 0, Math.PI * 2);
ctx.fill();
ctx.strokeStyle = '#fff';
ctx.lineWidth = 2;
ctx.stroke();
}

// Draw ghosts
for (const g of ghosts) {
if (g.eaten) {
ctx.globalAlpha = 0.5;
drawGhost(g.x, g.y, g.radius, '#888');
ctx.globalAlpha = 1;
} else {
drawGhost(g.x, g.y, g.radius, g.color);
}
}

// Draw pac-man
drawPacMan(pac.x, pac.y, pac.radius);

// Particles
for (const p of particles) {
ctx.globalAlpha = p.life / 30;
ctx.fillStyle = p.color;
ctx.beginPath();
ctx.arc(p.x, p.y, 3, 0, Math.PI * 2);
ctx.fill();
}
ctx.globalAlpha = 1;

// Popups
for (const p of popups) {
ctx.globalAlpha = p.life / 60;
ctx.fillStyle = p.color;
ctx.font = 'bold 18px monospace';
ctx.textAlign = 'center';
ctx.fillText(p.text, p.x, p.y);
}
ctx.globalAlpha = 1;

// HUD
drawHUD();

if (gameState === 'gameover') drawGameOver();
}

function drawPacMan(x, y, r) {
ctx.fillStyle = '#ffff00';
ctx.beginPath();
const m = pac.mouthAngle * 2;
ctx.moveTo(x, y);
ctx.arc(x, y, r, m, Math.PI * 2 - m);
ctx.closePath();
ctx.fill();

// Eye
ctx.fillStyle = '#000';
ctx.beginPath();
ctx.arc(x + r * 0.3, y - r * 0.4, r * 0.15, 0, Math.PI * 2);
ctx.fill();
}

function drawGhost(x, y, r, color) {
// Body
ctx.fillStyle = color;
ctx.beginPath();
ctx.arc(x, y - r * 0.2, r, Math.PI, 0);
ctx.lineTo(x + r, y + r * 0.6);
// Wavy bottom
const waveCount = 3;
for (let i = 0; i < waveCount; i++) {
const wx = x + r - (i * 2 * r) / waveCount;
const wy = y + r * 0.4;
ctx.quadraticCurveTo(wx - r / waveCount, y + r * 0.8, wx - r / waveCount * 2, wy);
}
ctx.lineTo(x - r, y + r * 0.6);
ctx.closePath();
ctx.fill();

// Eyes
const eyeOffset = Math.sin(g_time * 0.1) * 3;
ctx.fillStyle = '#fff';
ctx.beginPath();
ctx.arc(x - r * 0.3, y - r * 0.2, r * 0.25, 0, Math.PI * 2);
ctx.arc(x + r * 0.3, y - r * 0.2, r * 0.25, 0, Math.PI * 2);
ctx.fill();
ctx.fillStyle = '#00f';
ctx.beginPath();
ctx.arc(x - r * 0.35, y - r * 0.15, r * 0.12, 0, Math.PI * 2);
ctx.arc(x + r * 0.25, y - r * 0.15, r * 0.12, 0, Math.PI * 2);
ctx.fill();
}

let g_time = 0;

function drawHUD() {
ctx.fillStyle = '#fff';
ctx.font = 'bold 20px monospace';
ctx.textAlign = 'left';
ctx.fillText('SCORE: ' + score, 20, 35);
ctx.textAlign = 'right';
ctx.fillText('HI: ' + highScore, W - 20, 35);

// Lives
for (let i = 0; i < lives; i++) {
ctx.fillStyle = '#ffff00';
ctx.beginPath();
ctx.arc(W / 2 - 40 + i * 30, 30, 10, 0.3, Math.PI * 2 - 0.3);
ctx.lineTo(W / 2 - 40 + i * 30, 30);
ctx.closePath();
ctx.fill();
}

// Speed indicator
ctx.fillStyle = '#888';
ctx.font = '14px monospace';
ctx.textAlign = 'left';
ctx.fillText('SPEED: ' + Math.floor(speed * 10) / 10, 20, 60);
}

function drawStartScreen() {
ctx.fillStyle = '#ffff00';
ctx.font = 'bold 48px monospace';
ctx.textAlign = 'center';
ctx.fillText('ENDLESS PAC-MAN', W / 2, H * 0.3);

ctx.font = '20px monospace';
ctx.fillStyle = '#fff';
ctx.fillText('Arrow Keys / W,S to move between lanes', W / 2, H * 0.45);
ctx.fillText('Collect dots (+10) and power pellets (+50)', W / 2, H * 0.5);
ctx.fillText('Eat ghosts when vulnerable (+100)', W / 2, H * 0.55);
ctx.fillText('Avoid ghosts or lose a life!', W / 2, H * 0.6);

ctx.fillStyle = '#ffff00';
ctx.font = '24px monospace';
const blink = Math.floor(Date.now() / 500) % 2;
if (blink) ctx.fillText('PRESS ANY KEY TO START', W / 2, H * 0.75);

if (highScore > 0) {
ctx.fillStyle = '#0f0';
ctx.font = '18px monospace';
ctx.fillText('HIGH SCORE: ' + highScore, W / 2, H * 0.85);
}

// Draw decorative pac-man
drawPacMan(W / 2 - 200, H * 0.3, 25);
g_time = Date.now();
for (let i = 0; i < 4; i++) {
const colors = ['#ff0000', '#00ffff', '#ff8c00', '#ff69b4'];
drawGhost(W / 2 + 150 + i * 50, H * 0.3, 20, colors[i]);
}
}

function drawGameOver() {
ctx.fillStyle = 'rgba(0,0,0,0.7)';
ctx.fillRect(0, 0, W, H);

ctx.fillStyle = '#ff0000';
ctx.font = 'bold 48px monospace';
ctx.textAlign = 'center';
ctx.fillText('GAME OVER', W / 2, H * 0.35);

ctx.fillStyle = '#fff';
ctx.font = '24px monospace';
ctx.fillText('SCORE: ' + score, W / 2, H * 0.48);
ctx.fillText('DISTANCE: ' + Math.floor(distance), W / 2, H * 0.55);

if (score >= highScore && score > 0) {
ctx.fillStyle = '#ff0';
ctx.fillText('NEW HIGH SCORE!', W / 2, H * 0.63);
}

ctx.fillStyle = '#ffff00';
ctx.font = '20px monospace';
const blink = Math.floor(Date.now() / 500) % 2;
if (blink) ctx.fillText('PRESS SPACE / R TO RESTART', W / 2, H * 0.78);
}

// Game loop
function gameLoop() {
g_time = Date.now();
update();
draw();
requestAnimationFrame(gameLoop);
}
gameLoop();
})();
</script>
</body>
</html>
```
