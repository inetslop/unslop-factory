function playSound(type) {
if (!audioCtx) return;
const now = audioCtx.currentTime;
switch(type) {
case 'move':
playTone(200, 0.05, 'square', 0.08);
break;
case 'push':
playTone(120, 0.08, 'square', 0.12);
break;
case 'goal':
playTone(440, 0.1, 'sine', 0.15);
setTimeout(()=>{ if(audioCtx) playTone(660, 0.15, 'sine', 0.15); }, 80);
break;
case 'win':
const notes = [523, 659, 784, 1047];
notes.forEach((f,i) => {
if (audioCtx) playTone(f, 0.2, 'sine', 0.2 - i*0.03);
});
break;
case 'undo':
playTone(300, 0.06, 'triangle', 0.1);
break;
}
}

function playTone(freq, dur, type, vol) {
const osc = audioCtx.createOscillator();
const gain = audioCtx.createGain();
osc.type = type;
osc.frequency.value = freq;
gain.gain.setValueAtTime(vol, audioCtx.currentTime);
gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
osc.connect(gain);
gain.connect(audioCtx.destination);
osc.start();
osc.stop(audioCtx.currentTime + dur);
}

function parseLevel(lvIdx) {
const lv = LEVELS[lvIdx];
grid = [];
boxes = [];
goals = [];
walls = [];
let rows = lv.length;
let cols = 0;
for (let r = 0; r < rows; r++) {
cols = Math.max(cols, lv[r].length);
}
grid = Array.from({length: rows}, () => Array(cols).fill(' '));

for (let r = 0; r < rows; r++) {
for (let c = 0; c < cols; c++) {
const ch = lv[r][c] || ' ';
switch(ch) {
case '#': grid[r][c] = '#'; walls.push([c,r]); break;
case '.': grid[r][c] = '.'; goals.push([c,r]); break;
case '$': grid[r][c] = '$'; boxes.push({x:c, y:r, px:c, py:r}); break;
case '@': grid[r][c] = ' '; playerX = c; playerY = r; break;
case '*': grid[r][c] = '.'; goals.push([c,r]); boxes.push({x:c,y:r,px:c,py:r}); break;
case '+': grid[r][c] = '.'; goals.push([c,r]); playerX=c; playerY=r; break;
default: grid[r][c] = ' ';
}
}
}

rows = grid.length; cols = grid[0].length;
calcTileSize(rows, cols);
moves = 0;
history = [];
levelComplete = false;
animProgress = 1;
prevPlayerX = playerX; prevPlayerY = playerY;
prevBoxes = boxes.map(b=>({x:b.x,y:b.y}));
hudLevel.textContent = lvIdx + 1;
hudMoves.textContent = '0';
}

function calcTileSize(rows, cols) {
const w = window.innerWidth, h = window.innerHeight;
tileSize = Math.min(Math.floor((w - 40) / cols), Math.floor((h - 80) / rows));
if (tileSize < 20) tileSize = 20;
gridOffsetX = Math.floor((w - cols * tileSize) / 2);
gridOffsetY = Math.floor((h - rows * tileSize) / 2);
canvas.width = w;
canvas.height = h;
}

function resize() {

canvas.height = window.innerHeight;
gridOffsetX = Math.floor((canvas.width - (grid[0]?.length||10) * tileSize) / 2);
gridOffsetY = Math.floor((canvas.height - (grid.length||6) * tileSize) / 2);
}

window.addEventListener('resize', resize);

function isWall(x, y) {
if (y < 0 || y >= grid.length || x < 0 || x >= grid[0].length) return true;
return grid[y][x] === '#';
}

function isBox(x, y) {
return boxes.some(b => b.x === x && b.y === y);
}

function isGoal(x, y) {
return goals.some(g => g[0] === x && g[1] === y);
}

function checkWin() {
return boxes.every(b => isGoal(b.x, b.y));
}

function saveState() {
history.push({
px: playerX, py: playerY,
boxes: boxes.map(b=>({x:b.x, y:b.y}))
});
if (history.length > 100) history.shift();
}

function undo() {
if (history.length === 0 || levelComplete) return;
const s = history.pop();
playerX = s.px; playerY = s.py;
boxes.forEach((b,i) => { b.x = s.boxes[i].x; b.y = s.boxes[i].y; });
moves--;
hudMoves.textContent = moves;
playSound('undo');
}

function tryMove(dx, dy) {
if (!gameStarted || levelComplete || allComplete) return;
const nx = playerX + dx, ny = playerY + dy;
if (isWall(nx, ny)) return;

saveState();
prevPlayerX = playerX; prevPlayerY = playerY;
prevBoxes = boxes.map(b=>({x:b.x,y:b.y}));

if (isBox(nx, ny)) {
const bx = nx + dx, by = ny + dy;
if (isWall(bx, by) || isBox(bx, by)) return;
// Push box
const box = boxes.find(b => b.x === nx && b.y === ny);
box.px = box.x; box.py = box.y;
box.x = bx; box.y = by;
playerX = nx; playerY = ny;
moves++;
hudMoves.textContent = moves;
animProgress = 0;
if (isGoal(bx, by)) {
playSound('goal');
} else {
playSound('push');
}
if (checkWin()) {
levelComplete = true;
totalMoves += moves;
setTimeout(()=> playSound('win'), 300);
}
} else {
playerX = nx; playerY = ny;
moves++;
hudMoves.textContent = moves;
animProgress = 0;
playSound('move');
}
}

function draw() {

}
}
}

// Draw goals

const cy = gridOffsetY + gy * tileSize + tileSize/2;

ctx.arc(cx, cy, tileSize*0.18, 0, Math.PI*2);

ctx.arc(cx, cy, tileSize*0.1, 0, Math.PI*2);
ctx.fillStyle = `rgba(78, 205, 196, ${pulse})`;
ctx.fill();
});

boxes.forEach((b) => {

const y = gridOffsetY + by * tileSize;
const pad = tileSize * 0.1;
const size = tileSize - pad*2;

// Box shadow
ctx.fillStyle = 'rgba(0,0,0,0.3)';
ctx.fillRect(x+pad+2, y+pad+2, size, size);
// Box body
const onGoal = isGoal(b.x, b.y);
ctx.fillStyle = onGoal ? '#4ecdc4' : '#e8a838';
ctx.fillRect(x+pad, y+pad, size, size);
// Box border
ctx.strokeStyle = onGoal ? '#2d9e96' : '#c07818';
ctx.lineWidth = 2;
ctx.strokeRect(x+pad, y+pad, size, size);
// Box inner cross
ctx.strokeStyle = onGoal ? 'rgba(255,255,255,0.3)' : 'rgba(255,255,255,0.2)';
ctx.lineWidth = 1;
const mid = tileSize/2;
ctx.beginPath();
ctx.moveTo(x+pad+size*0.3, y+pad+size*0.3);
ctx.lineTo(x+pad+size*0.7, y+pad+size*0.7);
ctx.moveTo(x+pad+size*0.7, y+pad+size*0.3);
ctx.lineTo(x+pad+size*0.3, y+pad+size*0.7);
ctx.stroke();
});

// Draw player
const px = lerp(prevPlayerX !== undefined ? prevPlayerX : playerX, playerX);
const py = lerp(prevPlayerY !== undefined ? prevPlayerY : playerY, playerY);
const pcx = gridOffsetX + px * tileSize + tileSize/2;
const pcy = gridOffsetY + py * tileSize + tileSize/2;

// Player glow
ctx.beginPath();
ctx.arc(pcx, pcy, tileSize*0.35, 0, Math.PI*2);
ctx.fillStyle = 'rgba(65, 180, 240, 0.2)';
ctx.fill();
// Player body
ctx.beginPath();
ctx.arc(pcx, pcy, tileSize*0.3, 0, Math.PI*2);
const grad = ctx.createRadialGradient(pcx - tileSize*0.08, pcy - tileSize*0.08, 0, pcx, pcy, tileSize*0.3);
grad.addColorStop(0, '#7dd3fc');
grad.addColorStop(1, '#2196f3');
ctx.fillStyle = grad;
ctx.fill();
// Player eyes
const eyeOff = tileSize * 0.1;
ctx.fillStyle = '#fff';
ctx.beginPath();
ctx.arc(pcx - eyeOff*0.5, pcy - eyeOff*0.3, tileSize*0.06, 0, Math.PI*2);
ctx.arc(pcx + eyeOff*0.5, pcy - eyeOff*0.3, tileSize*0.06, 0, Math.PI*2);
ctx.fill();

// Level complete overlay
if (levelComplete) {
const alpha = Math.min(1, animFrame * 0.02);
ctx.fillStyle = `rgba(78, 205, 196, ${alpha * 0.1})`;
ctx.fillRect(0, 0, canvas.width, canvas.height);
ctx.fillStyle = `rgba(255,255,255,${alpha})`;
ctx.font = `bold ${Math.min(tileSize*1.2, 48)}px Courier New`;

ctx.fillText(level < LEVELS.length - 1 ? 'LEVEL COMPLETE! Press N' : 'ALL CLEAR!', canvas.width/2, canvas.height/2);

return;

switch(e.key) {
case 'ArrowUp': case 'w': case 'W': e.preventDefault(); tryMove(0,-1); break;
case 'ArrowDown': case 's': case 'S': e.preventDefault(); tryMove(0,1); break;
case 'ArrowLeft': case 'a': case 'A': e.preventDefault(); tryMove(-1,0); break;
case 'ArrowRight': case 'd': case 'D': e.preventDefault(); tryMove(1,0); break;
case 'u': case 'U': undo(); break;
case 'r': case 'R': restartLevel(); break;
case 'n': case 'N': if (levelComplete) nextLevel(); break;
case 'p': case 'P': if (!levelComplete && level > 0) prevLevel(); break;
}
});

// Touch support - swipe
let touchStartX = 0, touchStartY = 0;
document.addEventListener('touchstart', (e) => {
if (!gameStarted) { initAudio(); startGame(); return; }
touchStartX = e.touches[0].clientX;
touchStartY = e.touches[0].clientY;
});
document.addEventListener('touchend', (e) => {
if (!gameStarted || allComplete) return;
const dx = e.changedTouches[0].clientX - touchStartX;
const dy = e.changedTouches[0].clientY - touchStartY;
if (Math.abs(dx) > Math.abs(dy)) {
if (Math.abs(dx) > 20) tryMove(Math.sign(dx), 0);
} else {
if (Math.abs(dy) > 20) tryMove(0, Math.sign(dy));
}
});

// Init
canvas.width = window.innerWidth;
canvas.height = window.innerHeight;
gameLoop();
