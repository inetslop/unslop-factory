// Audio
let audioCtx = null;
function initAudio() {
    if (!audioCtx) {
        audioCtx = new (window.AudioContext || window.webkitAudioContext)();
    }
}
function playSound(freq, type, duration, vol) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = type;
    osc.frequency.value = freq;
    gain.gain.setValueAtTime(vol || 0.1, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + duration);
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + duration);
}
function playHit() { playSound(200, 'sawtooth', 0.3, 0.15); }
function playPhase() { playSound(800, 'sine', 0.2, 0.08); }
function playDash() { playSound(400, 'square', 0.1, 0.06); }
function playDeath() { playSound(100, 'sawtooth', 0.8, 0.2); }
function playShoot() { playSound(300 + Math.random()*200, 'square', 0.05, 0.03); }

let particles = [];
function spawnParticles(x, y, color, count, speed, life) {
    for (let i = 0; i < count; i++) {
        const angle = Math.random() * Math.PI * 2;
        const spd = (Math.random() * 0.7 + 0.3) * speed;
        particles.push({
            x, y,
            vx: Math.cos(angle) * spd,

let enemies = [];
function createEnemy(type, x, y) {
    let e = { type, x, y, angle: Math.random() * Math.PI * 2, fireTimer: 0, detectionRadius: 150 + Math.random() * 80, phase: Math.random() * Math.PI * 2 };
    if (type === 'spiral') { e.fireRate = 3; e.bulletSpeed = 2.5; }
    else if (type === 'ring') { e.fireRate = 90; e.bulletsPerShot = 16; e.bulletSpeed = 2; }
    else if (type === 'aimed') { e.fireRate = 45; e.spread = 3; e.bulletSpeed = 3.5; }
    else if (type === 'wave') { e.fireRate = 12; e.bulletSpeed = 2.8; }
    return e;
}

function spawnWave() {
    wave++;
    let count = Math.min(2 + wave, 8);
    for (let i = 0; i < count; i++) {
        let x, y, attempts = 0;
        do {
            x = 60 + Math.random() * (W - 120);
            y = 60 + Math.random() * (H - 120);
            attempts++;
        } while (Math.hypot(x - player.x, y - player.y) < 200 && attempts < 50);
        
        let types = ['spiral', 'ring', 'aimed', 'wave'];
        let type = types[Math.floor(Math.random() * Math.min(types.length, wave + 1))];
        if (wave < 2) type = 'spiral';
        else if (wave === 2) type = Math.random() < 0.5 ? 'ring' : 'spiral';
        enemies.push(createEnemy(type, x, y));
    }
}

function updateEnemy(e) {
    e.angle += 0.01;
    e.phase += 0.02;
    
    const dx = player.x - e.x;
    const dy = player.y - e.y;
    const dist = Math.hypot(dx, dy);
    const detected = !player.phased && dist < e.detectionRadius;
    
    if (detected) {
        player.exposure = Math.min(player.exposure + 0.3, 100);
    } else {
        player.exposure = Math.max(player.exposure - 0.5, 0);
    }
    
    e.fireTimer++;
    const rate = detected ? e.fireRate * 0.6 : e.fireRate;
    
    if (e.fireTimer >= rate) {
        e.fireTimer = 0;
        fireEnemy(e, detected);
    }
}

function fireEnemy(e, detected) {
    const angleToPlayer = Math.atan2(player.y - e.y, player.x - e.x);
    
    if (e.type === 'spiral') {
        const baseAngle = e.phase * 3;
        for (let i = 0; i < 2; i++) {
            const a = baseAngle + i * Math.PI;
            spawnBullet(e.x, e.y, Math.cos(a) * e.bulletSpeed, Math.sin(a) * e.bulletSpeed, 4, '#ff6633');
        }
    } else if (e.type === 'ring') {
        const count = e.bulletsPerShot + wave;
        for (let i = 0; i < count; i++) {
            const a = (i / count) * Math.PI * 2 + e.phase;
            spawnBullet(e.x, e.y, Math.cos(a) * e.bulletSpeed, Math.sin(a) * e.bulletSpeed, 5, '#ff3366');
        }
    } else if (e.type === 'aimed') {
        const spread = detected ? 5 : 3;
        for (let i = 0; i < spread; i++) {
            const offset = (i - (spread-1)/2) * 0.25;
            const a = angleToPlayer + offset;
            spawnBullet(e.x, e.y, Math.cos(a) * e.bulletSpeed, Math.sin(a) * e.bulletSpeed, 4, '#ffaa00');
        }
    } else if (e.type === 'wave') {
        const a = angleToPlayer + (Math.random() - 0.5) * 0.5;
        spawnBullet(e.x, e.y, Math.cos(a) * e.bulletSpeed, Math.sin(a) * e.bulletSpeed, 6, '#cc44ff');
    }
    
    if (audioCtx && Math.random() < 0.3) playShoot();
}

function updatePlayer() {
    let mx = 0, my = 0;
    if (keys['KeyW'] || keys['ArrowUp']) my -= 1;
    if (keys['KeyS'] || keys['ArrowDown']) my += 1;
    if (keys['KeyA'] || keys['ArrowLeft']) mx -= 1;
    if (keys['KeyD'] || keys['ArrowRight']) mx += 1;
    
    if (mx !== 0 && my !== 0) { mx *= 0.707; my *= 0.707; }
    
    const speed = player.phased ? 2 : 4;
    player.vx = mx * speed;
    player.vy = my * speed;
    
    if (player.dashTimer > 0) {
        player.dashTimer--;
        player.vx = player.dashVX * 12;
        player.vy = player.dashVY * 12;
    } else {
        player.dashCooldown = Math.max(0, player.dashCooldown - 1);
    }
    
    if ((keys['ShiftLeft'] || keys['ShiftRight']) && player.dashCooldown <= 0 && (mx !== 0 || my !== 0)) {
        const len = Math.hypot(mx, my);
        player.dashVX = mx / len;
        player.dashVY = my / len;
        player.dashTimer = 8;
        player.dashCooldown = 30;
        playDash();
    }
    
    if ((keys['Space']) && !player.phased && player.phaseCooldown <= 0) {
        player.phased = true;
        player.phaseTimer = 45;
        player.phaseCooldown = 90;
        playPhase();
        spawnParticles(player.x, player.y, '#00ffff', 12, 3, 25);
    }
    
    if (player.phased) {
        player.phaseTimer--;
        if (player.phaseTimer <= 0) {
            player.phased = false;
        }
    }
    player.phaseCooldown = Math.max(0, player.phaseCooldown - 1);
    
    player.x += player.vx;
    player.y += player.vy;
    
    player.x = Math.max(player.radius, Math.min(W - player.radius, player.x));
    player.y = Math.max(player.radius, Math.min(H - player.radius, player.y));
    
    player.invincible = Math.max(0, player.invincible - 1);
    
    // Exposure damage
    if (player.exposure > 75 && player.invincible <= 0) {
        // Extra vulnerability when very exposed
    }
}

function checkCollisions() {
    if (player.invincible > 0) return;
    
    for (let i = bullets.length - 1; i >= 0; i--) {
        const b = bullets[i];
        const dx = player.x - b.x;
        const dy = player.y - b.y;
        if (Math.hypot(dx, dy) < player.radius + b.radius) {
            if (!player.phased) {
                player.hp--;
                player.invincible = 60;
                screenShake = 10;
                playHit();
                spawnParticles(player.x, player.y, '#ff4444', 20, 5, 30);
                if (player.hp <= 0) {
                    gameState = STATE.DEAD;
                    playDeath();
                    spawnParticles(player.x, player.y, '#00ffff', 40, 6, 50);
                    try { localStorage.setItem('ghostProtocolHigh', String(Math.max(highScore, score))); } catch(e) {}
                }
            }
            bullets.splice(i, 1);
        }
    }
}

function updateBullets() {
    for (let i = bullets.length - 1; i >= 0; i--) {
        const b = bullets[i];
        b.x += b.vx;
        b.y += b.vy;
        b.life--;
        if (b.x < -20 || b.x > W + 20 || b.y < -20 || b.y > H + 20 || b.life <= 0) {
            bullets.splice(i, 1);
        }
    }
}

function updateParticles() {
    for (let i = particles.length - 1; i >= 0; i--) {
        const p = particles[i];
        p.x += p.vx;
        p.y += p.vy;
        p.vx *= 0.92;
        p.vy *= 0.92;
        p.life--;
        if (p.life <= 0) particles.splice(i, 1);
    }
}

function update() {
    gameTime++;
    score += Math.floor(gameTime / 10);
    
    // Wave timer
    waveTimer--;
    if (waveTimer <= 0) {
        spawnWave();
        waveTimer = 600 + wave * 30;
    }
    
    updatePlayer();
    for (let e of enemies) updateEnemy(e);
    updateBullets();
    checkCollisions();
    updateParticles();
    
    screenShake *= 0.85;
    shakeX = (Math.random() - 0.5) * screenShake;
    shakeY = (Math.random() - 0.5) * screenShake;
}

function drawEnemies() {
    for (let e of enemies) {
        const dx = player.x - e.x;
        const dy = player.y - e.y;
        const dist = Math.hypot(dx, dy);
        const detected = !player.phased && dist < e.detectionRadius;
        
        ctx.save();
        ctx.translate(e.x + shakeX, e.y + shakeY);
        
        // Detection range (faint)
        if (detected) {
            ctx.beginPath();
            ctx.arc(0, 0, e.detectionRadius, 0, Math.PI * 2);
            ctx.strokeStyle = 'rgba(255, 100, 50, 0.08)';
            ctx.lineWidth = 1;
            ctx.stroke();
        }
        
        // Body
        let color = detected ? '#ff4444' : '#667788';
        if (e.type === 'ring') color = detected ? '#ff3366' : '#556677';
        if (e.type === 'aimed') color = detected ? '#ffaa00' : '#556677';
        if (e.type === 'wave') color = detected ? '#cc44ff' : '#556677';
        
        // Glow
        const glow = ctx.createRadialGradient(0, 0, 0, 0, 0, 20);
        glow.addColorStop(0, detected ? 'rgba(255,100,50,0.3)' : 'rgba(100,120,140,0.15)');
        glow.addColorStop(1, 'rgba(0,0,0,0)');
        ctx.fillStyle = glow;
        ctx.beginPath();
        ctx.arc(0, 0, 20, 0, Math.PI * 2);
        ctx.fill();
        
        // Rotating shape
        const sides = e.type === 'spiral' ? 4 : e.type === 'ring' ? 6 : e.type === 'aimed' ? 3 : 5;
        ctx.beginPath();
        for (let i = 0; i < sides; i++) {
            const a = (i / sides) * Math.PI * 2 + e.angle;
            const r = 12;
            if (i === 0) ctx.moveTo(Math.cos(a) * r, Math.sin(a) * r);
            else ctx.lineTo(Math.cos(a) * r, Math.sin(a) * r);
        }
        ctx.closePath();
        ctx.strokeStyle = color;
        ctx.lineWidth = 2;
        ctx.stroke();
        
        // Center dot
        ctx.fillStyle = color;
        ctx.beginPath();
        ctx.arc(0, 0, 4, 0, Math.PI * 2);
        ctx.fill();
        
        ctx.restore();
    }
}

        ctx.save();
        ctx.translate(b.x + shakeX, b.y + shakeY);
        
        const glow = ctx.createRadialGradient(0, 0, 0, 0, 0, b.radius * 2);
        glow.addColorStop(0, b.color);
        glow.addColorStop(1, 'rgba(0,0,0,0)');
        ctx.fillStyle = glow;
        ctx.beginPath();
        ctx.arc(0, 0, b.radius * 2, 0, Math.PI * 2);
        ctx.fill();
        
        ctx.fillStyle = b.color;
        ctx.beginPath();
        ctx.arc(0, 0, b.radius, 0, Math.PI * 2);
        ctx.fill();
        
        ctx.restore();
    }
}

function drawParticles() {
    for (let p of particles) {
        const alpha = p.life / p.maxLife;
        ctx.fillStyle = p.color + Math.floor(alpha * 255).toString(16).padStart(2, '0');
        ctx.beginPath();
        ctx.arc(p.x + shakeX, p.y + shakeY, 2 * alpha, 0, Math.PI * 2);
        ctx.fill();
    }
}

    ctx.fillText('DETECTED', W/2, H/3);
    
    ctx.fillStyle = '#ffffff';
    ctx.font = '20px monospace';
    ctx.fillText('SCORE: ' + score, W/2, H/2 - 10);
    
    if (score >= highScore && score > 0) {
        ctx.fillStyle = '#ffaa00';
        ctx.fillText('NEW HIGH SCORE!', W/2, H/2 + 25);
    } else {
        ctx.fillStyle = 'rgba(255,255,255,0.5)';
        ctx.font = '14px monospace';
        ctx.fillText('HIGH: ' + highScore, W/2, H/2 + 25);
    }
    
    ctx.fillStyle = Math.sin(Date.now() * 0.005) > 0 ? '#ffffff' : 'rgba(255,255,255,0.4)';
    ctx.font = '16px monospace';
    ctx.fillText('PRESS R TO RESTART', W/2, H - 40);
}

function render() {
    ctx.fillStyle = '#0a0a12';
    ctx.fillRect(0, 0, W, H);
    
    if (gameState === STATE.MENU) {
        drawMenu();
        return;
    }
    
    drawGrid();
    drawEnemies();
    drawBullets();
    drawPlayer();
    drawParticles();
    drawHUD();
    
    if (gameState === STATE.DEAD) {
        drawDead();
    }
}

// Main loop
function loop() {
    if (gameState === STATE.PLAYING) {
        update();
    }
    render();
    requestAnimationFrame(loop);
}

loop();
