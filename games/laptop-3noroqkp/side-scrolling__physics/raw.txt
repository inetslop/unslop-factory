let audioCtx = null;
function initAudio() {
    if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
}
function playSound(type) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    const now = audioCtx.currentTime;
    switch(type) {
        case 'jump':
            osc.type = 'sine';
            osc.frequency.setValueAtTime(300, now);
            osc.frequency.linearRampToValueAtTime(600, now + 0.1);
            gain.gain.setValueAtTime(0.15, now);
            gain.gain.linearRampToValueAtTime(0, now + 0.15);
            osc.start(now); osc.stop(now + 0.15);
            break;
        case 'land':
            osc.type = 'sine';
            osc.frequency.setValueAtTime(150, now);
            osc.frequency.linearRampToValueAtTime(80, now + 0.1);
            gain.gain.setValueAtTime(0.1, now);
            gain.gain.linearRampToValueAtTime(0, now + 0.1);
            osc.start(now); osc.stop(now + 0.1);
            break;
        case 'collect':
            osc.type = 'sine';
            osc.frequency.setValueAtTime(800, now);
            osc.frequency.linearRampToValueAtTime(1200, now + 0.1);
            gain.gain.setValueAtTime(0.12, now);
            gain.gain.linearRampToValueAtTime(0, now + 0.2);
            osc.start(now); osc.stop(now + 0.2);
            break;
        case 'die':
            osc.type = 'sawtooth';
            osc.frequency.setValueAtTime(400, now);
            osc.frequency.linearRampToValueAtTime(100, now + 0.3);
            gain.gain.setValueAtTime(0.15, now);
            gain.gain.linearRampToValueAtTime(0, now + 0.4);
            osc.start(now); osc.stop(now + 0.4);
            break;
        case 'win':
            osc.type = 'sine';
            osc.frequency.setValueAtTime(523, now);
            osc.frequency.setValueAtTime(659, now + 0.1);
            osc.frequency.setValueAtTime(784, now + 0.2);
            gain.gain.setValueAtTime(0.15, now);
            gain.gain.linearRampToValueAtTime(0, now + 0.4);
            osc.start(now); osc.stop(now + 0.4);
            break;
    }
}

// Physics constants
const GRAVITY = 0.6;
const FRICTION = 0.85;
const ACCEL = 0.7;
const MAX_SPEED = 8;
const JUMP_FORCE = -12;
const BALL_RADIUS = 18;

// Player
let player = { x: 0, y: 0, vx: 0, vy: 0, onGround: false, squash: 1, stretch: 1 };

// Particles
let particles = [];

// Level data
function generateLevel(lvl) {
    const platforms = [];
    const orbs = [];
    const spikes = [];
    const movingPlatforms = [];
    const goal = null;
    
    let x = 200;
    let y = H - 100;
    
    // Starting platform
    platforms.push({ x: 0, y: y, w: 300, h: 40 });
    
    let segments = 8 + lvl * 3;
    for (let i = 0; i < segments; i++) {
        const gap = 80 + Math.random() * (60 + lvl * 15);
        x += gap;
        y += (Math.random() - 0.5) * 120;
        y = Math.max(100, Math.min(H - 150, y));
        
        const pw = 80 + Math.random() * 120;
        platforms.push({ x: x, y: y, w: pw, h: 30 });
        
        // Add orbs above platforms
        if (Math.random() > 0.4) {
            orbs.push({ x: x + pw/2, y: y - 50, collected: false });
        }
        if (Math.random() > 0.6) {
            orbs.push({ x: x + pw/2, y: y - 100, collected: false });
        }
        
        // Add spikes on some platforms
        if (i > 2 && Math.random() > 0.5) {
            const sx = x + pw * 0.3;
            spikes.push({ x: sx, y: y - 15, w: 30, h: 15 });
        }
        
        // Moving platforms
        if (Math.random() > 0.6 && lvl >= 1) {
            const mp = { 
                x: x + pw/2, y: y - 80, w: 70, h: 15, 
                baseY: y - 80, amplitude: 30 + Math.random() * 30, 
                speed: 0.02 + Math.random() * 0.02, phase: Math.random() * Math.PI * 2
            };
            movingPlatforms.push(mp);
        }
        
        x += pw;
    }
    
    // Goal platform
    const goalY = Math.max(150, Math.min(H - 200, y));
    platforms.push({ x: x + 50, y: goalY, w: 150, h: 40 });
    goal = { x: x + 125, y: goalY - 60 };
    
    return { platforms, orbs, spikes, movingPlatforms, goal };
}

let levelData = null;
let time = 0;

function startGame() {
    gameState = 'playing';
    score = 0;
    level = 0;
    loadLevel();
}

function restartLevel() {
    gameState = 'playing';
    deathTimer = 0;
    winTimer = 0;
    loadLevel();
}

function nextLevel() {
    level++;
    gameState = 'playing';
    winTimer = 0;
    loadLevel();
}

function loadLevel() {
    levelData = generateLevel(level);
    player.x = 150;
    player.y = H - 150;
    player.vx = 0;
    player.vy = 0;
    player.onGround = false;
    player.squash = 1;
    player.stretch = 1;
    particles = [];
}

function spawnParticles(x, y, color, count) {
    for (let i = 0; i < count; i++) {
        const angle = Math.random() * Math.PI * 2;
        const speed = 2 + Math.random() * 4;
        particles.push({
            x: x, y: y,
            vx: Math.cos(angle) * speed,
            vy: Math.sin(angle) * speed - 2,
            life: 1,
            color: color,
            size: 3 + Math.random() * 4
        });
    }
}

function update() {
    if (gameState !== 'playing') return;
    
    time += 0.016;
    
    // Input
    let moveX = 0;
    if (keys['ArrowLeft'] || keys['KeyA']) moveX = -ACCEL;
    if (keys['ArrowRight'] || keys['KeyD']) moveX = ACCEL;
    
    player.vx += moveX;
    player.vx *= FRICTION;
    player.vx = Math.max(-MAX_SPEED, Math.min(MAX_SPEED, player.vx));
    
    // Jump
    if ((keys['Space'] || keys['ArrowUp'] || keys['KeyW']) && player.onGround) {
        player.vy = JUMP_FORCE;
        player.onGround = false;
        playSound('jump');
        player.stretch = 1.3;
        player.squash = 0.7;
    }
    
    // Gravity
    player.vy += GRAVITY;
    player.vy = Math.min(player.vy, 20);
    
    // Move
    player.x += player.vx;
    player.y += player.vy;
    
    // Squash/stretch recovery
    player.squash += (1 - player.squash) * 0.15;
    player.stretch += (1 - player.stretch) * 0.15;
    
    // Platform collision
    let wasOnGround = false;
    const allPlatforms = [...levelData.platforms];
    
    // Update moving platforms
    for (let mp of levelData.movingPlatforms) {
        mp.y = mp.baseY + Math.sin(time * mp.speed * 100 + mp.phase) * mp.amplitude;
        allPlatforms.push(mp);
    }
    
    for (let p of allPlatforms) {
        if (player.x + BALL_RADIUS > p.x && player.x - BALL_RADIUS < p.x + p.w) {
            // Landing on top
            if (player.y + BALL_RADIUS > p.y && player.y + BALL_RADIUS < p.y + p.h + Math.abs(player.vy) + 2) {
                if (player.vy >= 0) {
                    player.y = p.y - BALL_RADIUS;
                    player.vy = 0;
                    player.onGround = true;
                    wasOnGround = true;
                }
            }
            // Hitting bottom
            else if (player.y - BALL_RADIUS < p.y + p.h && player.y - BALL_RADIUS > p.y) {
                if (player.vy <= 0) {
                    player.y = p.y + p.h + BALL_RADIUS;
                    player.vy = 0;
                }
            }
        }
        
        // Side collision
        if (player.y + BALL_RADIUS > p.y && player.y - BALL_RADIUS < p.y + p.h) {
            if (Math.abs(player.x + BALL_RADIUS - p.x) < Math.abs(player.vx)) {
                if (player.vx > 0) { player.x = p.x - BALL_RADIUS; player.vx = 0; }
            }
            if (Math.abs(player.x - BALL_RADIUS - (p.x + p.w)) < Math.abs(player.vx)) {
                if (player.vx < 0) { player.x = p.x + p.w + BALL_RADIUS; player.vx = 0; }
            }
        }
    }
    
    // Landing effect
    if (wasOnGround && !player.onGround) {
        // fell off
    }
    if (!player.onGround && wasOnGround === false) {}
    
    // Check spikes
    for (let s of levelData.spikes) {
        if (player.x + BALL_RADIUS * 0.6 > s.x && player.x - BALL_RADIUS * 0.6 < s.x + s.w &&
            player.y + BALL_RADIUS > s.y && player.y - BALL_RADIUS < s.y + s.h) {
            die();
            return;
        }
    }
    
    // Check orbs
    for (let o of levelData.orbs) {
        if (!o.collected) {
            const dx = player.x - o.x;
            const dy = player.y - o.y;
            if (dx * dx + dy * dy < 400) {
                o.collected = true;
                score += 10;
                playSound('collect');
                spawnParticles(o.x, o.y, '#ffd700', 8);
            }
        }
    }
    
    // Check goal
    if (levelData.goal) {
        const dx = player.x - levelData.goal.x;
        const dy = player.y - levelData.goal.y;
        if (dx * dx + dy * dy < 900) {
            gameState = 'win';
            winTimer = 0;
            playSound('win');
            score += 50;
        }
    }
    
    // Fall off screen
    if (player.y > H + 100) {
        die();
        return;
    }
    
    // Particles
    for (let i = particles.length - 1; i >= 0; i--) {
        const p = particles[i];
        p.x += p.vx;
        p.y += p.vy;
        p.vy += 0.2;
        p.life -= 0.03;
        if (p.life <= 0) particles.splice(i, 1);
    }
    
    // Screen shake decay
    screenShake.intensity *= 0.85;
    if (screenShake.intensity < 0.5) screenShake.intensity = 0;
}

function die() {
    gameState = 'dead';
    deathTimer = 0;
    playSound('die');
    spawnParticles(player.x, player.y, '#ff4444', 12);
    screenShake.intensity = 8;
}

function drawBackground() {
    const grad = ctx.createLinearGradient(0, 0, 0, H);
    grad.addColorStop(0, '#1a1a2e');
    grad.addColorStop(0.5, '#16213e');
    grad.addColorStop(1, '#0f3460');
    ctx.fillStyle = grad;
    ctx.fillRect(0, 0, W, H);
    
    // Stars
    for (let i = 0; i < 50; i++) {
        const sx = ((i * 137 + 50) % (W + 200)) - cameraX * 0.1 % (W + 200);
        const sy = (i * 97 + 30) % H;
        const brightness = 0.3 + 0.3 * Math.sin(time * 2 + i);
        ctx.fillStyle = `rgba(255,255,255,${brightness})`;
        ctx.beginPath();
        ctx.arc(((sx % (W+200)) + W + 200) % (W + 200), sy, 1.5, 0, Math.PI * 2);
        ctx.fill();
    }
    
    // Parallax hills
    ctx.fillStyle = '#1a2744';
    for (let i = 0; i < 8; i++) {
        const hx = i * 300 - cameraX * 0.2 % (W + 600);
        const hy = H - 50;
        ctx.beginPath();
        ctx.arc(hx, hy, 100 + (i % 3) * 40, Math.PI, 0);
        ctx.fill();
    }
}

function drawPlatforms() {
    for (let p of levelData.platforms) {
        const px = p.x - cameraX;
        if (px > W + 50 || px + p.w < -50) continue;
        
        ctx.fillStyle = '#4a6fa5';
        ctx.fillRect(px, p.y, p.w, p.h);
        ctx.fillStyle = '#5d8bc7';
        ctx.fillRect(px, p.y, p.w, 4);
        ctx.fillStyle = '#3a5a8a';
        ctx.fillRect(px, p.y + p.h - 4, p.w, 4);
    }
    
    for (let mp of levelData.movingPlatforms) {
        const px = mp.x - cameraX;
        if (px > W + 50 || px + mp.w < -50) continue;
        ctx.fillStyle = '#7b68ee';
        ctx.fillRect(px, mp.y, mp.w, mp.h);
        ctx.fillStyle = '#9d8ff2';
        ctx.fillRect(px, mp.y, mp.w, 3);
    }
}

function drawSpikes() {
    for (let s of levelData.spikes) {
        const sx = s.x - cameraX;
        if (sx > W + 50 || sx + s.w < -50) continue;
        ctx.fillStyle = '#ff4444';
        ctx.beginPath();
        ctx.moveTo(sx, s.y + s.h);
        ctx.lineTo(sx + s.w / 2, s.y);
        ctx.lineTo(sx + s.w, s.y + s.h);
        ctx.closePath();
        ctx.fill();
    }
}

function drawOrbs() {
    for (let o of levelData.orbs) {
        if (o.collected) continue;
        const ox = o.x - cameraX;
        if (ox > W + 50 || ox < -50) continue;
        
        const pulse = 1 + 0.1 * Math.sin(time * 5);
        ctx.fillStyle = '#ffd700';
        ctx.shadowColor = '#ffd700';
        ctx.shadowBlur = 15;
        ctx.beginPath();
        ctx.arc(ox, o.y, 10 * pulse, 0, Math.PI * 2);
        ctx.fill();
        ctx.shadowBlur = 0;
        
        ctx.fillStyle = '#fff8dc';
        ctx.beginPath();
        ctx.arc(ox - 3, o.y - 3, 4, 0, Math.PI * 2);
        ctx.fill();
    }
}

function drawGoal() {
    if (!levelData.goal) return;
    const gx = levelData.goal.x - cameraX;
    if (gx > W + 50 || gx < -50) return;
    
    // Flag pole
    ctx.fillStyle = '#ccc';
    ctx.fillRect(gx - 2, levelData.goal.y - 20, 4, 60);
    
    // Flag
    const wave = Math.sin(time * 3) * 5;
    ctx.fillStyle = '#ff6b6b';
    ctx.beginPath();
    ctx.moveTo(gx + 2, levelData.goal.y - 20);
    ctx.lineTo(gx + 30 + wave, levelData.goal.y - 10);
    ctx.lineTo(gx + 2, levelData.goal.y);
    ctx.closePath();
    ctx.fill();
}

function drawPlayer() {
    const px = player.x - cameraX;
    const py = player.y;
    
    // Shadow
    ctx.fillStyle = 'rgba(0,0,0,0.3)';
    ctx.beginPath();
    ctx.ellipse(px, py + BALL_RADIUS + 5, 12, 4, 0, 0, Math.PI * 2);
    ctx.fill();
    
    // Ball with squash/stretch
    ctx.save();
    ctx.translate(px, py);
    ctx.scale(player.squash, player.stretch);
    
    const ballGrad = ctx.createRadialGradient(-5, -5, 2, 0, 0, BALL_RADIUS);
    ballGrad.addColorStop(0, '#ff9a56');
    ballGrad.addColorStop(0.7, '#ff6b35');
    ballGrad.addColorStop(1, '#e84520');
    ctx.fillStyle = ballGrad;
    ctx.beginPath();
    ctx.arc(0, 0, BALL_RADIUS, 0, Math.PI * 2);
    ctx.fill();
    
    // Shine
    ctx.fillStyle = 'rgba(255,255,255,0.4)';
    ctx.beginPath();
    ctx.arc(-6, -8, 6, 0, Math.PI * 2);
    ctx.fill();
    
    // Eyes
    const lookDir = player.vx > 1 ? 3 : player.vx < -1 ? -3 : 0;
    ctx.fillStyle = '#fff';
    ctx.beginPath();
    ctx.arc(-5, -3, 5, 0, Math.PI * 2);
    ctx.fill();
    ctx.beginPath();
    ctx.arc(5, -3, 5, 0, Math.PI * 2);
    ctx.fill();
    ctx.fillStyle = '#222';
    ctx.beginPath();
    ctx.arc(-5 + lookDir, -3, 2.5, 0, Math.PI * 2);
    ctx.fill();
    ctx.beginPath();
    ctx.arc(5 + lookDir, -3, 2.5, 0, Math.PI * 2);
    ctx.fill();
    
    // Mouth
    if (player.onGround) {
        ctx.strokeStyle = '#c0392b';
        ctx.lineWidth = 2;
        ctx.beginPath();
        ctx.arc(0, 5, 4, 0, Math.PI * 0.8);
        ctx.stroke();
    } else {
        ctx.fillStyle = '#c0392b';
        ctx.beginPath();
        ctx.ellipse(0, 6, 3, 4, 0, 0, Math.PI * 2);
        ctx.fill();
    }
    
    ctx.restore();
}

function drawParticles() {
    for (let p of particles) {
        const px = p.x - cameraX;
        ctx.globalAlpha = p.life;
        ctx.fillStyle = p.color;
        ctx.beginPath();
        ctx.arc(px, p.y, p.size * p.life, 0, Math.PI * 2);
        ctx.fill();
    }
    ctx.globalAlpha = 1;
}

// Main loop
function gameLoop() {
    time += 0.016;
    
    if (gameState === 'start') {
        drawStartScreen();
    } else {
        update();
        updateCamera();
        
        // Apply screen shake
        ctx.save();
        if (screenShake.intensity > 0) {
            ctx.translate(
                (Math.random() - 0.5) * screenShake.intensity,
                (Math.random() - 0.5) * screenShake.intensity
            );
        }
        
        drawBackground();
        drawPlatforms();
        drawSpikes();
        drawOrbs();
        drawGoal();
        drawPlayer();
        drawParticles();
        ctx.restore();
        
        drawHUD();
        
        if (gameState === 'dead') drawDeath();
        if (gameState === 'win') drawWin();
    }
    
    requestAnimationFrame(gameLoop);
}

loadLevel();
cameraX = 0;
gameLoop();
</script>
