<script>
(function(){
let audioCtx = null;
function initAudio() {
  if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
}
function playBeep(freq, dur, type, vol) {
  if (!audioCtx) return;
  const o = audioCtx.createOscillator();
  const g = audioCtx.createGain();
  o.type = type || 'square';
  o.frequency.value = freq;
  g.gain.value = vol || 0.1;
  g.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
  o.connect(g); g.connect(audioCtx.destination);
  o.start(); o.stop(audioCtx.currentTime + dur);
}
function sfxCorrect() { playBeep(880, 0.08, 'sine', 0.12); }
function sfxWrong() { playBeep(150, 0.3, 'sawtooth', 0.15); }
function sfxLevelUp() { playBeep(440,0.1,'sine',0.1); setTimeout(()=>playBeep(660,0.1,'sine',0.1),120); setTimeout(()=>playBeep(880,0.15,'sine',0.1),240); }
function sfxGameOver() { playBeep(200,0.5,'sawtooth',0.2); }

let state = 'start'; // start, playing, over
let grid = [];
let gridSize = 5;
let mineCount = 3;
let safeCells = []; // list of {r,c,val} in reading order
let safeIndex = 0;
let lives = 3;
let score = 0;
let level = 1;
let cellsDOM = [];

const gridContainer = document.getElementById('grid-container');
const typedDisplay = document.getElementById('typed-display');
const msgEl = document.getElementById('msg');
const hudLevel = document.getElementById('hud-level');
const hudScore = document.getElementById('hud-score');
const livesDisplay = document.getElementById('lives-display');
const startScreen = document.getElementById('start-screen');
const endScreen = document.getElementById('end-screen');
const gameContainer = document.getElementById('game-container');
const btnStart = document.getElementById('btn-start');
const btnRestart = document.getElementById('btn-restart');
const finalScoreEl = document.getElementById('final-score');

function getLevelParams(lv) {
  const size = Math.min(5 + lv - 1, 9);
  const mines = Math.floor(size * size * (0.25 + lv * 0.03));
  return { size: Math.min(size, 9), mines: Math.min(mines, size * size - 4) };
}

function generateGrid(size, mineCount) {
  const g = Array.from({length: size}, () => Array.from({length: size}, () => ({ mine: false, val: 0 })));
  let placed = 0;
  while (placed < mineCount) {
    const r = Math.floor(Math.random() * size);
    const c = Math.floor(Math.random() * size);
    if (!g[r][c].mine) { g[r][c].mine = true; placed++; }
  }
  for (let r = 0; r < size; r++) {
    for (let c = 0; c < size; c++) {
      if (g[r][c].mine) continue;
      let n = 0;
      for (let dr = -1; dr <= 1; dr++) {
        for (let dc = -1; dc <= 1; dc++) {
          const nr = r+dr, nc = c+dc;
          if (nr>=0 && nr<size && nc>=0 && nc<size && g[nr][nc].mine) n++;
        }
      }
      g[r][c].val = n;
    }
  }
  return g;
}

function buildLevel() {
  const params = getLevelParams(level);
  gridSize = params.size;
  mineCount = params.mines;
  grid = generateGrid(gridSize, mineCount);
  
  safeCells = [];
  for (let r = 0; r < gridSize; r++) {
    for (let c = 0; c < gridSize; c++) {
      if (!grid[r][c].mine) safeCells.push({r, c, val: grid[r][c].val});
    }
  }
  safeIndex = 0;

  // Build DOM
  gridContainer.innerHTML = '';
  gridContainer.style.gridTemplateColumns = `repeat(${gridSize}, 1fr)`;
  cellsDOM = [];
  for (let r = 0; r < gridSize; r++) {
    for (let c = 0; c < gridSize; c++) {
      const cell = document.createElement('div');
      cell.className = 'cell';
      if (grid[r][c].mine) {
        cell.classList.add('mine');
      } else {
        cell.classList.add('safe');
        cell.textContent = grid[r][c].val;
      }
      gridContainer.appendChild(cell);
      cellsDOM.push({ el: cell, r, c });
    }
  }
  updateTargetHighlight();
  renderTypedDisplay();
}

function updateTargetHighlight() {
  // Remove old highlights
  for (const cd of cellsDOM) {
    cd.el.classList.remove('target', 'done');
  }
  if (safeIndex < safeCells.length) {
    const cur = safeCells[safeIndex];
    const idx = cur.r * gridSize + cur.c;
    cellsDOM[idx].el.classList.add('target');
  }
}

function renderTypedDisplay() {
  let html = '';
  for (let i = 0; i < safeCells.length; i++) {
    if (i === safeIndex) {
      html += `<span class="current">${safeCells[i].val}</span>`;
    } else if (i < safeIndex) {
      html += `<span class="correct">${safeCells[i].val}</span>`;
    } else {
      html += `<span>${safeCells[i].val}</span>`;
    }
  }
  typedDisplay.innerHTML = html;
}

function handleKey(e) {
  if (state !== 'playing') return;
  const ch = e.key;
  if (ch >= '0' && ch <= '9' || ch === ' ') {
    e.preventDefault();
    if (safeIndex >= safeCells.length) return;
    const expected = safeCells[safeIndex].val;
    if (String(expected) === ch) {
      sfxCorrect();
      score += 10 + level * 2;
      // Mark done
      const cur = safeCells[safeIndex];
      const idx = cur.r * gridSize + cur.c;
      cellsDOM[idx].el.classList.remove('target');
      cellsDOM[idx].el.classList.add('done');
      safeIndex++;
      hudScore.textContent = score;
      msgEl.textContent = '';
      if (safeIndex >= safeCells.length) {
        levelComplete();
      } else {
        updateTargetHighlight();
        renderTypedDisplay();
      }
    } else {
      loseLife();
    }
  }
}

function loseLife() {
  sfxWrong();
  lives--;
  livesDisplay.textContent = '♥ '.repeat(Math.max(0, lives)).trim() || '—';
  msgEl.textContent = '✗ Wrong! (' + lives + ' left)';
  // Flash the target cell red briefly
  if (safeIndex < safeCells.length) {
    const cur = safeCells[safeIndex];
    const idx = cur.r * gridSize + cur.c;
    const el = cellsDOM[idx].el;
    el.style.background = '#ff4444';
    setTimeout(() => { el.style.background = ''; }, 300);
  }
  if (lives <= 0) {
    gameOver();
  }
}

function levelComplete() {
  sfxLevelUp();
  score += 100 * level;
  hudScore.textContent = score;
  msgEl.textContent = '✓ Level Complete! +'+(100*level);
  // Reveal mines with animation
  for (let r = 0; r < gridSize; r++) {
    for (let c = 0; c < gridSize; c++) {
      if (grid[r][c].mine) {
        const idx = r * gridSize + c;
        cellsDOM[idx].el.classList.add('revealed-mine');
        cellsDOM[idx].el.textContent = '💣';
      }
    }
  }
  setTimeout(() => {
    level++;
    hudLevel.textContent = level;
    buildLevel();
    msgEl.textContent = '';
  }, 1200);
}

function gameOver() {
  state = 'over';
  sfxGameOver();
  // Reveal all mines
  for (let r = 0; r < gridSize; r++) {
    for (let c = 0; c < gridSize; c++) {
      if (grid[r][c].mine) {
        const idx = r * gridSize + c;
        cellsDOM[idx].el.classList.add('revealed-mine');
        cellsDOM[idx].el.textContent = '💣';
      }
    }
  }
  finalScoreEl.textContent = 'Final Score: ' + score;
  // Try to save high score
  try {
    const prev = parseInt(localStorage.getItem('mswTypingHigh') || '0');
    if (score > prev) localStorage.setItem('mswTypingHigh', String(score));
  } catch(e){}
  setTimeout(() => {
    endScreen.style.display = 'flex';
  }, 800);
}

function startGame() {
  initAudio();
  state = 'playing';
  lives = 3;
  score = 0;
  level = 1;
  hudScore.textContent = '0';
  hudLevel.textContent = '1';
  livesDisplay.textContent = '♥ ♥ ♥';
  msgEl.textContent = '';
  startScreen.style.display = 'none';
  endScreen.style.display = 'none';
  gameContainer.classList.remove('hidden');
  buildLevel();
}

btnStart.addEventListener('click', startGame);
btnRestart.addEventListener('click', startGame);
window.addEventListener('keydown', (e) => {
  if (state === 'start' && (e.key === 'Enter' || e.key === ' ')) { e.preventDefault(); startGame(); return; }
  if (state === 'over' && (e.key === 'Enter' || e.key === ' ')) { e.preventDefault(); startGame(); return; }
  handleKey(e);
});

// Also allow starting with click on start screen
startScreen.addEventListener('click', (e) => {
  if (e.target.tagName !== 'BUTTON') startGame();
});

})();
</script>
</body>
