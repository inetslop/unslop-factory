<!DOCTYPE html>

<script>
(function() {
    const canvas = document.getElementById('game');
    const ctx = canvas.getContext('2d');

    let W, H;
    let gameState = 'start'; // start, playing, scored
    let scoreLeft = 0, scoreRight = 0;
    let winScore = 7;
    let winner = null;
    let audioCtx = null;

    const BALL_R = 12;
    const PADDLE_R = 28;
    const WALL_T = 14;
    const GOAL_H = 200;

    let ball, paddleL, paddleR;
    let keys = {};
    let mouse = { x: 0, y: 0 };
    let lastTime = 0;

    function resize() {
        W = window.innerWidth;
        H = window.innerHeight;
        canvas.width = W;
        canvas.height = H;
    }
    window.addEventListener('resize', resize);
    resize();

    function initGame() {
        scoreLeft = 0;
        scoreRight = 0;
        winner = null;
        resetRound();
    }

    function resetRound() {
        const angle = (Math.random() - 0.5) * Math.PI * 0.4;
        const speed = Math.min(W, H) * 0.006;
        ball = { x: W / 2, y: H / 2, vx: Math.cos(angle) * speed, vy: Math.sin(angle) * speed };
        paddleL = { x: W * 0.15, y: H / 2, vx: 0, vy: 0 };
        paddleR = { x: W * 0.85, y: H / 2, vx: 0, vy: 0 };
    }

    function playSound(freq, dur, type) {
        if (!audioCtx) return;
        try {
            const osc = audioCtx.createOscillator();
            const gain = audioCtx.createGain();
            osc.type = type || 'sine';
            osc.frequency.value = freq;
            gain.gain.setValueAtTime(0.3, audioCtx.currentTime);
            gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + (dur || 0.1));
            osc.connect(gain);
            gain.connect(audioCtx.destination);
            osc.start();
            osc.stop(audioCtx.currentTime + (dur || 0.1));
        } catch(e) {}
    }

    function initAudio() {
        if (!audioCtx) {
            audioCtx = new (window.AudioContext || window.webkitAudioContext)();
        }
        if (audioCtx.state === 'suspended') audioCtx.resume();
    }

    window.addEventListener('keydown', function(e) {
        keys[e.code] = true;
        if (!audioCtx) initAudio();
        if (gameState === 'start' && (e.code === 'Space' || e.code === 'Enter')) {
            gameState = 'playing';
            initGame();
        }
        if (gameState === 'scored' && (e.code === 'Space' || e.code === 'Enter')) {
            if (winner) {
                initGame();
                gameState = 'playing';
            } else {
                gameState = 'playing';
            }
        }
        if (gameState === 'playing' && e.code === 'KeyR') {
            initGame();
            gameState = 'playing';
        }
    });

    window.addEventListener('keyup', function(e) { keys[e.code] = false; });

    canvas.addEventListener('mousemove', function(e) {
        mouse.x = e.clientX;
        mouse.y = e.clientY;
    });

    canvas.addEventListener('touchmove', function(e) {
        e.preventDefault();
        if (e.touches.length > 0) {
            mouse.x = e.touches[0].clientX;
            mouse.y = e.touches[0].clientY;
        }
    }, { passive: false });

    canvas.addEventListener('click', function() {
        if (!audioCtx) initAudio();
        if (gameState === 'start') {
            gameState = 'playing';
            initGame();
        } else if (gameState === 'scored') {
            if (winner) {
                initGame();
                gameState = 'playing';
            } else {
                gameState = 'playing';
            }
        }
    });

    function update(dt) {
        if (gameState !== 'playing') return;

        const speed = Math.min(W, H) * 0.008;

        // Player paddle (left) - WASD + mouse
        let tx = 0, ty = 0;
        if (keys['KeyW'] || keys['ArrowUp']) ty -= 1;
        if (keys['KeyS'] || keys['ArrowDown']) ty += 1;
        if (keys['KeyA'] || keys['ArrowLeft']) tx -= 1;
        if (keys['KeyD'] || keys['ArrowRight']) tx += 1;

        if (tx !== 0 || ty !== 0) {
            const len = Math.sqrt(tx * tx + ty * ty);
            paddleL.vx = (tx / len) * speed;
            paddleL.vy = (ty / len) * speed;
        } else {
            // Mouse control for left paddle
            const dx = mouse.x - paddleL.x;
            const dy = mouse.y - paddleL.y;
            const dist = Math.sqrt(dx * dx + dy * dy);
            if (dist > 2) {
                const mspeed = Math.min(dist * 0.3, speed * 1.5);
                paddleL.vx = (dx / dist) * mspeed;
                paddleL.vy = (dy / dist) * mspeed;
            } else {
                paddleL.vx *= 0.8;
                paddleL.vy *= 0.8;
            }
        }

        // AI paddle (right)
        const aiSpeed = speed * 0.75;
        let targetX = W * 0.85, targetY = H / 2;
        if (ball.vx > 0) {
            const timeToBall = (paddleR.x - ball.x) / (ball.vx || 1);
            targetY = ball.y + ball.vy * timeToBall * 0.8;
            targetX = W * 0.85 - Math.min(40, Math.abs(ball.x - paddleR.x) * 0.2);
        }
        const adx = targetX - paddleR.x;
        const ady = targetY - paddleR.y;
        const adist = Math.sqrt(adx * adx + ady * ady);
        if (adist > 3) {
            paddleR.vx = (adx / adist) * aiSpeed;
            paddleR.vy = (ady / adist) * aiSpeed;
        } else {
            paddleR.vx *= 0.8;
            paddleR.vy *= 0.8;
        }

        // Move paddles
        paddleL.x += paddleL.vx;
        paddleL.y += paddleL.vy;
        paddleR.x += paddleR.vx;
        paddleR.y += paddleR.vy;

        // Constrain paddles to their halves
        const pMaxX_L = W / 2 - PADDLE_R;
        const pMinX_R = W / 2 + PADDLE_R;
        if (paddleL.x < PADDLE_R) paddleL.x = PADDLE_R;
        if (paddleL.x > pMaxX_L) paddleL.x = pMaxX_L;
        if (paddleL.y < WALL_T + PADDLE_R) paddleL.y = WALL_T + PADDLE_R;
        if (paddleL.y > H - WALL_T - PADDLE_R) paddleL.y = H - WALL_T - PADDLE_R;

        if (paddleR.x > W - PADDLE_R) paddleR.x = W - PADDLE_R;
        if (paddleR.x < pMinX_R) paddleR.x = pMinX_R;
        if (paddleR.y < WALL_T + PADDLE_R) paddleR.y = WALL_T + PADDLE_R;
        if (paddleR.y > H - WALL_T - PADDLE_R) paddleR.y = H - WALL_T - PADDLE_R;

        // Move ball
        ball.x += ball.vx;
        ball.y += ball.vy;

        // Ball wall collisions (top/bottom)
        if (ball.y < WALL_T + BALL_R) {
            ball.y = WALL_T + BALL_R;
            ball.vy = Math.abs(ball.vy);
            playSound(400, 0.05, 'square');
        }
        if (ball.y > H - WALL_T - BALL_R) {
            ball.y = H - WALL_T - BALL_R;
            ball.vy = -Math.abs(ball.vy);
            playSound(400, 0.05, 'square');
        }

        // Ball-paddle collisions
        collidePaddle(paddleL);
        collidePaddle(paddleR);

        // Scoring
        if (ball.x < -BALL_R) {
            scoreRight++;
            playSound(200, 0.3, 'sawtooth');
            checkWin();
            resetRound();
        }
        if (ball.x > W + BALL_R) {
            scoreLeft++;
            playSound(200, 0.3, 'sawtooth');
            checkWin();
            resetRound();
        }
    }

    function collidePaddle(p) {
        const dx = ball.x - p.x;
        const dy = ball.y - p.y;
        const dist = Math.sqrt(dx * dx + dy * dy);
        const minDist = BALL_R + PADDLE_R;
        if (dist < minDist) {
            const nx = dx / dist;
            const ny = dy / dist;
            // Separate
            const overlap = minDist - dist;
            ball.x += nx * overlap;
            ball.y += ny * overlap;
            // Reflect + add paddle velocity
            const relVx = ball.vx - p.vx;
            const relVy = ball.vy - p.vy;
            const dot = relVx * nx + relVy * ny;
            ball.vx = (relVx - 2 * dot * nx) * 0.9 + p.vx * 1.2;
            ball.vy = (relVy - 2 * dot * ny) * 0.9 + p.vy * 1.2;
            // Speed limit
            const spd = Math.sqrt(ball.vx * ball.vx + ball.vy * ball.vy);
            const maxSpd = Math.min(W, H) * 0.012;
            if (spd > maxSpd) {
                ball.vx = (ball.vx / spd) * maxSpd;
                ball.vy = (ball.vy / spd) * maxSpd;
            }
            playSound(600, 0.08, 'triangle');
        }
    }

    function checkWin() {
        if (scoreLeft >= winScore) { winner = 'left'; gameState = 'scored'; }
        else if (scoreRight >= winScore) { winner = 'right'; gameState = 'scored'; }
        else if (scoreLeft >= 3 || scoreRight >= 3) { gameState = 'scored'; }
    }

    function draw() {
        ctx.fillStyle = '#16213e';
        ctx.fillRect(0, 0, W, H);

        // Table surface
        const tableX = WALL_T;
        const tableY = WALL_T;
        const tableW = W - WALL_T * 2;
        const tableH = H - WALL_T * 2;

        ctx.fillStyle = '#1f4064';
        ctx.fillRect(tableX, tableY, tableW, tableH);

        // Center line
        ctx.strokeStyle = 'rgba(255,255,255,0.3)';
        ctx.lineWidth = 3;
        ctx.setLineDash([12, 12]);
        ctx.beginPath();
        ctx.moveTo(W / 2, tableY);
        ctx.lineTo(W / 2, tableY + tableH);
        ctx.stroke();
        ctx.setLineDash([]);

        // Center circle
        ctx.beginPath();
        ctx.arc(W / 2, H / 2, Math.min(W, H) * 0.08, 0, Math.PI * 2);
        ctx.strokeStyle = 'rgba(255,255,255,0.2)';
        ctx.lineWidth = 2;
        ctx.stroke();

        // Walls
        ctx.fillStyle = '#e94560';
        ctx.fillRect(0, 0, W, WALL_T);
        ctx.fillRect(0, H - WALL_T, W, WALL_T);
        ctx.fillRect(0, 0, WALL_T, H);
        ctx.fillRect(W - WALL_T, 0, WALL_T, H);

        // Goals (openings on left and right)
        const goalTop = (H - GOAL_H) / 2;
        const goalBot = goalTop + GOAL_H;
        ctx.fillStyle = '#0f0f23';
        ctx.fillRect(0, goalTop, WALL_T, GOAL_H);
        ctx.fillRect(W - WALL_T, goalTop, WALL_T, GOAL_H);

        // Goal glow
        const goalGrad_L = ctx.createLinearGradient(0, 0, WALL_T, 0);
        goal_grad_L.addColorStop(0, 'rgba(233,69,96,0.4)');
        goal_grad_L.addColorStop(1, 'transparent');
        ctx.fillStyle = goalGrad_L;
        ctx.fillRect(0, goalTop, WALL_T, GOAL_H);

        const goalGrad_R = ctx.createLinearGradient(W, 0, W - WALL_T, 0);
        goalGrad_R.addColorStop(0, 'rgba(233,69,96,0.4)');
        goalGrad_R.addColorStop(1, 'transparent');
        ctx.fillStyle = goalGrad_R;
        ctx.fillRect(W - WALL_T, goalTop, WALL_T, GOAL_H);

        // Ball
        if (gameState === 'playing' || gameState === 'scored') {
            ctx.beginPath();
            ctx.arc(ball.x, ball.y, BALL_R, 0, Math.PI * 2);
            const ballGrad = ctx.createRadialGradient(ball.x - 3, ball.y - 3, 2, ball.x, ball.y, BALL_R);
            ballGrad.addColorStop(0, '#fff');
            ballGrad.addColorStop(1, '#f9d423');
            ctx.fillStyle = ballGrad;
            ctx.fill();
            ctx.strokeStyle = '#e94560';
            ctx.lineWidth = 2;
            ctx.stroke();

            // Ball shadow
            ctx.beginPath();
            ctx.ellipse(ball.x, ball.y + BALL_R * 0.7, BALL_R * 0.8, BALL_R * 0.3, 0, 0, Math.PI * 2);
            ctx.fillStyle = 'rgba(0,0,0,0.3)';
            ctx.fill();
        }

        // Paddles
        if (gameState === 'playing' || gameState === 'scored') {
            drawPaddle(paddleL, '#4ecdc4');
            drawPaddle(paddleR, '#ff6b6b');
        }

        // Score
        ctx.font = `bold ${Math.min(W, H) * 0.08}px monospace`;
        ctx.textAlign = 'center';
        ctx.fillStyle = 'rgba(255,255,255,0.8)';
        ctx.fillText(scoreLeft, W / 4, tableY + Math.min(W, H) * 0.1);
        ctx.fillText(scoreRight, W * 3 / 4, tableY + Math.min(W, H) * 0.1);

        // Overlays
        if (gameState === 'start') {
            drawStartScreen();
        } else if (gameState === 'scored') {
            drawScoreScreen();
        }
    }

    function drawPaddle(p, color) {
        ctx.beginPath();
        ctx.arc(p.x, p.y, PADDLE_R, 0, Math.PI * 2);
        const grad = ctx.createRadialGradient(p.x - 5, p.y - 5, 3, p.x, p.y, PADDLE_R);
        grad.addColorStop(0, '#fff');
        grad.addColorStop(0.3, color);
        grad.addColorStop(1, shadeColor(color, -40));
        ctx.fillStyle = grad;
        ctx.fill();
        ctx.strokeStyle = 'rgba(255,255,255,0.5)';
        ctx.lineWidth = 2;
        ctx.stroke();

        // Inner ring
        ctx.beginPath();
        ctx.arc(p.x, p.y, PADDLE_R * 0.5, 0, Math.PI * 2);
        ctx.strokeStyle = 'rgba(255,255,255,0.4)';
        ctx.lineWidth = 1.5;
        ctx.stroke();
    }

    function shadeColor(color, percent) {
        const num = parseInt(color.replace('#', ''), 16);
        const r = Math.min(255, Math.max(0, (num >> 16) + percent));
        const g = Math.min(255, Math.max(0, ((num >> 8) & 0xff) + percent));
        const b = Math.min(255, Math.max(0, (num & 0xff) + percent));
        return `rgb(${r},${g},${b})`;
    }

    function drawStartScreen() {
        ctx.fillStyle = 'rgba(10,10,30,0.85)';
        ctx.fillRect(0, 0, W, H);

        ctx.textAlign = 'center';
        const fs = Math.min(W, H);

        ctx.font = `bold ${fs * 0.1}px monospace`;
        ctx.fillStyle = '#4ecdc4';
        ctx.fillText('TOP-DOWN PONG', W / 2, H * 0.3);

        ctx.font = `${fs * 0.035}px monospace`;
        ctx.fillStyle = '#fff';
        const lines = [
            'Move: WASD / Arrow Keys / Mouse',
            'First to 7 wins!',
            '',
            'Press SPACE or CLICK to start'
        ];
        lines.forEach((line, i) => {
            ctx.fillText(line, W / 2, H * 0.45 + i * fs * 0.05);
        });

        // Animated paddles on start screen
        const t = performance.now() / 1000;
        const demoX1 = W / 2 - W * 0.2;
        const demoX2 = W / 2 + W * 0.2;
        const demoY = H / 2 + Math.sin(t * 2) * H * 0.15;
        drawPaddle({ x: demoX1, y: demoY }, '#4ecdc4');
        drawPaddle({ x: demoX2, y: H / 2 - Math.sin(t * 2) * H * 0.15 }, '#ff6b6b');
    }

    function drawScoreScreen() {
        ctx.fillStyle = 'rgba(10,10,30,0.7)';
        ctx.fillRect(0, 0, W, H);

        ctx.textAlign = 'center';
        const fs = Math.min(W, H);

        if (winner) {
            ctx.font = `bold ${fs * 0.08}px monospace`;
            ctx.fillStyle = winner === 'left' ? '#4ecdc4' : '#ff6b6b';
            ctx.fillText(winner === 'left' ? 'YOU WIN!' : 'AI WINS!', W / 2, H * 0.35);

            ctx.font = `${fs * 0.05}px monospace`;
            ctx.fillStyle = '#fff';
            ctx.fillText(`${scoreLeft} - ${scoreRight}`, W / 2, H * 0.48);

            ctx.font = `${fs * 0.03}px monospace`;
            ctx.fillText('Press SPACE or CLICK to play again', W / 2, H * 0.6);
        } else {
            ctx.font = `bold ${fs * 0.05}px monospace`;
            ctx.fillStyle = '#fff';
            ctx.fillText(`Score: ${scoreLeft} - ${scoreRight}`, W / 2, H * 0.4);
            ctx.font = `${fs * 0.03}px monospace`;
            ctx.fillText('Press SPACE or CLICK to continue', W / 2, H * 0.55);
        }
    }

    let scoreTimer = 0;

    function gameLoop(timestamp) {
        const dt = Math.min((timestamp - lastTime) / 16.67, 3);
        lastTime = timestamp;

        if (gameState === 'playing') {
            update(dt);
        }

        draw();
        requestAnimationFrame(gameLoop);
    }

    initGame();
    requestAnimationFrame(gameLoop);
})();
