```html

<script>
const canvas = document.getElementById('game');
const ctx = canvas.getContext('2d');

let W, H;
function resize() {
    W = window.innerWidth;
    H = window.innerHeight;
    canvas.width = W;
    canvas.height = H;
}
resize();
window.addEventListener('resize', resize);

// Audio
let audioCtx = null;
function initAudio() {
    if (!audioCtx) {
        audioCtx = new (window.AudioContext || window.webkitAudioContext)();
    }
}

function playSound(freq, duration, type, volume) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = type || 'square';
    osc.frequency.value = freq;
    gain.gain.setValueAtTime(volume || 0.15, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + duration);
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + duration);
}

function playBounce() { playSound(440, 0.08, 'square', 0.1); }
function playBlockBreak() { playSound(660, 0.15, 'sawtooth', 0.12); }
function playBlockHit() { playSound(330, 0.06, 'triangle', 0.1); }
function playLifeLost() { playSound(150, 0.4, 'sawtooth', 0.2); }
function playWin() { playSound(880, 0.3, 'sine', 0.15); setTimeout(() => playSound(1100, 0.4, 'sine', 0.15), 150); }

// Game State
const STATE = { START: 0, PLAYING: 1, GAME_OVER: 2, WIN: 3 };
let gameState = STATE.START;
let score = 0;
let lives = 3;
let highScore = 0;
try { highScore = parseInt(localStorage.getItem('breakout_high') || '0'); } catch(e) {}

// Paddle
const paddle = { w: 120, h: 20, x: 0, y: 0, speed: 8 };

// Ball
let ball = { x: 0, y: 0, r: 10, vx: 0, vy: 0, speed: 6, active: false };

// Blocks
let blocks = [];
const BLOCK_COLS = 10;
const BLOCK_ROWS = 5;
const BLOCK_W = 80;
const BLOCK_H = 30;
const BLOCK_GAP = 4;
const BLOCK_TOP = 60;

function createBlocks() {
    blocks = [];
    const totalW = BLOCK_COLS * (BLOCK_W + BLOCK_GAP) - BLOCK_GAP;
    const startX = (W - totalW) / 2;
    for (let r = 0; r < BLOCK_ROWS; r++) {
        for (let c = 0; c < BLOCK_COLS; c++) {
            let hp = 1;
            if (r === 0) hp = 3;
            else if (r === 1) hp = 2;
            blocks.push({
                x: startX + c * (BLOCK_W + BLOCK_GAP),
                y: BLOCK_TOP + r * (BLOCK_H + BLOCK_GAP),
                w: BLOCK_W, h: BLOCK_H,
                hp: hp, maxHp: hp,
                alive: true
            });
        }
    }
}

// Particles
let particles = [];
function spawnParticles(x, y, color, count) {
    for (let i = 0; i < count; i++) {
        const angle = Math.random() * Math.PI * 2;
        const speed = 2 + Math.random() * 4;
        particles.push({
            x: x, y: y,
            vx: Math.cos(angle) * speed,
            vy: Math.sin(angle) * speed - 1,
            size: 3 + Math.random() * 4,
            color: color,
            life: 1
        });
    }
}

function getBlockColor(hp, maxHp) {
    const ratio = hp / maxHp;
    if (ratio > 0.66) return '#ff3366';
    if (ratio > 0.33) return '#ff9933';
    return '#ffcc00';
}

// Input
let keys = {};
window.addEventListener('keydown', e => {
    keys[e.code] = true;
    if (e.code === 'Space' || e.code === 'Enter') {
        initAudio();
        if (gameState === STATE.START) startGame();
        else if (gameState === STATE.GAME_OVER || gameState === STATE.WIN) resetToStart();
    }
});
window.addEventListener('keyup', e => { keys[e.code] = false; });

// Mouse
let mouseX = W / 2;
window.addEventListener('mousemove', e => { mouseX = e.clientX; });
window.addEventListener('click', () => {
    initAudio();
    if (gameState === STATE.START) startGame();
    else if (gameState === STATE.GAME_OVER || gameState === STATE.WIN) resetToStart();
});

// Game Functions
function startGame() {
    score = 0;
    lives = 3;
    createBlocks();
    resetBall();
    paddle.x = W / 2 - paddle.w / 2;
    paddle.y = H - 50;
    gameState = STATE.PLAYING;
}

function resetToStart() {
    gameState = STATE.START;
}

function resetBall() {
    ball.active = false;
    ball.x = W / 2;
    ball.y = H - 80;
    ball.vx = 0;
    ball.vy = 0;
}

function launchBall() {
    ball.active = true;
    const angle = (-Math.PI / 4) + (Math.random() - 0.5) * 0.5;
    ball.vx = Math.cos(angle) * ball.speed;
    ball.vy = Math.sin(angle) * ball.speed;
}

function loseLife() {
    lives--;
    playLifeLost();
    resetBall();
    if (lives <= 0) {
        gameState = STATE.GAME_OVER;
        try { localStorage.setItem('breakout_high', String(highScore)); } catch(e) {}
    }
}

function checkWin() {
    const remaining = blocks.filter(b => b.alive).length;
    if (remaining === 0) {
        gameState = STATE.WIN;
        playWin();
        try { localStorage.setItem('breakout_high', String(highScore)); } catch(e) {}
    }
}

// Update
function update() {
    if (gameState !== STATE.PLAYING) return;

    // Paddle movement
    if (keys['ArrowLeft'] || keys['KeyA']) {
        paddle.x -= paddle.speed;
    } else if (keys['ArrowRight'] || keys['KeyD']) {
        paddle.x += paddle.speed;
    } else {
        // Smooth mouse follow
        const targetX = mouseX - paddle.w / 2;
        paddle.x += (targetX - paddle.x) * 0.3;
    }
    paddle.x = Math.max(0, Math.min(W - paddle.w, paddle.x));
    paddle.y = H - 50;

    if (!ball.active) return;

    ball.x += ball.vx;
    ball.y += ball.vy;

    // Wall collisions
    if (ball.x - ball.r <= 0) { ball.x = ball.r; ball.vx = Math.abs(ball.vx); playBounce(); }
    if (ball.x + ball.r >= W) { ball.x = W - ball.r; ball.vx = -Math.abs(ball.vx); playBounce(); }
    if (ball.y - ball.r <= 0) { ball.y = ball.r; ball.vy = Math.abs(ball.vy); playBounce(); }

    // Paddle collision
    if (ball.vy > 0 && ball.y + ball.r >= paddle.y && ball.y + ball.r <= paddle.y + paddle.h + 5 &&
        ball.x >= paddle.x - ball.r && ball.x <= paddle.x + paddle.w + ball.r) {
        const hitPos = (ball.x - paddle.x) / paddle.w;
        const angle = (hitPos - 0.5) * Math.PI * 0.7;
        ball.vx = Math.sin(angle) * ball.speed;
        ball.vy = -Math.cos(angle) * ball.speed;
        ball.y = paddle.y - ball.r;
        playBounce();
    }

    // Block collisions
    for (let i = 0; i < blocks.length; i++) {
        const b = blocks[i];
        if (!b.alive) continue;
        if (ball.x + ball.r > b.x && ball.x - ball.r < b.x + b.w &&
            ball.y + ball.r > b.y && ball.y - ball.r < b.y + b.h) {
            // Determine side
            const overlapL = ball.x + ball.r - b.x;
            const overlapR = b.x + b.w - (ball.x - ball.r);
            const overlapT = ball.y + ball.r - b.y;
            const overlapB = b.y + b.h - (ball.y - ball.r);
            const minOverlap = Math.min(overlapL, overlapR, overlapT, overlapB);

            if (minOverlap === overlapL || minOverlap === overlapR) {
                ball.vx = -ball.vx;
            } else {
                ball.vy = -ball.vy;
            }

            b.hp--;
            if (b.hp <= 0) {
                b.alive = false;
                score += 10 * (BLOCK_ROWS - Math.floor(b.y / (BLOCK_H + BLOCK_GAP)));
                spawnParticles(b.x + b.w/2, b.y + b.h/2, getBlockColor(1, b.maxHp), 8);
                playBlockBreak();
            } else {
                score += 5;
                spawnParticles(ball.x, ball.y, '#fff', 3);
                playBlockHit();
            }

            if (score > highScore) {
                highScore = score;
                try { localStorage.setItem('breakout_high', String(highScore)); } catch(e) {}
            }

            checkWin();
            break;
        }
    }

    // Ball lost
    if (ball.y - ball.r > H) {
        loseLife();
    }

    // Particles
    for (let i = particles.length - 1; i >= 0; i--) {
        const p = particles[i];
        p.x += p.vx;
        p.y += p.vy;
        p.vy += 0.2;
        p.life -= 0.03;
        if (p.life <= 0) particles.splice(i, 1);
    }

    // Speed up over time
    const elapsed = score / 10;
    ball.speed = 6 + Math.min(elapsed * 0.05, 4);
}

// Render
function drawGlow(x, y, r, color) {
    ctx.save();
    ctx.shadowColor = color;
    ctx.shadowBlur = r;
    ctx.fillStyle = color;
    ctx.beginPath();
    ctx.arc(x, y, r * 0.6, 0, Math.PI * 2);
    ctx.fill();
    ctx.restore();
}

function render() {
    ctx.fillStyle = '#0a0a1a';
    ctx.fillRect(0, 0, W, H);

    if (gameState === STATE.START) {
        drawTitle('NEON BREAKOUT', W/2, H/2 - 60, 48, '#ff3366');
        drawSubtitle('Move: Mouse / Arrow Keys | Launch: Space', W/2, H/2 + 10, 16, '#aaccff');
        drawSubtitle(`High Score: ${highScore}`, W/2, H/2 + 40, 14, '#66ccff');
        drawSubtitle('Click or Press SPACE to Start', W/2, H/2 + 90, 18, '#ffffff');
        // Animated blocks preview
        const t = Date.now() * 0.003;
        for (let i = 0; i < 5; i++) {
            const bx = W/2 - 100 + i * 50;
            const by = H/2 + 140 + Math.sin(t + i) * 10;
            ctx.fillStyle = ['#ff3366','#ff9933','#ffcc00','#66ff99','#6699ff'][i];
            ctx.globalAlpha = 0.7;
            ctx.fillRect(bx - 20, by - 10, 40, 20);
            ctx.globalAlpha = 1;
        }
    } else if (gameState === STATE.PLAYING) {
        // Blocks
        for (const b of blocks) {
            if (!b.alive) continue;
            const color = getBlockColor(b.hp, b.maxHp);
            ctx.fillStyle = color;
            ctx.globalAlpha = 0.3 + 0.7 * (b.hp / b.maxHp);
            ctx.fillRect(b.x, b.y, b.w, b.h);
            ctx.globalAlpha = 1;
            ctx.strokeStyle = color;
            ctx.lineWidth = 1;
            ctx.strokeRect(b.x, b.y, b.w, b.h);
        }

        // Paddle
        const grad = ctx.createLinearGradient(paddle.x, paddle.y, paddle.x, paddle.y + paddle.h);
        grad.addColorStop(0, '#66ccff');
        grad.addColorStop(1, '#3366ff');
        ctx.fillStyle = grad;
        ctx.fillRect(paddle.x, paddle.y, paddle.w, paddle.h);
        ctx.shadowColor = '#66ccff';
        ctx.shadowBlur = 15;
        ctx.strokeStyle = '#aaddff';
        ctx.strokeRect(paddle.x, paddle.y, paddle.w, paddle.h);
        ctx.shadowBlur = 0;

        // Ball
        if (ball.active) {
            drawGlow(ball.x, ball.y, ball.r, '#ffffff');
            ctx.fillStyle = '#ffffff';
            ctx.beginPath();
            ctx.arc(ball.x, ball.y, ball.r * 0.5, 0, Math.PI * 2);
            ctx.fill();
        } else {
            // Ball waiting
            const bobY = ball.y + Math.sin(Date.now() * 0.005) * 5;
            drawGlow(ball.x, bobY, ball.r, '#ff6699');
        }

        // Particles
        for (const p of particles) {
            ctx.globalAlpha = p.life;
            ctx.fillStyle = p.color;
            ctx.fillRect(p.x - p.size/2, p.y - p.size/2, p.size, p.size);
        }
        ctx.globalAlpha = 1;

        // HUD
        drawSubtitle(`Score: ${score}`, 20, 30, 18, '#ffffff');
        drawSubtitle(`Lives: ${'❤'.repeat(lives)}`, W - 120, 30, 18, '#ff3366');
    } else if (gameState === STATE.GAME_OVER) {
        drawTitle('GAME OVER', W/2, H/2 - 40, 42, '#ff3366');
        drawSubtitle(`Score: ${score}`, W/2, H/2 + 10, 20, '#ffffff');
        drawSubtitle(`High Score: ${highScore}`, W/2, H/2 + 40, 16, '#66ccff');
        drawSubtitle('Click or Press SPACE to Play Again', W/2, H/2 + 90, 18, '#ffffff');
    } else if (gameState === STATE.WIN) {
        drawTitle('YOU WIN!', W/2, H/2 - 40, 42, '#66ff99');
        drawSubtitle(`Final Score: ${score}`, W/2, H/2 + 10, 20, '#ffffff');
        drawSubtitle(`High Score: ${highScore}`, W/2, H/2 + 40, 16, '#66ccff');
        drawSubtitle('Click or Press SPACE to Play Again', W/2, H/2 + 90, 18, '#ffffff');
    }
}

function drawTitle(text, x, y, size, color) {
    ctx.font = `bold ${size}px 'Courier New', monospace`;
    ctx.textAlign = 'center';
    ctx.fillStyle = color;
    ctx.shadowColor = color;
    ctx.shadowBlur = 20;
    ctx.fillText(text, x, y);
    ctx.shadowBlur = 0;
}

function drawSubtitle(text, x, y, size, color) {
    ctx.font = `${size}px 'Courier New', monospace`;
    ctx.textAlign = 'center';
    ctx.fillStyle = color;
    ctx.fillText(text, x, y);
}

// Main Loop
let lastTime = 0;
function gameLoop(time) {
    const dt = time - lastTime;
    lastTime = time;
    update();
    render();
    requestAnimationFrame(gameLoop);
}

requestAnimationFrame(gameLoop);
```
