```html
<!DOCTYPE html>
<html lang="en">
<head>
<meta charset="UTF-8">
<meta name="viewport" content="width=device-width, initial-scale=1.0">
<title>Bounce Clicker</title>
<style>
* { margin: 0; padding: 0; box-sizing: border-box; }
html, body { width: 100%; height: 100%; overflow: hidden; background: #0a0a1a; font-family: 'Segoe UI', Arial, sans-serif; }
canvas { display: block; width: 100%; height: 100%; cursor: crosshair; }
</style>

<body>
<canvas id="game"></canvas>
<script>
(function() {
const canvas = document.getElementById('game');
const ctx = canvas.getContext('2d');

let W, H;
function resize() {
W = window.innerWidth;
H = window.innerHeight;
canvas.width = W;
canvas.height = H;
}
window.addEventListener('resize', resize);
resize();

// Audio
let audioCtx = null;
function initAudio() {
if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
}
function playSound(freq, type, duration, vol) {
if (!audioCtx) return;
const osc = audioCtx.createOscillator();
const gain = audioCtx.createGain();
osc.type = type;
osc.frequency.value = freq;
gain.gain.setValueAtTime(vol || 0.15, audioCtx.currentTime);
gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + duration);
osc.connect(gain);
gain.connect(audioCtx.destination);
osc.start();
osc.stop(audioCtx.currentTime + duration);
}
function playBounce() { playSound(220, 'sine', 0.1, 0.1); }
function playCollect() { playSound(880, 'square', 0.15, 0.12); }
function playLaunch() { playSound(440, 'triangle', 0.2, 0.1); }
function playGameOver() { playSound(150, 'sawtooth', 0.5, 0.1); }

// Game state
let state = 'start'; // start, playing, gameover
let score = 0;
let highScore = 0;
try { highScore = parseInt(localStorage.getItem('bounceClickerHigh') || '0') || 0; } catch(e) {}
let combo = 0;
let maxCombo = 0;
let bounces = 0;
let distanceTraveled = 0;
let starsCollected = 0;

// Ball
let ball = { x: 0, y: 0, vx: 0, vy: 0, r: 14, active: false, resting: true };
const GRAVITY = 0.55;
const RESTITUTION = 0.82;
const FRICTION = 0.999;
const STOP_THRESHOLD = 0.8;

// Stars
let stars = [];
let particles = [];
let floatTexts = [];
let trailPoints = [];

function spawnStars() {
stars = [];
const count = Math.floor(W / 120) + 3;
for (let i = 0; i < count; i++) {
stars.push({
x: 60 + Math.random() * (W - 120),
y: 80 + Math.random() * (H - 200),
r: 10,
vx: (Math.random() - 0.5) * 1.5,
vy: (Math.random() - 0.5) * 1.5,
phase: Math.random() * Math.PI * 2,
collected: false
});
}
}

function resetGame() {
score = 0; combo = 0; maxCombo = 0; bounces = 0; distanceTraveled = 0; starsCollected = 0;
ball.x = W / 2;
ball.y = H - ball.r - 10;
ball.vx = 0; ball.vy = 0;
ball.active = false;
ball.resting = true;
trailPoints = [];
particles = [];
floatTexts = [];
spawnStars();
}

function startGame() {
initAudio();
resetGame();
state = 'playing';
}

// Particles
function spawnParticles(x, y, color, count) {
for (let i = 0; i < count; i++) {
const angle = Math.random() * Math.PI * 2;
const speed = 1 + Math.random() * 4;
particles.push({
x, y,
vx: Math.cos(angle) * speed,
vy: Math.sin(angle) * speed - 2,
r: 2 + Math.random() * 3,
life: 1,
color
});
}
}

function addFloatText(x, y, text, color) {
floatTexts.push({ x, y, text, color, life: 1 });
}

// Input
canvas.addEventListener('click', function(e) {
initAudio();
if (state === 'start' || state === 'gameover') {
startGame();
return;
}
if (state === 'playing' && ball.resting) {
const rect = canvas.getBoundingClientRect();
const mx = e.clientX - rect.left;
const my = e.clientY - rect.top;
const dx = mx - ball.x;
const dy = my - ball.y;
const dist = Math.sqrt(dx * dx + dy * dy);
if (dist < 30) return;
const power = Math.min(dist / 15, 28);
ball.vx = (dx / dist) * power;
ball.vy = (dy / dist) * power;
ball.active = true;
ball.resting = false;
trailPoints = [];
playLaunch();
}
});

canvas.addEventListener('touchstart', function(e) {
e.preventDefault();
const touch = e.touches[0];
const mouseEvent = new MouseEvent('click', {
clientX: touch.clientX,
clientY: touch.clientY
});
canvas.dispatchEvent(mouseEvent);
});

// Update
function update() {
if (state !== 'playing') return;

// Stars movement
for (let s of stars) {
if (s.collected) continue;
s.phase += 0.02;
s.x += s.vx + Math.sin(s.phase) * 0.5;
s.y += s.vy + Math.cos(s.phase) * 0.3;
if (s.x < s.r) { s.x = s.r; s.vx *= -1; }
if (s.x > W - s.r) { s.x = W - s.r; s.vx *= -1; }
if (s.y < s.r) { s.y = s.r; s.vy *= -1; }
if (s.y > H - s.r) { s.y = H - s.r; s.vy *= -1; }
}

// Particles
for (let p of particles) {
p.x += p.vx;
p.y += p.vy;
p.vy += 0.3;
p.life -= 0.025;
}
particles = particles.filter(p => p.life > 0);

// Float texts
for (let ft of floatTexts) {
ft.y -= 1.5;
ft.life -= 0.02;
}
floatTexts = floatTexts.filter(ft => ft.life > 0);

if (!ball.active) return;

const prevX = ball.x;
const prevY = ball.y;

ball.vy += GRAVITY;
ball.vx *= FRICTION;
ball.vy *= FRICTION;
ball.x += ball.vx;
ball.y += ball.vy;

distanceTraveled += Math.sqrt((ball.x - prevX) ** 2 + (ball.y - prevY) ** 2);

// Trail
trailPoints.push({ x: ball.x, y: ball.y, life: 1 });
if (trailPoints.length > 30) trailPoints.shift();
for (let tp of trailPoints) tp.life -= 0.03;

// Wall collisions
let bounced = false;
if (ball.x - ball.r < 0) { ball.x = ball.r; ball.vx *= -RESTITUTION; bounced = true; }
if (ball.x + ball.r > W) { ball.x = W - ball.r; ball.vx *= -RESTITUTION; bounced = true; }
if (ball.y - ball.r < 0) { ball.y = ball.r; ball.vy *= -RESTITUTION; bounced = true; }
if (ball.y + ball.r > H) { ball.y = H - ball.r; ball.vy *= -RESTITUTION; bounced = true; }

if (bounced) {
bounces++;
playBounce();
const impactSpeed = Math.sqrt(ball.vx * ball.vx + ball.vy * ball.vy);
if (impactSpeed > 5) {
spawnParticles(ball.x, ball.y, '#ff8844', 3);
}
}

// Star collection
for (let s of stars) {
if (s.collected) continue;
const dx = ball.x - s.x;
const dy = ball.y - s.y;
const dist = Math.sqrt(dx * dx + dy * dy);
if (dist < ball.r + s.r) {
s.collected = true;
starsCollected++;
combo++;
maxCombo = Math.max(maxCombo, combo);
const points = 10 * combo;
score += points;
addFloatText(s.x, s.y, '+' + points, '#44ff88');
spawnParticles(s.x, s.y, '#44ff88', 8);
playCollect();
}
}

// Distance score
const distScore = Math.floor(distanceTraveled / 50);
score += 0; // small continuous scoring via distance

// Check stop
const speed = Math.sqrt(ball.vx * ball.vx + ball.vy * ball.vy);
if (speed < STOP_THRESHOLD && ball.y + ball.r >= H - 2) {
ball.vx = 0;
ball.vy = 0;
ball.active = false;
ball.resting = true;

// Distance bonus
const distBonus = Math.floor(distanceTraveled / 10);
score += distBonus;
addFloatText(W / 2, H / 2 - 50, 'Distance: +' + distBonus, '#88aaff');

// Bounce bonus
if (bounces > 5) {
const bounceBonus = bounces * 2;
score += bounceBonus;
addFloatText(W / 2, H / 2 - 30, 'Bounces: +' + bounceBonus, '#ffaa44');
}

playGameOver();

// Save high score
if (score > highScore) {
highScore = score;
try { localStorage.setItem('bounceClickerHigh', String(highScore)); } catch(e) {}
}

setTimeout(() => { state = 'gameover'; }, 800);
}
}

// Render
function drawStar(ctx, x, y, r, points) {
ctx.beginPath();
for (let i = 0; i < points * 2; i++) {
const angle = (i * Math.PI) / points - Math.PI / 2;
const radius = i % 2 === 0 ? r : r * 0.4;
const px = x + Math.cos(angle) * radius;
const py = y + Math.sin(angle) * radius;
if (i === 0) ctx.moveTo(px, py);
else ctx.lineTo(px, py);
}
ctx.closePath();
}

function render() {
ctx.fillStyle = '#0a0a1a';
ctx.fillRect(0, 0, W, H);

// Grid lines subtle
ctx.strokeStyle = 'rgba(40,60,100,0.15)';
ctx.lineWidth = 1;
for (let x = 0; x < W; x += 60) { ctx.beginPath(); ctx.moveTo(x,0); ctx.lineTo(x,H); ctx.stroke(); }
for (let y = 0; y < H; y += 60) { ctx.beginPath(); ctx.moveTo(0,y); ctx.lineTo(W,y); ctx.stroke(); }

if (state === 'start') {
ctx.fillStyle = '#fff';
ctx.font = 'bold 48px Segoe UI, Arial';
ctx.textAlign = 'center';
ctx.fillText('BOUNCE CLICKER', W/2, H/2 - 80);
ctx.font = '20px Segoe UI, Arial';
ctx.fillStyle = '#aaccee';
ctx.fillText('Click anywhere to launch the ball!', W/2, H/2 - 30);
ctx.fillText('Collect stars for combo points.', W/2, H/2 + 10);
ctx.fillText('Bounce around - earn distance & bounce bonuses!', W/2, H/2 + 50);
ctx.fillStyle = '#ffaa44';
ctx.font = '16px Segoe UI, Arial';
ctx.fillText('High Score: ' + highScore, W/2, H/2 + 100);
ctx.fillStyle = '#fff';
ctx.font = 'bold 24px Segoe UI, Arial';
ctx.fillText('[ CLICK TO START ]', W/2, H/2 + 160);
return;
}

// Stars
for (let s of stars) {
if (s.collected) continue;
const glow = 0.5 + Math.sin(s.phase * 2) * 0.3;
ctx.fillStyle = `rgba(68,255,136,${glow})`;
drawStar(ctx, s.x, s.y, s.r, 5);
ctx.fill();
ctx.strokeStyle = 'rgba(68,255,136,0.6)';
ctx.lineWidth = 1;
ctx.stroke();
}

// Trail
for (let i = 0; i < trailPoints.length; i++) {
const tp = trailPoints[i];
if (tp.life <= 0) continue;
const alpha = tp.life * 0.4;
const r = ball.r * tp.life * 0.6;
ctx.fillStyle = `rgba(255,136,68,${alpha})`;
ctx.beginPath();
ctx.arc(tp.x, tp.y, r, 0, Math.PI*2);
ctx.fill();
}

// Ball
const ballGlow = ball.active ? 1 : 0.5;
ctx.shadowColor = '#ff8844';
ctx.shadowBlur = ball.active ? 15 : 5;
ctx.fillStyle = `rgba(255,136,68,${ballGlow})`;
ctx.beginPath();
ctx.arc(ball.x, ball.y, ball.r, 0, Math.PI*2);
ctx.fill();
ctx.shadowBlur = 0;
ctx.strokeStyle = '#ffcc88';
ctx.lineWidth = 2;
ctx.stroke();

// Inner highlight
ctx.fillStyle = 'rgba(255,220,180,0.4)';
ctx.beginPath();
ctx.arc(ball.x - ball.r*0.3, ball.y - ball.r*0.3, ball.r*0.4, 0, Math.PI*2);
ctx.fill();

// Particles
for (let p of particles) {
ctx.fillStyle = `rgba(${p.color === '#ff8844' ? '255,136,68' : '68,255,136'},${p.life})`;
ctx.beginPath();
ctx.arc(p.x, p.y, p.r * p.life, 0, Math.PI*2);
ctx.fill();
}

// Float texts
for (let ft of floatTexts) {
ctx.fillStyle = `rgba(${ft.color === '#44ff88' ? '68,255,136' : ft.color === '#ffaa44' ? '255,170,68' : '136,170,255'},${ft.life})`;
ctx.font = 'bold 18px Segoe UI';
ctx.textAlign = 'center';
ctx.fillText(ft.text, ft.x, ft.y);
}

// HUD
if (state === 'playing') {
ctx.fillStyle = 'rgba(0,0,0,0.5)';
ctx.fillRect(10, 10, 200, 90);
ctx.strokeStyle = 'rgba(255,136,68,0.3)';
ctx.strokeRect(10, 10, 200, 90);

ctx.fillStyle = '#fff';
ctx.font = 'bold 16px Segoe UI';
ctx.textAlign = 'left';
ctx.fillText('Score: ' + score, 20, 35);
ctx.fillStyle = combo > 1 ? '#44ff88' : '#aaa';
ctx.fillText('Combo: x' + combo, 20, 57);
ctx.fillStyle = '#88aaff';
ctx.fillText('Bounces: ' + bounces, 20, 79);
}

// Game Over
if (state === 'gameover') {
ctx.fillStyle = 'rgba(0,0,0,0.7)';
ctx.fillRect(0, 0, W, H);
ctx.fillStyle = '#fff';
ctx.font = 'bold 42px Segoe UI';
ctx.textAlign = 'center';
ctx.fillText('GAME OVER', W/2, H/2 - 100);

ctx.font = '28px Segoe UI';
ctx.fillStyle = '#ffaa44';
ctx.fillText('Score: ' + score, W/2, H/2 - 50);
ctx.fillStyle = '#44ff88';
ctx.font = '20px Segoe UI';
ctx.fillText('Stars: ' + starsCollected + '   Max Combo: x' + maxCombo, W/2, H/2 - 15);
ctx.fillStyle = '#88aaff';
ctx.fillText('Bounces: ' + bounces + '   Distance: ' + Math.floor(distanceTraveled), W/2, H/2 + 20);

if (score >= highScore && score > 0) {
ctx.fillStyle = '#ffdd44';
ctx.font = 'bold 22px Segoe UI';
ctx.fillText('★ NEW HIGH SCORE! ★', W/2, H/2 + 65);
} else {
ctx.fillStyle = '#aaa';
ctx.font = '18px Segoe UI';
ctx.fillText('High Score: ' + highScore, W/2, H/2 + 65);
}

ctx.fillStyle = '#fff';
ctx.font = 'bold 22px Segoe UI';
ctx.fillText('[ CLICK TO RESTART ]', W/2, H/2 + 120);
}
}

function loop() {
update();
render();
requestAnimationFrame(loop);
}

// Init
ball.x = W / 2;
ball.y = H - ball.r - 10;
spawnStars();
loop();
})();
</script>
```
