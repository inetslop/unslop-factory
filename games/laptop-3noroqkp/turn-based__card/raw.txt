function initAudio() {
  if (!audioCtx) {
    try { audioCtx = new (window.AudioContext || window.webkitAudioContext)(); } catch(e) {}
  }
}

function playSound(type) {
  if (!audioCtx) return;
  const osc = audioCtx.createOscillator();
  const gain = audioCtx.createGain();
  osc.connect(gain);
  gain.connect(audioCtx.destination);
  
  if (type === 'attack') {
    osc.type = 'sawtooth'; osc.frequency.setValueAtTime(200, audioCtx.currentTime);
    osc.frequency.linearRampToValueAtTime(80, audioCtx.currentTime + 0.3);
    gain.gain.setValueAtTime(0.3, audioCtx.currentTime);
    gain.gain.linearRampToValueAtTime(0, audioCtx.currentTime + 0.3);
    osc.start(); osc.stop(audioCtx.currentTime + 0.3);
  } else if (type === 'defense') {
    osc.type = 'sine'; osc.frequency.setValueAtTime(400, audioCtx.currentTime);
    osc.frequency.linearRampToValueAtTime(600, audioCtx.currentTime + 0.2);
    gain.gain.setValueAtTime(0.2, audioCtx.currentTime);
    gain.gain.linearRampToValueAtTime(0, audioCtx.currentTime + 0.2);
    osc.start(); osc.stop(audioCtx.currentTime + 0.2);
  } else if (type === 'heal') {
    osc.type = 'sine'; osc.frequency.setValueAtTime(500, audioCtx.currentTime);
    osc.frequency.linearRampToValueAtTime(800, audioCtx.currentTime + 0.3);
    gain.gain.setValueAtTime(0.2, audioCtx.currentTime);
    gain.gain.linearRampToValueAtTime(0, audioCtx.currentTime + 0.4);
    osc.start(); osc.stop(audioCtx.currentTime + 0.4);
  } else if (type === 'special') {
    osc.type = 'square'; osc.frequency.setValueAtTime(300, audioCtx.currentTime);
    osc.frequency.linearRampToValueAtTime(900, audioCtx.currentTime + 0.2);
    gain.gain.setValueAtTime(0.15, audioCtx.currentTime);
    gain.gain.linearRampToValueAtTime(0, audioCtx.currentTime + 0.3);
    osc.start(); osc.stop(audioCtx.currentTime + 0.3);
  } else if (type === 'win') {
    osc.type = 'sine'; osc.frequency.setValueAtTime(523, audioCtx.currentTime);
    gain.gain.setValueAtTime(0.2, audioCtx.currentTime);
    gain.gain.linearRampToValueAtTime(0, audioCtx.currentTime + 1);
    osc.start(); osc.stop(audioCtx.currentTime + 1);
  } else if (type === 'lose') {
    osc.type = 'sine'; osc.frequency.setValueAtTime(300, audioCtx.currentTime);
    osc.frequency.linearRampToValueAtTime(150, audioCtx.currentTime + 0.8);
    gain.gain.setValueAtTime(0.2, audioCtx.currentTime);
    gain.gain.linearRampToValueAtTime(0, audioCtx.currentTime + 1);
    osc.start(); osc.stop(audioCtx.currentTime + 1);
  } else if (type === 'draw') {
    osc.type = 'triangle'; osc.frequency.setValueAtTime(600, audioCtx.currentTime);
    gain.gain.setValueAtTime(0.1, audioCtx.currentTime);
    gain.gain.linearRampToValueAtTime(0, audioCtx.currentTime + 0.1);
    osc.start(); osc.stop(audioCtx.currentTime + 0.1);
  }
}

function drawCards(n) {
  const pool = getCardPool();
  const cards = [];
  for (let i = 0; i < n; i++) {
    const c = pool[Math.floor(Math.random() * pool.length)];
    cards.push({ ...c, id: Date.now() + i + Math.random() });
  }
  return cards;
}

function playCard(index) {
  if (!state.isPlayerTurn || state.gameOver) return;
  const card = state.hand[index];
  if (card.cost > state.mana) return;
  
  initAudio();
  state.mana -= card.cost;
  state.cardsPlayed++;
  
  switch(card.type) {
    case 'attack':
      let dmg = card.value;
      let blocked = Math.min(state.enemyShield, dmg);
      state.enemyShield -= blocked;
      const actualDmg = dmg - blocked;
      state.enemyHP -= actualDmg;
      setBattleMsg(`${card.art} ${card.name}: ${dmg} damage (${blocked} blocked)!`);
      playSound('attack');
      document.querySelector('.enemy-area').classList.add('damage-effect');
      setTimeout(() => document.querySelector('.enemy-area').classList.remove('damage-effect'), 500);
      addLog(`You played ${card.name}: dealt ${actualDmg} damage.`);
      break;
    case 'defense':
      state.playerShield += card.value;
      setBattleMsg(`${card.art} ${card.name}: +${card.value} shield!`);
      playSound('defense');
      document.querySelector('.player-area').classList.add('shield-effect');
      setTimeout(() => document.querySelector('.player-area').classList.remove('shield-effect'), 500);
      addLog(`You played ${card.name}: gained ${card.value} shield.`);
      break;
    case 'heal':
      const healAmt = Math.min(card.value, state.playerMaxHP - state.playerHP);
      state.playerHP += healAmt;
      setBattleMsg(`${card.art} ${card.name}: +${healAmt} HP!`);
      playSound('heal');
      addLog(`You played ${card.name}: healed ${healAmt} HP.`);
      break;
    case 'special':
      if (card.name === "Drain") {
        const d = Math.min(card.value, state.enemyHP);
        state.enemyHP -= card.value;
        const h = Math.min(card.value, state.playerMaxHP - state.playerHP);
        state.playerHP += h;
        setBattleMsg(`${card.art} Drain: ${card.value} dmg + ${h} heal!`);
      } else if (card.name === "Rampage") {
        const rDmg = Math.min(card.value, state.enemyShield);
        state.enemyShield -= rDmg;
        state.enemyHP -= card.value - rDmg;
        state.playerHP = Math.max(0, state.playerHP - 3);
        setBattleMsg(`${card.art} Rampage: ${card.value} dmg, -3 HP!`);
      } else if (card.name === "Fortress") {
        state.playerShield += card.value;
        const fHeal = Math.min(5, state.playerMaxHP - state.playerHP);
        state.playerHP += fHeal;
        setBattleMsg(`${card.art} Fortress: +${card.value} shield, +${fHeal} HP!`);
      }
      playSound('special');
      addLog(`You played ${card.name}.`);
      break;
  }
  
  state.hand.splice(index, 1);
  checkWin();
  if (!state.gameOver) updateUI();
}

function endTurn() {

  setBattleMsg("Enemy is thinking...");

function enemyTurn() {
  if (state.gameOver) return;
  
  const pool = getCardPool();
  let plays = Math.floor(Math.random() * 3) + 1;
  let enemyMana = state.turn;
  
  for (let i = 0; i < plays; i++) {
    const affordable = pool.filter(c => c.cost <= enemyMana);
    if (affordable.length === 0) break;
    
    // AI logic: prioritize based on situation
    let chosen;
    if (state.enemyHP < 12 && Math.random() < 0.5) {
      const heals = affordable.filter(c => c.type === 'heal');
      chosen = heals.length ? heals[Math.floor(Math.random()*heals.length)] : null;
    } else if (state.playerShield > 3 && Math.random() < 0.4) {
      const attacks = affordable.filter(c => c.type === 'attack');
      chosen = attacks.length ? attacks[Math.floor(Math.random()*attacks.length)] : null;
    } else if (state.enemyHP < 15 && Math.random() < 0.3) {
      const defs = affordable.filter(c => c.type === 'defense');
      chosen = defs.length ? defs[Math.floor(Math.random()*defs.length)] : null;
    } else {
      const attacks = affordable.filter(c => c.type === 'attack' || c.type === 'special');
      if (attacks.length && Math.random() < 0.7) {
        chosen = attacks[Math.floor(Math.random()*attacks.length)];
      } else {
        chosen = affordable[Math.floor(Math.random()*affordable.length)];
      }
    }
    
    if (!chosen) break;
    enemyMana -= chosen.cost;
    
    switch(chosen.type) {
      case 'attack':
        let dmg = chosen.value;
        let blocked = Math.min(state.playerShield, dmg);
        state.playerShield -= blocked;
        state.enemyHP -= 0; // no self damage for enemy attacks
        const actualDmg = dmg - blocked;
        state.playerHP -= actualDmg;
        setBattleMsg(`👹 Enemy plays ${chosen.art} ${chosen.name}: ${actualDmg} damage!`);
        playSound('attack');
        document.querySelector('.player-area').classList.add('damage-effect');
        setTimeout(() => document.querySelector('.player-area').classList.remove('damage-effect'), 500);
        break;
      case 'defense':
        state.enemyShield += chosen.value;
        setBattleMsg(`👹 Enemy plays ${chosen.art}: +${chosen.value} shield`);
        playSound('defense');
        break;
      case 'heal':
        const eh = Math.min(chosen.value, state.enemyMaxHP - state.enemyHP);
        state.enemyHP += eh;
        setBattleMsg(`👹 Enemy plays ${chosen.art}: +${eh} HP`);
        playSound('heal');
        break;
      case 'special':
        if (chosen.name === "Drain") {
          state.playerHP -= chosen.value;
          const eh2 = Math.min(chosen.value, state.enemyMaxHP - state.enemyHP);
          state.enemyHP += eh2;
        } else if (chosen.name === "Rampage") {
          state.playerHP -= chosen.value;
          state.enemyHP = Math.max(0, state.enemyHP - 3);
        } else if (chosen.name === "Fortress") {
          state.enemyShield += chosen.value;
          const eh3 = Math.min(5, state.enemyMaxHP - state.enemyHP);
          state.enemyHP += eh3;
        }
        setBattleMsg(`👹 Enemy plays ${chosen.art} ${chosen.name}!`);
        playSound('special');
        break;
    }
    
    checkWin();
    if (state.gameOver) return;
  }
  
  // End of enemy turn -> new player turn
  state.turn++;
  state.maxMana = Math.min(10, state.turn);
  state.mana = state.maxMana;
  state.isPlayerTurn = true;
  
  // Draw a card
  const drawn = drawCards(1);
  if (state.hand.length < 7) {
    state.hand.push(drawn[0]);
    playSound('draw');
  }
  
  setBattleMsg("Your Turn! Play cards.");
  addLog(`--- Your Turn ${state.turn} ---`);
  updateUI();
}

function checkWin() {
  if (state.enemyHP <= 0) {
    state.gameOver = true;
    setTimeout(() => showEnd(true), 600);
    return true;
  }
  if (state.playerHP <= 0) {
    state.gameOver = true;
    setTimeout(() => showEnd(false), 600);
    return true;
  }
  return false;
}

function showEnd(won) {
  const endScreen = document.getElementById('end-screen');
  endScreen.style.display = 'flex';
  const msgEl = document.getElementById('end-msg');
  if (won) {
    msgEl.textContent = "🏆 Victory!";
    msgEl.className = 'result';
    playSound('win');
  } else {
    msgEl.textContent = "💀 Defeat!";
    msgEl.className = 'loss';
    playSound('lose');
  }
  document.getElementById('end-stats').textContent = `Turns: ${state.turn} | Cards Played: ${state.cardsPlayed} | Your HP: ${Math.max(0,state.playerHP)}`;
}

document.getElementById('end-turn-btn').addEventListener('click', endTurn);

// Keyboard controls
window.addEventListener('keydown', (e) => {
  if (e.key === 'e' || e.key === 'Enter') {
    if (state.isPlayerTurn && !state.gameOver) endTurn();
  }
  if (e.key >= '1' && e.key <= '9') {
    const idx = parseInt(e.key) - 1;
    if (idx < state.hand.length && state.isPlayerTurn && !state.gameOver) {
      playCard(idx);
    }
  }
  if (e.key === 'r' || e.key === 'R') {
    restartGame();
  }
});

});
