```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Snake Asteroids</title>
    <style>
        * { margin: 0; padding: 0; box-sizing: border-box; }
        html, body { width: 100%; height: 100%; overflow: hidden; background: #0a0a1a; }

<canvas id="game"></canvas>
<script>
(function() {
    const canvas = document.getElementById('game');
    const ctx = canvas.getContext('2d');

    let W, H, cellSize, cols, rows;
    let gameState = 'start'; // start, playing, gameover
    let snake = [];
    let dir = { x: 1, y: 0 };
    let nextDir = { x: 1, y: 0 };
    let asteroid = null;
    let comet = null;
    let score = 0;
    let highScore = 0;
    let speed = 150;
    let lastTick = 0;
    let stars = [];
    let particles = [];
    let time = 0;
    let audioCtx = null;

    try {
        const saved = localStorage.getItem('snakeAsteroidsHigh');
        if (saved) highScore = parseInt(saved) || 0;
    } catch(e) {}

    function resize() {
        W = window.innerWidth;
        H = window.innerHeight;
        canvas.width = W;
        canvas.height = H;
        cellSize = Math.floor(Math.min(W, H) / 24);
        cols = Math.floor((W - 40) / cellSize);
        rows = Math.floor((H - 80) / cellSize);
        initStars();
    }

    function initStars() {
        stars = [];
        for (let i = 0; i < 120; i++) {
            stars.push({
                x: Math.random() * W,
                y: Math.random() * H,
                r: Math.random() * 1.5 + 0.3,
                twinkle: Math.random() * Math.PI * 2,
                speed: Math.random() * 0.02 + 0.005
            });
        }
    }

    window.addEventListener('resize', resize);
    resize();

    function initGame() {
        const cx = Math.floor(cols / 2);
        const cy = Math.floor(rows / 2);
        snake = [
            { x: cx, y: cy },
            { x: cx - 1, y: cy },
            { x: cx - 2, y: cy }
        ];
        dir = { x: 1, y: 0 };
        nextDir = { x: 1, y: 0 };
        score = 0;
        speed = 150;
        particles = [];
        spawnAsteroid();
        comet = null;
    }

    function getGridOffset() {
        return { x: (W - cols * cellSize) / 2, y: (H - rows * cellSize) / 2 + 30 };
    }

    function spawnAsteroid() {
        let ax, ay;
        do {
            ax = Math.floor(Math.random() * cols);
            ay = Math.floor(Math.random() * rows);
        } while (snake.some(s => s.x === ax && s.y === ay));
        asteroid = { x: ax, y: ay, vertices: generateAsteroidVertices(), rotation: 0 };
    }

    function spawnComet() {
        let cx, cy;
        do {
            cx = Math.floor(Math.random() * cols);
            cy = Math.floor(Math.random() * rows);
        } while (snake.some(s => s.x === cx && s.y === cy) || (asteroid && asteroid.x === cx && asteroid.y === cy));
        comet = { x: cx, y: cy, life: 8 };
    }

    function generateAsteroidVertices() {
        const verts = [];
        const n = 7 + Math.floor(Math.random() * 4);
        for (let i = 0; i < n; i++) {
            const angle = (i / n) * Math.PI * 2;
            const r = 0.3 + Math.random() * 0.25;
            verts.push({ x: Math.cos(angle) * r, y: Math.sin(angle) * r });
        }
        return verts;
    }

    function playSound(type) {
        if (!audioCtx) return;
        const osc = audioCtx.createOscillator();
        const gain = audioCtx.createGain();
        osc.connect(gain);
        gain.connect(audioCtx.destination);

        if (type === 'eat') {
            osc.type = 'square';
            osc.frequency.setValueAtTime(200, audioCtx.currentTime);
            osc.frequency.exponentialRampToValueAtTime(600, audioCtx.currentTime + 0.1);
            gain.gain.setValueAtTime(0.15, audioCtx.currentTime);
            gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.2);
            osc.start(audioCtx.currentTime);
            osc.stop(audioCtx.currentTime + 0.2);
        } else if (type === 'comet') {
            osc.type = 'sine';
            osc.frequency.setValueAtTime(400, audioCtx.currentTime);
            osc.frequency.exponentialRampToValueAtTime(800, audioCtx.currentTime + 0.15);
            gain.gain.setValueAtTime(0.2, audioCtx.currentTime);
            gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.3);
            osc.start(audioCtx.currentTime);
            osc.stop(audioCtx.currentTime + 0.3);
        } else if (type === 'die') {
            osc.type = 'sawtooth';
            osc.frequency.setValueAtTime(300, audioCtx.currentTime);
            osc.frequency.exponentialRampToValueAtTime(50, audioCtx.currentTime + 0.5);
            gain.gain.setValueAtTime(0.2, audioCtx.currentTime);
            gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.5);
            osc.start(audioCtx.currentTime);
            osc.stop(audioCtx.currentTime + 0.5);
        } else if (type === 'start') {
            osc.type = 'sine';
            osc.frequency.setValueAtTime(300, audioCtx.currentTime);
            osc.frequency.exponentialRampToValueAtTime(600, audioCtx.currentTime + 0.1);
            gain.gain.setValueAtTime(0.15, audioCtx.currentTime);
            gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.15);
            osc.start(audioCtx.currentTime);
            osc.stop(audioCtx.currentTime + 0.15);
        }
    }

    function addParticles(px, py, color, count) {
        for (let i = 0; i < count; i++) {
            const angle = Math.random() * Math.PI * 2;
            const speed = Math.random() * 3 + 1;
            particles.push({
                x: px, y: py,
                vx: Math.cos(angle) * speed,
                vy: Math.sin(angle) * speed,
                life: 1,
                color: color,
                r: Math.random() * 3 + 1
            });
        }
    }

    function updateParticles(dt) {
        for (let i = particles.length - 1; i >= 0; i--) {
            const p = particles[i];
            p.x += p.vx;
            p.y += p.vy;
            p.life -= dt * 2;
            if (p.life <= 0) particles.splice(i, 1);
        }
    }

    function tick() {
        dir = nextDir;
        const head = { x: snake[0].x + dir.x, y: snake[0].y + dir.y };

        // Wall collision
        if (head.x < 0 || head.x >= cols || head.y < 0 || head.y >= rows) {
            gameOver();
            return;
        }

        // Self collision
        for (let i = 0; i < snake.length; i++) {
            if (head.x === snake[i].x && head.y === snake[i].y) {
                gameOver();
                return;
            }
        }

        snake.unshift(head);

        // Eat asteroid
        if (asteroid && head.x === asteroid.x && head.y === asteroid.y) {
            score += 10;
            speed = Math.max(60, speed - 5);
            const off = getGridOffset();
            addParticles(off.x + (head.x + 0.5) * cellSize, off.y + (head.y + 0.5) * cellSize, '#c9a87c', 12);
            playSound('eat');
            spawnAsteroid();
            if (score % 50 === 0 && !comet) spawnComet();
        }

        // Eat comet
        if (comet && head.x === comet.x && head.y === comet.y) {
            score += 25;
            const off = getGridOffset();
            addParticles(off.x + (head.x + 0.5) * cellSize, off.y + (head.y + 0.5) * cellSize, '#ff6644', 15);
            playSound('comet');
            comet = null;
        }

        snake.pop();
    }

    function gameOver() {
        gameState = 'gameover';
        if (score > highScore) {
            highScore = score;
            try { localStorage.setItem('snakeAsteroidsHigh', String(highScore)); } catch(e) {}
        }
        const off = getGridOffset();
        addParticles(off.x + (snake[0].x + 0.5) * cellSize, off.y + (snake[0].y + 0.5) * cellSize, '#44ff88', 20);
        playSound('die');
    }

    function drawStars(dt) {
        for (const s of stars) {
            s.twinkle += s.speed;
            const alpha = 0.3 + Math.sin(s.twinkle) * 0.4;
            ctx.fillStyle = `rgba(255, 255, 255, ${alpha})`;
            ctx.beginPath();
            ctx.arc(s.x, s.y, s.r, 0, Math.PI * 2);
            ctx.fill();
        }
    }

    function drawAsteroid(ax, ay, vertices, rot, scale) {
        const off = getGridOffset();
        const px = off.x + (ax + 0.5) * cellSize;
        const py = off.y + (ay + 0.5) * cellSize;
        const r = cellSize * 0.4 * scale;

        ctx.save();
        ctx.translate(px, py);
        ctx.rotate(rot);

        // Body
        ctx.beginPath();
        for (let i = 0; i < vertices.length; i++) {
            const v = vertices[i];
            if (i === 0) ctx.moveTo(v.x * r * 2, v.y * r * 2);
            else ctx.lineTo(v.x * r * 2, v.y * r * 2);
        }
        ctx.closePath();

        const grad = ctx.createRadialGradient(-r*0.3, -r*0.3, 0, 0, 0, r*1.5);
        grad.addColorStop(0, '#d4a872');
        grad.addColorStop(0.5, '#a0784a');
        grad.addColorStop(1, '#6b4c2a');
        ctx.fillStyle = grad;
        ctx.fill();

        // Craters
        ctx.fillStyle = 'rgba(80, 50, 30, 0.5)';
        ctx.beginPath();
        ctx.arc(-r*0.3, -r*0.2, r*0.25, 0, Math.PI * 2);
        ctx.fill();
        ctx.beginPath();
        ctx.arc(r*0.2, r*0.3, r*0.18, 0, Math.PI * 2);
        ctx.fill();

        // Highlight
        ctx.fillStyle = 'rgba(255, 220, 160, 0.3)';
        ctx.beginPath();
        ctx.arc(-r*0.4, -r*0.4, r*0.15, 0, Math.PI * 2);
        ctx.fill();

        ctx.restore();
    }

    function drawComet(cx, cy) {
        const off = getGridOffset();
        const px = off.x + (cx + 0.5) * cellSize;
        const py = off.y + (cy + 0.5) * cellSize;
        const pulse = 1 + Math.sin(time * 6) * 0.2;

        ctx.save();
        ctx.translate(px, py);

        // Glow
        ctx.shadowColor = '#ff6644';
        ctx.shadowBlur = 15;

        // Comet body
        const grad = ctx.createRadialGradient(0, 0, 0, 0, 0, cellSize * 0.35 * pulse);
        grad.addColorStop(0, '#ffffff');
        grad.addColorStop(0.3, '#ffaa44');
        grad.addColorStop(0.7, '#ff4422');
        grad.addColorStop(1, 'rgba(255, 68, 34, 0)');
        ctx.fillStyle = grad;
        ctx.beginPath();
        ctx.arc(0, 0, cellSize * 0.35 * pulse, 0, Math.PI * 2);
        ctx.fill();

        // Tail
        ctx.shadowBlur = 0;
        ctx.strokeStyle = 'rgba(255, 100, 50, 0.6)';
        ctx.lineWidth = 2;
        for (let i = 0; i < 3; i++) {
            ctx.beginPath();
            const tailAngle = time * 2 + i * 2;
            ctx.moveTo(0, 0);
            ctx.quadraticCurveTo(
                Math.cos(tailAngle) * cellSize * 0.5,
                Math.sin(tailAngle) * cellSize * 0.5,
                Math.cos(tailAngle + 0.5) * cellSize * 1,
                Math.sin(tailAngle + 0.5) * cellSize * 1
            );
            ctx.stroke();
        }

        ctx.restore();
    }

    function drawSnake() {
        const off = getGridOffset();

        for (let i = snake.length - 1; i >= 0; i--) {
            const seg = snake[i];
            const px = off.x + (seg.x + 0.5) * cellSize;
            const py = off.y + (seg.y + 0.5) * cellSize;

            const t = i / snake.length;
            const r = Math.floor(68 + t * 40);
            const g = Math.floor(255 - t * 100);
            const b = Math.floor(136 - t * 50);

            // Glow for head
            if (i === 0) {
                ctx.shadowColor = '#44ff88';
                ctx.shadowBlur = 12;
            } else {
                ctx.shadowBlur = 0;
            }

            const size = cellSize * (i === 0 ? 0.45 : 0.38 - t * 0.1);
            const wobble = Math.sin(time * 4 + i * 0.5) * 1.5;

            ctx.fillStyle = `rgb(${r}, ${g}, ${b})`;
            ctx.beginPath();
            ctx.arc(px, py + wobble, size, 0, Math.PI * 2);
            ctx.fill();

            // Head detail
            if (i === 0) {
                ctx.shadowBlur = 0;
                ctx.fillStyle = '#1a3a2a';
                const eyeOffX = dir.x * cellSize * 0.15;
                const eyeOffY = dir.y * cellSize * 0.15;
                ctx.beginPath();
                ctx.arc(px + eyeOffX - 3, py + eyeOffY - 3, 2.5, 0, Math.PI * 2);
                ctx.fill();
                ctx.beginPath();
                ctx.arc(px + eyeOffX + 3, py + eyeOffY - 3, 2.5, 0, Math.PI * 2);
                ctx.fill();
            }

            // Segment texture dots
            if (i > 0 && i % 3 === 0) {
                ctx.fillStyle = 'rgba(0,0,0,0.2)';
                ctx.beginPath();
                ctx.arc(px - size * 0.3, py + wobble - size * 0.2, size * 0.2, 0, Math.PI * 2);
                ctx.fill();
            }
        }
    }

    function drawParticles() {
        for (const p of particles) {
            ctx.globalAlpha = p.life;
            ctx.fillStyle = p.color;
            ctx.beginPath();
            ctx.arc(p.x, p.y, p.r * p.life, 0, Math.PI * 2);
            ctx.fill();
        }
        ctx.globalAlpha = 1;
    }

    function drawHUD() {
        ctx.fillStyle = 'rgba(10, 10, 30, 0.8)';
        ctx.fillRect(0, 0, W, 40);

        ctx.font = 'bold 16px monospace';
        ctx.fillStyle = '#44ff88';
        ctx.textAlign = 'left';
        ctx.fillText(`SCORE: ${score}`, 20, 25);
        ctx.fillStyle = '#c9a87c';
        ctx.textAlign = 'right';
        ctx.fillText(`BEST: ${highScore}`, W - 20, 25);

        // Speed indicator
        const speedPct = Math.round((150 - speed) / 9 * 100);
        ctx.fillStyle = '#ff6644';
        ctx.textAlign = 'center';
        ctx.font = '12px monospace';
        ctx.fillText(`SPEED +${speedPct}%`, W / 2, 25);
    }

    function drawStartScreen() {
        ctx.fillStyle = '#0a0a1a';
        ctx.fillRect(0, 0, W, H);
        drawStars(0);

        const titleY = H * 0.3;

        // Title
        ctx.font = `bold ${Math.min(48, W * 0.06)}px monospace`;
        ctx.textAlign = 'center';
        ctx.fillStyle = '#44ff88';
        ctx.shadowColor = '#44ff88';
        ctx.shadowBlur = 20;
        ctx.fillText('SNAKE ASTEROIDS', W / 2, titleY);
        ctx.shadowBlur = 0;

        // Subtitle
        ctx.font = `${Math.min(16, W * 0.03)}px monospace`;
        ctx.fillStyle = '#c9a87c';
        ctx.fillText('Navigate the void. Devour the rocks.', W / 2, titleY + 40);

        // Draw a decorative asteroid
        drawAsteroid(W / 2 - cellSize * 1.5, H / 2 - cellSize, [
            {x: 0.3, y: 0}, {x: 0.25, y: 0.25}, {x: 0, y: 0.3}, {x: -0.25, y: 0.2}, {x: -0.3, y: -0.1}, {x: -0.15, y: -0.3}, {x: 0.1, y: -0.28}
        ], time * 0.5, 2);

        // Controls
        const cy = H * 0.6;
        ctx.font = `${Math.min(14, W * 0.025)}px monospace`;
        ctx.fillStyle = '#8888aa';
        ctx.fillText('ARROW KEYS / WASD — Move', W / 2, cy);
        ctx.fillText('Eat asteroids to grow. Avoid walls & yourself!', W / 2, cy + 25);
        ctx.fillText('Catch comets for bonus points!', W / 2, cy + 50);

        // Start prompt
        const blink = Math.sin(time * 3) > 0;
        if (blink) {
            ctx.fillStyle = '#ffffff';
            ctx.font = `bold ${Math.min(18, W * 0.03)}px monospace`;
            ctx.fillText('PRESS ANY KEY TO START', W / 2, H * 0.82);
        }

        if (highScore > 0) {
            ctx.fillStyle = '#c9a87c';
            ctx.font = '14px monospace';
            ctx.fillText(`HIGH SCORE: ${highScore}`, W / 2, H * 0.9);
        }
    }

    function drawGameOver() {
        ctx.fillStyle = 'rgba(0, 0, 0, 0.7)';
        ctx.fillRect(0, 0, W, H);

        const cy = H * 0.35;
        ctx.font = `bold ${Math.min(40, W * 0.05)}px monospace`;
        ctx.textAlign = 'center';
        ctx.fillStyle = '#ff4444';
        ctx.shadowColor = '#ff4444';
        ctx.shadowBlur = 15;
        ctx.fillText('GAME OVER', W / 2, cy);
        ctx.shadowBlur = 0;

        ctx.font = `${Math.min(20, W * 0.03)}px monospace`;
        ctx.fillStyle = '#ffffff';
        ctx.fillText(`Score: ${score}`, W / 2, cy + 45);

        if (score >= highScore && score > 0) {
            ctx.fillStyle = '#ffaa00';
            ctx.fillText('★ NEW HIGH SCORE! ★', W / 2, cy + 75);
        }

        const blink = Math.sin(time * 3) > 0;
        if (blink) {
            ctx.fillStyle = '#44ff88';
            ctx.font = `${Math.min(16, W * 0.025)}px monospace`;
            ctx.fillText('PRESS ANY KEY TO RESTART', W / 2, cy + 120);
        }
    }

    let lastTime = 0;

    function loop(timestamp) {
        const dt = Math.min((timestamp - lastTime) / 1000, 0.1);
        lastTime = timestamp;
        time += dt;

        ctx.fillStyle = '#0a0a1a';
        ctx.fillRect(0, 0, W, H);

        drawStars(dt);

        if (gameState === 'start') {
            drawStartScreen();
        } else if (gameState === 'playing') {
            // Draw grid lines subtly
            const off = getGridOffset();
            ctx.strokeStyle = 'rgba(40, 40, 80, 0.3)';
            ctx.lineWidth = 0.5;
            for (let x = 0; x <= cols; x++) {
                ctx.beginPath();
                ctx.moveTo(off.x + x * cellSize, off.y);
                ctx.lineTo(off.x + x * cellSize, off.y + rows * cellSize);
                ctx.stroke();
            }
            for (let y = 0; y <= rows; y++) {
                ctx.beginPath();
                ctx.moveTo(off.x, off.y + y * cellSize);
                ctx.lineTo(off.x + cols * cellSize, off.y + y * cellSize);
                ctx.stroke();
            }

            // Border glow
            ctx.strokeStyle = 'rgba(68, 255, 136, 0.3)';
            ctx.lineWidth = 2;
            ctx.strokeRect(off.x - 2, off.y - 2, cols * cellSize + 4, rows * cellSize + 4);

            // Asteroid
            if (asteroid) {
                asteroid.rotation += dt * 0.5;
                drawAsteroid(asteroid.x, asteroid.y, asteroid.vertices, asteroid.rotation, 1);
            }

            // Comet
            if (comet) {
                comet.life -= dt;
                if (comet.life <= 0) comet = null;
                else drawComet(comet.x, comet.y);
            }

            // Snake
            drawSnake();

            // Particles
            updateParticles(dt);
            drawParticles();

            // HUD
            drawHUD();

            // Game tick
            if (timestamp - lastTick >= speed) {
                lastTick = timestamp;
                tick();
            }
        } else if (gameState === 'gameover') {
            // Still show the game state behind
            const off = getGridOffset();
            drawSnake();
            updateParticles(dt);
            drawParticles();
            drawHUD();
            drawGameOver();
        }

        requestAnimationFrame(loop);
    }

    function handleKey(e) {
        if (!audioCtx) {
            try { audioCtx = new (window.AudioContext || window.webkitAudioContext)(); } catch(err) {}
        }

        if (gameState === 'start') {
            gameState = 'playing';
            initGame();
            lastTick = performance.now();
            playSound('start');
            return;
        }

        if (gameState === 'gameover') {
            gameState = 'playing';
            initGame();
            lastTick = performance.now();
            playSound('start');
            return;
        }

        // Direction handling
        const k = e.key || e.keyCode;
        let newDir = null;

        if (k === 'ArrowUp' || k === 'w' || k === 'W') newDir = { x: 0, y: -1 };
        else if (k === 'ArrowDown' || k === 's' || k === 'S') newDir = { x: 0, y: 1 };
        else if (k === 'ArrowLeft' || k === 'a' || k === 'A') newDir = { x: -1, y: 0 };
        else if (k === 'ArrowRight' || k === 'd' || k === 'D') newDir = { x: 1, y: 0 };

        if (newDir) {
            // Prevent reversing
            if (!(newDir.x === -dir.x && newDir.y === -dir.y)) {
                nextDir = newDir;
            }
            e.preventDefault();
        }
    }

    window.addEventListener('keydown', handleKey);

    requestAnimationFrame(loop);
})();
</script>
</body>
</html>
```
