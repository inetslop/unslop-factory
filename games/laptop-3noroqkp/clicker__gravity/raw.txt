```html
<!DOCTYPE html>
<html lang="en">
<head>

<style>

<script>
const canvas = document.getElementById('game');
const ctx = canvas.getContext('2d');

let W, H;
function resize() {
    W = canvas.width = window.innerWidth;
    H = canvas.height = window.innerHeight;
}
window.addEventListener('resize', resize);
resize();

// Audio
let audioCtx = null;
function initAudio() {
    if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
}
function playSound(freq, type, duration, vol) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = type;
    osc.frequency.value = freq;
    gain.gain.setValueAtTime(vol || 0.3, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + duration);
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + duration);
}

// Game State
let state = 'start'; // start, playing, gameover
let score = 0;
let lives = 3;
let combo = 0;
let comboTimer = 0;
let time = 0;
let difficulty = 1;
let shakeTime = 0;
let shakeIntensity = 0;

// Player
let player = { x: 0, y: 0, vx: 0, vy: 0, radius: 15, glow: 0 };
let clickPulses = [];
let ripples = [];
let particles = [];
let stars = [];
let hazards = [];
let starfield = [];
let spawnTimer = 0;
let hazardTimer = 0;

// Starfield
function initStarfield() {
    starfield = [];
    for (let i = 0; i < 150; i++) {
        starfield.push({
            x: Math.random() * W,
            y: Math.random() * H,
            size: Math.random() * 2 + 0.5,
            speed: Math.random() * 0.3 + 0.1,
            twinkle: Math.random() * Math.PI * 2
        });
    }
}

function resetGame() {
    player.x = W / 2;
    player.y = H / 2;
    player.vx = 0;
    player.vy = 0;
    score = 0;
    lives = 3;
    combo = 0;
    comboTimer = 0;
    time = 0;
    difficulty = 1;
    clickPulses = [];
    ripples = [];
    particles = [];
    stars = [];
    hazards = [];
    spawnTimer = 0;
    hazardTimer = 0;
    initStarfield();
    for (let i = 0; i < 5; i++) spawnStar();
}

function spawnStar() {
    const side = Math.floor(Math.random() * 4);
    let x, y;
    if (side === 0) { x = -20; y = Math.random() * H; }
    else if (side === 1) { x = W + 20; y = Math.random() * H; }
    else if (side === 2) { x = Math.random() * W; y = -20; }
    else { x = Math.random() * W; y = H + 20; }
    
    const type = Math.random() < 0.15 && difficulty > 2 ? 'purple' : 'gold';
    stars.push({
        x, y, vx: (Math.random() - 0.5) * 1.5, vy: (Math.random() - 0.5) * 1.5,
        radius: type === 'purple' ? 12 : 10, type, pulse: Math.random() * Math.PI * 2
    });
}

function spawnHazard() {
    const side = Math.floor(Math.random() * 4);
    let x, y;
    if (side === 0) { x = -20; y = Math.random() * H; }
    else if (side === 1) { x = W + 20; y = Math.random() * H; }
    else if (side === 2) { x = Math.random() * W; y = -20; }
    else { x = Math.random() * W; y = H + 20; }
    
    const angle = Math.atan2(player.y - y, player.x - x) + (Math.random() - 0.5);
    const speed = 1 + difficulty * 0.3;
    hazards.push({
        x, y, vx: Math.cos(angle) * speed, vy: Math.sin(angle) * speed,
        radius: 14, pulse: 0
    });
}

function spawnParticles(x, y, color, count) {
    for (let i = 0; i < count; i++) {
        const angle = Math.random() * Math.PI * 2;
        const speed = Math.random() * 4 + 1;
        particles.push({
            x, y, vx: Math.cos(angle) * speed, vy: Math.sin(angle) * speed,
            life: 1, decay: Math.random() * 0.03 + 0.02,
            size: Math.random() * 4 + 2, color
        });
    }
}

function addRipple(x, y) {
    ripples.push({ x, y, radius: 0, maxRadius: 60, life: 1 });
}

// Input
let mouseX = 0, mouseY = 0;
window.addEventListener('mousemove', e => { mouseX = e.clientX; mouseY = e.clientY; });
window.addEventListener('click', handleClick);
window.addEventListener('touchstart', e => {
    e.preventDefault();
    const t = e.touches[0];
    handleInput(t.clientX, t.clientY);
});

function handleClick(e) {
    initAudio();
    if (state === 'start') { state = 'playing'; resetGame(); return; }
    if (state === 'gameover') { state = 'playing'; resetGame(); return; }
    handleInput(e.clientX, e.clientY);
}

function handleInput(cx, cy) {
    if (state !== 'playing') return;
    
    const dx = cx - player.x;
    const dy = cy - player.y;
    const dist = Math.sqrt(dx * dx + dy * dy);
    if (dist < 5) return;
    
    const force = Math.min(dist / 30, 8);
    const angle = Math.atan2(dy, dx);
    player.vx += Math.cos(angle) * force;
    player.vy += Math.sin(angle) * force;
    
    // Cap velocity
    const maxSpeed = 15;
    const speed = Math.sqrt(player.vx * player.vx + player.vy * player.vy);
    if (speed > maxSpeed) {
        player.vx = (player.vx / speed) * maxSpeed;
        player.vy = (player.vy / speed) * maxSpeed;
    }
    
    addRipple(cx, cy);
    playSound(200 + Math.random() * 100, 'sine', 0.1, 0.15);
}

window.addEventListener('keydown', e => {
    if (e.code === 'Space' || e.code === 'Enter') {
        initAudio();
        if (state === 'start') { state = 'playing'; resetGame(); }
        else if (state === 'gameover') { state = 'playing'; resetGame(); }
    }
});

// Update
function update(dt) {
    if (state !== 'playing') return;
    
    time += dt;
    difficulty = 1 + Math.floor(time / 30);
    
    // Combo timer
    comboTimer -= dt;
    if (comboTimer <= 0) combo = 0;
    
    // Player physics
    player.vx *= 0.97;
    player.vy *= 0.97;
    player.x += player.vx;
    player.y += player.vy;
    
    // Bounds
    if (player.x < player.radius) { player.x = player.radius; player.vx *= -0.5; }
    if (player.x > W - player.radius) { player.x = W - player.radius; player.vx *= -0.5; }
    if (player.y < player.radius) { player.y = player.radius; player.vy *= -0.5; }
    if (player.y > H - player.radius) { player.y = H - player.radius; player.vy *= -0.5; }
    
    // Glow based on speed
    const speed = Math.sqrt(player.vx * player.vx + player.vy * player.vy);
    player.glow = Math.min(speed / 10, 1);
    
    // Spawn stars
    spawnTimer += dt;
    if (spawnTimer > Math.max(2 - difficulty * 0.3, 0.8)) {
        spawnTimer = 0;
        spawnStar();
    }
    
    // Spawn hazards
    hazardTimer += dt;
    const hazardInterval = Math.max(3 - difficulty * 0.4, 1);
    if (hazardTimer > hazardInterval && hazards.length < 3 + difficulty) {
        hazardTimer = 0;
        spawnHazard();
    }
    
    // Update stars
    for (let i = stars.length - 1; i >= 0; i--) {
        const s = stars[i];
        s.pulse += dt * 3;
        s.x += s.vx;
        s.y += s.vy;
        
        // Bounce off walls
        if (s.x < s.radius) { s.x = s.radius; s.vx *= -1; }
        if (s.x > W - s.radius) { s.x = W - s.radius; s.vx *= -1; }
        if (s.y < s.radius) { s.y = s.radius; s.vy *= -1; }
        if (s.y > H - s.radius) { s.y = H - s.radius; s.vy *= -1; }
        
        // Collect check
        const dx = player.x - s.x;
        const dy = player.y - s.y;
        const dist = Math.sqrt(dx * dx + dy * dy);
        if (dist < player.radius + s.radius) {
            const points = (s.type === 'purple' ? 50 : 10) * (1 + combo);
            score += points;
            combo++;
            comboTimer = 3;
            spawnParticles(s.x, s.y, s.type === 'purple' ? '#a855f7' : '#fbbf24', 15);
            playSound(s.type === 'purple' ? 600 : 440 + combo * 50, 'sine', 0.2, 0.3);
            stars.splice(i, 1);
        }
        
        // Remove if off screen too long
        if (s.x < -100 || s.x > W + 100 || s.y < -100 || s.y > H + 100) {
            stars.splice(i, 1);
        }
    }
    
    // Update hazards
    for (let i = hazards.length - 1; i >= 0; i--) {
        const h = hazards[i];
        h.pulse += dt * 5;
        h.x += h.vx;
        h.y += h.vy;
        
        if (h.x < -50 || h.x > W + 50 || h.y < -50 || h.y > H + 50) {
            hazards.splice(i, 1);
            continue;
        }
        
        // Hit player
        const dx = player.x - h.x;
        const dy = player.y - h.y;
        const dist = Math.sqrt(dx * dx + dy * dy);
        if (dist < player.radius + h.radius) {
            lives--;
            combo = 0;
            shakeTime = 0.3;
            shakeIntensity = 8;
            spawnParticles(player.x, player.y, '#ef4444', 25);
            playSound(100, 'sawtooth', 0.3, 0.4);
            hazards.splice(i, 1);
            
            if (lives <= 0) {
                state = 'gameover';
                try { localStorage.setItem('gravityClickerHighScore', Math.max(parseInt(localStorage.getItem('gravityClickerHighScore') || '0'), score)); } catch(e) {}
            }
        }
    }
    
    // Update ripples
    for (let i = ripples.length - 1; i >= 0; i--) {
        ripples[i].radius += 4;
        ripples[i].life -= 0.05;
        if (ripples[i].life <= 0) ripples.splice(i, 1);
    }
    
    // Update particles
    for (let i = particles.length - 1; i >= 0; i--) {
        const p = particles[i];
        p.x += p.vx;
        p.y += p.vy;
        p.vx *= 0.95;
        p.vy *= 0.95;
        p.life -= p.decay;
        if (p.life <= 0) particles.splice(i, 1);
    }
    
    // Shake
    if (shakeTime > 0) shakeTime -= dt;
}

// Render
function render() {
    ctx.save();
    
    // Screen shake
    if (shakeTime > 0) {
        const sx = (Math.random() - 0.5) * shakeIntensity * (shakeTime / 0.3);
        const sy = (Math.random() - 0.5) * shakeIntensity * (shakeTime / 0.3);
        ctx.translate(sx, sy);
    }
    
    // Background
    ctx.fillStyle = '#0a0a1a';
    ctx.fillRect(-10, -10, W + 20, H + 20);
    
    // Starfield
    for (const s of starfield) {
        s.twinkle += 0.02;
        const alpha = 0.3 + Math.sin(s.twinkle) * 0.3;
        ctx.fillStyle = `rgba(255,255,255,${alpha})`;
        ctx.beginPath();
        ctx.arc(s.x, s.y, s.size, 0, Math.PI * 2);
        ctx.fill();
    }
    
    if (state === 'playing' || state === 'gameover') {
        // Ripples
        for (const r of ripples) {
            const alpha = r.life * 0.5;
            ctx.strokeStyle = `rgba(100,200,255,${alpha})`;
            ctx.lineWidth = 2;
            ctx.beginPath();
            ctx.arc(r.x, r.y, r.radius, 0, Math.PI * 2);
            ctx.stroke();
        }
        
        // Stars
        for (const s of stars) {
            const pulse = 1 + Math.sin(s.pulse) * 0.15;
            const color = s.type === 'purple' ? '#a855f7' : '#fbbf24';
            
            ctx.shadowColor = color;
            ctx.shadowBlur = 15;
            ctx.fillStyle = color;
            ctx.beginPath();
            ctx.arc(s.x, s.y, s.radius * pulse, 0, Math.PI * 2);
            ctx.fill();
            ctx.shadowBlur = 0;
            
            // Inner glow
            ctx.fillStyle = s.type === 'purple' ? '#d8b4fe' : '#fef3c7';
            ctx.beginPath();
            ctx.arc(s.x, s.y, s.radius * pulse * 0.5, 0, Math.PI * 2);
            ctx.fill();
        }
        
        // Hazards
        for (const h of hazards) {
            const pulse = 1 + Math.sin(h.pulse) * 0.2;
            ctx.shadowColor = '#ef4444';
            ctx.shadowBlur = 20;
            ctx.fillStyle = '#ef4444';
            ctx.beginPath();
            ctx.arc(h.x, h.y, h.radius * pulse, 0, Math.PI * 2);
            ctx.fill();
            ctx.shadowBlur = 0;
            
            // Spikes
            for (let i = 0; i < 6; i++) {
                const a = (i / 6) * Math.PI * 2 + h.pulse * 0.5;
                const sx = h.x + Math.cos(a) * h.radius * pulse;
                const sy = h.y + Math.sin(a) * h.radius * pulse;
                ctx.fillStyle = '#fca5a5';
                ctx.beginPath();
                ctx.arc(sx, sy, 3, 0, Math.PI * 2);
                ctx.fill();
            }
        }
        
        // Particles
        for (const p of particles) {
            ctx.globalAlpha = p.life;
            ctx.fillStyle = p.color;
            ctx.beginPath();
            ctx.arc(p.x, p.y, p.size * p.life, 0, Math.PI * 2);
            ctx.fill();
        }
        ctx.globalAlpha = 1;
        
        // Player
        const pGlow = 10 + player.glow * 20;
        ctx.shadowColor = '#60a5fa';
        ctx.shadowBlur = pGlow;
        ctx.fillStyle = '#3b82f6';
        ctx.beginPath();
        ctx.arc(player.x, player.y, player.radius, 0, Math.PI * 2);
        ctx.fill();
        ctx.shadowBlur = 0;
        
        // Player inner
        ctx.fillStyle = '#93c5fd';
        ctx.beginPath();
        ctx.arc(player.x, player.y, player.radius * 0.6, 0, Math.PI * 2);
        ctx.fill();
        
        // Player ring
        ctx.strokeStyle = `rgba(147,197,253,${0.5 + player.glow * 0.5})`;
        ctx.lineWidth = 2;
        ctx.beginPath();
        ctx.arc(player.x, player.y, player.radius + 4, 0, Math.PI * 2);
        ctx.stroke();
        
        // HUD
        ctx.fillStyle = '#fff';
        ctx.font = 'bold 24px monospace';
        ctx.textAlign = 'left';
        ctx.fillText(`Score: ${score}`, 20, 35);
        ctx.font = '16px monospace';
        ctx.fillStyle = '#94a3b8';
        ctx.fillText(`Lives: ${'❤'.repeat(lives)}${'♡'.repeat(Math.max(0, 3 - lives))}`, 20, 60);
        
        if (combo > 1) {
            ctx.fillStyle = '#fbbf24';
            ctx.font = 'bold 18px monospace';
            ctx.textAlign = 'center';
            ctx.fillText(`Combo x${combo}!`, W / 2, 35);
        }
        
        // Difficulty indicator
        ctx.fillStyle = '#64748b';
        ctx.font = '12px monospace';
        ctx.textAlign = 'right';
        ctx.fillText(`Wave ${difficulty}`, W - 20, 30);
    }
    
    // Start Screen
    if (state === 'start') {
        drawStartScreen();
    }
    
    // Game Over
    if (state === 'gameover') {
        drawGameOver();
    }
    
    ctx.restore();
}

function drawStartScreen() {
    ctx.fillStyle = 'rgba(10,10,26,0.85)';
    ctx.fillRect(0, 0, W, H);
    
    ctx.textAlign = 'center';
    ctx.fillStyle = '#60a5fa';
    ctx.font = 'bold 48px monospace';
    ctx.fillText('GRAVITY CLICKER', W / 2, H / 3);
    
    ctx.fillStyle = '#e2e8f0';
    ctx.font = '18px monospace';
    ctx.fillText('Click to pull yourself toward the cursor', W / 2, H / 3 + 60);
    ctx.fillText('Collect gold stars (+10) and purple gems (+50)', W / 2, H / 3 + 95);
    ctx.fillText('Avoid red hazards!', W / 2, H / 3 + 130);
    ctx.fillText('Chain collections for combo multipliers', W / 2, H / 3 + 165);
    
    ctx.fillStyle = '#fbbf24';
    ctx.font = 'bold 22px monospace';
    const pulse = 0.7 + Math.sin(time * 3) * 0.3;
    ctx.globalAlpha = pulse;
    ctx.fillText('[ CLICK TO START ]', W / 2, H / 3 + 240);
    ctx.globalAlpha = 1;
    
    try {
        const hi = localStorage.getItem('gravityClickerHighScore') || '0';
        if (parseInt(hi) > 0) {
            ctx.fillStyle = '#94a3b8';
            ctx.font = '14px monospace';
            ctx.fillText(`High Score: ${hi}`, W / 2, H / 3 + 280);
        }
    } catch(e) {}
}

function drawGameOver() {
    ctx.fillStyle = 'rgba(10,10,26,0.75)';
    ctx.fillRect(0, 0, W, H);
    
    ctx.textAlign = 'center';
    ctx.fillStyle = '#ef4444';
    ctx.font = 'bold 42px monospace';
    ctx.fillText('GAME OVER', W / 2, H / 3);
    
    ctx.fillStyle = '#e2e8f0';
    ctx.font = '24px monospace';
    ctx.fillText(`Final Score: ${score}`, W / 2, H / 3 + 50);
    
    try {
        const hi = localStorage.getItem('gravityClickerHighScore') || '0';
        if (score >= parseInt(hi)) {
            ctx.fillStyle = '#fbbf24';
            ctx.font = '18px monospace';
            ctx.fillText('NEW HIGH SCORE!', W / 2, H / 3 + 90);
        } else {
            ctx.fillStyle = '#94a3b8';
            ctx.font = '16px monospace';
            ctx.fillText(`High Score: ${hi}`, W / 2, H / 3 + 90);
        }
    } catch(e) {}
    
    ctx.fillStyle = '#60a5fa';
    ctx.font = 'bold 20px monospace';
    const pulse = 0.7 + Math.sin(time * 3) * 0.3;
    ctx.globalAlpha = pulse;
    ctx.fillText('[ CLICK TO RESTART ]', W / 2, H / 3 + 150);
    ctx.globalAlpha = 1;
}

// Game Loop
let lastTime = 0;
function gameLoop(timestamp) {
    const dt = Math.min((timestamp - lastTime) / 1000, 0.05);
    lastTime = timestamp;
    time += dt * (state === 'start' || state === 'gameover' ? 1 : 0); // keep time running for animations
    
    update(dt);
    render();
    requestAnimationFrame(gameLoop);
}

initStarfield();
requestAnimationFrame(gameLoop);
</script>
</body>
</html>
```
