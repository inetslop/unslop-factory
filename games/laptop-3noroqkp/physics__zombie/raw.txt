const canvas = document.getElementById('game');
const ctx = canvas.getContext('2d');

let W, H;
function resize() {

// Audio
let audioCtx = null;
function initAudio() {
    if (!audioCtx) audioCtx = new (window.AudioContext || webkitAudioContext)();
}
function playSound(type) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    if (type === 'shoot') {
        osc.type = 'sawtooth';
        osc.frequency.setValueAtTime(150, audioCtx.currentTime);
        osc.frequency.exponentialRampToValueAtTime(30, audioCtx.currentTime + 0.15);
        gain.gain.setValueAtTime(0.3, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.15);
        osc.start(); osc.stop(audioCtx.currentTime + 0.15);
    } else if (type === 'zombieHit') {
        osc.type = 'square';
        osc.frequency.setValueAtTime(200, audioCtx.currentTime);
        osc.frequency.exponentialRampToValueAtTime(80, audioCtx.currentTime + 0.1);
        gain.gain.setValueAtTime(0.2, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.1);
        osc.start(); osc.stop(audioCtx.currentTime + 0.1);
    } else if (type === 'zombieDie') {
        osc.type = 'sawtooth';
        osc.frequency.setValueAtTime(300, audioCtx.currentTime);
        osc.frequency.exponentialRampToValueAtTime(50, audioCtx.currentTime + 0.4);
        gain.gain.setValueAtTime(0.25, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.4);
        osc.start(); osc.stop(audioCtx.currentTime + 0.4);
    } else if (type === 'playerHit') {
        osc.type = 'sawtooth';
        osc.frequency.setValueAtTime(100, audioCtx.currentTime);
        osc.frequency.exponentialRampToValueAtTime(40, audioCtx.currentTime + 0.2);
        gain.gain.setValueAtTime(0.35, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.2);
        osc.start(); osc.stop(audioCtx.currentTime + 0.2);
    } else if (type === 'jump') {
        osc.type = 'sine';
        osc.frequency.setValueAtTime(200, audioCtx.currentTime);
        osc.frequency.exponentialRampToValueAtTime(400, audioCtx.currentTime + 0.1);
        gain.gain.setValueAtTime(0.15, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.1);
        osc.start(); osc.stop(audioCtx.currentTime + 0.1);
    } else if (type === 'wave') {
        osc.type = 'sine';
        osc.frequency.setValueAtTime(440, audioCtx.currentTime);
        gain.gain.setValueAtTime(0.2, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.5);
        osc.start(); osc.stop(audioCtx.currentTime + 0.5);
    }
}

// Game state
let gameState = 'start'; // start, playing, gameover
let score = 0;
let wave = 0;
let zombiesRemaining = 0;
let zombiesToSpawn = 0;
let spawnTimer = 0;
let shakeTime = 0;
let shakeMag = 0;

// Input
const keys = {};
let mouseX = 0, mouseY = 0;
let mouseDown = false;
window.addEventListener('keydown', e => { keys[e.code] = true; if (gameState === 'start' || gameState === 'gameover') startGame(); });
window.addEventListener('keyup', e => { keys[e.code] = false; });
canvas.addEventListener('mousemove', e => { mouseX = e.clientX; mouseY = e.clientY; });
canvas.addEventListener('mousedown', e => { initAudio(); mouseDown = true; if (gameState === 'start' || gameState === 'gameover') startGame(); });
canvas.addEventListener('mouseup', () => { mouseDown = false; });

// Player
let player = {};
function resetPlayer() {
    player = {
        x: W / 2, y: H - 80, vx: 0, vy: 0,
        w: 30, h: 50, health: 100, maxHealth: 100,
        onGround: false, facing: 1, shootCooldown: 0,
        invincible: 0, hitFlash: 0
    };
}

// Entities
let zombies = [];
let bullets = [];
let particles = [];
let muzzleFlashes = [];

const GRAVITY = 0.6;
const GROUND_Y = () => H - 40;

function spawnZombie() {
    const side = Math.random() < 0.5 ? -1 : 1;
    const x = side === -1 ? -30 : W + 30;
    const speed = 0.8 + wave * 0.15 + Math.random() * 0.5;
    zombies.push({
        x, y: GROUND_Y() - 45, vx: side * speed, vy: 0,
        w: 30, h: 50, health: 2 + Math.floor(wave / 3), maxHealth: 2 + Math.floor(wave / 3),
        rotation: 0, rotVel: 0, dead: false, deadTimer: 0,
        hitFlash: 0, attackCooldown: 0, speed
    });
}

function startGame() {
    initAudio();
    gameState = 'playing';
    score = 0;
    wave = 0;
    zombiesRemaining = 0;
    zombiesToSpawn = 0;
    spawnTimer = 0;
    shakeTime = 0;
    resetPlayer();
    zombies = [];
    bullets = [];
    particles = [];
    muzzleFlashes = [];
    nextWave();
}

function nextWave() {
    wave++;
    const count = 3 + wave * 2;
    zombiesToSpawn = count;
    zombiesRemaining = count;
    spawnTimer = 0;
    playSound('wave');
}

function shootBullet(px, py, angle) {
    const speed = 12;
    bullets.push({
        x: px, y: py, vx: Math.cos(angle) * speed, vy: Math.sin(angle) * speed,
        radius: 4, life: 0
    });
    muzzleFlashes.push({ x: px, y: py, angle: angle, life: 5 });
    playSound('shoot');
}

function spawnParticles(x, y, color, count, spread) {
    for (let i = 0; i < count; i++) {
        const a = Math.random() * Math.PI * 2;
        const s = Math.random() * spread + 1;
        particles.push({
            x, y, vx: Math.cos(a) * s, vy: Math.sin(a) * s - 2,
            life: 30 + Math.random() * 20, maxLife: 50,
            color, size: Math.random() * 4 + 2
        });
    }
}

function addShake(mag) { shakeMag = mag; shakeTime = 8; }

function circleCollide(ax, ay, ar, bx, by, br) {
    const dx = ax - bx, dy = ay - by;
    return dx * dx + dy * dy < (ar + br) * (ar + br);
}

function update() {
    if (gameState !== 'playing') return;

    // Spawn zombies
    if (zombiesToSpawn > 0) {
        spawnTimer++;
        if (spawnTimer >= 40 - wave * 2) {
            spawnZombie();
            zombiesToSpawn--;
            spawnTimer = 0;
        }
    }

    // Player physics
    const moveSpeed = 5;
    if (keys['KeyA'] || keys['ArrowLeft']) player.vx = -moveSpeed;
    else if (keys['KeyD'] || keys['ArrowRight']) player.vx = moveSpeed;
    else player.vx *= 0.7;

    if ((keys['KeyW'] || keys['Space'] || keys['ArrowUp']) && player.onGround) {
        player.vy = -13;
        player.onGround = false;
        playSound('jump');
    }

    player.vy += GRAVITY;
    player.x += player.vx;
    player.y += player.vy;

    if (player.y + player.h / 2 >= GROUND_Y()) {
        player.y = GROUND_Y() - player.h / 2;
        player.vy = 0;
        player.onGround = true;
    }
    player.x = Math.max(player.w / 2, Math.min(W - player.w / 2, player.x));

    if (player.invincible > 0) player.invincible--;
    if (player.hitFlash > 0) player.hitFlash--;

    // Shooting
    if (player.shootCooldown > 0) player.shootCooldown--;
    const aimAngle = Math.atan2(mouseY - player.y, mouseX - player.x);
    if (mouseDown && player.shootCooldown <= 0) {
        shootBullet(player.x + Math.cos(aimAngle) * 25, player.y + Math.sin(aimAngle) * 25, aimAngle);
        player.shootCooldown = 10;
        player.vx -= Math.cos(aimAngle) * 1.5; // recoil
    }

    // Update bullets
    for (let i = bullets.length - 1; i >= 0; i--) {
        const b = bullets[i];
        b.vy += GRAVITY * 0.15; // slight arc
        b.x += b.vx;
        b.y += b.vy;
        b.life++;
        if (b.x < -50 || b.x > W + 50 || b.y > H + 50 || b.life > 120) {
            bullets.splice(i, 1);
            continue;
        }
        // Hit zombies
        let hit = false;
        for (let j = 0; j < zombies.length; j++) {
            const z = zombies[j];
            if (z.dead) continue;
            if (circleCollide(b.x, b.y, b.radius, z.x, z.y, 25)) {
                hit = true;
                z.health--;
                z.hitFlash = 8;
                // Knock back
                const knockDir = Math.sign(z.x - player.x) || 1;
                z.vx += knockDir * 4 + b.vx * 0.3;
                z.vy -= 3;
                playSound('zombieHit');
                spawnParticles(b.x, b.y, '#4a7', 5, 3);
                if (z.health <= 0) {
                    z.dead = true;
                    z.rotVel = knockDir * 0.15;
                    z.vy = -8;
                    z.vx += knockDir * 5;
                    score += 10 * wave;
                    playSound('zombieDie');
                    spawnParticles(z.x, z.y, '#3a3', 12, 5);
                }
                break;
            }
        }
        if (hit) { bullets.splice(i, 1); continue; }
    }

    // Update zombies
    for (let i = zombies.length - 1; i >= 0; i--) {
        const z = zombies[i];
        if (z.dead) {
            z.vy += GRAVITY;
            z.x += z.vx;
            z.y += z.vy;
            z.rotation += z.rotVel;
            z.vx *= 0.95;
            z.deadTimer++;
            if (z.y + z.h / 2 >= GROUND_Y()) {
                z.y = GROUND_Y() - z.h / 2;
                z.vy = 0;
                z.rotVel *= 0.8;
            }
            if (z.deadTimer > 90) { zombies.splice(i, 1); }
            continue;
        }

        if (z.hitFlash > 0) z.hitFlash--;

        // Walk toward player
        const dir = Math.sign(player.x - z.x) || 1;
        z.vx += dir * 0.05;
        z.vx = Math.max(-z.speed, Math.min(z.speed, z.vx));
        z.vy += GRAVITY;

        // Jump occasionally if blocked or for fun
        if (Math.random() < 0.01 && z.y + z.h / 2 >= GROUND_Y() - 2) {
            z.vy = -8;
        }

        z.x += z.vx;
        z.y += z.vy;
        z.rotation *= 0.9; // return to upright

        if (z.y + z.h / 2 >= GROUND_Y()) {
            z.y = GROUND_Y() - z.h / 2;
            z.vy = 0;
        }

        // Attack player
        if (z.attackCooldown > 0) z.attackCooldown--;
        const dx = z.x - player.x, dy = z.y - player.y;
        const dist = Math.sqrt(dx * dx + dy * dy);
        if (dist < 40 && z.attackCooldown <= 0 && player.invincible <= 0) {
            player.health -= 10;
            player.invincible = 30;
            player.hitFlash = 10;
            player.vx += Math.sign(player.x - z.x) * 8;
            player.vy -= 5;
            player.onGround = false;
            addShake(5);
            playSound('playerHit');
            spawnParticles(player.x, player.y, '#a33', 6, 4);
            z.attackCooldown = 60;
            if (player.health <= 0) {
                gameState = 'gameover';
            }
        }

        // Zombie-zombie collision (simple push)
        for (let j = i + 1; j < zombies.length; j++) {
            const z2 = zombies[j];
            if (z2.dead) continue;
            const ddx = z.x - z2.x, ddy = z.y - z2.y;
            const dd = Math.sqrt(ddx * ddx + ddy * ddy);
            if (dd < 35 && dd > 0) {
                const push = (35 - dd) * 0.3;
                z.x += (ddx / dd) * push;
                z2.x -= (ddx / dd) * push;
            }
        }
    }

    // Check wave complete
    if (zombiesToSpawn === 0 && zombies.filter(z => !z.dead).length === 0 && zombies.length <= 5) {
        nextWave();
    }

    // Particles
    for (let i = particles.length - 1; i >= 0; i--) {
        const p = particles[i];
        p.vy += GRAVITY * 0.3;
        p.x += p.vx;
        p.y += p.vy;
        p.life--;
        if (p.life <= 0) particles.splice(i, 1);
    }

    // Muzzle flashes
    for (let i = muzzleFlashes.length - 1; i >= 0; i--) {
        muzzleFlashes[i].life--;
        if (muzzleFlashes[i].life <= 0) muzzleFlashes.splice(i, 1);
    }

    // Shake
    if (shakeTime > 0) shakeTime--;
}

function draw() {
    ctx.save();
    if (shakeTime > 0) {
        const sx = (Math.random() - 0.5) * shakeMag;
        const sy = (Math.random() - 0.5) * shakeMag;
        ctx.translate(sx, sy);
    }

    // Background
    const grad = ctx.createLinearGradient(0, 0, 0, H);
    grad.addColorStop(0, '#1a1a2e');
    grad.addColorStop(0.7, '#16213e');
    grad.addColorStop(1, '#0f3460');
    ctx.fillStyle = grad;
    ctx.fillRect(0, 0, W, H);

    // Stars
    ctx.fillStyle = 'rgba(255,255,255,0.3)';
    for (let i = 0; i < 50; i++) {
        const sx = (i * 137 + 50) % W;
        const sy = (i * 97 + 20) % (H * 0.6);
        ctx.fillRect(sx, sy, 2, 2);
    }

    // Moon
    ctx.beginPath();
    ctx.arc(W - 100, 80, 35, 0, Math.PI * 2);
    ctx.fillStyle = '#e8d5b7';
    ctx.fill();
    ctx.beginPath();
    ctx.arc(W - 90, 75, 30, 0, Math.PI * 2);
    ctx.fillStyle = '#1a1a2e';
    ctx.fill();

    // Ground
    const gy = GROUND_Y();
    ctx.fillStyle = '#2d4a2d';
    ctx.fillRect(0, gy, W, H - gy);
    ctx.fillStyle = '#3a5f3a';
    for (let x = 0; x < W; x += 20) {
        ctx.fillRect(x, gy, 12, 4);
    }

    // Particles
    for (const p of particles) {
        const alpha = p.life / p.maxLife;
        ctx.fillStyle = p.color + Math.floor(alpha * 99).toString(16).padStart(2, '0');
        ctx.globalAlpha = alpha;
        ctx.beginPath();
        ctx.arc(p.x, p.y, p.size * alpha, 0, Math.PI * 2);
        ctx.fill();
    }
    ctx.globalAlpha = 1;

    // Zombies
    for (const z of zombies) {
        ctx.save();
        ctx.translate(z.x, z.y);
        ctx.rotate(z.rotation);

        const color = z.dead ? '#3a5f3a' : (z.hitFlash > 0 ? '#ff6644' : '#4a7a4a');
        // Body
        ctx.fillStyle = color;
        ctx.fillRect(-12, -15, 24, 35);
        // Head
        ctx.beginPath();
        ctx.arc(0, -28, 12, 0, Math.PI * 2);
        ctx.fill();
        // Eyes
        if (!z.dead) {
            ctx.fillStyle = '#ff0';
            ctx.fillRect(-6, -32, 4, 4);
            ctx.fillRect(3, -32, 4, 4);
            // Arms reaching
            const armAngle = Math.sin(Date.now() * 0.01 + z.x) * 0.3;
            ctx.strokeStyle = color;
            ctx.lineWidth = 5;
            ctx.beginPath();
            ctx.moveTo(10, -5);
            ctx.lineTo(25, -10 + armAngle * 10);
            ctx.stroke();
            ctx.beginPath();
            ctx.moveTo(-10, -5);
            ctx.lineTo(-25, -8 - armAngle * 10);
            ctx.stroke();
        } else {
            // X eyes for dead
            ctx.strokeStyle = '#fff';
            ctx.lineWidth = 2;
            ctx.beginPath(); ctx.moveTo(-8, -32); ctx.lineTo(-4, -28); ctx.stroke();
            ctx.beginPath(); ctx.moveTo(-4, -32); ctx.lineTo(-8, -28); ctx.stroke();
            ctx.beginPath(); ctx.moveTo(5, -32); ctx.lineTo(9, -28); ctx.stroke();
            ctx.beginPath(); ctx.moveTo(9, -32); ctx.lineTo(5, -28); ctx.stroke();
        }
        // Legs
        ctx.strokeStyle = color;
        ctx.lineWidth = 5;
        const legPhase = z.dead ? 0 : Math.sin(Date.now() * 0.008 + z.x) * (z.dead ? 0 : 1);
        ctx.beginPath(); ctx.moveTo(-6, 20); ctx.lineTo(-6 - legPhase * 5, 38); ctx.stroke();
        ctx.beginPath(); ctx.moveTo(6, 20); ctx.lineTo(6 + legPhase * 5, 38); ctx.stroke();

        ctx.restore();

        // Health bar for non-dead zombies
        if (!z.dead && z.health < z.maxHealth) {
            const bw = 30;
            ctx.fillStyle = '#333';
            ctx.fillRect(z.x - bw / 2, z.y - 48, bw, 5);
            ctx.fillStyle = '#e74c3c';
            ctx.fillRect(z.x - bw / 2, z.y - 48, bw * (z.health / z.maxHealth), 5);
        }
    }

    // Bullets
    ctx.fillStyle = '#ffdd44';
    for (const b of bullets) {
        ctx.beginPath();
        ctx.arc(b.x, b.y, b.radius, 0, Math.PI * 2);
        ctx.fill();
        // Trail
        ctx.strokeStyle = 'rgba(255,221,68,0.4)';
        ctx.lineWidth = 2;
        ctx.beginPath();
        ctx.moveTo(b.x, b.y);
        ctx.lineTo(b.x - b.vx * 0.5, b.y - b.vy * 0.5);
        ctx.stroke();
    }

    // Muzzle flash
    for (const m of muzzleFlashes) {
        const a = m.life / 5;
        ctx.fillStyle = `rgba(255,255,150,${a})`;
        ctx.beginPath();
        ctx.arc(m.x + Math.cos(m.angle) * 10, m.y + Math.sin(m.angle) * 10, 8 * a, 0, Math.PI * 2);
        ctx.fill();
    }

    // Player
    if (player.invincible <= 0 || Math.floor(player.invincible / 4) % 2 === 0) {
        ctx.save();
        ctx.translate(player.x, player.y);
        const pColor = player.hitFlash > 0 ? '#ff6666' : '#ddd';
        // Body
        ctx.fillStyle = pColor;
        ctx.fillRect(-13, -15, 26, 35);
        // Head
        ctx.beginPath();
        ctx.arc(0, -28, 13, 0, Math.PI * 2);
        ctx.fill();
        // Eyes
        ctx.fillStyle = '#333';
        const eyeDir = Math.sign(mouseX - player.x) || 1;
        ctx.fillRect(-4 + eyeDir * 2, -32, 3, 3);
        ctx.fillRect(4 + eyeDir * 2, -32, 3, 3);
        // Gun arm
        const gunAngle = Math.atan2(mouseY - player.y, mouseX - player.x);
        ctx.strokeStyle = '#555';
        ctx.lineWidth = 6;
        ctx.beginPath();
        ctx.moveTo(0, -5);
        ctx.lineTo(Math.cos(gunAngle) * 25, Math.sin(gunAngle) * 25);
        ctx.stroke();
        // Legs
        const legAnim = player.onGround ? Math.abs(player.vx) > 1 ? Math.sin(Date.now() * 0.01) * 5 : 0 : 3;
        ctx.strokeStyle = pColor;
        ctx.lineWidth = 5;
        ctx.beginPath(); ctx.moveTo(-6, 20); ctx.lineTo(-6 - legAnim, 38); ctx.stroke();
        ctx.beginPath(); ctx.moveTo(6, 20); ctx.lineTo(6 + legAnim, 38); ctx.stroke();
        ctx.restore();
    }

    // HUD
    if (gameState === 'playing') {
        // Health bar
        ctx.fillStyle = '#333';
        ctx.fillRect(20, 20, 200, 25);
        const hpPct = Math.max(0, player.health / player.maxHealth);
        ctx.fillStyle = hpPct > 0.5 ? '#2ecc71' : hpPct > 0.25 ? '#f39c12' : '#e74c3c';
        ctx.fillRect(20, 20, 200 * hpPct, 25);
        ctx.strokeStyle = '#fff';
        ctx.lineWidth = 2;
        ctx.strokeRect(20, 20, 200, 25);
        ctx.fillStyle = '#fff';
        ctx.font = 'bold 14px monospace';
        ctx.fillText(`HP: ${Math.max(0, Math.round(player.health))}`, 25, 37);

        // Wave & Score
        ctx.font = 'bold 18px monospace';
        ctx.fillStyle = '#fff';
        ctx.textAlign = 'right';
        ctx.fillText(`WAVE ${wave}`, W - 20, 35);
        ctx.fillText(`SCORE: ${score}`, W - 20, 60);
        ctx.textAlign = 'left';

        // Zombie count
        const alive = zombies.filter(z => !z.dead).length;
        if (alive > 0) {
            ctx.fillStyle = '#a3a3a3';
            ctx.font = '14px monospace';
            ctx.fillText(`Zombies: ${alive + zombiesToSpawn}`, W - 20, 85);
        }
    }

    // Start screen
    if (gameState === 'start') {
        ctx.fillStyle = 'rgba(0,0,0,0.7)';
        ctx.fillRect(0, 0, W, H);
        ctx.textAlign = 'center';
        ctx.fillStyle = '#4a7a4a';
        ctx.font = 'bold 60px monospace';
        ctx.fillText('PHYSICS ZOMBIE', W / 2, H / 2 - 80);
        ctx.fillStyle = '#fff';
        ctx.font = '20px monospace';
        ctx.fillText('Physics-based zombie survival', W / 2, H / 2 - 30);
        ctx.font = '16px monospace';
        ctx.fillStyle = '#ccc';
        ctx.fillText('A/D or ←/→ : Move', W / 2, H / 2 + 20);
        ctx.fillText('W / Space : Jump', W / 2, H / 2 + 45);
        ctx.fillText('Mouse : Aim & Click to Shoot', W / 2, H / 2 + 70);
        ctx.fillStyle = '#8f8';
        ctx.font = '18px monospace';
        ctx.fillText('[ Press any key or click to start ]', W / 2, H / 2 + 130);
        ctx.textAlign = 'left';
    }

    // Game over
    if (gameState === 'gameover') {
        ctx.fillStyle = 'rgba(80,0,0,0.7)';
        ctx.fillRect(0, 0, W, H);
        ctx.textAlign = 'center';
        ctx.fillStyle = '#e74c3c';
        ctx.font = 'bold 50px monospace';
        ctx.fillText('YOU DIED', W / 2, H / 2 - 40);
        ctx.fillStyle = '#fff';
        ctx.font = '24px monospace';
        ctx.fillText(`Final Score: ${score}`, W / 2, H / 2 + 10);
        ctx.fillText(`Reached Wave: ${wave}`, W / 2, H / 2 + 45);
        ctx.fillStyle = '#8f8';
        ctx.font = '18px monospace';
        ctx.fillText('[ Press any key or click to restart ]', W / 2, H / 2 + 100);
        ctx.textAlign = 'left';
    }

    // Crosshair
    if (gameState === 'playing') {
        ctx.strokeStyle = '#ff4444';
        ctx.lineWidth = 1.5;
        const cs = 10;
        ctx.beginPath();
        ctx.moveTo(mouseX - cs, mouseY); ctx.lineTo(mouseX + cs, mouseY);
        ctx.moveTo(mouseX, mouseY - cs); ctx.lineTo(mouseX, mouseY + cs);
        ctx.stroke();
    }

    ctx.restore();
}

function loop() {
    update();
    draw();
    requestAnimationFrame(loop);
}

resetPlayer();
loop();
