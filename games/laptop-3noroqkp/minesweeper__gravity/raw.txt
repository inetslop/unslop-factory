<script>
const COLS = 10, ROWS = 10, MINES = 25;
let grid = [], revealed = [], flagged = [];
let gameOver = false, won = false;
let minesLeft = MINES;
let shiftTimer = 0, shiftInterval = 8000;
let gravityDir = [0,1]; // dx,dy
const dirs = [[0,1],[1,0],[0,-1],[-1,0]];
let lastTime = 0, animating = false;
let audioCtx = null;

function initAudio() {
  if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
}

function playTone(freq, dur, type='sine', vol=0.15) {
  if (!audioCtx) return;
  const o = audioCtx.createOscillator();
  const g = audioCtx.createGain();
  o.type = type; o.frequency.value = freq;
  g.gain.setValueAtTime(vol, audioCtx.currentTime);
  g.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
  o.connect(g); g.connect(audioCtx.destination);
  o.start(); o.stop(audioCtx.currentTime + dur);
}

function playReveal() { playTone(600, 0.1, 'sine', 0.1); }
function playFlag() { playTone(400, 0.08, 'square', 0.08); }
function playShift() { playTone(200, 0.3, 'sawtooth', 0.06); }
function playLose() { playTone(150, 0.5, 'sawtooth', 0.2); setTimeout(()=>playTone(100,0.8,'sawtooth',0.2),200); }
function playWin() { playTone(523,0.15);setTimeout(()=>playTone(659,0.15),150);setTimeout(()=>playTone(784,0.3),300); }

function startGame() {
  initAudio();
  document.getElementById('startScreen').style.display = 'none';
  document.getElementById('gameArea').style.display = 'flex';
  gameOver = false; won = false;
  minesLeft = MINES;
  shiftTimer = 0;
  gravityDir = [0,1];
  updateGravityIndicator();
  
  grid = Array.from({length:ROWS},()=>Array(COLS).fill(0));
  revealed = Array.from({length:ROWS},()=>Array(COLS).fill(false));
  flagged = Array.from({length:ROWS},()=>Array(COLS).fill(false));
  
  // Place mines randomly (avoiding center 3x3)
  let placed = 0;
  while (placed < MINES) {
    const r = Math.floor(Math.random()*ROWS);
    const c = Math.floor(Math.random()*COLS);
    if (grid[r][c] === 0 && !(r>=4&&r<=5&&c>=4&&c<=5)) {
      grid[r][c] = -1; placed++;
    }
  }
  
  document.getElementById('mineCount').textContent = MINES;
  document.getElementById('revealedCount').textContent = '0/75';
  document.getElementById('msg').textContent = '';
  renderGrid();
  lastTime = performance.now();
  requestAnimationFrame(gameLoop);
}

function updateGravityIndicator() {
  const arrows = ['↓','→','↑','←'];
  const idx = dirs.findIndex(d => d[0]===gravityDir[0] && d[1]===gravityDir[1]);
  document.getElementById('gravityIndicator').textContent = arrows[idx>=0?idx:0];
}

function renderGrid() {
  const gridEl = document.getElementById('grid');
  gridEl.innerHTML = '';
  gridEl.style.gridTemplateColumns = `repeat(${COLS}, 36px)`;
  
  for (let r=0;r<ROWS;r++) {
    for (let c=0;c<COLS;c++) {
      const cell = document.createElement('div');
      cell.className = 'cell';
      cell.dataset.r = r; cell.dataset.c = c;
      
      if (revealed[r][c]) {
        cell.classList.add('revealed');
        if (grid[r][c] === -1) {
          cell.textContent = '💀';
          cell.classList.add('mine-hit');
        } else {
          const n = getMineCount(r,c);
          if (n > 0) {
            cell.textContent = n;
            const span = document.createElement('span');
            span.className = `num-${n}`;
            span.textContent = n;
            cell.textContent = '';
            cell.appendChild(span);
          }
        }
      } else {
        cell.classList.add('hidden');
        if (flagged[r][c]) {
          cell.classList.add('flagged');
          cell.textContent = '🚩';
        }
      }
      
      cell.addEventListener('click', () => handleClick(r,c));
      cell.addEventListener('contextmenu', e => { e.preventDefault(); handleFlag(r,c); });
      gridEl.appendChild(cell);
    }
  }
}

function getMineCount(r,c) {
  let count = 0;
  for (let dr=-1;dr<=1;dr++) {
    for (let dc=-1;dc<=1;dc++) {
      if (dr===0&&dc===0) continue;
      const nr=r+dr, nc=c+dc;
      if (nr>=0&&nr<ROWS&&nc>=0&&nc<COLS&&grid[nr][nc]===-1) count++;
    }
  }
  return count;
}

function handleClick(r,c) {
  if (gameOver || flagged[r][c]) return;
  if (revealed[r][c]) return;
  
  initAudio();
  
  if (grid[r][c] === -1) {
    // Hit a mine!
    revealed[r][c] = true;
    gameOver = true;
    playLose();
    document.getElementById('msg').textContent = '💥 BOOM!';
    document.body.classList.add('shake');
    setTimeout(()=>document.body.classList.remove('shake'),500);
    revealAllMines();
    renderGrid();
    return;
  }
  
  // Reveal (flood fill for 0s)
  floodReveal(r,c);
  playReveal();
  checkWin();
  renderGrid();
}

function floodReveal(sr,sc) {
  const stack = [[sr,sc]];
  while (stack.length) {
    const [r,c] = stack.pop();
    if (r<0||r>=ROWS||c<0||c>=COLS) continue;
    if (revealed[r][c] || flagged[r][c]) continue;
    if (grid[r][c] === -1) { revealed[r][c]=true; gameOver=true; revealAllMines(); return; }
    revealed[r][c] = true;
    if (getMineCount(r,c) === 0) {
      for(let dr=-1;dr<=1;dr++)for(let dc=-1;dc<=1;dc++){
        if(dr!==0||dc!==0) stack.push([r+dr,c+dc]);
      }
    }
  }
}

function handleFlag(r,c) {
  if (gameOver || revealed[r][c]) return;
  initAudio();
  flagged[r][c] = !flagged[r][c];
  playFlag();
  renderGrid();
}

function revealAllMines() {
  for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++){
    if(grid[r][c]===-1) revealed[r][c]=true;
  }
}

function checkWin() {
  let revealedCount = 0;
  for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++) if(revealed[r][c]&&grid[r][c]!==-1) revealedCount++;
  document.getElementById('revealedCount').textContent = `${revealedCount}/75`;
  
  if (revealedCount === ROWS*COLS - MINES) {
    won = true; gameOver = true;
    playWin();
    document.getElementById('msg').textContent = '🏆 YOU WIN!';
  } else if (!gameOver) {
    document.getElementById('revealedCount').textContent = `${revealedCount}/75`;
  }
}

function gameLoop(time) {
  if (gameOver) return;
  
  const dt = time - lastTime;
  lastTime = time;
  shiftTimer += dt;
  
  // Update timer bar
  const pct = Math.max(0, 100 - (shiftTimer/shiftInterval)*100);
  document.getElementById('timerFill').style.width = pct + '%';
  
  if (shiftTimer >= shiftInterval) {
    performShift();
    shiftTimer = 0;
    // Change gravity direction occasionally
    if (Math.random() < 0.35) {
      const newDir = dirs[Math.floor(Math.random()*dirs.length)];
      gravityDir = newDir;
      updateGravityIndicator();
    }
  }
  
  requestAnimationFrame(gameLoop);
}

function performShift() {
  if (gameOver) return;
  playShift();
  
  const dx = gravityDir[0], dy = gravityDir[1];
  
  // Collect all mine positions
  let mines = [];
  for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++) if(grid[r][c]===-1) mines.push([r,c]);
  
  // Sort mines so they move in order (to handle stacking)
  // Process in the direction of movement first
  const sorted = [...mines].sort((a,b) => {
    if (dy > 0) return a[0]-b[0];
    if (dy < 0) return b[0]-a[0];
    if (dx > 0) return a[1]-b[1];
    return b[1]-a[1];
  });
  
  const mineSet = new Set(mines.map(m=>m[0]*COLS+m[1]));
  let movedAny = false;
  
  for (const [r,c] of sorted) {
    const nr = r+dy, nc = c+dx;
    if (nr<0||nr>=ROWS||nc<0||nc>=COLS) continue;
    const key = nr*COLS+nc;
    
    // Can't move into another mine or a revealed cell
    if (mineSet.has(key)) continue;
    if (revealed[nr][nc] && grid[nr][nc] !== -1) {
      // Mine would move into revealed safe cell = LOSE!
      gameOver = true;
      grid[r][c]=0; grid[nr][nc]=-1;
      mineSet.delete(r*COLS+c); mineSet.add(key);
      revealAllMines();
      playLose();
      document.getElementById('msg').textContent = '🌍 GRAVITY CRUSHED YOU!';
      document.body.classList.add('shake');
      setTimeout(()=>document.body.classList.remove('shake'),500);
      renderGrid();
      return;
    }
    
    // Move the mine
    grid[r][c] = 0;
    grid[nr][nc] = -1;
    mineSet.delete(r*COLS+c);
    mineSet.add(key);
    movedAny = true;
  }
  
  if (movedAny && !gameOver) {
    // Add shift animation class
    renderGrid();
    const cells = document.querySelectorAll('.cell');
    setTimeout(()=>{
      cells.forEach(cell=>{
        cell.classList.add('shifting');
        setTimeout(()=>cell.classList.remove('shifting'),400);
      });
    },10);
  }
  
  // Update revealed count display
  let rc = 0;
  for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++) if(revealed[r][c]&&grid[r][c]!==-1) rc++;
  document.getElementById('revealedCount').textContent = `${rc}/75`;
}

// Keyboard support
window.addEventListener('keydown', e => {
  if (e.key === 'r' || e.key === 'R') startGame();
});
</script>
