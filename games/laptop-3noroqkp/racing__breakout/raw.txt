<!DOCTYPE html>

<canvas id="game"></canvas>
<script>
(function() {
const canvas = document.getElementById('game');
const ctx = canvas.getContext('2d');

let W, H;
function resize() {
W = window.innerWidth; H = window.innerHeight;
canvas.width = W; canvas.height = H;
}
window.addEventListener('resize', resize);
resize();

// Audio
let audioCtx = null;
function initAudio() {
if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
}
function playSound(freq, duration, type, vol) {
if (!audioCtx) return;
const osc = audioCtx.createOscillator();
const gain = audioCtx.createGain();
osc.type = type || 'square';
osc.frequency.value = freq;
gain.gain.setValueAtTime(vol || 0.15, audioCtx.currentTime);
gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + duration);
osc.connect(gain); gain.connect(audioCtx.destination);
osc.start(); osc.stop(audioCtx.currentTime + duration);
}

// Game state
let state = 'start'; // start, playing, ballheld, gameover, win
let score = 0;
let lives = 3;
let level = 1;
let particles = [];
let speedLines = [];

// Car
const car = { x: 0, y: 0, w: 80, h: 40, vx: 0, speed: 8 };

// Ball
const ball = { x: 0, y: 0, r: 8, vx: 0, vy: 0, speed: 6, held: true };

// Bricks
let bricks = [];
const brickColors = [
['#ff4444','#cc2222'],['#ff8800','#cc6600'],['#ffcc00','#cc9900'],
['#44ff44','#22cc22'],['#44aaff','#2277cc'],['#cc44ff','#9922cc']
];

function createBricks(lv) {
bricks = [];
const rows = 5 + lv;
const cols = Math.min(10, 8 + Math.floor(lv / 2));
const bw = (W - 60) / cols;
const bh = 28;
const by = 70;
for (let r = 0; r < rows; r++) {
for (let c = 0; c < cols; c++) {
if (Math.random() > 0.15 || lv <= 1) {
bricks.push({ x: 30 + c * bw, y: by + r * bh, w: bw - 4, h: bh - 4, alive: true, hp: lv > 2 ? 2 : 1 });
}
}
}
}

function resetBall() {
ball.held = true;
ball.x = car.x + car.w / 2;
ball.y = car.y - ball.r - 5;
ball.vx = 0; ball.vy = 0;
state = 'ballheld';
}

function launchBall() {
if (ball.held) {
ball.held = false;
const angle = (-60 + Math.random() * 40) * Math.PI / 180;
ball.vx = Math.sin(angle) * ball.speed;
ball.vy = -Math.cos(angle) * ball.speed;
state = 'playing';
playSound(440, 0.1, 'sine', 0.2);
}
}

function startGame() {
score = 0; lives = 3; level = 1;
car.x = W / 2 - car.w / 2;
car.y = H - 60;
particles = [];
createBricks(1);
resetBall();
state = 'ballheld';
}

function nextLevel() {
level++;
if (level > 5) { level = 1; }
createBricks(level);
ball.speed += 0.3;
resetBall();
}

// Input
let keys = {};
window.addEventListener('keydown', e => {
keys[e.code] = true;
initAudio();
if (state === 'start' || state === 'gameover') {
if (e.code === 'Space' || e.code === 'Enter') startGame();
}
if (state === 'ballheld' && e.code === 'Space') launchBall();
if (e.code === 'KeyR' && (state === 'playing' || state === 'gameover')) startGame();
});
window.addEventListener('keyup', e => { keys[e.code] = false; });

// Particles
function spawnParticles(x, y, color, count) {
for (let i = 0; i < count; i++) {
const angle = Math.random() * Math.PI * 2;
const spd = 1 + Math.random() * 4;
particles.push({ x, y, vx: Math.cos(angle)*spd, vy: Math.sin(angle)*spd - 1, life: 30 + Math.random()*20, color, size: 2+Math.random()*3 });
}
}

function updateParticles() {
for (let i = particles.length - 1; i >= 0; i--) {
const p = particles[i];
p.x += p.vx; p.y += p.vy; p.vy += 0.15; p.life--;
if (p.life <= 0) particles.splice(i, 1);
}
}

function spawnSpeedLines() {
for (let i = 0; i < 3; i++) {
speedLines.push({ x: Math.random() * W, y: -20, len: 20 + Math.random()*40, speed: 8+Math.random()*6 });
}
}

function updateSpeedLines() {
for (let i = speedLines.length-1; i>=0; i--) {
speedLines[i].y += speedLines[i].speed;
if (speedLines[i].y > H) speedLines.splice(i, 1);
}
}

// Drawing
function drawCar(x, y, w, h) {
ctx.save();
// Body
ctx.fillStyle = '#e63946';
ctx.beginPath();
ctx.roundRect(x + 5, y + 10, w - 10, h - 10, 8);
ctx.fill();
// Roof
ctx.fillStyle = '#b22234';
ctx.fillRect(x + w*0.25, y + 5, w*0.5, 12);
// Windshield
ctx.fillStyle = '#a8dadc';
ctx.beginPath();
ctx.moveTo(x + w*0.3, y + 12);
ctx.lineTo(x + w*0.7, y + 12);
ctx.lineTo(x + w*0.65, y + 22);
ctx.lineTo(x + w*0.35, y + 22);
ctx.closePath();
ctx.fill();
// Wheels
ctx.fillStyle = '#222';
ctx.beginPath(); ctx.arc(x + 12, y + h - 8, 8, 0, Math.PI*2); ctx.fill();
ctx.beginPath(); ctx.arc(x + w - 12, y + h - 8, 8, 0, Math.PI*2); ctx.fill();
// Wheel caps
ctx.fillStyle = '#666';
ctx.beginPath(); ctx.arc(x + 12, y + h - 8, 4, 0, Math.PI*2); ctx.fill();
ctx.beginPath(); ctx.arc(x + w - 12, y + h - 8, 4, 0, Math.PI*2); ctx.fill();
// Headlights
ctx.fillStyle = '#fff59a';
ctx.fillRect(x + 3, y + 14, 4, 6);
ctx.fillRect(x + w - 7, y + 14, 4, 6);
// Taillight stripe
ctx.fillStyle = '#ff0';
ctx.fillRect(x + 5, y + h - 5, w - 10, 3);
ctx.restore();
}

function drawBrick(b) {
const colors = brickColors[(level-1) % brickColors.length];
ctx.fillStyle = b.hp > 1 ? colors[1] : colors[0];
ctx.beginPath();
ctx.roundRect(b.x, b.y, b.w, b.h, 4);
ctx.fill();
ctx.strokeStyle = 'rgba(255,255,255,0.3)';
ctx.lineWidth = 1;
ctx.stroke();
// Shine
ctx.fillStyle = 'rgba(255,255,255,0.15)';
ctx.fillRect(b.x + 2, b.y + 2, b.w - 4, 6);
}

function drawBall() {
ctx.beginPath();
ctx.arc(ball.x, ball.y, ball.r, 0, Math.PI * 2);
const grad = ctx.createRadialGradient(ball.x-2, ball.y-2, 1, ball.x, ball.y, ball.r);
grad.addColorStop(0, '#fff');
grad.addColorStop(0.4, '#ffcc00');
grad.addColorStop(1, '#ff6600');
ctx.fillStyle = grad;
ctx.fill();
// Glow
ctx.shadowBlur = 12;
ctx.shadowColor = '#ff8800';
ctx.beginPath();
ctx.arc(ball.x, ball.y, ball.r - 2, 0, Math.PI*2);
ctx.stroke();
ctx.shadowBlur = 0;
}

function drawRoad() {
// Background gradient
const bg = ctx.createLinearGradient(0, 0, 0, H);
bg.addColorStop(0, '#1a1a2e');
bg.addColorStop(1, '#16213e');
ctx.fillStyle = bg;
ctx.fillRect(0, 0, W, H);

// Road at bottom
const roadY = H - 80;
ctx.fillStyle = '#333';
ctx.fillRect(0, roadY, W, 80);
// Road lines
ctx.strokeStyle = '#ffcc00';
ctx.lineWidth = 2;
ctx.setLineDash([20, 15]);
ctx.beginPath();
ctx.moveTo(0, roadY + 40);
ctx.lineTo(W, roadY + 40);
ctx.stroke();
ctx.setLineDash([]);

// Speed lines
ctx.strokeStyle = 'rgba(255,255,255,0.1)';
ctx.lineWidth = 1;
for (const sl of speedLines) {
ctx.beginPath();
ctx.moveTo(sl.x, sl.y);
ctx.lineTo(sl.x, sl.y + sl.len);
ctx.stroke();
}
}

function drawHUD() {
ctx.fillStyle = 'rgba(0,0,0,0.6)';
ctx.fillRect(0, 0, W, 50);
ctx.font = 'bold 18px Segoe UI';
ctx.fillStyle = '#fff';
ctx.textAlign = 'left';
ctx.fillText(`SCORE: ${score}`, 20, 32);
ctx.textAlign = 'center';
ctx.fillText(`LVL ${level}`, W/2, 32);
ctx.textAlign = 'right';
// Lives as mini cars
let lifeX = W - 40;
for (let i = 0; i < lives; i++) {
ctx.fillStyle = '#e63946';
ctx.fillRect(lifeX - i * 25, 18, 18, 12);
ctx.fillStyle = '#a8dadc';
ctx.fillRect(lifeX - i*25 + 4, 20, 10, 5);
}
}

function drawStartScreen() {
drawRoad();
updateSpeedLines();
spawnSpeedLines();

ctx.fillStyle = 'rgba(0,0,0,0.7)';
ctx.fillRect(0, 0, W, H);

ctx.textAlign = 'center';
ctx.font = 'bold 52px Segoe UI';
ctx.fillStyle = '#ff4444';
ctx.fillText('RACING BREAKOUT', W/2, H/3);

ctx.font = '24px Segoe UI';
ctx.fillStyle = '#fff';
ctx.fillText('Drive your car. Smash the bricks.', W/2, H/3 + 50);

ctx.font = '18px Segoe UI';
ctx.fillStyle = '#aaa';
ctx.fillText('← → or A/D : Move Car', W/2, H/2 + 20);
ctx.fillText('SPACE : Launch Ball', W/2, H/2 + 50);
ctx.fillText('R : Restart', W/2, H/2 + 80);

ctx.font = 'bold 20px Segoe UI';
ctx.fillStyle = '#ffcc00';
const blink = Math.sin(Date.now() * 0.005) > 0;
if (blink) ctx.fillText('PRESS SPACE TO START', W/2, H * 0.75);

// Draw a sample car
drawCar(W/2 - 40, H * 0.8, 80, 40);
}

function drawGameOver() {
ctx.fillStyle = 'rgba(0,0,0,0.8)';
ctx.fillRect(0, 0, W, H);
ctx.textAlign = 'center';
ctx.font = 'bold 48px Segoe UI';
ctx.fillStyle = '#ff4444';
ctx.fillText('GAME OVER', W/2, H/3);
ctx.font = '24px Segoe UI';
ctx.fillStyle = '#fff';
ctx.fillText(`Final Score: ${score}`, W/2, H/3 + 50);
ctx.font = '18px Segoe UI';
ctx.fillStyle = '#aaa';
const blink = Math.sin(Date.now() * 0.005) > 0;
if (blink) ctx.fillText('PRESS SPACE TO RESTART', W/2, H * 0.6);
}

function drawWinScreen() {
ctx.fillStyle = 'rgba(0,0,0,0.8)';
ctx.fillRect(0, 0, W, H);
ctx.textAlign = 'center';
ctx.font = 'bold 48px Segoe UI';
ctx.fillStyle = '#44ff44';
ctx.fillText('YOU WIN!', W/2, H/3);
ctx.font = '24px Segoe UI';
ctx.fillStyle = '#fff';
ctx.fillText(`Score: ${score}`, W/2, H/3 + 50);
}

// Update
function update() {
if (state === 'ballheld') {
car.y = H - 60;
ball.x = car.x + car.w / 2;
ball.y = car.y - ball.r - 5;
updateCar();
updateParticles();
return;
}

if (state !== 'playing') return;

updateCar();

// Ball movement
ball.x += ball.vx;
ball.y += ball.vy;

// Wall collisions
if (ball.x - ball.r < 0) { ball.x = ball.r; ball.vx = Math.abs(ball.vx); playSound(200, 0.05, 'square', 0.1); }
if (ball.x + ball.r > W) { ball.x = W - ball.r; ball.vx = -Math.abs(ball.vx); playSound(200, 0.05, 'square', 0.1); }
if (ball.y - ball.r < 0) { ball.y = ball.r; ball.vy = Math.abs(ball.vy); playSound(200, 0.05, 'square', 0.1); }

// Car collision
if (ball.vy > 0 && ball.y + ball.r >= car.y && ball.y + ball.r <= car.y + car.h && ball.x >= car.x && ball.x <= car.x + car.w) {
const hitPos = (ball.x - car.x) / car.w - 0.5;
const angle = hitPos * 120 * Math.PI / 180;
ball.vy = -Math.cos(angle) * ball.speed;
ball.vx = Math.sin(angle) * ball.speed;
ball.y = car.y - ball.r;
playSound(600, 0.08, 'sine', 0.2);
}

// Brick collisions
for (let i = 0; i < bricks.length; i++) {
const b = bricks[i];
if (!b.alive) continue;
if (ball.x + ball.r > b.x && ball.x - ball.r < b.x + b.w && ball.y + ball.r > b.y && ball.y - ball.r < b.y + b.h) {
b.hp--;
if (b.hp <= 0) {
b.alive = false;
score += 10 * level;
spawnParticles(ball.x, ball.y, brickColors[(level-1)%brickColors.length][0], 8);
playSound(800, 0.1, 'square', 0.2);
} else {
playSound(500, 0.05, 'triangle', 0.15);
}
// Bounce direction
const overlapX = Math.min(ball.x + ball.r - b.x, b.x + b.w - (ball.x - ball.r));
const overlapY = Math.min(ball.y + ball.r - b.y, b.y + b.h - (ball.y - ball.r));
if (overlapX < overlapY) {
ball.vx = -ball.vx;
} else {
ball.vy = -ball.vy;
}
break;
}
}

// Ball lost
if (ball.y - ball.r > H) {
lives--;
playSound(150, 0.3, 'sawtooth', 0.2);
spawnParticles(ball.x, H, '#ff4444', 15);
if (lives <= 0) {
state = 'gameover';
} else {
ball.speed += 0.2;
resetBall();
}
}

// Check level clear
let aliveBricks = bricks.filter(b => b.alive).length;
if (aliveBricks === 0) {
if (level >= 5) {
state = 'win';
} else {
nextLevel();
}
}

updateParticles();
spawnSpeedLines();
updateSpeedLines();
}

function updateCar() {
let moveDir = 0;
if (keys['ArrowLeft'] || keys['KeyA']) moveDir = -1;
if (keys['ArrowRight'] || keys['KeyD']) moveDir = 1;
car.x += car.speed * moveDir;
car.x = Math.max(0, Math.min(W - car.w, car.x));
}

// Render
function render() {
ctx.clearRect(0, 0, W, H);

if (state === 'start') {
drawStartScreen();
return;
}

drawRoad();

// Draw bricks
for (const b of bricks) {
if (b.alive) drawBrick(b);
}

// Draw car
drawCar(car.x, car.y, car.w, car.h);

// Draw ball
if (state === 'playing' || state === 'ballheld') drawBall();

// Particles
for (const p of particles) {
ctx.globalAlpha = p.life / 50;
ctx.fillStyle = p.color;
ctx.fillRect(p.x, p.y, p.size, p.size);
}
ctx.globalAlpha = 1;

drawHUD();

if (state === 'gameover') drawGameOver();
if (state === 'win') drawWinScreen();
}

// Main loop
function loop() {
update();
render();
requestAnimationFrame(loop);
}

car.x = W / 2 - car.w / 2;
car.y = H - 60;
loop();

})();
</script>
</body>
