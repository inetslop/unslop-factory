let audioCtx = null;
function initAudio() {
    if (!audioCtx) {

function playSound(type) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    const now = audioCtx.currentTime;
    switch (type) {
        case 'move':
            osc.type = 'sine';
            osc.frequency.setValueAtTime(220, now);
            osc.frequency.exponentialRampToValueAtTime(180, now + 0.08);
            gain.gain.setValueAtTime(0.15, now);
            gain.gain.exponentialRampToValueAtTime(0.001, now + 0.08);
            osc.start(now);
            osc.stop(now + 0.08);
            break;
        case 'push':
            osc.type = 'square';
            osc.frequency.setValueAtTime(120, now);
            osc.frequency.exponentialRampToValueAtTime(90, now + 0.12);
            gain.gain.setValueAtTime(0.12, now);
            gain.gain.exponentialRampToValueAtTime(0.001, now + 0.12);
            osc.start(now);
            osc.stop(now + 0.12);
            break;
        case 'placed':
            osc.type = 'sine';
            osc.frequency.setValueAtTime(440, now);
            osc.frequency.exponentialRampToValueAtTime(660, now + 0.15);
            gain.gain.setValueAtTime(0.2, now);
            gain.gain.exponentialRampToValueAtTime(0.001, now + 0.2);
            osc.start(now);
            osc.stop(now + 0.2);
            break;
        case 'win':
            for (let i = 0; i < 4; i++) {
                const o = audioCtx.createOscillator();
                const g = audioCtx.createGain();
                o.connect(g);
                g.connect(audioCtx.destination);
                o.type = 'sine';
                o.frequency.setValueAtTime([523, 659, 784, 1047][i], now + i * 0.12);
                g.gain.setValueAtTime(0.15, now + i * 0.12);
                g.gain.exponentialRampToValueAtTime(0.001, now + i * 0.12 + 0.15);
                o.start(now + i * 0.12);
                o.stop(now + i * 0.12 + 0.15);
            }
            return;
        case 'complete':
            for (let i = 0; i < 6; i++) {
                const o = audioCtx.createOscillator();
                const g = audioCtx.createGain();
                o.connect(g);
                g.connect(audioCtx.destination);
                o.type = 'sine';
                o.frequency.setValueAtTime([523, 659, 784, 1047, 1319, 1568][i], now + i * 0.1);
                g.gain.setValueAtTime(0.18, now + i * 0.1);
                g.gain.exponentialRampToValueAtTime(0.001, now + i * 0.1 + 0.2);
                o.start(now + i * 0.1);
                o.stop(now + i * 0.1 + 0.25);
            }

    }

let state = 'start'; // start, playing, levelComplete, gameComplete

function parseLevel(lvl) {
    grid = [];
    boxes = [];
    targets = [];
    player = { x: 0, y: 0 };
    placedCount = 0;
    for (let r = 0; r < lvl.length; r++) {
        grid[r] = [];
        for (let c = 0; c < lvl[r].length; c++) {
            const ch = lvl[r][c];
            if (ch === '#') grid[r][c] = 1;
            else if (ch === '@') { grid[r][c] = 0; player.x = c; player.y = r; }
            else if (ch === '+') { grid[r][c] = 0; player.x = c; player.y = r; targets.push({x:c,y:r}); }
            else if (ch === '$') { grid[r][c] = 0; boxes.push({x:c,y:r}); }
            else if (ch === '*') { grid[r][c] = 0; boxes.push({x:c,y:r,onTarget:true}); }
            else if (ch === '%') { grid[r][c] = 0; targets.push({x:c,y:r}); }
            else { grid[r][c] = 0; }
        }
    }
    // Update box on-target status
    for (let b of boxes) {
        b.onTarget = targets.some(t => t.x === b.x && t.y === b.y);
    }
    placedCount = boxes.filter(b => b.onTarget).length;
}

function loadLevel(idx) {
    parseLevel(levels[idx]);
    camX = 0;
    targetCamX = 0;
    moveAnim = 0;
    particles = [];
}

function isWall(x, y) {
    if (y < 0 || y >= grid.length) return true;
    if (x < 0 || x >= grid[y].length) return true;
    return grid[y][x] === 1;
}

function hasBox(x, y) {
    return boxes.some(b => b.x === x && b.y === y);
}

function tryMove(dx, dy) {
    if (state !== 'playing') return;
    const nx = player.x + dx;
    const ny = player.y + dy;
    
    if (isWall(nx, ny)) return;
    
    if (hasBox(nx, ny)) {
        const bx = nx + dx;
        const by = ny + dy;
        if (isWall(bx, by) || hasBox(bx, by)) return;
        // Push box
        const box = boxes.find(b => b.x === nx && b.y === ny);
        const wasOnTarget = box.onTarget;
        box.x = bx;
        box.y = by;
        box.onTarget = targets.some(t => t.x === bx && t.y === by);
        if (!wasOnTarget && box.onTarget) {
            placedCount++;
            playSound('placed');
            spawnParticles(bx, by, '#4ecdc4');
        } else if (wasOnTarget && !box.onTarget) {
            placedCount--;
            playSound('push');
        } else {
            playSound('push');
        }
    } else {
        playSound('move');
    }
    
    player.x = nx;
    player.y = ny;
    moveAnim = 8;
    
    // Check win
    if (boxes.every(b => b.onTarget)) {
        state = 'levelComplete';
        playSound('win');
        setTimeout(() => {
            if (currentLevel >= levels.length - 1) {
                state = 'gameComplete';
                playSound('complete');
            }
        }, 800);
    }
}

function spawnParticles(gx, gy, color) {
    for (let i = 0; i < 12; i++) {
        const angle = Math.random() * Math.PI * 2;
        const speed = 1 + Math.random() * 3;
        particles.push({
            x: gx * TILE + TILE / 2,
            y: gy * TILE + TILE / 2,
            vx: Math.cos(angle) * speed,
            vy: Math.sin(angle) * speed - 2,
            life: 1,
            color: color
        });
    }
}

window.addEventListener('keydown', (e) => {

    if (state === 'levelComplete') {
        currentLevel++;

    if (state === 'gameComplete') {
        currentLevel = 0;

    }

        loadLevel(currentLevel);
        state = 'playing';
        return;
    }
    
    let dx = 0, dy = 0;
    switch (e.key) {
        case 'ArrowLeft': case 'a': case 'A': dx = -1; break;
        case 'ArrowRight': case 'd': case 'D': dx = 1; break;
        case 'ArrowUp': case 'w': case 'W': dy = -1; break;
        case 'ArrowDown': case 's': case 'S': dy = 1; break;
        default: return;
    }
    e.preventDefault();
    tryMove(dx, dy);
});

function drawBackground() {

function drawTile(gx, gy, camOffsetX) {
    const sx = gx * TILE - camOffsetX;
    const sy = gy * TILE;
    
    if (sx < -TILE || sx > W + TILE) return;
    
    const ch = grid[gy][gx];
    if (ch === 1) {
        // Wall
        ctx.fillStyle = '#3a3a5c';
        ctx.fillRect(sx, sy, TILE, TILE);
        ctx.fillStyle = '#4a4a6e';
        ctx.fillRect(sx + 2, sy + 2, TILE - 4, TILE - 4);
        // Brick pattern
        ctx.strokeStyle = '#2a2a4c';
        ctx.lineWidth = 1;
        ctx.strokeRect(sx + 2, sy + 2, TILE / 2 - 2, TILE / 2 - 2);
        ctx.strokeRect(sx + TILE / 2, sy + 2, TILE / 2 - 2, TILE / 2 - 2);
        ctx.strokeRect(sx + 2, sy + TILE / 2, TILE / 2 - 2, TILE / 2 - 2);
        ctx.strokeRect(sx + TILE / 2, sy + TILE / 2, TILE / 2 - 2, TILE / 2 - 2);
    } else {
        // Floor
        ctx.fillStyle = (gx + gy) % 2 === 0 ? '#2a2a45' : '#2e2e4a';
        ctx.fillRect(sx, sy, TILE, TILE);
    }
}

function drawTarget(gx, gy, camOffsetX) {
    const sx = gx * TILE - camOffsetX;
    const sy = gy * TILE;
    if (sx < -TILE || sx > W + TILE) return;
    
    const pulse = Math.sin(animFrame * 0.08) * 0.3 + 0.7;
    ctx.fillStyle = `rgba(78, 205, 196, ${pulse * 0.4})`;
    ctx.beginPath();
    ctx.arc(sx + TILE / 2, sy + TILE / 2, TILE * 0.35, 0, Math.PI * 2);
    ctx.fill();
    
    ctx.strokeStyle = `rgba(78, 205, 196, ${pulse})`;
    ctx.lineWidth = 2;
    ctx.beginPath();
    ctx.arc(sx + TILE / 2, sy + TILE / 2, TILE * 0.3, 0, Math.PI * 2);
    ctx.stroke();
}

function drawBox(gx, gy, camOffsetX) {
    const sx = gx * TILE - camOffsetX;
    const sy = gy * TILE;
    if (sx < -TILE || sx > W + TILE) return;
    
    const box = boxes.find(b => b.x === gx && b.y === gy);
    if (!box) return;
    
    // Wooden crate
    ctx.fillStyle = box.onTarget ? '#4ecdc4' : '#c8956c';
    ctx.fillRect(sx + 3, sy + 3, TILE - 6, TILE - 6);
    ctx.strokeStyle = box.onTarget ? '#2a9d94' : '#8b6914';
    ctx.lineWidth = 2;
    ctx.strokeRect(sx + 3, sy + 3, TILE - 6, TILE - 6);
    
    // Cross pattern
    ctx.beginPath();
    ctx.moveTo(sx + 3, sy + 3);
    ctx.lineTo(sx + TILE - 3, sy + TILE - 3);
    ctx.moveTo(sx + TILE - 3, sy + 3);
    ctx.lineTo(sx + 3, sy + TILE - 3);
    ctx.stroke();
    
    if (box.onTarget) {
        const glow = Math.sin(animFrame * 0.1) * 0.2 + 0.3;
        ctx.fillStyle = `rgba(78, 205, 196, ${glow})`;
        ctx.fillRect(sx, sy, TILE, TILE);
    }
}

function update() {
    animFrame++;
    
    if (moveAnim > 0) moveAnim--;
    
    // Update particles
    for (let i = particles.length - 1; i >= 0; i--) {
        const p = particles[i];
        p.x += p.vx;
        p.y += p.vy;
        p.vy += 0.3;
        p.life -= 0.025;
        if (p.life <= 0) particles.splice(i, 1);
    }
    
    // Camera follow
    const levelWidth = grid.length > 0 ? grid[0].length * TILE : W;
    targetCamX = player.x * TILE - W / 2 + TILE / 2;
    targetCamX = Math.max(0, Math.min(targetCamX, levelWidth - W));
    camX += (targetCamX - camX) * 0.1;
}

function render() {
    if (state === 'start') {
        drawStartScreen();
        return;
    }
    
    if (state === 'gameComplete') {
        drawGameComplete();
        return;
    }
    
    drawBackground();
    
    // Draw tiles
    const levelCols = grid[0] ? grid[0].length : 30;
    const startCol = Math.max(0, Math.floor(camX / TILE) - 1);
    const endCol = Math.min(levelCols, Math.ceil((camX + W) / TILE) + 1);
    
    for (let gy = 0; gy < grid.length; gy++) {
        for (let gx = startCol; gx < endCol; gx++) {
            drawTile(gx, gy, camX);
        }
    }
    
    // Draw targets
    for (let t of targets) {
        drawTarget(t.x, t.y, camX);
    }
    
    // Draw boxes
    for (let b of boxes) {
        drawBox(b.x, b.y, camX);
    }
    
    // Draw player
    drawPlayer(camX);
    
    // Particles
    drawParticles(camX);
    
    // HUD
    drawHUD();
    
    if (state === 'levelComplete') {
        drawLevelComplete();
    }
}

function gameLoop() {
    update();
    render();
    requestAnimationFrame(gameLoop);
}

gameLoop();
