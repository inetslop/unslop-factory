function playSound(freq, type, duration, volume) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = type;
    osc.frequency.value = freq;
    gain.gain.setValueAtTime(volume || 0.15, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + duration);
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + duration);
}

function flipGravity() {
    if (player.flipCooldown > 0) return;

function isPlayerInShadow() {
    for (let s of currentShadows) {
        if (player.x + player.w > s.x && player.x < s.x + s.w &&
            player.y + player.h > s.y && player.y < s.y + s.h) {
            return true;
        }
    }
    return false;
}

function guardCanSee(guard) {
    if (player.inShadow) return false;
    const gx = guard.x + guard.w / 2;
    const gy = guard.y + guard.h / 2;
    const px = player.x + player.w / 2;
    const py = player.y + player.h / 2;
    const dx = px - gx;
    const dy = py - gy;
    const dist = Math.hypot(dx, dy);
    if (dist > 180) return false;
    // Check if in direction guard faces
    const facingX = guard.dir;
    if (facingX * dx < -20) return false;
    // Angle check
    const angle = Math.atan2(dy, dx);
    const facingAngle = facingX > 0 ? 0 : Math.PI;
    let diff = Math.abs(angle - facingAngle);
    if (diff > Math.PI) diff = 2 * Math.PI - diff;
    return diff < 1.2;
}

function update() {
    time++;
    if (gameState !== 'playing') {
        if (gameState === 'dead') {

                } else {

                    gameState = 'playing';

        return;

    // Player movement

    if (keys['ArrowLeft'] || keys['KeyA']) player.vx = -moveSpeed;
    else if (keys['ArrowRight'] || keys['KeyD']) player.vx = moveSpeed;
    else player.vx *= 0.7;

    if ((keys['Space'] || keys['KeyW'] || keys['ArrowUp']) && player.flipCooldown <= 0) {
        flipGravity();
    }

    if (player.flipCooldown > 0) player.flipCooldown--;

    // Gravity
    const gravity = 0.6 * player.gravityDir;
    player.vy += gravity;
    player.vy = Math.max(-12, Math.min(12, player.vy));

    player.x += player.vx;
    player.y += player.vy;

    // Platform collision
    player.onGround = false;
    for (let p of currentPlatforms) {
        if (player.gravityDir === 1) {
            // Normal gravity - land on top
            if (player.x + player.w > p.x && player.x < p.x + p.w) {
                if (player.vy > 0 && player.y + player.h > p.y && player.y + player.h - player.vy <= p.y + 5) {
                    player.y = p.y - player.h;
                    player.vy = 0;
                    player.onGround = true;
                }
            }
        } else {
            // Inverted gravity - land on bottom (ceiling)
            if (player.x + player.w > p.x && player.x < p.x + p.w) {
                if (player.vy < 0 && player.y < p.y + p.h && player.y - player.vy >= p.y + p.h - 5) {
                    player.y = p.y + p.h;
                    player.vy = 0;
                    player.onGround = true;
                }
            }
        }
    }

    // World bounds
    if (player.x < 0) player.x = 0;
    if (player.x > 1000 - player.w) player.x = 1000 - player.w;
    if (player.y < -50 || player.y > 550) {
        killPlayer();
        return;
    }

    // Shadow check
    player.inShadow = isPlayerInShadow();

    // Update guards
    for (let g of guards) {
        if (g.alertTimer > 0) g.alertTimer--;
        const effectiveSpeed = g.speed + (g.alertTimer > 0 ? 1 : 0);
        g.x += g.dir * effectiveSpeed;
        if (g.x < 0) { g.x = 0; g.dir = 1; }
        if (g.x > 976) { g.x = 976; g.dir = -1; }

        // Patrol range from baseY position (horizontal patrol)
        const patrolCenter = levels[currentLevel].guards[guards.indexOf(g)] ? levels[currentLevel].guards[guards.indexOf(g)].x : 0;
        // Simpler: just bounce off edges of screen for now

        if (guardCanSee(g)) {
            g.alertTimer = 90;
            player.detected += 2;
            if (player.detected >= 100) {
                killPlayer();
                return;
            }
        } else {
            player.detected = Math.max(0, player.detected - 0.5);
        }
    }

    // Update noise pulses
    for (let i = noisePulses.length - 1; i >= 0; i--) {
        noisePulses[i].radius += 3;
        noisePulses[i].alpha -= 0.015;
        if (noisePulses[i].alpha <= 0 || noisePulses[i].radius > noisePulses[i].maxRadius) {
            noisePulses.splice(i, 1);
        }
    }

    // Update particles
    for (let i = particles.length - 1; i >= 0; i--) {
        const p = particles[i];
        p.x += p.vx;
        p.y += p.vy;
        p.life--;
        if (p.life <= 0) particles.splice(i, 1);
    }

    // Exit check
    const ex = currentExit.x;
    const ey = currentExit.y;
    if (player.x + player.w > ex && player.x < ex + 40 &&
        player.y + player.h > ey && player.y < ey + 40) {
        score += 100 - Math.floor(player.detected);
        loadLevel(currentLevel);
        nextLevel();
    }
}

    for (let p of particles) {
        const alpha = p.life / p.maxLife;
        ctx.fillStyle = p.color.replace(')', `,${alpha})`).replace('#', 'rgba(').replace(/rgba\(([^,]+),([^,]+),([^,]+)\),?/, '');
        // Simpler:
        ctx.globalAlpha = alpha;
        ctx.fillStyle = p.color;
        ctx.fillRect(p.x - 2, p.y - 2, 4, 4);
        ctx.globalAlpha = 1;
    }

    ctx.restore();

    // HUD
    if (gameState === 'playing') {
        drawHUD();
    }
    if (gameState === 'dead') {
        ctx.fillStyle = `rgba(255,0,0,${deathTimer / 60 * 0.3})`;
        ctx.fillRect(0, 0, canvas.width, canvas.height);
        ctx.fillStyle = '#fff';
        ctx.font = 'bold 24px monospace';
        ctx.textAlign = 'center';
        ctx.fillText('DETECTED!', canvas.width / 2, canvas.height / 2);
        if (lives <= 0) {
            ctx.font = '18px monospace';
            ctx.fillText('Game Over - Press R to Restart', canvas.width / 2, canvas.height / 2 + 40);
        } else {
            ctx.font = '16px monospace';
            ctx.fillText(`Lives: ${lives}`, canvas.width / 2, canvas.height / 2 + 40);
        }
        ctx.textAlign = 'left';
    }
    if (gameState === 'win') {
        ctx.fillStyle = '#0a0a12';
        ctx.fillRect(0, 0, canvas.width, canvas.height);
        ctx.fillStyle = '#5f5';
        ctx.font = 'bold 36px monospace';
        ctx.textAlign = 'center';
        if (lives <= 0) {
            ctx.fillText('GAME OVER', canvas.width / 2, canvas.height / 2 - 40);
            ctx.fillStyle = '#fff';
            ctx.font = '18px monospace';
            ctx.fillText(`Score: ${score}`, canvas.width / 2, canvas.height / 2 + 10);
        } else {
            ctx.fillText('MISSION COMPLETE!', canvas.width / 2, canvas.height / 2 - 40);
            ctx.fillStyle = '#fff';
            ctx.font = '18px monospace';
            ctx.fillText(`Final Score: ${score}`, canvas.width / 2, canvas.height / 2 + 10);
        }
        ctx.fillStyle = '#aaa';
        ctx.font = '16px monospace';
        ctx.fillText('Press R to Play Again', canvas.width / 2, canvas.height / 2 + 50);
        ctx.textAlign = 'left';
    }
}

function drawHUD() {
    ctx.fillStyle = 'rgba(0,0,0,0.6)';
    ctx.fillRect(10, 10, 200, 80);
    ctx.strokeStyle = '#4a4a6a';
    ctx.strokeRect(10, 10, 200, 80);

    ctx.fillStyle = '#fff';
    ctx.font = '14px monospace';
    ctx.fillText(`Level: ${currentLevel + 1}/${levels.length}`, 20, 35);
    ctx.fillText(`Score: ${score}`, 20, 55);
    ctx.fillText(`Lives: ${'❤'.repeat(Math.max(0, lives)) || '—'}`, 20, 75);

    // Gravity indicator
    const gDir = player.gravityDir === 1 ? '↓ NORMAL' : '↑ INVERTED';
    ctx.fillStyle = player.gravityDir === 1 ? '#4af' : '#f4a';
    ctx.fillText(`Gravity: ${gDir}`, 120, 35);

    // Stealth status
    ctx.fillStyle = player.inShadow ? '#8f8' : '#f88';
    ctx.fillText(player.inShadow ? 'Hidden' : 'Exposed', 120, 75);
}
