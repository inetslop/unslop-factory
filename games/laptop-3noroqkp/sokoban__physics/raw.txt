// Physics
const PLAYER_ACCEL = 0.8;
const PLAYER_FRICTION = 0.93;
const BOX_FRICTION = 0.965;
const RESTITUTION = 0.1;
const MAX_VEL = 12;
const SUBSTEPS = 4;

function physicsStep() {
  // Player input
  let ax = 0, ay = 0;
  if (keys['ArrowLeft'] || keys['KeyA']) ax -= PLAYER_ACCEL;
  if (keys['ArrowRight'] || keys['KeyD']) ax += PLAYER_ACCEL;
  if (keys['ArrowUp'] || keys['KeyW']) ay -= PLAYER_ACCEL;
  if (keys['ArrowDown'] || keys['KeyS']) ay += PLAYER_ACCEL;

  player.vx += ax;
  player.vy += ay;
  player.vx *= PLAYER_FRICTION;
  player.vy *= PLAYER_FRICTION;

  // Clamp velocity
  const pSpeed = Math.sqrt(player.vx * player.vx + player.vy * player.vy);
  if (pSpeed > MAX_VEL) {
    player.vx = player.vx / pSpeed * MAX_VEL;
    player.vy = player.vy / pSpeed * MAX_VEL;
  }

  // Integrate player
  player.x += player.vx;
  player.y += player.vy;

  // Box friction and integration
  for (let b of boxes) {
    b.vx *= BOX_FRICTION;
    b.vy *= BOX_FRICTION;
    if (Math.abs(b.vx) < 0.01) b.vx = 0;
    if (Math.abs(b.vy) < 0.01) b.vy = 0;
    b.x += b.vx;
    b.y += b.vy;
  }

  // Collisions
  resolveCollisions();
}

function circleRectCollision(cx, cy, cr, rx, ry, rw, rh) {
  const closestX = Math.max(rx, Math.min(cx, rx + rw));
  const closestY = Math.max(ry, Math.min(cy, ry + rh));
  const dx = cx - closestX;
  const dy = cy - closestY;
  return (dx * dx + dy * dy) < cr * cr;
}

function resolveCollisions() {
  for (let iter = 0; iter < 6; iter++) {
    // Player vs Walls
    resolveCircleRect(player, walls);
    // Boxes vs Walls
    for (let b of boxes) resolveCircleRect(b, walls);
    // Player vs Boxes
    for (let b of boxes) resolveCircleCircle(player, b);
    // Box vs Box
    for (let i = 0; i < boxes.length; i++) {
      for (let j = i + 1; j < boxes.length; j++) {
        resolveCircleCircle(boxes[i], boxes[j]);
      }
    }
  }
}

function resolveCircleRect(obj, wallArr) {
  for (let w of wallArr) {
    const closestX = Math.max(w.x, Math.min(obj.x, w.x + w.w));
    const closestY = Math.max(w.y, Math.min(obj.y, w.y + w.h));
    let dx = obj.x - closestX;
    let dy = obj.y - closestY;
    const dist = Math.sqrt(dx * dx + dy * dy);
    if (dist < obj.r) {
      const overlap = obj.r - dist;
      if (dist > 0.001) {
        const nx = dx / dist;
        const ny = dy / dist;
        obj.x += nx * overlap;
        obj.y += ny * overlap;
        // Reflect velocity
        const dot = obj.vx * (-nx) + obj.vy * (-ny);
        if (dot > 0) {
          obj.vx -= (1 + RESTITUTION) * dot * (-nx);
          obj.vy -= (1 + RESTITUTION) * dot * (-ny);
        }
      } else {
        // Center inside rect - push out nearest face
        const left = obj.x - w.x;
        const right = w.x + w.w - obj.x;
        const top = obj.y - w.y;
        const bottom = w.y + w.h - obj.y;
        const min = Math.min(left, right, top, bottom);
        if (min === left) { obj.x -= left + obj.r; obj.vx = -Math.abs(obj.vx) * RESTITUTION; }
        else if (min === right) { obj.x += right + obj.r; obj.vx = Math.abs(obj.vx) * RESTITUTION; }
        else if (min === top) { obj.y -= top + obj.r; obj.vy = -Math.abs(obj.vy) * RESTITUTION; }
        else { obj.y += bottom + obj.r; obj.vy = Math.abs(obj.vy) * RESTITUTION; }
      }
    }
  }
}

function resolveCircleCircle(a, b) {
  const dx = b.x - a.x;
  const dy = b.y - a.y;
  const dist = Math.sqrt(dx * dx + dy * dy);
  const minDist = a.r + b.r;
  if (dist < minDist && dist > 0.001) {
    const nx = dx / dist;
    const ny = dy / dist;
    const overlap = minDist - dist;

    // Mass-weighted separation
    const totalMass = a.mass + b.mass;
    a.x -= nx * overlap * (b.mass / totalMass);
    a.y -= ny * overlap * (b.mass / totalMass);
    b.x += nx * overlap * (a.mass / totalMass);
    b.y += ny * overlap * (a.mass / totalMass);

    // Relative velocity
    const rvx = a.vx - b.vx;
    const rvy = a.vy - b.vy;
    const velAlongNormal = rvx * nx + rvy * ny;

    if (velAlongNormal < 0) {
      const e = RESTITUTION;
      const impulse = -(1 + e) * velAlongNormal / totalMass;
      a.vx += impulse * b.mass * nx;
      a.vy += impulse * b.mass * ny;
      b.vx -= impulse * a.mass * nx;
      b.vy -= impulse * a.mass * ny;

      // Sound on push
      if (a === player && Math.abs(velAlongNormal) > 1.5) {
        playPush();
      }
    }
  }
}

function checkTargets() {
  for (let i = 0; i < boxes.length; i++) {
    const b = boxes[i];
    const t = targets[i];
    const dx = b.x - t.x;
    const dy = b.y - t.y;
    const dist = Math.sqrt(dx * dx + dy * dy);
    const speed = Math.sqrt(b.vx * b.vx + b.vy * b.vy);

    if (dist < t.r * 0.7 && speed < 0.5) {
      if (!b.onTarget) {
        b.onTarget = true;
        playTarget();
      }
    } else {
      b.onTarget = false;
    }
  }

  // Win check
  const allOnTarget = boxes.every(b => b.onTarget);
  if (allOnTarget) {
    winTimer++;
    if (winTimer > 30) {
      playWin();
      gameState = 'levelcomplete';
    }
  } else {
    winTimer = 0;
  }
}

  ctx.fillStyle = '#fff';
  ctx.font = `${Math.min(W * 0.025, 20)}px monospace`;
  ctx.fillText(`${levels[currentLevel].name} cleared!`, W / 2, H * 0.45);

  ctx.fillStyle = `rgba(255,255,255,${0.5 + Math.sin(frames * 0.05) * 0.3})`;

  if (currentLevel < levels.length - 1) {
    ctx.fillText('Press SPACE for next level', W / 2, H * 0.6);
  } else {
    ctx.fillText('Final level complete!', W / 2, H * 0.6);
  }
}

  ctx.fillStyle = 'rgba(0,0,0,0.7)';
  ctx.fillRect(0, 0, W, H);

  ctx.textAlign = 'center';
  ctx.fillStyle = '#ff9900';
  ctx.font = `bold ${Math.min(W * 0.05, 48)}px monospace`;
  ctx.fillText('YOU BEAT THE GAME!', W / 2, H * 0.3);

  ctx.fillStyle = '#fff';
  ctx.font = `${Math.min(W * 0.025, 20)}px monospace`;
  ctx.fillText('All levels complete. Incredible physics!?', W / 2, H * 0.42);

  // Confetti-like particles
  for (let i = 0; i < 30; i++) {
    const px = ((i * 137 + frames * 2) % W);

    ctx.fillStyle = `hsl(${(i * 47 + frames) % 360}, 80%, 60%)`;
    ctx.fillRect(px, py, 6, 6);
  }

  ctx.fillStyle = `rgba(255,255,255,${0.5 + Math.sin(frames * 0.05) * 0.3})`;
  ctx.font = `${Math.min(W * 0.025, 20)}px monospace`;

// Game Loop
function update() {
  if (gameState === 'playing') {
    for (let i = 0; i < SUBSTEPS; i++) {
      physicsStep();
    }
    checkTargets();
  }
  frames++;
  draw();
  requestAnimationFrame(update);
}
