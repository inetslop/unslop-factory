function initAudio() {

function playMatchSound(comboLevel) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.type = 'sine';
    const baseFreq = 400 + comboLevel * 100;
    osc.frequency.setValueAtTime(baseFreq, audioCtx.currentTime);
    osc.frequency.exponentialRampToValueAtTime(baseFreq * 2, audioCtx.currentTime + 0.1);
    gain.gain.setValueAtTime(0.3, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.2);
    osc.start(audioCtx.currentTime);
    osc.stop(audioCtx.currentTime + 0.2);
}

function playSwapSound() {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.type = 'triangle';
    osc.frequency.setValueAtTime(300, audioCtx.currentTime);
    osc.frequency.exponentialRampToValueAtTime(500, audioCtx.currentTime + 0.08);
    gain.gain.setValueAtTime(0.2, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.1);
    osc.start(audioCtx.currentTime);
    osc.stop(audioCtx.currentTime + 0.1);
}

function playInvalidSound() {

function initGrid() {
    grid = [];
    for (let y = 0; y < GRID_SIZE; y++) {
        grid[y] = [];
        for (let x = 0; x < GRID_SIZE; x++) {
            grid[y][x] = Math.floor(Math.random() * NUM_TYPES);
            // Prevent initial matches
            while (hasMatchAt(x, y)) {
                grid[y][x] = Math.floor(Math.random() * NUM_TYPES);
            }
        }
    }
}

function hasMatchAt(x, y) {
    const v = grid[y][x];
    // Check left
    if (x >= 2 && grid[y][x-1] === v && grid[y][x-2] === v) return true;
    // Check up
    if (y >= 2 && grid[y-1][x] === v && grid[y-2][x] === v) return true;
    return false;
}

function startGame() {
    initGrid();
    score = 0;
    moves = 30;
    targetScore = 5000 + (level - 1) * 2000;
    combo = 0;
    particles = [];
    fallingPieces = [];
    swapAnim = null;
    popups = [];
    selected = null;
    cursor = {x: 3, y: 3};
    state = 'playing';
}

function findMatches() {
    let matches = [];
    // Horizontal
    for (let y = 0; y < GRID_SIZE; y++) {
        for (let x = 0; x < GRID_SIZE - 2; x++) {
            if (grid[y][x] !== null && grid[y][x] === grid[y][x+1] && grid[y][x] === grid[y][x+2]) {
                let set = new Set([y * GRID_SIZE + x, y * GRID_SIZE + x + 1, y * GRID_SIZE + x + 2]);
                matches.push(...set);
            }
        }
    }
    // Vertical
    for (let x = 0; x < GRID_SIZE; x++) {
        for (let y = 0; y < GRID_SIZE - 2; y++) {
            if (grid[y][x] !== null && grid[y][x] === grid[y+1][x] && grid[y][x] === grid[y+2][x]) {
                let set = new Set([y * GRID_SIZE + x, (y+1) * GRID_SIZE + x, (y+2) * GRID_SIZE + x]);
                matches.push(...set);
            }
        }
    }
    return [...new Set(matches)];
}

function processMatches() {
    let matches = findMatches();
    if (matches.length > 0) {
        combo++;
        let points = matches.length * 10 * combo;
        score += points;
        playMatchSound(combo);
        
        // Add popup
        const cx = GRID_SIZE / 2;
        const cy = GRID_SIZE / 2;
        popups.push({text: '+' + points, x: cx, y: cy - combo * 0.5, life: 1});
        if (combo > 1) {
            popups.push({text: 'x' + combo + ' COMBO!', x: cx, y: cy - 1, life: 1, small: true});
        }
        
        // Spawn particles
        matches.forEach(idx => {
            const mx = idx % GRID_SIZE;
            const my = Math.floor(idx / GRID_SIZE);
            const type = grid[my][mx];
            spawnParticles(mx, my, COLORS[type].body);
            grid[my][mx] = null;
        });
        
        shakeTimer = 0.2;
        shakeIntensity = combo * 2;
        
        setTimeout(() => {
            dropPieces();
        }, 150);
    } else {
        combo = 0;
        moves--;
        if (moves <= 0) {
            state = 'gameover';
            playGameOverSound();
            try { localStorage.setItem('pacmatch_score', score); } catch(e) {}
        } else if (score >= targetScore) {
            level++;
            targetScore = 5000 + (level - 1) * 2000;
            moves = 30;
            playWinSound();
            popups.push({text: 'LEVEL UP!', x: GRID_SIZE/2, y: GRID_SIZE/2 - 2, life: 1.5, big: true});
        }
    }
}

function dropPieces() {
    // Drop existing pieces down
    for (let x = 0; x < GRID_SIZE; x++) {
        let writeRow = GRID_SIZE - 1;
        for (let y = GRID_SIZE - 1; y >= 0; y--) {
            if (grid[y][x] !== null) {
                grid[writeRow][x] = grid[y][x];
                if (writeRow !== y) grid[y][x] = null;
                writeRow--;
            }
        }
        // Fill top
        for (let y = writeRow; y >= 0; y--) {
            grid[y][x] = Math.floor(Math.random() * NUM_TYPES);
            fallingPieces.push({x, y, fromY: -1 - (writeRow - y), type: grid[y][x], progress: 0});
        }
    }
    
    // Animate falls then check matches
    setTimeout(() => {
        fallingPieces = [];
        processMatches();
    }, 250);
}

function spawnParticles(gx, gy, color) {
    const cellSize = getCellSize();
    const cx = (gx + 0.5) * cellSize;
    const cy = (gy + 0.5) * cellSize;
    for (let i = 0; i < 6; i++) {
        const angle = Math.PI * 2 * i / 6;
        particles.push({
            x: cx, y: cy,
            vx: Math.cos(angle) * (1 + Math.random()) * cellSize * 0.5,
            vy: Math.sin(angle) * (1 + Math.random()) * cellSize * 0.5,
            life: 1, color: color, size: cellSize * 0.15
        });
    }
}

function trySwap(x1, y1, x2, y2) {
    if (x2 < 0 || x2 >= GRID_SIZE || y2 < 0 || y2 >= GRID_SIZE) return false;
    const dx = Math.abs(x2 - x1);
    const dy = Math.abs(y2 - y1);
    if (dx + dy !== 1) return false;
    
    // Swap
    let temp = grid[y1][x1];
    grid[y1][x1] = grid[y2][x2];
    grid[y2][x2] = temp;
    
    // Check if valid
    const matches = findMatches();
    if (matches.length === 0) {
        // Swap back
        grid[y2][x2] = grid[y1][x1];
        grid[y1][x1] = temp;
        playInvalidSound();
        return false;
    }
    
    playSwapSound();
    moves--;
    combo = 0;
    state = 'animating';
    setTimeout(() => {
        processMatches();
        state = 'playing';
    }, 100);
    return true;
}

function getCellSize() {
    const boardSize = Math.min(W, H) * 0.85;
    return boardSize / GRID_SIZE;
}

function getBoardOffset() {
    const cellSize = getCellSize();
    const boardW = cellSize * GRID_SIZE;
    const boardH = cellSize * GRID_SIZE;
    return {x: (W - boardW) / 2, y: (H - boardH) / 2 + 10 * SCALE};
}

function drawPiece(cx, cy, size, type, alpha) {
    if (alpha === undefined) alpha = 1;
    const r = size * 0.4;
    ctx.globalAlpha = alpha;
    
    if (type === 0) {
        // Pac-Man
        ctx.fillStyle = COLORS[0].body;
        ctx.beginPath();
        ctx.arc(cx, cy, r, 0.3 * Math.PI, 1.7 * Math.PI);
        ctx.closePath();
        ctx.fill();
        // Eye
        ctx.fillStyle = '#000';
        ctx.beginPath();
        ctx.arc(cx - r*0.2, cy - r*0.2, r*0.15, 0, Math.PI*2);
        ctx.fill();
    } else {
        // Ghost shape
        const col = COLORS[type].body;
        ctx.fillStyle = col;
        ctx.beginPath();
        ctx.arc(cx, cy - r*0.2, r, Math.PI, 0);
        ctx.lineTo(cx + r, cy + r*0.8);
        // Wavy bottom
        const waveCount = 3;
        for (let i = 0; i < waveCount; i++) {
            const wx = cx + r - (i * 2 * r / waveCount);
            const wy = cy + r*0.8 - (i % 2 === 0 ? r*0.15 : 0);
            ctx.quadraticCurveTo(wx - r/waveCount, wy + r*0.3, wx - r/waveCount - r/waveCount, cy + r*0.8);
        }
        ctx.closePath();
        ctx.fill();
        
        // Eyes
        const eyeColor = type === 1 ? '#fff' : (type === 2 ? '#fff' : (type === 3 ? '#000' : '#fff'));
        ctx.fillStyle = eyeColor;
        ctx.beginPath();
        ctx.arc(cx - r*0.3, cy - r*0.1, r*0.25, 0, Math.PI*2);
        ctx.fill();
        ctx.beginPath();
        ctx.arc(cx + r*0.3, cy - r*0.1, r*0.25, 0, Math.PI*2);
        ctx.fill();
        
        // Pupils
        ctx.fillStyle = '#000';
        ctx.beginPath();
        ctx.arc(cx - r*0.25, cy - r*0.05, r*0.1, 0, Math.PI*2);
        ctx.fill();
        ctx.beginPath();
        ctx.arc(cx + r*0.35, cy - r*0.05, r*0.1, 0, Math.PI*2);
        ctx.fill();
    }
    ctx.globalAlpha = 1;
}

function drawBoard() {
    const cellSize = getCellSize();
    const off = getBoardOffset();
    
    // Board background
    ctx.fillStyle = '#0a0a2a';
    ctx.fillRect(off.x - 5, off.y - 5, cellSize * GRID_SIZE + 10, cellSize * GRID_SIZE + 10);
    
    // Grid lines (maze-style)
    ctx.strokeStyle = 'rgba(30, 60, 120, 0.4)';
    ctx.lineWidth = 1;
    for (let i = 0; i <= GRID_SIZE; i++) {
        ctx.beginPath();
        ctx.moveTo(off.x + i * cellSize, off.y);
        ctx.lineTo(off.x + i * cellSize, off.y + GRID_SIZE * cellSize);
        ctx.stroke();
        ctx.beginPath();
        ctx.moveTo(off.x, off.y + i * cellSize);
        ctx.lineTo(off.x + GRID_SIZE * cellSize, off.y + i * cellSize);
        ctx.stroke();
    }
    
    // Draw dots in cells (subtle)
    ctx.fillStyle = 'rgba(255, 200, 0, 0.1)';
    for (let y = 0; y < GRID_SIZE; y++) {
        for (let x = 0; x < GRID_SIZE; x++) {
            const cx = off.x + (x + 0.5) * cellSize;
            const cy = off.y + (y + 0.5) * cellSize;
            ctx.beginPath();
            ctx.arc(cx, cy, cellSize * 0.05, 0, Math.PI*2);
            ctx.fill();
        }
    }
    
    // Draw pieces
    for (let y = 0; y < GRID_SIZE; y++) {
        for (let x = 0; x < GRID_SIZE; x++) {
            if (grid[y] && grid[y][x] !== null) {
                const cx = off.x + (x + 0.5) * cellSize;
                const cy = off.y + (y + 0.5) * cellSize;
                let alpha = 1;
                
                // Check if this is a falling piece
                let isFalling = false;
                for (let fp of fallingPieces) {
                    if (fp.x === x && fp.y === y) {
                        isFalling = true;
                        const startCy = off.y + (fp.fromY + 0.5) * cellSize;
                        const targetCy = off.y + (y + 0.5) * cellSize;
                        const t = Math.min(1, performance.now() / 250);
                        const eased = 1 - Math.pow(1 - t, 3);
                        const drawCy = startCy + (targetCy - startCy) * eased;
                        if (y < fp.fromY || fp.fromY < 0) {
                            drawPiece(cx, off.y + (y + 0.5) * cellSize, cellSize, grid[y][x], alpha);
                        }
                    }
                }
                
                if (!isFalling) {
                    drawPiece(cx, cy, cellSize, grid[y][x]);
                }
            }
        }
    }
    
    // Selection highlight
    if (selected) {
        const cx = off.x + (selected.x + 0.5) * cellSize;
        const cy = off.y + (selected.y + 0.5) * cellSize;
        ctx.strokeStyle = '#fff';
        ctx.lineWidth = 3;
        ctx.setLineDash([5, 5]);
        ctx.strokeRect(off.x + selected.x * cellSize + 3, off.y + selected.y * cellSize + 3, cellSize - 6, cellSize - 6);
        ctx.setLineDash([]);
    }
    
    // Cursor highlight (keyboard)
    if (state === 'playing' || state === 'animating') {
        ctx.strokeStyle = 'rgba(255, 255, 100, 0.6)';
        ctx.lineWidth = 2;
        ctx.strokeRect(off.x + cursor.x * cellSize + 2, off.y + cursor.y * cellSize + 2, cellSize - 4, cellSize - 4);
    }
}

function drawParticles() {

        ctx.fill();
    });
    ctx.globalAlpha = 1;
}

function drawPopups() {
    const cellSize = getCellSize();
    const off = getBoardOffset();
    
    popups.forEach(p => {
        ctx.globalAlpha = p.life;
        ctx.fillStyle = p.small ? '#FF69B4' : (p.big ? '#00FFFF' : '#FFD700');
        const size = p.big ? 30 * SCALE : (p.small ? 16 * SCALE : 20 * SCALE);
        ctx.font = `bold ${size}px monospace`;
        ctx.textAlign = 'center';
        const cx = off.x + p.x * cellSize;
        const cy = off.y + p.y * cellSize - (1 - p.life) * 30 * SCALE;
        ctx.fillText(p.text, cx, cy);
    });
    ctx.globalAlpha = 1;
}

function drawGameOver() {
    ctx.fillStyle = 'rgba(0,0,0,0.7)';
    ctx.fillRect(0, 0, W, H);
    
    const titleSize = Math.max(24, 40 * SCALE);
    ctx.font = `bold ${titleSize}px monospace`;

        ctx.fillText('LEVEL COMPLETE!', W/2, H*0.35);

        ctx.fillStyle = '#FF3333';
        ctx.fillText('GAME OVER', W/2, H*0.35);
    }
    
    const fs = Math.max(14, 20 * SCALE);
    ctx.font = `${fs}px monospace`;
    ctx.fillStyle = '#fff';
    ctx.fillText('Final Score: ' + score, W/2, H*0.5);
    ctx.fillText('Target was: ' + targetScore, W/2, H*0.5 + fs * 1.5);
    
    const pulse = 0.5 + 0.5 * Math.sin(performance.now() / 500);
    ctx.globalAlpha = pulse;
    ctx.fillStyle = '#FFD700';
    ctx.font = `bold ${fs}px monospace`;
    ctx.fillText('Press R or Click to Play Again', W/2, H*0.7);
    ctx.globalAlpha = 1;
}

function update(dt) {
    // Update particles
    const cellSize = getCellSize();
    particles.forEach(p => {
        p.x += p.vx * dt;
        p.y += p.vy * dt;
        p.life -= dt * 2;
    });
    particles = particles.filter(p => p.life > 0);
    
    // Update popups
    popups.forEach(p => {
        p.life -= dt * 1.5;
    });
    popups = popups.filter(p => p.life > 0);
    
    // Shake
    if (shakeTimer > 0) {
        shakeTimer -= dt;
    }
}

function render() {
    ctx.clearRect(0, 0, W, H);
    
    if (state === 'start') {
        drawStartScreen();
        return;
    }
    
    // Apply shake
    let sx = 0, sy = 0;
    if (shakeTimer > 0) {
        sx = (Math.random() - 0.5) * shakeIntensity * SCALE;
        sy = (Math.random() - 0.5) * shakeIntensity * SCALE;
    }
    
    ctx.save();
    ctx.translate(sx, sy);
    
    // Background
    ctx.fillStyle = '#0a0a2a';
    ctx.fillRect(0, 0, W, H);
    
    drawBoard();
    drawParticles();
    drawPopups();
    drawHUD();
    
    ctx.restore();
    
    if (state === 'gameover') {
        drawGameOver();
    }
}

let lastTime = performance.now();
function gameLoop(time) {
    const dt = Math.min((time - lastTime) / 1000, 0.05);
    lastTime = time;
    
    update(dt);
    render();
    requestAnimationFrame(gameLoop);
}

// Input handling
function getGridPos(clientX, clientY) {
    const off = getBoardOffset();
    const cellSize = getCellSize();
    const gx = Math.floor((clientX - off.x) / cellSize);
    const gy = Math.floor((clientY - off.y) / cellSize);
    if (gx >= 0 && gx < GRID_SIZE && gy >= 0 && gy < GRID_SIZE) {
        return {x: gx, y: gy};
    }
    return null;
}

canvas.addEventListener('click', (e) => {
    initAudio();
    
    if (state === 'start') {
        startGame();
        return;
    }
    
    if (state === 'gameover') {
        level = 1;
        startGame();
        return;
    }
    
    if (state !== 'playing') return;
    
    const pos = getGridPos(e.clientX, e.clientY);
    if (!pos) return;
    
    cursor = pos;
    
    if (!selected) {
        selected = pos;
    } else {
        if (selected.x === pos.x && selected.y === pos.y) {
            selected = null;
        } else {
            const valid = trySwap(selected.x, selected.y, pos.x, pos.y);
            if (!valid) {
                playInvalidSound();
            }
            selected = null;
        }
    }
});

document.addEventListener('keydown', (e) => {
    initAudio();
    
    if (state === 'start') {
        if (e.key === ' ' || e.key === 'Enter') startGame();
        return;
    }
    
    if (state === 'gameover') {
        if (e.key === 'r' || e.key === 'R' || e.key === ' ' || e.key === 'Enter') {
            level = 1;
            startGame();
        }
        return;
    }
    
    if (state !== 'playing') return;
    
    if (e.key === 'r' || e.key === 'R') {
        level = 1;
        startGame();
        return;
    }
    
    switch(e.key) {
        case 'ArrowUp': cursor.y = Math.max(0, cursor.y - 1); e.preventDefault(); break;
        case 'ArrowDown': cursor.y = Math.min(GRID_SIZE-1, cursor.y + 1); e.preventDefault(); break;
        case 'ArrowLeft': cursor.x = Math.max(0, cursor.x - 1); e.preventDefault(); break;
        case 'ArrowRight': cursor.x = Math.min(GRID_SIZE-1, cursor.x + 1); e.preventDefault(); break;
        case ' ':
        case 'Enter':
            if (!selected) {
                selected = {x: cursor.x, y: cursor.y};
            } else {
                const valid = trySwap(selected.x, selected.y, cursor.x, cursor.y);
                if (!valid) playInvalidSound();
                selected = null;
            }
            e.preventDefault();
            break;
        case 'Escape':
            selected = null;
            break;
    }
});
