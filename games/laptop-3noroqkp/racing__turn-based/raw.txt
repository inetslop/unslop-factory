const canvas = document.getElementById('game');
const ctx = canvas.getContext('2d');

let W, H;
function resize() {
    W = canvas.width = window.innerWidth;
    H = canvas.height = window.innerHeight;
}
window.addEventListener('resize', resize);
resize();

// Audio
let audioCtx = null;
function initAudio() {
    if (!audioCtx) {
        audioCtx = new (window.AudioContext || window.webkitAudioContext)();
    }
}
function playSound(freq, dur, type) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = type || 'square';
    osc.frequency.value = freq;
    gain.gain.setValueAtTime(0.1, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + dur);
}

// Game State
const SEGMENTS = 30;
const CAR_COLORS = ['#ff4444', '#44ff44', '#4488ff', '#ffaa22'];
const CAR_NAMES = ['You', 'Rival Red', 'Rival Blue', 'Rival Yellow'];

let gameState = 'start'; // start, playing, finished
let track = [];
let cars = [];
let currentTurn = 0;
let maxTurns = 40;
let nitroUses = 3;
let message = '';
let messageTimer = 0;
let animFrame = 0;
let scrollOffset = 0;
let targetScroll = 0;

function generateTrack() {
    track = [];
    const types = ['STRAIGHT', 'LEFT', 'RIGHT', 'HAIRPIN'];
    for (let i = 0; i < SEGMENTS; i++) {
        if (i < 2 || i >= SEGMENTS - 2) {
            track.push('STRAIGHT');
        } else {
            track.push(types[Math.floor(Math.random() * types.length)]);
        }
    }
}

function initRace() {
    generateTrack();
    cars = [];
    for (let i = 0; i < 4; i++) {
        cars.push({
            name: CAR_NAMES[i],
            color: CAR_COLORS[i],
            pos: 0, // meters along track
            speed: 50,
            nitro: i === 0 ? 3 : 2,
            finished: false,
            finishTime: null
        });
    }
    currentTurn = 1;
    nitroUses = 3;
    gameState = 'playing';
    message = 'Race Start! Turn 1/40';
    messageTimer = 60;
}

function getCarDistance(car) {
    return car.pos + (car.speed / 10);
}

function processTurn(playerAction) {
    if (gameState !== 'playing') return;

    // Process player
    const p = cars[0];
    applyAction(p, playerAction);

    // Process AI
    for (let i = 1; i < cars.length; i++) {
        const ai = cars[i];
        const action = getAIAction(ai);
        applyAction(ai, action);
    }

    // Advance positions
    for (let i = 0; i < cars.length; i++) {
        const c = cars[i];
        if (!c.finished) {
            c.pos += c.speed / 10;
            if (c.pos >= SEGMENTS * 10) {
                c.finished = true;
                c.finishTime = currentTurn;
                message = `${c.name} finished! Turn ${currentTurn}`;
                messageTimer = 90;
                playSound(c === cars[0] ? 880 : 440, 0.5, 'sine');
            }
        }
    }

    currentTurn++;
    if (messageTimer <= 0) {
        message = `Turn ${currentTurn}/${maxTurns}`;
        messageTimer = 30;
    }

    // Check end conditions
    if (currentTurn > maxTurns || cars.every(c => c.finished)) {
        gameState = 'finished';
        playSound(cars[0].pos >= SEGMENTS * 10 ? 880 : 220, 0.8, 'sine');
    }

    playSound(200 + Math.random() * 200, 0.15, 'square');
}

function applyAction(car, action) {
    const segIdx = Math.min(Math.floor(car.pos / 10), SEGMENTS - 1);
    const segType = track[segIdx] || 'STRAIGHT';

    switch (action) {
        case 'ACCELERATE':
            if (segType === 'STRAIGHT') {
                car.speed = Math.min(car.speed + 15, 180);
            } else {
                car.speed = Math.max(car.speed - 5, 20);
            }
            break;
        case 'BRAKE':
            car.speed = Math.max(car.speed - 20, 20);
            break;
        case 'STEER_LEFT':
            if (segType === 'LEFT') {
                car.speed = Math.min(car.speed + 10, 180);
            } else if (segType === 'RIGHT' || segType === 'HAIRPIN') {
                car.speed = Math.max(car.speed - 25, 20);
            } else {
                car.speed = Math.max(car.speed - 5, 20);
            }
            break;
        case 'STEER_RIGHT':
            if (segType === 'RIGHT') {
                car.speed = Math.min(car.speed + 10, 180);
            } else if (segType === 'LEFT' || segType === 'HAIRPIN') {
                car.speed = Math.max(car.speed - 25, 20);
            } else {
                car.speed = Math.max(car.speed - 5, 20);
            }
            break;
        case 'NITRO':
            if (car.nitro > 0) {
                car.nitro--;
                car.speed = Math.min(car.speed + 30, 200);
                playSound(600, 0.4, 'sawtooth');
            } else {
                car.speed = Math.max(car.speed - 10, 20);
            }
            break;
    }
}

function getAIAction(car) {
    const segIdx = Math.min(Math.floor(car.pos / 10), SEGMENTS - 1);
    const segType = track[segIdx] || 'STRAIGHT';

    if (car.nitro > 0 && segType === 'STRAIGHT' && car.speed < 150 && Math.random() < 0.3) {
        return 'NITRO';
    }
    if (segType === 'LEFT') {
        return Math.random() < 0.8 ? 'STEER_LEFT' : 'ACCELERATE';
    }
    if (segType === 'RIGHT') {
        return Math.random() < 0.8 ? 'STEER_RIGHT' : 'ACCELERATE';
    }
    if (segType === 'HAIRPIN') {
        return Math.random() < 0.5 ? 'STEER_LEFT' : 'STEER_RIGHT';
    }
    if (car.speed < 140) return 'ACCELERATE';
    return 'BRAKE';
}

function getPositionRank(car) {
    let rank = 1;
    for (let i = 0; i < cars.length; i++) {
        if (i !== cars.indexOf(car)) {
            const other = cars[i];
            if ((other.finished && !car.finished) || (!other.finished && !car.finished && other.pos > car.pos) || (other.finished && other.finishTime < car.finishTime && car.finished)) {
                rank++;
            }
        }
    }
    return rank;
}

// Input
let pendingAction = null;
window.addEventListener('keydown', (e) => {
    initAudio();
    if (gameState === 'start') {
        if (e.key === ' ' || e.key === 'Enter') {
            initRace();
        }
        return;
    }
    if (gameState === 'finished') {
        if (e.key === ' ' || e.key === 'Enter' || e.key === 'r') {
            gameState = 'start';
        }
        return;
    }
    if (gameState === 'playing') {
        let action = null;
        switch (e.key) {
            case 'w': case 'ArrowUp': action = 'ACCELERATE'; break;
            case 's': case 'ArrowDown': action = 'BRAKE'; break;
            case 'a': case 'ArrowLeft': action = 'STEER_LEFT'; break;
            case 'd': case 'ArrowRight': action = 'STEER_RIGHT'; break;
            case ' ': case 'n': action = 'NITRO'; break;
        }
        if (action) {
            pendingAction = action;
        }
    }
});

// Rendering
function draw() {
    ctx.fillStyle = '#1a1a2e';
    ctx.fillRect(0, 0, W, H);

    if (gameState === 'start') {
        drawStartScreen();
    } else if (gameState === 'playing' || gameState === 'finished') {
        drawGame();
    }

    animFrame++;
    requestAnimationFrame(draw);
}

function drawStartScreen() {
    ctx.fillStyle = '#fff';
    ctx.font = `bold ${Math.min(W/10, 48)}px Courier New`;
    ctx.textAlign = 'center';
    ctx.fillText('TURN-BASED RACING', W/2, H/4);

    ctx.font = `${Math.min(W/30, 18)}px Courier New`;
    ctx.fillStyle = '#aaa';
    const lines = [
        'Navigate the track in 40 turns to win!',
        '',
        'Controls:',
        'W / ↑ - Accelerate',
        'S / ↓ - Brake',
        'A / ← - Steer Left (for left curves)',
        'D / → - Steer Right (for right curves)',
        'Space / N - Nitro Boost (3 per race)',
        '',
        'Match your steering to the track curve!',
        'Wrong steering slows you down.',
        'Hairpins require either left or right steer.',
        '',
        'Press SPACE or ENTER to Start'
    ];
    const startY = H/2 - 10;
    lines.forEach((line, i) => {
        ctx.fillStyle = i < 2 ? '#aaa' : (i === 2 || i === 7 ? '#4fc3f7' : '#888');
        ctx.fillText(line, W/2, startY + i * 24);
    });

    // Animated cars
    const t = animFrame * 0.03;
    for (let i = 0; i < 4; i++) {
        const x = W/2 - 150 + i * 80;
        const y = H - 80 + Math.sin(t + i) * 10;
        drawCar(x, y, CAR_COLORS[i], 30, 18);
    }
}

function drawGame() {
    // Draw track
    const segH = H / 8;
    const roadW = Math.min(W * 0.5, 400);
    const roadX = W/2 - roadW/2;

    for (let i = 0; i < 8; i++) {
        const segIdx = Math.floor(cars[0].pos / 10) + i - 3;
        if (segIdx < 0 || segIdx >= SEGMENTS) continue;

        const y = H - (i * segH) - segH/2;
        const type = track[segIdx];

        // Road background
        ctx.fillStyle = '#333';
        ctx.fillRect(roadX, y - segH/2 + 5, roadW, segH - 10);

        // Road markings based on type
        if (type === 'LEFT') {
            ctx.strokeStyle = '#ff0';
            ctx.lineWidth = 4;
            ctx.beginPath();
            ctx.moveTo(roadX + roadW * 0.3, y + segH/4);
            ctx.quadraticCurveTo(roadX + roadW * 0.5, y - segH/4, roadX + roadW * 0.7, y + segH/4);
            ctx.stroke();
        } else if (type === 'RIGHT') {
            ctx.strokeStyle = '#ff0';
            ctx.lineWidth = 4;
            ctx.beginPath();
            ctx.moveTo(roadX + roadW * 0.7, y + segH/4);
            ctx.quadraticCurveTo(roadX + roadW * 0.5, y - segH/4, roadX + roadW * 0.3, y + segH/4);
            ctx.stroke();
        } else if (type === 'HAIRPIN') {
            ctx.strokeStyle = '#f0f';
            ctx.lineWidth = 4;
            ctx.beginPath();
            ctx.moveTo(roadX + roadW * 0.2, y + segH/4);
            ctx.quadraticCurveTo(roadX + roadW * 0.5, y - segH/3, roadX + roadW * 0.8, y + segH/4);
            ctx.stroke();
        } else {
            // Straight - dashed center line
            ctx.strokeStyle = '#ff0';
            ctx.lineWidth = 2;
            ctx.setLineDash([10, 15]);
            ctx.beginPath();
            ctx.moveTo(W/2, y - segH/4);
            ctx.lineTo(W/2, y + segH/4);
            ctx.stroke();
            ctx.setLineDash([]);
        }

        // Segment label
        ctx.fillStyle = '#666';
        ctx.font = '12px Courier New';
        ctx.textAlign = 'left';
        ctx.fillText(`Seg ${segIdx + 1}: ${type}`, roadX + 10, y + 4);

        // Current segment highlight
        if (i === 3) {
            ctx.strokeStyle = '#fff';
            ctx.lineWidth = 2;
            ctx.strokeRect(roadX, y - segH/2 + 5, roadW, segH - 10);
        }
    }

    // Draw cars
    for (let i = 0; i < cars.length; i++) {
        const c = cars[i];
        if (c.finished) continue;

        // Find relative position to player
        const relPos = c.pos - cars[0].pos;
        const carY = H - 120 - relPos * 5;
        if (carY < 30 || carY > H - 30) continue;

        const laneX = W/2 + (i - 1.5) * 60;
        drawCar(laneX, carY, c.color, i === 0 ? 34 : 28, i === 0 ? 20 : 16);

        // Speed indicator
        ctx.fillStyle = '#fff';
        ctx.font = '11px Courier New';
        ctx.textAlign = 'center';
        ctx.fillText(`${Math.round(c.speed)}km/h`, laneX, carY - 18);
    }

    // HUD
    const p = cars[0];
    ctx.fillStyle = 'rgba(0,0,0,0.7)';
    ctx.fillRect(10, 10, 220, 130);
    ctx.strokeStyle = '#4fc3f7';
    ctx.lineWidth = 2;
    ctx.strokeRect(10, 10, 220, 130);

    ctx.fillStyle = '#fff';
    ctx.font = 'bold 16px Courier New';
    ctx.textAlign = 'left';
    const rank = getPositionRank(p);
    ctx.fillText(`Pos: ${rank}th / 4`, 20, 35);
    ctx.fillText(`Speed: ${Math.round(p.speed)} km/h`, 20, 58);
    ctx.fillText(`Nitro: ${'🔥'.repeat(p.nitro) || '(none)'}`, 20, 81);
    ctx.fillText(`Turn: ${currentTurn}/${maxTurns}`, 20, 104);

    const segIdx = Math.min(Math.floor(p.pos / 10), SEGMENTS - 1);
    const segType = track[segIdx];
    ctx.fillStyle = segType === 'STRAIGHT' ? '#4fc3f7' : (segType === 'HAIRPIN' ? '#f0f' : '#ff0');
    ctx.fillText(`Next: ${segType}`, 20, 127);

    // Progress bar
    const progW = W - 60;
    ctx.fillStyle = '#333';
    ctx.fillRect(30, H - 30, progW, 14);
    for (let i = 0; i < cars.length; i++) {
        const c = cars[i];
        const px = 30 + (c.pos / (SEGMENTS * 10)) * progW;
        ctx.fillStyle = c.color;
        ctx.beginPath();
        ctx.arc(px, H - 23, 6, 0, Math.PI * 2);
        ctx.fill();
    }

    // Message
    if (messageTimer > 0) {
        messageTimer--;
        ctx.fillStyle = 'rgba(0,0,0,0.8)';
        const mw = ctx.measureText(message).width + 40;
        ctx.fillRect(W/2 - mw/2, 60, mw, 36);
        ctx.fillStyle = '#fff';
        ctx.font = 'bold 18px Courier New';
        ctx.textAlign = 'center';
        ctx.fillText(message, W/2, 84);
    }

    // Action prompt
    if (gameState === 'playing') {
        ctx.fillStyle = 'rgba(0,0,0,0.6)';
        ctx.fillRect(W/2 - 180, H - 70, 360, 30);
        ctx.fillStyle = '#4fc3f7';
        ctx.font = '14px Courier New';
        ctx.textAlign = 'center';
        ctx.fillText('W:Accel S:Brake A:◄ D:► Space:Nitro', W/2, H - 50);
    }

    // Finished screen
    if (gameState === 'finished') {
        ctx.fillStyle = 'rgba(0,0,0,0.8)';
        ctx.fillRect(W/2 - 200, H/2 - 120, 400, 240);

        ctx.fillStyle = '#fff';
        ctx.font = 'bold 28px Courier New';
        ctx.textAlign = 'center';
        ctx.fillText('RACE COMPLETE', W/2, H/2 - 80);

        // Sort by finish time or position
        const sorted = [...cars].sort((a, b) => {
            if (a.finished && b.finished) return a.finishTime - b.finishTime;
            if (a.finished) return -1;
            if (b.finished) return 1;
            return b.pos - a.pos;
        });

        ctx.font = '18px Courier New';
        sorted.forEach((c, i) => {
            ctx.fillStyle = c.color;
            const status = c.finished ? `T${c.finishTime}` : `${Math.round(c.pos)}m`;
            ctx.fillText(`${i + 1}. ${c.name} - ${status}`, W/2, H/2 - 40 + i * 35);
        });

        ctx.fillStyle = '#aaa';
        ctx.font = '16px Courier New';
        ctx.fillText('Press R or SPACE to play again', W/2, H/2 + 100);
    }
}

function drawCar(x, y, color, w, h) {
    ctx.fillStyle = color;
    ctx.fillRect(x - w/2, y - h/2, w, h);
    ctx.fillStyle = 'rgba(255,255,255,0.3)';
    ctx.fillRect(x - w/4, y - h/4, w/2, h/2);
}

// Main loop with turn processing
let lastTime = 0;
function gameLoop(time) {
    if (gameState === 'playing' && pendingAction) {
        processTurn(pendingAction);
        pendingAction = null;
    }
    draw();
    requestAnimationFrame(gameLoop);
}

generateTrack();
requestAnimationFrame(draw);
