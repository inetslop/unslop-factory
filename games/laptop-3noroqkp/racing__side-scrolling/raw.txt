const canvas = document.getElementById('game');
const ctx = canvas.getContext('2d');

let W, H;

// Audio
let audioCtx = null;
let engineOsc = null;
let engineGain = null;

function initAudio() {
    if (audioCtx) return;
    audioCtx = new (window.AudioContext || window.webkitAudioContext)();
    engineOsc = audioCtx.createOscillator();
    engineOsc.type = 'sawtooth';
    engineOsc.frequency.value = 60;
    engineGain = audioCtx.createGain();
    engineGain.gain.value = 0;
    engineOsc.connect(engineGain);
    engineGain.connect(audioCtx.destination);
    engineOsc.start();
}

function playEngine(speed) {
    if (!audioCtx) return;
    const freq = 50 + speed * 8;
    engineOsc.frequency.value = freq;
    engineGain.gain.value = 0.03;
}

function stopEngine() {
    if (engineGain) engineGain.gain.value = 0;
}

function playCrash() {
    if (!audioCtx) return;
    const buf = audioCtx.createBuffer(1, 4410, 44100);
    const data = buf.getChannelData(0);
    for (let i = 0; i < data.length; i++) data[i] = (Math.random() * 2 - 1) * Math.exp(-i / 500);
    const src = audioCtx.createBufferSource();
    src.buffer = buf;
    const gain = audioCtx.createGain();
    gain.gain.value = 0.4;
    src.connect(gain);
    gain.connect(audioCtx.destination);
    src.start();
}

function playJump() {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    osc.type = 'sine';
    osc.frequency.setValueAtTime(300, audioCtx.currentTime);
    osc.frequency.linearRampToValueAtTime(600, audioCtx.currentTime + 0.1);
    const gain = audioCtx.createGain();
    gain.gain.setValueAtTime(0.1, audioCtx.currentTime);
    gain.gain.linearRampToValueAtTime(0, audioCtx.currentTime + 0.15);
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + 0.15);
}

function playCoin() {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    osc.type = 'sine';
    osc.frequency.setValueAtTime(800, audioCtx.currentTime);
    osc.frequency.linearRampToValueAtTime(1200, audioCtx.currentTime + 0.08);
    const gain = audioCtx.createGain();
    gain.gain.setValueAtTime(0.15, audioCtx.currentTime);
    gain.gain.linearRampToValueAtTime(0, audioCtx.currentTime + 0.12);
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + 0.12);
}

// Game state
let gameState = 'start'; // start, playing, gameover
let score = 0;
let highScore = 0;
try { highScore = parseInt(localStorage.getItem('sideRushHigh') || '0'); } catch(e) {}
let speed = 0;
let baseSpeed = 4;
let maxSpeed = 18;
let frameCount = 0;

// Player
const player = {
    x: 0, y: 0, w: 60, h: 30, vy: 0, grounded: true, jumping: false, ducking: false
};

const GRAVITY = 0.8;
const JUMP_FORCE = -14;

// Road
let roadY = 0;
let scrollX = 0;

// Obstacles
let obstacles = [];
let coins = [];
let particles = [];

// Background layers
let bgOffset1 = 0, bgOffset2 = 0, bgOffset3 = 0;

function resetGame() {
    score = 0;
    speed = baseSpeed;
    frameCount = 0;
    roadY = H * 0.75;
    player.x = W * 0.15;
    player.y = roadY - player.h;
    player.vy = 0;
    player.grounded = true;
    player.jumping = false;
    player.ducking = false;
    obstacles = [];
    coins = [];
    particles = [];
    bgOffset1 = 0; bgOffset2 = 0; bgOffset3 = 0;
}

function spawnObstacle() {
    const type = Math.random();
    let obs = { x: W + 50, y: roadY - 40, w: 40, h: 40, type: '' };
    
    if (type < 0.3) {
        obs.type = 'barrier';
        obs.w = 20; obs.h = 50;
        obs.y = roadY - obs.h;
    } else if (type < 0.6) {
        obs.type = 'car';
        obs.w = 55; obs.h = 30;
        obs.y = roadY - obs.h;
    } else if (type < 0.8) {
        obs.type = 'spike';
        obs.w = 35; obs.h = 25;
        obs.y = roadY - obs.h;
    } else {
        obs.type = 'tall';
        obs.w = 25; obs.h = 70;
        obs.y = roadY - obs.h;
    }
    
    // Coin above some obstacles
    if (Math.random() < 0.4) {
        coins.push({ x: W + 80, y: roadY - 90, w: 20, h: 20, collected: false });
    }
    
    obstacles.push(obs);
}

function spawnCoinRow() {
    const startX = W + 100;
    for (let i = 0; i < 4; i++) {
        coins.push({ x: startX + i * 35, y: roadY - 80, w: 20, h: 20, collected: false });
    }
}

function addParticles(x, y, color, count) {
    for (let i = 0; i < count; i++) {
        particles.push({
            x: x, y: y,
            vx: (Math.random() - 0.5) * 8,
            vy: (Math.random() - 1) * 6,
            life: 30 + Math.random() * 20,
            color: color,
            size: 3 + Math.random() * 4
        });
    }
}

function checkCollision(a, b) {
    const shrink = 5;
    return a.x + shrink < b.x + b.w &&
           a.x + a.w - shrink > b.x &&
           a.y + shrink < b.y + b.h &&
           a.y + a.h - shrink > b.y;
}

function update() {
    if (gameState !== 'playing') return;
    
    frameCount++;
    speed = Math.min(baseSpeed + frameCount * 0.003, maxSpeed);
    score += speed * 0.1;
    
    // Background scroll
    bgOffset1 += speed * 0.2;
    bgOffset2 += speed * 0.5;
    bgOffset3 += speed * 0.8;
    scrollX = (scrollX + speed) % 80;
    
    // Player physics
    if (!player.grounded) {
        player.vy += GRAVITY;
        player.y += player.vy;
        if (player.y >= roadY - player.h) {
            player.y = roadY - player.h;
            player.vy = 0;
            player.grounded = true;
            player.jumping = false;
            addParticles(player.x + player.w/2, player.y + player.h, '#8B7355', 5);
        }
    }
    
    if (player.ducking && player.grounded) {
        player.h = 18;
        player.y = roadY - player.h;
    } else if (!player.ducking) {
        if (player.grounded) player.h = 30;
        if (player.grounded) player.y = roadY - player.h;
    }
    
    // Spawn obstacles
    const spawnRate = Math.max(60 - Math.floor(speed * 2), 30);
    if (frameCount % spawnRate === 0) {
        if (Math.random() < 0.7) spawnObstacle();
        else if (Math.random() < 0.5) spawnCoinRow();
    }
    
    // Move obstacles
    for (let i = obstacles.length - 1; i >= 0; i--) {
        obstacles[i].x -= speed;
        if (obstacles[i].x + obstacles[i].w < -50) obstacles.splice(i, 1);
    }
    
    // Move coins
    for (let i = coins.length - 1; i >= 0; i--) {
        coins[i].x -= speed;
        if (!coins[i].collected && checkCollision(player, coins[i])) {
            coins[i].collected = true;
            score += 50;
            playCoin();
            addParticles(coins[i].x, coins[i].y, '#FFD700', 8);
        }
        if (coins[i].x < -30) coins.splice(i, 1);
    }
    
    // Check collisions with obstacles
    for (let i = obstacles.length - 1; i >= 0; i--) {
        if (checkCollision(player, obstacles[i])) {
            gameState = 'gameover';
            playCrash();
            stopEngine();
            addParticles(player.x + player.w/2, player.y + player.h/2, '#FF4444', 20);
            try { localStorage.setItem('sideRushHigh', Math.floor(score)); } catch(e) {}
            highScore = Math.max(highScore, Math.floor(score));
            return;
        }
    }
    
    // Particles
    for (let i = particles.length - 1; i >= 0; i--) {
        const p = particles[i];
        p.x += p.vx;
        p.y += p.vy;
        p.vy += 0.3;
        p.life--;
        if (p.life <= 0) particles.splice(i, 1);
    }
    
    // Engine sound
    playEngine(speed);
}

function drawBackground() {
    // Sky gradient
    const skyGrad = ctx.createLinearGradient(0, 0, 0, roadY);
    skyGrad.addColorStop(0, '#87CEEB');
    skyGrad.addColorStop(0.6, '#B0E0FF');
    skyGrad.addColorStop(1, '#E0F0FF');
    ctx.fillStyle = skyGrad;
    ctx.fillRect(0, 0, W, roadY);
    
    // Clouds (layer 1 - slowest)
    ctx.fillStyle = 'rgba(255,255,255,0.8)';
    const cloudSpacing = 300;
    const cOffset = bgOffset1 % cloudSpacing;
    for (let i = -1; i < W / cloudSpacing + 2; i++) {
        const cx = i * cloudSpacing - cOffset + 150;
        const cy = H * 0.1 + Math.sin(i * 2.3) * 30;
        drawCloud(cx, cy, 40);
    }
    
    // Mountains (layer 2)
    ctx.fillStyle = '#6B8E9B';
    const mOffset = bgOffset2 % 200;
    for (let i = -1; i < W / 100 + 2; i++) {
        const mx = i * 100 - mOffset;
        const mh = 80 + Math.sin(i * 1.7) * 40;
        ctx.beginPath();
        ctx.moveTo(mx, roadY);
        ctx.lineTo(mx + 50, roadY - mh);
        ctx.lineTo(mx + 100, roadY);
        ctx.fill();
    }
    
    // Trees (layer 3 - faster)
    const tOffset = bgOffset3 % 120;
    for (let i = -1; i < W / 60 + 2; i++) {
        const tx = i * 60 - tOffset + 30;
        const th = 40 + Math.sin(i * 3.1) * 15;
        // Trunk
        ctx.fillStyle = '#5D4037';
        ctx.fillRect(tx - 3, roadY - th, 6, th);
        // Leaves
        ctx.fillStyle = '#2E7D32';
        ctx.beginPath();
        ctx.arc(tx, roadY - th, 18, 0, Math.PI * 2);
        ctx.fill();
    }
}

function drawCloud(x, y, size) {
    ctx.beginPath();
    ctx.arc(x, y, size * 0.6, 0, Math.PI * 2);
    ctx.arc(x + size * 0.5, y - size * 0.2, size * 0.5, 0, Math.PI * 2);
    ctx.arc(x + size, y, size * 0.4, 0, Math.PI * 2);
    ctx.fill();
}

function drawRoad() {
    // Road surface
    const roadH = H - roadY;
    ctx.fillStyle = '#555555';
    ctx.fillRect(0, roadY, W, roadH);
    
    // Road top edge
    ctx.fillStyle = '#333333';
    ctx.fillRect(0, roadY, W, 4);
    
    // Lane markings
    ctx.fillStyle = '#FFD700';
    const markW = 40;
    const gap = 40;
    for (let i = -1; i < W / (markW + gap) + 2; i++) {
        const mx = i * (markW + gap) - scrollX;
        ctx.fillRect(mx, roadY + roadH / 2, markW, 4);
    }
    
    // Road texture lines
    ctx.fillStyle = 'rgba(0,0,0,0.1)';
    for (let i = 0; i < W / 30 + 1; i++) {
        const lx = i * 30 - scrollX % 30;
        ctx.fillRect(lx, roadY + 5, 15, 2);
    }
}

    ctx.fillStyle = '#E53935';
    ctx.beginPath();

        ctx.fill();

        ctx.fillRect(px + pw * 0.3, py + 3, pw * 0.35, ph * 0.3);

    // Wheels
    const wheelR = 8;
    const wheelY = py + ph - 2;
    ctx.fillStyle = '#212121';
    ctx.beginPath();
    ctx.arc(px + pw * 0.2, wheelY, wheelR, 0, Math.PI * 2);
    ctx.fill();
    ctx.beginPath();
    ctx.arc(px + pw * 0.8, wheelY, wheelR, 0, Math.PI * 2);
    ctx.fill();
    
    // Wheel hubs
    ctx.fillStyle = '#757575';
    ctx.beginPath();
    ctx.arc(px + pw * 0.2, wheelY, 4, 0, Math.PI * 2);
    ctx.fill();

    ctx.arc(px + pw * 0.8, wheelY, 4, 0, Math.PI * 2);

    if (speed > 10) {

        ctx.lineWidth = 2;
        for (let i = 0; i < 3; i++) {

        }

function drawObstacle(obs) {
    if (obs.type === 'barrier') {
        ctx.fillStyle = '#FF9800';
        ctx.fillRect(obs.x, obs.y, obs.w, obs.h);
        ctx.fillStyle = '#212121';
        for (let i = 0; i < obs.h / 15; i++) {
            ctx.fillRect(obs.x, obs.y + i * 15, obs.w, 7);
        }
    } else if (obs.type === 'car') {
        // Enemy car
        ctx.fillStyle = '#42A5F5';
        ctx.beginPath();
        ctx.roundRect(obs.x, obs.y + obs.h * 0.3, obs.w, obs.h * 0.7, 5);
        ctx.fill();
        ctx.fillStyle = '#1E88E5';
        ctx.beginPath();
        ctx.roundRect(obs.x + obs.w * 0.2, obs.y, obs.w * 0.6, obs.h * 0.45, 3);
        ctx.fill();
        // Wheels
        ctx.fillStyle = '#212121';
        ctx.beginPath();
        ctx.arc(obs.x + obs.w * 0.2, obs.y + obs.h - 2, 7, 0, Math.PI * 2);
        ctx.fill();
        ctx.beginPath();
        ctx.arc(obs.x + obs.w * 0.8, obs.y + obs.h - 2, 7, 0, Math.PI * 2);
        ctx.fill();
    } else if (obs.type === 'spike') {
        ctx.fillStyle = '#9E9E9E';
        for (let i = 0; i < 3; i++) {
            const sx = obs.x + i * (obs.w / 3);
            ctx.beginPath();
            ctx.moveTo(sx, obs.y + obs.h);
            ctx.lineTo(sx + obs.w / 6, obs.y);
            ctx.lineTo(sx + obs.w / 3, obs.y + obs.h);
            ctx.fill();
        }
    } else if (obs.type === 'tall') {
        ctx.fillStyle = '#FF5722';
        ctx.fillRect(obs.x, obs.y, obs.w, obs.h);
        // X pattern
        ctx.strokeStyle = '#FFF';
        ctx.lineWidth = 3;
        ctx.beginPath();
        ctx.moveTo(obs.x + 4, obs.y + 4);
        ctx.lineTo(obs.x + obs.w - 4, obs.y + obs.h - 4);
        ctx.moveTo(obs.x + obs.w - 4, obs.y + 4);
        ctx.lineTo(obs.x + 4, obs.y + obs.h - 4);
        ctx.stroke();
    }
}

function drawCoin(c) {
    if (c.collected) return;
    const bob = Math.sin(frameCount * 0.1) * 3;
    ctx.fillStyle = '#FFD700';
    ctx.beginPath();
    ctx.arc(c.x + c.w/2, c.y + c.h/2 + bob, c.w/2, 0, Math.PI * 2);
    ctx.fill();
    ctx.fillStyle = '#FFA000';
    ctx.font = 'bold 12px monospace';
    ctx.textAlign = 'center';
    ctx.fillText('$', c.x + c.w/2, c.y + c.h/2 + bob + 4);
}

function drawParticles() {
    for (const p of particles) {
        ctx.globalAlpha = p.life / 50;
        ctx.fillStyle = p.color;
        ctx.fillRect(p.x, p.y, p.size, p.size);
    }
    ctx.globalAlpha = 1;
}

function drawHUD() {
    // Score
    ctx.fillStyle = 'rgba(0,0,0,0.5)';
    ctx.beginPath();
    ctx.roundRect(W - 220, 10, 210, 80, 10);
    ctx.fill();
    
    ctx.fillStyle = '#FFF';
    ctx.font = 'bold 16px monospace';
    ctx.textAlign = 'left';
    ctx.fillText('SCORE: ' + Math.floor(score), W - 205, 35);
    ctx.fillText('BEST:  ' + Math.max(highScore, Math.floor(score)), W - 205, 55);
    
    // Speed bar
    const speedPct = (speed - baseSpeed) / (maxSpeed - baseSpeed);
    ctx.fillStyle = '#333';
    ctx.fillRect(W - 205, 68, 180, 12);
    const spdColor = speedPct > 0.7 ? '#FF5252' : speedPct > 0.4 ? '#FFD740' : '#69F0AE';
    ctx.fillStyle = spdColor;
    ctx.fillRect(W - 205, 68, 180 * Math.min(speedPct, 1), 12);
}

function drawGameOver() {
    // Darken
    ctx.fillStyle = 'rgba(0,0,0,0.7)';
    ctx.fillRect(0, 0, W, H);
    
    ctx.fillStyle = '#FF5252';
    ctx.font = 'bold 48px monospace';
    ctx.textAlign = 'center';
    ctx.fillText('CRASHED!', W/2, H * 0.3);

    ctx.font = '24px monospace';
    ctx.fillText('Score: ' + Math.floor(score), W/2, H * 0.42);

        ctx.fillText('★ NEW HIGH SCORE! ★', W/2, H * 0.5);
    } else {
        ctx.fillStyle = '#90A4AE';
        ctx.font = '18px monospace';
        ctx.fillText('Best: ' + highScore, W/2, H * 0.5);
    }
    
    const pulse = Math.sin(Date.now() * 0.005) * 0.3 + 0.7;
    ctx.globalAlpha = pulse;
    ctx.fillStyle = '#4FC3F7';
    ctx.font = 'bold 20px monospace';
    ctx.fillText('Press SPACE or ENTER to Restart', W/2, H * 0.7);
    ctx.globalAlpha = 1;

function draw() {
    if (gameState === 'start') {
        drawStartScreen();
        return;
    }
    
    ctx.clearRect(0, 0, W, H);
    drawBackground();
    drawRoad();
    
    // Coins
    for (const c of coins) drawCoin(c);
    
    // Obstacles
    for (const obs of obstacles) drawObstacle(obs);
    
    // Player
    drawPlayer();
    
    // Particles
    drawParticles();
    
    // HUD
    drawHUD();
    
    if (gameState === 'gameover') {
        drawGameOver();
    }
}

            player.jumping = true;

        }

        if (e.code === 'Space' || e.code === 'Enter') {
            resetGame();
            gameState = 'playing';
        }
    }
});
