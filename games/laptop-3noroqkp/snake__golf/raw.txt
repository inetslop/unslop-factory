function playTone(freq, dur, type, vol) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = type || 'sine';
    osc.frequency.value = freq;
    gain.gain.setValueAtTime(vol || 0.1, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + dur);
}

function loadLevel(idx) {
    const lvl = levels[idx];
    snake = [];
    const s = lvl.start;
    for (let i = 0; i < 3; i++) {

function isWall(x, y) {
    const lvl = levels[currentLevel];
    return lvl.walls.some(w => w[0] === x && w[1] === y);
}
function isSand(x, y) {
    const lvl = levels[currentLevel];
    return lvl.sand.some(s => s[0] === x && s[1] === y);
}

function moveSnake() {
    if (state !== 'playing') return;
    if (pendingDir.x !== 0 || pendingDir.y !== 0) {
        // Don't allow 180 turn
        if (!(pendingDir.x === -dir.x && pendingDir.y === -dir.y)) {
            dir = {...pendingDir};
        }
    } else {
        return;
    }
    const head = snake[0];
    const nx = head.x + dir.x;
    const ny = head.y + dir.y;

    // Bounds check
    if (nx < 0 || nx >= gridW || ny < 0 || ny >= gridH) {
        sfxFail();
        return;
    }
    // Wall check
    if (isWall(nx, ny)) {
        sfxFail();
        return;
    }
    // Self collision
    if (snake.some(s => s.x === nx && s.y === ny)) {
        sfxFail();
        return;
    }

    // Move
    snake.unshift({x: nx, y: ny});
    snake.pop();
    
    const cost = isSand(nx, ny) ? 2 : 1;
    levelStrokes += cost;
    if (isSand(nx, ny)) sfxSand(); else sfxMove();

    // Check cup
    if (nx === cupPos.x && ny === cupPos.y) {
        sfxWin();
        state = 'levelComplete';
        totalStrokes += levelStrokes;
    }
}

        drawStartScreen();
    } else if (state === 'playing') {
        ctx.fillStyle = '#1a4a2e';
        ctx.fillRect(0, 0, W, H);
        drawGrid();
        drawHUD();
    } else if (state === 'levelComplete') {
        ctx.fillStyle = '#1a4a2e';
        ctx.fillRect(0, 0, W, H);
        drawGrid();
        drawHUD();
        drawLevelComplete();
    } else if (state === 'finished') {
        ctx.fillStyle = '#1a4a2e';
        ctx.fillRect(0, 0, W, H);
        drawGrid();
        drawHUD();
        drawFinished();

// Input
let moveCooldown = 0;
function handleKey(e) {
    if (state === 'start') {
        if (e.key === 'Enter' || e.key === ' ') {
            initAudio();
            state = 'playing';
            currentLevel = 0;
            totalStrokes = 0;
            loadLevel(0);
        }
        return;
    }

    if (state === 'levelComplete') {
        if (e.key === 'Enter' || e.key === ' ') {
            currentLevel++;
            if (currentLevel >= levels.length) {
                state = 'finished';
            } else {
                state = 'playing';
                loadLevel(currentLevel);
            }
        }
        return;
    }

    if (state === 'finished') {
        if (e.key === 'Enter' || e.key === ' ') {
            state = 'start';
        }
        return;
    }

    // Playing state
    let dx = 0, dy = 0;
    if (e.key === 'ArrowUp' || e.key === 'w' || e.key === 'W') { dx = 0; dy = -1; }
    else if (e.key === 'ArrowDown' || e.key === 's' || e.key === 'S') { dx = 0; dy = 1; }
    else if (e.key === 'ArrowLeft' || e.key === 'a' || e.key === 'A') { dx = -1; dy = 0; }
    else if (e.key === 'ArrowRight' || e.key === 'd' || e.key === 'D') { dx = 1; dy = 0; }
    else return;

    e.preventDefault();
    
    // Debounce rapid keypresses
    const now = Date.now();
    if (now - moveCooldown < 80) return;
    moveCooldown = now;

    pendingDir = {x: dx, y: dy};
    moveSnake();
}

window.addEventListener('keydown', handleKey);
