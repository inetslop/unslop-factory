// Audio
let audioCtx = null;
function initAudio() {
    if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
}
function playSound(freq, duration, type, vol) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = type || 'square';
    osc.frequency.value = freq;
    gain.gain.setValueAtTime(vol || 0.1, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + duration);
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + duration);
}
function sfxCoin() { playSound(880, 0.15, 'sine', 0.12); playSound(1320, 0.1, 'sine', 0.08); }
function sfxHit() { playSound(200, 0.1, 'square', 0.15); }
function sfxPunch() { playSound(150, 0.08, 'sawtooth', 0.2); }
function sfxDeath() { playSound(100, 0.4, 'sawtooth', 0.3); }
function sfxBuy() { playSound(660, 0.1, 'sine', 0.1); playSound(990, 0.15, 'sine', 0.1); }

// Game State
let gameState = 'start'; // start, playing, dead, shop
let score = 0;
let coins = 0;

let scrollSpeed = 3;

let scrollDist = 0;

// Upgrades
let upgradePunchRange = 0;
let upgradeHealth = 0;
let upgradeCoinMagnet = 0;
let upgradeSpeed = 0;

const UPGRADES = [
    { name: 'Bigger Fist', cost: 50, max: 5, desc: '+30px punch range' },
    { name: 'Extra Life', cost: 100, max: 5, desc: '+1 life' },
    { name: 'Coin Magnet', cost: 75, max: 5, desc: 'Coins fly to you' },
    { name: 'Turbo Scroll', cost: 60, max: 5, desc: 'Faster = more coins' }
];

// Player
let player = { x: 150, y: 0, vy: 0, grounded: true, punchTimer: 0, punchDir: 1, invincible: 0 };
const GROUND_Y_RATIO = 0.75;

// Entities
let coinEntities = [];
let enemyEntities = [];
let particles = [];
let floatingTexts = [];
let clouds = [];
let mountainsFar = [];
let mountainsNear = [];
let groundTiles = [];

function initWorld() {
    clouds = [];
    for (let i = 0; i < 8; i++) {
        clouds.push({ x: Math.random() * W * 1.5, y: Math.random() * H * 0.25 + 20, w: 60 + Math.random() * 80, speed: 0.3 + Math.random() * 0.3 });
    }
    mountainsFar = [];
    for (let i = 0; i < 12; i++) {
        mountainsFar.push({ x: i * (W / 6), h: 80 + Math.random() * 120, w: 150 + Math.random() * 100 });
    }
    mountainsNear = [];
    for (let i = 0; i < 15; i++) {
        mountainsNear.push({ x: i * (W / 8), h: 40 + Math.random() * 60, w: 80 + Math.random() * 60 });
    }
    groundTiles = [];
    for (let i = 0; i < 20; i++) {
        groundTiles.push({ x: i * 80, type: Math.floor(Math.random() * 3) });
    }
}

function resetGame() {
    score = 0;
    coins = 0;
    lives = 3 + upgradeHealth;
    gameTime = 0;
    scrollDist = 0;
    scrollSpeed = baseScrollSpeed + upgradeSpeed * 0.5;
    player.x = W * 0.2;
    player.y = getGroundY() - 40;
    player.vy = 0;
    player.grounded = true;
    player.punchTimer = 0;
    player.invincible = 0;
    coinEntities = [];
    enemyEntities = [];
    particles = [];
    floatingTexts = [];
    initWorld();
}

function getGroundY() { return H * GROUND_Y_RATIO; }

// Spawning
let spawnTimer = 0;
function spawnEntities(dt) {
    spawnTimer -= dt;
    if (spawnTimer <= 0) {
        // Spawn coins
        if (Math.random() < 0.6) {
            const count = 1 + Math.floor(Math.random() * 3);
            for (let i = 0; i < count; i++) {
                coinEntities.push({
                    x: W + 50,
                    y: getGroundY() - 40 - Math.random() * (H * 0.3),
                    r: 12,
                    vy: 0,
                    bobPhase: Math.random() * Math.PI * 2
                });
            }
        }
        // Spawn enemies
        if (Math.random() < 0.4) {
            const type = Math.random() < 0.7 ? 'slime' : 'bat';
            enemyEntities.push({
                x: W + 40,
                y: type === 'slime' ? getGroundY() - 35 : getGroundY() - 80 - Math.random() * 60,
                w: type === 'slime' ? 35 : 30,
                h: type === 'slime' ? 30 : 25,
                type: type,
                hp: type === 'slime' ? 1 : 2,
                vy: type === 'bat' ? (Math.random() - 0.5) * 2 : 0,
                phase: Math.random() * Math.PI * 2,
                hit: false
            });
        }
        spawnTimer = 0.8 + Math.random() * 0.8;
    }
}

function addParticles(x, y, color, count) {
    for (let i = 0; i < count; i++) {
        particles.push({
            x: x, y: y,
            vx: (Math.random() - 0.5) * 6,
            vy: (Math.random() - 0.5) * 6 - 2,
            life: 1,
            color: color,
            r: 3 + Math.random() * 4
        });
    }
}

function addFloatingText(x, y, text, color) {
    floatingTexts.push({ x, y, text, color, life: 1, vy: -2 });
}

// Input

let mouseDown = false;

window.addEventListener('mousemove', e => { mouseX = e.clientX; mouseY = e.clientY; });
window.addEventListener('mousedown', e => {

    if (gameState === 'start') { startGame(); return; }
    if (gameState === 'dead') { showShop(); return; }
    if (gameState === 'shop') { handleShopClick(e.clientX, e.clientY); return; }

    e.preventDefault();
    const t = e.touches[0];
    mouseX = t.clientX; mouseY = t.clientY;
    initAudio();
    if (gameState === 'start') { startGame(); return; }
    if (gameState === 'dead') { showShop(); return; }
    if (gameState === 'shop') { handleShopClick(t.clientX, t.clientY); return; }
    if (gameState === 'playing') { playerPunch(); }
});

function startGame() {
    resetGame();
    gameState = 'playing';
}

function showShop() { gameState = 'shop'; }

let shopButtons = [];
function handleShopClick(mx, my) {
    for (let i = 0; i < shopButtons.length; i++) {
        const b = shopButtons[i];
        if (mx >= b.x && mx <= b.x + b.w && my >= b.y && my <= b.y + b.h) {
            if (i < UPGRADES.length) {
                buyUpgrade(i);
            } else if (i === UPGRADES.length) {
                startGame();
            }
        }
    }
}

function buyUpgrade(idx) {
    const up = UPGRADES[idx];
    const currentLevel = [upgradePunchRange, upgradeHealth, upgradeCoinMagnet, upgradeSpeed][idx];
    if (currentLevel < up.max && coins >= up.cost) {
        coins -= up.cost;
        sfxBuy();
        if (idx === 0) upgradePunchRange++;
        else if (idx === 1) upgradeHealth++;
        else if (idx === 2) upgradeCoinMagnet++;
        else if (idx === 3) upgradeSpeed++;
    }
}

function playerPunch() {
    if (player.punchTimer > 0) return;
    player.punchTimer = 0.4;
    sfxPunch();
    
    const punchRange = 60 + upgradePunchRange * 30;
    const px = player.x + 20;
    const py = player.y - 10;
    
    // Hit enemies
    for (let e of enemyEntities) {
        if (!e.hit && Math.abs(e.x - px) < punchRange && Math.abs(e.y - py) < 60) {
            e.hp--;
            e.hit = true;
            sfxHit();
            addParticles(e.x, e.y, '#ff6b6b', 8);
            if (e.hp <= 0) {
                score += 10;
                coins += 5;
                addFloatingText(e.x, e.y, '+10', '#ffd700');
                addParticles(e.x, e.y, '#ff6b6b', 12);
            } else {
                addFloatingText(e.x, e.y, 'HIT', '#fff');
            }
        }
    }
    
    // Collect coins
    const magnetRange = 40 + upgradeCoinMagnet * 30;
    for (let c of coinEntities) {
        if (Math.abs(c.x - px) < magnetRange && Math.abs(c.y - py) < 80) {
            collectCoin(c);
        }
    }
}

function collectCoin(c) {
    c.collected = true;
    sfxCoin();
    score += 5;
    coins += 1 + upgradeCoinMagnet;
    addParticles(c.x, c.y, '#ffd700', 6);
    addFloatingText(c.x, c.y, '+' + (1 + upgradeCoinMagnet), '#ffd700');
}

// Update
function update(dt) {
    if (gameState !== 'playing') return;
    
    gameTime += dt;
    scrollDist += scrollSpeed * dt * 60;
    
    // Player physics
    const groundY = getGroundY();
    player.vy += 0.5;
    player.y += player.vy;
    if (player.y > groundY - 40) {
        player.y = groundY - 40;
        player.vy = 0;
        player.grounded = true;
    }
    
    if (player.punchTimer > 0) player.punchTimer -= dt;
    if (player.invincible > 0) player.invincible -= dt;
    
    // Spawn
    spawnEntities(dt);
    
    // Update coins
    for (let i = coinEntities.length - 1; i >= 0; i--) {
        const c = coinEntities[i];
        c.x -= scrollSpeed * dt * 60;
        c.bobPhase += dt * 3;
        c.y += Math.sin(c.bobPhase) * 0.5;
        
        // Magnet effect
        if (upgradeCoinMagnet > 0) {
            const dx = player.x + 20 - c.x;
            const dy = player.y - c.y;
            const dist = Math.sqrt(dx*dx + dy*dy);
            if (dist < 150 * upgradeCoinMagnet) {
                c.x += dx * 0.03;
                c.y += dy * 0.03;
            }
        }
        
        // Auto collect if touching player
        const dx = player.x + 20 - c.x;
        const dy = player.y - c.y;
        if (Math.abs(dx) < 30 && Math.abs(dy) < 40) {
            collectCoin(c);
        }
        
        if (c.collected || c.x < -50) coinEntities.splice(i, 1);
    }
    
    // Update enemies
    for (let i = enemyEntities.length - 1; i >= 0; i--) {
        const e = enemyEntities[i];
        e.x -= scrollSpeed * dt * 60 * 0.5;
        if (e.type === 'bat') {
            e.phase += dt * 4;
            e.y += Math.sin(e.phase) * 1.5;
        }
        if (e.hit && !e.dead) {
            e.x -= 3;
            e.vy = -2;
            e.y += e.vy;
            e.vy += 0.3;
            if (e.hp <= 0) { e.dead = true; enemyEntities.splice(i, 1); continue; }
        }
        
        // Hit player
        if (player.invincible <= 0 && !e.hit) {
            const dx = player.x + 20 - e.x;
            const dy = player.y - e.y;
            if (Math.abs(dx) < 35 && Math.abs(dy) < 40) {
                lives--;
                player.invincible = 1.5;
                sfxDeath();
                addParticles(player.x, player.y, '#ff0000', 10);
                if (lives <= 0) {
                    gameState = 'dead';
                    showShop();
                }
            }
        }
        
        if (e.x < -60 || e.dead) enemyEntities.splice(i, 1);
    }
    
    // Update particles
    for (let i = particles.length - 1; i >= 0; i--) {
        const p = particles[i];
        p.x += p.vx;
        p.y += p.vy;
        p.vy += 0.2;
        p.life -= dt * 2;
        if (p.life <= 0) particles.splice(i, 1);
    }
    
    // Update floating texts
    for (let i = floatingTexts.length - 1; i >= 0; i--) {
        const f = floatingTexts[i];
        f.y += f.vy;
        f.life -= dt * 1.5;
        if (f.life <= 0) floatingTexts.splice(i, 1);
    }
    
    // Update clouds
    for (let c of clouds) {
        c.x -= c.speed * scrollSpeed * dt * 60 * 0.3;
        if (c.x < -c.w - 50) { c.x = W + Math.random() * 100; }
    }
    
    // Update mountains
    for (let m of mountainsFar) {
        m.x -= scrollSpeed * dt * 60 * 0.2;
        if (m.x < -m.w) m.x = W + Math.random() * 50;
    }
    for (let m of mountainsNear) {
        m.x -= scrollSpeed * dt * 60 * 0.4;
        if (m.x < -m.w) m.x = W + Math.random() * 50;
    }
    
    // Update ground tiles
    for (let g of groundTiles) {
        g.x -= scrollSpeed * dt * 60;
        if (g.x < -80) { g.x += 20 * 80; g.type = Math.floor(Math.random() * 3); }
    }
    
    // Score over time
    score += dt * 10;
}

// Render
function render() {
    ctx.clearRect(0, 0, W, H);
    
    if (gameState === 'start') {
        drawStartScreen();
        return;
    }
    
    // Sky gradient
    const skyGrad = ctx.createLinearGradient(0, 0, 0, H);
    skyGrad.addColorStop(0, '#87CEEB');
    skyGrad.addColorStop(1, '#e0f0ff');
    ctx.fillStyle = skyGrad;
    ctx.fillRect(0, 0, W, H);
    
    // Clouds
    ctx.fillStyle = 'rgba(255,255,255,0.8)';
    for (let c of clouds) {
        drawCloud(c.x, c.y, c.w);
    }
    
    // Far mountains
    ctx.fillStyle = '#a8d8ea';
    for (let m of mountainsFar) {
        ctx.beginPath();
        ctx.moveTo(m.x, getGroundY());
        ctx.lineTo(m.x + m.w / 2, getGroundY() - m.h);
        ctx.lineTo(m.x + m.w, getGroundY());
        ctx.fill();
    }
    
    // Near mountains
    ctx.fillStyle = '#7ec8a0';
    for (let m of mountainsNear) {
        ctx.beginPath();
        ctx.moveTo(m.x, getGroundY());
        ctx.lineTo(m.x + m.w / 2, getGroundY() - m.h);
        ctx.lineTo(m.x + m.w, getGroundY());
        ctx.fill();
    }
    
    // Ground
    const gy = getGroundY();
    ctx.fillStyle = '#5a8f3c';
    ctx.fillRect(0, gy, W, H - gy);
    ctx.fillStyle = '#4a7f2c';
    for (let g of groundTiles) {
        ctx.fillRect(g.x, gy, 60, 8);
    }
    // Grass tufts
    ctx.strokeStyle = '#3d6b1e';
    ctx.lineWidth = 2;
    for (let i = 0; i < 30; i++) {
        const gx = ((i * 97 + scrollDist * 0.5) % W);
        ctx.beginPath();
        ctx.moveTo(gx, gy);
        ctx.lineTo(gx - 3, gy - 8);
        ctx.stroke();
        ctx.beginPath();
        ctx.moveTo(gx + 5, gy);
        ctx.lineTo(gx + 7, gy - 6);
        ctx.stroke();
    }
    
    // Coins
    for (let c of coinEntities) {
        const bobY = Math.sin(c.bobPhase) * 4;
        ctx.fillStyle = '#ffd700';
        ctx.beginPath();
        ctx.arc(c.x, c.y + bobY, c.r, 0, Math.PI * 2);
        ctx.fill();
        ctx.fillStyle = '#ffaa00';
        ctx.font = 'bold 14px monospace';
        ctx.textAlign = 'center';
        ctx.fillText('$', c.x, c.y + bobY + 5);
    }
    
    // Enemies
    for (let e of enemyEntities) {
        if (e.type === 'slime') {
            const squish = Math.sin(gameTime * 5 + e.phase) * 3;
            ctx.fillStyle = '#66cc33';
            ctx.beginPath();
            ctx.ellipse(e.x, e.y + e.h/2, e.w/2 + squish, e.h/2 - squish, 0, 0, Math.PI * 2);
            ctx.fill();
            // Eyes
            ctx.fillStyle = '#fff';
            ctx.beginPath();
            ctx.arc(e.x - 6, e.y + 5, 5, 0, Math.PI * 2);
            ctx.arc(e.x + 6, e.y + 5, 5, 0, Math.PI * 2);
            ctx.fill();
            ctx.fillStyle = '#000';
            ctx.beginPath();
            ctx.arc(e.x - 6, e.y + 5, 2, 0, Math.PI * 2);
            ctx.arc(e.x + 6, e.y + 5, 2, 0, Math.PI * 2);
            ctx.fill();
        } else {
            const wingY = Math.sin(gameTime * 10) * 8;
            ctx.fillStyle = '#8b4513';
            ctx.beginPath();
            ctx.ellipse(e.x, e.y, 12, 10, 0, 0, Math.PI * 2);
            ctx.fill();
            // Wings
            ctx.strokeStyle = '#654321';
            ctx.lineWidth = 3;
            ctx.beginPath();
            ctx.moveTo(e.x - 10, e.y);
            ctx.lineTo(e.x - 22, e.y - wingY);
            ctx.stroke();
            ctx.beginPath();
            ctx.moveTo(e.x + 10, e.y);
            ctx.lineTo(e.x + 22, e.y + wingY);
            ctx.stroke();
            // Eyes
            ctx.fillStyle = '#ff0';
            ctx.beginPath();
            ctx.arc(e.x - 4, e.y - 3, 3, 0, Math.PI * 2);
            ctx.arc(e.x + 4, e.y - 3, 3, 0, Math.PI * 2);
            ctx.fill();
        }
    }
    
    // Player
    drawPlayer();
    
    // Particles
    for (let p of particles) {
        ctx.globalAlpha = p.life;
        ctx.fillStyle = p.color;
        ctx.beginPath();
        ctx.arc(p.x, p.y, p.r * p.life, 0, Math.PI * 2);
        ctx.fill();
    }
    ctx.globalAlpha = 1;
    
    // Floating texts
    for (let f of floatingTexts) {
        ctx.globalAlpha = f.life;
        ctx.fillStyle = f.color;
        ctx.font = 'bold 18px monospace';
        ctx.textAlign = 'center';
        ctx.fillText(f.text, f.x, f.y);
    }
    ctx.globalAlpha = 1;
    
    // HUD
    drawHUD();
    
    if (gameState === 'shop') drawShop();
}

function drawCloud(x, y, w) {
    ctx.beginPath();
    ctx.arc(x, y, w * 0.3, 0, Math.PI * 2);
    ctx.arc(x + w * 0.25, y - 10, w * 0.25, 0, Math.PI * 2);
    ctx.arc(x + w * 0.5, y, w * 0.3, 0, Math.PI * 2);
    ctx.fill();
}

function drawPlayer() {
    const px = player.x;
    const py = player.y;
    const runFrame = Math.sin(gameTime * 12) * 5;
    
    // Invincible flash
    if (player.invincible > 0 && Math.floor(player.invincible * 10) % 2 === 0) return;
    
    ctx.save();
    // Body
    ctx.fillStyle = '#ff6b35';
    ctx.beginPath();
    ctx.roundRect(px, py, 40, 40, 8);
    ctx.fill();
    
    // Face
    ctx.fillStyle = '#fff';
    ctx.beginPath();
    ctx.arc(px + 28, py + 15, 8, 0, Math.PI * 2);
    ctx.fill();
    ctx.fillStyle = '#333';
    ctx.beginPath();
    ctx.arc(px + 30, py + 15, 3, 0, Math.PI * 2);
    ctx.fill();
    
    // Arm (punch)
    if (player.punchTimer > 0.2) {
        ctx.fillStyle = '#ff6b35';
        ctx.fillRect(px + 35, py + 10, 40, 15);
        ctx.fillStyle = '#ffd700';
        ctx.beginPath();
        ctx.arc(px + 80, py + 17, 12, 0, Math.PI * 2);
        ctx.fill();
    } else {
        ctx.fillStyle = '#ff6b35';
        ctx.fillRect(px + 30, py + 15 + runFrame, 15, 10);
    }
    
    // Legs
    ctx.fillStyle = '#cc4a20';
    ctx.fillRect(px + 5, py + 38, 12, 15 - runFrame);
    ctx.fillRect(px + 25, py + 38, 12, 15 + runFrame);
    
    // Shoes
    ctx.fillStyle = '#fff';
    ctx.fillRect(px + 3, py + 48 - runFrame, 16, 6);
    ctx.fillRect(px + 23, py + 48 + runFrame, 16, 6);
    
    ctx.restore();
}

function drawHUD() {
    // Background bar
    ctx.fillStyle = 'rgba(0,0,0,0.5)';
    ctx.fillRect(0, 0, W, 50);
    
    ctx.fillStyle = '#fff';
    ctx.font = 'bold 16px monospace';
    ctx.textAlign = 'left';
    ctx.fillText(`SCORE: ${Math.floor(score)}`, 20, 32);
    ctx.fillText(`COINS: $${coins}`, 200, 32);
    
    // Lives
    for (let i = 0; i < lives; i++) {
        ctx.fillStyle = '#ff4444';
        ctx.beginPath();
        const hx = W - 40 - i * 30;
        ctx.moveTo(hx, 25);
        ctx.lineTo(hx + 8, 15);
        ctx.lineTo(hx + 16, 25);
        ctx.lineTo(hx + 8, 35);
        ctx.closePath();
        ctx.fill();
    }
    
    // Distance
    ctx.fillStyle = '#aaa';
    ctx.textAlign = 'right';
    ctx.font = '14px monospace';
    ctx.fillText(`${Math.floor(scrollDist / 10)}m`, W - 20, 32);
}

function drawStartScreen() {
    const skyGrad = ctx.createLinearGradient(0, 0, 0, H);
    skyGrad.addColorStop(0, '#1a1a2e');
    skyGrad.addColorStop(1, '#16213e');
    ctx.fillStyle = skyGrad;
    ctx.fillRect(0, 0, W, H);
    
    // Stars
    for (let i = 0; i < 50; i++) {
        const sx = (i * 137 + 50) % W;
        const sy = (i * 211 + 30) % (H * 0.6);

function drawShop() {
    ctx.fillStyle = 'rgba(0,0,0,0.85)';
    ctx.fillRect(0, 0, W, H);
    
    ctx.fillStyle = '#ff4444';
    ctx.font = 'bold 32px monospace';
    ctx.textAlign = 'center';
    ctx.fillText('GAME OVER', W/2, 60);
    
    ctx.fillStyle = '#ffd700';
    ctx.font = '20px monospace';
    ctx.fillText(`Final Score: ${Math.floor(score)}  |  Coins: $${coins}`, W/2, 100);
    
    ctx.fillStyle = '#fff';
    ctx.font = 'bold 24px monospace';
    ctx.fillText('SHOP', W/2, 150);
    
    shopButtons = [];
    const bw = Math.min(300, W - 60);
    const bh = 50;
    const startX = (W - bw) / 2;
    const startY = 180;
    
    for (let i = 0; i < UPGRADES.length; i++) {
        const up = UPGRADES[i];
        const currentLevel = [upgradePunchRange, upgradeHealth, upgradeCoinMagnet, upgradeSpeed][i];
        const by = startY + i * (bh + 15);
        
        shopButtons.push({ x: startX, y: by, w: bw, h: bh });
        
        // Button bg
        ctx.fillStyle = currentLevel >= up.max ? '#2a4a2a' : (coins >= up.cost ? '#3a3a6e' : '#4a4a4a');
        ctx.beginPath();
        ctx.roundRect(startX, by, bw, bh, 8);
        ctx.fill();
        
        ctx.fillStyle = currentLevel >= up.max ? '#6a6' : (coins >= up.cost ? '#fff' : '#888');
        ctx.font = '14px monospace';
        ctx.textAlign = 'left';
        ctx.fillText(`${up.name} (${currentLevel}/${up.max})`, startX + 15, by + 20);
        ctx.font = '12px monospace';
        ctx.fillStyle = currentLevel >= up.max ? '#4a6a4a' : (coins >= up.cost ? '#ffd700' : '#666');
        ctx.fillText(`${up.desc} - $${up.cost}`, startX + 15, by + 38);
    }
    
    // Play Again button
    const py = startY + UPGRADES.length * (bh + 15) + 20;
    shopButtons.push({ x: startX, y: py, w: bw, h: bh });
    ctx.fillStyle = '#2e7d32';
    ctx.beginPath();
    ctx.roundRect(startX, py, bw, bh, 8);
    ctx.fill();
    ctx.fillStyle = '#fff';
    ctx.font = 'bold 18px monospace';
    ctx.textAlign = 'center';
    ctx.fillText('PLAY AGAIN', W/2, py + 30);
}

// Game Loop
let lastTime = 0;
function gameLoop(timestamp) {
    const dt = Math.min((timestamp - lastTime) / 1000, 0.05);
    lastTime = timestamp;
    gameTime += dt;
    
    update(dt);
    render();
    
    requestAnimationFrame(gameLoop);
}
