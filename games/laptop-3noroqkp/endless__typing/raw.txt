let audioCtx = null;
function initAudio() { if (!audioCtx) audioCtx = new (window.AudioContext || webkitAudioContext)(); }
function playType() {
if (!audioCtx) return;
const o = audioCtx.createOscillator();
const g = audioCtx.createGain();
o.type = 'sine'; o.frequency.value = 800 + Math.random()*200;
g.gain.setValueAtTime(0.1, audioCtx.currentTime);
g.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime+0.05);
o.connect(g); g.connect(audioCtx.destination);
o.start(); o.stop(audioCtx.currentTime+0.05);
}
function playError() {
if (!audioCtx) return;
const o = audioCtx.createOscillator();
const g = audioCtx.createGain();
o.type = 'square'; o.frequency.value = 150;
g.gain.setValueAtTime(0.15, audioCtx.currentTime);
g.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime+0.2);
o.connect(g); g.connect(audioCtx.destination);
o.start(); o.stop(audioCtx.currentTime+0.2);
}
function playComplete() {
if (!audioCtx) return;
const notes = [523,659,784];
notes.forEach((f,i)=>{
const o=audioCtx.createOscillator();const g=audioCtx.createGain();
o.type='sine';o.frequency.value=f;
g.gain.setValueAtTime(0.08,audioCtx.currentTime+i*0.05);
g.gain.exponentialRampToValueAtTime(0.001,audioCtx.currentTime+i*0.05+0.15);

function showScreen(id) {

});

function updateHighDisplay() {
document.getElementById('hs-display').textContent = highScore > 0 ? 'HIGH SCORE: '+highScore : '';
}

function getWord() {
const diff = Math.min(wordIndex, WORDS.length-1);
const pool = WORDS.slice(0, diff+20);
return pool[Math.floor(Math.random()*pool.length)];
}

function nextWord() {
currentWord = getWord();
typed = '';
wordIndex++;
timerMax = Math.max(2000, 5000 - wordIndex * 100);
timerStart = performance.now();
renderWord();
updateHUD();
startTimer();
}

function renderWord() {
const el = document.getElementById('word-display');
let html = '';
for (let i=0;i<currentWord.length;i++){
let cls='word-char';
if(i<typed.length){
cls += typed[i]===currentWord[i]?' correct':' incorrect';
} else if(i===typed.length){
cls+=' current';
}
html+=`<span class="${cls}">${currentWord[i]}</span>`;
}
el.innerHTML=html;
document.getElementById('input-display').textContent = typed.replace(/ /g,'·');
}

function updateHUD() {
document.getElementById('score-val').textContent = score;
document.getElementById('combo-val').textContent = combo;
let hearts='';
for(let i=0;i<3;i++) hearts+= i<lives?'♥':'♡';
document.getElementById('lives-val').textContent = hearts;
}

function startTimer() {
cancelAnimationFrame(timerRAF);
timerStart = performance.now();
const bar = document.getElementById('timer-bar');
function tick(now){
if(state!=='play') return;
const elapsed = now - timerStart;
const remaining = Math.max(0, 1 - elapsed/timerMax);
bar.style.width = (remaining*100)+'%';
if(remaining < 0.3) bar.classList.add('danger'); else bar.classList.remove('danger');
if(elapsed >= timerMax){ onTimeout(); return; }
timerRAF = requestAnimationFrame(tick);
}
timerRAF = requestAnimationFrame(tick);
}

function onTimeout() {
playError();
lives--;
combo=1;
updateHUD();
spawnParticles('red');
if(lives<=0) gameOver(); else nextWord();
}

function onKey(e) {
if(state==='start' && (e.key==='Enter')){ initAudio(); startGame(); return; }
if(state==='gameover' && (e.key==='Enter')){ startGame(); return; }
if(state!=='play') return;
if(e.key===' '){ e.preventDefault(); }
const ch = e.key.length===1 ? e.key : '';
if(!ch) return;
typed += ch;
playType();
renderWord();
if(typed === currentWord){
onComplete();
} else if(typed.length > currentWord.length || !currentWord.startsWith(typed)){
onError();
}
}

function onComplete() {
playComplete();
score += currentWord.length * combo;
combo = Math.min(combo+1, 9);
updateHUD();
spawnParticles('green');
nextWord();
}

function onError() {
playError();
lives--;
combo=1;
typed='';
updateHUD();
spawnParticles('red');
if(lives<=0){ gameOver(); } else { renderWord(); }
}

function spawnParticles(color) {
const cont = document.getElementById('particles');
for(let i=0;i<8;i++){
const p=document.createElement('div');
p.className='particle';
p.style.background = color==='green'?'#00ff88':'#ff4444';
p.style.left='50%';p.style.top='50%';
const angle=Math.random()*Math.PI*2;
const dist=50+Math.random()*80;
p.style.setProperty('--px', Math.cos(angle)*dist+'px');
p.style.setProperty('--py', Math.sin(angle)*dist+'px');
cont.appendChild(p);
setTimeout(()=>p.remove(),600);
}
}

function startGame() {
state='play';
score=0; lives=3; combo=1; wordIndex=0;
showScreen('play-screen');
nextWord();
}

document.getElementById('gameover-score').textContent = 'SCORE: '+score;
document.getElementById('gameover-high').textContent = highScore===score ? '★ NEW HIGH SCORE! ★' : 'HIGH: '+highScore;
updateHighDisplay();
showScreen('gameover-screen');
}

document.getElementById('restart-btn').addEventListener('click', startGame);
