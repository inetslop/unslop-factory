```html
<!DOCTYPE html>
<html>
<head>
<meta charset="UTF-8">
<title>Side Scroller Snake</title>
<style>
* { margin: 0; padding: 0; box-sizing: border-box; }
body { overflow: hidden; background: #0f0f23; }
canvas { display: block; width: 100vw; height: 100vh; }
</style>
</head>
<body>
<canvas id="game"></canvas>
<script>
const canvas = document.getElementById('game');
const ctx = canvas.getContext('2d');

let W, H;
function resize() {
    W = canvas.width = window.innerWidth;
    H = canvas.height = window.innerHeight;
}
window.addEventListener('resize', resize);
resize();

// Audio
let audioCtx = null;
function initAudio() {
    if (!audioCtx) audioCtx = new (window.AudioContext || webkitAudioContext)();
}
function playSound(freq, dur, type, vol) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = type || 'sine';
    osc.frequency.value = freq;
    gain.gain.setValueAtTime(vol || 0.3, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + dur);
}
function eatSound() { playSound(880, 0.15, 'sine', 0.4); setTimeout(() => playSound(1320, 0.1, 'sine', 0.3), 60); }
function hitSound() { playSound(150, 0.3, 'sawtooth', 0.5); }
function scoreSound() { playSound(660, 0.1, 'triangle', 0.2); }

// Game state
let state = 'START'; // START, PLAYING, GAMEOVER
let score = 0;
let highScore = 0;
try { highScore = parseInt(localStorage.getItem('sideScrollSnakeHigh') || '0'); } catch(e) {}
let gameTime = 0;
let speed = 4;
let baseSpeed = 4;

// Snake
const SNAKE_HEAD_X_RATIO = 0.18;
const SEGMENT_COUNT = 20;
const SEGMENT_SPACING = 12;
let snakeY, snakeVelY;
let segments = [];
let snakeRadius = 14;

// Keys
let keys = {};
window.addEventListener('keydown', e => { keys[e.key] = true; if (state === 'START' && (e.key === ' ' || e.key === 'Enter')) startGame(); else if (state === 'GAMEOVER' && (e.key === ' ' || e.key === 'Enter')) startGame(); initAudio(); });
window.addEventListener('keyup', e => { keys[e.key] = false; });

// Obstacles
let obstacles = [];
let foods = [];
let particles = [];
let clouds = [];
let stars = [];

function initBackground() {
    stars = [];
    for (let i = 0; i < 80; i++) {
        stars.push({ x: Math.random() * W, y: Math.random() * H * 0.7, size: Math.random() * 2 + 0.5, speed: Math.random() * 0.3 + 0.1 });
    }
    clouds = [];
    for (let i = 0; i < 5; i++) {
        clouds.push({ x: Math.random() * W, y: Math.random() * H * 0.4, w: 80 + Math.random() * 120, speed: 0.3 + Math.random() * 0.3, alpha: 0.05 + Math.random() * 0.05 });
    }
}

function startGame() {
    state = 'PLAYING';
    score = 0;
    gameTime = 0;
    speed = baseSpeed;
    snakeY = H / 2;
    snakeVelY = 0;
    segments = [];
    const headX = W * SNAKE_HEAD_X_RATIO;
    for (let i = 0; i < SEGMENT_COUNT; i++) {
        segments.push({ x: headX - i * SEGMENT_SPACING, y: H / 2 });
    }
    obstacles = [];
    foods = [];
    particles = [];
    initBackground();
    // Initial food
    spawnFood();
}

function spawnObstacle() {
    const type = Math.random();
    if (type < 0.5) {
        // Ground spike
        const h = 40 + Math.random() * 60;
        obstacles.push({ x: W + 50, y: H - h, w: 25, h: h, type: 'spike' });
    } else if (type < 0.75) {
        // Ceiling spike
        const h = 40 + Math.random() * 60;
        obstacles.push({ x: W + 50, y: 0, w: 25, h: h, type: 'spike' });
    } else if (type < 0.9) {
        // Floating block
        const y = H * 0.2 + Math.random() * H * 0.5;
        obstacles.push({ x: W + 50, y: y, w: 40, h: 35, type: 'block' });
    } else {
        // Double spike (gap in middle)
        const gapY = H * 0.25 + Math.random() * H * 0.5;
        const gapH = 100 + Math.random() * 40;
        obstacles.push({ x: W + 50, y: 0, w: 30, h: gapY - gapH / 2, type: 'spike' });
        obstacles.push({ x: W + 50, y: gapY + gapH / 2, w: 30, h: H - (gapY + gapH / 2), type: 'spike' });
    }
}

function spawnFood() {
    const y = 60 + Math.random() * (H - 120);
    foods.push({ x: W + 50, y: y, r: 12, pulse: 0 });
}

function spawnParticles(x, y, color) {
    for (let i = 0; i < 8; i++) {
        const angle = Math.random() * Math.PI * 2;
        const spd = 2 + Math.random() * 4;
        particles.push({ x: x, y: y, vx: Math.cos(angle) * spd, vy: Math.sin(angle) * spd, life: 1, color: color });
    }
}

let obstacleTimer = 0;
let foodTimer = 0;

function update(dt) {
    if (state !== 'PLAYING') return;
    
    gameTime += dt;
    speed = baseSpeed + gameTime * 0.3;
    if (speed > 12) speed = 12;
    
    // Snake movement
    const accel = 0.8;
    const maxVel = 7;
    if (keys['ArrowUp'] || keys['w'] || keys['W']) {
        snakeVelY -= accel;
    }
    if (keys['ArrowDown'] || keys['s'] || keys['S']) {
        snakeVelY += accel;
    }
    snakeVelY *= 0.85; // damping
    if (snakeVelY > maxVel) snakeVelY = maxVel;
    if (snakeVelY < -maxVel) snakeVelY = -maxVel;
    
    const headX = W * SNAKE_HEAD_X_RATIO;
    segments[0].x = headX;
    segments[0].y += snakeVelY;
    segments[0].y = Math.max(snakeRadius, Math.min(H - snakeRadius, segments[0].y));
    
    // Body follow
    for (let i = 1; i < SEGMENT_COUNT; i++) {
        const prev = segments[i - 1];
        const curr = segments[i];
        const dx = prev.x - curr.x;
        const dy = prev.y - curr.y;
        const dist = Math.sqrt(dx * dx + dy * dy);
        if (dist > SEGMENT_SPACING) {
            const ratio = SEGMENT_SPACING / dist;
            curr.x = prev.x - dx * ratio;
            curr.y = prev.y - dy * ratio;
        }
    }
    
    // Spawn obstacles
    obstacleTimer -= dt;
    if (obstacleTimer <= 0) {
        spawnObstacle();
        obstacleTimer = Math.max(0.8, 2.5 - gameTime * 0.02);
    }
    
    // Spawn food
    foodTimer -= dt;
    if (foodTimer <= 0) {
        spawnFood();
        foodTimer = 1.5 + Math.random() * 1.5;
    }
    
    // Move obstacles
    for (let i = obstacles.length - 1; i >= 0; i--) {
        obstacles[i].x -= speed;
        if (obstacles[i].x + obstacles[i].w < -50) obstacles.splice(i, 1);
    }
    
    // Move foods
    for (let i = foods.length - 1; i >= 0; i--) {
        foods[i].x -= speed;
        foods[i].pulse += dt * 5;
        if (foods[i].x < -50) foods.splice(i, 1);
    }
    
    // Move particles
    for (let i = particles.length - 1; i >= 0; i--) {
        particles[i].x += particles[i].vx - speed * 0.5;
        particles[i].y += particles[i].vy;
        particles[i].life -= dt * 2;
        if (particles[i].life <= 0) particles.splice(i, 1);
    }
    
    // Move background
    for (let s of stars) { s.x -= speed * s.speed; if (s.x < -5) { s.x = W + 5; s.y = Math.random() * H * 0.7; } }
    for (let c of clouds) { c.x -= speed * c.speed; if (c.x + c.w < -10) { c.x = W + 50; c.y = Math.random() * H * 0.4; } }
    
    // Collision: food
    const headY = segments[0].y;
    for (let i = foods.length - 1; i >= 0; i--) {
        const f = foods[i];
        const dx = headX - f.x;
        const dy = headY - f.y;
        if (Math.sqrt(dx * dx + dy * dy) < snakeRadius + f.r) {
            score++;
            eatSound();
            spawnParticles(f.x, f.y, '#4cff4c');
            foods.splice(i, 1);
            if (score > highScore) {
                highScore = score;
                try { localStorage.setItem('sideScrollSnakeHigh', highScore.toString()); } catch(e) {}
            }
        }
    }
    
    // Collision: obstacles
    for (let obs of obstacles) {
        const dx = headX - obs.x;
        const dy = headY - obs.y;
        
        if (obs.type === 'spike') {
            // Check if head is within the spike area
            if (headX > obs.x - 5 && headX < obs.x + obs.w + 5) {
                if (obs.y < H / 2) {
                    // Top spike - check if snake is in top part
                    if (headY < obs.y + obs.h - 10) {
                        gameOver();
                        return;
                    }
                } else {
                    // Bottom spike
                    if (headY > obs.y + 10) {
                        gameOver();
                        return;
                    }
                }
            }
        } else if (obs.type === 'block') {
            const closestX = Math.max(obs.x, Math.min(headX, obs.x + obs.w));
            const closestY = Math.max(obs.y, Math.min(headY, obs.y + obs.h));
            const ddx = headX - closestX;
            const ddy = headY - closestY;
            if (ddx * ddx + ddy * ddy < (snakeRadius - 4) * (snakeRadius - 4)) {
                gameOver();
                return;
            }
        }
    }
    
    // Wall collision (top/bottom)
    if (headY <= snakeRadius || headY >= H - snakeRadius) {
        // Just clamp, don't kill (more forgiving)
    }
}

function gameOver() {
    state = 'GAMEOVER';
    hitSound();
    spawnParticles(W * SNAKE_HEAD_X_RATIO, segments[0].y, '#ff4444');
    for (let i = 1; i < SEGMENT_COUNT; i++) {
        spawnParticles(segments[i].x, segments[i].y, '#ff6644');
    }
}

function draw() {
    // Background gradient
    const bg = ctx.createLinearGradient(0, 0, 0, H);
    bg.addColorStop(0, '#0f0f23');
    bg.addColorStop(1, '#1a1a3e');
    ctx.fillStyle = bg;
    ctx.fillRect(0, 0, W, H);
    
    // Stars
    ctx.fillStyle = 'rgba(255,255,255,0.6)';
    for (let s of stars) {
        ctx.beginPath();
        ctx.arc(s.x, s.y, s.size, 0, Math.PI * 2);
        ctx.fill();
    }
    
    // Clouds
    for (let c of clouds) {
        ctx.fillStyle = `rgba(100,120,180,${c.alpha})`;
        ctx.beginPath();
        ctx.ellipse(c.x + c.w / 2, c.y, c.w / 2, c.w * 0.3, 0, 0, Math.PI * 2);
        ctx.fill();
    }
    
    if (state === 'PLAYING' || state === 'GAMEOVER') {
        // Obstacles
        for (let obs of obstacles) {
            if (obs.type === 'spike') {
                const isTop = obs.y < 10;
                ctx.fillStyle = '#e74c3c';
                ctx.beginPath();
                if (isTop) {
                    ctx.moveTo(obs.x, obs.y);
                    ctx.lineTo(obs.x + obs.w, obs.y);
                    ctx.lineTo(obs.x + obs.w / 2, obs.y + obs.h);
                } else {
                    ctx.moveTo(obs.x, obs.y + obs.h);
                    ctx.lineTo(obs.x + obs.w, obs.y + obs.h);
                    ctx.lineTo(obs.x + obs.w / 2, obs.y);
                }
                ctx.fill();
            } else if (obs.type === 'block') {
                ctx.fillStyle = '#e74c3c';
                ctx.fillRect(obs.x, obs.y, obs.w, obs.h);
                ctx.strokeStyle = '#c0392b';
                ctx.lineWidth = 2;
                ctx.strokeRect(obs.x, obs.y, obs.w, obs.h);
            }
        }
        
        // Foods
        for (let f of foods) {
            const pulse = 1 + Math.sin(f.pulse) * 0.15;
            ctx.fillStyle = '#4cff4c';
            ctx.beginPath();
            ctx.arc(f.x, f.y, f.r * pulse, 0, Math.PI * 2);
            ctx.fill();
            // Stem
            ctx.strokeStyle = '#2d8a2d';
            ctx.lineWidth = 3;
            ctx.beginPath();
            ctx.moveTo(f.x, f.y - f.r * pulse);
            ctx.lineTo(f.x + 3, f.y - f.r * pulse - 8);
            ctx.stroke();
        }
        
        // Snake body
        for (let i = SEGMENT_COUNT - 1; i >= 0; i--) {
            const seg = segments[i];
            const t = i / SEGMENT_COUNT;
            const r = Math.max(4, snakeRadius * (1 - t * 0.5));
            const hue = 120 + t * 30;
            ctx.fillStyle = `hsl(${hue}, 70%, ${45 - t * 10}%)`;
            ctx.beginPath();
            ctx.arc(seg.x, seg.y, r, 0, Math.PI * 2);
            ctx.fill();
            
            if (i === 0) {
                // Eyes
                const eyeOffY = snakeVelY > 1 ? -3 : snakeVelY < -1 ? 3 : 0;
                ctx.fillStyle = '#fff';
                ctx.beginPath();
                ctx.arc(seg.x + 4, seg.y - 5 + eyeOffY, 4, 0, Math.PI * 2);
                ctx.fill();
                ctx.fillStyle = '#000';
                ctx.beginPath();
                ctx.arc(seg.x + 5, seg.y - 5 + eyeOffY, 2, 0, Math.PI * 2);
                ctx.fill();
            }
        }
        
        // Particles
        for (let p of particles) {
            ctx.globalAlpha = p.life;
            ctx.fillStyle = p.color;
            ctx.beginPath();
            ctx.arc(p.x, p.y, 4 * p.life, 0, Math.PI * 2);
            ctx.fill();
        }
        ctx.globalAlpha = 1;
    }
    
    // HUD
    if (state === 'PLAYING') {
        ctx.fillStyle = '#fff';
        ctx.font = 'bold 24px monospace';
        ctx.textAlign = 'left';
        ctx.fillText(`Score: ${score}`, 20, 35);
        ctx.font = '16px monospace';
        ctx.fillStyle = 'rgba(255,255,255,0.5)';
        ctx.fillText(`Best: ${highScore}`, 20, 58);
    }
    
    // Start screen
    if (state === 'START') {
        ctx.fillStyle = 'rgba(0,0,0,0.7)';
        ctx.fillRect(0, 0, W, H);
        
        ctx.textAlign = 'center';
        ctx.fillStyle = '#4cff4c';
        ctx.font = `bold ${Math.min(W * 0.06, 60)}px monospace`;
        ctx.fillText('SIDE SCROLLER SNAKE', W / 2, H * 0.3);
        
        ctx.fillStyle = '#fff';
        ctx.font = `${Math.min(W * 0.025, 22)}px monospace`;
        ctx.fillText('Guide your snake to eat food and avoid obstacles!', W / 2, H * 0.42);
        
        ctx.fillStyle = 'rgba(255,255,255,0.7)';
        ctx.font = `${Math.min(W * 0.02, 18)}px monospace`;
        ctx.fillText('Arrow Keys / W S — Move Up & Down', W / 2, H * 0.52);
        ctx.fillText('Speed increases over time!', W / 2, H * 0.57);
        
        ctx.fillStyle = '#4cff4c';
        ctx.font = `bold ${Math.min(W * 0.03, 28)}px monospace`;
        const blink = Math.sin(Date.now() * 0.005) > 0;
        if (blink) ctx.fillText('Press SPACE or ENTER to Start', W / 2, H * 0.7);
        
        if (highScore > 0) {
            ctx.fillStyle = 'rgba(255,255,255,0.6)';
            ctx.font = `${Math.min(W * 0.02, 18)}px monospace`;
            ctx.fillText(`High Score: ${highScore}`, W / 2, H * 0.78);
        }
        
        // Draw decorative snake
        const t = Date.now() * 0.003;
        for (let i = 0; i < 15; i++) {
            const sx = W / 2 - 100 + i * 14;
            const sy = H * 0.85 + Math.sin(t + i * 0.4) * 10;
            ctx.fillStyle = `hsl(${120 + i * 3}, 70%, ${45 - i}%)`;
            ctx.beginPath();
            ctx.arc(sx, sy, 10 - i * 0.3, 0, Math.PI * 2);
            ctx.fill();
        }
    }
    
    // Game Over screen
    if (state === 'GAMEOVER') {
        ctx.fillStyle = 'rgba(0,0,0,0.6)';
        ctx.fillRect(0, 0, W, H);
        
        ctx.textAlign = 'center';
        ctx.fillStyle = '#ff4444';
        ctx.font = `bold ${Math.min(W * 0.05, 50)}px monospace`;
        ctx.fillText('GAME OVER', W / 2, H * 0.35);
        
        ctx.fillStyle = '#fff';
        ctx.font = `${Math.min(W * 0.03, 28)}px monospace`;
        ctx.fillText(`Score: ${score}`, W / 2, H * 0.47);
        ctx.fillText(`Best: ${highScore}`, W / 2, H * 0.53);
        
        ctx.fillStyle = '#4cff4c';
        ctx.font = `bold ${Math.min(W * 0.025, 22)}px monospace`;
        const blink = Math.sin(Date.now() * 0.005) > 0;
        if (blink) ctx.fillText('Press SPACE or ENTER to Restart', W / 2, H * 0.68);
    }
}

let lastTime = 0;
function loop(time) {
    const dt = Math.min((time - lastTime) / 1000, 0.05);
    lastTime = time;
    
    update(dt);
    draw();
    
    requestAnimationFrame(loop);
}

initBackground();
requestAnimationFrame(loop);
</script>
</body>
</html>
```
