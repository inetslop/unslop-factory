const canvas = document.getElementById('game');
const ctx = canvas.getContext('2d');

let W, H;
function resize() {
    W = window.innerWidth;
    H = window.innerHeight;
    canvas.width = W;
    canvas.height = H;
}
window.addEventListener('resize', resize);
resize();

// Audio
let audioCtx = null;
function initAudio() {
    if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
}
function playSound(freq, duration, type, vol) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = type || 'square';
    osc.frequency.value = freq;
    gain.gain.setValueAtTime(vol || 0.1, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + duration);
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + duration);
}

// Game state
let gameState = 'start'; // start, playing, gameover
let score = 0;
let balls = 3;
let tableX, tableY, tableW, tableH;
let ball = null;
let flippers = [];
let asteroids = [];
let particles = [];
let stars = [];
let plungerActive = false;
let plungerAngle = 0;
let keys = {};
let lastTime = 0;

// Generate stars
function genStars() {
    stars = [];
    for (let i = 0; i < 150; i++) {
        stars.push({ x: Math.random() * W, y: Math.random() * H, size: Math.random() * 2 + 0.5, speed: Math.random() * 0.3 + 0.1 });
    }
}

function initTable() {
    tableW = Math.min(W * 0.6, 400);
    tableH = H * 0.85;
    tableX = (W - tableW) / 2;
    tableY = H * 0.075;

    // Flippers
    const flipperLen = tableW * 0.32;
    const gapW = tableW * 0.18;
    const flipperY = tableY + tableH - 40;
    flippers = [
        { x: tableX + gapW, y: flipperY, len: flipperLen, angle: 0.35, restAngle: 0.35, hitAngle: -0.6, active: false, side: 'L' },
        { x: tableX + tableW - gapW, y: flipperY, len: flipperLen, angle: Math.PI - 0.35, restAngle: Math.PI - 0.35, hitAngle: Math.PI + 0.6, active: false, side: 'R' }
    ];

    // Asteroids - hexagonal shapes at various positions
    asteroids = [];
    const positions = [
        { x: tableX + tableW * 0.35, y: tableY + tableH * 0.2 },
        { x: tableX + tableW * 0.65, y: tableY + tableH * 0.2 },
        { x: tableX + tableW * 0.25, y: tableY + tableH * 0.38 },
        { x: tableX + tableW * 0.75, y: tableY + tableH * 0.38 },
        { x: tableX + tableW * 0.5, y: tableY + tableH * 0.3 },
        { x: tableX + tableW * 0.4, y: tableY + tableH * 0.52 },
        { x: tableX + tableW * 0.6, y: tableY + tableH * 0.52 },
        { x: tableX + tableW * 0.3, y: tableY + tableH * 0.68 },
        { x: tableX + tableW * 0.7, y: tableY + tableH * 0.68 },
    ];
    for (let p of positions) {
        asteroids.push({
            x: p.x, y: p.y,
            radius: Math.min(tableW * 0.12, 35),
            hp: 3, maxHp: 3,
            rot: Math.random() * Math.PI * 2,
            rotSpeed: (Math.random() - 0.5) * 0.02,
            alive: true
        });
    }

    // Ball start position at plunger
    ball = { x: tableX + tableW * 0.82, y: tableY + tableH - 60, vx: 0, vy: 0, radius: Math.min(tableW * 0.045, 12), active: false };

    particles = [];
}

function launchBall() {
    if (!ball.active) {
        ball.active = true;
        ball.vx = (Math.random() - 0.5) * 2;
        ball.vy = -14;
        plungerActive = false;
        playSound(200, 0.1, 'sine', 0.15);
    }
}

function spawnParticles(x, y, color, count) {
    for (let i = 0; i < count; i++) {
        const angle = Math.random() * Math.PI * 2;
        const speed = Math.random() * 4 + 1;
        particles.push({ x, y, vx: Math.cos(angle) * speed, vy: Math.sin(angle) * speed, life: 1, color, size: Math.random() * 3 + 1 });
    }
}

function updateBall(dt) {
    if (!ball.active) return;

    // Gravity
    ball.vy += 0.25;
    ball.x += ball.vx;
    ball.y += ball.vy;

    // Table walls
    const leftWall = tableX + 15;
    const rightWall = tableX + tableW - 15;
    const topWall = tableY + 15;

    if (ball.x - ball.radius < leftWall) { ball.x = leftWall + ball.radius; ball.vx = Math.abs(ball.vx) * 0.8; playSound(300, 0.05, 'square', 0.05); }
    if (ball.x + ball.radius > rightWall) { ball.x = rightWall - ball.radius; ball.vx = -Math.abs(ball.vx) * 0.8; playSound(300, 0.05, 'square', 0.05); }
    if (ball.y - ball.radius < topWall) { ball.y = topWall + ball.radius; ball.vy = Math.abs(ball.vy) * 0.8; playSound(300, 0.05, 'square', 0.05); }

    // Tapered walls at top
    const taperStart = tableY + tableH * 0.15;
    if (ball.y < taperStart) {
        const progress = (taperStart - ball.y) / (taperStart - topWall);
        const taperLeft = leftWall + progress * tableW * 0.2;
        const taperRight = rightWall - progress * tableW * 0.2;
        if (ball.x - ball.radius < taperLeft) { ball.x = taperLeft + ball.radius; ball.vx = Math.abs(ball.vx) * 0.8; }
        if (ball.x + ball.radius > taperRight) { ball.x = taperRight - ball.radius; ball.vx = -Math.abs(ball.vx) * 0.8; }
    }

    // Asteroid collisions
    for (let a of asteroids) {
        if (!a.alive) continue;
        const dx = ball.x - a.x;
        const dy = ball.y - a.y;
        const dist = Math.sqrt(dx * dx + dy * dy);
        if (dist < ball.radius + a.radius) {
            // Bounce
            const nx = dx / dist, ny = dy / dist;
            ball.vx = nx * 5 + (Math.random() - 0.5) * 2;
            ball.vy = ny * 5 - 2;
            a.hp--;
            playSound(150 + Math.random() * 100, 0.15, 'sawtooth', 0.1);
            spawnParticles(a.x, a.y, '#ff6633', 8);
            if (a.hp <= 0) {
                a.alive = false;
                score += 100 * a.maxHp;
                spawnParticles(a.x, a.y, '#ffaa00', 20);
                playSound(80, 0.3, 'sawtooth', 0.2);
            }
        }
    }

    // Flipper collisions
    for (let f of flippers) {
        const tipX = f.x + Math.cos(f.angle) * f.len;
        const tipY = f.y + Math.sin(f.angle) * f.len;
        // Line segment collision
        const fx = ball.x - f.x, fy = ball.y - f.y;
        const tx = tipX - f.x, ty = tipY - f.y;
        const len2 = tx * tx + ty * ty;
        let t = (fx * tx + fy * ty) / len2;
        t = Math.max(0, Math.min(1, t));
        const cx = f.x + t * tx, cy = f.y + t * ty;
        const dx = ball.x - cx, dy = ball.y - cy;
        const dist = Math.sqrt(dx * dx + dy * dy);
        if (dist < ball.radius + 8) {
            const nx = dx / dist, ny = dy / dist;
            const bounceSpeed = f.active ? 12 : 6;
            ball.vx = nx * bounceSpeed;
            ball.vy = ny * bounceSpeed - 3;
            playSound(400, 0.05, 'square', 0.08);
        }
    }

    // Ball lost
    if (ball.y > tableY + tableH + 50) {
        ball.active = false;
        balls--;
        playSound(100, 0.4, 'sawtooth', 0.15);
        if (balls <= 0) {
            gameState = 'gameover';
        } else {
            ball.x = tableX + tableW * 0.82;
            ball.y = tableY + tableH - 60;
            ball.vx = 0;
            ball.vy = 0;
        }
    }

    // Speed cap
    const speed = Math.sqrt(ball.vx * ball.vx + ball.vy * ball.vy);
    if (speed > 15) {
        ball.vx = (ball.vx / speed) * 15;
        ball.vy = (ball.vy / speed) * 15;
    }
}

function updateFlippers() {
    for (let f of flippers) {
        const targetAngle = f.active ? f.hitAngle : f.restAngle;
        const diff = targetAngle - f.angle;
        if (Math.abs(diff) > 0.01) {
            f.angle += diff * 0.4;
        } else {
            f.angle = targetAngle;
        }
    }
}

function updateParticles() {
    for (let i = particles.length - 1; i >= 0; i--) {
        const p = particles[i];
        p.x += p.vx;
        p.y += p.vy;
        p.vy += 0.1;
        p.life -= 0.02;
        if (p.life <= 0) particles.splice(i, 1);
    }
}

function updateAsteroids() {
    for (let a of asteroids) {
        if (!a.alive) continue;
        a.rot += a.rotSpeed;
    }
}

function drawStars() {
    ctx.fillStyle = '#0a0a1a';
    ctx.fillRect(0, 0, W, H);
    for (let s of stars) {
        s.y += s.speed;
        if (s.y > H) { s.y = 0; s.x = Math.random() * W; }
        ctx.fillStyle = `rgba(255,255,255,${0.3 + s.size * 0.3})`;
        ctx.beginPath();
        ctx.arc(s.x, s.y, s.size * 0.5, 0, Math.PI * 2);
        ctx.fill();
    }
}

function drawTable() {
    // Table background
    ctx.fillStyle = 'rgba(10, 15, 40, 0.8)';
    ctx.fillRect(tableX, tableY, tableW, tableH);

    // Walls with glow
    ctx.strokeStyle = '#00ccff';
    ctx.lineWidth = 3;
    ctx.shadowColor = '#00ccff';
    ctx.shadowBlur = 15;
    ctx.strokeRect(tableX + 5, tableY + 5, tableW - 10, tableH - 10);

    // Tapered top walls
    const taperStart = tableY + tableH * 0.15;
    ctx.beginPath();
    ctx.moveTo(tableX + 15, taperStart);
    ctx.lineTo(tableX + tableW * 0.3, tableY + 15);
    ctx.stroke();
    ctx.beginPath();
    ctx.moveTo(tableX + tableW - 15, taperStart);
    ctx.lineTo(tableX + tableW * 0.7, tableY + 15);
    ctx.stroke();

    // Plunger area
    const plungerX = tableX + tableW * 0.82;
    const plungerY = tableY + tableH - 30;
    ctx.strokeStyle = '#ff6600';
    ctx.shadowColor = '#ff6600';
    ctx.lineWidth = 4;
    ctx.beginPath();
    ctx.moveTo(plungerX, tableY + tableH - 5);
    ctx.lineTo(plungerX, plungerY);
    ctx.stroke();

    // Plunger spring
    ctx.strokeStyle = '#ffaa00';
    ctx.lineWidth = 2;
    for (let i = 0; i < 4; i++) {
        const sy = plungerY + i * 8;
        ctx.beginPath();
        ctx.moveTo(plungerX - 5, sy);
        ctx.lineTo(plungerX + 5, sy + 4);
        ctx.stroke();
    }

    ctx.shadowBlur = 0;
}

function drawAsteroid(a) {
    if (!a.alive) return;
    ctx.save();
    ctx.translate(a.x, a.y);
    ctx.rotate(a.rot);

    const glowColor = a.hp === 3 ? '#ff4444' : a.hp === 2 ? '#ff8800' : '#ffff00';
    ctx.strokeStyle = glowColor;
    ctx.shadowColor = glowColor;
    ctx.shadowBlur = 10;
    ctx.lineWidth = 2.5;

    // Hexagon-ish asteroid shape
    ctx.beginPath();
    const sides = 7;
    for (let i = 0; i <= sides; i++) {
        const angle = (i / sides) * Math.PI * 2;
        const r = a.radius * (0.8 + Math.sin(i * 3.7) * 0.15);
        const px = Math.cos(angle) * r;
        const py = Math.sin(angle) * r;
        if (i === 0) ctx.moveTo(px, py);
        else ctx.lineTo(px, py);
    }
    ctx.closePath();
    ctx.stroke();

    // Inner detail
    ctx.strokeStyle = glowColor;
    ctx.globalAlpha = 0.5;
    ctx.beginPath();
    for (let i = 0; i <= sides; i++) {
        const angle = (i / sides) * Math.PI * 2 + 0.3;
        const r = a.radius * 0.5 * (0.8 + Math.sin(i * 2.3) * 0.1);
        const px = Math.cos(angle) * r;
        const py = Math.sin(angle) * r;
        if (i === 0) ctx.moveTo(px, py);
        else ctx.lineTo(px, py);
    }
    ctx.closePath();
    ctx.stroke();

    ctx.restore();
}

function drawFlippers() {
    for (let f of flippers) {
        const tipX = f.x + Math.cos(f.angle) * f.len;
        const tipY = f.y + Math.sin(f.angle) * f.len;

        ctx.strokeStyle = '#ff3366';
        ctx.shadowColor = '#ff3366';
        ctx.shadowBlur = 12;
        ctx.lineWidth = 8;
        ctx.lineCap = 'round';
        ctx.beginPath();
        ctx.moveTo(f.x, f.y);
        ctx.lineTo(tipX, tipY);
        ctx.stroke();

        // Pivot circle
        ctx.fillStyle = '#ff3366';
        ctx.beginPath();
        ctx.arc(f.x, f.y, 8, 0, Math.PI * 2);
        ctx.fill();

        ctx.shadowBlur = 0;
    }
}

function drawBall() {
    if (!ball.active && gameState !== 'playing') return;
    ctx.fillStyle = '#ffffff';
    ctx.shadowColor = '#aaddff';
    ctx.shadowBlur = 15;
    ctx.beginPath();
    ctx.arc(ball.x, ball.y, ball.radius, 0, Math.PI * 2);
    ctx.fill();
    ctx.strokeStyle = '#88ccff';
    ctx.lineWidth = 2;
    ctx.stroke();
    ctx.shadowBlur = 0;
}

function drawParticles() {
    for (let p of particles) {
        ctx.globalAlpha = p.life;
        ctx.fillStyle = p.color;
        ctx.beginPath();
        ctx.arc(p.x, p.y, p.size * p.life, 0, Math.PI * 2);
        ctx.fill();
    }
    ctx.globalAlpha = 1;
}

function drawHUD() {
    ctx.fillStyle = '#ffffff';
    ctx.font = 'bold 18px monospace';
    ctx.textAlign = 'left';
    ctx.fillText(`SCORE: ${score}`, tableX + 10, tableY - 15);
    ctx.textAlign = 'right';
    ctx.fillText(`BALLS: ${'●'.repeat(Math.max(0, balls))}`, tableX + tableW - 10, tableY - 15);

    // Asteroids remaining
    const remaining = asteroids.filter(a => a.alive).length;
    ctx.textAlign = 'center';
    ctx.fillStyle = '#ff6633';
    ctx.fillText(`ASTEROIDS: ${remaining}`, W / 2, tableY - 15);
}

function drawStartScreen() {
    drawStars();
    ctx.fillStyle = '#00ccff';
    ctx.font = 'bold 48px monospace';
    ctx.textAlign = 'center';
    ctx.shadowColor = '#00ccff';
    ctx.shadowBlur = 20;
    ctx.fillText('ASTEROIDS', W / 2, H * 0.3);
    ctx.font = 'bold 36px monospace';
    ctx.fillStyle = '#ff6633';
    ctx.shadowColor = '#ff6633';
    ctx.fillText('PINBALL', W / 2, H * 0.38);

    ctx.shadowBlur = 0;
    ctx.font = '16px monospace';
    ctx.fillStyle = '#aaccdd';
    ctx.fillText('Destroy all asteroids with the pinball!', W / 2, H * 0.5);
    ctx.fillText('', W / 2, H * 0.55);
    ctx.fillText('← → or A/D : Flippers', W / 2, H * 0.6);
    ctx.fillText('SPACE : Launch Ball', W / 2, H * 0.64);
    ctx.fillText('R : Restart', W / 2, H * 0.68);

    ctx.font = 'bold 20px monospace';
    ctx.fillStyle = `rgba(255,255,255,${0.5 + Math.sin(Date.now() * 0.005) * 0.5})`;
    ctx.fillText('PRESS SPACE TO START', W / 2, H * 0.8);
}

function drawGameOver() {
    ctx.fillStyle = 'rgba(0,0,0,0.7)';
    ctx.fillRect(0, 0, W, H);
    ctx.fillStyle = '#ff3333';
    ctx.font = 'bold 42px monospace';
    ctx.textAlign = 'center';
    ctx.shadowColor = '#ff3333';
    ctx.shadowBlur = 15;
    ctx.fillText('GAME OVER', W / 2, H * 0.35);
    ctx.shadowBlur = 0;
    ctx.fillStyle = '#ffffff';
    ctx.font = '24px monospace';
    ctx.fillText(`Final Score: ${score}`, W / 2, H * 0.45);
    const remaining = asteroids.filter(a => a.alive).length;
    ctx.fillStyle = '#ff6633';
    ctx.fillText(`Asteroids Left: ${remaining}`, W / 2, H * 0.52);

    ctx.font = '18px monospace';
    ctx.fillStyle = `rgba(255,255,255,${0.5 + Math.sin(Date.now() * 0.005) * 0.5})`;
    ctx.fillText('PRESS SPACE TO RESTART', W / 2, H * 0.65);
}

function drawWinScreen() {
    ctx.fillStyle = 'rgba(0,0,0,0.6)';
    ctx.fillRect(0, 0, W, H);
    ctx.fillStyle = '#ffdd00';
    ctx.font = 'bold 38px monospace';
    ctx.textAlign = 'center';
    ctx.shadowColor = '#ffdd00';
    ctx.shadowBlur = 20;
    ctx.fillText('ALL CLEARED!', W / 2, H * 0.35);
    ctx.shadowBlur = 0;
    ctx.fillStyle = '#ffffff';
    ctx.font = '24px monospace';
    ctx.fillText(`Score: ${score}`, W / 2, H * 0.45);
    ctx.font = '18px monospace';
    ctx.fillStyle = `rgba(255,255,255,${0.5 + Math.sin(Date.now() * 0.005) * 0.5})`;
    ctx.fillText('PRESS SPACE FOR NEXT WAVE', W / 2, H * 0.6);
}

function drawGame() {
    drawStars();
    drawTable();
    for (let a of asteroids) drawAsteroid(a);
    drawFlippers();
    drawBall();
    drawParticles();
    drawHUD();

    // Win check
    if (asteroids.every(a => !a.alive)) {
        drawWinScreen();
    }
}

function gameLoop(time) {
    const dt = Math.min((time - lastTime) / 16, 3);
    lastTime = time;

    if (gameState === 'start') {
        drawStartScreen();
    } else if (gameState === 'playing') {
        updateFlippers();
        updateBall(dt);
        updateParticles();
        updateAsteroids();
        drawGame();
    } else if (gameState === 'gameover') {
        drawGame();
        drawGameOver();
    }

    requestAnimationFrame(gameLoop);
}

function startGame() {
    initAudio();
    score = 0;
    balls = 3;
    initTable();
    genStars();
    gameState = 'playing';
}

window.addEventListener('keydown', (e) => {
    keys[e.code] = true;
    if (e.code === 'Space') {
        e.preventDefault();
        if (gameState === 'start') startGame();
        else if (gameState === 'playing') launchBall();
        else if (gameState === 'gameover') startGame();
    }
    if (e.code === 'KeyR' && gameState !== 'start') startGame();
});

window.addEventListener('keyup', (e) => { keys[e.code] = false; });

// Handle flipper input in game loop via interval
setInterval(() => {
    if (gameState !== 'playing') return;
    flippers[0].active = keys['ArrowLeft'] || keys['KeyA'];
    flippers[1].active = keys['ArrowRight'] || keys['KeyD'];
}, 16);

genStars();
initTable();
requestAnimationFrame(gameLoop);
</script>
