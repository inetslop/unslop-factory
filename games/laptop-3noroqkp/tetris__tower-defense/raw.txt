function resize() {
    W = window.innerWidth;
    H = window.innerHeight;
    canvas.width = W;
    canvas.height = H;
    cellSize = Math.min(Math.floor((H * 0.8) / ROWS), Math.floor((W * 0.5) / COLS));

// Audio
let audioCtx = null;
function initAudio() {
    if (!audioCtx) {
        audioCtx = new (window.AudioContext || window.webkitAudioContext)();
    }
}
function playSound(freq, dur, type, vol) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = type || 'square';
    osc.frequency.value = freq;
    gain.gain.setValueAtTime(vol || 0.1, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + dur);
}

// Tetromino definitions
const SHAPES = [
    [[1,1,1,1]], // I
    [[1,1],[1,1]], // O
    [[0,1,0],[1,1,1]], // T
    [[1,0,0],[1,1,1]], // L
    [[0,0,1],[1,1,1]], // J
    [[0,1,1],[1,1,0]], // S
    [[1,1,0],[0,1,1]], // Z
    [[1,0],[1,1],[0,1]] // L2
];
const COLORS = ['#00f5ff','#ffee00','#b400ff','#ff6600','#0044ff','#00ff44','#ff0044','#ff8800'];

function rotateCW(shape) {
    const rows = shape.length, cols = shape[0].length;
    const result = [];
    for (let c = 0; c < cols; c++) {
        const row = [];
        for (let r = rows - 1; r >= 0; r--) row.push(shape[r][c]);
        result.push(row);
    }
    return result;
}

function getShape(type) {
    return SHAPES[type].map(r => r.slice());
}

function resetBoard() {
    board = [];
    for (let r = 0; r < ROWS; r++) {
        board[r] = [];
        for (let c = 0; c < COLS; c++) board[r][c] = 0;
    }
}

function spawnPiece() {
    const type = Math.floor(Math.random() * 8);
    const shape = getShape(type);
    const piece = {
        type: type,
        shape: shape,
        x: Math.floor((COLS - shape[0].length) / 2),
        y: 0,
        color: COLORS[type]
    };
    return piece;
}

function validPosition(shape, px, py) {
    for (let r = 0; r < shape.length; r++) {
        for (let c = 0; c < shape[r].length; c++) {
            if (shape[r][c]) {
                const nx = px + c, ny = py + r;
                if (nx < 0 || nx >= COLS || ny >= ROWS) return false;
                if (ny >= 0 && board[ny][nx]) return false;
            }
        }
    }
    return true;
}

function lockPiece() {
    for (let r = 0; r < currentPiece.shape.length; r++) {
        for (let c = 0; c < currentPiece.shape[r].length; c++) {
            if (currentPiece.shape[r][c]) {
                const ny = currentPiece.y + r, nx = currentPiece.x + c;
                if (ny >= 0) board[ny][nx] = currentPiece.type + 1;
            }
        }
    }
    checkLines();
    // Check game over - piece locked above row 0
    for (let r = 0; r < ROWS; r++) {
        for (let c = 0; c < COLS; c++) {
            if (board[r][c] && currentPiece.y + r < 0) { /* already handled */ }
        }
    }
    // Check if any block is in row 0 area that would be "topped out"
    let toppedOut = false;
    for (let c = 0; c < COLS; c++) {
        if (board[0][c]) toppedOut = true;
    }
    if (toppedOut) {
        state = 'GAME_OVER';
        saveHigh();
        return;
    }
    currentPiece = nextPiece;
    nextPiece = spawnPiece();
}

function checkLines() {
    let linesCleared = 0;
    for (let r = ROWS - 1; r >= 0; r--) {
        let full = true;
        for (let c = 0; c < COLS; c++) {
            if (!board[r][c]) { full = false; break; }
        }
        if (full) {
            linesCleared++;
            // Damage enemies in this row
            damageEnemiesInRow(r);
            // Remove row
            board.splice(r, 1);
            board.unshift(new Array(COLS).fill(0));
            // Particles
            for (let c = 0; c < COLS; c++) {
                spawnParticles(gridX + c * cellSize + cellSize/2, gridY + r * cellSize + cellSize/2, '#fff', 3);
            }
        }
    }
    if (linesCleared > 0) {
        const points = [0, 100, 300, 500, 800];
        score += points[linesCleared] * wave;
        comboCount++;
        playSound(400 + linesCleared * 100, 0.2, 'square', 0.15);
    } else {
        comboCount = 0;
    }
}

function damageEnemiesInRow(row) {
    for (let i = enemies.length - 1; i >= 0; i--) {
        const e = enemies[i];
        if (e.row === row) {
            e.hp--;
            if (e.hp <= 0) {
                score += 50 * wave;
                spawnParticles(gridX + e.col * cellSize + cellSize/2, gridY + e.row * cellSize + cellSize/2, e.color, 8);
                enemies.splice(i, 1);
                playSound(600, 0.15, 'sawtooth', 0.1);
            } else {
                e.hitFlash = 10;
            }
        }
    }
}

function spawnEnemy() {
    const types = [
        { hp: 1, speed: 800, color: '#ff4444', size: 0.7 },
        { hp: 2, speed: 600, color: '#ff8800', size: 0.85 },
        { hp: 1, speed: 500, color: '#44ff44', size: 0.6 }, // fast
        { hp: 3, speed: 700, color: '#cc44ff', size: 0.9 }  // tanky
    ];
    const type = types[Math.floor(Math.random() * Math.min(types.length, wave + 1))];
    // Find a column that's not blocked at the bottom
    let col = Math.floor(Math.random() * COLS);
    if (board[ROWS-1][col]) {
        // Try other columns
        for (let tries = 0; tries < 10; tries++) {
            col = Math.floor(Math.random() * COLS);
            if (!board[ROWS-1][col]) break;
        }
    }
    enemies.push({
        col: col,
        row: ROWS - 1,
        hp: type.hp,
        maxHp: type.hp,
        speed: type.speed / (1 + wave * 0.1),
        color: type.color,
        size: type.size,
        moveTimer: Math.random() * type.speed,
        hitFlash: 0
    });
}

function spawnParticles(x, y, color, count) {
    for (let i = 0; i < count; i++) {
        const angle = Math.random() * Math.PI * 2;
        const speed = 1 + Math.random() * 3;
        particles.push({
            x: x, y: y,
            vx: Math.cos(angle) * speed,
            vy: Math.sin(angle) * speed - 2,
            life: 30 + Math.random() * 20,
            color: color,
            size: 2 + Math.random() * 4
        });
    }
}

function startGame() {
    resetBoard();
    score = 0;
    baseHP = maxBaseHP;

    particles = [];
    wave = 1;
    waveTimer = 0;
    enemySpawnTimer = 0;
    comboCount = 0;
    currentPiece = spawnPiece();
    nextPiece = spawnPiece();
    state = 'PLAYING';
}

    } else if (e.key === 'ArrowRight') {
        if (validPosition(currentPiece.shape, currentPiece.x + 1, currentPiece.y)) {
            currentPiece.x++;
            playSound(200, 0.05, 'sine', 0.05);
        }
    } else if (e.key === 'ArrowUp') {
        const rotated = rotateCW(currentPiece.shape);
        // Try normal rotation, then kicks
        if (validPosition(rotated, currentPiece.x, currentPiece.y)) {
            currentPiece.shape = rotated;
        } else if (validPosition(rotated, currentPiece.x - 1, currentPiece.y)) {
            currentPiece.shape = rotated;
            currentPiece.x--;
        } else if (validPosition(rotated, currentPiece.x + 1, currentPiece.y)) {
            currentPiece.shape = rotated;
            currentPiece.x++;
        } else if (validPosition(rotated, currentPiece.x, currentPiece.y - 1)) {
            currentPiece.shape = rotated;
            currentPiece.y--;
        }
        playSound(300, 0.05, 'sine', 0.05);
    } else if (e.key === 'ArrowDown') {
        if (validPosition(currentPiece.shape, currentPiece.x, currentPiece.y + 1)) {
            currentPiece.y++;
            score += 1;
        }
    } else if (e.key === ' ') {
        // Hard drop
        while (validPosition(currentPiece.shape, currentPiece.x, currentPiece.y + 1)) {
            currentPiece.y++;
        }
        lockPiece();
        playSound(500, 0.1, 'triangle', 0.1);
    } else if (e.key === 'p' || e.key === 'P') {
        state = state === 'PAUSED' ? 'PLAYING' : 'PAUSED';
    }
});

window.addEventListener('keyup', function(e) {
    keys[e.key] = false;
});

// Game loop
let lastTime = 0;
let dropTimer = 0;
const baseDropInterval = 800;

function update(dt) {
    if (state !== 'PLAYING') return;

    // Wave progression
    waveTimer += dt;
    if (waveTimer > 30000) { // New wave every 30 seconds
        wave++;
        waveTimer = 0;
        enemySpawnInterval = Math.max(1500, 3000 - wave * 200);
        playSound(800, 0.3, 'square', 0.15);
    }

    // Enemy spawning
    enemySpawnTimer += dt;
    if (enemySpawnTimer >= enemySpawnInterval) {
        enemySpawnTimer = 0;
        spawnEnemy();
    }

    // Update enemies
    for (let i = enemies.length - 1; i >= 0; i--) {
        const e = enemies[i];
        e.moveTimer += dt;
        if (e.hitFlash > 0) e.hitFlash--;

        if (e.moveTimer >= e.speed) {
            e.moveTimer = 0;
            // Try to move up
            if (e.row <= 0) {
                // Reached the base!
                baseHP -= e.maxHp;
                shakeTime = 200;
                spawnParticles(gridX + e.col * cellSize + cellSize/2, gridY, '#ff0000', 10);
                enemies.splice(i, 1);
                playSound(150, 0.3, 'sawtooth', 0.2);
                if (baseHP <= 0) {
                    state = 'GAME_OVER';
                    saveHigh();
                }
            } else if (!board[e.row - 1][e.col]) {
                e.row--;
            }
        }
    }

    // Drop piece
    dropTimer += dt;
    const dropInterval = baseDropInterval / (1 + Math.floor(score / 500) * 0.2);
    if (dropTimer >= dropInterval) {
        dropTimer = 0;
        if (validPosition(currentPiece.shape, currentPiece.x, currentPiece.y + 1)) {
            currentPiece.y++;
        } else {
            lockPiece();
        }
    }

    // Particles
    for (let i = particles.length - 1; i >= 0; i--) {
        const p = particles[i];
        p.x += p.vx;
        p.y += p.vy;
        p.vy += 0.15;
        p.life--;
        if (p.life <= 0) particles.splice(i, 1);
    }

    // Shake
    if (shakeTime > 0) shakeTime -= dt;
}

    ctx.save();

        renderMenu();
    } else if (state === 'GAME_OVER') {
        renderGame();
        renderGameOver();

    ctx.restore();
}

function renderGame() {
    // Grid
    ctx.strokeStyle = '#1a1a3a';
    ctx.lineWidth = 1;
    for (let r = 0; r <= ROWS; r++) {
        ctx.beginPath();
        ctx.moveTo(gridX, gridY + r * cellSize);
        ctx.lineTo(gridX + COLS * cellSize, gridY + r * cellSize);
        ctx.stroke();
    }
    for (let c = 0; c <= COLS; c++) {
        ctx.beginPath();
        ctx.moveTo(gridX + c * cellSize, gridY);
        ctx.lineTo(gridX + c * cellSize, gridY + ROWS * cellSize);
        ctx.stroke();
    }

    // Board blocks
    for (let r = 0; r < ROWS; r++) {
        for (let c = 0; c < COLS; c++) {
            if (board[r] && board[r][c]) {
                const colorIdx = board[r][c] - 1;
                ctx.fillStyle = COLORS[colorIdx % COLORS.length];
                ctx.fillRect(gridX + c * cellSize + 1, gridY + r * cellSize + 1, cellSize - 2, cellSize - 2);
                // Highlight
                ctx.fillStyle = 'rgba(255,255,255,0.2)';
                ctx.fillRect(gridX + c * cellSize + 1, gridY + r * cellSize + 1, cellSize - 2, 4);
            }
        }
    }

    // Enemies
    for (const e of enemies) {
        const ex = gridX + e.col * cellSize + cellSize/2;
        const ey = gridY + e.row * cellSize + cellSize/2;
        const radius = cellSize * 0.4 * e.size;

        if (e.hitFlash > 0) {
            ctx.fillStyle = '#fff';
        } else {
            ctx.fillStyle = e.color;
        }
        ctx.beginPath();
        ctx.arc(ex, ey, radius, 0, Math.PI * 2);
        ctx.fill();

        // HP indicator for multi-HP enemies
        if (e.maxHp > 1) {
            ctx.fillStyle = '#fff';
            ctx.font = `bold ${Math.floor(cellSize * 0.4)}px monospace`;
            ctx.textAlign = 'center';
            ctx.fillText(e.hp, ex, ey + cellSize * 0.15);
        }

        // Eyes
        ctx.fillStyle = '#fff';
        const eyeR = radius * 0.2;
        ctx.beginPath();
        ctx.arc(ex - radius * 0.3, ey - radius * 0.2, eyeR, 0, Math.PI * 2);
        ctx.fill();
        ctx.beginPath();
        ctx.arc(ex + radius * 0.3, ey - radius * 0.2, eyeR, 0, Math.PI * 2);
        ctx.fill();
    }

    // Current piece (ghost + actual)
    if (currentPiece) {
        // Ghost
        let ghostY = currentPiece.y;
        while (validPosition(currentPiece.shape, currentPiece.x, ghostY + 1)) ghostY++;
        ctx.globalAlpha = 0.3;
        drawShape(currentPiece.shape, currentPiece.color, currentPiece.x, ghostY);
        ctx.globalAlpha = 1;

        // Actual piece
        drawShape(currentPiece.shape, currentPiece.color, currentPiece.x, currentPiece.y);
    }

    // Next piece preview
    if (nextPiece) {
        const px = gridX + COLS * cellSize + 20;
        const py = gridY + 20;
        ctx.fillStyle = '#fff';
        ctx.font = `${Math.min(12, cellSize*0.4)}px monospace`;
        ctx.textAlign = 'left';
        ctx.fillText('NEXT:', px, py);
        drawShapePreview(nextPiece.shape, nextPiece.color, px + 5, py + 10, cellSize * 0.5);
    }

    // Base (top)
    const baseY = gridY - cellSize * 0.8;
    ctx.fillStyle = '#00ffcc';
    ctx.font = `bold ${Math.min(14, cellSize*0.5)}px monospace`;
    ctx.textAlign = 'center';
    ctx.fillText('⬆ BASE ⬆', gridX + COLS * cellSize / 2, baseY);

    // HP Bar
    const hpBarW = COLS * cellSize;
    const hpBarH = 8;
    const hpBarX = gridX;
    const hpBarY = gridY - cellSize * 0.5;
    ctx.fillStyle = '#333';
    ctx.fillRect(hpBarX, hpBarY, hpBarW, hpBarH);
    const hpRatio = Math.max(0, baseHP / maxBaseHP);
    ctx.fillStyle = hpRatio > 0.5 ? '#0f0' : hpRatio > 0.25 ? '#ff0' : '#f00';
    ctx.fillRect(hpBarX, hpBarY, hpBarW * hpRatio, hpBarH);

    // Particles
    for (const p of particles) {
        ctx.globalAlpha = p.life / 40;
        ctx.fillStyle = p.color;
        ctx.fillRect(p.x - p.size/2, p.y - p.size/2, p.size, p.size);
    }
    ctx.globalAlpha = 1;

    // HUD
    const hudY = gridY + ROWS * cellSize + 15;
    ctx.fillStyle = '#fff';
    ctx.font = `bold ${Math.min(16, cellSize*0.55)}px monospace`;
    ctx.textAlign = 'left';
    ctx.fillText(`Score: ${score}`, gridX, hudY);
    ctx.fillText(`Wave: ${wave}`, gridX, hudY + 20);
    ctx.fillText(`Enemies: ${enemies.length}`, gridX, hudY + 40);

    // High score
    ctx.textAlign = 'right';
    ctx.fillStyle = '#ff0';
    ctx.fillText(`Best: ${Math.max(highScore, score)}`, gridX + COLS * cellSize, hudY);

    if (state === 'PAUSED') {
        ctx.fillStyle = 'rgba(0,0,0,0.7)';
        ctx.fillRect(0, 0, W, H);
        ctx.fillStyle = '#fff';
        ctx.font = `bold ${Math.min(36, W*0.05)}px monospace`;
        ctx.textAlign = 'center';
        ctx.fillText('PAUSED', W/2, H/2);
        ctx.font = `${Math.min(14, W*0.02)}px monospace`;
        ctx.fillText('Press P to resume', W/2, H/2 + 30);
    }
}

function drawShape(shape, color, px, py) {
    for (let r = 0; r < shape.length; r++) {
        for (let c = 0; c < shape[r].length; c++) {
            if (shape[r][c]) {
                const sx = gridX + (px + c) * cellSize;
                const sy = gridY + (py + r) * cellSize;
                if (sy >= gridY && sy <= gridY + ROWS * cellSize) {
                    ctx.fillStyle = color;
                    ctx.fillRect(sx + 1, sy + 1, cellSize - 2, cellSize - 2);
                    ctx.fillStyle = 'rgba(255,255,255,0.2)';
                    ctx.fillRect(sx + 1, sy + 1, cellSize - 2, 4);
                }
            }
        }
    }
}

function drawShapePreview(shape, color, px, py, size) {
    for (let r = 0; r < shape.length; r++) {
        for (let c = 0; c < shape[r].length; c++) {
            if (shape[r][c]) {
                ctx.fillStyle = color;
                ctx.fillRect(px + c * size, py + r * size, size - 1, size - 1);
            }
        }
    }
}

function renderGameOver() {
    ctx.fillStyle = 'rgba(0,0,0,0.8)';
    ctx.fillRect(0, 0, W, H);
    ctx.fillStyle = '#f44';
    ctx.font = `bold ${Math.min(42, W*0.06)}px monospace`;
    ctx.textAlign = 'center';
    ctx.fillText('GAME OVER', W/2, H * 0.35);

    ctx.fillStyle = '#fff';
    ctx.font = `${Math.min(18, W*0.03)}px monospace`;
    ctx.fillText(`Final Score: ${score}`, W/2, H * 0.45);
    ctx.fillText(`Waves Survived: ${wave}`, W/2, H * 0.5);

    if (score >= highScore && score > 0) {
        ctx.fillStyle = '#ff0';
        ctx.fillText('★ NEW HIGH SCORE! ★', W/2, H * 0.58);
    }

    ctx.fillStyle = '#aaa';
    ctx.font = `${Math.min(14, W*0.02)}px monospace`;
    ctx.fillText('Press ENTER to play again', W/2, H * 0.68);
}

function gameLoop(timestamp) {
    const dt = Math.min(timestamp - lastTime, 50);
    lastTime = timestamp;

    update(dt);
    render();

    requestAnimationFrame(gameLoop);
}

requestAnimationFrame(gameLoop);
})();
