const canvas = document.getElementById('game');
const ctx = canvas.getContext('2d');

let W, H;
function resize() {
    W = canvas.width = window.innerWidth;
    H = canvas.height = window.innerHeight;
}
window.addEventListener('resize', resize);
resize();

// Audio
let audioCtx = null;
function initAudio() {
    if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
}
function playSound(type) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    switch(type) {
        case 'laser':
            osc.type = 'square';
            osc.frequency.setValueAtTime(880, audioCtx.currentTime);
            osc.frequency.exponentialRampToValueAtTime(220, audioCtx.currentTime + 0.1);
            gain.gain.setValueAtTime(0.15, audioCtx.currentTime);
            gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.1);
            osc.start(); osc.stop(audioCtx.currentTime + 0.1);
            break;
        case 'explosion':
            osc.type = 'sawtooth';
            osc.frequency.setValueAtTime(150, audioCtx.currentTime);
            osc.frequency.exponentialRampToValueAtTime(30, audioCtx.currentTime + 0.3);
            gain.gain.setValueAtTime(0.25, audioCtx.currentTime);
            gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.3);
            osc.start(); osc.stop(audioCtx.currentTime + 0.3);
            break;
        case 'hit':
            osc.type = 'triangle';
            osc.frequency.setValueAtTime(200, audioCtx.currentTime);
            osc.frequency.exponentialRampToValueAtTime(80, audioCtx.currentTime + 0.15);
            gain.gain.setValueAtTime(0.3, audioCtx.currentTime);
            gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.15);
            osc.start(); osc.stop(audioCtx.currentTime + 0.15);
            break;
        case 'powerup':
            osc.type = 'sine';
            osc.frequency.setValueAtTime(440, audioCtx.currentTime);
            osc.frequency.exponentialRampToValueAtTime(880, audioCtx.currentTime + 0.15);
            gain.gain.setValueAtTime(0.2, audioCtx.currentTime);
            gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.2);
            osc.start(); osc.stop(audioCtx.currentTime + 0.2);
            break;
    }
}

// Game state
let gameState = 'start'; // start, playing, gameover
let score = 0;
let highScore = 0;
try { highScore = parseInt(localStorage.getItem('nebulaStrikerHigh')) || 0; } catch(e) {}
let wave = 1;

// Stars background
let stars = [];
function initStars() {
    stars = [];
    for (let i = 0; i < 200; i++) {
        stars.push({
            x: Math.random() * W,
            y: Math.random() * H,
            speed: 0.5 + Math.random() * 3,
            size: 0.5 + Math.random() * 2,
            brightness: 0.3 + Math.random() * 0.7
        });
    }
}
initStars();

// Player
let player = { x: 100, y: 0, w: 40, h: 24, speed: 5, health: 5, maxHealth: 5, shootCooldown: 0, invincible: 0 };
let bullets = [];
let enemies = [];
let particles = [];
let powerups = [];
let enemyBullets = [];

// Input
const keys = {};
window.addEventListener('keydown', e => {
    keys[e.code] = true;
    if (gameState === 'start' && (e.code === 'Space' || e.code === 'Enter')) {
        initAudio();
        startGame();
    } else if (gameState === 'gameover' && (e.code === 'Space' || e.code === 'Enter' || e.code === 'KeyR')) {
        startGame();
    }
    if (!audioCtx) initAudio();
});
window.addEventListener('keyup', e => { keys[e.code] = false; });

function startGame() {
    gameState = 'playing';
    score = 0;
    wave = 1;
    player.x = 80;
    player.y = H / 2;
    player.health = 5;
    player.invincible = 0;
    player.shootCooldown = 0;
    bullets = [];
    enemies = [];
    particles = [];
    powerups = [];
    enemyBullets = [];
}

function spawnEnemy() {
    let type = Math.random();
    let e = { x: W + 50, y: Math.random() * (H - 60) + 30 };
    if (type < 0.4) {
        // Basic drone
        e.w = 30; e.h = 20; e.hp = 1; e.speed = 2 + wave * 0.3; e.type = 'drone';
        e.score = 10;
    } else if (type < 0.7) {
        // Fighter
        e.w = 40; e.h = 28; e.hp = 2; e.speed = 1.5 + wave * 0.2; e.type = 'fighter';
        e.score = 25;
        e.y = Math.random() * (H - 80) + 40;
    } else if (type < 0.9) {
        // Tank
        e.w = 50; e.h = 36; e.hp = 4; e.speed = 1 + wave * 0.15; e.type = 'tank';
        e.score = 50;
    } else {
        // Boss-ish large
        e.w = 70; e.h = 50; e.hp = Math.floor(5 + wave); e.speed = 0.8 + wave * 0.1; e.type = 'large';
        e.score = 100;
    }
    e.maxHp = e.hp;
    e.shootTimer = Math.random() * 60 + 30;
    enemies.push(e);
}

function spawnPowerup(x, y) {
    let types = ['health', 'rapid'];
    let type = types[Math.floor(Math.random() * types.length)];
    powerups.push({ x, y, w: 20, h: 20, type, speed: 1.5 });
}

function createExplosion(x, y, color, count) {
    for (let i = 0; i < (count || 12); i++) {
        let angle = Math.random() * Math.PI * 2;
        let speed = 1 + Math.random() * 4;
        particles.push({
            x, y,
            vx: Math.cos(angle) * speed,
            vy: Math.sin(angle) * speed,
            life: 30 + Math.random() * 20,
            maxLife: 50,
            color: color || '#ff6600',
            size: 2 + Math.random() * 4
        });
    }
}

function createMuzzleFlash(x, y) {
    for (let i = 0; i < 4; i++) {
        particles.push({
            x, y,
            vx: 2 + Math.random() * 3,
            vy: (Math.random() - 0.5) * 2,
            life: 8, maxLife: 8,
            color: '#ffff66',
            size: 2 + Math.random() * 2
        });
    }
}

let spawnTimer = 0;
let rapidFire = false;
let rapidTimer = 0;

function update() {
    if (gameState !== 'playing') return;

    // Scroll stars
    for (let s of stars) {
        s.x -= s.speed;
        if (s.x < 0) { s.x = W; s.y = Math.random() * H; }
    }

    // Player movement
    let dx = 0, dy = 0;
    if (keys['ArrowUp'] || keys['KeyW']) dy = -player.speed;
    if (keys['ArrowDown'] || keys['KeyS']) dy = player.speed;
    if (keys['ArrowLeft'] || keys['KeyA']) dx = -player.speed;
    if (keys['ArrowRight'] || keys['KeyD']) dx = player.speed;
    player.x += dx;
    player.y += dy;
    player.x = Math.max(0, Math.min(W - player.w, player.x));
    player.y = Math.max(0, Math.min(H - player.h, player.y));

    if (player.invincible > 0) player.invincible--;

    // Shooting
    player.shootCooldown--;
    if (rapidTimer > 0) rapidTimer--;
    if (rapidTimer <= 0) rapidFire = false;

    if ((keys['Space'] || keys['KeyZ']) && player.shootCooldown <= 0) {
        let cooldown = rapidFire ? 5 : 12;
        if (player.shootCooldown <= 0) {
            bullets.push({ x: player.x + player.w, y: player.y + player.h / 2 - 2, w: 16, h: 4, speed: 12 });
            player.shootCooldown = cooldown;
            createMuzzleFlash(player.x + player.w, player.y + player.h / 2);
            playSound('laser');
        }
    }

    // Update bullets
    for (let i = bullets.length - 1; i >= 0; i--) {
        bullets[i].x += bullets[i].speed;
        if (bullets[i].x > W + 20) bullets.splice(i, 1);
    }

    // Spawn enemies
    spawnTimer--;
    if (spawnTimer <= 0) {
        spawnEnemy();
        let interval = Math.max(30, 90 - wave * 5);
        spawnTimer = interval + Math.random() * 20;
        if (wave > 3 && Math.random() < 0.3) spawnEnemy();
    }

    // Update enemies
    for (let i = enemies.length - 1; i >= 0; i--) {
        let e = enemies[i];
        e.x -= e.speed;
        if (e.type === 'fighter') {
            e.y += Math.sin(Date.now() * 0.003 + i) * 1.5;
        }
        if (e.x < -80) { enemies.splice(i, continue); }

        // Enemy shooting
        e.shootTimer--;
        if (e.shootTimer <= 0 && e.type !== 'drone') {
            let angle = Math.atan2(player.y - e.y, player.x - e.x);
            enemyBullets.push({
                x: e.x, y: e.y + e.h / 2,
                vx: Math.cos(angle) * 4, vy: Math.sin(angle) * 4,
                w: 8, h: 8
            });
            e.shootTimer = 60 + Math.random() * 60;
        }

        // Bullet collision
        for (let j = bullets.length - 1; j >= 0; j--) {
            let b = bullets[j];
            if (b.x < e.x + e.w && b.x + b.w > e.x && b.y < e.y + e.h && b.y + b.h > e.y) {
                e.hp--;
                bullets.splice(j, 1);
                createExplosion(b.x, b.y, '#ffff00', 4);
                if (e.hp <= 0) {
                    score += e.score;
                    createExplosion(e.x + e.w / 2, e.y + e.h / 2, e.type === 'large' ? '#ff3300' : '#ff6600', e.type === 'large' ? 25 : 12);
                    playSound('explosion');
                    if (Math.random() < 0.2) spawnPowerup(e.x, e.y);
                    enemies.splice(i, 1);
                } else {
                    playSound('hit');
                }
            }
        }

        // Player collision
        if (player.invincible <= 0 &&
            player.x < e.x + e.w && player.x + player.w > e.x &&
            player.y < e.y + e.h && player.y + player.h > e.y) {
            player.health--;
            player.invincible = 60;
            createExplosion(player.x, player.y, '#ff0000', 8);
            playSound('explosion');
            enemies.splice(i, 1);
            if (player.health <= 0) {
                gameState = 'gameover';
                try {
                    if (score > highScore) {
                        highScore = score;
                        localStorage.setItem('nebulaStrikerHigh', highScore);
                    }
                } catch(e2) {}
            }
        }
    }

    // Update enemy bullets
    for (let i = enemyBullets.length - 1; i >= 0; i--) {
        let eb = enemyBullets[i];
        eb.x += eb.vx;
        eb.y += eb.vy;
        if (eb.x < -20 || eb.x > W + 20 || eb.y < -20 || eb.y > H + 20) {
            enemyBullets.splice(i, 1);
            continue;
        }
        // Hit player
        if (player.invincible <= 0 &&
            eb.x < player.x + player.w && eb.x + eb.w > player.x &&
            eb.y < player.y + player.h && eb.y + eb.h > player.y) {
            player.health--;
            player.invincible = 60;
            enemyBullets.splice(i, 1);
            createExplosion(player.x, player.y, '#ff0000', 8);
            playSound('explosion');
            if (player.health <= 0) {
                gameState = 'gameover';
                try {
                    if (score > highScore) {
                        highScore = score;
                        localStorage.setItem('nebulaStrikerHigh', highScore);
                    }
                } catch(e2) {}
            }
        }
    }

    // Update powerups
    for (let i = powerups.length - 1; i >= 0; i--) {
        let p = powerups[i];
        p.x -= p.speed;
        if (p.x < -30) { powerups.splice(i, 1); continue; }
        if (player.x < p.x + p.w && player.x + player.w > p.x &&
            player.y < p.y + p.h && player.y + player.h > p.y) {
            if (p.type === 'health' && player.health < player.maxHealth) {
                player.health++;
            } else if (p.type === 'rapid') {
                rapidFire = true;
                rapidTimer = 300;
            }
            playSound('powerup');
            powerups.splice(i, 1);
        }
    }

    // Update particles
    for (let i = particles.length - 1; i >= 0; i--) {
        let p = particles[i];
        p.x += p.vx;
        p.y += p.vy;
        p.life--;
        if (p.life <= 0) particles.splice(i, 1);
    }

    // Wave progression
    if (score > wave * 200) {
        wave++;
    }
}

function drawShip(x, y, w, h) {
    ctx.save();
    ctx.translate(x + w / 2, y + h / 2);
    // Body
    ctx.fillStyle = '#4488ff';
    ctx.beginPath();
    ctx.moveTo(w / 2, 0);
    ctx.lineTo(-w / 2, -h / 2);
    ctx.lineTo(-w / 4, 0);
    ctx.lineTo(-w / 2, h / 2);
    ctx.closePath();
    ctx.fill();
    // Cockpit
    ctx.fillStyle = '#aaddff';
    ctx.beginPath();
    ctx.arc(w / 6, 0, h / 5, 0, Math.PI * 2);
    ctx.fill();
    // Engine glow
    let flicker = Math.random() * 5;
    ctx.fillStyle = `rgba(255, ${150 + flicker * 5}, 0, 0.8)`;
    ctx.beginPath();
    ctx.moveTo(-w / 2, -h / 4);
    ctx.lineTo(-w / 2 - 8 - flicker, 0);
    ctx.lineTo(-w / 2, h / 4);
    ctx.closePath();
    ctx.fill();
    ctx.restore();
}

function drawEnemy(e) {
    let cx = e.x + e.w / 2, cy = e.y + e.h / 2;
    ctx.save();
    ctx.translate(cx, cy);
    switch (e.type) {
        case 'drone':
            ctx.fillStyle = '#ff4444';
            ctx.fillRect(-e.w / 2, -e.h / 2, e.w, e.h);
            ctx.fillStyle = '#ff8888';
            ctx.fillRect(-e.w / 4, -e.h / 4, e.w / 2, e.h / 2);
            break;
        case 'fighter':
            ctx.fillStyle = '#cc44ff';
            ctx.beginPath();
            ctx.moveTo(-e.w / 2, 0);
            ctx.lineTo(0, -e.h / 2);
            ctx.lineTo(e.w / 2, 0);
            ctx.lineTo(0, e.h / 2);
            ctx.closePath();
            ctx.fill();
            ctx.fillStyle = '#ff66ff';
            ctx.beginPath();
            ctx.arc(0, 0, e.h / 4, 0, Math.PI * 2);
            ctx.fill();
            break;
        case 'tank':
            ctx.fillStyle = '#ff8800';
            ctx.fillRect(-e.w / 2, -e.h / 2, e.w, e.h);
            ctx.fillStyle = '#ffcc00';
            ctx.fillRect(-e.w / 3, -e.h / 4, e.w * 0.66, e.h / 2);
            ctx.strokeStyle = '#ffaa00';
            ctx.lineWidth = 2;
            ctx.strokeRect(-e.w / 2 + 5, -e.h / 2 + 5, e.w - 10, e.h - 10);
            break;
        case 'large':
            ctx.fillStyle = '#ff2200';
            ctx.beginPath();
            ctx.moveTo(-e.w / 2, 0);
            ctx.lineTo(-e.w / 4, -e.h / 2);
            ctx.lineTo(e.w / 4, -e.h / 3);
            ctx.lineTo(e.w / 2, 0);
            ctx.lineTo(e.w / 4, e.h / 3);
            ctx.lineTo(-e.w / 4, e.h / 2);
            ctx.closePath();
            ctx.fill();
            ctx.fillStyle = '#ff6644';
            ctx.beginPath();
            ctx.arc(0, 0, e.h / 4, 0, Math.PI * 2);
            ctx.fill();
            break;
    }
    // Health bar
    if (e.hp < e.maxHp) {
        let bw = e.w;
        ctx.fillStyle = '#333';
        ctx.fillRect(cx - bw / 2, cy - e.h / 2 - 8, bw, 4);
        ctx.fillStyle = '#ff3333';
        ctx.fillRect(cx - bw / 2, cy - e.h / 2 - 8, bw * (e.hp / e.maxHp), 4);
    }
    ctx.restore();
}

function draw() {
    // Background
    let grad = ctx.createLinearGradient(0, 0, W, H);
    grad.addColorStop(0, '#0a0a1a');
    grad.addColorStop(1, '#1a0a2a');
    ctx.fillStyle = grad;
    ctx.fillRect(0, 0, W, H);

    // Stars
    for (let s of stars) {
        ctx.fillStyle = `rgba(255,255,255,${s.brightness})`;
        ctx.fillRect(s.x, s.y, s.size, s.size);
    }

    if (gameState === 'start') {
        drawStartScreen();
        return;
    }

    // Powerups
    for (let p of powerups) {
        ctx.fillStyle = p.type === 'health' ? '#44ff44' : '#ffff44';
        ctx.beginPath();
        ctx.arc(p.x + 10, p.y + 10, 10, 0, Math.PI * 2);
        ctx.fill();
        ctx.fillStyle = '#000';
        ctx.font = 'bold 12px monospace';
        ctx.textAlign = 'center';
        ctx.fillText(p.type === 'health' ? '+' : 'R', p.x + 10, p.y + 14);
    }

    // Enemies
    for (let e of enemies) drawEnemy(e);

    // Player bullets
    ctx.fillStyle = '#66ccff';
    for (let b of bullets) {
        ctx.fillRect(b.x, b.y, b.w, b.h);
        ctx.fillStyle = '#ffffff';
        ctx.fillRect(b.x + 2, b.y + 1, b.w - 4, b.h - 2);
        ctx.fillStyle = '#66ccff';
    }

    // Enemy bullets
    ctx.fillStyle = '#ff4444';
    for (let eb of enemyBullets) {
        ctx.beginPath();
        ctx.arc(eb.x + 4, eb.y + 4, 5, 0, Math.PI * 2);
        ctx.fill();
    }

    // Player
    if (player.invincible <= 0 || Math.floor(Date.now() / 100) % 2 === 0) {
        drawShip(player.x, player.y, player.w, player.h);
    }

    // Particles
    for (let p of particles) {
        let alpha = p.life / p.maxLife;
        ctx.fillStyle = p.color.replace(')', `,${alpha})`).replace('#', 'rgba(').replace(/'/g, '');
        // Simpler approach
        ctx.globalAlpha = alpha;
        ctx.fillStyle = p.color;
        ctx.fillRect(p.x - p.size / 2, p.y - p.size / 2, p.size, p.size);
        ctx.globalAlpha = 1;
    }

    // HUD
    ctx.fillStyle = '#fff';
    ctx.font = 'bold 18px monospace';
    ctx.textAlign = 'left';
    ctx.fillText(`SCORE: ${score}`, 20, 30);
    ctx.fillText(`WAVE: ${wave}`, 20, 55);

    // Health bar
    let hbX = 20, hbY = 70, hbW = 120, hbH = 16;
    ctx.fillStyle = '#333';
    ctx.fillRect(hbX, hbY, hbW, hbH);
    let hpRatio = player.health / player.maxHealth;
    ctx.fillStyle = hpRatio > 0.5 ? '#44ff44' : hpRatio > 0.25 ? '#ffaa00' : '#ff3333';
    ctx.fillRect(hbX, hbY, hbW * hpRatio, hbH);
    ctx.strokeStyle = '#fff';
    ctx.lineWidth = 1;
    ctx.strokeRect(hbX, hbY, hbW, hbH);

    if (rapidFire) {
        ctx.fillStyle = '#ffff44';
        ctx.fillText('RAPID FIRE!', 20, 110);
    }

    if (gameState === 'gameover') {
        drawGameOver();
    }
}

function drawStartScreen() {
    ctx.fillStyle = 'rgba(0,0,0,0.7)';
    ctx.fillRect(0, 0, W, H);

    ctx.textAlign = 'center';
    ctx.fillStyle = '#66ccff';
    ctx.font = 'bold 48px monospace';
    ctx.fillText('NEBULA STRIKER', W / 2, H / 2 - 80);

    ctx.fillStyle = '#fff';
    ctx.font = '18px monospace';
    ctx.fillText('A side-scrolling space shooter', W / 2, H / 2 - 40);

    ctx.fillStyle = '#aaaacc';
    ctx.font = '16px monospace';
    let lines = [
        'WASD / Arrow Keys - Move',
        'SPACE / Z - Shoot',
        '',
        'Destroy enemies. Collect power-ups.',
        'Green = Health  |  Yellow = Rapid Fire'
    ];
    lines.forEach((line, i) => {
        ctx.fillText(line, W / 2, H / 2 + 10 + i * 28);
    });

    if (highScore > 0) {
        ctx.fillStyle = '#ffcc44';
        ctx.font = '16px monospace';
        ctx.fillText(`High Score: ${highScore}`, W / 2, H / 2 + 170);
    }

    let pulse = Math.sin(Date.now() * 0.005) * 0.3 + 0.7;
    ctx.fillStyle = `rgba(255,255,255,${pulse})`;
    ctx.font = 'bold 20px monospace';
    ctx.fillText('PRESS SPACE TO START', W / 2, H / 2 + 210);
}

function drawGameOver() {
    ctx.fillStyle = 'rgba(0,0,0,0.6)';
    ctx.fillRect(0, 0, W, H);

    ctx.textAlign = 'center';
    ctx.fillStyle = '#ff4444';
    ctx.font = 'bold 48px monospace';
    ctx.fillText('GAME OVER', W / 2, H / 2 - 60);

    ctx.fillStyle = '#fff';
    ctx.font = '24px monospace';
    ctx.fillText(`Final Score: ${score}`, W / 2, H / 2 - 10);

    if (score >= highScore && score > 0) {
        ctx.fillStyle = '#ffcc44';
        ctx.font = '20px monospace';
        ctx.fillText('NEW HIGH SCORE!', W / 2, H / 2 + 30);
    }

    let pulse = Math.sin(Date.now() * 0.005) * 0.3 + 0.7;
    ctx.fillStyle = `rgba(255,255,255,${pulse})`;
    ctx.font = '18px monospace';
    ctx.fillText('PRESS SPACE TO RESTART', W / 2, H / 2 + 80);
}

let lastTime = 0;
function gameLoop(time) {
    let dt = time - lastTime;
    lastTime = time;

    update();
    draw();

    requestAnimationFrame(gameLoop);
}
requestAnimationFrame(gameLoop);
</script>
</body>
</html>
