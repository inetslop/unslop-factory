<script>
const canvas = document.getElementById('game');
const ctx = canvas.getContext('2d');

let W, H;
function resize() {
    W = canvas.width = window.innerWidth;
    H = canvas.height = window.innerHeight;
}
window.addEventListener('resize', resize);
resize();

// Audio
let audioCtx = null;
function initAudio() {
    if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
}
function playSound(freq, duration, type, vol) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = type || 'square';
    osc.frequency.value = freq;
    gain.gain.setValueAtTime(vol || 0.1, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + duration);
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + duration);
}
function sBounce() { playSound(440, 0.08, 'square', 0.08); }
function sBreak() { playSound(880, 0.15, 'sawtooth', 0.12); }
function sPaddle() { playSound(220, 0.06, 'triangle', 0.1); }
function sLife() { playSound(150, 0.4, 'sawtooth', 0.15); }
function sWin() { playSound(660, 0.3, 'sine', 0.12); setTimeout(()=>playSound(880,0.3,'sine',0.12),150); setTimeout(()=>playSound(1100,0.4,'sine',0.12),300); }
function sLose() { playSound(200, 0.5, 'sawtooth', 0.15); }

// Game State
let state = 'start'; // start, playing, win, lose
let score = 0;
let lives = 3;
let balls = [];
let particles = [];
let bricks = [];
let trailPoints = [];

const BRICK_COLS = 8;
const BRICK_ROWS = 5;

function initBricks() {
    bricks = [];
    const bw = Math.min(60, (W - 100) / BRICK_COLS);
    const bh = Math.min(24, H * 0.03);
    const gap = 4;
    const totalW = BRICK_COLS * (bw + gap) - gap;
    const startX = (W - totalW) / 2;
    const startY = H * 0.06;
    for (let r = 0; r < BRICK_ROWS; r++) {
        for (let c = 0; c < BRICK_COLS; c++) {
            bricks.push({
                x: startX + c * (bw + gap),
                y: startY + r * (bh + gap),
                w: bw, h: bh,
                alive: true,
                row: r,
                color: `hsl(${(r * 40) + 200}, 80%, ${60 - r * 5}%)`
            });
        }
    }
}

// Paddles
let playerPaddle = { w: 0, h: 12, x: 0, y: 0 };
let aiPaddle = { w: 0, h: 14, x: 0, y: 0, targetX: 0, speed: 3 };

function initPaddles() {
    playerPaddle.w = Math.min(120, W * 0.15);
    playerPaddle.h = 12;
    playerPaddle.x = (W - playerPaddle.w) / 2;
    playerPaddle.y = H - 40;

    aiPaddle.w = Math.min(100, W * 0.13);
    aiPaddle.h = 14;
    aiPaddle.x = (W - aiPaddle.w) / 2;
    aiPaddle.y = H * 0.55;
    aiPaddle.targetX = aiPaddle.x;
}

function initBall() {
    const speed = Math.min(6, W * 0.008);
    balls = [{ x: W / 2, y: playerPaddle.y - 10, vx: (Math.random() > 0.5 ? 1 : -1) * speed * 0.7, vy: -speed }];
}

function resetGame() {
    score = 0;
    lives = 3;
    particles = [];
    trailPoints = [];
    initBricks();
    initPaddles();
    initBall();
    state = 'playing';
}

// Input
let mouseX = W / 2;
let keys = {};

window.addEventListener('mousemove', e => { mouseX = e.clientX; });
window.addEventListener('touchmove', e => { if (e.touches.length) mouseX = e.touches[0].clientX; }, { passive: true });
window.addEventListener('keydown', e => {
    keys[e.code] = true;
    if (state === 'start' || state === 'win' || state === 'lose') {
        initAudio();
        resetGame();
    }
});
window.addEventListener('keyup', e => { keys[e.code] = false; });

canvas.addEventListener('click', () => {
    if (state !== 'playing') {
        initAudio();
        resetGame();
    }
});

// Particles
function spawnParticles(x, y, color, count) {
    for (let i = 0; i < count; i++) {
        const angle = Math.random() * Math.PI * 2;
        const speed = Math.random() * 4 + 1;
        particles.push({ x, y, vx: Math.cos(angle) * speed, vy: Math.sin(angle) * speed, life: 1, color, size: Math.random() * 4 + 2 });
    }
}

// Update
function update() {
    if (state !== 'playing') return;

    // Player paddle
    const targetX = mouseX - playerPaddle.w / 2;
    playerPaddle.x += (targetX - playerPaddle.x) * 0.3;
    playerPaddle.x = Math.max(0, Math.min(W - playerPaddle.w, playerPaddle.x));

    if (keys['ArrowLeft']) playerPaddle.x -= 8;
    if (keys['ArrowRight']) playerPaddle.x += 8;
    playerPaddle.x = Math.max(0, Math.min(W - playerPaddle.w, playerPaddle.x));

    // AI paddle - tracks ball with some delay and imperfection
    const ballForAI = balls[0];
    if (ballForAI.vy < 0) {
        aiPaddle.targetX = ballForAI.x - aiPaddle.w / 2;
    } else {
        aiPaddle.targetX += (Math.random() - 0.5) * 2;
    }
    const diff = aiPaddle.targetX - aiPaddle.x;
    aiPaddle.x += Math.sign(diff) * Math.min(Math.abs(diff), aiPaddle.speed);
    aiPaddle.x = Math.max(0, Math.min(W - aiPaddle.w, aiPaddle.x));

    // Balls
    for (let i = balls.length - 1; i >= 0; i--) {
        const b = balls[i];
        b.x += b.vx;
        b.y += b.vy;

        // Trail
        trailPoints.push({ x: b.x, y: b.y });
        if (trailPoints.length > 12) trailPoints.shift();

        // Walls
        if (b.x <= 5 || b.x >= W - 5) {
            b.vx *= -1;
            sBounce();
        }
        if (b.y <= 5) {
            b.vy *= -1;
            sBounce();
        }

        // Player paddle collision
        if (b.vy > 0 && b.y >= playerPaddle.y && b.y <= playerPaddle.y + playerPaddle.h &&
            b.x >= playerPaddle.x && b.x <= playerPaddle.x + playerPaddle.w) {
            b.vy = -Math.abs(b.vy);
            // Angle based on hit position
            const hitPos = (b.x - playerPaddle.x) / playerPaddle.w - 0.5;
            b.vx = hitPos * 8;
            sPaddle();
        }

        // AI paddle collision
        if (b.vy < 0 && b.y >= aiPaddle.y && b.y <= aiPaddle.y + aiPaddle.h &&
            b.x >= aiPaddle.x && b.x <= aiPaddle.x + aiPaddle.w) {
            b.vy = Math.abs(b.vy);
            const hitPos = (b.x - aiPaddle.x) / aiPaddle.w - 0.5;
            b.vx = hitPos * 6;
            sBounce();
        }

        // Brick collision
        for (let j = bricks.length - 1; j >= 0; j--) {
            const br = bricks[j];
            if (!br.alive) continue;
            if (b.x + 5 > br.x && b.x - 5 < br.x + br.w && b.y + 5 > br.y && b.y - 5 < br.y + br.h) {
                // Determine side
                const overlapLeft = b.x + 5 - br.x;
                const overlapRight = br.x + br.w - (b.x - 5);
                const overlapTop = b.y + 5 - br.y;
                const overlapBottom = br.y + br.h - (b.y - 5);
                const minOverlap = Math.min(overlapLeft, overlapRight, overlapTop, overlapBottom);

                if (minOverlap === overlapTop || minOverlap === overlapBottom) {
                    b.vy *= -1;
                } else {
                    b.vx *= -1;
                }

                br.alive = false;
                score += 10;
                sBreak();
                spawnParticles(br.x + br.w / 2, br.y + br.h / 2, br.color, 8);
            }
        }

        // Ball lost
        if (b.y > H + 20) {
            balls.splice(i, 1);
            lives--;
            sLife();
            if (lives <= 0) {
                state = 'lose';
                sLose();
            } else {
                initBall();
            }
        }
    }

    // Win check
    if (bricks.every(b => !b.alive)) {
        state = 'win';
        sWin();
    }

    // Particles
    for (let i = particles.length - 1; i >= 0; i--) {
        const p = particles[i];
        p.x += p.vx;
        p.y += p.vy;
        p.vy += 0.2;
        p.life -= 0.02;
        if (p.life <= 0) particles.splice(i, 1);
    }
}

// Draw
function draw() {
    ctx.fillStyle = '#0a0a1a';
    ctx.fillRect(0, 0, W, H);

    // Grid lines (subtle)
    ctx.strokeStyle = 'rgba(50, 50, 100, 0.2)';
    ctx.lineWidth = 1;
    for (let x = 0; x < W; x += 40) {
        ctx.beginPath(); ctx.moveTo(x, 0); ctx.lineTo(x, H); ctx.stroke();
    }
    for (let y = 0; y < H; y += 40) {
        ctx.beginPath(); ctx.moveTo(0, y); ctx.lineTo(W, y); ctx.stroke();
    }

    // Bricks
    for (const br of bricks) {
        if (!br.alive) continue;
        ctx.fillStyle = br.color;
        ctx.shadowColor = br.color;
        ctx.shadowBlur = 8;
        ctx.fillRect(br.x, br.y, br.w, br.h);
        ctx.shadowBlur = 0;
        // Highlight
        ctx.fillStyle = 'rgba(255,255,255,0.2)';
        ctx.fillRect(br.x + 2, br.y + 2, br.w - 4, 3);
    }

    // AI Paddle
    ctx.shadowColor = '#ff4444';
    ctx.shadowBlur = 12;
    ctx.fillStyle = '#ff4444';
    ctx.fillRect(aiPaddle.x, aiPaddle.y, aiPaddle.w, aiPaddle.h);
    ctx.shadowBlur = 0;
    // Label
    ctx.fillStyle = 'rgba(255,68,68,0.7)';
    ctx.font = '10px monospace';
    ctx.textAlign = 'center';
    ctx.fillText('AI', aiPaddle.x + aiPaddle.w / 2, aiPaddle.y - 5);

    // Player Paddle
    ctx.shadowColor = '#44ff88';
    ctx.shadowBlur = 12;
    ctx.fillStyle = '#44ff88';
    ctx.fillRect(playerPaddle.x, playerPaddle.y, playerPaddle.w, playerPaddle.h);
    ctx.shadowBlur = 0;

    // Ball trail
    if (balls.length > 0) {
        for (let i = 0; i < trailPoints.length; i++) {
            const t = trailPoints[i];
            const alpha = (i / trailPoints.length) * 0.5;
            ctx.fillStyle = `rgba(255, 255, 100, ${alpha})`;
            const size = 4 * (i / trailPoints.length);
            ctx.beginPath();
            ctx.arc(t.x, t.y, size, 0, Math.PI * 2);
            ctx.fill();
        }
    }

    // Ball
    for (const b of balls) {
        ctx.shadowColor = '#ffff44';
        ctx.shadowBlur = 15;
        ctx.fillStyle = '#ffffaa';
        ctx.beginPath();
        ctx.arc(b.x, b.y, 6, 0, Math.PI * 2);
        ctx.fill();
        ctx.shadowBlur = 0;
    }

    // Particles
    for (const p of particles) {
        ctx.fillStyle = p.color.replace(')', `, ${p.life})`).replace('hsl', 'hsla');
        ctx.globalAlpha = p.life;
        ctx.fillRect(p.x - p.size / 2, p.y - p.size / 2, p.size, p.size);
        ctx.globalAlpha = 1;
    }

    // HUD
    if (state === 'playing' || state === 'win' || state === 'lose') {
        ctx.fillStyle = '#fff';
        ctx.font = 'bold 16px monospace';
        ctx.textAlign = 'left';
        ctx.fillText(`SCORE: ${score}`, 20, 30);
        ctx.textAlign = 'right';
        ctx.fillText(`LIVES: ${'♥'.repeat(lives)}`, W - 20, 30);

        // Bricks remaining
        const remaining = bricks.filter(b => b.alive).length;
        ctx.textAlign = 'center';
        ctx.fillStyle = 'rgba(255,255,255,0.5)';
        ctx.font = '12px monospace';
        ctx.fillText(`BRICKS: ${remaining}`, W / 2, 30);
    }

    // Overlays
    if (state === 'start') {
        drawOverlay('PONG BREAKOUT', [
            'Break all the bricks to win!',
            'But be careful - an AI Pong paddle guards them.',
            '',
            'CONTROLS:',
            'Mouse / Arrow Keys - Move paddle',
            'Click / Any Key - Start Game'
        ]);
    } else if (state === 'win') {
        drawOverlay('YOU WIN!', [
            `Final Score: ${score}`,
            'All bricks destroyed!',
            '',
            'Click or press any key to play again'
        ]);
    } else if (state === 'lose') {
        drawOverlay('GAME OVER', [
            `Score: ${score}`,
            '',
            'Click or press any key to try again'
        ]);
    }
}

function drawOverlay(title, lines) {
    ctx.fillStyle = 'rgba(10, 10, 26, 0.85)';
    ctx.fillRect(0, 0, W, H);

    ctx.textAlign = 'center';
    ctx.shadowColor = '#44ff88';
    ctx.shadowBlur = 20;
    ctx.fillStyle = '#44ff88';
    ctx.font = `bold ${Math.min(48, W * 0.06)}px monospace`;
    ctx.fillText(title, W / 2, H * 0.3);
    ctx.shadowBlur = 0;

    ctx.fillStyle = '#ccc';
    ctx.font = `${Math.min(16, W * 0.02)}px monospace`;
    lines.forEach((line, i) => {
        ctx.fillText(line, W / 2, H * 0.4 + i * 28);
    });
}

// Game Loop
let lastTime = 0;
function loop(time) {
    const dt = time - lastTime;
    lastTime = time;
    update();
    draw();
    requestAnimationFrame(loop);
}

initBricks();
initPaddles();
requestAnimationFrame(loop);
</script>
