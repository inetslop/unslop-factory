const canvas = document.getElementById('c');
const ctx = canvas.getContext('2d');
let W, H;

function resize() {
W = window.innerWidth; H = window.innerHeight;
canvas.width = W; canvas.height = H;
}
window.addEventListener('resize', resize);
resize();

// 3D Camera (top-down angled view)
const cam = { x: 0, y: -8, z: 12, fov: 500 };
let playerX = 0, playerZ = 0;

function project(x, y, z) {
// Translate relative to camera
let dx = x - cam.x;
let dy = y - cam.y;
let dz = z - cam.z;
if (dz < 0.1) dz = 0.1;
let scale = cam.fov / dz;
let sx = W / 2 + dx * scale;
let sy = H / 2 - dy * scale;
return { x: sx, y: sy, s: scale };
}

function drawPoly3D(pts, color, alpha) {
ctx.beginPath();
let projected = pts.map(p => project(p[0], p[1], p[2]));
if (projected.some(p => p.s < 0)) return;
ctx.moveTo(projected[0].x, projected[0].y);
for (let i = 1; i < projected.length; i++) ctx.lineTo(projected[i].x, projected[i].y);
ctx.closePath();
if (alpha !== undefined) {
ctx.globalAlpha = alpha;
}
ctx.fillStyle = color;
ctx.fill();
ctx.globalAlpha = 1;
}

function drawBox(cx, cz, w, h, d, colorTop, colorSide, yOff) {
yOff = yOff || 0;
let x0 = cx - w/2, x1 = cx + w/2;
let z0 = cz - d/2, z1 = cz + d/2;
let y0 = yOff, y1 = yOff + h;
// Top face
drawPoly3D([[x0,y1,z0],[x1,y1,z0],[x1,y1,z1],[x0,y1,z1]], colorTop);
// Front face (facing -z)
drawPoly3D([[x0,y0,z0],[x1,y0,z0],[x1,y1,z0],[x0,y1,z0]], colorSide);
// Right face
drawPoly3D([[x1,y0,z0],[x1,y0,z1],[x1,y1,z1],[x1,y1,z0]], shadeColor(colorSide, -20));
// Left face
drawPoly3D([[x0,y0,z1],[x0,y0,z0],[x0,y1,z0],[x0,y1,z1]], shadeColor(colorSide, -40));
}

function shadeColor(hex, amt) {
let r = parseInt(hex.slice(1,3),16)+amt;
let g = parseInt(hex.slice(3,5),16)+amt;
let b = parseInt(hex.slice(5,7),16)+amt;
r = Math.max(0,Math.min(255,r)); g = Math.max(0,Math.min(255,g)); b = Math.max(0,Math.min(255,b));
return `rgb(${r},${g},${b})`;
}

function drawSphere(cx, cy, cz, r, color) {
let p = project(cx, cy, cz);
let sr = r * p.s;
if (sr < 0) return;
let grad = ctx.createRadialGradient(p.x - sr*0.3, p.y - sr*0.3, sr*0.1, p.x, p.y, sr);
grad.addColorStop(0, '#fff');
grad.addColorStop(0.4, color);
grad.addColorStop(1, 'rgba(0,0,0,0)');
ctx.beginPath();
ctx.arc(p.x, p.y, sr, 0, Math.PI*2);
ctx.fillStyle = grad;
ctx.fill();
}

// Game state
let score = 0, timeLeft = 60, gameRunning = false, gameOver = false;

let audioCtx = null;
function playSound(freq, dur, type) {
if (!audioCtx) return;
let osc = audioCtx.createOscillator();
let gain = audioCtx.createGain();
osc.type = type || 'sine';
osc.frequency.value = freq;
gain.gain.setValueAtTime(0.15, audioCtx.currentTime);
gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
osc.connect(gain); gain.connect(audioCtx.destination);
osc.start(); osc.stop(audioCtx.currentTime + dur);
}

function drawGround() {
let gridSize = 40;
let step = 2;
for (let x = -gridSize/2; x < gridSize/2; x += step) {
for (let z = -gridSize/2; z < gridSize/2; z += step) {

let color = `rgb(${brightness},${brightness+5},${brightness+15})`;
drawPoly3D([[x,0,z],[x+step,0,z],[x+step,0,z+step],[x,0,z+step]], color);
// Grid lines

let p1 = project(x, 0, z);
let p2 = project(x+step, 0, z);
let p3 = project(x+step, 0, z+step);
let p4 = project(x, 0, z+step);
ctx.strokeStyle = 'rgba(0,255,255,0.1)';
ctx.lineWidth = 0.5;
ctx.moveTo(p1.x,p1.y); ctx.lineTo(p2.x,p2.y); ctx.lineTo(p3.x,p3.y); ctx.lineTo(p4.x,p4.y); ctx.closePath(); ctx.stroke();
}
}
// Border walls
drawBox(0, -gridSize/2-0.5, gridSize, 2, 1, '#1a3a5a', '#0f2a4a');

}

function drawPlayer() {
// Player is a 3D block with glow

drawBox(playerX, playerZ, 1.2, 1.4+bobY, 1.2, '#00ffcc', '#00aa88');

grad.addColorStop(1, 'rgba(0,255,200,0)');
ctx.beginPath(); ctx.arc(p.x, p.y, glowR*2, 0, Math.PI*2);
ctx.fillStyle = grad; ctx.fill();
}

function drawOrbs(t) {
for (let o of orbs) {
let bobY = 0.8 + Math.sin(t*0.003 + o.phase)*0.3;
drawSphere(o.x, bobY, o.z, 0.4, '#00ff88');
}
}

function drawHazards(t) {
for (let h of hazards) {
let pulse = 1 + Math.sin(t*0.005)*0.1;
drawBox(h.x, h.z, 1*pulse, 1.5, 1*pulse, '#ff4444', '#cc2222');
// Danger glow
let p = project(h.x, 1.5, h.z);
let gr = 20 * p.s / cam.fov * 5;
let g = ctx.createRadialGradient(p.x, p.y, 0, p.x, p.y, gr*2);
g.addColorStop(0, 'rgba(255,50,50,0.25)');
g.addColorStop(1, 'rgba(255,50,50,0)');
ctx.beginPath(); ctx.arc(p.x, p.y, gr*2, 0, Math.PI*2);
ctx.fillStyle = g; ctx.fill();
}
}

function update(dt, t) {
// Player movement
let speed = 8 * dt;
if (keys['w'] || keys['arrowup']) playerZ -= speed;
if (keys['s'] || keys['arrowdown']) playerZ += speed;
if (keys['a'] || keys['arrowleft']) playerX -= speed;
if (keys['d'] || keys['arrowright']) playerX += speed;

// Bounds
playerX = Math.max(-18, Math.min(18, playerX));
playerZ = Math.max(-18, Math.min(18, playerZ));

// Move hazards
for (let h of hazards) {
h.angle += h.speed * dt;
h.x += Math.cos(h.angle) * h.speed * 0.5 * dt;
h.z += Math.sin(h.angle) * h.speed * 0.5 * dt;
if (Math.abs(h.x) > 17) h.angle = Math.PI - h.angle;
if (Math.abs(h.z) > 17) h.angle = -h.angle;
h.x = Math.max(-17, Math.min(17, h.x));
h.z = Math.max(-17, Math.min(17, h.z));
}

// Collect orbs
for (let i = orbs.length-1; i >= 0; i--) {
let o = orbs[i];
let dx = playerX - o.x, dz = playerZ - o.z;
if (dx*dx+dz*dz < 2.5) {
score += 10;
orbs.splice(i, 1);
if (audioCtx) playSound(880, 0.15, 'sine');
setTimeout(() => { if(audioCtx) playSound(1100, 0.1, 'sine'); }, 50);
// Respawn
orbs.push({ x: (Math.random()-0.5)*30, z: (Math.random()-0.5)*30, phase: Math.random()*Math.PI*2 });
}
}

// Hazard collision
for (let h of hazards) {
let dx = playerX - h.x, dz = playerZ - h.z;
if (dx*dx+dz*dz < 1.8) {
score = Math.max(0, score - 5);
if (audioCtx) playSound(200, 0.3, 'sawtooth');
// Push player away
let dist = Math.sqrt(dx*dx+dz*dz)||1;
playerX += (dx/dist)*2;
playerZ += (dz/dist)*2;
}
}

// Timer
timeLeft -= dt;
if (timeLeft <= 0) { timeLeft = 0; endGame(); }
}

function render(t) {
ctx.fillStyle = '#0a0a1a';
ctx.fillRect(0, 0, W, H);

// Camera follows player with smoothing
cam.x += (playerX - cam.x) * 0.08;
cam.z += (playerZ - cam.z) * 0.08;
cam.y = -8;

drawGround();
drawObstacles();

// Sort draw order by z (painter's algorithm approximation)
let allEntities = [];
for (let o of orbs) allEntities.push({ type:'orb', obj:o, z: o.z });
for (let h of hazards) allEntities.push({ type:'haz', obj:h, z: h.z });
allEntities.push({ type:'player', z: playerZ });
allEntities.sort((a,b) => a.z - b.z);

for (let e of allEntities) {
if (e.type === 'orb') drawOrbs(t); // draw all orbs in one pass for simplicity
else if (e.type === 'haz') drawHazards(t);
else if (e.type === 'player') drawPlayer();
}

}

function loop(now) {
if (!gameRunning) return;
let dt = Math.min((now - lastTime)/1000, 0.05);
lastTime = now;

update(dt, now);
render(now);

// UI
document.getElementById('score').textContent = 'SCORE: ' + score;
document.getElementById('timer').textContent = 'TIME: ' + Math.ceil(timeLeft);

}

}
