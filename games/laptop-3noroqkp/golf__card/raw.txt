let audioCtx = null;
function initAudio() {
    if (!audioCtx) audioCtx = new (window.AudioContext || webkitAudioContext)();
}
function playSound(freq, dur, type, vol) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = type || 'sine';
    osc.frequency.value = freq;
    gain.gain.setValueAtTime(vol || 0.3, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
    osc.connect(gain); gain.connect(audioCtx.destination);
    osc.start(); osc.stop(audioCtx.currentTime + dur);
}

// Game State
let gameState = 'start'; // start, playing, holeComplete, gameOver
let currentHole = 0;
let totalStrokes = 0;
let holeStrokes = 0;
let ball = { x: 0, y: 0, vx: 0, vy: 0 };

const CARD_SUITS = ['♠','♥','♦','♣'];

function createDeck() {
    let d = [];
    for (let s = 0; s < 4; s++)
        for (let c = 0; c < 13; c++)
            d.push({ suit: CARD_SUITS[s], name: CARD_NAMES[c], value: c + 1, red: s === 1 || s === 2 });
    return shuffle(d);
}

function shuffle(arr) {
    for (let i = arr.length - 1; i > 0; i--) {
        const j = Math.floor(Math.random() * (i + 1));
        [arr[i], arr[j]] = [arr[j], arr[i]];
    }
    return arr;
}

function drawCards(n) {
    for (let i = 0; i < n; i++) {
        if (deck.length === 0) { deck = createDeck(); }
        hand.push(deck.pop());
    }
}

function shootBall() {
    if (hand.length === 0) return;
    const card = hand[selectedCard];
    const power = card.value * 2.5;

    ball.vy = Math.sin(aimAngle) * power;

    playSound(card.value * 40, 0.2, 'triangle', 0.2);

// Physics
const FRICTION = 0.985;
const SAND_FRICTION = 0.94;
const BALL_RADIUS = 10;
const HOLE_RADIUS = 16;

function updatePhysics() {
    if (!ballActive) return;
    
    ball.x += ball.vx;
    ball.y += ball.vy;
    
    // Friction
    let inSand = false;
    const h = holes[currentHole];
    for (let s of h.sand) {
        const sx = s.x * W, sy = s.y * H, sw = s.w * W, sh = s.h * H;
        if (ball.x > sx && ball.x < sx + sw && ball.y > sy && ball.y < sy + sh) inSand = true;
    }
    const fric = inSand ? SAND_FRICTION : FRICTION;
    ball.vx *= fric;
    ball.vy *= fric;
    
    // Walls (boundary)
    if (ball.x - BALL_RADIUS < 0) { ball.x = BALL_RADIUS; ball.vx *= -0.7; playSound(200, 0.1, 'square', 0.1); }
    if (ball.x + BALL_RADIUS > W) { ball.x = W - BALL_RADIUS; ball.vx *= -0.7; playSound(200, 0.1, 'square', 0.1); }
    if (ball.y - BALL_RADIUS < 0) { ball.y = BALL_RADIUS; ball.vy *= -0.7; playSound(200, 0.1, 'square', 0.1); }
    if (ball.y + BALL_RADIUS > H) { ball.y = H - BALL_RADIUS; ball.vy *= -0.7; playSound(200, 0.1, 'square', 0.1); }
    
    // Obstacle walls
    for (let w of h.walls) {
        const wx = w.x * W, wy = w.y * H, ww = w.w * W, wh = w.h * H;
        if (ball.x + BALL_RADIUS > wx && ball.x - BALL_RADIUS < wx + ww && ball.y + BALL_RADIUS > wy && ball.y - BALL_RADIUS < wy + wh) {
            // Find closest edge and bounce
            const overlapL = ball.x + BALL_RADIUS - wx;
            const overlapR = (wx + ww) - (ball.x - BALL_RADIUS);
            const overlapT = ball.y + BALL_RADIUS - wy;
            const overlapB = (wy + wh) - (ball.y - BALL_RADIUS);
            const minO = Math.min(overlapL, overlapR, overlapT, overlapB);
            if (minO === overlapL) { ball.x = wx - BALL_RADIUS; ball.vx *= -0.7; }
            else if (minO === overlapR) { ball.x = wx + ww + BALL_RADIUS; ball.vx *= -0.7; }
            else if (minO === overlapT) { ball.y = wy - BALL_RADIUS; ball.vy *= -0.7; }
            else if (minO === overlapB) { ball.y = wy + wh + BALL_RADIUS; ball.vy *= -0.7; }
            playSound(200, 0.1, 'square', 0.1);
        }
    }
    
    // Water hazards
    for (let wt of h.water) {
        const wx = wt.x * W, wy = wt.y * H, ww = wt.w * W, wh = wt.h * H;
        if (ball.x > wx && ball.x < wx + ww && ball.y > wy && ball.y < wy + wh) {
            // Penalty: reset to start
            playSound(150, 0.4, 'sawtooth', 0.3);
            messages.push({ text: '💦 Water! +1 penalty', x: ball.x, y: ball.y - 30, life: 90 });
            holeStrokes++;
            ball.x = h.ballStart * W;
            ball.y = h.ballStart * H;
            ball.vx = 0; ball.vy = 0;
        }
    }
    
    // Check hole
    const hx = h.holePos.x * W, hy = h.holePos.y * H;
    const dist = Math.hypot(ball.x - hx, ball.y - hy);
    if (dist < HOLE_RADIUS) {
        ball.vx = 0; ball.vy = 0;
        ballActive = false;
        totalStrokes += holeStrokes;
        gameState = 'holeComplete';
        holeCompleteTimer = 0;
        playSound(600, 0.3, 'sine', 0.4);
        setTimeout(() => playSound(800, 0.3, 'sine', 0.3), 150);
        // Particles
        for (let i = 0; i < 20; i++) {
            const a = Math.random() * Math.PI * 2;
            particles.push({ x: hx, y: hy, vx: Math.cos(a) * (2 + Math.random()*3), vy: Math.sin(a) * (2 + Math.random()*3), life: 40, color: '#FFD700' });
        }
    }
    
    // Stop ball
    if (Math.abs(ball.vx) < 0.1 && Math.abs(ball.vy) < 0.1) {
        ball.vx = 0; ball.vy = 0;
        ballActive = false;
        // Replace played card
        hand.splice(selectedCard, 1);
        drawCards(1);
    }
}

function updateParticles() {
    for (let i = particles.length - 1; i >= 0; i--) {
        const p = particles[i];
        p.x += p.vx; p.y += p.vy;
        p.vy += 0.1;
        p.life--;
        if (p.life <= 0) particles.splice(i, 1);
    }
}

function updateMessages() {
    for (let i = messages.length - 1; i >= 0; i--) {
        messages[i].y -= 0.5;
        messages[i].life--;
        if (messages[i].life <= 0) messages.splice(i, 1);
    }
}

}

function drawAimLine() {
    if (ballActive) return;
    const card = hand[selectedCard];
    if (!card) return;
    
    aimAngle = Math.atan2(mousePos.y - ball.y, mousePos.x - ball.x);

    ctx.strokeStyle = 'rgba(255,255,255,0.6)';
    ctx.lineWidth = 3;
    ctx.setLineDash([5, 5]);
    ctx.beginPath();
    ctx.moveTo(ball.x, ball.y);
    ctx.lineTo(ball.x + Math.cos(aimAngle) * len, ball.y + Math.sin(aimAngle) * len);
    ctx.stroke();
    ctx.setLineDash([]);
    
    // Arrow head
    const ax = ball.x + Math.cos(aimAngle) * len;
    const ay = ball.y + Math.sin(aimAngle) * len;
    ctx.fillStyle = 'rgba(255,255,255,0.6)';
    ctx.beginPath();
    ctx.moveTo(ax, ay);
    ctx.lineTo(ax - Math.cos(aimAngle - 0.4) * 12, ay - Math.sin(aimAngle - 0.4) * 12);
    ctx.lineTo(ax - Math.cos(aimAngle + 0.4) * 12, ay - Math.sin(aimAngle + 0.4) * 12);
    ctx.fill();
}

    ctx.fillStyle = '#fff';
    ctx.font = '24px Arial';
    ctx.fillText('A Golf Game Played with Cards', W/2, H/3 + 50);

function drawHoleComplete() {
    holeCompleteTimer++;
    const h = holes[currentHole];
    const diff = holeStrokes - h.par;
    
    ctx.fillStyle = 'rgba(0,0,0,0.7)';
    ctx.fillRect(0, 0, W, H);
    
    ctx.fillStyle = '#FFD700';
    ctx.font = 'bold 48px Arial';
    ctx.textAlign = 'center';
    ctx.fillText('HOLE COMPLETE!', W/2, H/3);
    
    let resultText = '';
    if (diff < -1) resultText = '🏆 ACE!!';
    else if (diff === -1) resultText = 'Birdie! 🐦';
    else if (diff === 0) resultText = 'Par ✅';
    else if (diff === 1) resultText = 'Bogey 😅';
    else resultText = '+' + diff + ' Over Par 😬';
    
    ctx.fillStyle = '#fff';
    ctx.font = '32px Arial';
    ctx.fillText(resultText, W/2, H/3 + 60);
    ctx.font = '24px Arial';
    ctx.fillText('Strokes: ' + holeStrokes + ' / Par: ' + h.par, W/2, H/3 + 110);
    
    const pulse = Math.sin(Date.now() / 500) * 0.3 + 0.7;
    ctx.fillStyle = 'rgba(255,255,255,' + pulse + ')';
    ctx.font = '20px Arial';
    ctx.fillText('Press SPACE for next hole', W/2, H - 100);
}

function drawGameOver() {
    ctx.fillStyle = 'rgba(0,0,0,0.8)';
    ctx.fillRect(0, 0, W, H);
    
    // Calculate total par
    let totalPar = 0;
    for (let h of holes) totalPar += h.par;
    const diff = totalStrokes - totalPar;
    
    ctx.fillStyle = '#FFD700';
    ctx.font = 'bold 56px Arial';
    ctx.textAlign = 'center';
    ctx.fillText('🏆 GAME COMPLETE 🏆', W/2, H/4);
    
    ctx.fillStyle = '#fff';
    ctx.font = '32px Arial';
    ctx.fillText('Total Strokes: ' + totalStrokes, W/2, H/4 + 70);
    ctx.fillText('Total Par: ' + totalPar, W/2, H/4 + 115);
    
    let finalText;
    if (diff < -2) finalText = '⭐ GOLF GENIUS! ⭐';
    else if (diff < 0) finalText = '🌟 Excellent Golf! 🌟';
    else if (diff === 0) finalText = '✅ Solid Game! ✅';
    else if (diff < 4) finalText = '👍 Not Bad! 👍';
    else finalText = '💪 Keep Practicing! 💪';
    
    ctx.fillStyle = '#2ecc71';
    ctx.font = 'bold 36px Arial';
    ctx.fillText(finalText, W/2, H/4 + 180);
    
    // Score breakdown
    ctx.fillStyle = '#ccc';
    ctx.font = '18px Arial';
    for (let i = 0; i < holes.length; i++) {
        ctx.fillText('Hole ' + (i+1) + ': Par ' + holes[i].par, W/2 - 100, H/4 + 230 + i * 30);
    }
    
    const pulse = Math.sin(Date.now() / 500) * 0.3 + 0.7;
    ctx.fillStyle = 'rgba(255,255,255,' + pulse + ')';
    ctx.font = '24px Arial';
    ctx.fillText('Press SPACE to Play Again', W/2, H - 80);
}

function drawParticles() {
    for (let p of particles) {
        ctx.fillStyle = p.color;
        ctx.globalAlpha = p.life / 40;
        ctx.beginPath();
        ctx.arc(p.x, p.y, 3, 0, Math.PI * 2);
        ctx.fill();
    }
    ctx.globalAlpha = 1;
}

function drawMessages() {
    for (let m of messages) {
        ctx.fillStyle = 'rgba(255,255,255,' + (m.life / 60) + ')';
        ctx.font = 'bold 16px Arial';
        ctx.textAlign = 'center';
        ctx.fillText(m.text, m.x, m.y);
    }
}

// Main loop
function gameLoop() {
    ctx.clearRect(0, 0, W, H);
    
    if (gameState === 'start') {
        drawStartScreen();
    } else if (gameState === 'playing') {
        updatePhysics();
        drawGolfCourse();
        drawBall();
        drawAimLine();
        drawParticles();
        drawMessages();
        drawCards();
        drawHUD();
    } else if (gameState === 'holeComplete') {
        drawGolfCourse();
        drawBall();
        drawParticles();
        drawHoleComplete();
    } else if (gameState === 'gameOver') {
        drawGameOver();
    }
    
    updateParticles();
    updateMessages();
    
    requestAnimationFrame(gameLoop);
}

// Polyfill roundRect if needed
if (!CanvasRenderingContext2D.prototype.roundRect) {
    CanvasRenderingContext2D.prototype.roundRect = function(x, y, w, h, r) {
        if (typeof r === 'number') r = [r, r, r, r];
        const [tl, tr, br, bl] = Array.isArray(r) ? r : [r, r, r, r];
        this.moveTo(x + tl, y);
        this.lineTo(x + w - tr, y);
        this.arcTo(x + w, y, x + w, y + tr, tr);
        this.lineTo(x + w, y + h - br);
        this.arcTo(x + w, y + h, x + w - br, y + h, br);
        this.lineTo(x + bl, y + h);
        this.arcTo(x, y + h, x, y + h - bl, bl);
        this.lineTo(x, y + tl);
        this.arcTo(x, y, x + tl, y, tl);
        this.closePath();
    };
}

gameLoop();
})();
