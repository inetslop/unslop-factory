const canvas = document.getElementById('game');
const ctx = canvas.getContext('2d');

let W, H;
function resize() {
    W = window.innerWidth;
    H = window.innerHeight;
    canvas.width = W;
    canvas.height = H;
}
window.addEventListener('resize', resize);
resize();

// Audio
let audioCtx = null;
function initAudio() {
    if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
}
function playSound(freq, dur, type) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = type || 'sine';
    osc.frequency.value = freq;
    gain.gain.setValueAtTime(0.3, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + dur);
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + dur);
}

// Game state
let gameState = 'start'; // start, playing, levelComplete, gameOver
let score = 0;
let totalLevels = 5;
let currentLevel = 0;
let shotsUsed = 0;
let maxShots = 3;
let ball = null;
let targets = [];
let obstacles = [];
let particles = [];
let aimAngle = -Math.PI / 4;
let power = 0;
let charging = false;
let mouseDown = false;
let mouseX = 0, mouseY = 0;
let levelCompleteTimer = 0;

const GRAVITY = 0.4;
const BOUNCE_DAMPING = 0.6;
const BALL_RADIUS = 12;

function createLevel(lvl) {
    shotsUsed = 0;
    particles = [];
    ball = null;
    targets = [];
    obstacles = [];
    
    const groundY = H - 80;
    
    // Tee position
    const teeX = 60;
    const teeY = groundY - BALL_RADIUS;
    
    // Create targets based on level
    const numTargets = Math.min(2 + lvl, 5);
    for (let i = 0; i < numTargets; i++) {
        const tx = 200 + Math.random() * (W - 400);
        const ty = groundY - BALL_RADIUS - 10;
        targets.push({ x: tx, y: ty, radius: 18, hit: false, bounce: 0 });
    }
    
    // Create obstacles
    const numObs = lvl + 1;
    for (let i = 0; i < numObs; i++) {
        if (Math.random() > 0.4) {
            const ox = 150 + Math.random() * (W - 300);
            const oy = groundY - 60 - Math.random() * (H * 0.3);
            obstacles.push({ x: ox, y: oy, w: 20, h: 80 + Math.random() * 60 });
        } else {
            const ox = 150 + Math.random() * (W - 300);
            const oy = groundY - 40;
            obstacles.push({ x: ox, y: oy, w: 80 + Math.random() * 60, h: 20 });
        }
    }
    
    // Reset ball to tee
    ball = { x: teeX, y: teeY, vx: 0, vy: 0, active: false, onTee: true };
}

function shootBall() {
    if (!ball || !ball.onTee || shotsUsed >= maxShots) return;
    const speed = power * 15;
    ball.vx = Math.cos(aimAngle) * speed;
    ball.vy = Math.sin(aimAngle) * speed;
    ball.active = true;
    ball.onTee = false;
    shotsUsed++;
    playSound(400, 0.1, 'square');
}

function updateBall() {
    if (!ball || !ball.active) return;
    
    ball.vy += GRAVITY;
    ball.x += ball.vx;
    ball.y += ball.vy;
    
    const groundY = H - 80 - BALL_RADIUS;
    
    // Floor bounce
    if (ball.y > groundY) {
        ball.y = groundY;
        ball.vy *= -BOUNCE_DAMPING;
        ball.vx *= 0.9;
        if (Math.abs(ball.vy) < 1) { ball.vy = 0; }
        playSound(200, 0.05, 'triangle');
    }
    
    // Wall bounces
    if (ball.x < BALL_RADIUS) { ball.x = BALL_RADIUS; ball.vx *= -BOUNCE_DAMPING; }
    if (ball.x > W - BALL_RADIUS) { ball.x = W - BALL_RADIUS; ball.vx *= -BOUNCE_DAMPING; }
    
    // Obstacle collisions
    for (let obs of obstacles) {
        if (ball.x + BALL_RADIUS > obs.x && ball.x - BALL_RADIUS < obs.x + obs.w &&
            ball.y + BALL_RADIUS > obs.y && ball.y - BALL_RADIUS < obs.y + obs.h) {
            // Determine side
            const overlapL = ball.x + BALL_RADIUS - obs.x;
            const overlapR = obs.x + obs.w - (ball.x - BALL_RADIUS);
            const overlapT = ball.y + BALL_RADIUS - obs.y;
            const overlapB = obs.y + obs.h - (ball.y - BALL_RADIUS);
            
            const minOverlap = Math.min(overlapL, overlapR, overlapT, overlapB);
            
            if (minOverlap === overlapL) { ball.x = obs.x - BALL_RADIUS; ball.vx *= -BOUNCE_DAMPING; }
            else if (minOverlap === overlapR) { ball.x = obs.x + obs.w + BALL_RADIUS; ball.vx *= -BOUNCE_DAMPING; }
            else if (minOverlap === overlapT) { ball.y = obs.y - BALL_RADIUS; ball.vy *= -BOUNCE_DAMPING; }
            else if (minOverlap === overlapB) { ball.y = obs.y + obs.h + BALL_RADIUS; ball.vy *= -BOUNCE_DAMPING; }
            
            playSound(300, 0.05, 'triangle');
        }
    }
    
    // Target collisions
    for (let t of targets) {
        const dx = ball.x - t.x;
        const dy = ball.y - t.y;
        const dist = Math.sqrt(dx * dx + dy * dy);
        if (dist < BALL_RADIUS + t.radius && !t.hit) {
            t.hit = true;
            t.bounce = 1;
            score += 100;
            playSound(800, 0.2, 'sine');
            for (let i = 0; i < 15; i++) {
                const angle = Math.random() * Math.PI * 2;
                const speed = 2 + Math.random() * 4;
                particles.push({ x: t.x, y: t.y, vx: Math.cos(angle) * speed, vy: Math.sin(angle) * speed - 3, life: 1, color: '#ffd700' });
            }
        }
    }
    
    // Stop ball
    const speed = Math.sqrt(ball.vx * ball.vx + ball.vy * ball.vy);
    if (speed < 0.5 && ball.y >= groundY - 2) {
        ball.vx = 0;
        ball.vy = 0;
        ball.active = false;
        
        // Check if all targets hit
        if (targets.every(t => t.hit)) {
            score += maxShots - shotsUsed + 1;
            gameState = 'levelComplete';
            levelCompleteTimer = 0;
        } else if (shotsUsed >= maxShots) {
            gameState = 'gameOver';
        } else {
            // Reset ball to tee
            setTimeout(() => {
                ball.onTee = true;
                ball.x = 60;
                ball.y = H - 80 - BALL_RADIUS;
                ball.vx = 0;
                ball.vy = 0;
            }, 500);
        }
    }
}

function updateParticles() {
    for (let p of particles) {
        p.x += p.vx;
        p.y += p.vy;
        p.vy += 0.2;
        p.life -= 0.02;
    }
    particles = particles.filter(p => p.life > 0);
}

function drawBackground() {
    // Sky gradient
    const grad = ctx.createLinearGradient(0, 0, 0, H);
    grad.addColorStop(0, '#1a1a2e');
    grad.addColorStop(0.6, '#16213e');
    grad.addColorStop(1, '#0f3460');
    ctx.fillStyle = grad;
    ctx.fillRect(0, 0, W, H);
    
    // Stars
    for (let i = 0; i < 50; i++) {
        const sx = (i * 137.5) % W;
        const sy = (i * 97.3) % (H * 0.5);
        ctx.fillStyle = `rgba(255,255,255,${0.3 + Math.sin(Date.now() / 1000 + i) * 0.2})`;
        ctx.fillRect(sx, sy, 2, 2);
    }
    
    // Ground
    const groundY = H - 80;
    ctx.fillStyle = '#2d5a27';
    ctx.fillRect(0, groundY, W, 80);
    ctx.fillStyle = '#1e4a1a';
    ctx.fillRect(0, groundY, W, 5);
    
    // Grass tufts
    for (let i = 0; i < W / 20; i++) {
        const gx = i * 20 + 10;
        ctx.strokeStyle = '#3d7a37';
        ctx.lineWidth = 1;
        ctx.beginPath();
        ctx.moveTo(gx, groundY);
        ctx.lineTo(gx - 3, groundY - 8);
        ctx.stroke();
    }
}

function drawObstacles() {
    for (let obs of obstacles) {
        ctx.fillStyle = '#5c4033';
        ctx.fillRect(obs.x, obs.y, obs.w, obs.h);
        ctx.strokeStyle = '#3d2914';
        ctx.lineWidth = 2;
        ctx.strokeRect(obs.x, obs.y, obs.w, obs.h);
        
        // Wood grain
        ctx.strokeStyle = 'rgba(0,0,0,0.2)';
        ctx.lineWidth = 1;
        if (obs.h > obs.w) {
            for (let i = 0; i < obs.h / 15; i++) {
                ctx.beginPath();
                ctx.moveTo(obs.x + 3, obs.y + i * 15);
                ctx.lineTo(obs.x + obs.w - 3, obs.y + i * 15);
                ctx.stroke();
            }
        } else {
            for (let i = 0; i < obs.w / 15; i++) {
                ctx.beginPath();
                ctx.moveTo(obs.x + i * 15, obs.y + 3);
                ctx.lineTo(obs.x + i * 15, obs.y + obs.h - 3);
                ctx.stroke();
            }
        }
    }
}

function drawTargets() {
    for (let t of targets) {
        if (t.hit) {
            // Hit marker
            ctx.fillStyle = 'rgba(255,215,0,0.3)';
            ctx.beginPath();
            ctx.arc(t.x, t.y, t.radius, 0, Math.PI * 2);
            ctx.fill();
            
            ctx.fillStyle = '#ffd700';
            ctx.font = 'bold 16px Arial';
            ctx.textAlign = 'center';
            ctx.fillText('✓', t.x, t.y + 5);
        } else {
            // Target cup
            const bounceOffset = Math.sin(t.bounce * Math.PI) * 20;
            t.bounce -= 0.05;
            
            ctx.fillStyle = '#ff4444';
            ctx.beginPath();
            ctx.arc(t.x, t.y + bounceOffset, t.radius, 0, Math.PI * 2);
            ctx.fill();
            
            ctx.fillStyle = '#ffffff';
            ctx.beginPath();
            ctx.arc(t.x, t.y + bounceOffset, t.radius * 0.6, 0, Math.PI * 2);
            ctx.fill();
            
            ctx.fillStyle = '#ff4444';
            ctx.beginPath();
            ctx.arc(t.x, t.y + bounceOffset, t.radius * 0.3, 0, Math.PI * 2);
            ctx.fill();
            
            // Glow
            ctx.shadowColor = '#ff4444';
            ctx.shadowBlur = 10;
            ctx.strokeStyle = '#ff6666';
            ctx.lineWidth = 2;
            ctx.beginPath();
            ctx.arc(t.x, t.y + bounceOffset, t.radius + 3, 0, Math.PI * 2);
            ctx.stroke();
            ctx.shadowBlur = 0;
        }
    }
}

function drawBall() {
    if (!ball) return;
    
    // Ball
    const grad = ctx.createRadialGradient(ball.x - 3, ball.y - 3, 2, ball.x, ball.y, BALL_RADIUS);
    grad.addColorStop(0, '#ffffff');
    grad.addColorStop(1, '#cccccc');
    ctx.fillStyle = grad;
    ctx.beginPath();
    ctx.arc(ball.x, ball.y, BALL_RADIUS, 0, Math.PI * 2);
    ctx.fill();
    
    // Dimpples
    ctx.fillStyle = 'rgba(0,0,0,0.1)';
    for (let i = 0; i < 5; i++) {
        const a = (i / 5) * Math.PI * 2;
        ctx.beginPath();
        ctx.arc(ball.x + Math.cos(a) * 5, ball.y + Math.sin(a) * 5, 2, 0, Math.PI * 2);
        ctx.fill();
    }
    
    // Aim line when on tee
    if (ball.onTee && gameState === 'playing') {
        const aimLen = 80 + power * 100;
        ctx.strokeStyle = 'rgba(255,255,255,0.6)';
        ctx.lineWidth = 2;
        ctx.setLineDash([5, 5]);
        ctx.beginPath();
        ctx.moveTo(ball.x, ball.y);
        ctx.lineTo(ball.x + Math.cos(aimAngle) * aimLen, ball.y + Math.sin(aimAngle) * aimLen);
        ctx.stroke();
        ctx.setLineDash([]);
        
        // Power indicator
        ctx.fillStyle = power > 0.8 ? '#ff4444' : power > 0.5 ? '#ffaa00' : '#44ff44';
        ctx.fillRect(20, H - 30, power * 100, 10);
        ctx.strokeStyle = '#fff';
        ctx.strokeRect(20, H - 30, 100, 10);
    }
}

function drawParticles() {
    for (let p of particles) {
        ctx.fillStyle = `rgba(255,215,0,${p.life})`;
        ctx.beginPath();
        ctx.arc(p.x, p.y, 3 * p.life, 0, Math.PI * 2);
        ctx.fill();
    }
}

function drawHUD() {
    // Score
    ctx.fillStyle = '#fff';
    ctx.font = 'bold 20px Arial';
    ctx.textAlign = 'left';
    ctx.fillText(`Score: ${score}`, 20, 35);
    
    // Level
    ctx.fillText(`Level: ${currentLevel + 1}/${totalLevels}`, 20, 65);
    
    // Shots remaining
    const shotsLeft = maxShots - shotsUsed;
    ctx.fillText(`Shots: ${shotsLeft}`, W - 150, 35);
    
    // Targets remaining
    const targetsLeft = targets.filter(t => !t.hit).length;
    ctx.fillText(`Targets: ${targetsLeft}/${targets.length}`, W - 150, 65);
}

function drawStartScreen() {
    ctx.fillStyle = 'rgba(0,0,0,0.8)';
    ctx.fillRect(0, 0, W, H);
    
    ctx.fillStyle = '#ffd700';
    ctx.font = 'bold 48px Arial';
    ctx.textAlign = 'center';
    ctx.fillText('⛳ GOLF SHOOTER', W / 2, H / 3);
    
    ctx.fillStyle = '#fff';
    ctx.font = '20px Arial';
    ctx.fillText('Aim and shoot the golf ball to hit all targets!', W / 2, H / 3 + 50);
    ctx.fillText('', W / 2, H / 3 + 90);
    ctx.font = '18px Arial';
    ctx.fillText('CONTROLS:', W / 2, H / 3 + 120);
    ctx.fillText('Mouse: Aim', W / 2, H / 3 + 150);
    ctx.fillText('Hold SPACE or Click: Charge Power', W / 2, H / 3 + 180);
    ctx.fillText('Release: Shoot!', W / 2, H / 3 + 210);
    ctx.fillText('R: Restart Level', W / 2, H / 3 + 240);
    
    // Blinking start text
    if (Math.floor(Date.now() / 500) % 2 === 0) {
        ctx.fillStyle = '#44ff44';
        ctx.font = 'bold 24px Arial';

    }

function drawLevelComplete() {
    levelCompleteTimer += 0.016;
    
    ctx.fillStyle = 'rgba(0,0,0,0.7)';
    ctx.fillRect(0, 0, W, H);
    
    ctx.fillStyle = '#44ff44';
    ctx.font = 'bold 42px Arial';
    ctx.textAlign = 'center';
    ctx.fillText('LEVEL COMPLETE!', W / 2, H / 3);
    
    ctx.fillStyle = '#fff';
    ctx.font = '24px Arial';
    ctx.fillText(`Score: ${score}`, W / 2, H / 3 + 50);
    
    if (levelCompleteTimer > 1.5) {
        if (Math.floor(Date.now() / 500) % 2 === 0) {
            ctx.fillStyle = '#44ff44';
            ctx.font = 'bold 20px Arial';
            ctx.fillText('Press SPACE for Next Level', W / 2, H - 80);
        }
    }
}

function drawGameOver() {
    ctx.fillStyle = 'rgba(0,0,0,0.8)';
    ctx.fillRect(0, 0, W, H);
    
    ctx.fillStyle = '#ff4444';
    ctx.font = 'bold 42px Arial';
    ctx.textAlign = 'center';
    ctx.fillText('GAME OVER', W / 2, H / 3);
    
    ctx.fillStyle = '#fff';
    ctx.font = '24px Arial';
    ctx.fillText(`Final Score: ${score}`, W / 2, H / 3 + 50);
    
    if (Math.floor(Date.now() / 500) % 2 === 0) {
        ctx.fillStyle = '#ffd700';
        ctx.font = 'bold 20px Arial';
        ctx.fillText('Press R to Restart', W / 2, H - 80);
    }
}

function drawVictory() {
    ctx.fillStyle = 'rgba(0,0,0,0.8)';
    ctx.fillRect(0, 0, W, H);
    
    ctx.fillStyle = '#ffd700';
    ctx.font = 'bold 42px Arial';
    ctx.textAlign = 'center';
    ctx.fillText('🏆 YOU WIN! 🏆', W / 2, H / 3);
    
    ctx.fillStyle = '#fff';
    ctx.font = '24px Arial';
    ctx.fillText(`Final Score: ${score}`, W / 2, H / 3 + 50);
    
    if (Math.floor(Date.now() / 500) % 2 === 0) {
        ctx.fillStyle = '#44ff44';
        ctx.font = 'bold 20px Arial';
        ctx.fillText('Press R to Play Again', W / 2, H - 80);
    }
}

function gameLoop() {
    updateBall();
    updateParticles();
    
    drawBackground();
    drawObstacles();
    drawTargets();
    drawBall();
    drawParticles();
    drawHUD();
    
    if (gameState === 'start') drawStartScreen();
    else if (gameState === 'levelComplete') drawLevelComplete();
    else if (gameState === 'gameOver') drawGameOver();
    else if (gameState === 'victory') drawVictory();
    
    requestAnimationFrame(gameLoop);
}

// Input
function getAimFromMouse() {
    if (!ball || !ball.onTee) return;
    const dx = mouseX - ball.x;
    const dy = mouseY - ball.y;
    aimAngle = Math.atan2(dy, dx);
    // Clamp to upper half mostly
    if (aimAngle > 0.3) aimAngle = 0.3;
}

canvas.addEventListener('mousemove', (e) => {
    mouseX = e.clientX;
    mouseY = e.clientY;
    getAimFromMouse();
});

canvas.addEventListener('mousedown', (e) => {
    initAudio();
    if (gameState === 'start') {
        gameState = 'playing';
        currentLevel = 0;
        score = 0;
        createLevel(0);
        return;
    }
    if (gameState === 'levelComplete' && levelCompleteTimer > 1.5) {
        currentLevel++;
        if (currentLevel >= totalLevels) {
            gameState = 'victory';
        } else {
            createLevel(currentLevel);
            gameState = 'playing';
        }
        return;
    }
    if (gameState === 'gameOver' || gameState === 'victory') {
        score = 0;
        currentLevel = 0;
        createLevel(0);
        gameState = 'playing';
        return;
    }
    mouseDown = true;
    charging = true;
});

canvas.addEventListener('mouseup', (e) => {
    if (charging && gameState === 'playing') {
        shootBall();
    }
    mouseDown = false;
    charging = false;
    power = 0;
});

window.addEventListener('keydown', (e) => {
    initAudio();
    if (e.code === 'Space') {
        e.preventDefault();
        if (gameState === 'start') {
            gameState = 'playing';
            currentLevel = 0;
            score = 0;
            createLevel(0);
        } else if (gameState === 'levelComplete' && levelCompleteTimer > 1.5) {
            currentLevel++;
            if (currentLevel >= totalLevels) {
                gameState = 'victory';
            } else {
                createLevel(currentLevel);
                gameState = 'playing';
            }
        } else if (gameState === 'gameOver' || gameState === 'victory') {
            score = 0;
            currentLevel = 0;
            createLevel(0);
            gameState = 'playing';
        } else if (!charging) {
            charging = true;
        }
    }
    if (e.code === 'KeyR') {
        if (gameState === 'playing' || gameState === 'gameOver' || gameState === 'victory') {
            score = 0;
            currentLevel = 0;
            createLevel(0);
            gameState = 'playing';
        }
    }
});

window.addEventListener('keyup', (e) => {
    if (e.code === 'Space' && charging && gameState === 'playing') {
        shootBall();
        charging = false;
        power = 0;
    }
});

// Power charging in game loop
const origUpdateBall = updateBall;
function gameLoopWrapper() {
    if (charging) {
        power += 0.015;
        if (power > 1) power = 1;
    }
    origUpdateBall();
}

// Override the game loop to include power charging
function mainLoop() {
    if (charging && gameState === 'playing') {
        power += 0.015;
        if (power > 1) power = 1;
    }
    
    updateBall();
    updateParticles();
    
    drawBackground();
    drawObstacles();
    drawTargets();
    drawBall();
    drawParticles();
    drawHUD();
    
    if (gameState === 'start') drawStartScreen();
    else if (gameState === 'levelComplete') drawLevelComplete();
    else if (gameState === 'gameOver') drawGameOver();
    else if (gameState === 'victory') drawVictory();
    
    requestAnimationFrame(mainLoop);
}

// Start the game
mainLoop();
