// Audio
let audioCtx = null;
function initAudio() {
    if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
}
function playSound(type, freq, dur, vol) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.type = type;
    osc.frequency.value = freq;
    gain.gain.setValueAtTime(vol, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
    osc.start();
    osc.stop(audioCtx.currentTime + dur);
}
function sfxShoot() { playSound('square', 400, 0.1, 0.15); }
function sfxHit() { playSound('sawtooth', 150, 0.2, 0.3); }
function sfxPowerup() { playSound('sine', 800, 0.3, 0.2); }
function sfxRound() { playSound('triangle', 600, 0.4, 0.25); }

// Game State
let gameState = 'menu'; // menu, playing, roundEnd, gameOver
let frameCount = 0;

    else if (gameState === 'roundEnd' && (e.code === 'Space' || e.code === 'Enter')) nextRound();
    else if (gameState === 'gameOver' && (e.code === 'KeyR')) resetGame();
});

// Particles
let particles = [];
function spawnParticles(x, y, color, count, speed) {
    for (let i = 0; i < count; i++) {
        const a = Math.random() * Math.PI * 2;
        const s = speed * (0.5 + Math.random() * 0.5);
        particles.push({ x, y, vx: Math.cos(a) * s, vy: Math.sin(a) * s, life: 30 + Math.random() * 20, maxLife: 50, color });
    }
}

// Power-ups
let powerups = [];
const POWERUP_TYPES = ['speed', 'triple', 'shield', 'heal'];

function spawnPowerup() {
    const type = POWERUP_TYPES[Math.floor(Math.random() * POWERUP_TYPES.length)];
    powerups.push({
        x: 100 + Math.random() * (W - 200),
        y: 150 + Math.random() * (H - 300),
        type,
        life: 600,
        bob: Math.random() * Math.PI * 2
    });

    p2 = createPlayer(W / 2 + 150, H / 2, '#ff7043', { up: 'ArrowUp', down: 'ArrowDown', left: 'ArrowLeft', right: 'ArrowRight', shoot: 'KeyL' });
    projectiles = [];

function endRound(winner) {
    roundScores[winner - 1]++;
    gameState = 'roundEnd';

function updatePlayer(p, opponent) {
    const speed = 4 + (p.speedBoost > 0 ? 3 : 0);
    let ax = 0, ay = 0;
    if (keys[p.controls.up]) ay -= 1;
    if (keys[p.controls.down]) ay += 1;
    if (keys[p.controls.left]) ax -= 1;
    if (keys[p.controls.right]) ax += 1;

    if (ax !== 0 || ay !== 0) {
        const len = Math.sqrt(ax * ax + ay * ay);
        ax /= len; ay /= len;
        p.vx += ax * speed * 0.3;
        p.vy += ay * speed * 0.3;
        p.facing = Math.atan2(ay, ax);
    }

    // Friction
    p.vx *= 0.88;
    p.vy *= 0.88;

    p.x += p.vx;
    p.y += p.vy;

    // Bounds
    const m = p.radius;
    if (p.x < m) { p.x = m; p.vx *= -0.5; }
    if (p.x > W - m) { p.x = W - m; p.vx *= -0.5; }
    if (p.y < 80 + m) { p.y = 80 + m; p.vy *= -0.5; }
    if (p.y > H - m) { p.y = H - m; p.vy *= -0.5; }

    // Trail
    p.trail.push({ x: p.x, y: p.y });
    if (p.trail.length > 12) p.trail.shift();

    // Timers
    if (p.shootCooldown > 0) p.shootCooldown--;
    if (p.speedBoost > 0) p.speedBoost--;
    if (p.tripleShot > 0) p.tripleShot--;
    if (p.shield > 0) p.shield--;
    if (p.invincible > 0) p.invincible--;

    // Shoot
    if (keys[p.controls.shoot] && p.shootCooldown <= 0) {
        shoot(p);
        sfxShoot();
    }
}

function shoot(p) {
    const speed = 8;
    const cos = Math.cos(p.facing);
    const sin = Math.sin(p.facing);
    const isTriple = p.tripleShot > 0;

    if (isTriple) {
        for (let i = -1; i <= 1; i++) {
            const a = p.facing + i * 0.25;
            projectiles.push({ x: p.x + cos * p.radius, y: p.y + sin * p.radius, vx: Math.cos(a) * speed, vy: Math.sin(a) * speed, owner: p === p1 ? 1 : 2, life: 90 });
        }
    } else {
        projectiles.push({ x: p.x + cos * p.radius, y: p.y + sin * p.radius, vx: cos * speed, vy: sin * speed, owner: p === p1 ? 1 : 2, life: 90 });
    }
    p.shootCooldown = isTriple ? 25 : 15;
}

function updateProjectiles() {
    for (let i = projectiles.length - 1; i >= 0; i--) {
        const pr = projectiles[i];
        pr.x += pr.vx;
        pr.y += pr.vy;
        pr.life--;

        if (pr.life <= 0 || pr.x < 0 || pr.x > W || pr.y < 80 || pr.y > H) {
            spawnParticles(pr.x, pr.y, pr.owner === 1 ? '#4fc3f7' : '#ff7043', 5, 2);
            projectiles.splice(i, 1);
            continue;
        }

        // Hit check
        const target = pr.owner === 1 ? p2 : p1;
        const dx = pr.x - target.x;
        const dy = pr.y - target.y;
        if (Math.sqrt(dx * dx + dy * dy) < target.radius + 5) {
            if (target.invincible > 0) {
                projectiles.splice(i, 1);
                continue;
            }
            if (target.shield > 0) {
                spawnParticles(pr.x, pr.y, '#ffeb3b', 8, 3);
                target.shield = 0;
                sfxHit();
                projectiles.splice(i, 1);
                continue;
            }
            const dmg = 15;
            target.hp -= dmg;
            target.invincible = 30;
            target.vx += pr.vx * 0.4;
            target.vy += pr.vy * 0.4;
            sfxHit();
            shakeMag = 8;
            spawnParticles(pr.x, pr.y, pr.owner === 1 ? '#4fc3f7' : '#ff7043', 12, 4);
            projectiles.splice(i, 1);

            if (target.hp <= 0) {
                target.hp = 0;
                endRound(pr.owner);
            }
        }
    }
}

function updatePowerups() {
    for (let i = powerups.length - 1; i >= 0; i--) {
        const pu = powerups[i];
        pu.life--;
        pu.bob += 0.05;
        if (pu.life <= 0) { powerups.splice(i, 1); continue; }

        for (const p of [p1, p2]) {
            const dx = p.x - pu.x;
            const dy = p.y - pu.y;
            if (Math.sqrt(dx * dx + dy * dy) < p.radius + 15) {
                sfxPowerup();
                spawnParticles(pu.x, pu.y, '#ffeb3b', 10, 3);
                applyPowerup(p, pu.type);
                powerups.splice(i, 1);
                break;
            }
        }
    }
}

function applyPowerup(p, type) {
    switch (type) {
        case 'speed': p.speedBoost = 300; break;
        case 'triple': p.tripleShot = 300; break;
        case 'shield': p.shield = 600; break;
        case 'heal': p.hp = Math.min(p.maxHp, p.hp + 25); break;
    }
}

function updateParticles() {
    for (let i = particles.length - 1; i >= 0; i--) {
        const pt = particles[i];
        pt.x += pt.vx;
        pt.y += pt.vy;
        pt.vx *= 0.92;
        pt.vy *= 0.92;
        pt.life--;
        if (pt.life <= 0) particles.splice(i, 1);
    }
}

function drawPlayer(p) {
    // Trail
    for (let i = 0; i < p.trail.length; i++) {

        ctx.arc(t.x, t.y, p.radius * (i / p.trail.length) * 0.7, 0, Math.PI * 2);
        ctx.fillStyle = p.color + Math.floor(alpha * 255).toString(16).padStart(2, '0');
        ctx.fill();
    }

    // Shield ring
    if (p.shield > 0) {

        ctx.strokeStyle = `rgba(255,235,59,${0.5 + Math.sin(frameCount * 0.1) * 0.3})`;
        ctx.lineWidth = 3;
        ctx.stroke();

    // Body
    ctx.beginPath();
    ctx.arc(p.x, p.y, p.radius, 0, Math.PI * 2);
    const grad = ctx.createRadialGradient(p.x - 5, p.y - 5, 2, p.x, p.y, p.radius);
    grad.addColorStop(0, '#ffffff');
    grad.addColorStop(0.3, p.color);
    grad.addColorStop(1, p.color + '88');

    ctx.fill();

    // Direction indicator
    const fx = Math.cos(p.facing) * (p.radius + 8);
    const fy = Math.sin(p.facing) * (p.radius + 8);

    ctx.lineTo(p.x + fx, p.y + fy);

    ctx.lineWidth = 3;
    ctx.lineCap = 'round';
    ctx.stroke();

        ctx.beginPath();
        ctx.arc(p.x, p.y, p.radius + 3, 0, Math.PI * 2);

// Main loop
let powerupTimer = 0;

function update() {
    frameCount++;

    if (gameState === 'playing') {
        updatePlayer(p1);
        updatePlayer(p2);
        updateProjectiles();
        updatePowerups();
        updateParticles();

        // Powerup spawn
        powerupTimer++;
        if (powerupTimer > 300 && powerups.length < 3) {
            spawnPowerup();
            powerupTimer = 0;
        }

        // Shake decay
        shakeMag *= 0.85;
        if (shakeMag < 0.5) shakeMag = 0;
    } else {
        updateParticles();
        shakeMag *= 0.9;
    }

    shakeX = (Math.random() - 0.5) * shakeMag;
    shakeY = (Math.random() - 0.5) * shakeMag;
}

function draw() {
    ctx.save();
    ctx.translate(shakeX, shakeY);

    // Background
    ctx.fillStyle = '#0a0a1a';
    ctx.fillRect(-10, -10, W + 20, H + 20);

    if (gameState === 'menu') {
        drawGrid();
        drawMenu();
    } else if (gameState === 'playing' || gameState === 'roundEnd' || gameState === 'gameOver') {
        drawGrid();
        drawArena();

        // Power-ups
        powerups.forEach(drawPowerup);

        // Projectiles
        projectiles.forEach(drawProjectile);

        // Players
        if (p1 && p2) {
            drawPlayer(p1);
            drawPlayer(p2);
        }

        // Particles
        particles.forEach(pt => {
            const alpha = pt.life / pt.maxLife;
            ctx.beginPath();
            ctx.arc(pt.x, pt.y, 3 * alpha + 1, 0, Math.PI * 2);
            ctx.fillStyle = pt.color + Math.floor(alpha * 255).toString(16).padStart(2, '0');
            ctx.fill();
        });

        drawHUD();

        if (gameState === 'roundEnd') drawRoundEnd();
        if (gameState === 'gameOver') drawGameOver();
    }

    // Check for game over after round end
    if (gameState === 'roundEnd' && (roundScores[0] >= ROUNDS_TO_WIN || roundScores[1] >= ROUNDS_TO_WIN)) {
        gameState = 'gameOver';
    }

    ctx.restore();
}

function loop() {
    update();
    draw();
    requestAnimationFrame(loop);
}

initRound();
loop();
</script>
