<script>
const canvas = document.getElementById('game');
const ctx = canvas.getContext('2d');
let W, H;

const COLS = 14, ROWS = 9;
let cellW, cellH, offX, offY;

// Path cells (row,col) - enemies travel through these
const pathCells = [
[0,0],[1,0],[2,0],[3,0],[3,1],[3,2],[3,3],[4,3],[5,3],[5,4],[5,5],[5,6],[5,7],[5,8],[5,9],[5,10],[5,11],[5,12],[5,13],
[6,13],[7,13],[8,13]
];

// Path waypoints in pixel coords (center of each path cell)
let pathWaypoints = [];

function isPathCell(r, c) {
return pathCells.some(p => p[0]===r && p[1]===c);
}

let grid = []; // grid[r][c] = null or tower object
let enemies = [];
let projectiles = [];
let particles = [];
let money = 100;
let lives = 20;
let wave = 0;
let kills = 0;
let gameRunning = false;
let audioCtx = null;

function resize() {
W = window.innerWidth;
H = window.innerHeight;
canvas.width = W;
canvas.height = H;
cellW = Math.floor((W - 40) / COLS);
cellH = Math.floor((H - 80) / ROWS);
const scale = Math.min(cellW, cellH);
cellW = scale; cellH = scale;
offX = Math.floor((W - COLS * cellW) / 2);
offY = Math.floor((H - ROWS * cellH) / 2) + 10;

pathWaypoints = pathCells.map(([r,c]) => ({
x: offX + c * cellW + cellW/2,
y: offY + r * cellH + cellH/2
}));
}
window.addEventListener('resize', resize);
resize();

function initGrid() {
grid = [];
for (let r = 0; r < ROWS; r++) {
grid[r] = [];
for (let c = 0; c < COLS; c++) grid[r][c] = null;
}
}

function getAdjacentMineCount(r, c) {
let count = 0;
for (let dr = -1; dr <= 1; dr++) {
for (let dc = -1; dc <= 1; dc++) {
if (dr === 0 && dc === 0) continue;
const nr = r + dr, nc = c + dc;
if (nr >= 0 && nr < ROWS && nc >= 0 && nc < COLS && grid[nr][nc]) count++;
}
}
return count;
}

function playSound(freq, dur, type) {
if (!audioCtx) return;
const osc = audioCtx.createOscillator();
const gain = audioCtx.createGain();
osc.type = type || 'square';
osc.frequency.value = freq;
gain.gain.setValueAtTime(0.1, audioCtx.currentTime);
gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
osc.connect(gain).connect(audioCtx.destination);
osc.start();
osc.stop(audioCtx.currentTime + dur);
}

function startGame() {
if (!audioCtx) {
try { audioCtx = new (window.AudioContext || webkitAudioContext)(); } catch(e) {}
}
initGrid();
enemies = [];
projectiles = [];
particles = [];
money = 100;
lives = 20;
wave = 0;
kills = 0;
gameRunning = true;
document.getElementById('startScreen').style.display = 'none';
document.getElementById('gameOver').style.display = 'none';
document.getElementById('hud').style.display = 'flex';
}

function gameOver() {
gameRunning = false;
document.getElementById('gameOver').style.display = 'flex';
document.getElementById('finalWave').textContent = wave;
document.getElementById('finalKills').textContent = kills;
}

function startWave() {
wave++;
const count = 3 + wave * 2;
const hp = 50 + wave * 25;
const speed = Math.max(30, 80 - wave * 2);
for (let i = 0; i < count; i++) {
enemies.push({
x: pathWaypoints[0].x,
y: pathWaypoints[0].y,
hp: hp, maxHp: hp,
speed: speed + Math.random() * 20,
wpIndex: 1,
alive: true,
radius: cellW * 0.3,
color: `hsl(${(wave * 40 + i * 20) % 360}, 70%, 50%)`
});
}
playSound(200, 0.2, 'sawtooth');
}

function updateEnemies(dt) {
for (let e of enemies) {
if (!e.alive) continue;
const wp = pathWaypoints[e.wpIndex];
if (!wp) { e.alive = false; lives--; playSound(100, 0.3, 'sawtooth'); if (lives <= 0) gameOver(); continue; }
const dx = wp.x - e.x, dy = wp.y - e.y;
const dist = Math.sqrt(dx*dx + dy*dy);
if (dist < e.speed * dt) {
e.wpIndex++;
if (e.wpIndex >= pathWaypoints.length) {
e.alive = false;
lives--;
playSound(100, 0.3, 'sawtooth');
if (lives <= 0) gameOver();
}
} else {
e.x += (dx/dist) * e.speed * dt;
e.y += (dy/dist) * e.speed * dt;
}
}
enemies = enemies.filter(e => e.alive);
}

function updateTowers(dt) {
for (let r = 0; r < ROWS; r++) {
for (let c = 0; c < COLS; c++) {
const t = grid[r][c];
if (!t) continue;
t.cooldown -= dt;
if (t.cooldown <= 0) {
// Find target in range
const tx = offX + c * cellW + cellW/2;
const ty = offY + r * cellH + cellH/2;
let bestEnemy = null, bestDist = Infinity;
for (let e of enemies) {
if (!e.alive) continue;
const dx = e.x - tx, dy = e.y - ty;
const d = Math.sqrt(dx*dx+dy*dy);
if (d < t.range && d < bestDist) { bestDist = d; bestEnemy = e; }
}
if (bestEnemy) {
t.cooldown = 0.8;
projectiles.push({ x: tx, y: ty, tx: bestEnemy.x, ty: bestEnemy.y, target: bestEnemy, speed: 300, damage: t.damage, color: '#4ecca3' });
playSound(800, 0.1, 'square');
}
}
}
}
}

function updateProjectiles(dt) {
for (let p of projectiles) {
if (!p.target || !p.target.alive) { p.dead = true; continue; }
const dx = p.target.x - p.x, dy = p.target.y - p.y;
const dist = Math.sqrt(dx*dx+dy*dy);
if (dist < 10) {
p.target.hp -= p.damage;
p.dead = true;
// Hit particles
for (let i = 0; i < 4; i++) {
particles.push({ x: p.x, y: p.y, vx: (Math.random()-0.5)*80, vy: (Math.random()-0.5)*80, life: 0.3, color: p.color });
}
if (p.target.hp <= 0) {
p.target.alive = false;
kills++;
money += 15 + Math.floor(wave * 2);
playSound(400, 0.15, 'triangle');
for (let i = 0; i < 8; i++) {
particles.push({ x: p.target.x, y: p.target.y, vx: (Math.random()-0.5)*120, vy: (Math.random()-0.5)*120, life: 0.5, color: p.target.color });
}
}
} else {
p.x += (dx/dist) * p.speed * dt;
p.y += (dy/dist) * p.speed * dt;
}
}
projectiles = projectiles.filter(p => !p.dead);
enemies = enemies.filter(e => e.alive);
}

function updateParticles(dt) {
for (let p of particles) {
p.x += p.vx * dt;
p.y += p.vy * dt;
p.life -= dt;
}
particles = particles.filter(p => p.life > 0);
}

function draw() {
ctx.fillStyle = '#1a1a2e';
ctx.fillRect(0, 0, W, H);

// Draw grid
for (let r = 0; r < ROWS; r++) {
for (let c = 0; c < COLS; c++) {
const x = offX + c * cellW, y = offY + r * cellH;
const isPath = isPathCell(r, c);

if (isPath) {
ctx.fillStyle = '#2d2d44';
} else {
ctx.fillStyle = (r+c)%2===0 ? '#16213e' : '#1a1a2e';
}
ctx.fillRect(x, y, cellW-1, cellH-1);

// Draw minesweeper number
if (!isPath && grid[r][c]) {
const adj = getAdjacentMineCount(r, c);
if (adj > 0) {
ctx.fillStyle = '#4ecca3';
ctx.font = `bold ${Math.min(cellW,cellH)*0.25}px Courier New`;
ctx.textAlign = 'center';
ctx.textBaseline = 'middle';
// Draw in corner to not overlap tower
ctx.fillText(adj, x + cellW*0.15, y + cellH*0.15);
}
}

if (!isPath && grid[r][c]) {
// Draw mine/tower
const cx = x + cellW/2, cy = y + cellH/2;
const rad = cellW * 0.3;
ctx.fillStyle = '#4ecca3';
ctx.beginPath();
ctx.arc(cx, cy, rad, 0, Math.PI*2);
ctx.fill();
// Spikes
for (let a = 0; a < 8; a++) {
const angle = a * Math.PI/4;
ctx.strokeStyle = '#4ecca3';
ctx.lineWidth = 2;
ctx.beginPath();
ctx.moveTo(cx + Math.cos(angle)*rad*0.6, cy + Math.sin(angle)*rad*0.6);
ctx.lineTo(cx + Math.cos(angle)*rad, cy + Math.sin(angle)*rad);
ctx.stroke();
}
// Damage number
ctx.fillStyle = '#fff';
ctx.font = `bold ${Math.min(cellW,cellH)*0.28}px Courier New`;
ctx.textAlign = 'center';
ctx.textBaseline = 'middle';
ctx.fillText(grid[r][c].damage, cx, cy);

} else if (!isPath && !grid[r][c]) {
// Show adjacent count for empty non-path cells
const adj = getAdjacentMineCount(r, c);
if (adj > 0) {
ctx.fillStyle = `rgba(78,204,163,${Math.min(adj*0.3, 0.9)})`;
ctx.font = `bold ${Math.min(cellW,cellH)*0.35}px Courier New`;
ctx.textAlign = 'center';
ctx.textBaseline = 'middle';
ctx.fillText(adj, x + cellW/2, y + cellH/2);
}
}
}
}

// Draw path direction arrows
ctx.strokeStyle = 'rgba(78,204,163,0.2)';
ctx.lineWidth = 2;
for (let i = 0; i < pathWaypoints.length - 1; i++) {
const a = pathWaypoints[i], b = pathWaypoints[i+1];
ctx.beginPath();
ctx.moveTo(a.x, a.y);
ctx.lineTo(b.x, b.y);
ctx.stroke();
}

// Draw enemies
for (let e of enemies) {
if (!e.alive) continue;
ctx.fillStyle = e.color;
ctx.beginPath();
ctx.arc(e.x, e.y, e.radius, 0, Math.PI*2);
ctx.fill();
// HP bar
const bw = e.radius * 2;
const hpPct = e.hp / e.maxHp;
ctx.fillStyle = '#333';
ctx.fillRect(e.x - bw/2, e.y - e.radius - 8, bw, 5);
ctx.fillStyle = hpPct > 0.5 ? '#4ecca3' : hpPct > 0.25 ? '#f39c12' : '#e74c3c';
ctx.fillRect(e.x - bw/2, e.y - e.radius - 8, bw * hpPct, 5);
}

// Draw projectiles
for (let p of projectiles) {
ctx.fillStyle = p.color;
ctx.beginPath();
ctx.arc(p.x, p.y, 4, 0, Math.PI*2);
ctx.fill();
}

// Draw particles
for (let p of particles) {
ctx.globalAlpha = p.life * 2;
ctx.fillStyle = p.color;
ctx.fillRect(p.x - 3, p.y - 3, 6, 6);
}
ctx.globalAlpha = 1;

// Update HUD
document.getElementById('money').textContent = money;
document.getElementById('lives').textContent = lives;
document.getElementById('wave').textContent = wave;
document.getElementById('towerInfo').textContent = enemies.length > 0 ? enemies.length + ' enemies' : '';
}

let lastTime = 0;
function gameLoop(time) {
if (!gameRunning) return;
const dt = Math.min((time - lastTime) / 1000, 0.05);
lastTime = time;

updateTowers(dt);
updateEnemies(dt);
updateProjectiles(dt);
updateParticles(dt);
draw();
requestAnimationFrame(gameLoop);
}

canvas.addEventListener('click', (e) => {
if (!gameRunning) return;
const mx = e.clientX, my = e.clientY;
const c = Math.floor((mx - offX) / cellW);
const r = Math.floor((my - offY) / cellH);
if (r < 0 || r >= ROWS || c < 0 || c >= COLS) return;
if (isPathCell(r, c)) return;
if (grid[r][c]) return;
if (money < 50) { playSound(150, 0.2, 'sawtooth'); return; }
money -= 50;
grid[r][c] = { damage: 10 + Math.floor(wave/3), range: cellW * 2.2, cooldown: 0 };
playSound(600, 0.1, 'square');
});

canvas.addEventListener('contextmenu', (e) => {
e.preventDefault();
if (!gameRunning) return;
const mx = e.clientX, my = e.clientY;
const c = Math.floor((mx - offX) / cellW);
const r = Math.floor((my - offY) / cellH);
if (r < 0 || r >= ROWS || c < 0 || c >= COLS) return;
if (grid[r][c]) {
grid[r][c] = null;
money += 25;
playSound(300, 0.15, 'triangle');
}
});

window.addEventListener('keydown', (e) => {
if (e.code === 'Space' && gameRunning) {
e.preventDefault();
startWave();
}
if (e.code === 'KeyR') {
startGame();
lastTime = performance.now();
requestAnimationFrame(gameLoop);
}
});

document.getElementById('startBtn').addEventListener('click', () => {
startGame();
lastTime = performance.now();
requestAnimationFrame(gameLoop);
});

document.getElementById('restartBtn').addEventListener('click', () => {
startGame();
lastTime = performance.now();
requestAnimationFrame(gameLoop);
});

// Initial draw
initGrid();
draw();
</script>
</body>
</html>
