<script>
(function(){
const canvas = document.getElementById('game');
const ctx = canvas.getContext('2d');

let W, H;
function resize() {
    W = window.innerWidth;
    H = window.innerHeight;
    canvas.width = W;
    canvas.height = H;
}
window.addEventListener('resize', resize);
resize();

// Audio
let audioCtx = null;
function initAudio() {
    if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
}
function playSound(freq, dur, type, vol) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = type || 'square';
    osc.frequency.value = freq;
    gain.gain.setValueAtTime(vol || 0.1, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + dur);
}
function sHit() { playSound(440, 0.08, 'square', 0.15); }
function sBrick() { playSound(660, 0.12, 'triangle', 0.2); playSound(880, 0.15, 'sine', 0.15); }
function sWin() { for(let i=0;i<5;i++) setTimeout(()=>playSound(440+i*110,0.2,'triangle',0.2),i*100); }
function sLose() { playSound(200, 0.4, 'sawtooth', 0.2); }

// Game state
let gameState = 'start'; // start, playing, typing, gameover, win
let score = 0;
let lives = 3;
let highScore = 0;
try { highScore = parseInt(localStorage.getItem('typingBreakoutHS')) || 0; } catch(e){}

// Words pool
const words = [
    'code','type','word','brick','ball','game','play','test',
    'fire','star','moon','rain','wind','tree','rock','wave',
    'jump','run','fly','swim','walk','climb','crawl','slide',
    'crash','blast','spark','flash','glow','beam','laser','punch',
    'sword','shield','armor','magic','power','speed','force','energy',
    'ghost','shadow','storm','thunder','lightning','frozen','melt'
];

// Bricks
let bricks = [];
const BRICK_ROWS = 5;
const BRICK_COLS = 8;
let brickW, brickH, brickPad, brickOffsetX, brickOffsetY;

function calcBrickSize() {
    const sidePad = Math.max(20, W * 0.05);
    brickW = (W - sidePad * 2) / BRICK_COLS;
    brickH = Math.max(30, Math.min(45, H * 0.04));
    brickPad = 4;
    brickOffsetX = sidePad;
    brickOffsetY = 60;
}

function initBricks() {
    calcBrickSize();
    bricks = [];
    const usedWords = [];
    for (let r = 0; r < BRICK_ROWS; r++) {
        for (let c = 0; c < BRICK_COLS; c++) {
            let word;
            do { word = words[Math.floor(Math.random() * words.length)]; } while (usedWords.includes(word));
            usedWords.push(word);
            const lenBonus = Math.max(0, word.length - 4);
            bricks.push({
                x: brickOffsetX + c * (brickW + brickPad),
                y: brickOffsetY + r * (brickH + brickPad),
                w: brickW,
                h: brickH,
                word: word,
                alive: true,
                points: 10 + lenBonus * 5,
                color: `hsl(${(r * 60 + c * 10) % 360}, 70%, 50%)`
            });
        }
    }
}

// Ball
let ball = { x: 0, y: 0, vx: 0, vy: 0, r: 8, speed: 5 };
function resetBall() {
    ball.x = W / 2;
    ball.y = H - 60;
    const angle = -Math.PI/4 + (Math.random() - 0.5) * Math.PI/4;
    ball.vx = Math.cos(angle) * ball.speed;
    ball.vy = Math.sin(angle) * ball.speed;
}

// Paddle
let paddle = { w: 120, h: 16, x: 0, y: 0 };
function resetPaddle() {
    paddle.w = Math.max(100, W * 0.12);
    paddle.h = 16;
    paddle.x = (W - paddle.w) / 2;
    paddle.y = H - 40;
}

// Typing state
let typingTarget = null; // brick reference
let typedChars = '';
let typeTimer = 0;
const TYPE_TIME = 5000; // ms to type the word
let particles = [];

// Particles
function spawnParticles(x, y, color) {
    for (let i = 0; i < 12; i++) {
        const angle = Math.random() * Math.PI * 2;
        const speed = 2 + Math.random() * 4;
        particles.push({
            x: x, y: y,
            vx: Math.cos(angle) * speed,
            vy: Math.sin(angle) * speed,
            life: 1,
            color: color
        });
    }
}

// Input
let mouseX = W / 2;
window.addEventListener('mousemove', e => { mouseX = e.clientX; });
window.addEventListener('touchmove', e => { mouseX = e.touches[0].clientX; }, {passive:true});

window.addEventListener('keydown', e => {
    if (gameState === 'start') {
        initAudio();
        startGame();
        return;
    }
    if (gameState === 'gameover' || gameState === 'win') {
        if (e.key === 'Enter' || e.key === ' ') {
            startGame();
        }
        return;
    }
    if (gameState !== 'typing') return;
    
    const key = e.key.toLowerCase();
    if (key === 'backspace' || key === 'Backspace') {
        typedChars = typedChars.slice(0, -1);
        return;
    }
    if (key.length === 1 && /[a-z]/i.test(key)) {
        initAudio();
        typedChars += key.toLowerCase();
        sHit();
        
        const targetWord = typingTarget.word;
        if (typedChars === targetWord) {
            // Success!
            breakBrick(typingTarget);
            typingTarget = null;
            typedChars = '';
            gameState = 'playing';
        } else if (!targetWord.startsWith(typedChars)) {
            // Wrong - reset typed chars
            typedChars = '';
            sLose();
        }
    }
});

// Touch start for mobile
window.addEventListener('touchstart', e => {
    initAudio();
    if (gameState === 'start') startGame();
    else if (gameState === 'gameover' || gameState === 'win') startGame();
}, {passive:true});

function breakBrick(brick) {
    brick.alive = false;
    score += brick.points;
    spawnParticles(brick.x + brick.w/2, brick.y + brick.h/2, brick.color);
    sBrick();
    
    if (score > highScore) {
        highScore = score;
        try { localStorage.setItem('typingBreakoutHS', String(highScore)); } catch(e){}
    }
    
    // Check win
    if (!bricks.some(b => b.alive)) {
        gameState = 'win';
        sWin();
    }
}

function loseLife() {
    lives--;
    sLose();
    typingTarget = null;
    typedChars = '';
    gameState = 'playing';
    if (lives <= 0) {
        gameState = 'gameover';
    } else {
        resetBall();
    }
}

function startGame() {
    score = 0;
    lives = 3;
    initBricks();
    resetPaddle();
    resetBall();
    particles = [];
    typingTarget = null;
    typedChars = '';
    gameState = 'playing';
}

// Physics
function update(dt) {
    if (gameState !== 'playing' && gameState !== 'typing') return;
    
    // Paddle follows mouse
    paddle.x = Math.max(0, Math.min(W - paddle.w, mouseX - paddle.w/2));
    
    if (gameState === 'typing') {
        typeTimer += dt;
        if (typeTimer >= TYPE_TIME) {
            loseLife();
            return;
        }
    }
    
    // Ball physics
    ball.x += ball.vx;
    ball.y += ball.vy;
    
    // Wall bounces
    if (ball.x - ball.r < 0) { ball.x = ball.r; ball.vx = Math.abs(ball.vx); sHit(); }
    if (ball.x + ball.r > W) { ball.x = W - ball.r; ball.vx = -Math.abs(ball.vx); sHit(); }
    if (ball.y - ball.r < 0) { ball.y = ball.r; ball.vy = Math.abs(ball.vy); sHit(); }
    
    // Paddle bounce
    if (ball.y + ball.r >= paddle.y && ball.y + ball.r <= paddle.y + paddle.h + 10 &&
        ball.x >= paddle.x && ball.x <= paddle.x + paddle.w) {
        ball.y = paddle.y - ball.r;
        const hitPos = (ball.x - paddle.x) / paddle.w;
        const angle = (hitPos - 0.5) * Math.PI * 0.7;
        ball.vx = Math.sin(angle) * ball.speed;
        ball.vy = -Math.cos(angle) * ball.speed;
        sHit();
    }
    
    // Ball lost
    if (ball.y + ball.r > H) {
        typingTarget = null;
        typedChars = '';
        loseLife();
        return;
    }
    
    // Brick collision
    if (gameState === 'playing') {
        for (let b of bricks) {
            if (!b.alive) continue;
            if (ball.x + ball.r > b.x && ball.x - ball.r < b.x + b.w &&
                ball.y + ball.r > b.y && ball.y - ball.r < b.y + b.h) {
                // Bounce off brick
                const overlapX = Math.min(ball.x + ball.r - b.x, b.x + b.w - (ball.x - ball.r));
                const overlapY = Math.min(ball.y + ball.r - b.y, b.y + b.h - (ball.y - ball.r));
                if (overlapX < overlapY) {
                    ball.vx = -ball.vx;
                } else {
                    ball.vy = -ball.vy;
                }
                
                // Start typing mode
                gameState = 'typing';
                typingTarget = b;
                typedChars = '';
                typeTimer = 0;
                break;
            }
        }
    }
    
    // Particles
    for (let p of particles) {
        p.x += p.vx;
        p.y += p.vy;
        p.vy += 0.15;
        p.life -= dt / 800;
    }
    particles = particles.filter(p => p.life > 0);
}

// Render
function draw() {
    ctx.fillStyle = '#0a0a1a';
    ctx.fillRect(0, 0, W, H);
    
    if (gameState === 'start') {
        drawStartScreen();
        return;
    }
    
    // Bricks
    for (let b of bricks) {
        if (!b.alive) continue;
        ctx.fillStyle = gameState === 'typing' && typingTarget === b ? '#fff' : b.color;
        ctx.fillRect(b.x, b.y, b.w, b.h);
        ctx.strokeStyle = 'rgba(255,255,255,0.3)';
        ctx.strokeRect(b.x, b.y, b.w, b.h);
        
        // Word on brick
        ctx.fillStyle = '#fff';
        ctx.font = `bold ${Math.min(16, b.w / (b.word.length * 0.7))}px Courier New`;
        ctx.textAlign = 'center';
        ctx.textBaseline = 'middle';
        ctx.fillText(b.word.toUpperCase(), b.x + b.w/2, b.y + b.h/2);
    }
    
    // Paddle
    ctx.fillStyle = '#4a9eff';
    ctx.fillRect(paddle.x, paddle.y, paddle.w, paddle.h);
    ctx.fillStyle = '#8bc4ff';
    ctx.fillRect(paddle.x, paddle.y, paddle.w, 4);
    
    // Ball
    ctx.beginPath();
    ctx.arc(ball.x, ball.y, ball.r, 0, Math.PI * 2);
    ctx.fillStyle = '#fff';
    ctx.fill();
    ctx.shadowColor = '#4a9eff';
    ctx.shadowBlur = 15;
    ctx.fill();
    ctx.shadowBlur = 0;
    
    // Particles
    for (let p of particles) {
        ctx.globalAlpha = p.life;
        ctx.fillStyle = p.color;
        ctx.fillRect(p.x - 3, p.y - 3, 6, 6);
    }
    ctx.globalAlpha = 1;
    
    // Typing prompt
    if (gameState === 'typing' && typingTarget) {
        const targetWord = typingTarget.word;
        const progress = typedChars.length / targetWord.length;
        
        // Prompt box
        const pw = Math.min(500, W * 0.7);
        const ph = 80;
        const px = (W - pw) / 2;
        const py = H * 0.35;
        
        ctx.fillStyle = 'rgba(10,10,30,0.9)';
        ctx.strokeStyle = '#4a9eff';
        ctx.lineWidth = 2;
        ctx.fillRect(px, py, pw, ph);
        ctx.strokeRect(px, py, pw, ph);
        
        // Target word
        ctx.font = 'bold 28px Courier New';
        ctx.textAlign = 'center';
        ctx.textBaseline = 'middle';
        for (let i = 0; i < targetWord.length; i++) {
            const cx = px + pw/2 - (targetWord.length * 16) / 2 + i * 16 + 8;
            if (i < typedChars.length && typedChars[i] === targetWord[i]) {
                ctx.fillStyle = '#4aff9e';
            } else if (i === typedChars.length) {
                ctx.fillStyle = '#fff';
                ctx.font = 'bold 28px Courier New';
                ctx.fillText('_', cx, py + 25);
                ctx.font = 'bold 28px Courier New';
            } else {
                ctx.fillStyle = targetWord[i] === typedChars[i] ? '#4aff9e' : '#ff6b6b';
            }
            if (i < typedChars.length) {
                ctx.fillText(targetWord[i], cx, py + 25);
            } else {
                ctx.fillStyle = 'rgba(255,255,255,0.3)';
                ctx.fillText(targetWord[i].toUpperCase(), cx, py + 25);
            }
        }
        
        // Timer bar
        const timeLeft = 1 - (typeTimer / TYPE_TIME);
        const barW = pw * Math.max(0, timeLeft);
        ctx.fillStyle = timeLeft > 0.3 ? '#4aff9e' : '#ff6b6b';
        ctx.fillRect(px, py + ph - 8, barW, 8);
        
        // "TYPE TO BREAK" label
        ctx.fillStyle = '#aaa';
        ctx.font = '12px Courier New';
        ctx.fillText('TYPE THE WORD TO BREAK THE BRICK', W/2, py - 15);
    }
    
    // HUD
    ctx.fillStyle = '#fff';
    ctx.font = 'bold 16px Courier New';
    ctx.textAlign = 'left';
    ctx.textBaseline = 'top';
    ctx.fillText(`SCORE: ${score}`, 20, 20);
    ctx.fillText(`LIVES: ${'♥'.repeat(Math.max(0,lives))}`, 20, 42);
    ctx.textAlign = 'right';
    ctx.fillText(`HI: ${highScore}`, W - 20, 20);
    
    if (gameState === 'gameover') {
        drawOverlay('GAME OVER', `Final Score: ${score}`);
    } else if (gameState === 'win') {
        drawOverlay('YOU WIN!', `Score: ${score} - All bricks destroyed!`);
    }
    
    // Particles drawn on top for typing state too
    for (let p of particles) {
        ctx.globalAlpha = p.life;
        ctx.fillStyle = p.color;
        ctx.fillRect(p.x - 3, p.y - 3, 6, 6);
    }
    ctx.globalAlpha = 1;
}

function drawStartScreen() {
    ctx.fillStyle = '#0a0a1a';
    ctx.fillRect(0, 0, W, H);
    
    // Title
    ctx.fillStyle = '#4a9eff';
    ctx.font = `bold ${Math.min(60, W*0.08)}px Courier New`;
    ctx.textAlign = 'center';
    ctx.textBaseline = 'middle';
    ctx.fillText('TYPING BREAKOUT', W/2, H * 0.25);
    
    // Subtitle
    ctx.fillStyle = '#fff';
    ctx.font = `${Math.min(18, W*0.03)}px Courier New`;
    ctx.fillText('Break bricks by typing the words on them!', W/2, H * 0.4);
    
    // Instructions
    const lines = [
        '• Move mouse to control the paddle',
        '• The ball bounces off bricks',
        '• Type the word shown to break the brick',
        '• You have limited time - hurry!',
        '• Lose a life if you fail or the ball drops',
        '',
        `High Score: ${highScore}`
    ];
    ctx.font = `${Math.min(15, W*0.02)}px Courier New`;
    ctx.fillStyle = '#aaa';
    lines.forEach((line, i) => {
        ctx.fillText(line, W/2, H * 0.5 + i * 28);

    const pulse = 0.5 + Math.sin(Date.now() / 500) * 0.5;
    ctx.fillStyle = `rgba(74,158,255,${pulse})`;

function drawOverlay(title, subtitle) {
    ctx.fillStyle = 'rgba(0,0,0,0.7)';
    ctx.fillRect(0, 0, W, H);
    
    ctx.fillStyle = '#fff';
    ctx.font = `bold ${Math.min(50, W*0.07)}px Courier New`;
    ctx.textAlign = 'center';

    ctx.font = `${Math.min(16, W*0.025)}px Courier New`;

}

// Game loop
let lastTime = 0;
function gameLoop(timestamp) {
    const dt = Math.min(timestamp - lastTime, 33);
    lastTime = timestamp;
    
    update(dt);
    draw();
    
    requestAnimationFrame(gameLoop);
}

calcBrickSize();
initBricks();
resetPaddle();
resetBall();
