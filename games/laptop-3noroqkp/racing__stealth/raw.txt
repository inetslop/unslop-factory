// Game state
let gameState = 'start'; // start, playing, caught, win
let gameTime = 0;
let exposure = 0;
let maxExposure = 100;
let audioCtx = null;
let firstInput = false;

// World
const WORLD_W = 4000;
const WORLD_H = 2500;
const TILE = 80;

// Player
let player = { x: 200, y: WORLD_H / 2, angle: 0, speed: 0, maxSpeed: 320, accel: 400, friction: 0.97, turnSpeed: 3.2 };

// Finish
let finishX = WORLD_W - 150;

// Buildings (obstacles)
let buildings = [];
let roads = [];

// Patrols
let patrols = [];

// Particles
let particles = [];

function initWorld() {
    buildings = [];
    roads = [];
    patrols = [];
    particles = [];

    // Generate city grid - roads every 200px, buildings in between
    for (let gx = 0; gx < WORLD_W / TILE; gx++) {
        for (let gy = 0; gy < WORLD_H / TILE; gy++) {
            let x = gx * TILE;
            let y = gy * TILE;
            // Roads: every 3 tiles in each direction is a road
            let isRoadX = (gx % 3 === 2);
            let isRoadY = (gy % 3 === 2);
            if (!isRoadX && !isRoadY) {
                // Building block with some randomness
                let hasBuilding = Math.random() > 0.2;
                if (hasBuilding) {
                    buildings.push({ x: x + 5, y: y + 5, w: TILE - 10, h: TILE - 10, color: `hsl(${220 + Math.random()*40}, ${15+Math.random()*10}%, ${8+Math.random()*8}%)` });
                }
            }
        }
    }

    // Ensure start area is clear
    buildings = buildings.filter(b => !(b.x < 400 && Math.abs(b.y - WORLD_H/2) < 200));

    // Generate patrols
    for (let i = 0; i < 18; i++) {
        let px, py;
        let angle;
        let range = 120 + Math.random() * 80;
        let arc = (Math.PI / 3) + Math.random() * (Math.PI / 5);

        // Place patrols along the route but not too close to start
        px = 600 + Math.random() * (WORLD_W - 900);
        py = 100 + Math.random() * (WORLD_H - 200);
        angle = Math.random() * Math.PI * 2;

        let speed = 40 + Math.random() * 60;
        let turnRate = 0.3 + Math.random() * 0.5;
        let changeTimer = 2 + Math.random() * 4;

        patrols.push({ x: px, y: py, angle: angle, range: range, arc: arc, speed: speed, turnRate: turnRate, changeTimer: changeTimer, targetAngle: angle });
    }
}

function initGame() {
    player = { x: 200, y: WORLD_H / 2, angle: 0, speed: 0, maxSpeed: 320, accel: 400, friction: 0.97, turnSpeed: 3.2 };
    exposure = 0;
    gameTime = 0;
    particles = [];
    initWorld();

// Sound
function playTone(freq, dur, type='sine', vol=0.1) {
    if (!audioCtx) return;
    let osc = audioCtx.createOscillator();
    let gain = audioCtx.createGain();
    osc.type = type;
    osc.frequency.value = freq;
    gain.gain.setValueAtTime(vol, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + dur);
}

function playAlarm() {
    playTone(880, 0.2, 'square', 0.05);
    setTimeout(() => playTone(660, 0.3, 'square', 0.04), 150);
}

function playWin() {
    playTone(523, 0.2, 'sine', 0.08);
    setTimeout(() => playTone(659, 0.2, 'sine', 0.08), 150);
    setTimeout(() => playTone(784, 0.3, 'sine', 0.08), 300);

// Update
let lastTime = 0;
function update(dt) {
    if (gameState !== 'playing') return;

    gameTime += dt;

    // Player movement
    let throttle = 0;
    let steer = 0;
    if (keys['KeyW'] || keys['ArrowUp']) throttle = 1;
    if (keys['KeyS'] || keys['ArrowDown']) throttle = -0.5;
    if (keys['KeyA'] || keys['ArrowLeft']) steer = -1;
    if (keys['KeyD'] || keys['ArrowRight']) steer = 1;

    player.speed += throttle * player.accel * dt;
    player.speed *= Math.pow(player.friction, dt * 60);
    player.speed = Math.max(-80, Math.min(player.maxSpeed, player.speed));

    if (Math.abs(player.speed) > 5) {
        let turnFactor = steer * player.turnSpeed * (player.speed > 0 ? 1 : -1) * dt;
        player.angle += turnFactor * (1 - Math.abs(player.speed) / player.maxSpeed * 0.3);
    }

    player.x += Math.cos(player.angle) * player.speed * dt;
    player.y += Math.sin(player.angle) * player.speed * dt;

    // Clamp to world
    player.x = Math.max(20, Math.min(WORLD_W - 20, player.x));
    player.y = Math.max(20, Math.min(WORLD_H - 20, player.y));

    // Building collision
    for (let b of buildings) {
        if (player.x > b.x && player.x < b.x + b.w && player.y > b.y && player.y < b.y + b.h) {
            // Push out
            let overlapX = Math.min(player.x - b.x, b.x + b.w - player.x);
            let overlapY = Math.min(player.y - b.y, b.y + b.h - player.y);
            if (overlapX < overlapY) {
                player.x += (player.x - b.x < b.x + b.w - player.x) ? -overlapX : overlapX;
            } else {
                player.y += (player.y - b.y < b.y + b.h - player.y) ? -overlapY : overlapY;
            }
            player.speed *= 0.3;
        }
    }

    // Patrol AI
    for (let p of patrols) {
        p.changeTimer -= dt;
        if (p.changeTimer <= 0) {
            p.targetAngle = Math.random() * Math.PI * 2;
            p.changeTimer = 3 + Math.random() * 5;
        }

        let diff = p.targetAngle - p.angle;
        while (diff > Math.PI) diff -= Math.PI * 2;
        while (diff < -Math.PI) diff += Math.PI * 2;
        p.angle += Math.sign(diff) * p.turnRate * dt;

        p.x += Math.cos(p.angle) * p.speed * dt;
        p.y += Math.sin(p.angle) * p.speed * dt;

        // Bounce off world edges
        if (p.x < 50 || p.x > WORLD_W - 50) { p.targetAngle = Math.PI - p.targetAngle; }
        if (p.y < 50 || p.y > WORLD_H - 50) { p.targetAngle = -p.targetAngle; }

        // Check player visibility
        let dx = player.x - p.x;
        let dy = player.y - p.y;
        let dist = Math.sqrt(dx * dx + dy * dy);

        if (dist < p.range) {
            let toPlayer = Math.atan2(dy, dx);
            let angleDiff = toPlayer - p.angle;
            while (angleDiff > Math.PI) angleDiff -= Math.PI * 2;
            while (angleDiff < -Math.PI) angleDiff += Math.PI * 2;

            if (Math.abs(angleDiff) < p.arc / 2) {
                // Check line of sight (no building between)
                let visible = true;
                for (let b of buildings) {
                    if (lineIntersectsRect(p.x, p.y, player.x, player.y, b)) {
                        visible = false;
                        break;
                    }
                }
                if (visible) {
                    exposure += 30 * dt * (1 - dist / p.range);
                    playAlarm();
                    break;
                }
            }
        }
    }

    // Exposure decay
    let inCone = false;
    for (let p of patrols) {
        let dx = player.x - p.x;
        let dy = player.y - p.y;
        let dist = Math.sqrt(dx * dx + dy * dy);
        if (dist < p.range) {
            let toPlayer = Math.atan2(dy, dx);
            let angleDiff = toPlayer - p.angle;
            while (angleDiff > Math.PI) angleDiff -= Math.PI * 2;
            while (angleDiff < -Math.PI) angleDiff += Math.PI * 2;
            if (Math.abs(angleDiff) < p.arc / 2) { inCone = true; break; }
        }
    }
    if (!inCone) {
        exposure -= 8 * dt;
    }
    exposure = Math.max(0, Math.min(maxExposure, exposure));

    if (exposure >= maxExposure) {
        gameState = 'caught';
        playTone(200, 0.5, 'sawtooth', 0.1);
    }

    // Win check
    if (player.x >= finishX) {
        gameState = 'win';
        playWin();
    }

    // Particles (engine trail)
    if (Math.abs(player.speed) > 50 && Math.random() < 0.3) {
        particles.push({
            x: player.x - Math.cos(player.angle) * 15,
            y: player.y - Math.sin(player.angle) * 15,
            life: 0.5 + Math.random() * 0.3,
            maxLife: 0.8,
            size: 3 + Math.random() * 3
        });
    }

    for (let i = particles.length - 1; i >= 0; i--) {
        particles[i].life -= dt;
        if (particles[i].life <= 0) particles.splice(i, 1);
    }
}

function lineIntersectsRect(x1, y1, x2, y2, rect) {
    // Simple check: sample points along the line
    let steps = Math.max(Math.abs(x2-x1), Math.abs(y2-y1)) / 20;
    for (let i = 0; i <= steps; i++) {
        let t = i / steps;
        let px = x1 + (x2 - x1) * t;
        let py = y1 + (y2 - y1) * t;
        if (px > rect.x && px < rect.x + rect.w && py > rect.y && py < rect.y + rect.h) return true;
    }
    return false;
}

        ctx.beginPath(); ctx.moveTo(0, y); ctx.lineTo(WORLD_W, y); ctx.stroke();
    }

        }

    for (let gy = 0; gy < WORLD_H / TILE; gy++) {
        if (gy % 3 === 2) {
            ctx.fillRect(0, gy * TILE, WORLD_W, TILE);
        }
    }

    // Road center lines
    ctx.strokeStyle = '#2a2a4e';
    ctx.lineWidth = 2;
    ctx.setLineDash([15, 15]);

            ctx.beginPath(); ctx.moveTo(0, ry); ctx.lineTo(WORLD_W, ry); ctx.stroke();
        }

    for (let b of buildings) {

        ctx.fillStyle = i % 2 === 0 ? '#ffffff' : '#333';

    }
    ctx.fillStyle = '#fff';
    ctx.font = 'bold 14px monospace';
    ctx.textAlign = 'center';

    for (let p of patrols) {

        ctx.beginPath();

        ctx.arc(0, 0, p.range, -p.arc / 2, p.arc / 2);

        ctx.fillStyle = gradient;
        ctx.fill();

        // Cone edge lines
        ctx.beginPath();
        ctx.moveTo(0, 0);
        ctx.lineTo(Math.cos(-p.arc/2) * p.range, Math.sin(-p.arc/2) * p.range);
        ctx.strokeStyle = 'rgba(255, 80, 80, 0.3)';
        ctx.lineWidth = 1;
        ctx.stroke();
        ctx.beginPath();
        ctx.moveTo(0, 0);
        ctx.lineTo(Math.cos(p.arc/2) * p.range, Math.sin(p.arc/2) * p.range);

    ctx.fillRect(6, -5, 6, 10);

function renderCaughtScreen() {
