function applyGravity() {
    let moved = false;
    for (let x = 0; x < 7; x++) {
        // Bottom to top, move gems down
        for (let y = 6; y > 0; y--) {
            if (grid[y][x] === 0 && (typeof grid[y-1][x] === 'number' && grid[y-1][x] > 0)) {
                // Check if cell below is not wall
                grid[y][x] = grid[y-1][x];
                grid[y-1][x] = 0;
                moved = true;
            }
        }
    }
    return moved;
}

function resolveMatches() {
    let matched = checkMatches();
    if (matched.size === 0) return false;
    
    matched.forEach(key => {
        const [x, y] = key.split(',').map(Number);
        const color = grid[y][x];
        gemsCleared++;
        score += 10 * color;
        // Add particles
        for (let i = 0; i < 5; i++) {
            particles.push({
                x: gridX + x * cellSize + cellSize/2,
                y: gridY + y * cellSize + cellSize/2,
                vx: (Math.random()-0.5) * 4,
                vy: (Math.random()-0.5) * 4 - 2,
                life: 30,
                color: COLORS[color]
            });
        }
        grid[y][x] = 0;
    });
    
    // Play sound
    playMatchSound();
    
    return true;
}

function tryMove(dx, dy) {
    if (state !== 'playing') return;
    
    const nx = player.x + dx;
    const ny = player.y + dy;
    
    if (nx < 0 || nx >= 7 || ny < 0 || ny >= 7) return;
    if (grid[ny][nx] === 'X') return;
    
    if (typeof grid[ny][nx] === 'number' && grid[ny][nx] > 0) {
        // Try to push gem
        const px = nx + dx;
        const py = ny + dy;
        if (px < 0 || px >= 7 || py < 0 || py >= 7) return;
        if (grid[py][px] === 'X') return;
        if (typeof grid[py][px] === 'number' && grid[py][px] > 0) return;
        // Push
        grid[ny][nx] = grid[ny+dy][nx+dx];
        grid[ny+dy][nx+dx] = 0;
    }
    
    player.x = nx;
    player.y = ny;
    movesLeft--;
    
    playMoveSound();
    
    // Resolve matches with animation
    state = 'animating';
    animTimer = 0;
    resolveAnimation();
}

function resolveAnimation() {
    if (resolveMatches()) {
        applyGravity();
        setTimeout(resolveAnimation, 300);
    } else {
        // Check win/lose
        let remaining = 0;
        for (let y = 0; y < 7; y++)
            for (let x = 0; x < 7; x++)
                if (typeof grid[y][x] === 'number' && grid[y][x] > 0) remaining++;

function drawRoundedRect(x, y, w, h, r) {

    ctx.moveTo(x+r, y);
    ctx.lineTo(x+w-r, y);
    ctx.quadraticCurveTo(x+w, y, x+w, y+r);
    ctx.lineTo(x+w, y+h-r);

    ctx.quadraticCurveTo(x, y+h, x, y+h-r);
    ctx.lineTo(x, y+r);
    ctx.quadraticCurveTo(x, y, x+r, y);
    ctx.closePath();
