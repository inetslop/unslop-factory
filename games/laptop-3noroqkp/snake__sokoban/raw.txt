function playSound(freq, dur, type) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = type || 'square';
    osc.frequency.value = freq;
    gain.gain.setValueAtTime(0.15, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + dur);
}

function resize() {
    W = window.innerWidth;
    H = window.innerHeight;
    canvas.width = W;
    canvas.height = H;
    cellSize = Math.floor(Math.min(W / 16, H / 14));
    cols = Math.floor(W / cellSize);
    rows = Math.floor(H / cellSize);
}
window.addEventListener('resize', resize);
resize();

// Game states: 'start', 'playing', 'win', 'gameover'

];

function initLevel(lvl) {
    const L = LEVELS[lvl % LEVELS.length];
    snake = L.snakeStart.map(c => ({x:c[0], y:c[1]}));
    dir = L.dir === 'R' ? {x:1,y:0} : L.dir === 'L' ? {x:-1,y:0} : L.dir === 'U' ? {x:0,y:-1} : {x:0,y:1};
    nextDir = {...dir};
    boxes = L.boxes.map(c => ({x:c[0], y:c[1]}));
    targets = L.targets.map(c => ({x:c[0], y:c[1]}));
    food = L.food.map(c => ({x:c[0], y:c[1]}));
    walls = L.walls.map(c => ({x:c[0], y:c[1]}));
    gameSpeed = Math.max(4, 8 - lvl);
    lastMove = 0;
}

function startGame() {
    state = 'playing';
    score = 0;
    level = 0;
    initLevel(0);
    if (!audioCtx) {
        audioCtx = new (window.AudioContext || window.webkitAudioContext)();
    }

function nextLevel() {
    level++;
    if (level >= LEVELS.length) {
        state = 'win';
        return;
    }
    initLevel(level);
    state = 'playing';
}

function isWall(x, y) {
    const L = LEVELS[level % LEVELS.length];
    if (x < 0 || x >= L.size[0] || y < 0 || y >= L.size[1]) return true;
    return walls.some(w => w.x === x && w.y === y);
}

function isSnake(x, y) {
    for (let i = 0; i < snake.length - 1; i++) {
        if (snake[i].x === x && snake[i].y === y) return true;
    }
    return false;
}

function isBoxAt(x, y) {
    return boxes.findIndex(b => b.x === x && b.y === y);
}

function canMove(nx, ny) {
    if (isWall(nx, ny)) return false;
    const boxIdx = isBoxAt(nx, ny);
    if (boxIdx !== -1) {
        const bx = nx + nextDir.x;
        const by = ny + nextDir.y;
        if (isWall(bx, by) || isBoxAt(bx, by) !== -1) return false;
        if (isSnake(bx, by)) return false;
    } else {
        if (isSnake(nx, ny)) return false;
    }
    return true;
}

function move() {
    const head = snake[0];
    const nx = head.x + nextDir.x;
    const ny = head.y + nextDir.y;

    if (!canMove(nx, ny)) {
        state = 'gameover';
        playSound(120, 0.3, 'sawtooth');
        return;
    }

    // Check if pushing a box
    const boxIdx = isBoxAt(nx, ny);
    if (boxIdx !== -1) {
        boxes[boxIdx].x += nextDir.x;
        boxes[boxIdx].y += nextDir.y;
        playSound(200, 0.1, 'triangle');
        // Check box on target
        if (targets.some(t => t.x === boxes[boxIdx].x && t.y === boxes[boxIdx].y)) {
            score += 100;
            playSound(600, 0.15, 'sine');
        }
    }

    // Move snake
    snake.unshift({x: nx, y: ny});

    // Check food
    const fi = food.findIndex(f => f.x === nx && f.y === ny);
    if (fi !== -1) {
        food.splice(fi, 1);
        score += 50;
        playSound(800, 0.1, 'sine');
        // Snake grows (keep tail)
    } else {
        snake.pop();
    }

    dir = {...nextDir};

    // Check win: all boxes on targets
    if (boxes.every(b => targets.some(t => t.x === b.x && t.y === b.y))) {
        score += 500;
        nextLevel();
    }
}

function updateParticles() {
    for (let i = particles.length - 1; i >= 0; i--) {
        particles[i].life -= 0.02;
        particles[i].x += particles[i].vx;
        particles[i].y += particles[i].vy;
        if (particles[i].life <= 0) particles.splice(i, 1);
    }
}

function spawnParticles(x, y, color) {
    for (let i = 0; i < 8; i++) {
        const angle = Math.random() * Math.PI * 2;
        const speed = 1 + Math.random() * 2;
        particles.push({x: x*cellSize+cellSize/2, y: y*cellSize+cellSize/2, vx: Math.cos(angle)*speed, vy: Math.sin(angle)*speed, life: 1, color});
    }
}

    ctx.fillStyle = '#0a0a1a';

    const L = LEVELS[level % LEVELS.length];

    // Draw grid
    for (let y = 0; y < L.size[1]; y++) {
        for (let x = 0; x < L.size[0]; x++) {
            const px = offX + x * cellSize;

            ctx.fillStyle = (x+y) % 2 === 0 ? '#1a1a2e' : '#16162a';
            ctx.fillRect(px, py, cellSize, cellSize);

        ctx.strokeStyle = 'rgba(0, 200, 255, 0.8)';
        ctx.lineWidth = 2;
        ctx.strokeRect(px+4, py+4, cellSize-8, cellSize-8);
        // Pulse
        const pulse = Math.sin(Date.now() * 0.003) * 0.15 + 0.15;
        ctx.fillStyle = `rgba(0, 200, 255, ${pulse})`;
        ctx.fillRect(px+6, py+6, cellSize-12, cellSize-12);
    }

    // Draw walls
    for (const w of walls) {
        const px = offX + w.x * cellSize;
        const py = offY + w.y * cellSize;
        ctx.fillStyle = '#4a4a6a';
        ctx.fillRect(px+1, py+1, cellSize-2, cellSize-2);
        ctx.fillStyle = '#5a5a7a';
        ctx.fillRect(px+3, py+3, cellSize-6, cellSize-6);
    }

    // Draw food
    for (const f of food) {
        const px = offX + f.x * cellSize;

        ctx.arc(px+cellSize/2-2, py+cellSize/2-2, cellSize*0.12, 0, Math.PI*2);

function drawOverlay(title, subtitle, hint) {
    ctx.fillStyle = 'rgba(0,0,0,0.7)';

let lastTime = 0;
function gameLoop(time) {
    if (state === 'playing') {
        if (time - lastMove > 1000 / gameSpeed) {
            move();
            lastMove = time;
        }
    }
    updateParticles();
    draw();
    requestAnimationFrame(gameLoop);
}

window.addEventListener('keydown', function(e) {
    if (!audioCtx) {
        audioCtx = new (window.AudioContext || window.webkitAudioContext)();
    }

    if (state === 'start') {
        if (e.key === 'Enter' || e.key === ' ') startGame();
        return;
    }

    if (state === 'gameover' || state === 'win') {
        if (e.key === 'r' || e.key === 'R') startGame();
        return;
    }

    if (state !== 'playing') return;

    let handled = true;
    switch(e.key) {
        case 'ArrowUp': case 'w': case 'W': nextDir = {x:0,y:-1}; break;
        case 'ArrowDown': case 's': case 'S': nextDir = {x:0,y:1}; break;
        case 'ArrowLeft': case 'a': case 'A': nextDir = {x:-1,y:0}; break;
        case 'ArrowRight': case 'd': case 'D': nextDir = {x:1,y:0}; break;
        case 'r': case 'R': startGame(); return;
        default: handled = false;
    }
    if (handled) e.preventDefault();

    // Prevent 180 turn
    if (nextDir.x === -dir.x && nextDir.y === -dir.y && snake.length > 1) {
        nextDir = {...dir};
    }
});

requestAnimationFrame(gameLoop);
})();
