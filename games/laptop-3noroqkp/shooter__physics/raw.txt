const canvas = document.getElementById('game');
const ctx = canvas.getContext('2d');

let W, H;
function resize() {
    W = window.innerWidth;
    H = window.innerHeight;
    canvas.width = W;
    canvas.height = H;
}
window.addEventListener('resize', resize);
resize();

// Audio
let audioCtx = null;
function initAudio() {
    if (!audioCtx) {
        audioCtx = new (window.AudioContext || window.webkitAudioContext)();
    }
}
function playSound(type) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    const now = audioCtx.currentTime;
    if (type === 'shoot') {
        osc.type = 'sine';
        osc.frequency.setValueAtTime(200, now);
        osc.frequency.exponentialRampToValueAtTime(600, now + 0.1);
        gain.gain.setValueAtTime(0.3, now);
        gain.gain.exponentialRampToValueAtTime(0.01, now + 0.15);
        osc.start(now);
        osc.stop(now + 0.15);
    } else if (type === 'hit') {
        osc.type = 'square';
        osc.frequency.setValueAtTime(300, now);
        osc.frequency.exponentialRampToValueAtTime(100, now + 0.1);
        gain.gain.setValueAtTime(0.2, now);
        gain.gain.exponentialRampToValueAtTime(0.01, now + 0.12);
        osc.start(now);
        osc.stop(now + 0.12);
    } else if (type === 'destroy') {
        osc.type = 'sawtooth';
        osc.frequency.setValueAtTime(400, now);
        osc.frequency.exponentialRampToValueAtTime(80, now + 0.3);
        gain.gain.setValueAtTime(0.25, now);
        gain.gain.exponentialRampToValueAtTime(0.01, now + 0.3);
        osc.start(now);
        osc.stop(now + 0.3);
    } else if (type === 'bounce') {
        osc.type = 'triangle';
        osc.frequency.setValueAtTime(500, now);
        gain.gain.setValueAtTime(0.1, now);
        gain.gain.exponentialRampToValueAtTime(0.01, now + 0.05);
        osc.start(now);
        osc.stop(now + 0.05);
    } else if (type === 'wave') {
        osc.type = 'sine';
        osc.frequency.setValueAtTime(440, now);
        gain.gain.setValueAtTime(0.2, now);
        gain.gain.exponentialRampToValueAtTime(0.01, now + 0.5);
        osc.start(now);
        osc.stop(now + 0.5);
    }
}

// Game state
let gameState = 'start'; // start, playing, gameover
let score = 0;
let wave = 1;
let shotsLeft = 10;
let maxShots = 10;
let mouseX = W / 2;
let mouseY = H / 3;
let bullets = [];
let targets = [];
let particles = [];
let floatTexts = [];
let cannonAngle = -Math.PI / 2;
let combo = 0;
let comboTimer = 0;

const GRAVITY = 0.15;
const BOUNCE_RESTITUTION = 0.7;
const CANNON_X_RATIO = 0.5;
const CANNON_Y_RATIO = 0.92;

function getCannonX() { return W * CANNON_X_RATIO; }
function getCannonY() { return H * CANNON_Y_RATIO; }

// Target types
const TARGET_TYPES = {
    wood: { hp: 1, color: '#c8a050', points: 10, size: 30 },
    stone: { hp: 2, color: '#8899aa', points: 25, size: 35 },
    metal: { hp: 3, color: '#aaccff', points: 50, size: 40 },
    gold: { hp: 1, color: '#ffd700', points: 100, size: 25 }
};

function generateWave(w) {
    targets = [];
    const numTargets = Math.min(5 + w * 2, 20);
    const types = ['wood'];
    if (w >= 2) types.push('stone');
    if (w >= 3) types.push('metal');
    if (w >= 4) types.push('gold');

    for (let i = 0; i < numTargets; i++) {
        const type = types[Math.floor(Math.random() * types.length)];
        const t = TARGET_TYPES[type];
        let x, y, size = t.size;
        let attempts = 0;
        do {
            x = 60 + Math.random() * (W - 120);
            y = 60 + Math.random() * (H * 0.55);
            attempts++;
        } while (targets.some(t => Math.hypot(t.x - x, t.y - y) < 80) && attempts < 50);
        targets.push({
            x, y, type,
            hp: t.hp, maxHp: t.hp,
            size, points: t.points,
            color: t.color,
            wobble: Math.random() * Math.PI * 2
        });
    }
}

function startGame() {
    score = 0;
    wave = 1;
    shotsLeft = maxShots;
    bullets = [];
    particles = [];
    floatTexts = [];
    combo = 0;
    comboTimer = 0;
    generateWave(wave);
    gameState = 'playing';
}

function nextWave() {
    wave++;
    shotsLeft = maxShots + Math.floor(wave / 2);
    bullets = [];
    generateWave(wave);
    playSound('wave');
}

function fireBullet() {
    if (gameState !== 'playing' || shotsLeft <= 0) return;
    initAudio();
    shotsLeft--;
    const cx = getCannonX();
    const cy = getCannonY();
    const speed = 12;
    bullets.push({
        x: cx + Math.cos(cannonAngle) * 40,
        y: cy + Math.sin(cannonAngle) * 40,
        vx: Math.cos(cannonAngle) * speed,
        vy: Math.sin(cannonAngle) * speed,
        radius: 6,
        trail: []
    });
    playSound('shoot');
}

function spawnParticles(x, y, color, count) {
    for (let i = 0; i < count; i++) {
        const angle = Math.random() * Math.PI * 2;
        const speed = 1 + Math.random() * 4;
        particles.push({
            x, y,
            vx: Math.cos(angle) * speed,
            vy: Math.sin(angle) * speed - 2,
            life: 1,
            decay: 0.02 + Math.random() * 0.03,
            color,
            size: 2 + Math.random() * 4
        });
    }
}

function addFloatText(x, y, text, color) {
    floatTexts.push({ x, y, text, color, life: 1 });
}

function update() {
    if (gameState !== 'playing') return;

    comboTimer -= 0.016;
    if (comboTimer <= 0) combo = 0;

    // Update bullets
    for (let i = bullets.length - 1; i >= 0; i--) {
        const b = bullets[i];
        b.trail.push({ x: b.x, y: b.y });
        if (b.trail.length > 15) b.trail.shift();

        b.vy += GRAVITY;
        b.x += b.vx;
        b.y += b.vy;

        // Wall bouncing
        let bounced = false;
        if (b.x - b.radius < 0) { b.x = b.radius; b.vx = -b.vx * BOUNCE_RESTITUTION; bounced = true; }
        if (b.x + b.radius > W) { b.x = W - b.radius; b.vx = -b.vx * BOUNCE_RESTITUTION; bounced = true; }
        if (b.y - b.radius < 0) { b.y = b.radius; b.vy = -b.vy * BOUNCE_RESTITUTION; bounced = true; }

        if (bounced) playSound('bounce');

        // Off bottom
        if (b.y > H + 50) {
            bullets.splice(i, 1);
            continue;
        }

        // Speed check (too slow means it's stuck)
        if (Math.hypot(b.vx, b.vy) < 0.5 && b.y < H - 20) {
            bullets.splice(i, 1);
            continue;
        }

        // Hit targets
        for (let j = targets.length - 1; j >= 0; j--) {
            const t = targets[j];
            const dx = b.x - t.x;
            const dy = b.y - t.y;
            const dist = Math.hypot(dx, dy);
            if (dist < t.size + b.radius) {
                t.hp--;
                playSound('hit');
                spawnParticles(b.x, b.y, t.color, 8);

                // Bounce off target slightly
                const nx = dx / (dist || 1);
                const ny = dy / (dist || 1);
                b.vx += nx * 2;
                b.vy += ny * 2;

                if (t.hp <= 0) {
                    combo++;
                    comboTimer = 2;
                    const pts = t.points * combo;
                    score += pts;
                    addFloatText(t.x, t.y, `+${pts}`, '#fff');
                    if (combo > 1) addFloatText(t.x, t.y - 20, `x${combo} COMBO!`, '#ff6644');
                    spawnParticles(t.x, t.y, t.color, 20);
                    targets.splice(j, 1);
                    playSound('destroy');
                } else {
                    addFloatText(t.x, t.y, 'HIT', '#aaa');
                }
                break;
            }
        }
    }

    // Update particles
    for (let i = particles.length - 1; i >= 0; i--) {
        const p = particles[i];
        p.x += p.vx;
        p.y += p.vy;
        p.vy += GRAVITY * 0.5;
        p.life -= p.decay;
        if (p.life <= 0) particles.splice(i, 1);
    }

    // Update float texts
    for (let i = floatTexts.length - 1; i >= 0; i--) {
        const f = floatTexts[i];
        f.y -= 1;
        f.life -= 0.02;
        if (f.life <= 0) floatTexts.splice(i, 1);
    }

    // Check wave complete
    if (targets.length === 0 && bullets.length === 0) {
        nextWave();
    }

    // Game over
    if (shotsLeft <= 0 && bullets.length === 0 && targets.length > 0) {
        gameState = 'gameover';
    }
}

function drawCannon() {
    const cx = getCannonX();
    const cy = getCannonY();

    // Base
    ctx.fillStyle = '#334';
    ctx.beginPath();
    ctx.arc(cx, cy, 25, 0, Math.PI * 2);
    ctx.fill();
    ctx.fillStyle = '#556';
    ctx.beginPath();
    ctx.arc(cx, cy, 18, 0, Math.PI * 2);
    ctx.fill();

    // Barrel
    ctx.save();
    ctx.translate(cx, cy);
    ctx.rotate(cannonAngle);
    ctx.fillStyle = '#667';
    ctx.fillRect(0, -5, 40, 10);
    ctx.fillStyle = '#889';
    ctx.fillRect(30, -7, 12, 14);
    ctx.restore();

    // Muzzle glow
    const glowX = cx + Math.cos(cannonAngle) * 42;
    const glowY = cy + Math.sin(cannonAngle) * 42;
    const grad = ctx.createRadialGradient(glowX, glowY, 0, glowX, glowY, 10);
    grad.addColorStop(0, 'rgba(100,180,255,0.6)');
    grad.addColorStop(1, 'rgba(100,180,255,0)');
    ctx.fillStyle = grad;
    ctx.beginPath();
    ctx.arc(glowX, glowY, 10, 0, Math.PI * 2);
    ctx.fill();
}

function drawTargets() {
    for (const t of targets) {
        const wobble = Math.sin(t.wobble + performance.now() * 0.003) * 2;
        ctx.save();
        ctx.translate(t.x, t.y + wobble);

        // Shadow
        ctx.fillStyle = 'rgba(0,0,0,0.3)';
        ctx.beginPath();
        ctx.arc(3, 3, t.size, 0, Math.PI * 2);
        ctx.fill();

        // Body
        ctx.fillStyle = t.color;
        ctx.beginPath();
        ctx.arc(0, 0, t.size, 0, Math.PI * 2);
        ctx.fill();

        // Damage indicator
        if (t.hp < t.maxHp) {
            const ratio = t.hp / t.maxHp;
            ctx.strokeStyle = `rgba(255,255,255,${ratio})`;
            ctx.lineWidth = 3;
            ctx.beginPath();
            ctx.arc(0, 0, t.size - 4, -Math.PI / 2, -Math.PI / 2 + Math.PI * 2 * ratio);
            ctx.stroke();

            // Cracks
            if (t.hp < t.maxHp) {
                ctx.strokeStyle = 'rgba(0,0,0,0.4)';
                ctx.lineWidth = 1.5;
                for (let c = 0; c < (t.maxHp - t.hp); c++) {
                    const a = t.wobble + c * 2.3;
                    ctx.beginPath();
                    ctx.moveTo(Math.cos(a) * 5, Math.sin(a) * 5);
                    ctx.lineTo(Math.cos(a) * t.size * 0.7, Math.sin(a) * t.size * 0.7);
                    ctx.stroke();
                }
            }
        }

        // Highlight
        ctx.fillStyle = 'rgba(255,255,255,0.3)';
        ctx.beginPath();
        ctx.arc(-t.size * 0.3, -t.size * 0.3, t.size * 0.3, 0, Math.PI * 2);
        ctx.fill();

        // Points label
        ctx.fillStyle = 'rgba(255,255,255,0.7)';
        ctx.font = 'bold 12px monospace';
        ctx.textAlign = 'center';
        ctx.fillText(t.points, 0, 4);

        ctx.restore();
    }
}

function drawBullets() {
    for (const b of bullets) {
        // Trail
        for (let i = 0; i < b.trail.length; i++) {
            const t = b.trail[i];
            const alpha = (i / b.trail.length) * 0.5;
            const r = (i / b.trail.length) * b.radius * 0.8;
            ctx.fillStyle = `rgba(100,180,255,${alpha})`;
            ctx.beginPath();
            ctx.arc(t.x, t.y, r, 0, Math.PI * 2);
            ctx.fill();
        }

        // Bullet
        const grad = ctx.createRadialGradient(b.x, b.y, 0, b.x, b.y, b.radius);
        grad.addColorStop(0, '#fff');
        grad.addColorStop(0.5, '#6bf');
        grad.addColorStop(1, 'rgba(100,180,255,0)');
        ctx.fillStyle = grad;
        ctx.beginPath();
        ctx.arc(b.x, b.y, b.radius * 1.5, 0, Math.PI * 2);
        ctx.fill();

        ctx.fillStyle = '#fff';
        ctx.beginPath();
        ctx.arc(b.x, b.y, b.radius * 0.5, 0, Math.PI * 2);
        ctx.fill();
    }
}

function drawParticles() {
    for (const p of particles) {
        ctx.globalAlpha = p.life;
        ctx.fillStyle = p.color;
        ctx.beginPath();
        ctx.arc(p.x, p.y, p.size * p.life, 0, Math.PI * 2);
        ctx.fill();
    }
    ctx.globalAlpha = 1;
}

function drawFloatTexts() {
    for (const f of floatTexts) {
        ctx.globalAlpha = f.life;
        ctx.fillStyle = f.color;
        ctx.font = 'bold 16px monospace';
        ctx.textAlign = 'center';
        ctx.fillText(f.text, f.x, f.y);
    }
    ctx.globalAlpha = 1;
}

function drawAimLine() {
    if (gameState !== 'playing') return;
    const cx = getCannonX();
    const cy = getCannonY();
    const speed = 12;
    let px = cx + Math.cos(cannonAngle) * 40;
    let py = cy + Math.sin(cannonAngle) * 40;
    let pvx = Math.cos(cannonAngle) * speed;
    let pvy = Math.sin(cannonAngle) * speed;

    ctx.setLineDash([4, 6]);
    ctx.strokeStyle = 'rgba(100,180,255,0.3)';
    ctx.lineWidth = 1;
    ctx.beginPath();
    ctx.moveTo(px, py);

    for (let i = 0; i < 60; i++) {
        pvy += GRAVITY;
        px += pvx;
        py += pvy;
        if (px < 0 || px > W || py < 0) break;
        ctx.lineTo(px, py);
    }
    ctx.stroke();
    ctx.setLineDash([]);
}

function drawHUD() {
    // Score
    ctx.fillStyle = '#fff';
    ctx.font = 'bold 24px monospace';
    ctx.textAlign = 'left';
    ctx.fillText(`SCORE: ${score}`, 20, 35);

    // Wave
    ctx.font = '16px monospace';
    ctx.fillText(`WAVE ${wave}`, 20, 60);

    // Shots
    ctx.textAlign = 'right';
    ctx.font = 'bold 20px monospace';
    const shotColor = shotsLeft <= 3 ? '#f44' : '#4f8';
    ctx.fillStyle = shotColor;
    ctx.fillText(`SHOTS: ${shotsLeft}`, W - 20, 35);

    // Combo
    if (combo > 1 && comboTimer > 0) {
        ctx.textAlign = 'center';
        ctx.font = 'bold 28px monospace';
        ctx.fillStyle = `rgba(255,100,60,${Math.min(comboTimer / 2, 1)})`;
        ctx.fillText(`COMBO x${combo}`, W / 2, 40);
    }

    // Shot indicator bar
    const barW = 200;
    const barX = W / 2 - barW / 2;
    const barY = H - 30;
    ctx.fillStyle = 'rgba(255,255,255,0.1)';
    ctx.fillRect(barX, barY, barW, 8);
    ctx.fillStyle = shotsLeft <= 3 ? '#f44' : '#4f8';
    ctx.fillRect(barX, barY, barW * (shotsLeft / maxShots), 8);
}

function drawBackground() {
    // Sky gradient
    const bg = ctx.createLinearGradient(0, 0, 0, H);
    bg.addColorStop(0, '#0a0a2a');
    bg.addColorStop(0.7, '#141428');
    bg.addColorStop(1, '#1a1a30');
    ctx.fillStyle = bg;
    ctx.fillRect(0, 0, W, H);

    // Stars
    const starSeed = 12345;
    for (let i = 0; i < 80; i++) {
        const sx = ((starSeed * (i + 1) * 7919) % 10000) / 10000 * W;
        const sy = ((starSeed * (i + 1) * 6271) % 10000) / 10000 * H * 0.7;
        const twinkle = Math.sin(performance.now() * 0.001 + i) * 0.5 + 0.5;
        ctx.fillStyle = `rgba(255,255,255,${twinkle * 0.6})`;
        ctx.beginPath();
        ctx.arc(sx, sy, 1, 0, Math.PI * 2);
        ctx.fill();
    }

    // Ground
    ctx.fillStyle = '#1a1a2e';
    ctx.fillRect(0, H - 15, W, 15);
    ctx.strokeStyle = 'rgba(100,180,255,0.2)';
    ctx.lineWidth = 1;
    ctx.beginPath();
    ctx.moveTo(0, H - 15);
    ctx.lineTo(W, H - 15);
    ctx.stroke();
}

function drawStartScreen() {
    drawBackground();

    ctx.fillStyle = '#fff';
    ctx.font = 'bold 48px monospace';
    ctx.textAlign = 'center';
    ctx.fillText('GRAVITY BLAST', W / 2, H * 0.3);

    ctx.font = '18px monospace';
    ctx.fillStyle = '#aaccf';
    ctx.fillText('A physics-based target shooter', W / 2, H * 0.4);

    ctx.font = '16px monospace';
    ctx.fillStyle = '#889';
    const lines = [
        'AIM with your mouse',
        'CLICK to fire projectiles',
        'Bullets obey GRAVITY and BOUNCE off walls',
        'Destroy all targets before running out of shots',
        'Chain kills for COMBO multipliers!',
        '',
        'Wood (10pts) | Stone (25pts) | Metal (50pts) | Gold (100pts)'
    ];
    lines.forEach((line, i) => {
        ctx.fillText(line, W / 2, H * 0.48 + i * 28);
    });

    // Blinking prompt
    const blink = Math.sin(performance.now() * 0.005) * 0.5 + 0.5;
    ctx.fillStyle = `rgba(100,180,255,${blink})`;
    ctx.font = 'bold 22px monospace';
    ctx.fillText('CLICK TO START', W / 2, H * 0.85);
}

function drawGameOver() {
    ctx.fillStyle = 'rgba(0,0,0,0.7)';
    ctx.fillRect(0, 0, W, H);

    ctx.fillStyle = '#f44';
    ctx.font = 'bold 42px monospace';
    ctx.textAlign = 'center';
    ctx.fillText('GAME OVER', W / 2, H * 0.35);

    ctx.fillStyle = '#fff';
    ctx.font = '24px monospace';
    ctx.fillText(`Final Score: ${score}`, W / 2, H * 0.45);
    ctx.fillText(`Reached Wave: ${wave}`, W / 2, H * 0.52);

    const blink = Math.sin(performance.now() * 0.005) * 0.5 + 0.5;
    ctx.fillStyle = `rgba(100,180,255,${blink})`;
    ctx.font = 'bold 20px monospace';
    ctx.fillText('CLICK TO RESTART', W / 2, H * 0.7);
}

function render() {
    if (gameState === 'start') {
        drawStartScreen();
        return;
    }

    drawBackground();
    drawTargets();
    drawBullets();
    drawParticles();
    drawFloatTexts();
    drawAimLine();
    drawCannon();
    drawHUD();

    if (gameState === 'gameover') {
        drawGameOver();
    }
}

function gameLoop() {
    update();
    render();
    requestAnimationFrame(gameLoop);
}

// Input
window.addEventListener('mousemove', (e) => {
    mouseX = e.clientX;
    mouseY = e.clientY;
    const cx = getCannonX();
    const cy = getCannonY();
    cannonAngle = Math.atan2(mouseY - cy, mouseX - cx);
    // Clamp angle to upper half
    if (cannonAngle > 0 && cannonAngle < Math.PI) {
        cannonAngle = mouseY > cy ? (mouseX > cx ? 0.1 : Math.PI - 0.1) : cannonAngle;
    }
});

window.addEventListener('click', (e) => {
    initAudio();
    if (gameState === 'start') {
        startGame();
    } else if (gameState === 'playing') {
        fireBullet();
    } else if (gameState === 'gameover') {
        startGame();
    }
});

window.addEventListener('keydown', (e) => {
    initAudio();
    if (e.code === 'Space' && gameState === 'playing') {
        e.preventDefault();
        fireBullet();
    }
    if (e.code === 'Enter' || e.code === 'KeyR') {
        if (gameState === 'start' || gameState === 'gameover') {
            startGame();
        }
    }
});

// High score
function saveHighScore() {
    try {
        const hs = JSON.parse(localStorage.getItem('gravityBlastHS') || '0');
        if (score > hs) localStorage.setItem('gravityBlassHS', JSON.stringify(score));
    } catch (e) {}
}
window.addEventListener('beforeunload', saveHighScore);

// Start loop
gameLoop();
</script>
</body>
