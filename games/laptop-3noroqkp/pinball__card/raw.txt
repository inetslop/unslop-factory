// Audio
let audioCtx = null;
function initAudio() {
    if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
}
function playSound(freq, duration, type, vol) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = type || 'sine';
    osc.frequency.value = freq;
    gain.gain.setValueAtTime(vol || 0.3, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + duration);
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + duration);
}

// Game state
let gameState = 'start'; // start, playing, gameover
let score = 0;
let lives = 3;
let level = 1;
let cards = [];
let ball = null;
let flippers = [];
let particles = [];
let popups = [];
let animTime = 0;

const CARD_SUITS = ['♠', '♥', '♦', '♣'];
const CARD_VALUES = [
    { label: 'A', points: 50, color: '#fff' },
    { label: 'K', points: 30, color: '#ff6b6b' },
    { label: 'Q', points: 20, color: '#4ecdc4' },
    { label: 'J', points: 15, color: '#ffd93d' },
    { label: '10', points: 10, color: '#a855f7' }
];

function createCards() {
    cards = [];
    const tableW = Math.min(W * 0.8, 500);
    const tableH = H * 0.7;
    const offsetX = (W - tableW) / 2;
    const offsetY = 30;

    const positions = [
        { x: 0.2, y: 0.1 }, { x: 0.5, y: 0.08 }, { x: 0.8, y: 0.1 },
        { x: 0.3, y: 0.3 }, { x: 0.7, y: 0.3 },
        { x: 0.5, y: 0.45 },
        { x: 0.2, y: 0.6 }, { x: 0.8, y: 0.6 }
    ];

    const shuffled = [...CARD_VALUES].sort(() => Math.random() - 0.5);
    for (let i = 0; i < positions.length; i++) {
        const val = CARD_VALUES[i % CARD_VALUES.length];
        cards.push({
            x: offsetX + positions[i].x * tableW,
            y: offsetY + positions[i].y * tableH,
            w: 50,
            h: 70,
            value: val,
            suit: CARD_SUITS[i % 4],
            hit: false,
            hitAnim: 0,
            glow: 0
        });
    }
}

function createFlippers() {
    const tableW = Math.min(W * 0.8, 500);
    const offsetX = (W - tableW) / 2;
    const flipperY = H - 60;
    const gap = 40;
    flippers = [
        { x: W/2 - gap, y: flipperY, len: 80, angle: 0.4, targetAngle: 0.4, restAngle: 0.4, activeAngle: -0.5, side: 'left' },
        { x: W/2 + gap, y: flipperY, len: 80, angle: -0.4, targetAngle: -0.4, restAngle: -0.4, activeAngle: 0.5, side: 'right' }
    ];
}

function createBall(x, y) {
    return { x: x || W/2, y: y || 80, vx: (Math.random() - 0.5) * 3, vy: 0, r: 12 };
}

function startGame() {
    score = 0;
    lives = 3;
    level = 1;
    createCards();
    createFlippers();
    ball = createBall();
    particles = [];
    popups = [];
    gameState = 'playing';

function nextLevel() {
    level++;

// Particles
function spawnParticles(x, y, color, count) {
    for (let i = 0; i < count; i++) {
        const angle = Math.random() * Math.PI * 2;
        const speed = 1 + Math.random() * 4;
        particles.push({
            x, y,
            vx: Math.cos(angle) * speed,
            vy: Math.sin(angle) * speed - 2,
            life: 1,
            color,
            size: 3 + Math.random() * 5
        });
    }
}

function spawnPopup(x, y, text, color) {
    popups.push({ x, y, text, color, life: 1 });

// Physics
function update(dt) {
    if (gameState !== 'playing') return;

    const gravity = 0.35;
    const damping = 0.998;
    const bounce = 0.75;

    // Update flippers
    for (let f of flippers) {
        if (f.side === 'left') {
            if (keys['ArrowLeft'] || keys['KeyA']) f.targetAngle = -0.5;
            else f.targetAngle = 0.4;
        } else {
            if (keys['ArrowRight'] || keys['KeyD']) f.targetAngle = 0.5;
            else f.targetAngle = -0.4;
        }
        f.angle += (f.targetAngle - f.angle) * 0.3;
    }

    // Ball physics
    if (ball) {
        ball.vy += gravity;
        ball.vx *= damping;
        ball.vy *= damping;
        ball.x += ball.vx;
        ball.y += ball.vy;

        const tableW = Math.min(W * 0.8, 500);
        const offsetX = (W - tableW) / 2;
        const leftWall = offsetX + 10;
        const rightWall = offsetX + tableW - 10;
        const topWall = 30;

        // Wall collisions
        if (ball.x - ball.r < leftWall) { ball.x = leftWall + ball.r; ball.vx = Math.abs(ball.vx) * bounce; playSound(200, 0.1, 'triangle', 0.2); }
        if (ball.x + ball.r > rightWall) { ball.x = rightWall - ball.r; ball.vx = -Math.abs(ball.vx) * bounce; playSound(200, 0.1, 'triangle', 0.2); }
        if (ball.y - ball.r < topWall) { ball.y = topWall + ball.r; ball.vy = Math.abs(ball.vy) * bounce; playSound(200, 0.1, 'triangle', 0.2); }

        // Card collisions
        for (let c of cards) {
            if (c.hit) continue;
            const cx = c.x, cy = c.y, hw = c.w/2 + ball.r, hh = c.h/2 + ball.r;
            const dx = ball.x - cx, dy = ball.y - cy;
            if (Math.abs(dx) < hw && Math.abs(dy) < hh) {
                // Determine which side hit
                const overlapX = hw - Math.abs(dx);
                const overlapY = hh - Math.abs(dy);
                if (overlapX < overlapY) {
                    ball.vx = (dx > 0 ? 1 : -1) * Math.abs(ball.vx) * bounce + (dx > 0 ? 2 : -2);
                    ball.x = cx + (dx > 0 ? hw : -hw);
                } else {
                    ball.vy = (dy > 0 ? 1 : -1) * Math.abs(ball.vy) * bounce + (dy > 0 ? 2 : -2);
                    ball.y = cy + (dy > 0 ? hh : -hh);
                }
                c.hit = true;
                c.hitAnim = 1;
                score += c.value.points * level;
                spawnParticles(c.x, c.y, c.value.color, 15);
                spawnPopup(c.x, c.y - 20, '+' + (c.value.points * level), c.value.color);
                playSound(400 + c.value.points * 5, 0.3, 'square', 0.25);

                // Check if all cards hit
                if (cards.every(cc => cc.hit)) {
                    setTimeout(() => nextLevel(), 800);
                    spawnPopup(W/2, H/2, 'LEVEL UP!', '#ffd700');
                    playSound(600, 0.5, 'sine', 0.3);
                }
            }
        }

        // Flipper collisions
        for (let f of flippers) {
            const tipX = f.x + Math.cos(f.angle * (f.side === 'left' ? -1 : 1)) * f.len;
            const tipY = f.y + Math.sin(f.angle * (f.side === 'left' ? -1 : 1)) * f.len;

            // Line segment collision (simplified as circle at tip and base)
            const distToBase = Math.hypot(ball.x - f.x, ball.y - f.y);
            if (distToBase < ball.r + 8) {
                const angle = Math.atan2(ball.y - f.y, ball.x - f.x);
                ball.vx = Math.cos(angle) * 5;
                ball.vy = Math.sin(angle) * 5 - 3;
                playSound(300, 0.1, 'triangle', 0.2);
            }

            const distToTip = Math.hypot(ball.x - tipX, ball.y - tipY);
            if (distToTip < ball.r + 8) {
                const angle = Math.atan2(ball.y - tipY, ball.x - tipX);
                ball.vx = Math.cos(angle) * 5;
                ball.vy = Math.sin(angle) * 5 - 3;
                playSound(300, 0.1, 'triangle', 0.2);
            }

            // Flipper body collision (simplified)
            const midX = (f.x + tipX) / 2;
            const midY = (f.y + tipY) / 2;
            const distToMid = Math.hypot(ball.x - midX, ball.y - midY);
            if (distToMid < ball.r + 6) {
                const angle = Math.atan2(ball.y - midY, ball.x - midX);
                ball.vx = Math.cos(angle) * 4;
                ball.vy = Math.sin(angle) * 4 - 3;
            }
        }

        // Ball lost
        if (ball.y > H + 50) {
            lives--;
            playSound(150, 0.5, 'sawtooth', 0.3);
            if (lives <= 0) {
                gameState = 'gameover';
            } else {
                ball = createBall();
            }
        }
    }

    // Update particles
    for (let p of particles) {
        p.x += p.vx;
        p.y += p.vy;
        p.vy += 0.2;
        p.life -= 0.03;
    }
    particles = particles.filter(p => p.life > 0);

    // Update popups
    for (let p of popups) {
        p.y -= 1.5;
        p.life -= 0.02;
    }
    popups = popups.filter(p => p.life > 0);

    // Card hit animation
    for (let c of cards) {
        if (c.hit && c.hitAnim > 0) c.hitAnim -= 0.05;
    }
}

// Rendering
function drawCard(c) {
    ctx.save();
    const x = c.x - c.w/2, y = c.y - c.h/2;

    if (c.hit) {
        ctx.globalAlpha = Math.max(0, 1 - c.hitAnim);
        const scale = 1 + c.hitAnim * 0.5;
        ctx.translate(c.x, c.y);
        ctx.scale(scale, scale);
        ctx.translate(-c.x, -c.y);
    }

    // Card body
    ctx.fillStyle = c.hit ? 'rgba(255,255,255,0.3)' : '#fff';
    ctx.strokeStyle = c.hit ? 'rgba(100,100,100,0.5)' : c.value.color;
    ctx.lineWidth = 3;

    // Rounded rect
    const r = 8;
    ctx.beginPath();
    ctx.roundRect(x, y, c.w, c.h, r);
    ctx.fill();
    ctx.stroke();

    // Card content
    if (!c.hit) {
        ctx.fillStyle = c.suit === '♥' || c.suit === '♦' ? '#e63946' : '#1a1a2e';
        ctx.font = 'bold 14px monospace';
        ctx.textAlign = 'center';
        ctx.fillText(c.value.label, x + 15, y + 20);
        ctx.font = '20px serif';
        ctx.fillText(c.suit, c.x, c.y + 8);

        // Glow effect
        if (c.glow > 0) {
            ctx.shadowColor = c.value.color;
            ctx.shadowBlur = 15;
            ctx.strokeStyle = c.value.color;
            ctx.lineWidth = 2;
            ctx.strokeRect(x+3, y+3, c.w-6, c.h-6);
        }

        // Points label
        ctx.fillStyle = '#666';
        ctx.font = '10px monospace';
        ctx.fillText(c.value.points + 'pts', x + c.w/2, y + c.h - 8);
    } else {
        ctx.fillStyle = 'rgba(150,150,150,0.5)';
        ctx.font = '24px serif';
        ctx.textAlign = 'center';
        ctx.fillText('✓', c.x, c.y + 8);
    }

    ctx.restore();
}

function drawFlipper(f) {
    const dir = f.side === 'left' ? -1 : 1;
    const tipX = f.x + Math.cos(f.angle * dir) * f.len;
    const tipY = f.y + Math.sin(f.angle * dir) * f.len;

    ctx.strokeStyle = '#ffd700';
    ctx.lineWidth = 8;
    ctx.lineCap = 'round';
    ctx.beginPath();
    ctx.moveTo(f.x, f.y);
    ctx.lineTo(tipX, tipY);
    ctx.stroke();

    // Base circle
    ctx.fillStyle = '#ffd700';
    ctx.beginPath();
    ctx.arc(f.x, f.y, 8, 0, Math.PI * 2);
    ctx.fill();
}

function drawBall() {
    if (!ball) return;
    const grad = ctx.createRadialGradient(ball.x - 3, ball.y - 3, 2, ball.x, ball.y, ball.r);
    grad.addColorStop(0, '#fff');
    grad.addColorStop(0.5, '#c0c0ff');
    grad.addColorStop(1, '#4a4a8a');
    ctx.fillStyle = grad;
    ctx.beginPath();
    ctx.arc(ball.x, ball.y, ball.r, 0, Math.PI * 2);
    ctx.fill();

    // Shadow
    ctx.fillStyle = 'rgba(0,0,0,0.3)';
    ctx.beginPath();
    ctx.ellipse(ball.x + 4, ball.y + 6, ball.r * 0.8, ball.r * 0.4, 0, 0, Math.PI * 2);
    ctx.fill();
}

function drawTable() {
    const tableW = Math.min(W * 0.8, 500);
    const offsetX = (W - tableW) / 2;

    // Table background
    const grad = ctx.createLinearGradient(offsetX, 0, offsetX + tableW, H);
    grad.addColorStop(0, '#1a1a3e');
    grad.addColorStop(1, '#0d0d2b');
    ctx.fillStyle = grad;
    ctx.beginPath();
    ctx.roundRect(offsetX, 30, tableW, H - 90, 15);
    ctx.fill();

    // Table border
    ctx.strokeStyle = '#4a4a8a';
    ctx.lineWidth = 4;
    ctx.stroke();

    // Inner glow lines
    ctx.strokeStyle = 'rgba(100,100,255,0.1)';
    ctx.lineWidth = 1;
    for (let i = 0; i < 5; i++) {
        ctx.beginPath();
        ctx.moveTo(offsetX + 20, 30 + i * 40);
        ctx.lineTo(offsetX + tableW - 20, 30 + i * 40);
        ctx.stroke();
    }
}

function drawHUD() {
    // Score
    ctx.fillStyle = '#fff';
    ctx.font = 'bold 24px monospace';
    ctx.textAlign = 'left';
    ctx.fillText('Score: ' + score, 20, 30);

    // Level
    ctx.textAlign = 'center';
    ctx.fillText('Level: ' + level, W/2, 30);

    // Lives
    ctx.textAlign = 'right';
    let livesStr = '';
    for (let i = 0; i < lives; i++) livesStr += '● ';

    // Remaining cards
    const remaining = cards.filter(c => !c.hit).length;
    ctx.textAlign = 'center';
    ctx.fillStyle = '#aaa';
    ctx.font = '14px monospace';

}

function drawStartScreen() {
    // Background
    const grad = ctx.createLinearGradient(0, 0, 0, H);
    grad.addColorStop(0, '#1a0a2e');
    grad.addColorStop(1, '#0d0520');
    ctx.fillStyle = grad;
    ctx.fillRect(0, 0, W, H);

    // Title
    ctx.textAlign = 'center';
    ctx.fillStyle = '#ffd700';
    ctx.font = 'bold 48px monospace';
    ctx.fillText('PINBALL CARDS', W/2, H * 0.25);

    // Subtitle
    ctx.fillStyle = '#a855f7';
    ctx.font = '18px monospace';
    ctx.fillText('Hit all the cards to clear each level!', W/2, H * 0.35);

    // Controls
    ctx.fillStyle = '#ccc';
    ctx.font = '16px monospace';
    ctx.fillText('← → or A/D : Flippers', W/2, H * 0.48);
    ctx.fillText('Ball bounces off walls and cards', W/2, H * 0.54);
    ctx.fillText('Face cards & Aces score more points!', W/2, H * 0.60);
    ctx.fillText('Don\'t let the ball fall off the bottom!', W/2, H * 0.66);

    // Card preview
    const cardLabels = ['A', 'K', 'Q', 'J', '10'];
    const cardColors = ['#fff', '#ff6b6b', '#4ecdc4', '#ffd93d', '#a855f7'];
    for (let i = 0; i < 5; i++) {
        const cx = W/2 - 120 + i * 50;
        const cy = H * 0.78;
        ctx.fillStyle = '#fff';
        ctx.strokeStyle = cardColors[i];
        ctx.lineWidth = 2;
        ctx.beginPath();
        ctx.roundRect(cx - 18, cy - 25, 36, 50, 5);
        ctx.fill();
        ctx.stroke();
        ctx.fillStyle = i === 1 ? '#e63946' : '#1a1a2e';
        ctx.font = 'bold 16px monospace';
        ctx.fillText(cardLabels[i], cx, cy + 5);
    }

    // Start prompt
    const pulse = 0.5 + Math.sin(animTime * 3) * 0.5;
    ctx.globalAlpha = pulse;
    ctx.fillStyle = '#fff';
    ctx.font = 'bold 20px monospace';
    ctx.fillText('Press SPACE to Start', W/2, H * 0.92);
    ctx.globalAlpha = 1;

function drawGameOver() {
    ctx.fillStyle = 'rgba(0,0,0,0.8)';
    ctx.fillRect(0, 0, W, H);

    ctx.textAlign = 'center';
    ctx.fillStyle = '#ff6b6b';
    ctx.font = 'bold 42px monospace';
    ctx.fillText('GAME OVER', W/2, H * 0.35);

    ctx.fillStyle = '#ffd700';
    ctx.font = '28px monospace';
    ctx.fillText('Final Score: ' + score, W/2, H * 0.48);
    ctx.fillStyle = '#aaa';
    ctx.font = '18px monospace';
    ctx.fillText('Reached Level: ' + level, W/2, H * 0.56);

    const pulse = 0.5 + Math.sin(animTime * 3) * 0.5;
    ctx.globalAlpha = pulse;
    ctx.fillStyle = '#fff';
    ctx.font = '18px monospace';
    ctx.fillText('Press SPACE or R to Restart', W/2, H * 0.7);
    ctx.globalAlpha = 1;
}

function render() {
    ctx.clearRect(0, 0, W, H);

    if (gameState === 'start') {
        drawStartScreen();
        return;
    }

    // Background
    const bgGrad = ctx.createLinearGradient(0, 0, 0, H);
    bgGrad.addColorStop(0, '#1a0a2e');
    bgGrad.addColorStop(1, '#0d0520');
    ctx.fillStyle = bgGrad;
    ctx.fillRect(0, 0, W, H);

    drawTable();

    // Cards
    for (let c of cards) drawCard(c);

    // Flippers
    for (let f of flippers) drawFlipper(f);

    // Ball
    drawBall();

    // Particles
    for (let p of particles) {
        ctx.globalAlpha = p.life;
        ctx.fillStyle = p.color;
        ctx.beginPath();
        ctx.arc(p.x, p.y, p.size * p.life, 0, Math.PI * 2);
        ctx.fill();
    }
    ctx.globalAlpha = 1;

    // Popups
    for (let p of popups) {
        ctx.globalAlpha = p.life;
        ctx.fillStyle = p.color;
        ctx.font = 'bold 20px monospace';
        ctx.textAlign = 'center';
        ctx.fillText(p.text, p.x, p.y);
    }
    ctx.globalAlpha = 1;

    // HUD
    drawHUD();

    if (gameState === 'gameover') {
        drawGameOver();
    }
}

// Game loop
let lastTime = 0;
function gameLoop(time) {
    animTime += 0.016;
    const dt = Math.min((time - lastTime) / 16.67, 3);
    lastTime = time;

    update(dt);
    render();

    requestAnimationFrame(gameLoop);
}

requestAnimationFrame(gameLoop);
