const canvas = document.getElementById('c');
const ctx = canvas.getContext('2d');

let W, H;
function resize() {
  W = canvas.width = window.innerWidth;
  H = canvas.height = window.innerHeight;
}
window.addEventListener('resize', resize);
resize();

// Audio
let audioCtx = null;
function initAudio() {
  if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
}
function playSound(freq, type, duration, vol) {
  if (!audioCtx) return;
  const osc = audioCtx.createOscillator();
  const gain = audioCtx.createGain();
  osc.type = type;
  osc.frequency.value = freq;
  gain.gain.setValueAtTime(vol || 0.3, audioCtx.currentTime);
  gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + (duration || 0.1));
  osc.connect(gain);
  gain.connect(audioCtx.destination);
  osc.start();
  osc.stop(audioCtx.currentTime + (duration || 0.1));
}

// Game state
const STATE = { START: 0, PLAYING: 1, LEVEL_COMPLETE: 2, GAME_OVER: 3, WIN: 4 };
let gameState = STATE.START;
let score = 0;
let lives = 3;
let level = 0;
let particles = [];
let shakeTime = 0;

// Paddle
let paddle = { x: 0, w: 120, h: 18, speed: 8 };

// Crates (Sokoban boxes)
let crates = [];
const CRATE_SIZE = 36;
const FLOOR_Y_RATIO = 0.85;

// Bricks
let bricks = [];
const BRICK_COLS = 8;
const BRICK_ROWS_MAX = 6;

// Level definitions
const levelDefs = [
  // Level 1: Simple rows
  () => {
    return Array.from({length: 4}, (_, r) => 
      Array.from({length: BRICK_COLS}, (_, c) => ({hp: 1, maxHp: 1, alive: true}))
    );
  },
  // Level 2: Gap in middle
  () => {
    return Array.from({length: 5}, (_, r) => 
      Array.from({length: BRICK_COLS}, (_, c) => {
        if (r === 2 && (c === 3 || c === 4)) return {hp: 0, maxHp: 0, alive: false};
        return {hp: 1, maxHp: 1, alive: true};
      })
    );
  },
  // Level 3: Diamond pattern
  () => {
    return Array.from({length: 6}, (_, r) => 
      Array.from({length: BRICK_COLS}, (_, c) => {
        const dist = Math.abs(c - 3.5) + Math.abs(r - 2.5);
        if (dist > 4) return {hp: 0, maxHp: 0, alive: false};
        return {hp: r < 2 ? 2 : 1, maxHp: r < 2 ? 2 : 1, alive: true};
      })
    );
  },
  // Level 4: Fortified
  () => {
    return Array.from({length: 6}, (_, r) => 
      Array.from({length: BRICK_COLS}, (_, c) => {
        if (r === 5 && (c < 2 || c > 5)) return {hp: 0, maxHp: 0, alive: false};
        return {hp: r <= 1 ? 3 : 2, maxHp: r <= 1 ? 3 : 2, alive: true};
      })
    );
  },
  // Level 5: Full fortress
  () => {
    return Array.from({length: 6}, (_, r) => 
      Array.from({length: BRICK_COLS}, (_, c) => {
        if (r === 0 && c % 2 === 0) return {hp: 4, maxHp: 4, alive: true};
        if (r === 5 && c % 3 === 1) return {hp: 0, maxHp: 0, alive: false};
        return {hp: 3, maxHp: 3, alive: true};
      })
    );
  }
];

function loadLevel(lvl) {
  level = lvl;
  const def = levelDefs[lvl]();
  bricks = [];
  const brickW = Math.min(70, (W - 80) / BRICK_COLS);
  const brickH = 28;
  const startX = (W - BRICK_COLS * (brickW + 6) + 6 * (BRICK_COLS-1)) / 2;
  const startY = 60;

  for (let r = 0; r < def.length; r++) {
    for (let c = 0; c < BRICK_COLS; c++) {
      if (def[r][c].alive) {
        bricks.push({
          x: startX + c * (brickW + 6),
          y: startY + r * (brickH + 6),
          w: brickW, h: brickH,
          hp: def[r][c].hp, maxHp: def[r][c].maxHp
        });
      }
    }
  }

  paddle.x = W / 2 - paddle.w / 2;
  initCrates();
}

function initCrates() {
  const floorY = H * FLOOR_Y_RATIO;
  const count = Math.min(3, 2 + Math.floor(level / 2));
  crates = [];
  for (let i = 0; i < count; i++) {
    crates.push({
      x: W / 2 - (count - 1) * (CRATE_SIZE + 10) / 2 + i * (CRATE_SIZE + 10),
      y: floorY,
      vx: 0, vy: 0,
      launched: false,
      active: true
    });
  }
}

function resetGame() {
  score = 0;
  lives = 3;
  loadLevel(0);
  gameState = STATE.PLAYING;
}

// Input
const keys = {};
window.addEventListener('keydown', e => {
  keys[e.code] = true;
  initAudio();
  if (gameState === STATE.START && (e.code === 'Space' || e.code === 'Enter')) {
    resetGame();
  } else if (gameState === STATE.GAME_OVER && (e.code === 'Space' || e.code === 'Enter')) {
    resetGame();
  } else if (gameState === STATE.WIN && (e.code === 'Space' || e.code === 'Enter')) {
    resetGame();
  } else if (gameState === STATE.LEVEL_COMPLETE && (e.code === 'Space' || e.code === 'Enter')) {
    loadLevel(level + 1);
    gameState = STATE.PLAYING;
  }
  if (gameState === STATE.PLAYING && e.code === 'Space') {
    launchCrates();
  }
});
window.addEventListener('keyup', e => { keys[e.code] = false; });

function launchCrates() {
  let launched = false;
  for (let c of crates) {
    if (!c.launched && c.active) {
      c.launched = true;
      const relX = (c.x - paddle.x) / paddle.w;
      c.vx = (relX - 0.5) * 6;
      c.vy = -12 - Math.random() * 3;
      launched = true;
    }
  }
  if (launched) playSound(200, 'square', 0.15, 0.2);
}

function spawnParticles(x, y, color, count) {
  for (let i = 0; i < count; i++) {
    const angle = Math.random() * Math.PI * 2;
    const speed = 1 + Math.random() * 4;
    particles.push({
      x, y,
      vx: Math.cos(angle) * speed,
      vy: Math.sin(angle) * speed - 2,
      life: 30 + Math.random() * 20,
      maxLife: 50,
      color,
      size: 3 + Math.random() * 4
    });
  }
}

function update() {
  if (gameState !== STATE.PLAYING) return;

  // Paddle movement
  let paddleSpeed = paddle.speed;
  if (keys['KeyA'] || keys['ArrowLeft']) paddle.x -= paddleSpeed;
  if (keys['KeyD'] || keys['ArrowRight']) paddle.x += paddleSpeed;
  paddle.x = Math.max(0, Math.min(W - paddle.w, paddle.x));

  // Crate pushing (Sokoban)
  const floorY = H * FLOOR_Y_RATIO;
  let pushLeft = false, pushRight = false;
  if (keys['KeyA']) pushLeft = true;
  if (keys['KeyD']) pushRight = true;

  for (let c of crates) {
    if (!c.launched && c.active) {
      if (pushLeft) c.x -= 3;
      if (pushRight) c.x += 3;
      c.y = floorY;
      // Keep in bounds
      c.x = Math.max(CRATE_SIZE/2, Math.min(W - CRATE_SIZE/2, c.x));
    }
  }

  // Prevent crates from overlapping when unlaunched
  const unlaunched = crates.filter(c => !c.launched && c.active);
  for (let i = 0; i < unlaunched.length; i++) {
    for (let j = i+1; j < unlaunched.length; j++) {
      const a = unlaunched[i], b = unlaunched[j];
      if (Math.abs(a.x - b.x) < CRATE_SIZE) {
        const mid = (a.x + b.x) / 2;
        if (a.x < b.x) { a.x = mid - CRATE_SIZE/2; b.x = mid + CRATE_SIZE/2; }
        else { a.x = mid + CRATE_SIZE/2; b.x = mid - CRATE_SIZE/2; }
      }
    }
  }

  // Update launched crates
  const paddleY = H - 50;
  let lostLife = false;

  for (let c of crates) {
    if (!c.launched || !c.active) continue;

    c.x += c.vx;
    c.y += c.vy;
    c.vy += 0.15; // gravity

    // Wall bounce
    if (c.x < CRATE_SIZE/2) { c.x = CRATE_SIZE/2; c.vx = Math.abs(c.vx); playSound(300, 'triangle', 0.05, 0.1); }
    if (c.x > W - CRATE_SIZE/2) { c.x = W - CRATE_SIZE/2; c.vx = -Math.abs(c.vx); playSound(300, 'triangle', 0.05, 0.1); }
    if (c.y < 20) { c.y = 20; c.vy = Math.abs(c.vy); }

    // Brick collision
    for (let b of bricks) {
      if (!b.alive) continue;
      const cx = c.x, cy = c.y, cs = CRATE_SIZE/2;
      if (cx + cs > b.x && cx - cs < b.x + b.w && cy + cs > b.y && cy - cs < b.y + b.h) {
        // Determine bounce direction
        const overlapX = Math.min(cx + cs - b.x, b.x + b.w - (cx - cs));
        const overlapY = Math.min(cy + cs - b.y, b.y + b.h - (cy - cs));
        if (overlapX < overlapY) {
          c.vx = -c.vx;
        } else {
          c.vy = -c.vy;
        }
        hitBrick(b);
      }
    }

    // Paddle collision
    if (c.vy > 0 && c.y + CRATE_SIZE/2 >= paddleY && c.y + CRATE_SIZE/2 <= paddleY + paddle.h &&
        c.x >= paddle.x - CRATE_SIZE/2 && c.x <= paddle.x + paddle.w + CRATE_SIZE/2) {
      const relX = (c.x - paddle.x) / paddle.w;
      c.vy = -(10 + Math.random() * 3);
      c.vx = (relX - 0.5) * 8;
      playSound(440, 'sine', 0.1, 0.2);
    }

    // Fell below
    if (c.y > H + 50) {
      c.active = false;
      lostLife = true;
      playSound(100, 'sawtooth', 0.4, 0.3);
    }
  }

  if (lostLife) {
    lives--;
    shakeTime = 15;
    if (lives <= 0) {
      gameState = STATE.GAME_OVER;
    } else {
      // Reset crates
      initCrates();
    }
  }

  // Check level complete
  const remaining = bricks.filter(b => b.alive).length;
  if (remaining === 0 && gameState === STATE.PLAYING) {
    gameState = STATE.LEVEL_COMPLETE;
    playSound(600, 'sine', 0.3, 0.3);
    setTimeout(() => playSound(800, 'sine', 0.3, 0.3), 150);
    setTimeout(() => playSound(1000, 'sine', 0.4, 0.3), 300);
    if (level + 1 >= levelDefs.length) {
      gameState = STATE.WIN;
    }
  }

  // Particles
  for (let p of particles) {
    p.x += p.vx;
    p.y += p.vy;
    p.vy += 0.2;
    p.life--;
  }
  particles = particles.filter(p => p.life > 0);

  if (shakeTime > 0) shakeTime--;
}

function hitBrick(b) {
  b.hp--;
  if (b.hp <= 0) {
    b.alive = false;
    score += b.maxHp * 10;
    spawnParticles(b.x + b.w/2, b.y + b.h/2, getBrickColor(b.maxHp), 8);
    playSound(500 + Math.random() * 200, 'square', 0.1, 0.2);
  } else {
    score += 5;
    spawnParticles(b.x + b.w/2, b.y + b.h/2, getBrickColor(b.maxHp), 3);
    playSound(350, 'square', 0.08, 0.15);
  }
}

function getBrickColor(maxHp) {
  if (maxHp >= 4) return '#ff3366';
  if (maxHp === 3) return '#ff6633';
  if (maxHp === 2) return '#ffaa00';
  return '#44cc88';
}

function draw() {
  ctx.save();
  if (shakeTime > 0) {
    const sx = (Math.random() - 0.5) * shakeTime;
    const sy = (Math.random() - 0.5) * shakeTime;
    ctx.translate(sx, sy);
  }

  // Background
  ctx.fillStyle = '#0a0a1a';
  ctx.fillRect(-20, -20, W + 40, H + 40);

  // Grid lines (subtle)
  ctx.strokeStyle = 'rgba(30, 40, 80, 0.3)';
  ctx.lineWidth = 1;
  for (let x = 0; x < W; x += 40) {
    ctx.beginPath(); ctx.moveTo(x, 0); ctx.lineTo(x, H); ctx.stroke();
  }
  for (let y = 0; y < H; y += 40) {
    ctx.beginPath(); ctx.moveTo(0, y); ctx.lineTo(W, y); ctx.stroke();
  }

  if (gameState === STATE.START) {
    drawStartScreen();
    ctx.restore();
    return;
  }

  // Bricks
  for (let b of bricks) {
    if (!b.alive) continue;
    const color = getBrickColor(b.maxHp);
    ctx.fillStyle = color;
    ctx.fillRect(b.x, b.y, b.w, b.h);
    // Highlight
    ctx.fillStyle = 'rgba(255,255,255,0.2)';
    ctx.fillRect(b.x, b.y, b.w, 4);
    // Damage cracks
    if (b.hp < b.maxHp) {
      ctx.strokeStyle = 'rgba(0,0,0,0.4)';
      ctx.lineWidth = 1;
      const cx2 = b.x + b.w/2, cy2 = b.y + b.h/2;
      for (let i = 0; i < (b.maxHp - b.hp); i++) {
        ctx.beginPath();
        ctx.moveTo(cx2 - 10 + i*8, b.y);
        ctx.lineTo(cx2 - 5 + i*5, cy2);
        ctx.lineTo(cx2 + 5 - i*3, b.y + b.h);
        ctx.stroke();
      }
    }
    // Border
    ctx.strokeStyle = 'rgba(0,0,0,0.5)';
    ctx.lineWidth = 1;
    ctx.strokeRect(b.x, b.y, b.w, b.h);
  }

  // Crates (Sokoban boxes)
  for (let c of crates) {
    if (!c.active) continue;
    const x = c.x - CRATE_SIZE/2;
    const y = c.y - CRATE_SIZE/2;
    // Box body
    ctx.fillStyle = c.launched ? '#8B5E3C' : '#A0724A';
    ctx.fillRect(x, y, CRATE_SIZE, CRATE_SIZE);
    // Planks
    ctx.strokeStyle = '#6B4226';
    ctx.lineWidth = 2;
    ctx.strokeRect(x+2, y+2, CRATE_SIZE-4, CRATE_SIZE-4);
    ctx.beginPath();
    ctx.moveTo(x, y); ctx.lineTo(x + CRATE_SIZE, y + CRATE_SIZE);
    ctx.moveTo(x + CRATE_SIZE, y); ctx.lineTo(x, y + CRATE_SIZE);
    ctx.stroke();
    // Highlight if ready to launch
    if (!c.launched) {
      ctx.strokeStyle = 'rgba(255, 220, 100, ' + (0.3 + 0.2 * Math.sin(Date.now()/300)) + ')';
      ctx.lineWidth = 2;
      ctx.strokeRect(x-2, y-2, CRATE_SIZE+4, CRATE_SIZE+4);
    }
  }

  // Paddle
  const paddleY = H - 50;
  ctx.fillStyle = '#667788';
  ctx.fillRect(paddle.x, paddleY, paddle.w, paddle.h);
  ctx.fillStyle = 'rgba(255,255,255,0.3)';
  ctx.fillRect(paddle.x, paddleY, paddle.w, 4);
  ctx.strokeStyle = '#aabbcc';
  ctx.lineWidth = 1;
  ctx.strokeRect(paddle.x, paddleY, paddle.w, paddle.h);

  // Particles
  for (let p of particles) {
    const alpha = p.life / p.maxLife;
    ctx.fillStyle = p.color + Math.floor(alpha * 255).toString(16).padStart(2,'0');
    ctx.fillRect(p.x - p.size/2, p.y - p.size/2, p.size, p.size);
  }

  // HUD
  ctx.fillStyle = '#fff';
  ctx.font = 'bold 18px monospace';
  ctx.textAlign = 'left';
  ctx.fillText('SCORE: ' + score, 20, 30);
  ctx.textAlign = 'right';
  ctx.fillText('LIVES: ' + '❤'.repeat(Math.max(0, lives)), W - 20, 30);
  ctx.textAlign = 'center';
  ctx.fillText('LEVEL ' + (level + 1), W/2, 30);

  // Overlays
  if (gameState === STATE.GAME_OVER) {
    drawOverlay('GAME OVER', 'Press SPACE to restart');
  } else if (gameState === STATE.LEVEL_COMPLETE) {
    drawOverlay('LEVEL COMPLETE!', 'Press SPACE for next level');
  } else if (gameState === STATE.WIN) {
    drawOverlay('YOU WIN! 🎉', 'Score: ' + score + ' — Press SPACE to play again');
  }

  ctx.restore();
}

  ctx.textAlign = 'center';
  ctx.fillStyle = '#ffaa00';
  ctx.font = 'bold 48px monospace';
  ctx.fillText('SOKOBAN', W/2, H/3);
  ctx.fillStyle = '#44cc88';
  ctx.font = 'bold 48px monospace';
  ctx.fillText('BREAKOUT', W/2, H/3 + 55);

  ctx.fillStyle = '#aabbcc';
  ctx.font = '16px monospace';
  const lines = [
    '',
    'Push crates with A/D keys (Sokoban style)',
    'Move paddle with Arrow Keys',
    'Press SPACE to launch crates at bricks!',
    '',
    'Crates bounce off the paddle — don\'t lose them below!',
    '',
    'Break all bricks to advance through 5 levels.',
    '',
    'Press SPACE or ENTER to start'
  ];
  lines.forEach((line, i) => {
    ctx.fillText(line, W/2, H/3 + 120 + i * 24);
  });

  // Animated crates
  const t = Date.now() / 500;
  for (let i = 0; i < 3; i++) {
    const cx = W/2 - 60 + i * 60;
    const cy = H - 100 + Math.sin(t + i) * 10;
    ctx.fillStyle = '#A0724A';
    ctx.fillRect(cx-18, cy-18, 36, 36);
    ctx.strokeStyle = '#6B4226';
    ctx.lineWidth = 2;
    ctx.strokeRect(cx-14, cy-14, 28, 28);
  }
}

function drawOverlay(title, subtitle) {
  ctx.fillStyle = 'rgba(0,0,0,0.7)';
  ctx.fillRect(0, 0, W, H);
  ctx.textAlign = 'center';
  ctx.fillStyle = '#fff';
  ctx.font = 'bold 42px monospace';
  ctx.fillText(title, W/2, H/2 - 20);
  ctx.font = '18px monospace';
  ctx.fillStyle = '#aabbcc';
  ctx.fillText(subtitle, W/2, H/2 + 30);
}

  update();
  draw();
  requestAnimationFrame(loop);
