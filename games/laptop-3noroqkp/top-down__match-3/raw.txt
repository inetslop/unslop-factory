(function(){
const SIZE = 8;
const COLORS = ['#ff6b6b','#feca57','#48dbfb','#1dd1a1','#f368e0','#ff9f43'];
const GLOW = ['rgba(255,107,107,','rgba(254,202,87,','rgba(72,219,251,','rgba(29,209,161,','rgba(243,104,224,','rgba(255,159,67,'];
let canvas, ctx, cellSize, boardW;
let grid = [], score = 0, combo = 0, animating = false;
let selected = null, dragging = false, dragStart = null;
let particles = [], fallingGems = [];
let audioCtx = null;

function initAudio(){
  if(!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
}

function playSound(type){
  if(!audioCtx) return;
  const osc = audioCtx.createOscillator();
  const gain = audioCtx.createGain();
  osc.connect(gain);
  gain.connect(audioCtx.destination);
  if(type === 'swap'){
    osc.type = 'sine'; osc.frequency.setValueAtTime(400, audioCtx.currentTime);
    osc.frequency.linearRampToValueAtTime(600, audioCtx.currentTime + 0.1);
    gain.gain.setValueAtTime(0.15, audioCtx.currentTime);
    gain.gain.linearRampToValueAtTime(0, audioCtx.currentTime + 0.1);
    osc.start(); osc.stop(audioCtx.currentTime + 0.1);
  } else if(type === 'match'){
    osc.type = 'sine'; osc.frequency.setValueAtTime(500 + combo * 100, audioCtx.currentTime);
    osc.frequency.linearRampToValueAtTime(800 + combo * 100, audioCtx.currentTime + 0.15);
    gain.gain.setValueAtTime(0.2, audioCtx.currentTime);
    gain.gain.linearRampToValueAtTime(0, audioCtx.currentTime + 0.2);
    osc.start(); osc.stop(audioCtx.currentTime + 0.2);
  } else if(type === 'cascade'){
    osc.type = 'triangle'; osc.frequency.setValueAtTime(300, audioCtx.currentTime);
    gain.gain.setValueAtTime(0.1, audioCtx.currentTime);
    gain.gain.linearRampToValueAtTime(0, audioCtx.currentTime + 0.08);
    osc.start(); osc.stop(audioCtx.currentTime + 0.08);
  } else if(type === 'invalid'){
    osc.type = 'square'; osc.frequency.setValueAtTime(200, audioCtx.currentTime);
    gain.gain.setValueAtTime(0.1, audioCtx.currentTime);
    gain.gain.linearRampToValueAtTime(0, audioCtx.currentTime + 0.15);
    osc.start(); osc.stop(audioCtx.currentTime + 0.15);
  }
}

function randomColor(){ return Math.floor(Math.random() * COLORS.length); }

function initGrid(){
  grid = [];
  for(let r = 0; r < SIZE; r++){
    grid[r] = [];
    for(let c = 0; c < SIZE; c++){
      grid[r][c] = randomColor();
    }
  }
  // Remove initial matches
  let hasMatch = true;
  while(hasMatch){
    hasMatch = false;
    for(let r = 0; r < SIZE; r++){
      for(let c = 0; c < SIZE; c++){
        if(c >= 2 && grid[r][c] === grid[r][c-1] && grid[r][c] === grid[r][c-2]){
          grid[r][c] = randomColor(); hasMatch = true;
        }
        if(r >= 2 && grid[r][c] === grid[r-1][c] && grid[r][c] === grid[r-2][c]){
          grid[r][c] = randomColor(); hasMatch = true;
        }
      }
    }
  }
}

function findMatches(){
  let matches = new Set();
  // Horizontal
  for(let r = 0; r < SIZE; r++){
    for(let c = 0; c < SIZE - 2; c++){
      if(grid[r][c] !== null && grid[r][c] === grid[r][c+1] && grid[r][c] === grid[r][c+2]){
        matches.add(r + ',' + c);
        let cc = c;
        while(cc < SIZE - 1 && grid[r][cc] === grid[r][cc+1]) cc++;
        for(let i = c; i <= cc; i++) matches.add(r + ',' + i);
      }
    }
  }
  // Vertical
  for(let c = 0; c < SIZE; c++){
    for(let r = 0; r < SIZE - 2; r++){
      if(grid[r][c] !== null && grid[r][c] === grid[r+1][c] && grid[r][c] === grid[r+2][c]){
        matches.add(r + ',' + c);
        let rr = r;
        while(rr < SIZE - 1 && grid[rr][c] === grid[rr+1][c]) rr++;
        for(let i = r; i <= rr; i++) matches.add(i + ',' + c);
      }
    }
  }
  return Array.from(matches).map(s => s.split(',').map(Number));
}

function applyGravity(){
  let moved = false;
  for(let c = 0; c < SIZE; c++){
    for(let r = SIZE - 1; r >= 0; r--){
      if(grid[r][c] === null){
        for(let rr = r - 1; rr >= 0; rr--){
          if(grid[rr][c] !== null){
            grid[r][c] = grid[rr][c];
            grid[rr][c] = null;
            moved = true;
            break;
          }
        }
      }
    }
  }
  // Fill top
  for(let c = 0; c < SIZE; c++){
    for(let r = 0; r < SIZE; r++){
      if(grid[r][c] === null){
        grid[r][c] = randomColor();
        moved = true;
      }
    }
  }
  return moved;
}

function processMatches(){
  let matches = findMatches();
  if(matches.length > 0){
    combo++;
    score += matches.length * 10 * combo;
    updateHUD();
    playSound('match');
    // Spawn particles
    matches.forEach(([r,c]) => {
      spawnParticles(r, c, grid[r][c]);
    });
    // Clear
    matches.forEach(([r,c]) => { grid[r][c] = null; });
    return true;
  } else {
    combo = 0;
    return false;
  }
}

function spawnParticles(r, c, colorIdx){
  const cx = c * cellSize + cellSize/2;
  const cy = r * cellSize + cellSize/2;
  for(let i = 0; i < 6; i++){
    const angle = Math.random() * Math.PI * 2;
    const speed = 2 + Math.random() * 3;
    particles.push({ x: cx, y: cy, vx: Math.cos(angle)*speed, vy: Math.sin(angle)*speed, life: 1, color: COLORS[colorIdx] || '#fff' });
  }
}

function updateParticles(){
  for(let i = particles.length - 1; i >= 0; i--){
    const p = particles[i];
    p.x += p.vx;
    p.y += p.vy;
    p.vy += 0.3;
    p.life -= 0.03;
    if(p.life <= 0) particles.splice(i, 1);
  }
}

function drawGem(x, y, colorIdx, scale, alpha){
  const s = cellSize * (scale || 1) * 0.85;
  const cx = x + cellSize/2;
  const cy = y + cellSize/2;
  ctx.save();
  ctx.globalAlpha = alpha !== undefined ? alpha : 1;
  
  // Shadow for top-down depth
  ctx.fillStyle = 'rgba(0,0,0,0.3)';
  ctx.beginPath();
  ctx.arc(cx, cy + 4, s/2, 0, Math.PI*2);
  ctx.fill();
  
  // Gem body - octagonal shape
  const r = s/2;
  ctx.beginPath();
  for(let i = 0; i < 8; i++){
    const angle = (i * Math.PI / 4) + Math.PI/8;
    const px = cx + Math.cos(angle) * r;
    const py = cy + Math.sin(angle) * r;
    if(i === 0) ctx.moveTo(px, py); else ctx.lineTo(px, py);
  }
  ctx.closePath();
  
  // Fill with gradient for 3D top-down look
  const grad = ctx.createRadialGradient(cx - r*0.2, cy - r*0.2, 0, cx, cy, r);
  grad.addColorStop(0, 'rgba(255,255,255,0.6)');
  grad.addColorStop(0.3, COLORS[colorIdx]);
  grad.addColorStop(1, GLOW[colorIdx] + '0.4)');
  ctx.fillStyle = grad;
  ctx.fill();
  
  // Inner highlight
  ctx.beginPath();
  for(let i = 0; i < 8; i++){
    const angle = (i * Math.PI / 4) + Math.PI/8;
    const px = cx + Math.cos(angle) * r * 0.5;
    const py = cy + Math.sin(angle) * r * 0.5;
    if(i === 0) ctx.moveTo(px, py); else ctx.lineTo(px, py);
  }
  ctx.closePath();
  ctx.fillStyle = 'rgba(255,255,255,0.15)';
  ctx.fill();
  
  // Top light reflection
  const hlGrad = ctx.createRadialGradient(cx - r*0.3, cy - r*0.4, 0, cx, cy, r*0.6);
  hlGrad.addColorStop(0, 'rgba(255,255,255,0.5)');
  hlGrad.addColorStop(1, 'rgba(255,255,255,0)');
  ctx.fillStyle = hlGrad;
  ctx.beginPath();
  ctx.arc(cx - r*0.1, cy - r*0.1, r*0.4, 0, Math.PI*2);
  ctx.fill();
  
  ctx.restore();
}

function drawBoard(){
  // Board background
  ctx.fillStyle = '#0f3460';
  ctx.fillRect(0, 0, boardW, boardW);
  
  // Grid lines - subtle top-down grid
  ctx.strokeStyle = 'rgba(255,255,255,0.08)';
  ctx.lineWidth = 1;
  for(let i = 0; i <= SIZE; i++){
    ctx.beginPath(); ctx.moveTo(i * cellSize, 0); ctx.lineTo(i * cellSize, boardW); ctx.stroke();
    ctx.beginPath(); ctx.moveTo(0, i * cellSize); ctx.lineTo(boardW, i * cellSize); ctx.stroke();
  }
  
  // Draw gems
  for(let r = 0; r < SIZE; r++){
    for(let c = 0; c < SIZE; c++){
      if(grid[r][c] !== null){
        let scale = 1, alpha = 1;
        if(selected && selected.r === r && selected.c === c){
          scale = 1.1;
        }
        drawGem(c * cellSize, r * cellSize, grid[r][c], scale, alpha);
      }
    }
  }
  
  // Selection highlight
  if(selected){
    ctx.strokeStyle = '#fff';
    ctx.lineWidth = 3;
    ctx.shadowColor = 'rgba(255,255,255,0.8)';
    ctx.shadowBlur = 10;
    ctx.strokeRect(selected.c * cellSize + 3, selected.r * cellSize + 3, cellSize - 6, cellSize - 6);
    ctx.shadowBlur = 0;
  }
  
  // Particles
  particles.forEach(p => {
    ctx.globalAlpha = p.life;
    ctx.fillStyle = p.color;
    ctx.beginPath();
    ctx.arc(p.x, p.y, 4 * p.life, 0, Math.PI*2);
    ctx.fill();
  });
  ctx.globalAlpha = 1;
}

function gameLoop(){
  if(!running) return;
  updateParticles();
  drawBoard();
  requestAnimationFrame(gameLoop);
}

function getCellFromPos(x, y){
  const rect = canvas.getBoundingClientRect();
  const px = x - rect.left;
  const py = y - rect.top;
  const c = Math.floor(px / cellSize);
  const r = Math.floor(py / cellSize);
  if(r >= 0 && r < SIZE && c >= 0 && c < SIZE) return {r, c};
  return null;
}

function isValidSwap(r1, c1, r2, c2){
  return (Math.abs(r1-r2) + Math.abs(c1-c2)) === 1;
}

function wouldMatch(r1, c1, r2, c2){
  // Swap temporarily and check
  let tmp = grid[r1][c1];
  grid[r1][c1] = grid[r2][c2];
  grid[r2][c2] = tmp;
  const matches = findMatches();
  grid[r1][c1] = tmp;
  grid[r2][c2] = grid[r1][c1]; // undo
  return matches.length > 0;
}

function doSwap(r1, c1, r2, c2){
  if(!isValidSwap(r1,c1,r2,c2)) return;
  if(grid[r1][c1] === null || grid[r2][c2] === null) return;
  
  playSound('swap');
  
  // Animate swap - simple instant swap for now with particles
  const tmp = grid[r1][c1];
  grid[r1][c1] = grid[r2][c2];
  grid[r2][c2] = tmp;
  
  if(!wouldMatch(r1, c1, r2, c2)){
    // Undo - actually we already swapped, let's swap back
    const t = grid[r1][c1];
    grid[r1][c1] = grid[r2][c2];
    grid[r2][c2] = t;
    playSound('invalid');
  } else {
    resolveBoard();
  }
}

function resolveBoard(){
  animating = true;
  combo = 0;
  
  const step = () => {
    if(processMatches()){
      setTimeout(() => {
        applyGravity();
        playSound('cascade');
        setTimeout(step, 200);
      }, 300);
    } else {
      animating = false;
      updateHUD();
    }
  };
  step();
}

function handleInput(x, y){
  if(animating) return;
  const cell = getCellFromPos(x, y);
  if(!cell) { selected = null; drawBoard(); return; }
  
  if(selected === null){
    selected = cell;
    drawBoard();
  } else {
    if(selected.r === cell.r && selected.c === cell.c){
      selected = null;
      drawBoard();
    } else if(isValidSwap(selected.r, selected.c, cell.r, cell.c)){
      doSwap(selected.r, selected.c, cell.r, cell.c);
      selected = null;
    } else {
      selected = cell;
      drawBoard();
    }
  }
}

function setupInput(){
  canvas.addEventListener('click', (e) => {
    if(!running || animating) return;
    handleInput(e.clientX, e.clientY);
  });
  
  canvas.addEventListener('touchstart', (e) => {
    e.preventDefault();
    if(!running || animating) return;
    const t = e.touches[0];
    handleInput(t.clientX, t.clientY);
  }, {passive: false});
  
  // Keyboard support - arrow keys + space to select/swap
  let keySelected = null;
  window.addEventListener('keydown', (e) => {
    if(!running || animating) return;
    if(e.key === 'Escape') { selected = null; drawBoard(); return; }
    
    const cell = getCellFromPos(
      canvas.getBoundingClientRect().left + 4*cellSize,
      canvas.getBoundingClientRect().top + 4*cellSize
    );
    
    // Use a cursor-based approach with arrow keys
    if(!keySelected) keySelected = {r: 0, c: 0};
    
    switch(e.key){
      case 'ArrowUp': keySelected.r = Math.max(0, keySelected.r - 1); break;
      case 'ArrowDown': keySelected.r = Math.min(SIZE-1, keySelected.r + 1); break;
      case 'ArrowLeft': keySelected.c = Math.max(0, keySelected.c - 1); break;
      case 'ArrowRight': keySelected.c = Math.min(SIZE-1, keySelected.c + 1); break;
      case ' ':
      case 'Enter':
        e.preventDefault();
        if(selected === null){
          selected = {...keySelected};
        } else {
          if(isValidSwap(selected.r, selected.c, keySelected.r, keySelected.c)){
            doSwap(selected.r, selected.c, keySelected.r, keySelected.c);
            selected = null;
          } else {
            selected = {...keySelected};
          }
        }
        break;
      case 'r': case 'R': restart(); return;
    }
    if(keySelected){
      // Draw cursor
      ctx.save();
      ctx.strokeStyle = 'rgba(255,255,255,0.6)';
      ctx.lineWidth = 2;
      ctx.setLineDash([4,4]);
      ctx.strokeRect(keySelected.c * cellSize + 2, keySelected.r * cellSize + 2, cellSize - 4, cellSize - 4);
      ctx.restore();
    }
    if(running) drawBoard();
  });
}

  });
