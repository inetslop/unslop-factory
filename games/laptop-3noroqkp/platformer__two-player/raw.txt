// Game states

// Audio
let audioCtx = null;
function initAudio() {
    if (!audioCtx) {
        audioCtx = new (window.AudioContext || window.webkitAudioContext)();
    }
}

function playSound(freq, type, duration, vol) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.type = type;
    osc.frequency.value = freq;
    gain.gain.setValueAtTime(vol || 0.1, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + duration);
    osc.start();
    osc.stop(audioCtx.currentTime + duration);
}

function playJump() { playSound(300, 'square', 0.15, 0.08); }
function playCollect() { playSound(800, 'sine', 0.2, 0.1); playSound(1200, 'sine', 0.3, 0.08); }
function playHurt() { playSound(150, 'sawtooth', 0.3, 0.15); }
function playWin() { playSound(523, 'sine', 0.3, 0.1); setTimeout(()=>playSound(659,'sine',0.3,0.1),150); setTimeout(()=>playSound(784,'sine',0.4,0.1),300); }

// Players
function createPlayer(x, y, color, name) {
    return {
        x, y, w: 30, h: 40,
        vx: 0, vy: 0,
        onGround: false,
        color, name,
        lives: 3,
        invincible: 0,
        facing: 1,
        squash: 0
    };
}

let player1, player2;

function resetGame() {
    buildLevel();
    const groundY = levelHeight - TILE;
    player1 = createPlayer(80, groundY - 40, '#ff6b6b', 'P1');
    player2 = createPlayer(140, groundY - 40, '#4dabf7', 'P2');
    score = 0;

// Input
let keys = {};
window.addEventListener('keydown', e => {
    keys[e.code] = true;
    if (gameState === STATE.MENU) {
        initAudio();
        startGame();
    } else if ((gameState === STATE.WIN || gameState === STATE.LOSE) && e.code === 'KeyR') {
        startGame();
    }
    e.preventDefault();
});
window.addEventListener('keyup', e => { keys[e.code] = false; });

function startGame() {
    resetGame();
    gameState = STATE.PLAYING;

// Physics constants
const GRAVITY = 0.6;
const JUMP_FORCE = -12;
const MOVE_SPEED = 5;
const FRICTION = 0.85;
const MAX_FALL = 14;

function updatePlayer(p, leftKey, rightKey, jumpKey) {
    // Movement
    if (keys[leftKey]) { p.vx -= 0.8; p.facing = -1; }
    if (keys[rightKey]) { p.vx += 0.8; p.facing = 1; }
    
    p.vx *= FRICTION;
    if (Math.abs(p.vx) > MOVE_SPEED) p.vx = Math.sign(p.vx) * MOVE_SPEED;
    
    // Jump
    if (keys[jumpKey] && p.onGround) {
        p.vy = JUMP_FORCE;
        p.onGround = false;
        p.squash = 5;
        playJump();
    }
    
    // Gravity
    p.vy += GRAVITY;
    if (p.vy > MAX_FALL) p.vy = MAX_FALL;
    
    // Move X
    p.x += p.vx;
    resolveCollisionX(p);
    
    // Move Y
    p.y += p.vy;
    p.onGround = false;
    resolveCollisionY(p);
    
    // Clamp to level
    if (p.x < 0) { p.x = 0; p.vx = 0; }
    if (p.x + p.w > levelWidth) { p.x = levelWidth - p.w; p.vx = 0; }
    
    // Fall off level
    if (p.y > levelHeight + 50) {
        hurtPlayer(p);
    }
    
    // Invincibility timer
    if (p.invincible > 0) p.invincible--;
    if (p.squash > 0) p.squash--;
}

function resolveCollisionX(p) {
    for (let plat of platforms) {
        if (p.x + p.w > plat.x && p.x < plat.x + plat.w &&
            p.y + p.h > plat.y && p.y < plat.y + plat.h) {
            if (p.vx > 0) p.x = plat.x - p.w;
            else if (p.vx < 0) p.x = plat.x + plat.w;
            p.vx = 0;
        }
    }
}

function resolveCollisionY(p) {
    for (let plat of platforms) {
        if (p.x + p.w > plat.x && p.x < plat.x + plat.w &&
            p.y + p.h > plat.y && p.y < plat.y + plat.h) {
            if (p.vy > 0) {
                p.y = plat.y - p.h;
                p.vy = 0;
                p.onGround = true;
            } else if (p.vy < 0) {
                p.y = plat.y + plat.h;
                p.vy = 0;
            }
        }
    }
}

function checkSpikes(p) {
    if (p.invincible > 0) return;
    for (let spike of spikes) {
        if (p.x + p.w > spike.x && p.x < spike.x + spike.w &&
            p.y + p.h > spike.y && p.y < spike.y + spike.h) {
            hurtPlayer(p);
            break;
        }
    }
}

function checkGems(p) {
    for (let gem of gems) {
        if (!gem.collected &&
            p.x + p.w > gem.x && p.x < gem.x + gem.w &&
            p.y + p.h > gem.y && p.y < gem.y + gem.h) {
            gem.collected = true;
            gemsCollected++;
            score += 100;
            playCollect();
            spawnParticles(gem.x + gem.w/2, gem.y + gem.h/2, '#ffd700', 8);
        }
    }
}

function hurtPlayer(p) {
    if (p.invincible > 0) return;
    p.lives--;
    p.invincible = 90;
    shakeTime = 15;
    playHurt();
    spawnParticles(p.x + p.w/2, p.y + p.h/2, p.color, 6);
    
    // Reset position
    const groundY = levelHeight - TILE;
    p.x = Math.max(80, p.x - 200);
    p.y = groundY - p.h - 10;
    p.vx = 0;
    p.vy = 0;
    
    if (p.lives <= 0) {
        gameState = STATE.LOSE;
    }
}

function spawnParticles(x, y, color, count) {
    for (let i = 0; i < count; i++) {
        particles.push({
            x, y,
            vx: (Math.random() - 0.5) * 6,
            vy: (Math.random() - 0.5) * 6 - 2,
            life: 30 + Math.random() * 20,
            color,
            size: 3 + Math.random() * 4
        });
    }
}

function updateParticles() {
    for (let i = particles.length - 1; i >= 0; i--) {
        let pt = particles[i];
        pt.x += pt.vx;
        pt.y += pt.vy;
        pt.vy += 0.2;
        pt.life--;
        if (pt.life <= 0) particles.splice(i, 1);
    }
}

// Camera
let camX = 0, camY = 0;

function updateCamera() {
    let targetX = (player1.x + player2.x) / 2 - W / 2;
    let targetY = (player1.y + player2.y) / 2 - H / 2;
    
    camX += (targetX - camX) * 0.08;
    camY += (targetY - camY) * 0.08;
    
    camX = Math.max(0, Math.min(camX, levelWidth - W));
    camY = Math.max(-50, Math.min(camY, levelHeight - H));
}

// Rendering
function drawBackground() {
    const grad = ctx.createLinearGradient(0, 0, 0, H);
    grad.addColorStop(0, '#16213e');
    grad.addColorStop(0.5, '#1a1a2e');
    grad.addColorStop(1, '#0f3460');
    ctx.fillStyle = grad;
    ctx.fillRect(0, 0, W, H);
    
    // Stars
    ctx.fillStyle = 'rgba(255,255,255,0.3)';
    for (let i = 0; i < 50; i++) {
        let sx = ((i * 137 + 50) % levelWidth) - camX * 0.3;
        let sy = (i * 97 + 20) % 200;
        if (sx > -5 && sx < W) {
            ctx.beginPath();
            ctx.arc(sx, sy, 1.5, 0, Math.PI * 2);
            ctx.fill();
        }
    }
    
    // Distant mountains
    ctx.fillStyle = 'rgba(30, 60, 90, 0.4)';
    for (let i = 0; i < 8; i++) {
        let mx = (i * 400 + 100) - camX * 0.2;

        ctx.beginPath();
        ctx.moveTo(mx, H);
        ctx.lineTo(mx + 200, H - mh);
        ctx.lineTo(mx + 400, H);
        ctx.fill();
    }
}

function drawPlatforms() {
    for (let plat of platforms) {
        let px = plat.x - camX;
        let py = plat.y - camY;
        
        if (px + plat.w < 0 || px > W) continue;
        
        if (plat.type === 'ground') {
            ctx.fillStyle = '#4a7c59';
            ctx.fillRect(px, py, plat.w, plat.h);
            ctx.fillStyle = '#2d6a4f';
            ctx.fillRect(px, py + 10, plat.w, plat.h - 10);
            
            // Grass top
            ctx.fillStyle = '#67b85e';
            ctx.fillRect(px, py, plat.w, 8);
        } else {
            ctx.fillStyle = '#8b6f47';
            ctx.fillRect(px, py, plat.w, plat.h);
            ctx.fillStyle = '#a0825a';
            ctx.fillRect(px, py, plat.w, 6);
            
            // Platform edges
            ctx.fillStyle = 'rgba(0,0,0,0.2)';
            ctx.fillRect(px, py + plat.h - 4, plat.w, 4);
        }
    }
}

function drawSpikes() {
    for (let spike of spikes) {
        let sx = spike.x - camX;
        let sy = spike.y - camY;
        
        if (sx + spike.w < 0 || sx > W) continue;
        
        ctx.fillStyle = '#c0392b';
        const sCount = Math.floor(spike.w / 10);
        for (let i = 0; i < sCount; i++) {
            let bx = sx + i * 10;
            ctx.beginPath();
            ctx.moveTo(bx, sy + spike.h);
            ctx.lineTo(bx + 5, sy);
            ctx.lineTo(bx + 10, sy + spike.h);
            ctx.fill();
        }
    }
}

function drawGems() {
    const time = Date.now() * 0.003;
    for (let gem of gems) {
        if (gem.collected) continue;
        
        let gx = gem.x - camX;
        let gy = gem.y - camY + Math.sin(time + gem.bob) * 5;
        
        if (gx + gem.w < 0 || gx > W) continue;
        
        ctx.save();
        ctx.translate(gx + gem.w/2, gy + gem.h/2);
        ctx.rotate(Math.PI / 4);
        
        // Glow
        ctx.shadowColor = '#ffd700';
        ctx.shadowBlur = 10;
        
        ctx.fillStyle = '#ffd700';
        ctx.fillRect(-8, -8, 16, 16);
        ctx.fillStyle = '#fff8dc';
        ctx.fillRect(-5, -5, 10, 10);
        
        ctx.restore();
    }
}

function drawPlayer(p) {
    let px = p.x - camX;
    let py = p.y - camY;
    
    if (p.invincible > 0 && Math.floor(p.invincible / 5) % 2 === 0) return;
    
    ctx.save();
    ctx.translate(px + p.w/2, py + p.h/2);
    
    // Squash/stretch
    let sx = 1, sy = 1;
    if (p.squash > 0) {
        sx = 1.2;
        sy = 0.8;
    }
    ctx.scale(sx, sy);
    
    // Body
    ctx.fillStyle = p.color;
    ctx.fillRect(-p.w/2, -p.h/2, p.w, p.h);
    
    // Face
    let eyeX = p.facing * 5;
    ctx.fillStyle = '#fff';
    ctx.fillRect(eyeX - 4, -10, 8, 8);
    ctx.fillRect(eyeX + 4, -10, 8, 8);
    
    ctx.fillStyle = '#222';
    ctx.fillRect(eyeX - 2 + p.facing * 2, -8, 4, 4);
    ctx.fillRect(eyeX + 6 + p.facing * 2, -8, 4, 4);
    
    // Mouth
    ctx.fillStyle = '#222';
    if (p.lives > 1) {
        ctx.fillRect(-3, 5, 6, 2);
    } else {
        ctx.beginPath();
        ctx.arc(0, 8, 4, 0, Math.PI);
        ctx.fill();
    }
    
    // Label
    ctx.fillStyle = '#fff';
    ctx.font = 'bold 10px monospace';
    ctx.textAlign = 'center';
    ctx.fillText(p.name, 0, -p.h/2 - 5);
    
    ctx.restore();
}

function drawParticles() {
    for (let pt of particles) {
        let alpha = pt.life / 40;
        ctx.fillStyle = pt.color + Math.floor(alpha * 255).toString(16).padStart(2, '0');
        ctx.fillRect(pt.x - camX, pt.y - camY, pt.size, pt.size);
    }
}

function drawEndScreen(won) {

    ctx.fillRect(0, 0, W, H);
    
    ctx.textAlign = 'center';
    
    if (won) {
        ctx.fillStyle = '#ffd700';

        ctx.font = '20px monospace';

        ctx.fillText(`Final Score: ${score}`, W/2, H/3 + 85);

        ctx.fillStyle = '#c0392b';
        ctx.font = 'bold 42px monospace';
        ctx.fillText('GAME OVER', W/2, H/3);
        ctx.fillStyle = '#fff';

        ctx.fillText(`Score: ${score}`, W/2, H/3 + 85);

    ctx.font = 'bold 18px monospace';

// Main loop
function update() {
    if (gameState !== STATE.PLAYING) return;
    
    updatePlayer(player1, 'KeyA', 'KeyD', 'KeyW');
    updatePlayer(player2, 'ArrowLeft', 'ArrowRight', 'ArrowUp');
    
    checkSpikes(player1);
    checkSpikes(player2);
    checkGems(player1);
    checkGems(player2);
    
    // Check win
    if (gemsCollected >= totalGems) {
        gameState = STATE.WIN;
        playWin();
    }
    
    updateParticles();
    updateCamera();
    
    if (shakeTime > 0) shakeTime--;
}

function render() {
    ctx.save();
    
    // Screen shake
    if (shakeTime > 0) {
        let sx = (Math.random() - 0.5) * shakeTime;
        let sy = (Math.random() - 0.5) * shakeTime;
        ctx.translate(sx, sy);
    }
    
    drawBackground();
    
    if (gameState === STATE.PLAYING || gameState === STATE.WIN || gameState === STATE.LOSE) {
        drawPlatforms();
        drawSpikes();
        drawGems();
        drawPlayer(player1);
        drawPlayer(player2);
        drawParticles();
        drawHUD();
    }
    
    ctx.restore();
    
    if (gameState === STATE.MENU) drawMenu();
    if (gameState === STATE.WIN) drawEndScreen(true);
    if (gameState === STATE.LOSE) drawEndScreen(false);
}

function gameLoop() {
    update();
    render();
    requestAnimationFrame(gameLoop);
}

resetGame();
gameLoop();
