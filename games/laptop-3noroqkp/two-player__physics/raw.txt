function playSound(freq, type, duration, volume) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = type;
    osc.frequency.value = freq;
    gain.gain.setValueAtTime(volume || 0.3, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + duration);
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + duration);
}

// Game State
const GRAVITY = 1500;
const FRICTION = 0.98;
const RESTITUTION = 0.6;
const SLAM_COOLDOWN = 800;
const SLAM_FORCE = 800;
const MOVE_ACCEL = 2000;
const JUMP_FORCE = -600;
const MAX_SPEED = 1200;

class Body {
    constructor(x, y, r, mass) {
        this.x = x;
        this.y = y;
        this.r = r;
        this.mass = mass;
        this.vx = 0;
        this.vy = 0;
        this.onGround = false;
    }

    update(dt) {
        this.vy += GRAVITY * dt;
        this.x += this.vx * dt;
        this.y += this.vy * dt;

        // Wall collisions
        if (this.x - this.r < 0) { this.x = this.r; this.vx = Math.abs(this.vx) * RESTITUTION; }
        if (this.x + this.r > W) { this.x = W - this.r; this.vx = -Math.abs(this.vx) * RESTITUTION; }
        if (this.y - this.r < 0) { this.y = this.r; this.vy = Math.abs(this.vy) * RESTITUTION; }

        // Floor collision (platform)
        const floorY = H * 0.75;
        if (this.y + this.r > floorY) {
            this.y = floorY - this.r;
            this.vy = Math.min(0, this.vy);
            if (Math.abs(this.vy) < 100) this.vy = 0;
            this.vx *= FRICTION;
            this.onGround = true;
        } else {
            this.onGround = false;
        }

        // Speed cap
        const speed = Math.sqrt(this.vx * this.vx + this.vy * this.vy);
        if (speed > MAX_SPEED) {
            this.vx = (this.vx / speed) * MAX_SPEED;
            this.vy = (this.vy / speed) * MAX_SPEED;
        }
    }

    applyForce(fx, fy) {
        this.vx += fx / this.mass;
        this.vy += fy / this.mass;
    }
}

class Player extends Body {
    constructor(x, y, r, mass, color, id) {
        super(x, y, r, mass);
        this.color = color;
        this.id = id;
        this.slamTimer = 0;
        this.alive = true;
        this.fallTimer = 0;
    }

    update(dt) {
        // Slam cooldown
        if (this.slamTimer > 0) this.slamTimer -= dt * 1000;

        super.update(dt);

        // Check if fallen off platform
        const floorY = H * 0.75;
        if (this.y - this.r > floorY + 20) {
            this.fallTimer += dt;
            if (this.fallTimer > 0.5 && this.alive) {
                this.alive = false;
                playFall();
                spawnParticles(this.x, this.y, this.color, 20);
            }
        } else {
            this.fallTimer = 0;
        }
    }

    slam() {
        if (this.slamTimer > 0) return;
        this.slamTimer = SLAM_COOLDOWN;
        playSlam();
        shockwaves.push({ x: this.x, y: this.y, r: 10, maxR: 200, color: this.color });

        // Apply radial force to all bodies
        for (let b of allBodies) {
            if (b === this) continue;
            const dx = b.x - this.x;
            const dy = b.y - this.y;
            const dist = Math.sqrt(dx * dx + dy * dy);
            if (dist < 200 && dist > 0) {
                const force = SLAM_FORCE * (1 - dist / 200);
                b.applyForce((dx / dist) * force, (dy / dist) * force - 50);
            }
        }
        // Self lift
        this.vy -= 200;
    }

    draw(ctx) {
        if (!this.alive) return;
        ctx.save();
        ctx.beginPath();
        ctx.arc(this.x, this.y, this.r, 0, Math.PI * 2);
        const gradient = ctx.createRadialGradient(this.x - this.r * 0.3, this.y - this.r * 0.3, 0, this.x, this.y, this.r);
        gradient.addColorStop(0, this.color.light);
        gradient.addColorStop(1, this.color.dark);
        ctx.fillStyle = gradient;
        ctx.fill();
        ctx.strokeStyle = this.color.stroke;
        ctx.lineWidth = 3;
        ctx.stroke();

        // Face
        const eyeY = this.y - this.r * 0.2;
        const eyeR = this.r * 0.15;
        ctx.fillStyle = '#fff';
        ctx.beginPath();
        ctx.arc(this.x - this.r * 0.25, eyeY, eyeR, 0, Math.PI * 2);
        ctx.arc(this.x + this.r * 0.25, eyeY, eyeR, 0, Math.PI * 2);
        ctx.fill();

        // Pupils look toward center of action
        const px = (this.id === 0 ? -1 : 1) * 3;
        ctx.fillStyle = '#000';
        ctx.beginPath();
        ctx.arc(this.x - this.r * 0.25 + px, eyeY + 1, eyeR * 0.6, 0, Math.PI * 2);
        ctx.arc(this.x + this.r * 0.25 + px, eyeY + 1, eyeR * 0.6, 0, Math.PI * 2);
        ctx.fill();

        // Slam cooldown indicator
        if (this.slamTimer > 0) {
            const ratio = this.slamTimer / SLAM_COOLDOWN;
            ctx.beginPath();
            ctx.arc(this.x, this.y, this.r + 5, -Math.PI / 2, -Math.PI / 2 + Math.PI * 2 * (1 - ratio));
            ctx.strokeStyle = 'rgba(255,255,255,0.4)';
            ctx.lineWidth = 3;
            ctx.stroke();
        }

        ctx.restore();
    }
}

    const platformX = (W - platformW) / 2;

function spawnParticles(x, y, color, count) {
    for (let i = 0; i < count; i++) {
        const angle = Math.random() * Math.PI * 2;
        const speed = 100 + Math.random() * 300;
        particles.push({
            x, y,
            vx: Math.cos(angle) * speed,
            vy: Math.sin(angle) * speed - 100,
            r: 3 + Math.random() * 5,
            color,
            life: 1
        });
    }
}

function resolveCollision(a, b) {
    const dx = b.x - a.x;
    const dy = b.y - a.y;
    const dist = Math.sqrt(dx * dx + dy * dy);
    const minDist = a.r + b.r;

    if (dist < minDist && dist > 0) {
        // Separate
        const overlap = minDist - dist;
        const nx = dx / dist;
        const ny = dy / dist;
        const totalMass = a.mass + b.mass;
        a.x -= nx * overlap * (b.mass / totalMass);
        a.y -= ny * overlap * (b.mass / totalMass);
        b.x += nx * overlap * (a.mass / totalMass);
        b.y += ny * overlap * (a.mass / totalMass);

        // Impulse
        const rvx = a.vx - b.vx;
        const rvy = a.vy - b.vy;
        const velAlongNormal = rvx * nx + rvy * ny;

        if (velAlongNormal > 0) {
            const impulse = (1 + RESTITUTION) * velAlongNormal / totalMass;
            a.vx -= impulse * b.mass * nx;
            a.vy -= impulse * b.mass * ny;
            b.vx += impulse * a.mass * nx;
            b.vy += impulse * a.mass * ny;

            if (velAlongNormal > 200) {
                playHit();
                const cx = (a.x + b.x) / 2;
                const cy = (a.y + b.y) / 2;
                spawnParticles(cx, cy, '#fff', 3);
            }
        }
    }
}

function updatePhysics(dt) {
    for (let b of allBodies) {
        if (!b.alive && b instanceof Player) continue;
        b.update(dt);
    }

    // Collision detection - multiple iterations for stability
    for (let iter = 0; iter < 3; iter++) {
        for (let i = 0; i < allBodies.length; i++) {
            for (let j = i + 1; j < allBodies.length; j++) {
                const a = allBodies[i];
                const b = allBodies[j];
                if (!a.alive && a instanceof Player) continue;
                if (!b.alive && b instanceof Player) continue;
                resolveCollision(a, b);
            }
        }
    }

    // Remove fallen objects
    for (let i = objects.length - 1; i >= 0; i--) {
        if (objects[i].y > H + 100) {
            allBodies.splice(allBodies.indexOf(objects[i]), 1);
            objects.splice(i, 1);
        }
    }

    // Update particles
    for (let i = particles.length - 1; i >= 0; i--) {
        const p = particles[i];
        p.x += p.vx * dt;
        p.y += p.vy * dt;
        p.vy += GRAVITY * 0.5 * dt;
        p.life -= dt * 2;
        if (p.life <= 0) particles.splice(i, 1);
    }

    // Update shockwaves
    for (let i = shockwaves.length - 1; i >= 0; i--) {
        const s = shockwaves[i];
        s.r += 500 * dt;
        if (s.r > s.maxR) shockwaves.splice(i, 1);
    }

    // Check win condition
    if (!player1.alive || !player2.alive) {
        if (gameState === 'playing') {
            gameState = 'roundover';
            if (!player1.alive && !player2.alive) {
                winner = null; // draw
            } else if (!player1.alive) {
                winner = 1;
                scores[1]++;
            } else {
                winner = 0;
                scores[0]++;
            }
            playWin();
        }
    }
}

function handleInput(dt) {
    // Player 1: A, D, W, S
    if (player1.alive) {
        if (keys['KeyA']) player1.applyForce(-MOVE_ACCEL * dt * player1.mass, 0);
        if (keys['KeyD']) player1.applyForce(MOVE_ACCEL * dt * player1.mass, 0);
        if (keys['KeyW'] && player1.onGround) {
            player1.vy = JUMP_FORCE;
            playJump();
        }
        if (keys['KeyS']) player1.slam();
    }

    // Player 2: Arrows
    if (player2.alive) {
        if (keys['ArrowLeft']) player2.applyForce(-MOVE_ACCEL * dt * player2.mass, 0);
        if (keys['ArrowRight']) player2.applyForce(MOVE_ACCEL * dt * player2.mass, 0);
        if (keys['ArrowUp'] && player2.onGround) {
            player2.vy = JUMP_FORCE;
            playJump();
        }
        if (keys['ArrowDown']) player2.slam();
    }
}

}

function render() {
    if (gameState === 'start') {
        drawStartScreen();
        return;
    }

    drawBackground();
    drawObjects();
    drawShockwaves();
    player1.draw(ctx);
    player2.draw(ctx);
    drawParticles();
    drawHUD();

    if (gameState === 'roundover') {
        drawRoundOver();
    } else if (gameState === 'gameover') {
        drawGameOver();
    }
}

function gameLoop(timestamp) {
    const dt = Math.min((timestamp - lastTime) / 1000, 0.05);
    lastTime = timestamp;
    time += dt;

    if (gameState === 'playing') {
        handleInput(dt);
        updatePhysics(dt);
    } else if (gameState === 'roundover' || gameState === 'gameover') {
        // Still simulate physics for visual effect
        for (let b of allBodies) {
            if (b instanceof Player && !b.alive) continue;
            b.update(dt);
        }
        for (let i = particles.length - 1; i >= 0; i--) {
            const p = particles[i];
            p.x += p.vx * dt;
            p.y += p.vy * dt;
            p.vy += GRAVITY * 0.5 * dt;
            p.life -= dt * 2;
            if (p.life <= 0) particles.splice(i, 1);
        }
    }

    render();
    requestAnimationFrame(gameLoop);
}

function startGame() {
    initRound();
    gameState = 'playing';
}

window.addEventListener('keydown', (e) => {
    keys[e.code] = true;

    if (gameState === 'start' && e.code === 'Space') {
        initAudio();
        startGame();
    } else if (gameState === 'roundover' && e.code === 'Space') {
        if (scores[0] >= 3 || scores[1] >= 3) {
            gameState = 'gameover';
        } else {
            initRound();
            gameState = 'playing';
        }
    } else if ((gameState === 'roundover' || gameState === 'gameover') && e.code === 'KeyR') {
        scores = [0, 0];
        startGame();
    }

    // Prevent scrolling with arrow keys
    if (['Space', 'ArrowUp', 'ArrowDown', 'ArrowLeft', 'ArrowRight'].includes(e.code)) {
        e.preventDefault();
    }
});

window.addEventListener('keyup', (e) => {
    keys[e.code] = false;
});

// Initialize and start loop
initRound();
lastTime = performance.now();
requestAnimationFrame(gameLoop);
