(function() {
const canvas = document.getElementById('game');
const ctx = canvas.getContext('2d');

let W, H;
function resize() {
W = window.innerWidth;
H = window.innerHeight;
canvas.width = W;
canvas.height = H;

// Audio
let audioCtx = null;
function initAudio() {
if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
}
function playSound(type) {
if (!audioCtx) return;
const osc = audioCtx.createOscillator();
const gain = audioCtx.createGain();
osc.connect(gain);
gain.connect(audioCtx.destination);
switch(type) {
case 'bounce':
osc.type = 'square';
osc.frequency.setValueAtTime(220, audioCtx.currentTime);
osc.frequency.exponentialRampToValueAtTime(110, audioCtx.currentTime + 0.1);
gain.gain.setValueAtTime(0.15, audioCtx.currentTime);
gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.1);
osc.start(); osc.stop(audioCtx.currentTime + 0.1); break;
case 'destroy':
osc.type = 'sawtooth';
osc.frequency.setValueAtTime(440, audioCtx.currentTime);
osc.frequency.exponentialRampToValueAtTime(880, audioCtx.currentTime + 0.15);
gain.gain.setValueAtTime(0.2, audioCtx.currentTime);
gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.15);
osc.start(); osc.stop(audioCtx.currentTime + 0.15); break;
case 'shoot':
osc.type = 'square';
osc.frequency.setValueAtTime(600, audioCtx.currentTime);
osc.frequency.exponentialRampToValueAtTime(200, audioCtx.currentTime + 0.08);
gain.gain.setValueAtTime(0.1, audioCtx.currentTime);
gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.08);
osc.start(); osc.stop(audioCtx.currentTime + 0.08); break;
case 'hit':
osc.type = 'sawtooth';
osc.frequency.setValueAtTime(100, audioCtx.currentTime);
gain.gain.setValueAtTime(0.25, audioCtx.currentTime);
gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.3);
osc.start(); osc.stop(audioCtx.currentTime + 0.3); break;
case 'win':
osc.type = 'sine';
osc.frequency.setValueAtTime(523, audioCtx.currentTime);
gain.gain.setValueAtTime(0.2, audioCtx.currentTime);
gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 1);
osc.start(); osc.stop(audioCtx.currentTime + 1); break;
}
}

// Game state
let gameState = 'start'; // start, playing, gameover, win
let score = 0;
let lives = 3;
let highScore = 0;
try { highScore = parseInt(localStorage.getItem('siBreakoutHigh') || '0'); } catch(e) {}

// Stars
let stars = [];
function initStars() {
stars = [];
for (let i = 0; i < 120; i++) {
stars.push({ x: Math.random() * W, y: Math.random() * H, s: Math.random() * 2 + 0.5, speed: Math.random() * 0.3 + 0.1 });
}
}
initStars();

// Paddle
let paddle = { w: 100, h: 16, x: 0, y: 0 };
function resetPaddle() {
paddle.w = Math.min(120, W * 0.15);
paddle.h = 16;
paddle.x = (W - paddle.w) / 2;
paddle.y = H - 40;
}

// Ball
let ball = { x: 0, y: 0, r: 8, dx: 0, dy: 0, speed: 6, active: false };
function resetBall() {
ball.x = paddle.x + paddle.w / 2;
ball.y = paddle.y - ball.r - 4;
ball.dx = 0;
ball.dy = 0;
ball.speed = Math.max(5, H * 0.007);
ball.active = false;
}

// Aliens (grid)
let aliens = [];
let alienSpeed = 1;
let alienDir = 1;
let alienTimer = 0;
let shootTimer = 0;

function initAliens() {
aliens = [];
const cols = Math.min(8, Math.floor(W / 70));
const rows = 5;
const cellW = Math.min(60, (W - 100) / cols);
const cellH = 35;
const startX = (W - cols * cellW) / 2;
const startY = 80;
for (let r = 0; r < rows; r++) {
for (let c = 0; c < cols; c++) {
aliens.push({
x: startX + c * cellW + cellW / 2,
y: startY + r * cellH + cellH / 2,
w: cellW - 10,
h: cellH - 8,
row: r,
col: c,
hp: 1,
alive: true,
type: r === 0 ? 'boss' : (r < 3 ? 'grunt' : 'elite'),
points: (rows - r) * 10
});
}
}
alienSpeed = Math.max(0.5, W / 800);
alienDir = 1;
alienTimer = 0;
shootTimer = 60;
}

// Bullets (enemy shots)
let bullets = [];

// Particles
let particles = [];
function spawnParticles(x, y, color, count) {
for (let i = 0; i < count; i++) {
const angle = Math.random() * Math.PI * 2;
const speed = Math.random() * 4 + 1;
particles.push({ x, y, dx: Math.cos(angle) * speed, dy: Math.sin(angle) * speed, life: 30, color });
}
}

// Input
let keys = {};
let mouseX = W / 2;
window.addEventListener('keydown', e => {
keys[e.code] = true;
if (gameState === 'start' && (e.code === 'Space' || e.code === 'Enter')) startGame();
else if ((gameState === 'gameover' || gameState === 'win') && (e.code === 'KeyR' || e.code === 'Enter')) startGame();
});
window.addEventListener('keyup', e => { keys[e.code] = false; });
canvas.addEventListener('mousemove', e => { mouseX = e.clientX; });
canvas.addEventListener('click', () => {
initAudio();
if (gameState === 'start') startGame();
else if (gameState === 'gameover' || gameState === 'win') startGame();
else if (gameState === 'playing' && !ball.active) launchBall();
});

function startGame() {
score = 0;
lives = 3;
resetPaddle();
initAliens();
bullets = [];
particles = [];
gameState = 'playing';
}

function launchBall() {
ball.active = true;
const angle = -Math.PI / 2 + (Math.random() - 0.5) * 0.5;
ball.dx = Math.cos(angle) * ball.speed;
ball.dy = Math.sin(angle) * ball.speed;
}

function saveHighScore() {
try { localStorage.setItem('siBreakoutHigh', highScore); } catch(e) {}
}

// Update
let frameCount = 0;
function update() {
frameCount++;

// Stars
for (let s of stars) {
s.y += s.speed;
if (s.y > H) { s.y = 0; s.x = Math.random() * W; }
}

if (gameState !== 'playing') return;

// Paddle control
if (keys['ArrowLeft'] || keys['KeyA']) paddle.x -= 8;
if (keys['ArrowRight'] || keys['KeyD']) paddle.x += 8;
if (!keys['ArrowLeft'] && !keys['ArrowRight'] && !keys['KeyA'] && !keys['KeyD']) {
const target = mouseX - paddle.w / 2;
paddle.x += (target - paddle.x) * 0.3;
}
paddle.x = Math.max(0, Math.min(W - paddle.w, paddle.x));

// Ball physics
if (ball.active) {
ball.x += ball.dx;
ball.y += ball.dy;

// Walls
if (ball.x - ball.r < 0) { ball.x = ball.r; ball.dx = -ball.dx; playSound('bounce'); }
if (ball.x + ball.r > W) { ball.x = W - ball.r; ball.dx = -ball.dx; playSound('bounce'); }
if (ball.y - ball.r < 0) { ball.y = ball.r; ball.dy = -ball.dy; playSound('bounce'); }

// Paddle bounce
if (ball.dy > 0 && ball.y + ball.r >= paddle.y && ball.y + ball.r <= paddle.y + paddle.h + 5 && ball.x >= paddle.x && ball.x <= paddle.x + paddle.w) {
const hitPos = (ball.x - paddle.x) / paddle.w;
const angle = -Math.PI / 2 + (hitPos - 0.5) * Math.PI * 0.7;
ball.dx = Math.cos(angle) * ball.speed;
ball.dy = Math.sin(angle) * ball.speed;
playSound('bounce');
}

// Ball lost
if (ball.y > H + 20) {
lives--;
playSound('hit');
if (lives <= 0) {
gameState = 'gameover';
if (score > highScore) { highScore = score; saveHighScore(); }
} else {
resetBall();
}
}

// Ball vs Aliens
for (let a of aliens) {
if (!a.alive) continue;
if (ball.x + ball.r > a.x - a.w/2 && ball.x - ball.r < a.x + a.w/2 &&
    ball.y + ball.r > a.y - a.h/2 && ball.y - ball.r < a.y + a.h/2) {
a.alive = false;
score += a.points;
playSound('destroy');
spawnParticles(a.x, a.y, getAlienColor(a.type), 12);
// Reflect ball
ball.dy = -ball.dy;
ball.dx *= (Math.random() > 0.5 ? 1 : -1);
}
}

// Check win
if (aliens.every(a => !a.alive)) {
gameState = 'win';
playSound('win');
if (score > highScore) { highScore = score; saveHighScore(); }
}
}

// Alien movement
alienTimer++;
const moveInterval = Math.max(10, 40 - frameCount * 0.001);
if (alienTimer >= moveInterval) {
alienTimer = 0;
for (let a of aliens) {
if (!a.alive) continue;
a.x += alienDir * alienSpeed;
}
// Check edges
let minEdge = W, maxEdge = 0;
for (let a of aliens) {
if (a.alive) {
minEdge = Math.min(minEdge, a.x - a.w/2);
maxEdge = Math.max(maxEdge, a.x + a.w/2);
}
}
if (minEdge < 5 || maxEdge > W - 5) alienDir *= -1;

// Aliens descend occasionally
if (frameCount % 180 === 0) {
for (let a of aliens) { if (a.alive) a.y += 3; }
}
}

// Enemy shooting
shootTimer--;
if (shootTimer <= 0) {
shootTimer = Math.max(40, 90 - Math.floor(score / 200));
const aliveAliens = aliens.filter(a => a.alive);
if (aliveAliens.length > 0) {
const shooter = aliveAliens[Math.floor(Math.random() * aliveAliens.length)];
bullets.push({ x: shooter.x, y: shooter.y + shooter.h/2, dy: 3 + Math.random() * 2, r: 4 });
playSound('shoot');
}
}

// Bullets update
for (let i = bullets.length - 1; i >= 0; i--) {
const b = bullets[i];
b.y += b.dy;
if (b.y > H) { bullets.splice(i, 1); continue; }
// Bullet hits paddle
if (ball.active === false || true) {
if (b.x - b.r < paddle.x + paddle.w && b.x + b.r > paddle.x && b.y + b.r > paddle.y && b.y - b.r < paddle.y + paddle.h) {
lives--;
playSound('hit');
spawnParticles(b.x, b.y, '#ff4444', 8);
bullets.splice(i, 1);
if (lives <= 0) {
gameState = 'gameover';
if (score > highScore) { highScore = score; saveHighScore(); }
} else {
resetBall();
}
continue;
}
}
}

// Particles
for (let i = particles.length - 1; i >= 0; i--) {
const p = particles[i];
p.x += p.dx;
p.y += p.dy;
p.life--;
if (p.life <= 0) particles.splice(i, 1);
}
}

function getAlienColor(type) {
switch(type) {
case 'boss': return '#ff4444';
case 'elite': return '#ff8800';
case 'grunt': return '#44ff44';
default: return '#ffffff';
}
}

// Draw
function drawAlien(a) {
const type = a.type;
const color = getAlienColor(type);
const w = a.w, h = a.h;
ctx.save();
ctx.translate(a.x, a.y);

// Body
ctx.fillStyle = color;
ctx.shadowColor = color;
ctx.shadowBlur = 8;

if (type === 'boss') {
// Boss - wider, with horns
ctx.beginPath();
ctx.moveTo(-w/2, -h/4);
ctx.lineTo(-w/3, -h/2);
ctx.lineTo(-w/6, -h/4);
ctx.lineTo(w/6, -h/4);
ctx.lineTo(w/3, -h/2);
ctx.lineTo(w/2, -h/4);
ctx.lineTo(w/2, h/4);
ctx.lineTo(w/4, h/2);
ctx.lineTo(-w/4, h/2);
ctx.lineTo(-w/2, h/4);
ctx.closePath();
ctx.fill();
// Eyes
ctx.fillStyle = '#fff';
ctx.fillRect(-w/5, -h/6, w/8, h/6);
ctx.fillRect(w/10, -h/6, w/8, h/6);
} else if (type === 'elite') {
// Elite - diamond shape
ctx.beginPath();
ctx.moveTo(0, -h/2);
ctx.lineTo(w/2, 0);
ctx.lineTo(0, h/2);
ctx.lineTo(-w/2, 0);
ctx.closePath();
ctx.fill();
ctx.fillStyle = '#fff';
ctx.fillRect(-w/6, -h/5, w/5, h/4);
} else {
// Grunt - classic invader shape
ctx.beginPath();
ctx.moveTo(0, -h/2);
ctx.lineTo(w/3, -h/4);
ctx.lineTo(w/2, 0);
ctx.lineTo(w/3, h/3);
ctx.lineTo(w/5, h/2);
ctx.lineTo(-w/5, h/2);
ctx.lineTo(-w/3, h/3);
ctx.lineTo(-w/2, 0);
ctx.lineTo(-w/3, -h/4);
ctx.closePath();
ctx.fill();
// Eyes
ctx.fillStyle = '#fff';
ctx.fillRect(-w/5, -h/6, w/7, h/5);
ctx.fillRect(w/12, -h/6, w/7, h/5);
}
ctx.restore();
}

function draw() {
ctx.fillStyle = '#0a0a1a';
ctx.fillRect(0, 0, W, H);

// Stars
ctx.fillStyle = '#fff';
for (let s of stars) {
ctx.globalAlpha = 0.4 + s.s * 0.3;
ctx.fillRect(s.x, s.y, s.s, s.s);
}
ctx.globalAlpha = 1;

if (gameState === 'start') {
drawStartScreen();
return;
}

// Particles
for (let p of particles) {
ctx.fillStyle = p.color;
ctx.globalAlpha = p.life / 30;
ctx.fillRect(p.x - 2, p.y - 2, 4, 4);
}
ctx.globalAlpha = 1;

// Aliens
for (let a of aliens) {
if (!a.alive) continue;
drawAlien(a);
}

// Bullets
ctx.fillStyle = '#ff3333';
ctx.shadowColor = '#ff3333';
ctx.shadowBlur = 6;
for (let b of bullets) {
ctx.beginPath();
ctx.arc(b.x, b.y, b.r, 0, Math.PI * 2);
ctx.fill();
}
ctx.shadowBlur = 0;

// Ball
if (ball.active || gameState === 'playing') {
ctx.fillStyle = '#00ffff';
ctx.shadowColor = '#00ffff';
ctx.shadowBlur = 12;
ctx.beginPath();
ctx.arc(ball.x, ball.y, ball.r, 0, Math.PI * 2);
ctx.fill();
ctx.shadowBlur = 0;

if (!ball.active) {
ctx.fillStyle = '#888';
ctx.font = '14px monospace';
ctx.textAlign = 'center';
ctx.fillText('Click or SPACE to launch', W/2, H - 70);
}
}

// Paddle
const grad = ctx.createLinearGradient(paddle.x, paddle.y, paddle.x, paddle.y + paddle.h);
grad.addColorStop(0, '#66ccff');
grad.addColorStop(1, '#336699');
ctx.fillStyle = grad;
ctx.shadowColor = '#66ccff';
ctx.shadowBlur = 8;
roundRect(paddle.x, paddle.y, paddle.w, paddle.h, 4);
ctx.fill();
ctx.shadowBlur = 0;

// HUD
ctx.fillStyle = '#fff';
ctx.font = 'bold 18px monospace';
ctx.textAlign = 'left';
ctx.fillText('SCORE: ' + score, 20, 30);
ctx.textAlign = 'right';
ctx.fillText('HI: ' + highScore, W - 20, 30);

// Lives
ctx.textAlign = 'left';
ctx.font = '16px monospace';
for (let i = 0; i < lives; i++) {
ctx.fillStyle = '#ff4444';
ctx.fillRect(20 + i * 25, 45, 18, 14);
}

if (gameState === 'gameover') drawGameOver();
if (gameState === 'win') drawWin();
}

function roundRect(x, y, w, h, r) {
ctx.beginPath();
ctx.moveTo(x + r, y);
ctx.lineTo(x + w - r, y);
ctx.quadraticCurveTo(x + w, y, x + w, y + r);
ctx.lineTo(x + w, y + h - r);
ctx.quadraticCurveTo(x + w, y + h, x + w - r, y + h);
ctx.lineTo(x + r, y + h);
ctx.quadraticCurveTo(x, y + h, x, y + h - r);
ctx.lineTo(x, y + r);
ctx.quadraticCurveTo(x, y, x + r, y);
ctx.closePath();
}

function drawStartScreen() {
ctx.fillStyle = '#0a0a1a';
ctx.fillRect(0, 0, W, H);
for (let s of stars) {
ctx.globalAlpha = 0.4 + s.s * 0.3;
ctx.fillStyle = '#fff';
ctx.fillRect(s.x, s.y, s.s, s.s);
}
ctx.globalAlpha = 1;

ctx.textAlign = 'center';
ctx.fillStyle = '#00ffff';
ctx.font = 'bold ' + Math.min(48, W/12) + 'px monospace';
ctx.shadowColor = '#00ffff';
ctx.shadowBlur = 20;
ctx.fillText('SPACE INVADERS', W/2, H * 0.3);
ctx.fillStyle = '#ff4444';
ctx.shadowColor = '#ff4444';
ctx.fillText('BREAKOUT', W/2, H * 0.3 + Math.min(50, W/12));
ctx.shadowBlur = 0;

ctx.fillStyle = '#aaa';
ctx.font = '16px monospace';
const lines = [
'← → or A/D to move paddle (or mouse)',
'CLICK / SPACE to launch ball',
'Destroy all aliens before losing your lives!',
'Aliens shoot back - dodge their fire!',
'',
'High Score: ' + highScore
];
lines.forEach((line, i) => {
ctx.fillText(line, W/2, H * 0.5 + i * 24);
});

ctx.fillStyle = '#fff';
ctx.font = 'bold 20px monospace';
if (Math.floor(frameCount / 30) % 2 === 0) {
ctx.fillText('[ CLICK TO START ]', W/2, H * 0.85);
}
}

function drawGameOver() {
ctx.fillStyle = 'rgba(0,0,0,0.7)';
ctx.fillRect(0, 0, W, H);
ctx.textAlign = 'center';
ctx.fillStyle = '#ff4444';
ctx.font = 'bold 48px monospace';
ctx.shadowColor = '#ff4444';
ctx.shadowBlur = 15;
ctx.fillText('GAME OVER', W/2, H * 0.4);
ctx.shadowBlur = 0;
ctx.fillStyle = '#fff';
ctx.font = '24px monospace';
ctx.fillText('Final Score: ' + score, W/2, H * 0.55);
ctx.fillStyle = '#aaa';
ctx.font = '18px monospace';
ctx.fillText('[ R or ENTER to restart ]', W/2, H * 0.7);
}

function drawWin() {
ctx.fillStyle = 'rgba(0,0,0,0.7)';
ctx.fillRect(0, 0, W, H);
ctx.textAlign = 'center';
ctx.fillStyle = '#44ff44';
ctx.font = 'bold 48px monospace';
ctx.shadowColor = '#44ff44';
ctx.shadowBlur = 15;
ctx.fillText('SECTOR CLEARED!', W/2, H * 0.4);
ctx.shadowBlur = 0;
ctx.fillStyle = '#fff';
ctx.font = '24px monospace';
ctx.fillText('Score: ' + score, W/2, H * 0.55);
ctx.fillStyle = '#aaa';
ctx.font = '18px monospace';
ctx.fillText('[ R or ENTER to play again ]', W/2, H * 0.7);
}

// Game loop
function gameLoop() {
update();
draw();
requestAnimationFrame(gameLoop);
}

resetPaddle();
initAliens();
resetBall();
gameLoop();
})();
