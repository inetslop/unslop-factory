// Audio
let audioCtx = null;
function initAudio() {
if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();

function playSound(freq, dur, type, vol) {
if (!audioCtx) return;
const osc = audioCtx.createOscillator();
const gain = audioCtx.createGain();
osc.type = type || 'square';
osc.frequency.value = freq;
gain.gain.setValueAtTime(vol || 0.15, audioCtx.currentTime);
gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + (dur || 0.1));
osc.connect(gain); gain.connect(audioCtx.destination);
osc.start(); osc.stop(audioCtx.currentTime + (dur || 0.1));
}

// Game state
let state = 'start'; // start, playing, gameover, win

function createBricks() {

const type = getZombieType(r, c);
const bw = BRICK_W - BRICK_GAP;

bricks.push({ x, y, w: bw, h: bh, hp: type.hp, maxHp: type.hp, type: type.name, alive: true });
}

function getZombieType(row, col) {
const rand = Math.random();
if (row === 0 && rand > 0.3) return { name: 'boss', hp: 5 };
if (row <= 2 && rand > 0.5) return { name: 'tank', hp: 3 };
if (row <= 4 && rand > 0.7) return { name: 'fast', hp: 2 };
return { name: 'walker', hp: 1 };
}

ball.dy = Math.sin(a) * ball.speed;
}

paddle.x = W / 2 - paddle.w / 2;
paddle.y = H - 50;

particles = [];

function spawnPowerUp(x, y) {

// Drawing helpers
function drawZombieFace(x, y, w, h, type, hp) {
const ratio = hp / (type === 'boss' ? 5 : type === 'tank' ? 3 : type === 'fast' ? 2 : 1);
ctx.save();
// Body
let bodyColor;
if (type === 'boss') bodyColor = '#8B0000';
else if (type === 'tank') bodyColor = '#4a7a2e';
else if (type === 'fast') bodyColor = '#6b8e23';
else bodyColor = '#556b2f';

ctx.fillStyle = bodyColor;
ctx.fillRect(x, y, w, h);
// Damage overlay

const cx = x + w/2, cy = y + h/2;

ctx.beginPath(); ctx.arc(cx - 10, cy - 3, 3, 0, Math.PI*2); ctx.fill();
ctx.beginPath(); ctx.arc(cx + 10, cy - 3, 3, 0, Math.PI*2); ctx.fill();

// Game logic
function updateBricks() {
const speed = ZOMBIE_SPEED * (1 + level * 0.3);
let allDead = true;
for (let b of bricks) {
if (!b.alive) continue;
allDead = false;
b.y += speed;
// Check if brick reached paddle level
if (b.y + b.h >= H - 60) {
return 'gameover';
}
}
if (allDead) return 'win';
return null;
}

function updateBall() {
if (ball.stuck) {
ball.x = paddle.x + paddle.w / 2;
ball.y = paddle.y - ball.r - 5;
return;
}
ball.x += ball.dx;
ball.y += ball.dy;

// Walls
if (ball.x - ball.r < 0) { ball.x = ball.r; ball.dx = -ball.dx; playSound(200, 0.05, 'square', 0.1); }
if (ball.x + ball.r > W) { ball.x = W - ball.r; ball.dx = -ball.dx; playSound(200, 0.05, 'square', 0.1); }
if (ball.y - ball.r < 0) { ball.y = ball.r; ball.dy = -ball.dy; playSound(200, 0.05, 'square', 0.1); }

// Paddle collision
if (ball.dy > 0 && ball.y + ball.r >= paddle.y && ball.y + ball.r <= paddle.y + paddle.h + 5 &&
    ball.x >= paddle.x && ball.x <= paddle.x + paddle.w) {
ball.dy = -Math.abs(ball.dy);
// Angle based on hit position
const hitPos = (ball.x - (paddle.x + paddle.w/2)) / (paddle.w/2);
ball.dx = hitPos * 4;
ball.dy = -Math.sqrt(ball.speed * ball.speed - ball.dx * ball.dx);
playSound(400, 0.08, 'square', 0.15);
}

// Ball lost
if (ball.y - ball.r > H) {
lives--;
playSound(100, 0.3, 'sawtooth', 0.2);
if (lives <= 0) {
state = 'gameover';
try { localStorage.setItem('zombieBreakoutHS', String(Math.max(highScore, score))); } catch(e) {}
highScore = Math.max(highScore, score);
} else {
resetBall();
}
return;
}

// Brick collision
for (let b of bricks) {
if (!b.alive) continue;
if (ball.x + ball.r > b.x && ball.x - ball.r < b.x + b.w &&
    ball.y + ball.r > b.y && ball.y - ball.r < b.y + b.h) {
// Determine side
const overlapX = Math.min(ball.x + ball.r - b.x, b.x + b.w - (ball.x - ball.r));
const overlapY = Math.min(ball.y + ball.r - b.y, b.y + b.h - (ball.y - ball.r));
if (overlapX < overlapY) { ball.dx = -ball.dx; } else { ball.dy = -ball.dy; }

b.hp--;
playSound(300, 0.1, 'square', 0.2);
spawnParticles(ball.x, ball.y, '#4a7a2e', 6);

if (b.hp <= 0) {
b.alive = false;
score += b.type === 'boss' ? 50 : b.type === 'tank' ? 30 : b.type === 'fast' ? 20 : 10;
spawnParticles(b.x + b.w/2, b.y + b.h/2, '#4a7a2e', 12);
playSound(150, 0.2, 'sawtooth', 0.15);
spawnPowerUp(b.x + b.w/2, b.y + b.h/2);
} else {
score += 5;
}
break;
}
}
}

function updatePowerUps() {
for (let i = powerUps.length - 1; i >= 0; i--) {
const p = powerUps[i];
p.y += p.vy;
// Catch with paddle
if (p.y + p.h > paddle.y && p.y < paddle.y + paddle.h &&
    p.x + p.w/2 > paddle.x && p.x + p.w/2 < paddle.x + paddle.w) {
applyPowerUp(p.type);
powerUps.splice(i, 1);
playSound(600, 0.15, 'sine', 0.2);
continue;
}
if (p.y > H + 30) powerUps.splice(i, 1);
}
}

function applyPowerUp(type) {
switch(type) {
case 'widen': paddle.w = Math.min(paddle.w + 20, W * 0.4); break;
case 'multi': {
const newBall = { x: ball.x, y: ball.y, dx: -ball.dx, dy: ball.dy, r: 8, speed: ball.speed, stuck: false };
extraBalls.push(newBall);
playSound(800, 0.1, 'sine', 0.15);
break;
}
case 'slow': zombieSlowTimer = 300; break;
case 'life': lives++; break;
}
}

let extraBalls = [];
let zombieSlowTimer = 0;

function updateExtraBalls() {
for (let i = extraBalls.length - 1; i >= 0; i--) {
const b = extraBalls[i];
b.x += b.dx; b.y += b.dy;
if (b.x - b.r < 0) { b.x = b.r; b.dx = -b.dx; }
if (b.x + b.r > W) { b.x = W - b.r; b.dx = -b.dx; }
if (b.y - b.r < 0) { b.y = b.r; b.dy = -b.dy; }
// Paddle
if (b.dy > 0 && b.y + b.r >= paddle.y && b.y + b.r <= paddle.y + paddle.h + 5 &&
    b.x >= paddle.x && b.x <= paddle.x + paddle.w) {
b.dy = -Math.abs(b.dy);
const hitPos = (b.x - (paddle.x + paddle.w/2)) / (paddle.w/2);
b.dx = hitPos * 4;
b.dy = -Math.sqrt(b.speed * b.speed - b.dx * b.dx);
}
if (b.y - b.r > H) { extraBalls.splice(i, 1); continue; }
// Brick collision (simplified)
for (let br of bricks) {
if (!br.alive) continue;
if (b.x + b.r > br.x && b.x - b.r < br.x + br.w && b.y + b.r > br.y && b.y - b.r < br.y + br.h) {
const overlapX = Math.min(b.x + b.r - br.x, br.x + br.w - (b.x - b.r));
const overlapY = Math.min(b.y + b.r - br.y, br.y + br.h - (b.y - b.r));
if (overlapX < overlapY) b.dx = -b.dx; else b.dy = -b.dy;
br.hp--;
playSound(300, 0.1, 'square', 0.2);
spawnParticles(b.x, b.y, '#4a7a2e', 4);
if (br.hp <= 0) { br.alive = false; score += 15; spawnPowerUp(br.x + br.w/2, br.y + br.h/2); }
break;
}
}
}
}

function updateParticles() {
for (let i = particles.length - 1; i >= 0; i--) {
const p = particles[i];
p.x += p.dx; p.y += p.dy;
p.dy += 0.1;
p.life--;
if (p.life <= 0) particles.splice(i, 1);
}
}

function nextLevel() {
level++;
paddle.w = Math.max(80, 120 - level * 5);
ball.speed = 6 + level * 0.5;
createBricks();
powerUps = [];
extraBalls = [];
resetBall();
state = 'playing';
}

// Main loop
function update() {
if (state !== 'playing') return;

// Paddle movement
if (keysDown['ArrowLeft'] || keysDown['a']) paddle.x -= 8;
if (keysDown['ArrowRight'] || keysDown['d']) paddle.x += 8;
else if (!keysDown['ArrowLeft'] && !keysDown['ArrowRight']) {
paddle.x += (mouseX - paddle.w/2 - paddle.x) * 0.3;
}
paddle.x = Math.max(0, Math.min(W - paddle.w, paddle.x));

// Zombie slow timer
if (zombieSlowTimer > 0) zombieSlowTimer--;

const result = updateBricks();
if (result === 'gameover') { state = 'gameover'; try{localStorage.setItem('zombieBreakoutHS',String(Math.max(highScore,score)));}catch(e){} highScore=Math.max(highScore,score); return; }
if (result === 'win') { state = 'win'; try{localStorage.setItem('zombieBreakoutHS',String(Math.max(highScore,score)));}catch(e){} highScore=Math.max(highScore,score); return; }

updateBall();
updateExtraBalls();
updatePowerUps();
updateParticles();
}

function draw() {
ctx.fillStyle = '#1a0a0a';
ctx.fillRect(0, 0, W, H);

if (state === 'start') { drawStartScreen(); return; }

// Bricks
for (let b of bricks) {
if (!b.alive) continue;
drawZombieFace(b.x, b.y, b.w, b.h, b.type, b.hp);
}

// Power-ups
for (let p of powerUps) {
ctx.fillStyle = '#222';
ctx.fillRect(p.x - 15, p.y - 10, 30, 20);
ctx.fillStyle = '#0f0';
ctx.font = 'bold 12px monospace';
ctx.textAlign = 'center';
const labels = {widen:'W', multi:'M', slow:'S', life:'+'};
ctx.fillText(labels[p.type] || '?', p.x, p.y + 5);
}

// Particles
for (let p of particles) {
ctx.globalAlpha = p.life / 50;
ctx.fillStyle = p.color;
ctx.fillRect(p.x - p.size/2, p.y - p.size/2, p.size, p.size);
}
ctx.globalAlpha = 1;

// Paddle
drawPaddle();

// Ball
if (!ball.stuck || state === 'playing') drawBall();
for (let b of extraBalls) { ctx.fillStyle='#fff'; ctx.beginPath(); ctx.arc(b.x,b.y,b.r,0,Math.PI*2); ctx.fill(); }

// HUD
drawHUD();

if (state === 'gameover') drawGameOver();
if (state === 'win') drawWin();
}

function loop() {
update();
draw();
requestAnimationFrame(loop);
}
loop();
})();
