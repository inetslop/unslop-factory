  audio();

  const r=cv.getBoundingClientRect();
  let cx,cy;
  if(e.touches&&e.touches[0]){cx=e.touches[0].clientX-r.left; cy=e.touches[0].clientY-r.top;}
  else {const m=new MouseEvent('');cx=e.clientX-r.left;cy=e.clientY-r.top;}
  const gx=Math.floor(cx/cell.size), gy=Math.floor(cy/cell.size);
  if(gx<0||gy<0||gx>=W||gy>=H) return;
  if(grid[gy*W+gx]===2){ grid[gy*W+gx]=0; sBreak(); }
}
