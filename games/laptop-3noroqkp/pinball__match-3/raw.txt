function addParticles(x, y, color) {
    for (let i = 0; i < 8; i++) {
        const angle = (i / 8) * Math.PI * 2;
        particles.push({
            x: x, y: y,
            vx: Math.cos(angle) * (2 + Math.random() * 3),
            vy: Math.sin(angle) * (2 + Math.random() * 3),
            life: 1,
            color: color
        });
    }
}

function checkMatchAndPop() {
    if (!collecting) return;
    const counts = {};
    for (let c of collectedColors) {
        counts[c] = (counts[c] || 0) + 1;
    }
    for (let c in counts) {
        if (counts[c] >= 3) {
            const pts = counts[c] === 3 ? 100 : counts[c] === 4 ? 300 : 600;
            score += pts;
            // Pop from grid - remove collected positions
            for (let i = collectedColors.length - 1; i >= 0; i--) {
                const item = collectedColors[i];
                if (item.color === c) {
                    const pos = getGridPos(item.r, item.c);
                    addParticles(pos.x, pos.y, COLORS[c]);
                    grid[item.r][item.c] = -1; // mark empty
                }
            }
            collectedColors = [];
            applyGravity();
        }
    }
}

function applyGravity() {
    for (let c = 0; c < GRID_COLS; c++) {
        let writeRow = GRID_ROWS - 1;
        for (let r = GRID_ROWS - 1; r >= 0; r--) {
            if (grid[r][c] !== -1) {
                grid[writeRow][c] = grid[r][c];
                if (writeRow !== r) grid[r][c] = -1;
                writeRow--;
            }
        }
        // Fill top with new balls
        for (let r = writeRow; r >= 0; r--) {
            grid[r][c] = Math.floor(Math.random() * COLORS.length);
        }
    }
}

function updatePhysics(dt) {
    if (!ball.active) {
        // Ball rests at bottom
        ball.x += (VW/2 - ball.x) * 0.1;
        ball.y = VH - 55;
        return;
    }

    ball.vy += 0.35; // gravity
    ball.x += ball.vx;
    ball.y += ball.vy;

    // Wall bounces (pinball area walls)
    const leftWall = GRID_X - 10;
    const rightWall = GRID_X + GRID_W + 10;
    if (ball.x - BALL_R < leftWall) { ball.x = leftWall + BALL_R; ball.vx = Math.abs(ball.vx); }
    if (ball.x + BALL_R > rightWall) { ball.x = rightWall - BALL_R; ball.vx = -Math.abs(ball.vx); }

    // Top wall
    if (ball.y - BALL_R < 10) { ball.y = 10 + BALL_R; ball.vy = Math.abs(ball.vy) * 0.8; }

    // Flipper collisions
    if (!ball.active) return;

    // Left flipper
    const lfAngle = leftFlipper.active ? -0.3 : 0.6;
    const lfEndX = FLIPPER_LEFT_PIVOT.x + Math.cos(lfAngle + Math.PI/2) * FLIPPER_LEN;
    const lfEndY = FLIPPER_LEFT_PIVOT.y + Math.sin(lfAngle + Math.PI/2) * FLIPPER_LEN;
    if (ball.y > PINBALL_TOP && ball.x < VW/2) {
        // Simple: if ball is near left flipper line, bounce it up
        const dx = lfEndX - FLIPPER_LEFT_PIVOT.x;
        const dy = lfEndY - FLIPPER_LEFT_PIVOT.y;
        const t = ((ball.x - FLIPPER_LEFT_PIVOT.x) * dx + (ball.y - FLIPPER_LEFT_PIVOT.y) * dy) / (dx*dx+dy*dy);
        if (t > 0 && t < 1) {
            const projX = FLIPPER_LEFT_PIVOT.x + t * dx;
            const projY = FLIPPER_LEFT_PIVOT.y + t * dy;
            const dist = Math.hypot(ball.x - projX, ball.y - projY);
            if (dist < BALL_R + 5) {
                ball.vy = -Math.abs(ball.vy) * 0.9 - 3;
                ball.vx += 2;
            }
        }
    }

    // Right flipper
    const rfAngle = rightFlipper.active ? 0.3 : -0.6;
    const rfEndX = FLIPPER_RIGHT_PIVOT.x + Math.cos(rfAngle - Math.PI/2) * FLIPPER_LEN;
    const rfEndY = FLIPPER_RIGHT_PIVOT.y + Math.sin(rfAngle - Math.PI/2) * FLIPPER_LEN;
    if (ball.y > PINBALL_TOP && ball.x >= VW/2) {
        const dx = rfEndX - FLIPPER_RIGHT_PIVOT.x;
        const dy = rfEndY - FLIPPER_RIGHT_PIVOT.y;
        const t = ((ball.x - FLIPPER_RIGHT_PIVOT.x) * dx + (ball.y - FLIPPER_RIGHT_PIVOT.y) * dy) / (dx*dx+dy*dy);
        if (t > 0 && t < 1) {
            const projX = FLIPPER_RIGHT_PIVOT.x + t * dx;
            const projY = FLIPPER_RIGHT_PIVOT.y + t * dy;
            const dist = Math.hypot(ball.x - projX, ball.y - projY);
            if (dist < BALL_R + 5) {
                ball.vy = -Math.abs(ball.vy) * 0.9 - 3;
                ball.vx -= 2;
            }
        }
    }

    // Grid ball collisions and collection
    if (ball.y > GRID_Y && ball.y < PINBALL_TOP) {
        for (let r = 0; r < GRID_ROWS; r++) {
            for (let c = 0; c < GRID_COLS; c++) {
                if (grid[r][c] === -1) continue;
                const pos = getGridPos(r, c);
                const dx = ball.x - pos.x;
                const dy = ball.y - pos.y;
                const dist = Math.hypot(dx, dy);
                if (dist < BALL_R * 2) {
                    // Bounce off
                    const nx = dx / (dist || 1);
                    const ny = dy / (dist || 1);
                    ball.vx += nx * 4;
                    ball.vy += ny * 4;

                    // Collect if same color
                    if (grid[r][c] === ball.color) {
                        collectedColors.push({ r, c, color: grid[r][c] });
                        addParticles(pos.x, pos.y, COLORS[ball.color]);
                        grid[r][c] = -1;
                        checkMatchAndPop();
                    }
                }
            }
        }
    }

    // Ball falls back to pinball area
    if (ball.y > VH - 40 && ball.vy > 0) {
        ball.active = false;
        collecting = false;
        checkMatchAndPop();
        collectedColors = [];
    }

    // Check game over
    if (shotsLeft <= 0 && !ball.active) {
        gameState = 'GAMEOVER';
        if (score > bestScore) {
            bestScore = score;
            try { localStorage.setItem('pinballMatch3Best', bestScore.toString()); } catch(e) {}
        }
    }

    // Update particles
    for (let i = particles.length - 1; i >= 0; i--) {
        particles[i].x += particles[i].vx;
        particles[i].y += particles[i].vy;
        particles[i].life -= 0.03;
        if (particles[i].life <= 0) particles.splice(i, 1);
    }
}

function drawBall(x, y, r, color) {
    const grad = ctx.createRadialGradient(x - r*0.3, y - r*0.3, 0, x, y, r);
    grad.addColorStop(0, lighten(color, 40));
    grad.addColorStop(1, color);
    ctx.fillStyle = grad;
    ctx.beginPath();
    ctx.arc(x, y, r, 0, Math.PI * 2);
    ctx.fill();
}

function lighten(color, amt) {
    let r = parseInt(color.slice(1,3),16), g = parseInt(color.slice(3,5),16), b = parseInt(color.slice(5,7),16);
    r = Math.min(255, r + amt); g = Math.min(255, g + amt); b = Math.min(255, b + amt);
    return `rgb(${r},${g},${b})`;
}

let lastTime = 0;
function gameLoop(time) {
    const dt = Math.min((time - lastTime) / 16.67, 3);
    lastTime = time;
    animTime += dt;

    if (gameState === 'PLAYING') {
        updatePhysics(dt);
    }

    draw();
    requestAnimationFrame(gameLoop);
}
