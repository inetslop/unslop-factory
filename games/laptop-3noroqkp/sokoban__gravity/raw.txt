function getDirVec(dir) {
    return [
        {x:0,y:-1}, // up
        {x:1,y:0},  // right
        {x:0,y:1},  // down
        {x:-1,y:0}  // left
    ][dir];
}

function applyGravity() {
    const dv = getDirVec(gravityDir);
    
    // Collect all movable objects (boxes + player)
    let objects = [];
    boxes.forEach(b => objects.push({type:'box', ref:b, x:b.x, y:b.y}));
    objects.push({type:'player', ref:playerPos, x:playerPos.x, y:playerPos.y});
    
    // Sort: objects furthest in gravity direction go first
    const sorted = objects.sort((a,b) => {
        if (dv.y !== 0) return a.y - b.y; // for down(1), larger y first -> reverse; for up(-1), smaller y first
        else return a.x - b.x;
    });
    
    // For "down" gravity (dv.y=1): process from bottom to top (larger y first)
    // For "up" gravity (dv.y=-1): process from top to bottom (smaller y first)
    // For "right" (dv.x=1): process from right to left (larger x first)
    // For "left" (dv.x=-1): process from left to right (smaller x first)
    
    if (dv.y === 1 || dv.x === 1) sorted.reverse();
    
    // Build occupancy grid
    let occ = {};
    walls.forEach(w => { occ[w.x+','+w.y] = true; });
    boxes.forEach(b => { occ[b.x+','+b.y] = true; });
    occ[playerPos.x+','+playerPos.y] = true;
    
    sorted.forEach(obj => {
        // Remove current position
        delete occ[obj.x+','+obj.y];
        
        let newX = obj.x, newY = obj.y;
        while (true) {
            const tx = newX + dv.x;
            const ty = newY + dv.y;
            if (isWall(tx, ty)) break;
            if (occ[tx+','+ty]) break;
            newX = tx; newY = ty;
        }
        
        // Update position
        obj.ref.x = newX;
        obj.ref.y = newY;
        occ[newX+','+newY] = true;
    });
    
    // Start animation
    boxes.forEach(b => {
        b.animX = b.x; b.animY = b.y;
    });
    animTimer = 200; // ms
}

function spawnParticles(gx, gy, color, count) {
    const cx = (gx + 0.5) * cellSize + offsetX;
    const cy = (gy + 0.5) * cellSize + offsetY;
    for (let i = 0; i < count; i++) {
        const angle = Math.random() * Math.PI * 2;
        const speed = 1 + Math.random() * 3;
        particles.push({
            x: cx, y: cy,
            vx: Math.cos(angle) * speed,
            vy: Math.sin(angle) * speed,
            life: 1,
            color: color,
            size: 2 + Math.random() * 4
        });
    }
}

        boxes.forEach(b => {
            b.animX += (b.x - b.animX) * (1 - t) * 0.3 + (1-t)*0.1;
            b.animY += (b.y - b.animY) * (1 - t) * 0.3 + (1-t)*0.1;
        });
        if (animTimer <= 0) {
            boxes.forEach(b => { b.animX = b.x; b.animY = b.y; });
        }
    }
    
    // Update particles
    particles.forEach(p => {
        p.x += p.vx;
        p.y += p.vy;
        p.vy += 0.1;
        p.life -= dt * 2;
    });
    particles = particles.filter(p => p.life > 0);
    
    // Shake
    let shakeX = 0, shakeY = 0;
    if (shakeTimer > 0) {
        shakeTimer--;
        shakeX = (Math.random() - 0.5) * shakeIntensity;
        shakeY = (Math.random() - 0.5) * shakeIntensity;
    }
    
    ctx.save();
    ctx.translate(shakeX, shakeY);
    
    // Draw grid
    const cs = cellSize;
    
    // Floor
    for (let y = 0; y < gridSize.h; y++) {
        for (let x = 0; x < gridSize.w; x++) {
            if (!isWall(x, y)) {
                ctx.fillStyle = '#1a1a35';
                ctx.fillRect(offsetX + x*cs, offsetY + y*cs, cs, cs);
                // Subtle grid
                ctx.strokeStyle = '#252545';
                ctx.lineWidth = 1;
                ctx.strokeRect(offsetX + x*cs, offsetY + y*cs, cs, cs);
            }
        }
    }
    
    // Walls
    walls.forEach(w => {
        const px = offsetX + w.x * cs;
        const py = offsetY + w.y * cs;
        ctx.fillStyle = '#2a2a4a';
        ctx.fillRect(px, py, cs, cs);
        ctx.strokeStyle = '#3a3a6a';
        ctx.lineWidth = 1;
        ctx.strokeRect(px+2, py+2, cs-4, cs-4);
        // Inner detail
        ctx.fillStyle = '#222244';
        ctx.fillRect(px + cs*0.2, py + cs*0.2, cs*0.6, cs*0.6);
    });
    
    // Targets
    targets.forEach(t => {
        const px = offsetX + (t.x + 0.5) * cs;
        const py = offsetY + (t.y + 0.5) * cs;
        const pulse = 0.7 + 0.3 * Math.sin(time * 3);
        ctx.beginPath();
        ctx.arc(px, py, cs * 0.25 * pulse, 0, Math.PI*2);
        ctx.fillStyle = `rgba(180, 200, 255, ${0.4 * pulse})`;
        ctx.fill();
        ctx.beginPath();
        ctx.arc(px, py, cs * 0.15, 0, Math.PI*2);
        ctx.fillStyle = '#aaccff';
        ctx.fill();
    });
    
    // Boxes
    boxes.forEach(b => {
        const bx = offsetX + (b.animX + 0.5) * cs;
        const by = offsetY + (b.animY + 0.5) * cs;
        const onTarget = targets.some(t => t.x === b.x && t.y === b.y);
        
        ctx.fillStyle = onTarget ? '#4caf50' : '#ff9800';
        ctx.fillRect(bx - cs*0.4, by - cs*0.4, cs*0.8, cs*0.8);
        ctx.strokeStyle = onTarget ? '#81c784' : '#ffb74d';
        ctx.lineWidth = 2;
        ctx.strokeRect(bx - cs*0.35, by - cs*0.35, cs*0.7, cs*0.7);
        
        // Box detail
        ctx.fillStyle = onTarget ? 'rgba(255,255,255,0.2)' : 'rgba(255,255,255,0.15)';
        ctx.fillRect(bx - cs*0.2, by - cs*0.2, cs*0.4, cs*0.4);
    });
    
    // Player
    const ppx = offsetX + (playerPos.x + 0.5) * cs;
    const ppy = offsetY + (playerPos.y + 0.5) * cs;
    ctx.beginPath();
    ctx.arc(ppx, ppy, cs * 0.32, 0, Math.PI*2);
    ctx.fillStyle = '#4fc3f7';
    ctx.fill();
    ctx.strokeStyle = '#81d4fa';
    ctx.lineWidth = 2;
    ctx.stroke();
    
    // Player eye (direction indicator)
    const pdv = lastMoveDir || {x:0,y:1};
    ctx.beginPath();
    ctx.arc(ppx + pdv.x * cs*0.1, ppy + pdv.y * cs*0.1, cs * 0.1, 0, Math.PI*2);
    ctx.fillStyle = '#fff';
    ctx.fill();
    
    // Particles
    particles.forEach(p => {
        ctx.globalAlpha = p.life;
        ctx.fillStyle = p.color;
        ctx.fillRect(p.x - p.size/2, p.y - p.size/2, p.size * p.life, p.size * p.life);
    });
    ctx.globalAlpha = 1;
    
    // UI
    ctx.restore();
    
    // Level indicator
    ctx.fillStyle = '#aaa';
    ctx.font = '16px monospace';
    ctx.textAlign = 'left';
    ctx.fillText(`Level ${currentLevel + 1} / ${levels.length}`, 20, 30);
    
    // Gravity indicator
    const gravLabel = ['↑ UP', '→ RIGHT', '↓ DOWN', '← LEFT'][gravityDir];
    ctx.fillStyle = '#ffeb3b';
    ctx.font = 'bold 18px monospace';
    ctx.textAlign = 'right';
    ctx.fillText(`GRAVITY: ${gravLabel}`, W - 20, 30);
    
    // Controls hint (small)
    ctx.fillStyle = 'rgba(255,255,255,0.3)';
    ctx.font = '12px monospace';
    ctx.textAlign = 'center';
    ctx.fillText('[G/Space] Rotate Gravity  [R] Restart  [N/P] Level', W/2, H - 15);
    
    // Win overlay
    if (gameState === 'win') {
        winTimer += dt;
        const alpha = Math.min(winTimer * 2, 0.7);
        ctx.fillStyle = `rgba(15, 15, 35, ${alpha})`;
        ctx.fillRect(0, 0, W, H);
        
        ctx.fillStyle = '#fff';
        ctx.font = 'bold 48px monospace';
        ctx.textAlign = 'center';
        ctx.fillText('LEVEL COMPLETE!', W/2, H/2 - 30);
        ctx.font = '24px monospace';
        ctx.fillStyle = '#aaa';
        ctx.fillText('Press [G] or [Enter] for next level', W/2, H/2 + 20);
        
        if (winTimer > 1) {
            // Auto-advance hint
            ctx.fillStyle = 'rgba(255,255,255,0.3)';
            ctx.font = '16px monospace';
            ctx.fillText('(or wait...)', W/2, H/2 + 50);
        }
    }
    
    requestAnimationFrame(render);
}
