const NUM_RAYS = W; // will be recalculated
function castRay(angle, maxDist) {
  const sinA = Math.sin(angle), cosA = Math.cos(angle);
  for(let d=0.1;d<maxDist;d+=0.05){
    const x = Math.floor(player.x + cosA*d);
    const y = Math.floor(player.y + sinA*d);
    if(x<0||x>=MAP_SIZE||y<0||y>=MAP_SIZE) return d;
    if(map[y][x]===1) return d;
  }
  return maxDist;
}

function isWall(x,y){
  if(x<0||x>=MAP_SIZE||y<0||y>=MAP_SIZE) return true;
  return map[Math.floor(y)][Math.floor(x)]===1;
}

function hasLOS(x1,y1,x2,y2){
  const dx=x2-x1, dy=y2-y1;
  const dist=Math.sqrt(dx*dx+dy*dy);
  const steps=Math.ceil(dist*8);
  for(let i=0;i<=steps;i++){
    const t=i/steps;
    const px=x1+dx*t, py=y1+dy*t;
    if(isWall(px,py)) return false;
  }
  return true;
}

function update(dt) {
  if(gameState!=='playing') return;
  
  // Player movement
  const speed = player.sneaking ? 1.2 : (player.crouching ? 1.8 : 3.0);
  const turnSpeed = 2.5;
  
  if(keys['q']) player.angle -= turnSpeed*dt;
  if(keys['e']) player.angle += turnSpeed*dt;
  
  let dx=0, dy=0;
  const cosA=Math.cos(player.angle), sinA=Math.sin(player.angle);
  if(keys['w']||keys['arrowup']){dx+=cosA*speed*dt;dy+=sinA*speed*dt;}
  if(keys['s']||keys['arrowdown']){dx-=cosA*speed*dt;dy-=sinA*speed*dt;}
  if(keys['a']||keys['arrowleft']){dx+=-sinA*speed*dt;dy+=cosA*speed*dt;}
  if(keys['d']||keys['arrowright']){dx+=sinA*speed*dt;dy+=-cosA*speed*dt;}
  
  player.crouching = !!keys['shift'];
  player.sneaking = !!keys[' '];
  
  // Collision
  const newX=player.x+dx, newY=player.y+dy;
  if(!isWall(newX+0.3*Math.sign(dx),player.y)) player.x=newX;
  if(!isWall(player.x,newY+0.3*Math.sign(dy))) player.y=newY;
  
  // Footstep sounds
  const moving = dx!==0||dy!==0;
  if(moving && Math.random()<dt*4){
    playSound(player.sneaking?'sneak':'step');
  }
  
  // Check win
  const ptx=Math.floor(player.x), pty=Math.floor(player.y);
  if(pty>=0&&pty<MAP_SIZE&&ptx>=0&&ptx<MAP_SIZE&&map[pty][ptx]===4){
    gameState='won';
    showMsg('EXTRACTED!',' #4f4');
    playSound('win');
  }
  
  // Enemy AI
  detectionLevel = Math.max(0, detectionLevel - dt*0.3);
  
  enemies.forEach(e=>{
    // Patrol movement
    const target = e.patrol[e.pi];
    const tdx=target[0]+0.5-e.x, tdy=target[1]+0.5-e.y;
    const tdist=Math.sqrt(tdx*tdx+tdy*tdy);
    
    if(e.state==='patrol'){
      if(tdist>0.2){
        const ta=Math.atan2(tdy,tdx);
        let da=ta-e.angle;
        while(da>Math.PI)da-=2*Math.PI;while(da<-Math.PI)da+=2*Math.PI;
        e.angle+=da*2*dt;
        if(Math.abs(da)<0.3){
          e.x+=Math.cos(e.angle)*e.speed*dt;
          e.y+=Math.sin(e.angle)*e.speed*dt;
        }
      } else {
        e.pi=(e.pi+1)%e.patrol.length;
      }
      
      // Detection check
      const pdx=player.x-e.x, pdy=player.y-e.y;
      const pdist=Math.sqrt(pdx*pdx+pdy*pdy);
      const viewRange = player.crouching ? 3.5 : (player.sneaking ? 4.5 : 5.5);
      
      if(pdist<viewRange){
        const pdAngle=Math.atan2(pdy,pdx);
        let da=pdAngle-e.angle;
        while(da>Math.PI)da-=2*Math.PI;while(da<-Math.PI)da+=2*Math.PI;
        
        if(Math.abs(da)<Math.PI/3 && hasLOS(e.x,e.y,player.x,player.y)){
          // Check if in light (more visible) or shadow (less visible)
          const inShadow = map[Math.floor(player.y)][Math.floor(player.x)]===3;
          const inLight = map[Math.floor(player.y)][Math.floor(player.x)]===2;
          let detectRate = 1.5;
          if(inShadow) detectRate *= 0.4;
          if(inLight) detectRate *= 1.8;
          if(player.crouching) detectRate *= 0.7;
          if(player.sneaking) detectRate *= 0.8;
          
          e.detectTimer += dt * detectRate;
          detectionLevel = Math.min(1, e.detectTimer / 2);
          
          if(e.detectTimer >= 2){
            e.state='alert';
            e.detectTimer=0;
            alertTimer=5;
            playSound('alert');
          }
        } else {
          e.detectTimer = Math.max(0, e.detectTimer - dt*0.5);
        }
      } else {
        e.detectTimer = Math.max(0, e.detectTimer - dt);
      }
    }
    
    if(e.state==='alert'){
      alertTimer -= dt;
      // Chase player
      const pdx2=player.x-e.x, pdy2=player.y-e.y;
      const pdist2=Math.sqrt(pdx2*pdx2+pdy2*pdy2);
      
      if(pdist2>0.5){
        const ta=Math.atan2(pdy2,pdx2);
        let da=ta-e.angle;
        while(da>Math.PI)da-=2*Math.PI;while(da<-Math.PI)da+=2*Math.PI;
        e.angle+=da*3*dt;
        if(Math.abs(da)<0.5){
          e.x+=Math.cos(e.angle)*1.5*dt;
          e.y+=Math.sin(e.angle)*1.5*dt;
        }
      } else {
        // Caught!
        gameState='caught';
        playSound('caught');
        showMsg('DETECTED - MISSION FAILED','#f44');
      }
      
      if(alertTimer<=0){
        e.state='search';
        e.detectTimer=0;
      }
    }
    
    // Alert others
    if(e.state==='alert'){
      enemies.forEach(other=>{
        if(other!==e){
          const od=Math.sqrt((other.x-e.x)**2+(other.y-e.y)**2);
          if(od<5) other.state='alert';
        }
      });
    }
    
    // Search state - look around then return to patrol
    if(e.state==='search'){
      e.angle += dt*2;
      if(Math.random()<dt*0.3){
        e.state='patrol';
        e.detectTimer=0;
      }
    }
  });
  
  // HUD updates
  document.getElementById('statusHud').textContent = detectionLevel>0.5?'STATUS: ⚠ DETECTING':(detectionLevel>0?'STATUS: SUSPECTED':'STATUS: HIDDEN');
  document.getElementById('statusHud').style.color = detectionLevel>0.5?'#f44':'#4a9eff';
  document.getElementById('posHud').textContent = `POS: ${Math.floor(player.x)},${Math.floor(player.y)}`;
  
  const alertEl = document.getElementById('alertLevel');
  if(detectionLevel>0.3){
    alertEl.style.opacity='1';
    alertEl.textContent = detectionLevel>0.7?'⚠ DETECTED!':'? SUSPECTED...';
  } else {
    alertEl.style.opacity='0';
  }
  
  renderMinimap();
}

let lastTime = 0;
