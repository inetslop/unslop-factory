let audioCtx = null;
function initAudio() {

function playSound(freq, dur, type, vol) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = type || 'sine';
    osc.frequency.value = freq;
    gain.gain.setValueAtTime(vol || 0.3, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + dur);
}

// Physics
const GRAVITY = 0.15;
const FRICTION = 0.998;
const RESTITUTION = 0.75;
const BUMPER_KICK = 6;

function distToSegment(px, py, x1, y1, x2, y2) {
    const dx = x2 - x1, dy = y2 - y1;
    const len2 = dx * dx + dy * dy;
    let t = ((px - x1) * dx + (py - y1) * dy) / len2;
    t = Math.max(0, Math.min(1, t));
    const cx = x1 + t * dx, cy = y1 + t * dy;
    return { x: cx, y: cy, dist: Math.hypot(px - cx, py - cy) };
}

function updatePhysics() {
    if (gameState !== 'PLAYING') return;

    // Gravity
    ball.vy += GRAVITY;
    ball.vx *= FRICTION;
    ball.vy *= FRICTION;

    // Speed cap
    const spd = Math.hypot(ball.vx, ball.vy);
    if (spd > 15) { ball.vx *= 15 / spd; ball.vy *= 15 / spd; }

    ball.x += ball.vx;
    ball.y += ball.vy;

    // Wall collisions
    for (const w of walls) {
        const hit = distToSegment(ball.x, ball.y, w.x1, w.y1, w.x2, w.y2);
        if (hit.dist < BALL_R) {
            const nx = (ball.x - hit.x) / hit.dist;
            const ny = (ball.y - hit.y) / hit.dist;
            ball.x += nx * (BALL_R - hit.dist);
            ball.y += ny * (BALL_R - hit.dist);
            const dot = ball.vx * nx + ball.vy * ny;
            if (dot < 0) {
                ball.vx -= 2 * dot * nx * RESTITUTION;
                ball.vy -= 2 * dot * ny * RESTITUTION;
                playSound(200, 0.1, 'triangle', 0.15);
            }
        }
    }

    // Bumper collisions
    for (const b of bumpers) {
        const dx = ball.x - b.x, dy = ball.y - b.y;
        const dist = Math.hypot(dx, dy);
        if (dist < BALL_R + b.r) {
            const nx = dx / dist, ny = dy / dist;
            ball.x = b.x + nx * (BALL_R + b.r);
            ball.y = b.y + ny * (BALL_R + b.r);
            ball.vx += nx * BUMPER_KICK;
            ball.vy += ny * BUMPER_KICK;
            score += 100;
            comboCount++;
            if (comboCount > 3) { rollovers.forEach(r => r.lit = true); }
            playSound(400 + Math.random() * 200, 0.15, 'square', 0.25);
        }
    }

    // Rollover sensors
    for (const r of rollovers) {
        const dist = Math.hypot(ball.x - r.x, ball.y - r.y);
        if (dist < BALL_R + r.r && !r.lit) {
            r.lit = true;
            score += 50;
            playSound(600, 0.2, 'sine', 0.3);
            if (rollovers.every(rr => rr.lit)) {
                score += 500;
                rollovers.forEach(rr => rr.lit = false);
                comboCount = 0;
                playSound(800, 0.4, 'sine', 0.4);
            }
        }
    }

    // Flipper collision
    collideFlipper(flipperL);
    collideFlipper(flipperR);

    // Drain check
    if (ball.y > TABLE_H + 30) {
        loseBall();
        playSound(150, 0.5, 'sawtooth', 0.2);
    }
}

function collideFlipper(f) {
    const ex = f.px + Math.cos(f.angle) * f.len;
    const ey = f.py + Math.sin(f.angle) * f.len;
    const hit = distToSegment(ball.x, ball.y, f.px, f.py, ex, ey);
    if (hit.dist < BALL_R + 5) {
        const nx = (ball.x - hit.x) / (hit.dist || 1);
        const ny = (ball.y - hit.y) / (hit.dist || 1);
        ball.x += nx * (BALL_R + 5 - hit.dist);
        ball.y += ny * (BALL_R + 5 - hit.dist);
        const dot = ball.vx * nx + ball.vy * ny;
        if (dot < 0) {
            ball.vx -= 2 * dot * nx * RESTITUTION;
            ball.vy -= 2 * dot * ny * RESTITUTION;
        }
        // Add kick from flipper motion
        const t = hit.dist > 0 ? Math.hypot(ball.x - f.px, ball.y - f.py) / f.len : 0;
        const kickSpeed = Math.abs(f.angVel) * (BALL_R + 5);
        if (Math.abs(f.angVel) > 0.01) {
            const perpX = -Math.sin(f.angle);
            const perpY = Math.cos(f.angle);
            ball.vx += perpX * f.angVel * t * 8;
            ball.vy += perpY * f.angVel * t * 8;
        }
    }
}

function updateFlippers() {
    // Left flipper
    const targetL = leftPressed ? flipperL.activeAngle : flipperL.restAngle;
    const diffL = targetL - flipperL.angle;

    flipperL.angle += flipperL.angVel;

    // Right flipper
    const targetR = rightPressed ? flipperR.activeAngle : flipperR.restAngle;
    const diffR = targetR - flipperR.angle;
    flipperR.angVel = diffR * 0.35;
    flipperR.angle += flipperR.angVel;
}

// Main loop
let lastTime = 0;
function gameLoop(time) {
    const dt = Math.min((time - lastTime) / 16.67, 3);
    lastTime = time;

    if (gameState === 'PLAYING') {
        updateFlippers();
        for (let i = 0; i < Math.ceil(dt); i++) {
            updatePhysics();
        }
    } else if (gameState === 'BALL_LOST') {
        ballLostTimer--;
        if (ballLostTimer <= 0) {
            resetBall();
            gameState = 'PLAYING';
        }
    }

    if (gameState === 'START') {
        drawStartScreen();
    } else if (gameState === 'GAME_OVER') {
        drawTable();
        drawHUD();
        drawGameOver();
    } else {
        drawTable();
        drawHUD();
        if (gameState === 'BALL_LOST') drawBallLost();
    }

    requestAnimationFrame(gameLoop);
}
