const canvas = document.getElementById('game');
const ctx = canvas.getContext('2d');

let W, H, cellSize, gridW = 21, gridH = 21;
let maze = [];
let snake = [];
let dir = {x: 1, y: 0};
let nextDir = {x: 1, y: 0};
let food = null;
let score = 0;
let bestScore = 0;
let state = 'start'; // start, playing, over
let lastMove = 0;
const MOVE_INTERVAL = 140;
let audioCtx = null;

function resize() {
    W = window.innerWidth;
    H = window.innerHeight;
    canvas.width = W;
    canvas.height = H;
    cellSize = Math.min(W, H) / (gridW + 2);
}
window.addEventListener('resize', resize);

function initAudio() {
    if (!audioCtx) {
        audioCtx = new (window.AudioContext || window.webkitAudioContext)();
    }
}

function playSound(type) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    
    if (type === 'eat') {
        osc.type = 'sine';
        osc.frequency.setValueAtTime(600, audioCtx.currentTime);
        osc.frequency.linearRampToValueAtTime(900, audioCtx.currentTime + 0.08);
        gain.gain.setValueAtTime(0.3, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.15);
        osc.start();
        osc.stop(audioCtx.currentTime + 0.15);
    } else if (type === 'die') {
        osc.type = 'sawtooth';
        osc.frequency.setValueAtTime(300, audioCtx.currentTime);
        osc.frequency.linearRampToValueAtTime(80, audioCtx.currentTime + 0.4);
        gain.gain.setValueAtTime(0.3, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.4);
        osc.start();
        osc.stop(audioCtx.currentTime + 0.4);
    } else if (type === 'start') {
        osc.type = 'sine';
        osc.frequency.setValueAtTime(400, audioCtx.currentTime);
        osc.frequency.linearRampToValueAtTime(800, audioCtx.currentTime + 0.1);
        gain.gain.setValueAtTime(0.2, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.15);
        osc.start();
        osc.stop(audioCtx.currentTime + 0.15);
    }
}

function generateMaze() {
    // Grid: 21x21, rooms at odd positions (1,3,5,...,19) -> 10x10 rooms
    maze = [];
    for (let y = 0; y < gridH; y++) {
        maze[y] = [];
        for (let x = 0; x < gridW; x++) {
            maze[y][x] = 1; // wall
        }
    }
    
    // Carve passages using recursive backtracker
    const roomsX = 10, roomsY = 10;
    const visited = [];
    for (let y = 0; y < roomsY; y++) {
        visited[y] = [];
        for (let x = 0; x < roomsX; x++) visited[y][x] = false;
    }
    
    function roomToGrid(rx, ry) {
        return { x: rx * 2 + 1, y: ry * 2 + 1 };
    }
    
    const stack = [];
    let startR = { x: 0, y: 0 };
    visited[0][0] = true;
    const g = roomToGrid(0, 0);
    maze[g.y][g.x] = 0;
    stack.push({ x: 0, y: 0 });
    
    const dirs4 = [{x:1,y:0},{x:-1,y:0},{x:0,y:1},{x:0,y:-1}];
    
    while (stack.length > 0) {
        let cur = stack[stack.length - 1];
        let neighbors = [];
        for (let d of dirs4) {
            let nx = cur.x + d.x, ny = cur.y + d.y;
            if (nx >= 0 && nx < roomsX && ny >= 0 && ny < roomsY && !visited[ny][nx]) {
                neighbors.push({x: nx, y: ny, d: d});
            }
        }
        if (neighbors.length === 0) {
            stack.pop();
        } else {
            let next = neighbors[Math.floor(Math.random() * neighbors.length)];
            visited[next.y][next.x] = true;
            // Carve wall between current and next
            const cg = roomToGrid(cur.x, cur.y);
            const ng = roomToGrid(next.x, next.y);
            const wx = (cg.x + ng.x) / 2;
            const wy = (cg.y + ng.y) / 2;
            maze[wy][wx] = 0;
            maze[ng.y][ng.x] = 0;
            stack.push({x: next.x, y: next.y});
        }
    }
    
    // Remove some extra walls to create loops (~15%)
    const extraRemovals = Math.floor(roomsX * roomsY * dirs4.length * 0.12);
    for (let i = 0; i < extraRemovals; i++) {
        let rx = Math.floor(Math.random() * roomsX);
        let ry = Math.floor(Math.random() * roomsY);
        let d = dirs4[Math.floor(Math.random() * dirs4.length)];
        let nx = rx + d.x, ny = ry + d.y;
        if (nx >= 0 && nx < roomsX && ny >= 0 && ny < roomsY) {
            const cg = roomToGrid(rx, ry);
            const ng = roomToGrid(nx, ny);
            const wx = (cg.x + ng.x) / 2;
            const wy = (cg.y + ng.y) / 2;
            maze[wy][wx] = 0;
        }
    }
}

function getOpenCells() {
    let cells = [];
    for (let y = 1; y < gridH - 1; y++) {
        for (let x = 1; x < gridW - 1; x++) {
            if (maze[y][x] === 0) cells.push({x, y});
        }
    }
    return cells;
}

function placeFood() {
    const open = getOpenCells().filter(c => !snake.some(s => s.x === c.x && s.y === c.y));
    if (open.length > 0) {
        food = open[Math.floor(Math.random() * open.length)];
    } else {
        // Win condition - no more space
        state = 'over';
    }
}

function startGame() {
    generateMaze();
    snake = [{x: 1, y: 1}];
    dir = {x: 1, y: 0};
    nextDir = {x: 1, y: 0};
    score = 0;
    placeFood();
    state = 'playing';
    lastMove = performance.now();
}

function moveSnake() {
    dir = nextDir;
    const head = snake[0];
    const nx = head.x + dir.x;
    const ny = head.y + dir.y;
    
    // Check wall collision
    if (nx < 0 || nx >= gridW || ny < 0 || ny >= gridH || maze[ny][nx] === 1) {
        state = 'over';
        playSound('die');
        return;
    }
    
    // Check self collision
    for (let i = 0; i < snake.length - 1; i++) {
        if (snake[i].x === nx && snake[i].y === ny) {
            state = 'over';
            playSound('die');
            return;
        }
    }
    
    snake.unshift({x: nx, y: ny});
    
    if (food && nx === food.x && ny === food.y) {
        score++;
        try { bestScore = Math.max(bestScore, score); localStorage.setItem('snakeMazeBest', bestScore); } catch(e) {}
        playSound('eat');
        placeFood();
    } else {
        snake.pop();
    }
}

function update(timestamp) {
    if (state === 'playing') {
        if (timestamp - lastMove >= MOVE_INTERVAL) {
            moveSnake();
            lastMove = timestamp;
        }
    }
}

function drawMaze() {
    const offX = (W - gridW * cellSize) / 2;
    const offY = (H - gridH * cellSize) / 2 + 10;
    
    // Background
    ctx.fillStyle = '#0f0f1a';
    ctx.fillRect(0, 0, W, H);
    
    // Draw maze walls
    for (let y = 0; y < gridH; y++) {
        for (let x = 0; x < gridW; x++) {
            const px = offX + x * cellSize;
            const py = offY + y * cellSize;
            if (maze[y] && maze[y][x] === 1) {
                ctx.fillStyle = '#2d2d44';
                ctx.fillRect(px, py, cellSize, cellSize);
                // Subtle inner glow
                ctx.fillStyle = '#3a3a5c';
                ctx.fillRect(px + 1, py + 1, cellSize - 2, cellSize - 2);
            } else {
                // Path cells - subtle floor
                ctx.fillStyle = '#16162a';
                ctx.fillRect(px, py, cellSize, cellSize);
            }
        }
    }
    
    // Grid lines (subtle)
    ctx.strokeStyle = 'rgba(255,255,255,0.03)';
    ctx.lineWidth = 1;
    for (let i = 0; i <= gridW; i++) {
        ctx.beginPath();
        ctx.moveTo(offX + i * cellSize, offY);
        ctx.lineTo(offX + i * cellSize, offY + gridH * cellSize);
        ctx.stroke();
    }
    for (let i = 0; i <= gridH; i++) {
        ctx.beginPath();
        ctx.moveTo(offX, offY + i * cellSize);
        ctx.lineTo(offX + gridW * cellSize, offY + i * cellSize);
        ctx.stroke();
    }
    
    // Food
    if (food) {
        const fx = offX + food.x * cellSize + cellSize / 2;
        const fy = offY + food.y * cellSize + cellSize / 2;
        const pulse = 1 + 0.15 * Math.sin(performance.now() * 0.005);
        
        ctx.beginPath();
        ctx.arc(fx, fy, cellSize * 0.3 * pulse, 0, Math.PI * 2);
        ctx.fillStyle = '#ff4757';
        ctx.fill();
        ctx.beginPath();
        ctx.arc(fx, fy, cellSize * 0.2, 0, Math.PI * 2);
        ctx.fillStyle = '#ff6b81';
        ctx.fill();
        
        // Glow
        const grd = ctx.createRadialGradient(fx, fy, 0, fx, fy, cellSize * 0.5);
        grd.addColorStop(0, 'rgba(255,71,87,0.3)');
        grd.addColorStop(1, 'rgba(255,71,87,0)');
        ctx.fillStyle = grd;
        ctx.beginPath();
        ctx.arc(fx, fy, cellSize * 0.5, 0, Math.PI * 2);
        ctx.fill();
    }
    
    // Snake
    for (let i = snake.length - 1; i >= 0; i--) {
        const sx = offX + snake[i].x * cellSize;
        const sy = offY + snake[i].y * cellSize;
        
        if (i === 0) {
            // Head
            ctx.fillStyle = '#2ed573';
            ctx.fillRect(sx + 1, sy + 1, cellSize - 2, cellSize - 2);
            // Eyes
            const ex1 = sx + cellSize * 0.65;
            const ey1 = sy + cellSize * 0.35;
            const ex2 = sx + cellSize * 0.65;
            const ey2 = sy + cellSize * 0.65;
            if (dir.y !== 0) {
                // Vertical movement - eyes on sides
                ctx.fillStyle = '#fff';
                ctx.fillRect(sx + cellSize * 0.2, ey1, cellSize * 0.2, cellSize * 0.2);
                ctx.fillRect(sx + cellSize * 0.6, ey1, cellSize * 0.2, cellSize * 0.2);
            } else {
                ctx.fillStyle = '#fff';
                ctx.fillRect(ex1, sy + cellSize * 0.25, cellSize * 0.2, cellSize * 0.2);
                ctx.fillRect(ex2, sy + cellSize * 0.6, cellSize * 0.2, cellSize * 0.2);
            }
        } else {
            const t = i / snake.length;
            const g = Math.floor(180 - t * 80);
            ctx.fillStyle = `rgb(46, ${g}, 115)`;
            ctx.fillRect(sx + 2, sy + 2, cellSize - 4, cellSize - 4);
        }
    }
    
    // Score
    ctx.fillStyle = '#fff';
    ctx.font = `bold ${Math.min(24, W * 0.03)}px monospace`;
    ctx.textAlign = 'left';
    ctx.fillText(`Score: ${score}`, 15, 30);
    
    try {
        bestScore = parseInt(localStorage.getItem('snakeMazeBest') || '0');
    } catch(e) {}
    ctx.fillStyle = '#888';
    ctx.font = `${Math.min(16, W * 0.02)}px monospace`;
    ctx.fillText(`Best: ${bestScore}`, 15, 52);
}

function render() {
    if (state === 'start') {
        drawStartScreen();
    } else {
        drawMaze();
        if (state === 'over') drawGameOver();
    }
}

function gameLoop(timestamp) {
    update(timestamp);
    render();
    requestAnimationFrame(gameLoop);
}

document.addEventListener('keydown', (e) => {
    initAudio();
    
    if (state === 'start' || state === 'over') {
        if (e.key === ' ' || e.key === 'Enter') {
            playSound('start');
            startGame();
        }
        return;
    }
    
    if (state !== 'playing') return;
    
    let nd = null;
    switch(e.key) {
        case 'ArrowUp': case 'w': case 'W': nd = {x:0, y:-1}; break;
        case 'ArrowDown': case 's': case 'S': nd = {x:0, y:1}; break;
        case 'ArrowLeft': case 'a': case 'A': nd = {x:-1, y:0}; break;
        case 'ArrowRight': case 'd': case 'D': nd = {x:1, y:0}; break;
    }
    
    if (nd) {
        // Prevent 180-degree turns
        if (nd.x !== -dir.x || nd.y !== -dir.y) {
            nextDir = nd;
        }
        e.preventDefault();
    }
});
