// Hit ball
function tryHit() {
    const dx = mouseX - player.x;
    const dy = mouseY - player.y;
    const dist = Math.sqrt(dx*dx + dy*dy);
    if (dist > 80) return; // too far from ball to hit
    
    const bdx = ball.x - player.x;
    const bdy = ball.y - player.y;
    const bdist = Math.sqrt(bdx*bdx + bdy*bdy);
    if (bdist > 80) return; // ball too far from player
    
    // Hit direction: from ball toward mouse
    const hdx = mouseX - ball.x;
    const hdy = mouseY - ball.y;
    const hlen = Math.sqrt(hdx*hdx + hdy*hdy) || 1;
    
    const power = Math.min(8, 3 + dist * 0.05);
    ball.vx = (hdx / hlen) * power;
    ball.vy = (hdy / hlen) * power;
    ball.inHole = false;
    strokes++;
    playSound('hit');
}

// Line of sight check
function hasLineOfSight(gx, gy, px, py) {
    // Check if any bush blocks the line
    for (const b of bushes) {
        const cx = b.x, cy = b.y;
        const dx = px - gx, dy = py - gy;
        const len = Math.sqrt(dx*dx + dy*dy);
        if (len === 0) continue;
        // Distance from bush center to line segment
        const t = ((cx-gx)*dx + (cy-gy)*dy) / (len*len);
        if (t < 0 || t > 1) continue;
        const projX = gx + t * dx;
        const projY = gy + t * dy;
        const distToLine = Math.sqrt((cx-projX)*(cx-projX) + (cy-projY)*(cy-projY));
        if (distToLine < b.r) return false; // blocked
    }
    for (const tr of trees) {
        const cx = tr.x, cy = tr.y;
        const dx = px - gx, dy = py - gy;
        const len = Math.sqrt(dx*dx + dy*dy);
        if (len === 0) continue;
        const t = ((cx-gx)*dx + (cy-gy)*dy) / (len*len);
        if (t < 0 || t > 1) continue;
        const projX = gx + t * dx;
        const projY = gy + t * dy;
        const distToLine = Math.sqrt((cx-projX)*(cx-projX) + (cy-projY)*(cy-projY));
        if (distToLine < tr.r) return false;
    }
    return true;
}

function guardCanSeePlayer(guard) {
    const dx = player.x - guard.x;
    const dy = player.y - guard.y;
    const dist = Math.sqrt(dx*dx + dy*dy);
    if (dist > guard.visionRange) return false;
    
    // Check angle
    const angleToPlayer = Math.atan2(dy, dx);
    let diff = Math.abs(angleDiff(guard.angle, angleToPlayer));
    if (diff > guard.visionArc / 2) return false;
    
    return hasLineOfSight(guard.x, guard.y, player.x, player.y);
}

function angleDiff(a, b) {
    let d = a - b;
    while (d > Math.PI) d -= 2*Math.PI;
    while (d < -Math.PI) d += 2*Math.PI;
    return d;
}

function update(dt) {
    if (gameState !== 'playing') return;
    
    // Player movement
    const speed = 3;
    let px = 0, py = 0;
    if (keys['w'] || keys['W'] || keys['ArrowUp']) py -= 1;
    if (keys['s'] || keys['S'] || keys['ArrowDown']) py += 1;
    if (keys['a'] || keys['A'] || keys['ArrowLeft']) px -= 1;
    if (keys['d'] || keys['D'] || keys['ArrowRight']) px += 1;
    
    const len = Math.sqrt(px*px + py*py);
    if (len > 0) {
        player.x += (px/len) * speed;
        player.y += (py/len) * speed;
    }
    
    // Clamp player
    player.x = Math.max(20, Math.min(W-20, player.x));
    player.y = Math.max(20, Math.min(H-20, player.y));
    
    // Ball physics
    if (!ball.inHole) {
        ball.x += ball.vx;
        ball.y += ball.vy;
        ball.vx *= 0.97;
        ball.vy *= 0.97;
        
        // Bounce off walls
        if (ball.x < 15 || ball.x > W-15) { ball.vx *= -0.8; playSound('bump'); }
        if (ball.y < 15 || ball.y > H-15) { ball.vy *= -0.8; playSound('bump'); }
        ball.x = Math.max(15, Math.min(W-15, ball.x));
        ball.y = Math.max(15, Math.min(H-15, ball.y));
        
        // Tree collision
        for (const tr of trees) {
            const dx = ball.x - tr.x;
            const dy = ball.y - tr.y;
            const d = Math.sqrt(dx*dx + dy*dy);
            if (d < tr.r + 8) {
                const nx = dx/d, ny = dy/d;
                ball.vx = nx * Math.abs(ball.vx) * 0.7;
                ball.vy = ny * Math.abs(ball.vy) * 0.7;
                playSound('bump');
            }
        }
        
        // Check hole
        const hdx = ball.x - hole.x;
        const hdy = ball.y - hole.y;
        if (Math.sqrt(hdx*hdx + hdy*hdy) < 18 && Math.abs(ball.vx) + Math.abs(ball.vy) < 4) {
            ball.inHole = true;
            ball.vx = 0; ball.vy = 0;
            ball.x = hole.x; ball.y = hole.y;
            playSound('sunk');
            const pts = Math.max(0, 100 - strokes * 15);
            score += pts;
            gameState = 'levelComplete';
        }
        
        if (Math.abs(ball.vx) < 0.1 && Math.abs(ball.vy) < 0.1) {
            ball.vx = 0; ball.vy = 0;
        }
    }
    
    // Guard movement
    for (const g of guards) {
        const target = g.path[g.index];
        const dx = target.x - g.x;
        const dy = target.y - g.y;
        const dist = Math.sqrt(dx*dx + dy*dy);
        
        if (dist < 5) {
            g.index = (g.index + 1) % g.path.length;
        } else {
            g.x += (dx/dist) * g.speed;
            g.y += (dy/dist) * g.speed;
        }
        
        // Rotate vision toward movement direction or player if close
        const moveAngle = Math.atan2(dy, dx);
        g.angle = lerpAngle(g.angle, moveAngle, 0.05);
        
        // Look at player if in range
        const pdx = player.x - g.x;
        const pdy = player.y - g.y;
        const pdist = Math.sqrt(pdx*pdx + pdy*pdy);
        if (pdist < g.visionRange * 0.7) {
            const lookAngle = Math.atan2(pdy, pdx);
            g.angle = lerpAngle(g.angle, lookAngle, 0.08);
        }
        
        // Check if spotted
        if (guardCanSeePlayer(g)) {
            gameState = 'caught';
            playSound('alarm');
        }
    }
}

function lerpAngle(a, b, t) {
    let d = b - a;
    while (d > Math.PI) d -= 2*Math.PI;
    while (d < -Math.PI) d += 2*Math.PI;
    return a + d * t;
}

    ctx.fillStyle = 'rgba(0,0,0,0.6)';
    ctx.fillRect(10, 10, 220, 90);
    ctx.fillStyle = '#fff';
    ctx.font = '14px monospace';

    ctx.fillText(`Strokes: ${strokes}  |  Score: ${score}`, 20, 50);
    ctx.fillStyle = '#8f8';
    const detected = guards.some(g => guardCanSeePlayer(g));
    ctx.fillText(detected ? '⚠ DETECTED!' : '✓ Stealth', 20, 75);
    
    // Caught overlay
    if (gameState === 'caught') {
        ctx.fillStyle = 'rgba(180,0,0,0.4)';
        ctx.fillRect(0, 0, W, H);
        ctx.fillStyle = '#fff';
        ctx.font = 'bold 48px monospace';
        ctx.textAlign = 'center';
        ctx.fillText('SPOTTED!', W/2, H/2 - 20);
        ctx.font = '20px monospace';
        ctx.fillText('Press SPACE to retry', W/2, H/2 + 30);
    }
    
    // Level complete overlay
    if (gameState === 'levelComplete') {
        ctx.fillStyle = 'rgba(0,40,0,0.6)';
        ctx.fillRect(0, 0, W, H);
        ctx.fillStyle = '#fff';
        ctx.font = 'bold 42px monospace';
        ctx.textAlign = 'center';
        ctx.fillText('HOLE IN ONE!', W/2, H/2 - 40);
        ctx.font = '24px monospace';
        ctx.fillText(`${strokes} stroke${strokes!==1?'s':''} | +${Math.max(0, 100-strokes*15)} pts`, W/2, H/2 + 10);
        ctx.font = '18px monospace';
        ctx.fillText('Press SPACE to continue', W/2, H/2 + 50);
    }
    
    // Won overlay
    if (gameState === 'won') {
        ctx.fillStyle = 'rgba(0,30,60,0.7)';
        ctx.fillRect(0, 0, W, H);
        ctx.fillStyle = '#ffd700';
        ctx.font = 'bold 48px monospace';
        ctx.textAlign = 'center';
        ctx.fillText('COURSE COMPLETE!', W/2, H/2 - 30);
        ctx.fillStyle = '#fff';
        ctx.font = '28px monospace';
        ctx.fillText(`Final Score: ${score}`, W/2, H/2 + 20);
        ctx.font = '18px monospace';
        ctx.fillText('Press SPACE to play again', W/2, H/2 + 60);
    }
}
