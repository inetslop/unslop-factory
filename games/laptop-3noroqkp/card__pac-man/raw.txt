<script>
let audioCtx = null;
function initAudio() {
  if (!audioCtx) {
    try { audioCtx = new (window.AudioContext || webkitAudioContext)(); } catch(e) {}
  }
}
function playTone(freq, dur, type, vol) {
  if (!audioCtx) return;
  const osc = audioCtx.createOscillator();
  const gain = audioCtx.createGain();
  osc.type = type || 'square';
  osc.frequency.value = freq;
  gain.gain.setValueAtTime(vol || 0.1, audioCtx.currentTime);
  gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
  osc.connect(gain).connect(audioCtx.destination);
  osc.start();
  osc.stop(audioCtx.currentTime + dur);
}
function sfxFlip() { playTone(440, 0.1, 'square', 0.08); }
function sfxMatch() { playTone(523, 0.1); setTimeout(()=>playTone(659, 0.1), 80); setTimeout(()=>playTone(784, 0.15), 160); }
function sfxMismatch() { playTone(200, 0.2, 'sawtooth', 0.06); }
function sfxWin() { for(let i=0;i<8;i++) setTimeout(()=>playTone(400+i*100, 0.15, 'square', 0.07), i*100); }

const CARDS = ['🟡','🔴','🩷','🔵','🟠','⚪','✨','💀'];
let grid = [], flipped = [], moves = 0, pairs = 0, timerInterval = null, seconds = 0, locked = false, started = false;

function getBest() { try { return JSON.parse(localStorage.getItem('pacmanCardsBest') || 'null'); } catch(e) { return null; } }
function setBest(data) { try { localStorage.setItem('pacmanCardsBest', JSON.stringify(data)); } catch(e) {} }

function shuffle(arr) { for(let i=arr.length-1;i>0;i--){ const j=Math.floor(Math.random()*(i+1));[arr[i],arr[j]]=[arr[j],arr[i]];} return arr; }

function createGrid() {
  const deck = [];
  for (let i = 0; i < CARDS.length; i++) { deck.push(CARDS[i], CARDS[i]); }
  grid = shuffle(deck);
}

function renderGrid() {
  const g = document.getElementById('grid');
  g.innerHTML = '';
  grid.forEach((val, idx) => {
    const card = document.createElement('div');
    card.className = 'card';
    card.dataset.index = idx;
    card.innerHTML = `<div class="card-inner"><div class="card-back"></div><div class="card-face">${val}</div></div>`;
    card.addEventListener('click', () => flipCard(idx));
    g.appendChild(card);
  });
}

function flipCard(idx) {
  if (locked || started === false) return;
  const cards = document.querySelectorAll('.card');
  const card = cards[idx];
  if (card.classList.contains('flipped') || card.classList.contains('matched')) return;
  
  initAudio();
  sfxFlip();
  card.classList.add('flipped');
  flipped.push(idx);
  
  if (flipped.length === 2) {
    locked = true;
    moves++;
    document.getElementById('moves').textContent = moves;
    
    const [a, b] = flipped;
    if (grid[a] === grid[b]) {
      setTimeout(() => {
        cards[a].classList.add('matched');
        cards[b].classList.add('matched');
        sfxMatch();
        pairs++;
        document.getElementById('pairs').textContent = pairs + '/8';
        flipped = [];
        locked = false;
        if (pairs === 8) winGame();
      }, 400);
    } else {
      setTimeout(() => {
        sfxMismatch();
        cards[a].classList.remove('flipped');
        cards[b].classList.remove('flipped');
        flipped = [];
        locked = false;
      }, 800);
    }
  }
}

function startTimer() {
  seconds = 0;
  clearInterval(timerInterval);
  timerInterval = setInterval(() => {
    seconds++;
    const m = Math.floor(seconds / 60);
    const s = seconds % 60;
    document.getElementById('timer').textContent = m + ':' + (s < 10 ? '0' : '') + s;
  }, 1000);
}

function stopTimer() { clearInterval(timerInterval); }

function winGame() {
  stopTimer();
  const best = getBest();
  let msg = '🎉 PAC-MAN! You matched all pairs! ';
  if (!best || moves < best.moves) {
    setBest({ moves, time: seconds });
    msg += '<br>🏆 NEW BEST!';
  } else {
    msg += `<br>Best: ${best.moves} moves`;
  }
  document.getElementById('message').innerHTML = msg;
  document.getElementById('restartBtn').style.display = 'inline-block';
  sfxWin();
}

function startGame() {
  initAudio();
  document.getElementById('startScreen').style.display = 'none';
  const ga = document.getElementById('gameArea');
  ga.style.display = 'flex';
  
  moves = 0; pairs = 0; seconds = 0; flipped = []; locked = false; started = true;
  document.getElementById('moves').textContent = '0';
  document.getElementById('pairs').textContent = '0/8';
  document.getElementById('timer').textContent = '0:00';
  document.getElementById('message').innerHTML = '';
  document.getElementById('restartBtn').style.display = 'none';
  
  createGrid();
  renderGrid();
  startTimer();
}

function restart() {
  stopTimer();
  startGame();
}

document.getElementById('startBtn').addEventListener('click', startGame);
document.getElementById('restartBtn').addEventListener('click', restart);

window.addEventListener('keydown', (e) => {
  if (e.key === 'r' || e.key === 'R') {
    if (started) restart();
  }
  if (e.key === 'Escape') {
    stopTimer();
    document.getElementById('gameArea').style.display = 'none';
    document.getElementById('startScreen').style.display = 'flex';
    started = false;
  }
});

// Dots row decoration

  if (i === 5 || i === 6) { d.style.width = '14px'; d.style.height = '14px'; d.style.background = '#fff'; }
  dotsRow.appendChild(d);
}
