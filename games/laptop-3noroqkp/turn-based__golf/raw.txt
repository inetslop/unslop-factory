function playSound(freq, dur, type, vol) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = type || 'sine';
    osc.frequency.value = freq;
    gain.gain.setValueAtTime(vol || 0.3, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + dur);
}

function getLevelPixels() {

function resetBall() {
    const lvl = getLevelPixels();

    ball.y = lastPos.y;

    ball.vy = 0;

function isMoving() {
    return Math.abs(ball.vx) > 0.1 || Math.abs(ball.vy) > 0.1;
}

function updateBall() {
    if (!isMoving()) return;
    
    const lvl = getLevelPixels();
    const friction = 0.985;
    
    // Check sand
    let inSand = false;
    for (const s of lvl.sand) {
        if (ball.x > s.x && ball.x < s.x + s.w && ball.y > s.y && ball.y < s.y + s.h) {
            inSand = true; break;
        }
    }
    
    const f = inSand ? 0.95 : friction;
    ball.vx *= f;
    ball.vy *= f;
    
    if (Math.abs(ball.vx) < 0.1 && Math.abs(ball.vy) < 0.1) {
        ball.vx = 0; ball.vy = 0;
        lastPos = { x: ball.x, y: ball.y };
        return;
    }
    
    ball.x += ball.vx;
    ball.y += ball.vy;
    
    // Wall collisions
    const pad = 40;
    if (ball.x - ball.r < pad) { ball.x = pad + ball.r; ball.vx = Math.abs(ball.vx) * 0.7; playSound(300, 0.1, 'square', 0.1); }
    if (ball.x + ball.r > W - pad) { ball.x = W - pad - ball.r; ball.vx = -Math.abs(ball.vx) * 0.7; playSound(300, 0.1, 'square', 0.1); }
    if (ball.y - ball.r < pad) { ball.y = pad + ball.r; ball.vy = Math.abs(ball.vy) * 0.7; playSound(300, 0.1, 'square', 0.1); }
    if (ball.y + ball.r > H - pad) { ball.y = H - pad - ball.r; ball.vy = -Math.abs(ball.vy) * 0.7; playSound(300, 0.1, 'square', 0.1); }
    
    // Obstacle walls
    for (const w of lvl.walls) {
        if (ball.x + ball.r > w.x && ball.x - ball.r < w.x + w.w && ball.y + ball.r > w.y && ball.y - ball.r < w.y + w.h) {
            const overlapL = ball.x + ball.r - w.x;
            const overlapR = w.x + w.w - (ball.x - ball.r);
            const overlapT = ball.y + ball.r - w.y;
            const overlapB = w.y + w.h - (ball.y - ball.r);
            const minOverlap = Math.min(overlapL, overlapR, overlapT, overlapB);
            if (minOverlap === overlapL) { ball.x = w.x - ball.r; ball.vx = -Math.abs(ball.vx) * 0.7; }
            else if (minOverlap === overlapR) { ball.x = w.x + w.w + ball.r; ball.vx = Math.abs(ball.vx) * 0.7; }
            else if (minOverlap === overlapT) { ball.y = w.y - ball.r; ball.vy = -Math.abs(ball.vy) * 0.7; }
            else { ball.y = w.y + w.h + ball.r; ball.vy = Math.abs(ball.vy) * 0.7; }
            playSound(300, 0.1, 'square', 0.1);
        }
    }
    
    // Water check
    for (const wt of lvl.water) {
        if (ball.x > wt.x && ball.x < wt.x + wt.w && ball.y > wt.y && ball.y < wt.y + wt.h) {
            resetBall();
            return;
        }
    }
    
    // Hole check
    const hp = lvl.holePos;
    const dist = Math.sqrt((ball.x - hp.x) ** 2 + (ball.y - hp.y) ** 2);
    const speed = Math.sqrt(ball.vx * ball.vx + ball.vy * ball.vy);
    if (dist < 18 && speed < 5) {
        gameState = STATE.SINKING;
        sinkAnim = 0;
        playSound(600, 0.3, 'sine', 0.4);
        setTimeout(() => playSound(800, 0.2, 'sine', 0.3), 150);
    }
}

function updateParticles() {
    for (let i = particles.length - 1; i >= 0; i--) {
        particles[i].x += particles[i].vx;
        particles[i].y += particles[i].vy;
        particles[i].life -= 0.02;
        if (particles[i].life <= 0) particles.splice(i, 1);
    }
}

    const lvl = getLevelPixels();

        particles.push({
            x: lvl.holePos.x, y: lvl.holePos.y,
            vx: Math.cos(angle) * speed, vy: Math.sin(angle) * speed,
            life: 1, color: `hsl(${40 + Math.random() * 20}, 80%, 60%)`
        });
    }
}

function drawBall() {
    let bx = ball.x, by = ball.y;
    if (gameState === STATE.SINKING) {
        const scale = 1 - sinkAnim * 0.8;
        const lvl = getLevelPixels();
        bx = bx + (lvl.holePos.x - bx) * sinkAnim;
        by = by + (lvl.holePos.y - by) * sinkAnim;
        ctx.globalAlpha = 1 - sinkAnim * 0.5;
        ctx.beginPath();
        ctx.arc(bx, by, ball.r * scale, 0, Math.PI * 2);
        ctx.fillStyle = '#fff';
        ctx.fill();
        ctx.strokeStyle = '#ccc';
        ctx.lineWidth = 1;
        ctx.stroke();
        ctx.globalAlpha = 1;
        return;
    }
    
    // Shadow
    ctx.fillStyle = 'rgba(0,0,0,0.3)';
    ctx.beginPath();
    ctx.ellipse(bx + 2, by + 2, ball.r, ball.r * 0.7, 0, 0, Math.PI * 2);
    ctx.fill();
    
    // Ball
    const grad = ctx.createRadialGradient(bx - 3, by - 3, 1, bx, by, ball.r);
    grad.addColorStop(0, '#fff');
    grad.addColorStop(1, '#ddd');
    ctx.fillStyle = grad;
    ctx.beginPath();
    ctx.arc(bx, by, ball.r, 0, Math.PI * 2);
    ctx.fill();
    ctx.strokeStyle = '#aaa';
    ctx.lineWidth = 1;
    ctx.stroke();
    
    // Dimple pattern
    ctx.fillStyle = 'rgba(0,0,0,0.1)';
    for (let i = 0; i < 6; i++) {
        const a = (i / 6) * Math.PI * 2;
        ctx.beginPath();
        ctx.arc(bx + Math.cos(a) * 4, by + Math.sin(a) * 4, 1.5, 0, Math.PI * 2);
        ctx.fill();
    }
}

function drawAim() {
    if (!dragging || gameState !== STATE.PLAYING) return;
    
    const dx = dragStart.x - dragEnd.x;
    const dy = dragStart.y - dragEnd.y;
    const dist = Math.sqrt(dx * dx + dy * dy);
    const power = Math.min(dist / 8, 15);
    
    // Arrow
    ctx.strokeStyle = 'rgba(255,255,255,0.7)';
    ctx.lineWidth = 3;
    ctx.setLineDash([5, 5]);
    ctx.beginPath();
    ctx.moveTo(ball.x, ball.y);
    const arrowLen = Math.min(dist * 0.8, 120);
    const angle = Math.atan2(dy, dx);
    ctx.lineTo(ball.x + Math.cos(angle) * arrowLen, ball.y + Math.sin(angle) * arrowLen);
    ctx.stroke();
    ctx.setLineDash([]);
    
    // Power indicator
    const powerPct = power / 15;
    const barW = 60;
    const barH = 8;
    const bx2 = ball.x - barW / 2;
    const by2 = ball.y - 35;
    ctx.fillStyle = 'rgba(0,0,0,0.5)';
    ctx.fillRect(bx2 - 1, by2 - 1, barW + 2, barH + 2);
    const r = Math.floor(255 * powerPct);
    const g = Math.floor(255 * (1 - powerPct));
    ctx.fillStyle = `rgb(${r},${g},0)`;
    ctx.fillRect(bx2, by2, barW * powerPct, barH);
    ctx.strokeStyle = '#fff';
    ctx.lineWidth = 1;
    ctx.strokeRect(bx2, by2, barW, barH);
}

function drawParticles() {
    for (const p of particles) {

        ctx.beginPath();
        ctx.arc(p.x, p.y, 3 * p.life, 0, Math.PI * 2);
        ctx.fill();
    }
    ctx.globalAlpha = 1;
}

    ctx.strokeRect(btnX, btnY, btnW, btnH);
    ctx.fillStyle = '#fff';
    ctx.font = 'bold 20px monospace';
    ctx.fillText('START GAME', W / 2, btnY + 32);
    
    ctx.textAlign = 'left';
}

function drawGameOver() {
    ctx.fillStyle = 'rgba(0,0,0,0.8)';
    ctx.fillRect(0, 0, W, H);
    
    const totalPar = levels.reduce((a, l) => a + l.par, 0);
    let label = '';
    if (totalStrokes <= totalPar - 4) label = 'GOLF GOD!';
    else if (totalStrokes <= totalPar - 2) label = 'AMAZING!';
    else if (totalStrokes === totalPar) label = 'PAR - WELL DONE!';
    else if (totalStrokes <= totalPar + 2) label = 'NOT BAD!';
    else label = 'KEEP PRACTICING!';
    
    ctx.fillStyle = '#fff';
    ctx.font = 'bold 40px monospace';
    ctx.textAlign = 'center';
    ctx.fillText('GAME COMPLETE!', W / 2, H / 2 - 100);
    
    ctx.font = '24px monospace';
    ctx.fillStyle = '#aaffaa';
    ctx.fillText(label, W / 2, H / 2 - 50);
    
    ctx.font = '20px monospace';
    ctx.fillStyle = '#fff';
    ctx.fillText(`Total Strokes: ${totalStrokes}  (Par: ${totalPar})`, W / 2, H / 2);
    
    // Per hole scores
    ctx.font = '16px monospace';
    for (let i = 0; i < levels.length; i++) {
        const s = `H${i+1}: ${holeStrokes}`; // This is simplified, just showing last
    }
    
    // Restart button
    const btnW = 200, btnH = 50;
    const btnX = W / 2 - btnW / 2, btnY = H / 2 + 80;
    ctx.fillStyle = '#4a9';
    ctx.fillRect(btnX, btnY, btnW, btnH);
    ctx.strokeStyle = '#6fc';
    ctx.lineWidth = 2;
    ctx.strokeRect(btnX, btnY, btnW, btnH);
    ctx.fillStyle = '#fff';
    ctx.font = 'bold 20px monospace';
    ctx.fillText('PLAY AGAIN', W / 2, btnY + 32);
    
    ctx.textAlign = 'left';
}

function draw() {
    if (gameState === STATE.MENU) {
        drawMenu();
        return;
    }
    
    if (gameState === STATE.GAME_OVER) {
        drawCourse();
        drawBall();
        drawGameOver();
        return;
    }
    
    drawCourse();
    drawParticles();
    drawBall();
    drawAim();
    drawHUD();
}

function update() {
    if (gameState === STATE.PLAYING) {
        updateBall();
        updateParticles();
    } else if (gameState === STATE.SINKING) {
        sinkAnim += 0.04;
        if (sinkAnim >= 1 && gameState === STATE.SINKING) {
            addSinkParticles();
            gameState = STATE.HOLE_COMPLETE;
            holeCompleteTimer = 0;
        }
        updateParticles();
    } else if (gameState === STATE.HOLE_COMPLETE) {
        holeCompleteTimer += 1 / 60;
        updateParticles();
        if (holeCompleteTimer > 2) {
            totalStrokes += holeStrokes;
            currentLevel++;
            if (currentLevel >= levels.length) {
                gameState = STATE.GAME_OVER;
            } else {
                startLevel(currentLevel);
            }
        }
    }
}

function gameLoop() {
    update();
    draw();
    
    if (gameState === STATE.HOLE_COMPLETE) {
        const lvl = getLevelPixels();
        ctx.fillStyle = 'rgba(0,0,0,0.5)';
        ctx.fillRect(W / 2 - 150, H / 2 - 30, 300, 60);
        ctx.fillStyle = '#fff';
        ctx.font = 'bold 24px monospace';
        ctx.textAlign = 'center';
        const score = holeStrokes - levels[currentLevel].par;
        let label = `Hole ${currentLevel + 1} Complete!`;
        if (score <= -2) label += ' EAGLE!';
        else if (score === -1) label += ' BIRDIE!';
        else if (score === 0) label += ' PAR';
        else if (score === 1) label += ' BOGEY';
        else label += ` +${score}`;
        ctx.fillText(label, W / 2, H / 2 + 8);
        ctx.textAlign = 'left';
    }
    
    requestAnimationFrame(gameLoop);
}

// Input
function getPos(e) {
    const rect = canvas.getBoundingClientRect();
    const x = (e.clientX || (e.touches && e.touches[0].clientX)) - rect.left;
    const y = (e.clientY || (e.touches && e.touches[0].clientY)) - rect.top;
    return { x, y };
}

function onDown(e) {
    initAudio();
    const pos = getPos(e);
    
    if (gameState === STATE.MENU) {
        const btnW = 200, btnH = 50;
        const btnX = W / 2 - btnW / 2, btnY = H / 2 + 9 * 24 + 20;
        if (pos.x > btnX && pos.x < btnX + btnW && pos.y > btnY && pos.y < btnY + btnH) {
            totalStrokes = 0;
            startLevel(0);
        }
        return;
    }
    
    if (gameState === STATE.GAME_OVER) {
        const btnW = 200, btnH = 50;
        const btnX = W / 2 - btnW / 2, btnY = H / 2 + 80;
        if (pos.x > btnX && pos.x < btnX + btnW && pos.y > btnY && pos.y < btnY + btnH) {
            totalStrokes = 0;
            startLevel(0);
        }
        return;
    }
    
    if (gameState === STATE.PLAYING && !isMoving()) {
        const dist = Math.sqrt((pos.x - ball.x) ** 2 + (pos.y - ball.y) ** 2);
        if (dist < 80) {
            dragging = true;
            dragStart = { x: ball.x, y: ball.y };
            dragEnd = { x: pos.x, y: pos.y };
        }
    }
}

function onMove(e) {
    if (!dragging) return;
    const pos = getPos(e);
    dragEnd = pos;
}

function onUp(e) {
    if (!dragging) return;
    dragging = false;
    
    const dx = dragStart.x - dragEnd.x;
    const dy = dragStart.y - dragEnd.y;
    const dist = Math.sqrt(dx * dx + dy * dy);
    
    if (dist < 10) return;
    
    const power = Math.min(dist / 8, 15);
    const angle = Math.atan2(dy, dx);
    ball.vx = Math.cos(angle) * power;
    ball.vy = Math.sin(angle) * power;
    holeStrokes++;
    
    addHitParticles(ball.x, ball.y);
    playSound(440, 0.15, 'sine', 0.3);
}

canvas.addEventListener('mousedown', onDown);
window.addEventListener('mousemove', onMove);
window.addEventListener('mouseup', onUp);
canvas.addEventListener('touchstart', (e) => { e.preventDefault(); onDown(e.touches[0]); });
window.addEventListener('touchmove', (e) => { e.preventDefault(); onMove(e.touches[0]); });
window.addEventListener('touchend', (e) => { e.preventDefault(); onUp(e.changedTouches[0]); });

// Keyboard restart
window.addEventListener('keydown', (e) => {
    if (e.key === 'r' || e.key === 'R') {
        if (gameState === STATE.GAME_OVER || gameState === STATE.MENU) {
            totalStrokes = 0;
            startLevel(0);
        } else if (gameState === STATE.PLAYING) {
            // Reset ball to last pos with penalty
            resetBall();
        }
    }
});

startLevel(0);
gameState = STATE.MENU;
gameLoop();
