<script>
const canvas = document.getElementById('c');
const ctx = canvas.getContext('2d');

let W, H;
function resize() {
    W = window.innerWidth;
    H = window.innerHeight;
    canvas.width = W;
    canvas.height = H;
}
resize();
window.addEventListener('resize', resize);

// Audio
let audioCtx = null;
function initAudio() {
    if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
}
function playSound(freq, dur, type, vol) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = type || 'square';
    osc.frequency.value = freq;
    gain.gain.setValueAtTime(vol || 0.1, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + dur);
}
function playExplosion() {
    if (!audioCtx) return;
    const buf = audioCtx.createBufferSource();
    const buffer = audioCtx.createBuffer(1, 4410, 44100);
    const data = buffer.getChannelData(0);
    for (let i = 0; i < 4410; i++) data[i] = (Math.random() * 2 - 1) * Math.pow(1 - i/4410, 2);
    buf.buffer = buffer;
    const gain = audioCtx.createGain();
    gain.gain.setValueAtTime(0.3, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.4);
    buf.connect(gain);
    gain.connect(audioCtx.destination);
    buf.start();
}

// Game state
let gameState = 'start'; // start, playing, gameover
let score = 0;
let wave = 0;
let playerHP = 5;
let maxHP = 10;
let invulnTimer = 0;
let screenShake = 0;
let time = 0;

// Player
let px, py, pz;
let pvx, pvy;
const PLAYER_Z = 200;
const FOCAL = 300;

// Input
let mouseX = 0, mouseY = 0;
let keys = {};
window.addEventListener('mousemove', e => { mouseX = e.clientX; mouseY = e.clientY; });
window.addEventListener('keydown', e => {
    keys[e.key] = true;
    if (gameState === 'start' && (e.key === ' ' || e.key === 'Enter')) startGame();
    if (gameState === 'gameover' && (e.key === ' ' || e.key === 'Enter')) startGame();
});
window.addEventListener('keyup', e => { keys[e.key] = false; });

// Entities
let playerBullets = [];
let enemyBullets = [];
let enemies = [];
let particles = [];
let stars = [];

function initStars() {
    stars = [];
    for (let i = 0; i < 150; i++) {
        stars.push({
            x: (Math.random() - 0.5) * 2,
            y: -(Math.random() * 0.5 + 0.2),
            z: Math.random() * 800 + 100,
            speed: Math.random() * 0.5 + 0.3
        });
    }
}

function project(x, y, z) {
    const scale = FOCAL / z;
    return {
        x: W/2 + x * scale,
        y: H * 0.35 + y * scale,
        s: scale
    };
}

function startGame() {
    initAudio();
    gameState = 'playing';
    score = 0;
    wave = 1;
    playerHP = maxHP;
    invulnTimer = 0;
    px = 0; py = -0.3; pz = PLAYER_Z;
    pvx = 0; pvy = 0;
    playerBullets = [];
    enemyBullets = [];
    enemies = [];
    particles = [];
    screenShake = 0;
    initStars();
    spawnWave();
}

function spawnWave() {
    const count = Math.min(2 + wave, 8);
    for (let i = 0; i < count; i++) {
        const type = wave >= 3 && i === 0 ? 'boss' : (Math.random() < 0.4 ? 'spiral' : 'fan');
        enemies.push({
            x: (Math.random() - 0.5) * 1.5,
            y: -(Math.random() * 0.3 + 0.2),
            z: 600 + Math.random() * 200,
            hp: type === 'boss' ? 20 + wave * 5 : (type === 'spiral' ? 8 : 5) + wave,
            maxHP: type === 'boss' ? 20 + wave * 5 : (type === 'spiral' ? 8 : 5) + wave,
            type: type,
            timer: Math.random() * 100,
            fireTimer: 60 + Math.random() * 60,
            size: type === 'boss' ? 30 : 15,
            alive: true
        });
    }
}

function fireEnemyBullets(e) {
    const ex = e.x, ey = e.y, ez = e.z;
    if (e.type === 'fan') {
        for (let i = -2; i <= 2; i++) {
            enemyBullets.push({
                x: ex, y: ey, z: ez,
                vx: i * 0.003, vy: 0.005, vz: 0.8,
                life: 1
            });
        }
    } else if (e.type === 'spiral') {
        const angle = e.timer * 0.1;
        for (let i = 0; i < 3; i++) {
            const a = angle + i * (Math.PI * 2 / 3);
            enemyBullets.push({
                x: ex, y: ey, z: ez,
                vx: Math.cos(a) * 0.004, vy: Math.sin(a) * 0.002 + 0.004, vz: 0.7,
                life: 1
            });
        }
    } else if (e.type === 'boss') {
        const angle = e.timer * 0.05;
        for (let i = 0; i < 8; i++) {
            const a = angle + i * (Math.PI * 2 / 8);
            enemyBullets.push({
                x: ex, y: ey, z: ez,
                vx: Math.cos(a) * 0.005, vy: Math.sin(a) * 0.003 + 0.004, vz: 0.6,
                life: 1
            });
        }
    }
}

function addParticle(x, y, z, color) {
    for (let i = 0; i < 8; i++) {
        const a = Math.random() * Math.PI * 2;
        const sp = 0.01 + Math.random() * 0.02;
        particles.push({
            x, y, z,
            vx: Math.cos(a) * sp, vy: Math.sin(a) * sp * 0.5, vz: (Math.random() - 0.5) * 0.3,
            life: 1, color: color || '#ff6600'
        });
    }
}

function update() {
    if (gameState !== 'playing') return;
    time++;
    
    // Player movement
    const targetX = ((mouseX / W) - 0.5) * 1.2;
    const targetY = -(((mouseY / H) - 0.3) * 0.8);
    px += (targetX - px) * 0.08;
    py += (targetY - py) * 0.08;
    
    if (invulnTimer > 0) invulnTimer--;
    if (screenShake > 0) screenShake *= 0.9;
    
    // Player fire
    if (time % 4 === 0) {
        playerBullets.push({ x: px, y: py + 0.1, z: pz, vx: 0, vy: 0.02, vz: 3 });
        playSound(880, 0.05, 'square', 0.03);
    }
    
    // Update player bullets
    for (let i = playerBullets.length - 1; i >= 0; i--) {
        const b = playerBullets[i];
        b.x += b.vx; b.y += b.vy; b.z -= b.vz;
        if (b.z < 50) { playerBullets.splice(i, 1); continue; }
        
        // Hit enemies
        for (let j = enemies.length - 1; j >= 0; j--) {
            const e = enemies[j];
            if (!e.alive) continue;
            const dx = b.x - e.x, dy = b.y - e.y, dz = b.z - e.z;
            const dist = Math.sqrt(dx*dx + dy*dy + dz*dz);
            if (dist < 0.15) {
                e.hp--;
                playerBullets.splice(i, 1);
                playSound(200, 0.1, 'sawtooth', 0.05);
                if (e.hp <= 0) {
                    e.alive = false;
                    enemies.splice(j, 1);
                    score += e.type === 'boss' ? 500 : 100;
                    addParticle(e.x, e.y, e.z, '#ff4400');
                    playExplosion();
                }
                break;
            }
        }
    }
    
    // Update enemies
    for (let i = enemies.length - 1; i >= 0; i--) {
        const e = enemies[i];
        if (!e.alive) continue;
        e.timer++;
        e.z -= 0.3 + wave * 0.05;
        e.y += 0.002;
        
        // Slight homing
        e.x += (px - e.x) * 0.001;
        
        if (e.z < PLAYER_Z - 30) {
            enemies.splice(i, 1);
            continue;
        }
        
        e.fireTimer--;
        if (e.fireTimer <= 0) {
            fireEnemyBullets(e);
            e.fireTimer = e.type === 'boss' ? 40 : 80 - Math.min(wave * 3, 40);
        }
    }
    
    // Update enemy bullets
    for (let i = enemyBullets.length - 1; i >= 0; i--) {
        const b = enemyBullets[i];
        b.x += b.vx; b.y += b.vy; b.z -= b.vz;
        if (b.z < 50) { enemyBullets.splice(i, 1); continue; }
        
        // Hit player
        if (invulnTimer <= 0) {
            const dx = b.x - px, dy = b.y - py;
            const dist = Math.sqrt(dx*dx + dy*dy + (b.z - pz)*(b.z - pz));
            if (dist < 0.12) {
                playerHP--;
                invulnTimer = 60;
                screenShake = 10;
                enemyBullets.splice(i, 1);
                playSound(150, 0.3, 'sawtooth', 0.2);
                if (playerHP <= 0) {
                    gameState = 'gameover';
                    addParticle(px, py, pz, '#ff0000');
                    playExplosion();
                }
            }
        }
    }
    
    // Update particles
    for (let i = particles.length - 1; i >= 0; i--) {
        const p = particles[i];
        p.x += p.vx; p.y += p.vy; p.z -= p.vz;
        p.life -= 0.03;
        if (p.life <= 0) particles.splice(i, 1);
    }
    
    // Stars
    for (let s of stars) {
        s.z -= s.speed * 2;
        if (s.z < 50) { s.z = 800; s.x = (Math.random()-0.5)*2; s.y = -(Math.random()*0.5+0.2); }
    }
    
    // Wave management
    if (enemies.length === 0) {
        wave++;
        setTimeout(() => { if (gameState === 'playing') spawnWave(); }, 1500);
    }
}

function draw() {
    ctx.save();
    if (screenShake > 0.5) {
        ctx.translate((Math.random()-0.5)*screenShake, (Math.random()-0.5)*screenShake);
    }
    
    // Background
    const grad = ctx.createRadialGradient(W/2, H*0.35, 0, W/2, H*0.35, W*0.6);
    grad.addColorStop(0, '#1a0a2e');
    grad.addColorStop(0.5, '#0d0221');
    grad.addColorStop(1, '#000000');
    ctx.fillStyle = grad;
    ctx.fillRect(-20, -20, W+40, H+40);
    
    // Grid floor
    ctx.strokeStyle = 'rgba(80, 0, 255, 0.15)';
    ctx.lineWidth = 1;
    for (let i = 0; i < 20; i++) {
        const z = i * 50 + 50;
        const p1 = project(-1.5, -0.5, z);
        const p2 = project(1.5, -0.5, z);
        ctx.beginPath();
        ctx.moveTo(p1.x, p1.y);
        ctx.lineTo(p2.x, p2.y);
        ctx.stroke();
    }
    for (let i = -6; i <= 6; i++) {
        const p1 = project(i * 0.25, -0.5, 50);
        const p2 = project(i * 0.25, -0.5, 800);
        ctx.beginPath();
        ctx.moveTo(p1.x, p1.y);
        ctx.lineTo(p2.x, p2.y);
        ctx.stroke();
    }
    
    // Stars
    for (let s of stars) {
        const p = project(s.x, s.y, s.z);
        const size = Math.max(0.5, 3 * p.s);
        ctx.fillStyle = `rgba(200, 200, 255, ${Math.min(1, p.s * 2)})`;
        ctx.beginPath();
        ctx.arc(p.x, p.y, size, 0, Math.PI*2);
        ctx.fill();
    }
    
    // Sort all entities by z (far to near)
    let drawList = [];
    
    for (let e of enemies) {
        if (e.alive) drawList.push({ type: 'enemy', ref: e, z: e.z });
    }
    for (let b of enemyBullets) drawList.push({ type: 'ebullet', ref: b, z: b.z });
    for (let b of playerBullets) drawList.push({ type: 'pbullet', ref: b, z: b.z });
    for (let p of particles) drawList.push({ type: 'particle', ref: p, z: p.z });
    drawList.sort((a, b) => b.z - a.z);
    
    // Draw entities back to front
    for (let item of drawList) {
        const obj = item.ref;
        const p = project(obj.x, obj.y, obj.z);
        
        if (item.type === 'enemy') {
            const size = obj.size * p.s;
            ctx.save();
            ctx.translate(p.x, p.y);
            
            // Glow
            ctx.shadowColor = obj.type === 'boss' ? '#ff0000' : '#ff8800';
            ctx.shadowBlur = 15;
            
            if (obj.type === 'boss') {
                ctx.fillStyle = '#ff2222';
                ctx.beginPath();
                for (let i = 0; i < 6; i++) {
                    const a = i * Math.PI/3 + obj.timer * 0.02;
                    const r = size * (i % 2 === 0 ? 1 : 0.7);
                    ctx.lineTo(Math.cos(a)*r, Math.sin(a)*r);
                }
                ctx.closePath();
                ctx.fill();
                ctx.strokeStyle = '#ff6666';
                ctx.lineWidth = 2;
                ctx.stroke();
                
                // HP bar
                ctx.shadowBlur = 0;
                ctx.fillStyle = 'rgba(0,0,0,0.5)';
                ctx.fillRect(-size, -size-15, size*2, 6);
                ctx.fillStyle = '#ff3333';
                ctx.fillRect(-size, -size-15, size*2*(obj.hp/obj.maxHP), 6);
            } else {
                ctx.fillStyle = obj.type === 'spiral' ? '#aa44ff' : '#ffaa00';
                ctx.beginPath();
                for (let i = 0; i < 5; i++) {
                    const a = i * Math.PI*2/5 + obj.timer * 0.05;
                    ctx.lineTo(Math.cos(a)*size, Math.sin(a)*size);
                }
                ctx.closePath();
                ctx.fill();
                ctx.strokeStyle = '#fff';
                ctx.lineWidth = 1;
                ctx.stroke();
            }
            ctx.restore();
        }
        else if (item.type === 'ebullet') {
            const size = Math.max(2, 8 * p.s);
            ctx.shadowColor = '#ff0066';
            ctx.shadowBlur = 10;
            ctx.fillStyle = '#ff3399';
            ctx.beginPath();
            ctx.arc(p.x, p.y, size, 0, Math.PI*2);
            ctx.fill();
            ctx.fillStyle = '#fff';
            ctx.beginPath();
            ctx.arc(p.x, p.y, size * 0.4, 0, Math.PI*2);
            ctx.fill();
        }
        else if (item.type === 'pbullet') {
            const size = Math.max(1, 5 * p.s);
            ctx.shadowColor = '#00ffff';
            ctx.shadowBlur = 8;
            ctx.fillStyle = '#00ffff';
            ctx.fillRect(p.x - size/2, p.y - size*2, size, size*4);
        }
        else if (item.type === 'particle') {
            const size = Math.max(1, 5 * p.s * obj.life);
            ctx.fillStyle = obj.color;
            ctx.globalAlpha = obj.life;
            ctx.beginPath();
            ctx.arc(p.x, p.y, size, 0, Math.PI*2);
            ctx.fill();
            ctx.globalAlpha = 1;
        }
    }
    
    // Player ship
    if (gameState === 'playing') {
        const pp = project(px, py, pz);
        const psz = FOCAL / pz * 20;
        
        ctx.save();
        ctx.translate(pp.x, pp.y);
        
        if (invulnTimer > 0 && Math.floor(time/4) % 2 === 0) {
            ctx.globalAlpha = 0.5;
        }
        
        // Engine glow
        ctx.shadowColor = '#00aaff';
        ctx.shadowBlur = 15;
        ctx.fillStyle = 'rgba(0, 170, 255, 0.6)';
        ctx.beginPath();
        ctx.ellipse(0, psz*0.8, psz*0.3, psz*0.6 + Math.random()*4, 0, 0, Math.PI*2);
        ctx.fill();
        
        // Ship body
        ctx.fillStyle = '#ffffff';
        ctx.beginPath();
        ctx.moveTo(0, -psz);
        ctx.lineTo(-psz*0.5, psz*0.6);
        ctx.lineTo(-psz*0.2, psz*0.4);
        ctx.lineTo(psz*0.2, psz*0.4);
        ctx.lineTo(psz*0.5, psz*0.6);
        ctx.closePath();
        ctx.fill();
        ctx.strokeStyle = '#00ccff';
        ctx.lineWidth = 1.5;
        ctx.stroke();
        
        ctx.restore();
    }
    
    // HUD
    if (gameState === 'playing') {
        ctx.fillStyle = 'rgba(0,0,0,0.5)';
        ctx.fillRect(10, 10, 200, 80);
        
        ctx.fillStyle = '#fff';
        ctx.font = '14px monospace';
        ctx.fillText(`SCORE: ${score}`, 20, 30);
        ctx.fillText(`WAVE: ${wave}`, 20, 50);
        
        // HP bar
        ctx.fillStyle = '#333';
        ctx.fillRect(20, 65, 150, 12);
        const hpPct = playerHP / maxHP;
        ctx.fillStyle = hpPct > 0.5 ? '#0f0' : hpPct > 0.25 ? '#ff0' : '#f00';
        ctx.fillRect(20, 65, 150 * hpPct, 12);
        ctx.strokeStyle = '#fff';
        ctx.strokeRect(20, 65, 150, 12);
    }
    
    // Start screen
    if (gameState === 'start') {
        ctx.fillStyle = 'rgba(0,0,0,0.7)';
        ctx.fillRect(0, 0, W, H);
        
        ctx.textAlign = 'center';
        ctx.fillStyle = '#ff3399';
        ctx.font = 'bold 48px monospace';
        ctx.shadowColor = '#ff3399';
        ctx.shadowBlur = 20;
        ctx.fillText('BULLET HELL 3D', W/2, H*0.3);
        
        ctx.shadowBlur = 0;
        ctx.fillStyle = '#fff';
        ctx.font = '18px monospace';
        ctx.fillText('Move: Mouse', W/2, H*0.45);
        ctx.fillText('Auto-fire: Enabled', W/2, H*0.5);
        ctx.fillText('Survive the waves!', W/2, H*0.55);
        
        ctx.fillStyle = '#0ff';
        ctx.font = '24px monospace';
        if (Math.floor(time/30) % 2 === 0) {
            ctx.fillText('[ PRESS SPACE TO START ]', W/2, H*0.7);
        }
        ctx.textAlign = 'left';
    }
    
    // Game over
    if (gameState === 'gameover') {
        ctx.fillStyle = 'rgba(0,0,0,0.6)';
        ctx.fillRect(0, 0, W, H);
        
        ctx.textAlign = 'center';
        ctx.fillStyle = '#ff3333';
        ctx.font = 'bold 42px monospace';
        ctx.shadowColor = '#ff0000';
        ctx.shadowBlur = 20;
        ctx.fillText('GAME OVER', W/2, H*0.35);
        
        ctx.shadowBlur = 0;
        ctx.fillStyle = '#fff';
        ctx.font = '20px monospace';
        ctx.fillText(`FINAL SCORE: ${score}`, W/2, H*0.48);
        ctx.fillText(`WAVES SURVIVED: ${wave}`, W/2, H*0.54);
        
        ctx.fillStyle = '#0ff';
        ctx.font = '18px monospace';
        if (Math.floor(time/30) % 2 === 0) {
            ctx.fillText('[ PRESS SPACE TO RESTART ]', W/2, H*0.7);
        }
        ctx.textAlign = 'left';
    }
    
    ctx.restore();
}

function loop() {
    update();
    draw();
    requestAnimationFrame(loop);
}

initStars();
px = 0; py = -0.3; pz = PLAYER_Z;
loop();
</script>
