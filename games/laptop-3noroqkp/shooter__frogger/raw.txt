(function() {
const canvas = document.getElementById('game');

// Audio
let audioCtx = null;
function initAudio() {
if (!audioCtx) {
audioCtx = new (window.AudioContext || window.webkitAudioContext)();
}
}

function playSound(type) {
if (!audioCtx) return;
const osc = audioCtx.createOscillator();
const gain = audioCtx.createGain();
osc.connect(gain);
gain.connect(audioCtx.destination);
switch(type) {
case 'shoot':
osc.type = 'square';
osc.frequency.setValueAtTime(800, audioCtx.currentTime);
osc.frequency.exponentialRampToValueAtTime(200, audioCtx.currentTime + 0.1);
gain.gain.setValueAtTime(0.15, audioCtx.currentTime);
gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.1);
osc.start(); osc.stop(audioCtx.currentTime + 0.1);
break;
case 'hit':
osc.type = 'sawtooth';
osc.frequency.setValueAtTime(300, audioCtx.currentTime);
osc.frequency.exponentialRampToValueAtTime(50, audioCtx.currentTime + 0.2);
gain.gain.setValueAtTime(0.2, audioCtx.currentTime);
gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.2);
osc.start(); osc.stop(audioCtx.currentTime + 0.2);
break;
case 'die':
osc.type = 'sawtooth';
osc.frequency.setValueAtTime(400, audioCtx.currentTime);
osc.frequency.exponentialRampToValueAtTime(30, audioCtx.currentTime + 0.5);
gain.gain.setValueAtTime(0.3, audioCtx.currentTime);
gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.5);
osc.start(); osc.stop(audioCtx.currentTime + 0.5);
break;
case 'score':
osc.type = 'sine';
osc.frequency.setValueAtTime(600, audioCtx.currentTime);
osc.frequency.exponentialRampToValueAtTime(1200, audioCtx.currentTime + 0.1);
gain.gain.setValueAtTime(0.15, audioCtx.currentTime);
gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.15);
osc.start(); osc.stop(audioCtx.currentTime + 0.15);
break;
}
}

// Game State
let gameState = 'start'; // start, playing, gameover
let score = 0;
let highScore = 0;
let lives = 3;
let wave = 1;
let enemiesKilledThisWave = 0;
let enemiesNeededForWave = 10;

try {
highScore = parseInt(localStorage.getItem('froggerShooterHigh') || '0');
} catch(e) {}

// Player
const player = { x: 0, y: 0, w: 40, h: 35, speed: 5, shootCooldown: 0 };

// Arrays
let bullets = [];
let enemies = [];
let particles = [];
let spawnTimer = 0;
let spawnInterval = 90;

// Input
const keys = {};
window.addEventListener('keydown', (e) => {
keys[e.key] = true;
if (gameState === 'start' && (e.key === ' ' || e.key === 'Enter')) {
initAudio();
startGame();
} else if (gameState === 'gameover' && (e.key === ' ' || e.key === 'Enter')) {
startGame();
}
if (e.key === ' ') e.preventDefault();
});
window.addEventListener('keyup', (e) => { keys[e.key] = false; });

function startGame() {
gameState = 'playing';
score = 0;
lives = 3;
wave = 1;
enemiesKilledThisWave = 0;
enemiesNeededForWave = 10;
bullets = [];
enemies = [];
particles = [];
spawnTimer = 0;
player.x = W / 2 - player.w / 2;
player.y = H - 60;
initAudio();
}

// Enemy types
function spawnEnemy() {
const types = ['bike', 'car', 'turtle', 'motorcycle'];
let type, hp, speed, w, h, points;
const r = Math.random();
if (r < 0.3) { type='bike'; hp=1; speed=1.5+Math.random()*1.5; w=30; h=25; points=10; }
else if (r < 0.6) { type='car'; hp=2; speed=1+Math.random(); w=45; h=28; points=20; }
else if (r < 0.8) { type='turtle'; hp=3; speed=0.5+Math.random()*0.5; w=35; h=30; points=40; }
else { type='motorcycle'; hp=2; speed=2.5+Math.random()*1.5; w=28; h=22; points=30; }

const fromLeft = Math.random() < 0.5;
const laneY = 60 + Math.floor(Math.random() * 5) * (H / 7);
const e = {
type, hp, maxHp: hp, speed, w, h, points,
x: fromLeft ? -w : W,
y: laneY,
vx: fromLeft ? speed : -speed,
alive: true,
flash: 0
};
enemies.push(e);
}

function shoot() {
if (player.shootCooldown <= 0) {
bullets.push({ x: player.x + player.w / 2, y: player.y, w: 4, h: 12, vy: -8 });
player.shootCooldown = 12;
playSound('shoot');
}
}

function createParticles(x, y, color, count) {
for (let i = 0; i < count; i++) {
particles.push({
x, y,
vx: (Math.random()-0.5)*6,
vy: (Math.random()-0.5)*6 - 2,
life: 30 + Math.random()*20,
maxLife: 50,
color, size: 3+Math.random()*4
});
}
}

function update() {
if (gameState !== 'playing') return;

// Player movement
let dx = 0;
if (keys['ArrowLeft'] || keys['a'] || keys['A']) dx = -player.speed;
if (keys['ArrowRight'] || keys['d'] || keys['D']) dx = player.speed;
player.x += dx;
player.x = Math.max(0, Math.min(W - player.w, player.x));

// Shoot
if (keys[' '] || keys['w'] || keys['W'] || keys['ArrowUp']) {
shoot();
}
player.shootCooldown--;

// Spawn enemies
spawnTimer++;
if (spawnTimer >= spawnInterval) {
spawnTimer = 0;
spawnEnemy();
if (wave > 1 && Math.random() < 0.3) spawnEnemy();
if (enemiesKilledThisWave + enemies.length >= 5) {
// speed up spawning
}
}

// Update bullets
for (let i = bullets.length - 1; i >= 0; i--) {
bullets[i].y += bullets[i].vy;
if (bullets[i].y < -20) bullets.splice(i, 1);
}

// Update enemies
for (let i = enemies.length - 1; i >= 0; i--) {
const e = enemies[i];
e.x += e.vx;
if (e.flash > 0) e.flash--;
// Remove if off screen
if ((e.vx > 0 && e.x > W + 50) || (e.vx < 0 && e.x < -50)) {
enemies.splice(i, 1);
continue;
}
// Check if enemy reaches bottom
if (e.y + e.h >= H - 20) {
// Enemy reached the frog's level
createParticles(e.x + e.w/2, e.y + e.h/2, '#ff4444', 8);
enemies.splice(i, 1);
lives--;
playSound('die');
if (lives <= 0) {
gameState = 'gameover';
try { localStorage.setItem('froggerShooterHigh', String(highScore)); } catch(e2) {}
}
continue;
}

// Bullet collision
for (let j = bullets.length - 1; j >= 0; j--) {
const b = bullets[j];
if (b.x < e.x + e.w && b.x + b.w > e.x && b.y < e.y + e.h && b.y + b.h > e.y) {
e.hp--;
e.flash = 5;
bullets.splice(j, 1);
playSound('hit');
if (e.hp <= 0) {
e.alive = false;
enemies.splice(i, 1);
score += e.points;
enemiesKilledThisWave++;
playSound('score');
createParticles(e.x + e.w/2, e.y + e.h/2, '#44ff44', 10);
if (enemiesKilledThisWave >= enemiesNeededForWave) {
wave++;
enemiesKilledThisWave = 0;
enemiesNeededForWave += 5;
spawnInterval = Math.max(30, spawnInterval - 5);
}
}
break;
}
}

// Update particles
for (let i = particles.length - 1; i >= 0; i--) {
const p = particles[i];
p.x += p.vx;
p.y += p.vy;
p.vy += 0.2;
p.life--;
if (p.life <= 0) particles.splice(i, 1);
}

// Update high score
if (score > highScore) {
highScore = score;
try { localStorage.setItem('froggerShooterHigh', String(highScore)); } catch(e) {}
}
}

function drawFrog(x, y, w, h) {
ctx.fillStyle = '#33cc33';

ctx.fill();
// Head
ctx.beginPath();
ctx.ellipse(x + w/2, y + h*0.3, w/3, h/4, 0, 0, Math.PI*2);
ctx.fill();

ctx.fill();
ctx.fillStyle = '#000';
ctx.beginPath();
ctx.arc(x + w*0.3, y + h*0.25, 2.5, 0, Math.PI*2);
ctx.arc(x + w*0.7, y + h*0.25, 2.5, 0, Math.PI*2);
ctx.fill();
// Legs
ctx.fillStyle = '#22aa22';
ctx.fillRect(x, y + h*0.8, 8, 8);
ctx.fillRect(x + w - 12, y + h*0.8, 8, 8);
}

function drawEnemy(e) {
const x = e.x, y = e.y, w = e.w, h = e.h;
if (e.flash > 0 && Math.floor(e.flash * 2) % 2 === 0) return;

switch(e.type) {
case 'bike':
ctx.fillStyle = '#ff6633';
ctx.fillRect(x + 5, y, w - 10, h);
ctx.fillStyle = '#ff9966';
ctx.beginPath();
ctx.arc(x + 8, y + h, 6, 0, Math.PI*2);
ctx.arc(x + w - 8, y + h, 6, 0, Math.PI*2);
ctx.fill();
// Rider
ctx.fillStyle = '#ffcc00';
ctx.beginPath();
ctx.arc(x + w/2, y + 5, 7, 0, Math.PI*2);
ctx.fill();
break;
case 'car':
ctx.fillStyle = '#4488ff';
ctx.fillRect(x, y + 5, w, h - 10);
ctx.fillStyle = '#66aaff';
ctx.fillRect(x + 5, y, w - 10, h - 5);
ctx.fillStyle = '#333';
ctx.beginPath();
ctx.arc(x + 10, y + h, 6, 0, Math.PI*2);
ctx.arc(x + w - 10, y + h, 6, 0, Math.PI*2);
ctx.fill();
break;
case 'turtle':
ctx.fillStyle = '#88aa44';
ctx.beginPath();
ctx.ellipse(x + w/2, y + h/2, w/2, h/2.5, 0, 0, Math.PI*2);
ctx.fill();
ctx.fillStyle = '#669933';
ctx.fillRect(x - 3, y + h/2 - 4, 8, 8);
ctx.fillRect(x + w - 5, y + h/2 - 4, 8, 8);
// Shell pattern
ctx.fillStyle = '#557722';
ctx.beginPath();
ctx.arc(x + w/2, y + h/2, w/3, 0, Math.PI*2);
ctx.fill();
break;
case 'motorcycle':
ctx.fillStyle = '#cc44ff';
ctx.fillRect(x + 3, y, w - 6, h);
ctx.fillStyle = '#9933cc';
ctx.beginPath();
ctx.arc(x + 5, y + h, 5, 0, Math.PI*2);
ctx.arc(x + w - 5, y + h, 5, 0, Math.PI*2);
ctx.fill();
// Exhaust flame
ctx.fillStyle = '#ffaa00';
ctx.beginPath();
ctx.moveTo(x + (e.vx > 0 ? 0 : w), y + h/2);
ctx.lineTo(x + (e.vx > 0 ? -8 : w+8), y + h/2 - 3);
ctx.lineTo(x + (e.vx > 0 ? -8 : w+8), y + h/2 + 3);
ctx.fill();
break;
}

// HP bar
if (e.maxHp > 1) {
const bw = w;
ctx.fillStyle = '#333';
ctx.fillRect(x, y - 8, bw, 5);
ctx.fillStyle = e.hp > 1 ? '#44ff44' : '#ff4444';
ctx.fillRect(x, y - 8, bw * (e.hp / e.maxHp), 5);
}
}

function draw() {

ctx.fillStyle = '#1a2a1a';
ctx.fillRect(0, 0, W, H);

ctx.setLineDash([15, 15]);
ctx.beginPath();
ctx.moveTo(0, ly + H/7/2);
ctx.lineTo(W, ly + H/7/2);
ctx.stroke();
ctx.setLineDash([]);
}

// Bottom zone (danger)
ctx.fillStyle = '#3a1a1a';
ctx.fillRect(0, H - 25, W, 25);
ctx.fillStyle = '#ff4444';
ctx.font = 'bold 12px monospace';
ctx.textAlign = 'center';
ctx.fillText('⚠ DANGER ZONE ⚠', W/2, H - 8);

if (gameState === 'playing' || gameState === 'gameover') {
// Draw enemies
for (const e of enemies) drawEnemy(e);

// Draw bullets
ctx.fillStyle = '#ffff44';
for (const b of bullets) {
ctx.fillRect(b.x, b.y, b.w, b.h);
ctx.fillStyle = '#ffaa00';
ctx.fillRect(b.x - 1, b.y + b.h - 4, b.w + 2, 4);
ctx.fillStyle = '#ffff44';
}

// Draw player
drawFrog(player.x, player.y, player.w, player.h);

// Draw particles
for (const p of particles) {
const alpha = p.life / p.maxLife;
ctx.globalAlpha = alpha;
ctx.fillStyle = p.color;
ctx.fillRect(p.x, p.y, p.size, p.size);
}
ctx.globalAlpha = 1;

// HUD
ctx.fillStyle = '#fff';
ctx.font = 'bold 18px monospace';
ctx.textAlign = 'left';
ctx.fillText('SCORE: ' + score, 20, 30);
ctx.fillText('WAVE: ' + wave, 160, 30);
ctx.fillText('LIVES: ', 270, 30);
for (let i = 0; i < lives; i++) {
ctx.fillStyle = '#44ff44';
ctx.beginPath();
ctx.arc(360 + i * 25, 25, 8, 0, Math.PI*2);
ctx.fill();
}
ctx.fillStyle = '#aaa';
ctx.font = '14px monospace';
ctx.textAlign = 'right';
ctx.fillText('HIGH: ' + highScore, W - 20, 30);
}

if (gameState === 'start') {
// Overlay
ctx.fillStyle = 'rgba(0,0,0,0.7)';
ctx.fillRect(0, 0, W, H);

ctx.textAlign = 'center';
ctx.fillStyle = '#44ff44';
ctx.font = 'bold 48px monospace';
ctx.fillText('FROGGER SHOOTER', W/2, H/3 - 20);

ctx.fillStyle = '#fff';
ctx.font = '18px monospace';
ctx.fillText('Defend the road! Shoot enemies before they reach you.', W/2, H/3 + 40);

ctx.font = '16px monospace';
ctx.fillStyle = '#ccc';
ctx.fillText('← → / A D  —  Move', W/2, H/3 + 90);
ctx.fillText('SPACE / W / ↑  —  Shoot', W/2, H/3 + 120);

ctx.fillStyle = '#ffaa00';
ctx.font = 'bold 20px monospace';
ctx.fillText('Press SPACE to Start', W/2, H/3 + 180);

ctx.fillStyle = '#aaa';
ctx.font = '14px monospace';
ctx.fillText('High Score: ' + highScore, W/2, H/3 + 220);

// Draw a frog on start screen
drawFrog(W/2 - 20, H/3 + 240, 40, 35);
}

if (gameState === 'gameover') {
ctx.fillStyle = 'rgba(0,0,0,0.7)';
ctx.fillRect(0, 0, W, H);

ctx.textAlign = 'center';
ctx.fillStyle = '#ff4444';
ctx.font = 'bold 42px monospace';
ctx.fillText('GAME OVER', W/2, H/3);

ctx.fillStyle = '#fff';
ctx.font = '22px monospace';
ctx.fillText('Final Score: ' + score, W/2, H/3 + 50);
ctx.fillText('Wave Reached: ' + wave, W/2, H/3 + 85);
ctx.fillStyle = '#ffaa00';
ctx.fillText('High Score: ' + highScore, W/2, H/3 + 120);

ctx.fillStyle = '#44ff44';
ctx.font = 'bold 20px monospace';
ctx.fillText('Press SPACE to Restart', W/2, H/3 + 170);
}
}

function gameLoop() {
update();
draw();
requestAnimationFrame(gameLoop);
}
