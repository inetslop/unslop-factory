<script>
const canvas = document.getElementById('game');
const ctx = canvas.getContext('2d');

let W, H;
function resize() {
W = window.innerWidth; H = window.innerHeight;
canvas.width = W; canvas.height = H;
}
window.addEventListener('resize', resize);
resize();

let audioCtx = null;
function initAudio() {
if (!audioCtx) audioCtx = new (window.AudioContext || webkitAudioContext)();
}
function playSound(freq, dur, type, vol) {
if (!audioCtx) return;
const osc = audioCtx.createOscillator();
const gain = audioCtx.createGain();
osc.type = type || 'square';
osc.frequency.value = freq;
gain.gain.setValueAtTime(vol || 0.1, audioCtx.currentTime);
gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
osc.connect(gain); gain.connect(audioCtx.destination);
osc.start(); osc.stop(audioCtx.currentTime + dur);
}

let gameState = 'start';
let keys = {};
window.addEventListener('keydown', e => { keys[e.key] = true; if (gameState === 'playing') initAudio(); });
window.addEventListener('keyup', e => { keys[e.key] = false; });

document.getElementById('startBtn').addEventListener('click', startGame);
document.getElementById('restartBtn').addEventListener('click', startGame);

let player, enemies, playerBullets, enemyBullets, particles, cards, score, wave, waveTimer, maxHp, shieldTimer, gameLoop;

const CARD_TYPES = [
{ name: 'FIRE', color: '#ff6633', desc: 'Spread shot' },
{ name: 'DASH', color: '#66ccff', desc: 'Quick dodge' },
{ name: 'SHIELD', color: '#ffff33', desc: 'Invincible 2s' },
{ name: 'BOMB', color: '#ff33ff', desc: 'Clear bullets' },
{ name: 'HEAL', color: '#33ff66', desc: '+20 HP' }
];

function startGame() {
initAudio();
document.getElementById('startScreen').style.display = 'none';
document.getElementById('gameOverScreen').style.display = 'none';
player = { x: W/2, y: H - 80, r: 12, hp: 100, maxHp: 100, speed: 5 };
maxHp = 100;
enemies = []; playerBullets = []; enemyBullets = []; particles = [];
score = 0; wave = 1; waveTimer = 0; shieldTimer = 0;
cards = drawHand();
gameState = 'playing';
if (gameLoop) cancelAnimationFrame(gameLoop);
lastTime = performance.now();
gameLoop = requestAnimationFrame(update);
playSound(440, 0.2, 'sine', 0.15);
}

function drawHand() {
let hand = [];
for (let i = 0; i < 5; i++) {
hand.push({ type: Math.floor(Math.random() * 5), used: false });
}
return hand;
}

function drawNewCard() {
let idx = Math.floor(Math.random() * cards.length);
cards[idx] = { type: Math.floor(Math.random() * 5), used: false };
}

function spawnEnemy() {
const types = ['shooter', 'spiral', 'rusher'];
const type = types[Math.floor(Math.random() * types.length)];
const x = 60 + Math.random() * (W - 120);
let e = {
x, y: -30, r: 18, hp: 3 + wave, maxHp: 3 + wave, type,
fireTimer: 0, fireInterval: Math.max(60 - wave * 3, 20), angle: 0, speed: 1 + wave * 0.2
};
enemies.push(e);
}

function playCard(idx) {
if (idx < 0 || idx >= 5) return;
const card = cards[idx];
if (!card) return;
const t = card.type;
playSound(300 + t * 100, 0.15, 'square', 0.1);

if (t === 0) { // FIRE
for (let i = -2; i <= 2; i++) {
const a = -Math.PI/2 + i * 0.3;
playerBullets.push({ x: player.x, y: player.y, vx: Math.cos(a)*12, vy: Math.sin(a)*12, r: 5 });
}
} else if (t === 1) { // DASH
const dirs = [[-1,0],[1,0],[0,-1],[0,1]];
let dir = dirs[Math.floor(Math.random()*4)];
player.x += dir[0] * 80;
player.y += dir[1] * 80;
player.x = Math.max(20, Math.min(W-20, player.x));
player.y = Math.max(20, Math.min(H-20, player.y));
for (let i = 0; i < 8; i++) {
particles.push({ x: player.x, y: player.y, vx: (Math.random()-0.5)*6, vy: (Math.random()-0.5)*6, life: 15, color: '#66ccff' });
}
} else if (t === 2) { // SHIELD
shieldTimer = 120;
for (let i = 0; i < 12; i++) {
particles.push({ x: player.x + Math.cos(i/12*Math.PI*2)*25, y: player.y + Math.sin(i/12*Math.PI*2)*25, vx: 0, vy: 0, life: 30, color: '#ffff33' });
}
} else if (t === 3) { // BOMB
for (let b of enemyBullets) {
particles.push({ x: b.x, y: b.y, vx: (Math.random()-0.5)*4, vy: (Math.random()-0.5)*4, life: 20, color: '#ff33ff' });
}
enemyBullets = [];
playSound(100, 0.4, 'sawtooth', 0.2);
} else if (t === 4) { // HEAL
player.hp = Math.min(player.hp + 20, maxHp);
for (let i = 0; i < 6; i++) {
particles.push({ x: player.x, y: player.y, vx: (Math.random()-0.5)*3, vy: -Math.random()*3, life: 25, color: '#33ff66' });
}
}
card.used = true;
drawNewCard();
}

function update(time) {
gameLoop = requestAnimationFrame(update);
const dt = Math.min((time - (lastTime||time)) / 16.67, 2);
lastTime = time;

if (gameState !== 'playing') return;

// Wave logic
waveTimer++;
if (waveTimer % (Math.max(180 - wave*15, 60)) === 0) {
spawnEnemy();
}
if (waveTimer > 600) { wave++; waveTimer = 0; playSound(200, 0.3, 'triangle', 0.15); }

// Player movement
let dx = 0, dy = 0;
if (keys['w'] || keys['W'] || keys['ArrowUp']) dy -= 1;
if (keys['s'] || keys['S'] || keys['ArrowDown']) dy += 1;
if (keys['a'] || keys['A'] || keys['ArrowLeft']) dx -= 1;
if (keys['d'] || keys['D'] || keys['ArrowRight']) dx += 1;
if (dx !== 0 && dy !== 0) { dx *= 0.707; dy *= 0.707; }
player.x += dx * player.speed * dt;
player.y += dy * player.speed * dt;
player.x = Math.max(15, Math.min(W-15, player.x));
player.y = Math.max(15, Math.min(H-15, player.y));

// Shield timer
if (shieldTimer > 0) shieldTimer -= dt;

// Enemy AI
for (let e of enemies) {
const angleToPlayer = Math.atan2(player.y - e.y, player.x - e.x);
if (e.type === 'rusher') {
e.x += Math.cos(angleToPlayer) * e.speed * 1.5 * dt;
e.y += Math.sin(angleToPlayer) * e.speed * 1.5 * dt;
} else {
if (e.y < 150) e.y += e.speed * dt;
else { e.x += Math.cos(e.angle) * e.speed * 0.5 * dt; e.angle += 0.02 * dt; }
}
e.x = Math.max(20, Math.min(W-20, e.x));
e.y = Math.max(-30, Math.min(H-30, e.y));

// Enemy fire
e.fireTimer -= dt;
if (e.fireTimer <= 0) {
e.fireTimer = e.fireInterval;
if (e.type === 'spiral') {
for (let i = 0; i < 5; i++) {
const a = e.angle + i * Math.PI*2/5;
enemyBullets.push({ x: e.x, y: e.y, vx: Math.cos(a)*3.5, vy: Math.sin(a)*3.5, r: 6 });
}
e.angle += 0.4;
} else if (e.type === 'shooter') {
for (let i = -1; i <= 1; i++) {
const a = angleToPlayer + i * 0.25;
enemyBullets.push({ x: e.x, y: e.y, vx: Math.cos(a)*4, vy: Math.sin(a)*4, r: 6 });
}
} else {
const a = angleToPlayer;
for (let i = -3; i <= 3; i++) {
enemyBullets.push({ x: e.x, y: e.y, vx: Math.cos(a+i*0.15)*4.5, vy: Math.sin(a+i*0.15)*4.5, r: 7 });
}
}
playSound(200 + Math.random()*100, 0.08, 'square', 0.03);
}
}

// Player bullets
for (let i = playerBullets.length-1; i >= 0; i--) {
const b = playerBullets[i];
b.x += b.vx * dt; b.y += b.vy * dt;
if (b.x < -20 || b.x > W+20 || b.y < -20 || b.y > H+20) { playerBullets.splice(i,1); continue; }
// Hit enemies
for (let j = enemies.length-1; j >= 0; j--) {
const e = enemies[j];
const ddx = b.x - e.x, ddy = b.y - e.y;
if (ddx*ddx + ddy*ddy < (b.r+e.r)*(b.r+e.r)) {
e.hp--;
playerBullets.splice(i,1);
for (let p = 0; p < 4; p++) particles.push({ x: e.x, y: e.y, vx: (Math.random()-0.5)*5, vy: (Math.random()-0.5)*5, life: 15, color: '#ff6633' });
playSound(500, 0.05, 'square', 0.05);
if (e.hp <= 0) {
score += 10;
for (let p = 0; p < 12; p++) particles.push({ x: e.x, y: e.y, vx: (Math.random()-0.5)*8, vy: (Math.random()-0.5)*8, life: 30, color: '#ffaa00' });
enemies.splice(j,1);
playSound(600, 0.2, 'triangle', 0.15);
}
break;
}
}
}

// Enemy bullets
for (let i = enemyBullets.length-1; i >= 0; i--) {
const b = enemyBullets[i];
b.x += b.vx * dt; b.y += b.vy * dt;
if (b.x < -30 || b.x > W+30 || b.y < -30 || b.y > H+30) { enemyBullets.splice(i,1); continue; }
// Hit player
const ddx = b.x - player.x, ddy = b.y - player.y;
if (ddx*ddx + ddy*ddy < (b.r+player.r)*(b.r+player.r)) {
if (shieldTimer > 0) { enemyBullets.splice(i,1); continue; }
player.hp -= 10;
enemyBullets.splice(i,1);
for (let p = 0; p < 6; p++) particles.push({ x: player.x, y: player.y, vx: (Math.random()-0.5)*5, vy: (Math.random()-0.5)*5, life: 20, color: '#ff3333' });
playSound(150, 0.2, 'sawtooth', 0.2);
if (player.hp <= 0) { gameOver(); }
}
}

// Particles
for (let i = particles.length-1; i >= 0; i--) {
const p = particles[i];
p.x += p.vx * dt; p.y += p.vy * dt; p.life -= dt;
if (p.life <= 0) particles.splice(i,1);
}

// Score by survival
score += 0.02 * dt;

render();
}

let lastTime = 0;

function gameOver() {
gameState = 'gameover';
document.getElementById('gameOverScreen').style.display = 'flex';
document.getElementById('score').textContent = 'Score: ' + Math.floor(score);
playSound(100, 0.5, 'sawtooth', 0.3);
}

function render() {
ctx.fillStyle = '#0a0a1a';
ctx.fillRect(0,0,W,H);

// Grid
ctx.strokeStyle = 'rgba(30,30,80,0.3)';
ctx.lineWidth = 1;
for (let x = 0; x < W; x += 50) { ctx.beginPath(); ctx.moveTo(x,0); ctx.lineTo(x,H); ctx.stroke(); }
for (let y = 0; y < H; y += 50) { ctx.beginPath(); ctx.moveTo(0,y); ctx.lineTo(W,y); ctx.stroke(); }

// Particles
for (const p of particles) {
ctx.globalAlpha = p.life / 20;
ctx.fillStyle = p.color;
ctx.beginPath(); ctx.arc(p.x, p.y, 3, 0, Math.PI*2); ctx.fill();
}
ctx.globalAlpha = 1;

// Enemy bullets
for (const b of enemyBullets) {
ctx.fillStyle = '#ff3366';
ctx.shadowColor = '#ff3366'; ctx.shadowBlur = 8;
ctx.beginPath(); ctx.arc(b.x, b.y, b.r, 0, Math.PI*2); ctx.fill();
}
ctx.shadowBlur = 0;

// Player bullets
for (const b of playerBullets) {
ctx.fillStyle = '#66ffcc';
ctx.shadowColor = '#66ffcc'; ctx.shadowBlur = 6;
ctx.beginPath(); ctx.arc(b.x, b.y, b.r, 0, Math.PI*2); ctx.fill();
}
ctx.shadowBlur = 0;

// Enemies
for (const e of enemies) {
ctx.fillStyle = e.type === 'rusher' ? '#ff6633' : e.type === 'spiral' ? '#cc66ff' : '#ff9933';
ctx.shadowColor = ctx.fillStyle; ctx.shadowBlur = 10;
ctx.beginPath(); ctx.arc(e.x, e.y, e.r, 0, Math.PI*2); ctx.fill();
ctx.shadowBlur = 0;
// HP bar
const bw = 30;
ctx.fillStyle = '#333';
ctx.fillRect(e.x-bw/2, e.y-e.r-10, bw, 4);
ctx.fillStyle = '#ff3366';
ctx.fillRect(e.x-bw/2, e.y-e.r-10, bw*(e.hp/e.maxHp), 4);
}

// Player
if (shieldTimer > 0) {
ctx.strokeStyle = '#ffff33'; ctx.lineWidth = 3;
ctx.shadowColor = '#ffff33'; ctx.shadowBlur = 15;
ctx.beginPath(); ctx.arc(player.x, player.y, player.r+8, 0, Math.PI*2); ctx.stroke();
ctx.shadowBlur = 0;
}
ctx.fillStyle = '#ffffff';
ctx.shadowColor = '#66aaff'; ctx.shadowBlur = 12;
ctx.beginPath(); ctx.arc(player.x, player.y, player.r, 0, Math.PI*2); ctx.fill();
ctx.shadowBlur = 0;

// HP Bar
const barW = 200, barH = 20;
ctx.fillStyle = '#222';
ctx.fillRect(W/2-barW/2, 10, barW, barH);
ctx.fillStyle = player.hp > 30 ? '#33ff66' : '#ff3333';
ctx.fillRect(W/2-barW/2, 10, barW*(player.hp/maxHp), barH);
ctx.strokeStyle = '#666'; ctx.lineWidth = 1;
ctx.strokeRect(W/2-barW/2, 10, barW, barH);
ctx.fillStyle = '#fff'; ctx.font = '12px monospace'; ctx.textAlign = 'center';
ctx.fillText(Math.ceil(player.hp) + '/' + maxHp, W/2, 25);

// Score & Wave
ctx.fillStyle = '#66ff66'; ctx.font = '16px monospace'; ctx.textAlign = 'left';
ctx.fillText('SCORE: ' + Math.floor(score), 15, H - 15);
ctx.fillStyle = '#ff9933';
ctx.fillText('WAVE: ' + wave, 15, H - 35);

// Cards
const cardW = 90, cardH = 70, gap = 8;
const totalW = 5 * (cardW + gap) - gap;
const startX = W/2 - totalW/2;
const cardY = H - cardH - 10;

for (let i = 0; i < 5; i++) {
const cx = startX + i * (cardW + gap);
const ct = CARD_TYPES[cards[i].type];
ctx.fillStyle = cards[i].used ? '#1a1a2a' : 'rgba(20,20,40,0.9)';
ctx.fillRect(cx, cardY, cardW, cardH);
ctx.strokeStyle = ct.color; ctx.lineWidth = 2;
ctx.strokeRect(cx, cardY, cardW, cardH);

if (!cards[i].used) {
ctx.fillStyle = ct.color; ctx.font = 'bold 11px monospace'; ctx.textAlign = 'center';
ctx.fillText(ct.name, cx + cardW/2, cardY + 20);
ctx.fillStyle = '#aaa'; ctx.font = '9px monospace';
ctx.fillText(ct.desc, cx + cardW/2, cardY + 35);
} else {
ctx.fillStyle = '#444'; ctx.font = '10px monospace'; ctx.textAlign = 'center';
ctx.fillText('USED', cx + cardW/2, cardY + 35);
}

// Key number
ctx.fillStyle = '#fff'; ctx.font = 'bold 14px monospace'; ctx.textAlign = 'left';
ctx.fillText((i+1) + '', cx + 5, cardY + 18);
}

// Card names at bottom
ctx.fillStyle = '#555'; ctx.font = '9px monospace'; ctx.textAlign = 'center';
for (let i = 0; i < 5; i++) {
const cx = startX + i * (cardW + gap) + cardW/2;
ctx.fillText(CARD_TYPES[cards[i].type].name, cx, cardY + cardH - 5);
}
}

// Key handling for cards
window.addEventListener('keydown', e => {
if (gameState !== 'playing') return;
const n = parseInt(e.key);
if (n >= 1 && n <= 5) playCard(n-1);
});

function loop(time) {
if (gameState === 'playing') update(time);
else render();
requestAnimationFrame(loop);
}
lastTime = performance.now();
loop(lastTime);
</script>
