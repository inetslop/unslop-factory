function update(dt){
  time+=dt;
  // waves
  waveTimer+=dt;
  if(waveTimer>=25){ waveTimer=0; wave++; }

  // spawn
  spawnTimer-=dt;
  const interval=Math.max(0.45, 1.6 - wave*0.13);
  if(spawnTimer<=0){ spawnTimer=interval; spawnEnemy(); }

  // enemies
  for(let i=enemies.length-1;i>=0;i--){
    const e=enemies[i];
    e.row += e.speed*dt;
    // drift toward target column
    const d=e.targetCol-e.col;
    if(Math.abs(d)>0.02){ e.col += Math.sign(d)*Math.min(1.4,Math.abs(d))*dt; }
    else { if(Math.random()<dt*0.6){ const nc=Math.max(0,Math.min(COLS-1,e.targetCol+(Math.random()<0.5?-1:1))); e.targetCol=nc; } }
    if(e.row>=ROWS-0.35){ enemies.splice(i,1); leak(); if(gameOver) break; }
  }

  // frogs shoot
  for(const k in frogs){
    const f=frogs[k];
    f.cd-=dt;
    if(f.cd>0) continue;
    const c=frogCenter(f);
    const rangePx=FROG.rangeRows[f.level-1]*ch;
    let bestT=null, bd=1e9;
    for(const e of enemies){
      const p=enemyPx(e); const dx=p.x-c.x, dy=p.y-c.y; const dist=Math.hypot(dx,dy);
      if(dist<=rangePx && dist<bd){ bd=dist; bestT=e; }
    }

    } else if(p.life>2){ projectiles.splice(i,1); }
  }

  // remove dead enemies not from projectile (safety)
  for(let i=enemies.length-1;i>=0;i--){ if(enemies[i].hp<=0) killEnemy(enemies[i]); }
}

function drawHover(){
  const c=Math.floor(mouse.x/cw), r=Math.floor(mouse.y/ch);
  if(c<0||c>=COLS||r<0||r>=ROWS) return;
  if(isBuildRow(r)){
    const f=frogAt(c,r);
    if(!f){
      if(coins>=BUILD_COST){ drawFrog({col:c,row:r,level:1},true); }
      ctx.fillStyle= coins>=BUILD_COST?'#7CFC00':'#ff5252';
      ctx.font='bold '+(ch*0.34)+'px Arial'; ctx.textAlign='center';
      ctx.fillText('🪙'+BUILD_COST, (c+0.5)*cw, r*ch+ch*0.4);
    } else {
      if(f.level<3 && coins>=UP_COST[f.level-1]){
        ctx.strokeStyle='#7CFC00'; ctx.lineWidth=3; ctx.strokeRect(c*cw+2,r*ch+2,cw-4,ch-4);
        ctx.fillStyle='#7CFC00'; ctx.font='bold '+(ch*0.3)+'px Arial'; ctx.textAlign='center';
        ctx.fillText('⬆ '+UP_COST[f.level-1], (c+0.5)*cw, r*ch+ch*0.4);
      } else {
        ctx.fillStyle='#ffe08a'; ctx.font='bold '+(ch*0.3)+'px Arial'; ctx.textAlign='center';
        ctx.fillText('MAX', (c+0.5)*cw, r*ch+ch*0.4);
      }
    }
  }
}
