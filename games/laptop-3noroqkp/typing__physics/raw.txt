<script>
const canvas = document.getElementById('game');
const ctx = canvas.getContext('2d');

let W, H;
function resize() {
    W = canvas.width = window.innerWidth;
    H = canvas.height = window.innerHeight;
}
window.addEventListener('resize', resize);
resize();

// Audio
let audioCtx = null;
function initAudio() {
    if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
}
function playSound(freq, dur, type, vol) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = type || 'square';
    osc.frequency.value = freq;
    gain.gain.setValueAtTime(vol || 0.1, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + dur);
}
function soundType() { playSound(440, 0.05, 'square', 0.05); }
function soundComplete() { playSound(880, 0.15, 'sine', 0.12); playSound(1320, 0.15, 'sine', 0.08); }
function soundFail() { playSound(120, 0.4, 'sawtooth', 0.15); }
function soundDestroy() { playSound(600, 0.1, 'triangle', 0.1); }

// Word lists
const words = [
    "fire","star","moon","wind","rain","snow","ice","ocean","cliff","stone",
    "light","sound","force","mass","time","space","energy","matter","motion","speed",
    "gravity","vector","scalar","momentum","friction","tension","spring","damper","pivot","torque",
    "angular","velocity","inertia","impulse","elastic","plastic","rigid","fluid","solid","gas",
    "photon","electron","proton","neutron","quark","gluon","boson","muon","tau","charm",
    "quantum","relativity","singularity","horizon","warp","flux","field","wave","pulse","echo"
];

// Game state
let gameState = 'start'; // start, playing, gameover
let score = 0;
let lives = 3;
let level = 1;
let wordsOnScreen = [];
let particles = [];
let inputBuffer = "";
let targetWord = null;
let spawnTimer = 0;
let spawnInterval = 2500;
let shakeTimer = 0;
let shakeIntensity = 0;
let comboCount = 0;
let lastScoreTime = 0;
let gameTime = 0;

// Physics constants
const GRAVITY = 300; // px/s^2
const BOUNCE_DAMPING = 0.85;
const WORD_MIN_SPEED_X = 30;
const WORD_MAX_SPEED_X = 120;

class Word {
    constructor(text) {
        this.text = text;
        this.x = Math.random() * (W - 200) + 50;
        this.y = -40;
        this.vx = (Math.random() > 0.5 ? 1 : -1) * (WORD_MIN_SPEED_X + Math.random() * (WORD_MAX_SPEED_X - WORD_MIN_SPEED_X));
        this.vy = 20 + Math.random() * 60;
        this.alive = true;
        this.matched = 0;
        this.alpha = 1;
        this.scale = 1;
        this.rotVel = (Math.random() - 0.5) * 0.02;
        this.rot = 0;
    }
    update(dt) {
        this.vy += GRAVITY * dt;
        this.x += this.vx * dt;
        this.y += this.vy * dt;
        this.rot += this.rotVel;

        // Bounce off walls
        const w = this.text.length * 14 + 20;
        if (this.x < 10) { this.x = 10; this.vx = Math.abs(this.vx) * BOUNCE_DAMPING; }
        if (this.x + w > W - 10) { this.x = W - w - 10; this.vx = -Math.abs(this.vx) * BOUNCE_DAMPING; }

        // Hit ground
        if (this.y > H - 60) {
            this.alive = false;
            return 'ground';
        }
        return null;
    }
    draw(ctx, isTarget) {
        ctx.save();
        ctx.translate(this.x + this.text.length * 7, this.y);
        ctx.rotate(this.rot * 0.3);

        const fontSize = 24;
        ctx.font = `bold ${fontSize}px 'Courier New', monospace`;

        for (let i = 0; i < this.text.length; i++) {
            const charX = -this.text.length * 7 + i * 14 + 7;
            if (isTarget && i < this.matched) {
                ctx.fillStyle = '#00ff88';
                ctx.shadowColor = '#00ff88';
                ctx.shadowBlur = 12;
            } else if (isTarget && i === this.matched) {
                ctx.fillStyle = '#ffff00';
                ctx.shadowColor = '#ffff00';
                ctx.shadowBlur = 8;
            } else {
                ctx.fillStyle = `rgba(255, 255, 255, ${this.alpha * 0.9})`;
                ctx.shadowColor = 'transparent';
                ctx.shadowBlur = 0;
            }
            ctx.fillText(this.text[i], charX, 12);
        }

        // Underline for target
        if (isTarget) {
            ctx.strokeStyle = '#00ff88';
            ctx.lineWidth = 2;
            ctx.shadowColor = '#00ff88';
            ctx.shadowBlur = 6;
            const lw = this.text.length * 14;
            ctx.beginPath();
            ctx.moveTo(-lw / 2 + 7, 20);
            ctx.lineTo(-lw / 2 + 7 + (this.matched / this.text.length) * lw, 20);
            ctx.stroke();
        }

        ctx.restore();
    }
}

class Particle {
    constructor(x, y, color) {
        this.x = x;
        this.y = y;
        const angle = Math.random() * Math.PI * 2;
        const speed = 100 + Math.random() * 300;
        this.vx = Math.cos(angle) * speed;
        this.vy = Math.sin(angle) * speed - 50;
        this.life = 1;
        this.decay = 1 + Math.random() * 2;
        this.size = 2 + Math.random() * 4;
        this.color = color;
    }
    update(dt) {
        this.vy += GRAVITY * 0.5 * dt;
        this.x += this.vx * dt;
        this.y += this.vy * dt;
        this.life -= this.decay * dt;
        return this.life > 0;
    }
    draw(ctx) {
        ctx.globalAlpha = Math.max(0, this.life);
        ctx.fillStyle = this.color;
        ctx.shadowColor = this.color;
        ctx.shadowBlur = 8;
        ctx.fillRect(this.x - this.size / 2, this.y - this.size / 2, this.size, this.size);
        ctx.globalAlpha = 1;
    }
}

function spawnWord() {
    const word = words[Math.floor(Math.random() * words.length)];
    wordsOnScreen.push(new Word(word));
}

function destroyWord(word) {
    word.alive = false;
    const cx = word.x + word.text.length * 7;
    const cy = word.y;
    for (let i = 0; i < 25; i++) {
        particles.push(new Particle(cx, cy, Math.random() > 0.3 ? '#00ff88' : '#ffffff'));
    }
    score += word.text.length * 10 + comboCount * 5;
    comboCount++;
    lastScoreTime = gameTime;
    soundDestroy();
    if (targetWord === word) { targetWord = null; inputBuffer = ""; }
}

function loseLife() {
    lives--;
    comboCount = 0;
    shakeTimer = 0.3;
    shakeIntensity = 8;
    soundFail();
    if (lives <= 0) {
        gameState = 'gameover';
    }
}

function startGame() {
    score = 0;
    lives = 3;
    level = 1;
    wordsOnScreen = [];
    particles = [];
    inputBuffer = "";
    targetWord = null;
    spawnTimer = 0;
    spawnInterval = 2500;
    comboCount = 0;
    gameTime = 0;
    gameState = 'playing';
}

// Input handling
window.addEventListener('keydown', (e) => {
    if (gameState === 'start') {
        initAudio();
        startGame();
        return;
    }
    if (gameState === 'gameover') {
        if (e.key === 'Enter' || e.key === ' ') {
            startGame();
        }
        return;
    }
    if (gameState !== 'playing') return;

    if (e.key === 'Backspace') {
        inputBuffer = inputBuffer.slice(0, -1);
        if (targetWord) targetWord.matched = inputBuffer.length;
        e.preventDefault();
        return;
    }

    if (e.key.length === 1 && /[a-z]/i.test(e.key)) {
        const key = e.key.toLowerCase();
        initAudio();

        // Try to find a matching word
        let bestMatch = null;
        let bestLen = 0;

        for (const w of wordsOnScreen) {
            if (!w.alive) continue;
            const currentProgress = targetWord === w ? w.matched : 0;
            if (w.text[currentProgress] === key) {
                if (currentProgress + 1 > bestLen) {
                    bestLen = currentProgress + 1;
                    bestMatch = w;
                }
            }
        }

        if (bestMatch) {
            targetWord = bestMatch;
            inputBuffer += key;
            targetWord.matched = inputBuffer.length;
            soundType();

            if (inputBuffer === targetWord.text) {
                destroyWord(targetWord);
            }
        } else {
            // Wrong key - reset
            inputBuffer = "";
            if (targetWord) targetWord.matched = 0;
            playSound(200, 0.1, 'sawtooth', 0.05);
        }
    }
});

// Rendering
function drawBackground() {
    ctx.fillStyle = '#0a0a1a';
    ctx.fillRect(0, 0, W, H);

    // Grid
    ctx.strokeStyle = 'rgba(30, 40, 80, 0.5)';
    ctx.lineWidth = 1;
    const gridSize = 60;
    for (let x = 0; x < W; x += gridSize) {
        ctx.beginPath(); ctx.moveTo(x, 0); ctx.lineTo(x, H); ctx.stroke();
    }
    for (let y = 0; y < H; y += gridSize) {
        ctx.beginPath(); ctx.moveTo(0, y); ctx.lineTo(W, y); ctx.stroke();
    }

    // Danger zone at bottom
    const dangerGrad = ctx.createLinearGradient(0, H - 60, 0, H);
    dangerGrad.addColorStop(0, 'rgba(255, 0, 0, 0)');
    dangerGrad.addColorStop(1, 'rgba(255, 0, 0, 0.3)');
    ctx.fillStyle = dangerGrad;
    ctx.fillRect(0, H - 60, W, 60);

    // Bottom line
    ctx.strokeStyle = '#ff3333';
    ctx.lineWidth = 2;
    ctx.shadowColor = '#ff3333';
    ctx.shadowBlur = 10;
    ctx.beginPath();
    ctx.moveTo(0, H - 60);
    ctx.lineTo(W, H - 60);
    ctx.stroke();
    ctx.shadowBlur = 0;
}

function drawHUD() {
    ctx.font = 'bold 20px Courier New';
    ctx.fillStyle = '#ffffff';
    ctx.textAlign = 'left';
    ctx.fillText(`SCORE: ${score}`, 20, 35);
    ctx.fillText(`LEVEL: ${level}`, 20, 65);

    // Lives as hearts
    ctx.textAlign = 'right';
    let heartStr = '';
    for (let i = 0; i < lives; i++) heartStr += '♥ ';
    ctx.fillStyle = '#ff4466';
    ctx.fillText(heartStr, W - 20, 35);

    // Combo
    if (comboCount > 1) {
        ctx.textAlign = 'center';
        ctx.fillStyle = '#ffff00';
        ctx.font = 'bold 18px Courier New';
        ctx.fillText(`COMBO x${comboCount}`, W / 2, 35);
    }

    // Input buffer display
    if (inputBuffer.length > 0 && gameState === 'playing') {
        ctx.textAlign = 'center';
        ctx.font = 'bold 28px Courier New';
        ctx.fillStyle = '#00ff88';
        ctx.shadowColor = '#00ff88';
        ctx.shadowBlur = 10;
        ctx.fillText(inputBuffer, W / 2, H - 20);
        ctx.shadowBlur = 0;
    }
}

function drawStartScreen() {
    drawBackground();
    ctx.textAlign = 'center';
    ctx.fillStyle = '#ffffff';
    ctx.font = 'bold 48px Courier New';
    ctx.shadowColor = '#00ff88';
    ctx.shadowBlur = 20;
    ctx.fillText('PHYSICS TYPING', W / 2, H / 2 - 60);
    ctx.shadowBlur = 0;

    ctx.font = '18px Courier New';
    ctx.fillStyle = '#aaaaaa';
    const lines = [
        'Words fall with gravity and bounce off walls!',
        'Type them before they hit the danger zone.',
        '',
        'CONTROLS:',
        '  Type letters to match falling words',
        '  Backspace to undo',
        '  Complete words to destroy them',
        '',
        'PRESS ANY KEY TO START'
    ];
    lines.forEach((line, i) => {
        ctx.fillText(line, W / 2, H / 2 + i * 28);
    });

    // Animated demo words
    const t = Date.now() / 1000;
    for (let i = 0; i < 5; i++) {
        const dx = W / 2 + Math.sin(t + i * 1.3) * 150;
        const dy = H / 2 - 120 + Math.cos(t * 0.7 + i) * 30;
        ctx.font = '16px Courier New';
        ctx.fillStyle = `rgba(0, 255, 136, ${0.4 + Math.sin(t * 2 + i) * 0.3})`;
        ctx.fillText(words[i], dx, dy);
    }
}

function drawGameOver() {
    ctx.fillStyle = 'rgba(0, 0, 0, 0.7)';
    ctx.fillRect(0, 0, W, H);
    ctx.textAlign = 'center';
    ctx.fillStyle = '#ff4466';
    ctx.font = 'bold 48px Courier New';
    ctx.fillText('GAME OVER', W / 2, H / 2 - 40);

    ctx.fillStyle = '#ffffff';
    ctx.font = '24px Courier New';
    ctx.fillText(`Final Score: ${score}`, W / 2, H / 2 + 10);
    ctx.fillText(`Level Reached: ${level}`, W / 2, H / 2 + 45);

    ctx.fillStyle = '#aaaaaa';
    ctx.font = '18px Courier New';
    ctx.fillText('Press ENTER to restart', W / 2, H / 2 + 100);
}

// Game loop
let lastTime = 0;
function gameLoop(timestamp) {
    const dt = Math.min((timestamp - lastTime) / 1000, 0.05);
    lastTime = timestamp;

    if (gameState === 'start') {
        drawStartScreen();
        requestAnimationFrame(gameLoop);
        return;
    }

    if (gameState === 'gameover') {
        drawBackground();
        // Update particles still
        particles = particles.filter(p => p.update(dt));
        particles.forEach(p => p.draw(ctx));
        drawGameOver();
        ctx.textAlign = 'center';
        requestAnimationFrame(gameLoop);
        return;
    }

    // Playing state
    gameTime += dt;

    // Difficulty scaling
    const newLevel = Math.floor(gameTime / 30) + 1;
    if (newLevel > level) {
        level = newLevel;
        spawnInterval = Math.max(800, 2500 - (level - 1) * 200);
    }

    // Combo decay
    if (gameTime - lastScoreTime > 3) comboCount = 0;

    // Spawn words
    spawnTimer += dt * 1000;
    if (spawnTimer >= spawnInterval) {
        spawnTimer = 0;
        spawnWord();
    }

    // Update words
    for (const w of wordsOnScreen) {
        if (!w.alive) continue;
        const result = w.update(dt);
        if (result === 'ground') {
            loseLife();
            // Particles on ground hit
            for (let i = 0; i < 15; i++) {
                particles.push(new Particle(w.x + w.text.length * 7, H - 60, '#ff4466'));
            }
        }
    }
    wordsOnScreen = wordsOnScreen.filter(w => w.alive);

    // Update particles
    particles = particles.filter(p => p.update(dt));

    // Screen shake
    let sx = 0, sy = 0;
    if (shakeTimer > 0) {
        shakeTimer -= dt;
        sx = (Math.random() - 0.5) * shakeIntensity;
        sy = (Math.random() - 0.5) * shakeIntensity;
    }

    ctx.save();
    ctx.translate(sx, sy);

    drawBackground();

    // Draw words
    for (const w of wordsOnScreen) {
        w.draw(ctx, targetWord === w);
    }

    // Draw particles
    for (const p of particles) {
        p.draw(ctx);
    }

    ctx.shadowBlur = 0;
    drawHUD();

    ctx.restore();

    requestAnimationFrame(gameLoop);
}

requestAnimationFrame(gameLoop);
