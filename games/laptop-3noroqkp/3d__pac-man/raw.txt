      else { e.moving=false; e.p=0; e.dirx=0; e.diry=0; e.nx=e.gx; e.ny=e.gy; return; }
    }
  } else {
    const nd=chooser();
    if(nd){ e.dirx=nd[0]; e.diry=nd[1]; e.nx=e.gx+nd[0]; e.ny=e.gy+nd[1]; e.moving=true; e.p=0; }
  }
}
function pacChooser(){
  const d=lastHeld();
  if(d && isWalk(pac.gx+d.x,pac.gy+d.y)) return [d.x,d.y];
  if((pac.dirx||pac.diry) && isWalk(pac.gx+pac.dirx,pac.gy+pac.diry)) return [pac.dirx,pac.diry];
  return null;
}
function ghostTarget(g){
  const p=posOf(pac);
  switch(g.i){
    case 0: return [Math.round(p.x),Math.round(p.y)];
    case 1: return [Math.round(p.x+pac.dirx*3),Math.round(p.y+pac.diry*3)];
    case 2: { const b=posOf(ghosts[0]); const ax=p.x+pac.dirx*2, ay=p.y+pac.diry*2; return [Math.round(2*b.x-ax),Math.round(2*b.y-ay)]; }
    case 3: { const ppos=Math.round(p.x)+Math.round(p.y); if(Math.abs(p.x-g.gx)+Math.abs(p.y-g.gy)>8) return [Math.round(p.x),Math.round(p.y)]; return [1,H-2]; }
  }
}
function ghostChooser(g){
  const fright = frightTimer>0 && !g.eaten;
  let target; if(g.eaten) target=[spawns[g.i].x,spawns[g.i].y]; else if(fright) target=[Math.round(posOf(pac).x),Math.round(posOf(pac).y)]; else target=ghostTarget(g);
  const cur=[g.dirx,g.diry];
  let opts=openN(g.gx,g.gy);
  const allowReverse = g.eaten || !(cur[0]||cur[1]);
  if(!allowReverse){ const rev=[-cur[0],-cur[1]]; if(opts.length>1) opts=opts.filter(o=>!(o[0]===rev[0]&&o[1]===rev[1])); }
  if(!opts.length) return null;
  let best=null,bs=Infinity; const tx=target[0],ty=target[1];
  for(const o of opts){ const nx=g.gx+o[0],ny=g.gy+o[1]; let d=Math.abs(nx-tx)+Math.abs(ny-ty); if(fright) d=-d; d+=Math.random()*(fright?1.5:0.4); if(d<bs){bs=d;best=o;} }
  return best;
}

function ghostSpeed(){ return Math.min(4.2+(level-1)*0.25, 5.0); }

function update(dt){
  if(state==='ready'){ readyTimer-=dt; if(readyTimer<=0){ state='playing'; } return; }
  if(state!=='playing') { if(state==='dying'){ dieTimer-=dt; if(dieTimer<=0){ if(lives>0){ resetPositions(); state='playing'; } else { state='over'; saveHi(); } } } shake=Math.max(0,shake-dt*3); return; }

  // pac
  pac.speed=5.0; moveNode(pac,dt,pacChooser);
  const pp=posOf(pac);
  const cx=Math.round(pp.x), cy=Math.round(pp.y);
  if(isWalk(cx,cy)){
    if(pellet[cy][cx]===1){ pellet[cy][cx]=0; score+=10; remaining--; sfxWaka(); }
    else if(pellet[cy][cx]===2){ pellet[cy][cx]=0; score+=50; remaining--; combo=0; frightTimer=Math.max(3,8-level); sfxPower(); }
  }

  // ghosts
  const gs=ghostSpeed();
  for(const g of ghosts){
    if(!g.active){ g.spawnTimer-=dt; if(g.spawnTimer<=0) g.active=true; continue; }
    if(g.eaten){ g.speed=Math.min(gs+2.5,7); moveNode(g,dt,ghostChooser); if(g.gx===spawns[g.i].x&&g.gy===spawns[g.i].y){ g.eaten=false; g.moving=false; g.dirx=0;g.diry=0; g.nx=g.gx;g.ny=g.gy; } continue; }
    const fright=frightTimer>0; g.speed = fright?3.0:gs; moveNode(g,dt,ghostChooser);
  }

  // collisions
  for(const g of ghosts){
    if(!g.active||g.eaten) continue;
    const gp=posOf(g); const dx=gp.x-pp.x, dy=gp.y-pp.y;
    if(dx*dx+dy*dy < 0.45*0.45){
      if(frightTimer>0){ // eat ghost
        g.eaten=true; const pts=200<<Math.min(combo,3); score+=pts; combo++; addPop(gp.x,gp.y,'+'+pts,'#8dff9a'); sfxGhost();
      } else { loseLife(); break; }
    }
  }

  if(frightTimer>0) frightTimer-=dt; else combo=0;

  // win
  if(remaining<=0){ level++; newLevel(); remaining=pelletCount(); state='ready'; readyTimer=1.6; sfxLevel(); }
}
function loseLife(){ lives--; sfxDeath(); shake=1; deathPos={x:posOf(pac).x,y:posOf(pac).y}; state='dying'; dieTimer=1.2; if(lives<=0){ /*over handled after anim*/ } }
