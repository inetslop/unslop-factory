  function renderWord(el, word, progress) {
    let html = '';
    for (let i = 0; i < word.length; i++) {
      if (i < progress) {
        html += '<span class="done">' + word[i] + '</span>';
      } else if (i === progress) {
        html += '<span class="current">' + word[i] + '</span>';
      } else {
        html += '<span>' + word[i] + '</span>';
      }
    }
    el.innerHTML = html;
  }

  function renderAll() {
    if (p1Progress < p1Word.length) renderWord(p1WordEl, p1Word, p1Progress);
    else p1WordEl.textContent = '✓';
    if (p2Progress < p2Word.length) renderWord(p2WordEl, p2Word, p2Progress);
    else p2WordEl.textContent = '✓';
  }

  function newWords() {
    p1Word = randomWord();
    p2Word = randomWord();
    p1Progress = 0;
    p2Progress = 0;
    renderAll();
  }

  function updateTurn() {
    turnEl.textContent = "P" + currentTurn + "'s Turn";
    turnEl.className = '';
    if (currentTurn === 1) turnEl.id = 'turnP1'; else turnEl.id = 'turnP2';
  }

  function startGame() {
    state = 'playing';
    p1Score = 0; p2Score = 0;
    timeLeft = 60;
    currentTurn = 1;
    startScreen.style.display = 'none';
    endScreen.style.display = 'none';
    gameArea.style.display = 'flex';
    timerEl.textContent = timeLeft;
    p1ScoreEl.textContent = 'Score: 0';
    p2ScoreEl.textContent = 'Score: 0';
    newWords();
    updateTurn();

    if (timerInterval) clearInterval(timerInterval);
    timerInterval = setInterval(() => {
      timeLeft--;
      timerEl.textContent = timeLeft;
      if (timeLeft <= 10) timerEl.style.color = '#f44';
      else timerEl.style.color = '#fff';
      if (timeLeft <= 0) endGame();
    }, 1000);
  }

  function endGame() {
    state = 'end';
    clearInterval(timerInterval);
    timerEl.style.color = '#fff';
    gameArea.style.display = 'none';
    endScreen.style.display = 'block';
    
    let winText, winColor;
    if (p1Score > p2Score) { winText = '🏆 PLAYER 1 WINS!'; winColor = '#f0f'; }
    else if (p2Score > p1Score) { winText = '🏆 PLAYER 2 WINS!'; winColor = '#0af'; }
    else { winText = "IT'S A TIE!"; winColor = '#ff0'; }
    
    winnerText.textContent = winText;
    winnerText.style.color = winColor;
    finalScores.textContent = 'P1: ' + p1Score + '  |  P2: ' + p2Score;
  }

  function handleKey(e) {
    if (state === 'start') {
      if (e.key === 'Enter') startGame();
      return;
    }
    if (state === 'end') {
      if (e.key.toLowerCase() === 'r') startGame();
      return;
    }
    if (state !== 'playing') return;

    const key = e.key.toUpperCase();
    if (!/^[A-Z]$/.test(key)) return;

    let playerProgress, playerWord;
    if (currentTurn === 1) {
      playerProgress = p1Progress;
      playerWord = p1Word;
    } else {
      playerProgress = p2Progress;
      playerWord = p2Word;
    }

    if (key === playerWord[playerProgress]) {
      // Correct!
      if (currentTurn === 1) p1Progress++; else p2Progress++;
      
      if (p1Progress >= p1Word.length || p2Progress >= p2Word.length) {
        if (p1Progress >= p1Word.length) p1Score++;
        if (p2Progress >= p2Word.length) p2Score++;
        p1ScoreEl.textContent = 'Score: ' + p1Score;
        p2ScoreEl.textContent = 'Score: ' + p2Score;
        newWords();
      } else {
        renderAll();
      }
    } else {
      // Wrong key - flash and skip turn
      const el = currentTurn === 1 ? p1WordEl : p2WordEl;
      el.style.animation = 'none';
      el.offsetHeight; // reflow
      el.style.animation = '';
      
      // Brief visual feedback
      const spans = el.querySelectorAll('span');
      if (spans[playerProgress]) {
        spans[playerProgress].classList.add('wrong');
        setTimeout(() => spans[playerProgress] && spans[playerProgress].classList.remove('wrong'), 300);
      }
    }

    // Switch turn
    currentTurn = currentTurn === 1 ? 2 : 1;
    updateTurn();
  }

  window.addEventListener('keydown', handleKey);

  // Audio
  let audioCtx = null;
  function playSound(freq, dur) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.type = 'sine';
    osc.frequency.value = freq;
    gain.gain.setValueAtTime(0.15, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
    osc.start();
    osc.stop(audioCtx.currentTime + dur);
  }

  let firstInput = false;
  window.addEventListener('keydown', function() {
    if (!firstInput) {
      firstInput = true;
      try { audioCtx = new (window.AudioContext || window.webkitAudioContext)(); } catch(e) {}
    }
  });

  // Override handleKey to add sound
  const origHandle = handleKey;
  window.removeEventListener('keydown', origHandle);
  window.addEventListener('keydown', function(e) {
    if (!firstInput) {
      firstInput = true;
      try { audioCtx = new (window.AudioContext || window.webkitAudioContext)(); } catch(err) {}
    }
    if (state === 'start' && e.key === 'Enter') { playSound(440, 0.1); startGame(); return; }
    if (state === 'end' && e.key.toLowerCase() === 'r') { playSound(440, 0.1); startGame(); return; }
    if (state !== 'playing') return;

    const key = e.key.toUpperCase();
    if (!/^[A-Z]$/.test(key)) return;

    let playerProgress, playerWord;
    if (currentTurn === 1) { playerProgress = p1Progress; playerWord = p1Word; }
    else { playerProgress = p2Progress; playerWord = p2Word; }

    const correct = key === playerWord[playerProgress];

    if (correct) {
      playSound(currentTurn === 1 ? 523 : 659, 0.08);
      if (currentTurn === 1) p1Progress++; else p2Progress++;

      if (p1Progress >= p1Word.length || p2Progress >= p2Word.length) {
        if (p1Progress >= p1Word.length) { p1Score++; playSound(880, 0.2); }
        if (p2Progress >= p2Word.length) { p2Score++; playSound(880, 0.2); }
        p1ScoreEl.textContent = 'Score: ' + p1Score;
        p2ScoreEl.textContent = 'Score: ' + p2Score;
        newWords();
      } else {
        renderAll();
      }
    } else {
      playSound(150, 0.15);
      const el = currentTurn === 1 ? p1WordEl : p2WordEl;
      const spans = el.querySelectorAll('span');
      if (spans[playerProgress]) {
        spans[playerProgress].classList.add('wrong');
        setTimeout(() => { if (spans[playerProgress]) spans[playerProgress].classList.remove('wrong'); }, 300);
      }
    }

    currentTurn = currentTurn === 1 ? 2 : 1;
    updateTurn();
  });

})();
