<script>
const canvas = document.getElementById('canvas');
const ctx = canvas.getContext('2d');
let W, H, cellSize, gridSize, offsetX, offsetY;
let snake = [];
let filled = new Set();
let moves = 0;
let level = 1;
let maxLevel = 5;
let gameActive = false;
let solutionPath = [];
let audioCtx = null;

const levels = [
  { size: 4 },
  { size: 6 },
  { size: 8 },
  { size: 10 },
  { size: 12 }
];

function initAudio() {
  if (!audioCtx) {
    audioCtx = new (window.AudioContext || webkitAudioContext)();
  }
}

function playTone(freq, dur, type) {
  if (!audioCtx) return;
  const osc = audioCtx.createOscillator();
  const gain = audioCtx.createGain();
  osc.type = type || 'sine';
  osc.frequency.value = freq;
  gain.gain.setValueAtTime(0.15, audioCtx.currentTime);
  gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
  osc.connect(gain);
  gain.connect(audioCtx.destination);
  osc.start();
  osc.stop(audioCtx.currentTime + dur);
}

function playMove() { playTone(440, 0.08, 'sine'); }
function playWin() { playTone(523, 0.15); setTimeout(()=>playTone(659, 0.15), 100); setTimeout(()=>playTone(784, 0.3), 200); }
function playLose() { playTone(200, 0.3, 'sawtooth'); }

function generatePath(size) {
  const path = [];
  for (let row = 0; row < size; row++) {
    if (row % 2 === 0) {
      for (let col = 0; col < size; col++) path.push([col, row]);
    } else {
      for (let col = size - 1; col >= 0; col--) path.push([col, row]);
    }
  }
  return path;
}

function startLevel(lvl) {
  const cfg = levels[lvl - 1];
  gridSize = cfg.size;
  solutionPath = generatePath(gridSize);
  
  // Calculate sizing
  W = window.innerWidth;
  H = window.innerHeight;
  canvas.width = W;
  canvas.height = H;
  
  cellSize = Math.min((W - 40) / gridSize, (H - 120) / gridSize);
  offsetX = Math.floor((W - gridSize * cellSize) / 2);
  offsetY = Math.floor((H - gridSize * cellSize) / 2) + 30;
  
  // Initialize snake at start position
  const startCol = solutionPath[0][0];
  const startRow = solutionPath[0][1];
  snake = [{ col: startCol, row: startRow }];
  filled = new Set();
  filled.add(startCol + ',' + startRow);
  moves = 0;
  gameActive = true;
  
  document.getElementById('level-num').textContent = lvl;
  document.getElementById('move-count').textContent = '0';
  updateProgress();
  draw();
}

function getHead() { return snake[0]; }

function canMove(dc, dr) {
  const head = getHead();
  const nc = head.col + dc;
  const nr = head.row + dr;
  
  // Wall check
  if (nc < 0 || nc >= gridSize || nr < 0 || nr >= gridSize) return false;
  
  // Self-collision check
  for (let i = 0; i < snake.length; i++) {
    if (snake[i].col === nc && snake[i].row === nr) return false;
  }
  
  // No reverse check: can't go back to the cell we just came from
  if (snake.length > 1) {
    const neck = snake[1];
    if (neck.col === nc && neck.row === nr) return false;
  }
  
  return true;
}

function move(dc, dr) {
  if (!gameActive) return;
  
  if (!canMove(dc, dr)) {
    gameLose();
    return;
  }
  
  const head = getHead();
  const nc = head.col + dc;
  const nr = head.row + dr;
  
  snake.unshift({ col: nc, row: nr });
  filled.add(nc + ',' + nr);
  moves++;
  
  document.getElementById('move-count').textContent = moves;
  playMove();
  updateProgress();
  draw();
  
  if (filled.size === gridSize * gridSize) {
    gameWin();
  }
}

function updateProgress() {
  const pct = Math.floor((filled.size / (gridSize * gridSize)) * 100);
  document.getElementById('progress-fill').style.width = pct + '%';
}

function gameWin() {
  gameActive = false;
  playWin();
  document.getElementById('win-screen').classList.remove('hidden');
  const optimal = (gridSize * gridSize) - 1;
  document.getElementById('win-stats').textContent = 
    `Filled ${gridSize*gridSize} cells in ${moves} moves!` +
    `\nOptimal: ${optimal} moves.` +
    `\nEfficiency: ${Math.floor((optimal/moves)*100)}%`;
}

function gameLose() {
  gameActive = false;
  playLose();
  document.getElementById('lose-screen').classList.remove('hidden');
  draw();
}

function draw() {
  ctx.clearRect(0, 0, W, H);
  
  // Draw grid background
  for (let r = 0; r < gridSize; r++) {
    for (let c = 0; c < gridSize; c++) {
      const x = offsetX + c * cellSize;
      const y = offsetY + r * cellSize;
      
      if (filled.has(c + ',' + r)) {
        // Filled cell
        const gradient = ctx.createLinearGradient(x, y, x + cellSize, y + cellSize);
        gradient.addColorStop(0, '#1a4a3a');
        gradient.addColorStop(1, '#0d2e22');
        ctx.fillStyle = gradient;
      } else {
        ctx.fillStyle = '#0f0f2a';
      }
      
      ctx.fillRect(x + 1, y + 1, cellSize - 2, cellSize - 2);
      
      // Grid lines
      ctx.strokeStyle = 'rgba(68,255,255,0.1)';
      ctx.strokeRect(x, y, cellSize, cellSize);
    }
  }
  
  // Draw solution path hint (subtle dots on unfilled cells that are part of the next few steps)
  for (let i = 0; i < Math.min(5, solutionPath.length); i++) {
    const [pc, pr] = solutionPath[filled.size + i];
    if (!filled.has(pc + ',' + pr)) {
      const px = offsetX + pc * cellSize + cellSize / 2;
      const py = offsetY + pr * cellSize + cellSize / 2;
      ctx.fillStyle = 'rgba(68,255,255,0.15)';
      ctx.beginPath();
      ctx.arc(px, py, 3, 0, Math.PI * 2);
      ctx.fill();
    }
  }
  
  // Draw snake body (tail to head)
  for (let i = snake.length - 1; i >= 0; i--) {
    const seg = snake[i];
    const x = offsetX + seg.col * cellSize;
    const y = offsetY + seg.row * cellSize;
    
    const t = 1 - i / Math.max(snake.length, 1);
    const r = Math.floor(40 + t * 20);
    const g = Math.floor(200 + t * 55);
    const b = Math.floor(80 + t * 155);
    
    ctx.fillStyle = `rgb(${r},${g},${b})`;
    const pad = 3;
    const radius = 6;
    roundRect(ctx, x + pad, y + pad, cellSize - pad*2, cellSize - pad*2, radius);
    ctx.fill();
    
    // Glow on head
    if (i === 0) {
      ctx.shadowColor = '#4ff';
      ctx.shadowBlur = 15;
      roundRect(ctx, x + pad, y + pad, cellSize - pad*2, cellSize - pad*2, radius);
      ctx.fill();
      ctx.shadowBlur = 0;
      
      // Eyes
      const head = getHead();
      let eyeDx = 0, eyeDy = 0;
      if (snake.length > 1) {
        const neck = snake[1];
        eyeDx = (head.col - neck.col) * 3;
        eyeDy = (head.row - neck.row) * 3;
      }
      const ex = x + cellSize/2 + eyeDx;
      const ey = y + cellSize/2 + eyeDy;
      ctx.fillStyle = '#fff';
      ctx.beginPath();
      ctx.arc(ex - 5, ey - 5, 3, 0, Math.PI*2);
      ctx.arc(ex + 5, ey - 5, 3, 0, Math.PI*2);
      ctx.fill();
    }
  }
}

function roundRect(ctx, x, y, w, h, r) {
  ctx.beginPath();
  ctx.moveTo(x + r, y);
  ctx.lineTo(x + w - r, y);
  ctx.quadraticCurveTo(x + w, y, x + w, y + r);
  ctx.lineTo(x + w, y + h - r);
  ctx.quadraticCurveTo(x + w, y + h, x + w - r, y + h);
  ctx.lineTo(x + r, y + h);
  ctx.quadraticCurveTo(x, y + h, x, y + h - r);
  ctx.lineTo(x, y + r);
  ctx.quadraticCurveTo(x, y, x + r, y);
  ctx.closePath();
}

  }
