<script>
const canvas = document.getElementById('game');
const ctx = canvas.getContext('2d');
const overlay = document.getElementById('overlay');
const startBtn = document.getElementById('startBtn');

const COLS = 10, ROWS = 16;
let cellSize = 40;
let grid = [];
let ball = {};
let leftPaddle = {}, rightPaddle = {};
let score = 0, linesCleared = 0, highScore = 0;
let gameState = 'menu'; // menu, playing, gameover
let particles = [];
let clearingRows = [];
let clearAnimT = 0;
let audioCtx = null;

try { highScore = parseInt(localStorage.getItem('pongTetrisHigh') || '0'); } catch(e) {}

function resize() {
    const maxW = window.innerWidth, maxH = window.innerHeight;
    cellSize = Math.floor(Math.min(maxW / (COLS + 4), maxH / (ROWS + 2)));
    canvas.width = (COLS + 4) * cellSize;
    canvas.height = (ROWS + 2) * cellSize;
}
window.addEventListener('resize', resize);
resize();

function initAudio() {
    if (!audioCtx) {
        audioCtx = new (window.AudioContext || window.webkitAudioContext)();
    }
    if (audioCtx.state === 'suspended') audioCtx.resume();
}

function playSound(type) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    if (type === 'bounce') {
        osc.type = 'sine'; osc.frequency.value = 440;
        gain.gain.setValueAtTime(0.15, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.1);
        osc.start(); osc.stop(audioCtx.currentTime + 0.1);
    } else if (type === 'place') {
        osc.type = 'square'; osc.frequency.value = 220;
        gain.gain.setValueAtTime(0.1, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.08);
        osc.start(); osc.stop(audioCtx.currentTime + 0.08);
    } else if (type === 'clear') {
        osc.type = 'sine'; osc.frequency.value = 660;
        gain.gain.setValueAtTime(0.2, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.3);
        osc.start(); osc.stop(audioCtx.currentTime + 0.3);
    } else if (type === 'gameover') {
        osc.type = 'sawtooth'; osc.frequency.value = 150;
        gain.gain.setValueAtTime(0.2, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.5);
        osc.start(); osc.stop(audioCtx.currentTime + 0.5);
    }
}

function resetGame() {
    grid = Array.from({length: ROWS}, () => Array(COLS).fill(0));
    score = 0; linesCleared = 0;
    particles = [];
    clearingRows = [];
    clearAnimT = 0;
    
    leftPaddle = { y: Math.floor(ROWS / 2), height: 3 };
    rightPaddle = { y: Math.floor(ROWS / 2), height: 3 };
    
    ball = {
        x: (COLS * cellSize) / 2 + 2 * cellSize,
        y: (ROWS * cellSize) / 2 + cellSize,
        vx: cellSize * 0.8 * (Math.random() > 0.5 ? 1 : -1),
        vy: cellSize * 0.6 * (Math.random() > 0.5 ? 1 : -1)
    };
}

function startGame() {
    initAudio();
    resetGame();
    gameState = 'playing';
    overlay.classList.add('hidden');
}

startBtn.addEventListener('click', startGame);

let keys = {};
window.addEventListener('keydown', e => {
    keys[e.code] = true;
    if (e.code === 'KeyR' && gameState !== 'menu') startGame();
    if ((gameState === 'playing') && !audioCtx) initAudio();
});
window.addEventListener('keyup', e => { keys[e.code] = false; });

function getGridPos(x, y) {
    const gx = Math.floor((x - 2 * cellSize) / cellSize);
    const gy = Math.floor((y - cellSize) / cellSize);
    return { x: Math.max(0, Math.min(COLS - 1, gx)), y: Math.max(0, Math.min(ROWS - 1, gy)) };
}

function placeBlock(gx, gy) {
    if (gy < 0 || gy >= ROWS || gx < 0 || gx >= COLS) return;
    if (grid[gy][gx] === 0) {
        grid[gy][gx] = 1;
        playSound('place');
        spawnParticles(2 * cellSize + gx * cellSize + cellSize/2, cellSize + gy * cellSize + cellSize/2);
        checkRowClear();
        if (grid[0].some(c => c === 1)) triggerGameOver();
    } else {
        // Cell already occupied - game over risk
        if (gy <= 2) triggerGameOver();
    }
}

function checkRowClear() {
    for (let r = ROWS - 1; r >= 0; r--) {
        if (grid[r].every(c => c === 1)) {
            clearingRows.push(r);
        }
    }
    if (clearingRows.length > 0 && clearAnimT === 0) {
        clearAnimT = 30;
        playSound('clear');
    }
}

function processClear() {
    if (clearingRows.length > 0 && clearAnimT > 0) {
        clearAnimT--;
        if (clearAnimT === 0) {
            for (let r of clearingRows) {
                for (let c = 0; c < COLS; c++) {
                    grid[r][c] = 0;
                    spawnParticles(2 * cellSize + c * cellSize + cellSize/2, cellSize + r * cellSize + cellSize/2);
                }
            }
            // Shift rows down
            const clearedCount = clearingRows.length;
            for (let r of clearingRows) grid.splice(r, 1);
            for (let i = 0; i < clearedCount; i++) grid.unshift(Array(COLS).fill(0));
            linesCleared += clearedCount;
            score += clearedCount * 100 * clearedCount;
            if (score > highScore) {
                highScore = score;
                try { localStorage.setItem('pongTetrisHigh', highScore.toString()); } catch(e) {}
            }
            clearingRows = [];
        }
    }
}

function triggerGameOver() {
    gameState = 'gameover';
    playSound('gameover');
    if (score > highScore) {
        highScore = score;
        try { localStorage.setItem('pongTetrisHigh', highScore.toString()); } catch(e) {}
    }
    showGameOver();
}

function showGameOver() {
    overlay.classList.remove('hidden');
    overlay.innerHTML = `
        <h2>GAME OVER</h2>
        <p>Score: ${score}</p>
        <p>Lines Cleared: ${linesCleared}</p>
        <p>High Score: ${highScore}</p>
        <button class="start-btn" id="restartBtn">RESTART (R)</button>
    `;
    document.getElementById('restartBtn').addEventListener('click', startGame);
}

function spawnParticles(x, y) {
    for (let i = 0; i < 6; i++) {
        particles.push({
            x, y,
            vx: (Math.random() - 0.5) * cellSize * 0.3,
            vy: (Math.random() - 0.5) * cellSize * 0.3,
            life: 20 + Math.random() * 15,
            color: `hsl(${40 + Math.random() * 60}, 100%, ${60 + Math.random() * 30}%)`
        });
    }
}

function update() {
    if (gameState !== 'playing') return;
    
    // Paddle movement
    const paddleSpeed = cellSize * 0.7;
    if (keys['KeyW']) leftPaddle.y = Math.max(0, Math.min(ROWS - leftPaddle.height, leftPaddle.y - paddleSpeed));
    if (keys['KeyS']) leftPaddle.y = Math.max(0, Math.min(ROWS - leftPaddle.height, leftPaddle.y + paddleSpeed));
    if (keys['ArrowUp']) rightPaddle.y = Math.max(0, Math.min(ROWS - rightPaddle.height, rightPaddle.y - paddleSpeed));
    if (keys['ArrowDown']) rightPaddle.y = Math.max(0, Math.min(ROWS - rightPaddle.height, rightPaddle.y + paddleSpeed));
    
    // Ball movement
    ball.x += ball.vx;
    ball.y += ball.vy;
    
    const leftWall = 2 * cellSize;
    const rightWall = (COLS + 2) * cellSize;
    const topWall = cellSize;
    const bottomWall = (ROWS + 1) * cellSize;
    
    // Bounce off walls and deposit blocks
    if (ball.x <= leftWall) {
        ball.x = leftWall;
        ball.vx = Math.abs(ball.vx);
        const pos = getGridPos(ball.x, ball.y);
        placeBlock(0, pos.y);
        playSound('bounce');
    }
    if (ball.x >= rightWall - cellSize) {
        ball.x = rightWall - cellSize;
        ball.vx = -Math.abs(ball.vx);
        const pos = getGridPos(ball.x, ball.y);
        placeBlock(COLS - 1, pos.y);
        playSound('bounce');
    }
    if (ball.y <= topWall) {
        ball.y = topWall;
        ball.vy = Math.abs(ball.vy);
        const pos = getGridPos(ball.x, ball.y);
        placeBlock(pos.x, 0);
        playSound('bounce');
    }
    if (ball.y >= bottomWall - cellSize) {
        ball.y = bottomWall - cellSize;
        ball.vy = -Math.abs(ball.vy);
    }
    
    // Paddle collision detection (ball bounces off paddles too)
    const lpLeft = 2 * cellSize;
    const lpTop = cellSize + Math.floor(leftPaddle.y) * cellSize;
    const lpBottom = lpTop + leftPaddle.height * cellSize;
    
    const rpLeft = (COLS + 2) * cellSize - cellSize;
    const rpTop = cellSize + Math.floor(rightPaddle.y) * cellSize;
    const rpBottom = rpTop + rightPaddle.height * cellSize;
    
    // Left paddle collision
    if (ball.x <= lpLeft + cellSize && ball.x >= lpLeft - cellSize && 
        ball.y >= lpTop && ball.y <= lpBottom && ball.vx < 0) {
        ball.vx = Math.abs(ball.vx);
        const pos = getGridPos(ball.x, ball.y);
        placeBlock(0, pos.y);
        playSound('bounce');
    }
    
    // Right paddle collision
    if (ball.x >= rpLeft - cellSize && ball.x <= rpLeft + cellSize && 
        ball.y >= rpTop && ball.y <= rpBottom && ball.vx > 0) {
        ball.vx = -Math.abs(ball.vx);
        const pos = getGridPos(ball.x, ball.y);
        placeBlock(COLS - 1, pos.y);
        playSound('bounce');
    }
    
    // Slight random angle variation
    if (Math.random() < 0.02) {
        ball.vy += (Math.random() - 0.5) * cellSize * 0.3;
        const maxVy = cellSize * 1.2;
        ball.vy = Math.max(-maxVy, Math.min(maxVy, ball.vy));
    }
    
    // Process row clears
    processClear();
    
    // Update particles
    for (let i = particles.length - 1; i >= 0; i--) {
        const p = particles[i];
        p.x += p.vx;
        p.y += p.vy;
        p.life--;
        if (p.life <= 0) particles.splice(i, 1);
    }
}

function draw() {
    ctx.fillStyle = '#0a0a1a';
    ctx.fillRect(0, 0, canvas.width, canvas.height);
    
    // Grid area
    const gridLeft = 2 * cellSize;
    const gridTop = cellSize;
    
    // Draw grid lines
    ctx.strokeStyle = 'rgba(79, 195, 247, 0.1)';
    ctx.lineWidth = 1;
    for (let c = 0; c <= COLS; c++) {
        ctx.beginPath();
        ctx.moveTo(gridLeft + c * cellSize, gridTop);
        ctx.lineTo(gridLeft + c * cellSize, gridTop + ROWS * cellSize);
        ctx.stroke();
    }
    for (let r = 0; r <= ROWS; r++) {
        ctx.beginPath();
        ctx.moveTo(gridLeft, gridTop + r * cellSize);
        ctx.lineTo(gridLeft + COLS * cellSize, gridTop + r * cellSize);
        ctx.stroke();
    }
    
    // Draw blocks
    for (let r = 0; r < ROWS; r++) {
        for (let c = 0; c < COLS; c++) {
            if (grid[r][c] === 1) {
                const isClearing = clearingRows.includes(r);
                const alpha = isClearing ? (clearAnimT / 30) : 1;
                const hue = 200 + (r * 8) % 40;
                ctx.fillStyle = `hsla(${hue}, 70%, 55%, ${alpha})`;
                ctx.fillRect(gridLeft + c * cellSize + 2, gridTop + r * cellSize + 2, cellSize - 4, cellSize - 4);
                ctx.strokeStyle = `hsla(${hue}, 80%, 75%, ${alpha})`;
                ctx.lineWidth = 1;
                ctx.strokeRect(gridLeft + c * cellSize + 2, gridTop + r * cellSize + 2, cellSize - 4, cellSize - 4);
            }
        }
    }
    
    // Draw paddles
    const lpY = gridTop + Math.floor(leftPaddle.y) * cellSize;
    ctx.fillStyle = '#4fc3f7';
    ctx.shadowColor = '#4fc3f7';
    ctx.shadowBlur = 10;
    ctx.fillRect(gridLeft - cellSize, lpY, cellSize - 6, leftPaddle.height * cellSize);
    
    const rpY = gridTop + Math.floor(rightPaddle.y) * cellSize;
    ctx.fillStyle = '#ff5252';
    ctx.shadowColor = '#ff5252';
    ctx.fillRect(gridLeft + COLS * cellSize + 6, rpY, cellSize - 6, rightPaddle.height * cellSize);
    ctx.shadowBlur = 0;
    
    // Draw ball
    if (gameState === 'playing') {
        ctx.beginPath();
        ctx.arc(ball.x, ball.y, cellSize * 0.25, 0, Math.PI * 2);
        ctx.fillStyle = '#fff';
        ctx.shadowColor = '#fff';
        ctx.shadowBlur = 15;
        ctx.fill();
        ctx.shadowBlur = 0;
    }
    
    // Draw particles
    for (const p of particles) {
        ctx.fillStyle = p.color;
        ctx.globalAlpha = p.life / 35;
        ctx.fillRect(p.x - 3, p.y - 3, 6, 6);
    }
    ctx.globalAlpha = 1;
    
    // Draw HUD
    if (gameState === 'playing') {
        ctx.fillStyle = '#4fc3f7';
        ctx.font = `bold ${Math.floor(cellSize * 0.5)}px Courier New`;
        ctx.textAlign = 'left';
        ctx.fillText(`SCORE: ${score}`, gridLeft, cellSize * 0.6);
        ctx.textAlign = 'right';
        ctx.fillStyle = '#ff5252';
        ctx.fillText(`HIGH: ${highScore}`, gridLeft + COLS * cellSize, cellSize * 0.6);
        ctx.textAlign = 'center';
        ctx.fillStyle = '#b0bec5';
        ctx.font = `${Math.floor(cellSize * 0.4)}px Courier New`;
        ctx.fillText(`LINES: ${linesCleared}`, canvas.width / 2, canvas.height - cellSize * 0.3);
    }
}

function gameLoop() {
    update();
    draw();
    requestAnimationFrame(gameLoop);
}

resetGame();
gameLoop();
</script>
