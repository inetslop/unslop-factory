function playSound(freq, dur, type, vol) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = type || 'square';
    osc.frequency.value = freq;
    gain.gain.setValueAtTime(vol || 0.1, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + dur);
}

class Obstacle {

        this.x = x; // 0-1 in track

function spawnParticles(x, y, color, count) {
    for (let i = 0; i < count; i++) {
        particles.push({
            x: x, y: y,
            vx: (Math.random() - 0.5) * 8,
            vy: (Math.random() - 0.5) * 8 - 2,
            life: 1,
            color: color,
            size: 3 + Math.random() * 4
        });
    }
}

function checkCollisions(dt) {
    const player = cars[0];
    if (!player.alive) return;
    
    // Check obstacles
    for (let obs of obstacles) {
        const relDist = obs.y - player.distance;
        if (Math.abs(relDist) < 25) {
            const obsScreenX = trackLeft + obs.x * trackW;
            const playerScreenX = trackLeft + player.x * trackW;
            const hitDist = (trackW / LANE_COUNT) * 0.4;
            
            if (Math.abs(obsScreenX - playerScreenX) < hitDist) {
                if (player.shieldTimer > 0) {
                    // Shield absorbs hit, destroy obstacle
                    obs.y = -9999;
                    playSound(200, 0.1, 'sine', 0.15);
                    spawnParticles(playerScreenX, H * 0.75, '#44ccff', 8);
                } else {
                    player.alive = false;
                    playCrash();
                    spawnParticles(playerScreenX, H * 0.75, player.color, 20);
                    gameState = 'gameover';
                    raceResult = 'CRASHED!';
                    try { localStorage.setItem('tetrisRacingBest', String(score)); } catch(e) {}
                }
            }
        }
    }
    
    // Check power-ups
    for (let p of powerups) {
        if (p.collected) continue;
        const relDist = p.y - player.distance;
        if (Math.abs(relDist) < 30) {
            const pScreenX = trackLeft + p.x * trackW;
            const playerScreenX = trackLeft + player.x * trackW;
            if (Math.abs(pScreenX - playerScreenX) < (trackW / LANE_COUNT) * 0.4) {
                p.collected = true;
                playPowerup();
                if (p.type === 'boost') {
                    player.boostTimer = 2;
                } else {
                    player.shieldTimer = 3;
                }
                score += 50;
            }
        }
    }
    
    // Check car collisions
    for (let i = 1; i < cars.length; i++) {
        const car = cars[i];
        if (!car.alive || car.finished) continue;
        const relDist = car.distance - player.distance;
        if (Math.abs(relDist) < 20 && Math.abs(car.x - player.x) < 0.1) {
            if (player.shieldTimer > 0) {
                car.alive = false;
                spawnParticles(trackLeft + car.x * trackW, H * 0.75, car.color, 12);
                playSound(300, 0.2, 'sawtooth', 0.1);
            } else {
                player.alive = false;
                playCrash();
                spawnParticles(trackLeft + player.x * trackW, H * 0.75, player.color, 20);
                gameState = 'gameover';
                raceResult = 'COLLISION!';
                try { localStorage.setItem('tetrisRacingBest', String(score)); } catch(e) {}
            }
        }
    }
    
    // Check finish
    if (!player.finished && player.distance >= FINISH_DISTANCE) {
        player.finished = true;
        player.finishTime = gameTime;
        const rank = cars.filter(c => c.finished || !c.alive).length + 1;
        gameState = 'win';
        raceResult = `FINISHED! Place: #${rank}`;
        playWin();
        try { localStorage.setItem('tetrisRacingBest', String(Math.max(score, bestScore))); } catch(e) {}
    }
    
    // AI cars finish
    for (let i = 1; i < cars.length; i++) {
        const car = cars[i];
        if (!car.alive || car.finished) continue;
        if (car.distance >= FINISH_DISTANCE) {
            car.finished = true;
            car.finishTime = gameTime;
        }
    }
}

function update(dt) {
    if (gameState !== 'playing') return;
    
    gameTime += dt;
    
    const player = cars[0];
    
    // Player movement
    if (player.alive && !player.finished) {
        const steerSpeed = 2.5 * dt;
        if (keys['ArrowLeft'] || keys['a'] || keys['A']) {
            player.x -= steerSpeed;
        }
        if (keys['ArrowRight'] || keys['d'] || keys['D']) {
            player.x += steerSpeed;
        }
    }
    
    // Update all cars
    for (let car of cars) {
        car.update(dt, trackW);
    }
    
    // Scroll
    if (player.alive) {
        scrollOffset = player.distance;
    }
    
    // Score
    score += dt * 10;
    
    checkCollisions(dt);
    
    // Update particles
    for (let p of particles) {
        p.x += p.vx * dt * 60;
        p.y += p.vy * dt * 60;
        p.life -= dt * 2;
    }
    particles = particles.filter(p => p.life > 0);
}

function drawTrack() {
    // Background
    ctx.fillStyle = '#0a0a1a';
    ctx.fillRect(0, 0, W, H);
    
    // Track background
    const grad = ctx.createLinearGradient(trackLeft, 0, trackLeft + trackW, 0);
    grad.addColorStop(0, '#1a1a2e');
    grad.addColorStop(0.5, '#16213e');
    grad.addColorStop(1, '#1a1a2e');
    ctx.fillStyle = grad;
    ctx.fillRect(trackLeft, 0, trackW, H);
    
    // Track borders
    ctx.strokeStyle = '#ff4444';
    ctx.lineWidth = 3;
    ctx.beginPath();
    ctx.moveTo(trackLeft, 0);
    ctx.lineTo(trackLeft, H);
    ctx.stroke();
    ctx.beginPath();
    ctx.moveTo(trackLeft + trackW, 0);
    ctx.lineTo(trackLeft + trackW, H);
    ctx.stroke();
    
    // Lane lines (scrolling)
    const laneW = trackW / LANE_COUNT;
    const lineSpacing = 80;
    const offset = (scrollOffset * 0.5) % lineSpacing;
    
    ctx.strokeStyle = 'rgba(255,255,255,0.1)';
    ctx.lineWidth = 1;
    for (let i = 1; i < LANE_COUNT; i++) {
        const x = trackLeft + i * laneW;
        ctx.beginPath();
        ctx.moveTo(x, 0);
        ctx.lineTo(x, H);
        ctx.stroke();
    }
    
    // Dashes on lanes
    ctx.strokeStyle = 'rgba(255,255,255,0.15)';
    ctx.setLineDash([20, 30]);
    for (let i = 1; i < LANE_COUNT; i++) {
        const x = trackLeft + i * laneW;
        ctx.beginPath();
        ctx.moveTo(x, -offset);
        ctx.lineTo(x, H);
        ctx.stroke();
    }
    ctx.setLineDash([]);
    
    // Finish line
    const finishRelDist = FINISH_DISTANCE - scrollOffset;
    const finishY = H * 0.75 + finishRelDist * 0.5;
    if (finishY > -100 && finishY < H + 100) {
        ctx.fillStyle = 'rgba(255,255,255,0.8)';
        const checkerSize = trackW / 20;
        for (let row = 0; row < 3; row++) {
            for (let col = 0; col < Math.ceil(trackW / checkerSize); col++) {
                if ((row + col) % 2 === 0) {
                    ctx.fillRect(trackLeft + col * checkerSize, finishY + row * checkerSize, checkerSize, checkerSize);
                }
            }
        }
    }
}

        ctx.fillStyle = '#ffaa00';
        ctx.textAlign = 'left';
        ctx.fillText(`⚡ BOOST: ${player.boostTimer.toFixed(1)}s`, trackLeft + 10, 80);
    }
    if (player.shieldTimer > 0) {
        ctx.fillStyle = '#44ccff';
        ctx.textAlign = 'left';
        ctx.fillText(`🛡 SHIELD: ${player.shieldTimer.toFixed(1)}s`, trackLeft + 10, 105);
    }
    
    // Position indicator (which car is ahead)
    const sortedCars = [...cars].sort((a,b) => b.distance - a.distance);

    // Mini-map / progress bar

    }
}

function drawParticles() {
    for (let p of particles) {
        ctx.globalAlpha = p.life;
        ctx.fillStyle = p.color;
        ctx.fillRect(p.x - p.size/2, p.y - p.size/2, p.size, p.size);
    }
    ctx.globalAlpha = 1;
}

        cars[i].draw(trackW, scrollOffset, trackLeft);
    }

}

function drawStartScreen() {
    ctx.fillStyle = '#0a0a1a';
    ctx.fillRect(0, 0, W, H);
    
    // Title
    const titleSize = Math.min(W * 0.08, 60);
    ctx.font = `bold ${titleSize}px monospace`;
    ctx.textAlign = 'center';
    ctx.fillStyle = '#ff4444';
    ctx.fillText('TETRIS', W/2, H*0.25);
    ctx.fillStyle = '#44ff44';
    ctx.fillText('RACING', W/2, H*0.25 + titleSize * 1.1);
    
    // Animated tetrominoes in background
    const t = gameTime * 2;
    for (let i = 0; i < 8; i++) {
        const bx = (W * 0.1) + (i % 4) * W * 0.25;
        const by = H * 0.5 + Math.sin(t + i) * 30 + (Math.floor(i/4)) * H * 0.15;
        const s = 15;
        ctx.fillStyle = CAR_COLORS[i % 4] + '40';
        for (let b of SHAPES[CAR_SHAPES[i % 4]]) {
            ctx.fillRect(bx + b[0]*s, by + b[1]*s, s*0.9, s*0.9);
        }
    }
    
    // Controls
    const ctrlSize = Math.min(W * 0.03, 18);
    ctx.font = `${ctrlSize}px monospace`;
    ctx.fillStyle = '#aaa';
    ctx.fillText('← → or A/D to steer', W/2, H*0.55);
    ctx.fillText('Avoid obstacles & other racers!', W/2, H*0.55 + ctrlSize*1.5);
    ctx.fillText('Collect ⚡ (boost) and 🛡 (shield)', W/2, H*0.55 + ctrlSize*3);
    ctx.fillText('First to cross the finish line wins!', W/2, H*0.55 + ctrlSize*4.5);
    
    // Start prompt
    const pulse = 0.5 + Math.sin(gameTime * 3) * 0.5;
    ctx.globalAlpha = pulse;
    ctx.fillStyle = '#fff';
    ctx.font = `bold ${Math.min(W*0.04, 28)}px monospace`;
    ctx.fillText('PRESS SPACE TO RACE', W/2, H*0.8);
    ctx.globalAlpha = 1;
    
    // Best score
    if (bestScore > 0) {
        ctx.fillStyle = '#ffaa00';
        ctx.font = `${ctrlSize}px monospace`;
        ctx.fillText(`BEST: ${bestScore}`, W/2, H*0.9);
    }
}

function drawGameOver() {
    draw(); // Draw the game state in background
    
    // Overlay
    ctx.fillStyle = 'rgba(0,0,0,0.7)';
    ctx.fillRect(0, 0, W, H);
    
    const titleSize = Math.min(W * 0.06, 48);
    ctx.font = `bold ${titleSize}px monospace`;
    ctx.textAlign = 'center';
    ctx.fillStyle = '#ff4444';
    ctx.fillText('RACE OVER', W/2, H*0.3);
    
    ctx.font = `${Math.min(W*0.035, 28)}px monospace`;

    ctx.fillText(`Score: ${Math.floor(score)}`, W/2, H*0.52);

    ctx.globalAlpha = pulse;
    ctx.fillStyle = '#44ff44';
    ctx.font = `bold ${Math.min(W*0.04, 26)}px monospace`;
    ctx.fillText('PRESS SPACE TO RESTART', W/2, H*0.7);
    ctx.globalAlpha = 1;
}

function drawWin() {
    draw();
    
    // Overlay
    ctx.fillStyle = 'rgba(0,0,0,0.6)';
    ctx.fillRect(0, 0, W, H);
    
    const titleSize = Math.min(W * 0.06, 48);
    ctx.font = `bold ${titleSize}px monospace`;
    ctx.textAlign = 'center';
    ctx.fillStyle = '#44ff44';
    ctx.fillText('YOU FINISHED!', W/2, H*0.3);
    
    ctx.font = `${Math.min(W*0.035, 28)}px monospace`;
    ctx.fillStyle = '#fff';
    ctx.fillText(raceResult, W/2, H*0.42);
    ctx.fillText(`Score: ${Math.floor(score)}`, W/2, H*0.52);
    
    // Confetti particles
    for (let i = 0; i < 30; i++) {
        const cx = Math.random() * W;
        const cy = (gameTime * 100 + i * 50) % H;
        ctx.fillStyle = CAR_COLORS[i % 4];
        ctx.fillRect(cx, cy, 6, 6);
    }
    
    const pulse = 0.5 + Math.sin(gameTime * 3) * 0.5;
    ctx.globalAlpha = pulse;
    ctx.fillStyle = '#fff';
    ctx.font = `bold ${Math.min(W*0.04, 26)}px monospace`;
    ctx.fillText('PRESS SPACE TO RACE AGAIN', W/2, H*0.7);
    ctx.globalAlpha = 1;
}

function gameLoop(timestamp) {
    const dt = Math.min((timestamp - lastTime) / 1000, 0.05);
    lastTime = timestamp;
    
    if (gameState === 'start') {
        gameTime += dt;
        drawStartScreen();
    } else if (gameState === 'playing') {
        update(dt);
        draw();
    } else if (gameState === 'gameover' || gameState === 'win') {
        gameTime += dt;
        if (gameState === 'gameover') drawGameOver();
        else drawWin();
    }
    
    requestAnimationFrame(gameLoop);
}

// Input
window.addEventListener('keydown', (e) => {
    keys[e.key] = true;
    initAudio();
    
    if (e.key === ' ' || e.key === 'Enter') {
        if (gameState === 'start' || gameState === 'gameover' || gameState === 'win') {
            initGame();
            gameState = 'playing';
        }
    }
    
    e.preventDefault();
});

window.addEventListener('keyup', (e) => {
    keys[e.key] = false;
});

// Also support touch/click to start
canvas.addEventListener('click', () => {
    initAudio();
    if (gameState === 'start' || gameState === 'gameover' || gameState === 'win') {
        initGame();
        gameState = 'playing';
    }
});

// Start
initGame();
lastTime = performance.now();
requestAnimationFrame(gameLoop);
