<!DOCTYPE html>
<html lang="en">
<head>
<meta charset="UTF-8">
<meta name="viewport" content="width=device-width, initial-scale=1.0">

<script>
const canvas = document.getElementById('game');
const ctx = canvas.getContext('2d');

let W, H, CELL, GRID_W = 16, GRID_H = 12;
let offsetX, offsetY;

function resize() {
    W = window.innerWidth;
    H = window.innerHeight;
    canvas.width = W;
    canvas.height = H;
    CELL = Math.min(Math.floor((W * 0.8) / GRID_W), Math.floor((H * 0.7) / GRID_H));
    offsetX = Math.floor((W - GRID_W * CELL) / 2);
    offsetY = Math.floor((H - GRID_H * CELL) / 2);
}
window.addEventListener('resize', resize);
resize();

const COLORS = {
    I: '#00e5ff', O: '#ffe500', T: '#b400ff',
    S: '#00ff66', Z: '#ff3355', J: '#3366ff', L: '#ff8800'
};

const SHAPES = {
    I: [[1,1,1,1]],
    O: [[1,1],[1,1]],
    T: [[1,1,1],[0,1,0]],
    S: [[0,1,1],[1,1,0]],
    Z: [[1,1,0],[0,1,1]],
    J: [[1,0,0],[1,1,1]],
    L: [[0,0,1],[1,1,1]]
};

let grid, currentPiece, nextType, score, level, linesCleared, speed, gameOver, paused, state;
let particles = [], lastTime = 0, spawnTimer = 0, highScore = 0;
let audioCtx = null;

try { highScore = parseInt(localStorage.getItem('sideTetrisHigh') || '0'); } catch(e) {}

function init() {
    grid = Array.from({length: GRID_H}, () => Array(GRID_W).fill(null));
    score = 0; level = 1; linesCleared = 0; speed = 2.5;
    gameOver = false; paused = false; state = 'start';
    particles = []; currentPiece = null; nextType = randomType();
}

function randomType() {
    const types = Object.keys(SHAPES);
    return types[Math.floor(Math.random() * types.length)];
}

function getShape(type, rotation) {
    let shape = SHAPES[type].map(r => [...r]);
    for (let i = 0; i % 4 === rotation % 4; i++) {
        shape = transpose(shape);
        shape = shape.map(row => [...row]).reverse();
    }
    return shape;
}

function transpose(m) {
    return m[0].map((_, col) => m.map(row => row[col]));
}

function spawnPiece() {
    const type = nextType;
    nextType = randomType();
    const shape = getShape(type, 0);
    const h = shape.length;
    const w = shape[0].length;
    const x = GRID_W - 1;
    const y = Math.floor(Math.random() * (GRID_H - h));
    
    if (collides(shape, x, y)) {
        gameOver = true;
        return;
    }
    
    currentPiece = { type, shape, x, y, rotation: 0 };
}

function collides(shape, px, py) {
    for (let r = 0; r < shape.length; r++) {
        for (let c = 0; c < shape[r].length; c++) {
            if (shape[r][c]) {
                const nx = px + c;
                const ny = py + r;
                if (nx < 0 || nx >= GRID_W || ny < 0 || ny >= GRID_H) return true;
                if (grid[ny][nx]) return true;
            }
        }
    }
    return false;
}

function lockPiece() {
    const { shape, x, y, type } = currentPiece;
    for (let r = 0; r < shape.length; r++) {
        for (let c = 0; c < shape[r].length; c++) {
            if (shape[r][c]) {
                grid[y + r][x + c] = type;
            }
        }
    }
    addParticles(x, y, type);
    clearLines();
    playLockSound();
    spawnPiece();
}

function clearLines() {
    let cleared = 0;
    for (let r = GRID_H - 1; r >= 0; r--) {
        if (grid[r].every(cell => cell !== null)) {
            grid.splice(r, 1);
            grid.unshift(Array(GRID_W).fill(null));
            cleared++;
            for (let c = 0; c < GRID_W; c++) {
                addParticle(offsetX + c * CELL, offsetY + r * CELL, COLORS['S']);
            }
        }
    }
    if (cleared > 0) {
        linesCleared += cleared;
        score += cleared * cleared * 100 * level;
        level = Math.floor(linesCleared / 5) + 1;
        speed = 2.5 + (level - 1) * 0.8;
        playClearSound(cleared);
    }
}

function addParticles(gx, gy, type) {
    const color = COLORS[type];
    for (let i = 0; i < 6; i++) {
        particles.push({
            x: offsetX + gx * CELL + Math.random() * CELL,
            y: offsetY + gy * CELL + Math.random() * CELL,
            vx: (Math.random() - 0.5) * 120,
            vy: (Math.random() - 0.5) * 120,
            life: 1, color
        });
    }
}

function addParticle(x, y, color) {
    particles.push({ x, y, vx: (Math.random()-0.5)*200, vy: (Math.random()-0.5)*200 - 50, life: 1, color });
}

function tryMove(dx, dy) {
    if (!currentPiece || gameOver) return;
    const nx = currentPiece.x + dx;
    const ny = currentPiece.y + dy;
    if (!collides(currentPiece.shape, nx, ny)) {
        currentPiece.x = nx;
        currentPiece.y = ny;
    }
}

function tryRotate() {
    if (!currentPiece || gameOver) return;
    const newRot = (currentPiece.rotation + 1) % 4;
    const newShape = getShape(currentPiece.type, newRot);
    if (!collides(newShape, currentPiece.x, currentPiece.y)) {
        currentPiece.shape = newShape;
        currentPiece.rotation = newRot;
    } else if (!collides(newShape, currentPiece.x - 1, currentPiece.y)) {
        currentPiece.x--;
        currentPiece.shape = newShape;
        currentPiece.rotation = newRot;
    } else if (!collides(newShape, currentPiece.x + 1, currentPiece.y)) {
        currentPiece.x++;
        currentPiece.shape = newShape;
        currentPiece.rotation = newRot;
    }
}

function playLockSound() {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = 'square';
    osc.frequency.value = 120;
    gain.gain.setValueAtTime(0.15, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.1);
    osc.connect(gain).connect(audioCtx.destination);
    osc.start(); osc.stop(audioCtx.currentTime + 0.1);
}

function playClearSound(lines) {
    if (!audioCtx) return;
    for (let i = 0; i < lines; i++) {
        const osc = audioCtx.createOscillator();
        const gain = audioCtx.createGain();
        osc.type = 'square';
        osc.frequency.value = 300 + i * 150;
        gain.gain.setValueAtTime(0.12, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.15);
        osc.connect(gain).connect(audioCtx.destination);
        osc.start(audioCtx.currentTime + i * 0.05);
        osc.stop(audioCtx.currentTime + 0.2);
    }
}

function initAudio() {
    if (!audioCtx) {
        audioCtx = new (window.AudioContext || window.webkitAudioContext)();
    }
}

let keys = {};
window.addEventListener('keydown', (e) => {
    keys[e.code] = true;
    if (state === 'start' && (e.code === 'Space' || e.code === 'Enter')) {
        initAudio();
        state = 'playing';
        spawnPiece();
    }
    if (state === 'gameover' && (e.code === 'KeyR' || e.code === 'Enter')) {
        init();
        state = 'start';
    }
    if (state === 'playing') {
        if (e.code === 'KeyP') paused = !paused;
        if (!paused) {
            if (e.code === 'Space') tryRotate();
            if (e.code === 'ArrowUp') tryMove(0, -1);
            if (e.code === 'ArrowDown') tryMove(0, 1);
            if (e.code === 'ArrowLeft') tryMove(-1, 0);
            if (e.code === 'ArrowRight') tryMove(1, 0);
        }
    }
});
window.addEventListener('keyup', (e) => { keys[e.code] = false; });

function update(dt) {
    if (state !== 'playing' || paused || gameOver) return;
    
    if (currentPiece) {
        const moveAmount = speed * dt;
        let remaining = moveAmount;
        while (remaining > 0) {
            const step = Math.min(remaining, 0.5);
            const nx = currentPiece.x - step;
            const roundedX = Math.floor(nx);
            
            if (roundedX < currentPiece.x && collides(currentPiece.shape, roundedX, currentPiece.y)) {
                lockPiece();
                break;
            } else {
                currentPiece.x -= step;
                remaining -= step;
            }
        }
    }
    
    // Update particles
    for (let i = particles.length - 1; i >= 0; i--) {
        const p = particles[i];
        p.x += p.vx * dt;
        p.y += p.vy * dt;
        p.life -= dt * 2;
        if (p.life <= 0) particles.splice(i, 1);
    }
}

function draw() {
    ctx.fillStyle = '#0a0a1a';
    ctx.fillRect(0, 0, W, H);
    
    // Background stars
    ctx.fillStyle = 'rgba(255,255,255,0.3)';
    for (let i = 0; i < 50; i++) {
        const sx = ((i * 137 + 53) % W);
        const sy = ((i * 251 + 97) % H);
        ctx.fillRect(sx, sy, 1, 1);
    }
    
    if (state === 'start') {
        drawStartScreen();
        return;
    }
    
    // Grid
    ctx.strokeStyle = 'rgba(100,140,255,0.1)';
    ctx.lineWidth = 1;
    for (let r = 0; r <= GRID_H; r++) {
        ctx.beginPath();
        ctx.moveTo(offsetX, offsetY + r * CELL);
        ctx.lineTo(offsetX + GRID_W * CELL, offsetY + r * CELL);
        ctx.stroke();
    }
    for (let c = 0; c <= GRID_W; c++) {
        ctx.beginPath();
        ctx.moveTo(offsetX + c * CELL, offsetY);
        ctx.lineTo(offsetX + c * CELL, offsetY + GRID_H * CELL);
        ctx.stroke();
    }
    
    // Grid blocks
    for (let r = 0; r < GRID_H; r++) {
        for (let c = 0; c < GRID_W; c++) {
            if (grid[r][c]) {
                drawBlock(offsetX + c * CELL, offsetY + r * CELL, COLORS[grid[r][c]], 0.9);
            }
        }
    }
    
    // Current piece
    if (currentPiece && !gameOver) {
        const { shape, x, y, type } = currentPiece;
        for (let r = 0; r < shape.length; r++) {
            for (let c = 0; c < shape[r].length; c++) {
                if (shape[r][c]) {
                    drawBlock(offsetX + (x + c) * CELL, offsetY + (y + r) * CELL, COLORS[type], 1);
                }
            }
        }
    }
    
    // Particles
    for (const p of particles) {
        ctx.globalAlpha = p.life;
        ctx.fillStyle = p.color;
        ctx.fillRect(p.x - 3, p.y - 3, 6, 6);
    }
    ctx.globalAlpha = 1;
    
    // HUD
    ctx.fillStyle = '#fff';
    ctx.font = `${Math.max(14, CELL * 0.4)}px Courier New`;
    ctx.textAlign = 'left';
    ctx.fillText(`SCORE: ${score}`, offsetX, offsetY - 20);
    ctx.fillText(`LEVEL: ${level}`, offsetX + 200, offsetY - 20);
    ctx.fillText(`LINES: ${linesCleared}`, offsetX + 400, offsetY - 20);
    
    // Next piece preview
    const nextShape = SHAPES[nextType];
    const prevX = offsetX + GRID_W * CELL + 30;
    const prevY = offsetY + 10;
    ctx.fillStyle = 'rgba(255,255,255,0.6)';
    ctx.fillText('NEXT:', prevX, prevY);
    for (let r = 0; r < nextShape.length; r++) {
        for (let c = 0; c < nextShape[r].length; c++) {
            if (nextShape[r][c]) {
                drawBlock(prevX + c * CELL * 0.7, prevY + 10 + r * CELL * 0.7, COLORS[nextType], 0.8);
            }
        }
    }
    
    // Speed indicator
    ctx.fillStyle = 'rgba(255,255,255,0.4)';
    ctx.fillText(`SPEED: ${speed.toFixed(1)}x`, prevX, prevY + nextShape.length * CELL * 0.7 + 30);
    
    if (paused) {
        ctx.fillStyle = 'rgba(0,0,0,0.7)';
        ctx.fillRect(0, 0, W, H);
        ctx.fillStyle = '#fff';
        ctx.font = `${CELL}px Courier New`;
        ctx.textAlign = 'center';
        ctx.fillText('PAUSED', W/2, H/2);
        ctx.font = `${CELL*0.4}px Courier New`;
        ctx.fillText('Press P to resume', W/2, H/2 + CELL);
    }
    
    if (gameOver) {
        drawGameOver();
    }
}

function drawBlock(x, y, color, alpha) {
    const s = CELL * 0.85;
    const pad = (CELL - s) / 2;
    ctx.globalAlpha = alpha;
    ctx.fillStyle = color;
    ctx.fillRect(x + pad, y + pad, s, s);
    
    // Highlight
    ctx.fillStyle = 'rgba(255,255,255,0.3)';
    ctx.fillRect(x + pad, y + pad, s, 3);
    ctx.fillRect(x + pad, y + pad, 3, s);
    
    // Shadow
    ctx.fillStyle = 'rgba(0,0,0,0.3)';
    ctx.fillRect(x + pad + s - 3, y + pad, 3, s);
    ctx.fillRect(x + pad, y + pad + s - 3, s, 3);
    
    // Inner glow
    ctx.fillStyle = 'rgba(255,255,255,0.1)';
    ctx.fillRect(x + pad + 4, y + pad + 4, s - 8, s - 8);
    
    ctx.globalAlpha = 1;
}

function drawStartScreen() {
    ctx.fillStyle = '#fff';
    ctx.textAlign = 'center';
    ctx.font = `bold ${Math.min(W*0.06, 48)}px Courier New`;
    ctx.fillText('SIDE-SCROLLING TETRIS', W/2, H * 0.3);
    
    ctx.font = `${Math.min(W*0.03, 20)}px Courier New`;
    ctx.fillStyle = 'rgba(255,255,255,0.8)';
    const lines = [
        'Pieces scroll in from the right — fill the grid!',
        '',
        'CONTROLS:',
        '← →  Fine horizontal adjustment',
        '↑ ↓  Move piece up / down',
        'SPACE  Rotate',
        'P  Pause',
        '',
        'Fit pieces into gaps before they reach the left wall!',
        'Clear full rows to score. Speed increases with level.',
        '',
        `HIGH SCORE: ${highScore}`,
        '',
        '[ PRESS SPACE TO START ]'
    ];
    lines.forEach((line, i) => {
        ctx.fillText(line, W/2, H * 0.4 + i * 28);
    });
}

function drawGameOver() {
    ctx.fillStyle = 'rgba(0,0,0,0.75)';
    ctx.fillRect(0, 0, W, H);
    ctx.fillStyle = '#ff3355';
    ctx.font = `bold ${Math.min(W*0.06, 48)}px Courier New`;
    ctx.textAlign = 'center';
    ctx.fillText('GAME OVER', W/2, H * 0.35);
    
    ctx.fillStyle = '#fff';
    ctx.font = `${Math.min(W*0.03, 22)}px Courier New`;
    ctx.fillText(`Score: ${score}   Lines: ${linesCleared}   Level: ${level}`, W/2, H * 0.45);
    
    if (score >= highScore && score > 0) {
        ctx.fillStyle = '#ffe500';
        ctx.fillText('★ NEW HIGH SCORE! ★', W/2, H * 0.52);
    } else {
        ctx.fillStyle = 'rgba(255,255,255,0.6)';
        ctx.fillText(`High Score: ${highScore}`, W/2, H * 0.52);
    }
    
    ctx.fillStyle = '#fff';
    ctx.font = `${Math.min(W*0.03, 20)}px Courier New`;
    ctx.fillText('[ PRESS R TO RESTART ]', W/2, H * 0.65);
}

function gameLoop(time) {
    const dt = Math.min((time - lastTime) / 1000, 0.05);
    lastTime = time;
    
    update(dt);
    draw();
    
    if (gameOver && state === 'playing') {
        state = 'gameover';
        if (score > highScore) {
            highScore = score;
            try { localStorage.setItem('sideTetrisHigh', highScore); } catch(e) {}
        }
    }
    
    requestAnimationFrame(gameLoop);
}

init();
requestAnimationFrame(gameLoop);
</script>
