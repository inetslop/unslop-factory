```html
<!DOCTYPE html>
<html lang="en">
<head>
<meta charset="UTF-8">
<meta name="viewport" content="width=device-width, initial-scale=1.0">
<title>Asteroids Endless</title>
<style>
* { margin: 0; padding: 0; box-sizing: border-box; }
html, body { width: 100%; height: 100%; overflow: hidden; background: #000; }
canvas { display: block; width: 100%; height: 100%; }
</style>
</head>
<body>
<canvas id="game"></canvas>
<script>
(function() {
const canvas = document.getElementById('game');
const ctx = canvas.getContext('2d');

let W, H;
function resize() {
W = window.innerWidth;
H = window.innerHeight;
canvas.width = W;
canvas.height = H;
}
window.addEventListener('resize', resize);
resize();

// Audio
let audioCtx = null;
function initAudio() {
if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
}
function playSound(freq, duration, type, vol) {
if (!audioCtx) return;
const osc = audioCtx.createOscillator();
const gain = audioCtx.createGain();
osc.type = type || 'square';
osc.frequency.value = freq;
gain.gain.setValueAtTime(vol || 0.15, audioCtx.currentTime);
gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + duration);
osc.connect(gain);
gain.connect(audioCtx.destination);
osc.start();
osc.stop(audioCtx.currentTime + duration);
}

// Game state
let state = 'start'; // start, playing, gameover
let score = 0;
let highScore = 0;
try {
highScore = parseInt(localStorage.getItem('asteroidsEndlessHigh') || '0');
} catch(e) {}

let ship, bullets, asteroids, particles, stars;
let keys = {};
let timeAlive = 0;
let lastTime = 0;
let spawnTimer = 0;
let difficulty = 1;

function initStars() {
stars = [];
for (let i = 0; i < 150; i++) {
stars.push({ x: Math.random() * W, y: Math.random() * H, s: Math.random() * 2 + 0.5, b: Math.random() });
}
}

function resetGame() {
ship = { x: W/2, y: H/2, angle: -Math.PI/2, radius: 14 };
bullets = [];
asteroids = [];
particles = [];
score = 0;
timeAlive = 0;
spawnTimer = 0;
difficulty = 1;
initStars();
}

function startGame() {
resetGame();
state = 'playing';
}

// Input
window.addEventListener('keydown', function(e) {
keys[e.code] = true;
if (!audioCtx) initAudio();
if (e.code === 'Space' || e.code === 'Enter') {
if (state === 'start' || state === 'gameover') {
startGame();
}
e.preventDefault();
}
});
window.addEventListener('keyup', function(e) {
keys[e.code] = false;
});

// Asteroid spawning
function spawnAsteroid() {
const size = Math.random() < 0.3 ? 50 : (Math.random() < 0.6 ? 35 : 20);
const side = Math.floor(Math.random() * 4);
let x, y;
if (side === 0) { x = -size; y = Math.random() * H; }
else if (side === 1) { x = W + size; y = Math.random() * H; }
else if (side === 2) { x = Math.random() * W; y = -size; }
else { x = Math.random() * W; y = H + size; }

const speed = (30 + Math.random() * 40) * difficulty / 60;
const angle = Math.atan2(H/2 - y, W/2 - x) + (Math.random() - 0.5) * 1.2;
const vx = Math.cos(angle) * speed;
const vy = Math.sin(angle) * speed;

// Generate asteroid shape
const verts = [];
const numVerts = 8 + Math.floor(Math.random() * 4);
for (let i = 0; i < numVerts; i++) {
const a = (i / numVerts) * Math.PI * 2;
const r = size * (0.7 + Math.random() * 0.3);
verts.push({ x: Math.cos(a) * r, y: Math.sin(a) * r });
}

asteroids.push({ x, y, vx, vy, radius: size, verts, rot: 0, rotSpeed: (Math.random()-0.5)*2 });
}

function createParticles(x, y, color, count) {
for (let i = 0; i < count; i++) {
const angle = Math.random() * Math.PI * 2;
const speed = Math.random() * 4 + 1;
particles.push({
x, y,
vx: Math.cos(angle) * speed,
vy: Math.sin(angle) * speed,
life: 1,
decay: 0.02 + Math.random() * 0.03,
color: color || '#fff',
size: Math.random() * 4 + 1
});
}
}

function update(dt) {
if (state !== 'playing') return;

timeAlive += dt;
difficulty = 1 + timeAlive / 30; // ramps up over time

// Ship movement
const thrust = 300 * difficulty;
const maxSpeed = 250;
const turnSpeed = 4;

if (keys['ArrowLeft'] || keys['KeyA']) ship.angle -= turnSpeed * dt;
if (keys['ArrowRight'] || keys['KeyD']) ship.angle += turnSpeed * dt;
if (keys['ArrowUp'] || keys['KeyW']) {
ship.x += Math.cos(ship.angle) * 200 * dt;
ship.y += Math.sin(ship.angle) * 200 * dt;
}
if (keys['ArrowDown'] || keys['KeyS']) {
ship.x -= Math.cos(ship.angle) * 100 * dt;
ship.y -= Math.sin(ship.angle) * 100 * dt;
}

// Wrap ship
if (ship.x < -20) ship.x = W + 20;
if (ship.x > W + 20) ship.x = -20;
if (ship.y < -20) ship.y = H + 20;
if (ship.y > H + 20) ship.y = -20;

// Shooting
if (keys['Space'] && Math.random() < 0.15) {
const bx = ship.x + Math.cos(ship.angle) * 18;
const by = ship.y + Math.sin(ship.angle) * 18;
bullets.push({ x: bx, y: by, vx: Math.cos(ship.angle) * 400, vy: Math.sin(ship.angle) * 400, life: 2 });
playSound(800, 0.05, 'square', 0.08);
}

// Update bullets
for (let i = bullets.length - 1; i >= 0; i--) {
const b = bullets[i];
b.x += b.vx * dt;
b.y += b.vy * dt;
b.life -= dt;
if (b.life <= 0 || b.x < -20 || b.x > W+20 || b.y < -20 || b.y > H+20) {
bullets.splice(i, 1);
}
}

// Spawn asteroids
const spawnRate = Math.max(0.3, 2 - timeAlive * 0.02);
spawnTimer += dt;
if (spawnTimer > spawnRate) {
spawnTimer = 0;
const count = 1 + Math.floor(Math.random() * Math.min(3, difficulty));
for (let i = 0; i < count; i++) spawnAsteroid();
}

// Update asteroids
for (let i = asteroids.length - 1; i >= 0; i--) {
const a = asteroids[i];
a.x += a.vx * dt * 60;
a.y += a.vy * dt * 60;
a.rot += a.rotSpeed * dt;

// Remove if way off screen
if (a.x < -200 || a.x > W+200 || a.y < -200 || a.y > H+200) {
asteroids.splice(i, 1);
}
}

// Collision: bullets vs asteroids
for (let i = bullets.length - 1; i >= 0; i--) {
const b = bullets[i];
for (let j = asteroids.length - 1; j >= 0; j--) {
const a = asteroids[j];
const dx = b.x - a.x, dy = b.y - a.y;
if (dx*dx + dy*dy < (a.radius * 0.8) * (a.radius * 0.8)) {
createParticles(a.x, a.y, '#aaa', 15);
score += Math.max(10, Math.floor(60 - a.radius));
asteroids.splice(j, 1);
bullets.splice(i, 1);
playSound(200 + Math.random()*100, 0.2, 'sawtooth', 0.12);
break;
}
}
}

// Collision: ship vs asteroids
for (let j = asteroids.length - 1; j >= 0; j--) {
const a = asteroids[j];
const dx = ship.x - a.x, dy = ship.y - a.y;
if (dx*dx + dy*dy < (ship.radius * 0.7 + a.radius * 0.6) * (ship.radius * 0.7 + a.radius * 0.6)) {
createParticles(ship.x, ship.y, '#4af', 30);
asteroids.splice(j, 1);
gameOver();
break;
}
}

// Update particles
for (let i = particles.length - 1; i >= 0; i--) {
const p = particles[i];
p.x += p.vx;
p.y += p.vy;
p.life -= p.decay;
if (p.life <= 0) particles.splice(i, 1);
}
}

function gameOver() {
state = 'gameover';
playSound(100, 0.5, 'sawtooth', 0.2);
try {
if (score > highScore) {
highScore = score;
localStorage.setItem('asteroidsEndlessHigh', String(highScore));
}
} catch(e) {}
}

function draw() {
ctx.fillStyle = '#0a0a1a';
ctx.fillRect(0, 0, W, H);

// Stars
for (const s of stars) {
const brightness = 0.3 + Math.sin(timeAlive * 2 + s.b * 10) * 0.3;
ctx.fillStyle = `rgba(255,255,255,${brightness})`;
ctx.fillRect(s.x, s.y, s.s, s.s);
}

if (state === 'playing' || state === 'gameover') {
// Particles
for (const p of particles) {
ctx.fillStyle = `rgba(${p.color}, ${p.life})`;
ctx.beginPath();
ctx.arc(p.x, p.y, p.size * p.life, 0, Math.PI*2);
ctx.fill();
}

// Asteroids
for (const a of asteroids) {
ctx.save();
ctx.translate(a.x, a.y);
ctx.rotate(a.rot);
ctx.beginPath();
ctx.moveTo(a.verts[0].x, a.verts[0].y);
for (let i = 1; i < a.verts.length; i++) {
ctx.lineTo(a.verts[i].x, a.verts[i].y);
}
ctx.closePath();
ctx.fillStyle = '#3a3a4a';
ctx.fill();
ctx.strokeStyle = '#888';
ctx.lineWidth = 1.5;
ctx.stroke();
ctx.restore();
}

// Bullets
for (const b of bullets) {
ctx.fillStyle = '#ff6644';
ctx.beginPath();
ctx.arc(b.x, b.y, 3, 0, Math.PI*2);
ctx.fill();
ctx.fillStyle = '#ffaa44';
ctx.beginPath();
ctx.arc(b.x, b.y, 1.5, 0, Math.PI*2);
ctx.fill();
}

// Ship
if (state === 'playing') {
ctx.save();
ctx.translate(ship.x, ship.y);
ctx.rotate(ship.angle);
ctx.beginPath();
ctx.moveTo(18, 0);
ctx.lineTo(-12, -10);
ctx.lineTo(-6, 0);
ctx.lineTo(-12, 10);
ctx.closePath();
ctx.fillStyle = '#4af';
ctx.fill();
ctx.strokeStyle = '#fff';
ctx.lineWidth = 1.5;
ctx.stroke();
ctx.restore();

// Engine flame
if (keys['ArrowUp'] || keys['KeyW']) {
const fx = ship.x - Math.cos(ship.angle) * 14;
const fy = ship.y - Math.sin(ship.angle) * 14;
createParticles(fx, fy, '#f80', 1);
}
}
}

// HUD
ctx.fillStyle = '#fff';
ctx.font = 'bold 20px monospace';
ctx.textAlign = 'left';
ctx.fillText('SCORE: ' + score, 20, 35);
ctx.fillText('TIME: ' + Math.floor(timeAlive) + 's', 20, 60);
ctx.font = '14px monospace';
ctx.fillStyle = '#888';
ctx.fillText('HI: ' + highScore, 20, 80);

if (state === 'start') {
drawStartScreen();
} else if (state === 'gameover') {
drawGameOverScreen();
}
}

function drawStartScreen() {
ctx.fillStyle = 'rgba(0,0,0,0.7)';
ctx.fillRect(0, 0, W, H);

ctx.textAlign = 'center';
ctx.fillStyle = '#fff';
ctx.font = 'bold 48px monospace';
ctx.fillText('ASTEROIDS ENDLESS', W/2, H/3);

ctx.font = '18px monospace';
ctx.fillStyle = '#aaa';
const lines = [
'← → or A/D : Rotate Ship',
'↑ or W : Thrust Forward',
'↓ or S : Move Back',
'SPACE : Shoot',
'',
'Survive as long as you can!',
'Difficulty increases over time.',
'',
'Press SPACE to Start'
];
lines.forEach((line, i) => {
ctx.fillText(line, W/2, H/3 + 50 + i * 28);
});

if (highScore > 0) {
ctx.fillStyle = '#ff0';
ctx.font = '16px monospace';
ctx.fillText('HIGH SCORE: ' + highScore, W/2, H - 40);
}
}

function drawGameOverScreen() {
ctx.fillStyle = 'rgba(0,0,0,0.6)';
ctx.fillRect(0, 0, W, H);

ctx.textAlign = 'center';
ctx.fillStyle = '#f44';
ctx.font = 'bold 42px monospace';
ctx.fillText('GAME OVER', W/2, H/3);

ctx.fillStyle = '#fff';
ctx.font = '24px monospace';
ctx.fillText('Score: ' + score, W/2, H/3 + 50);
ctx.fillText('Time Survived: ' + Math.floor(timeAlive) + 's', W/2, H/3 + 85);

if (score >= highScore && score > 0) {
ctx.fillStyle = '#ff0';
ctx.font = '20px monospace';
ctx.fillText('NEW HIGH SCORE!', W/2, H/3 + 125);
}

ctx.fillStyle = '#aaa';
ctx.font = '18px monospace';
ctx.fillText('Press SPACE to Restart', W/2, H/3 + 170);
}

function gameLoop(timestamp) {
const dt = Math.min((timestamp - lastTime) / 1000, 0.05);
lastTime = timestamp;

if (state === 'playing') {
update(dt);
}

draw();
requestAnimationFrame(gameLoop);
}

resetGame();
initStars();
requestAnimationFrame(function(t) { lastTime = t; gameLoop(t); });

})();
</script>
</body>
</html>
```
