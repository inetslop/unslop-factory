<!DOCTYPE html>

<script>
const canvas = document.getElementById('game');
const ctx = canvas.getContext('2d');

let W, H;
function resize() {
    W = window.innerWidth;
    H = window.innerHeight;
    canvas.width = W;
    canvas.height = H;
}
window.addEventListener('resize', resize);
resize();

// Audio
let audioCtx = null;
function initAudio() {
    if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
}
function playSound(type) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    if (type === 'shoot') {
        osc.type = 'square';
        osc.frequency.setValueAtTime(200, audioCtx.currentTime);
        osc.frequency.exponentialRampToValueAtTime(50, audioCtx.currentTime + 0.1);
        gain.gain.setValueAtTime(0.15, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.1);
        osc.start(); osc.stop(audioCtx.currentTime + 0.1);
    } else if (type === 'hit') {
        osc.type = 'sawtooth';
        osc.frequency.setValueAtTime(150, audioCtx.currentTime);
        osc.frequency.exponentialRampToValueAtTime(30, audioCtx.currentTime + 0.15);
        gain.gain.setValueAtTime(0.2, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.15);
        osc.start(); osc.stop(audioCtx.currentTime + 0.15);
    } else if (type === 'death') {
        osc.type = 'sawtooth';
        osc.frequency.setValueAtTime(100, audioCtx.currentTime);
        osc.frequency.exponentialRampToValueAtTime(20, audioCtx.currentTime + 0.4);
        gain.gain.setValueAtTime(0.25, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.4);
        osc.start(); osc.stop(audioCtx.currentTime + 0.4);
    } else if (type === 'wave') {
        osc.type = 'sine';
        osc.frequency.setValueAtTime(300, audioCtx.currentTime);
        osc.frequency.exponentialRampToValueAtTime(600, audioCtx.currentTime + 0.2);
        gain.gain.setValueAtTime(0.15, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.3);
        osc.start(); osc.stop(audioCtx.currentTime + 0.3);
    } else if (type === 'gameover') {
        osc.type = 'sawtooth';
        osc.frequency.setValueAtTime(200, audioCtx.currentTime);
        osc.frequency.exponentialRampToValueAtTime(30, audioCtx.currentTime + 1);
        gain.gain.setValueAtTime(0.3, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 1);
        osc.start(); osc.stop(audioCtx.currentTime + 1);
    }
}

// Game State
let gameState = 'start'; // start, playing, gameover
let score = 0;
let wave = 1;
let zombiesToSpawn = 0;
let spawnTimer = 0;
let frameCount = 0;

const keys = {};
window.addEventListener('keydown', e => {
    keys[e.key.toLowerCase()] = true;
    if (gameState === 'start' && (e.key === ' ' || e.key === 'Enter')) startGame();
    if (gameState === 'gameover' && (e.key === ' ' || e.key === 'Enter')) restart();
});
window.addEventListener('keyup', e => { keys[e.key.toLowerCase()] = false; });

// Particles
let particles = [];
function addParticles(x, y, color, count) {
    for (let i = 0; i < count; i++) {
        const angle = Math.random() * Math.PI * 2;
        const speed = Math.random() * 4 + 1;
        particles.push({ x, y, vx: Math.cos(angle) * speed, vy: Math.sin(angle) * speed, life: 30 + Math.random() * 20, color });
    }
}

// Players
let players = [];
function createPlayer(x, y, id, color, controls) {
    return { x, y, id, color, hp: 100, maxHp: 100, alive: true, radius: 18, shootCooldown: 0, controls, dead: false };
}

// Zombies
let zombies = [];
function spawnZombie() {
    const side = Math.floor(Math.random() * 4);
    let x, y;
    if (side === 0) { x = -30; y = Math.random() * H; }
    else if (side === 1) { x = W + 30; y = Math.random() * H; }
    else if (side === 2) { x = Math.random() * W; y = -30; }
    else { x = Math.random() * W; y = H + 30; }
    const isElite = wave > 3 && Math.random() < 0.15;
    zombies.push({
        x, y, hp: isElite ? 6 : (wave >= 5 ? 3 : 2), maxHp: isElite ? 6 : (wave >= 5 ? 3 : 2),
        radius: isElite ? 22 : 16, speed: (isElite ? 0.8 : 1) + wave * 0.05,
        color: isElite ? '#ff4444' : '#44aa44', elite: isElite, wobble: Math.random() * Math.PI * 2
    });
}

// Bullets
let bullets = [];

function startGame() {
    initAudio();
    gameState = 'playing';
    score = 0;
    wave = 1;
    frameCount = 0;
    zombiesToSpawn = 5;
    spawnTimer = 0;
    players = [
        createPlayer(W / 3, H / 2, 0, '#4488ff', { up: 'w', down: 's', left: 'a', right: 'd', shoot: 'f' }),
        createPlayer(2 * W / 3, H / 2, 1, '#ff8844', { up: 'arrowup', down: 'arrowdown', left: 'arrowleft', right: 'arrowright', shoot: 'l' })
    ];
    zombies = [];
    bullets = [];
    particles = [];
    playSound('wave');
}

function restart() { startGame(); }

function nextWave() {
    wave++;
    zombiesToSpawn = 4 + wave * 2;
    spawnTimer = 0;
    playSound('wave');
}

function update() {
    if (gameState !== 'playing') return;
    frameCount++;

    // Spawn zombies
    if (zombiesToSpawn > 0) {
        spawnTimer++;
        const interval = Math.max(20, 60 - wave * 3);
        if (spawnTimer >= interval) {
            spawnZombie();
            zombiesToSpawn--;
            spawnTimer = 0;
        }
    } else if (zombies.length === 0) {
        nextWave();
    }

    // Update players
    let anyAlive = false;
    for (const p of players) {
        if (!p.alive) continue;
        anyAlive = true;
        const speed = 4;
        if (keys[p.controls.up]) p.y -= speed;
        if (keys[p.controls.down]) p.y += speed;
        if (keys[p.controls.left]) p.x -= speed;
        if (keys[p.controls.right]) p.x += speed;
        p.x = Math.max(p.radius, Math.min(W - p.radius, p.x));
        p.y = Math.max(p.radius, Math.min(H - p.radius, p.y));

        if (p.shootCooldown > 0) p.shootCooldown--;
        if (keys[p.controls.shoot] && p.shootCooldown <= 0) {
            let dx = 0, dy = 0;
            if (keys[p.controls.left]) dx -= 1;
            if (keys[p.controls.right]) dx += 1;
            if (keys[p.controls.up]) dy -= 1;
            if (keys[p.controls.down]) dy += 1;
            if (dx === 0 && dy === 0) {
                // Default shoot direction: away from center or random
                const angle = Math.random() * Math.PI * 2;
                dx = Math.cos(angle); dy = Math.sin(angle);
            }
            const len = Math.sqrt(dx * dx + dy * dy);
            bullets.push({ x: p.x, y: p.y, vx: (dx / len) * 8, vy: (dy / len) * 8, radius: 4, color: p.color });
            p.shootCooldown = 12;
            playSound('shoot');
        }

        if (p.hp <= 0 && !p.dead) {
            p.dead = true;
            p.alive = false;
            addParticles(p.x, p.y, p.color, 30);
            playSound('death');
        }
    }

    // Update bullets
    for (let i = bullets.length - 1; i >= 0; i--) {
        const b = bullets[i];
        b.x += b.vx;
        b.y += b.vy;
        if (b.x < -20 || b.x > W + 20 || b.y < -20 || b.y > H + 20) {
            bullets.splice(i, 1);
            continue;
        }
        // Hit zombie
        for (let j = zombies.length - 1; j >= 0; j--) {
            const z = zombies[j];
            const dx = b.x - z.x, dy = b.y - z.y;
            if (Math.sqrt(dx * dx + dy * dy) < b.radius + z.radius) {
                z.hp--;
                addParticles(z.x, z.y, '#88ff44', 5);
                playSound('hit');
                bullets.splice(i, 1);
                if (z.hp <= 0) {
                    score += z.elite ? 25 : 10;
                    addParticles(z.x, z.y, z.color, 15);
                    zombies.splice(j, 1);
                }
                break;
            }
        }
    }

    // Update zombies
    for (const z of zombies) {
        z.wobble += 0.05;
        // Find nearest alive player
        let target = null, minDist = Infinity;
        for (const p of players) {
            if (!p.alive) continue;
            const dx = p.x - z.x, dy = p.y - z.y;
            const d = Math.sqrt(dx * dx + dy * dy);
            if (d < minDist) { minDist = d; target = p; }
        }
        if (!target) continue;
        const dx = target.x - z.x, dy = target.y - z.y;
        const len = Math.sqrt(dx * dx + dy * dy);
        if (len > 0) {
            z.x += (dx / len) * z.speed;
            z.y += (dy / len) * z.speed;
        }
        // Check collision with players
        for (const p of players) {
            if (!p.alive) continue;
            const pdx = p.x - z.x, pdy = p.y - z.y;
            if (Math.sqrt(pdx * pdx + pdy * pdy) < p.radius + z.radius) {
                p.hp -= z.elite ? 15 : 8;
                addParticles(p.x, p.y, '#ff0000', 5);
                // Knock zombie back a bit
                const kd = Math.sqrt(pdx * pdx + pdy * pdy) || 1;
                z.x -= (pdx / kd) * 20;
                z.y -= (pdy / kd) * 20;
            }
        }
    }

    // Update particles
    for (let i = particles.length - 1; i >= 0; i--) {
        const pt = particles[i];
        pt.x += pt.vx;
        pt.y += pt.vy;
        pt.vx *= 0.92;
        pt.vy *= 0.92;
        pt.life--;
        if (pt.life <= 0) particles.splice(i, 1);
    }

    // Game over check
    if (!anyAlive) {
        gameState = 'gameover';
        playSound('gameover');
    }
}

function draw() {
    ctx.fillStyle = '#1a1a2e';
    ctx.fillRect(0, 0, W, H);

    // Grid
    ctx.strokeStyle = '#252540';
    ctx.lineWidth = 1;
    for (let x = 0; x < W; x += 50) { ctx.beginPath(); ctx.moveTo(x, 0); ctx.lineTo(x, H); ctx.stroke(); }
    for (let y = 0; y < H; y += 50) { ctx.beginPath(); ctx.moveTo(0, y); ctx.lineTo(W, y); ctx.stroke(); }

    if (gameState === 'start') {
        drawStartScreen();
        return;
    }

    // Particles
    for (const pt of particles) {
        ctx.globalAlpha = pt.life / 50;
        ctx.fillStyle = pt.color;
        ctx.beginPath();
        ctx.arc(pt.x, pt.y, 3, 0, Math.PI * 2);
        ctx.fill();
    }
    ctx.globalAlpha = 1;

    // Bullets
    for (const b of bullets) {
        ctx.fillStyle = b.color;
        ctx.beginPath();
        ctx.arc(b.x, b.y, b.radius, 0, Math.PI * 2);
        ctx.fill();
    }

    // Zombies
    for (const z of zombies) {
        const wobbleX = Math.sin(z.wobble) * 3;
        ctx.fillStyle = z.color;
        ctx.beginPath();
        ctx.arc(z.x + wobbleX, z.y, z.radius, 0, Math.PI * 2);
        ctx.fill();
        // Eyes
        ctx.fillStyle = '#ff0';
        const eyeOff = z.elite ? 5 : 4;
        ctx.beginPath();
        ctx.arc(z.x + wobbleX - eyeOff, z.y - 3, 3, 0, Math.PI * 2);
        ctx.fill();
        ctx.beginPath();
        ctx.arc(z.x + wobbleX + eyeOff, z.y - 3, 3, 0, Math.PI * 2);
        ctx.fill();
        // HP bar
        if (z.hp < z.maxHp) {
            ctx.fillStyle = '#333';
            ctx.fillRect(z.x - 15, z.y - z.radius - 10, 30, 5);
            ctx.fillStyle = '#f44';
            ctx.fillRect(z.x - 15, z.y - z.radius - 10, 30 * (z.hp / z.maxHp), 5);
        }
    }

    // Players
    for (const p of players) {
        if (!p.alive) {
            ctx.fillStyle = '#555';
            ctx.beginPath();
            ctx.arc(p.x, p.y, p.radius, 0, Math.PI * 2);
            ctx.fill();
            ctx.fillStyle = '#fff';
            ctx.font = '14px monospace';
            ctx.textAlign = 'center';
            ctx.fillText('X', p.x, p.y + 5);
            continue;
        }
        ctx.fillStyle = p.color;
        ctx.beginPath();
        ctx.arc(p.x, p.y, p.radius, 0, Math.PI * 2);
        ctx.fill();
        // Face
        ctx.fillStyle = '#fff';
        ctx.beginPath();
        ctx.arc(p.x - 5, p.y - 4, 3, 0, Math.PI * 2);
        ctx.fill();
        ctx.beginPath();
        ctx.arc(p.x + 5, p.y - 4, 3, 0, Math.PI * 2);
        ctx.fill();
        // HP bar
        ctx.fillStyle = '#333';
        ctx.fillRect(p.x - 20, p.y - p.radius - 12, 40, 6);
        ctx.fillStyle = p.hp > 50 ? '#4f4' : (p.hp > 25 ? '#ff4' : '#f44');
        ctx.fillRect(p.x - 20, p.y - p.radius - 12, 40 * (p.hp / p.maxHp), 6);
    }

    // HUD
    ctx.fillStyle = '#fff';
    ctx.font = 'bold 18px monospace';
    ctx.textAlign = 'left';
    ctx.fillText(`WAVE ${wave}`, 20, 30);
    ctx.fillText(`SCORE: ${score}`, 20, 55);
    ctx.textAlign = 'right';
    ctx.fillText(`ZOMBIES: ${zombies.length + zombiesToSpawn}`, W - 20, 30);

    if (gameState === 'gameover') {
        drawGameOver();
    }
}

function drawStartScreen() {
    ctx.fillStyle = '#1a1a2e';
    ctx.fillRect(0, 0, W, H);

    // Title
    ctx.fillStyle = '#44aa44';
    ctx.font = 'bold 60px monospace';
    ctx.textAlign = 'center';
    ctx.fillText('ZOMBIE SQUAD', W / 2, H / 3);

    ctx.fillStyle = '#fff';
    ctx.font = '18px monospace';
    ctx.fillText('A Two-Player Zombie Survival Game', W / 2, H / 3 + 40);

    ctx.fillStyle = '#888';
    ctx.font = '16px monospace';
    const lines = [
        '',
        'PLAYER 1 (Blue)',
        '  Move: W A S D',
        '  Shoot: F (shoots in movement direction)',
        '',
        'PLAYER 2 (Orange)',
        '  Move: Arrow Keys',
        '  Shoot: L (shoots in movement direction)',
        '',
        'Survive the zombie horde! Waves get harder each round.',
        'Don\'t let zombies touch you - they deal damage!',
        '',
        'Press SPACE or ENTER to Start'
    ];
    const startY = H / 2 + 20;
    for (let i = 0; i < lines.length; i++) {
        ctx.fillText(lines[i], W / 2, startY + i * 24);
    }

    // Animated zombies on start screen
    const t = Date.now() / 1000;
    for (let i = 0; i < 5; i++) {
        const zx = W / 2 - 150 + i * 70 + Math.sin(t + i) * 10;
        const zy = H - 60 + Math.cos(t * 1.5 + i) * 5;
        ctx.fillStyle = '#44aa44';
        ctx.beginPath();
        ctx.arc(zx, zy, 16, 0, Math.PI * 2);
        ctx.fill();
        ctx.fillStyle = '#ff0';
        ctx.beginPath();
        ctx.arc(zx - 4, zy - 3, 2, 0, Math.PI * 2);
        ctx.fill();
        ctx.beginPath();
        ctx.arc(zx + 4, zy - 3, 2, 0, Math.PI * 2);
        ctx.fill();
    }
}

function drawGameOver() {
    ctx.fillStyle = 'rgba(0,0,0,0.7)';
    ctx.fillRect(0, 0, W, H);
    ctx.fillStyle = '#f44';
    ctx.font = 'bold 50px monospace';
    ctx.textAlign = 'center';
    ctx.fillText('GAME OVER', W / 2, H / 3);
    ctx.fillStyle = '#fff';
    ctx.font = '24px monospace';
    ctx.fillText(`Final Score: ${score}`, W / 2, H / 3 + 50);
    ctx.fillText(`Waves Survived: ${wave}`, W / 2, H / 3 + 85);
    ctx.fillStyle = '#aaa';
    ctx.font = '18px monospace';
    ctx.fillText('Press SPACE or ENTER to Restart', W / 2, H / 2 + 60);
}

function loop() {
    update();
    draw();
    requestAnimationFrame(loop);
}
loop();
</script>
