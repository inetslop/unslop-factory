```html
<!DOCTYPE html>

<canvas id="game"></canvas>
<script>
const canvas = document.getElementById('game');
const ctx = canvas.getContext('2d');

let W, H;
function resize() {
    W = canvas.width = window.innerWidth;
    H = canvas.height = window.innerHeight;
}
window.addEventListener('resize', resize);
resize();

// Audio
let audioCtx = null;
function initAudio() {
    if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
}
function playSound(freq, type, duration, vol) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = type;
    osc.frequency.value = freq;
    gain.gain.setValueAtTime(vol || 0.15, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + duration);
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + duration);
}
function sfxJump() { playSound(280, 'square', 0.15, 0.1); }
function sfxGem() { playSound(880, 'sine', 0.2, 0.15); setTimeout(() => playSound(1320, 'sine', 0.15, 0.1), 60); }
function sfxDeath() { playSound(120, 'sawtooth', 0.4, 0.2); }
function sfxSpike() { playSound(200, 'square', 0.3, 0.15); }

// Game state
let gameState = 'start'; // start, playing, dead
let score = 0;
let gems = 0;
let distance = 0;
let highScore = 0;
try { highScore = parseInt(localStorage.getItem('gravityRunnerHigh') || '0'); } catch(e) {}

// Input
const keys = {};
window.addEventListener('keydown', e => {
    keys[e.code] = true;
    if (gameState === 'start' && (e.code === 'Space' || e.code === 'Enter')) startGame();
    if (gameState === 'dead' && (e.code === 'Space' || e.code === 'Enter' || e.code === 'KeyR')) startGame();
});
window.addEventListener('keyup', e => { keys[e.code] = false; });

// Player
let player = {};
function resetPlayer() {
    player = {
        x: 100, y: 300, vx: 0, vy: 0,
        w: 32, h: 40, grounded: false, dead: false,
        runFrame: 0, facing: 1
    };
}

// World
let platforms = [];
let spikes = [];
let gemItems = [];
let particles = [];
let camera = { x: 0, y: 0 };
let worldEnd = 0;
let timeScale = 1;
let deathTimer = 0;
let shakeAmount = 0;

function generateWorld(startX) {
    let lastY = H - 150;
    let lastX = startX;
    for (let i = 0; i < 60; i++) {
        const gap = 80 + Math.random() * 120;
        const w = 120 + Math.random() * 180;
        const yDiff = (Math.random() - 0.5) * 120;
        lastY += yDiff;
        lastY = Math.max(100, Math.min(H - 100, lastY));
        lastX += gap;
        platforms.push({ x: lastX, y: lastY, w: w, h: 24 });

        // Spikes on some platforms
        if (Math.random() < 0.35 && i > 2) {
            const sx = lastX + 20 + Math.random() * (w - 60);
            spikes.push({ x: sx, y: lastY - 20, w: 24, h: 20 });
        }

        // Gems
        if (Math.random() < 0.5) {
            const gx = lastX + w / 2;
            const gy = lastY - 60 - Math.random() * 60;
            gemItems.push({ x: gx, y: gy, w: 18, h: 18, collected: false, bobOffset: Math.random() * Math.PI * 2 });
        }
    }
    worldEnd = lastX + w + 500;
}

function startGame() {
    initAudio();
    gameState = 'playing';
    score = 0; gems = 0; distance = 0;
    platforms = []; spikes = []; gemItems = []; particles = [];
    resetPlayer();
    player.y = H - 200;
    // Starting platform
    platforms.push({ x: 0, y: H - 150, w: 300, h: 24 });
    generateWorld(200);
    camera.x = 0; camera.y = 0;
    shakeAmount = 0;
}

// Particles
function spawnParticles(x, y, color, count) {
    for (let i = 0; i < count; i++) {
        const angle = Math.random() * Math.PI * 2;
        const speed = 1 + Math.random() * 4;
        particles.push({
            x, y, vx: Math.cos(angle) * speed, vy: Math.sin(angle) * speed - 2,
            life: 1, decay: 0.02 + Math.random() * 0.03, color, size: 3 + Math.random() * 4
        });
    }
}

// Stars background
let stars = [];
for (let i = 0; i < 100; i++) {
    stars.push({ x: Math.random() * 2000, y: Math.random() * 600, size: 1 + Math.random() * 2, speed: 0.2 + Math.random() * 0.5 });
}

function update(dt) {
    if (gameState !== 'playing' && gameState !== 'dead') return;

    const gravity = 0.6;
    const friction = 0.85;
    const moveSpeed = 0.7;
    const maxSpeed = 12;
    const jumpForce = -13;

    if (gameState === 'playing') {
        // Input
        let moveX = 0;
        if (keys['ArrowLeft'] || keys['KeyA']) { moveX = -moveSpeed; player.facing = -1; }
        if (keys['ArrowRight'] || keys['KeyD']) { moveX = moveSpeed; player.facing = 1; }

        player.vx += moveX;
        player.vx *= friction;
        if (Math.abs(player.vx) > maxSpeed) player.vx = Math.sign(player.vx) * maxSpeed;
        if (Math.abs(player.vx) < 0.1) player.vx = 0;

        // Jump
        if ((keys['Space'] || keys['ArrowUp'] || keys['KeyW']) && player.grounded) {
            player.vy = jumpForce;
            player.grounded = false;
            sfxJump();
            spawnParticles(player.x + player.w/2, player.y + player.h, '#4af', 5);
        }

        // Physics
        player.vy += gravity;
        if (player.vy > 18) player.vy = 18;

        player.x += player.vx;
        player.y += player.vy;

        // Run animation
        if (Math.abs(player.vx) > 0.5 && player.grounded) {
            player.runFrame += Math.abs(player.vx) * dt * 0.3;
        }

        // Platform collision
        player.grounded = false;
        for (let p of platforms) {
            if (player.x + player.w > p.x && player.x < p.x + p.w) {
                // Landing on top
                if (player.vy >= 0 && player.y + player.h >= p.y && player.y + player.h - player.vy <= p.y + 10) {
                    player.y = p.y - player.h;
                    player.vy = 0;
                    player.grounded = true;
                }
            }
        }

        // Spike collision
        for (let s of spikes) {
            if (player.x + player.w > s.x && player.x < s.x + s.w &&
                player.y + player.h > s.y && player.y < s.y + s.h) {
                killPlayer();
                return;
            }
        }

        // Gem collection
        for (let g of gemItems) {
            if (!g.collected) {
                const dx = (player.x + player.w/2) - (g.x + g.w/2);
                const dy = (player.y + player.h/2) - (g.y + g.h/2);
                if (Math.abs(dx) < 30 && Math.abs(dy) < 35) {
                    g.collected = true;
                    gems++;
                    score += 10;
                    sfxGem();
                    spawnParticles(g.x + g.w/2, g.y + g.h/2, '#ff0', 8);
                }
            }
        }

        // Fall death
        if (player.y > H + 50) {
            killPlayer();
            return;
        }

        // Distance score
        distance = Math.max(distance, Math.floor(player.x / 10));
        score += 0; // gems give score

        // Camera
        camera.x += (player.x - W * 0.35 - camera.x) * 0.08;
        camera.y += (player.y - H * 0.5 - camera.y) * 0.05;
        if (camera.y > 100) camera.y = 100;
    }

    if (gameState === 'dead') {
        deathTimer += dt;
        shakeAmount = Math.max(0, shakeAmount - dt * 2);
    }

    // Particles
    for (let i = particles.length - 1; i >= 0; i--) {
        const p = particles[i];
        p.x += p.vx;
        p.y += p.vy;
        p.vy += 0.3;
        p.life -= p.decay;
        if (p.life <= 0) particles.splice(i, 1);
    }

    // Generate more world
    if (gameState === 'playing') {
        const rightEdge = camera.x + W * 2;
        if (worldEnd - rightEdge < 800) {
            generateWorld(worldEnd);
        }
    }
}

function killPlayer() {
    gameState = 'dead';
    player.dead = true;
    sfxDeath();
    shakeAmount = 10;
    spawnParticles(player.x + player.w/2, player.y + player.h/2, '#f44', 20);
    if (score > highScore) {
        highScore = score;
        try { localStorage.setItem('gravityRunnerHigh', String(highScore)); } catch(e) {}
    }
}

function drawBackground() {
    // Gradient sky
    const grad = ctx.createLinearGradient(0, 0, 0, H);
    grad.addColorStop(0, '#0a0a2e');
    grad.addColorStop(1, '#1a1a3e');
    ctx.fillStyle = grad;
    ctx.fillRect(0, 0, W, H);

    // Stars (parallax)
    ctx.fillStyle = '#fff';
    for (let s of stars) {
        const sx = ((s.x - camera.x * s.speed * 0.1) % (W + 100) + W + 100) % (W + 100) - 50;
        const sy = (s.y - camera.y * 0.05) % H;
        ctx.globalAlpha = 0.3 + Math.sin(Date.now() * 0.001 + s.x) * 0.2;
        ctx.beginPath();
        ctx.arc(sx, sy, s.size, 0, Math.PI * 2);
        ctx.fill();
    }
    ctx.globalAlpha = 1;

    // Distant mountains (parallax layer)
    ctx.fillStyle = '#1a1a4e';
    for (let i = 0; i < 8; i++) {
        const mx = ((i * 300 - camera.x * 0.1) % (W + 600) + W + 600) % (W + 600) - 300;
        const my = H - 80 + Math.sin(i * 2.5) * 40;
        ctx.beginPath();
        ctx.moveTo(mx, my);
        ctx.lineTo(mx + 150, my - 100 - (i % 3) * 40);
        ctx.lineTo(mx + 300, my);
        ctx.fill();
    }

    // Mid mountains
    ctx.fillStyle = '#22226e';
    for (let i = 0; i < 10; i++) {
        const mx = ((i * 250 - camera.x * 0.2) % (W + 500) + W + 500) % (W + 500) - 250;
        const my = H - 30 + Math.sin(i * 1.7) * 30;
        ctx.beginPath();
        ctx.moveTo(mx, my);
        ctx.lineTo(mx + 120, my - 60 - (i % 4) * 20);
        ctx.lineTo(mx + 250, my);
        ctx.fill();
    }
}

function drawPlatform(p) {
    const px = p.x - camera.x;
    const py = p.y - camera.y;
    if (px > W + 50 || px + p.w < -50) return;

    // Platform body
    ctx.fillStyle = '#3a3a6e';
    ctx.fillRect(px, py, p.w, p.h);
    // Top highlight
    ctx.fillStyle = '#6a6aff';
    ctx.fillRect(px, py, p.w, 4);
    // Bottom shadow
    ctx.fillStyle = '#2a2a4e';
    ctx.fillRect(px, py + p.h - 4, p.w, 4);
}

function drawSpike(s) {
    const sx = s.x - camera.x;
    const sy = s.y - camera.y;
    if (sx > W + 50 || sx < -50) return;

    ctx.fillStyle = '#ff3333';
    ctx.beginPath();
    ctx.moveTo(sx, sy + s.h);
    ctx.lineTo(sx + s.w / 2, sy);
    ctx.lineTo(sx + s.w, sy + s.h);
    ctx.closePath();
    ctx.fill();

    ctx.fillStyle = '#ff6666';
    ctx.beginPath();
    ctx.moveTo(sx + 4, sy + s.h);
    ctx.lineTo(sx + s.w / 2, sy + 4);
    ctx.lineTo(sx + s.w - 4, sy + s.h);
    ctx.closePath();
    ctx.fill();
}

function drawGem(g) {
    if (g.collected) return;
    const bob = Math.sin(Date.now() * 0.003 + g.bobOffset) * 5;
    const gx = g.x - camera.x;
    const gy = g.y - camera.y + bob;
    if (gx > W + 50 || gx < -50) return;

    ctx.fillStyle = '#ffdd00';
    ctx.beginPath();
    ctx.moveTo(gx + 9, gy);
    ctx.lineTo(gx + 18, gy + 9);
    ctx.lineTo(gx + 9, gy + 18);
    ctx.lineTo(gx, gy + 9);
    ctx.closePath();
    ctx.fill();

    ctx.fillStyle = '#ffffaa';
    ctx.beginPath();
    ctx.moveTo(gx + 9, gy + 3);
    ctx.lineTo(gx + 14, gy + 9);
    ctx.lineTo(gx + 9, gy + 14);
    ctx.lineTo(gx + 4, gy + 9);
    ctx.closePath();
    ctx.fill();

    // Glow
    ctx.globalAlpha = 0.3;
    ctx.fillStyle = '#ffdd00';
    ctx.beginPath();
    ctx.arc(gx + 9, gy + 9, 14, 0, Math.PI * 2);
    ctx.fill();
    ctx.globalAlpha = 1;
}

function drawPlayer() {
    const px = player.x - camera.x;
    const py = player.y - camera.y;

    if (gameState === 'dead') return;

    // Body
    ctx.fillStyle = '#4488ff';
    ctx.fillRect(px, py + 10, player.w, player.h - 10);

    // Head
    ctx.fillStyle = '#66aaff';
    ctx.fillRect(px + 4, py, player.w - 8, 18);

    // Eyes
    const eyeOff = player.facing * 4;
    ctx.fillStyle = '#fff';
    ctx.fillRect(px + 10 + eyeOff, py + 4, 6, 6);
    ctx.fillRect(px + 20 + eyeOff, py + 4, 6, 6);
    ctx.fillStyle = '#000';
    ctx.fillRect(px + 12 + eyeOff, py + 6, 3, 3);
    ctx.fillRect(px + 22 + eyeOff, py + 6, 3, 3);

    // Legs (animation)
    const legOffset = Math.sin(player.runFrame * 0.5) * 6;
    ctx.fillStyle = '#3366cc';
    if (!player.grounded) {
        ctx.fillRect(px + 4, py + player.h - 12, 10, 12);
        ctx.fillRect(px + 18, py + player.h - 12, 10, 12);
    } else {
        ctx.fillRect(px + 4 + legOffset, py + player.h - 12, 10, 12);
        ctx.fillRect(px + 18 - legOffset, py + player.h - 12, 10, 12);
    }

    // Glow trail
    if (Math.abs(player.vx) > 3) {
        ctx.globalAlpha = 0.3;
        ctx.fillStyle = '#4488ff';
        ctx.fillRect(px - player.vx * 2, py + 10, player.w, player.h - 10);
        ctx.globalAlpha = 1;
    }
}

function drawParticles() {
    for (let p of particles) {
        const px = p.x - camera.x;
        const py = p.y - camera.y;
        ctx.globalAlpha = p.life;
        ctx.fillStyle = p.color;
        ctx.fillRect(px - p.size/2, py - p.size/2, p.size, p.size);
    }
    ctx.globalAlpha = 1;
}

function drawHUD() {
    // Score
    ctx.fillStyle = '#fff';
    ctx.font = 'bold 24px monospace';
    ctx.textAlign = 'left';
    ctx.fillText(`SCORE: ${score}`, 20, 35);
    ctx.fillText(`GEMS: ${gems}`, 20, 65);
    ctx.fillText(`DIST: ${distance}m`, 20, 95);

    // High score
    ctx.textAlign = 'right';
    ctx.fillStyle = '#aaa';
    ctx.font = '16px monospace';
    ctx.fillText(`BEST: ${highScore}`, W - 20, 35);
}

function drawStartScreen() {
    drawBackground();

    ctx.fillStyle = 'rgba(0,0,0,0.5)';
    ctx.fillRect(0, 0, W, H);

    // Title
    ctx.textAlign = 'center';
    ctx.font = 'bold 64px monospace';
    ctx.fillStyle = '#4af';
    ctx.fillText('GRAVITY', W/2, H/2 - 100);
    ctx.fillStyle = '#ff0';
    ctx.fillText('RUNNER', W/2, H/2 - 40);

    // Controls
    ctx.font = '20px monospace';
    ctx.fillStyle = '#ccc';
    ctx.fillText('← → / A D  —  Move', W/2, H/2 + 30);
    ctx.fillText('SPACE / ↑ / W  —  Jump', W/2, H/2 + 60);
    ctx.fillText('Collect gems • Avoid spikes • Don\'t fall!', W/2, H/2 + 100);

    // Start prompt
    const pulse = 0.5 + Math.sin(Date.now() * 0.005) * 0.5;
    ctx.globalAlpha = pulse;
    ctx.font = 'bold 28px monospace';
    ctx.fillStyle = '#fff';
    ctx.fillText('PRESS SPACE TO START', W/2, H/2 + 170);
    ctx.globalAlpha = 1;

    if (highScore > 0) {
        ctx.font = '16px monospace';
        ctx.fillStyle = '#888';
        ctx.fillText(`High Score: ${highScore}`, W/2, H/2 + 220);
    }
}

function drawDeathScreen() {
    // Red flash
    const alpha = Math.max(0, 1 - deathTimer * 2) * 0.4;
    if (alpha > 0) {
        ctx.fillStyle = `rgba(255,0,0,${alpha})`;
        ctx.fillRect(0, 0, W, H);
    }

    // Overlay
    const overlayAlpha = Math.min(deathTimer * 3, 0.6);
    ctx.fillStyle = `rgba(0,0,0,${overlayAlpha})`;
    ctx.fillRect(0, 0, W, H);

    if (deathTimer > 0.5) {
        ctx.textAlign = 'center';
        ctx.font = 'bold 48px monospace';
        ctx.fillStyle = '#f44';
        ctx.fillText('GAME OVER', W/2, H/2 - 40);

        ctx.font = '24px monospace';
        ctx.fillStyle = '#fff';
        ctx.fillText(`Score: ${score}   Gems: ${gems}   Distance: ${distance}m`, W/2, H/2 + 10);

        if (score >= highScore && score > 0) {
            ctx.fillStyle = '#ff0';
            ctx.fillText('★ NEW HIGH SCORE! ★', W/2, H/2 + 50);
        }

        const pulse = 0.5 + Math.sin(Date.now() * 0.005) * 0.5;
        ctx.globalAlpha = pulse;
        ctx.font = '22px monospace';
        ctx.fillStyle = '#aaa';
        ctx.fillText('PRESS SPACE TO RESTART', W/2, H/2 + 100);
        ctx.globalAlpha = 1;
    }
}

let lastTime = 0;
function gameLoop(timestamp) {
    const dt = Math.min((timestamp - lastTime) / 16.67, 3);
    lastTime = timestamp;

    update(dt);

    // Screen shake
    ctx.save();
    if (shakeAmount > 0) {
        const sx = (Math.random() - 0.5) * shakeAmount;
        const sy = (Math.random() - 0.5) * shakeAmount;
        ctx.translate(sx, sy);
    }

    drawBackground();

    if (gameState === 'start') {
        drawStartScreen();
    } else {
        // Draw world
        for (let p of platforms) drawPlatform(p);
        for (let s of spikes) drawSpike(s);
        for (let g of gemItems) drawGem(g);
        drawPlayer();
        drawParticles();
        drawHUD();

        if (gameState === 'dead') drawDeathScreen();
    }

    ctx.restore();

    requestAnimationFrame(gameLoop);
}

resetPlayer();
requestAnimationFrame(gameLoop);
</script>
</body>
</html>
```
