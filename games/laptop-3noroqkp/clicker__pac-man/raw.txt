<script>
const canvas = document.getElementById('gameCanvas');
const ctx = canvas.getContext('2d');
let W, H;

function resize() {
  W = window.innerWidth; H = window.innerHeight;
  canvas.width = W; canvas.height = H;
}
window.addEventListener('resize', resize);
resize();

// Audio
let audioCtx = null;
function initAudio() { if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)(); }
function playSound(freq, dur, type, vol) {
  if (!audioCtx) return;
  const osc = audioCtx.createOscillator();
  const gain = audioCtx.createGain();
  osc.type = type || 'square';
  osc.frequency.value = freq;
  gain.gain.setValueAtTime(vol || 0.1, audioCtx.currentTime);
  gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
  osc.connect(gain); gain.connect(audioCtx.destination);
  osc.start(); osc.stop(audioCtx.currentTime + dur);
}

// Game State
let game = {
  running: false,
  score: 0,
  totalScore: 0,
  perClick: 1,
  autoPerSec: 0,
  ghostDamage: 0.1,
  ghostSpeed: 1,
  ghostSpawnRate: 8,
  clickGhostBonus: 5,
  pacX: 0, pacY: 0,
  pacMouth: 0,
  pacPulse: 0,
  dots: [],
  ghosts: [],
  particles: [],
  floatTexts: [],
  upgrades: [
    { name: 'Power Pac', desc: '+1 dot per click', cost: 10, level: 0, max: 50, apply: g => g.perClick += 1 },
    { name: 'Auto Eater', desc: '+1 auto dot/sec', cost: 25, level: 0, max: 30, apply: g => g.autoPerSec += 1 },
    { name: 'Waka Shield', desc: '-5% ghost damage', cost: 50, level: 0, max: 15, apply: g => g.ghostDamage = Math.max(0.02, g.ghostDamage - 0.05) },
    { name: 'Ghost Bane', desc: '+5 bonus per ghost click', cost: 40, level: 0, max: 20, apply: g => g.clickGhostBonus += 5 },
    { name: 'Slow Mo', desc: '-10% ghost speed', cost: 60, level: 0, max: 10, apply: g => g.ghostSpeed = Math.max(0.3, g.ghostSpeed - 0.1) },
    { name: 'Less Ghosts', desc: '-1 sec spawn delay', cost: 80, level: 0, max: 5, apply: g => g.ghostSpawnRate += 1 }
  ],
  lastTime: 0,
  autoTimer: 0,
  ghostTimer: 0,
  gameOver: false
};

function costOf(up) { return Math.floor(up.cost * Math.pow(1.5, up.level)); }

// Drawing
function drawPac(x, y, r, mouth) {
  ctx.save();
  ctx.translate(x, y);
  const open = mouth * 0.4;
  ctx.beginPath();
  ctx.moveTo(0, 0);
  ctx.arc(0, 0, r, -Math.PI/2 + open, Math.PI/2 - open, false);
  ctx.closePath();
  ctx.fillStyle = '#ffe44d';
  ctx.fill();
  ctx.strokeStyle = '#ccaa00';
  ctx.lineWidth = 2;
  ctx.stroke();
  // Eye
  ctx.beginPath();
  ctx.arc(0, -r*0.2, r*0.12, 0, Math.PI*2);
  ctx.fillStyle = '#fff';
  ctx.fill();
  ctx.restore();
}

function drawGhost(x, y, r, color, t) {
  const bob = Math.sin(t * 4) * 3;
  ctx.save();
  ctx.translate(x, y + bob);
  // Body
  ctx.beginPath();
  ctx.arc(0, -r*0.2, r*0.8, Math.PI, 0);
  ctx.lineTo(r*0.8, r*0.6);
  for(let i = 0; i < 4; i++) {
    const sx = r*0.8 - (i+1)*r*0.4;
    ctx.quadraticCurveTo(sx + r*0.2, r*0.3, sx, r*0.6);
  }
  ctx.closePath();
  ctx.fillStyle = color;
  ctx.fill();
  // Eyes
  ctx.beginPath(); ctx.arc(-r*0.25, -r*0.1, r*0.18, 0, Math.PI*2); ctx.fillStyle='#fff'; ctx.fill();
  ctx.beginPath(); ctx.arc(r*0.25, -r*0.1, r*0.18, 0, Math.PI*2); ctx.fillStyle='#fff'; ctx.fill();
  ctx.beginPath(); ctx.arc(-r*0.2, -r*0.05, r*0.08, 0, Math.PI*2); ctx.fillStyle='#300'; ctx.fill();
  ctx.beginPath(); ctx.arc(r*0.3, -r*0.05, r*0.08, 0, Math.PI*2); ctx.fillStyle='#300'; ctx.fill();
  ctx.restore();
}

function drawDot(x, y) {
  ctx.beginPath();
  ctx.arc(x, y, 4, 0, Math.PI*2);
  ctx.fillStyle = '#fff';
  ctx.shadowColor = '#ffe44d';
  ctx.shadowBlur = 6;
  ctx.fill();
  ctx.shadowBlur = 0;
}

function drawGrid() {
  ctx.strokeStyle = 'rgba(30,30,80,0.5)';
  ctx.lineWidth = 1;
  const gs = 40;
  for(let x = 0; x < W; x += gs) { ctx.beginPath(); ctx.moveTo(x,0); ctx.lineTo(x,H); ctx.stroke(); }
  for(let y = 0; y < H; y += gs) { ctx.beginPath(); ctx.moveTo(0,y); ctx.lineTo(W,y); ctx.stroke(); }
}

function drawUI() {
  // Score
  ctx.fillStyle = '#ffe44d';
  ctx.font = 'bold 24px monospace';
  ctx.textAlign = 'left';
  ctx.fillText(`DOTS: ${Math.floor(game.score)}`, 20, 35);
  ctx.font = '14px monospace';
  ctx.fillStyle = '#aaa';
  ctx.fillText(`Per Click: ${game.perClick} | Auto: ${game.autoPerSec}/s`, 20, 55);

  // Upgrade buttons drawn on canvas
  const bw = Math.min(220, W*0.28), bh = 36;
  const bx = W - bw - 15, by = 15;
  ctx.fillStyle = 'rgba(10,10,46,0.9)';
  ctx.fillRect(bx-5, by-5, bw+10, game.upgrades.length*(bh+8)+10);
  
  for(let i = 0; i < game.upgrades.length; i++) {
    const u = game.upgrades[i];
    const uy = by + i*(bh+8);
    const c = costOf(u);
    const canBuy = game.score >= c && u.level < u.max;
    ctx.fillStyle = canBuy ? 'rgba(255,228,77,0.15)' : 'rgba(60,60,80,0.3)';
    ctx.fillRect(bx, uy, bw, bh);
    ctx.strokeStyle = canBuy ? '#ffe44d' : '#444';
    ctx.strokeRect(bx, uy, bw, bh);
    ctx.fillStyle = canBuy ? '#ffe44d' : '#888';
    ctx.font = '12px monospace';
    ctx.textAlign = 'left';
    const label = u.level >= u.max ? 'MAX' : `Lv${u.level} | ${c}`;
    ctx.fillText(`${u.name} [${label}]`, bx+8, uy+14);
    ctx.fillStyle = '#aaa';
    ctx.font = '10px monospace';
    ctx.fillText(u.desc, bx+8, uy+28);
  }
  
  // Store button rects for click detection
  game.upgradeRects = game.upgrades.map((u,i) => ({x:bx, y:by+i*(bh+8), w:bw, h:bh}));
}

function spawnGhost() {
  const colors = ['#ff0040','#00e5ff','#ff69b4','#ffa500'];
  game.ghosts.push({
    x: W + 30,
    y: H*0.2 + Math.random()*H*0.6,
    r: 20,
    color: colors[Math.floor(Math.random()*colors.length)],
    speed: (1+Math.random()*0.5) * game.ghostSpeed * 60,
    alive: true
  });
}

function spawnParticle(x, y, color) {
  for(let i = 0; i < 6; i++) {
    const a = Math.random()*Math.PI*2;
    game.particles.push({ x, y, vx: Math.cos(a)*(2+Math.random()*3), vy: Math.sin(a)*(2+Math.random()*3), life: 1, color });
  }
}

function spawnFloatText(x, y, text, color) {
  game.floatTexts.push({ x, y, text, color, life: 1 });
}

let lastFrame = 0;
function update(dt) {
  if (!game.running || game.gameOver) return;
  
  // Auto dots
  game.autoTimer += dt;
  if (game.autoPerSec > 0 && game.autoTimer >= 1/game.autoPerSec) {
    game.score += game.autoPerSec / game.autoPerSec;
    game.totalScore += 1;
    game.autoTimer = 0;
  }
  
  // Ghost spawn
  game.ghostTimer += dt;
  if (game.ghostTimer >= game.ghostSpawnRate) {
    game.ghostTimer = 0;
    spawnGhost();
  }
  
  // Update ghosts
  for(let i = game.ghosts.length-1; i >= 0; i--) {
    const g = game.ghosts[i];
    g.x -= g.speed * dt;
    if (g.x <= game.pacX + 30) {
      // Ghost hits Pac-Man
      const loss = Math.max(1, Math.floor(game.score * game.ghostDamage));
      game.score = Math.max(0, game.score - loss);
      spawnFloatText(game.pacX, game.pacY - 40, `-${loss}`, '#ff4444');
      playSound(150, 0.3, 'sawtooth', 0.15);
      game.ghosts.splice(i, 1);
    }
  }
  
  // Update particles
  for(let i = game.particles.length-1; i >= 0; i--) {
    const p = game.particles[i];
    p.x += p.vx; p.y += p.vy; p.life -= dt*2;
    if (p.life <= 0) game.particles.splice(i,1);
  }
  
  // Float texts
  for(let i = game.floatTexts.length-1; i >= 0; i--) {
    const f = game.floatTexts[i];
    f.y -= 40*dt; f.life -= dt*1.5;
    if (f.life <= 0) game.floatTexts.splice(i,1);
  }
  
  // Dots animation
  for(let i = game.dots.length-1; i >= 0; i--) {
    const d = game.dots[i];
    d.x -= 80*dt; d.life -= dt*3;
    if (d.life <= 0 || d.x < game.pacX) game.dots.splice(i,1);
  }
  
  // Pac mouth animation
  game.pacPulse += dt * 5;
}

function render(t) {
  ctx.fillStyle = '#0a0a2e';
  ctx.fillRect(0, 0, W, H);
  drawGrid();
  
  if (!game.running) return;
  
  // Draw dots trail
  for(const d of game.dots) { drawDot(d.x, d.y); }
  
  // Pac-Man
  const mouth = (Math.sin(game.pacPulse) + 1) * 0.5;
  drawPac(game.pacX, game.pacY, 28, mouth);
  
  // Ghosts
  for(const g of game.ghosts) { drawGhost(g.x, g.y, g.r, g.color, t/1000); }
  
  // Particles
  for(const p of game.particles) {
    ctx.globalAlpha = p.life;
    ctx.fillStyle = p.color;
    ctx.fillRect(p.x-3, p.y-3, 6, 6);
    ctx.globalAlpha = 1;
  }
  
  // Float texts
  for(const f of game.floatTexts) {
    ctx.globalAlpha = f.life;
    ctx.fillStyle = f.color;
    ctx.font = 'bold 18px monospace';
    ctx.textAlign = 'center';
    ctx.fillText(f.text, f.x, f.y);
    ctx.globalAlpha = 1;
  }
  
  drawUI();
  
  // Game Over
  if (game.gameOver) {
    ctx.fillStyle = 'rgba(0,0,0,0.7)';
    ctx.fillRect(0,0,W,H);
    ctx.fillStyle = '#ff4444';
    ctx.font = 'bold 3rem monospace';
    ctx.textAlign = 'center';
    ctx.fillText('GAME OVER', W/2, H/2 - 30);
    ctx.fillStyle = '#ffe44d';
    ctx.font = '1.5rem monospace';
    ctx.fillText(`Final Score: ${Math.floor(game.totalScore)}`, W/2, H/2 + 20);
    ctx.font = '1rem monospace';
    ctx.fillStyle = '#fff';
    ctx.fillText('Press R to restart', W/2, H/2 + 60);
  }
}

function loop(t) {
  const dt = Math.min((t - lastFrame)/1000, 0.05);
  lastFrame = t;
  update(dt);
  render(t);
  requestAnimationFrame(loop);
}

// Input
canvas.addEventListener('click', (e) => {
  if (!game.running || game.gameOver) return;
  const mx = e.clientX, my = e.clientY;
  
  // Check upgrade clicks
  if (game.upgradeRects) {
    for(let i = 0; i < game.upgradeRects.length; i++) {
      const r = game.upgradeRects[i];
      if (mx >= r.x && mx <= r.x+r.w && my >= r.y && my <= r.y+r.h) {
        const u = game.upgrades[i];
        const c = costOf(u);
        if (game.score >= c && u.level < u.max) {
          game.score -= c;
          u.level++;
          u.apply(game);
          playSound(600, 0.15, 'square', 0.08);
          spawnFloatText(mx, my, 'UPGRADE!', '#00ff88');
        } else {
          playSound(200, 0.1, 'sawtooth', 0.05);
        }
        return;
      }
    }
  }
  
  // Check ghost clicks
  for(let i = game.ghosts.length-1; i >= 0; i--) {
    const g = game.ghosts[i];
    const dx = mx - g.x, dy = my - (g.y + Math.sin(t*4)*3);
    if (dx*dx+dy*dy < (g.r+15)*(g.r+15)) {
      const bonus = game.clickGhostBonus;
      game.score += bonus;
      game.totalScore += bonus;
      spawnFloatText(g.x, g.y, `+${bonus}`, '#00ff88');
      spawnParticle(g.x, g.y, g.color);
      playSound(800, 0.1, 'square', 0.1);
      game.ghosts.splice(i, 1);
      return;
    }
  }
  
  // Eat dots (main click)
  const earned = game.perClick;
  game.score += earned;
  game.totalScore += earned;
  spawnFloatText(game.pacX + 20, game.pacY - 30, `+${earned}`, '#ffe44d');
  
  // Spawn dot particles toward pac
  for(let i = 0; i < 3; i++) {
    game.dots.push({ x: game.pacX + 60 + Math.random()*80, y: game.pacY - 20 + Math.random()*40, life: 1 });
  }
  
  playSound(400 + Math.random()*200, 0.08, 'square', 0.06);
});

window.addEventListener('keydown', (e) => {
  if (e.key === 'r' || e.key === 'R') {
    if (game.gameOver) resetGame();
  }
});

function resetGame() {
  game.score = 0;
  game.totalScore = 0;
  game.perClick = 1;
  game.autoPerSec = 0;
  game.ghostDamage = 0.1;
  game.ghostSpeed = 1;
  game.ghostSpawnRate = 8;
  game.clickGhostBonus = 5;
  game.dots = [];
  game.ghosts = [];
  game.particles = [];
  game.floatTexts = [];
  game.autoTimer = 0;
  game.ghostTimer = 0;
  game.gameOver = false;
  game.upgrades.forEach(u => u.level = 0);
}

document.getElementById('startBtn').addEventListener('click', () => {
  initAudio();
  document.getElementById('startScreen').classList.add('hidden');
  game.running = true;
  game.pacX = W*0.15;
  game.pacY = H*0.5;
  playSound(523, 0.1, 'square', 0.1);
});

// Game over check - if score drops to 0 and totalScore > 0, game over
const origUpdate = update;
function checkGameOver() {
  if (game.running && !game.gameOver && game.totalScore > 50 && game.score <= 0) {
    // Give a second chance but show warning
    game.score = 1;
  }
}

requestAnimationFrame(loop);
</script>
