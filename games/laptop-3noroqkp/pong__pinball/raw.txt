```javascript
function closestPointOnSegment(px, py, ax, ay, bx, by) {
    const dx = bx - ax, dy = by - ay;
    const len2 = dx*dx + dy*dy;
    let t = ((px - ax)*dx + (py - ay)*dy) / len2;
    t = Math.max(0, Math.min(1, t));
    return { x: ax + t*dx, y: ay + t*dy, t };
}

function flipperCollision(ball, flipper) {
    const tipX = flipper.px + Math.cos(flipper.angle) * flipper.len;
    const tipY = flipper.py + Math.sin(flipper.angle) * flipper.len;
    
    const cp = closestPointOnSegment(ball.x, ball.y, flipper.px, flipper.py, tipX, tipY);
    const dx = ball.x - cp.x;
    const dy = ball.y - cp.y;
    const dist = Math.sqrt(dx*dx + dy*dy);
    const minDist = ball.r + 5; // ball radius + flipper half-width
    
    if (dist < minDist) {
        // Push out
        const nx = dx / dist, ny = dy / dist;
        ball.x = cp.x + nx * minDist;
        ball.y = cp.y + ny * minDist;
        
        // Flipper surface velocity at contact point
        const relX = cp.x - flipper.px;
        const relY = cp.y - flipper.py;
        const svx = -flipper.angVel * relY;
        const svy = flipper.angVel * relX;
        
        // Relative velocity
        const rvx = ball.vx - svx;
        const rvy = ball.vy - svy;
        
        // Reflect
        const dot = rvx * nx + rvy * ny;
        if (dot < 0) {
            const j = -1.5 * dot; // restitution > 1 for "hit" effect
            ball.vx += j * nx;
            ball.vy += j * ny;
        }
    }
}
```

// Physics helpers
function closestPt(px,py,ax,ay,bx,by){
  const dx=bx-ax,dy=by-ay,l2=dx*dx+dy*dy;
  let t=((px-ax)*dx+(py-ay)*dy)/l2;
  t=Math.max(0,Math.min(1,t));
  return{x:ax+t*dx,y:ay+t*dy,t};
}

function segCollide(ball,ax,ay,bx,by,hw,surfaceVX,surfaceVY){
  const cp=closestPt(ball.x,ball.y,ax,ay,bx,by);
  const dx=ball.x-cp.x,dy=ball.y-cp.y;
  const d=Math.sqrt(dx*dx+dy*dy);
  const minD=ball.r+hw;
  if(d<minD&&d>0){
    const nx=dx/d,ny=dy/d;
    ball.x=cp.x+nx*minD;
    ball.y=cp.y+ny*minD;
    const rvx=ball.vx-(surfaceVX||0),rvy=ball.vy-(surfaceVY||0);
    const dot=rvx*nx+rvy*ny;
    if(dot<0){
      const e=.55;
      ball.vx=rvx-(1+e)*dot*nx+(surfaceVX||0);
      ball.vy=rvy-(1+e)*dot*ny+(surfaceVY||0);
      return true;
    }
  }
  return false;
}

function updateFlipper(f,active,dt){
  const targetA=active?f.activeA:f.restA;

  else f.angle=Math.max(f.angle-spd*dt,targetA);
  f.angVel=(f.angle-prev)/Math.max(dt,.001);
}

function flipperCollision(f){
  const tx=f.px+Math.cos(f.angle)*f.len;
  const ty=f.py+Math.sin(f.angle)*f.len;

  const dx=ball.x-cp.x,dy=ball.y-cp.y;

  if(d<minD&&d>0){
    const nx=dx/d,ny=dy/d;

    // Surface velocity at contact

  }

  // Pong Paddle AI
  const aiSpd=250+Math.min(score/8,150);
  let targetX=ball.x;
  if(ball.y<400&&ball.vy<0)targetX=ball.x;else targetX=W/2;
  const dx=targetX-pongPaddle.x;
  if(Math.abs(dx)>5){
    pongPaddle.x+=Math.sign(dx)*Math.min(aiSpd*dt,Math.abs(dx));
  }
  pongPaddle.x=Math.max(pongPaddle.w/2+15,Math.min(W-pongPaddle.w/2-15,pongPaddle.x));

  // Trail
  trail.push({x:ball.x,y:ball.y});
  if(trail.length>18)trail.shift();

  // Particles
  for(let i=particles.length-1;i>=0;i--){
    const p=particles[i];
    p.x+=p.vx*dt;p.y+=p.vy*dt;p.vy+=300*dt;
    p.life-=dt;
    if(p.life<=0)particles.splice(i,1);
  }

  // Score popups
  for(let i=scorePopups.length-1;i>=0;i--){
    const sp=scorePopups[i];
    sp.y+=sp.vy*dt;sp.life-=dt;
    if(sp.life<=0)scorePopups.splice(i,1);
  }

  // Drain check
  if(ball.y>790){
    loseLife();
  }
}

// Rendering
function drawGlow(color,blur,fn){
  ctx.save();ctx.shadowColor=color;ctx.shadowBlur=blur||12;fn();ctx.restore();
}

function render(){
  ctx.fillStyle='#08081a';ctx.fillRect(0,0,W,H);

  // Subtle grid dots
  ctx.fillStyle='rgba(30,40,80,.4)';
  for(let x=30;x<W;x+=40)for(let y=30;y<H;y+=40){
    ctx.beginPath();ctx.arc(x,y,1.2,0,Math.PI*2);ctx.fill();
  }

  // Walls
  drawGlow('#00ccff',8,()=>{
    ctx.strokeStyle='#00ccff';ctx.lineWidth=3;ctx.lineCap='round';
    for(const w of walls){
      ctx.beginPath();ctx.moveTo(w.ax,w.ay);ctx.lineTo(w.bx,w.by);ctx.stroke();
    }
  });

  // Slopes
  drawGlow('#00ccff',8,()=>{
    ctx.strokeStyle='#00b8d4';ctx.lineWidth=4;
    for(const s of slopes){
      ctx.beginPath();ctx.moveTo(s.ax,s.ay);ctx.lineTo(s.bx,s.by);ctx.stroke();
    }
  });

  // Pong Paddle
  drawGlow('#ffffff',15,()=>{
    ctx.fillStyle='#eef';
    const pp=pongPaddle;
    ctx.fillRect(pp.x-pp.w/2,pp.y,pp.w,pp.h);
    ctx.strokeStyle='#88eeff';ctx.lineWidth=2;
    ctx.strokeRect(pp.x-pp.w/2,pp.y,pp.w,pp.h);
  });
  ctx.fillStyle='#88eeff';ctx.font='bold 10px monospace';ctx.textAlign='center';
  ctx.fillText('PONG',pongPaddle.x,pongPaddle.y+9);

  // Bumpers
  for(const b of bumpers){
    const pr=1+b.pulse*.3;
    drawGlow('#ff4488',12+b.pulse*20,()=>{
      ctx.fillStyle=`rgba(255,68,136,${.3+b.pulse*.5})`;
      ctx.beginPath();ctx.arc(b.x,b.y,b.r*pr,0,Math.PI*2);ctx.fill();
      ctx.strokeStyle='#ff66aa';ctx.lineWidth=2;
      ctx.beginPath();ctx.arc(b.x,b.y,b.r*pr,0,Math.PI*2);ctx.stroke();
    });
    ctx.fillStyle='#fff';ctx.font='bold 14px monospace';ctx.textAlign='center';ctx.textBaseline='middle';
    ctx.fillText('●',b.x,b.y);
  }

  // Flippers
  for(let i=0;i<2;i++){
    const f=flippers[i];
    const tx=f.px+Math.cos(f.angle)*f.len;
    const ty=f.py+Math.sin(f.angle)*f.len;
    drawGlow('#ffaa22',14,()=>{
      ctx.strokeStyle='#ffaa22';ctx.lineWidth=flipperHalfW*2;ctx.lineCap='round';
      ctx.beginPath();ctx.moveTo(f.px,f.py);ctx.lineTo(tx,ty);ctx.stroke();
    });
    // Pivot dot
    ctx.fillStyle='#ffcc66';
    ctx.beginPath();ctx.arc(f.px,f.py,5,0,Math.PI*2);ctx.fill();
  }

  // Ball trail
  for(let i=0;i<trail.length;i++){
    const a=i/trail.length*.4;
    ctx.fillStyle=`rgba(100,220,255,${a})`;
    ctx.beginPath();ctx.arc(trail[i].x,trail[i].y,BALL_R*(i/trail.length)*.8,0,Math.PI*2);ctx.fill();
  }

  // Ball
  drawGlow('#66eeff',18,()=>{
    ctx.fillStyle='#fff';
    ctx.beginPath();ctx.arc(ball.x,ball.y,BALL_R,0,Math.PI*2);ctx.fill();
  });

  // Particles
  for(const p of particles){
    const a=p.life/p.maxLife;
    ctx.fillStyle=p.color.replace(')',`a)`).replace('rgba','rgba').replace(/,\s*\)/,'').length>0?
      `rgba(${parseInt(p.color.slice(1,3),16)},${parseInt(p.color.slice(3,5),16)},${parseInt(p.color.slice(5,7),16)},${a})`:p.color;
    ctx.globalAlpha=a;
    ctx.fillStyle=p.color;
    ctx.beginPath();ctx.arc(p.x,p.y,3*a+1,0,Math.PI*2);ctx.fill();
  }
  ctx.globalAlpha=1;

  // Score popups
  for(const sp of scorePopups){
    ctx.globalAlpha=Math.min(1,sp.life);
    ctx.fillStyle='#ffdd44';ctx.font='bold 16px monospace';ctx.textAlign='center';
    ctx.fillText(sp.text,sp.x,sp.y);
  }
  ctx.globalAlpha=1;

  // HUD
  if(state==='playing'||state==='gameover'){
    ctx.fillStyle='#aaddff';ctx.font='bold 22px monospace';ctx.textAlign='left';
    ctx.fillText('SCORE: '+score,20,45);
    if(combo>1){
      ctx.fillStyle='#ff88cc';ctx.font='bold 16px monospace';ctx.textAlign='center';
      ctx.fillText('COMBO x'+combo,W/2,45);
    }
    // Lives as small balls
    for(let i=0;i<lives;i++){
      ctx.fillStyle='#66eeff';
      ctx.beginPath();ctx.arc(W-30-i*25,40,7,0,Math.PI*2);ctx.fill();
    }
  }

  // Start screen
  if(state==='start'){
    ctx.fillStyle='rgba(6,6,18,.85)';ctx.fillRect(0,0,W,H);
    ctx.textAlign='center';
    ctx.fillStyle='#66eeff';ctx.font='bold 48px monospace';
    ctx.fillText('PONG',W/2,200);
    ctx.fillStyle='#ff66aa';ctx.font='bold 48px monospace';
    ctx.fillText('PINBALL',W/2,260);

    ctx.fillStyle='#aaddff';ctx.font='16px monospace';
    ctx.fillText('← / A : Left Flipper',W/2,350);
    ctx.fillText('→ / D : Right Flipper',W/2,380);
    ctx.fillText('SPACE : Start / Restart',W/2,420);

    ctx.fillStyle='#ffaa33';ctx.font='14px monospace';
    ctx.fillText('Keep the ball aloft with your flippers!',W/2,470);
    ctx.fillText('Hit bumpers for combos. The PONG paddle',W/2,495);
    ctx.fillText('at the top bounces the ball back down.',W/2,520);

    if(bestScore>0){ctx.fillStyle='#ffdd44';ctx.font='16px monospace';ctx.fillText('Best: '+bestScore,W/2,560)}

    const blink=Math.sin(Date.now()/400)>.2;
    if(blink){ctx.fillStyle='#fff';ctx.font='bold 18px monospace';ctx.fillText('Press SPACE to Start',W/2,630)}
  }

  // Game over
  if(state==='gameover'){
    ctx.fillStyle='rgba(6,6,18,.75)';ctx.fillRect(0,0,W,H);
    ctx.textAlign='center';
    ctx.fillStyle='#ff4444';ctx.font='bold 42px monospace';
    ctx.fillText('GAME OVER',W/2,300);
    ctx.fillStyle='#aaddff';ctx.font='24px monospace';
    ctx.fillText('Score: '+score,W/2,360);
    if(score>=bestScore&&score>0){ctx.fillStyle='#ffdd44';ctx.fillText('★ NEW BEST! ★',W/2,400)}
    const blink=Math.sin(Date.now()/400)>.2;
    if(blink){ctx.fillStyle='#fff';ctx.font='bold 18px monospace';ctx.fillText('Press SPACE to Play Again',W/2,470)}
  }
}

  const dt=Math.min((ts-lastTime)/1000,.035);
  lastTime=ts;
  update(dt);
  render();
  requestAnimationFrame(loop);
}
requestAnimationFrame(t=>{lastTime=t;loop(t)});
