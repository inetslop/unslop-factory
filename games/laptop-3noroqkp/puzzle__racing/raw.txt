function getCell(x, y) {
    if (x < 0 || x >= 10 || y < 0 || y >= 10) return 1;
    return levels[level].grid[y][x];

function tryMove(dx, dy) {
    if (animating || state !== STATE.PLING && state !== STATE.PLAYING) return;
    if (state !== STATE.PLAYING) return;
    
    const nx = carX + dx;
    const ny = carY + dy;
    const cell = getCell(nx, ny);
    
    if (cell === 1) { playFail(); shakeTime = 0.2; return; }
    
    let cost = 1;
    if (cell === 3) cost = 3; // oil costs 3 total
    
    if (fuel < cost) { 
        playFail(); 
        state = STATE.GAME_OVER;
        shakeTime = 0.3;
        return; 
    }
    
    fuel -= cost;
    moves++;
    
    animFromX = carX;
    animFromY = carY;
    animToX = nx;
    animToY = ny;
    animProgress = 0;
    animating = true;
    
    playMove();
}

function completeAnimation() {
    carX = animToX;
    carY = animToY;
    carPixelX = carX;
    carPixelY = carY;
    animating = false;
    
    const cell = getCell(carX, carY);
    if (cell === 2) {
        fuel = Math.min(fuel + 3, maxFuel + 5);
        levels[level].grid[carY][carX] = 0;
        playNitro();
        spawnParticles(carX, carY, '#00ff88', 12);
    } else if (cell === 4) {
        playGoal();
        state = STATE.LEVEL_COMPLETE;
        const bonus = fuel * 10 + Math.max(0, levels[level].par - moves) * 5;
        totalScore += 100 + bonus;
        spawnParticles(carX, carY, '#ffff00', 20);
    } else if (cell === 3) {
        playFail(); // extra sound for oil
    }
}

function spawnParticles(gx, gy, color, count) {
    const cell = getCellSize();
    const cx = (gx + 0.5) * cell;
    const cy = (gy + 0.5) * cell;
    for (let i = 0; i < count; i++) {
        const angle = Math.random() * Math.PI * 2;
        const speed = 1 + Math.random() * 3;
        particles.push({
            x: cx, y: cy,
            vx: Math.cos(angle) * speed,
            vy: Math.sin(angle) * speed,
            life: 1,
            color: color,
            size: 2 + Math.random() * 4
        });
    }
}

function getCellSize() {
    const margin = 80;
    return Math.min((W - margin*2) / 10, (H - margin*2) / 10);
}

function getOrigin() {
    const cs = getCellSize();
    return { x: (W - cs * 10) / 2, y: (H - cs * 10) / 2 };

let lastTime = 0;
function gameLoop(time) {
    const dt = Math.min((time - lastTime) / 1000, 0.05);
    lastTime = time;
    
    update(dt);
    render();
    requestAnimationFrame(gameLoop);
}

function update(dt) {
    if (animating) {
        animProgress += dt * 8;
        if (animProgress >= 1) {
            animProgress = 1;
            completeAnimation();
        }
        const t = easeOut(animProgress);
        carPixelX = animFromX + (animToX - animFromX) * t;
        carPixelY = animFromY + (animToY - animFromY) * t;
    }
    
    if (shakeTime > 0) shakeTime -= dt;
    
    for (let i = particles.length - 1; i >= 0; i--) {
        const p = particles[i];
        p.x += p.vx;
        p.y += p.vy;
        p.vy += 0.1;
        p.life -= dt * 2;
        if (p.life <= 0) particles.splice(i, 1);
    }
}

function easeOut(t) {
    return 1 - Math.pow(1 - t, 3);
}

function renderHUD() {
    const cs = getCellSize();
    const ox = getOrigin().x;
    const oy = getOrigin().y;
    
    // Fuel bar
    const barW = cs * 4;
    const barH = 20;
    const barX = ox + (10*cs - barW)/2;
    const barY = oy - 45;
    
    ctx.fillStyle = '#1a1a2e';
    ctx.fillRect(barX, barY, barW, barH);
    ctx.strokeStyle = '#444';
    ctx.lineWidth = 2;
    ctx.strokeRect(barX, barY, barW, barH);
    
    const fuelPct = Math.max(0, fuel / maxFuel);
    let fuelColor = fuelPct > 0.5 ? '#44ff44' : fuelPct > 0.25 ? '#ffaa00' : '#ff3333';
    ctx.fillStyle = fuelColor;
    ctx.fillRect(barX + 2, barY + 2, (barW - 4) * fuelPct, barH - 4);
    
    ctx.fillStyle = '#fff';
    ctx.font = `bold 14px monospace`;
    ctx.textAlign = 'left';
    ctx.textBaseline = 'middle';
    ctx.fillText(`FUEL: ${fuel}/${maxFuel}`, barX + 5, barY + barH/2);
    
    // Level & Moves
    ctx.textAlign = 'right';
    ctx.font = `bold 16px monospace`;
    ctx.fillStyle = '#aaa';
    ctx.fillText(`LVL ${level + 1}/${levels.length}`, ox + 10*cs - 10, oy - 45);
    ctx.fillText(`MOVES: ${moves}`, ox + 10*cs - 10, oy - 22);
    
    // Score
    ctx.textAlign = 'left';
    ctx.fillStyle = '#ffdd00';
    ctx.fillText(`SCORE: ${totalScore}`, ox + 5, oy - 22);
}

function renderMenu() {
    ctx.fillStyle = '#0a0a1a';

    ctx.fillRect(carX - 20, carY - 12, 40, 24);
    ctx.fillStyle = '#88ccff';
    ctx.fillRect(carX - 10, carY - 8, 20, 8);

    ctx.font = `${Math.min(W*0.025, 20)}px monospace`;
    ctx.fillStyle = '#aaa';

        'Navigate the circuit from START to FINISH',
        'Arrow Keys / WASD to drive',

        'Each move costs 1 FUEL',

        'NITRO (N) gives +3 fuel',

    ];
    lines.forEach((line, i) => {
        ctx.fillText(line, W/2, H * 0.52 + i * 28);

            ctx.fillRect(W - (i+1)*sq + sq, H - (j+1)*sq + sq, sq, sq);
        }
    }
}
