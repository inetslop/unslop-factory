  ctx.lineTo(sx-hw,sy+hh);
  ctx.closePath();
  ctx.fillStyle='#1e1e3a';ctx.fill();
  ctx.strokeStyle='#2a2a4a';ctx.lineWidth=1;ctx.stroke();
}

function drawGoalTile(sx,sy,active){
  const hw=TW/2,hh=TH/2;
  ctx.beginPath();
  ctx.moveTo(sx,sy);
  ctx.lineTo(sx+hw,sy+hh);
  ctx.lineTo(sx,sy+hh*2);
  ctx.lineTo(sx-hw,sy+hh);
  ctx.closePath();
  const glow=active?0.3:0.6;
  ctx.fillStyle=`rgba(50,230,120,${glow})`;ctx.fill();
  ctx.strokeStyle=`rgba(50,230,120,${glow+0.2})`;ctx.lineWidth=2;ctx.stroke();
  // Inner diamond
  const ihw=hw*0.5,ihh=hh*0.5;
  ctx.beginPath();
  ctx.moveTo(sx,sy+hh-ihh);
  ctx.lineTo(sx+ihw,sy+hh);
  ctx.lineTo(sx,sy+hh+ihh);
  ctx.lineTo(sx-ihw,sy+hh);
  ctx.closePath();
  ctx.fillStyle=`rgba(50,230,120,${glow*0.5})`;ctx.fill();
}

function render(){
  ctx.clearRect(0,0,W,H);
  
  // Background gradient
  const bg=ctx.createLinearGradient(0,0,0,H);
  bg.addColorStop(0,'#0a0a1a');bg.addColorStop(1,'#151530');
  ctx.fillStyle=bg;ctx.fillRect(0,0,W,H);
  
  if(gameState==='start'){drawStartScreen();return;}
  
  const cam=getCamOffset();
  const{ox,oy,scale}=cam;
  
  ctx.save();
  ctx.translate(ox,oy);
  ctx.scale(scale,scale);
  
  // Collect all drawable objects with depth
  const drawables=[];
  
  // Floor tiles
  for(const k of Object.keys(floorSet)){
    const[x,y]=k.split(',').map(Number);
    drawables.push({depth:x+y,type:'floor',x,y});
  }
  // Also add floor under walls, boxes, goals
  for(const[x,y]of walls)drawables.push({depth:x+y,type:'floor',x,y});
  for(const[x,y]of goals)drawables.push({depth:x+y,type:'floor',x,y});
  for(const b of boxes)drawables.push({depth:b.x+b.y,type:'floor',x:b.x,y:b.y});
  
  // Goals
  for(const[x,y]of goals){
    const hasBox=boxes.some(b=>b.x===x&&b.y===y);
    drawables.push({depth:x+y+0.1,type:'goal',x,y,active:!hasBox});
  }
  
  // Walls
  for(const[x,y]of walls)drawables.push({depth:x+y+0.2,type:'wall',x,y});
  
  // Boxes
  for(const b of boxes)drawables.push({depth:b.x+b.y+0.3,type:'box',x:b.x,y:b.y});
  
  // Player (with animation)
  let dispX=px,dispY=py;
  if(animT<1){
    const t=1-animT;
    const ease=t*t*(3-2*t); // smoothstep
    dispX=ax+(bx-ax)*ease;
    dispY=ay+(by-ay)*ease;
  }
  drawables.push({depth:dispX+dispY+0.4,type:'player',x:dispX,y:dispY});
  
  // Sort by depth
  drawables.sort((a,b)=>a.depth-b.depth);
  
  // Draw all
  for(const d of drawables){
    const sx=isoX(d.x,d.y),sy=isoY(d.x,d.y);
    if(d.type==='floor'){drawFloorTile(sx,sy);}
    else if(d.type==='goal'){drawGoalTile(sx,sy,d.active);}
    else if(d.type==='wall'){drawCube(sx,sy,CH*0.7,'#5a5a7a','#4a4a6a','#525272');}
    else if(d.type==='box'){
      const onGoal=goals.some(g=>g[0]===Math.round(d.x)&&g[1]===Math.round(d.y));
      if(onGoal)drawCube(sx,sy,CH*0.5,'#30e860','#20c050','#28d858');
      else drawCube(sx,sy,CH*0.5,'#e8a030','#c08020','#d89028');
    }
    else if(d.type==='player'){
      const bob=Math.sin(Date.now()*0.005)*2;
      drawCube(sx,sy,CH*0.6+bob,'#40c0ff','#30a0e0','#38b8f0');
      // Eyes
      ctx.fillStyle='#fff';
      ctx.fillRect(sx-4,sy-CH*0.6+bob+4,3,3);
      ctx.fillRect(sx+2,sy-CH*0.6+bob+4,3,3);
    }
  }
  
  // Particles
  for(const p of particles){
    ctx.globalAlpha=p.life;
    ctx.fillStyle=p.color;
    ctx.beginPath();
    ctx.arc(p.x,p.y,p.r,0,Math.PI*2);
    ctx.fill();
  }
  ctx.globalAlpha=1;
  
  ctx.restore();
  
  // HUD
  if(gameState==='playing'||gameState==='levelComplete'){
    drawHUD();
  }
}

// Input
let keys={};
window.addEventListener('keydown',e=>{
  if(e.repeat)return;
  initAudio();
  
  if(gameState==='start'){
    if(e.key==='Enter'||e.key===' '){startGame();}
    return;
  }
  
  if(gameState==='levelComplete'){
    if(e.key==='Enter'||e.key===' '){nextLevel();}
    return;
  }
  
  if(gameState==='gameWin'){
    if(e.key==='r'||e.key==='R'){startGame();}
    return;
  }
  
  if(gameState!=='playing')return;
  
  let dx=0,dy=0;
  if(e.key==='w'||e.key==='W'||e.key==='ArrowUp'){dx=0;dy=-1;}
  else if(e.key==='s'||e.key==='S'||e.key==='ArrowDown'){dx=0;dy=1;}
  else if(e.key==='a'||e.key==='A'||e.key==='ArrowLeft'){dx=-1;dy=0;}
  else if(e.key==='d'||e.key==='D'||e.key==='ArrowRight'){dx=1;dy=0;}
  else if(e.key==='r'||e.key==='R'){loadLevel(curLevel);return;}
  
  if(dx||dy)tryMove(dx,dy);
});

function tryMove(dx,dy){
  if(animT<1)return; // Still animating
  
  const nx=px+dx,ny=py+dy;
  
  // Check wall
  if(walls.some(w=>w[0]===nx&&w[1]===ny)){sfxBump();return;}
  
  // Check box
  const bi=boxes.findIndex(b=>b.x===nx&&b.y===ny);
  if(bi!==-1){
    const bxx=nx+dx,byy=ny+dy;
    // Can't push into wall or another box
    if(walls.some(w=>w[0]===bxx&&w[1]===byy)){sfxBump();return;}
    if(boxes.some(b=>b.x===bxx&&b.y===byy)){sfxBump();return;}
    // Push!
    ax=px;ay=py;bx=nx;by=ny;animT=0;
    boxes[bi].x=bxx;boxes[bi].y=byy;
    px=nx;py=ny;moveCount++;
    sfxPush();
    // Check if box landed on goal
    if(goals.some(g=>g[0]===bxx&&g[1]===byy)){
      setTimeout(()=>{sfxGoal();spawnParticles(isoX(bxx,byy),isoY(bxx,byy),'#30e860');},150);
    }
    return;
  }
  
  // Walk
  ax=px;ay=py;bx=nx;by=ny;animT=0;
  px=nx;py=ny;moveCount++;
  sfxMove();
  
  // Check win
  setTimeout(checkWin,180);
}

function checkWin(){
  const allOnGoals=boxes.every(b=>goals.some(g=>g[0]===b.x&&g[1]===b.y));
  if(allOnGoals){
    gameState='levelComplete';
    sfxWin();
    spawnParticles(W/2,H/2,'#30e860');
    spawnParticles(W/2,H/2,'#e8a030');
    spawnParticles(W/2,H/2,'#40c0ff');
  }
}

function startGame(){
  curLevel=0;
  loadLevel(0);
  gameState='playing';
}

function nextLevel(){
  curLevel++;
  if(curLevel>=RAW_LEVELS.length){gameState='gameWin';return;}
  loadLevel(curLevel);
  gameState='playing';
}

function spawnParticles(sx,sy,color){
  for(let i=0;i<20;i++){
    const angle=Math.random()*Math.PI*2;
    const speed=1+Math.random()*3;
    particles.push({
      x:sx,y:sy-20,
      vx:Math.cos(angle)*speed,vy:Math.sin(angle)*speed-2,
      life:1,r:2+Math.random()*4,color
    });
  }
}

// Particles in screen space for overlay effects
function updateParticles(){
  for(const p of particles){
    p.x+=p.vx;p.y+=p.vy;
    p.vy+=0.15;
    p.life-=0.015;
  }
  particles=particles.filter(p=>p.life>0);
}

// Main loop
let lastTime=0;
function gameLoop(time){
  const dt=(time-lastTime)/1000;lastTime=time;
  
  if(animT<1){animT=Math.min(1,animT+dt/0.15);}
  
  updateParticles();
  
  render();
  
  if(gameState==='levelComplete'||gameState==='gameWin'){drawLevelComplete();}
  
  requestAnimationFrame(gameLoop);
}

loadLevel(0);
requestAnimationFrame(gameLoop);
