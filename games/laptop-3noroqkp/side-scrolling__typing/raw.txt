let audioCtx = null;

    if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();

function playSound(type, freq, dur, vol) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = type;
    osc.frequency.value = freq;
    gain.gain.setValueAtTime(vol || 0.1, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + (dur || 0.1));
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + (dur || 0.1));
}

function update() {
    if (gameState !== 'playing') return;

    frameCount++;
    speed = Math.min(12, 4 + frameCount / 300);
    scrollX += speed;

    // Player physics
    player.vy += GRAVITY;

    if (player.y >= groundY) {
        player.y = groundY;
        player.vy = 0;
        player.jumping = false;
        player.grounded = true;
    }

    // Word timer - if you don't type in time, lose a life
    wordTimer++;
    const maxWordTime = currentWord.length * 45 + 60;
    if (wordTimer > maxWordTime) {
        loseLife();
        nextWord();
    }

    // Spawn obstacles
    if (scrollX >= nextObstacleDist) {
        const type = Math.random() < 0.6 ? 'rock' : 'spike';
        const size = type === 'rock' ? 25 + Math.random() * 15 : 20 + Math.random() * 10;
        obstacles.push({
            x: canvas.width + 50, y: groundY, w: size, h: type === 'rock' ? size : size * 0.7,
            type: type, passed: false
        });
        nextObstacleDist = scrollX + 200 + Math.random() * 300 - speed * 15;
    }

    // Update obstacles
    for (let i = obstacles.length - 1; i >= 0; i--) {
        obstacles[i].x -= speed;
        if (obstacles[i].x < -100) {
            obstacles.splice(i, 1);
            continue;
        }

        // Collision check
        const obs = obstacles[i];
        const px = player.x + 10, py = player.y - 50, pw = 30, ph = 50;
        const ox = obs.x, oy = obs.y - obs.h, ow = obs.w, oh = obs.h;

        if (px < ox + ow && px + pw > ox && py < oy + oh && py + ph > oy) {
            if (!obs.passed) {
                loseLife();
                obs.passed = true;
                shakeTimer = 12;
                shakeIntensity = 8;
                spawnHitParticles();
                playSound('square', 100, 0.3, 0.2);
            }
        }

        // Score for passing obstacle
        if (!obs.passed && obs.x + obs.w < player.x) {
            obs.passed = true;
            score += 5;
        }
    }

    // Clouds
    clouds.forEach(c => {
        c.x -= c.speed;
        if (c.x < -200) { c.x = canvas.width + 100; c.y = 30 + Math.random() * 80; }
    });

    // Mountains parallax
    mountains.forEach(m => {
        m.x -= speed * 0.3;
        if (m.x < -400) { m.x = canvas.width + 200; m.h = 100 + Math.random() * 150; }
    });

    // Particles
    for (let i = particles.length - 1; i >= 0; i--) {
        const p = particles[i];
        p.x += p.vx;
        p.y += p.vy;
        p.vy += 0.3;
        p.life--;
        if (p.life <= 0) particles.splice(i, 1);
    }

    // Stars twinkle
    stars.forEach(s => { s.twinkle += 0.05; });

    // Shake decay
    if (shakeTimer > 0) shakeTimer--;
    if (flashTimer > 0) flashTimer--;
}

function roundRect(ctx, x, y, w, h, r) {
    ctx.beginPath();
    ctx.moveTo(x + r, y);
    ctx.lineTo(x + w - r, y);
    ctx.quadraticCurveTo(x + w, y, x + w, y + r);
    ctx.lineTo(x + w, y + h - r);
    ctx.quadraticCurveTo(x + w, y + h, x + w - r, y + h);
    ctx.lineTo(x + r, y + h);
    ctx.quadraticCurveTo(x, y + h, x, y + h - r);
    ctx.lineTo(x, y + r);
    ctx.quadraticCurveTo(x, y, x + r, y);
    ctx.closePath();
}

// Game loop
function gameLoop() {
    update();
    draw();
    requestAnimationFrame(gameLoop);
}
