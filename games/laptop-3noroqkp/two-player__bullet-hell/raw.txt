<script>
const canvas = document.getElementById('game');
const ctx = canvas.getContext('2d');

let W, H;
function resize() {
    W = window.innerWidth;
    H = window.innerHeight;
    canvas.width = W;
    canvas.height = H;
}
window.addEventListener('resize', resize);
resize();

// Audio
let audioCtx = null;
function initAudio() {
    if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
}
function playSound(type, freq, dur, vol) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = type;
    osc.frequency.value = freq || 440;
    gain.gain.setValueAtTime(vol || 0.15, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + (dur || 0.1));
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + (dur || 0.1));
}
function playShoot() { playSound('square', 880, 0.05, 0.08); }
function playExplode() { playSound('sawtooth', 120, 0.3, 0.2); }
function playHit() { playSound('square', 200, 0.2, 0.25); }
function playEnemyDie() { playSound('triangle', 600, 0.15, 0.12); }

// Game State
let gameState = 'menu'; // menu, playing, gameover
let score = [0, 0];
let totalKills = [0, 0];
let wave = 0;
let waveTimer = 0;
let difficulty = 1;

// Input
const keys = {};
window.addEventListener('keydown', e => {
    keys[e.code] = true;
    if (gameState === 'menu') startGame();
    if (gameState === 'gameover' && e.code === 'Space') startGame();
});
window.addEventListener('keyup', e => { keys[e.code] = false; });

// Particles
let particles = [];
function spawnParticles(x, y, color, count) {
    for (let i = 0; i < count; i++) {
        const angle = Math.random() * Math.PI * 2;
        const speed = Math.random() * 4 + 1;
        particles.push({ x, y, vx: Math.cos(angle)*speed, vy: Math.sin(angle)*speed, life: 30+Math.random()*20, maxLife: 50, color, size: Math.random()*4+1 });
    }
}

// Bullets (enemy)
let enemyBullets = [];
// Player bullets
let playerBullets = [];
// Enemies
let enemies = [];

// Players
function createPlayer(id, x, y) {
    return {
        id, x, y, w: 20, h: 20,
        lives: 3, maxLives: 5,
        invincible: 0, // frames of invincibility
        shootCooldown: 0,
        color: id === 0 ? '#00ffcc' : '#ff6699',
        glowColor: id === 0 ? 'rgba(0,255,204,' : 'rgba(255,102,153,',
        dead: false
    };
}

let players = [];

function startGame() {
    initAudio();
    gameState = 'playing';
    score = [0, 0];
    totalKills = [0, 0];
    wave = 0;
    waveTimer = 180;
    difficulty = 1;
    enemyBullets = [];
    playerBullets = [];
    enemies = [];
    particles = [];
    players = [createPlayer(0, W*0.25, H*0.75), createPlayer(1, W*0.75, H*0.75)];
}

// Enemy types and bullet patterns
function spawnEnemy() {
    const type = Math.floor(Math.random() * 4);
    const x = 60 + Math.random() * (W - 120);
    const e = { x, y: -30, w: 24, h: 24, hp: 1 + Math.floor(difficulty/3), maxHp: 1 + Math.floor(difficulty/3), type, shootTimer: 0, shootInterval: Math.max(60, 90 - difficulty*5), angle: 0, vx: (Math.random()-0.5)*1.5, vy: 1 + Math.random()*0.5 };
    enemies.push(e);
}

function enemyShoot(e) {
    const pattern = e.type;
    if (pattern === 0) {
        // Aimed shot at nearest player
        let target = null; let minDist = Infinity;
        for (const p of players) { if (!p.dead) { const d = Math.hypot(p.x-e.x, p.y-e.y); if (d<minDist){minDist=d;target=p;} } }
        if (target) {
            const angle = Math.atan2(target.y - e.y, target.x - e.x);
            for (let i = -1; i <= 1; i++) {
                enemyBullets.push({ x: e.x, y: e.y+10, vx: Math.cos(angle+i*0.2)*4, vy: Math.sin(angle+i*0.2)*4, size: 5, color: '#ff3366' });
            }
        }
    } else if (pattern === 1) {
        // Fan spread downward
        const count = 5;
        for (let i = 0; i < count; i++) {
            const angle = Math.PI/2 + (i - (count-1)/2) * 0.3;
            enemyBullets.push({ x: e.x, y: e.y+10, vx: Math.cos(angle)*3.5, vy: Math.sin(angle)*3.5, size: 4, color: '#ffaa00' });
        }
    } else if (pattern === 2) {
        // Spiral burst
        const count = 8;
        for (let i = 0; i < count; i++) {
            const angle = e.angle + (i / count) * Math.PI * 2;
            enemyBullets.push({ x: e.x, y: e.y, vx: Math.cos(angle)*3, vy: Math.sin(angle)*3, size: 4, color: '#cc44ff' });
        }
    } else {
        // Ring burst
        const count = 12;
        for (let i = 0; i < count; i++) {
            const angle = (i / count) * Math.PI * 2;
            enemyBullets.push({ x: e.x, y: e.y, vx: Math.cos(angle)*2.5, vy: Math.sin(angle)*2.5, size: 3.5, color: '#44ff88' });
        }
    }
}

function update() {
    if (gameState !== 'playing') return;

    // Wave management
    waveTimer--;
    if (waveTimer <= 0) {
        wave++;
        difficulty = wave;
        waveTimer = Math.max(120, 300 - wave*10);
        const count = Math.min(3 + wave, 8);
        for (let i = 0; i < count; i++) setTimeout(() => spawnEnemy(), i * 400);
    }

    // Player movement
    const speed = 5;
    // P1: WASD
    if (!players[0].dead) {
        if (keys['KeyW']) players[0].y -= speed;
        if (keys['KeyS']) players[0].y += speed;
        if (keys['KeyA']) players[0].x -= speed;
        if (keys['KeyD']) players[0].x += speed;
    }
    // P2: Arrow keys
    if (!players[1].dead) {
        if (keys['ArrowUp']) players[1].y -= speed;
        if (keys['ArrowDown']) players[1].y += speed;
        if (keys['ArrowLeft']) players[1].x -= speed;
        if (keys['ArrowRight']) players[1].x += speed;
    }

    // Clamp players
    for (const p of players) {
        p.x = Math.max(15, Math.min(W-15, p.x));
        p.y = Math.max(15, Math.min(H-15, p.y));
        if (p.invincible > 0) p.invincible--;
        if (p.shootCooldown > 0) p.shootCooldown--;
    }

    // Player shooting
    if (!players[0].dead && players[0].shootCooldown <= 0) {
        playerBullets.push({ x: players[0].x, y: players[0].y-15, vx: 0, vy: -8, size: 4, color: '#00ffcc', owner: 0 });
        playerBullets.push({ x: players[0].x-6, y: players[0].y-10, vx: -1, vy: -7.5, size: 3, color: '#00ffcc', owner: 0 });
        playerBullets.push({ x: players[0].x+6, y: players[0].y-10, vx: 1, vy: -7.5, size: 3, color: '#00ffcc', owner: 0 });
        players[0].shootCooldown = 8;
        playShoot();
    }
    if (!players[1].dead && players[1].shootCooldown <= 0) {
        playerBullets.push({ x: players[1].x, y: players[1].y-15, vx: 0, vy: -8, size: 4, color: '#ff6699', owner: 1 });
        playerBullets.push({ x: players[1].x-6, y: players[1].y-10, vx: -1, vy: -7.5, size: 3, color: '#ff6699', owner: 1 });
        playerBullets.push({ x: players[1].x+6, y: players[1].y-10, vx: 1, vy: -7.5, size: 3, color: '#ff6699', owner: 1 });
        players[1].shootCooldown = 8;
        playShoot();
    }

    // Update player bullets
    for (let i = playerBullets.length-1; i >= 0; i--) {
        const b = playerBullets[i];
        b.x += b.vx; b.y += b.vy;
        if (b.y < -20 || b.y > H+20 || b.x < -20 || b.x > W+20) playerBullets.splice(i, 1);
    }

    // Update enemies
    for (let i = enemies.length-1; i >= 0; i--) {
        const e = enemies[i];
        e.x += e.vx;
        e.y += e.vy;
        e.angle += 0.05;
        if (e.x < 20 || e.x > W-20) e.vx *= -1;
        e.shootTimer++;
        if (e.shootTimer >= e.shootInterval && e.y > 30) {
            enemyShoot(e);
            e.shootTimer = 0;
        }
        // Remove if off screen bottom
        if (e.y > H + 50) { enemies.splice(i, 1); continue; }

        // Player bullet hits enemy
        for (let j = playerBullets.length-1; j >= 0; j--) {
            const b = playerBullets[j];
            const dx = b.x - e.x, dy = b.y - e.y;
            if (Math.abs(dx) < 20 && Math.abs(dy) < 20) {
                e.hp--;
                playerBullets.splice(j, 1);
                spawnParticles(b.x, b.y, '#ffffff', 3);
                if (e.hp <= 0) {
                    score[b.owner] += 10 * difficulty;
                    totalKills[b.owner]++;
                    enemies.splice(i, 1);
                    spawnParticles(e.x, e.y, e.type===0?'#ff3366':e.type===1?'#ffaa00':e.type===2?'#cc44ff':'#44ff88', 15);
                    playEnemyDie();
                }
                break;
            }
        }
    }

    // Update enemy bullets
    for (let i = enemyBullets.length-1; i >= 0; i--) {
        const b = enemyBullets[i];
        b.x += b.vx; b.y += b.vy;
        if (b.x < -30 || b.x > W+30 || b.y < -30 || b.y > H+30) { enemyBullets.splice(i, 1); continue; }

        // Hit players
        for (const p of players) {
            if (p.dead || p.invincible > 0) continue;
            const dx = b.x - p.x, dy = b.y - p.y;
            if (Math.abs(dx) < 12 && Math.abs(dy) < 12) {
                enemyBullets.splice(i, 1);
                p.lives--;
                p.invincible = 90;
                spawnParticles(p.x, p.y, p.color, 20);
                playHit();
                if (p.lives <= 0) p.dead = true;
                break;
            }
        }
    }

    // Update particles
    for (let i = particles.length-1; i >= 0; i--) {
        const p = particles[i];
        p.x += p.vx; p.y += p.vy;
        p.vx *= 0.95; p.vy *= 0.95;
        p.life--;
        if (p.life <= 0) particles.splice(i, 1);
    }

    // Check game over
    if (players[0].dead && players[1].dead) {
        gameState = 'gameover';
    }
}

function draw() {
    ctx.fillStyle = '#0a0a1a';
    ctx.fillRect(0, 0, W, H);

    // Grid background
    ctx.strokeStyle = 'rgba(30,40,80,0.3)';
    ctx.lineWidth = 1;
    const gridSize = 50;
    for (let x = 0; x < W; x += gridSize) { ctx.beginPath(); ctx.moveTo(x,0); ctx.lineTo(x,H); ctx.stroke(); }
    for (let y = 0; y < H; y += gridSize) { ctx.beginPath(); ctx.moveTo(0,y); ctx.lineTo(W,y); ctx.stroke(); }

    if (gameState === 'menu') {
        drawMenu();
        return;
    }

    // Draw particles
    for (const p of particles) {
        const alpha = p.life / p.maxLife;
        ctx.fillStyle = p.color + Math.floor(alpha*255).toString(16).padStart(2,'0');
        ctx.beginPath();
        ctx.arc(p.x, p.y, p.size * alpha, 0, Math.PI*2);
        ctx.fill();
    }

    // Draw player bullets
    for (const b of playerBullets) {
        ctx.fillStyle = b.color;
        ctx.shadowColor = b.color;
        ctx.shadowBlur = 8;
        ctx.beginPath();
        ctx.arc(b.x, b.y, b.size, 0, Math.PI*2);
        ctx.fill();
    }
    ctx.shadowBlur = 0;

    // Draw enemies
    for (const e of enemies) {
        const colors = ['#ff3366','#ffaa00','#cc44ff','#44ff88'];
        ctx.fillStyle = colors[e.type];
        ctx.shadowColor = colors[e.type];
        ctx.shadowBlur = 12;
        // Draw as rotated square
        ctx.save();
        ctx.translate(e.x, e.y);
        ctx.rotate(e.angle);
        ctx.fillRect(-e.w/2, -e.h/2, e.w, e.h);
        // HP bar
        ctx.restore();
        ctx.shadowBlur = 0;
        if (e.maxHp > 1) {
            ctx.fillStyle = '#333';
            ctx.fillRect(e.x-15, e.y+18, 30, 4);
            ctx.fillStyle = colors[e.type];
            ctx.fillRect(e.x-15, e.y+18, 30*(e.hp/e.maxHp), 4);
        }
    }

    // Draw enemy bullets
    for (const b of enemyBullets) {
        ctx.fillStyle = b.color;
        ctx.shadowColor = b.color;
        ctx.shadowBlur = 6;
        ctx.beginPath();
        ctx.arc(b.x, b.y, b.size, 0, Math.PI*2);
        ctx.fill();
    }
    ctx.shadowBlur = 0;

    // Draw players
    for (const p of players) {
        if (p.dead) continue;
        const blink = p.invincible > 0 && Math.floor(p.invincible/5) % 2 === 0;
        if (blink) continue;
        ctx.fillStyle = p.color;
        ctx.shadowColor = p.color;
        ctx.shadowBlur = 15;
        // Ship shape
        ctx.beginPath();
        ctx.moveTo(p.x, p.y - 15);
        ctx.lineTo(p.x - 12, p.y + 10);
        ctx.lineTo(p.x, p.y + 5);
        ctx.lineTo(p.x + 12, p.y + 10);
        ctx.closePath();
        ctx.fill();
        ctx.shadowBlur = 0;

        // Lives display
        for (let i = 0; i < p.lives; i++) {
            const lx = p.id === 0 ? 20 + i*18 : W - 60 + i*18;
            const ly = p.id === 0 ? H-20 : H-20;
            ctx.fillStyle = p.color;
            ctx.beginPath();
            ctx.arc(lx, ly, 5, 0, Math.PI*2);
            ctx.fill();
        }
    }

    // HUD
    ctx.fillStyle = '#fff';
    ctx.font = 'bold 14px monospace';
    ctx.textAlign = 'left';
    ctx.fillText(`P1: ${score[0]}`, 20, 25);
    ctx.textAlign = 'right';
    ctx.fillText(`P2: ${score[1]}`, W-20, 25);
    ctx.textAlign = 'center';
    ctx.fillText(`WAVE ${wave}`, W/2, 25);

    if (gameState === 'gameover') {
        drawGameOver();
    }
}

function drawMenu() {
    ctx.fillStyle = '#00ffcc';
    ctx.font = `bold ${Math.min(48, W*0.05)}px monospace`;
    ctx.textAlign = 'center';
    ctx.shadowColor = '#00ffcc';
    ctx.shadowBlur = 20;
    ctx.fillText('TWIN BULLET HELL', W/2, H*0.3);
    ctx.shadowBlur = 0;

    ctx.fillStyle = '#aaa';
    ctx.font = '16px monospace';
    ctx.fillText('PLAYER 1: WASD to move (auto-shoot)', W/2, H*0.45);
    ctx.fillText('PLAYER 2: ARROW KEYS to move (auto-shoot)', W/2, H*0.52);
    ctx.fillText('Dodge enemy bullets! Kill enemies for points!', W/2, H*0.60);
    ctx.fillText('Each player has 3 lives. Last one standing wins!', W/2, H*0.67);

    ctx.fillStyle = '#fff';
    ctx.font = 'bold 20px monospace';
    const pulse = Math.sin(Date.now()*0.005)*0.5+0.5;
    ctx.globalAlpha = pulse;
    ctx.fillText('PRESS ANY KEY TO START', W/2, H*0.8);
    ctx.globalAlpha = 1;

    // Decorative bullets
    const t = Date.now()*0.002;
    for (let i = 0; i < 20; i++) {
        const x = W/2 + Math.cos(t+i*0.5)*150;
        const y = H*0.3 + Math.sin(t*1.3+i*0.7)*80;
        ctx.fillStyle = `hsl(${(i*20+Date.now()*0.05)%360}, 100%, 60%)`;
        ctx.beginPath();
        ctx.arc(x, y, 3, 0, Math.PI*2);
        ctx.fill();
    }
}

function drawGameOver() {
    ctx.fillStyle = 'rgba(0,0,0,0.7)';
    ctx.fillRect(0, 0, W, H);

    ctx.fillStyle = '#ff3366';
    ctx.font = `bold ${Math.min(42, W*0.04)}px monospace`;
    ctx.textAlign = 'center';
    ctx.shadowColor = '#ff3366';
    ctx.shadowBlur = 15;
    ctx.fillText('GAME OVER', W/2, H*0.3);
    ctx.shadowBlur = 0;

    ctx.fillStyle = '#fff';
    ctx.font = '20px monospace';
    const winner = players[0].dead && players[1].dead ? (score[0]===score[1]?'TIE!': score[0]>score[1]?'PLAYER 1 WINS!':'PLAYER 2 WINS!') : '';
    ctx.fillText(winner, W/2, H*0.4);
    ctx.fillText(`P1: ${score[0]} pts (${totalKills[0]} kills)`, W/2-80, H*0.5);
    ctx.fillText(`P2: ${score[1]} pts (${totalKills[1]} kills)`, W/2+80, H*0.5);

    const pulse = Math.sin(Date.now()*0.004)*0.5+0.5;
    ctx.globalAlpha = pulse;
    ctx.font = 'bold 18px monospace';
    ctx.fillText('PRESS SPACE TO RESTART', W/2, H*0.65);
    ctx.globalAlpha = 1;
}

let lastTime = 0;
function gameLoop(time) {
    const dt = time - lastTime;
    lastTime = time;
    update();
    draw();
    requestAnimationFrame(gameLoop);
}
requestAnimationFrame(gameLoop);
</script>
</body>
