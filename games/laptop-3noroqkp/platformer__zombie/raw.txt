let audioCtx = null;
function initAudio() {
    if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
}
function playSound(type) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    
    if (type === 'shoot') {
        osc.type = 'square';
        osc.frequency.setValueAtTime(400, audioCtx.currentTime);
        osc.frequency.exponentialRampToValueAtTime(100, audioCtx.currentTime + 0.1);
        gain.gain.setValueAtTime(0.3, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.1);
        osc.start();
        osc.stop(audioCtx.currentTime + 0.1);
    } else if (type === 'hit') {
        osc.type = 'sawtooth';
        osc.frequency.setValueAtTime(200, audioCtx.currentTime);
        osc.frequency.exponentialRampToValueAtTime(50, audioCtx.currentTime + 0.2);
        gain.gain.setValueAtTime(0.4, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.2);
        osc.start();
        osc.stop(audioCtx.currentTime + 0.2);
    } else if (type === 'zombieDie') {
        osc.type = 'sawtooth';
        osc.frequency.setValueAtTime(300, audioCtx.currentTime);
        osc.frequency.exponentialRampToValueAtTime(30, audioCtx.currentTime + 0.4);
        gain.gain.setValueAtTime(0.5, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.4);
        osc.start();
        osc.stop(audioCtx.currentTime + 0.4);
    } else if (type === 'jump') {
        osc.type = 'sine';
        osc.frequency.setValueAtTime(200, audioCtx.currentTime);
        osc.frequency.exponentialRampToValueAtTime(500, audioCtx.currentTime + 0.1);
        gain.gain.setValueAtTime(0.2, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.1);
        osc.start();
        osc.stop(audioCtx.currentTime + 0.1);
    } else if (type === 'playerHit') {
        osc.type = 'sawtooth';
        osc.frequency.setValueAtTime(150, audioCtx.currentTime);
        osc.frequency.exponentialRampToValueAtTime(40, audioCtx.currentTime + 0.3);
        gain.gain.setValueAtTime(0.5, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.3);
        osc.start();

        osc.type = 'sine';

        gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.5);
        osc.start();

    }
}

// Game State
let gameState = 'start'; // start, playing, gameover
let score = 0;
let wave = 1;

// Particles
let particles = [];
function spawnParticles(x, y, color, count) {
    for (let i = 0; i < count; i++) {
        const angle = Math.random() * Math.PI * 2;
        const speed = Math.random() * 4 + 1;
        particles.push({
            x, y,
            vx: Math.cos(angle) * speed,
            vy: Math.sin(angle) * speed - 2,
            life: 30 + Math.random() * 20,
            maxLife: 50,
            color,
            size: Math.random() * 4 + 2
        });
    }
}

function spawnZombie(x) {
    const type = Math.random() < 0.2 && wave >= 3 ? 'brute' : (Math.random() < 0.15 && wave >= 2 ? 'runner' : 'normal');
    const baseSize = type === 'brute' ? 40 : 30;
    zombies.push({

        y: H - 85,
        vx: 0, vy: 0,
        w: baseSize + 10, h: 50,
        health: type === 'brute' ? 5 : (type === 'runner' ? 2 : 3),
        maxHealth: type === 'brute' ? 5 : (type === 'runner' ? 2 : 3),
        speed: type === 'runner' ? 2.5 : (type === 'brute' ? 0.8 : 1.2),

        onGround: false,
        attackCooldown: 0,
        hitFlash: 0
    });

    initAudio();
    gameState = 'playing';
    score = 0;
    wave = 1;

function nextWave() {
    wave++;
    zombiesKilledThisWave = 0;
    zombiesToKillForWave = 4 + wave * 2;
    waveTransition = 120;
    playSound('wave');

let camX = 0;

function update() {
    if (gameState !== 'playing') return;

    // Wave transition
    if (waveTransition > 0) waveTransition--;

    // Player movement
    let moveSpeed = 4;
    if (keys['KeyA'] || keys['ArrowLeft']) { player.vx = -moveSpeed; player.facing = -1; }
    else if (keys['KeyD'] || keys['ArrowRight']) { player.vx = moveSpeed; player.facing = 1; }
    else player.vx *= 0.8;

    if ((keys['KeyW'] || keys['ArrowUp'] || keys['Space']) && player.onGround) {
        player.vy = -12;
        player.onGround = false;
        playSound('jump');
    }

    // Shooting
    if (player.shootCooldown > 0) player.shootCooldown--;
    if ((keys['KeyF'] || keys['KeyJ']) && player.shootCooldown <= 0) {
        bullets.push({
            x: player.x + player.w / 2 + player.facing * 20,
            y: player.y + 15,
            vx: player.facing * 10,
            vy: 0,
            life: 60
        });
        player.shootCooldown = 12;
        playSound('shoot');
    }

    // Player physics
    player.vy += GRAVITY;
    player.x += player.vx;
    player.y += player.vy;

    // Platform collision for player
    player.onGround = false;
    for (let p of platforms) {
        if (player.x + player.w > p.x && player.x < p.x + p.w) {
            if (player.vy >= 0 && player.y + player.h >= p.y && player.y + player.h - player.vy <= p.y + 10) {
                player.y = p.y - player.h;
                player.vy = 0;
                player.onGround = true;
            }
        }
    }

    // World bounds
    if (player.x < 0) player.x = 0;
    if (player.x > WORLD_WIDTH - player.w) player.x = WORLD_WIDTH - player.w;
    if (player.y > H + 100) {
        player.health -= 20;
        player.y = 0;
        player.vy = 0;
        playSound('playerHit');
    }

    // Invincibility timer
    if (player.invincible > 0) player.invincible--;

    // Update bullets
    for (let i = bullets.length - 1; i >= 0; i--) {
        bullets[i].x += bullets[i].vx;
        bullets[i].y += bullets[i].vy;
        bullets[i].life--;
        if (bullets[i].life <= 0 || bullets[i].x < 0 || bullets[i].x > WORLD_WIDTH) {
            bullets.splice(i, 1);
        }
    }

    // Update zombies
    for (let i = zombies.length - 1; i >= 0; i--) {
        let z = zombies[i];
        if (z.hitFlash > 0) z.hitFlash--;

        // Move toward player
        let dx = player.x - z.x;
        if (Math.abs(dx) > 5) {
            z.vx = Math.sign(dx) * z.speed;
        } else {
            z.vx = 0;
        }

        // Physics
        z.vy += GRAVITY;
        z.x += z.vx;
        z.y += z.vy;

        // Platform collision for zombie
        z.onGround = false;
        for (let p of platforms) {
            if (z.x + z.w > p.x && z.x < p.x + p.w) {
                if (z.vy >= 0 && z.y + z.h >= p.y && z.y + z.h - z.vy <= p.y + 10) {
                    z.y = p.y - z.h;
                    z.vy = 0;
                    z.onGround = true;
                }
            }
        }

        // Zombie attacks player
        if (player.invincible <= 0) {
            if (z.x + z.w > player.x && z.x < player.x + player.w &&
                z.y + z.h > player.y && z.y < player.y + player.h) {
                let dmg = z.type === 'brute' ? 25 : 10;
                player.health -= dmg;
                player.invincible = 30;
                player.vx = Math.sign(player.x - z.x) * 8;
                player.vy = -5;
                playSound('playerHit');
                spawnParticles(player.x + player.w/2, player.y + player.h/2, '#ff4444', 5);
            }
        }

        // Bullet collision with zombie
        for (let j = bullets.length - 1; j >= 0; j--) {
            let b = bullets[j];
            if (b.x > z.x && b.x < z.x + z.w && b.y > z.y && b.y < z.y + z.h) {
                z.health--;
                z.hitFlash = 8;
                bullets.splice(j, 1);
                playSound('hit');
                spawnParticles(b.x, b.y, '#44ff44', 3);
                if (z.health <= 0) {
                    zombies.splice(i, 1);
                    score += z.type === 'brute' ? 50 : (z.type === 'runner' ? 30 : 20);
                    zombiesKilledThisWave++;
                    playSound('zombieDie');
                    spawnParticles(z.x + z.w/2, z.y + z.h/2, '#44ff44', 12);
                }
                break;
            }
        }

        // Remove if fell off world
        if (z.y > H + 200) {
            zombies.splice(i, 1);
        }
    }

    // Spawn zombies to maintain population
    let targetCount = Math.min(3 + wave * 2, 15);
    if (zombies.length < targetCount && waveTransition <= 0) {
        if (Math.random() < 0.02) {
            let spawnX = player.x > WORLD_WIDTH / 2 ? Math.random() * (player.x - 400) : player.x + 300 + Math.random() * 400;
            spawnZombie(Math.max(0, Math.min(WORLD_WIDTH - 50, spawnX)));
        }
    }

    // Wave check
    if (zombiesKilledThisWave >= zombiesToKillForWave) {
        nextWave();
    }

    // Update particles
    for (let i = particles.length - 1; i >= 0; i--) {
        let p = particles[i];
        p.x += p.vx;
        p.y += p.vy;
        p.vy += 0.3;
        p.life--;
        if (p.life <= 0) particles.splice(i, 1);
    }

    // Camera
    camX = Math.max(0, Math.min(WORLD_WIDTH - W, player.x - W / 3));

    // Game over
    if (player.health <= 0) {
        gameState = 'gameover';
        try { localStorage.setItem('zombiePlatformerHighScore', score); } catch(e) {}
    }
}

function draw() {
    ctx.clearRect(0, 0, W, H);

    // Background gradient
    let grad = ctx.createLinearGradient(0, 0, 0, H);
    grad.addColorStop(0, '#1a1a2e');
    grad.addColorStop(1, '#16213e');
    ctx.fillStyle = grad;
    ctx.fillRect(0, 0, W, H);

    // Stars
    ctx.fillStyle = '#ffffff';
    for (let i = 0; i < 50; i++) {
        let sx = ((i * 137 + 50) % WORLD_WIDTH - camX * 0.3) % W;
        let sy = (i * 97) % (H * 0.6);
        ctx.globalAlpha = 0.3 + Math.sin(Date.now() / 1000 + i) * 0.2;
        ctx.fillRect(sx, sy, 2, 2);
    }
    ctx.globalAlpha = 1;

    // Moon
    ctx.beginPath();
    ctx.arc(W - 100, 80, 30, 0, Math.PI * 2);
    ctx.fillStyle = '#e8d5a3';
    ctx.fill();
    ctx.beginPath();
    ctx.arc(W - 90, 75, 25, 0, Math.PI * 2);
    ctx.fillStyle = '#1a1a2e';
    ctx.fill();

    if (gameState === 'start') {
        drawStartScreen();
        return;
    }

    ctx.save();
    ctx.translate(-camX, 0);

    // Platforms
    for (let p of platforms) {
        if (p.x + p.w < camX - 50 || p.x > camX + W + 50) continue;
        if (p.type === 'ground') {
            ctx.fillStyle = '#2d4a2d';
            ctx.fillRect(p.x, p.y, p.w, p.h);
            ctx.fillStyle = '#3d6b3d';
            for (let gx = 0; gx < p.w; gx += 20) {
                ctx.fillRect(p.x + gx, p.y, 15, 5);
            }
        } else {
            ctx.fillStyle = '#5a4a3a';
            ctx.fillRect(p.x, p.y, p.w, p.h);
            ctx.fillStyle = '#7a6a5a';
            ctx.fillRect(p.x, p.y, p.w, 5);
        }
    }

    // Bullets
    for (let b of bullets) {
        ctx.fillStyle = '#ffdd44';
        ctx.beginPath();
        ctx.arc(b.x, b.y, 4, 0, Math.PI * 2);
        ctx.fill();
        ctx.fillStyle = '#ffffff';
        ctx.beginPath();
        ctx.arc(b.x, b.y, 2, 0, Math.PI * 2);
        ctx.fill();
    }

    // Player
    if (player.invincible <= 0 || Math.floor(player.invincible / 4) % 2 === 0) {
        drawPlayer();
    }

    // Zombies
    for (let z of zombies) {
        drawZombie(z);
    }

    // Particles
    for (let p of particles) {
        ctx.globalAlpha = p.life / p.maxLife;
        ctx.fillStyle = p.color;
        ctx.fillRect(p.x - p.size/2, p.y - p.size/2, p.size, p.size);
    }
    ctx.globalAlpha = 1;

    ctx.restore();

    // HUD
    drawHUD();

    if (gameState === 'gameover') {
        drawGameOver();
    }
}

function drawPlayer() {
    let px = player.x, py = player.y;
    let f = player.facing;

    // Body
    ctx.fillStyle = '#4488ff';
    ctx.fillRect(px + 5, py + 15, 20, 25);
    
    // Head
    ctx.fillStyle = '#ffcc99';
    ctx.beginPath();
    ctx.arc(px + player.w/2, py + 10, 10, 0, Math.PI * 2);
    ctx.fill();

    // Eyes
    ctx.fillStyle = '#333';
    ctx.fillRect(px + player.w/2 + f * 4 - 1, py + 8, 3, 3);
    ctx.fillRect(px + player.w/2 + f * 4 + 4 - 1, py + 8, 3, 3);

    // Legs
    ctx.fillStyle = '#3355aa';
    let legAnim = Math.abs(player.vx) > 0.5 ? Math.sin(Date.now() / 100) * 5 : 0;
    ctx.fillRect(px + 7, py + 40, 6, 10 + legAnim);
    ctx.fillRect(px + 17, py + 40, 6, 10 - legAnim);

    // Gun arm
    ctx.fillStyle = '#888';
    ctx.fillRect(px + player.w/2 + f * 15 - 2, py + 20, 15 * f, 5);
}

function drawZombie(z) {
    let flash = z.hitFlash > 0;
    
    // Body
    if (flash) ctx.fillStyle = '#ffffff';
    else if (z.type === 'brute') ctx.fillStyle = '#6b3a6b';
    else if (z.type === 'runner') ctx.fillStyle = '#cc4444';
    else ctx.fillStyle = '#55aa55';

    let bw = z.w, bh = z.h;
    ctx.fillRect(z.x + 5, z.y + 12, bw - 10, bh - 20);

    // Head
    if (flash) ctx.fillStyle = '#ffffff';
    else ctx.fillStyle = z.type === 'brute' ? '#8b5a8b' : '#7bc47b';
    ctx.beginPath();
    ctx.arc(z.x + bw/2, z.y + 10, bw/2 - 3, 0, Math.PI * 2);
    ctx.fill();

    // Eyes (glowing)
    ctx.fillStyle = z.type === 'brute' ? '#ff00ff' : '#ffff00';
    let eyeX = z.x + bw/2 + (player.x > z.x ? 3 : -3);
    ctx.fillRect(eyeX - 1, z.y + 8, 4, 4);
    ctx.fillRect(eyeX + 5, z.y + 8, 4, 4);

    // Legs
    ctx.fillStyle = flash ? '#ffffff' : '#3a6b3a';
    let legAnim = Math.sin(Date.now() / 150 + z.x) * 4;
    ctx.fillRect(z.x + 7, z.y + bh - 12, 7, 12 + legAnim);
    ctx.fillRect(z.x + bw - 14, z.y + bh - 12, 7, 12 - legAnim);

    // Health bar
    let hpPct = z.health / z.maxHealth;
    ctx.fillStyle = '#333';
    ctx.fillRect(z.x, z.y - 10, bw, 5);
    ctx.fillStyle = hpPct > 0.5 ? '#44ff44' : (hpPct > 0.25 ? '#ffaa00' : '#ff4444');
    ctx.fillRect(z.x, z.y - 10, bw * hpPct, 5);
}

function drawHUD() {
    // Health bar
    ctx.fillStyle = 'rgba(0,0,0,0.5)';
    ctx.fillRect(15, 15, 204, 30);
    ctx.fillStyle = '#333';
    ctx.fillRect(20, 20, 200, 20);
    let hpPct = Math.max(0, player.health) / player.maxHealth;
    ctx.fillStyle = hpPct > 0.5 ? '#44ff44' : (hpPct > 0.25 ? '#ffaa00' : '#ff4444');
    ctx.fillRect(20, 20, 200 * hpPct, 20);
    ctx.fillStyle = '#fff';
    ctx.font = 'bold 12px monospace';
    ctx.fillText('HP', 85, 34);

    // Score
    ctx.fillStyle = 'rgba(0,0,0,0.5)';
    ctx.fillRect(W/2 - 80, 15, 160, 30);
    ctx.fillStyle = '#fff';
    ctx.font = 'bold 16px monospace';
    ctx.textAlign = 'center';
    ctx.fillText('SCORE: ' + score, W/2, 35);

    // Wave
    ctx.fillStyle = 'rgba(0,0,0,0.5)';
    ctx.fillRect(W - 180, 15, 165, 30);
    ctx.fillStyle = '#ffcc44';
    ctx.font = 'bold 14px monospace';
    ctx.fillText('WAVE ' + wave, W - 97, 35);

    // Wave progress
    let progPct = zombiesKilledThisWave / zombiesToKillForWave;
    ctx.fillStyle = '#333';
    ctx.fillRect(W - 180, 48, 165, 6);
    ctx.fillStyle = '#ffcc44';
    ctx.fillRect(W - 180, 48, 165 * progPct, 6);

    ctx.textAlign = 'left';
}

function drawStartScreen() {
    ctx.fillStyle = 'rgba(0,0,0,0.7)';
    ctx.fillRect(0, 0, W, H);

    ctx.textAlign = 'center';
    ctx.fillStyle = '#44ff44';
    ctx.font = 'bold 48px monospace';
    ctx.fillText('ZOMBIE PLATFORMER', W/2, H/3);

    ctx.fillStyle = '#fff';
    ctx.font = '18px monospace';
    ctx.fillText('Survive the zombie hordes!', W/2, H/3 + 50);

    ctx.font = '16px monospace';
    ctx.fillStyle = '#aaa';
    let startY = H/2 + 20;
    let lines = [
        'A/D or ←/→ - Move',
        'W or ↑ or SPACE - Jump',
        'F or J - Shoot',
        '',
        'Press ENTER to Start'
    ];
    lines.forEach((line, i) => {
        ctx.fillText(line, W/2, startY + i * 28);
    });

    // High score
    try {
        let hs = localStorage.getItem('zombiePlatformerHighScore') || 0;
        if (hs > 0) {
            ctx.fillStyle = '#ffcc44';

            ctx.fillText('HIGH SCORE: ' + hs, W/2, H - 60);

function drawGameOver() {
    ctx.fillStyle = 'rgba(0,0,0,0.8)';
    ctx.fillRect(0, 0, W, H);

    ctx.textAlign = 'center';
    ctx.fillStyle = '#ff4444';
    ctx.font = 'bold 52px monospace';
    ctx.fillText('YOU DIED', W/2, H/3);

    ctx.fillStyle = '#fff';
    ctx.font = '20px monospace';
    ctx.fillText('Score: ' + score, W/2, H/3 + 60);
    ctx.fillText('Wave: ' + wave, W/2, H/3 + 95);

    try {
        let hs = localStorage.getItem('zombiePlatformerHighScore') || 0;
        if (score > hs) {
            ctx.fillStyle = '#ffcc44';
            ctx.fillText('NEW HIGH SCORE!', W/2, H/3 + 135);
        } else if (hs > 0) {
            ctx.fillStyle = '#aaa';
            ctx.font = '16px monospace';
            ctx.fillText('High Score: ' + hs, W/2, H/3 + 135);
        }
    } catch(e) {}

    ctx.fillStyle = '#44ff44';
    ctx.font = '18px monospace';
    let blink = Math.sin(Date.now() / 400) > 0;
    if (blink) ctx.fillText('Press ENTER to Restart', W/2, H - 80);

    ctx.textAlign = 'left';
}

// Game loop
function gameLoop() {
    update();
    draw();
    requestAnimationFrame(gameLoop);
}

generatePlatforms();
gameLoop();
