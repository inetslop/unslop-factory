function initAudio() {

function playSound(freq, dur, type, vol) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = type || 'sine';
    osc.frequency.value = freq;
    gain.gain.setValueAtTime(vol || 0.3, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + dur);
}

function isInObstacle(x, y, obs) {
    const ox = obs.x * W, oy = obs.y * H;
    const ow = obs.w * W, oh = obs.h * H;
    return x >= ox && x <= ox + ow && y >= oy && y <= oy + oh;
}

function update() {
    if (gameState !== 'playing' || ballInHole) return;

    // Apply friction (more in sand)
    let fric = FRICTION;
    for (const obs of getCourse().obstacles) {
        if (obs.type === 'sand' && isInObstacle(ball.x, ball.y, obs)) {
            fric = 0.92;
            break;
        }
    }
    ball.vx *= fric;
    ball.vy *= fric;

    const speed = Math.sqrt(ball.vx * ball.vx + ball.vy * ball.vy);
    if (speed < MIN_SPEED) {
        ball.vx = 0;
        ball.vy = 0;
    } else {
        ball.x += ball.vx;
        ball.y += ball.vy;

        // Trail
        if (speed > 2) {
            trail.push({ x: ball.x, y: ball.y, life: 1 });
        }
    }

    // Wall bounce
    if (ball.x - ball.r < 0) { ball.x = ball.r; ball.vx *= -0.7; playSound(300, 0.1, 'triangle', 0.15); }
    if (ball.x + ball.r > W) { ball.x = W - ball.r; ball.vx *= -0.7; playSound(300, 0.1, 'triangle', 0.15); }
    if (ball.y - ball.r < 0) { ball.y = ball.r; ball.vy *= -0.7; playSound(300, 0.1, 'triangle', 0.15); }
    if (ball.y + ball.r > H) { ball.y = H - ball.r; ball.vy *= -0.7; playSound(300, 0.1, 'triangle', 0.15); }

    // Water penalty
    for (const obs of getCourse().obstacles) {
        if (obs.type === 'water' && isInObstacle(ball.x, ball.y, obs)) {
            ball.vx = 0;
            ball.vy = 0;
            strokes++;
            const course = getCourse();
            ball.x = course.tee.x * W;
            ball.y = course.tee.y * H;
            playSound(150, 0.3, 'sawtooth', 0.2);
            addParticles(ball.x, ball.y, '#4fc3f7', 10);
        }
    }

    // Check hole
    const course = getCourse();
    const cx = course.cup.x * W;
    const cy = course.cup.y * H;
    const dxc = ball.x - cx;
    const dyc = ball.y - cy;
    const distC = Math.sqrt(dxc * dxc + dyc * dyc);
    if (distC < 18 && speed < 5) {
        // Drop in hole
        ballInHole = true;
        ball.vx = 0;
        ball.vy = 0;
        playSound(80, 0.4, 'sine', 0.3);
        playSound(160, 0.3, 'sine', 0.2);
        addParticles(cx, cy, '#FFD700', 20);
    } else if (distC < 18 && speed >= 5) {
        // Deflect slightly
        const angle = Math.atan2(dyc, dxc);
        ball.vx += Math.cos(angle) * 1;
        ball.vy += Math.sin(angle) * 1;
    }

    // Update trail
    for (let i = trail.length - 1; i >= 0; i--) {
        trail[i].life -= 0.05;
        if (trail[i].life <= 0) trail.splice(i, 1);
    }

    // Update particles
    for (let i = particles.length - 1; i >= 0; i--) {
        const p = particles[i];
        p.x += p.vx;
        p.y += p.vy;
        p.vy += 0.3;
        p.life -= 0.02;
        if (p.life <= 0) particles.splice(i, 1);
    }
}

function addParticles(x, y, color, count) {
    for (let i = 0; i < count; i++) {
        const angle = Math.random() * Math.PI * 2;
        const speed = Math.random() * 4 + 1;
        particles.push({ x, y, vx: Math.cos(angle) * speed, vy: Math.sin(angle) * speed - 2, life: 1, color });
    }
}

    if (diff === 0) return 'Par';
    if (diff === 1) return 'Bogey';
    if (diff === 2) return 'Double Bogey';
    return '+' + diff;
}

function roundRect(x, y, w, h, r) {
    ctx.beginPath();
    ctx.moveTo(x + r, y);
    ctx.lineTo(x + w - r, y);
    ctx.quadraticCurveTo(x + w, y, x + w, y + r);
    ctx.lineTo(x + w, y + h - r);
    ctx.quadraticCurveTo(x + w, y + h, x + w - r, y + h);
    ctx.lineTo(x + r, y + h);
    ctx.quadraticCurveTo(x, y + h, x, y + h - r);
    ctx.lineTo(x, y + r);
    ctx.quadraticCurveTo(x, y, x + r, y);
    ctx.closePath();
}

    if (gameState === 'holeComplete') {

        if (lastBallInHoleCheck === 0) lastBallInHoleCheck = now;
        if (now - lastBallInHoleCheck > 1500) {
            gameState = 'holeComplete';
        }
