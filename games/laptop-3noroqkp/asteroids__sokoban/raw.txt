let audioCtx = null;
function initAudio() {
  if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();

function playSound(freq, type, dur, vol) {
  if (!audioCtx) return;
  const o = audioCtx.createOscillator();
  const g = audioCtx.createGain();
  o.type = type;
  o.frequency.value = freq;
  g.gain.setValueAtTime(vol || 0.15, audioCtx.currentTime);
  g.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
  o.connect(g);
  g.connect(audioCtx.destination);
  o.start();
  o.stop(audioCtx.currentTime + dur);
}

function loadLevel(idx) {
  const map = LEVELS[idx].map;
  const rows = map.length;
  const cols = map[0].length;
  grid = [];
  player = { x: 1, y: 1 };
  asteroids = [];
  targets = [];

  for (let r = 0; r < rows; r++) {
    grid[r] = [];
    for (let c = 0; c < cols; c++) {
      const ch = map[r][c];
      if (ch === '#') { walls.push({x: c, y: r}); }
      else if (ch === '@' || ch === 'P') { player = {x: c, y: r}; floorTiles.push({x:c,y:r}); }
      else if (ch === '$') { asteroids.push({x: c, y: r}); floorTiles.push({x:c,y:r}); }
      else if (ch === 'T' || ch === '*') { targets.push({x: c, y: r}); floorTiles.push({x:c,y:r}); }
      else if (ch === '.' || ch === '+' || ch === ' ') { floorTiles.push({x:c,y:r}); }
    }

function getCell(x, y) {
  const map = LEVELS[currentLevel].map;
  if (y < 0 || y >= map.length || x < 0 || x >= map[0].length) return '#';

function isWall(x, y) {
  const map = LEVELS[currentLevel].map;
  if (y < 0 || y >= map.length || x < 0 || x >= map[0].length) return true;
  return map[y][x] === '#';
}

function hasAsteroid(x, y) {
  return asteroids.find(a => a.x === x && a.y === y);

function isTarget(x, y) {
  return targets.some(t => t.x === x && t.y === y);

function checkWin() {
  return asteroids.every(a => isTarget(a.x, a.y));

function move(dx, dy) {
  if (gameState !== 'playing') return;
  const nx = player.x + dx;
  const ny = player.y + dy;

  if (isWall(nx, ny)) { sfxBlock(); return; }

  const ast = hasAsteroid(nx, ny);
  if (ast) {
    const ax = nx + dx;
    const ay = ny + dy;
    if (isWall(ax, ay) || hasAsteroid(ax, ay)) { sfxBlock(); return; }
    ast.x = ax;
    ast.y = ay;
    sfxPush();
    if (isTarget(ax, ay)) { sfxGoal(); spawnParticles(ax, ay); }
  } else {
    sfxMove();

  player.x = nx;
  player.y = ny;
  moves++;

  if (checkWin()) {
    totalMoves += moves;
    gameState = 'levelComplete';
    levelCompleteTimer = 0;
    sfxWin();
  }
}

    const speed = 1 + Math.random() * 3;
    particles.push({ x: ox, y: oy, vx: Math.cos(angle)*speed, vy: Math.sin(angle)*speed, life: 1, color: `hsl(${160+Math.random()*40}, 100%, 60%)` });
  }
}

function getCellSize() {
  const map = LEVELS[currentLevel].map;
  const rows = map.length;
  const cols = Math.max(...map.map(r => r.length));
  const margin = 40;
  return Math.min((W - margin*2) / cols, (H - margin*2 - 60) / rows);
}

function getOffsetX() {
  const map = LEVELS[currentLevel].map;
  const cols = Math.max(...map.map(r => r.length));
  return (W - cols * getCellSize()) / 2;
}

function getOffsetY() {
  const map = LEVELS[currentLevel].map;
  const rows = map.length;
  return (H - rows * getCellSize()) / 2 + 30;

function drawAsteroids() {
  const cs = getCellSize();
  const ox = getOffsetX();
  const oy = getOffsetY();
  asteroids.forEach(a => {
    const cx = ox + a.x * cs + cs/2;
    const cy = oy + a.y * cs + cs/2;
    const r = cs * 0.38;
    const onTarget = isTarget(a.x, a.y);

    // Rock body - irregular shape
    ctx.beginPath();
    for (let i = 0; i < 10; i++) {
      const angle = (i / 10) * Math.PI * 2;
      const wobble = 0.85 + 0.15 * Math.sin(i * 3.7 + a.x * 2 + a.y);
      const px = cx + Math.cos(angle) * r * wobble;
      const py = cy + Math.sin(angle) * r * wobble;
      if (i === 0) ctx.moveTo(px, py);
      else ctx.lineTo(px, py);
    }
    ctx.closePath();

    const grad = ctx.createRadialGradient(cx-r*0.2, cy-r*0.2, 0, cx, cy, r);
    if (onTarget) {
      grad.addColorStop(0, '#8affc0');
      grad.addColorStop(1, '#4a9f6a');
    } else {
      grad.addColorStop(0, '#a08860');
      grad.addColorStop(1, '#5a4a30');
    }
    ctx.fillStyle = grad;
    ctx.fill();
    ctx.strokeStyle = onTarget ? 'rgba(0,255,180,0.8)' : 'rgba(160,130,80,0.6)';
    ctx.lineWidth = 2;
    ctx.stroke();

    // Craters
    for (let i = 0; i < 3; i++) {
      const ca = i * 2.1 + a.x;
      const cr = r * 0.7;
      const ccx = cx + Math.cos(ca) * cr * 0.4;
      const ccy = cy + Math.sin(ca) * cr * 0.4;
      ctx.fillStyle = 'rgba(0,0,0,0.2)';
      ctx.beginPath();
      ctx.arc(ccx, ccy, r*0.15, 0, Math.PI*2);
      ctx.fill();
    }

    if (onTarget) {
      const pulse = 0.5 + 0.5 * Math.sin(Date.now()*0.005);
      ctx.strokeStyle = `rgba(0,255,180,${pulse*0.5})`;
      ctx.lineWidth = 3;
      ctx.beginPath();
      ctx.arc(cx, cy, r + 4, 0, Math.PI*2);
      ctx.stroke();
    }
  });
}

function render() {
  ctx.clearRect(0, 0, W, H);

  if (gameState === 'start') {
    drawStartScreen();
  } else if (gameState === 'playing') {
    drawBackground();
    drawFloor();
    drawWalls();
    drawTargets();
    drawAsteroids();
    drawPlayer();
    drawParticles();
    drawHUD();
  } else if (gameState === 'levelComplete') {
    drawLevelComplete();
  } else if (gameState === 'gameWin') {
    drawGameWin();
  }

  animFrame = requestAnimationFrame(render);
}

function startGame() {
  initAudio();
  currentLevel = 0;
  totalMoves = 0;
  moves = 0;
  loadLevel(0);
  gameState = 'playing';
}

function nextLevel() {
  currentLevel++;
  if (currentLevel >= LEVELS.length) {
    gameState = 'gameWin';
  } else {
    moves = 0;
    loadLevel(currentLevel);
    gameState = 'playing';
  }
}

function resetLevel() {
  moves = 0;
  particles = [];
  loadLevel(currentLevel);
}

// Input
window.addEventListener('keydown', (e) => {
  if (gameState === 'start') {
    startGame();
    return;
  }

  if (gameState === 'levelComplete' && levelCompleteTimer > 1) {
    if (e.key === 'Enter' || e.key === ' ') nextLevel();
    return;
  }

  if (gameState === 'gameWin') {
    if (e.key === 'Enter' || e.key === ' ') startGame();
    return;
  }

  if (gameState !== 'playing') return;

  let dx = 0, dy = 0;
  switch(e.key) {
    case 'ArrowUp': case 'w': case 'W': dy = -1; break;
    case 'ArrowDown': case 's': case 'S': dy = 1; break;
    case 'ArrowLeft': case 'a': case 'A': dx = -1; break;
    case 'ArrowRight': case 'd': case 'D': dx = 1; break;
    case 'r': case 'R': resetLevel(); return;
    default: return;
  }
  e.preventDefault();
  move(dx, dy);
});

// Touch / click support for mobile
let touchStartX = 0, touchStartY = 0;
canvas.addEventListener('touchstart', (e) => {
  initAudio();
  if (gameState === 'start') { startGame(); return; }
  if (gameState === 'levelComplete' && levelCompleteTimer > 1) { nextLevel(); return; }
  if (gameState === 'gameWin') { startGame(); return; }
  touchStartX = e.touches[0].clientX;
  touchStartY = e.touches[0].clientY;
});
canvas.addEventListener('touchend', (e) => {
  if (gameState !== 'playing') return;
  const dx = e.changedTouches[0].clientX - touchStartX;
  const dy = e.changedTouches[0].clientY - touchStartY;
  if (Math.abs(dx) > Math.abs(dy)) {
    if (Math.abs(dx) > 20) move(Math.sign(dx), 0);
  } else {
    if (Math.abs(dy) > 20) move(0, Math.sign(dy));
  }
});

// Try localStorage for high score
try {
  window.addEventListener('beforeunload', () => {
    localStorage.setItem('sokobanAsteroids_best', JSON.stringify(totalMoves));
  });
} catch(e) {}

render();
