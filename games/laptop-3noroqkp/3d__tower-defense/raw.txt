function project(x,y,z){
let cx=Math.cos(camAngle),sx=Math.sin(camAngle);
let rx=x-12,ry=y,rz=z-12;
let px=rx*cx-rz*sx,pz=rx*sx+rz*cx+camHeight;
let py=-ry;
let scale=fov/(pz+camHeight*0.5);
return{x:W/2+px*scale,y:H/2-py*scale-100,scale:scale,depth:pz};
}

function getPathPoint(t){
// t is distance along path in grid units
let totalLen=0;
let segLens=[];
for(let i=0;i<PATH.length-1;i++){
let dx=PATH[i+1][0]-PATH[i][0],dy=PATH[i+1][1]-PATH[i][1];
let l=Math.abs(dx)+Math.abs(dy);
segLens.push(l);totalLen+=l;
}
let dist=t*1.0;
for(let i=0;i<segLens.length;i++){
if(dist<=segLens[i]){
let f=dist/segLens[i];
let x=PATH[i][0]+(PATH[i+1][0]-PATH[i][0])*f;
let y=PATH[i][1]+(PATH[i+1][1]-PATH[i][1])*f;
return{x,y};
}
dist-=segLens[i];
}
return{x:PATH[PATH.length-1][0],y:PATH[PATH.length-1][1]};
}

function getTotalPathLen(){
let total=0;
for(let i=0;i<PATH.length-1;i++){
total+=Math.abs(PATH[i+1][0]-PATH[i][0])+Math.abs(PATH[i+1][1]-PATH[i][1]);
}
return total;
}

o.connect(g);g.connect(audioCtx.destination);
o.start();o.stop(audioCtx.currentTime+dur);
}

}
enemies.splice(i,1);
playSound(880,0.1,'sine',0.08);
updateUI();
}
}

// Tower shooting
for(let t of towers){
t.cooldown-=dt;
if(t.cooldown>0)continue;
let def=TOWER_DEFS[t.type];

let best=null,bestDist=Infinity;
for(let e of enemies){
let dx=e.x-t.x,dy=e.y-t.y;
let d=Math.sqrt(dx*dx+dy*dy);
if(d<range&&d<bestDist){best=e;bestDist=d}
}
if(best){
t.cooldown=def.rate/(1+t.level*0.2);
let dmg=def.dmg*(1+(t.level-1)*0.5);
projectiles.push({x:t.x,y:0.8,z:t.y,tgt:best,dmg:dmg,type:t.type,speed:8,life:2});

}
}

p.life-=dt;

if(!p.tgt||p.tgt.hp<=0){projectiles.splice(i,1);continue}

let d=Math.sqrt(dx*dx+dz*dz);
if(d<0.3){
p.tgt.hp-=p.dmg;

}
// Particles
for(let p of particles){
objects.push({type:'part',data:p,z:project(p.x,0,p.y).depth});
}

objects.sort((a,b)=>b.z-a.z);

for(let obj of objects){
if(obj.type==='tower')drawTower(obj.data);
else if(obj.type==='enemy')drawEnemy(obj.data);
else if(obj.type==='proj')drawProjectile(obj.data);
else if(obj.type==='part')drawParticle(obj.data);

}
ctx.closePath();ctx.stroke();
}

// Ghost tower preview

for(let x=0;x<GRID;x++)for(let y=0;y<GRID;y++){
