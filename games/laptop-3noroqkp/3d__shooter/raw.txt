const FOV = Math.PI / 3;
const NEAR = 0.1, FAR = 200;

function project(x, y, z) {
    // Transform to camera space
    let dx = x - playerX;
    let dy = y - playerY;
    let dz = z - playerZ;
    
    // Rotate by yaw (camera looks in +z direction at yaw=0)
    let cosY = Math.cos(yaw), sinY = Math.sin(yaw);
    let rx = dx * cosY - dz * sinY;
    let rz = dx * sinY + dz * cosY;
    
    // Actually let's use a simpler convention: yaw rotates around Y axis
    // Camera faces -Z at yaw=0, so we rotate world by -yaw
    rx = dx * Math.cos(-yaw) - dz * Math.sin(-yaw);
    rz = dx * Math.sin(-yaw) + dz * Math.cos(-yaw);
    
    let ry = dy;
    
    // Pitch rotation (around X axis)
    let cosP = Math.cos(pitch), sinP = Math.sin(pitch);
    let finalY = ry * cosP - rz * sinP;
    let finalZ = ry * sinP + rz * cosP;
    
    if (finalZ < NEAR) return null;
    
    let sx = (rx / finalZ) * (H / Math.tan(FOV / 2)) + W / 2;
    let sy = -(finalY / finalZ) * (H / Math.tan(FOV / 2)) + H / 2;
    
    return { x: sx, y: sy, z: finalZ, scale: H / Math.tan(FOV / 2) / finalZ };
}

class Enemy {
    constructor(x, y, type) {
        this.x = x;
        this.y = y;
        this.z = 1.0 + Math.random() * 2; // float height
        this.type = type || 'drone'; // drone, tank, fast
        this.health = type === 'tank' ? 5 : (type === 'fast' ? 1 : 3);
        this.maxHealth = this.health;
        this.speed = type === 'fast' ? 4 : (type === 'tank' ? 0.8 : 1.5);
        this.size = type === 'tank' ? 0.8 : (type === 'fast' ? 0.3 : 0.5);
        this.color = type === 'tank' ? '#ff4444' : (type === 'fast' ? '#ffff44' : '#ff8800');
        this.hitTimer = 0;
        this.bobPhase = Math.random() * Math.PI * 2;
        this.alive = true;
    }
    
    update(dt) {
        if (!this.alive) return;
        let dx = -this.x; // move toward player at (0,0)
        let dy = -this.y;
        let dist = Math.sqrt(dx * dx + dy * dy);
        if (dist > 0.5) {
            this.x += (dx / dist) * this.speed * dt;
            this.y += (dy / dist) * this.speed * dt;
        } else {
            // Attack player
            health -= 20 * dt;
            shakeTime = 0.2;
            if (Math.random() < dt * 3) playSound('hurt');
        }
        this.bobPhase += dt * 3;
        this.z = 1.0 + Math.sin(this.bobPhase) * 0.3;
        if (this.hitTimer > 0) this.hitTimer -= dt;
    }
    
    hit() {
        this.health--;
        this.hitTimer = 0.2;
        playSound('hit');
        if (this.health <= 0) {
            this.alive = false;
            score += this.type === 'tank' ? 50 : (this.type === 'fast' ? 30 : 10);
            playSound('explode');
            // Spawn particles
            for (let i = 0; i < 8; i++) {
                let angle = Math.random() * Math.PI * 2;
                let speed = 2 + Math.random() * 3;

                    vx: Math.cos(angle) * speed,
                    vy: Math.sin(angle) * speed,
                    vz: (Math.random() - 0.5) * 2,
                    life: 1,
                    color: this.color
                });
            }
        }
    }
}

let shootCooldown = 0;

function shoot() {
    if (reloading || ammo <= 0) return;
    ammo--;
    muzzleFlash = 0.1;
    playSound('shoot');
    
    // Hitscan: check each enemy, see if it's in the crosshair area
    let bestDist = Infinity;
    let hitEnemy = null;
    
    for (let e of enemies) {
        if (!e.alive) continue;
        let p = project(e.x, e.y, e.z);
        if (!p) continue;
        
        // Check if enemy is near crosshair center
        let dx = p.x - W / 2;
        let dy = p.y - H / 2;
        let distSq = dx * dx + dy * dy;
        let hitRadius = e.size * p.scale * 0.5;
        
        if (distSq < hitRadius * hitRadius && p.z < bestDist) {
            bestDist = p.z;
            hitEnemy = e;
        }
    }
    
    if (hitEnemy) {
        hitMarker = 0.3;
        hitEnemy.hit();
    }
    
    if (ammo <= 0) {
        reloading = true;
        reloadTime = 1.5;
    }
}

        reloadTime -= dt;

            reloading = false;
            ammo = maxAmmo;
        }

    if (keys['KeyR'] && !reloading && ammo < maxAmmo) {
        reloading = true;
        reloadTime = 1.5;
    }
    
    // Wave management
    waveTimer -= dt;
    if (waveTimer <= 0 && enemiesToSpawn === 0 && enemies.filter(e => e.alive).length === 0) {

        spawnTimer -= dt;

            enemiesToSpawn--;
            spawnTimer = Math.max(0.3, 1.5 - wave * 0.1);
        }
    }

    for (let e of enemies) e.update(dt);
    enemies = enemies.filter(e => e.alive);
    
    // Update particles
    for (let p of particles) {
        p.x += p.vx * dt;
        p.y += p.vy * dt;
        p.z += p.vz * dt;
        p.life -= dt * 2;
    }
    particles = particles.filter(p => p.life > 0);
    
    // Timers
    if (muzzleFlash > 0) muzzleFlash -= dt;
    if (shakeTime > 0) shakeTime -= dt;
    if (hitMarker > 0) hitMarker -= dt;
    
    // Check game over
    if (health <= 0) {
        health = 0;
        gameState = 'gameover';
    }
}

function drawEnemy(enemy, proj) {
    let size = enemy.size * proj.scale * 0.08;
    if (size < 1) return;
    
    let x = proj.x, y = proj.y;
    let flash = enemy.hitTimer > 0;

    ctx.translate(x, y);
    
    // Body

    ctx.shadowColor = enemy.color;
    ctx.shadowBlur = 15;
    
    if (enemy.type === 'drone') {
        // Diamond shape
        ctx.beginPath();
        ctx.moveTo(0, -size);
        ctx.lineTo(size * 0.7, 0);
        ctx.lineTo(0, size);
        ctx.lineTo(-size * 0.7, 0);
        ctx.closePath();
        ctx.fill();
        
        // Eye
        ctx.fillStyle = flash ? '#fff' : '#ff0';
        ctx.beginPath();
        ctx.arc(0, -size * 0.2, size * 0.25, 0, Math.PI * 2);
        ctx.fill();
    } else if (enemy.type === 'tank') {
        // Large hexagon
        ctx.beginPath();
        for (let i = 0; i < 6; i++) {
            let a = (i / 6) * Math.PI * 2 - Math.PI / 2;
            let px = Math.cos(a) * size * 0.8;
            let py = Math.sin(a) * size * 0.8;
            if (i === 0) ctx.moveTo(px, py); else ctx.lineTo(px, py);
        }
        ctx.closePath();
        ctx.fill();
        
        // Core
        ctx.fillStyle = flash ? '#fff' : '#ff2200';
        ctx.beginPath();
        ctx.arc(0, 0, size * 0.3, 0, Math.PI * 2);
        ctx.fill();
    } else if (enemy.type === 'fast') {
        // Arrow/triangle
        ctx.beginPath();
        ctx.moveTo(0, -size);
        ctx.lineTo(size * 0.6, size * 0.5);
        ctx.lineTo(-size * 0.6, size * 0.5);
        ctx.closePath();
        ctx.fill();
    }
    
    // Health bar
    if (enemy.health < enemy.maxHealth) {
        let bw = size;
        ctx.fillStyle = 'rgba(0,0,0,0.5)';
        ctx.fillRect(-bw/2, -size * 1.4, bw, 3);
        ctx.fillStyle = '#0f0';
        ctx.fillRect(-bw/2, -size * 1.4, bw * (enemy.health / enemy.maxHealth), 3);
    }
    
    ctx.restore();
}

    ctx.fillText(`HP: ${Math.ceil(health)}`, hbX + 5, hbY - 8);
    
    // Ammo
    ctx.font = 'bold 24px monospace';
    ctx.fillStyle = reloading ? '#ff0' : '#fff';
    let ammoText = reloading ? 'RELOADING...' : `${ammo} / ${maxAmmo}`;
    ctx.fillText(ammoText, W - 180, H - 35);
    
    // Score & Wave
    ctx.font = 'bold 18px monospace';
    ctx.fillStyle = '#0ff';
    ctx.textAlign = 'center';
    ctx.fillText(`WAVE ${wave}`, W / 2, 40);
    ctx.fillStyle = '#fff';
    ctx.fillText(`SCORE: ${score}`, W / 2, 65);
    ctx.textAlign = 'left';
    
    // Enemy count
    let aliveCount = enemies.filter(e => e.alive).length;
    if (aliveCount > 0) {
        ctx.fillStyle = '#f80';
        ctx.font = '14px monospace';
        ctx.fillText(`Enemies: ${aliveCount}`, W / 2, 85);
    }
}

function renderStartScreen() {
    // Background effect
    let t = Date.now() * 0.001;
    for (let i = 0; i < 50; i++) {
        let x = (Math.sin(t + i * 0.7) * 0.5 + 0.5) * W;
        let y = (Math.cos(t * 0.7 + i * 1.3) * 0.5 + 0.5) * H;
        ctx.fillStyle = `rgba(0, 255, 150, ${0.1 + Math.sin(t + i) * 0.05})`;
        ctx.beginPath();
        ctx.arc(x, y, 2, 0, Math.PI * 2);
        ctx.fill();
    }
    
    // Title
    ctx.textAlign = 'center';
    ctx.font = 'bold 64px monospace';
    ctx.fillStyle = '#0ff';
    ctx.shadowColor = '#0ff';
    ctx.shadowBlur = 20;
    ctx.fillText('NEON STRIKE', W / 2, H * 0.3);
    ctx.shadowBlur = 0;
    
    ctx.font = '20px monospace';
    ctx.fillStyle = '#fff';
    ctx.fillText('A 3D Arena Shooter', W / 2, H * 0.4);
    
    // Controls
    ctx.font = '16px monospace';
    ctx.fillStyle = '#aaa';
    let y = H * 0.52;
    let lines = [
        'WASD - Move',
        'MOUSE - Aim',
        'CLICK - Shoot',
        'R - Reload'
    ];
    lines.forEach((line, i) => {
        ctx.fillText(line, W / 2, y + i * 25);
    });
    
    // Start prompt
    let blink = Math.sin(t * 3) > 0;
    if (blink) {
        ctx.font = 'bold 24px monospace';
        ctx.fillStyle = '#0f0';
        ctx.fillText('PRESS ENTER TO START', W / 2, H * 0.8);
    }
    
    ctx.textAlign = 'left';
}

function renderGameOver() {
    // Dark overlay
    ctx.fillStyle = 'rgba(0, 0, 0, 0.7)';
    ctx.fillRect(0, 0, W, H);
    
    ctx.textAlign = 'center';
    ctx.font = 'bold 56px monospace';
    ctx.fillStyle = '#f00';
    ctx.shadowColor = '#f00';
    ctx.shadowBlur = 15;
    ctx.fillText('GAME OVER', W / 2, H * 0.35);
    ctx.shadowBlur = 0;
    
    ctx.font = '24px monospace';
    ctx.fillStyle = '#fff';
    ctx.fillText(`Final Score: ${score}`, W / 2, H * 0.48);
    ctx.fillText(`Waves Survived: ${wave}`, W / 2, H * 0.55);
    
    let t = Date.now() * 0.001;
    if (Math.sin(t * 3) > 0) {
        ctx.font = '20px monospace';
        ctx.fillStyle = '#0f0';
        ctx.fillText('PRESS R OR ENTER TO RESTART', W / 2, H * 0.7);
    }
    ctx.textAlign = 'left';
}

// Main loop
function gameLoop(timestamp) {
    let dt = Math.min((timestamp - lastTime) / 1000, 0.05);
    lastTime = timestamp;
    
    update(dt);
    render();
    
    requestAnimationFrame(gameLoop);
}

requestAnimationFrame(gameLoop);
