let audioCtx = null;
function initAudio(){
  if(!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();

function playSound(type){
  if(!audioCtx) return;
  const osc = audioCtx.createOscillator();
  const gain = audioCtx.createGain();
  osc.connect(gain);
  gain.connect(audioCtx.destination);
  if(type === 'swap'){
    osc.type = 'sine'; osc.frequency.setValueAtTime(400, audioCtx.currentTime);
    osc.frequency.linearRampToValueAtTime(600, audioCtx.currentTime + 0.1);
    gain.gain.setValueAtTime(0.2, audioCtx.currentTime);
    gain.gain.linearRampToValueAtTime(0, audioCtx.currentTime + 0.15);
    osc.start(); osc.stop(audioCtx.currentTime + 0.15);
  } else if(type === 'match'){
    osc.type = 'square'; osc.frequency.setValueAtTime(523, audioCtx.currentTime);
    osc.frequency.linearRampToValueAtTime(784, audioCtx.currentTime + 0.1);
    gain.gain.setValueAtTime(0.15, audioCtx.currentTime);
    gain.gain.linearRampToValueAtTime(0, audioCtx.currentTime + 0.2);
    osc.start(); osc.stop(audioCtx.currentTime + 0.2);
  } else if(type === 'cascade'){
    osc.type = 'triangle'; osc.frequency.setValueAtTime(659, audioCtx.currentTime);
    osc.frequency.linearRampToValueAtTime(1047, audioCtx.currentTime + 0.15);
    gain.gain.setValueAtTime(0.2, audioCtx.currentTime);
    gain.gain.linearRampToValueAtTime(0, audioCtx.currentTime + 0.25);
    osc.start(); osc.stop(audioCtx.currentTime + 0.25);
  } else if(type === 'win'){
    [523,659,784,1047].forEach((f,i)=>{
      const o = audioCtx.createOscillator(); const g = audioCtx.createGain();
      o.connect(g); g.connect(audioCtx.destination);
      o.type='square'; o.frequency.value=f;
      g.gain.setValueAtTime(0.15, audioCtx.currentTime + i*0.15);
      g.gain.linearRampToValueAtTime(0, audioCtx.currentTime + i*0.15 + 0.2);
      o.start(audioCtx.currentTime + i*0.15); o.stop(audioCtx.currentTime + i*0.15 + 0.2);
    });
  } else if(type === 'lose'){
    osc.type='sawtooth'; osc.frequency.setValueAtTime(300, audioCtx.currentTime);
    osc.frequency.linearRampToValueAtTime(150, audioCtx.currentTime+0.5);

// Game constants
const GRID_SIZE = 8;
const TILE_TYPES = 5;
const RACE_DISTANCE = 100;

// Tile types: colors and shapes (racing themed)
const TILE_INFO = [
  { color: '#ff4757', shape: 'car', name: 'Car' },
  { color: '#2ed573', shape: 'flag', name: 'Flag' },
  { color: '#1e90ff', shape: 'wrench', name: 'Wrench' },
  { color: '#ffa502', shape: 'bolt', name: 'Bolt' },
  { color: '#a29bfe', shape: 'shield', name: 'Shield' }
];

// Game state
let gameState = 'start'; // start, playing, won, lost
let grid = [];
let selectedTile = null;
let animating = false;
let playerProgress = 0;
let aiProgress = [0, 0, 0];
let aiSpeeds = [0.35, 0.45, 0.55];
let cascadeCount = 0;
let particles = [];
let raceTime = 0;
let matchScore = 0;
let comboText = [];

function initGrid(){
  grid = [];
  for(let r=0;r<GRID_SIZE;r++){
    grid[r]=[];
    for(let c=0;c<GRID_SIZE;c++){
      grid[r][c] = Math.floor(Math.random()*TILE_TYPES);
      // Avoid initial matches
      while(checkMatchAt(r,c)){
        grid[r][c] = Math.floor(Math.random()*TILE_TYPES);
      }
    }
  }
}

function checkMatchAt(r,c){
  const t = grid[r][c];
  if(c>=2 && grid[r][c-1]===t && grid[r][c-2]===t) return true;
  if(r>=2 && grid[r-1][c]===t && grid[r-2][c]===t) return true;
  return false;
}

function findMatches(){
  let matches = [];
  // Horizontal
  for(let r=0;r<GRID_SIZE;r++){
    for(let c=0;c<GRID_SIZE-2;c++){
      if(grid[r][c]!==-1 && grid[r][c]===grid[r][c+1] && grid[r][c]===grid[r][c+2]){
        let len=3;
        while(c+len<GRID_SIZE && grid[r][c+len]===grid[r][c]) len++;
        for(let i=0;i<len;i++) matches.push({r:r,c:c+i});
        c += len-1;
      }
    }
  }
  // Vertical
  for(let c=0;c<GRID_SIZE;c++){
    for(let r=0;r<GRID_SIZE-2;r++){
      if(grid[r][c]!==-1 && grid[r][c]===grid[r+1][c] && grid[r][c]===grid[r+2][c]){
        let len=3;
        while(r+len<GRID_SIZE && grid[r+len][c]===grid[r][c]) len++;
        for(let i=0;i<len;i++) matches.push({r:r+i,c:c});
        r += len-1;
      }
    }
  }
  // Deduplicate
  const seen = new Set();
  return matches.filter(m=>{const k=m.r*GRID_SIZE+m.c; if(seen.has(k))return false; seen.add(k); return true;});
}

function removeMatches(matches){
  for(let m of matches) grid[m.r][m.c] = -1;
}

function dropTiles(){
  let dropped = false;
  for(let c=0;c<GRID_SIZE;c++){
    let writeRow = GRID_SIZE-1;
    for(let r=GRID_SIZE-1;r>=0;r--){
      if(grid[r][c]!==-1){
        grid[writeRow][c]=grid[r][c];
        if(writeRow!==r) dropped=true;
        writeRow--;
      }
    }
    for(let r=writeRow;r>=0;r--){
      grid[r][c]=Math.floor(Math.random()*TILE_TYPES);
      dropped=true;
    }
  }
  return dropped;
}

function spawnParticles(r,c,color,count){
  const cellSize = getCellSize();
  const cx = getGridX() + (c+0.5)*cellSize;
  const cy = getGridY() + (r+0.5)*cellSize;
  for(let i=0;i<count;i++){
    particles.push({
      x:cx, y:cy,
      vx:(Math.random()-0.5)*4,
      vy:(Math.random()-0.5)*4 - 2,
      life:1,
      color:color,
      size: Math.random()*4+2
    });
  }
}

function getGridX(){ return W/2 - (GRID_SIZE*getCellSize())/2; }
function getGridY(){ return getTrackHeight() + 20; }
function getCellSize(){ return Math.min((W-40)/GRID_SIZE, (H-getTrackHeight()-80)/GRID_SIZE); }
function getTrackHeight(){ return Math.max(60, H*0.15); }

function processBoard(){
  let matches = findMatches();
  if(matches.length===0){
    animating=false;
    cascadeCount=0;
    return;
  }
  cascadeCount++;
  const pts = matches.length * cascadeCount * 2;
  playerProgress = Math.min(RACE_DISTANCE, playerProgress + pts);
  matchScore += pts;
  
  if(cascadeCount > 1){
    playSound('cascade');
    comboText.push({text: 'x'+cascadeCount+' COMBO!', life:1});
  } else {
    playSound('match');
  }
  
  for(let m of matches) spawnParticles(m.r, m.c, TILE_INFO[grid[m.r][m.c]]?.color || '#fff', 4);
  removeMatches(matches);
  
  // Add combo text
  if(cascadeCount>1){
    const cellSize=getCellSize();
    comboText.push({text: cascadeCount+'x COMBO!', life:1, x:W/2, y:getGridY()+cellSize*3});
  }
  
  setTimeout(()=>{ dropTiles(); processBoard(); }, 200);
}

function trySwap(r1,c1,r2,c2){
  if(animating) return;
  initAudio();
  const temp = grid[r1][c1];
  grid[r1][c1] = grid[r2][c2];
  grid[r2][c2] = temp;
  
  const matches = findMatches();
  if(matches.length===0){
    // Swap back
    grid[r2][c2] = grid[r1][c1];
    grid[r1][c1] = temp;
    playSound('invalid');
  } else {
    playSound('swap');
    animating=true;
    cascadeCount=0;
    processBoard();
  }
}

function canSwap(r1,c1,r2,c2){
  const dr=Math.abs(r1-r2), dc=Math.abs(c1-c2);
  return (dr+dc===1) && r1>=0 && r1<GRID_SIZE && c1>=0 && c1<GRID_SIZE;
}

function getGridPos(x,y){
  const cellSize=getCellSize();
  const gx=getGridX(), gy=getGridY();
  const c=Math.floor((x-gx)/cellSize);
  const r=Math.floor((y-gy)/cellSize);
  if(r>=0&&r<GRID_SIZE&&c>=0&&c<GRID_SIZE) return {r,c};
  return null;
}

function handleInput(x,y){
  if(gameState!=='playing') return;
  const pos=getGridPos(x,y);
  if(!pos) return;
  
  if(selectedTile===null){
    selectedTile=pos;
  } else {
    if(pos.r===selectedTile.r && pos.c===selectedTile.c){
      selectedTile=null;
    } else if(canSwap(selectedTile.r,selectedTile.c,pos.r,pos.c)){
      trySwap(selectedTile.r,selectedTile.c,pos.r,pos.c);
      selectedTile=null;
    } else {
      playSound('invalid');
      selectedTile=pos;
    }
  }
}

    startGame();
  } else if(gameState==='won'||gameState==='lost'){
    startGame();
  }
}

function startGame(){
  gameState='playing';
  playerProgress=0;
  aiProgress=[0,0,0];
  raceTime=0;
  matchScore=0;
  cascadeCount=0;
  animating=false;
  selectedTile=null;
  particles=[];
  comboText=[];
  initGrid();
}

function update(dt){
  if(gameState!=='playing') return;
  
  raceTime += dt;
  
  // AI progress
  for(let i=0;i<3;i++){
    aiProgress[i] = Math.min(RACE_DISTANCE, aiProgress[i] + aiSpeeds[i]*dt);
  }
  
  // Check win/lose
  if(playerProgress>=RACE_DISTANCE){ gameState='won'; playSound('win'); }
  else if(aiProgress[0]>=RACE_DISTANCE || aiProgress[1]>=RACE_DISTANCE || aiProgress[2]>=RACE_DISTANCE){
    gameState='lost'; playSound('lose');
  }
  
  // Particles
  for(let p of particles){
    p.x+=p.vx; p.y+=p.vy;
    p.vy+=0.15;
    p.life-=dt*2;
  }
  particles=particles.filter(p=>p.life>0);
  
  // Combo text
  for(let ct of comboText) ct.life-=dt*1.5;
  comboText=comboText.filter(ct=>ct.life>0);
}

function drawTile(x,y,type,size){
  if(type===undefined||type<0||type>=TILE_TYPES) return;
  const info=TILE_INFO[type];
  const s=size*0.35;
  
  ctx.save();
  ctx.translate(x,y);
  
  // Background circle
  ctx.beginPath();
  ctx.arc(0,0,s+2,0,Math.PI*2);
  ctx.fillStyle=info.color+'33';
  ctx.fill();
  
  // Shape
  ctx.fillStyle=info.color;
  ctx.strokeStyle=info.color;
  ctx.lineWidth=2;
  
  switch(info.shape){
    case 'car':
      // Car body
      ctx.beginPath();
      ctx.roundRect(-s*0.8,-s*0.4,s*1.6,s*0.8,3);
      ctx.fill();
      // Windshield
      ctx.fillStyle='#1a1a2e';
      ctx.fillRect(-s*0.3,-s*0.25,s*0.6,s*0.3);
      // Wheels
      ctx.fillStyle=info.color;
      ctx.beginPath(); ctx.arc(-s*0.5,s*0.4,s*0.18,0,Math.PI*2); ctx.fill();
      ctx.beginPath(); ctx.arc(s*0.5,s*0.4,s*0.18,0,Math.PI*2); ctx.fill();
      break;
    case 'flag':
      // Pole
      ctx.fillRect(-s*0.1,-s*0.8,s*0.2,s*1.6);
      // Flag
      ctx.beginPath();
      ctx.moveTo(s*0.1,-s*0.7);
      ctx.lineTo(s*0.7,-s*0.4);
      ctx.lineTo(s*0.1,-s*0.1);
      ctx.closePath();
      ctx.fill();
      break;
    case 'wrench':
      // Wrench shape
      ctx.beginPath();
      ctx.arc(-s*0.3,-s*0.4,s*0.25,0,Math.PI*2);
      ctx.fill();
      ctx.fillRect(-s*0.1,-s*0.3,s*0.6,s*0.8);
      ctx.beginPath();
      ctx.arc(s*0.3,s*0.4,s*0.25,0,Math.PI*2);
      ctx.fill();
      break;
    case 'bolt':
      // Hex bolt
      ctx.beginPath();
      for(let i=0;i<6;i++){
        const a=i*Math.PI/3 - Math.PI/6;
        const px=Math.cos(a)*s*0.7, py=Math.sin(a)*s*0.7;
        if(i===0) ctx.moveTo(px,py); else ctx.lineTo(px,py);
      }
      ctx.closePath();
      ctx.fill();
      // Center hole
      ctx.fillStyle='#1a1a2e';
      ctx.beginPath(); ctx.arc(0,0,s*0.2,0,Math.PI*2); ctx.fill();
      break;
    case 'shield':
      // Shield shape
      ctx.beginPath();
      ctx.moveTo(0,-s*0.7);
      ctx.lineTo(s*0.5,-s*0.3);
      ctx.lineTo(s*0.4,s*0.5);
      ctx.lineTo(0,s*0.8);
      ctx.lineTo(-s*0.4,s*0.5);
      ctx.lineTo(-s*0.5,-s*0.3);
      ctx.closePath();
      ctx.fill();
      // Star
      ctx.fillStyle='#1a1a2e';
      ctx.beginPath();
      for(let i=0;i<5;i++){
        const a=i*4*Math.PI/5-Math.PI/2;
        const px=Math.cos(a)*s*0.2, py=Math.sin(a)*s*0.2;
        if(i===0) ctx.moveTo(px,py); else ctx.lineTo(px,py);
      }
      ctx.closePath();
      ctx.fill();
      break;
  }
  
  ctx.restore();
}

function drawTrack(){
  const th=getTrackHeight();
  const trackY=10;
  const trackH=th-20;
  const trackX=20;
  const trackW=W-40;
  
  // Track background
  ctx.fillStyle='#2d2d44';
  ctx.beginPath();
  ctx.roundRect(trackX,trackY,trackW,trackH,8);
  ctx.fill();
  
  // Track lanes
  const laneH=trackH/3.5;
  for(let i=0;i<3;i++){
    const ly=trackY+10+i*laneH;
    ctx.fillStyle='#1a1a2e';
    ctx.fillRect(trackX+10,ly,trackW-20,laneH-4);
  }
  
  // Player lane (top)
  const playerLaneY=trackY+10+laneH/2;
  const aiLane1Y=trackY+10+laneH*1.5;
  const aiLane2Y=trackY+10+laneH*2.5;
  
  // Progress markers
  for(let i=0;i<10;i++){
    const mx=trackX+10+(trackW-40)*(i/9);
    ctx.fillStyle='#ffffff33';
    ctx.fillRect(mx,trackY+5,2,trackH-10);
  }
  
  // Finish line
  const finishX=trackX+trackW-30;
  for(let i=0;i<Math.floor(trackH/8);i++){
    for(let j=0;j<2;j++){
      ctx.fillStyle=((i+j)%2===0)?'#fff':'#000';
      ctx.fillRect(finishX+j*4,trackY+5+i*8,4,8);
    }
  }
  
  // Draw cars
  function drawCar(x,y,color,scale){
    ctx.save();
    ctx.translate(x,y);
    ctx.scale(scale,scale);
    ctx.fillStyle=color;
    ctx.beginPath();
    ctx.roundRect(-12,-6,24,12,3);
    ctx.fill();
    ctx.fillStyle='#fff';
    ctx.fillRect(-8,-3,8,6);
    ctx.restore();
  }
  
  // Player car
  const px=trackX+10+(trackW-50)*(playerProgress/RACE_DISTANCE);
  drawCar(px, playerLaneY, '#ff4757', 1.2);
  
  // AI cars
  const aiColors=['#2ed573','#1e90ff','#ffa502'];
  for(let i=0;i<3;i++){
    const ax=trackX+10+(trackW-50)*(aiProgress[i]/RACE_DISTANCE);
    drawCar(ax, aiLane1Y+i*(laneH), aiColors[i], 1);
  }
  
  // Labels
  ctx.fillStyle='#fff';
  ctx.font='bold 10px sans-serif';
  ctx.textAlign='left';
  ctx.fillText('YOU',trackX+5,playerLaneY+3);
  ctx.font='9px sans-serif';
  ctx.fillText('AI 1',trackX+5,aiLane1Y+3);
  ctx.fillText('AI 2',trackX+5,aiLane2Y-laneH/2+3);
}

function drawGrid(){
  const cellSize=getCellSize();
  const gx=getGridX(), gy=getGridY();
  
  // Grid background
  ctx.fillStyle='#16213e';
  ctx.beginPath();
  ctx.roundRect(gx-4,gy-4,cellSize*GRID_SIZE+8,cellSize*GRID_SIZE+8,8);
  ctx.fill();
  
  // Cells
  for(let r=0;r<GRID_SIZE;r++){
    for(let c=0;c<GRID_SIZE;c++){
      const x=gx+c*cellSize, y=gy+r*cellSize;
      
      // Cell background
      ctx.fillStyle=(r+c)%2===0?'#1a1a3e':'#16213e';
      ctx.fillRect(x+1,y+1,cellSize-2,cellSize-2);
      
      // Tile
      if(grid[r]&&grid[r][c]!==-1){
        drawTile(x+cellSize/2, y+cellSize/2, grid[r][c], cellSize);
      }
    }
  }
  
  // Selection highlight
  if(selectedTile && gameState==='playing'){
    const sx=gx+selectedTile.c*cellSize;
    const sy=gy+selectedTile.r*cellSize;
    ctx.strokeStyle='#fff';
    ctx.lineWidth=3;
    ctx.strokeRect(sx+2,sy+2,cellSize-4,cellSize-4);
  }
  
  // Particles
  for(let p of particles){
    ctx.globalAlpha=p.life;
    ctx.fillStyle=p.color;
    ctx.beginPath();
    ctx.arc(p.x,p.y,p.size,0,Math.PI*2);
    ctx.fill();
  }
  ctx.globalAlpha=1;
  
  // Combo text
  for(let ct of comboText){
    ctx.globalAlpha=ct.life;
    ctx.fillStyle='#fff';
    ctx.font='bold '+Math.min(36,cellSize*0.8)+'px sans-serif';
    ctx.textAlign='center';
    const tx=ct.x||W/2;
    const ty=(ct.y||gy+cellSize*4) - (1-ct.life)*30;
    ctx.fillText(ct.text,tx,ty);
  }
  ctx.globalAlpha=1;
}

function drawHUD(){
  // Score

let lastTime=performance.now();
function gameLoop(now){
  const dt=(now-lastTime)/1000;
  lastTime=now;
  
  ctx.clearRect(0,0,W,H);
  
  if(gameState==='start'){
    drawStartScreen();
  } else {
    update(dt);
    
    // Background
    ctx.fillStyle='#0f0f23';
    ctx.fillRect(0,0,W,H);
    
    // Speed lines when cascading
    if(animating && cascadeCount>2){
      ctx.strokeStyle='#ffffff15';
      ctx.lineWidth=1;
      const t=performance.now()/100;
      for(let i=0;i<10;i++){
        const x=(t*3+i*80)%(W+100)-50;
        ctx.beginPath();
        ctx.moveTo(x,0);
        ctx.lineTo(x-20,H);
        ctx.stroke();
      }
    }
    
    drawTrack();
    drawGrid();
    drawHUD();
    
    if(gameState==='won') drawEndScreen(true);
    else if(gameState==='lost') drawEndScreen(false);
  }
  
  requestAnimationFrame(gameLoop);
}

// Polyfill roundRect if needed
if(!CanvasRenderingContext2D.prototype.roundRect){
  CanvasRenderingContext2D.prototype.roundRect = function(x,y,w,h,r){
    if(typeof r==='number') r=[r,r,r,r];
    const [tl,tr,br,bl]=Array.isArray(r)?r:[r,r,r,r];
    this.moveTo(x+tl,y);
    this.lineTo(x+w-tr,y);
    this.arcTo(x+w,y,x+w,y+tr,tr);
    this.lineTo(x+w,y+h-br);
    this.arcTo(x+w,y+h,x+w-br,y+h,br);
    this.lineTo(x+bl,y+h);
    this.arcTo(x,y+h,x,y+h-bl,bl);
    this.lineTo(x,y+tl);
    this.arcTo(x,y,x+tl,y,tl);
    return this;
  };
}

initGrid();
