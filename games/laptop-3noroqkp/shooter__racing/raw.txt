const canvas = document.getElementById('game');
const ctx = canvas.getContext('2d');

let W, H;
function resize() {
    W = window.innerWidth;
    H = window.innerHeight;
    canvas.width = W;
    canvas.height = H;
}
window.addEventListener('resize', resize);
resize();

// Audio
let audioCtx = null;
function initAudio() {
    if (!audioCtx) audioCtx = new (window.AudioContext || webkitAudioContext)();
}
function playSound(type) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    switch(type) {
        case 'shoot':
            osc.type = 'square'; osc.frequency.setValueAtTime(400, audioCtx.currentTime);
            osc.frequency.exponentialRampToValueAtTime(200, audioCtx.currentTime + 0.1);
            gain.gain.setValueAtTime(0.15, audioCtx.currentTime);
            gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.1);
            osc.start(); osc.stop(audioCtx.currentTime + 0.1); break;
        case 'hit':
            osc.type = 'sawtooth'; osc.frequency.setValueAtTime(150, audioCtx.currentTime);
            osc.frequency.exponentialRampToValueAtTime(50, audioCtx.currentTime + 0.2);
            gain.gain.setValueAtTime(0.3, audioCtx.currentTime);
            gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.2);
            osc.start(); osc.stop(audioCtx.currentTime + 0.2); break;
        case 'explode':
            osc.type = 'sawtooth'; osc.frequency.setValueAtTime(100, audioCtx.currentTime);
            osc.frequency.exponentialRampToValueAtTime(30, audioCtx.currentTime + 0.4);
            gain.gain.setValueAtTime(0.4, audioCtx.currentTime);
            gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.4);
            osc.start(); osc.stop(audioCtx.currentTime + 0.4); break;
        case 'powerup':
            osc.type = 'sine'; osc.frequency.setValueAtTime(600, audioCtx.currentTime);
            osc.frequency.exponentialRampToValueAtTime(1200, audioCtx.currentTime + 0.15);
            gain.gain.setValueAtTime(0.2, audioCtx.currentTime);
            gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.15);
            osc.start(); osc.stop(audioCtx.currentTime + 0.15); break;
        case 'gameover':
            osc.type = 'sawtooth'; osc.frequency.setValueAtTime(300, audioCtx.currentTime);
            osc.frequency.exponentialRampToValueAtTime(50, audioCtx.currentTime + 0.8);
            gain.gain.setValueAtTime(0.4, audioCtx.currentTime);
            gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.8);
            osc.start(); osc.stop(audioCtx.currentTime + 0.8); break;
    }
}

// Game State
const STATE = { MENU: 0, PLAYING: 1, GAMEOVER: 2 };
let state = STATE.MENU;
let score = 0, highScore = 0;
try { highScore = parseInt(localStorage.getItem('turboStrikeHS') || '0'); } catch(e) {}

let player, bullets, enemies, particles, powerups, roadMarkers;
let gameTime, spawnTimer, difficulty, speedBoost, shieldTimer, spreadTimer;

function resetGame() {
    score = 0; gameTime = 0; spawnTimer = 0; difficulty = 1;
    speedBoost = 0; shieldTimer = 0; spreadTimer = 0;
    player = { x: W/2, y: H - 80, w: 36, h: 50, speed: 0, maxSpeed: 8, hp: 100, invincible: 0 };
    bullets = []; enemies = []; particles = []; powerups = [];
    roadMarkers = [];
    for (let i = 0; i < 20; i++) roadMarkers.push({ x: Math.random() * W, y: Math.random() * H });
}

// Input
const keys = {};
window.addEventListener('keydown', e => {
    keys[e.code] = true;
    if (state === STATE.MENU && (e.code === 'Space' || e.code === 'Enter')) {
        initAudio(); resetGame(); state = STATE.PLAYING;
    } else if (state === STATE.GAMEOVER && (e.code === 'Space' || e.code === 'Enter' || e.code === 'KeyR')) {
        initAudio(); resetGame(); state = STATE.PLAYING;
    }
});
window.addEventListener('keyup', e => { keys[e.code] = false; });

// Spawning
function spawnEnemy() {
    const types = ['car', 'tank', 'fast'];
    const type = types[Math.floor(Math.random() * Math.min(types.length, difficulty + 1))];
    let ex, ey, ew, eh, evy, ehp;
    switch(type) {
        case 'car':
            ew = 34; eh = 46; ehp = 1; evy = 2 + Math.random() * 2;
            break;
        case 'tank':
            ew = 40; eh = 40; ehp = 3; evy = 1 + Math.random();
            break;
        case 'fast':
            ew = 28; eh = 36; ehp = 1; evy = 4 + Math.random() * 2;
            break;
    }
    ex = Math.random() * (W - ew); ey = -eh;
    enemies.push({ x: ex, y: ey, w: ew, h: eh, hp: ehp, vy: evy, type, shootTimer: Math.random() * 100 });
}

function spawnPowerup(x, y) {
    const types = ['shield', 'spread', 'heal'];
    const type = types[Math.floor(Math.random() * types.length)];
    powerups.push({ x, y, w: 24, h: 24, type, vy: 1.5 });
}

function addParticles(x, y, color, count) {
    for (let i = 0; i < count; i++) {
        const angle = Math.random() * Math.PI * 2;
        const speed = 1 + Math.random() * 4;
        particles.push({ x, y, vx: Math.cos(angle) * speed, vy: Math.sin(angle) * speed, life: 30 + Math.random() * 30, color });
    }
}

function drawCar(x, y, w, h, color, isPlayer) {
    ctx.fillStyle = color;
    ctx.fillRect(x - w/2, y - h/2, w, h);
    // Windshield
    ctx.fillStyle = isPlayer ? '#4af' : '#f44';
    ctx.fillRect(x - w/4, y - h/4, w/2, h/5);
    // Wheels
    ctx.fillStyle = '#222';
    ctx.fillRect(x - w/2 - 3, y - h/3, 5, h/4);
    ctx.fillRect(x + w/2 - 2, y - h/3, 5, h/4);
    ctx.fillRect(x - w/2 - 3, y + h/6, 5, h/4);
    ctx.fillRect(x + w/2 - 2, y + h/6, 5, h/4);
    // Taillights
    if (isPlayer) {
        ctx.fillStyle = '#f00';
        ctx.fillRect(x - w/3, y + h/2 - 4, w/6, 4);
        ctx.fillRect(x + w/6, y + h/2 - 4, w/6, 4);
    }
}

function drawTank(x, y, w, h) {
    ctx.fillStyle = '#5a7';
    ctx.fillRect(x - w/2, y - h/2, w, h);
    ctx.fillStyle = '#384';
    ctx.fillRect(x - w/4, y - h/4, w/2, h/2);
    ctx.fillStyle = '#fff';
    ctx.font = '10px monospace';
    ctx.textAlign = 'center';
    ctx.fillText('T', x, y + 3);
}

function drawFastCar(x, y, w, h) {
    ctx.fillStyle = '#f80';
    ctx.beginPath();
    ctx.moveTo(x, y - h/2);
    ctx.lineTo(x - w/2, y + h/2);
    ctx.lineTo(x + w/2, y + h/2);
    ctx.closePath();
    ctx.fill();
}

function update() {
    if (state !== STATE.PLAYING) return;
    gameTime++;
    difficulty = 1 + Math.floor(gameTime / 500);

    // Player movement
    let dx = 0, dy = 0;
    if (keys['ArrowLeft'] || keys['KeyA']) dx = -player.maxSpeed;
    if (keys['ArrowRight'] || keys['KeyD']) dx = player.maxSpeed;
    if (keys['ArrowUp'] || keys['KeyW']) dy = -player.maxSpeed;
    if (keys['ArrowDown'] || keys['KeyS']) dy = player.maxSpeed;

    if (speedBoost > 0) { dx *= 1.5; dy *= 1.5; speedBoost--; }

    player.x += dx; player.y += dy;
    player.x = Math.max(player.w/2, Math.min(W - player.w/2, player.x));
    player.y = Math.max(H * 0.3, Math.min(H - player.h/2, player.y));

    if (player.invincible > 0) player.invincible--;
    if (shieldTimer > 0) shieldTimer--;
    if (spreadTimer > 0) spreadTimer--;

    // Shooting
    if ((keys['Space'] || keys['KeyZ']) && gameTime % 8 === 0) {
        playSound('shoot');
        if (spreadTimer > 0) {
            bullets.push({ x: player.x, y: player.y - 25, vx: -1.5, vy: -10, w: 4, h: 10 });
            bullets.push({ x: player.x, y: player.y - 25, vx: 0, vy: -10, w: 4, h: 10 });
            bullets.push({ x: player.x, y: player.y - 25, vx: 1.5, vy: -10, w: 4, h: 10 });
        } else {
            bullets.push({ x: player.x, y: player.y - 25, vx: 0, vy: -10, w: 4, h: 10 });
        }
    }

    // Update bullets
    for (let i = bullets.length - 1; i >= 0; i--) {
        const b = bullets[i];
        b.x += b.vx; b.y += b.vy;
        if (b.y < -20 || b.x < -20 || b.x > W + 20) bullets.splice(i, 1);
    }

    // Spawn enemies
    spawnTimer++;
    const spawnRate = Math.max(30, 80 - difficulty * 5);
    if (spawnTimer >= spawnRate) {
        spawnTimer = 0;
        spawnEnemy();
    }

    // Update enemies
    for (let i = enemies.length - 1; i >= 0; i--) {
        const e = enemies[i];
        e.y += e.vy;
        if (e.type === 'tank' && gameTime % 90 === 0) {
            playSound('shoot');
            bullets.push({ x: e.x, y: e.y + e.h/2, vx: 0, vy: 5, w: 4, h: 8, enemy: true });
        }
        if (e.y > H + 50) enemies.splice(i, 1);
    }

    // Bullet-enemy collision
    for (let i = bullets.length - 1; i >= 0; i--) {
        const b = bullets[i];
        if (b.enemy) continue;
        for (let j = enemies.length - 1; j >= 0; j--) {
            const e = enemies[j];
            if (Math.abs(b.x - e.x) < (b.w + e.w)/2 && Math.abs(b.y - e.y) < (b.h + e.h)/2) {
                bullets.splice(i, 1);
                e.hp--;
                playSound('hit');
                addParticles(e.x, e.y, '#ff0', 5);
                if (e.hp <= 0) {
                    score += e.type === 'tank' ? 50 : e.type === 'fast' ? 30 : 20;
                    playSound('explode');
                    addParticles(e.x, e.y, '#f80', 15);
                    if (Math.random() < 0.2) spawnPowerup(e.x, e.y);
                    enemies.splice(j, 1);
                }
                break;
            }
        }
    }

    // Enemy bullet hit player
    for (let i = bullets.length - 1; i >= 0; i--) {
        const b = bullets[i];
        if (!b.enemy) continue;
        if (Math.abs(b.x - player.x) < (b.w + player.w)/2 && Math.abs(b.y - player.y) < (b.h + player.h)/2) {
            bullets.splice(i, 1);
            if (player.invincible <= 0 && shieldTimer <= 0) {
                player.hp -= 15;
                playSound('hit');
                addParticles(player.x, player.y, '#f00', 8);
            }
        }
    }

    // Enemy-player collision
    for (let i = enemies.length - 1; i >= 0; i--) {
        const e = enemies[i];
        if (Math.abs(e.x - player.x) < (e.w + player.w)/2 && Math.abs(e.y - player.y) < (e.h + player.h)/2) {
            if (player.invincible <= 0 && shieldTimer <= 0) {
                player.hp -= 25;
                playSound('hit');
                addParticles(player.x, player.y, '#f00', 10);
                player.invincible = 60;
            } else if (player.invincible > 0 || shieldTimer > 0) {
                // Destroy enemy
                enemies.splice(i, 1);
                addParticles(e.x, e.y, '#ff0', 10);
            } else {
                enemies.splice(i, 1);
                addParticles(e.x, e.y, '#f80', 15);
            }
        }
    }

    // Powerups
    for (let i = powerups.length - 1; i >= 0; i--) {
        const p = powerups[i];
        p.y += p.vy;
        if (p.y > H + 30) { powerups.splice(i, 1); continue; }
        if (Math.abs(p.x - player.x) < 30 && Math.abs(p.y - player.y) < 30) {
            playSound('powerup');
            switch(p.type) {
                case 'shield': shieldTimer = 600; break;
                case 'spread': spreadTimer = 400; break;
                case 'heal': player.hp = Math.min(100, player.hp + 30); break;
            }
            powerups.splice(i, 1);
        }
    }

    // Particles
    for (let i = particles.length - 1; i >= 0; i--) {
        const p = particles[i];
        p.x += p.vx; p.y += p.vy; p.life--;
        if (p.life <= 0) particles.splice(i, 1);
    }

    // Road markers
    const scrollSpeed = speedBoost > 0 ? 6 : 3;
    for (let m of roadMarkers) {
        m.y += scrollSpeed;
        if (m.y > H + 10) { m.y = -10; m.x = Math.random() * W; }
    }

    // Score from speed
    score += speedBoost > 0 ? 2 : 1;

    // Game over
    if (player.hp <= 0) {
        state = STATE.GAMEOVER;
        playSound('gameover');
        if (score > highScore) {
            highScore = score;
            try { localStorage.setItem('turboStrikeHS', String(highScore)); } catch(e) {}
        }
    }
}

function render() {
    ctx.fillStyle = '#1a1a2e';
    ctx.fillRect(0, 0, W, H);

    // Road background
    ctx.fillStyle = '#252540';
    const roadW = Math.min(W * 0.8, 600);
    const roadX = (W - roadW) / 2;
    ctx.fillRect(roadX, 0, roadW, H);

    // Road edges
    ctx.fillStyle = '#444';
    ctx.fillRect(roadX, 0, 4, H);
    ctx.fillRect(roadX + roadW - 4, 0, 4, H);

    // Lane markers
    ctx.fillStyle = '#555';
    for (let m of roadMarkers) {
        if (m.x > roadX && m.x < roadX + roadW) {
            ctx.fillRect(m.x, m.y, 3, 20);
        }
    }

    // Center dashes
    ctx.fillStyle = '#666';
    const dashOffset = (gameTime * 3) % 80;
    for (let y = -80 + dashOffset; y < H; y += 80) {
        ctx.fillRect(W/2 - 1, y, 2, 40);
    }

    if (state === STATE.PLAYING || state === STATE.GAMEOVER) {
        // Powerups
        for (let p of powerups) {
            ctx.fillStyle = p.type === 'shield' ? '#4af' : p.type === 'spread' ? '#f0f' : '#0f0';
            ctx.beginPath();
            ctx.arc(p.x, p.y, 12, 0, Math.PI * 2);
            ctx.fill();
            ctx.fillStyle = '#fff';
            ctx.font = 'bold 10px monospace';
            ctx.textAlign = 'center';
            ctx.fillText(p.type === 'shield' ? 'S' : p.type === 'spread' ? 'W' : '+', p.x, p.y + 4);
        }

        // Enemies
        for (let e of enemies) {
            if (e.type === 'tank') drawTank(e.x, e.y, e.w, e.h);
            else if (e.type === 'fast') drawFastCar(e.x, e.y, e.w, e.h);
            else drawCar(e.x, e.y, e.w, e.h, '#a44', false);
        }

        // Bullets
        for (let b of bullets) {
            ctx.fillStyle = b.enemy ? '#f44' : '#ff0';
            ctx.fillRect(b.x - b.w/2, b.y - b.h/2, b.w, b.h);
        }

        // Player
        if (state === STATE.PLAYING && (player.invincible <= 0 || gameTime % 8 < 4)) {
            drawCar(player.x, player.y, player.w, player.h, '#4af', true);
            if (shieldTimer > 0) {
                ctx.strokeStyle = 'rgba(100,200,255,0.6)';
                ctx.lineWidth = 2;
                ctx.beginPath();
                ctx.arc(player.x, player.y, 35, 0, Math.PI * 2);
                ctx.stroke();
            }
        }

        // Particles
        for (let p of particles) {
            ctx.fillStyle = p.color;
            ctx.globalAlpha = p.life / 60;
            ctx.fillRect(p.x - 3, p.y - 3, 6, 6);
            ctx.globalAlpha = 1;
        }

        // HUD
        ctx.fillStyle = '#fff';
        ctx.font = 'bold 16px monospace';
        ctx.textAlign = 'left';
        ctx.fillText('SCORE: ' + score, 20, 30);
        ctx.fillText('SPEED: ' + (speedBoost > 0 ? 'BOOST!' : 'NORMAL'), 20, 52);

        // Health bar
        ctx.fillStyle = '#333';
        ctx.fillRect(20, H - 40, 150, 16);
        ctx.fillStyle = player.hp > 50 ? '#0f0' : player.hp > 25 ? '#ff0' : '#f00';
        ctx.fillRect(20, H - 40, 150 * (player.hp / 100), 16);
        ctx.strokeStyle = '#fff';
        ctx.strokeRect(20, H - 40, 150, 16);

        // Shield timer
        if (shieldTimer > 0) {
            ctx.fillStyle = '#4af';
            ctx.fillText('SHIELD: ' + Math.ceil(shieldTimer / 60), 20, H - 50);
        }
        if (spreadTimer > 0) {
            ctx.fillStyle = '#f0f';
            ctx.fillText('SPREAD: ' + Math.ceil(spreadTimer / 60), 20, H - 70);
        }

        // Difficulty indicator
        ctx.fillStyle = '#888';
        ctx.textAlign = 'right';
        ctx.font = '12px monospace';
        ctx.fillText('LEVEL: ' + difficulty, W - 20, 30);
    }

    if (state === STATE.MENU) {
        // Title screen overlay
        ctx.fillStyle = 'rgba(10,10,26,0.85)';
        ctx.fillRect(0, 0, W, H);

        ctx.textAlign = 'center';
        ctx.fillStyle = '#4af';
        ctx.font = 'bold 48px monospace';
        ctx.fillText('TURBO STRIKE', W/2, H/3 - 20);

        ctx.fillStyle = '#fff';
        ctx.font = '18px monospace';
        ctx.fillText('A Racing Shooter', W/2, H/3 + 20);

        ctx.font = '14px monospace';
        ctx.fillStyle = '#aaa';
        const lines = [
            'ARROWS / WASD - Drive',
            'SPACE / Z - Shoot',
            '',
            'Dodge enemies. Collect power-ups.',
            'S = Shield | W = Spread | + = Heal',
            '',
            'High Score: ' + highScore,
            '',
            'Press SPACE to Start'
        ];
        lines.forEach((line, i) => {
            ctx.fillText(line, W/2, H/2 + i * 24);
        });

        // Animated car
        const t = Date.now() / 500;
        drawCar(W/2, H - 80, 36, 50, '#4af', true);
    }

    if (state === STATE.GAMEOVER) {
        ctx.fillStyle = 'rgba(10,10,26,0.7)';
        ctx.fillRect(0, 0, W, H);
        ctx.textAlign = 'center';
        ctx.fillStyle = '#f44';
        ctx.font = 'bold 42px monospace';
        ctx.fillText('GAME OVER', W/2, H/3);

        ctx.fillStyle = '#fff';
        ctx.font = '20px monospace';
        ctx.fillText('Score: ' + score, W/2, H/3 + 50);
        ctx.fillText('High Score: ' + highScore, W/2, H/3 + 80);

        ctx.fillStyle = '#aaa';
        ctx.font = '16px monospace';
        ctx.fillText('Press SPACE or R to Restart', W/2, H/2 + 40);
    }
}

function loop() {
    update();
    render();
    requestAnimationFrame(loop);
}

resetGame();
loop();
</script>
