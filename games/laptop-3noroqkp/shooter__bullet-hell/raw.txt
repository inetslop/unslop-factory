// Audio
let audioCtx = null;
function initAudio() {
    if (!audioCtx) {
        audioCtx = new (window.AudioContext || window.webkitAudioContext)();
    }
}

function playLaser() {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = 'square';
    osc.frequency.setValueAtTime(880, audioCtx.currentTime);
    osc.frequency.exponentialRampToValueAtTime(220, audioCtx.currentTime + 0.1);
    gain.gain.setValueAtTime(0.08, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.1);
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + 0.1);
}

function playExplosion() {
    if (!audioCtx) return;
    const bufferSize = audioCtx.sampleRate * 0.3;
    const buffer = audioCtx.createBuffer(1, bufferSize, audioCtx.sampleRate);
    const data = buffer.getChannelData(0);
    for (let i = 0; i < bufferSize; i++) {
        data[i] = (Math.random() * 2 - 1) * Math.pow(1 - i / bufferSize, 2);
    }
    const source = audioCtx.createBufferSource();
    source.buffer = buffer;
    const gain = audioCtx.createGain();
    gain.gain.setValueAtTime(0.3, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.3);
    source.connect(gain);
    gain.connect(audioCtx.destination);
    source.start();
}

function playPlayerHit() {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = 'sawtooth';
    osc.frequency.setValueAtTime(200, audioCtx.currentTime);
    osc.frequency.exponentialRampToValueAtTime(50, audioCtx.currentTime + 0.4);
    gain.gain.setValueAtTime(0.3, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.4);
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + 0.4);
}

function updatePlayer(dt) {
    const focus = keys['ShiftLeft'] || keys['ShiftRight'];
    const spd = focus ? player.focusSpeed : player.speed;
    
    if (keys['KeyW'] || keys['ArrowUp']) player.y -= spd;
    if (keys['KeyS'] || keys['ArrowDown']) player.y += spd;
    if (keys['KeyA'] || keys['ArrowLeft']) player.x -= spd;
    if (keys['KeyD'] || keys['ArrowRight']) player.x += spd;
    
    player.x = Math.max(15, Math.min(W - 15, player.x));
    player.y = Math.max(20, Math.min(H - 20, player.y));
    
    if (player.invincible > 0) player.invincible -= dt;
    
    // Auto shoot
    player.shootTimer -= dt;
    if (player.shootTimer <= 0) {
        player.shootTimer = focus ? 18 : 12;
        const bx = player.x;
        const by = player.y - 20;
        
        if (player.power >= 3) {
            playerBullets.push({ x: bx, y: by, vx: 0, vy: -12, r: 4 });
            playerBullets.push({ x: bx - 8, y: by + 5, vx: -1, vy: -12, r: 3 });
            playerBullets.push({ x: bx + 8, y: by + 5, vx: 1, vy: -12, r: 3 });
        } else if (player.power >= 2) {
            playerBullets.push({ x: bx - 5, y: by, vx: -0.5, vy: -12, r: 3 });
            playerBullets.push({ x: bx + 5, y: by, vx: 0.5, vy: -12, r: 3 });
        } else {
            playerBullets.push({ x: bx, y: by, vx: 0, vy: -12, r: 4 });
        }
        playLaser();
    }
}

function updateEnemies(dt) {
    for (let i = enemies.length - 1; i >= 0; i--) {
        const e = enemies[i];
        e.x += e.vx;
        e.y += e.vy;
        
        if (e.type === 'spinner') e.angle += 0.05;
        if (e.type === 'boss') {
            e.angle += 0.03;
            if (e.y > H * 0.3) e.vy = 0;
            e.x += Math.sin(e.angle * 2) * 1;
        }
        
        // Keep in bounds horizontally
        if (e.x < e.w/2) { e.x = e.w/2; e.vx *= -1; }
        if (e.x > W - e.w/2) { e.x = W - e.w/2; e.vx *= -1; }
        
        // Remove if off screen bottom
        if (e.y > H + 60) { enemies.splice(i, continue); continue; }
        
        // Shooting
        e.shootTimer -= dt;
        if (e.shootTimer <= 0 && player) {
            switch(e.type) {
                case 'basic':
                    e.shootTimer = 80 - Math.min(30, wave * 3);
                    const angle = Math.atan2(player.y - e.y, player.x - e.x);
                    for (let a = -0.2; a <= 0.2; a += 0.2) {
                        enemyBullets.push({
                            x: e.x, y: e.y + e.h/2,
                            vx: Math.cos(angle + a) * 4,
                            vy: Math.sin(angle + a) * 4,
                            r: 5, color: '#ff6666'
                        });
                    }
                    break;
                    
                case 'spinner':
                    e.shootTimer = 12;
                    const sa = e.angle;
                    for (let j = 0; j < 4; j++) {
                        enemyBullets.push({
                            x: e.x, y: e.y,
                            vx: Math.cos(sa + j * Math.PI/2) * 3.5,
                            vy: Math.sin(sa + j * Math.PI/2) * 3.5,
                            r: 4, color: '#ffaa00'
                        });
                    }
                    break;
                    
                case 'spreader':
                    e.shootTimer = 60 - Math.min(20, wave * 2);
                    const count = 5 + Math.min(4, Math.floor(wave / 3));
                    for (let j = 0; j < count; j++) {
                        const a = -Math.PI/2 + (j - (count-1)/2) * (Math.PI / count);
                        enemyBullets.push({
                            x: e.x, y: e.y + e.h/2,
                            vx: Math.cos(a) * 3,
                            vy: Math.sin(a) * 3 + 1,
                            r: 5, color: '#66ff66'
                        });
                    }
                    break;
                    
                case 'boss':
                    e.shootTimer = 20;
                    const bc = 8 + Math.min(8, wave * 2);
                    for (let j = 0; j < bc; j++) {
                        const a = e.angle + (j / bc) * Math.PI * 2;
                        enemyBullets.push({
                            x: e.x, y: e.y,
                            vx: Math.cos(a) * 2.5,
                            vy: Math.sin(a) * 2.5,
                            r: 4, color: '#ff66ff'
                        });
                    }
                    // Aimed shot
                    const aa = Math.atan2(player.y - e.y, player.x - e.x);
                    enemyBullets.push({
                        x: e.x, y: e.y,
                        vx: Math.cos(aa) * 5,
                        vy: Math.sin(aa) * 5,
                        r: 6, color: '#ffffff'
                    });
                    break;
            }
        }
    }

function updateBullets(dt) {
    // Player bullets
    for (let i = playerBullets.length - 1; i >= 0; i--) {
        const b = playerBullets[i];
        b.x += b.vx;
        b.y += b.vy;
        if (b.y < -20 || b.x < -20 || b.x > W + 20) playerBullets.splice(i, 1);
    }
    
    // Enemy bullets
    for (let i = enemyBullets.length - 1; i >= 0; i--) {
        const b = enemyBullets[i];
        b.x += b.vx;
        b.y += b.vy;
        if (b.y < -30 || b.y > H + 30 || b.x < -30 || b.x > W + 30) {
            enemyBullets.splice(i, 1);
        }
    }
}

function checkCollisions() {
    // Player bullets vs enemies
    for (let i = playerBullets.length - 1; i >= 0; i--) {
        const b = playerBullets[i];
        for (let j = enemies.length - 1; j >= 0; j--) {
            const e = enemies[j];
            if (Math.abs(b.x - e.x) < e.w/2 + b.r && Math.abs(b.y - e.y) < e.h/2 + b.r) {
                playerBullets.splice(i, 1);
                e.hp--;
                // Hit particles
                for (let p = 0; p < 3; p++) {
                    particles.push({
                        x: b.x, y: b.y,
                        vx: (Math.random()-0.5)*4, vy: (Math.random()-0.5)*4,
                        life: 15, maxLife: 15, color: '#fff', size: 2
                    });
                }
                if (e.hp <= 0) {
                    score += e.type === 'boss' ? 500 : e.type === 'spinner' ? 200 : 100;
                    screenShake = 5;
                    playExplosion();
                    for (let p = 0; p < 20; p++) {
                        const a = Math.random() * Math.PI * 2;
                        const sp = 1 + Math.random() * 4;
                        particles.push({
                            x: e.x, y: e.y,
                            vx: Math.cos(a)*sp, vy: Math.sin(a)*sp,
                            life: 30 + Math.random()*20, maxLife: 50,
                            color: e.color, size: 3 + Math.random()*3
                        });
                    }
                    enemies.splice(j, 1);
                }
                break;
            }
        }
    }
    
    // Enemy bullets vs player
    if (player && player.invincible <= 0) {
        const hitR = keys['ShiftLeft'] || keys['ShiftRight'] ? 4 : 12;
        for (let i = enemyBullets.length - 1; i >= 0; i--) {
            const b = enemyBullets[i];
            if (Math.abs(b.x - player.x) < hitR + b.r && Math.abs(b.y - player.y) < hitR + b.r) {
                enemyBullets.splice(i, 1);
                hitPlayer();
                break;
            }
        }
        
        // Enemy body collision
        for (let j = enemies.length - 1; j >= 0; j--) {
            const e = enemies[j];
            if (Math.abs(e.x - player.x) < (e.w/2 + hitR) && Math.abs(e.y - player.y) < (e.h/2 + hitR)) {
                hitPlayer();
                break;
            }
        }
    }
}

function hitPlayer() {
    if (player.invincible > 0) return;
    player.lives--;
    player.invincible = 120;
    screenShake = 15;
    playPlayerHit();
    
    for (let p = 0; p < 30; p++) {
        const a = Math.random() * Math.PI * 2;
        const sp = 2 + Math.random() * 5;
        particles.push({
            x: player.x, y: player.y,
            vx: Math.cos(a)*sp, vy: Math.sin(a)*sp,
            life: 40, maxLife: 40, color: '#00ffff', size: 3
        });
    }
    
    if (player.lives <= 0) {
        gameState = 'gameover';
        try {
            if (score > highScore) {
                highScore = score;
                localStorage.setItem('bulletStormHS', highScore);
            }
        } catch(e) {}
    }
}

function updateParticles(dt) {
    for (let i = particles.length - 1; i >= 0; i--) {
        const p = particles[i];
        p.x += p.vx;
        p.y += p.vy;
        p.vx *= 0.95;
        p.vy *= 0.95;
        p.life--;
        if (p.life <= 0) particles.splice(i, 1);
    }
}

function updateStars() {
    for (const s of stars) {
        s.y += s.speed;
        if (s.y > H) { s.y = -2; s.x = Math.random() * W; }
    }
}

function drawStars() {
    for (const s of stars) {
        ctx.fillStyle = `rgba(150, 200, 255, ${s.brightness})`;
        ctx.fillRect(s.x, s.y, s.size, s.size);
    }
}

function drawPlayer() {
    if (!player) return;
    if (player.invincible > 0 && Math.floor(player.invincible / 4) % 2 === 0) return;
    
    const px = player.x, py = player.y;
    ctx.save();
    ctx.translate(px, py);
    
    // Ship body
    ctx.fillStyle = '#00e5ff';
    ctx.beginPath();
    ctx.moveTo(0, -20);
    ctx.lineTo(-14, 16);
    ctx.lineTo(-6, 12);
    ctx.lineTo(0, 18);
    ctx.lineTo(6, 12);
    ctx.lineTo(14, 16);
    ctx.closePath();
    ctx.fill();
    
    // Glow
    ctx.shadowColor = '#00e5ff';
    ctx.shadowBlur = 10;
    ctx.strokeStyle = '#00ffff';
    ctx.lineWidth = 1.5;
    ctx.stroke();
    ctx.shadowBlur = 0;
    
    // Focus mode indicator
    if (keys['ShiftLeft'] || keys['ShiftRight']) {
        ctx.beginPath();
        ctx.arc(0, 0, 5, 0, Math.PI * 2);
        ctx.strokeStyle = '#ff0';
        ctx.lineWidth = 1;
        ctx.stroke();
    }
    
    ctx.restore();
}

function drawEnemies() {
    for (const e of enemies) {
        ctx.save();
        ctx.translate(e.x, e.y);
        
        const hpRatio = e.hp / e.maxHp;
        ctx.fillStyle = e.color;
        ctx.shadowColor = e.color;
        ctx.shadowBlur = 8;
        
        if (e.type === 'basic') {
            ctx.beginPath();
            ctx.moveTo(0, -e.h/2);
            ctx.lineTo(e.w/2, e.h/2);
            ctx.lineTo(-e.w/2, e.h/2);
            ctx.closePath();
            ctx.fill();
        } else if (e.type === 'spinner') {
            ctx.rotate(e.angle);
            for (let i = 0; i < 4; i++) {
                ctx.save();
                ctx.rotate(i * Math.PI / 2);
                ctx.fillRect(-3, -e.h/2, 6, e.h/2);
                ctx.restore();
            }
        } else if (e.type === 'spreader') {
            ctx.beginPath();
            for (let i = 0; i < 6; i++) {
                const a = (i / 6) * Math.PI * 2;
                const r = e.w / 2;
                if (i === 0) ctx.moveTo(Math.cos(a)*r, Math.sin(a)*r);
                else ctx.lineTo(Math.cos(a)*r, Math.sin(a)*r);
            }
            ctx.closePath();
            ctx.fill();
        } else { // boss
            ctx.beginPath();
            for (let i = 0; i < 8; i++) {
                const a = (i / 8) * Math.PI * 2 + e.angle;
                const r = e.w / 2;
                if (i === 0) ctx.moveTo(Math.cos(a)*r, Math.sin(a)*r);
                else ctx.lineTo(Math.cos(a)*r, Math.sin(a)*r);
            }
            ctx.closePath();
            ctx.fill();
            // Inner
            ctx.fillStyle = '#fff';
            ctx.beginPath();
            ctx.arc(0, 0, 8, 0, Math.PI * 2);
            ctx.fill();
        }
        
        ctx.shadowBlur = 0;
        
        // HP bar
        if (hpRatio < 1) {
            ctx.fillStyle = '#333';
            ctx.fillRect(-e.w/2, -e.h/2 - 10, e.w, 4);
            ctx.fillStyle = hpRatio > 0.5 ? '#0f0' : hpRatio > 0.25 ? '#ff0' : '#f00';
            ctx.fillRect(-e.w/2, -e.h/2 - 10, e.w * hpRatio, 4);
        }
        
        ctx.restore();
    }
}

function drawBullets() {
    // Player bullets
    ctx.fillStyle = '#00ffff';
    ctx.shadowColor = '#00ffff';
    ctx.shadowBlur = 6;
    for (const b of playerBullets) {
        ctx.beginPath();
        ctx.arc(b.x, b.y, b.r, 0, Math.PI * 2);
        ctx.fill();
    }
    
    // Enemy bullets
    ctx.shadowBlur = 4;
    for (const b of enemyBullets) {
        ctx.fillStyle = b.color || '#ff4444';
        ctx.shadowColor = b.color || '#ff4444';
        ctx.beginPath();
        ctx.arc(b.x, b.y, b.r, 0, Math.PI * 2);
        ctx.fill();
    }
    ctx.shadowBlur = 0;
}

function drawParticles() {
    for (const p of particles) {
        const alpha = p.life / p.maxLife;
        ctx.globalAlpha = alpha;
        ctx.fillStyle = p.color;
        ctx.beginPath();
        ctx.arc(p.x, p.y, p.size * alpha, 0, Math.PI * 2);
        ctx.fill();
    }
    ctx.globalAlpha = 1;
}

    ctx.fillStyle = `rgba(255,255,255,${pulse})`;
    ctx.font = 'bold 20px monospace';
    ctx.fillText('PRESS SPACE TO START', W/2, H - 60);
}

function drawGameOver() {
    ctx.fillStyle = 'rgba(0,0,0,0.7)';
    ctx.fillRect(0, 0, W, H);
    
    ctx.textAlign = 'center';
    ctx.fillStyle = '#ff4444';
    ctx.font = 'bold 48px monospace';
    ctx.shadowColor = '#ff4444';
    ctx.shadowBlur = 15;
    ctx.fillText('GAME OVER', W/2, H/2 - 40);
    ctx.shadowBlur = 0;
    
    ctx.fillStyle = '#fff';
    ctx.font = '20px monospace';
    ctx.fillText(`FINAL SCORE: ${score}`, W/2, H/2 + 10);
    ctx.fillStyle = '#888';
    ctx.font = '16px monospace';
    ctx.fillText(`HIGH SCORE: ${highScore}`, W/2, H/2 + 45);
    
    const pulse = 0.5 + Math.sin(Date.now() / 500) * 0.5;
    ctx.fillStyle = `rgba(255,255,255,${pulse})`;
    ctx.font = '18px monospace';
    ctx.fillText('PRESS SPACE TO RESTART', W/2, H/2 + 100);
}

let lastTime = 0;
function gameLoop(timestamp) {
    const dt = Math.min((timestamp - lastTime) / 16.67, 3);
    lastTime = timestamp;
    
    // Clear
    ctx.fillStyle = '#0a0a1a';
    ctx.fillRect(0, 0, W, H);
    
    if (gameState === 'start') {
        updateStars();
        drawStartScreen();
    } else if (gameState === 'playing') {
        // Screen shake
        if (screenShake > 0) {
            screenShake -= dt;
            shakeX = (Math.random() - 0.5) * screenShake;
            shakeY = (Math.random() - 0.5) * screenShake;
        } else {
            shakeX = 0; shakeY = 0;
        }
        
        ctx.save();
        ctx.translate(shakeX, shakeY);
        
        updateStars();
        drawStars();
        
        // Spawning
        spawnTimer -= dt;
        if (spawnTimer <= 0) {
            const interval = Math.max(40, 120 - wave * 8);
            spawnTimer = interval + Math.random() * 30;
            spawnEnemy();
            wave += 0.5;
        }
        
        updatePlayer(dt);
        updateEnemies(dt);
        updateBullets(dt);
        checkCollisions();
        updateParticles(dt);
        
        drawBullets();
        drawEnemies();
        drawPlayer();
        drawParticles();
        drawHUD();
        
        ctx.restore();
    } else if (gameState === 'gameover') {
        updateStars();
        updateParticles(dt);
        drawStars();
        drawBullets();
        drawParticles();
        drawGameOver();
    }
    
    requestAnimationFrame(gameLoop);
}

createStars();
requestAnimationFrame(gameLoop);
