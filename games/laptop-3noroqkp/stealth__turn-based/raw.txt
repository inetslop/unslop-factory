function playTone(freq, dur, type, vol) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = type || 'square';
    osc.frequency.value = freq;
    gain.gain.setValueAtTime(vol || 0.1, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + dur);
}

function getGuardVision(g) {
    let visible = [];
    const facing = [
        {dx:0,dy:-1}, // N
        {dx:1,dy:0},  // E
        {dx:0,dy:1},  // S
        {dx:-1,dy:0}  // W
    ][g.dir];

    // See 3 tiles directly in front
    for (let d = 1; d <= 3; d++) {
        let tx = g.x + facing.dx * d;
        let ty = g.y + facing.dy * d;
        if (tx < 0 || tx >= MAP_W || ty < 0 || ty >= MAP_H) break;
        if (map[ty][tx] === 1) break;
        visible.push({x: tx, y: ty});
    }

    // See 1 tile to each side at distance 1 and 2
    const sideL = { dx: facing.dy, dy: -facing.dx };
    const sideR = { dx: -facing.dy, dy: facing.dx };

    for (let d = 1; d <= 2; d++) {
        let lx = g.x + facing.dx * d + sideL.dx;
        let ly = g.y + facing.dy * d + sideL.dy;
        if (lx >= 0 && lx < MAP_W && ly >= 0 && ly < MAP_H && map[ly][lx] !== 1) {
            visible.push({x: lx, y: ly});
        }
        let rx = g.x + facing.dx * d + sideR.dx;
        let ry = g.y + facing.dy * d + sideR.dy;
        if (rx >= 0 && rx < MAP_W && ry >= 0 && ry < MAP_H && map[ry][rx] !== 1) {
            visible.push({x: rx, y: ry});
        }
    }

    return visible;
}

function isPlayerVisible() {
    // Player in shadow is hidden
    if (map[player.y][player.x] === 2) return false;
    for (let g of guards) {
        let vision = getGuardVision(g);
        for (let v of vision) {
            if (v.x === player.x && v.y === player.y) return true;
        }
        // Also check if guard is adjacent to player
        if (Math.abs(g.x - player.x) + Math.abs(g.y - player.y) <= 1) return true;
    }
    return false;
}

function movePlayer(dx, dy) {
    let nx = player.x + dx;
    let ny = player.y + dy;
    if (nx < 0 || nx >= MAP_W || ny < 0 || ny >= MAP_H) return;
    if (map[ny][nx] === 1) return;
    // Can't walk into a guard
    for (let g of guards) {
        if (g.x === nx && g.y === ny) return;
    }
    player.x = nx;
    player.y = ny;
    playMove();
    resolveTurn();
}

function waitPlayer() {
    playWait();
    resolveTurn();
}

function moveGuards() {
    for (let g of guards) {
        let next = g.pathIdx + 1;
        if (next >= g.path.length) next = 0;
        let target = g.path[next];
        // Check if path is clear (no player in the way)
        g.x = target[0];
        g.y = target[1];
        g.pathIdx = next;
        // Update facing based on movement
        let prevX = g.path[g.pathIdx === 0 ? g.path.length - 1 : g.pathIdx - 1][0];
        let prevY = g.path[g.pathIdx === 0 ? g.path.length - 1 : g.pathIdx - 1][1];
        let mdx = g.x - prevX;
        let mdy = g.y - prevY;
        if (mdy === -1) g.dir = 0;
        else if (mdx === 1) g.dir = 1;
        else if (mdy === 1) g.dir = 2;
        else if (mdx === -1) g.dir = 3;
    }
}

function resolveTurn() {
    moveGuards();
    turn++;
    // Check if caught
    if (isPlayerVisible()) {
        gameState = 'lost';
        playLose();
    }
    // Check win
    if (map[player.y][player.x] === 3) {
        gameState = 'won';
        playWin();
    }
}

            let type = map[y][x];
            if (type === 0) {
                ctx.fillStyle = '#1a1a2e';
                ctx.fillRect(0, 0, TILE, TILE);
                ctx.strokeStyle = '#252540';
                ctx.lineWidth = 1;
                ctx.strokeRect(0.5, 0.5, TILE-1, TILE-1);
            } else if (type === 1) {
                ctx.fillStyle = '#2d2d44';
                ctx.fillRect(0, 0, TILE, TILE);
                ctx.fillStyle = '#3a3a5c';
                ctx.fillRect(2, 2, TILE-4, TILE-4);
            } else if (type === 2) {
                let pulse = Math.sin(animFrame * 0.05) * 0.15 + 0.35;
                ctx.fillStyle = '#0f0f23';
                ctx.fillRect(0, 0, TILE, TILE);
                ctx.fillStyle = `rgba(100, 50, 180, ${pulse})`;
                ctx.fillRect(2, 2, TILE-4, TILE-4);
            } else if (type === 3) {
                ctx.fillStyle = '#1a1a2e';
                ctx.fillRect(0, 0, TILE, TILE);
                let docPulse = Math.sin(animFrame * 0.08) * 0.3 + 0.7;
                ctx.fillStyle = `rgba(255, 215, 0, ${docPulse * 0.4})`;
                ctx.fillRect(TILE*0.2, TILE*0.2, TILE*0.6, TILE*0.6);
                ctx.fillStyle = `rgba(255, 215, 0, ${docPulse})`;
                ctx.font = `${Math.floor(TILE*0.5)}px monospace`;
                ctx.textAlign = 'center';
                ctx.textBaseline = 'middle';
                ctx.fillText('◆', TILE/2, TILE/2);
            }
            ctx.restore();
        }
    }

    // Draw guard vision cones
    for (let g of guards) {
        let vision = getGuardVision(g);
        for (let v of vision) {
            let vx = offsetX + v.x * TILE;
            let vy = offsetY + v.y * TILE;
            let dist = Math.abs(v.x - g.x) + Math.abs(v.y - g.y);
            let alpha = 0.12 / dist;
            ctx.fillStyle = `rgba(255, 80, 80, ${alpha})`;
            ctx.fillRect(vx+2, vy+2, TILE-4, TILE-4);
        }
    }

    // Draw guards
    for (let g of guards) {
        let px = offsetX + g.x * TILE + TILE/2;
        let py = offsetY + g.y * TILE + TILE/2;

        ctx.fillStyle = '#ff4444';
        ctx.beginPath();
        ctx.arc(px, py, TILE*0.3, 0, Math.PI*2);
        ctx.fill();
        ctx.fillStyle = '#cc2222';
        ctx.beginPath();
        ctx.arc(px, py, TILE*0.18, 0, Math.PI*2);
        ctx.fill();

        const facing = [{dx:0,dy:-1},{dx:1,dy:0},{dx:0,dy:1},{dx:-1,dy:0}][g.dir];
        let ex = px + facing.dx * TILE*0.18;
        let ey = py + facing.dy * TILE*0.18;
        ctx.fillStyle = '#ffaaaa';
        ctx.beginPath();
        ctx.arc(ex, ey, 3, 0, Math.PI*2);
        ctx.fill();
    }

    // Draw player
    let ppx = offsetX + player.x * TILE + TILE/2;
    let ppy = offsetY + player.y * TILE + TILE/2;
    let glow = Math.sin(animFrame * 0.1) * 0.3 + 0.7;

    ctx.fillStyle = `rgba(0, 255, 136, ${glow * 0.25})`;
    ctx.beginPath();
    ctx.arc(ppx, ppy, TILE*0.45, 0, Math.PI*2);
    ctx.fill();

    ctx.fillStyle = '#00ff88';
    ctx.beginPath();
    ctx.arc(ppx, ppy, TILE*0.25, 0, Math.PI*2);
    ctx.fill();
    ctx.fillStyle = '#aaffdd';
    ctx.beginPath();
    ctx.arc(ppx, ppy, TILE*0.12, 0, Math.PI*2);
    ctx.fill();

    if (map[player.y][player.x] === 2) {
        ctx.strokeStyle = `rgba(140, 80, 220, ${glow})`;
        ctx.lineWidth = 2;
        ctx.beginPath();
        ctx.arc(ppx, ppy, TILE*0.35, 0, Math.PI*2);
        ctx.stroke();
    }

    // HUD
    hud.textContent = `TURN ${turn}  |  Shadow Protocol`;
    let statusText = map[player.y][player.x] === 2 ? 'HIDDEN' : (isPlayerVisible() ? 'EXPOSED!' : 'CONCEALED');
    msgEl.textContent = `[${statusText}]  WASD/Arrows:Move  SPACE:Wait`;
    msgEl.style.color = statusText === 'HIDDEN' ? '#aa77ff' : (statusText === 'EXPOSED!' ? '#ff4444' : '#00ff88');

    // Danger indicator
    if (isPlayerVisible() && gameState === 'playing') {
        let flash = Math.sin(animFrame * 0.3) > 0;
        if (flash) {
            ctx.strokeStyle = 'rgba(255, 50, 50, 0.6)';
            ctx.lineWidth = 4;
            ctx.strokeRect(offsetX - 4, offsetY - 4, MAP_W*TILE + 8, MAP_H*TILE + 8);
        }
    }

    // End screens
    if (gameState === 'won' || gameState === 'lost') {
        drawEndScreen(gameState === 'won');
    }
}

function gameLoop() {
    animFrame++;
    ctx.clearRect(0, 0, W, H);

    if (gameState === 'start') {
        drawStartScreen();
    } else {
        drawGame();
    }

    requestAnimationFrame(gameLoop);
}

buildMap();
gameLoop();
})();
