<script>
const canvas = document.getElementById('c');
const ctx = canvas.getContext('2d');
let W, H;
function resize() {
W = window.innerWidth; H = window.innerHeight;
canvas.width = W; canvas.height = H;
}
window.addEventListener('resize', resize);
resize();

let audioCtx = null;
function initAudio() { if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)(); }
function playSound(freq, dur, type='sine', vol=0.3) {
if (!audioCtx) return;
const o = audioCtx.createOscillator();
const g = audioCtx.createGain();
o.type = type; o.frequency.value = freq;
g.gain.setValueAtTime(vol, audioCtx.currentTime);
g.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
o.connect(g); g.connect(audioCtx.destination);
o.start(); o.stop(audioCtx.currentTime + dur);
}

let game = { state: 'start', level: 0, score: 0, strokes: 0, maxStrokes: 5 };
let ball = { x: 0, y: 0, vx: 0, vy: 0, r: 10, moving: false };
let hole = { x: 0, y: 0, r: 22 };
let zombies = [];
let walls = [];
let aiming = false, aimStart = null;
let animFrame = 0;
let msgTimer = null;

function showMsg(text) {
const el = document.getElementById('msg');
el.textContent = text; el.style.opacity = '1';
clearTimeout(msgTimer);
msgTimer = setTimeout(() => { el.style.opacity = '0'; }, 2000);
}

function startGame() {
initAudio();
document.getElementById('startScreen').style.display = 'none';
game.state = 'playing';
game.score = 0; game.level = 0;
loadLevel(0);
playSound(440, 0.2, 'sine', 0.2);
}

function loadLevel(lv) {
game.level = lv;
game.strokes = 0;
game.maxStrokes = 5 + Math.floor(lv / 2);
ball.moving = false; ball.vx = 0; ball.vy = 0;
zombies = []; walls = [];

const courseW = W * 0.8, courseH = H * 0.7;
const ox = (W - courseW) / 2, oy = (H - courseH) / 2 + 30;

ball.x = ox + 60; ball.y = oy + courseH - 60;
hole.x = ox + courseW - 80; hole.y = oy + 80;

// Walls around course
walls.push({x:ox, y:oy, w:courseW, h:10});
walls.push({x:ox, y:oy+courseH-10, w:courseW, h:10});
walls.push({x:ox, y:oy, w:10, h:courseH});
walls.push({x:ox+courseW-10, y:oy, w:10, h:courseH});

// Add obstacles
if (lv >= 1) walls.push({x:ox+courseW/2-5, y:oy+50, w:10, h:courseH-100});
if (lv >= 2) { const cx=ox+courseW*0.3, cy=oy+courseH*0.4; walls.push({x:cx-40,y:cy-40,w:80,h:80}); }
if (lv >= 3) { const cx=ox+courseW*0.6, cy=oy+courseH*0.6; walls.push({x:cx-50,y:cy-50,w:100,h:100}); }

// Zombies
const zc = Math.min(2 + lv, 7);
for (let i = 0; i < zc; i++) {
const zx = ox + 80 + Math.random() * (courseW - 160);
const zy = oy + 60 + Math.random() * (courseH - 120);
const dir = Math.random() < 0.5 ? 'h' : 'v';
zombies.push({ x:zx, y:zy, r:18, vx:(dir==='h'?1:0)*(Math.random()<0.5?1:-1), vy:(dir==='v'?1:0)*(Math.random()<0.5?1:-1), speed:0.8+lv*0.15, phase: Math.random()*Math.PI*2 });
}

updateHUD();
}

function updateHUD() {
document.getElementById('hud').textContent = `Level ${game.level+1} | Strokes: ${game.strokes}/${game.maxStrokes} | Score: ${game.score}`;
}

function checkWin() {
const dx = ball.x - hole.x, dy = ball.y - hole.y;
const dist = Math.sqrt(dx*dx+dy*dy);
if (dist < hole.r && Math.sqrt(ball.vx*ball.vx+ball.vy*ball.vy) < 4) {
ball.moving = false; ball.vx = 0; ball.vy = 0;
playSound(200, 0.5, 'sine', 0.4);
const par = game.maxStrokes - game.strokes;
game.score += Math.max(1, par * 2);
showMsg('⛳ SUNK! +'+Math.max(1,par*2));
if (game.level < 9) { setTimeout(() => { loadLevel(game.level+1); }, 1500); } 
else { game.state = 'won'; showMsg('🏆 YOU BEAT THE COURSE! Score: '+game.score); }
}
}

function checkLose() {
if (!ball.moving && game.strokes >= game.maxStrokes) {
setTimeout(() => {
showMsg('💀 Out of strokes! Restarting level...');
playSound(100, 0.5, 'sawtooth', 0.3);
setTimeout(() => loadLevel(game.level), 1800);
}, 800);
}
}

function update() {
animFrame++;
if (game.state !== 'playing') return;

// Zombies move
for (let z of zombies) {
z.phase += 0.05;
z.x += z.vx * z.speed + Math.sin(z.phase)*0.3;
z.y += z.vy * z.speed + Math.cos(z.phase*1.3)*0.2;
const ox = W*0.1, oy = H*0.15, cw = W*0.8, ch = H*0.7;
if (z.x < ox+30) { z.vx = Math.abs(z.vx); }
if (z.x > ox+cw-30) { z.vx = -Math.abs(z.vx); }
if (z.y < oy+30) { z.vy = Math.abs(z.vy); }
if (z.y > oy+ch-30) { z.vy = -Math.abs(z.vy); }
}

// Ball physics
if (ball.moving) {
ball.x += ball.vx; ball.y += ball.vy;
ball.vx *= 0.975; ball.vy *= 0.975;
const spd = Math.sqrt(ball.vx*ball.vx+ball.vy*ball.vy);
if (spd < 0.3) { ball.moving = false; ball.vx=0; ball.vy=0; checkLose(); }

// Wall collisions
for (let w of walls) {
if (ball.x + ball.r > w.x && ball.x - ball.r < w.x+w.w && ball.y + ball.r > w.y && ball.y - ball.r < w.y+w.h) {
// Bounce off nearest edge
const cx = Math.max(w.x, Math.min(ball.x, w.x+w.w));
const cy = Math.max(w.y, Math.min(ball.y, w.y+w.h));
const dx = ball.x-cx, dy = ball.y-cy;
const d = Math.sqrt(dx*dx+dy*dy);
if (d < ball.r) {
const nx=dx/d, ny=dy/d;
ball.vx = -ball.vx * 0.7; ball.vy = -ball.vy * 0.7;
playSound(300, 0.1, 'square', 0.2);
}
}
}

// Zombie collisions
for (let z of zombies) {
const dx = ball.x - z.x, dy = ball.y - z.y;
const d = Math.sqrt(dx*dx+dy*dy);
if (d < ball.r + z.r) {
const nx=dx/d, ny=dy/d;
ball.vx = nx * 5 + ball.vx * 0.3;
ball.vy = ny * 5 + ball.vy * 0.3;
z.x += nx*20; z.y += ny*20;
playSound(150, 0.15, 'sawtooth', 0.3);
}
}

checkWin();
}
}

function draw() {
ctx.fillStyle = '#1a3a1a'; ctx.fillRect(0,0,W,H);

if (game.state === 'start') return;

// Course
const ox=W*0.1, oy=H*0.15, cw=W*0.8, ch=H*0.7;
ctx.fillStyle = '#2d6b2d'; ctx.fillRect(ox,oy,cw,ch);
// Stripes
for(let i=0;i<cw/40;i++){ if(i%2===0) { ctx.fillStyle='rgba(0,0,0,0.05)'; ctx.fillRect(ox+i*40,oy,40,ch);} }

// Walls
ctx.fillStyle = '#555';
for (let w of walls) ctx.fillRect(w.x,w.y,w.w,w.h);

// Hole
ctx.beginPath(); ctx.arc(hole.x,hole.y,hole.r,0,Math.PI*2); ctx.fillStyle='#111'; ctx.fill();
ctx.beginPath(); ctx.arc(hole.x,hole.y,hole.r-3,0,Math.PI*2); ctx.fillStyle='#000'; ctx.fill();

// Zombies
for (let z of zombies) {
const bob = Math.sin(z.phase*2)*3;
ctx.save(); ctx.translate(z.x, z.y+bob);
// Body
ctx.fillStyle = '#4a7a4a'; ctx.fillRect(-12,-8,24,20);
// Head
ctx.beginPath(); ctx.arc(0,-14,10,0,Math.PI*2); ctx.fillStyle='#6a9a6a'; ctx.fill();
// Eyes
ctx.fillStyle='#ff3333'; ctx.fillRect(-5,-17,3,3); ctx.fillRect(3,-17,3,3);
// Arms
ctx.strokeStyle='#4a7a4a'; ctx.lineWidth=4;
ctx.beginPath(); ctx.moveTo(-12,0); ctx.lineTo(-20,8+Math.sin(z.phase*3)*5); ctx.stroke();
ctx.beginPath(); ctx.moveTo(12,0); ctx.lineTo(20,-5+Math.cos(z.phase*2.7)*4); ctx.stroke();
ctx.restore();
}

// Ball
ctx.beginPath(); ctx.arc(ball.x, ball.y, ball.r, 0, Math.PI*2); ctx.fillStyle='#fff'; ctx.fill();
ctx.strokeStyle='#ccc'; ctx.lineWidth=1; ctx.stroke();
// Dimple lines
ctx.beginPath(); ctx.arc(ball.x-3,ball.y-3,4,0,Math.PI*2); ctx.strokeStyle='rgba(0,0,0,0.2)'; ctx.stroke();

// Aim line
if (aiming && aimStart) {
const dx = aimStart.x - ball.x, dy = aimStart.y - ball.y;
const d = Math.sqrt(dx*dx+dy*dy);
const power = Math.min(d, 150);
ctx.setLineDash([5,5]);
ctx.beginPath(); ctx.moveTo(ball.x,ball.y);
ctx.lineTo(ball.x + (dx/d)*power*1.5, ball.y + (dy/d)*power*1.5);
ctx.strokeStyle='rgba(255,255,0,0.7)'; ctx.lineWidth=3; ctx.stroke();
ctx.setLineDash([]);
}

// Flag at hole
ctx.beginPath(); ctx.moveTo(hole.x,hole.y-hole.r); ctx.lineTo(hole.x,hole.y-hole.r-30); ctx.strokeStyle='#fff'; ctx.lineWidth=2; ctx.stroke();
ctx.beginPath(); ctx.moveTo(hole.x,hole.y-hole.r-30); ctx.lineTo(hole.x+15,hole.y-hole.r-22); ctx.lineTo(hole.x,hole.y-hole.r-14); ctx.closePath(); ctx.fillStyle='#ff4444'; ctx.fill();
}

function loop() {
update(); draw(); requestAnimationFrame(loop);
}

// Input
let mouseDown = false, mouseX=0, mouseY=0;
canvas.addEventListener('mousedown', e => {
if (game.state !== 'playing') return;
mouseX = e.clientX; mouseY = e.clientY;
const dx = mouseX - ball.x, dy = mouseY - ball.y;
if (Math.sqrt(dx*dx+dy*dy) < 60 && !ball.moving) { aiming=true; aimStart={x:mouseX,y:mouseY}; }
});
window.addEventListener('mousemove', e => { mouseX=e.clientX; mouseY=e.clientY; });
window.addEventListener('mouseup', e => {
if (!aiming) return;
aiming=false;
const dx = aimStart.x - ball.x, dy = aimStart.y - ball.y;
const d = Math.sqrt(dx*dx+dy*dy);
if (d < 5) { aimStart=null; return; }
const power = Math.min(d, 150) * 0.12;
ball.vx = (dx/d)*power; ball.vy = (dy/d)*power;
ball.moving = true; game.strokes++;
updateHUD();
playSound(600, 0.1, 'sine', 0.3);
aimStart=null;
});

// Touch support
canvas.addEventListener('touchstart', e => {
e.preventDefault();
const t=e.touches[0];
mouseX=t.clientX; mouseY=t.clientY;
if (game.state!=='playing') return;
const dx=mouseX-ball.x, dy=mouseY-ball.y;
if (Math.sqrt(dx*dx+dy*dy)<60 && !ball.moving) { aiming=true; aimStart={x:mouseX,y:mouseY}; }
});
window.addEventListener('touchmove', e => { const t=e.touches[0]; mouseX=t.clientX; mouseY=t.clientY; });
window.addEventListener('touchend', e => {
e.preventDefault();
if (!aiming) return;
aiming=false;
const dx=aimStart.x-ball.x, dy=aimStart.y-ball.y;
const d=Math.sqrt(dx*dx+dy*dy);
if(d<5){aimStart=null;return;}
const power=Math.min(d,150)*0.12;
ball.vx=(dx/d)*power; ball.vy=(dy/d)*power;
ball.moving=true; game.strokes++; updateHUD();
playSound(600,0.1,'sine',0.3); aimStart=null;
});

// Restart key
window.addEventListener('keydown', e => {
if (e.key === 'r' || e.key === 'R') { if (game.state==='playing'||game.state==='won') loadLevel(0); game.score=0; updateHUD(); }
if (e.key === 'Escape') { document.getElementById('startScreen').style.display='flex'; game.state='start'; }
});

// Win detection for end screen
setInterval(() => { if(game.state==='won') {} }, 100);

loop();
</script>
