const canvas = document.getElementById('c');
const ctx = canvas.getContext('2d');

let W, H, cellSize, cols, rows;
let gameStarted = false;
let gameOver = false;
let winner = '';
let playerScore = 0, aiScore = 0;
let audioCtx = null;

function resize() {
    W = window.innerWidth;
    H = window.innerHeight;
    canvas.width = W;
    canvas.height = H;
    cellSize = Math.floor(Math.min(W / 14, H / 10));
    cols = 12;
    rows = 8;
    const gridW = cols * cellSize;
    const gridH = rows * cellSize;
    offset.x = (W - gridW) / 2;
    offset.y = (H - gridH) / 2;
}

const offset = { x: 0, y: 0 };

function playBeep(freq, dur, vol) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.frequency.value = freq;
    osc.type = 'square';
    gain.gain.setValueAtTime(vol || 0.1, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + (dur || 0.1));
    osc.start();
    osc.stop(audioCtx.currentTime + (dur || 0.1));
}

let player, ai, blocks, ball;

function initGame() {
    playerScore = 0;
    aiScore = 0;
    gameOver = false;
    winner = '';
    resetRound();
}

function resetRound() {
    player = { x: 1, y: 3 };
    ai = { x: 10, y: 4 };
    blocks = [
        { x: 5, y: 2 },
        { x: 6, y: 5 },
        { x: 5, y: 7 }
    ];
    resetBall();
}

function resetBall() {
    const speed = Math.max(cellSize * 0.03, 2);
    const angle = (Math.random() - 0.5) * Math.PI * 0.6;
    ball = {
        x: offset.x + cols * cellSize / 2,
        y: offset.y + rows * cellSize / 2,
        vx: Math.cos(angle) * speed,
        vy: Math.sin(angle) * speed,
        radius: cellSize * 0.3
    };
}

function startGame() {
    if (!audioCtx) {
        audioCtx = new (window.AudioContext || window.webkitAudioContext)();
    }
    gameStarted = true;
    initGame();
}

let keys = {};
window.addEventListener('keydown', e => {
    keys[e.key] = true;
    if (!gameStarted && !gameOver) startGame();
    if (gameOver && (e.key === 'r' || e.key === 'R')) initGame();
});
window.addEventListener('keyup', e => { keys[e.key] = false; });

function canPush(x, y) {
    if (x < 0 || x >= cols || y < 0 || y >= rows) return false;
    if (player.x === x && player.y === y) return false;
    if (ai.x === x && ai.y === y) return false;
    for (let b of blocks) if (b.x === x && b.y === y) return true;
    return false;
}

function isOccupied(x, y) {
    if (player.x === x && player.y === y) return true;
    if (ai.x === x && ai.y === y) return true;
    for (let b of blocks) if (b.x === x && b.y === y) return true;
    return false;
}

function tryMove(entity, dx, dy) {
    const nx = entity.x + dx;
    const ny = entity.y + dy;
    if (nx < 0 || nx >= cols || ny < 0 || ny >= rows) return;
    // Check if hitting a block - push it
    let hitBlock = null;
    for (let b of blocks) {
        if (b.x === nx && b.y === ny) {
            hitBlock = b;
            break;
        }
    }
    if (hitBlock) {
        const bx = hitBlock.x + dx;
        const by = hitBlock.y + dy;
        if (bx >= 0 && bx < cols && by >= 0 && by < rows && !isOccupied(bx, by)) {
            hitBlock.x = bx;
            hitBlock.y = by;
            entity.x = nx;
            entity.y = ny;
            playBeep(220, 0.05, 0.08);
        }
    } else if (!isOccupied(nx, ny)) {
        entity.x = nx;
        entity.y = ny;
        playBeep(150, 0.03, 0.05);
    }
}

function aiThink() {
    // AI tries to position itself or push blocks to block the ball
    const ballGridX = Math.floor((ball.x - offset.x) / cellSize);
    const ballGridY = Math.floor((ball.y - offset.y) / cellSize);
    
    let bestMove = null;
    let bestScore = -Infinity;

    // Try to position between ball and right wall, or push blocks
    for (let dx = -1; dx <= 1; dx++) {
        for (let dy = -1; dy <= 1; dy++) {
            if (dx === 0 && dy === 0) continue;
            const nx = ai.x + dx;
            const ny = ai.y + dy;
            if (nx < 6 || nx >= cols || ny < 0 || ny >= rows) continue;
            
            let score = 0;
            // Prefer being between ball and right edge
            if (ballGridX > 6 && ballGridX < 10) {
                const distToBall = Math.abs(nx - ballGridX) + Math.abs(ny - ballGridY);
                score += 5 / (distToBall + 1);
            }
            // Prefer center column for AI
            score += 3 - Math.abs(nx - 9);
            // Prefer middle rows
            score += 2 - Math.abs(ny - 3.5);
            
            if (score > bestScore) {
                bestScore = score;
                bestMove = { dx, dy };
            }
        }
    }

    if (bestMove && Math.random() < 0.7) {
        tryMove(ai, bestMove.dx, bestMove.dy);
    }
}

let aiTimer = 0;

function update(dt) {
    if (!gameStarted || gameOver) return;

    // Player movement
    const pSpeed = 1 / 12;
    if (keys['ArrowUp'] || keys['w'] || keys['W']) tryMove(player, 0, -1);
    else if (keys['ArrowDown'] || keys['s'] || keys['S']) tryMove(player, 0, 1);
    else if (keys['ArrowLeft'] || keys['a'] || keys['A']) tryMove(player, -1, 0);
    else if (keys['ArrowRight'] || keys['d'] || keys['D']) tryMove(player, 1, 0);

    // AI movement (slower)
    aiTimer += dt;
    if (aiTimer > 250) {
        aiThink();
        aiTimer = 0;
    }

    // Ball physics
    ball.x += ball.vx;
    ball.y += ball.vy;

    const gridLeft = offset.x;
    const gridRight = offset.x + cols * cellSize;
    const gridTop = offset.y;
    const gridBottom = offset.y + rows * cellSize;

    // Bounce off top/bottom
    if (ball.y - ball.radius < gridTop) {
        ball.y = gridTop + ball.radius;
        ball.vy = Math.abs(ball.vy);
        playBeep(330, 0.05, 0.08);
    }
    if (ball.y + ball.radius > gridBottom) {
        ball.y = gridBottom - ball.radius;
        ball.vy = -Math.abs(ball.vy);
        playBeep(330, 0.05, 0.08);
    }

    // Bounce off blocks (AABB vs circle)
    for (let b of blocks) {
        const bx = offset.x + b.x * cellSize;
        const by = offset.y + b.y * cellSize;
        const bw = cellSize;
        const bh = cellSize;

        const cx = Math.max(bx, Math.min(ball.x, bx + bw));
        const cy = Math.max(by, Math.min(ball.y, by + bh));
        const dx = ball.x - cx;
        const dy = ball.y - cy;
        const dist = Math.sqrt(dx * dx + dy * dy);

        if (dist < ball.radius) {
            if (dist === 0) { ball.vy = -ball.vy; continue; }
            const nx = dx / dist;
            const ny = dy / dist;
            const dot = ball.vx * nx + ball.vy * ny;
            ball.vx -= 2 * dot * nx;
            ball.vy -= 2 * dot * ny;
            ball.x += nx * (ball.radius - dist);
            ball.y += ny * (ball.radius - dist);
            playBeep(440, 0.05, 0.1);
        }
    }

    // Bounce off player
    const px = offset.x + player.x * cellSize;
    const py = offset.y + player.y * cellSize;
    let cx = Math.max(px, Math.min(ball.x, px + cellSize));
    let cy = Math.max(py, Math.min(ball.y, py + cellSize));
    let dx2 = ball.x - cx;
    let dy2 = ball.y - cy;
    let dist2 = Math.sqrt(dx2 * dx2 + dy2 * dy2);
    if (dist2 < ball.radius) {
        if (dist2 === 0) { ball.vx = -ball.vx; }
        else {
            const nx = dx2 / dist2, ny = dy2 / dist2;
            const dot = ball.vx * nx + ball.vy * ny;
            ball.vx -= 2 * dot * nx;
            ball.vy -= 2 * dot * ny;
            ball.x += nx * (ball.radius - dist2);
            ball.y += ny * (ball.radius - dist2);
        }
        playBeep(500, 0.04, 0.1);
    }

    // Bounce off AI
    const ax = offset.x + ai.x * cellSize;
    const ay = offset.y + ai.y * cellSize;
    cx = Math.max(ax, Math.min(ball.x, ax + cellSize));
    cy = Math.max(ay, Math.min(ball.y, ay + cellSize));
    dx2 = ball.x - cx;
    dy2 = ball.y - cy;
    dist2 = Math.sqrt(dx2 * dx2 + dy2 * dy2);
    if (dist2 < ball.radius) {
        if (dist2 === 0) { ball.vx = -ball.vx; }
        else {
            const nx = dx2 / dist2, ny = dy2 / dist2;
            const dot = ball.vx * nx + ball.vy * ny;
            ball.vx -= 2 * dot * nx;
            ball.vy -= 2 * dot * ny;
            ball.x += nx * (ball.radius - dist2);
            ball.y += ny * (ball.radius - dist2);
        }
        playBeep(500, 0.04, 0.1);
    }

    // Scoring
    if (ball.x < gridLeft - ball.radius) {
        playerScore++;
        playBeep(100, 0.3, 0.15);
        if (playerScore >= 5) { gameOver = true; winner = 'PLAYER'; }
        resetBall();
    }
    if (ball.x > gridRight + ball.radius) {
        aiScore++;
        playBeep(80, 0.3, 0.15);
        if (aiScore >= 5) { gameOver = true; winner = 'AI'; }
        resetBall();
    }

    // Speed up over time
    const speed = Math.sqrt(ball.vx * ball.vx + ball.vy * ball.vy);
    const targetSpeed = Math.max(cellSize * 0.03, 2) + (playerScore + aiScore) * 0.3;
    if (speed > 0 && targetSpeed > speed) {
        ball.vx *= targetSpeed / speed;
        ball.vy *= targetSpeed / speed;
    }
}

    }

    // Player
    const ppx = gridLeft + player.x * cellSize;

    ctx.fillStyle = '#4ecdc4';
    ctx.fillRect(ppx + 4, ppy + 4, cellSize - 8, cellSize - 8);
