function isWall(x, y) {
  if (x < 0 || x >= cols || y < 0 || y >= rows) return true;
  return gameGrid[y][x] === 1;
}

function canMove(x, y, dir) {
  let nx = x, ny = y;
  if (dir === 'up') ny--;
  else if (dir === 'down') ny++;
  else if (dir === 'left') nx--;
  else if (dir === 'right') nx++;
  // Tunnel wrap
  if (nx < 0) nx = cols - 1;
  if (nx >= cols) nx = 0;
  return !isWall(nx, ny);
}

function moveEntity(e, dir) {
  let nx = e.x, ny = e.y;
  if (dir === 'up') ny--;
  else if (dir === 'down') ny++;
  else if (dir === 'left') nx--;
  else if (dir === 'right') nx++;
  if (nx < 0) nx = cols - 1;
  if (nx >= cols) nx = 0;
  if (!isWall(nx, ny)) { e.x = nx; e.y = ny; return true; }
  return false;
}

function getOpposite(dir) {
  const opp = { up: 'down', down: 'up', left: 'right', right: 'left' };
  return opp[dir];
}

function ghostAI(g, idx) {
  if (invincibleTimer > 0) g.dir = getOpposite(g.dir);
  // Rival racing AI: try to follow track and stay competitive
  let dirs = [];
  const opp = getOpposite(g.dir);
  if (canMove(g.x, g.y, 'left')) dirs.push('left');
  if (canMove(g.x, g.y, 'right')) dirs.push('right');
  if (canMove(g.x, g.y, 'up')) dirs.push('up');
  if (canMove(g.x, g.y, 'down')) dirs.push('down');

  // Weight toward pac-man direction for racing feel
  let target = null;
  if (Math.random() < 0.3) target = { x: pacX, y: pacY };

  if (target) {
    let bestDir = g.dir; let bestDist = Infinity;
    dirs.forEach(d => {
      let nx = g.x, ny = g.y;
      if (d === 'up') ny--; else if (d === 'down') ny++;
      else if (d === 'left') nx--; else if (d === 'right') nx++;
      const dist = Math.abs(nx - target.x) + Math.abs(ny - target.y);
      if (dist < bestDist && d !== opp) { bestDir = d; bestDist = dist; }
    });
    g.dir = bestDir;
  } else {
    // Follow track: prefer not to reverse, prefer forward/side based on lap
    const valid = dirs.filter(d => d !== opp);
    if (valid.length > 0) {
      if (lap === 1 && idx === 0) g.dir = valid[0];
      else g.dir = valid[Math.floor(Math.random() * valid.length)];
    }
  }
}

    gameGrid[pacY][pacX] = 2; powerPelletsActive++; score += 50; invincibleTimer = 300;
    playSound('power');

}

    else if (type === 'power') { osc.frequency.value = 600; osc.type='square'; gain.gain.value=0.15; osc.start(); osc.stop(ac.currentTime+0.2); }
    else if (type === 'eat') { osc.frequency.value = 800; osc.type='sawtooth'; gain.gain.value=0.2; osc.start(); osc.stop(ac.currentTime+0.3); }
    else if (type === 'die') { osc.frequency.value = 150; osc.type='sawtooth'; gain.gain.value=0.3; osc.start(); osc.stop(ac.currentTime+0.5); }
    else if (type === 'lap') { osc.frequency.value = 500; osc.type='triangle'; gain.gain.value=0.2; osc.start(); osc.stop(ac.currentTime+0.15); }
    else if (type === 'win') { osc.frequency.value = 800; osc.type='square'; gain.gain.value=0.2; osc.start(); osc.stop(ac.currentTime+0.5); }
  } catch(e) {}
}

    } else {
      ctx.fillStyle = '#f00';
      ctx.font = `bold ${Math.max(24,t*1.5)}px monospace`;
      ctx.fillText('GAME OVER', cx, cy - t*2);
      ctx.fillStyle = '#fff';
      ctx.font = `${Math.max(14,t*0.7)}px monospace`;
      ctx.fillText(`Time: ${formatTime(elapsedTime)}`, cx, cy - t);
      ctx.fillText(`Score: ${score}`, cx, cy);
    }
    ctx.fillStyle = '#fff';
    ctx.font = `${Math.max(12,t*0.55)}px monospace`;
    if (animFrame % 40 < 25) ctx.fillText('PRESS R TO RESTART', cx, cy + t*2);
  }
}

function loop() {
  animFrame++;
  if (gameState === 'playing') {
    elapsedTime = performance.now() - startTime;
    updatePac();
    updateGhosts();
  }
  draw();
  requestAnimationFrame(loop);
}

resetGame();
loop();
