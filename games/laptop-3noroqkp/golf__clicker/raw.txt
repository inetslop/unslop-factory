function playSound(type) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    if (type === 'hit') {
        osc.type = 'sine';
        osc.frequency.setValueAtTime(200, audioCtx.currentTime);
        osc.frequency.exponentialRampToValueAtTime(80, audioCtx.currentTime + 0.15);
        gain.gain.setValueAtTime(0.3, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.15);
        osc.start(audioCtx.currentTime);
        osc.stop(audioCtx.currentTime + 0.15);
    } else if (type === 'cup') {
        osc.type = 'sine';
        osc.frequency.setValueAtTime(400, audioCtx.currentTime);
        osc.frequency.exponentialRampToValueAtTime(800, audioCtx.currentTime + 0.2);
        gain.gain.setValueAtTime(0.25, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.3);
        osc.start(audioCtx.currentTime);
        osc.stop(audioCtx.currentTime + 0.3);

        osc.type = 'triangle';

function getMousePos(e) {
    const rect = canvas.getBoundingClientRect();
    const clientX = e.touches ? e.touches[0].clientX : e.clientX;
    const clientY = e.touches ? e.touches[0].clientY : e.clientY;
    return { x: (clientX - rect.left) * (W / rect.width), y: (clientY - rect.top) * (H / rect.height) };
}

function onDown(e) {
    if (gameState !== 'playing') return;
    initAudio();
    const pos = getMousePos(e);
    const dx = pos.x - ball.x;
    const dy = pos.y - ball.y;
    if (Math.sqrt(dx*dx + dy*dy) < 50 && Math.abs(ball.vx) < 1 && Math.abs(ball.vy) < 1) {
        isDragging = true;
        dragStart = { x: ball.x, y: ball.y };
        dragCurrent = { ...pos };
    }
}

function onMove(e) {
    if (!isDragging) return;
    dragCurrent = getMousePos(e);
}

function onUp(e) {
    if (!isDragging) return;
    isDragging = false;
    const dx = dragStart.x - dragCurrent.x;
    const dy = dragStart.y - dragCurrent.y;
    const dist = Math.sqrt(dx*dx + dy*dy);
    if (dist > 15) {
        const power = Math.min(dist / 10, 18);
        ball.vx = dx / dist * power;
        ball.vy = dy / dist * power;
        strokes++;
        totalStrokes++;
        playSound('hit');
        spawnParticles(ball.x, ball.y, '#fff', 5);
        updateScoreBar();
    }
}

function spawnParticles(x, y, color, count) {
    for (let i = 0; i < count; i++) {
        const angle = Math.random() * Math.PI * 2;
        const speed = 1 + Math.random() * 3;
        particles.push({ x, y, vx: Math.cos(angle)*speed, vy: Math.sin(angle)*speed, life: 1, color });
    }
}

function isOnSand(x, y) {
    const h = holes[holeIndex];
    for (const s of h.sand) {
        if (x > s.x*W && x < (s.x+s.w)*W && y > s.y*H && y < (s.y+s.h)*H) return true;
    }
    return false;
}

function isOnWater(x, y) {
    const h = holes[holeIndex];
    for (const w of h.water) {
        if (x > w.x*W && x < (w.x+w.w)*W && y > w.y*H && y < (w.y+w.h)*H) return true;
    }
    return false;
}

function checkWalls() {
    const h = holes[holeIndex];
    for (const wall of h.walls) {
        const wx = wall.x * W, wy = wall.y * H, ww = wall.w * W, wh = wall.h * H;
        // Ball vs AABB collision
        const closestX = Math.max(wx, Math.min(ball.x, wx + ww));
        const closestY = Math.max(wy, Math.min(ball.y, wy + wh));
        const dx = ball.x - closestX;
        const dy = ball.y - closestY;
        const dist = Math.sqrt(dx*dx + dy*dy);
        if (dist < ball.r) {
            playSound('wall');
            // Determine which side
            const overlapX = ball.r - Math.abs(dx);
            const overlapY = ball.r - Math.abs(dy);
            if (overlapX > overlapY) {
                ball.vx = -ball.vx * 0.7;
                ball.x += (dx > 0 ? overlapX : -overlapX);
            } else {
                ball.vy = -ball.vy * 0.7;
                ball.y += (dy > 0 ? overlapY : -overlapY);
            }
        }
    }
}

function update(dt) {
    if (gameState !== 'playing') return;
    
    const speed = Math.sqrt(ball.vx*ball.vx + ball.vy*ball.vy);
    
    if (speed > 0.1) {
        inSand = isOnSand(ball.x, ball.y);
        let friction = 0.985;
        if (inSand) friction = 0.94;
        
        ball.vx *= friction;
        ball.vy *= friction;
        
        // Wind
        ball.vx += windX * 0.02;
        ball.vy += windY * 0.02;
        
        ball.x += ball.vx;
        ball.y += ball.vy;
        
        // Boundary collision
        if (ball.x < ball.r) { ball.x = ball.r; ball.vx *= -0.7; playSound('wall'); }
        if (ball.x > W - ball.r) { ball.x = W - ball.r; ball.vx *= -0.7; playSound('wall'); }
        if (ball.y < ball.r) { ball.y = ball.r; ball.vy *= -0.7; playSound('wall'); }
        if (ball.y > H - ball.r) { ball.y = H - ball.r; ball.vy *= -0.7; playSound('wall'); }
        
        checkWalls();
        
        // Water check
        if (isOnWater(ball.x, ball.y)) {
            playSound('water');
            spawnParticles(ball.x, ball.y, '#4fc3f7', 10);
            ball.vx = 0; ball.vy = 0;
            const h = holes[holeIndex];
            ball.x = h.ballStart[0] * W;
            ball.y = h.ballStart[1] * H;
            totalStrokes++; // penalty
            updateScoreBar();
        }
        
        if (inSand && speed > 2) {
            playSound('sand');
        }
    } else {
        ball.vx = 0;
        ball.vy = 0;
    }
    
    // Cup check
    const cupDist = Math.sqrt((ball.x - cup.x)**2 + (ball.y - cup.y)**2);
    if (cupDist < cup.r * 0.7 && speed > 1 && speed < 12) {
        ballInCupAnim += dt;
        if (ballInCupAnim > 0.5) {
            ball.x = cup.x;
            ball.y = cup.y;
            ball.vx = 0; ball.vy = 0;
            playSound('cup');
            spawnParticles(cup.x, cup.y, '#ffd700', 15);
            gameState = 'holeDone';
            showHoleComplete();
        }
    } else {
        ballInCupAnim = 0;
    }
    
    // Particles
    for (let i = particles.length - 1; i >= 0; i--) {
        particles[i].x += particles[i].vx;
        particles[i].y += particles[i].vy;
        particles[i].life -= dt * 2;
        if (particles[i].life <= 0) particles.splice(i, 1);
    }
}

function showHoleComplete() {
    const par = holes[holeIndex].par;

    else if (strokes < par) { result = `🎉 Under Par! (${par - strokes} under)`; }
    else if (strokes === par) { result = '✅ Par!'; }
    else if (strokes === par + 1) { result = '➕ One Over Par'; }
    else { result = `📈 ${strokes - par} Over Par`; }
    
    document.getElementById('holeResult').textContent = result;
    document.getElementById('holeDetail').textContent = `Hole ${holeIndex+1}: ${strokes} strokes (Par ${par}) | Total: ${totalStrokes}`;
    holeComplete.classList.remove('hidden');
}

        const dx = dragStart.x - dragCurrent.x;
        const dy = dragStart.y - dragCurrent.y;
        const dist = Math.sqrt(dx*dx + dy*dy);
        const power = Math.min(dist / 10, 18);
        
        // Direction line
        ctx.beginPath();
        ctx.moveTo(ball.x, ball.y);
        ctx.lineTo(ball.x + dx * 0.5, ball.y + dy * 0.5);
        ctx.strokeStyle = 'rgba(255,255,255,0.6)';
        ctx.lineWidth = 3;
        ctx.setLineDash([8, 6]);
        ctx.stroke();
        ctx.setLineDash([]);
        
        // Power bar
        const barX = ball.x - 25;
        const barY = ball.y - 30;
        ctx.fillStyle = 'rgba(0,0,0,0.5)';
        ctx.fillRect(barX - 2, barY - 2, 54, 12);
        const pct = power / 18;
        const r = Math.floor(255 * pct);
        const g = Math.floor(255 * (1 - pct));
        ctx.fillStyle = `rgb(${r},${g},0)`;
        ctx.fillRect(barX, barY, 50 * pct, 8);
        
        // Power text
        ctx.fillStyle = '#fff';
        ctx.font = '12px monospace';
        ctx.textAlign = 'center';
        ctx.fillText(`${Math.round(pct*100)}%`, ball.x, ball.y - 35);
    }
    
    // Particles
    for (const p of particles) {
        ctx.globalAlpha = p.life;
        ctx.fillStyle = p.color;
        ctx.beginPath();
        ctx.arc(p.x, p.y, 4 * p.life, 0, Math.PI * 2);
        ctx.fill();
    }
    ctx.globalAlpha = 1;
    
    // Wind indicator
    if (Math.abs(windX) > 0 || Math.abs(windY) > 0) {
        const wx = W - 50, wy = H - 30;
        ctx.fillStyle = 'rgba(0,0,0,0.4)';
        ctx.beginPath();
        ctx.arc(wx, wy, 20, 0, Math.PI*2);
        ctx.fill();
        const angle = Math.atan2(windY, windX);
        ctx.strokeStyle = '#fff';
        ctx.lineWidth = 2;
        ctx.beginPath();
        ctx.moveTo(wx - Math.cos(angle)*12, wy - Math.sin(angle)*12);
        ctx.lineTo(wx + Math.cos(angle)*12, wy + Math.sin(angle)*12);
        ctx.stroke();
        // Arrow head
        ctx.beginPath();
        ctx.moveTo(wx + Math.cos(angle)*12, wy + Math.sin(angle)*12);
        ctx.lineTo(wx + Math.cos(angle)*12 - Math.cos(angle-Math.PI/4)*6, wy + Math.sin(angle)*12 - Math.sin(angle-Math.PI/4)*6);
        ctx.moveTo(wx + Math.cos(angle)*12, wy + Math.sin(angle)*12);
        ctx.lineTo(wx + Math.cos(angle)*12 - Math.cos(angle+Math.PI/4)*6, wy + Math.sin(angle)*12 - Math.sin(angle+Math.PI/4)*6);
        ctx.stroke();
        
        ctx.fillStyle = '#fff';
        ctx.font = '10px sans-serif';
        ctx.textAlign = 'center';
        ctx.fillText('WIND', wx, wy + 32);
    }
    
    // Hole number display on course
    ctx.fillStyle = 'rgba(255,255,255,0.15)';
    ctx.font = `bold ${Math.min(W,H)/8}px sans-serif`;
    ctx.textAlign = 'center';
    ctx.textBaseline = 'middle';
    ctx.fillText(`${holeIndex + 1}`, W/2, H/2);
}

let lastTime = 0;
function gameLoop(time) {
    const dt = Math.min((time - lastTime) / 1000, 0.05);
    lastTime = time;
    
    update(dt);
    draw();
    
    requestAnimationFrame(gameLoop);
}

requestAnimationFrame(gameLoop);
