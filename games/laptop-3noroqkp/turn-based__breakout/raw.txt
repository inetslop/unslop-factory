<script>
const canvas = document.getElementById('game');
const ctx = canvas.getContext('2d');
let W, H, cellW, cellH;
const COLS = 13, ROWS = 12;
const BRICK_ROWS = 5;

let state = 'start'; // start, player_turn, ball_moving, game_over, win
let score = 0, lives = 3;
let bricks = [];
let paddleCol = 6, paddleWidth = 3;
let ball = { col: 6, row: 10, dx: 0, dy: -1 };
let ballAtPaddle = true;
let animFrame = 0;
let animTimer = null;
let particles = [];

function resize() {
    const maxW = window.innerWidth;
    const maxH = window.innerHeight;
    const aspect = COLS / ROWS;
    if (maxW / maxH > aspect) { H = maxH; W = H * aspect; }
    else { W = maxW; H = W / aspect; }
    canvas.width = W;
    canvas.height = H;
    cellW = W / COLS;
    cellH = H / ROWS;
}

function initBricks() {
    bricks = [];
    for (let r = 1; r <= BRICK_ROWS; r++) {
        for (let c = 0; c < COLS; c++) {
            if (Math.random() > 0.15 || r === 1) {
                bricks.push({ col: c, row: r, hp: r <= 2 ? 2 : 1, maxHp: r <= 2 ? 2 : 1 });
            }
        }
    }
}

function getBrickAt(c, r) {
    return bricks.find(b => b.col === c && b.row === r);
}

function startGame() {
    score = 0; lives = 3;
    paddleCol = 6;
    ballAtPaddle = true;
    ball = { col: 6, row: 10, dx: 0, dy: -1 };
    initBricks();
    state = 'player_turn';
    updateHUD();
    showTurnActions();
}

function updateHUD() {
    document.getElementById('score').textContent = score;
    document.getElementById('lives').textContent = lives;
    document.getElementById('bricks').textContent = bricks.length;
}

function showTurnActions() {
    const actionsDiv = document.getElementById('actions');
    actionsDiv.innerHTML = '';
    const indicator = document.getElementById('turn-indicator');
    
    if (ballAtPaddle) {
        indicator.textContent = 'SERVE: Choose angle';
        addBtn(actionsDiv, '⟵ Up-Left', () => serveBall(-1, -1));
        addBtn(actionsDiv, '↑ Up', () => serveBall(0, -1));
        addBtn(actionsDiv, 'Up-Right ⟶', () => serveBall(1, -1));
    } else {
        indicator.textContent = 'POSITION: Move paddle';
        addBtn(actionsDiv, '← Left', () => movePaddle(-1));
        addBtn(actionsDiv, '→ Right', () => movePaddle(1));
    }
}

function addBtn(parent, label, fn) {
    const btn = document.createElement('button');
    btn.className = 'action-btn';
    btn.textContent = label;
    btn.onclick = fn;
    parent.appendChild(btn);
}

function serveBall(dx, dy) {
    ball.dx = dx;
    ball.dy = dy;
    ballAtPaddle = false;
    state = 'ball_moving';
    document.getElementById('turn-indicator').textContent = 'BALL IN MOTION...';
    animateBall();
}

function movePaddle(dir) {
    const newCol = paddleCol + dir;
    if (newCol < 1 || newCol > COLS - 2) return;
    paddleCol = newCol;
    state = 'ball_moving';
    document.getElementById('turn-indicator').textContent = 'BALL IN MOTION...';
    animateBall();
}

function animateBall() {
    animTimer = setInterval(() => {
        stepBall();
        draw();
        if (state !== 'ball_moving') {
            clearInterval(animTimer);
        }
    }, 120);
}

function stepBall() {
    ball.col += ball.dx;
    ball.row += ball.dy;
    
    // Check bounds and collisions
    // Top wall
    if (ball.row <= 0) {
        ball.row = 1;
        ball.dy = -ball.dy;
        spawnParticles(ball.col, 0);
    }
    // Side walls
    if (ball.col <= 0) {
        ball.col = 1;
        ball.dx = -ball.dx;
        spawnParticles(0, ball.row);
    }
    if (ball.col >= COLS - 1) {
        ball.col = COLS - 2;
        ball.dx = -ball.dx;
    }
    
    // Check brick collision
    const brick = getBrickAt(ball.col, ball.row);
    if (brick) {
        brick.hp--;
        spawnParticles(ball.col, ball.row);
        if (brick.hp <= 0) {
            bricks.splice(bricks.indexOf(brick), 1);
            score += brick.maxHp * 10;
        } else {
            score += 5;
        }
        // Bounce: reverse both components for grid simplicity
        ball.dx = -ball.dx;
        ball.dy = -ball.dy;
        updateHUD();
    }
    
    // Check paddle row
    if (ball.row >= 10) {
        const pL = paddleCol, pR = paddleCol + paddleWidth - 1;
        if (ball.col >= pL && ball.col <= pR) {
            // Bounce off paddle
            ballAtPaddle = true;
            ball.row = 10;
            state = 'player_turn';
            document.getElementById('turn-indicator').textContent = 'PAUSE: Re-serve';
            showTurnActions();
            updateHUD();
            return;
        } else {
            // Miss! Lose life
            lives--;
            updateHUD();
            if (lives <= 0) {
                state = 'game_over';
                document.getElementById('msg').innerHTML = 'GAME OVER<small>Press R to restart</small>';
                document.getElementById('turn-indicator').textContent = '';
                document.getElementById('actions').innerHTML = '';
            } else {
                ballAtPaddle = true;
                ball.col = paddleCol + 1;
                ball.row = 10;
                state = 'player_turn';
                showTurnActions();
                document.getElementById('turn-indicator').textContent = 'PAUSE: Re-serve';
            }
            return;
        }
    }
    
    // Check win
    if (bricks.length === 0) {
        state = 'win';
        score += 100;
        updateHUD();
        document.getElementById('msg').innerHTML = 'LEVEL CLEARED!<small>Press R for next level</small>';
        document.getElementById('turn-indicator').textContent = '';
        document.getElementById('actions').innerHTML = '';
    }
}

function spawnParticles(c, r) {
    for (let i = 0; i < 6; i++) {
        particles.push({
            x: c * cellW + cellW/2,
            y: r * cellH + cellH/2,
            vx: (Math.random()-0.5)*4,
            vy: (Math.random()-0.5)*4,
            life: 30,
            color: `hsl(${180+Math.random()*60}, 80%, 70%)`
        });
    }
}

function draw() {
    ctx.fillStyle = '#0a0e1a';
    ctx.fillRect(0, 0, W, H);
    
    // Draw grid (subtle)
    ctx.strokeStyle = 'rgba(50,80,120,0.15)';
    ctx.lineWidth = 1;
    for (let c = 0; c <= COLS; c++) {
        ctx.beginPath(); ctx.moveTo(c*cellW, 0); ctx.lineTo(c*cellW, H); ctx.stroke();
    }
    for (let r = 0; r <= ROWS; r++) {
        ctx.beginPath(); ctx.moveTo(0, r*cellH); ctx.lineTo(W, r*cellH); ctx.stroke();
    }
    
    // Draw top wall
    ctx.fillStyle = 'rgba(60,120,200,0.3)';
    ctx.fillRect(0, 0, W, cellH * 0.3);
    
    // Draw bricks
    for (const b of bricks) {
        const x = b.col * cellW + 2;
        const y = b.row * cellH + 2;
        const w = cellW - 4;
        const h = cellH - 4;
        const hue = (b.row - 1) * 30 + 180;
        ctx.fillStyle = `hsla(${hue}, 70%, ${b.hp >= b.maxHp ? 55 : 35}%, 0.9)`;
        ctx.fillRect(x, y, w, h);
        ctx.strokeStyle = `hsla(${hue}, 80%, 70%, 0.6)`;
        ctx.lineWidth = 1;
        ctx.strokeRect(x, y, w, h);
        if (b.maxHp > 1) {
            ctx.fillStyle = 'rgba(255,255,255,0.3)';
            ctx.font = `${cellH*0.4}px sans-serif`;
            ctx.textAlign = 'center';
            ctx.fillText(b.hp, x + w/2, y + h*0.65);
        }
    }
    
    // Draw paddle
    const px = (paddleCol) * cellW;
    const py = 10 * cellH;

        ctx.strokeStyle = 'rgba(150,230,255,0.9)';

    // Draw ball

        ctx.arc(bx, by, Math.min(cellW,cellH)*0.5, 0, Math.PI*2);
        ctx.strokeStyle = 'rgba(255,255,100,0.3)';
        ctx.lineWidth = 2;
        ctx.stroke();
    }
    
    // Draw direction preview when at paddle

        ctx.strokeStyle = 'rgba(255,255,100,0.4)';
        ctx.lineWidth = 2;
        for (const [dx,dy] of [[-1,-1],[0,-1],[1,-1]]) {
            const bx = ball.col * cellW + cellW/2;
            const by = ball.row * cellH + cellH/2;
            ctx.beginPath();
            ctx.moveTo(bx, by);
            ctx.lineTo(bx + dx*cellW*3, by + dy*cellH*3);
            ctx.stroke();
        }
        ctx.setLineDash([]);
    }
    
    // Particles
    for (let i = particles.length-1; i >= 0; i--) {

        ctx.textAlign = 'center';
        ctx.fillText('TURN BREAKOUT', W/2, H*0.3);
        ctx.font = `${Math.min(W,H)*0.035}px sans-serif`;
        ctx.fillStyle = '#8cf';
        ctx.fillText('A strategic grid-based breakout', W/2, H*0.4);
        ctx.fillStyle = '#aac';
        ctx.font = `${Math.min(W,H)*0.028}px sans-serif`;
        ctx.fillText('• Ball moves in 8 directions (like chess queen)', W/2, H*0.52);

    if (ballAtPaddle) {
        if (e.key === 'ArrowUp' || e.key === 'w') serveBall(0, -1);
        else if (e.key === 'q' || (e.shiftKey && e.key === 'ArrowLeft')) serveBall(-1, -1);
        else if (e.key === 'e' || (e.shiftKey && e.key === 'ArrowRight')) serveBall(1, -1);
    } else {
        if (e.key === 'ArrowLeft' || e.key === 'a') movePaddle(-1);
        else if (e.key === 'ArrowRight' || e.key === 'd') movePaddle(1);
    }
});

canvas.addEventListener('click', () => {
    if (state === 'start') startGame();
});

document.getElementById('msg').addEventListener('click', () => {
    if (state === 'game_over' || state === 'win') startGame();
});

draw();
