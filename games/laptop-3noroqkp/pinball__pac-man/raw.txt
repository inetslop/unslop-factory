function playSound(freq, dur, type, vol) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = type || 'square';
    osc.frequency.value = freq;
    gain.gain.setValueAtTime(vol || 0.1, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + dur);
}

const GRAVITY = 0.35;
const DAMPING = 0.998;
const MAX_SPEED = 12;

function update() {
    if (state !== 'PLAYING') return;
    animTimer++;

    // Flippers
    const lPressed = keys['KeyA'] || keys['ArrowLeft'];
    const rPressed = keys['KeyD'] || keys['ArrowRight'];

    if (lPressed) leftFlipper.angle -= FLIPPER_SPEED;
    else leftFlipper.angle += FLIPPER_SPEED * 0.6;
    leftFlipper.angle = Math.max(leftFlipper.activeAngle, Math.min(leftFlipper.angle, leftFlipper.restAngle));

    if (rPressed) rightFlipper.angle += FLIPPER_SPEED;
    else rightFlipper.angle -= FLIPPER_SPEED * 0.6;
    rightFlipper.angle = Math.min(rightFlipper.activeAngle, Math.max(rightFlipper.angle, rightFlipper.restAngle));

    // Ball physics
    if (launcherActive) {
        ball.vy += GRAVITY * 0.3;
    } else {
        ball.vy += GRAVITY;
    }
    ball.vx *= DAMPING;
    ball.vy *= DAMPING;
    const speed = Math.sqrt(ball.vx*ball.vx + ball.vy*ball.vy);
    if (speed > MAX_SPEED) {
        ball.vx = (ball.vx / speed) * MAX_SPEED;
        ball.vy = (ball.vy / speed) * MAX_SPEED;
    }
    ball.x += ball.vx;
    ball.y += ball.vy;

    // Wall collisions
    for (let w of walls) {
        collideWall(w);
    }

    // Bumper collisions
    for (let i = 0; i < bumpers.length; i++) {
        const b = bumpers[i];
        const dx = ball.x - b.x;
        const dy = ball.y - b.y;
        const dist = Math.sqrt(dx*dx + dy*dy);
        if (dist < BALL_R + b.r) {
            const nx = dx / dist, ny = dy / dist;
            ball.x = b.x + nx * (BALL_R + b.r);
            ball.y = b.y + ny * (BALL_R + b.r);
            const dot = ball.vx * nx + ball.vy * ny;
            ball.vx -= 2 * dot * nx;
            ball.vy -= 2 * dot * ny;
            ball.vx += nx * 4;
            ball.vy += ny * 4;
            score += 50;
            sfxBumper();
            spawnParticles(ball.x, ball.y, bumperColors[i], 8);
        }
    }

    // Flipper collisions
    collideFlipper(leftFlipper);
    collideFlipper(rightFlipper);

    // Dot collection
    for (let d of dots) {
        if (!d.eaten) {
            const dx = ball.x - d.x;
            const dy = ball.y - d.y;
            if (dx*dx + dy*dy < (BALL_R + d.r) * (BALL_R + d.r)) {
                d.eaten = true;
                dotsEaten++;
                score += 10;
                sfxDot();
                spawnParticles(d.x, d.y, '#FFE66D', 5);
            }
        }
    }

    // Ball lost
    if (ball.y > GAME_H + 20) {
        lives--;
        sfxLose();
        if (lives <= 0 || dotsEaten >= totalDots) {
            state = 'GAME_OVER';
            if (score > highScore) {
                highScore = score;
                try { localStorage.setItem('pinpac_high', highScore.toString()); } catch(e){}
            }
        } else {
            resetBall();
        }
    }

    // All dots eaten - next level
    if (dotsEaten >= totalDots && state === 'PLAYING') {
        score += 100;
        spawnDots();
        dotsEaten = 0;
    }

    // Particles
    for (let i = particles.length - 1; i >= 0; i--) {
        particles[i].x += particles[i].vx;
        particles[i].y += particles[i].vy;
        particles[i].vy += 0.1;
        particles[i].life--;
        if (particles[i].life <= 0) particles.splice(i, 1);
    }
}

function collideWall(w) {
    const dx = w.x2 - w.x1;
    const dy = w.y2 - w.y1;
    const len = Math.sqrt(dx*dx + dy*dy);
    if (len === 0) return;
    const nx = -dy / len, ny = dx / len;

    // Closest point on segment to ball
    const t = Math.max(0, Math.min(1, ((ball.x - w.x1)*dx + (ball.y - w.y1)*dy) / (len*len)));
    const cx = w.x1 + t * dx;
    const cy = w.y1 + t * dy;
    const ddx = ball.x - cx;
    const ddy = ball.y - cy;
    const dist = Math.sqrt(ddx*ddx + ddy*ddy);

    if (dist < BALL_R) {
        const nnx = ddx / dist, nny = ddy / dist;
        ball.x = cx + nnx * BALL_R;
        ball.y = cy + nny * BALL_R;
        const dot = ball.vx * nnx + ball.vy * nny;
        if (dot < 0) {
            ball.vx -= 2 * dot * nnx;
            ball.vy -= 2 * dot * nny;
        }
        sfxBounce();
    }
}

function collideFlipper(f) {
    const tipX = f.pivotX + Math.cos(f.angle) * FLIPPER_LEN;
    const tipY = f.pivotY + Math.sin(f.angle) * FLIPPER_LEN;
    const dx = tipX - f.pivotX, dy = tipY - f.pivotY;
    const len = Math.sqrt(dx*dx + dy*dy);
    if (len === 0) return;

    const t = Math.max(0, Math.min(1, ((ball.x - f.pivotX)*dx + (ball.y - f.pivotY)*dy) / (len*len)));
    const cx = f.pivotX + t * dx;
    const cy = f.pivotY + t * dy;
    const ddx = ball.x - cx;
    const ddy = ball.y - cy;
    const dist = Math.sqrt(ddx*ddx + ddy*ddy);

    if (dist < BALL_R + 5) {
        const nnx = ddx / dist, nny = ddy / dist;
        ball.x = cx + nnx * (BALL_R + 5);
        ball.y = cy + nny * (BALL_R + 5);
        const dot = ball.vx * nnx + ball.vy * nny;
        if (dot < 0) {
            ball.vx -= 2 * dot * nnx;
            ball.vy -= 2 * dot * nny;
        }
        // Add flipper impulse
        const fvx = (f.angle - (f._prevAngle || f.angle)) * FLIPPER_LEN * 0.5;
        ball.vy -= Math.abs(fvx) * 0.8;
        sfxBounce();
    }
    f._prevAngle = f.angle;
}

function drawGhost(x, y, r, color, t) {
    const bob = Math.sin(t * 0.08) * 3;
    ctx.fillStyle = color;

        const wx = x + r - (i * r / 3);
        const wy = y + bob + r * 0.6 + (i % 2 === 0 ? -5 : 5);
        ctx.lineTo(wx, wy);

    ctx.closePath();
    ctx.fill();

    // Eyes
    ctx.fillStyle = '#fff';
    ctx.beginPath();
    ctx.arc(x - r*0.3, y + bob - r*0.2, r*0.25, 0, Math.PI*2);
    ctx.arc(x + r*0.3, y + bob - r*0.2, r*0.25, 0, Math.PI*2);

}
