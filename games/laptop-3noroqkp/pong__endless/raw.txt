```html
<!DOCTYPE html>
<html lang="en">
<head>
<meta charset="UTF-8">
<meta name="viewport" content="width=device-width, initial-scale=1.0">
<title>Endless Pong</title>
<style>
* { margin: 0; padding: 0; box-sizing: border-box; }
html, body { width: 100%; height: 100%; overflow: hidden; background: #0a0a1a; }
canvas { display: block; width: 100vw; height: 100vh; }
</style>
</head>
<body>
<canvas id="game"></canvas>
<script>
(function() {
const canvas = document.getElementById('game');
const ctx = canvas.getContext('2d');

let W, H;
function resize() {
W = window.innerWidth;
H = window.innerHeight;
canvas.width = W;
canvas.height = H;
}
window.addEventListener('resize', resize);
resize();

// Audio
let audioCtx = null;
function initAudio() {
if (!audioCtx) {
audioCtx = new (window.AudioContext || window.webkitAudioContext)();
}
}
function playSound(freq, duration, type, volume) {
if (!audioCtx) return;
const osc = audioCtx.createOscillator();
const gain = audioCtx.createGain();
osc.type = type || 'square';
osc.frequency.value = freq;
gain.gain.setValueAtTime(volume || 0.15, audioCtx.currentTime);
gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + duration);
osc.connect(gain);
gain.connect(audioCtx.destination);
osc.start();
osc.stop(audioCtx.currentTime + duration);
}

// Game state
let state = 'start'; // start, playing, gameover
let score = 0;
let highScore = 0;
try {
highScore = parseInt(localStorage.getItem('endlessPongHigh') || '0');
} catch(e) { highScore = 0; }

// Paddle and ball
const PADDLE_W = 14;
const PADDLE_H = 90;
const BALL_SIZE = 12;
let playerY, aiY, ballX, ballY, ballVX, ballVY;
let baseSpeed, maxSpeed, speedMultiplier;
let aiSpeed, aiReactivity;
let particles = [];
let flashTimer = 0;
let trailPoints = [];

function resetGame() {
score = 0;
baseSpeed = Math.max(W, H) * 0.004;
maxSpeed = baseSpeed * 5;
speedMultiplier = 1;
aiSpeed = 3.5;
aiReactivity = 0.6;
playerY = H / 2 - PADDLE_H / 2;
aiY = H / 2 - PADDLE_H / 2;
ballX = W / 2;
ballY = H / 2;
let angle = (Math.random() - 0.5) * Math.PI / 3;
ballVX = baseSpeed * Math.cos(angle);
ballVY = baseSpeed * Math.sin(angle);
particles = [];
trailPoints = [];
flashTimer = 0;
}

function serveBall(direction) {
ballX = W / 2;
ballY = H / 2;
let angle = (Math.random() - 0.5) * Math.PI / 3;
let dir = direction || (Math.random() > 0.5 ? 1 : -1);
ballVX = baseSpeed * speedMultiplier * Math.cos(angle) * dir;
ballVY = baseSpeed * speedMultiplier * Math.sin(angle);
trailPoints = [];
}

// Input
let keys = {};
window.addEventListener('keydown', function(e) {
keys[e.code] = true;
if (state === 'start' && (e.code === 'Space' || e.code === 'Enter')) {
initAudio();
resetGame();
state = 'playing';
} else if (state === 'gameover' && (e.code === 'Space' || e.code === 'Enter')) {
resetGame();
state = 'playing';
}
});
window.addEventListener('keyup', function(e) { keys[e.code] = false; });

// Particles
function spawnParticles(x, y, color, count) {
for (let i = 0; i < count; i++) {
let angle = Math.random() * Math.PI * 2;
let speed = Math.random() * 4 + 1;
particles.push({
x: x, y: y,
vx: Math.cos(angle) * speed,
vy: Math.sin(angle) * speed,
life: 1,
decay: 0.02 + Math.random() * 0.03,
color: color,
size: Math.random() * 4 + 2
});
}
}

function updateParticles() {
for (let i = particles.length - 1; i >= 0; i--) {
let p = particles[i];
p.x += p.vx;
p.y += p.vy;
p.life -= p.decay;
if (p.life <= 0) particles.splice(i, 1);
}
}

function drawParticles() {
for (let p of particles) {
ctx.globalAlpha = p.life;
ctx.fillStyle = p.color;
ctx.beginPath();
ctx.arc(p.x, p.y, p.size * p.life, 0, Math.PI * 2);
ctx.fill();
}
ctx.globalAlpha = 1;
}

// Drawing
function drawBackground() {
ctx.fillStyle = '#0a0a1a';
ctx.fillRect(0, 0, W, H);

// Center line
ctx.strokeStyle = 'rgba(100, 200, 255, 0.15)';
ctx.lineWidth = 2;
ctx.setLineDash([10, 15]);
ctx.beginPath();
ctx.moveTo(W / 2, 0);
ctx.lineTo(W / 2, H);
ctx.stroke();
ctx.setLineDash([]);

// Center circle
ctx.strokeStyle = 'rgba(100, 200, 255, 0.08)';
ctx.lineWidth = 2;
ctx.beginPath();
ctx.arc(W / 2, H / 2, Math.min(W, H) * 0.15, 0, Math.PI * 2);
ctx.stroke();
}

function drawPaddle(x, y, color) {
ctx.shadowColor = color;
ctx.shadowBlur = 15;
ctx.fillStyle = color;
roundRect(ctx, x, y, PADDLE_W, PADDLE_H, 7);
ctx.fill();
ctx.shadowBlur = 0;
}

function roundRect(ctx, x, y, w, h, r) {
ctx.beginPath();
ctx.moveTo(x + r, y);
ctx.lineTo(x + w - r, y);
ctx.quadraticCurveTo(x + w, y, x + w, y + r);
ctx.lineTo(x + w, y + h - r);
ctx.quadraticCurveTo(x + w, y + h, x + w - r, y + h);
ctx.lineTo(x + r, y + h);
ctx.quadraticCurveTo(x, y + h, x, y + h - r);
ctx.lineTo(x, y + r);
ctx.quadraticCurveTo(x, y, x + r, y);
ctx.closePath();
}

function drawBall() {
// Trail
for (let i = 0; i < trailPoints.length; i++) {
let t = trailPoints[i];
let alpha = (i / trailPoints.length) * 0.4;
ctx.globalAlpha = alpha;
ctx.fillStyle = '#ff6644';
ctx.beginPath();
ctx.arc(t.x, t.y, BALL_SIZE * (i / trailPoints.length) * 0.8, 0, Math.PI * 2);
ctx.fill();
}
ctx.globalAlpha = 1;

ctx.shadowColor = '#ff6644';
ctx.shadowBlur = 20;
ctx.fillStyle = '#ff6644';
ctx.beginPath();
ctx.arc(ballX, ballY, BALL_SIZE / 2, 0, Math.PI * 2);
ctx.fill();
ctx.shadowBlur = 0;
}

function drawHUD() {
// Score
ctx.font = 'bold 32px monospace';
ctx.textAlign = 'center';
ctx.fillStyle = 'rgba(100, 200, 255, 0.9)';
ctx.fillText(score, W / 2, 50);

// Speed indicator
let speedPct = Math.min((speedMultiplier - 1) / (maxSpeed / baseSpeed - 1), 1);
ctx.font = '14px monospace';
ctx.fillStyle = 'rgba(255, 100, 68, 0.7)';
let barW = 120;
let barX = W / 2 - barW / 2;
ctx.fillText('SPEED', W / 2, 70);
ctx.strokeStyle = 'rgba(255, 100, 68, 0.3)';
ctx.lineWidth = 1;
ctx.strokeRect(barX, 76, barW, 6);
ctx.fillStyle = '#ff6644';
ctx.fillRect(barX, 76, barW * speedPct, 6);

// High score
ctx.font = '12px monospace';
ctx.textAlign = 'right';
ctx.fillStyle = 'rgba(255, 255, 255, 0.4)';
ctx.fillText('BEST: ' + highScore, W - 20, 30);
}

function drawStartScreen() {
drawBackground();

ctx.textAlign = 'center';
ctx.fillStyle = '#ff6644';
ctx.shadowColor = '#ff6644';
ctx.shadowBlur = 20;
ctx.font = 'bold 72px monospace';
ctx.fillText('ENDLESS', W / 2, H / 2 - 80);
ctx.fillStyle = '#64c8ff';
ctx.shadowColor = '#64c8ff';
ctx.fillText('PONG', W / 2, H / 2 - 10);
ctx.shadowBlur = 0;

ctx.font = '18px monospace';
ctx.fillStyle = 'rgba(255,255,255,0.7)';
ctx.fillText('W / S  or  ↑ / ↓  to move your paddle', W / 2, H / 2 + 60);
ctx.fillText('Ball speeds up every hit — how long can you last?', W / 2, H / 2 + 90);

ctx.font = '24px monospace';
ctx.fillStyle = 'rgba(100, 200, 255, ' + (0.5 + 0.5 * Math.sin(Date.now() / 500)).toFixed(2) + ')';
ctx.fillText('PRESS SPACE TO START', W / 2, H / 2 + 160);

if (highScore > 0) {
ctx.font = '16px monospace';
ctx.fillStyle = 'rgba(255,255,255,0.4)';
ctx.fillText('HIGH SCORE: ' + highScore, W / 2, H / 2 + 210);
}
}

function drawGameOver() {
drawBackground();
updateParticles();
drawParticles();

ctx.textAlign = 'center';
ctx.fillStyle = '#ff3344';
ctx.shadowColor = '#ff3344';
ctx.shadowBlur = 20;
ctx.font = 'bold 56px monospace';
ctx.fillText('GAME OVER', W / 2, H / 2 - 60);
ctx.shadowBlur = 0;

ctx.font = '28px monospace';
ctx.fillStyle = '#64c8ff';
ctx.fillText('SCORE: ' + score, W / 2, H / 2);

if (score >= highScore && score > 0) {
ctx.font = '20px monospace';
ctx.fillStyle = '#ffcc00';
ctx.fillText('★ NEW HIGH SCORE! ★', W / 2, H / 2 + 45);
} else if (highScore > 0) {
ctx.font = '16px monospace';
ctx.fillStyle = 'rgba(255,255,255,0.5)';
ctx.fillText('BEST: ' + highScore, W / 2, H / 2 + 45);
}

ctx.font = '20px monospace';
ctx.fillStyle = 'rgba(255,255,255,' + (0.5 + 0.5 * Math.sin(Date.now() / 500)).toFixed(2) + ')';
ctx.fillText('PRESS SPACE TO RESTART', W / 2, H / 2 + 110);
}

// Update
function update(dt) {
if (state !== 'playing') return;

let dtSec = dt / 1000;

// Player movement
let pSpeed = H * 0.8 * dtSec;
if (keys['KeyW'] || keys['ArrowUp']) playerY -= pSpeed;
if (keys['KeyS'] || keys['ArrowDown']) playerY += pSpeed;
playerY = Math.max(0, Math.min(H - PADDLE_H, playerY));

// AI movement
let targetY = ballY - PADDLE_H / 2;
let diff = targetY - aiY;
let aiMove = diff * aiReactivity * dtSec * 8;
aiY += aiMove;
aiY = Math.max(0, Math.min(H - PADDLE_H, aiY));

// Ball movement
ballX += ballVX * dtSec * 60;
ballY += ballVY * dtSec * 60;

// Trail
trailPoints.push({x: ballX, y: ballY});
if (trailPoints.length > 15) trailPoints.shift();

// Wall bounce
if (ballY - BALL_SIZE / 2 <= 0 || ballY + BALL_SIZE / 2 >= H) {
ballVY = -ballVY;
ballY = Math.max(BALL_SIZE / 2, Math.min(H - BALL_SIZE / 2, ballY));
playSound(300, 0.1, 'square', 0.1);
spawnParticles(ballX, ballY, '#64c8ff', 5);
}

// Paddle collisions
let playerX = 30;
let aiX = W - 30 - PADDLE_W;

// Player paddle
if (ballVX < 0 && ballX - BALL_SIZE / 2 <= playerX + PADDLE_W && ballX + BALL_SIZE / 2 >= playerX) {
if (ballY >= playerY - BALL_SIZE && ballY <= playerY + PADDLE_H + BALL_SIZE) {
let hitPos = (ballY - playerY) / PADDLE_H;
hitPos = Math.max(0, Math.min(1, hitPos));
let angle = (hitPos - 0.5) * Math.PI * 0.7;
speedMultiplier = Math.min(speedMultiplier + 0.08, maxSpeed / baseSpeed);

ballVX = Math.abs(baseSpeed * speedMultiplier) * Math.cos(angle);
ballVY = baseSpeed * speedMultiplier * Math.sin(angle);
ballX = playerX + PADDLE_W + BALL_SIZE / 2;

score++;
playSound(500, 0.12, 'square', 0.2);
spawnParticles(ballX, ballY, '#64c8ff', 10);

// Increase difficulty
aiSpeed += 0.3;
aiReactivity = Math.min(aiReactivity + 0.005, 0.95);
}
}

// AI paddle
if (ballVX > 0 && ballX + BALL_SIZE / 2 >= aiX && ballX - BALL_SIZE / 2 <= aiX + PADDLE_W) {
if (ballY >= aiY - BALL_SIZE && ballY <= aiY + PADDLE_H + BALL_SIZE) {
let hitPos = (ballY - aiY) / PADDLE_H;
hitPos = Math.max(0, Math.min(1, hitPos));
let angle = Math.PI - (hitPos - 0.5) * Math.PI * 0.7;

ballVX = -Math.abs(baseSpeed * speedMultiplier) * Math.cos(angle);
ballVY = baseSpeed * speedMultiplier * Math.sin(angle);
ballX = aiX - BALL_SIZE / 2;

playSound(400, 0.12, 'square', 0.15);
spawnParticles(ballX, ballY, '#ff6644', 8);
}
}

// Score / Game over
if (ballX < -BALL_SIZE) {
// AI scores - not game over for us in endless mode, but let's make it: player loses
state = 'gameover';
if (score > highScore) {
highScore = score;
try { localStorage.setItem('endlessPongHigh', String(highScore)); } catch(e) {}
}
playSound(150, 0.4, 'sawtooth', 0.2);
spawnParticles(ballX + BALL_SIZE, ballY, '#ff3344', 20);
}

if (ballX > W + BALL_SIZE) {
// Player scored! Ball passes AI side
serveBall(-1);
playSound(700, 0.15, 'sine', 0.15);
spawnParticles(W - 10, ballY, '#33ff66', 12);
}

updateParticles();
}

// Main loop
let lastTime = 0;
function gameLoop(timestamp) {
let dt = timestamp - lastTime;
lastTime = timestamp;
if (dt > 100) dt = 16;

if (state === 'start') {
drawStartScreen();
} else if (state === 'playing') {
update(dt);
drawBackground();
drawPaddle(30, playerY, '#64c8ff');
drawPaddle(W - 30 - PADDLE_W, aiY, '#ff6644');
drawBall();
drawParticles();
drawHUD();
} else if (state === 'gameover') {
drawGameOver();
}

requestAnimationFrame(gameLoop);
}

resetGame();
requestAnimationFrame(gameLoop);
})();
</script>
</body>
```
