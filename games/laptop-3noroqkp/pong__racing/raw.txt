// Audio
let audioCtx = null;
function initAudio() {
  if (!audioCtx) audioCtx = new (window.AudioContext || webkitAudioContext)();
}
function playSound(freq, dur, type, vol) {
  if (!audioCtx) return;
  const osc = audioCtx.createOscillator();
  const gain = audioCtx.createGain();
  osc.type = type || 'square';
  osc.frequency.value = freq;
  gain.gain.setValueAtTime(vol || 0.15, audioCtx.currentTime);
  gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
  osc.connect(gain);
  gain.connect(audioCtx.destination);
  osc.start();
  osc.stop(audioCtx.currentTime + dur);
}

// Game state
let gameState = 'menu'; // menu, playing, score, gameOver
let scorePlayer = 0, scoreAI = 0;
let winsPlayer = 0, winsAI = 0;
const WINS_NEEDED = 3;
const POINTS_NEEDED = 5;

// Particles
let particles = [];

class Particle {
  constructor(x, y, vx, vy, color, life) {
    this.x = x; this.y = y; this.vx = vx; this.vy = vy;
    this.color = color; this.life = life; this.maxLife = life;
  }
  update(dt) {
    this.x += this.vx * dt;
    this.y += this.vy * dt;
    this.life -= dt;
  }
  draw() {
    const a = this.life / this.maxLife;
    ctx.globalAlpha = a;
    ctx.fillStyle = this.color;
    ctx.beginPath();
    ctx.arc(this.x, this.y, 3 * a, 0, Math.PI * 2);
    ctx.fill();
    ctx.globalAlpha = 1;
  }
}

function spawnParticles(x, y, color, count) {
  for (let i = 0; i < count; i++) {
    const angle = Math.random() * Math.PI * 2;
    const speed = 50 + Math.random() * 200;
    particles.push(new Particle(x, y, Math.cos(angle) * speed, Math.sin(angle) * speed, color, 0.3 + Math.random() * 0.5));
  }
}

// Ball
let ball = { x: W/2, y: H/2, vx: 0, vy: 0, r: 10, trail: [] };

function resetBall(direction) {
  ball.x = W / 2;
  ball.y = H / 2;
  const angle = (Math.random() - 0.5) * Math.PI * 0.6;
  const speed = 300 + Math.random() * 100;
  ball.vx = Math.sin(angle) * speed;
  ball.vy = Math.cos(angle) * speed * (direction || (Math.random() > 0.5 ? 1 : -1));
  ball.trail = [];
}

let keys = {};

function startGame() {
  scorePlayer = 0;
  scoreAI = 0;

  gameState = 'playing';

// Nitro
let nitroFuel = 100;
let nitroActive = false;
const NITRO_MAX = 100;
const NITRO_DRAIN = 60; // per second
const NITRO_REGEN = 25; // per second

// AI difficulty scaling with score
function getAISpeed() {
  return 200 + (scorePlayer + scoreAI) * 15;
}

function update(dt) {
  if (gameState !== 'playing') return;

  // Nitro
  nitroActive = keys['ShiftLeft'] || keys['ShiftRight'];
  if (nitroActive && nitroFuel > 0) {
    nitroFuel -= NITRO_DRAIN * dt;
    if (nitroFuel < 0) nitroFuel = 0;
  } else {
    nitroFuel += NITRO_REGEN * dt;
    if (nitroFuel > NITRO_MAX) nitroFuel = NITRO_MAX;
  }

  // Player movement
  let pSpeed = nitroActive ? 500 : 320;
  let pVx = 0, pVy = 0;
  if (keys['KeyA'] || keys['ArrowLeft']) pVx -= pSpeed;
  if (keys['KeyD'] || keys['ArrowRight']) pVx += pSpeed;
  if (keys['KeyW'] || keys['ArrowUp']) pVy -= pSpeed;
  if (keys['KeyS'] || keys['ArrowDown']) pVy += pSpeed;

  playerPaddle.x += pVx * dt;
  playerPaddle.y += pVy * dt;
  playerPaddle.x = Math.max(playerPaddle.w/2, Math.min(W - playerPaddle.w/2, playerPaddle.x));
  playerPaddle.y = Math.max(H * 0.55, Math.min(H - 30, playerPaddle.y));

  // AI movement
  let aiSpeed = getAISpeed();
  let dx = ball.x - aiPaddle.x;
  let dy = (ball.y < H/2 ? ball.y : H * 0.3) - aiPaddle.y;
  
  if (Math.abs(dx) > 5 || Math.abs(dy) > 5) {
    aiPaddle.x += Math.sign(dx) * Math.min(aiSpeed, Math.abs(dx) * 3) * dt;
    aiPaddle.y += Math.sign(dy) * Math.min(aiSpeed, Math.abs(dy) * 3) * dt;
  }
  aiPaddle.x = Math.max(aiPaddle.w/2, Math.min(W - aiPaddle.w/2, aiPaddle.x));
  aiPaddle.y = Math.max(30, Math.min(H * 0.45, aiPaddle.y));

  // Ball trail
  ball.trail.push({x: ball.x, y: ball.y});
  if (ball.trail.length > 20) ball.trail.shift();

  // Ball movement
  ball.x += ball.vx * dt;
  ball.y += ball.vy * dt;

  // Wall bounce
  if (ball.x - ball.r < 0) { ball.x = ball.r; ball.vx = Math.abs(ball.vx); playSound(200, 0.1, 'square', 0.1); spawnParticles(ball.x, ball.y, '#ff6600', 5); }
  if (ball.x + ball.r > W) { ball.x = W - ball.r; ball.vx = -Math.abs(ball.vx); playSound(200, 0.1, 'square', 0.1); spawnParticles(ball.x, ball.y, '#ff6600', 5); }

  // Paddle collision
  // Player paddle
  if (ball.vy > 0) {
    let px = playerPaddle.x, py = playerPaddle.y;
    if (ball.x > px - playerPaddle.w/2 && ball.x < px + playerPaddle.w/2 &&
        ball.y + ball.r > py - playerPaddle.h/2 && ball.y - ball.r < py + playerPaddle.h/2) {
      let hitPos = (ball.x - px) / (playerPaddle.w/2);
      ball.vy = -Math.abs(ball.vy) * 1.05;
      ball.vx += hitPos * 200;
      playSound(400, 0.1, 'square', 0.2);
      spawnParticles(ball.x, ball.y, '#00ffff', 8);
    }
  }

  // AI paddle
  if (ball.vy < 0) {
    let ax = aiPaddle.x, ay = aiPaddle.y;
    if (ball.x > ax - aiPaddle.w/2 && ball.x < ax + aiPaddle.w/2 &&
        ball.y - ball.r < ay + aiPaddle.h/2 && ball.y + ball.r > ay - aiPaddle.h/2) {
      let hitPos = (ball.x - ax) / (aiPaddle.w/2);
      ball.vy = Math.abs(ball.vy) * 1.05;
      ball.vx += hitPos * 200;
      playSound(350, 0.1, 'square', 0.2);
      spawnParticles(ball.x, ball.y, '#ff0066', 8);
    }
  }

  // Speed cap
  let maxSpeed = 700;
  let spd = Math.sqrt(ball.vx * ball.vx + ball.vy * ball.vy);
  if (spd > maxSpeed) {
    ball.vx = (ball.vx / spd) * maxSpeed;
    ball.vy = (ball.vy / spd) * maxSpeed;
  }

  // Scoring
  if (ball.y - ball.r > H + 20) {
    scoreAI++;
    playSound(150, 0.4, 'sawtooth', 0.3);
    spawnParticles(ball.x, H, '#ff0000', 20);
    checkScore();
  }
  if (ball.y + ball.r < -20) {
    scorePlayer++;
    playSound(600, 0.4, 'sine', 0.3);
    spawnParticles(ball.x, 0, '#00ff88', 20);
    checkScore();
  }

  // Update particles
  for (let i = particles.length - 1; i >= 0; i--) {
    particles[i].update(dt);
    if (particles[i].life <= 0) particles.splice(i, 1);
  }
}

function checkScore() {
  if (scorePlayer >= POINTS_NEEDED) {
    winsPlayer++;
    if (winsPlayer >= WINS_NEEDED) {
      gameState = 'gameOver';
      message = 'YOU WIN THE CHAMPIONSHIP!';
    } else {
      gameState = 'score';
      message = `RACE ${winsPlayer + 1} - Press SPACE to start`;
    }
  } else if (scoreAI >= POINTS_NEEDED) {
    winsAI++;
    if (winsAI >= WINS_NEEDED) {
      gameState = 'gameOver';
      message = 'AI WINS THE CHAMPIONSHIP!';
    } else {
      gameState = 'score';
      message = `RACE ${winsAI + 1} - Press SPACE to start`;
    }
  }
  if (gameState !== 'playing') {
    scorePlayer = 0;
    scoreAI = 0;
    resetBall(Math.random() > 0.5 ? 1 : -1);
    resetPaddles();
  }
}

function drawCheckered(x, y, w, h) {
  const size = 10;
  for (let i = 0; i < w / size + 1; i++) {
    for (let j = 0; j < h / size + 1; j++) {
      ctx.fillStyle = (i + j) % 2 === 0 ? '#fff' : '#000';
      ctx.fillRect(x + i * size, y + j * size, Math.min(size, w - i * size), Math.min(size, h - j * size));
    }
  }
}

function drawCar(x, y, color, boosted) {
  ctx.save();
  ctx.translate(x, y);
  
  // Body
  ctx.fillStyle = color;
  ctx.shadowColor = color;
  ctx.shadowBlur = boosted ? 20 : 10;
  
  // Car shape
  ctx.beginPath();
  ctx.moveTo(-playerPaddle.w/2, -8);
  ctx.lineTo(-playerPaddle.w/2 + 15, -12);
  ctx.lineTo(playerPaddle.w/2 - 15, -12);
  ctx.lineTo(playerPaddle.w/2, -8);
  ctx.lineTo(playerPaddle.w/2, 8);
  ctx.lineTo(playerPaddle.w/2 - 15, 12);
  ctx.lineTo(-playerPaddle.w/2 + 15, 12);
  ctx.lineTo(-playerPaddle.w/2, 8);
  ctx.closePath();
  ctx.fill();

  // Windshield
  ctx.fillStyle = 'rgba(0,0,0,0.6)';
  ctx.shadowBlur = 0;
  ctx.fillRect(-15, -6, 30, 12);

  // Headlights
  ctx.fillStyle = '#fff';
  ctx.shadowColor = '#fff';
  ctx.shadowBlur = boosted ? 15 : 5;
  ctx.beginPath();
  ctx.arc(-playerPaddle.w/2 + 8, -6, 3, 0, Math.PI * 2);
  ctx.arc(playerPaddle.w/2 - 8, -6, 3, 0, Math.PI * 2);
  ctx.fill();

  // Exhaust flames when boosted
  if (boosted) {
    ctx.fillStyle = '#ff6600';
    ctx.shadowColor = '#ff6600';

      ctx.beginPath();
      ctx.arc(fx, fy, 4 + Math.random() * 4, 0, Math.PI * 2);
      ctx.fill();

  ctx.restore();
}
