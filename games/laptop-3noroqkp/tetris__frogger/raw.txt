```html
<!DOCTYPE html>
<html lang="en">
<head>
<meta charset="UTF-8">
<title>Frogger Tetris</title>
<style>
* { margin: 0; padding: 0; box-sizing: border-box; }
body { background: #0a0a1a; overflow: hidden; width: 100vw; height: 100vh; display: flex; align-items: center; justify-content: center; font-family: monospace; }
canvas { display: block; image-rendering: pixelated; }

<canvas id="c"></canvas>
<script>
const canvas = document.getElementById('c');
const ctx = canvas.getContext('2d');

const COLS = 7, ROWS = 14;
let cellSize, offsetX, offsetY;

function resize() {
    const w = window.innerWidth, h = window.innerHeight;
    canvas.width = w; canvas.height = h;
    cellSize = Math.min(Math.floor(w / (COLS + 2)), Math.floor(h / (ROWS + 4)));
    offsetX = Math.floor((w - COLS * cellSize) / 2);
    offsetY = Math.floor((h - ROWS * cellSize) / 2) - 20;
}
window.addEventListener('resize', resize);
resize();

// Game state
let gameState = 'start'; // start, playing, gameover, win
let frogX = 3, frogY = 0;
let blocks = []; // active falling blocks
let score = 0, highScore = 0;
let fallSpeed = 800; // ms per row
let spawnTimer = 0;
let lastTime = 0;
let particles = [];
let shakeTime = 0;

try { highScore = parseInt(localStorage.getItem('frogTetrisHigh') || '0'); } catch(e) {}

const PIECES = [
    { shape: [[1,1,1,1]], color: '#4a4', name: 'I' },
    { shape: [[1,1],[1,1]], color: '#ee4', name: 'O' },
    { shape: [[0,1,0],[1,1,1]], color: '#c4f', name: 'T' },
    { shape: [[0,1,1],[1,1,0]], color: '#4dd', name: 'S' },
    { shape: [[1,1,0],[0,1,1]], color: '#f44', name: 'Z' },
    { shape: [[1,0,0],[1,1,1]], color: '#48f', name: 'J' },
    { shape: [[0,0,1],[1,1,1]], color: '#fa4', name: 'L' }
];

function spawnPiece() {
    const p = PIECES[Math.floor(Math.random() * PIECES.length)];
    const cells = [];
    for (let r = 0; r < p.shape.length; r++) {
        for (let c = 0; c < p.shape[r].length; c++) {
            if (p.shape[r][c]) cells.push([r, c]);
        }
    }
    const maxCol = COLS - 1;
    let startX = Math.floor(Math.random() * (COLS - 3)) + 1;
    // Clamp to fit
    const width = p.shape[0].length;
    if (startX + width > COLS) startX = COLS - width;
    if (startX < 0) startX = 0;

    for (let [r, c] of cells) {
        blocks.push({ x: startX + c, y: -p.shape.length + r, color: p.color });
    }
}

function resetGame() {
    frogX = 3; frogY = 0;
    blocks = [];
    score = 0;
    fallSpeed = Math.max(400, 800 - Math.floor(score / 50) * 30);
    spawnTimer = 0;
    particles = [];
    shakeTime = 0;
    for (let i = 0; i < 2; i++) spawnPiece();
}

function addParticle(x, y, color) {
    for (let i = 0; i < 6; i++) {
        particles.push({ x: offsetX + x * cellSize + cellSize/2, y: offsetY + y * cellSize + cellSize/2, vx: (Math.random()-0.5)*4, vy: (Math.random()-0.5)*4 - 2, life: 1, color });
    }
}

function checkCollision() {
    for (let b of blocks) {
        if (b.x === frogX && b.y === frogY) return true;
    }
    return false;
}

function update(dt) {
    if (gameState !== 'playing') return;

    spawnTimer += dt;
    const spawnInterval = Math.max(1200, 2500 - score * 10);
    if (spawnTimer >= spawnInterval) {
        spawnTimer = 0;
        spawnPiece();
    }

    // Fall blocks
    fallSpeed = Math.max(300, 800 - Math.floor(score / 40) * 25);
    // We use a simple approach: each block falls at fixed interval via accumulated time
    // Actually let's do it per-frame with delta
    for (let b of blocks) {
        b._timer = (b._timer || 0) + dt;
        if (b._timer >= fallSpeed) {
            b._timer -= fallSpeed;
            b.y++;
        }
    }

    // Remove blocks below grid
    blocks = blocks.filter(b => b.y < ROWS);

    // Collision
    if (checkCollision()) {
        gameState = 'gameover';
        shakeTime = 500;
        addParticle(frogX, frogY, '#4f4');
        try { if (score > highScore) { highScore = score; localStorage.setItem('frogTetrisHigh', highScore); } } catch(e) {}
        playSound(150, 0.3, 'sawtooth');
    }

    // Win check
    if (frogY >= ROWS - 1) {
        gameState = 'win';
        score += 100;
        try { if (score > highScore) { highScore = score; localStorage.setItem('frogTetrisHigh', highScore); } } catch(e) {}
        for (let i = 0; i < 20; i++) addParticle(frogX, frogY, '#ff0');
        playSound(600, 0.5, 'sine');
    }

    // Particles
    for (let p of particles) {
        p.x += p.vx; p.y += p.vy; p.vy += 0.2; p.life -= dt / 1000;
    }
    particles = particles.filter(p => p.life > 0);

    if (shakeTime > 0) shakeTime -= dt;
}

function playSound(freq, dur, type) {
    try {
        const ac = new (window.AudioContext || webkitAudioContext)();
        const osc = ac.createOscillator();
        const gain = ac.createGain();
        osc.type = type || 'sine';
        osc.frequency.value = freq;
        gain.gain.setValueAtTime(0.3, ac.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.01, ac.currentTime + dur);
        osc.connect(gain).connect(ac.destination);
        osc.start();
        osc.stop(ac.currentTime + dur);
    } catch(e) {}
}

function playMoveSound() { playSound(440, 0.08, 'square'); }

let audioStarted = false;
window.addEventListener('keydown', (e) => {
    if (!audioStarted) { audioStarted = true; }

    if (gameState === 'start' || gameState === 'gameover' || gameState === 'win') {
        if (e.key === ' ' || e.key === 'Enter') {
            resetGame();
            gameState = 'playing';
            playMoveSound();
        }
        return;
    }

    if (gameState !== 'playing') return;

    let moved = false;
    if (e.key === 'ArrowLeft' && frogX > 0) { frogX--; moved = true; }
    if (e.key === 'ArrowRight' && frogX < COLS - 1) { frogX++; moved = true; }
    if (e.key === 'ArrowDown' && frogY < ROWS - 1) { frogY++; score += 10; moved = true; }
    if (moved) playMoveSound();

    e.preventDefault();
});

// Also support WASD
window.addEventListener('keydown', (e) => {
    if (gameState !== 'playing') return;
    let moved = false;
    if (e.key === 'a' && frogX > 0) { frogX--; moved = true; }
    if (e.key === 'd' && frogX < COLS - 1) { frogX++; moved = true; }
    if (e.key === 's' && frogY < ROWS - 1) { frogY++; score += 10; moved = true; }
    if (moved) playMoveSound();
});

function drawFrog(x, y) {
    const px = offsetX + x * cellSize;
    const py = offsetY + y * cellSize;
    const s = cellSize;

    // Glow
    ctx.shadowColor = '#4f4';
    ctx.shadowBlur = 10;

    // Body
    ctx.fillStyle = '#2a8a2a';
    ctx.fillRect(px + s*0.2, py + s*0.35, s*0.6, s*0.45);
    // Head
    ctx.fillStyle = '#3cba3b';
    ctx.fillRect(px + s*0.15, py + s*0.2, s*0.7, s*0.3);
    // Eyes
    ctx.fillStyle = '#ff0';
    ctx.fillRect(px + s*0.25, py + s*0.22, s*0.15, s*0.15);
    ctx.fillRect(px + s*0.6, py + s*0.22, s*0.15, s*0.15);
    // Pupils
    ctx.fillStyle = '#000';
    ctx.fillRect(px + s*0.3, py + s*0.27, s*0.07, s*0.07);
    ctx.fillRect(px + s*0.65, py + s*0.27, s*0.07, s*0.07);
    // Legs
    ctx.fillStyle = '#1a6a1a';
    ctx.fillRect(px + s*0.1, py + s*0.7, s*0.15, s*0.2);
    ctx.fillRect(px + s*0.75, py + s*0.7, s*0.15, s*0.2);

    ctx.shadowBlur = 0;
}

function drawBlock(b) {
    const px = offsetX + b.x * cellSize;
    const py = offsetY + b.y * cellSize;
    const s = cellSize;
    const pad = 2;

    ctx.fillStyle = b.color;
    ctx.fillRect(px + pad, py + pad, s - pad*2, s - pad*2);
    // Highlight
    ctx.fillStyle = 'rgba(255,255,255,0.3)';
    ctx.fillRect(px + pad, py + pad, s - pad*2, 4);
    ctx.fillRect(px + pad, py + pad, 4, s - pad*2);
    // Shadow
    ctx.fillStyle = 'rgba(0,0,0,0.3)';
    ctx.fillRect(px + pad, py + s - pad - 4, s - pad*2, 4);
}

function drawGrid() {
    for (let r = 0; r < ROWS; r++) {
        for (let c = 0; c < COLS; c++) {
            const px = offsetX + c * cellSize;
            const py = offsetY + r * cellSize;
            ctx.strokeStyle = 'rgba(50,80,50,0.4)';
            ctx.strokeRect(px, py, cellSize, cellSize);
        }
    }

    // Bottom row - lily pads
    for (let c = 0; c < COLS; c++) {
        const px = offsetX + c * cellSize;
        const py = offsetY + (ROWS-1) * cellSize;
        ctx.fillStyle = 'rgba(34,170,34,0.3)';
        ctx.fillRect(px + 2, py + 2, cellSize - 4, cellSize - 4);
        // Pad shape
        ctx.fillStyle = '#2a8a2a';
        ctx.beginPath();
        ctx.arc(px + cellSize/2, py + cellSize/2, cellSize*0.35, 0, Math.PI*2);
        ctx.fill();
    }

    // Top row - danger zone indicator
    ctx.fillStyle = 'rgba(255,50,50,0.1)';
    ctx.fillRect(offsetX, offsetY, COLS * cellSize, cellSize);
}

function drawHUD() {
    ctx.fillStyle = '#fff';
    ctx.font = `bold ${Math.min(cellSize*0.6, 24)}px monospace`;
    ctx.textAlign = 'left';
    ctx.fillText(`SCORE: ${score}`, offsetX, offsetY - 15);
    ctx.textAlign = 'right';
    ctx.fillText(`BEST: ${highScore}`, offsetX + COLS * cellSize, offsetY - 15);

    // Level indicator
    const level = Math.floor(score / 50) + 1;
    ctx.textAlign = 'center';
    ctx.fillStyle = '#8f8';
    ctx.fillText(`LVL ${level}`, offsetX + COLS * cellSize / 2, offsetY - 15);
}

function drawStartScreen() {
    ctx.fillStyle = 'rgba(0,0,0,0.85)';
    ctx.fillRect(0, 0, canvas.width, canvas.height);

    const cx = canvas.width / 2;
    const cy = canvas.height / 2;

    // Title
    ctx.textAlign = 'center';
    ctx.font = `bold ${Math.min(48, cellSize * 1.5)}px monospace`;
    ctx.fillStyle = '#4f4';
    ctx.shadowColor = '#4f4';
    ctx.shadowBlur = 20;
    ctx.fillText('FROGGER TETRIS', cx, cy - 80);
    ctx.shadowBlur = 0;

    // Subtitle
    ctx.font = `${Math.min(16, cellSize*0.5)}px monospace`;
    ctx.fillStyle = '#aaa';
    ctx.fillText('Help the frog cross the falling Tetris blocks!', cx, cy - 40);

    // Instructions
    ctx.fillStyle = '#8f8';
    ctx.font = `${Math.min(14, cellSize*0.45)}px monospace`;
    const lines = [
        '← → Move Left/Right',
        '↓ Move Down (score points)',
        '',
        'Avoid the falling blocks!',
        'Reach the lily pads at the bottom to WIN.',
        '',
        'Press SPACE or ENTER to start'
    ];
    lines.forEach((l, i) => {
        ctx.fillText(l, cx, cy + 10 + i * 24);
    });

    // Draw a little frog
    const fs = cellSize * 0.8;
    ctx.fillStyle = '#3cba3b';
    ctx.fillRect(cx - fs*0.35, cy + lines.length * 24 + 10, fs*0.7, fs*0.5);
    ctx.fillStyle = '#ff0';
    ctx.fillRect(cx - fs*0.25, cy + lines.length * 24 + 10, fs*0.12, fs*0.12);
    ctx.fillRect(cx + fs*0.13, cy + lines.length * 24 + 10, fs*0.12, fs*0.12);
}

function drawEndScreen(text, color) {
    ctx.fillStyle = 'rgba(0,0,0,0.7)';
    ctx.fillRect(0, 0, canvas.width, canvas.height);

    const cx = canvas.width / 2;
    const cy = canvas.height / 2;

    ctx.textAlign = 'center';
    ctx.font = `bold ${Math.min(40, cellSize * 1.3)}px monospace`;
    ctx.fillStyle = color;
    ctx.shadowColor = color;
    ctx.shadowBlur = 15;
    ctx.fillText(text, cx, cy - 20);
    ctx.shadowBlur = 0;

    ctx.font = `${Math.min(18, cellSize*0.6)}px monospace`;
    ctx.fillStyle = '#fff';
    ctx.fillText(`Score: ${score}`, cx, cy + 30);
    ctx.fillStyle = '#aaa';
    ctx.fillText('Press SPACE to play again', cx, cy + 70);
}

function draw(timestamp) {
    const dt = timestamp - lastTime;
    lastTime = timestamp;

    update(dt);

    // Shake effect
    let sx = 0, sy = 0;
    if (shakeTime > 0) {
        sx = (Math.random() - 0.5) * 6;
        sy = (Math.random() - 0.5) * 6;
    }

    ctx.save();
    ctx.translate(sx, sy);

    // Background
    ctx.fillStyle = '#0a0a1a';
    ctx.fillRect(0, 0, canvas.width, canvas.height);

    if (gameState === 'start') {
        drawGrid();
        drawStartScreen();
    } else {
        drawGrid();

        // Draw blocks
        for (let b of blocks) {
            if (b.y >= 0 && b.y < ROWS) drawBlock(b);
        }

        // Draw frog
        drawFrog(frogX, frogY);

        // Particles
        for (let p of particles) {
            ctx.globalAlpha = p.life;
            ctx.fillStyle = p.color;
            ctx.fillRect(p.x - 3, p.y - 3, 6, 6);
            ctx.globalAlpha = 1;
        }

        drawHUD();

        if (gameState === 'gameover') drawEndScreen('SPLAT! GAME OVER', '#f44');
        if (gameState === 'win') drawEndScreen('LILY PAD REACHED!', '#4f4');
    }

    ctx.restore();

    requestAnimationFrame(draw);
}

// Initial state
resetGame();
gameState = 'start';
requestAnimationFrame((t) => { lastTime = t; requestAnimationFrame(draw); });
</script>
</body>
```
