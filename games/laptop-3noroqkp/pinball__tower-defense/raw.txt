let audioCtx = null;
function initAudio() {
    if (!audioCtx) {
        audioCtx = new (window.AudioContext || window.webkitAudioContext)();
    }
}
function playSound(freq, dur, type, vol) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = type || 'square';
    osc.frequency.value = freq;
    gain.gain.setValueAtTime(vol || 0.1, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + dur);
}

// Game state
let gameState = 'start'; // start, playing, gameover
let score = 0;
let points = 100;
let wave = 0;
let lives = 5;
let enemies = [];
let towers = [];
let projectiles = [];
let particles = [];

function setupTable() {
    // Bumpers

function nextWave() {
    wave++;
    const count = 3 + wave * 2;
    for (let i = 0; i < count; i++) {
        const type = wave > 3 && Math.random() < 0.3 ? 'tank' : (wave > 5 && Math.random() < 0.2 ? 'fast' : 'normal');
        enemies.push({
            x: W * (0.1 + Math.random() * 0.8),

            hp: type === 'tank' ? 5 : 2,

            speed: type === 'fast' ? 1.8 : 0.8 + wave * 0.1,
            radius: type === 'tank' ? 22 : 16,
            type: type,
            hitFlash: 0
        });
    }
}

// Physics
function updateBall() {
    if (!ball || !ball.active) return;
    
    ball.vy += GRAVITY;
    ball.x += ball.vx;
    ball.y += ball.vy;
    
    // Speed cap
    const spd = Math.hypot(ball.vx, ball.vy);
    if (spd > MAX_BALL_SPEED) {
        ball.vx = (ball.vx / spd) * MAX_BALL_SPEED;
        ball.vy = (ball.vy / spd) * MAX_BALL_SPEED;
    }
    
    // Walls
    const wallPad = 15;
    if (ball.x - BALL_RADIUS < wallPad) { ball.x = wallPad + BALL_RADIUS; ball.vx = Math.abs(ball.vx) * BOUNCE_DAMPING; playSound(200, 0.05, 'square', 0.05); }
    if (ball.x + BALL_RADIUS > W - wallPad) { ball.x = W - wallPad - BALL_RADIUS; ball.vx = -Math.abs(ball.vx) * BOUNCE_DAMPING; playSound(200, 0.05, 'square', 0.05); }
    if (ball.y - BALL_RADIUS < wallPad) { ball.y = wallPad + BALL_RADIUS; ball.vy = Math.abs(ball.vy) * BOUNCE_DAMPING; }
    
    // Bumpers
    for (const b of bumpers) {
        const dx = ball.x - b.x;
        const dy = ball.y - b.y;
        const dist = Math.hypot(dx, dy);
        if (dist < BALL_RADIUS + b.r) {
            const nx = dx / dist;
            const ny = dy / dist;
            ball.x = b.x + nx * (BALL_RADIUS + b.r + 1);
            ball.y = b.y + ny * (BALL_RADIUS + b.r + 1);
            const dot = ball.vx * nx + ball.vy * ny;
            ball.vx -= 2 * dot * nx;
            ball.vy -= 2 * dot * ny;
            // Boost
            ball.vx += nx * 5;
            ball.vy += ny * 5;
            playSound(300 + Math.random() * 200, 0.1, 'sine', 0.15);
            spawnParticle(ball.x, ball.y, '#ffff00', 4);
        }
    }
    
    // Flippers collision
    for (const key of ['left', 'right']) {
        const f = flippers[key];
        const tipX = key === 'left' ? f.x + Math.cos(f.angle) * f.len : f.x - Math.cos(f.angle) * f.len;
        const tipY = f.y + Math.sin(f.angle) * f.len;
        
        // Line segment collision (base to tip)
        const bx = key === 'left' ? f.x : f.x;
        const by = f.y;
        const ex = tipX;
        const ey = tipY;
        
        const dist = pointToLineDist(ball.x, ball.y, bx, by, ex, ey);
        if (dist < BALL_RADIUS + 5) {
            const nx = (ball.y - by) / (Math.hypot(ex - bx, ey - by) || 1);
            const ny = -(ball.x - bx) / (Math.hypot(ex - bx, ey - by) || 1);
            
            // Push ball out
            if (dist < BALL_RADIUS + 5) {
                const pushDirX = nx >= 0 ? 1 : -1;
                ball.y -= ny * (BALL_RADIUS + 6 - dist);
                ball.x += pushDirX * (BALL_RADIUS + 6 - dist) * 0.3;
            }
            
            // Add flipper velocity boost if active
            const isFlipping = (key === 'left' && keys['ArrowLeft']) || (key === 'right' && keys['ArrowRight']);
            const boost = isFlipping ? 8 : 2;
            ball.vy = -boost;
            ball.vx += (Math.random() - 0.5) * 3;
            playSound(500, 0.08, 'sine', 0.1);
        }
    }
    
    // Enemy collision
    for (let i = enemies.length - 1; i >= 0; i--) {
        const e = enemies[i];
        const dist = Math.hypot(ball.x - e.x, ball.y - e.y);
        if (dist < BALL_RADIUS + e.radius) {
            e.hp--;
            e.hitFlash = 10;
            playSound(200, 0.1, 'sawtooth', 0.15);
            spawnParticle(e.x, e.y, '#ff4444', 6);
            shakeTime = 5;
            shakeMag = 3;
            
            // Bounce ball off enemy
            const nx = (ball.x - e.x) / dist;
            const ny = (ball.y - e.y) / dist;
            ball.vx += nx * 4;
            ball.vy -= Math.abs(ball.vy) * 0.5 + 3;
            
            if (e.hp <= 0) {
                score += e.type === 'tank' ? 50 : 20;
                points += 10;
                spawnParticle(e.x, e.y, '#ff8800', 12);
                enemies.splice(i, 1);
            }
        }
    }
    
    // Ball lost
    if (ball.y > H + 50) {
        ball = null;
        lives--;
        playSound(100, 0.3, 'sawtooth', 0.2);
        shakeTime = 10;
        shakeMag = 5;
        if (lives <= 0) {
            gameState = 'gameover';
        }
    }
}

function pointToLineDist(px, py, x1, y1, x2, y2) {
    const dx = x2 - x1;
    const dy = y2 - y1;
    const len2 = dx * dx + dy * dy;
    let t = ((px - x1) * dx + (py - y1) * dy) / len2;
    t = Math.max(0, Math.min(1, t));
    const cx = x1 + t * dx;
    const cy = y1 + t * dy;
    return Math.hypot(px - cx, py - cy);
}

function updateFlippers() {
    // Left flipper
    if (keys['ArrowLeft']) {
        flippers.left.angle += (flippers.left.activeAngle - flippers.left.angle) * 0.4;
    } else {
        flippers.left.angle += (flippers.left.restAngle - flippers.left.angle) * 0.25;
    }
    // Right flipper
    if (keys['ArrowRight']) {
        flippers.right.angle += (flippers.right.activeAngle - flippers.right.angle) * 0.4;
    } else {
        flippers.right.angle += (flippers.right.restAngle - flippers.right.angle) * 0.25;
    }
}

function updateEnemies() {
    for (let i = enemies.length - 1; i >= 0; i--) {
        const e = enemies[i];
        e.y += e.speed;
        if (e.hitFlash > 0) e.hitFlash--;
        
        // Reached bottom
        if (e.y > H * 0.82) {
            spawnParticle(e.x, e.y, '#ff0000', 10);
            enemies.splice(i, 1);
        }
    }
}

function updateTowers() {
    for (const t of towers) {
        if (t.cooldown > 0) { t.cooldown--; continue; }
        
        // Find nearest enemy in range
        let best = null, bestDist = Infinity;
        for (const e of enemies) {
            const d = Math.hypot(e.x - t.x, e.y - t.y);
            if (d < t.range && d < bestDist) { best = e; bestDist = d; }
        }
        
        if (best) {
            const angle = Math.atan2(best.y - t.y, best.x - t.x);
            projectiles.push({ x: t.x, y: t.y, vx: Math.cos(angle) * 5, vy: Math.sin(angle) * 5, damage: t.damage, life: 60 });
            t.cooldown = t.fireRate;
            playSound(800 + Math.random() * 200, 0.05, 'square', 0.05);
        }
    }
}

function updateProjectiles() {
    for (let i = projectiles.length - 1; i >= 0; i--) {
        const p = projectiles[i];
        p.x += p.vx;
        p.y += p.vy;
        p.life--;
        
        if (p.life <= 0) { projectiles.splice(i, 1); continue; }
        
        // Hit enemy
        for (let j = enemies.length - 1; j >= 0; j--) {
            const e = enemies[j];
            if (Math.hypot(p.x - e.x, p.y - e.y) < e.radius + 5) {
                e.hp -= p.damage;
                e.hitFlash = 8;
                playSound(300, 0.05, 'square', 0.08);
                spawnParticle(p.x, p.y, '#44ff88', 3);
                
                if (e.hp <= 0) {
                    score += e.type === 'tank' ? 50 : 20;
                    points += 10;
                    spawnParticle(e.x, e.y, '#ff8800', 12);
                    enemies.splice(j, 1);
                }
                projectiles.splice(i, 1);
                break;
            }
        }
    }
}

function updateParticles() {
    for (let i = particles.length - 1; i >= 0; i--) {
        const p = particles[i];
        p.x += p.vx;
        p.y += p.vy;
        p.vy += 0.1;
        p.life--;
        if (p.life <= 0) particles.splice(i, 1);
    }
}

function checkWaveComplete() {
    if (enemies.length === 0 && projectiles.length === 0) {
        // Small delay then next wave
        setTimeout(() => { if (gameState === 'playing') nextWave(); }, 1500);
        enemies.push({ x: -999, y: -999, hp: 1, maxHp: 1, speed: 0, radius: 1, type: 'wait', hitFlash: 0 });
    }
}

        const bgrad = ctx.createRadialGradient(ball.x - 3, ball.y - 3, 0, ball.x, ball.y, BALL_RADIUS);
        bgrad.addColorStop(0, '#ffffff');
        bgrad.addColorStop(1, '#88ccff');
        ctx.fillStyle = bgrad;
        ctx.beginPath();
        ctx.arc(ball.x, ball.y, BALL_RADIUS, 0, Math.PI * 2);
        ctx.fill();
    } else if (!ball) {
        // Launch indicator
        ctx.fillStyle = '#88ccff';
        ctx.font = '14px monospace';
        ctx.textAlign = 'center';
        ctx.fillText('SPACE to launch', W * 0.92, H * 0.88);
    }
    
    // Particles
    for (const p of particles) {
        const alpha = p.life / 50;
        ctx.fillStyle = p.color + Math.floor(alpha * 255).toString(16).padStart(2, '0');
        ctx.beginPath();
        ctx.arc(p.x, p.y, 3, 0, Math.PI * 2);
        ctx.fill();
    }
    
    // HUD
    drawHUD();
    
    ctx.restore();
}

let lastTime = 0;
function gameLoop(timestamp) {
    const dt = timestamp - lastTime;
    lastTime = timestamp;
    
    if (gameState === 'playing') {
        updateFlippers();
        updateBall();
        updateEnemies();
        updateTowers();
        updateProjectiles();
        updateParticles();
        
        // Auto-launch ball after delay
        if (!ball && Math.random() < 0.02) launchBall();
    } else {
        updateParticles();
    }
    
    draw();
    requestAnimationFrame(gameLoop);
}
