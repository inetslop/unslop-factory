// Audio
let audioCtx = null;
function initAudio() {
    if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
}
function playLaser() {
    if (!audioCtx) return;
    const o = audioCtx.createOscillator();
    const g = audioCtx.createGain();
    o.type = 'sawtooth';
    o.frequency.setValueAtTime(880, audioCtx.currentTime);
    o.frequency.exponentialRampToValueAtTime(220, audioCtx.currentTime + 0.15);
    g.gain.setValueAtTime(0.3, audioCtx.currentTime);
    g.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.15);
    o.connect(g); g.connect(audioCtx.destination);
    o.start(); o.stop(audioCtx.currentTime + 0.15);
}
function playExplosion() {
    if (!audioCtx) return;
    const bufferSize = audioCtx.sampleRate * 0.4;
    const buffer = audioCtx.createBuffer(1, bufferSize, audioCtx.sampleRate);
    const data = buffer.getChannelData(0);
    for (let i = 0; i < bufferSize; i++) data[i] = (Math.random() * 2 - 1) * Math.pow(1 - i / bufferSize, 3);
    const src = audioCtx.createBufferSource();
    const g = audioCtx.createGain();
    src.buffer = buffer;
    g.gain.setValueAtTime(0.4, audioCtx.currentTime);
    g.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.4);
    src.connect(g); g.connect(audioCtx.destination);
    src.start();
}
function playHit() {
    if (!audioCtx) return;
    const o = audioCtx.createOscillator();
    const g = audioCtx.createGain();
    o.type = 'square';
    o.frequency.setValueAtTime(150, audioCtx.currentTime);
    o.frequency.exponentialRampToValueAtTime(50, audioCtx.currentTime + 0.3);
    g.gain.setValueAtTime(0.4, audioCtx.currentTime);
    g.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.3);
    o.connect(g); g.connect(audioCtx.destination);
    o.start(); o.stop(audioCtx.currentTime + 0.3);
}

function startGame() {
    gameState = 'playing';
    score = 0;
    lives = 3;
    wave = 1;
    waveTimer = 0;
    player.x = 0;
    player.y = -2;
    player.vx = 0;
    player.vy = 0;
    bullets = [];
    asteroids = [];

function spawnWave() {
    const count = 3 + wave * 2;
    for (let i = 0; i < count; i++) {
        setTimeout(() => {
            if (gameState !== 'playing') return;
            spawnAsteroid();
        }, i * 400);
    }
}

function spawnAsteroid() {
    const size = Math.random() * 1.5 + 0.8;
    const verts = [];
    const numVerts = 7 + Math.floor(Math.random() * 5);
    for (let i = 0; i < numVerts; i++) {
        const a = (i / numVerts) * Math.PI * 2;
        const r = size * (0.6 + Math.random() * 0.4);
        verts.push({ x: Math.cos(a) * r, y: Math.sin(a) * r, z: 0 });
    }
    asteroids.push({
        x: (Math.random() - 0.5) * 20,
        y: -(Math.random() * 3 + 2),
        z: 40 + Math.random() * 20,
        vx: (Math.random() - 0.5) * 0.02,
        vy: 0.01 + Math.random() * 0.02,
        rotX: Math.random() * Math.PI * 2,
        rotY: Math.random() * Math.PI * 2,
        rotSpdX: (Math.random() - 0.5) * 0.03,
        rotSpdY: (Math.random() - 0.5) * 0.03,
        verts: verts,
        size: size
    });
}

function project(x, y, z) {
    if (z <= 0.1) return null;
    const fov = 400;
    const sx = W / 2 + (x - player.x * 0.3) * fov / z;
    const sy = H / 2 - (y + 2) * fov / z + player.y * fov / z * 0.5;
    return { x: sx, y: sy, scale: fov / z };
}

function fireBullet() {
    if (bullets.length > 4) return;
    bullets.push({ x: player.x * 0.3, y: -2, z: 2 });
    playLaser();
}

function update(dt) {
    if (gameState !== 'playing') return;

    // Player movement
    const speed = 4;
    if (keys['ArrowLeft'] || keys['KeyA']) player.x -= speed * dt;
    if (keys['ArrowRight'] || keys['KeyD']) player.x += speed * dt;
    if (keys['ArrowUp'] || keys['KeyW']) player.y -= speed * 0.5 * dt;
    if (keys['ArrowDown'] || keys['KeyS']) player.y += speed * 0.5 * dt;
    player.x = Math.max(-8, Math.min(8, player.x));
    player.y = Math.max(-4, Math.min(2, player.y));

    // Fire
    if (keys['Space'] && bullets.length < 5) {
        fireBullet();
        keys['Space'] = false;
    }

    // Bullets
    for (let i = bullets.length - 1; i >= 0; i--) {
        bullets[i].z += 30 * dt;
        if (bullets[i].z > 70) bullets.splice(i, 1);
    }

    // Asteroids
    for (let i = asteroids.length - 1; i >= 0; i--) {
        const a = asteroids[i];
        a.z -= (5 + wave * 0.5) * dt;
        a.x += a.vx;
        a.rotX += a.rotSpdX;
        a.rotY += a.rotSpdY;

        if (a.z < 1) {
            // Hit player!
            asteroids.splice(i, 1);
            createExplosion(a.x, a.y, a.z, 'red');
            playHit();
            lives--;
            shakeTime = 0.5;
            shakeMag = 10;
            if (lives <= 0) {
                gameState = 'gameover';
                try { 
                    if (score > highScore) {
                        highScore = score;
                        localStorage.setItem('asteroids3d_hs', highScore);
                    }
                } catch(e) {}
            }
            continue;
        }

        // Check bullet collisions
        for (let j = bullets.length - 1; j >= 0; j--) {
            const b = bullets[j];
            const dx = a.x - b.x, dy = a.y - b.y, dz = a.z - b.z;
            if (Math.abs(dz) < 3 && Math.abs(dx) < a.size + 1 && Math.abs(dy) < a.size + 1) {
                asteroids.splice(i, 1);
                bullets.splice(j, 1);
                score += Math.floor(100 / a.z);
                createExplosion(a.x, a.y, a.z, 'cyan');
                playExplosion();
                // Split into smaller ones
                if (a.size > 1) {
                    for (let k = 0; k < 2; k++) {
                        const verts = [];
                        const nv = 6 + Math.floor(Math.random() * 4);
                        for (let v = 0; v < nv; v++) {
                            const ang = (v / nv) * Math.PI * 2;
                            const r = a.size * 0.5 * (0.6 + Math.random() * 0.4);
                            verts.push({ x: Math.cos(ang) * r, y: Math.sin(ang) * r, z: 0 });
                        }
                        asteroids.push({
                            x: a.x + (Math.random() - 0.5),
                            y: a.y,
                            z: a.z + (Math.random() - 0.5) * 2,
                            vx: (Math.random() - 0.5) * 0.05,
                            vy: 0.01 + Math.random() * 0.02,
                            rotX: Math.random() * Math.PI * 2,
                            rotY: Math.random() * Math.PI * 2,
                            rotSpdX: (Math.random() - 0.5) * 0.05,
                            rotSpdY: (Math.random() - 0.5) * 0.05,
                            verts: verts,
                            size: a.size * 0.5
                        });
                    }
                }
                break;
            }
        }
    }

    // Wave progression
    waveTimer += dt;
    if (asteroids.length === 0 && bullets.length <= 1) {
        if (waveTimer > 2) {
            wave++;
            waveTimer = 0;
            spawnWave();
        }
    }

    // Particles
    for (let i = particles.length - 1; i >= 0; i--) {
        const p = particles[i];
        p.x += p.vx * dt;
        p.y += p.vy * dt;
        p.z -= p.vz * dt;
        p.life -= dt;
        if (p.life <= 0) particles.splice(i, 1);
    }

    // Shake decay
    shakeTime = Math.max(0, shakeTime - dt);
}

function createExplosion(x, y, z, color) {
    for (let i = 0; i < 20; i++) {
        const a = Math.random() * Math.PI * 2;
        const spd = Math.random() * 3 + 1;
        particles.push({
            x: x, y: y, z: z,
            vx: Math.cos(a) * spd,
            vy: (Math.random() - 0.5) * 2,
            vz: Math.sin(a) * spd,
            life: 0.8 + Math.random() * 0.5,
            color: color
        });
    }
}

function draw3DWireframe(verts, ox, oy, oz, rotX, rotY, scale) {
    const cosX = Math.cos(rotX), sinX = Math.sin(rotX);
    const cosY = Math.cos(rotY), sinY = Math.sin(rotY);
    const pts = [];
    for (const v of verts) {
        let x = v.x, y = v.y, z = v.z;
        // Rotate Y
        let tx = x * cosY - z * sinY;
        let tz = x * sinY + z * cosY;
        x = tx; z = tz;
        // Rotate X
        let ty = y * cosX - z * sinX;
        tz = y * sinX + z * cosX;
        y = ty; z = tz;
        const px = ox + x * scale;
        const py = oy + y * scale;
        pts.push({ x: px, y: py });
    }
    ctx.beginPath();
    ctx.moveTo(pts[0].x, pts[0].y);
    for (let i = 1; i < pts.length; i++) {
        ctx.lineTo(pts[i].x, pts[i].y);
    }
    ctx.closePath();
    ctx.stroke();
}

        if (score >= highScore && score > 0) {
            ctx.fillStyle = '#ff0';
            ctx.fillText('★ NEW HIGH SCORE! ★', W/2, H/2 + 90);
        }

        ctx.fillStyle = '#0ff';
        ctx.font = '20px monospace';
        ctx.fillText('PRESS SPACE OR R TO RESTART', W/2, H/2 + 140);
    }

    ctx.restore();
}

let lastTime = 0;
function gameLoop(time) {
    const dt = Math.min((time - lastTime) / 1000, 0.05);
    lastTime = time;

    update(dt);
    draw();

    requestAnimationFrame(gameLoop);
}
