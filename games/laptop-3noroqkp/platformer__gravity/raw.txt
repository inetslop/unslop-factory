// Audio
let audioCtx = null;
function initAudio() {
    if (!audioCtx) {
        audioCtx = new (window.AudioContext || window.webkitAudioContext)();
    }
}
function playSound(type) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    
    if (type === 'jump') {
        osc.type = 'sine';
        osc.frequency.setValueAtTime(300, audioCtx.currentTime);
        osc.frequency.linearRampToValueAtTime(600, audioCtx.currentTime + 0.1);
        gain.gain.setValueAtTime(0.15, audioCtx.currentTime);
        gain.gain.linearRampToValueAtTime(0, audioCtx.currentTime + 0.1);
        osc.start();
        osc.stop(audioCtx.currentTime + 0.1);
    } else if (type === 'collect') {
        osc.type = 'sine';
        osc.frequency.setValueAtTime(800, audioCtx.currentTime);
        osc.frequency.linearRampToValueAtTime(1200, audioCtx.currentTime + 0.1);
        gain.gain.setValueAtTime(0.15, audioCtx.currentTime);
        gain.gain.linearRampToValueAtTime(0, audioCtx.currentTime + 0.2);
        osc.start();
        osc.stop(audioCtx.currentTime + 0.2);
    } else if (type === 'die') {
        osc.type = 'sawtooth';
        osc.frequency.setValueAtTime(400, audioCtx.currentTime);
        osc.frequency.linearRampToValueAtTime(100, audioCtx.currentTime + 0.4);
        gain.gain.setValueAtTime(0.2, audioCtx.currentTime);
        gain.gain.linearRampToValueAtTime(0, audioCtx.currentTime + 0.4);
        osc.start();
        osc.stop(audioCtx.currentTime + 0.4);
    } else if (type === 'win') {
        osc.type = 'sine';
        const notes = [523, 659, 784, 1047];
        notes.forEach((freq, i) => {
            const o = audioCtx.createOscillator();
            const g = audioCtx.createGain();
            o.connect(g); g.connect(audioCtx.destination);
            o.type = 'sine';
            o.frequency.setValueAtTime(freq, audioCtx.currentTime + i * 0.15);
            g.gain.setValueAtTime(0.15, audioCtx.currentTime + i * 0.15);
            g.gain.linearRampToValueAtTime(0, audioCtx.currentTime + i * 0.15 + 0.14);
            o.start(audioCtx.currentTime + i * 0.15);
            o.stop(audioCtx.currentTime + i * 0.15 + 0.15);
        });
    }

let gameState = 'start'; // start, playing, dead, win
let score = 0;
let highScore = 0;

let deathTimer = 0;
let winTimer = 0;

let camX = 0;
let time = 0;

// Player

    onGround: false, facing: 1,
    runFrame: 0, runTimer: 0,
    dead: false

const GRAVITY = 0.6;
const JUMP_FORCE = -12;
const MOVE_SPEED = 5;
const MAX_FALL = 14;

let platforms = [];
let gems = [];
let spikes = [];
let finishLine = null;
let decorations = [];

function generateLevel() {
    platforms = [];
    gems = [];
    spikes = [];
    decorations = [];
    
    const groundY = H - 60;
    
    // Starting platform
    platforms.push({ x: 0, y: groundY, w: 250, h: 60 });
    
    let lastX = 200;
    let lastY = groundY;
    const levelWidth = 3500;
    
    for (let i = 0; i < 40; i++) {
        const gap = 80 + Math.random() * 120;
        const heightChange = (Math.random() - 0.5) * 160;
        const newY = Math.max(80, Math.min(H - 120, lastY + heightChange));
        const newX = lastX + gap;
        const platW = 80 + Math.random() * 150;
        
        platforms.push({ x: newX, y: newY, w: platW, h: 24 });
        
        // Add gem above some platforms
        if (Math.random() < 0.6) {
            gems.push({ x: newX + platW/2 - 10, y: newY - 50, w: 20, h: 20, collected: false, bob: Math.random() * Math.PI * 2 });
        }
        
        // Add spikes on some platforms
        if (Math.random() < 0.3 && platW > 120) {
            const spikeCount = Math.floor(platW / 30);
            for (let s = 0; s < spikeCount; s++) {
                spikes.push({ x: newX + s * 30, y: newY - 16, w: 24, h: 16 });
            }
        }
        
        lastX = newX + platW;
        lastY = newY;
    }
    
    // Finish platform
    const finishPlat = { x: lastX + 100, y: lastY, w: 200, h: 24 };
    platforms.push(finishPlat);
    finishLine = { x: lastX + 180, y: lastY - 60, w: 20, h: 60 };
    
    // Ground sections (longer, with spikes)
    for (let i = 0; i < 5; i++) {
        const gx = 400 + i * 700;
        if (!platforms.find(p => Math.abs(p.x - gx) < 200)) {
            // Add a ground-level platform
        }
    }
    
    // Decorations (background elements)
    for (let i = 0; i < 30; i++) {
        decorations.push({
            x: Math.random() * levelWidth,
            y: Math.random() * H * 0.6 + H * 0.2,
            size: 20 + Math.random() * 40,
            type: Math.floor(Math.random() * 3),
            phase: Math.random() * Math.PI * 2
        });
    }
    
    // Reset player
    player.x = 100;
    player.y = groundY - player.h;
    player.vx = 0;
    player.vy = 0;
    player.onGround = false;
    player.dead = false;
    score = 0;
    camX = 0;
    particles = [];
}

const keys = {};
window.addEventListener('keydown', e => {
    keys[e.code] = true;
    if (gameState === 'start' || gameState === 'dead' || gameState === 'win') {

            generateLevel();
            gameState = 'playing';

    }

// Particles
function spawnParticles(x, y, color, count) {
    for (let i = 0; i < count; i++) {
        particles.push({
            x, y,
            vx: (Math.random() - 0.5) * 6,
            vy: (Math.random() - 1) * 5,
            life: 1,
            color,
            size: 3 + Math.random() * 4
        });
    }
}

// Update
function update(dt) {
    time += dt;
    
    if (gameState !== 'playing') return;
    
    // Player input
    if (keys['ArrowLeft'] || keys['KeyA']) {
        player.vx = -MOVE_SPEED;
        player.facing = -1;
    } else if (keys['ArrowRight'] || keys['KeyD']) {
        player.vx = MOVE_SPEED;
        player.facing = 1;
    } else {
        player.vx *= 0.8;
    }
    
    if ((keys['Space'] || keys['ArrowUp'] || keys['KeyW']) && player.onGround) {
        player.vy = JUMP_FORCE;
        player.onGround = false;
        playSound('jump');
    }
    
    // Gravity
    player.vy += GRAVITY;
    if (player.vy > MAX_FALL) player.vy = MAX_FALL;
    
    // Move
    player.x += player.vx;
    player.y += player.vy;
    
    // Run animation
    if (Math.abs(player.vx) > 0.5 && player.onGround) {
        player.runTimer += dt;
        if (player.runTimer > 0.1) {
            player.runFrame = (player.runFrame + 1) % 4;
            player.runTimer = 0;
        }
    } else {
        player.runFrame = 0;
    }
    
    // Collision with platforms
    player.onGround = false;
    for (const p of platforms) {
        if (player.x + player.w > p.x && player.x < p.x + p.w) {
            // Landing on top
            if (player.vy >= 0 && player.y + player.h >= p.y && player.y + player.h - player.vy <= p.y + 10) {
                player.y = p.y - player.h;
                player.vy = 0;
                player.onGround = true;
            }
            // Hitting bottom
            else if (player.vy < 0 && player.y <= p.y + p.h && player.y >= p.y + p.h - 10) {
                player.y = p.y + p.h;
                player.vy = 0;
            }
        }
        // Side collision
        if (player.y + player.h > p.y && player.y < p.y + p.h) {
            if (player.vx > 0 && player.x + player.w >= p.x && player.x + player.w - player.vx <= p.x + 5) {
                player.x = p.x - player.w;
                player.vx = 0;
            } else if (player.vx < 0 && player.x <= p.x + p.w && player.x - player.vx >= p.x + p.w - 5) {
                player.x = p.x + p.w;
                player.vx = 0;
            }
        }
    }
    
    // Fall death
    if (player.y > H + 100) {
        die();
        return;
    }
    
    // Spike collision
    for (const s of spikes) {
        if (player.x + player.w - 5 > s.x && player.x + 5 < s.x + s.w &&
            player.y + player.h > s.y + 4 && player.y + player.h < s.y + s.h) {
            die();
            return;
        }
    }
    
    // Gem collection
    for (const g of gems) {
        if (!g.collected) {
            g.bob += dt * 3;
            const gx = g.x, gy = g.y + Math.sin(g.bob) * 5;
            if (player.x + player.w > gx && player.x < gx + g.w &&
                player.y + player.h > gy && player.y < gy + g.h) {
                g.collected = true;
                score += 10;
                playSound('collect');
                spawnParticles(gx + 10, gy + 10, '#ffd700', 8);
            }
        }
    }
    
    // Finish line
    if (finishLine) {
        if (player.x + player.w > finishLine.x && player.x < finishLine.x + finishLine.w &&
            player.y + player.h > finishLine.y && player.y < finishLine.y + finishLine.h) {
            gameState = 'win';
            winTimer = 0;
            playSound('win');
            if (score > highScore) {
                highScore = score;
                try { localStorage.setItem('gravityDashHigh', highScore); } catch(e) {}
            }
        }
    }
    
    // Camera
    const targetCamX = player.x - W * 0.35;
    camX += (targetCamX - camX) * 0.1;
    if (camX < 0) camX = 0;
    
    // Particles
    for (let i = particles.length - 1; i >= 0; i--) {
        const p = particles[i];
        p.x += p.vx;
        p.y += p.vy;
        p.vy += 0.3;
        p.life -= dt * 2;
        if (p.life <= 0) particles.splice(i, 1);
    }
}

function die() {
    player.dead = true;
    gameState = 'dead';
    deathTimer = 0;
    playSound('die');
    spawnParticles(player.x + player.w/2, player.y + player.h/2, '#ff4444', 15);
}

function drawBackground() {
    // Sky gradient
    const grad = ctx.createLinearGradient(0, 0, 0, H);
    grad.addColorStop(0, '#0f0c29');
    grad.addColorStop(0.5, '#1a1a3e');
    grad.addColorStop(1, '#24243e');
    ctx.fillStyle = grad;
    ctx.fillRect(0, 0, W, H);
    
    // Stars
    ctx.fillStyle = 'rgba(255,255,255,0.6)';
    for (let i = 0; i < 50; i++) {
        const sx = ((i * 137 + 50) % W);
        const sy = ((i * 97 + 30) % (H * 0.6));
        const twinkle = Math.sin(time * 2 + i) * 0.5 + 0.5;
        ctx.globalAlpha = 0.3 + twinkle * 0.5;
        ctx.fillRect(sx, sy, 2, 2);
    }
    ctx.globalAlpha = 1;
    
    // Distant mountains
    ctx.fillStyle = '#1a1a3e';
    for (let i = 0; i < 8; i++) {
        const mx = ((i * 400 - camX * 0.2) % (W + 400)) - 200;
        const mh = 100 + (i * 73 % 150);
        ctx.beginPath();
        ctx.moveTo(mx, H - 60);
        ctx.lineTo(mx + 80, H - 60 - mh);
        ctx.lineTo(mx + 160, H - 60);
        ctx.fill();
    }
}

function drawDecorations() {
    for (const d of decorations) {
        const dx = d.x - camX * 0.5;
        if (dx < -d.size || dx > W + d.size) continue;
        
        ctx.globalAlpha = 0.15;
        if (d.type === 0) {
            ctx.fillStyle = '#4a90d9';
            ctx.fillRect(dx, d.y, d.size * 0.6, d.size);
            ctx.fillRect(dx - 5, d.y + d.size * 0.3, d.size * 0.8, 4);
        } else if (d.type === 1) {
            ctx.fillStyle = '#9b59b6';
            ctx.beginPath();
            ctx.arc(dx, d.y, d.size * 0.3, 0, Math.PI * 2);
            ctx.fill();
        } else {
            ctx.fillStyle = '#2ecc71';
            ctx.beginPath();
            ctx.moveTo(dx, d.y + d.size);
            ctx.lineTo(dx + d.size * 0.5, d.y);
            ctx.lineTo(dx + d.size, d.y + d.size);
            ctx.fill();
        }
        ctx.globalAlpha = 1;
    }
}

    for (const p of platforms) {
        const px = p.x - camX;
        if (px > W + 50 || px + p.w < -50) continue;

        ctx.fillStyle = '#3d3d6b';
        ctx.fillRect(px, p.y, p.w, p.h);
        
        // Top highlight
        ctx.fillStyle = '#5a5a9e';
        ctx.fillRect(px, p.y, p.w, 4);

        for (let i = 0; i < p.w; i += 20) {

        }

function drawSpikes() {
    for (const s of spikes) {
        const sx = s.x - camX;
        if (sx > W || sx + s.w < 0) continue;
        
        ctx.fillStyle = '#e74c3c';
        ctx.beginPath();
        ctx.moveTo(sx, s.y + s.h);
        ctx.lineTo(sx + s.w / 2, s.y);
        ctx.lineTo(sx + s.w, s.y + s.h);
        ctx.fill();
        
        // Glow
        ctx.fillStyle = 'rgba(231, 76, 60, 0.2)';
        ctx.beginPath();
        ctx.arc(sx + s.w/2, s.y + s.h/2, 15, 0, Math.PI * 2);
        ctx.fill();
    }
}

function drawGems() {
    for (const g of gems) {
        if (g.collected) continue;
        const gx = g.x - camX;
        if (gx > W || gx + g.w < 0) continue;

        ctx.beginPath();
        ctx.arc(gx + 10, gy + 10, 18, 0, Math.PI * 2);
        ctx.fill();
        
        // Gem shape
        ctx.fillStyle = '#ffd700';
        ctx.beginPath();
        ctx.moveTo(gx + 10, gy);
        ctx.lineTo(gx + 20, gy + 10);
        ctx.lineTo(gx + 10, gy + 20);
        ctx.lineTo(gx, gy + 10);

        // Highlight

        ctx.beginPath();
        ctx.moveTo(gx + 10, gy + 3);
        ctx.lineTo(gx + 15, gy + 10);
        ctx.lineTo(gx + 10, gy + 12);
        ctx.lineTo(gx + 5, gy + 10);
        ctx.fill();
    }

function drawFinish() {

    if (fx > W || fx < -50) return;

function drawPlayer() {

    const py = player.y;

    ctx.fillStyle = 'rgba(0,0,0,0.3)';
    ctx.beginPath();
    ctx.ellipse(px + player.w/2, py + player.h + 4, 14, 5, 0, 0, Math.PI * 2);

        }
    }
    
    // Update particles even in win state
    for (let i = particles.length - 1; i >= 0; i--) {
        const p = particles[i];
        p.x += p.vx;
        p.y += p.vy;
        p.life -= 0.005;
        if (p.life <= 0) particles.splice(i, 1);
    }
    
    ctx.fillStyle = 'rgba(0,0,0,0.6)';
    ctx.fillRect(0, 0, W, H);
    
    ctx.fillStyle = '#2ecc71';
    ctx.font = 'bold 48px monospace';
    ctx.textAlign = 'center';
    ctx.fillText('YOU WIN!', W/2, H * 0.35);
    
    ctx.fillStyle = '#ffd700';
    ctx.font = '24px monospace';
    ctx.fillText(`Gems Collected: ${score}`, W/2, H * 0.48);
    
    if (score >= highScore && score > 0) {
        ctx.fillStyle = '#ff6bcb';
        ctx.font = '18px monospace';
        ctx.fillText('NEW HIGH SCORE!', W/2, H * 0.55);
    }
    
    const pulse = Math.sin(time * 3) * 0.5 + 0.5;
    ctx.globalAlpha = 0.5 + pulse * 0.5;
    ctx.fillStyle = '#4a90d9';
    ctx.font = 'bold 20px monospace';
    ctx.fillText('Press SPACE to Play Again', W/2, H * 0.7);
    ctx.globalAlpha = 1;
}

// Main loop
let lastTime = 0;
function gameLoop(timestamp) {
    const dt = Math.min((timestamp - lastTime) / 1000, 0.05);
    lastTime = timestamp;
    
    update(dt);
    
    // Draw
    ctx.clearRect(0, 0, W, H);
    
    if (gameState === 'start') {
        drawStartScreen();
    } else {
        drawBackground();
        drawDecorations();
        drawPlatforms();
        drawSpikes();
        drawGems();
        drawFinish();
        drawPlayer();
        drawParticles();
        drawHUD();
        
        if (gameState === 'dead') drawDeathScreen();
        if (gameState === 'win') drawWinScreen();
    }
    
    requestAnimationFrame(gameLoop);
}

// Initial setup
generateLevel();
