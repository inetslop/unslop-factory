function spawnParticles(x, y, color) {
  for (let i = 0; i < 8; i++) {
    const angle = Math.random() * Math.PI * 2;
    const speed = 1 + Math.random() * 3;
    particles.push({
      x: x, y: y,
      vx: Math.cos(angle) * speed,
      vy: Math.sin(angle) * speed - 2,
      r: 3 + Math.random() * 4,
      color: color,
      life: 1
    });
  }
}

function updateParticles() {
  particles = particles.filter(p => {
    p.x += p.vx;
    p.y += p.vy;
    p.vy += 0.2;
    p.life -= 0.03;
    return p.life > 0;
  });
}

function findMatches() {
  const matches = [];
  
  // Horizontal
  for (let y = 0; y < GRID; y++) {
    for (let x = 0; x < GRID; x++) {
      if (grid[y][x].type >= 0 && grid[y][x].type < TYPES) {
        const t = grid[y][x].type;
        let count = 1;
        while (x + count < GRID && grid[y][x + count].type === t) count++;
        if (count >= 3) matches.push({ x, y, type: t, length: count, horizontal: true });
      }
    }
  }
  
  // Vertical
  for (let x = 0; x < GRID; x++) {
    for (let y = 0; y < GRID; y++) {
      if (grid[y][x].type >= 0 && grid[y][x].type < TYPES) {
        const t = grid[y][x].type;
        let count = 1;
        while (y + count < GRID && grid[y + count][x].type === t) count++;
        if (count >= 3) matches.push({ x, y, type: t, length: count, horizontal: false });
      }
    }
  }
  
  return matches;
}

function processMatches() {
  const matches = findMatches();
  if (matches.length === 0) {
    comboCount = 0;
    return;
  }
  
  const now = Date.now();
  if (now - lastMatchTime > 1000) comboCount = 0;
  lastMatchTime = now;
  comboCount++;
  
  const toRemove = new Set();
  
  matches.forEach(m => {
    let bonus = m.length >= 5 ? 3 : m.length === 4 ? 2 : 1;
    score += bonus * comboCount;
    
    if (m.horizontal) {
      for (let i = 0; i < m.length; i++) toRemove.add(m.x + i + ',' + m.y);
    } else {
      for (let i = 0; i < m.length; i++) toRemove.add(m.x + ',' + (m.y + i));
    }
    
    const pos = getTilePos(m.x, m.y);
    spawnParticles(pos.x + pos.size / 2, pos.y + pos.size / 2, TILE_COLORS[m.type]);
  });
  
  // Remove matched
  toRemove.forEach(key => {
    const [x, y] = key.split(',').map(Number);
    grid[y][x].type = -1;
  });
  
  if (comboCount > 1) playSound('bonus');
  else playSound('match');
  
  applyGravity();
  fillGrid();
  
  setTimeout(processMatches, 300);
}

function applyGravity() {
  for (let x = 0; x < GRID; x++) {
    let writeRow = GRID - 1;
    for (let y = GRID - 1; y >= 0; y--) {
      if (grid[y][x].type >= 0) {
        grid[writeRow][x].type = grid[y][x].type;
        if (writeRow !== y) grid[y][x].type = -1;
        writeRow--;
      }
    }
  }
}

function fillGrid() {
  for (let x = 0; x < GRID; x++) {
    for (let y = 0; y < GRID; y++) {
      if (grid[y][x].type < 0) {
        grid[y][x].type = Math.floor(Math.random() * TYPES);
      }
    }
  }
}

function swap(x1, y1, x2, y2) {
  const t = grid[y1][x1].type;
  grid[y1][x1].type = grid[y2][x2].type;
  grid[y2][x2].type = t;
}

function isValidMove(x, y) {
  return x >= 0 && x < GRID && y >= 0 && y < GRID;
}

function isAdjacent(x1, y1, x2, y2) {
  return Math.abs(x1 - x2) + Math.abs(y1 - y2) === 1;
}

function hasValidMoves() {
  for (let y = 0; y < GRID; y++) {
    for (let x = 0; x < GRID; x++) {
      if (x + 1 < GRID && grid[y][x].type === grid[y][x+1].type) return true;
      if (y + 1 < GRID && grid[y][x].type === grid[y+1][x].type) return true;
    }
  }
  return false;
}

function shuffleGrid() {
  const types = [];
  for (let y = 0; y < GRID; y++)
    for (let x = 0; x < GRID; x++)
      types.push(grid[y][x].type);
  
  for (let i = types.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    [types[i], types[j]] = [types[j], types[i]];
  }
  
  let idx = 0;
  for (let y = 0; y < GRID; y++)
    for (let x = 0; x < GRID; x++)
      grid[y][x].type = types[idx++];
}

  par = Math.min(3 + lvl, 7);
  comboCount = 0;
  initGrid();
  if (!hasValidMoves()) shuffleGrid();
}

function startGame() {
  totalScore = 0;
  level = 1;
  startLevel(1);
  state = 'playing';
}

// Input handling
let mouseX = 0, mouseY = 0;
let isDown = false;

function getGridPos(px, py) {
  const o = getGridOrigin();
  const gx = Math.floor((px - o.x) / o.tileSize);
  const gy = Math.floor((py - o.y) / o.tileSize);
  return { x: gx, y: gy };
}

function handleInput(px, py) {
  if (state === 'start') {
    initAudio();
    startGame();
    return;
  }
  
  if (state === 'levelComplete') {
    const cx = W / 2;
    const cy = H / 2;
    if (px > cx - 120 * scale && px < cx + 120 * scale && py > cy + 65 * scale && py < cy + 115 * scale) {
      totalScore += score;
      startLevel(level + 1);
      state = 'playing';
    }
    return;
  }
  
  if (state === 'gameOver') {
    const cx = W / 2;
    const cy = H / 2;
    if (px > cx - 100 * scale && px < cx + 100 * scale && py > cy + 35 * scale && py < cy + 85 * scale) {
      startGame();
    }
    return;
  }
  
  if (state === 'playing') {
    const pos = getGridPos(px, py);
    if (!isValidMove(pos.x, pos.y)) return;
    
    if (selected === null) {
      selected = { x: pos.x, y: pos.y };
    } else {
      if (pos.x === selected.x && pos.y === selected.y) {
        selected = null;
      } else if (isAdjacent(selected.x, selected.y, pos.x, pos.y)) {
        // Swap
        swap(selected.x, selected.y, pos.x, pos.y);
        
        const matches = findMatches();
        if (matches.length > 0) {
          playSound('swap');
          movesLeft--;
          processMatches();
        } else {
          // Invalid - swap back
          setTimeout(() => swap(selected.x, selected.y, pos.x, pos.y), 150);
          playSound('fail');
        }
        selected = null;
      } else {
        selected = { x: pos.x, y: pos.y };
      }
    }
  }
}

canvas.addEventListener('mousedown', (e) => {

  mouseY = (e.clientY - rect.top) * (H / rect.height);

  mouseY = (t.clientY - rect.top) * (H / rect.height);

});

// Main loop
function draw() {
  ctx.clearRect(0, 0, W, H);
  
  if (state === 'start') {
    drawStartScreen();
    return;
  }
  
  // Apply shake
  ctx.save();
  if (shakeTimer > 0) {
    const sx = (Math.random() - 0.5) * 4;
    const sy = (Math.random() - 0.5) * 4;
    ctx.translate(sx, sy);
    shakeTimer -= 0.05;
  }
  
  drawBackground();
  
  // Draw grid
  const o = getGridOrigin();
  
  // Grid background
  ctx.fillStyle = 'rgba(0,0,0,0.15)';
  ctx.fillRect(o.x - 5, o.y - 5, o.tileSize * GRID + 10, o.tileSize * GRID + 10);
  
  // Checkerboard
  for (let y = 0; y < GRID; y++) {
    for (let x = 0; x < GRID; x++) {
      const pos = getTilePos(x, y);
      ctx.fillStyle = (x + y) % 2 === 0 ? 'rgba(255,255,255,0.03)' : 'rgba(0,0,0,0.05)';
      ctx.fillRect(pos.x, pos.y, pos.size, pos.size);
    }
  }
  
  // Draw tiles
  for (let y = 0; y < GRID; y++) {
    for (let x = 0; x < GRID; x++) {
      if (grid[y][x].type >= 0) {
        drawTile(x, y, grid[y][x].type);
      }
    }
  }
  
  // Selection highlight
  if (selected && state === 'playing') {
    const pos = getTilePos(selected.x, selected.y);
    ctx.strokeStyle = '#fff';
    ctx.lineWidth = 3;
    ctx.strokeRect(pos.x + 2, pos.y + 2, pos.size - 4, pos.size - 4);
    
    // Glow effect
    ctx.shadowColor = '#fff';
    ctx.shadowBlur = 10;
    ctx.strokeRect(pos.x + 2, pos.y + 2, pos.size - 4, pos.size - 4);
    ctx.shadowBlur = 0;
  }
  
  drawHUD();
  drawParticles();
  
  ctx.restore();
  
  if (state === 'levelComplete') {
    drawLevelComplete();
  } else if (state === 'gameOver') {
    drawGameOver();
  }
}

function update() {
  updateParticles();
  
  if (state === 'playing') {
    // Check for level complete or game over after matches resolve
    setTimeout(() => {
      if (score >= par) {
        state = 'levelComplete';
        playSound('win');
      } else if (movesLeft <= 0) {
        totalScore += score;
        state = 'gameOver';
        shakeTimer = 1;
      }
    }, 500);
  }
}

let lastTime = 0;
function gameLoop(time) {
  update();
  draw();
  requestAnimationFrame(gameLoop);
}
