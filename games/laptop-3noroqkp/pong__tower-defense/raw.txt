<!DOCTYPE html>

<script>
const canvas = document.getElementById('c');
const ctx = canvas.getContext('2d');

let W, H;
function resize() {
    W = canvas.width = window.innerWidth;
    H = canvas.height = window.innerHeight;
}
window.addEventListener('resize', resize);
resize();

// Audio
let audioCtx = null;
function initAudio() {
    if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
}
function playSound(freq, dur, type, vol) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = type || 'square';
    osc.frequency.value = freq;
    gain.gain.setValueAtTime(vol || 0.15, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + dur);
}
function sBounce() { playSound(440, 0.1, 'square', 0.2); }
function sKill() { playSound(880, 0.2, 'sawtooth', 0.15); playSound(1200, 0.15, 'sine', 0.1); }
function sBreach() { playSound(120, 0.4, 'sawtooth', 0.3); }
function sPlace() { playSound(660, 0.15, 'sine', 0.2); }
function sWave() { playSound(330, 0.3, 'triangle', 0.2); }

// Game State
const STATE = { MENU: 0, PLAYING: 1, GAMEOVER: 2 };
let gameState = STATE.MENU;
let health = 20;
let maxHealth = 20;
let currency = 0;
let wave = 0;
let enemies = [];
let paddles = [];
let particles = [];
let floatTexts = [];
let shakeTime = 0;
let shakeMag = 0;

// Paddle columns (x positions as fractions)
const COLS = [0.15, 0.3, 0.45, 0.6, 0.75];
const PADDLE_W = 12;
const PADDLE_H = 60;

// Enemy config
function getEnemyConfig(wave) {
    const hp = 20 + wave * 8 + Math.floor(Math.random() * wave * 4);
    const speed = 0.8 + wave * 0.15 + Math.random() * 0.3;
    const r = 12 + Math.min(wave, 15);
    return { hp, maxHp: hp, speed, r };
}

// Wave system
let waveTimer = 0;
let spawnTimer = 0;
let enemiesToSpawn = 0;
let betweenWaves = true;

function startWave() {
    wave++;
    sWave();
    enemiesToSpawn = 3 + Math.floor(wave * 1.5);
    spawnTimer = 0;
    betweenWaves = false;
}

function spawnEnemy() {
    const cfg = getEnemyConfig(wave);
    const y = 80 + Math.random() * (H - 200);
    enemies.push({
        x: W + 30,
        y,
        vy: 0,
        vx: -cfg.speed,
        r: cfg.r,
        hp: cfg.hp,
        maxHp: cfg.maxHp,
        hitFlash: 0,
        wobble: Math.random() * Math.PI * 2
    });
}

// Paddle management
function placePaddle(colIdx) {
    const x = COLS[colIdx] * W;
    if (paddles.find(p => p.col === colIdx)) return false;
    if (currency < 50) return false;
    currency -= 50;
    paddles.push({
        col: colIdx,
        x,
        y: H / 2,
        vy: 0,
        level: 1,
        power: 1
    });
    sPlace();
    return true;
}

function upgradePaddle(colIdx) {
    const p = paddles.find(pp => pp.col === colIdx);
    if (!p || currency < 30) return false;
    currency -= 30;
    p.level++;
    p.power += 0.5;
    sPlace();
    return true;
}

// Particles
function burst(x, y, color, count) {
    for (let i = 0; i < count; i++) {
        const a = Math.random() * Math.PI * 2;
        const sp = 1 + Math.random() * 4;
        particles.push({
            x, y,
            vx: Math.cos(a) * sp,
            vy: Math.sin(a) * sp,
            life: 1,
            decay: 0.02 + Math.random() * 0.03,
            color,
            r: 2 + Math.random() * 3
        });
    }
}

function addFloatText(x, y, text, color) {
    floatTexts.push({ x, y, text, color, life: 1 });
}

// Input
let mouseX = 0, mouseY = 0;
window.addEventListener('mousemove', e => { mouseX = e.clientX; mouseY = e.clientY; });
window.addEventListener('click', e => {
    initAudio();
    if (gameState === STATE.MENU || gameState === STATE.GAMEOVER) {
        startGame();
        return;
    }
    // Try to place or upgrade paddle at nearest column
    let bestCol = -1, bestDist = 80;
    for (let i = 0; i < COLS.length; i++) {
        const cx = COLS[i] * W;
        const d = Math.abs(e.clientX - cx);
        if (d < bestDist) { bestDist = d; bestCol = i; }
    }
    if (bestCol >= 0) {
        const existing = paddles.find(p => p.col === bestCol);
        if (existing) upgradePaddle(bestCol);
        else placePaddle(bestCol);
    }
});

// Touch support
window.addEventListener('touchstart', e => {
    initAudio();
    const t = e.touches[0];
    mouseX = t.clientX; mouseY = t.clientY;
    if (gameState === STATE.MENU || gameState === STATE.GAMEOVER) { startGame(); return; }
    let bestCol = -1, bestDist = 80;
    for (let i = 0; i < COLS.length; i++) {
        const cx = COLS[i] * W;
        const d = Math.abs(t.clientX - cx);
        if (d < bestDist) { bestDist = d; bestCol = i; }
    }
    if (bestCol >= 0) {
        const existing = paddles.find(p => p.col === bestCol);
        if (existing) upgradePaddle(bestCol);
        else placePaddle(bestCol);
    }
});

// Keyboard restart
window.addEventListener('keydown', e => {
    if (e.key === 'r' || e.key === 'R') startGame();
    if (e.key === ' ' || e.key === 'Enter') { initAudio(); if (gameState !== STATE.PLAYING) startGame(); }
});

function startGame() {
    gameState = STATE.PLAYING;
    health = 20;
    currency = 50;
    wave = 0;
    enemies = [];
    paddles = [];
    particles = [];
    floatTexts = [];
    betweenWaves = true;
    waveTimer = 60;
}

// Update
function update() {
    if (gameState !== STATE.PLAYING) return;

    // Wave management
    if (betweenWaves) {
        waveTimer--;
        if (waveTimer <= 0) startWave();
    } else {
        spawnTimer--;
        if (spawnTimer <= 0 && enemiesToSpawn > 0) {
            spawnEnemy();
            enemiesToSpawn--;
            spawnTimer = Math.max(20, 60 - wave * 3);
        }
        if (enemiesToSpawn === 0 && enemies.length === 0) {
            betweenWaves = true;
            waveTimer = 120;
            currency += 20 + wave * 5;
        }
    }

    // Paddle AI
    for (const p of paddles) {
        let targetY = H / 2;
        let nearest = null, minDist = Infinity;
        for (const e of enemies) {
            const d = Math.abs(e.x - p.x);
            if (d < 300 && d < minDist) { minDist = d; nearest = e; }
        }
        if (nearest) targetY = nearest.y;
        const speed = 3 + p.level * 0.5;
        const dy = targetY - p.y;
        if (Math.abs(dy) > 2) {
            p.vy += Math.sign(dy) * 0.8;
            p.vy = Math.max(-speed, Math.min(speed, p.vy));
        } else {
            p.vy *= 0.8;
        }
        p.y += p.vy;
        p.y = Math.max(40, Math.min(H - 40, p.y));
    }

    // Enemies
    for (let i = enemies.length - 1; i >= 0; i--) {
        const e = enemies[i];
        e.x += e.vx;
        e.wobble += 0.05;
        e.y += Math.sin(e.wobble) * 0.3;
        if (e.hitFlash > 0) e.hitFlash--;

        // Paddle collision
        for (const p of paddles) {
            const px = p.x - PADDLE_W / 2;
            if (e.x + e.r > px && e.x - e.r < px + PADDLE_W &&
                Math.abs(e.y - p.y) < PADDLE_H / 2 + e.r) {
                // Bounce!
                e.vx = Math.abs(e.vx) * 1.05;
                e.hp -= 5 * p.power;
                e.hitFlash = 10;
                sBounce();
                burst(e.x, e.y, '#ff6644', 8);
                addFloatText(e.x, e.y - 20, `-${Math.round(5 * p.power)}`, '#ffaa44');
                shakeTime = 3;
                shakeMag = 3;

                if (e.hp <= 0) {
                    sKill();
                    burst(e.x, e.y, '#44ff88', 20);
                    const reward = 15 + wave * 2;
                    currency += reward;
                    addFloatText(e.x, e.y - 35, `+${reward}`, '#44ff88');
                    enemies.splice(i, 1);
                }
            }
        }

        // Breach
        if (e.x - e.r < 0) {
            health -= Math.ceil(e.r / 5);
            sBreach();
            shakeTime = 10;
            shakeMag = 8;
            burst(10, e.y, '#ff2244', 15);
            addFloatText(30, e.y, `-${Math.ceil(e.r / 5)}`, '#ff2244');
            enemies.splice(i, 1);
        }

        // Off screen right (bounced away)
        if (e.x > W + 60) {
            burst(e.x, e.y, '#88ccff', 5);
            enemies.splice(i, 1);
        }
    }

    // Particles
    for (let i = particles.length - 1; i >= 0; i--) {
        const pt = particles[i];
        pt.x += pt.vx;
        pt.y += pt.vy;
        pt.vy += 0.1;
        pt.life -= pt.decay;
        if (pt.life <= 0) particles.splice(i, 1);
    }

    // Float texts
    for (let i = floatTexts.length - 1; i >= 0; i--) {
        floatTexts[i].y -= 1;
        floatTexts[i].life -= 0.02;
        if (floatTexts[i].life <= 0) floatTexts.splice(i, 1);
    }

    // Shake
    if (shakeTime > 0) shakeTime--;

    // Game over
    if (health <= 0) {
        health = 0;
        gameState = STATE.GAMEOVER;
    }
}

// Render
function render() {
    ctx.save();
    if (shakeTime > 0) {
        const sx = (Math.random() - 0.5) * shakeMag;
        const sy = (Math.random() - 0.5) * shakeMag;
        ctx.translate(sx, sy);
    }

    // Background
    ctx.fillStyle = '#0a0a1a';
    ctx.fillRect(-10, -10, W + 20, H + 20);

    // Court lines
    ctx.strokeStyle = 'rgba(60,80,120,0.3)';
    ctx.lineWidth = 1;
    for (let i = 0; i < COLS.length; i++) {
        const x = COLS[i] * W;
        ctx.beginPath();
        ctx.moveTo(x, 60);
        ctx.lineTo(x, H - 20);
        ctx.stroke();
    }
    // Base line
    ctx.strokeStyle = 'rgba(255,50,80,0.4)';
    ctx.lineWidth = 3;
    ctx.beginPath();
    ctx.moveTo(2, 60);
    ctx.lineTo(2, H - 20);
    ctx.stroke();

    if (gameState === STATE.MENU || gameState === STATE.GAMEOVER) {
        renderMenu();
    } else {
        renderGame();
    }

    ctx.restore();
}

function renderMenu() {
    ctx.fillStyle = '#fff';
    ctx.font = 'bold 52px monospace';
    ctx.textAlign = 'center';
    ctx.fillText('PONG DEFENSE', W / 2, H / 2 - 80);

    ctx.font = '18px monospace';
    ctx.fillStyle = '#aaa';
    ctx.fillText('Enemies march from the right. Place pong paddles to bounce them away!', W / 2, H / 2 - 30);
    ctx.fillText('Click a column to place a paddle (50 coins)', W / 2, H / 2 + 10);
    ctx.fillText('Click an existing paddle to upgrade it (30 coins)', W / 2, H / 2 + 40);
    ctx.fillText('Earn coins by destroying enemies. Survive the waves!', W / 2, H / 2 + 70);

    ctx.font = '24px monospace';
    ctx.fillStyle = '#4fc';
    const blink = Math.sin(Date.now() * 0.005) > 0;
    if (blink || gameState === STATE.GAMEOVER) {
        ctx.fillText(gameState === STATE.GAMEOVER ? '[ CLICK OR PRESS R TO RESTART ]' : '[ CLICK TO START ]', W / 2, H / 2 + 130);
    }

    if (gameState === STATE.GAMEOVER) {
        ctx.font = '28px monospace';
        ctx.fillStyle = '#ff4466';
        ctx.fillText(`WAVES SURVIVED: ${wave}`, W / 2, H / 2 + 180);
    }
}

function renderGame() {
    // Paddles
    for (const p of paddles) {
        const grad = ctx.createLinearGradient(p.x - PADDLE_W/2, 0, p.x + PADDLE_W/2, 0);
        const hue = 140 + p.level * 20;
        grad.addColorStop(0, `hsl(${hue},80%,40%)`);
        grad.addColorStop(0.5, `hsl(${hue},90%,70%)`);
        grad.addColorStop(1, `hsl(${hue},80%,40%)`);
        ctx.fillStyle = grad;
        ctx.fillRect(p.x - PADDLE_W/2, p.y - PADDLE_H/2, PADDLE_W, PADDLE_H);

        // Glow
        ctx.shadowColor = `hsl(${hue},100%,60%)`;
        ctx.shadowBlur = 10;
        ctx.fillRect(p.x - PADDLE_W/2, p.y - PADDLE_H/2, PADDLE_W, PADDLE_H);
        ctx.shadowBlur = 0;

        // Level indicator
        ctx.fillStyle = '#fff';
        ctx.font = '12px monospace';
        ctx.textAlign = 'center';
        ctx.fillText(`L${p.level}`, p.x, p.y + PADDLE_H/2 + 16);
    }

    // Enemies
    for (const e of enemies) {
        const flash = e.hitFlash > 0;
        ctx.beginPath();
        ctx.arc(e.x, e.y, e.r, 0, Math.PI * 2);
        if (flash) {
            ctx.fillStyle = '#fff';
        } else {
            const hpFrac = e.hp / e.maxHp;
            const r = Math.floor(180 + 75 * (1 - hpFrac));
            const g = Math.floor(40 + 60 * hpFrac);
            ctx.fillStyle = `rgb(${r},${g},40)`;
        }
        ctx.fill();

        // HP bar
        const bw = e.r * 2;
        ctx.fillStyle = 'rgba(0,0,0,0.5)';
        ctx.fillRect(e.x - bw/2, e.y - e.r - 10, bw, 4);
        ctx.fillStyle = '#4f8';
        ctx.fillRect(e.x - bw/2, e.y - e.r - 10, bw * (e.hp / e.maxHp), 4);

        // Angry face
        ctx.fillStyle = '#fff';
        const eyeY = e.y - e.r * 0.2;
        ctx.beginPath(); ctx.arc(e.x - e.r*0.3, eyeY, 2, 0, Math.PI*2); ctx.fill();
        ctx.beginPath(); ctx.arc(e.x + e.r*0.3, eyeY, 2, 0, Math.PI*2); ctx.fill();
    }

    // Particles
    for (const pt of particles) {
        ctx.globalAlpha = pt.life;
        ctx.fillStyle = pt.color;
        ctx.beginPath();
        ctx.arc(pt.x, pt.y, pt.r * pt.life, 0, Math.PI * 2);
        ctx.fill();
    }
    ctx.globalAlpha = 1;

    // Float texts
    for (const ft of floatTexts) {
        ctx.globalAlpha = ft.life;
        ctx.fillStyle = ft.color;
        ctx.font = 'bold 16px monospace';
        ctx.textAlign = 'center';
        ctx.fillText(ft.text, ft.x, ft.y);
    }
    ctx.globalAlpha = 1;

    // HUD
    ctx.fillStyle = 'rgba(0,0,0,0.7)';
    ctx.fillRect(0, 0, W, 50);

    ctx.font = 'bold 16px monospace';
    ctx.textAlign = 'left';

    // Health bar
    ctx.fillStyle = '#446';
    ctx.fillRect(10, 18, 200, 18);
    const hFrac = health / maxHealth;
    ctx.fillStyle = hFrac > 0.5 ? '#4f8' : hFrac > 0.25 ? '#fa4' : '#f44';
    ctx.fillRect(10, 18, 200 * hFrac, 18);
    ctx.fillStyle = '#fff';
    ctx.fillText(`${health}/${maxHealth}`, 12, 32);

    // Wave
    ctx.textAlign = 'center';
    ctx.fillStyle = '#8cf';
    ctx.font = 'bold 18px monospace';
    const waveText = betweenWaves ? `NEXT WAVE IN ${Math.ceil(waveTimer / 60)}` : `WAVE ${wave}`;
    ctx.fillText(waveText, W / 2, 32);

    // Currency
    ctx.textAlign = 'right';
    ctx.fillStyle = '#ff8';
    ctx.font = 'bold 18px monospace';
    ctx.fillText(`$${currency}`, W - 15, 32);

    // Column hints
    if (betweenWaves) {
        ctx.textAlign = 'center';
        ctx.font = '14px monospace';
        for (let i = 0; i < COLS.length; i++) {
            const x = COLS[i] * W;
            const has = paddles.find(p => p.col === i);
            if (!has) {
                ctx.fillStyle = 'rgba(100,255,150,0.4)';
                ctx.fillText(`$${50}`, x, 75);
            } else {
                ctx.fillStyle = 'rgba(255,200,100,0.4)';
                ctx.fillText(`UP $30`, x, 75);
            }
        }
    }

    // Hover highlight
    let hoverCol = -1;
    for (let i = 0; i < COLS.length; i++) {
        if (Math.abs(mouseX - COLS[i] * W) < 40) hoverCol = i;
    }
    if (hoverCol >= 0 && gameState === STATE.PLAYING) {
        const hx = COLS[hoverCol] * W;
        ctx.strokeStyle = 'rgba(255,255,255,0.3)';
        ctx.lineWidth = 2;
        ctx.strokeRect(hx - 20, 60, 40, H - 80);
    }
}

// Main loop
function loop() {
    update();
    render();
    requestAnimationFrame(loop);
}
loop();
</script>
