let audioCtx = null;
function initAudio() { if (!audioCtx) audioCtx = new (window.AudioContext || webkitAudioContext)(); }
function playSound(type) {
if (!audioCtx) return;
const osc = audioCtx.createOscillator();
const gain = audioCtx.createGain();
osc.connect(gain); gain.connect(audioCtx.destination);
if (type === 'shoot') { osc.type = 'square'; osc.frequency.setValueAtTime(200, audioCtx.currentTime); osc.frequency.exponentialRampToValueAtTime(80, audioCtx.currentTime + 0.1); gain.gain.setValueAtTime(0.08, audioCtx.currentTime); gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.1); osc.start(); osc.stop(audioCtx.currentTime + 0.1); }
else if (type === 'explosion') { osc.type = 'sawtooth'; osc.frequency.setValueAtTime(150, audioCtx.currentTime); osc.frequency.exponentialRampToValueAtTime(30, audioCtx.currentTime + 0.3); gain.gain.setValueAtTime(0.12, audioCtx.currentTime); gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.3); osc.start(); osc.stop(audioCtx.currentTime + 0.3); }
else if (type === 'build') { osc.type = 'sine'; osc.frequency.setValueAtTime(440, audioCtx.currentTime); osc.frequency.exponentialRampToValueAtTime(880, audioCtx.currentTime + 0.15); gain.gain.setValueAtTime(0.1, audioCtx.currentTime); gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.15); osc.start(); osc.stop(audioCtx.currentTime + 0.15); }
else if (type === 'hit') { osc.type = 'sawtooth'; osc.frequency.setValueAtTime(60, audioCtx.currentTime); gain.gain.setValueAtTime(0.2, audioCtx.currentTime); gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.4); osc.start(); osc.stop(audioCtx.currentTime + 0.4); }
else if (type === 'wave') { osc.type = 'sine'; osc.frequency.setValueAtTime(330, audioCtx.currentTime); osc.frequency.exponentialRampToValueAtTime(660, audioCtx.currentTime + 0.2); gain.gain.setValueAtTime(0.1, audioCtx.currentTime); gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.3); osc.start(); osc.stop(audioCtx.currentTime + 0.3); }
}

// Game State

running: false, planetHP: 100, maxPlanetHP: 100, credits: 150, wave: 0, kills: 0,

{ name: 'Laser', cost: 80, damage: 35, range: 220, fireRate: 70, color: '#f64', bulletSpeed: 12 },

function initStars() {
game.stars = [];
for (let i = 0; i < 200; i++) {
game.stars.push({ x: Math.random() * W, y: Math.random() * H, size: Math.random() * 2 + 0.5, speed: Math.random() * 0.3 + 0.1, twinkle: Math.random() * Math.PI * 2 });
}
}

function initPads() {
game.pads = [];
const rows = 3, cols = 4;
const startY = H * 0.25, endY = H * 0.65;
const startX = W * 0.15, endX = W * 0.85;
for (let r = 0; r < rows; r++) {
for (let c = 0; c < cols; c++) {
const x = startX + (endX - startX) * c / (cols - 1);
const y = startY + (endY - startY) * r / (rows - 1);
game.pads.push({ x: x + (Math.random() - 0.5) * 30, y: y + (Math.random() - 0.5) * 20, tower: null });
}
}

function resetGame() {
game.planetHP = 100; game.maxPlanetHP = 100; game.credits = 150; game.wave = 0; game.kills = 0;
game.asteroids = []; game.bullets = []; game.particles = []; game.towers = [];

function startWave() {
game.wave++;
game.betweenWaves = false;
const base = 3 + game.wave * 2;
game.waveAsteroidsLeft = Math.min(base, 30);
game.spawnInterval = Math.max(800, 2000 - game.wave * 150);
playSound('wave');
}

function spawnAsteroid() {
const isBoss = game.wave % 5 === 0 && game.waveAsteroidsLeft === 1;
const size = isBoss ? 40 + Math.random() * 10 : 12 + Math.random() * (game.wave > 5 ? 15 : 8);
const x = Math.random() * (W - 100) + 50;
const hp = isBoss ? 200 + game.wave * 20 : Math.floor(size * 2 + game.wave * 3);
const speed = isBoss ? 0.4 : 0.8 + Math.random() * 0.6 + game.wave * 0.1;
const drift = (Math.random() - 0.5) * 0.5;
game.asteroids.push({ x, y: -size, vx: drift, vy: speed, size, hp, maxHP: hp, rotation: Math.random() * Math.PI * 2, rotSpeed: (Math.random() - 0.5) * 0.03, vertices: generateAsteroidShape(size), isBoss, flash: 0 });
}

function generateAsteroidShape(size) {
const pts = [];
const n = 8 + Math.floor(Math.random() * 4);
for (let i = 0; i < n; i++) {
const angle = (i / n) * Math.PI * 2;
const r = size * (0.7 + Math.random() * 0.3);
pts.push({ x: Math.cos(angle) * r, y: Math.sin(angle) * r });
}
return pts;
}

function spawnParticles(x, y, color, count, speed) {
for (let i = 0; i < count; i++) {
const angle = Math.random() * Math.PI * 2;
const s = Math.random() * speed;
game.particles.push({ x, y, vx: Math.cos(angle) * s, vy: Math.sin(angle) * s, life: 1, decay: 0.02 + Math.random() * 0.03, size: 2 + Math.random() * 4, color });
}
}

ctx.beginPath();

ctx.fill();
}
}

function drawPlanet() {

const grd = ctx.createRadialGradient(px, py, 20, px, py, 80);
grd.addColorStop(0, 'rgba(68,170,255,0.3)');
grd.addColorStop(1, 'rgba(68,170,255,0)');
ctx.fillStyle = grd;
ctx.beginPath(); ctx.arc(px, py, 80, 0, Math.PI * 2); ctx.fill();
// Planet
const pg = ctx.createRadialGradient(px - 10, py - 10, 5, px, py, 45);
pg.addColorStop(0, '#6cf'); pg.addColorStop(0.7, '#28a'); pg.addColorStop(1, '#145');

// Atmosphere ring
ctx.strokeStyle = 'rgba(68,170,255,0.4)';

ctx.beginPath(); ctx.arc(px, py, 48, 0, Math.PI * 2); ctx.stroke();
// HP bar

ctx.strokeStyle = '#fff'; ctx.lineWidth = 1; ctx.strokeRect(px - bw/2, py + 55, bw, bh);

function drawPad(pad) {
if (pad.tower) return;
const pulse = 0.5 + Math.sin(game.time * 0.003 + pad.x) * 0.3;
ctx.strokeStyle = `rgba(68,170,255,${pulse})`;
ctx.lineWidth = 2;
ctx.beginPath(); ctx.arc(pad.x, pad.y, 18, 0, Math.PI * 2); ctx.stroke();
ctx.fillStyle = `rgba(68,170,255,${pulse * 0.2})`;
ctx.beginPath(); ctx.arc(pad.x, pad.y, 18, 0, Math.PI * 2); ctx.fill();
// Cross
ctx.beginPath();
ctx.moveTo(pad.x - 8, pad.y); ctx.lineTo(pad.x + 8, pad.y);
ctx.moveTo(pad.x, pad.y - 8); ctx.lineTo(pad.x, pad.y + 8);
ctx.stroke();
}

function drawTower(tower) {
const type = TOWER_TYPES[tower.type];
// Base
ctx.fillStyle = '#1a2a3a';
ctx.beginPath(); ctx.arc(tower.x, tower.y, 16, 0, Math.PI * 2); ctx.fill();
ctx.strokeStyle = type.color; ctx.lineWidth = 2;
ctx.stroke();
// Turret direction
const angle = tower.angle || 0;
ctx.save();
ctx.translate(tower.x, tower.y);
ctx.rotate(angle);
ctx.fillStyle = type.color;
ctx.fillRect(0, -3, 18 + tower.level * 2, 6);
ctx.restore();
// Level indicator
for (let i = 0; i < tower.level; i++) {
ctx.fillStyle = type.color;
ctx.beginPath(); ctx.arc(tower.x - 10 + i * 7, tower.y + 12, 3, 0, Math.PI * 2); ctx.fill();
}
// Range circle (faint)
ctx.strokeStyle = `rgba(${hexToRgb(type.color)},0.1)`;
ctx.lineWidth = 1;
ctx.beginPath(); ctx.arc(tower.x, tower.y, type.range + tower.level * 15, 0, Math.PI * 2); ctx.stroke();
}

function hexToRgb(hex) {
const r = parseInt(hex.slice(1,3), 16), g = parseInt(hex.slice(3,5), 16), b = parseInt(hex.slice(5,7), 16);
return `${r},${g},${b}`;
}

function drawAsteroid(a) {
ctx.save();
ctx.translate(a.x, a.y);
ctx.rotate(a.rotation);
const flash = a.flash > 0 ? 0.5 : 0;
ctx.fillStyle = flash ? '#fff' : (a.isBoss ? '#f44' : '#876');
ctx.strokeStyle = flash ? '#fff' : (a.isBoss ? '#f88' : '#a97');
ctx.lineWidth = a.isBoss ? 3 : 1.5;
ctx.beginPath();
const v = a.vertices;
ctx.moveTo(v[0].x, v[0].y);
for (let i = 1; i < v.length; i++) ctx.lineTo(v[i].x, v[i].y);
ctx.closePath();
ctx.fill(); ctx.stroke();
ctx.restore();
// HP bar for damaged asteroids
if (a.hp < a.maxHP) {
const w = a.size * 2;
ctx.fillStyle = '#333'; ctx.fillRect(a.x - w/2, a.y - a.size - 12, w, 5);
ctx.fillStyle = a.isBoss ? '#f44' : '#4f8';
ctx.fillRect(a.x - w/2, a.y - a.size - 12, w * (a.hp / a.maxHP), 5);
}
}

function drawBullet(b) {
ctx.fillStyle = b.color;
ctx.shadowColor = b.color; ctx.shadowBlur = 8;
ctx.beginPath(); ctx.arc(b.x, b.y, b.size, 0, Math.PI * 2); ctx.fill();
ctx.shadowBlur = 0;
}

function drawParticle(p) {
const alpha = p.life;
ctx.fillStyle = p.color.replace(')', `,${alpha})`).replace('rgb', 'rgba');
ctx.beginPath(); ctx.arc(p.x, p.y, p.size * p.life, 0, Math.PI * 2);
ctx.fill();
}

ctx.fillStyle = 'rgba(10,10,30,0.8)';

ctx.strokeStyle = 'rgba(68,170,255,0.3)'; ctx.lineWidth = 1;

function update(dt) {
game.time += dt;
// Shake decay
if (game.shake > 0) game.shake *= 0.9;

// Wave management
if (game.betweenWaves) {
game.waveCountdown -= dt / 1000;
if (game.waveCountdown <= 0) startWave();
return;
}

// Spawn asteroids
game.spawnTimer += dt;
if (game.waveAsteroidsLeft > 0 && game.spawnTimer >= game.spawnInterval) {
game.spawnTimer = 0;
spawnAsteroid();
game.waveAsteroidsLeft--;
}
if (game.waveAsteroidsLeft === 0 && game.asteroids.length === 0) {
game.betweenWaves = true;
game.waveCountdown = 4;
game.credits += 30 + game.wave * 10;
}

// Asteroids
for (let i = game.asteroids.length - 1; i >= 0; i--) {
const a = game.asteroids[i];
a.x += a.vx; a.y += a.vy;
a.rotation += a.rotSpeed;
if (a.flash > 0) a.flash -= 0.1;
// Check planet hit
const dx = a.x - W/2, dy = a.y - (H - 60);
if (Math.sqrt(dx*dx + dy*dy) < 50 + a.size * 0.3) {
game.planetHP -= Math.ceil(a.size / 3);
game.shake = 15;
playSound('hit');
spawnParticles(a.x, a.y, 'rgb(255,100,50)', 15, 4);
game.asteroids.splice(i, 1);
if (game.planetHP <= 0) { gameOver(); return; }
continue;
}
// Off screen sides
if (a.y > H + 50 || a.x < -100 || a.x > W + 100) {
game.asteroids.splice(i, 1);
}
}

// Towers shoot
for (const tower of game.towers) {
tower.cooldown -= dt;
if (tower.cooldown <= 0) {
const type = TOWER_TYPES[tower.type];
const range = type.range + tower.level * 15;
let target = null, bestDist = Infinity;
for (const a of game.asteroids) {
const d = Math.sqrt((a.x - tower.x)**2 + (a.y - tower.y)**2);
if (d < range && d < bestDist) { bestDist = d; target = a; }
}
if (target) {
const angle = Math.atan2(target.y - tower.y, target.x - tower.x);
tower.angle = angle;
const dmg = type.damage * (1 + tower.level * 0.3);
game.bullets.push({ x: tower.x, y: tower.y, vx: Math.cos(angle) * type.bulletSpeed, vy: Math.sin(angle) * type.bulletSpeed, damage: dmg, color: type.color, size: tower.type === 1 ? 5 : 3 });
tower.cooldown = (type.fireRate - tower.level * 3);
playSound('shoot');
}
}
}

// Bullets
for (let i = game.bullets.length - 1; i >= 0; i--) {
const b = game.bullets[i];
b.x += b.vx; b.y += b.vy;
if (b.x < -20 || b.x > W + 20 || b.y < -20 || b.y > H + 20) {
game.bullets.splice(i, 1); continue;
}
// Hit asteroid
for (let j = game.asteroids.length - 1; j >= 0; j--) {
const a = game.asteroids[j];
const dx = b.x - a.x, dy = b.y - a.y;
if (Math.sqrt(dx*dx+dy*dy) < a.size + b.size) {
a.hp -= b.damage;
a.flash = 1;
game.bullets.splice(i, 1);
spawnParticles(b.x, b.y, 'rgb(255,200,100)', 3, 2);
if (a.hp <= 0) {
const reward = Math.ceil(a.size / 2) + (a.isBoss ? 50 : 0);
game.credits += reward;
game.kills++;
playSound('explosion');
spawnParticles(a.x, a.y, 'rgb(255,150,50)', a.isBoss ? 30 : 12, 4);
game.asteroids.splice(j, 1);
}
break;
}
}
}

// Particles
for (let i = game.particles.length - 1; i >= 0; i--) {
const p = game.particles[i];
p.x += p.vx; p.y += p.vy;
p.vy += 0.1;
p.life -= p.decay;
if (p.life <= 0) game.particles.splice(i, 1);
}
}

function draw() {
ctx.save();
if (game.shake > 0.5) {
ctx.translate((Math.random()-0.5)*game.shake, (Math.random()-0.5)*game.shake);
}
// Background
const bg = ctx.createLinearGradient(0, 0, 0, H);
bg.addColorStop(0, '#0a0a1a'); bg.addColorStop(1, '#141428');
ctx.fillStyle = bg;
ctx.fillRect(-20, -20, W+40, H+40);

drawStarfield();

// Pads
for (const pad of game.pads) drawPad(pad);

// Towers
for (const t of game.towers) drawTower(t);

// Asteroids
for (const a of game.asteroids) drawAsteroid(a);

// Bullets
for (const b of game.bullets) drawBullet(b);

// Particles
for (const p of game.particles) drawParticle(p);

drawPlanet();
drawHUD();
ctx.restore();
}

function loop(timestamp) {
if (!game.running) return;
const dt = Math.min(timestamp - game.lastTime, 50);
game.lastTime = timestamp;
update(dt);
draw();
requestAnimationFrame(loop);
}

function gameOver() {
game.running = false;
document.getElementById('final-wave').textContent = game.wave;
document.getElementById('final-kills').textContent = game.kills;
document.getElementById('gameover-screen').classList.remove('hidden');
}

function startGame() {
initAudio();
resetGame();
initStars();
game.running = true;
game.lastTime = performance.now();
document.getElementById('start-screen').classList.add('hidden');
document.getElementById('gameover-screen').classList.add('hidden');
requestAnimationFrame(loop);
}

// Input

if (!game.running) return;
if (e.key === '1') game.selectedTowerType = 0;
if (e.key === '2') game.selectedTowerType = 1;
if (e.key === '3') game.selectedTowerType = 2;
});

canvas.addEventListener('click', (e) => {
if (!game.running) return;
const rect = canvas.getBoundingClientRect();
const mx = e.clientX - rect.left, my = e.clientY - rect.top;
// Check pads
