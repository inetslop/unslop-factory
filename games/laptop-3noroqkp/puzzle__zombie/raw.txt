function playTone(freq, dur, type, vol) {
  if (!audioCtx) return;

  osc.connect(gain);
  gain.connect(audioCtx.destination);

  osc.stop(audioCtx.currentTime + dur);
}

function drawPath(path, color) {

  ctx.beginPath();
  const first = path[0];
  ctx.moveTo(offsetX + first[1]*cellSize + cellSize/2, offsetY + first[0]*cellSize + cellSize/2);
  for (let i = 1; i < path.length; i++) {
    ctx.lineTo(offsetX + path[i][1]*cellSize + cellSize/2, offsetY + path[i][0]*cellSize + cellSize/2);
  }
  ctx.strokeStyle = color;
  ctx.lineWidth = cellSize * 0.5;
  ctx.lineCap = 'round';
  ctx.lineJoin = 'round';
  ctx.stroke();
}

function getCellFromPos(px, py) {
  const c = Math.floor((px - offsetX) / cellSize);
  const r = Math.floor((py - offsetY) / cellSize);
  if (r < 0 || r >= gridSize || c < 0 || c >= gridSize) return null;
  return [r, c];

function isAdjacent(a, b) {
  return Math.abs(a[0]-b[0]) + Math.abs(a[1]-b[1]) === 1;
}

function inPath(path, r, c) {
  for (let i = 0; i < path.length; i++) {
    if (path[i][0] === r && path[i][1] === c) return true;
  }
  return false;
}

function canExtend(path, otherPath, r, c) {
  // Can't go on other path
  if (inPath(otherPath, r, c)) return false;
  // Can't revisit same path (except backtracking)
  if (path.length > 0 && inPath(path, r, c) && !(path.length === 1)) {
    // Allow backtracking to last cell
    const last = path[path.length-1];
    if (!(path.length === 2 && path[0][0]===r && path[0][1]===c)) return false;
  }
  if (inPath(path, r, c) && path.length > 1) {
    const last = path[path.length-1];
    if (path.length === 2 && path[0][0]===r && path[0][1]===c) return true; // backtrack
    return false;
  }
  return true;
}

function handleStart(r, c) {
  const val = LEVELS[levelIdx].grid[r][c];
  if (val === 1) {
    currentPath = 'red';
    redPath = [[r,c]];
    isDrawing = true;
    playDraw();
  } else if (val === 2) {
    currentPath = 'blue';
    bluePath = [[r,c]];
    isDrawing = true;
    playDraw();
  } else {
    // Empty cell - check if adjacent to existing path
    if (redPath.length > 0 && canExtend(redPath, bluePath, r, c)) {
      const last = redPath[redPath.length-1];
      if (isAdjacent(last, [r,c])) {
        currentPath = 'red';
        redPath.push([r,c]);
        isDrawing = true;
        playDraw();
        return;
      }
    }
    if (bluePath.length > 0 && canExtend(bluePath, redPath, r, c)) {
      const last = bluePath[bluePath.length-1];
      if (isAdjacent(last, [r,c])) {
        currentPath = 'blue';
        bluePath.push([r,c]);
        isDrawing = true;
        playDraw();
        return;
      }
    }
  }
}

function handleExtend(r, c) {
  if (!currentPath) return;
  const path = currentPath === 'red' ? redPath : bluePath;
  const other = currentPath === 'red' ? bluePath : redPath;
  const last = path[path.length-1];
  
  if (!isAdjacent(last, [r,c])) return;
  
  // Backtrack
  if (path.length >= 2) {
    const prev = path[path.length-2];
    if (prev[0]===r && prev[1]===c) {
      path.pop();
      playDraw();
      drawBoard();
      return;
    }
  }
  
  if (!canExtend(path, other, r, c)) {
    playError();
    return;
  }
  
  path.push([r,c]);
  playDraw();
  drawBoard();
}

function handleEnd() {
  isDrawing = false;
  currentPath = null;
  checkWin();

function checkWin() {
  const grid = LEVELS[levelIdx].grid;
  
  // Check all reds connected
  let redCount = 0, blueCount = 0;
  for (let r = 0; r < gridSize; r++) {
    for (let c = 0; c < gridSize; c++) {
      if (grid[r][c] === 1) {
        redCount++;
        if (!inPath(redPath, r, c)) return false;
      } else if (grid[r][c] === 2) {
        blueCount++;
        if (!inPath(bluePath, r, c)) return false;
      }
    }
  }
  
  // Check paths are valid (connected chains)
  if (!isValidPath(redPath)) return false;
  if (!isValidPath(bluePath)) return false;
  
  // Calculate efficiency
  const totalCells = redPath.length + bluePath.length;
  const minCells = redCount + blueCount - 2; // minimum is just the dots minus overlap
  const maxCells = gridSize * gridSize;
  const efficiency = Math.round(((maxCells - totalCells) / (maxCells - minCells)) * 100);
  
  document.getElementById('efficiencyDisplay').textContent = 'Efficiency: ' + Math.min(efficiency, 100) + '%';
  playComplete();
  
  setTimeout(() => {
    document.getElementById('winStats').textContent = 
      `Brains used: ${redPath.length} | Skulls used: ${bluePath.length}\nTotal cells: ${totalCells}`;
    document.getElementById('winScreen').style.display = 'flex';
  }, 500);
  
  return true;
}

function isValidPath(path) {
  if (path.length === 0) return false;
  for (let i = 1; i < path.length; i++) {
    if (!isAdjacent(path[i-1], path[i])) return false;
  }
  return true;
}

function undo() {
  const path = currentPath === 'red' ? redPath : bluePath;
  // Try to undo last segment of whichever path was most recently drawn
  if (redPath.length > 1) {
    redPath.pop();
  } else if (bluePath.length > 1) {
    bluePath.pop();
  } else if (redPath.length === 1) {
    redPath = [];
  } else if (bluePath.length === 1) {
    bluePath = [];
  }
  drawBoard();
}

  if (!gameActive || !isDrawing) return;
  const [x,y] = getCanvasPos(e);
  const cell = getCellFromPos(x, y);
  if (!cell) return;
  handleExtend(cell[0], cell[1]);
});

canvas.addEventListener('touchend', (e) => {
  e.preventDefault();
  if (!gameActive) return;
  handleEnd();
});

window.addEventListener('keydown', (e) => {
  if (!gameActive) return;
  const key = e.key.toLowerCase();
  if (key === 'r') resetLevel();
  if (key === 'u') undo();
  if (key === 'n') nextLevel();
});

document.getElementById('undoBtn').addEventListener('click', undo);
document.getElementById('resetBtn').addEventListener('click', resetLevel);
document.getElementById('nextLevelBtn').addEventListener('click', nextLevel);
document.getElementById('startBtn').addEventListener('click', () => {
  initAudio();
  document.getElementById('startScreen').style.display = 'none';
  gameActive = true;
  loadLevel(0);
});

window.addEventListener('resize', () => {
  if (gameActive) {
    const s = sizeCanvas();
    cellSize = Math.floor(s / (gridSize + 1));
    offsetX = Math.floor((s - cellSize * gridSize) / 2);
    offsetY = Math.floor((s - cellSize * gridSize) / 2);
    drawBoard();
  }
});
