const canvas = document.getElementById('game');
const ctx = canvas.getContext('2d');

let W, H, CELL, COLS, ROWS;
function resize() {
  W = window.innerWidth;
  H = window.innerHeight;
  canvas.width = W;
  canvas.height = H;
  const aspect = 16/9;
  if (W/H > aspect) {
    CELL = Math.floor(H / 20);
    COLS = Math.floor(W / CELL);
    ROWS = 20;
  } else {
    CELL = Math.floor(W / 18);
    COLS = 18;
    ROWS = Math.floor(H / CELL);
  }
}
resize();
window.addEventListener('resize', () => { resize(); if (gameState === 'playing') generateMaze(); });

let audioCtx = null;
function initAudio() {
  if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
}
function playSound(freq, dur, type, vol) {
  if (!audioCtx) return;
  const osc = audioCtx.createOscillator();
  const gain = audioCtx.createGain();
  osc.type = type || 'square';
  osc.frequency.value = freq;
  gain.gain.setValueAtTime(vol || 0.1, audioCtx.currentTime);
  gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
  osc.connect(gain);
  gain.connect(audioCtx.destination);
  osc.start();
  osc.stop(audioCtx.currentTime + dur);
}
function shootSound() { playSound(880, 0.1, 'square', 0.08); }
function hitSound() { playSound(200, 0.2, 'sawtooth', 0.15); }
function explodeSound() { playSound(100, 0.4, 'sawtooth', 0.2); playSound(60, 0.3, 'square', 0.1); }
function waveSound() { playSound(440, 0.3, 'sine', 0.1); setTimeout(() => playSound(660, 0.3, 'sine', 0.1), 150); }

let gameState = 'start'; // start, playing, gameover, waveclear
let score = 0;
let lives = 3;
let wave = 1;
let maze = [];
let player = {};
let aliens = [];
let bullets = [];
let enemyBullets = [];
let particles = [];
let keys = {};
let lastShot = 0;
let waveTimer = 0;
let alienDir = 1;
let alienSpeed = 2;
let frameCount = 0;

function generateMaze() {
  maze = Array.from({length: ROWS}, () => Array(COLS).fill(1));
  // Recursive backtracker
  const stack = [];
  const sx = Math.floor(COLS / 2);
  maze[ROWS - 1][sx] = 0;
  stack.push([sx, ROWS - 1]);
  while (stack.length > 0) {
    const [cx, cy] = stack[stack.length - 1];
    const neighbors = [];
    if (cx > 0 && maze[cy][cx-1] === 1) neighbors.push([cx-1, cy]);
    if (cx < COLS-1 && maze[cy][cx+1] === 1) neighbors.push([cx+1, cy]);
    if (cy > 0 && maze[cy-1][cx] === 1) neighbors.push([cx, cy-1]);
    if (cy < ROWS-1 && maze[cy+1][cx] === 1) neighbors.push([cx, cy+1]);
    if (neighbors.length > 0) {
      const [nx, ny] = neighbors[Math.floor(Math.random() * neighbors.length)];
      maze[ny][nx] = 0;
      maze[cy + (ny - cy)/2][cx + (nx - cx)/2] = 0;
      stack.push([nx, ny]);
    } else {
      stack.pop();
    }
  }
  // Open extra passages for gameplay
  const openCount = Math.floor(COLS * ROWS * 0.15);
  for (let i = 0; i < openCount; i++) {
    const x = Math.floor(Math.random() * COLS);
    const y = Math.floor(Math.random() * ROWS);
    if (x > 0 && x < COLS-1) maze[y][x] = 0;
  }
  // Ensure bottom row is mostly open for player start
  for (let x = 0; x < COLS; x++) {
    if (Math.random() < 0.7) maze[ROWS-1][x] = 0;
  }
}

function resetPlayer() {
  player = {
    x: (COLS / 2) * CELL + CELL/2,
    y: (ROWS - 1) * CELL + CELL/2,
    w: CELL * 0.7,
    h: CELL * 0.5,
    speed: 4
  };
}

function spawnAliens() {
  aliens = [];
  const rows = Math.min(4, 3 + Math.floor(wave / 2));
  const cols = Math.min(8, 6 + wave);
  const startX = (COLS - cols) / 2;
  for (let r = 0; r < rows; r++) {
    for (let c = 0; c < cols; c++) {
      aliens.push({
        x: (startX + c) * CELL + CELL/2,
        y: (r + 1) * CELL + CELL/2,
        w: CELL * 0.6,
        h: CELL * 0.5,
        type: r % 3,
        alive: true
      });
    }
  }
  alienDir = 1;
  alienSpeed = Math.min(2 + wave * 0.5, 5);
}

  resetPlayer();
  spawnAliens();
  bullets = [];
  enemyBullets = [];
  particles = [];
  gameState = 'playing';
}

function isWall(x, y) {
  const gx = Math.floor(x / CELL);
  const gy = Math.floor(y / CELL);
  if (gx < 0 || gx >= COLS || gy < 0 || gy >= ROWS) return true;
  return maze[gy][gx] === 1;
}

function bulletHitsWall(bx, by) {
  return isWall(bx, by);
}

function updatePlayer(dt) {
  let dx = 0, dy = 0;
  if (keys['ArrowLeft'] || keys['KeyA']) dx = -1;
  if (keys['ArrowRight'] || keys['KeyD']) dx = 1;
  if (keys['ArrowUp'] || keys['KeyW']) dy = -1;
  if (keys['ArrowDown'] || keys['KeyS']) dy = 1;
  
  let nx = player.x + dx * player.speed;
  let ny = player.y + dy * player.speed;
  
  // Check wall collision for movement
  const checkX = isWall(nx - player.w/2, player.y) || isWall(nx + player.w/2, player.y);
  const checkY = isWall(player.x, ny - player.h/2) || isWall(player.x, ny + player.h/2);
  
  if (!checkX) player.x = nx;
  if (!checkY) player.y = ny;
  
  player.x = Math.max(player.w/2, Math.min(W - player.w/2, player.x));
  player.y = Math.max(player.h/2, Math.min(H - player.h/2, player.y));
  
  // Shooting
  if (keys['Space'] && frameCount - lastShot > 15) {
    bullets.push({ x: player.x, y: player.y - player.h/2, vy: -7, w: 3, h: 10 });
    lastShot = frameCount;
    shootSound();
  }
}

function updateBullets() {
  for (let i = bullets.length - 1; i >= 0; i--) {
    const b = bullets[i];
    b.y += b.vy;
    if (b.y < 0) { bullets.splice(i, 1); continue; }
    // Check wall collision
    if (isWall(b.x, b.y)) {
      spawnParticles(b.x, b.y, '#4af', 3);
      bullets.splice(i, 1);
      continue;
    }
    // Check alien hit
    for (let a of aliens) {
      if (!a.alive) continue;
      if (Math.abs(b.x - a.x) < (b.w + a.w)/2 && Math.abs(b.y - a.y) < (b.h + a.h)/2) {
        a.alive = false;
        score += 10 * wave;
        bullets.splice(i, 1);
        spawnParticles(a.x, a.y, '#f44', 8);
        hitSound();
        break;
      }
    }
  }
  
  for (let i = enemyBullets.length - 1; i >= 0; i--) {
    const b = enemyBullets[i];
    b.y += b.vy;
    if (b.y > H) { enemyBullets.splice(i, 1); continue; }
    if (isWall(b.x, b.y)) {
      spawnParticles(b.x, b.y, '#f4a', 3);
      enemyBullets.splice(i, 1);
      continue;
    }
    // Hit player
    if (Math.abs(b.x - player.x) < (b.w + player.w)/2 && Math.abs(b.y - player.y) < (b.h + player.h)/2) {
      enemyBullets.splice(i, 1);
      lives--;
      explodeSound();
      spawnParticles(player.x, player.y, '#4af', 15);
      if (lives <= 0) gameState = 'gameover';
    }
  }
}

function updateAliens() {
  let aliveCount = aliens.filter(a => a.alive).length;
  if (aliveCount === 0) {
    gameState = 'waveclear';
    waveTimer = 120;
    wave++;
    waveSound();
    return;
  }
  
  frameCount++;
  const interval = Math.max(3, 8 - wave);
  if (frameCount % interval !== 0) return;
  
  let minDistX = Infinity, maxDistX = 0, minDistY = Infinity;
  for (let a of aliens) {
    if (!a.alive) continue;
    const edgeX = alienDir > 0 ? a.x + a.w/2 : a.x - a.w/2;
    minDistX = Math.min(minDistX, W - edgeX);
    maxDistX = Math.max(maxDistX, edgeX);
    minDistY = Math.min(minDistY, H - a.y);
  }
  
  if (minDistX < CELL * 0.5 || maxDistX > W - CELL * 0.5) {
    alienDir *= -1;
    for (let a of aliens) { if (a.alive) a.y += CELL * 0.3; }
  } else {
    for (let a of aliens) { if (a.alive) a.x += alienDir * alienSpeed; }
  }
  
  // Aliens reaching bottom = game over
  for (let a of aliens) {
    if (a.alive && a.y > H - CELL) {
      gameState = 'gameover';
      return;
    }
  }
  
  // Enemy shooting
  const shootChance = Math.min(0.1 + wave * 0.02, 0.3);
  if (Math.random() < shootChance) {
    const alive = aliens.filter(a => a.alive);
    if (alive.length > 0) {
      const shooter = alive[Math.floor(Math.random() * alive.length)];
      enemyBullets.push({ x: shooter.x, y: shooter.y + shooter.h/2, vy: 3 + wave * 0.3, w: 3, h: 8 });
    }
  }
}

function spawnParticles(x, y, color, count) {
  for (let i = 0; i < count; i++) {
    const angle = Math.random() * Math.PI * 2;
    const speed = 1 + Math.random() * 3;
    particles.push({
      x, y,
      vx: Math.cos(angle) * speed,
      vy: Math.sin(angle) * speed,
      life: 30,
      color
    });
  }
}

function updateParticles() {
  for (let i = particles.length - 1; i >= 0; i--) {
    const p = particles[i];
    p.x += p.vx;
    p.y += p.vy;
    p.life--;
    if (p.life <= 0) particles.splice(i, 1);
  }
}

function drawMaze() {
  ctx.fillStyle = '#1a1a2e';
  ctx.fillRect(0, 0, W, H);
  
  for (let y = 0; y < ROWS; y++) {
    for (let x = 0; x < COLS; x++) {
      if (maze[y][x] === 1) {
        ctx.fillStyle = '#3a3a6e';
        ctx.fillRect(x * CELL, y * CELL, CELL, CELL);
        ctx.fillStyle = '#2a2a4e';
        ctx.fillRect(x * CELL + 2, y * CELL + 2, CELL - 4, CELL - 4);
      }
    }
  }
  
  // Grid lines for style
  ctx.strokeStyle = 'rgba(100, 100, 200, 0.1)';
  ctx.lineWidth = 1;
  for (let x = 0; x <= COLS; x++) {
    ctx.beginPath();
    ctx.moveTo(x * CELL, 0);
    ctx.lineTo(x * CELL, H);
    ctx.stroke();
  }
  for (let y = 0; y <= ROWS; y++) {
    ctx.beginPath();
    ctx.moveTo(0, y * CELL);
    ctx.lineTo(W, y * CELL);
    ctx.stroke();
  }
}

function drawPlayer() {
  const px = player.x, py = player.y;
  ctx.fillStyle = '#4af';
  ctx.beginPath();
  ctx.moveTo(px, py - player.h/2);
  ctx.lineTo(px + player.w/2, py + player.h/2);
  ctx.lineTo(px - player.w/2, py + player.h/2);
  ctx.closePath();
  ctx.fill();
  
  // Engine glow
  ctx.fillStyle = 'rgba(100, 200, 255, 0.5)';
  ctx.beginPath();
  ctx.arc(px, py + player.h/2, 4, 0, Math.PI * 2);
  ctx.fill();
}

function drawAliens() {
  const colors = ['#f44', '#4f4', '#ff4'];
  for (let a of aliens) {
    if (!a.alive) continue;
    const c = colors[a.type];
    ctx.fillStyle = c;
    // Draw alien shape based on type
    if (a.type === 0) {
      // Crab-like
      ctx.fillRect(a.x - a.w/2, a.y - a.h/4, a.w/4, a.h/2);
      ctx.fillRect(a.x + a.w/4, a.y - a.h/4, a.w/4, a.h/2);
      ctx.fillRect(a.x - a.w/4, a.y - a.h/2, a.w/2, a.h/4);
    } else if (a.type === 1) {
      // Squid-like
      ctx.beginPath();
      ctx.arc(a.x, a.y - a.h/4, a.w/3, 0, Math.PI * 2);
      ctx.fill();
      ctx.fillRect(a.x - a.w/2, a.y + a.h/4, a.w/4, a.h/4);
      ctx.fillRect(a.x + a.w/4, a.y + a.h/4, a.w/4, a.h/4);
    } else {
      // Diamond
      ctx.beginPath();
      ctx.moveTo(a.x, a.y - a.h/2);
      ctx.lineTo(a.x + a.w/2, a.y);
      ctx.lineTo(a.x, a.y + a.h/2);
      ctx.lineTo(a.x - a.w/2, a.y);
      ctx.closePath();
      ctx.fill();
    }
  }
}

function drawBullets() {
  ctx.fillStyle = '#4af';
  for (let b of bullets) {
    ctx.fillRect(b.x - b.w/2, b.y - b.h/2, b.w, b.h);
  }
  ctx.fillStyle = '#f4a';
  for (let b of enemyBullets) {
    ctx.fillRect(b.x - b.w/2, b.y - b.h/2, b.w, b.h);
  }
}

function drawParticles() {
  for (let p of particles) {
    ctx.globalAlpha = p.life / 30;
    ctx.fillStyle = p.color;
    ctx.fillRect(p.x - 2, p.y - 2, 4, 4);
  }
  ctx.globalAlpha = 1;
}

    ctx.fill();
  }
}

function gameLoop() {
  frameCount++;
  
  if (gameState === 'start') {
    drawStartScreen();
  } else if (gameState === 'playing') {
    updatePlayer();
    updateAliens();
    updateBullets();
    updateParticles();
    
    drawMaze();
    drawAliens();
    drawPlayer();
    drawBullets();
    drawParticles();
    drawHUD();
  } else if (gameState === 'gameover') {
    drawMaze();
    drawAliens();
    drawPlayer();
    drawBullets();
    drawGameOver();
  } else if (gameState === 'waveclear') {
