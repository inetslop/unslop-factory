function isWall(gx, gy) {
  if (gx < 0 || gx >= 15 || gy < 0 || gy >= 15) return true;
  return maze[gy][gx] === 1;
}

function canMove(gx, gy) {
  return !isWall(gx, gy);
}

function doTick() {
  // Apply gravity to pacman
  const gd = GRAV_DIRS[gravityDir];
  if (canMove(pacman.gx + gd.x, pacman.gy + gd.y)) {
    pacman.gx += gd.x;
    pacman.gy += gd.y;
  }

  // Check pellet
  if (maze[pacman.gy][pacman.gx] === 2) {
    // check if pellet exists at this position
    const idx = pellets.findIndex(p => p.gx===pacman.gx && p.gy===pacman.gy);
    if (idx !== -1) {
      pellets.splice(idx, 1);
      pelletsLeft--;
      score += 10;
      playEat();
    }
  }

  // Move ghosts
  for (let g of ghosts) {
    const gd2 = GRAV_DIRS[gravityDir];
    if (canMove(g.gx + gd2.x, g.gy + gd2.y)) {
      g.gx += gd2.x;
      g.gy += gd2.y;
    }

    // Ghost AI: move toward pacman (one step)
    const dx = pacman.gx - g.gx;
    const dy = pacman.gy - g.gy;
    let moves = [];
    if (canMove(g.gx+1, g.gy)) moves.push({x:1,y:0});
    if (canMove(g.gx-1, g.gy)) moves.push({x:-1,y:0});
    if (canMove(g.gx, g.gy+1)) moves.push({x:0,y:1});
    if (canMove(g.gx, g.gy-1)) moves.push({x:0,y:-1});
    
    if (moves.length > 0) {
      // Prefer direction toward pacman
      let best = null;
      let bestScore = -999;
      for (let m of moves) {
        const nd = Math.abs(pacman.gx-(g.gx+m.x)) + Math.abs(pacman.gy-(g.gy+m.y));
        const s = -nd + Math.random()*2;
        if (s > bestScore) { bestScore = s; best = m; }
      }
      if (best) { g.gx += best.x; g.gy += best.y; }
    }

    // Check collision with pacman
    if (g.gx === pacman.gx && g.gy === pacman.gy) {
      gameState = 'dead';
      playDeath();
      return;
    }
  }

  // Win check
  if (pelletsLeft <= 0) {
    gameState = 'won';
    playWin();
  }
}

let inputQueue = [];
window.addEventListener('keydown', (e) => {
  initAudio();
  if (gameState === 'start') {
    if (e.key === 'Enter' || e.key === ' ') {
      initGame();
    }
    return;
  }
  if (gameState === 'dead' || gameState === 'won') {
    if (e.key === 'r' || e.key === 'R') {
      initGame();
    }
    return;
  }
  // Playing
  const k = e.key.toLowerCase();
  if (k === 'arrowup' || k === 'w') inputQueue.push({x:0,y:-1});
  else if (k === 'arrowdown' || k === 's') inputQueue.push({x:0,y:1});
  else if (k === 'arrowleft' || k === 'a') inputQueue.push({x:-1,y:0});
  else if (k === 'arrowright' || k === 'd') inputQueue.push({x:1,y:0});
  else if (k === 'q') { gravityDir = (gravityDir + 3) % 4; }
  else if (k === 'e') { gravityDir = (gravityDir + 1) % 4; }
  else if (k === ' ') { gravityDir = (gravityDir + 1) % 4; e.preventDefault(); }
});

function gameLoop(t) {
  if (gameState === 'start') {
    drawStartScreen(t);
  } else if (gameState === 'playing') {
    // Process input queue for movement (after gravity tick)
    if (t - lastTick >= TICK_MS) {
      doTick();
      
      // Apply player input
      while (inputQueue.length > 0 && gameState === 'playing') {
        const m = inputQueue.shift();
        if (canMove(pacman.gx + m.x, pacman.gy + m.y)) {
          pacman.gx += m.x;
          pacman.gy += m.y;
        }
      }
      
      lastTick = t;
    }
    
    drawGame(t);
  } else if (gameState === 'dead') {
    drawEndScreen(t, false);
  } else if (gameState === 'won') {
    drawEndScreen(t, true);
  }
  
  requestAnimationFrame(gameLoop);
}

requestAnimationFrame(gameLoop);
