```html
<!DOCTYPE html>
<html lang="en">
<head>

<title>Match-3 Shooter</title>
<style>
* { margin: 0; padding: 0; box-sizing: border-box; }

<body>
<canvas id="game"></canvas>
<script>
(() => {
const canvas = document.getElementById('game');
const ctx = canvas.getContext('2d');

let W, H;
function resize() {
W = window.innerWidth;
H = window.innerHeight;
canvas.width = W;
canvas.height = H;
}
window.addEventListener('resize', resize);
resize();

// Audio
let audioCtx = null;
function initAudio() { if (!audioCtx) audioCtx = new (window.AudioContext || webkitAudioContext)(); }
function playSound(freq, dur, type, vol) {
if (!audioCtx) return;
const osc = audioCtx.createOscillator();
const gain = audioCtx.createGain();
osc.type = type || 'square';
osc.frequency.value = freq;
gain.gain.setValueAtTime(vol || 0.15, audioCtx.currentTime);
gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
osc.connect(gain).connect(audioCtx.destination);
osc.start();
osc.stop(audioCtx.currentTime + dur);
}

// Game state
const COLORS = [
{ name: 'red', hex: '#ff4455', glow: '#ff2233' },
{ name: 'green', hex: '#44ff55', glow: '#22ff33' },
{ name: 'blue', hex: '#4466ff', glow: '#2244ff' },
{ name: 'yellow', hex: '#ffdd44', glow: '#ffcc22' }
];

let state = 'start'; // start, playing, gameover
let score = 0;
let lives = 3;
let wave = 1;
let ammo = [];
let selectedSlot = 3;
let bullets = [];
let enemies = [];
let particles = [];
let floatTexts = [];
let mouseX = W / 2, mouseY = H / 2;
let lastFire = 0;
let shakeTime = 0;
let comboCount = 0;
let comboTimer = 0;

function randomColor() { return Math.floor(Math.random() * COLORS.length); }

function initAmmo() {
ammo = [];
for (let i = 0; i < 7; i++) ammo.push(randomColor());
}

function spawnEnemy() {
const col = randomColor();
const x = 60 + Math.random() * (W - 120);
const speed = 0.3 + wave * 0.08 + Math.random() * 0.4;
enemies.push({
x, y: -30,
vx: (Math.random() - 0.5) * 0.8,
vy: speed,
color: col,
hp: 3,
maxHp: 3,
radius: 22 + Math.min(wave * 1.5, 12),
wobble: Math.random() * Math.PI * 2,
alive: true
});
}

function spawnParticles(x, y, color, count) {
for (let i = 0; i < count; i++) {
const angle = Math.random() * Math.PI * 2;
const speed = 1 + Math.random() * 4;
particles.push({
x, y,
vx: Math.cos(angle) * speed,
vy: Math.sin(angle) * speed,
life: 30 + Math.random() * 20,
maxLife: 50,
color
});
}
}

function addFloatText(x, y, text, color) {
floatTexts.push({ x, y, text, color, life: 60 });
}

let spawnTimer = 0;
let waveEnemiesSpawned = 0;
let waveEnemiesTotal = 5;

function startGame() {
state = 'playing';
score = 0;
lives = 3;
wave = 1;
bullets = [];
enemies = [];
particles = [];
floatTexts = [];
comboCount = 0;
comboTimer = 0;
spawnTimer = 0;
waveEnemiesSpawned = 0;
waveEnemiesTotal = 5 + wave * 2;
initAmmo();
}

function nextWave() {
wave++;
waveEnemiesSpawned = 0;
waveEnemiesTotal = 5 + wave * 2;
addFloatText(W / 2, H / 3, `WAVE ${wave}`, '#ffffff');
}

// Input
let keys = {};
window.addEventListener('keydown', e => {
keys[e.key] = true;
if (state === 'start' && (e.key === ' ' || e.key === 'Enter')) { initAudio(); startGame(); }
else if (state === 'gameover' && (e.key === ' ' || e.key === 'Enter')) { initAudio(); startGame(); }
});
window.addEventListener('keyup', e => { keys[e.key] = false; });

canvas.addEventListener('mousemove', e => {
const r = canvas.getBoundingClientRect();
mouseX = (e.clientX - r.left) * (W / r.width);
mouseY = (e.clientY - r.top) * (H / r.height);
});

canvas.addEventListener('click', e => {
initAudio();
const r = canvas.getBoundingClientRect();
const mx = (e.clientX - r.left) * (W / r.width);
const my = (e.clientY - r.top) * (H / r.height);

if (state === 'start' || state === 'gameover') { startGame(); return; }
if (state !== 'playing') return;

// Check if clicking ammo row
const ammoY = H - 50;
const slotW = Math.min(60, W / 10);
const startX = W / 2 - (7 * slotW) / 2;

if (my > ammoY - 10 && my < ammoY + 50) {
for (let i = 0; i < 7; i++) {
const sx = startX + i * slotW + slotW / 2;
const sy = ammoY + 20;
if (Math.abs(mx - sx) < slotW / 2 && Math.abs(my - sy) < 30) {
if (i === selectedSlot) return;
if (Math.abs(i - selectedSlot) === 1) {
// Swap
const tmp = ammo[selectedSlot];
ammo[selectedSlot] = ammo[i];
ammo[i] = tmp;
selectedSlot = i;
playSound(400, 0.1, 'sine', 0.1);
}
return;
}
}
}

// Fire bullet
const now = Date.now();
if (now - lastFire > 200) {
lastFire = now;
const colorIdx = ammo[selectedSlot];
ammo[selectedSlot] = randomColor();
const ax = W / 2 + (selectedSlot - 3) * Math.min(60, W / 10);
const ay = H - 30;
const dx = mx - ax;
const dy = my - ay;
const dist = Math.sqrt(dx * dx + dy * dy);
if (dist > 5) {
bullets.push({ x: ax, y: ay, vx: dx / dist * 8, vy: dy / dist * 8, color: colorIdx, radius: 10 });
playSound(200 + colorIdx * 100, 0.08, 'square', 0.1);
}
}
});

// Update
function update() {
if (state !== 'playing') return;

// Combo timer
if (comboTimer > 0) comboTimer--;
else comboCount = 0;

// Spawn enemies
spawnTimer++;
const spawnInterval = Math.max(60 - wave * 3, 20);
if (spawnTimer >= spawnInterval && waveEnemiesSpawned < waveEnemiesTotal) {
spawnTimer = 0;
waveEnemiesSpawned++;
spawnEnemy();
}

// Check wave complete
if (waveEnemiesSpawned >= waveEnemiesTotal && enemies.length === 0) {
nextWave();
}

// Update bullets
for (let i = bullets.length - 1; i >= 0; i--) {
const b = bullets[i];
b.x += b.vx;
b.y += b.vy;
if (b.y < -20 || b.x < -20 || b.x > W + 20) { bullets.splice(i, 1); continue; }

// Hit enemies
for (let j = enemies.length - 1; j >= 0; j--) {
const en = enemies[j];
if (!en.alive) continue;
const dx = b.x - en.x;
const dy = b.y - en.y;
const dist = Math.sqrt(dx * dx + dy * dy);
if (dist < b.radius + en.radius) {
bullets.splice(i, 1);
if (b.color === en.color) {
en.hp--;
playSound(300 + (3 - en.hp) * 100, 0.15, 'triangle', 0.2);
addFloatText(en.x, en.y - 10, COLORS[b.color].name.toUpperCase(), COLORS[b.color].hex);
if (en.hp <= 0) {
en.alive = false;
enemies.splice(j, 1);
comboCount++;
comboTimer = 90;
const pts = comboCount >= 3 ? 50 * comboCount : 20 + wave * 5;
score += pts;
addFloatText(en.x, en.y, `+${pts}`, '#ffffff');
if (comboCount >= 3) addFloatText(en.x, en.y - 25, `COMBO x${comboCount}!`, '#ffdd00');
spawnParticles(en.x, en.y, COLORS[en.color].hex, 15);
shakeTime = 8;
playSound(500, 0.3, 'sawtooth', 0.2);
} else {
spawnParticles(b.x, b.y, COLORS[b.color].hex, 5);
}
} else {
// Wrong color - small damage
en.hp -= 0; // No damage for wrong color
playSound(150, 0.1, 'sawtooth', 0.08);
addFloatText(en.x, en.y - 10, 'MISS', '#888888');
}
break;
}
}

// Update enemies
for (let i = enemies.length - 1; i >= 0; i--) {
const en = enemies[i];
if (!en.alive) { enemies.splice(i, 1); continue; }
en.wobble += 0.05;
en.x += en.vx + Math.sin(en.wobble) * 0.3;
en.y += en.vy;
if (en.x < 20) en.x = 20;
if (en.x > W - 20) en.x = W - 20;

// Reached bottom
if (en.y > H - 70) {
enemies.splice(i, 1);
lives--;
shakeTime = 15;
playSound(100, 0.4, 'sawtooth', 0.3);
addFloatText(en.x, H - 80, '-1 LIFE', '#ff4444');
if (lives <= 0) { state = 'gameover'; }
}
}

// Update particles
for (let i = particles.length - 1; i >= 0; i--) {
const p = particles[i];
p.x += p.vx;
p.y += p.vy;
p.vy += 0.15;
p.life--;
if (p.life <= 0) particles.splice(i, 1);
}

// Update float texts
for (let i = floatTexts.length - 1; i >= 0; i--) {
floatTexts[i].y -= 0.8;
floatTexts[i].life--;
if (floatTexts[i].life <= 0) floatTexts.splice(i, 1);
}

// Shake decay
if (shakeTime > 0) shakeTime--;
}

// Draw
function draw() {
ctx.fillStyle = '#0a0a1a';
ctx.fillRect(0, 0, W, H);

if (state === 'start') {
drawStartScreen();
return;
}
if (state === 'gameover') {
drawGameOver();
return;
}

// Apply shake
let sx = 0, sy = 0;
if (shakeTime > 0) {
sx = (Math.random() - 0.5) * shakeTime;
sy = (Math.random() - 0.5) * shakeTime;
}
ctx.save();
ctx.translate(sx, sy);

// Draw grid lines (subtle)
ctx.strokeStyle = 'rgba(255,255,255,0.03)';
ctx.lineWidth = 1;
for (let x = 0; x < W; x += 40) { ctx.beginPath(); ctx.moveTo(x, 0); ctx.lineTo(x, H); ctx.stroke(); }
for (let y = 0; y < H; y += 40) { ctx.beginPath(); ctx.moveTo(0, y); ctx.lineTo(W, y); ctx.stroke(); }

// Draw enemies
for (const en of enemies) {
if (!en.alive) continue;
const c = COLORS[en.color];

// Glow
ctx.shadowColor = c.glow;
ctx.shadowBlur = 15;

// Body
ctx.beginPath();
ctx.arc(en.x, en.y, en.radius, 0, Math.PI * 2);
ctx.fillStyle = c.hex + '33';
ctx.fill();
ctx.strokeStyle = c.hex;
ctx.lineWidth = 2.5;
ctx.stroke();

// HP pips
for (let h = 0; h < en.maxHp; h++) {
const px = en.x - (en.maxHp - 1) * 6 + h * 12;
const py = en.y;
ctx.beginPath();
ctx.arc(px, py, 4, 0, Math.PI * 2);
if (h < en.hp) {
ctx.fillStyle = c.hex;
ctx.fill();
} else {
ctx.strokeStyle = c.hex + '88';
ctx.lineWidth = 1;
ctx.stroke();
}
}

// Color letter
ctx.shadowBlur = 0;
ctx.fillStyle = c.hex;
ctx.font = `bold ${Math.floor(en.radius * 0.7)}px monospace`;
ctx.textAlign = 'center';
ctx.textBaseline = 'middle';
ctx.fillText(c.name[0].toUpperCase(), en.x, en.y + (en.maxHp > 0 ? 12 : 0));

ctx.shadowBlur = 0;
}

// Draw bullets
for (const b of bullets) {
const c = COLORS[b.color];
ctx.shadowColor = c.glow;
ctx.shadowBlur = 12;
ctx.beginPath();
ctx.arc(b.x, b.y, b.radius, 0, Math.PI * 2);
ctx.fillStyle = c.hex;
ctx.fill();
ctx.strokeStyle = '#ffffff';
ctx.lineWidth = 1.5;
ctx.stroke();
ctx.shadowBlur = 0;
}

// Draw particles
for (const p of particles) {
const alpha = p.life / p.maxLife;
ctx.globalAlpha = alpha;
ctx.beginPath();
ctx.arc(p.x, p.y, 3 * alpha, 0, Math.PI * 2);
ctx.fillStyle = p.color;
ctx.fill();
}
ctx.globalAlpha = 1;

// Draw float texts
for (const ft of floatTexts) {
ctx.globalAlpha = ft.life / 60;
ctx.fillStyle = ft.color;
ctx.font = 'bold 16px monospace';
ctx.textAlign = 'center';
ctx.fillText(ft.text, ft.x, ft.y);
}
ctx.globalAlpha = 1;

// Draw ammo row
const slotW = Math.min(60, W / 10);
const startX = W / 2 - (7 * slotW) / 2;
const ammoY = H - 50;

// Background bar
ctx.fillStyle = 'rgba(20,20,40,0.8)';
ctx.fillRect(startX - 10, ammoY - 5, 7 * slotW + 20, 55);

for (let i = 0; i < 7; i++) {
const sx2 = startX + i * slotW;
const c = COLORS[ammo[i]];

// Slot bg
ctx.fillStyle = 'rgba(30,30,60,0.8)';
ctx.fillRect(sx2 + 2, ammoY, slotW - 4, 45);

if (i === selectedSlot) {
ctx.strokeStyle = '#ffffff';
ctx.lineWidth = 2;
ctx.strokeRect(sx2 + 2, ammoY, slotW - 4, 45);
ctx.shadowColor = '#ffffff';
ctx.shadowBlur = 8;
}

// Orb
const ox = sx2 + slotW / 2;
const oy = ammoY + 22;
ctx.shadowColor = c.glow;
ctx.shadowBlur = i === selectedSlot ? 15 : 8;
ctx.beginPath();
ctx.arc(ox, oy, Math.min(slotW * 0.3, 16), 0, Math.PI * 2);
ctx.fillStyle = c.hex;
ctx.fill();
ctx.strokeStyle = '#ffffff44';
ctx.lineWidth = 1;
ctx.stroke();
ctx.shadowBlur = 0;

// Letter
ctx.fillStyle = '#fff';
ctx.font = 'bold 12px monospace';
ctx.textAlign = 'center';
ctx.textBaseline = 'middle';
ctx.fillText(c.name[0].toUpperCase(), ox, oy);
}

// Crosshair
ctx.strokeStyle = 'rgba(255,255,255,0.6)';
ctx.lineWidth = 1.5;
ctx.beginPath();
ctx.arc(mouseX, mouseY, 12, 0, Math.PI * 2);
ctx.stroke();
ctx.beginPath();
ctx.moveTo(mouseX - 18, mouseY); ctx.lineTo(mouseX - 6, mouseY);
ctx.moveTo(mouseX + 6, mouseY); ctx.lineTo(mouseX + 18, mouseY);
ctx.moveTo(mouseX, mouseY - 18); ctx.lineTo(mouseX, mouseY - 6);
ctx.moveTo(mouseX, mouseY + 6); ctx.lineTo(mouseX, mouseY + 18);
ctx.stroke();

// HUD
ctx.shadowBlur = 0;
ctx.fillStyle = '#ffffff';
ctx.font = 'bold 20px monospace';
ctx.textAlign = 'left';
ctx.fillText(`SCORE: ${score}`, 20, 35);
ctx.fillText(`WAVE: ${wave}`, 20, 60);

// Lives
for (let i = 0; i < 3; i++) {
ctx.beginPath();
ctx.arc(W - 80 + i * 30, 30, 10, 0, Math.PI * 2);
if (i < lives) {
ctx.fillStyle = '#ff4455';
ctx.fill();
} else {
ctx.strokeStyle = '#ff445588';
ctx.lineWidth = 1.5;
ctx.stroke();
}
}

// Combo display
if (comboCount >= 2 && comboTimer > 0) {
ctx.fillStyle = '#ffdd00';
ctx.font = 'bold 24px monospace';
ctx.textAlign = 'center';
ctx.fillText(`COMBO x${comboCount}`, W / 2, 50);
}

ctx.restore();
}

function drawStartScreen() {
ctx.fillStyle = '#0a0a1a';
ctx.fillRect(0, 0, W, H);

// Title glow
ctx.shadowColor = '#4466ff';
ctx.shadowBlur = 20;
ctx.fillStyle = '#ffffff';
ctx.font = `bold ${Math.min(W * 0.08, 60)}px monospace`;
ctx.textAlign = 'center';
ctx.fillText('MATCH-3 SHOOTER', W / 2, H * 0.25);
ctx.shadowBlur = 0;

// Subtitle
ctx.font = `${Math.min(W * 0.03, 20)}px monospace`;
ctx.fillStyle = '#8888ff';
ctx.fillText('Match colors to destroy enemies!', W / 2, H * 0.35);

// Instructions
const lines = [
'Click enemies with matching colored orbs (need 3 hits)',
'Swap adjacent ammo slots by clicking next to selected',
'Mouse to aim - Click to fire',
'',
'Top Left/Right Arrows: Select slot',
'Q/E: Swap left/right',
'',
'Press SPACE or CLICK to start!'
];
ctx.font = `${Math.min(W * 0.02, 16)}px monospace`;
ctx.fillStyle = '#aaaaaa';
lines.forEach((line, i) => {
ctx.fillText(line, W / 2, H * 0.45 + i * 28);
});

// Animated orbs
const t = Date.now() * 0.003;
for (let i = 0; i < 7; i++) {
const ox = W / 2 - 140 + i * 44;
const oy = H * 0.85 + Math.sin(t + i) * 8;
const c = COLORS[i % 4];
ctx.shadowColor = c.glow;
ctx.shadowBlur = 12;
ctx.beginPath();
ctx.arc(ox, oy, 14, 0, Math.PI * 2);
ctx.fillStyle = c.hex;
ctx.fill();
ctx.shadowBlur = 0;
}
}

function drawGameOver() {
ctx.fillStyle = 'rgba(10,10,26,0.9)';
ctx.fillRect(0, 0, W, H);

ctx.shadowColor = '#ff4455';
ctx.shadowBlur = 20;
ctx.fillStyle = '#ffffff';
ctx.font = `bold ${Math.min(W * 0.07, 50)}px monospace`;
ctx.textAlign = 'center';
ctx.fillText('GAME OVER', W / 2, H * 0.3);
ctx.shadowBlur = 0;

ctx.font = `${Math.min(W * 0.04, 30)}px monospace`;
ctx.fillStyle = '#ffdd44';
ctx.fillText(`Final Score: ${score}`, W / 2, H * 0.45);
ctx.fillStyle = '#aaaaaa';
ctx.font = `${Math.min(W * 0.025, 20)}px monospace`;
ctx.fillText(`Waves Survived: ${wave}`, W / 2, H * 0.55);

ctx.fillStyle = '#8888ff';
ctx.fillText('Press SPACE or CLICK to restart', W / 2, H * 0.7);
}

// Keyboard controls for ammo selection
function handleKeys() {
if (state !== 'playing') return;
if (keys['ArrowLeft']) {
selectedSlot = Math.max(0, selectedSlot - 1);
keys['ArrowLeft'] = false;
playSound(350, 0.05, 'sine', 0.08);
}
if (keys['ArrowRight']) {
selectedSlot = Math.min(6, selectedSlot + 1);
keys['ArrowRight'] = false;
playSound(350, 0.05, 'sine', 0.08);
}
if (keys['q'] || keys['Q']) {
if (selectedSlot > 0) {
const tmp = ammo[selectedSlot];
ammo[selectedSlot] = ammo[selectedSlot - 1];
ammo[selectedSlot - 1] = tmp;
selectedSlot--;
playSound(400, 0.1, 'sine', 0.1);
}
keys['q'] = keys['Q'] = false;
}
if (keys['e'] || keys['E']) {
if (selectedSlot < 6) {
const tmp = ammo[selectedSlot];
ammo[selectedSlot] = ammo[selectedSlot + 1];
ammo[selectedSlot + 1] = tmp;
selectedSlot++;
playSound(400, 0.1, 'sine', 0.1);
}
keys['e'] = keys['E'] = false;
}
}

// Game loop
function loop() {
handleKeys();
update();
draw();
requestAnimationFrame(loop);
}
loop();

})();
</script>
</body>
```
