<script>
const canvas = document.getElementById('c');
const ctx = canvas.getContext('2d');

let W, H;
function resize() {
    W = window.innerWidth;
    H = window.innerHeight;
    canvas.width = W;
    canvas.height = H;
}
resize();
window.addEventListener('resize', resize);

// Audio
let audioCtx = null;
function initAudio() {
    if (!audioCtx) {
        audioCtx = new (window.AudioContext || window.webkitAudioContext)();
    }
}
function playSound(type, freq, dur, vol) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = type;
    osc.frequency.value = freq || 440;
    gain.gain.setValueAtTime(vol || 0.1, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + (dur || 0.1));
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + (dur || 0.1));
}
function playBounce() { playSound('square', 220, 0.05, 0.08); }
function playHit() { playSound('sawtooth', 100, 0.3, 0.15); }
function playShoot() { playSound('triangle', 880, 0.05, 0.05); }
function playExplode() { playSound('sawtooth', 60, 0.4, 0.2); }

// Game state
let gameState = 'start'; // start, playing, gameover
let score = 0;
let lives = 3;
let wave = 1;
let time = 0;
let shakeMag = 0;
let shakeX = 0, shakeY = 0;

// Pinball
const ball = { x: 0, y: 0, vx: 0, vy: 0, r: 12, trail: [] };
const WALL_THICK = 8;
const RESTITUTION = 0.95;
const MAX_SPEED = 600;

// Bullets (enemy)
let bullets = [];
// Particles
let particles = [];
// Powerups
let powerups = [];
// Enemies (emitters)
let emitters = [];

// Input
const keys = {};
window.addEventListener('keydown', e => {
    keys[e.code] = true;
    if (gameState === 'start' && (e.code === 'Space' || e.code === 'Enter')) {
        initAudio();
        startGame();
    } else if (gameState === 'gameover' && (e.code === 'Space' || e.code === 'Enter')) {
        startGame();
    }
    e.preventDefault();
});
window.addEventListener('keyup', e => { keys[e.code] = false; });

// Mouse for aiming
let mouseX = 0, mouseY = 0;
window.addEventListener('mousemove', e => { mouseX = e.clientX; mouseY = e.clientY; });

function startGame() {
    gameState = 'playing';
    score = 0;
    lives = 3;
    wave = 1;
    time = 0;
    bullets = [];
    particles = [];
    powerups = [];
    emitters = [];
    ball.x = W / 2;
    ball.y = H / 2;
    ball.vx = (Math.random() - 0.5) * 200;
    ball.vy = (Math.random() - 0.5) * 200;
    ball.trail = [];
    spawnWave();
}

function spawnWave() {
    emitters = [];
    const patterns = ['spiral', 'radial', 'aimed', 'zigzag'];
    const numEmitters = Math.min(1 + Math.floor(wave / 2), 4);
    
    for (let i = 0; i < numEmitters; i++) {
        const side = i % 4;
        let ex, ey;
        if (side === 0) { ex = W * 0.25; ey = -30; }
        else if (side === 1) { ex = W * 0.75; ey = -30; }
        else if (side === 2) { ex = -30; ey = H * 0.5; }
        else { ex = W + 30; ey = H * 0.5; }
        
        const pattern = patterns[Math.floor(Math.random() * patterns.length)];
        emitters.push({ x: ex, y: ey, pattern, angle: 0, timer: 0, count: 0 });
    }
}

function update(dt) {
    if (gameState !== 'playing') return;
    
    time += dt;
    wave = Math.floor(time / 15) + 1;
    
    // Ball physics - apply input force
    const FORCE = 800;
    if (keys['KeyW'] || keys['ArrowUp']) ball.vy -= FORCE * dt;
    if (keys['KeyS'] || keys['ArrowDown']) ball.vy += FORCE * dt;
    if (keys['KeyA'] || keys['ArrowLeft']) ball.vx -= FORCE * dt;
    if (keys['KeyD'] || keys['ArrowRight']) ball.vx += FORCE * dt;
    
    // Space for dash
    if (keys['Space']) {
        const speed = Math.sqrt(ball.vx * ball.vx + ball.vy * ball.vy);
        if (speed > 10) {
            ball.vx *= 1.5;
            ball.vy *= 1.5;
        } else {
            ball.vx += (Math.random() - 0.5) * 200;
            ball.vy += (Math.random() - 0.5) * 200;
        }
    }
    
    // Clamp speed
    const spd = Math.sqrt(ball.vx * ball.vx + ball.vy * ball.vy);
    if (spd > MAX_SPEED) {
        ball.vx = (ball.vx / spd) * MAX_SPEED;
        ball.vy = (ball.vy / spd) * MAX_SPEED;
    }
    
    // Move ball
    ball.x += ball.vx * dt;
    ball.y += ball.vy * dt;
    
    // Wall collisions
    const r = ball.r;
    if (ball.x - r < WALL_THICK) { ball.x = WALL_THICK + r; ball.vx = Math.abs(ball.vx) * RESTITUTION; playBounce(); spawnBounceParticles(WALL_THICK, ball.y); }
    if (ball.x + r > W - WALL_THICK) { ball.x = W - WALL_THICK - r; ball.vx = -Math.abs(ball.vx) * RESTITUTION; playBounce(); spawnBounceParticles(W - WALL_THICK, ball.y); }
    if (ball.y - r < WALL_THICK) { ball.y = WALL_THICK + r; ball.vy = Math.abs(ball.vy) * RESTITUTION; playBounce(); spawnBounceParticles(ball.x, WALL_THICK); }
    if (ball.y + r > H - WALL_THICK) { ball.y = H - WALL_THICK - r; ball.vy = -Math.abs(ball.vy) * RESTITUTION; playBounce(); spawnBounceParticles(ball.x, H - WALL_THICK); }
    
    // Trail
    ball.trail.push({ x: ball.x, y: ball.y });
    if (ball.trail.length > 20) ball.trail.shift();
    
    // Update emitters
    const difficulty = 1 + wave * 0.3;
    for (let e of emitters) {
        e.timer += dt;
        e.angle += dt * 2 * difficulty;
        
        const fireRate = Math.max(0.05, 0.2 - wave * 0.01);
        if (e.timer >= fireRate) {
            e.timer = 0;
            e.count++;
            
            switch (e.pattern) {
                case 'spiral':
                    for (let i = 0; i < 3; i++) {
                        const a = e.angle + i * (Math.PI * 2 / 3);
                        bullets.push({ x: e.x, y: e.y, vx: Math.cos(a) * 150 * difficulty, vy: Math.sin(a) * 150 * difficulty, r: 6, color: '#ff4488', life: 8 });
                    }
                    break;
                case 'radial':
                    if (e.count % 20 === 0) {
                        const n = 8 + wave;
                        for (let i = 0; i < n; i++) {
                            const a = (i / n) * Math.PI * 2;
                            bullets.push({ x: e.x, y: e.y, vx: Math.cos(a) * 120 * difficulty, vy: Math.sin(a) * 120 * difficulty, r: 5, color: '#44ff88', life: 8 });
                        }
                    }
                    break;
                case 'aimed':
                    {
                        const dx = ball.x - e.x;
                        const dy = ball.y - e.y;
                        const d = Math.sqrt(dx * dx + dy * dy);
                        const spread = 0.15;
                        for (let i = -1; i <= 1; i++) {
                            const a = Math.atan2(dy, dx) + i * spread;
                            bullets.push({ x: e.x, y: e.y, vx: Math.cos(a) * 200 * difficulty, vy: Math.sin(a) * 200 * difficulty, r: 7, color: '#ffcc44', life: 8 });
                        }
                    }
                    break;
                case 'zigzag':
                    {
                        const a = Math.atan2(ball.y - e.y, ball.x - e.x) + Math.sin(e.angle * 3) * 0.5;
                        bullets.push({ x: e.x, y: e.y, vx: Math.cos(a) * 180 * difficulty, vy: Math.sin(a) * 180 * difficulty, r: 6, color: '#cc44ff', life: 8 });
                    }
                    break;
            }
            playShoot();
        }
    }
    
    // Update bullets
    for (let i = bullets.length - 1; i >= 0; i--) {
        const b = bullets[i];
        b.x += b.vx * dt;
        b.y += b.vy * dt;
        b.life -= dt;
        
        if (b.life <= 0 || b.x < -50 || b.x > W + 50 || b.y < -50 || b.y > H + 50) {
            bullets.splice(i, 1);
            continue;
        }
        
        // Wall bounce for bullets
        if (b.x - b.r < WALL_THICK) { b.x = WALL_THICK + b.r; b.vx = Math.abs(b.vx); }
        if (b.x + b.r > W - WALL_THICK) { b.x = W - WALL_THICK - b.r; b.vx = -Math.abs(b.vx); }
        if (b.y - b.r < WALL_THICK) { b.y = WALL_THICK + b.r; b.vy = Math.abs(b.vy); }
        if (b.y + b.r > H - WALL_THICK) { b.y = H - WALL_THICK - b.r; b.vy = -Math.abs(b.vy); }
        
        // Collision with ball
        const dx = ball.x - b.x;
        const dy = ball.y - b.y;
        const dist = Math.sqrt(dx * dx + dy * dy);
        if (dist < ball.r + b.r) {
            bullets.splice(i, 1);
            hitBall();
        }
    }
    
    // Update particles
    for (let i = particles.length - 1; i >= 0; i--) {
        const p = particles[i];
        p.x += p.vx * dt;
        p.y += p.vy * dt;
        p.life -= dt;
        if (p.life <= 0) particles.splice(i, 1);
    }
    
    // Score over time
    score += Math.floor(dt * 10 * wave);
    
    // Spawn powerups rarely
    if (Math.random() < 0.002) {
        powerups.push({ x: WALL_THICK + Math.random() * (W - WALL_THICK * 2), y: WALL_THICK + Math.random() * (H - WALL_THICK * 2), r: 15, type: 'shield', life: 10 });
    }
    
    // Update powerups
    for (let i = powerups.length - 1; i >= 0; i--) {
        const p = powerups[i];
        p.life -= dt;
        if (p.life <= 0) { powerups.splice(i, 1); continue; }
        const dx = ball.x - p.x;
        const dy = ball.y - p.y;
        if (Math.sqrt(dx * dx + dy * dy) < ball.r + p.r) {
            if (p.type === 'shield') { lives++; }
            powerups.splice(i, 1);
            playBounce();
        }
    }
    
    // Shake decay
    shakeMag *= 0.9;
    if (shakeMag > 0.5) {
        shakeX = (Math.random() - 0.5) * shakeMag;
        shakeY = (Math.random() - 0.5) * shakeMag;
    } else {
        shakeX = 0; shakeY = 0;
    }
    
    // Respawn wave
    if (time % 15 < dt) {
        spawnWave();
    }
}

function hitBall() {
    lives--;
    shakeMag = 20;
    playHit();
    playExplode();
    for (let i = 0; i < 30; i++) {
        const a = Math.random() * Math.PI * 2;
        const s = 100 + Math.random() * 300;
        particles.push({ x: ball.x, y: ball.y, vx: Math.cos(a) * s, vy: Math.sin(a) * s, life: 0.5 + Math.random() * 0.5, color: '#ffffff', r: 3 });
    }
    if (lives <= 0) {
        gameState = 'gameover';
    } else {
        // Brief invulnerability - just push ball to center-ish
        ball.vx = (Math.random() - 0.5) * 400;
        ball.vy = (Math.random() - 0.5) * 400;
    }
}

function spawnBounceParticles(x, y) {
    for (let i = 0; i < 5; i++) {
        const a = Math.random() * Math.PI * 2;
        const s = 50 + Math.random() * 100;
        particles.push({ x, y, vx: Math.cos(a) * s, vy: Math.sin(a) * s, life: 0.3, color: '#66ccff', r: 2 });
    }
}

function draw() {
    ctx.save();
    ctx.translate(shakeX, shakeY);
    
    // Background
    ctx.fillStyle = '#0a0a12';
    ctx.fillRect(-10, -10, W + 20, H + 20);
    
    // Grid
    ctx.strokeStyle = 'rgba(30,40,80,0.3)';
    ctx.lineWidth = 1;
    const gridSize = 60;
    for (let x = 0; x < W; x += gridSize) {
        ctx.beginPath(); ctx.moveTo(x, 0); ctx.lineTo(x, H); ctx.stroke();
    }
    for (let y = 0; y < H; y += gridSize) {
        ctx.beginPath(); ctx.moveTo(0, y); ctx.lineTo(W, y); ctx.stroke();
    }
    
    // Walls
    ctx.strokeStyle = '#4488ff';
    ctx.lineWidth = WALL_THICK;
    ctx.strokeRect(WALL_THICK / 2, WALL_THICK / 2, W - WALL_THICK, H - WALL_THICK);
    ctx.strokeStyle = 'rgba(68,136,255,0.3)';
    ctx.lineWidth = WALL_THICK + 4;
    ctx.strokeRect(WALL_THICK / 2, WALL_THICK / 2, W - WALL_THICK, H - WALL_THICK);
    
    // Particles
    for (const p of particles) {
        ctx.globalAlpha = Math.min(1, p.life * 2);
        ctx.fillStyle = p.color;
        ctx.beginPath();
        ctx.arc(p.x, p.y, p.r, 0, Math.PI * 2);
        ctx.fill();
    }
    ctx.globalAlpha = 1;
    
    // Bullets
    for (const b of bullets) {
        ctx.shadowColor = b.color;
        ctx.shadowBlur = 8;
        ctx.fillStyle = b.color;
        ctx.beginPath();
        ctx.arc(b.x, b.y, b.r, 0, Math.PI * 2);
        ctx.fill();
    }
    ctx.shadowBlur = 0;
    
    // Powerups
    for (const p of powerups) {
        ctx.shadowColor = '#44ff88';
        ctx.shadowBlur = 15;
        ctx.fillStyle = '#44ff88';
        ctx.font = '20px monospace';
        ctx.textAlign = 'center';
        ctx.fillText('+', p.x, p.y + 7);
        ctx.strokeStyle = '#44ff88';
        ctx.lineWidth = 2;
        ctx.beginPath();
        ctx.arc(p.x, p.y, p.r, 0, Math.PI * 2);
        ctx.stroke();
    }
    ctx.shadowBlur = 0;
    
    // Emitters (glowing dots)
    for (const e of emitters) {
        ctx.shadowColor = '#ff4488';
        ctx.shadowBlur = 15;
        ctx.fillStyle = '#ff4488';
        ctx.beginPath();
        ctx.arc(e.x, e.y, 8, 0, Math.PI * 2);
        ctx.fill();
    }
    ctx.shadowBlur = 0;
    
    // Ball trail
    for (let i = 0; i < ball.trail.length; i++) {
        const t = ball.trail[i];
        const alpha = (i / ball.trail.length) * 0.5;
        ctx.fillStyle = `rgba(255,255,255,${alpha})`;
        ctx.beginPath();
        ctx.arc(t.x, t.y, ball.r * (i / ball.trail.length), 0, Math.PI * 2);
        ctx.fill();
    }
    
    // Ball
    ctx.shadowColor = '#ffffff';
    ctx.shadowBlur = 20;
    ctx.fillStyle = '#ffffff';
    ctx.beginPath();
    ctx.arc(ball.x, ball.y, ball.r, 0, Math.PI * 2);
    ctx.fill();
    ctx.fillStyle = 'rgba(100,180,255,0.6)';
    ctx.beginPath();
    ctx.arc(ball.x - 3, ball.y - 3, ball.r * 0.4, 0, Math.PI * 2);
    ctx.fill();
    ctx.shadowBlur = 0;
    
    // HUD
    ctx.fillStyle = '#ffffff';
    ctx.font = 'bold 16px monospace';
    ctx.textAlign = 'left';
    ctx.fillText(`SCORE: ${score}`, 20, H - 20);
    ctx.fillText(`WAVE: ${wave}`, 20, H - 45);
    
    // Lives as hearts
    for (let i = 0; i < lives; i++) {
        ctx.fillStyle = '#ff4488';
        ctx.font = '20px monospace';
        ctx.fillText('♥', W - 30 - i * 30, H - 20);
    }
    
    ctx.restore();
}

function drawStartScreen() {
    ctx.fillStyle = '#0a0a12';
    ctx.fillRect(0, 0, W, H);
    
    // Title
    ctx.shadowColor = '#ff4488';
    ctx.shadowBlur = 30;
    ctx.fillStyle = '#ffffff';
    ctx.font = 'bold 48px monospace';
    ctx.textAlign = 'center';
    ctx.fillText('PINBALL', W / 2, H / 2 - 80);
    ctx.fillStyle = '#ff4488';
    ctx.fillText('BULLET HELL', W / 2, H / 2 - 30);
    ctx.shadowBlur = 0;
    
    // Controls
    ctx.font = '18px monospace';
    ctx.fillStyle = '#aaccff';
    ctx.fillText('WASD / Arrow Keys — Move Ball', W / 2, H / 2 + 40);
    ctx.fillText('SPACE — Dash', W / 2, H / 2 + 70);
    ctx.fillText('Bounce off walls to survive the bullet storm!', W / 2, H / 2 + 110);
    
    // Blinking prompt
    if (Math.floor(time * 2) % 2 === 0) {
        ctx.fillStyle = '#ffffff';
        ctx.font = '24px monospace';
        ctx.fillText('PRESS SPACE TO START', W / 2, H / 2 + 180);
    }
    
    // Decorative bullets
    for (let i = 0; i < 20; i++) {
        const a = time * 2 + i * 0.31;
        const r = 200 + Math.sin(time + i) * 50;
        const x = W / 2 + Math.cos(a) * r;
        const y = H / 2 - 30 + Math.sin(a) * r * 0.6;
        ctx.fillStyle = `rgba(255,68,136,${0.3 + Math.sin(time * 3 + i) * 0.2})`;
        ctx.beginPath();
        ctx.arc(x, y, 4, 0, Math.PI * 2);
        ctx.fill();
    }
}

function drawGameOver() {
    ctx.fillStyle = 'rgba(10,10,18,0.8)';
    ctx.fillRect(0, 0, W, H);
    
    ctx.shadowColor = '#ff4488';
    ctx.shadowBlur = 20;
    ctx.fillStyle = '#ff4488';
    ctx.font = 'bold 48px monospace';
    ctx.textAlign = 'center';
    ctx.fillText('GAME OVER', W / 2, H / 2 - 40);
    ctx.shadowBlur = 0;
    
    ctx.fillStyle = '#ffffff';
    ctx.font = '24px monospace';
    ctx.fillText(`FINAL SCORE: ${score}`, W / 2, H / 2 + 10);
    ctx.fillText(`WAVES SURVIVED: ${wave}`, W / 2, H / 2 + 45);
    
    if (Math.floor(time * 2) % 2 === 0) {
        ctx.fillStyle = '#aaccff';
        ctx.font = '20px monospace';
        ctx.fillText('PRESS SPACE TO RESTART', W / 2, H / 2 + 110);
    }
}

let lastTime = 0;
function loop(timestamp) {
    const dt = Math.min((timestamp - lastTime) / 1000, 0.05);
    lastTime = timestamp;
    
    if (gameState === 'playing') {
        update(dt);
    } else {
        time += dt;
    }
    
    if (gameState === 'start') {
        drawStartScreen();
    } else if (gameState === 'playing') {
        draw();
    } else if (gameState === 'gameover') {
        draw();
        drawGameOver();
    }
    
    requestAnimationFrame(loop);
}
requestAnimationFrame(loop);
</script>
