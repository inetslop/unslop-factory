function resize() {
    const ratio = GAME_W / GAME_H;
    if (window.innerWidth / window.innerHeight > ratio) {
        H = window.innerHeight;
        W = H * ratio;
    } else {
        W = window.innerWidth;
        H = W / ratio;
    }
    canvas.width = Math.floor(W);
    canvas.height = Math.floor(H);
    scale = W / GAME_W;
}

// Audio
let audioCtx = null;
function initAudio() {
    if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
}
function playSound(freq, dur, type, vol) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = type || 'square';
    osc.frequency.value = freq;
    gain.gain.setValueAtTime(vol || 0.1, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + dur);
}

let lives = 3;

let frog = null;

let time = 0;

function startGame() {
    gameState = 'playing';
    score = 0;
    lives = 3;
    time = 0;
    particles = [];
    frog = { x: GAME_W / 2, y: GAME_H - 50, vx: 0, vy: 0, r: 18, inWater: false };

    cars = [];

            });

    gators = [];
    for (let i = 0; i < 4; i++) {
        const dir = Math.random() > 0.5 ? 1 : -1;
        const speed = 0.8 + Math.random() * 0.6;
        const count = 2 + Math.floor(Math.random() * 2);
        for (let j = 0; j < count; j++) {
            gators.push({
                x: (j / count) * GAME_W + Math.random() * 30,
                y: WATER_START_Y + i * LANE_H + LANE_H / 2,
                vx: dir * speed,
                w: 55, h: 28
            });
        }
    }

    // Logs
    logs = [];
    for (let i = 0; i < 3; i++) {
        const dir = Math.random() > 0.5 ? 1 : -1;
        const speed = 0.6 + Math.random() * 0.4;
        const count = 2;
        for (let j = 0; j < count; j++) {
            logs.push({
                x: (j / count) * GAME_W + Math.random() * 50,

        });

function spawnParticles(x, y, color, count) {
    for (let i = 0; i < count; i++) {
        const angle = Math.random() * Math.PI * 2;
        const speed = 1 + Math.random() * 3;
        particles.push({
            x, y,
            vx: Math.cos(angle) * speed,
            vy: Math.sin(angle) * speed,
            life: 1,
            color
        });
    }
}

function update() {
    if (gameState !== 'playing') return;
    time++;

    // Frog physics
    const accel = 0.6;
    const friction = frog.inWater ? 0.92 : 0.95;
    const maxSpeed = 7;

    if (keys['ArrowLeft'] || keys['KeyA']) frog.vx -= accel;
    if (keys['ArrowRight'] || keys['KeyD']) frog.vx += accel;
    if (keys['ArrowUp'] || keys['KeyW']) frog.vy -= accel;
    if (keys['ArrowDown'] || keys['KeyS']) frog.vy += accel;

    frog.vx *= friction;
    frog.vy *= friction;

    const speed = Math.sqrt(frog.vx * frog.vx + frog.vy * frog.vy);
    if (speed > maxSpeed) {
        frog.vx = (frog.vx / speed) * maxSpeed;
        frog.vy = (frog.vy / speed) * maxSpeed;
    }

    // Check if in water
    frog.inWater = frog.y > WATER_START_Y - 20 && frog.y < ROAD_START_Y + LANE_H * 5 - 20;

    frog.x += frog.vx;
    frog.y += frog.vy;

    // Bounds
    if (frog.x < frog.r) { frog.x = frog.r; frog.vx = 0; }
    if (frog.x > GAME_W - frog.r) { frog.x = GAME_W - frog.r; frog.vx = 0; }
    if (frog.y < 15) { frog.y = 15; frog.vy = 0; }
    if (frog.y > GAME_H - 20) { frog.y = GAME_H - 20; frog.vy = 0; }

    // Move cars
    for (let car of cars) {
        car.x += car.vx;
        if (car.vx > 0 && car.x > GAME_W + 60) car.x = -60;
        if (car.vx < 0 && car.x < -60) car.x = GAME_W + 60;
    }

    // Move gators
    for (let g of gators) {
        g.x += g.vx;
        if (g.vx > 0 && g.x > GAME_W + 70) g.x = -70;
        if (g.vx < 0 && g.x < -70) g.x = GAME_W + 70;
    }

    // Move logs
    for (let l of logs) {
        l.x += l.vx;
        if (l.vx > 0 && l.x > GAME_W + 80) l.x = -80;
        if (l.vx < 0 && l.x < -80) l.x = GAME_W + 80;
    }

    // Collision: frog vs cars
    for (let car of cars) {
        if (rectCircle(car.x - car.w/2, car.y - car.h/2, car.w, car.h, frog.x, frog.y, frog.r)) {
            killFrog();
            return;
        }
    }

    // Collision: frog vs gators
    for (let g of gators) {
        if (rectCircle(g.x - g.w/2, g.y - g.h/2, g.w, g.h, frog.x, frog.y, frog.r)) {
            killFrog();
            return;
        }
    }

    // Lily pad check
    for (let pad of lilyPads) {
        const dx = frog.x - pad.x;
        const dy = frog.y - pad.y;
        if (Math.sqrt(dx*dx + dy*dy) < frog.r + pad.r && !pad.filled) {
            pad.filled = true;
            score += 100;
            playSound(880, 0.3, 'sine', 0.2);
            spawnParticles(pad.x, pad.y, '#7fff7f', 15);
            
            if (lilyPads.every(p => p.filled)) {
                // New round
                lilyPads.forEach(p => p.filled = false);
                score += 500;
                playSound(1200, 0.4, 'sine', 0.2);
            }
            
            // Respawn frog
            respawnFrog();
        }
    }

    // Particles
    for (let i = particles.length - 1; i >= 0; i--) {
        particles[i].x += particles[i].vx;
        particles[i].y += particles[i].vy;
        particles[i].life -= 0.03;
        if (particles[i].life <= 0) particles.splice(i, 1);
    }
}

function killFrog() {
    playSound(150, 0.3, 'sawtooth', 0.15);
    spawnParticles(frog.x, frog.y, '#ff4444', 20);
    lives--;
    if (lives <= 0) {
        gameState = 'gameover';
        try { 
            if (score > highScore) {
                highScore = score;
                localStorage.setItem('physicsFroggerHS', highScore.toString());
            }
        } catch(e) {}
    } else {
        respawnFrog();
    }
}

function respawnFrog() {
    frog.x = GAME_W / 2;
    frog.y = GAME_H - 50;
    frog.vx = 0;
    frog.vy = 0;
}

function rectCircle(rx, ry, rw, rh, cx, cy, cr) {
    const dx = Math.abs(cx - (rx + rw/2));
    const dy = Math.abs(cy - (ry + rh/2));
    if (dx > rw/2 + cr || dy > rh/2 + cr) return false;
    if (dx <= rw/2 || dy <= rh/2) return true;
    const cornerX = dx - rw/2;
    const cornerY = dy - rh/2;
    return cornerX*cornerX + cornerY*cornerY <= cr*cr;
}

    ctx.restore();
}

function drawGameOver() {
