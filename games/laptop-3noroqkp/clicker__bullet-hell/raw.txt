const canvas = document.getElementById('c');
const ctx = canvas.getContext('2d');

let W, H;

// Audio
let audioCtx = null;
function initAudio() {
    if (!audioCtx) {
        audioCtx = new (window.AudioContext || window.webkitAudioContext)();
    }
}

function playSound(freq, type, duration, volume) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = type;
    osc.frequency.value = freq;
    gain.gain.setValueAtTime(volume, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + duration);
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + duration);
}

function playShoot() { playSound(800, 'square', 0.05, 0.08); }
function playHit() { playSound(200, 'sawtooth', 0.3, 0.15); }
function playExplode() { playSound(100, 'sawtooth', 0.4, 0.2); }
function playDie() { playSound(80, 'square', 0.6, 0.25); }

// Game state
let state = 'START'; // START, PLAYING, GAMEOVER
let score = 0;
let highScore = 0;
try { highScore = parseInt(localStorage.getItem('bulletStormHigh') || '0'); } catch(e) {}
let time = 0;
let shakeTime = 0;
let shakeMag = 0;

// Input
let mouseX = W / 2;
let mouseY = H - 80;
let mouseDown = false;
window.addEventListener('mousemove', e => { mouseX = e.clientX; mouseY = e.clientY; });
window.addEventListener('mousedown', e => {
    if (e.button === 0) {
        initAudio();
        mouseDown = true;
        if (state === 'START') startGame();
        else if (state === 'GAMEOVER') startGame();
    }
});
window.addEventListener('mouseup', e => { if (e.button === 0) mouseDown = false; });
window.addEventListener('keydown', e => {
    if (e.key === 'r' || e.key === 'R') {
        if (state !== 'PLAYING') startGame();
    }
});

// Player
let player = {};
function resetPlayer() {
    player = { x: W/2, y: H - 80, hp: 5, maxHp: 5, invincible: 0, shootTimer: 0 };
}

// Entities
let playerBullets = [];
let enemyBullets = [];
let enemies = [];
let particles = [];
let floatTexts = [];

function startGame() {
    state = 'PLAYING';
    score = 0;
    time = 0;
    playerBullets = [];
    enemyBullets = [];
    enemies = [];
    particles = [];
    floatTexts = [];
    resetPlayer();
}

// Enemy spawning
let spawnTimer = 0;
function spawnEnemy() {
    const type = Math.random();
    let e = { x: Math.random() * (W - 80) + 40, y: -30, hp: 3, maxHp: 3, r: 25, shootTimer: 0, fireRate: 60, pattern: 'aimed', patternPhase: 0, life: 0, vx: (Math.random()-0.5)*1.5, vy: 1 + Math.random()*1.5 };
    
    const diff = Math.min(time / 3000, 3);
    
    if (type < 0.2) {
        e.pattern = 'radial';
        e.fireRate = 90 - diff * 15;
        e.hp = 4 + Math.floor(diff * 2);
        e.maxHp = e.hp;
        e.r = 30;
    } else if (type < 0.4) {
        e.pattern = 'spiral';
        e.fireRate = 8 - diff * 2;
        e.hp = 5 + Math.floor(diff * 2);
        e.maxHp = e.hp;
        e.r = 28;
    } else if (type < 0.6) {
        e.pattern = 'fan';
        e.fireRate = 40 - diff * 8;
        e.hp = 3 + Math.floor(diff);
        e.maxHp = e.hp;
        e.r = 22;
    } else if (type < 0.8) {
        e.pattern = 'aimed';
        e.fireRate = 50 - diff * 10;
        e.hp = 3 + Math.floor(diff);
        e.maxHp = e.hp;
        e.r = 24;
    } else {
        e.pattern = 'wall';
        e.fireRate = 70 - diff * 15;
        e.hp = 6 + Math.floor(diff * 3);
        e.maxHp = e.hp;
        e.r = 35;
        e.vy = 0.3;
    }
    
    enemies.push(e);
}

function firePattern(e) {
    const px = player.x, py = player.y;
    const dx = px - e.x, dy = py - e.y;
    const angle = Math.atan2(dy, dx);
    const speed = 3 + Math.min(time / 5000, 2);
    
    if (e.pattern === 'aimed') {
        for (let i = -1; i <= 1; i++) {
            enemyBullets.push({ x: e.x, y: e.y + e.r, vx: Math.cos(angle + i * 0.3) * speed, vy: Math.sin(angle + i * 0.3) * speed, r: 5, color: '#ff4466' });
        }
    } else if (e.pattern === 'radial') {
        const count = 8 + Math.floor(Math.min(time / 2000, 4));
        for (let i = 0; i < count; i++) {
            const a = (Math.PI * 2 / count) * i + e.patternPhase;
            enemyBullets.push({ x: e.x, y: e.y, vx: Math.cos(a) * speed, vy: Math.sin(a) * speed, r: 4, color: '#ff8844' });
        }
    } else if (e.pattern === 'spiral') {
        const a = e.patternPhase;
        for (let i = 0; i < 3; i++) {
            enemyBullets.push({ x: e.x, y: e.y, vx: Math.cos(a + i * 2.09) * speed, vy: Math.sin(a + i * 2.09) * speed, r: 4, color: '#aa44ff' });
        }
        e.patternPhase += 0.5;
    } else if (e.pattern === 'fan') {
        for (let i = -2; i <= 2; i++) {
            const a = Math.PI/2 + i * 0.4;
            enemyBullets.push({ x: e.x, y: e.y + e.r, vx: Math.cos(a) * speed, vy: Math.sin(a) * speed, r: 5, color: '#ffcc33' });
        }
    } else if (e.pattern === 'wall') {
        const count = Math.floor(W / 40);
        for (let i = 0; i < count; i++) {
            if (Math.random() > 0.3) {
                enemyBullets.push({ x: i * 40 + 20, y: e.y + e.r, vx: 0, vy: speed * 0.7, r: 5, color: '#ff2222' });
            }
        }
    }
}

function addParticles(x, y, color, count) {
    for (let i = 0; i < count; i++) {
        const a = Math.random() * Math.PI * 2;
        const s = Math.random() * 4 + 1;
        particles.push({ x, y, vx: Math.cos(a)*s, vy: Math.sin(a)*s, life: 30 + Math.random()*20, maxLife: 50, r: Math.random()*3+1, color });
    }
}

function addFloatText(x, y, text, color) {
    floatTexts.push({ x, y, text, color, life: 40, vy: -1.5 });
}

// Update
function update() {
    if (state !== 'PLAYING') return;
    
    time++;
    score += Math.floor(time / 3);
    
    // Player movement (smooth follow)
    player.x += (mouseX - player.x) * 0.15;
    player.y += (mouseY - player.y) * 0.15;
    player.x = Math.max(20, Math.min(W - 20, player.x));
    player.y = Math.max(H/2, Math.min(H - 30, player.y));
    
    if (player.invincible > 0) player.invincible--;
    
    // Shooting
    player.shootTimer--;
    if (mouseDown && player.shootTimer <= 0) {
        playerBullets.push({ x: player.x, y: player.y - 15, vx: 0, vy: -8, r: 4, color: '#44ffcc' });
        player.shootTimer = 8;
        playShoot();
    }
    
    // Spawn enemies
    spawnTimer--;
    const spawnRate = Math.max(120 - Math.floor(time / 50), 30);
    if (spawnTimer <= 0) {
        spawnEnemy();
        spawnTimer = spawnRate;
    }
    
    // Update player bullets
    for (let i = playerBullets.length - 1; i >= 0; i--) {
        const b = playerBullets[i];
        b.y += b.vy;
        if (b.y < -20) { playerBullets.splice(i, 1); continue; }
        
        // Hit enemies
        for (let j = enemies.length - 1; j >= 0; j--) {
            const e = enemies[j];
            const dx = b.x - e.x, dy = b.y - e.y;
            if (dx*dx + dy*dy < (b.r + e.r) * (b.r + e.r)) {
                e.hp--;
                playerBullets.splice(i, 1);
                addParticles(b.x, b.y, '#44ffcc', 5);
                playHit();
                if (e.hp <= 0) {
                    addParticles(e.x, e.y, '#ff8844', 20);
                    addFloatText(e.x, e.y, '+' + (100 + Math.floor(time/100)), '#fff');
                    score += 100;
                    playExplode();
                    enemies.splice(j, 1);
                }
                break;
            }
        }
    }
    
    // Update enemies
    for (let i = enemies.length - 1; i >= 0; i--) {
        const e = enemies[i];
        e.x += e.vx;
        e.y += e.vy;
        e.life++;
        
        if (e.x < e.r || e.x > W - e.r) e.vx *= -1;
        
        // Fire
        e.shootTimer--;
        if (e.shootTimer <= 0 && e.y > 0) {
            firePattern(e);
            e.shootTimer = e.fireRate;
        }
        
        // Remove if off screen
        if (e.y > H + 50) enemies.splice(i, 1);
    }
    
    // Update enemy bullets
    for (let i = enemyBullets.length - 1; i >= 0; i--) {
        const b = enemyBullets[i];
        b.x += b.vx;
        b.y += b.vy;
        
        if (b.x < -20 || b.x > W + 20 || b.y < -20 || b.y > H + 20) {
            enemyBullets.splice(i, 1);
            continue;
        }
        
        // Hit player
        if (player.invincible <= 0) {
            const dx = b.x - player.x, dy = b.y - player.y;
            if (dx*dx + dy*dy < (b.r + 8) * (b.r + 8)) {
                enemyBullets.splice(i, 1);
                player.hp--;
                player.invincible = 60;
                shakeTime = 10;
                shakeMag = 5;
                addParticles(player.x, player.y, '#ff4444', 15);
                playDie();
                if (player.hp <= 0) {
                    state = 'GAMEOVER';
                    addParticles(player.x, player.y, '#ffffff', 40);
                    try { 
                        if (score > highScore) {
                            highScore = score;
                            localStorage.setItem('bulletStormHigh', highScore);
                        }
                    } catch(e2) {}
                }
            }
        }
    }
    
    // Update particles
    for (let i = particles.length - 1; i >= 0; i--) {
        const p = particles[i];
        p.x += p.vx;
        p.y += p.vy;
        p.vx *= 0.95;
        p.vy *= 0.95;
        p.life--;
        if (p.life <= 0) particles.splice(i, 1);
    }
    
    // Update float texts
    for (let i = floatTexts.length - 1; i >= 0; i--) {
        const t = floatTexts[i];
        t.y += t.vy;
        t.life--;
        if (t.life <= 0) floatTexts.splice(i, 1);
    }
    
    // Shake
    if (shakeTime > 0) shakeTime--;
}

// Render
function draw() {
    ctx.save();
    
    // Screen shake
    if (shakeTime > 0) {
        const sx = (Math.random() - 0.5) * shakeMag;
        const sy = (Math.random() - 0.5) * shakeMag;
        ctx.translate(sx, sy);
    }
    
    // Background
    ctx.fillStyle = '#0a0a1a';
    ctx.fillRect(-10, -10, W + 20, H + 20);
    
    // Grid
    ctx.strokeStyle = 'rgba(30, 40, 80, 0.3)';
    ctx.lineWidth = 1;
    const gridSize = 50;
    for (let x = 0; x < W; x += gridSize) {
        ctx.beginPath(); ctx.moveTo(x, 0); ctx.lineTo(x, H); ctx.stroke();
    }
    for (let y = 0; y < H; y += gridSize) {
        ctx.beginPath(); ctx.moveTo(0, y); ctx.lineTo(W, y); ctx.stroke();
    }
    
    if (state === 'START') {
        drawStartScreen();
        ctx.restore();
        return;
    }
    
    // Particles
    for (const p of particles) {
        const alpha = p.life / p.maxLife;
        ctx.fillStyle = p.color + Math.floor(alpha * 255).toString(16).padStart(2,'0');
        ctx.beginPath();
        ctx.arc(p.x, p.y, p.r * alpha, 0, Math.PI*2);
        ctx.fill();
    }
    
    // Player bullets
    for (const b of playerBullets) {
        ctx.fillStyle = b.color;
        ctx.shadowColor = b.color;
        ctx.shadowBlur = 8;
        ctx.beginPath();
        ctx.arc(b.x, b.y, b.r, 0, Math.PI*2);
        ctx.fill();
        ctx.shadowBlur = 0;
    }
    
    // Enemy bullets
    for (const b of enemyBullets) {
        ctx.fillStyle = b.color;
        ctx.shadowColor = b.color;
        ctx.shadowBlur = 6;
        ctx.beginPath();
        ctx.arc(b.x, b.y, b.r, 0, Math.PI*2);
        ctx.fill();
        ctx.shadowBlur = 0;
    }
    
    // Enemies
    for (const e of enemies) {
        const pulse = 1 + Math.sin(time * 0.1) * 0.05;
        const r = e.r * pulse;
        
        // Body
        ctx.fillStyle = '#2a1a3a';
        ctx.strokeStyle = e.pattern === 'radial' ? '#ff8844' : e.pattern === 'spiral' ? '#aa44ff' : e.pattern === 'fan' ? '#ffcc33' : e.pattern === 'wall' ? '#ff2222' : '#ff4466';
        ctx.lineWidth = 2;
        ctx.shadowColor = ctx.strokeStyle;
        ctx.shadowBlur = 10;
        
        // Draw as hexagon
        ctx.beginPath();
        for (let i = 0; i < 6; i++) {
            const a = (Math.PI * 2 / 6) * i + time * 0.02;
            const px = e.x + Math.cos(a) * r;
            const py = e.y + Math.sin(a) * r;
            if (i === 0) ctx.moveTo(px, py); else ctx.lineTo(px, py);
        }
        ctx.closePath();
        ctx.fill();
        ctx.stroke();
        
        // HP bar
        ctx.shadowBlur = 0;
        ctx.fillStyle = '#333';
        ctx.fillRect(e.x - 20, e.y + r + 5, 40, 4);
        ctx.fillStyle = '#ff4466';
        ctx.fillRect(e.x - 20, e.y + r + 5, 40 * (e.hp / e.maxHp), 4);
        
        ctx.shadowBlur = 0;
    }
    
    // Player
    if (state === 'PLAYING') {
        const blink = player.invincible > 0 && Math.floor(player.invincible / 5) % 2 === 0;
        if (!blink) {
            ctx.shadowColor = '#44ffcc';
            ctx.shadowBlur = 12;
            ctx.fillStyle = '#44ffcc';
            ctx.beginPath();
            ctx.moveTo(player.x, player.y - 15);
            ctx.lineTo(player.x - 10, player.y + 10);
            ctx.lineTo(player.x + 10, player.y + 10);
            ctx.closePath();
            ctx.fill();
            
            // Engine glow
            const eg = Math.random() * 5;
            ctx.fillStyle = `rgba(68, 255, 204, ${0.3 + Math.random()*0.3})`;
            ctx.beginPath();
            ctx.arc(player.x, player.y + 12 + eg * 0.3, 4 + Math.random()*2, 0, Math.PI*2);
            ctx.fill();
            
            ctx.shadowBlur = 0;
        }
    }
    
    // Float texts
    for (const t of floatTexts) {
        const alpha = t.life / 40;
        ctx.fillStyle = t.color + Math.floor(alpha * 255).toString(16).padStart(2,'0');
        ctx.font = 'bold 16px monospace';
        ctx.textAlign = 'center';
        ctx.fillText(t.text, t.x, t.y);
    }
    
    // HUD
    if (state === 'PLAYING' || state === 'GAMEOVER') {
        ctx.fillStyle = '#fff';
        ctx.font = 'bold 18px monospace';
        ctx.textAlign = 'left';
        ctx.fillText('SCORE: ' + score, 20, 30);
        ctx.font = '14px monospace';
        ctx.fillText('BEST: ' + Math.max(highScore, score), 20, 50);
        
        // HP
        for (let i = 0; i < player.maxHp; i++) {
            ctx.fillStyle = i < player.hp ? '#ff4466' : '#333';
            ctx.beginPath();
            ctx.arc(W - 120 + i * 25, 25, 8, 0, Math.PI*2);
            ctx.fill();
        }
        
        // Difficulty indicator
        const diff = Math.min(time / 3000, 3);
        ctx.fillStyle = '#888';
        ctx.font = '12px monospace';
        ctx.textAlign = 'right';
        ctx.fillText('THREAT LVL: ' + (1 + Math.floor(diff)), W - 20, 50);
    }
    
    if (state === 'GAMEOVER') {
        drawGameOver();
    }
    
    ctx.restore();
}

function drawStartScreen() {
    ctx.fillStyle = '#fff';
    ctx.font = 'bold 48px monospace';
    ctx.textAlign = 'center';
    ctx.shadowColor = '#44ffcc';
    ctx.shadowBlur = 20;
    ctx.fillText('BULLET STORM', W/2, H/3);
    ctx.shadowBlur = 0;
    
    ctx.font = '18px monospace';
    ctx.fillStyle = '#aaa';
    ctx.fillText('Move: Mouse | Shoot: Hold Click', W/2, H/3 + 50);
    ctx.fillText('Dodge the bullets. Destroy the enemies.', W/2, H/3 + 80);
    ctx.fillText('Survive as long as you can!', W/2, H/3 + 110);
    
    if (highScore > 0) {
        ctx.fillStyle = '#ffcc33';
        ctx.font = '16px monospace';
        ctx.fillText('HIGH SCORE: ' + highScore, W/2, H/3 + 150);
    }
    
    const pulse = 0.5 + Math.sin(Date.now() * 0.005) * 0.5;
    ctx.fillStyle = `rgba(68, 255, 204, ${pulse})`;
    ctx.font = 'bold 22px monospace';
    ctx.fillText('CLICK TO START', W/2, H/3 + 210);
}

function drawGameOver() {
    ctx.fillStyle = 'rgba(0,0,0,0.7)';
    ctx.fillRect(0, 0, W, H);
    
    ctx.fillStyle = '#ff4466';
    ctx.font = 'bold 48px monospace';
    ctx.textAlign = 'center';
    ctx.shadowColor = '#ff4466';
    ctx.shadowBlur = 20;
    ctx.fillText('GAME OVER', W/2, H/3);
    ctx.shadowBlur = 0;
    
    ctx.fillStyle = '#fff';
    ctx.font = '24px monospace';
    ctx.fillText('Score: ' + score, W/2, H/3 + 50);
    
    if (score >= highScore && score > 0) {
        ctx.fillStyle = '#ffcc33';
        ctx.font = '18px monospace';
        ctx.fillText('NEW HIGH SCORE!', W/2, H/3 + 85);
    }
    
    const pulse = 0.5 + Math.sin(Date.now() * 0.005) * 0.5;
    ctx.fillStyle = `rgba(68, 255, 204, ${pulse})`;
    ctx.font = '18px monospace';
    ctx.fillText('Click to Restart (or press R)', W/2, H/3 + 140);
}

// Main loop
function loop() {
    update();
    draw();
    requestAnimationFrame(loop);
}
loop();
</script>
