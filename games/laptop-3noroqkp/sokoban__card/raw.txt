function parseLevel(idx) {
  const grid = LEVELS[idx].grid;
  playerX = -1; playerY = -1;
  cards = []; goals = []; walls = [];
  moves = 0; pushes = 0;
  
  for (let r = 0; r < grid.length; r++) {
    for (let c = 0; c < grid[r].length; c++) {
      const ch = grid[r][c];
      if (ch === '#') walls.push({x: c, y: r});
      if (ch === '@' || ch === '+') { playerX = c; playerY = r; }
      if (ch === '$') cards.push({x: c, y: r, cardIdx: cards.length, onGoal: false});
      if (ch === '.') goals.push({x: c, y: r});
    }
  }
  
  // Assign cards cyclically
  for (let i = 0; i < cards.length; i++) {
    cards[i].cardIdx = i % CARDS.length;
  }
}

function isWall(x, y) {
  return walls.some(w => w.x === x && w.y === y);
}

function getGridBounds() {
  let minX = Infinity, minY = Infinity, maxX = -Infinity, maxY = -Infinity;
  const all = [...walls, ...goals];
  all.forEach(p => {
    minX = Math.min(minX, p.x); minY = Math.min(minY, p.y);
    maxX = Math.max(maxX, p.x); maxY = Math.max(maxY, p.y);
  });
  return {minX, minY, maxX, maxY};
}

function checkWin() {
  return cards.every(c => c.onGoal);
}

function movePlayer(dx, dy) {
  const nx = playerX + dx;
  const ny = playerY + dy;
  
  if (isWall(nx, ny)) return;
  
  const cardIdx = cards.findIndex(c => c.x === nx && c.y === ny);
  
  if (cardIdx !== -1) {
    const card = cards[cardIdx];
    const bx = nx + dx;
    const by = ny + dy;
    
    if (isWall(bx, by)) return;
    if (cards.some(c => c.x === bx && c.y === by)) return;
    
    card.x = bx;
    card.y = by;
    card.onGoal = goals.some(g => g.x === bx && g.y === by);
    playerX = nx;
    playerY = ny;
    moves++;
    pushes++;
    playSound('push');
    
    if (card.onGoal) setTimeout(() => playSound('place'), 100);
  } else {
    playerX = nx;
    playerY = ny;
    moves++;
    playSound('move');
  }
  
  if (checkWin()) {
    winTimer = 60;
    playSound('win');
  }
}

function drawCard(x, y, size, cardData, onGoal, glow) {
  const pad = size * 0.08;
  const cx = x + size/2;
  const cy = y + size/2;
  
  ctx.save();

  ctx.shadowColor = 'transparent';

  ctx.lineWidth = Math.max(1, size * 0.03);
  roundRect(x + pad, y + pad, size - pad*2, size - pad*2, size * 0.1);
  ctx.stroke();
  
  // Glow if on goal
  if (glow) {
    ctx.shadowColor = cardData.color;
    ctx.shadowBlur = 15;
    ctx.strokeStyle = cardData.color;
    ctx.lineWidth = 2;
    roundRect(x + pad, y + pad, size - pad*2, size - pad*2, size * 0.1);
    ctx.stroke();
    ctx.shadowColor = 'transparent';
  }
  
  // Suit symbol
  const fontSize = size * 0.35;
  ctx.font = `bold ${fontSize}px Georgia`;
  ctx.textAlign = 'center';
  ctx.textBaseline = 'middle';
  ctx.fillStyle = cardData.color;
  ctx.fillText(cardData.suit, cx, cy - fontSize * 0.1);
  
  // Label
  const labelSize = size * 0.25;
  ctx.font = `bold ${labelSize}px Georgia`;
  ctx.fillStyle = cardData.color;
  ctx.fillText(cardData.label, x + pad + labelSize * 0.6, y + pad + labelSize * 0.7);
  
  ctx.restore();
}

function roundRect(x, y, w, h, r) {
  ctx.beginPath();
  ctx.moveTo(x + r, y);
  ctx.lineTo(x + w - r, y);
  ctx.quadraticCurveTo(x + w, y, x + w, y + r);
  ctx.lineTo(x + w, y + h - r);
  ctx.quadraticCurveTo(x + w, y + h, x + w - r, y + h);
  ctx.lineTo(x + r, y + h);
  ctx.quadraticCurveTo(x, y + h, x, y + h - r);
  ctx.lineTo(x, y + r);
  ctx.quadraticCurveTo(x, y, x + r, y);
  ctx.closePath();
}

    for (let c = bounds.minX; c <= bounds.maxX; c++) {
      if (!isWall(c, r)) {
        drawFloor(offsetX + (c - bounds.minX) * cellSize, offsetY + (r - bounds.minY) * cellSize, cellSize);
      }
    }
  }
  
  // Draw goals
  goals.forEach(g => {
    drawGoal(
      offsetX + (g.x - bounds.minX) * cellSize,
      offsetY + (g.y - bounds.minY) * cellSize,
      cellSize
    );
  });
  
  // Draw walls
  walls.forEach(w => {
    drawWall(
      offsetX + (w.x - bounds.minX) * cellSize,
      offsetY + (w.y - bounds.minY) * cellSize,
      cellSize
    );
  });
  
  // Draw cards
  cards.forEach(c => {
    const cardData = CARDS[c.cardIdx];
    drawCard(
      offsetX + (c.x - bounds.minX) * cellSize,
      offsetY + (c.y - bounds.minY) * cellSize,
      cellSize,
      cardData,
      c.onGoal,
      c.onGoal && animFrame % 60 < 30
    );
  });
  
  // Draw player
  drawPlayer(
    offsetX + (playerX - bounds.minX) * cellSize,
    offsetY + (playerY - bounds.minY) * cellSize,
    cellSize
  );
  
  // HUD
  ctx.fillStyle = 'rgba(0,0,0,0.6)';
  ctx.fillRect(0, 0, W, 50);
  
  ctx.fillStyle = '#f1c40f';
  ctx.font = `bold ${Math.min(24, W*0.03)}px Georgia`;
  ctx.textAlign = 'left';
  ctx.textBaseline = 'middle';
  ctx.fillText(`♥ ${LEVELS[level].name} ♥`, 20, 25);
  
  ctx.fillStyle = '#ecf0f1';
  ctx.font = `${Math.min(16, W*0.02)}px Georgia`;
  ctx.textAlign = 'right';
  const onGoalCount = cards.filter(c => c.onGoal).length;
  ctx.fillText(`Cards: ${onGoalCount}/${cards.length}  Moves: ${moves}`, W - 20, 25);
  
  // Win overlay
  if (winTimer > 0) {
    winTimer--;
    const alpha = Math.min(1, (60 - winTimer) / 20);
    ctx.fillStyle = `rgba(0,0,0,${alpha * 0.7})`;
    ctx.fillRect(0, 0, W, H);
    
    if (winTimer < 40) {
      ctx.fillStyle = '#f1c40f';
      ctx.font = `bold ${Math.min(48, W*0.06)}px Georgia`;
      ctx.textAlign = 'center';
      ctx.textBaseline = 'middle';
      ctx.fillText('★ Perfect Hand! ★', W/2, H/2 - 30);
      
      ctx.fillStyle = '#ecf0f1';
      ctx.font = `${Math.min(24, W*0.03)}px Georgia`;
      ctx.fillText(`Solved in ${moves} moves, ${pushes} pushes`, W/2, H/2 + 20);
      
      if (level < LEVELS.length - 1) {
        ctx.fillStyle = '#f39c12';
        ctx.font = `${Math.min(20, W*0.025)}px Georgia`;
        ctx.fillText('Press SPACE for next hand', W/2, H/2 + 70);
      } else {
        ctx.fillStyle = '#f39c12';
        ctx.font = `${Math.min(20, W*0.025)}px Georgia`;
        ctx.fillText('You\'ve collected the Royal Flush!', W/2, H/2 + 60);
        ctx.fillText('Press SPACE to play again', W/2, H/2 + 100);
      }
    }
    
    if (winTimer <= 0) {
      gameState = 'levelComplete';
    }
  }
}

function render() {
  animFrame++;
  
  if (gameState === 'start') {
    drawStartScreen();
  } else if (gameState === 'playing' || gameState === 'levelComplete') {
    drawFeltBackground();
    drawGame();
  } else if (gameState === 'allWin') {
    drawAllWin();
  }
  
  requestAnimationFrame(render);
}

// Input handling
window.addEventListener('keydown', (e) => {
  initAudio();
  
  if (gameState === 'start') {
    if (e.key === ' ' || e.key === 'Enter') {
      playSound('click');
      level = 0;
      parseLevel(level);
      gameState = 'playing';
    }
  } else if (gameState === 'playing') {
    let dx = 0, dy = 0;
    if (e.key === 'ArrowUp' || e.key === 'w' || e.key === 'W') dy = -1;
    if (e.key === 'ArrowDown' || e.key === 's' || e.key === 'S') dy = 1;
    if (e.key === 'ArrowLeft' || e.key === 'a' || e.key === 'A') dx = -1;
    if (e.key === 'ArrowRight' || e.key === 'd' || e.key === 'D') dx = 1;
    
    if (dx !== 0 || dy !== 0) {
      e.preventDefault();
      movePlayer(dx, dy);
    }
    
    if (e.key === 'u' || e.key === 'U') {
      parseLevel(level);
      playSound('click');
    }
  } else if (gameState === 'levelComplete') {
    if (e.key === ' ' || e.key === 'Enter') {
      playSound('click');
      level++;
      if (level >= LEVELS.length) {
        gameState = 'allWin';
      } else {
        parseLevel(level);
        gameState = 'playing';
      }
    }
  } else if (gameState === 'allWin') {
    if (e.key === ' ' || e.key === 'Enter') {
      playSound('click');
      level = 0;
      parseLevel(level);
      gameState = 'playing';
    }
  }
});
