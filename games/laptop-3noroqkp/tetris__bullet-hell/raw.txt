(function(){
const canvas = document.getElementById('game');
const ctx = canvas.getContext('2d');

let W, H;
function resize() {
    W = window.innerWidth;
    H = window.innerHeight;
    canvas.width = W;
    canvas.height = H;
}
window.addEventListener('resize', resize);
resize();

// Audio
let audioCtx = null;
function initAudio() {
    if (!audioCtx) {
        audioCtx = new (window.AudioContext || window.webkitAudioContext)();
    }
}
function playSound(freq, type, duration, vol) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = type;
    osc.frequency.value = freq;
    gain.gain.setValueAtTime(vol || 0.15, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + (duration || 0.1));
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + (duration || 0.1));
}

// Game constants
const COLS = 10, ROWS = 20;
const PIECES = [
    [[0,0],[1,0],[2,0],[3,0]], // I
    [[0,0],[1,0],[0,1],[1,1]], // O
    [[1,0],[1,1],[0,1],[2,1]], // T
    [[0,0],[1,0],[1,1],[2,1]], // S
    [[1,0],[1,1],[0,1],[1,2]], // Z
    [[0,0],[0,1],[1,1],[2,1]], // J
    [[1,0],[1,1],[1,2],[2,2]]  // L
];
const COLORS = ['#00f5ff','#ffee00','#b400ff','#00ff88','#ff3366','#ff6600','#ff00aa'];

// Game state
let gameState = 'start'; // start, playing, gameover
let grid, score, lives, level, combo;
let currentPiece, curX, curY, curRot;
let nextPiece;
let bullets, enemies, particles, floatTexts;
let lastTime, spawnTimer, bulletTimer, gameTime;
let shake = 0;

function initGame() {
    grid = Array.from({length: ROWS}, () => Array(COLS).fill(null));
    score = 0; lives = 3; level = 1; combo = 0;
    bullets = []; enemies = []; particles = []; floatTexts = [];
    gameTime = 0; spawnTimer = 0; bulletTimer = 0;
    shake = 0;
    nextPiece = randomPiece();
    spawnPiece();
}

function randomPiece() {
    return Math.floor(Math.random() * PIECES.length);
}

function spawnPiece() {
    currentPiece = nextPiece;
    nextPiece = randomPiece();
    curRot = 0;
    curX = Math.floor((COLS - getShape(currentPiece, 0).width) / 2);
    curY = 0;
    if (collides(getShape(currentPiece, 0), curX, curY)) {
        gameState = 'gameover';
    }
}

function getShape(type, rot) {
    const base = PIECES[type];
    let cells = base.map(c => [...c]);
    for (let r = 0; r < rot % 4; r++) {
        cells = cells.map(c => [c[1], -c[0]]);
        const minX = Math.min(...cells.map(c => c[0]));
        const minY = Math.min(...cells.map(c => c[1]));
        cells = cells.map(c => [c[0] - minX, c[1] - minY]);
    }
    const w = Math.max(...cells.map(c => c[0])) + 1;
    const h = Math.max(...cells.map(c => c[1])) + 1;
    return { cells, width: w, height: h };
}

function collides(shape, px, py) {
    for (const [cx, cy] of shape.cells) {
        const gx = px + cx, gy = py + cy;
        if (gx < 0 || gx >= COLS || gy >= ROWS) return true;
        if (gy >= 0 && grid[gy][gx]) return true;
    }
    return false;
}

function lockPiece() {
    const shape = getShape(currentPiece, curRot);
    for (const [cx, cy] of shape.cells) {
        const gx = curX + cx, gy = curY + cy;
        if (gy >= 0 && gy < ROWS && gx >= 0 && gx < COLS) {
            grid[gy][gx] = currentPiece;
        }
    }
    clearLines();
    spawnPiece();
}

function clearLines() {
    let linesCleared = 0;
    for (let r = ROWS - 1; r >= 0; r--) {
        if (grid[r].every(c => c !== null)) {
            grid.splice(r, 1);
            grid.unshift(Array(COLS).fill(null));
            linesCleared++;
            // Particles for line clear
            for (let x = 0; x < COLS; x++) {
                spawnBurst(getCellX(x), getCellY(r - r), COLORS[grid[r] ? currentPiece : 0], 3);
            }
        }
    }
    if (linesCleared > 0) {
        combo++;
        const pts = linesCleared * 100 * level * (combo > 1 ? combo : 1);
        score += pts;
        addFloatText('+' + pts, getCellX(COLS/2), getCellY(ROWS - 3), '#fff');
        // Destroy all bullets on screen!
        for (let i = bullets.length - 1; i >= 0; i--) {
            spawnBurst(bullets[i].x, bullets[i].y, '#ff0', 2);
            bullets.splice(i, 1);
        }
        playSound(880, 'square', 0.3, 0.2);
        shake = 5;
    } else {
        combo = 0;
    }
}

function getCellX(col) { return gridOffsetX + col * cellSize + cellSize/2; }
function getCellY(row) { return gridOffsetY + row * cellSize + cellSize/2; }

let cellSize, gridOffsetX, gridOffsetY;

// Player hitbox (center of current piece)
function getPlayerPos() {
    const shape = getShape(currentPiece, curRot);
    let cx = 0, cy = 0;
    for (const [px, py] of shape.cells) {
        cx += px + curX;
        cy += py + curY;
    }
    cx /= shape.cells.length;
    cy /= shape.cells.length;
    return { x: getCellX(cx), y: getCellY(cy) };
}

// Enemies
function spawnEnemy() {
    const types = ['spread', 'spiral', 'rain'];
    const type = types[Math.floor(Math.random() * types.length)];
    const ex = gridOffsetX + Math.random() * (COLS * cellSize - cellSize);
    enemies.push({
        x: ex,
        y: 20,
        type: type,
        hp: 3,
        timer: 0,
        life: 300 + Math.random() * 200,
        angle: 0
    });
}

function updateEnemies(dt) {
    for (let i = enemies.length - 1; i >= 0; i--) {
        const e = enemies[i];
        e.timer += dt;
        e.life -= dt;
        if (e.life <= 0 || e.hp <= 0) {
            spawnBurst(e.x, e.y, '#f0f', 4);
            enemies.splice(i, 1);
            continue;
        }
        // Fire patterns
        const fireRate = Math.max(20, 60 - level * 5);
        if (e.timer % fireRate < dt * 1000) {
            e.timer = 0;
            const pp = getPlayerPos();
            switch (e.type) {
                case 'spread': {
                    const baseAngle = Math.atan2(pp.y - e.y, pp.x - e.x);
                    for (let a = -1; a <= 1; a++) {
                        const ang = baseAngle + a * 0.3;
                        bullets.push({ x: e.x, y: e.y, vx: Math.cos(ang) * 3, vy: Math.sin(ang) * 3, r: 5, color: '#ff4444' });
                    }
                    break;
                }
                case 'spiral': {
                    e.angle += 0.4;
                    for (let a = 0; a < 6; a++) {
                        const ang = e.angle + a * Math.PI / 3;
                        bullets.push({ x: e.x, y: e.y, vx: Math.cos(ang) * 2.5, vy: Math.sin(ang) * 2.5, r: 4, color: '#ff8800' });
                    }
                    break;
                }
                case 'rain': {
                    const bx = gridOffsetX + Math.random() * (COLS * cellSize);
                    bullets.push({ x: bx, y: 10, vx: 0, vy: 3, r: 5, color: '#44ff44' });
                    break;
                }
            }
        }
    }
}

// Particles
function spawnBurst(x, y, color, count) {
    for (let i = 0; i < (count * 4); i++) {
        const ang = Math.random() * Math.PI * 2;
        const spd = 1 + Math.random() * 3;
        particles.push({ x, y, vx: Math.cos(ang) * spd, vy: Math.sin(ang) * spd, life: 30 + Math.random() * 30, color, r: 2 + Math.random() * 3 });
    }
}

function addFloatText(text, x, y, color) {
    floatTexts.push({ text, x, y, life: 60, color: color || '#fff' });
}

// Input
let keys = {};
window.addEventListener('keydown', (e) => {
    if (gameState === 'start') {
        initAudio();
        gameState = 'playing';
        initGame();
        return;
    }
    if (gameState === 'gameover' && e.code === 'Space') {
        gameState = 'playing';
        initGame();
        return;
    }
    keys[e.code] = true;
    if (gameState !== 'playing') return;
    if (!audioCtx) initAudio();

    const shape = getShape(currentPiece, curRot);
    switch (e.code) {
        case 'ArrowLeft':
            if (!collides(shape, curX - 1, curY)) { curX--; }
            break;
        case 'ArrowRight':
            if (!collides(shape, curX + 1, curY)) { curX++; }
            break;
        case 'ArrowUp':
            const newRot = (curRot + 1) % 4;
            const newShape = getShape(currentPiece, newRot);
            // Wall kick
            const kicks = [[0,-1],[0,1],[-1,0],[1,0],[0,0]];
            for (const [kx, ky] of kicks) {
                if (!collides(newShape, curX + kx, curY + ky)) {
                    curRot = newRot;
                    curX += kx;
                    curY += ky;
                    break;
                }
            }
            playSound(300, 'square', 0.05, 0.1);
            break;
        case 'ArrowDown':
            if (!collides(shape, curX, curY + 1)) { curY++; score += 1; }
            else lockPiece();
            break;
        case 'Space': {
            while (!collides(shape, curX, curY + 1)) { curY++; }
            lockPiece();
            playSound(200, 'square', 0.1, 0.15);
            break;
        }
    }
});
window.addEventListener('keyup', (e) => { keys[e.code] = false; });

// Main loop
let lastFrame = performance.now();
function gameLoop(now) {
    requestAnimationFrame(gameLoop);
    const dt = Math.min((now - lastFrame) / 16.67, 3);
    lastFrame = now;

    // Calculate layout
    cellSize = Math.min(Math.floor(W * 0.45 / COLS), Math.floor(H * 0.7 / ROWS));
    gridOffsetX = Math.floor((W - COLS * cellSize) / 2);
    gridOffsetY = Math.floor(H * 0.08);

    if (gameState === 'playing') {
        gameTime += dt;
        level = Math.floor(gameTime / 300) + 1;

        // Gravity
        const gravity = Math.max(2, 8 - level);
        bulletTimer += dt;
        if (bulletTimer >= gravity) {
            bulletTimer = 0;
            const shape = getShape(currentPiece, curRot);
            if (!collides(shape, curX, curY + 1)) {
                curY++;
            } else {
                lockPiece();
            }
        }

        // Spawn enemies
        spawnTimer += dt;
        const spawnRate = Math.max(80, 200 - level * 15);
        if (spawnTimer >= spawnRate && enemies.length < 3 + level) {
            spawnTimer = 0;
            spawnEnemy();
        }

        updateEnemies(dt * 16.67);

        // Update bullets
        for (let i = bullets.length - 1; i >= 0; i--) {
            const b = bullets[i];
            b.x += b.vx * dt;
            b.y += b.vy * dt;
            if (b.y > H + 20 || b.y < -50 || b.x < -50 || b.x > W + 50) {
                bullets.splice(i, 1);
                continue;
            }
            // Hit player?
            const pp = getPlayerPos();
            const dx = b.x - pp.x, dy = b.y - pp.y;
            if (dx * dx + dy * dy < (b.r + 8) * (b.r + 8)) {
                bullets.splice(i, 1);
                lives--;
                shake = 10;
                spawnBurst(pp.x, pp.y, '#ff0', 5);
                addFloatText('HIT!', pp.x, pp.y - 20, '#f44');
                playSound(100, 'sawtooth', 0.3, 0.3);
                if (lives <= 0) {
                    gameState = 'gameover';
                }
            }
        }

        // Update particles
        for (let i = particles.length - 1; i >= 0; i--) {
            const p = particles[i];
            p.x += p.vx * dt;
            p.y += p.vy * dt;
            p.life -= dt;
            if (p.life <= 0) particles.splice(i, 1);
        }

        // Float texts
        for (let i = floatTexts.length - 1; i >= 0; i--) {
            floatTexts[i].y -= 1 * dt;
            floatTexts[i].life -= dt;
            if (floatTexts[i].life <= 0) floatTexts.splice(i, 1);
        }

        // Shake decay
        shake *= 0.85;
    }

    // Render
    ctx.fillStyle = '#0a0a1a';
    ctx.fillRect(0, 0, W, H);

    if (gameState === 'start') {
        renderStartScreen();
    } else {
        renderGame();
    }
}

function renderStartScreen() {
    ctx.textAlign = 'center';
    ctx.fillStyle = '#0ff';
    ctx.font = `bold ${Math.min(W*0.08, 60)}px monospace`;
    ctx.fillText('BULLET HELL TETRIS', W/2, H * 0.3);

    ctx.fillStyle = '#fff';
    ctx.font = `${Math.min(W*0.03, 18)}px monospace`;
    ctx.fillText('Tetris meets Danmaku!', W/2, H * 0.42);
    ctx.fillText('', W/2, H * 0.5);
    ctx.fillText('← → Move  |  ↑ Rotate  |  ↓ Soft Drop', W/2, H * 0.52);
    ctx.fillText('SPACE Hard Drop', W/2, H * 0.57);
    ctx.fillText('', W/2, H * 0.64);
    ctx.fillText('Dodge bullets with your falling piece!', W/2, H * 0.64);
    ctx.fillText('Clear lines to destroy all bullets on screen!', W/2, H * 0.68);

    ctx.fillStyle = `hsl(${(Date.now()/20)%360},100%,70%)`;
    ctx.font = `bold ${Math.min(W*0.04, 28)}px monospace`;
    ctx.fillText('PRESS ANY KEY TO START', W/2, H * 0.8);

function renderGame() {
    ctx.save();
    if (shake > 0.5) {
        ctx.translate((Math.random()-0.5)*shake, (Math.random()-0.5)*shake);
    }

    // Grid background
    ctx.strokeStyle = 'rgba(255,255,255,0.05)';
    ctx.lineWidth = 1;
    for (let x = 0; x <= COLS; x++) {
        ctx.beginPath();
        ctx.moveTo(gridOffsetX + x * cellSize, gridOffsetY);
        ctx.lineTo(gridOffsetX + x * cellSize, gridOffsetY + ROWS * cellSize);
        ctx.stroke();
    }
    for (let y = 0; y <= ROWS; y++) {
        ctx.beginPath();
        ctx.moveTo(gridOffsetX, gridOffsetY + y * cellSize);
        ctx.lineTo(gridOffsetX + COLS * cellSize, gridOffsetY + y * cellSize);
        ctx.stroke();
    }

    // Grid border glow
    ctx.strokeStyle = 'rgba(0,255,255,0.3)';
    ctx.lineWidth = 2;
    ctx.strokeRect(gridOffsetX, gridOffsetY, COLS * cellSize, ROWS * cellSize);

    // Locked blocks
    for (let r = 0; r < ROWS; r++) {
        for (let c = 0; c < COLS; c++) {
            if (grid[r][c] !== null) {
                drawCell(c, r, COLORS[grid[r][c]], 1);
            }
        }
    }

    // Current piece
    const shape = getShape(currentPiece, curRot);
    for (const [cx, cy] of shape.cells) {
        const gx = curX + cx, gy = curY + cy;
        if (gy >= 0) drawCell(gx, gy, COLORS[currentPiece], 0.9);
    }

    // Player hitbox indicator
    const pp = getPlayerPos();
    ctx.beginPath();
    ctx.arc(pp.x, pp.y, 8, 0, Math.PI * 2);
    ctx.strokeStyle = 'rgba(255,255,255,0.8)';
    ctx.lineWidth = 2;
    ctx.stroke();
    ctx.beginPath();
    ctx.arc(pp.x, pp.y, 3, 0, Math.PI * 2);
    ctx.fillStyle = '#fff';
    ctx.fill();

    // Bullets
    for (const b of bullets) {
        ctx.beginPath();
        ctx.arc(b.x, b.y, b.r, 0, Math.PI * 2);
        ctx.fillStyle = b.color;
        ctx.shadowColor = b.color;
        ctx.shadowBlur = 8;
        ctx.fill();
        ctx.shadowBlur = 0;
    }

    // Enemies
    for (const e of enemies) {
        ctx.save();
        ctx.translate(e.x, e.y);
        ctx.fillStyle = '#f0f';
        ctx.shadowColor = '#f0f';
        ctx.shadowBlur = 10;
        // Draw as diamond
        ctx.beginPath();
        ctx.moveTo(0, -12);
        ctx.lineTo(12, 0);
        ctx.lineTo(0, 12);
        ctx.lineTo(-12, 0);
        ctx.closePath();
        ctx.fill();
        ctx.shadowBlur = 0;
        // HP bar
        ctx.fillStyle = '#44f';
        ctx.fillRect(-15, -20, 30 * (e.hp / 3), 3);
        ctx.restore();
    }

    // Particles
    for (const p of particles) {
        const alpha = p.life / 60;
        ctx.globalAlpha = Math.min(1, alpha);
        ctx.beginPath();
        ctx.arc(p.x, p.y, p.r * (p.life / 40), 0, Math.PI * 2);
        ctx.fillStyle = p.color;
        ctx.fill();
    }
    ctx.globalAlpha = 1;

    // Float texts
    for (const ft of floatTexts) {
        ctx.textAlign = 'center';
        ctx.font = `bold ${Math.min(W*0.025, 20)}px monospace`;
        ctx.fillStyle = ft.color;
        ctx.globalAlpha = Math.min(1, ft.life / 30);
        ctx.fillText(ft.text, ft.x, ft.y);
    }
    ctx.globalAlpha = 1;

    // HUD
    ctx.textAlign = 'left';
    ctx.font = `bold ${Math.min(W*0.025, 18)}px monospace`;
    ctx.fillStyle = '#fff';
    ctx.fillText(`SCORE: ${score}`, gridOffsetX, gridOffsetY - 10);
    ctx.textAlign = 'right';
    ctx.fillText(`LEVEL: ${level}`, gridOffsetX + COLS * cellSize, gridOffsetY - 10);

    // Lives
    for (let i = 0; i < lives; i++) {
        ctx.beginPath();
        ctx.arc(gridOffsetX + 15 + i * 25, H - 20, 8, 0, Math.PI * 2);
        ctx.fillStyle = '#f44';
        ctx.fill();
    }

    // Game over overlay
    if (gameState === 'gameover') {
        ctx.fillStyle = 'rgba(0,0,0,0.7)';
        ctx.fillRect(0, 0, W, H);
        ctx.textAlign = 'center';
        ctx.fillStyle = '#f44';
        ctx.font = `bold ${Math.min(W*0.06, 50)}px monospace`;
        ctx.fillText('GAME OVER', W/2, H * 0.35);
        ctx.fillStyle = '#fff';
        ctx.font = `${Math.min(W*0.03, 22)}px monospace`;
        ctx.fillText(`Final Score: ${score}`, W/2, H * 0.45);
        ctx.fillText(`Level Reached: ${level}`, W/2, H * 0.52);
        ctx.fillStyle = `hsl(${(Date.now()/30)%360},100%,70%)`;
        ctx.font = `bold ${Math.min(W*0.035, 24)}px monospace`;
        ctx.fillText('PRESS SPACE TO RESTART', W/2, H * 0.68);
    }

    ctx.restore();
}

function drawCell(col, row, color, alpha) {
    const x = gridOffsetX + col * cellSize;
    const y = gridOffsetY + row * cellSize;
    const pad = 2;
    ctx.globalAlpha = alpha;
    ctx.fillStyle = color;
    ctx.shadowColor = color;
    ctx.shadowBlur = 6;
    ctx.fillRect(x + pad, y + pad, cellSize - pad * 2, cellSize - pad * 2);
