function parseLevel(levelStr) {
    const rows = levelStr.split('\n');
    // Remove leading spaces for uniform width
    let maxW = 0;
    rows.forEach(r => { const t = r.trimEnd(); if(t.length > maxW) maxW = t.length; });

    let pX = 0, pY = 0;
    let boxes_ = [], targets_ = [], walls_ = [];
    let gridH_ = rows.length;
    let gridW_ = 0;

    rows.forEach((row, y) => {
        const r = row.padEnd(maxW, ' ');
        for (let x = 0; x < maxW; x++) {
            const c = r[x];
            if (c === '#') walls_.push({x, y});
            if (c === '@' || c === '+') { pX = x; pY = y; }
            if (c === '$' || c === '*') boxes_.push({x, y});
            if (c === '.' || c === '+' || c === '*') targets_.push({x, y});
        }
    });

    return { playerX: pX, playerY: pY, boxes: boxes_, targets: targets_, walls: walls_, gridW: maxW, gridH: gridH_ };
}

function loadLevel(idx) {
    const parsed = parseLevel(LEVELS[idx]);
    currentLevel = idx;
    playerX = parsed.playerX;
    playerY = parsed.playerY;
    boxes = parsed.boxes.map(b => ({x: b.x, y: b.y}));
    targets = parsed.targets;
    walls = parsed.walls;
    gridW = parsed.gridW;
    gridH = parsed.gridH;
    moves = 0;
    particles = [];
    gameState = 'playing';
}

function isWall(x, y) {
    return walls.some(w => w.x === x && w.y === y);
}

function isBox(x, y) {
    return boxes.findIndex(b => b.x === x && b.y === y);
}

function isTarget(x, y) {
    return targets.some(t => t.x === x && t.y === y);
}

function canMove(x, y) {
    if (isWall(x, y)) return false;
    const bi = isBox(x, y);
    if (bi !== -1) {
        // Check if box can be pushed
        const b = boxes[bi];
        const dx = x - playerX;
        const dy = y - playerY;
        const nx = b.x + dx;
        const ny = b.y + dy;
        if (isWall(nx, ny)) return false;
        if (isBox(nx, ny) !== -1) return false;
    }
    return true;
}

function move(dx, dy) {
    if (gameState !== 'playing') return;
    const nx = playerX + dx;
    const ny = playerY + dy;

    if (isWall(nx, ny)) return;

    const bi = isBox(nx, ny);
    if (bi !== -1) {
        const b = boxes[bi];
        const bx = b.x + dx;
        const by = b.y + dy;
        if (isWall(bx, by) || isBox(bx, by) !== -1) return;
        b.x = bx;
        b.y = by;
        playerX = nx;
        playerY = ny;
        moves++;
    } else {
        playerX = nx;
        playerY = ny;
        moves++;
    }

    // Check win
    if (boxes.every(b => isTarget(b.x, b.y))) {
        spawnWinParticles();
        gameState = 'levelComplete';
        levelTransition = 0;
    }
}

function spawnWinParticles() {
    for (let i = 0; i < 30; i++) {
        particles.push({
            x: playerX * cellSize + cellSize/2,
            y: playerY * cellSize + cellSize/2,
            vx: (Math.random()-0.5)*8,
            vy: (Math.random()-0.5)*8 - 3,
            life: 1,
            color: `hsl(${Math.random()*60+40}, 90%, 60%)`
        });
    }
}

let audioCtx = null;
function initAudio() {
    if (!audioCtx) {
        audioCtx = new (window.AudioContext || window.webkitAudioContext)();

function playMoveSound() {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = 'sine';
    osc.frequency.value = 200 + Math.random()*50;
    gain.gain.setValueAtTime(0.1, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.1);
    osc.connect(gain).connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + 0.1);
}

function playBoxSound() {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = 'square';
    osc.frequency.value = 120;
    gain.gain.setValueAtTime(0.08, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.15);
    osc.connect(gain).connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + 0.15);
}

function playWinSound() {
    if (!audioCtx) return;
    const notes = [262, 330, 392, 523];
    notes.forEach((f, i) => {
        const osc = audioCtx.createOscillator();
        const gain = audioCtx.createGain();
        osc.type = 'sine';
        osc.frequency.value = f;
        gain.gain.setValueAtTime(0.12, audioCtx.currentTime + i*0.1);
        gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + i*0.1 + 0.3);
        osc.connect(gain).connect(audioCtx.destination);
        osc.start(audioCtx.currentTime + i*0.1);
        osc.stop(audioCtx.currentTime + i*0.1 + 0.3);
    });
}

    let dx = 0, dy = 0;

        case 'ArrowRight': case 'd': case 'D': dx = 1; break;
        case 'r': case 'R': loadLevel(currentLevel); return;

    const oldBoxPos = boxes.map(b => ({x:b.x, y:b.y}));
    move(dx, dy);
    const newBoxPos = boxes.map(b => ({x:b.x, y:b.y}));
    if (JSON.stringify(oldBoxPos) !== JSON.stringify(newBoxPos)) {
        playBoxSound();
    } else if (gameState === 'playing') {
        playMoveSound();
    }
});

function drawWall(x, y) {
    const px = x * cellSize;
    const py = y * cellSize;
    // Check neighbors for 3D effect
    const topOpen = !isWall(x, y-1);
    const bottomOpen = !isWall(x, y+1);
    const leftOpen = !isWall(x-1, y);
    const rightOpen = !isWall(x+1, y);

    ctx.fillStyle = '#5c4033';
    ctx.fillRect(px, py, cellSize, cellSize);

    // Top face
    if (topOpen) {
        ctx.fillStyle = '#6b4c3b';
        ctx.fillRect(px, py, cellSize, 4);
    }
    // Left face
    if (leftOpen) {
        ctx.fillStyle = '#4a3528';
        ctx.fillRect(px, py, 4, cellSize);
    }
    // Right face
    if (rightOpen) {
        ctx.fillStyle = '#4a3528';
        ctx.fillRect(px + cellSize - 4, py, 4, cellSize);
    }
    // Bottom face
    if (bottomOpen) {
        ctx.fillStyle = '#6b4c3b';
        ctx.fillRect(px, py + cellSize - 4, cellSize, 4);
    }

    // Cross pattern
    ctx.strokeStyle = 'rgba(0,0,0,0.2)';
    ctx.lineWidth = 1;
    ctx.beginPath();
    ctx.moveTo(px+6, py+6);
    ctx.lineTo(px+cellSize-6, py+cellSize-6);
    ctx.moveTo(px+cellSize-6, py+6);
    ctx.lineTo(px+6, py+cellSize-6);
    ctx.stroke();
}

function drawTarget(x, y) {
    const px = x * cellSize + cellSize/2;
    const py = y * cellSize + cellSize/2;
    const r = cellSize * 0.25;
    const pulse = Math.sin(animTimer * 3) * 0.15 + 0.85;

    ctx.beginPath();

    ctx.fillStyle = `rgba(255, 200, 50, ${0.4 * pulse})`;
    ctx.fill();
    ctx.strokeStyle = `rgba(255, 200, 50, ${0.7 * pulse})`;
    ctx.lineWidth = 2;
    ctx.stroke();

    // Inner dot
    ctx.beginPath();
    ctx.arc(px, py, r * 0.3, 0, Math.PI*2);
    ctx.fillStyle = `rgba(255, 220, 80, ${pulse})`;
    ctx.fill();
}

function drawBox(x, y) {
    const px = x * cellSize;
    const py = y * cellSize;
    const onTarget = isTarget(x, y);

    // Box body

        ctx.fillStyle = '#cd853f';

    ctx.fillRect(px+3, py+3, cellSize-6, cellSize-6);

    // Box edges (top-down crate look)
    ctx.strokeStyle = onTarget ? '#1a5c32' : '#8b5e2a';
    ctx.lineWidth = 2;
    ctx.strokeRect(px+3, py+3, cellSize-6, cellSize-6);

    // Cross planks

    ctx.lineTo(px+cellSize-8, py+cellSize-8);

    ctx.lineTo(px+8, py+cellSize-8);

        ctx.fillRect(px+3, py+3, cellSize-6, cellSize-6);
    }
}

        ctx.fillText('You completed all Sokoban levels!', W/2, H/2 + 20);
        ctx.fillStyle = '#fff';
        const blink = Math.sin(animTimer * 4) > 0;
        if (blink) {
            ctx.fillText('Press ENTER to return to menu', W/2, H/2 + 80);
        }
    }
}

function updateParticles() {
    particles.forEach(p => {
        p.x += p.vx * 0.5;
        p.y += p.vy * 0.5;
        p.vy += 0.3;
        p.life -= 0.015;
    });
    particles = particles.filter(p => p.life > 0);
}

let lastTime = 0;
function gameLoop(time) {
    const dt = (time - lastTime) / 1000;
    lastTime = time;
    animTimer += dt;

    resize();

    if (gameState === 'menu') {
        drawMenu();
    } else {
        updateParticles();
        drawGame();
    }

    requestAnimationFrame(gameLoop);
}

// Handle Enter for transitions in playing state
window.addEventListener('keydown', (e) => {
    if (gameState === 'levelComplete' && levelTransition > 0.5) {
        if (currentLevel < LEVELS.length - 1) {
            loadLevel(currentLevel + 1);
        } else {
            gameState = 'gameComplete';
        }
        return;
    }
});

// Initialize
ctx.fillStyle = '#1a1a2e';
ctx.fillRect(0, 0, W, H);
requestAnimationFrame(gameLoop);

})();
