function playSound(type) {
  if (!audioCtx) return;
  const osc = audioCtx.createOscillator();
  const gain = audioCtx.createGain();
  osc.connect(gain);
  gain.connect(audioCtx.destination);
  
  if (type === 'step') {
    osc.type = 'sine';
    osc.frequency.setValueAtTime(220, audioCtx.currentTime);
    osc.frequency.exponentialRampToValueAtTime(330, audioCtx.currentTime + 0.05);
    gain.gain.setValueAtTime(0.1, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.1);
    osc.start();
    osc.stop(audioCtx.currentTime + 0.1);
  } else if (type === 'explode') {
    osc.type = 'sawtooth';
    osc.frequency.setValueAtTime(80, audioCtx.currentTime);
    osc.frequency.exponentialRampToValueAtTime(30, audioCtx.currentTime + 0.5);
    gain.gain.setValueAtTime(0.4, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.6);
    osc.start();
    osc.stop(audioCtx.currentTime + 0.6);
  } else if (type === 'flag') {
    osc.type = 'triangle';
    osc.frequency.setValueAtTime(440, audioCtx.currentTime);
    gain.gain.setValueAtTime(0.15, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.15);
    osc.start();

  } else if (type === 'win') {
    const notes = [262, 330, 392, 523];

      const o = audioCtx.createOscillator();
      const g = audioCtx.createGain();
      o.connect(g); g.connect(audioCtx.destination);
      o.type = 'sine';
      o.frequency.value = f;
      g.gain.setValueAtTime(0.15, audioCtx.currentTime + i * 0.12);
      g.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + i * 0.12 + 0.3);
      o.start(audioCtx.currentTime + i * 0.12);
      o.stop(audioCtx.currentTime + i * 0.12 + 0.3);
    });
  }
}

function generateLevel(lvl) {
  const cols = LEVEL_WIDTHS[lvl];
  const density = MINE_DENSITIES[lvl];
  grid = [];
  
  for (let y = 0; y < ROWS; y++) {
    grid[y] = [];
    for (let x = 0; x < cols; x++) {
      grid[y][x] = {
        mine: Math.random() < density,
        revealed: false,
        flagged: false,
        number: 0
      };
    }
  }
  
  // Calculate numbers
  for (let y = 0; y < ROWS; y++) {
    for (let x = 0; x < cols; x++) {
      if (!grid[y][x].mine) {
        let n = 0;
        for (let dy = -1; dy <= 1; dy++) {
          for (let dx = -1; dx <= 1; dx++) {
            if (dx === 0 && dy === 0) continue;
            const ny = y + dy, nx = x + dx;
            if (ny >= 0 && ny < ROWS && nx >= 0 && nx < cols && grid[ny][nx].mine) n++;
          }
        }
        grid[y][x].number = n;
      }
    }
  }
  
  // Ensure start area is safe (first 2 columns, middle rows)
  for (let y = 0; y < ROWS; y++) {
    for (let x = 0; x < 3; x++) {
      grid[y][x].mine = false;
    }
  }
  
  // Recalculate numbers after clearing start area
  for (let y = 0; y < ROWS; y++) {
    for (let x = 0; x < cols; x++) {
      if (!grid[y][x].mine) {
        let n = 0;

function getCellSize() {
  return Math.min(Math.floor((H - 100) / ROWS), 64);
}

function addParticles(x, y, color, count) {
  for (let i = 0; i < count; i++) {
    const angle = Math.random() * Math.PI * 2;
    const speed = 1 + Math.random() * 4;
    particles.push({
      x, y,
      vx: Math.cos(angle) * speed,
      vy: Math.sin(angle) * speed - 2,
      life: 1,
      color,
      size: 2 + Math.random() * 4
    });
  }
}

function updateParticles(dt) {
  for (let i = particles.length - 1; i >= 0; i--) {
    const p = particles[i];
    p.x += p.vx;
    p.y += p.vy;
    p.vy += 0.3;
    p.life -= dt * 2;
    if (p.life <= 0) particles.splice(i, 1);
  }
}

function drawParticles() {
  for (const p of particles) {
    ctx.globalAlpha = p.life;
    ctx.fillStyle = p.color;
    ctx.fillRect(p.x - p.size/2, p.y - p.size/2, p.size * p.life, p.size * p.life);
  }
  ctx.globalAlpha = 1;
}

function getNumberColor(n) {
  const colors = ['#000', '#4fc3f7', '#69f0ae', '#ff5252', '#ffd740', '#e040fb', '#ff9800', '#fff'];
  return colors[n] || '#fff';
}

function draw() {
  const cellSize = getCellSize();
  
  // Screen shake
  let sx = 0, sy = 0;
  if (shakeTime > 0) {
    sx = (Math.random() - 0.5) * shakeIntensity;
    sy = (Math.random() - 0.5) * shakeIntensity;
  }
  
  ctx.save();
  ctx.translate(sx, sy);
  
  // Background
  const grad = ctx.createLinearGradient(0, 0, 0, H);
  grad.addColorStop(0, '#0d1b2a');
  grad.addColorStop(1, '#1b2838');
  ctx.fillStyle = grad;
  ctx.fillRect(-10, -10, W + 20, H + 20);
  
  // Stars in background

  }
  
  if (gameState === 'menu') {
    drawMenu();
  } else {
    drawGame(cellSize);
  }
  
  ctx.restore();
}

let lastTime = 0;
function gameLoop(time) {
  const dt = Math.min((time - lastTime) / 1000, 0.05);
  lastTime = time;
  
  if (gameState === 'playing') {
    levelTimer += dt * 1000;
    
    // Camera follow
    const cellSize = getCellSize();
    targetCameraX = playerX * cellSize - W / 2 + cellSize / 2;
    targetCameraX = Math.max(0, targetCameraX);
    const maxCam = LEVEL_WIDTHS[level] * cellSize - W;
    targetCameraX = Math.min(targetCameraX, Math.max(0, maxCam));
    cameraX += (targetCameraX - cameraX) * 5 * dt;
    
    // Shake decay
    if (shakeTime > 0) {
      shakeTime -= dt;
      shakeIntensity *= 0.9;
    }
    
    updateParticles(dt);
  }
  
  draw();
  requestAnimationFrame(gameLoop);
}

function startGame() {
  level = 0;
  generateLevel(0);
  gameState = 'playing';
  levelTimer = 0;
  particles = [];
}

function nextLevel() {
  level++;
  if (level >= LEVEL_WIDTHS.length) {
    gameState = 'gameComplete';
    return;
  }
  generateLevel(level);
  gameState = 'playing';
  levelTimer = 0;
}

function checkLevelComplete() {
  const safeCells = grid.flat().filter(c => !c.mine).length;
  const visitedSafe = grid.flat().filter(c => c.revealed && !c.mine).length;
  
  if (visitedSafe >= safeCells) {
    gameState = 'levelComplete';
    playSound('win');
    if (level + 1 > bestLevel) {
      bestLevel = level + 1;
      try { localStorage.setItem('minescroll_best', bestLevel); } catch(e) {}
    }
  }
}

function movePlayer(dx, dy) {
  const nx = playerX + dx;
  const ny = playerY + dy;
  const cols = LEVEL_WIDTHS[level];
  
  if (nx < 0 || nx >= cols || ny < 0 || ny >= ROWS) return;
  
  const cell = grid[ny][nx];
  
  if (!cell.revealed && !cell.mine) {
    cell.revealed = true;
    playSound('step');
    playerX = nx;
    playerY = ny;
    checkLevelComplete();
  } else if (cell.revealed && !cell.mine) {
    // Already revealed safe cell - can move onto it
    playerX = nx;
    playerY = ny;
    playSound('step');
  } else if (!cell.revealed && cell.mine) {
    // HIT A MINE!
    cell.revealed = true;
    gameState = 'dead';
    shakeTime = 0.5;
    shakeIntensity = 15;
    playSound('explode');
    
    const cellSize = getCellSize();
    const topOffset = (H - ROWS * cellSize) / 2;
    const ex = nx * cellSize - cameraX + cellSize/2;
    const ey = topOffset + ny * cellSize + cellSize/2;
    addParticles(ex, ey, '#ff5252', 30);
    addParticles(ex, ey, '#ffd740', 20);
    addParticles(ex, ey, '#fff', 15);
  }
}

function toggleFlag() {
  const cell = grid[playerY][playerX];
  if (!cell.mine) {
    cell.flagged = !cell.flagged;
    playSound('flag');
  }
}

document.addEventListener('keydown', (e) => {
  initAudio();
  
  if (gameState === 'menu') {
    if (e.key === 'Enter' || e.key === ' ') {
      startGame();
    }
    return;
  }
  
  if (gameState === 'dead' || gameState === 'levelComplete' || gameState === 'gameComplete') {
    if (e.key === 'Enter' || e.key === ' ') {
      if (gameState === 'dead') startGame();
      else nextLevel();
    }
    return;
  }
  
  if (gameState !== 'playing') return;
  
  switch(e.key) {
    case 'ArrowUp': case 'w': case 'W': movePlayer(0, -1); break;
    case 'ArrowDown': case 's': case 'S': movePlayer(0, 1); break;
    case 'ArrowLeft': case 'a': case 'A': movePlayer(-1, 0); break;
    case 'ArrowRight': case 'd': case 'D': movePlayer(1, 0); break;
    case ' ': e.preventDefault(); toggleFlag(); break;
  }
});

requestAnimationFrame(gameLoop);
