const canvas = document.getElementById('game');
const ctx = canvas.getContext('2d');

let W, H;
function resize() {

// Audio
let audioCtx = null;
function initAudio() {
    if (!audioCtx) {
        audioCtx = new (window.AudioContext || window.webkitAudioContext)();
    }
}
function playSound(type) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    const now = audioCtx.currentTime;
    switch(type) {
        case 'hit':
            osc.type = 'square';
            osc.frequency.setValueAtTime(200, now);
            osc.frequency.exponentialRampToValueAtTime(80, now + 0.1);
            gain.gain.setValueAtTime(0.15, now);
            gain.gain.exponentialRampToValueAtTime(0.001, now + 0.1);
            osc.start(now); osc.stop(now + 0.1);
            break;
        case 'kill':
            osc.type = 'square';
            osc.frequency.setValueAtTime(400, now);
            osc.frequency.exponentialRampToValueAtTime(100, now + 0.15);
            gain.gain.setValueAtTime(0.2, now);
            gain.gain.exponentialRampToValueAtTime(0.001, now + 0.15);
            osc.start(now); osc.stop(now + 0.15);
            break;
        case 'towerHit':
            osc.type = 'sawtooth';
            osc.frequency.setValueAtTime(80, now);
            osc.frequency.exponentialRampToValueAtTime(40, now + 0.3);
            gain.gain.setValueAtTime(0.25, now);
            gain.gain.exponentialRampToValueAtTime(0.001, now + 0.3);
            osc.start(now); osc.stop(now + 0.3);
            break;
        case 'wave':
            osc.type = 'sine';
            osc.frequency.setValueAtTime(300, now);
            osc.frequency.exponentialRampToValueAtTime(600, now + 0.2);
            gain.gain.setValueAtTime(0.15, now);
            gain.gain.exponentialRampToValueAtTime(0.001, now + 0.25);
            osc.start(now); osc.stop(now + 0.25);
            break;
        case 'bomb':
            osc.type = 'sawtooth';
            osc.frequency.setValueAtTime(60, now);
            osc.frequency.exponentialRampToValueAtTime(20, now + 0.4);
            gain.gain.setValueAtTime(0.3, now);
            gain.gain.exponentialRampToValueAtTime(0.001, now + 0.4);
            osc.start(now); osc.stop(now + 0.4);
            break;
        case 'gameover':
            osc.type = 'sawtooth';
            osc.frequency.setValueAtTime(200, now);
            osc.frequency.exponentialRampToValueAtTime(50, now + 0.8);
            gain.gain.setValueAtTime(0.2, now);
            gain.gain.exponentialRampToValueAtTime(0.001, now + 0.8);
            osc.start(now); osc.stop(now + 0.8);
            break;
    }
}

// Game State
let state = 'MENU'; // MENU, PLAYING, WAVE_BREAK, GAME_OVER
let score = 0;
let combo = 0;
let maxCombo = 0;
let wave = 0;
let bombs = 3;
let shakeTime = 0;
let shakeIntensity = 0;
let particles = [];
let enemies = [];
let towers = [];
let spawnQueue = [];
let spawnTimer = 0;
let waveBreakTimer = 0;
let time = 0;
let lastTime = 0;
let clickPos = { x: 0, y: 0 };
let ripples = [];

const ENEMY_TYPES = [
    { name: 'grunt', hp: 1, speed: 60, size: 14, color: '#ff4444', score: 10 },
    { name: 'runner', hp: 1, speed: 120, size: 10, color: '#ffaa00', score: 25 },
    { name: 'tank', hp: 5, speed: 40, size: 22, color: '#4488ff', score: 50 },
    { name: 'boss', hp: 20, speed: 30, size: 32, color: '#aa44ff', score: 200 },
];

function initTowers() {
    towers = [];
    const towerY = H * 0.5;
    for (let i = 0; i < 3; i++) {
        towers.push({
            x: W * 0.12 + i * W * 0.08,
            y: towerY - 40,
            w: 40,
            h: 80,
            maxHp: 100,
            hp: 100,
            color: ['#ff6644', '#44aaff', '#44ff88'][i],
            hitFlash: 0,
            alive: true
        });
    }
}

function spawnWave() {
    wave++;
    spawnQueue = [];
    const count = 3 + Math.floor(wave * 1.5);
    for (let i = 0; i < count; i++) {
        let typeIdx = 0;
        const r = Math.random();
        if (wave >= 2 && r > 0.7) typeIdx = 1;
        if (wave >= 3 && r > 0.85) typeIdx = 2;
        if (wave >= 5 && i === count - 1 && wave % 5 === 0) typeIdx = 3;
        const t = ENEMY_TYPES[typeIdx];
        spawnQueue.push({
            type: typeIdx,
            delay: i * (800 / Math.max(1, wave * 0.3)),
            hp: t.hp + Math.floor(wave * 0.5),
            maxHp: t.hp + Math.floor(wave * 0.5),
            speed: t.speed + wave * 2,
            size: t.size,
            color: t.color,
            scoreVal: t.score + wave * 2
        });
    }
    spawnTimer = 0;
    playSound('wave');
}

function resetGame() {
    score = 0;
    combo = 0;
    maxCombo = 0;
    wave = 0;
    bombs = 3;
    enemies = [];
    particles = [];
    ripples = [];
    spawnQueue = [];
    shakeTime = 0;
    initTowers();
    state = 'PLAYING';
    spawnWave();
}

function spawnEnemy(data) {
    const laneY = H * 0.3 + Math.random() * H * 0.4;
    enemies.push({
        x: W + data.size,
        y: laneY,
        hp: data.hp,
        maxHp: data.maxHp,
        speed: data.speed,
        size: data.size,
        color: data.color,
        scoreVal: data.scoreVal,
        hitFlash: 0,
        wobble: Math.random() * Math.PI * 2
    });
}

function addParticles(x, y, color, count) {
    for (let i = 0; i < count; i++) {
        const angle = Math.random() * Math.PI * 2;
        const speed = 50 + Math.random() * 150;
        particles.push({
            x, y,
            vx: Math.cos(angle) * speed,
            vy: Math.sin(angle) * speed,
            life: 0.4 + Math.random() * 0.3,
            maxLife: 0.4 + Math.random() * 0.3,
            size: 2 + Math.random() * 4,
            color
        });
    }
}

function addRipple(x, y) {
    ripples.push({ x, y, radius: 5, maxRadius: 30, life: 0.3, maxLife: 0.3 });
}

function update(dt) {
    time += dt;
    
    if (state === 'PLAYING') {
        // Spawn enemies
        if (spawnQueue.length > 0) {
            spawnTimer += dt * 1000;
            const next = spawnQueue[0];
            if (spawnTimer >= next.delay) {
                spawnEnemy(next);
                spawnQueue.shift();
                spawnTimer = 0;
            }
        }

        // Update enemies
        for (let i = enemies.length - 1; i >= 0; i--) {
            const e = enemies[i];
            e.x -= e.speed * dt;
            e.wobble += dt * 5;
            if (e.hitFlash > 0) e.hitFlash -= dt * 8;

            // Check if enemy reached tower line
            const towerX = W * 0.12 + 60;
            if (e.x <= towerX) {
                // Hit the nearest alive tower
                let hitTower = null;
                for (const t of towers) {
                    if (t.alive && e.y > t.y - 30 && e.y < t.y + t.h + 30) {
                        hitTower = t;
                        break;
                    }
                }
                if (!hitTower) {
                    for (const t of towers) {
                        if (t.alive) { hitTower = t; break; }
                    }
                }
                if (hitTower) {
                    hitTower.hp -= 10;
                    hitTower.hitFlash = 1;
                    shakeTime = 0.2;
                    shakeIntensity = 5;
                    playSound('towerHit');
                    addParticles(hitTower.x + hitTower.w/2, e.y, '#ff0', 8);
                }
                enemies.splice(i, 1);
            }
        }

        // Update towers
        for (const t of towers) {
            if (t.hitFlash > 0) t.hitFlash -= dt * 5;
            if (t.hp <= 0 && t.alive) {
                t.alive = false;
                addParticles(t.x + t.w/2, t.y + t.h/2, t.color, 30);
            }
        }

        // Check game over
        if (!towers.some(t => t.alive)) {
            state = 'GAME_OVER';
            playSound('gameover');
        }

        // Check wave complete
        if (spawnQueue.length === 0 && enemies.length === 0) {
            state = 'WAVE_BREAK';
            waveBreakTimer = 2;
        }
    } else if (state === 'WAVE_BREAK') {
        waveBreakTimer -= dt;
        if (waveBreakTimer <= 0) {
            state = 'PLAYING';
            spawnWave();
        }
    }

    // Update particles
    for (let i = particles.length - 1; i >= 0; i--) {
        const p = particles[i];
        p.x += p.vx * dt;
        p.y += p.vy * dt;
        p.life -= dt;
        if (p.life <= 0) particles.splice(i, 1);
    }

    // Update ripples
    for (let i = ripples.length - 1; i >= 0; i--) {
        const r = ripples[i];
        r.radius += (r.maxRadius - r.radius) * dt * 8;
        r.life -= dt;
        if (r.life <= 0) ripples.splice(i, 1);
    }

    // Shake
    if (shakeTime > 0) shakeTime -= dt;
}

function draw() {
    ctx.save();
    
    // Screen shake
    if (shakeTime > 0) {
        const sx = (Math.random() - 0.5) * shakeIntensity;
        const sy = (Math.random() - 0.5) * shakeIntensity;
        ctx.translate(sx, sy);
    }

    // Background
    const grad = ctx.createLinearGradient(0, 0, 0, H);
    grad.addColorStop(0, '#0a0a2a');
    grad.addColorStop(1, '#1a1a3a');
    ctx.fillStyle = grad;
    ctx.fillRect(-10, -10, W + 20, H + 20);

    // Ground line
    ctx.strokeStyle = '#2a2a4a';
    ctx.lineWidth = 2;
    for (let y = H * 0.3; y < H * 0.75; y += 40) {
        ctx.beginPath();
        ctx.moveTo(0, y);
        ctx.lineTo(W, y);
        ctx.globalAlpha = 0.15;
        ctx.stroke();
    }
    ctx.globalAlpha = 1;

    // Draw towers
    for (const t of towers) {
        if (!t.alive) {
            // Rubble
            ctx.fillStyle = '#333';
            ctx.fillRect(t.x + 5, t.y + t.h - 15, t.w - 10, 15);
            continue;
        }
        
        const flash = t.hitFlash > 0 ? Math.min(1, t.hitFlash) : 0;
        ctx.fillStyle = flash > 0.5 ? '#fff' : t.color;
        ctx.globalAlpha = 0.9;
        
        // Tower body
        ctx.fillRect(t.x, t.y + 20, t.w, t.h - 20);
        // Tower top (battlements)
        for (let bx = 0; bx < t.w; bx += 12) {
            ctx.fillRect(t.x + bx, t.y + 10, 8, 15);
        }
        // Tower base
        ctx.fillStyle = '#333';
        ctx.fillRect(t.x - 4, t.y + t.h - 10, t.w + 8, 10);
        
        ctx.globalAlpha = 1;

        // HP bar
        const hpW = t.w;
        const hpX = t.x;
        const hpY = t.y - 10;
        ctx.fillStyle = '#333';
        ctx.fillRect(hpX, hpY, hpW, 6);
        ctx.fillStyle = t.hp > 50 ? '#4f4' : t.hp > 25 ? '#ff4' : '#f44';
        ctx.fillRect(hpX, hpY, hpW * (t.hp / t.maxHp), 6);
    }

    // Draw enemies
    for (const e of enemies) {
        const wobbleY = Math.sin(e.wobble) * 3;
        const flash = e.hitFlash > 0 ? Math.min(1, e.hitFlash) : 0;
        
        ctx.fillStyle = flash > 0.5 ? '#fff' : e.color;
        ctx.globalAlpha = 0.9;
        
        if (e.size > 25) {
            // Boss - draw as diamond
            ctx.beginPath();
            ctx.moveTo(e.x, e.y + wobbleY - e.size);
            ctx.lineTo(e.x + e.size, e.y + wobbleY);
            ctx.lineTo(e.x, e.y + wobbleY + e.size);
            ctx.lineTo(e.x - e.size, e.y + wobbleY);
            ctx.closePath();
            ctx.fill();
        } else if (e.size > 18) {
            // Tank - draw as square
            ctx.fillRect(e.x - e.size/2, e.y + wobbleY - e.size/2, e.size, e.size);
        } else {
            // Normal - draw as circle
            ctx.beginPath();
            ctx.arc(e.x, e.y + wobbleY, e.size, 0, Math.PI * 2);
            ctx.fill();
        }
        
        ctx.globalAlpha = 1;

        // HP bar for multi-hit enemies
        if (e.maxHp > 1) {
            const bw = e.size * 1.5;
            ctx.fillStyle = '#333';
            ctx.fillRect(e.x - bw/2, e.y + wobbleY - e.size - 8, bw, 4);
            ctx.fillStyle = '#ff6';
            ctx.fillRect(e.x - bw/2, e.y + wobbleY - e.size - 8, bw * (e.hp / e.maxHp), 4);
        }
    }

    // Draw particles
    for (const p of particles) {
        const alpha = p.life / p.maxLife;
        ctx.globalAlpha = alpha;
        ctx.fillStyle = p.color;
        ctx.fillRect(p.x - p.size/2, p.y - p.size/2, p.size, p.size);
    }
    ctx.globalAlpha = 1;

    // Draw ripples
    for (const r of ripples) {
        const alpha = r.life / r.maxLife;
        ctx.strokeStyle = `rgba(255, 200, 100, ${alpha})`;
        ctx.lineWidth = 2;
        ctx.beginPath();
        ctx.arc(r.x, r.y, r.radius, 0, Math.PI * 2);
        ctx.stroke();
    }

    // UI
    if (state === 'PLAYING' || state === 'WAVE_BREAK') {
        // Score
        ctx.fillStyle = '#fff';
        ctx.font = 'bold 24px monospace';
        ctx.textAlign = 'left';
        ctx.fillText(`SCORE: ${score}`, 20, 35);
        
        // Combo
        if (combo > 1) {
            ctx.fillStyle = combo > 10 ? '#ff0' : '#fa0';
            ctx.font = 'bold 18px monospace';
            ctx.fillText(`x${combo} COMBO`, 20, 60);
        }

        // Wave
        ctx.textAlign = 'right';
        ctx.fillStyle = '#aaa';
        ctx.font = '16px monospace';
        ctx.fillText(`WAVE ${wave}`, W - 20, 35);

        // Bombs
        ctx.textAlign = 'left';
        ctx.fillStyle = bombs > 0 ? '#ff0' : '#666';
        ctx.font = '16px monospace';
        let bombStr = '';
        for (let i = 0; i < bombs; i++) bombStr += '💣 ';
        ctx.fillText(`BOMBS [B]: ${bombStr}`, W - 200, 60);

        if (state === 'WAVE_BREAK') {
            ctx.textAlign = 'center';
            ctx.fillStyle = '#fff';
            ctx.font = 'bold 28px monospace';
            ctx.fillText(`WAVE ${wave} CLEARED!`, W/2, H/2 - 20);
            ctx.font = '16px monospace';
            ctx.fillStyle = '#aaa';
            ctx.fillText(`Next wave in ${Math.ceil(waveBreakTimer)}...`, W/2, H/2 + 15);
        }
    }

    // Menu / Game Over screens
    if (state === 'MENU') {
        drawOverlay();
        ctx.fillStyle = '#fff';
        ctx.textAlign = 'center';
        ctx.font = 'bold 48px monospace';
        ctx.fillText('TOWER DEFENSE', W/2, H/2 - 60);
        ctx.font = 'bold 36px monospace';
        ctx.fillStyle = '#ff6644';
        ctx.fillText('CLICKER', W/2, H/2 - 20);

        ctx.font = '16px monospace';
        ctx.fillStyle = '#aaa';
        ctx.fillText('Click enemies to damage them!', W/2, H/2 + 30);
        ctx.fillText('Defend your towers from incoming waves.', W/2, H/2 + 55);
        ctx.fillText('[B] - Use bomb (damages all enemies)', W/2, H/2 + 80);
        ctx.fillText('', W/2, H/2 + 110);
        ctx.fillStyle = '#fff';
        ctx.font = 'bold 20px monospace';
        ctx.fillText('CLICK TO START', W/2, H/2 + 150);
    }

    if (state === 'GAME_OVER') {
        drawOverlay();
        ctx.fillStyle = '#f44';
        ctx.textAlign = 'center';
        ctx.font = 'bold 48px monospace';
        ctx.fillText('GAME OVER', W/2, H/2 - 40);

        ctx.fillStyle = '#fff';
        ctx.font = '24px monospace';
        ctx.fillText(`Final Score: ${score}`, W/2, H/2 + 10);
        ctx.fillText(`Waves Survived: ${wave}`, W/2, H/2 + 45);
        ctx.fillText(`Max Combo: x${maxCombo}`, W/2, H/2 + 80);

        ctx.fillStyle = '#aaa';
        ctx.font = '18px monospace';
        ctx.fillText('CLICK TO RESTART', W/2, H/2 + 140);
    }

    ctx.restore();
}

function drawOverlay() {
    ctx.fillStyle = 'rgba(5, 5, 20, 0.85)';
    ctx.fillRect(0, 0, W, H);
}

function handleClick(x, y) {
    initAudio();
    
    if (state === 'MENU' || state === 'GAME_OVER') {
        resetGame();
        return;
    }

    if (state !== 'PLAYING') return;

    addRipple(x, y);
    
    let hitAny = false;
    
    for (let i = enemies.length - 1; i >= 0; i--) {
        const e = enemies[i];
        const dx = x - e.x;
        const dy = y - e.y;
        const dist = Math.sqrt(dx * dx + dy * dy);
        
        if (dist <= e.size + 15) {
            hitAny = true;
            e.hp--;
            e.hitFlash = 1;
            addParticles(e.x, e.y, e.color, 5);
            playSound('hit');
            
            if (e.hp <= 0) {
                // Kill
                combo++;
                maxCombo = Math.max(maxCombo, combo);
                const comboMult = Math.min(5, 1 + Math.floor(combo / 5));
                score += e.scoreVal * comboMult;
                addParticles(e.x, e.y, '#fff', 12);
                playSound('kill');
                enemies.splice(i, 1);
            } else {
                // Reset combo if not killing (actually keep it going for clicker feel)
                // Small combo increment handled above only on kill
            }
        }
    }

    if (!hitAny) {
        combo = Math.max(0, combo - 1);
    }
}

function useBomb() {
    if (state !== 'PLAYING' || bombs <= 0) return;
    bombs--;
    playSound('bomb');
    shakeTime = 0.3;
    shakeIntensity = 8;
    
    for (let i = enemies.length - 1; i >= 0; i--) {
        const e = enemies[i];
        e.hp -= 5;
        e.hitFlash = 1;
        addParticles(e.x, e.y, '#ff0', 8);
        
        if (e.hp <= 0) {
            combo++;
            maxCombo = Math.max(maxCombo, combo);
            const comboMult = Math.min(5, 1 + Math.floor(combo / 5));
            score += e.scoreVal * comboMult;
            addParticles(e.x, e.y, '#fff', 12);
            enemies.splice(i, 1);
        }
    }
}

// Input
canvas.addEventListener('click', (e) => {
    const rect = canvas.getBoundingClientRect();
    const x = (e.clientX - rect.left) * (W / rect.width);
    const y = (e.clientY - rect.top) * (H / rect.height);
    clickPos.x = x;
    clickPos.y = y;
    handleClick(x, y);
});

canvas.addEventListener('touchstart', (e) => {
    e.preventDefault();
    const touch = e.touches[0];
    const rect = canvas.getBoundingClientRect();
    const x = (touch.clientX - rect.left) * (W / rect.width);
    const y = (touch.clientY - rect.top) * (H / rect.height);
    handleClick(x, y);
});

window.addEventListener('keydown', (e) => {
    if (e.code === 'Space') {
        e.preventDefault();
        initAudio();
        if (state === 'MENU' || state === 'GAME_OVER') resetGame();
    }
    if (e.code === 'KeyB') {
        useBomb();
    }
});

// Game loop
function gameLoop(timestamp) {
    const dt = Math.min((timestamp - lastTime) / 1000, 0.05);
    lastTime = timestamp;
    
    update(dt);
    draw();
    
    requestAnimationFrame(gameLoop);
}

initTowers();
requestAnimationFrame(gameLoop);
</script>
