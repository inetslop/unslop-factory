// Audio
let audioCtx = null;
function initAudio(){
  if(!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
}
function playSound(type){
  if(!audioCtx) return;
  const osc = audioCtx.createOscillator();
  const gain = audioCtx.createGain();
  osc.connect(gain);
  gain.connect(audioCtx.destination);
  osc.type = 'square';
  switch(type){
    case 'jump': osc.frequency.value=400; gain.gain.setValueAtTime(0.15, audioCtx.currentTime); gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime+0.15); osc.start(); osc.stop(audioCtx.currentTime+0.15); break;
    case 'collect': osc.frequency.value=800; gain.gain.setValueAtTime(0.2, audioCtx.currentTime); gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime+0.2); osc.start(); osc.stop(audioCtx.currentTime+0.2); break;
    case 'hurt': osc.frequency.value=150; gain.gain.setValueAtTime(0.3, audioCtx.currentTime); gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime+0.3); osc.start(); osc.stop(audioCtx.currentTime+0.3); break;
    case 'win': osc.frequency.value=600; gain.gain.setValueAtTime(0.2, audioCtx.currentTime); gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime+0.5); osc.start(); osc.stop(audioCtx.currentTime+0.5); break;
    case 'splash': osc.frequency.value=100; gain.gain.setValueAtTime(0.25, audioCtx.currentTime); gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime+0.4); osc.start(); osc.stop(audioCtx.currentTime+0.4); break;
  }
}

  // Player movement

    player.vx = -MOVE_SPEED;
    player.facing = -1;

    player.vx = MOVE_SPEED;
    player.facing = 1;
  } else {
    player.vx *= 0.7;
  }
  
  // Jumping with coyote time
  if(player.onGround){
    player.coyoteTimer = 8;
  } else {
    player.coyoteTimer -= 1;

    player.vy = JUMP_VEL;
    player.onGround = false;

    playSound('jump');
    // Jump particles
    for(let i=0;i<5;i++){
      particles.push({x:player.x+player.w/2, y:player.y+player.h, vx:(Math.random()-0.5)*3, vy:-Math.random()*2, life:20, color:'#8f8'});
    }

  // Variable jump height
  if(!(keys['Space'] || keys['ArrowUp'] || keys['KeyW']) && player.vy < -6){

  if(player.vy > MAX_FALL) player.vy = MAX_FALL;
  
  // Move player
  player.x += player.vx;
  player.y += player.vy;
  
  // Platform collision
  player.onGround = false;
  for(let p of platforms){

    if(p.moving){
      py = p.baseY + Math.sin(time * 0.02 * p.moveSpeed + p.phase) * p.moveRange;

    // Landing on top

       player.y + player.h >= py &&

      player.y = py - player.h;
      player.vy = 0;
      player.onGround = true;
    }
    
    // Side collision (simple push)
    if(player.vy < 0 && player.y < py && player.y > py - 40){
      if(player.x + player.w > p.x && player.x < p.x + p.w){
        // Don't do much for side, just let gravity handle it
      }
    }
  }
  
  // Water death
  if(player.y > H - 30){
    playSound('splash');
    lives--;
    if(lives <= 0){
      gameState = 'gameover';
    } else {
      player.x = Math.max(50, cameraX + 50);
      player.y = 100;
      player.vy = 0;
      // Find nearest platform above
      let bestPlat = platforms[0];
      let bestDist = Infinity;
      for(let p of platforms){
        const d = Math.abs(p.x - player.x);
        if(d < bestDist && p.currentY > 50) { bestDist = d; bestPlat = p; }
      }
      player.x = bestPlat.x + bestPlat.w/2 - player.w/2;
      player.y = (bestPlat.currentY || bestPlat.y) - player.h - 60;
    }
  }
  
  // Collectibles
  for(let c of collectibles){
    if(c.collected) continue;
    const bobY = c.y + Math.sin(time * 0.05 + c.bobPhase) * 8;
    if(player.x < c.x + c.w && player.x + player.w > c.x && player.y < bobY + c.h && player.y + player.h > bobY){
      c.collected = true;
      score += 100;
      playSound('collect');
      for(let i=0;i<8;i++){
        particles.push({x:c.x+c.w/2, y:bobY+c.h/2, vx:(Math.random()-0.5)*4, vy:(Math.random()-0.5)*4, life:25, color:'#ff0'});
      }
    }
  }
  
  // Enemies
  for(let e of enemies){
    e.x += e.vx * e.dir;
    if(e.x < e.patrolL) { e.dir = 1; }
    if(e.x + e.w > e.patrolR) { e.dir = -1; }
    
    // Stomp or hurt
    if(player.x < e.x + e.w && player.x + player.w > e.x && player.y < e.y + e.h && player.y + player.h > e.y){
      if(player.vy > 0 && player.y + player.h - player.vy <= e.y + 10){
        // Stomp!
        e.dead = true;
        player.vy = -8;
        score += 200;
        playSound('collect');
        for(let i=0;i<6;i++){
          particles.push({x:e.x+e.w/2, y:e.y+e.h/2, vx:(Math.random()-0.5)*4, vy:-Math.random()*3, life:20, color:'#f80'});
        }

        if(lives <= 0){ gameState = 'gameover'; }
        else { player.vy = -10; player.x -= 30; }
      }
    }
  }
  enemies = enemies.filter(e => !e.dead);

  if(goal && player.x + player.w > goal.x && player.x < goal.x + goal.w && player.y + player.h > goal.y && player.y < goal.y + goal.h){

    generateLevel(level);
  }

    p.y += p.vy;
    p.vy += 0.2;
    p.life--;
  }

  if(cameraX < 0) cameraX = 0;
