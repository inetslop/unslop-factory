const COLORS = [

const SHAPES = [
    [[0,0],[1,0],[2,0],[3,0]], // I
    [[0,0],[1,0],[0,1],[1,1]], // O
    [[0,0],[1,0],[2,0],[1,1]], // T
    [[0,0],[0,1],[0,2],[1,2]], // L

function initAudio() {
    if (!audioCtx) {

function playSound(type) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    
    if (type === 'drop') {
        osc.type = 'sine';
        osc.frequency.setValueAtTime(200, audioCtx.currentTime);
        osc.frequency.exponentialRampToValueAtTime(100, audioCtx.currentTime + 0.1);
        gain.gain.setValueAtTime(0.3, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.1);
        osc.start(audioCtx.currentTime);
        osc.stop(audioCtx.currentTime + 0.1);
    } else if (type === 'match') {
        osc.type = 'square';
        osc.frequency.setValueAtTime(400, audioCtx.currentTime);
        osc.frequency.exponentialRampToValueAtTime(800, audioCtx.currentTime + 0.15);
        gain.gain.setValueAtTime(0.2, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.2);
        osc.start(audioCtx.currentTime);
        osc.stop(audioCtx.currentTime + 0.2);
    } else if (type === 'chain') {
        osc.type = 'sawtooth';
        osc.frequency.setValueAtTime(300, audioCtx.currentTime);
        osc.frequency.exponentialRampToValueAtTime(600, audioCtx.currentTime + 0.2);
        gain.gain.setValueAtTime(0.15, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.3);
        osc.start(audioCtx.currentTime);
        osc.stop(audioCtx.currentTime + 0.3);
    } else if (type === 'gameover') {
        osc.type = 'sawtooth';
        osc.frequency.setValueAtTime(300, audioCtx.currentTime);
        osc.frequency.exponentialRampToValueAtTime(100, audioCtx.currentTime + 0.5);
        gain.gain.setValueAtTime(0.2, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.5);
        osc.start(audioCtx.currentTime);
        osc.stop(audioCtx.currentTime + 0.5);
    } else if (type === 'rotate') {
        osc.type = 'sine';
        osc.frequency.setValueAtTime(500, audioCtx.currentTime);
        gain.gain.setValueAtTime(0.15, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.05);
        osc.start(audioCtx.currentTime);
        osc.stop(audioCtx.currentTime + 0.05);
    } else if (type === 'levelup') {
        osc.type = 'sine';
        osc.frequency.setValueAtTime(400, audioCtx.currentTime);

        gain.gain.setValueAtTime(0.25, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.4);
        osc.start(audioCtx.currentTime);
        osc.stop(audioCtx.currentTime + 0.4);
    }

function resetGrid() {
    grid = Array.from({ length: ROWS }, () => Array(COLS).fill(-1));
}

function randomPiece() {
    const shapeIdx = Math.floor(Math.random() * SHAPES.length);
    const colorIdx = Math.floor(Math.random() * COLORS.length);
    return {
        shape: SHAPES[shapeIdx].map(c => [...c]),
        color: colorIdx,
        x: 0, y: -1,
        rotation: 0
    };
}

function getPieceCells(piece) {
    let cells = [];
    for (let c of piece.shape) {
        cells.push({ x: piece.x + c[0], y: piece.y + c[1] });
    }
    return cells;
}

function rotateShape(shape, rot) {
    // For simplicity, we only use 2 rotation states for non-I shapes
    if (rot % 2 === 0) return shape.map(c => [...c]);
    else return shape.map(c => [c[1], c[0]]);
}

function canPlace(piece, px, py, shapeOverride) {
    let cells = [];
    for (let c of (shapeOverride || piece.shape)) {
        cells.push({ x: px + c[0], y: py + c[1] });
    }
    for (let cell of cells) {
        if (cell.x < 0 || cell.x >= COLS || cell.y >= ROWS) return false;
        if (cell.y >= 0 && grid[cell.y][cell.x] !== -1) return false;
    }
    return true;
}

function tryMove(dx, dy) {
    if (!currentPiece) return false;
    let nx = currentPiece.x + dx;
    let ny = currentPiece.y + dy;
    if (canPlace(currentPiece, nx, ny)) {
        currentPiece.x = nx;
        currentPiece.y = ny;
        return true;
    }
    return false;
}

function tryRotate() {
    if (!currentPiece) return;
    let newRot = (currentPiece.rotation + 1) % 2;
    let newShape = rotateShape(currentPiece.shape, newRot);
    
    // Try original position first, then shift left/right
    for (let offset of [0, -1, 1, -2, 2]) {
        if (canPlace(currentPiece, currentPiece.x + offset, currentPiece.y, newShape)) {
            currentPiece.x += offset;
            currentPiece.rotation = newRot;
            currentPiece.shape = newShape;
            playSound('rotate');
            return;
        }
    }
}

function hardDrop() {
    if (!currentPiece) return;
    while (tryMove(0, 1)) {}
    lockPiece();
}

function lockPiece() {
    let cells = getPieceCells(currentPiece);
    for (let cell of cells) {
        if (cell.y >= 0 && cell.y < ROWS && cell.x >= 0 && cell.x < COLS) {
            grid[cell.y][cell.x] = currentPiece.color;
        }
    }
    playSound('drop');
    spawnLockParticles(cells);
    
    currentPiece = null;
    checkMatches();
}

function findMatches() {
    let visited = Array.from({ length: ROWS }, () => Array(COLS).fill(false));
    let matched = [];
    
    for (let y = 0; y < ROWS; y++) {
        for (let x = 0; x < COLS; x++) {
            if (grid[y][x] !== -1 && !visited[y][x]) {
                // Flood fill
                let group = [];
                let queue = [{ x, y }];
                visited[y][x] = true;
                let color = grid[y][x];
                
                while (queue.length > 0) {
                    let { x: cx, y: cy } = queue.shift();
                    group.push({ x: cx, y: cy });
                    
                    const dirs = [[0,-1],[0,1],[-1,0],[1,0]];
                    for (let [dx, dy] of dirs) {
                        let nx = cx + dx, ny = cy + dy;
                        if (nx >= 0 && nx < COLS && ny >= 0 && ny < ROWS && 
                            !visited[ny][nx] && grid[ny][nx] === color) {
                            visited[ny][nx] = true;
                            queue.push({ x: nx, y: ny });
                        }
                    }
                }
                
                if (group.length >= 3) {
                    matched.push(...group);
                }
            }
        }
    }
    
    return matched;
}

function checkMatches() {
    let matches = findMatches();
    if (matches.length > 0) {
        gameState = 'clearing';
        startClearAnimation(matches, 1);
    } else {
        nextPiece();
    }
}

function startClearAnimation(matches, chain) {
    playSound(chain > 1 ? 'chain' : 'match');
    
    let points = matches.length * 10 * chain;
    score += points;
    if (score > highScore) {
        highScore = score;
        try { localStorage.setItem('tetrisMatch3High', highScore.toString()); } catch(e) {}
    }
    
    // Level up every 500 points
    let newLevel = Math.floor(score / 500) + 1;
    if (newLevel > level) {
        level = newLevel;
        dropInterval = Math.max(300, 800 - (level - 1) * 60);
        playSound('levelup');
        chainTexts.push({ text: `LEVEL ${level}!`, x: W/2, y: gridY + ROWS*cellSize/2, life: 60, maxLife: 60 });
    }
    
    for (let cell of matches) {
        clearAnimations.push({
            x: cell.x, y: cell.y,
            color: grid[cell.y][cell.x],
            scale: 1, alpha: 1, life: 20
        });
        spawnClearParticles(cell.x, cell.y, grid[cell.y][cell.x]);
    }
    
    chainTexts.push({
        text: `+${points}${chain > 1 ? ` x${chain} CHAIN!` : ''}`,
        x: W/2, y: gridY + ROWS*cellSize * 0.6, life: 45, maxLife: 45
    });
    
    // Remove matched cells
    for (let cell of matches) {
        grid[cell.y][cell.x] = -1;
    }
    
    // Apply gravity
    applyGravity();
    
    // Check for more matches after gravity (chain reaction)
    setTimeout(() => {
        let newMatches = findMatches();
        if (newMatches.length > 0) {
            startClearAnimation(newMatches, chain + 1);
        } else {
            nextPiece();
        }
    }, 300);
}

function applyGravity() {
    for (let x = 0; x < COLS; x++) {
        let writeRow = ROWS - 1;
        for (let y = ROWS - 1; y >= 0; y--) {
            if (grid[y][x] !== -1) {
                grid[writeRow][x] = grid[y][x];
                if (writeRow !== y) grid[y][x] = -1;
                writeRow--;
            }
        }
        for (let y = writeRow; y >= 0; y--) {
            grid[y][x] = -1;
        }
    }
}

function nextPiece() {
    gameState = 'playing';
    currentPiece = nextPiece || randomPiece();
    let np = randomPiece();
    
    // Check if piece can be placed
    let startShape = currentPiece.shape;
    let startX = Math.floor((COLS - 4) / 2);
    
    if (!canPlace(currentPiece, startX, -1, startShape)) {
        // Try different starting positions
        let found = false;
        for (let sx = 0; sx < COLS; sx++) {
            if (canPlace(currentPiece, sx, -1, startShape)) {
                currentPiece.x = sx;
                found = true;
                break;
            }
        }
        if (!found) {
            // Try at y=0
            for (let sx = 0; sx < COLS; sx++) {
                if (canPlace(currentPiece, sx, 0, startShape)) {
                    currentPiece.x = sx;
                    currentPiece.y = 0;
                    found = true;
                    break;
                }
            }
        }
        if (!found) {
            gameOver();
            return;
        }
    }
    
    currentPiece.x = startX;
    currentPiece.y = -1;
    nextPiece = np;
}

function gameOver() {
    gameState = 'gameover';
    playSound('gameover');
}

function spawnLockParticles(cells) {
    for (let cell of cells) {
        for (let i = 0; i < 3; i++) {
            particles.push({
                x: gridX + (cell.x + 0.5) * cellSize,
                y: gridY + (cell.y + 0.5) * cellSize,
                vx: (Math.random() - 0.5) * 3,
                vy: (Math.random() - 0.5) * 3 - 2,
                life: 20, maxLife: 20,
                color: COLORS[currentPiece ? currentPiece.color : 0].main
            });
        }
    }
}

function spawnClearParticles(gx, gy, colorIdx) {
    for (let i = 0; i < 6; i++) {
        particles.push({
            x: gridX + (gx + 0.5) * cellSize,
            y: gridY + (gy + 0.5) * cellSize,
            vx: (Math.random() - 0.5) * 6,
            vy: (Math.random() - 0.5) * 6 - 2,
            life: 30, maxLife: 30,
            color: COLORS[colorIdx].main
        });
    }
}

function drawCell(x, y, colorIdx, scale = 1, alpha = 1) {
    if (colorIdx < 0) return;
    let px = gridX + x * cellSize;
    let py = gridY + y * cellSize;
    let s = cellSize * scale;
    let ox = (cellSize - s) / 2;
    
    ctx.globalAlpha = alpha;
    ctx.fillStyle = COLORS[colorIdx].dark;
    ctx.fillRect(px + ox, py + ox, s, s);
    
    ctx.fillStyle = COLORS[colorIdx].main;
    let inset = Math.max(1, s * 0.1);
    ctx.fillRect(px + ox + inset, py + ox + inset, s - inset*2, s - inset*2);
    
    ctx.fillStyle = COLORS[colorIdx].light;
    let hi = Math.max(1, s * 0.15);
    ctx.fillRect(px + ox + inset + hi, py + ox + inset + hi, (s - inset*2) * 0.4, (s - inset*2) * 0.3);
    
    ctx.globalAlpha = 1;
}

function drawGrid() {
    // Grid background
    ctx.fillStyle = '#16213e';
    ctx.fillRect(gridX - 4, gridY - 4, COLS * cellSize + 8, ROWS * cellSize + 8);
    
    // Grid lines
    ctx.strokeStyle = 'rgba(255,255,255,0.05)';
    ctx.lineWidth = 1;
    for (let x = 0; x <= COLS; x++) {
        ctx.beginPath();
        ctx.moveTo(gridX + x * cellSize, gridY);
        ctx.lineTo(gridX + x * cellSize, gridY + ROWS * cellSize);
        ctx.stroke();
    }
    for (let y = 0; y <= ROWS; y++) {
        ctx.beginPath();
        ctx.moveTo(gridX, gridY + y * cellSize);
        ctx.lineTo(gridX + COLS * cellSize, gridY + y * cellSize);
        ctx.stroke();
    }
    
    // Placed blocks
    for (let y = 0; y < ROWS; y++) {
        for (let x = 0; x < COLS; x++) {
            if (grid[y][x] !== -1) drawCell(x, y, grid[y][x]);
        }
    }
}

function drawPiece() {
    if (!currentPiece) return;
    
    // Ghost position
    let ghostY = currentPiece.y;
    while (canPlace(currentPiece, currentPiece.x, ghostY + 1)) ghostY++;
    
    ctx.globalAlpha = 0.3;
    for (let c of currentPiece.shape) {
        let gx = currentPiece.x + c[0];
        let gy = ghostY + c[1];
        if (gy >= 0 && gy < ROWS && gx >= 0 && gx < COLS) {
            drawCell(gx, gy, currentPiece.color, 1, 0.3);
        }
    }
    ctx.globalAlpha = 1;
    
    // Current piece
    for (let c of currentPiece.shape) {
        let px = currentPiece.x + c[0];
        let py = currentPiece.y + c[1];
        if (py >= 0 && py < ROWS && px >= 0 && px < COLS) {
            drawCell(px, py, currentPiece.color);
        }
    }
}

function drawClearAnimations() {
    for (let anim of clearAnimations) {
        let scale = 1 + anim.scale * 0.5;
        ctx.globalAlpha = anim.alpha;
        ctx.fillStyle = COLORS[anim.color].main;
        let px = gridX + anim.x * cellSize;
        let py = gridY + anim.y * cellSize;
        let s = cellSize * scale;
        let ox = (cellSize - s) / 2;
        ctx.fillRect(px + ox, py + ox, s, s);
        ctx.globalAlpha = 1;
    }
}

function drawParticles() {
    for (let p of particles) {
        ctx.globalAlpha = p.life / p.maxLife;
        ctx.fillStyle = p.color;
        let size = 4 * (p.life / p.maxLife);
        ctx.fillRect(p.x - size/2, p.y - size/2, size, size);
    }
    ctx.globalAlpha = 1;
}

function drawChainTexts() {
    for (let ct of chainTexts) {
        let alpha = ct.life / ct.maxLife;
        ctx.globalAlpha = alpha;
        ctx.fillStyle = '#fff';
        ctx.font = `bold ${Math.floor(cellSize * 0.6)}px Segoe UI, sans-serif`;
        ctx.textAlign = 'center';
        ctx.fillText(ct.text, ct.x, ct.y);
        ctx.globalAlpha = 1;
    }
}

        ctx.font = `${Math.min(W * 0.025, 18)}px Segoe UI, sans-serif`;
        ctx.fillStyle = '#ffcc33';
        ctx.fillText(`High Score: ${highScore}`, cx, cy + 180);
    }
}

function drawGameOver() {
    ctx.fillStyle = 'rgba(0,0,0,0.7)';
    ctx.fillRect(0, 0, W, H);
    
    let cx = W / 2;
    let cy = H / 2;
    
    ctx.textAlign = 'center';
    ctx.font = `bold ${Math.min(W * 0.06, 50)}px Segoe UI, sans-serif`;
    ctx.fillStyle = '#ff4444';
    ctx.fillText('GAME OVER', cx, cy - 40);
    
    ctx.font = `${Math.min(W * 0.03, 24)}px Segoe UI, sans-serif`;
    ctx.fillStyle = '#fff';
    ctx.fillText(`Final Score: ${score}`, cx, cy + 10);
    ctx.fillText(`High Score: ${highScore}`, cx, cy + 45);
    
    let pulse = 0.5 + 0.5 * Math.sin(Date.now() / 500);
    ctx.globalAlpha = pulse;
    ctx.font = `bold ${Math.min(W * 0.03, 22)}px Segoe UI, sans-serif`;
    ctx.fillStyle = '#44dd44';
    ctx.fillText('Press ENTER or R to Restart', cx, cy + 100);
    ctx.globalAlpha = 1;
}

function update() {
    // Update clear animations
    for (let i = clearAnimations.length - 1; i >= 0; i--) {
        let a = clearAnimations[i];
        a.scale += 0.1;
        a.alpha -= 0.05;
        if (a.life > 0) a.life--;
        else clearAnimations.splice(i, 1);
    }
    
    // Update particles
    for (let i = particles.length - 1; i >= 0; i--) {
        let p = particles[i];
        p.x += p.vx;
        p.y += p.vy;
        p.vy += 0.3;
        p.life--;
        if (p.life <= 0) particles.splice(i, 1);
    }
    
    // Update chain texts
    for (let i = chainTexts.length - 1; i >= 0; i--) {
        let ct = chainTexts[i];
        ct.y -= 0.5;
        ct.life--;
        if (ct.life <= 0) chainTexts.splice(i, 1);
    }
    
    // Game logic
    if (gameState === 'playing' && currentPiece) {
        let now = Date.now();
        if (now - lastDrop > dropInterval) {
            if (!tryMove(0, 1)) {
                lockPiece();
            }
            lastDrop = now;
        }
    }
}

function draw() {
    ctx.fillStyle = '#1a1a2e';
    ctx.fillRect(0, 0, W, H);
    
    if (gameState === 'start') {
        drawStartScreen();
        return;
    }
    
    drawGrid();
    drawClearAnimations();
    drawPiece();
    drawParticles();
    drawChainTexts();
    drawHUD();
    
    if (gameState === 'gameover') {
        drawGameOver();
    }
}

function gameLoop() {
    update();
    draw();
    requestAnimationFrame(gameLoop);
}

function startGame() {
    resetGrid();
    score = 0;
    level = 1;
    dropInterval = 800;
    lastDrop = Date.now();
    clearAnimations = [];
    particles = [];
    chainTexts = [];
    currentPiece = null;
    nextPiece = randomPiece();
    gameState = 'playing';
    
    let sp = randomPiece();
    sp.x = Math.floor((COLS - 4) / 2);
    sp.y = -1;
    currentPiece = sp;
    lastDrop = Date.now();
}

function handleKey(e) {
    initAudio();
    
    if (gameState === 'start') {
        if (e.code === 'Enter' || e.code === 'Space') {
            startGame();
        }
        return;
    }
    
    if (gameState === 'gameover') {
        if (e.code === 'Enter' || e.code === 'KeyR') {
            startGame();
        }
        return;
    }
    
    if (gameState !== 'playing' || !currentPiece) return;
    
    switch (e.code) {
        case 'ArrowLeft':
            tryMove(-1, 0);
            e.preventDefault();
            break;
        case 'ArrowRight':
            tryMove(1, 0);
            e.preventDefault();
            break;
        case 'ArrowDown':
            if (tryMove(0, 1)) {
                score += 2;
                lastDrop = Date.now();
            } else {
                lockPiece();
            }
            e.preventDefault();
            break;
        case 'ArrowUp':
        case 'KeyX':
            tryRotate();
            e.preventDefault();
            break;
        case 'Space':
            hardDrop();
            score += 5;
            e.preventDefault();
            break;
    }
}

window.addEventListener('keydown', handleKey);

// Mobile / touch support - tap left/right halves
canvas.addEventListener('touchstart', (e) => {
    initAudio();
    if (gameState === 'start' || gameState === 'gameover') {
        startGame();
        return;
    }
    
    let touch = e.touches[0];
    let rect = canvas.getBoundingClientRect();
    let x = touch.clientX - rect.left;
    let y = touch.clientY - rect.top;
    
    if (x < W * 0.3) {
        tryMove(-1, 0);
    } else if (x > W * 0.7) {
        tryMove(1, 0);
    } else if (y > H * 0.5) {
        hardDrop();
    } else {
        tryRotate();
    }
    e.preventDefault();
});

gameLoop();
