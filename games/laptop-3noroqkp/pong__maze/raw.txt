<script>
(function() {
const canvas = document.getElementById('game');
const ctx = canvas.getContext('2d');

let W, H;
function resize() {
    W = window.innerWidth;
    H = window.innerHeight;
    canvas.width = W;
    canvas.height = H;
}
window.addEventListener('resize', resize);
resize();

// Audio
let audioCtx = null;
function initAudio() {
    if (!audioCtx) audioCtx = new (window.AudioContext || webkitAudioContext)();
}
function playSound(freq, duration, type, vol) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = type || 'sine';
    osc.frequency.value = freq;
    gain.gain.setValueAtTime(vol || 0.3, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + duration);
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + duration);
}
function sfxPaddle() { playSound(440, 0.08, 'square', 0.2); }
function sfxWall() { playSound(220, 0.06, 'triangle', 0.15); }
function sfxGoal() { playSound(330, 0.4, 'sawtooth', 0.25); setTimeout(() => playSound(220, 0.5, 'sawtooth', 0.2), 150); }
function sfxStart() { playSound(660, 0.15, 'sine', 0.2); setTimeout(() => playSound(880, 0.2, 'sine', 0.2), 100); }

// Game state
let gameState = 'START'; // START, COUNTDOWN, PLAYING, SCORED, GAMEOVER
let score1 = 0, score2 = 0;
const WIN_SCORE = 7;
let countdownTimer = 0;
let scoredTimer = 0;
let lastScorer = 0;

// Particles
let particles = [];

function spawnParticles(x, y, color, count) {
    for (let i = 0; i < count; i++) {
        const angle = Math.random() * Math.PI * 2;
        const speed = 1 + Math.random() * 4;
        particles.push({ x, y, vx: Math.cos(angle)*speed, vy: Math.sin(angle)*speed, life: 1, color });
    }
}

// Paddles
let paddleW, paddleH, ballR, wallThick;
function initSizes() {
    paddleW = Math.max(12, W * 0.015);
    paddleH = Math.max(60, H * 0.18);
    ballR = Math.max(8, W * 0.012);
    wallThick = Math.max(8, W * 0.01);
}

let p1 = { x: 30, y: 0, vy: 0 };
let p2 = { x: 0, y: 0, vy: 0 };
let ball = { x: 0, y: 0, vx: 0, vy: 0 };

function resetBall(direction) {
    initSizes();
    ball.x = W / 2;
    ball.y = H / 2;
    const angle = (Math.random() - 0.5) * Math.PI / 3;
    const speed = Math.max(4, W * 0.008);
    ball.vx = Math.cos(angle) * speed * direction;
    ball.vy = Math.sin(angle) * speed;
}

function resetGame() {
    score1 = 0; score2 = 0;
    p1.y = H / 2 - paddleH / 2;
    p2.y = H / 2 - paddleH / 2;
    resetBall(Math.random() < 0.5 ? 1 : -1);
}

// Maze walls (relative positions)
let mazeWalls = [];
function buildMaze() {
    mazeWalls = [];
    const t = wallThick;
    const cx = W / 2;
    // Central vertical wall segments with gaps
    const segH = H * 0.25;
    const gap = paddleH * 1.2;
    // Top center block
    mazeWalls.push({ x: cx - t/2, y: H*0.1, w: t, h: segH });
    // Bottom center block
    mazeWalls.push({ x: cx - t/2, y: H*0.9 - segH, w: t, h: segH });
    // Left-center horizontal walls
    const hw = W * 0.15;
    mazeWalls.push({ x: W*0.3 - hw/2, y: H*0.3, w: hw, h: t });
    mazeWalls.push({ x: W*0.7 - hw/2, y: H*0.7, w: hw, h: t });
    // Additional center obstacles
    const obsSize = Math.min(W * 0.04, 30);
    mazeWalls.push({ x: cx - obsSize/2, y: H*0.5 - obsSize/2, w: obsSize, h: obsSize });
}

// AI
let aiTarget = H / 2;
function updateAI() {
    const predictY = ball.y + ball.vy * 10;
    const clampedY = Math.max(paddleH/2, Math.min(H - paddleH/2, predictY));
    aiTarget += (clampedY - aiTarget) * 0.08;
    const diff = aiTarget - p2.y;
    const speed = Math.max(3, H * 0.006);
    if (Math.abs(diff) > 5) {
        p2.y += Math.sign(diff) * Math.min(speed, Math.abs(diff));
    }
}

// Input
let keys = {};
window.addEventListener('keydown', e => { keys[e.key] = true; initAudio(); });
window.addEventListener('keyup', e => { keys[e.key] = false; });

function handleInput() {
    if (gameState === 'START' || gameState === 'GAMEOVER') {
        if (keys[' '] || keys['Enter']) {
            sfxStart();
            resetGame();
            gameState = 'COUNTDOWN';
            countdownTimer = 60;
        }
    } else if (gameState === 'PLAYING') {
        const pSpeed = Math.max(5, H * 0.01);
        if (keys['w'] || keys['W']) p1.y -= pSpeed;
        if (keys['s'] || keys['S']) p1.y += pSpeed;
        // P2 AI or human
        if (gameMode === 'ai') {
            updateAI();
        } else {
            if (keys['ArrowUp']) p2.y -= pSpeed;
            if (keys['ArrowDown']) p2.y += pSpeed;
        }
        // Clamp paddles
        p1.y = Math.max(0, Math.min(H - paddleH, p1.y));
        p2.y = Math.max(0, Math.min(H - paddleH, p2.y));
    }
}

let gameMode = 'ai'; // 'ai' or '2p'
window.addEventListener('keydown', e => {
    if (gameState === 'START' && (e.key === 'r' || e.key === 'R')) {
        gameMode = '2p';
    }
});

// Collision detection
function circleRectCollision(cx, cy, r, rx, ry, rw, rh) {
    const closestX = Math.max(rx, Math.min(cx, rx + rw));
    const closestY = Math.max(ry, Math.min(cy, ry + rh));
    const dx = cx - closestX;
    const dy = cy - closestY;
    return (dx * dx + dy * dy) < r * r;
}

function bounceOffRect(rx, ry, rw, rh) {
    // Determine which side the ball hit
    const bcx = rx + rw / 2;
    const bcy = ry + rh / 2;
    const dx = ball.x - bcx;
    const dy = ball.y - bcy;
    
    if (Math.abs(dx) / (rw/2 + ballR) > Math.abs(dy) / (rh/2 + ballR)) {
        ball.vx = Math.abs(ball.vx);
        ball.x += (dx > 0 ? 1 : -1) * ballR;
    } else {
        ball.vy = Math.abs(ball.vy);
        ball.y += (dy > 0 ? 1 : -1) * ballR;
    }
}

function updateBall() {
    ball.x += ball.vx;
    ball.y += ball.vy;

    // Top/Bottom walls
    if (ball.y - ballR < 0) { ball.y = ballR; ball.vy = Math.abs(ball.vy); sfxWall(); }
    if (ball.y + ballR > H) { ball.y = H - ballR; ball.vy = -Math.abs(ball.vy); sfxWall(); }

    // Paddle 1 collision
    const p1Rect = { x: p1.x, y: p1.y, w: paddleW, h: paddleH };
    if (circleRectCollision(ball.x, ball.y, ballR, p1Rect.x, p1Rect.y, p1Rect.w, p1Rect.h)) {
        bounceOffRect(p1Rect.x, p1Rect.y, p1Rect.w, p1Rect.h);
        // Add spin based on where it hits
        const hitPos = (ball.y - (p1.y + paddleH/2)) / (paddleH/2);
        ball.vy += hitPos * 2;
        sfxPaddle();
        spawnParticles(ball.x, ball.y, '#00ffcc', 5);
    }

    // Paddle 2 collision
    const p2Rect = { x: p2.x, y: p2.y, w: paddleW, h: paddleH };
    if (circleRectCollision(ball.x, ball.y, ballR, p2Rect.x, p2Rect.y, p2Rect.w, p2Rect.h)) {
        bounceOffRect(p2Rect.x, p2Rect.y, p2Rect.w, p2Rect.h);
        const hitPos = (ball.y - (p2.y + paddleH/2)) / (paddleH/2);
        ball.vy += hitPos * 2;
        sfxPaddle();
        spawnParticles(ball.x, ball.y, '#ff6699', 5);
    }

    // Maze walls
    for (const wall of mazeWalls) {
        if (circleRectCollision(ball.x, ball.y, ballR, wall.x, wall.y, wall.w, wall.h)) {
            bounceOffRect(wall.x, wall.y, wall.w, wall.h);
            sfxWall();
            spawnParticles(ball.x, ball.y, '#ffaa00', 3);
        }
    }

    // Goals (left and right)
    const goalTop = H * 0.2;
    const goalBottom = H * 0.8;
    
    if (ball.x - ballR < 0) {
        if (ball.y > goalTop && ball.y < goalBottom) {
            score2++;
            sfxGoal();
            spawnParticles(10, ball.y, '#ff4444', 20);
            gameState = 'SCORED';
            scoredTimer = 60;
            lastScorer = 2;
            if (score2 >= WIN_SCORE) { gameState = 'GAMEOVER'; }
        } else {
            ball.x = ballR; ball.vx = Math.abs(ball.vx); sfxWall();
        }
    }
    
    if (ball.x + ballR > W) {
        if (ball.y > goalTop && ball.y < goalBottom) {
            score1++;
            sfxGoal();
            spawnParticles(W-10, ball.y, '#44ff44', 20);
            gameState = 'SCORED';
            scoredTimer = 60;
            lastScorer = 1;
            if (score1 >= WIN_SCORE) { gameState = 'GAMEOVER'; }
        } else {
            ball.x = W - ballR; ball.vx = -Math.abs(ball.vx); sfxWall();
        }
    }

    // Speed limit
    const maxSpeed = Math.max(8, W * 0.012);
    const speed = Math.sqrt(ball.vx*ball.vx + ball.vy*ball.vy);
    if (speed > maxSpeed) {
        ball.vx = (ball.vx / speed) * maxSpeed;
        ball.vy = (ball.vy / speed) * maxSpeed;
    }
}

function updateParticles() {
    for (let i = particles.length - 1; i >= 0; i--) {
        const p = particles[i];
        p.x += p.vx;
        p.y += p.vy;
        p.life -= 0.03;
        if (p.life <= 0) particles.splice(i, 1);
    }
}

// Rendering
function drawMaze() {
    ctx.fillStyle = '#ffaa00';
    for (const wall of mazeWalls) {
        ctx.fillRect(wall.x, wall.y, wall.w, wall.h);
        // Glow
        ctx.save();
        ctx.shadowColor = '#ffaa00';
        ctx.shadowBlur = 8;
        ctx.fillRect(wall.x, wall.y, wall.w, wall.h);
        ctx.restore();
    }
}

function drawGoals() {
    const goalTop = H * 0.2;
    const goalBottom = H * 0.8;
    // Left goal (P1 scores here)
    ctx.fillStyle = '#44ff44';
    ctx.shadowColor = '#44ff44';
    ctx.shadowBlur = 15;
    ctx.fillRect(0, goalTop, wallThick * 2, goalBottom - goalTop);
    // Right goal (P2 scores here)
    ctx.fillStyle = '#ff4466';
    ctx.shadowColor = '#ff4466';
    ctx.shadowBlur = 15;
    ctx.fillRect(W - wallThick * 2, goalTop, wallThick * 2, goalBottom - goalTop);
    ctx.shadowBlur = 0;

    // Goal lines
    ctx.strokeStyle = 'rgba(255,255,255,0.3)';
    ctx.lineWidth = 1;
    ctx.setLineDash([5, 5]);
    ctx.beginPath();
    ctx.moveTo(wallThick * 2, goalTop);
    ctx.lineTo(W - wallThick * 2, goalTop);
    ctx.stroke();
    ctx.beginPath();
    ctx.moveTo(wallThick * 2, goalBottom);
    ctx.lineTo(W - wallThick * 2, goalBottom);
    ctx.stroke();
    ctx.setLineDash([]);
}

function drawPaddles() {
    // P1
    ctx.fillStyle = '#00ffcc';
    ctx.shadowColor = '#00ffcc';
    ctx.shadowBlur = 12;
    ctx.fillRect(p1.x, p1.y, paddleW, paddleH);
    // P2
    ctx.fillStyle = '#ff6699';
    ctx.shadowColor = '#ff6699';
    ctx.shadowBlur = 12;
    ctx.fillRect(p2.x, p2.y, paddleW, paddleH);
    ctx.shadowBlur = 0;
}

function drawBall() {
    ctx.fillStyle = '#ffffff';
    ctx.shadowColor = '#ffffff';
    ctx.shadowBlur = 15;
    ctx.beginPath();
    ctx.arc(ball.x, ball.y, ballR, 0, Math.PI * 2);
    ctx.fill();
    ctx.shadowBlur = 0;
}

function drawParticles() {
    for (const p of particles) {
        ctx.globalAlpha = p.life;
        ctx.fillStyle = p.color;
        ctx.fillRect(p.x - 2, p.y - 2, 4, 4);
    }
    ctx.globalAlpha = 1;
}

function drawScore() {
    ctx.font = `bold ${Math.min(W*0.08, 60)}px monospace`;
    ctx.textAlign = 'center';
    ctx.fillStyle = '#00ffcc';
    ctx.fillText(score1, W * 0.35, H * 0.12);
    ctx.fillStyle = '#ff6699';
    ctx.fillText(score2, W * 0.65, H * 0.12);
}

function drawStartScreen() {
    ctx.fillStyle = 'rgba(10,10,26,0.85)';
    ctx.fillRect(0, 0, W, H);
    
    ctx.textAlign = 'center';
    ctx.font = `bold ${Math.min(W*0.08, 72)}px monospace`;
    ctx.fillStyle = '#ffffff';
    ctx.shadowColor = '#00ffcc';
    ctx.shadowBlur = 20;
    ctx.fillText('MAZE PONG', W/2, H*0.3);
    ctx.shadowBlur = 0;

    ctx.font = `${Math.min(W*0.03, 24)}px monospace`;
    ctx.fillStyle = '#aaaacc';
    const lines = [
        'P1: W / S to move',
        gameMode === 'ai' ? 'P2: AI (press R on start screen for 2P)' : 'P2: Arrow Up / Down',
        '',
        'First to ' + WIN_SCORE + ' wins!',
        '',
        'Press SPACE or ENTER to start',
        'Press R to toggle 1P/2P mode'
    ];
    lines.forEach((line, i) => {
        ctx.fillText(line, W/2, H*0.42 + i * Math.min(W*0.035, 30));
    });

    // Draw decorative ball
    const t = Date.now() / 1000;
    ctx.beginPath();
    ctx.arc(W/2 + Math.sin(t)*20, H*0.7 + Math.cos(t*1.3)*10, 12, 0, Math.PI*2);
    ctx.fillStyle = '#ffffff';
    ctx.shadowColor = '#ffffff';
    ctx.shadowBlur = 20;
    ctx.fill();
    ctx.shadowBlur = 0;
}

function drawGameOver() {
    ctx.fillStyle = 'rgba(10,10,26,0.8)';
    ctx.fillRect(0, 0, W, H);
    ctx.textAlign = 'center';
    ctx.font = `bold ${Math.min(W*0.07, 64)}px monospace`;
    
    const winner = score1 > score2 ? 'PLAYER 1' : 'PLAYER 2';
    const color = score1 > score2 ? '#00ffcc' : '#ff6699';
    ctx.fillStyle = color;
    ctx.shadowColor = color;
    ctx.shadowBlur = 20;
    ctx.fillText(winner + ' WINS!', W/2, H*0.4);
    ctx.shadowBlur = 0;

    ctx.font = `${Math.min(W*0.035, 28)}px monospace`;
    ctx.fillStyle = '#ffffff';
    ctx.fillText(score1 + ' - ' + score2, W/2, H*0.5);
    ctx.fillStyle = '#aaaacc';
    ctx.font = `${Math.min(W*0.03, 22)}px monospace`;
    ctx.fillText('Press SPACE to play again', W/2, H*0.62);
}

function drawCountdown() {
    const num = Math.ceil(countdownTimer / 20);
    ctx.textAlign = 'center';
    ctx.font = `bold ${Math.min(W*0.15, 120)}px monospace`;
    ctx.fillStyle = '#ffffff';
    ctx.shadowColor = '#ffffff';
    ctx.shadowBlur = 30;
    ctx.fillText(num > 0 ? num : 'GO!', W/2, H/2 + 20);
    ctx.shadowBlur = 0;
}

function drawCenterLine() {
    ctx.strokeStyle = 'rgba(255,255,255,0.1)';
    ctx.lineWidth = 2;
    ctx.setLineDash([10, 15]);
    ctx.beginPath();
    ctx.moveTo(W/2, 0);
    ctx.lineTo(W/2, H);
    ctx.stroke();
    ctx.setLineDash([]);
}

// Main loop
let lastTime = 0;
function gameLoop(timestamp) {
    const dt = timestamp - lastTime;
    lastTime = timestamp;

    handleInput();

    if (gameState === 'COUNTDOWN') {
        countdownTimer--;
        if (countdownTimer <= 0) gameState = 'PLAYING';
    }

    if (gameState === 'PLAYING') {
        updateBall();
    }

    if (gameState === 'SCORED') {
        scoredTimer--;
        if (scoredTimer <= 0) {
            resetBall(lastScorer === 1 ? -1 : 1);
            gameState = 'PLAYING';
        }
    }

    updateParticles();

    // Render
    ctx.fillStyle = '#0a0a1a';
    ctx.fillRect(0, 0, W, H);

    drawCenterLine();
    drawGoals();
    drawMaze();
    drawPaddles();
    drawBall();
    drawParticles();
    drawScore();

    if (gameState === 'START') drawStartScreen();
    if (gameState === 'COUNTDOWN') drawCountdown();
    if (gameState === 'GAMEOVER') drawGameOver();

    // Mode indicator
    if (gameState === 'PLAYING' || gameState === 'SCORED') {
        ctx.font = `${Math.min(W*0.02, 14)}px monospace`;
        ctx.textAlign = 'left';
        ctx.fillStyle = 'rgba(255,255,255,0.3)';
        ctx.fillText(gameMode === 'ai' ? '1P vs AI' : '2P', 10, H - 10);
    }

    requestAnimationFrame(gameLoop);
}

// Init
initSizes();
p1.y = H/2 - paddleH/2;
p2.x = W - 30 - paddleW;
p2.y = H/2 - paddleH/2;
buildMaze();
resetBall(1);
p1.x = 30;
p2.x = W - 30 - paddleW;

window.addEventListener('resize', () => {
    p1.x = 30;
    p2.x = W - 30 - paddleW;
    buildMaze();
});

requestAnimationFrame(gameLoop);
})();
