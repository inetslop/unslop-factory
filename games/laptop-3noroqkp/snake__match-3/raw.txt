const canvas = document.getElementById('game');
const ctx = canvas.getContext('2d');

let W, H, cellSize, gridX, gridY, gridSize = 8;
let gems = [];
let snakeHead = {x: 3, y: 6};
let snakeBody = [];
let score = 0;
let timer = 60;
let lastTime = 0;
let state = 'start'; // start, playing, gameover
let particles = [];
let comboMultiplier = 1;
let animating = false;
let shakeTime = 0;
let flashCells = [];

const GEM_COLORS = [
    {main: '#ff4466', light: '#ff8899', dark: '#cc2244'},
    {main: '#44aaff', light: '#88ccff', dark: '#2277cc'},
    {main: '#44ff88', light: '#88ffbb', dark: '#22cc66'},
    {main: '#ffdd44', light: '#ffee88', dark: '#ccaa22'}
];

function resize() {
    W = window.innerWidth;
    H = window.innerHeight;
    canvas.width = W;
    canvas.height = H;
    cellSize = Math.min(Math.floor((W - 40) / gridSize), Math.floor((H - 120) / gridSize));
    gridX = Math.floor((W - cellSize * gridSize) / 2);
    gridY = Math.floor((H - cellSize * gridSize) / 2) + 30;
}
window.addEventListener('resize', resize);
resize();

function initGems() {
    gems = [];
    for (let y = 0; y < gridSize; y++) {
        gems[y] = [];
        for (let x = 0; x < gridSize; x++) {
            gems[y][x] = Math.floor(Math.random() * GEM_COLORS.length);
        }
    }
    // Remove initial matches
    let hasMatch = true;
    while (hasMatch) {
        hasMatch = false;
        for (let y = 0; y < gridSize; y++) {
            for (let x = 0; x < gridSize; x++) {
                if (x + 2 < gridSize && gems[y][x] === gems[y][x+1] && gems[y][x] === gems[y][x+2]) {
                    gems[y][x+2] = Math.floor(Math.random() * GEM_COLORS.length);
                    hasMatch = true;
                }
                if (y + 2 < gridSize && gems[y][x] === gems[y+1][x] && gems[y][x] === gems[y+2][x]) {
                    gems[y+2][x] = Math.floor(Math.random() * GEM_COLORS.length);
                    hasMatch = true;
                }
            }
        }
    }
}

function startGame() {
    initGems();
    snakeHead = {x: 3, y: 6};
    snakeBody = [];
    score = 0;
    timer = 60;
    comboMultiplier = 1;
    particles = [];
    state = 'playing';
    lastTime = performance.now();
}

function getRunAt(x, y) {
    let color = gems[y][x];
    let runs = [];
    
    // Horizontal run
    let hStart = x;
    while (hStart > 0 && gems[y][hStart-1] === color) hStart--;
    let hEnd = x;
    while (hEnd < gridSize-1 && gems[y][hEnd+1] === color) hEnd++;
    if (hEnd - hStart + 1 >= 3) {
        for (let i = hStart; i <= hEnd; i++) runs.push({x: i, y: y});
    }
    
    // Vertical run
    let vStart = y;
    while (vStart > 0 && gems[vStart-1][x] === color) vStart--;
    let vEnd = y;
    while (vEnd < gridSize-1 && gems[vEnd+1][x] === color) vEnd++;
    if (vEnd - vStart + 1 >= 3) {
        for (let i = vStart; i <= vEnd; i++) runs.push({x: x, y: i});
    }
    
    return runs;
}

function findBoardMatches() {
    let matches = [];
    // Horizontal
    for (let y = 0; y < gridSize; y++) {
        for (let x = 0; x < gridSize - 2; x++) {
            if (gems[y][x] !== null && gems[y][x] === gems[y][x+1] && gems[y][x] === gems[y][x+2]) {
                let set = new Set();
                for (let i = x; i < gridSize && gems[y][i] === gems[y][x]; i++) set.add(y * gridSize + i);
                set.forEach(v => matches.push({x: v % gridSize, y: Math.floor(v / gridSize)}));
            }
        }
    }
    // Vertical
    for (let x = 0; x < gridSize; x++) {
        for (let y = 0; y < gridSize - 2; y++) {
            if (gems[y][x] !== null && gems[y][x] === gems[y+1][x] && gems[y][x] === gems[y+2][x]) {
                let set = new Set();
                for (let i = y; i < gridSize && gems[i][x] === gems[y][x]; i++) set.add(i * gridSize + x);
                set.forEach(v => matches.push({x: v % gridSize, y: Math.floor(v / gridSize)}));
            }
        }
    }
    // Deduplicate
    let seen = new Set();
    return matches.filter(m => {
        let key = m.x + ',' + m.y;
        if (seen.has(key)) return false;
        seen.add(key);
        return true;
    });
}

function clearMatches(matches, isCascade) {
    if (matches.length === 0) return false;
    
    let cleared = new Set();
    matches.forEach(m => {
        cleared.add(m.x + ',' + m.y);
        flashCells.push({x: m.x, y: m.y, time: 1});
        spawnParticles(m.x, m.y, gems[m.y][m.x]);
    });
    
    let points = cleared.size * 10 * comboMultiplier;
    score += points;
    
    if (isCascade) {
        comboMultiplier++;
    } else {
        comboMultiplier = 1;
    }
    
    shakeTime = 0.2;
    
    cleared.forEach(key => {
        let [x, y] = key.split(',').map(Number);
        gems[y][x] = null;
    });
    
    return true;
}

function applyGravity() {
    for (let x = 0; x < gridSize; x++) {
        let writeY = gridSize - 1;
        for (let y = gridSize - 1; y >= 0; y--) {
            if (gems[y][x] !== null) {
                gems[writeY][x] = gems[y][x];
                if (writeY !== y) gems[y][x] = null;
                writeY--;
            }
        }
        for (let y = writeY; y >= 0; y--) {
            gems[y][x] = null;
        }
    }
}

function refillGems() {
    for (let x = 0; x < gridSize; x++) {
        for (let y = 0; y < gridSize; y++) {
            if (gems[y][x] === null) {
                gems[y][x] = Math.floor(Math.random() * GEM_COLORS.length);
            }
        }
    }
}

function spawnParticles(gx, gy, colorIdx) {
    let cx = gridX + gx * cellSize + cellSize / 2;
    let cy = gridY + gy * cellSize + cellSize / 2;
    for (let i = 0; i < 6; i++) {
        let angle = Math.random() * Math.PI * 2;
        let speed = 50 + Math.random() * 100;
        particles.push({
            x: cx, y: cy,
            vx: Math.cos(angle) * speed,
            vy: Math.sin(angle) * speed - 50,
            life: 0.8,
            color: GEM_COLORS[colorIdx].main,
            size: cellSize * 0.15 + Math.random() * cellSize * 0.1
        });
    }
}

function moveSnake(dx, dy) {
    if (state !== 'playing' || animating) return;
    
    let nx = snakeHead.x + dx;
    let ny = snakeHead.y + dy;
    
    if (nx < 0 || nx >= gridSize || ny < 0 || ny >= gridSize) return;
    
    // Check if moving into body (small penalty - just block)
    for (let seg of snakeBody) {
        if (seg.x === nx && seg.y === ny) return;
    }
    
    snakeBody.unshift({x: snakeHead.x, y: snakeHead.y});
    if (snakeBody.length > 5) snakeBody.pop();
    
    snakeHead = {x: nx, y: ny};
    
    // Check for match at new position
    let runs = getRunAt(nx, ny);
    if (runs.length > 0) {
        animating = true;
        comboMultiplier = 1;
        
        processMatches(runs, false);
    }
}

function processMatches(matches, isCascade) {
    let cleared = clearMatches(matches, isCascade);
    if (!cleared) {
        animating = false;
        return;
    }
    
    setTimeout(() => {
        applyGravity();
        refillGems();
        
        // Check for cascades
        let newMatches = findBoardMatches();
        if (newMatches.length > 0) {
            processMatches(newMatches, true);
        } else {
            animating = false;
        }
    }, 250);
}

let audioCtx = null;
function initAudio() {
    if (!audioCtx) {
        audioCtx = new (window.AudioContext || window.webkitAudioContext)();
    }
}

function playMatchSound(combo) {
    if (!audioCtx) return;
    let osc = audioCtx.createOscillator();
    let gain = audioCtx.createGain();
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.type = 'sine';
    osc.frequency.value = 300 + combo * 100;
    gain.gain.setValueAtTime(0.15, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.2);
    osc.start(audioCtx.currentTime);
    osc.stop(audioCtx.currentTime + 0.2);
}

function playMoveSound() {
    if (!audioCtx) return;
    let osc = audioCtx.createOscillator();
    let gain = audioCtx.createGain();
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.type = 'triangle';
    osc.frequency.value = 200;
    gain.gain.setValueAtTime(0.05, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.08);
    osc.start(audioCtx.currentTime);
    osc.stop(audioCtx.currentTime + 0.08);
}

window.addEventListener('keydown', (e) => {
    if (!audioCtx) initAudio();
    
    if (state === 'start' || state === 'gameover') {
        if (e.key === 'Enter' || e.key === ' ') startGame();
        return;
    }
    
    if (state !== 'playing') return;
    
    let dx = 0, dy = 0;
    if (e.key === 'ArrowRight' || e.key === 'd' || e.key === 'D') dx = 1;
    else if (e.key === 'ArrowLeft' || e.key === 'a' || e.key === 'A') dx = -1;
    else if (e.key === 'ArrowDown' || e.key === 's' || e.key === 'S') dy = 1;
    else if (e.key === 'ArrowUp' || e.key === 'w' || e.key === 'W') dy = -1;
    else return;
    
    e.preventDefault();
    playMoveSound();
    moveSnake(dx, dy);
});

function drawGem(x, y, colorIdx, scale) {
    if (colorIdx === null || colorIdx === undefined) return;
    let px = gridX + x * cellSize;
    let py = gridY + y * cellSize;
    let s = cellSize * (scale || 1);
    let ox = (cellSize - s) / 2;
    
    let color = GEM_COLORS[colorIdx];
    let r = s * 0.2;
    
    ctx.save();
    ctx.beginPath();
    ctx.roundRect(px + ox + 2, py + ox + 2, s - 4, s - 4, r);
    ctx.fillStyle = color.dark;
    ctx.fill();
    
    ctx.beginPath();
    ctx.roundRect(px + ox + 4, py + ox + 4, s - 8, s - 8, r - 2);
    ctx.fillStyle = color.main;
    ctx.fill();
    
    // Highlight
    let grad = ctx.createLinearGradient(px + ox, py + ox, px + ox + s, py + ox + s);
    grad.addColorStop(0, 'rgba(255,255,255,0.3)');
    grad.addColorStop(0.5, 'rgba(255,255,255,0)');
    ctx.beginPath();
    ctx.roundRect(px + ox + 4, py + ox + 4, s - 8, (s-8)/2, r - 2);
    ctx.fillStyle = grad;
    ctx.fill();
    
    ctx.restore();
}

function drawSnake() {
    // Body
    for (let i = snakeBody.length - 1; i >= 0; i--) {
        let seg = snakeBody[i];
        let alpha = 1 - (i / snakeBody.length) * 0.7;
        let size = cellSize * 0.6 * (1 - i / snakeBody.length * 0.3);
        let cx = gridX + seg.x * cellSize + cellSize / 2;
        let cy = gridY + seg.y * cellSize + cellSize / 2;
        
        ctx.beginPath();
        ctx.arc(cx, cy, size / 2, 0, Math.PI * 2);
        ctx.fillStyle = `rgba(255, 255, 100, ${alpha * 0.5})`;
        ctx.fill();
    }
    
    // Head
    let hx = gridX + snakeHead.x * cellSize;
    let hy = gridY + snakeHead.y * cellSize;
    let hcx = hx + cellSize / 2;
    let hcy = hy + cellSize / 2;
    
    // Glow
    ctx.beginPath();
    ctx.arc(hcx, hcy, cellSize * 0.45, 0, Math.PI * 2);
    ctx.fillStyle = 'rgba(255, 255, 100, 0.3)';
    ctx.fill();
    
    // Head circle
    ctx.beginPath();
    ctx.arc(hcx, hcy, cellSize * 0.35, 0, Math.PI * 2);
    ctx.fillStyle = '#fff';
    ctx.fill();
    ctx.beginPath();
    ctx.arc(hcx, hcy, cellSize * 0.3, 0, Math.PI * 2);
    ctx.fillStyle = '#ffe066';
    ctx.fill();
    
    // Eyes
    let eyeOff = cellSize * 0.1;
    ctx.beginPath();
    ctx.arc(hcx - eyeOff, hcy - eyeOff * 0.5, cellSize * 0.06, 0, Math.PI * 2);
    ctx.fillStyle = '#333';
    ctx.fill();
    ctx.beginPath();
    ctx.arc(hcx + eyeOff, hcy - eyeOff * 0.5, cellSize * 0.06, 0, Math.PI * 2);
    ctx.fillStyle = '#333';
    ctx.fill();
}

function drawParticles(dt) {
    for (let i = particles.length - 1; i >= 0; i--) {
        let p = particles[i];
        p.x += p.vx * dt;
        p.y += p.vy * dt;
        p.vy += 300 * dt;
        p.life -= dt;
        
        if (p.life <= 0) {
            particles.splice(i, 1);
            continue;
        }
        
        ctx.beginPath();
        ctx.arc(p.x, p.y, p.size * (p.life / 0.8), 0, Math.PI * 2);
        ctx.fillStyle = p.color + Math.floor(p.life / 0.8 * 255).toString(16).padStart(2, '0');
        ctx.fill();
    }
}

function drawGrid() {
    // Grid background
    ctx.fillStyle = '#1a1a3e';
    ctx.beginPath();
    ctx.roundRect(gridX - 8, gridY - 8, cellSize * gridSize + 16, cellSize * gridSize + 16, 8);
    ctx.fill();
    
    // Grid lines
    ctx.strokeStyle = 'rgba(255,255,255,0.05)';
    ctx.lineWidth = 1;
    for (let i = 0; i <= gridSize; i++) {
        ctx.beginPath();
        ctx.moveTo(gridX + i * cellSize, gridY);
        ctx.lineTo(gridX + i * cellSize, gridY + gridSize * cellSize);
        ctx.stroke();
        ctx.beginPath();
        ctx.moveTo(gridX, gridY + i * cellSize);
        ctx.lineTo(gridX + gridSize * cellSize, gridY + i * cellSize);
        ctx.stroke();
    }
    
    // Gems
    for (let y = 0; y < gridSize; y++) {
        for (let x = 0; x < gridSize; x++) {
            if (gems[y][x] !== null) {
                drawGem(x, y, gems[y][x]);
            }
        }
    }
    
    // Flash cells
    for (let i = flashCells.length - 1; i >= 0; i--) {
        let fc = flashCells[i];
        fc.time -= 1/60;
        if (fc.time <= 0) {
            flashCells.splice(i, 1);
            continue;
        }
        ctx.fillStyle = `rgba(255,255,255,${fc.time * 0.5})`;
        ctx.fillRect(gridX + fc.x * cellSize, gridY + fc.y * cellSize, cellSize, cellSize);
    }
}

function drawHUD() {
    // Score
    ctx.font = `bold ${Math.max(16, cellSize * 0.4)}px monospace`;
    ctx.fillStyle = '#fff';
    ctx.textAlign = 'left';
    ctx.fillText(`Score: ${score}`, gridX, gridY - 15);
    
    // Timer
    ctx.textAlign = 'right';
    let timerColor = timer < 10 ? '#ff4466' : '#fff';
    ctx.fillStyle = timerColor;
    ctx.fillText(`Time: ${Math.ceil(timer)}`, gridX + cellSize * gridSize, gridY - 15);
    
    // Combo indicator
    if (comboMultiplier > 1) {
        ctx.textAlign = 'center';
        ctx.fillStyle = '#ffdd44';
        ctx.font = `bold ${cellSize * 0.35}px monospace`;
        ctx.fillText(`x${comboMultiplier} COMBO!`, W / 2, gridY - 15);
    }
}

    ctx.fillStyle = '#ffe066';
    ctx.fillText(`Final Score: ${score}`, cx, cy);
    
    let rank = score > 1000 ? 'LEGENDARY!' : score > 500 ? 'GREAT!' : score > 200 ? 'NICE!' : 'GOOD TRY!';
    ctx.font = `${Math.min(20, W * 0.03)}px monospace`;
    ctx.fillStyle = '#44ff88';
    ctx.fillText(rank, cx, cy + 50);
    
    if (Math.floor(performance.now() / 500) % 2 === 0) {
        ctx.fillStyle = '#fff';
        ctx.font = `bold ${Math.min(18, W * 0.025)}px monospace`;
        ctx.fillText('Press ENTER to Play Again', cx, cy + 120);
    }
}

function update(dt) {
    if (state !== 'playing') return;
    
    timer -= dt;
    if (timer <= 0) {
        timer = 0;
        state = 'gameover';
    }
    
    // Shake decay
    if (shakeTime > 0) shakeTime -= dt;
}

function render(dt) {
    ctx.clearRect(0, 0, W, H);
    
    if (state === 'start') {
        drawStartScreen();
        return;
    }
    
    // Apply screen shake
    let sx = 0, sy = 0;
    if (shakeTime > 0) {
        sx = (Math.random() - 0.5) * shakeTime * 20;
        sy = (Math.random() - 0.5) * shakeTime * 20;
    }
    
    ctx.save();
    ctx.translate(sx, sy);
    
    drawGrid();
    drawSnake();
    drawParticles(dt);
    drawHUD();
    
    ctx.restore();
    
    if (state === 'gameover') {
        drawGameOver();
    }
}

function gameLoop(timestamp) {
    let dt = Math.min((timestamp - lastTime) / 1000, 0.05);
    lastTime = timestamp;
    
    update(dt);
    render(dt);
    
    requestAnimationFrame(gameLoop);
}

// Polyfill for roundRect
if (!CanvasRenderingContext2D.prototype.roundRect) {
    CanvasRenderingContext2D.prototype.roundRect = function(x, y, w, h, r) {
        if (typeof r === 'number') r = [r, r, r, r];
        let [tl, tr, br, bl] = Array.isArray(r) ? r : [r, r, r, r];
        this.moveTo(x + tl, y);
        this.lineTo(x + w - tr, y);
        this.quadraticCurveTo(x + w, y, x + w, y + tr);
        this.lineTo(x + w, y + h - br);
        this.quadraticCurveTo(x + w, y + h, x + w - br, y + h);
        this.lineTo(x + bl, y + h);
        this.quadraticCurveTo(x, y + h, x, y + h - bl);
        this.lineTo(x, y + tl);
        this.quadraticCurveTo(x, y, x + tl, y);
        this.closePath();
    };
}

// Play match sound via observer
let lastScore = 0;
setInterval(() => {
    if (state === 'playing' && score !== lastScore) {
        playMatchSound(comboMultiplier);
        lastScore = score;
    }
}, 100);

lastTime = performance.now();
requestAnimationFrame(gameLoop);
