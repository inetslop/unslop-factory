function playSound(freq, dur, type='sine', vol=0.15) {
  if (!audioCtx) return;
  const osc = audioCtx.createOscillator();
  const gain = audioCtx.createGain();
  osc.type = type;
  osc.frequency.value = freq;
  gain.gain.setValueAtTime(vol, audioCtx.currentTime);
  gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
  osc.connect(gain);
  gain.connect(audioCtx.destination);
  osc.start();
  osc.stop(audioCtx.currentTime + dur);
}

  enemyPlay();
  resolveTurn();
  
  selectedCard = -1;

function playCard(col, row) {
  if(selectedCard < 0 || selectedCard >= hand.length) return;
  const cardName = hand[selectedCard];
  const c = CARD_TYPES[cardName];
  if(playerMana < c.cost) { addMessage('Not enough mana!','#ff6'); return; }
  
  if(c.type === 'unit') {
    if(row >= 4 && !grid[row][col]) {
      grid[row][col] = { type:cardName, atk:c.atk, hp:c.hp, maxHp:c.hp, side:'player', color:c.color };
      playerMana -= c.cost;
      hand.splice(selectedCard, 1);
      selectedCard = -1;
      addMessage(c.name + ' deployed!','#4a90d9');
      sfxPlace();
      nextTurn();
    } else {
      addMessage('Invalid position!','#f66');
    }
  } else if(cardName === 'fireball') {
    // Target: find an enemy unit in range (rows 0-3)
    let target = null;
    for(let r=0;r<4;r++) {
      for(let cc=0;cc<COLS;cc++) {
        if(grid[r][cc] && grid[r][cc].side==='enemy') { target={r,c:cc}; break; }
      }
      if(target) break;
    }
    if(!target) { addMessage('No enemy in range!','#f66'); return; }
    playerMana -= c.cost;
    grid[target.r][target.c].hp -= c.atk;
    addParticles(gridX+target.c*cellW+cellW/2, gridY+target.r*cellH+cellH/2, '#ff4400', 15);
    if(grid[target.r][target.c].hp <= 0) {
      grid[target.r][target.c] = null;
      sfxDeath();
    }
    hand.splice(selectedCard, 1);
    selectedCard = -1;
    addMessage('Fireball!','#ff4444');
    sfxSpell();
    nextTurn();
  } else if(cardName === 'heal') {
    // Heal lowest HP player unit
    let best = null;
    for(let r=3;r<ROWS;r++) {
      for(let cc=0;cc<COLS;cc++) {
        const u = grid[r][cc];
        if(u && u.side==='player' && u.hp < u.maxHp) {
          if(!best || u.hp/u.maxHp < best.u.hp/best.u.maxHp) best={r:r,c:cc,u:u};
        }
      }
    }
    if(!best) { addMessage('No unit to heal!','#f66'); return; }
    playerMana -= c.cost;
    best.u.hp = Math.min(best.u.maxHp, best.u.hp + 5);
    addParticles(gridX+best.c*cellW+cellW/2, gridY+best.r*cellH+cellH/2, '#44ff88', 10);
    hand.splice(selectedCard, 1);
    selectedCard = -1;
    addMessage('Healed!','#4f8');
    sfxSpell();
    nextTurn();
  }
}
