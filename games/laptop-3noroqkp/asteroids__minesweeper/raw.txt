<script>
(function(){
const canvas = document.getElementById('gameCanvas');
const ctx = canvas.getContext('2d');
let W, H;
let audioCtx = null;
let gameStarted = false;
let gameOver = false;
let won = false;
let timerInterval = null;
let seconds = 0;

const ROWS = 12, COLS = 12, MINES = 24;
let grid = [];
let cellSize, offsetX, offsetY;
let stars = [];
let particles = [];
let shakeTimer = 0;
let shakeIntensity = 0;

function resize(){
W = window.innerWidth; H = window.innerHeight;
canvas.width = W; canvas.height = H;
cellSize = Math.min((W - 40) / COLS, (H - 80) / ROWS);
offsetX = (W - cellSize * COLS) / 2;
offsetY = (H - cellSize * ROWS) / 2 + 20;
}
window.addEventListener('resize', resize);
resize();

function initStars(){
stars = [];
for(let i=0;i<150;i++){
stars.push({x:Math.random()*W, y:Math.random()*H, r:Math.random()*1.5+0.3, speed:Math.random()*0.3+0.05, twinkle:Math.random()*Math.PI*2});
}
}

function initAudio(){
if(!audioCtx) audioCtx = new (window.AudioContext||webkitAudioContext)();
}

function playTone(freq, dur, type='sine', vol=0.15){
if(!audioCtx) return;
const osc = audioCtx.createOscillator();
const gain = audioCtx.createGain();
osc.type = type;
osc.frequency.value = freq;
gain.gain.setValueAtTime(vol, audioCtx.currentTime);
gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
osc.connect(gain); gain.connect(audioCtx.destination);
osc.start(); osc.stop(audioCtx.currentTime + dur);
}

function playReveal(){ playTone(400+Math.random()*200, 0.1, 'sine', 0.1); }
function playFlag(){ playTone(600, 0.08, 'square', 0.08); }
function playExplosion(){
if(!audioCtx) return;
const buf = audioCtx.createBufferSource();
const buffer = audioCtx.createBuffer(1, audioCtx.sampleRate*0.5, audioCtx.sampleRate);
const data = buffer.getChannelData(0);
for(let i=0;i<data.length;i++) data[i] = (Math.random()*2-1)*Math.pow(1-i/data.length, 3);
buf.buffer = buffer;
const gain = audioCtx.createGain();
gain.gain.value = 0.4;
buf.connect(gain); gain.connect(audioCtx.destination);
buf.start();
}
function playWin(){
[523,659,784,1047].forEach((f,i)=>setTimeout(()=>playTone(f,0.3,'sine',0.15),i*120));
}

function initGrid(){
grid = [];
for(let r=0;r<ROWS;r++){
grid[r]=[];
for(let c=0;c<COLS;c++){
grid[r][c] = {mine:false, revealed:false, flagged:false, count:0};
}
}
let placed = 0;
while(placed < MINES){
const r = Math.floor(Math.random()*ROWS);
const c = Math.floor(Math.random()*COLS);
if(!grid[r][c].mine){ grid[r][c].mine = true; placed++; }
}
for(let r=0;r<ROWS;r++) for(let c=0;c<COLS;c++){
let n=0;
for(let dr=-1;dr<=1;dr++) for(let dc=-1;dc<=1;dc++){
if(dr===0&&dc===0) continue;
const nr=r+dr, nc=c+dc;
if(nr>=0&&nr<ROWS&&nc>=0&&nc<COLS&&grid[nr][nc].mine) n++;
}
grid[r][c].count = n;
}
document.getElementById('minesLeft').textContent = MINES;
}

function startGame(){
gameStarted = true; gameOver = false; won = false; seconds = 0;
initGrid();
particles = [];
shakeTimer = 0;
document.getElementById('startScreen').style.display = 'none';
document.getElementById('hud').style.display = 'flex';
const msg = document.getElementById('message');
msg.classList.remove('show','win');
if(timerInterval) clearInterval(timerInterval);
timerInterval = setInterval(()=>{ if(!gameOver){ seconds++; document.getElementById('timer').textContent=seconds; } }, 1000);
document.getElementById('timer').textContent = '0';
}

function floodReveal(sr, sc){
const stack = [[sr,sc]];
while(stack.length){
const [r,c] = stack.pop();
if(r<0||r>=ROWS||c<0||c>=COLS) continue;
if(grid[r][c].revealed || grid[r][c].flagged) continue;
grid[r][c].revealed = true;
playReveal();
if(grid[r][c].mine){ triggerExplosion(r,c); return; }
if(grid[r][c].count === 0){
for(let dr=-1;dr<=1;dr++) for(let dc=-1;dc<=1;dc++){
if(dr===0&&dc===0) continue;
stack.push([r+dr, c+dc]);
}
}
}
checkWin();
}

function triggerExplosion(r,c){
gameOver = true;
won = false;
playExplosion();
shakeTimer = 20; shakeIntensity = 8;
for(let i=0;i<30;i++){
const angle = Math.random()*Math.PI*2;
const speed = Math.random()*4+1;
particles.push({x:offsetX+c*cellSize+cellSize/2, y:offsetY+r*cellSize+cellSize/2, vx:Math.cos(angle)*speed, vy:Math.sin(angle)*speed, life:60+Math.random()*30, color:`hsl(${Math.random()*40},100%,${50+Math.random()*30}%)`});
}
for(let rr=0;rr<ROWS;rr++) for(let cc=0;cc<COLS;cc++) grid[rr][cc].revealed = true;
const msg = document.getElementById('message');
msg.textContent = '💥 ASTEROID IMPACT!';
msg.classList.add('show');
msg.classList.remove('win');
if(timerInterval) clearInterval(timerInterval);
}

function checkWin(){
let safe = 0, total = ROWS*COLS - MINES;
for(let r=0;r<ROWS;r++) for(let c=0;c<COLS;c++){
if(!grid[r][c].mine && grid[r][c].revealed) safe++;
}
if(safe === total){
gameOver = true; won = true;
playWin();
const msg = document.getElementById('message');
msg.textContent = '✓ SECTOR CLEARED!';
msg.classList.add('show','win');
if(timerInterval) clearInterval(timerInterval);
}
}

function countFlags(){
let f=0; for(let r=0;r<ROWS;r++) for(let c=0;c<COLS;c++) if(grid[r][c].flagged) f++;
document.getElementById('minesLeft').textContent = MINES - f;
}

function handleClick(r,c){
if(gameOver || !gameStarted) return;
initAudio();
if(grid[r][c].flagged) return;
floodReveal(r,c);
}

function handleRightClick(r,c){
if(gameOver || !gameStarted) return;
initAudio();
if(grid[r][c].revealed) return;
grid[r][c].flagged = !grid[r][c].flagged;
playFlag();
countFlags();
}

canvas.addEventListener('contextmenu', e=>e.preventDefault());

const rect = canvas.getBoundingClientRect();
const mx = e.clientX - rect.left, my = e.clientY - rect.top;
const c = Math.floor((mx - offsetX) / cellSize);
const r = Math.floor((my - offsetY) / cellSize);
if(r>=0&&r<ROWS&&c>=0&&c<COLS) handleClick(r,c);
});
canvas.addEventListener('contextmenu', e=>{
e.preventDefault();
if(!gameStarted||gameOver) return;
const rect = canvas.getBoundingClientRect();
const mx = e.clientX - rect.left, my = e.clientY - rect.top;
const c = Math.floor((mx - offsetX) / cellSize);
const r = Math.floor((my - offsetY) / cellSize);
if(r>=0&&r<ROWS&&c>=0&&c<COLS) handleRightClick(r,c);
});

document.getElementById('startBtn').addEventListener('click', ()=>{ initAudio(); startGame(); });
document.getElementById('restartBtn').addEventListener('click', ()=>{ initAudio(); startGame(); });
window.addEventListener('keydown', e=>{
if(e.key==='r'||e.key==='R'){ if(gameStarted) startGame(); }
});

function drawStar(x,y,r,alpha){
ctx.beginPath(); ctx.arc(x,y,r,0,Math.PI*2);
ctx.fillStyle = `rgba(200,220,255,${alpha})`; ctx.fill();
}

function drawBackground(t){
ctx.fillStyle = '#0a0a1a';
ctx.fillRect(0,0,W,H);
for(let s of stars){
s.twinkle += 0.02;
const a = 0.3 + Math.sin(s.twinkle)*0.4;
drawStar(s.x, s.y, s.r, a);
}
// Nebula glow
const grd = ctx.createRadialGradient(W*0.7,H*0.3,0,W*0.7,H*0.3,W*0.4);
grd.addColorStop(0,'rgba(60,20,80,0.15)');
grd.addColorStop(1,'rgba(0,0,0,0)');
ctx.fillStyle = grd; ctx.fillRect(0,0,W,H);
const grd2 = ctx.createRadialGradient(W*0.2,H*0.7,0,W*0.2,H*0.7,W*0.3);
grd2.addColorStop(0,'rgba(20,40,80,0.1)');
grd2.addColorStop(1,'rgba(0,0,0,0)');
ctx.fillStyle = grd2; ctx.fillRect(0,0,W,H);
}

function drawGrid(t){
for(let r=0;r<ROWS;r++) for(let c=0;c<COLS;c++){
const x = offsetX + c*cellSize, y = offsetY + r*cellSize;
const cell = grid[r][c];
if(cell.revealed && !cell.mine){
// Revealed safe - dark space tile
ctx.fillStyle = 'rgba(15,20,45,0.7)';
ctx.fillRect(x+1,y+1,cellSize-2,cellSize-2);
ctx.strokeStyle = 'rgba(40,60,100,0.3)';
ctx.lineWidth = 1;
ctx.strokeRect(x+1,y+1,cellSize-2,cellSize-2);
if(cell.count > 0){
const colors = ['#fff','#ff8844','#ffcc44','#ff66aa','#44aaff','#44ffaa','#ff4444','#ffffff'];
ctx.fillStyle = colors[cell.count-1];
ctx.font = `bold ${cellSize*0.5}px monospace`;
ctx.textAlign = 'center'; ctx.textBaseline = 'middle';
ctx.fillText(cell.count, x+cellSize/2, y+cellSize/2);
}
} else if(cell.revealed && cell.mine){
// Revealed mine - dangerous asteroid
ctx.fillStyle = '#1a0a0a';
ctx.fillRect(x+1,y+1,cellSize-2,cellSize-2);
ctx.strokeStyle = 'rgba(255,50,30,0.6)';
ctx.lineWidth = 1;
ctx.strokeRect(x+1,y+1,cellSize-2,cellSize-2);
// Asteroid shape
const cx=x+cellSize/2, cy=y+cellSize/2;
const pulse = 0.8 + Math.sin(t*0.05+c*3+r*7)*0.2;
ctx.beginPath();
for(let i=0;i<8;i++){
const a = (i/8)*Math.PI*2;
const rad = cellSize*0.3*pulse*(0.7+Math.sin(a*3)*0.3);
const px = cx + Math.cos(a)*rad, py = cy + Math.sin(a)*rad;
if(i===0) ctx.moveTo(px,py); else ctx.lineTo(px,py);
}
ctx.closePath();
ctx.fillStyle = `rgba(255,60,20,${0.7+Math.sin(t*0.08)*0.3})`;
ctx.fill();
ctx.strokeStyle = '#ff4400';
ctx.lineWidth = 1.5;
ctx.stroke();
// Glow
const grd = ctx.createRadialGradient(cx,cy,0,cx,cy,cellSize*0.4);
grd.addColorStop(0,'rgba(255,80,20,0.3)');
grd.addColorStop(1,'rgba(0,0,0,0)');
ctx.fillStyle = grd;
ctx.fillRect(x,y,cellSize,cellSize);
} else if(cell.flagged){
// Flagged - beacon marker
ctx.fillStyle = 'rgba(20,30,60,0.85)';
ctx.fillRect(x+1,y+1,cellSize-2,cellSize-2);
ctx.strokeStyle = 'rgba(80,120,200,0.5)';
ctx.lineWidth = 1;
ctx.strokeRect(x+1,y+1,cellSize-2,cellSize-2);
const cx=x+cellSize/2, cy=y+cellSize/2;
// Beacon
ctx.beginPath(); ctx.moveTo(cx,cy-cellSize*0.3); ctx.lineTo(cx+cellSize*0.15,cy+cellSize*0.2); ctx.lineTo(cx-cellSize*0.15,cy+cellSize*0.2); ctx.closePath();
ctx.fillStyle = '#4488ff'; ctx.fill();
ctx.strokeStyle = '#66aaff'; ctx.lineWidth=1; ctx.stroke();
const glow = 0.3+Math.sin(t*0.06)*0.2;
const grd = ctx.createRadialGradient(cx,cy,0,cx,cy,cellSize*0.3);
grd.addColorStop(0,`rgba(80,140,255,${glow})`);
grd.addColorStop(1,'rgba(0,0,0,0)');
ctx.fillStyle = grd; ctx.fillRect(x,y,cellSize,cellSize);
} else {
// Hidden - asteroid surface
const shade = 0.85 + Math.sin(c*2.1+r*3.7)*0.08;
ctx.fillStyle = `rgba(40,50,80,${shade})`;
ctx.fillRect(x+1,y+1,cellSize-2,cellSize-2);
// Cracks/texture
ctx.strokeStyle = 'rgba(60,80,120,0.3)';
ctx.lineWidth = 0.5;
const seed = r*7+c*13;
for(let i=0;i<2;i++){
ctx.beginPath();
ctx.moveTo(x+((seed*17+i*31)%cellSize), y+((seed*23+i*47)%cellSize));
ctx.lineTo(x+((seed*31+i*13)%cellSize), y+((seed*43+i*29)%cellSize));
ctx.stroke();
}
ctx.strokeStyle = 'rgba(80,100,150,0.4)';
ctx.lineWidth = 1;
ctx.strokeRect(x+1,y+1,cellSize-2,cellSize-2);
// Highlight edge
ctx.fillStyle = 'rgba(100,130,180,0.1)';
ctx.fillRect(x+2,y+2,cellSize-4,2);
}
}
}

function drawParticles(){
for(let i=particles.length-1;i>=0;i--){
const p = particles[i];
p.x += p.vx; p.y += p.vy;
p.vy += 0.05;
p.life--;
if(p.life<=0){ particles.splice(i,1); continue; }
ctx.beginPath();
ctx.arc(p.x,p.y,2+Math.random()*2,0,Math.PI*2);
ctx.fillStyle = p.color;
ctx.globalAlpha = p.life/60;
ctx.fill();
ctx.globalAlpha = 1;
}
}

function draw(){
const t = performance.now();
let sx=0, sy=0;
if(shakeTimer>0){
shakeTimer--;
sx = (Math.random()-0.5)*shakeIntensity*(shakeTimer/20);
sy = (Math.random()-0.5)*shakeIntensity*(shakeTimer/20);
}
ctx.save();
ctx.translate(sx,sy);
drawBackground(t);
if(gameStarted){
drawGrid(t);
drawParticles();
}
ctx.restore();
requestAnimationFrame(draw);
}

initStars();
resize();
draw();
})();
