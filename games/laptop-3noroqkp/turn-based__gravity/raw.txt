(function() {

const ctx = canvas.getContext('2d');

const GRID_SIZE = 7;
const COLORS = ['#ff4466','#44aaff','#44ff88','#ffcc44','#cc66ff'];
const COLOR_NAMES = ['red','blue','green','yellow','purple'];
const TARGET_SCORES = [100, 250, 500, 900, 1400, 2000, 3000, 4500, 6500, 9000];
const MOVES_PER_LEVEL = 25;

let state = 'start'; // start, playing, animating, levelComplete, gameOver
let grid = [];
let score = 0;
let targetScore = 100;
let movesLeft = MOVES_PER_LEVEL;
let level = 1;
let combo = 0;
let particles = [];
let animQueue = [];
let shakeTimer = 0;
let shakeIntensity = 0;
let floatTexts = [];

function initGrid() {
grid = [];
for (let r = 0; r < GRID_SIZE; r++) {
grid[r] = [];
for (let c = 0; c < GRID_SIZE; c++) {
grid[r][c] = Math.floor(Math.random() * COLORS.length);
}
}
// Remove initial matches
while (findMatches().length > 0) {
clearMatches(false);
refill();
}
}

function findMatches() {
let matched = new Set();
// Horizontal
for (let r = 0; r < GRID_SIZE; r++) {
for (let c = 0; c < GRID_SIZE - 2; c++) {
if (grid[r][c] !== -1 && grid[r][c] === grid[r][c+1] && grid[r][c] === grid[r][c+2]) {
matched.add(r + ',' + c); matched.add(r + ',' + (c+1)); matched.add(r + ',' + (c+2));
}
}
}
// Vertical
for (let c = 0; c < GRID_SIZE; c++) {
for (let r = 0; r < GRID_SIZE - 2; r++) {
if (grid[r][c] !== -1 && grid[r][c] === grid[r+1][c] && grid[r][c] === grid[r+2][c]) {
matched.add(r + ',' + c); matched.add((r+1) + ',' + c); matched.add((r+2) + ',' + c);
}
}
}
let arr = [];
matched.forEach(v => { let [r,c] = v.split(','); arr.push({r:+r, c:+c}); });
return arr;
}

function clearMatches(animate) {
let matches = findMatches();
if (matches.length === 0) return 0;
combo++;
let pts = matches.length * 10 * combo;
score += pts;
if (animate) {
floatTexts.push({ x: W/2, y: H/2 - 50*scale, text: '+' + pts + (combo > 1 ? ' x' + combo : ''), life: 60, maxLife: 60 });
}
matches.forEach(m => {
let cellSize = getCellSize();
let x = getGridX() + m.c * cellSize + cellSize/2;
let y = getGridY() + m.r * cellSize + cellSize/2;
for (let i = 0; i < 6; i++) {
particles.push({
x: x, y: y,
vx: (Math.random()-0.5)*4*scale, vy: (Math.random()-0.5)*4*scale - 2*scale,
life: 30 + Math.random()*20, maxLife: 50,
color: COLORS[grid[m.r][m.c]], size: (4+Math.random()*4)*scale
});
}
grid[m.r][m.c] = -1;
});
if (animate) { shakeTimer = 8; shakeIntensity = 3 * scale; }
return matches.length;
}

function refill() {
for (let c = 0; c < GRID_SIZE; c++) {
// Collect non-empty from bottom up
let blocks = [];
for (let r = GRID_SIZE - 1; r >= 0; r--) {
if (grid[r][c] !== -1) blocks.push(grid[r][c]);
}
while (blocks.length < GRID_SIZE) {
blocks.push(Math.floor(Math.random() * COLORS.length));
}
for (let r = GRID_SIZE - 1; r >= 0; r--) {
grid[r][c] = blocks[GRID_SIZE - 1 - r];
}
}
}

function shiftGrid(dir) {
// dir: 'up','down','left','right'
if (dir === 'down') {
for (let c = 0; c < GRID_SIZE; c++) {
let blocks = [];
for (let r = 0; r < GRID_SIZE; r++) if (grid[r][c] !== -1) blocks.push(grid[r][c]);
for (let r = 0; r < GRID_SIZE; r++) grid[r][c] = (r >= GRID_SIZE - blocks.length) ? blocks[r - (GRID_SIZE - blocks.length)] : -1;
}
} else if (dir === 'up') {
for (let c = 0; c < GRID_SIZE; c++) {
let blocks = [];
for (let r = GRID_SIZE-1; r >= 0; r--) if (grid[r][c] !== -1) blocks.push(grid[r][c]);
for (let r = 0; r < GRID_SIZE; r++) grid[r][c] = (r < blocks.length) ? blocks[r] : -1;
}
} else if (dir === 'right') {
for (let r = 0; r < GRID_SIZE; r++) {
let blocks = [];
for (let c = 0; c < GRID_SIZE; c++) if (grid[r][c] !== -1) blocks.push(grid[r][c]);
for (let c = 0; c < GRID_SIZE; c++) grid[r][c] = (c >= GRID_SIZE - blocks.length) ? blocks[c - (GRID_SIZE - blocks.length)] : -1;
}
} else if (dir === 'left') {
for (let r = 0; r < GRID_SIZE; r++) {
let blocks = [];
for (let c = 0; c < GRID_SIZE; c++) if (grid[r][c] !== -1) blocks.push(grid[r][c]);
for (let c = 0; c < GRID_SIZE; c++) grid[r][c] = (c < blocks.length) ? blocks[c] : -1;
}
}
}

function getCellSize() { return Math.min(W, H) * 0.065; }
function getGridX() { return (W - GRID_SIZE * getCellSize()) / 2; }
function getGridY() { return (H - GRID_SIZE * getCellSize()) / 2 + 30 * scale; }

let animFrame = 0;
let animPhase = 'none'; // shift, cascade
let animDir = '';
let shiftProgress = 0;
let displayGrid = [];

function startLevel() {
initGrid();
movesLeft = MOVES_PER_LEVEL;
combo = 0;
state = 'playing';
}

function doTurn(dir) {
if (state !== 'playing') return;
shiftGrid(dir);
animDir = dir;
animPhase = 'cascade';
shiftProgress = 0;
displayGrid = grid.map(row => row.slice());
combo = 0;
movesLeft--;
setTimeout(cascade, 150);
}

function cascade() {
let cleared = clearMatches(true);
if (cleared > 0) {
refill();
// Check for more matches after refill
setTimeout(() => {
let more = clearMatches(true);
if (more > 0) {
refill();
cascade();
} else {
checkEnd();
}
}, 250);
} else {
checkEnd();
}
combo = 0;
}

function checkEnd() {
if (score >= targetScore) {
state = 'levelComplete';
setTimeout(() => { level++; targetScore = TARGET_SCORES[Math.min(level, TARGET_SCORES.length)-1]; startLevel(); }, 2500);
} else if (movesLeft <= 0) {
state = 'gameOver';
} else {
state = 'playing';
}
}

function drawGrid() {
let cellSize = getCellSize();
let gx = getGridX();
let gy = getGridY();

// Grid background
ctx.fillStyle = 'rgba(20,20,40,0.8)';
ctx.fillRect(gx - 8*scale, gy - 8*scale, GRID_SIZE * cellSize + 16*scale, GRID_SIZE * cellSize + 16*scale);
ctx.strokeStyle = 'rgba(100,100,200,0.3)';
ctx.lineWidth = 2;
ctx.strokeRect(gx - 8*scale, gy - 8*scale, GRID_SIZE * cellSize + 16*scale, GRID_SIZE * cellSize + 16*scale);

for (let r = 0; r < GRID_SIZE; r++) {
for (let c = 0; c < GRID_SIZE; c++) {
let x = gx + c * cellSize;
let y = gy + r * cellSize;
// Cell border
ctx.strokeStyle = 'rgba(80,80,140,0.2)';
ctx.lineWidth = 1;
ctx.strokeRect(x, y, cellSize, cellSize);

if (grid[r][c] !== -1) {
let color = COLORS[grid[r][c]];
let pad = 3 * scale;
// Glow
ctx.shadowColor = color;
ctx.shadowBlur = 8 * scale;
ctx.fillStyle = color;
roundRect(ctx, x + pad, y + pad, cellSize - pad*2, cellSize - pad*2, 6*scale);
ctx.fill();
ctx.shadowBlur = 0;
// Highlight
ctx.fillStyle = 'rgba(255,255,255,0.3)';
roundRect(ctx, x + pad + 4*scale, y + pad + 4*scale, cellSize - pad*2 - 12*scale, (cellSize - pad*2)*0.3, 4*scale);
ctx.fill();
}
}
}

// Particles
for (let i = particles.length - 1; i >= 0; i--) {
let p = particles[i];
p.x += p.vx; p.y += p.vy; p.vy += 0.3 * scale;
p.life--;
if (p.life <= 0) { particles.splice(i, 1); continue; }
let alpha = p.life / p.maxLife;
ctx.globalAlpha = alpha;
ctx.fillStyle = p.color;
ctx.beginPath();
ctx.arc(p.x, p.y, p.size * alpha, 0, Math.PI*2);
ctx.fill();
ctx.globalAlpha = 1;
}

// Float texts
for (let i = floatTexts.length - 1; i >= 0; i--) {
let ft = floatTexts[i];
ft.y -= 1.5 * scale;
ft.life--;
if (ft.life <= 0) { floatTexts.splice(i, 1); continue; }
let alpha = ft.life / ft.maxLife;
ctx.globalAlpha = alpha;
ctx.fillStyle = '#fff';
ctx.font = `bold ${24*scale}px 'Segoe UI', sans-serif`;
ctx.textAlign = 'center';
ctx.fillText(ft.text, ft.x, ft.y);
ctx.globalAlpha = 1;
}
}

function roundRect(ctx, x, y, w, h, r) {
ctx.beginPath();
ctx.moveTo(x + r, y);
ctx.lineTo(x + w - r, y);
ctx.quadraticCurveTo(x + w, y, x + w, y + r);
ctx.lineTo(x + w, y + h - r);
ctx.quadraticCurveTo(x + w, y + h, x + w - r, y + h);
ctx.lineTo(x + r, y + h);
ctx.quadraticCurveTo(x, y + h, x, y + h - r);
ctx.lineTo(x, y + r);
ctx.quadraticCurveTo(x, y, x + r, y);
ctx.closePath();
}

function draw() {

// Background

drawStartScreen();
return;
}

shakeTimer--;
let sx = (Math.random()-0.5) * shakeIntensity * 2;

ctx.translate(sx, sy);
}

if (state === 'gameOver') { drawGameOver(); if(shakeTimer>0) ctx.restore(); return; }

}

particles = []; floatTexts = [];
initGrid(); startLevel();
return;

}
});
