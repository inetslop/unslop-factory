        ctx.beginPath(); ctx.moveTo(cx,cy); ctx.lineTo(cx+cellSize*0.2,cy+cellSize*0.1); ctx.stroke();
        if(isTarget(pad.x,pad.y)){
            ctx.strokeStyle='#f6e05e'; ctx.lineWidth=3;
            ctx.beginPath(); ctx.arc(cx,cy,cellSize*0.44,0,Math.PI*2); ctx.stroke();
        }
    });

    // Player
    if(gameState!=='dead'||deadTimer%12<7){
        drawFrog(offsetX+player.x*cellSize,offsetY+player.y*cellSize,cellSize);
    }

    // HUD
    const hfs=Math.floor(cellSize*0.4);
    ctx.fillStyle='#fff'; ctx.font=`${hfs}px monospace`; ctx.textAlign='left';
    ctx.fillText(`♥ ${lives}`,offsetX,offsetY-cellSize*0.5);
    ctx.textAlign='right';
    const done=pads.filter(p=>isTarget(p.x,p.y)).length;
    ctx.fillText(`Pads: ${done}/${pads.length}`,offsetX+COLS*cellSize,offsetY-cellSize*0.5);

    // Overlays
    if(gameState==='dead'){
        ctx.fillStyle='rgba(0,0,0,0.4)'; ctx.fillRect(0,0,canvas.width,canvas.height);
        ctx.fillStyle='#e53e3e'; ctx.font=`bold ${cellSize}px monospace`; ctx.textAlign='center';
        ctx.fillText('SPLAT!',canvas.width/2,canvas.height/2);
    }
    if(gameState==='win'){
        ctx.fillStyle='rgba(0,0,0,0.5)'; ctx.fillRect(0,0,canvas.width,canvas.height);
        ctx.fillStyle='#48bb78'; ctx.font=`bold ${cellSize}px monospace`; ctx.textAlign='center';
        ctx.fillText('LEVEL COMPLETE!',canvas.width/2,canvas.height/2);
        ctx.font=`${hfs}px monospace`; ctx.fillStyle='#fff';
        ctx.fillText('Press R to play again',canvas.width/2,canvas.height/2+cellSize*1.2);
    }
    if(gameState==='gameover'){
        ctx.fillStyle='rgba(0,0,0,0.7)'; ctx.fillRect(0,0,canvas.width,canvas.height);
        ctx.fillStyle='#e53e3e'; ctx.font=`bold ${cellSize}px monospace`; ctx.textAlign='center';
        ctx.fillText('GAME OVER',canvas.width/2,canvas.height/2);
        ctx.font=`${hfs}px monospace`; ctx.fillStyle='#fff';
        ctx.fillText('Press R to restart',canvas.width/2,canvas.height/2+cellSize*1.2);
    }
}

function loop(){
    frame++;
    if(gameState==='playing'){
        updateCars();
        if(isCarLane(player.y)&&carAt(player.x,player.y)) killPlayer();
    } else if(gameState==='dead'){
        deadTimer--;
        if(deadTimer<=0){
            if(lives>0){ player.x=4; player.y=7; gameState='playing'; }
            else gameState='gameover';
        }
    }
    draw();
    requestAnimationFrame(loop);
}

function startGame(){ initLevel(); lives=3; gameState='playing'; }

window.addEventListener('keydown',e=>{
    if(!audioCtx) initAudio();
    e.preventDefault();
    if(gameState==='start'){
        if(e.code==='Space'||e.code==='Enter') startGame();
        return;
    }
    if(gameState==='win'||gameState==='gameover'){
        if(e.code==='KeyR') startGame();
        return;
    }
    if(gameState!=='playing') return;
    switch(e.code){
        case'ArrowUp': movePlayer(0,-1); break;
        case'ArrowDown': movePlayer(0,1); break;
        case'ArrowLeft': movePlayer(-1,0); break;
        case'ArrowRight': movePlayer(1,0); break;
    }
});

initLevel();
loop();
