let audioCtx = null;
function initAudio() {
    if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
}
function playSound(type, freq, dur, vol) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.type = type;
    osc.frequency.value = freq || 440;
    gain.gain.setValueAtTime(vol || 0.1, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + (dur || 0.1));
    osc.start();
    osc.stop(audioCtx.currentTime + (dur || 0.1));
}

        particles.push({ x, y, vx: Math.cos(a)*s, vy: Math.sin(a)*s, life: 1, color, size: 2+Math.random()*3 });

}

function enemyAI(e, dt) {
    e.timer += dt;
    if (e.type === 'turret') {
        // Aimed spread shots

            playSound('square', 200, 0.1, 0.05);
        }
    } else if (e.type === 'flyer') {
        // Move toward player, fire spiral
        const dx = player.x - e.x;
        const dy = player.y - e.y;
        e.vx += dx * 0.02;
        e.vy += dy * 0.02;
        e.vx *= 0.95; e.vy *= 0.95;
        e.x += e.vx; e.y += e.vy;
        if (e.timer > 0.8) {
            e.timer = 0;

            }

    } else if (e.type === 'boss') {
        // Boss patterns
        const pattern = Math.floor(e.timer / 2) % 3;
        if (pattern === 0) {

            if (e.timer > 1.2) {
                e.timer = 0;

                    fireBullet(e.x + e.w/2, e.y + e.h/2, i * (Math.PI*2/16), 2.8, '#ff44aa', 5);
                }
                playSound('sawtooth', 100, 0.2, 0.08);
            }
        } else if (pattern === 1) {
            // Aimed fan

                e.timer = 0;
                const dx = player.x - e.x;
                const dy = player.y - e.y;
                const baseAngle = Math.atan2(dy, dx);
                for (let i = -3; i <= 3; i++) {
                    fireBullet(e.x + e.w/2, e.y + e.h/2, baseAngle + i * 0.15, 4, '#ffaa00', 4);
                }
            }
        } else {
            // Stream

                e.timer = 0;
                const dx = player.x - e.x;
                const dy = player.y - e.y;
                const baseAngle = Math.atan2(dy, dx);
                fireBullet(e.x + e.w/2, e.y + e.h/2, baseAngle, 5, '#ff2200', 6);
            }
        }
        // Boss moves slowly toward player
        const dx = player.x - e.x;
        e.x += Math.sign(dx) * 0.5;
    }

function update(dt) {
    if (gameState !== 'playing') return;
    gameTime += dt;

    // Player movement
    const speed = 4;
    if (player.dashing > 0) {
        player.dashing -= dt;
        player.x += player.dashDir * 12;
        player.y += 0;
    } else {
        if (keys['KeyA'] || keys['ArrowLeft']) { player.vx = -speed; player.facing = -1; }
        else if (keys['KeyD'] || keys['ArrowRight']) { player.vx = speed; player.facing = 1; }
        else player.vx *= 0.7;

        // Jump
        if ((keys['KeyW'] || keys['Space'] || keys['ArrowUp']) && player.onGround) {
            player.vy = -12;
            player.onGround = false;
            playSound('square', 300, 0.08, 0.06);
        }

        // Dash
        if (keys['ShiftLeft'] || keys['ShiftRight']) {
            if (player.dashCooldown <= 0) {
                player.dashing = 0.15;
                player.dashCooldown = 0.8;
                player.dashDir = player.vx !== 0 ? Math.sign(player.vx) : player.facing;
                playSound('sawtooth', 500, 0.1, 0.08);
            }
        }
    }

    player.dashCooldown -= dt;
    if (player.dashCooldown < 0) player.dashCooldown = 0;

    // Gravity
    player.vy += 0.6;
    player.x += player.vx;
    player.y += player.vy;

    // Platform collision
    player.onGround = false;
    for (const p of platforms) {
        if (player.x + player.w > p.x && player.x < p.x + p.w) {
            if (player.vy >= 0 && player.y + player.h >= p.y && player.y + player.h - player.vy <= p.y + 5) {
                player.y = p.y - player.h;
                player.vy = 0;
                player.onGround = true;
            }
        }
    }

    // Fall off screen
    if (player.y > H + 50) {
        player.health--;
        playSound('sawtooth', 100, 0.3, 0.15);
        if (player.health <= 0) { gameState = 'gameover'; saveHi(); }
        else { player.y = H - 200; player.vy = 0; player.invincible = 2; }
    }

    // Bounds
    if (player.x < 0) player.x = 0;
    if (player.x > levelWidth - player.w) player.x = levelWidth - player.w;

    // Invincibility
    if (player.invincible > 0) player.invincible -= dt;

    // Camera
    const targetCam = player.x - W * 0.35;
    cameraX += (targetCam - cameraX) * 0.08;
    if (cameraX < 0) cameraX = 0;
    if (cameraX > levelWidth - W) cameraX = levelWidth - W;

    // Spawn waves
    waveTimer += dt;
    if (waveTimer > 3 && enemies.length < 5 + wave) {
        waveTimer = 0;
        wave++;
        const type = Math.random() < 0.4 ? 'turret' : 'flyer';
        const ex = cameraX + W + 50 + Math.random() * 200;
        const ey = H - 100 - Math.random() * (H * 0.5);
        if (type === 'turret') {
            // Find a platform to place on
            let placed = false;
            for (const p of platforms) {
                if (Math.abs(p.x + p.w/2 - ex) < 150 && p.y > H * 0.3) {
                    spawnEnemy('turret', p.x + p.w/2, p.y - 40);
                    placed = true;
                    break;
                }
            }
            if (!placed) spawnEnemy('turret', ex, H - 80);
        } else {
            spawnEnemy('flyer', ex, ey);
        }
    }

    // Boss at end of level
    if (player.x > levelWidth - 400 && !bossDefeated && enemies.length < 1) {
        spawnEnemy('boss', cameraX + W - 150, H / 2 - 80);
    }

    // Update enemies
    for (let i = enemies.length - 1; i >= 0; i--) {
        const e = enemies[i];
        enemyAI(e, dt);
        if (e.health <= 0) {
            spawnParticles(e.x + e.w/2, e.y + e.h/2, '#ff6644', 15, 4);
            playSound('square', 80, 0.2, 0.12);
            score += e.type === 'boss' ? 500 : e.type === 'turret' ? 50 : 30;
            if (e.type === 'boss') bossDefeated = true;
            enemies.splice(i, 1);
        }
    }

    // Player shooting (auto-fire toward nearest enemy)
    player.shootTimer -= dt;
    if (player.shootTimer <= 0 && enemies.length > 0) {
        player.shootTimer = 0.25;
        let nearest = null, nd = Infinity;
        for (const e of enemies) {
            const d = Math.hypot(e.x - player.x, e.y - player.y);
            if (d < nd && d < 500) { nd = d; nearest = e; }
        }
        if (nearest) {
            const a = Math.atan2(nearest.y + nearest.h/2 - player.y, nearest.x + nearest.w/2 - player.x);
            playerBullets.push({ x: player.x + player.w/2, y: player.y + player.h/2, vx: Math.cos(a)*8, vy: Math.sin(a)*8, radius: 4 });
            playSound('square', 600, 0.05, 0.03);
        }
    }

    // Update player bullets
    for (let i = playerBullets.length - 1; i >= 0; i--) {
        const b = playerBullets[i];
        b.x += b.vx; b.y += b.vy;
        if (b.x < cameraX - 50 || b.x > cameraX + W + 50 || b.y < -50 || b.y > H + 50) {
            playerBullets.splice(i, 1);
            continue;
        }
        // Hit enemy
        for (let j = enemies.length - 1; j >= 0; j--) {
            const e = enemies[j];
            if (b.x > e.x && b.x < e.x + e.w && b.y > e.y && b.y < e.y + e.h) {
                e.health--;
                playerBullets.splice(i, 1);
                spawnParticles(b.x, b.y, '#ffffff', 3, 2);
                break;
            }
        }
    }

    // Update enemy bullets
    for (let i = bullets.length - 1; i >= 0; i--) {
        const b = bullets[i];
        b.x += b.vx; b.y += b.vy;
        if (b.x < cameraX - 100 || b.x > cameraX + W + 100 || b.y < -100 || b.y > H + 100) {
            bullets.splice(i, 1);
            continue;
        }
    }

    // Bullet-player collision (dodge mode: small hitbox when not pressing move keys)
    const dodgeMode = !keys['KeyA'] && !keys['KeyD'] && !keys['ArrowLeft'] && !keys['ArrowRight'];
    const hitR = player.dashing > 0 ? 2 : (dodgeMode ? 4 : 10);
    const px = player.x + player.w/2;
    const py = player.y + player.h/2;

    if (player.invincible <= 0 && player.dashing <= 0) {
        for (const b of bullets) {
            const dx = b.x - px, dy = b.y - py;
            if (dx*dx + dy*dy < (b.radius + hitR) * (b.radius + hitR)) {
                player.health--;
                player.invincible = 1.5;
                playSound('sawtooth', 150, 0.3, 0.2);
                spawnParticles(px, py, '#ff0044', 10, 3);
                if (player.health <= 0) { gameState = 'gameover'; saveHi(); }
                break;
            }
        }
    }

    // Particles
    for (let i = particles.length - 1; i >= 0; i--) {
        const p = particles[i];
        p.x += p.vx; p.y += p.vy;
        p.vy += 0.1;
        p.life -= dt * 2;
        if (p.life <= 0) particles.splice(i, 1);
    }

    // Win condition
    if (bossDefeated && enemies.length === 0 && player.x > levelWidth - 400) {
        gameState = 'win';
        saveHi();
    }
}

// Game loop
let lastTime = 0;
function gameLoop(time) {
    const dt = Math.min((time - lastTime) / 1000, 0.05);
    lastTime = time;
    gameTime += dt;

    if (gameState === 'menu' || gameState === 'gameover' || gameState === 'win') {
        if (keys['Enter'] || keys['Space']) {
            resetGame();
            gameState = 'playing';
            playSound('square', 440, 0.15, 0.1);
        }
    }

    update(dt);
    draw();

    requestAnimationFrame(gameLoop);
}

gameState = 'menu';
