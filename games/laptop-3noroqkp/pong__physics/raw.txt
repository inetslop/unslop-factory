<script>
const canvas = document.getElementById('game');
const ctx = canvas.getContext('2d');

let W, H;
function resize() {
    W = window.innerWidth;
    H = window.innerHeight;
    canvas.width = W;
    canvas.height = H;
}
window.addEventListener('resize', resize);
resize();

// Audio
let audioCtx = null;
function initAudio() {
    if (!audioCtx) {
        audioCtx = new (window.AudioContext || window.webkitAudioContext)();
    }
}
function playSound(freq, duration, type, vol) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = type || 'sine';
    osc.frequency.value = freq;
    gain.gain.setValueAtTime(vol || 0.3, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + (duration || 0.1));
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + (duration || 0.1));
}

// Game state
const STATE = { MENU: 0, PLAYING: 1, SCORE: 2 };
let gameState = STATE.MENU;
let mode = 0; // 0 = vs AI, 1 = 2 player
let scores = [0, 0];
let winner = -1;

// Physics constants
const GRAVITY = 0;
const DRAG = 0.9995;
const RESTITUTION = 0.85;
const SPIN_EFFECT = 0.3;
const PADDLE_SPIN_TRANSFER = 0.4;

// Ball
let ball = { x: 0, y: 0, vx: 0, vy: 0, spin: 0, r: 10, trail: [] };

// Paddles
let paddles = [
    { x: 30, y: 0, w: 14, h: 80, vy: 0 },
    { x: 0, y: 0, w: 14, h: 80, vy: 0 }
];

// Input
let keys = {};
window.addEventListener('keydown', e => {
    keys[e.code] = true;
    if (gameState === STATE.MENU && (e.code === 'Enter' || e.code === 'Space')) {
        initAudio();
        startGame(mode);
    } else if (gameState === STATE.MENU && (e.code === 'Digit1' || e.code === 'Numpad1')) {
        mode = 0;
    } else if (gameState === STATE.MENU && (e.code === 'Digit2' || e.code === 'Numpad2')) {
        mode = 1;
    } else if (gameState === STATE.SCORE && (e.code === 'Enter' || e.code === 'Space')) {
        initAudio();
        startGame(mode);
    } else if (gameState === STATE.PLAYING && e.code === 'KeyR') {
        startGame(mode);
    }
});
window.addEventListener('keyup', e => { keys[e.code] = false; });

function startGame(m) {
    mode = m;
    scores = [0, 0];
    winner = -1;
    paddles[0].y = H / 2 - paddles[0].h / 2;
    paddles[1].y = H / 2 - paddles[1].h / 2;
    paddles[0].x = 30;
    paddles[1].x = W - 30 - paddles[1].w;
    resetBall();
    gameState = STATE.PLAYING;
}

function resetBall() {
    ball.x = W / 2;
    ball.y = H / 2;
    let angle = (Math.random() - 0.5) * Math.PI * 0.4;
    let speed = 7;
    ball.vx = Math.cos(angle) * speed;
    ball.vy = Math.sin(angle) * speed;
    ball.spin = 0;
    ball.trail = [];
}

function updateBall() {
    // Apply drag
    ball.vx *= DRAG;
    ball.vy *= DRAG;

    // Magnus effect - spin curves the ball
    let magnusX = -ball.spin * SPIN_EFFECT * ball.vy * 0.01;
    let magnusY = ball.spin * SPIN_EFFECT * ball.vx * 0.01;
    ball.vx += magnusX;
    ball.vy += magnusY;

    // Spin decays
    ball.spin *= 0.998;

    ball.x += ball.vx;
    ball.y += ball.vy;

    // Trail
    ball.trail.push({ x: ball.x, y: ball.y, spin: ball.spin });
    if (ball.trail.length > 30) ball.trail.shift();

    // Top/Bottom bounce
    if (ball.y - ball.r < 0) {
        ball.y = ball.r;
        ball.vy = Math.abs(ball.vy) * RESTITUTION;
        ball.spin *= 0.8;
        playSound(200, 0.1, 'triangle', 0.2);
    }
    if (ball.y + ball.r > H) {
        ball.y = H - ball.r;
        ball.vy = -Math.abs(ball.vy) * RESTITUTION;
        ball.spin *= 0.8;
        playSound(200, 0.1, 'triangle', 0.2);
    }

    // Paddle collisions
    for (let i = 0; i < 2; i++) {
        let p = paddles[i];
        if (ball.x - ball.r < p.x + p.w && ball.x + ball.r > p.x &&
            ball.y + ball.r > p.y && ball.y - ball.r < p.y + p.h) {

            // Determine hit position relative to paddle center
            let hitPos = (ball.y - (p.y + p.h / 2)) / (p.h / 2); // -1 to 1
            hitPos = Math.max(-1, Math.min(1, hitPos));

            // Reflect
            if (i === 0) {
                ball.vx = Math.abs(ball.vx) * RESTITUTION + 1;
            } else {
                ball.vx = -Math.abs(ball.vx) * RESTITUTION - 1;
            }

            // Angle based on hit position
            let baseAngle = (i === 0 ? 1 : -1);
            let angleOffset = hitPos * 0.8;
            let speed = Math.sqrt(ball.vx * ball.vx + ball.vy * ball.vy);
            speed = Math.min(speed * 1.05, 20);

            let newAngle = baseAngle > 0 ? Math.asin(Math.max(-0.9, Math.min(0.9, hitPos * 0.6))) : -Math.asin(Math.max(-0.9, Math.min(0.9, hitPos * 0.6)));
            ball.vx = Math.cos(newAngle) * speed * baseAngle;
            ball.vy = Math.sin(newAngle) * speed + p.vy * 0.3;

            // Impart spin from paddle movement and hit position
            ball.spin = p.vy * PADDLE_SPIN_TRANSFER + hitPos * 2;

            // Push ball out of paddle
            if (i === 0) ball.x = p.x + p.w + ball.r;
            else ball.x = p.x - ball.r;

            playSound(440, 0.08, 'square', 0.3);
        }
    }

    // Score
    if (ball.x < -ball.r * 2) {
        scores[1]++;
        playSound(200, 0.3, 'sawtooth', 0.4);
        checkWin();
        resetBall();
    }
    if (ball.x > W + ball.r * 2) {
        scores[0]++;
        playSound(200, 0.3, 'sawtooth', 0.4);
        checkWin();
        resetBall();
    }

    // Speed cap
    let spd = Math.sqrt(ball.vx * ball.vx + ball.vy * ball.vy);
    if (spd > 20) {
        ball.vx = (ball.vx / spd) * 20;
        ball.vy = (ball.vy / spd) * 20;
    }
}

function checkWin() {
    if (scores[0] >= 5 || scores[1] >= 5) {
        winner = scores[0] >= 5 ? 0 : 1;
        gameState = STATE.SCORE;
    }
}

function updatePaddles(dt) {
    const speed = 8;
    // Paddle 0 (left) - W/S
    if (keys['KeyW']) paddles[0].vy = -speed;
    else if (keys['KeyS']) paddles[0].vy = speed;
    else paddles[0].vy *= 0.8;

    // Paddle 1 (right)
    if (mode === 1) {
        // 2 player: ArrowUp/ArrowDown
        if (keys['ArrowUp']) paddles[1].vy = -speed;
        else if (keys['ArrowDown']) paddles[1].vy = speed;
        else paddles[1].vy *= 0.8;
    } else {
        // AI
        let targetY = ball.y - paddles[1].h / 2;
        let diff = targetY - paddles[1].y;
        let aiSpeed = Math.min(6, Math.abs(diff) * 0.1 + 3);
        paddles[1].vy = Math.sign(diff) * aiSpeed;
    }

    for (let i = 0; i < 2; i++) {
        paddles[i].y += paddles[i].vy;
        if (paddles[i].y < 0) paddles[i].y = 0;
        if (paddles[i].y + paddles[i].h > H) paddles[i].y = H - paddles[i].h;
    }
}

function draw() {
    ctx.fillStyle = '#0a0a1a';
    ctx.fillRect(0, 0, W, H);

    // Center line
    ctx.setLineDash([10, 15]);
    ctx.strokeStyle = 'rgba(100, 140, 255, 0.3)';
    ctx.lineWidth = 2;
    ctx.beginPath();
    ctx.moveTo(W / 2, 0);
    ctx.lineTo(W / 2, H);
    ctx.stroke();
    ctx.setLineDash([]);

    // Center circle
    ctx.strokeStyle = 'rgba(100, 140, 255, 0.15)';
    ctx.beginPath();
    ctx.arc(W / 2, H / 2, 60, 0, Math.PI * 2);
    ctx.stroke();

    if (gameState === STATE.MENU) {
        drawMenu();
        return;
    }

    // Ball trail
    for (let i = 0; i < ball.trail.length; i++) {
        let t = ball.trail[i];
        let alpha = (i / ball.trail.length) * 0.5;
        let size = ball.r * (i / ball.trail.length) * 0.7;
        ctx.fillStyle = `rgba(255, 180, 50, ${alpha})`;
        ctx.beginPath();
        ctx.arc(t.x, t.y, size, 0, Math.PI * 2);
        ctx.fill();
    }

    // Ball
    let glow = Math.min(1, Math.sqrt(ball.vx * ball.vx + ball.vy * ball.vy) / 15);
    ctx.shadowColor = `rgba(255, 200, 80, ${glow})`;
    ctx.shadowBlur = 15 * glow;
    ctx.fillStyle = '#fff';
    ctx.beginPath();
    ctx.arc(ball.x, ball.y, ball.r, 0, Math.PI * 2);
    ctx.fill();

    // Spin indicator
    if (Math.abs(ball.spin) > 0.5) {
        let spinAlpha = Math.min(1, Math.abs(ball.spin) / 5);
        ctx.strokeStyle = `rgba(100, 200, 255, ${spinAlpha})`;
        ctx.lineWidth = 2;
        let spinAngle = performance.now() * 0.005 * ball.spin;
        for (let a = 0; a < 3; a++) {
            ctx.beginPath();
            ctx.arc(ball.x, ball.y, ball.r + 4, spinAngle + a * Math.PI * 2 / 3, spinAngle + a * Math.PI * 2 / 3 + 1.2);
            ctx.stroke();
        }
    }

    ctx.shadowBlur = 0;

    // Paddles
    for (let i = 0; i < 2; i++) {
        let p = paddles[i];
        let glowPaddle = Math.min(1, Math.abs(p.vy) / 8);
        ctx.shadowColor = i === 0 ? 'rgba(80, 200, 255, 0.6)' : 'rgba(255, 100, 150, 0.6)';
        ctx.shadowBlur = 10 + glowPaddle * 10;
        ctx.fillStyle = i === 0 ? '#4fc8ff' : '#ff6496';
        roundRect(p.x, p.y, p.w, p.h, 7);
        ctx.fill();

        // Paddle velocity indicator
        if (Math.abs(p.vy) > 1) {
            ctx.strokeStyle = `rgba(255, 255, 255, ${Math.min(0.6, Math.abs(p.vy) / 10)})`;
            ctx.lineWidth = 2;
            let cx = i === 0 ? p.x + p.w / 2 : p.x + p.w / 2;
            let cy = p.y + p.h / 2;
            let dir = Math.sign(p.vy);
            ctx.beginPath();
            ctx.moveTo(cx, cy - dir * 15);
            ctx.lineTo(cx, cy - dir * 8);
            ctx.stroke();
        }
    }

    ctx.shadowBlur = 0;

    // Score
    ctx.font = 'bold 48px monospace';
    ctx.textAlign = 'center';
    ctx.fillStyle = 'rgba(255, 255, 255, 0.6)';
    ctx.fillText(scores[0], W / 2 - 60, 60);
    ctx.fillText(scores[1], W / 2 + 60, 60);

    // Physics info
    let speed = Math.sqrt(ball.vx * ball.vx + ball.vy * ball.vy);
    ctx.font = '12px monospace';
    ctx.textAlign = 'left';
    ctx.fillStyle = 'rgba(150, 200, 255, 0.5)';
    ctx.fillText(`v: ${speed.toFixed(1)}  spin: ${ball.spin.toFixed(2)}`, 10, H - 10);

    if (gameState === STATE.SCORE) {
        drawScore();
    }
}

function roundRect(x, y, w, h, r) {
    ctx.beginPath();
    ctx.moveTo(x + r, y);
    ctx.lineTo(x + w - r, y);
    ctx.quadraticCurveTo(x + w, y, x + w, y + r);
    ctx.lineTo(x + w, y + h - r);
    ctx.quadraticCurveTo(x + w, y + h, x + w - r, y + h);
    ctx.lineTo(x + r, y + h);
    ctx.quadraticCurveTo(x, y + h, x, y + h - r);
    ctx.lineTo(x, y + r);
    ctx.quadraticCurveTo(x, y, x + r, y);
    ctx.closePath();
}

function drawMenu() {
    ctx.textAlign = 'center';
    ctx.font = 'bold 52px monospace';
    ctx.fillStyle = '#fff';
    ctx.shadowColor = 'rgba(100, 200, 255, 0.8)';
    ctx.shadowBlur = 20;
    ctx.fillText('PHYSICS PONG', W / 2, H / 2 - 80);
    ctx.shadowBlur = 0;

    ctx.font = '18px monospace';
    ctx.fillStyle = 'rgba(255, 255, 255, 0.7)';
    ctx.fillText('A physics-based Pong with spin & Magnus effect', W / 2, H / 2 - 30);

    ctx.font = '16px monospace';
    ctx.fillStyle = 'rgba(150, 200, 255, 0.8)';
    ctx.fillText('[1] Player vs AI', W / 2, H / 2 + 30);
    ctx.fillText('[2] Two Players', W / 2, H / 2 + 55);

    ctx.fillStyle = 'rgba(255, 255, 255, 0.6)';
    ctx.font = '14px monospace';
    ctx.fillText('Left:  W/S    Right: ↑/↓ (or AI)', W / 2, H / 2 + 100);
    ctx.fillText('Paddle movement imparts SPIN on the ball!', W / 2, H / 2 + 125);

    ctx.fillStyle = 'rgba(255, 255, 255, 0.9)';
    ctx.font = '16px monospace';
    let blink = Math.sin(performance.now() * 0.004) > 0;
    if (blink) ctx.fillText('Press ENTER to start', W / 2, H / 2 + 170);

    // Decorative ball
    let t = performance.now() * 0.002;
    let bx = W / 2 + Math.cos(t) * 100;
    let by = H / 2 - 80 + Math.sin(t * 1.5) * 20;
    ctx.fillStyle = '#fff';
    ctx.shadowColor = 'rgba(255, 200, 80, 0.8)';
    ctx.shadowBlur = 15;
    ctx.beginPath();
    ctx.arc(bx, by, 6, 0, Math.PI * 2);
    ctx.fill();
    ctx.shadowBlur = 0;
}

function drawScore() {
    ctx.fillStyle = 'rgba(0, 0, 0, 0.7)';
    ctx.fillRect(0, 0, W, H);

    ctx.textAlign = 'center';
    ctx.font = 'bold 48px monospace';
    let wName = mode === 1 ? (winner === 0 ? 'P1 WINS!' : 'P2 WINS!') : (winner === 0 ? 'YOU WIN!' : 'AI WINS!');
    ctx.fillStyle = winner === 0 ? '#4fc8ff' : '#ff6496';
    ctx.fillText(wName, W / 2, H / 2 - 30);

    ctx.font = '24px monospace';
    ctx.fillStyle = '#fff';
    ctx.fillText(`${scores[0]} - ${scores[1]}`, W / 2, H / 2 + 20);

    let blink = Math.sin(performance.now() * 0.004) > 0;
    if (blink) {
        ctx.font = '16px monospace';
        ctx.fillStyle = 'rgba(255, 255, 255, 0.7)';
        ctx.fillText('Press ENTER to play again', W / 2, H / 2 + 70);
    }
}

let lastTime = 0;
function gameLoop(time) {
    let dt = Math.min((time - lastTime) / 16.67, 3);
    lastTime = time;

    if (gameState === STATE.PLAYING) {
        updatePaddles(dt);
        updateBall();
    }

    draw();
    requestAnimationFrame(gameLoop);
}
requestAnimationFrame(gameLoop);
</script>
