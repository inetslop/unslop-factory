<script>
const canvas = document.getElementById('game');
const ctx = canvas.getContext('2d');

let W, H, cellSize, cols, rows;
let snake, dir, nextDir, food, gravityWells, particles, score, highScore, gameSpeed, lastTime, animTime;
let state = 'start'; // start, playing, dead
let audioCtx = null;
let screenShake = 0;

try { highScore = parseInt(localStorage.getItem('snakeGravityHigh') || '0'); } catch(e) { highScore = 0; }

function resize() {
    W = canvas.width = window.innerWidth;
    H = canvas.height = window.innerHeight;
    cellSize = Math.floor(Math.min(W, H) / 24);
    cols = Math.floor(W / cellSize);
    rows = Math.floor(H / cellSize);
}
window.addEventListener('resize', resize);
resize();

function initAudio() {
    if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
}

function playSound(freq, duration, type, vol) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = type || 'square';
    osc.frequency.value = freq;
    gain.gain.setValueAtTime(vol || 0.1, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + (duration || 0.1));
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + (duration || 0.1));
}

function playEat() { playSound(600, 0.1, 'sine', 0.15); playSound(900, 0.15, 'sine', 0.1); }
function playDeath() { playSound(150, 0.4, 'sawtooth', 0.2); playSound(80, 0.6, 'square', 0.15); }
function playGravityShift() { playSound(200, 0.3, 'triangle', 0.08); }

function startGame() {
    const cx = Math.floor(cols / 2);
    const cy = Math.floor(rows / 2);
    snake = [{x: cx, y: cy}, {x: cx - 1, y: cy}, {x: cx - 2, y: cy}];
    dir = {x: 1, y: 0};
    nextDir = {x: 1, y: 0};
    score = 0;
    gameSpeed = 8;
    lastTime = 0;
    animTime = 0;
    food = spawnFood();
    gravityWells = [];
    for (let i = 0; i < 2; i++) spawnGravityWell();
    particles = [];
    state = 'playing';
}

function spawnFood() {
    let x, y, tries = 0;
    do {
        x = Math.floor(Math.random() * cols);
        y = Math.floor(Math.random() * rows);
        tries++;
    } while (tries < 100 && snake.some(s => s.x === x && s.y === y));
    return {x, y};
}

function spawnGravityWell() {
    let x, y, tries = 0;
    do {
        x = Math.floor(Math.random() * cols);
        y = Math.floor(Math.random() * rows);
        tries++;
    } while (tries < 100 && snake.some(s => s.x === x && s.y === y));
    gravityWells.push({x, y, phase: Math.random() * Math.PI * 2, life: 300 + Math.random() * 500});
}

function getGravityInfluence(headX, headY) {
    for (let well of gravityWells) {
        const dx = well.x - headX;
        const dy = well.y - headY;
        const dist = Math.sqrt(dx * dx + dy * dy);
        if (dist < 4 && dist > 0.5) {
            // Pull toward the well
            const angle = Math.atan2(dy, dx);
            const strength = 1 - (dist / 4);
            if (Math.random() < strength * 0.6) {
                let bestDir = null;
                let bestDot = -2;
                const dirs = [{x:1,y:0},{x:-1,y:0},{x:0,y:1},{x:0,y:-1}];
                for (let d of dirs) {
                    if (d.x === -dir.x && d.y === -dir.y) continue; // no reverse
                    const dot = (d.x * Math.cos(angle)) + (d.y * Math.sin(angle));
                    if (dot > bestDot) { bestDot = dot; bestDir = d; }
                }
                return bestDir || null;
            }
        }
    }
    return null;
}

function update(dt) {
    if (state !== 'playing') return;
    
    animTime += dt;
    
    // Gravity well movement and lifecycle
    for (let i = gravityWells.length - 1; i >= 0; i--) {
        const w = gravityWells[i];
        w.phase += dt * 2;
        w.life -= dt;
        if (w.life <= 0) {
            gravityWells.splice(i, 1);
            if (gravityWells.length < 3 && Math.random() < 0.5) spawnGravityWell();
        } else if (Math.random() < 0.02) {
            const ox = Math.floor(Math.random() * 3) - 1;
            const oy = Math.floor(Math.random() * 3) - 1;
            w.x += ox; w.y += oy;
            w.x = Math.max(0, Math.min(cols - 1, w.x));
            w.y = Math.max(0, Math.min(rows - 1, w.y));
        }
    }
    
    if (Math.random() < 0.005) spawnGravityWell();
    
    // Particles for gravity wells
    for (let well of gravityWells) {
        if (Math.random() < 0.3) {
            const angle = Math.random() * Math.PI * 2;
            const r = cellSize * 3;
            particles.push({
                x: well.x * cellSize + Math.cos(angle) * r,
                y: well.y * cellSize + Math.sin(angle) * r,
                vx: -Math.cos(angle) * 0.5,
                vy: -Math.sin(angle) * 0.5,
                life: 1,
                well: well
            });
        }
    }
    
    // Update particles
    for (let i = particles.length - 1; i >= 0; i--) {
        const p = particles[i];
        p.x += p.vx;
        p.y += p.vy;
        p.life -= dt * 0.8;
        if (p.life <= 0) particles.splice(i, 1);
    }
    
    // Screen shake decay
    if (screenShake > 0) screenShake *= 0.9;
    
    // Snake movement (frame-based at gameSpeed fps)
    const moveInterval = 1000 / gameSpeed;
    if (!update._lastMove) update._lastMove = 0;
    update._lastMove += dt * 1000;
    
    while (update._lastMove >= moveInterval) {
        update._lastMove -= moveInterval;
        
        // Apply gravity influence
        const gravDir = getGravityInfluence(snake[0].x, snake[0].y);
        const moveDir = gravDir || nextDir;
        
        dir = moveDir;
        const head = {x: snake[0].x + dir.x, y: snake[0].y + dir.y};
        
        // Collision check
        if (head.x < 0 || head.x >= cols || head.y < 0 || head.y >= rows) {
            die();
            return;
        }
        for (let s of snake) {
            if (s.x === head.x && s.y === head.y) {
                die();
                return;
            }
        }
        
        snake.unshift(head);
        
        // Eat food
        if (head.x === food.x && head.y === food.y) {
            score += 10;
            gameSpeed = Math.min(15, 8 + Math.floor(score / 50));
            playEat();
            screenShake = 3;
            food = spawnFood();
        } else {
            snake.pop();
        }
    }
}

function die() {
    state = 'dead';
    playDeath();
    screenShake = 8;
    if (score > highScore) {
        highScore = score;
        try { localStorage.setItem('snakeGravityHigh', String(highScore)); } catch(e) {}
    }
}

function draw() {
    ctx.save();
    
    // Screen shake
    if (screenShake > 0.5) {
        ctx.translate(
            (Math.random() - 0.5) * screenShake,
            (Math.random() - 0.5) * screenShake
        );
    }
    
    // Background
    ctx.fillStyle = '#0a0a1a';
    ctx.fillRect(-10, -10, W + 20, H + 20);
    
    // Grid (subtle)
    ctx.strokeStyle = 'rgba(30, 30, 60, 0.5)';
    ctx.lineWidth = 1;
    for (let x = 0; x <= cols; x++) {
        ctx.beginPath(); ctx.moveTo(x * cellSize, 0); ctx.lineTo(x * cellSize, rows * cellSize); ctx.stroke();
    }
    for (let y = 0; y <= rows; y++) {
        ctx.beginPath(); ctx.moveTo(0, y * cellSize); ctx.lineTo(cols * cellSize, y * cellSize); ctx.stroke();
    }
    
    if (state === 'start') {
        drawStartScreen();
        ctx.restore();
        return;
    }
    
    // Gravity wells
    for (let well of gravityWells) {
        const cx = well.x * cellSize + cellSize / 2;
        const cy = well.y * cellSize + cellSize / 2;
        const pulse = Math.sin(well.phase) * 0.3 + 1;
        const radius = cellSize * 2.5 * pulse;
        
        // Glow
        const grad = ctx.createRadialGradient(cx, cy, 0, cx, cy, radius);
        grad.addColorStop(0, 'rgba(120, 80, 255, 0.3)');
        grad.addColorStop(0.5, 'rgba(100, 50, 200, 0.15)');
        grad.addColorStop(1, 'rgba(80, 40, 180, 0)');
        ctx.fillStyle = grad;
        ctx.beginPath();
        ctx.arc(cx, cy, radius, 0, Math.PI * 2);
        ctx.fill();
        
        // Core
        ctx.fillStyle = 'rgba(180, 140, 255, 0.6)';
        ctx.beginPath();
        ctx.arc(cx, cy, cellSize * 0.4, 0, Math.PI * 2);
        ctx.fill();
        
        // Swirl lines
        ctx.strokeStyle = 'rgba(150, 100, 255, 0.4)';
        ctx.lineWidth = 1.5;
        for (let i = 0; i < 3; i++) {
            const a = well.phase * 1.5 + i * (Math.PI * 2 / 3);
            ctx.beginPath();
            ctx.arc(cx, cy, cellSize * (1 + i * 0.4), a, a + 1.2);
            ctx.stroke();
        }
    }
    
    // Particles
    for (let p of particles) {
        ctx.fillStyle = `rgba(150, 100, 255, ${p.life * 0.6})`;
        ctx.beginPath();
        ctx.arc(p.x, p.y, 2, 0, Math.PI * 2);
        ctx.fill();
    }
    
    // Food
    const fx = food.x * cellSize + cellSize / 2;
    const fy = food.y * cellSize + cellSize / 2;
    const fpulse = Math.sin(animTime * 5) * 0.15 + 0.85;
    ctx.fillStyle = '#ff6633';
    ctx.shadowColor = '#ff6633';
    ctx.shadowBlur = 10;
    ctx.beginPath();
    ctx.arc(fx, fy, cellSize * 0.35 * fpulse, 0, Math.PI * 2);
    ctx.fill();
    ctx.shadowBlur = 0;
    
    // Snake
    for (let i = snake.length - 1; i >= 0; i--) {
        const s = snake[i];
        const t = i / snake.length;
        const r = Math.floor(50 + (1 - t) * 50);
        const g = Math.floor(220 - t * 80);
        const b = Math.floor(80 + t * 40);
        ctx.fillStyle = `rgb(${r},${g},${b})`;
        
        if (i === 0) {
            // Head - glow
            ctx.shadowColor = '#66ff99';
            ctx.shadowBlur = 8;
            ctx.fillRect(s.x * cellSize + 1, s.y * cellSize + 1, cellSize - 2, cellSize - 2);
            ctx.shadowBlur = 0;
        } else {
            const shrink = 2 + t * 3;
            ctx.fillRect(s.x * cellSize + shrink, s.y * cellSize + shrink, cellSize - shrink * 2, cellSize - shrink * 2);
        }
    }
    
    // Score
    ctx.fillStyle = '#aaffcc';
    ctx.font = `bold ${Math.max(14, cellSize)}px monospace`;
    ctx.textAlign = 'left';
    ctx.fillText(`SCORE: ${score}`, 10, cellSize + 5);
    ctx.fillStyle = '#888';
    ctx.font = `${Math.max(12, cellSize * 0.7)}px monospace`;
    ctx.fillText(`HI: ${highScore}`, 10, cellSize * 1.8);
    
    if (state === 'dead') {
        drawDeathScreen();
    }
    
    ctx.restore();
}

    ctx.fillStyle = '#66ff99';
    ctx.font = `bold ${Math.min(W, H) * 0.08}px monospace`;

function drawDeathScreen() {
    ctx.fillStyle = 'rgba(10, 10, 26, 0.7)';
    ctx.fillRect(0, 0, W, H);
    
    ctx.fillStyle = '#ff4444';
    ctx.font = `bold ${Math.min(W, H) * 0.07}px monospace`;
    ctx.textAlign = 'center';
    ctx.fillText('GAME OVER', W / 2, H * 0.35);

    ctx.font = `${Math.min(W, H) * 0.04}px monospace`;

    if (blink) {

    }

let lastFrame = 0;
function gameLoop(timestamp) {
    const dt = Math.min((timestamp - lastFrame) / 1000, 0.1);
    lastFrame = timestamp;
    
    if (state === 'start' || state === 'dead') animTime += dt;
    else update(dt);
    
    draw();
    requestAnimationFrame(gameLoop);
}

window.addEventListener('keydown', (e) => {
    initAudio();
    
    if (state === 'start') {
        if (e.key === 'Enter' || e.key === ' ') startGame();
        return;
    }
    
    if (state === 'dead') {
        if (e.key === 'r' || e.key === 'R' || e.key === 'Enter' || e.key === ' ') startGame();
        return;
    }
    
    if (state === 'playing') {
        let d = null;
        switch(e.key) {
            case 'ArrowUp': case 'w': case 'W': d = {x:0, y:-1}; break;
            case 'ArrowDown': case 's': case 'S': d = {x:0, y:1}; break;
            case 'ArrowLeft': case 'a': case 'A': d = {x:-1, y:0}; break;
            case 'ArrowRight': case 'd': case 'D': d = {x:1, y:0}; break;
        }
        if (d) {
            // Prevent 180 turn
            if (!(d.x === -dir.x && d.y === -dir.y)) {
                nextDir = d;
            }
        }
        if (e.key === 'r' || e.key === 'R') startGame();
    }
    
    e.preventDefault();
});

requestAnimationFrame(gameLoop);
</script>
