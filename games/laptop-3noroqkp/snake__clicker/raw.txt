const canvas = document.getElementById('gameCanvas');
const ctx = canvas.getContext('2d');
let W, H;

function resize() {
W = window.innerWidth; H = window.innerHeight;

}

let audioCtx = null;
function initAudio() { if (!audioCtx) audioCtx = new (window.AudioContext || webkitAudioContext)(); }
function playSound(freq, dur, type) {
if (!audioCtx) return;
const osc = audioCtx.createOscillator();
const gain = audioCtx.createGain();
osc.type = type || 'sine';
osc.frequency.value = freq;
gain.gain.setValueAtTime(0.15, audioCtx.currentTime);
gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
osc.connect(gain); gain.connect(audioCtx.destination);
osc.start(); osc.stop(audioCtx.currentTime + dur);
}

let game = {
running: false,
mouseX: W/2, mouseY: H/2,
snake: [],
foods: [],
particles: [],
floatTexts: [],
coins: 0,
eaten: 0,
combo: 1.0,
comboTimer: 0,
lastEatTime: 0,
upgrades: { speed: 0, multi: 0, value: 0, auto: 0, magnet: 0 },
autoSpawnTimer: 0,
time: 0
};

const SNAKE_LEN = 12;
const SEG_SPACING = 14;

function resetGame() {
game.coins = 0; game.eaten = 0; game.combo = 1.0; game.comboTimer = 0;
game.upgrades = { speed: 0, multi: 0, value: 0, auto: 0, magnet: 0 };
game.foods = []; game.particles = []; game.floatTexts = [];
game.autoSpawnTimer = 0;
const cx = W/2, cy = H/2;
for (let i = 0; i < SNAKE_LEN; i++) {
game.snake.push({ x: cx - i * SEG_SPACING, y: cy });
}
updateUI();
}

function spawnFood(x, y) {
const value = Math.floor(Math.random() * 3) + 1;
game.foods.push({ x, y, value, radius: 8 + value * 2, pulse: 0, life: 0 });
}

function eatFood(idx) {
const f = game.foods[idx];
const val = Math.floor((f.value * (1 + game.upgrades.value * 0.5)) * game.combo);
game.coins += Math.floor(val);
game.eaten++;

game.comboTimer = 3;
game.combo = Math.min(5.0, game.combo + 0.25);

const colors = ['#7fff7f','#ffd700','#ff7fff','#7fffff'];
for (let i = 0; i < 8; i++) {
game.particles.push({
x: f.x, y: f.y,
vx: (Math.random()-0.5)*4, vy: (Math.random()-0.5)*4 - 1,
life: 1, color: colors[Math.floor(Math.random()*colors.length)]
});
}

game.floatTexts.push({ x: f.x, y: f.y - 20, text: '+' + Math.floor(val), life: 1, vy: -1.5 });
game.foods.splice(idx, 1);
playSound(400 + Math.random()*200, 0.1, 'sine');

if (game.upgrades.multi > 0) {
const head = game.snake[0];
let eaten = 1;
for (let i = game.foods.length - 1; i >= 0 && eaten < game.upgrades.multi; i--) {
const fd = game.foods[i];
const dx = fd.x - head.x, dy = fd.y - head.y;
if (Math.sqrt(dx*dx+dy*dy) < 60 + game.upgrades.magnet * 20) {
spawnFood(fd.x, fd.y);
eatFood(i);
eaten++;
}
}
}
updateUI();
}

function update(dt) {
if (!game.running) return;
game.time += dt;

// Combo decay
if (game.comboTimer > 0) {
game.comboTimer -= dt;
if (game.comboTimer <= 0) { game.combo = 1.0; }
} else {
game.combo = Math.max(1.0, game.combo - dt * 0.5);
}

// Auto spawn
if (game.upgrades.auto > 0) {
game.autoSpawnTimer -= dt;
if (game.autoSpawnTimer <= 0) {
spawnFood(Math.random()*W, Math.random()*H);
game.autoSpawnTimer = 1.5 / game.upgrades.auto;
}
}

// Snake follows mouse
const head = game.snake[0];
const speed = (2 + game.upgrades.speed * 0.8) * dt * 60;
const dx = game.mouseX - head.x;
const dy = game.mouseY - head.y;
const dist = Math.sqrt(dx*dx+dy*dy);
if (dist > 1) {
const move = Math.min(speed, dist);
head.x += (dx/dist) * move;
head.y += (dy/dist) * move;
}

// Snake segments follow
for (let i = 1; i < game.snake.length; i++) {
const prev = game.snake[i-1];
const seg = game.snake[i];
const sdx = seg.x - prev.x;
const sdy = seg.y - prev.y;
const sd = Math.sqrt(sdx*sdx+sdy*sdy) || 1;
const targetDist = SEG_SPACING;
if (sd > targetDist) {
const t = (sd - targetDist) / sd;
seg.x += sdx * t;
seg.y += sdy * t;
} else if (sd < targetDist * 0.5) {
const t = (targetDist - sd) / (sd || 1);
seg.x -= sdx * t * 0.5;
seg.y -= sdy * t * 0.5;
}
}

// Eat food
const eatRange = 20 + game.upgrades.magnet * 8;
for (let i = game.foods.length - 1; i >= 0; i--) {
const f = game.foods[i];
const fdx = f.x - head.x, fdy = f.y - head.y;
if (Math.sqrt(fdx*fdx+fdfy*fdfy) < eatRange + f.radius) {
eatFood(i);
}
f.pulse += dt * 3;
f.life += dt;
if (f.life > 15) game.foods.splice(i, 1);
}

// Particles
for (let i = game.particles.length - 1; i >= 0; i--) {
const p = game.particles[i];
p.x += p.vx; p.y += p.vy; p.vy += 0.1;
p.life -= dt * 2;
if (p.life <= 0) game.particles.splice(i, 1);
}

// Float texts
for (let i = game.floatTexts.length - 1; i >= 0; i--) {
const t = game.floatTexts[i];
t.y += t.vy; t.life -= dt * 1.5;
if (t.life <= 0) game.floatTexts.splice(i, 1);
}
}

function draw() {
ctx.fillStyle = '#0a0a1a';
ctx.fillRect(0, 0, W, H);

// Grid
ctx.strokeStyle = 'rgba(42,110,63,0.1)';
ctx.lineWidth = 1;
for (let x = 0; x < W; x += 40) { ctx.beginPath(); ctx.moveTo(x,0); ctx.lineTo(x,H); ctx.stroke(); }
for (let y = 0; y < H; y += 40) { ctx.beginPath(); ctx.moveTo(0,y); ctx.lineTo(W,y); ctx.stroke(); }

// Foods
for (const f of game.foods) {
const pulse = 1 + Math.sin(f.pulse) * 0.2;
const r = f.radius * pulse;
ctx.beginPath();
ctx.arc(f.x, f.y, r, 0, Math.PI*2);
ctx.fillStyle = `hsl(${(f.value * 60)}, 80%, 50%)`;
ctx.fill();
ctx.strokeStyle = 'rgba(255,255,255,0.4)';
ctx.lineWidth = 2;
ctx.stroke();

// Glow
const grad = ctx.createRadialGradient(f.x, f.y, 0, f.x, f.y, r*2);
grad.addColorStop(0, 'rgba(255,255,100,0.3)');
grad.addColorStop(1, 'rgba(255,255,100,0)');
ctx.fillStyle = grad;
ctx.beginPath(); ctx.arc(f.x, f.y, r*2, 0, Math.PI*2); ctx.fill();

ctx.fillStyle = '#fff';
ctx.font = 'bold 12px sans-serif';
ctx.textAlign = 'center';
ctx.fillText(f.value, f.x, f.y + 4);
}

// Snake
for (let i = game.snake.length - 1; i >= 0; i--) {
const seg = game.snake[i];
const t = i / game.snake.length;
const size = Math.max(4, 10 * (1 - t * 0.5));
const hue = 120 - t * 40;
ctx.beginPath();
ctx.arc(seg.x, seg.y, size, 0, Math.PI*2);
ctx.fillStyle = `hsl(${hue}, 70%, ${50 - t*10}%)`;
ctx.fill();

if (i === 0) {
// Eyes
const dx = game.mouseX - seg.x;
const dy = game.mouseY - seg.y;
const d = Math.sqrt(dx*dx+dy*dy) || 1;
const ex = dx/d * 3, ey = dy/d * 3;
ctx.fillStyle = '#fff';
ctx.beginPath(); ctx.arc(seg.x + ex - 2, seg.y + ey - 2, 2.5, 0, Math.PI*2); ctx.fill();
ctx.beginPath(); ctx.arc(seg.x + ex + 2, seg.y + ey - 2, 2.5, 0, Math.PI*2); ctx.fill();
ctx.fillStyle = '#000';
ctx.beginPath(); ctx.arc(seg.x + ex - 2, seg.y + ey - 1.5, 1, 0, Math.PI*2); ctx.fill();
ctx.beginPath(); ctx.arc(seg.x + ex + 2, seg.y + ey - 1.5, 1, 0, Math.PI*2); ctx.fill();

// Tongue
const tx = seg.x + (dx/d)*12;
const ty = seg.y + (dy/d)*12;
ctx.strokeStyle = '#ff3333';
ctx.lineWidth = 1.5;
ctx.beginPath();
ctx.moveTo(seg.x + (dx/d)*8, seg.y + (dy/d)*8);
ctx.lineTo(tx, ty);
ctx.lineTo(tx - 3, ty - 2);
ctx.moveTo(tx, ty);
ctx.lineTo(tx + 3, ty - 1);
ctx.stroke();
}
}

// Particles
for (const p of game.particles) {
ctx.globalAlpha = p.life;
ctx.fillStyle = p.color;
ctx.beginPath(); ctx.arc(p.x, p.y, 3 * p.life, 0, Math.PI*2); ctx.fill();
ctx.globalAlpha = 1;
}

// Float texts
for (const t of game.floatTexts) {
ctx.globalAlpha = t.life;
ctx.fillStyle = '#ffd700';
ctx.font = 'bold 16px sans-serif';
ctx.textAlign = 'center';
ctx.fillText(t.text, t.x, t.y);
ctx.globalAlpha = 1;
}

// Combo display
if (game.combo > 1.5) {
const cd = document.getElementById('comboDisplay');
cd.textContent = `🔥 x${game.combo.toFixed(1)}`;
cd.style.opacity = '1';
} else {
document.getElementById('comboDisplay').style.opacity = '0';
}
}

let lastTime = 0;

}

else if (upg === 'multi') cost = 100 * (lvl+1);
else if (upg === 'value') cost = 75 * (lvl+1);
else if (upg === 'auto') cost = 200 * (lvl+1);
else if (upg === 'magnet') cost = 150 * (lvl+1);
btn.innerHTML = `${upg==='speed'?'⚡':upg==='multi'?'🍴':upg==='value'?'💰':upg==='auto'?'🔄':'🧲'} Lv${lvl} <span class="cost">${cost}c</span>`;
});
}

document.querySelectorAll('.upgradeBtn').forEach(btn => {

const upg = btn.dataset.upg;
const lvl = game.upgrades[upg];
let cost = 0;
if (upg === 'speed') cost = 50 * (lvl+1);
else if (upg === 'multi') cost = 100 * (lvl+1);
else if (upg === 'value') cost = 75 * (lvl+1);
else if (upg === 'auto') cost = 200 * (lvl+1);

}
