<script>
(function(){
"use strict";

const canvas = document.getElementById('gameCanvas');
const ctx = canvas.getContext('2d');
let W, H;

function resize() {
W = window.innerWidth; H = window.innerHeight;
canvas.width = W; canvas.height = H;
}
window.addEventListener('resize', resize);
resize();

// Audio
let audioCtx = null;
function initAudio() {
if (!audioCtx) {
try { audioCtx = new (window.AudioContext || window.webkitAudioContext)(); } catch(e) {}
}
}

function playShot() {
if (!audioCtx) return;
const o = audioCtx.createOscillator();
const g = audioCtx.createGain();
o.type = 'sawtooth';
o.frequency.setValueAtTime(150, audioCtx.currentTime);
o.frequency.exponentialRampToValueAtTime(40, audioCtx.currentTime + 0.12);
g.gain.setValueAtTime(0.3, audioCtx.currentTime);
g.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.12);
o.connect(g); g.connect(audioCtx.destination);
o.start(); o.stop(audioCtx.currentTime + 0.12);
}

function playZombieHit() {
if (!audioCtx) return;
const o = audioCtx.createOscillator();
const g = audioCtx.createGain();
o.type = 'square';
o.frequency.setValueAtTime(80, audioCtx.currentTime);
o.frequency.exponentialRampToValueAtTime(30, audioCtx.currentTime + 0.15);
g.gain.setValueAtTime(0.2, audioCtx.currentTime);
g.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.15);
o.connect(g); g.connect(audioCtx.destination);
o.start(); o.stop(audioCtx.currentTime + 0.15);
}

function playZombieDeath() {
if (!audioCtx) return;
const o = audioCtx.createOscillator();
const g = audioCtx.createGain();
o.type = 'sawtooth';
o.frequency.setValueAtTime(200, audioCtx.currentTime);
o.frequency.exponentialRampToValueAtTime(50, audioCtx.currentTime + 0.3);
g.gain.setValueAtTime(0.15, audioCtx.currentTime);
g.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.3);
o.connect(g); g.connect(audioCtx.destination);
o.start(); o.stop(audioCtx.currentTime + 0.3);
}

function playPlayerHit() {
if (!audioCtx) return;
const o = audioCtx.createOscillator();
const g = audioCtx.createGain();
o.type = 'sawtooth';
o.frequency.setValueAtTime(100, audioCtx.currentTime);
o.frequency.exponentialRampToValueAtTime(60, audioCtx.currentTime + 0.2);
g.gain.setValueAtTime(0.25, audioCtx.currentTime);
g.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.2);
o.connect(g); g.connect(audioCtx.destination);
o.start(); o.stop(audioCtx.currentTime + 0.2);
}

function playWaveStart() {
if (!audioCtx) return;
const o = audioCtx.createOscillator();
const g = audioCtx.createGain();
o.type = 'sine';
o.frequency.setValueAtTime(440, audioCtx.currentTime);
o.frequency.exponentialRampToValueAtTime(880, audioCtx.currentTime + 0.3);
g.gain.setValueAtTime(0.15, audioCtx.currentTime);
g.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.4);
o.connect(g); g.connect(audioCtx.destination);
o.start(); o.stop(audioCtx.currentTime + 0.4);
}

// Game State
let gameState = 'start'; // start, playing, gameover
let player, bullets, zombies, particles, pickups;
let score, kills, wave, waveTimer, waveZombiesLeft, spawnTimer;
let lastTime = 0;

const keys = {};
let mouseX = 0, mouseY = 0, mouseDown = false;

window.addEventListener('keydown', e => {
keys[e.code] = true;
if (e.code === 'Space') e.preventDefault();
initAudio();
});
window.addEventListener('keyup', e => { keys[e.code] = false; });
window.addEventListener('mousemove', e => { mouseX = e.clientX; mouseY = e.clientY; });
window.addEventListener('mousedown', e => { mouseDown = true; initAudio(); });
window.addEventListener('mouseup', e => { mouseDown = false; });

document.getElementById('start-btn').addEventListener('click', startGame);
document.getElementById('restart-btn').addEventListener('click', startGame);

function startGame() {
initAudio();
if (audioCtx && audioCtx.state === 'suspended') audioCtx.resume();
player = {
x: W / 2, y: H / 2, r: 14, speed: 220,
hp: 100, maxHp: 100, angle: 0,
stamina: 100, maxStamina: 100,
invulnTimer: 0, shootCooldown: 0
};
bullets = []; zombies = []; particles = []; pickups = [];
score = 0; kills = 0; wave = 1;
waveZombiesLeft = getWaveCount(1);
spawnTimer = 0; waveTimer = 3;
gameState = 'playing';
document.getElementById('start-screen').style.display = 'none';
document.getElementById('gameover-screen').style.display = 'none';
document.getElementById('hud').style.display = 'block';
document.getElementById('health-bar-container').style.display = 'block';
playWaveStart();
}

function getWaveCount(w) { return 3 + Math.floor(w * 1.5); }
function getZombieSpeed(w) { return 40 + w * 3; }
function getZombieHp(w) { return 20 + Math.floor(w * 3); }

function spawnZombie() {
const side = Math.floor(Math.random() * 4);
let x, y;
switch(side) {
case 0: x = -20; y = Math.random() * H; break;
case 1: x = W + 20; y = Math.random() * H; break;
case 2: x = Math.random() * W; y = -20; break;
case 3: x = Math.random() * W; y = H + 20; break;
}
const hp = getZombieHp(wave);
zombies.push({
x, y, r: 14 + Math.floor(Math.random() * 6),
hp, maxHp: hp, speed: getZombieSpeed(wave) * (0.8 + Math.random() * 0.4),
angle: 0, hitFlash: 0, wobble: Math.random() * Math.PI * 2
});
}

function spawnParticles(x, y, color, count) {
for (let i = 0; i < count; i++) {
const a = Math.random() * Math.PI * 2;
const s = 50 + Math.random() * 150;
particles.push({
x, y, vx: Math.cos(a) * s, vy: Math.sin(a) * s,
life: 0.4 + Math.random() * 0.4, maxLife: 0.4 + Math.random() * 0.4,
color, r: 2 + Math.random() * 3
});
}
}

function spawnPickup(x, y) {
pickups.push({ x, y, r: 8, life: 15, type: 'health' });
}

function shoot() {
const spread = (Math.random() - 0.5) * 0.06;
const a = player.angle + spread;
bullets.push({
x: player.x + Math.cos(a) * 20, y: player.y + Math.sin(a) * 20,
vx: Math.cos(a) * 500, vy: Math.sin(a) * 500, r: 3, life: 1.5
});
}

function update(dt) {
if (gameState !== 'playing') return;

// Player movement
let dx = 0, dy = 0;
if (keys['KeyW'] || keys['ArrowUp']) dy -= 1;
if (keys['KeyS'] || keys['ArrowDown']) dy += 1;
if (keys['KeyA'] || keys['ArrowLeft']) dx -= 1;
if (keys['KeyD'] || keys['ArrowRight']) dx += 1;

const sprinting = keys['ShiftLeft'] || keys['ShiftRight'];
let spd = player.speed;
if (sprinting && player.stamina > 0) {
spd *= 1.6;
player.stamina -= 30 * dt;
} else {
player.stamina += 20 * dt;
}
player.stamina = Math.max(0, Math.min(player.maxStamina, player.stamina));

const len = Math.sqrt(dx*dx + dy*dy);
if (len > 0) { dx /= len; dy /= len; }
player.x += dx * spd * dt;
player.y += dy * spd * dt;
player.x = Math.max(player.r, Math.min(W - player.r, player.x));
player.y = Math.max(player.r, Math.min(H - player.r, player.y));

// Aim
player.angle = Math.atan2(mouseY - player.y, mouseX - player.x);

// Shoot
player.shootCooldown -= dt;
if ((mouseDown || keys['Space']) && player.shootCooldown <= 0) {
shoot();
player.shootCooldown = 0.18;
playShot();
}

// Invulnerability
player.invulnTimer -= dt;

// Bullets
for (let i = bullets.length - 1; i >= 0; i--) {
const b = bullets[i];
b.x += b.vx * dt;
b.y += b.vy * dt;
b.life -= dt;
if (b.life <= 0 || b.x < -20 || b.x > W+20 || b.y < -20 || b.y > H+20) {
bullets.splice(i, 1); continue;
}
// Hit zombies
for (let j = zombies.length - 1; j >= 0; j--) {
const z = zombies[j];
const ddx = b.x - z.x, ddy = b.y - z.y;
if (ddx*ddx + ddy*ddy < (b.r + z.r) * (b.r + z.r)) {
z.hp -= 15;
z.hitFlash = 0.15;
playZombieHit();
spawnParticles(z.x, z.y, '#4a2', 3);
bullets.splice(i, 1);
break;
}
}
}

// Zombie death check
for (let i = zombies.length - 1; i >= 0; i--) {
if (zombies[i].hp <= 0) {
const z = zombies[i];
spawnParticles(z.x, z.y, '#2a6', 8);
playZombieDeath();
score += 10 + wave * 2;
kills++;
waveZombiesLeft--;
if (Math.random() < 0.25) spawnPickup(z.x, z.y);
zombies.splice(i, 1);
}
}

// Wave spawning
spawnTimer -= dt;
if (spawnTimer <= 0 && waveZombiesLeft > 0) {
const batch = Math.min(waveZombiesLeft, 1 + Math.floor(Math.random() * 2));
for (let i = 0; i < batch; i++) spawnZombie();
waveZombiesLeft -= batch;
spawnTimer = Math.max(0.4, 1.5 - wave * 0.1);
}

// Wave complete
if (waveZombiesLeft <= 0 && zombies.length === 0) {
wave++;
waveZombiesLeft = getWaveCount(wave);
waveTimer = 3;
playWaveStart();
}

// Update zombies
for (const z of zombies) {
z.wobble += dt * 5;
z.hitFlash -= dt;
const a = Math.atan2(player.y - z.y, player.x - z.x);
z.angle = a;
z.x += Math.cos(a) * z.speed * dt + Math.sin(z.wobble) * 10 * dt;
z.y += Math.sin(a) * z.speed * dt + Math.cos(z.wobble) * 10 * dt;

// Zomb hit player
if (player.invulnTimer <= 0) {
const ddx = z.x - player.x, ddy = z.y - player.y;
if (ddx*ddx + ddy*ddy < (z.r + player.r) * (z.r + player.r)) {
player.hp -= 15;
player.invulnTimer = 0.8;
playPlayerHit();
spawnParticles(player.x, player.y, '#f44', 5);
if (player.hp <= 0) { player.hp = 0; gameOver(); }
}
}
}

// Pickups
for (let i = pickups.length - 1; i >= 0; i--) {
const p = pickups[i];
p.life -= dt;
if (p.life <= 0) { pickups.splice(i, 1); continue; }
const ddx = p.x - player.x, ddy = p.y - player.y;
if (ddx*ddx + ddy*ddy < (p.r + player.r)*(p.r + player.r)) {
player.hp = Math.min(player.maxHp, player.hp + 25);
spawnParticles(p.x, p.y, '#0f0', 5);
pickups.splice(i, 1);
}
}

// Particles
for (let i = particles.length - 1; i >= 0; i--) {
const p = particles[i];
p.x += p.vx * dt;
p.y += p.vy * dt;
p.life -= dt;
if (p.life <= 0) particles.splice(i, 1);
}

// HUD
document.getElementById('hud-score').textContent = 'Score: ' + score;
document.getElementById('hud-wave').textContent = 'Wave: ' + wave;
document.getElementById('hud-kills').textContent = 'Kills: ' + kills;
document.getElementById('health-bar').style.width = (player.hp / player.maxHp * 100) + '%';
}

function gameOver() {
gameState = 'gameover';
document.getElementById('final-score').textContent = score;
document.getElementById('final-wave').textContent = wave;
document.getElementById('final-kills').textContent = kills;
document.getElementById('gameover-screen').style.display = 'flex';
document.getElementById('hud').style.display = 'none';
document.getElementById('health-bar-container').style.display = 'none';
}

function draw() {
ctx.fillStyle = '#1a1a2e';
ctx.fillRect(0, 0, W, H);

if (gameState !== 'playing' && gameState !== 'gameover') return;

// Grid
ctx.strokeStyle = 'rgba(50,50,80,0.3)';
ctx.lineWidth = 1;
const gs = 60;
for (let x = 0; x < W; x += gs) { ctx.beginPath(); ctx.moveTo(x,0); ctx.lineTo(x,H); ctx.stroke(); }
for (let y = 0; y < H; y += gs) { ctx.beginPath(); ctx.moveTo(0,y); ctx.lineTo(W,y); ctx.stroke(); }

// Pickups
for (const p of pickups) {
ctx.fillStyle = '#2ecc71';
ctx.beginPath();
ctx.arc(p.x, p.y, p.r, 0, Math.PI*2);
ctx.fill();
ctx.fillStyle = '#fff';
ctx.font = 'bold 12px sans-serif';
ctx.textAlign = 'center';
ctx.fillText('+', p.x, p.y + 4);
}

// Zombies
for (const z of zombies) {
ctx.save();
ctx.translate(z.x, z.y);
// Body
if (z.hitFlash > 0) ctx.fillStyle = '#fff';
else if (z.hp < z.maxHp * 0.3) ctx.fillStyle = '#1a4a1a';
else ctx.fillStyle = '#2d5a2d';
ctx.beginPath();
ctx.arc(0, 0, z.r, 0, Math.PI*2);
ctx.fill();
// Eyes
const ex = Math.cos(z.angle) * 6;
const ey = Math.sin(z.angle) * 6;
ctx.fillStyle = z.hitFlash > 0 ? '#f00' : '#ff6';
ctx.beginPath(); ctx.arc(ex - 4, ey - 2, 3, 0, Math.PI*2); ctx.fill();
ctx.beginPath(); ctx.arc(ex + 4, ey - 2, 3, 0, Math.PI*2); ctx.fill();
// HP bar
if (z.hp < z.maxHp) {
ctx.fillStyle = 'rgba(0,0,0,0.5)';
ctx.fillRect(-15, -z.r - 8, 30, 4);
ctx.fillStyle = '#e74c3c';
ctx.fillRect(-15, -z.r - 8, 30 * (z.hp / z.maxHp), 4);
}
ctx.restore();
}

// Bullets
for (const b of bullets) {
ctx.fillStyle = '#ffdd44';
ctx.beginPath();
ctx.arc(b.x, b.y, b.r, 0, Math.PI*2);
ctx.fill();
ctx.fillStyle = 'rgba(255,221,68,0.3)';
ctx.beginPath();
ctx.arc(b.x, b.y, b.r * 2, 0, Math.PI*2);
ctx.fill();
}

// Player
if (gameState === 'playing') {
ctx.save();
ctx.translate(player.x, player.y);
// Invuln flash
if (player.invulnTimer > 0 && Math.floor(player.invulnTimer * 10) % 2 === 0) {
ctx.globalAlpha = 0.5;
}
// Body
ctx.fillStyle = '#3498db';
ctx.beginPath();
ctx.arc(0, 0, player.r, 0, Math.PI*2);
ctx.fill();
// Gun
ctx.strokeStyle = '#ddd';
ctx.lineWidth = 4;
ctx.beginPath();
ctx.moveTo(Math.cos(player.angle) * 10, Math.sin(player.angle) * 10);
ctx.lineTo(Math.cos(player.angle) * 24, Math.sin(player.angle) * 24);
ctx.stroke();
// Eye
ctx.fillStyle = '#fff';
ctx.beginPath();
ctx.arc(Math.cos(player.angle)*5, Math.sin(player.angle)*5 - 3, 3, 0, Math.PI*2);
ctx.fill();
ctx.restore();

// Crosshair
ctx.strokeStyle = 'rgba(255,100,100,0.7)';
ctx.lineWidth = 1.5;
const cx = mouseX, cy = mouseY;
ctx.beginPath(); ctx.moveTo(cx-10,cy); ctx.lineTo(cx+10,cy); ctx.stroke();
ctx.beginPath(); ctx.moveTo(cx,cy-10); ctx.lineTo(cx,cy+10); ctx.stroke();
}

// Particles
for (const p of particles) {
const a = p.life / p.maxLife;
ctx.globalAlpha = a;
ctx.fillStyle = p.color;
ctx.beginPath();
ctx.arc(p.x, p.y, p.r * a, 0, Math.PI*2);
ctx.fill();
}
ctx.globalAlpha = 1;

// Stamina bar (small)
if (gameState === 'playing') {
ctx.fillStyle = 'rgba(0,0,0,0.5)';
ctx.fillRect(player.x - 20, player.y + 22, 40, 4);
ctx.fillStyle = '#f39c12';
ctx.fillRect(player.x - 20, player.y + 22, 40 * (player.stamina / player.maxStamina), 4);
}

// Wave announcement
if (waveTimer > 0 && gameState === 'playing') {
ctx.fillStyle = 'rgba(255,255,255,' + Math.min(waveTimer, 1) + ')';
ctx.font = 'bold clamp(1.5rem, 3vw, 2.5rem) sans-serif';
ctx.textAlign = 'center';
ctx.fillText('WAVE ' + wave, W/2, H/2 - 50);
}
}

function gameLoop(timestamp) {
const dt = Math.min((timestamp - lastTime) / 1000, 0.05);
lastTime = timestamp;
update(dt);
draw();
requestAnimationFrame(gameLoop);
}

// Wave timer update in main loop
(function waveUpdate() {
if (gameState === 'playing') {
waveTimer -= 1/60;
}
})();

requestAnimationFrame(gameLoop);
})();
</script>
