    if (!audioCtx) return;

// Physics
function circleLineCollision(cx, cy, r, x1, y1, x2, y2) {
    let dx = x2 - x1, dy = y2 - y1;
    let len = Math.sqrt(dx * dx + dy * dy);
    let nx = dy / len, ny = -dx / len;

    // Distance from point to line segment
    let px = cx - x1, py = cy - y1;
    let t = px * (x2 - x1) / (len * len) + py * (y2 - y1) / (len * len);
    t = Math.max(0, Math.min(1, t));

    let closestX = x1 + t * dx;
    let closestY = y1 + t * dy;

    let distX = cx - closestX;
    let distY = cy - closestY;
    let dist = Math.sqrt(distX * distX + distY * distY);

    if (dist < r) {
        let overlap = r - dist;
        let pushX = (distX / (dist || 1)) * overlap;
        let pushY = (distY / (dist || 1)) * overlap;
        return { hit: true, nx: (distX / (dist || 1)), ny: (distY / (dist || 1)), overlap };
    }
    return { hit: false };
}

function updatePhysics() {
    if (!ball.active) return;

    ball.vy += GRAVITY;
    ball.vx *= FRICTION;
    ball.vy *= FRICTION;

    ball.x += ball.vx;
    ball.y += ball.vy;

    // Wall collisions
    for (let w of walls) {
        let col = circleLineCollision(ball.x, ball.y, ball.r, w.x1, w.y1, w.x2, w.y2);
        if (col.hit) {
            ball.x += col.nx * col.overlap;
            ball.y += col.ny * col.overlap;
            let dot = ball.vx * col.nx + ball.vy * col.ny;
            ball.vx -= 1.8 * dot * col.nx;
            ball.vy -= 1.8 * dot * col.ny;
            if (Math.abs(dot) > 3) sndBounce();
        }
    }

    // Bumper collisions
    for (let b of bumpers) {
        let dx = ball.x - b.x, dy = ball.y - b.y;
        let dist = Math.sqrt(dx * dx + dy * dy);
        if (dist < ball.r + b.r) {
            let nx = dx / (dist || 1), ny = dy / (dist || 1);
            let overlap = ball.r + b.r - dist;
            ball.x += nx * overlap;
            ball.y += ny * overlap;
            let dot = ball.vx * nx + ball.vy * ny;
            if (dot < 0) {
                ball.vx -= 2.5 * dot * nx;
                ball.vy -= 2.5 * dot * ny;
                // Bumper boost
                ball.vx += nx * 4;
                ball.vy += ny * 4;
            }
            score += 10;
            sndBounce();
        }
    }

    // Target detection
    for (let t of targets) {
        if (!t.lit && ball.x + ball.r > t.x && ball.x - ball.r < t.x + t.w &&
            ball.y + ball.r > t.y && ball.y - ball.r < t.y + t.h) {
            t.lit = true;
            score += 100;
            sndTarget();
        }
    }

    // Flipper collisions
    updateFlipperCollision(flipperL);
    updateFlipperCollision(flipperR);

    // Ball lost
    if (ball.y > H + 20) {
        ball.active = false;
        ball.launched = false;
        ballsLeft--;
        sndLose();
        if (ballsLeft <= 0) {
            setTimeout(() => { state = STATE.GAME_OVER; }, 500);
        }
    }

    // Level complete check
    let allLit = targets.every(t => t.lit);
    if (allLit && ball.active) {
        state = STATE.LEVEL_COMPLETE;
        score += 500;
        sndWin();
    }
}

function updateFlipperCollision(flip) {
    let tipX, tipY;
    if (flip === flipperL) {
        tipX = flip.x + Math.cos(flip.angle) * flip.len;
        tipY = flip.y + Math.sin(flip.angle) * flip.len;
    } else {
        tipX = flip.x + Math.cos(flip.angle) * flip.len;
        tipY = flip.y + Math.sin(flip.angle) * flip.len;
    }

    let col = circleLineCollision(ball.x, ball.y, ball.r, flip.x, flip.y, tipX, tipY);
    if (col.hit) {
        ball.x += col.nx * col.overlap;
        ball.y += col.ny * col.overlap;
        let dot = ball.vx * col.nx + ball.vy * col.ny;
        ball.vx -= 1.9 * dot * col.nx;
        ball.vy -= 1.9 * dot * col.ny;
        // Add flipper power if active
        let isLeft = (flip === flipperL);
        let isActive = isLeft ? keys['ArrowLeft'] || keys['KeyA'] : keys['ArrowRight'] || keys['KeyD'];
        if (isActive) {
            ball.vy -= 3;
            ball.vx += (isLeft ? 1.5 : -1.5);
        }
    }
}

function updateFlippers() {
    let lActive = keys['ArrowLeft'] || keys['KeyA'];
    let rActive = keys['ArrowRight'] || keys['KeyD'];

    if (lActive) {
        flipperL.angle += (flipperL.activeAngle - flipperL.angle) * 0.4;
    } else {
        flipperL.angle += (flipperL.restAngle - flipperL.angle) * 0.3;
    }

    if (rActive) {
        flipperR.angle -= (flipperR.activeAngle - flipperR.angle) * 0.4;
    } else {
        flipperR.angle += (flipperR.restAngle - flipperR.angle) * 0.3;
    }

    // Normalize right flipper
    if (flipperR.angle > Math.PI + 0.6) flipperR.angle = Math.PI + 0.5;
    if (flipperR.angle < Math.PI - 0.6) flipperR.angle = Math.PI - 0.5;

    // Plunger
    if (keys['Space'] && !ball.active && !ball.launched) {
        plunger.pulling = true;
        plunger.power = Math.min(plunger.power + 0.03, 1);
        plunger.y -= 2;
    } else if (!keys['Space'] && plunger.pulling) {
        plunger.pulling = false;
        launchBall();
        plunger.y = 700;
        plunger.power = 0;
    }
}

function drawFlipper(f) {
    let tipX = f.x + Math.cos(f.angle) * f.len;
    let tipY = f.y + Math.sin(f.angle) * f.len;
    ctx.strokeStyle = '#ffcc44';
    ctx.lineWidth = 10;
    ctx.lineCap = 'round';
    ctx.beginPath();
    ctx.moveTo(f.x, f.y);
    ctx.lineTo(tipX, tipY);
    ctx.stroke();
    // Base circle
    ctx.fillStyle = '#ffcc44';
    ctx.beginPath();
    ctx.arc(f.x, f.y, 8, 0, Math.PI * 2);
    ctx.fill();
}

function drawMenu() {
    ctx.fillStyle = '#1a1a3a';
    ctx.fillRect(0, 0, W, H);

    // Title
    ctx.fillStyle = '#ffcc00';
    ctx.font = 'bold 28px monospace';
    ctx.textAlign = 'center';
    ctx.fillText('PINBALL', W / 2, 150);
    ctx.fillStyle = '#44ff88';
    ctx.fillText('SOKOBAN', W / 2, 190);

    // Instructions
    ctx.fillStyle = '#aaccff';
    ctx.font = '13px monospace';
    let lines = [
        '',
        'Bounce the ball to light up',
        'all TARGETS (diamond pads)',
        '',
        'Controls:',
        '  Arrow Left / A - Left flipper',
        '  Arrow Right / D - Right flipper',
        '  Hold SPACE - Charge plunger',
        '  Release SPACE - Launch ball!',
        '',
        'Hit all targets before losing',
        'all 3 balls to clear the level',
        '',
        '4 Levels of increasing difficulty',
        '',
    ];
    let y = 250;
    for (let line of lines) {
        ctx.fillText(line, W / 2, y);
        y += 22;
    }

    // Blinking prompt
    if (Math.floor(Date.now() / 500) % 2 === 0) {
        ctx.fillStyle = '#fff';
        ctx.font = 'bold 16px monospace';
        ctx.fillText('Press SPACE to Start', W / 2, H - 80);
    }

    // Decorative ball
    let t = Date.now() / 1000;
    let bx = W / 2 + Math.sin(t * 2) * 50;
    let by = H - 130 + Math.cos(t * 3) * 10;
    let grad = ctx.createRadialGradient(bx - 2, by - 2, 0, bx, by, 8);
    grad.addColorStop(0, '#fff');
    grad.addColorStop(1, '#888');
    ctx.fillStyle = grad;
    ctx.beginPath();
    ctx.arc(bx, by, 8, 0, Math.PI * 2);
    ctx.fill();
}
