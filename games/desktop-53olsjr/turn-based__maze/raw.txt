function genMaze(w,h){ // w,h odd
  const g = Array.from({length:h},()=>Array(w).fill(1));
  const stack=[[1,1]]; g[1][1]=0;
  const dirs=[[2,0],[-2,0],[0,2],[0,-2]];
  while(stack.length){
    const [x,y]=stack[stack.length-1];
    const opts=dirs.map(([dx,dy])=>[x+dx,y+dy,dx,dy]).filter(([nx,ny])=>nx>0&&ny>0&&nx<w-1&&ny<h-1&&g[ny][nx]===1);
    if(!opts.length){stack.pop();continue;}
    const [nx,ny,dx,dy]=opts[(Math.random()*opts.length)|0];
    g[y+dy/2][x+dx/2]=0; g[ny][nx]=0; stack.push([nx,ny]);
  }
  // braid: knock some dead-end walls to create loops
  let knocked=0, target=Math.floor(w*h*0.03);
  for(let i=0;i<target*20 && knocked<target;i++){
    const x=1+((Math.random()*(w-2))|0), y=1+((Math.random()*(h-2))|0);
    if(g[y][x]===1){
      const horiz = g[y][x-1]===0 && g[y][x+1]===0;
      const vert = g[y-1][x]===0 && g[y+1][x]===0;
      if((horiz||vert) && !(horiz&&vert)){ g[y][x]=0; knocked++; }
    }
  }
  return g;
}

function genMaze(w,h){
  const gg=Array.from({length:h},()=>new Array(w).fill(1));
  gg[1][1]=0; const stack=[[1,1]];
  while(stack.length){
    const [x,y]=stack[stack.length-1];
    const opts=[];
    for(const [dx,dy] of [[2,0],[-2,0],[0,2],[0,-2]]){
      const nx=x+dx,ny=y+dy;
      if(nx>0&&ny>0&&nx<w-1&&ny<h-1&&gg[ny][nx]===1)opts.push([nx,ny,dx,dy]);
    }
    if(!opts.length){stack.pop();continue;}
    const o=opts[(Math.random()*opts.length)|0];
    gg[y+o[3]/2][x+o[2]/2]=0; gg[o[1]][o[0]]=0; stack.push([o[0],o[1]]);
  }
  let knocked=0,target=Math.max(2,Math.floor(w*h*0.02));
  for(let i=0;i<80&&knocked<target;i++){
    const x=1+((Math.random()*(w-2))|0), y=1+((Math.random()*(h-2))|0);

      if((hz||vt)&&!(hz&&vt)){gg[y][x]=0;knocked++;}
    }
  }
  return gg;
}

function pickType(n){const r=Math.random();
  if(n===1)return 'gloop';
  if(n===2)return r<0.6?'gloop':'fang';
  if(r<0.4)return 'gloop'; if(r<0.78)return 'fang'; return 'gravemaw';}
function newLevel(n){
  level=n;hasKey=false;keyPos=null;
  H=Math.min(9+(n-1)*2,17); W=Math.min(H+6,25)|1; if(W%2===0)W++;
  g=genMaze(W,H);
  const d=bfs(1,1),cells=[];let maxD=1;
  for(let y=1;y<H-1;y++)for(let x=1;x<W-1;x++){if(g[y][x]===0){cells.push([x,y,d[y][x]]);if(d[y][x]>maxD)maxD=d[y][x];}}
  cells.sort((a,b)=>b[2]-a[2]);
  gatePos={x:cells[0][0],y:cells[0][1]};
  const kc=cells.filter(c=>c[2]>=maxD*0.45&&!(c[0]===gatePos.x&&c[1]===gatePos.y));
  const kk=kc[(Math.random()*kc.length)|0]; keyPos={x:kk[0],y:kk[1]};
  const cand=[];
  for(const [x,y,dd] of cells){
    if(x===1&&y===1)continue;
    if(x===gatePos.x&&y===gatePos.y)continue;
    if(x===keyPos.x&&y===keyPos.y)continue;
    if(dd<3)continue; cand.push([x,y]);
  }
  shuffle(cand);
  const nc=Math.min(6+n,14); coins=[];
  for(let i=0;i<nc&&cand.length;i++){const c=cand.pop();coins.push({x:c[0],y:c[1],v:3+((Math.random()*6)|0),s:Math.random()*6});}
  const nh=n<3?1:2; hitems=[];
  for(let i=0;i<nh&&cand.length;i++){const c=cand.pop();hitems.push({x:c[0],y:c[1]});}
  const ecount=Math.min(1+n,8); enemies=[];
  let epool=shuffle(cand.filter(([x,y])=>d[y][x]>=8));
  if(epool.length<ecount){for(const c of cand)if(!epool.includes(c))epool.push(c);shuffle(epool);}
  for(let i=0;i<ecount&&epool.length;i++){
    const [x,y]=epool.pop(),type=pickType(n),T=ETYPES[type];
    enemies.push({type,x,y,rx:x,ry:y,hp:T.hp,maxHp:T.hp,tick:Math.random()<0.5?1:0,seed:Math.random()*10,alive:true,spot:false});
  }
  P={x:1,y:1,rx:1,ry:1,hp:5,maxHp:5,fx:0,fy:1,hurtT:0,seed:Math.random()*10};
  seen=new Set();lightF=new Float32Array(W*H);parts.length=0;floats.length=0;shake=0;
  wallC=new Array(W*H);floorC=new Array(W*H);
  for(let y=0;y<H;y++)for(let x=0;x<W;x++){const i=y*W+x,h=hsh(x,y);
    if(g[y][x]===1){wallC[i]='rgb('+((36+h*14)|0)+','+((32+h*12)|0)+','+((50+h*18)|0)+')';}
    else{floorC[i]='rgb('+((21+h*7)|0)+','+((17+h*5)|0)+','+((27+h*9)|0)+')';}}
  fit();updateHUD();
}

/* ================= actions (turn-based) ================= */
function act(dx,dy){
  if(state!=='play')return;
  P.fx=dx;P.fy=dy;
  const nx=P.x+dx,ny=P.y+dy;
  if(nx<0||ny<0||nx>=W||ny>=H||g[ny][nx]===1){sfx.bump();return;}
  let spent=false;
  const e=enemies.find(en=>en.alive&&en.x===nx&&en.y===ny);
  if(e){attack(e);spent=true;}
  else{P.x=nx;P.y=ny;sfx.move();onCell();spent=true;}
  if(spent&&state==='play'){enemyPhase();turns++;hTn.textContent=turns;if(P.hp<=0)die();}
}
function waitTurn(){if(state!=='play')return;sfx.wait();enemyPhase();turns++;hTn.textContent=turns;if(P.hp<=0)die();}
function attack(e){
  e.hp--;sfx.hit();shake=Math.max(shake,0.15);
  const col=e.type==='gloop'?'#8fd45f':e.type==='fang'?'#ff5d5d':'#cfc3e0';
  burst(e.x,e.y,col,10,2);float(e.x,e.y,'-1','#ffffff');
  if(e.hp<=0){e.alive=false;kills++;const sc=ETYPES[e.type].score;gold+=sc;sfx.kill();
    burst(e.x,e.y,col,22,3);log('You slay the '+ETYPES[e.type].name+' (+'+sc+' embers).','good');updateHUD();}
}
function onCell(){
  for(let i=coins.length-1;i>=0;i--){const c=coins[i];
    if(c.x===P.x&&c.y===P.y){coins.splice(i,1);gold+=c.v;sfx.coin();float(P.x,P.y,'+'+c.v,'#ffd45e');
      burst(P.x,P.y,'#ffd45e',8,1.2);log('You pocket '+c.v+' embers.','gold');updateHUD();}}
  for(let i=hitems.length-1;i>=0;i--){const h=hitems[i];
    if(h.x===P.x&&h.y===P.y){hitems.splice(i,1);
      if(P.hp<P.maxHp){P.hp++;sfx.heart();float(P.x,P.y,'+♥','#ff7d92');log('Warmth returns. +1 heart.','good');}
      else{gold+=8;sfx.coin();float(P.x,P.y,'+8','#ffd45e');log('Your heart is full — it crumbles into 8 embers.','gold');}
      updateHUD();}}
  if(keyPos&&P.x===keyPos.x&&P.y===keyPos.y){hasKey=true;keyPos=null;sfx.key();
    burst(P.x,P.y,'#ffd45e',20,2.4);log('✦ The Ember Key is yours. The gate will open.','key');
    toast('THE EMBER KEY','the gate will open');updateHUD();}
  if(gatePos&&P.x===gatePos.x&&P.y===gatePos.y){
    if(!hasKey){sfx.locked();shake=0.5;log('The Ember Gate is sealed. It craves the key.','dmg');}
    else levelClear();
  }
}
function occAt(x,y){return enemies.some(e=>e.alive&&e.x===x&&e.y===y);}
function hitPlayer(dmg,name){P.hp-=dmg;P.hurtT=0.7;shake=Math.max(shake,0.5);sfx.hurt();
  float(P.x,P.y,'-'+dmg,'#ff6b6b');log('The '+name+' hits you for '+dmg+'.','dmg');updateHUD();}
function enemyPhase(){
  if(state!=='play')return;
  const pd=bfs(P.x,P.y);
  for(const e of enemies){
    if(!e.alive)continue;
    const T=ETYPES[e.type];
    if(T.speed>1){e.tick=(e.tick+1)%T.speed;if(e.tick!==0)continue;}
    const d=pd[e.y][e.x];let nx=null,ny=null;
    if(d>=0&&d<=T.aggro){
      for(const [dx,dy] of shuffled(DIRS)){
        const ax=e.x+dx,ay=e.y+dy;
        if(ax<0||ay<0||ax>=W||ay>=H)continue;
        if(g[ay][ax]===0&&pd[ay][ax]===d-1){nx=ax;ny=ay;break;}
      }
    }else{
      const opts=[];
      for(const [dx,dy] of DIRS){const ax=e.x+dx,ay=e.y+dy;
        if(ax<0||ay<0||ax>=W||ay>=H)continue;
        if(g[ay][ax]===1)continue;
        if(ax===P.x&&ay===P.y)continue;
        if(occAt(ax,ay))continue;opts.push([ax,ay]);}
      if(opts.length&&Math.random()<0.6){const o=opts[(Math.random()*opts.length)|0];nx=o[0];ny=o[1];}
    }
    if(nx===null)continue;
    if(nx===P.x&&ny===P.y)hitPlayer(T.dmg,T.name);
    else{e.x=nx;e.y=ny;}
  }
}

  c.beginPath();c.moveTo(x-s*0.035,kh+s*0.02);c.lineTo(x+s*0.035,kh+s*0.02);c.lineTo(x,kh+s*0.11);c.closePath();c.fill();
}

function drawGame(ts,dt){
  const cw=cell*W,chh=cell*H;
  ctx.clearRect(0,0,cw,chh);
  computeLight();
  for(const e of enemies){if(!e.alive||e.spot)continue;
    if(lightF[e.y*W+e.x]>0.25){e.spot=true;log('A '+ETYPES[e.type].name+' stirs in the dark.','sys');}}
  const k=1-Math.exp(-dt*13);
  P.rx+=(P.x-P.rx)*k;P.ry+=(P.y-P.ry)*k;P.hurtT=Math.max(0,P.hurtT-dt);
  for(const e of enemies){if(!e.alive)continue;e.rx+=(e.x-e.rx)*k;e.ry+=(e.y-e.ry)*k;}
  shake*=Math.exp(-dt*5);
  const ox=(Math.random()-0.5)*shake*cell*0.3,oy=(Math.random()-0.5)*shake*cell*0.3;
  ctx.save();ctx.translate(ox,oy);
  for(let y=0;y<H;y++)for(let x=0;x<W;x++){
    const i=y*W+x,X=x*cell,Y=y*cell;
    if(g[y][x]===1){
      ctx.fillStyle=wallC[i];ctx.fillRect(X,Y,cell,cell);
      if(y+1<H&&g[y+1][x]===0){ctx.fillStyle='rgba(255,220,170,0.10)';ctx.fillRect(X,Y+cell-2,cell,2);}
      if(hsh(x*3,y*7)>0.94){ctx.strokeStyle='rgba(0,0,0,.35)';ctx.lineWidth=1;
        ctx.beginPath();ctx.moveTo(X+cell*0.3,Y+cell*0.2);ctx.lineTo(X+cell*0.6,Y+cell*0.55);ctx.lineTo(X+cell*0.45,Y+cell*0.8);ctx.stroke();}
    }else{
      ctx.fillStyle=floorC[i];ctx.fillRect(X,Y,cell,cell);
      const h1=hsh(x,y);
      if(h1<0.16){ctx.fillStyle='rgba(255,230,190,0.05)';
        ctx.fillRect(X+cell*(0.2+h1*4),Y+cell*(0.25+hsh(y,x)*0.5),Math.max(1,cell*0.06),Math.max(1,cell*0.06));}
    }
  }
  for(const c of coins)sprCoin(ctx,(c.x+0.5)*cell,(c.y+0.5)*cell,cell,ts,c.s);
  for(const h of hitems)sprHeart(ctx,(h.x+0.5)*cell,(h.y+0.5)*cell,cell,ts);
  if(keyPos)sprKey(ctx,(keyPos.x+0.5)*cell,(keyPos.y+0.5)*cell,cell,ts);
  if(gatePos)sprGate(ctx,(gatePos.x+0.5)*cell,(gatePos.y+0.5)*cell,cell,ts,hasKey);
  for(const e of enemies){if(!e.alive)continue;
    const sx=(e.rx+0.5)*cell,sy=(e.ry+0.5)*cell;
    const dd=Math.hypot(P.x-e.x,P.y-e.y)||1,lk={x:(P.x-e.x)/dd,y:(P.y-e.y)/dd};
    if(e.type==='gloop')sprGloop(ctx,sx,sy,cell,ts,e.seed,lk);
    else if(e.type==='fang')sprFang(ctx,sx,sy,cell,ts,e.seed,lk);
    else sprGravemaw(ctx,sx,sy,cell,ts,e.seed,lk);
    if(e.maxHp>1&&e.hp<e.maxHp){const n=e.maxHp,gw=cell*0.14;
      for(let i=0;i<n;i++){ctx.fillStyle=i<e.hp?'#ffd45e':'rgba(255,255,255,.15)';
        ctx.fillRect(sx-(n*gw)/2+i*gw+gw*0.1,sy-cell*0.44,gw*0.8,cell*0.07);}}
  }
  if(state!=='dead')sprPlayer(ctx,(P.rx+0.5)*cell,(P.ry+0.5)*cell,cell,ts,P);
  for(let i=parts.length-1;i>=0;i--){const p=parts[i];
    p.life-=dt/p.tot;if(p.life<=0){parts.splice(i,1);continue;}
    p.vy+=3.5*dt;p.x+=p.vx*dt;p.y+=p.vy*dt;
    ctx.globalAlpha=Math.max(0,p.life)*0.9;ctx.fillStyle=p.color;
    ctx.beginPath();ctx.arc(p.x*cell,p.y*cell,Math.max(0.5,p.r*p.life),0,TAU);ctx.fill();}
  ctx.globalAlpha=1;
  for(let i=0;i<W*H;i++){const L=lightF[i];let a;
    if(L>0)a=(1-L)*0.82;else if(seen.has(i))a=0.78;else a=1;
    if(a>0.02){ctx.fillStyle='rgba(6,4,12,'+a.toFixed(3)+')';ctx.fillRect((i%W)*cell,((i/W)|0)*cell,cell,cell);}}
  const px=(P.rx+0.5)*cell,py=(P.ry+0.5)*cell;
  ctx.globalCompositeOperation='lighter';
  const tg=ctx.createRadialGradient(px,py,cell*0.2,px,py,cell*(3.1*flick));
  tg.addColorStop(0,'rgba(255,160,70,'+(0.14*flick).toFixed(3)+')');tg.addColorStop(1,'rgba(255,120,40,0)');
  ctx.fillStyle=tg;ctx.beginPath();ctx.arc(px,py,cell*3.4,0,TAU);ctx.fill();
  ctx.globalCompositeOperation='source-over';
  for(let i=floats.length-1;i>=0;i--){const f=floats[i];
    f.life-=dt*1.05;f.y-=dt*0.8;if(f.life<=0){floats.splice(i,1);continue;}
    ctx.globalAlpha=Math.max(0,f.life);
    ctx.font='700 '+Math.max(12,(cell*0.42)|0)+'px ui-monospace,Consolas,monospace';
    ctx.textAlign='center';ctx.fillStyle=f.color;
    ctx.fillText(f.txt,(f.x+0.5)*cell,f.y*cell);}
  ctx.globalAlpha=1;
  if(vig){ctx.fillStyle=vig;ctx.fillRect(0,0,cw,chh);}
  ctx.restore();
}
