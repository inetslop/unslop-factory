    const n=found.length;
    const pts=n*10*depth+(n>=6?(n-6)*5*depth:0);
    score+=pts;if(score>best)best=score;
    bestEl.textContent=best;bump(scoreEl);
    cleared+=n;depthCheck();
    let cx=0,cy=0;for(const cl of found){cx+=bx+cl.c*T;cy+=by+cl.r*T;}cx/=n;cy/=n;
    addFloat(cx,cy-T*0.3,'+'+pts,Math.max(15,T*(n>=6?0.46:0.32)),n>=6?'#ffd166':'#ffe9c4');
    if(n>=6)addFloat(cx,cy-T*0.78,'CASCADE ×'+depth,Math.max(11,T*0.24),'#ff9d3b');
  }
  firstClick=false;
}
function depthCheck(){
  while(cleared>=nextAt){
    depth++;density=Math.min(density+0.02,0.36);nextAt+=DIFFS[diff].every;
    hudDepth();sDepth();toast('DEPTH '+depth+' · CHARGES +2%');
  }
}
function startSettle(){
  for(let c=0;c<COLS;c++){
    const pins=[];
    for(let r=0;r<ROWS;r++)if(grid[r][c].flagged&&!grid[r][c].revealed)pins.push(r);
    const segs=[];let s=0;
    for(const p of pins){segs.push([s,p-1]);s=p+1;}
    segs.push([s,ROWS-1]);
    const newCol=new Array(ROWS);
    for(const sg of segs){
      const a=sg[0],b=sg[1];if(a>b)continue;
      const list=[];
      for(let r=a;r<=b;r++){const cell=grid[r][c];if(!cell.revealed&&!cell.flagged)list.push(cell);}
      const L=list.length,top=b-L+1;
      for(let i=0;i<L;i++){
        const nr=top+i,cell=list[i];
        cell.r=nr;cell.c=c;cell.ty=by+nr*T;
        cell.settled=(cell.vy===0&&Math.abs(cell.py-cell.ty)<1);
        newCol[nr]=cell;
      }
      let k=0;
      for(let r=a;r<top;r++){
        const above=(!grid[r][c].revealed)||(function(){for(let rr2=r-1;rr2>=0;rr2--)if(!grid[rr2][c].revealed)return true;return false;})();
        newCol[r]={mine:Math.random()<density,flagged:false,revealed:false,num:0,d:0,r,c,
          py:above?by+r*T-T*(2+Math.random()*0.6):by-(1.3+k*0.5)*T-Math.random()*10,
          ty:by+r*T,vy:0,settled:false,crum:0,crumbling:false,flash:0,squash:0,revealStart:-1,seed:(Math.random()*1e9)|0,
          base:'hsl('+(26+((Math.random()*14)|0))+','+(18+((Math.random()*12)|0))+'%,'+(30+((Math.random()*9)|0))+'%)',hi:'hsl(30,26%,52%)',lo:'hsl(30,20%,16%)'};
        k++;
      }
    }
    for(const p of pins){const cell=grid[p][c];cell.ty=by+p*T;cell.settled=true;newCol[p]=cell;}
    for(let r=0;r<ROWS;r++)grid[r][c]=newCol[r];
  }
  phase='settling';
}

/* ================= UPDATE ================= */
function update(dt){
  if(!paused){
    gt+=dt;
    if(down&&down.r>=0&&!down.done&&state==='playing'&&performance.now()-down.wall>340){down.done=true;toggleFlag(down.r,down.c);}
    if(state!=='over'){
      for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++){
        const cell=grid[r][c];
        if(cell.revealed&&!cell.crumbling&&gt>=cell.revealStart){
          cell.crumbling=true;spawnShards(bx+cell.c*T,cell.py);
          if(state==='playing')sTick(cell.d);
        }
      }
    }
    if(state!=='over'&&phase==='revealing'&&gt>=revealDeadline)startSettle();
    let landed=0;const landers=[];
    for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++){
      const cell=grid[r][c];if(cell.revealed)continue;
      if(!cell.settled){
        cell.vy=Math.min(cell.vy+G*dt,T*22);cell.py+=cell.vy*dt;
        if(cell.py>=cell.ty){cell.py=cell.ty;cell.vy=0;cell.settled=true;cell.squash=1;if(landers.length<5)landers.push(cell);}
        landed++;
      }
    }
    if(landers.length){
      if(gt>landCd){sLand(landers.length);landCd=gt+0.08;}
      for(const cl of landers)spawnDust(bx+cl.c*T,cl.py+T);
    }
    if(state==='playing'&&phase==='settling'&&allSettled())finalizeSettle();
    if(state==='menu'){attractT+=dt;if(attractT>0.62){attractT=0;autoReveal();}}
    if(state==='over'&&!panelShown&&gt>=panelAt){panelShown=true;showOverPanel();}
    for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++){
      const cell=grid[r][c];
      if(cell.flash>0)cell.flash=Math.max(0,cell.flash-dt*2.6);
      if(cell.squash>0)cell.squash=Math.max(0,cell.squash-dt*5);
      if(cell.crumbling&&cell.crum<1)cell.crum=Math.min(1,cell.crum+dt/0.16);
    }
    for(const d of dust){d.y-=d.v*dt;d.x+=Math.sin(gt*0.6+d.ph)*6*dt;if(d.y<-6){d.y=H+6;d.x=Math.random()*W;}if(d.x<-8)d.x=W+8;if(d.x>W+8)d.x=-8;}
    for(const p of pebbles){p.y+=p.v*dt;if(p.y>H+12){p.y=-12;p.x=Math.random()*W;}}
    for(let i=parts.length-1;i>=0;i--){
      const p=parts[i];p.life+=dt;
      if(p.life>=p.max){parts.splice(i,1);continue;}
      if(p.g)p.vy+=p.g*dt;p.x+=p.vx*dt;p.y+=p.vy*dt;
      if(p.vr!==undefined)p.rot+=p.vr*dt;
      if(p.type==='ring')p.r+=p.vr*dt;
    }
    for(let i=floats.length-1;i>=0;i--){const f=floats[i];f.t+=dt;f.y-=T*0.7*dt;if(f.t>f.life)floats.splice(i,1);}
    if(shakeT>0)shakeT=Math.max(0,shakeT-dt);
    if(flashA>0)flashA=Math.max(0,flashA-dt*2.2);
  }
  dispScore+=(score-dispScore)*Math.min(1,dt*9);
  if(Math.abs(score-dispScore)<0.6)dispScore=score;
  const sTxt=String(Math.round(dispScore));
  if(sTxt!==lastScoreTxt){lastScoreTxt=sTxt;scoreEl.textContent=sTxt;}
}

    corner(X+o,Y+o,1,1);corner(X+w-o,Y+o,-1,1);corner(X+o,Y+h-o,1,-1);corner(X+w-o,Y+h-o,-1,-1);
  }
  if(hov){ctx.fillStyle='rgba(255,230,180,.1)';rr(X,Y,w,h,rad);ctx.fill();}
  ctx.restore();
}
function drawMine(cx,cy,s,pop){
  pop=pop===undefined?1:pop;
  ctx.save();ctx.translate(cx,cy);ctx.scale(pop,pop);
  const R=s*0.4;
  ctx.strokeStyle='#d8cbb6';ctx.lineWidth=Math.max(2,s*0.13);ctx.lineCap='round';
  for(let a=0;a<8;a++){const an=a*Math.PI/4+Math.PI/8;
    ctx.beginPath();ctx.moveTo(Math.cos(an)*R*0.5,Math.sin(an)*R*0.5);ctx.lineTo(Math.cos(an)*R*1.7,Math.sin(an)*R*1.7);ctx.stroke();}
  ctx.fillStyle='#20242b';ctx.beginPath();ctx.arc(0,0,R,0,7);ctx.fill();
  ctx.strokeStyle='rgba(0,0,0,.6)';ctx.lineWidth=1;ctx.stroke();
  ctx.fillStyle='rgba(255,255,255,.4)';ctx.beginPath();ctx.arc(-R*0.32,-R*0.36,R*0.3,0,7);ctx.fill();
  ctx.restore();
}
function drawParts(){
  for(const p of parts){
    const a=Math.max(0,1-p.life/p.max);
    if(p.type==='shard'){
      ctx.save();ctx.translate(p.x,p.y);ctx.rotate(p.rot||0);ctx.globalAlpha=a;
      ctx.fillStyle=p.col;const s=p.size;ctx.fillRect(-s/2,-s/2,s,s);ctx.restore();
    }else if(p.type==='dust'){
      ctx.globalAlpha=0.25*a;ctx.fillStyle='#c9ae8b';
      ctx.beginPath();ctx.arc(p.x,p.y,p.size*(1+p.life*4),0,7);ctx.fill();
    }else if(p.type==='spark'){
      ctx.globalAlpha=a;ctx.strokeStyle=p.col;ctx.lineWidth=2;
      ctx.beginPath();ctx.moveTo(p.x,p.y);ctx.lineTo(p.x-p.vx*0.03,p.y-p.vy*0.03);ctx.stroke();
    }else if(p.type==='ring'){
      ctx.globalAlpha=a;ctx.strokeStyle='#ffb45e';ctx.lineWidth=3+6*a;
      ctx.beginPath();ctx.arc(p.x,p.y,p.r,0,7);ctx.stroke();
      ctx.globalAlpha=a*0.6;ctx.strokeStyle='#fff2d8';ctx.lineWidth=2;
      ctx.beginPath();ctx.arc(p.x,p.y,p.r*0.72,0,7);ctx.stroke();
    }
  }
  ctx.globalAlpha=1;
}
function drawFloats(){
  for(const f of floats){
    const a=Math.max(0,1-f.t/f.life);
    ctx.globalAlpha=a;ctx.font='700 '+f.size+'px '+MONO;ctx.textAlign='center';
    ctx.shadowColor='rgba(0,0,0,.8)';ctx.shadowBlur=4;
    ctx.fillStyle=f.col;ctx.fillText(f.txt,f.x,f.y);ctx.shadowBlur=0;
  }
  ctx.globalAlpha=1;
}
function draw(){
  ctx.clearRect(0,0,W,H);
  let ox=0,oy=0;
  if(shakeT>0){const k=shakeT/0.6;ox=(Math.random()*2-1)*shakeAmp*k;oy=(Math.random()*2-1)*shakeAmp*k;}
  ctx.save();ctx.translate(ox,oy);
  drawBG();drawFrame();drawPit();
  ctx.save();ctx.beginPath();ctx.rect(pitX,pitY,pitW,pitH);ctx.clip();
  for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++){
    const cell=grid[r][c];
    if(cell.revealed){if(cell.crumbling&&cell.crum<1)drawRock(cell,cell.crum);continue;}
    drawRock(cell,0);
  }
  if(state==='over'){
    for(const m of minesShown){
      const t=gt-m.t0;if(t<=0)continue;
      drawMine(bx+m.cell.c*T+T/2,m.cell.py+T/2,T*0.8,Math.min(1,t/0.16));
    }
    if(hitCell){
      const hx=bx+hitCell.c*T+T/2,hy=hitCell.py+T/2,pu=0.5+0.5*Math.sin(gt*9);
      ctx.fillStyle='rgba(255,60,25,'+(0.18+0.14*pu)+')';
      ctx.beginPath();ctx.arc(hx,hy,T*(0.55+0.1*pu),0,7);ctx.fill();
      drawMine(hx,hy,T*0.9,1);
      ctx.fillStyle='rgba(120,10,5,.45)';ctx.beginPath();ctx.arc(hx,hy,T*0.42,0,7);ctx.fill();
    }
  }
  if(down&&down.r>=0&&!down.done&&state==='playing'){
    const cell=grid[down.r][down.c];
    if(cell&&!cell.revealed){
      const pr=Math.min(1,(performance.now()-down.wall)/340);
      ctx.strokeStyle='rgba(255,180,43,.85)';ctx.lineWidth=3;
      ctx.beginPath();ctx.arc(bx+down.c*T+T/2,cell.py+T/2,T*(0.55-0.22*pr),0,7);ctx.stroke();
    }
  }
  ctx.restore();
  drawParts();drawFloats();
  if(flashA>0){ctx.fillStyle='rgba(255,240,210,'+flashA+')';ctx.fillRect(-24,-24,W+48,H+48);}
  let vg=ctx.createRadialGradient(W/2,H/2,Math.min(W,H)*0.42,W/2,H/2,Math.hypot(W,H)*0.62);
  vg.addColorStop(0,'rgba(0,0,0,0)');vg.addColorStop(1,'rgba(0,0,0,.5)');
  ctx.fillStyle=vg;ctx.fillRect(-24,-24,W+48,H+48);
  ctx.restore();
}
