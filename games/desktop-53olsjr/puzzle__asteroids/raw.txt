function bounce(A,B){
  let dx=B.x-A.x, dy=B.y-A.y; let d=Math.hypot(dx,dy)||0.001;
  const nx=dx/d, ny=dy/d, min=A.r+B.r;
  const overlap=min-d;
  const mA=A.r*A.r, mB=B.r*B.r, tot=mA+mB;
  A.x -= nx*overlap*(mB/tot); A.y -= ny*overlap*(mB/tot);
  B.x += nx*overlap*(mA/tot); B.y += ny*overlap*(mA/tot);
  const rvx=B.vx-A.vx, rvy=B.vy-A.vy;
  const rel=rvx*nx+rvy*ny;
  if(rel<0){

    A.vx -= imp*nx/mA; A.vy -= imp*ny/mA;
    B.vx += imp*nx/mB; B.vy += imp*ny/mB;
  }
}

function bounce(A,B){
  let dx=B.x-A.x, dy=B.y-A.y; let d=Math.hypot(dx,dy)||0.001;
  const nx=dx/d, ny=dy/d, min=A.r+B.r;
  const overlap=min-d;
  const mA=A.r*A.r, mB=B.r*B.r, tot=mA+mB;
  A.x -= nx*overlap*(mB/tot); A.y -= ny*overlap*(mB/tot);
  B.x += nx*overlap*(mA/tot); B.y += ny*overlap*(mA/tot);
  const rvx=B.vx-A.vx, rvy=B.vy-A.vy;
  const rel=rvx*nx+rvy*ny;
  if(rel<0){

    A.vx -= imp*nx/mA; A.vy -= imp*ny/mA;
    B.vx += imp*nx/mB; B.vy += imp*ny/mB;
  }
}

function bounce(A,B){
 const dx=B.x-A.x,dy=B.y-A.y,d=Math.hypot(dx,dy)||.001,nx=dx/d,ny=dy/d,min=A.r+B.r;

}

function hitAsteroid(a,b){
  b.dead=true;
  const h=a.hue;
  if(a.value%2===0&&a.value>=4){
    a.dead=true;
    const half=a.value/2,ba=Math.atan2(b.vy,b.vx),spd=70+a.value*9;
    for(const s of[-1,1]){
      const dir=ba+s*.85+rand(-.2,.2);
      makeAsteroid(a.x+Math.cos(dir)*a.r*.7,a.y+Math.sin(dir)*a.r*.7,half,
        a.vx*.55+Math.cos(dir)*spd,a.vy*.55+Math.sin(dir)*spd);
    }
    burst(a.x,a.y,h,10,160,2.5);ring(a.x,a.y,a.r*.5,a.r*1.7,.4,hs(h,85,60),2);
    if(inPlay())score+=15;

const tractorR=()=>Math.min(210,Math.min(W,H)*.42);
function gameplayUpdate(dt){
  if(chain.timer>0){chain.timer-=dt;if(chain.timer<=0)chain.count=0;}
  if(wavePending){waveTimer-=dt;if(waveTimer<=0){wavePending=false;startWave(wave+1);}}

  }else resup=0;

  if(ship.respawn>0){ship.respawn-=dt;if(ship.respawn<=0)respawnShip();}
  else{
    ship.invuln=Math.max(0,ship.invuln-dt);
    let ax=0,ay=0;
    if(keys.KeyW||keys.ArrowUp)ay--;if(keys.KeyS||keys.ArrowDown)ay++;
    if(keys.KeyA||keys.ArrowLeft)ax--;if(keys.KeyD||keys.ArrowRight)ax++;
    const il=Math.hypot(ax,ay);thrusting=il>0;
    if(il){
      ax/=il;ay/=il;ship.vx+=ax*470*dt;ship.vy+=ay*470*dt;
      for(let i=0;i<2;i++)particles.push({x:ship.x-ax*12+rand(-3,3),y:ship.y-ay*12+rand(-3,3),
        vx:-ax*rand(60,140)+rand(-20,20),vy:-ay*rand(60,140)+rand(-20,20),life:rand(.15,.35),max:.35,size:rand(1,2.4),c:hs(rand(18,40),95,60),drag:1.5});
    }
    const damp=Math.exp(-dt*(il?1.1:1.7));ship.vx*=damp;ship.vy*=damp;
    const sp=Math.hypot(ship.vx,ship.vy);if(sp>340){ship.vx*=340/sp;ship.vy*=340/sp;}
    ship.x+=ship.vx*dt;ship.y+=ship.vy*dt;
    if(ship.x<-SHIP_R)ship.x+=W+2*SHIP_R;else if(ship.x>W+SHIP_R)ship.x-=W+2*SHIP_R;
    if(ship.y<-SHIP_R)ship.y+=H+2*SHIP_R;else if(ship.y>H+SHIP_R)ship.y-=H+2*SHIP_R;
    aimAng=Math.atan2(mouse.y-ship.y,mouse.x-ship.x);
  }

  const beamOn=(keys.ShiftLeft||keys.ShiftRight)&&energy>0&&ship.respawn<=0;
  for(const a of asteroids)a._beam=false;
  if(beamOn){
    energy=Math.max(0,energy-30*dt);
    const TR=tractorR();
    for(const a of asteroids){const dx=ship.x-a.x,dy=ship.y-a.y,d=Math.hypot(dx,dy);
      if(d<TR&&d>1){a._beam=true;let f=430*(1-d/TR)*Math.min(1,26/a.r);if(d<70)f*=d/70;
        a.vx+=dx/d*f*dt;a.vy+=dy/d*f*dt;}}
  }else energy=Math.min(100,energy+13*dt);

  if(mouse.down||keys.Space)fire();

  for(const b of bullets){
    b.x+=b.vx*dt;b.y+=b.vy*dt;b.life-=dt;
    if(b.life<=0||b.x<-30||b.x>W+30||b.y<-30||b.y>H+30){b.dead=true;continue;}
    for(const a of asteroids){if(a.dead)continue;const dx=a.x-b.x,dy=a.y-b.y;
      if(dx*dx+dy*dy<(a.r+4)*(a.r+4)){hitAsteroid(a,b);break;}}
  }
  bullets=bullets.filter(b=>!b.dead);

  if(!wavePending&&asteroids.length===0){

    const bonus=ammo*40+wave*250;score+=bonus;
    banner('WAVE '+wave+' CLEAR','BONUS +'+bonus,2.3);sfx.clear();
  }
}
function moveAsteroids(dt){
  for(const a of asteroids){if(a.dead)continue;a.x+=a.vx*dt;a.y+=a.vy*dt;a.rot+=a.vr*dt;
    if(a.x<-a.r)a.x+=W+2*a.r;else if(a.x>W+a.r)a.x-=W+2*a.r;
    if(a.y<-a.r)a.y+=H+2*a.r;else if(a.y>H+a.r)a.y-=H+2*a.r;}
  asteroids=asteroids.filter(a=>!a.dead);
}
function bounce(A,B){
  const dx=B.x-A.x,dy=B.y-A.y,d=Math.hypot(dx,dy)||.001,nx=dx/d,ny=dy/d,min=A.r+B.r;
  const mA=A.r*A.r,mB=B.r*B.r,tot=mA+mB,ov=min-d;
  A.x-=nx*ov*(mB/tot);A.y-=ny*ov*(mB/tot);B.x+=nx*ov*(mA/tot);B.y+=ny*ov*(mA/tot);
  const rel=(B.vx-A.vx)*nx+(B.vy-A.vy)*ny;
  if(rel<0){const imp=-1.85*rel/(1/mA+1/mB);
    A.vx-=imp*nx/mA;A.vy-=imp*ny/mA;B.vx+=imp*nx/mB;B.vy+=imp*ny/mB;}
}
function collideAsteroids(){
  for(let i=0;i<asteroids.length;i++){const A=asteroids[i];if(A.dead)continue;
    for(let j=i+1;j<asteroids.length;j++){const B=asteroids[j];if(B.dead)continue;
      const dx=B.x-A.x,dy=B.y-A.y,min=A.r+B.r,d2=dx*dx+dy*dy;
      if(d2<min*min&&d2>1e-4){
        if(A.value===B.value&&T-A.birth>.55&&T-B.birth>.55){fuse(A,B);break;}
        else bounce(A,B);
      }}}
  asteroids=asteroids.filter(a=>!a.dead);
}
function shipCollide(){
  if(ship.respawn>0)return;
  for(const a of asteroids){if(a.dead)continue;
    const dx=a.x-ship.x,dy=a.y-ship.y,d=Math.hypot(dx,dy),min=a.r+SHIP_R-2;
    if(d<min&&d>1e-3){
      if(ship.invuln>0){const nx=dx/d,ny=dy/d;a.vx+=nx*170;a.vy+=ny*170;
        a.x=ship.x+nx*(a.r+SHIP_R);a.y=ship.y+ny*(a.r+SHIP_R);}
      else{shipHit();return;}
    }}
}
function update(dt){
  T+=dt;
  shake=Math.max(0,shake-dt*(shake*2.5+3));
  for(const st of starsFar){st.x-=7*dt;st.y+=3*dt;if(st.x<0)st.x+=W;if(st.y>H)st.y-=H;}
  for(const st of starsNear){st.x-=16*dt;st.y+=8*dt;if(st.x<0)st.x+=W;if(st.y>H)st.y-=H;}
  if(!meteor.on&&T>nextMeteor){meteor.on=true;const dir=Math.random()<.5?-1:1;
    meteor.x=rand(W*.2,W*.8);meteor.y=-30;meteor.vx=dir*rand(300,520);meteor.vy=rand(420,640);}
  if(meteor.on){meteor.x+=meteor.vx*dt;meteor.y+=meteor.vy*dt;
    if(meteor.y>H+60||meteor.x<-80||meteor.x>W+80){meteor.on=false;nextMeteor=T+rand(5,11);}}
  for(let i=particles.length-1;i>=0;i--){const p=particles[i];p.life-=dt;if(p.life<=0){particles.splice(i,1);continue;}
    const d=Math.exp(-p.drag*dt);p.vx*=d;p.vy*=d;p.x+=p.vx*dt;p.y+=p.vy*dt;}
  for(let i=rings.length-1;i>=0;i--){const r=rings[i];r.life-=dt;if(r.life<=0){rings.splice(i,1);continue;}r.r+=r.vr*dt;}
  for(let i=popups.length-1;i>=0;i--){const p=popups[i];p.life-=dt;p.y-=26*dt;if(p.life<=0)popups.splice(i,1);}

  if(state==='menu'){if(!asteroids.length){attractT-=dt;if(attractT<=0)spawnAttract();}}
  else if(state==='playing')gameplayUpdate(dt);

  moveAsteroids(dt);collideAsteroids();
  if(state==='playing')shipCollide();
  syncHud();
}

**SPLIT CASCADE** turns classic Asteroids into a number puzzle: every rock carries a value, and **even values split in half when shot** (8 → 4 + 4), while **equal values fuse on contact** — chain fuses within the combo window to multiply your score and earn bonus shots. Hold **Shift** to tractor-herd fragments together (energy-limited, and dangerous up close). You get a finite shot budget per wave, so reading the board matters as much as aiming. Controls: WASD/arrows to thrust, mouse to aim, click/Space to fire, P pause, R restart, M mute.
