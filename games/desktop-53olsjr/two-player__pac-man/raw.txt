  }

    for(const p of pacs){if(!p.alive)continue;
      if((g.x-p.x)**2+(g.y-p.y)**2<0.34){
        if(frightT>0){
          const pts=CHAIN[Math.min(chainIdx,3)];chainIdx++;p.score+=pts;
          S.ghost();floaters.push({x:g.x,y:g.y-.4,text:'+'+pts,color:p.color,t:1});
          burst(g.x,g.y,14,'#bcd2ff',3,.6);
          g.mode='house';g.gx=g.homeX;g.gy=9;g.x=g.homeX+.5;g.y=9.5;g.exitT=3;
        }else if(p.invulnT<=0)killPac(p);
        break;
      }}}
  if(state!=='play')return;
  if(dotsLeft<=0)endGame('dots');
  else if(gameT<=0){gameT=0;endGame('time');}
}

/* ============================== RENDER ============================== */
let W=0,H=0,dpr=1,T=16,mx=0,my=0,mcz=null,mzPad=0;
function resize(){
  dpr=Math.min(2,window.devicePixelRatio||1);
  W=window.innerWidth;H=window.innerHeight;
  cv.width=W*dpr;cv.height=H*dpr;cv.style.width=W+'px';cv.style.height=H+'px';
  ctx.setTransform(dpr,0,0,dpr,0,0);
  const hud=78,pad=16;

  mx=Math.round((W-COLS*T)/2);
  my=Math.round(hud+Math.max(0,(H-hud-ROWS*T)/2));

}
function rr(c,x,y,w,h,r){c.moveTo(x+r,y);c.arcTo(x+w,y,x+w,y+h,r);c.arcTo(x+w,y+h,x,y+h,r);c.arcTo(x,y+h,x,y,r);c.arcTo(x,y,x+w,y,r);c.closePath();}
function buildMaze(){
  mzPad=Math.ceil(T*.8)+12;
  mcz=document.createElement('canvas');
  mcz.width=(COLS*T+2*mzPad)*dpr;mcz.height=(ROWS*T+2*mzPad)*dpr;
  const c=mcz.getContext('2d');c.setTransform(dpr,0,0,dpr,0,0);c.translate(mzPad,mzPad);
  // board backdrop
  c.beginPath();rr(c,-T*.6,-T*.6,COLS*T+1.2*T,ROWS*T+1.2*T,T);
  c.fillStyle='rgba(7,11,40,.85)';c.fill();
  c.lineWidth=2;c.strokeStyle='rgba(99,132,255,.4)';c.shadowColor='rgba(80,120,255,.6)';c.shadowBlur=T*.8;c.stroke();c.shadowBlur=0;
  // wall mass
  c.fillStyle='#0d1550';
  for(let y=0;y<ROWS;y++)for(let x=0;x<COLS;x++)
    if(wall[y][x]&&!(x===DOOR.x&&y===DOOR.y))c.fillRect(x*T,y*T,T+.6,T+.6);
  // glowing edges facing corridors
  c.strokeStyle='#5f7dff';c.lineWidth=Math.max(1.5,T*.14);c.lineCap='round';
  c.shadowColor='#7d9bff';c.shadowBlur=T*.4;
  for(let y=0;y<ROWS;y++)for(let x=0;x<COLS;x++){
    if(!wall[y][x]||(x===DOOR.x&&y===DOOR.y))continue;
    const X=x*T,Y=y*T;
    if(y>0&&!wall[y-1][x]){c.beginPath();c.moveTo(X,Y);c.lineTo(X+T,Y);c.stroke();}
    if(y<ROWS-1&&!wall[y+1][x]){c.beginPath();c.moveTo(X,Y+T);c.lineTo(X+T,Y+T);c.stroke();}
    if(x>0&&!wall[y][x-1]){c.beginPath();c.moveTo(X,Y);c.lineTo(X,Y+T);c.stroke();}
    if(x<COLS-1&&!wall[y][x+1]){c.beginPath();c.moveTo(X+T,Y);c.lineTo(X+T,Y+T);c.stroke();}
  }
  // house door
  const dx=DOOR.x*T,dy=DOOR.y*T;
  c.fillStyle='#ff9ad5';c.shadowColor='#ff9ad5';c.shadowBlur=T*.4;
  c.fillRect(dx+1,dy+T*.42,T-2,T*.16);c.shadowBlur=0;
}
function angOf(d){return d.x===1?0:(d.x===-1?Math.PI:(d.y===1?Math.PI/2:-Math.PI/2));}
function drawPac(px,py,r,color,ang,mouth,alpha){
  ctx.save();ctx.globalAlpha=alpha==null?1:alpha;
  ctx.translate(px,py);ctx.rotate(ang);
  ctx.fillStyle=color;ctx.shadowColor=color;ctx.shadowBlur=r*.8;
  ctx.beginPath();ctx.moveTo(0,0);ctx.arc(0,0,r,mouth,Math.PI*2-mouth);ctx.closePath();ctx.fill();
  ctx.restore();
}
function drawGhost(g,px,py){
  const r=T*.44,fr=frightT>0&&g.mode==='out';
  const flash=fr&&frightT<2&&(Math.floor(now*8)%2);
  const body=fr?(flash?'#e8f0ff':'#2b3cff'):g.color;
  const cy=py-r*.1,feetY=py+r*.72,w=r*2/3,ph=Math.sin(now*10+g.i)*r*.06;
  ctx.save();
  ctx.shadowColor=fr?(flash?'#ffffff':'#4658ff'):g.color;ctx.shadowBlur=T*.45;
  ctx.fillStyle=body;ctx.beginPath();
  ctx.arc(px,cy,r,Math.PI,Math.PI*2);
  ctx.lineTo(px+r,feetY+ph);
  for(let k=0;k<3;k++){const cx2=px+r-w*(k+.5);ctx.arc(cx2,feetY+ph,w/2,0,Math.PI,true);}
  ctx.closePath();ctx.fill();
  ctx.shadowBlur=0;
  const ec=(fr&&flash)?'#2b3cff':'#ffffff';
  const ex=r*.36,ey=cy-r*.05,ox=g.dir.x*r*.14,oy=g.dir.y*r*.14;
  ctx.fillStyle=ec;
  ctx.beginPath();ctx.arc(px-ex,ey,r*.26,0,6.283);ctx.arc(px+ex,ey,r*.26,0,6.283);ctx.fill();
  ctx.fillStyle=(fr&&flash)?'#1a2a99':'#101433';
  ctx.beginPath();ctx.arc(px-ex+ox,ey+oy,r*.13,0,6.283);ctx.arc(px+ex+ox,ey+oy,r*.13,0,6.283);ctx.fill();
  ctx.restore();
}
function render(){
  ctx.clearRect(0,0,W,H);
  if(!grid||!mcz)return;
  ctx.save();
  if(shakeT>0){const a=shakeT*12;ctx.translate((Math.random()-.5)*a,(Math.random()-.5)*a);}
  ctx.drawImage(mcz,mx-mzPad,my-mzPad);
  // dots (batched)
  ctx.fillStyle='#ffd9b8';ctx.shadowColor='rgba(255,190,140,.7)';ctx.shadowBlur=T*.3;
  ctx.beginPath();let has=false;
  for(let y=0;y<ROWS;y++)for(let x=0;x<COLS;x++){
    if(grid[y][x]===1){const px=mx+(x+.5)*T,py=my+(y+.5)*T,r=T*.09;ctx.moveTo(px+r,py);ctx.arc(px,py,r,0,6.283);has=true;}
  }
  if(has)ctx.fill();
  // pellets (pulsing)
  for(let y=0;y<ROWS;y++)for(let x=0;x<COLS;x++){
    if(grid[y][x]===2){
      const px=mx+(x+.5)*T,py=my+(y+.5)*T,pr=T*(.26+.05*Math.sin(now*7+x*.7+y));
      ctx.shadowBlur=T*.7;ctx.beginPath();ctx.arc(px,py,pr,0,6.283);ctx.fill();
    }
  }
  ctx.shadowBlur=0;
  // rings
  for(const r of rings){ctx.globalAlpha=Math.max(0,r.t);ctx.strokeStyle=r.color;ctx.lineWidth=2;
    ctx.beginPath();ctx.arc(mx+r.x*T,my+r.y*T,r.r*T,0,6.283);ctx.stroke();}
  ctx.globalAlpha=1;
  // ghosts
  for(const g of ghosts){const bob=g.mode==='house'?Math.sin(now*4+g.bobP)*.1:0;drawGhost(g,mx+g.x*T,my+(g.y+bob)*T);}
  // pacs
  for(const p of pacs){
    if(!p.alive)continue;
    let al=1;if(p.invulnT>0&&state==='play')al=(Math.floor(now*14)%2)?.35:1;
    drawPac(mx+p.x*T,my+p.y*T,T*.42,p.color,angOf(p.dir),.18+.4*Math.abs(Math.sin(now*9+p.ph)),al);
  }
  // death animation
  if(state==='dying'&&deadPac){
    const ph=1-dyingT/1.35;
    const mo=Math.min(1,ph*1.7);
    const sc=ph<.7?1:Math.max(.05,1-(ph-.7)/.3);
    drawPac(mx+deadPac.x*T,my+deadPac.y*T,T*.42*sc,deadPac.color,angOf(deadPac.dir),.3+mo*(Math.PI-.3));
  }
  // particles
  for(const p of parts){ctx.globalAlpha=Math.max(0,p.t/p.life);ctx.fillStyle=p.color;
    ctx.beginPath();ctx.arc(mx+p.x*T,my+p.y*T,T*.07,0,6.283);ctx.fill();}
  ctx.globalAlpha=1;
  // floaters
  for(const f of floaters){
    const px=mx+f.x*T,py=my+f.y*T;
    ctx.globalAlpha=Math.max(0,f.t);
    ctx.font='700 '+Math.round(T*.95)+'px Verdana';ctx.textAlign='center';
    ctx.lineWidth=3;ctx.strokeStyle='rgba(0,0,0,.6)';
    ctx.strokeText(f.text,px,py);ctx.fillStyle=f.color;ctx.fillText(f.text,px,py);
  }
  ctx.globalAlpha=1;
  ctx.restore();
}

**How it plays:** P1 (cyan) steers with **WASD**, P2 (pink) with the **arrow keys**. Both race across a shared glowing maze grabbing dots (10 pts). Pellets (50 pts) turn all three AI ghosts blue for 7 seconds — eat them for 200, 400, 800… in chain. Each player has 3 lives; run out and it's an instant K.O. Otherwise the round ends when the maze clears or the 2:30 clock hits zero, and the higher score wins. **P** pauses, **M** mutes, **R** restarts instantly — no reload needed. The menu even runs a live attract-mode demo behind the title, and best scores persist between sessions.
