/* ================= CONSTANTS ================= */
const COLS=22, ROWS=15;
const BASE_TICK=150, MIN_TICK=76, TICK_DROP=4;
const START_MINES=10, MAX_MINES=42, MAX_CHARGE=5;
const FS_A='#3a4426', FS_B='#343e21', SW_A='#212a19', SW_B='#1d2515';
const NUMC=['','#74b8ff','#8fd96f','#ff7a6b','#c79bff','#ffb454','#5fd8d0','#ff9ad5','#eef1e4'];

/* ================= DOM ================= */
const cv=document.getElementById('game'), ctx=cv.getContext('2d');
const ov=document.getElementById('overlay');
const scoreEl=document.getElementById('score'), bestTagEl=document.getElementById('bestTag');
const mineEl=document.getElementById('mineCount'), lvEl=document.getElementById('lvBadge');
const pipEl=document.getElementById('pips'), muteBtn=document.getElementById('muteBtn');

/* ================= STATE ================= */
let W=0,H=0,dpr=1,cell=24,ox=0,oy=0;
let state='menu', snake=[], segs=[], dir={x:1,y:0}, queue=[];
let growth=0, tickMs=BASE_TICK, acc=0, elapsed=0, steps=0;
let foods=0, score=0, level=1, charges=3;
let mines=new Set(), swept=new Map(), food=null;
let particles=[], popups=[], pulses=[];
let shakeT=0, shakeDur=1, shakeMag=0;
let hitMine=null, deathReason='', winState=false, overAt=0, dieAt=0, menuDead=false;
let best=loadBest(), runStartBest=best;
const key=(x,y)=>x*100+y;

/* ================= AUDIO (WebAudio, starts on first input) ================= */
let AC=null, master=null, nbuf=null, muted=false;
function audio(){
  if(!AC){
    try{
      AC=new (window.AudioContext||window.webkitAudioContext)();
      master=AC.createGain(); master.gain.value=muted?0:0.55; master.connect(AC.destination);
    }catch(e){}
  }
  if(AC&&AC.state==='suspended')AC.resume();
}
function tone(f0,f1,dur,type,vol,delay=0){
  if(!AC||muted)return; const t=AC.currentTime+delay;
  const o=AC.createOscillator(), g=AC.createGain();
  o.type=type; o.frequency.setValueAtTime(Math.max(1,f0),t);
  if(f1&&f1!==f0)o.frequency.exponentialRampToValueAtTime(Math.max(1,f1),t+dur);
  g.gain.setValueAtTime(vol,t); g.gain.exponentialRampToValueAtTime(0.0001,t+dur);
  o.connect(g).connect(master); o.start(t); o.stop(t+dur+0.03);
}
function getNoise(){
  if(!nbuf){ nbuf=AC.createBuffer(1,AC.sampleRate,AC.sampleRate);
    const d=nbuf.getChannelData(0); for(let i=0;i<d.length;i++)d[i]=Math.random()*2-1; }
  return nbuf;
}
function noise(dur,vol,f0,f1,delay=0){
  if(!AC||muted)return; const t=AC.currentTime+delay;
  const s=AC.createBufferSource(); s.buffer=getNoise();
  const fl=AC.createBiquadFilter(); fl.type='lowpass';
  fl.frequency.setValueAtTime(f0,t); fl.frequency.exponentialRampToValueAtTime(Math.max(30,f1),t+dur);
  const g=AC.createGain(); g.gain.setValueAtTime(vol,t); g.gain.exponentialRampToValueAtTime(0.0001,t+dur);
  s.connect(fl).connect(g).connect(master); s.start(t); s.stop(t+dur+0.05);
}
const stepSnd   =()=>{ if(++steps%2)return; tone(92+(steps%4?6:0),70,.05,'square',.035); };
const eatSnd    =()=>{ tone(620,880,.09,'triangle',.18); tone(880,1320,.12,'triangle',.15,.07); };
const pulseSnd  =()=>{ noise(.35,.22,300,1500); tone(220,90,.3,'sine',.12); };
const boomSnd   =()=>{ noise(.7,.85,1200,60); tone(140,30,.55,'sine',.6); noise(.09,.5,4000,900); };
const crashSnd  =()=>{ noise(.28,.5,700,120); tone(190,55,.28,'sawtooth',.3); };
const levelSnd  =()=>{ [523,659,784].forEach((f,i)=>tone(f,f,.09,'square',.12,i*.07)); };
const jingleSnd =()=>{ [392,311,262,196].forEach((f,i)=>tone(f,f*0.98,.22,'triangle',.15,i*.16)); };
const fanfareSnd=()=>{ [392,523,659,784,1046].forEach((f,i)=>tone(f,f,.14,'triangle',.16,i*.09)); };
const uiSnd     =()=>{ tone(740,980,.08,'square',.12); };

/* ================= LAYOUT / PATTERN ================= */
let stripePat=null;
(function makeStripes(){
  const sp=document.createElement('canvas'); sp.width=sp.height=14;
  const g=sp.getContext('2d');
  g.fillStyle='#191b12'; g.fillRect(0,0,14,14);
  g.strokeStyle='#ffd23f'; g.lineWidth=7; g.lineCap='butt';
  g.beginPath();g.moveTo(-5,19);g.lineTo(19,-5);g.stroke();
  g.beginPath();g.moveTo(-5,5);g.lineTo(5,-5);g.stroke();
  g.beginPath();g.moveTo(9,19);g.lineTo(19,9);g.stroke();
  stripePat=ctx.createPattern(sp,'repeat');
})();

function resize(){
  dpr=Math.min(2,window.devicePixelRatio||1);
  W=window.innerWidth; H=window.innerHeight;
  cv.width=W*dpr; cv.height=H*dpr;
  cv.style.width=W+'px'; cv.style.height=H+'px';
  ctx.setTransform(dpr,0,0,dpr,0,0);
  const HUD_H=64, availW=W-28, availH=H-HUD_H-30;
  cell=Math.max(10,Math.floor(Math.min(availW/COLS,availH/ROWS)));
  ox=Math.round((W-cell*COLS)/2);
  oy=Math.round(HUD_H+(availH-cell*ROWS)/2);
}

try{ new ResizeObserver(resize).observe(document.body); }catch(e){}

/* grass twinkle seeds */
const twk=new Array(COLS*ROWS).fill(null);
for(let i=0;i<COLS*ROWS;i++) if(Math.random()<0.16)
  twk[i]={ph:Math.random()*6.283,dx:0.1+Math.random()*0.8,dy:0.1+Math.random()*0.8};

/* ================= WORLD HELPERS ================= */
function adjMines(x,y){
  let c=0;
  for(let dy=-1;dy<=1;dy++)for(let dx=-1;dx<=1;dx++){
    if(!dx&&!dy)continue;
    if(mines.has(key((x+dx+COLS)%COLS,(y+dy+ROWS)%ROWS)))c++;
  }
  return c;
}
function sweep(cx,cy,r){
  for(let dy=-r;dy<=r;dy++)for(let dx=-r;dx<=r;dx++){
    const x=(cx+dx+COLS)%COLS, y=(cy+dy+ROWS)%ROWS, k=key(x,y);
    if(!swept.has(k)) swept.set(k,{n:adjMines(x,y),m:mines.has(k)});
  }
}
function refreshAround(x,y){
  for(let dy=-1;dy<=1;dy++)for(let dx=-1;dx<=1;dx++){
    const xx=(x+dx+COLS)%COLS, yy=(y+dy+ROWS)%ROWS, s=swept.get(key(xx,yy));
    if(s){ if(mines.has(key(xx,yy)))s.m=true; else s.n=adjMines(xx,yy); }
  }
}
function addMine(){
  const hd=snake[0];
  for(let t=0;t<250;t++){
    const x=(Math.random()*COLS)|0, y=(Math.random()*ROWS)|0, k=key(x,y);
    if(mines.has(k))continue;
    if(snake.some(s=>s.x===x&&s.y===y))continue;
    if(food&&food.x===x&&food.y===y)continue;
    if(Math.max(Math.abs(x-hd.x),Math.abs(y-hd.y))<3)continue;
    mines.add(k); refreshAround(x,y); return;
  }
}
function spawnFood(){
  for(let t=0;t<600;t++){
    const x=(Math.random()*COLS)|0, y=(Math.random()*ROWS)|0, k=key(x,y);
    if(mines.has(k))continue;
    if(snake.some(s=>s.x===x&&s.y===y))continue;
    food={x,y}; return;
  }
  food=null;
}

/* ================= GAME FLOW ================= */
function reset(){
  const cx=(COLS/2)|0, cy=(ROWS/2)|0;
  snake=[]; for(let i=0;i<3;i++)snake.push({x:cx-i,y:cy});
  segs=snake.map(s=>({x:s.x,y:s.y,px:s.x,py:s.y}));
  dir={x:1,y:0}; queue=[]; growth=0; foods=0; score=0; level=1; charges=3;
  tickMs=BASE_TICK; acc=0; elapsed=0; steps=0;
  mines=new Set(); swept=new Map(); food=null;
  particles.length=0; popups.length=0; pulses.length=0;
  shakeT=0; hitMine=null; deathReason=''; winState=false; overAt=0; dieAt=0; menuDead=false;
  for(let i=0;i<START_MINES;i++)addMine();
  spawnFood();
  updateHUD(false);
}
function startGame(){
  audio(); runStartBest=best; reset();
  state='playing'; ov.style.display='none'; uiSnd();
}
function pauseGame(){ if(state!=='playing')return; state='paused'; showPause(); }
function resumeGame(){ if(state!=='paused')return; state='playing'; ov.style.display='none'; uiSnd(); }

function doDie(cause,x,y){
  state='dying'; dieAt=performance.now(); deathReason=cause; winState=false;
  if(cause==='mine'){ hitMine=key(x,y); boomSnd(); shakeIt(16,650); explodeAt(x,y); }
  else { crashSnd(); shakeIt(9,420); explodeAt(x,y); }
}
function finalizeOver(){
  state='over'; overAt=performance.now();
  if(score>runStartBest&&score>0){ best=score; saveBest(best); }
  jingleSnd(); showOver(); updateHUD(false);
}
function doWin(){
  winState=true; state='over'; overAt=performance.now();
  if(score>runStartBest&&score>0){ best=score; saveBest(best); }
  fanfareSnd(); showOver(); updateHUD(false);
}
function menuDeath(){
  menuDead=true; acc=0;
  puffAt(snake[0].x,snake[0].y,'#b6e14b');
  setTimeout(()=>{ if(state==='menu')reset(); },750);
}

/* ================= MOVEMENT ================= */
function turnCW(d){return {x:-d.y,y:d.x};}
function turnCCW(d){return {x:d.y,y:-d.x};}
function queueDir(d){
  const last=queue.length?queue[queue.length-1]:dir;
  if(d.x===last.x&&d.y===last.y)return;
  if(d.x===-last.x&&d.y===-last.y)return;
  if(queue.length<3)queue.push(d);
}
function advance(){
  const h=snake[0];
  const nx=(h.x+dir.x+COLS)%COLS, ny=(h.y+dir.y+ROWS)%ROWS, nk=key(nx,ny);
  const onMine=mines.has(nk);
  let selfHit=false; const willGrow=growth>0;
  if(!onMine){
    for(let i=0;i<snake.length;i++){
      if(i===snake.length-1&&!willGrow)continue;
      if(snake[i].x===nx&&snake[i].y===ny){selfHit=true;break;}
    }
  }
  const old=snake.map(s=>({x:s.x,y:s.y}));
  snake.unshift({x:nx,y:ny});
  if(onMine||selfHit)snake.pop();
  else if(willGrow)growth--;
  else snake.pop();
  segs=snake.map((s,i)=>{const p=old[i]||s;return{x:s.x,y:s.y,px:p.x,py:p.y};});

  if(onMine){ state==='menu'?menuDeath():doDie('mine',nx,ny); return; }
  if(selfHit){ state==='menu'?menuDeath():doDie('self',nx,ny); return; }

  sweep(nx,ny,1);
  if(state==='playing')stepSnd();
  if(food&&nx===food.x&&ny===food.y){
    if(state==='playing')eatFood(nx,ny);
    else { growth+=1; puffAt(nx,ny,'#b6e14b'); spawnFood(); }
  }
}
function eatFood(x,y){
  growth+=1; foods++;
  const pts=10*level; score+=pts;
  popupCell(x,y,'+'+pts); eatSnd(); puffAt(x,y,'#ff8a5c');
  const nl=1+Math.floor(foods/3);
  if(nl>level){ level=nl; charges=Math.min(MAX_CHARGE,charges+1); levelSnd(); popupCenter('LEVEL '+level); }
  tickMs=Math.max(MIN_TICK,BASE_TICK-foods*TICK_DROP);
  const target=Math.min(MAX_MINES,START_MINES+Math.floor(foods/2));
  while(mines.size<target)addMine();
  spawnFood();
  updateHUD(nl>level);
  if(snake.length+mines.size>=COLS*ROWS)doWin();
}
function pulse(){
  if(state!=='playing'||charges<=0)return;
  charges--; const h=snake[0];
  sweep(h.x,h.y,3); pulses.push({x:h.x,y:h.y,t:0});
  pulseSnd(); updateHUD(false);
}

/* menu autopilot */
function aiStep(){
  const h=snake[0];
  const cands=[{d:dir,s:-0.6},{d:turnCW(dir),s:0},{d:turnCCW(dir),s:0}];
  let bestD=dir,bestS=1e9;
  for(const c of cands){
    const nx=(h.x+c.d.x+COLS)%COLS, ny=(h.y+c.d.y+ROWS)%ROWS;
    let s=c.s;
    if(snake.some((p,i)=>i<snake.length-1&&p.x===nx&&p.y===ny))s+=1000;
    for(let dy=-1;dy<=1;dy++)for(let dx=-1;dx<=1;dx++){
      const xx=(nx+dx+COLS)%COLS, yy=(ny+dy+ROWS)%ROWS;
      if(snake.some(p=>p.x===xx&&p.y===yy))s+=2;
    }
    if(food)s+=Math.abs(nx-food.x)+Math.abs(ny-food.y);
    if(Math.random()<0.08)s+=4;
    if(s<bestS){bestS=s;bestD=c.d;}
  }
  dir=bestD; advance();
}

/* ================= FX ================= */
function shakeIt(mag,durMs){ shakeMag=mag; shakeDur=durMs; shakeT=durMs; }
function explodeAt(x,y){
  const cx=ox+(x+0.5)*cell, cy=oy+(y+0.5)*cell;
  const cols=['#ffd23f','#ff9a3d','#ff5c4d','#fff1b8'];
  for(let i=0;i<30;i++){
    const a=Math.random()*6.283, sp=1.5+Math.random()*4.5;
    particles.push({type:'spark',x:cx,y:cy,vx:Math.cos(a)*sp,vy:Math.sin(a)*sp-1,
      life:.5+Math.random()*.5,t:0,size:1.5+Math.random()*2.5,col:cols[i&3]});
  }
  for(let i=0;i<16;i++){
    const a=Math.random()*6.283, sp=.4+Math.random()*1.2;
    particles.push({type:'smoke',x:cx,y:cy,vx:Math.cos(a)*sp,vy:Math.sin(a)*sp-0.5,
      life:.9+Math.random()*.8,t:0,r0:4+Math.random()*5,grow:16+Math.random()*12});
  }
  pulses.push({x,y,t:0,shock:true});
}
function puffAt(x,y,col){
  const cx=ox+(x+0.5)*cell, cy=oy+(y+0.5)*cell;
  for(let i=0;i<9;i++){
    const a=Math.random()*6.283, sp=1+Math.random()*2.4;
    particles.push({type:'spark',x:cx,y:cy,vx:Math.cos(a)*sp,vy:Math.sin(a)*sp-.6,
      life:.35+Math.random()*.3,t:0,size:1.5+Math.random()*2,col});
  }
}
function popupCell(x,y,txt){ popups.push({x:ox+(x+.5)*cell,y:oy+y*cell,txt,t:0,life:.9,size:15,color:'#ffe063',big:false}); }
function popupCenter(txt){ popups.push({x:ox+COLS*cell/2,y:oy+ROWS*cell*.32,txt,t:0,life:1.1,size:30,color:'#ffd23f',big:true}); }

const KEYS={ArrowUp:[0,-1],KeyW:[0,-1],ArrowDown:[0,1],KeyS:[0,1],ArrowLeft:[-1,0],KeyA:[-1,0],ArrowRight:[1,0],KeyD:[1,0]};

  }

    if(c==='KeyP'||c==='Enter'||c==='Space')resumeGame();
    else if(c==='KeyR')startGame();

/* ================= MAIN LOOP ================= */
let lastTs=0;
function frame(ts){
  requestAnimationFrame(frame);
  const dt=Math.min(50,ts-(lastTs||ts)); lastTs=ts;
  if(shakeT>0)shakeT=Math.max(0,shakeT-dt);

  if(state==='playing'||state==='menu'){
    if(!(state==='menu'&&menuDead)){
      acc+=dt;
      if(state==='playing')elapsed+=dt;
      let guard=0;
      while(acc>=tickMs&&guard++<4){
        acc-=tickMs;
        if(state==='playing'){
          while(queue.length){ const d=queue.shift(); if(d.x===-dir.x&&d.y===-dir.y)continue; dir=d; break; }
          advance();
        }else aiStep();
      }
    }
  }
  if(state==='dying'&&performance.now()-dieAt>950)finalizeOver();

  for(let i=particles.length-1;i>=0;i--){
    const p=particles[i]; p.t+=dt/1000;
    if(p.t>=p.life){particles.splice(i,1);continue;}
    p.x+=p.vx*dt*0.06; p.y+=p.vy*dt*0.06;
    if(p.type==='spark')p.vy+=dt*0.004;
  }
  for(let i=popups.length-1;i>=0;i--){ popups[i].t+=dt/1000; if(popups[i].t>=popups[i].life)popups.splice(i,1); }
  for(let i=pulses.length-1;i>=0;i--){ pulses[i].t+=dt/450; if(pulses[i].t>=1)pulses.splice(i,1); }

  draw(ts);
}

/* ================= DRAWING ================= */
function draw(ts){
  ctx.clearRect(0,0,W,H);
  let sx=0,sy=0;
  if(shakeT>0){ const m=shakeMag*(shakeT/shakeDur); sx=(Math.random()*2-1)*m; sy=(Math.random()*2-1)*m; }
  ctx.save(); ctx.translate(sx,sy);

  const bw=COLS*cell, bh=ROWS*cell;

  /* board shadow + hazard frame */
  ctx.save();
  ctx.shadowColor='rgba(0,0,0,.5)'; ctx.shadowBlur=26; ctx.shadowOffsetY=10;
  ctx.fillStyle='#0c100a'; ctx.fillRect(ox-10,oy-10,bw+20,bh+20);
  ctx.restore();
  ctx.fillStyle=stripePat; ctx.fillRect(ox-10,oy-10,bw+20,bh+20);
  ctx.strokeStyle='#0a0c07'; ctx.lineWidth=2; ctx.strokeRect(ox-11,oy-11,bw+22,bh+22);

  /* cells */
  ctx.textAlign='center'; ctx.textBaseline='middle';
  const numSize=Math.round(cell*0.52);
  for(let y=0;y<ROWS;y++)for(let x=0;x<COLS;x++){
    const k=key(x,y), s=swept.get(k), px=ox+x*cell, py=oy+y*cell;
    ctx.fillStyle=s?((x+y)&1?SW_A:SW_B):((x+y)&1?FS_A:FS_B);
    ctx.fillRect(px,py,cell+0.7,cell+0.7);
    if(s){
      if(!s.m&&s.n>0){
        ctx.fillStyle=NUMC[s.n];
        ctx.font='bold '+numSize+'px Consolas,Menlo,monospace';
        ctx.fillText(s.n,px+cell/2,py+cell/2+1);
      }
    }else{
      const tw=twk[k];
      if(tw){
        const a=0.04+0.05*(0.5+0.5*Math.sin(ts*0.0012+tw.ph));
        ctx.fillStyle='rgba(220,240,170,'+a.toFixed(3)+')';
        ctx.fillRect(px+tw.dx*cell,py+tw.dy*cell,2.4,2.4);
      }
    }
  }
  ctx.strokeStyle='rgba(0,0,0,.55)'; ctx.lineWidth=2; ctx.strokeRect(ox-1,oy-1,bw+2,bh+2);

  drawFood(ts);
  drawPulses();
  drawSnake(ts);
  drawMines(ts);
  drawParticles();
  drawPopups();

  /* dying flash */
  if(state==='dying'&&deathReason==='mine'){
    const k=1-(performance.now()-dieAt)/950;
    if(k>0){
      ctx.fillStyle='rgba(255,120,40,'+(0.2*k*Math.abs(Math.sin(performance.now()*0.03))).toFixed(3)+')';
      ctx.fillRect(ox,oy,bw,bh);
    }
  }
  ctx.restore();
}

function drawFood(ts){
  if(!food)return;
  const cx=ox+(food.x+0.5)*cell, cy=oy+(food.y+0.5)*cell+Math.sin(ts*0.004)*cell*0.05;
  const r=cell*0.28*(1+0.07*Math.sin(ts*0.006));
  let g=ctx.createRadialGradient(cx,cy,1,cx,cy,cell*0.95);
  g.addColorStop(0,'rgba(255,110,70,.4)'); g.addColorStop(1,'rgba(255,110,70,0)');
  ctx.fillStyle=g; ctx.beginPath(); ctx.arc(cx,cy,cell*0.95,0,7); ctx.fill();
  ctx.strokeStyle='#6b4a22'; ctx.lineWidth=2; ctx.lineCap='round';
  ctx.beginPath(); ctx.moveTo(cx,cy-r); ctx.lineTo(cx+1,cy-r-4); ctx.stroke();
  ctx.save(); ctx.translate(cx+3,cy-r-3); ctx.rotate(-0.5);
  ctx.fillStyle='#7cbf3f'; ctx.beginPath(); ctx.ellipse(0,0,r*0.42,r*0.18,0,0,7); ctx.fill(); ctx.restore();
  ctx.fillStyle='#ff5c4d'; ctx.beginPath(); ctx.arc(cx,cy,r,0,7); ctx.fill();
  ctx.strokeStyle='rgba(90,10,5,.6)'; ctx.lineWidth=1.5; ctx.stroke();
  ctx.fillStyle='rgba(255,255,255,.8)'; ctx.beginPath(); ctx.arc(cx-r*0.35,cy-r*0.35,r*0.2,0,7); ctx.fill();
}

function drawPulses(){
  for(const p of pulses){
    const cx=ox+(p.x+0.5)*cell, cy=oy+(p.y+0.5)*cell;
    if(p.shock){
      const r=cell*(0.4+2.6*p.t), a=(1-p.t);
      ctx.strokeStyle='rgba(255,179,92,'+(a*0.9).toFixed(3)+')';
      ctx.lineWidth=6*a+1; ctx.beginPath(); ctx.arc(cx,cy,r,0,7); ctx.stroke();
      ctx.strokeStyle='rgba(255,80,40,'+(a*0.7).toFixed(3)+')';
      ctx.lineWidth=2; ctx.beginPath(); ctx.arc(cx,cy,r*0.7,0,7); ctx.stroke();
    }else{
      const r=cell*(0.3+3.1*p.t), a=(1-p.t);
      ctx.strokeStyle='rgba(255,210,63,'+(a*0.85).toFixed(3)+')';
      ctx.lineWidth=3; ctx.beginPath(); ctx.arc(cx,cy,r,0,7); ctx.stroke();
      ctx.fillStyle='rgba(255,210,63,'+(0.06*a).toFixed(3)+')';
      ctx.beginPath(); ctx.arc(cx,cy,r,0,7); ctx.fill();
    }
  }
}

function drawSnake(ts){
  const lt=(state==='playing'||state==='menu')?Math.min(1,acc/tickMs):1;
  const n=segs.length; let hx=0,hy=0;
  for(let i=n-1;i>=0;i--){
    const s=segs[i];
    let gx,gy;
    if(Math.abs(s.px-s.x)>1||Math.abs(s.py-s.y)>1){gx=s.x;gy=s.y;}
    else {gx=s.px+(s.x-s.px)*lt; gy=s.py+(s.y-s.py)*lt;}
    const cx=ox+gx*cell+cell/2, cy=oy+gy*cell+cell/2;
    if(i===0){hx=cx;hy=cy;}
    const t=n>1?i/(n-1):0;
    const r=cell*(i===0?0.54:0.5-0.13*t);
    ctx.beginPath(); ctx.arc(cx,cy,r,0,7);
    ctx.fillStyle=i===0?'hsl(72,68%,64%)':'hsl('+(76-10*t)+','+56+'%,'+(58-28*t)+'%)';
    ctx.fill();
    ctx.lineWidth=1.5; ctx.strokeStyle='rgba(8,12,3,.55)'; ctx.stroke();
    ctx.beginPath(); ctx.arc(cx-r*0.3,cy-r*0.34,r*0.32,0,7);
    ctx.fillStyle='rgba(255,255,225,.14)'; ctx.fill();
  }
  /* tongue */
  if(state==='playing'&&(ts%1700)<240){
    const fl=Math.sin(ts*0.06)*cell*0.1;
    const tx=hx+dir.x*(cell*0.78+fl), ty=hy+dir.y*(cell*0.78+fl);
    ctx.strokeStyle='#ff6a55'; ctx.lineWidth=2; ctx.lineCap='round';
    ctx.beginPath();
    ctx.moveTo(hx+dir.x*cell*0.4,hy+dir.y*cell*0.4); ctx.lineTo(tx,ty);
    const fx=-dir.y,fy=dir.x;
    ctx.moveTo(tx,ty); ctx.lineTo(tx+dir.x*cell*0.14+fx*3,ty+dir.y*cell*0.14+fy*3);
    ctx.moveTo(tx,ty); ctx.lineTo(tx+dir.x*cell*0.14-fx*3,ty+dir.y*cell*0.14-fy*3);
    ctx.stroke();
  }
  /* eyes */
  const eo=cell*0.2, fwd=cell*0.14, epx=-dir.y, epy=dir.x;
  for(const sgn of [-1,1]){
    const ex=hx+epx*eo*sgn+fwd*dir.x, ey=hy+epy*eo*sgn+fwd*dir.y;
    ctx.fillStyle='#f4f7e8'; ctx.beginPath(); ctx.arc(ex,ey,cell*0.135,0,7); ctx.fill();
    ctx.fillStyle='#141a08'; ctx.beginPath(); ctx.arc(ex+dir.x*cell*0.045,ey+dir.y*cell*0.045,cell*0.06,0,7); ctx.fill();
  }
}

function drawMines(ts){
  if(state!=='over'&&state!=='dying')return;
  const now=performance.now();
  for(const k of mines){
    const x=(k/100)|0, y=k%100;
    if(state==='dying'){ if(k===hitMine)mine(x,y,1,true,now); continue; }
    if(k===hitMine)mine(x,y,1,true,now);
    else mine(x,y,Math.min(1,(now-overAt)/500),false,now);
  }
}
function mine(x,y,alpha,hit,now){
  const cx=ox+(x+0.5)*cell, cy=oy+(y+0.5)*cell, r=cell*0.34;
  ctx.save(); ctx.globalAlpha=alpha; ctx.translate(cx,cy);
  if(hit){
    ctx.fillStyle='rgba(255,70,40,.4)'; ctx.beginPath(); ctx.arc(0,0,r*1.9,0,7); ctx.fill();
    ctx.strokeStyle='rgba(255,90,50,.85)'; ctx.lineWidth=2;
    ctx.beginPath(); ctx.arc(0,0,r*(1.35+0.15*Math.sin(now*0.02)),0,7); ctx.stroke();
  }
  ctx.strokeStyle=hit?'#2b0f08':'#171a10'; ctx.lineWidth=r*0.4; ctx.lineCap='round';
  for(let i=0;i<8;i++){
    const a=i*Math.PI/4+Math.PI/8;
    ctx.beginPath();
    ctx.moveTo(Math.cos(a)*r*0.5,Math.sin(a)*r*0.5);
    ctx.lineTo(Math.cos(a)*r*1.15,Math.sin(a)*r*1.15);
    ctx.stroke();
  }
  ctx.fillStyle=hit?'#5a1610':'#232819';
  ctx.beginPath(); ctx.arc(0,0,r,0,7); ctx.fill();
  ctx.strokeStyle='rgba(0,0,0,.5)'; ctx.lineWidth=1.5; ctx.stroke();
  ctx.fillStyle='rgba(230,236,200,.75)';
  ctx.beginPath(); ctx.arc(-r*0.3,-r*0.3,r*0.28,0,7); ctx.fill();
  ctx.restore();
}

function drawParticles(){
  for(const p of particles){
    const k=p.t/p.life;
    if(p.type==='smoke'){
      ctx.globalAlpha=(1-k)*0.35;
      ctx.fillStyle='#3a4030';
      ctx.beginPath(); ctx.arc(p.x,p.y,p.r0+p.grow*p.t,0,7); ctx.fill();
      ctx.globalAlpha=1;
    }
  }
  if(particles.some(p=>p.type==='spark')){
    ctx.save(); ctx.globalCompositeOperation='lighter';
    for(const p of particles){
      if(p.type!=='spark')continue;
      const k=p.t/p.life;
      ctx.globalAlpha=1-k;
      ctx.fillStyle=p.col;
      ctx.beginPath(); ctx.arc(p.x,p.y,Math.max(0.5,p.size*(1-k*0.6)),0,7); ctx.fill();
    }
    ctx.restore(); ctx.globalAlpha=1;
  }
}

function drawPopups(){
  for(const p of popups){
    const k=Math.min(1,p.t/p.life), e=1-Math.pow(1-k,3);
    ctx.globalAlpha=1-k*k;
    ctx.font=p.big?('bold '+p.size+'px Impact,"Arial Narrow",sans-serif')
                :('bold '+p.size+'px Consolas,Menlo,monospace');
    const y=p.y-e*28;
    ctx.fillStyle='rgba(0,0,0,.6)'; ctx.fillText(p.txt,p.x+1.5,y+1.5);
    ctx.fillStyle=p.color; ctx.fillText(p.txt,p.x,y);
  }
  ctx.globalAlpha=1;
}
