/* ============================== update ============================== */
function updateFx(dt){
  for(const p of parts){ p.x+=p.vx*dt; p.y+=p.vy*dt; p.vx*=(1-2.5*dt); p.vy*=(1-2.5*dt); p.life-=dt; }
  parts=parts.filter(p=>p.life>0);
  for(const q of pops){ q.y-=34*dt; q.life-=dt*0.9; }
  pops=pops.filter(q=>q.life>0);
  shake=Math.max(0,shake-(24+shake)*3*dt);
  redFlash=Math.max(0,redFlash-1.4*dt);
}
function updatePlay(dt){
  /* timers */
  fireCd-=dt; boostT=Math.max(0,boostT-dt); shieldT=Math.max(0,shieldT-dt);
  tripleT=Math.max(0,tripleT-dt); invulnT=Math.max(0,invulnT-dt);

  /* dying sequence */
  if(dead){
    overDelay-=dt; deadTick+=dt; spd=Math.max(0,spd-320*dt);
    if(deadTick>0.15){ deadTick=0;
      boom(player.x+rand(-34,34),player.y+rand(-26,26),['#ffb03a','#3df2ff','#ff4fa3','#ffffff'][Math.floor(rand(0,4))],false); }
    if(overDelay<=0){ finishOver(); return; }
  }

  /* steering */
  if(!dead){
    if(touching){ player.x+=(touchX-player.x)*Math.min(1,14*dt); player.vx=(touchX-player.x)*8; }
    else{
      const dir=((keys['arrowright']||keys['d'])?1:0)-((keys['arrowleft']||keys['a'])?1:0);
      player.vx+=dir*2600*dt;
    }
  }
  player.vx-=player.vx*Math.min(1,7*dt);
  player.vx=clamp(player.vx,-560,560);
  player.x+=player.vx*dt;
  const minx=roadL+22,maxx=roadR-22;
  if(player.x<minx){ player.x=minx; if(Math.abs(player.vx)>140&&!dead) burst(minx,player.y,6,'#ffd94d',160,2.5); player.vx*=0.2; }
  if(player.x>maxx){ player.x=maxx; if(Math.abs(player.vx)>140&&!dead) burst(maxx,player.y,6,'#ffd94d',160,2.5); player.vx*=0.2; }

  /* speed */
  const up=keys['arrowup']||keys['w'], down=keys['arrowdown']||keys['s'];
  const targetMax=boostT>0?MAXS*1.5:MAXS;
  if(!dead){
    if(touching) spd+=(MAXS*0.86-spd)*Math.min(1,2.2*dt);
    else if(up) spd+=340*dt;
    else if(down) spd-=520*dt;
    else spd-=140*dt;
  }
  spd=clamp(spd,0,targetMax);
  if(boostT<=0&&spd>MAXS) spd=Math.max(MAXS,spd-260*dt);
  topSpd=Math.max(topSpd,spd);

  /* world scroll + passive score */
  worldY+=spd*dt; distance+=spd*dt;
  score+=spd*dt*0.02*(1+spd/MAXS);

  /* firing */
  if((keys[' ']||touching)&&!dead) tryFire();

  /* pads */
  padDist+=spd*dt;
  if(padDist>nextPad){ padDist=0; nextPad=rand(520,1050);
    const lw=RW/3, li=Math.floor(rand(0,3));
    pads.push({x:roadL+lw*(li+0.5),y:-60,w:lw*0.7,used:false}); }
  for(const p of pads) p.y+=spd*dt;
  pads=pads.filter(p=>p.y<H+80);
  for(const p of pads){
    if(!p.used&&!dead&&Math.abs(player.x-p.x)<p.w/2&&Math.abs(player.y-p.y)<34){
      p.used=true; boostT=1.5; spd=Math.min(spd+90,MAXS*1.5); sfx.boost();
      popup(player.x,player.y-46,'BOOST','#b8ff57'); burst(p.x,p.y,20,'#b8ff57',260,3); shake=Math.max(shake,5);
    }
  }

  /* pickups */
  for(const k of pickups){ k.y+=(spd+46)*dt; k.t+=dt*3; }
  pickups=pickups.filter(k=>k.y<H+40);
  for(let i=pickups.length-1;i>=0;i--){ const k=pickups[i];
    const dx=player.x-k.x, dy=player.y-k.y;
    if(dx*dx+dy*dy<30*30&&!dead){
      pickups.splice(i,1); sfx.power(); burst(k.x,k.y,14,k.type==='T'?'#ffd94d':'#3df2ff',200,3);
      if(k.type==='T'){ tripleT=8; popup(k.x,k.y,'TRIPLE SHOT','#ffd94d'); }
      else{ shieldT=6; popup(k.x,k.y,'SHIELD','#3df2ff'); }
    } }

  /* spawn queue */
  if(spawnQueue.length){ spawnQueue[0].t-=dt; if(spawnQueue[0].t<=0){ spawnFormation(); spawnQueue.shift(); } }

  /* formations */
  for(const f of formations){
    const sp=f.speed*(1+(1-f.alive/f.total)*1.4);
    f.x+=f.dir*sp*dt;
    const minX=roadL+4,maxX=roadR-4-f.totalW;
    if(f.x<minX){ f.x=minX; f.dir=1; } else if(f.x>maxX){ f.x=maxX; f.dir=-1; }
    f.y+=(f.baseDrop+spd*0.5)*dt;
    for(const m of f.members){
      if(!m.alive) continue;
      const mx=f.x+m.c*f.spX+SW/2, my=f.y+m.r*38+SH/2;
      if(my>H-46){ /* escaped the track */
        m.alive=false; f.alive--; waveEscapes++; score=Math.max(0,score-25);
        popup(mx,H-70,'ESCAPED -25','#ff8f8f'); burst(mx,my,'#5a6b8c',140,2.5);
        if(waveEscapes>=4){ breach(); }
      }
    }
  }
  formations=formations.filter(f=>f.alive>0);

  /* bullets */
  for(const b of bullets){ b.x+=b.vx*dt; b.y+=b.vy*dt; }
  bullets=bullets.filter(b=>b.y>-30&&b.x>-20&&b.x<W+20);

  /* bullet hits */
  for(const b of bullets){
    if(b.dead) continue;
    for(const f of formations){
      for(const m of f.members){
        if(!m.alive) continue;
        const mx=f.x+m.c*f.spX+SW/2, my=f.y+m.r*38+SH/2;
        const dx=b.x-mx, dy=b.y-my;
        if(dx*dx+dy*dy<20*20){ b.dead=true; killMember(m,f,true); break; }
      }
      if(b.dead) break;
    }
  }
  bullets=bullets.filter(b=>!b.dead);

  /* player collisions */
  if(!dead&&invulnT<=0){
    for(const f of formations){
      let hit=false;
      for(const m of f.members){
        if(!m.alive) continue;
        const mx=f.x+m.c*f.spX+SW/2, my=f.y+m.r*38+SH/2;
        const dx=player.x-mx, dy=player.y-my;
        if(dx*dx+dy*dy<34*34){ hit=true; m.alive=false; f.alive--;
          if(shieldT>0){ kills++; const mult=1+(spd/MAXS)*2; score+=Math.round(50*mult);
            popup(mx,my,'ZAP','#9ff3ff'); burst(mx,my,16,'#3df2ff',240,3); shake=Math.max(shake,6); sfx.boom(false); }
          else{ boom(mx,my,SPRC[m.type],false); damagePlayer(false); }
          break; }
      }
      if(hit) break;
    }
  }

  /* wave clear */
  if(!dead&&spawnQueue.length===0&&formations.length===0){
    if(!waveClearing){
      waveClearing=true;
      const bonus=Math.round((400+wave*150)*(1+spd/MAXS));
      score+=bonus; banner('WAVE CLEAR','BONUS +'+bonus); sfx.fanfare();
    } else { waveDelay+=dt; if(waveDelay>=1.6){ waveClearing=false; waveDelay=0; startWave(wave+1); } }
  }

  /* speed lines drift */
  for(const l of speedLines) l.y=(l.y+spd*3.2*dt)%H;

  updateFx(dt);
}

function drawStars(arr,f){
  for(const s of arr){
    const y=(s.y+worldY*f)%H;
    const a=0.35+0.55*(0.5+0.5*Math.sin(t*2+s.t));
    ctx.fillStyle='rgba(190,230,255,'+a.toFixed(2)+')';
    ctx.fillRect(s.x,y,s.r,s.r);
  }
}
function sideHatch(x0,x1,dir){
  const step=46, off=worldY%step;
  ctx.strokeStyle='rgba(61,242,255,.05)'; ctx.lineWidth=3; ctx.beginPath();
  for(let y=-step+off;y<H+step;y+=step){ ctx.moveTo(x0,y); ctx.lineTo(x1,y+18*dir); }
  ctx.stroke();
}
function drawWorld(){

  for(let y=-40+o2;y<H+40;y+=sp2){ ctx.moveTo(roadL,y); ctx.lineTo(roadR,y); }
  ctx.stroke();

  const lw=RW/3, per=112, off=worldY%per;
  ctx.fillStyle='rgba(220,240,255,.4)';
  for(let li=1;li<3;li++){ const x=roadL+lw*li;
    for(let y=-per+off;y<H;y+=per) ctx.fillRect(x-2.5,y,5,46); }

  /* edge lines + ticks */
  ctx.save(); ctx.lineWidth=3;
  ctx.shadowBlur=12; ctx.shadowColor='#3df2ff'; ctx.strokeStyle='#7ff4ff';
  ctx.beginPath(); ctx.moveTo(roadL+1.5,0); ctx.lineTo(roadL+1.5,H); ctx.stroke();
  ctx.shadowColor='#ffb03a'; ctx.strokeStyle='#ffc46e';

  }

  for(let k=-1;k<Math.ceil(H/pper)+1;k++){
    const y=((k*pper+popf)%(H+160)+H+160)%(H+160)-80;
    const on=((k%2)+2)%2===0;
    [roadL-34,roadR+34].forEach(x=>{
      ctx.fillStyle='#1c2a4d'; ctx.fillRect(x-2,y,4,26);
      ctx.fillStyle=on?'#3df2ff':'#ffb03a';
      ctx.beginPath(); ctx.arc(x,y-4,3.5,0,TAU); ctx.fill();
      ctx.fillStyle='rgba(255,255,255,.8)';
      ctx.beginPath(); ctx.arc(x,y-4,1.4,0,TAU); ctx.fill();
    });
  }

  if(state!=='menu'){
    /* pads */
    ctx.globalCompositeOperation='lighter';
    for(const p of pads){
      const a=p.used?0.25:0.9;
      ctx.strokeStyle='rgba(184,255,87,'+a+')'; ctx.lineWidth=5; ctx.lineJoin='round';
      for(let i=0;i<3;i++){ const y=p.y-14+i*14;
        ctx.beginPath(); ctx.moveTo(p.x-p.w/2,y+9); ctx.lineTo(p.x,y); ctx.lineTo(p.x+p.w/2,y+9); ctx.stroke(); }
    }
    /* pickups */
    for(const k of pickups){
      const c=k.type==='T'?'#ffd94d':'#3df2ff';
      ctx.save(); ctx.translate(k.x,k.y+Math.sin(k.t)*3); ctx.rotate(k.t*0.7);
      ctx.strokeStyle=c; ctx.lineWidth=2.5;
      ctx.beginPath(); ctx.moveTo(0,-12); ctx.lineTo(12,0); ctx.lineTo(0,12); ctx.lineTo(-12,0); ctx.closePath();
      ctx.stroke(); ctx.fillStyle='rgba(255,255,255,.15)'; ctx.fill(); ctx.restore();
      ctx.fillStyle='#fff'; ctx.font='700 11px Consolas,monospace'; ctx.textAlign='center';
      ctx.fillText(k.type,k.x,k.y+4);
    }
    ctx.globalCompositeOperation='source-over';

    /* enemies */
    for(const f of formations){
      const img=f; 
      for(const m of f.members){
        if(!m.alive) continue;
        const s=spr(m.type,f.frame);
        ctx.drawImage(s, f.x+m.c*f.spX+SW/2-s.width/2, f.y+m.r*38+SH/2-s.height/2);
      }
    }

    /* bullets */
    ctx.globalCompositeOperation='lighter';
    for(const b of bullets){
      ctx.strokeStyle='rgba(120,240,255,.9)'; ctx.lineWidth=3;
      ctx.beginPath(); ctx.moveTo(b.x,b.y+16); ctx.lineTo(b.x+b.vx*0.012,b.y); ctx.stroke();
      ctx.fillStyle='#fff'; ctx.beginPath(); ctx.arc(b.x,b.y,2.6,0,TAU); ctx.fill();
    }

    /* player */
    if(!dead&&!(invulnT>0&&(Math.floor(invulnT*14)%2===0))) drawPlayer();

    /* particles */
    for(const p of parts){ const a=Math.max(0,p.life/p.max);
      ctx.globalAlpha=a; ctx.fillStyle=p.c;
      ctx.beginPath(); ctx.arc(p.x,p.y,p.size*a+0.4,0,TAU); ctx.fill(); }
    ctx.globalAlpha=1; ctx.globalCompositeOperation='source-over';

    /* danger line */
    ctx.save(); ctx.setLineDash([12,10]); ctx.strokeStyle='rgba(255,93,93,.4)'; ctx.lineWidth=2;
    ctx.beginPath(); ctx.moveTo(roadL,H-46); ctx.lineTo(roadR,H-46); ctx.stroke(); ctx.restore();
    ctx.fillStyle='rgba(255,140,140,.5)'; ctx.font='9px Consolas,monospace'; ctx.textAlign='right';
    ctx.fillText('TRACK LIMIT',roadR-8,H-52);

    /* popups */
    for(const q of pops){ const a=Math.max(0,q.life/q.max);
      ctx.globalAlpha=Math.min(1,a*1.5); ctx.fillStyle=q.c;
      ctx.font=q.big?'26px Impact,sans-serif':'700 13px Consolas,monospace'; ctx.textAlign='center';
      ctx.fillText(q.txt,q.x,q.y); }
    ctx.globalAlpha=1;
  }
  ctx.restore();

  /* speed lines */
  if(spd>MAXS*0.55){
    const a=(spd/MAXS-0.55)*1.8;
    ctx.lineWidth=2;
    for(const l of speedLines){
      const x=l.side<0?14+l.off%Math.max(6,roadL-40):W-14-(l.off%Math.max(6,roadL-40));
      ctx.strokeStyle='rgba(180,240,255,'+(a*l.a*0.35).toFixed(3)+')';
      ctx.beginPath(); ctx.moveTo(x,l.y); ctx.lineTo(x,l.y+l.len*(0.5+spd/MAXS)); ctx.stroke();
    }
  }
  if(redFlash>0){ ctx.fillStyle='rgba(255,60,60,'+(redFlash*0.22).toFixed(3)+')'; ctx.fillRect(0,0,W,H); }
}
