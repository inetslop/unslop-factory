function tone({f=440,f2,t=0.1,type='sine',g=0.2,delay=0}){
  const a=ac(); if(!a||muted)return;
  const o=a.createOscillator(),gn=a.createGain();
  const t0=a.currentTime+delay;
  o.type=type;o.frequency.setValueAtTime(f,t0);
  if(f2)o.frequency.exponentialRampToValueAtTime(Math.max(1,f2),t0+t);
  gn.gain.setValueAtTime(0.0001,t0);
  gn.gain.exponentialRampToValueAtTime(g,t0+0.008);
  gn.gain.exponentialRampToValueAtTime(0.0001,t0+t);
  o.connect(gn).connect(master);o.start(t0);o.stop(t0+t+0.05);
}
function noise({t=0.2,g=0.3,f=800,q=1,type='lowpass',delay=0,slide}){
  const a=ac(); if(!a||muted)return;
  const n=Math.floor(a.sampleRate*t), buf=a.createBuffer(1,n,a.sampleRate), d=buf.getChannelData(0);
  for(let i=0;i<n;i++)d[i]=(Math.random()*2-1)*(1-i/n);
  const s=a.createBufferSource();s.buffer=buf;
  const fl=a.createBiquadFilter();fl.type=type;fl.frequency.value=f;if(slide)fl.frequency.exponentialRampToValueAtTime(slide,a.currentTime+delay+t);fl.Q.value=q;
  const gn=a.createGain();const t0=a.currentTime+delay;
  gn.gain.setValueAtTime(g,t0);gn.gain.exponentialRampToValueAtTime(0.0001,t0+t);
  s.connect(fl).connect(gn).connect(master);s.start(t0);
}

  }else if(c.flag){
    c.flag=false;c.q=true;flags--;el.classList.remove("flag");el.classList.add("q");

}

function lose(hit){
  if(state!=="play")return;
  state="lose";stopTimer();
  const c=cells[hit],t=tiles[hit];
  c.flag=false;c.q=false;t.classList.remove("flag","q");

  spawnExplosion(cx,cy);shake();flashAt(cx,cy);sfx.boom();vib(70);
  let k=0;
  cells.forEach((cc,i)=>{
    if(cc.mine&&i!==hit){
      setTimeout(()=>{tiles[i].classList.add("rev");tiles[i].querySelector("i").innerHTML=mineSVG(false)},50+k*9);k++;
    }
  });
  cells.forEach((cc,i)=>{
    if(!cc.mine&&cc.flag){
      const el=tiles[i];el.classList.remove("flag");el.classList.add("badflag");
      el.querySelector("i").textContent="✕";
    }
  });

}
function win(){
  if(state!=="play")return;
  state="win";stopTimer();
  let newBest=false,prev=getBest(D.id);
  if(prev===null||Math.ceil(finalTime)<prev){store.set("szb_"+D.id,String(Math.ceil(finalTime)));newBest=true}
  flags=D.mines;updateMineLED();
  let k=0;
  cells.forEach((c,i)=>{
    if(c.mine&&!c.flag){
      setTimeout(()=>{
        c.flag=true;tiles[i].classList.add("flag");sfx.tick(k);
        const b=tiles[i].getBoundingClientRect();
        pings.push({x:b.left+b.width/2,y:b.top+b.height/2,r:1,vr:60,life:0,max:.45,c:"150,230,140"});
      },k*30);k++;
    }
  });
  setTimeout(()=>sfx.win(),260);
  setTimeout(spawnConfetti,380);
  setTimeout(()=>showEnd(true,newBest),Math.min(1700,k*30+450));
}

let lastNow=performance.now();
function loop(now){
  requestAnimationFrame(loop);
  const dt=Math.min(.05,(now-lastNow)/1000);lastNow=now;
  if(timerOn){
    const s=Math.floor((now-t0)/1000);
    if(s!==lastSec){lastSec=s;timeLed.textContent=String(Math.min(999,s)).padStart(3,"0");timeLed.classList.toggle("hot",s>=90)}
  }
  ctx.clearRect(0,0,innerWidth,innerHeight);
  for(let i=pings.length-1;i>=0;i--){
    const p=pings[i];p.life+=dt;p.r+=p.vr*dt;
    if(p.life>p.max){pings.splice(i,1);continue}
    ctx.beginPath();ctx.arc(p.x,p.y,p.r,0,6.283);
    ctx.strokeStyle=`rgba(${p.c},${((1-p.life/p.max)*.5).toFixed(3)})`;
    ctx.lineWidth=1.5;ctx.stroke();
  }
  for(let i=parts.length-1;i>=0;i--){
    const p=parts[i];p.life+=dt;
    if(p.life>p.max){parts.splice(i,1);continue}
    p.vy+=p.g*dt;p.x+=p.vx*dt;p.y+=p.vy*dt;
    const a=1-p.life/p.max;
    if(p.type==="spark"){
      ctx.globalAlpha=a;ctx.fillStyle=p.color;
      ctx.beginPath();ctx.arc(p.x,p.y,Math.max(.4,p.size*a),0,6.283);ctx.fill();
    }else if(p.type==="smoke"){
      p.size+=26*dt;
      ctx.globalAlpha=a*.3;ctx.fillStyle=`rgb(${p.color})`;
      ctx.beginPath();ctx.arc(p.x,p.y,p.size,0,6.283);ctx.fill();
    }else{
      p.rot+=p.vr*dt;
      ctx.save();ctx.globalAlpha=Math.min(1,a*1.4);ctx.translate(p.x,p.y);ctx.rotate(p.rot);
      ctx.fillStyle=p.color;ctx.fillRect(-p.size/2,-p.size*.8,p.size,p.size*1.6);ctx.restore();
    }
  }
  ctx.globalAlpha=1;
}
