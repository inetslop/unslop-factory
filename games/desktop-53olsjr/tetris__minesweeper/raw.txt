function confetti(){const t=G.tile;
  for(let i=0;i<140;i++)G.particles.push({type:'conf',x:G.ox+Math.random()*G.w*t,y:G.oy-20-Math.random()*60,
    vx:(Math.random()-.5)*90,vy:40+Math.random()*130,g:240,rot:Math.random()*6.28,vr:(Math.random()-.5)*9,
    life:2.2+Math.random()*1.6,max:3.8,size:4+Math.random()*5,col:PIECES[PTYPES[(Math.random()*7)|0]].c});}
function floatText(x,y,txt,col){G.floats.push({x:G.ox+x*G.tile+G.tile/2,y:G.oy+y*G.tile-G.tile*.2,txt,col,life:1.25,max:1.25});}
function updateFx(dt){
  for(let i=G.particles.length-1;i>=0;i--){const p=G.particles[i];p.life-=dt;
    if(p.life<=0){G.particles.splice(i,1);continue;}
    if(p.type==='ring'){p.r+=p.vr*dt;continue;}
    p.vy+=p.g*dt;p.x+=p.vx*dt;p.y+=p.vy*dt;if(p.rot!==undefined)p.rot+=p.vr*dt;}
  for(let i=G.floats.length-1;i>=0;i--){const f=G.floats[i];f.life-=dt;f.y-=32*dt;if(f.life<=0)G.floats.splice(i,1);}}

/* ================= render ================= */
function bombIcon(c,cx,cy,r,t,seed){
  c.strokeStyle='#c8a06a';c.lineWidth=Math.max(1.5,r*.3);
  c.beginPath();c.moveTo(cx+r*.5,cy-r*.5);c.lineTo(cx+r*1.0,cy-r*1.1);c.stroke();
  const fl=1+.3*Math.sin(t/90+seed),sr=r*.5*fl;
  c.fillStyle='#ffd23f';star4(c,cx+r*1.05,cy-r*1.15,sr);
  c.fillStyle='#e7edf8';c.beginPath();c.arc(cx,cy,r,0,7);c.fill();
  c.strokeStyle='rgba(20,26,44,.8)';c.lineWidth=Math.max(1,r*.18);c.beginPath();c.arc(cx,cy,r,0,7);c.stroke();
  c.fillStyle='#fff';c.beginPath();c.arc(cx-r*.35,cy-r*.35,r*.28,0,7);c.fill();}

function drawCell(c,x,y,t){
  const cell=G.grid[y*G.w+x],tile=G.tile,g=Math.max(1,tile*.055),px=G.ox+x*tile+g,py=G.oy+y*tile+g,s=tile-2*g;
  if(cell.cov){
    let sc=1;const dt=t-cell.covT;if(dt<180)sc=1+.28*(1-dt/180);
    const ss=s*sc,xp=px+(s-ss)/2,yp=py+(s-ss)/2;
    c.fillStyle=cell.color;rr(c,xp,yp,ss,ss,3);c.fill();
    c.fillStyle='rgba(255,255,255,.34)';c.fillRect(xp+1.5,yp+1.5,ss-3,Math.max(2,ss*.16));
    c.fillStyle='rgba(0,0,0,.30)';c.fillRect(xp+1.5,yp+ss-Math.max(2,ss*.18)-1.5,ss-3,Math.max(2,ss*.18));
    if(cell.mine){c.fillStyle='rgba(6,9,17,.45)';rr(c,xp,yp,ss,ss,3);c.fill();
      bombIcon(c,xp+ss/2,yp+ss*.56,ss*.26,t,x*13+y*7);}
  }else if(cell.rev){
    c.fillStyle='#1a2440';rr(c,px,py,s,s,3);c.fill();
    c.fillStyle='rgba(0,0,0,.35)';c.fillRect(px+1.5,py+1.5,s-3,Math.max(2,s*.14));
    if(cell.n>0){const dt=t-cell.revT,sc=dt<240?.6+.4*(dt/240):1;
      c.save();c.translate(px+s/2,py+s/2);c.scale(sc,sc);
      c.font='700 '+Math.round(s*.58)+'px Consolas,Menlo,monospace';
      c.textAlign='center';c.textBaseline='middle';c.fillStyle=NUMCOL[cell.n];c.fillText(cell.n,0,s*.04);c.restore();}
  }else{
    c.fillStyle='#465a85';rr(c,px,py,s,s,3);c.fill();
    c.strokeStyle='rgba(255,255,255,.16)';c.lineWidth=1.5;
    c.beginPath();c.moveTo(px+2,py+s-3);c.lineTo(px+2,py+2);c.lineTo(px+s-3,py+2);c.stroke();
    c.strokeStyle='rgba(0,0,0,.35)';
    c.beginPath();c.moveTo(px+s-2,py+3);c.lineTo(px+s-2,py+s-2);c.lineTo(px+3,py+s-2);c.stroke();
    if(G.revealAll&&cell.mine){c.fillStyle='rgba(170,30,45,.45)';rr(c,px,py,s,s,3);c.fill();
      bombIcon(c,px+s/2,py+s*.56,s*.26,t,x*13+y*7);
      c.strokeStyle='#ff5c5c';c.lineWidth=2.5;rr(c,px+1,py+1,s-2,s-2,3);c.stroke();}}}

function draw(t){
  const c=ctx;c.setTransform(dpr,0,0,dpr,0,0);c.clearRect(0,0,cw,chh);
  if(!G.grid)return;
  const w=G.w,h=G.h,tile=G.tile,ox=G.ox,oy=G.oy;
  c.save();
  if(G.shake>.4){c.translate((Math.random()-.5)*G.shake,(Math.random()-.5)*G.shake);G.shake*=.86;}else G.shake=0;
  /* plate */
  c.beginPath();rr(c,ox-13,oy-13,w*tile+26,h*tile+26,10);
  c.fillStyle='#111a31';c.shadowColor='rgba(0,0,0,.55)';c.shadowBlur=26;c.shadowOffsetY=9;c.fill();
  c.shadowColor='transparent';c.shadowOffsetY=0;c.lineWidth=2;c.strokeStyle='#2f3f68';c.stroke();
  for(let y=0;y<h;y++)for(let x=0;x<w;x++)drawCell(c,x,y,t);
  /* ghost */
  if(G.state==='playing'&&G.tray.length){
    const [ax,ay]=effAnchor(),type=G.tray[0],m=ROT[type][G.rot],cells=[];
    for(let j=0;j<m.length;j++)for(let i=0;i<m[j].length;i++)if(m[j][i])cells.push([ax+i,ay+j]);
    let ok=true;
    for(const[cx,cy] of cells)if(cx<0||cy<0||cx>=w||cy>=h||G.grid[cy*w+cx].cov){ok=false;break;}
    const g=Math.max(1,tile*.055),s=tile-2*g;
    for(const[cx,cy] of cells){if(cx<0||cy<0||cx>=w||cy>=h)continue;
      const px=ox+cx*tile+g,py=oy+cy*tile+g;
      if(ok){c.globalAlpha=.42;c.fillStyle=PIECES[type].c;rr(c,px,py,s,s,3);c.fill();
        c.globalAlpha=1;c.strokeStyle='rgba(255,255,255,.5)';c.lineWidth=1.5;rr(c,px,py,s,s,3);c.stroke();}
      else{c.globalAlpha=.48;c.fillStyle='#ff4646';rr(c,px,py,s,s,3);c.fill();c.globalAlpha=1;
        c.strokeStyle='rgba(0,0,0,.7)';c.lineWidth=2;
        c.beginPath();c.moveTo(px+s*.3,py+s*.3);c.lineTo(px+s*.7,py+s*.7);
        c.moveTo(px+s*.7,py+s*.3);c.lineTo(px+s*.3,py+s*.7);c.stroke();}}
    if(!G.pointer.active){const cxp=ox+G.cursor.x*tile,cyp=oy+G.cursor.y*tile,L=tile*.28;
      c.strokeStyle='rgba(63,216,255,.9)';c.lineWidth=2.5;
      c.beginPath();
      c.moveTo(cxp+2,cyp+2+L);c.lineTo(cxp+2,cyp+2);c.lineTo(cxp+2+L,cyp+2);
      c.moveTo(cxp+tile-2-L,cyp+2);c.lineTo(cxp+tile-2,cyp+2);c.lineTo(cxp+tile-2,cyp+2+L);
      c.moveTo(cxp+tile-2,cyp+tile-2-L);c.lineTo(cxp+tile-2,cyp+tile-2);c.lineTo(cxp+tile-2-L,cyp+tile-2);
      c.moveTo(cxp+2+L,cyp+tile-2);c.lineTo(cxp+2,cyp+tile-2);c.lineTo(cxp+2,cyp+tile-2-L);c.stroke();}}
  /* hint */
  if(G.hint&&t<G.hint.until){
    const ht=G.hint,cells=[],m=ROT[ht.type][ht.rot];
    for(let j=0;j<m.length;j++)for(let i=0;i<m[j].length;i++)if(m[j][i])cells.push([ht.x+i,ht.y+j]);
    let alive=true;for(const[cx,cy] of cells)if(cx<0||cy<0||cx>=w||cy>=h||G.grid[cy*w+cx].cov)alive=false;
    if(alive){const a=.28+.2*Math.sin(t/150),g=Math.max(1,tile*.055),s=tile-2*g;
      c.globalAlpha=a;c.fillStyle='#ffd23f';
      for(const[cx,cy] of cells)if(cx>=0&&cy>=0&&cx<w&&cy<h){rr(c,ox+cx*tile+g,oy+cy*tile+g,s,s,3);c.fill();}
      c.globalAlpha=1;c.strokeStyle='#ffd23f';c.lineWidth=3;c.shadowColor='#ffd23f';c.shadowBlur=9;
      rr(c,ox+ht.x*tile+1,oy+ht.y*tile+1,tile-2,tile-2,4);c.stroke();c.shadowBlur=0;}
    else G.hint=null;}
  /* particles */
  for(const p of G.particles){const al=Math.max(0,p.life/p.max);
    if(p.type==='ring'){c.globalAlpha=al*.85;c.strokeStyle=p.col;c.lineWidth=3;
      c.beginPath();c.arc(p.x,p.y,p.r,0,7);c.stroke();}
    else if(p.type==='spark'){c.globalAlpha=al;c.strokeStyle=p.col;c.lineWidth=2.2;
      c.beginPath();c.moveTo(p.x,p.y);c.lineTo(p.x-p.vx*.025,p.y-p.vy*.025);c.stroke();}
    else{c.save();c.globalAlpha=al;c.translate(p.x,p.y);if(p.rot!==undefined)c.rotate(p.rot);
      c.fillStyle=p.col;c.fillRect(-p.size/2,-p.size/2,p.size,p.size);c.restore();}}
  c.globalAlpha=1;
  /* floats */
  for(const f of G.floats){const al=Math.max(0,f.life/f.max);
    c.font='700 '+Math.round(13+tile*.09)+'px Consolas,Menlo,monospace';
    c.textAlign='center';c.globalAlpha=al;
    c.lineWidth=4;c.strokeStyle='rgba(6,9,17,.85)';c.strokeText(f.txt,f.x,f.y);
    c.fillStyle=f.col;c.fillText(f.txt,f.x,f.y);}
  c.globalAlpha=1;c.restore();}
