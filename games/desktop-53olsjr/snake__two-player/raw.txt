  const f=o.f||440,t0=AC.currentTime+(o.delay||0),t=o.t||.1,v=o.v||.25,f2=o.f2;
  const osc=AC.createOscillator(),g=AC.createGain();
  osc.type=o.type||'square';osc.frequency.setValueAtTime(f,t0);
  if(f2)osc.frequency.exponentialRampToValueAtTime(Math.max(1,f2),t0+t);
  g.gain.setValueAtTime(.0001,t0);g.gain.exponentialRampToValueAtTime(v,t0+.012);g.gain.exponentialRampToValueAtTime(.0001,t0+t);
  osc.connect(g).connect(master);osc.start(t0);osc.stop(t0+t+.03);
}
function noiseBurst(dur,vol,freq){
  if(!AC||muted)return;
  const n=(AC.sampleRate*dur)|0,buf=AC.createBuffer(1,n,AC.sampleRate),d=buf.getChannelData(0);
  for(let i=0;i<n;i++)d[i]=(Math.random()*2-1)*(1-i/n);
  const src=AC.createBufferSource();src.buffer=buf;
  const f=AC.createBiquadFilter();f.type='lowpass';f.frequency.value=freq;
  const g=AC.createGain();g.gain.value=vol;
  src.connect(f).connect(g).connect(master);src.start();
}

function makeSnake(id){
  const def=PDEF[id],mid=Math.floor(ROWS/2);
  let hx,hy,tx;
  if(id===0){hx=4;hy=mid-3;}else{hx=COLS-5;hy=mid+3;}
  const cells=[];for(let i=0;i<CFG.startLen;i++)cells.push({x:hx+(id===0?-i:i),y:hy});
  return {id,name:def.name,cMain:def.cMain,cDark:def.cDark,glow:def.glow,hex:def.hex,
    cells,prevCells:cells.map(c=>({x:c.x,y:c.y})),dir:{x:id===0?1:-1,y:0},queue:[],grow:0,score:0,alive:true,deathAt:0,cause:null,maxLen:CFG.startLen};
}
const key=(c)=>c.x+','+c.y;
const sameCell=(a,b)=>a&&b&&a.x===b.x&&a.y===b.y;
function occupied(x,y){
  for(const s of [p1,p2])for(const c of s.cells)if(c.x===x&&c.y===y)return true;
  for(const f of foods)if(f.x===x&&f.y===y)return true;
  if(bonus&&bonus.x===x&&bonus.y===y)return true;return false;
}
function spawnFood(){
  for(let t=0;t<600;t++){const x=(Math.random()*COLS)|0,y=(Math.random()*ROWS)|0;if(!occupied(x,y)){foods.push({x,y});return;}}
}
function spawnBonus(){
  for(let t=0;t<300;t++){const x=(Math.random()*COLS)|0,y=(Math.random()*ROWS)|0;if(!occupied(x,y)){bonus={x,y,born:gameT};return;}}
}
function resetRound(){
  p1=makeSnake(0);p2=makeSnake(1);foods=[];spawnFood();spawnFood();
  bonus=null;nextBonusAt=CFG.bonusEvery;totalFruits=0;gameT=0;elapsedMs=0;acc=0;menuAcc=0;
  particles.length=0;floaters.length=0;shakeMag=0;
}

/* ============================== AI (attract mode) ============================== */
function aiPick(s){
  const o=(s===p1)?p2:p1,h=s.cells[0];
  const occ=new Set();const body=s.grow>0?s.cells:s.cells.slice(0,-1);
  for(const c of body)occ.add(key(c));
  if(o.alive)for(const c of o.cells)occ.add(key(c));
  let target=null,bd=1e9;
  for(const f of foods){const d=Math.abs(f.x-h.x)+Math.abs(f.y-h.y);if(d<bd){bd=d;target=f;}}
  if(bonus&&Math.random()<.75)target=bonus;
  if(!target)target=h;
  const cands=[s.dir,{x:-s.dir.y,y:s.dir.x},{x:s.dir.y,y:-s.dir.x}];
  let best=s.dir,bs=1e9;
  for(const d of cands){
    const nx=h.x+d.x,ny=h.y+d.y;let sc;
    if(nx<0||ny<0||nx>=COLS||ny>=ROWS)sc=1e6;
    else if(occ.has(nx+','+ny))sc=1e5+(d===s.dir?1e4:0);
    else sc=Math.abs(nx-target.x)+Math.abs(ny-target.y)+(d===s.dir?-2:1);
    if(sc<bs){bs=sc;best=d;}
  }
  return best;
}

/* ============================== GAME STEP ============================== */
const interval=()=>Math.max(CFG.minInterval,CFG.baseInterval-totalFruits*CFG.ramp);

function step(){
  const A=p1,B=p2;
  if(state==='menu'){if(A.alive)A.queue=[aiPick(A)];if(B.alive)B.queue=[aiPick(B)];}
  for(const s of [A,B]){
    if(!s.alive)continue;const d=s.queue.shift();
    if(d&&!(d.x===-s.dir.x&&d.y===-s.dir.y))s.dir=d;
  }
  for(const s of [A,B])if(s.alive)s._nh={x:s.cells[0].x+s.dir.x,y:s.cells[0].y+s.dir.y};

  const dc={a:null,b:null};
  for(const s of [A,B]){
    if(!s.alive)continue;const n=s._nh;
    if(n.x<0||n.y<0||n.x>=COLS||n.y>=ROWS)dc[s.id==='a'?'a':'b']='wall';
  }
  if(A.alive&&B.alive&&!dc.a&&!dc.b){
    if(sameCell(A._nh,B._nh)){dc.a=dc.b='head';}
    else if(sameCell(A._nh,B.cells[0])&&sameCell(B._nh,A.cells[0])){dc.a=dc.b='swap';}
  }
  for(const s of [A,B]){
    if(!s.alive||dc[s.id])continue;
    const o=(s===p1)?p2:p1,n=s._nh,ok=o.id;
    let ob = dc[ok]?o.cells:(o.grow>0?o.cells:o.cells.slice(0,-1));
    if(ob.some(c=>sameCell(c,n))){dc[s.id]='body';continue;}
    const sb=s.grow>0?s.cells:s.cells.slice(0,-1);
    if(sb.some(c=>sameCell(c,n)))dc[s.id]='self';
  }
  for(const s of [A,B])if(s.alive&&dc[s.id])killSnake(s,dc[s.id]);

  for(const s of [A,B]){
    if(!s.alive)continue;const n=s._nh;
    s.prevCells=s.cells.map(c=>({x:c.x,y:c.y}));
    let ate=null;
    const fi=foods.findIndex(f=>f.x===n.x&&f.y===n.y);
    if(fi>=0){ate='food';foods.splice(fi,1);}
    else if(bonus&&bonus.x===n.x&&bonus.y===n.y){ate='bonus';bonus=null;}
    s.cells.unshift(n);
    if(ate==='food'){s.score++;totalFruits++;s.grow+=1;spawnFood();onEat(s,n,1);}
    else if(ate==='bonus'){s.score+=3;totalFruits++;s.grow+=2;nextBonusAt=gameT+CFG.bonusEvery;onEat(s,n,3);}
    if(s.grow>0)s.grow--;else s.cells.pop();
    if(s.cells.length>s.maxLen)s.maxLen=s.cells.length;
  }

  if(state==='playing'){
    if(gameT>=nextBonusAt&&!bonus)spawnBonus();
    if(bonus&&gameT-bonus.born>CFG.bonusLife){burst(boardX+(bonus.x+.5)*cell,boardY+(bonus.y+.5)*cell,8,['#ffe36e','#ffffff'],60);bonus=null;nextBonusAt=gameT+CFG.bonusEvery;}
  }
  if(!A.alive||!B.alive){
    if(state==='menu')resetRound();
    else finishRound(dc);
  }
}

function killSnake(s,cause){
  s.alive=false;s.cause=cause;s.deathAt=performance.now();
  for(const c of s.cells)burst(boardX+(c.x+.5)*cell,boardY+(c.y+.5)*cell,2,[s.hex,'#ffffff','#ff6b4a'],170);
  shakeMag=Math.max(shakeMag,13);flashAt=performance.now();
  sDeath(state==='menu');updateScores();
}
function onEat(s,c,pts){
  const px=boardX+(c.x+.5)*cell,py=boardY+(c.y+.5)*cell;
  burst(px,py,pts===3?26:14,[s.hex,'#ffd257','#ffffff'],pts===3?200:130);
  floaters.push({x:px,y:py-6,txt:'+'+pts,color:pts===3?'#ffe36e':'#ffffff',t:900,max:900});
  if(pts===3)sBonus();else sEat(s.id);
  updateScores();
}

}

function queueDir(s,x,y){
  if(!s.alive)return;
  const eff=s.queue.length?s.queue[s.queue.length-1]:s.dir;
  if(eff.x===x&&eff.y===y)return;if(eff.x===-x&&eff.y===-y)return;
  if(s.queue.length<3)s.queue.push({x,y});
}

function segPos(s,i,t){
  const to=s.cells[i];
  if(t>=1||!s.prevCells.length)return to;
  const fr=i===0?s.prevCells[0]:(i-1<s.prevCells.length?s.prevCells[i-1]:s.prevCells[s.prevCells.length-1]);
  return {x:fr.x+(to.x-fr.x)*t,y:fr.y+(to.y-fr.y)*t};
}

  ctx.beginPath();ctx.arc(px,py,cell*.62,-Math.PI/2,-Math.PI/2+rem*Math.PI*2);ctx.stroke();
}

function render(now){
  const pdt=Math.min(.04,(now-lastRender)/1000||.016);lastRender=now;
  ctx.clearRect(0,0,W,H);
  // ambient spores
  for(const a of ambients){
    a.x+=a.vx*pdt;a.y+=a.vy*pdt;
    if(a.x<-5)a.x=W+5;if(a.x>W+5)a.x=-5;if(a.y<-5)a.y=H+5;if(a.y>H+5)a.y=-5;
    ctx.fillStyle=a.c+(a.a*(.7+.3*Math.sin(now/900+a.x)))+')';circle(a.x,a.y,a.r);
  }
  const t=(state==='playing'||state==='paused')?Math.min(1,acc/interval()):state==='menu'?Math.min(1,menuAcc/210):1;
  ctx.save();
  if(shakeMag>0){ctx.translate((Math.random()-.5)*shakeMag,(Math.random()-.5)*shakeMag);shakeMag*=.86;if(shakeMag<.3)shakeMag=0;}
  const F=10;
  if(boardCanvas)ctx.drawImage(boardCanvas,boardX-F,boardY-F,boardW+2*F,boardH+2*F);
  ctx.strokeStyle='rgba(255,105,80,.3)';ctx.lineWidth=2;
  ctx.setLineDash([12,9]);ctx.lineDashOffset=-now/40;
  ctx.strokeRect(boardX-3,boardY-3,boardW+6,boardH+6);ctx.setLineDash([]);
  if(now-flashAt<180){ctx.fillStyle='rgba(255,255,255,'+(0.24*(1-(now-flashAt)/180)).toFixed(3)+')';ctx.fillRect(boardX-F,boardY-F,boardW+2*F,boardH+2*F);}
  for(const f of foods)drawFood(f,now);
  if(bonus)drawBonus(now);
  drawSnake(p2,now,t);drawSnake(p1,now,t);
  // particles
  ctx.save();ctx.globalCompositeOperation='lighter';
  for(let i=particles.length-1;i>=0;i--){const p=particles[i];
    p.life-=pdt*1000;if(p.life<=0){particles.splice(i,1);continue;}
    p.x+=p.vx*pdt;p.y+=p.vy*pdt;p.vy+=p.g*pdt;
    ctx.globalAlpha=Math.max(0,p.life/p.max);ctx.fillStyle=p.color;circle(p.x,p.y,p.size*(.4+.6*p.life/p.max));
  }
  ctx.restore();
  // floaters
  for(let i=floaters.length-1;i>=0;i--){const f=floaters[i];
    f.t-=pdt*1000;if(f.t<=0){floaters.splice(i,1);continue;}
    f.y-=34*pdt;
    ctx.globalAlpha=Math.max(0,f.t/f.max);
    ctx.font='700 '+Math.max(12,cell*.55)+'px '+getComputedStyle(document.body).getPropertyValue('--mono');
    ctx.textAlign='center';ctx.lineWidth=4;ctx.strokeStyle='rgba(6,14,19,.8)';
    ctx.strokeText(f.txt,f.x,f.y);ctx.fillStyle=f.color;ctx.fillText(f.txt,f.x,f.y);
    ctx.globalAlpha=1;
  }
  ctx.restore();
  // HUD live values
  if(state==='playing'||state==='over'){
    const sp='×'+(CFG.baseInterval/interval()).toFixed(1);
    if($('speedTag').textContent!==sp)$('speedTag').textContent=sp;
    const tm=fmt(state==='over'?finalElapsed:elapsedMs);
    if($('timeTag').textContent!==tm)$('timeTag').textContent=tm;
  }
}

/* ============================== MAIN LOOP ============================== */
function frame(ts){
  requestAnimationFrame(frame);
  const dt=Math.min(50,ts-lastTs||16);lastTs=ts;
  if(state==='playing'){
    gameT+=dt;elapsedMs+=dt;acc+=dt;let n=0;
    while(acc>=interval()&&n<4){step();acc-=interval();n++;}
  }else if(state==='menu'){
    menuAcc+=dt;const mi=210;
    while(menuAcc>=mi){if(state==='menu')step();menuAcc-=mi;}
  }
  render(ts);
}
