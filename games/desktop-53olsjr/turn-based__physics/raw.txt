Collision routine (circle vs box):
```
function collide(b, rect, res){ // res: restitution, friction; returns {nx,ny,vn} or null
  const cx = clamp(b.x, rect.x, rect.x+rect.w);
  const cy = clamp(b.y, rect.y, rect.y+rect.h);
  let dx=b.x-cx, dy=b.y-cy; let d2=dx*dx+dy*dy;
  if(d2 > b.r*b.r) return null;
  let d=Math.sqrt(d2);
  let nx,ny;
  if(d<1e-4){ // center inside box: push out along min axis
    const l=b.x-rect.x, r=rect.x+rect.w-b.x, t=b.y-rect.y, bo=rect.y+rect.h-b.y;
    const m=Math.min(l,r,t,bo);
    if(m===l){nx=-1;ny=0;} else if(m===r){nx=1;ny=0;} else if(m===t){nx=0;ny=-1;} else {nx=0;ny=1;}
    dx=nx*0.01; dy=ny*0.01; d=0.01;
  } else { nx=dx/d; ny=dy/d; }
  const pen=b.r-d; b.x+=nx*pen; b.y+=ny*pen;
  const vn=b.vx*nx+b.vy*ny;
  if(vn<0){
    const rest=res, fric=0.985? use tangential damp:
    const tx=-ny, ty=nx;
    let vt=b.vx*tx+b.vy*ty;
    vt*=0.96; // friction
    b.vx = -vn*rest*nx + vt*tx;
    b.vy = -vn*rest*ny + vt*ty;
  }
  return {nx,ny,vn:Math.max(0,-vn)};
}
```

```js
function resolveCircle(b, rc, rest){
  const cx = b.x < rc.x ? rc.x : (b.x > rc.x+rc.w ? rc.x+rc.w : b.x);
  const cy = b.y < rc.y ? rc.y : (b.y > rc.y+rc.h ? rc.y+rc.h : b.y);
  let dx=b.x-cx, dy=b.y-cy; let d2=dx*dx+dy*dy;
  if(d2 > b.r*b.r) return 0;
  let nx,ny,d=Math.sqrt(d2);
  if(d<1e-5){
    const l=b.x-rc.x, r=(rc.x+rc.w)-b.x, t=b.y-rc.y, bo=(rc.y+rc.h)-b.y;
    const m=Math.min(l,r,t,bo);
    if(m===l){nx=-1;ny=0;} else if(m===r){nx=1;ny=0;} else if(m===t){nx=0;ny=-1;} else {nx=0;ny=1;}
  } else { nx=dx/d; ny=dy/d; }
  const pen=b.r-(d||0); b.x+=nx*pen; b.y+=ny*pen;
  const vn=b.vx*nx+b.vy*ny;
  if(vn<0){
    const tx=-ny, ty=nx;
    const vt=(b.vx*tx+b.vy*ty)*0.965;
    const nv=(-vn)*rest;
    b.vx=nx*nv+tx*vt; b.vy=ny*nv+ty*vt;
    return -vn;
  }
  return 0;
}
```

```html
  let dx=b.x-cx,dy=b.y-cy,d2=dx*dx+dy*dy;
  if(d2>b.r*b.r)return 0;
  let nx,ny,d=Math.sqrt(d2);
  if(d<1e-5){
    const l=b.x-rc.x,r=(rc.x+rc.w)-b.x,t=b.y-rc.y,bo=(rc.y+rc.h)-b.y,m=Math.min(l,r,t,bo);
    if(m===l){nx=-1;ny=0}else if(m===r){nx=1;ny=0}else if(m===t){nx=0;ny=-1}else{nx=0;ny=1}
  }else{nx=dx/d;ny=dy/d}
  const pen=b.r-d; b.x+=nx*pen; b.y+=ny*pen;
  const vn=b.vx*nx+b.vy*ny;
  if(vn<0){
    const tx=-ny,ty=nx,vt=(b.vx*tx+b.vy*ty)*.965,nv=-vn*rest;
    b.vx=nx*nv+tx*vt; b.vy=ny*nv+ty*vt; return -vn;
  }
  return 0;
}
function rectHit(x,y,r,rc){
  const cx=clamp(x,rc.x,rc.x+rc.w),cy=clamp(y,rc.y,rc.y+rc.h);
  const dx=x-cx,dy=y-cy; return dx*dx+dy*dy<r*r;
}
function blockedAt(x,y,r){
  const i0=Math.max(0,Math.floor((x-r)/COLW)-1),i1=Math.min(COLS-1,Math.floor((x+r)/COLW)+1);
  for(let i=i0;i<=i1;i++)if(rectHit(x,y,r,world.cols[i]))return true;
  for(const b of world.blocks)if(rectHit(x,y,r,b))return true;
  for(const c of world.cores)if(c.hp>0&&rectHit(x,y,r,c))return true;
  return false;
}

function stepBall(dt){
  const b=ball,o=b.o; b.t+=dt;
  b.vy+=GRAV*dt; b.vx+=world.wind*dt;
  b.trail.push({x:b.x,y:b.y}); if(b.trail.length>15)b.trail.shift();
  let speed=Math.hypot(b.vx,b.vy);
  const sub=clamp(Math.ceil(speed*dt/(b.r*.5)),1,10), sdt=dt/sub;
  for(let s=0;s<sub;s++){
    b.x+=b.vx*sdt; b.y+=b.vy*sdt;
    // terrain
    const i0=Math.max(0,Math.floor((b.x-b.r)/COLW)-1),i1=Math.min(COLS-1,Math.floor((b.x+b.r)/COLW)+1);
    for(let i=i0;i<=i1;i++){
      const vn=resolveCircle(b,world.cols[i],o.rest*.85);
      if(vn>60){ SFX.hit(vn*.6); puff(b.x,b.y+ b.r*.4,vn); if(vn>520)shake+=Math.min(7,vn/140); ringFx(b.x,b.y+ b.r*.4,vn); }
    }
    // blocks
    for(const bl of world.blocks.slice()){
      const vn=resolveCircle(b,bl,o.rest);
      if(vn>60){ ringFx(b.x,b.y,vn); SFX.hit(vn); shake+=Math.min(5,vn/180); applyDamage(bl,vn,o,false); }
    }
    // cores
    for(let s=0;s<2;s++){ const c=world.cores[s]; if(c.hp<=0)continue;
      const vn=resolveCircle(b,c,o.rest);
      if(vn>45){ ringFx(b.x,b.y,vn); SFX.hit(vn); SFX.coreHit(); shake+=Math.min(9,vn/120); applyDamage(c,vn,o,true); }
    }
    if(state!=='fly')return;
  }
  speed=Math.hypot(b.vx,b.vy);
  b.idle = speed<38 ? b.idle+dt : 0;
  if(b.idle>.7 || b.t>12 || b.x<-70 || b.x>VW+70 || b.y>VH+120) endTurn();
}

function applyDamage(bl,vn,o,isCore){
  const d=vn*.09*o.dmg, shooter=turnSide;
  bl.hp-=d;
  if(bl.side!==shooter)stats.dmg[shooter]+=d;
  if(d>=6)texts.push({x:bl.x+bl.w/2,y:bl.y-8,txt:String(Math.round(d)),col:PCOL[shooter],t:0});
  if(bl.hp<=0){ isCore?destroyCore(bl.side):destroyBlock(bl); }
  else { SFX.crack(); hud(); }
}
function destroyBlock(bl){
  const i=world.blocks.indexOf(bl); if(i>=0)world.blocks.splice(i,1);
  shatter(bl.x,bl.y,bl.w,bl.h,bl.mat,turnSide);
  if(bl.side!==turnSide){ stats.brk[turnSide]++; texts.push({x:bl.x+bl.w/2,y:bl.y-14,txt:'+'+bl.score,col:PCOL[turnSide],t:0,big:true}); SFX.score(); }
  else texts.push({x:bl.x+bl.w/2,y:bl.y-14,txt:'OWN GOAL',col:'#9db2d6',t:0});
  SFX.brk(bl.mat); shake+=5; hud();
}

/* ============================== CPU ============================== */
function simShot(x,y,vx,vy,o){
  const dyn=[];
  for(const b of world.blocks)dyn.push({x:b.x,y:b.y,w:b.w,h:b.h,side:b.side,hp:Math.max(1,b.hp),core:false});
  for(const c of world.cores)if(c.hp>0)dyn.push({x:c.x,y:c.y,w:c.w,h:c.h,side:c.side,hp:c.hp,core:true});
  const b={x,y,vx,vy,r:o.r}; let score=0,contact=false; const dt=1/90;
  for(let s=0;s<430;s++){
    b.vy+=GRAV*dt; b.vx+=world.wind*dt;
    const sp=Math.hypot(b.vx,b.vy), sub=sp>950?2:1, sd=dt/sub;
    for(let k=0;k<sub;k++){
      b.x+=b.vx*sd; b.y+=b.vy*sd;
      const i0=Math.max(0,Math.floor((b.x-b.r)/COLW)-1),i1=Math.min(COLS-1,Math.floor((b.x+b.r)/COLW)+1);
      for(let i=i0;i<=i1;i++)resolveCircle(b,world.cols[i],.5);
      for(const d of dyn){ if(d.hp<=0)continue;
        const vn=resolveCircle(b,d,.5);
        if(vn>60){ contact=true;
          if(d.side===0){ let dm=Math.min(vn*.09*o.dmg,d.hp); score+=dm*(d.core?2:1); d.hp-=dm; } }
      }
    }
    if(b.y>VH+80||b.x<-90||b.x>VW+90)break;
    if(sp<30&&s>24)break;
  }
  return score+(contact?0:-70);
}
function cpuPlan(){
  const sp=SPAWNS[1]; let best={score:-1e9,oi:1,a:62,pf:.8};
  for(let oi=0;oi<3;oi++){ const o=ORBS[oi];
    for(let a=42;a<=84;a+=3)for(const pf of [.5,.68,.86,1]){
      const v=o.max*pf, sc=simShot(sp.x,sp.y,-Math.cos(a*Math.PI/180)*v,-Math.sin(a*Math.PI/180)*v,o);
      if(sc>best.score)best={score:sc,oi,a,pf};
    }}
  const o=ORBS[best.oi], ang=(best.a+rnd(-2,2))*Math.PI/180, v=o.max*best.pf*rnd(.96,1.03);
  return {oi:best.oi,vx:-Math.cos(ang)*v,vy:-Math.sin(ang)*v};
}
function scheduleCpu(){
  cpuPending=true; const id=matchId;
  setTimeout(()=>{ if(id!==matchId||state!=='aim'||paused)return; banner('CPU IS LOADING…','cpu'); },650);
  setTimeout(()=>{
    if(id!==matchId||state!=='aim')return;
    if(paused){ return; } // will reschedule on resume
    const p=cpuPlan(); fire(p.oi,p.vx,p.vy);
  },1550+Math.random()*500);
}

function ringFx(x,y,vn){ parts.push({type:'ring',x,y,t:0,life:.32,r0:6,r1:clamp(8+vn*.045,12,60)}); }
function shatter(x,y,w,h,mat,bySide){
  const pal= mat==='wood'?['#c98a4b','#a5683a','#8a5430'] : mat==='stone'?['#a7b3c2','#8895a5','#6e7c8d'] : [PCOL[bySide],'#ffffff',PCOL[bySide]];
  const n=mat==='core'?26:13;
  for(let i=0;i<n;i++)parts.push({type:'chunk',x:x+rnd(0,w),y:y+rnd(0,h),vx:rnd(-240,240),vy:rnd(-380,-40),rot:rnd(0,6.3),vr:rnd(-9,9),size:rnd(4,mat==='core'?11:8),col:pal[i%3],t:0,life:rnd(.8,1.4)});
}
function spawnConfetti(w){
  const pal=[PCOL[w],'#ffd9a0','#ffffff','#ff5d7a',PCOL[1-w]];
  for(let i=0;i<130;i++)parts.push({type:'conf',x:Math.random()*VW,y:-Math.random()*280,vx:rnd(-40,40),vy:rnd(40,130),rot:rnd(0,6.3),vr:rnd(-6,6),size:rnd(5,11),col:pal[i%pal.length],t:0,life:rnd(3.2,5),ph:rnd(0,6)});
}
function updateParts(dt){
  for(let i=parts.length-1;i>=0;i--){ const p=parts[i]; p.t+=dt;
    if(p.t>p.life){parts.splice(i,1);continue;}
    if(p.type==='chunk'){p.vy+=GRAV*.8*dt;p.x+=p.vx*dt;p.y+=p.vy*dt;p.rot+=p.vr*dt;}
    else if(p.type==='dust'){p.x+=p.vx*dt;p.y+=p.vy*dt;p.r+=14*dt;}
    else if(p.type==='conf'){p.vy=Math.min(170,p.vy+160*dt);p.x+=(p.vx+Math.sin(p.t*3.2+p.ph)*34)*dt;p.y+=p.vy*dt;p.rot+=p.vr*dt;}
  }
  for(let i=texts.length-1;i>=0;i--){ const t=texts[i]; t.t+=dt; t.y-=42*dt; if(t.t>.95)texts.splice(i,1); }
}

}
function drawCracks(o,cr,stage,col){
  ctx.strokeStyle='rgba(20,12,8,.55)';ctx.lineWidth=1.6;ctx.lineJoin='round';
  const n=stage===1?2:cr.length;
  for(let i=0;i<n;i++){ const p=cr[i]; ctx.beginPath();
    for(let j=0;j<p.length;j++){const px=o.x+p[j][0]*o.w,py=o.y+p[j][1]*o.h; j?ctx.lineTo(px,py):ctx.moveTo(px,py);} ctx.stroke(); }
}
function drawBlock(bl){
  const f=bl.hp/bl.max;
  ctx.fillStyle=bl.mat==='wood'?'#c98a4b':'#a7b3c2';
  ctx.fillRect(bl.x,bl.y,bl.w,bl.h);
  ctx.strokeStyle=bl.mat==='wood'?'#7c4e26':'#5d6b7c';ctx.lineWidth=3;
  ctx.strokeRect(bl.x+1.5,bl.y+1.5,bl.w-3,bl.h-3);
  ctx.fillStyle='rgba(255,255,255,.18)';ctx.fillRect(bl.x+4,bl.y+4,bl.w-8,3);
  if(f<.7)drawCracks(bl,bl.cr,1);
  if(f<.35)drawCracks(bl,bl.cr,2);
}
function drawCore(c,t){
  const cx=c.x+c.w/2,cy=c.y+c.h/2,side=c.side,col=PCOL[side];
  // pedestal
  ctx.fillStyle='#5d6b7c';ctx.fillRect(cx-30,c.y+c.h-10,60,12);
  ctx.strokeStyle='#3e4a58';ctx.lineWidth=2;ctx.strokeRect(cx-30,c.y+c.h-10,60,12);
  // glow
  const pulse=.5+.3*Math.sin(t*2.6+side*2);
  const gg=ctx.createRadialGradient(cx,cy,4,cx,cy,64);
  gg.addColorStop(0,col.replace(')',',0)').length?col:col); // noop guard
  ctx.globalAlpha=.3*pulse;
  const gg2=ctx.createRadialGradient(cx,cy,4,cx,cy,64);
  gg2.addColorStop(0,'rgba(255,255,255,.9)');gg2.addColorStop(.35,col);gg2.addColorStop(1,'rgba(0,0,0,0)');
  ctx.fillStyle=gg2;ctx.beginPath();ctx.arc(cx,cy,64,0,7);ctx.fill();
  ctx.globalAlpha=1;
  // gem
  const hw=c.w/2,hh=c.h/2;
  const pts=[[cx,cy-hh],[cx+hw*.85,cy-hh*.42],[cx+hw*.85,cy+hh*.42],[cx,cy+hh],[cx-hw*.85,cy+hh*.42],[cx-hw*.85,cy-hh*.42]];
  ctx.beginPath();pts.forEach((p,i)=>i?ctx.lineTo(p[0],p[1]):ctx.moveTo(p[0],p[1]));ctx.closePath();
  const fg=ctx.createLinearGradient(cx,cy-hh,cx,cy+hh);
  if(side===0){fg.addColorStop(0,'#ffd9b0');fg.addColorStop(.5,'#ff8a4d');fg.addColorStop(1,'#c2410c');}
  else{fg.addColorStop(0,'#d8fff7');fg.addColorStop(.5,'#4fd8c4');fg.addColorStop(1,'#0f766e');}
  ctx.fillStyle=fg;ctx.fill();
  ctx.strokeStyle='rgba(20,12,8,.5)';ctx.lineWidth=3;ctx.lineJoin='round';ctx.stroke();
  ctx.strokeStyle='rgba(255,255,255,.3)';ctx.lineWidth=1.5;
  for(let i=0;i<pts.length;i+=2){ctx.beginPath();ctx.moveTo(cx,cy);ctx.lineTo(pts[i][0],pts[i][1]);ctx.stroke();}
  const f=c.hp/c.max;
  if(f<.7)drawCracks(c,c.cr,1);
  if(f<.35)drawCracks(c,c.cr,2);
}
function ballPos(){
  const sp=SPAWNS[turnSide];
  if(!dragging)return {x:sp.x,y:sp.y};
  let dx=dragPt.x-sp.x,dy=dragPt.y-sp.y;const L=Math.hypot(dx,dy);
  if(L>MAXPULL){dx*=MAXPULL/L;dy*=MAXPULL/L;}
  return {x:sp.x+dx,y:sp.y+dy};
}
function drawLauncher(t){
  const s=turnSide, sp=SPAWNS[s], gy=470;
  ctx.fillStyle='#3a2418';ctx.beginPath();ctx.ellipse(sp.x,gy+2,36,10,0,0,7);ctx.fill();
  ctx.strokeStyle='#5a3d27';ctx.lineWidth=8;ctx.lineCap='round';
  ctx.beginPath();ctx.moveTo(sp.x,gy-4);ctx.lineTo(sp.x,gy-26);ctx.stroke();
  ctx.beginPath();ctx.moveTo(sp.x,gy-24);ctx.lineTo(sp.x-15,gy-58);ctx.moveTo(sp.x,gy-24);ctx.lineTo(sp.x+15,gy-58);ctx.stroke();
  if(state==='aim'&&s===turnSide&&!paused){
    const can=canAim();
    // aim pulse ring
    ctx.globalAlpha=.35+.2*Math.sin(t*4);
    ctx.strokeStyle=PCOL[s];ctx.lineWidth=2;
    ctx.beginPath();ctx.ellipse(sp.x,gy,44+3*Math.sin(t*4),11,0,0,7);ctx.stroke();
    ctx.globalAlpha=1;
    const bp=ballPos(), o=ORBS[orbIdx];
    if(can){
      // bands
      ctx.strokeStyle='#e8d5ae';ctx.lineWidth=4.5;
      ctx.beginPath();ctx.moveTo(sp.x-15,gy-58);ctx.lineTo(bp.x,bp.y);ctx.moveTo(sp.x+15,gy-58);ctx.lineTo(bp.x,bp.y);ctx.stroke();
      // preview + power (while dragging)
      if(dragging){
        const L=Math.min(MAXPULL,Math.hypot(bp.x-sp.x,bp.y-sp.y));
        let vx=-(bp.x-sp.x)*(o.max/MAXPULL),vy=-(bp.y-sp.y)*(o.max/MAXPULL);
        const sp2=Math.hypot(vx,vy); if(sp2>o.max){vx*=o.max/sp2;vy*=o.max/sp2;}
        let px=sp.x,py=sp.y,pvx=vx,pvy=vy;const dt=.03;
        for(let k=0;k<16;k++){
          pvy+=GRAV*dt;pvx+=world.wind*dt;px+=pvx*dt;py+=pvy*dt;
          const hit=blockedAt(px,py,o.r);
          ctx.globalAlpha=(hit?.9:.75)*(1-k/18);
          ctx.fillStyle=hit?'#ffd9a0':PCOL[s];
          ctx.beginPath();ctx.arc(px,py,3.2,0,7);ctx.fill();
          if(hit)break;
        }
        ctx.globalAlpha=1;
        const pct=Math.round(L/MAXPULL*100);
        ctx.font='bold 17px "Trebuchet MS",sans-serif';ctx.textAlign='center';
        ctx.lineWidth=4;ctx.strokeStyle='rgba(0,0,0,.6)';
        ctx.strokeText(pct+'%',bp.x,bp.y-26);ctx.fillStyle='#fff';ctx.fillText(pct+'%',bp.x,bp.y-26);
      }
    }
    // the orb
    const bg=ctx.createRadialGradient(sp.x-o.r*.35,sp.y-o.r*.4,o.r*.15,sp.x===bp.x?bp.x:bp.x,bp.y,o.r);
    ctx.beginPath();ctx.arc(bp.x,bp.y,o.r,0,7);
    const bg2=ctx.createRadialGradient(bp.x-o.r*.35,bp.y-o.r*.4,o.r*.15,bp.x,bp.y,o.r);
    bg2.addColorStop(0,o.c0);bg2.addColorStop(.55,o.c1);bg2.addColorStop(1,o.c2);
    ctx.fillStyle=bg2;ctx.fill();
    ctx.strokeStyle='rgba(0,0,0,.4)';ctx.lineWidth=2.5;ctx.stroke();
  }
}

function drawParts(){
  for(const p of parts){ const k=p.t/p.life;
    if(p.type==='chunk'||p.type==='conf'){
      ctx.globalAlpha=p.type==='chunk'?(k>.7?1-(k-.7)/.3:1):1;
      ctx.save();ctx.translate(p.x,p.y);ctx.rotate(p.rot);
      ctx.fillStyle=p.col;ctx.fillRect(-p.size/2,-p.size/2,p.size,p.size*.7);ctx.restore();
    }else if(p.type==='dust'){
      ctx.globalAlpha=(1-k)*.5;ctx.fillStyle='#c9a06b';
      ctx.beginPath();ctx.arc(p.x,p.y,p.r,0,7);ctx.fill();
    }else if(p.type==='ring'){
      ctx.globalAlpha=(1-k)*.8;ctx.strokeStyle='#ffd9a0';ctx.lineWidth=3*(1-k)+1;
      ctx.beginPath();ctx.arc(p.x,p.y,p.r0+(p.r1-p.r0)*k,0,7);ctx.stroke();
    }
  }
  ctx.globalAlpha=1;
}

}
function render(t){
  ctx.setTransform(dpr,0,0,dpr,0,0);
  drawSky(t);
  const sx=(Math.random()*2-1)*shake, sy=(Math.random()*2-1)*shake;
  ctx.setTransform(dpr*scale,0,0,dpr*scale,dpr*(ox+sx),dpr*(oy+sy));
  drawTerrain();
  for(const bl of world.blocks)drawBlock(bl);
  for(let s=0;s<2;s++)if(world.cores[s].hp>0)drawCore(world.cores[s],t);
  drawLauncher(t);
  drawBall(t);
  drawParts();
  drawTexts();
  // vignette (screen space)
  ctx.setTransform(dpr,0,0,dpr,0,0);
  const vg=ctx.createRadialGradient(W/2,H*.45,Math.min(W,H)*.38,W/2,H*.5,Math.max(W,H)*.78);
  vg.addColorStop(0,'rgba(4,8,18,0)');vg.addColorStop(1,'rgba(4,8,18,.42)');
  ctx.fillStyle=vg;ctx.fillRect(0,0,W,H);
  // timer
  const sec=Math.max(0,Math.ceil(timeLeft));
  if(sec!==lastSec){ lastSec=sec; const el=byId('timer');
    el.textContent=Math.floor(sec/60)+':'+String(sec%60).padStart(2,'0');
    el.classList.toggle('low',sec<=30&&state!=='over');
  }
}
```
