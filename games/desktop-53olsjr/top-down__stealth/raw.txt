function findPath(sx,sy,tx,ty){
  if(sx===tx&&sy===ty) return [];
  const N=COLS*ROWS;
  const g=new Float32Array(N).fill(Infinity);
  const came=new Int32Array(N).fill(-1);
  const closed=new Uint8Array(N);
  const open=[]; 
  const h=(i)=>Math.abs((i%COLS)-tx)+Math.abs(Math.floor(i/COLS)-ty);
  const s=sy*COLS+sx, goal=ty*COLS+tx;
  g[s]=0; open.push(s); let fOpen=new Float32Array(N).fill(Infinity); fOpen[s]=h(s);
  while(open.length){
    // find min f
    let bi=0,bf=fOpen[open[0]];
    for(let k=1;k<open.length;k++){ if(fOpen[open[k]]<bf){bf=fOpen[open[k]];bi=k;} }
    const cur=open.splice(bi,1)[0];
    if(cur===goal){ // reconstruct
      const path=[]; let c=cur;
      while(c!==s && c!==-1){ path.push({x:c%COLS,y:(c/COLS)|0}); c=came[c]; }
      path.reverse(); return path;
    }
    closed[cur]=1;
    const cx=cur%COLS, cy=(cur/COLS)|0;
    const nb=[[cx+1,cy],[cx-1,cy],[cx,cy+1],[cx,cy-1]];
    for(const [nx,ny] of nb){ if(solidAt(nx,ny)) continue; const ni=ny*COLS+nx; if(closed[ni]) continue; const ng=g[cur]+1; if(ng<g[ni]){ g[ni]=ng; came[ni]=cur; fOpen[ni]=ng+h(ni); if(fOpen[ni]===Infinity) open.push(ni); } }
  }
  return null;
}

/* ================= HELPERS ================= */
const dist=(a,b,c,d)=>Math.hypot(c-a,d-b);
function angleDiff(a,b){ let d=a-b; while(d>Math.PI)d-=6.28319; while(d<-Math.PI)d+=6.28319; return d; }
function losBlocked(x0,y0,x1,y1){ const d=dist(x0,y0,x1,y1), n=Math.ceil(d/6);
  for(let i=1;i<n;i++){ const t=i/n, px=x0+(x1-x0)*t, py=y0+(y1-y0)*t;
    if(solidAt(Math.floor(px/T),Math.floor(py/T))) return true; } return false; }
function circleFree(x,y,r){
  const x0=Math.floor((x-r)/T),x1=Math.floor((x+r)/T),y0=Math.floor((y-r)/T),y1=Math.floor((y+r)/T);
  for(let ty=y0;ty<=y1;ty++)for(let tx=x0;tx<=x1;tx++){ if(!solidAt(tx,ty))continue;
    const cx=Math.max(tx*T,Math.min(x,(tx+1)*T)), cy=Math.max(ty*T,Math.min(y,(ty+1)*T));
    if((cx-x)*(cx-x)+(cy-y)*(cy-y)<r*r) return false; }
  return true; }
function moveCircle(o,dx,dy,r){ let nx=o.x+dx; if(circleFree(nx,o.y,r)) o.x=nx;
  let ny=o.y+dy; if(circleFree(o.x,ny,r)) o.y=ny; }
function inBush(p){ const c=MAP[Math.floor(p.y/T)]&&MAP[Math.floor(p.y/T)][Math.floor(p.x/T)]; return c==='B'; }
function nearestFloor(tx,ty){ if(!solidAt(tx,ty)) return {x:tx,y:ty};
  const q=[[tx,ty]], seen=new Set([tx+','+ty]); let ring=0;
  while(q.length&&ring<5){ const [x,y]=q.shift(); ring++;
    for(const n of [[x+1,y],[x-1,y],[x,y+1],[x,y-1]]){ if(solidAt(n[0],n[1]))continue;
      const k=n[0]+','+n[1]; if(!seen.has(k)){ seen.add(k); return {x:n[0],y:n[1]}; } } }
  return {x:tx,y:ty}; }

/* ================= A* (4-way, tile grid) ================= */
function findPath(sx,sy,tx,ty){
  if(sx===tx&&sy===ty) return [];
  const N=COLS*ROWS, gs=new Float32Array(N).fill(Infinity), fs=new Float32Array(N).fill(Infinity),
        came=new Int32Array(N).fill(-1), closed=new Uint8Array(N), open=[];
  const h=i=>Math.abs((i%COLS)-tx)+Math.abs(Math.floor(i/COLS)-ty);
  const s=sy*COLS+sx, goal=ty*COLS+tx;
  gs[s]=0; fs[s]=h(s); open.push(s);
  while(open.length){
    let bi=0,bf=fs[open[0]];
    for(let k=1;k<open.length;k++){ if(fs[open[k]]<bf){ bf=fs[open[k]]; bi=k; } }
    const cur=open.splice(bi,1)[0];
    if(cur===goal){ const path=[]; let c=cur;
      while(c!==s&&c!==-1){ path.push({x:c%COLS,y:(c/COLS)|0}); c=came[c]; }
      return path.reverse(); }
    closed[cur]=1;
    const cx=cur%COLS, cy=(cur/COLS)|0;
    for(const n of [[cx+1,cy],[cx-1,cy],[cx,cy+1],[cx,cy-1]]){
      if(solidAt(n[0],n[1])) continue;
      const ni=n[1]*COLS+n[0]; if(closed[ni]) continue;
      const ng=gs[cur]+1;
      if(ng<gs[ni]){ gs[ni]=ng; came[ni]=cur; fs[ni]=ng+h(ni); if(gs[ni]===ng&&fs[ni]===ng+h(ni)&&came[ni]===cur) if(!open.includes(ni)) open.push(ni); }
    }
  }
  return null;
}
function pathTo(g,tx,ty){ const f=nearestFloor(tx,ty);
  const p=findPath(Math.floor(g.x/T),Math.floor(g.y/T),f.x,f.y);
  if(!p) return [];
  return p.map(n=>({x:n.x*T+16,y:n.y*T+16})); }
function moveAlong(g,dt,spd){
  const n=g.path[0]; if(!n) return false;
  const dx=n.x-g.x, dy=n.y-g.y, d=Math.hypot(dx,dy), st=spd*dt; let reached=false;
  if(d<=st){ g.x=n.x; g.y=n.y; g.path.shift(); if(!g.path.length) reached=true; }
  else { g.x+=dx/d*st; g.y+=dy/d*st; }
  const nn=g.path[0];
  g.face = nn? Math.atan2(nn.y-g.y,nn.x-g.x) : Math.atan2(dy,dx);
  return reached;
}

function trySubdue(){
  if(state!=='playing') return;
  let best=null,bd=52;
  for(const g of guards){ if(g.state==='down') continue;
    const d=dist(player.x,player.y,g.x,g.y);

  if(best){ best.state='down'; best.downT=0; subdues++; sfx.subdue();
    addNoise(best.x,best.y,100); burst(best.x,best.y,'220,235,255',10); shake(2.5); toast('GUARD NEUTRALIZED'); }
}

}

/* ================= UPDATE ================= */
function updPlayer(dt){
  let ix=((keys['d']||keys['arrowright'])?1:0)-((keys['a']||keys['arrowleft'])?1:0);
  let iy=((keys['s']||keys['arrowdown'])?1:0)-((keys['w']||keys['arrowup'])?1:0);
  if(ix||iy){ const l=Math.hypot(ix,iy); ix/=l; iy/=l; player.face=Math.atan2(iy,ix); }
  player.crouch=!!keys['shift'];
  const spd=player.crouch?85:155;
  moveCircle(player,ix*spd*dt,iy*spd*dt,8);
  player.moving=!!(ix||iy);
  if(player.moving){ pStepT-=dt*(player.crouch?0.72:1);
    if(pStepT<=0){ pStepT=player.crouch?0.42:0.3; sfx.step(player.crouch);
      fpSide^=1; foots.push({x:player.x-Math.cos(player.face)*4+Math.cos(player.face+1.57)*fpSide*3.4,
        y:player.y-Math.sin(player.face)*4+Math.sin(player.face+1.57)*fpSide*3.4,a:player.face,life:1});
      if(foots.length>48) foots.shift(); } }
  for(const d of drives){ if(d.got) continue;
    if(dist(player.x,player.y,d.x,d.y)<20){ d.got=true; gotCount++; sfx.pick(); burst(d.x,d.y,'63,224,255',16);
      toast(gotCount===3?'ALL DRIVES — REACH THE HATCH':'DRIVE SECURED '+gotCount+'/3');
      if(gotCount===3){ exitOpen=true; sfx.openExit(); objLabel.textContent='EXFIL HATCH'; objLabel.classList.add('open'); }
      syncPips(); } }
  const ed=dist(player.x,player.y,exitPos.x,exitPos.y);
  if(exitOpen&&ed<20){ winGame(); return; }
  if(!exitOpen&&!sealedTold&&ed<28){ sealedTold=true; toast('HATCH SEALED — SECURE ALL DRIVES'); }
}

function updGuard(g,dt){
  g.pop=Math.max(0,g.pop-dt*1.3); g.flash=Math.max(0,g.flash-dt*2);
  if(g.state==='down'){ g.downT+=dt; return; }
  let spot=false;
  const dx=player.x-g.x, dy=player.y-g.y, d=Math.hypot(dx,dy);
  if(state==='playing'){
    const rng=g.rng*(player.crouch?0.72:1)*(inBush(player)?0.6:1);
    if(d<rng && !(player.crouch&&inBush(player)&&d>80)){
      if(Math.abs(angleDiff(Math.atan2(dy,dx),g.face))<g.fov/2 && !losBlocked(g.x,g.y,player.x,player.y)) spot=true; } }
  g.spotting=spot;
  if(spot){ g.seeT+=dt; g.lastSeen.x=player.x; g.lastSeen.y=player.y;
    if(g.seeT>0.32 && g.state!=='chase'){ g.state='chase'; g.loseT=0; g.pop=1; g.flash=1; sfx.sting(); shake(3); } }
  else g.seeT=Math.max(0,g.seeT-dt*1.5);

  if(g.state==='chase'){
    if(!spot){ g.loseT+=dt; if(g.loseT>3.2){ g.state='search'; g.searchT=4.2; g.target={x:g.lastSeen.x,y:g.lastSeen.y}; } }
    else g.loseT=0;
    const tt=spot?player:g.lastSeen;
    g.repath-=dt;
    if(g.repath<=0||!g.path.length){ g.path=pathTo(g,Math.floor(tt.x/T),Math.floor(tt.y/T)); g.repath=0.4; }
    moveAlong(g,dt,g.spd*2.1); return;
  }
  if(g.state==='investigate'){
    if(!g.path.length){ g.state='search'; g.searchT=2.8; }
    else moveAlong(g,dt,g.spd*1.5); return;
  }
  if(g.state==='search'){
    g.searchT-=dt; g.face+=dt*2.6;
    if(g.searchT<=0){ g.state='patrol'; g.dwell=0; } return;
  }
  /* patrol */
  if(!g.path.length){
    if(g.dwell>0){ g.dwell-=dt; g.face+=dt*1.3; return; }
    const wp=g.wps[g.wpi%g.wps.length];
    g.path=pathTo(g,Math.floor(wp.x/T),Math.floor(wp.y/T));
  } else {
    const reached=moveAlong(g,dt,g.spd);
    if(reached){ g.dwell=.9+Math.random()*1.4; g.wpi++; }
  }
}

function updDecoy(dt){
  if(!decoy) return;
  if(decoy.landT>0){ decoy.landT-=dt; if(decoy.landT<=0) decoy=null; return; }
  const st=360*dt, nx=decoy.x+decoy.vx*dt, ny=decoy.y+decoy.vy*dt;
  decoy.trav+=st;
  if(solidAt(Math.floor(nx/T),Math.floor(ny/T))||decoy.trav>=decoy.max){
    decoy.landT=2.4; addNoise(decoy.x,decoy.y,250); sfx.ping(); return; }
  decoy.x=nx; decoy.y=ny;
}

function updEffects(dt){
  for(const r of rings){ r.r+=r.max*1.15*dt; r.a=1-r.r/r.max; }
  rings=rings.filter(r=>r.a>0);
  for(const p of parts){ p.x+=p.vx*dt; p.y+=p.vy*dt; p.vx*=0.92; p.vy*=0.92; p.life-=dt*1.7; }
  parts=parts.filter(p=>p.life>0);
  for(const f of foots) f.life-=dt*0.62;
  foots=foots.filter(f=>f.life>0);
  shakeMag=Math.max(0,shakeMag-8*dt);
}

function update(dt){
  updPlayer(dt); if(state!=='playing') return;
  guards.forEach(g=>updGuard(g,dt));
  updDecoy(dt);
  const seen=guards.some(g=>g.spotting);
  det+=dt*(seen?(player.crouch?0.45:1.05):-0.5);
  det=Math.max(0,Math.min(1,det));
  if(det>0.3){ hbT-=dt; if(hbT<=0){ sfx.thump(); hbT=0.28+(1-det)*0.5; } }
  cd=Math.max(0,cd-dt);
  updEffects(dt);
  if(det>=1) caughtGame();
}

function drawCone(g){
  const N=26, rng=g.rng, base=[[g.x,g.y]];
  for(let i=0;i<=N;i++){
    const a=g.face-g.fov/2+g.fov*i/N, ca=Math.cos(a), sa=Math.sin(a);
    let px=g.x,py=g.y;
    for(let s=4;s<=rng;s+=4){ px=g.x+ca*s; py=g.y+sa*s;
      if(solidAt(Math.floor(px/T),Math.floor(py/T))) break; }
    base.push([px,py]);
  }
  const ch=g.state==='chase', pulse=0.26+0.08*Math.sin(tSec*3+g.ph);
  const a=Math.min(.55,pulse+g.flash*.3);
  const col=ch?'255,70,60':'255,196,84';

  ctx.closePath(); ctx.fill();
}

    ctx.beginPath(); ctx.arc(r.x,r.y,r.r,0,7); ctx.stroke(); }
  for(const p of parts){ ctx.fillStyle='rgba('+p.col+','+Math.max(0,p.life)+')';
    ctx.fillRect(p.x-p.sz/2,p.y-p.sz/2,p.sz,p.sz); }

  if(decoy&&decoy.landT>0){ const bl=Math.sin(tSec*10)>0;
    ctx.fillStyle=bl?'#3fe0ff':'rgba(63,224,255,.3)';
    ctx.beginPath(); ctx.arc(decoy.x,decoy.y,3,0,7); ctx.fill(); }

  /* subdue prompt + hidden tag */
  if(state==='playing'){
    let best=null,bd=52;
    for(const g of guards){ if(g.state==='down') continue;
      const d=dist(player.x,player.y,g.x,g.y);
      if(d<bd&&!g.spotting&&g.state!=='chase'){ best=g; bd=d; } }
    if(best){ ctx.strokeStyle='rgba(63,224,255,'+(0.5+0.3*Math.sin(tSec*6))+')';
      ctx.setLineDash([3,4]); ctx.lineWidth=1.5;
      ctx.beginPath(); ctx.arc(best.x,best.y,15,0,7); ctx.stroke(); ctx.setLineDash([]);
      ctx.font='bold 12px Consolas, monospace'; ctx.textAlign='center';
      ctx.fillStyle='#3fe0ff'; ctx.fillText('E',best.x,best.y-22); }
    if(player.crouch&&inBush(player)&&guards.some(g=>dist(player.x,player.y,g.x,g.y)<240)){
      ctx.font='10px Consolas, monospace'; ctx.textAlign='center';
      ctx.fillStyle='rgba(92,255,157,'+(0.6+0.3*Math.sin(tSec*4))+')';
      ctx.fillText('HIDDEN',player.x,player.y-18); } }

  ctx.restore();

  /* vignette + detection pulse */
  let vg=ctx.createRadialGradient(W/2,H/2,H*0.42,W/2,H/2,H*0.85);
  vg.addColorStop(0,'rgba(0,0,0,0)'); vg.addColorStop(1,'rgba(0,0,0,.42)');
  ctx.fillStyle=vg; ctx.fillRect(0,0,W,H);
  if(det>0.02){ const a=det*(0.5+0.22*Math.sin(tSec*7));
    vg=ctx.createRadialGradient(W/2,H/2,H*0.3,W/2,H/2,H*0.8);
    vg.addColorStop(0,'rgba(255,40,60,0)'); vg.addColorStop(1,'rgba(255,40,60,'+(a*0.5)+')');
    ctx.fillStyle=vg; ctx.fillRect(0,0,W,H); }
}
