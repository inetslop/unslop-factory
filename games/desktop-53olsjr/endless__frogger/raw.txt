  tAmb+=dt;
  for(const f of flies){ f.ph+=dt*f.sp*2;
    f.x=(f.x+Math.sin(f.ph)*.01*dt+.0035*((f.layer===1)?2:1)); if(f.x>1)f.x-=1; if(f.x<0)f.x+=1;
    f.y=f.y+Math.cos(f.ph*.7)*.008*dt; if(f.y>1)f.y-=1; if(f.y<0)f.y+=1; }
  if(paused)return;
  if(state==='menu'){ tW+=dt; camY-=46*dt; updateTraffic(1,dt); }
  else if(state==='playing'){
    tW+=dt; rampT+=dt;
    const sp=22+Math.min(maxLane,60)*1.05;
    const e=Math.min(1,rampT/1.3), sm=e*e*(3-2*e);
    camY-=sp*sm*dt;
    updateTraffic(trafficMul(),dt);
    updateFrog(dt);
  }else if(state==='dying'){
    tW+=dt*.3; deathTimer-=dt; shake=Math.max(0,shake-42*dt);
    if(deathTimer<=0){ (lives<=0)?gameOver():respawn(); }
  }
  for(const p of particles){ p.life-=dt; p.x+=p.vx*dt; p.y+=p.vy*dt; p.vy+=(p.g||0)*dt; if(p.type==='ring')p.r+=p.gr*dt; }
  particles=particles.filter(p=>p.life>0);
  for(const p of popups){ p.life-=dt; p.y-=36*dt; }
  popups=popups.filter(p=>p.life>0);
}

const FD='Impact, "Arial Black", sans-serif', FB='"Trebuchet MS", Verdana, sans-serif';
function rr(x,y,w,h,r){ r=Math.min(r,w/2,h/2); g.beginPath();
  g.moveTo(x+r,y); g.arcTo(x+w,y,x+w,y+h,r); g.arcTo(x+w,y+h,x,y+h,r);
  g.arcTo(x,y+h,x,y,r); g.arcTo(x,y,x+w,y,r); g.closePath(); }
function heartPath(x,y,s){ g.beginPath(); g.moveTo(x,y+s*.6);
  g.bezierCurveTo(x-s*1.2,y-s*.3,x-s*.6,y-s*1.0,x,y-s*.4);
  g.bezierCurveTo(x+s*.6,y-s*1.0,x+s*1.2,y-s*.3,x,y+s*.6); g.closePath(); }

const flies=Array.from({length:26},()=>({x:Math.random(),y:Math.random(),ph:Math.random()*7,sp:.3+Math.random()*.5,s:.6+Math.random(),layer:Math.random()<.5?0:1}));
function drawFlies(layer){
