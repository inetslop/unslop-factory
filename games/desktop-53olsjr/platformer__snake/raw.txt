function spawnP(o){if(parts.length>420)parts.shift();parts.push(Object.assign({vx:0,vy:0,g:0,life:0,max:.6,size:3,color:'#fff',type:'dot',rot:0,vr:0},o));}
function updateParts(dt){for(let i=parts.length-1;i>=0;i--){const p=parts[i];p.life+=dt;if(p.life>=p.max){parts.splice(i,1);continue;}p.x+=p.vx*dt;p.y+=p.vy*dt;p.vy+=p.g*dt;p.rot+=p.vr*dt;}}

/* ============================== trail / snake math ============================== */
function pushTrail(){const s=game.snake,tr=s.trail,last=tr[tr.length-1];
  if(!last){tr.push({x:s.x,y:s.y,s:0});return;}
  const d=Math.hypot(s.x-last.x,s.y-last.y);if(d<.5)return;
  s.dist+=d;tr.push({x:s.x,y:s.y,s:s.dist});
  const cut=s.dist-(s.curLen+SPACING*2);while(tr.length>2&&tr[0].s<cut)tr.shift();}
function segPos(off){const s=game.snake,tr=s.trail,target=s.dist-off;
  if(target<=tr[0].s)return{x:tr[0].x,y:tr[0].y};
  for(let i=tr.length-1;i>0;i--){const a=tr[i],b=tr[i-1];
    if(a.s>=target&&b.s<=target){const t=(target-b.s)/((a.s-b.s)||1);return{x:b.x+(a.x-b.x)*t,y:b.y+(a.y-b.y)*t};}}
  return{x:tr[0].x,y:tr[0].y};}

/* ============================== physics ============================== */
function circleRect(cx,cy,r,rx,ry,rw,rh){const nx=clamp(cx,rx,rx+rw),ny=clamp(cy,ry,ry+rh);const dx=cx-nx,dy=cy-ny;return dx*dx+dy*dy<r*r;}

function updateSnake(dt){
  const s=game.snake;
  // coyote / buffer (from previous frame's grounded state)
  s.coyote=s.onGround?COYOTE:s.coyote-dt;s.jumpBuf-=dt;
  if(s.jumpBuf>0&&(s.onGround||s.coyote>0)){
    s.vy=-JUMPV;s.onGround=false;s.coyote=0;s.jumpBuf=0;s.sy=1.3;s.sx=.78;SFX.jump();
    for(let i=0;i<6;i++)spawnP({x:s.x+rand(-8,8),y:s.y+HEAD_R,vx:rand(-90,90),vy:rand(-30,40),max:.4,size:rand(2,4),color:'rgba(220,240,220,.7)',type:'dot'});
  }
  // horizontal
  let ax=0;if(keys.left)ax--;if(keys.right)ax++;
  if(ax){s.vx+=ax*MOVE*(s.onGround?1:AIRC)*dt;s.face=ax;}
  else{const f=(s.onGround?FRICT:260)*dt;if(Math.abs(s.vx)<=f)s.vx=0;else s.vx-=Math.sign(s.vx)*f;}
  s.vx=clamp(s.vx,-MAXV,MAXV);
  // gravity / dive / jump-cut
  let gm=G;
  if(!s.onGround){if(keys.down)gm*=DIVE_G;else if(s.vy<0&&!keys.jumpHeld)gm*=2.1;}
  s.vy=Math.min(s.vy+gm*dt,FALLMAX);
  // carry by moving platform
  if(s.groundPlat&&s.groundPlat.move){s.x+=s.groundPlat.dx;s.y+=s.groundPlat.dy;}
  // integrate + collide
  const wasAir=!s.onGround,prevVy=s.vy;
  s.x+=s.vx*dt;s.y+=s.vy*dt;
  s.x=clamp(s.x,6,game.LW-6);
  s.groundPlat=null;let grounded=false;
  for(const p of game.plats){
    if(p.kind==='crumble'&&p.state===2)continue;
    const nx0=clamp(s.x,p.x,p.x+p.w),ny0=clamp(s.y,p.y,p.y+p.h);
    let dx=s.x-nx0,dy=s.y-ny0,d2=dx*dx+dy*dy;
    if(d2<HEAD_R*HEAD_R){
      if(d2>.0001){const d=Math.sqrt(d2),nx=dx/d,ny=dy/d;s.x+=nx*(HEAD_R-d);s.y+=ny*(HEAD_R-d);
        if(ny<-.5&&s.vy>0){s.vy=0;grounded=true;s.groundPlat=p;}
        if(nx<-0.5&&s.vx>0)s.vx=0;if(nx>.5&&s.vx<0)s.vx=0;
        if(ny>.5&&s.vy<0)s.vy=0;}
      else{const l=s.x-p.x,r=p.x+p.w-s.x,t=s.y-p.y,b=p.y+p.h-s.y,m=Math.min(l,r,t,b);
        if(m===t){s.y=p.y-HEAD_R;if(s.vy>0)s.vy=0;grounded=true;s.groundPlat=p;}
        else if(m===b){s.y=p.y+p.h+HEAD_R;if(s.vy<0)s.vy=0;}
        else if(m===l){s.x=p.x-HEAD_R;if(s.vx>0)s.vx=0;}else{s.x=p.x+p.w+HEAD_R;if(s.vx<0)s.vx=0;}}
    }
  }
  s.onGround=grounded;
  if(grounded&&wasAir&&prevVy>420){s.sy=.68;s.sx=1.32;SFX.land();game.shake=Math.max(game.shake,2);
    for(let i=0;i<7;i++)spawnP({x:s.x+rand(-12,12),y:s.y+HEAD_R,vx:rand(-140,140),vy:rand(-60,10),max:.45,size:rand(2,5),color:'rgba(220,240,220,.7)',type:'dot'});}
  if(s.groundPlat&&s.groundPlat.kind==='crumble'&&s.groundPlat.state===0){s.groundPlat.state=1;s.groundPlat.t=0;}
  // trail & growth
  pushTrail();
  s.curLen+=(s.targetLen-s.curLen)*Math.min(1,dt*5);
  s.grace-=dt;
  // self collision
  if(s.grace<=0){const dl=Math.min(s.curLen,s.dist),n=Math.floor(dl/SPACING);
    for(let k=5;k<n;k++){const p=segPos(k*SPACING);if(Math.hypot(p.x-s.x,p.y-s.y)<(HEAD_R+10)*.72){die('self');return;}}}
  // orbs
  for(const o of game.orbs){if(o.got)continue;
    if(Math.hypot(o.x-s.x,o.y-s.y)<HEAD_R+13){o.got=true;game.collected++;game.score+=100;s.targetLen=Math.min(s.targetLen+GROWTH,26*SPACING);
      SFX.orb();SFX.grow();updateHUD();
      spawnP({x:o.x,y:o.y,life:0,max:.8,type:'text',text:'+100',color:'#ffd76a',vy:-55});
      for(let i=0;i<12;i++){const a=rand(TAU);spawnP({x:o.x,y:o.y,vx:Math.cos(a)*rand(60,240),vy:Math.sin(a)*rand(60,240),max:rand(.35,.7),size:rand(2,4),color:'#ffd76a',type:'spark'});}
      spawnP({x:o.x,y:o.y,life:0,max:.4,size:8,type:'ring',r1:44,color:'#ffd76a'});}}
  // spikes
  for(const sp of game.spikes)if(circleRect(s.x,s.y,HEAD_R*.8,sp.x+3,sp.y-24,sp.w-6,24)){die('spike');return;}
  // portal
  if(game.collected>=game.required&&Math.hypot(s.x-game.portal.x,s.y-game.portal.y)<36){clearLevel();return;}
  // fell
  if(s.y>KILLY)die('fell');
}

function updateLevel(dt){
  for(const p of game.plats){
    if(p.move){const s=.5-.5*Math.cos(TAU*game.time/p.move.period+p.move.phase),v=p.move.range*s;
      const nx=p.move.axis==='x'?p.bx+v:p.bx,ny=p.move.axis==='y'?p.by+v:p.by;
      p.dx=nx-p.x;p.dy=ny-p.y;p.x=nx;p.y=ny;}
    if(p.kind==='crumble'&&p.state===1){p.t+=dt;if(Math.random()<.4)spawnP({x:p.x+rand(p.w),y:p.y+p.h,vx:rand(-20,20),vy:rand(10,50),g:300,max:.4,size:2,color:'#c9a06b',type:'dot'});
      if(p.t>.55){p.state=2;SFX.crumble();game.shake=Math.max(game.shake,4);
        for(let i=0;i<10;i++)spawnP({x:p.x+rand(p.w),y:p.y+rand(p.h),vx:rand(-160,160),vy:rand(-220,40),g:900,max:rand(.5,.9),size:rand(3,7),color:'#b98d5f',type:'debris',rot:rand(TAU),vr:rand(-8,8)});}}
  }
}

/* ============================== update ============================== */
function update(dt){
  game.time+=dt;
  game.shake=Math.max(0,game.shake-dt*26);
  game.flash=Math.max(0,game.flash-dt*2.4);
  const s=game.snake;
  if(s){s.sx+=(1-s.sx)*Math.min(1,dt*10);s.sy+=(1-s.sy)*Math.min(1,dt*10);
    s.blinkT-=dt;if(s.blinkT<-.12)s.blinkT=rand(2,4.5);
    if(s.onGround&&Math.abs(s.vx)<40&&Math.random()<dt*1.2)s.tongue=.18;s.tongue-=dt;}
  updateParts(dt);

  if(game.state==='title'){
    const span=Math.max(0,game.LW-viewW);
    game.camX=(0.5+0.5*Math.sin(game.time*.1))*span;
  }
  else if(game.state==='playing'){
    game.runTime+=dt;updateLevel(dt);updateSnake(dt);
    if(game.state==='playing'&&s){
      const tx=clamp(s.x+s.vx*.26-viewW/2,0,Math.max(0,game.LW-viewW));
      game.camX+=(tx-game.camX)*Math.min(1,dt*6);}
  }
  else if(game.state==='dying'){
    game.statT+=dt;
    if(game.statT>.95){if(game.lives>0)respawn();else gameover();}
  }
  else if(game.state==='clear'){
    game.statT+=dt;
    if(s){s.curLen=Math.max(0,s.curLen-dt*SPACING*16);
      s.x=lerp(s.x,game.portal.x,Math.min(1,dt*5));s.y=lerp(s.y,game.portal.y,Math.min(1,dt*5));pushTrail();
      const a=rand(TAU),r=rand(30,46);spawnP({x:game.portal.x+Math.cos(a)*r,y:game.portal.y+Math.sin(a)*r,vx:-Math.cos(a)*120,vy:-Math.sin(a)*120,max:.5,size:3,color:'#7df2d6',type:'spark'});}
    if(game.statT>.85){
      if(game.levelIndex+1<LEVELS.length){loadLevel(game.levelIndex+1);game.state='playing';banner('LV '+(game.levelIndex+1)+' · '+LEVELS[game.levelIndex].name);}
      else victory();}
  }
}

function ridge(par,baseFrac,amp,color){

  g.lineTo(W,H);g.closePath();g.fill();
}

function drawPlat(p){
  let ox=0;if(p.kind==='crumble'&&p.state===1)ox=Math.sin(p.t*70)*Math.min(3,p.t*9);
  const x=p.x+ox,y=p.y;
  if(p.kind==='mover'){
    g.fillStyle='rgba(255,179,71,.16)';g.beginPath();g.ellipse(x+p.w/2,y+p.h+10,p.w*.42,8,0,0,TAU);g.fill();
  }
  const body=p.kind==='crumble'?'#b98d5f':'#28434c';
  g.fillStyle=body;rr(x,y,p.w,p.h,5);g.fill();
  g.fillStyle='rgba(0,0,0,.28)';g.fillRect(x+3,y+p.h-6,p.w-6,Math.min(6,p.h-8));
  if(p.kind==='crumble'){
    g.strokeStyle='#7a5a34';g.lineWidth=1.5;g.beginPath();
    g.moveTo(x+p.w*.3,y+2);g.lineTo(x+p.w*.42,y+p.h*.5);g.lineTo(x+p.w*.3,y+p.h-3);
    g.moveTo(x+p.w*.68,y+2);g.lineTo(x+p.w*.6,y+p.h*.6);g.stroke();
  }else{
    const moss=g.createLinearGradient(0,y,0,y+12);moss.addColorStop(0,'#8fd14f');moss.addColorStop(1,'#5ea63e');
    g.fillStyle=moss;rr(x,y,p.w,Math.min(11,p.h),5);g.fill();
    // grass tufts
    g.strokeStyle='rgba(190,240,140,.8)';g.lineWidth=2;const n=Math.floor(p.w/70);
    for(let i=0;i<n;i++){const tx=x+8+pr(i*3.7+p.x)*(p.w-16),sw=Math.sin(game.time*2+i)*1.5;
      g.beginPath();g.moveTo(tx,y+1);g.lineTo(tx-3+sw,y-6);g.moveTo(tx+2,y+1);g.lineTo(tx+2+sw,y-7);g.stroke();
      if(pr(i*9.1+p.y)>.8){g.fillStyle='#ffd76a';g.beginPath();g.arc(tx+5,y-4,2,0,TAU);g.fill();}}
  }
  if(p.kind==='mover'){g.fillStyle='#ffb347';g.beginPath();g.arc(x+10,y+p.h/2,2.5,0,TAU);g.arc(x+p.w-10,y+p.h/2,2.5,0,TAU);g.fill();}
}
function rr(x,y,w,h,r){r=Math.min(r,w/2,h/2);g.beginPath();g.moveTo(x+r,y);g.arcTo(x+w,y,x+w,y+h,r);g.arcTo(x+w,y+h,x,y+h,r);g.arcTo(x,y+h,x,y,r);g.arcTo(x,y,x+w,y,r);g.closePath();}

function drawSpikes(){
  for(const sp of game.spikes){const n=Math.max(1,Math.round(sp.w/20)),step=sp.w/n;
    g.fillStyle='#ff6b57';g.strokeStyle='rgba(90,20,10,.5)';g.lineWidth=2;
    g.beginPath();g.moveTo(sp.x,sp.y);
    for(let i=0;i<n;i++){g.lineTo(sp.x+i*step+step/2,sp.y-28);g.lineTo(sp.x+(i+1)*step,sp.y);}
    g.closePath();g.fill();g.stroke();
    g.fillStyle='#7e2418';g.fillRect(sp.x,sp.y-4,sp.w,5);}
}
function drawOrbs(){
  for(const o of game.orbs){if(o.got)continue;const bob=Math.sin(game.time*3+o.i)*5,y=o.y+bob;
    g.globalCompositeOperation='lighter';
    const gl=g.createRadialGradient(o.x,y,2,o.x,y,26);gl.addColorStop(0,'rgba(255,215,106,.8)');gl.addColorStop(1,'rgba(255,179,71,0)');
    g.fillStyle=gl;g.beginPath();g.arc(o.x,y,26,0,TAU);g.fill();g.globalCompositeOperation='source-over';
    g.fillStyle='#ffd76a';g.beginPath();g.arc(o.x,y,8,0,TAU);g.fill();
    g.fillStyle='#fff6d8';g.beginPath();g.arc(o.x-2.5,y-2.5,3,0,TAU);g.fill();
    g.save();g.translate(o.x,y);g.rotate(game.time*1.5+o.i);g.strokeStyle='rgba(255,246,216,.7)';g.lineWidth=1.5;
    g.beginPath();g.moveTo(-13,0);g.lineTo(13,0);g.moveTo(0,-13);g.lineTo(0,13);g.stroke();g.restore();}
}
function drawPortal(){
  const p=game.portal,on=game.collected>=game.required;
  g.save();g.translate(p.x,p.y);
  if(on){g.globalCompositeOperation='lighter';const gl=g.createRadialGradient(0,0,4,0,0,60);gl.addColorStop(0,'rgba(125,242,214,.5)');gl.addColorStop(1,'rgba(125,242,214,0)');
    g.fillStyle=gl;g.beginPath();g.arc(0,0,60,0,TAU);g.fill();g.globalCompositeOperation='source-over';}
  g.lineWidth=5;g.setLineDash([10,8]);g.lineDashOffset=-game.time*40;
  g.strokeStyle=on?'#7df2d6':'#54707a';g.beginPath();g.arc(0,0,34,0,TAU);g.stroke();g.setLineDash([]);
  for(let i=0;i<3;i++){const ph=(game.time*.8+i/3)%1,r=26*(1-ph)+4;
    g.globalAlpha=(on?.8:.3)*(1-ph);g.strokeStyle=on?'#c9fff0':'#7c98a1';g.lineWidth=2.5;
    g.beginPath();g.arc(0,0,r,ph*TAU+i*2,ph*TAU+i*2+2.2);g.stroke();}
  g.globalAlpha=1;
  if(!on){const rem=game.required-game.collected;
    for(let i=0;i<rem;i++){const a=-Math.PI/2+(i-(rem-1)/2)*.5;
      g.fillStyle='#ffd76a';g.beginPath();g.arc(Math.cos(a)*52,Math.sin(a)*52,4,0,TAU);g.fill();}}
  else{g.fillStyle='#c9fff0';g.beginPath();g.arc(0,0,6+Math.sin(game.time*6)*2,0,TAU);g.fill();}
  g.restore();
}

function segR(k,n){return lerp(6.5,12,(n-k)/Math.max(1,n));}
function drawSnakeBody(sx,sy,face,trailPts){
  for(let i=trailPts.length-1;i>=0;i--){const p=trailPts[i];
    g.fillStyle=i%2?'#98c93c':'#b9e14d';g.strokeStyle='#2c4a16';g.lineWidth=2.5;
    g.beginPath();g.arc(p.x,p.y,p.r,0,TAU);g.fill();g.stroke();}
  // head
  const br=s.blinkT<0?.15:1;
  g.save();g.translate(sx,sy);g.scale(face*s.sx,s.sy);
  g.fillStyle='#c9ec5f';g.strokeStyle='#2c4a16';g.lineWidth=3;
  g.beginPath();g.ellipse(3,0,HEAD_R+2,HEAD_R-1,0,0,TAU);g.fill();g.stroke();
  if(s.tongue>0){g.strokeStyle='#ff5d73';g.lineWidth=2;const w=Math.sin(game.time*40)*2;
    g.beginPath();g.moveTo(HEAD_R+2,3);g.lineTo(HEAD_R+11,4+w);g.moveTo(HEAD_R+11,4+w);g.lineTo(HEAD_R+15,1+w);g.moveTo(HEAD_R+11,4+w);g.lineTo(HEAD_R+15,7+w);g.stroke();}
  g.fillStyle='#fff';g.beginPath();g.ellipse(5,-5,5,5*br,0,0,TAU);g.fill();
  if(br>.5){const lx=clamp(s.vx/MAXV,-1,1),ly=clamp(s.vy/600,-1,1);
    g.fillStyle='#1c2a10';g.beginPath();g.arc(5+lx*2.2,-5+ly*1.8,2.6,0,TAU);g.fill();}
  g.fillStyle='#2c4a16';g.beginPath();g.arc(HEAD_R+1,-1,1.3,0,TAU);g.fill();
  g.restore();
}
function drawSnake(){
  const s=game.snake,dl=Math.min(s.curLen,s.dist),n=Math.max(1,Math.floor(dl/SPACING)),frac=(dl/SPACING)%1;
  const pts=[];if(frac>.2)pts.push({p:segPos(dl),r:4+6*frac});
  for(let k=n;k>=1;k--)pts.push({p:segPos(k*SPACING),r:segR(k,n)});
  drawSnakeBody(s.x,s.y,s.face,pts.map(q=>({x:q.p.x,y:q.p.y,r:q.r})));
}
function drawDemoSnake(){
  const s=game.snake,pts=[];
  for(let k=7;k>=1;k--)pts.push({x:s.x-k*SPACING,y:s.y+Math.sin(game.time*2.4+k*.55)*4,r:segR(k,7)});
  drawSnakeBody(s.x,s.y,1,pts);
}

function drawParts(){
  for(const p of parts){const a=1-p.life/p.max;
    if(p.type==='dot'){g.globalAlpha=a;g.fillStyle=p.color;g.beginPath();g.arc(p.x,p.y,p.size*a+0.5,0,TAU);g.fill();}
    else if(p.type==='spark'){g.globalCompositeOperation='lighter';g.globalAlpha=a;g.fillStyle=p.color;g.beginPath();g.arc(p.x,p.y,p.size,0,TAU);g.fill();g.globalCompositeOperation='source-over';}
    else if(p.type==='debris'){g.globalAlpha=a;g.fillStyle=p.color;g.save();g.translate(p.x,p.y);g.rotate(p.rot);g.fillRect(-p.size/2,-p.size/2,p.size,p.size);g.restore();}
    else if(p.type==='ring'){const r=lerp(p.size||8,p.r1,p.life/p.max);g.globalAlpha=a;g.strokeStyle=p.color;g.lineWidth=3;g.beginPath();g.arc(p.x,p.y,r,0,TAU);g.stroke();}
    else if(p.type==='text'){g.globalAlpha=Math.min(1,a*1.6);g.font='900 22px "Arial Black",Impact,sans-serif';g.textAlign='center';
      g.strokeStyle='rgba(6,20,26,.9)';g.lineWidth=4;g.strokeText(p.text,p.x,p.y);g.fillStyle=p.color;g.fillText(p.text,p.x,p.y);}
    g.globalAlpha=1;}
}

function render(){
  g.setTransform(DPR,0,0,DPR,0,0);
  drawBG();
  const shx=(Math.random()-.5)*game.shake,shy=(Math.random()-.5)*game.shake;
  g.save();g.translate(W/2+shx,H/2+shy);g.scale(ZOOM,ZOOM);g.translate(-game.camX-viewW/2,-viewH/2);
  for(const p of game.plats)if(p.kind!=='crumble'||p.state!==2)drawPlat(p);
  drawSpikes();drawPortal();drawOrbs();
  if(game.state==='title')drawDemoSnake();
  else if(game.state!=='dying')drawSnake();
  drawParts();
  g.restore();
  // flash & vignette
  if(game.flash>0){g.fillStyle='rgba('+game.flashCol+','+(game.flash*.45).toFixed(3)+')';g.fillRect(0,0,W,H);}
  const vg=g.createRadialGradient(W/2,H/2,Math.min(W,H)*.42,W/2,H/2,Math.max(W,H)*.75);
  vg.addColorStop(0,'rgba(0,10,14,0)');vg.addColorStop(1,'rgba(0,10,14,.38)');g.fillStyle=vg;g.fillRect(0,0,W,H);
  const tg=g.createLinearGradient(0,0,0,80);tg.addColorStop(0,'rgba(4,16,20,.5)');tg.addColorStop(1,'rgba(4,16,20,0)');g.fillStyle=tg;g.fillRect(0,0,W,80);
}
