function G(x,y){return (x<0||y<0||x>=GW||y>=GH)?1:maze[y*GW+x];}
function makeMaze(){
  const g=new Uint8Array(GW*GH).fill(1),I=function(x,y){return y*GW+x;};
  g[I(1,1)]=0;const st=[[1,1]];
  while(st.length){
    const c=st[st.length-1],cx=c[0],cy=c[1],opts=[];
    for(let d=0;d<4;d++){const nx=cx+2*DIRS[d][0],ny=cy+2*DIRS[d][1];
      if(nx>0&&ny>0&&nx<GW-1&&ny<GH-1&&g[I(nx,ny)])opts.push([nx,ny,cx+DIRS[d][0],cy+DIRS[d][1]]);}
    if(!opts.length){st.pop();continue;}
    const o=opts[(Math.random()*opts.length)|0];
    g[I(o[2],o[3])]=0;g[I(o[0],o[1])]=0;st.push([o[0],o[1]]);
  }
  for(let y=1;y<GH-1;y++)for(let x=1;x<GW-1;x++){ // braid: open some walls into loops
    if(g[I(x,y)]===1){
      const h=g[I(x-1,y)]===0&&g[I(x+1,y)]===0, v=g[I(x,y-1)]===0&&g[I(x,y+1)]===0;
      if((h||v)&&Math.random()<0.32)g[I(x,y)]=0;
    }
  }
  return g;
}

  }
  return [1,1];
}

    if(state==='play'&&Math.hypot(pw.x-p.x,pw.y-p.y)<0.62){applyPower(pw.type);explosion(pw.x,pw.y,'#7dff9e',8,false);powers.splice(i,1);}
  }
}
function updatePlayer(dt){
  let ax=(keys.ArrowRight||keys.KeyD?1:0)-(keys.ArrowLeft||keys.KeyA?1:0);
  let ay=(keys.ArrowDown||keys.KeyS?1:0)-(keys.ArrowUp||keys.KeyW?1:0);
  if(touch.active){const dx=touch.gx-p.x,dy=touch.gy-p.y,d=Math.hypot(dx,dy);
    if(d>0.35){ax=dx/d;ay=dy/d;}else{ax=0;ay=0;}}
  if(ax&&ay){ax*=0.7071;ay*=0.7071;}
  const sp=7.4,k=Math.min(1,dt*13);
  p.vx+=(ax*sp-p.vx)*k;p.vy+=(ay*sp-p.vy)*k;
  if(ay)p.dir=[0,Math.sign(ay)];else if(ax)p.dir=[Math.sign(ax),0];
  const nx=p.x+p.vx*dt;if(!hitsWall(nx,p.y,p.r))p.x=nx;else p.vx=0;
  const ny=p.y+p.vy*dt;if(!hitsWall(p.x,ny,p.r))p.y=ny;else p.vy=0;
  const ta=Math.atan2(p.dir[1],p.dir[0]);let da=ta-p.ang;
  while(da>Math.PI)da-=6.283;while(da<-Math.PI)da+=6.283;p.ang+=da*Math.min(1,dt*14);
  p.cool-=dt;p.invuln=Math.max(0,p.invuln-dt);p.rapidT=Math.max(0,p.rapidT-dt);
  p.doubleT=Math.max(0,p.doubleT-dt);p.shieldT=Math.max(0,p.shieldT-dt);p.mf=Math.max(0,p.mf-dt);
  if((keys.Space||touch.fire)&&p.cool<=0){
    p.cool=p.rapidT>0?0.13:0.3;p.mf=0.06;
    const d=p.dir,sx=p.x+d[0]*0.5,sy=p.y+d[1]*0.5,per=[-d[1],d[0]];
    if(p.doubleT>0){bullets.push({x:sx+per[0]*0.24,y:sy+per[1]*0.24,dx:d[0],dy:d[1]});
                    bullets.push({x:sx-per[0]*0.24,y:sy-per[1]*0.24,dx:d[0],dy:d[1]});}
    else bullets.push({x:sx,y:sy,dx:d[0],dy:d[1]});
    sfx('shoot');
  }
}
function hitsWall(x,y,r){
  const x0=Math.floor(x-r),x1=Math.floor(x+r),y0=Math.floor(y-r),y1=Math.floor(y+r);
  for(let yy=y0;yy<=y1;yy++)for(let xx=x0;xx<=x1;xx++)if(G(xx,yy)===1)return true;
  return false;
}
function updateParticles(dt){
  for(let i=particles.length-1;i>=0;i--){const q=particles[i];q.life+=dt;
    if(q.kind==='spark'){q.x+=q.vx*dt;q.y+=q.vy*dt;q.vx*=0.92;q.vy*=0.92;}
    else q.r+=q.vr*dt;
    if(q.life>=q.max)particles.splice(i,1);}
  for(let i=popups.length-1;i>=0;i--){popups[i].t+=dt;if(popups[i].t>0.95)popups.splice(i,1);}
}
