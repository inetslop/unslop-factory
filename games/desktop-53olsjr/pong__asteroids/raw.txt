function foldY(y){const r=BALL_R,span=VH-2*r;let m=(y-r)%(2*span);if(m<0)m+=2*span;return r+(m<=span?m:2*span-m);}
function aiSteer(p,dt,skill){
  p.errT-=dt;if(p.errT<=0){p.err=(Math.random()*2-1)*(1-skill)*90;p.errT=.25+Math.random()*.35;}
  let target=VH/2;
  const toward=p.side===1?ball.vx<0:ball.vx>0;
  if(serveTimer<=0&&toward){const tHit=Math.abs((p.x-ball.x)/ball.vx);target=foldY(ball.y+ball.vy*tHit);}
  else target=VH/2+Math.sin(tGlobal*.8+p.side)*36;
  target+=p.err||0;
  const maxV=300+skill*260,d=target-p.y,mv=clamp(d*10,-maxV,maxV);
  p.vy+=(mv-p.vy)*Math.min(1,dt*12);
}
function fire(){
  if(pCd>0)return;pCd=.24;shots++;
  bolts.push({x:P.x+P.w/2+8,y:P.y,vx:820});
  sparks(P.x+P.w/2+8,P.y,'#8cfbe4',4,120);P.recoil=1;SFX.fire();
}
function paddleHit(p,prevX){
  const face=p.x+p.w/2*(p.side===1?1:-1);
  if(p.side===1?!(ball.vx<0):!(ball.vx>0))return false;
  const crossed=p.side===1?(prevX-ball.r>=face&&ball.x-ball.r<=face):(prevX+ball.r<=face&&ball.x+ball.r>=face);
  if(!crossed||Math.abs(ball.y-p.y)>p.h/2+ball.r)return false;
  const rel=clamp((ball.y-p.y)/(p.h/2),-1,1),ang=rel*1.05;
  const sp=Math.min(BALL_MAX,Math.hypot(ball.vx,ball.vy)*1.045+16);
  ball.x=face+(p.side===1?ball.r:-ball.r);
  ball.vx=Math.cos(ang)*sp*(p.side===1?1:-1);ball.vy=Math.sin(ang)*sp;
  lastTouch=p.side===1?'P':'A';
  p.hitPulse=1;if(p.side===1)P.recoil=1;
  sparks(face,ball.y,p.side===1?'#5cf2d8':'#ffb45e',6,0);
  shake=Math.min(18,shake+2);if(liveSfx())SFX.paddle();
  return true;
}
function rockHitsPaddle(a,p){
  const cx=clamp(a.x,p.x-p.w/2,p.x+p.w/2),cy=clamp(a.y,p.y-p.h/2,p.y+p.h/2);
  const dx=a.x-cx,dy=a.y-cy;return dx*dx+dy*dy<a.R*a.R;
}

function stepParts(dt){for(let i=parts.length-1;i>=0;i--){const p=parts[i];const dr=Math.max(0,1-2.4*dt);p.x+=p.vx*dt;p.y+=p.vy*dt;p.vx*=dr;p.vy*=dr;p.life-=dt;if(p.life<=0)parts.splice(i,1);}}
function stepRings(dt){for(let i=rings.length-1;i>=0;i--){const r=rings[i];r.r+=(r.max-r.r)*Math.min(1,dt*7);r.life-=dt*2;if(r.life<=0)rings.splice(i,1);}}
function stepFloaters(dt){for(let i=floaters.length-1;i>=0;i--){const f=floaters[i];f.y+=f.vy*dt;f.life-=dt*1.1;if(f.life<=0)floaters.splice(i,1);}}
function stepRocksMotion(dt,cull){for(let i=asteroids.length-1;i>=0;i--){const a=asteroids[i];a.x+=a.vx*dt;a.y+=a.vy*dt;a.rot+=a.vr*dt;if(cull&&(a.x<-160||a.x>VW+160||a.y<-160||a.y>VH+160))asteroids.splice(i,1);}}

function worldStep(dt,dtr,demo){
  t+=dt;pCd-=dt;invuln-=dt;
  flashRed=Math.max(0,flashRed-dtr*1.8);
  shake=Math.max(0,shake-dtr*(shake*6+4));
  P.hitPulse=Math.max(0,P.hitPulse-dtr*4);A.hitPulse=Math.max(0,A.hitPulse-dtr*4);
  P.recoil=Math.max(0,P.recoil-dtr*5);

  if(!demo){
    let desired;
    if(pointerActive&&(tGlobal-pointerT<3))desired=clamp((pointerY-P.y)*10,-820,820);
    else{const dir=(held('ArrowDown','s','S')?1:0)-(held('ArrowUp','w','W')?1:0);desired=dir*560;}
    P.vy+=(desired-P.vy)*Math.min(1,dtr*14);
  }else aiSteer(P,dt,.5);
  aiSteer(A,dt,demo?.6:(.5+threat*.32));
  P.y=clamp(P.y+P.vy*dt,30,VH-30);A.y=clamp(A.y+A.vy*dt,30,VH-30);

  if(!demo&&held(' ')&&pCd<=0)fire();

  for(let i=bolts.length-1;i>=0;i--){const b=bolts[i];b.x+=b.vx*dt;let kill=b.x>VW+40;
    if(!kill)for(const a of asteroids){if(a.dead)continue;const dx=a.x-b.x,dy=a.y-b.y;if(dx*dx+dy*dy<(a.R+5)*(a.R+5)){shatter(a,'P');kill=true;break;}}
    if(kill)bolts.splice(i,1);
  }

  for(let i=asteroids.length-1;i>=0;i--){const a=asteroids[i];if(a.dead){asteroids.splice(i,1);continue;}
    a.x+=a.vx*dt;a.y+=a.vy*dt;a.rot+=a.vr*dt;
    if(a.x<-160||a.x>VW+160||a.y<-160||a.y>VH+160){asteroids.splice(i,1);continue;}
    if(serveTimer<=0&&!a.dead){const dx=ball.x-a.x,dy=ball.y-a.y,rr=a.R+ball.r;
      if(dx*dx+dy*dy<rr*rr){const d=Math.hypot(dx,dy)||.001,nx=dx/d,ny=dy/d;
        ball.x=a.x+nx*(a.R+ball.r);ball.y=a.y+ny*(a.R+ball.r);
        const dot=ball.vx*nx+ball.vy*ny;if(dot<0){ball.vx-=2*dot*nx;ball.vy-=2*dot*ny;}
        const sp=Math.min(BALL_MAX,Math.hypot(ball.vx,ball.vy)*1.03);const cur=Math.hypot(ball.vx,ball.vy)||1;
        ball.vx*=sp/cur;ball.vy*=sp/cur;shatter(a,lastTouch||'P');}}
    if(!a.dead&&!(P===P&&invuln>0)&&rockHitsPaddle(a,P)){if(demo)shatter(a,null);else{loseLife();shatter(a,null);}}
    if(!a.dead&&rockHitsPaddle(a,A))shatter(a,null);
  }

  if(serveTimer>0){serveTimer-=dt;if(serveTimer<=0)launchBall();}
  else{
    const prevX=ball.x;
    ball.x+=ball.vx*dt;ball.y+=ball.vy*dt;
    trail.push({x:ball.x,y:ball.y});if(trail.length>15)trail.shift();
    if(ball.y<ball.r){ball.y=ball.r;ball.vy=Math.abs(ball.vy);sparks(ball.x,4,'#cfe2ff',3);if(liveSfx())SFX.wall();}
    else if(ball.y>VH-ball.r){ball.y=VH-ball.r;ball.vy=-Math.abs(ball.vy);sparks(ball.x,VH-4,'#cfe2ff',3);if(liveSfx())SFX.wall();}
    paddleHit(P,prevX);paddleHit(A,prevX);
    if(ball.x<-40)onGoal('A',demo);else if(ball.x>VW+40)onGoal('P',demo);
  }

  spawnT-=dt;const cap=demo?6:Math.min(9,3+Math.round(threat*6));
  if(spawnT<=0){if(asteroids.length<cap)spawnAsteroid();spawnT=lerp(3.4,1.4,threat)*(.7+Math.random()*.6);}

  stepParts(dt);stepRings(dt);stepFloaters(dt);
}
function cosmeticStep(dt){
  for(let i=asteroids.length-1;i>=0;i--){const a=asteroids[i];a.x+=a.vx*dt;a.y+=a.vy*dt;a.rot+=a.vr*dt;if(a.x<-200||a.x>VW+200||a.y<-200||a.y>VH+200)asteroids.splice(i,1);}
  stepParts(dt);stepRings(dt);stepFloaters(dt);
}

function tracePath(v){ctx.beginPath();for(let i=0;i<v.length;i++){i?ctx.lineTo(v[i].x,v[i].y):ctx.moveTo(v[i].x,v[i].y);}ctx.closePath();}
