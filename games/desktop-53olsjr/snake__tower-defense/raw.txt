function tone(f,dur,type,vol,f2,when){
  if(!AC||muted)return; type=type||'sine'; vol=vol||0.15; when=when||0;
  const t=AC.currentTime+when,o=AC.createOscillator(),g=AC.createGain();
  o.type=type;o.frequency.setValueAtTime(f,t);
  if(f2)o.frequency.exponentialRampToValueAtTime(Math.max(1,f2),t+dur);
  g.gain.setValueAtTime(0.0001,t);
  g.gain.exponentialRampToValueAtTime(vol,t+0.012);
  g.gain.exponentialRampToValueAtTime(0.0001,t+dur);
  o.connect(g).connect(master);o.start(t);o.stop(t+dur+0.05);
}
function noise(dur,vol,fq){
  if(!AC||muted)return;
  const n=Math.max(1,Math.floor(AC.sampleRate*dur)),b=AC.createBuffer(1,n,AC.sampleRate),d=b.getChannelData(0);
  for(let i=0;i<n;i++)d[i]=(Math.random()*2-1)*(1-i/n);
  const s=AC.createBufferSource();s.buffer=b;
  const f=AC.createBiquadFilter();f.type='lowpass';f.frequency.value=fq||800;
  const g=AC.createGain();g.gain.value=vol||0.2;
  s.connect(f).connect(g).connect(master);s.start();
}

function pickType(){
  const w=G.wave,pool=[['cr',1]];
  if(w>=2)pool.push(['sw',0.35]);
  if(w>=3)pool.push(['sp',0.3]);
  if(w>=4)pool.push(['br',0.28]);
  let tot=0; pool.forEach(p=>tot+=p[1]);
  let r=rand(tot);
  for(const p of pool){ if((r-=p[1])<=0)return p[0]; }
  return 'cr';
}

function spawnEnemy(type,gate,noSplit){
  const b=ETYPES[type], mult=1+(G.wave-1)*0.18, sm=Math.min(1.35,1+(G.wave-1)*0.025);
  let sx,sy;
  if(gate.s==='N'){sx=px(gate.x);sy=OY-CELL*0.7;}
  else if(gate.s==='S'){sx=px(gate.x);sy=OY+BH+CELL*0.7;}
  else if(gate.s==='W'){sx=OX-CELL*0.7;sy=py(gate.y);}
  else{sx=OX+BW+CELL*0.7;sy=py(gate.y);}
  G.enemies.push({type,x:sx,y:sy,hp:Math.ceil(b.hp*mult),maxhp:Math.ceil(b.hp*mult),
    spd:b.spd*sm*CELL,r:b.r*CELL,dmg:b.dmg,score:b.score,en:b.en,col:b.col,
    slowT:0,ramCd:0,flash:0,ph:rand(TAU),wob:type==='sw'?1.5:type==='cr'?0.7:0.25,split:!noSplit&&!type.endsWith('x')});
  gate.glow=0.8;
}
function hurt(e,dm){
  if(e.dead||e.hp<=0)return;
  e.hp-=dm; e.flash=0.1;
  if(e.hp<=0)killEnemy(e);
}
function killEnemy(e){
  e.dead=true; G.score+=e.score; G.kills++;

  while(en>0){const v=Math.min(8,en);spawnOrb('en',v,e.x+rand(-9,9),e.y+rand(-9,9));en-=v;}

  spark(e.x,e.y,e.col,10,150); SFX.pop();
  if(e.split){
    const mult=1+(G.wave-1)*0.18;
    for(let i=0;i<2;i++)spawnEnemy('sw',{s:'N'},true), G.enemies[G.enemies.length-1].x=e.x+rand(-CELL*0.5,CELL*0.5),
      G.enemies[G.enemies.length-1].y=e.y+rand(-CELL*0.4,CELL*0.4),
      G.enemies[G.enemies.length-1].hp=Math.ceil(mult), G.enemies[G.enemies.length-1].maxhp=Math.ceil(mult);
  }
}

/* ============================== snake actions ============================== */
function turn(dv){
  const sn=G.snake;
  const last=sn.queue.length?sn.queue[sn.queue.length-1]:sn.dir;
  if((dv.x===-last.x&&dv.y===-last.y)||(dv.x===last.x&&dv.y===last.y))return;
  if(sn.queue.length<3)sn.queue.push(dv);
}
function build(i){
  const T=TYPES[i], sn=G.snake, h=sn.cells[0];
  const bx=h.x+sn.dir.x, by=h.y+sn.dir.y;
  const okB=(bx>=0&&by>=0&&bx<COLS&&by<ROWS);
  const okC=!(bx===CORE.x&&by===CORE.y);
  const okT=!G.towers.some(t=>t.cx===bx&&t.cy===by);
  const okS=!sn.cells.some(c=>c.x===bx&&c.y===by);
  if(!okB||!okC||!okT||!okS){ SFX.invalid(); addFloat(px(clamp(bx,0,COLS-1)),py(clamp(by,0,ROWS-1)),'BLOCKED','#ff6b5d'); return; }
  if(G.energy<T.cost){ SFX.invalid(); addFloat(px(bx),py(by),'NEED '+T.cost+'⚡','#ff6b5d'); return; }
  G.energy-=T.cost; sn.built++;
  G.towers.push({cx:bx,cy:by,x:px(bx),y:py(by),type:i,cd:0,angle:-Math.PI/2,reveal:1.2});
  SFX.build(); spark(px(bx),py(by),T.col,8,90); addFloat(px(bx),py(by)-CELL*0.5,'-'+T.cost+'⚡',T.col);
}
function sell(){
  const h=G.snake.cells[0];
  const i=G.towers.findIndex(t=>t.cx===h.x&&t.cy===h.y);
  if(i<0){SFX.invalid();return;}
  const t=G.towers[i], ref=Math.round(TYPES[t.type].cost*0.6);
  G.energy+=ref; G.towers.splice(i,1);
  SFX.click(); spark(t.x,t.y,'#9cb8a6',8,90); addFloat(t.x,t.y-CELL*0.5,'+'+ref+'⚡','#ffd76a');
}
function stepSnake(){
  const sn=G.snake,c=sn.cells;
  if(sn.queue.length)sn.dir=sn.queue.shift();
  const h=c[0], nx=h.x+sn.dir.x, ny=h.y+sn.dir.y;
  if(nx<0||ny<0||nx>=COLS||ny>=ROWS){
    if(sn.dir.x)sn.dir.x*=-1; else sn.dir.y*=-1;
    G.shake=Math.max(G.shake,2); SFX.bump(); return;
  }
  for(let i=1;i<c.length;i++)if(c[i].x===nx&&c[i].y===ny){wound();return;}
  const growing=sn.grow>0;
  sn.prev=c.map(p=>({x:p.x,y:p.y}));
  c.unshift({x:nx,y:ny});
  if(growing)sn.grow--; else {c.pop();sn.prev.pop();}
}
function wound(){
  const sn=G.snake,cut=Math.max(3,Math.floor(sn.cells.length*0.4));
  sn.cells.length-=cut; sn.grow=0;
  sn.prev=sn.cells.map(p=>({x:p.x,y:p.y}));
  G.core=Math.max(0,G.core-5); sn.flash=0.4; G.shake=7; SFX.wound();
  const h=sn.cells[0]; addFloat(px(h.x),py(h.y)-CELL*0.6,'WOUND  -5','#ff6b5d');
  spark(px(h.x),py(h.y),'#ff6b5d',12,160);
  if(G.core<=0)gameOver();
}
function computePts(){
  const sn=G.snake,f=clamp(sn.acc/TICK,0,1);
  G.pts=sn.cells.map((c,i)=>{
    const p=sn.prev&&sn.prev[i];
    return p?{x:px(p.x+(c.x-p.x)*f),y:py(p.y+(c.y-p.y)*f)}:{x:px(c.x),y:py(c.y)};
  });
}

/* ============================== update ============================== */
function update(dt){
  const sn=G.snake, cx=px(CORE.x), cy=py(CORE.y);

  /* wave phase */
  if(G.phase==='prep'){
    G.prepT-=dt;
    if(G.prepT<=0)G.phase='active';
  }else{
    if(G.toSpawn.length){
      G.spawnT-=dt;
      if(G.spawnT<=0){
        const gs=activeGates(); spawnEnemy(G.toSpawn.shift(),gs[Math.floor(rand(gs.length))]);
        G.spawnT=Math.max(0.35,1.05-G.wave*0.05)*rand(0.7,1.3);
      }
    }else if(!G.enemies.length){
      const bonus=15+G.wave*5;
      G.energy+=bonus; G.score+=40+G.wave*10;
      G.msg='WAVE '+G.wave+' CLEAR   +'+bonus+'⚡'; G.msgT=1.6; SFX.clear();
      nextWave();
    }
  }
  G.msgT=Math.max(0,G.msgT-dt);

  /* snake tick */
  sn.acc+=dt;
  while(sn.acc>=TICK){sn.acc-=TICK;stepSnake();}
  computePts();
  sn.flash=Math.max(0,sn.flash-dt*2);

  /* enemies */
  for(const e of G.enemies){
    if(e.dead)continue;
    e.slowT=Math.max(0,e.slowT-dt); e.ramCd=Math.max(0,e.ramCd-dt); e.flash=Math.max(0,e.flash-dt);
    const slow=e.slowT>0?0.45:1, sp=e.spd*slow;
    let dx=cx-e.x, dy=cy-e.y; const dl=Math.hypot(dx,dy)||1; dx/=dl; dy/=dl;
    e.ph+=dt*(e.type==='sw'?9:5);
    const wv=Math.sin(e.ph)*e.wob*sp;
    e.x+=(dx-w? 0:0); // (kept simple below)
    e.x+= (dx + (-dy)*e.wob*0.9)*sp*dt*0 + (dx)*sp*dt + (-dy)*wv*dt*1;
    e.y+= (dy)*sp*dt + (dx)*wv*dt;
    if(d2(e.x,e.y,cx,cy)<Math.pow(CELL*0.85,2)){
      e.dead=true; G.core=Math.max(0,G.core-e.dmg); G.flashR=0.35; G.shake=7; SFX.leak();
      spark(cx,cy,'#ff6b5d',14,170); addFloat(cx,cy-CELL*1.2,'-'+e.dmg+' EGG','#ff6b5d');
      if(G.core<=0){gameOver();return;}
    }
  }
  /* separation */
  for(let i=0;i<G.enemies.length;i++)for(let j=i+1;j<G.enemies.length;j++){
    const a=G.enemies[i],b=G.enemies[j]; if(a.dead||b.dead)continue;
    const md=(a.r+b.r)*0.9, dd=d2(a.x,a.y,b.x,b.y);
    if(dd>0&&dd<md*md){
      const dist=Math.sqrt(dd),ov=(md-dist)*0.35,nx=(b.x-a.x)/dist,ny=(b.y-a.y)/dist;
      a.x-=nx*ov;a.y-=ny*ov;b.x+=nx*ov;b.y+=ny*ov;
    }
  }

  /* ramming */
  const pts=G.pts;
  for(const e of G.enemies){
    if(e.dead||e.ramCd>0)continue;
    let bi=-1,bd=1e18;
    for(let i=0;i<pts.length;i++){
      const q=pts[i], rr=(i===0?CELL*0.55:CELL*0.42)+e.r*0.8, dd=d2(e.x,e.y,q.x,q.y);
      if(dd<rr*rr&&dd<bd){bd=dd;bi=i;}
    }
    if(bi>=0){
      hurt(e,bi===0?2:1); e.ramCd=0.45;
      const q=pts[bi]; let nx=e.x-q.x,ny=e.y-q.y; const l=Math.hypot(nx,ny)||1;
      e.x+=nx/l*CELL*0.28; e.y+=ny/l*CELL*0.28;
      G.shake=Math.max(G.shake,1.6); spark(q.x,q.y,'#d9f7a1',5,110);
      if(G.time-G.lastRamSfx>0.09){SFX.ram();G.lastRamSfx=G.time;}
    }
  }

  /* towers */
  for(const t of G.towers){
    t.cd-=dt; t.reveal=Math.max(0,t.reveal-dt);
    const T=TYPES[t.type], R=T.range*CELL, R2=R*R;
    if(t.type===0){
      if(t.cd<=0){
        let tgt=null,bdc=1e18;
        for(const e of G.enemies){if(e.dead)continue;
          const dd=d2(e.x,e.y,t.x,t.y);
          if(dd<R2){const dc=d2(e.x,e.y,cx,cy); if(dc<bdc){bdc=dc;tgt=e;}}}
        if(tgt){
          t.cd=T.cd; const ang=Math.atan2(tgt.y-t.y,tgt.x-t.x); t.angle=ang;
          G.bolts.push({x:t.x+Math.cos(ang)*CELL*0.4,y:t.y+Math.sin(ang)*CELL*0.4,
            vx:Math.cos(ang)*560,vy:Math.sin(ang)*560,dmg:T.dmg,life:1.2});
          SFX.shoot();
        }
      }
    }else if(t.type===1){
      if(t.cd<=0){
        let any=false;
        for(const e of G.enemies)if(!e.dead&&d2(e.x,e.y,t.x,t.y)<R2){any=true;hurt(e,T.dmg);}
        if(any){t.cd=T.cd;G.rings.push({x:t.x,y:t.y,r:CELL*0.3,maxR:R,col:'63,214,192'});SFX.pulse();}
      }
    }else{
      if(t.cd<=0){
        let any=false;
        for(const e of G.enemies)if(!e.dead&&d2(e.x,e.y,t.x,t.y)<R2){any=true;e.slowT=1.2;}
        if(any){t.cd=T.cd;G.rings.push({x:t.x,y:t.y,r:CELL*0.3,maxR:R,col:'143,216,255'});SFX.frost();}
      }
    }
  }

  /* bolts */
  for(const b of G.bolts){
    b.life-=dt; b.x+=b.vx*dt; b.y+=b.vy*dt;
    if(b.life<=0||b.x<OX-30||b.x>OX+BW+30||b.y<OY-30||b.y>OY+BH+30){b.dead=true;continue;}
    for(const e of G.enemies){
      if(e.dead)continue;
      if(d2(b.x,b.y,e.x,e.y)<Math.pow(e.r+4,2)){hurt(e,b.dmg);b.dead=true;spark(b.x,b.y,'#ffe08a',5,100);break;}
    }
  }

  /* orbs */
  const hp=pts[0];
  for(const o of G.orbs){
    o.t+=dt;
    const dx=hp.x-o.x, dy=hp.y-o.y, dist=Math.hypot(dx,dy)||1;
    if(dist<CELL*2.6){o.vx+=dx/dist*1500*dt;o.vy+=dy/dist*1500*dt;}
    else{o.vx*=0.94;o.vy*=0.94;o.x+=Math.sin(o.t*3+o.ph)*8*dt;}
    const s2=o.vx*o.vx+o.vy*o.vy, mx=CELL*7;
    if(s2>mx*mx){const s=mx/Math.sqrt(s2);o.vx*=s;o.vy*=s;}
    o.x+=o.vx*dt; o.y+=o.vy*dt;
    if(dist<CELL*0.5){
      o.dead=true;
      if(o.kind==='en'){G.energy+=o.val;addFloat(hp.x,hp.y-CELL*0.5,'+'+o.val,'#ffd76a');SFX.eat();}
      else{
        if(sn.cells.length>=46){G.energy+=10;addFloat(hp.x,hp.y-CELL*0.5,'+10⚡','#8fd8ff');}
        else{sn.grow++;addFloat(hp.x,hp.y-CELL*0.5,'GROW!','#b7f34d');SFX.fruit();}
      }
      spark(o.x,o.y,o.kind==='en'?'#ffd76a':'#b7f34d',5,80);
    }
  }

  /* fx */
  for(const r of G.rings)r.r+=r.maxR*2.4*dt;
  G.rings=G.rings.filter(r=>r.r<r.maxR);
  for(const p of G.parts){p.t+=dt;p.x+=p.vx*dt;p.y+=p.vy*dt;p.vx*=0.92;p.vy*=0.92;}
  G.parts=G.parts.filter(p=>p.t<p.life);
  for(const f of G.floats)f.t+=dt;
  G.floats=G.floats.filter(f=>f.t<f.life);

  G.enemies=G.enemies.filter(e=>!e.dead);
  G.bolts=G.bolts.filter(b=>!b.dead);
  G.orbs=G.orbs.filter(o=>!o.dead);
}

function render(){
  ctx.clearRect(0,0,W,H);
  const bg=ctx.createLinearGradient(0,0,0,H);
  bg.addColorStop(0,'#0d1b14');bg.addColorStop(1,'#0a1510');
  ctx.fillStyle=bg;ctx.fillRect(0,0,W,H);

  ctx.save();
  if(G.shake>0)ctx.translate(rand(-G.shake,G.shake),rand(-G.shake,G.shake));

  /* ambient specks */
  for(const s of SPECKS){ctx.fillStyle='rgba(150,220,170,'+s.a+')';ctx.fillRect(s.x,s.y,s.r,s.r);}
  drawMargin();

  /* board */
  for(let y=0;y<ROWS;y++)for(let x=0;x<COLS;x++){
    ctx.fillStyle=((x+y)&1)?'#152a21':'#173026';

  }

function drawSnake(){
  const pts=G.pts||G.snake.cells.map(c=>({x:px(c.x),y:py(c.y)}));
  if(pts.length<2)return;
  ctx.lineJoin='round';ctx.lineCap='round';
  const path=()=>{ctx.beginPath();ctx.moveTo(pts[0].x,pts[0].y);for(let i=1;i<pts.length;i++)ctx.lineTo(pts[i].x,pts[i].y);};
  /* shadow */
  ctx.save();ctx.translate(0,3);path();ctx.strokeStyle='rgba(0,0,0,0.2)';ctx.lineWidth=CELL*0.76;ctx.stroke();ctx.restore();
  path();ctx.strokeStyle='#5d9c2a';ctx.lineWidth=CELL*0.74;ctx.stroke();
  path();ctx.strokeStyle='#a9e653';ctx.lineWidth=CELL*0.54;ctx.stroke();
  /* scale dots */
  ctx.fillStyle='rgba(80,140,40,0.55)';
  for(let i=2;i<pts.length-1;i+=2){ctx.beginPath();ctx.arc(pts[i].x,pts[i].y,CELL*0.11,0,TAU);ctx.fill();}
  /* head */
  const h=pts[0],dv=G.snake.dir;
  let look=Math.atan2(dv.y,dv.x);
  let ne=null,nd=CELL*3*CELL*3;
  for(const e of G.enemies){const dd=d2(e.x,e.y,h.x,h.y);if(dd<nd){nd=dd;ne=e;}}
  if(ne)look=Math.atan2(ne.y-h.y,ne.x-h.x);
  ctx.fillStyle='#bdf05a';ctx.strokeStyle='#4c8322';ctx.lineWidth=2.5;
  ctx.beginPath();ctx.arc(h.x,h.y,CELL*0.46,0,TAU);ctx.fill();ctx.stroke();
  if(G.snake.flash>0){ctx.fillStyle='rgba(255,80,70,'+G.snake.flash+')';ctx.beginPath();ctx.arc(h.x,h.y,CELL*0.46,0,TAU);ctx.fill();}
  /* eyes */
  const pxp=-dv.y,pyp=dv.x;
  for(const s of [-1,1]){
    ctx.fillStyle='#fff';ctx.beginPath();
    ctx.arc(h.x+dv.x*CELL*0.16+pxp*s*CELL*0.2,h.y+dv.y*CELL*0.16+pyp*s*CELL*0.2,CELL*0.13,0,TAU);ctx.fill();
    ctx.fillStyle='#141a10';ctx.beginPath();
    ctx.arc(h.x+dv.x*CELL*0.24+pxp*s*CELL*0.2+Math.cos(look)*1.8,h.y+dv.y*CELL*0.24+pyp*s*CELL*0.2+Math.sin(look)*1.8,CELL*0.065,0,TAU);ctx.fill();
  }
  /* tongue */
  if((G.time*1.7)%1>0.9){
    ctx.strokeStyle='#ff5d5d';ctx.lineWidth=2;
    const tx=h.x+dv.x*CELL*0.46,ty=h.y+dv.y*CELL*0.46;
    ctx.beginPath();ctx.moveTo(tx,ty);
    ctx.lineTo(tx+dv.x*CELL*0.3,ty+dv.y*CELL*0.3);
    ctx.moveTo(tx+dv.x*CELL*0.22,ty+dv.y*CELL*0.22);ctx.lineTo(tx+dv.x*CELL*0.34+pxp*3,ty+dv.y*CELL*0.34+pyp*3);
    ctx.moveTo(tx+dv.x*CELL*0.22,ty+dv.y*CELL*0.22);ctx.lineTo(tx+dv.x*CELL*0.34-pxp*3,ty+dv.y*CELL*0.34-pyp*3);
    ctx.stroke();
  }
}

window.addEventListener('keydown',e=>{
  audioInit();
  const c=e.code;
  if(['ArrowUp','ArrowDown','ArrowLeft','ArrowRight','Space'].includes(c))e.preventDefault();
  if(state==='menu'){if(c==='Enter'||c==='Space')startGame();return;}
  if(state==='over'){if(c==='KeyR'||c==='Enter')startGame();return;}
  if(c==='KeyM'){toggleMute();return;}
