function deadly(p,r){
  if(p.x<0||p.x>=COLS||p.y<0||p.y>=ROWS) return true;
  const o=other(r);

  for(let i=0;i<lim;i++){const c=r.body[i]; if(c.x===p.x&&c.y===p.y) return true;}
  return false;
}
function openness(p,r){ let n=0; for(const d of DIRS4) if(!deadly({x:p.x+d.x,y:p.y+d.y},r)) n++; return n; }
function aiChoose(r){
  const h=r.body[0]; let cands=[];
  for(const d of DIRS4){
    if(d.x===-r.dir.x&&d.y===-r.dir.y) continue;
    const nh={x:h.x+d.x,y:h.y+d.y};
    if(!deadly(nh,r)) cands.push({d,nh});
  }
  if(!cands.length) return DIRS4[(Math.random()*4)|0];
  if(Math.random()<r.err) return cands[(Math.random()*cands.length)|0].d;
  let tg=null,bd=1e9;
  for(const f of foods){ const dd=Math.abs(f.x-h.x)+Math.abs(f.y-h.y); if(dd<bd){bd=dd;tg=f;} }
  if(!tg) tg={x:COLS-1,y:h.y};
  let bestC=cands[0],bs=-1e9;
  for(const c of cands){
    const dd=Math.abs(tg.x-c.nh.x)+Math.abs(tg.y-c.nh.y);
    const s=-dd*2.1+(c.nh.x-h.x)*7+openness(c.nh,r)*1.8+Math.random()*.6;
    if(s>bs){bs=s;bestC=c;}
  }
  return bestC.d;
}

function stepRacer(r){
  if(r.state!=='alive') return;
  if(r.isPlayer && !attract){
    while(q.length){ const d=q.shift();
      if(!(d.x===-r.dir.x&&d.y===-r.dir.y)){ r.dir={x:d.x,y:d.y}; break; } }
  } else r.dir=aiChoose(r);

  const h=r.body[0], nh={x:h.x+r.dir.x, y:h.y+r.dir.y};
  if(nh.x<0||nh.x>=COLS||nh.y<0||nh.y>=ROWS){ crash(r); return; }
  const o=other(r);
  for(let i=0;i<o.body.length;i++){
    const c=o.body[i];
    if(c.x===nh.x&&c.y===nh.y){
      if(i===0 && o.state==='alive') break;   // head-to-head: crossing allowed
      crash(r); return;
    }
  }
  const lim=r.growth>0?r.body.length:r.body.length-1;
  for(let i=0;i<lim;i++){ const c=r.body[i]; if(c.x===nh.x&&c.y===nh.y){ crash(r); return; } }

  r.body.unshift(nh);
  if(r.growth>0) r.growth--; else r.body.pop();

  const fi=foods.findIndex(f=>f.x===nh.x&&f.y===nh.y);
  if(fi>=0){
    const f=foods[fi]; foods.splice(fi,1); spawnFood();
    r.growth+=2; r.eaten++; r.interval=baseInt(r); r.boost=650;
    burst(f.x,f.y,f.c,12,200); sfx.eat();
  }
  for(let i=0;i<4;i++) if(!r.cpFlags[i]&&nh.x>=CPX[i]) r.cpFlags[i]=true;
  r.lastCp=r.cpFlags.filter(Boolean).length;

  if(nh.x>=FINX){
    r.state='finished'; r.finishTime=raceTime;
    if(attract){ placeStart(r); return; }
    finishRace(r);
  }
}

function burst(x,y,color,n,spd){
  const px=(x+.5)*cell, py=(y+.5)*cell;
  for(let i=0;i<n;i++){

      s:cell*(.1+Math.random()*.2), c:Math.random()<.3?'#ffffff':color});
  }
}
function confettiBurst(){
  const W=COLS*cell;
  for(let i=0;i<90;i++) confetti.push({
    x:Math.random()*W, y:-20-Math.random()*ROWS*cell*.4,
    vx:(Math.random()-.5)*60, vy:60+Math.random()*140,
    rot:Math.random()*TAU, vr:(Math.random()-.5)*8,
    s:cell*(.2+Math.random()*.22), c:CONF[i%CONF.length], life:2400+Math.random()*1400
  });
}
function updateFx(dt){
  const k=dt/1000;
  for(let i=particles.length-1;i>=0;i--){ const p=particles[i];
    p.life-=dt; if(p.life<=0){particles.splice(i,1);continue;}
    p.x+=p.vx*k; p.y+=p.vy*k; p.vx*=(1-2.4*k); p.vy*=(1-2.4*k); }
  for(let i=confetti.length-1;i>=0;i--){ const c=confetti[i];
    c.life-=dt; if(c.life<=0||c.y>ROWS*cell+40){confetti.splice(i,1);continue;}
    c.x+=c.vx*k; c.y+=c.vy*k; c.vy+=520*k; c.rot+=c.vr*k; }
}

        }
      }
      if(r.boost>0) r.boost=Math.max(0,r.boost-dt);
    }
    if(state==='RACE') maybeEnd();
  }
  updateFx(dt);
  if(shake>0) shake=Math.max(0,shake-dt*.025);
}

/* ============================== RENDER ============================== */
function circle(x,y,r){ ctx.beginPath(); ctx.arc(x,y,r,0,TAU); ctx.fill(); }
function drawGate(x0,w,cA,cB){
  for(let cy=0;cy<ROWS;cy++) for(let cx=x0;cx<x0+w;cx++){
    ctx.fillStyle=((cx+cy)&1)?cA:cB; ctx.fillRect(cx*cell,cy*cell,cell,cell);
  }
}
function drawFace(r){
  const h=r.body[0], d=r.dir, e={x:-d.y,y:d.x};
  const cx=(h.x+.5)*cell+d.x*cell*.16, cy=(h.y+.5)*cell+d.y*cell*.16;
  for(const s of [1,-1]){
    const ox=cx+e.x*cell*.2*s, oy=cy+e.y*cell*.2*s;
    ctx.fillStyle='#fff'; circle(ox,oy,cell*.14);
    ctx.fillStyle='#0d1119'; circle(ox+d.x*cell*.05,oy+d.y*cell*.05,cell*.065);
  }
}
function drawSnake(r,now){
  const b=r.body; if(!b.length) return;
  const flick=(r.state==='crashed')?(Math.floor(now/80)%2?.35:.7):1;
  ctx.globalAlpha=flick;

  // boost speed-lines behind tail
  if(r.boost>0 && r.state!=='crashed'){
    const k=r.boost/650, t=b[b.length-1];
    ctx.strokeStyle=r.glow; ctx.globalAlpha=.45*k*flick; ctx.lineWidth=cell*.12;
    for(let i=0;i<3;i++){
      const bx=(t.x+.5)*cell-r.dir.x*cell*(1.2+i*.9), by=(t.y+.5)*cell-r.dir.y*cell*(1.2+i*.9);
      ctx.beginPath(); ctx.moveTo(bx,by);
      ctx.lineTo(bx-r.dir.x*cell*1.4,by-r.dir.y*cell*1.4); ctx.stroke();
    }
    ctx.globalAlpha=flick;
  }
  // outline pass
  ctx.fillStyle='#0a0e15';
  for(let i=b.length-1;i>=0;i--){
    const t=i/(b.length-1||1), rad=cell*(i===0?.52:lerp(.42,.18,t));
    circle((b[i].x+.5)*cell,(b[i].y+.5)*cell,rad+Math.max(1.5,cell*.07));
  }
  // body pass
  ctx.shadowColor=r.glow; ctx.shadowBlur=(r.boost>0?26:14);
  for(let i=b.length-1;i>=0;i--){
    const t=i/(b.length-1||1), rad=cell*(i===0?.5:lerp(.4,.17,t));
    ctx.fillStyle=(r.state==='crashed')?'#3a4356':mix(r.cHead,r.cTail,t);
    circle((b[i].x+.5)*cell,(b[i].y+.5)*cell,rad);
  }
  ctx.shadowBlur=0;
  // racing stripe
  if(r.state!=='crashed'){
    ctx.fillStyle='rgba(255,255,255,.9)';
    for(let i=2;i<b.length;i++){
      const t=i/(b.length-1||1), rad=cell*lerp(.4,.17,t)*.3;
      circle((b[i].x+.5)*cell,(b[i].y+.5)*cell,rad);
    }
    drawFace(r);
  }
  ctx.globalAlpha=1;
}
function render(now){
  const W=COLS*cell, H=ROWS*cell;
  ctx.setTransform(dpr,0,0,dpr,0,0);
  if(shake>0) ctx.translate((Math.random()-.5)*shake,(Math.random()-.5)*shake);

  const g=ctx.createLinearGradient(0,0,0,H);
  g.addColorStop(0,'#141b2b'); g.addColorStop(1,'#0c1120');
  ctx.fillStyle=g; ctx.fillRect(-20,-20,W+40,H+40);

  ctx.fillStyle='rgba(255,255,255,.02)';
  for(let y=0;y<ROWS;y++) for(let x=(y&1);x<COLS;x+=2) ctx.fillRect(x*cell,y*cell,cell,cell);

  ctx.save(); ctx.strokeStyle='rgba(255,255,255,.05)'; ctx.lineWidth=2;
  ctx.setLineDash([cell*1.2,cell*.9]);
  for(const ly of [4,11,17]){ ctx.beginPath(); ctx.moveTo(cell*3,(ly+.5)*cell); ctx.lineTo((COLS-3)*cell,(ly+.5)*cell); ctx.stroke(); }
  ctx.restore();

  drawGate(3,1,'rgba(233,238,245,.26)','rgba(20,26,38,.26)');   // start line
  drawGate(FINX,2,'#e9edf5','#161b26');                        // finish gate

  ctx.save(); ctx.font='700 '+Math.max(9,cell*.5)+'px Impact,"Arial Black",sans-serif';
  ctx.textAlign='center'; ctx.fillStyle='rgba(255,182,72,.85)';
  ctx.translate((COLS-1)*cell+cell*.4,H*.5); ctx.rotate(Math.PI/2); ctx.fillText('FINISH',0,0); ctx.restore();
  ctx.save(); ctx.font='700 '+Math.max(8,cell*.4)+'px Impact,"Arial Black",sans-serif';
  ctx.textAlign='center'; ctx.fillStyle='rgba(164,245,54,.35)';
  ctx.translate(cell*3+cell*.4,H*.5); ctx.rotate(Math.PI/2); ctx.fillText('START',0,0); ctx.restore();

  for(const f of foods){
    const cxp=(f.x+.5)*cell, cyp=(f.y+.5)*cell;
    const s=cell*.62*(1+.14*Math.sin(now/280+f.ph));
    ctx.save(); ctx.translate(cxp,cyp); ctx.rotate(now/900);
    ctx.shadowColor=f.c; ctx.shadowBlur=12; ctx.fillStyle=f.c;
    ctx.fillRect(-s/2,-s/2,s,s); ctx.shadowBlur=0;
    ctx.fillStyle='rgba(255,255,255,.85)'; const s2=s*.36; ctx.fillRect(-s2/2,-s2/2,s2,s2);
    ctx.restore();
  }

  drawSnake(racers[1],now);
  drawSnake(racers[0],now);

  for(const p of particles){
    ctx.globalAlpha=Math.max(0,p.life/p.max); ctx.fillStyle=p.c;
    ctx.fillRect(p.x-p.s/2,p.y-p.s/2,p.s,p.s);
  }
  ctx.globalAlpha=1;
  for(const c of confetti){
    ctx.save(); ctx.globalAlpha=Math.min(1,c.life/400);
    ctx.translate(c.x,c.y); ctx.rotate(c.rot); ctx.fillStyle=c.c;
    ctx.fillRect(-c.s/2,-c.s/2,c.s,c.s*.7); ctx.restore();
  }
}

}
