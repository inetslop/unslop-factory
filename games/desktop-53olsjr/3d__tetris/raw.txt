/* ================= 3D PROJECTION ================= */
function prepCam(c){
 c.cyw=Math.cos(c.yaw);c.syw=Math.sin(c.yaw);
 c.cxp=Math.cos(c.pitch);c.sxp=Math.sin(c.pitch);
}
function proj(c,x,y,z){
 const X=x*c.cyw+z*c.syw, Z1=-x*c.syw+z*c.cyw;
 const Y=y*c.cxp-Z1*c.sxp, Z2=y*c.sxp+Z1*c.cxp;
 const k=c.D/(c.D-Z2), s=c.s*k;
 return{x:c.cx+X*s,y:c.cy-Y*s,z:Z2};
}
function rotN(n,c){
 const X=n[0]*c.cyw+n[2]*c.syw, Z1=-n[0]*c.syw+n[2]*c.cyw;
 const Y=n[1]*c.cxp-Z1*c.sxp, Z2=n[1]*c.sxp+Z1*c.cxp;
 return[X,Y,Z2];
}

function drawCube(c,cb){
 const h=.48*cb.scl; if(h<=.015)return;
 const P=[];
 for(let i=0;i<8;i++){
  const sx=i&1?1:-1,sy=i&2?1:-1,sz=i&4?1:-1;
  P.push(proj(c,cb.wx+sx*h,cb.wy+sy*h,(cb.wz||0)+sz*h));
 }
 const vis=[];
 for(const f of FACES){
  const nr=rotN(f.n,c); if(nr[2]<=.04)continue;
  let zc=0;for(const v of f.v)zc+=P[v].z; zc/=4;
  let diff=Math.max(0,nr[0]*LIGHT[0]+nr[1]*LIGHT[1]+nr[2]*LIGHT[2]);
  let col=mixc(cb.c,[255,255,255],.3+.75*diff);
  if(cb.fl)col=mixc(col,[255,255,255],cb.fl*.85);
  vis.push({f,zc,col,nr});
 }
 vis.sort((a,b)=>a.zc-b.zc);
 ctx.lineWidth=1;
 for(const v of vis){
  const idx=v.f.v;
  ctx.beginPath();ctx.moveTo(P[idx[0]].x,P[idx[0]].y);
  for(let i=1;i<4;i++)ctx.lineTo(P[idx[i]].x,P[idx[i]].y);
  ctx.closePath();
  if(cb.ghost){
   ctx.fillStyle='rgba('+cb.c[0]+','+cb.c[1]+','+cb.c[2]+',.13)';ctx.fill();
   ctx.strokeStyle='rgba('+cb.c[0]+','+cb.c[1]+','+cb.c[2]+',.5)';ctx.stroke();
  }else{
   ctx.globalAlpha=cb.a;
   ctx.fillStyle='rgb('+(v.col[0]|0)+','+(v.col[1]|0)+','+(v.col[2]|0)+')';
   ctx.fill();
   ctx.strokeStyle='rgba(5,8,14,.5)';ctx.stroke();
   if(v.nr[1]>.7){ /* top bevel highlight */
    let mx=0,my=0;for(const i of idx){mx+=P[i].x;my+=P[i].y;}mx/=4;my/=4;
    ctx.beginPath();
    for(let i=0;i<4;i++){const q=P[idx[i]],px=q.x+(mx-q.x)*.3,py=q.y+(my-q.y)*.3;i?ctx.lineTo(px,py):ctx.moveTo(px,py);}
    ctx.closePath();ctx.fillStyle='rgba(255,255,255,.16)';ctx.fill();
   }
  }
 }
 ctx.globalAlpha=1;
}

/* ================= GAME LOGIC ================= */
function refill(){bag='IJLOSTZ'.split('').sort(()=>Math.random()-.5);}
function qNext(){if(!bag.length)refill();return bag.pop();}
function fillQueue(){while(queue.length<3)queue.push(qNext());}
function collides(type,rot,x,y){
 const cs=SHAPES[type].cells[rot];
 for(let i=0;i<4;i++){
  const gx=x+cs[i][0],gy=y+cs[i][1];
  if(gx<0||gx>=COLS||gy>=ROWS)return true;
  if(gy>=0&&grid[gy][gx])return true;
 }
 return false;
}
function baseIv(){return Math.max(70,1000*Math.pow(.8,level-1));}
function spawn(type){
 cur={type,rot:0,x:(COLS-SHAPES[type].size>>1),y:0};
 gravT=0;lockT=0;grounded=false;resets=0;
 if(collides(type,0,cur.x,cur.y))gameOver();
}
function startNext(){const t=queue.shift();fillQueue();spawn(t);}
function nudge(){if(grounded&&resets<MAXRST){lockT=0;resets++;}}
function tryMove(dx){
 if(state!=='playing'||clearing||!cur)return false;
 if(!collides(cur.type,cur.rot,cur.x+dx,cur.y)){cur.x+=dx;SFX.move();nudge();return true;}
 return false;
}
function tryRot(dir){
 if(state!=='playing'||clearing||!cur)return false;
 if(cur.type==='O')return false;
 const nr=(cur.rot+dir+4)%4;
 const tbl=(cur.type==='I'?KICK_I:KICK_N)[cur.rot+''+nr];
 for(let i=0;i<5;i++){
  const nx=cur.x+tbl[i][0],ny=cur.y+tbl[i][1];
  if(!collides(cur.type,nr,nx,ny)){cur.rot=nr;cur.x=nx;cur.y=ny;SFX.rot();nudge();return true;}
 }
 return false;
}
function stepDown(){
 if(collides(cur.type,cur.rot,cur.x,cur.y+1)){grounded=true;return false;}
 cur.y++;grounded=false;resets=0;lockT=0;return true;
}
function ghostY(){let gy=cur.y;while(!collides(cur.type,cur.rot,cur.x,gy+1))gy++;return gy;}
function hardDrop(){
 if(state!=='playing'||clearing||!cur)return;
 let d=0;while(!collides(cur.type,cur.rot,cur.x,cur.y+1)){cur.y++;d++;}
 if(d>0){score+=2*d;hud();}
 shake=Math.min(9,2.5+d*.4);SFX.drop();
 for(const cc of SHAPES[cur.type].cells[cur.rot]){
  const gx=cur.x+cc[0],gy=cur.y+cc[1];
  if(gy+1>=ROWS||grid[gy+1][gx])spawnBurst(gx-4.5,9.5-gy,COLORS[cur.type],3,120);
 }
 lockPiece();
}
function doHold(){
 if(state!=='playing'||clearing||!cur||holdUsed)return;
 SFX.hold();const t=cur.type;
 if(hold){spawn(hold);}else{hold=t;startNext();}
 holdUsed=true;
}
function lockPiece(){
 if(!cur)return;
 const {type,rot,x,y}=cur,cells=SHAPES[type].cells[rot];
 let topOut=false;
 for(const c of cells){
  const gx=x+c[0],gy=y+c[1];
  if(gy<0){topOut=true;continue;}
  grid[gy][gx]=type;
 }
 holdUsed=false;
 if(topOut){gameOver();return;}
 const full=[];
 for(let r=0;r<ROWS;r++){let f=true;for(let c2=0;c2<COLS;c2++)if(!grid[r][c2]){f=false;break;}if(f)full.push(r);}
 SFX.lock();
 if(full.length)doClear(full);
 else{combo=0;updComboBadge();startNext();}
}
function doClear(rows){
 const n=rows.length;combo++;
 let base=[0,100,300,500,800][n]*level,label=['','SINGLE','DOUBLE','TRIPLE','TETRIS!'][n];
 if(n===4&&b2b){base=Math.round(base*1.5);label='B2B TETRIS!';}
 score+=base;
 const midY=9.5-rows.reduce((a,b)=>a+b,0)/rows.length;
 popups.push({wx:0,wy:midY,t:0,life:1.3,text:label,color:n===4?'#ffd23f':'#7fe3ff',size:n===4?26:18});
 if(combo>=2){
  const b=50*(combo-1)*level;score+=b;
  popups.push({wx:0,wy:midY-1.5,t:0,life:1.4,text:'COMBO ×'+(combo-1),color:'#ffb638',size:15});
 }
 b2b=(n===4);lines+=n;hud();updComboBadge();
 clearing={rows:new Set(rows),t:0};
 SFX.clear(n);flash=Math.min(.5,n*.12);shake=Math.max(shake,n===4?9:4);
 const nl=Math.floor(lines/10)+1;
 if(nl>level){
  level=nl;SFX.level();hud();
  popups.push({wx:0,wy:midY+1.6,t:0,life:1.5,text:'LEVEL '+level,color:'#3fe07a',size:20});
 }
}
function finishClear(){
 const rows=clearing.rows;
 for(const r of rows)for(let c2=0;c2<COLS;c2++)spawnBurst(c2-4.5,9.5-r,COLORS[grid[r][c2]],6,180);
 const keep=[];
 for(let r=ROWS-1;r>=0;r--)if(!rows.has(r))keep.push(grid[r]);
 while(keep.length<ROWS)keep.unshift(new Array(COLS).fill(0));
 grid=keep;clearing=null;startNext();
}

 else elNewBest.classList.remove('show');
 elFinal.textContent=fmt(score);elFinalBest.textContent=fmt(best);hud();showScreen('over');
}
function startGame(){
 grid=Array.from({length:ROWS},()=>new Array(COLS).fill(0));

 bag=[];queue=[];fillQueue();hold=null;holdUsed=false;
 popups.length=0;parts.length=0;shake=0;flash=0;
 state='playing';startNext();hud();updComboBadge();showScreen(null);
}

});

function update(dt){
 const ty=-.32+mouse.x*.42,tp=.26-mouse.y*.3;
 mouse.x+=(mouse.tx-mouse.x)*Math.min(1,dt*6);
 mouse.y+=(mouse.ty-mouse.y)*Math.min(1,dt*6);
 cam.yaw+=(ty-cam.yaw)*Math.min(1,dt*5);
 cam.pitch+=(tp-cam.pitch)*Math.min(1,dt*5);
 shake=Math.max(0,shake-shake*Math.min(1,dt*7)-dt*2);
 flash=Math.max(0,flash-dt*1.6);
 for(let i=popups.length-1;i>=0;i--){const p=popups[i];p.t+=dt;if(p.t>p.life)popups.splice(i,1);}
 for(let i=parts.length-1;i>=0;i--){const p=parts[i];p.t+=dt;p.x+=p.vx*dt;p.y+=p.vy*dt;p.vy+=p.g*dt;if(p.t>p.life)parts.splice(i,1);}
 if(state!=='playing')return;
 const ms=dt*1000;
 if(moveDir){
  dasT+=ms;
  if(dasT>=DAS){arrT+=ms;while(arrT>=ARR){arrT-=ARR;tryMove(moveDir);}}
 }
 if(clearing){
  clearing.t+=ms;
  if(clearing.t>=CLEARMS)finishClear();
 }else if(cur){
  const iv=keys.down?Math.min(SOFT,baseIv()):baseIv();
  gravT+=ms;
  while(gravT>=iv){
   gravT-=iv;
   if(stepDown()){if(keys.down){score++;hud();}}
   else break;
  }
  if(grounded){lockT+=ms;if(lockT>=LOCK)lockPiece();}
 }
}
