function solvePath(m,sx,sy,ex,ey){

    for(let d=0;d<4;d++){

      const nx=x+DIRS[d].dx,ny=y+DIRS[d].dy,k=idx(nx,ny);
      if(seen.has(k))continue;seen.add(k);parent.set(k,idx(x,y));st.push([nx,ny]);pushed=true;break;
    }

  while(cur!==undefined){path.unshift([m.cells[cur].x,m.cells[cur].y]);cur=parent.get(cur);}

}

    for(let i=0;i<6;i++)push({x:sx+rand(-6,6),y:sy+rand(-6,6),vx:rand(-18,18),vy:rand(-30,-8),g:-20,life:rand(.4,.8),t:0,size:rand(4,9),col:'rgba(150,162,176,',smoke:true});
  }else if(kind==='win'){

  }
}

function loop(t){
  requestAnimationFrame(loop);
  const dt=Math.min(.05,(t-lastT)/1000||0);lastT=t;tNow+=dt;
  for(const mo of motes){mo.x+=mo.vx*dt;mo.y+=mo.vy*dt;
    if(mo.x<-4)mo.x=L.w+4;if(mo.x>L.w+4)mo.x=-4;if(mo.y<-4)mo.y=L.h+4;if(mo.y>L.h+4)mo.y=-4;}
  if(G){
    if(G.state==='playing')G.time+=dt;
    G.shake=Math.max(0,G.shake-dt*34);G.flash=Math.max(0,G.flash-dt*2.2);G.nudge=Math.max(0,G.nudge-dt*4.5);
    for(const f of G.fx)f.t+=dt;G.fx=G.fx.filter(f=>f.t<.45);

      G.endTimer-=dt;
      if(G.endTimer<=0){if(!G.won)SFX.lose();showEnd();}}
  }else demoUpdate(dt);
  updateParts(dt);
  if(G){const s=Math.floor(G.time);if(s!==lastSec){lastSec=s;$('#timeVal').textContent=fmt(s);}}
