  }catch(e){}
}
function noiseBurst(t,vol,fq){
  if(!AC||muted||!noiseBuf)return;
  try{
    const s=AC.createBufferSource();s.buffer=noiseBuf;
    const f=AC.createBiquadFilter();f.type='bandpass';f.frequency.value=fq;f.Q.value=.8;
    const g=AC.createGain(),t0=AC.currentTime;
    g.gain.setValueAtTime(vol,t0);g.gain.exponentialRampToValueAtTime(0.001,t0+t);
    s.connect(f);f.connect(g);g.connect(master);s.start(t0);s.stop(t0+t+0.05);
  }catch(e){}
}

function addBoxSolid(cx,cy,cz,sx,sy,sz,col,em){boxes.push({cx,cy,cz,sx,sy,sz,col,em:em||0,solid:true});}

function spawnPattern(x){
  const r=rnd();
  if(r<.24){ // ember crystal spikes — jump over (or steer around)
    const w=3+rnd()*3, z0=-5.4+rnd()*(10.8-w), h=1.45+rnd()*.3;
    for(let z=z0;z<=z0+w;z+=1.05){
      addBoxSolid(x,(h)/2,z,.8,h,.8,[196,74,58],.12);
      addDecoBox(x,h+.22,z,.42,.5,.42,[255,120,70]);
    }
    if(rnd()<.45){const hz=h+1;for(let i=0;i<5;i++)orbs.push({x:x+i*1.5-3,y:hz+.6-Math.abs(i-2)*.28,z:z0+w/2,ph:rnd()*7});}
    run.nextSpawn=x+gap(6);
  } else if(r<.40){ // full-width light bar — slide under
    addBoxSolid(x,1.85,0,.7,1.3,12.6,[255,120,60],.55);
    addDecoBox(x,2.3,-6.35,.9,4.6,.9,[120,70,64]);
    addDecoBox(x,2.3, 6.35,.9,4.6,.9,[120,70,64]);
    orbs.push({x:x-1.6,y:.6,z:0,ph:rnd()*7});orbs.push({x:x+1.6,y:.6,z:0,ph:rnd()*7});
    run.nextSpawn=x+gap(5);
  } else if(r<.60){ // canyon wall with a gap — steer through
    const gz=(rnd()*2-1)*4.2, gw=3.1;
    addBoxSolid(x,2.3,(-6.5+(gz-gw/2))/2, .8,4.6,gz-gw/2+6.5,[172,120,86],0);
    addBoxSolid(x,2.3,(gz+gw/2+6.5)/2,  .8,4.6,6.5-(gz+gw/2),[172,120,86],0);
    if(rnd()<.4)for(let i=0;i<4;i++)orbs.push({x:x+i*1.3-1.9,y:.7,z:gz,ph:rnd()*7});
    run.nextSpawn=x+gap(7);
  } else if(r<.74){ // low rock — hop it
    const zc=(rnd()*2-1)*4.5;
    addBoxSolid(x,.62,zc,2.2,1.25,2.2,[120,84,74],0);
    addDecoBox(x,1.42,zc-.4,.9,.35,.9,[150,104,88]);
    run.nextSpawn=x+gap(6);
  } else if(r<.86){ // hanging flier — slide or jump
    const zc=(rnd()*2-1)*4.2;
    addBoxSolid(x,1.6,zc,3.4,1.3,2.6,[128,92,110],.05);
    addDecoBox(x,.72,zc,1.1,.45,1.1,[160,120,138]);
    run.nextSpawn=x+gap(6);
  } else { // ember gate ring — fly through for bonus
    const gz=(rnd()*2-1)*4;
    gates.push({x,z:gz,yc:1.35,r:1.35,got:false});
    for(let i=0;i<3;i++)orbs.push({x:x+i*1.2-.6,y:.8,z:gz,ph:rnd()*7});
    run.nextSpawn=x+gap(4);
  }
}
function gap(extra){return 12+run.speed*.5+extra;}

  }
  if(rnd()<.6){
    const s=rnd()<.5?-1:1, z=s*(12+rnd()*16);
    if(rnd()<.22)addDecoBox(x,(4+rnd()*4)/2,z,.9,4+rnd()*4,1.4,[88,60,70]);
    else addDecoBox(x,(.8+rnd()*2.4)/2,z,1+rnd()*3.5,.8+rnd()*2.4,1+rnd()*3,[rnd()<.5?96:112,66,72]);
  }
}

/* ============================== UPDATE ============================== */
function update(dt){
  tGlobal+=dt;
  zS+=(player.z-zS)*Math.min(1,dt*10);
  shake*=Math.exp(-4.5*dt);camDip*=Math.exp(-5*dt);

  if(state==='playing'){
    run.t+=dt;
    const target=Math.min(MAXS,14+run.t*.45);
    run.speed+=(target-run.speed)*Math.min(1,dt*2);
    player.x+=run.speed*dt;
    run.dist+=run.speed*dt/3;

    // steering
    let az=0;
    if(keys.left)az-=1; if(keys.right)az+=1;
    player.vz+=az*42*dt;
    if(!az)player.vz*=Math.exp(-8*dt);
    player.vz=clamp(player.vz,-13,13);
    player.z+=player.vz*dt;
    if(player.z<-5.3){player.z=-5.3;player.vz=0;}
    if(player.z> 5.3){player.z= 5.3;player.vz=0;}

    // jumping
    if(wantJumpAt>0&&run.t-wantJumpAt<.12){
      wantJumpAt=-9;
      if(player.grounded||player.coyote>0){player.vy=11.5;player.grounded=false;player.coyote=0;player.airJumps=1;sfx.jump();}
      else if(player.airJumps>0){player.vy=10.8;player.airJumps--;sfx.djump();}
    }
    const g=player.down?84:32;
    player.vy-=g*dt;player.y+=player.vy*dt;
    if(player.y<=REST){
      if(!player.grounded&&player.vy<-6){sfx.land();camDip=.14;dustBurst();}
      player.y=REST;player.vy=0;player.grounded=true;player.airJumps=1;player.coyote=.09;
    }else{player.grounded=false;player.coyote-=dt;}
    player.down=!!keys.down&&true;

    if(player.invuln>0)player.invuln-=dt;

    // spawning
    while(player.x+86>run.nextSpawn)spawnPattern(run.nextSpawn);
    while(player.x+95>decoX){addDeco(decoX);decoX+=12+rnd()*14;}

    // collisions
    if(player.invuln<=0){
      const head=player.down?.85:1.5, hx=.30, hz=.27;
      for(const b of boxes){
        if(!b.solid||Math.abs(b.cx-player.x)>3)continue;
        if(Math.abs(player.x-b.cx)<b.sx/2+hx && Math.abs(player.z-b.cz)<b.sz/2+hz
           && player.y+head>b.cy-b.sy/2+.06 && player.y+.08<b.cy+b.sy/2-.06){doHit();break;}
      }
    }
    // orbs
    for(let i=orbs.length-1;i>=0;i--){const o=orbs[i];
      if(Math.abs(o.x-player.x)>1.5||Math.abs(o.z-player.z)>1.7)continue;
      const dx=o.x-player.x,dy=o.y-(player.y+.7),dz=o.z-player.z;
      if(dx*dx+dy*dy+dz*dz<1){orbs.splice(i,1);run.orbs++;sfx.collect(run.orbs);orbBurst(o);}
    }
    // gates
    for(const gt of gates){
      if(gt.got||Math.abs(gt.x-player.x)>.8)continue;
      const dz=gt.z-player.z,dy=gt.yc-(player.y+.7);
      if(dz*dz+dy*dy<1.32){gt.got=true;run.gates++;sfx.gate();gatePulse(gt);
        floats.push({x:gt.x,y:3,z:gt.z,txt:'+40',life:.9,max:.9,sz:.7,col:'#ffd9a0'});}
    }
    run.score=run.dist+run.orbs*25+run.gates*40;
    if(run.score>run.best&&!newBestShown){newBestShown=true;hNew.classList.remove('hidden');}

  } else if(state==='menu'){
    run.speed=7.5;player.x+=run.speed*dt;
    player.y=REST+.1*Math.sin(tGlobal*2);
    player.vz*=Math.exp(-6*dt);player.z=clamp(player.z+player.vz*dt,-5,5);
    while(player.x+86>run.nextSpawn)spawnPattern(run.nextSpawn);
    while(player.x+95>decoX){addDeco(decoX);decoX+=12+rnd()*14;}
  } else if(state==='over'){
    run.speed+=(3-run.speed)*Math.min(1,dt*1.5);
    player.x+=run.speed*dt;
  }

  // cleanup
  const cut=cam.x-12;
  boxes=boxes.filter(b=>b.cx>cut);
  orbs=orbs.filter(o=>o.x>cut);
  gates=gates.filter(g=>g.x>cut-4);

  // thruster trail
  if(state==='playing'||state==='menu'){
    emitAcc+=dt*(6+run.speed*.35);
    while(emitAcc>1){emitAcc--;
      parts.push({x:player.x-.55,y:player.y+.2+rnd()*.2,z:player.z+(rnd()-.5)*.3,
        vx:(rnd()-.5)*2,vy:.3+rnd()*.4,vz:(rnd()-.5)*2,life:.5,max:.5,sz:.09,col:rnd()<.5?[255,170,90]:[120,220,210]});
    }
  }
  // particles / floats / motes
  for(let i=parts.length-1;i>=0;i--){const p=parts[i];p.life-=dt;
    if(p.life<=0){parts.splice(i,1);continue;}
    p.x+=p.vx*dt;p.y+=p.vy*dt;p.z+=p.vz*dt;}
  for(let i=floats.length-1;i>=0;i--){const f=floats[i];f.life-=dt;f.y+=.6*dt;if(f.life<=0)floats.splice(i,1);}
  for(const m of motes){if(m.x<cam.x)m.x+=72;}

  // wind
  if(AC&&windFilter){
    const tgt=state==='playing'?240+run.speed*6:180;
    windFilter.frequency.value=lerp(windFilter.frequency.value,tgt,dt*3);
    windGain.gain.setTargetAtTime(muted?0:(state==='playing'?.05:.028),AC.currentTime,.3);
  }
}

const FACES=[{n:[0,0,1],i:[4,5,6,7]},{n:[0,0,-1],i:[1,0,3,2]},{n:[1,0,0],i:[5,1,2,6]},
             {n:[-1,0,0],i:[0,4,7,3]},{n:[0,1,0],i:[2,3,7,6]},{n:[0,-1,0],i:[0,1,5,4]}];

function addBox(cx,cy,cz,sx,sy,sz,col,ry,rx,opt){
  ry=ry||0;rx=rx||0;
  const cY=Math.cos(ry),sY=Math.sin(ry),cX=Math.cos(rx),sX=Math.sin(rx);
  const hx=sx/2,hy=sy/2,hz=sz/2;
  const loc=[[-hx,-hy,-hz],[hx,-hy,-hz],[hx,hy,-hz],[-hx,hy,-hz],[-hx,-hy,hz],[hx,-hy,hz],[hx,hy,hz],[-hx,hy,hz]];
  const pts=loc.map(p=>{let x=p[0],y=p[1],z=p[2];
    const x1=x*cY+z*sY,z1=-x*sY+z*cY;
    return [cx+x1,cy+y*cX-z1*sX,cz+y*sX+z1*cX];});
  const fn=FACES.map(f=>{let x=f.n[0],y=f.n[1],z=f.n[2];
    const x1=x*cY+z*sY,z1=-x*sY+z*cY;
    return {n:[x1,y*cX-z1*sX,y*sX+z1*cX],i:f.i};});
  let cxm=0,cym=0,czm=0;for(const p of pts){cxm+=p[0];cym+=p[1];czm+=p[2];}
  const c=[cxm/8,cym/8,czm/8];
  const zc=(c[0]-cam.x)*fwd[0]+(c[1]-cam.y)*fwd[1]+(c[2]-cam.z)*fwd[2];
  items.push({type:'box',pts,fn,col,zc,a:opt&&opt.a!==undefined?opt.a:1});
}
function addQuad(p1,p2,p3,p4,fill){

  items.push({type:'q',pts:[p1,p2,p3,p4],zc:(p1[2]+p2[2]+p3[2]+p4[2])/4,fill});
}

    const z=s*8.7, seed=s>0?31:57;
    for(let x0=x0s;x0<cam.x+95;x0+=3){
      const t=clamp((x0-cam.x)/95,0,1);
      const hL=4+vnoise(x0*.11+seed)*7.5, hR=4+vnoise((x0+3)*.11+seed)*7.5;
      addQuad(P(x0,0,z),P(x0+3,0,z),P(x0+3,hR,z),P(x0,hL,z),mix([62,45,56],HAZE,t*.95));
      const q=addQuad;
      const a1=P(x0,hL-.35,z),a2=P(x0+3,hR-.35,z),a3=P(x0+3,hR,z),a4=P(x0,hL,z);
      if(a1&&a2&&a3&&a4)items.push({type:'q',pts:[a1,a2,a3,a4],zc:(a1[2]+a2[2]+a3[2]+a4[2])/4+.05,fill:mix([216,128,84],HAZE,t*.9)});
    }
  }
  for(const b of boxes)addBox(b.cx,b.cy,b.cz,b.sx,b.sy,b.sz,b.col,0,0,{a:1});

  // player (hover-runner) — occluded correctly by the sort
  const blink=player.invuln>0&&Math.floor(tGlobal*16)%2===0;
  const syB=player.down?.5:.8, rxp=clamp(player.vy*.05,-.4,.75), ryp=clamp(-player.vz*.06,-.5,.5);
  const yb=player.y+syB*.55;
  const zg=P(px,.02,player.z);
  if(zg)items.push({type:'glow',p:zg,r:.9,zc:zg[2]+.3,col:'rgba(255,170,80,'+(blink?.12:.3)+')'});
  if(!blink){
    addBox(px,yb,player.z,.95,syB,.62,[238,224,205],ryp,rxp);          // body
    addBox(px+.42,yb,player.z,.34,syB*.8,.66,[255,142,64],ryp,rxp);     // ember nose
    addBox(px-.3,yb+syB*.28,player.z,.26,.24,.5,[36,92,96],ryp,rxp);    // rear glass
    addBox(px-.5,yb+syB*.18,player.z,.14,syB*1.2,.36,[230,120,70],ryp,rxp); // fin
  }

  for(const o of orbs)items.push({type:'orb',o,zc:0});
  for(const gt of gates)items.push({type:'gate',gt,zc:0});

  // resolve depths
  for(const it of items){
    if(it.type==='box')continue;
    if(it.type==='q')continue;
    if(it.type==='glow')continue;
    const p=P(it.o?it.o.x:(it.gt.x), it.o?it.o.y:it.gt.yc, it.o?it.o.z:it.gt.z);
    it.p=p; it.zc=p?p[2]:-1;
  }
  items.sort((a,b)=>b.zc-a.zc);

  /* --- draw list --- */
  for(const it of items){
    if(it.type==='q'){
      ctx.fillStyle=it.fill;ctx.beginPath();
      ctx.moveTo(it.pts[0][0],it.pts[0][1]);
      for(let i=1;i<4;i++)ctx.lineTo(it.pts[i][0],it.pts[i][1]);
      ctx.closePath();ctx.fill();
    }
    else if(it.type==='box'){
      const c=[(it.pts[0][0]+it.pts[2][0])/2,(it.pts[0][1]+it.pts[2][1])/2,(it.pts[0][2]+it.pts[2][2])/2];
      if(it.zc<NEAR)continue;
      const t=clamp((it.zc-16)/95,0,.88);
      ctx.globalAlpha=it.a;
      for(const f of it.fn){
        let vx=0,vy=0,vz=0;for(const idx of f.i){vx+=it.pts[idx][0];vy+=it.pts[idx][1];vz+=it.pts[idx][2];}
        vx/=4;vy/=4;vz/=4;
        if((f.n[0]*(cam.x-vx)+f.n[1]*(cam.y-vy)+f.n[2]*(cam.z-vz))<=0)continue;
        const pp=[];let ok=true;
        for(const idx of f.i){const q=P(it.pts[idx][0],it.pts[idx][1],it.pts[idx][2]);if(!q){ok=false;break;}pp.push(q);}
        if(!ok)continue;
        const shade=.42+.62*Math.max(0,f.n[0]*LIGHT[0]+f.n[1]*LIGHT[1]+f.n[2]*LIGHT[2]);
        ctx.fillStyle=mix([it.col[0]*shade,it.col[1]*shade,it.col[2]*shade],HAZE,t);
        ctx.beginPath();ctx.moveTo(pp[0][0],pp[0][1]);
        for(let i=1;i<4;i++)ctx.lineTo(pp[i][0],pp[i][1]);
        ctx.closePath();ctx.fill();
      }
      ctx.globalAlpha=1;
    }
    else if(it.type==='glow'){
      const rpx=it.r*FOC/it.p[2];
      ctx.save();ctx.globalCompositeOperation='lighter';
      const rg=ctx.createRadialGradient(it.p[0],it.p[1],0,it.p[0],it.p[1],rpx);
      rg.addColorStop(0,it.col);rg.addColorStop(1,'rgba(255,170,80,0)');
      ctx.fillStyle=rg;ctx.beginPath();ctx.arc(it.p[0],it.p[1],rpx,0,7);ctx.fill();ctx.restore();
    }
    else if(it.type==='orb'&&it.p){
      const o=it.o,pulse=1+.15*Math.sin(tGlobal*5+o.ph), rpx=.3*pulse*FOC/it.zc;
      ctx.save();ctx.globalCompositeOperation='lighter';
      const rg=ctx.createRadialGradient(it.p[0],it.p[1],0,it.p[0],it.p[1],rpx*2.4);
      rg.addColorStop(0,'rgba(255,190,110,.85)');rg.addColorStop(1,'rgba(255,170,80,0)');
      ctx.fillStyle=rg;ctx.beginPath();ctx.arc(it.p[0],it.p[1],rpx*2.4,0,7);ctx.fill();
      ctx.fillStyle='#fff3d6';ctx.beginPath();ctx.arc(it.p[0],it.p[1],Math.max(1,rpx*.55),0,7);ctx.fill();
      ctx.restore();
    }
    else if(it.type==='gate'&&it.p){
      const gt=it.gt,pts=[];let ok=true;
      for(let i=0;i<16;i++){const a=i/16*6.283;
        const q=P(gt.x,gt.yc+Math.cos(a)*gt.r,gt.z+Math.sin(a)*gt.r);
        if(!q){ok=false;break;}pts.push(q);}
      if(!ok)continue;
      ctx.lineJoin='round';ctx.beginPath();ctx.moveTo(pts[0][0],pts[0][1]);
      for(let i=1;i<16;i++)ctx.lineTo(pts[i][0],pts[i][1]);
      ctx.closePath();
      const w1=.22*FOC/it.zc;
      ctx.strokeStyle='rgba(255,140,70,.3)';ctx.lineWidth=w1*2.8;ctx.stroke();
      ctx.strokeStyle=gt.got?'rgba(160,220,205,.9)':'#ffb35c';ctx.lineWidth=Math.max(1.5,w1*.9);ctx.stroke();
      for(let k=0;k<3;k++){const a=tGlobal*2+k*2.09;
        const q=P(gt.x,gt.yc+Math.cos(a)*gt.r,gt.z+Math.sin(a)*gt.r);
        if(q){ctx.fillStyle='#ffe9c4';ctx.beginPath();ctx.arc(q[0],q[1],Math.max(1.5,w1*.35),0,7);ctx.fill();}}
    }
  }

  /* --- additive particles & motes --- */
  ctx.save();ctx.globalCompositeOperation='lighter';
  for(const p of parts){
    const q=P(p.x,p.y,p.z);if(!q)continue;
    const a=p.life/p.max, r=Math.max(.5,p.sz*FOC/q[2]);
    ctx.fillStyle='rgba('+p.col[0]+','+p.col[1]+','+p.col[2]+','+(a*.8).toFixed(2)+')';
    ctx.beginPath();ctx.arc(q[0],q[1],r,0,7);ctx.fill();
  }
  for(const m of motes){
    const q=P(m.x,m.y,m.z);if(!q)continue;
    ctx.fillStyle='rgba(255,220,180,.1)';
    ctx.beginPath();ctx.arc(q[0],q[1],Math.max(.6,.05*FOC/q[2]),0,7);ctx.fill();
  }
  ctx.restore();

  /* --- floating texts --- */
  for(const f of floats){
    const q=P(f.x,f.y,f.z);if(!q)continue;
    const a=f.life/f.max, pxs=clamp(f.sz*90*(FOC/q[2])*(H/900),12,84);
    ctx.globalAlpha=a;ctx.font='400 '+pxs+'px Impact,"Arial Black",sans-serif';
    ctx.textAlign='center';ctx.lineWidth=pxs*.14;ctx.strokeStyle='rgba(30,10,10,.65)';
    ctx.strokeText(f.txt,q[0],q[1]);ctx.fillStyle=f.col;ctx.fillText(f.txt,q[0],q[1]);
    ctx.globalAlpha=1;
  }

  /* --- HUD --- */
  const sc=Math.floor(run.score);
  if(hScore._v!==sc){hScore._v=sc;hScore.textContent=sc;}
  const dm=Math.floor(run.dist);
  if(hDist._v!==dm){hDist._v=dm;hDist.textContent=dm+' m';}
  hSpd.textContent=Math.round(run.speed*5.4);
  hBar.style.width=clamp((run.speed-14)/(MAXS-14)*100,0,100)+'%';
  hintEl.style.opacity=(state==='playing')?clamp(1-(run.t-4)/2,0,1)*.85:0;
}

/* ============================== RIDGE (2D mesas) ============================== */
function ridge(par,amp,seed,col,horizonY){
  ctx.fillStyle=col;ctx.beginPath();ctx.moveTo(-30,H+20);
  for(let sx=-30;sx<=W+30;sx+=12){
    const nc=sx*.006+cam.x*par;
    const h=(vnoise(nc*.35+seed)*.7+vnoise(nc*1.7+seed*3)*.3)*amp;
    ctx.lineTo(sx,horizonY-h);
  }
  ctx.lineTo(W+30,H+20);ctx.closePath();ctx.fill();
}
