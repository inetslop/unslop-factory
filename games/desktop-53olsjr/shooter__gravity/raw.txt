  if(d>24){ax=dx/d;ay=dy/d;}}
 const l=Math.hypot(ax,ay)||1;P.vx+=ax/l*1500*dt;P.vy+=ay/l*1500*dt;
 for(const c of G.cores){const dx=c.x-P.x,dy=c.y-P.y,d2=dx*dx+dy*dy;
  if(d2<4e5&&d2>1){const d=Math.sqrt(d2),a=Math.min(1700,K/d2)*.35;P.vx+=dx/d*a*dt;P.vy+=dy/d*a*dt;}}

 if(P.y<16){P.y=16;P.vy*=-.3}if(P.y>H-16){P.y=H-16;P.vy*=-.3}
 P.thrust=(ax||ay)?1:0;
 const aimPt=touchMode&&touchPt?touchPt:mouse;
 P.a=Math.atan2(aimPt.y-P.y,aimPt.x-P.x);
 if(G.rapidT>0)G.rapidT-=dt;
 if(P.inv>0)P.inv-=dt;
 G.fireT-=dt;const firing=mouse.down||(touchMode&&touchPt);

 b.life-=dt;if(b.life<=0){G.bullets.splice(i,1);continue;}
 b.vy+=G.field*dt;
 for(const c of G.cores){const dx=c.x-b.x,dy=c.y-b.y,d2=dx*dx+dy*dy;

  const ar=c.r*.55;if(d2<ar*ar){c.flash=1;sfx.absorb();burst(b.x,b.y,4,['#4fe3cf','#fff'],120,.3,1.6);G.bullets.splice(i,1);break;}}
 if(!G.bullets[i])continue;b.x+=b.vx*dt;b.y+=b.vy*dt;

 if(g.x<-70||g.x>W+70||g.y<-70||g.y>H+70)G.gravel.splice(i,1);}}

function updateEnemies(dt){for(let i=G.enemies.length-1;i>=0;i--){const e=G.enemies[i];
 e.t+=dt;e.flash=Math.max(0,e.flash-dt*6);
 if(G.state==='over'){if(e.boomAt&&G.t>=e.boomAt){explode(e,false);G.enemies.splice(i,1);}continue;}
 const tgt=G.state==='menu'?{x:W/2,y:H*.45}:P;
 switch(e.type){
  case 'mite':{let dx=tgt.x-e.x,dy=tgt.y-e.y,d=Math.hypot(dx,dy)||1;let ux=dx/d,uy=dy/d;
   if(G.state==='menu'&&d<90){ux=-ux;uy=-uy;}
   const wob=Math.sin(e.t*4+e.ph)*.7,px=-uy*wob,py=ux*wob,nx=ux+px,ny=uy+py,nl=Math.hypot(nx,ny)||1;
   e.vx+=((nx/nl)*e.spd-e.vx)*Math.min(1,dt*2.5);e.vy+=((ny/nl)*e.spd-e.vy)*Math.min(1,dt*2.5);break;}
  case 'sinker':{if(e.y<e.ty)e.vy+=(90-e.vy)*Math.min(1,dt*2);
   else e.vy+=(Math.sin(e.t*2+e.ph)*14-e.vy)*Math.min(1,dt*3);
   const dxp=P.x-e.x;e.vx+=((clamp(dxp,-1,1)*55)-e.vx)*Math.min(1,dt*1.5);
   e.fireT-=dt;if(e.fireT<=0&&dist2(e.x,e.y,P.x,P.y)<520*520){
    fireGravel(e,Math.atan2(P.y-e.y,P.x-e.x),260);e.fireT=rand(2,2.8);}break;}
  case 'orbiter':{e.oang+=e.ospd*dt;const c=e.core;
   e.x=c.x+Math.cos(e.oang)*e.orad;e.y=c.y+Math.sin(e.oang)*e.orad;
   e.fireT-=dt;if(e.fireT<=0&&dist2(e.x,e.y,P.x,P.y)<560*560){
    fireGravel(e,Math.atan2(P.y-e.y,P.x-e.x),240);e.fireT=rand(2.6,3.4);}break;}
  case 'brute':{const dx=tgt.x-e.x,dy=tgt.y-e.y,d=Math.hypot(dx,dy)||1;
   e.vx+=((dx/d)*e.spd-e.vx)*Math.min(1,dt*1.2);e.vy+=((dy/d)*e.spd-e.vy)*Math.min(1,dt*1.2);
   e.fireT-=dt;if(e.fireT<=0&&dist2(e.x,e.y,P.x,P.y)<600*600){
    const base=Math.atan2(P.y-e.y,P.x-e.x);
    for(let k=-1;k<=1;k++)fireGravel(e,base+k*.28,240);e.fireT=rand(2.6,3.4);}break;}}
 if(e.type!=='orbiter'){
  for(const c of G.cores){const dx=c.x-e.x,dy=c.y-e.y,d2=dx*dx+dy*dy;
   if(d2<4e5&&d2>1){const d=Math.sqrt(d2),a=Math.min(1700,K/d2)*.35;e.vx+=dx/d*a*dt;e.vy+=dy/d*a*dt;}}
  e.x+=e.vx*dt;e.y+=e.vy*dt;}
 if(G.state==='play'&&dist2(e.x,e.y,P.x,P.y)<(e.r+P.r)*(e.r+P.r)){
  const dmg=e.type==='brute'?2:1;hurt(dmg);
  const dx=P.x-e.x,dy=P.y-e.y,d=Math.hypot(dx,dy)||1;P.vx+=dx/d*260;P.vy+=dy/d*260;
  if(e.type==='mite'||e.type==='orbiter'){explode(e,false);G.enemies.splice(i,1);}}}}

function collisions(){if(G.state!=='play'&&!G.bullets.length)return;
 for(let i=G.bullets.length-1;i>=0;i--){const b=G.bullets[i];let hit=false;
  for(let j=G.enemies.length-1;j>=0;j--){const e=G.enemies[j];
   const rr=e.r+4;if(dist2(b.x,b.y,e.x,e.y)<rr*rr){hit=true;e.hp--;e.flash=1;
    if(e.type!=='orbiter'){e.vx+=b.vx*.03;e.vy+=b.vy*.03;}
    burst(b.x,b.y,5,['#ffd166','#fff'],180,.3,1.8);sfx.hit();
    if(e.hp<=0){explode(e,G.state==='play');G.enemies.splice(j,1);}break;}}
  if(hit)G.bullets.splice(i,1);}
 for(let i=G.gravel.length-1;i>=0;i--){const g=G.gravel[i];
  if(G.state==='play'&&dist2(g.x,g.y,P.x,P.y)<(4+P.r)*(4+P.r)){G.gravel.splice(i,1);hurt(1);}}}

function update(dt){G.t+=dt;

 G.field+=(ft-G.field)*Math.min(1,dt*5);
 if(f.charges<f.max&&f.t<=0){f.rt+=dt;if(f.rt>=f.regen){f.charges++;f.rt=0;}}
 if(G.banner){G.banner.t-=dt;if(G.banner.t<=0)G.banner=null;}
 G.shake=Math.max(0,G.shake-dt*26);G.flash=Math.max(0,G.flash-dt*2.4);G.retKick=Math.max(0,G.retKick-dt*6);
 if(G.comboT>0){G.comboT-=dt;if(G.comboT<=0)G.combo=0;}
 for(const c of G.cores)c.flash=Math.max(0,c.flash-dt*3);

 if(G.state==='play')updatePlayer(dt);
 else if(G.state==='over'){G.overT+=dt;
  if(G.overT>1.5&&!G.shown){G.shown=true;showOver();}}

 if(G.state==='play'){
  if(G.interT>0){G.interT-=dt;if(G.interT<=0){placeCores(G.wave+1);startWave(G.wave+1);}}
  else if(G.spawnQ.length){G.spawnT-=dt;
   if(G.spawnT<=0){spawnEnemy(G.spawnQ.pop());G.spawnT=G.spawnInt*rand(.7,1.3);}}

 updateBullets(dt);updateGravel(dt);updateEnemies(dt);collisions();
 updatePowers(dt);updateParts(dt);updateTexts(dt);updateDust(dt);}
