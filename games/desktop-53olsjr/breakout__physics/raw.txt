function updatePaddle(dt){
  const wantW=fx.wide>G.t?178:paddle.baseW;paddle.w+=(wantW-paddle.w)*Math.min(1,dt*10);
  let nx=paddle.x;
  if(G.mode==='play'&&paddle.target!=null)nx=paddle.target;
  else{const dir=(keys.right?1:0)-(keys.left?1:0);if(dir){paddle.target=null;nx+=dir*980*dt;}}
  if(G.inMenu&&G.mode==='menu'){const b=balls[0];
    if(b&&!b.stuck)nx=paddle.x+clamp((b.x+Math.sin(G.t*2.3)*46)-paddle.x,-600*dt,600*dt);}
  nx=clamp(nx,F.x+paddle.w/2+8,F.right-paddle.w/2-8);
  paddle.vx=dt>0?clamp((nx-paddle.x)/dt,-1400,1400):0;paddle.x=nx;
  paddle.squash=Math.max(0,paddle.squash-dt*5);
  if(fx.laser>G.t&&G.mode==='play'){G.fireT-=dt;if(G.fireT<=0){fireLaser();G.fireT=.3;}}
}
function updateBalls(dt){
  for(const b of balls){
    if(b.stuck){b.stuckT+=dt;b.x=paddle.x+clamp(b.attach,-1,1)*(paddle.w/2-14);b.y=F.bottom-paddle.h-b.r-2;
      if((G.inMenu&&b.stuckT>0.9)||(!G.inMenu&&b.stuckT>6))launch(b);continue;}
    const steps=Math.max(1,Math.ceil(Math.hypot(b.vx,b.vy)*dt/(b.r*0.8))),sdt=dt/steps;
    for(let i=0;i<steps;i++){b.x+=b.vx*sdt;b.y+=b.vy*sdt;collideWalls(b);if(b.dead)break;brickHit(b);paddleHit(b);}
    b.trail.push({x:b.x,y:b.y});if(b.trail.length>12)b.trail.shift();
  }
  const before=balls.length;balls=balls.filter(b=>!b.dead);
  if(before&&balls.length===0){if(G.mode==='play')loseBall();else newStuckBall();}
}
function updatePows(dt){for(const p of pows){p.t+=dt;p.y+=p.vy*dt;p.x=p.baseX+Math.sin(p.t*2.6+p.ph)*10;
    if(p.y>F.bottom-paddle.h-14&&p.y<F.bottom+4&&Math.abs(p.x-paddle.x)<paddle.w/2+18){applyPow(p);p.dead=true;}
    if(p.y>F.bottom+30)p.dead=true;}pows=pows.filter(p=>!p.dead);}
function updateShots(dt){for(const s of shots){s.y-=960*dt;if(s.y<F.y+4){s.dead=true;continue;}
    for(const br of bricks){if(br.hp<=0)continue;
      if(s.x>br.x&&s.x<br.x+br.w&&s.y>br.y&&s.y<br.y+br.h){damageBrick(br);sparkAt(s.x,br.y,'#bfe6ff',4);s.dead=true;break;}}}
  shots=shots.filter(s=>!s.dead);}
function updateParts(dt){for(const p of parts){p.life-=dt;
    if(p.type==='shard'||p.type==='confetti'){p.vy+=(p.type==='confetti'?240:520)*dt;p.x+=p.vx*dt;p.y+=p.vy*dt;p.rot+=p.spin*dt;}
    else if(p.type==='spark'){p.x+=p.vx*dt;p.y+=p.vy*dt;p.vy+=160*dt;}
    else if(p.type==='ring')p.r+=(p.r1-p.r0)*Math.min(1,dt*7);}
  parts=parts.filter(p=>p.life>0);if(parts.length>420)parts.splice(0,parts.length-420);}
function updatePops(dt){for(const p of pops){p.y-=26*dt;p.life-=dt;}pops=pops.filter(p=>p.life>0);}

function update(dt){G.t+=dt;
  G.shake*=Math.exp(-7*dt);if(G.shake<.1)G.shake=0;
  G.flash=Math.max(0,G.flash-dt*1.6);
  for(const d of dust){d.y-=d.s*dt;d.x+=Math.sin(G.t*.8+d.ph)*4*dt;if(d.y<F.y)d.y=F.bottom;}
  updatePaddle(dt);
  if(G.mode!=='over')updateBalls(dt);
  if(G.mode==='clear'){G.clearT-=dt;
    if(G.clearT<=0){if(G.inMenu){buildLevel((Math.floor(G.t*0)%1)||0,0);buildLevel(0,0);G.mode='menu';}else loadLevel(G.level+1);}}
  updatePows(dt);updateShots(dt);updateParts(dt);updatePops(dt);
  if(G.banner){G.banner.t-=dt;if(G.banner.t<=0)G.banner=null;}
  while(G.score>=G.nextLife){G.nextLife+=10000;
    if(G.lives<6&&G.mode==='play'){G.lives++;popText(GW/2,F.y+F.h*.4,'1UP!',COL.mint,30);sfx('lifeup');}}
}
