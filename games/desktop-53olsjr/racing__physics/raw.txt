const SURF={road:{grip:9,cap:1},curb:{grip:6.2,cap:.84},off:{grip:6.5,cap:.5}};
class Car{
  constructor(o){Object.assign(this,o);this.vx=0;this.vy=0;this.vF=0;this.vL=0;this.prog=0;this.lapsDone=0;
    this.drifting=false;this.surface='road';this.d=0;this.stuckT=0;this.hitV=0;this.lastLap=null;this.finishTime=null;
    this.inp={throttle:0,brake:0,steer:0,hb:false};this.pw1=null;this.pw2=null;}
  reset(x,y,h,idx,gap){this.x=x;this.y=y;this.h=h;this.vx=0;this.vy=0;this.vF=0;this.vL=0;
    this.sIdx=idx;this.prevS=idx;this.gap=gap;this.prog=0;this.lapsDone=0;this.lastLap=null;this.finishTime=null;
    this.drifting=false;this.stuckT=0;this.hitV=0;this.pw1=null;this.pw2=null;this.lapTimes=[];}
  get raceDist(){return this.prog-this.gap;}
  update(dt){
    const inp=this.inp,np=nearestTrack(this.x,this.y,this.sIdx);
    this.prevS=this.sIdx;this.sIdx=np.idx;this.d=np.d;this.surface=surfaceOf(this.d);
    const surf=SURF[this.surface];
    let fx=Math.cos(this.h),fy=Math.sin(this.h),rx=-fy,ry=fx;
    let vF=this.vx*fx+this.vy*fy,vL=this.vx*rx+this.vy*ry;
    const maxV=this.maxV*surf.cap;
    if(inp.throttle)vF+=this.accel*(1-vF/maxV)*dt;
    if(inp.brake){if(vF>20)vF-=980*dt;else vF-=340*dt;}
    if(inp.hb)vF*=Math.exp(-1.15*dt);
    const cd=this.surface==='road'?.4:(this.surface==='curb'?1.1:2.2);
    vF*=Math.exp(-cd*dt);if(vF<-230)vF=-230;
    let grip=surf.grip*(inp.hb?.22:1);
    const sat=clamp((Math.abs(vL)-85)/260,0,1);grip*=(1-.5*sat);
    vL*=Math.exp(-grip*dt);
    const sf=clamp(Math.abs(vF)/150,0,1)*(1-.38*clamp(vF/this.maxV,0,1));
    this.h+=inp.steer*this.steerRate*sf*dt*(inp.hb?1.25:1);
    fx=Math.cos(this.h);fy=Math.sin(this.h);rx=-fy;ry=fx;
    this.vx=fx*vF+rx*vL;this.vy=fy*vF+ry*vL;
    this.x+=this.vx*dt;this.y+=this.vy*dt;
    this.x=clamp(this.x,20,WORLD.w-20);this.y=clamp(this.y,20,WORLD.h-20);
    this.vF=vF;this.vL=vL;
    // wall
    const lim=WALL_D-13;this.hitV=0;
    if(this.d>lim){const nx=(this.x-np.px)/this.d,ny=(this.y-np.py)/this.d;
      this.x=np.px+nx*lim;this.y=np.py+ny*lim;
      const vn=this.vx*nx+this.vy*ny;
      if(vn>0){this.vx-=nx*vn*1.35;this.vy-=ny*vn*1.35;this.hitV=vn;}}
    // progress
    const dI=wrapIdx(this.sIdx-this.prevS);if(dI>0&&dI<N/2)this.prog+=dI;
    // drift hysteresis
    if(!this.drifting&&Math.abs(vL)>85&&Math.hypot(this.vx,this.vy)>130)this.drifting=true;
    else if(this.drifting&&Math.abs(vL)<45)this.drifting=false;
  }
}

function aiInput(c,dt){
  const la=Math.max(4,Math.round((150+c.vF*.3)/SP));
  let ti=(c.sIdx+la)%N;
  const a1=c.sIdx,a2=(c.sIdx+la)%N,a3=(c.sIdx+2*la)%N;
  const angA=Math.atan2(C[a2][1]-C[a1][1],C[a2][0]-C[a1][0]);
  const angB=Math.atan2(C[a3][1]-C[a2][1],C[a3][0]-C[a2][0]);
  const sharp=Math.abs(wrapAng(angB-angA));
  let base=c.maxV*c.skill;
  // rubber band vs player
  const diff=player.raceDist-c.raceDist;
  if(diff>40)base*=1.10;else if(diff<-60)base*=.93;
  base*=1+.03*Math.sin(nowS*1.7+c.seed);
  let targetV=clamp(base-Math.max(0,sharp-.1)*540,235,base);
  if(c.d>WALL_D-26||c.stuckT>1.2){ti=(c.sIdx+8)%N;targetV=Math.min(targetV,base*.7);}
  const tp=C[ti];
  const tx=tp[0]+NM[ti][0]*c.lineOff,ty=tp[1]+NM[ti][1]*c.lineOff;
  let dh=wrapAng(Math.atan2(ty-c.y,tx-c.x)-c.h);
  c.inp.steer=clamp(dh*2.6-c.vL*.004,-1,1)+(c.stuckT>2?(Math.random()-.5)*.3:0);
  if(c.finishTime!=null){c.inp.throttle=.4;c.inp.brake=0;}
  else if(c.vF>targetV+25){c.inp.throttle=0;c.inp.brake=1;}
  else if(c.vF<targetV-15){c.inp.throttle=1;c.inp.brake=0;}
  else{c.inp.throttle=.55;c.inp.brake=0;}
  c.inp.hb=false;
  if(Math.hypot(c.vx,c.vy)<40&&c.finishTime==null)c.stuckT+=dt;else c.stuckT=Math.max(0,c.stuckT-dt*2);
}

}
function render(){

  ctx.fillStyle='#3f7a29';ctx.fillRect(0,0,view.w,view.h);
  const shx=(Math.random()-.5)*cam.shake*2,shy=(Math.random()-.5)*cam.shake*2;
  ctx.save();ctx.translate(view.w/2,view.h/2);ctx.scale(cam.z,cam.z);ctx.translate(-cam.x+shx,-cam.y+shy);
  ctx.drawImage(worldCv,0,0);
  drawMarks(cam.z);drawParticles();
  for(const c of cars)drawCar(c);
  ctx.restore();
  // speed streaks
  const spd=Math.hypot(player.vx,player.vy),sf=clamp(spd/player.maxV,0,1);
  if(sf>.78&&(state==='racing')){const a=(sf-.78)/.22*.4;ctx.strokeStyle='rgba(255,255,255,'+a.toFixed(3)+')';ctx.lineWidth=2;
    for(let i=0;i<14;i++){const ang=i/14*TAU+nowS*.6,r0=Math.max(view.w,view.h)*.55,r1=r0-90-a*160;
      const cx=view.w/2,cy=view.h/2;ctx.beginPath();
      ctx.moveTo(cx+Math.cos(ang)*r0,cy+Math.sin(ang)*r0);ctx.lineTo(cx+Math.cos(ang)*r1,cy+Math.sin(ang)*r1);ctx.stroke();}}
  if(state==='racing'&&raceT<.35){ctx.fillStyle='rgba(255,255,255,'+((.35-raceT)*1.4).toFixed(3)+')';ctx.fillRect(0,0,view.w,view.h);}
  drawMinimap();
}
