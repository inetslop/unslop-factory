  AC.init();AC.resume();
  pB.length=eB.length=enemies.length=cores.length=pows.length=parts.length=floats.length=warns.length=spawnQ.length=ghosts.length=0;

/* ================= waves / spawns ================= */
function startWave(n){
  wave=n;interT=-1;spawnT=1.35;spawnQ.length=0;
  let brutes=n>=4?Math.min(1+((n-5)/3|0),4):0;
  for(let i=0;i<brutes;i++)spawnQ.push('brute');
  const pool=[['drone',1]];
  if(n>=2)pool.push(['striker',.9]);
  if(n>=3)pool.push(['spitter',.75]);
  if(n>=4)pool.push(['splitter',.7]);
  const tw=pool.reduce((s,p)=>s+p[1],0);
  let budget=Math.min(6+n*2,44)-brutes;
  while(budget-->0){let r=Math.random()*tw;for(const[t,w]of pool){if(r<w){spawnQ.push(t);break}r-=w}}
  shuffle(spawnQ);
  showBanner('WAVE '+pad2(n),n===1?'DEFEND THE GRID':'HOSTILES INBOUND');
  AC.waveStart();
}
function addWarn(type){
  let x=0,y=0,tries=0;
  do{const s=(Math.random()*4)|0;
    if(s===0){x=rand(60,WORLD.w-60);y=52}else if(s===1){x=rand(60,WORLD.w-60);y=WORLD.h-52}
    else if(s===2){x=52;y=rand(60,WORLD.h-60)}else{x=WORLD.w-52;y=rand(60,WORLD.h-60)}
  }while(dist2(x,y,player.x,player.y)<300*300&&++tries<10);
  warns.push({type,x,y,t:0,dur:.6});
}
function makeEnemy(type,x,y){
  const e={type,x,y,vx:0,vy:0,ph:rand(TAU),flash:0,dead:false};
  switch(type){
    case 'drone':e.r=13;e.hp=2+((wave/4)|0);e.sc=10;e.sp=120+Math.min(wave*6,90);break;
    case 'mini': e.r=9; e.hp=1;e.sc=5; e.sp=200+Math.min(wave*4,60);break;
    case 'striker':e.r=12;e.hp=3+(wave>=6?1:0);e.sc=15;e.mode='orbit';e.dir=Math.random()<.5?-1:1;e.cd=rand(1,2);e.t=0;e.tx=x;e.ty=y;break;
    case 'spitter':e.r=15;e.hp=4+((wave/5)|0);e.sc=20;e.cd=rand(.9,1.7);e.wf=0;e.strT=rand(1.4,2.6);e.dir=Math.random()<.5?-1:1;break;
    case 'splitter':e.r=16;e.hp=3;e.sc=12;e.sp=95;break;
    case 'brute': e.r=27;e.hp=Math.min(12+wave*2,42);e.sc=75;e.ringCd=rand(2,3);e.wf=0;break;
  }
  e.maxHp=e.hp;return e;
}

    addPart({k:'shard',x,y,vx:Math.cos(a)*sp,vy:Math.sin(a)*sp,rot:rand(TAU),vr:rand(-9,9),life:rand(.3,.7),max:.7,size:rand(4,9),col})}

/* ================= player actions ================= */
const DASH_SP=980,DASH_CD=2.2;
function tryDash(){
  const p=player;if(p.dashCd>0||p.dashT>0)return;
  let ix=(keys['d']||keys['arrowright']?1:0)-(keys['a']||keys['arrowleft']?1:0);
  let iy=(keys['s']||keys['arrowdown']?1:0)-(keys['w']||keys['arrowup']?1:0);
  const l=Math.hypot(ix,iy);
  if(l>0){p.dvX=ix/l;p.dvY=iy/l}else{p.dvX=Math.cos(p.aim);p.dvY=Math.sin(p.aim)}
  p.dashT=.16;p.dashCd=DASH_CD;p.inv=Math.max(p.inv,.34);AC.dash();
}
function fire(){
  const p=player,od=p.od>0;
  p.fireCd=Math.max(.06,(od?.085:.17)-(p.weapon-1)*.014);
  const spd=od?830:660,tipx=p.x+Math.cos(p.aim)*17,tipy=p.y+Math.sin(p.aim)*17;
  for(const a of PATTERN[p.weapon]){const an=p.aim+a;
    pB.push({x:tipx,y:tipy,vx:Math.cos(an)*spd,vy:Math.sin(an)*spd,r:3.4,life:1.1})}
  p.vx-=Math.cos(p.aim)*(od?26:14);p.vy-=Math.sin(p.aim)*(od?26:14);
  addPart({k:'flash',x:tipx,y:tipy,life:.05,max:.05,size:9,col:'#ffd23f'});
  for(let i=0;i<3;i++){const a=p.aim+rand(-.5,.5),sp=rand(120,320);
    addPart({k:'spark',x:tipx,y:tipy,vx:Math.cos(a)*sp,vy:Math.sin(a)*sp,life:.14,max:.14,size:1.6,col:'#ffd9a0'})}
  AC.shot(od);
}
function damagePlayer(dmg,sx,sy){
  const p=player;if(!p.alive||p.inv>0||p.dashT>0)return;
  let d=dmg;
  if(p.shield>0){const a=Math.min(p.shield,d);p.shield-=a;d-=a;
    if(d<=0){p.inv=.5;addPart({k:'ring',x:p.x,y:p.y,life:.3,max:.3,size:44,col:'#35e0ff'});AC.shieldHit();return}}
  p.hp-=d;combo=0;redFlash=.6;shake=Math.max(shake,10);p.inv=.95;
  const dx=p.x-sx,dy=p.y-sy,dd=Math.hypot(dx,dy)||1;
  p.vx+=dx/dd*340;p.vy+=dy/dd*340;AC.hurt();
  if(p.hp<=0){p.hp=0;playerDeath()}
}
function playerDeath(){
  player.alive=false;deathT=1.25;
  explosion(player.x,player.y,3,'#ffb02e');shake=20;AC.boom();
}

function tierCol(m){return m>=5?'#ff4d5e':m===4?'#ff7a1a':m===3?'#ffd23f':m===2?'#ffe9b0':'#cfe9f5'}
function killEnemy(e){
  if(e.dead)return;e.dead=true;
  const m=mult(),pts=e.sc*m;
  score+=pts;combo++;comboT=3;if(combo>maxCombo)maxCombo=combo;
  const nm=mult();if(nm>maxMult)maxMult=nm;kills++;
  floatText(e.x,e.y,'+'+pts,tierCol(m),12+m*2);
  const cs=CORES[e.type]||0;
  for(let i=0;i<cs;i++)cores.push({x:e.x+rand(-8,8),y:e.y+rand(-8,8),vx:rand(-90,90),vy:rand(-90,90),life:12});
  if(cores.length>140)cores.splice(0,cores.length-140);
  if(e.type==='brute'){dropPowerAt(e.x,e.y);explosion(e.x,e.y,ESIZE.brute,ECOL.brute)}
  else{if(Math.random()<.12)dropPowerAt(e.x,e.y);explosion(e.x,e.y,ESIZE[e.type],ECOL[e.type])}
  if(e.type==='splitter')for(let i=0;i<2;i++){const a=rand(TAU);
    const mi=makeEnemy('mini',e.x+Math.cos(a)*10,e.y+Math.sin(a)*10);
    mi.vx=Math.cos(a)*260;mi.vy=Math.sin(a)*260;enemies.push(mi)}
}

/* ================= update ================= */
function update(rdt){
  const ts=player.alive?1:.32,dt=rdt*ts;time+=rdt;
  if(combo>0){comboT-=dt;if(comboT<=0)combo=0}
  redFlash=Math.max(0,redFlash-rdt*1.8);
  shake=Math.max(0,shake-rdt*26);

  /* --- player --- */
  const p=player;
  if(p.alive){
    p.inv=Math.max(0,p.inv-dt);p.fireCd-=dt;p.dashCd=Math.max(0,p.dashCd-dt);p.od=Math.max(0,p.od-dt);
    let ix=(keys['d']||keys['arrowright']?1:0)-(keys['a']||keys['arrowleft']?1:0);
    let iy=(keys['s']||keys['arrowdown']?1:0)-(keys['w']||keys['arrowup']?1:0);
    const il=Math.hypot(ix,iy)||1;ix/=il;iy/=il;
    if(p.dashT>0){
      p.dashT-=dt;p.x+=p.dvX*DASH_SP*dt;p.y+=p.dvY*DASH_SP*dt;
      ghosts.push({x:p.x,y:p.y,a:p.aim,t:1});if(ghosts.length>12)ghosts.shift();
      addPart({k:'spark',x:p.x-p.dvX*10,y:p.y-p.dvY*10,vx:-p.dvX*rand(60,160),vy:-p.dvY*rand(60,160),life:.25,max:.25,size:2,col:'#35e0ff'});
      for(const e of enemies){if(e.dead)continue;const rr=e.r+22;
        if(dist2(p.x,p.y,e.x,e.y)<rr*rr){e.vx+=p.dvX*300;e.vy+=p.dvY*300;
          addPart({k:'spark',x:(p.x+e.x)/2,y:(p.y+e.y)/2,vx:rand(-160,160),vy:rand(-160,160),life:.2,max:.2,size:2,col:'#35e0ff'})}}
    }else{
      p.vx+=ix*2400*dt;p.vy+=iy*2400*dt;
      const f=Math.pow(.0015,dt);p.vx*=f;p.vy*=f;
      const sp=Math.hypot(p.vx,p.vy);if(sp>330){p.vx*=330/sp;p.vy*=330/sp}
      p.x+=p.vx*dt;p.y+=p.vy*dt;
      if(sp>70&&Math.random()<dt*50)addPart({k:'spark',x:p.x-Math.cos(p.aim)*12,y:p.y-Math.sin(p.aim)*12,
        vx:-p.dvX*0,vx:rand(-30,30)-p.vx*.1,vy:rand(-30,30)-p.vy*.1,life:.2,max:.2,size:1.6,col:'#ff9a3d'});
    }
    p.x=clamp(p.x,p.r,WORLD.w-p.r);p.y=clamp(p.y,p.r,WORLD.h-p.r);
    const wx=cam.x+mouse.x,wy=cam.y+mouse.y;
    p.aim=Math.atan2(wy-p.y,wx-p.x);
    if(mouse.down&&p.fireCd<=0)fire();
  }

  /* --- spawn queue / warnings --- */
  if(spawnQ.length&&interT<0&&player.alive){
    spawnT-=dt;
    if(spawnT<=0&&enemies.length<55){spawnT=Math.max(.28,1.0-wave*.04);addWarn(spawnQ.shift())}
  }
  for(let i=warns.length-1;i>=0;i--){const w=warns[i];w.t+=dt;
    if(w.t>=w.dur){enemies.push(makeEnemy(w.type,w.x,w.y));
      addPart({k:'ring',x:w.x,y:w.y,life:.3,max:.3,size:40,col:ECOL[w.type]});warns.splice(i,1)}}

  /* --- enemies --- */
  for(const e of enemies){
    if(e.dead)continue;
    const dx=p.x-e.x,dy=p.y-e.y,d=Math.hypot(dx,dy)||1,ux=dx/d,uy=dy/d;
    e.flash=Math.max(0,e.flash-dt);
    switch(e.type){
      case 'drone':case 'mini':{
        const wob=Math.sin(time*3+e.ph)*40/Math.max(60,d);
        const tx=ux-wob*-uy,ty=uy+wob*ux;
        e.vx=lerp(e.vx,tx*e.sp,Math.min(1,dt*3));e.vy=lerp(e.vy,ty*e.sp,Math.min(1,dt*3));break}
      case 'splitter':{
        const wob=Math.sin(time*2.2+e.ph)*50/Math.max(60,d);
        e.vx=lerp(e.vx,(ux-wob*-uy)*e.sp,Math.min(1,dt*2.4));e.vy=lerp(e.vy,(uy+wob*ux)*e.sp,Math.min(1,dt*2.4));break}
      case 'striker':{
        e.cd-=dt;const RING=250;
        if(e.mode==='orbit'){
          const want=d>RING+40?1:d<RING-40?-1:0;
          const tx=ux*want-uy*e.dir*.9,ty=uy*want+ux*e.dir*.9,tl=Math.hypot(tx,ty)||1;
          e.vx=lerp(e.vx,tx/tl*235,Math.min(1,dt*3));e.vy=lerp(e.vy,ty/tl*235,Math.min(1,dt*3));
          if(d<340&&e.cd<=0){e.mode='tele';e.t=0;const lead=.5;e.tx=p.x+p.vx*lead;e.ty=p.y+p.vy*lead}
        }else if(e.mode==='tele'){
          e.t+=dt;e.vx*=Math.pow(.001,dt);e.vy*=Math.pow(.001,dt);
          if(e.t>=.5){e.mode='dash';e.t=0;const ax=e.tx-e.x,ay=e.ty-e.y,al=Math.hypot(ax,ay)||1;
            e.vx=ax/al*780;e.vy=ay/al*780;AC.strike()}
        }else if(e.mode==='dash'){
          e.t+=dt;addPart({k:'spark',x:e.x,y:e.y,vx:rand(-40,40),vy:rand(-40,40),life:.18,max:.18,size:2,col:'#ffc247'});
          if(e.t>=.3){e.mode='rest';e.t=0}
        }else{ e.t+=dt;e.vx*=Math.pow(.05,dt);e.vy*=Math.pow(.05,dt);
          if(e.t>=.7){e.mode='orbit';e.cd=rand(1.2,2.2);e.dir=Math.random()<.5?-1:1} }
        break}
      case 'spitter':{
        const pref=310;let radial=d<pref-60?-1:d>pref+60?1:0;
        e.strT-=dt;if(e.strT<=0){e.dir*=-1;e.strT=rand(1.8,3)}
        const tx=ux*radial-uy*e.dir*.8,ty=uy*radial+ux*e.dir*.8,tl=Math.hypot(tx,ty)||1;
        e.vx=lerp(e.vx,tx/tl*170,Math.min(1,dt*2.6));e.vy=lerp(e.vy,ty/tl*170,Math.min(1,dt*2.6));
        if(e.cd>0)e.cd-=dt;
        else{e.wf+=dt;
          if(e.wf>=.5){const spE=Math.min(400,250+wave*6),base=Math.atan2(p.y-e.y,p.x-e.x);
            for(const a of[-.22,0,.22])eShot(e.x,e.y,base+a,spE,9);
            e.wf=0;e.cd=rand(1.9,2.6);AC.spit()}}
        break}
      case 'brute':{
        const sp=55+Math.min(wave*2,35);
        e.vx=lerp(e.vx,ux*sp,Math.min(1,dt*1.8));e.vy=lerp(e.vy,uy*sp,Math.min(1,dt*1.8));
        if(wave>=5){e.ringCd-=dt;
          if(e.ringCd<=0&&d<430){e.wf+=dt;
            if(e.wf>=.6){const off=rand(TAU);
              for(let i=0;i<8;i++)eShot(e.x,e.y,off+i/8*TAU,195,12);
              e.wf=0;e.ringCd=rand(2.8,3.6);AC.ring()}}
          else if(e.ringCd>0)e.wf=0}
        break}
    }
    e.x+=e.vx*dt;e.y+=e.vy*dt;
    e.x=clamp(e.x,e.r,WORLD.w-e.r);e.y=clamp(e.y,e.r,WORLD.h-e.r);
    if(p.alive&&d<e.r+p.r+2){
      const nx=(p.x-e.x)/(d||1),ny=(p.y-e.y)/(d||1);
      p.x=e.x+nx*(e.r+p.r+3);p.y=e.y+ny*(e.r+p.r+3);
      if(p.inv<=0&&p.dashT<=0){
        damagePlayer(CDAM[e.type],e.x,e.y);
        if(KAMIKAZE[e.type])killEnemy(e);
      }
    }
  }
  /* separation */
  for(let i=0;i<enemies.length;i++){const a=enemies[i];if(a.dead)continue;
    for(let j=i+1;j<enemies.length;j++){const b=enemies[j];if(b.dead)continue;
      const dx=b.x-a.x,dy=b.y-a.y,min=a.r+b.r+4,d2=dx*dx+dy*dy;
      if(d2>0.01&&d2<min*min){const d=Math.sqrt(d2),push=(min-d)/d*.5;
        a.x-=dx*push;a.y-=dy*push;b.x+=dx*push;b.y+=dy*push}}}
  if(enemies.some(e=>e.dead))enemies=enemies.filter(e=>!e.dead);

  /* --- bullets --- */
  for(let i=pB.length-1;i>=0;i--){const b=pB[i];b.x+=b.vx*dt;b.y+=b.vy*dt;b.life-=dt;
    let dead=b.life<=0||b.x<6||b.x>WORLD.w-6||b.y<6||b.y>WORLD.h-6;
    if(!dead)for(const e of enemies){if(e.dead)continue;const rr=e.r+b.r;
      if(dist2(b.x,b.y,e.x,e.y)<rr*rr){e.hp--;e.flash=.14;
        const kb=e.type==='brute'?25:90,bl=Math.hypot(b.vx,b.vy)||1;
        e.vx+=b.vx/bl*kb;e.vy+=b.vy/bl*kb;AC.hit();

function buildShape(e){
  const p=new Path2D(),r=e.r;
  switch(e.type){
    case 'drone':case 'mini':p.moveTo(r,0);p.lineTo(-r*.8,r*.75);p.lineTo(-r*.4,0);p.lineTo(-r*.8,-r*.75);p.closePath();break;
    case 'striker':p.moveTo(r*1.5,0);p.lineTo(-r,.85*r);p.lineTo(-r*.5,0);p.lineTo(-r,-.85*r);p.closePath();break;
    case 'spitter':{const rr=r*(1+.06*Math.sin(time*4+e.ph));p.moveTo(0,-rr);p.lineTo(rr,0);p.lineTo(0,rr);p.lineTo(-rr,0);p.closePath();break}
    case 'splitter':{const a=e.ph+time*2.4,ox=Math.cos(a)*r*.45,oy=Math.sin(a)*r*.45;
      p.arc(ox,oy,r*.62,0,TAU);p.moveTo(-ox+r*.62,-oy);p.arc(-ox,-oy,r*.62,0,TAU);break}
    case 'brute':for(let i=0;i<6;i++){const a=i/6*TAU-Math.PI/6;const px=Math.cos(a)*r,py=Math.sin(a)*r;i?p.lineTo(px,py):p.moveTo(px,py)}p.closePath();break;
  }
  return p;
}

    else if(q.k==='flash')drawGlow(q.x,q.y,q.size*(1.3-t),q.col,t);
    else if(q.k==='ring'){const r=q.size*(1-t*t);
      ctx.strokeStyle=q.col;ctx.globalAlpha=t;ctx.lineWidth=1+3*t;
      ctx.beginPath();ctx.arc(q.x,q.y,r,0,TAU);ctx.stroke();ctx.globalAlpha=1}
    else if(q.k==='shard'){ctx.save();ctx.translate(q.x,q.y);ctx.rotate(q.rot);
      ctx.strokeStyle=q.col;ctx.globalAlpha=t;ctx.lineWidth=2.5;
      ctx.beginPath();ctx.moveTo(-q.size/2,0);ctx.lineTo(q.size/2,0);ctx.stroke();ctx.restore();ctx.globalAlpha=1}}
  ctx.restore();
}

function drawFloats(){
  ctx.save();ctx.textAlign='center';
  for(const f of floats){const t=f.life/f.max;
    ctx.font=`700 ${f.size}px ui-monospace, Menlo, Consolas, monospace`;
    ctx.globalAlpha=Math.min(1,t*1.6);ctx.fillStyle=f.col;
    ctx.fillText(f.text,f.x,f.y)}
  ctx.restore();ctx.globalAlpha=1;
}

function drawAmbient(){
  for(const s of ambient){const col=ECOL[s.type];
    ctx.save();ctx.translate(s.x,s.y);ctx.rotate(s.a);ctx.globalAlpha=.45;
    const p=new Path2D();p.moveTo(14,0);p.lineTo(-9,8);p.lineTo(-4,0);p.lineTo(-9,-8);p.closePath();
    ctx.fillStyle=col;ctx.fill(p);ctx.restore()}
  ctx.globalAlpha=1;
}

function drawReticle(){
  const x=mouse.x,y=mouse.y,r0=9+(player.fireCd>0?3:0);
  ctx.save();ctx.translate(x,y);ctx.rotate(time*1.2);
  ctx.strokeStyle='rgba(255,210,63,.95)';ctx.lineWidth=1.6;
  for(let i=0;i<4;i++){ctx.rotate(Math.PI/2);ctx.beginPath();ctx.moveTo(r0,0);ctx.lineTo(r0+7,0);ctx.stroke()}
  ctx.restore();
  ctx.fillStyle='rgba(255,210,63,.9)';ctx.fillRect(x-1.5,y-1.5,3,3);
  ctx.strokeStyle='rgba(53,224,255,.3)';ctx.lineWidth=1;
  ctx.beginPath();ctx.arc(x,y,r0+13,0,TAU);ctx.stroke();
}

function draw(){
  ctx.fillStyle='#040b12';ctx.fillRect(0,0,W,H);
  const sx=(Math.random()*2-1)*shake,sy=(Math.random()*2-1)*shake;
  drawDust();
  ctx.save();ctx.translate(-cam.x+sx,-cam.y+sy);
  drawFloor();
  if(state==='menu')drawAmbient();
  else{
    drawWarns();drawCores();drawPows();
    drawBullets.call(null);
    drawEnemies();
    drawPlayer();
    drawParts();drawFloats();
  }
  ctx.restore();
  if(vign){ctx.fillStyle=vign;ctx.fillRect(0,0,W,H)}
  if(redFlash>0){ctx.fillStyle=`rgba(255,64,72,${redFlash*.3})`;ctx.fillRect(0,0,W,H)}
  if(state==='play'&&player.alive&&player.hp<35){
    const a=(.1+.08*Math.sin(time*7))*(1-player.hp/35);
    ctx.fillStyle=`rgba(255,40,60,${a})`;ctx.fillRect(0,0,W,H)}
  if(state==='play')drawReticle();
}

/* ================= main loop ================= */
let last=performance.now();
function frame(now){
  requestAnimationFrame(frame);
  let dt=(now-last)/1000;last=now;
  if(dt>0.05)dt=0.05;
  if(state==='play')update(dt);
  else if(state==='menu')menuUpdate(dt);
  else if(state==='over')fxOnly(dt);
  draw();
}
