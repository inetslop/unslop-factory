function carWorld(c){
  const N=track.N; let d=c.rdist; let i=Math.floor(d)%N; if(i<0)i+=N; const f=d-Math.floor(d);
  const p=track.pts[i],q=track.pts[(i+1)%N];

  return {x:(view.ox+(x+nx*c.rlat*HALFW)*view.s), y:(view.oy+(y+ny*c.rlat*HALFW)*view.s), a};
}

function botAct(c){
  if(c.finished)return null;
  const seg=curvAt(c.dist+Math.max(1,c.speed)*0.5);
  const ahead2=curvAt(c.dist+Math.max(1,c.speed)+1.5);
  if(Math.abs(c.lat)>0.8&&seg!==0&&Math.sign(c.lat)===-seg)return 'B'; // stuck inside: brake
  if(c.nitro>0&&seg===0&&ahead2===0&&curvAt(c.dist+c.speed+3)===0&&Math.random()<c.npro)return 'N';

    return seg===-1?'L':'R';
  }

}

function buildTrack(){
  const M=64,pts=[];
  for(let i=0;i<M;i++){
    const t=i/M*Math.PI*2;
    const r=1+0.42*Math.sin(2*t+1.2)+0.17*Math.sin(3*t+0.5);
    pts.push([Math.cos(t)*r,Math.sin(t)*r]);
  }
  const raw=[];
  for(let i=0;i<M;i++){
    const a1=Math.atan2(pts[(i+1)%M][1]-pts[i][1],pts[(i+1)%M][0]-pts[i][0]);
    const a2=Math.atan2(pts[(i+2)%M][1]-pts[(i+1)%M][1],pts[(i+2)%M][0]-pts[(i+1)%M][0]);
    let d=a2-a1;while(d>Math.PI)d-=2*Math.PI;while(d<-Math.PI)d+=2*Math.PI;
    raw.push(d);
  }
  const sm=raw.map((_,i)=>(raw[(i-1+M)%M]+raw[i]*2+raw[(i+1)%M])/4);
  let curv=sm.map(v=>v>0.11?1:v<-0.11?-1:0);
  // rotate so segment 0 sits mid-way through the longest straight
  let bestRun=0,bestStart=0,run=0;
  for(let i=0;i<M*2;i++){if(curv[i%M]===0){run++;if(run>bestRun){bestRun=run;bestStart=i-run+1;}}else run=0;}
  const off=((bestStart+Math.floor(bestRun/2))%M+M)%M;
  const rPts=[],rCurv=[];
  for(let i=0;i<M;i++){rPts.push(pts[(off+i)%M]);rCurv.push(curv[(off+i)%M]);}
  const ang=rPts.map((p,i)=>{const q=rPts[(i+1)%M];return Math.atan2(q[1]-p[1],q[0]-p[0]);});
  let minX=1e9,maxX=-1e9,minY=1e9,maxY=-1e9;
  rPts.forEach(p=>{minX=Math.min(minX,p[0]);maxX=Math.max(maxX,p[0]);minY=Math.min(minY,p[1]);maxY=Math.max(maxY,p[1]);});
  track={N:M,pts:rPts,curv:rCurv,ang:ang,bw:maxX-minX,bh:maxY-minY,cx:(minX+maxX)/2,cy:(minY+maxY)/2};
}

    }else{
      sp=Math.max(2,sp0-0.5*pow);

    else{sp=Math.min(sp0+2*pow,(seg?6.5:8)*pow);}

  car.lat+=latD;
  if(Math.abs(car.lat)>1){car.lat=Math.sign(car.lat)*0.85;sp=2;car.offs++;type='off';}

  car.speed=sp;

  if(newLap>oldLap){

}

  if(Math.abs(c.lat)>0.8&&seg!==0&&Math.sign(c.lat)===-seg)return 'B';
  if(c.nitro>0&&seg===0&&a2===0&&a3===0&&Math.random()<c.npro)return 'N';
  if(seg!==0){
    if(Math.random()<c.mistake)return Math.random()<0.5?'T':'B';
    if(c.speed<3.6&&Math.abs(c.lat)<0.65)return 'T';
    return seg===-1?'L':'R';
  }
  if(Math.abs(c.lat)>0.72)return c.lat>0?'R':'L';
  return 'T';
}

  // start/finish checkers
  const p0=scr(track.pts[0]),a=track.ang[0],hw=view.halfW,c=Math.max(3,hw/4);
  ctx.save();ctx.translate(p0[0],p0[1]);ctx.rotate(a);
  for(let r2=0;r2<4;r2++)for(let q=0;q<2;q++){ctx.fillStyle=(r2+q)%2?'#10201A':'#F2EAD8';ctx.fillRect(q*c-c,-hw+r2*(hw*2/4),c,hw*2/4);}
  ctx.restore();
}
function drawLookahead(now){
  const N=track.N,base=((Math.floor(player.dist)%N)+N)%N;
  // current segment tint
  const a1=scr(track.pts[base]),a2=scr(track.pts[(base+1)%N]);
  ctx.strokeStyle='rgba(255,210,63,.14)';ctx.lineWidth=Math.max(3,view.halfW*2-6);
  ctx.beginPath();ctx.moveTo(a1[0],a1[1]);ctx.lineTo(a2[0],a2[1]);ctx.stroke();
  // upcoming corner markers
  for(let j=1;j<=6;j++){
    const i=(base+j)%N;
    if(track.curv[i]!==0){
      const p=scr(track.pts[i]),pu=4.5+Math.sin(now/180+i*1.7)*1.6;
      ctx.fillStyle='rgba(255,122,26,.22)';ctx.beginPath();ctx.arc(p[0],p[1],pu+4,0,7);ctx.fill();
      ctx.strokeStyle='#FF7A1A';ctx.lineWidth=2;ctx.beginPath();ctx.arc(p[0],p[1],pu,0,7);ctx.stroke();
      ctx.fillStyle='#FF7A1A';ctx.beginPath();ctx.arc(p[0],p[1],1.8,0,7);ctx.fill();
    }
  }
}
function drawTrail(c){
  if(c.trail.length<1)return;
  const pts=c.trail.map(d=>{const N=track.N;let i=Math.floor(d)%N;if(i<0)i+=N;const f=d-Math.floor(d);const p=track.pts[i],q=track.pts[(i+1)%N];return[view.ox+(p[0]+(q[0]-p[0])*f)*view.s,view.oy+(p[1]+(q[1]-p[1])*f)*view.s];});
  const w=carWorld(c);pts.push([w.x,w.y]);
  for(let i=1;i<pts.length;i++){
    ctx.strokeStyle='rgba('+c.rgb+','+(i/pts.length*0.3).toFixed(3)+')';
    ctx.lineWidth=2;ctx.beginPath();ctx.moveTo(pts[i-1][0],pts[i-1][1]);ctx.lineTo(pts[i][0],pts[i][1]);ctx.stroke();
  }
}
function drawCar(c,now){
  const w=carWorld(c);
  if(c.finished){
    ctx.strokeStyle=c.color;ctx.globalAlpha=.5+.3*Math.sin(now/220);
    ctx.lineWidth=2;ctx.beginPath();ctx.arc(w.x,w.y,10+Math.sin(now/220)*2,0,7);ctx.stroke();ctx.globalAlpha=1;
  }
  ctx.save();ctx.translate(w.x,w.y);ctx.rotate(w.a);
  ctx.fillStyle=c.color;ctx.strokeStyle='#0C1713';ctx.lineWidth=2;
  ctx.beginPath();ctx.moveTo(8,0);ctx.lineTo(-5,5.2);ctx.lineTo(-2.5,0);ctx.lineTo(-5,-5.2);ctx.closePath();ctx.fill();ctx.stroke();
  if(c.isPlayer){ctx.strokeStyle='rgba(242,234,216,.85)';ctx.lineWidth=1.5;ctx.beginPath();ctx.arc(0,0,9,0,7);ctx.stroke();}
  ctx.restore();
  if(c.isPlayer){
    ctx.font='600 9px '+getComputedStyle(document.body).getPropertyValue('--mono');
    ctx.textAlign='center';ctx.fillStyle='rgba(242,234,216,.85)';
    ctx.fillText('YOU',w.x,w.y-14);
  }
}
/* ---- particles ---- */
function burst(x,y,n,colors,spd,life,g,add,size){
  for(let i=0;i<n;i++){
    const a=Math.random()*Math.PI*2,v=spd*(0.4+Math.random()*0.8);
    particles.push({x,y,vx:Math.cos(a)*v,vy:Math.sin(a)*v,g,life:life*(0.6+Math.random()*.7),tl:life,color:colors[i%colors.length],size:size*(0.6+Math.random()*.9),add});
  }
  if(particles.length>420)particles.splice(0,particles.length-420);
}
function spawnFx(c){
  const w=carWorld(c);
  if(c.fx==='apex')burst(w.x,w.y,16,['#FFD23F','#3ED598','#F2EAD8'],110,.55,60,true,2.6);
  else if(c.fx==='off'){burst(w.x,w.y,18,['#9AA8A0','#7d8a82','#B8C4BC'],55,.8,-14,false,4);shake=c.isPlayer?9:4;}
  else if(c.fx==='nitro')burst(w.x,w.y,12,['#FFD23F','#FF7A1A','#FF4B3E'],130,.4,30,true,3);
  else if(c.fx==='lap')burst(w.x,w.y,10,['#3ED598','#F2EAD8'],90,.5,20,true,2.4);
}
let lastNow=performance.now();
function stepParticles(now){
  const dt=Math.min(.05,(now-lastNow)/1000);lastNow=now;
  for(let i=particles.length-1;i>=0;i--){
    const p=particles[i];p.life-=dt;if(p.life<=0){particles.splice(i,1);continue;}
    p.x+=p.vx*dt;p.y+=p.vy*dt;p.vy+=p.g*dt;
    ctx.globalCompositeOperation=p.add?'lighter':'source-over';
    ctx.globalAlpha=clamp(p.life/p.tl,0,1);
    ctx.fillStyle=p.color;ctx.beginPath();ctx.arc(p.x,p.y,p.size,0,7);ctx.fill();
  }
  ctx.globalAlpha=1;ctx.globalCompositeOperation='source-over';
}
function draw(now){
  requestAnimationFrame(draw);
  if(state==='menu'&&!paused&&now>nextAuto){
    if(cars.every(c=>c.finished)){createCars();turn=0;}
    else doTurn(botAct(player)||'T');
  }
  const w=cv.clientWidth,h=cv.clientHeight;
  ctx.setTransform(dpr,0,0,dpr,0,0);
  ctx.clearRect(0,0,w,h);
  ctx.drawImage(bgCanvas,0,0,w,h);
  ctx.save();
  if(shake>0.4){ctx.translate((Math.random()-.5)*shake,(Math.random()-.5)*shake);shake*=0.86;}else shake=0;
  drawTrack();
  drawLookahead(now);
  const pr=clamp((now-tStart)/dur,0,1),e=easeOutCubic(pr);
  for(const c of cars){c.rdist=c.fromD+(c.dist-c.fromD)*e;c.rlat=c.fromL+(c.lat-c.fromL)*e;}
  for(const c of cars){
    if(c.fx&&!c.fxDone&&now-tStart>dur*0.45){spawnFx(c);c.fxDone=true;}
    if(c.fx==='nitro'&&now-tStart<dur*0.8){
      const w2=carWorld(c),bx=w2.x-Math.cos(w2.a)*12,by=w2.y-Math.sin(w2.a)*12;
      burst(bx,by,2,['#FFD23F','#FF7A1A'],60,.28,0,true,2.4);
    }
    if(now-tStart>dur+50)c.fx=null;
  }
  for(const c of cars)drawTrail(c);
  for(const c of cars)drawCar(c,now);
  stepParticles(now);
  ctx.restore();
}
