function raySeg(ox,oy,dx,dy,ax,ay,bx,by){
 const sx=bx-ax, sy=by-ay;
 const den = dx*sy - dy*sx;

 const t = ((ax-ox)*sy - (ay-oy)*sx)/den; // solve along ray

 return Infinity;
}

function raySeg(ox,oy,dx,dy,s){
  const sx=s[2]-s[0],sy=s[3]-s[1];
  const den=dx*sy-dy*sx;
  if(Math.abs(den)<1e-9)return Infinity;
  const t=((s[0]-ox)*sy-(s[1]-oy)*sx)/den;
  const u=((s[0]-ox)*dy-(s[1]-oy)*dx)/den;
  return (t>0.5&&u>=0&&u<=1)?t:Infinity;
}
function castRay(ox,oy,dx,dy,segs,maxD){
  let t=maxD;
  for(const s of segs){const tt=raySeg(ox,oy,dx,dy,s);if(tt<t)t=tt;}
  return t;
}
function visPoly(o,segs,a0,a1,maxD){
  const angs=[];
  if(a0===null){
    for(const s of segs)for(const p of [[s[0],s[1]],[s[2],s[3]]]){
      const a=Math.atan2(p[1]-o.y,p[0]-o.x);
      angs.push(a,a-0.0035,a+0.0035);
    }
    angs.sort((x,y)=>x-y);
  }else{
    const w=a1-a0;
    for(const s of segs)for(const p of [[s[0],s[1]],[s[2],s[3]]]){
      let rel=Math.atan2(p[1]-o.y,p[0]-o.x)-a0;
      while(rel<-Math.PI)rel+=TAU;while(rel>Math.PI)rel-=TAU;
      if(rel>=-1e-6&&rel<=w+1e-6)angs.push(rel);
    }
    angs.push(0,w);
    angs.sort((x,y)=>x-y);
  }

  for(const ra of angs){
    const ang=(a0===null)?ra:a0+ra;

    pts.push([o.x+dx*t,o.y+dy*t]);
  }
  return pts;
}
function pip(x,y,pts){

  }

}
function collideCircle(x,y,r,segs){

    const dx=s[2]-s[0],dy=s[3]-s[1];const L2=dx*dx+dy*dy||1e-9;

    const cx=s[0]+dx*u,cy=s[1]+dy*u;

  }

}

function isLit(x,y){
  const L=LEVELS[cur];
  const lights=[...L.lamps,{x:0,y:0,r:150,g:1}]; // placeholder replaced below
  lights.length=0;
  for(const l of L.lamps)lights.push({x:l.x,y:l.y,r:l.r});
  for(const d of drones)lights.push({x:d.x,y:d.y,r:d.r});
  for(const g of guards)lights.push({x:g.x,y:g.y,r:150});
  for(const l of lights){
    const dx=x-l.x,dy=y-l.y,d2=dx*dx+dy*dy;
    if(d2>l.r*l.r)continue;
    const d=Math.sqrt(d2)||1e-6;
    if(castRay(l.x,l.y,dx/d,dy/d,segs,d)>=d-4)return true;
  }
  return false;
}
function guardSees(g,px,py,lit){
  const dx=px-g.x,dy=py-g.y,d=Math.hypot(dx,dy);
  if(d>(lit?g.range:g.close))return false;
  const da=normAng(Math.atan2(dy,dx)-g.facing);
  if(Math.abs(da)>g.fov/2)return false;
  const dd=d||1e-6;
  return castRay(g.x,g.y,dx/dd,dy/dd,segs,dd)>=dd-9;
}
function moveAlong(o,dt){
  const wp=o.path[o.wp];
  const dx=wp[0]-o.x,dy=wp[1]-o.y,d=Math.hypot(dx,dy);
  const step=o.speed*dt;
  if(d<=step){o.x=wp[0];o.y=wp[1];o.wp=(o.wp+1)%o.path.length;o.pauseLeft=o.pause||0;}
  else{o.x+=dx/d*step;o.y+=dy/d*step;}
}
function simGuards(dt,t,play){
  for(const g of guards){
    if(!(g.pauseLeft>0)){moveAlong(g,dt);if(g.pauseLeft>0)g.pauseLeft-=dt;else g.pauseLeft=0;}
    else g.pauseLeft-=dt;
    let vis=false;
    if(play){vis=guardSees(g,player.x,player.y,litNow);}
    if(g.alert>0.32&&vis){g.staring=true;g.facing=angLerp(g.facing,Math.atan2(player.y-g.y,player.x-g.x),1-Math.exp(-6*dt));}
    else{
      g.staring=false;
      const wp=g.path[g.wp];
      g.facing=angLerp(g.facing,Math.atan2(wp[1]-g.y,wp[0]-g.x),1-Math.exp(-3*dt));
    }
    if(play){
      if(vis){
        const rate=litNow?(sneak?0.55:1.0):(sneak?0.32:0.5);
        g.alert+=rate*dt;
        if(g.alert>0.3&&g.spotT<=0){g.spotT=1.4;sfx.sting();}
      }else{g.alert-=0.5*dt;if(g.alert<0)g.alert=0;}
      if(g.spotT>0)g.spotT-=dt*0.9;
    }
  }
}
function simDrones(dt){for(const d of drones)moveAlong(d,dt);}

  }
}

const keys={};
function update(dt,t){
  if(state!=='caught')simGuards(dt,t,state==='play');
  simDrones(dt);simMotes(dt,t);
  for(const p of parts){p.t-=dt;p.x+=p.vx*dt;p.y+=p.vy*dt;}
  parts=parts.filter(p=>p.t>0);
  for(const f of floats)f.t-=dt;
  floats=floats.filter(f=>f.t>0);
  shake*=Math.exp(-5*dt);redFlash*=Math.exp(-3.2*dt);

  if(state==='play'){
    let ix=(keys.KeyD||keys.ArrowRight?1:0)-(keys.KeyA||keys.ArrowLeft?1:0);
    let iy=(keys.KeyS||keys.ArrowDown?1:0)-(keys.KeyW||keys.ArrowUp?1:0);
    const il=Math.hypot(ix,iy);if(il>0){ix/=il;iy/=il;}
    const sp=sneak?88:170;
    const k=1-Math.exp(-10*dt);
    player.vx=lerp(player.vx,ix*sp,k);player.vy=lerp(player.vy,iy*sp,k);
    [player.x,player.y]=collideCircle(player.x+player.vx*dt,player.y+player.vy*dt,player.r,segs);
    const spd=Math.hypot(player.vx,player.vy);
    if(spd>12)player.facing=angLerp(player.facing,Math.atan2(player.vy,player.vx),1-Math.exp(-12*dt));

    stepAcc+=spd*dt;
    if(stepAcc>36){stepAcc=0;if(spd>25)sfx.step(sneak);}
    if(sneak&&spd>30){dustT-=dt;if(dustT<=0){dustT=.12;parts.push({x:player.x-ix*10,y:player.y+6,vx:-ix*8,vy:-4,t:.5,max:.5});}}

    litNow=isLit(player.x,player.y);
    if(prevLit!==null&&prevLit!==litNow)sfx.whoosh(sneak);
    prevLit=litNow;

    let maxA=0;
    for(const g of guards){maxA=Math.max(maxA,g.alert);if(g.alert>=1){doCaught();break;}}
    if(state==='play'){
      // heartbeat
      if(maxA>0.22){const iv=lerp(1.15,.42,maxA);if(t-lastBeat>iv){lastBeat=t;sfx.beat(maxA);}}

      const L=LEVELS[cur];
      if(!itemTaken&&Math.hypot(player.x-L.item[0],player.y-L.item[1])<28){
        itemTaken=true;exitActive=true;setObjective();sfx.pickup();
        floats.push({x:L.item[0],y:L.item[1]-14,t:1.2,max:1.2,txt:'◆ ARTIFACT SECURED',c:'#ffd977'});
      }
      if(itemTaken&&Math.hypot(player.x-L.exit[0],player.y-L.exit[1])<34)levelClear();

      time+=dt*1000;
    }
    // HUD
    const total=times.reduce((a,b)=>a+b,0)+time;
    el('clock').textContent=fmt(total);
    el('susp').firstElementChild.style.width=(clamp(maxA,0,1)*100)+'%';
    const chip=el('chip');let cls='hidden',txt='HIDDEN';
    if(state==='play'){
      if(maxA>0.32){cls='panic';txt='DETECTED';}
      else if(litNow){cls='exposed';txt='EXPOSED';}
    }
    if(chip.dataset.s!==cls+txt){chip.dataset.s=cls+txt;chip.className=cls;chip.textContent=txt;}
  }
  else if(state==='caught'){
    caughtT+=dt;
    if(caughtT>0.7&&!el('caught').classList.contains('shown')){el('caught').dataset.shown='1';showScreen('caught');}
    else if(caughtT<=0.7)delete el('caught').dataset.shown;
  }
  else if(state==='clear'){
    clearT+=dt;
    if(clearT>1.6){
      time=0;
      if(cur+1<LEVELS.length){resetLevel(cur+1);state='play';banner('0'+(cur+1),LEVELS[cur].name,LEVELS[cur].hint,'amber');sfx.intro();}
      else showWon();
    }
  }
}

    ctx.beginPath();ctx.arc(player.x,player.y,9,0,TAU);ctx.fillStyle='rgba(185,196,222,.5)';ctx.fill();
  }

  /* drones */
  for(const d of drones){
    ctx.save();ctx.translate(d.x,d.y-6);
    ctx.beginPath();
    for(let i=0;i<6;i++){const a=i/6*TAU+t*.4;const px=Math.cos(a)*10,py=Math.sin(a)*10;i?ctx.lineTo(px,py):ctx.moveTo(px,py);}
    ctx.closePath();ctx.fillStyle='#182741';ctx.fill();
    ctx.strokeStyle='#6fa8dd';ctx.lineWidth=1.5;ctx.stroke();
    ctx.beginPath();ctx.arc(0,0,3,0,TAU);ctx.fillStyle='#dff1ff';ctx.fill();
    ctx.restore();
  }

  /* dust puffs */
  for(const p of parts){
    ctx.globalAlpha=(p.t/p.max)*.35;
    ctx.beginPath();ctx.arc(p.x,p.y,2+(1-p.t/p.max)*3,0,TAU);ctx.fillStyle='#9fb0d6';ctx.fill();
  }
  ctx.globalAlpha=1;

  /* ---------- darkness ---------- */
  dctx.setTransform(1,0,0,1,0,0);
  dctx.globalCompositeOperation='source-over';
  dctx.clearRect(0,0,960,600);
  dctx.globalAlpha=.94;dctx.fillStyle='#080d1a';dctx.fillRect(0,0,960,600);dctx.globalAlpha=1;
  dctx.globalCompositeOperation='destination-out';
  for(const l of lightPolys){
    const g=dctx.createRadialGradient(l.x,l.y,0,l.x,l.y,l.r);
    g.addColorStop(0,'rgba(255,255,255,'+l.s+')');
    g.addColorStop(.6,'rgba(255,255,255,'+(l.s*.8)+')');
    g.addColorStop(1,'rgba(255,255,255,0)');
    dctx.fillStyle=g;trace(dctx,l.poly);dctx.fill();
  }
  dctx.globalCompositeOperation='source-over';
  ctx.drawImage(darkC,0,0,960,600);

  /* ---------- light tints ---------- */
  ctx.save();ctx.globalCompositeOperation='lighter';
  for(const l of lightPolys){
    if(l.kind==='lamp'){
      const g=ctx.createRadialGradient(l.x,l.y,0,l.x,l.y,l.r);
      g.addColorStop(0,'rgba(255,178,84,.22)');g.addColorStop(1,'rgba(255,178,84,0)');
      ctx.fillStyle=g;trace(ctx,l.poly);ctx.fill();
    }else if(l.kind==='drone'){
      const g=ctx.createRadialGradient(l.x,l.y,0,l.x,l.y,l.r);
      g.addColorStop(0,'rgba(130,195,255,.18)');g.addColorStop(1,'rgba(130,195,255,0)');
      ctx.fillStyle=g;trace(ctx,l.poly);ctx.fill();
    }else{
      const a=clamp(l.alert||0,0,1);
      const g=ctx.createRadialGradient(l.x,l.y,0,l.x,l.y,l.r*1.8);
      const cr=Math.round(255),cg=Math.round(232-140*a),cb=Math.round(160-120*a);
      g.addColorStop(0,'rgba('+cr+','+cg+','+cb+','+(0.10+0.20*a)+')');g.addColorStop(1,'rgba('+cr+','+cg+','+cb+',0)');
      ctx.fillStyle=g;trace(ctx,l.poly);ctx.fill();
      ctx.strokeStyle='rgba('+cr+','+cg+','+cb+','+(0.14+0.2*a)+')';ctx.lineWidth=1.5;ctx.stroke();
    }
  }
  ctx.restore();

  /* motes */
  for(const m of motes){
    let litF=0;
    for(const l of lightPolys){if(l.kind==='guard')continue;if(Math.hypot(m.x-l.x,m.y-l.y)>l.r)continue;if(pip(m.x,m.y,l.poly)){litF=1;break;}}
    if(!litF)continue;
    ctx.globalAlpha=.22*litF*(.6+.4*Math.sin(t*2+m.ph));
    ctx.fillStyle='#fff3dc';ctx.fillRect(m.x,m.y,1.7,1.7);
  }
  ctx.globalAlpha=1;

  /* lamp cores */
  for(const l of L.lamps){
    const fl=.85+.15*Math.sin(t*11+l.x);
    ctx.beginPath();ctx.arc(l.x,l.y,3.5,0,TAU);ctx.fillStyle='rgba(255,243,220,'+fl+')';ctx.fill();
  }
  /* drone rotor ring */
  for(const d of drones){
    ctx.setLineDash([4,6]);ctx.lineDashOffset=-t*30;
    ctx.strokeStyle='rgba(143,211,255,.6)';ctx.lineWidth=1.5;
    ctx.beginPath();ctx.arc(d.x,d.y-6,14,0,TAU);ctx.stroke();
    ctx.setLineDash([]);ctx.lineDashOffset=0;
  }

  /* item glow + sparkle */
  if(!itemTaken){
    const [ix,iy]=L.item,bob=Math.sin(t*3)*2,puls=.5+.5*Math.sin(t*4);
    const g=ctx.createRadialGradient(ix,iy+bob,0,ix,iy+bob,36);
    g.addColorStop(0,'rgba(255,217,119,'+(0.35+0.2*puls)+')');g.addColorStop(1,'rgba(255,217,119,0)');
    ctx.fillStyle=g;ctx.beginPath();ctx.arc(ix,iy+bob,36,0,TAU);ctx.fill();
    ctx.save();ctx.translate(ix,iy+bob);ctx.rotate(t*1.2);
    ctx.strokeStyle='rgba(255,244,214,.7)';ctx.lineWidth=1.5;
    ctx.beginPath();ctx.moveTo(-16,0);ctx.lineTo(16,0);ctx.moveTo(0,-16);ctx.lineTo(0,16);ctx.stroke();
    ctx.restore();
  }

  /* exit overlay */
  {
    const [ex,ey]=L.exit;
    if(exitActive){
      const pr=34+5*Math.sin(t*3);
      ctx.strokeStyle='rgba(111,242,200,'+(0.4+0.25*Math.sin(t*3))+')';ctx.lineWidth=2;
      ctx.beginPath();ctx.arc(ex,ey,pr,0,TAU);ctx.stroke();
      const g=ctx.createRadialGradient(ex,ey,0,ex,ey,46);
      g.addColorStop(0,'rgba(111,242,200,.28)');g.addColorStop(1,'rgba(111,242,200,0)');
      ctx.fillStyle=g;ctx.beginPath();ctx.arc(ex,ey,46,0,TAU);ctx.fill();
    }
    ctx.font='700 9px '+FONT_M;ctx.textAlign='center';
    ctx.fillStyle=exitActive?'rgba(111,242,200,.9)':'rgba(159,176,214,.5)';
    ctx.fillText(exitActive?'EXTRACTION':'EXIT',ex,ey-40);
  }

  /* guard alert arcs + threat rim + ! pops */
  for(const g of guards){
    const a=clamp(g.alert,0,1);
    if(a>0.02){
      ctx.strokeStyle='rgba(255,255,255,.12)';ctx.lineWidth=3;
      ctx.beginPath();ctx.arc(g.x,g.y,19,0,TAU);ctx.stroke();
      ctx.strokeStyle=a<0.5?'#ffd977':'#ff5d6c';
      ctx.beginPath();ctx.arc(g.x,g.y,19,-Math.PI/2,-Math.PI/2+a*TAU);ctx.stroke();
    }
    ctx.setLineDash([3,5]);ctx.lineDashOffset=-t*10;
    ctx.strokeStyle='rgba(255,93,108,'+(0.3+0.4*a)+')';ctx.lineWidth=1.5;
    ctx.beginPath();ctx.arc(g.x,g.y,16,0,TAU);ctx.stroke();
    ctx.setLineDash([]);ctx.lineDashOffset=0;
    if(g.spotT>0){
      const sa=Math.min(1,g.spotT);
      ctx.globalAlpha=sa;
      ctx.font='24px '+FONT_D;ctx.textAlign='center';
      const yy=g.y-28-(1-sa)*6;
      ctx.strokeStyle='#0d1425';ctx.lineWidth=5;ctx.strokeText('!',g.x,yy);
      ctx.fillStyle='#ff5d6c';ctx.fillText('!',g.x,yy);
      ctx.globalAlpha=1;
    }
  }

  /* player beacon */
  if(state!=='menu'){
    const g=ctx.createRadialGradient(player.x,player.y,0,player.x,player.y,30);
    g.addColorStop(0,'rgba(191,239,255,.18)');g.addColorStop(1,'rgba(191,239,255,0)');
    ctx.fillStyle=g;ctx.beginPath();ctx.arc(player.x,player.y,30,0,TAU);ctx.fill();
    ctx.beginPath();ctx.arc(player.x,player.y,2,0,TAU);ctx.fillStyle='rgba(214,246,255,.9)';ctx.fill();
  }

  /* floaters */
  for(const f of floats){
    const k=f.t/f.max;
    ctx.globalAlpha=k;ctx.font='700 11px '+FONT_M;ctx.textAlign='center';
    ctx.fillStyle=f.c;ctx.fillText(f.txt,f.x,f.y-(1-k)*26);
  }
  ctx.globalAlpha=1;

  /* vignette + grain + frame */
  ctx.fillStyle=vigGrad;ctx.fillRect(0,0,960,600);
  ctx.save();ctx.globalCompositeOperation='overlay';ctx.globalAlpha=.05;
  ctx.fillStyle=grains[Math.floor(t*12)%3];ctx.fillRect(0,0,960,600);ctx.restore();
  ctx.strokeStyle='rgba(255,180,84,.18)';ctx.lineWidth=2;ctx.strokeRect(10,10,940,580);
  ctx.strokeStyle='rgba(255,180,84,.6)';ctx.lineWidth=3;
  const T2=[[10,10,1,1],[950,10,-1,1],[10,590,1,-1],[950,590,-1,-1]];
  for(const [x,y,sx2,sy2] of T2){ctx.beginPath();ctx.moveTo(x+sx2*16,y);ctx.lineTo(x,y);ctx.lineTo(x,y+sy2*16);ctx.stroke();}

  /* red flash */
  if(redFlash>0.01){
    ctx.setTransform(dpr,0,0,dpr,0,0);
    ctx.fillStyle='rgba(255,60,70,'+(redFlash*.35)+')';ctx.fillRect(0,0,W,H);
  }
}
