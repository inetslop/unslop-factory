function isWall(c,r){return c<0||r<0||c>=cols||r>=rows||walls.has(key(c,r));}
function crateAt(c,r){for(const cr of crates)if(!cr.sunk&&cr.c===c&&cr.r===r)return cr;return null;}

function loadHole(i){
  holeIdx=i;strokes=0;particles.length=0;popups.length=0;completeT=-1;pendingDir=null;introT=1.8;shake=0;deadShown=false;
  walls=new Set();waters=new Set();sands=new Set();greens=new Set();targets=[];crates=[];player=null;grid=[];
  const map=LEVELS[i].map;rows=map.length;cols=Math.max.apply(null,map.map(r=>r.length));
  for(let r=0;r<rows;r++){grid[r]=[];
    for(let c=0;c<cols;c++){
      const ch=(map[r]&&map[r][c])||'.';let t=0;
      if(ch==='#')t=1;else if(ch==='W')t=2;else if(ch==='S')t=3;else if(ch==='g'||ch==='T')t=4;
      grid[r][c]=t;
      if(t===1)walls.add(key(c,r));if(t===2)waters.add(key(c,r));if(t===3)sands.add(key(c,r));if(t===4)greens.add(key(c,r));
      if(ch==='T')targets.push({c,r,cupped:false,lift:1});
      if(ch==='B')crates.push({c,r,x:0,y:0,startC:c,startR:r,stuck:false,sunk:0,pop:0});
      if(ch==='P')player={c,r,x:0,y:0,dir:[1,0],phase:9};
    }}
  resize();
  crates.forEach(cr=>{cr.x=ox+(cr.c+.5)*tile;cr.y=oy+(cr.r+.5)*tile;});
  player.x=ox+(player.c+.5)*tile;player.y=oy+(player.r+.5)*tile;
  updateHUD(true);updateDots();
}

function respawnSpot(cr){
  const free=(c,r)=>c>=0&&r>=0&&c<cols&&r<rows&&!walls.has(key(c,r))&&!waters.has(key(c,r))
    &&!crates.some(o=>o!==cr&&!o.sunk&&o.c===c&&o.r===r);
  if(free(cr.startC,cr.startR))return[cr.startC,cr.startR];
  for(let d=1;d<8;d++)for(let dr=-d;dr<=d;dr++)for(let dc=-d;dc<=d;dc++){
    if(Math.max(Math.abs(dc),Math.abs(dr))!==d)continue;
    const c=cr.startC+dc,r=cr.startR+dr;if(free(c,r))return[c,r];}
  return[cr.startC,cr.startR];
}

/* ============================= MOVEMENT =========================== */
function doMove(dc,dr){
  if(mode!=='play'||completeT>=0)return;
  player.dir=[dc,dr];deadShown=false;
  const nc=player.c+dc,nr=player.r+dr;
  if(isWall(nc,nr)){sfx.bump();shake=Math.max(shake,2);return;}
  const cr=crateAt(nc,nr);
  if(cr){
    if(cr.stuck){strokes++;sfx.sand();popup('SAND +1',cr.x,cr.y-tile*.3,'#ffe9a8',tile*.36);puff(cr.x,cr.y,tile*.42);shake=2;updateHUD();return;}
    const cc=nc+dc,rr=nr+dr;
    if(isWall(cc,rr)||crateAt(cc,rr)){sfx.bump();shake=Math.max(shake,2);return;}
    if(grid[rr][cc]===2){ // pond hazard
      sfx.splash();strokes+=2;shake=6;cr.sunk=.55;
      popup('WATER +2',ox+(nc+.5)*tile,oy+(nr+.5)*tile-tile*.3,'#ff9d8a',tile*.44);
      splashFx(ox+(nc+.5)*tile,oy+(nr+.5)*tile,tile);updateHUD();return;
    }
    cr.c=cc;cr.r=rr;strokes++;sfx.push();shake=Math.max(shake,3);
    popup('+1',ox+(cc+.5)*tile,oy+(rr+.5)*tile-tile*.28,'#ffe9a8',tile*.34);
    if(grid[rr][cc]===3)cr.stuck=true;
  }
  const px=player.x,py=player.y;
  player.c=nc;player.r=nr;player.phase=0;
  sfx.walk();grassFx(px,py);updateHUD();
}

/* ============================== UPDATE ============================ */
function update(dt){
  time+=dt;shake=Math.max(0,shake-dt*30);
  if(introT>0)introT-=dt;
  // player
  const txp=ox+(player.c+.5)*tile,typ=oy+(player.r+.5)*tile,f=Math.min(1,dt*16);
  player.x+=(txp-player.x)*f;player.y+=(typ-player.y)*f;
  if(Math.abs(txp-player.x)<.4&&Math.abs(typ-player.y)<.4){player.x=txp;player.y=typ;}else player.phase+=dt*13;
  if(pendingDir&&settledPlayer()){const d=pendingDir;pendingDir=null;doMove(d[0],d[1]);}
  // crates
  for(const cr of crates){
    if(cr.sunk>0){cr.sunk-=dt;if(cr.sunk<=0){const s=respawnSpot(cr);cr.c=s[0];cr.r=s[1];cr.pop=.4;cr.stuck=grid[s[1]][s[0]]===3;sfx.pop();}}
    if(cr.pop>0)cr.pop-=dt;
    const tx=ox+(cr.c+.5)*tile,ty=oy+(cr.r+.5)*tile,gf=Math.min(1,dt*13);
    cr.x+=(tx-cr.x)*gf;cr.y+=(ty-cr.y)*gf;
    if(Math.abs(tx-cr.x)<.4&&Math.abs(ty-cr.y)<.4){cr.x=tx;cr.y=ty;}
  }
  // cupping + flags
  for(const t of targets){
    const occ=crateAt(t.c,t.r)&&settled(crateAt(t.c,t.r));
    if(occ&&!t.cupped){t.cupped=true;sfx.cup();sparkle(ox+(t.c+.5)*tile,oy+(t.r+.5)*tile);}
    if(!occ)t.cupped=false;
    const tgt=t.cupped?.14:1;t.lift+=(tgt-t.lift)*Math.min(1,dt*6);
  }
  // win
  if(mode==='play'&&completeT<0){
    const all=crates.every(cr=>cr.sunk<=0&&settled(cr))&&targets.every(t=>crateAt(t.c,t.r));
    if(all)completeT=.95;
    else{ // deadlock hint
      for(const cr of crates){
        const onT=targets.some(t=>t.c===cr.c&&t.r===cr.r);
        if(!onT&&cr.sunk<=0&&settled(cr)){
          const u=isWall(cr.c,cr.r-1),d2=isWall(cr.c,cr.r+1),l=isWall(cr.c-1,cr.r),r2=isWall(cr.c+1,cr.r);
          if((u&&l)||(u&&r2)||(d2&&l)||(d2&&r2)&&!deadShown){deadShown=true;toast('Cornered! Press R to replay the hole.');}
        }}
    }
  }
  if(completeT>=0){completeT-=dt;if(completeT<0)completeHole();}
  // particles / popups
  for(let i=particles.length-1;i>=0;i--){const p=particles[i];p.t-=dt;
    p.vy+=p.g*dt;p.x+=p.vx*dt;p.y+=p.vy*dt;if(p.rot!==undefined)p.rot+=p.vr*dt;
    if(p.t<=0)particles.splice(i,1);}
  for(let i=popups.length-1;i>=0;i--){const p=popups[i];p.t-=dt;p.y+=p.vy*dt;if(p.t<=0)popups.splice(i,1);}
}

    ctx.beginPath();ctx.arc(x+tile*(.2+.16*jx+(i%2)*.24),y+tile*(.28+.2*jy),tile*.3,0,7);ctx.fill();}
  ctx.fillStyle='rgba(90,170,110,.5)';
  for(let i=0;i<3;i++){const jx=hash(c*9+i,r),jy=hash(r*9+i,c);
    ctx.beginPath();ctx.arc(x+tile*(.28+.4*jx),y+tile*(.16+.2*jy),tile*.1,0,7);ctx.fill();}
}
function drawTree(c,r){
  const x=ox+c*tile,y=oy+r*tile;
  ctx.fillStyle='rgba(10,30,16,.3)';ctx.beginPath();ctx.ellipse(x+tile/2,y+tile*.92,tile*.44,tile*.15,0,0,7);ctx.fill();
  ctx.fillStyle='#6b4a2a';ctx.fillRect(x+tile*.42,y+tile*.5,tile*.16,tile*.42);
  ctx.fillStyle='#27663a';ctx.beginPath();ctx.arc(x+tile/2,y+tile*.42,tile*.4,0,7);ctx.fill();
  ctx.fillStyle='#2f7a44';ctx.beginPath();ctx.arc(x+tile*.38,y+tile*.36,tile*.28,0,7);ctx.arc(x+tile*.64,y+tile*.4,tile*.25,0,7);ctx.fill();
  ctx.fillStyle='rgba(96,178,118,.6)';ctx.beginPath();ctx.arc(x+tile*.4,y+tile*.3,tile*.11,0,7);ctx.fill();
}

function drawCrate(cr){
  if(cr.sunk>0)return;
  const sunken=cr.stuck?tile*.08:0;
  const x=cr.x,y=cr.y+sunken,s=Math.max(.05,tile*.76);
  let sc=1;if(cr.pop>0){const p=1-cr.pop/.4;sc=1+.35*Math.sin(p*Math.PI)*(1-p)+.08;}
  ctx.save();ctx.translate(x,y);ctx.scale(sc,sc);
  ctx.fillStyle='rgba(10,30,16,.28)';ctx.beginPath();ctx.ellipse(0,s*.46,s*.42,s*.14,0,0,7);ctx.fill();
  const cupped=targets.some(t=>t.cupped&&t.c===cr.c&&t.r===cr.r);
  if(cupped){ctx.shadowColor='rgba(231,200,119,.95)';ctx.shadowBlur=tile*.4;}
  rr(-s/2,-s/2,s,s,s*.16);ctx.fillStyle='#b5803f';ctx.fill();ctx.shadowBlur=0;
  ctx.strokeStyle=cupped?'#e7c877':'#7a5226';ctx.lineWidth=Math.max(2,tile*.07);rr(-s/2,-s/2,s,s,s*.16);ctx.stroke();
  ctx.strokeStyle='rgba(122,82,38,.85)';ctx.lineWidth=Math.max(1.5,tile*.045);
  ctx.beginPath();ctx.moveTo(-s*.3,-s*.3);ctx.lineTo(s*.3,s*.3);ctx.moveTo(s*.3,-s*.3);ctx.lineTo(-s*.3,s*.3);ctx.stroke();
  ctx.strokeStyle='rgba(255,235,190,.4)';ctx.lineWidth=Math.max(1.5,tile*.05);
  ctx.beginPath();ctx.moveTo(-s*.34,-s*.36);ctx.lineTo(s*.34,-s*.36);ctx.stroke();
  if(cr.stuck){ctx.fillStyle='rgba(233,215,149,.8)';
    ctx.beginPath();ctx.ellipse(0,s*.5,s*.55,s*.16,0,0,7);ctx.fill();}
  ctx.restore();
}

function drawPlayer(){
  const x=player.x,y=player.y,moving=Math.abs(player.x-(ox+(player.c+.5)*tile))>2||Math.abs(player.y-(oy+(player.r+.5)*tile))>2;
  const bob=moving?Math.sin(player.phase*2)*tile*.04:0;
  const dx=player.dir[0],dy=player.dir[1];
  ctx.save();ctx.translate(x,y+bob);
  ctx.fillStyle='rgba(10,30,16,.28)';ctx.beginPath();ctx.ellipse(0,tile*.4-bob,tile*.3,tile*.11,0,0,7);ctx.fill();
  const sw=moving?Math.sin(player.phase*2)*tile*.09:0;
  ctx.fillStyle='#5d4a2e';
  ctx.fillRect(-tile*.16+sw,tile*.18,tile*.13,tile*.2);
  ctx.fillRect(tile*.03-sw,tile*.18,tile*.13,tile*.2);
  rr(-tile*.24,-tile*.14,tile*.48,tile*.38,tile*.12);ctx.fillStyle='#f7f3e8';ctx.fill();
  ctx.strokeStyle='#c9bfa4';ctx.lineWidth=1.5;rr(-tile*.24,-tile*.14,tile*.48,tile*.38,tile*.12);ctx.stroke();
  ctx.strokeStyle='#c0392b';ctx.lineWidth=Math.max(1.5,tile*.04);
  ctx.beginPath();ctx.moveTo(-tile*.24,0);ctx.lineTo(tile*.24,0);ctx.stroke();
  const hy=-tile*.26;
  ctx.fillStyle='#f1c9a0';ctx.beginPath();ctx.arc(0,hy,tile*.17,0,7);ctx.fill();
  ctx.fillStyle='#d0452f';ctx.beginPath();ctx.arc(0,hy-tile*.02,tile*.17,Math.PI,0);ctx.fill();
  rr(dx*tile*.1- tile*.16,hy-tile*.07+dy*tile*.02,tile*.32,tile*.09,tile*.04);ctx.fillStyle='#b53423';ctx.fill();
  ctx.fillStyle='#2b2418';
  const eo=dx*tile*.05,eyo=dy*tile*.03;
  ctx.beginPath();ctx.arc(-tile*.06+eo,hy+tile*.02+eyo,tile*.022,0,7);ctx.arc(tile*.06+eo,hy+tile*.02+eyo,tile*.022,0,7);ctx.fill();
  ctx.restore();
}

function drawFlag(t){
  const x=ox+(t.c+.5)*tile,y=oy+(t.r+.5)*tile;
  const ph=tile*1.12*t.lift,w=Math.sin(time*3.4+t.c+t.r)*tile*.07*(1-t.lift*.8);
  ctx.strokeStyle='rgba(60,50,20,.3)';ctx.lineWidth=2;ctx.beginPath();ctx.moveTo(x,y+2);ctx.lineTo(x+w*.3,y-ph-2);ctx.stroke();
  ctx.strokeStyle='#efe9d6';ctx.lineWidth=Math.max(1.5,tile*.045);
  ctx.beginPath();ctx.moveTo(x,y);ctx.lineTo(x+w,y-ph);ctx.stroke();
  if(t.lift>.2){
    const fy=y-ph;
    ctx.fillStyle='#d0452f';ctx.strokeStyle='#8c2a1c';ctx.lineWidth=1.5;
    ctx.beginPath();ctx.moveTo(x+w,fy);
    ctx.quadraticCurveTo(x+w+tile*.34,fy-tile*.06+w,x+w+tile*.55,fy+tile*.03+w);
    ctx.lineTo(x+w,fy+tile*.16);ctx.closePath();ctx.fill();ctx.stroke();
  }
}

function render(){
  ctx.clearRect(0,0,W,H);
  const g=ctx.createLinearGradient(0,0,0,H);g.addColorStop(0,'#17402a');g.addColorStop(1,'#0d2314');
  ctx.fillStyle=g;ctx.fillRect(0,0,W,H);
  if(!cols)return;
  const sx=(Math.random()*2-1)*shake,sy=(Math.random()*2-1)*shake;
  ctx.save();ctx.translate(sx,sy);
  // course mat
  rr(ox-tile*.55,oy-tile*.55,cols*tile+tile*1.1,rows*tile+tile*1.1,tile*.4);
  ctx.fillStyle='rgba(8,24,13,.6)';ctx.fill();
  ctx.strokeStyle='rgba(201,162,74,.5)';ctx.lineWidth=2;rr(ox-tile*.55,oy-tile*.55,cols*tile+tile*1.1,rows*tile+tile*1.1,tile*.4);ctx.stroke();
  drawGrounds();
  // depth-sorted entities + hedges
  const items=[];
  for(let r=0;r<rows;r++)for(let c=0;c<cols;c++)if(walls.has(key(c,r)))items.push({r,f:hash(c,r),k:'w',c,r});
  crates.forEach(cr=>items.push({r:cr.r+.5,f:1,k:'b',o:cr}));
  items.push({r:player.r+.6,f:1,k:'p'});
  items.sort((a,b)=>a.r-b.r||a.f-b.f);
  for(const it of items){
    if(it.k==='w'){if(hash(it.c,it.r)<.2)drawTree(it.c,it.r);else drawHedge(it.c,it.r);}
    else if(it.k==='b')drawCrate(it.o);
    else drawPlayer();
  }
  targets.forEach(drawFlag);
  // particles
  for(const p of particles){
    ctx.globalAlpha=Math.max(0,Math.min(1,p.t/(p.life*.6)));
    if(p.type==='c'){ctx.save();ctx.translate(p.x,p.y);ctx.rotate(p.rot);ctx.fillStyle=p.c;ctx.fillRect(-p.s/2,-p.s/3,p.s,p.s*.66);ctx.restore();}
    else{ctx.fillStyle=p.c;ctx.beginPath();ctx.arc(p.x,p.y,p.s,0,7);ctx.fill();}
  }
  ctx.globalAlpha=1;
  // popups
  for(const p of popups){
    ctx.globalAlpha=Math.max(0,Math.min(1,p.t/p.life*1.6));
    ctx.font='700 '+p.size+'px Georgia, serif';ctx.textAlign='center';
    ctx.lineWidth=4;ctx.strokeStyle='rgba(30,25,10,.8)';ctx.strokeText(p.text,p.x,p.y);
    ctx.fillStyle=p.color;ctx.fillText(p.text,p.x,p.y);
  }
  ctx.globalAlpha=1;
  // hole intro
  if(introT>0){
    const a=introT<.4?introT/.4:(introT>1.4?(1.8-introT)/.4:1);
    ctx.globalAlpha=Math.max(0,Math.min(1,a));
    const cx=ox+cols*tile/2,cy=oy+rows*tile/2-tile*.3;
    ctx.textAlign='center';
    ctx.font='700 '+(tile*.62)+'px Georgia, serif';
    ctx.lineWidth=5;ctx.strokeStyle='rgba(20,40,25,.85)';
    const h1='HOLE '+(holeIdx+1);ctx.strokeText(h1,cx,cy);ctx.fillStyle='#e7c877';ctx.fillText(h1,cx,cy);
    ctx.font='italic '+(tile*.3)+'px Georgia, serif';
    ctx.lineWidth=4;ctx.strokeText(LEVELS[holeIdx].name,cx,cy+tile*.42);ctx.fillStyle='#f6efdb';ctx.fillText(LEVELS[holeIdx].name,cx,cy+tile*.42);
    ctx.font='700 '+(tile*.22)+'px Georgia, serif';
    ctx.strokeText('PAR '+LEVELS[holeIdx].par,cx,cy+tile*.72);ctx.fillStyle='#b9d3ae';ctx.fillText('PAR '+LEVELS[holeIdx].par,cx,cy+tile*.72);
    ctx.globalAlpha=1;
  }
  // drifting cloud shadow
  const cx2=((time*26)%(W+900))-450;
  const cg=ctx.createRadialGradient(cx2,H*.3,50,cx2,H*.3,W*.4);
  cg.addColorStop(0,'rgba(12,30,18,.1)');cg.addColorStop(1,'rgba(12,30,18,0)');
  ctx.fillStyle=cg;ctx.fillRect(0,0,W,H);
  ctx.restore();
}
