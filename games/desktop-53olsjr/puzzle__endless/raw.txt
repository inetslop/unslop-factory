 {c:[[0,0],[1,0],[2,0],[0,1],[1,1],[2,1],[0,2],[1,2],[2,2]],w:1.2}
];
function norm(cells){
  var mx=99,my=99;
  cells.forEach(function(p){mx=Math.min(mx,p[0]);my=Math.min(my,p[1]);});
  return cells.map(function(p){return [p[0]-mx,p[1]-my];})
              .sort(function(a,b){return a[1]-b[1]||a[0]-b[0];});
}
function rotCells(cells){
  var h=0; cells.forEach(function(p){h=Math.max(h,p[1]+1);});
  return norm(cells.map(function(p){return [h-1-p[1],p[0]];}));
}
function dims(cells){
  var w=0,h=0;
  cells.forEach(function(p){w=Math.max(w,p[0]+1);h=Math.max(h,p[1]+1);});
  return [w,h];
}
function pickShape(){
  var big=1+Math.min(2.5,lines/30), mid=1+Math.min(1.2,lines/80), tot=0;
  var ws=SHAPES.map(function(s){
    var w=s.w, n=s.c.length;
    if(n>=9)w*=big; else if(n>=4)w*=mid;
    tot+=w; return w;
  });
  var r=Math.random()*tot;
  for(var i=0;i<ws.length;i++){ r-=ws[i]; if(r<=0) return SHAPES[i].c.map(function(p){return [p[0],p[1]];}); }
  return [[0,0]];
}
function spawnPiece(){
  var c=pickShape(), k=(Math.random()*4)|0;
  for(var i=0;i<k;i++) c=rotCells(c);
  return {cells:c, color:1+((Math.random()*6)|0)};
}

// ---------------- BOARD OPS ----------------
function newBoard(){
  board=[];
  for(var y=0;y<N;y++){var r=[];for(var x=0;x<N;x++)r.push(0);board.push(r);}
}
function validAt(cells,x,y){
  if(x<0||y<0) return false;
  var d=dims(cells);
  if(x+d[0]>N||y+d[1]>N) return false;
  for(var i=0;i<cells.length;i++) if(board[y+cells[i][1]][x+cells[i][0]]!==0) return false;
  return true;
}
function anyFit(cells){
  var d=dims(cells);
  for(var x=0;x<=N-d[0];x++)for(var y=0;y<=N-d[1];y++) if(validAt(cells,x,y)) return true;
  return false;
}

function reset(){
  newBoard();
  score=0; shownScore=0; lines=0; combo=0; maxCombo=1; pieces=0; level=1; elapsed=0; newBest=false; scorePop=0;
  piece=spawnPiece(); nextPiece=spawnPiece();
  var d=dims(piece.cells); piece.x=(N-d[0])>>1; piece.y=0; pieceT=0;
  particles.length=0; floats.length=0; clearing.length=0; rings.length=0; placedFx.length=0; shakeT=0; dragInvalid=0;
}

function tryMove(dx,dy){
  if(state!=='playing'||!piece) return;
  var d=dims(piece.cells);
  var nx=Math.max(0,Math.min(N-d[0],piece.x+dx)), ny=Math.max(0,Math.min(N-d[1],piece.y+dy));
  if(nx===piece.x&&ny===piece.y) return;
  if(validAt(piece.cells,nx,ny)){ piece.x=nx; piece.y=ny; sfx.move(); }
  else { sfx.bad(); dragInvalid=0.18; }
}
function tryRotate(){
  if(state!=='playing'||!piece) return;
  var rc=rotCells(piece.cells);
  if(validAt(rc,piece.x,piece.y)){ piece.cells=rc; sfx.rotate(); }
  else { sfx.bad(); dragInvalid=0.18; }
}

function placePiece(){
  if(state!=='playing'||!piece) return;
  var p=piece;
  for(var i=0;i<p.cells.length;i++){
    var c=p.cells[i];
    board[p.y+c[1]][p.x+c[0]]=p.color;
    placedFx.push({x:p.x+c[0],y:p.y+c[1],t:0});
  }
  pieces++; score+=p.cells.length*5; scorePop=1; sfx.place();

  var rows=[],cols=[];
  for(var y=0;y<N;y++){var f=true;for(var x=0;x<N;x++)if(!board[y][x]){f=false;break;}if(f)rows.push(y);}
  for(x=0;x<N;x++){f=true;for(y=0;y<N;y++)if(!board[y][x]){f=false;break;}if(f)cols.push(x);}
  var n=rows.length+cols.length;

  if(n>0){
    combo++; maxCombo=Math.max(maxCombo,combo);
    var base=[0,100,300,800,2000][Math.min(n,4)], pts=base*combo;
    score+=pts; scorePop=1;
    var set={}, gx=0, gy=0, cnt=0;
    rows.forEach(function(ry){for(var x=0;x<N;x++)set[ry*N+x]=1;});
    cols.forEach(function(cx2){for(var y=0;y<N;y++)set[y*N+cx2]=1;});
    for(var key in set){
      var px2=key%N, py2=(key/N)|0; gx+=px2; gy+=py2; cnt++;
      var col=board[py2][px2]; board[py2][px2]=0;
      clearing.push({x:px2,y:py2,color:col,t:0});
      burst(px2,py2,col);
    }
    addFloat(gx/cnt,gy/cnt,'+'+fmt(pts),'#ffc233',1.0);
    if(combo>=2) addFloat(gx/cnt,gy/cnt-1.4,'COMBO ×'+combo,'#ff5d73',0.8);
    lines+=n; sfx.clear(n,combo);
    if(n>=2){ shakeIt(3+2*n,0.28); rings.push({x:bx+(gx/cnt+0.5)*cell,y:by+(gy/cnt+0.5)*cell,r:6,max:cell*(1.5+n*0.9),t:0}); }
    var nl=Math.floor(lines/10)+1;
    if(nl>level){ level=nl; addFloat((N-1)/2,(N-1)/2-0.5,'LEVEL '+level,'#3ce6a6',1.3); sfx.level(); }
  } else combo=0;

  piece=nextPiece; nextPiece=spawnPiece(); pieceT=0;
  if(!anyFit(piece.cells)) gameOver();
}

function gameOver(){
  state='gameover'; overShown=false;
  if(score>best){ best=score; newBest=true; try{localStorage.setItem('gridfever_best',String(best));}catch(e){} }
  shakeIt(6,0.4); sfx.over();
  if(newBest) setTimeout(function(){sfx.best();},700);
  setTimeout(function(){
    if(state!=='gameover') return;
    overShown=true;
    $('overScore').textContent=fmt(score);
    $('overLines').textContent=lines;
    $('overCombo').textContent='×'+maxCombo;
    $('overTime').textContent=fmtTime(elapsed);
    $('overPieces').textContent=pieces;
    $('overBest').textContent=fmt(best);
    $('newBest').classList.toggle('hidden',!newBest);
    show(scrOver);
  },900);
}

function burst(gx,gy,color){
  var px=bx+(gx+0.5)*cell, py=by+(gy+0.5)*cell;
  for(var i=0;i<7;i++){
    var a=Math.random()*Math.PI*2, sp=60+Math.random()*240;

  }
}

function easeOutCubic(x){return 1-Math.pow(1-x,3);}
function easeOutBack(x){var c1=1.70158,c3=c1+1;return 1+c3*Math.pow(x-1,3)+c1*Math.pow(x-1,2);}
function fmt(n){return n.toLocaleString('en-US');}
function fmtTime(s){s=Math.floor(s);var m=(s/60)|0;return m+':'+String(s%60).padStart(2,'0');}

window.addEventListener('keydown',function(e){
  initAudio();
  var k=e.key;
  if(k==='ArrowLeft'||k==='ArrowRight'||k==='ArrowUp'||k==='ArrowDown'||k===' ') e.preventDefault();
  switch(k){
    case 'ArrowLeft': case 'a': case 'A': tryMove(-1,0); break;
    case 'ArrowRight': case 'd': case 'D': tryMove(1,0); break;
    case 'ArrowUp': case 'w': case 'W': tryMove(0,-1); break;
    case 'ArrowDown': case 's': case 'S': tryMove(0,1); break;
    case 'r': case 'R': case 'x': case 'X': if(state==='playing'&&!e.repeat) tryRotate(); break;
    case ' ': case 'Enter':
      if(state==='menu') startGame();
      else if(state==='gameover'&&overShown) startGame();
      else if(state==='paused'&&k==='Enter') resumeGame();
      else if(state==='playing'&&!e.repeat) placePiece();
      break;
    case 'p': case 'P': case 'Escape': if(!e.repeat) togglePause(); break;
    case 'm': case 'M': if(!e.repeat) toggleMute(); break;
  }
});

function pointerToBoard(px,py){
  if(!piece||state!=='playing') return;
  var d=dims(piece.cells);
  var nx=Math.round((px-bx)/cell-(d[0]-1)/2), ny=Math.round((py-by)/cell-(d[1]-1)/2);
  nx=Math.max(0,Math.min(N-d[0],nx)); ny=Math.max(0,Math.min(N-d[1],ny));
  if(validAt(piece.cells,nx,ny)){ piece.x=nx; piece.y=ny; dragInvalid=0; }
  else{
    var found=false, r,ox2,oy2;
    for(r=1;r<=3&&!found;r++)
      for(oy2=-r;oy2<=r&&!found;oy2++)
        for(ox2=-r;ox2<=r&&!found;ox2++){
          var tx=nx+ox2, ty=ny+oy2;
          if(tx>=0&&ty>=0&&tx+d[0]<=N&&ty+d[1]<=N&&validAt(piece.cells,tx,ty)){piece.x=tx;piece.y=ty;found=true;}
        }
    dragInvalid=found?0:0.15;
  }
}

// ---------------- UPDATE ----------------
function update(dt){
  t+=dt;
  for(var i=0;i<amb.length;i++){
    var a=amb[i]; a.y-=a.v*dt/H; a.rot+=a.vr*dt;
    if(a.y<-0.06){a.y=1.06;a.x=Math.random();}
  }
  if(state==='playing'){ elapsed+=dt; pieceT+=dt; }
  dragInvalid=Math.max(0,dragInvalid-dt);
  scorePop=Math.max(0,scorePop-dt*3);
  shownScore+=(score-shownScore)*Math.min(1,dt*9);
  if(Math.abs(score-shownScore)<1) shownScore=score;

  for(i=placedFx.length-1;i>=0;i--){ placedFx[i].t+=dt; if(placedFx[i].t>0.2) placedFx.splice(i,1); }
  for(i=clearing.length-1;i>=0;i--){ clearing[i].t+=dt; if(clearing[i].t>0.3) clearing.splice(i,1); }
  for(i=rings.length-1;i>=0;i--){ var rg=rings[i]; rg.t+=dt/0.5; if(rg.t>=1) rings.splice(i,1); else rg.r=6+(rg.max-6)*easeOutCubic(rg.t); }
  for(i=particles.length-1;i>=0;i--){
    var p=particles[i]; p.t+=dt;
    if(p.t>p.life){ particles.splice(i,1); continue; }
    p.vy+=980*dt; p.x+=p.vx*dt; p.y+=p.vy*dt; p.rot+=p.vr*dt;
  }
  for(i=floats.length-1;i>=0;i--){ floats[i].t+=dt; if(floats[i].t>floats[i].life) floats.splice(i,1); }
  if(shakeT>0) shakeT=Math.max(0,shakeT-dt);
}

// ---------------- DRAW HELPERS ----------------
function rr(x,y,w,h,r){
  r=Math.min(r,w/2,h/2);
  ctx.beginPath();
  ctx.moveTo(x+r,y);
  ctx.arcTo(x+w,y,x+w,y+h,r);
  ctx.arcTo(x+w,y+h,x,y+h,r);
  ctx.arcTo(x,y+h,x,y,r);
  ctx.arcTo(x,y,x+w,y,r);
  ctx.closePath();
}
function drawBlock(x,y,size,colorIdx,alpha,scale,rot){
  var p=PAL[colorIdx]; if(!p) return;
  scale=scale||1; if(alpha==null)alpha=1;
  var s=size*scale, cx=x+size/2, cy=y+size/2;
  ctx.save(); ctx.globalAlpha=alpha;
  if(rot){ ctx.translate(cx,cy); ctx.rotate(rot); ctx.translate(-cx,-cy); }
  var bxr=cx-s/2, byr=cy-s/2, r=Math.max(2,s*0.18);
  rr(bxr,byr,s,s,r); ctx.fillStyle=p.b; ctx.fill();
  rr(bxr+s*0.09,byr+s*0.06,s*0.82,s*0.34,Math.max(2,s*0.12)); ctx.globalAlpha=alpha*0.5; ctx.fillStyle=p.l; ctx.fill();
  rr(bxr+s*0.09,byr+s*0.6,s*0.82,s*0.3,Math.max(2,s*0.12)); ctx.globalAlpha=alpha*0.4; ctx.fillStyle=p.d; ctx.fill();
  ctx.restore();
}
function drawMini(pl,cx,cy,cs){
  if(!pl) return;
  var d=dims(pl.cells), w=d[0]*cs, h=d[1]*cs;
  for(var i=0;i<pl.cells.length;i++){
    var c=pl.cells[i];
    drawBlock(cx-w/2+c[0]*cs, cy-h/2+c[1]*cs, cs-3, pl.color, 1, 1, 0);
  }
}

// ---------------- DRAW: BOARD & PIECE ----------------
function drawBoard(){
  var pad=10;
  rr(bx-pad,by-pad,boardPx+pad*2,boardPx+pad*2,14);
  var fg=ctx.createLinearGradient(0,by-pad,0,by+boardPx+pad);
  fg.addColorStop(0,'#1a2f55'); fg.addColorStop(1,'#132340');
  ctx.fillStyle=fg; ctx.fill();
  var empty=0;
  for(var y=0;y<N;y++)for(var x=0;x<N;x++)if(!board[y][x])empty++;
  var danger = state==='playing'&&empty<=10;
  ctx.lineWidth=danger?4:3;
  ctx.strokeStyle=danger?('rgba(255,93,115,'+(0.55+0.4*Math.sin(t*8)).toFixed(3)+')'):'#31507f';
  ctx.stroke();

  // empty cells
  for(y=0;y<N;y++)for(x=0;x<N;x++){
    rr(bx+x*cell+gap/2, by+y*cell+gap/2, cell-gap, cell-gap, Math.max(3,cell*0.16));
    ctx.fillStyle='#141f38'; ctx.fill();
  }

  // hot lines (one block away from full) → gold hint
  var rowCnt=[], colCnt=[], x2, y2;
  for(y=0;y<N;y++)rowCnt.push(0); for(x=0;x<N;x++)colCnt.push(0);
  for(y=0;y<N;y++)for(x=0;x<N;x++){ if(board[y][x]){rowCnt[y]++;colCnt[x]++;} }
  var hot={};
  for(y=0;y<N;y++)if(rowCnt[y]===N-1)for(x=0;x<N;x++)if(board[y][x])hot[y*N+x]=1;
  for(x=0;x<N;x++)if(colCnt[x]===N-1)for(y=0;y<N;y++)if(board[y][x])hot[y*N+x]=1;
  if(Object.keys(hot).length){
    ctx.strokeStyle='rgba(255,194,51,'+(0.3+0.25*Math.sin(t*6)).toFixed(3)+')';
    ctx.lineWidth=2;
    for(var hk in hot){ var hx=hk%N, hy=(hk/N)|0;
      rr(bx+hx*cell+2, by+hy*cell+2, cell-4, cell-4, Math.max(3,cell*0.14)); ctx.stroke(); }
  }

  // filled blocks (with placement pulse)
  var pf={};
  for(var i=0;i<placedFx.length;i++)pf[placedFx[i].y*N+placedFx[i].x]=placedFx[i].t;
  for(y=0;y<N;y++)for(x=0;x<N;x++){
    if(!board[y][x]) continue;
    var s=1, k=pf[y*N+x];
    if(k!=null) s=1+0.16*(1-k/0.2);
    drawBlock(bx+x*cell+gap/2, by+y*cell+gap/2, cell-gap, board[y][x], 1, s, 0);
  }

  // clearing animation
  for(i=0;i<clearing.length;i++){
    var c=clearing[i], kk=c.t/0.3;
    var sc=Math.max(0.02,1-Math.pow(kk,1.6)), al=Math.max(0,1-kk);
    var rot=kk*0.7*((c.x+c.y)%2?1:-1);
    drawBlock(bx+c.x*cell+gap/2, by+c.y*cell+gap/2, cell-gap, c.color, al, sc, rot);
  }
}
function drawPiece(){
  if(!piece) return;
  var s=easeOutBack(Math.min(1,pieceT/0.22));
  var bad=dragInvalid>0;
  for(var i=0;i<piece.cells.length;i++){
    var c=piece.cells[i];
    drawBlock(bx+(piece.x+c[0])*cell+gap/2, by+(piece.y+c[1])*cell+gap/2, cell-gap, piece.color, bad?(0.7+0.3*Math.sin(t*40)):1, s, 0);
  }
}

  for(i=0;i<floats.length;i++){
    var f=floats[i], k=f.t/f.life;

    ctx.save(); ctx.globalAlpha=fa; ctx.translate(px,py); ctx.scale(sc,sc);
    var fs=Math.max(16,cell*0.9)*f.scale;
    ctx.font=fs+'px '+DISP; ctx.textAlign='center'; ctx.textBaseline='middle';
    ctx.lineWidth=4; ctx.strokeStyle='rgba(10,18,34,0.85)'; ctx.strokeText(f.text,0,0);
    ctx.fillStyle=f.color; ctx.fillText(f.text,0,0);
    ctx.restore();
  }
}

function draw(){
  if(bgCv) ctx.drawImage(bgCv,0,0,W,H);
  // drifting dot grid
  var sp=30, ox=(t*4)%sp, oy=(t*8)%sp;
  ctx.fillStyle='rgba(150,190,255,0.05)';
  for(var y=-sp;y<H+sp;y+=sp){ for(var x=-sp;x<W+sp;x+=sp){ ctx.fillRect(x+ox,y+oy,2,2); } }
  // ambient floating blocks
  var aa=state==='menu'?0.16:0.07;
  for(var i=0;i<amb.length;i++){
    var a=amb[i];
    drawBlock(a.x*W-a.s/2,a.y*H-a.s/2,a.s,a.c,aa,1,a.rot);
  }
  // board layer (shakes)
  var ox2=0,oy2=0;
  if(shakeT>0){ var m=shakeMag*(shakeT/shakeDur); ox2=(Math.random()*2-1)*m; oy2=(Math.random()*2-1)*m; }
  ctx.save(); ctx.translate(ox2,oy2);
  drawBoard();
  if(state==='playing'||state==='paused') drawPiece();
  ctx.restore();
  drawFx();
  drawHUD();
}
