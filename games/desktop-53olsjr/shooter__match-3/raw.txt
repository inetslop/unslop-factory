    g.gain.setValueAtTime(v,now); g.gain.exponentialRampToValueAtTime(.001,now+d);
    osc.connect(g); g.connect(master); osc.start(now); osc.stop(now+d+.03);}catch(e){} }

    n.buffer=buf; const f=AC.createBiquadFilter(); f.type=type; f.frequency.value=fq;
    const g=AC.createGain(); g.gain.value=v; n.connect(f); f.connect(g); g.connect(master); n.start(AC.currentTime+t);}catch(e){} }

/* ---------- grid math ---------- */
const ck=(c,r)=>c+','+r;
function kp(k){const i=k.indexOf(',');return [+k.slice(0,i), +k.slice(i+1)];}
const xOf=(c,r)=>boardX+R+c*2*R+(r&1?R:0);
const yOf=r=>boardY+R+r*HROW;
function neighbors(c,r){ const p=r&1;
  const n = p===0 ? [[c-1,r],[c+1,r],[c-1,r-1],[c,r-1],[c-1,r+1],[c,r+1]]
                  : [[c-1,r],[c+1,r],[c,r-1],[c+1,r-1],[c,r+1],[c+1,r+1]];
  return n.filter(([cc])=>cc>=0&&cc<COLS); }
const hasOccNeighbor=(c,r)=>neighbors(c,r).some(([cc,rr])=>grid.has(ck(cc,rr)));
function colorsOnBoard(){ const s=new Set(grid.values()); const a=[...s]; return a.length?a:[...Array(levelColors()).keys()]; }
const levelColors=()=> level<=1?4 : level===2?5 : 6;
const pickColor=()=>{const p=colorsOnBoard();return p[irand(p.length)];};

function flood(c0,r0,col){ const out=[],seen=new Set([ck(c0,r0)]),st=[[c0,r0]];
  while(st.length){ const [c,r]=st.pop(); if(grid.get(ck(c,r))!==col)continue; out.push([c,r]);
    for(const [cc,rr] of neighbors(c,r)){ const k=ck(cc,rr); if(!seen.has(k)&&grid.has(k)){seen.add(k);st.push([cc,rr]);} } }
  return out; }
function findFloaters(){ const seen=new Set(), st=[];
  for(const [k] of grid){ const r=+k.slice(k.indexOf(',')+1); if(r===0){seen.add(k);st.push(kp(k));} }
  while(st.length){ const [c,r]=st.pop();
    for(const [cc,rr] of neighbors(c,r)){ const k=ck(cc,rr); if(grid.has(k)&&!seen.has(k)){seen.add(k);st.push([cc,rr]);} } }
  const fl=[]; for(const [k,v] of grid){ if(!seen.has(k)){const [c,r]=kp(k);fl.push({c,r,col:v});} } return fl; }
function snapCell(px,py){ let r0=Math.max(0,Math.round((py-boardY-R)/HROW));
  const p=r0&1; let c0=clamp(Math.round((px-boardX-R-(p?R:0))/(2*R)),0,COLS-1);
  let best=null,bd=Infinity;
  for(let d=0;d<=2&&!best;d++)
    for(let rr=Math.max(0,r0-d);rr<=r0+d;rr++)
      for(let cc=Math.max(0,c0-d);cc<=c0+d;cc++){
        if(Math.max(Math.abs(cc-c0),Math.abs(rr-r0))!==d||cc>COLS-1)continue;
        const k=ck(cc,rr); if(grid.has(k))continue;
        if(rr!==0&&!hasOccNeighbor(cc,rr))continue;
        const dx=xOf(cc,rr)-px, dy=yOf(rr)-py, dist=dx*dx+dy*dy;
        if(dist<bd){bd=dist;best={c:cc,r:rr};} }
  return best; }
function recomputeDanger(){ let m=-1e9;
  for(const [k] of grid){ const r=+k.slice(k.indexOf(',')+1); const y=yOf(r); if(y>m)m=y; }
  maxCenter=m; }

function loadLevel(l){ grid.clear(); projectile=null; falling=[]; popAnims=[]; popups=[]; combo=null;
  shotsFired=0; pushEvery = l<3?5:4; recomputeDanger();
  const rows=Math.min(5+l,9), lc=levelColors(), pal=[...Array(lc).keys()];
  for(let r=0;r<rows;r++)for(let c=0;c<COLS;c++){
    const bad=new Set();
    if(c>=2){const a=grid.get(ck(c-1,r));if(a!==undefined&&a===grid.get(ck(c-2,r)))bad.add(a);}
    if(r>0){const p=r&1, a=p?grid.get(ck(c,r-1)):grid.get(ck(c-1,r-1)), b=p?grid.get(ck(c+1,r-1)):grid.get(ck(c,r-1));
      if(a!==undefined&&a===b)bad.add(a);}
    const opts=pal.filter(i=>!bad.has(i)).sort(()=>Math.random()-.5);
    grid.set(ck(c,r),opts[0]);
  }
  curCol=irand(lc); nextCol=irand(lc); }

function fire(){ if(state!=='playing'||paused||projectile)return;
  const a=aim*Math.PI/180, dx=Math.cos(a), dy=-Math.sin(a), spd=clamp(R*38,560,1250);
  projectile={x:lx+dx*R*2.1,y:ly+dy*R*2.1,vx:dx*spd,vy:dy*spd,col:curCol};
  curCol=nextCol; nextCol=pickColor(); shotsFired++; recoil=1; sfx.shoot(); }

function resolveShot(c,r,col){ grid.set(ck(c,r),col); sfx.place();
  popAnims.push({kind:'ring',x:xOf(c,r),y:yOf(r),col,t:0,life:.3});
  const cl=flood(c,r,col);
  if(cl.length>=3){
    let sx=0,sy=0; for(const [c2,r2] of cl){sx+=xOf(c2,r2);sy+=yOf(r2);spawnPopFX(xOf(c2,r2),yOf(r2),col);grid.delete(ck(c2,r2));}
    const g1=cl.length*10; score+=g1; scorePulse=1;
    addPopup(sx/cl.length,sy/cl.length,'+'+g1,R*.85,'#ffffff'); sfx.pop(cl.length);
    shakeAmp = cl.length>=5 ? Math.min(9,2+cl.length*.6) : 3;
    const fl=findFloaters(); let dropped=0;
    if(fl.length){ dropped=fl.length; let dxs=0,dys=0;
      for(const f of fl){ dxs+=xOf(f.c,f.r); dys+=yOf(f.r); grid.delete(ck(f.c,f.r));
        falling.push({x:xOf(f.c,f.r),y:yOf(f.r),vx:rand(-60,60),vy:-40-rand(60),col:f.col,delay:rand(.25),rot:0}); }
      const g2=fl.length*25; score+=g2; scorePulse=1;
      addPopup(dxs/fl.length,dys/fl.length,'DROP +'+g2,R*.8,'#7ef0ff'); sfx.drop(fl.length); shakeAmp=Math.max(shakeAmp,5); }
    const total=cl.length+dropped; let word='POP!',wcol='#7ef0ff';
    if(total>=8){word='BLASTING!';wcol='#ffd23f'} else if(total>=6||total===5){word='GREAT!';wcol='#4be3a0'}
    else if(dropped>0){word='NICE DROP!';wcol='#7ef0ff'}
    combo={txt:word,col:wcol,t:0,life:1.1};
  }
  recomputeDanger();
  if(grid.size===0){ startClearing(); return; }
  if(dangerY-R*.4-maxCenter<=0){ startDying(); return; }
  if(shotsFired%pushEvery===0){ state='pushing'; pushT=0; pushPhase='warn'; sfx.warn(); } }

  const pool=colorsOnBoard();
  for(let c=0;c<COLS;c++)nm.set(ck(c,0),pool[irand(pool.length)]);
  grid=nm; sfx.slide(); }

/* ---------- update ---------- */
function tickFx(dt){
  recoil=Math.max(0,recoil-dt*7); swapPulse=Math.max(0,swapPulse-dt*4);
  scorePulse=Math.max(0,scorePulse-dt*3); shakeAmp*=Math.exp(-8*dt); if(shakeAmp<.05)shakeAmp=0;
  scoreShown+=(score-scoreShown)*Math.min(1,dt*12); if(Math.abs(score-scoreShown)<1)scoreShown=score;
  for(let i=falling.length-1;i>=0;i--){ const f=falling[i];
    if(f.delay>0){f.delay-=dt;continue;} f.vy+=1400*dt; f.x+=f.vx*dt; f.y+=f.vy*dt;
    if(f.y>H+R*2)falling.splice(i,1); }
  for(let i=particles.length-1;i>=0;i--){ const p=particles[i]; p.t+=dt;
    if(p.t>=p.life){particles.splice(i,1);continue;} p.vy+=p.g*dt; p.x+=p.vx*dt; p.y+=p.vy*dt; }
  for(let i=popAnims.length-1;i>=0;i--){ popAnims[i].t+=dt; if(popAnims[i].t>=popAnims[i].life)popAnims.splice(i,1); }
  for(let i=popups.length-1;i>=0;i--){ popups[i].t+=dt; if(popups[i].t>=popups[i].life)popups.splice(i,1); }
  if(combo){ combo.t+=dt; if(combo.t>=combo.life)combo=null; }
}
function stepProjectile(dt){ const p=projectile; if(!p)return;
  const dist=Math.hypot(p.vx,p.vy)*dt, n=Math.max(1,Math.ceil(dist/(R*.45)));
  for(let i=0;i<n;i++){ p.x+=p.vx*dt/n; p.y+=p.vy*dt/n;
    if(p.x<boardX+R){p.x=boardX+R;p.vx=-p.vx;sfx.bounce();}
    else if(p.x>boardX+boardW-R){p.x=boardX+boardW-R;p.vx=-p.vx;sfx.bounce();}
    let hit = p.y-R<=boardY+1;
    if(!hit) for(const [k] of grid){ const [c,r]=kp(k);
      const dx=p.x-xOf(c,r), dy=p.y-yOf(r);
      if(dx*dx+dy*dy < (2*R-3)*(2*R-3)){hit=true;break;} }
    if(hit) break; }
  if(p.y-R<=boardY+1)p.y=boardY+R;
  let cell=snapCell(p.x,p.y);
  if(!cell){ // fallback: nearest empty anywhere near
    let bd=Infinity; for(const [k] of grid){const [c,r]=kp(k);
      const dx=p.x-xOf(c,r),dy=p.y-yOf(r),d=dx*dx+dy*dy; if(d<bd){bd=d;} } }
  projectile=null;
  if(cell)resolveShot(cell.c,cell.r,p.col); }

function update(dt){
  if(state==='playing'){
    aim=clamp(aim+(keysR?1:0)*70*dt-(keysL?1:0)*70*dt,12,168);
    stepProjectile(dt);
  } else if(state==='pushing'){
    pushT+=dt;
    if(pushPhase==='warn'&&pushT>0.55){pushPhase='slide';pushT=0;doPushRow();}
    else if(pushPhase==='slide'&&pushT>0.34){ state='playing'; recomputeDanger();
      if(dangerY-R*.4-maxCenter<=0)startDying(); }
  } else if(state==='clearing'){
    clearT+=dt;
    if(clearT>1.6){ level++; loadLevel(level); state='playing';
      banner('LEVEL '+level,FLAVOR[(level-1)%FLAVOR.length]); sfx.ui(); }
  } else if(state==='dying'){
    dieT+=dt; while(dieT>=dieNext){ dieNext+=.085; burstLowestRow(); }
    if(grid.size===0)finalizeOver();
  }
}

function traceGuide(){ const a=aim*Math.PI/180; let dx=Math.cos(a),dy=-Math.sin(a);
  let x=lx+dx*R*2.1, y=ly+dy*R*2.1; const step=R*.85, pts=[[x,y]];
  for(let i=0;i<240;i++){ x+=dx*step; y+=dy*step;
    if(x<boardX+R){x=boardX+R;dx=-dx;} else if(x>boardX+boardW-R){x=boardX+boardW-R;dx=-dx;}
    if(y<=boardY+R){pts.push([x,y]);break;}
    let hit=false; const lim=(2*R-4)*(2*R-4);
    for(const [k] of grid){ const [c,r]=kp(k); const ddx=xOf(c,r)-x, ddy=yOf(r)-y;
      if(ddx*ddx+ddy*ddy<lim){hit=true;break;} }
    if(hit){pts.push([x,y]);break;}
    if(i%3===0)pts.push([x,y]); }
  const end=pts[pts.length-1];
  return {pts,cell:snapCell(end[0],end[1])}; }
