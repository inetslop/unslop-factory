    const x0=m.side===1?IL+MR:CX,x1=m.side===1?CX:IR-MR;
    if(m.x<x0){m.x=x0;if(m.vx<0)m.vx=0;}else if(m.x>x1){m.x=x1;if(m.vx>0)m.vx=0;}
    if(m.y<IT+MR){m.y=IT+MR;if(m.vy<0)m.vy=0;}else if(m.y>IB-MR){m.y=IB-MR;if(m.vy>0)m.vy=0;}
    m.sting*=Math.exp(-7*dt);
  }
}
function malletCollide(){
  const dx=p2.x-p1.x,dy=p2.y-p1.y,d=Math.hypot(dx,dy),minD=MR*2;
  if(d>=minD||d===0)return;
  const nx=dx/d,ny=dy/d,ov=(minD-d)/2;
  p1.x-=nx*ov;p1.y-=ny*ov;p2.x+=nx*ov;p2.y+=ny*ov;
  const rvn=(p2.vx-p1.vx)*nx+(p2.vy-p1.vy)*ny;
  if(rvn<0){const j=-(1+.55)*rvn/2;p1.vx-=j*nx;p1.vy-=j*ny;p2.vx+=j*nx;p2.vy+=j*ny;
    if(-rvn>140){sparks((p1.x+p2.x)/2,(p1.y+p2.y)/2,8,['#ffffff','#9fd8ff'],300);blip(150,.09,'square',.22);addShake(3);}}
  const c=(m,a,b)=>{if(m.x<a){m.x=a;if(m.vx<0)m.vx=0;}else if(m.x>b){m.x=b;if(m.vx>0)m.vx=0;}};
  c(p1,IL+MR,CX);c(p2,CX,IR-MR);
}
function malletHit(m){
  const dx=puck.x-m.x,dy=puck.y-m.y,minD=PR+MR,d2=dx*dx+dy*dy;
  if(d2>=minD*minD)return;
  const d=Math.sqrt(d2)||.001,nx=dx/d,ny=dy/d;
  puck.x=m.x+nx*minD;puck.y=m.y+ny*minD;
  const rvn=(puck.vx-m.vx)*nx+(puck.vy-m.vy)*ny;if(rvn>=0)return;
  const j=-(1+MREST)*rvn;puck.vx+=j*nx;puck.vy+=j*ny;m.sting=1;
  if(j>70){const spd=Math.hypot(puck.vx,puck.vy);
    blip(clamp(240+spd*.32,240,980),.05,'triangle',clamp(.08+j/1600,.08,.4));
    sparks(puck.x-nx*PR,puck.y-ny*PR,Math.min(12,(j/70)|0),[m.cGlow,'#ffffff'],clamp(j*.5,120,620));
    addShake(clamp((j-200)/130,0,9));}
}
function wallFX(v,x,y){
  thud(clamp(v/1500,.05,.3),90+clamp(v*.02,0,60));
  if(v>240)sparks(x,y,Math.min(8,(v/130)|0),['#dcecff'],clamp(v*.4,100,420));
  if(v>1050)addShake(clamp((v-1050)/160,0,7));
}
function walls(){
  const inMouthL=Math.abs(puck.y-CY)<GH-6;
  if(puck.y<IT+PR){puck.y=IT+PR;if(puck.vy<0){wallFX(-puck.vy,puck.x,IT);puck.vy*=-REST;}}
  else if(puck.y>IB-PR){puck.y=IB-PR;if(puck.vy>0){wallFX(puck.vy,puck.x,IB);puck.vy*=-REST;}}
  if(inMouthL){
    if(puck.x<IL-8){scoreGoal(2);return true;}
    const lo=CY-GH+PR,hi=CY+GH-PR;
    if(puck.y<lo){puck.y=lo;if(puck.vy<0)puck.vy*=-.5;}else if(puck.y>hi){puck.y=hi;if(puck.vy>0)puck.vy*=-.5;}
  }else{
    if(puck.x<IL+PR){puck.x=IL+PR;if(puck.vx<0){wallFX(-puck.vx,IL,puck.y);puck.vx*=-REST;}}
  }
  const inMouthR=Math.abs(puck.y-CY)<GH-6;
  if(inMouthR){
    if(puck.x>IR+8){scoreGoal(1);return true;}
    const lo=CY-GH+PR,hi=CY+GH-PR;
    if(puck.y<lo){puck.y=lo;if(puck.vy<0)puck.vy*=-.5;}else if(puck.y>hi){puck.y=hi;if(puck.vy>0)puck.vy*=-.5;}
  }else{
    if(puck.x>IR-PR){puck.x=IR-PR;if(puck.vx>0){wallFX(puck.vx,IR,puck.y);puck.vx*=-REST;}}
  }
  return false;
}
function stepPuck(dt){
  const sp=Math.hypot(puck.vx,puck.vy);
  const n=Math.min(8,Math.max(1,Math.ceil(sp*dt/(PR*.6)))),sdt=dt/n;
  for(let i=0;i<n;i++){puck.x+=puck.vx*sdt;puck.y+=puck.vy*sdt;if(walls())return;malletHit(p1);malletHit(p2);}
  const f=Math.exp(-PFRIC*dt);puck.vx*=f;puck.vy*=f;
  let s2=Math.hypot(puck.vx,puck.vy);
  if(s2>MAXV){puck.vx*=MAXV/s2;puck.vy*=MAXV/s2;s2=MAXV;}
  if(s2>1500&&!wasFast){wasFast=true;blip(320,.18,'sine',.14,0,90);}else if(s2<1100)wasFast=false;
  trail.push({x:puck.x,y:puck.y});if(trail.length>18)trail.shift();
}
function scoreGoal(who){
  if(!puckLive)return;
  puckLive=false;lastScorer=who;serveSide=who===1?2:1;scores[who]++;
  flash=1;addShake(15);horn();trail.length=0;
  const gx=who===1?IR-6:IL+6,gy=clamp(puck.y,CY-GH+10,CY+GH-10);
  const col=who===1?'#35dfe8':'#ff6a3c';
  sparks(gx,gy,26,[col,'#ffffff'],760);ring(gx,gy,col);
  state='goal';stateT=0;
  if(mode==='match')updateHUD(who);
}
function addShake(m){shakeMag=Math.max(shakeMag,m);}

/* ================= state machine / step ================= */
function tickState(){
  if(state==='ready'&&stateT>=.85){state='play';stateT=0;blip(990,.07,'square',.25);}
  else if(state==='goal'&&stateT>=1.5){
    if(mode==='attract'){resetPositions(Math.random()<.5?1:2,true);state='menu';stateT=0;puckLive=true;}
    else if(scores.p1>=WINNER||scores.p2>=WINNER)enterWin();
    else{resetPositions(serveSide);state='ready';stateT=0;puckLive=true;updateHUD();}
  }
  else if(state==='win'&&!winShown&&stateT>=.9){winShown=true;show(winOv,true);}
}
function step(dt){
  t+=dt;updateSnow(dt);updateParts(dt);
  shakeMag*=Math.exp(-5.2*dt);if(shakeMag<.05)shakeMag=0;
  flash*=Math.exp(-4.5*dt);if(flash<.01)flash=0;
  if(state==='pause')return;
  stateT+=dt;
  if(state==='win'){
    winAcc+=dt;const w=scores.p1>scores.p2?1:2;
    const cols=w===1?['#ff6a3c','#ffb08e','#ffffff']:['#35dfe8','#bdf8fb','#ffffff'];
    while(winAcc>.14){winAcc-=.14;confetti(rand(IL+40,IR-40),IT+10,2,cols);}
    const f=Math.exp(-3*dt);for(const m of [p1,p2]){m.vx*=f;m.vy*=f;}
    integrateMallets(dt);tickState();return;
  }
  if(state==='menu')aiControl(dt);else if(state==='play'||state==='goal')controlMallets(dt);
  if(state!=='ready'){integrateMallets(dt);malletCollide();}
  if(puckLive&&state!=='ready')stepPuck(dt);
  tickState();
}
