function genMaze(){

  const seen=new Set([0]),stack=[0];
  while(stack.length){
    const cur=stack[stack.length-1],cc=cur%COLS,cr=(cur/COLS)|0,opts=[];
    if(cr>0&&!seen.has(idx(cc,cr-1)))opts.push(0);
    if(cc<COLS-1&&!seen.has(idx(cc+1,cr)))opts.push(1);
    if(cr<ROWS-1&&!seen.has(idx(cc,cr+1)))opts.push(2);
    if(cc>0&&!seen.has(idx(cc-1,cr)))opts.push(3);
    if(!opts.length){stack.pop();continue;}
    const d=pick(opts),n=[[cc,cr-1],[cc+1,cr],[cc,cr+1],[cc-1,cr]][d],ni=idx(n[0],n[1]);
    cells[cur].walls[d]=0;cells[ni].walls[(d+2)%4]=0;seen.add(ni);stack.push(ni);
  }
}

  for(let i=0;i<nH;i++)cards.push({suit:'h',rank:Math.random()<0.18?'10':pick(['5','6','7','8','9','A'])});
  for(let i=0;i<nC;i++)cards.push({suit:'c',rank:pick(hi)});
  while(cards.length<pool.length)cards.push({suit:pick(['s','h','c']),rank:pick(['2','3','4'])});
  pool.forEach((ci,i)=>cells[ci].card=cards[i]);
  cells[n-1].isExit=true;cells[n-1].card={suit:'j',rank:'J'};
  for(const cl of cells){if(cl.card)cl.card.dud=!(isGem(cl.card)||isIce(cl.card)||isHeart(cl.card)||isThorn(cl.card));}
}

/* ================= game flow ================= */
function startGame(){
  const landscape=W>=H;
  COLS=landscape?13:9;ROWS=landscape?9:13;
  computeLayout();rebuildCaches();
  genMaze();assignCards();
  score=0;gems=0;lives=3;shields=0;timeLeft=240;lastSec=240;
  unlocked=false;loseReason='';winBonus=0;endShown=false;
  particles=[];floats=[];flashRed=0;shakeMag=0;
  player={cx:0,cy:0,moving:false,path:[],pi:0,t:0,dir:[0,1],buf:null,bump:null};
  for(const cl of cells)cl.dealD=(cl.c+cl.r)*38+rand(24);
  gameStart=performance.now();
  dealEnd=gameStart+(COLS+ROWS)*38+750;
  state='deal';
  $('veil').classList.add('hidden');
  renderScore();renderGems();renderHearts(false);renderShields();renderTime(true);
  noise(0.15,0.14,1200);noise(0.15,0.12,900,0.12);noise(0.18,0.12,700,0.26);
}
function pauseGame(){if(state!=='play')return;state='paused';$('veil').classList.remove('hidden');$('pausePanel').classList.remove('hidden');}
function unpauseGame(){if(state!=='paused')return;state='play';$('veil').classList.add('hidden');$('pausePanel').classList.add('hidden');}

  if(state!=='play')return;
  loseReason=reason;state='over';overT=performance.now();sfx.lose();shakeMag=6;
}
function winGame(){
  if(state!=='play')return;
  winBonus=Math.round(timeLeft)*5;score+=winBonus;renderScore();
  state='win';winT=performance.now();sfx.win();confettiBurst(exitCenter());
}
function unlockJoker(){
  unlocked=true;sfx.open();

  particles.push({type:'ring',x:p.x,y:p.y,r0:CS*0.4,r1:CS*4,life:0,dur:0.8,col:'#ffd76a'});

}
const exitCenter=()=>({x:ox+(COLS-0.5)*CS,y:oy+(ROWS-0.5)*CS});

}

/* ================= movement & effects ================= */
const DVEC={up:[0,-1],down:[0,1],left:[-1,0],right:[1,0]};
const wallIdxFor=(dx,dy)=>dy===-1?0:dx===1?1:dy===1?2:3;
function tryMove(dirName){
  if(state!=='play'||!player)return;
  const d=DVEC[dirName],dx=d[0],dy=d[1];
  if(player.moving){player.buf=dirName;return;}
  const cur=cellAt(player.cx,player.cy);
  if(cur.walls[wallIdxFor(dx,dy)]){bump(dirName);return;}
  const path=[];let c=cur.c,r=cur.r;
  let budget=isIce(cur.card)?Math.min(valOf(cur.card.rank),9):0;
  while(budget>0){
    if(cellAt(c,r).walls[wallIdxFor(dx,dy)])break;
    c+=dx;r+=dy;budget--;
    path.push([c,r]);
    const nc=cellAt(c,r);
    if(isIce(nc.card))budget=Math.min(budget+Math.min(valOf(nc.card.rank),9),20);
    if(nc.isExit&&!unlocked)break;
  }
  if(!path.length)return;
  player.path=path;player.pi=0;player.t=0;player.dir=[dx,dy];player.moving=true;
  if(path.length>1)sfx.slide();else sfx.tick();
}
function bump(dirName){
  const d=DVEC[dirName];
  player.bump={dx:d[0],dy:d[1],t:0};
  sfx.bump();
  const p=chipPos();
  for(let i=0;i<5;i++)particles.push({type:'puff',x:p.x+d[0]*CS*0.4,y:p.y+d[1]*CS*0.4,vx:(Math.random()-0.5)*30,vy:(Math.random()-0.5)*30,r:2+Math.random()*3,dur:0.4,t:0,col:'rgba(220,215,190,0.5)'});
}
function commitStep(){
  const p=player.path[player.pi];
  player.cx=p[0];player.cy=p[1];
  arrive(cellAt(p[0],p[1]));
  if(state!=='play'){player.moving=false;return;}
  if(player.stopAfter){player.stopAfter=false;player.moving=false;player.buf=null;return;}
  player.pi++;
  if(player.pi>=player.path.length)player.moving=false;
}
function arrive(cl){
  if(!cl.revealed){
    cl.revealed=true;cl.flipT=0;sfx.flip();
    applyEffect(cl);
  }
  if(cl.isExit&&state==='play'){
    if(unlocked)winGame();
    else{addFloat(ox+(cl.c+0.5)*CS,oy+(cl.r-0.1)*CS,'LOCKED \u00B7 '+(gemTotal-gems)+' \u2666 LEFT','#ffd76a',Math.min(15,W*0.02));sfx.lock();player.stopAfter=true;}
  }
}
function applyEffect(cl){
  const card=cl.card;
  if(!card||card.suit==='j')return;
  const cx=ox+(cl.c+0.5)*CS,cy=oy+(cl.r+0.5)*CS;
  if(isGem(card)){
    const pts=valOf(card.rank)*10;
    score+=pts;gems++;renderScore();renderGems();
    addFloat(cx,cy,'+'+pts,'#ffd76a',Math.min(16,W*0.024));
    sparkle(cx,cy,14,['#ffd76a','#fff3c8','#e8b545']);
    sfx.gem();
    if(gems>=gemTotal)unlockJoker();
  }else if(isHeart(card)){
    if(card.rank==='10'){lives=Math.min(5,lives+1);renderHearts(true);addFloat(cx,cy,'+1 \u2665','#7ee2a8',Math.min(15,W*0.022));sfx.life();}
    else{shields=Math.min(3,shields+1);renderShields();addFloat(cx,cy,'SHIELD +1','#ffe9b0',Math.min(14,W*0.02));sparkle(cx,cy,7,['#a8d8ff','#ffe9b0']);sfx.shield();}
  }else if(isThorn(card)){
    player.stopAfter=true;
    if(shields>0){shields--;card.nullified=true;renderShields();addFloat(cx,cy,'BLOCKED','#ffe9b0',Math.min(14,W*0.02));sparkle(cx,cy,8,['#ffd76a','#fff3c8']);sfx.block();}
    else{lives--;renderHearts(false);flashRed=1;shakeMag=9;addFloat(cx,cy,'-1 \u2665','#ff8f84',Math.min(16,W*0.024));sparkle(cx,cy,12,['#e05a50','#8c1d24']);sfx.hurt();
      if(lives<=0)gameOver('thorns');}
  }
  /* spades: no immediate effect \u2014 they slide the next step */
}
function sparkle(x,y,n,cols){
  for(let i=0;i<n;i++){
    const a=Math.random()*Math.PI*2,sp=40+Math.random()*130;
    particles.push({type:'spark',x,y,vx:Math.cos(a)*sp,vy:Math.sin(a)*sp-30,dur:0.5+Math.random()*0.4,t:0,col:pick(cols),r:1.5+Math.random()*2});
  }
}
function confettiBurst(p){
  const cols=['#f7f1e0','#c8352f','#d9a441','#22335c'];
  for(let i=0;i<90;i++){
    const a=Math.random()*Math.PI*2,sp=120+Math.random()*330;
    particles.push({type:'conf',x:p.x,y:p.y,vx:Math.cos(a)*sp,vy:Math.sin(a)*sp-140,rot:Math.random()*7,vr:(Math.random()-0.5)*12,w:4+Math.random()*6,h:8+Math.random()*8,col:pick(cols),dur:1.4+Math.random()*0.9,t:0});
  }
}
function addFloat(x,y,txt,col,size){floats.push({x,y,txt,col,size:size||14,t:0,dur:1.25});}

/* ================= update ================= */
function chipPos(){
  let x=ox+(player.cx+0.5)*CS,y=oy+(player.cy+0.5)*CS;
  if(player.moving){
    const prev=player.pi===0?[player.cx,player.cy]:player.path[player.pi-1];
    const nxt=player.path[player.pi];
    x=lerp(ox+(prev[0]+0.5)*CS,ox+(nxt[0]+0.5)*CS,clamp(player.t,0,1));
    y=lerp(oy+(prev[1]+0.5)*CS,oy+(nxt[1]+0.5)*CS,clamp(player.t,0,1));
  }
  if(player.bump){const bt=Math.min(1,player.bump.t/0.16),o=Math.sin(Math.PI*bt)*CS*0.18;x+=player.bump.dx*o;y+=player.bump.dy*o;}
  return {x,y};
}
function update(dt){
  const now=performance.now();
  if(state==='deal'&&now>=dealEnd){
    state='play';
    const sc=cells[0];if(!sc.revealed){sc.revealed=true;sc.flipT=0;}
  }
  if(state==='play'){
    timeLeft-=dt;renderTime(false);
    const s=Math.ceil(timeLeft);
    if(s!==lastSec&&s<=10&&s>0)sfx.tickT();
    lastSec=s;
    if(timeLeft<=0){timeLeft=0;gameOver('time');}
  }
  if(player&&state==='play'){
    if(player.moving){
      const dur=player.path.length>1?72:130;
      player.t+=dt*1000/dur;
      while(player.t>=1&&player.moving&&state==='play'){player.t-=1;commitStep();}
      if(state!=='play')player.moving=false;
    }else if(player.buf){const b=player.buf;player.buf=null;tryMove(b);}
  }
  if(player&&player.bump){player.bump.t+=dt;if(player.bump.t>0.16)player.bump=null;}
  for(const cl of cells)if(cl.revealed&&cl.flipT<1)cl.flipT=Math.min(1,cl.flipT+dt/0.24);
  if(unlocked&&state==='play'&&Math.random()<dt*3){const p=exitCenter();sparkle(p.x+(Math.random()-0.5)*CS,p.y+(Math.random()-0.5)*CS,1,['#ffd76a','#fff3c8']);}
  for(let i=particles.length-1;i>=0;i--){
    const p=particles[i];p.t+=dt;
    if(p.t>p.dur||(p.type==='ring'&&p.t>0.8)){particles.splice(i,1);continue;}
    if(p.type==='spark'){p.x+=p.vx*dt;p.y+=p.vy*dt;p.vy+=220*dt;p.vx*=0.96;}
    else if(p.type==='puff'){p.x+=p.vx*dt;p.y+=p.vy*dt;}
    else if(p.type==='conf'){p.x+=p.vx*dt;p.y+=p.vy*dt;p.vy+=560*dt;p.rot+=p.vr*dt;p.vx*=0.99;}
  }
  for(let i=floats.length-1;i>=0;i--){floats[i].t+=dt;if(floats[i].t>floats[i].dur)floats.splice(i,1);}
  flashRed=Math.max(0,flashRed-dt*2.2);
  shakeMag=Math.max(0,shakeMag-dt*26);
  if(state==='win'&&!endShown&&now-winT>1150){endShown=true;showEnd();}
  if(state==='over'&&!endShown&&now-overT>950){endShown=true;showEnd();}
}

/* ================= render ================= */
function render(){
  const now=performance.now();
  ctx.clearRect(0,0,W,H);
  ctx.fillStyle='#12372b';ctx.fillRect(0,0,W,H);
  if(noisePat){ctx.globalAlpha=0.5;ctx.fillStyle=noisePat;ctx.fillRect(0,0,W,H);ctx.globalAlpha=1;}
  const gx=W/2,gy=cells.length?oy+boardH/2:H/2;
  const rg=ctx.createRadialGradient(gx,gy,CS*1.4,gx,gy,Math.max(W,H)*0.78);
  rg.addColorStop(0,'rgba(255,244,214,0.10)');rg.addColorStop(1,'rgba(0,0,0,0.4)');
  ctx.fillStyle=rg;ctx.fillRect(0,0,W,H);

  if(!cells.length){return;}

  const sx=(Math.random()*2-1)*shakeMag,sy=(Math.random()*2-1)*shakeMag;
  ctx.save();ctx.translate(sx,sy);

  /* board base */
  const th=Math.max(5,CS*0.14);
  ctx.fillStyle='rgba(0,0,0,.16)';rr(ctx,ox-th-8,oy-th+2,boardW+th*2+16,boardH+th*2+10,14);ctx.fill();
  ctx.fillStyle='rgba(0,0,0,.3)';rr(ctx,ox-th-5,oy-th+5,boardW+th*2+10,boardH+th*2+8,12);ctx.fill();
  for(const cl of cells){
    ctx.fillStyle=(cl.c+cl.r)%2?'#1d4a3b':'#1b4637';
    ctx.fillRect(ox+cl.c*CS,oy+cl.r*CS,CS,CS);
  }
  ctx.strokeStyle='rgba(214,178,94,.45)';ctx.lineWidth=2;
  rr(ctx,ox-th-6,oy-th-6,boardW+th*2+12,boardH+th*2+12,th);ctx.stroke();

  /* walls as card-back strips */
  ctx.fillStyle=wallPat;
  const rects=[];
  for(const cl of cells){
    const x=ox+cl.c*CS,y=oy+cl.r*CS;
    if(cl.walls[0])rects.push([x-th/2-0.5,y-th/2,CS+th+1,th]);
    if(cl.walls[3])rects.push([x-th/2,y-th/2-0.5,th,CS+th+1]);
    if(cl.walls[2]&&cl.r===ROWS-1)rects.push([x-th/2-0.5,y+CS-th/2,CS+th+1,th]);
    if(cl.walls[1]&&cl.c===COLS-1)rects.push([x+CS-th/2,y-th/2-0.5,th,CS+th+1]);
  }
  for(const r of rects)ctx.fillRect(r[0],r[1],r[2],r[3]);
  ctx.strokeStyle='rgba(30,8,12,.7)';ctx.lineWidth=1;
  for(const r of rects)ctx.strokeRect(r[0]+.5,r[1]+.5,r[2]-1,r[3]-1);
  ctx.fillStyle='rgba(255,226,170,.1)';
  for(const r of rects)ctx.fillRect(r[0],r[1],r[2],Math.min(2,r[3]*0.4));

  /* cards */
  const s=CS*0.82;
  for(const cl of cells){
    const cx=ox+(cl.c+0.5)*CS,cy=oy+(cl.r+0.5)*CS;
    if(!cl.revealed){
      const dtv=clamp((now-gameStart-cl.dealD)/280,0,1);
      if(dtv<=0)continue;
      const t=easeOutBack(dtv);
      ctx.save();ctx.translate(cx,cy-(1-dtv)*9);ctx.scale(t,t);ctx.globalAlpha=clamp(dtv*3,0,1);
      ctx.fillStyle='rgba(0,0,0,.25)';ctx.beginPath();ctx.ellipse(0,s*0.46+2,s*0.42,s*0.16,0,0,7);ctx.fill();
      ctx.drawImage(cl.isExit?jokerBackImg:backImg,-s/2,-s/2,s,s);
      ctx.restore();continue;
    }
    const f=cl.flipT,scX=Math.max(0.03,Math.abs(Math.cos(Math.PI*f)));
    const lift=Math.sin(Math.PI*f)*CS*0.16;
    ctx.save();ctx.translate(cx,cy-lift);
    ctx.fillStyle='rgba(0,0,0,.25)';ctx.beginPath();ctx.ellipse(0,s*0.46+2-lift*0.3,s*0.42*scX,s*0.16,0,0,7);ctx.fill();
    ctx.scale(scX,1);
    const img=cl.isExit?getJokerFace():(f<0.5?(cl.card&&cl.card.suit==='j'?jokerBackImg:backImg):getFace(cl.card));
    ctx.drawImage(img,-s/2,-s/2,s,s);
    ctx.restore();
    if(f>=1){
      const card=cl.card;
      if(isIce(card)){
        ctx.fillStyle='rgba(140,200,255,.16)';rr(ctx,cx-s/2,cy-s/2,s,s,s*0.1);ctx.fill();
        ctx.strokeStyle='rgba(255,255,255,.3)';ctx.lineWidth=1.5;
        ctx.beginPath();ctx.moveTo(cx-s*0.28,cy+s*0.3);ctx.lineTo(cx+s*0.1,cy-s*0.34);ctx.stroke();
        ctx.beginPath();ctx.moveTo(cx-s*0.1,cy+s*0.34);ctx.lineTo(cx+s*0.28,cy-s*0.26);ctx.stroke();
      }else if(isThorn(card)){
        ctx.strokeStyle=card.nullified?'rgba(217,164,65,.5)':'rgba(200,45,55,.45)';
        ctx.lineWidth=2;rr(ctx,cx-s/2+2.5,cy-s/2+2.5,s-5,s-5,s*0.08);ctx.stroke();
        if(card.nullified){ctx.fillStyle='#d9a441';ctx.beginPath();ctx.arc(cx,cy,s*0.11,0,7);ctx.fill();}
      }
      if(cl.isExit&&!unlocked)drawLock(cx,cy,s);
    }
  }

  /* unlocked door glow */
  if(unlocked){
    const p=exitCenter(),pw=0.55+0.45*Math.sin(now/280);
    const g=ctx.createRadialGradient(p.x,p.y,2,p.x,p.y,CS*1.5);
    g.addColorStop(0,'rgba(255,214,110,'+(0.5*pw).toFixed(3)+')');g.addColorStop(1,'rgba(255,214,110,0)');
    ctx.fillStyle=g;ctx.fillRect(p.x-CS*1.6,p.y-CS*1.6,CS*3.2,CS*3.2);
  }

  drawChip(now);

  /* particles */
  for(const p of particles){
    const k=1-p.t/p.dur;
    if(p.type==='spark'){
      ctx.save();ctx.globalCompositeOperation='lighter';ctx.globalAlpha=k;
      ctx.fillStyle=p.col;ctx.beginPath();ctx.arc(p.x,p.y,p.r*k+0.5,0,7);ctx.fill();ctx.restore();
    }else if(p.type==='puff'){
      ctx.globalAlpha=k*0.8;ctx.fillStyle=p.col;
      ctx.beginPath();ctx.arc(p.x,p.y,p.r*(1+(1-k)*2),0,7);ctx.fill();ctx.globalAlpha=1;
    }else if(p.type==='conf'){
      ctx.save();ctx.translate(p.x,p.y);ctx.rotate(p.rot);
      ctx.globalAlpha=Math.min(1,k*3);ctx.fillStyle=p.col;
      ctx.fillRect(-p.w/2,-p.h/2,p.w,p.h*(0.5+0.5*Math.abs(Math.sin(now/90+p.rot))));
      ctx.restore();
    }else if(p.type==='ring'){
      const e=1-Math.pow(1-p.t/0.8,3);
      ctx.globalAlpha=k*0.8;ctx.strokeStyle=p.col;ctx.lineWidth=4;
      ctx.beginPath();ctx.arc(p.x,p.y,lerp(p.r0,p.r1,e),0,7);ctx.stroke();ctx.globalAlpha=1;
    }
  }

  /* floating text */
  for(const f of floats){
    const e=f.t/f.dur;
    ctx.save();ctx.globalAlpha=e<0.1?e/0.1:Math.max(0,1-Math.pow(e,2));
    ctx.font='700 '+f.size+'px Georgia,"Times New Roman",serif';
    ctx.textAlign='center';ctx.textBaseline='middle';
    const y=f.y-30*easeOutBack(Math.min(1,e));
    ctx.lineWidth=3;ctx.strokeStyle='rgba(15,20,16,.6)';ctx.strokeText(f.txt,f.x,y);
    ctx.fillStyle=f.col;ctx.fillText(f.txt,f.x,y);
    ctx.restore();
  }
  ctx.restore();

  if(flashRed>0){ctx.fillStyle='rgba(190,35,45,'+(flashRed*0.32).toFixed(3)+')';ctx.fillRect(0,0,W,H);}
  if(state==='over'){
    const a=clamp((now-overT)/700,0,1)*0.4;
    ctx.fillStyle='rgba(8,16,12,'+a.toFixed(3)+')';ctx.fillRect(0,0,W,H);
  }
}

}
function drawLock(cx,cy,s){
  ctx.save();
  ctx.fillStyle='rgba(18,20,26,.45)';rr(ctx,cx-s/2,cy-s/2,s,s,s*0.1);ctx.fill();
  const ly=cy+s*0.02;
  ctx.strokeStyle='#e8dfc8';ctx.lineWidth=Math.max(2,s*0.05);
  ctx.beginPath();ctx.arc(cx,ly,s*0.13,Math.PI,0);ctx.stroke();

  ctx.fillStyle='#23252d';ctx.beginPath();ctx.arc(cx,ly+s*0.1,s*0.045,0,7);ctx.fill();
  ctx.font='700 '+(s*0.15)+'px Georgia,serif';ctx.textAlign='center';ctx.textBaseline='middle';
  ctx.fillStyle='#f3e9cf';ctx.fillText('LOCKED',cx,cy+s*0.42);
  ctx.restore();
}
function drawChip(now){
  if(state==='menu'||!player)return;
  const {x,y}=chipPos();
  let pop=1;
  if(now<gameStart+750){pop=easeOutBack(clamp((now-gameStart-380)/260,0,1));if(pop<=0)return;}
  const dead=state==='over'?clamp((now-overT)/700,0,1):0;
  const R=CS*0.34;
  ctx.save();ctx.translate(x,y);
  ctx.fillStyle='rgba(0,0,0,.3)';ctx.beginPath();ctx.ellipse(0,R*0.55+3,R*0.95,R*0.4,0,0,7);ctx.fill();
  if(dead){ctx.rotate(-dead*1.2);ctx.scale(1,Math.max(0.18,1-dead));ctx.globalAlpha=1-dead*0.3;}
  if(player.moving&&!dead){const st=1.1;if(player.dir[0])ctx.scale(st,1/st);else ctx.scale(1/st,st);}
  ctx.scale(pop,pop);
  const gr=ctx.createRadialGradient(-R*0.3,-R*0.35,R*0.2,0,0,R*1.05);
  gr.addColorStop(0,'#d8565e');gr.addColorStop(1,'#9c2a33');
  ctx.fillStyle=gr;ctx.beginPath();ctx.arc(0,0,R,0,7);ctx.fill();
  ctx.strokeStyle='#f4ecd8';ctx.lineWidth=R*0.26;
  const seg=Math.PI*2*R/10;
  ctx.setLineDash([seg*0.5,seg*0.5]);ctx.lineDashOffset=(now/60)%seg;
  ctx.beginPath();ctx.arc(0,0,R-R*0.13,0,7);ctx.stroke();ctx.setLineDash([]);
  ctx.strokeStyle='rgba(217,164,65,.9)';ctx.lineWidth=R*0.08;
  ctx.beginPath();ctx.arc(0,0,R*0.58,0,7);ctx.stroke();
  ctx.fillStyle='#f4ecd8';ctx.beginPath();ctx.arc(0,0,R*0.34,0,7);ctx.fill();
  ctx.fillStyle='#2a2d36';suitPath(ctx,'s',0,R*0.02,R*0.34);ctx.fill();
  if(shields>0&&!dead){
    ctx.strokeStyle='rgba(233,196,105,'+(0.55+0.3*Math.sin(now/180)).toFixed(3)+')';
    ctx.lineWidth=3;ctx.setLineDash([7,8]);ctx.lineDashOffset=-now/40;
    ctx.beginPath();ctx.arc(0,0,R*1.28,0,7);ctx.stroke();ctx.setLineDash([]);
  }
  ctx.restore();
}
