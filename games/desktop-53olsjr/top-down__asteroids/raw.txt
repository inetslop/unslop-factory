  for (let i=particles.length-1;i>=0;i--){
    const p=particles[i]; p.life-=dt;
    if (p.life<=0){ particles.splice(i,1); continue; }
    if (p.type==='ring'){ p.r+=p.vr*dt; continue; }
    p.x+=p.vx*dt; p.y+=p.vy*dt; p.vx*=drag; p.vy*=drag;
  }
  if (particles.length>450) particles.splice(0, particles.length-450);
}
function updatePopups(dt){
  for (let i=popups.length-1;i>=0;i--){ const p=popups[i]; p.t+=dt; if (p.t>=p.dur) popups.splice(i,1); }
}
function updateRocks(dt){
  for (const r of rocks){ r.x+=r.vx*dt; r.y+=r.vy*dt; r.rot+=r.va*dt; wrap(r,r.r+20); }
}
function updateShip(dt){
  if (!ship.alive) return;
  if (input.left) ship.angle-=4.4*dt;
  if (input.right) ship.angle+=4.4*dt;
  if (input.thrust){
    ship.vx+=Math.cos(ship.angle)*400*SCALE*dt;
    ship.vy+=Math.sin(ship.angle)*400*SCALE*dt;
    const back=ship.angle+Math.PI;
    for (let i=0;i<2;i++){
      const a=back+(Math.random()-.5)*.7, sp=(120+Math.random()*160)*SCALE;
      particles.push({type:'dot', x:ship.x+Math.cos(ship.angle)*-14*SCALE, y:ship.y+Math.sin(ship.angle)*-14*SCALE,
        vx:Math.cos(a)*sp+ship.vx*.3, vy:Math.sin(a)*sp+ship.vy*.3,
        life:.18+Math.random()*.22, dur:.4, s:1.5+Math.random()*1.6,
        col:['#ffb54d','#ff7b2e','#ffe9a8'][i%3]});
    }
  }
  SND.thrust(input.thrust);
  const damp=Math.exp(-.45*dt);
  ship.vx*=damp; ship.vy*=damp;
  const sp=Math.hypot(ship.vx,ship.vy), MAXV=430*SCALE;
  if (sp>MAXV){ ship.vx*=MAXV/sp; ship.vy*=MAXV/sp; }
  ship.x+=ship.vx*dt; ship.y+=ship.vy*dt; wrap(ship,30);
  if (ship.invuln>0) ship.invuln-=dt;
  if (input.fire){
    fireCd-=dt;
    if (fireCd<=0){ doFire(); fireCd = pTime.rapid>0 ? .1 : .22; }
  }
}
function updateBullets(dt){
  for (let i=bullets.length-1;i>=0;i--){
    const b=bullets[i]; b.x+=b.vx*dt; b.y+=b.vy*dt; b.life-=dt; wrap(b,16);
    if (b.life<=0) bullets.splice(i,1);
  }
}
function updateSBullets(dt){
  for (let i=sBullets.length-1;i>=0;i--){
    const b=sBullets[i]; b.x+=b.vx*dt; b.y+=b.vy*dt; b.life-=dt; wrap(b,16);
    if (b.life<=0) sBullets.splice(i,1);
  }
}
function updatePowerups(dt){
  for (let i=powerups.length-1;i>=0;i--){
    const p=powerups[i];
    p.x+=p.vx*dt; p.y+=p.vy*dt; wrap(p,26);
    p.rot+=dt*1.5; p.life-=dt;
    if (p.life<=0){ powerups.splice(i,1); continue; }
    if (ship.alive && Math.hypot(p.x-ship.x,p.y-ship.y)<26*SCALE){
      pTime[p.type] = CHIP_MAX[p.type];
      powerups.splice(i,1);
      SND.powerup();
      popup(ship.x,ship.y-22*SCALE,'+ '+POW[p.type].name,POW[p.type].col);
      burst(ship.x,ship.y,POW[p.type].col,12,.8);
    }
  }
}
function updateSaucer(dt){
  if (!saucer) return;
  const s=saucer;
  s.t+=dt;
  s.x+=s.dir*(95+wave*4)*SCALE*dt;
  s.y=s.baseY+Math.sin(s.t*1.7)*55*SCALE;
  if (ship.alive){
    s.fireT-=dt;
    if (s.fireT<=0){ saucerShoot(); s.fireT = s.small ? 1.1+Math.random()*.8 : 1.6+Math.random()*1.2; }
  }
  if ((s.dir>0&&s.x>W+60)||(s.dir<0&&s.x<-60)){ SND.saucerStop(); saucer=null; scheduleNext(); }
}
function splitRock(i){
  const r=rocks[i];
  addScore(R_SCORE[r.size], r.x, r.y);
  stats.rocks++;
  burst(r.x,r.y,r.col,10+r.size*8,(.5+.4*r.size),r.vx*.4,r.vy*.4);
  ring(r.x,r.y,r.col,(90+r.size*70)*SCALE,.4);
  SND.boom(r.size);
  setShake(r.size===3?7:4, r.size===3?.35:.2);
  const dropC = r.size===1?.1 : r.size===2?.06 : .03;
  if (Math.random()<dropC) spawnPowerup(r.x,r.y);
  if (r.size>1){
    const pa=Math.atan2(r.vy,r.vx);
    for (let k=0;k<2;k++){
      const a=pa+(Math.random()-.5)*1.8, sp=rockSpeed(r.size-1)*(.7+Math.random()*.6);
      rocks.push(makeRock(r.size-1, r.x+(Math.random()-.5)*10, r.y+(Math.random()-.5)*10, Math.cos(a)*sp, Math.sin(a)*sp));
    }
  }
  rocks.splice(i,1);
}
function collisions(){
  for (let i=bullets.length-1;i>=0;i--){
    const b=bullets[i]; let hit=false;
    for (let j=rocks.length-1;j>=0 && !hit;j--){
      const r=rocks[j];
      if (Math.hypot(b.x-r.x,b.y-r.y) < r.r*.82+3){ splitRock(j); hit=true; }
    }
    if (!hit && saucer && Math.hypot(b.x-saucer.x,b.y-saucer.y)<saucer.r*1.05){ killSaucer(); hit=true; }
    if (hit) bullets.splice(i,1);
  }
  const prot = ship.invuln>0 || pTime.shield>0;
  if (ship.alive && !prot){
    for (const r of rocks){
      if (Math.hypot(ship.x-r.x,ship.y-r.y) < r.r*.8+9*SCALE){ killShip(); break; }
    }
  }
  if (ship.alive){
    for (let i=sBullets.length-1;i>=0;i--){
      const b=sBullets[i];
      if (Math.hypot(b.x-ship.x,b.y-ship.y) < 10*SCALE){
        sBullets.splice(i,1);
        if (prot){ burst(ship.x,ship.y,'#6fd3ff',8,.7); } else { killShip(); break; }
      }
    }
    if (saucer && Math.hypot(ship.x-saucer.x,ship.y-saucer.y) < saucer.r+12*SCALE){
      killSaucer();
      if (!prot) killShip();
    }
  }
}
function waveClearCheck(dt){
  if (transitioning){
    transT-=dt;
    if (transT<=0){ transitioning=false; wave++; spawnWave(wave); banner('WAVE '+wave,''); }
    return;
  }
  if (rocks.length===0){
    transitioning=true; transT=1.5;
    const bonus=50*wave;
    addScore(bonus, W/2, H*.4);
    banner('FIELD CLEARED','BONUS +'+bonus);
    SND.waveSnd();
    clearSaucer();
  }
}
function update(dt){
  updateStarsCam(dt);
  updateParticles(dt);
  updatePopups(dt);
  shakeT=Math.max(0,shakeT-dt);
  flashA=Math.max(0,flashA-dt*1.5);
  if (state==='menu' || state==='over'){ updateRocks(dt); return; }
  if (state!=='playing') return;

  for (const k in pTime) pTime[k]=Math.max(0,pTime[k]-dt);

  updateShip(dt);
  updateBullets(dt);
  updateRocks(dt);
  updateSaucer(dt);
  updateSBullets(dt);
  updatePowerups(dt);
  collisions();

  if (!ship.alive){
    if (lives>0 && deathT>0){ deathT-=dt; if (deathT<=0) respawn(); }
    else if (lives<=0 && overTimer>0){ overTimer-=dt; if (overTimer<=0) doGameOver(); }
  }
  waveClearCheck(dt);

  if (!saucer && !transitioning && state==='playing'){
    nextSaucer-=dt;
    if (nextSaucer<=0) spawnSaucer();
  }
}

/* ================= RENDER ================= */
function drawRocks(){
  for (const r of rocks){
    ctx.save(); ctx.translate(r.x,r.y); ctx.rotate(r.rot);
    ctx.beginPath();
    ctx.moveTo(r.pts[0][0],r.pts[0][1]);
    for (let i=1;i<r.pts.length;i++) ctx.lineTo(r.pts[i][0],r.pts[i][1]);
    ctx.closePath();
    ctx.shadowColor=r.col; ctx.shadowBlur=10;
    ctx.fillStyle=r.fill; ctx.fill();
    ctx.strokeStyle=r.col; ctx.lineWidth=r.size===3?2.4:2; ctx.stroke();
    ctx.restore();
  }
}
function drawBullets(){
  for (const b of bullets){
    ctx.save();
    ctx.shadowColor='#ffb54d'; ctx.shadowBlur=8;
    ctx.strokeStyle='rgba(255,233,168,.7)'; ctx.lineWidth=2;
    ctx.beginPath(); ctx.moveTo(b.x,b.y); ctx.lineTo(b.x-b.vx*.018,b.y-b.vy*.018); ctx.stroke();
    ctx.fillStyle='#fff2c9';
    ctx.beginPath(); ctx.arc(b.x,b.y,2.6*SCALE,0,TAU); ctx.fill();
    ctx.restore();
  }
}
function drawSBullets(){
  for (const b of sBullets){
    ctx.save();
    ctx.shadowColor='#ff5fa8'; ctx.shadowBlur=9;
    ctx.fillStyle='#ff8fc0';
    ctx.beginPath(); ctx.arc(b.x,b.y,3.6*SCALE,0,TAU); ctx.fill();
    ctx.restore();
  }
}
function drawSaucer(t){
  if (!saucer) return;
  const s=saucer, r=s.r;
  ctx.save(); ctx.translate(s.x,s.y);
  ctx.strokeStyle='#ff5fa8'; ctx.shadowColor='#ff5fa8'; ctx.shadowBlur=14; ctx.lineWidth=2;
  ctx.beginPath(); ctx.arc(0,-r*.25,r*.55,Math.PI,0); ctx.stroke();
  ctx.beginPath(); ctx.ellipse(0,r*.15,r,r*.42,0,0,TAU); ctx.stroke();
  for (let i=0;i<3;i++){
    const on=Math.sin(t*6+i*2)>0;
    ctx.globalAlpha=on?.9:.25;
    ctx.fillStyle='#ffd7ea';
    ctx.beginPath(); ctx.arc(-r*.5+i*r*.5, r*.15+r*.38, 1.8*SCALE, 0, TAU); ctx.fill();
  }
  ctx.restore();
}
function drawPowerups(t){
  for (const p of powerups){
    const info=POW[p.type];
    ctx.save(); ctx.translate(p.x,p.y);
    if (p.life<3) ctx.globalAlpha = Math.sin(t*16)>0 ? 1 : .25;
    ctx.rotate(p.rot);
    const pulse=1+Math.sin(t*5)*.08, R=12*SCALE*pulse;
    ctx.strokeStyle=info.col; ctx.shadowColor=info.col; ctx.shadowBlur=12; ctx.lineWidth=2;
    ctx.beginPath();
    for (let i=0;i<6;i++){
      const a=i/6*TAU;
      if (i===0) ctx.moveTo(Math.cos(a)*R,Math.sin(a)*R); else ctx.lineTo(Math.cos(a)*R,Math.sin(a)*R);
    }
    ctx.closePath(); ctx.stroke();
    ctx.rotate(-p.rot);
    ctx.shadowBlur=6; ctx.fillStyle=info.col;
    ctx.font='700 '+Math.round(11*SCALE+2)+'px ui-monospace,Menlo,monospace';
    ctx.textAlign='center'; ctx.textBaseline='middle';
    ctx.fillText(info.letter,0,1);
    ctx.restore();
  }
}
function drawShip(t){
  if (!ship.alive) return;
  const prot = ship.invuln>0 || pTime.shield>0;
  ctx.save(); ctx.translate(ship.x,ship.y);
  if (ship.invuln>0 && Math.sin(t*24)<0) ctx.globalAlpha=.35;
  ctx.rotate(ship.angle);
  const k=SCALE;
  if (input.thrust && state==='playing'){
    const fl=(10+Math.random()*12)*k;
    ctx.strokeStyle='#ffb54d'; ctx.shadowColor='#ff7b2e'; ctx.shadowBlur=10; ctx.lineWidth=2.5*k;
    ctx.beginPath(); ctx.moveTo(-12*k,-3.5*k); ctx.lineTo(-12*k-fl,0); ctx.lineTo(-12*k,3.5*k); ctx.stroke();
    ctx.strokeStyle='#ffe9a8'; ctx.lineWidth=1.4*k;
    ctx.beginPath(); ctx.moveTo(-12*k,-1.6*k); ctx.lineTo(-12*k-fl*.55,0); ctx.lineTo(-12*k,1.6*k); ctx.stroke();
  }
  ctx.strokeStyle='#7df9ff'; ctx.shadowColor='#7df9ff'; ctx.shadowBlur=12; ctx.lineWidth=2*k;
  ctx.fillStyle='rgba(125,249,255,.08)';
  ctx.beginPath();
  ctx.moveTo(17*k,0); ctx.lineTo(-6*k,-9.5*k); ctx.lineTo(-3*k,0); ctx.lineTo(-6*k,9.5*k); ctx.closePath();
  ctx.fill(); ctx.stroke();
  ctx.restore();
  if (prot){
    ctx.save();
    ctx.strokeStyle='rgba(125,249,255,.55)'; ctx.shadowColor='#7df9ff'; ctx.shadowBlur=14; ctx.lineWidth=1.6;
    ctx.beginPath(); ctx.arc(ship.x,ship.y,(21+Math.sin(t*6)*2.5)*k,0,TAU); ctx.stroke();
    ctx.restore();
  }
}
function drawParticles(){
  for (const p of particles){
    const a=Math.max(0,p.life/p.dur);
    ctx.globalAlpha=a;
    if (p.type==='ring'){
      ctx.strokeStyle=p.col; ctx.lineWidth=2*a+.5;
      ctx.beginPath(); ctx.arc(p.x,p.y,p.r,0,TAU); ctx.stroke();
    } else if (p.type==='spark'){
      ctx.strokeStyle=p.col; ctx.lineWidth=1.6;
      ctx.beginPath(); ctx.moveTo(p.x,p.y); ctx.lineTo(p.x-p.vx*.025,p.y-p.vy*.025); ctx.stroke();
    } else {
      ctx.fillStyle=p.col;
      ctx.fillRect(p.x-p.s/2,p.y-p.s/2,p.s,p.s);
    }
  }
  ctx.globalAlpha=1;
}
function drawPopups(){
  const fs=Math.round(13*SCALE+3);
  ctx.font='700 '+fs+'px ui-monospace,Menlo,Consolas,monospace';
  ctx.textAlign='center'; ctx.textBaseline='middle';
  for (const p of popups){
    const a=1-p.t/p.dur;
    ctx.globalAlpha=a;
    ctx.fillStyle=p.col;
    ctx.fillText(p.txt,p.x,p.y-p.t*26);
  }
  ctx.globalAlpha=1;
}
function render(t){
  ctx.setTransform(DPR,0,0,DPR,0,0);
  ctx.save();
  if (shakeT>0){
    const k=shakeT/shakeDur;
    ctx.translate((Math.random()*2-1)*shakeMag*k,(Math.random()*2-1)*shakeMag*k);
  }
  ctx.fillStyle='#05070f'; ctx.fillRect(-30,-30,W+60,H+60);
  if (neb) ctx.drawImage(neb, -W*.03+Math.sin(t*.05)*14, -H*.03+Math.cos(t*.04)*10, W*1.06, H*1.06);
  for (const s of stars){
    const x=((s.x+camX*s.f)%W+W)%W, y=((s.y+camY*s.f)%H+H)%H;
    ctx.globalAlpha=s.a*(.65+.35*Math.sin(t*s.tw+s.ph));
    ctx.fillStyle=s.c;
    ctx.fillRect(x,y,s.r,s.r);
  }
  ctx.globalAlpha=1;
  drawPowerups(t);
  drawRocks();
  drawSaucer(t);
  drawBullets();
  drawSBullets();
  if (state==='playing'||state==='paused') drawShip(t);
  drawParticles();
  drawPopups();
  ctx.restore();
  if (flashA>0){
    const g=ctx.createRadialGradient(W/2,H/2,Math.min(W,H)*.25,W/2,H/2,Math.max(W,H)*.72);
    g.addColorStop(0,'rgba(255,64,96,0)');
    g.addColorStop(1,'rgba(255,64,96,'+(flashA*.5).toFixed(3)+')');
    ctx.fillStyle=g; ctx.fillRect(0,0,W,H);
  }
}

    if (on) c.querySelector('i').style.width=(tt/CHIP_MAX[k]*100)+'%';
  }
}
