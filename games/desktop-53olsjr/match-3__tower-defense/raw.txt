function startResolve(grps){
  const removed=new Set();
  for(const g of grps)for(const p of g.cells)removed.add(p.r*COLS+p.c);
  for(const g of grps){queueGroup(g);popFx(g);}
  sPop(board.chain);
  let any=false;const gg=board.g;
  for(let c=0;c<COLS;c++){
    let write=ROWS-1;
    for(let r=ROWS-1;r>=0;r--){
      if(removed.has(r*COLS+c))continue;
      const gem=gg[r][c];
      if(write!==r){gg[write][c]=gem;gg[r][c]=null;gem.yOff=r-write;any=true;}
      write--;
    }
    const empties=write+1;
    for(let i=0;i<empties;i++){gg[write-i][c]={c:irand(NC),yOff:-empties,vy:0,squash:0,flash:0};any=true;}
  }
  if(any)board.state='fall';else afterLand();
}

function updateBoard(dt){

    board.t+=dt/board.dur;const sw=board.swap;
    if(board.t>=1){
      swapCells(sw.a,sw.b);
      if(sw.phase==='fwd'){
        const grps=findMatches();
        if(grps.length){board.chain=1;startResolve(grps);}
        else{sw.phase='back';board.t=0;}
      }else{board.state='idle';sBad();}
    }
  }else if(board.state==='fall'){
    let done=true;
    for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++){
      const g=board.g[r][c];
      if(g.yOff<0||g.vy>0){
        g.vy+=130*dt;g.yOff+=g.vy*dt;
        if(g.yOff>=0){g.yOff=0;g.vy=0;g.squash=1;}else done=false;
      }

  }

const ETYPES={
  grunt:{hp:26,spd:44,dmg:1,score:10,r:11,col:'#b9d98a'},
  runner:{hp:15,spd:82,dmg:1,score:15,r:9,col:'#ffb36b'},
  brute:{hp:95,spd:27,dmg:2,score:30,r:15,col:'#c3ccd6'},
  boss:{hp:700,spd:19,dmg:4,score:500,r:24,col:'#b06bff'}
};
const FLAV=['THEY CREEP FROM THE DARK','FASTER NOW · HUNGERIER','THE DEEP ONES FOLLOW','THEY LEARN YOUR PATTERN'];
function waveMul(n){return Math.pow(1.14,n-1)*(1+.12*(n-1));}
function buildWave(n){
  const list=[];const count=5+n*2;
  if(n%5===0){for(let i=0;i<count/2|0;i++)list.push(i%3===2?'brute':'grunt');list.push('boss');return list;}
  for(let i=0;i<count;i++){
    const r=Math.random();let t='grunt';
    if(n>=2&&r<Math.min(.4,.12+n*.02))t='runner';
    else if(n>=3&&r>.85-Math.min(.2,n*.01))t='brute';
    list.push(t);
  }return list;
}
function spawnEnemy(type){
  const d=ETYPES[type],mul=waveMul(S.wave);
  const hp=type==='boss'?Math.round(700*Math.pow(1.28,S.wave/5-1)):Math.round(d.hp*mul);
  const p0=posAt(0);
  enemies.push({type,hp,maxhp:hp,spd:d.spd,dmg:d.dmg,score:d.score,r:d.r,col:d.col,
    f:0,x:p0.x,y:p0.y,ang:p0.ang,flash:0,slowT:0,poisT:0,poisDps:0,wob:rand(TAU),dead:false,gone:false});
}
function aimEnemy(){let b=null,bf=-1;for(const e of enemies)if(!e.dead&&e.f>bf){bf=e.f;b=e;}return b;}
function hitEnemy(e,d){
  if(e.dead)return;
  e.hp-=d;e.flash=1;
  if(d>=10)addText(e.x+rand(-8,8),e.y-e.r-4,String(Math.round(d)),'rgba(255,255,255,.8)',10,.45);
  if(e.hp<=0)killEnemy(e);
}
function killEnemy(e){
  e.dead=true;S.score+=e.score;
  addText(e.x,e.y-e.r-6,'+'+e.score,'#ffd97a',13,.7);
  const n=e.type==='boss'?26:10;
  for(let i=0;i<n;i++){const a=rand(TAU),s=rand(50,200);
    parts.push({x:e.x,y:e.y,vx:Math.cos(a)*s,vy:Math.sin(a)*s,t:0,life:.5,size:rand(2,4.5),col:Math.random()<.3?'#fff':e.col,grav:220});}
  parts.push({x:e.x,y:e.y,type:'ring',t:0,life:.35,r0:e.r*.5,r1:e.r*(e.type==='boss'?3.4:2.2),col:e.col});
  if(e.type==='boss'){S.shake=12;sBomb();}else sDie();
}
function explode(x,y,dmg,r){
  S.shake=Math.min(14,S.shake+6);sBomb();
  parts.push({x,y,type:'ring',t:0,life:.38,r0:8,r1:r+20,col:'#ffd27a'});
  parts.push({x,y,type:'flash',t:0,life:.22,r:r*.7,col:'#ffe9b0'});
  for(let i=0;i<14;i++){const a=rand(TAU),s=rand(60,240);
    parts.push({x,y,vx:Math.cos(a)*s,vy:Math.sin(a)*s-30,t:0,life:.5,size:rand(2,4.5),col:i%2?'#ffb347':'#ff6b3d',grav:260});}
  for(const e of enemies)if(!e.dead&&Math.hypot(e.x-x,e.y-y)<r+e.r)hitEnemy(e,dmg);
}
function fireArc(dmg,k){
  const alive=enemies.filter(e=>!e.dead).sort((a,b)=>b.f-a.f);
  if(!alive.length)return;
  const hit=[alive[0]];let curE=alive[0];
  while(hit.length<k){
    let bestE=null,bd=230;
    for(const e of alive){if(hit.includes(e))continue;const d=Math.hypot(e.x-curE.x,e.y-curE.y);if(d<bd){bd=d;bestE=e;}}
    if(!bestE)break;hit.push(bestE);curE=bestE;
  }
  const anchors=[{x:turret.x,y:turret.y}];anchors.push(...hit.map(e=>({x:e.x,y:e.y})));
  const poly=[];
  for(let i=0;i<anchors.length-1;i++){
    const a=anchors[i],b=anchors[i+1],dx=b.x-a.x,dy=b.y-a.y,L=Math.hypot(dx,dy)||1,nx=-dy/L,ny=dx/L;
    poly.push(a);
    for(let s2=1;s2<5;s2++){const t=s2/5;poly.push({x:lerp(a.x,b.x,t)+nx*rand(-7,7),y:lerp(a.y,b.y,t)+ny*rand(-7,7)});}
  }
  poly.push(anchors[anchors.length-1]);
  beams.push({poly,t:0,life:.16});
  for(const e of hit)hitEnemy(e,dmg);
  sZap();S.shake=Math.min(14,S.shake+2);
}
function spawnProj(target,type,dmg,r,dot){
  projs.push({x:turret.x,y:turret.y,tx:target.x,ty:target.y,tg:target,type,dmg,r,dot,spd:type===3?380:440,sx:1,sy:0});
  turret.recoil=1;turret.lastCol=type;
}
function executePending(){
  while(pending.length&&pending[0].at<=S.time){
    const s=pending.shift();
    if(s.kind==='proj'){const t=aimEnemy();if(t)spawnProj(t,s.type,s.dmg,0,s.dot);else parts.push({x:turret.x,y:turret.y,vx:rand(-30,30),vy:rand(-30,30),t:0,life:.3,size:2,col:'#8fae97',grav:0});}
    else if(s.kind==='bomb'){const t=aimEnemy();if(t)spawnProj(t,3,s.dmg,s.r,0);else parts.push({x:turret.x,y:turret.y,vx:rand(-30,30),vy:rand(-30,30),t:0,life:.3,size:2,col:'#8fae97',grav:0});}
    else if(s.kind==='arc')fireArc(s.dmg,s.k);
  }
}

/* ================= update ================= */
function update(dt){
  S.time+=dt;
  // phases
  if(phase==='prewave'){
    phaseT-=dt;
    if(phaseT<=0){phase='spawning';spawnQ=buildWave(S.wave);spawnT=.5;sWave();
      banner('WAVE '+S.wave,S.wave%5===0?'THE MIRE WARDEN RISES':FLAV[(S.wave-1)%4]);}
  }else if(phase==='spawning'){
    spawnT-=dt;
    if(spawnT<=0&&spawnQ.length){spawnEnemy(spawnQ.shift());spawnT=Math.max(.32,1-S.wave*.04);}
    if(!spawnQ.length)phase='active';
  }else{
    if(!enemies.length){
      const bonus=25*S.wave;S.score+=bonus;
      addText(fieldRect.x+fieldRect.w/2,fieldRect.y+fieldRect.h*.4,'WAVE CLEAR +'+bonus,'#ffd97a',18,.9);
      if(S.baseHP<S.maxHP){S.baseHP++;sHeal();const kp=pts[pts.length-1];addText(kp.x,kp.y-40,'KEEP +1','#59e6c9',13,.8);}
      S.wave++;phase='prewave';phaseT=4;
    }
  }
  updateBoard(dt);
  // enemies
  for(let i=enemies.length-1;i>=0;i--){
    const e=enemies[i];
    if(e.dead||e.gone)continue;
    if(e.poisT>0){
      e.poisT-=dt;e.hp-=e.poisDps*dt;
      if(Math.random()<dt*7)parts.push({x:e.x+rand(-6,6),y:e.y,vx:0,vy:40,t:0,life:.5,size:2,col:'#8dff57',grav:120});
      if(e.hp<=0){killEnemy(e);continue;}
    }
    if(e.slowT>0)e.slowT-=dt;
    const sp=e.spd*(e.slowT>0?.55:1);
    e.f+=sp*dt/totalLen;
    e.flash=Math.max(0,e.flash-dt*4);
    if(e.f>=1){
      e.gone=true;S.baseHP-=e.dmg;S.shake=9;S.flash=.8;sLeak();
      const kp=pts[pts.length-1];
      parts.push({x:kp.x,y:kp.y,type:'ring',t:0,life:.4,r0:10,r1:46,col:'#ff5a5a'});
      addText(kp.x,kp.y-34,'-'+e.dmg+' HP','#ff6b6b',16,.8);
      if(S.baseHP<=0){S.baseHP=0;setPips(0);gameOver();return;}
    }else{const p=posAt(e.f);e.x=p.x;e.y=p.y;e.ang=p.ang;}
  }
  enemies=enemies.filter(e=>!e.dead&&!e.gone);
  executePending();
  // projectiles
  for(let i=projs.length-1;i>=0;i--){
    const p=projs[i];
    if(p.tg&&!p.tg.dead){p.tx=p.tg.x;p.ty=p.tg.y;}
    const dx=p.tx-p.x,dy=p.ty-p.y,d=Math.hypot(dx,dy);
    if(d<13){
      if(p.type===3)explode(p.x,p.y,p.dmg,p.r);
      else{
        if(p.tg&&!p.tg.dead){
          hitEnemy(p.tg,p.dmg);
          if(p.type===1&&p.tg.slowT!==undefined)p.tg.slowT=Math.max(p.tg.slowT,2.6);
          if(p.type===2){p.tg.poisT=3.5;p.tg.poisDps=p.dot;}
        }
        const col=GEMS[p.type].col;
        for(let k=0;k<5;k++){const a=rand(TAU),s=rand(40,120);
          parts.push({x:p.x,y:p.y,vx:Math.cos(a)*s,vy:Math.sin(a)*s,t:0,life:.3,size:2,col,grav:160});}
        sHitT(p.type);
      }
      projs.splice(i,1);continue;
    }
    const s=p.spd*dt/d;p.x+=dx*s;p.y+=dy*s;p.sx=dx/d;p.sy=dy/d;
  }
  // turret aim/recoil
  turret.recoil=Math.max(0,turret.recoil-dt*6);
  const ae=aimEnemy();
  if(ae)turret.aim=lerpAng(turret.aim,Math.atan2(ae.y-turret.y,ae.x-turret.x),1-Math.pow(.002,dt));
  updateFx(dt);
}
function fxOnly(dt){turret.recoil=Math.max(0,turret.recoil-dt*6);updateFx(dt);}
function updateFx(dt){
  for(let i=parts.length-1;i>=0;i--){const p=parts[i];p.t+=dt;
    if(p.t>=p.life){parts.splice(i,1);continue;}
    if(!p.type&&p!==null){p.x+=p.vx*dt;p.y+=p.vy*dt;if(p.grav)p.vy+=p.grav*dt;}}
  for(let i=texts.length-1;i>=0;i--){texts[i].t+=dt;if(texts[i].t>=texts[i].life)texts.splice(i,1);}
  for(let i=beams.length-1;i>=0;i--){beams[i].t+=dt;if(beams[i].t>=beams[i].life)beams.splice(i,1);}
  if(bannerMsg){bannerMsg.t+=dt;if(bannerMsg.t>=bannerMsg.dur)bannerMsg=null;}
}

function drawFx(){
  // projectiles
  for(const p of projs){
    const col=GEMS[p.type].col;
    ctx.globalCompositeOperation='lighter';
    ctx.strokeStyle=col;ctx.globalAlpha=.5;ctx.lineWidth=3;
    ctx.beginPath();ctx.moveTo(p.x-p.sx*14,p.y-p.sy*14);ctx.lineTo(p.x,p.y);ctx.stroke();
    ctx.globalAlpha=1;
    let g=ctx.createRadialGradient(p.x,p.y,0,p.x,p.y,p.type===3?12:9);
    g.addColorStop(0,'#fff');g.addColorStop(.4,col);g.addColorStop(1,'rgba(0,0,0,0)');
    ctx.fillStyle=g;ctx.beginPath();ctx.arc(p.x,p.y,p.type===3?12:9,0,TAU);ctx.fill();
    if(p.type===3){
      ctx.strokeStyle='rgba(255,198,58,.8)';ctx.lineWidth=1.5;
      ctx.beginPath();ctx.arc(p.x,p.y,7+Math.sin(T*22)*1.6,0,TAU);ctx.stroke();
    }
    ctx.globalCompositeOperation='source-over';
  }
  // beams
  for(const bm of beams){
    const a=1-bm.t/bm.life;
    ctx.save();ctx.globalCompositeOperation='lighter';
    const passes=[[9,'rgba(255,92,232,'+(.28*a)+')'],[3.5,'rgba(255,170,245,'+(.85*a)+')'],[1.4,'rgba(255,255,255,'+a+')']];
    for(const[lw,col]of passes){
      ctx.lineWidth=lw;ctx.strokeStyle=col;ctx.beginPath();
      bm.poly.forEach((p,i)=>i?ctx.lineTo(p.x,p.y):ctx.moveTo(p.x,p.y));ctx.stroke();
    }
    ctx.restore();
  }
  // particles
  ctx.save();ctx.globalCompositeOperation='lighter';
  for(const p of parts){
    const e=p.t/p.life,al=1-e;
    if(p.type==='ring'){
      ctx.strokeStyle=p.col;ctx.globalAlpha=al*.85;ctx.lineWidth=3*(1-e)+1;
      ctx.beginPath();ctx.arc(p.x,p.y,lerp(p.r0,p.r1,1-Math.pow(1-e,3)),0,TAU);ctx.stroke();
    }else if(p.type==='flash'){
      let g=ctx.createRadialGradient(p.x,p.y,0,p.x,p.y,p.r);
      g.addColorStop(0,'rgba(255,240,210,'+(al*.7)+')');g.addColorStop(1,'rgba(255,240,210,0)');
      ctx.fillStyle=g;ctx.beginPath();ctx.arc(p.x,p.y,p.r,0,TAU);ctx.fill();
    }else{
      ctx.globalAlpha=al;ctx.fillStyle=p.col;
      ctx.beginPath();ctx.arc(p.x,p.y,Math.max(.5,p.size*(1-e*.6)),0,TAU);ctx.fill();
    }
  }
  // fireflies
  for(const a of amb){
    const al=Math.sin(Math.PI*a.t/a.life)*.7;
    ctx.globalAlpha=al;ctx.fillStyle=a.col;
    ctx.beginPath();ctx.arc(a.x,a.y,1.6,0,TAU);ctx.fill();
  }
  ctx.restore();ctx.globalAlpha=1;
}

let last=0;
function frame(ts){
  requestAnimationFrame(frame);
  const dt=clamp((ts-last)/1000,0,.033);last=ts;T+=dt;
  S.shake=Math.max(0,S.shake-45*dt);
  S.flash=Math.max(0,S.flash-dt*1.6);
  // ambient fireflies
  if(amb.length<34&&Math.random()<dt*7)
    amb.push({x:fieldRect.x+rand(fieldRect.w),y:fieldRect.y+rand(fieldRect.h),vx:rand(-8,8),vy:rand(-14,-4),t:0,life:rand(4,7),col:Math.random()<.5?'#ffd97a':'#8ffce0'});
  for(let i=amb.length-1;i>=0;i--){const a=amb[i];a.t+=dt;a.x+=a.vx*dt;a.y+=a.vy*dt;if(a.t>=a.life)amb.splice(i,1);}
  if(S.state==='play'&&!S.paused)update(dt);
  else if(S.state==='over')fxOnly(dt);

  ctx.setTransform(dpr,0,0,dpr,0,0);
  ctx.clearRect(0,0,W,H);
  const sx=(Math.random()*2-1)*S.shake,sy=(Math.random()*2-1)*S.shake;
  ctx.save();ctx.translate(sx,sy);
  if(bg)ctx.drawImage(bg,0,0,W,H);
  drawEntry();drawKeep();
  for(const e of enemies)if(!e.gone)drawEnemy(e);
  drawTurret();
  drawFx();
  drawBoard();
  drawTexts();
  drawBanner();
  ctx.restore();
  if(S.flash>0){
    const g=ctx.createRadialGradient(W/2,H/2,Math.min(W,H)*.3,W/2,H/2,Math.max(W,H)*.7);
    g.addColorStop(0,'rgba(255,40,40,0)');g.addColorStop(1,'rgba(255,40,40,'+(S.flash*.4)+')');
    ctx.fillStyle=g;ctx.fillRect(0,0,W,H);
  }
  hud();
}
