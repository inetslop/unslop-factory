function pointAt(p){ // p in cell units along path; returns {x,y,a}
  if(p<=0){p=0;}
  let d=p;
  for(let i=0;i<WAYPTS.length-1;i++){
    const [x1,y1]=WAYPTS[i],[x2,y2]=WAYPTS[i+1];
    const len=Math.abs(x2-x1)+Math.abs(y2-y1);

      return {x:x1+(x2-x1)*t, y:y1+(y2-y1)*t, a:Math.atan2(y2-y1,x2-x1)};

}

  for(let i=0;i<6;i++)part({type:'smoke',x,y,vx:rand(-16,16),vy:rand(-34,-10),life:rand(.5,.9),max:.9,size:rand(7,13),color:'90,85,80'});
  ringFx(x,y,r,'255,170,80',5,tw(360));}
function deathFx(e,x,y){const c=e.color;for(let i=0;i<9;i++){const a=rand(0,TAU),s=rand(30,140);part({type:'shard',x,y,vx:Math.cos(a)*s,vy:Math.sin(a)*s-60,g:380,life:rand(.4,.75),max:.75,size:rand(2.5,5),color:c,rot:rand(0,TAU),vr:rand(-9,9)});}
  for(let i=0;i<4;i++)part({type:'smoke',x,y,vx:rand(-14,14),vy:rand(-30,-8),life:.6,max:.6,size:rand(5,9),color:'70,70,75'});

function snowBurst(x,y){for(let i=0;i<6;i++){const a=rand(0,TAU);part({type:'snow',x,y,vx:Math.cos(a)*rand(20,70),vy:Math.sin(a)*rand(20,70)-20,life:rand(.4,.8),max:.8,size:rand(1.5,3),ph:rand(0,9)});}}

const sleep=ms=>new Promise(r=>setTimeout(r,Math.max(24,ms/G.speed)));
const tw=ms=>Math.max(24,ms/G.speed);
const ABORT={aborted:true};
function towerAt(c,r){return G.towers.find(t=>t.col===c&&t.row===r);}
function validCell(cs){return !PKEY.has(cs.c+','+cs.r)&&!towerAt(cs.c,cs.r);}
function towerPx(t){return{x:ox+(t.col+.5)*cell,y:oy+(t.row+.5)*cell};}
function fpx(e){const p=pointAt(e.pos);return{x:ox+(p.x+.5)*cell,y:oy+(p.y+.5)*cell};}
function makeFoe(k){const type=WKEY[k],d=FOES[type];
  const hp=Math.round(d.hp*(type==='boss'?1:(1+(G.wave-1)*.09)));
  return{type,hp,max:hp,spd:d.spd,armor:d.armor,r:d.r,color:d.color,pos:0,drawPos:0,chillT:0,flash:0,frozenNow:false,seed:rand(0,9),alive:true,anim:null};}

/* ================= combat ================= */
function pickFirst(t,rng){let best=null;for(const e of G.enemies){if(!e.alive)continue;const p=pointAt(e.pos);if(Math.hypot(p.x-t.col,p.y-t.row)<=rng&&(!best||e.pos>best.pos))best=e;}return best;}
function hit(e,d,color){if(!e||!e.alive)return;e.hp-=d;e.flash=1;const p=fpx(e);addText(p.x,p.y-e.r*cell-6,d,color);if(e.hp<=0)killFoe(e);}
function killFoe(e){e.alive=false;const i=G.enemies.indexOf(e);if(i>=0)G.enemies.splice(i,1);G.kills++;
  const rw=FOES[e.type].reward;G.gold+=rw;G.earned+=rw;goldPop();
  const p=fpx(e);addText(p.x,p.y-e.r*cell-16,'+'+rw,'#e8c25a');deathFx(e,p.x,p.y);SFX.coin();}

async function fireTower(t){
  if(G.runId!==G._run)throw ABORT;
  const d=TOWERS[t.type],l=t.level-1,tp=towerPx(t);
  if(t.type==='arrow'){
    const e=pickFirst(t,d.range[l]);if(!e)return;
    t.aim=Math.atan2(pointAt(e.pos).y-t.row,pointAt(e.pos).x-t.col);await sleep(150);
    if(G.runId!==G._run)throw ABORT;
    const bx=tp.x+Math.cos(t.angle)*cell*.3,by=tp.y+Math.sin(t.angle)*cell*.3,p1=fpx(e);
    t.flashT=1;SFX.arrow();
    G.proj.push({kind:'arrow',x0:bx,y0:by,x1:p1.x,y1:p1.y,t0:now(),dur:tw(170)});
    await sleep(170);if(G.runId!==G._run)throw ABORT;
    hit(e,Math.max(1,d.dmg[l]-e.armor),'#ffffff');sparkAt(p1.x,p1.y,'62,193,176');
  }else if(t.type==='cannon'){
    const e=pickFirst(t,d.range[l]);if(!e)return;
    t.aim=Math.atan2(pointAt(e.pos).y-t.row,pointAt(e.pos).x-t.col);await sleep(140);
    if(G.runId!==G._run)throw ABORT;
    const bx=tp.x+Math.cos(t.angle)*cell*.3,by=tp.y+Math.sin(t.angle)*cell*.3,p1=fpx(e);
    t.recoil=1;SFX.cannon();dust(bx,by,'120,95,60');
    G.proj.push({kind:'ball',x0:bx,y0:by,x1:p1.x,y1:p1.y,t0:now(),dur:tw(400)});
    await sleep(400);if(G.runId!==G._run)throw ABORT;
    boomFx(p1.x,p1.y,d.splash[l]*cell);SFX.boom();shake(5,.25);
    for(const f of [...G.enemies]){const fp=fpx(f);if(Math.hypot(fp.x-p1.x,fp.y-p1.y)<=d.splash[l]*cell)hit(f,Math.max(1,d.dmg[l]-f.armor),'#ffb36b');}
  }else if(t.type==='frost'){
    const list=G.enemies.filter(e=>e.alive&&Math.hypot(pointAt(e.pos).x-t.col,pointAt(e.pos).y-t.row)<=d.range[l]).sort((a,b)=>b.pos-a.pos).slice(0,d.hits[l]);
    if(!list.length)return;
    ringFx(tp.x,tp.y,d.range[l]*cell*.5,'140,220,255',3,tw(300));SFX.frost();
    for(const f of list){const fp=fpx(f);snowBurst(fp.x,fp.y);f.chillT=Math.max(f.chillT,2);hit(f,d.dmg[l],'#9fe4ff');if(f.alive&&Math.max(0,f.spd-(f.chillT>0?1:0))===0)SFX.freeze();await sleep(95);if(G.runId!==G._run)throw ABORT;}
  }else{ /* tesla */
    const start=pickFirst(t,d.range[l]);if(!start)return;
    const list=[start];let last=start;
    while(list.length<d.chains[l]){let best=null,bd=1.7;
      for(const e of G.enemies){if(!e.alive||list.indexOf(e)>=0)continue;const d2=Math.hypot(pointAt(e.pos).x-pointAt(last.pos).x,pointAt(e.pos).y-pointAt(last.pos).y);if(d2<bd){bd=d2;best=e;}}
      if(!best)break;list.push(best);last=best;}
    G.bolts.push({pts:[tp].concat(list.map(fpx)),t0:now(),dur:tw(260)});SFX.zap();
    for(const f of list){const fp=fpx(f);sparkAt(fp.x,fp.y,'255,210,62');hit(f,d.dmg[l],'#ffe066');await sleep(75);if(G.runId!==G._run)throw ABORT;}
  }
}

async function moveFoe(e){
  const spd=Math.max(0,e.spd-(e.chillT>0?1:0));
  if(spd===0){e.frozenNow=true;const p=fpx(e);snowBurst(p.x,p.y);await sleep(150);}
  else{e.anim={from:e.pos,to:e.pos+spd,t0:now(),dur:tw(165*spd)};e.pos=e.anim.to;if(e.chillT>0)e.chillT--;await sleep(spd===2?345:175);}
  e.frozenNow=false;
}

async function endTurn(){
  if(G.state!=='build')return;
  G._run=++G.runId;G.state='resolving';hidePanel();updateHUD();
  try{
    const rate=G.wave<4?2:G.wave<7?3:4,n=Math.min(rate,G.queue.length);
    if(n>0){for(let i=0;i<n;i++){const e=makeFoe(G.queue.shift());e.pos=-.75-i*.65;e.drawPos=e.pos;G.enemies.push(e);}
      ringFx(ox+cell/2,oy+1.5*cell,cell*.9,'176,108,255',4,tw(340));SFX.spawn();await sleep(230);}
    if(G.runId!==G._run)throw ABORT;
    showBanner('TOWER PHASE','',true);
    for(const t of G.towers){if(G.runId!==G._run)throw ABORT;await fireTower(t);}
    showBanner('ENEMY PHASE','',true);
    const march=[...G.enemies].sort((a,b)=>b.pos-a.pos);
    for(const e of march){if(G.runId!==G._run)throw ABORT;if(!e.alive)continue;await moveFoe(e);
      if(G.state==='gameover')throw ABORT;
      if(e.alive&&e.pos>=LEAK){leakFoe(e);if(G.state==='gameover')throw ABORT;}}
    G.gold+=STIPEND;G.earned+=STIPEND;goldPop();
    if(G.queue.length===0&&G.enemies.length===0)waveCleared();
    if(G.runId!==G._run||G.state==='gameover'||G.state==='victory')throw ABORT;
    G.turn++;G.state='build';updateHUD();
  }catch(err){if(!err||!err.aborted)console.error(err);}
}

function leakFoe(e){e.alive=false;const i=G.enemies.indexOf(e);if(i>=0)G.enemies.splice(i,1);
  G.hp=Math.max(0,G.hp-FOES[e.type].leak);G.keepFlash=1;shake(7,.35);SFX.hurt();
  const gx=ox+12.5*cell,gy=oy+2.5*cell;boomFx(gx,gy,cell*.8);
  addText(gx,gy-cell,'-'+FOES[e.type].leak+' KEEP','#ff6a55');updateHUD();
  if(G.hp<=0){G.state='gameover';showBanner('THE KEEP FALLS','',true);SFX.lose();
    G.best.wave=Math.max(G.best.wave,G.wave);saveBest();
    setTimeout(()=>showEnd(false),900);}}

function waveCleared(){
  const bonus=20+10*G.wave;G.gold+=bonus;G.earned+=bonus;goldPop();SFX.wave();
  if(G.wave>=TOTAL){victory();return;}
  G.hp=Math.min(MAX_HP,G.hp+1);G.wave++;G.queue=WAVES[G.wave-1].slice();
  showBanner('WAVE '+G.wave,WAVENAMES[G.wave-1]+'   ·   +'+bonus+' GOLD · KEEP +1 HP',false);updateHUD();}

function victory(){G.state='victory';SFX.win();showBanner('EMBERHOLD STANDS','',true);
  G.best.wave=Math.max(G.best.wave,TOTAL);G.best.wins++;saveBest();
  for(let i=0;i<130;i++)part({type:'conf',x:rand(ox,ox+COLS*cell),y:rand(-40,-5),vx:rand(-14,14),vy:rand(40,95),life:rand(2,3.4),max:3.4,size:rand(3,6),color:['#e8c25a','#ff7a3d','#3ec1b0','#7fd6ff','#ff4d4d'][i%5],rot:rand(0,TAU),vr:rand(-6,6)});
  setTimeout(()=>showEnd(true),950);}

function update(dt){
  T+=dt;
  if(G.shakeT>0)G.shakeT=Math.max(0,G.shakeT-dt);
  G.keepFlash=Math.max(0,G.keepFlash-dt*1.6);
  for(const t of G.towers){t.recoil*=Math.exp(-dt*9);t.flashT=Math.max(0,t.flashT-dt*5);
    if(t.aim!=null){const d=normAng(t.aim-t.angle);t.angle+=d*Math.min(1,dt*12);if(Math.abs(d)<.02)t.aim=null;}
    if(t.type==='tesla'&&Math.random()<dt*.9){const p=towerPx(t);sparkAt(p.x+rand(-4,4),p.y-cell*.3,'255,220,90');}}
  for(const e of G.enemies){e.flash=Math.max(0,e.flash-dt*4);
    if(e.anim){const p=(now()-e.anim.t0)/e.anim.dur;
      if(p>=1){e.drawPos=e.pos;e.anim=null;}else e.drawPos=lerp(e.anim.from,e.anim.to,1-Math.pow(1-p,2));}
    else e.drawPos=e.pos;}
  for(const d of G.deco){d.pos+=dt*1.05;if(d.pos>L)d.pos=-4-Math.random()*3;d.drawPos=d.pos;}
  G.emberT-=dt;
  if(G.emberT<=0){G.emberT=rand(.12,.35);const gx=ox+12.5*cell,gy=oy+1.9*cell;
    part({type:'ember',x:gx+rand(-6,6),y:gy,vx:rand(-6,6),vy:rand(-38,-16),life:rand(.7,1.4),max:1.4,size:rand(1.2,2.4),color:'255,'+(140+((Math.random()*80)|0))+',60',ph:rand(0,9)});}
  if(Math.random()<dt*7){const x=ox+cell/2,y=oy+1.5*cell,a=rand(0,TAU),rr2=cell*rand(.35,.6);
    part({type:'spark',x:x+Math.cos(a)*rr2,y:y+Math.sin(a)*rr2,vx:-Math.cos(a)*40,vy:-Math.sin(a)*40,life:.4,max:.4,size:1.6,color:'176,108,255'});}
  for(let i=G.parts.length-1;i>=0;i--){const p=G.parts[i];p.life-=dt;
    if(p.life<=0){G.parts.splice(i,1);continue;}
    p.x+=p.vx*dt;p.y+=p.vy*dt;if(p.g)p.vy+=p.g*dt;if(p.vr)p.rot+=p.vr*dt;
    if(p.type==='snow')p.x+=Math.sin(T*3+p.ph||0)*dt*24;}
  for(let i=G.texts.length-1;i>=0;i--){G.texts[i].t+=dt;if(G.texts[i].t>G.texts[i].dur)G.texts.splice(i,1);}
  G.proj=G.proj.filter(p=>now()-p.t0<p.dur+60);
  G.bolts=G.bolts.filter(b=>now()-b.t0<b.dur+40);
  G.ringsFx=G.ringsFx.filter(r=>now()-r.t0<r.dur+40);
  for(const f of G.fog){f.x+=f.vx*dt;f.y+=f.vy*dt;
    if(f.x<-f.r)f.x=W+f.r;if(f.x>W+f.r)f.x=-f.r;if(f.y<-f.r)f.y=H+f.r;if(f.y>H+f.r)f.y=-f.r;}
}
