/* ================= projection ================= */
const cP=Math.cos(CAM.pitch),sP=Math.sin(CAM.pitch);
function proj(x,y,z){
  const dx=x-CAM.x,dy=y-CAM.y,dz=z-CAM.z;
  const yy=dy*cP+dz*sP,zz=dz*cP-dy*sP;
  if(zz<0.35)return null;
  return {x:cx+dx*f/zz,y:cy-yy*f/zz,s:f/zz,dz:zz};
}
function path4(a,b,c,d){if(!a||!b||!c||!d)return false;ctx.beginPath();ctx.moveTo(a.x,a.y);ctx.lineTo(b.x,b.y);ctx.lineTo(c.x,c.y);ctx.lineTo(d.x,d.y);ctx.closePath();return true;}
const fog=dz=>clamp(1.25-dz/95,.05,1);

function reflectX(x){const L=FLD.hw-R,per=4*L;let m=((x+L)%per+per)%per;return m<=2*L?m-L:3*L-m;}
function aiMove(p,dt,skill){
  let des=0,threat=false;
  if(state==='rally'&&ball.vz!==0){
    const t=(p.z-ball.z)/ball.vz;
    if(t>0.15&&t<4){des=reflectX(ball.x+ball.vx*t)+p.err;threat=true;}
  }
  if(!threat)des=Math.sin(time*.9+p.z)*1.3;
  des=clamp(des,-(FLD.hw-.3),FLD.hw-.3);
  const maxV=6.5+skill*4.5,d=des-p.x;
  p.x+=clamp(d,-maxV*dt,maxV*dt);
}

  ball.speed=Math.min(MAX_SPD,ball.speed*1.05+.35);maxSpd=Math.max(maxSpd,ball.speed);

function stepBall(dt){
  const prevZ=ball.z;
  ball.vy-=G*dt;ball.x+=ball.vx*dt;ball.y+=ball.vy*dt;ball.z+=ball.vz*dt;
  if(ball.y<R&&ball.vy<0){const imp=-ball.vy;ball.y=R;ball.vy=imp*.74;ball.vx*=.985;
    if(imp>2){sfx.floor(Math.min(1,imp/16));burst(ball.x,.15,ball.z,3,['rgba(200,220,255,.8)'],2.5,true);}}
  if(ball.x>FLD.hw-R){ball.x=FLD.hw-R;if(ball.vx>0){ball.vx*=-.92;sfx.wall();burst(ball.x,ball.y,ball.z,6,['#ffd166','#fff'],4,false);}}
  else if(ball.x<-(FLD.hw-R)){ball.x=-(FLD.hw-R);if(ball.vx<0){ball.vx*=-.92;sfx.wall();burst(ball.x,ball.y,ball.z,6,['#ffd166','#fff'],4,false);}}
  if(ball.vz<0&&prevZ>=pPaddle.z&&ball.z<=pPaddle.z&&Math.abs(ball.x-pPaddle.x)<PAD.hw+R*.75&&ball.y<PAD.y1+R)hitPlayer();
  else if(ball.vz>0&&prevZ<=aiPaddle.z&&ball.z>=aiPaddle.z&&Math.abs(ball.x-aiPaddle.x)<PAD.hw+R*.75&&ball.y<PAD.y1+R)hitAI();
  if(state==='rally'){if(ball.z<-2)pointScored('cpu');else if(ball.z>FLD.len+2)pointScored('you');}
}

/* ================= update ================= */
function update(dt){
  time+=dt;shake=Math.max(0,shake-dt*26);
  for(let i=parts.length-1;i>=0;i--){const p=parts[i];p.life-=dt;if(p.life<=0){parts.splice(i,1);continue;}
    p.vy-=16*dt;p.x+=p.vx*dt;p.y+=p.vy*dt;p.z+=p.vz*dt;}
  for(const d of dust){d.y+=dt*.12;d.x+=Math.sin(time*.4+d.ph)*dt*.15;if(d.y>5.6)d.y=.3;}

  if(state==='point'){stateT-=dt;
    if(stateT<=0){
      if(attractPoint){scores.you=0;scores.cpu=0;$('sYou').textContent='0';$('sCpu').textContent='0';}
      else if(scores.you>=7||scores.cpu>=7){endMatch();return;}
      server=other(server);startServe();
    }}
  else if(state==='serve'&&server==='cpu'){stateT-=dt;if(stateT<=0)launch(-1);}

  if(state==='menu'){aiMove(pPaddle,dt,.42);pPaddle.z=3;}
  else if(state!=='over'){
    const kx=(keys.a||keys.ArrowLeft?-1:0)+(keys.d||keys.ArrowRight?1:0);
    const kz=(keys.w||keys.ArrowUp?1:0)+(keys.s||keys.ArrowDown?-1:0);
    if(kx)pTarget.x=clamp(pTarget.x+kx*15*dt,-(FLD.hw-.5),FLD.hw-.5);
    if(kz)pTarget.z=clamp(pTarget.z+kz*10*dt,PAD.zmin,PAD.zmax);
    const ox=pPaddle.x;
    pPaddle.x+=(pTarget.x-pPaddle.x)*Math.min(1,dt*16);
    pPaddle.z+=(pTarget.z-pPaddle.z)*Math.min(1,dt*14);
    pSpin+=(clamp((pPaddle.x-ox)/dt,-30,30)-pSpin)*Math.min(1,dt*8);
  }
  if(state!=='over')aiMove(aiPaddle,dt,state==='menu'?.55:aiSkill);

  if(state==='serve'){
    if(server==='you'){ball.x=pPaddle.x;ball.z=pPaddle.z;}else{ball.x=aiPaddle.x;ball.z=aiPaddle.z-.4;}
    ball.y=(server==='you'?2.5:3.1)+Math.sin(time*3)*.22;
  }
  if(state==='rally')stepBall(dt);
}

    shadowAt(pPaddle.x,pPaddle.z,PAD.hw,.3);shadowAt(aiPaddle.x,aiPaddle.z,PAD.hw,.25);}

  /* paddles + ball (depth order near camera) */
  drawPaddle(aiPaddle,0,'#4a5878','#242e49','#9fd8ff');
  if(state==='rally'||state==='serve'){
    const tilt=clamp(pSpin*.03,-.4,.4);
    if(ball.z<pPaddle.z){drawBall();drawPaddle(pPaddle,tilt,'#ff6b7a','#b81f3d','#ffd9a8');}
    else{drawPaddle(pPaddle,tilt,'#ff6b7a','#b81f3d','#ffd9a8');drawBall();}
  }

  /* particles */
  for(const p of parts){const pr=proj(p.x,p.y,p.z);if(!pr)continue;
    ctx.globalAlpha=p.life/p.max;ctx.fillStyle=p.col;
    const r=Math.max(1,p.sz*pr.s);ctx.beginPath();ctx.arc(pr.x,pr.y,r,0,7);ctx.fill();}
  ctx.globalAlpha=1;
  ctx.restore();

  /* speed meter */
  const pct=state==='rally'?clamp((ball.speed-BASE_SPD)/(MAX_SPD-BASE_SPD),0,1):0;
  $('speedFill').style.width=(pct*100).toFixed(1)+'%';
  $('speedFill').style.background='hsl('+(48-pct*52)+' 96% '+(60-pct*8)+'%)';
}
