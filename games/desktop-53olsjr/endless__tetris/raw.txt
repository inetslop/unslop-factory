function update(dt){
 for(const a of amb){a.y+=a.vy*dt/1000*(1+level*.03);a.rot+=a.vr*dt/1000;
  if(a.y>H+a.s*4){a.y=-a.s*4;a.x=Math.random()*W}}
 if(mode==='playing'){
  const dir=heldDirs.length?heldDirs[heldDirs.length-1]:0;
  if(dir){dasT+=dt;if(dasT>=DAS){arrT+=dt;while(arrT>=ARR){arrT-=ARR;tryMove(dir)}}}
  else{dasT=0;arrT=0}
  if(clearing){clearT+=dt;if(clearT>=CLEAR_MS)finishClear()}
  else if(piece){
   let iv=gravityMs(level);if(downHeld)iv=Math.min(iv,35);
   gravAcc+=dt;
   while(gravAcc>=iv){gravAcc-=iv;
    if(!collidesAt(piece.t,piece.rot,piece.x,piece.y+1)){piece.y++;if(downHeld)score+=1;lastAction='gravity'}
    else break}
   if(collidesAt(piece.t,piece.rot,piece.x,piece.y+1)){lockTimer+=dt;if(lockTimer>=LOCK_MS)lockPiece()}
   else lockTimer=0}
  displayScore+=(score-displayScore)*Math.min(1,dt*.02);
  if(Math.abs(score-displayScore)<1)displayScore=score;
 }
 if(mode==='playing'||mode==='over'){
  for(let i=particles.length-1;i>=0;i--){const p=particles[i];p.life-=dt;
   if(p.life<=0){particles.splice(i,1);continue}
   p.vy+=p.g*dt/1000;p.x+=p.vx*dt/1000;p.y+=p.vy*dt/1000}
  if(particles.length>420)particles.splice(0,particles.length-420);
  for(let i=popups.length-1;i>=0;i--){popups[i].life-=dt;if(popups[i].life<=0)popups.splice(i,1)}
  shake.t=Math.max(0,shake.t-dt);flashT=Math.max(0,flashT-dt);scorePulse=Math.max(0,scorePulse-dt/280);
 }
}
