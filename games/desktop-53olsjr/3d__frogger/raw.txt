/* ================= FROG LOGIC ================= */
function standZ(row,x){
  const l=LANES[row];
  if(l.t==='log'||l.t==='turtle'){const p=findPlatform(row,x);return p?p.top:-1;}
  return 0;
}
function findPlatform(row,x){
  const l=LANES[row];
  if(l.t==='log'){const arr=row===4?logsR[0]:logsR[1];
    for(const o of arr)if(Math.abs(o.x-x)<1.92)return{vx:o.dir*l.sp*spdMul,top:0.72};}
  else if(l.t==='turtle'){const arr=row===5?turtR[0]:turtR[1];
    for(const o of arr)if(Math.abs(o.x-x)<0.64)return{vx:o.dir*l.sp*spdMul,top:0.30};}
  return null;
}
function attemptHop(d){
  if(frog.dead||frog.hidden)return;
  if(state!=='play'&&state!=='start')return;
  if(frog.hop){frog.pending=d;return;}
  let row=frog.row,x=Math.round(frog.x);
  if(d==='up')row=Math.min(8,row+1);else if(d==='down')row=Math.max(0,row-1);
  else if(d==='left')x=Math.max(-5,x-1);else x=Math.min(5,x+1);
  frog.hop={x0:frog.x,y0:frog.y,x1:x,y1:ROWS[row],z0:frog.z,z1:standZ(row,x),row,t:0};
  frog.row=row;frog.facing=FACING[d];sfx.hop();hopGrace=0.07;
}
function updateFrog(dt){
  hopGrace-=dt;
  if(frog.dead){
    frog.deadT+=dt;
    const lim=state==='play'?1.25:0.9;
    if(frog.deadT>=lim){
      if(state==='play'){if(lives<=0)gameOver();else respawn();}
      else respawn();
    }
    return;
  }
  if(frog.hop){
    const h=frog.hop;h.t+=dt/HOPTIME;
    if(h.t>=1){frog.x=h.x1;frog.y=ROWS[h.row];frog.z=h.z1<0?0:h.z1;frog.hop=null;onLand();}
    else{const zb=h.z1<0?0:h.z1;
      frog.x=h.x0+(h.x1-h.x0)*h.t;frog.y=h.y0+(h.y1-h.y0)*h.t;
      frog.z=(h.z0+(zb-h.z0)*h.t)+Math.sin(Math.PI*h.t)*0.5;}
  }else{
    const lane=LANES[frog.row];
    if(lane.t==='log'||lane.t==='turtle'){
      const p=findPlatform(frog.row,frog.x);
      if(p){frog.x+=p.vx*dt;frog.z+=(p.top-frog.z)*Math.min(1,dt*14);
        if(Math.abs(frog.x)>7.3)die('drown');}
      else if(hopGrace<=0)die('drown');
    }else frog.z+=(0-frog.z)*Math.min(1,dt*14);
  }
  for(const c of cars){
    if(Math.abs(c.y-frog.y)<0.42&&Math.abs(c.x-frog.x)<c.hx+0.27){die('squash');break;}
  }
}
function botBrain(dt){
  if(frog.dead||frog.hidden)return;
  botT-=dt;if(botT>0)return;
  botT=0.3+Math.random()*0.45;
  const r=Math.random();
  attemptHop(r<0.62?'up':r<0.82?'left':'right');
}

/* ================= WORLD UPDATE ================= */
function updateWorld(dt){
  for(const c of cars){c.x+=c.dir*c.sp*spdMul*dt;
    if(c.dir>0&&c.x>9.5)c.x-=c.span;else if(c.dir<0&&c.x<-9.5)c.x+=c.span;}
  const wrap=(a,bound,span)=>{for(const o of a){o.x+=o.dir*LANES[o.y===ROWS[4]||o.y===ROWS[6]?(o.dir>0?6:4):(o.dir>0?5:7)].sp*spdMul*dt;
      if(o.x>bound)o.x-=span;else if(o.x<-bound)o.x+=span;}};
  wrap(logsR[0],10,22);wrap(logsR[1],10,22);wrap(turtR[0],10,26.4);wrap(turtR[1],10,26.4);
  for(const s of streaks){s.x-=0.7*dt;if(s.x<-8.5)s.x+=17;}
  for(const d of drops){d.vz-=5.5*dt;d.x+=d.vx*dt;d.y+=d.vy*dt;d.z+=d.vz*dt;d.life-=dt;if(d.z<0)d.life=0;}
  drops=drops.filter(d=>d.life>0);
  for(const r of ripples)r.t+=dt;
  ripples=ripples.filter(r=>r.t<0.9);
}

const FBOX=[{n:[0,0,1],v:[4,5,6,7]},{n:[0,0,-1],v:[0,1,5,4]},

function drawBox(cx,cy,cz,hx,hy,hz,col){
  const C=[];for(let i=0;i<8;i++)C.push([cx+((i&1)?hx:-hx),cy+((i&2)?hy:-hy),cz+((i&4)?hz:-hz)]);
  for(const f of FBOX){
    const fc=[cx+f.n[0]*hx,cy+f.n[1]*hy,cz+f.n[2]*hz];
    const dx=fc[0]-cam.x,dy=fc[1]-cam.y,dz=fc[2]-cam.z;
    if(f.n[0]*dx+f.n[1]*dy+f.n[2]*dz<=0.02)continue;
    const ndl=Math.max(0,f.n[0]*LX+f.n[1]*LY+f.n[2]*LZ),b=0.52+0.5*ndl;
    const P=[];let ok=true;
    for(const vi of f.v){const p=project(C[vi][0],C[vi][1],C[vi][2]);if(!p){ok=false;break;}P.push(p);}
    if(!ok)continue;
    ctx.beginPath();ctx.moveTo(P[0][0],P[0][1]);
    for(let i=1;i<4;i++)ctx.lineTo(P[i][0],P[i][1]);

function drawCyl(x,y,z,hx,ry,rz,col){
  const S=10,P=[],Q=[],A=[];
  for(let i=0;i<S;i++){const a=i/S*Math.PI*2;A[i]=a;
    P.push([x-hx,y+Math.cos(a)*ry,z+Math.sin(a)*rz]);
    Q.push([x+hx,y+Math.cos(a)*ry,z+Math.sin(a)*rz]);}
  for(let i=0;i<S;i++){
    const j=(i+1)%S,am=(A[i]+A[j])/2;
    let nx=Math.cos(am)/ry,nz=Math.sin(am)/rz;const nl=Math.hypot(nx,nz);nx/=nl;nz/=nl;
    const cxp=x,cyp=y+Math.cos(am)*ry,czp=z+Math.sin(am)*rz;
    if(nx*(cxp-cam.x)+nz*(czp-cam.z)<=0.03)continue;
    const ndl=Math.max(0,nx*LX+nz*LZ),b=0.52+0.5*ndl;
    const pts=[P[i],Q[i],Q[j],P[j]].map(v=>project(v[0],v[1],v[2]));
    if(pts.some(!q=>!q))continue;
    ctx.beginPath();ctx.moveTo(pts[0][0],pts[0][1]);
    for(let k=1;k<4;k++)ctx.lineTo(pts[k][0],pts[k][1]);
    ctx.closePath();
    const s=shade(col,b,ndl);ctx.fillStyle=s;ctx.fill();ctx.strokeStyle=s;ctx.lineWidth=1;ctx.stroke();
  }
  // end caps
  [[-1,P],[1,Q]].forEach(([s,ring])=>{
    const cxp=x+s*hx;
    if(s*(cxp-cam.x)<=0.03)return;
    const pts=ring.map(v=>project(v[0],v[1],v[2]));if(pts.some(!q=>!q))return;
    ctx.beginPath();ctx.moveTo(pts[0][0],pts[0][1]);
    for(let k=1;k<S;k++)ctx.lineTo(pts[k][0],pts[k][1]);
    ctx.closePath();
    const b=0.52+0.5*Math.max(0,s*LX);
    ctx.fillStyle=shade(col,b,0);ctx.fill();
  });
}

}
function drawFireflies(){
  ctx.save();ctx.globalCompositeOperation='lighter';
  for(const f of flies){
    const x=f.bx+Math.sin(gt*0.5+f.ph)*0.4,y=f.by+Math.cos(gt*0.4+f.ph*2)*0.3,z=f.bz+Math.sin(gt*f.sp+f.ph)*0.15;
    const a=0.15+0.85*Math.pow(0.5+0.5*Math.sin(gt*f.sp*1.7+f.ph),2);
    glowDot(x,y,z,0.16,[255,226,138],a*0.5);
    worldDot(x,y,z,0.035,[255,244,190],a);
  }
  ctx.restore();
}

      if(frog.dead){
        if(frog.deadType==='squash'){
          const q=Math.min(1,frog.deadT/0.4);
          drawFrog(frog.x,frog.y,0,frog.facing,1,true,1-0.82*q);
          strokeCircleW(frog.x,frog.y,0.03,0.25+q*0.9,'rgba(40,70,40,'+Math.max(0,0.6-frog.deadT*0.5)+')',3);
        }
        return;
      }
      const bob=(!frog.hop)?Math.sin(gt*4)*0.02:0;
      drawFrog(frog.x,frog.y,frog.z+bob,frog.facing,1,false,1);
    }});
  }
  items.sort((a,b)=>b.y-a.y);

  ctx.fillStyle=vg;ctx.fillRect(0,0,W,H);
}
