function fire(e){

  const a=Math.atan2((H-64)-e.y,ship.x-e.x),d=e.type;
  if(d==='drone')addB(e.x,e.y,Math.cos(a)*bs,Math.sin(a)*bs,C_MAG);
  else if(d==='weaver'){for(let i=-1;i<=1;i++)addB(e.x,e.y,Math.cos(a+i*.3)*bs*.9,Math.sin(a+i*.3)*bs*.9,C_ORG);}
  else{e.sa+=.6;const s=bs*.55;addB(e.x,e.y,Math.cos(e.sa)*s,Math.sin(e.sa)*s,C_VIO);addB(e.x,e.y,-Math.cos(e.sa)*s,-Math.sin(e.sa)*s,C_VIO);}
  if(Math.random()<.5)SFX.fire();

function completeWord(e){
  G.combo++;G.bestCombo=Math.max(G.bestCombo,G.combo);
  G.mult=Math.min(5,1+Math.floor(G.combo/4));
  const pts=e.word.length*10*G.mult+(e.power?30*G.mult:0);
  G.score+=pts;G.words++;
  pops.push({x:e.x,y:e.y-24,txt:'+'+pts,c:e.power?POWERS[e.power].c:[255,224,138],l:1});
  const c=e.power?POWERS[e.power].c:C_MAG;
  explode(e.x,e.y,c,e.power?42:26);SFX.zap();SFX.boom(false);
  strikes.push({x:e.x,y:e.y,mx:ship.x,my:H-80,t:.13});
  for(let i=bullets.length-1;i>=0;i--){const b=bullets[i],dx=b.x-e.x,dy=b.y-e.y;
    if(dx*dx+dy*dy<105*105){spark(b.x,b.y,c,1);bullets.splice(i,1);}}
  if(e.power)applyPower(e.power);
  enemies.splice(enemies.indexOf(e),1);if(activeTarget===e)activeTarget=null;
}
function applyPower(p){
  const P=POWERS[p];G.flashC=P.c;G.flashA=.26;
  pops.push({x:ship.x,y:H-110,txt:p.toUpperCase(),c:P.c,l:1.4,big:true});
  if(p==='shield')G.shieldT=5;
  else if(p==='heal'){G.hp=Math.min(6,G.hp+2);setPips();}
  else if(p==='slow')G.slowT=3.2;
  else{for(const b of bullets)spark(b.x,b.y,P.c,1);bullets.length=0;}
  SFX.power(p);
}

  if(G.hp<=0){G.hp=0;die(px||ship.x,py||(H-64));}

function typeLetter(ch){

  for(const e of enemies)if(e.word[e.idx]===ch&&(bestE===null||e.y>bestE.y))bestE=e;

function worldUpdate(dt,live){
  const ts=G.slowT>0?.45:1,t=live?G.time:T;
  if(live){
    G.time+=dt;G.goT-=dt;
    if(!G.goFired&&G.goT<=.35){G.goFired=true;for(let i=0;i<3;i++)spawnEnemy(i*85);}
    if(G.goT<.4&&!G.dead){G.spawnT-=dt;
      if(G.spawnT<=0){if(enemies.length<cap())spawnEnemy(0);
        G.spawnT=lerp(2.3,.8,clamp(G.time/95,0,1))*rnd(.75,1.25);}}
    G.shieldT=Math.max(0,G.shieldT-dt);G.slowT=Math.max(0,G.slowT-dt);
    G.invT=Math.max(0,G.invT-dt);G.hitFlash=Math.max(0,G.hitFlash-dt*1.6);
    G.flashA=Math.max(0,G.flashA-dt*1.4);
    if(!G.dead){const tv=(mvR?1:0)-(mvL?1:0);
      ship.vx=lerp(ship.vx,tv*430,clamp(dt*10,0,1));ship.x=clamp(ship.x+ship.vx*dt,26,W-26);}
  }
  for(let i=enemies.length-1;i>=0;i--){const e=enemies[i];
    e.y+=e.vy*dt*ts;
    if(e.type==='weaver')e.x=clamp(e.bx+Math.sin(t*e.wf+e.phase)*e.amp,30,W-30);
    if(e.y>H+80){enemies.splice(i,1);continue;}
    if(live&&!G.dead&&G.goT<.8&&e.y>10){e.ft-=dt*ts;if(e.ft<=0){fire(e);e.ft=e.fi*rnd(.8,1.25);}}
    if(live&&!G.dead&&e.y>H-150){enemies.splice(i,1);
      shocks.push({x:e.x,y:H-150,r:6,vr:340,l:.5,c:C_MAG});shake=Math.min(16,shake+5);SFX.boom(false);
      damage(2,e.x,H-150);if(activeTarget===e)activeTarget=null;continue;}
  }
  for(let i=bullets.length-1;i>=0;i--){const b=bullets[i];
    b.l+=dt;b.x+=b.vx*dt*ts;b.y+=b.vy*dt*ts;
    if(b.x<-40||b.x>W+40||b.y<-60||b.y>H+40||b.l>9){bullets.splice(i,1);continue;}
    if(live&&!G.dead){const dx=b.x-ship.x,dy=b.y-(H-64),sr=G.shieldT>0?30:13;
      if(dx*dx+dy*dy<sr*sr){
        if(G.shieldT>0)spark(b.x,b.y,[88,255,208],3);else{damage(1,b.x,b.y);spark(b.x,b.y,C_MAG,6);}
        bullets.splice(i,1);}}
  }
  if(activeTarget&&enemies.indexOf(activeTarget)<0)activeTarget=null;
}
function updateFx(dt){
  for(let i=parts.length-1;i>=0;i--){const p=parts[i];p.l-=dt;if(p.l<=0){parts.splice(i,1);continue;}
    p.x+=p.vx*dt;p.y+=p.vy*dt;p.vx*=(1-2.5*dt);p.vy*=(1-2.5*dt);}
  for(let i=pops.length-1;i>=0;i--){const p=pops[i];p.l-=dt;p.y-=32*dt;if(p.l<=0)pops.splice(i,1);}
  for(let i=shocks.length-1;i>=0;i--){const s=shocks[i];s.l-=dt;s.r+=s.vr*dt;if(s.l<=0)shocks.splice(i,1);}
  for(let i=strikes.length-1;i>=0;i--){strikes[i].t-=dt;if(strikes[i].t<=0)strikes.splice(i,1);}
  shake*=Math.exp(-6*dt);if(shake<.1)shake=0;
}
function ambient(dt){
  for(const s of stars){s.y+=(s.z?14:6)*dt;if(s.y>H+2){s.y=-2;s.x=Math.random()*W;}}
  if(G.state==='menu'){G.mbT=(G.mbT||0)-dt;
    if(G.mbT<=0){G.mbT=.45;mB.push({x:rnd(0,W),y:-10,vy:rnd(50,130),c:pick([C_MAG,C_VIO,C_CYA]),r:rnd(2.2,4)});}
    for(let i=mB.length-1;i>=0;i--){const b=mB[i];b.y+=b.vy*dt;if(b.y>H+20)mB.splice(i,1);}}
  updateFx(dt);
}
