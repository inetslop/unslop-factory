function mix(a,b,t){ return [ a[0]+(b[0]-a[0])*t|0, a[1]+(b[1]-a[1])*t|0, a[2]+(b[2]-a[2])*t|0 ]; }
const rgb=(c,a=1)=>`rgba(${c[0]},${c[1]},${c[2]},${a})`;

function genTerrain(){
  terrain=document.createElement('canvas'); terrain.width=W*dpr; terrain.height=H*dpr;
  const c=terrain.getContext('2d'); c.scale(dpr,dpr);
  // far hills
  c.beginPath(); c.moveTo(0,H); let y=H*.62; c.lineTo(0,y);
  for(let x=0;x<=W+30;x+=30){ y=H*.62 - Math.sin(x*.011)*10 - Math.sin(x*.004+2)*16 - (Math.random()*8); c.lineTo(x,y); }
  c.lineTo(W,H); c.closePath(); c.fillStyle='#0d150e'; c.fill();
  // dead tree, left
  c.strokeStyle='#0a0f0a'; c.lineWidth=7; c.lineCap='round';
  const tx=W*.12, ty=H*.66;
  c.beginPath(); c.moveTo(tx,ty); c.lineTo(tx-4,ty-58); c.stroke();
  c.lineWidth=3.5;
  c.beginPath(); c.moveTo(tx-2,ty-30); c.lineTo(tx-26,ty-52); c.moveTo(tx-3,ty-44); c.lineTo(tx+18,ty-70); c.moveTo(tx-1,ty-22); c.lineTo(tx+14,ty-34); c.stroke();
  // near ground + stones
  c.beginPath(); c.moveTo(0,H); y=H*.75; c.lineTo(0,y);
  for(let x=0;x<=W+30;x+=30){ y=H*.75 - Math.sin(x*.016+4)*7 - (Math.random()*5); c.lineTo(x,y); }
  c.lineTo(W,H); c.closePath(); c.fillStyle='#060a06'; c.fill();
  for(let k=0;k<5;k++){
    const sx=W*(.2+k*.16)+rnd(-30,30), sw=rnd(14,30), sh=rnd(18,34), sy=H*.75-rnd(0,8);
    c.fillStyle='#0b110c'; c.beginPath();
    c.moveTo(sx,sy+sh); c.lineTo(sx,sy+sw/2); c.arc(sx+sw/2,sy+sw/2,sw/2,Math.PI,0); c.lineTo(sx+sw,sy+sh); c.closePath(); c.fill();
    if(k===1){ c.strokeStyle='#0b110c'; c.lineWidth=5; c.beginPath(); c.moveTo(sx+sw/2,sy-14); c.lineTo(sx+sw/2,sy+sh); c.moveTo(sx+sw/2-7,sy-6); c.lineTo(sx+sw/2+7,sy-6); c.stroke(); }
  }
}
function genVignette(){

  const c=vign.getContext('2d'); c.scale(dpr,dpr);

}

let dawnT=0, redFlash=0, last=performance.now();
function drawBG(dt,t){
  dawnT = state.dawn ? Math.min(1,dawnT+dt/2.5) : Math.max(0,dawnT-dt/.6);
  const top=mix([7,10,15],[64,38,52],dawnT), bot=mix([15,23,16],[216,134,72],dawnT);
  const g=ctx.createLinearGradient(0,0,0,H); g.addColorStop(0,rgb(top)); g.addColorStop(1,rgb(bot));
  ctx.fillStyle=g; ctx.fillRect(0,0,W,H);
  // moon
  if(dawnT<.98){
    const mx=W*.82,my=H*.16,mr=Math.min(W,H)*.05;
    let mg=ctx.createRadialGradient(mx,my,mr*.3,mx,my,mr*4);
    mg.addColorStop(0,`rgba(220,230,200,${.28*(1-dawnT)})`); mg.addColorStop(1,'rgba(220,230,200,0)');
    ctx.fillStyle=mg; ctx.fillRect(mx-mr*4,my-mr*4,mr*8,mr*8);
    ctx.beginPath(); ctx.arc(mx,my,mr,0,7); ctx.fillStyle=`rgba(226,232,210,${.9*(1-dawnT)})`; ctx.fill();
    ctx.beginPath(); ctx.arc(mx-mr*.3,my-mr*.2,mr*.85,0,7); ctx.fillStyle=rgb(top,.55*(1-dawnT)); ctx.fill();
  }
  // dawn sun
  if(dawnT>0){
    const sx=W*.6, sy=H*.72;
    let sg=ctx.createRadialGradient(sx,sy,4,sx,sy,H*.5);
    sg.addColorStop(0,`rgba(255,196,120,${.8*dawnT})`); sg.addColorStop(1,'rgba(255,196,120,0)');
    ctx.fillStyle=sg; ctx.fillRect(0,0,W,H);
  }
  ctx.drawImage(terrain,0,0,W,H);
  // fog
  const fc=mix([150,180,130],[226,176,142],dawnT);
  for(const f of fog){
    f.x+=f.v*dt; if(f.x<-f.r) f.x=W+f.r; if(f.x>W+f.r) f.x=-f.r;
    let fg=ctx.createRadialGradient(f.x,f.y,0,f.x,f.y,f.r);
    fg.addColorStop(0,rgb(fc,f.a)); fg.addColorStop(1,rgb(fc,0));
    ctx.fillStyle=fg; ctx.beginPath(); ctx.arc(f.x,f.y,f.r,0,7); ctx.fill();
  }
  // spores
  const sc=mix([170,215,130],[255,214,150],dawnT);
  for(const s of spores){
    s.y+=s.vy*dt; s.x+=Math.sin(t*s.sp+s.ph)*9*dt;
    if(s.y<-4){ s.y=H+4; s.x=Math.random()*W; }
    ctx.globalAlpha=.22+.2*Math.sin(t*1.4+s.ph);
    ctx.beginPath(); ctx.arc(s.x,s.y,s.r,0,7); ctx.fillStyle=rgb(sc); ctx.fill();
  }
  ctx.globalAlpha=1;
  ctx.drawImage(vign,0,0,W,H);
  if(redFlash>0){ redFlash=Math.max(0,redFlash-dt*2); ctx.fillStyle=`rgba(150,25,20,${redFlash*.28})`; ctx.fillRect(0,0,W,H); }
}
(function loop(now){
  const dt=Math.min(.05,(now-last)/1000); last=now; drawBG(dt,now/1000); requestAnimationFrame(loop);
})(performance.now());

/* ================= CARDS ================= */
function pickCard(){
  let total=0; const ws=CARDS.map(c=>{
    let w=c.w;
    if(c.id==='cure'&&state.infection>0) w=16;
    if(c.id==='medkit'&&state.hp>=state.maxHp-2) w=4;
    if(c.id==='barricade'&&state.barricade>=8) w=5;
    total+=w; return w;
  });
  let r=Math.random()*total;
  for(let i=0;i<CARDS.length;i++){ r-=ws[i]; if(r<=0) return CARDS[i]; }
  return CARDS[0];
}

function enterTargeting(i){
  state.tCard=i; state.tIdx=0;

  state.zombies.forEach(z=>z.el.classList.add('targetable'));

function moveT(d){
  if(state.tCard<0||!state.zombies.length) return;
  state.tIdx=(state.tIdx+d+state.zombies.length)%state.zombies.length;
  applyCursor(); SFX.tick();

function confirmT(){
  if(state.tCard<0) return;
  const z=state.zombies[state.tIdx]; if(!z) return clearTargeting();
  const i=state.tCard; clearTargeting(); doPlay(i,z);
}
function clearTargeting(){
  state.tCard=-1;
  state.hand.forEach(c=>c.el&&c.el.classList.remove('sel'));
  state.zombies.forEach(z=>{ z.el.classList.remove('targetable','cursor'); });

}

/* ================= COMBAT / PLAY ================= */
function centerOf(el){ const r=el.getBoundingClientRect(); return [r.left+r.width/2, r.top+r.height*.35]; }

function tryPlay(i){
  if(state.mode!=='play'||state.phase!=='player') { setHint('Not your turn…',true); return; }
  const c=state.hand[i]; if(!c||c.spent) return;
  if(c.cost>state.energy){ SFX.deny(); setHint(`Need ${c.cost} brain${c.cost>1?'s':''}.`,true); return; }
  if(c.type==='target'&&state.zombies.length===0){ SFX.deny(); setHint('Nothing to hit.',true); return; }
  if(c.type==='all'&&state.zombies.length===0){ SFX.deny(); setHint('Nothing to hit.',true); return; }
  if(c.type==='front'&&state.zombies.length===0){ SFX.deny(); setHint('No front zombie to spike.',true); return; }
  if(c.id==='medkit'&&state.hp>=state.maxHp){ SFX.deny(); setHint("You're healthy as a horse.",true); return; }
  if(c.id==='barricade'&&state.barricade>=10){ SFX.deny(); setHint('Barricade is solid enough.',true); return; }
  if(c.id==='cure'&&state.infection===0&&state.hp>=state.maxHp){ SFX.deny(); setHint('Nothing to cure.',true); return; }
  if(c.type==='draw'&&state.hand.length>=6){ SFX.deny(); setHint('Hand full.',true); return; }
  if(c.type==='target'){ enterTargeting(i); return; }
  doPlay(i,null);
}

function doPlay(i,target){
  const c=state.hand[i]; if(!c) return;
  c.spent=true; state.energy-=c.cost; renderHUD(); SFX.card();
  // fly animation
  const [sx,sy]=centerOf(c.el.querySelector('.cinner'));
  let tx,ty;
  if(target){ [tx,ty]=centerOf(target.el); }
  else if(c.type==='all'){ const z=state.zombies[0]; z?([tx,ty]=centerOf(z.el)):([tx,ty]=centerOf(youPanel)); }
  else{ [tx,ty]=centerOf(youPanel); }
  c.el.querySelector('.cinner').animate(
    [{transform:'translate(0,0) scale(1)',opacity:1},
     {transform:`translate(${tx-sx}px,${ty-sy}px) scale(.35) rotate(10deg)`,opacity:0}],
    {duration:260,easing:'ease-in'});
  setTimeout(()=>{ c.el.remove(); layoutHand(); },270);
  state.hand.splice(i,1);
  setTimeout(()=>applyEffect(c,target),150);
}

function applyEffect(c,target){
  switch(c.type){
    case 'target': damageZombie(target,c.dmg); break;
    case 'all': state.zombies.forEach(z=>damageZombie(z,c.dmg)); break;
    case 'front':
      if(state.zombies.length) damageZombie(state.zombies[0],c.dmg);
      if(c.bd){ state.barricade=Math.min(10,state.barricade+c.bd); SFX.wood();
        const [x,y]=centerOf(youPanel); popup(x,y+40,'+'+c.bd+' BARR','woodp'); }
      break;
    case 'self':{
      const [x,y]=centerOf(youPanel);
      if(c.heal){ state.hp=Math.min(state.maxHp,state.hp+c.heal); SFX.heal(); popup(x,y,'+'+c.heal,'heal'); }
      if(c.bd){ state.barricade=Math.min(10,state.barricade+c.bd); SFX.wood(); popup(x,y+34,'+'+c.bd+' BARR','woodp'); }
      if(c.cure&&state.infection>0){ state.infection=0; SFX.heal(); popup(x,y-20,'PURGED','plaguep'); }
      break; }
    case 'draw':{
      const nc=pickCard(); state.hand.push(nc); addCardEl(nc,0); layoutHand(); SFX.draw();
      setHint('Drew '+nc.name+'.',true); break; }
  }
  renderHUD();
}

function damageZombie(z,dmg){
  if(!z.alive) return;
  z.hp-=dmg;
  const [x,y]=centerOf(z.el);
  popup(x,y,'-'+dmg,'dmg'); blood(x,y,7); SFX.hit();
  z.el.querySelector('.z-art').animate([{filter:'brightness(2.4)'},{filter:'brightness(1)'}],{duration:180});
  [...z.el.querySelectorAll('.pz')].forEach((p,i)=>p.classList.toggle('on',i<z.hp));
  if(z.hp<=0){
    z.alive=false; state.zombies.splice(state.zombies.indexOf(z),1);
    SFX.die(); blood(x,y,16);
    state.kills++; state.score+=15;
    if(z.type==='wailer'){ state.scream++; SFX.scream(); popup(x,y-26,'SCREAMS!','dim'); setHint("The Wailer's scream echoes… −1 brain next round.",true); }
    z.el.style.pointerEvents='none';
    z.el.animate([{transform:'rotate(0) translateY(0)',opacity:1},
      {transform:'rotate(70deg) translateY(34px)',opacity:0}],{duration:420,easing:'ease-in'}).onfinish=()=>z.el.remove();
    updateFront(); renderHUD();
  }
}

function hitPlayer(dmg,from){
  const abs=Math.min(state.barricade,dmg); state.barricade-=abs;
  const rest=dmg-abs; state.hp-=rest;
  const [x,y]=centerOf(youPanel);
  if(abs>0) popup(x,y+34,'-'+abs+' BARR','woodp');
  if(rest>0){ popup(x,y,'-'+rest,'hurt'); SFX.hurt(); redFlash=Math.min(1,.25+dmg/6); shake(4+dmg*2); }
  else{ SFX.wood(); shake(3); }
  youPanel.animate([{boxShadow:'0 0 0 rgba(201,58,46,0)'},{boxShadow:'0 0 26px rgba(201,58,46,.8)'},{boxShadow:'0 0 0 rgba(201,58,46,0)'}],{duration:450});
  renderHUD();
}

/* ================= TURN FLOW ================= */
async function endTurn(){
  if(state.mode!=='play'||state.phase!=='player') return;
  clearTargeting(); state.phase='enemy'; renderHUD(); SFX.whoosh();
  await sleep(420);
  const snap=[...state.zombies];
  for(let i=0;i<snap.length;i++){
    const z=snap[i]; if(!z.alive) continue;
    z.el.animate([{transform:'translateX(0)'},{transform:'translateX(-26px)',offset:.4},{transform:'translateX(0)'}],{duration:380});
    await sleep(360);
    if(z.atk>0){ hitPlayer(z.atk*(i===0?2:1),z); }
    else { const [x,y]=centerOf(z.el); popup(x,y,'screams…','dim'); SFX.groan(); }
    await sleep(260);
    if(state.hp<=0) break;
  }
  if(state.hp>0){
    const inf=state.zombies.find(z=>z.type==='infector'&&z.alive);
    if(inf){
      state.infection++; SFX.infect(); renderHUD();
      const [x,y]=centerOf(youPanel); popup(x,y-30,'INFECTION +1','plaguep');
      setHint("The Infector's spores take hold — Infection +1.",true);
      await sleep(650);
    }
  }
  if(state.hp<=0){ die(); return; }
  // next round / dawn
  state.round++;
  if(state.round>8){ winGame(); return; }
  state.score+=25;
  const base=3+(state.round>=4?1:0)+(state.round>=7?1:0);
  let e=Math.max(1,base-state.infection-state.scream);
  if(state.scream>0) setHint("Screaming horde: −1 brain.",true);
  state.scream=0; state.energy=e;
  banner('ROUND '+state.round, ROUND_LINES[state.round-1]);
  SFX.pop();
  spawnWave(state.round);
  drawHand();
  state.phase='player'; renderHUD();
}
