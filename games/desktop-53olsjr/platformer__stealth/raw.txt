function segRectT(x1,y1,x2,y2,r){
  const dx=x2-x1, dy=y2-y1;
  let t0=0.02, t1=0.98; // we only care about intersection before reaching target (t<~1)
  if(Math.abs(dx)<1e-9){ if(x1<r.x||x1>r.x+r.w) return 1e9; }
  else{
    let ta=(r.x-x1)/dx, tb=(r.x+r.w-x1)/dx; if(ta>tb){const q=ta;ta=tb;tb=q;}
    t0=Math.max(t0,ta); t1=Math.min(t1,tb); if(t0>t1) return 1e9;
  }
  if(Math.abs(dy)<1e-9){ if(y1<r.y||y1>r.y+r.h) return 1e9; }
  else{
    let ta=(r.y-y1)/dy, tb=(r.y+r.h-y1)/dy; if(ta>tb){const q=ta;ta=tb;tb=q;}
    t0=Math.max(t0,ta); t1=Math.min(t1,tb); if(t0>t1) return 1e9;
  }
  return t0;
}

function segRectT(x1,y1,x2,y2,r){

 else{let a=(r.y-y1)/dy,b=(r.y+r.h-y1)/dy;if(a>b){const q=a;a=b;b=q}t0=Math.max(t0,a);t1=Math.min(t1,b);if(t0>t1)return 1e9}
