function circleRect(rx,ry,rw,rh){
  const cxp=clamp(ball.x,rx,rx+rw),cyp=clamp(ball.y,ry,ry+rh);
  let dx=ball.x-cxp,dy=ball.y-cyp;const d2=dx*dx+dy*dy;
  if(d2>=ball.r*ball.r)return;
  let nx,ny;
  if(d2<1e-6){
    const l=ball.x-rx,rg=rx+rw-ball.x,tp=ball.y-ry,bt=ry+rh-ball.y,m=Math.min(l,rg,tp,bt);
    if(m===l){nx=-1;ny=0;}else if(m===rg){nx=1;ny=0;}else if(m===tp){nx=0;ny=-1;}else{nx=0;ny=1;}
  }else{const d=Math.sqrt(d2);nx=dx/d;ny=dy/d;}
  ball.x=cxp+nx*ball.r;ball.y=cyp+ny*ball.r;
  const vn=ball.vx*nx+ball.vy*ny;
  if(vn<0){const j=-(1+REST)*vn;ball.vx+=j*nx;ball.vy+=j*ny;
    if(-vn>90){sfx.wall(-vn);puffAt(cxp,cyp,nx,ny,-vn);shake=Math.min(5,-vn/260);}}
}
function stepBall(h){
  const cx=(ball.x-OX)/CELL|0,cy=(ball.y-OY)/CELL|0;
  let fr=260,inSand=false;
  if(cx>=0&&cx<COLS&&cy>=0&&cy<ROWS&&sandCells.some(c=>c.x===cx&&c.y===cy)){fr=950;inSand=true;}
  ball.vx+=wind.x*h;ball.vy+=wind.y*h;
  const sp=Math.hypot(ball.vx,ball.vy);
  if(sp>0){const dec=fr*h;if(dec>=sp){ball.vx=0;ball.vy=0;}else{const k=(sp-dec)/sp;ball.vx*=k;ball.vy*=k;}}
  ball.x+=ball.vx*h;ball.y+=ball.vy*h;
  // frame
  let hit=0;
  if(ball.x<OX+ball.r){ball.x=OX+ball.r;if(ball.vx<0){hit=Math.abs(ball.vx);ball.vx*=-REST;}}
  else if(ball.x>OX+FW-ball.r){ball.x=OX+FW-ball.r;if(ball.vx>0){hit=Math.abs(ball.vx);ball.vx*=-REST;}}
  if(ball.y<OY+ball.r){ball.y=OY+ball.r;if(ball.vy<0){if(hit===0)hit=Math.abs(ball.vy);ball.vy*=-REST;}}
  else if(ball.y>OY+FH-ball.r){ball.y=OY+FH-ball.r;if(ball.vy>0){if(hit===0)hit=Math.abs(ball.vy);ball.vy*=-REST;}}
  if(hit>120){sfx.wall(hit);shake=Math.min(5,hit/260);}
  // wall cells
  const x0=Math.max(0,((ball.x-OX-ball.r)/CELL)|0),x1=Math.min(COLS-1,((ball.x-OX+ball.r)/CELL)|0);
  const y0=Math.max(0,((ball.y-OY-ball.r)/CELL)|0),y1=Math.min(ROWS-1,((ball.y-OY+ball.r)/CELL)|0);
  for(let y=y0;y<=y1;y++)for(let x=x0;x<=x1;x++)if(grid[y][x]===1)circleRect(OX+x*CELL,OY+y*CELL,CELL,CELL);
  // gates
  for(const g of gates){const r=gateRect(g);circleRect(r.x,r.y,r.w,r.h);}
  // bumpers
  for(const b of bumpers){
    const dx=ball.x-b.x,dy=ball.y-b.y,rr=b.r+ball.r,d2=dx*dx+dy*dy;
    if(d2<rr*rr&&d2>1e-6){
      const d=Math.sqrt(d2),nx=dx/d,ny=dy/d;
      ball.x=b.x+nx*rr;ball.y=b.y+ny*rr;
      const vn=ball.vx*nx+ball.vy*ny;if(vn<0){ball.vx-=2*vn*nx;ball.vy-=2*vn*ny;}
      const s=Math.hypot(ball.vx,ball.vy);if(s<380){const k=380/(s||1);ball.vx*=k;ball.vy*=k;}
      b.flash=1;sfx.bump();shake=4;sparkAt(b.x,b.y);
    }
  }
  // water
  if(cx>=0&&cx<COLS&&cy>=0&&cy<ROWS&&waterCells.some(c=>c.x===cx&&c.y===cy)){
    splashFx(ball.x,ball.y);sfx.water();
    floats.push({x:ball.x,y:ball.y-24,txt:'SPLASH!',t:0,life:1.2,col:'#7cc0ff',size:18});
    ball.x=restPos.x;ball.y=restPos.y;ball.vx=0;ball.vy=0;state='aim';
    if(strokes>=maxStrokes)loseHeart();
    return;
  }
  // sand dust
  if(inSand&&Math.hypot(ball.vx,ball.vy)>50&&Math.random()<.35)dustAt(ball.x,ball.y);
  // cup
  const dx2=ball.x-cup.x,dy2=ball.y-cup.y,d=Math.hypot(dx2,dy2),sp2=Math.hypot(ball.vx,ball.vy);
  if(d<CUP_R+BALL_R&&d>0.1){
    if(sp2<380){const pull=(1-d/(CUP_R+BALL_R))*2400;ball.vx+=dx2/d*pull*h;ball.vy+=dy2/d*pull*h;}
    if(d<CUP_R*.95&&sp2<260){doSink();return;}
  }
}
function doSink(){state='sinking';sinkT=0;scored=false;sinkFrom={x:ball.x,y:ball.y};sfx.sink();}

/* ============================ GAME FLOW ============================ */
function doPutt(power,ux,uy){
  const sp=power*4.7;ball.vx=ux*sp;ball.vy=uy*sp;
  strokes++;totalStrokes++;restPos={x:ball.x,y:ball.y};
  state='rolling';trail.length=0;sfx.putt(Math.min(1,power/MAXDRAG));
}
function applyScore(){
  const diff=strokes-par;let txt,col,pts;
  if(diff<=-2){txt='EAGLE!';col='#ffd166';pts=400;}
  else if(diff===-1){txt='BIRDIE!';col='#3ec56b';pts=250;}
  else if(diff===0){txt='PAR';col='#f2e7c9';pts=100;}
  else if(diff===1){txt='BOGEY';col='#ff9f43';pts=50;}
  else{txt='DOUBLE BOGEY';col='#ff5a4e';pts=25;}
  if(diff<=0){streak++;pts+=(streak-1)*50;}else streak=0;
  score+=pts;
  floats.push({x:cup.x,y:cup.y-36,txt,t:0,life:1.5,col,size:26});
  floats.push({x:cup.x,y:cup.y-10,txt:'+'+pts,t:0,life:1.5,col:'#ffffff',size:15});
  if(diff<0)confetti(diff<=-2?46:30);else if(diff===0)confetti(10);
}

}

/* ============================ UPDATE ============================ */
function gameUpdate(dt){
  for(const g of gates)g.t+=g.speed*dt;
  for(const b of bumpers)b.flash=Math.max(0,b.flash-dt*2.5);
  ballPop=Math.max(0,ballPop-dt/.4);
  shownScore+=(score-shownScore)*Math.min(1,dt*8);if(Math.abs(score-shownScore)<1)shownScore=score;
  shake=Math.max(0,shake-dt*14);flashT=Math.max(0,flashT-dt*2);
  updParticles(dt);updFloats(dt);

  if(state==='banner'){bannerT+=dt;if(bannerT>=1.35){state='aim';ballPop=1;}}
  else if(state==='rolling'){
    const n=Math.max(1,Math.ceil(dt*120)),h=dt/n;
    for(let i=0;i<n&&state==='rolling';i++)stepBall(h);
    if(state==='rolling'){
      trail.push({x:ball.x,y:ball.y});if(trail.length>24)trail.shift();
      if(Math.hypot(ball.vx,ball.vy)<10){
        ball.vx=0;ball.vy=0;
        if(strokes>=maxStrokes)loseHeart();else state='aim';
      }
    }
  }
  else if(state==='sinking'){
    sinkT=Math.min(1,sinkT+dt/.5);flagDrop=sinkT;
    const k=sinkT*sinkT;
    ball.x=sinkFrom.x+(cup.x-sinkFrom.x)*k;ball.y=sinkFrom.y+(cup.y-sinkFrom.y)*k;
    if(sinkT>=1&&!scored){scored=true;applyScore();nextAt=time+.85;}
    if(scored&&time>=nextAt)nextHole();
  }
  if(state!=='rolling'&&trail.length)trail.splice(0,2);
}

function drawAim(){
  if(state!=='aim'||!aim.on)return;
  const dx=ball.x-aim.px,dy=ball.y-aim.py,len=Math.hypot(dx,dy);
  if(len<8)return;
  const ux=dx/len,uy=dy/len,p=Math.min(MAXDRAG,len);
  const t=p/MAXDRAG,col=t<.5?mixCol('#6fe08f','#ffd166',t*2):mixCol('#ffd166','#ff5a4e',(t-.5)*2);
  // trajectory dots (bounces once)
  let bx=ball.x+ux*(BALL_R+2),by=ball.y+uy*(BALL_R+2),dxn=ux,dyn=uy,used=0,bnc=0,segUsed=0;
  const maxL=50+p*1.6;
  ctx.fillStyle=col;
  while(used<maxL){
    let nx=bx+dxn*5,ny=by+dyn*5;
    if(solidPx(nx,ny)){
      if(bnc>=1)break;bnc++;
      if(!solidPx(nx,by)){dyn=-dyn;ny=by-dyn*2;}else{dxn=-dxn;nx=bx-dxn*2;}
      segUsed=0;
    }
    bx=nx;by=ny;used+=5;segUsed+=5;
    if(bnc===1&&segUsed>60)break;
    ctx.globalAlpha=Math.max(.12,.85-used/maxL*1.3);
    ctx.beginPath();ctx.arc(bx,by,2.6,0,TAU);ctx.fill();
  }
  ctx.globalAlpha=1;
  // power ring
  if(len>=6){ctx.strokeStyle=col;ctx.lineWidth=4;ctx.lineCap='round';
    ctx.beginPath();ctx.arc(ball.x,ball.y,17,-Math.PI/2,-Math.PI/2+(p/MAXDRAG)*TAU);ctx.stroke();ctx.lineCap='butt';}
}
