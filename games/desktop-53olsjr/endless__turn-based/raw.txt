function los(x0,y0,x1,y1){
 let dx=Math.abs(x1-x0),dy=Math.abs(y1-y0);
 let sx=x0<x1?1:-1,sy=y0<y1?1:-1;
 let err=dx-dy;

 while(!(x===x1&&y===y1)){

  if(e2<dx){err+=dx;y+=sy;}
  if(x===x1&&y===y1)break;

 }
 return true;
}

function update(dt,t){
  const k=1-Math.exp(-dt*15),P=S.p;
  P.px+=(P.x*tile+tile/2-P.px)*k;P.py+=(P.y*tile+tile/2-P.py)*k;
  P.lunge=Math.max(0,(P.lunge||0)-dt*7);
  for(const e of S.enemies){
    e.px+=(e.x*tile+tile/2-e.px)*k;e.py+=(e.y*tile+tile/2-e.py)*k;
    e.flash=Math.max(0,e.flash-dt*4);
    if(e.dead)e.dieT+=dt;
    if(e.shape==='drake'&&!e.dead&&S.mode==='play'&&Math.random()<dt*1.2)
      S.parts.push({x:e.px+tile*.3,y:e.py-tile*.1,vx:R(-8,8),vy:R(-40,-15),life:.6,max:.6,size:R(1,2.2),color:'#ff9a4d',grav:-20});
  }
  S.enemies=S.enemies.filter(e=>!e.dead||e.dieT<.5);
  for(const it of S.items){it.px+=(it.x*tile+tile/2-it.px)*k;it.py+=(it.y*tile+tile/2-it.py)*k;}
  const mwpx=S.mw*tile,mhpx=S.mh*tile;
  let tx=mwpx<=W?mwpx/2:clamp(P.px,W/2,mwpx-W/2);
  let ty=mhpx<=H?mhpx/2:clamp(P.py,H/2,mhpx-H/2);
  const ck=1-Math.exp(-dt*6);
  S.camX+=(tx-S.camX)*ck;S.camY+=(ty-S.camY)*ck;
  S.shake=Math.max(0,S.shake-dt*26);
  for(const p of S.parts){
    if(p.ring)p.r+=p.vr*dt;
    else{p.vy+=(p.grav||0)*dt;p.x+=p.vx*dt;p.y+=p.vy*dt;}
    p.life-=dt;
  }
  S.parts=S.parts.filter(p=>p.life>0);
  if(S.parts.length>420)S.parts.splice(0,S.parts.length-420);
  for(const f of S.floats)f.t+=dt;
  S.floats=S.floats.filter(f=>f.t<f.life);
  if(S.mode==='dead')S.deadT+=dt;
  if(S.mode==='start'){
    if(Math.random()<dt*38)wellE.push({x:R(W),y:H+8,vy:-R(24,66),ph:R(TAU),r:R(1,3),life:1,max:1,col:Math.random()<.85?'#f5a83c':'#9fd6ff'});
    for(const e of wellE){e.y+=e.vy*dt;e.x+=Math.sin(t*2+e.ph)*14*dt;e.life-=dt*.22;}
    for(let i=wellE.length-1;i>=0;i--)if(wellE[i].life<=0||wellE[i].y<-10)wellE.splice(i,1);
  }
  refreshHUD();renderLog();
}

  dot(s*.02,-s*.3,s*.125*sc,'#101826');
  ctx.shadowColor='#ffcf7d';ctx.shadowBlur=6;
  dot(0,-s*.31,s*.02,'#ffcf7d');dot(s*.055,-s*.31,s*.02,'#ffcf7d');
  ctx.shadowBlur=0;
  if(S.mode!=='dead'){
    ctx.strokeStyle='#d9d4c3';ctx.lineWidth=s*.055;
    ctx.beginPath();ctx.moveTo(s*.2,s*.1);ctx.lineTo(s*.48,-s*.14);ctx.stroke();

function drawDungeon(t){
  ctx.fillStyle='#05070d';ctx.fillRect(0,0,W,H);
  const shx=(Math.random()*2-1)*S.shake,shy=(Math.random()*2-1)*S.shake;
  const ox=W/2-S.camX+shx,oy=H/2-S.camY+shy;
  ctx.save();ctx.translate(ox,oy);
  const x0=Math.max(0,Math.floor((S.camX-W/2)/tile)-1),x1=Math.min(S.mw-1,Math.ceil((S.camX+W/2)/tile)+1);
  const y0=Math.max(0,Math.floor((S.camY-H/2)/tile)-1),y1=Math.min(S.mh-1,Math.ceil((S.camY+H/2)/tile)+1);
  for(let y=y0;y<=y1;y++)for(let x=x0;x<=x1;x++){
    const kk=K(x,y);if(!S.explored.has(kk))continue;
    const L=S.visible.has(kk)?(S.lightMap.get(kk)??1):.3;
    if(S.grid[y][x]===1)drawFloor(x,y,L);
    else if(adjFloor(x,y))drawWall(x,y,L);
  }
  drawStairs(t);
  for(const it of S.items)if(S.explored.has(K(it.x,it.y)))drawItem(it,t);
  for(const e of S.enemies){
    const kk=K(e.x,e.y);
    if(S.visible.has(kk)||e.dead)drawEnemy(e,t);
  }
  drawPlayer(t);
  ctx.globalCompositeOperation='lighter';
  for(const p of S.parts){
    const al=clamp(p.life/p.max,0,1);
    if(p.ring){ctx.strokeStyle=p.color;ctx.globalAlpha=al*.8;ctx.lineWidth=3;
      ctx.beginPath();ctx.arc(p.x,p.y,p.r,0,TAU);ctx.stroke();}
    else{ctx.globalAlpha=al;dot(p.x,p.y,p.size*(p.life/p.max),p.color);}
  }
  ctx.globalAlpha=1;ctx.globalCompositeOperation='source-over';
  for(const f of S.floats){
    const pr=f.t/f.life,al=1-pr*pr;
    ctx.font=(f.big?'800 ':'700 ')+Math.round(f.big?16:13)+'px '+MONO;ctx.textAlign='center';
    ctx.globalAlpha=al;ctx.lineWidth=3;ctx.strokeStyle='rgba(0,0,0,.8)';
    const yy=f.y-f.t*30;
    ctx.strokeText(f.text,f.x,yy);ctx.fillStyle=f.color;ctx.fillText(f.text,f.x,yy);
  }
  ctx.globalAlpha=1;
  ctx.restore();
  const sx=S.p.px+ox,sy=S.p.py+oy;
  const flick=.82+.13*Math.sin(t*11)+.05*Math.sin(t*23.7);
  ctx.globalCompositeOperation='lighter';
  const g=ctx.createRadialGradient(sx,sy,0,sx,sy,tile*7*flick);
  g.addColorStop(0,'rgba(255,166,64,'+(.1*flick).toFixed(3)+')');g.addColorStop(1,'rgba(255,166,64,0)');
  ctx.fillStyle=g;ctx.fillRect(sx-tile*8,sy-tile*8,tile*16,tile*16);
  ctx.globalCompositeOperation='source-over';
  if(S.mode==='dead'){
    ctx.fillStyle='rgba(2,3,8,'+clamp(S.deadT*.45,0,.6).toFixed(2)+')';
    ctx.fillRect(0,0,W,H);
  }
}

function octa(x,y,r,rot){
  ctx.beginPath();
  for(let i=0;i<8;i++){const a=i/8*TAU+Math.PI/8+rot;
    const px=x+Math.cos(a)*r,py=y+Math.sin(a)*r;
    i?ctx.lineTo(px,py):ctx.moveTo(px,py);}
  ctx.closePath();
}
function drawWell(t){
  ctx.fillStyle='#05070d';ctx.fillRect(0,0,W,H);
  const cx=W*.5,cy=H*.68,maxR=Math.hypot(W,H)*.7;
  for(let i=0;i<14;i++){
    const f=((i/14)+t*.045)%1;
    const r=maxR*Math.pow(1-f,1.9);
    if(r<8)continue;
    const a=Math.sin(f*Math.PI)*.5+.06;
    ctx.strokeStyle='rgba(104,112,138,'+(a*.4).toFixed(3)+')';
    ctx.lineWidth=Math.max(2,r*.14);octa(cx,cy,r,t*.03);ctx.stroke();
    ctx.strokeStyle='rgba(5,7,13,'+(a*.8).toFixed(3)+')';
    ctx.lineWidth=Math.max(1,r*.05);octa(cx,cy,r*1.06,t*.03);ctx.stroke();
  }
  const g=ctx.createRadialGradient(cx,cy,0,cx,cy,maxR*.42);
  g.addColorStop(0,'rgba(0,0,0,.95)');g.addColorStop(.7,'rgba(3,4,8,.6)');g.addColorStop(1,'rgba(3,4,8,0)');
  ctx.fillStyle=g;ctx.beginPath();ctx.arc(cx,cy,maxR*.42,0,TAU);ctx.fill();
  ctx.globalCompositeOperation='lighter';
  for(const e of wellE){
    ctx.globalAlpha=clamp(e.life,0,1)*.8;
    dot(e.x,e.y,e.r,e.col);
  }
  ctx.globalAlpha=1;ctx.globalCompositeOperation='source-over';
}
