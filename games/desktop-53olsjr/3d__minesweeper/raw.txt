let eye=[0,0,0], Rw=[1,0,0], Uw=[0,1,0], Fw=[0,0,1];
function basis(){
  const cp=Math.cos(cam.pitch),sp=Math.sin(cam.pitch),cyw=Math.cos(cam.yaw),syw=Math.sin(cam.yaw);
  eye=[cam.tgt[0]+cam.dist*cp*syw, cam.tgt[1]+cam.dist*sp, cam.tgt[2]+cam.dist*cp*cyw];
  const f=norm3(sub3(cam.tgt,eye));
  Rw=norm3(cross(f,[0,1,0])); Uw=cross(Rw,f); Fw=f;
}
const sub3=(a,b)=>[a[0]-b[0],a[1]-b[1],a[2]-b[2]];
const cross=(a,b)=>[a[1]*b[2]-a[2]*b[1], a[2]*b[0]-a[0]*b[2], a[0]*b[1]-a[1]*b[0]];
const norm3=a=>{const l=Math.hypot(a[0],a[1],a[2])||1;return [a[0]/l,a[1]/l,a[2]/l];};
let shX=0,shY=0;
function project(p){ const dx=p[0]-eye[0],dy=p[1]-eye[1],dz=p[2]-eye[2];
  const z=dx*Fw[0]+dy*Fw[1]+dz*Fw[2]; if(z<.15)return null;
  return [cxS+shX+F*(dx*Rw[0]+dy*Rw[1]+dz*Rw[2])/z, cyS+shY-F*(dx*Uw[0]+dy*Uw[1]+dz*Uw[2])/z, z]; }
function rayDir(mx,my){ const x=(mx-cxS)/F, y=-(my-cyS)/F;
  return norm3([Rw[0]*x+Uw[0]*y+Fw[0], Rw[1]*x+Uw[1]*y+Fw[1], Rw[2]*x+Uw[2]*y+Fw[2]]); }
function rayBox(ox,oy,oz,dx,dy,dz,cx,cy,cz,h){
  let tmin=.02,tmax=1e9;
  if(Math.abs(dx)<1e-8){if(ox<cx-h||ox>cx+h)return -1;}else{let a=(cx-h-ox)/dx,b=(cx+h-ox)/dx;if(a>b)[a,b]=[b,a];tmin=Math.max(tmin,a);tmax=Math.min(tmax,b);}
  if(Math.abs(dy)<1e-8){if(oy<cy-h||oy>cy+h)return -1;}else{let a=(cy-h-oy)/dy,b=(cy+h-oy)/dy;if(a>b)[a,b]=[b,a];tmin=Math.max(tmin,a);tmax=Math.min(tmax,b);}
  if(Math.abs(dz)<1e-8){if(oz<cz-h||oz>cz+h)return -1;}else{let a=(cz-h-oz)/dz,b=(cz+h-oz)/dz;if(a>b)[a,b]=[b,a];tmin=Math.max(tmin,a);tmax=Math.min(tmax,b);}
  return tmax>tmin?tmin:-1;
}

}

function frame(now){

  if(keys.l)cam.yaw+=dt*1.6; if(keys.r)cam.yaw-=dt*1.6;

  cam.pitch=clamp(cam.pitch,.08,1.42);
  if(mode==='menu')cam.yaw+=dt*.12;
  // hover lift anim
  for(let i=lifts.length-1;i>=0;i--){const l=lifts[i];
    const tgt=l.idx===hoverIdx?.05:0; l.a=lerp(l.a,tgt,1-Math.exp(-dt*14)); if(tgt===0&&l.a<.003)lifts.splice(i,1);}
  if(hoverIdx>=0&&!lifts.some(l=>l.idx===hoverIdx))lifts.push({idx:hoverIdx,a:0});
  // shake/flash
  shX=(Math.random()-.5)*shakeA; shY=(Math.random()-.5)*shakeA; shakeA*=Math.exp(-dt*5); if(shakeA<.1)shakeA=0;
  flashA=Math.max(0,flashA-dt*2.2);
  // bubbles
  for(const b of bubbles){ b.y-=b.s*dt/vh; if(b.y<-.02)b.y=1.02; }
  // particles
  for(let i=particles.length-1;i>=0;i--){ const p=particles[i];
    p.life-=dt; if(p.life<=0){particles.splice(i,1);continue;}
    const g=p.type==='shard'?6:p.type==='spark'?2.4:3; p.vy-=g*dt;
    p.x+=p.vx*dt; p.y+=p.vy*dt; p.z+=p.vz*dt; p.rot+=p.vr*dt; }
  if(mode==='won'&&confettiQ>0){spawnConfetti(Math.min(6,confettiQ));confettiQ-=6;}

  // render
  ctx.clearRect(0,0,vw,vh);
  drawBubbles(t);
  basis();
  drawFloor();
  const items=[]; collect(items,t);
  items.sort((a,b)=>b.z-a.z);
  for(const it of items){ if(it.a<.05)continue; ctx.globalAlpha=it.a; it.draw(); }
  ctx.globalAlpha=1;
  drawParticles();
  if(flashA>0){ctx.fillStyle=`rgba(255,64,32,${flashA*.3})`;ctx.fillRect(0,0,vw,vh);}

  // HUD time (cheap)
  let s; if(mode==='play'&&startTime)s=(now-startTime)/1000; else if(timeFinal)s=timeFinal/1000; else s=0;
  const str=fmtT(s); if(str!==lastTimeStr){lastTimeStr=str;$('timeV').textContent=str;}
  requestAnimationFrame(frame);
}
