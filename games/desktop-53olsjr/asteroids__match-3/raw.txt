try{muted=localStorage.getItem('ac_mute')==='1';}catch(e){}
function initAudio(){
  if(AC){ if(AC.state==='suspended')AC.resume(); return; }
  try{
    const Ctx=window.AudioContext||window.webkitAudioContext;
    AC=new Ctx(); master=AC.createGain();
    master.gain.value=muted?0:0.3; master.connect(AC.destination);
  }catch(e){}
}
function beep(f,d,type,vol,slide,delay){
  if(!AC)return;
  const t0=AC.currentTime+(delay||0), o=AC.createOscillator(), g=AC.createGain();
  o.type=type||'sine'; o.frequency.setValueAtTime(f,t0);
  if(slide)o.frequency.exponentialRampToValueAtTime(Math.max(30,slide),t0+d);
  g.gain.setValueAtTime(0.0001,t0);
  g.gain.exponentialRampToValueAtTime(vol||0.4,t0+0.012);
  g.gain.exponentialRampToValueAtTime(0.0001,t0+d);
  o.connect(g); g.connect(master); o.start(t0); o.stop(t0+d+0.05);
}
function hiss(d,vol,fq,delay){
  if(!AC)return;
  const t0=AC.currentTime+(delay||0), n=Math.max(1,(AC.sampleRate*d)|0);
  const buf=AC.createBuffer(1,n,AC.sampleRate), data=buf.getChannelData(0);
  for(let i=0;i<n;i++)data[i]=(Math.random()*2-1)*(1-i/n);
  const src=AC.createBufferSource(); src.buffer=buf;
  const f=AC.createBiquadFilter(); f.type='lowpass'; f.frequency.value=fq||900;
  const g=AC.createGain(); g.gain.value=vol||0.3;
  src.connect(f); f.connect(g); g.connect(master); src.start(t0);
}

function rockPts(){const n=9+((Math.random()*3)|0),a=[];for(let i=0;i<n;i++){const t=i/n*6.283,rr=0.7+Math.random()*0.4;a.push([Math.cos(t)*rr,Math.sin(t)*rr]);}return a;}

}

/* ================= PIECES ================= */
function makePiece(type,special,r,c,ay){
  const n=8+((Math.random()*3)|0), rock=[];
  for(let i=0;i<n;i++){const a=(i/n)*6.283+(Math.random()-0.5)*0.3,rr=0.34+Math.random()*0.10;rock.push([Math.cos(a)*rr,Math.sin(a)*rr]);}
  return {type,special,r,c,ax:c,ay:(ay===undefined?r:ay),vy:0,alpha:1,popT:-1,shake:0,
    spinP:Math.random()*6.283,spinS:0.4+Math.random()*0.6,rock};
}
function renderGem(g,x,y,p,t,sz){
  if(p.popT>=0&&p.popT>=1)return;
  const sc=(p.popT>=0)?1+p.popT*0.5:1;
  const al=p.alpha*(p.popT>=0?Math.max(0,1-p.popT):1);
  if(al<=0.01)return;
  const col=GEMS[p.type].c, R=sz*0.46;
  g.save();g.translate(x,y);
  if(p.shake>0)g.translate(Math.sin(t*68+p.spinP)*p.shake*3.2,Math.cos(t*59+p.spinP)*p.shake*2.2);
  g.rotate(Math.sin(t*p.spinS+p.spinP)*0.13);
  g.scale(sc,sc);g.globalAlpha*=al;
  g.beginPath();
  for(let i=0;i<p.rock.length;i++){const q=p.rock[i];i?g.lineTo(q[0]*R,q[1]*R):g.moveTo(q[0]*R,q[1]*R);}
  g.closePath();
  const rg=g.createLinearGradient(-R,-R,R,R);
  rg.addColorStop(0,'#42537a');rg.addColorStop(0.55,'#232e49');rg.addColorStop(1,'#121a2c');
  g.fillStyle=rg;g.fill();
  g.lineWidth=Math.max(1,sz*0.02);g.strokeStyle='rgba(180,215,255,0.18)';g.stroke();
  const cr=sz*0.22*(p.special==='b'?1+0.1*Math.sin(t*6+p.spinP):1);
  g.shadowColor=col;g.shadowBlur=sz*0.3;
  g.fillStyle=col;
  g.beginPath();g.moveTo(0,-cr);g.lineTo(cr*0.62,0);g.lineTo(0,cr);g.lineTo(-cr*0.62,0);g.closePath();g.fill();
  g.shadowBlur=0;
  g.fillStyle='rgba(255,255,255,0.7)';
  g.beginPath();g.moveTo(0,-cr*0.45);g.lineTo(cr*0.28,0);g.lineTo(0,cr*0.45);g.lineTo(-cr*0.28,0);g.closePath();g.fill();
  g.fillStyle='rgba(255,255,255,0.9)';
  g.beginPath();g.arc(-cr*0.3,-cr*0.34,sz*0.032,0,6.283);g.fill();
  if(p.special==='h'||p.special==='v'){
    const dx=p.special==='h'?1:0, dy=p.special==='v'?1:0;
    g.strokeStyle='rgba(255,240,200,'+(0.5+0.3*Math.sin(t*7)).toFixed(3)+')';
    g.lineWidth=sz*0.05;g.lineCap='round';
    for(let i=0;i<2;i++){const off=(i-0.5)*cr*1.6;
      const cxo=off*dx, cyo=off*dy, w=cr*0.42;
      g.beginPath();
      g.moveTo(cxo-dx*w-dy*w, cyo-dy*w+dx*w);
      g.lineTo(cxo+dx*w,    cyo+dy*w);
      g.lineTo(cxo-dx*w+dy*w,cyo-dy*w-dx*w);
      g.stroke();
    }
  }else if(p.special==='b'){
    g.strokeStyle='rgba(255,255,255,'+(0.35+0.3*Math.sin(t*6)).toFixed(3)+')';
    g.lineWidth=sz*0.035;
    g.beginPath();g.arc(0,0,cr*1.75,0,6.283);g.stroke();
    g.fillStyle='rgba(255,255,255,0.9)';
    for(let i=0;i<4;i++){const a=i*1.5708+Math.PI/4;
      g.beginPath();g.arc(Math.cos(a)*cr*1.75,Math.sin(a)*cr*1.75,sz*0.032,0,6.283);g.fill();}
  }else if(p.special==='x'){
    g.save();g.rotate(t*1.4);
    g.strokeStyle=col;g.globalAlpha*= (0.55+0.3*Math.sin(t*5));
    g.lineWidth=sz*0.04;g.lineCap='round';
    for(let i=0;i<4;i++){g.beginPath();g.moveTo(sz*0.1,0);g.lineTo(sz*0.46,0);g.stroke();g.rotate(Math.PI/2);}
    g.restore();
  }
  g.restore();
}

/* ================= BOARD LOGIC ================= */
function initBoard(){
  for(let tries=0;tries<60;tries++){
    grid=[];
    for(let r=0;r<ROWS;r++){const row=[];
      for(let c=0;c<COLS;c++){let t;
        do{t=(Math.random()*NTYPES)|0;}
        while((c>=2&&row[c-1].type===t&&row[c-2].type===t)||(r>=2&&grid[r-1][c].type===t&&grid[r-2][c].type===t));
        row.push(makePiece(t,'-',r,c));}
      grid.push(row);}
    if(hasMove())break;
  }
  for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++){
    const p=grid[r][c]; p.ay=r-ROWS-Math.random()*2+c*0.14; p.vy=0;
  }
}
function findRuns(){
  const runs=[];
  for(let r=0;r<ROWS;r++){let c=0;while(c<COLS){const p=grid[r][c];if(!p){c++;continue;}
    let e=c+1;while(e<COLS&&grid[r][e]&&grid[r][e].type===p.type)e++;
    if(e-c>=3)runs.push({dir:'h',a:r,b:c,len:e-c});c=e;}}
  for(let c=0;c<COLS;c++){let r=0;while(r<ROWS){const p=grid[r][c];if(!p){r++;continue;}
    let e=r+1;while(e<ROWS&&grid[e][c]&&grid[e][c].type===p.type)e++;
    if(e-r>=3)runs.push({dir:'v',a:c,b:r,len:e-r});r=e;}}
  return runs;
}
function inRun(run,r,c){return run.dir==='h'?(r===run.a&&c>=run.b&&c<run.b+run.len):(c===run.a&&r>=run.b&&r<run.b+run.len);}
function hasMove(){
  const tg=grid.map(row=>row.map(p=>p?p.type:-1));
  function runAt(r,c){const t=tg[r][c];let n=1,i;
    i=c-1;while(i>=0&&tg[r][i]===t)n++,i--;i=c+1;while(i<COLS&&tg[r][i]===t)n++,i++;if(n>=3)return true;
    n=1;i=r-1;while(i>=0&&tg[i][c]===t)n++,i--;i=r+1;while(i<ROWS&&tg[i][c]===t)n++,i++;return n>=3;}
  for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++){
    if(c+1<COLS&&tg[r][c]!==tg[r][c+1]){const a=tg[r][c];tg[r][c]=tg[r][c+1];tg[r][c+1]=a;
      const ok=runAt(r,c)||runAt(r,c+1);tg[r][c+1]=tg[r][c];tg[r][c]=a;if(ok)return true;}
    if(r+1<ROWS&&tg[r][c]!==tg[r+1][c]){const a=tg[r][c];tg[r][c]=tg[r+1][c];tg[r+1][c]=a;
      const ok=runAt(r,c)||runAt(r+1,c);tg[r+1][c]=tg[r][c];tg[r][c]=a;if(ok)return true;}
  }
  return false;
}
function reshuffleBoard(){
  toast('NO PATHS — BELT RESHUFFLED'); sfx.shuffle();
  let tries=0;
  do{
    const flat=[];grid.forEach(row=>row.forEach(p=>flat.push(p.type)));
    for(let i=flat.length-1;i>0;i--){const j=(Math.random()*(i+1))|0;const tmp=flat[i];flat[i]=flat[j];flat[j]=tmp;}
    let i=0;grid.forEach(row=>row.forEach(p=>{p.type=flat[i++];p.spinP=Math.random()*6.283;p.popT=-1;}));
    tries++;
  }while((!hasMove()||findRuns().length>0)&&tries<40);
  if(!hasMove()){initBoard();state='resolve';introDrop=true;resolvePhase='fall';}
}

/* ================= RESOLVE MACHINE ================= */
function beginResolve(swapCells){state='resolve';combo=1;doStep(swapCells);}
function doStep(swapCells){
  const runs=findRuns();
  if(!runs.length){finishResolve();return;}
  const key=(r,c)=>r*COLS+c, toClear=new Set(), createMap=new Map();
  for(const run of runs)for(let i=0;i<run.len;i++){
    const r=run.dir==='h'?run.a:run.b+i, c=run.dir==='h'?run.b+i:run.a;
    toClear.add(key(r,c));
  }
  const rank={h:1,v:1,b:2,x:3};
  function setCreate(r,c,sp){const k=key(r,c),old=createMap.get(k);if(!old||rank[sp]>rank[old])createMap.set(k,sp);}
  for(const run of runs){
    if(run.len<4)continue;
    const sp=run.len>=5?'x':(run.dir==='h'?'h':'v');
    let pick=null;
    if(swapCells)for(const s of swapCells){if(inRun(run,s.r,s.c)){pick=s;break;}}
    if(!pick){const mid=(run.len/2)|0;pick=run.dir==='h'?{r:run.a,c:run.b+mid}:{r:run.b+mid,c:run.a};}
    setCreate(pick.r,pick.c,sp);
  }
  for(const h of runs)if(h.dir==='h')for(const v of runs)if(v.dir==='v'){
    if(v.b<=h.a&&h.a<v.b+v.len&&h.b<=v.a&&v.a<h.b+h.len)setCreate(h.a,v.a,'b');
  }
  let trig=0;const triggered=new Set(),q=[...toClear];
  while(q.length){
    const k=q.pop(), r=(k/COLS)|0, c=k%COLS, p=grid[r][c];
    if(!p||!p.special||triggered.has(k))continue;
    triggered.add(k);trig++;
    expand(r,c,p.special,toClear,q,key);
  }
  createMap.forEach((sp,k)=>toClear.delete(k));
  const pts=toClear.size*10*combo+trig*50*combo+createMap.size*30*combo;
  score+=pts;levelScore+=pts;updateHUD();
  let sx=0,sy=0,n=0;
  toClear.forEach(k=>{const r=(k/COLS)|0,c=k%COLS;sx+=c+0.5;sy+=r+0.5;n++;});
  if(n){
    const px=board.ox+(sx/n)*board.cell, py=board.oy+(sy/n)*board.cell;
    floatText(px,py,'+'+pts,combo>1?'#ffe9a8':'#bfefff',combo>1);
    if(combo>1)floatText(px,py-board.cell*0.75,'CHAIN ×'+combo,'#ff9d5c',false);
  }
  let spawned=0;
  toClear.forEach(k=>{const r=(k/COLS)|0,c=k%COLS,p=grid[r][c];if(!p)return;
    p.popT=0;
    if(spawned<46){spawnShards(board.ox+(c+0.5)*board.cell,board.oy+(r+0.5)*board.cell,p.type);spawned++;}});
  triggered.forEach(k=>{const r=(k/COLS)|0,c=k%COLS,p=grid[r][c];
    spawnRing(board.ox+(c+0.5)*board.cell,board.oy+(r+0.5)*board.cell,GEMS[p.type].c,board.cell*1.9);});
  sfx.pop(combo); if(trig>0)sfx.boom();
  popCells=toClear; newSpecials=createMap; resolvePhase='pop'; popTimer=0;
}
function expand(r,c,sp,set,q,key){
  const add=(rr,cc)=>{if(rr<0||cc<0||rr>=ROWS||cc>=COLS)return;const k=key(rr,cc);if(!set.has(k)){set.add(k);q.push(k);}};
  if(sp==='h')for(let i=0;i<COLS;i++)add(r,i);
  else if(sp==='v')for(let i=0;i<ROWS;i++)add(i,c);
  else if(sp==='b'){for(let rr=r-1;rr<=r+1;rr++)for(let cc=c-1;cc<=c+1;cc++)add(rr,cc);}
  else if(sp==='x'){for(let i=0;i<COLS;i++)add(r,i);for(let i=0;i<ROWS;i++)add(i,c);}
}
function commitPop(){
  popCells.forEach(k=>{const r=(k/COLS)|0,c=k%COLS;if(newSpecials.has(k))return;grid[r][c]=null;});
  let created=0;
  newSpecials.forEach((sp,k)=>{const r=(k/COLS)|0,c=k%COLS,p=grid[r][c];
    if(p){p.special=sp;p.popT=-1;created++;
      spawnRing(board.ox+(c+0.5)*board.cell,board.oy+(r+0.5)*board.cell,GEMS[p.type].c,board.cell*1.4);}});
  if(created)sfx.create();
  setupFall();
}
function setupFall(){
  let any=false;
  for(let c=0;c<COLS;c++){
    let write=ROWS-1;
    for(let r=ROWS-1;r>=0;r--){const p=grid[r][c];
      if(p){if(write!==r){grid[write][c]=p;grid[r][c]=null;p.r=write;p.vy=0;any=true;}write--;}}
    const empty=write+1;
    for(let r=write;r>=0;r--){const p=makePiece((Math.random()*NTYPES)|0,'-',r,c,r-empty);grid[r][c]=p;any=true;}
  }
  resolvePhase='fall';
  if(!any)onFallSettled();
}
function stepFall(dt){
  let settled=true;
  for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++){const p=grid[r][c];if(!p)continue;
    if(p.ay<p.r-1e-6){
      p.vy=Math.min(16,p.vy+78*dt);p.ay+=p.vy*dt;settled=false;
      if(p.ay>=p.r){
        if(p.vy>8)spawnShards(board.ox+(c+0.5)*board.cell,board.oy+(r+0.42)*board.cell,p.type,3);
        p.ay=p.r;p.vy=p.vy>3.4?-p.vy*0.16:0;
      }
    }}
  if(settled)onFallSettled();
}
function onFallSettled(){
  if(introDrop){introDrop=false;finishResolve();return;}
  combo++;doStep(null);
}
function finishResolve(){
  combo=0;
  if(levelScore>=target){levelClear();state='playing';return;}
  if(moves<=0){gameOver();return;}
  if(!hasMove())reshuffleBoard();
  state='playing';updateHUD();
}
function levelClear(){
  const done=level;level++;target=targetFor(level);moves=Math.min(30,moves+12);levelScore=0;
  banner('BELT '+String(done).padStart(2,'0')+' CLEARED','+12 MOVES · NEW TARGET '+fmt(target));
  sfx.level();updateHUD();
}

/* ================= SWAP ================= */
function trySwap(a,b){
  if(state!=='playing')return;
  if(b.r<0||b.c<0||b.r>=ROWS||b.c>=COLS)return;
  const pA=grid[a.r][a.c],pB=grid[b.r][b.c];if(!pA||!pB)return;
  sel=null;state='swap';
  swapAnim={A:pA,B:pB,t:0,dur:0.16,back:false,
    from:[[pA.ax,pA.ay],[pB.ax,pB.ay]],to:[[b.c,b.r],[a.c,a.r]]};
  sfx.swap();
}
function stepSwap(dt){
  swapAnim.t+=dt;
  const k=Math.min(1,swapAnim.t/swapAnim.dur);
  const e=k<0.5?2*k*k:1-Math.pow(-2*k+2,2)/2;
  const A=swapAnim.A,B=swapAnim.B,f=swapAnim.from,to=swapAnim.to;
  A.ax=f[0][0]+(to[0][0]-f[0][0])*e;A.ay=f[0][1]+(to[0][1]-f[0][1])*e;
  B.ax=f[1][0]+(to[1][0]-f[1][0])*e;B.ay=f[1][1]+(to[1][1]-f[1][1])*e;
  if(k<1)return;
  const r1=A.r,c1=A.c,r2=B.r,c2=B.c;
  if(!swapAnim.back){
    grid[r1][c1]=B;grid[r2][c2]=A;
    A.r=r2;A.c=c2;A.ax=c2;A.ay=r2;B.r=r1;B.c=c1;B.ax=c1;B.ay=r1;
    if(findRuns().length>0){moves--;pulseMoves();updateHUD();beginResolve([{r:r1,c:c1},{r:r2,c:c2}]);}
    else{
      grid[r1][c1]=A;grid[r2][c2]=B;A.r=r1;A.c=c1;B.r=r2;B.c=c2;
      sfx.bad();A.shake=1;B.shake=1;
      swapAnim={A,B,t:0,dur:0.15,back:true,from:[[A.ax,A.ay],[B.ax,B.ay]],to:[[c1,r1],[c2,r2]]};
    }
  }else{
    A.ax=c1;A.ay=r1;B.ax=c2;B.ay=r2;swapAnim=null;state='playing';
  }
}

}

function bevelPath(g,x,y,w,h,b){g.beginPath();g.moveTo(x+b,y);g.lineTo(x+w,y);g.lineTo(x+w,y+h-b);
  g.lineTo(x+w-b,y+h);g.lineTo(x,y+h);g.lineTo(x,y+b);g.closePath();}
function brackets(cx,cy,s,col,alpha){
  const h=s/2,L=s*0.18;
  ctx.strokeStyle=col;ctx.globalAlpha=alpha;ctx.lineWidth=2;
  ctx.beginPath();
  ctx.moveTo(cx-h,cy-h+L);ctx.lineTo(cx-h,cy-h);ctx.lineTo(cx-h+L,cy-h);
  ctx.moveTo(cx+h-L,cy-h);ctx.lineTo(cx+h,cy-h);ctx.lineTo(cx+h,cy-h+L);
  ctx.moveTo(cx+h,cy+h-L);ctx.lineTo(cx+h,cy+h);ctx.lineTo(cx+h-L,cy+h);
  ctx.moveTo(cx-h+L,cy+h);ctx.lineTo(cx-h,cy+h);ctx.lineTo(cx-h,cy+h-L);
  ctx.stroke();ctx.globalAlpha=1;
}

let gtime=0,last=performance.now();
function update(dt){
  for(const d of drift){d.x+=d.vx*dt;d.y+=d.vy*dt;
    if(d.x<-0.12)d.x=1.12;if(d.x>1.12)d.x=-0.12;if(d.y<-0.12)d.y=1.12;if(d.y>1.12)d.y=-0.12;}
  for(const rk of bg.rocks){rk.x-=rk.spd*dt;rk.rot+=rk.vr*dt;if(rk.x<-0.16)rk.x=1.16;}
  if(Math.random()<dt*0.22&&bg.shoots.length<3)
    bg.shoots.push({x:Math.random()*W,y:Math.random()*H*0.45,
      vx:(Math.random()<0.5?-1:1)*(260+Math.random()*240),vy:80+Math.random()*120,life:0,max:0.7+Math.random()*0.3});
  for(let i=bg.shoots.length-1;i>=0;i--){const s=bg.shoots[i];s.life+=dt;s.x+=s.vx*dt;s.y+=s.vy*dt;if(s.life>s.max)bg.shoots.splice(i,1);}
  updateParts(dt);
  if(state==='start'||state==='gameover')return;
  for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++){const p=grid[r][c];if(p&&p.shake>0)p.shake=Math.max(0,p.shake-dt*2.2);}
  if(state==='swap'){stepSwap(dt);return;}
  if(state==='resolve'){
    if(resolvePhase==='pop'){
      popTimer+=dt;const k=Math.min(1,popTimer/POP_DUR);
      popCells.forEach(kk=>{if(newSpecials.has(kk))return;
        const p=grid[(kk/COLS)|0][kk%COLS];if(p)p.popT=k;});
      if(popTimer>=POP_DUR)commitPop();
    }else if(resolvePhase==='fall'){stepFall(dt);}
  }
}
