  grid=Array.from({length:ROWS},()=>Array(COLS).fill(null));
  bag=[]; score=0; lives=3; linesCleared=0; level=1; pieces=0; chain=0;
  particles=[]; floaters=[]; clearingRows=[]; banner=null; shake=0; flashRed=0; dangerOn=false;
  paddleX=(boardX+boardX+boardW)/2; prevPX=paddleX; mouseX=null;
  ball.held=true; ball.trail=[];
  spawnPiece();
}
function startGame(){
  resetGame(); state='playing';
  $('menu').classList.add('hidden'); $('overOv').classList.add('hidden'); $('pauseOv').classList.add('hidden');
  updateHUD(); sfx.serve&&tone(330,.08,'square',.12,0,520);
}
function serve(){
  if(!ball.held)return;
  ball.held=false;
  const a=(-Math.PI/2)+(Math.random()*.6-.3);
  const sp=ballSpeed();
  ball.vx=Math.cos(a)*sp; ball.vy=Math.sin(a)*sp;
  sfx.serve();
}
function loseLife(){
  lives--; chain=0; flashRed=.85; shake=Math.max(shake,14); sfx.life(); updateHUD();
  burst(ball.x,playBottom-4,'#ff5f56',16,CELL*3);
  if(lives<=0){gameOver();}
  else{ball.held=true;ball.trail=[];}
}
function gameOver(){
  state='over'; overAt=performance.now(); shake=Math.max(shake,20); flashRed=.9; sfx.over();
  let nb=false;
  if(score>bestStored()){nb=true;try{localStorage.setItem('tetroSmashBest',String(best));}catch(e){}}
  $('fScore').textContent=fmt(score); $('fBest').textContent=fmt(Math.max(best,bestStored()));
  $('fLines').textContent=linesCleared; $('fLevel').textContent=level; $('fPieces').textContent=pieces;
  $('newBest').classList.toggle('hidden',!nb);
  setTimeout(()=>{ if(state==='over')$('overOv').classList.remove('hidden'); },500);
}
function bestStored(){try{return parseInt(localStorage.getItem('tetroSmashBest')||'0',10)||0;}catch(e){return 0;}}

/* ---------- physics ---------- */
function solidCell(c,r){ return (r>=0&&r<ROWS&&grid[r][c])?grid[r][c]:null; }
function pieceCells(){
  const out=[]; if(!piece)return out;
  for(let y=0;y<piece.mat.length;y++)for(let x=0;x<piece.mat[y].length;x++)
    if(piece.mat[y][x])out.push({cx:piece.x+x,cy:piece.y+y,type:piece.type});
  return out;
}
function stepBall(dt){
  if(ball.held)return;
  const sp=Math.hypot(ball.vx,ball.vy)||1, steps=Math.max(1,Math.ceil((sp*dt)/(ball.r*0.8)));
  for(let s=0;s<steps;s++){
    ball.x+=ball.vx*dt/steps; ball.y+=ball.vy*dt/steps;
    // walls
    if(ball.x-ball.r<boardX){ball.x=boardX+ball.r;ball.vx=Math.abs(ball.vx);sfx.wall();}
    else if(ball.x+ball.r>boardX+boardW){ball.x=boardX+boardW-ball.r;ball.vx=-Math.abs(ball.vx);sfx.wall();}
    if(ball.y-ball.r<boardY){ball.y=boardY+ball.r;ball.vy=Math.abs(ball.vy);sfx.wall();}
    // paddle
    const ph=12, pl=paddleX-paddleW/2, pt=paddleY;
    if(ball.vy>0&&ball.y+ball.r>pt&&ball.y-ball.r<pt+ph&&ball.x>pl-ball.r&&ball.x<pl+paddleW+ball.r){
      const off=Math.max(-1,Math.min(1,(ball.x-paddleX)/(paddleW/2)));
      const ang=off*1.05, spd=ballSpeed();
      ball.vx=Math.sin(ang)*spd+paddleVX*.3;
      ball.vy=-Math.abs(Math.cos(ang)*spd);
      ball.y=pt-ball.r-.5;
      chain=0; updateHUD(); sfx.paddle();
    }
    // stacked cells
    const c0=Math.max(0,((ball.x-ball.r-boardX)/CELL|0)-1), c1=Math.min(COLS-1,((ball.x+ball.r-boardX)/CELL|0)+1);
    const r0=Math.max(0,((ball.y-ball.r-boardY)/CELL|0)-1), r1=Math.min(ROWS-1,((ball.y+ball.r-boardY)/CELL|0)+1);
    for(let r=r0;r<=r1;r++)for(let c=c0;c<=c1;c++){
      const col=solidCell(c,r); if(!col)continue;
      collideBox(boardX+c*CELL,boardY+r*CELL,CELL,CELL,col,true,c,r);
    }
    // falling piece (solid too — smash it mid-air!)
    for(const pc of pieceCells()){
      if(pc.cy<0)continue;
      collideBox(boardX+pc.cx*CELL,boardY+pc.cy*CELL,CELL,CELL,COLORS[pc.type],false,pc.cx,pc.cy,pc);
    }
  }
  // keep constant arcade speed
  const cur=Math.hypot(ball.vx,ball.vy)||1, tgt=ballSpeed();
  ball.vx*=tgt/cur; ball.vy*=tgt/cur;
  if(ball.y-ball.r>playBottom)loseLife();
}
function collideBox(bx,by,bw,bh,color,isGrid,c,r,pc){
  const px=Math.max(bx,Math.min(ball.x,bx+bw)), py=Math.max(by,Math.min(ball.y,by+bh));
  let dx=ball.x-px, dy=ball.y-py; const d2=dx*dx+dy*dy;
  if(d2>ball.r*ball.r)return;
  let nx,ny,pen;
  if(d2>1e-6){const d=Math.sqrt(d2);nx=dx/d;ny=dy/d;pen=ball.r-d;}
  else{ // center inside: push out shortest face
    const l=ball.x-bx,rt=bx+bw-ball.x,tp=ball.y-by,bt=by+bh-ball.y,m=Math.min(l,rt,tp,bt);
    if(m===l){nx=-1;ny=0;}else if(m===rt){nx=1;ny=0;}else if(m===tp){nx=0;ny=-1;}else{nx=0;ny=1;}
    pen=ball.r+m;
  }
  ball.x+=nx*pen; ball.y+=ny*pen;
  const vn=ball.vx*nx+ball.vy*ny;
  if(vn<0){ball.vx-=2*vn*nx;ball.vy-=2*vn*ny;sfx.wall();}
  // smash!
  chain++;
  const pts=10*Math.min(chain,10); addScore(pts);
  sfx.smash(chain);
  burst(ball.x- nx*ball.r,ball.y-ny*ball.r,color,8,CELL*4.5);
  floater(bx+bw/2,by,'+'+pts,chain>=4?'#ffc857':'#ffffff',13);
  if(chain===4)banner={text:'CHAIN ×4!',t:0,dur:.9,c:'#ffc857'};
  if(chain===8)banner={text:'CHAIN ×8!!',t:0,dur:1.1,c:'#ff9838'};
  const gx=boardX+c*CELL+CELL/2, gy=boardY+r*CELL+CELL/2;
  if(isGrid){ grid[r][c]=null; burst(gx,gy,color,10,CELL*4); }
  else{ pc.type=null; checkShatter(); }
}
function checkShatter(){
  if(!piece)return;
  let alive=false;
  for(const row of piece.mat)for(const v of row)if(v){alive=true;break;}
  if(!alive){ addScore(50); floater(boardX+boardW/2,boardY+boardH*.4,'PIECE SHATTERED +50','#ffd23e',18);
    burst(boardX+boardW/2,ball.y,'#ffd23e',20,CELL*5); sfx.tetris(); spawnPiece(); }
}

/* ---------- lock & clears ---------- */
function lockPiece(){
  let above=false;
  for(let y=0;y<piece.mat.length;y++)for(let x=0;x<piece.mat[y].length;x++){
    if(!piece.mat[y][x])continue;
    const cy=piece.y+y,cx=piece.x+x;
    if(cy<0){above=true;continue;}
    grid[cy][cx]=COLORS[piece.type];
  }
  pieces++; sfx.lock();
  burst(boardX+(piece.x+1.5)*CELL,boardY+Math.max(0,piece.y+1)*CELL,COLORS[piece.type],6,CELL*3);
  // level up every 7 pieces
  const nl=1+Math.floor(pieces/7);
  if(nl>level){level=nl;banner={text:'LEVEL '+level,t:0,dur:1.4,c:'#57d7ff'};sfx.levelup();}
  // full rows
  const full=[];
  for(let r=0;r<ROWS;r++){let f=true;for(let c=0;c<COLS;c++)if(!grid[r][c]){f=false;break;}if(f)full.push(r);}
  if(full.length){
    clearingRows=full.map(r=>({r,t:0}));
    const pts=[0,40,100,300,800][Math.min(full.length,4)];
    addScore(pts); linesCleared+=full.length;
    const names=['','SINGLE','DOUBLE','TRIPLE','TETRIS!'];
    banner={text:names[full.length]+'  +'+pts,t:0,dur:1.2,c:full.length===4?'#ffd23e':'#5be07a'};
    if(full.length===4)sfx.tetris();else sfx.clear(full.length);
    shake=Math.max(shake,full.length*4);
  }
  // top-out?
  let topped=above;
  for(let r=0;r<2&&!topped;r++)for(let c=0;c<COLS;c++)if(grid[r][c]){topped=true;break;}
  if(topped){gameOver();return;}
  spawnPiece();
}
function finishClears(){
  const rows=clearingRows.map(o=>o.r).sort((a,b)=>b-a);
  for(const r of rows)for(let c=0;c<COLS;c++)if(grid[r][c])burst(boardX+c*CELL+CELL/2,boardY+r*CELL+CELL/2,grid[r][c],3,CELL*3);
  grid=grid.filter((_,i)=>!clearingRows.some(o=>o.r===i));
  while(grid.length<ROWS)grid.unshift(Array(COLS).fill(null));
  clearingRows=[];
}

/* ---------- update ---------- */
function update(dt){
  // paddle
  prevPX=paddleX;
  const ps=Math.max(480,W*0.55);
  if(keys.l)paddleX-=ps*dt;
  if(keys.r)paddleX+=ps*dt;
  if(mouseX!=null&&!keys.l&&!keys.r){const t=mouseX;paddleX+=(t-paddleX)*Math.min(1,dt*18);}
  paddleX=Math.max(boardX+paddleW/2,Math.min(boardX+boardW-paddleW/2,paddleX));
  paddleVX=(paddleX-prevPX)/Math.max(dt,.001);

  // falling piece
  if(piece){
    dropTimer+=dt;
    const iv=fallInterval()*(softDrop?0.16:1);
    if(dropTimer>=iv){dropTimer=0;
      if(fits(piece.mat,piece.x,piece.y+1))piece.y++;
      else lockPiece();
    }
  }
  // clears anim
  for(const o of clearingRows)o.t+=dt;
  if(clearingRows.length&&clearingRows.every(o=>o.t>=0.28))finishClears();

  stepBall(dt);

  // held ball follows paddle
  if(ball.held){ball.x=paddleX;ball.y=paddleY-ball.r-2;}
  else{ball.trail.push({x:ball.x,y:ball.y});if(ball.trail.length>10)ball.trail.shift();}

  // danger state
  let top=ROWS;
  for(let r=0;r<ROWS;r++){for(let c=0;c<COLS;c++)if(grid[r][c]){top=r;break;}if(top!==ROWS)break;}
  const dn=top<=DANGER_ROW;
  if(dn&&!dangerOn){dangerOn=true;sfx.danger();}
  if(!dn)dangerOn=false;

  // fx decay
  for(let i=particles.length-1;i>=0;i--){const p=particles[i];p.life-=dt;p.x+=p.vx*dt;p.y+=p.vy*dt;p.vy+=CELL*9*dt;if(p.life<=0)particles.splice(i,1);}
  for(let i=floaters.length-1;i>=0;i--){const f=floaters[i];f.life-=dt;f.y-=34*dt;if(f.life<=0)floaters.splice(i,1);}
  if(banner){banner.t+=dt;if(banner.t>banner.dur)banner=null;}
  shake=Math.max(0,shake-dt*38);
  flashRed=Math.max(0,flashRed-dt*1.6);
}

/* ---------- drawing helpers ---------- */
function shade(hex,amt){
  const n=parseInt(hex.slice(1),16);
  let r=(n>>16)+amt,g=((n>>8)&255)+amt,b=(n&255)+amt;
  r=Math.max(0,Math.min(255,r));g=Math.max(0,Math.min(255,g));b=Math.max(0,Math.min(255,b));
  return 'rgb('+r+','+g+','+b+')';
}
function block(x,y,s,color,alpha){
  ctx.globalAlpha=alpha==null?1:alpha;
  ctx.fillStyle=color; ctx.fillRect(x,y,s,s);
  ctx.fillStyle=shade(color,52); ctx.fillRect(x,y,s,Math.max(2,s*.14)); ctx.fillRect(x,y,Math.max(2,s*.14),s);
  ctx.fillStyle=shade(color,-58); ctx.fillRect(x,y+s-Math.max(2,s*.16),s,Math.max(2,s*.16)); ctx.fillRect(x+s-Math.max(2,s*.16),y,Math.max(2,s*.16),s);
  ctx.globalAlpha=1;
}
function ghostBlock(x,y,s,color){
  ctx.globalAlpha=.14;ctx.fillStyle=color;ctx.fillRect(x+1,y+1,s-2,s-2);
  ctx.globalAlpha=.5;ctx.strokeStyle=color;ctx.lineWidth=1.5;ctx.strokeRect(x+1.5,y+1.5,s-3,s-3);ctx.globalAlpha=1;
}

/* ---------- draw ---------- */
function draw(){
  const t=tNow;
  // background layers
  ctx.fillStyle='#081222'; ctx.fillRect(0,0,W,H);
  let g=ctx.createRadialGradient(W*.3+Math.sin(t*.1)*40,H*.35,10,W*.3,H*.35,W*.7);
  g.addColorStop(0,'rgba(38,92,140,.28)');g.addColorStop(1,'rgba(0,0,0,0)');
  ctx.fillStyle=g;ctx.fillRect(0,0,W,H);
  g=ctx.createRadialGradient(W*.75+Math.cos(t*.13)*50,H*.7,10,W*.75,H*.7,W*.6);
  g.addColorStop(0,'rgba(140,92,38,.16)');g.addColorStop(1,'rgba(0,0,0,0)');
  ctx.fillStyle=g;ctx.fillRect(0,0,W,H);
  for(const s of stars){
    const a=.25+.5*Math.abs(Math.sin(t*s.tw+s.ph));
    ctx.globalAlpha=a;ctx.fillStyle='#cfe6ff';
    ctx.fillRect(s.x+Math.sin(t*.2+s.ph)*3,s.y,s.r,s.r);
  }
  ctx.globalAlpha=1;

  // menu attract: drifting tetrominoes
  if(state==='menu'){
    for(const gh of ghosts){
      const gx=(gh.x+.05*Math.sin(t*.3+gh.ph))*W, gy=((gh.y+t*.02)%1.2-0.1)*H;
      ctx.save();ctx.translate(gx,gy);ctx.rotate(gh.rot+t*gh.rs);
      const m=SHAPES[gh.t],sc=Math.max(8,CELL*.5),col=COLORS[gh.t];
      for(let y=0;y<m.length;y++)for(let x=0;x<m[y].length;x++)if(m[y][x]){
        ctx.globalAlpha=.1;ctx.fillStyle=col;
        ctx.fillRect((x-m.length/2)*sc,(y-m.length/2)*sc,sc-1,sc-1);
      }
      ctx.restore();ctx.globalAlpha=1;
    }
  }

  // world (with shake)
  ctx.save();
  if(shake>0)ctx.translate((Math.random()-.5)*shake,(Math.random()-.5)*shake);

  // arena frame
  const fy=boardY,fh=(playBottom-14)-fy;
  ctx.strokeStyle='rgba(87,215,255,.35)';ctx.lineWidth=2;
  ctx.beginPath();ctx.moveTo(boardX,fy);ctx.lineTo(boardX+boardW,fy);ctx.lineTo(boardX+boardW,playBottom-8);
  ctx.moveTo(boardX,fy);ctx.lineTo(boardX,playBottom-8);ctx.stroke();
  ctx.strokeStyle='rgba(120,170,230,.09)';ctx.lineWidth=1;
  for(let c=1;c<COLS;c++){ctx.beginPath();ctx.moveTo(boardX+c*CELL,fy);ctx.lineTo(boardX+c*CELL,fy+boardH);ctx.stroke();}
  for(let r=1;r<ROWS;r++){ctx.beginPath();ctx.moveTo(boardX,fy+r*CELL);ctx.lineTo(boardX+boardW,fy+r*CELL);ctx.stroke();}
  // floor line (where pieces land)
  ctx.strokeStyle='rgba(255,200,87,.35)';ctx.lineWidth=2;
  ctx.beginPath();ctx.moveTo(boardX,fy+boardH);ctx.lineTo(boardX+boardW,fy+boardH);ctx.stroke();

  if(state!=='menu'){
    // danger line
    if(dangerOn){
      const a=.35+.35*Math.sin(t*8);
      ctx.strokeStyle='rgba(255,84,104,'+a+')';ctx.lineWidth=2;ctx.setLineDash([8,6]);
      ctx.beginPath();ctx.moveTo(boardX,fy+DANGER_ROW*CELL);ctx.lineTo(boardX+boardW,fy+DANGER_ROW*CELL);ctx.stroke();
      ctx.setLineDash([]);
      ctx.font='700 10px Consolas,monospace';ctx.fillStyle='rgba(255,84,104,'+(a+.3)+')';
      ctx.fillText('⚠ DANGER',boardX+6,fy+DANGER_ROW*CELL-5);
    }
    // stacked blocks
    for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++){
      const col=grid[r][c]; if(col)block(boardX+c*CELL+1,fy+r*CELL+1,CELL-2,CELL-2,col);
    }
    // clearing flash
    for(const o of clearingRows){
      const a=Math.abs(Math.sin(o.t/0.28*Math.PI));
      ctx.fillStyle='rgba(255,255,255,'+(a*.75)+')';
      ctx.fillRect(boardX,fy+o.r*CELL,boardW,CELL);
    }
    // ghost + active piece
    if(piece){
      const gy=ghostY();
      for(let y=0;y<piece.mat.length;y++)for(let x=0;x<piece.mat[y].length;x++){
        if(!piece.mat[y][x])continue;
        const cy=gy+y; if(cy<0)continue;
        ghostBlock(boardX+(piece.x+x)*CELL+1,fy+(cy)*CELL+1,CELL-2,CELL-2,COLORS[piece.type]);
      }
      for(let y=0;y<piece.mat.length;y++)for(let x=0;x<piece.mat[y].length;x++){
        if(!piece.mat[y][x])continue;
        const cy=piece.y+y; if(cy<0)continue;
        block(boardX+(piece.x+x)*CELL+1,fy+cy*CELL+1,CELL-2,CELL-2,COLORS[piece.type]);
      }
    }
    // paddle (with motion smear)
    const ph=12,px=paddleX-paddleW/2;
    if(Math.abs(paddleVX)>60){ctx.globalAlpha=.25;ctx.fillStyle='#ffc857';
      ctx.fillRect(px-Math.sign(paddleVX)*8,paddleY,paddleW,ph);}
    ctx.globalAlpha=1;
    const pg=ctx.createLinearGradient(0,paddleY,0,paddleY+ph);
    pg.addColorStop(0,'#ffe3a1');pg.addColorStop(.5,'#ffc857');pg.addColorStop(1,'#c47a1e');
    ctx.fillStyle=pg;ctx.fillRect(px,paddleY,paddleW,ph);
    ctx.fillStyle='rgba(255,255,255,.8)';ctx.fillRect(px+3,paddleY+2,paddleW-6,2);
    ctx.shadowColor='rgba(255,200,87,.7)';ctx.shadowBlur=14;
    ctx.fillStyle='rgba(255,200,87,.25)';ctx.fillRect(px,paddleY,paddleW,ph);ctx.shadowBlur=0;

    // ball + trail
    for(let i=0;i<ball.trail.length;i++){
      const p=ball.trail[i],a=(i/ball.trail.length)*.35;
      ctx.globalAlpha=a;ctx.fillStyle='#8fe0ff';
      ctx.beginPath();ctx.arc(p.x,p.y,ball.r*(0.4+0.6*i/ball.trail.length),0,7);ctx.fill();
    }
    ctx.globalAlpha=1;
    const bg2=ctx.createRadialGradient(ball.x-ball.r*.3,ball.y-ball.r*.3,1,ball.x,ball.y,ball.r);
    bg2.addColorStop(0,'#ffffff');bg2.addColorStop(.6,'#bfeeff');bg2.addColorStop(1,'#4fb7e8');
    ctx.shadowColor='rgba(120,220,255,.9)';ctx.shadowBlur=16;
    ctx.fillStyle=bg2;ctx.beginPath();ctx.arc(ball.x,ball.y,ball.r,0,7);ctx.fill();ctx.shadowBlur=0;

    // serve hint
    if(ball.held&&state==='playing'){
      const a=.5+.4*Math.sin(t*6);
      ctx.font='700 13px Consolas,monospace';ctx.textAlign='center';
      ctx.fillStyle='rgba(255,255,255,'+a+')';
      ctx.fillText('SPACE / CLICK — LAUNCH',paddleX,paddleY-26);
      ctx.textAlign='left';
    }
  }

  // particles
  for(const p of particles){
    ctx.globalAlpha=Math.max(0,p.life/p.max);ctx.fillStyle=p.c;
    ctx.fillRect(p.x-p.sz/2,p.y-p.sz/2,p.sz,p.sz);
  }
  ctx.globalAlpha=1;
  // floaters
  for(const f of floaters){
    ctx.globalAlpha=Math.max(0,f.life/f.max);
    ctx.font='700 '+f.size+'px Consolas,monospace';ctx.textAlign='center';
    ctx.fillStyle=f.c;ctx.fillText(f.text,f.x,f.y);
  }
  ctx.globalAlpha=1;ctx.textAlign='left';
  // banner
  if(banner){
    const p=banner.t/banner.dur, a=p<.15?p/.15:(p>.7?(1-p)/.3:1);
    const sc=1+ (p<.2?(0.2-p)*3:0);
    ctx.save();ctx.translate(W/2,H*.32);ctx.scale(sc,sc);
    ctx.globalAlpha=Math.max(0,a);
    ctx.font='400 '+Math.max(30,CELL*1.5)+'px Impact,"Arial Narrow",sans-serif';
    ctx.textAlign='center';ctx.fillStyle=banner.c;
    ctx.shadowColor=banner.c;ctx.shadowBlur=24;
    ctx.fillText(banner.text,0,0);
    ctx.restore();ctx.globalAlpha=1;ctx.textAlign='left';
  }
  ctx.restore(); // shake

  // red damage vignette
  if(flashRed>0){
    const rg=ctx.createRadialGradient(W/2,H/2,H*.3,W/2,H/2,H*.85);
    rg.addColorStop(0,'rgba(255,60,70,0)');rg.addColorStop(1,'rgba(255,60,70,'+flashRed*.5+')');
    ctx.fillStyle=rg;ctx.fillRect(0,0,W,H);
  }
}

/* ---------- HUD ---------- */
const cache={};
function setTxt(el,v){if(cache[el.id]!==v){cache[el.id]=v;el.textContent=v;}}
function updateHUD(){
  setTxt(scoreEl,fmt(score)); setTxt(bestEl,'BEST '+fmt(Math.max(best,bestStored())));
  setTxt(linesEl,String(linesCleared)); setTxt(levelEl,String(level));
  setTxt(livesEl,'● '.repeat(Math.max(0,lives)).trim()||'—');
  const c=chain<2?1:Math.min(chain,10);
  setTxt(chainEl,'×'+c);
  chainEl.classList.toggle('on',chain>=2);
}

/* ---------- loop ---------- */
let last=performance.now();
function frame(now){
  tNow=now/1000;
  let dt=Math.min(.033,(now-last)/1000); last=now;
  if(state==='playing')update(dt);
  else{ // keep fx alive on menu/over
    for(let i=particles.length-1;i>=0;i--){const p=particles[i];p.life-=dt;p.x+=p.vx*dt;p.y+=p.vy*dt;p.vy+=CELL*9*dt;if(p.life<=0)particles.splice(i,1);}
    shake=Math.max(0,shake-dt*38);flashRed=Math.max(0,flashRed-dt*1.6);
  }
  draw();
  requestAnimationFrame(frame);
}

    else if(state==='playing'&&ball.held)serve();
    else if(state==='over'&&performance.now()-overAt>600)startGame();
  }
  if(k==='Enter'){
    if(state==='menu'||(state==='over'&&performance.now()-overAt>400))startGame();
    else if(state==='paused')togglePause();
  }
  if((k==='ArrowUp'||k==='x'||k==='X'||k==='w'||k==='W')&&state==='playing')rotatePiece();
  if(k==='ArrowDown'||k==='s'||k==='S')softDrop=true;
  if(k==='p'||k==='P'||k==='Escape'){if(state==='playing'||state==='paused')togglePause();}
  if(k==='r'||k==='R'){if(state!=='menu')startGame();}
  if(k==='m'||k==='M'){muted=!muted;if(master)master.gain.value=muted?0:.5;
    try{localStorage.setItem('tetroSmashMuted',muted?'1':'0');}catch(e){}}
});
