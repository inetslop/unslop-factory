Circle vs AABB:
```
function collideBox(b, box){ // box {x,y,w,h}
  const cx = clamp(b.x, box.x, box.x+box.w);
  const cy = clamp(b.y, box.y, box.y+box.h);
  let dx = b.x-cx, dy = b.y-cy;
  let d2 = dx*dx+dy*dy;
  if(d2 > b.r*b.r) return null;
  let nx, ny, pen;
  if(d2 > 1e-9){ const d=Math.sqrt(d2); nx=dx/d; ny=dy/d; pen=b.r-d; }
  else { // center inside box: push out along min axis
    const l = b.x-box.x, rgt = box.x+box.w-b.x, t = b.y-box.y, bt = box.y+box.h-b.y;
    const m = Math.min(l,rgt,t,bt);
    if(m===l){nx=-1;ny=0;pen=b.r+l;} else if(m===rgt){nx=1;ny=0;pen=b.r+rgt;}
    else if(m===t){nx=0;ny=-1;pen=b.r+t;} else {nx=0;ny=1;pen=b.r+bt;}
  }
  return {nx,ny,pen,cx,cy};
}
```

Positional: b.x += nx*pen etc (full correction for static, times 0.9). Record impact = |j| for sound/particles/crumble.

  const bodies=[orb,...rocks].filter(b=>b.alive);
  for(const b of bodies){
    b.vx+=gvec.x*gm*h; b.vy+=gvec.y*gm*h;
    if(d.well){ const dx=d.well.x-b.x, dy=d.well.y-b.y; const dd=Math.hypot(dx,dy); if(dd<d.well.R&&dd>0.5){ const a=d.well.P*(1-dd/d.well.R); b.vx+=dx/dd*a*h; b.vy+=dy/dd*a*h; } }
    const dr=Math.max(0,1-0.12*h); b.vx*=dr; b.vy*=dr; // mild air drag... 0.12/s is fine for long flights
    const sp=Math.hypot(b.vx,b.vy); if(sp>1600){b.vx*=1600/sp;b.vy*=1600/sp;}
    b.x+=b.vx*h; b.y+=b.vy*h;
  }
  // body-body
  for(let i=0;i<bodies.length;i++)for(let j=i+1;j<bodies.length;j++) collideCC(bodies[i],bodies[j]);
  // body-solid
  const boxes=[...solids]; crumbles.forEach(c=>{if(!c.broken)boxes.push(c);});
  for(const b of bodies){
    for(const bx of boxes){ const hit=collideBox(b,bx); if(hit){ if(bx.crumble && hit.vn < -160 && hit.j > 2400){ bx.broken=true; SFX.crumble(); spawnDebris(...); shake=Math.max(shake,10);} } }
    // spikes (orb only)

  }
}

 let nx,ny,pen;

 // positional
 b.x+=nx*pen*0.85; b.y+=ny*pen*0.85;
 const vn=b.vx*nx+b.vy*ny; let j=0;

 }
