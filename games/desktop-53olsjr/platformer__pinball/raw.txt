let AC=null, master=null, muted=false, noiseBuf=null;
function initAudio(){
  if(AC) return;
  try{
    AC=new (window.AudioContext||window.webkitAudioContext)();
    master=AC.createGain(); master.gain.value=muted?0:0.5; master.connect(AC.destination);
    const len=AC.sampleRate|0; noiseBuf=AC.createBuffer(1,len,AC.sampleRate);
    const d=noiseBuf.getChannelData(0); for(let i=0;i<len;i++)d[i]=Math.random()*2-1;
  }catch(e){}
}
function tone(f,f2,dur,type,vol,delay){
  if(!AC||muted) return; delay=delay||0;
  try{
    const t=AC.currentTime+delay, o=AC.createOscillator(), g=AC.createGain();
    o.type=type; o.frequency.setValueAtTime(Math.max(1,f),t);
    if(f2) o.frequency.exponentialRampToValueAtTime(Math.max(1,f2),t+dur);
    g.gain.setValueAtTime(.0001,t); g.gain.linearRampToValueAtTime(vol,t+.006);
    g.gain.exponentialRampToValueAtTime(.0001,t+dur);
    o.connect(g); g.connect(master); o.start(t); o.stop(t+dur+.03);
  }catch(e){}
}
function noise(dur,vol,f0,f1,delay,type){
  if(!AC||muted) return; delay=delay||0; type=type||'lowpass';
  try{
    const t=AC.currentTime+delay, s=AC.createBufferSource(); s.buffer=noiseBuf; s.loop=true;
    const f=AC.createBiquadFilter(); f.type=type; f.frequency.setValueAtTime(f0,t);
    if(f1) f.frequency.exponentialRampToValueAtTime(Math.max(1,f1),t+dur);
    const g=AC.createGain(); g.gain.setValueAtTime(vol,t);
    g.gain.exponentialRampToValueAtTime(.0001,t+dur);
    s.connect(f); f.connect(g); g.connect(master); s.start(t); s.stop(t+dur+.03);
  }catch(e){}
}

function updFlip(f,dt){
  const d=f.tgt-f.a;
  if(Math.abs(d)<0.001){ f.w=0; return; }
  const flipping=(f===flippers.L)?d<0:d>0;
  const sp=(flipping?26:15)*(G.tilt>0?0.45:1);
  const step=Math.sign(d)*Math.min(Math.abs(d),sp*dt);
  f.a+=step; f.w=step/dt;
}
function segCol(ax,ay,bx,by,e){
  const b=G.ball, abx=bx-ax, aby=by-ay;
  let t=((b.x-ax)*abx+(b.y-ay)*aby)/(abx*abx+aby*aby);
  t=Math.max(0,Math.min(1,t));
  const cx=ax+abx*t, cy=ay+aby*t;
  let dx=b.x-cx, dy=b.y-cy; const d2=dx*dx+dy*dy;
  if(d2>=BALL_R*BALL_R) return null;
  const d=Math.sqrt(d2)||.0001, nx=dx/d, ny=dy/d;
  b.x=cx+nx*BALL_R; b.y=cy+ny*BALL_R;
  const vn=b.vx*nx+b.vy*ny;
  if(vn<0){ b.vx-=(1+e)*vn*nx; b.vy-=(1+e)*vn*ny; }
  return {nx,ny,vn};
}

function boxCol(x,y,w,h,e,isPad,po){
  const b=G.ball;

  if(d2>=BALL_R*BALL_R) return;
  let d=Math.sqrt(d2), nx, ny;
  if(d<.0001){
    const l=b.x-x,r=x+w-b.x,t=b.y-y,bo=y+h-b.y,m=Math.min(l,r,t,bo);
    if(m===l){nx=-1;ny=0;b.x=x-BALL_R;} else if(m===r){nx=1;ny=0;b.x=x+w+BALL_R;}
    else if(m===t){nx=0;ny=-1;b.y=y-BALL_R;} else {nx=0;ny=1;b.y=y+h+BALL_R;}
  } else { nx=dx/d; ny=dy/d; b.x=cx+nx*BALL_R; b.y=cy+ny*BALL_R; }
  const vn=b.vx*nx+b.vy*ny;

function flipCol(f){
  const b=G.ball;
  const tx=f.px+Math.cos(f.a)*FLIP_LEN, ty=f.py+Math.sin(f.a)*FLIP_LEN;
  const abx=tx-f.px, aby=ty-f.py;
  let t=((b.x-f.px)*abx+(b.y-f.py)*aby)/(abx*abx+aby*aby);
  t=Math.max(0,Math.min(1,t));
  const cx=f.px+abx*t, cy=f.py+aby*t;
  let dx=b.x-cx, dy=b.y-cy; const d2=dx*dx+dy*dy, lim=BALL_R+FLIP_R;
  if(d2>=lim*lim) return;
  const d=Math.sqrt(d2)||.0001, nx=dx/d, ny=dy/d;
  b.x=cx+nx*lim; b.y=cy+ny*lim;
  const rx=cx-f.px, ry=cy-f.py, svx=-f.w*ry, svy=f.w*rx;
  let rvx=b.vx-svx, rvy=b.vy-svy;

function update(dt){
  G.time+=dt;
  const inL=!!(keys.KeyA||keys.ArrowLeft||keys.KeyZ||ptrL), inR=!!(keys.KeyD||keys.ArrowRight||keys.KeyX||ptrR);
  flippers.L.tgt=inL?-0.42:0.5; flippers.R.tgt=inR?Math.PI+0.42:Math.PI-0.5;

  piston.y=piston.base+piston.amp*Math.sin(G.time*piston.sp);
  piston.vy=piston.amp*Math.cos(G.time*piston.sp)*piston.sp;

  if(G.charging&&G.phase==='ready'&&!paused){ G.charge=Math.min(1,G.charge+dt/0.85); }
  if(G.phase==='ready'){ G.ball.x=CX; G.ball.y=938-G.charge*26; }

  for(const k of [flippers.L,flippers.R]) updFlip(k,dt);
  bumpers.forEach(b=>{b.cd-=dt;if(b.flash>0)b.flash=Math.max(0,b.flash-dt*5);});
  pads.forEach(p=>{p.cd-=dt;if(p.flash>0)p.flash=Math.max(0,p.flash-dt*5);});
  slings.forEach(s=>{s.cd-=dt;if(s.flash>0)s.flash=Math.max(0,s.flash-dt*5);});
  flippers.L.hitCd-=dt; flippers.R.hitCd-=dt;
  if(G.comboT>0){ G.comboT-=dt; if(G.comboT<=0)G.combo=0; }
  if(G.comboPop>0)G.comboPop=Math.max(0,G.comboPop-dt*4);
  if(G.bannerT>0)G.bannerT-=dt;
  if(G.tilt>0)G.tilt-=dt;
  G.shake=Math.max(0,G.shake-dt*14);
  if(G.flash>0)G.flash=Math.max(0,G.flash-dt*2.2);

  /* particles */
  for(let i=parts.length-1;i>=0;i--){
    const p=parts[i]; p.life-=dt;
    if(p.life<=0){ parts.splice(i,1); continue; }
    if(p.type==='spark'||p.type==='conf'||p.type==='drop') p.vy+=900*dt;
    if(p.type==='smoke'){ p.vx*=1-2*dt; }
    p.x+=p.vx*dt; p.y+=p.vy*dt;
  }
  if(parts.length>320) parts.splice(0,parts.length-320);
  for(let i=texts.length-1;i>=0;i--){ const t=texts[i]; t.t-=dt*.9; t.y-=28*dt; if(t.t<=0)texts.splice(i,1); }

  /* bubbles */
  if(G.acidY<BOT&&Math.random()<dt*7) parts.push(bubble(TL+10+Math.random()*(TR-TL-20),G.acidY+8));

  if(G.phase==='play'){
    const SUB=5, h=dt/SUB;
    for(let i=0;i<SUB;i++){
      const b=G.ball;
      b.vy+=GRAV*h;
      const sp=Math.hypot(b.vx,b.vy);
      if(sp>MAXV){ b.vx*=MAXV/sp; b.vy*=MAXV/sp; }
      b.x+=b.vx*h; b.y+=b.vy*h; b.rot+=b.vx*h/BALL_R;

      domeCol();
      segCol(TL,TOP,TL,BOT,.75); segCol(TR,TOP,TR,BOT,.75);
      for(const d of deflects) segCol(d.a[0],d.a[1],d.b[0],d.b[1],.72);
      for(const s of slings){
        const r=segCol(s.a[0],s.a[1],s.b[0],s.b[1],.5);
        if(r&&r.vn<-90&&s.cd<=0){
          s.cd=.18; s.flash=1; G.score+=10; SFX.sling();
          b.vx+=r.nx*430; b.vy+=r.ny*430;
        }
        segCol(s.b[0],s.b[1],s.c[0],s.c[1],.5);
      }
      for(const p of pads) boxCol(p.x,p.y,p.w,p.h,.5,true,p);
      pistonCol();
      for(const bm of bumpers) bumperCol(bm);
      flipCol(flippers.L); flipCol(flippers.R);

      if(G.clearArmed&&b.y<CLEAR_Y) clearFloor();
      else if(!G.clearArmed&&b.y>CLEAR_Y+50) G.clearArmed=true;
    }
    const b=G.ball;
    for(const c of coins){
      if(c.gone) continue;
      if((b.x-c.x)**2+(b.y-c.y)**2<24*24){
        c.gone=true; G.score+=50; G.coinsGot++;
        addText(c.x,c.y,'+50','#ffd97a',16); SFX.coin();
        for(let i=0;i<6;i++) parts.push(spark(c.x,c.y,'#ffd97a'));
      }
    }
    /* acid */
    if(G.acidReset>0){
      G.acidY=Math.min(968,G.acidY+1300*dt);
      if(G.acidY>=968)G.acidReset=0;
      if(Math.random()<.6) parts.push(bubble(TL+Math.random()*(TR-TL),G.acidY-4));
    } else {
      if(G.acidGrace>0) G.acidGrace-=dt;
      else { G.acidY-= (16+(G.floor-1)*2.8)*dt; if(G.acidY<258)G.acidY=258; }
    }
    if(b.y+BALL_R>G.acidY) gameOver('acid');
    if(b.y-BALL_R>1005) gameOver('drain');
    if(!G.acidReset&&G.acidY<760){ G.tickT-=dt; if(G.tickT<=0){ SFX.tick(); G.tickT=.9; } }

    /* trail */
    const sp=Math.hypot(b.vx,b.vy);
    if(sp>420){ trail.push({x:b.x,y:b.y}); if(trail.length>16)trail.shift(); }
    else if(trail.length) trail.splice(0,2);
  }
}

const FD='Impact,"Arial Black",sans-serif', FB='"Trebuchet MS",Verdana,sans-serif';
function rr(c,x,y,w,h,r){
  c.beginPath(); c.moveTo(x+r,y);
  c.arcTo(x+w,y,x+w,y+h,r); c.arcTo(x+w,y+h,x,y+h,r);
  c.arcTo(x,y+h,x,y,r); c.arcTo(x,y,x+w,y,r); c.closePath();
}
function label(c,txt,x,y,col){
  c.font='12px '+FB; c.fillStyle=col||'#8b96a8'; c.textAlign='center'; c.fillText(txt,x,y);
}
function value(c,txt,x,y,size,col){
  c.font=size+'px '+FD; c.fillStyle=col||'#fff'; c.textAlign='center'; c.fillText(txt,x,y);
}
function kbdDraw(c,txt,x,y){

}

    }
  }

  /* particles */
  for(const p of parts){
    const a=Math.max(0,p.life/p.max);
    if(p.type==='smoke'){
      c.beginPath(); c.arc(p.x,p.y,p.r*(1+(1-a)*2),0,7);
      c.fillStyle='rgba(159,180,200,'+(a*.25)+')'; c.fill();
    } else if(p.type==='conf'){
      c.save(); c.translate(p.x,p.y); c.rotate(p.life*10);
      c.globalAlpha=a; c.fillStyle=p.c; c.fillRect(-2,-3.5,4,7); c.restore(); c.globalAlpha=1;
    } else {
      c.beginPath(); c.arc(p.x,p.y,p.r*(p.type==='bubble'?a:1),0,7);
      c.fillStyle=p.c; c.globalAlpha=a*(p.type==='bubble'?.8:1); c.fill(); c.globalAlpha=1;
    }
  }

  /* floating texts */
  for(const t of texts){
    c.font=t.size+'px '+FD; c.textAlign='center';
    c.globalAlpha=Math.min(1,t.t);
    c.strokeStyle='rgba(0,0,0,.6)'; c.lineWidth=3; c.strokeText(t.txt,t.x,t.y);
    c.fillStyle=t.c; c.fillText(t.txt,t.x,t.y);
  }
  c.globalAlpha=1;

  /* banner */
  if(G.bannerT>0){
    const t=G.bannerT, a=Math.min(1,t/.4), s=1+Math.max(0,t-1.2)*1.1;
    c.save(); c.translate(CX,268); c.scale(s,s); c.globalAlpha=a;
    c.font='54px '+FD; c.textAlign='center';
    c.strokeStyle='rgba(0,0,0,.7)'; c.lineWidth=6; c.strokeText(G.banner,0,0);
    c.fillStyle='#ffd97a'; c.shadowColor='#ffd97a'; c.shadowBlur=18; c.fillText(G.banner,0,0);
    c.restore(); c.globalAlpha=1;
  }

  /* TILT */
  if(G.tilt>0&&Math.sin(G.time*20)>0){
    c.font='40px '+FD; c.textAlign='center';
    c.strokeStyle='rgba(0,0,0,.7)'; c.lineWidth=5; c.strokeText('T I L T !',CX,548);
    c.fillStyle='#ff5c49'; c.fillText('T I L T !',CX,548);
  }

  /* danger rim pulse */
  if(G.phase==='play'&&!G.acidReset&&((968-G.acidY)/(968-862))>.75){
    c.strokeStyle='rgba(255,92,73,'+(.25+.2*Math.sin(G.time*10))+')';
    c.lineWidth=7; c.stroke(PF);
  }

  /* clear flash */
  if(G.flash>0){ c.save(); c.clip(PF); c.fillStyle='rgba(255,217,122,'+(G.flash*.28)+')'; c.fillRect(TL-20,TOP-ARC_R-10,TR-TL+40,BOT-TOP+ARC_R+20); c.restore(); }

  /* ready hint */
  if(G.phase==='ready'){
    const a=.6+.4*Math.sin(G.time*5);
    c.font='17px '+FD; c.textAlign='center';
    c.globalAlpha=a;
    c.strokeStyle='rgba(0,0,0,.7)'; c.lineWidth=4;
    c.strokeText(G.charging?'RELEASE!':'HOLD SPACE \u2014 RELEASE TO LAUNCH',CX,822);
    c.fillStyle='#ffd97a'; c.fillText(G.charging?'RELEASE!':'HOLD SPACE \u2014 RELEASE TO LAUNCH',CX,822);
    c.globalAlpha=1;
  }

  c.restore();

  /* vignette */
  c.setTransform(DPR,0,0,DPR,0,0);
  const vg=c.createRadialGradient(innerWidth/2,innerHeight/2,Math.min(innerWidth,innerHeight)*.45,innerWidth/2,innerHeight/2,Math.max(innerWidth,innerHeight)*.75);
  vg.addColorStop(0,'rgba(0,0,0,0)'); vg.addColorStop(1,'rgba(0,0,0,.4)');
  c.fillStyle=vg; c.fillRect(0,0,innerWidth,innerHeight);
}

let last=performance.now();
function frame(now){
  requestAnimationFrame(frame);
  let dt=(now-last)/1000; last=now;
  if(dt>0.05)dt=0.05;
  if(!paused) update(dt);
  render();
}
spawnCoins();
requestAnimationFrame(frame);
})();
