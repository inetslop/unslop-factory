```js
function collideWalls(o,r,rest){
  const cx=Math.floor(o.x), cy=Math.floor(o.y);
  let hit=0, impact=0;
  for(let yy=cy-1;yy<=cy+1;yy++)for(let xx=cx-1;xx<=cx+1;xx++){
    if(!isWall(xx,yy)) continue;
    const nx=clamp(o.x, xx, xx+1), ny=clamp(o.y, yy, yy+1);
    let dx=o.x-nx, dy=o.y-ny; let d2=dx*dx+dy*dy;
    if(d2 > r*r) continue;
    let d=Math.sqrt(d2);
    if(d<1e-6){ // center inside wall: push out along smallest axis
      const l=o.x-xx, rr=xx+1-o.x, t=o.y-yy, b=yy+1-o.y;
      const m=Math.min(l,rr,t,b);
      if(m===l){o.x=xx-r;} else if(m===rr){o.x=xx+1+r;} else if(m===t){o.y=yy-r;} else {o.y=yy+1+r;}
      continue;
    }
    const px=dx/d, py=dy/d;
    o.x=nx+px*r; o.y=ny+py*r;
    const vn=o.vx*px+o.vy*py;
    if(vn<0){ impact=Math.max(impact,-vn); o.vx-=(1+rest)*vn*px; o.vy-=(1+rest)*vn*py; }
  }
  return impact;
}
```

```js
function updateBall(dt){
  const b=ball;
  if(state!=='rolling') return; // sinking handled separately
  // friction
  const cell=floorAt(b.x,b.y);
  let k = cell==='S'?4.2:0.75;
  const f=Math.exp(-k*dt);
  b.vx*=f; b.vy*=f;
  // hole pull
  const dx=hole.x-b.x, dy=hole.y-b.y; const d=Math.hypot(dx,dy);
  if(d<1.35 && d>0.0001){
    const pull=14*(1-d/1.35);
    b.vx+=dx/d*pull*dt; b.vy+=dy/d*pull*dt;
  }
  let sp=Math.hypot(b.vx,b.vy);
  if(sp>MAXSPEED*1.3){...clamp}

  const steps=clamp(Math.ceil((sp*dt)/(ballR*0.6)),1,8);
```

```html
function collideWalls(o,r,rest){
  const cx=Math.floor(o.x),cy=Math.floor(o.y);let impact=0;
  for(let yy=cy-1;yy<=cy+1;yy++)for(let xx=cx-1;xx<=cx+1;xx++){
    if(!isWall(xx,yy))continue;
    const nx=clamp(o.x,xx,xx+1),ny=clamp(o.y,yy,yy+1);
    let dx=o.x-nx,dy=o.y-ny,d2=dx*dx+dy*dy;
    if(d2>r*r)continue;
    let d=Math.sqrt(d2);
    if(d<1e-6){
      const l=o.x-xx,rg=xx+1-o.x,t=o.y-yy,b=yy+1-o.y,m=Math.min(l,rg,t,b);
      if(m===l)o.x=xx-r;else if(m===rg)o.x=xx+1+r;else if(m===t)o.y=yy-r;else o.y=yy+1+r;
      continue;
    }
    const px=dx/d,py=dy/d;
    o.x=nx+px*r;o.y=ny+py*r;
    const vn=o.vx*px+o.vy*py;
    if(vn<0){impact=Math.max(impact,-vn);o.vx-=(1+rest)*vn*px;o.vy-=(1+rest)*vn*py;}
  }
  return impact;
}

    if(imp>0.6)sfx.bounce(imp);
    if(floorAt(b.x,b.y)==='~'){end='water';break;}
    const hd=Math.hypot(hole.x-b.x,hole.y-b.y);
    const hsp=Math.hypot(b.vx,b.vy);
    if(hd<holeR&&hsp<captureSpeed){end='sink';break;}
    else if(hd<holeR*.9&&hsp>=captureSpeed&&T-lastClack>0.12){b.vx*=.94;b.vy*=.94;lastClack=T;sfx.rim();}
    const cx=Math.floor(b.x),cy=Math.floor(b.y);
    if(dots[cy]&&dots[cy][cx])eatDot(cx,cy);
  }
  if(end==='water'){doSplash();return;}
  if(end==='sink'){startSink();return;}
  b.mouth+=Math.hypot(b.vx,b.vy)*dt*3;
  ghostCheck();
  sp=Math.hypot(b.vx,b.vy);
  if(sp<0.2){
    aimAngle=Math.atan2(b.vy,b.vx);
    b.vx=0;b.vy=0;phase='aim';origin={x:b.x,y:b.y};
  }
}
function startSink(){
  state='sinking';sinkT=0;sfx.sink();
  rings.push({x:hole.x,y:hole.y,r:.3,max:1.8,c:'rgba(255,230,0,',t:0});
}
function stepSink(dt){
  sinkT+=dt;
  const k=1-Math.exp(-9*dt);
  ball.x+=(hole.x-ball.x)*k;ball.y+=(hole.y-ball.y)*k;
  ball.spin+=14*dt;ball.scale=Math.max(0,1-sinkT/.5);
  if(sinkT>=.55)finishHole();
}
function finishHole(){
  if(state!=='sinking')return;
  const par=MAPS[holeIndex].par,d=strokes-par;
  let word='PAR',cls='winfo';
  if(strokes===1){word='ACE!';cls='wace';}
  else if(d<=-2){word='EAGLE!';cls='wgood';}
  else if(d===-1){word='BIRDIE!';cls='wgood';}
  else if(d===0){word='PAR';cls='winfo';}
  else if(d===1){word='BOGEY';cls='wbad';}
  else{word='+'+d;cls='wbad';}
  const bonus=Math.max(0,-d)*300+(strokes===1?500:0)+50;
  score+=bonus;results.push({par,strokes,word,cls});scored=true;
  state='done';doneT=0;updateHUD();
  toast(word+' · '+strokes+(strokes>1?' STROKES':' STROKE')+(d<0?'  ('+d+')':d>0?'  (+'+d+')':''),'big '+(cls==='wgood'||cls==='wace'?'good':cls==='wbad'?'bad':'info'),2100);
  floatText('+'+bonus,hole.x,hole.y-1,'#6dff8a');
  if(d<0)confetti();
}

/* ================= GHOSTS ================= */
function stepGhost(g,dt,chase){
  if(g.pop>0){g.pop-=dt;if(g.pop<=0){g.x=g.spawn.x;g.y=g.spawn.y;g.frozen=1;}return;}
  if(g.frozen>0){g.frozen-=dt;return;}
  const tx=chase?ball.x:start.x,ty=chase?ball.y:start.y;
  let ang=Math.atan2(ty-g.y,tx-g.x)+Math.sin(T*1.4+g.ph)*.55;
  g.look=ang;
  const v=MAPS[holeIndex].gs;
  g.vx=Math.cos(ang)*v;g.vy=Math.sin(ang)*v;
  g.x+=g.vx*dt;g.y+=g.vy*dt;
  collideWalls(g,ghostR,0);
}

/* ================= SHOT ================= */
function fire(ang,p){
  if(state!=='play'||phase!=='aim')return;
  strokes++;updateHUD();
  const v=p*MAXSPEED;
  ball.vx=Math.cos(ang)*v;ball.vy=Math.sin(ang)*v;
  phase='roll';charging=false;power=0;
  sfx.shot(p);spark(ball.x,ball.y,'#ffe9c9',5);
}

/* ================= FX (screen space) ================= */
const w2sx=wx=>wx*cs+ox, w2sy=wy=>wy*cs+oy;
function spark(wx,wy,c,n){
  for(let i=0;i<n;i++){const a=rnd(6.28),s=rnd(30,140);
    particles.push({x:w2sx(wx),y:w2sy(wy),vx:Math.cos(a)*s,vy:Math.sin(a)*s,g:0,life:rnd(.3,.7),max:.7,c,sz:rnd(1.5,3.5)});}
}
function confetti(){
  const csx=w2sx(hole.x),csy=w2sy(hole.y);
  const cols=['#ffe600','#7df9ff','#ffb8de','#6dff8a','#ffa245'];
  for(let i=0;i<46;i++){const a=rnd(-Math.PI,. )||rnd(0,6.28),s=rnd(60,230);
    particles.push({x:csx,y:csy,vx:Math.cos(a)*s,vy:Math.sin(a)*s-120,g:420,life:rnd(.7,1.4),max:1.4,c:cols[i%5],sz:rnd(2,4)});}
}
function floatText(txt,wx,wy,c){floats.push({x:w2sx(wx),y:w2sy(wy)-cs*.4,txt,c,t:0});}
function updateFx(dt){
  particles=particles.filter(p=>{p.life-=dt;p.vy+=p.g*dt;p.x+=p.vx*dt;p.y+=p.vy*dt;const dr=Math.max(0,1-3*dt);p.vx*=dr;p.vy*=dr;return p.life>0;});
  if(particles.length>320)particles.splice(0,particles.length-320);
  rings=rings.filter(r=>{r.t+=dt;r.r+=(r.max-.2)*dt/0.5;return r.t<0.5;});
  floats=floats.filter(f=>{f.t+=dt;return f.t<1;});
}

/* ================= UPDATE LOOP ================= */
function update(dt){
  T+=dt;
  if(shake>0)shake=Math.max(0,shake-dt*2.2);
  updateFx(dt);
  if(state==='menu'){ghosts.forEach(g=>stepGhost(g,dt,false));return;}
  if(paused||state==='final')return;
  ball.invuln=Math.max(0,ball.invuln-dt);
  if(state==='sinking'){stepSink(dt);ghosts.forEach(g=>stepGhost(g,dt,true));return;}
  if(state==='done'){
    doneT+=dt;ghosts.forEach(g=>stepGhost(g,dt,true));
    if(doneT>2){if(holeIndex+1<MAPS.length)loadHole(holeIndex+1);else showFinal();}
    return;
  }
  ghosts.forEach(g=>stepGhost(g,dt,true));
  if(phase==='aim'){
    const rot=2.8*dt;
    if(keys.has('ArrowLeft')||keys.has('a'))aimAngle-=rot;
    if(keys.has('ArrowRight')||keys.has('d'))aimAngle+=rot;
    if(charging){const p=((T-chargeStart)*0.75)%2;power=p<1?p:2-p;}
    ghostCheck();
    return;
  }
  updateBall(dt);
}

}
function drawCourse(){

function drawPellets(){
  for(let y=0;y<rows;y++)for(let x=0;x<cols;x++){
    const v=dots[y][x];if(!v)continue;
    const X=w2sx(x+.5),Y=w2sy(y+.5);
    if(v===1){
      const r=cs*.11*(1+.14*Math.sin(T*4+x*1.3+y));
      ctx.fillStyle='#ffe3c9';ctx.beginPath();ctx.arc(X,Y,r,0,7);ctx.fill();
    }else{
      const s=cs*.95*(1+.2*Math.sin(T*5));
      ctx.drawImage(tiles.glow,X-s/2,Y-s/2,s,s);
      ctx.fillStyle='#ffd84d';ctx.beginPath();ctx.arc(X,Y,cs*.26,0,7);ctx.fill();
    }
  }
}
function drawGhost(g){
  if(g.pop>0)return;
  const frozen=g.frozen>0;
  let jx=0,jy=0;if(frozen){jx=rnd(-1,1);jy=rnd(-1,1);}
  const x=w2sx(g.x)+jx,y=w2sy(g.y)+jy,r=ghostR*cs;
  ctx.save();ctx.translate(x,y);
  if(!frozen){ctx.shadowColor=g.c;ctx.shadowBlur=r*.5;}
  ctx.fillStyle=frozen?'#2b3fe0':g.c;
  const by=r*.75,seg=r*.5;
  ctx.beginPath();ctx.arc(0,-r*.15,r,Math.PI,0);ctx.lineTo(r,by);
  for(let i=1;i<4;i++)ctx.lineTo(r-i*seg,by-(i%2?r*.22*(.6+.4*Math.sin(T*9+g.ph)):0));
  ctx.lineTo(-r,by);ctx.closePath();ctx.fill();
  ctx.shadowBlur=0;
  const ex=Math.cos(g.look)*r*.15,ey=Math.sin(g.look)*r*.15;
  for(const s of[-1,1]){
    ctx.fillStyle='#fff';ctx.beginPath();
    ctx.ellipse(s*r*.32+ex,-r*.25+ey,r*.2,r*.26,0,0,7);ctx.fill();
    ctx.fillStyle=frozen?'#8fb4ff':'#1b2a6b';ctx.beginPath();
    ctx.arc(s*r*.32+ex*1.8,-r*.25+ey*1.8,r*.1,0,7);ctx.fill();
  }
  ctx.restore();
}
function drawPac(){
  const b=ball,x=w2sx(b.x),y=w2sy(b.y),r=Math.max(.1,ballR*cs*b.scale);
  let ang;
  if(state==='sinking')ang=aimAngle;
  else if(phase==='aim'&&state==='play')ang=aimAngle;
  else{const s=Math.hypot(b.vx,b.vy);ang=s>0.15?Math.atan2(b.vy,b.vx):aimAngle;}
  const moving=Math.hypot(b.vx,b.vy)>0.15&&state==='play';
  let m=state==='sinking'?Math.PI*.8:moving?(0.3+0.4*Math.abs(Math.sin(b.mouth)))*Math.PI:0.16*Math.PI*(1+.3*Math.sin(T*3));
  ctx.save();ctx.translate(x,y);
  if(state==='sinking')ctx.rotate(b.spin);
  ctx.globalAlpha=(b.invuln>0&&Math.floor(T*14)%2===0)?.35:1;
  ctx.shadowColor='rgba(255,230,0,.4)';ctx.shadowBlur=r*.6;
  ctx.fillStyle='#ffe600';
  ctx.beginPath();ctx.moveTo(0,0);ctx.arc(0,0,r,m,Math.PI*2-m,false);ctx.closePath();
  ctx.rotate(ang);
  ctx.fill();
  ctx.restore();
  if(state==='sinking'&&b.scale<.6){
    ctx.fillStyle='#04070c';ctx.beginPath();ctx.arc(x,y,holeR*cs*.9,0,7);ctx.fill();
  }
}
function drawAim(){
  if(state!=='play'||phase!=='aim'||paused)return;
  const bx=w2sx(ball.x),by=w2sy(ball.y);
  // spot ring
  ctx.save();
  ctx.strokeStyle='rgba(255,255,255,.28)';ctx.lineWidth=1.5;
  ctx.setLineDash([5,6]);ctx.lineDashOffset=-T*20;
  ctx.beginPath();ctx.arc(bx,by,ballR*cs+7,0,7);ctx.stroke();
  ctx.restore();
  let ang=aimAngle,pw=charging?power:0,dragging=false;
  if(drag){
    const dx=drag.sx-drag.cx,dy=drag.sy-drag.cy,len=Math.hypot(dx,dy);
    if(len>0.15){ang=Math.atan2(dy,dx);pw=clamp(len/4.5,.06,1);dragging=true;}
  }
  if(pw<=0)return;
  // slingshot band
  if(dragging){
    ctx.save();ctx.strokeStyle='rgba(200,210,255,.35)';ctx.lineWidth=2;ctx.setLineDash([4,6]);
    ctx.beginPath();ctx.moveTo(bx,by);ctx.lineTo(w2sx(drag.cx),w2sy(drag.cy));ctx.stroke();ctx.restore();
  }
  const L=(0.9+pw*2.6)*cs,col='hsl('+(120-120*pw)+',95%,60%)';
  ctx.save();ctx.strokeStyle=col;ctx.fillStyle=col;ctx.lineWidth=3;ctx.globalAlpha=.9;
  const ex=bx+Math.cos(ang)*(ballR*cs+L),ey=by+Math.sin(ang)*(ballR*cs+L);
  ctx.beginPath();ctx.moveTo(bx+Math.cos(ang)*(ballR*cs+4),by+Math.sin(ang)*(ballR*cs+4));ctx.lineTo(ex,ey);ctx.stroke();
  ctx.beginPath();ctx.moveTo(ex+Math.cos(ang)*10,ey+Math.sin(ang)*10);
  ctx.lineTo(ex+Math.cos(ang+2.5)*9,ey+Math.sin(ang+2.5)*9);
  ctx.lineTo(ex+Math.cos(ang-2.5)*9,ey+Math.sin(ang-2.5)*9);ctx.closePath();ctx.fill();
  if(charging||dragging){
    ctx.font='700 12px Trebuchet MS,sans-serif';ctx.textAlign='center';
    ctx.fillText(Math.round(pw*100)+'%',ex+Math.cos(ang)*24,ey+Math.sin(ang)*24+4);
  }
  ctx.restore();
}
```

**Pac-Golf** — a complete mash-up where Pac-Man is your golf ball across 5 hand-built maze courses:

- **Shoot** by drag-slingshot (mouse/touch) *or* arrow keys + hold Space to charge a ping-pong power meter. Every shot counts as a stroke; beat par on each hole.
- **Pellets boost your roll** as the ball chomps them; big power dots freeze ghosts blue — hit a frozen ghost to pop it for +200.
- Classic hazards turned golf rules: sand bunkers slow you, water and ghosts cost +1 stroke (ball returns to where you shot from), and fast balls rattle over the rim of the cup instead of dropping.
- Full arcade presentation: mowed-stripe fairways, rippling water, waving numbered flags, ghost AI that hunts along the walls, Waka-style sound synthesized with Web Audio, scorecard with ACE/BIRDIE/BOGEY ratings, and a saved best round — all in one self-contained file.
