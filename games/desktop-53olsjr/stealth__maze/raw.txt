DDA implementation:
```js
function rayDist(x,y,dx,dy,maxD){
  let cx = Math.floor(x/T), cy = Math.floor(y/T);
  const stepX = dx>0?1:-1, stepY = dy>0?1:-1;
  const tDeltaX = T/Math.abs(dx||1e-9), tDeltaY = T/Math.abs(dy||1e-9);
  let tMaxX = ((stepX>0 ? (cx+1)*T - x : x - cx*T)) / Math.abs(dx||1e-9);
  let tMaxY = ((stepY>0 ? (cy+1)*T - y : y - cy*T)) / Math.abs(dy||1e-9);
  let d = 0;
  while(d < maxD){
    if(tMaxX < tMaxY){ 
      const nx=cx+stepX; // crossing into nx
      if(!inBounds(nx,cy) || wallBetween(cx,cy,nx,cy)){ return d + tMaxX; }
      cx=nx; d += tMaxX; tMaxX += tDeltaX;
    } else { similar y }
  }
  return maxD;
}
```

  while(st.length){ const [c,r]=st[st.length-1]; const opts=[];
    for(let d=0;d<4;d++){ const nc=c+DIRS[d][0], nr=r+DIRS[d][1]; if(inb(nc,nr)&&!seen[nr][nc]) opts.push(d); }
    if(!opts.length){ st.pop(); continue; }
    const d=opts[(Math.random()*opts.length)|0]; carve(c,r,d);
    seen[r+DIRS[d][1]][c+DIRS[d][0]]=true; st.push([c+DIRS[d][0], r+DIRS[d][1]]);
  }
  let extra=(MW*MH*0.05)|0;
  while(extra-->0){ const c=1+((Math.random()*(MW-2))|0), r=1+((Math.random()*(MH-2))|0), d=(Math.random()*4)|0;
    if(grid[r][c]&BIT[d]){ const nc=c+DIRS[d][0], nr=r+DIRS[d][1]; if(inb(nc,nr)) carve(c,r,d); } }
}
function bfs(sc,sr){ const d=Array.from({length:MH},()=>new Array(MW).fill(-1)); const q=[[sc,sr]]; d[sr][sc]=0; let h=0;
  while(h<q.length){ const [c,r]=q[h++]; for(let k=0;k<4;k++){ if(grid[r][c]&BIT[k])continue;
    const nc=c+DIRS[k][0], nr=r+DIRS[k][1]; if(inb(nc,nr)&&d[nr][nc]===-1){ d[nr][nc]=d[r][c]+1; q.push([nc,nr]); } } } return d; }
function bfsPath(sc,sr,tc,tr){ const s0=sc+sr*MW, t0=tc+tr*MW; if(s0===t0) return [s0];
  const prev=new Int32Array(MW*MH).fill(-1), q=new Array(MW*MH); let h=0,t=0;
  prev[s0]=s0; q[t++]=s0;
  while(h<t){ const cur=q[h++]; if(cur===t0)break; const c=cur%MW, r=(cur-c)/MW;
    for(let d=0;d<4;d++){ if(grid[r][c]&BIT[d])continue; const ni=c+DIRS[d][0]+(r+DIRS[d][1])*MW;
      if(prev[ni]===-1){ prev[ni]=cur; q[t++]=ni; } } }
  if(prev[t0]===-1) return null;
  const path=[]; let cur=t0; while(cur!==s0){ path.push(cur); cur=prev[cur]; } path.push(s0); path.reverse(); return path; }
function shuffle(a){ for(let i=a.length-1;i>0;i--){ const j=(Math.random()*(i+1))|0; const t=a[i]; a[i]=a[j]; a[j]=t; } return a; }
function pickCells(cands,n,minD){ const out=[], pool=shuffle(cands.slice());
  for(const c of pool){ if(out.every(o=>Math.abs(o.c-c.c)+Math.abs(o.r-c.r)>=minD)){ out.push(c); if(out.length===n)break; } }
  if(out.length<n) for(const c of pool){ if(!out.includes(c)){ out.push(c); if(out.length===n)break; } } return out; }

function rayDist(x0,y0,dx,dy,maxD){
  let cx=Math.floor(x0/T), cy=Math.floor(y0/T);
  const tDX=Math.abs(dx)>1e-9?T/Math.abs(dx):1e9, tDY=Math.abs(dy)>1e-9?T/Math.abs(dy):1e9;
  let tX=dx>0?((cx+1)*T-x0)/dx:(dx<0?(cx*T-x0)/dx:1e9);
  let tY=dy>0?((cy+1)*T-y0)/dy:(dy<0?(cy*T-y0)/dy:1e9);
  let d=0;
  while(d<maxD){
    if(tX<tY){ const nx=cx+(dx>0?1:-1);
      if(!inb(nx,cy)||(grid[cy][cx]&BIT[nx>cx?1:3])) return Math.min(d+tX,maxD);
      cx=nx; d+=tX; tX+=tDX; }
    else { const ny=cy+(dy>0?1:-1);
      if(!inb(cx,ny)||(grid[cy][cx]&BIT[ny>cy?2:0])) return Math.min(d+tY,maxD);
      cy=ny; d+=tY; tY+=tDY; } }
  return maxD;
}
function canSee(g){ const dx=player.x-g.x, dy=player.y-g.y, d=Math.hypot(dx,dy);
  if(d>RANGE) return 0;
  let da=Math.atan2(dy,dx)-g.face; while(da>Math.PI)da-=2*Math.PI; while(da<-Math.PI)da+=2*Math.PI;
  if(Math.abs(da)>FOV/2) return 0;
  const rd=rayDist(g.x,g.y,dx/d,dy/d,d); if(rd<d-3) return 0;
  return 1-d/RANGE; }

function collideP(){ const p=player;
  const c=Math.max(0,Math.min(MW-1,Math.floor(p.x/T))), r=Math.max(0,Math.min(MH-1,Math.floor(p.y/T)));
  const g=grid[r][c];
  if(g&1&&p.y-p.r<r*T) p.y=r*T+p.r;
  if(g&4&&p.y+p.r>(r+1)*T) p.y=(r+1)*T-p.r;
  if(g&8&&p.x-p.r<c*T) p.x=c*T+p.r;
  if(g&2&&p.x+p.r>(c+1)*T) p.x=(c+1)*T-p.r; }
function turnToward(o,ang,k){ let d=ang-o.face; while(d>Math.PI)d-=2*Math.PI; while(d<-Math.PI)d+=2*Math.PI;
  o.face+=Math.sign(d)*Math.min(k,Math.abs(d)); }
function moveToward(g,tx,ty,s){ const dx=tx-g.x, dy=ty-g.y, d=Math.hypot(dx,dy);
  if(d<=s||d<1){ g.x=tx; g.y=ty; return true; }
  g.x+=dx/d*s; g.y+=dy/d*s; return false; }
function patrolPick(g){ const c=Math.floor(g.x/T), r=Math.floor(g.y/T); const opts=[];
  for(let d=0;d<4;d++) if(!hasWall(c,r,d)) opts.push(d);
  let bd=-1;
  if(c!==g.fromC||r!==g.fromR){ if(g.fromC>c)bd=3; else if(g.fromC<c)bd=1; else if(g.fromR>r)bd=0; else bd=2; }
  const pool=opts.length>1?opts.filter(d=>d!==bd):opts;
  const d=pool[(Math.random()*pool.length)|0];
  g.tc=c+DIRS[d][0]; g.tr=r+DIRS[d][1]; g.fromC=c; g.fromR=r; }
function markExploredAt(c,r,rad){ for(let dr=-3;dr<=3;dr++)for(let dc=-3;dc<=3;dc++){
  if(dc*dc+dr*dr<=rad*rad&&inb(c+dc,r+dr)) explored[r+dr][c+dc]=true; } }

function updatePlayer(dt){
  let ax=((keys['d']||keys['ArrowRight'])?1:0)-((keys['a']||keys['ArrowLeft'])?1:0);
  let ay=((keys['s']||keys['ArrowDown'])?1:0)-((keys['w']||keys['ArrowUp'])?1:0);
  player.crouch=!!keys['Shift'];
  const sp=(player.crouch?2.6:4.2)*T;
  if(ax||ay){ const l=Math.hypot(ax,ay); ax/=l; ay/=l;
    player.x+=ax*sp*dt; collideP(); player.y+=ay*sp*dt; collideP();
    player.moving=true; turnToward(player,Math.atan2(ay,ax),15*dt);
    stepTimer-=dt;
    if(stepTimer<=0){ sfx.step(player.crouch);
      particles.push({type:'ripple',x:player.x-Math.cos(player.face)*6,y:player.y-Math.sin(player.face)*6,r0:3,r1:15,life:.4,max:.4,col:'95,232,196'});
      stepTimer=player.crouch?.5:.33; }
  } else player.moving=false;
  markExploredAt(Math.floor(player.x/T),Math.floor(player.y/T),2.4);
  for(const k of cores){ if(!k.got&&Math.hypot(k.x-player.x,k.y-player.y)<T*.55){
    k.got=true; coresGot++; sfx.pickup(); burst(k.x,k.y,'95,232,196',14);
    particles.push({type:'txt',x:k.x,y:k.y-14,vy:-28,life:1.3,max:1.3,txt:'+INTEL',col:'#5fe8c4'});
    markExploredAt(k.c,k.r,3); updatePips();
    if(coresGot===CORES){ sfx.unlock();
      particles.push({type:'txt',x:exitCell.x,y:exitCell.y-26,vy:-20,life:1.8,max:1.8,txt:'EXTRACTION OPEN',col:'#7dff9a'}); } } }
  if(coresGot===CORES&&Math.hypot(exitCell.x-player.x,exitCell.y-player.y)<T*.5) winGame();
}
function updateGuards(dt){
  for(const g of guards){ let tx,ty;
    if(g.chasing&&mode==='playing'){ const sp=CHASE*T;
      const pcx=Math.floor(player.x/T), pcy=Math.floor(player.y/T);
      const gcx=Math.floor(g.x/T), gcy=Math.floor(g.y/T);
      g.reT-=dt;
      if(!g.path||g.reT<=0||(gcx===pcx&&gcy===pcy)){ const p=bfsPath(gcx,gcy,pcx,pcy);
        g.path=p||[gcx+gcy*MW]; g.pi=0; g.reT=.65+(g.seed%1)*.3; }
      if(g.path.length>1&&g.pi<g.path.length-1){ const n=g.path[g.pi+1]; tx=n%MW*T+T/2; ty=((n/MW)|0)*T+T/2;
        turnToward(g,Math.atan2(ty-g.y,tx-g.x),9*dt); if(moveToward(g,tx,ty,sp*dt)) g.pi++; }
      else { tx=player.x; ty=player.y; turnToward(g,Math.atan2(ty-g.y,tx-g.x),9*dt); moveToward(g,tx,ty,sp*dt); }
    } else { const sp=PATROL*T; tx=g.tc*T+T/2; ty=g.tr*T+T/2;
      turnToward(g,Math.atan2(ty-g.y,tx-g.x),5.5*dt); if(moveToward(g,tx,ty,sp*dt)) patrolPick(g); } }
}
function detect(){
  let vis=0; for(const g of guards){ const v=canSee(g); g.sees=v>0; if(v>vis)vis=v; }
  if(vis>0){
    if(det===0&&!alarm){ sfx.ping(); particles.push({type:'ring',x:player.x,y:player.y,r0:6,r1:34,life:.5,max:.5,col:'95,232,196'}); }
    const inLight=guards.some(g=>Math.hypot(g.x-player.x,g.y-player.y)<2.5*T);
    const f=(player.crouch?.45:1)*(inLight?1.5:1)*(player.moving?1.12:1);
    det=Math.min(1,det+dt*vis*f/1.05);
    if(!alarm&&det>=1) startAlarm();
  } else det=Math.max(0,det-dt*.85);
  if(alarm){ if(vis>0) lastSeen=time; else if(time-lastSeen>2.6) endAlarm();
    for(const g of guards){ if(g.chasing&&Math.hypot(g.x-player.x,g.y-player.y)<T*.55){ caughtPlayer(); break; } } }
  if(!alarm&&det>.5){ heartT-=dt; if(heartT<=0){ sfx.thump(); heartT=.35+(1-det)*.4; } } else heartT=0;
}

function punch(c,x,y,r,a){ const g=c.createRadialGradient(x,y,0,x,y,r);

function glow(c,x,y,r,rgb,a){ const g=c.createRadialGradient(x,y,0,x,y,r);

function buildLight(){ const c=lctx; c.setTransform(dpr,0,0,dpr,0,0); c.clearRect(0,0,W,H);
  const dark=mode==='menu'?.84:(mode==='playing'?(alarm?.92:.93):.88);

  if(mode!=='menu'){ const f=player.face, cr=player.crouch?.78:1;

function drawPlayer(){ const p=player, r=p.crouch?T*.26:T*.3;

function render(){
  cam.z=Math.max(.7,Math.min(1.5,Math.min(W/(T*12.5),H/(T*8.5)))); clampCam();
  ctx.setTransform(dpr,0,0,dpr,0,0);
  ctx.fillStyle='#04100b'; ctx.fillRect(0,0,W,H);
  const gs=T*cam.z; let ox=((W/2-cam.x*cam.z)%gs+gs)%gs, oy=((H/2-cam.y*cam.z)%gs+gs)%gs;
  ctx.strokeStyle='rgba(95,232,196,.05)'; ctx.lineWidth=1; ctx.beginPath();
  for(let x=ox;x<W;x+=gs){ ctx.moveTo(x,0); ctx.lineTo(x,H); }

  for(const g of guards) drawGuard(g);
  if(mode!=='menu') drawPlayer();
  else { const pr=10+6*Math.sin(animT*2.5);

    ctx.beginPath(); ctx.moveTo(player.x-5,player.y); ctx.lineTo(player.x+5,player.y);
    ctx.moveTo(player.x,player.y-5); ctx.lineTo(player.x,player.y+5); ctx.stroke();
    ctx.font='9px Consolas,monospace'; ctx.textAlign='center'; ctx.fillStyle='rgba(95,232,196,.7)';

  ctx.globalCompositeOperation='source-over';
  for(const p of particles){ const a=p.life/p.max;
    if(p.type==='spark'){ ctx.fillStyle='rgba('+p.col+','+a+')'; ctx.fillRect(p.x-p.size/2,p.y-p.size/2,p.size,p.size); }
    else if(p.type==='ring'){ const t=1-a; ctx.beginPath(); ctx.arc(p.x,p.y,p.r0+(p.r1-p.r0)*t,0,7);
      ctx.strokeStyle='rgba('+p.col+','+a+')'; ctx.lineWidth=2.5; ctx.stroke(); }
    else if(p.type==='txt'){ ctx.font='700 14px Consolas,monospace'; ctx.textAlign='center';
      ctx.globalAlpha=a; ctx.lineWidth=3; ctx.strokeStyle='rgba(0,0,0,.6)';

      const tx=dx>0?(W-m)/dx:(dx<0?m/dx:1e9), ty=dy>0?(H-m)/dy:(dy<0?m/dy:1e9), t=Math.min(tx,ty,1e9);
      const qx=W/2+dx*t, qy=H/2+dy*t;
      ctx.save(); ctx.translate(qx,qy); ctx.rotate(ang);
      ctx.fillStyle='#7dff9a'; ctx.beginPath(); ctx.moveTo(10,0); ctx.lineTo(-6,-8); ctx.lineTo(-6,8); ctx.closePath(); ctx.fill();
      ctx.rotate(-ang); ctx.font='9px Consolas,monospace'; ctx.textAlign='center'; ctx.fillText('EXTRACT',0,24); ctx.restore(); } }
  if(alarm){ ctx.fillStyle='rgba(255,40,30,'+(.07+.05*Math.sin(animT*9))+')'; ctx.fillRect(0,0,W,H); }
  if(flash>0){ ctx.fillStyle='rgba(255,120,110,'+(flash*.5)+')'; ctx.fillRect(0,0,W,H); }
}

  for(let r=0;r<MH;r++)for(let c=0;c<MW;c++){ if(!explored[r][c]) continue;
    mctx.fillStyle=((r+c)&1)?'#12332a':'#0f2b23'; mctx.fillRect(c*s,r*s,s,s); }
  mctx.strokeStyle='rgba(5,18,14,.95)'; mctx.lineWidth=1.6; mctx.beginPath();
  for(let r=0;r<MH;r++)for(let c=0;c<MW;c++){ if(!explored[r][c]) continue; const g=grid[r][c];
    if(g&1){ mctx.moveTo(c*s,r*s); mctx.lineTo((c+1)*s,r*s); }
    if(g&4){ mctx.moveTo(c*s,(r+1)*s); mctx.lineTo((c+1)*s,(r+1)*s); }
    if(g&8){ mctx.moveTo(c*s,r*s); mctx.lineTo(c*s,(r+1)*s); }
    if(g&2){ mctx.moveTo((c+1)*s,r*s); mctx.lineTo((c+1)*s,(r+1)*s); } }
  mctx.stroke();
  for(const k of cores){ mctx.fillStyle=k.got?'rgba(127,168,154,.4)':'#5fe8c4';
    mctx.fillRect(k.c*s+s/2-2,k.r*s+s/2-2,4,4); }
  const open=coresGot===CORES;
  mctx.fillStyle=open?(animT%1<.5?'#7dff9a':'rgba(125,255,154,.35)'):'#ff5347';
  mctx.fillRect((MW-1)*s+1,(MH-1)*s+1,s-2,s-2);
  if(alarm) for(const g of guards){ mctx.fillStyle='#ff5347';
    mctx.beginPath(); mctx.arc(g.x/T*s,g.y/T*s,2.6,0,7); mctx.fill(); }
  if(mode!=='menu'){ const pr=2.4+1.2*Math.sin(animT*6);

function hud(dt){ hudT-=dt; if(hudT>0) return; hudT=.12;

function frame(now){ requestAnimationFrame(frame);
  let dt=(now-last)/1000; last=now; if(dt>.06) dt=.06; animT+=dt;
  if(mode==='playing'){ time+=dt; updatePlayer(dt); updateGuards(dt); detect(); }
  else if(mode==='menu'){ updateGuards(dt); menuCamT+=dt;
    const g=guards[(Math.floor(menuCamT/6)%guards.length)]||guards[0];
    if(g){ cam.x+=(g.x-cam.x)*(1-Math.exp(-1.2*dt)); cam.y+=(g.y-cam.y)*(1-Math.exp(-1.2*dt)); } }
  else if(mode==='caught'||mode==='won'){ updateGuards(dt); }
  shake*=Math.exp(-5*dt); flash=Math.max(0,flash-dt*1.8);
  updateParticles(dt); updateMotes(dt);
  render(); hud(dt);
}
