function circRect(px,py,r,rx,ry,rw,rh){ const nx=Math.max(rx,Math.min(px,rx+rw)),ny=Math.max(ry,Math.min(py,ry+rh));
  const dx=px-nx,dy=py-ny;return dx*dx+dy*dy<r*r; }
function nearRects(p,rad,fn){
  const x0=Math.max(0,Math.floor((p.x-rad-8)/P)-1),x1=Math.min(cols,Math.floor((p.x+rad+8)/P)+1);
  const y0=Math.max(0,Math.floor((p.y-rad-8)/P)-1),y1=Math.min(rows,Math.floor((p.y+rad+8)/P)+1);
  for(let cj=y0;cj<=y1;cj++)for(let cxi=x0;cxi<=x1;cxi++)
    if(vW[cj*(cols+1)+cxi]&&fn(cxi*P-4,cj*P-4,8,P+8))return true;
  for(let rwi=y0;rwi<=y1;rwi++)for(let ci=x0;ci<=x1;ci++)
    if(hW[rwi*cols+ci]&&fn(ci*P-4,rwi*P-4,P+8,8))return true;
  return false;
}
function moveAxis(p,axis,d){
  if(!d)return false;
  const prev=p[axis];p[axis]+=d;const r=p.r;
  let hit=nearRects(p,r,(rx,ry,rw,rh)=>circRect(p.x,p.y,r,rx,ry,rw,rh));
  if(hit){p[axis]=prev;if(axis==='x')p.vx=0;else p.vy=0;return true;}
  return false;
}
function clearPos(p){ let bad=false;

function stepPlayer(p,ix,iy,dt){
  const sp=p.boost>0?SPEED_B:SPEED_N;
  if(ix||iy){const l=Math.hypot(ix,iy);p.vx+=ix/l*ACC*dt;p.vy+=iy/l*ACC*dt;}
  else{const f=Math.exp(-9.5*dt);p.vx*=f;p.vy*=f;}
  const v=Math.hypot(p.vx,p.vy);if(v>sp){p.vx*=sp/v;p.vy*=sp/v;}
  const n=Math.max(1,Math.ceil(v*dt/(P*.4)));let hit=false;
  for(let i=0;i<n;i++){ if(moveAxis(p,'x',p.vx*dt/n))hit=true; if(moveAxis(p,'y',p.vy*dt/n))hit=true; }
  if(hit){sWall();const a=Math.random()*7;
    parts.push({x:p.x+Math.cos(a)*p.r,y:p.y+Math.sin(a)*p.r,vx:Math.cos(a)*30,vy:Math.sin(a)*30-20,

}

  if(cx<cols-1&&!vW[cy*(cols+1)+cx+1])opts.push([cx+1,cy]);
  if(cy>0&&!hW[cy*cols+cx])opts.push([cx,cy-1]);
  if(cy<rows-1&&!hW[(cy+1)*cols+cx])opts.push([cx,cy+1]);
  const o=opts.length?opts[(Math.random()*opts.length)|0]:[cx,cy];
  b.tx=o[0]*P+P/2+4;b.ty=o[1]*P+P/2+4;b.timer=.5+Math.random()*1.3;
}
function updateBots(dt){
  for(const b of bots){
    const dx=b.tx-b.x,dy=b.ty-b.y,d=Math.hypot(dx,dy);
    b.timer-=dt;
    if(d<6||b.timer<=0)botPick(b);
    else{const sp=95;b.x+=dx/d*sp*dt;b.y+=dy/d*sp*dt;}
  }
}

/* ================= UPDATE ================= */
function update(dt){
  for(let i=parts.length-1;i>=0;i--){const q=parts[i];q.life-=dt;
    if(q.life<=0){parts.splice(i,1);continue;}
    q.x+=q.vx*dt;q.y+=q.vy*dt;q.vy+=(q.g||0)*dt;q.vx*=Math.exp(-2*dt);if(q.vr)q.rot+=q.vr*dt;}
  for(let i=rings.length-1;i>=0;i--){rings[i].t+=dt;if(rings[i].t>rings[i].dur)rings.splice(i,1);}
  shake*=Math.exp(-7*dt);if(shake<.15)shake=0;
  goT=Math.max(0,goT-dt);clashCd=Math.max(0,clashCd-dt);
  for(const s of spores){s.ph+=dt;s.y-=s.s*dt;if(s.y<-4){s.y=mh+4;s.x=Math.random()*mw;}}

  if(state==='menu'){
    updateBots(dt);
    menuPingT-=dt;
    if(menuPingT<=0&&gems.length){menuPingT=1.5+Math.random();const g=gems[(Math.random()*gems.length)|0];
      rings.push({x:g.x,y:g.y,t:0,dur:.8,rgb:GEMRGB});}
    return;
  }
  if(state==='count'){
    const prev=Math.ceil(countT);countT-=dt;const cur=Math.ceil(countT);
    if(cur!==prev&&cur>=1)sCount();
    if(countT<=0){state='play';goT=.6;sGo();}
    return;
  }
  if(state==='pause')return;
  if(state==='over'){
    winDelay-=dt;if(winDelay<=0&&!overlayShown){overlayShown=true;showWin();}
    confT-=dt;const wc=winner===0?P1C:P2C;
    while(confT<=0){confT+=.13;
      for(let i=0;i<3;i++)parts.push({x:Math.random()*mw,y:-8,vx:(Math.random()-.5)*70,vy:60+Math.random()*90,g:150,
        life:2.4,max:2.4,size:3+Math.random()*3.5,rgb:[wc.rgb,GEMRGB,'255,255,255'][(Math.random()*3)|0],rot:Math.random()*7,vr:(Math.random()-.5)*9,glow:false});}
    return;
  }
  /* ---- play ---- */
  matchTime+=dt;
  const i1x=(keys.KeyD?1:0)-(keys.KeyA?1:0),i1y=(keys.KeyS?1:0)-(keys.KeyW?1:0);
  const i2x=(keys.ArrowRight?1:0)-(keys.ArrowLeft?1:0),i2y=(keys.ArrowDown?1:0)-(keys.ArrowUp?1:0);
  stepPlayer(p1,i1x,i1y,dt);stepPlayer(p2,i2x,i2y,dt);
  for(const p of [p1,p2]){
    if(p.boost>0){p.boost-=dt;
      parts.push({x:p.x+(Math.random()*8-4),y:p.y+(Math.random()*8-4),vx:-p.vx*.12,vy:-p.vy*.12,
        life:.3,max:.3,size:2.1,rgb:p.rgb,glow:true});}
  }
  handleClash();
  let gt=0;
  while(gems.length<GEMN&&gt<8){if(spawnGem([p1,p2]))break;gt++;}
  for(let i=gems.length-1;i>=0;i--){const gm=gems[i];gm.t=Math.min(1,gm.t+dt/.45);
    let taken=false;
    for(let pi=0;pi<2&&!taken;pi++){const p=pi===0?p1:p2;
      if((p.x-gm.x)**2+(p.y-gm.y)**2<(p.r+9)**2){gems.splice(i,1);onPickup(p,pi,gm.x,gm.y,false);taken=true;}}
    if(state!=='play')return;
  }
  surge.t=Math.min(1,surge.t+dt/.45);
  if(surge.state==='idle'){surge.timer-=dt;if(surge.timer<=0)spawnSurge();}
  else{
    surge.timer-=dt;
    if(Math.random()<.3)parts.push({x:surge.x+(Math.random()*16-8),y:surge.y+(Math.random()*16-8),
      vx:0,vy:-25,life:.4,max:.4,size:1.6,rgb:GEMRGB,glow:true});
    for(let pi=0;pi<2;pi++){const p=pi===0?p1:p2;
      if((p.x-surge.x)**2+(p.y-surge.y)**2<(p.r+13)**2){onPickup(p,pi,surge.x,surge.y,true);break;}}
    if(state!=='play')return;
    if(surge.timer<=0&&surge.state==='live'){surge.state='idle';surge.timer=SURGE_IDLE;}
  }
}
