function stepSnake(dt,isAI){
  const s=SNAKE,h=s.segs[0];
  let turn=0,boost=false,brake=false;
  if(!isAI){
    if(keys.has('arrowleft')||keys.has('a'))turn-=1;
    if(keys.has('arrowright')||keys.has('d'))turn+=1;
    boost=(keys.has('arrowup')||keys.has('w')||keys.has('shift'));
    brake=(keys.has('arrowdown')||keys.has('s'));
  }else{
    s.aiT+=dt;
    turn=Math.sin(s.aiT*0.85)*0.65+Math.sin(s.aiT*2.17+1.3)*0.45;
    const m=120;let ax=0,ay=0;
    if(h.x<WALL+m)ax+=1;if(h.x>W-WALL-m)ax-=1;
    if(h.y<WALL+m)ay+=1;if(h.y>H-WALL-m)ay-=1;
    if(ax||ay){const ta=Math.atan2(ay,ax);turn=clamp(angNorm(ta-s.heading)*2.6,-1.8,1.8);}
    const kL=keys.has('arrowleft')||keys.has('a'),kR=keys.has('arrowright')||keys.has('d');
    if(kL||kR)turn=kR-kL;
    boost=keys.has('shift')||keys.has('arrowup')||keys.has('w');
  }
  s.boosting=boost;
  if(boost!==s.boostPrev){if(boost)sBoost();}
  s.boostPrev=boost;
  s.turnCur+=(turn-s.turnCur)*Math.min(1,9*dt);
  s.heading=angNorm(s.heading+s.turnCur*(boost?2.4:3.0)*dt);

  const base=isAI?265:Math.min(240+s.eaten*6,430);
  const target=boost?Math.min(base*1.7,640):brake?base*0.5:base;
  const k=boost?4.5:2.6,e=1-Math.exp(-k*dt);
  s.vx+=(Math.cos(s.heading)*target-s.vx)*e;
  s.vy+=(Math.sin(s.heading)*target-s.vy)*e;
  h.x+=s.vx*dt;h.y+=s.vy*dt;

  const r=s.headR;let hit=0;
  if(h.x<WALL+r){h.x=WALL+r;s.vx=Math.abs(s.vx)*0.82;hit=Math.abs(s.vx);}
  else if(h.x>W-WALL-r){h.x=W-WALL-r;s.vx=-Math.abs(s.vx)*0.82;hit=Math.abs(s.vx);}
  if(h.y<WALL+r){h.y=WALL+r;s.vy=Math.abs(s.vy)*0.82;hit=Math.max(hit,Math.abs(s.vy));}
  else if(h.y>H-WALL-r){h.y=H-WALL-r;s.vy=-Math.abs(s.vy)*0.82;hit=Math.max(hit,Math.abs(s.vy));}
  if(hit>140&&state==='play'){sBounce(hit);addShake(clamp(hit/60,3,9));burst(h.x,h.y,rgbStr(segRGB(0)),8,160,.5,2.5);}

  for(let it=0;it<2;it++){
    for(let i=1;i<s.segs.length;i++){
      const a=s.segs[i-1],b=s.segs[i];
      let dx=b.x-a.x,dy=b.y-a.y;const d=Math.hypot(dx,dy);
      if(d>0.0001){const f=s.L/d;b.x=a.x+dx*f;b.y=a.y+dy*f;}
    }
  }
  for(let i=1;i<s.segs.length;i++){
    const p=s.segs[i],rr=bodyRadius(i)+2;
    if(p.x<WALL+rr)p.x=WALL+rr;if(p.x>W-WALL-rr)p.x=W-WALL-rr;
    if(p.y<WALL+rr)p.y=WALL+rr;if(p.y>H-WALL-rr)p.y=H-WALL-rr;
  }

  for(let i=food.length-1;i>=0;i--){
    const o=food[i];
    if(Math.hypot(h.x-o.x,h.y-o.y)<r+o.r+2){food.splice(i,1);eatOrb(o,isAI);}
  }

  s.tongueWait-=dt;
  if(s.tongue<1){s.tongue=Math.min(1,s.tongue+dt*5.5);if(s.tongue>=1)s.tongueWait=rand(0.8,2.6);}
  else if(s.tongueWait<=0)s.tongue=0;

  if(!isAI){
    for(let i=4;i<s.segs.length;i++){
      const p=s.segs[i];
      if(Math.hypot(h.x-p.x,h.y-p.y)<(r+bodyRadius(i))*0.72){die();return;}
    }
  }
}

**Slithermass** is a physics-driven snake: the head carries real momentum (acceleration, drag, and drift into tight turns), while the body follows as a chained distance-constraint coil. Steer with **←→/AD**, hold **↑/W/Shift** to boost, **↓/S** to brake — walls *bounce* with impact thuds and dust, but biting your own tail ends the run. Glowing larvae drift and pulse; chaining eats within 6 seconds builds a score combo. The menu runs a live AI demo snake you can hijack with the arrow keys, best score persists via localStorage, and **R** restarts instantly.
