function tone(f,dur,type,vol,slide,delay){ if(!AC||muted)return; type=type||'square'; vol=vol||0.4; delay=delay||0;
  var t0=AC.currentTime+delay, o=AC.createOscillator(), g=AC.createGain();
  o.type=type; o.frequency.setValueAtTime(f,t0);
  if(slide) o.frequency.exponentialRampToValueAtTime(Math.max(30,slide),t0+dur);
  g.gain.setValueAtTime(vol,t0); g.gain.exponentialRampToValueAtTime(0.001,t0+dur);
  o.connect(g); g.connect(master); o.start(t0); o.stop(t0+dur+0.03); }
function noise(dur,vol,freq,type){ if(!AC||muted)return; type=type||'lowpass';
  var t0=AC.currentTime, n=AC.createBufferSource(), buf=AC.createBuffer(1,(AC.sampleRate*dur)|0,AC.sampleRate), d=buf.getChannelData(0);
  for(var i=0;i<d.length;i++)d[i]=Math.random()*2-1; n.buffer=buf;
  var f=AC.createBiquadFilter(); f.type=type; f.frequency.value=freq;
  var g=AC.createGain(); g.gain.setValueAtTime(vol,t0); g.gain.exponentialRampToValueAtTime(0.001,t0+dur);
  n.connect(f); f.connect(g); g.connect(master); n.start(t0); }

function spawnLanes(){
  lanes=[]; laneByRow={};
  for(var r=0;r<ROWS;r++){
    var d=ROW_DEF[r];
    if(d.t==='water'){
      var items=[], n=d.lens.length, span=(COLS+2)/(n);
      for(var i=0;i<n;i++){var len=d.lens[i%2];
        items.push({x:i*span+(Math.random()*1.6-0.4),len:len,v:d.dir*d.sp*mult});}
      var lane={row:r,items:items}; lanes.push(lane); laneByRow[r]=lane;
    } else if(d.t==='road'){
      var it2=[], n2=d.n, sp2=COLS/n2;
      for(var j=0;j<n2;j++){it2.push({x:j*sp2+(Math.random()*1.4-0.5),w:d.w,v:d.dir*d.sp*mult});}
      var lane2={row:r,items:it2}; lanes.push(lane2); laneByRow[r]=lane2;
    }
  }
}
function genMines(){
  mines=new Set(); revealed=new Set(); visited=new Set(); flags=new Set();
  var safe=new Set(), c=3+((Math.random()*4)|0);
  for(var r=10;r>=0;r--){
    for(var dc=-1;dc<=1;dc++){var nc=c+dc;if(nc>=0&&nc<COLS)safe.add(key(nc,r));}
    if(r>0&&Math.random()<0.45)c+=(Math.random()<0.5?-1:1);
    c=clamp(c,1,COLS-2);
  }
  var cand=[];
  for(var rr2=1;rr2<=9;rr2++)for(var cc=0;cc<COLS;cc++){if(!safe.has(key(cc,rr2)))cand.push(key(cc,rr2));}
  for(var i=cand.length-1;i>0;i--){var j=(Math.random()*(i+1))|0,t=cand[i];cand[i]=cand[j];cand[j]=t;}
  var target=Math.min(6+level*3,24);
  for(var k=0;k<target&&k<cand.length;k++)mines.add(cand[k]);
  recount();
}
function recount(){
  counts=new Int8Array(COLS*ROWS);
  mines.forEach(function(k){ if(revealed.has(k))return; var p=k.split(','),c=+p[0],r=+p[1];
    for(var dc=-1;dc<=1;dc++)for(var dr=-1;dr<=1;dr++){ if(!dc&&!dr)continue;
      var nc=c+dc,nr=r+dr; if(nc<0||nc>=COLS||nr<0||nr>=ROWS)continue;
      if(mines.has(key(nc,nr)))continue; counts[nr*COLS+nc]++; } });
}

function jump(dx,dy){
  if(state!=='playing'||!frog.alive||frog.jumping)return;
  var nc=clamp(frog.col+dx,0,COLS-1), nr=clamp(frog.row+dy,0,ROWS-1);
  if(nc===frog.col&&nr===frog.row)return;
  frog.sx=frog.x; frog.srow=frog.row; frog.tcol=nc; frog.trow=nr; frog.jt=0; frog.jumping=true;
  frog.dx=dx; frog.dy=dy; hops++; sfx.jump();
}
function land(){
  frog.jumping=false; frog.col=frog.tcol; frog.row=frog.trow; frog.x=frog.col+0.5; frog.squash=1;
  var k=key(frog.col,frog.row); visited.add(k);
  if(mines.has(k)&&!revealed.has(k)){die('boom');return;}
  if(frog.row===0&&PADS.indexOf(frog.col)>=0){doWin();return;}
  if(frog.row<maxRow){var g=maxRow-frog.row; addScore(30*g); popup('+'+(30*g)+' ROW'); maxRow=frog.row;}
}
function die(cause){
  if(state!=='playing')return;
  state='dying'; deathT=1.05; frog.alive=false; lastCause=cause;
  var gx=frog.x, gy=(frog.jumping?frog.trow:frog.row)+0.5;
  if(cause==='boom'){ revealed.add(key(frog.col,frog.row)); recount(); shake=1; sfx.boom(); minesExploded++;
    burst(gx,gy,30,['#ffb340','#ff6b3a','#8d8d8d','#ffd75e'],cell*2.4,cell*2.2,cell*0.1);
    ring(gx,gy,cell*2.4,0.5,'#ffb340',4); popup('BOOM!','#ff5540',gx,gy);
  } else if(cause==='drown'){ sfx.splash(); burst(gx,gy,18,['#7fd4ff','#bdeaff','#3f9ecf'],cell*2.2,cell*3,cell*0.07);
    ring(gx,gy,cell*1.6,0.55,'#7fd4ff',3); popup('SPLASH!','#7fd4ff',gx,gy);
  } else if(cause==='hit'){ sfx.hit(); burst(gx,gy,20,['#c9503e','#8d8d8d','#58c93a'],cell*2.4,cell*2,cell*0.09);
    popup('CRUNCH!','#ff8f6b',gx,gy);
  } else if(cause==='time'){ tone(220,0.3,'sawtooth',0.3,110); popup('TIME UP!','#ffb340',COLS/2,1); }
}

/* ============================== UPDATE ============================== */
function update(dt){
  tGlobal+=dt;
  fireflies.forEach(function(f){f.x+=f.vx*dt;f.y+=Math.sin(tGlobal*f.w+f.p)*8*dt;if(f.x>W+20)f.x=-20;if(f.x<-20)f.x=W+20;});
  particles=particles.filter(function(p){p.t+=dt;
    if(p.type==='p'){p.vy+=p.g*dt;p.x+=p.vx*dt;p.y+=p.vy*dt;}return p.t<p.dur;});
  popups=popups.filter(function(p){p.t+=dt;p.y-=cell*0.6*dt;return p.t<1.1;});
  shake=Math.max(0,shake-dt*2.4);

  if(state==='playing'){
    time-=dt;
    var sec=Math.ceil(time);
    if(time<=5.5&&sec!==lastTick){lastTick=sec;sfx.tick();}
    if(time<=0){time=0;die('time');}
    updateFrog(dt); bugUpdate(dt); sonarT=Math.max(0,sonarT-dt);
    frog.squash=Math.max(0,frog.squash-dt*6);
  } else if(state==='dying'){
    deathT-=dt;
    if(deathT<=0){lives--;uiLives(); if(lives<=0)gameOver(); else respawn();}
  } else if(state==='won'){
    winT-=dt; frog.squash=Math.max(0,frog.squash-dt*6);
    if(winT<=0&&!ovShown){ovShown=true;showOv($('ovWin'));}
  }
  if(frog){frog.blinkT-=dt;if(frog.blinkT<-0.12)frog.blinkT=2+Math.random()*3;}
  if(state==='playing'||state==='dying'||state==='won'){
    lanes.forEach(function(l){l.items.forEach(function(it){
      it.x+=it.v*dt;
      var len=(laneByRow[l.row]===l&&ROW_DEF[l.row].t==='water')?it.len:it.w;
      if(it.v>0&&it.x>COLS)it.x=-len-Math.random()*1.5;
      if(it.v<0&&it.x<-len)it.x=COLS+Math.random()*1.5;
    });});
  }
}
function updateFrog(dt){
  var f=frog; if(!f.alive)return;
  if(f.jumping){ f.jt+=dt/JUMP_MS; if(f.jt>=1)land(); return; }
  var row=f.row, def=ROW_DEF[row];
  if(def.t==='water'){
    var lane=laneByRow[row], under=null;
    for(var i=0;i<lane.items.length;i++){var it=lane.items[i];
      if(f.x>=it.x+0.12&&f.x<=it.x+it.len-0.12&&f.x>=0.35&&f.x<=COLS-0.35){under=it;break;}}
    if(!under){die('drown');return;}
    f.x+=under.v*dt;
  } else if(def.t==='road'){
    var l2=laneByRow[row];
    for(var j=0;j<l2.items.length;j++){var c=l2.items[j], cen=c.x+c.w/2;
      if(Math.abs(f.x-cen)<c.w/2+0.42){die('hit');return;}}
  }
}
function bugUpdate(dt){
  if(!bug.active){bug.t-=dt;
    if(bug.t<=0){var lane=laneByRow[3];bug.log=lane.items[(Math.random()*lane.items.length)|0];bug.active=true;}
  } else {
    bug.x=bug.log.x+bug.log.len/2;
    if(frog.alive&&frog.row===3&&!frog.jumping&&Math.abs(frog.x-bug.x)<0.65){
      bug.active=false; bug.t=14+Math.random()*10; sfx.bug();
      burst(bug.x,3.5,12,['#ffd75e','#fff'],cell*1.8,cell,cell*0.06);
      if(charges<3){charges++;popup('SONAR +1','#ffd75e',frog.x,3.5);}
      else{addScore(75);popup('+75','#ffd75e',frog.x,3.5);}
      uiPips();
    }
  }
}

**Frog Sweeper** fuses both classics: every tile shows how many hidden mines touch it (Minesweeper deduction), while you hop across drifting logs and oncoming traffic in classic Frogger style. Blow a mine up and its neighbors' hints drop by one; flag correct reads for +10, grab the firefly bug for sonar charges (E) that pulse nearby mines, then reach a glowing pad before the clock runs out. Levels ramp up speed, mine density, and shrink the timer — press R anytime to restart.
