/* ============================== SPAWNING ============================== */
const dNow=()=>Math.min(1, 0.32*(race-1) + (dist/RACE_M)*0.75);
function waveGapM(){ return lerp(9,4.6,dNow())+Math.random()*2.2; }
function mkObs(type,l,y,dd){
  const o={type,lane:l,laneF:l,y};
  if(type==='beetle'){ o.w=LANEW*0.46; o.ph=Math.random()*6.28; o.amp=0.22+dd*0.18; }
  else if(type==='tire'){ o.w=LANEW*0.74; o.rot=Math.random()*6.28; }
  else if(type==='log'){ const lf=l<4?l+1:l-1; o.laneF=lf; o.w=LANEW*2-14; }
  else if(type==='pud'){ o.w=LANEW*1.5; o.ph=Math.random()*6.28; }
  else { o.w=30; o.ph=Math.random()*6.28; } // fly / flyAir
  return o;
}
function spawnWave(y){
  const dd=dNow();
  const order=[0,1,2,3,4].sort(()=>Math.random()-0.5);
  const occ=new Set(); let added=0;
  const want=1 + (Math.random()>0.78-dd*0.25?1:0) + (dd>0.5&&Math.random()>0.88?1:0);
  for(const l of order){
    if(added>=want) break; if(occ.has(l)) continue;
    const r=Math.random(); let type = r<0.20+0.2*dd ? 'log' : (r<0.62?'tire':'beetle');
    if(type==='log'){ const l2=l<4?l+1:l-1; if(occ.has(l2)) type='tire'; else occ.add(l2); }
    occ.add(l); obstacles.push(mkObs(type,l,y,dd)); added++;
  }
  if(Math.random()<0.65){
    const free=[0,1,2,3,4].filter(l=>!occ.has(l));
    if(free.length){ const l=free[Math.floor(Math.random()*free.length)];
      obstacles.push(mkObs(Math.random()<0.42?'flyAir':'fly', l, y+(Math.random()*50-25), dd)); }
  }
  if(Math.random()<0.38){
    const free=[0,1,2,3,4].filter(l=>!occ.has(l));
    if(free.length){ const l=free[Math.floor(Math.random()*free.length)];
      obstacles.push(mkObs('pud',l,y+70,dd)); }
  }
}

/* ============================== PARTICLES / FLOATERS ============================== */
function addPart(p){ if(parts.length<320) parts.push(p); }
function burst(x,y,n,kind,col){
  for(let i=0;i<n;i++){ const a=Math.random()*6.28, sp=40+Math.random()*160;
    addPart({t:kind,x,y,vx:Math.cos(a)*sp,vy:Math.sin(a)*sp-(kind==='drop'?-40:0),life:0.5+Math.random()*0.4,max:0.9,col,r:2+Math.random()*3,rot:Math.random()*6.28,vr:(Math.random()-0.5)*10}); }
}
function confetti(){
  const cols=['#ffd83d','#7ee04a','#ff8a3d','#2fd4c5','#ffffff'];
  for(let i=0;i<70;i++) addPart({t:'conf',x:roadX+Math.random()*roadW,y:-20-Math.random()*120,vx:(Math.random()-0.5)*60,vy:60+Math.random()*120,life:2.4,max:2.4,col:cols[i%5],r:3,rot:Math.random()*6.28,vr:(Math.random()-0.5)*12});
}
function floater(x,y,txt){ floaters.push({x,y,txt,life:0.9}); }

/* ============================== HIT / CATCH ============================== */
function hit(o){
  lives--; P.invuln=2; P.slow=1.5; shake=13; flash=0.5; sfx.hit();
  burst(P.x,playerY,10,'leaf','#4fae3a'); burst(P.x,playerY,6,'dot','#cfc79b');
  if(o.type==='beetle'){ o.dead=true; burst(o.x||laneX(o.laneF),o.y,8,'dot','#3b2a1e'); }
  floater(P.x,playerY-40,'OUCH!');
  if(lives<=0){ finishInfo={dnf:true}; result={score:flyCount*25+Math.floor(dist/10),dnf:true,nb:false}; mode='over'; overTimer=1.0; }
}
function catchFly(o){
  o.dead=true; flyCount++; P.boost=2.1; sfx.fly();
  burst(o.x||laneX(o.laneF),o.y,8,'spark','#ffd83d'); floater(P.x,playerY-46,'+25');
}

/* ============================== UPDATE ============================== */
function update(dt){
  goFlash=Math.max(0,goFlash-dt);
  shake*=Math.pow(0.001,dt); if(shake<0.3) shake=0;
  flash=Math.max(0,flash-dt*1.6);
  for(let i=parts.length-1;i>=0;i--){ const p=parts[i]; p.life-=dt;
    if(p.life<=0){ parts.splice(i,1); continue; }
    p.x+=p.vx*dt; p.y+=p.vy*dt; p.rot+=(p.vr||0)*dt;
    if(p.t==='drop') p.vy+=500*dt; if(p.t==='conf'){ p.vy+=80*dt; p.vx*=0.995; }
    if(p.t==='leaf'||p.t==='dot') p.vy+=240*dt;
  }
  for(let i=floaters.length-1;i>=0;i--){ const f=floaters[i]; f.life-=dt; f.y-=34*dt; if(f.life<=0) floaters.splice(i,1); }

  if(mode==='menu'){ t+=dt; dist+=3.5*dt; return; }

  if(mode==='count'){
    cd-=dt;
    const n=Math.ceil(cd);
    if(n<lastCdInt){ lastCdInt=n; if(n>=1&&n<=3) sfx.count(); }
    if(cd<=0){ mode='play'; goFlash=0.85; sfx.go(); }
    return;
  }

  if(mode==='pause') return;

  t+=dt;
  if(mode==='over'){ v*=Math.pow(0.04,dt); dist+=v*PXPM*dt/PXPM; }
  else {
    const V0=Math.min(14,10+(race-1)*0.7), V1=Math.min(23,15.5+(race-1)*0.9);
    let base=V0+(V1-V0)*smooth(Math.min(1,t/42));
    if(P.slow>0){ P.slow-=dt; base*=0.5; }
    if(P.boost>0){ P.boost-=dt; base*=1.3; }
    if(inPud) base*=0.85;
    v=base; dist+=v*dt;
  }

  // player movement
  const tx=laneX(P.lane);
  P.x+=(tx-P.x)*(1-Math.exp(-dt*22));
  if(P.hop>0) P.hop=Math.max(0,P.hop-dt/0.15);
  if(P.landPop>0) P.landPop=Math.max(0,P.landPop-dt/0.18);
  if(P.jump>0){ P.jump+=dt/JUMP_DUR;
    if(P.jump>=1){ P.jump=0; P.landPop=1; sfx.land();
      burst(P.x,playerY,4,'dot','#b9c7a6'); } }
  P.z=P.jump>0?4*P.jump*(1-P.jump):0;
  if(P.invuln>0) P.invuln-=dt;

  // obstacles: drift + scroll + cull
  for(const o of obstacles){
    o.y+=v*PXPM*dt;
    if(o.type==='beetle') o.laneF=o.lane+Math.sin(t*1.5+o.ph)*o.amp, o.x=laneX(o.laneF);
    else if(o.type==='fly'||o.type==='flyAir') o.laneF=o.lane+Math.sin(t*2.2+o.ph)*0.18, o.x=laneX(o.laneF);
    else o.x=laneX(o.laneF);
    if(o.type==='tire') o.rot+=(v*PXPM*dt)/(o.w/2);
  }
  obstacles=obstacles.filter(o=>!o.dead && o.y<H+160);

  // spawn ahead (world-space density)
  if(mode==='play'){ const aheadM=(playerY+90)/PXPM;
    while(lastWave<dist+aheadM){ spawnWave(playerY-(lastWave-dist)*PXPM); lastWave+=waveGapM(); } }

  // collisions
  inPud=false;
  for(const o of obstacles){
    const dy=o.y-playerY; if(dy<-60||dy>60) continue;
    const overlap=Math.abs(o.x-P.x)<(o.w/2+12-4);
    if(!overlap) continue;
    if(o.type==='fly'||o.type==='flyAir'){
      const need=(o.type==='flyAir')?P.z>0.48:P.z<0.35;
      if(need){ catchFly(o); }
    } else if(o.type==='pud'){ if(P.z<0.18){ inPud=true; if(Math.random()<dt*9) burst(P.x,playerY+8,1,'drop','#7cc4ee'); } }
    else {
      const th=o.type==='log'?0.62:0.42;
      if(P.z<th && P.invuln<=0) hit(o);
    }
  }

  // AI racers
  for(const a of ai){
    a.p=a.p0+(RACE_M-a.p0)*Math.min(1,t/a.T);
    a.hopT-=dt; if(a.hopT<0){ a.hopT=2+Math.random()*3; a.lane=clamp(a.lane+(Math.random()<0.5?-1:1),0,LANES-1); }
    a.x+=(laneX(a.lane)-a.x)*(1-Math.exp(-dt*6));
  }

  // finish line
  if(mode==='play' && dist>=RACE_M){
    const ft=t, rank=1+ai.filter(a=>a.p>=RACE_M).length;
    const rb=[500,300,150,50][rank-1]||0, par=(ai[0].T+ai[2].T)/2;
    const tb=Math.max(0,Math.round((par-ft)*8));
    finishInfo={time:ft,rank};
    result={score:1000+rb+flyCount*25+tb, time:ft, rank, rb, tb, nb:false};
    mode='over'; overTimer=1.4; sfx.finish(); confetti(); shake=6;
  }

  if(mode==='over'){
    overTimer-=dt;
    if(overTimer<=0 && !overShown){ overShown=true; result.nb=saveBest(result.score); showOverlay(panelOver()); }
  }

  // wind volume
  if(windG&&AC) windG.gain.setTargetAtTime(mode==='play'?(0.015+(v/24)*0.06):0, AC.currentTime, 0.15);
}

function drawFrog(x,y,s,tilt,c1,c2,alpha){
  ctx.save(); ctx.translate(x,y); ctx.rotate(tilt||0); ctx.scale(s,s);
  if(alpha!==undefined) ctx.globalAlpha*=alpha;
  ctx.fillStyle=c2;
  E(-13,10,7.5,5,-0.6); E(13,10,7.5,5,0.6);            // back legs
  E(-11,-11,4.6,4.6); E(11,-11,4.6,4.6);               // front legs
  const g=ctx.createRadialGradient(-4,-7,3,0,0,19);
  g.addColorStop(0,c1); g.addColorStop(1,c2);
  ctx.fillStyle=g; E(0,0,13,16.5);                      // body
  ctx.fillStyle='rgba(0,0,0,0.14)';
  E(-5,3,1.8,1.8); E(4,7,1.6,1.6); E(2,-3,1.5,1.5);     // spots
  ctx.fillStyle='rgba(255,255,255,0.16)'; E(-2,-2,7,9); // back sheen
  ctx.fillStyle='#fff'; E(-6,-13,4.6,4.6); E(6,-13,4.6,4.6);   // eyes
  ctx.fillStyle='#1c2417'; E(-7,-14.5,2.2,2.2); E(5,-14.5,2.2,2.2);
  ctx.fillStyle='#fff'; E(-7.6,-15,0.9,0.9); E(4.4,-15,0.9,0.9);
  ctx.restore();
}
function drawObs(o){
  const x=o.x, y=o.y;
  if(o.type==='tire'){
    const r=o.w/2;
    ctx.fillStyle='rgba(0,0,0,0.2)'; E(x+3,y+4,r,r*0.9);
    ctx.fillStyle='#20242a'; E(x,y,r,r);
    ctx.fillStyle='#3a4048'; E(x,y,r*0.58,r*0.58);
    ctx.fillStyle='#9aa2ad'; E(x,y,r*0.17,r*0.17);
    ctx.fillStyle='#14171b';
    for(let i=0;i<8;i++){ const a=o.rot+i*Math.PI/4; E(x+Math.cos(a)*r*0.8, y+Math.sin(a)*r*0.8, 5.5,2.6, a); }
    ctx.strokeStyle='rgba(255,255,255,0.1)'; ctx.lineWidth=3;
    ctx.beginPath(); ctx.arc(x,y,r*0.8,-2.4,-1.2); ctx.stroke();
  }
  else if(o.type==='beetle'){
    ctx.save(); ctx.translate(x,y);
    ctx.strokeStyle='#241811'; ctx.lineWidth=2;
    for(let i=0;i<3;i++){ const wig=Math.sin(t*18+o.ph+i)*0.25;
      ctx.beginPath(); ctx.moveTo(-9,-6+i*6); ctx.lineTo(-17,-9+i*7+wig*6); ctx.stroke();
      ctx.beginPath(); ctx.moveTo(9,-6+i*6); ctx.lineTo(17,-9+i*7-wig*6); ctx.stroke(); }
    ctx.fillStyle='#3b2a1e'; E(0,0,o.w*0.4,o.w*0.58);
    ctx.strokeStyle='#241811'; ctx.lineWidth=1.5;
    ctx.beginPath(); ctx.moveTo(0,-o.w*0.5); ctx.lineTo(0,o.w*0.55); ctx.stroke();
    ctx.fillStyle='#241811'; E(0,o.w*0.6,5,4.5);
    ctx.fillStyle='rgba(255,255,255,0.15)'; E(-3,-4,3,5);
    ctx.restore();
  }
  else if(o.type==='log'){
    const w=o.w;
    ctx.fillStyle='rgba(0,0,0,0.28)'; rr(x-w/2,y-6,w,38,17);
    const g=ctx.createLinearGradient(x,y-17,x,y+17);
    g.addColorStop(0,'#9a6a3a'); g.addColorStop(1,'#6d4423');
    ctx.fillStyle=g; rr(x-w/2,y-17,w,34,17);
    ctx.strokeStyle='rgba(60,35,15,0.5)'; ctx.lineWidth=2;
    for(let i=-1;i<=1;i++){ ctx.beginPath(); ctx.moveTo(x-w*0.3,y+i*8); ctx.lineTo(x+w*0.3,y+i*8+2); ctx.stroke(); }
    for(const ex of [x-w/2+14,x+w/2-14]){
      ctx.fillStyle='#caa06a'; E(ex,y,15,15);
      ctx.strokeStyle='#7c5330'; ctx.lineWidth=3; ctx.beginPath(); ctx.arc(ex,y,9,0,6.28); ctx.stroke();
    }
  }
  else if(o.type==='pud'){
    const w=o.w;
    ctx.fillStyle='rgba(64,145,205,0.5)'; E(x,y,w*0.5,19);
    ctx.fillStyle='rgba(150,210,240,0.35)'; E(x-6,y-3,w*0.3,10);
    const rr2=((t*1.4+o.ph)%1);
    ctx.strokeStyle=`rgba(255,255,255,${0.3*(1-rr2)})`; ctx.lineWidth=2;
    ctx.beginPath(); ctx.ellipse(x,y,w*0.15+rr2*w*0.32,6+rr2*9,0,0,6.28); ctx.stroke();
  }
  else { // flies
    const lift=o.type==='flyAir'?46+Math.sin(t*3+o.ph)*6:10;
    ctx.fillStyle='rgba(0,0,0,0.18)'; E(x,y,7,3);
    const fy=y-lift, flap=Math.abs(Math.sin(t*42+o.ph));
    ctx.fillStyle='rgba(255,216,61,0.28)'; E(x,fy,11,11);
    ctx.fillStyle='rgba(255,255,255,0.75)';
    E(x-5,fy-2,6*flap+1.5,3,-0.5); E(x+5,fy-2,6*flap+1.5,3,0.5);
    ctx.fillStyle='#ffcf3f'; E(x,fy,4.5,6);
    ctx.fillStyle='#7a5b10'; E(x,fy-6,3,3);
  }
}

  const lift=P.z*HJ+hopArc+(mode==='menu'||mode==='count'?Math.abs(Math.sin(t*4))*3:0);
  const sx=(1+P.z*0.22)*(1+P.landPop*0.3), sy=(1-P.z*0.3)*(1-P.landPop*0.3);
  ctx.fillStyle=`rgba(0,0,0,${0.28*(1-P.z*0.5)})`; E(P.x,playerY+12,15*(1-P.z*0.3),6*(1-P.z*0.3));

  let alpha=1; if(P.invuln>0 && Math.floor(t*12)%2===0) alpha=0.35;

      const s=3+4*a; ctx.beginPath(); ctx.moveTo(-s,0); ctx.lineTo(s,0); ctx.moveTo(0,-s); ctx.lineTo(0,s); ctx.stroke(); ctx.restore(); }
    else { ctx.globalAlpha=a; ctx.fillStyle=p.col; E(p.x,p.y,p.r,p.r); ctx.globalAlpha=1; }
  }

  ctx.font='bold 15px Consolas, monospace'; ctx.textAlign='center';
  for(const f of floaters){ ctx.globalAlpha=clamp(f.life/0.9,0,1);
    ctx.fillStyle='#081a0e'; ctx.fillText(f.txt,f.x+1,f.y+1);
    ctx.fillStyle='#ffd83d'; ctx.fillText(f.txt,f.x,f.y); }
  ctx.globalAlpha=1;

  if(v>12.5 && mode==='play'){ const n=Math.min(6,Math.floor((v-12)*0.9));
    for(let i=0;i<n;i++){ const x=Math.random()*W, y=Math.random()*H, l=26+v;

  const tg=ctx.createLinearGradient(0,0,0,90);
  tg.addColorStop(0,'rgba(0,0,0,0.35)'); tg.addColorStop(1,'rgba(0,0,0,0)');

  if(flash>0){ ctx.fillStyle=`rgba(255,70,45,${flash*0.32})`; ctx.fillRect(0,0,W,H); }

  // countdown / GO
  ctx.textAlign='center'; ctx.textBaseline='middle';
  if(mode==='count'){
    const n=Math.ceil(cd);
    if(n>=1&&n<=3){ const age=1-(cd-Math.floor(cd)), s=(1.25-age*0.25)*Math.min(1,W/420);
      ctx.font=`italic 900 ${Math.round(120*s)}px Impact, "Arial Black", sans-serif`;

  }
  if(mode==='play'&&goFlash>0){ const a=goFlash/0.85, s=(1+(0.85-goFlash)*0.7)*Math.min(1,W/420);
    ctx.globalAlpha=a; ctx.font=`italic 900 ${Math.round(96*s)}px Impact, "Arial Black", sans-serif`;
    ctx.fillStyle='rgba(0,0,0,0.5)'; ctx.fillText('HOP!',W/2+5,H*0.38+5);
    ctx.fillStyle='#c8ff5e'; ctx.fillText('HOP!',W/2,H*0.38); ctx.globalAlpha=1; }
  if(mode==='over'&&!overShown){ const s=1+(1.4-overTimer)*0.1;
    ctx.font=`italic 900 ${Math.round(72*Math.min(1,W/380))}px Impact, "Arial Black", sans-serif`;
    ctx.fillStyle='rgba(0,0,0,0.5)'; ctx.fillText(finishInfo.dnf?'WRECKED!':'FINISH!',W/2+4,H*0.4+4);
    ctx.fillStyle=finishInfo.dnf?'#ff6b52':'#ffd83d'; ctx.fillText(finishInfo.dnf?'WRECKED!':'FINISH!',W/2,H*0.4); }
  ctx.textBaseline='alphabetic';

  if('ontouchstart' in window && mode==='play'){
    ctx.font='12px Consolas, monospace'; ctx.fillStyle='rgba(220,255,190,0.3)';
    ctx.fillText('◀ HOP', W*0.16, H-24); ctx.fillText('JUMP ▶', W*0.5, H-24); ctx.textAlign='center';
  }
}

}
