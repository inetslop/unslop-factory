const MUSIC={step:0,next:0,dur:0.21}; // ~143bpm eighth
function scheduleMusic(){
  if(!AC||muted) return;
  if(state!=='play'){ return; }
  if(MUSIC.next < AC.currentTime-0.3) MUSIC.next=AC.currentTime+0.1;
  while(MUSIC.next < AC.currentTime+0.18){
    const t=MUSIC.next, s=MUSIC.step%16;
    // kick every 2 steps
    if(s%2===0) thumpAt(t, s===0? .30:.22);
    // hat off-beats
    if(s%2===1) hatAt(t,.05);
    const bass={0:55,3:55,6:65.41,8:55,11:49.0,14:43.65}[s];
    if(bass) bassAt(t,bass);
    // arp pluck on even steps cycling scale
    const arp=[220,329.63,261.63,392,329.63,440,392,523.25];
    if(s%4===2) pluckAt(t, arp[(s/4|0)%8]); // steps 2,6,10,14
    if(s===0) padAt(t);
    MUSIC.step++; MUSIC.next+=MUSIC.dur;
  }
}

function nextWave(){
  wave++;
  $('hWave').textContent=String(wave).padStart(2,'0');

}
function buildSpawns(){
  spawnQ=[];let t=1.6;const budget=Math.min(6+wave*2,26);
  while(budget>0){
    const r=Math.random();
    if(wave>=3&&r<.2){spawnQ.push({t,type:'tank'});budget-=4;t+=rand(1,1.6);}
    else if(wave>=2&&r<.55){const n=irand(2,3);for(let i=0;i<n;i++)spawnQ.push({t:t+i*.3,type:'weaver',i,n});budget-=n;t+=rand(1.1,1.7);}
    else{const n=Math.min(irand(3,5),budget),pat=irand(0,1);for(let i=0;i<n;i++)spawnQ.push({t:t+i*.24,type:'drone',i,n,pat});budget-=n;t+=rand(1,1.6);}
  }
  spawnQ.sort((a,b)=>a.t-b.t);
}
function mkEnemy(kind,x,y,extra){
  const e={kind,x,y,t:0,vx:0,hp:1,maxHp:1,r:15,fireCd:rand(.8,1.6),flash:0,score:100,dead:false};
  if(kind==='drone'){e.hp=e.maxHp=2+Math.floor(wave/3);e.r=15;e.vx=-(150+wave*6);e.score=100;}
  else if(kind==='weaver'){e.hp=e.maxHp=3+Math.floor(wave/3);e.r=15;e.baseY=y;e.amp=rand(50,95);e.frq=rand(1.6,2.6);e.vx=-(125+wave*5);e.score=150;}
  else if(kind==='tank'){e.hp=e.maxHp=10+wave*2;e.r=27;e.vx=-58;e.score=400;e.fireCd=rand(1,1.8);}
  Object.assign(e,extra||{});enemies.push(e);return e;
}
function doSpawn(it){
  if(it.type==='drone'){
    const y=it.pat===0?rand(H*.18,H*.82):H/2+(it.i-(it.n-1)/2)*70;
    mkEnemy('drone',W+30,clamp(y,40,H-40));
  }else if(it.type==='weaver'){
    const y=clamp(H/2+(it.i-(it.n-1)/2)*120,60,H-60);
    mkEnemy('weaver',W+30,y,{baseY:y});
  }else mkEnemy('tank',W+40,rand(80,H-80));
}

function mkEB(x,y,a,spd){
  if(ebullets.length>260)return;
  ebullets.push({x,y,vx:Math.cos(a)*spd,vy:Math.sin(a)*spd,r:5,dead:false});
}
function aimShot(x,y,spd,jit){
  const a=Math.atan2(player.y-y,player.x-x)+rand(-jit,jit);
  mkEB(x,y,a,spd);sfxEShoot();
}
function shoot(){
  const px=player.x+20,py=player.y;
  const mk=(ox,oy,a)=>{const c=Math.cos(a),s=Math.sin(a);bullets.push({x:px+ox*c,y:py+oy+s,vx:c*720,vy:s*720,r:4,dead:false});};
  sfxShoot();
  parts.push({k:'ring',x:px,y:py,t:0,maxT:.1,R:13,col:'rgba(180,240,255,'});
  const w=player.weapon;
  if(w===1)mk(0,0,0);
  else{mk(0,-6,0);mk(0,6,0);
    if(w>=3){mk(0,-2,.15);mk(0,2,-.15);}
    if(w===4){mk(0,-9,.28);mk(0,9,-.28);}
  }
}
function useBomb(){
  if(player.bombs<=0||!player.alive)return;
  player.bombs--;sfxBombS();shake=Math.max(shake,16);bombFlash=.5;hitstop=.14;
  bombWave={x:player.x,y:player.y,r:20};
  for(const b of ebullets)if(!b.dead){b.dead=true;spark(b.x,b.y);}
  for(const e of enemies){
    if(e.dead||e.dying)continue;
    e.hp-=e.kind==='boss'?7:4;e.flash=1;
    if(e.hp<=0){if(e.kind==='boss')e.dying=1.6;else killEnemy(e);}
  }
}
function damagePlayer(){
  if(player.invulnT>0||!player.alive)return;
  if(player.shield){
    player.shield=false;sfxPower();
    parts.push({k:'ring',x:player.x,y:player.y,t:0,maxT:.4,R:52,col:'rgba(83,230,255,'});
    addFloat(player.x,player.y-26,'SHIELD DOWN','#53e6ff');
    player.invulnT=1.2;shake=Math.max(shake,8);return;
  }
  lives--;streak=0;mult=1;syncCombo(true);
  if(lives<=0){
    player.alive=false;player.deathT=1.7;
    explodeAt(player.x,player.y,2.6);sfxBoom(true);shake=24;hitstop=.2;
  }else{
    sfxPlayerHit();shake=Math.max(shake,14);hitstop=.12;player.invulnT=2.3;
    explodeAt(player.x,player.y,.8);
    for(const b of ebullets){const dx=b.x-player.x,dy=b.y-player.y;if(dx*dx+dy*dy<170*170){b.dead=true;spark(b.x,b.y);}}
  }
}
function killEnemy(e){
  if(e.dead)return;e.dead=true;kills++;streak++;mult=Math.min(8,1+((streak/5)|0));syncCombo();
  const pts=e.score*mult;score+=pts;addFloat(e.x,e.y,'+'+pts,'#ffd9a0');
  explodeAt(e.x,e.y,e.r/14);
  const roll=Math.random();
  if(e.kind==='tank')dropPower(e.x,e.y,['w','s','b'][irand(0,2)]);
  else if(roll<.06)dropPower(e.x,e.y,'c');
  else if(roll<.11)dropPower(e.x,e.y,['w','s','b'][irand(0,2)]);
}
function killBoss(e){
  e.dead=true;boss=null;$('bossBar').classList.remove('on');
  const pts=(2000+wave*300)*mult;score+=pts;kills++;
  addFloat(e.x,e.y,'+'+pts,'#ffb35c');
  for(let i=0;i<6;i++)explodeAt(rand(e.x-70,e.x+70),rand(e.y-60,e.y+60),rand(.8,1.6));
  sfxBoom(true);shake=28;hitstop=.25;
  dropPower(e.x-30,e.y-40,'w');dropPower(e.x-30,e.y+40,Math.random()<.5?'s':'b');
}

  }
}
function spark(x,y){
  const a=rand(0,TAU),s=rand(60,240);
  parts.push({k:'spark',x,y,vx:Math.cos(a)*s,vy:Math.sin(a)*s,t:0,maxT:rand(.2,.45),col:SPARKC[irand(0,3)],w:rand(1.5,2.6)});
}
function explodeAt(x,y,s){
  const n=Math.min(40,Math.floor(14*s+6));
  for(let i=0;i<n;i++){const a=rand(0,TAU),sp=rand(60,320)*clamp(s,.5,1.8);
    parts.push({k:'spark',x,y,vx:Math.cos(a)*sp,vy:Math.sin(a)*sp,t:0,maxT:rand(.3,.7),col:SPARKC[irand(0,3)],w:rand(1.5,3)});}
  for(let i=0;i<Math.min(8,4*s|0);i++){const a=rand(0,TAU),sp=rand(40,160)*s;
    parts.push({k:'deb',x,y,vx:Math.cos(a)*sp,vy:Math.sin(a)*sp,rot:rand(0,TAU),vr:rand(-6,6),t:0,maxT:rand(.5,1.1),sz:rand(3,7)*s});}
  for(let i=0;i<3;i++)parts.push({k:'smoke',x:x+rand(-8,8),y:y+rand(-8,8),vx:rand(-20,20),vy:rand(-20,20),t:0,maxT:rand(.5,.9),r:rand(8,14)*s});
  parts.push({k:'ring',x,y,t:0,maxT:.35,R:44*s,col:'rgba(255,179,92,'});
  shake=Math.min(30,shake+5*s);sfxBoom(s>1.3);
}

}

/* ================= update ================= */
function updWorldFx(dt){
  decorT-=dt;
  if(decorT<=0){decorT=rand(2.5,5);
    decorArr.push({img:rockSprites[irand(0,2)],x:W+90,y:rand(40,H-40),vx:-rand(50,110),vy:rand(-8,8),rot:rand(0,TAU),vr:rand(-.6,.6),sc:rand(.5,1.3)});}
  for(const d of decorArr){d.x+=d.vx*dt;d.y+=d.vy*dt;d.rot+=d.vr*dt;}
  decorArr=decorArr.filter(d=>d.x>-140);
  for(let i=parts.length-1;i>=0;i--){const p=parts[i];p.t+=dt;
    if(p.t>=p.maxT){parts.splice(i,1);continue;}
    p.x+=(p.vx||0)*dt;p.y+=(p.vy||0)*dt;
    if(p.k==='spark'){p.vx*=Math.exp(-3*dt);p.vy*=Math.exp(-3*dt);}
    if(p.k==='deb')p.rot+=p.vr*dt;}
  if(parts.length>520)parts.splice(0,parts.length-520);
  for(let i=floats.length-1;i>=0;i--){const f=floats[i];f.t+=dt;f.y-=32*dt;if(f.t>.95)floats.splice(i,1);}
}

  scroll+=26*dt;
  player.x=W*.15;player.y=H/2+Math.sin(time*1.3)*24;player.tilt=Math.cos(time*1.3)*.1;
  if(player.alive)spawnFlame();
  updWorldFx(dt);
}
function updateBoss(e,dt){
  const tx=Math.max(W*.74,W-210);
  if(e.dying>0){
    e.dying-=dt;
    if(Math.random()<dt*14)explodeAt(rand(e.x-65,e.x+65),rand(e.y-58,e.y+58),rand(.5,1.1));
    if(e.dying<=0)killBoss(e);
    return;
  }
  const ph=e.hp/e.maxHp;
  if(e.x>tx){e.x+=((tx-e.x)*Math.min(1,dt*1.2)-40*dt);}
  else{e.x=tx+Math.sin(e.t*.5)*8;}
  e.y=H/2+Math.sin(e.t*(ph<.3?1.1:.6))*H*.3;
  const ready=e.x<=tx+10&&player.alive;
  e.atkCd-=dt;
  if(e.atkCd<=0&&ready){
    const base=Math.atan2(player.y-e.y,player.x-e.x);
    const offs=ph>.6?[-.22,0,.22]:ph>.3?[-.3,-.15,0,.15,.3]:[-.4,-.2,0,.2,.4];
    for(const o of offs)mkEB(e.x-70,e.y,base+o,ph>.6?300:ph>.3?320:350);
    e.atkCd=ph>.6?1.5:ph>.3?1.1:.8;sfxEShoot();
  }
  if(ph<=.6){e.podCd-=dt;
    if(e.podCd<=0&&ready){
      const py=e.y+(Math.sin(e.t*2)>0?-58:58);
      const a=Math.atan2(player.y-py,player.x-(e.x+30));
      for(const o of[-.18,-.06,.06,.18])mkEB(e.x+30,py,a+o,270);
      e.podCd=2.1;sfxEShoot();}}
  if(ph<=.3){e.streamCd-=dt;
    if(e.streamCd<=0&&ready){mkEB(e.x-60,e.y+Math.sin(e.t*9)*40,Math.PI+rand(-.55,.55),330);e.streamCd=.22;}}
  if(ph<=.6&&ph>.3){e.summonCd-=dt;
    if(e.summonCd<=0){mkEnemy('drone',e.x-90,clamp(e.y-140,40,H-40));mkEnemy('drone',e.x-90,clamp(e.y+140,40,H-40));e.summonCd=5;}}
}
function updatePlay(dt){
  scroll+=(boss?130:92)*dt;
  if(bombFlash>0)bombFlash-=dt*1.6;
  if(shake>0)shake=Math.max(0,shake*Math.exp(-7*dt));

  /* player */
  if(player.alive){
    const sp=430;
    let mx=(keys.right?1:0)-(keys.left?1:0),my=(keys.down?1:0)-(keys.up?1:0);
    if(mx&&my){mx*=.7071;my*=.7071;}
    player.x=clamp(player.x+mx*sp*dt,26,W-40);
    player.y=clamp(player.y+my*sp*dt,30,H-30);
    player.tilt+=((my*.22)-player.tilt)*Math.min(1,dt*10);
    if(keys.fire){player.fireCd-=dt;if(player.fireCd<=0){shoot();player.fireCd=player.weapon>=4?.12:.14;}}
    else player.fireCd=Math.min(player.fireCd,.03);
    if(player.invulnT>0)player.invulnT-=dt;
    spawnFlame();
  }else{
    player.deathT-=dt;
    if(player.deathT<=0)gameOver();
  }

  /* spawns */
  for(let i=spawnQ.length-1;i>=0;i--){spawnQ[i].t-=dt;if(spawnQ[i].t<=0){doSpawn(spawnQ[i]);spawnQ.splice(i,1);}}
  if(bossWarnT>0){bossWarnT-=dt;if(bossWarnT<=0)spawnBoss();}

  /* enemies */
  for(const e of enemies){
    if(e.dead)continue;
    e.t+=dt;if(e.flash>0)e.flash=Math.max(0,e.flash-dt*5);
    if(e.kind==='boss'){updateBoss(e,dt);continue;}
    e.x+=e.vx*dt;
    if(e.kind==='weaver')e.y=e.baseY+Math.sin(e.t*e.frq)*e.amp;
    else if(e.kind==='tank')e.y=clamp(e.y+Math.sin(e.t*.8)*14*dt,60,H-60);
    e.fireCd-=dt;
    if(e.fireCd<=0&&player.alive&&e.x<W-30){
      const spd=Math.min(420,250+wave*8);
      if(e.kind==='tank'){
        const a=Math.atan2(player.y-e.y,player.x-e.x);
        for(const o of[-.28,0,.28])mkEB(e.x-e.r,e.y,a+o,spd);
        e.fireCd=rand(1.7,2.3);sfxEShoot();
      }else{aimShot(e.x-e.r,e.y,spd+.3*rand(0,1),e.kind==='drone'?.05:.09);e.fireCd=rand(1.4,2.3);}
    }
    if(e.x<-70)e.dead=true;
  }

  /* bullets */
  for(const b of bullets){b.x+=b.vx*dt;b.y+=b.vy*dt;if(b.x>W+40||b.y<-40||b.y>H+40)b.dead=true;}
  for(const e of enemies){
    if(e.dead||e.dying)continue;
    for(const b of bullets){
      if(b.dead)continue;
      const dx=e.x-b.x,dy=e.y-b.y,rr=e.r+b.r;
      if(dx*dx+dy*dy<rr*rr){
        b.dead=true;spark(b.x,b.y);sfxHit();e.hp--;e.flash=1;
        if(e.hp<=0){if(e.kind==='boss')e.dying=1.6;else killEnemy(e);}
        break;
      }
    }
  }
  for(const b of ebullets){b.x+=b.vx*dt;b.y+=b.vy*dt;if(b.x<-40||b.x>W+40||b.y<-40||b.y>H+40)b.dead=true;}

  /* player collisions */
  if(player.alive&&player.invulnT<=0){
    for(const b of ebullets){
      if(b.dead)continue;
      const dx=player.x-b.x,dy=player.y-b.y,rr=player.r+b.r;
      if(dx*dx+dy*dy<rr*rr){b.dead=true;damagePlayer();break;}
    }
  }
  if(player.alive&&player.invulnT<=0){
    for(const e of enemies){
      if(e.dead||e.dying)continue;
      const rr=player.r+(e.kind==='boss'?e.r*.65:e.r*.8);
      const dx=player.x-e.x,dy=player.y-e.y;
      if(dx*dx+dy*dy<rr*rr){
        damagePlayer();
        if(e.kind!=='boss'){if(e.hp<=2)killEnemy(e);else{e.hp-=2;e.flash=1;}}
        break;
      }
    }
  }

  /* powerups */
  for(const p of powerups){
    p.t+=dt;p.x-=85*dt;p.y=p.baseY+Math.sin(p.t*2.6)*10;
    const dx=player.x-p.x,dy=player.y-p.y,d2=dx*dx+dy*dy;
    if(player.alive&&d2<130*130){const d=Math.sqrt(d2)||1;p.x+=dx/d*300*dt;p.baseY+=dy/d*300*dt;}
    if(p.x<-40)p.dead=true;
    else if(player.alive&&d2<(player.r+14)*(player.r+14)){p.dead=true;applyPower(p);}
  }

  /* bomb wave */
  if(bombWave){bombWave.r+=950*dt;if(bombWave.r>Math.hypot(W,H))bombWave=null;}

  updWorldFx(dt);

  /* cull */
  bullets=bullets.filter(b=>!b.dead);
  ebullets=ebullets.filter(b=>!b.dead);
  enemies=enemies.filter(e=>!e.dead);
  powerups=powerups.filter(p=>!p.dead);

  /* wave progression */
  if(!boss&&spawnQ.length===0&&bossWarnT<0&&enemies.length===0&&player.alive){
    interT+=dt;if(interT>1.6){interT=0;nextWave();}
  }else interT=0;

  music();syncHUD(false);
}

    const el=$('hShield');el.textContent=sh?'ONLINE':'OFFLINE';el.className='st '+(sh?'on':'off');hudCache.sh=sh;

}
