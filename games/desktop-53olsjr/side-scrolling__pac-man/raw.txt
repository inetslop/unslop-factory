let AC=null,master=null,muted=false;

function ensureAudio(){
  try{
    if(!AC){ const C=window.AudioContext||window.webkitAudioContext; if(!C)return;
      AC=new C(); master=AC.createGain(); master.gain.value=muted?0:0.16; master.connect(AC.destination); }
    if(AC.state==='suspended') AC.resume();
  }catch(e){}
}
function tone(f0,f1,dur,type,vol,delay){
  if(!AC||muted)return;
  try{
    const t0=AC.currentTime+(delay||0), o=AC.createOscillator(), g=AC.createGain();
    o.type=type; o.frequency.setValueAtTime(f0,t0);
    if(f1) o.frequency.exponentialRampToValueAtTime(Math.max(30,f1),t0+dur);
    g.gain.setValueAtTime(vol,t0); g.gain.exponentialRampToValueAtTime(0.001,t0+dur);
    o.connect(g); g.connect(master); o.start(t0); o.stop(t0+dur+0.03);
  }catch(e){}
}

const R=13;                                  // rows incl. border rails
const wk=(c,r)=>c+','+r;
let walls,pellets,lastGen,cleanCol,seed=0;
function mulberry32(a){return function(){a|=0;a=a+0x6D2B79F5|0;let t=Math.imul(a^a>>>15,1|a);t=t+Math.imul(t^t>>>7,61|t)^t;return((t^t>>>14)>>>0)/4294967296;};}
const isWall=(c,r)=>walls.has(wk(c,r));

function genColumn(c){
  if(walls.has(wk(c,0)))return;
  walls.set(wk(c,0),1); walls.set(wk(c,R-1),1);
  const rng=mulberry32((seed^Math.imul(c+1,2654435761))|0);
  if(c>=5){
    const dens=0.40+0.20*Math.min(1,c/500), roll=rng();
    if(roll<dens){
      const L1=2+((rng()*4)|0);                 // bar length 2..5
      let a1=2+((rng()*(9-L1))|0);              // rows a1..a1+L1-1, inside 2..9
      for(let r=a1;r<a1+L1;r++) walls.set(wk(c,r),1);
      if(rng()<0.35){                            // optional second bar
        const L2=2+((rng()*3)|0); let a2=-1;
        if(a1+L1-1<=5){ if(4-L2>0) a2=7+((rng()*(4-L2))|0); }
        else if(a1-L2-3>0) a2=2+((rng()*(a1-L2-3))|0);
        if(a2>=0) for(let r=a2;r<a2+L2;r++) walls.set(wk(c,r),1);
      }
    } else if(roll<dens+0.28){
      const r=2+((rng()*8)|0);                   // single pillar
      walls.set(wk(c,r),1);
    }
  }
  for(let r=1;r<R-1;r++) if(!walls.has(wk(c,r))) pellets.set(wk(c,r),1);
  if(c>6 && c%23===9){                            // power pellet, mid rows
    const cand=[4,5,6,7].filter(r=>!walls.has(wk(c,r)));
    if(cand.length) pellets.set(wk(c,cand[(rng()*cand.length)|0]),2);
  }
}
function ensureGen(untilC){ while(lastGen<untilC) genColumn(++lastGen); }
function cleanup(){
  const cutoff=Math.floor(G.camX)-14;
  if(cutoff-cleanCol>6){
    for(const k of walls.keys())   if(+k.split(',')[0]<cutoff) walls.delete(k);
    for(const k of pellets.keys()) if(+k.split(',')[0]<cutoff) pellets.delete(k);
    cleanCol=cutoff;
  }
}

/* ================= MOVEMENT / COLLISION ================= */
function hitWall(px,py,rw,c2,r2){
  const nx=Math.max(c2,Math.min(px,c2+1)), ny=Math.max(r2,Math.min(py,r2+1));
  const dx=px-nx, dy=py-ny; return dx*dx+dy*dy<rw*rw;
}
function moveAxis(axis,delta){
  if(!delta)return;
  let p=P[axis]+delta; const r=P.r;
  if(axis==='x'){
    const rr0=Math.floor(P.y-r), rr1=Math.floor(P.y+r);
    for(let c=Math.floor(p-r);c<=Math.floor(p+r);c++)
      for(let rr=rr0;rr<=rr1;rr++)
        if(isWall(c,rr)&&hitWall(p,P.y,r,c,rr)) p=delta>0?c-r:c+1+r;
    P.x=p;
  }else{
    const cc0=Math.floor(P.x-r), cc1=Math.floor(P.x+r);
    for(let rr=Math.floor(p-r);rr<=Math.floor(p+r);rr++)
      for(let c=cc0;c<=cc1;c++)
        if(isWall(c,rr)&&hitWall(P.x,p,r,c,rr)) p=delta>0?rr-r:rr+1+r;
    P.y=p;
  }
}
function lerpAngle(a,b,k){ let d=(b-a)%6.2831853; if(d>Math.PI)d-=6.2831853; if(d<-Math.PI)d+=6.2831853; return a+d*Math.min(1,k); }

/* ================= GHOST AI (BFS field from player) ================= */
const DIRS=[[1,0],[-1,0],[0,1],[0,-1]];
function computeField(){
  const c0=Math.floor(G.camX)-3, span=Math.ceil(viewW)+14, c1=c0+span-1;
  const arr=new Int32Array(span*R).fill(-1);
  let sc=Math.max(c0,Math.min(c1,Math.floor(P.x))), sr=Math.max(0,Math.min(R-1,Math.floor(P.y)));
  if(isWall(sc,sr))sr=1;
  const si=(sc-c0)*R+sr; arr[si]=0; let q=[si],head=0;
  while(head<q.length){
    const idx=q[head++], r=idx%R, c=c0+(idx/R|0), dv=arr[idx]+1;
    for(let i=0;i<4;i++){
      const nc=c+DIRS[i][0], nr=r+DIRS[i][1];
      if(nc<c0||nc>c1||nr<0||nr>=R)continue;
      const ni=(nc-c0)*R+nr;
      if(arr[ni]===-1&&!isWall(nc,nr)){ arr[ni]=dv; q.push(ni); }
    }
  }
  field={c0,c1,span,arr,t:G.t};
}
function pickNext(g){
  let best=Infinity, cands=[];
  const gc=Math.max(field.c0,Math.min(field.c1,Math.floor(g.x)));
  const gr=Math.max(0,Math.min(R-1,Math.floor(g.y)));
  for(let i=0;i<4;i++){
    const nc=gc+DIRS[i][0], nr=gr+DIRS[i][1];
    if(nc<field.c0||nc>field.c1||nr<0||nr>=R||isWall(nc,nr))continue;
    const dval=field.arr[(nc-field.c0)*R+nr]; if(dval<0)continue;
    const m=G.powerT>0?-dval:dval;
    if(m<best){best=m;cands=[[nc,nr,DIRS[i]]];} else if(m===best)cands.push([nc,nr,DIRS[i]]);
  }
  let pick=null;
  if(cands.length){
    const nr=cands.filter(c=>!(c[2][0]===-g.dir.x&&c[2][1]===-g.dir.y));
    const pool=nr.length?nr:cands;
    pick=pool[(Math.random()*pool.length)|0];
  }else{
    // fallback: keep going if open, else hold
    const nc=gc+g.dir.x, nr=gr+g.dir.y;
    if(g.dir.x||g.dir.y){ if(nc>=field.c0&&nc<=field.c1&&nr>=0&&nr<R&&!isWall(nc,nr)) pick=[nc,nr,g.dir]; else pick=[gc,gr,{x:0,y:0}]; }
    else pick=[gc,gr,{x:0,y:0}];
  }
  g.tx=pick[0]+0.5; g.ty=pick[1]+0.5; g.dir={x:pick[2][0],y:pick[2][1]};
}
function pickKind(){ const r=Math.random(); return r<0.3?'b':r<0.55?'p':r<0.8?'i':'c'; }
function spawnGhost(ramp){
  const elite=G.t>25&&Math.random()<0.18&&!ghosts.some(g=>g.kind==='e');
  const kind=elite?'e':pickKind();
  const c=Math.max(0,Math.floor(G.voidX+1)); ensureGen(c);
  const open=[]; for(let r=1;r<R-1;r++) if(!isWall(c,r)) open.push(r);
  const r=open[(Math.random()*open.length)|0];
  const spd=elite?5.7+ramp*0.6:(4.5+ramp*0.8)*(0.92+Math.random()*0.18);
  ghosts.push({x:c+0.5,y:r+0.5,tx:c+0.5,ty:r+0.5,dir:{x:1,y:0},spd,kind,color:COLORS[kind],ph:Math.random()*6});
  burst(c+0.5,r+0.5,'rgba(180,190,230,0.9)',6);
}

function burst(x,y,color,n){
  for(let i=0;i<n;i++){ const a=Math.random()*6.283, sp=1+Math.random()*3.5;
    particles.push({type:'dot',x,y,vx:Math.cos(a)*sp,vy:Math.sin(a)*sp,life:.45+Math.random()*.4,max:.85,size:S*(0.04+Math.random()*0.07),color}); }
  if(particles.length>380) particles.splice(0,particles.length-380);
}
function ring(x,y,color){ particles.push({type:'ring',x,y,life:.35,max:.35,color}); }
function soul(x,y){ particles.push({type:'soul',x,y,vy:-1.4,life:.9,max:.9}); }

function addScore(n,cx,cy){
  G.score+=n; floater('+'+n,cx,cy);
  while(G.nextLifeIdx<LIFE_MARKS.length&&G.score>=LIFE_MARKS[G.nextLifeIdx]){
    G.lives++; syncLives(); sfx.life(); ring(P.x,P.y,'#7dffb0');
    floater('EXTRA LIFE!',P.x,P.y-1.2,'#7dffb0'); G.nextLifeIdx++;
  }
}
function syncLives(){
  elLives.innerHTML=''; for(let i=0;i<Math.min(9,G.lives);i++){ const s=document.createElement('span'); s.className='pac-ico'; elLives.appendChild(s); }
}
function eatGhost(i){
  const g=ghosts[i];
  G.chain=Math.min(4,G.chain+1);
  addScore((g.kind==='e'?500:200)*G.chain,g.x,g.y);
  sfx.ghost(); burst(g.x,g.y,g.color,16); ring(g.x,g.y,'#ffffff'); soul(g.x,g.y);
  G.shake=Math.max(G.shake,5); ghosts.splice(i,1);
}
function killPlayer(msg){
  if(G.dead)return;
  G.dead=true; G.deadT=1.3; G.cause=msg; G.powerT=0; G.chain=0;
  sfx.death(); G.shake=14; burst(P.x,P.y,'#ffe14d',26); ring(P.x,P.y,'#ffe14d');
  if(G.score>G.hi){G.hi=G.score;saveHi(G.hi);}
}
function afterDeath(){
  G.lives--; syncLives();
  if(G.lives<=0){ gameOver(); return; }
  const c=Math.floor(Math.max(G.camX,G.voidX)+viewW*0.5); ensureGen(c);
  let bestRow=1,bestD=99;
  for(let r=1;r<R-1;r++){ if(!isWall(c,r)){const d=Math.abs(r+0.5-P.y); if(d<bestD){bestD=d;bestRow=r;}} }
  P.x=c+0.5; P.y=bestRow+0.5; P.face=0; G.invuln=2.5;
  for(const g of ghosts){
    const c2=Math.max(0,Math.floor(G.voidX)+(1+((Math.random()*3)|0))); ensureGen(c2);
    const open=[]; for(let r=1;r<R-1;r++) if(!isWall(c2,r))open.push(r);
    const r=open[(Math.random()*open.length)|0];
    g.x=c2+0.5;g.y=r+0.5;g.tx=g.x;g.ty=g.y;burst(g.x,g.y,'rgba(160,170,220,.8)',4);
  }
  floater('GO!',P.x,P.y-1.2,'#7dffb0');
}

/* ================= UPDATE ================= */
function updateFx(dt){
  for(let i=particles.length-1;i>=0;i--){ const p=particles[i]; p.life-=dt;
    if(p.type==='dot'){p.x+=p.vx*dt;p.y+=p.vy*dt;p.vx*=Math.max(0,1-3*dt);p.vy*=Math.max(0,1-3*dt);}
    else if(p.type==='soul'){p.y+=p.vy*dt;}
    if(p.life<=0)particles.splice(i,1); }
  for(let i=floaters.length-1;i>=0;i--){ const f=floaters[i]; f.life-=dt*1.1; f.y-=dt*0.9; if(f.life<=0)floaters.splice(i,1); }
  G.shake=Math.max(0,G.shake-dt*30);
}
function update(dt){
  G.t+=dt;
  const ramp=Math.min(1,G.t/150);
  const scroll=3.4+ramp*2.9;

  if(G.dead){
    G.deadT-=dt; if(G.deadT<=0) afterDeath();
  }else{
    G.voidX+=scroll*dt;
    // player
    const d=inputDir();
    moveAxis('x',d.x*P.speed*dt); moveAxis('y',d.y*P.speed*dt);
    P.phase+=dt*(6+(d.x||d.y?7:2));
    let tf=0; if(d.x<0)tf=Math.PI; else if(d.y<0)tf=-Math.PI/2; else if(d.y>0)tf=Math.PI/2;
    P.face=lerpAngle(P.face,tf,dt*16);
    // eat pellet in current cell
    const cc=Math.floor(P.x), cr=Math.floor(P.y), v=pellets.get(wk(cc,cr));
    if(v){
      pellets.delete(wk(cc,cr));
      if(v===2){ G.powerT=6.5; sfx.power(); burst(cc+0.5,cr+0.5,'#ff5fa2',12); ring(cc+0.5,cr+0.5,'#ff5fa2'); addScore(50,cc+0.5,cr+0.5); }
      else{ G.score+=10; if(G.t-G.lastWaka>0.08){sfx.waka();G.lastWaka=G.t;} burst(cc+0.5,cr+0.5,'#ffd97a',2); addScoreCheckOnly(); }
    }
    // void death + warning
    if(P.x-P.r < G.voidX+0.25) killPlayer('THE VOID GOT YOU');
    else{
      if(G.invuln>0)G.invuln-=dt;
      const gap=P.x-G.voidX;
      if(gap<4&&G.t-G.lastThump>0.55){sfx.thump();G.lastThump=G.t;G.shake=Math.max(G.shake,2);}
    }
  }

  // camera (smooth)
  const camTarget=Math.max(G.voidX,P.x-viewW*0.55);
  G.camX+=(camTarget-G.camX)*Math.min(1,dt*7);
  ensureGen(Math.floor(G.camX+viewW)+8); cleanup();

  if(G.powerT>0){ G.powerT-=dt; if(G.powerT<=0)G.chain=0; }

  // spawn ghosts
  if(!G.dead){
    G.ghostTimer-=dt;
    const maxG=Math.min(6,3+((G.t/50)|0));
    if(G.ghostTimer<=0&&ghosts.length<maxG){ spawnGhost(ramp); G.ghostTimer=4.2-ramp*2.4+Math.random()*1.2; }
  }

  // AI field + ghosts
  if(G.t-field.t>0.16) computeField();
  for(let i=ghosts.length-1;i>=0;i--){
    const g=ghosts[i];
    const dx=g.tx-g.x, dy=g.ty-g.y, dist=Math.hypot(dx,dy), step=g.spd*dt;
    if(dist<=step+1e-6){ g.x=g.tx; g.y=g.ty; if(field.arr)pickNext(g); }
    else{ g.x+=dx/dist*step; g.y+=dy/dist*step; }
    if(!G.dead&&G.invuln<=0){
      const pdx=P.x-g.x, pdy=P.y-g.y;
      if(pdx*pdx+pdy*pdy<0.36){ if(G.powerT>0) eatGhost(i); else{ killPlayer('GOTCHA!'); break; } }
    }
    if(g.x<G.voidX-3){ burst(g.x,g.y,'rgba(120,130,180,.7)',5); ghosts.splice(i,1); }
  }

  updateFx(dt);
}
function addScoreCheckOnly(){
  while(G.nextLifeIdx<LIFE_MARKS.length&&G.score>=LIFE_MARKS[G.nextLifeIdx]){
    G.lives++; syncLives(); sfx.life(); ring(P.x,P.y,'#7dffb0');
    floater('EXTRA LIFE!',P.x,P.y-1.2,'#7dffb0'); G.nextLifeIdx++;
  }
}

function drawMaze(ts){
  const cA=Math.floor(G.camX)-1, cB=cA+Math.ceil(viewW)+2;
  ctx.fillStyle='rgba(35,60,190,0.14)';
  for(let c=cA;c<=cB;c++)for(let r=0;r<R;r++) if(isWall(c,r)) ctx.fillRect((c-G.camX)*S+1,r*S+1,S-2,S-2);
  ctx.beginPath();
  for(let c=cA;c<=cB;c++)for(let r=0;r<R;r++){
    if(!isWall(c,r))continue;
    const x=(c-G.camX)*S, y=r*S;
    if(!isWall(c,r-1)){ctx.moveTo(x,y);ctx.lineTo(x+S,y);}
    if(!isWall(c,r+1)){ctx.moveTo(x,y+S);ctx.lineTo(x+S,y+S);}
    if(!isWall(c-1,r)){ctx.moveTo(x,y);ctx.lineTo(x,y+S);}
    if(!isWall(c+1,r)){ctx.moveTo(x+S,y);ctx.lineTo(x+S,y+S);}
  }
  ctx.lineCap='round';
  ctx.strokeStyle='rgba(70,110,255,0.3)'; ctx.lineWidth=Math.max(6,S*0.16); ctx.stroke();
  ctx.strokeStyle='#5b86ff'; ctx.lineWidth=2.5; ctx.stroke();
  ctx.strokeStyle='rgba(190,210,255,0.45)'; ctx.lineWidth=1; ctx.stroke();
}
function drawPellets(ts){

function drawGhost(g,ts){
  const x=(g.x-G.camX)*S, y=g.y*S, gr=S*0.38;
  const scared=G.powerT>0, blink=scared&&G.powerT<1.6&&(Math.floor(ts*9)%2===0);
  const body=scared?(blink?'#f4f7ff':'#3557ff'):g.color;
  ctx.beginPath();

  ctx.lineTo(x+gr,y+gr*0.85);
  const fw=(2*gr)/3;
  for(let i=0;i<3;i++){ const xe=x+gr-fw*(i+1), wob=Math.sin(ts*12+g.ph+i*1.7)*gr*0.08;
    ctx.quadraticCurveTo(x+gr-fw*(i+0.5),y+gr*0.85+gr*0.22+wob,xe,y+gr*0.85); }
  ctx.closePath();
  ctx.fillStyle=body; ctx.fill();
  if(g.kind==='e'){ ctx.strokeStyle='rgba(255,95,162,0.9)'; ctx.lineWidth=2; ctx.stroke();
    ctx.fillStyle='#ff5fa2'; ctx.beginPath();ctx.moveTo(x-gr*0.4,y-gr*0.95);ctx.lineTo(x,y-gr*1.35);ctx.lineTo(x+gr*0.4,y-gr*0.95);ctx.closePath();ctx.fill(); }
  if(scared){
    ctx.fillStyle='#ffe9a8';
    ctx.beginPath();ctx.arc(x-gr*0.3,y-gr*0.12,gr*0.1,0,6.283);ctx.arc(x+gr*0.3,y-gr*0.12,gr*0.1,0,6.283);ctx.fill();

    }
  }
}

  }
}

      for(let yy=H;yy>=0;yy-=12){ const jx=w+Math.sin(yy*0.05+ts*6)*3+Math.sin(yy*0.11-ts*9)*2; ctx.lineTo(jx,yy); }

  }
}

**How it plays:** the maze auto-scrolls and a jagged red **void** chases you from the left — fall behind it (or get touched by a ghost) and you lose a life. Arrows/WASD move you through the neon corridors; eat pellets for score, grab the pulsing gold power pellets to turn every ghost blue and chain them for 200→1600 points. White crowned wraiths are faster than you, so use walls and power-ups wisely. Extra life at 10,000; `P` pauses, `R` restarts, `M` mutes, and on touch devices a D-pad appears. Best score is saved locally.
