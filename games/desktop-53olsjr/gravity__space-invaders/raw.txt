  for(const s of stars){s.y+=(5+s.z*12)*dt*(state==='play'?1:0.55);if(s.y>H+2){s.y=-2;s.x=rand(0,W);}}
  const G=H*1.05;
  for(const m of motes){m.vy=Math.min(m.vy+G*0.35*dt,H*0.9);m.y+=m.vy*dt;
    if(m.y>H+8){m.y=rand(-40,-5);m.x=rand(0,W);m.vy=rand(20,60);}}
  meteorT-=dt;
  if(meteorT<=0){meteorT=rand(4,9);const d=Math.random()<.5?-1:1;
    meteors.push({x:rand(W*.15,W*.85),y:-30,vx:d*rand(60,120),vy:H*rand(.75,.95),life:1.6,max:1.6});}
  for(let i=meteors.length-1;i>=0;i--){const m=meteors[i];m.x+=m.vx*dt;m.y+=m.vy*dt;m.life-=dt;
    if(m.life<=0||m.y>H+40)meteors.splice(i,1);}
}

  spriteCache={};
  const gw=11*grid.unit;
  grid.ox=(W-gw)/2; grid.oy=Math.max(84,H*0.13); grid.oyCur=grid.oy;
  grid.dir=1; grid.animT=0; grid.fireT=rand(.9,1.5); grid.freeze=0; grid.pending=0;
  grid.total=rows*11; grid.aliveN=grid.total;
  grid.dy=Math.min(40,20+wave*2);
  aliens=[];
  for(let r=0;r<rows;r++)for(let c=0;c<11;c++){
    const R=ROWS[r], a={r,c,alive:true,x:0,y:0,hw:grid.unit*0.44,hh:grid.rowH*0.36,kind:R.k,color:R.c,pts:R.p};

  const cs=clamp(H*0.0085,4.5,7), bh=7*cs;
  doomY=(rimY-6-bh)+8;
  player.x=W/2; player.y=H-58; player.vx=0; player.vy=0;
}

  a.alive=false; grid.aliveN--;
  addScore(a.pts,a.x,a.y-8,a.color);
  puff(a.x,a.y,10,a.color,150,.5,3); puff(a.x,a.y,6,'#ffffff',220,.35,2); puff(a.x,a.y,3,'rgba(90,100,120,1)',60,.7,4,false);
  sHit(); shake=Math.max(shake,2.5);
  const s=grid.unit*0.42;
  debris.push({x:a.x,y:a.y,vx:rand(-30,30),vy:rand(20,60),rot:rand(0,TAU),vr:rand(-4,4),size:s,life:9});
  if(Math.random()<0.12){const r=Math.random(),type=r<0.4?'D':r<0.75?'S':'G';
    drops.push({bx:a.x,x:a.x,y:a.y+6,vy:40,type,timer:type==='G'?9:7,seed:rand(0,TAU),rest:false});}
  if(grid.aliveN===0&&grid.pending<=0){
    grid.pending=1.7; const bonus=100*wave;
    addScore(bonus,W/2,H*0.4,null); scoreEl.classList.remove('pop');void scoreEl.offsetWidth;scoreEl.classList.add('pop');
    showBanner('SECTOR '+pad2(wave)+' CLEARED','BONUS +'+bonus); sWave();
  }
}
function damagePlayer(){
  if(state!=='play'||!player.alive||player.invuln>0)return;
  if(powers.s>0){powers.s=0;rings.push({x:player.x,y:player.y,r:8,max:60,life:.4,maxL:.4,col:'126,240,255'});
    sShieldHit(); shake=Math.max(shake,7); player.invuln=1.3; return;}
  lives--; player.alive=false;
  puff(player.x,player.y,26,'#ffb45e',260,.8,3.4); puff(player.x,player.y,12,'#ffffff',340,.5,2.4);
  puff(player.x,player.y,9,'rgba(90,95,115,1)',90,1,6,false);
  shake=15; sBoom();
  for(const b of bombs)puff(b.x,b.y,5,'#ffb45e',90,.35,2.4);
  bombs.length=0;
  for(const d of debris)puff(d.x,d.y,4,'#7f9c86',70,.4,2.4,false);
  debris.length=0;
  powers.d=0; powers.s=0;
  if(lives<=0)gameOver('destroyed'); else player.respawnT=1.3;
}
function gravityBurst(x,y){
  rings.push({x,y,r:10,max:H*0.8,life:.7,maxL:.7,col:'192,132,252'});
  shake=Math.max(shake,12); sBurst();
  for(const b of bombs){addScore(15,b.x,b.y,'#ffd166');puff(b.x,b.y,8,'#ffb45e',180,.5,3);}
  bombs.length=0;

  debris.length=0;
}

/* ================= UPDATE ================= */
function update(dt){
  tG+=dt;
  updateBG(dt);
  for(let i=parts.length-1;i>=0;i--){const p=parts[i];p.vy+=(p.grav||0)*dt;p.x+=p.vx*dt;p.y+=p.vy*dt;p.life-=dt;if(p.life<=0)parts.splice(i,1);}
  for(let i=texts.length-1;i>=0;i--){const t=texts[i];t.y-=30*dt;t.life-=dt;if(t.life<=0)texts.splice(i,1);}
  for(let i=rings.length-1;i>=0;i--){const r=rings[i];r.r+=(H*1.6)*dt*(1-r.r/r.max*0.4);r.life-=dt;if(r.life<=0)rings.splice(i,1);}
  if(shake>0)shake=Math.max(0,shake-dt*32);

  if(state==='menu'){menuSprT-=dt;if(menuSprT<=0){menuSprT=.45;menuSprFrame^=1;}return;}
  if(state==='paused'||state==='over')return;

  /* --- powers --- */
  if(powers.d>0)powers.d-=dt; if(powers.s>0)powers.s-=dt;

  /* --- player --- */
  if(!player.alive){
    player.respawnT-=dt;
    if(player.respawnT<=0&&lives>0){player.alive=true;player.x=W/2;player.y=H-58;player.vx=0;player.vy=0;player.invuln=2.4;}
  }else{
    let ix=((keys.ArrowRight||keys.KeyD)?1:0)-((keys.ArrowLeft||keys.KeyA)?1:0);
    let iy=((keys.ArrowDown||keys.KeyS)?1:0)-((keys.ArrowUp||keys.KeyW)?1:0);
    if(pointer.down){
      const tx=clamp(pointer.x,20,W-20),ty=clamp(pointer.y,H*0.46,rimY-26);
      const dx=tx-player.x,dy=ty-player.y,dist=Math.hypot(dx,dy);
      ix=dist>6?Math.sign(dx):0; iy=dist>6?Math.sign(dy):0;
    }
    const k=Math.min(1,dt*10);
    player.vx+=(ix*440-player.vx)*k; player.vy+=(iy*330-player.vy)*k;
    player.x=clamp(player.x+player.vx*dt,20,W-20);
    player.y=clamp(player.y+player.vy*dt,H*0.46,rimY-26);
    if(player.invuln>0)player.invuln-=dt;
    player.cd-=dt;
    const firing=keys.Space||pointer.down;
    if(firing&&player.cd<=0){
      player.cd=powers.d>0?0.24:0.3;
      if(powers.d>0){shots.push({x:player.x-7,y:player.y-16,vy:-H*1.38,t:0});shots.push({x:player.x+7,y:player.y-16,vy:-H*1.38,t:0});}
      else shots.push({x:player.x,y:player.y-16,vy:-H*1.38,t:0});
      puff(player.x,player.y-20,3,'#bfe9ff',70,.15,1.8); sShoot();
    }
  }

  /* --- alien grid --- */
  if(grid.freeze>0){grid.freeze-=dt;}
  else{
    const base=grid.unit*(0.5+Math.min(wave,12)*0.05);
    const lf=1+(1-grid.aliveN/Math.max(1,grid.total))*2.4;
    const spd=Math.min(base*lf,grid.unit*3);
    grid.ox+=grid.dir*spd*dt;
    const gw=11*grid.unit, minOx=12, maxOx=W-12-gw;
    if(grid.ox>=maxOx){grid.ox=maxOx;if(grid.dir>0)flipGrid();}
    else if(grid.ox<=minOx){grid.ox=minOx;if(grid.dir<0)flipGrid();}
    grid.animT+=dt;
    grid.fireT-=dt;
    const maxB=Math.min(2+wave,9);
    if(grid.aliveN>0&&grid.fireT<=0&&bombs.length<maxB){
      grid.fireT=Math.max(0.32,1.05-wave*0.07)*rand(.7,1.3);
      const cols=[]; for(let c=0;c<11;c++){for(const a of aliens)if(a.alive&&a.c===c){cols.push(c);break;}}
      if(cols.length){const cc=cols[(Math.random()*cols.length)|0];let pick=null;
        for(const a of aliens)if(a.alive&&a.c===cc&&(!pick||a.r>pick.r))pick=a;
        if(pick)bombs.push({x:pick.x,y:pick.y+12,vy:H*(0.4+Math.min(wave,10)*0.015),seed:rand(0,TAU)});}
    }
  }
  grid.oyCur+=(grid.oy-grid.oyCur)*Math.min(1,dt*7);
  let maxBottom=0;
  for(const a of aliens)if(a.alive){placeAlien(a);const b=a.y+a.hh;if(b>maxBottom)maxBottom=b;}
  if(grid.aliveN>0&&maxBottom>doomY){gameOver('breach');return;}

  /* --- shots (gravity arcs) --- */
  const G=H*1.05;
  for(let i=shots.length-1;i>=0;i--){const s=shots[i];s.t+=dt;s.vy+=G*dt;s.y+=s.vy*dt;
    let dead=false;
    if(s.y>=killY||s.t>8){puff(s.x,Math.min(s.y,killY),3,'#ffd9a0',60,.25,1.8);dead=true;}
    if(!dead)for(const a of aliens){if(!a.alive)continue;
      if(Math.abs(s.x-a.x)<a.hw+2&&Math.abs(s.y-a.y)<a.hh+8){killAlien(a);dead=true;break;}}
    if(!dead)for(let j=bombs.length-1;j>=0;j--){const b=bombs[j];
      if(Math.abs(s.x-b.x)<7&&Math.abs(s.y-b.y)<9){addScore(25,b.x,b.y,'#ffd166');puff(b.x,b.y,6,'#ffb45e',120,.4,2.4);bombs.splice(j,1);dead=true;break;}}
    if(!dead)for(let j=debris.length-1;j>=0;j--){const d=debris[j];
      if(Math.abs(s.x-d.x)<d.size*0.7&&Math.abs(s.y-d.y)<d.size*0.7){addScore(5,d.x,d.y,'#bfe9c8');puff(d.x,d.y,6,'#8fae97',110,.4,2.4,false);debris.splice(j,1);dead=true;break;}}
    if(dead)shots.splice(i,1);
  }

  /* --- bombs --- */
  for(let i=bombs.length-1;i>=0;i--){const b=bombs[i];b.vy+=H*0.42*dt;b.y+=b.vy*dt;
    b.x+=Math.sin(tG*4.5+b.seed)*26*dt;
    let dead=false;
    if(b.y>=killY){puff(b.x,killY-4,8,'#ffb45e',130,.4,2.6);sBombFx();dead=true;}
    if(!dead&&player.alive&&Math.abs(b.x-player.x)<15&&Math.abs(b.y-player.y)<14){
      puff(b.x,b.y,10,'#ff9e4a',180,.45,3);damagePlayer();dead=true;}
    if(!dead){const hit=bunkerAt(b.x,b.y);if(hit){erode(hit,2);puff(b.x,b.y,7,'#ffb45e',120,.4,2.6);sBombFx();dead=true;}}
    if(dead)bombs.splice(i,1);
  }

  /* --- debris (falling corpses) --- */
  for(let i=debris.length-1;i>=0;i--){const d=debris[i];
    d.vy+=H*0.8*dt;d.x+=d.vx*dt;d.y+=d.vy*dt;d.rot+=d.vr*dt;d.life-=dt;
    let dead=d.life<=0;
    if(!dead&&d.y>killY-d.size*0.4){puff(d.x,killY-6,9,'#ff9e4a',150,.5,2.8);dead=true;}
    if(!dead&&player.alive&&Math.abs(d.x-player.x)<d.size*0.6+10&&Math.abs(d.y-player.y)<d.size*0.6+10){
      puff(d.x,d.y,8,'#8fae97',120,.4,2.6,false);damagePlayer();dead=true;}
    if(!dead){const hit=bunkerAt(d.x,d.y);if(hit){erode(hit,2);puff(d.x,d.y,5,'#8fae97',90,.35,2.2,false);dead=true;}}
    if(dead)debris.splice(i,1);
  }

  /* --- drops --- */
  for(let i=drops.length-1;i>=0;i--){const p=drops[i];
    if(!p.rest){p.vy=Math.min(p.vy+H*0.3*dt,240);p.y+=p.vy*dt;p.x=p.bx+Math.sin(tG*3+p.seed)*9;
      if(p.y>=killY-18){p.y=killY-18;p.rest=true;p.timer=2.6;}}
    else p.timer-=dt;
    let dead=p.timer<=0;
    if(!dead&&player.alive&&Math.hypot(p.x-player.x,p.y-player.y)<26){
      addScore(50,p.x,p.y-14,'#ffffff'); sPickup();
      if(p.type==='D'){powers.d=8;rings.push({x:p.x,y:p.y,r:6,max:40,life:.3,maxL:.3,col:'255,209,102'});}
      else if(p.type==='S'){powers.s=10;rings.push({x:p.x,y:p.y,r:6,max:40,life:.3,maxL:.3,col:'126,240,255'});}
      else gravityBurst(p.x,p.y);
      dead=true;}
    if(dead)drops.splice(i,1);
  }

  /* --- aliens vs player --- */
  if(player.alive)for(const a of aliens){if(!a.alive)continue;
    if(Math.abs(a.x-player.x)<a.hw+12&&Math.abs(a.y-player.y)<a.hh+9){killAlien(a);damagePlayer();break;}}

  /* --- next wave --- */
  if(grid.pending>0){grid.pending-=dt;
    if(grid.pending<=0){wave++;setupWave();showBanner('WAVE '+pad2(wave),'SECTOR '+pad2(wave)+' INBOUND');}}

  syncHUD();
}
