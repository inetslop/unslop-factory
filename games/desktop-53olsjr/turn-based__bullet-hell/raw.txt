/* ============ update ============ */
function movePlayer(dt){
  let ix=((keys.KeyD||keys.ArrowRight)?1:0)-((keys.KeyA||keys.ArrowLeft)?1:0);
  let iy=((keys.KeyS||keys.ArrowDown)?1:0)-((keys.KeyW||keys.ArrowUp)?1:0);
  const m=Math.hypot(ix,iy)||1,sp=275;
  player.vx=ix/m*sp;player.vy=iy/m*sp;
  player.x=clamp(player.x+player.vx*dt,34,W-34);
  player.y=clamp(player.y+player.vy*dt,Math.min(120,H*.3),H-30);
  const target=clamp(player.vx/sp,-1,1)*.32;
  player.tilt+=(target-player.tilt)*Math.min(1,dt*10);
  if(m>1&&Math.random()<dt*40)spark(player.x-Math.sin(player.tilt)*12,player.y+Math.cos(player.tilt)*12,rand(-15,15),rand(30,80),.3,'#ffd166',1.6);
  if(firing)tryFire();
}
function musicBeat(dt){
  if(!AC||state!=='play')return;
  const P=PHASES[boss.phase];beatAcc+=dt;
  while(beatAcc>=P.beat){beatAcc-=P.beat;beatStep=(beatStep+1)%4;
    const f=BASSF[P.bass[beatStep]];
    if(f)tone(f,f,.2,'triangle',.05);
    else noiseS(.04,.018,6000,'highpass');
  }
}
function updatePlay(dt,rdt){
  shotCd-=dt;player.inv-=dt;
  if(player.dead)return;
  movePlayer(dt);
  switch(turnPhase){
    case 'player':
      turnTimer-=dt;
      if(player.ammo<=0){endGrace+=dt;if(endGrace>.5)endPlayerTurn();}else endGrace=0;
      if(turnTimer<=0)endPlayerTurn();
      break;
    case 'telegraph':telT-=rdt;if(telT<=0)startPattern();break;
    case 'pattern':pat.emit(dt);if(pat.t>=pat.dur)endPattern();break;
    case 'stagger':stagT-=dt;if(stagT<=0)startPlayerTurn();break;
    case 'phasebreak':pbT-=rdt;if(pbT<=0){boss.swayT=rand(0,6);startPlayerTurn();}break;
    case 'winning':
      winBoomT-=rdt;
      if(winBoomT<=0){winBoomT=.18;const cols=['#ff5fa2','#ffc14d','#ff4d5e','#ffffff'];
        explosion(BOSSX()+rand(-55,55),BOSSY()+rand(-34,34),26,cols[(Math.random()*4)|0],260);tone(rand(200,700),60,.15,'sawtooth',.08);}
      overReal-=rdt;if(overReal<=0)showOver(true);
      break;
    case 'dying':overReal-=rdt;if(overReal<=0)showOver(false);break;
  }
  // player shots
  for(let i=shots.length-1;i>=0;i--){const s=shots[i];const px=s.x,py=s.y;s.x+=s.vx*dt;s.y+=s.vy*dt;
    if(!boss.dead&&segCircle(px,py,s.x,s.y,BOSSX(),BOSSY(),44)){
      shots.splice(i,1);
      const crit=s.mode==='reg'&&Math.random()<CFG.crit;
      applyBossDamage(s.mode==='surge'?CFG.dmgSurge:(crit?CFG.dmgN*2:CFG.dmgN),s.mode,crit);
    }else if(s.y<-60||s.x<-40||s.x>W+40)shots.splice(i,1);
  }
  // enemy bullets
  for(let i=bullets.length-1;i>=0;i--){const b=bullets[i];b.x+=b.vx*dt;b.y+=b.vy*dt;
    if(b.x<-34||b.x>W+34||b.y<-60||b.y>H+34){bullets.splice(i,1);continue;}
    if(!player.dead&&player.inv<=0){const dx=b.x-player.x,dy=b.y-player.y,rr=b.r+9;
      if(dx*dx+dy*dy<rr*rr){bullets.splice(i,1);hitPlayer();}}
  }
  // pickups
  for(let i=pickups.length-1;i>=0;i--){const p=pickups[i];p.t+=dt;
    if(Math.hypot(p.x-player.x,p.y-player.y)<28){collectPickup(p);pickups.splice(i,1);}}
  musicBeat(dt);
}
function updateFx(dt){
  for(let i=parts.length-1;i>=0;i--){const p=parts[i];p.life-=dt;
    if(p.life<=0){parts.splice(i,1);continue;}
    if(p.k==='s'){p.x+=p.vx*dt;p.y+=p.vy*dt;p.vx*=(1-3.2*dt);p.vy*=(1-3.2*dt);}
    else if(p.k==='r')p.r+=p.vr*dt;
  }
  for(let i=floaters.length-1;i>=0;i--){const f=floaters[i];f.t+=dt;f.y-=34*dt;if(f.t>=f.max)floaters.splice(i,1);}
  if(banner){banner.t+=dt;if(banner.t>=banner.dur)banner=null;}
  shake=Math.max(0,shake-shake*7*dt-2*dt);
  flash=Math.max(0,flash-2.4*dt);
}

function drawReticle(){
  const can=(turnPhase==='player'&&player.ammo>0)||(turnPhase==='stagger');
  const col=can?'#5ff5c8':'rgba(160,178,220,.35)';
  ctx.strokeStyle=col;ctx.lineWidth=1.5;
  ctx.beginPath();ctx.arc(mouse.x,mouse.y,9,0,TAU);ctx.stroke();
  for(let i=0;i<4;i++){const a=i*Math.PI/2;
    ctx.beginPath();ctx.moveTo(mouse.x+Math.cos(a)*12,mouse.y+Math.sin(a)*12);
    ctx.lineTo(mouse.x+Math.cos(a)*17,mouse.y+Math.sin(a)*17);ctx.stroke();}
  if(can&&shotCd>0){const cd=turnPhase==='stagger'?CFG.surgeCd:CFG.shotCd;
    ctx.strokeStyle='#ffd166';ctx.beginPath();
    ctx.arc(mouse.x,mouse.y,9,-Math.PI/2,-Math.PI/2+TAU*(1-shotCd/cd));ctx.stroke();}
  if(turnPhase==='player'){
    ctx.strokeStyle='rgba(95,245,200,.13)';ctx.lineWidth=1;ctx.setLineDash([5,8]);ctx.lineDashOffset=-t*30;
    ctx.beginPath();ctx.moveTo(player.x,player.y);ctx.lineTo(mouse.x,mouse.y);ctx.stroke();ctx.setLineDash([]);
  }
}
function drawBanner(){
  if(!banner)return;
  const p=banner.t/banner.dur;
  const a=p<.15?p/.15:p>.72?(1-p)/.28:1;
  const size=clamp(W*.05,30,48);
  ctx.textAlign='center';ctx.font=fD(size);
  const y=H*.42,s=1+(1-Math.min(1,banner.t/.16))*.12;
  ctx.save();ctx.translate(W/2,y);ctx.scale(s,s);
  ctx.globalAlpha=a*.35;ctx.fillStyle='#ff5fa2';ctx.fillText(banner.txt,-3,0);
  ctx.fillStyle='#7dd3fc';ctx.fillText(banner.txt,3,0);
  ctx.globalAlpha=a;ctx.fillStyle=banner.col;ctx.fillText(banner.txt,0,0);
  if(banner.sub){ctx.font=fM(12,700);ctx.fillStyle='#eaf2ff';ctx.globalAlpha=a*.85;ctx.fillText(banner.sub.toUpperCase(),0,26);}
  ctx.restore();ctx.globalAlpha=1;
}
function render(){
  ctx.setTransform(dpr,0,0,dpr,0,0);
  const sx=(Math.random()-.5)*shake,sy=(Math.random()-.5)*shake;
  ctx.save();ctx.translate(sx,sy);

}
