function makePiece(type){const p=PIECES[type];
  const cells=p.cells.map(c=>c.slice());

function lockPiece(){
  let topOut=false;
  for(const c of cur.cells){const by=cur.y+c[1];
    if(by<0){topOut=true;continue;}board[by][cur.x+c[0]]=cur.type;}
  SFX.lock();
  for(const c of cur.cells){const by=cur.y+c[1];if(by>=0)
    burst(WELL_X+(cur.x+c[0]+0.5)*CELL,WELL_Y+(by+1)*CELL,'#9fd8ff',2,40);}
  if(topOut){gameOver();return;}
  const full=[];for(let r=0;r<ROWS;r++){let f=true;for(let c=0;c<COLS;c++)if(!board[r][c]){f=false;break;}if(f)full.push(r);}
  if(full.length){
    for(const r of full)for(let c=0;c<COLS;c++){const t=board[r][c];
      burst(WELL_X+(c+0.5)*CELL,WELL_Y+(r+0.5)*CELL,PIECES[t].color,5,120);}
    for(const r of full)board.splice(r,1);
    while(board.length<ROWS)board.unshift(Array(COLS).fill(0));
    const n=full.length;
    const pts=[0,100,300,500,800][n]*level;score+=pts;lines+=n;
    borderFlash=1;shake=Math.max(shake,6+4*n);SFX.clear(n);
    addText(WELL_X+WELL_W/2,WELL_Y+WELL_H*0.35,'+'+pts,'#aef4ff',22);

  }
  canHold=true;
  cur=makePiece(queue.shift());queue.push(bagPop());
  if(collides(cur.x,cur.y,cur.cells))gameOver();
}

      if(!hit||d<hit.d){hit={t:'blk',col,row:r,d};hitY=WELL_Y+(r+1)*CELL;}break;}}}
  beams.push({x,y0,y1:hit?hitY:0,life:0.09});

/* ---------- update ---------- */
function update(dt){
  elapsed+=dt;
  if(borderFlash>0)borderFlash=Math.max(0,borderFlash-dt*2.2);
  if(shake>0)shake*=Math.exp(-6*dt);if(shake<0.15)shake=0;
  if(banner){banner.t+=dt;if(banner.t>banner.life)banner=null;}
  for(let i=beams.length-1;i>=0;i--){beams[i].life-=dt;if(beams[i].life<=0)beams.splice(i,1);}
  const dr=Math.exp(-3*dt);
  for(let i=particles.length-1;i>=0;i--){const p=particles[i];p.t+=dt;
    if(p.t>p.life){particles.splice(i,1);continue;}
    p.x+=p.vx*dt;p.y+=p.vy*dt;p.vx*=dr;p.vy*=dr;}
  for(let i=texts.length-1;i>=0;i--){const t=texts[i];t.t+=dt;t.y-=32*dt;if(t.t>t.life)texts.splice(i,1);}
  updateAsteroids(dt);

  if(state!=='playing')return;

  // ship
  const dir=(keys.KeyD?1:0)-(keys.KeyA?1:0);
  const tv=dir*560;ship.vx+=(tv-ship.vx)*Math.min(1,dt*8);
  ship.x+=ship.vx*dt;
  if(ship.x<34){ship.x=34;ship.vx=0;}if(ship.x>DW-34){ship.x=DW-34;ship.vx=0;}
  if(dir!==0&&Math.random()<dt*60)particles.push({x:ship.x-dir*10,y:SHIP_Y+12,vx:-dir*(30+Math.random()*50),vy:40+Math.random()*60,t:0,life:0.25,c:'#ff9a4d',r:2,k:'dot'});

  // laser energy
  if(laserCd>0)laserCd-=dt;
  if(energy<3){energyT+=dt;if(energyT>=4){energyT=0;energy++;energyFlash=0.5;SFX.charge();}}
  if(energyFlash>0)energyFlash-=dt;

  // piece DAS/ARR movement
  if(dasDir){dasT-=dt;while(dasT<=0&&state==='playing'){move(dasDir);dasT+=0.05;}}

  // gravity
  const iv=keys.ArrowDown?0.045:Math.max(0.11,0.8*Math.pow(0.86,level-1));
  dropAcc+=dt;
  while(dropAcc>=iv&&state==='playing'){dropAcc-=iv;
    if(!collides(cur.x,cur.y+1,cur.cells)){cur.y++;if(keys.ArrowDown)score+=1;}
    else{lockPiece();break;}}

  updateMeteor(dt);
}
