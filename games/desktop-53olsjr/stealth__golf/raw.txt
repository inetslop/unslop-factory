function segRect(x1,y1,x2,y2,r){ // Liang-Barsky
  const dx=x2-x1,dy=y2-y1,p=[-dx,dx,-dy,dy],q=[x1-r[0],r[0]+r[2]-x1,y1-r[1],r[1]+r[3]-y1];

      else{if(t<t0)return false;if(t<t1)t1=t;}}
  }
  return true;
}

function collideObj(o,w){
  const cx=clamp(o.x,w[0],w[0]+w[2]),cy=clamp(o.y,w[1],w[1]+w[3]);
  let dx=o.x-cx,dy=o.y-cy;const d2=dx*dx+dy*dy;
  if(d2>R*R)return 0;
  if(d2<1e-6){
    const l=o.x-w[0],r=w[0]+w[2]-o.x,t=o.y-w[1],b=w[1]+w[3]-o.y,m=Math.min(l,r,t,b);
    if(m===l){o.x=w[0]-R;o.vx=-Math.abs(o.vx)*REST;}
    else if(m===r){o.x=w[0]+w[2]+R;o.vx=Math.abs(o.vx)*REST;}
    else if(m===t){o.y=w[1]-R;o.vy=-Math.abs(o.vy)*REST;}
    else{o.y=w[1]+w[3]+R;o.vy=Math.abs(o.vy)*REST;}
    return Math.hypot(o.vx,o.vy);
  }
  const d=Math.sqrt(d2),nx=dx/d,ny=dy/d;
  o.x=cx+nx*R;o.y=cy+ny*R;
  const dot=o.vx*nx+o.vy*ny;
  if(dot<0){o.vx-=(1+REST)*dot*nx;o.vy-=(1+REST)*dot*ny;return -dot;}
  return 0;
}
function solids(){return[[0,0,W,16],[0,H-16,W,16],[0,0,16,H],[W-16,0,16,H]].concat(cur.walls);}

function physics(dt){
  const px=ball.x,py=ball.y;
  hidden=inRects(ball.x,ball.y,cur.bush);
  const dxC=cup.x-ball.x,dyC=cup.y-ball.y,dC=Math.hypot(dxC,dyC);
  if(dC<36&&dC>1){ball.vx+=dxC/dC*340*dt;ball.vy+=dyC/dC*340*dt;}
  const k=zoneK(ball.x,ball.y),f=Math.exp(-k*dt);
  ball.vx*=f;ball.vy*=f;
  ball.x+=ball.vx*dt;ball.y+=ball.vy*dt;
  let impact=0;
  for(const w of cur._solids)impact=Math.max(impact,collideObj(ball,w));
  if(impact>70){S.bounce(impact);shake=Math.min(8,shake+impact/260);}
  const sp=Math.hypot(ball.vx,ball.vy);
  ball.rot+=sp*dt*.05;
  if(sp>40){trail.push({x:ball.x,y:ball.y});if(trail.length>20)trail.shift();}
  // cup capture
  const dC2=Math.hypot(cup.x-ball.x,cup.y-ball.y);
  if(dC2<CUPR-2&&sp<130){startSink();return;}
  // water
  if(inRects(ball.x,ball.y,cur.water)){splashBack(px,py);return;}
  // lasers
  for(const lz of cur._l){
    if(lz.cd>0)lz.cd-=dt;
    if(laserOn(lz)&&lz.cd<=0&&sp>25){
      const hit=segSeg(px,py,ball.x,ball.y,lz.a[0],lz.a[1],lz.b[0],lz.b[1]);
      if(hit){alertLvl=Math.min(1,alertLvl+.4);lz.flash=1;lz.cd=.9;S.laser();sparks(hit.x,hit.y);
        if(alertLvl>=1){bust();return;}}
    }
  }
  // stopped?
  if(sp<5){
    ball.vx=ball.vy=0;restPos={x:ball.x,y:ball.y};hidden=inRects(ball.x,ball.y,cur.bush);mode='aim';
    if(strokes>cur.limit)cutSeq();
  }
}

function updGuards(dt){
  for(const g of cur._g){
    const wp=g.p[g.i];let dx=wp[0]-g.x,dy=wp[1]-g.y;
    const d=Math.hypot(dx,dy),step=g.sp*dt;
    if(d<=step){g.x=wp[0];g.y=wp[1];g.i=(g.i+1)%g.p.length;}
    else{g.x+=dx/d*step;g.y+=dy/d*step;g.dir=lerpAng(g.dir,Math.atan2(dy,dx),Math.min(1,dt*7));}
    g.bob+=dt*g.sp*.25;
  }
}
function detect(dt){
  if(mode!=='aim'&&mode!=='rolling')return;
  hidden=inRects(ball.x,ball.y,cur.bush);
  let seen=false;
  if(!hidden){
    for(const g of cur._g){
      const dx=ball.x-g.x,dy=ball.y-g.y,d=Math.hypot(dx,dy);
      if(d>g.rng+R)continue;
      let da=Math.atan2(dy,dx)-g.dir;
      while(da>Math.PI)da-=2*Math.PI;while(da<-Math.PI)da+=2*Math.PI;
      if(Math.abs(da)>g.h)continue;
      if(losBlocked(g.x,g.y,ball.x,ball.y))continue;
      seen=true;break;
    }
  }
  if(seen){
    alertLvl=Math.min(1,alertLvl+dt*1.0);
    tickT-=dt;
    if(tickT<=0&&alertLvl>.12){tone(480+alertLvl*620,460+alertLvl*560,'sine',.05,.15);tickT=.16;}
  }else alertLvl=Math.max(0,alertLvl-dt*.45);
  if(alertLvl>=1&&mode!=='busted')bust();
}

function updParticles(dt){
  for(let i=particles.length-1;i>=0;i--){const p=particles[i];p.life-=dt;
    if(p.life<=0){particles.splice(i,1);continue;}
    if(p.ring)p.r+=p.vr*dt;else{p.vy+=(p.g||0)*dt;p.x+=p.vx*dt;p.y+=p.vy*dt;}}
}

/* ============================== UPDATE ============================== */
function update(dt){
  time+=dt;
  for(let i=pending.length-1;i>=0;i--)if(time>=pending[i].at){const f=pending[i].fn;pending.splice(i,1);f();}
  shake=Math.max(0,shake-shake*7*dt-4*dt);
  flash=Math.max(0,flash-2.2*dt);
  if(['intro','aim','rolling','sinking'].includes(mode))updGuards(dt);
  for(const l of cur._l)l.flash=Math.max(0,l.flash-3*dt);
  updParticles(dt);
  if(mode==='rolling')physics(dt);
  else if(mode==='sinking')sinkStep(dt);
  if(['aim','rolling'].includes(mode))detect(dt);
  else if(!['busted','paused'].includes(mode))alertLvl=Math.max(0,alertLvl-dt*.45);
}

  }
}
