```js
function collideBrick(b){let best=null,bd=1e9;
 for(const k of bricks){if(!k.alive)continue;
  const cx=Math.max(k.x,Math.min(b.x,k.x+k.w)), cy=Math.max(k.y,Math.min(b.y,k.y+k.h));
  const dx=b.x-cx, dy=b.y-cy, d2=dx*dx+dy*dy;
  if(d2<=b.r*b.r && d2<bd){ // also require ball center within margin? standard
    best=k;bd=d2; }
 }
 if(!best)return null;
 const cx=Math.max(best.x,Math.min(b.x,best.x+best.w)), cy=...;
 let dx=b.x-cx, dy=b.y-cy;
 if(dx===0&&dy===0){ // center inside: push out along min axis
   const l=b.x-best.x, r2=best.x+best.w-b.x, t=b.y-best.y, bo=best.y+best.h-b.y;
   const m=Math.min(l,r2,t,bo);
   if(m===l)b.vx=-Math.abs(b.vx)||-1; else if(m===r2)b.vx=Math.abs(b.vx);
   else if(m===t)b.vy=-Math.abs(b.vy)||-1; else b.vy=Math.abs(b.vy);
 } else if(Math.abs(dx)>Math.abs(dy)){ b.vx = dx>0? Math.abs(b.vx): -Math.abs(b.vx); }
 else { b.vy = dy>0? Math.abs(b.vy): -Math.abs(b.vy); }
 // nudge out
 return best;
}
```

```html
  if(dx===0&&dy===0){const l=b.x-best.x,r2=best.x+best.w-b.x,t=b.y-best.y,bo=best.y+best.h-b.y,m=Math.min(l,r2,t,bo);
    if(m===l)b.vx=-Math.abs(b.vx)||-1;else if(m===r2)b.vx=Math.abs(b.vx)||1;

  else b.vy=dy>0?Math.abs(b.vy):-Math.abs(b.vy);
  return best;}

function stepBalls(dt){const ts=slowT>0?.45:1;
  for(let i=balls.length-1;i>=0;i--){const b=balls[i];let alive=true;
    const steps=Math.max(1,Math.ceil(b.sp*ts*dt/5));
    for(let s=0;s<steps&&alive;s++){b.x+=b.vx*ts*dt/steps;b.y+=b.vy*ts*dt/steps;
      if(b.x-b.r<PLAY.x){b.x=PLAY.x+b.r;b.vx=Math.abs(b.vx);wallTick();}
      else if(b.x+b.r>PLAY.x+PLAY.w){b.x=PLAY.x+PLAY.w-b.r;b.vx=-Math.abs(b.vx);wallTick();}
      if(b.y-b.r<PLAY.y){b.y=PLAY.y+b.r;b.vy=Math.abs(b.vy);wallTick();}
      if(demoMode()){if(b.y+b.r>H-64){b.y=H-64-b.r;b.vy=-Math.abs(b.vy);}}
      else if(b.y-b.r>H+20)alive=false;
      const hit=collideBrick(b);if(hit&&!demoMode())damageBrick(hit,b,false);}
    b.tintT=Math.max(0,b.tintT-dt);
    b.trail.unshift({x:b.x,y:b.y});if(b.trail.length>9)b.trail.pop();
    if(!alive){balls.splice(i,1);onBallLost();}}}

function stepDrops(dt){const ts=slowT>0?.45:1;
```
