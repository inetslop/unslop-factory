function getSpr(name,frame,px,color){
  const k=name+frame+'_'+px+'_'+color;if(sprCache[k])return sprCache[k];
  const rows=SPR[name][frame],w=Math.max.apply(null,rows.map(r=>r.length)),h=rows.length;
  const c=document.createElement('canvas');c.width=w*px;c.height=h*px;
  const g=c.getContext('2d');g.fillStyle=color;
  for(let y=0;y<h;y++)for(let x=0;x<w;x++)if(rows[y][x]==='#')g.fillRect(x*px,y*px,px,px);
  sprCache[k]=c;return c;
}

/* ================= board ================= */
function computeAdj(cells,rows,cols){
  for(let i=0;i<cells.length;i++)cells[i].adj=0;
  for(let r=0;r<rows;r++)for(let c=0;c<cols;c++){
    if(cells[r*cols+c].t!=='M')continue;
    for(let dr=-1;dr<=1;dr++)for(let dc=-1;dc<=1;dc++){
      const nr=r+dr,nc=c+dc;if(nr<0||nc<0||nr>=rows||nc>=cols)continue;
      cells[nr*cols+nc].adj++;
    }
  }
}
function reseedSafe(bad){
  const b=board,total=b.rows*b.cols,order=shuffle([...Array(total).keys()]);
  for(const c of b.cells)c.t='E';
  let m=0,a=0;
  for(const i of order){if(i===bad)continue;
    if(m<b.minesN){b.cells[i].t='M';m++;}else if(a<b.aliensN){b.cells[i].t='A';a++;}}
  computeAdj(b.cells,b.rows,b.cols);
}
function setupWave(){
  const rows=Math.min(4+wave,7);
  let cell=fitCell(9,rows);
  let cols=clamp(Math.floor((W-28)/cell),9,12);
  cell=Math.min(cell,fitCell(cols,rows));
  const total=rows*cols;
  const minesN=clamp(Math.round(total*.2)+(wave>1?wave-1:0),4,Math.floor(total*.36));
  const aliensN=clamp(Math.round(total*.38),5,total-minesN-2);
  const cells=new Array(total);
  for(let i=0;i<total;i++)cells[i]={t:'E',rev:false,flag:false,adj:0,revealAt:0,mineShown:false,wasA:false};
  const order=shuffle([...Array(total).keys()]);
  let m=0,a=0;
  for(const i of order){if(m<minesN){cells[i].t='M';m++;}else if(a<aliensN){cells[i].t='A';a++;}}
  computeAdj(cells,rows,cols);
  const bands=[];for(let r=0;r<rows;r++)bands.push(Math.min(2,(r*3/rows)|0));
  board={rows,cols,cell,fw:cols*cell,fx:(W-cols*cell)/2,formY:-rows*cell-10,homeY:78,
    entering:true,speed:0,cells,minesN,flags:0,aliensAlive:aliensN,bands,px:sprPx(cell),
    pxShip:Math.max(2,(cell*0.9/13)|0)};
  board.speed=Math.max(.5,(dangerY()-board.homeY)/Math.max(26,54-(wave-1)*5));
  firstShot=false;bolts.length=0;lasers.length=0;delayed.length=0;pops.length=0;rings.length=0;
  afT=rand(2.5,4);beatT=0;frame=0;hop=0;invuln=Math.min(invuln,.5);
  ship.tc=(cols/2)|0;ship.x=colCenter(ship.tc);
  state='play';
  banner('WAVE '+pad(wave,2),wave===1?'DEPLOY YOUR CANNON':'THE FLEET RETURNS',1700);
}

/* ================= gameplay ================= */
function beatInterval(){
  const b=board,base=clamp(.6-wave*.045,.3,.6);
  const p=clamp((b.formY-b.homeY)/(dangerY()-b.homeY),0,1);
  return lerp(base,base*.42,p);
}
function beatTick(interval){
  beatT+=1/60; // called with dt below
}
function simPlay(dt){
  const b=board;
  fireCd=Math.max(0,fireCd-dt);invuln=Math.max(0,invuln-dt);
  beatT+=dt;const bi=beatInterval();
  while(beatT>=bi){beatT-=bi;frame^=1;hop=1;sfx.beat(frame===1);}
  if(b.entering){b.formY+=520*dt;if(b.formY>=b.homeY){b.formY=b.homeY;b.entering=false;}}
  else{b.formY+=b.speed*dt;
    if(b.formY+b.rows*b.cell>=dangerY()){startDying('invasion');return;}}
  const dir=(keys.has('right')?1:0)-(keys.has('left')?1:0);
  if(dir){mvT+=dt;while(mvT>=.105){mvT-=.105;ship.tc=clamp(ship.tc+dir,0,b.cols-1);}}else mvT=0;
  ship.x+=(colCenter(ship.tc)-ship.x)*Math.min(1,dt*16);
  if(fireHeld)fire();
  if(wave>=2){afT-=dt;
    if(afT<=0&&bolts.length<Math.min(2+(wave>>1),4)){spawnBolt();afT=rand(3.4,6.2)*Math.max(.5,1-wave*.06);}}
  for(let i=bolts.length-1;i>=0;i--){const bl=bolts[i];
    bl.t+=dt;bl.y+=(150+wave*16)*dt;bl.x=bl.x0+Math.sin(bl.t*5+bl.ph)*16;
    if(bl.y>playerY()-10&&invuln<=0&&Math.abs(bl.x-ship.x)<b.cell*.42){bolts.splice(i,1);hitPlayer();continue;}
    if(bl.y>H+30)bolts.splice(i,1);}
  fxUpdate(dt);
  const now=performance.now();
  for(let i=delayed.length-1;i>=0;i--)if(now>=delayed[i].at){const f=delayed[i].fn;delayed.splice(i,1);f();}
}
function spawnBolt(){
  const b=board,alive=[];
  for(let i=0;i<b.cells.length;i++)if(b.cells[i].t==='A'&&!b.cells[i].rev)alive.push(i);
  if(!alive.length)return;
  const i=alive[(Math.random()*alive.length)|0],r=(i/b.cols)|0,c=i%b.cols;
  bolts.push({x0:colCenter(c),x:colCenter(c),y:b.formY+r*b.cell+b.cell*.75,t:0,ph:rand(0,6.28)});
  sfx.bolt();
}
function fire(){
  if(state!=='play'||paused||fireCd>0)return;
  const b=board,col=ship.tc,x=colCenter(col);
  let r=-1,blocked=false;
  for(let i=b.rows-1;i>=0;i--){const c=b.cells[i*b.cols+col];
    if(c.rev)continue;if(!flagMode&&c.flag)blocked=true;r=i;break;}
  if(r<0){sfx.fizz();lasers.push({x,y0:playerY()-16,y1:b.formY-14,t:0});return;}
  fireCd=.16;
  const cy=b.formY+r*b.cell+b.cell/2;
  lasers.push({x,y0:playerY()-16,y1:cy,t:0});
  if(flagMode){const c=b.cells[r*b.cols+col];
    if(c.flag){c.flag=false;b.flags--;sfx.low();}
    else{c.flag=true;b.flags++;sfx.flag();
      for(let i=0;i<6;i++)parts.push({x:cx2(x,cy),y:cy,vx:rand(-40,40),vy:rand(-60,10),g:0,life:.5,max:.5,s:2,col:'#ffb84d'});}
    syncHud();return;}
  if(blocked){popTxt(x,cy,'BLOCKED','#ffb84d');sfx.block();return;}
  const idx=r*b.cols+col;
  if(!firstShot){firstShot=true;if(b.cells[idx].t==='M')reseedSafe(idx);}
  const c=b.cells[idx];
  if(c.t==='M')hitMine(idx);
  else if(c.t==='A')killAlien(idx);
  else revealEmpty(idx,0);
}
const cx2=(a,b)=>a; // (kept simple: x passed where needed)
function hitMine(i){
  const b=board,c=b.cells[i],r=(i/b.cols)|0,cc=i%b.cols;
  const x=colCenter(cc),y=b.formY+r*b.cell+b.cell/2;
  c.rev=true;c.mineShown=true;c.revealAt=performance.now();
  sfx.boom();shake=16;flash=.7;spawnBoom(x,y,1);rings.push({x,y,r:4,vr:360,a:1,col:'255,140,80'});
  for(let dr=-1;dr<=1;dr++)for(let dc=-1;dc<=1;dc++){
    const nr=r+dr,ncc=cc+dc;if(dr===0&&dc===0)continue;
    if(nr<0||ncc<0||nr>=b.rows||ncc>=b.cols)continue;
    const n=b.cells[nr*b.cols+ncc];
    if(n.t==='M'&&!n.rev){const ni=nr*b.cols+ncc;
      delayed.push({at:performance.now()+140+Math.random()*220,fn:()=>{
        n.rev=true;n.mineShown=true;n.revealAt=performance.now();
        const nx=colCenter(ncc),ny=b.formY+nr*b.cell+b.cell/2;
        spawnBoom(nx,ny,.5);sfx.boomS();rings.push({x:nx,y:ny,r:3,vr:220,a:.8,col:'255,140,80'});}});}
  }
  damage(1);
}
function killAlien(i){
  const b=board,c=b.cells[i],r=(i/b.cols)|0,cc=i%b.cols;
  const x=colCenter(cc),y=b.formY+r*b.cell+b.cell/2;
  c.rev=true;c.wasA=true;
  const pts=30+wave*10;score+=pts;kills++;
  sfx.kill();spawnDebris(x,y,BCOL[b.bands[r]]);popTxt(x,y-6,'+'+pts,'#7dff9e');
  if(--b.aliensAlive<=0)waveClear();
}
function revealEmpty(i,d){
  const b=board,c=b.cells[i];
  c.rev=true;c.revealAt=performance.now()+d*45;
  if(c.adj!==0)return;
  const r=(i/b.cols)|0,cc=i%b.cols;
  for(let dr=-1;dr<=1;dr++)for(let dc=-1;dc<=1;dc++){
    const nr=r+dr,ncc=cc+dc;if(dr===0&&dc===0)continue;
    if(nr<0||ncc<0||nr>=b.rows||ncc>=b.cols)continue;
    const n=b.cells[nr*b.cols+ncc];
    if(!n.rev&&n.t==='E')revealEmpty(nr*b.cols+ncc,d+1);
  }
}
function waveClear(){
  state='clear';clearT=0;bolts.length=0;
  let cf=0;for(const c of board.cells)if(c.flag&&c.t==='M')cf++;
  const bonus=shields*150+cf*75;score+=bonus;
  banner('WAVE '+pad(wave,2)+' CLEAR','BONUS +'+pad(bonus,4),1900);
  sfx.jingle();
}
function damage(){
  if(invuln>0)return;
  shields--;invuln=1.6;renderShields();syncHud();
  if(shields<=0)startDying('shields');
}
function hitPlayer(){
  shields--;invuln=1.6;shake=12;flash=.5;sfx.zap();
  spawnBoom(ship.x,playerY(),.8);renderShields();syncHud();
  if(shields<=0)startDying('shields');
}
function startDying(why){
  state='dying';whyDying=why;dieT=0;bolts.length=0;
  sfx.boom();
  if(why==='shields'){shake=14;flash=.6;spawnBoom(ship.x,playerY(),1.6);
    rings.push({x:ship.x,y:playerY(),r:6,vr:420,a:1,col:'120,220,255'});}
  else{shake=20;flash=1;spawnBoom(W/2,dangerY(),2.2);
    rings.push({x:W/2,y:dangerY(),r:10,vr:560,a:1,col:'255,90,70'});}
}
function finishGameOver(){
  state='over';
  const nb=score>best;if(nb){best=score;saveBest(best);}
  $('overTitle').textContent=whyDying==='invasion'?'FLEET INVADED':'SHIELDS DOWN';
  setT($('ovScore'),pad(score));setT($('ovBest'),pad(best));
  setT($('ovWave'),String(wave).padStart(2,'0'));setT($('ovKills'),String(kills).padStart(3,'0'));
  show($('newRec'),nb);$('startBest').textContent=pad(best);
  show(overScr,true);sfx.over();
  if(nb)for(let i=0;i<130;i++)parts.push({x:W/2,y:H*.34,vx:rand(-280,280),vy:rand(-320,90),g:430,
    life:rand(.9,1.8),max:1.8,s:rand(2,4),col:['#ffb84d','#7dff9e','#59d8ff','#ffffff'][i%4]});
}

/* ================= tick ================= */
function fxUpdate(dt){
  shake=Math.max(0,shake-dt*34);flash=Math.max(0,flash-dt*1.5);hop*=Math.exp(-5*dt);
  for(let i=lasers.length-1;i>=0;i--){lasers[i].t+=dt;if(lasers[i].t>.14)lasers.splice(i,1);}
  for(let i=parts.length-1;i>=0;i--){const p=parts[i];
    p.x+=p.vx*dt;p.y+=p.vy*dt;p.vy+=(p.g||0)*dt;p.life-=dt;if(p.life<=0)parts.splice(i,1);}
  for(let i=rings.length-1;i>=0;i--){const r=rings[i];r.r+=r.vr*dt;r.a-=dt*1.8;if(r.a<=0)rings.splice(i,1);}
  for(let i=pops.length-1;i>=0;i--){const p=pops[i];p.y-=30*dt;p.life-=dt;if(p.life<=0)pops.splice(i,1);}
}
function tick(dt){
  gt+=dt;
  for(const s of stars){s.y+=dt*6*s.z;if(s.y>H+2){s.y=-2;s.x=Math.random()*W;}}
  if(paused)return;
  switch(state){
    case 'menu':menuFt+=dt;if(menuFt>.42){menuFt=0;menuF^=1;}break;
    case 'play':simPlay(dt);break;
    case 'clear':{clearT+=dt;beatT+=dt;if(beatT>=.55){beatT=0;frame^=1;hop=1;}fxUpdate(dt);
      if(clearT>2.35){wave++;setupWave();}break;}
    case 'dying':dieT+=dt;fxUpdate(dt);if(dieT>1.15)finishGameOver();break;
    case 'over':fxUpdate(dt);break;
  }
}

const easeOutBack=t=>{const c1=1.70158,c3=c1+1;return 1+c3*Math.pow(t-1,3)+c1*Math.pow(t-1,2);};
function drawBombShape(g,x,y,s,t){
  const p=.65+.35*Math.sin(t*7);
  const rg=g.createRadialGradient(x,y,0,x,y,s*1.5);
  rg.addColorStop(0,'rgba(255,82,82,'+(0.22*p)+')');rg.addColorStop(1,'rgba(255,82,82,0)');
  g.fillStyle=rg;g.beginPath();g.arc(x,y,s*1.5,0,7);g.fill();
  g.strokeStyle='#6b7690';g.lineWidth=Math.max(1.5,s*.09);
  for(let i=0;i<8;i++){const a=i*Math.PI/4+t*.3;
    g.beginPath();g.moveTo(x+Math.cos(a)*s*.5,y+Math.sin(a)*s*.5);
    g.lineTo(x+Math.cos(a)*s*.95,y+Math.sin(a)*s*.95);g.stroke();}
  const bg=g.createRadialGradient(x-s*.2,y-s*.2,s*.1,x,y,s*.62);
  bg.addColorStop(0,'#3d4763');bg.addColorStop(1,'#0e1119');
  g.fillStyle=bg;g.beginPath();g.arc(x,y,s*.62,0,7);g.fill();
  g.fillStyle='rgba(255,82,82,'+(.55+.4*p)+')';g.beginPath();g.arc(x,y,s*.2,0,7);g.fill();
  g.fillStyle='rgba(255,255,255,.5)';g.beginPath();g.arc(x-s*.22,y-s*.26,s*.1,0,7);g.fill();
}

function drawBoard(){
  const b=board,now=performance.now();
  for(let r=0;r<b.rows;r++)for(let c=0;c<b.cols;c++){
    const i=r*b.cols+c,cell=b.cells[i],x=b.fx+c*b.cell,y=b.formY+r*b.cell;
    if(cell.rev){
      ctx.fillStyle='rgba(80,120,220,.06)';ctx.fillRect(x+3,y+3,b.cell-6,b.cell-6);
      ctx.strokeStyle='rgba(120,170,255,.13)';ctx.lineWidth=1;
      ctx.strokeRect(x+3.5,y+3.5,b.cell-7,b.cell-7);
      const cx=x+b.cell/2,cy=y+b.cell/2;
      if(cell.mineShown){
        const age=(now-cell.revealAt)/220,s=b.cell*.46*(age>=1?1:Math.max(.05,easeOutBack(clamp(age,0,1))));
        drawBombShape(ctx,cx,cy,s,gt+i);
      }else if(cell.wasA){
        ctx.strokeStyle='rgba(107,124,154,.3)';ctx.lineWidth=2;
        const k=b.cell*.16;
        ctx.beginPath();ctx.moveTo(cx-k,cy-k);ctx.lineTo(cx+k,cy+k);
        ctx.moveTo(cx+k,cy-k);ctx.lineTo(cx-k,cy+k);ctx.stroke();
      }else if(cell.adj>0&&now>=cell.revealAt){
        const age=(now-cell.revealAt)/200,sc=age>=1?1:easeOutBack(clamp(age,0,1));
        ctx.save();ctx.translate(cx,cy);ctx.scale(sc,sc);
        const col=NCOL[cell.adj]||'#fff';
        ctx.font='700 '+Math.round(b.cell*.42)+"px Consolas,Menlo,'Courier New',monospace";
        ctx.textAlign='center';ctx.textBaseline='middle';
        ctx.shadowColor=col;ctx.shadowBlur=10;ctx.fillStyle=col;
        ctx.fillText(cell.adj,0,1);ctx.shadowBlur=0;ctx.restore();
      }
    }else{
      ctx.fillStyle='rgba(60,90,180,.05)';ctx.fillRect(x+2,y+2,b.cell-4,b.cell-4);
      const spr=getSpr(BANDS[b.bands[r]],frame,b.px,BCOL[b.bands[r]]);
      const wy=Math.sin(gt*2.4+i*.7)*1.5,hy=-hop*3;
      ctx.globalAlpha=cell.flag?.3:1;
      ctx.drawImage(spr,x+(b.cell-spr.width)/2,y+(b.cell-spr.height)/2+wy+hy);
      ctx.globalAlpha=1;
      if(cell.flag)drawBeaconShape(ctx,x+b.cell/2,y+b.cell-4,b.cell*.5,gt+i);
    }
  }
}
function drawDangerLine(){

function drawShip(){
  const b=board,spr=getSpr('ship',0,b.pxShip,'#e8f1ff');
  const x=ship.x,y=playerY();
  if(invuln>0&&((gt*14)|0)%2===0)ctx.globalAlpha=.3;
  ctx.drawImage(spr,x-spr.width/2,y-spr.height/2);
  const L=6+Math.random()*9,fl=flagMode?'rgba(255,184,77,':'rgba(96,220,255,';
  ctx.fillStyle=fl+'0.85)';ctx.beginPath();
  ctx.moveTo(x-5,y+spr.height/2);ctx.lineTo(x+5,y+spr.height/2);ctx.lineTo(x,y+spr.height/2+L);
  ctx.closePath();ctx.fill();

  ctx.moveTo(x-2.5,y+spr.height/2);ctx.lineTo(x+2.5,y+spr.height/2);ctx.lineTo(x,y+spr.height/2+L*.55);

function render(){

  if(bgCv)ctx.drawImage(bgCv,0,0,W,H);else{ctx.fillStyle='#05070f';ctx.fillRect(0,0,W,H);}

  drawStars();
  if(board&&state!=='menu'){if(state==='play')drawDangerLine();drawBoard();}

    ctx.fillStyle='rgba(96,220,255,'+(.3*a).toFixed(3)+')';ctx.fillRect(l.x-3,l.y1,6,l.y0-l.y1);
    ctx.fillStyle='rgba(255,255,255,'+(.85*a).toFixed(3)+')';ctx.fillRect(l.x-1,l.y1,2,l.y0-l.y1);
    ctx.fillStyle='rgba(160,240,255,'+a.toFixed(3)+')';

  for(const bl of bolts){
    ctx.fillStyle='rgba(255,95,208,.3)';ctx.fillRect(bl.x-4,bl.y-7,8,14);
    ctx.fillStyle='#ff9fe6';ctx.fillRect(bl.x-1.5,bl.y-5,3,10);
    ctx.fillStyle='#fff';ctx.fillRect(bl.x-.5,bl.y+4,1,4);}
  for(const p of parts){ctx.globalAlpha=Math.max(0,p.life/p.max);ctx.fillStyle=p.col;
    ctx.fillRect(p.x-p.s/2,p.y-p.s/2,p.s,p.s);}
  ctx.globalAlpha=1;
  for(const r of rings){ctx.strokeStyle='rgba('+r.col+','+Math.max(0,r.a).toFixed(3)+')';
    ctx.lineWidth=3;ctx.beginPath();ctx.arc(r.x,r.y,r.r,0,7);ctx.stroke();}
  ctx.font="700 13px Consolas,Menlo,monospace";ctx.textAlign='center';ctx.textBaseline='middle';

}
