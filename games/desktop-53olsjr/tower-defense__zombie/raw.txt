      else burst(s.lx,s.ly,3,['#5b6a4e'],1.2,.07,.3); }
    else { s.x+=dx/dist*step; s.y+=dy/dist*step;
      if(s.k==='o'&&Math.random()<dt*20)G.parts.push({x:s.x,y:s.y,vx:0,vy:-.4,life:.3,max:.3,col:'#8aff5e',sz:cell*.05}); } }
  else { s.t+=dt/s.dur; if(s.t>=1){s.done=true; explode(s.tx,s.ty,s.spl,s.dmg);} } }

 // bolts / parts / texts

 G.parts=G.parts.filter(p=>p.life>0);

 // wave end
