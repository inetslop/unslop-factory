function collideRect(e,c){
 const cx=Math.max(c.x,Math.min(e.x,c.x+c.w));
 const cy=Math.max(c.y,Math.min(e.y,c.y+c.h));
 let dx=e.x-cx, dy=e.y-cy; const d2=dx*dx+dy*dy;
 if(d2>=e.r*e.r) return;
 let d=Math.sqrt(d2);
 if(d<0.0001){ // center inside: push out along smallest penetration
  const l=e.x-c.x, rgt=c.x+c.w-e.x, t=e.y-c.y, bt=c.y+c.h-e.y;
  const m=Math.min(l,rgt,t,bt);
  if(m===l)e.x=c.x-e.r; else if(m===rgt)e.x=c.x+c.w+e.r; else if(m===t)e.y=c.y-e.r; else e.y=c.y+c.h+e.r;
  return;
 }
 const push=(e.r-d)/d; e.x+=dx*push; e.y+=dy*push;
}

/* ================= COLLISIONS ================= */
function collideRect(e,c){
  const cx=clamp(e.x,c.x,c.x+c.w),cy=clamp(e.y,c.y,c.y+c.h);
  let dx=e.x-cx,dy=e.y-cy;const d2=dx*dx+dy*dy;
  if(d2>=e.r*e.r)return;
  const d=Math.sqrt(d2);
  if(d<.001){const l=e.x-c.x,rt=c.x+c.w-e.x,t=e.y-c.y,bt=c.y+c.h-e.y,mn=Math.min(l,rt,t,bt);
    if(mn===l)e.x=c.x-e.r;else if(mn===rt)e.x=c.x+c.w+e.r;else if(mn===t)e.y=c.y-e.r;else e.y=c.y+c.h+e.r;return;}
  const push=(e.r-d)/d;e.x+=dx*push;e.y+=dy*push;
}
function collideWorld(e){
  for(const c of crates)collideRect(e,c);
  for(const b of barrels){ if(b.dead)continue; let dx=e.x-b.x,dy=e.y-b.y;const rr=e.r+b.r,d2=dx*dx+dy*dy;
    if(d2<rr*rr&&d2>.001){const d=Math.sqrt(d2),push=(rr-d)/d;e.x+=dx*push;e.y+=dy*push;}}
  e.x=clamp(e.x,WM+e.r,W-WM-e.r);e.y=clamp(e.y,WM+e.r,H-WM-e.r);
}
function findNearestZ(x,y,maxD){let best=null,bd=maxD*maxD;
  for(const z of zombies){if(z.dead)continue;const d2=(z.x-x)*(z.x-x)+(z.y-y)*(z.y-y);if(d2<bd){bd=d2;best=z;}}return best;}
function nearestPlayer(x,y){let best=null,bd=1e18;
  for(const p of players){if(!p.alive)continue;const d2=(p.x-x)*(p.x-x)+(p.y-y)*(p.y-y);if(d2<bd){bd=d2;best=p;}}return best;}

function damageZombie(z,dmg,kx,ky,owner){
  if(z.dead)return;z.hp-=dmg;z.flash=1;z.x+=kx||0;z.y+=ky||0;
  blood(z.x,z.y,6,1,(kx||0)*.02,(ky||0)*.02);sfx.hit();

}
function killZombie(z,owner){
  if(z.dead)return;z.dead=true;brains+=z.val;
  if(owner!=null&&owner>=0&&players[owner]){players[owner].kills++;
    effects.push({kind:'text',t:0,dur:.9,x:z.x,y:z.y-14,txt:'+'+z.val,col:'#ffd23f'});}

function hurtPlayer(p,dmg,kx,ky){

}

function waveLogic(d){

function updatePlayers(d){
  for(const p of players){if(!p.alive)continue;
    const inp=p.i===0?in1():in2();const ml=Math.hypot(inp.x,inp.y);
    if(ml>0){p.x+=inp.x/ml*PSPEED*d;p.y+=inp.y/ml*PSPEED*d;}
    collideWorld(p);
    const t=findNearestZ(p.x,p.y,560);
    if(t)p.dir=Math.atan2(t.y-p.y,t.x-p.x);else if(ml>0)p.dir=Math.atan2(inp.y,inp.x);
    p.fireT-=d;
    if(inp.fire&&p.fireT<=0){doShoot(p);p.fireT=p.stimT>0?.085:.16;}
    if(p.stimT>0)p.stimT=Math.max(0,p.stimT-d);
    if(p.hurtT>0)p.hurtT=Math.max(0,p.hurtT-d*1.4);
    if(p.flashT>0)p.flashT-=d*7;
  }
}
function updateZombie(z,d){
  z.hitCd-=d;if(z.flash>0)z.flash-=d*6;
  let tx,ty;const t=nearestPlayer(z.x,z.y);
  if(t){tx=t.x;ty=t.y;}
  else{z.wtT-=d;const dx=z.wt.x-z.x,dy=z.wt.y-z.y;
    if(z.wtT<=0||dx*dx+dy*dy<40*40){z.wt={x:WM+40+Math.random()*(W-2*WM-80),y:WM+40+Math.random()*(H-2*WM-80)};z.wtT=2+Math.random()*3;}
    tx=z.wt.x;ty=z.wt.y;}
  const a=Math.atan2(ty-z.y,tx-z.x);
  z.ph+=d*(z.spd>100?10:6);
  const va=a+Math.sin(z.ph)*.35;
  z.x+=Math.cos(va)*z.spd*d;z.y+=Math.sin(va)*z.spd*d;
  let df=a-z.face;while(df>Math.PI)df-=TAU;while(df<-Math.PI)df+=TAU;z.face+=df*Math.min(1,d*8);
  collideWorld(z);
  for(const p of players){if(!p.alive||z.hitCd>0)continue;
    const dx=p.x-z.x,dy=p.y-z.y,rr=z.r+p.r+2,d2=dx*dx+dy*dy;
    if(d2<rr*rr){const dd=Math.sqrt(d2)||1;hurtPlayer(p,z.dmg,-dx/dd*9,-dy/dd*9);z.hitCd=.75;z.x-=dx/dd*4;z.y-=dy/dd*4;break;}}
}
function update(d,r){
  runT+=d;shake*=Math.exp(-6*r);
  moanT-=d;if(moanT<=0){moanT=2.5+Math.random()*3.5;const z=zombies[(Math.random()*zombies.length)|0];if(z)moan(((z.x/W)*2-1)*.7);}
  if(state==='playing'){waveLogic(d);updatePlayers(d);}
  for(const z of zombies)if(!z.dead)updateZombie(z,d);
  for(let i=0;i<zombies.length;i++){const A=zombies[i];if(A.dead)continue;
    for(let j=i+1;j<zombies.length;j++){const B=zombies[j];if(B.dead)continue;
      let dx=B.x-A.x,dy=B.y-A.y;const rr=A.r+B.r,d2=dx*dx+dy*dy;
      if(d2<rr*rr&&d2>.01){const dd=Math.sqrt(d2),push=(rr-dd)/2;dx/=dd;dy/=dd;
        A.x-=dx*push;A.y-=dy*push;B.x+=dx*push;B.y+=dy*push;}}}
  if(state==='playing'){
    for(const b of bullets){b.life-=d;if(b.life<=0){b.dead=true;continue;}
      b.x+=b.vx*d;b.y+=b.vy*d;
      if(b.x<WM||b.x>W-WM||b.y<WM||b.y>H-WM){b.dead=true;sparks(clamp(b.x,WM,W-WM),clamp(b.y,WM,H-WM),3,'#9aa4a8');continue;}
      for(const c of crates){if(b.x>c.x&&b.x<c.x+c.w&&b.y>c.y&&b.y<c.y+c.h){b.dead=true;sparks(b.x,b.y,5,'#d9c08a');break;}}
      if(b.dead)continue;
      for(const bl of barrels){if(bl.dead)continue;const d2=(b.x-bl.x)*(b.x-bl.x)+(b.y-bl.y)*(b.y-bl.y);
        if(d2<(bl.r+3)*(bl.r+3)){b.dead=true;bl.hp-=25;bl.flash=1;sparks(b.x,b.y,4,'#ffcf6a');sfx.hit();if(bl.hp<=0)explode(bl);break;}}
      if(b.dead)continue;
      for(const z of zombies){if(z.dead)continue;const d2=(b.x-z.x)*(b.x-z.x)+(b.y-z.y)*(b.y-z.y);
        if(d2<(z.r+3)*(z.r+3)){b.dead=true;damageZombie(z,25,b.vx/BSPD*3,b.vy/BSPD*3,b.o);break;}}}
    bullets=bullets.filter(b=>!b.dead);
    for(const dr of drops){dr.t-=d;
      for(const p of players){if(!p.alive)continue;const d2=(p.x-dr.x)*(p.x-dr.x)+(p.y-dr.y)*(p.y-dr.y);
        if(d2<(p.r+13)*(p.r+13)){dr.gone=true;
          if(dr.type==='med'){p.hp=Math.min(100,p.hp+30);effects.push({kind:'text',t:0,dur:.9,x:p.x,y:p.y-20,txt:'+30 HP',col:'#6fe89a'});}
          else{p.stimT=6;effects.push({kind:'text',t:0,dur:.9,x:p.x,y:p.y-20,txt:'STIM!',col:'#ffd23f'});}
          sfx.pick(dr.type);break;}}}
    drops=drops.filter(dr=>!dr.gone&&dr.t>0);
  }
  zombies=zombies.filter(z=>!z.dead);
  for(const q of parts){q.life-=d;q.x+=q.vx*d;q.y+=q.vy*d;q.vx*=(1-3.2*d);q.vy*=(1-3.2*d);}
  parts=parts.filter(q=>q.life>0);
  for(const e of effects)e.t+=d;
  effects=effects.filter(e=>e.t<e.dur);
}

    case 'menu': if(c==='Enter'||c==='NumpadEnter'||c==='Space')startGame(); break;
    case 'playing': if(c==='KeyP'||c==='Escape')pause(); break;
    case 'paused':

      else if(c==='KeyR')startGame();
      else if(c==='Escape')showMenu(); break;

      if(c==='KeyR'||c==='Enter'||c==='NumpadEnter')startGame();
      else if(c==='Escape')showMenu(); break;
  }
}
