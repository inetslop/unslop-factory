function initAudio(){
  if(AC){if(AC.state==='suspended')AC.resume();return;}
  try{AC=new (window.AudioContext||window.webkitAudioContext)();
    master=AC.createGain();master.gain.value=muted?0:0.5;master.connect(AC.destination);}catch(e){AC=null;}
}
function tone(o){if(!AC||muted)return;const t=AC.currentTime+(o.at||0);
  const osc=AC.createOscillator(),g=AC.createGain();osc.type=o.type||'square';
  osc.frequency.setValueAtTime(Math.max(1,o.f),t);
  if(o.f2)osc.frequency.exponentialRampToValueAtTime(Math.max(1,o.f2),t+o.t);
  g.gain.setValueAtTime(o.v||.2,t);g.gain.exponentialRampToValueAtTime(.0001,t+o.t);
  osc.connect(g);g.connect(master);osc.start(t);osc.stop(t+o.t+.03);}
function noiseSfx(o){o=o||{};if(!AC||muted)return;const t0=AC.currentTime+(o.at||0),t=o.t||.3;
  const n=Math.max(1,Math.floor(AC.sampleRate*t)),buf=AC.createBuffer(1,n,AC.sampleRate),d=buf.getChannelData(0);
  for(let i=0;i<n;i++)d[i]=(Math.random()*2-1)*(1-i/n);
  const src=AC.createBufferSource();src.buffer=buf;
  const flt=AC.createBiquadFilter();flt.type=o.type||'lowpass';flt.frequency.setValueAtTime(o.f||800,t0);
  if(o.f2)flt.frequency.exponentialRampToValueAtTime(Math.max(1,o.f2),t0+t);flt.Q.value=o.q||1;
  const g=AC.createGain();g.gain.value=o.v||.4;src.connect(flt);flt.connect(g);g.connect(master);src.start(t0);}

      L.n=Math.ceil((W+180)/L.P);L.span=L.n*L.P;const off=R(0,L.P);L.cars=[];

function tryMove(dx,dy){
  if(G.state!=='playing'||frog.dead)return;
  frog.face={x:dx,y:dy};
  frog.fx=frog.px;frog.fy=frog.py;
  frog.gx+=dx;frog.gy+=dy;
  frog.tx=cellX(frog.gx);frog.ty=laneC(clamp(frog.gy,0,9));
  if(frog.gy<0)frog.ty=HUD-46;if(frog.gy>9)frog.ty=H+30;
  frog.hop=0;SFX.hop();
}
function laneBonus(){
  if(frog.gy<frog.bestRow){const g=(frog.bestRow-frog.gy)*10;G.score+=g;
    pop(frog.px,frog.py-26,'+'+g,'#9fe8cf');frog.bestRow=frog.gy;}
}
function die(cause){
  if(G.state!=='playing'||frog.dead)return;
  if(cause==='time'){G.lives--;frog.dead=true;frog.cause='time';frog.hidden=true;SFX.die();addShake(4,.3);setState('dying');G.dieT=0;return;}
  if(frog.shield){frog.shield=false;frog.invuln=1.5;shatter(frog.px,frog.py);pop(frog.px,frog.py-28,'BLOCKED!','#7deaff');return;}
  G.lives--;frog.dead=true;frog.cause=cause;G.dieT=0;
  if(cause==='water'){SFX.splash();splashFx(frog.px,frog.py);}else SFX.die();
  addShake(6,.35);setState('dying');
}
function fire(){
  const a=Math.atan2(frog.face.y,frog.face.x);
  const angs=frog.spreadT>0?[a-.3,a,a+.3]:[a];
  for(const an of angs)G.bullets.push({x:frog.px+Math.cos(an)*16,y:frog.py+Math.sin(an)*16-2,
    vx:Math.cos(an)*640,vy:Math.sin(an)*640,life:1.5});
  pushP({type:'flash',x:frog.px+frog.face.x*20,y:frog.py+frog.face.y*20-2,life:.08,max:.08,size:9,col:'#fff3b0'});
  SFX.shoot();fireCd=frog.spreadT>0?.21:.15;
}

/* ============ update ============ */
function moveLanes(dt,sf){
  for(const L of G.lanes){if(L.t!=='road'&&L.t!=='river')continue;
    const v=L.dir*L.sp*G.speedMul*sf,arr=L.t==='road'?L.cars:L.riv;
    for(const e of arr){e._dx=v*dt;e.x+=v*dt;
      if(e.x>W+80)e.x-=L.span;else if(e.x<-e.w-140)e.x+=L.span;
      if(L.t==='road'&&e.dead){e.rt-=dt;if(e.rt<=0)e.dead=false;}}}
}
function findCarrier(gy){const L=G.lanes[gy];if(!L||L.t!=='river')return null;
  for(const e of L.riv)if(frog.px>e.x+7&&frog.px<e.x+e.w-7)return e;return null;}
function stepFrog(dt){
  if(frog.hop>=1)return;
  frog.hop=Math.min(1,frog.hop+dt/0.14);const t=frog.hop;
  frog.px=lerp(frog.fx,frog.tx,t);frog.py=lerp(frog.fy,frog.ty,t);
  if(frog.hop>=1){frog.px=frog.tx;frog.py=frog.ty;landFx();
    if(G.state==='playing'){
      if(frog.gx<0||frog.gx>=COLS||frog.gy<0||frog.gy>9)die('out');else laneBonus();}}
}
function checkHazards(){
  if(frog.dead||G.state!=='playing')return;
  const gy=frog.gy,L=G.lanes[gy];
  if(!L)return;
  if(L.t==='road'){const cy=laneC(gy);
    for(const e of L.cars){if(e.dead)continue;
      const h=e.w>100?46:40;
      if(frog.px+15>e.x+4&&frog.px-15<e.x+e.w-4&&frog.py+13>cy-h/2+4&&frog.py-13<cy+h/2-4){die('car');return;}}}
  else if(L.t==='river'){const car=findCarrier(gy);
    if(!car){die('water');return;}
    const dx=car._dx||0;frog.px+=dx;
    if(frog.hop<1){frog.tx+=dx;frog.fx+=dx;}
    if(frog.px<-8||frog.px>W+8)die('out');}
  for(const c of G.crows){const dx=c.x-frog.px,dy=c.y-frog.py;
    if(dx*dx+dy*dy<900){die('crow');return;}}
}
function update(dt){
  T+=dt;G.shakeT=Math.max(0,G.shakeT-dt);if(G.shakeT===0)G.shakeMag=0;
  const worldOn=(G.state==='menu'||G.state==='playing'||G.state==='dying'||G.state==='clear');
  if(worldOn){
    const sf=(G.state==='menu'?.55:1)*(G.slowT>0?.55:1);
    moveLanes(dt,sf);
    // crows
    if(G.state==='playing'){
      G.crowTimer-=dt;const cap=Math.min(4,1+((G.lvl/2)|0));
      if(G.crowTimer<=0&&G.crows.length<cap){
        const fromL=Math.random()<.5,row=pick([5,6,7,8]),sp=R(140,190)+G.lvl*12;
        G.crows.push({x:fromL?-40:W+40,vx:(fromL?1:-1)*sp,baseY:laneC(row),y:laneC(row),t:R(0,3),ph:R(0,9),home:G.lvl>=3&&Math.random()<.5});
        G.crowTimer=Math.max(1.1,3.6-.28*(G.lvl-1))*R(.75,1.3);}
    }
    for(let i=G.crows.length-1;i>=0;i--){const c=G.crows[i];c.t+=dt;c.x+=c.vx*dt;
      if(c.home&&G.state==='playing')c.baseY+=clamp(frog.py-c.baseY,-90,90)*dt;
      c.y=c.baseY+Math.sin(c.t*4.2+c.ph)*13;
      if(c.x<-60||c.x>W+60)G.crows.splice(i,1);}
    // bullets
    for(let i=G.bullets.length-1;i>=0;i--){const b=G.bullets[i];
      b.x+=b.vx*dt;b.y+=b.vy*dt;b.life-=dt;
      if(b.life<=0||b.x<-20||b.x>W+20||b.y<HUD-20||b.y>H+20){G.bullets.splice(i,1);continue;}
      let hit=false;
      for(let li=0;li<G.lanes.length&&!hit;li++){const L=G.lanes[li];if(L.t!=='road')continue;
        const cy=laneC(li);
        for(const e of L.cars){if(e.dead)continue;const h=e.w>100?46:40;
          if(b.x>e.x-2&&b.x<e.x+e.w+2&&b.y>cy-h/2-2&&b.y<cy+h/2+2){destroyCar(L,e,b);hit=true;break;}}}
      if(!hit)for(let ci=G.crows.length-1;ci>=0;ci--){const c=G.crows[ci];
        const dx=c.x-b.x,dy=c.y-b.y;if(dx*dx+dy*dy<361){killCrow(c,b);G.crows.splice(ci,1);hit=true;break;}}
      if(hit)G.bullets.splice(i,1);}
    // powerups
    for(let i=G.powerups.length-1;i>=0;i--){const p=G.powerups[i];p.life-=dt;
      if(G.state==='playing'&&!frog.dead&&Math.abs(frog.px-p.x)<28&&Math.abs(frog.py-p.y)<28){
        G.score+=50;SFX.power();
        if(p.type==='shield'){frog.shield=true;pop(p.x,p.y-26,'SHIELD!','#7deaff');sparkle(p.x,p.y,'#35e0ff');}
        else if(p.type==='spread'){frog.spreadT=8;pop(p.x,p.y-26,'SPREAD x3!','#ffb020');sparkle(p.x,p.y,'#ffb020');}
        else{G.slowT=7;pop(p.x,p.y-26,'SLOW-MO!','#59e6a8');sparkle(p.x,p.y,'#59e6a8');}
        G.powerups.splice(i,1);continue;}
      if(p.life<=0)G.powerups.splice(i,1);}
    for(let i=G.scorches.length-1;i>=0;i--){G.scorches[i].life-=dt;if(G.scorches[i].life<=0)G.scorches.splice(i,1);}
  }
  if(G.state==='menu'){
    mHopT-=dt;
    if(mHopT<=0&&frog.hop>=1){mHopT=R(1.8,3);const tgt=frog.gx===5?6:5;
      frog.fx=frog.px;frog.fy=frog.py;frog.gx=tgt;frog.tx=cellX(tgt);frog.ty=frog.py;frog.hop=0;
      frog.face={x:Math.sign(tgt-frog.gx)||1,y:0};}
    stepFrog(dt);
  }
  if(G.state==='playing'){
    G.time-=dt;
    const s=Math.ceil(G.time);if(s!==G.tickSec){G.tickSec=s;if(s<=5&&s>0)SFX.tick();}
    if(G.time<=0){G.time=0;die('time');}
    frog.invuln=Math.max(0,frog.invuln-dt);
    frog.spreadT=Math.max(0,frog.spreadT-dt);
    G.slowT=Math.max(0,G.slowT-dt);
    fireCd-=dt;
    if(shootHeld&&!frog.dead&&fireCd<=0)fire();
    stepFrog(dt);
    checkHazards();
  }
  else if(G.state==='dying'){
    G.dieT+=dt;
    if(G.dieT>.95){if(G.lives<=0)gameOver();else{resetFrog(1.4);setState('playing');}}
  }
  else if(G.state==='clear'){
    G.clearT+=dt;
    if(G.padTarget){frog.px=lerp(frog.px,G.padTarget.x,Math.min(1,dt*8));frog.py=lerp(frog.py,G.padTarget.y,Math.min(1,dt*8));}
    if(G.clearT>1.7){G.lvl++;buildLevel(G.lvl);G.time=maxTime(G.lvl);G.maxT=G.time;G.tickSec=-1;
      resetFrog(.9);setState('playing');}
  }
  // particles & popups always
  for(let i=G.parts.length-1;i>=0;i--){const p=G.parts[i];p.life-=dt;
    if(p.life<=0){G.parts.splice(i,1);continue;}
    if(p.type==='ring'){p.r+=p.vr*dt;continue;}
    p.x+=(p.vx||0)*dt;p.y+=(p.vy||0)*dt;if(p.g)p.vy+=p.g*dt;if(p.vr!==undefined&&p.rot!==undefined)p.rot+=p.vr*dt;}
  for(let i=G.pops.length-1;i>=0;i--){const p=G.pops[i];p.life-=dt*1.1;p.y-=30*dt;
    if(p.life<=0)G.pops.splice(i,1);}
}

function easeOutBack(t){const c=1.70158;return 1+(c+1)*Math.pow(t-1,3)+c*Math.pow(t-1,2);}
