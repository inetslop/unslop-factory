let AC=null, master=null, muted=false, musicStep=0;
function initAudio(){
  if(AC){ if(AC.state==='suspended') AC.resume(); return; }
  try{
    AC=new (window.AudioContext||window.webkitAudioContext)();
    master=AC.createGain(); master.gain.value=muted?0:0.16; master.connect(AC.destination);
    const bass=[98,98,116.54,98,130.81,130.81,146.83,130.81];
    setInterval(()=>{ if(!AC||muted) return;
      if(state==='playing'||state==='menu'&&!paused){
        tone({type:'sine',f:bass[musicStep%8],dur:.22,vol:.05});
        if(musicStep%4===2) noise(.03,.02,6000,'highpass');
        musicStep++; } },140);
  }catch(e){}
}
function tone(o){ if(!AC||muted) return;
  const t0=AC.currentTime+(o.at||0), osc=AC.createOscillator(), g=AC.createGain();
  osc.type=o.type||'sine'; osc.frequency.setValueAtTime(o.f,t0);
  if(o.f2) osc.frequency.exponentialRampToValueAtTime(Math.max(1,o.f2),t0+o.dur);
  g.gain.setValueAtTime(.0001,t0);
  g.gain.exponentialRampToValueAtTime(o.vol,t0+.008);
  g.gain.exponentialRampToValueAtTime(.0001,t0+o.dur);
  osc.connect(g).connect(master); osc.start(t0); osc.stop(t0+o.dur+.05);
}
function noise(dur,vol,freq,type,at){ if(!AC||muted) return;
  const t0=AC.currentTime+(at||0), n=Math.max(1,Math.floor(AC.sampleRate*dur));
  const buf=AC.createBuffer(1,n,AC.sampleRate), d=buf.getChannelData(0);
  for(let i=0;i<n;i++) d[i]=(Math.random()*2-1)*(1-i/n);
  const src=AC.createBufferSource(); src.buffer=buf;
  const f=AC.createBiquadFilter(); f.type=type||'lowpass'; f.frequency.value=freq;
  const g=AC.createGain(); g.gain.value=vol;
  src.connect(f).connect(g).connect(master); src.start(t0);
}

function onKey(k){
  if(k==='m'){ muted=!muted; if(master) master.gain.value=muted?0:0.16; return; }
  switch(state){
    case 'menu': if(k===' '||k==='enter') startRun(); break;
    case 'playing':
      if(k==='p'||k==='escape') setPause(true);
      else if(k==='1') selectColor(0);
      else if(k==='2') selectColor(1);
      else if(k==='3') selectColor(2);

      else if(k==='e') cycle(1);

function moveEnemy(e,dt){
  switch(e.type){
    case 'drone': e.x=e.ax+Math.sin(e.t*.9+e.ph)*48; e.y=e.ay+Math.cos(e.t*.7+e.ph)*30; break;
    case 'shifter': { const a=e.t*.5+e.ph; e.x=e.ax+Math.cos(a)*62; e.y=e.ay+Math.sin(a)*40; break; }
    case 'orb': {
      if(state==='menu'){ const a=e.t*.3+e.ph; e.x=e.ax+Math.cos(a)*50; e.y=e.ay+Math.sin(a)*36; break; }
      const dx=player.x-e.x, dy=player.y-e.y, d=Math.hypot(dx,dy)||1;
      if(d>240){ e.x+=dx/d*48*dt; e.y+=dy/d*48*dt; }
      else { e.x+=(-dy/d)*36*dt; e.y+=(dx/d)*36*dt; }
      break; }
    case 'seed': e.x=e.ax+Math.sin(e.t*1.1+e.ph)*22; e.y=e.ay+Math.cos(e.t*.9+e.ph)*18; break;
  }
  e.x=clamp(e.x,30,W-30); e.y=clamp(e.y,30,H-30);
}

function enemyFire(e,dt){
  if(eb.length>720) return;
  const wv=state==='menu'?2:wave+1;
  const tx=state==='menu'?W*.5:player.x, ty=state==='menu'?H*.75:player.y;
  switch(e.type){
    case 'drone': e.ft-=dt; if(e.ft<=0){ e.ft=Math.max(1.5,2.6-wv*.09); const n=8+wv, sp=105+wv*9; for(let i=0;i<n;i++) eb.push(mkb(e.x,e.y,e.spin+i/n*TAU,sp,e.color)); e.spin+=.4; } break;
    case 'shifter': {
      e.phaseT+=dt; if(e.phaseT>=2.4){ e.phaseT=0; e.color=(e.color+1)%3; ring(e.x,e.y,e.r*.6,e.r*1.5,e.color,.25); }
      e.ft-=dt; if(e.ft<=0){ e.ft=Math.max(1.1,2.0-wv*.07); const base=Math.atan2(ty-e.y,tx-e.x); for(const off of[-.3,0,.3]) eb.push(mkb(e.x,e.y,base+off,150+wv*7,e.color)); }
      break; }
    case 'orb': e.sa+=dt*(2.4+wv*.12); e.ft-=dt; if(e.ft<=0){ e.ft=.13; const sp=92+wv*8; eb.push(mkb(e.x,e.y,e.sa,sp,e.color)); eb.push(mkb(e.x,e.y,e.sa+Math.PI,sp,e.color)); } break;
    case 'seed': break;
  }
}

/* ============================== WORLD UPDATE ============================== */
function updateWorld(dt){
  waveT+=dt; runTime+=dt;
  const inPlay=state==='playing';
  if(inPlay){ while(spawnQueue.length&&spawnQueue[0].t<=waveT){ const s=spawnQueue.shift(); spawnEnemy(s.type,s.col,s.fx,s.fy); } }

  // focus / time scale
  const wantFocus=(keys['shift']&&!player.dead&&focus>0);
  if(wantFocus){ if(timeScale>.8) SFX.focus(); timeScale+=( .35-timeScale)*Math.min(1,dt*10); focus=Math.max(0,focus-dt*45); }
  else { timeScale+=(1-timeScale)*Math.min(1,dt*8); focus=Math.min(100,focus+dt*22); }
  const wdt=dt*(player.dead?.25:timeScale);

  // player
  if(!player.dead){
    let dx=(keys['arrowright']||keys['d']?1:0)-(keys['arrowleft']||keys['a']?1:0);
    let dy=(keys['arrowdown']||keys['s']?1:0)-(keys['arrowup']||keys['w']?1:0);
    if(dx&&dy){ dx*=.7071; dy*=.7071; }
    player.x=clamp(player.x+dx*player.speed*dt,12,W-12);
    player.y=clamp(player.y+dy*player.speed*dt,12,H-12);
    if((dx||dy)&&Math.random()<dt*40) puff(player.x-dx*14,player.y-dy*14,1,1,60,.35,1.8);

    // aim
    const mouseFresh=(performance.now()/1000-mouse.lastMove)<3;
    let tx=null,ty=null;
    if(mouseFresh){ tx=mouse.x; ty=mouse.y; }
    else{
      let bd=1e9;
      for(const e of enemies){ const d=Math.hypot(e.x-player.x,e.y-player.y)+( (e.wild||e.color===curColor)?0:600 ); if(d<bd){ bd=d; tx=e.x; ty=e.y; } }
    }
    if(tx!==null) player.aim=lerpAngle(player.aim,Math.atan2(ty-player.y,tx-player.x),14*dt);

    // fire
    player.cool-=dt;
    if(inPlay&&(keys[' ']||mouse.down)){
      if(ammo[curColor]>0&&player.cool<=0){
        player.cool=.115; shots++; ammo[curColor]--;
        pb.push({x:player.x+Math.cos(player.aim)*16,y:player.y+Math.sin(player.aim)*16,vx:Math.cos(player.aim)*640,vy:Math.sin(player.aim)*640,c:curColor});
        SFX.fire(curColor); ring(player.x+Math.cos(player.aim)*18,player.y+Math.sin(player.aim)*18,2,9,curColor,.15);
      } else if(ammo[curColor]<=0&&player.cool<=0){ player.cool=.3; SFX.dry(); dryFlash=.3; }
    }
  }

  // enemies
  for(const e of enemies){ if(e.dead) continue; e.t+=wdt; moveEnemy(e,wdt); enemyFire(e,wdt); e.flash=Math.max(0,e.flash-wdt); }

  // player bullets + hits
  for(let i=pb.length-1;i>=0;i--){
    const p=pb[i]; p.x+=p.vx*dt; p.y+=p.vy*dt;
    if(p.x<-30||p.x>W+30||p.y<-30||p.y>H+30){ pb.splice(i,1); combo=0; continue; }
    let hit=false;
    for(const e of enemies){
      if(e.dead) continue;
      const rr=e.r-2+4, dx=p.x-e.x, dy=p.y-e.y;
      if(dx*dx+dy*dy<rr*rr){
        hit=true;
        if(e.wild||e.color===p.c){
          e.hp--; e.flash=.12; hits++; combo++; maxCombo=Math.max(maxCombo,combo);
          const pts=10*mult(); addScore(pts); SFX.hit();
          puff(p.x,p.y,e.wild?3:e.color,5,160,.35,2.2); ftext(p.x,p.y-8,'+'+pts,2);
          if(e.hp<=0) killEnemy(e);
        } else {
          clinks++; combo=0; SFX.clink();
          puff(p.x,p.y,3,4,140,.3,1.8); ftext(p.x,p.y-8,'MISS',0);
        }
        break;
      }
    }
    if(hit) pb.splice(i,1);
  }

  // enemy bullets + player
  for(let i=eb.length-1;i>=0;i--){
    const b=eb[i]; b.t+=wdt; b.x+=b.vx*wdt; b.y+=b.vy*wdt;
    if(b.x<-40||b.x>W+40||b.y<-40||b.y>H+40||b.t>14){ eb.splice(i,1); continue; }
    if(!player.dead){
      const dx=b.x-player.x, dy=b.y-player.y, hr=b.r*.8+player.r;
      if(dx*dx+dy*dy<hr*hr){ playerDie(); break; }
      if(!b.gz){ const gr=b.r+14; if(dx*dx+dy*dy<gr*gr){ b.gz=true; grazes++; focus=Math.min(100,focus+7); addScore(5); SFX.graze(); ring(b.x,b.y,2,16,3,.25); } }
    }
  }

  if(enemies.some(e=>e.dead)) enemies=enemies.filter(e=>!e.dead);

  // solvability (puzzle fail)
  if(inPlay&&!player.dead){
    const need=[0,0,0]; let sh=0;
    for(const e of enemies){ if(e.type==='shifter') sh+=e.hp; else need[e.color]+=e.hp; }
    let msg='';
    for(let c=0;c<3;c++) if(need[c]>ammo[c]) msg='The '+['ROSE','CYAN','GOLD'][c]+' cores were left unbroken — your resonance ran dry.';
    if(!msg&&sh>ammo[0]+ammo[1]+ammo[2]) msg='Your reserves fell before the rotators did. Read the field and spend each hue with intent.';
    if(msg) showFail('RESONANCE DEPLETED',msg);
  }

  // wave clear / death / interlude
  if(inPlay){
    if(!player.dead&&spawnQueue.length===0&&enemies.length===0) waveCleared();
    else if(player.dead){ deathT+=dt; if(deathT>1.5) showFail('SIGNAL LOST','Your core was overwhelmed on '+WAVES[wave].name+' — wave '+(wave+1)+'.'); }
  } else if(state==='waveclear'){
    clearT+=dt;
    if(clearT>=2.3){ if(wave+1<WAVES.length) startWave(wave+1); else showVictory(); }
  }

  updateParts(dt);
}
function updateParts(dt){
  for(let i=parts.length-1;i>=0;i--){
    const p=parts[i]; p.life-=dt;
    if(p.life<=0){ parts.splice(i,1); continue; }
    if(p.kind==='dot'){ p.x+=p.vx*dt; p.y+=p.vy*dt; p.vx*=Math.exp(-2.5*dt); p.vy*=Math.exp(-2.5*dt); }
    else if(p.kind==='txt') p.y+=p.vy*dt;
  }
}

/* ============================== AMBIENT (menu demo) ============================== */
function seedAmbient(){
  const types=['drone','shifter','orb','drone','seed','shifter'];
  for(let i=0;i<6;i++) spawnEnemy(types[i],i%3,.15+Math.random()*.7,.12+Math.random()*.42);
}
function updateAmbient(dt){
  for(const e of enemies){ e.t+=dt; moveEnemy(e,dt); enemyFire(e,dt); }
  for(let i=eb.length-1;i>=0;i--){ const b=eb[i]; b.x+=b.vx*dt; b.y+=b.vy*dt; if(b.x<-40||b.x>W+40||b.y<-40||b.y>H+40) eb.splice(i,1); }
}

/* ============================== RENDER ============================== */
function poly(n,r,rot){ ctx.beginPath(); for(let i=0;i<n;i++){ const a=rot+i/n*TAU; if(i)ctx.lineTo(Math.cos(a)*r,Math.sin(a)*r); else ctx.moveTo(Math.cos(a)*r,Math.sin(a)*r); } ctx.closePath(); }

function drawShip(x,y,ang,t){
  ctx.save(); ctx.translate(x,y); ctx.rotate(ang);
  ctx.globalCompositeOperation='lighter';
  ctx.fillStyle='rgba(160,220,255,.1)'; ctx.beginPath(); ctx.arc(0,0,17,0,TAU); ctx.fill();
  ctx.globalCompositeOperation='source-over';
  ctx.beginPath(); ctx.moveTo(15,0); ctx.lineTo(-9,8); ctx.lineTo(-4,0); ctx.lineTo(-9,-8); ctx.closePath();
  ctx.fillStyle='#eef7ff'; ctx.fill(); ctx.strokeStyle='rgba(140,210,255,.7)'; ctx.lineWidth=1; ctx.stroke();
  ctx.beginPath(); ctx.arc(0,0,5.5,0,TAU); ctx.strokeStyle='rgba(255,255,255,'+(.45+.3*Math.sin(t*6))+')'; ctx.stroke();
  ctx.fillStyle='#ffffff'; ctx.beginPath(); ctx.arc(0,0,2.6,0,TAU); ctx.fill();
  ctx.restore();
}

function drawEnemy(e,t){
  const c=e.color;
  ctx.save(); ctx.translate(e.x,e.y);
  if(e.type==='drone'){
    ctx.rotate(e.t*.5+e.ph); poly(6,e.r,0); ctx.fillStyle='#0d1826'; ctx.fill(); ctx.strokeStyle=tint(c,.5); ctx.lineWidth=1.5; ctx.stroke();
    ctx.rotate(-(e.t*.5+e.ph));
    ctx.beginPath(); ctx.arc(0,0,e.r*.62,0,TAU); ctx.strokeStyle=tint(c,.3); ctx.stroke();
    ctx.globalCompositeOperation='lighter'; ctx.fillStyle=tint(c,.28); ctx.beginPath(); ctx.arc(0,0,e.r*.75,0,TAU); ctx.fill();
    ctx.globalCompositeOperation='source-over';
    ctx.fillStyle=HEX[c]; ctx.beginPath(); ctx.arc(0,0,e.r*.4,0,TAU); ctx.fill();
    ctx.fillStyle='#fff'; ctx.beginPath(); ctx.arc(0,0,e.r*.15,0,TAU); ctx.fill();
  }
  else if(e.type==='shifter'){
    const prog=e.phaseT/2.4, next=(c+1)%3;
    ctx.setLineDash([6,6]); ctx.lineDashOffset=-t*30;
    ctx.beginPath(); ctx.arc(0,0,e.r+7,0,TAU);
    ctx.strokeStyle=tint(next,.12+(prog>.7?(prog-.7)/.3*.75:0)); ctx.lineWidth=2; ctx.stroke();
    ctx.setLineDash([]);
    ctx.rotate(-e.t*.4+e.ph); poly(5,e.r,0); ctx.fillStyle='#0d1826'; ctx.fill(); ctx.strokeStyle=tint(c,.55); ctx.lineWidth=1.5; ctx.stroke();
    ctx.rotate(e.t*.4-e.ph);
    ctx.globalCompositeOperation='lighter'; ctx.fillStyle=tint(c,.3); ctx.beginPath(); ctx.arc(0,0,e.r*.85,0,TAU); ctx.fill();
    ctx.globalCompositeOperation='source-over';
    const pr=e.r*.42*(1+Math.sin(t*6)*.08);
    ctx.fillStyle=HEX[c]; ctx.beginPath(); ctx.arc(0,0,pr,0,TAU); ctx.fill();
    ctx.fillStyle='#fff'; ctx.beginPath(); ctx.arc(0,0,e.r*.15,0,TAU); ctx.fill();
  }
  else if(e.type==='orb'){
    ctx.beginPath(); ctx.arc(0,0,e.r,0,TAU); ctx.strokeStyle=tint(c,.45); ctx.lineWidth=2; ctx.stroke();
    for(let i=0;i<3;i++){ ctx.beginPath(); ctx.arc(0,0,e.r-4,e.sa+i*TAU/3,e.sa+i*TAU/3+.9); ctx.strokeStyle=tint(c,.85); ctx.lineWidth=3; ctx.stroke(); }
    ctx.fillStyle='#0c1622'; ctx.beginPath(); ctx.arc(0,0,e.r*.6,0,TAU); ctx.fill();
    ctx.globalCompositeOperation='lighter'; ctx.fillStyle=tint(c,.35); ctx.beginPath(); ctx.arc(0,0,e.r*.55,0,TAU); ctx.fill();
    ctx.globalCompositeOperation='source-over';
    ctx.fillStyle=HEX[c]; ctx.beginPath(); ctx.arc(0,0,e.r*.34,0,TAU); ctx.fill();
    ctx.fillStyle='#fff'; ctx.beginPath(); ctx.arc(0,0,e.r*.13,0,TAU); ctx.fill();
  }
  else { // seed — wildcard
    ctx.rotate(e.t*.8+e.ph);
    ctx.beginPath(); for(let i=0;i<16;i++){ const r=i%2?e.r:e.r*.55,a=i/16*TAU; if(i)ctx.lineTo(Math.cos(a)*r,Math.sin(a)*r); else ctx.moveTo(Math.cos(a)*r,Math.sin(a)*r); } ctx.closePath();
    ctx.fillStyle='#0d1826'; ctx.fill(); ctx.strokeStyle='rgba(230,245,255,.4)'; ctx.lineWidth=1; ctx.stroke();
    ctx.rotate(-(e.t*.8+e.ph));
    for(let i=0;i<3;i++){ ctx.beginPath(); ctx.arc(0,0,e.r+3.5,i*TAU/3-.35+.6,i*TAU/3+.35); ctx.strokeStyle=tint(i,.7); ctx.lineWidth=2; ctx.stroke(); }
    ctx.fillStyle='#fff'; ctx.beginPath(); ctx.arc(0,0,e.r*.3,0,TAU); ctx.fill();
  }
  // hp ticks (fixed-hue enemies)
  if(!e.wild){
    for(let i=0;i<e.maxHp||i<((e.type==='orb')?5:(e.type==='shifter')?3:2);i++){
      const mx=(e.type==='orb')?5:(e.type==='shifter')?3:2;
      const a=-Math.PI/2+(i-(mx-1)/2)*.5, on=i<e.hp;
      ctx.strokeStyle=tint(c,on?.85:.15); ctx.lineWidth=2;
      ctx.beginPath(); ctx.moveTo(Math.cos(a)*(e.r+4),Math.sin(a)*(e.r+4)); ctx.lineTo(Math.cos(a)*(e.r+9),Math.sin(a)*(e.r+9)); ctx.stroke();
    }
  }
  if(e.flash>0){ ctx.globalAlpha=e.flash/.12*.75; ctx.fillStyle='#fff'; ctx.beginPath(); ctx.arc(0,0,e.r*1.05,0,TAU); ctx.fill(); ctx.globalAlpha=1; }
  ctx.restore();
}

function drawParts(){
  for(const p of parts){
    const a=Math.max(0,p.life/p.lmax);

    else if(p.kind==='txt'){ ctx.font='700 12px '+MONO; ctx.textAlign='center'; ctx.fillStyle=tint(p.h,a*.95); ctx.fillText(p.txt,p.x,p.y); }
  }
}

function render(t){
  // background
  const bg=ctx.createLinearGradient(0,0,0,H); bg.addColorStop(0,'#081221'); bg.addColorStop(.6,'#0a1a29'); bg.addColorStop(1,'#07101a');
  ctx.fillStyle=bg; ctx.fillRect(0,0,W,H);
  ctx.globalCompositeOperation='lighter';
  for(let i=0;i<3;i++){
    const nx=W*(.25+.5*i)+Math.sin(t*.05+i*2.1)*W*.1, ny=H*(.3+ (i%2)*.4)+Math.cos(t*.04+i)*H*.08;
    const g=ctx.createRadialGradient(nx,ny,0,nx,ny,H*.5);
    g.addColorStop(0,tint(i,.06)); g.addColorStop(1,'rgba(0,0,0,0)');
    ctx.fillStyle=g; ctx.fillRect(0,0,W,H);
  }
  for(const s of stars){
    const x=(s.fx*W+t*(s.ly?4:9))%W, y=s.fy*H;
    ctx.fillStyle='rgba(210,235,250,'+(.18+.2*Math.sin(t*s.tw+s.fx*40))+')';
    ctx.fillRect(x,y,s.r,s.r);
  }
  ctx.globalCompositeOperation='source-over';
  // drifting dot grid
  const off=(t*6)%72;
  ctx.fillStyle='rgba(150,210,235,.045)';
  for(let x=0;x<W+72;x+=72) for(let y=-72+off;y<H;y+=72) ctx.fillRect(x,y,1.6,1.6);

  if(state==='menu'){
    const gx=W/2+Math.sin(t*.4)*W*.18, gy=H*.75+Math.cos(t*.3)*H*.05;
    for(const e of enemies) drawEnemy(e,t);
    ctx.globalCompositeOperation='lighter';
    for(const b of eb){
      ctx.strokeStyle=tint(b.c,.25); ctx.lineWidth=2;
      ctx.beginPath(); ctx.moveTo(b.x-b.vx*.03,b.y-b.vy*.03); ctx.lineTo(b.x,b.y); ctx.stroke();
      ctx.fillStyle=tint(b.c,.2); ctx.beginPath(); ctx.arc(b.x,b.y,b.r*2.1,0,TAU); ctx.fill();
      ctx.fillStyle=tint(b.c,.75); ctx.beginPath(); ctx.arc(b.x,b.y,b.r*.9,0,TAU); ctx.fill();
    }
    ctx.globalCompositeOperation='source-over';
    drawShip(gx,gy,Math.sin(t*.4+1.6)*.6,t);
  } else {
    // shake
    const sx=(Math.random()-.5)*shake, sy=(Math.random()-.5)*shake;
    ctx.save(); ctx.translate(sx,sy);
    for(const e of enemies) drawEnemy(e,t);
    ctx.globalCompositeOperation='lighter';
    for(const b of eb){
      ctx.strokeStyle=tint(b.c,.28); ctx.lineWidth=2;
      ctx.beginPath(); ctx.moveTo(b.x-b.vx*.03,b.y-b.vy*.03); ctx.lineTo(b.x,b.y); ctx.stroke();
      ctx.fillStyle=tint(b.c,.18); ctx.beginPath(); ctx.arc(b.x,b.y,b.r*2.1,0,TAU); ctx.fill();
      ctx.fillStyle=tint(b.c,.75); ctx.beginPath(); ctx.arc(b.x,b.y,b.r*.9,0,TAU); ctx.fill();
      ctx.fillStyle='rgba(255,255,255,.85)'; ctx.beginPath(); ctx.arc(b.x,b.y,b.r*.4,0,TAU); ctx.fill();
    }
    for(const p of pb){
      const a=Math.atan2(p.vy,p.vx);
      ctx.save(); ctx.translate(p.x,p.y); ctx.rotate(a);
      ctx.fillStyle=tint(p.c,.25); ctx.beginPath(); ctx.arc(0,0,7,0,TAU); ctx.fill();
      ctx.fillStyle=HEX[p.c]; ctx.beginPath(); ctx.moveTo(10,0); ctx.lineTo(-6,3.4); ctx.lineTo(-3,0); ctx.lineTo(-6,-3.4); ctx.closePath(); ctx.fill();
      ctx.strokeStyle='rgba(255,255,255,.9)'; ctx.lineWidth=1.4; ctx.beginPath(); ctx.moveTo(8,0); ctx.lineTo(-4,0); ctx.stroke();
      ctx.restore();
    }
    ctx.globalCompositeOperation='source-over';
    if(!player.dead) drawShip(player.x,player.y,player.aim,t);
    ctx.globalCompositeOperation='lighter';
    drawParts();
    ctx.globalCompositeOperation='source-over';
    ctx.restore();

    // focus tint + ring
    if(timeScale<.98&&state==='playing'&&!player.dead){
      ctx.fillStyle='rgba(56,225,232,'+((1-timeScale)*.16)+')'; ctx.fillRect(0,0,W,H);
      ctx.strokeStyle='rgba(255,255,255,.3)'; ctx.lineWidth=1.5;
      ctx.beginPath(); ctx.arc(player.x,player.y,42+Math.sin(t*8)*5,0,TAU); ctx.stroke();
    }
    drawHUD(t);
  }

  // vignette
  if(vignGrad){ ctx.fillStyle=vignGrad; ctx.fillRect(0,0,W,H); }
}

let last=performance.now();
function frame(now){
  requestAnimationFrame(frame);
  const dt=Math.min(.05,(now-last)/1000)||0; last=now;
  if(freeze>0) freeze-=dt;
  shake=Math.max(0,shake-dt*26); dryFlash=Math.max(0,dryFlash-dt);
  if(state==='menu') updateAmbient(dt);
  else if((state==='playing'||state==='waveclear')&&!paused){ if(freeze<=0) updateWorld(dt); }
  else if(state==='failed'||state==='victory') updateParts(dt*.6);
  render(now/1000);
}
