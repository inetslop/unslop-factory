  p.x+=p.vx*dt; p.y+=p.vy*dt; if(p.t>=1)parts.splice(i,1);}
 for(let i=floats.length-1;i>=0;i--){floats[i].t+=dt; if(floats[i].t>1)floats.splice(i,1);}

 if(paused)return;

 updFlip(FLIP_L,flipL(),dt); updFlip(FLIP_R,flipR(),dt);
 if(charging)charge=Math.min(1,charge+dt*1.5);

 /* match-3 phase machine */
 if(phase==='clear'){ clearT-=dt; for(const p of clearing)p.anim+=dt/CLEAR_DUR;
  if(clearT<=0){clearing.length=0; applyGravity();} }
 else if(phase==='drop'){ dropT+=dt; const tt=Math.min(1,dropT/DROP_DUR), e=tt*tt; let done=true;
  for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++){const p=G[r][c];
   if(p&&p.state==='drop'){ p.y=p.fromY+(p.fy-p.fromY)*e;
    if(tt<1)done=false; else {p.state='idle'; p.y=p.fy;} } }
  if(done){ const g2=findMatches2();
   if(g2.length){chain++; startClear(g2);} else {phase='idle'; chain=0;} } }

 /* ball */
 if(state==='ready'){ const hy=headY(charge);
  ball.x=706; ball.y=hy-BR; ball.vx=ball.vy=0; ball.trail.length=0; }
 else if(state==='play'){ stepBall(dt);
  if(ball.y-BR>TH+20||ball.x<-80||ball.x>TW+80)loseBall(); }
 else if(state==='drain'){ drainT-=dt;
  if(drainT<=0){ if(lives>0){resetBall(); state='ready';} else gameOver(); } }
}

**How it plays:** Launch the ball down the right lane (hold **Space**, release to fire). It curves into a table full of gem pegs — when the ball knocks a peg, that gem swaps with its neighbor in the direction of travel. Build runs of 3+ to clear them; refills cascade from above and chain combo multipliers. Pop bumpers, slingshots and flipper saves keep the pinball side alive: **←/→** (or **A/D**) for flippers, **R** restarts, **P** pauses, **M** toggles sound. Extra balls are awarded every 60,000 points; your best score is saved locally.
