function makeShape(seed){ const r=rng(seed),n=9+Math.floor(r()*3),a=[]; for(let i=0;i<n;i++)a.push(.78+r()*.27); return a; }
function spawnRock(x,y,size,fx,fy,vr){
  const rk={x:x,y:y,px:fx!==undefined?fx:x+.5,py:fy!==undefined?fy:y+.5,
    size:size,r:RAD[size],seed:(Math.random()*1e9)|0,shape:makeShape((Math.random()*1e9)|0),
    rot:Math.random()*TAU,vr:vr||0,slide:null,wobble:0,heavyT:0,docked:false,born:(Math.random()*.15)};
  if(fx!==undefined)rk.slide={fx:fx,fy:fy,t:0,dur:.4};
  rocks.push(rk); occ.set(key(x,y),rk); return rk;
}
function loadLevel(i){
  const L=LEVELS[i]; H=L.map.length; W=Math.max(...L.map.map(r=>r.length));
  obstacles=new Set(); pads=new Map(); rocks=[]; occ=new Map(); bolts=[]; parts=[];
  let s0={x:2.5,y:2.5};
  for(let y=0;y<H;y++)for(let x=0;x<W;x++){ const c=L.map[y][x]||'.';
    if(c==='#')obstacles.add(key(x,y));
    else if(c==='P')pads.set(key(x,y),{x:x,y:y,pulse:0,covered:false});
    else if(c==='S')s0={x:x+.5,y:y+.5};
    else if(c==='b'||c==='m'||c==='B'){ const rk=spawnRock(x,y,c==='b'?'small':(c==='m'?'mid':'big')); rk.born=.15+rocks.length*.07; }
  }
  ship={x:s0.x,y:s0.y,vx:0,vy:0,a:-Math.PI/2};
  pushes=0;shots=0;levelTime=0;energy=100;cd=0;setThrust(false);thrustOn=false;
  updateHUD();
}

/* ============================== rules ============================== */
function cellFree(x,y){ return x>=0&&y>=0&&x<W&&y<H&&!obstacles.has(key(x,y))&&!occ.has(key(x,y)); }
function refreshCovered(){ let n=0; pads.forEach(p=>{ p.covered=occ.has(key(p.x,p.y)); if(p.covered)n++; }); return n; }
function doPush(rock,tx,ty,nx,ny){
  occ.delete(key(rock.x,rock.y)); rock.x=tx; rock.y=ty; occ.set(key(tx,ty),rock);
  rock.slide={fx:rock.px,fy:rock.py,t:0,dur:.13};
  rock.vr+=(Math.random()*4-2);
  const vn=ship.vx*nx+ship.vy*ny;
  if(vn<0){ ship.vx-=nx*vn*1.7; ship.vy-=ny*vn*1.7; }
  ship.vx-=nx*.9; ship.vy-=ny*.9;
  pushes++; S.push();
  for(let i=0;i<4;i++)addPart(rock.px,rock.py,-nx*(.5+Math.random()),-ny*(.5+Math.random()),.3,.12,'rgba(200,220,255,.8)');
  updateHUD();
}
function onSettle(rock){
  const k=key(rock.x,rock.y), p=pads.get(k);
  if(p){ if(!rock.docked){ rock.docked=true; p.pulse=1; S.dock();
      for(let i=0;i<12;i++){const a=Math.random()*TAU,s=.8+Math.random()*1.6;
        addPart(rock.px,rock.py,Math.cos(a)*s,Math.sin(a)*s,.5,.14,'#57ffa8');}
      addRing(rock.px,rock.py,'#57ffa8'); } }
  else rock.docked=false;
  if(refreshCovered()===pads.size){ clearLevel(); }
  updateHUD();
}
function pickFree(x,y,dir,excl){
  const offs=[]; for(let r=1;r<=2;r++)for(let dy=-r;dy<=r;dy++)for(let dx=-r;dx<=r;dx++){
    if(Math.max(Math.abs(dx),Math.abs(dy))!==r)continue; offs.push([dx,dy,r]); }
  offs.sort((a,b)=>{ const da=a[0]*dir.x+a[1]*dir.y, db=b[0]*dir.x+b[1]*dir.y;
    return (db-b[2]*.45+Math.random()*.3)-(da-a[2]*.45+Math.random()*.3); });
  for(const o of offs){ const tx=x+o[0],ty=y+o[1];
    if(tx<0||ty<0||tx>=W||ty>=H)continue;
    if(obstacles.has(key(tx,ty))||occ.has(key(tx,ty))||excl.has(key(tx,ty)))continue;
    return {x:tx,y:ty}; }
  return null;
}
function splitRock(rock,bolt){
  const i=rocks.indexOf(rock); if(i<0)return; rocks.splice(i,1); occ.delete(key(rock.x,rock.y));
  const cx=rock.x+.5,cy=rock.y+.5; let dx=cx-bolt.x,dy=cy-bolt.y; const L=Math.hypot(dx,dy)||1; dx/=L;dy/=L;
  const si=SIZES.indexOf(rock.size), kids=[]; const excl=new Set();
  for(let j=0;j<2;j++){ const dir=j===0?{x:dx,y:dy}:{x:-dx,y:-dy};
    const c=pickFree(rock.x,rock.y,dir,excl); if(!c)break; excl.add(key(c.x,c.y));
    kids.push(spawnRock(c.x,c.y,SIZES[si-1],cx,cy,j===0?3.5:-3.5)); }
  const col=RSTYLE[rock.size].stroke;
  for(let i=0;i<14+si*8;i++){const a=Math.random()*TAU,s=1+Math.random()*(1.5+si);
    addPart(cx,cy,Math.cos(a)*s,Math.sin(a)*s,.5+.3*Math.random(),.16+i%2*.06,i%3?col:'#ffffff');}
  addRing(cx,cy,col); shake=Math.min(1,shake+(.3+si*.18)); flashT=Math.max(flashT,.22); S.split(si); updateHUD();
}

function buildDrifters(){ drifters=[]; for(let i=0;i<5;i++)drifters.push({
  x:Math.random()*vw,y:Math.random()*vh,vx:(Math.random()-.5)*14,vy:(Math.random()-.5)*14,

function collideWalls(){
  const m=WALL; let hit=0;
  if(ship.x-SHIP_R<m){ship.x=m+SHIP_R; if(ship.vx<0){hit=Math.max(hit,-ship.vx); ship.vx=-ship.vx*.72;}}
  if(ship.x+SHIP_R>W-m){ship.x=W-m-SHIP_R; if(ship.vx>0){hit=Math.max(hit,ship.vx); ship.vx=-ship.vx*.72;}}
  if(ship.y-SHIP_R<m){ship.y=m+SHIP_R; if(ship.vy<0){hit=Math.max(hit,-ship.vy); ship.vy=-ship.vy*.72;}}
  if(ship.y+SHIP_R>H-m){ship.y=H-m-SHIP_R; if(ship.vy>0){hit=Math.max(hit,ship.vy); ship.vy=-ship.vy*.72;}}
  if(hit>1.4){ S.thump(Math.min(.5,hit*.1)); shake=Math.min(1,shake+hit*.06); }
}
function collideObstacles(){
  const gx=Math.floor(ship.x),gy=Math.floor(ship.y);
  for(let dy=-1;dy<=1;dy++)for(let dx=-1;dx<=1;dx++){
    const cx=gx+dx,cy=gy+dy; if(!obstacles.has(key(cx,cy)))continue;
    const nxp=clamp(ship.x,cx,cx+1),nyp=clamp(ship.y,cy,cy+1);
    let dxp=ship.x-nxp,dyp=ship.y-nyp; const d2=dxp*dxp+dyp*dyp;
    if(d2>=SHIP_R*SHIP_R)continue;
    let d=Math.sqrt(d2),ux,uy;
    if(d<1e-6){ ux=0; uy=ship.y-cy-.5<0?-1:1; d=1e-6; } else { ux=dxp/d; uy=dyp/d; }
    ship.x=nxp+ux*SHIP_R; ship.y=nyp+uy*SHIP_R;
    const vn=ship.vx*ux+ship.vy*uy;
    if(vn<0){ ship.vx-=ux*vn*1.72; ship.vy-=uy*vn*1.72;
      if(-vn>1.6){ S.thump(Math.min(.4,-vn*.09)); shake=Math.min(1,shake+.15); } }
  }
}
function updateRock(rk,dt){
  rk.born=Math.min(1,rk.born+dt*3);
  if(rk.slide){ const s=rk.slide; s.t+=dt/s.dur;
    if(s.t>=1){ rk.px=rk.x+.5; rk.py=rk.y+.5; rk.slide=null; onSettle(rk); }
    else { const e=1-(1-s.t)*(1-s.t); rk.px=lerp(s.fx,rk.x+.5,e); rk.py=lerp(s.fy,rk.y+.5,e); } }
  rk.rot+=rk.vr*dt; rk.vr*=Math.exp(-2.6*dt);
  rk.wobble=Math.max(0,rk.wobble-dt*3.4); rk.heavyT=Math.max(0,rk.heavyT-dt);
}
function tryFire(){
  if(cd>0)return;
  if(energy<SHOOT_COST){ S.deny(); toast('LOW ENERGY'); return; }
  cd=.16; energy-=SHOOT_COST; shots++;
  const a=ship.a,cx=Math.cos(a),cy=Math.sin(a);
  bolts.push({x:ship.x+cx*.5,y:ship.y+cy*.5,vx:cx*BOLT_V+ship.vx*.3,vy:cy*BOLT_V+ship.vy*.3,life:.7});
  for(let i=0;i<3;i++)addPart(ship.x+cx*.55,ship.y+cy*.55,cx*(2+Math.random())+(Math.random()-.5),cy*(2+Math.random())+(Math.random()-.5),.15,.1,'#ffd76a');
  shake=Math.min(1,shake+.08); S.laser(); updateHUD();
}
function update(dt){
  T+=dt;
  if(shake>0){ shake*=Math.exp(-5*dt); if(shake<.002)shake=0; }
  flashT=Math.max(0,flashT-dt*1.7); heavyToastT-=dt;
  for(let i=parts.length-1;i>=0;i--){ const p=parts[i]; p.life-=dt;
    if(p.life<=0){ parts.splice(i,1); continue; }
    if(p.ring){ p.r+=dt*3.2; continue; }
    p.x+=p.vx*dt; p.y+=p.vy*dt; p.vx*=Math.exp(-.8*dt); p.vy*=Math.exp(-.8*dt); }
  pads.forEach(p=>{ p.pulse=Math.max(0,p.pulse-dt*1.5); });
  if(state==='menu'||state==='win'){ for(const d of drifters){ d.x+=d.vx*dt; d.y+=d.vy*dt; d.rot+=d.vr*dt;
      if(d.x<-60)d.x=vw+60; if(d.x>vw+60)d.x=-60; if(d.y<-60)d.y=vh+60; if(d.y>vh+60)d.y=-60; } return; }
  if(state!=='play')return;

  levelTime+=dt; cd-=dt; energy=Math.min(100,energy+45*dt);
  if(K.left)ship.a-=TURN*dt; if(K.right)ship.a+=TURN*dt;
  let thr=false;
  if(K.up){ ship.vx+=Math.cos(ship.a)*THRUST*dt; ship.vy+=Math.sin(ship.a)*THRUST*dt; thr=true; }
  if(thr!==thrustOn){ thrustOn=thr; setThrust(thr); }
  const dmp=Math.exp(-DAMP*dt); ship.vx*=dmp; ship.vy*=dmp;
  const sp=Math.hypot(ship.vx,ship.vy);
  if(sp>MAXV){ ship.vx*=MAXV/sp; ship.vy*=MAXV/sp; }
  ship.x+=ship.vx*dt; ship.y+=ship.vy*dt;
  collideWalls(); collideObstacles();
  for(const rk of rocks)updateRock(rk,dt);
  if(state!=='play')return;

  /* ship ↔ rocks */
  for(const rk of rocks){
    if(rk.slide)continue;
    const dx=rk.px-ship.x, dy=rk.py-ship.y, d=Math.hypot(dx,dy), minD=SHIP_R+rk.r;
    if(d>=minD||d<1e-6)continue;
    const nx=dx/d, ny=dy/d, pen=minD-d;
    ship.x-=nx*pen; ship.y-=ny*pen;
    if(rk.size==='small'){
      const dom=Math.abs(nx)>Math.abs(ny)?[nx>0?1:-1,0]:[0,ny>0?1:-1];
      const tx=rk.x+dom[0], ty=rk.y+dom[1];
      if(cellFree(tx,ty))doPush(rk,tx,ty,nx,ny);
      else{ const vn=ship.vx*nx+ship.vy*ny; if(vn<0){ ship.vx-=nx*vn*1.72; ship.vy-=ny*vn*1.72; }
        rk.wobble=1; S.knock(); shake=Math.min(1,shake+.1); }
    } else {
      const vn=ship.vx*nx+ship.vy*ny; if(vn<0){ ship.vx-=nx*vn*1.72; ship.vy-=ny*vn*1.72; }
      rk.wobble=1; rk.heavyT=.5; S.heavy(); shake=Math.min(1,shake+.2);
      if(heavyToastT<=0){ heavyToastT=1.4; toast('TOO HEAVY — FIRE TO SPLIT'); }
    }
  }

  /* bolts */
  for(let i=bolts.length-1;i>=0;i--){ const b=bolts[i]; let dead=false;
    for(let s2=0;s2<2&&!dead;s2++){
      b.x+=b.vx*dt/2; b.y+=b.vy*dt/2; b.life-=dt/2;
      if(b.life<=0||b.x<-.3||b.y<-.3||b.x>W+.3||b.y>H+.3){ dead=true; break; }
      if(obstacles.has(key(Math.floor(b.x),Math.floor(b.y)))){ dead=true;
        for(let q=0;q<5;q++)addPart(b.x,b.y,(Math.random()-.5)*3,(Math.random()-.5)*3,.25,.1,'#ffd76a'); break; }
      for(const rk of rocks){ const dd=Math.hypot(rk.px-b.x,rk.py-b.y);
        if(dd<rk.r+.06){ dead=true;
          if(rk.size==='small'){ S.tink(); for(let q=0;q<5;q++)addPart(b.x,b.y,(Math.random()-.5)*3,(Math.random()-.5)*3,.25,.1,'#dcecff'); }
          else splitRock(rk,b);
          break; } }
    }
    if(dead)bolts.splice(i,1);
  }
  if(K.fire)tryFire();
}

function drawPoly(c,x,y,r,rot,shape){ c.beginPath(); const n=shape.length;
  for(let i=0;i<n;i++){ const a=rot+i/n*TAU, rr=r*shape[i];
    const px=x+Math.cos(a)*rr, py=y+Math.sin(a)*rr; i?c.lineTo(px,py):c.moveTo(px,py); }
  c.closePath(); }
function drawStars(offX,offY){
  ctx.fillStyle='#ffffff';
  for(const s of starsA){ const x=((s.x+offX)%vw+vw)%vw, y=((s.y+offY*1.4)%vh+vh)%vh;
    ctx.globalAlpha=.25+.4*(.5+.5*Math.sin(T*s.tw+s.ph));
    ctx.beginPath(); ctx.arc(x,y,s.s,0,TAU); ctx.fill(); }
  for(const s of starsB){ const x=((s.x+offX*1.9)%vw+vw)%vw, y=((s.y+offY*2.6)%vh+vh)%vh;
    ctx.globalAlpha=.35+.45*(.5+.5*Math.sin(T*s.tw+s.ph));
    ctx.beginPath(); ctx.arc(x,y,s.s,0,TAU); ctx.fill(); }
  ctx.globalAlpha=1;
}
function drawRockC(c,x,y,r,rot,shape,fill,stroke,lw){
  drawPoly(c,x,y,r,rot,shape); c.fillStyle=fill; c.fill();
  c.lineWidth=lw; c.strokeStyle=stroke; c.stroke(); }
function render(){
  ctx.setTransform(dpr,0,0,dpr,0,0);
  ctx.drawImage(bg,0,0,vw,vh);
  const inPlay=state==='play'||state==='pause'||state==='clear';
  drawStars(inPlay?ship.x*cell*.03:T*5, inPlay?ship.y*cell*.03:T*2);

  if(state==='menu'||state==='win'){
    for(const d of drifters){ drawRockC(ctx,d.x,d.y,d.r,d.rot,d.shape,'#8fa9cf','#dcecff',2);
      ctx.globalAlpha=.5; drawPoly(ctx,d.x,d.y,d.r*.55,d.rot*1.3,d.shape); ctx.fillStyle='#5c74a0'; ctx.fill(); ctx.globalAlpha=1; }
  } else {
    metrics();
    const sx=(Math.random()*2-1)*shake*9, sy=(Math.random()*2-1)*shake*9;
    ctx.save(); ctx.translate(ox+sx,oy+sy);
    /* floor */
    const fg=ctx.createLinearGradient(0,0,0,H*cell); fg.addColorStop(0,'#0d1531'); fg.addColorStop(1,'#0a0f24');
    ctx.fillStyle=fg; ctx.fillRect(WALL*cell,WALL*cell,(W-2*WALL)*cell,(H-2*WALL)*cell);
    ctx.strokeStyle='rgba(125,232,255,.06)'; ctx.lineWidth=1;
    for(let x=1;x<W;x++){ ctx.beginPath(); ctx.moveTo(x*cell,WALL*cell); ctx.lineTo(x*cell,(H-WALL)*cell); ctx.stroke(); }
    for(let y=1;y<H;y++){ ctx.beginPath(); ctx.moveTo(WALL*cell,y*cell); ctx.lineTo((W-WALL)*cell,y*cell); ctx.stroke(); }
    /* walls */
    ctx.strokeStyle='#233459'; ctx.lineWidth=WALL*cell*2; ctx.strokeRect(0,0,W*cell,H*cell);
    ctx.strokeStyle='rgba(125,232,255,.35)'; ctx.lineWidth=2;
    ctx.strokeRect(WALL*cell,WALL*cell,(W-2*WALL)*cell,(H-2*WALL)*cell);
    ctx.fillStyle='#7de8ff';
    [[0,0],[1,0],[0,1],[1,1]].forEach(c2=>{ const px=c2[0]? (W-WALL)*cell: WALL*cell, py=c2[1]? (H-WALL)*cell: WALL*cell;
      ctx.fillRect(px-2.5,py-2.5,5,5); });

    /* pads */
    pads.forEach(p=>{ const px=(p.x+.5)*cell, py=(p.y+.5)*cell, R=cell*.30;
      const pulse=p.covered?1:.45+.2*Math.sin(T*3+p.x*2+p.y);
      ctx.save();
      if(p.covered){ ctx.shadowColor='#57ffa8'; ctx.shadowBlur=16; }
      ctx.strokeStyle=p.covered?'#7dffc4':'rgba(87,255,168,'+pulse+')';
      ctx.lineWidth=Math.max(2,cell*.045);
      ctx.setLineDash([cell*.12,cell*.1]); ctx.lineDashOffset=-T*cell*.3;
      ctx.beginPath(); ctx.arc(px,py,R,0,TAU); ctx.stroke(); ctx.setLineDash([]);
      ctx.fillStyle=p.covered?'#b6ffdd':'rgba(87,255,168,'+(.3+.3*Math.sin(T*3+p.x))+')';
      ctx.beginPath(); ctx.arc(px,py,cell*.07*(1+.25*Math.sin(T*4)),0,TAU); ctx.fill();
      if(p.pulse>0){ ctx.strokeStyle='rgba(87,255,168,'+p.pulse+')'; ctx.lineWidth=3;
        ctx.beginPath(); ctx.arc(px,py,R*(1+(1-p.pulse)*.9),0,TAU); ctx.stroke(); }
      ctx.restore(); });

    /* obstacles */
    const osh=rng(77);
    obstacles.forEach(k=>{ const [gx,gy]=k.split(',').map(Number);
      const px=(gx+.5)*cell, py=(gy+.5)*cell;
      if(!osh._m)osh._m=new Map();
      let sh=osh._m.get(k); if(!sh){ sh=makeShape(gx*731+gy*1549); osh._m.set(k,sh); }
      drawRockC(ctx,px,py+cell*.06,cell*.52,(gx*3.7+gy*1.9),sh,'#1c2846','#314270',Math.max(1.5,cell*.03));
      drawRockC(ctx,px,py,cell*.5,(gx*2.1+gy*4.3),sh,'#2b3a60','#4d6398',Math.max(1.5,cell*.03)); });

    /* rocks */
    for(const rk of rocks){
      const px=rk.px*cell, py=rk.py*cell;
      const born=rk.born<1? (t=>1+2.7*(t-1)*(t-1)*(t-1)+1.70158*(t-1)*(t-1))(rk.born) : 1;
      const wob=1+.06*rk.wobble*Math.sin(T*34+rk.seed);
      const r=rk.r*cell*wob*Math.max(.05,born);
      const st=RSTYLE[rk.size];
      if(rk.docked){ const gg=ctx.createRadialGradient(px,py,0,px,py,r*1.9);
        gg.addColorStop(0,'rgba(87,255,168,.4)'); gg.addColorStop(1,'rgba(87,255,168,0)');
        ctx.fillStyle=gg; ctx.beginPath(); ctx.arc(px,py,r*1.9,0,TAU); ctx.fill(); }
      drawRockC(ctx,px,py+cell*.03,r*1.02,rk.rot,rk.shape,'#0e1526','rgba(0,0,0,.4)',cell*.05);
      drawRockC(ctx,px,py,r,rk.rot,rk.shape,st.fill,st.stroke,Math.max(1.5,cell*.035));
      if(rk.size!=='small'){ ctx.strokeStyle='rgba(20,10,5,.4)'; ctx.lineWidth=Math.max(1,cell*.02);
        for(let i=0;i<2;i++){ const a=rk.rot+rk.shape[i*3||0]*TAU;
          ctx.beginPath(); ctx.moveTo(px+Math.cos(a)*r*.7,py+Math.sin(a)*r*.7);
          ctx.lineTo(px-Math.cos(a+1.4)*r*.55,py-Math.sin(a+1.4)*r*.55); ctx.stroke(); } }
      if(rk.heavyT>0){ drawPoly(ctx,px,py,r,rk.rot,rk.shape);
        ctx.fillStyle='rgba(255,80,60,'+(rk.heavyT*.7)+')'; ctx.fill(); }
      if(rk.docked){ ctx.fillStyle='#57ffa8'; ctx.beginPath(); ctx.arc(px,py,cell*.09*(1+.2*Math.sin(T*5)),0,TAU); ctx.fill(); }
    }

    /* bolts */
    for(const b of bolts){ const l=Math.hypot(b.vx,b.vy)||1, ux=b.vx/l, uy=b.vy/l, L2=cell*.3;
      ctx.lineCap='round';
      ctx.strokeStyle='rgba(255,215,106,.4)'; ctx.lineWidth=6;
      ctx.beginPath(); ctx.moveTo(b.x*cell,b.y*cell); ctx.lineTo((b.x-ux*L2)*cell,(b.y-uy*L2)*cell); ctx.stroke();
      ctx.strokeStyle='#fff7d9'; ctx.lineWidth=2.5;
      ctx.beginPath(); ctx.moveTo(b.x*cell,b.y*cell); ctx.lineTo((b.x-ux*L2)*cell,(b.y-uy*L2)*cell); ctx.stroke(); }

    /* ship */
    { const px=ship.x*cell, py=ship.y*cell, L=cell*.55;
      if(thrustOn){ for(let i=0;i<2;i++)addPart(ship.x-Math.cos(ship.a)*.34,ship.y-Math.sin(ship.a)*.34,
          -Math.cos(ship.a)*(1.4+Math.random())+(Math.random()-.5),-Math.sin(ship.a)*(1.4+Math.random())+(Math.random()-.5),
          .25+Math.random()*.2,.1,['#ffd76a','#ff9d5c','#ff5d3a'][i%3]); }
      ctx.save(); ctx.translate(px,py); ctx.rotate(ship.a);
      if(thrustOn){ const fl=L*(.5+Math.random()*.4);
        ctx.fillStyle='rgba(255,150,60,.8)';
        ctx.beginPath(); ctx.moveTo(-L*.42,L*.17); ctx.lineTo(-L*.42-fl,0); ctx.lineTo(-L*.42,-L*.17); ctx.closePath(); ctx.fill();
        ctx.fillStyle='rgba(255,235,150,.9)';
        ctx.beginPath(); ctx.moveTo(-L*.42,L*.08); ctx.lineTo(-L*.42-fl*.55,0); ctx.lineTo(-L*.42,-L*.08); ctx.closePath(); ctx.fill(); }
      drawPoly(ctx,0,0,1,0,[1,1,1,1]); /* noop guard */
      ctx.beginPath(); ctx.moveTo(L,0); ctx.lineTo(-L*.55,L*.6); ctx.lineTo(-L*.3,L*.24); ctx.lineTo(-L*.55,-L*.6); ctx.closePath();
      ctx.fillStyle='#f2f7ff'; ctx.fill();
      ctx.lineWidth=Math.max(1.5,cell*.035); ctx.strokeStyle='rgba(125,232,255,.95)'; ctx.stroke();
      ctx.fillStyle='#7de8ff'; ctx.beginPath(); ctx.arc(L*.28,0,cell*.05,0,TAU); ctx.fill();
      ctx.restore(); }

    /* particles (world space) */
    for(const p of parts){ const a=p.life/p.max;
      if(p.ring){ ctx.strokeStyle=p.color; ctx.globalAlpha=a; ctx.lineWidth=2.5;
        ctx.beginPath(); ctx.arc(p.x*cell,p.y*cell,p.r*cell,0,TAU); ctx.stroke(); continue; }
      ctx.globalAlpha=a; ctx.fillStyle=p.color;
      ctx.beginPath(); ctx.arc(p.x*cell,p.y*cell,Math.max(.6,p.size*cell*a),0,TAU); ctx.fill(); }
    ctx.globalAlpha=1;
    ctx.restore();
  }
  if(flashT>0){ ctx.fillStyle='rgba(235,250,255,'+flashT*.7+')'; ctx.fillRect(0,0,vw,vh); }
}
