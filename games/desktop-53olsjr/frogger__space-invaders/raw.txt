function randX(li){let x=20+R()*(FW-40),p=G.player;

function updateLanes(dt){
  for(const lane of G.lanes){const m=lane.ents[0].wb;
    for(const e of lane.ents){
      if(!e.alive){e.rT-=dt;if(e.rT<=0){e.alive=true;e.x=randX(lane.i);}}
      else{e.x+=e.sp*lane.dir*dt;
        if(lane.dir>0&&e.x>FW+m)e.x=-m;else if(lane.dir<0&&e.x<-m)e.x=FW+m;}
    }}
}
function update(dt){
  T+=dt;
  for(const s of stars){s.y+=dt*0.006*s.z;if(s.y>1)s.y-=1;}
  if(G.state==='paused')return;
  G.shake=Math.max(0,G.shake-26*dt);
  // particles / rings / popups always tick
  for(let i=G.parts.length-1;i>=0;i--){const p=G.parts[i];p.t+=dt;
    if(p.t>p.life){G.parts.splice(i,1);continue;}
    const dr=Math.exp(-2.5*dt);p.vx*=dr;p.vy*=dr;p.x+=p.vx*dt;p.y+=p.vy*dt;}
  for(let i=G.rings.length-1;i>=0;i--){const r=G.rings[i];r.t+=dt;r.r+=r.vr*dt;if(r.t>r.life)G.rings.splice(i,1);}
  for(let i=G.pops.length-1;i>=0;i--){const p=G.pops[i];p.t+=dt;p.y-=30*dt;if(p.t>p.life)G.pops.splice(i,1);}
  G.pads.forEach(p=>{if(p.occ)p.bt+=dt;});

  if(G.state==='menu'){updateLanes(dt*0.5);return;}
  if(G.state!=='playing'&&G.state!=='waveclear')return;

  updateLanes(dt);
  if(G.state!=='playing')return;
  updateMusic();
  G.hintT-=dt;G.bannerT-=dt;G.invuln-=dt;

  const p=G.player;
  if(p.dead){G.deadT-=dt;
    if(G.deadT<=0){if(G.lives<=0){saveHi();G.state='gameover';sfx.over();}else{resetPlayer();G.invuln=1.6;}}
    return;}
  // horizontal
  const dir=(keys.has('ArrowRight')||keys.has('KeyD'))-(keys.has('ArrowLeft')||keys.has('KeyA'));
  p.vx+=(dir*270-p.vx)*Math.min(1,dt*18);
  p.x=clamp(p.x+p.vx*dt,14,FW-14);
  // hop tween + hold-to-cross
  if(p.hop.a){p.hop.t+=dt/0.13;
    if(p.hop.t>=1){p.row=p.hop.to;p.hop.a=false;G.landT=T;
      for(let k=0;k<4;k++)G.parts.push({x:p.x+(R()*16-8),y:yRow(p.row)+9,vx:R()*30-15,vy:-25,t:0,life:0.3,col:'rgba(140,255,170,0.8)',sz:2});}
  }else if(!p.docked&&T-G.landT>0.12){
    const up=keys.has('ArrowUp')||keys.has('KeyW'),dn=keys.has('ArrowDown')||keys.has('KeyS');
    if(up&&p.row<LN+1)hop(1);else if(dn&&p.row>0)hop(-1);}
  // lane bonus
  if(!p.hop.a&&p.row>G.crossMax&&p.row<=LN){G.crossMax=p.row;
    addScore(LB[p.row],p.x,py()-20,'#9fd8ff','+'+LB[p.row]);sfx.lane(p.row);}
  // dock
  if(!p.docked&&p.row===LN+1&&!p.hop.a){
    for(let i=0;i<4;i++){if(!G.pads[i].occ&&Math.abs(p.x-PADX[i])<30){
      p.x=PADX[i];G.pads[i].occ=true;G.pads[i].bt=0;p.docked=true;p.padIdx=i;
      const b=100+G.wave*20;addScore(b,p.x,py()-26,'#7dff9a','+'+b);
      G.rings.push({x:PADX[i],y:yRow(LN+1),r:6,vr:180,t:0,life:0.5,col:'#7dff9a'});
      sfx.dock();
      if(G.pads.every(q=>q.occ)){G.state='waveclear';G.wcT=0;
        addScore(200,FW/2,FH*0.4,'#ffe08a','+200 BONUS');sfx.wave();}
      break;}}}
  // collisions
  if(G.invuln<=0){const pyy=py();outer:
    for(let i=0;i<LN;i++){const ly=yRow(i+1);
      for(const e of G.lanes[i].ents){if(!e.alive)continue;
        if(Math.abs(pyy-ly)<7.5+e.eh/2&&Math.abs(p.x-e.x)<13+e.wb/2){die();break outer;}}}}

  // bolts
  const u=G.ufo;
  for(const b of G.bolts){b.y-=950*dt;
    if(u.a&&Math.abs(b.x-u.x)<36&&b.y<48+10&&b.y>48-18){killUfo();b.dead=true;break;}
    for(let i=0;i<LN;i++){const ly=yRow(i+1);
      if(b.y+7<ly-24||b.y-7>ly+24)continue;
      for(const e of G.lanes[i].ents){if(!e.alive)continue;
        if(Math.abs(b.x-e.x)<e.wb/2&&b.y+7>ly-e.eh/2&&b.y-7<ly+e.eh/2){killEnt(i,e);b.dead=true;break;}}
      if(b.dead)break;}
    if(b.y<-30)b.dead=true;}
  G.bolts=G.bolts.filter(b=>!b.dead);

  // ufo
  if(u.a){u.x+=u.dir*140*dt;u.tk-=dt;
    if(u.tk<=0){sfx.ufoTick(u.hi=!u.hi);u.tk=0.13;}
    if(u.x<-60||u.x>FW+60){u.a=false;u.tmr=9+R()*8;}}
  else{u.tmr-=dt;if(u.tmr<=0){u.a=true;u.dir=R()<0.5?1:-1;u.x=u.dir>0?-40:FW+40;u.pts=R()<0.3?100:50;u.tk=0;}}
}
