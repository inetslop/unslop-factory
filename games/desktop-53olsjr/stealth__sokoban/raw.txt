Now the guard behavior model:
- Patroller: moves continuously along path points (pixel space), speed in tiles/sec (~1.6). At each waypoint, dwells for `dwell` seconds while sweeping view from -sweep to +sweep around base facing. Between waypoints, faces movement direction.
- Sentinel: fixed position; continuously sweeps ±sweep around baseAngle with sinusoidal oscillation (period ~3.5s).

Detection math (per frame):
For each guard: compute vector to player center (pixels), distance d (in tiles), angle between facing and vector. If d <= range && |angleDelta| <= fov/2 && LOS clear → visible. LOS: sample along the line every 0.15 tile; if any sample is inside a wall cell → blocked. Also, if player is in shadow → not detected at all (full concealment) — simpler and clearer gameplay. Show an indicator when seen.

Suspicion: rises with rate = base * (1 - d/range*0.6) * (crouch? 0.45 : 1). Let's say max ~90/sec at close range → about 1s to get caught up close, ~3-4s at range. Decays ~25/sec when not seen (only starts decaying after grace of 0.4s). Suspicion ≥ 100 → CAUGHT.

Crouch: hold Shift — movement speed × 0.5, detection range halved (d vs range*0.5), suspicion rise × 0.45, and noise? Also crouching hides from sentinels?   }else{startTween(player,nx,ny,player.crouch?.2:.13);sfx.step(player.crouch);moves++;}
}
function undo(){
  if(state!=='playing'||player.moving||boxesMoving())return;
  const h=hist.pop();if(!h)return;
  player.x=player.y=0;player.fx=h.px;player.fy=h.py;player.tx=h.px;player.ty=h.py;player.x=h.px;player.y=h.py;player.moving=false;player.t=1;
  const b=boxes.find(b=>b.x===h.bx1&&b.y===h.by1);
  if(b){b.x=b.y=0;b.fx=h.bx0;b.fy=h.by0;b.tx=h.bx0;b.ty=h.by0;b.x=h.bx0;b.y=h.by0;b.moving=false;b.t=1;}
  moves=Math.max(0,moves-1);sfx.undo();
}
function winCheck(){
  if(state!=='playing')return;
  if(boxes.every(b=>isOnPad(b.x,b.y))){state='clearing';stateT=0;sfx.clear();
    pads.forEach(p=>confettiBurst(p.x+.5,p.y+.5));totalTime+=levelTime;totalMoves+=moves;}
}

/* ============================== GUARDS & DETECTION ============================== */
function stepGuards(dt){
  for(const g of guards){
    if(g.k==='s'){g.angle=g.base+g.sweep*Math.sin(nowT*TAU/g.per+g.seed);continue}
    const dx=g.b.x-g.a.x,dy=g.b.y-g.a.y,len=Math.hypot(dx,dy);
    if(g.ph==='dwell'){g.dwell-=dt;const prog=1-g.dwell/DWELL;
      g.angle=g.face+g.sweep*Math.sin(prog*TAU);
      if(g.dwell<=0){const t=g.a;g.a=g.b;g.b=t;g.ph='go';g.face=Math.atan2(g.b.y-g.a.y,g.b.x-g.a.x);g.angle=g.face;}
    }else{
      const step=g.sp*dt,dist=Math.hypot(g.b.x-g.pos.x,g.b.y-g.pos.y);
      if(dist<=step){g.pos.x=g.b.x;g.pos.y=g.b.y;g.ph='dwell';g.dwell=DWELL;}
      else{g.pos.x+=dx/len*step;g.pos.y+=dy/len*step;}
      g.angle=g.face;
    }
    if(g.spotT>0)g.spotT-=dt;
  }
}
function visPos(o){if(!o.moving)return{x:o.x,y:o.y};const t=o.t*t*t*(3-2*t);return{x:lerp(o.fx,o.x,t),y:lerp(o.fy,o.y,t)}}
function losClear(a,b){const dx=b.x-a.x,dy=b.y-a.y,d=Math.hypot(dx,dy);
  for(let s=.1;s<d;s+=.12)if(isSolid(a.x+dx/d*s,a.y+dy/d*s))return false;return true}
function detect(dt){
  const pv=visPos(player),inSh=cellAt(Math.round(pv.x-.5),Math.round(pv.y-.5))===',';
  seenNow=false;lastSeer=null;minD=99;
  if(!inSh){
    for(const g of guards){
      const dx=pv.x-g.pos.x,dy=pv.y-g.pos.y,d=Math.hypot(dx,dy);
      if(d<.72){seenNow=true;lastSeer=g;minD=Math.min(minD,d);g.spotT=.3;continue}
      const eff=RANGE*(player.crouch?CROUCH_R:1);
      if(d>eff+.05)continue;
      if(Math.abs(angDiff(Math.atan2(dy,dx),g.angle))>FOV/2)continue;
      if(!losClear(g.pos,pv))continue;
      seenNow=true;lastSeer=g;minD=Math.min(minD,d);g.spotT=.3;
    }
  }
  if(seenNow){
    const t=clamp(1-minD/RANGE,0,1),rate=(16+72*t)*(player.crouch?.4:1);
    suspicion=clamp(suspicion+rate*dt,0,100);graceT=.35;
    const lvl2=Math.floor(suspicion/10);if(lvl2>lastTick){sfx.tick(lvl2);lastTick=lvl2;}
  }else{graceT-=dt;if(graceT<=0){suspicion=clamp(suspicion-30*dt,0,100);lastTick=Math.floor(suspicion/10)}}
  if(suspicion>55){hbT-=dt;if(hbT<=0){sfx.heart(.2+(suspicion-55)/45*.3);hbT=.9-suspicion/100*.28}}
  if(suspicion>=100)caught();
}
function caught(){
  if(state!=='playing')return;
  player.moving=false;player.t=1;boxes.forEach(b=>{b.moving=false;b.t=1});
  state='caught';stateT=0;suspicion=100;sfx.alarm();shake=1;
}

function addP(x,y,vx,vy,life,size,c,kind){particles.push({x,y,vx,vy,life,max:life,size,c,kind,rot:Math.random()*TAU})}

  addP(x,y,Math.cos(a)*.5,Math.sin(a)*.5-.2,.35+Math.random()*.15,.05+Math.random()*.04,'rgba(170,180,195,','puff')}}
function addSparkle(x,y){const a=Math.random()*TAU;addP(x+Math.cos(a)*.2,y+Math.sin(a)*.2,-.1,.15,.5,.06,'#ffd9a0','spark')}
function confettiBurst(x,y){const cs=['#ffb454','#5fe3cf','#e8f1ef'];
  for(let i=0;i<26;i++){const a=Math.random()*TAU,sp=1+Math.random()*2.4;
    addP(x,y,Math.cos(a)*sp,-Math.abs(Math.sin(a))*sp-1.5,1+Math.random()*.6,.07+Math.random()*.05,cs[i%3],'conf')}}
function updateParticles(dt){
  let dust=0;for(const p of particles)if(p.kind==='dust')dust++;
  if(dust<24&&Math.random()<dt*8&&state!=='paused'){
    addP(Math.random()*cols,Math.random()*rows,-.06-Math.random()*.1,-.03-Math.random()*.05,4+Math.random()*3,.02+Math.random()*.02,'rgba(200,220,235,','dust');}
  for(let i=particles.length-1;i>=0;i--){const p=particles[i];
    p.life-=dt;if(p.life<=0){particles.splice(i,1);continue}
    if(p.kind==='conf')p.vy+=4.5*dt;
    p.x+=p.vx*dt;p.y+=p.vy*dt;p.rot+=dt*3;}
  const pv=visPos(player);
  if(state==='playing'&&cellAt(Math.round(pv.x-.5),Math.round(pv.y-.5))===','&&Math.random()<dt*5)addSparkle(pv.x,pv.y);
}

/* ============================== UPDATE ============================== */
function updatePlaying(dt){
  levelTime+=dt;
  if(player.moving){player.t+=dt/player.dur;
    if(player.t>=1){player.t=1;player.moving=false;
      if(buffer&&!boxesMoving()){const d=buffer;buffer=null;tryMove(d[0],d[1]);}}}
  for(const b of boxes){
    if(b.moving){b.t+=dt/.14;
      if(b.t>=1){b.t=1;b.moving=false;b.onP=isOnPad(b.x,b.y);
        if(b.onP&&!b.wasOn){sfx.ding();b.pop=.0001;}
        else if(!b.onP&&b.wasOn)sfx.offpad();
        b.wasOn=b.onP;winCheck();}}
    if(b.pop>0)b.pop=Math.min(.35,b.pop+dt);}
  stepGuards(dt);detect(dt);
  if(player.bump){player.bump.t+=dt;if(player.bump.t>.14)player.bump=null}
  shake=Math.max(0,shake-dt*2.2);
}
function updateWorldAlive(dt){stepGuards(dt);
  for(const b of boxes)if(b.moving)b.t=1;
  if(shake>0)shake=Math.max(0,shake-dt*2.2);}
