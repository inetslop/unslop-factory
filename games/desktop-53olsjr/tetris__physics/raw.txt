function recomputeWorld(b){const co=Math.cos(b.ang),si=Math.sin(b.ang);b.sq.length=0;
  let mnx=1e9,mny=1e9,mxx=-1e9,mxy=-1e9;
  for(let i=0;i<b.cells.length;i++){const c=b.cells[i];
    const wx=b.cx+co*c.x-si*c.y, wy=b.cy+si*c.x+co*c.y;
    const box={cx:wx,cy:wy,ang:b.ang,hw:b.hw,hh:b.hh};box.verts=boxCorners(box);b.sq.push(box);
    for(let v=0;v<4;v++){const p=box.verts[v];
      if(p.x<mnx)mnx=p.x;if(p.x>mxx)mxx=p.x;if(p.y<mny)mny=p.y;if(p.y>mxy)mxy=p.y;}}
  b.aabb={minx:mnx,miny:mny,maxx:mxx,maxy:mxy};}

function refill(){while(queue.length<4){ if(!bag.length){bag=[0,1,2,3,4,5,6];
    for(let i=bag.length-1;i>0;i--){const j=(Math.random()*(i+1))|0;const t=bag[i];bag[i]=bag[j];bag[j]=t;}}
  queue.push(bag.pop());}}
function spawnActive(){refill();const ti=queue.shift();refill();
  active=makePiece(ti,W/2+(Math.random()-0.5)*2.6,-1.3,(Math.random()-0.5)*0.6,2,(Math.random()-0.5)*1.4);bodies.push(active);}
function autoSpawn(){const ti=(Math.random()*7)|0;
  const b=makePiece(ti,2.5+Math.random()*(W-5),-1.6,(Math.random()-0.5)*1.8,1.5,(Math.random()-0.5)*4);bodies.push(b);}
function dynCount(){let n=0;for(let i=0;i<bodies.length;i++)if(!bodies[i].st)n++;return n;}

  const T=TYPES[b.type];recomputeWorld(b);
  for(let s=0;s<b.sq.length;s++){const sq=b.sq[s];
    for(let k=0;k<7;k++){const a=Math.random()*TAU,sp=1.5+Math.random()*6;
      shards.push({x:sq.cx+(Math.random()-0.5)*0.7,y:sq.cy+(Math.random()-0.5)*0.7,
        vx:Math.cos(a)*sp,vy:Math.sin(a)*sp-2.5,rot:Math.random()*TAU,w:(Math.random()-0.5)*16,
        size:0.12+Math.random()*0.18,life:0,maxLife:0.45+Math.random()*0.4,col:T.c});}}
  rings.push({x:b.cx,y:b.cy,r:0.5,vr:10,life:0.45,maxLife:0.45,col:'#ffffff'});
  flashT=0.3;ffT=0.5;ffCol=T.c;shake=Math.min(14,shake+3);
  if(b===active){active=null;canHold=true;spawnTimer=Math.min(spawnTimer,0.3);}
  if(state==='play'){
    combo=(gameTime-lastShatterT<COMBO_WIN)?combo+1:1;lastShatterT=gameTime;maxCombo=Math.max(maxCombo,combo);
    const hf=clamp((H*0.98-b.cy)/(H*0.98-DANGER_Y),0,1);

    const nl=1+Math.floor(clears/6);
    if(nl>level){level=nl;floats.push({x:W/2,y:H*0.45,txt:'LEVEL '+nl,col:'#3ce8dd',t:0,dur:1.4,size:0.9});audio.level();}
  }
  audio.shatter(Math.max(1,combo));
  if(shards.length>380)shards.splice(0,shards.length-380);}

const manifolds=[],pendingImp=[];
let lastImpactT=0;
function step(h,act){
  for(let i=0;i<bodies.length;i++){const b=bodies[i];if(b.st)continue;
    b.vy+=grav*h;
    const d=Math.max(0,1-AIR*h);b.vx*=d;b.vy*=d;b.w*=(1-ANGD*h);
    if(b===act){
      const dir=((keys.ArrowRight||keys.KeyD)?1:0)-((keys.ArrowLeft||keys.KeyA)?1:0);
      if(dir){const tv=dir*MOVE_V;b.vx+=clamp(tv-b.vx,-MOVE_ACC*h,MOVE_ACC*h);}
      if(keys.ArrowDown||keys.KeyS)b.vy=Math.min(b.vy+SOFT_G*h,15);}
    if(b.spinTarget!=null){
      const dl=normAng(b.spinTarget-b.ang);
      if(Math.abs(dl)<0.1&&Math.abs(b.w)<3){b.ang=b.spinTarget;b.w*=0.25;b.spinTarget=null;}
      else{b.w+=clamp(dl*SPINK,-MAXW,MAXW)*h;b.w*=(1-SPINDAMP*h);}}
    b.vx=clamp(b.vx,-60,60);b.vy=clamp(b.vy,-55,55);b.w=clamp(b.w,-30,30);}
  for(let i=0;i<bodies.length;i++)recomputeWorld(bodies[i]);
  manifolds.length=0;pendingImp.length=0;
  const n=bodies.length;
  for(let i=0;i<n;i++){const A=bodies[i];
    for(let j=i+1;j<n;j++){const B=bodies[j];
      if(A.st&&B.st)continue;
      const a=A.aabb,c=B.aabb;
      if(a.maxx<c.minx||a.minx>c.maxx||a.maxy<c.miny||a.miny>c.maxy)continue;
      for(let p=0;p<A.sq.length;p++)for(let q=0;q<B.sq.length;q++){
        const m=collideOBB(A.sq[p],B.sq[q]);if(m){m.a=A;m.b=B;manifolds.push(m);}}}}
  for(let it=0;it<ITER;it++)for(let i=0;i<manifolds.length;i++)solveM(manifolds[i]);
  for(let i=0;i<bodies.length;i++){const b=bodies[i];if(b.st)continue;
    b.cx+=b.vx*h;b.cy+=b.vy*h;b.ang+=b.w*h;}
  for(let i=0;i<manifolds.length;i++){const m=manifolds[i],a=m.a,b2=m.b;let pen=0;
    for(let k=0;k<m.pts.length;k++)if(m.pts[k].pen>pen)pen=m.pts[k].pen;
    const corr=Math.max(pen-SLOP,0)/(a.im+b2.im)*POS_PCT;
    a.cx-=m.n.x*corr*a.im;a.cy-=m.n.y*corr*a.im;b2.cx+=m.n.x*corr*b2.im;b2.cy+=m.n.y*corr*b2.im;}
  let bestI=0;for(let i=0;i<pendingImp.length;i++)if(pendingImp[i].s>bestI)bestI=pendingImp[i].s;
  if(bestI>9&&performance.now()-lastImpactT>80){lastImpactT=performance.now();
    audio.impact(Math.min(bestI,60));shake=Math.min(14,shake+(bestI>25?4:(bestI>15?2.5:1)));}}

/* ============================ UPDATE ================================ */
function update(dt){
  if(state==='menu'){menuTimer-=dt;if(menuTimer<=0&&dynCount()<14){autoSpawn();menuTimer=1.05;}}
  else if(state==='play'){gameTime+=dt;spawnTimer-=dt;
    if(spawnTimer<=0&&dynCount()<14){spawnActive();spawnTimer=Math.max(0.8,2.4-(level-1)*0.17);}}
  const h=1/90;let ns=0;acc+=dt;
  while(acc>=h&&ns<5){step(h,(state==='play')?active:null);acc-=h;ns++;}
  if(ns===5)acc=0;
  if(state==='play'||state==='menu'){
    for(let i=bodies.length-1;i>=0;i--){const b=bodies[i];if(b.st)continue;
      b.grace=Math.max(0,b.grace-dt);b.slamT=Math.max(0,b.slamT-dt);
      const sp2=b.vx*b.vx+b.vy*b.vy;
      if(sp2<STILL_V*STILL_V&&b.w*b.w<STILL_W*STILL_W){b.still+=dt;
        if(b.still>0.9&&Math.random()<dt*7){const sq=b.sq[(Math.random()*b.sq.length)|0];
          shards.push({x:sq.cx,y:sq.cy-0.3,vx:(Math.random()-0.5)*1.2,vy:-1.5,rot:0,w:4,
            size:0.07,life:0,maxLife:0.5,col:'#ffffff'});}}
      else b.still=Math.max(0,b.still-dt*2);
      if(b.still>=SHATTER_T)shatter(b);}
    if(state==='play'){
      let top=Infinity;
      for(let i=0;i<bodies.length;i++){const b=bodies[i];if(b.st||b.grace>0)continue;if(b.aabb.miny<top)top=b.aabb.miny;}
      if(top<DANGER_Y)overload+=dt;else overload=Math.max(0,overload-2*dt);
      if(overload>0){warnT-=dt;if(warnT<=0){audio.warn();warnT=0.7;}}else warnT=0;
      if(overload>=4)gameOver();}}}

function effectsUpdate(dt){
  flashT=Math.max(0,flashT-dt*2.2);ffT=Math.max(0,ffT-dt*1.6);
  for(let i=shards.length-1;i>=0;i--){const p=shards[i];p.life+=dt;
    if(p.life>p.maxLife){shards.splice(i,1);continue;}
    p.vy+=24*dt;p.vx*=(1-1.6*dt);p.x+=p.vx*dt;p.y+=p.vy*dt;p.rot+=p.w*dt;}
  for(let i=rings.length-1;i>=0;i--){const r=rings[i];r.life-=dt;
    if(r.life<=0){rings.splice(i,1);continue;}r.r+=r.vr*dt;}
  for(let i=floats.length-1;i>=0;i--){const f=floats[i];f.t+=dt;if(f.t>f.dur)floats.splice(i,1);}}

function layout(){
  const vw=innerWidth,vh=innerHeight,pad=Math.max(12,Math.min(vw,vh)*0.02);
  if(vw>=780){
    const panelW=clamp(vw*0.16,150,235),gap=20;
    const availW=vw-2*(pad+panelW+gap),availH=vh-2*pad;
    const s=Math.min(availW/W,availH/H);const w=W*s,h=H*s;
    return {mode:'wide',s:s,w:w,h:h,x:(vw-w)/2,y:(vh-h)/2,panelW:panelW};}
  const TH=Math.max(58,vh*0.09);
  const availW=vw-2*pad,availH=vh-TH-2*pad;
  const s=Math.min(availW/W,availH/H);const w=W*s,h=H*s;
  return {mode:'narrow',s:s,w:w,h:h,x:(vw-w)/2,y:TH+((vh-TH)-h)/2,panelW:0};}
