  world.dist=0; world.speed=13;
  pillars.length=0; cores.length=0; parts.length=0; pops.length=0;
  cursor=40; ship.alive=true; ship.x=world.camX; ship.boosting=false;
  pauseOv.classList.remove('show'); overOv.classList.remove('show');
  menuOv.classList.add('show'); hud.classList.add('hidden');
  if(best>0){ menuBest.textContent='BEST · '+best+' m'; menuBest.classList.remove('hidden'); } else menuBest.classList.add('hidden');
}
function pause(){ if(state!=='play') return; state='pause'; pauseOv.classList.add('show'); }
function resume(){ if(state!=='pause') return; state='play'; pauseOv.classList.remove('show'); }
function toggleMute(){
  muted=!muted; saveNum('emberMute',muted?1:0);
  if(master) master.gain.value=muted?0:0.55;
  muteBtn.textContent=muted?'SOUND OFF':'SOUND ON';
}

/* ============================== STEP ============================== */
function step(dt,rdt){
  shake=Math.max(0,shake*(1-Math.min(1,rdt*5))-rdt*2);
  flash=Math.max(0,flash-rdt*2.2);
  for(const p of pops) p.life-=rdt;
  pops=pops.filter(p=>p.life>0);
  if(state==='pause') return;

  if(state==='menu'){
    world.speed+=(13-world.speed)*Math.min(1,dt*2);
    const tx=Math.sin(clk*0.45)*2.4;
    world.camX+=(tx-world.camX)*Math.min(1,dt*2.5);
    ship.x=world.camX; ship.vx=0;
  }else if(state==='play'){
    /* steering */
    if(ship.touch!=null){
      const tv=clamp(ship.touch,-7.2,7.2);
      ship.vx+=((tv-ship.x)*10-ship.vx)*Math.min(1,dt*8);
    }else{
      const dir=(keys.R?1:0)-(keys.L?1:0);
      ship.vx+=(dir*13-ship.vx)*Math.min(1,dt*9);
    }
    ship.x+=ship.vx*dt;
    if(ship.x>7.2){ ship.x=7.2; ship.vx=Math.min(0,ship.vx)*-0.3; }
    if(ship.x<-7.2){ ship.x=-7.2; ship.vx=Math.max(0,ship.vx)*-0.3; }
    ship.bank+=(clamp(ship.vx/13,-1,1)-ship.bank)*Math.min(1,dt*8);
    /* boost */
    const want=(keys.B||tBoost)&&ship.alive;
    ship.boosting=want&&ship.boost>0;
    if(ship.boosting) ship.boost=Math.max(0,ship.boost-0.42*dt);
    else ship.boost=Math.min(1,ship.boost+0.05*dt);
    if(ship.boosting&&!prevB) whoosh(0.9);
    prevB=ship.boosting;
    /* speed */
    const base=17+Math.min(29,world.dist*0.016);
    const target=base*(ship.boosting?1.75:1)*(ship.alive?1:0.35);
    world.speed+=(target-world.speed)*Math.min(1,dt*(ship.boosting?3.4:1.9));
    maxSpeed=Math.max(maxSpeed,world.speed);
    world.camX+=(ship.x*0.55-world.camX)*Math.min(1,dt*6);
  }else if(state==='over'){
    world.speed+=(7-world.speed)*Math.min(1,dt*2);
    world.camX+=(0-world.camX)*Math.min(1,dt*1.5);
  }

  world.dist+=world.speed*dt;

  /* spawn ahead */
  const ambient=state!=='play';
  while(cursor<world.dist+FAR){
    const rel=cursor-world.dist;
    if(ambient) spawnAmbient(rel); else spawnChunk(rel);
    cursor += ambient ? (36+Math.random()*24) : Math.max(26, world.speed*1.15+Math.random()*12);
  }

  /* move + cull */
  for(const p of pillars){ p.prevZ=p.z; p.z-=world.speed*dt; }
  for(const c of cores){ c.prevZ=c.z; c.z-=world.speed*dt; }
  pillars=pillars.filter(p=>p.z>-3);
  cores=cores.filter(c=>!c.taken&&c.z>-3);

  /* particles */
  const dr=Math.exp(-1.6*dt);
  for(const q of parts){
    q.x+=q.vx*dt; q.y+=q.vy*dt; q.z+=q.vz*dt-world.speed*dt;
    q.vy-=5.5*dt; q.vx*=dr; q.vy*=dr; q.vz*=dr;
    if(q.y<0.04){ q.y=0.04; q.vy*=-0.35; }
    q.life-=dt;
  }
  parts=parts.filter(q=>q.life>0&&q.z>-2);

  /* play-only */
  if(state==='play'){
    if(ship.alive){
      const tn=ship.boosting?2:1;
      for(let i=0;i<tn;i++){ const lf=0.35+Math.random()*0.25;
        parts.push({x:ship.x+(Math.random()-0.5)*0.3,y:ship.y-0.15+(Math.random()-0.5)*0.1,z:PZ-0.8-Math.random()*0.4,
          vx:(Math.random()-0.5)*0.6,vy:-0.3-Math.random()*0.4,vz:-0.5,life:lf,max:lf,size:0.07+Math.random()*0.05,
          col:Math.random()<0.5?[255,190,110]:[255,235,200]}); }
      for(const p of pillars){
        if(p.prevZ>PZ&&p.z<=PZ){
          const dx=Math.abs(ship.x-p.x);
          if(dx<p.w/2+0.5){ crash(); break; }
          else if(dx<p.w/2+1.8) nearMiss(p);
        }
      }
      if(ship.alive){
        for(const c of cores){
          const hit=(c.prevZ>PZ&&c.z<=PZ)||Math.abs(c.z-PZ)<1.5;
          if(hit&&Math.abs(ship.x-c.x)<1.3) collectCore(c);
        }
      }
      const n2=Math.floor(world.dist/300);
      if(n2>mileN){ mileN=n2; popCenter('VELOCITY RISING','#ffb85c',20); blip(320,640,0.07); }
    }else{
      crashT+=rdt;
      if(crashT>1.15) showOver();
    }
  }
}

/* ============================== RENDER ============================== */
function drawSky(){
  const g=ctx.createLinearGradient(0,0,0,HOR+2);
  g.addColorStop(0,'#150911'); g.addColorStop(0.5,'#38121c'); g.addColorStop(0.84,'#7c2418'); g.addColorStop(1,'rgb(255,140,78)');
  ctx.fillStyle=g; ctx.fillRect(-12,-12,W+24,HOR+13);
}
function drawStars(){
  for(const st of stars){
    const sy=st.y*HOR*0.85;
    const a=(1-sy/HOR)*(0.3+0.3*Math.sin(clk*2+st.p));
    if(a<=0.03) continue;
    const sx=((st.x*W-world.camX*8)%W+W)%W;
    ctx.fillStyle='rgba(255,217,176,'+a.toFixed(3)+')';
    ctx.fillRect(sx,sy,st.r,st.r);
  }
}
function drawSun(){
  const sx=CX+W*0.24-world.camX*FOCAL*0.05, sy=HOR-H*0.115, R=H*0.085;
  const g=ctx.createRadialGradient(sx,sy,R*0.3,sx,sy,R*3.6);
  g.addColorStop(0,'rgba(255,170,90,0.5)'); g.addColorStop(1,'rgba(255,150,70,0)');
  ctx.fillStyle=g; ctx.beginPath(); ctx.arc(sx,sy,R*3.6,0,6.283); ctx.fill();
  const d=ctx.createRadialGradient(sx-R*0.3,sy-R*0.4,R*0.1,sx,sy,R);
  d.addColorStop(0,'#fff6d8'); d.addColorStop(1,'#ff9a4a');
  ctx.fillStyle=d; ctx.beginPath(); ctx.arc(sx,sy,R,0,6.283); ctx.fill();
  ctx.save(); ctx.beginPath(); ctx.arc(sx,sy,R,0,6.283); ctx.clip();
  ctx.fillStyle='rgba(58,17,22,0.55)';
  ctx.fillRect(sx-R,sy+R*0.12,R*2,R*0.09);
  ctx.fillRect(sx-R,sy+R*0.45,R*2,R*0.13);
  ctx.restore();
}
function drawRidge(arr,amp,par,color){
  const n=arr.length, sp=W/n, off=world.camX*par/sp;
  ctx.beginPath(); ctx.moveTo(-sp,H+8);
  for(let px=-sp;px<=W+sp*2;px+=sp){
    const f=px/sp+off, fl=Math.floor(f), i0=((fl%n)+n)%n, i1=(i0+1)%n, u=f-fl;
    const h=lerp(arr[i0],arr[i1],u);
    ctx.lineTo(px,HOR+2-h*amp);
  }
  ctx.lineTo(W+sp*2,H+8); ctx.closePath(); ctx.fillStyle=color; ctx.fill();
}
function drawGround(){
  const g=ctx.createLinearGradient(0,HOR,0,H+4);
  g.addColorStop(0,'rgb(255,140,78)'); g.addColorStop(0.12,'#7a3320'); g.addColorStop(0.45,'#3a1716'); g.addColorStop(1,'#170b0e');
  ctx.fillStyle=g; ctx.fillRect(-12,HOR,W+24,H-HOR+14);
}
function drawGrid(){
  const bst=state==='play'&&ship.boosting;
  ctx.lineWidth=1;
  ctx.strokeStyle='rgba(255,180,120,'+(bst?0.17:0.09)+')';
  ctx.beginPath();
  for(let k=-13;k<=13;k++){ const wx=k*3.6; const a=P(wx,0,5), b=P(wx,0,FAR+25); ctx.moveTo(a.x,a.y); ctx.lineTo(b.x,b.y); }
  ctx.stroke();
  const off=world.dist%8;
  for(let i=1;i<=24;i++){
    const z=i*8-off; if(z<4) continue;
    const y=P(0,0,z).y;
    if(y<HOR-1) break;
    if(y>H+60) continue;
    const a=clamp((110-z)/95,0,1)*0.32+(z<15?0.12:0);
    ctx.strokeStyle='rgba(255,190,140,'+a.toFixed(3)+')';
    ctx.lineWidth=z<12?2:1;
    ctx.beginPath(); ctx.moveTo(-8,y); ctx.lineTo(W+8,y); ctx.stroke();
  }
}
function drawPillar(p){
  if(p.z<2.1) return;
  const zN=Math.max(p.z,2.3), zB=p.z+4.5, f=fog(zN);
  const X0=p.x-p.w/2, X1=p.x+p.w/2;
  const A=P(X0,0,zN), B=P(X1,0,zN), Cc=P(X1,p.h,zN), D=P(X0,p.h,zN);
  const k=0.8+p.tone*0.35;
  if(world.camX<X0){ const Bb=P(X0,0,zB), Db=P(X0,p.h,zB); quad(A,D,Db,Bb,mixc([60*k,28*k,21*k],f)); }
  else if(world.camX>X1){ const Bb=P(X1,0,zB), Db=P(X1,p.h,zB); quad(B,Cc,Db,Bb,mixc([60*k,28*k,21*k],f)); }
  if(CAMH>p.h){ const Tl=P(X0,p.h,zB), Tr=P(X1,p.h,zB); quad(D,Cc,Tr,Tl,mixc([158*k,92*k,58*k],f)); }
  quad(A,B,Cc,D,mixc([104*k,48*k,32*k],f));
  ctx.strokeStyle='rgba(255,180,120,'+((1-f)*0.35).toFixed(3)+')';
  ctx.lineWidth=Math.max(1,A.s*0.045);
  ctx.beginPath(); ctx.moveTo(D.x,D.y); ctx.lineTo(Cc.x,Cc.y); ctx.stroke();
  if(p.vein){
    const vx=X0+p.w*(0.28+0.44*p.tone), v1=P(vx,0.12,zN), v2=P(vx,p.h*0.74,zN);
    ctx.strokeStyle='rgba(255,110,44,'+((1-f)*0.6).toFixed(3)+')';
    ctx.lineWidth=Math.max(1,v2.s*0.09);
    ctx.beginPath(); ctx.moveTo(v1.x,v1.y); ctx.lineTo(v2.x,v2.y); ctx.stroke();
    ctx.fillStyle='rgba(255,180,90,'+((1-f)*0.5).toFixed(3)+')';
    ctx.beginPath(); ctx.arc(v2.x,v2.y,Math.max(1.5,v2.s*0.13),0,6.283); ctx.fill();
  }
}
function drawCore(c){
  if(c.z<2.1) return;
  const f=fog(c.z), s=P(c.x,c.y,c.z);
  const r=Math.max(2,s.s*0.4*(1+0.1*Math.sin(clk*5+c.spin)));
  ctx.save(); ctx.globalCompositeOperation='lighter';
  const g=ctx.createRadialGradient(s.x,s.y,0,s.x,s.y,r*3);
  g.addColorStop(0,'rgba(96,240,224,'+(0.35*(1-f)).toFixed(3)+')'); g.addColorStop(1,'rgba(96,240,224,0)');
  ctx.fillStyle=g; ctx.beginPath(); ctx.arc(s.x,s.y,r*3,0,6.283); ctx.fill();
  ctx.restore();
  ctx.save(); ctx.translate(s.x,s.y); ctx.rotate(clk*1.8+c.spin);
  const rr=r*(1-f*0.5);
  ctx.fillStyle='#d9fff7';
  ctx.beginPath(); ctx.moveTo(0,-rr); ctx.lineTo(rr*0.62,0); ctx.lineTo(0,rr); ctx.lineTo(-rr*0.62,0); ctx.closePath(); ctx.fill();
  ctx.fillStyle='rgba(46,220,200,0.85)';
  ctx.beginPath(); ctx.moveTo(0,-rr*0.55); ctx.lineTo(rr*0.34,0); ctx.lineTo(0,rr*0.55); ctx.lineTo(-rr*0.34,0); ctx.closePath(); ctx.fill();
  ctx.fillStyle='#ffffff'; ctx.beginPath(); ctx.arc(0,0,Math.max(1,r*0.16),0,6.283); ctx.fill();
  ctx.restore();
}
function drawShip(){
  if(!ship.alive) return;
  const bob=Math.sin(clk*3.2)*0.07;
  const pos=P(ship.x,ship.y+bob,PZ), u=pos.s;
  const sh=P(ship.x,0,PZ);
  ctx.fillStyle='rgba(10,3,5,0.3)';
  ctx.beginPath(); ctx.ellipse(sh.x,sh.y,u*0.62,u*0.14,0,0,6.283); ctx.fill();
  ctx.save(); ctx.translate(pos.x,pos.y); ctx.rotate(ship.bank*0.35);
  const L=(0.45+(world.speed/46)*0.75+(ship.boosting?0.85:0))*u*(0.8+0.25*Math.sin(clk*41));
  ctx.globalCompositeOperation='lighter';
  for(const ex of[-1,1]){
    const x=ex*0.16*u, ey=0.30*u;
    const g=ctx.createLinearGradient(0,ey,0,ey+L);
    g.addColorStop(0,'rgba(255,244,214,0.95)'); g.addColorStop(0.4,'rgba(255,150,60,0.55)'); g.addColorStop(1,'rgba(255,120,40,0)');
    ctx.fillStyle=g; ctx.beginPath(); ctx.moveTo(x-0.05*u,ey); ctx.lineTo(x+0.05*u,ey); ctx.lineTo(x,ey+L); ctx.closePath(); ctx.fill();
  }
  ctx.globalCompositeOperation='source-over';
  const g2=ctx.createLinearGradient(0,-0.62*u,0,0.4*u);
  g2.addColorStop(0,'#4a2126'); g2.addColorStop(1,'#1c0d13');
  ctx.fillStyle=g2; ctx.beginPath();
  ctx.moveTo(0,-0.62*u); ctx.lineTo(-0.18*u,-0.08*u); ctx.lineTo(-0.56*u,0.30*u); ctx.lineTo(-0.30*u,0.34*u);
  ctx.lineTo(0,0.18*u); ctx.lineTo(0.30*u,0.34*u); ctx.lineTo(0.56*u,0.30*u); ctx.lineTo(0.18*u,-0.08*u);
  ctx.closePath(); ctx.fill();
  ctx.strokeStyle='rgba(255,158,92,0.55)'; ctx.lineWidth=Math.max(1,u*0.018); ctx.stroke();
  ctx.globalCompositeOperation='lighter';
  ctx.fillStyle='rgba(255,207,143,0.9)';
  ctx.beginPath(); ctx.ellipse(0,-0.30*u,0.055*u,0.10*u,0,0,6.283); ctx.fill();
  for(const ex of[-1,1]){
    const x=ex*0.16*u;
    ctx.globalCompositeOperation='source-over';
    ctx.fillStyle='#3a1418'; ctx.beginPath(); ctx.arc(x,0.28*u,0.10*u,0,6.283); ctx.fill();
    ctx.globalCompositeOperation='lighter';
    ctx.fillStyle=ship.boosting?'#fff3d0':'#ffcf8f';
    ctx.beginPath(); ctx.arc(x,0.28*u,0.055*u,0,6.283); ctx.fill();
  }
  ctx.restore();
}
function drawParts(){
  ctx.save(); ctx.globalCompositeOperation='lighter';
  for(const q of parts){
    if(q.z<2.2) continue;
    const s=P(q.x,q.y,q.z), a=clamp(q.life/q.max,0,1);
    const r=Math.max(1,q.size*s.s*0.6);
    ctx.fillStyle='rgba('+q.col[0]+','+q.col[1]+','+q.col[2]+','+(a*0.9).toFixed(3)+')';
    ctx.beginPath(); ctx.arc(s.x,s.y,r,0,6.283); ctx.fill();
  }
  ctx.restore();
}
function drawStreaks(){
  if(!(state==='play'&&(ship.boosting||world.speed>38))) return;
  ctx.save(); ctx.globalCompositeOperation='lighter';
  const R=Math.max(W,H)*0.75, n=14;
  for(let i=0;i<n;i++){
    const a=i/n*6.283+(i%3)*0.35;
    const prog=((clk*(1.5+world.speed/30)+i*0.61)%1);
    const r0=(0.1+prog*0.9)*R, r1=r0+(1-prog)*R*0.22+30;
    ctx.strokeStyle='rgba(255,217,168,'+((1-prog)*(ship.boosting?0.11:0.05)).toFixed(3)+')';
    ctx.lineWidth=1.5;
    ctx.beginPath();
    ctx.moveTo(CX+Math.cos(a)*r0,HOR+Math.sin(a)*r0);
    ctx.lineTo(CX+Math.cos(a)*r1,HOR+Math.sin(a)*r1);
    ctx.stroke();
  }
  ctx.restore();
}
function drawPops(){
  for(const p of pops){
    const a=clamp(p.life/p.max*1.4,0,1);
    const y=p.y-(p.max-p.life)*36;
    ctx.save(); ctx.globalAlpha=a;
    ctx.font='italic '+p.size+'px '+FD; ctx.textAlign='center';
    ctx.lineWidth=3; ctx.strokeStyle='rgba(20,6,10,0.7)';
    ctx.strokeText(p.txt,p.x,y);
    ctx.fillStyle=p.col; ctx.fillText(p.txt,p.x,y);
    ctx.restore();
  }
}
function render(){
  ctx.fillStyle='#150a0f'; ctx.fillRect(0,0,W,H);
  const sw=(state==='play')?Math.sin(clk*2.1)*Math.min(2,world.speed*0.04):0;
  const shx=(Math.random()*2-1)*shake, shy=(Math.random()*2-1)*shake+sw;
  ctx.save(); ctx.translate(shx,shy);
  drawSky(); drawStars(); drawSun();
  drawRidge(ridgeFar,H*0.10,5,'#2a0f16');
  drawRidge(ridgeNear,H*0.16,14,'#1d0a10');
  drawGround(); drawGrid();
  const list=[];
  for(const p of pillars) if(p.z>2.1&&p.z<FAR+10) list.push({z:p.z,f:drawPillar,o:p});
  for(const c of cores) if(!c.taken&&c.z>2.1&&c.z<FAR+10) list.push({z:c.z,f:drawCore,o:c});
  list.sort((a,b)=>b.z-a.z);
  for(const it of list) it.f(it.o);
  if(ship.alive&&(state==='play'||state==='menu')) drawShip();
  drawParts(); drawStreaks(); drawPops();
  ctx.restore();
  if(flash>0){ ctx.fillStyle='rgba(255,220,180,'+(flash*0.55).toFixed(3)+')'; ctx.fillRect(0,0,W,H); }
  ctx.drawImage(VIG,0,0);
}

  bestChip.textContent='BEST '+best+' m';

}

/* ============================== LOOP ============================== */
function frame(now){
  requestAnimationFrame(frame);
  let rdt=(now-last)/1000; last=now;
  if(!(rdt>0)) rdt=0.016; rdt=Math.min(rdt,0.05);
  clk+=rdt;
  if(!ship.alive&&state==='play') timeScale+=(0.22-timeScale)*Math.min(1,rdt*8);
  else timeScale+=(1-timeScale)*Math.min(1,rdt*6);
  step(rdt*timeScale,rdt);
  render(); hudTick(); audioTick();
}

const keys={L:false,R:false,B:false};
function canon(k){
  if(k==='arrowleft'||k==='a') return 'L';
  if(k==='arrowright'||k==='d') return 'R';
  if(k==='arrowup'||k==='w'||k==='shift') return 'B';
  return null;
}
window.addEventListener('keydown',e=>{
  const k=e.key.toLowerCase();
  if(k==='arrowleft'||k==='arrowright'||k==='arrowup'||k==='arrowdown'||k===' ') e.preventDefault();
  initAudio(); if(AC&&AC.state==='suspended') AC.resume();
  const c=canon(k);
  if(c){ keys[c]=true; ship.touch=null; }
  if(k==='m'){ toggleMute(); return; }
  if(state==='menu'&&(k===' '||k==='enter')) startGame();
  else if(state==='play'){
    if(k==='p'||k==='escape') pause();
    else if(k==='r') startGame();
  }else if(state==='pause'){
    if(k==='p'||k==='escape'||k===' '||k==='enter') resume();
    else if(k==='r') startGame();
  }else if(state==='over'&&performance.now()-overAt>600&&(k===' '||k==='enter'||k==='r')) startGame();
});
window.addEventListener('keyup',e=>{ const c=canon(e.key.toLowerCase()); if(c) keys[c]=false; });
window.addEventListener('blur',()=>{ keys.L=keys.R=keys.B=false; tBoost=false; if(state==='play') pause(); });
document.addEventListener('visibilitychange',()=>{ if(document.hidden&&state==='play') pause(); });
