function collide(e){
  const r=e.r;
  // resolve against each overlapping solid tile
  const x0=Math.floor((e.x-r)/T), x1=Math.floor((e.x+r)/T), y0=Math.floor((e.y-r)/T), y1=Math.floor((e.y+r)/T);
  for(let ty=y0;ty<=y1;ty++)for(let tx=x0;tx<=x1;tx++){
    if(!solidAt(tx,ty)) continue;
    const bx=tx*T, by=ty*T;
    // closest point
    let cx=Math.max(bx,Math.min(e.x,bx+T)), cy=Math.max(by,Math.min(e.y,by+T));
    let dx=e.x-cx, dy=e.y-cy;
    const d2=dx*dx+dy*dy;
    if(d2<r*r){
      if(d2>1e-6){const d=Math.sqrt(d2); e.x=cx+dx/d*r; e.y=cy+dy/d*r;}
      else { // center inside tile: push out along min axis
        const l=e.x-bx, rr=bx+T-e.x, t=e.y-by, b=by+T-e.y;
        const m=Math.min(l,rr,t,b);
        if(m===l) e.x=bx-r; else if(m===rr) e.x=bx+T+r; else if(m===t) e.y=by-r; else e.y=by+T+r;
      }
    }
  }
}

function beep(o){
  if(!audioOn||!AC) return;

    const t0=AC.currentTime+(o.at||0), osc=AC.createOscillator(), g=AC.createGain();
    osc.type=o.type||'sine'; osc.frequency.setValueAtTime(o.f,t0);
    if(o.f2) osc.frequency.exponentialRampToValueAtTime(Math.max(1,o.f2),t0+o.dur);
    g.gain.setValueAtTime(0.0001,t0);
    g.gain.exponentialRampToValueAtTime(o.vol||0.2,t0+(o.a||0.008));
    g.gain.exponentialRampToValueAtTime(0.0001,t0+o.dur);
    osc.connect(g); g.connect(master); osc.start(t0); osc.stop(t0+o.dur+0.06);
  }catch(e){}
}

function raySeg(x0,y0,ux,uy,x1,y1,x2,y2){
  const sx=x2-x1, sy=y2-y1, den=ux*sy-uy*sx;
  if(Math.abs(den)<1e-9) return null;
  const t=((x1-x0)*sy-(y1-y0)*sx)/den, u=((x1-x0)*uy-(y1-y0)*ux)/den;
  return (t>=0&&u>=0&&u<=1)?t:null;
}
function castRay(x0,y0,x1,y1){
  const dx=x1-x0, dy=y1-y0, L=Math.hypot(dx,dy)||1, ux=dx/L, uy=dy/L;
  let best=L;
  for(const s of segs){ const t=raySeg(x0,y0,ux,uy,s[0],s[1],s[2],s[3]); if(t!==null&&t<best) best=t; }
  return best;
}

    S.parts.push({x,y,vx:Math.cos(a)*s,vy:Math.sin(a)*s-60,c:col,t:0,life:1.2+Math.random()*.6}); }
}
function bust(p){
  if(state!=='play') return;
  state='over'; S.result={win:false,caught:p.id};
  S.flash=1; S.shake=12; SFX.lose();
  $('#alertEl').classList.remove('on');
  setTimeout(showOver,900);
}
function openVault(){
  S.vaultOpen=true; S.doorProg=0; buildSegs(); renderMap();
  emitNoise(VAULT.cx,VAULT.cy,340); SFX.unlock(); S.shake=6;
  floater(VAULT.cx-10,VAULT.y-8,'VAULT OPEN','#ff8c5a');
}
function openExit(){
  S.exitOpen=true; S.exitProg=0; emitNoise(PAD.cx,PAD.cy,300); SFX.done(); S.shake=5;
  floater(PAD.cx,PAD.y-10,'EXTRACTION LIVE','#49ffa5');
}

function resolveRect(e,bx,by,w,h){
  const cx=Math.max(bx,Math.min(e.x,bx+w)), cy=Math.max(by,Math.min(e.y,by+h));
  let dx=e.x-cx, dy=e.y-cy; const d2=dx*dx+dy*dy, r=e.r;
  if(d2>=r*r) return;
  if(d2>1e-9){ const d=Math.sqrt(d2); e.x=cx+dx/d*r; e.y=cy+dy/d*r; }
  else{
    const l=e.x-bx, rt=bx+w-e.x, tp=e.y-by, bt=by+h-e.y, m=Math.min(l,rt,tp,bt);
    if(m===l)e.x=bx-r; else if(m===rt)e.x=bx+w+r; else if(m===tp)e.y=by-r; else e.y=by+h+r;
  }
}
function collide(e){
  const r=e.r;
  const x0=Math.floor((e.x-r)/T), x1=Math.floor((e.x+r)/T), y0=Math.floor((e.y-r)/T), y1=Math.floor((e.y+r)/T);
  for(let ty=y0;ty<=y1;ty++)for(let tx=x0;tx<=x1;tx++){ if(solidAt(tx,ty)) resolveRect(e,tx*T,ty*T,T,T); }
  for(const c of crates) resolveRect(e,c.x-17,c.y-17,34,34);
}
function turnTo(g,ang,rate,dt){
  let d=ang-g.face; while(d>Math.PI)d-=2*Math.PI; while(d<-Math.PI)d+=2*Math.PI;
  g.face+=Math.max(-rate*dt,Math.min(rate*dt,d));
}

function updPlayer(p,dt){
  if(p.escaped) return;
  const ix=(keys[p.k.r]?1:0)-(keys[p.k.l]?1:0), iy=(keys[p.k.dn]?1:0)-(keys[p.k.up]?1:0);
  p.crouch=!!keys[p.k.cr];
  let sp=p.crouch?92:172; if(p.hasCore) sp*=0.8;
  if(ix&&iy){ ix*=0.7071; iy*=0.7071; }
  p.x+=ix*sp*dt; collide(p); p.y+=iy*sp*dt; collide(p);
  if(ix||iy) p.face=Math.atan2(iy,ix);
  if((ix||iy)&&!p.crouch){
    p.stepT-=dt;
    if(p.stepT<=0){ p.stepT=0.34; emitNoise(p.x,p.y,100); SFX.step(); }
  } else p.stepT=0;
  // pickups
  if(!S.cardTaken&&Math.hypot(p.x-CARD_POS.x,p.y-CARD_POS.y)<18){
    S.cardTaken=true; p.hasCard=true; SFX.card(); floater(CARD_POS.x,CARD_POS.y-14,'KEYCARD','#49ffa5');
  }
  if(!S.coreTaken&&S.vaultOpen&&Math.hypot(p.x-CORE_POS.x,p.y-CORE_POS.y)<20){
    S.coreTaken=true; p.hasCore=true; SFX.core(); floater(CORE_POS.x,CORE_POS.y-14,'CORE ACQUIRED','#3fe8ff');
    emitNoise(CORE_POS.x,CORE_POS.y,150);
  }
}

function updGuard(g,dt,detect){
  // --- perception
  let seen=null, seenD=1e9;
  if(detect&&g.state!=='chase'){
    for(const p of players){
      if(p.escaped) continue;
      const dx=p.x-g.x, dy=p.y-g.y, d=Math.hypot(dx,dy);
      if(d>g.range) continue;
      if(p.crouch&&d>g.range*0.55) continue;
      let a=Math.atan2(dy,dx)-g.face; a=Math.atan(a,Math.PI*2);
      if(Math.abs(a)>g.fov/2) continue;
      if(castRay(g.x,g.y,p.x,p.y)<d-4) continue;
      if(d<seenD){ seen=p; seenD=d; }
    }
    if(seen){
      if(seenD<27){ bust(seen); return; }
      const rate=(0.5+(1-seenD/g.range)*2.6)*(seen.crouch?0.55:1);
      g.prevSus=g.sus; g.sus=Math.min(1,g.sus+rate*dt);
      if(g.prevSus<0.5&&g.sus>=0.5) SFX.ping(g.sus);
      if(g.sus>=1){
        g.state='chase'; g.target=seen; g.lastSeen={x:seen.x,y:seen.y}; g.scanT=0;
        S.alarms++; floater(g.x,g.y-26,'SPOTTED!','#ff3b5c'); beep({f:880,type:'square',dur:.12,vol:.14});
      }
    } else {
      g.sus=Math.max(0,g.sus-dt*0.55);
    }
  }
  // --- state machine
  if(g.state==='patrol'){
    const wp=g.path[g.wp];
    if(g.pauseT>0){
      g.pauseT-=dt; g.face+=Math.sin(S.time*4+g.x)*dt*1.6;
    }else{
      turnTo(g,Math.atan2(wp[1]-g.y,wp[0]-g.x),5,dt);
      moveGuard(g,wp,62,dt);
      if(Math.hypot(wp[0]-g.x,wp[1]-g.y)<7){ g.wp=(g.wp+1)%g.path.length; g.pauseT=0.4; }
    }
  }
  else if(g.state==='investigate'){
    const t=g.inv;
    turnTo(g,Math.atan2(t.y-g.y,t.x-g.x),6,dt);
    moveGuard(g,t,100,dt);
    if(Math.hypot(t.x-g.x,t.y-g.y)<9){
      g.scanT+=dt; g.face+=Math.sin(S.time*5+g.y)*dt*2.2;
      if(g.scanT>1.8){
        g.state='patrol'; g.sus=0;
        let bi=0,bd=1e9;
        for(let i=0;i<g.path.length;i++){ const d=Math.hypot(g.path[i][0]-g.x,g.path[i][1]-g.y); if(d<bd){bd=d;bi=i;} }
        g.wp=bi;
      }
    } else g.scanT=0;
  }
  else if(g.state==='chase'){
    const t=g.target;
    if(!t||t.escaped){ g.state='patrol'; g.sus=0; g.target=null; let bi=0,bd=1e9;
      for(let i=0;i<g.path.length;i++){const d=Math.hypot(g.path[i][0]-g.x,g.path[i][1]-g.y);if(d<bd){bd=d;bi=i;}} g.wp=bi; }
    else{
      const dx=t.x-g.x, dy=t.y-g.y, d=Math.hypot(dx,dy);
      const vis = d<g.range*1.2 && castRay(g.x,g.y,t.x,t.y)>=d-4;
      if(vis){ g.lastSeen={x:t.x,y:t.y}; }
      const goal=vis?t:g.lastSeen;
      turnTo(g,Math.atan2(goal.y-g.y,goal.x-g.x),9,dt);
      moveGuard(g,goal,150,dt);
      if(!vis){
        if(Math.hypot(goal.x-g.x,goal.y-g.y)<8){ g.scanT+=dt; g.face+=Math.sin(S.time*6)*dt*2.6; if(g.scanT>2.6){ g.state='patrol'; g.sus=0; g.target=null;
          let bi=0,bd=1e9; for(let i=0;i<g.path.length;i++){const dd=Math.hypot(g.path[i][0]-g.x,g.path[i][1]-g.y);if(dd<bd){bd=dd;bi=i;}} g.wp=bi; } }
        else g.scanT=0;
      }
      for(const p of players){ if(!p.escaped&&Math.hypot(p.x-g.x,p.y-g.y)<g.r+p.r+3){ bust(p); return; } }
    }
  }
}
function moveGuard(g,goal,sp,dt){
  const dx=goal.x-g.x, dy=goal.y-g.y, d=Math.hypot(dx,dy)||1;
  g.x+=dx/d*sp*dt; collide(g); g.y+=dy/d*sp*dt; collide(g);
}

function updInteract(dt){
  // vault door
  if(!S.vaultOpen){
    const cardP=players.find(p=>p.hasCard&&!p.escaped);
    let active=false;
    if(cardP&&Math.hypot(cardP.x-VAULT.cx,cardP.y-VAULT.cy)<58){
      active=true;
      if(keys[cardP.k.act]){
        S.doorProg+=dt/0.9; S.tickT-=dt;
        if(S.tickT<=0){S.tickT=0.16;SFX.tick();}
        if(S.doorProg>=1) openVault();
      } else S.doorProg=Math.max(0,S.doorProg-dt*2);
    } else S.doorProg=0;
  }
  // exit breach
  if(!S.exitOpen){
    const near=players.find(p=>!p.escaped&&Math.hypot(p.x-PAD.cx,p.y-PAD.cy)<64);
    if(near&&S.coreTaken){
      if(keys[near.k.act]){
        S.exitProg+=dt/1.4; S.tickT-=dt;
        if(S.tickT<=0){S.tickT=0.16;SFX.tick();}
        if(S.exitProg>=1) openExit();
      } else S.exitProg=Math.max(0,S.exitProg-dt*2);
    } else S.exitProg=0;
  }
  // escape
  if(S.exitOpen){
    for(const p of players){
      if(p.escaped) continue;
      if(p.x>PAD.x-4&&p.x<PAD.x+PAD.w+4&&p.y>PAD.y-4&&p.y<PAD.y+PAD.h+4){
        p.escaped=true; SFX.safe(); burst(p.x,p.y,14,p.col); floater(p.x,p.y-20,'SAFE','#49ffa5');
      }
    }
    if(players.every(p=>p.escaped)&&S.winT===0) S.winT=0.9;
  }
}

function update(dt){
  // ambient decay
  S.shake=Math.max(0,S.shake-dt*26); S.flash=Math.max(0,S.flash-dt*2.2);
  for(const r of S.rings) r.t+=dt; S.rings=S.rings.filter(r=>r.t<r.life);
  for(const f of S.floats) f.t+=dt; S.floats=S.floats.filter(f=>f.t<1.3);
  for(const p of S.parts){ p.t+=dt; p.x+=p.vx*dt; p.y+=p.vy*dt; p.vy+=280*dt; } S.parts=S.parts.filter(p=>p.t<p.life);

  if(state==='menu'||state==='over'){ for(const g of guards) updGuard(g,dt,false); return; }
  if(state==='paused') return;

  S.time+=dt;
  $('#timerEl').textContent=fmt(S.time);

  for(const p of players) updPlayer(p,dt);
  // player-player separation
  const [a,b]=players;
  if(!a.escaped&&!b.escaped){
    const dx=b.x-a.x,dy=b.y-a.y,d=Math.hypot(dx,dy),m=a.r+b.r;
    if(d<m&&d>1e-6){ a.x-=dx/d*(m-d)/2; a.y-=dy/d*(m-d)/2; b.x+=dx/d*(m-d)/2; b.y+=dy/d*(m-d)/2; collide(a); collide(b); }
  }
  for(const g of guards) updGuard(g,dt,true);
  // guard-player physical push (non-chase)
  for(const g of guards){ if(g.state==='chase') continue;
    for(const p of players){ if(p.escaped) continue;
      const dx=p.x-g.x,dy=p.y-g.y,d=Math.hypot(dx,dy),m=g.r+p.r;
      if(d<m&&d>1e-6){ p.x+=dx/d*(m-d); p.y+=dy/d*(m-d); collide(p); }
    }}

  updInteract(dt);

  const p=phase(); if(p!==lastPhase){ lastPhase=p; setPhaseUI(p); }

  // alarm audio + UI
  const alarm=guards.some(g=>g.state==='chase');
  $('#alertEl').classList.toggle('on',alarm);
  if(alarm){
    S.sirenT-=dt;
    if(S.sirenT<=0){ S.sirenT=0.55; S.sirenFlip=!S.sirenFlip; SFX.siren(S.sirenFlip); }
    S.heartT-=dt;
    if(S.heartT<=0){ S.heartT=1.0; SFX.heart(); }
  }
  // win finalize
  if(S.winT>0){
    S.winT-=dt;
    if(S.winT<=0){ state='over'; S.result={win:true}; SFX.win(); burst(PAD.cx,PAD.cy,46,'#3fe8ff'); burst(PAD.cx,PAD.cy,46,'#ffd23f'); setTimeout(showOver,750); }
  }
}

  }

  for(const g of guards){
    const pts=[]; const n=26;

    ctx.closePath(); ctx.fill();
  }

  // guards
  for(const g of guards){
    ctx.fillStyle='rgba(0,0,0,.3)'; ctx.beginPath(); ctx.ellipse(g.x,g.y+3,g.r*.9,g.r*.5,0,0,7); ctx.fill();

  // floaters & particles
  for(const f of S.floats){
    const k=f.t/1.3;
    ctx.globalAlpha=1-k; ctx.fillStyle=f.c; ctx.font='bold 11px monospace'; ctx.textAlign='center';
    ctx.fillText(f.txt,f.x,f.y-16*k); ctx.globalAlpha=1;
  }
  for(const p of S.parts){
    ctx.globalAlpha=Math.max(0,1-p.t/p.life); ctx.fillStyle=p.c;
    ctx.fillRect(p.x-2,p.y-2,4,4); ctx.globalAlpha=1;
  }
  ctx.restore();

  // frame + corner ticks
  ctx.strokeStyle='#26345a'; ctx.lineWidth=1.5;
  ctx.strokeRect(ox-1,oy-1,WW*sc+2,WH*sc+2);
  ctx.strokeStyle='#3fe8ff'; ctx.lineWidth=2; const L=14;
  [[ox-1,oy-1,1,1],[ox+WW*sc+1,oy-1,-1,1],[ox-1,oy+WH*sc+1,1,-1],[ox+WW*sc+1,oy+WH*sc+1,-1,-1]].forEach(c=>{
    ctx.beginPath(); ctx.moveTo(c[0]+c[2]*L,c[1]); ctx.lineTo(c[0],c[1]); ctx.lineTo(c[0],c[1]+c[3]*L); ctx.stroke();
  });

  // overlays
  if(vigCv) ctx.drawImage(vigCv,0,0,w,h);
  const alarm=guards.some(g=>g.state==='chase');
  if(alarm&&redCv){ ctx.globalAlpha=.2+.14*Math.sin(S.time*6); ctx.drawImage(redCv,0,0,w,h); ctx.globalAlpha=1; }
  if(S.flash>0){ ctx.fillStyle=`rgba(255,30,60,${S.flash*.4})`; ctx.fillRect(0,0,w,h); }
}

function drawProg(x,y,prog,p){
  ctx.strokeStyle='rgba(255,210,63,.9)'; ctx.lineWidth=3;
  ctx.beginPath(); ctx.arc(x,y,11,-Math.PI/2,-Math.PI/2+prog*Math.PI*2); ctx.stroke();
  ctx.fillStyle='#ffd23f'; ctx.font='bold 9px monospace'; ctx.textAlign='center';
  ctx.fillText('HOLD ['+p.keyLabel+']',x,y-18);
}

/* ============================== LOOP ============================== */
let last=performance.now();
function frame(now){
  requestAnimationFrame(frame);
  const dt=Math.min(0.05,(now-last)/1000); last=now;
  update(dt); draw();
}
resize(); reset(); requestAnimationFrame(frame);
