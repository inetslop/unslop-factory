function pickWord(short){
  const frac=Math.min(1,dist/TRACK);
  let minL=short?3:3+Math.round(frac*3),maxL=minL+2;minL=clamp(minL,3,7);
  const c=[];for(const w of WORDS)if(w.length>=minL&&w.length<=maxL)c.push(w);
  let pick=c[(Math.random()*c.length)|0];let guard=0;
  while(pick===cur&&guard++<8)pick=c[(Math.random()*c.length)|0];
  return pick;
}
function buildWord(){

  caretEl=document.createElement('span');caretEl.className='caret';wordEl.appendChild(caretEl);

}
function syncWord(){
  for(let i=0;i<spans.length;i++)spans[i].classList.toggle('done',i<progress());
  const a=progress()<spans.length?spans[progress()]:null;
  if(a)wordEl.insertBefore(caretEl,a);else wordEl.appendChild(caretEl);
}
function progress(){return cur.length?Math.min(spans.length,spans.findIndex(s=>!s.classList.contains('done'))):0;}

/* ============ update ============ */
function carX(){return roadX+laneW*(PLAYER_LANE+.5)+Math.sin(tG*6.5)*2.2*(v/70);}
function update(dt){
  tG+=dt;shake=Math.max(0,shake-dt*3.2);boostT=Math.max(0,boostT-dt);brakeT=Math.max(0,brakeT-dt);
  // particles
  for(let i=particles.length-1;i>=0;i--){const p=particles[i];p.age+=dt;
    if(p.t==='conf'){p.y+=p.vy*dt;p.x+=Math.sin(tG*3+p.ph)*40*dt;p.rot+=p.vr*dt;if(p.y>H+20)particles.splice(i,1);continue;}
    p.x+=p.vx*dt;p.y+=p.vy*dt;if(p.age>=p.life)particles.splice(i,1);}
  for(let i=popups.length-1;i>=0;i--){const p=popups[i];p.t+=dt;p.y-=36*dt;if(p.t>=p.life)popups.splice(i,1);}

  if(ST==='menu'){ // idle grid life
    smokeAcc+=dt;const all=[{x:carX(),y:playerY}].concat(rivals.filter(r=>Math.abs(r.dist)<120).map(r=>({x:roadX+laneW*(r.lane+.5),y:distToY(r.dist)})));
    if(smokeAcc>.28){smokeAcc=0;const c=all[(Math.random()*all.length)|0];puff(c.x,c.y+carH*.5);}
    A.engine(0);return;
  }
  if(ST==='countdown'){
    const nl=Math.min(4,Math.floor(cdT/.65));
    if(nl>cdLight&&nl>=0){cdLight=nl;for(let i=0;i<=nl;i++)cdLights[i].classList.add('on');A.beep(520,.12);}
    cdT+=dt;if(cdT>=4*.65+.8)go();
    return;
  }
  if(ST==='pause'){A.engine(0);return;}

  // player physics
  const prevD=dist;
  let dec=v*DECAY*dt+(v>62?(v-62)*1.1*dt:0);
  if(ST==='finish')dec+=v*2.6*dt;           // braking after the line
  v=Math.max(0,v-dec);dist+=v*dt;stats.top=Math.max(stats.top,v);
  A.engine(v);

  // exhaust
  smokeAcc+=dt;const rate=v>4?(.16-v*.0009):0;
  if(rate>0&&smokeAcc>rate){smokeAcc=0;puff(carX()-(Math.random()<.5?1:-1)*carW*.3,playerY+carH*.5);}

  // rivals
  for(const r of rivals){
    r.t+=dt;
    if(r.t>=r.delay){
      r.nextBurst-=dt;
      if(r.nextBurst<=0){r.burstT=1.1+Math.random()*.8;r.nextBurst=4+Math.random()*5;}
      r.burstT=Math.max(0,r.burstT-dt);
      const target=r.base+Math.sin(r.t*r.wf+r.wp)*r.wa+(r.burstT>0?r.burstAmp:0);
      const acc=target>r.v?26:34;
      r.v=clamp(r.v+clamp(target-r.v,-acc*dt,acc*dt),0,85);
    } else r.v=0;
    r.dist+=r.v*dt;
    // overtake detect (player crossing rival upward)
    if(!finished&&prevD<=r.dist&&dist>r.dist){
      A.overtake();
      const pl=1+rivals.filter(o=>o.dist>dist).length;
      popup(roadX+laneW*(r.lane+.5),distToY(r.dist)-carH,'OVERTAKE! P'+pl,'#fff',26);
    }
  }
  const place=1+rivals.filter(r=>r.dist>dist).length;
  if(place<prevPlace){placeEl.classList.remove('pop');void placeEl.offsetWidth;placeEl.classList.add('pop');}
  prevPlace=place;

  if(!finished&&dist>=TRACK)finishRace();
  if(finished&&ST==='race'){ST='finish';finishDelay=.95;}
  if(ST==='finish'&&finishDelay>0){finishDelay-=dt;if(finishDelay<=0)showFinish();}

  updateHUD(place);
}

  $('medal').textContent='P'+finalPlace;
  $('medal').style.color=finalPlace===1?'#ffd75e':finalPlace===2?'#cfd6e4':finalPlace===3?'#e0a06b':'#8fa0b5';
  $('finishTitle').textContent=finalPlace===1?'VICTORY!':(finalPlace<=3?'PODIUM FINISH':'RACE COMPLETE');
  $('newBest').classList.toggle('hidden',!newBestFlag);
  $('fTime').textContent=fmt(raceT);$('fWpm').textContent=wpm;$('fAcc').textContent=acc+'%';
  $('fTop').textContent=Math.round(stats.top*3.6)+' km/h';$('fWords').textContent=stats.words;
  $('fCombo').textContent='×'+bestCombo;
  $('finishScr').classList.remove('hidden');updateBestChips();
}

  // road

  ctx.fillStyle='#e8ecf2';ctx.fillRect(roadX+2,-20,3,H+40);ctx.fillRect(roadX+roadW-5,-20,3,H+40);
  for(let l=1;l<4;l++){const x=roadX+laneW*l;let dA2=Math.floor((dist-(H-playerY)/pxPerM)/16)*16;
    for(let d=dA2;d<dist+(playerY+80)/pxPerM;d+=16){if(Math.floor(d/16)&1)continue;
      const y1=distToY(d),y2=distToY(d+8);ctx.fillStyle='rgba(215,224,238,.75)';ctx.fillRect(x-2,y2,4,y1-y2);}}
  // curbs near start / finish
  const curb=(d0,d1)=>{for(let d=d0;d<d1;d+=3){const y1=distToY(d),y2=distToY(d+3);
    ctx.fillStyle=(Math.floor(d/3)&1)?'#e5484d':'#f2f5f8';
    ctx.fillRect(roadX-8,y2,8,y1-y2);ctx.fillRect(roadX+roadW,y2,8,y1-y2);}};
  curb(-12,16);curb(TRACK-16,TRACK+14);
  // start & finish checker bands
  const bandDraw=(d0,rows)=>{const cs=Math.max(9,laneW*.22),cols=Math.ceil(roadW/cs);
    for(let r=0;r<rows;r++)for(let c=0;c<cols;c++){const y=distToY(d0+r*cs/pxPerM*0+ (d0 + (r)* (cs/pxPerM) ) );
      const yy=distToY(d0+(d0===0?0:0)+ (r)*(cs/pxPerM) ),xx=roadX+c*cs;
      ctx.fillStyle=((c+r)&1)?'#f2f5f8':'#10151d';ctx.fillRect(xx,yy+cs/pxPerM*0,cs,cs/pxPerM+.6);}};
  const band2=(dStart,rows)=>{const cs=Math.max(9,laneW*.22),cols=Math.ceil(roadW/cs),rowH=cs/pxPerM;
    for(let r=0;r<rows;r++)for(let c=0;c<cols;c++){ctx.fillStyle=((c+r)&1)?'#f2f5f8':'#10151d';
      ctx.fillRect(roadX+c*cs,distToY(dStart+(r+1)*rowH),cs,rowH+.6);}};
  band2(0,2);band2(TRACK,3);
  // finish gantry
  const gy=distToY(TRACK);
  if(gy>-40&&gy<H+40){ctx.fillStyle='#e8ecf2';ctx.fillRect(roadX-10,gy-16,9,26);ctx.fillRect(roadX+roadW+1,gy-16,9,26);
    ctx.fillStyle='#111a24';ctx.fillRect(roadX-10,gy-30,roadW+20,18);
    ctx.fillStyle='#ffb020';ctx.font='italic 700 15px Impact,"Arial Narrow",sans-serif';ctx.textAlign='center';ctx.textBaseline='middle';
    ctx.fillText('F I N I S H',roadX+roadW/2,gy-21);}

  drawScenery(dist+(playerY+60)/pxPerM,dist-(H+60-playerY)/pxPerM);

  // cloud shadows (drifting)
  for(const c of clouds){
    const cx=((c.x+tG*c.s+dist*pxPerM*.05)%(W+400))-200,cy=c.y%H;
    ctx.fillStyle='rgba(8,18,26,.05)';ctx.beginPath();ctx.ellipse(cx,cy,c.r*1.5,c.r*.8,0,0,6.29);ctx.fill();
    ctx.fillStyle='rgba(8,18,26,.05)';ctx.beginPath();ctx.ellipse(cx+c.r*.7,cy+14,c.r,c.r*.6,0,0,6.29);ctx.fill();
  }
  // speed streaks
  if(v>50){const n=Math.min(14,(v-50)|0);ctx.strokeStyle='rgba(255,255,255,'+Math.min(.2,(v-50)/160)+')';ctx.lineWidth=2;
    for(let k=0;k<n;k++){const x=k%2?roadX+laneW*(.2+.6*Math.abs(Math.sin(k*3.7))): (k<4?roadX-14-k*16:W-roadX+14+(k-4)*16);
      const y=((tG*(280+v*6)+k*173)%(H+120))-60,len=40+((v-50)/35)*70;
      ctx.beginPath();ctx.moveTo(x,y);ctx.lineTo(x,y+len);ctx.stroke();}}

  // separation (visual only)
  const cars=rivals.map(r=>({d:r.dist,x:roadX+laneW*(r.lane+.5),id:r.id,r}));cars.push({d:dist,x:carX(),id:0});
  for(const r of rivals){let t=0;const my=cars.find(c=>c.r===r);
    for(const o of cars){if(o.id===my.id)continue;
      if(Math.abs(o.d-my.d)<26&&Math.abs(o.x-my.x)<laneW*.6){const dir=o.id<my.id?-1:1;t+=dir*laneW*.62*(o.id===0?1:.5);}}
    r.xAdj+=(clamp(t,-laneW,laneW)-r.xAdj)*Math.min(1,dtCache*8);}

  // rivals (far first)
  const sorted=rivals.slice().sort((a,b)=>b.dist-a.dist);
  for(const r of sorted){const y=distToY(r.dist);if(y<-70||y>H+70)continue;
    const x=clamp(roadX+laneW*(r.lane+.5)+r.xAdj+Math.sin(tG*1.3+r.wp)*4,roadX+carW*.7,roadX+roadW-carW*.7);
    drawCar(x,y,r.color,false);
    if(Math.abs(r.dist-dist)<130){ctx.font='700 10px "Segoe UI",system-ui,sans-serif';ctx.textAlign='center';ctx.textBaseline='bottom';
      ctx.lineWidth=3;ctx.strokeStyle='rgba(8,12,18,.85)';ctx.strokeText(r.name,x,y-carH*.75);
      ctx.fillStyle=r.color;ctx.fillText(r.name,x,y-carH*.75);}}
  // player
  drawCar(carX(),playerY,'#e8eef5',true);
  if(boostT>0){const f=Math.random();ctx.fillStyle='rgba(58,214,232,.7)';
    ctx.beginPath();ctx.moveTo(carX()-carW*.26,playerY+carH*.5);ctx.lineTo(carX()+carW*.26,playerY+carH*.5);
    ctx.lineTo(carX(),playerY+carH*(1.1+f*.3));ctx.closePath();ctx.fill();
    ctx.fillStyle='#ffd166';ctx.beginPath();ctx.moveTo(carX()-carW*.13,playerY+carH*.5);ctx.lineTo(carX()+carW*.13,playerY+carH*.5);
    ctx.lineTo(carX(),playerY+carH*(.95+f*.2));ctx.closePath();ctx.fill();}

  // particles
  for(const p of particles){const a=1-p.age/p.life;
    if(p.t==='smoke'){ctx.fillStyle='rgba(205,214,224,'+(a*.35)+')';ctx.beginPath();ctx.arc(p.x,p.y,p.s+(p.age*14),0,6.29);ctx.fill();}
    else if(p.t==='spark'){ctx.strokeStyle=p.age<.15?'#fff':'#ffd76a';ctx.lineWidth=2;ctx.globalAlpha=a;
      ctx.beginPath();ctx.moveTo(p.x,p.y);ctx.lineTo(p.x-p.vx*.03,p.y-p.vy*.03);ctx.stroke();ctx.globalAlpha=1;}
    else if(p.t==='conf'){ctx.save();ctx.translate(p.x,p.y);ctx.rotate(p.rot);ctx.fillStyle=p.c;ctx.fillRect(-p.s/2,-p.s/2,p.s,p.s*.66);ctx.restore();}}

  // popups
  for(const p of popups){const a=1-p.t/p.life;ctx.globalAlpha=a;
    ctx.font='italic 700 '+p.s+'px Impact,"Arial Narrow",sans-serif';ctx.textAlign='center';ctx.textBaseline='middle';
    ctx.lineWidth=5;ctx.strokeStyle='rgba(8,12,18,.9)';ctx.strokeText(p.txt,p.x,p.y);ctx.fillStyle=p.c;ctx.fillText(p.txt,p.x,p.y);ctx.globalAlpha=1;}
  ctx.restore();
}
