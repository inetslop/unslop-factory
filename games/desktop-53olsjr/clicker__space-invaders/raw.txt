function threat(){let best=null,bv=-1e9;for(const e of enemies){const sc=e.y-Math.abs(e.x-W/2)*.02;if(sc>bv){bv=sc;best=e;}}return best;}

function buy(u,d){
 if(S.mode!=='play')return;
 const l=U[u.k];if(l>=u.max)return;
 const c=cost(u);
 if(S.coins<c){sfx.deny();d.classList.remove('deny');void d.offsetWidth;d.classList.add('deny');return;}
 S.coins-=c;U[u.k]=l+1;
 if(u.k==='shield')S.baseHP=Math.min(maxHP(),S.baseHP+30);
 sfx.buy();d.classList.remove('bought');void d.offsetWidth;d.classList.add('bought');
 shopSig='';save();
}

function startWave(n){
 S.wave=n;if(n>S.bestWave)S.bestWave=n;
 enemies.length=0;eB.length=0;pB.length=0;boss=null;S.cleared=false;S.interT=0;
 if(n%5===0){
  const i=Math.floor(n/5),hp=Math.round(380*(1+(i-1)*.7));
  const e={type:'boss',x:W/2,y:-70,hp,max:hp,flash:0,t:rnd(9),fireT:1.6,enraged:false};
  enemies.push(e);boss=e;waveTotal=1;
  banner('WAVE '+n,'⚠ BOSS — THE GOLDEN CRAWLER');
 }else{
  const cols=n<=2?5:(n<=5?6:7),rows=Math.min(6,2+Math.ceil(n/2));
  const fit=Math.min(1,(W-100)/(cols*52));fitS=fit;
  F={active:true,arrived:false,x:(W-(cols-1)*52*fit)/2,y:-(rows*44*fit)-50,targetY:70,dir:1,speed:26+n*3,sx:52*fit,sy:44*fit,cols,rows,fireT:1.3};
  const hpMul=1+(n-1)*.28;waveTotal=cols*rows;
  for(let r=0;r<rows;r++)for(let c=0;c<cols;c++){
   const t=r===0?'octo':(r===1?'crab':(c%2?'squid':'crab'));
   enemies.push({type:t,col:c,row:r,hp:Math.round(TYPES[t].hp*hpMul),max:Math.round(TYPES[t].hp*hpMul),flash:0,ph:rnd(9),x:-999,y:-999});
  }
  banner('WAVE '+n,'HOSTILES INBOUND — '+enemies.length+' CONTACTS');
 }
 sfx.wave();save();
}

/* ================= combat ================= */
function shoot(x,y,quiet){
 const cx=W/2,cy=baseLine-10;
 let dx=x-cx,dy=y-cy;if(dy>-6)dy=-6;
 const len=Math.hypot(dx,dy)||1;dx/=len;dy/=len;
 const n=1+U.split;
 for(let i=0;i<n;i++){const a=(i-(n-1)/2)*.09,ca=Math.cos(a),sa=Math.sin(a);
  pB.push({x:cx,y:cy-24,vx:(dx*ca-dy*sa)*560,vy:(dx*sa+dy*ca)*560,dmg:damage()});}
 muzzleT=.07;rec=3;retPulse=1;if(!quiet)sfx.shoot();
}
function popBullets(x,y){
 for(let i=eB.length-1;i>=0;i--){const b=eB[i];if(Math.hypot(b.x-x,b.y-y)<18){eB.splice(i,1);earn(5,x,y,false);spark(x,y,'#ff5fa8',6);sfx.pop();}}
}
function killEnemy(idx){
 const e=enemies[idx];enemies.splice(idx,1);
 S.kills++;S.runKills++;
 S.combo=Math.min(40,S.combo+1);S.comboT=3.5;
 const t=TYPES[e.type],big=e.type==='boss';
 earn(t.val*(1+(S.wave-1)*.07),e.x,e.y,big);
 explosion(e.x,e.y,t.color,big?46:14);sfx.boom(big);
 if(big){S.shake=16;redFlash=.25;save();}
}
function hurtBase(d){
 if(S.mode!=='play')return;
 S.baseHP=Math.max(0,S.baseHP-d);redFlash=.55;S.shake=Math.min(16,S.shake+7);sfx.hurt();
 if(S.baseHP<=0)gameOver();
}

/* ================= update ================= */
let autoAcc=0,droneAcc=0;
function step(dt){
 S.saveT+=dt;if(S.saveT>6){S.saveT=0;save();}
 streakT-=dt;if(streakT<0){streakT=rnd(3,7);parts.push({x:rnd(W),y:rnd(H*.4),vx:-rnd(200,320),vy:rnd(90,150),g:0,t:0,life:.8,s:1,c:'streak'});}
 if(S.comboT>0){S.comboT-=dt;if(S.comboT<=0)S.combo=0;}
 S.shake=Math.max(0,S.shake-dt*30);redFlash=Math.max(0,redFlash-dt*1.2);muzzleT=Math.max(0,muzzleT-dt);rec=Math.max(0,rec-dt*20);retPulse=Math.max(0,retPulse-dt*5);
 if(U.drone>0){S.coins+=U.drone*2*dt;S.total+=U.drone*2*dt;}

 // formation
 if(F.active&&!boss){
  if(!F.arrived){F.y+=(F.targetY-F.y)*Math.min(1,dt*1.6);if(F.targetY-F.y<3)F.arrived=true;}
  else{
   F.x+=F.dir*F.speed*dt;
   const min=46,maxx=W-46-(F.cols-1)*F.sx;
   if(F.x<min){F.x=min;F.dir=1;F.y+=16;}else if(F.x>maxx){F.x=maxx;F.dir=-1;F.y+=16;}
   if(S.wave>=4)F.y+=2*dt;
  }
  if(F.arrived){F.fireT-=dt;
   if(F.fireT<=0){F.fireT=clamp(2.3-S.wave*.07,.8,2.3);
    const cand=enemies.filter(e=>e.type!=='boss'&&e.y>10);
    if(cand.length){const e=pick(cand),d=eDim(e);
     const tx=W/2-e.x,ty=(baseLine-14)-e.y,tl=Math.hypot(tx,ty)||1;
     const sp=clamp(120+S.wave*6,120,300),ang=Math.atan2(ty,tx)+rnd(-.05,.05);
     eB.push({x:e.x,y:e.y+d.h/2,vx:Math.cos(ang)*sp,vy:Math.sin(ang)*sp,dmg:7});}
   }}
 }
 // positions + crash
 for(let i=enemies.length-1;i>=0;i--){const e=enemies[i];e.flash=Math.max(0,e.flash-dt*5);
  if(e.type==='boss'){
   e.t+=dt;e.y+=(105-e.y)*Math.min(1,dt*1.5);
   e.x=W/2+Math.sin(e.t*(e.enraged?1.4:.6))*(W*.32);
   if(!e.enraged&&e.hp<e.max*.3){e.enraged=true;banner('⚠ ENRAGED','THE CRAWLER RAGES');sfx.wave();}
   e.fireT-=dt*(e.enraged?1.8:1);
   if(e.fireT<=0&&e.y>40){e.fireT=1.5;
    const tx=W/2-e.x,ty=(baseLine-14)-e.y,ba=Math.atan2(ty,tx),sp=clamp(170+S.wave*6,170,340);
    for(let k=-1;k<=1;k++){const a=ba+k*.35;eB.push({x:e.x,y:e.y+24,vx:Math.cos(a)*sp,vy:Math.sin(a)*sp,dmg:13});}
   }
  }else{
   e.x=F.x+e.col*F.sx;e.y=F.y+e.row*F.sy+Math.sin(S.time*2.2+e.ph)*3;
   const d=eDim(e);
   if(e.y+d.h/2>baseLine){hurtBase(TYPES[e.type].dmg);explosion(e.x,baseLine-6,TYPES[e.type].color,18);enemies.splice(i,1);}
  }
 }
 // player bullets
 for(let i=pB.length-1;i>=0;i--){const b=pB[i];b.x+=b.vx*dt;b.y+=b.vy*dt;
  if(b.x<-20||b.x>W+20||b.y<-30||b.y>H+20){pB.splice(i,1);continue;}
  let hit=false;
  for(let j=enemies.length-1;j>=0;j--){const e=enemies[j],d=eDim(e);
   if(Math.abs(b.x-e.x)<d.w/2+3&&Math.abs(b.y-e.y)<d.h/2+3){hit=true;e.hp-=b.dmg;e.flash=1;spark(b.x,b.y,'#fff',4);sfx.hit();if(e.hp<=0)killEnemy(j);break;}}
  if(hit)pB.splice(i,1);
 }
 // enemy bullets
 for(let i=eB.length-1;i>=0;i--){const b=eB[i];b.x+=b.vx*dt;b.y+=b.vy*dt;
  if(b.x<-20||b.x>W+20||b.y>H+20){eB.splice(i,1);continue;}
  if(b.y>baseLine-2){hurtBase(b.dmg);explosion(b.x,baseLine-4,'#ff4d5e',8);eB.splice(i,1);}
 }
 // auto fire
 if(U.auto>0){autoAcc+=dt*U.auto;while(autoAcc>=1){autoAcc--;const t=threat();if(t)shoot(t.x,t.y,true);}}
 // drones
 if(U.drone>0){droneAcc+=dt;
  if(droneAcc>=.9){droneAcc=0;const t=threat();
   if(t){t.hp-=2.7*U.drone;t.flash=1;
    const cx=W/2,cy=baseLine-10,ang=S.time*1.8,dx=cx+Math.cos(ang)*46,dy=cy-10+Math.sin(ang)*16;
    beams.push({x1:dx,y1:dy,x2:t.x,y2:t.y,t:0,life:.14});sfx.hit();
    if(t.hp<=0)killEnemy(enemies.indexOf(t));}}}
 // wave clear
 if(S.mode==='play'&&enemies.length===0&&!S.cleared){
  S.cleared=true;const b=25+S.wave*12;earn(b,W/2,H*.4,false);
  popup(W/2,H*.34,'WAVE BONUS','#43ffb0',18);sfx.clear();S.interT=1.7;save();
 }
 if(S.cleared){S.interT-=dt;if(S.interT<=0)startWave(S.wave+1);}
 // particles / pops / flies / beams
 for(let i=parts.length-1;i>=0;i--){const p=parts[i];p.t+=dt;if(p.t>p.life){parts.splice(i,1);continue;}p.vy+=(p.g||0)*dt;p.x+=p.vx*dt;p.y+=p.vy*dt;}
 for(let i=pops.length-1;i>=0;i--){const p=pops[i];p.t+=dt;p.y-=32*dt;if(p.t>p.life)pops.splice(i,1);}
 const sr=stage.getBoundingClientRect(),br=bankEl.getBoundingClientRect();
 const tx=br.left-sr.left+8,ty=br.top-sr.top+8;
 for(let i=flies.length-1;i>=0;i--){const f=flies[i];f.t+=dt;
  const dx=tx-f.x,dy=ty-f.y;
  f.vx+=(dx*3.2-f.vx)*Math.min(1,dt*7);f.vy+=(dy*3.2-f.vy)*Math.min(1,dt*7);
  f.x+=f.vx*dt;f.y+=f.vy*dt;
  if(Math.hypot(dx,dy)<14||f.t>2){flies.splice(i,1);bumpBank();}}
 for(let i=beams.length-1;i>=0;i--){const b=beams[i];b.t+=dt;if(b.t>b.life)beams.splice(i,1);}
}
