const Snd={
  ctx:null,master:null,muted:false,
  init(){
    if(this.ctx){ if(this.ctx.state==='suspended') this.ctx.resume(); return; }
    try{
      const AC=window.AudioContext||window.webkitAudioContext; if(!AC) return;
      this.ctx=new AC();
      this.master=this.ctx.createGain();
      this.master.gain.value=this.muted?0:0.5;
      this.master.connect(this.ctx.destination);
    }catch(e){ this.ctx=null; }
  },
  setMuted(m){ this.muted=m; if(this.master) this.master.gain.value=m?0:0.5;
    try{localStorage.setItem('gp.mute',m?'1':'0');}catch(e){}
    sndChip.textContent='M · SOUND '+(m?'OFF':'ON'); sndChip.classList.toggle('off',m); },
  tone(f,dur,type,vol,slide,delay){
    if(!this.ctx) return; type=type||'sine'; vol=vol||.2; slide=slide||0; delay=delay||0;
    const c=this.ctx,t0=c.currentTime+delay,o=c.createOscillator(),g=c.createGain();
    o.type=type; o.frequency.setValueAtTime(f,t0);
    if(slide) o.frequency.exponentialRampToValueAtTime(Math.max(30,f+slide),t0+dur);
    g.gain.setValueAtTime(.0001,t0);
    g.gain.exponentialRampToValueAtTime(vol,t0+.012);
    g.gain.exponentialRampToValueAtTime(.0001,t0+dur);
    o.connect(g).connect(this.master); o.start(t0); o.stop(t0+dur+.05);
  },
  noise(dur,freq,vol,type,delay,sweep){
    if(!this.ctx) return; vol=vol||.2; type=type||'bandpass'; delay=delay||0; sweep=sweep||0;
    const c=this.ctx,t0=c.currentTime+delay,len=Math.max(1,(dur*c.sampleRate)|0);
    const buf=c.createBuffer(1,len,c.sampleRate),d=buf.getChannelData(0);
    for(let i=0;i<len;i++) d[i]=(Math.random()*2-1)*(1-i/len);
    const src=c.createBufferSource(); src.buffer=buf;
    const f=c.createBiquadFilter(); f.type=type; f.frequency.setValueAtTime(freq,t0);
    if(sweep) f.frequency.exponentialRampToValueAtTime(sweep,t0+dur);
    f.Q.value=1.2;
    const g=c.createGain(); g.gain.setValueAtTime(vol,t0); g.gain.exponentialRampToValueAtTime(.0001,t0+dur);
    src.connect(f).connect(g).connect(this.master); src.start(t0); src.stop(t0+dur+.02);
  },
  play(name,v){
    if(!this.ctx||this.muted) return; v=v||1;
    switch(name){
      case 'knock': this.noise(.07,240,.5*v,'bandpass'); this.tone(92+30*v,.1,'sine',.4*v,-30); break;
      case 'tick':  this.tone(300,.05,'triangle',.1*v,-80); this.noise(.03,2400,.06*v,'highpass'); break;
      case 'thud':  this.tone(70,.14,'sine',.5,-30); this.noise(.09,180,.3,'lowpass'); break;
      case 'whoosh':this.noise(.28,900,.14,'bandpass',0,260); break;
      case 'lip':   this.tone(500,.06,'square',.07,-200); break;
      case 'point': this.tone(659,.09,'triangle',.22); this.tone(880,.14,'triangle',.2,0,.09); break;
      case 'burr':  this.tone(170,.35,'sawtooth',.16,-90); this.noise(.2,120,.1,'lowpass'); break;
      case 'holein':[659,784,988].forEach((f,i)=>this.tone(f,.16,'triangle',.24,0,i*.09));
                   this.tone(1568,.3,'sine',.12,0,.3); this.noise(.4,5000,.05,'highpass',.3); break;
      case 'win':   [523,659,784,1047].forEach((f,i)=>this.tone(f,.18,'triangle',.24,0,i*.11));
                   this.tone(1319,.5,'sine',.18,0,.44); this.noise(.5,3000,.06,'highpass',.44); break;
      case 'ui':    this.tone(840,.04,'square',.06,-200); break;
    }
  }
};

function makeBallSprite(){
  const d=Math.max(8,Math.ceil(ball.r*4)),c=document.createElement('canvas');c.width=c.height=d;
  const g=c.getContext('2d'),r=d/2;
  const rg=g.createRadialGradient(r-r*.35,r-r*.4,r*.1,r,r,r);
  rg.addColorStop(0,'#ffffff');rg.addColorStop(.55,'#f2eee1');rg.addColorStop(.85,'#c9c3b0');rg.addColorStop(1,'#a29a86');
  g.fillStyle=rg;g.beginPath();g.arc(r,r,r,0,7);g.fill();
  const sp=d*.21,dr=d*.045;
  g.fillStyle='rgba(120,130,140,.3)';
  for(let row=-3;row<=3;row++)for(let col=-3;col<=3;col++){
    const x=r+col*sp+(row%2?sp*.5:0),y=r+row*sp;
    if(Math.hypot(x-r,y-r)<r*.8){g.beginPath();g.arc(x,y,dr,0,7);g.fill();}
  }
  g.strokeStyle='rgba(60,50,30,.3)';g.lineWidth=d*.02;g.beginPath();g.arc(r,r,r-g.lineWidth/2,0,7);g.stroke();
  ballSprite=c;
}

function updateParticles(dt){
  for(let i=parts.length-1;i>=0;i--){const p=parts[i];p.t+=dt;
    if(p.t>=p.life){parts.splice(i,1);continue;}
    p.vy+=p.g*dt;p.vx*=Math.exp(-p.drag*dt);p.x+=p.vx*dt;p.y+=p.vy*dt;p.rot+=p.vr*dt;}
}

    game.state='point';game.pointT=.6;game.serveTo=Math.random()<.5?'west':'east';return;
  }

    sfx('holein');
    if(side){game.scores[side]++;game.holesIn[side]++;hudScore(side);main='HOLE-IN ONE!';sub=cap(side)+' DROPS IT IN';}

const MAXSPD=110;
function humanPaddle(p,dt){
  let dir=0;
  if(p.side==='west'){if(keys['KeyW'])dir--;if(keys['KeyS'])dir++;}
  else{if(keys['ArrowUp'])dir--;if(keys['ArrowDown'])dir++;}
  let target=null;
  if(!game.attract&&game.mode==='1p'&&p.side==='west'&&mouse.y!=null&&performance.now()-mouse.t<2500)target=mouse.y;
  if(touchAim[p.side]!=null)target=touchAim[p.side];
  if(target!=null){
    const dy=clamp(target-p.y,-MAXSPD*U*dt,MAXSPD*U*dt);
    p.vy=dy/Math.max(dt,1e-4);p.y+=dy;
  }else{
    if(dir!==0)p.vy+=dir*2000*U*dt;else p.vy*=Math.exp(-14*dt);
    p.vy=clamp(p.vy,-MAXSPD*U,MAXSPD*U);p.y+=p.vy*dt;
  }
  p.y=clamp(p.y,TOP+p.h/2+4,H-BOT-p.h/2-4);
}
function aiPaddle(p,dt){
  const a=game.attract?AI.pro:AI[game.cpu];
  p.aiT-=dt;
  if(p.aiT<=0){
    p.aiT=a.react*rand(.7,1.3);
    const toward=(p.side==='east'&&ball.vx>0)||(p.side==='west'&&ball.vx<0);
    if(toward&&game.state==='play'){
      const tHit=Math.abs((p.x-ball.x)/ball.vx);
      let ty=ball.y+ball.vy*tHit*.6;
      ty=clamp(ty,TOP+p.h/2,H-BOT-p.h/2);
      p.aiY=ty+rand(-1,1)*a.err*U;
    }else p.aiY=H/2+Math.sin(game.t*.8+p.phase)*16*U;
  }
  const move=clamp(p.aiY-p.y,-a.spd*U*dt,a.spd*U*dt);
  p.vy=move/Math.max(dt,1e-4);p.y+=move;
  p.y=clamp(p.y,TOP+p.h/2+4,H-BOT-p.h/2-4);
}

function beginHoleIn(i,hitter){
  const c=cups()[i];
  ball.cupIdx=i;ball.cupT=0;ball.cupAng=Math.atan2(ball.y-c.y,ball.x-c.x);
  ball.cupR0=Math.hypot(ball.x-c.x,ball.y-c.y);ball.cupHitter=hitter;
  game.state='cup';
  if(!game.attract){game.tsTarget=.3;game.tsHold=.5;}
  sfx('tick');
}
function collidePaddle(p){
  const pw=p.w/2+ball.r,phh=p.h/2+ball.r;
  if(Math.abs(ball.x-p.x)<pw&&Math.abs(ball.y-p.y)<phh){
    const toward=p.side==='west'?ball.vx<0:ball.vx>0;
    if(toward||Math.abs(ball.vx)<5*U){
      const off=clamp((ball.y-p.y)/(p.h/2),-1.4,1.4);
      const dir=p.side==='west'?1:-1;
      ball.x=p.x+dir*pw;
      let sp=Math.max(Math.abs(ball.vx)*1.05+3*U,20*U);
      sp=Math.min(sp,96*U);
      ball.vx=dir*sp;
      ball.vy=off*47*U+p.vy*.38;
      const m=Math.hypot(ball.vx,ball.vy),capV=120*U;
      if(m>capV){ball.vx*=capV/m;ball.vy*=capV/m;}
      ball.spin=clamp(p.vy*.5+off*40*U,-70*U,70*U);
      ball.squash=Math.max(ball.squash,.5);p.squash=1;p.glow=1;
      game.lastHitter=p.side;game.rally++;onRally();
      sfx('knock',clamp(sp/(90*U),.4,1));
      sparks(ball.x,ball.y,dir);
      if(m>75*U)game.shake=Math.max(game.shake,3);
    }else ball.x=p.x+(ball.x>p.x?pw:-pw);
  }
}
function updateServe(dt){
  ball.vy+=1500*U*dt;ball.y+=ball.vy*dt;
  if(ball.y>=H/2){
    ball.y=H/2;ball.squash=1;sfx('thud');dust(ball.x,H/2+ball.r,8);
    ball.vx=game.serveDir*(46+rand(10))*U;ball.vy=rand(-15,9)*U;ball.spin=0;
    game.state='play';sfx('whoosh');
    if(!game.hintDone&&!game.attract){game.hintDone=true;showBanner('TEE OFF!','KEEP IT ALIVE — DROP IT SLOW OVER A CUP',2.2);}
  }
}
function updatePlay(dt){
  ball.vx*=Math.exp(-.5*dt);ball.vy*=Math.exp(-.5*dt);
  ball.vy+=ball.spin*dt;ball.spin*=Math.exp(-1.4*dt);
  ball.x+=ball.vx*dt;ball.y+=ball.vy*dt;
  const sp0=Math.hypot(ball.vx,ball.vy);
  ball.roll+=(sp0/ball.r)*dt*(ball.vx>=0?1:-1);
  if(sp0>20*U){ball.trail.push({x:ball.x,y:ball.y});if(ball.trail.length>22)ball.trail.shift();}
  // walls
  if(ball.y<TOP+ball.r){ball.y=TOP+ball.r;ball.vy=Math.abs(ball.vy)*.98;ball.squash=Math.max(ball.squash,.4);if(sp0>25*U){sfx('tick');grassPuff(ball.x,TOP,4);}}
  if(ball.y>H-BOT-ball.r){ball.y=H-BOT-ball.r;ball.vy=-Math.abs(ball.vy)*.98;ball.squash=Math.max(ball.squash,.4);if(sp0>25*U){sfx('tick');grassPuff(ball.x,H-BOT,4);}}
  // paddles
  collidePaddle(pads.west);collidePaddle(pads.east);
  if(game.state!=='play')return;
  const sp=Math.hypot(ball.vx,ball.vy);
  ball.lipCd-=dt;
  // cups
  for(let i=0;i<2;i++){
    const c=cups()[i],d=Math.hypot(ball.x-c.x,ball.y-c.y);
    if(d<c.r*.9&&sp<34*U){beginHoleIn(i,game.lastHitter);return;}
    if(d>c&&d<c.r*1.6&&sp>=34*U&&ball.lipCd<=0){
      const ang=Math.atan2(ball.y-c.y,ball.x-c.x),push=40*U;
      ball.vx+=Math.cos(ang)*push;ball.vy+=Math.sin(ang)*push;ball.lipCd=.35;
      sfx('lip');textPop(c.x,c.y-c.r-8,'LIP!');
    }
  }
  if(game.state!=='play')return;
  // burr (dead ball)
  if(sp<8*U)ball.deadT+=dt;else ball.deadT=0;
  if(ball.deadT>1.1){
    dust(ball.x,ball.y,14);
    if(game.lastHitter)awardPoint(opp(game.lastHitter),'burr');
    else{showBanner('DEAD BALL','RE-TEE THE BALL');game.state='point';game.pointT=1.2;game.serveTo=Math.random()<.5?'west':'east';}
    return;
  }
  // escapes
  if(ball.x<-40)awardPoint('east','escape');
  else if(ball.x>W+40)awardPoint('west','escape');
}
function updateCup(dt){
  const c=cups()[ball.cupIdx];
  ball.cupT+=dt/.55;const t=clamp(ball.cupT,0,1);
  const ang=ball.cupAng+t*7,rad=(1-t)*(1-t)*ball.cupR0;
  ball.x=c.x+Math.cos(ang)*rad;ball.y=c.y+Math.sin(ang)*rad*.6;
  ball.roll+=dt*25;ball.scale=1-t*.85;
  if(t>=1){confetti(c.x,c.y,34);game.shake=4;awardPoint(ball.cupHitter,'holein');}
}

/* ================= update ================= */
function update(dtRaw){
  game.t+=dtRaw;
  if(game.tsHold>0)game.tsHold-=dtRaw;else game.tsTarget=1;
  game.ts+=(game.tsTarget-game.ts)*Math.min(1,dtRaw*7);
  const dt=dtRaw*game.ts;
  game.shake*=Math.exp(-5.5*dtRaw);
  if(game.state==='paused')return;
  updateParticles(dt);
  if(banner.t>0)banner.t-=dtRaw;
  for(const s of ['west','east']){const p=pads[s];
    aiControlled(s)?aiPaddle(p,dt):humanPaddle(p,dt);
    p.squash*=Math.exp(-7*dtRaw);p.glow*=Math.exp(-3.5*dtRaw);}
  ball.squash*=Math.exp(-8*dtRaw);
  switch(game.state){
    case 'serve':updateServe(dt);break;
    case 'play':updatePlay(dt);break;
    case 'cup':updateCup(dt);break;
    case 'point':
      game.pointT-=dtRaw;
      if(game.pointT<=0){if(game.pendingOver)doGameOver();else startServe(game.serveTo);}
      break;
  }
}

function rr(x,y,w,h,r){ctx.beginPath();ctx.moveTo(x+r,y);ctx.arcTo(x+w,y,x+w,y+h,r);ctx.arcTo(x+w,y+h,x,y+h,r);ctx.arcTo(x,y+h,x,y,r);ctx.arcTo(x,y,x+w,y,r);ctx.closePath();}
function drawCup(i){
  const c=cups()[i],cs=cups(),near=(game.state==='play'||game.state==='cup')&&Math.hypot(ball.x-c.x,ball.y-c.y)<7*U;
  ctx.fillStyle='rgba(0,0,0,.28)';ctx.beginPath();ctx.arc(c.x,c.y,c.r*1.22,0,7);ctx.fill();
  const rg=ctx.createRadialGradient(c.x,c.y,0,c.x,c.y,c.r);
  rg.addColorStop(0,'#01130c');rg.addColorStop(.8,'#04231a');rg.addColorStop(1,'#0a3527');
  ctx.fillStyle=rg;ctx.beginPath();ctx.arc(c.x,c.y,c.r,0,7);ctx.fill();
  ctx.strokeStyle='rgba(242,233,212,.5)';ctx.lineWidth=1.5;ctx.beginPath();ctx.arc(c.x,c.y,c.r,0,7);ctx.stroke();
  // capture hint ring
  if(near&&game.state==='play'){
    const sp=Math.hypot(ball.vx,ball.vy),d=Math.hypot(ball.x-c.x,ball.y-c.y);
    if(d<c.r*1.8&&sp<45*U){
      ctx.strokeStyle='rgba(240,201,107,'+(.28+.2*Math.sin(game.t*10))+')';ctx.lineWidth=2;
      ctx.beginPath();ctx.arc(c.x,c.y,c.r*(1.15+.06*Math.sin(game.t*10)),0,7);ctx.stroke();
    }
  }

  ctx.font='italic 700 '+Math.max(9,U*1.7)+'px Georgia, serif';
  ctx.fillStyle='rgba(242,233,212,.4)';ctx.textAlign='left';
  ctx.fillText(String(i+1),c.x+c.r+5,c.y-c.r+2);

  ctx.strokeStyle='rgba(217,207,174,.9)';ctx.lineWidth=2;
  ctx.beginPath();ctx.moveTo(c.x,c.y-2);ctx.lineTo(topX,topY);ctx.stroke();

  ctx.moveTo(topX,topY);
  ctx.quadraticCurveTo(topX+L*.5,topY-2+wav*.4,topX+L,topY+wav*.7+1);
  ctx.lineTo(topX,topY+4.5);ctx.closePath();ctx.fill();
}
function drawPaddle(p){
  const sx=1+p.squash*.25,sy=1-p.squash*.22;
  ctx.save();ctx.translate(p.x,p.y);ctx.scale(sx,sy);
  if(p.glow>.05){ctx.shadowColor='rgba(240,201,107,'+p.glow*.8+')';ctx.shadowBlur=18*p.glow;}
  const w=p.w,h=p.h;
  const g=ctx.createLinearGradient(-w/2,0,w/2,0);
  g.addColorStop(0,'#8a5a2e');g.addColorStop(.5,'#9c6a38');g.addColorStop(1,'#5d3a1c');
  ctx.fillStyle=g;rr(-w/2,-h/2,w,h,Math.min(8,w*.4));ctx.fill();
  ctx.shadowBlur=0;
  ctx.strokeStyle='#3c2410';ctx.lineWidth=2;rr(-w/2,-h/2,w,h,Math.min(8,w*.4));ctx.stroke();
  ctx.strokeStyle='rgba(40,20,5,.25)';ctx.lineWidth=1;
  for(let i=-1;i<=1;i++){ctx.beginPath();ctx.moveTo(-w*.3,i*h*.26);ctx.lineTo(w*.3,i*h*.26+(i%2?2:0));ctx.stroke();}
  const bg=ctx.createLinearGradient(0,-h*.09,0,h*.09);
  bg.addColorStop(0,'#f0c96b');bg.addColorStop(1,'#b57f24');
  ctx.fillStyle=bg;ctx.fillRect(-w/2+1.5,-h*.09,w-3,h*.18);
  ctx.strokeStyle='rgba(0,0,0,.3)';ctx.strokeRect(-w/2+1.5,-h*.09,w-3,h*.18);

}
function render(){
  // background
  const bg=ctx.createLinearGradient(0,0,0,H);
  bg.addColorStop(0,'#114731');bg.addColorStop(.5,'#0c3527');bg.addColorStop(1,'#062018');
