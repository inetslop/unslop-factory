const sfx=(()=>{
  let ac=null, master=null, nbuf=null, muted=false;
  function init(){
    if(ac){ if(ac.state==='suspended') ac.resume(); return; }
    try{ ac=new (window.AudioContext||window.webkitAudioContext)(); }catch(e){ return; }
    master=ac.createGain(); master.gain.value=muted?0:0.55;
    const comp=ac.createDynamicsCompressor(); master.connect(comp); comp.connect(ac.destination);
    nbuf=ac.createBuffer(1, ac.sampleRate, ac.sampleRate);
    const d=nbuf.getChannelData(0); for(let i=0;i<d.length;i++) d[i]=Math.random()*2-1;
    // ambient wind
    const src=ac.createBufferSource(); src.buffer=nbuf; src.loop=true;
    const f=ac.createBiquadFilter(); f.type='bandpass'; f.frequency.value=240; f.Q.value=.5;
    const wg=ac.createGain(); wg.gain.value=0.018;
    const lfo=ac.createOscillator(); lfo.frequency.value=0.07;
    const lg=ac.createGain(); lg.gain.value=0.012;
    lfo.connect(lg); lg.connect(wg.gain);
    src.connect(f); f.connect(wg); wg.connect(master); src.start(); lfo.start();
  }
  function env(g,t0,peak,dur){ g.gain.setValueAtTime(0.0001,t0); g.gain.linearRampToValueAtTime(peak,t0+0.006); g.gain.exponentialRampToValueAtTime(0.0001,t0+dur); }
  function osc(type,f0,f1,dur,peak,delay=0){
    if(!ac) return; const t0=ac.currentTime+delay;
    const o=ac.createOscillator(); o.type=type; o.frequency.setValueAtTime(f0,t0);
    if(f1) o.frequency.exponentialRampToValueAtTime(Math.max(24,f1),t0+dur);
    const g=ac.createGain(); env(g,t0,peak,dur);
    o.connect(g); g.connect(master); o.start(t0); o.stop(t0+dur+0.06);
  }
  function noise(dur,type,freq,peak,delay=0,slideTo){
    if(!ac) return; const t0=ac.currentTime+delay;
    const s=ac.createBufferSource(); s.buffer=nbuf; s.loop=true;
    const f=ac.createBiquadFilter(); f.type=type; f.frequency.setValueAtTime(freq,t0);
    if(slideTo) f.frequency.exponentialRampToValueAtTime(slideTo,t0+dur);
    const g=ac.createGain(); env(g,t0,peak,dur);
    s.connect(f); f.connect(g); g.connect(master); s.start(t0); s.stop(t0+dur+0.12);
  }
  return {
    init,
    toggle(){ muted=!muted; if(ac) master.gain.setTargetAtTime(muted?0:0.55, ac.currentTime, .05); return muted; },
    tick(i){ osc('square', 240+Math.min(i,12)*28, 0, .045, .05); },
    kill(){ noise(.16,'lowpass',900,.5); osc('sine',150,55,.14,.5); noise(.3,'lowpass',320,.28,.03); },
    flare(){ noise(.32,'bandpass',700,.35,0,2600); osc('square',900,180,.3,.12); },
    error(){ osc('sawtooth',140,62,.18,.14); },
    hurt(){ osc('sine',90,36,.32,.6); noise(.22,'lowpass',220,.5); },
    wave(){ osc('square',392,0,.09,.12); osc('square',587,0,.14,.12,.11); },
    over(){ [330,262,196].forEach((f,i)=>osc('triangle',f,f*.97,.42,.2,i*.3)); }
  };
})();

function hiLen(){ return Math.min(9, 5+Math.floor((wave-1)/2)); }
function pickWord(minL,maxL){
  for(let t=0;t<14;t++){ const L=minL+((Math.random()*(maxL-minL+1))|0); const arr=WORDS[L]; if(!arr) continue;
    const w=arr[(Math.random()*arr.length)|0]; if(!usedWords.has(w)){ usedWords.add(w); return w; } }
  const L=minL+((Math.random()*(maxL-max(4,minL)+1))|0); const arr=WORDS[Math.max(4,L)]||WORDS[4];
  return arr[(Math.random()*arr.length)|0];
}
function spawnZombie(menuMode){
  if(zombies.length>=(menuMode?7:14)) return;
  let t='walker'; const r=Math.random();
  if(!menuMode && wave>=4 && r<.16+wave*.008) t='brute';
  else if(!menuMode && wave>=2 && r<.34+wave*.01) t='runner';
  let minL=4, maxL=hiLen();
  if(t==='runner') maxL=Math.max(4,hiLen()-1);
  if(t==='brute'){ minL=Math.min(9,Math.max(6,hiLen())); maxL=Math.min(11,hiLen()+2); }
  const word=pickWord(minL,maxL);
  const cfg=TYPES[t];
  const base = menuMode ? W*.03 : Math.min(W*(.045+.0045*Math.min(wave,14)), W*.16);
  const z={ type:t, x:W+50+Math.random()*90, s:cfg.s, speed:base*cfg.mul*(.85+Math.random()*.3),
            word, phase:Math.random()*6.28, dead:false };
  ctx.font='bold '+Math.max(13,Math.round(15*z.s))+'px '+MONO;
  z.bw=ctx.measureText(word).width;
  zombies.push(z);
}

/* ============================== COMBAT LOGIC ============================== */
function bestMatch(str){ let b=null; for(const z of zombies){ if(!z.dead && z.word.startsWith(str)){ if(!b||z.x<b.x) b=z; } } return b; }
function gunTip(){ const rec=fireT>0? 5*(fireT/0.09):0; return {x:playerX+58-rec, y:footY-47}; }

function onLetter(L){
  if(state!==ST.PLAY) return;
  const cand=input+L;
  const z=bestMatch(cand);
  if(z){ input=cand; correctChars++; sfx.tick(input.length);
    if(input===z.word) killZombie(z,'type');
  } else { errors++; combo=0; input=''; sfx.error(); flash($('flashRed'),.35); shake(3,.12); }
}

function killZombie(z,cause){
  z.dead=true; usedWords.delete(z.word);
  kills++; combo++; maxCombo=Math.max(maxCombo,combo);
  const mult=1+Math.floor(combo/5);
  const pts=z.word.length*15*mult; score+=pts; input='';
  popup(z.x, footY-104*z.s, '+'+pts, '#ffd23f');
  if(combo%5===0 && charges<5){ charges++; popup(playerX+70, footY-130, 'FLARE +1', '#9fe870', 16); }
  const cfg=TYPES[z.type];
  burst(z.x, footY-46*z.s, z.s, [cfg.skin,cfg.shirt,'#5c8a48'], 16);
  addSplat(z.x,z.s);
  fireT=0.09;
  if(cause==='flare'){ const t=gunTip(); flareFx={x1:t.x,y1:t.y,x2:z.x-10*z.s,y2:footY-50*z.s,life:.18}; sfx.flare(); shake(7,.2); }
  else { sfx.kill(); shake(4,.16); }
  const tip=gunTip();
  for(let i=0;i<3;i++) parts.push({x:tip.x+4,y:tip.y-2,vx:20+Math.random()*40,vy:-15-Math.random()*20,g:-20,life:.5,max:.5,size:3+Math.random()*3,col:'rgba(160,170,180,.5)'});
  killsInWave++;
  if(killsInWave>=target()){
    wave++; killsInWave=0; hp=Math.min(100,hp+8);
    popup(playerX+30, footY-120, '+8 HP', '#7fd25a', 16);
    flash($('flashGreen'),.4); sfx.wave(); banner('WAVE '+wave);
  }
}

function breach(z){
  z.dead=true; usedWords.delete(z.word);
  hp-=TYPES[z.type].dmg; combo=0; input='';
  burst(playerX+30, footY-40, 1, ['#7a2b22','#4a5c3a','#222'], 12);
  flash($('flashRed'),.55); shake(13,.32); sfx.hurt();
  if(hp<=0){ hp=0; gameOver(); }
}

function doFlare(){
  if(state!==ST.PLAY || charges<=0) return;
  let near=null; for(const z of zombies){ if(!z.dead && (!near||z.x<near.x)) near=z; }
  if(!near) return;
  charges--; killZombie(near,'flare');
}

/* ============================== UPDATE ============================== */
function update(dt){
  fireT=Math.max(0,fireT-dt);
  if(shakeT>0) shakeT-=dt;
  for(const c of clouds){ c.x-=c.v*dt; if(c.x<-260*c.s){ c.x=W+120; c.y=H*(.08+Math.random()*.2); } }

  if(state===ST.PLAY){
    gameT+=dt; spawnT-=dt*1000;
    if(spawnT<=0){
      spawnZombie(false);
      if(wave>=4 && Math.random()<.22) spawnZombie(false);
      spawnT=Math.max(650, 2300-wave*150)*(.7+Math.random()*.6);
    }
    for(const z of zombies){
      if(z.dead) continue;
      z.x-=z.speed*dt; z.phase+=dt*(2.2+z.speed*.05);
      if(z.x<=playerX+30) breach(z);
    }
    zombies=zombies.filter(z=>!z.dead);
  }
  else if(state===ST.MENU){
    menuSpawnT-=dt*1000;
    if(menuSpawnT<=0){ spawnZombie(true); menuSpawnT=2000+Math.random()*1500; }
    for(const z of zombies){ z.x-=z.speed*dt; z.phase+=dt*(2+z.speed*.05);
      if(z.x<-70){ z.x=W+60+Math.random()*80; const old=z.word; usedWords.delete(old);
        z.word=pickWord(4,7); ctx.font='bold '+Math.max(13,Math.round(15*z.s))+'px '+MONO; z.bw=ctx.measureText(z.word).width; } }
  }
  else if(state===ST.OVER){
    for(const z of zombies){ if(z.dead) continue;
      const lim=playerX+34; if(z.x>lim){ z.x=Math.max(lim, z.x-z.speed*.25*dt); z.phase+=dt*1.6; } }
  }

  for(const p of parts){ p.vy+=(p.g||0)*dt; p.x+=p.vx*dt; p.y+=p.vy*dt; if(p.rot!==undefined)p.rot+=p.vr*dt; p.life-=dt; }
  parts=parts.filter(p=>p.life>0); if(parts.length>320) parts.splice(0,parts.length-320);
  for(const p of popups){ p.y-=42*dt; p.life-=dt; }
  popups=popups.filter(p=>p.life>0);
  for(const s of splats) s.a-=dt/9;
  splats=splats.filter(s=>s.a>0);
  if(flareFx){ flareFx.life-=dt; if(flareFx.life<=0) flareFx=null; }

  $('pulse').classList.toggle('on', state===ST.PLAY && hp<=30);
}

function drawBubble(z){
  const fs=Math.max(13,Math.round(15*z.s));
  ctx.font='bold '+fs+'px '+MONO;
  const active = input && z.word.startsWith(input) && state===ST.PLAY;
  const bobY=Math.sin(time*2+z.phase)*2;
  const bw=z.bw+16, bh=fs+13;
  let bx=z.x-bw/2, by=footY-z.s*94-bh-8+bobY; if(by<4) by=4;
  const prox=Math.max(0,Math.min(1,(z.x-playerX-30)/260));
  const rC=Math.round(159+(255-159)*(1-prox)), gC=Math.round(232+(75-232)*(1-prox)), bC=Math.round(112+(58-112)*(1-prox));
  ctx.fillStyle='rgba(7,13,9,.85)';
  ctx.beginPath();
  if(ctx.roundRect) ctx.roundRect(bx,by,bw,bh,4); else ctx.rect(bx,by,bw,bh);
  ctx.fill();
  ctx.strokeStyle=active?'#ffd23f':`rgb(${rC},${gC},${bC})`;
  ctx.lineWidth=active?2:1.5; if(active){ ctx.save(); ctx.shadowColor='#ffd23f'; ctx.shadowBlur=10; }
  ctx.stroke(); if(active) ctx.restore();
  // pointer
  ctx.fillStyle='rgba(7,13,9,.85)';
  ctx.beginPath(); ctx.moveTo(z.x-5,by+bh); ctx.lineTo(z.x+5,by+bh); ctx.lineTo(z.x,by+bh+6); ctx.closePath(); ctx.fill();
  // text
  const typed=active?input.length:0;
  ctx.fillStyle='#ffd23f'; ctx.fillText(z.word.slice(0,typed), bx+8, by+bh-4);
  ctx.fillStyle='#cfe6c0'; if(typed<z.word.length) ctx.fillText(z.word.slice(typed), bx+8+typed*(ctx.measureText('M').width), by+bh-4);
  // reticle corners
  if(active && state===ST.PLAY){
    const ph=(time*3)%2, o=3+(ph<1?ph:2-ph)*2.5;
    ctx.strokeStyle='#ffd23f'; ctx.lineWidth=2;

  }
}

}
