  while(segs.length<sn.len && idx<pts.length){

  const e={type,x,y,angle:rand(Math.PI*2),t:rand(9),cd:rand(.8,1.8),strafe:Math.random()<.5?-1:1,rot:rand(9),hitT:0,dead:false};
  if(type==='mite'){e.r=8;e.hp=1;e.spd=Math.min(125,58+lvl*6);}
  if(type==='drone'){e.r=13;e.hp=2;e.spd=Math.min(115,70+lvl*4);e.cd=rand(1.2,2);}
  if(type==='mine'){e.r=9;e.hp=1;const a=rand(Math.PI*2);e.vx=Math.cos(a)*135;e.vy=Math.sin(a)*135;}
  if(type==='brute'){e.r=17;e.hp=5;e.spd=Math.min(82,40+lvl*4);}
  world.enemies.push(e);
}

function killEnemy(e){
  if(e.dead) return; e.dead=true; world.kills++;
  const mult=addChain();
  addScore(VALUES[e.type],mult,e.x,e.y,{mite:'#c95cff',mine:'#e08aff',drone:'#ff2e9a',brute:'#ff6ab0'}[e.type]);
  explode(e.x,e.y,e.type==='brute'||e.type==='drone', {mite:'#c95cff',mine:'#e08aff',drone:'#ff2e9a',brute:'#ff6ab0'}[e.type]);
  (e.type==='brute'||e.type==='drone')?sfx.boom():sfx.kill();
  if(e.type==='brute'){ spawnFood(); spawnFood(); if(Math.random()<.5) dropShield(e.x,e.y); }
  else if(Math.random()<.42) spawnFoodAt(e.x+rand(-14,14),e.y+rand(-14,14));
  else if(Math.random()<.12) dropShield(e.x,e.y);
  if(e.type==='mine') chainExplode(e);
}

}

/* ---------- snake update ---------- */
function computeSegs(sn){
  const pts=sn.path, segs=[]; let cum=0, idx=1, target=SEG_SP;
  while(segs.length<sn.len && idx<pts.length){
    const a=pts[idx-1], b=pts[idx];
    const d=Math.hypot(b.x-a.x,b.y-a.y);
    if(cum+d>=target){ const t=(target-cum)/d; segs.push({x:a.x+(b.x-a.x)*t,y:a.y+(b.y-a.y)*t}); target+=SEG_SP; }
    else { cum+=d; idx++; }
  }
  while(segs.length<sn.len) segs.push({x:pts[0].x,y:pts[0].y});
  sn.segs=segs;
}
function steerSnake(sn,dt){
  let tx=0,ty=0;
  if(world.attract){
    // autopilot: chase nearest food, hug the open water
    let tf=null,bd=1e9;
    for(const f of world.foods){ const d=dist2(f.x,f.y,sn.x,sn.y); if(d<bd){bd=d;tf=f;} }
    if(!tf) tf={x:W/2,y:H/2};
    tx=tf.x-sn.x; ty=tf.y-sn.y;
    const m=170; let ax=0,ay=0;
    if(sn.x<m)ax+=m-sn.x; if(sn.x>W-m)ax-=sn.x-(W-m);
    if(sn.y<m)ay+=m-sn.y; if(sn.y>H-m)ay-=sn.y-(H-m);
    const urg=clamp(1-Math.min(ax,ay)/m,0,1)*(ax||ay?1:0);
    if(ax||ay){ const cx=(W/2-sn.x),cy=(H/2-sn.y),cl=Math.hypot(cx,cy)||1;
      tx=tx*(1-urg)+cx/cl*urg*600; ty=ty*(1-urg)+cy/cl*urg*600; }
  } else if(input.mode==='key'){
    tx=(input.right?1:0)-(input.left?1:0); ty=(input.down?1:0)-(input.up?1:0);
  } else {
    tx=input.mx-sn.x; ty=input.my-sn.y;
    if(tx*tx+ty*ty<26*26){ tx=0; ty=0; }
  }
  if(tx||ty){
    const ta=Math.atan2(ty,tx), da=angDiff(ta,sn.angle);
    const grip=clamp(1-(sn.len-8)*.005,.58,1);
    sn.angle+=clamp(da,-TURN*grip*dt,TURN*grip*dt);
  }
}
function updateSnake(dt,withCombat){
  const sn=world.snake;
  steerSnake(sn,dt);
  sn.vx=Math.cos(sn.angle)*SPEED; sn.vy=Math.sin(sn.angle)*SPEED;
  sn.x+=sn.vx*dt; sn.y+=sn.vy*dt;
  sn.recoil=Math.max(0,sn.recoil-26*dt);
  sn.tongueT-=dt; if(sn.tongueT<0) sn.tongueT=rand(1.8,3.4);
  if(!world.attract){
    sn.inv=Math.max(0,sn.inv-dt);
    sn.fireT-=dt;
    if((input.space||input.down2)&&sn.fireT<=0){
      sn.fireT=FIRE_CD; sn.recoil=3;
      const nx=sn.x+Math.cos(sn.angle)*16, ny=sn.y+Math.sin(sn.angle)*16;
      world.bullets.push({x:nx,y:ny,vx:Math.cos(sn.angle)*BUL_SPEED+sn.vx*.25,vy:Math.sin(sn.angle)*BUL_SPEED+sn.vy*.25});
      spark(nx,ny,'#ffd75e',3,140); sfx.shoot();
    }
  }
  sn.path.unshift({x:sn.x,y:sn.y});
  const maxPts=Math.min(1600,Math.ceil((sn.len+4)*SEG_SP/2.2));
  if(sn.path.length>maxPts) sn.path.length=maxPts;
  computeSegs(sn);

  if(!withCombat) return;

  // walls — always lethal (edge glow warns you)
  if(sn.x<HEAD_R+5||sn.x>W-HEAD_R-5||sn.y<HEAD_R+5||sn.y>H-HEAD_R-5){ die('wall'); return; }

  // self collision (forgiving neck)
  if(!world.attract&&sn.inv<=0){
    const n=sn.segs.length;
    for(let i=9;i<n;i+=2){
      const t=i/n, r=lerp(9,3.5,t);
      if(dist2(sn.x,sn.y,sn.segs[i].x,sn.segs[i].y)<Math.pow((HEAD_R+r)*.82,2)){ die('self'); return; }
    }
  }

  // food (with magnet)
  for(let i=world.foods.length-1;i>=0;i--){
    const f=world.foods[i]; f.t+=dt*4;
    const d2=dist2(f.x,f.y,sn.x,sn.y);
    if(d2<95*95&&!world.attract){
      const d=Math.sqrt(d2)||1;
      f.x+=(sn.x-f.x)/d*Math.min(420,d*.6)*dt*3; f.y+=(sn.y-f.y)/d*Math.min(420,d*.6)*dt*3;
    }
    if(Math.random()<dt*1.5) world.parts.push({x:f.x+rand(-4,4),y:f.y,vx:0,vy:-18,life:.5,max:.5,size:1,col:'#4df0c8',drag:0});
    if(d2<(HEAD_R+13)*(HEAD_R+13)){
      world.foods.splice(i,1);
      if(!world.attract){
        sn.inv=Math.max(sn.inv,1.1);
        addScore(FOOD_VAL,world.combo.chain?Math.min(6,1+Math.floor(world.combo.chain/2)):1,f.x,f.y,'#4df0c8');
        sn.len++; sfx.eat(); spark(f.x,f.y,'#4df0c8',12,170); ring(f.x,f.y,'#4df0c8',260,30);
      } else { sn.len++; spark(f.x,f.y,'#4df0c8',8,120); }
    }
  }

  // shield pickups
  for(let i=world.pickups.length-1;i>=0;i--){
    const p=world.pickups[i]; p.t+=dt*3; p.life-=dt;
    if(p.life<=0){ world.pickups.splice(i,1); continue; }
    if(dist2(p.x,p.y,sn.x,sn.y)<(HEAD_R+15)*(HEAD_R+15)){
      world.pickups.splice(i,1); sn.inv=Math.max(sn.inv,3); sfx.shield();
      popup(p.x,p.y,'SHIELD','#6fd7ff'); ring(p.x,p.y,'#6fd7ff',420,54); spark(p.x,p.y,'#6fd7ff',14,200);
    }
  }

  // enemy contact
  if(!world.attract){
    for(const e of world.enemies){
      if(e.dead) continue;
      const rr=e.r+HEAD_R-2;
      if(dist2(e.x,e.y,sn.x,sn.y)<rr*rr){
        if(sn.inv>0){ killEnemy(e); popup(sn.x,sn.y-14,'SMASH','#6fd7ff'); }
        else { die(e.type==='mine'?'mine':e.type); return; }
      }
    }
  }
}

/* ---------- enemies ---------- */
function updateEnemies(dt){
  const sn=world.snake;
  for(const e of world.enemies){
    if(e.dead) continue;
    e.t+=dt; e.hitT=Math.max(0,e.hitT-dt);
    const dx=sn.x-e.x, dy=sn.y-e.y, d=Math.hypot(dx,dy)||1;
    if(e.type==='mite'){
      const a=Math.atan2(dy,dx);
      e.x+=Math.cos(a)*e.spd*dt+rand(-30,30)*dt; e.y+=Math.sin(a)*e.spd*dt+rand(-30,30)*dt;
    } else if(e.type==='drone'){
      let mx=0,my=0;
      if(d>250){mx=dx/d;my=dy/d;} else if(d<160){mx=-dx/d;my=-dy/d;}
      else {mx=-dy/d*e.strafe;my=dx/d*e.strafe;}
      if(e.x<70)mx+=.8; if(e.x>W-70)mx-=.8; if(e.y<70)my+=.8; if(e.y>H-70)my-=.8;
      const ml=Math.hypot(mx,my)||1;
      e.x+=mx/ml*e.spd*dt; e.y+=my/ml*e.spd*dt;
      e.cd-=dt;
      if(e.cd<=0&&d<560&&!world.attract){
        const espd=Math.min(275,165+world.level*9), lead=clamp(d/espd,0,1.2)*.55;
        const tx=sn.x+sn.vx*lead, ty=sn.y+sn.vy*lead, ta=Math.atan2(ty-e.y,tx-e.x);
        world.ebullets.push({x:e.x+Math.cos(ta)*(e.r+4),y:e.y+Math.sin(ta)*(e.r+4),vx:Math.cos(ta)*espd,vy:Math.sin(ta)*espd});
        e.cd=Math.max(1.25,2.3-world.level*.06)*rand(.8,1.2);
      }
    } else if(e.type==='mine'){
      e.rot+=3*dt;
      e.x+=e.vx*dt; e.y+=e.vy*dt;
      if(e.x<e.r+6){e.x=e.r+6;e.vx*=-1;} if(e.x>W-e.r-6){e.x=W-e.r-6;e.vx*=-1;}
      if(e.y<e.r+6){e.y=e.r+6;e.vy*=-1;} if(e.y>H-e.r-6){e.y=H-e.r-6;e.vy*=-1;}
    } else if(e.type==='brute'){
      const a=Math.atan2(dy,dx)+Math.sin(e.t*1.6)*.35;
      e.x+=Math.cos(a)*e.spd*dt; e.y+=Math.sin(a)*e.spd*dt;
    }
  }
}

/* ---------- bullets ---------- */
function updateBullets(dt){
  const sn=world.snake, B=world.bullets;
  for(let i=B.length-1;i>=0;i--){
    const b=B[i]; b.x+=b.vx*dt; b.y+=b.vy*dt;
    if(b.x<-20||b.x>W+20||b.y<-20||b.y>H+20){ B.splice(i,1); continue; }
    let hit=false;
    for(const e of world.enemies){
      if(e.dead) continue;
      const rr=e.r+4;
      if(dist2(b.x,b.y,e.x,e.y)<rr*rr){
        hit=true; e.hp--; e.hitT=.12; spark(b.x,b.y,'#fff',5,180);
        if(e.type==='mine'){ killEnemy(e); }
        else if(e.hp<=0) killEnemy(e);
        break;
      }
    }
    if(hit) B.splice(i,1);
  }
  const E=world.ebullets;
  for(let i=E.length-1;i>=0;i--){
    const b=E[i]; b.x+=b.vx*dt; b.y+=b.vy*dt;
    if(b.x<6||b.x>W-6||b.y<6||b.y>H-6){ spark(b.x,b.y,'#ff8ae0',4,90); E.splice(i,1); continue; }
    const rr=HEAD_R+3;
    if(dist2(b.x,b.y,sn.x,sn.y)<rr*rr){
      if(sn.inv>0){ spark(b.x,b.y,'#6fd7ff',6,140); E.splice(i,1); }
      else { E.splice(i,1); die('plasma'); return; }
    }
  }
}

/* ---------- particles / popups ---------- */
function updateFx(dt){
  const P=world.parts;
  for(let i=P.length-1;i>=0;i--){
    const p=P[i]; p.life-=dt;
    if(p.life<=0){ P.splice(i,1); continue; }
    if(p.type==='ring'){ p.r+=p.vr*dt; continue; }
    if(p.drag){ const k=Math.max(0,1-p.drag*dt); p.vx*=k; p.vy*=k; }
    p.x+=p.vx*dt; p.y+=p.vy*dt;
  }
  const O=world.pops;
  for(let i=O.length-1;i>=0;i--){ O[i].t+=dt; if(O[i].t>1.1) O.splice(i,1); }
}

/* ---------- master update ---------- */
function update(dt){
  world.t+=dt;
  const sdt=dt*world.timeScale;
  world.shake=Math.max(0,world.shake-42*dt);
  world.dmg=Math.max(0,world.dmg-1.6*dt);

  // plankton drift (real time — ambience always alive)
  for(const p of plankton){
    p.x+=p.vx*dt; p.y+=p.vy*dt;
    if(p.x<-10)p.x=W+10; if(p.x>W+10)p.x=-10; if(p.y<-10)p.y=H+10; if(p.y>H+10)p.y=-10;
  }

  if(state==='menu'){
    updateSnake(sdt,false);
    updateFx(sdt);
    if(world.foods.length<4&&Math.random()<dt*.5) spawnFood();
    return;
  }

  if(state==='play'){
    world.playT+=sdt;
    const nl=Math.floor(world.playT/25)+1;
    if(nl>world.level){ world.level=nl; banner('THREAT LEVEL '+nl,'HOSTILES ACCELERATED'); sfx.level(); }
    const c=world.combo;
    if(c.t>0){ c.t-=sdt; if(c.t<=0) c.chain=0; }
    // spawns
    world.spawnT-=sdt;
    const cap=Math.min(3+world.level,12);
    if(world.spawnT<=0&&world.enemies.length<cap){
      spawnEnemy();
      world.spawnT=Math.max(.55,1.9-(world.level-1)*.16)*rand(.7,1.3);
    }
    if(world.foods.length<3) spawnFood();

    updateSnake(sdt,true);
    if(state!=='play'){ /* died this frame */ } else {
      updateEnemies(sdt);
      updateBullets(sdt);
    }
    world.enemies=world.enemies.filter(e=>!e.dead);
  }
  else if(state==='dying'){
    const d=world.death; d.timer+=dt;
    if(d.timer>=d.next&&d.i<d.segs.length){
      const s=d.segs[d.i];
      explode(s.x,s.y,s.big, s.big?'#ffd75e':'#ff9d2e');
      if(d.i%2===0||s.big) (s.big?sfx.boom():sfx.kill());
      d.i++; d.next=d.timer+(d.i<3?.16:.028);
    }
    updateEnemies(sdt); updateBullets(sdt);
    if(d.i>=d.segs.length&&d.timer>1.4) finishDeath();
  }
  else if(state==='over'){
    world.overT+=dt;
    updateEnemies(sdt); updateFx(dt);
  }

  if(state!=='dying'||true) { if(state!=='play') {} }
  if(state==='play'||state==='pause'||state==='over') updateFx(sdt||dt*.25);
  hudSync();
}

const hudCache={};
function setTxt(el,v){ if(hudCache[el.id]!==v){ hudCache[el.id]=v; el.textContent=v; } }
function hudSync(){
  if(!world) return;
  const c=world.combo, mult=c.chain?Math.min(6,1+Math.floor(c.chain/2)):1;
  setTxt(ui.score,String(world.score).padStart(6,'0'));
  setTxt(ui.bestV,String(best).padStart(6,'0'));
  setTxt(ui.lenV,String(world.snake.len));
  setTxt(ui.comboV,'×'+mult);
  setTxt(ui.lvlV,String(world.level));
  ui.fill.style.width=(c.t>0?c.t/COMBO_WIN*100:0)+'%';
  ui.flash.style.opacity=world.dmg.toFixed(2);
}

function render(){
  ctx.clearRect(0,0,W,H);
  // background
  if(!bgGrad){ bgGrad=ctx.createRadialGradient(W/2,H*.45,0,W/2,H*.5,Math.max(W,H)*.75);
    bgGrad.addColorStop(0,'#0c231b'); bgGrad.addColorStop(.6,'#081a13'); bgGrad.addColorStop(1,'#040d0a'); }
  ctx.fillStyle=bgGrad; ctx.fillRect(0,0,W,H);

  const shX=(Math.random()-.5)*world.shake, shY=(Math.random()-.5)*world.shake;
  ctx.save(); ctx.translate(shX,shY);

  drawBackdrop();
  drawBorder();
  for(const f of world.foods) drawFood(f);
  for(const p of world.pickups) drawShield(p);
  for(const e of world.ebullets) drawEBullet(e);
  for(const e of world.enemies) if(!e.dead) drawEnemy(e);
  if(state!=='dying'&&state!=='over') drawSnake();
  for(const b of world.bullets) drawBullet(b);
  drawParts();
  drawPops();
  if(state==='play'&&input.mode==='mouse') drawReticle();

  ctx.restore();
}
function drawBackdrop(){
  // faint grid
  ctx.strokeStyle='rgba(90,220,180,.045)'; ctx.lineWidth=1; ctx.beginPath();
  for(let x=48;x<W;x+=48){ ctx.moveTo(x,0); ctx.lineTo(x,H); }
  for(let y=48;y<H;y+=48){ ctx.moveTo(0,y); ctx.lineTo(W,y); }
  ctx.stroke();
  // sonar pulses from center
  const maxR=Math.hypot(W,H)/2;
  for(let k=0;k<2;k++){
    const ph=((world.t/3.4)+k*.5)%1, r=ph*maxR;
    ctx.strokeStyle='rgba(80,240,200,'+((1-ph)*.06).toFixed(3)+')';
    ctx.beginPath(); ctx.arc(W/2,H/2,r,0,Math.PI*2); ctx.stroke();
  }
  // plankton
  for(const p of plankton){
    const a=.12+.12*Math.sin(world.t*p.tw+p.ph);
    ctx.fillStyle='rgba(80,230,190,'+a.toFixed(3)+')';
    ctx.beginPath(); ctx.arc(p.x,p.y,p.r,0,Math.PI*2); ctx.fill();
  }
  // edge danger glow when nose near wall
  if(state==='play'||state==='dying'){
    const sn=world.snake;
    const edges=[['l',sn.x],['r',W-sn.x],['t',sn.y],['b',H-sn.y]];
    ctx.globalCompositeOperation='lighter';
    for(const [side,d] of edges){
      if(d<95){
        const a=(1-d/95)*.32, g=ctx.createLinearGradient(side==='l'?0:side==='r'?W:0, side==='t'?0:side==='b'?H:0, side==='l'?70:side==='r'?W-70:0, side==='t'?70:side==='b'?H-70:0);
        g.addColorStop(0,'rgba(255,50,30,'+a.toFixed(3)+')'); g.addColorStop(1,'rgba(255,50,30,0)');
        ctx.fillStyle=g;
        if(side==='l')ctx.fillRect(0,0,70,H); else if(side==='r')ctx.fillRect(W-70,0,70,H);
        else if(side==='t')ctx.fillRect(0,0,W,70); else ctx.fillRect(0,H-70,W,70);
      }
    }
    ctx.globalCompositeOperation='source-over';
  }
}
function drawBorder(){
  ctx.strokeStyle='#24503f'; ctx.lineWidth=2; ctx.strokeRect(3,3,W-6,H-6);
  ctx.strokeStyle='rgba(255,190,80,.75)'; ctx.lineWidth=3; const L=18,o=8;
  ctx.beginPath();
  ctx.moveTo(o,L+o);ctx.lineTo(o,o);ctx.lineTo(L+o,o);
  ctx.moveTo(W-o-L,o);ctx.lineTo(W-o,o);ctx.lineTo(W-o,o+L);
  ctx.moveTo(W-o,H-o-L);ctx.lineTo(W-o,H-o);ctx.lineTo(W-o-L,H-o);
  ctx.moveTo(o+L,H-o);ctx.lineTo(o,H-o);ctx.lineTo(o,H-o-L);
  ctx.stroke();
}
function drawFood(f){
  const r=5.5+Math.sin(f.t)*1.6;
  ctx.globalCompositeOperation='lighter';
  ctx.fillStyle='rgba(70,240,200,.16)'; ctx.beginPath(); ctx.arc(f.x,f.y,r*3,0,Math.PI*2); ctx.fill();
  ctx.globalCompositeOperation='source-over';
  const g=ctx.createRadialGradient(f.x-1.5,f.y-1.5,0,f.x,f.y,r);
  g.addColorStop(0,'#eafff6'); g.addColorStop(.5,'#4df0c8'); g.addColorStop(1,'#159b7a');
  ctx.fillStyle=g; ctx.beginPath(); ctx.arc(f.x,f.y,r,0,Math.PI*2); ctx.fill();
}
function drawShield(p){
  const blink=p.life<3?(Math.sin(p.t*8)>0?1:.3):1;
  ctx.save(); ctx.translate(p.x,p.y); ctx.rotate(p.t*.8); ctx.globalAlpha=blink;
  ctx.strokeStyle='#6fd7ff'; ctx.lineWidth=2.5; ctx.setLineDash([5,5]);
  ctx.beginPath(); ctx.arc(0,0,10,0,Math.PI*2); ctx.stroke(); ctx.setLineDash([]);
  ctx.globalCompositeOperation='lighter';
  ctx.fillStyle='rgba(110,215,255,.35)'; ctx.beginPath(); ctx.arc(0,0,4.5,0,Math.PI*2); ctx.fill();
  ctx.restore();
}
function drawEBullet(b){

function drawEnemy(e){
  ctx.save(); ctx.translate(e.x,e.y);
  if(e.type==='mite'){
    const w=1+Math.sin(e.t*8)*.1;
    ctx.rotate(e.t*2);
    ctx.strokeStyle='#a44dff'; ctx.lineWidth=2; ctx.beginPath();
    for(let i=0;i<6;i++){ const a=i/6*Math.PI*2; ctx.moveTo(Math.cos(a)*e.r*.7,Math.sin(a)*e.r*.7); ctx.lineTo(Math.cos(a)*(e.r+5)*w,Math.sin(a)*(e.r+5)*w); }
    ctx.stroke();
    const g=ctx.createRadialGradient(-2,-2,0,0,0,e.r);
    g.addColorStop(0,'#f0c8ff'); g.addColorStop(.5,'#c95cff'); g.addColorStop(1,'#6a1fa8');
    ctx.fillStyle=g; ctx.beginPath(); ctx.arc(0,0,e.r*w,0,Math.PI*2); ctx.fill();
  } else if(e.type==='drone'){
    const charge=e.cd<.4?clamp(1-e.cd/.4,0,1):0;
    ctx.rotate(Math.sin(e.t*1.5)*.15);
    ctx.beginPath();
    for(let i=0;i<6;i++){ const a=i/6*Math.PI*2; i?ctx.lineTo(Math.cos(a)*e.r,Math.sin(a)*e.r):ctx.moveTo(Math.cos(a)*e.r,Math.sin(a)*e.r); }
    ctx.closePath();
    ctx.globalCompositeOperation='lighter';
    ctx.strokeStyle='rgba(255,46,154,'+(.2+.3*charge).toFixed(2)+')'; ctx.lineWidth=7; ctx.stroke();
    ctx.globalCompositeOperation='source-over';
    ctx.fillStyle='#1c0f2e'; ctx.fill();
    ctx.strokeStyle='#ff2e9a'; ctx.lineWidth=2; ctx.stroke();
    ctx.fillStyle=charge>0?'#fff':'#ff8ae0';
    ctx.beginPath(); ctx.arc(0,0,3+charge*3.5,0,Math.PI*2); ctx.fill();
  } else if(e.type==='mine'){
    ctx.rotate(e.rot);
    ctx.strokeStyle='#c95cff'; ctx.lineWidth=2.5; ctx.beginPath();
    for(let i=0;i<4;i++){ const a=i/4*Math.PI*2; ctx.moveTo(Math.cos(a)*e.r*.6,Math.sin(a)*e.r*.6); ctx.lineTo(Math.cos(a)*(e.r+7),Math.sin(a)*(e.r+7)); }
    ctx.stroke();
    ctx.fillStyle='#1a0f22'; ctx.beginPath(); ctx.arc(0,0,e.r,0,Math.PI*2); ctx.fill();
    ctx.strokeStyle='#c95cff'; ctx.lineWidth=2; ctx.stroke();
    ctx.globalCompositeOperation='lighter';
    const pr=3+Math.sin(e.t*10)*1.6;
    ctx.fillStyle='rgba(255,240,255,.9)'; ctx.beginPath(); ctx.arc(0,0,pr,0,Math.PI*2); ctx.fill();
  } else { // brute — armor arcs = remaining hp
    const g=ctx.createRadialGradient(-4,-4,0,0,0,e.r);
    g.addColorStop(0,'#7a1f4d'); g.addColorStop(.6,'#3a0e2c'); g.addColorStop(1,'#1c0716');
    ctx.fillStyle=g; ctx.beginPath(); ctx.arc(0,0,e.r,0,Math.PI*2); ctx.fill();
    ctx.strokeStyle='#ff2e9a'; ctx.lineWidth=2.5; ctx.stroke();
    for(let i=0;i<5;i++){
      const a0=i/5*Math.PI*2+e.t*.3, a1=a0+Math.PI*2/5*.72;
      ctx.strokeStyle=i<e.hp?'#ff6ab0':'rgba(255,106,176,.18)';
      ctx.lineWidth=4; ctx.beginPath(); ctx.arc(0,0,e.r-5,a0,a1); ctx.stroke();
    }
    ctx.globalCompositeOperation='lighter';
    ctx.fillStyle='rgba(255,80,120,'+(.25+.15*Math.sin(e.t*6)).toFixed(2)+')';
    ctx.beginPath(); ctx.arc(0,0,e.r*.45,0,Math.PI*2); ctx.fill();
  }
  if(e.hitT>0){ ctx.globalCompositeOperation='lighter'; ctx.fillStyle='rgba(255,255,255,'+(e.hitT*6).toFixed(2)+')';
    ctx.beginPath(); ctx.arc(0,0,e.r+2,0,Math.PI*2); ctx.fill(); }
  ctx.restore();
}
function drawSnake(){
  const sn=world.snake, n=sn.segs.length; if(!n) return;
  const P=[{x:sn.x,y:sn.y}].concat(sn.segs);
  let alpha=1;
  if(sn.inv>0&&!world.attract) alpha=.62+.38*Math.sin(world.t*40);
  ctx.save(); ctx.globalAlpha=alpha; ctx.lineCap='round';

  // outer glow (single pass, additive)
  ctx.globalCompositeOperation='lighter';
  ctx.strokeStyle='rgba(255,150,50,.10)'; ctx.lineWidth=HEAD_R*2+11;
  ctx.beginPath(); ctx.moveTo(P[0].x,P[0].y); for(let i=1;i<P.length;i++) ctx.lineTo(P[i].x,P[i].y); ctx.stroke();
  ctx.globalCompositeOperation='source-over';

  // body: chunked gradient strokes, tail → neck
  const N=P.length;
  for(let i=N-1;i>=1;i--){
    const t=i/N;
    ctx.strokeStyle=bodyCol(t);
    ctx.lineWidth=7+(HEAD_R*1.9-7)*(1-t);
    ctx.beginPath(); ctx.moveTo(P[i].x,P[i].y); ctx.lineTo(P[i-1].x,P[i-1].y); ctx.stroke();
  }
  // sheen line
  ctx.strokeStyle='rgba(255,245,210,.16)'; ctx.lineWidth=3;
  ctx.beginPath(); ctx.moveTo(P[0].x,P[0].y); for(let i=1;i<N;i+=2) ctx.lineTo(P[i].x,P[i].y); ctx.stroke();

  // head (with recoil kick)
  const hx=sn.x-Math.cos(sn.angle)*sn.recoil, hy=sn.y-Math.sin(sn.angle)*sn.recoil;
  if(sn.inv>0&&!world.attract){
    ctx.globalCompositeOperation='lighter';
    ctx.fillStyle='rgba(90,255,220,'+(.1+.06*Math.sin(world.t*12)).toFixed(3)+')';
    ctx.beginPath(); ctx.arc(hx,hy,HEAD_R+11,0,Math.PI*2); ctx.fill();
    ctx.strokeStyle='rgba(120,255,230,.8)'; ctx.lineWidth=2;
    ctx.setLineDash([6,6]); ctx.lineDashOffset=-world.t*40;
    ctx.beginPath(); ctx.arc(hx,hy,HEAD_R+7,0,Math.PI*2); ctx.stroke();
    ctx.setLineDash([]); ctx.globalCompositeOperation='source-over';
  }
  const g=ctx.createRadialGradient(hx-3,hy-3,1,hx,hy,HEAD_R+2);
  g.addColorStop(0,'#fff3c0'); g.addColorStop(.55,'#ffb347'); g.addColorStop(1,'#d95f1e');
  ctx.fillStyle=g; ctx.beginPath(); ctx.arc(hx,hy,HEAD_R+1,0,Math.PI*2); ctx.fill();
  // tongue flick
  if(sn.tongueT<.18){
    const p=1-sn.tongueT/.18, L=9*Math.sin(p*Math.PI);
    const sx=hx+Math.cos(sn.angle)*HEAD_R, sy=hy+Math.sin(sn.angle)*HEAD_R;
    const tip=sx+Math.cos(sn.angle)*L, tipy=sy+Math.sin(sn.angle)*L;
    ctx.strokeStyle='#ff3b5c'; ctx.lineWidth=1.6; ctx.beginPath();
    ctx.moveTo(sx,sy); ctx.lineTo(tip,tipy);
    ctx.moveTo(tip,tipy); ctx.lineTo(tip+Math.cos(sn.angle-.4)*4,tipy+Math.sin(sn.angle-.4)*4);
    ctx.moveTo(tip,tipy); ctx.lineTo(tip+Math.cos(sn.angle+.4)*4,tipy+Math.sin(sn.angle+.4)*4);
    ctx.stroke();
  }
  // eyes
  const pa=sn.angle+Math.PI/2;
  for(const s of [-1,1]){
    const ex=hx+Math.cos(sn.angle)*3.5+Math.cos(pa)*4.6*s, ey=hy+Math.sin(sn.angle)*3.5+Math.sin(pa)*4.6*s;
    ctx.fillStyle='#20100a'; ctx.beginPath(); ctx.arc(ex,ey,2.7,0,Math.PI*2); ctx.fill();
    ctx.fillStyle='rgba(255,255,255,.9)'; ctx.beginPath(); ctx.arc(ex-.8,ey-.8,.9,0,Math.PI*2); ctx.fill();
  }
  ctx.restore();
}
function drawBullet(b){
  const a=Math.atan2(b.vy,b.vx);
  ctx.globalCompositeOperation='lighter';
  ctx.strokeStyle='rgba(255,180,60,.35)'; ctx.lineWidth=7;
  ctx.beginPath(); ctx.moveTo(b.x-Math.cos(a)*8,b.y-Math.sin(a)*8); ctx.lineTo(b.x+Math.cos(a)*4,b.y+Math.sin(a)*4); ctx.stroke();
  ctx.strokeStyle='rgba(255,225,130,.95)'; ctx.lineWidth=3;
  ctx.beginPath(); ctx.moveTo(b.x-Math.cos(a)*7,b.y-Math.sin(a)*7); ctx.lineTo(b.x+Math.cos(a)*4,b.y+Math.sin(a)*4); ctx.stroke();
  ctx.strokeStyle='#fff8e0'; ctx.lineWidth=1.4;
  ctx.beginPath(); ctx.moveTo(b.x-Math.cos(a)*5,b.y-Math.sin(a)*5); ctx.lineTo(b.x+Math.cos(a)*3,b.y+Math.sin(a)*3); ctx.stroke();
  ctx.globalCompositeOperation='source-over';
}
function drawParts(){
  ctx.globalCompositeOperation='lighter';
  for(const p of world.parts){
    const a=clamp(p.life/p.max,0,1);
    if(p.type==='ring'){
      ctx.strokeStyle=p.col; ctx.globalAlpha=a*.7; ctx.lineWidth=2.5;
      ctx.beginPath(); ctx.arc(p.x,p.y,p.r,0,Math.PI*2); ctx.stroke();
    } else {
      ctx.fillStyle=p.col; ctx.globalAlpha=a;
      ctx.beginPath(); ctx.arc(p.x,p.y,p.size*a+.4,0,Math.PI*2); ctx.fill();
    }
  }
  ctx.globalAlpha=1; ctx.globalCompositeOperation='source-over';
}
function drawPops(){

    ctx.fillStyle=p.col; ctx.fillText(p.txt,p.x,y);
  }
  ctx.globalAlpha=1; ctx.textAlign='left';
}
function drawReticle(){

  ctx.fillStyle='rgba(140,255,220,.9)'; ctx.beginPath(); ctx.arc(0,0,1.6,0,Math.PI*2); ctx.fill();
  ctx.restore();
}
