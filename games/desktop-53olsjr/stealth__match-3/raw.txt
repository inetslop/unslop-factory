    else{th.vx=dx/d*SPD;th.vy=dy/d*SPD;th.moving=true;}
  }else{th.vx=th.vy=0;th.moving=false;}
  if(th.vx){th.x+=th.vx*dt;collideAxis('x',Math.sign(th.vx));}
  if(th.vy){th.y+=th.vy*dt;collideAxis('y',Math.sign(th.vy));}
  th.x=clamp(th.x,0.31,COLS-0.31);th.y=clamp(th.y,0.31,ROWS-0.31);
  if(th.moving){th.stepT+=dt*Math.hypot(th.vx,th.vy);th.facing=lerpAng(th.facing,Math.atan2(th.vy,th.vx),1-Math.exp(-14*dt));}
  // door
  const dcx=DOOR.c+0.5,dcy=DOOR.r+0.5,d=Math.hypot(th.x-dcx,th.y-dcy);
  if(d<0.6){
    if(S.doorOpen)win();
    else if(S.t-S.doorToastT>2.2){S.doorToastT=S.t;toast('THE LOCK NEEDS '+(GOAL-S.loot)+' MORE GEMS');sDeny();}
  }
}

/* ================= guards ================= */
function sightRange(){return 3.1+0.15*(S.shift-1);}
function chaseSpeed(){return 2.35*(1+0.1*(S.shift-1));}
function hasLOS(x1,y1,x2,y2){
  const dx=x2-x1,dy=y2-y1,len=Math.hypot(dx,dy),step=0.12;
  for(let d=step;d<len;d+=step){const px=x1+dx*d/len,py=y1+dy*d/len;if(isWall(Math.floor(px),Math.floor(py)))return false;}
  return true;
}
function canSeeRaw(g,x,y){
  const dx=x-g.x,dy=y-g.y,d=Math.hypot(dx,dy);
  if(d<0.5)return true;
  if(d>sightRange())return false;
  if(Math.abs(angDiff(Math.atan2(dy,dx),g.facing))>0.98)return false;
  return hasLOS(g.x,g.y,x,y);
}
function aStar(sr,sc,tr,tc){
  if(sr===tr&&sc===tc)return[];
  const N=COLS*ROWS,idx=(r,c)=>r*COLS+c;
  const gs=new Float32Array(N).fill(Infinity),fs=new Float32Array(N).fill(Infinity),came=new Int16Array(N).fill(-1);
  const open=[idx(sr,sc)];gs[open[0]]=0;fs[open[0]]=Math.abs(tr-sr)+Math.abs(tc-sc);
  const D=[[1,0],[-1,0],[0,1],[0,-1]];
  while(open.length){
    let bi=0;for(let i=1;i<open.length;i++)if(fs[open[i]]<fs[open[bi]])bi=i;
    const cur=open.splice(bi,1)[0],cr=(cur/COLS)|0,cc=cur%COLS;
    if(cr===tr&&cc===tc){const path=[];let n=cur;while(n!==idx(sr,sc)){path.push([n%COLS,(n/COLS)|0]);n=came[n];}return path.reverse();}
    for(const[dr,dc]of D){const nr=cr+dr,nc=cc+dc;if(nr<0||nc<0||nr>=ROWS||nc>=COLS||isWall(nc,nr))continue;
      const ni=idx(nr,nc),ng=gs[cur]+1;
      if(ng<gs[ni]){gs[ni]=ng;fs[ni]=ng+Math.abs(tr-nr)+Math.abs(tc-nc);came[ni]=cur;if(!open.includes(ni))open.push(ni);}}
  }
  return[];
}
function startChase(g){
  if(g.state==='chase')return;
  g.state='chase';g.path=[];g.repath=0;g.lastSeen=S.t;
  sAlarm();addFloat(g.x,g.y-0.5,'!','#ff4d5e',30,0.9);S.shake=Math.max(S.shake,4);
}
function busted(g){
  S.mode='busted';S.caughtBy=g.name;S.endT=0;S.endShown=false;
  sBust();S.shake=15;S.redFlash=1;fillEnd(false);
}
function updGuards(dt,full){
  const th=S.thief;
  for(const g of S.guards){
    if(g.state==='patrol'){
      const sp=g.speed*(1+0.12*(S.shift-1));
      const wp=g.pts[g.wpIdx],tx=wp[0]+0.5,ty=wp[1]+0.5;
      const dx=tx-g.x,dy=ty-g.y,d=Math.hypot(dx,dy);
      if(d>0.02){const mv=Math.min(d,sp*dt);g.x+=dx/d*mv;g.y+=dy/d*mv;g.facing=lerpAng(g.facing,Math.atan2(dy,dx),1-Math.exp(-9*dt));}
      if(d<0.06){g.wpIdx+=g.dir;if(g.wpIdx<0){g.dir=1;g.wpIdx=0;}else if(g.wpIdx>=g.pts.length){g.dir=-1;g.wpIdx=g.pts.length-2;}}
    }else if(g.state==='distraught'){
      g.distractT-=dt;
      g.facing=lerpAng(g.facing,Math.atan2(g.noisePt.y-g.y,g.noisePt.x-g.x),1-Math.exp(-8*dt));
      if(g.distractT<=0)g.state='patrol';
    }else if(g.state==='chase'){
      const d=Math.hypot(th.x-g.x,th.y-g.y);
      const sees=hasLOS(g.x,g.y,th.x,th.y)&&d<sightRange()+1.2;
      if(sees)g.lastSeen=S.t;
      g.repath-=dt;
      if(g.repath<=0||!g.path.length){
        g.path=aStar(Math.floor(g.x),Math.floor(g.y),Math.floor(th.x),Math.floor(th.y));g.repath=0.35;
      }
      const cs=chaseSpeed()*dt;let tx=null,ty=null;
      if(g.path.length){tx=g.path[0][0]+0.5;ty=g.path[0][1]+0.5;}else{tx=th.x;ty=th.y;}
      const dx=tx-g.x,dy=ty-g.y,d2=Math.hypot(dx,dy);
      if(d2>0.001){const mv=Math.min(d2,cs);g.x+=dx/d2*mv;g.y+=dy/d2*mv;g.facing=lerpAng(g.facing,Math.atan2(dy,dx),1-Math.exp(-12*dt));}
      if(g.path.length&&d2<=cs+0.01)g.path.shift();
      if(d<0.72){busted(g);return;}
      if(S.t-g.lastSeen>4&&!sees){
        g.state='patrol';g.path=[];let bi=0,bd=1e9;
        for(let i=0;i<g.pts.length;i++){const p=g.pts[i],b=Math.hypot(p[0]+0.5-g.x,p[1]+0.5-g.y);if(b<bd){bd=b;bi=i;}}
        g.wpIdx=bi;g.dir=1;addFloat(g.x,g.y-0.5,'lost you','#8fa0bf',13,1.2);
      }
    }
  }
  if(full)for(const g of S.guards){if(g.state!=='chase'&&canSeeRaw(g,th.x,th.y))startChase(g);}
}

function gemPath(g,t,s){
  g.beginPath();
  if(t===0){g.moveTo(0,-s);g.lineTo(s*.85,0);g.lineTo(0,s);g.lineTo(-s*.85,0);g.closePath();}
  else if(t===1){for(let i=0;i<6;i++){const a=-Math.PI/2+i*Math.PI/3,x=Math.cos(a)*s*.95,y=Math.sin(a)*s*.95;i?g.lineTo(x,y):g.moveTo(x,y);}g.closePath();}
  else if(t===2){const w=s*1.02,h=s*1.18,k=s*.42;g.moveTo(-w+k,-h);g.lineTo(w-k,-h);g.lineTo(w,-h+k);g.lineTo(w,h-k);g.lineTo(w-k,h);g.lineTo(-w+k,h);g.lineTo(-w,h-k);g.lineTo(-w,-h+k);g.closePath();}
  else if(t===3){g.arc(0,0,s*.82,0,TAU);}
  else{for(let i=0;i<10;i++){const a=-Math.PI/2+i*Math.PI/5,rr=i%2?s*.44:s*1.0,x=Math.cos(a)*rr,y=Math.sin(a)*rr;i?g.lineTo(x,y):g.moveTo(x,y);}g.closePath();}
}

  updGuards(dt,S.mode==='play');
  if(S.mode==='play'){
    if(S.phase==='swapping')updateSwap(dt);
    else if(S.phase==='clearing'){
      let done=true;
      for(const o of S.clearSet){o.gem.pop+=dt/0.26;if(o.gem.pop<1)done=false;}
      if(done){for(const o of S.clearSet)S.grid[o.r][o.c]=null;startFall();}
    }
    else if(S.phase==='falling')updateFall(dt);
  }
  S.hot=S.guards.some(g=>g.state==='chase');
  if(S.mode==='play')updHUD();
  // glints
  if(Math.random()<dt*1.4){
    const g=S.grid;let tries=0,cell2=null;
    while(tries++<12){const r=(Math.random()*ROWS)|0,c=(Math.random()*COLS)|0;if(g[r]&&g[r][c]){cell2={x:g[r][c].x,y:g[r][c].y};break;}}
    if(cell2)S.glints.push({x:cell2.x,y:cell2.y,t:0});
  }
}
function tickEffects(dt){
  S.t+=dt;
  for(let i=S.rings.length-1;i>=0;i--){S.rings[i].t+=dt;if(S.rings[i].t>S.rings[i].life)S.rings.splice(i,1);}
  for(let i=S.floats.length-1;i>=0;i--){S.floats[i].t+=dt;if(S.floats[i].t>S.floats[i].dur)S.floats.splice(i,1);}
  for(let i=S.glints.length-1;i>=0;i--){S.glints[i].t+=dt/0.45;if(S.glints[i].t>1)S.glints.splice(i,1);}
  S.shake*=Math.exp(-6*dt);if(S.shake<0.05)S.shake=0;
  S.redFlash=Math.max(0,S.redFlash-dt*1.1);S.goldFlash=Math.max(0,S.goldFlash-dt*0.9);
  for(const d of dust){d.y-=d.s*dt;d.x+=Math.sin(S.t*0.8+d.ph)*7*dt;if(d.y<-4){d.y=H+4;d.x=rand(W);}if(d.x<-4)d.x=W+4;if(d.x>W+4)d.x=-4;}
  if((S.mode==='busted'||S.mode==='won')&&!S.endShown){
    S.endT+=dt;const lim=S.mode==='won'?1.0:0.7;
    if(S.endT>lim){S.endShown=true;showScreen('end');if(S.mode==='won')showBest();}
  }
}

function beam(g,ox,oy,ang,len,a){
  const x2=ox+Math.cos(ang)*len,y2=oy+Math.sin(ang)*len;
  const gr=g.createLinearGradient(ox,oy,x2,y2);gr.addColorStop(0,'rgba(190,215,255,'+a+')');gr.addColorStop(1,'rgba(190,215,255,0)');
  g.fillStyle=gr;g.beginPath();g.moveTo(ox,oy);
  g.lineTo(ox+Math.cos(ang-0.16)*len,oy+Math.sin(ang-0.16)*len);
  g.lineTo(ox+Math.cos(ang+0.16)*len,oy+Math.sin(ang+0.16)*len);g.closePath();g.fill();
}
function render(){
  ctx.setTransform(dpr,0,0,dpr,0,0);
  // bg
  const bg=ctx.createLinearGradient(0,0,0,H);bg.addColorStop(0,'#0b1120');bg.addColorStop(.6,'#0a0f1c');bg.addColorStop(1,'#080d17');
  ctx.fillStyle=bg;ctx.fillRect(0,0,W,H);
  beam(ctx,-40,-30,Math.PI/2.6+Math.sin(S.t*0.13)*0.5,Math.max(W,H),0.07);
  beam(ctx,W+40,-30,Math.PI*0.8-Math.sin(S.t*0.11+2)*0.5,Math.max(W,H),0.06);
  ctx.fillStyle='rgba(200,220,255,.1)';
  for(const d of dust){ctx.beginPath();ctx.arc(d.x,d.y,d.r,0,TAU);ctx.fill();}
  // board (shake)
  const shx=(Math.random()*2-1)*S.shake,shy=(Math.random()*2-1)*S.shake;
  ctx.save();ctx.translate(shx,shy);
  if(floorCv)ctx.drawImage(floorCv,bx,by,boardW,boardH);
  drawDoor();
  drawReach();
  drawGems();
  drawCones();
  drawGuards();
  if(S.mode!=='menu')drawThief();else drawThief(); // idle figure on start screen too
  drawRings();
  drawFloats();
  ctx.restore();
  // over-beam pass
  ctx.save();ctx.globalCompositeOperation='screen';
  beam(ctx,-40,-30,Math.PI/2.6+Math.sin(S.t*0.13)*0.5,Math.max(W,H),0.028);
  beam(ctx,W+40,-30,Math.PI*0.8-Math.sin(S.t*0.11+2)*0.5,Math.max(W,H),0.024);
  ctx.restore();
  if(vigCv)ctx.drawImage(vigCv,0,0,W,H);
  if(S.hot){ctx.save();ctx.globalAlpha=0.16+0.08*Math.sin(S.t*7);if(redVigCv)ctx.drawImage(redVigCv,0,0,W,H);ctx.restore();}
  if(S.redFlash>0){ctx.fillStyle='rgba(255,40,60,'+(S.redFlash*0.3)+')';ctx.fillRect(0,0,W,H);}
  if(S.goldFlash>0){ctx.fillStyle='rgba(255,215,94,'+(S.goldFlash*0.22)+')';ctx.fillRect(0,0,W,H);}
}
function drawDoor(){
  const dx=bx+(DOOR.c+.5)*cell,dy=by+(DOOR.r+.5)*cell;
  if(S.doorOpen){
    const a=0.18+0.1*Math.sin(S.t*4);
    const gr=ctx.createRadialGradient(dx,dy,2,dx,dy,cell*0.9);
    gr.addColorStop(0,'rgba(255,215,94,'+a+')');gr.addColorStop(1,'rgba(255,215,94,0)');
    ctx.fillStyle=gr;ctx.beginPath();ctx.arc(dx,dy,cell*0.9,0,TAU);ctx.fill();
  }
  const rot=S.doorOpen?S.t*0.9:S.t*0.12;
  ctx.save();ctx.translate(dx,dy);ctx.rotate(rot);
  ctx.strokeStyle=S.doorOpen?'#ffd75e':'#5b6c94';ctx.lineWidth=Math.max(2,cell*0.045);
  for(let i=0;i<3;i++){ctx.beginPath();ctx.moveTo(-Math.cos(i*Math.PI/3)*cell*0.3, -Math.sin(i*Math.PI/3)*cell*0.3);
    ctx.lineTo(Math.cos(i*Math.PI/3)*cell*0.3,Math.sin(i*Math.PI/3)*cell*0.3);ctx.stroke();}
  ctx.restore();

  ctx.fillStyle=S.doorOpen?'rgba(255,215,94,'+la+')':'rgba(255,77,94,'+la+')';ctx.fill();
}
function drawReach(){
  const th=S.thief,tx=bx+th.x*cell,ty=by+th.y*cell;
  const a=(S.mode==='play'&&S.phase==='idle')?0.2:0.08;
  ctx.save();
  ctx.strokeStyle='rgba(255,215,94,'+a+')';ctx.lineWidth=1.5;
  ctx.setLineDash([6,7]);ctx.lineDashOffset=-S.t*22;
  ctx.beginPath();ctx.arc(tx,ty,1.42*cell,0,TAU);ctx.stroke();ctx.setLineDash([]);
  if(S.mode==='play'&&S.phase==='idle'){
    const p=thCell(),pulse=0.35+0.2*Math.sin(S.t*5),L=cell*0.16;
    ctx.strokeStyle='rgba(255,215,94,'+pulse+')';ctx.lineWidth=1.5;
    for(let r=p.r-1;r<=p.r+1;r++)for(let c=p.c-1;c<=p.c+1;c++){
      if(c<0||r<0||c>=COLS||r>=ROWS||noGem(c,r)||!S.grid[r][c])continue;
      const x=bx+c*cell,y=by+r*cell;
      ctx.beginPath();
      ctx.moveTo(x+2,y+2+L);ctx.lineTo(x+2,y+2);ctx.lineTo(x+2+L,y+2);
      ctx.moveTo(x+cell-2-L,y+2);ctx.lineTo(x+cell-2,y+2);ctx.lineTo(x+cell-2,y+2+L);
      ctx.moveTo(x+2,y+cell-2-L);ctx.lineTo(x+2,y+cell-2);ctx.lineTo(x+2+L,y+cell-2);
      ctx.moveTo(x+cell-2-L,y+cell-2);ctx.lineTo(x+cell-2,y+cell-2);ctx.lineTo(x+cell-2,y+cell-2-L);
      ctx.stroke();
    }
  }
  ctx.restore();
}
function drawGems(){
  const ds=cell*0.88;
  for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++){
    const gem=S.grid[r][c];if(!gem)continue;
    let sc=1,al=1,oy=0;
    if(gem.pop>=0){sc=1+0.45*Math.sin(Math.min(1,gem.pop)*Math.PI);al=1-Math.max(0,gem.pop-0.5)*2;}
    const isSel=S.sel&&S.sel.r===r&&S.sel.c===c;
    if(isSel){sc*=1.08;oy=-3;}
    const sx=bx+gem.x*cell,sy=by+gem.y*cell+oy;
    ctx.save();ctx.globalAlpha=al;
    if(isSel){ctx.shadowColor='rgba(255,255,255,.9)';ctx.shadowBlur=14;}
    const w=ds*sc;ctx.drawImage(gemSpr[gem.t],sx-w/2,sy-w/2,w,w);
    ctx.restore();
  }
  for(const gl of S.glints){
    const sx=bx+gl.x*cell,sy=by+gl.y*cell,a=Math.sin(gl.t*Math.PI)*0.75;
    ctx.save();ctx.translate(sx,sy);ctx.rotate(-0.6);
    ctx.fillStyle='rgba(255,255,255,'+a+')';
    const off=(gl.t-0.5)*cell*1.4;
    ctx.fillRect(off-cell*0.3,-cell*0.03,cell*0.16,cell*0.6);
    ctx.restore();
  }
}

}
