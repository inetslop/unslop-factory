    b.sp=Math.min(1040*SC,b.sp*1.015);}
  if(Math.random()<(br.tier===4?1:.16))caps.push({x:br.x+br.w/2,y:br.y+br.h/2,type:pickPup(),seed:rand(TAU)});}

function applyPup(c){const P=PUPDEF[c.type],px=paddle.x+paddle.w,py=clamp(paddle.cy,90,H-60);
  const say=(msg)=>addText(px+30*SC,py-26*SC,msg,P.c,Math.round(15*SC));
  switch(c.type){
    case 'multi':{let added=0;
      for(const b of balls){if(b.held||balls.length+added>=6)break;
        const a=(Math.random()<.5?-1:1)*rand(.5,.9),co=Math.cos(a),si=Math.sin(a);
        balls.push({x:b.x,y:b.y,vx:b.vx*co-b.vy*si,vy:b.vx*si+b.vy*co,r:b.r,sp:b.sp,trail:[],held:false});added++;}
      if(added){say('MULTIBALL!');SFX.good();}else{score+=250;say('MAX BALLS +250');SFX.good();}break;}
    case 'grow':effects.grow=9;effects.shrink=0;say('GROW PADDLE');SFX.good();break;
    case 'slow':effects.slow=8;say('SLOW DRIFT');SFX.good();break;
    case 'repair':if(integrity<5){integrity++;setPips();say('REPAIR +1');}else{score+=150;say('FULL SHIELD +150');}SFX.good();break;
    case 'pierce':effects.pierce=6;say('PIERCE!');SFX.good();break;
    case 'shrink':effects.shrink=7;effects.grow=0;say('SHRINK!');SFX.bad();break;}
  addBurst(px,py,P.c,8,180*SC);}

function updatePaddle(dt,attract){
  paddle.h=paddleH();
  if(attract){const b=balls.find(v=>!v.held);
    if(b){const t=b.y+Math.sin(time*2.1)*30,d=t-paddle.cy;
      paddle.cy+=clamp(d,-540*SC*dt,540*SC*dt);}}
  else{
    const vy=(keys.d?1:0)-(keys.u?1:0),vx=(keys.r?1:0)-(keys.l?1:0);
    paddle.cy+=vy*640*SC*dt;paddle.x+=vx*420*SC*dt;
    if(pointer.active){const dx=pointer.x-paddle.w/2-paddle.x,dy=pointer.y-paddle.cy;
      paddle.x+=clamp(dx,-1500*SC*dt,1500*SC*dt);paddle.cy+=clamp(dy,-1500*SC*dt,1500*SC*dt);}}
  paddle.x=clamp(paddle.x,16,Math.max(20,W*.24-paddle.w));
  paddle.cy=clamp(paddle.cy,paddle.h/2+56,H-paddle.h/2-8);
  paddle.y=paddle.cy-paddle.h/2;
  paddleFlash=Math.max(0,paddleFlash-dt*3);}

/* ================= balls ================= */
function updateBalls(dt,attract){
  for(const b of balls)if(b.held){b.x=paddle.x+paddle.w+3;b.y=clamp(paddle.cy,b.r,H-b.r);}
  respawnT-=dt;
  if(balls.length===0&&respawnT<=0){spawnHeld();respawnT=attract?1.8:2.4;}
  const pierce=effects.pierce>0&&!attract;
  for(let i=balls.length-1;i>=0;i--){const b=balls[i];if(b.held)continue;
    const steps=clamp(Math.ceil(Math.hypot(b.vx,b.vy)*dt/(b.r*.9)),1,5);
    let lost=false;
    for(let s=0;s<steps&&!lost;s++){
      b.x+=b.vx*dt/steps;b.y+=b.vy*dt/steps;
      if(b.y<b.r){b.y=b.r;b.vy=Math.abs(b.vy);}
      else if(b.y>H-b.r){b.y=H-b.r;b.vy=-Math.abs(b.vy);}
      if(b.x>W-b.r){b.x=W-b.r;b.vx=-Math.abs(b.vx);}
      if(b.vx<0&&b.x-b.r<=paddle.x+paddle.w&&b.x>paddle.x-14&&
         b.y>paddle.y-b.r&&b.y<paddle.y+paddle.h+b.r){
        b.x=paddle.x+paddle.w+b.r;
        const off=clamp((b.y-paddle.cy)/(paddle.h/2),-1,1),th=off*1.02;
        b.vx=Math.cos(th)*b.sp;b.vy=Math.sin(th)*b.sp;
        if(!attract){combo=0;}paddleFlash=.35;SFX.paddle();
        addBurst(b.x-b.r,b.y,'#9fe8ff',4,160*SC);}
      for(const br of bricks){if(br.dead)continue;
        const cx=clamp(b.x,br.x,br.x+br.w),cy=clamp(b.y,br.y,br.y+br.h);
        const dx=b.x-cx,dy=b.y-cy;
        if(dx*dx+dy*dy>b.r*b.r)continue;
        br.hp--;br.flash=.25;addBurst(cx,cy,TIER[br.tier].f,3,140*SC);
        if(!pierce){
          if(Math.abs(dx)>Math.abs(dy))b.vx=dx>=0?Math.abs(b.vx):-Math.abs(b.vx);
          else b.vy=dy>=0?Math.abs(b.vy):-Math.abs(b.vy);}
        if(br.hp<=0)destroyBrick(br,b,attract);
        else if(!attract)SFX.tick(br.tier);
        if(!pierce)break;}
      if(b.x<-b.r*2-4){lost=true;}}
    b.trail.push({x:b.x,y:b.y});if(b.trail.length>12)b.trail.shift();
    const cur=Math.hypot(b.vx,b.vy)||1e-6,k=b.sp/cur;b.vx*=k;b.vy*=k;
    if(Math.abs(b.vx)<b.sp*.18){const s=b.vx<0?-1:1;b.vx=s*b.sp*.3;
      b.vy=(b.vy>=0?1:-1)*Math.sqrt(Math.max(0,b.sp*b.sp-b.vx*b.vx));}
    if(lost){balls.splice(i,1);
      if(!attract){respawnT=Math.max(respawnT,.9);damage(1,'BALL LOST',26,b.y,SFX.lost);}}}}

/* ================= update ================= */
function update(dt){
  if(state===ST.PAUSE)return;
  for(const s of stars){s.x-=(s.l===1?14:42)*SC*dt;if(s.x<-5){s.x=W+5;s.y=rand(H);}}
  updateParts(dt);updateTexts(dt);
  comboPop=Math.max(0,comboPop-dt*3);shake=Math.max(0,shake-dt*2.2);flash=Math.max(0,flash-dt*1.8);

  const attract=state===ST.MENU;
  for(const k in effects)if(effects[k]>0)effects[k]=Math.max(0,effects[k]-dt);
  updatePaddle(dt,attract);

  const wp=waveParams();
  spawnT-=dt;
  if(spawnT<=0){spawnFormation(wp);spawnT=wp.interval*rand(.75,1.25);}

  const driftBase=attract?30:Math.min(120,26+(wave-1)*5.5);
  const drift=driftBase*SC*((effects.slow>0&&!attract)?.45:1);
  for(const b of bricks){b.x-=drift*dt;if(b.flash>0)b.flash=Math.max(0,b.flash-dt*4);}
  for(let i=bricks.length-1;i>=0;i--){const b=bricks[i];
    if(b.x+b.w<-4){bricks.splice(i,1);if(!attract)damage(1,'BREACH -1',30,b.y+b.h/2,SFX.breach);}}

  const capSp=(78+wave*2)*SC;
  for(let i=caps.length-1;i>=0;i--){const c=caps[i];c.x-=capSp*dt;
    if(c.x<-40){caps.splice(i,1);continue;}
    if(!attract&&c.x>paddle.x-16*SC&&c.x<paddle.x+paddle.w+16*SC&&
       Math.abs(c.y-paddle.cy)<paddle.h/2+14*SC){applyPup(c);caps.splice(i,1);}}

  updateBalls(dt,attract);
  bricks=bricks.filter(b=>!b.dead);

  if(!attract){waveT+=dt;
    if(waveT>26){waveT=0;wave++;score+=150*wave;
      addText(W/2,H*.3,'WAVE BONUS +'+(150*wave),'#ffd166',Math.round(18*SC));
      showBanner('WAVE '+wave,'DRIFT ACCELERATES');SFX.wave();updateHUD(true);}}
  updateHUD(false);renderChips();}

function drawBrick(b){const c=TIER[b.tier];
  ctx.fillStyle=c.f;rr(b.x,b.y,b.w,b.h,4);ctx.fill();
  ctx.fillStyle='rgba(255,255,255,.3)';ctx.fillRect(b.x+2,b.y+2,b.w-4,Math.max(2,b.h*.22));
  ctx.strokeStyle=c.e;ctx.lineWidth=1.5;rr(b.x+.75,b.y+.75,b.w-1.5,b.h-1.5,4);ctx.stroke();
  if(b.tier>=3){ctx.fillStyle='rgba(0,0,0,.35)';
    const o=3.5*SC;for(const [ox,oy] of [[o,o],[b.w-o,o],[o,b.h-o],[b.w-o,b.h-o]]){
      ctx.beginPath();ctx.arc(b.x+ox,b.y+oy,1.6*SC,0,TAU);ctx.fill();}}
  if(b.hp<b.maxHp){const d=b.maxHp-b.hp;
    ctx.fillStyle='rgba(10,20,28,.5)';rr(b.x,b.y,b.w,b.h,4);ctx.fill();
    ctx.strokeStyle='rgba(6,14,20,.7)';ctx.lineWidth=1.2;ctx.beginPath();
    for(let i=0;i<d&&i<b.cracks.length;i++){const k=b.cracks[i];
      ctx.moveTo(b.x+k.x1*b.w,b.y+k.y1*b.h);ctx.lineTo(b.x+k.x2*b.w,b.y+k.y2*b.h);}
    ctx.stroke();}
  if(b.tier===4){ctx.save();ctx.shadowColor='#3fe8cf';ctx.shadowBlur=12+6*Math.sin(time*6+b.seed);
    ctx.strokeStyle='rgba(182,255,242,.9)';ctx.lineWidth=2;rr(b.x-1,b.y-1,b.w+2,b.h+2,5);ctx.stroke();ctx.restore();}
  if(b.flash>0){ctx.fillStyle=`rgba(255,255,255,${Math.min(.85,b.flash*3)})`;rr(b.x,b.y,b.w,b.h,4);ctx.fill();}}

function drawBall(b){
  const n=b.trail.length;
  for(let i=0;i<n;i++){const t=b.trail[i],f=i/n;
    ctx.globalAlpha=f*.3;ctx.fillStyle=effects.pierce>0?'#ffffff':'#3fe8cf';
    ctx.beginPath();ctx.arc(t.x,t.y,b.r*(.35+.6*f),0,TAU);ctx.fill();}
  ctx.globalAlpha=1;
  if(effects.pierce>0){ctx.save();ctx.shadowColor='#fff';ctx.shadowBlur=18;
    const a=Math.atan2(b.vy,b.vx),ln=b.r*3.4;
    ctx.strokeStyle='rgba(255,255,255,.9)';ctx.lineWidth=b.r*1.4;ctx.lineCap='round';
    ctx.beginPath();ctx.moveTo(b.x-Math.cos(a)*ln,b.y-Math.sin(a)*ln);ctx.lineTo(b.x,b.y);ctx.stroke();ctx.restore();}
  const g=ctx.createRadialGradient(b.x-b.r*.3,b.y-b.r*.3,1,b.x,b.y,b.r*1.6);
  g.addColorStop(0,'#ffffff');g.addColorStop(.55,effects.pierce>0?'#dff4ff':'#8ff2df');g.addColorStop(1,'rgba(63,232,207,0)');
  ctx.fillStyle=g;ctx.beginPath();ctx.arc(b.x,b.y,b.r*1.6,0,TAU);ctx.fill();
  ctx.fillStyle='#fff';ctx.beginPath();ctx.arc(b.x,b.y,b.r*.55,0,TAU);ctx.fill();}
