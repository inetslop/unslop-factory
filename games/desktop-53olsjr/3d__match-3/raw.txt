    if(e-r>=3){const ids=[];for(let i=r;i<e;i++){ids.push(grid[i][c].id);cells.add(grid[i][c].id);}runs.push({axis:'v',type:t.type,ids});}
    r=e;}}
  return {runs,cells};
}
function planSpecials(runs,hints){
  const creates=[],used=new Set(),claimed=new Set();
  for(const hr of runs)if(hr.axis==='h')for(const vr of runs)
    if(vr.axis==='v'&&vr.type===hr.type){
      const shared=hr.ids.find(id=>vr.ids.includes(id));
      if(shared!=null){const t=byId.get(shared);
        if(t&&!t.sp&&!claimed.has(t.id)){creates.push({tile:t,sp:{kind:'bomb'}});claimed.add(t.id);used.add(hr);used.add(vr);}}
    }
  for(const run of runs){
    if(used.has(run))continue;
    const len=run.ids.length;
    const kind=len>=5?{kind:'nova'}:(len===4?{kind:'line',axis:run.axis==='h'?'v':'h'}:null);
    if(!kind)continue;
    let anchor=null;
    if(hints)for(const hid of hints){const t=byId.get(hid);if(t&&run.ids.includes(t.id)&&!t.sp&&!claimed.has(t.id)){anchor=t;break;}}
    if(!anchor){const mid=byId.get(run.ids[(len/2)|0]);if(mid&&!mid.sp&&!claimed.has(mid.id))anchor=mid;}
    if(!anchor)for(const id of run.ids){const t=byId.get(id);if(!t.sp&&!claimed.has(t.id)){anchor=t;break;}}
    if(anchor){creates.push({tile:anchor,sp:kind});claimed.add(anchor.id);}
  }
  return creates;
}
function blastCells(t){
  const out=[],k=t.sp.kind;
  if(k==='line'){if(t.sp.axis==='h')for(let c=0;c<N;c++)out.push([t.r,c]);else for(let r=0;r<N;r++)out.push([r,t.c]);}
  else if(k==='bomb'){for(let dr=-1;dr<=1;dr++)for(let dc=-1;dc<=1;dc++){const r=t.r+dr,c=t.c+dc;if(r>=0&&r<N&&c>=0&&c<N)out.push([r,c]);}}
  else if(k==='nova'){for(let r=0;r<N;r++)for(let c=0;c<N;c++){const o=grid[r][c];if(o&&o.type===t.type)out.push([r,c]);}}
  return out;
}

async function trySwap(a,b){
  if(lock||state!=='play'||!a||!b)return false;
  if(Math.abs(a.r-b.r)+Math.abs(a.c-b.c)!==1)return false;
  lock=true;deselect();
  sSwap();
  const ar=a.r,ac=a.c;a.r=b.r;a.c=b.c;b.r=ar;b.c=ac;grid[a.r][a.c]=a;grid[b.r][b.c]=b;
  place(a);place(b);
  await sleep(SWAP_MS);
  const m=findMatches();const q0=[];if(a.sp)q0.push(a);if(b.sp)q0.push(b);
  if(m.cells.size===0&&q0.length===0){
    const ar2=a.r,ac2=a.c;a.r=b.r;a.c=b.c;b.r=ar2;b.c=ac2;grid[a.r][a.c]=a;grid[b.r][b.c]=b;
    place(a);place(b);sBad();
    [a,b].forEach(t=>t.el.firstElementChild.animate([
      {transform:'translateX(0) rotate(0)'},{transform:'translateX(-6px) rotate(-4deg)'},
      {transform:'translateX(5px) rotate(3deg)'},{transform:'translateX(0) rotate(0)'}],{duration:280}));
    lock=false;return false;
  }
  moves--;hud();hideHint();
  queue.push(...q0);
  await cascade([a.id,b.id]);
  if(levelScore>=target){doLevelClear();return true;}
  if(moves<=0){doGameOver();return true;}
  if(!hasMoveLive())await reshuffle();
  lock=false;return true;
}

function expandBlasts(trig,clearSet){
  const q=[...trig],done=new Set();
  while(q.length){
    const t=q.pop();if(done.has(t.id))continue;done.add(t.id);
    beamFX(t);
    for(const [r,c] of blastCells(t)){
      const o=grid[r][c];if(!o)continue;
      clearSet.add(o.id);
      if(o.sp&&!done.has(o.id))q.push(o);
    }
  }
  return done.size;
}

async function cascade(hintIds){
  let combo=0;
  while(true){
    const m=findMatches();
    if(m.cells.size===0&&queue.length===0)break;
    combo++;
    const creates=m.runs.length?planSpecials(m.runs,combo===1?hintIds:null):[];
    const clearSet=new Set(m.cells);
    const trig=[];
    m.cells.forEach(id=>{const t=byId.get(id);if(t&&t.sp)trig.push(t);});
    queue.forEach(t=>{if(!trig.includes(t))trig.push(t)});queue.length=0;
    const nTrig=expandBlasts(trig,clearSet);
    creates.forEach(cr=>clearSet.delete(cr.tile.id));
    const gained=clearSet.size*60*combo+nTrig*250;
    addScore(gained);
    let cx=0,cy=0,n=0;
    clearSet.forEach(id=>{
      const t=byId.get(id);if(!t)return;
      popTile(t);n++;
      const x=t.c*cell+cell/2,y=t.r*cell+cell/2;cx+=x;cy+=y;
      sparks(x,y,GEMGLOW[t.type],5+((Math.random()*4)|0),nTrig>0);
    });
    if(n){popup('+'+gained.toLocaleString('en-US'),cx/n,cy/n,combo);if(combo>1)comboPopup(combo);}
    if(nTrig>0){sBlast();shake();}else sPop(combo);
    creates.forEach(cr=>{cr.tile.sp=cr.sp;addSpecialEl(cr.tile);});
    if(creates.length)sForge();
    await sleep(POP_MS+60);
    clearSet.forEach(id=>{const t=byId.get(id);if(t)removeTile(t);});
    dropAndFill();
    await sleep(DROP_MS+80);
  }
}
