function buildWave(w){
 const q=[]; let t=0.5;
 if(w%10===0){ q.push({type:'boss',t}); const n=4+Math.floor(w/4); for(let i=0;i<n;i++){q.push({type:i%2?'skitter':'drifter',t:t+3+i*2.4});} }
 else{
  const n=Math.round(6+w*1.7);
  const pool=[['drifter',10]];
  if(w>=3)pool.push(['skitter',7]);
  if(w>=5)pool.push(['splitter',6]);
  if(w>=4)pool.push(['brute',5]);
  if(w>=7)pool.push(['shimmer',6]);
  if(w>=9)pool.push(['wraith',6]);
  const tot=pool.reduce((s,p)=>s+p[1],0);
  // group spawn: bursts
  let burst=3+Math.min(4,Math.floor(w/4));
  for(let i=0;i<n;i++){
    if(i%burst===0) t+= (i===0?0: Math.max(.5, 1.6 - w*.02)); // pause between bursts
    else t+= Math.max(.22, .8 - w*.015);
    // weighted pick
    let r=Math.random()*tot, type='drifter';
    for(const [tp,wt] of pool){ if(r<wt){type=tp;break;} r-=wt; }
    q.push({type,t});
  }
 }
 q.sort((a,b)=>a.t-b.t); G.queue=q; G.spawnT=0;
}

    const dx=tg.px-s.x,dy=tg.py-s.y,d=Math.hypot(dx,dy),step=s.sp*dt;
    if(d<=step+6){damage(tg,s.dmg);burst(s.x,s.y,'#ffb38a',3,1.1);s.done=true;}else{s.x+=dx/d*step;s.y+=dy/d*step;}}
  else{const tg=s.tg;if(tg&&!tg.dead){s.ax=tg.px;s.ay=tg.py;}s.p+=dt/s.T;
    if(Math.random()<.5)G.parts.push({x:s.x+(Math.random()-.5)*4,y:s.y-18+Math.sin(s.p*Math.PI)*62,vx:(Math.random()-.5)*20,vy:30,life:.3,t:0,size:7,col:'#ffd166',a:.7});
    if(s.p>=1){s.done=true;SFX.boom();ringFx(s.ax,s.ay,8,s.splash,'#ffd166');burst(s.ax,s.ay,'#ffb347',12,2);
      for(const e of G.enemies){if(e.dead)continue;const d=Math.hypot(e.px-s.ax,e.py-s.ay);if(d<=s.splash)damage(e,s.dmg*(1-.5*d/s.splash));}}}}}
function moveEnemies(dt,menu){for(const e of G.enemies){if(e.dead)continue;
  const spd=e.def.spd*(1+Math.min(.5,(G.wave||3)*.008));e.dist+=spd*e.slowF*dt;
  if(e.def.phase)e.phased=(e.pt+=dt)%2.6>1.7;
  if(e.def.boss){e.summonT-=dt;if(e.summonT<=0&&!menu){e.summonT=7;for(let i=0;i<3;i++)spawnEnemy('mite',Math.max(0,e.dist-30-i*14));}}
  const p=pointAt(e.dist);e.px=p.x;e.py=p.y;e.pa=p.ang;e.flash=Math.max(0,e.flash-dt);
  if(e.dist>=TOTAL)leak(e);}
  G.enemies=G.enemies.filter(e=>!e.dead);}

function update(dt){if(G.banner){G.banner.t+=dt;if(G.banner.t>G.banner.life)G.banner=null;}
  if(!G.waveActive){if(G.countdown>0){G.countdown-=dt;if(G.countdown<=0)startWave();}}
  else{G.spawnT+=dt;while(G.queue.length&&G.queue[0].t<=G.spawnT)spawnEnemy(G.queue.shift().type);}
  for(const e of G.enemies)e.slowF=1;
  updateTowers(dt);updateShots(dt);moveEnemies(dt,false);
  G.shots=G.shots.filter(s=>!s.done);
  if(G.waveActive&&!G.queue.length&&!G.enemies.length){G.waveActive=false;const b=12+G.wave*5;G.gold+=b;G.score+=G.wave*30;
    addText(W/2,H*.46,'WAVE CLEAR  +'+b+'g','#3ff2d0');G.countdown=7;}}

  for(const t of G.texts){t.t+=dt;t.y-=30*dt;}G.texts=G.texts.filter(t=>t.t<t.life);}
