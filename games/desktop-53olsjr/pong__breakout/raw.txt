function buildWall(lv){
  bricks=[]; const pat=(lv-1)%4; let total=0;
  for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++){
    let skip=false;
    if(pat===1&&(r+c)%2)skip=true;
    if(pat===2&&(r===1||r===2)&&(c===1||c===4))skip=true;
    if(pat===3&&Math.abs(c-2.5)+Math.abs(r-1.5)>2.9)skip=true;
    if(skip)continue;
    let hp=1;
    if(lv>=2&&(r*COLS+c)%5===0)hp=2;

}

function predictY(b,pad){
  const m=Math.hypot(b.vx,b.vy); if(m<1)return LH/2;
  let x=b.x,y=b.y,ux=b.vx/m,uy=b.vy/m;
  const limit=pad===C?C.x-14:P.x+14;
  for(let i=0;i<800;i++){
    x+=ux*10; y+=uy*10;
    if(y<R+WALL){y=R+WALL;uy=-uy;} else if(y>LH-R-WALL){y=LH-R-WALL;uy=-uy;}
    if((pad===C&&x>=limit)||(pad!==C&&x<=limit))return y;
  }
  return y;
}
function aiMove(pad,dt,maxS){
  const b=balls[0]; let target;
  if(!b||b.stuck)target=LH/2+Math.sin(animT*0.7+(pad===C?2:0))*34;
  else{
    const toward=(pad===C&&b.vx>0)||(pad!==C&&b.vx<0);
    target=toward?predictY(b,pad)+(pad===C?C.err:(Math.sin(animT*1.3)*50)):LH/2+Math.sin(animT*0.6+(pad===C?4:0))*44;
  }
  const lo=WALL+pad.h/2+4, hi=LH-WALL-pad.h/2-4;
  target=Math.max(lo,Math.min(hi,target));
  const d=target-pad.y, mv=Math.sign(d)*Math.min(Math.abs(d),maxS*dt);
  pad.prevY=pad.y; pad.y+=mv; pad.vy=(pad.y-pad.prevY)/Math.max(dt,0.001);
}

function padHit(pad,isPlayer){
  const b=balls[0]||balls.find(b=>!b.stuck); if(!b)return;
  const hw=pad.w/2, hh=pad.h/2;
  const cx=Math.max(pad.x-hw,Math.min(b.x,pad.x+hw)), cy=Math.max(pad.y-hh,Math.min(b.y,pad.y+hh));
  let dx=b.x-cx, dy=b.y-cy; const d2=dx*dx+dy*dy;
  if(d2>R*R)return;
  let nx,ny,pen;
  if(d2===0){nx=isPlayer?-1:1;ny=0;pen=R;}
  else{const d=Math.sqrt(d2);nx=dx/d;ny=dy/d;pen=R-d;}
  b.x+=nx*pen; b.y+=ny*pen;
  const dot=b.vx*nx+b.vy*ny; if(dot<0){b.vx-=2*dot*nx;b.vy-=2*dot*ny;}
  lastHit=isPlayer?0:1; combo=0;
  if(!isPlayer)C.err=(Math.random()*2-1)*(95-diffNow()*65);
  const rel=Math.max(-1,Math.min(1,(b.y-pad.y)/hh));
  const ang=rel*MAXA, dir=isPlayer?1:-1;
  ballSpeed=Math.min(ballSpeed+10,MAXS);
  b.vx=Math.cos(ang)*ballSpeed*dir;
  b.vy=Math.sin(ang)*ballSpeed+pad.vy*0.18;
  const m=Math.hypot(b.vx,b.vy), cap=ballSpeed*1.15;
  if(m>cap){b.vx*=cap/m;b.vy*=cap/m;}
  burst(cx,cy,isPlayer?'#ffb057':'#3fe8cc',6,140,3);
  sfx.paddle(ballSpeed); shake=Math.min(shake+1.2,14);
}

      burst(br.x+BW/2,br.y+BH/2,'rgb('+col.join(',')+')',14,220,3.5);
      shake=Math.min(shake+3.5,14); sfx.brick(br.row,br.max);
      matchBricks++; topLevel=Math.max(topLevel,level);
      combo++;
      addScore(lastHit,1,br.x+BW/2,br.y+BH/2-6);
      if(combo>0&&combo%5===0){addScore(lastHit,1,LW/2,150,'COMBO +1');shake+=2;}
      if(Math.random()<0.13)spawnPower(br.x+BW/2,br.y+BH/2);
      let alive=0; for(const q of bricks)if(q.alive)alive++;
      if(alive===0){
        if(state==='menu'){buildWall(level);level++;}
        else{state='levelclear';clearT=1.9;level++;sfx.level();confetti();ftext('LEVEL '+level,LW/2,300,10,'#ffd23f','center');}
      }
    }else{
      burst(br.x+BW/2,br.y+BH/2,'rgba(255,255,255,0.9)',6,120,2.5);
      sfx.thud(); shake=Math.min(shake+1.5,14);
    }
  }else{
    br.hp--; if(br.hp<=0){br.alive=false;burst(br.x+BW/2,br.y+BH/2,'rgb('+ROWCOL[br.row].join(',')+')',10,200,3);}
    let alive=0;for(const q of bricks)if(q.alive)alive++;
    if(alive===0){buildWall(level);level++;}
  }

function stepBall(b,dt){
  const sp=Math.hypot(b.vx,b.vy)*slowF;
  const n=Math.max(1,Math.ceil(sp*dt/6));
  for(let s=0;s<n;s++){
    b.x+=b.vx*slowF*dt/n; b.y+=b.vy*slowF*dt/n;
    if(b.y<R+WALL){b.y=R+WALL;b.vy=Math.abs(b.vy);if(animT-lastWallSfx>0.06){sfx.wall();lastWallSfx=animT;}burst(b.x,R+2,'#9fd8e8',3,90,2);}
    else if(b.y>LH-R-WALL){b.y=LH-R-WALL;b.vy=-Math.abs(b.vy);if(animT-lastWallSfx>0.06){sfx.wall();lastWallSfx=animT;}burst(b.x,LH-R-2,'#9fd8e8',3,90,2);}
    padHit(P,true); padHit(C,false);
    for(const br of bricks){
      if(!br.alive)continue;
      const cx=Math.max(br.x,Math.min(b.x,br.x+BW)), cy=Math.max(br.y,Math.min(b.y,br.y+BH));
      let dx=b.x-cx, dy=b.y-cy; const d2=dx*dx+dy*dy;
      if(d2>R*R)continue;
      let nx,ny,pen;
      if(d2===0){
        const l=b.x-br.x,r2=br.x+BW-b.x,t=b.y-br.y,bt=br.y+BH-b.y,m=Math.min(l,r2,t,bt);
        nx=m===l?-1:m===r2?1:0; ny=m===t?-1:m===bt?1:0; pen=R+m;
      }else{const d=Math.sqrt(d2);nx=dx/d;ny=dy/d;pen=R-d;}
      b.x+=nx*pen;b.y+=ny*pen;
      const dot=b.vx*nx+b.vy*ny; if(dot<0){b.vx-=2*dot*nx;b.vy-=2*dot*ny;}
      if(br.cd<=0){br.cd=0.12;br.flash=1;hitBrick(br);}
    }
    if(b.x<WALL-R*2){if(state==='menu'){setupServe(1);}else{state='point';pendingServer=1;pendingWinner=-1;pointT=1.35;goalGlow={side:0,t:1};scoreGoal(1);}return;}
    if(b.x>LW-WALL+R*2){if(state==='menu'){setupServe(0);}else{state='point';pendingServer=0;pendingWinner=-1;pointT=1.35;goalGlow={side:1,t:1};scoreGoal(0);}return;}
  }
}

function update(dt){
  animT+=dt;
  shake=Math.max(0,shake-shake*7*dt-2*dt);
  shx=(Math.random()-0.5)*shake*2; shy=(Math.random()-0.5)*shake*2;
  for(const m of motes){m.y-=m.sp*dt;m.x+=Math.sin(animT+m.ph)*6*dt;if(m.y<-4){m.y=LH+4;m.x=Math.random()*LW;}}
  for(let i=parts.length-1;i>=0;i--){const p=parts[i];p.life-=dt;if(p.life<=0){parts.splice(i,1);continue;}p.x+=p.vx*dt;p.y+=p.vy*dt;p.vx*=Math.pow(0.9,dt*60);p.vy*=Math.pow(0.9,dt*60);}
  for(let i=texts.length-1;i>=0;i--){const t=texts[i];t.life-=dt;t.y+=t.vy*dt;if(t.life<=0)texts.splice(i,1);}
  goalGlow.t=Math.max(0,goalGlow.t-dt*0.9);
  popT[0]=Math.max(0,popT[0]-dt*3); popT[1]=Math.max(0,popT[1]-dt*3);

  if(state==='pause')return;

  updatePads(dt);
  for(const br of bricks){br.cd=Math.max(0,br.cd-dt);br.flash=Math.max(0,br.flash-dt*4);}

  if(state==='menu'||state==='serve'){
    const pad=serverSide?C:P, b=balls[0];
    if(b){b.x=serverSide?pad.x-(pad.w/2+R+6):pad.x+(pad.w/2+R+6);b.y=pad.y;}
    if(serveT>-100&&serveT<90){serveT-=dt;if(serveT<=0)launchBall();}
  }
  else if(state==='play'){
    slowTimer=Math.max(0,slowTimer-dt);
    const st=slowTimer>0?0.62:1; slowF+=(st-slowF)*Math.min(1,dt*3);
    for(const b of balls){if(state!=='play')break;stepBall(b,dt);}
    /* powerups */
    for(let i=powerups.length-1;i>=0;i--){
      const p=powerups[i]; p.t+=dt; p.x+=p.vx*dt;
      if(p.t>5||p.x<WALL+8||p.x>LW-WALL-8){powerups.splice(i,1);continue;}
      let caught=false;
      for(const pad of [P,C]){
        if(Math.abs(p.x-pad.x)<pad.w/2+16&&Math.abs(p.y-pad.y)<pad.h/2+12){
          caught=true;
          const mine=pad===P;
          if(p.type==='W')pad.wide=8;
          else if(p.type==='S')slowTimer=Math.max(slowTimer,6);
          else{
            const src=balls[0]||balls[0];
            if(balls.length<3&&src){
              for(let k=1;k<2;k++){
                const a=Math.atan2(src.vy,src.vx)+(k?0.5:-0.5)*(0.4+Math.random()*0.3);
                balls.push({x:src.x,y:src.y,vx:Math.cos(a)*ballSpeed,vy:Math.sin(a)*ballSpeed,stuck:false,trail:[]});
              }
            }
          }
          ftext(PU[p.type].n,pad.x+(mine?-70:70),pad.y-14,3,PU[p.type].c,'center');
          burst(p.x,p.y,PU[p.type].c,12,180,3); sfx.power(); shake=Math.min(shake+2,14);
          break;
        }
      }
      if(caught)powerups.splice(i,1);
    }
    /* trails */
    for(const b of balls){b.trail.push({x:b.x,y:b.y});if(b.trail.length>13)b.trail.shift();}
  }
  else if(state==='point'){
    pointT-=dt;
    if(pointT<=0){
      if(pendingWinner>=0)gameOver(pendingWinner);
      else setupServe(pendingServer);
    }
  }
  else if(state==='levelclear'){
    clearT-=dt;
    if(clearT<=0){buildWall(level);setupServe(0);}
  }
  else if(state==='over'){overT+=dt;}
}

    ctx.moveTo(br.x+8+(s%10),br.y+4); ctx.lineTo(br.x+16+(s%7),br.y+BH/2); ctx.lineTo(br.x+10+(s%9),br.y+BH-4);
    ctx.moveTo(br.x+BW-10-(s%5),br.y+3); ctx.lineTo(br.x+BW-18-(s%4),br.y+BH-6);
    ctx.stroke();
  }
  if(br.max>=2){ctx.strokeStyle='rgba(255,255,255,'+(0.35+wave*0.2)+')';ctx.lineWidth=2;rr(br.x+3,br.y+3,BW-6,BH-6,2);ctx.stroke();}
  if(br.max>=3){ctx.strokeStyle='rgba(255,255,255,'+(0.25+0.25*Math.sin(animT*4+br.seed))+')';ctx.lineWidth=1.5;rr(br.x-2,br.y-2,BW+4,BH+4,4);ctx.stroke();}
  if(br.flash>0){ctx.fillStyle='rgba(255,255,255,'+(br.flash*0.85)+')';rr(br.x,br.y,BW,BH,3);ctx.fill();}
}
function drawPad(pad,colStr){
  const w=pad.w,h=pad.h;
  if(Math.abs(pad.y-pad.prevY)>1.5){ctx.globalAlpha=0.22;ctx.fillStyle=colStr;rr(pad.x-w/2,pad.prevY-h/2,w,h,6);ctx.fill();ctx.globalAlpha=1;}
  const g=ctx.createLinearGradient(0,pad.y-h/2,0,pad.y+h/2);
  g.addColorStop(0,'rgba(255,255,255,0.95)');g.addColorStop(0.25,colStr);g.addColorStop(1,colStr);
  ctx.save();
  if(pad.wide>0){ctx.shadowColor=colStr;ctx.shadowBlur=18;}
  ctx.fillStyle=g; rr(pad.x-w/2,pad.y-h/2,w,h,6); ctx.fill();
  ctx.restore();
}
function drawBall(b){
  const n=b.trail.length;
  ctx.save(); ctx.globalCompositeOperation='lighter';
  for(let i=0;i<n;i++){const p=b.trail[i],f=i/n;
    ctx.fillStyle='rgba(255,214,140,'+(f*0.22)+')';
    ctx.beginPath();ctx.arc(p.x,p.y,R*(0.3+f*0.8),0,7);ctx.fill();}
  const gs=R*6*(1+0.06*Math.sin(animT*6));
  ctx.drawImage(glowCv,b.x-gs/2,b.y-gs/2,gs,gs);
  ctx.restore();
  ctx.fillStyle='#fff6e4';ctx.beginPath();ctx.arc(b.x,b.y,R,0,7);ctx.fill();
  ctx.fillStyle='rgba(255,255,255,0.9)';ctx.beginPath();ctx.arc(b.x-R*0.3,b.y-R*0.35,R*0.4,0,7);ctx.fill();
}
function drawKeycap(x,y,label){
  const w=Math.max(30,label.length*8+16), h=26;
  ctx.fillStyle='#182338'; rr(x,y,w,h,6); ctx.fill();
  ctx.strokeStyle='rgba(255,255,255,0.28)'; ctx.lineWidth=1.5; rr(x,y,w,h,6); ctx.stroke();
  body(label,x+w/2,y+17,12,'rgba(255,255,255,0.9)','center',700);
  return w;
}
function drawControlRow(items,y){
  let total=0; const ws=[];
  for(const it of items){const w=it.cap?Math.max(30,it.cap.length*8+16):it.txt.length*(y>520?7.4:8.2);ws.push(w);total+=w;}
  let x=LW/2-total/2;
  for(let i=0;i<items.length;i++){
    const it=items[i];
    if(it.cap)drawKeycap(x,y,it.cap);
    else body(it.txt,x+ws[i]/2,y+17,13,'rgba(255,255,255,0.65)','center',700);
    x+=ws[i];
  }
}
