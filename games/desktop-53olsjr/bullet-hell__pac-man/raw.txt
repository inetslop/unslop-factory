  for(const def of GHOST_DEFS)if(G.time>=def.at&&!ghosts.some(g=>g.def===def))spawnGhost(def);
  if(G.time>=92&&!G.gun){G.gun=true;floatTxt(W/2,H*.3,'GAUNTLET · FULL STORM','#ff4d6d',24);sfx.stage();}
  pelletT-=dt;
  if(pelletT<=0&&pellets.length<14){pelletT=.55;const s=placeSpot(false);if(s)pellets.push({x:s.x,y:s.y,life:10});}
  powerT-=dt;
  if(powerT<=0&&G.power<=0&&!P.dead){powerT=rand(10,15);const s=placeSpot(true);
    if(s)ppe={x:s.x,y:s.y,life:9};else powerT=2;}
}
function update(dt,rdt){
  for(const d of dust){d.y-=d.v*dt;if(d.y<-4){d.y=H+4;d.x=rand(0,W);}}
  if(G.state==='menu'||G.state==='playing'){
    const demo=G.state==='menu';
    G.shake=Math.max(0,G.shake-dt*26);G.flash=Math.max(0,G.flash-rdt*1.6);
    spawnSchedule(demo,dt);
    for(const g of ghosts)upGhost(g,dt,demo,false);
    upBullets(dt,demo);
    if(!demo){
      G.time+=dt;G.score+=dt*8;
      for(let i=pellets.length-1;i>=0;i--){pellets[i].life-=dt;if(pellets[i].life<=0)pellets.splice(i,1);}
      if(ppe){ppe.life-=dt;if(ppe.life<=0)ppe=null;}
      upPlayer(dt);
      if(P.dead){G.overT-=rdt;if(G.overT<=0)doGameOver();}
    }
  }else if(G.state==='gameover'){
    G.shake=Math.max(0,G.shake-dt*26);
    for(const g of ghosts)upGhost(g,dt,true,true);
    upBullets(dt,true);
  }
  for(let i=parts.length-1;i>=0;i--){const p=parts[i];p.life-=dt;
    if(p.life<=0){parts.splice(i,1);continue;}
    if(p.k==='s'){p.x+=p.vx*dt;p.y+=p.vy*dt;p.vx*=1-dt*2;p.vy*=1-dt*2;}
    else p.r+=(p.max-p.r)*Math.min(1,dt*9);}
  for(let i=floats.length-1;i>=0;i--){const f=floats[i];f.life-=dt;f.y-=28*dt;if(f.life<=0)floats.splice(i,1);}
}

/* ---------- render ---------- */
function drawGhost(g){
  if(g.born<0)return;
  const r=20,A=aim(),ta=Math.atan2(A.y-g.y,A.x-g.x);
  const sc=Math.min(1,g.born*1.7),sx=1+g.squash*.14,sy=1-g.squash*.22;
  ctx.save();ctx.translate(g.x,g.y);ctx.scale(sc*sx,sc*sy);
  ctx.shadowColor=g.def.col;ctx.shadowBlur=16;
  ctx.fillStyle=g.def.col;ctx.beginPath();
  ctx.arc(0,-r*.15,r,Math.PI,TAU);
  const by=r*.9;
  for(let i=1;i<=8;i++)ctx.lineTo(r-2*r*i/8,by+Math.sin(i*1.35+g.born*9)*4);
  ctx.closePath();ctx.fill();ctx.shadowBlur=0;
  const ex=Math.cos(ta)*2.6,ey=Math.sin(ta)*2.6;
  ctx.fillStyle='#fff';
  ctx.beginPath();ctx.ellipse(-6,-r*.3,4.4,5,0,0,TAU);ctx.fill();
  ctx.beginPath();ctx.ellipse(6,-r*.3,4.4,5,0,0,TAU);ctx.fill();
  ctx.fillStyle='#0a0e22';
  ctx.beginPath();ctx.arc(-6+ex,-r*.3+ey,2,0,TAU);ctx.fill();
  ctx.beginPath();ctx.arc(6+ex,-r*.3+ey,2,0,TAU);ctx.fill();
  if(g.flash>0){const m=g.flash*15;
    ctx.fillStyle=g.def.dark;ctx.beginPath();
    ctx.arc(Math.cos(ta)*r*.45,Math.sin(ta)*r*.45-r*.05,m*.62,0,TAU);ctx.fill();}
  ctx.restore();
}
function drawPlayer(){
  const power=G.power>0;
  for(let i=0;i<P.trail.length;i++){const t=P.trail[i],a=(i+1)/P.trail.length*.3*(power?1.7:1);
    ctx.globalAlpha=a;ctx.fillStyle=power?'#9ff2ff':'#ffd93b';
    ctx.beginPath();ctx.arc(t.x,t.y,P.r*.45*(i/P.trail.length+.3),0,TAU);ctx.fill();}
  ctx.globalAlpha=1;if(P.dead)return;
  let al=1;if(P.inv>0&&Math.floor(G.time*16)%2===0)al=.35;
  ctx.save();ctx.translate(P.x,P.y);ctx.globalAlpha=al;
  if(power){
    ctx.setLineDash([7,9]);ctx.lineDashOffset=-G.time*80;
    ctx.strokeStyle='rgba(120,240,255,.6)';ctx.lineWidth=2;
    ctx.beginPath();ctx.arc(0,0,P.r+8,0,TAU);ctx.stroke();ctx.setLineDash([]);
    ctx.shadowColor='#7ff0ff';ctx.shadowBlur=18;
  }else{ctx.shadowColor='rgba(255,217,59,.7)';ctx.shadowBlur=10;}
  ctx.rotate(P.ang);
  const ch=P.moving?.25+Math.abs(Math.sin(P.chomp))*.95:.18;
  ctx.fillStyle=power?'#eaffff':'#ffd93b';
  ctx.beginPath();ctx.moveTo(0,0);ctx.arc(0,0,P.r,ch,TAU-ch);ctx.closePath();ctx.fill();
  ctx.restore();ctx.globalAlpha=1;
}
function render(){
  ctx.drawImage(bgC,0,0,W,H);
  for(const d of dust){ctx.globalAlpha=d.a;ctx.fillStyle='#9db4ff';ctx.fillRect(d.x,d.y,d.r,d.r);}
  ctx.globalAlpha=1;
  let sx=0,sy=0;if(G.shake>.2){sx=(Math.random()*2-1)*G.shake;sy=(Math.random()*2-1)*G.shake;}
  ctx.save();ctx.translate(sx,sy);
  for(let i=0;i<pellets.length;i++){const p=pellets[i];
    const a=clamp(p.life<1.5?p.life/1.5:1,0,1)*(.7+.3*Math.sin(G.time*6+i));
    ctx.globalAlpha=a;ctx.fillStyle='#ffe9a3';ctx.beginPath();ctx.arc(p.x,p.y,3.2,0,TAU);ctx.fill();}
  ctx.globalAlpha=1;
  if(ppe){const blink=ppe.life<2&&Math.floor(G.time*8)%2===0;
    const pu=.75+.25*Math.sin(G.time*7),r=(8+pu*2)*2.2;
    const gr=ctx.createRadialGradient(ppe.x,ppe.y,0,ppe.x,ppe.y,r);
    gr.addColorStop(0,'#fff');gr.addColorStop(.4,'#ffd93b');gr.addColorStop(1,'rgba(255,217,59,0)');
    ctx.globalAlpha=blink?.35:1;ctx.fillStyle=gr;
    ctx.beginPath();ctx.arc(ppe.x,ppe.y,r,0,TAU);ctx.fill();
    for(let k=0;k<2;k++){const a=G.time*4+k*Math.PI;
      ctx.fillStyle='#fff';ctx.beginPath();
      ctx.arc(ppe.x+Math.cos(a)*15,ppe.y+Math.sin(a)*15,1.8,0,TAU);ctx.fill();}
    ctx.globalAlpha=1;}
  for(const g of ghosts)if(g.tel){const A=aim();
    ctx.save();ctx.setLineDash([6,7]);ctx.lineDashOffset=-G.time*90;
    ctx.strokeStyle=rgba(g.def.col,.35+.3*Math.sin(G.time*30));ctx.lineWidth=1.5;
    ctx.beginPath();ctx.moveTo(g.x,g.y);ctx.lineTo(A.x,A.y);ctx.stroke();ctx.restore();}
  for(const g of ghosts)drawGhost(g);
  if(G.state!=='menu')drawPlayer();
  for(const b of bullets){const s=bulletSprite(b.col,Math.round(b.r));
    ctx.drawImage(s.c,b.x-s.w/2,b.y-s.w/2,s.w,s.w);
    if(b.flash>0){ctx.globalAlpha=Math.min(1,b.flash*3);ctx.strokeStyle='#fff';ctx.lineWidth=1.5;
      ctx.beginPath();ctx.arc(b.x,b.y,b.r+4,0,TAU);ctx.stroke();ctx.globalAlpha=1;}}
  for(const p of parts){const a=clamp(p.life/p.t,0,1);ctx.globalAlpha=a;
    if(p.k==='s'){ctx.strokeStyle=p.col;ctx.lineWidth=p.r;
      ctx.beginPath();ctx.moveTo(p.x,p.y);ctx.lineTo(p.x-p.vx*.025,p.y-p.vy*.025);ctx.stroke();}
    else{ctx.strokeStyle=p.col;ctx.lineWidth=2.5;
      ctx.beginPath();ctx.arc(p.x,p.y,p.r,0,TAU);ctx.stroke();}}
  ctx.globalAlpha=1;
  for(const f of floats){const a=clamp(f.life/f.t*1.4,0,1);
    ctx.globalAlpha=a;ctx.font='700 '+f.size+'px Consolas,Menlo,monospace';ctx.textAlign='center';
    ctx.strokeStyle='rgba(0,0,0,.65)';ctx.lineWidth=3;ctx.strokeText(f.txt,f.x,f.y);
    ctx.fillStyle=f.col;ctx.fillText(f.txt,f.x,f.y);}
  ctx.globalAlpha=1;ctx.restore();
  if(G.flash>0){ctx.fillStyle='rgba(255,255,255,'+Math.min(.7,G.flash*4)+')';ctx.fillRect(0,0,W,H);}
}
