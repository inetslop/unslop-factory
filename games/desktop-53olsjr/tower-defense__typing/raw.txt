          const a=t2.ang;
          bullets.push({x:cx+Math.cos(a)*26,y:cy2+Math.sin(a)*26,px:cx,py:cy2,tx:target.sx,ty:target.sy,t:target,dmg:1,real:true,sp:540*scale,r:3,col:'#8dffb0',life:0,crit:false});
          t2.flash=.09; t2.rec=1; sfx.boltHit();
        }
      } else if(up.turret>0) t2.cd=Math.min(t2.cd,dt);
      if(!target||target.dead)target=enemies.length?lead():null;
    }
  }
  const sp=(22+wave*3.4)*scale;
  for(const e of enemies){
    let mul=1; if(e.type==='wraith'&&e.typed>=e.len-2)mul=1.9;
    e.x-=TYPEDEF[e.type].spMul*sp*mul*dt;
    e.sx=e.x; e.sy=lerp(topY+14,botY-16,e.laneFrac)+Math.sin(at*1.4+e.bobP)*e.bobA;
    if(e.x<=baseW+e.size*.35){
      e.dead=true; activeWords.delete(e.word);
      if(live)hitBase(e); else explode(e.sx,e.sy,TYPEDEF[e.type].col,e.size/12,.5);
    }
  }
  enemies=enemies.filter(e=>!e.dead);
  if(live&&interT<=0&&wave>0&&toSpawn===0&&enemies.length===0)waveCleared();
}
function stepFx(dt){
  shake=Math.max(0,shake-dt*(shake*6+10));
  wallHitT=Math.max(0,wallHitT-dt*1.2);
  muzzleT=Math.max(0,muzzleT-dt); muzzleRec=Math.max(0,muzzleRec-dt*7);
  t2.flash=Math.max(0,t2.flash-dt); t2.rec=Math.max(0,t2.rec-dt*7);
  for(const e of enemies){ if(e.dead)continue; e.flash=Math.max(0,e.flash-dt*4); e.errT=Math.max(0,e.errT-dt); e.selT=Math.max(0,e.selT-dt*1.6); }
  // cannon aim
  const desA=(target&&!target.dead)?Math.atan2(target.sy-cy,target.sx-cx):(-.15+Math.sin(at*.6)*.1);
  cannonAng+=(desA-cannonAng)*Math.min(1,dt*10);
  const desB=(target&&!target.dead&&up.turret>0)?Math.atan2(target.sy-cy2,target.sx-cx):(-.3+Math.sin(at*.8)*.15);
  t2.ang+=(desB-t2.ang)*Math.min(1,dt*10);
  // bullets
  for(const b of bullets){
    b.life+=dt;
    if(b.t&&!b.t.dead){ b.tx=b.t.sx; b.ty=b.t.sy; }
    const a=Math.atan2(b.ty-b.y,b.tx-b.x);
    b.px=b.x; b.py=b.y; b.x+=Math.cos(a)*b.sp*dt; b.y+=Math.sin(a)*b.sp*dt;
    const d=Math.hypot(b.tx-b.x,b.ty-b.y), hitR=(b.t&&!b.t.dead)?b.t.size*.5+10:8;
    if(d<hitR||b.life>1.7){
      b.done=true;
      if(b.real&&b.t&&!b.t.dead){
        b.t.typed=Math.min(b.t.len,b.t.typed+b.dmg); b.t.flash=1;
        for(let i=0;i<4;i++){const aa=rand(0,TAU),ss=rand(40,120);parts.push({type:'spark',x:b.x,y:b.y,vx:Math.cos(aa)*ss,vy:Math.sin(aa)*ss,g:300,life:.25,max:.25,size:2,col:'#8dffb0'});}
        sfx.boltHit();
        if(b.t.typed>=b.t.len)killEnemy(b.t);
      } else {
        for(let i=0;i<3;i++){const aa=rand(0,TAU),ss=rand(20,70);parts.push({type:'spark',x:b.x,y:b.y,vx:Math.cos(aa)*ss,vy:Math.sin(aa)*ss,g:200,life:.2,max:.2,size:1.6,col:b.col});}
      }
    }
  }
  bullets=bullets.filter(b=>!b.done);
  // particles
  for(const p of parts){
    p.life-=dt; if(p.type==='spark'||p.type==='shard')p.vy+=p.g*dt;
    p.x+=(p.vx||0)*dt; p.y+=(p.vy||0)*dt;
    if(p.type==='smoke')p.size+=13*dt;
    if(p.type==='ring')p.r+=p.vr*dt;
  }
  parts=parts.filter(p=>p.life>0);
  for(const f of floaters)f.t+=dt;
  floaters=floaters.filter(f=>f.t<.9);
  // torch embers (ambience)
  emberAcc+=dt;
  if(emberAcc>.12&&parts.length<420){
    emberAcc=0;
    const th=Math.random()<.5?{x:baseW-8,y:H*.24}:{x:baseW-8,y:H*.68};
    parts.push({type:'ember',x:th.x+rand(-3,3),y:th.y-14,vx:rand(-7,7),vy:-rand(9,24),g:0,life:rand(.8,1.5),max:1.5,size:rand(1,2.3),col:'#ffb35c'});
  }
}

    if(comb){ show(el); $('comboN').textContent=combo; el.classList.remove('pop'); void el.offsetWidth; el.classList.add('pop'); }
    else hide(el); }
  // shop
  for(const s of SHOP){
    const el=shopEls[s.id], l=up[s.id], maxed=l>=s.max, full=(s.id==='repair'&&baseHP>=10);
    const c=maxed?null:s.cost(l);

    const box=el.querySelector('.pips');
    if(box){ const pp=box.children; for(let i=0;i<pp.length;i++)pp[i].classList.toggle('on',i<l); }
  }
}
