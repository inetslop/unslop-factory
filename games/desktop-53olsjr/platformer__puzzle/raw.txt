function audio(){

 if(AC&&AC.state==='suspended')AC.resume().catch(()=>{});
 return AC;
}
function getNoise(){if(!AC)return null;if(!noiseBuf){noiseBuf=AC.createBuffer(1,AC.sampleRate,AC.sampleRate);const d=noiseBuf.getChannelData(0);for(let i=0;i<d.length;i++)d[i]=Math.random()*2-1;}return noiseBuf}
function tone(o){const ac=audio();if(!ac||muted)return;try{
 const {f0,f1=f0,t=.15,type='square',v=.25,d=0}=o,now=ac.currentTime+d;
 const osc=ac.createOscillator(),g=ac.createGain();
 osc.type=type;osc.frequency.setValueAtTime(Math.max(1,f0),now);
 if(f1!==f0)osc.frequency.exponentialRampToValueAtTime(Math.max(1,f1),now+t);
 g.gain.setValueAtTime(v,now);g.gain.exponentialRampToValueAtTime(.0001,now+t);
 osc.connect(g);g.connect(master);osc.start(now);osc.stop(now+t+.03);}catch(e){}}
function noiseS(o){const ac=audio();if(!ac||muted)return;try{
 const b=getNoise();if(!b)return;
 const {t=.2,v=.25,f=800,sweep=1,q=1,d=0}=o,now=ac.currentTime+d;
 const s=ac.createBufferSource();s.buffer=b;
 const fl=ac.createBiquadFilter();fl.type='bandpass';fl.Q.value=q;
 fl.frequency.setValueAtTime(Math.max(30,f),now);fl.frequency.exponentialRampToValueAtTime(Math.max(30,f*sweep),now+t);
 const g=ac.createGain();g.gain.setValueAtTime(v,now);g.gain.exponentialRampToValueAtTime(.0001,now+t);
 s.connect(fl);fl.connect(g);g.connect(master);s.start(now);s.stop(now+t+.03);}catch(e){}}

function updatePlayer(dt){
 if(!player.alive){player.respawnT-=dt;if(player.respawnT<=0)respawnPlayer();return;}
 const p=player;
 // input
 const dir=(keys.r?1:0)-(keys.l?1:0);
 if(dir)p.facing=dir;
 const acc=p.grounded?2700:1650;
 if(dir)p.vx+=dir*acc*dt;
 else{const f=p.grounded?14:3.5;p.vx-=p.vx*Math.min(1,f*dt);}
 // belt
 let onBelt=0;
 if(p.grounded){const bcy=Math.floor((p.y+p.h+2)/T);for(const bcx of [Math.floor((p.x+4)/T),Math.floor((p.x+p.w-4)/T)]){const ch=grid[bcy]&&grid[bcy][bcx];if(ch==='>')onBelt=1;else if(ch=='<')onBelt=-1;}}
 if(onBelt)p.vx+=onBelt*1500*dt;
 p.vx=clamp(p.vx,-460,460);
 if(Math.abs(p.vx)>320&&dir===0&&!onBelt)p.vx*= (1-Math.min(1,6*dt));
 // jump
 p.buf-=dt;p.coyote-=dt;
 if(jumpBuf>0&&(p.grounded||p.coyote>0)){p.vy=-745;p.grounded=false;p.coyote=0;jumpBuf=0;sfx.jump();burst(p.x+p.w/2,p.y+p.h,6,['#9fb7c8',PHASES[p.phase].main],120,.35,2.4,300);}
 // gravity
 p.vy=Math.min(800,p.vy+2350*dt);
 // X axis
 p.x+=p.vx*dt;
 {const t=Math.floor(p.y/T),b=Math.floor((p.y+p.h-.01)/T);
  if(p.vx>0){const c=Math.floor((p.x+p.w-.01)/T);for(let r=t;r<=b;r++)if(solidAt(c,r)){p.x=c*T-p.w;p.vx=0;break;}}
  else if(p.vx<0){const c=Math.floor(p.x/T);for(let r=t;r<=b;r++)if(solidAt(c,r)){p.x=(c+1)*T;p.vx=0;break;}}}
 // Y axis
 const wasAir=!p.grounded,prevVy=p.vy;
 p.y+=p.vy*dt;let landed=false;
 {const l=Math.floor(p.x/T),r=Math.floor((p.x+p.w-.01)/T);
  if(p.vy>0){const c=Math.floor((p.y+p.h-.01)/T);for(let c2=l;c2<=r;c2++)if(solidAt(c2,c)){p.y=c*T-p.h;landed=true;break;}}
  else if(p.vy<0){const c=Math.floor(p.y/T);for(let c2=l;c2<=r;c2++)if(solidAt(c2,c)){p.y=(c+1)*T;p.vy=0;break;}}}
 // ground probe (keeps grounded when walking)
 let groundedNow=landed;
 if(!groundedNow&&prevVy>=0){const py=p.y+p.h+1.5,l=Math.floor(p.x/T),r=Math.floor((p.x+p.w-.01)/T),c=Math.floor(py/T);for(let c2=l;c2<=r;c2++)if(solidAt(c2,c)){groundedNow=true;break;}}

  p.squash=clamp(prevVy/1500,.12,.5);
  if(prevVy>430){sfx.land(prevVy);burst(p.x+p.w/2,p.y+p.h,Math.min(12,4+prevVy/90),['#9fb7c8',PHASES[p.phase].main],prevVy*.3,.4,2.6,200);shake=Math.max(shake,clamp(prevVy/170,1,6));}
 }

function updateEntities(dt){

  if(!p.alive){pl.was=false;continue;}
  const box={x:pl.cx*T+6,y:pl.cy*T+14,w:T-12,h:T-10};
  const over=overlap(p,box);
  if(over&&!pl.was){pl.latch=!pl.latch;sfx.plate(pl.latch);const ph=PHASES[pl.color];ring(pl.cx*T+T/2,pl.cy*T+T/2,ph.main,34);burst(pl.cx*T+T/2,pl.cy*T+T/2,10,[ph.main,ph.hi],180,.5,3);}
  pl.was=over;
 }

  d.target=plates.some(p=>p.color===d.color&&p.latch)?1:0;
  const prev=d.openT;d.openT+=clamp(d.target-d.openT,-3.2*dt,3.2*dt);
  if(prev<.5&&d.openT>=.5)sfx.door();
 }
 for(const s of shards){
  if(s.got||!p.alive)continue;
  const sx=s.cx*T+T/2,sy=s.cy*T+T/2+Math.sin(time*2+s.seed)*3;
  const dx=p.x+p.w/2-sx,dy=p.y+p.h/2-sy;
  if(dx*dx+dy*dy<400){s.got=true;shardsGot++;sfx.shard();burst(sx,sy,16,['#ffffff','#ffe9b0','#8feaff','#c5f78e'],300,.6,3);ring(sx,sy,'#ffffff',38);}
 }
 // spikes
 if(p.alive)for(let cy=0;cy<HL;cy++)for(let cx=0;cx<WL;cx++){
  if(grid[cy][cx]==='^'){const box={x:cx*T+6,y:cy*T+10,w:T-12,h:T-10};if(overlap(p,box))die();}
 }
 // exit
 if(p.alive&&!clearing&&state==='play'){
  const box={x:exit.cx*T+3,y:exit.cy*T+3,w:T-6,h:T-6};
  if(overlap(p,box))beginClear();
 }
}

function drawTileBody(g,ox,oy,s,key){
 const p=PALS[key],r=Math.max(3,s*.15);
 g.fillStyle=p.glow;rr(g,ox-2,oy-2,s+4,s+4,r+2);g.fill();
 const gr=g.createLinearGradient(0,oy,0,oy+s);gr.addColorStop(0,p.hi);gr.addColorStop(.3,p.mid);gr.addColorStop(1,p.lo);
 g.fillStyle=gr;rr(g,ox+.5,oy+.5,s-1,s-1,r);g.fill();
 g.globalAlpha=.95;g.fillStyle=p.hi;g.fillRect(ox+2,oy+2,s-4,Math.max(2,s*.09));g.globalAlpha=1;
 g.fillStyle='rgba(0,0,0,.28)';g.fillRect(ox+2,oy+s-Math.max(3,s*.12),s-4,Math.max(2,s*.1)-1);
 g.strokeStyle='rgba(0,0,0,.35)';g.lineWidth=1;rr(g,ox+.5,oy+.5,s-1,s-1,r);g.stroke();
 if(p.motif==='stone'){g.fillStyle='rgba(255,255,255,.12)';g.fillRect(ox+s*.3,oy+s*.45,2.5,2.5);g.fillRect(ox+s*.68,oy+s*.62,2,2);}
 else if(p.motif==='solar'){g.strokeStyle='rgba(255,255,255,.28)';g.lineWidth=1.5;g.beginPath();g.moveTo(ox+s*.2,oy+s*.7);g.lineTo(ox+s*.42,oy+s*.32);g.lineTo(ox+s*.6,oy+s*.62);g.stroke();}
 else if(p.motif==='glace'){g.strokeStyle='rgba(255,255,255,.3)';g.lineWidth=1.5;g.beginPath();g.moveTo(ox+s*.18,oy+s*.5);g.lineTo(ox+s*.5,oy+s*.42);g.lineTo(ox+s*.78,oy+s*.55);g.stroke();}
 else{g.strokeStyle='rgba(255,255,255,.3)';g.lineWidth=1.5;g.beginPath();g.moveTo(ox+s*.3,oy+s*.62);g.quadraticCurveTo(ox+s*.5,oy+s*.28,ox+s*.72,oy+s*.5);g.stroke();}
}

let attractT=0;
function update(dt){
 time+=dt;shake-=shake*8*dt;if(shake<.05)shake=0;flash=Math.max(0,flash-dt*1.6);
 for(let i=particles.length-1;i>=0;i--){const p=particles[i];p.age+=dt;if(p.age>=p.life){particles.splice(i,1);continue;}p.vy+=p.g*dt;p.x+=p.vx*dt;p.y+=p.vy*dt;}
 for(let i=rings.length-1;i>=0;i--){rings[i].t+=dt;if(rings[i].t>=rings[i].dur)rings.splice(i,1);}
 if(state==='title'){
  attractT+=dt;
  if(attractT>1.8){attractT=0;setPhase((player.phase+1)%3);}
  player.blinkT-=dt;if(player.blinkT<=0){player.blinking=.12;player.blinkT=3;}if(player.blinking>0)player.blinking-=dt;
  const tx=spawn.x+16-viewW*.5+Math.sin(time*.25)*40,ty=spawn.y-viewH*.5+Math.cos(time*.2)*12;
  camX+=(clamp(tx,0,Math.max(0,WL*T-viewW))-camX)*Math.min(1,3*dt);
  camY+=(clamp(ty,(HL*T-viewH)/2,Math.max(0,HL*T-viewH))-camY)*Math.min(1,3*dt);
  updateHUD();return;
 }
 if(state!=='play')return;
 if(clearing){
  clearing.t+=dt/.65;const e=clearing.t>=1?1:1-Math.pow(1-clearing.t,2.2);
  player.x=lerp(clearing.sx,exit.cx*T+T/2-player.w/2,e);player.y=lerp(clearing.sy,exit.cy*T+T/2-player.h/2,e);
  burst(exit.cx*T+T/2,exit.cy*T+T/2,1,['#ffe9b0','#ffffff'],80,.4,2,-50);
  if(clearing.t>=1)finishLevel();
 }else{
  levelTime+=dt;jumpBuf=Math.max(0,jumpBuf-dt*10);
  updatePlayer(dt);updateEntities(dt);
 }
 // camera
 const tx=player.x+player.w/2-viewW*.45,ty=player.y+player.h/2-viewH*.5;
 camX+=(clamp(tx,(WL*T-viewW)/2,Math.max(0,WL*T-viewW))-camX)*Math.min(1,6*dt);
 camY+=(clamp(ty,(HL*T-viewH)/2,Math.max(0,HL*T-viewH))-camY)*Math.min(1,6*dt);
 updateHUD();
}

function drawBG(){
 ctx.fillStyle=bgGrad;ctx.fillRect(0,0,W,H);
 // ambient glows
 const gl=[['#ffb703',.28],['#2fc6ec',.52],['#78c842',.78]];
 for(let i=0;i<3;i++){const gx=W*gl[i][1]-camX*.1+Math.sin(time*.3+i*2)*60,gy=H*.35+Math.cos(time*.22+i)*40;
  const g=ctx.createRadialGradient(gx,gy,0,gx,gy,W*.4);g.addColorStop(0,gl[i][0]+'14');g.addColorStop(1,gl[i][0]+'00');
  ctx.fillStyle=g;ctx.fillRect(0,0,W,H);}
 // dot grid
 const off=( -camX*zoom*.12)%72;ctx.fillStyle='rgba(160,210,235,.045)';
 for(let x=off;x<W;x+=72)for(let y=(( -camY*zoom*.12)%72);y<H;y+=72)ctx.fillRect(x,y,2,2);
 // silhouettes
 for(const s of sils){const sx=s.x-camX*.45,sy=s.y-camY*.3;
  if(sx+s.w<0||sx>W)continue;
  ctx.fillStyle='#0c2230';ctx.fillRect(sx,sy,s.w,s.h);
  ctx.fillStyle=['rgba(255,183,3,.14)','rgba(47,198,236,.14)','rgba(120,200,66,.14)'][s.c];ctx.fillRect(sx,sy,s.w,2);}
 // motes
 for(const m of motes){const span=viewW+140;let sx=((m.x*span-camX*.5)%span+span)%span-70;let sy=((m.y*(viewH+140)+time*m.s*26)%(viewH+140))-70;
  ctx.globalAlpha=.05+m.r*.03;ctx.fillStyle='#bfe2f2';ctx.beginPath();ctx.arc(sx,sy,m.r,0,TAU);ctx.fill();}
 ctx.globalAlpha=1;
}
function drawWorld(){
 const x0=clamp(Math.floor(camX/T)-1,0,WL-1),x1=clamp(Math.ceil((camX+viewW)/T)+1,0,WL-1);
 const y0=clamp(Math.floor(camY/T)-1,0,HL-1),y1=clamp(Math.ceil((camY+viewH)/T)+1,0,HL-1);
 for(let cy=y0;cy<=y1;cy++)for(let cx=x0;cx<=x1;cx++){
  const ch=grid[cy][cx];if(ch==='.'||ch==='S'||ch==='*')continue;
  const x=cx*T,y=cy*T;
  if(ch==='X'||ch==='A'||ch==='B'||ch==='C'){const s=tileSpr[ch];ctx.drawImage(s.c,x-s.pad,y-s.pad);}
  else if(SOFT.hasOwnProperty(ch)){
   if(player.phase===SOFT[ch]){const s=tileSpr[PHASES[SOFT[ch]].key];ctx.drawImage(s.c,x-s.pad,y-s.pad);}
   else{const p=PALS[PHASES[SOFT[ch]].key];ctx.save();ctx.globalAlpha=.06;ctx.fillStyle=p.mid;rr(ctx,x+2,y+2,T-4,T-4,5);ctx.fill();
    ctx.globalAlpha=.38;ctx.strokeStyle=p.mid;ctx.setLineDash([5,4]);ctx.lineWidth=1.5;rr(ctx,x+2.5,y+2.5,T-5,T-5,5);ctx.stroke();ctx.restore();}
  }
  else if(ch==='>'||ch==='<'){
   ctx.fillStyle='#152530';rr(ctx,x+.5,y+.5,T-1,T-1,4);ctx.fill();
   ctx.fillStyle='#2c4557';ctx.fillRect(x+2,y+2,T-4,2.5);
   const dir=ch==='>'?1:-1;
   ctx.strokeStyle='rgba(200,228,242,.75)';ctx.lineWidth=2;ctx.lineCap='round';
   for(let k=0;k<3;k++){const cxp=x+((time*55*dir)%12+12*k)%T;
    ctx.beginPath();ctx.moveTo(cxp-4*dir,y+11);ctx.lineTo(cxp+3*dir,y+16);ctx.lineTo(cxp-4*dir,y+21);ctx.stroke();}
  }
  else if(ch==='^'){
   ctx.fillStyle='rgba(255,90,90,.1)';ctx.fillRect(x,y+T-3,T,3);
   for(let k=0;k<4;k++){const x0=x+k*T/4;
    const g=ctx.createLinearGradient(0,y+6,0,y+T);g.addColorStop(0,'#e8f1f5');g.addColorStop(1,'#71828d');
    ctx.fillStyle=g;ctx.beginPath();ctx.moveTo(x0+1,y+T-1);ctx.lineTo(x0+T/8,y+6);ctx.lineTo(x0+T/4-1,y+T-1);ctx.closePath();ctx.fill();}
  }
  else if('opq'.includes(ch)){
   const pl=plateMap[cx*16+cy];if(!pl)continue;const ph=PHASES[pl.color],on=pl.latch;
   ctx.fillStyle='#33454f';rr(ctx,x+T/2-2,y+T-10,4,9,2);ctx.fill();
   if(on){ctx.strokeStyle=ph.main;ctx.globalAlpha=.5+.3*Math.sin(time*6);ctx.lineWidth=2;ctx.beginPath();ctx.arc(x+T/2,y+T-8,9+2*Math.sin(time*6),0,TAU);ctx.stroke();ctx.globalAlpha=1;}
   ctx.fillStyle=on?ph.main:'#2a3b46';ctx.beginPath();ctx.ellipse(x+T/2,y+T-9,T/2-5,4.5,0,0,TAU);ctx.fill();
   ctx.strokeStyle='rgba(0,0,0,.4)';ctx.lineWidth=1;ctx.stroke();
   ctx.fillStyle=on?ph.hi:'#4a5c68';ctx.beginPath();ctx.arc(x+T/2,y+T-9,1.8,0,TAU);ctx.fill();
  }
  else if('OPQ'.includes(ch)){
   const d=doorMap[cx*16+cy];if(!d)continue;const ph=PALS[PHASES[d.color].key],h=T*(1-d.openT*.97);
   if(h>1){ctx.save();ctx.globalAlpha=1-d.openT*.65;
    const g=ctx.createLinearGradient(0,y,0,y+h);g.addColorStop(0,ph.lo);g.addColorStop(1,ph.mid);
    ctx.fillStyle=g;rr(ctx,x+3,y+1,h?T-6:1,h,3);ctx.fill();
    ctx.fillStyle=ph.hi;ctx.globalAlpha=(1-d.openT)*.9;ctx.fillRect(x+3,y+h-2,T-6,2);
    ctx.restore();}
   else{ctx.strokeStyle='rgba(160,210,235,.12)';ctx.setLineDash([4,5]);ctx.beginPath();ctx.moveTo(x+T/2,y+2);ctx.lineTo(x+T/2,y+T-2);ctx.stroke();ctx.setLineDash([]);}
  }
  else if(ch==='*'){
   const s=shards.find(s=>s.cx===cx&&s.cy===cy);if(!s||s.got)continue;
   const sx=x+T/2,sy=y+T/2+Math.sin(time*2+s.seed)*3,rot=time*1.6+s.seed;
   ctx.save();ctx.translate(sx,sy);ctx.rotate(rot);
   ctx.globalAlpha=.25+.15*Math.sin(time*4+s.seed);ctx.fillStyle='#fff';
   ctx.beginPath();ctx.moveTo(0,-13);ctx.lineTo(9,0);ctx.lineTo(0,13);ctx.lineTo(-9,0);ctx.closePath();ctx.fill();
   ctx.globalAlpha=1;const hue=(time*60+s.seed*80)%360;
   ctx.fillStyle='#ffffff';ctx.beginPath();ctx.moveTo(0,-8);ctx.lineTo(5.5,0);ctx.lineTo(0,8);ctx.lineTo(-5.5,0);ctx.closePath();ctx.fill();
   ctx.strokeStyle=`hsl(${hue} 90% 70%)`;ctx.lineWidth=2;ctx.stroke();ctx.restore();
  }
  else if(ch==='E'){
   const ex=x+T/2,ey=y+T/2;
   for(const [rad,spd,col,a] of [[13,1.4,'#ffe9b0',.8],[9,-2.1,'#8feaff',.7]]){
    ctx.save();ctx.translate(ex,ey);ctx.rotate(time*spd);ctx.strokeStyle=col;ctx.globalAlpha=a;ctx.lineWidth=2.5;
    for(let k=0;k<3;k++){ctx.beginPath();ctx.arc(0,0,rad,k*TAU/3,k*TAU/3+1.6);ctx.stroke();}ctx.restore();}
   const g=ctx.createRadialGradient(ex,ey,0,ex,ey,12);g.addColorStop(0,'rgba(255,247,224,'+(.9+.1*Math.sin(time*5))+')');g.addColorStop(1,'rgba(255,247,224,0)');
   ctx.fillStyle=g;ctx.beginPath();ctx.arc(ex,ey,12,0,TAU);ctx.fill();
   for(let k=0;k<3;k++){const a=time*2+k*TAU/3;ctx.fillStyle='#fff';ctx.globalAlpha=.8;ctx.beginPath();ctx.arc(ex+Math.cos(a)*15,ey+Math.sin(a)*15,1.6,0,TAU);ctx.fill();ctx.globalAlpha=1;}
  }
  else if(ch==='S'){
   ctx.save();ctx.globalAlpha=.14+.08*Math.sin(time*3);ctx.strokeStyle='#bfe2f2';ctx.setLineDash([4,4]);ctx.lineWidth=1.5;rr(ctx,x+5,y+5,T-10,T-10,5);ctx.stroke();ctx.restore();
  }
 }
 // trail
 for(const tr of trail){const ph=PHASES[tr.ph];ctx.globalAlpha=Math.max(0,tr.a)*.3;ctx.fillStyle=ph.main;rr(ctx,tr.x+2,tr.y+2,player.w-4,player.h-4,5);ctx.fill();}

 // rings

 ctx.globalAlpha=1;
 // player
 if(player.alive)drawPlayer();
 // particles

  if(p.type==='r')ctx.fillRect(p.x-p.size/2,p.y-p.size/2,p.size,p.size);
  else{ctx.beginPath();ctx.arc(p.x,p.y,p.size*a+0.4,0,TAU);ctx.fill();}}

function drawPlayer(){
 const p=player,ph=PHASES[p.phase],cx=p.x+p.w/2,cyB=p.y+p.h;
 let sc=1,rot=0;
 if(clearing){sc=Math.max(.05,1-clearing.t);rot=clearing.t*4;}
 const sq=p.squash,sx=sc*(1+sq*.9),sy=sc*(1-sq);
 ctx.save();ctx.translate(cx,cyB);ctx.rotate(rot);ctx.scale(sx,sy);
 ctx.globalAlpha=.2;ctx.fillStyle=ph.main;rr(ctx,-p.w/2-4,-p.h-4,p.w+8,p.h+8,9);ctx.fill();
 ctx.globalAlpha=1;
 const g=ctx.createLinearGradient(0,-p.h,0,0);g.addColorStop(0,ph.hi);g.addColorStop(.35,ph.main);g.addColorStop(1,ph.lo);
 ctx.fillStyle=g;rr(ctx,-p.w/2,-p.h,p.w,p.h,6);ctx.fill();
 ctx.strokeStyle='rgba(0,0,0,.4)';ctx.lineWidth=1.5;rr(ctx,-p.w/2,-p.h,p.w,p.h,6);ctx.stroke();
 // eyes
 const look=clamp(p.vx/300,-1,1)*2+p.facing*1.5,bl=p.blinking>0?.15:1;

  ctx.save();ctx.translate(e+look*.7,-p.h*.58);ctx.scale(1,bl);
  ctx.fillStyle='#fff';ctx.beginPath();ctx.arc(0,0,3.6,0,TAU);ctx.fill();
  ctx.fillStyle='#12242e';ctx.beginPath();ctx.arc(look*.55,.3,1.8,0,TAU);ctx.fill();ctx.restore();}
 ctx.restore();
}

}
