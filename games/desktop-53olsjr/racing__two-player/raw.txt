  if(c.offroad&&speed>90)spawn('dust',c.x-fx*14+(Math.random()-.5)*10,c.y-fy*14+(Math.random()-.5)*10,
    -fx*60+(Math.random()-.5)*70,-fy*60+(Math.random()-.5)*70,.5,3+Math.random()*3);
  if(c.skid&&!c.offroad&&Math.abs(vl)>130)spawn('smoke',c.x-fx*12,c.y-fy*12,
    (Math.random()-.5)*40,(Math.random()-.5)*40,.4,3+Math.random()*3);
  if(c.boosting&&speed>40){for(let k=0;k<2;k++)spawn('flame',c.x-fx*19+(Math.random()-.5)*6,c.y-fy*19+(Math.random()-.5)*6,
    -fx*230+(Math.random()-.5)*60,-fy*230+(Math.random()-.5)*60,.28,4+Math.random()*4);}
}

function collide(){
  const a=cars[0],b=cars[1];
  let dx=b.x-a.x,dy=b.y-a.y,d=Math.hypot(dx,dy);const minD=30;
  if(d<minD&&d>.001){
    const nx=dx/d,ny=dy/d,ov=(minD-d)/2;
    a.x-=nx*ov;a.y-=ny*ov;b.x+=nx*ov;b.y+=ny*ov;
    let dvx=a.vx-b.vx,dvy=a.vy-b.vy;const dot=dvx*nx+dvy*ny;
    if(dot>0){const j=dot*.68;
      a.vx-=nx*j;a.vy-=ny*j;b.vx+=nx*j;b.vy+=ny*j;
      const mx=(a.x+b.x)/2,my=(a.y+b.y)/2;
      for(let k=0;k<9;k++){const an=Math.random()*7,sp=120+Math.random()*220;
        spawn('spark',mx,my,Math.cos(an)*sp,Math.sin(an)*sp,.32,2.5);}
      shake=Math.min(7,shake+4);
      if(T-lastThud>.12){lastThud=T;thud();}
    }
  }
}

function spawn(type,x,y,vx,vy,life,size){
  if(parts.length>420)parts.splice(0,parts.length-420);
  parts.push({type,x,y,vx,vy,life,max:life,size,rot:Math.random()*7,vr:(Math.random()-.5)*10,
    col:type==='confetti'?CONF[Math.random()*CONF.length|0]:null});
}

function addSkid(w1,w2){
  if(skids.length>700)skids.splice(0,skids.length-700);
  skids.push({x1:w1.x,y1:w1.y,x2:w2.x,y2:w2.y,l:1});
}
function updateFx(dt){
  for(let i=parts.length-1;i>=0;i--){const p=parts[i];p.life-=dt;
    if(p.life<=0){parts.splice(i,1);continue;}
    if(p.type==='confetti'){p.vy+=520*dt;p.vx*=.99;}else{p.vx*=Math.pow(.02,dt);p.vy*=Math.pow(.02,dt);}
    p.x+=p.vx*dt;p.y+=p.vy*dt;p.rot+=p.vr*dt;}
  for(let i=skids.length-1;i>=0;i--){skids[i].l-=dt/4;if(skids[i].l<=0)skids.splice(i,1);}
  shake*=Math.max(0,1-6*dt);
}
