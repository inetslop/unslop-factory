function damageBunker(bk,cc,rr){

function pbStep(b,dt){
  b.prevY=b.y;b.y-=900*dt;
  const cross=top=>b.prevY>top&&b.y<=top;
  let best=null;
  if(G.ufo&&Math.abs(b.x-G.ufo.x)<30&&cross(G.ufo.y-10))best={t:G.ufo.y-10,k:'ufo'};
  const F=G.F;
  for(let j=0;j<F.rows;j++)for(let i=0;i<F.cols;i++){
    if(!F.alive[j][i])continue;
    const cx=(F.ox+i+0.5)*CELL;if(Math.abs(cx-b.x)>CELL*0.48)continue;
    const top=(F.oy+j)*CELL+6;
    if(cross(top)&&(!best||top<best.t))best={t:top,k:'alien',i,j};
  }
  for(const bk of G.shields){
    if(b.x<bk.left||b.x>bk.left+72)continue;
    const sc=Math.floor((b.x-bk.left)/8);let r=-1;
    for(let rr=0;rr<7;rr++)if(bk.cells[rr][sc]){r=rr;break;}
    if(r>=0){const top=BY+r*8;if(cross(top)&&(!best||top<best.t))best={t:top,k:'bunk',bk,sc,r};}
  }
  if(best){
    if(best.k==='ufo'){
      const bonus=[50,100,150,300][(Math.random()*4)|0];
      G.score+=bonus;G.hits++;
      popup(G.ufo.x,G.ufo.y,'MYSTERY +'+bonus,'#ff7ad9');
      burst(G.ufo.x,G.ufo.y,'#ff7ad9',22);sfx.ufoKill();
      setMsg('MYSTERY SIGNAL +'+bonus,'#ff7ad9');G.ufo=null;return false;
    }
    if(best.k==='alien'){killAlien(best.i,best.j);G.hits++;if(!b.rail)return false;}
    if(best.k==='bunk'){
      const n=damageBunker(best.bk,best.sc,best.r);
      burst(b.x,BY+best.r*8+4,'#3fe0b0',10+n);sfx.shield();setMsg('BUNKER DAMAGED','#3fe0b0');return false;
    }
  }
  if(b.y<12){burst(b.x,12,'#ffe9a8',5);return false;}
  return true;
}
function ebStep(b,dt){
  b.prevY=b.y;b.y+=560*dt;
  for(const bk of G.shields){
    if(b.x<bk.left||b.x>bk.left+72)continue;
    const sc=Math.floor((b.x-bk.left)/8);let r=-1;
    for(let rr=6;rr>=0;rr--)if(bk.cells[rr][sc]){r=rr;break;}
    if(r>=0){const blockY=BY+(r+1)*8;
      if(b.prevY<=blockY&&b.y>blockY){
        const n=damageBunker(bk,sc,r);burst(b.x,blockY,'#3fe0b0',10+n);sfx.shield();
        setMsg('BUNKER DAMAGED','#3fe0b0');return false;
      }}
  }
  const cx=(G.player.col+0.5)*CELL,bandTop=CANNON_ROW*CELL+8;
  if(Math.abs(b.x-cx)<CELL*0.46&&b.prevY<=bandTop&&b.y>bandTop){
    G.hull--;G.lastHurt=G.t;G.shake=12;
    burst(cx,bandTop+10,'#ff8a5c',26);popup(cx,bandTop-14,'-1 HULL','#ff5d73');
    sfx.hullHit();setMsg('CANNON HIT — HULL '+Math.max(0,G.hull),'#ff5d73');refreshHUD();
    if(G.hull<=0)gameover('hull');
    return false;
  }
  if(b.y>=GY){burst(b.x,GY,'#ff5d73',8);return false;}
  return true;
}

/* ================= UPDATE ================= */
function update(dt){
  G.t+=dt;
  if(G.timers.length){
    const due=G.timers.filter(tm=>tm.at<=G.t);
    G.timers=G.timers.filter(tm=>tm.at>G.t);
    due.forEach(tm=>tm.fn());
  }
  for(const s of stars){s.y+=s.sp*dt;if(s.y>view.h+2){s.y=-2;s.x=Math.random()*view.w;}}
  G.shake=Math.max(0,G.shake-dt*34);

  if(G.state==='player'){
    if(G.ap===0&&!G.autoEndAt)G.autoEndAt=G.t+(G.pB.length?1.15:0.7);
    if(G.autoEndAt&&G.t>=G.autoEndAt){G.autoEndAt=0;endPlayerTurn();}
    if(G.ufo){const u=G.ufo;u.x+=u.dir*95*dt;u.blip+=dt;
      if(u.blip>0.55){u.blip=0;sfx.ufoBlip();}
      if((u.dir>0&&u.x>LW+40)||(u.dir<0&&u.x<-40))G.ufo=null;}
  }else if(G.state==='enemy'){
    const e=G.enemy;
    if(e.stage==='tick'&&G.tickAnim&&G.t>=G.tickAnim.t0+G.tickAnim.dur)e.stage='shoot';
    else if(e.stage==='shoot'&&!G.eB.length&&G.t>=e.waitUntil){
      if(e.idx<e.shots.length)fireEnemy(e.shots[e.idx++]);else startPlayerTurn();
    }
  }

  let ebHit=false;
  G.pB=G.pB.filter(b=>pbStep(b,dt));
  G.eB=G.eB.filter(b=>{const k=ebStep(b,dt);if(!k)ebHit=true;return k;});
  if(ebHit&&G.state==='enemy')G.enemy.waitUntil=G.t+0.2;

  G.parts=G.parts.filter(p=>{p.x+=p.vx*dt;p.y+=p.vy*dt;p.vx*=0.985;return G.t-p.t0<p.life;});
  G.pops=G.pops.filter(p=>G.t-p.t0<1);
}

**Starfall Protocol** is a fully turn-based take on Space Invaders:

- **Your turn (3 AP):** MOVE costs 1, FIRE costs 2. Reposition behind bunkers, snipe the mystery UFO, or save your bolt — then the invasion steps in and fires back while you're locked in place.
- **Deterministic enemy fire:** each round's bolts are seeded, so the **Decryption** upgrade (earned at the Depot between waves) reveals exactly which lanes will shoot next turn.
- **Waves & upgrades:** formation grows wider and descends faster; clear a sector to pick one of three Depot upgrades (+Hull, +Action, Bunker Repair, Railbolt pierce, Decryption, Supply Drop).
- **Classic texture:** 2-frame pixel invaders in three ranks, eroding bunkers, screen shake, particle bursts, score popups, parallax nebula starfield, CRT scanlines, and full Web Audio SFX (M to mute). Best score persists via localStorage.
