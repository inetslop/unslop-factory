/* ================= UPDATE ================= */
function updBalls(dt){
  for(const b of balls){
    if(b.dead)continue;
    b.rel+=b.dir*b.speed*dt;
    if(b.rel<BR){b.rel=BR;b.dir=1;wallHit(b,0);}
    else if(b.rel>FW-BR){b.rel=FW-BR;b.dir=-1;wallHit(b,1);}
    b.trail.push(b.rel);if(b.trail.length>9)b.trail.shift();
    if(b.squash>0)b.squash=Math.max(0,b.squash-dt*6);
    if(b.hitCd>0)b.hitCd-=dt;
    if(b.flame){
      b.flameT-=dt;
      if(Math.random()<dt*28)parts.push({x:b.rel-b.dir*14,y:b.ly,vx:-b.dir*(30+Math.random()*50),vy:(Math.random()-.5)*60,g:0,
        life:.4,max:.4,size:2.2,col:Math.random()<.5?'#ff9d3a':'#ff5533',type:'dot',rot:0,vr:0,ph:0});
      if(b.flameT<=0){b.flame=false;b.bounces=0;b.speed=b.cap/2.6;popup(b.rel,b.ly-22,'COOLED','#7deaff',12);sfx.cool();burst(b.rel,b.ly,8,['#7deaff'],90,2,0);}
    }
  }
  balls=balls.filter(b=>!b.dead);
}

function updFrog(dt){

    frog.hopT+=dt*1000;const t=Math.min(1,frog.hopT/HOP_MS);
    frog.x=frog.fc.x+(frog.tc.x-frog.fc.x)*t;

    if(t>=1){
      frog.col=frog.nc;frog.row=frog.nr;const p=cellXY(frog.col,frog.row);frog.x=p.x;frog.y=p.y;
      frog.state='idle';lastHopEnd=tGlobal;
      if(frog.row===0)doGoal();
    }
  } else {const p=cellXY(frog.col,frog.row);frog.x=p.x;frog.y=p.y;}
}
function checkCollisions(){
  if(!frog.alive||frog.invuln>0)return;
  for(const b of balls){
    if(b.hitCd>0)continue;
    const dx=frog.x-b.rel,dy=frog.y-b.ly;
    if(dx*dx+dy*dy<27*27){
      if(b.flame){b.dead=true;burst(b.rel,b.ly,16,['#ff9d3a','#ff5533'],170,3,80);die('burn');return;}
      else smack(b);
    }
  }
  if(frog.state==='idle'&&frog.row===1){
    let on=false;for(const p of pads)if(Math.abs(frog.x-p.rel)<PAD_R+12){on=true;break;}
    if(!on)die('drown');
  }
}
function update(dt){
  for(const f of flies){f.x+=Math.sin(tGlobal*.35+f.ph)*.014*dt*60/60;f.y+=Math.cos(tGlobal*.29+f.ph*1.7)*.012;}
  for(const f of flies){f.x=(f.x%1+1)%1;f.y=(f.y%1+1)%1;}
  for(const w of wallFx)w.t+=dt;wallFx=wallFx.filter(w=>w.t<.45);
  for(const p of parts){p.life-=dt;p.x+=p.vx*dt;p.y+=p.vy*dt;p.vy+=(p.g||0)*dt;p.rot+=(p.vr||0)*dt;
    if(p.type==='petal')p.x+=Math.sin(tGlobal*3+p.ph)*24*dt;}
  parts=parts.filter(p=>p.life>0);
  for(const p of pops){p.life-=dt;p.y+=p.vy*dt;}
  pops=pops.filter(p=>p.life>0);
  if(flashA>0)flashA=Math.max(0,flashA-dt*1.8);
  if(shakeT>0)shakeT=Math.max(0,shakeT-dt);

  if(state==='attract'){updBalls(dt);updPads(dt);return;}
  updPads(dt);

  if(state==='playing'){
    timer-=dt;
    if(timer<=10.5&&Math.ceil(timer)!==lastTick){lastTick=Math.ceil(timer);sfx.tick();}
    hudMid.classList.toggle('low',timer<=10);
    if(timer<=0){timer=0;die('time');}
    if(frog.alive&&frog.state==='idle'&&tGlobal-lastHopEnd>0.045){
      const d=pendingDir||(heldOrder.length?heldOrder[heldOrder.length-1].d:null);
      if(d)tryHop(d[0],d[1]);
    }
    pendingDir=null;
    if(frog.invuln>0)frog.invuln-=dt;
    updFrog(dt);updBalls(dt);checkCollisions();
  }
  else if(state==='dying'){
    deadT+=dt;updBalls(dt);
    if(deadT>0.95){if(lives<=0)doGameOver();else respawn();}
  }
  else if(state==='goal'){
    goalT+=dt;updBalls(dt);
    if(goalT>1.1){wave++;resetLines();timer=TIME;lastTick=TIME+1;
      frog.col=HOME.col;frog.row=HOME.row;const p=cellXY(frog.col,frog.row);frog.x=p.x;frog.y=p.y;
      frog.state='idle';frog.invuln=1;state='playing';setWave();setComboUI();
      popup(FW/2,CH*.5,'WAVE '+wave,'#7deaff',30,true);}
  }
  else if(state==='gameover'){
    updBalls(dt);overT+=dt;
    if(overT>0.65&&!overShown){overShown=true;
      $('finalScore').textContent=score;$('stWave').textContent=wave;$('stCombo').textContent='×'+maxCombo;$('stPong').textContent=ponged;
      $('newHi').classList.toggle('hidden',!(score>=hi&&score>0));
      overOv.classList.remove('hidden');}
  }
  // HUD timer
  if(state==='playing'||state==='goal'){
    if(Math.ceil(timer)!==lastTShown){lastTShown=Math.ceil(timer);timeEl.textContent=Math.max(0,lastTShown);}
    timebarEl.style.width=(Math.max(0,timer)/TIME*100)+'%';
  }
}

function drawFrog(){

  }else{
    if(frog.state==='hop'){
      const t=Math.min(1,frog.hopT/HOP_MS);
      x=frog.fc.x+(frog.tc.x-frog.fc.x)*t;y=frog.fc.y+(frog.tc.y-frog.fc.y)*t;lift=Math.sin(Math.PI*t)*14;
      if(Math.abs(frog.face.y)>=Math.abs(frog.face.x)){sx=.9;sy=1.12;}else{sx=1.14;sy=.9;}
