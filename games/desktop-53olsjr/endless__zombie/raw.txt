  if(zombies.length>46)return;
  const type=force||pickType(),T=TYPES[type],m=T.r+8;
  const mot=[];for(let i=0;i<3;i++)mot.push({x:rand(-T.r*0.5,T.r*0.5),y:rand(-T.r*0.6,T.r*0.6),r:rand(2,4)});
  zombies.push({type:type,x:rand(roadL+m,roadR-m),y:-T.r-10,r:T.r,hp:T.hp,vy:rand(T.spd[0],T.spd[1])*(1+(wave-1)*0.05),phase:rand(0,TAU),wob:rand(0.5,1.4),biteCd:0,hitF:0,kb:0,seed:Math.random(),mot:mot,dead:false});
}
function spawnPickup(){
  const r=Math.random();const kind=r<0.45?'ammo':(r<0.8?'med':'double');
  pickups.push({kind:kind,x:rand(roadL+24,roadR-24),y:-24,vy:70,ph:rand(0,6)});
}
function damageZ(z,dmg){
  if(z.dead)return;
  z.hp-=dmg;z.hitF=1;z.kb=Math.min(260,(z.kb||0)+90);
  burst(z.x,z.y-4,4,['#7e2430','#5c3a2a','#6b7f3e']);
  if(sHitT<=0){sfx.hitZ();sHitT=0.06;}
  if(z.hp<=0)killZombie(z);
}
function killZombie(z){
  if(z.dead)return;z.dead=true;kills++;
  const T=TYPES[z.type];killScore+=T.val;
  addText('+'+T.val,z.x,z.y,'#e8e2cf');
  burst(z.x,z.y,10+z.r,['#7e2430','#5c3a2a','#6b7f3e']);
  sfx.deadZ();
  if(z.type==='bloater')explode(z.x,z.y);
}
function explode(x,y){
  const R=85;shake=Math.min(20,shake+13);sfx.boom();
  addPart({type:'ring',x:x,y:y,r:10,vr:430,life:0.45,color:'rgba(200,255,120,0.8)'});
  for(let i=0;i<9;i++)addPart({type:'smoke',x:x+rand(-14,14),y:y+rand(-14,14),vx:rand(-30,30),vy:rand(-50,-10),size:rand(10,22),life:rand(0.6,1)});
  for(let i=0;i<10;i++){const a=rand(0,TAU),s=rand(80,260);addPart({type:'spark',x:x,y:y,vx:Math.cos(a)*s,vy:Math.sin(a)*s,size:3,color:'#ffd23e',life:rand(0.15,0.3)});}
  for(const o of zombies){
    if(o===z||o.dead)continue;
    const d=Math.hypot(o.x-x,o.y-y);
    if(d<R+o.r){o.hp-=45;o.hitF=1;o.kb=220;if(o.hp<=0)killZombie(o);}
  }
  if(state==='play'){const d=Math.hypot(player.x-x,player.y-y);if(d<R+16)damagePlayer(Math.round(28*(1-d/(R+16))));}
}
function damagePlayer(d){
  if(state!=='play')return;
  player.hp-=d;flashRed=Math.min(0.5,flashRed+0.3);shake+=7;sfx.hurt();
  burst(player.x,player.y,8,['#b03030','#7e1f1f']);
  if(player.hp<=0){player.hp=0;gameOver();}
}
function shoot(){
  player.ammo--;player.fireCd=0.42;shake=Math.min(14,shake+3);
  player.flashT=0.06;sfx.shot();
  const gx=player.x,gy=player.y-27;
  const streams=player.buff>0?[-7,7]:[0];
  for(const off of streams)for(let i=0;i<3;i++){
    const a=-Math.PI/2+(i-1)*0.13+rand(-0.03,0.03);
    pellets.push({x:gx+off,y:gy,vx:Math.cos(a)*740,vy:Math.sin(a)*740,dmg:13,life:1.2});
  }
}
function knife(){
  player.knifeCd=0.42;shake=Math.min(14,shake+1);sfx.knife();
  addPart({type:'slash',x:player.x,y:player.y,vx:0,vy:0,size:38,life:0.18});
  for(const z of zombies){
    if(z.dead)continue;
    const dy=z.y-player.y,dx=z.x-player.x;
    if(dy>-64&&dy<-6&&Math.abs(dx)<46)damageZ(z,30);
  }
}

function updatePlayer(dt){
  const p=player;let ax=0,ay=0;
  if(keys['arrowleft']||keys['a'])ax-=1;if(keys['arrowright']||keys['d'])ax+=1;
  if(keys['arrowup']||keys['w'])ay-=1;if(keys['arrowdown']||keys['s'])ay+=1;
  const k=Math.min(1,dt*9);
  p.vx+=(ax*340-p.vx)*k;p.vy+=(ay*320-p.vy)*k;
  p.x+=p.vx*dt;p.y+=p.vy*dt;clampPlayer();
  p.moving=Math.abs(p.vx)+Math.abs(p.vy)>50||!!(ax||ay);
  p.anim+=dt*(p.moving?11:3);
  p.fireCd-=dt;p.knifeCd-=dt;p.buff=Math.max(0,p.buff-dt);p.flashT=Math.max(0,p.flashT-dt);
  if((keys[' ']||pointerFire)&&p.fireCd<=0){if(p.ammo>0)shoot();else if(p.knifeCd<=0)knife();}
}
function updZombie(z,dt,playing){
  z.phase+=dt*(3+z.wob*3);
  let vx=0;
  if(playing){
    if(z.type==='sprint')vx=clamp((player.x-z.x)*1.4,-150,150);
    else if(z.type==='brute')vx=clamp((player.x-z.x)*0.25,-40,40);
    else vx=Math.sin(z.phase*0.7)*(z.type==='bloater'?14:26);
  }
  z.x=clamp(z.x+vx*dt,roadL+z.r*0.6,roadR-z.r*0.6);
  z.y+=z.vy*dt;z.y-=z.kb*dt;
  z.kb=Math.max(0,z.kb-dt*420);
  z.hitF=Math.max(0,z.hitF-dt*5);
  z.biteCd=Math.max(0,z.biteCd-dt);
  if(playing&&z.biteCd<=0){
    const dx=z.x-player.x,dy=z.y-player.y,rr2=(z.r+13)*(z.r+13);
    if(dx*dx+dy*dy<rr2){z.biteCd=0.8;z.y-=26;damagePlayer(TYPES[z.type].dmg);}
  }
}

function update(dt){
  t+=dt;
  const playing=state==='play'&&!paused;
  if(playing){
    waveT+=dt;const nw=1+Math.floor(waveT/20);
    if(nw>wave){wave=nw;onWave();}
  }
  let sp=state==='menu'?60:state==='over'?45:Math.min(250,78+(wave-1)*9);
  scrollSpd+=(sp-scrollSpd)*Math.min(1,dt*2);
  const sc=scrollSpd*dt;totScroll+=sc;
  if(playing)meters+=sc/14;

  for(const s of specks){s.y+=sc;if(s.y>H+12){s.y-=H+24;s.x=rand(0,W);}}
  for(const p of props){p.y+=sc;if(p.y>H+60){p.y=-rand(40,120);p.offF=rand(0.05,0.9);}}
  for(const f of fogs){f.y+=sc*0.45;f.x+=f.vx*dt;if(f.x<-f.r)f.x=W+f.r;if(f.x>W+f.r)f.x=-f.r;if(f.y>H+f.r)f.y=-f.r;}

  if(state==='menu'){
    spawnT-=dt;if(spawnT<=0&&zombies.length<7){spawnZombie('sham');spawnT=2.4;}
  }else if(playing){
    spawnT-=dt;
    if(spawnT<=0){
      const burst=wave>=6&&Math.random()<0.3?2:1;
      for(let i=0;i<burst;i++)spawnZombie();
      spawnT=Math.max(0.28,1.0-(wave-1)*0.075)*(0.7+Math.random()*0.6);
    }
    pickT-=dt;if(pickT<=0){spawnPickup();pickT=2.4+Math.random()*3.2;}
  }

  if(playing)updatePlayer(dt);

  for(let i=zombies.length-1;i>=0;i--){
    const z=zombies[i];
    updZombie(z,dt,playing);
    if(z.dead||z.y>H+60)zombies.splice(i,1);
  }

  for(let i=pellets.length-1;i>=0;i--){
    const p=pellets[i];p.x+=p.vx*dt;p.y+=p.vy*dt;p.life-=dt;
    let hit=false;
    for(const z of zombies){
      if(z.dead)continue;
      const dx=z.x-p.x,dy=z.y-p.y,rr2=(z.r+4)*(z.r+4);
      if(dx*dx+dy*dy<rr2){damageZ(z,p.dmg);hit=true;break;}
    }
    if(hit||p.life<=0||p.y<-40||p.x<-40||p.x>W+40)pellets.splice(i,1);
  }

  for(let i=pickups.length-1;i>=0;i--){
    const p=pickups[i];p.y+=p.vy*dt;
    if(playing){
      const dx=player.x-p.x,dy=player.y-p.y,d=Math.hypot(dx,dy);
      if(d<85&&d>1){p.x+=dx/d*430*dt;p.y+=dy/d*430*dt;}
      if(d<24){
        if(p.kind==='ammo'){player.ammo=Math.min(60,player.ammo+12);addText('+12 SHELLS',p.x,p.y,'#e0a24d');}
        else if(p.kind==='med'){player.hp=Math.min(100,player.hp+30);addText('+30 VITALS',p.x,p.y,'#8fe6a0');}
        else{player.buff=9;addText('DOUBLE BARREL',p.x,p.y,'#ffd23e');}
        sfx.pick(p.kind);pickups.splice(i,1);continue;
      }
    }
    if(p.y>H+40)pickups.splice(i,1);
  }

  for(let i=parts.length-1;i>=0;i--){
    const p=parts[i];p.life-=dt;
    if(p.life<=0){parts.splice(i,1);continue;}
    if(p.type==='gore'||p.type==='spark'){p.x+=p.vx*dt;p.y+=p.vy*dt;const d=Math.min(1,dt*5);p.vx+=(0-p.vx)*d;p.vy+=(0-p.vy)*d;}
    else if(p.type==='smoke'){p.x+=p.vx*dt;p.y+=p.vy*dt;p.size+=dt*30;}
    else if(p.type==='ring')p.r+=p.vr*dt;
    else if(p.type==='text'){p.y+=p.vy*dt;}
  }

  shake=Math.max(0,shake-dt*26-shake*dt*5);
  shakeX=rand(-shake,shake);shakeY=rand(-shake,shake);
  flashRed=Math.max(0,flashRed-dt*1.4);
  sHitT-=dt;
  growlT-=dt;
  if(growlT<=0){growlT=1.5+Math.random()*2.5;if(zombies.length&&AC){const z=pick(zombies);sfx.growl(clamp((z.x/W)*2-1,-0.8,0.8));}}
  if(playing&&player.hp<35){hbT-=dt;if(hbT<=0){hbT=0.9;sfx.heart();}}
}

  ctx.fillStyle='#8a6a3f';ctx.beginPath();ctx.arc(0,-6,6.5,0,TAU);ctx.fill();
  ctx.fillStyle='#c9a07a';ctx.beginPath();ctx.arc(0,-4.5,5,0,TAU);ctx.fill();
  ctx.fillStyle='#8a6a3f';ctx.fillRect(-5,-12,10,3);
  if(p.flashT>0){
    ctx.globalCompositeOperation='lighter';
    const a=p.flashT/0.06;
    const g=ctx.createRadialGradient(0,-30,1,0,-30,24);
    g.addColorStop(0,'rgba(255,255,230,'+(0.9*a)+')');g.addColorStop(0.4,'rgba(255,200,90,'+(0.6*a)+')');g.addColorStop(1,'rgba(255,180,60,0)');
    ctx.fillStyle=g;ctx.beginPath();ctx.arc(0,-30,24,0,TAU);ctx.fill();
  }
  ctx.restore();
}
function drawLight(){
  const px=player.x,py=player.y-18,L=Math.min(H*0.6,520);
  const skew=clamp(player.vx*0.0009,-0.35,0.35);
  ctx.save();ctx.globalCompositeOperation='lighter';
  const g=ctx.createLinearGradient(0,py,0,py-L);
  g.addColorStop(0,'rgba(255,246,214,0.2)');g.addColorStop(1,'rgba(255,246,214,0)');
  ctx.fillStyle=g;ctx.beginPath();
  ctx.moveTo(px-12,py);ctx.lineTo(px+12,py);
  ctx.lineTo(px+95+skew*160,py-L);ctx.lineTo(px-95+skew*160,py-L);
  ctx.closePath();ctx.fill();
  const g2=ctx.createRadialGradient(px,py,4,px,py,70);
  g2.addColorStop(0,'rgba(255,244,214,0.12)');g2.addColorStop(1,'rgba(255,244,214,0)');
  ctx.fillStyle=g2;ctx.fillRect(px-72,py-72,144,144);
  ctx.restore();
}
function drawParts(){
  for(const p of parts){
    const a=Math.max(0,p.life/p.max);
    if(p.type==='gore'){
      ctx.globalAlpha=a;ctx.fillStyle=p.color;
      ctx.beginPath();ctx.arc(p.x,p.y,p.size*a+0.5,0,TAU);ctx.fill();
    }else if(p.type==='spark'){
      ctx.globalAlpha=a;ctx.strokeStyle=p.color;ctx.lineWidth=2;
      ctx.beginPath();ctx.moveTo(p.x,p.y);ctx.lineTo(p.x-p.vx*0.03,p.y-p.vy*0.03);ctx.stroke();
    }else if(p.type==='smoke'){
      ctx.globalAlpha=a*0.14;ctx.fillStyle='#9fb8a2';
      ctx.beginPath();ctx.arc(p.x,p.y,p.size,0,TAU);ctx.fill();
    }else if(p.type==='ring'){
      ctx.globalAlpha=a;ctx.strokeStyle=p.color;ctx.lineWidth=3*a+1;
      ctx.beginPath();ctx.arc(p.x,p.y,p.r,0,TAU);ctx.stroke();
    }else if(p.type==='slash'){
      const prog=1-a,start=-Math.PI/2-1.2;
      ctx.globalAlpha=a;ctx.strokeStyle='rgba(230,240,220,0.9)';ctx.lineWidth=4;
      ctx.beginPath();ctx.arc(p.x,p.y,p.size,start,start+2.4*prog);ctx.stroke();
    }else if(p.type==='text'){
      ctx.globalAlpha=a;ctx.fillStyle=p.color;
      ctx.font='700 13px ui-monospace,Consolas,monospace';ctx.textAlign='center';
      ctx.fillText(p.txt,p.x,p.y);
    }
  }
  ctx.globalAlpha=1;
}
function render(){
  ctx.save();
  ctx.translate(shakeX,shakeY);
  drawWorld();
  for(const p of props)drawProp(p);
  for(const f of fogs){
    const g=ctx.createRadialGradient(f.x,f.y,0,f.x,f.y,f.r);
    g.addColorStop(0,'rgba(150,190,160,'+(0.05+0.02*Math.sin(t*0.5+f.ph))+')');g.addColorStop(1,'rgba(150,190,160,0)');
    ctx.fillStyle=g;ctx.fillRect(f.x-f.r,f.y-f.r,f.r*2,f.r*2);
  }
  for(const p of pickups)drawPickup(p);
  for(const z of zombies)drawZombie(z);
  if(state!=='over')drawPlayer();
  ctx.lineCap='round';
  for(const p of pellets){
    ctx.strokeStyle='rgba(255,233,168,0.35)';ctx.lineWidth=5;
    ctx.beginPath();ctx.moveTo(p.x,p.y);ctx.lineTo(p.x-p.vx*0.02,p.y-p.vy*0.02);ctx.stroke();
    ctx.strokeStyle='#ffe9a8';ctx.lineWidth=2;
    ctx.beginPath();ctx.moveTo(p.x,p.y);ctx.lineTo(p.x-p.vx*0.02,p.y-p.vy*0.02);ctx.stroke();
  }
  drawParts();
  if(state!=='over')drawLight();
  // atmosphere overlays
  ctx.fillStyle=topG;ctx.fillRect(-20,-20,W+40,H*0.45);
  const tg=ctx.createLinearGradient(0,0,0,60);
  tg.addColorStop(0,'rgba(120,255,120,0.05)');tg.addColorStop(1,'rgba(120,255,120,0)');
  ctx.fillStyle=tg;ctx.fillRect(-20,-20,W+40,80);
  ctx.fillStyle=vigG;ctx.fillRect(-20,-20,W+40,H+40);
  if(flashRed>0){ctx.fillStyle='rgba(170,30,25,'+flashRed+')';ctx.fillRect(-20,-20,W+40,H+40);}
  if(state==='play'&&player.hp<35){
    const p=(Math.sin(t*5.5)+1)/2*(1-player.hp/35)*0.3;
    ctx.fillStyle='rgba(140,10,10,'+p+')';ctx.fillRect(-20,-20,W+40,H+40);
  }
  ctx.restore();
}

let lastScore=-1,lastM=-1,lastK=-1,lastW=-1,lastA=-1,lastH=-1,buffOn=null;
function hudSync(){
  if(state!=='play')return;
  const s=score();
  if(s!==lastScore){$('sScore').textContent=s;$('sBest').textContent=Math.max(best,s);lastScore=s;}
  const m=Math.floor(meters);
  if(m!==lastM){$('sDist').textContent=m+'m';lastM=m;}
  if(kills!==lastK){$('sKills').textContent=kills;lastK=kills;}
  if(wave!==lastW){$('sWave').textContent=wave;lastW=wave;}
  if(player.ammo!==lastA){
    $('sAmmo').textContent=player.ammo;lastA=player.ammo;
    const knife=player.ammo<=0;
    $('ammoLbl').textContent=knife?'KNIFE — SWING!':'SHELLS';
    $('ammoPanel').classList.toggle('knife',knife);
  }
  const hv=Math.ceil(player.hp);
  if(hv!==lastH){
    $('sHp').textContent=hv;
    const f=$('hpFill');f.style.width=hv+'%';
    f.className=hv>55?'':(hv>28?'warn':'crit');lastH=hv;
  }
  const on=player.buff>0;
  if(on!==buffOn){$('buffRow').classList.toggle('hidden',!on);buffOn=on;}
  if(on)$('sBuff').textContent=player.buff.toFixed(1);
}
