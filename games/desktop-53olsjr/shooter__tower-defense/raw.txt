function moveEnemy(e,dt){ let d=e.speed*(e.slowT>0?1-e.slowAmt:1)*dt;

 return e.wp>=WAYPTS.length-1; }

  e.slowT-=dt;

   e.shield=Math.min(e.maxShield,e.shield+ENEMIES.aegis.regen*dt);
