}

function drawPac(c,x,y,r,mouth,ang){
  c.save();c.translate(x,y);c.rotate(ang);
  c.shadowColor='rgba(255,230,61,.55)';c.shadowBlur=r*.9;
  c.beginPath();c.moveTo(0,0);c.arc(0,0,r,mouth,Math.PI*2-mouth);c.closePath();
  c.fillStyle='#ffe63d';c.fill();c.restore();
}
function drawGhost(c,x,y,r,col,t,ph,die,lx,ly){
  c.save();c.translate(x,y);
  if(die>0){const s=Math.max(0.01,1-die);c.globalAlpha=Math.max(0,1-die*1.15);c.scale(s,s);c.rotate(die*5);}
  else c.translate(0,Math.sin(t*2.6+ph)*r*.1);
  const w=r*.92,by=r*.78;
  c.beginPath();c.arc(0,-r*.12,w,Math.PI,0,true);c.lineTo(w,by);
  for(let i=0;i<3;i++){
    const xA=w-(i+1)*(2*w/3),xM=w-i*(2*w/3)-(w/3);
    c.quadraticCurveTo(xM,by+r*.16+Math.sin(t*7+ph+i)*r*.07,xA,by);
  }
  c.closePath();c.fillStyle=die>0&&die<.3?'#ffffff':col;c.fill();
  if(die<.5){
    const ex=r*.34,ey=-r*.22,er=r*.3;
    for(const s of [-1,1]){
      c.beginPath();c.arc(s*ex,ey,er,0,7);c.fillStyle='#fff';c.fill();
      c.beginPath();c.arc(s*ex+lx*er*.45,ey+ly*er*.45,er*.52,0,7);c.fillStyle='#18214d';c.fill();
    }
  }
  c.restore();
}

function drawTrail(c){
  if(path.length<2)return;
  const pts=path.map(p=>[cx(p.x),cy(p.y)]);
  if(closed)pts.push([cx(start.x),cy(start.y)]);
  c.lineJoin='round';c.lineCap='round';
  const passes=[[cell*.5,'rgba(90,240,255,.22)',16],[cell*.3,'#5fe8ff',8],[cell*.12,'#f0ffff',0]];
  for(const [w,col,bl] of passes){
    c.beginPath();c.moveTo(pts[0][0],pts[0][1]);
    for(let i=1;i<pts.length;i++)c.lineTo(pts[i][0],pts[i][1]);
    c.shadowColor='rgba(93,241,255,.8)';c.shadowBlur=bl;
    c.strokeStyle=col;c.lineWidth=w;c.stroke();
  }
  c.shadowBlur=0;
  if(closed){ // energy pulse around the closed loop
    c.beginPath();c.moveTo(pts[0][0],pts[0][1]);
    for(let i=1;i<pts.length;i++)c.lineTo(pts[i][0],pts[i][1]);
    c.setLineDash([cell*.45,cell*1.1]);c.lineDashOffset=-T*cell*2.4;
    c.strokeStyle='rgba(255,255,255,.8)';c.lineWidth=cell*.07;c.stroke();

    const e=path[path.length-1];

    c.fillStyle='rgba(255,255,255,.9)';c.fill();
  }
}

let frameDt=0,lastT=performance.now();
function update(dt){
  for(const d of dust){d.y-=d.v*dt;if(d.y<-6){d.y=cssH+6;d.x=Math.random()*cssW;}}
  shake=Math.max(0,shake-dt*1.7);
  for(let i=popups.length-1;i>=0;i--){popups[i].t+=dt;if(popups[i].t>=popups[i].life)popups.splice(i,1);}
  for(const g of ghosts)if(g.dead&&g.dieT<1)g.dieT=Math.min(1,g.dieT+dt*2.2);
  if(state==='run'){
    ra.t+=dt*SPEED;
    while(ra.t>=1){
      ra.t-=1;ra.cur=(ra.cur+1)%path.length;
      const ccell=path[ra.cur];
      if(ccell.x!==start.x||ccell.y!==start.y){
        const d=dots.find(dd=>dd.x===ccell.x&&dd.y===ccell.y);
        if(d&&!d.eaten){d.eaten=true;levelScore+=10;SFX.chomp();
          popups.push({x:cx(ccell.x),y:cy(ccell.y),txt:'+10',t:0,life:.7,size:cell*.32,color:'#ffd94a'});
          popups.push({x:cx(ccell.x),y:cy(ccell.y),ring:true,t:0,life:.35,color:'rgba(255,217,74,ALPHA)'});}
      }
      const gi=ghosts.findIndex(g=>g.x===ccell.x&&g.y===ccell.y&&!g.dead);
      if(gi>-1){const g=ghosts[gi];g.dead=true;g.dieT=0;levelScore+=200;shake=.35;SFX.ghost();
        popups.push({x:cx(ccell.x),y:cy(ccell.y)-cell*.1,txt:'+200',t:0,life:.9,size:cell*.45,color:'#fff'});}
      updateHUD();
      if(ra.cur===0){finishRun();break;}
    }
  }
}
