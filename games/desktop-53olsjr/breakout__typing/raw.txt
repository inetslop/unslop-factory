function pickTarget(initial){

  if(!initial){while(tapped.length){const c=tapped.shift();if(c.alive&&c.cool<=0){b=c;break;}}}
  if(!b){let maxR=-1;bricks.forEach(k=>{if(k.alive&&k.row>maxR)maxR=k.row;});
    const cand=bricks.filter(k=>k.alive&&k.row===maxR&&k.cool<=0);
    b=cand.length?pick(cand):null;}
  if(!b){onFieldCleared();return;}
  target=b;typed=0;
  timerMax=(1.4+b.word.length*.55)*Math.max(.65,1-(S.level-1)*.05);timer=timerMax;
  buildTape(b.word);
  tapeLabel.textContent='TARGET \u25b8 '+b.word.toUpperCase();

}

/* ================= update ================= */
function movePaddle(dt){
  if(keyL)pad.desired-=780*dt;
  if(keyR)pad.desired+=780*dt;
  pad.desired=clamp(pad.desired,pad.w/2,W-pad.w/2);
  pad.x+=(pad.desired-pad.x)*(1-Math.exp(-16*dt));
  pad.squash=Math.max(0,pad.squash-dt*5);
}
function stepBall(dt){
  const steps=Math.max(1,Math.ceil(ball.speed*dt/(ball.r*.9)));
  for(let s=0;s<steps;s++){
    ball.x+=ball.vx*dt/steps;ball.y+=ball.vy*dt/steps;
    if(ball.x-ball.r<0){ball.x=ball.r;ball.vx=Math.abs(ball.vx);sWall();spark(2,ball.y,'#4cc9e8',3);}
    if(ball.x+ball.r>W){ball.x=W-ball.r;ball.vx=-Math.abs(ball.vx);sWall();spark(W-2,ball.y,'#4cc9e8',3);}
    if(ball.y-ball.r<hudH){ball.y=hudH+ball.r;ball.vy=Math.abs(ball.vy);sWall();}
    for(const b of bricks){if(!b.alive)continue;
      const nx=clamp(ball.x,b.x,b.x+b.w),ny=clamp(ball.y,b.y,b.y+b.h);
      const dx=ball.x-nx,dy=ball.y-ny;
      if(dx*dx+dy*dy<ball.r*ball.r){
        if(Math.abs(dx)>Math.abs(dy)&&dx!==0){ball.vx=dx>0?Math.abs(ball.vx):-Math.abs(ball.vx);ball.x=nx+(dx>0?ball.r:-ball.r);}
        else{ball.vy=dy>0?Math.abs(ball.vy):-Math.abs(ball.vy);ball.y=ny+(dy>0?ball.r:-ball.r);}
        ballHitBrick(b,nx,ny);break;}}
    if(ball.vy>0&&ball.y+ball.r>=pad.y&&ball.y-ball.r<pad.y+pad.h+6&&Math.abs(ball.x-pad.x)<=pad.w/2+ball.r*.7){
      const rel=clamp((ball.x-pad.x)/(pad.w/2),-1,1);
      const a=-Math.PI/2+rel*(Math.PI*.36);
      ball.speed=Math.min(620,ball.speed+2);
      ball.vx=Math.cos(a)*ball.speed;ball.vy=Math.sin(a)*ball.speed;
      ball.y=pad.y-ball.r;pad.squash=1;sPaddle();spark(ball.x,pad.y,'#ffb454',5);}
  }
}
function update(dt){
  S.time+=dt;S.shake=Math.max(0,S.shake-dt*26);
  glyphs.forEach(g=>{g.y-=g.v*dt/H;if(g.y<-.06){g.y=1.05;g.x=Math.random();}});
  particles.forEach(p=>{p.t-=dt;p.x+=p.vx*dt;p.y+=p.vy*dt;p.vy+=(p.g||0)*dt;if(p.vr)p.rot+=p.vr*dt;});
  particles=particles.filter(p=>p.t>0);
  floaters.forEach(f=>{f.t-=dt*1.1;f.y-=42*dt;});floaters=floaters.filter(f=>f.t>0);
  beams.forEach(bm=>bm.t-=dt*3.5);beams=beams.filter(bm=>bm.t>0);
  if(S.mode==='serve'){movePaddle(dt);ball.x=pad.x;ball.y=pad.y-ball.r-2;
    serveT-=dt;if(serveT<=0)launch();}
  else if(S.mode==='play'){
    movePaddle(dt);stepBall(dt);
    ball.trail.push({x:ball.x,y:ball.y});if(ball.trail.length>12)ball.trail.shift();
    if(target){timer-=dt;
      timerFill.style.width=(clamp(timer/timerMax,0,1)*100)+'%';
      timerFill.classList.toggle('low',timer/timerMax<.3);
      if(timer<=0)failTarget();}
    bricks.forEach(b=>{if(b.cool>0)b.cool-=dt;});
    if(ball.y-ball.r>H-4){
      S.lives--;S.combo=0;S.shake=10;sLife();redFlash();updateHud(true);
      if(S.lives<=0)gameOver();else serve();}
  }
  else if(S.mode==='clear'){clearT-=dt;if(clearT<=0)startLevel(S.level+1);}
}

    ctx.fillStyle='rgb('+c.join(',')+')';
    ctx.beginPath();ctx.arc(ball.x,ball.y,ball.r,0,7);ctx.fill();ctx.restore();}
  /* paddle */
  ctx.save();ctx.translate(pad.x,pad.y+pad.h/2);
  const sq=pad.squash;ctx.scale(1+sq*.18,1-sq*.42);
  if(S.combo>=6){ctx.shadowColor='rgba(255,138,92,.8)';ctx.shadowBlur=18;}
  ctx.fillStyle='#ffb454';rr(-pad.w/2,-pad.h/2,pad.w,pad.h,7);ctx.fill();
  ctx.shadowBlur=0;
  ctx.fillStyle='rgba(255,255,255,.5)';rr(-pad.w/2+3,-pad.h/2+2,pad.w-6,4,2);ctx.fill();
  for(let k=-1;k<=1;k++){ctx.fillStyle='rgba(60,30,5,.35)';
    rr(k*pad.w*.27-9,-pad.h/2-1,18,5,2);ctx.fill();}
  ctx.restore();
  if(S.mode==='serve'){const al=.4+.6*(Math.sin(S.time*6)+1)/2;
    ctx.font='bold 13px "Courier New",monospace';ctx.textAlign='center';
    ctx.fillStyle='rgba(244,234,216,'+al.toFixed(3)+')';
    ctx.fillText('SPACE \u25b8 LAUNCH',pad.x,pad.y-26);}
  /* particles */
  particles.forEach(p=>{const a=clamp(p.t/p.max,0,1);
    ctx.save();ctx.globalAlpha=a;ctx.fillStyle=p.c;
    if(p.vr){ctx.translate(p.x,p.y);ctx.rotate(p.rot);ctx.fillRect(-p.s/2,-p.s/2,p.s,p.s);}
    else{ctx.beginPath();ctx.arc(p.x,p.y,p.s,0,7);ctx.fill();}
    ctx.restore();});
  /* floaters */
  floaters.forEach(f=>{ctx.save();ctx.globalAlpha=clamp(f.t,0,1);
    ctx.font='italic bold 20px Impact,"Arial Narrow",sans-serif';ctx.textAlign='center';
    ctx.fillStyle='#ffd23f';ctx.fillText(f.txt,f.x,f.y);ctx.restore();});
  ctx.restore();
}
