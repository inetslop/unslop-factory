function stepFx(dt){
  for(let i=parts.length-1;i>=0;i--){const p=parts[i];p.t-=dt;
    if(p.t<=0){parts.splice(i,1);continue;}
    const d=Math.exp(-p.drag*dt);p.vx*=d;p.vy*=d;p.x+=p.vx*dt;p.y+=p.vy*dt;}
  for(let i=rings.length-1;i>=0;i--){const r=rings[i];r.r+=r.vr*dt;r.vr*=Math.exp(-2.5*dt);r.a-=dt*1.1;if(r.a<=0)rings.splice(i,1);}
  for(let i=pops.length-1;i>=0;i--){const p=pops[i];p.t-=dt;p.y-=26*dt;if(p.t<=0)pops.splice(i,1);}
  for(let i=meteors.length-1;i>=0;i--){const m=meteors[i];m.x+=m.vx*dt;m.y+=m.vy*dt;
    if(m.x<-120||m.x>W+120||m.y>H+120)meteors.splice(i,1);}
  if(shakeT>0)shakeT=Math.max(0,shakeT-dt);else shakeMag=0;
}
function step(dt){
  time+=dt;
  for(const s of stars){s.y+=(3+s.L*6)*dt;if(s.y>H+2){s.y=-2;s.x=Math.random()*W;}}
  meteorT-=dt;if(meteorT<=0){meteorT=5+Math.random()*9;spawnMeteor();}
  stepFx(dt);
  if(bannerMsg){bannerMsg.t+=dt;if(bannerMsg.t>bannerMsg.dur)bannerMsg=null;}

  if(mode!=='play'){
    for(const a of asteroids){a.x+=a.vx*dt;a.y+=a.vy*dt;a.rot+=a.vr*dt;wrap(a);}
    return;
  }
  if(paused)return;

  /* ship */
  const L=keys.has('ArrowLeft')||keys.has('KeyA'),Rt=keys.has('ArrowRight')||keys.has('KeyD');
  if(!ship.dead){
    if(L)ship.a-=3.8*dt;if(Rt)ship.a+=3.8*dt;
    ship.thrust=(keys.has('ArrowUp')||keys.has('KeyW'));
    if(ship.thrust){
      ship.vx+=Math.cos(ship.a)*520*S*dt;ship.vy+=Math.sin(ship.a)*520*S*dt;
      const ba=ship.a+Math.PI+(Math.random()-.5)*.6;
      parts.push({x:ship.x-Math.cos(ship.a)*14*S,y:ship.y-Math.sin(ship.a)*14*S,
        vx:ship.vx*.5+Math.cos(ba)*(90+Math.random()*120)*S,vy:ship.vy*.5+Math.sin(ba)*(90+Math.random()*120)*S,
        t:.25+Math.random()*.15,max:.4,size:(1.5+Math.random()*1.8)*S,
        clr:Math.random()<.4?'#fff3c4':'#ff8b3d',drag:2});
    }
    const dr=Math.exp(-.6*dt);ship.vx*=dr;ship.vy*=dr;
    const sp=Math.hypot(ship.vx,ship.vy),mx=520*S;
    if(sp>mx){ship.vx*=mx/sp;ship.vy*=mx/sp;}
    ship.x+=ship.vx*dt;ship.y+=ship.vy*dt;wrap(ship);
  }else{
    ship.thrust=false;ship.deadT-=dt;
    if(lives>0&&ship.deadT<=0){
      if(safeSpot())Object.assign(ship,{x:W/2,y:H/2,vx:0,vy:0,a:-Math.PI/2,dead:false,invuln:2.5});
      else ship.deadT=.35;
    }
  }
  ship.invuln=Math.max(0,ship.invuln-dt);ship.hypCd=Math.max(0,ship.hypCd-dt);
  for(const k in PTYPES)if(ship[k]>0)ship[k]-=dt;
  if(thrustGain&&AC)thrustGain.gain.setTargetAtTime((ship.thrust&&!ship.dead)?.12:0,AC.currentTime,.06);

  /* fire */
  fireCd-=dt;
  if(keys.has('Space')&&!ship.dead&&fireCd<=0){
    fireCd=ship.rapid>0?.075:.16;
    const spread=ship.triple>0?[-.14,0,.14]:[0];
    for(const o of spread){const a=ship.a+o;
      bullets.push({x:ship.x+Math.cos(a)*18*S,y:ship.y+Math.sin(a)*18*S,px:0,py:0,
        vx:ship.vx*.35+Math.cos(a)*760*S,vy:ship.vy*.35+Math.sin(a)*760*S,t:1.05});}
    sfx.laser();
  }

  /* bullets */
  for(let i=bullets.length-1;i>=0;i--){const b=bullets[i];
    b.px=b.x;b.py=b.y;b.x+=b.vx*dt;b.y+=b.vy*dt;wrap(b);b.t-=dt;
    if(b.t<=0)bullets.splice(i,1);}

  /* asteroids */
  for(const a of asteroids){a.x+=a.vx*dt;a.y+=a.vy*dt;a.rot+=a.vr*dt;wrap(a);}

  /* collisions: bullets vs asteroids */
  for(let bi=bullets.length-1;bi>=0;bi--){const b=bullets[bi];let hit=-1;
    for(let i=0;i<asteroids.length;i++){const a=asteroids[i];
      if(dist2(b.x,b.y,a.x,a.y)<Math.pow(a.r*.9+3,2)){hit=i;break;}}
    if(hit>=0){bullets.splice(bi,1);hitAsteroid(hit,false);}}

  /* collisions: ship vs asteroids */
  if(!ship.dead&&ship.invuln<=0){
    for(let i=0;i<asteroids.length;i++){const a=asteroids[i];
      if(dist2(ship.x,ship.y,a.x,a.y)<Math.pow(a.r*.85+13*S,2)){
        if(ship.shield>0){
          hitAsteroid(i,true);
          const dx=ship.x-a.x,dy=ship.y-a.y,d=Math.hypot(dx,dy)||1;
          ship.vx+=dx/d*420*S;ship.vy+=dy/d*420*S;
          rings.push({x:ship.x,y:ship.y,r:14*S,vr:340*S,a:.7,lw:2.5});
          pops.push({x:ship.x,y:ship.y-34*S,txt:'DEFLECTED',t:.8,max:.8,clr:'#7ef0ff',fs:13});
          sfx.deflect();
        }else{killShip();}
        break;}}
  }

  /* powerups */
  for(let i=pows.length-1;i>=0;i--){const p=pows[i];
    p.x+=p.vx*dt;p.y+=p.vy*dt;wrap(p);p.t-=dt;
    if(p.t<=0){pows.splice(i,1);continue;}
    if(!ship.dead&&dist2(p.x,p.y,ship.x,ship.y)<Math.pow(30*S,2)){
      pows.splice(i,1);
      ship[p.type]=PTYPES[p.type].dur;
      sfx.power();
      pops.push({x:ship.x,y:ship.y-34*S,txt:PTYPES[p.type].name+'!',t:1.1,max:1.1,clr:'rgb('+PTYPES[p.type].rgb+')',fs:14});
    }}

  /* wave management */
  if(toSpawn>0){spawnT-=dt;if(spawnT<=0){spawnT=.32;toSpawn--;spawnAsteroid(3);}}
  else if(asteroids.length===0&&clearT===0){
    clearT=2.4;const bonus=100*wave;score+=bonus;
    banner('FIELD CLEARED','+'+bonus+' BONUS');sfx.wave();updateHud();
  }
  if(clearT>0){clearT-=dt;if(clearT<=0)startWave(wave+1);}

  /* death / game over */
  if(ship.dead&&lives<=0){overT-=dt;if(overT<=0)showOver();}
  warnEl.classList.toggle('on',lives===1&&!ship.dead);
  syncChips();
}

/* ============ RENDER ============ */
function drawStars(){
  const tx=mode==='play'&&!ship.dead?ship.x:W/2+Math.sin(time*.07)*60;
  const ty=mode==='play'&&!ship.dead?ship.y:H/2+Math.cos(time*.05)*40;
  cam.x+=(tx-cam.x)*Math.min(1,gdt*1.6);cam.y+=(ty-cam.y)*Math.min(1,gdt*1.6);
  const f=[.1,.25,.5];
  for(const s of stars){
    const x=mod(s.x-cam.x*f[s.L],W),y=mod(s.y-cam.y*f[s.L]*.6,H);
    const a=.35+.6*(.5+.5*Math.sin(time*s.ts+s.tw));
    ctx.globalAlpha=a;
    ctx.fillStyle=s.warm?'#ffd9a8':'#cfeeff';
    ctx.fillRect(x,y,s.r,s.r);
  }
  ctx.globalAlpha=1;
}
function drawMeteors(){
  ctx.lineWidth=1.5;ctx.strokeStyle='rgba(190,235,255,.5)';
  for(const m of meteors){
    ctx.beginPath();ctx.moveTo(m.x,m.y);ctx.lineTo(m.x-m.vx*.08,m.y-m.vy*.08);ctx.stroke();
    ctx.fillStyle='rgba(230,250,255,.8)';ctx.fillRect(m.x-1,m.y-1,2.4,2.4);
  }
}
function drawRings(){
  for(const r of rings){
    ctx.globalAlpha=Math.max(0,r.a);
    ctx.lineWidth=r.lw;ctx.strokeStyle='#bff3ff';
    ctx.beginPath();ctx.arc(r.x,r.y,r.r,0,TAU);ctx.stroke();
  }
  ctx.globalAlpha=1;
}
function drawParts(){
  for(const p of parts){
    const a=Math.max(0,p.t/p.max);
    ctx.globalAlpha=a;ctx.fillStyle=p.clr;
    ctx.beginPath();ctx.arc(p.x,p.y,Math.max(.4,p.size*a),0,TAU);ctx.fill();
  }
  ctx.globalAlpha=1;
}
function drawBullets(){
  for(const b of bullets){
    const d=Math.hypot(b.x-b.px,b.y-b.py);
    if(d<80){ctx.strokeStyle='rgba(255,246,216,.9)';ctx.lineWidth=2;
      ctx.beginPath();ctx.moveTo(b.px,b.py);ctx.lineTo(b.x,b.y);ctx.stroke();}
    ctx.fillStyle='rgba(255,246,216,.28)';
    ctx.beginPath();ctx.arc(b.x,b.y,5*S,0,TAU);ctx.fill();
    ctx.fillStyle='#fff6d8';
    ctx.beginPath();ctx.arc(b.x,b.y,2.2*S,0,TAU);ctx.fill();
  }
}
const ASTCLR=['','#bff3ff','#9fe4f2','#7fdcc8'];
function drawAsteroids(){
  for(const a of asteroids){
    ctx.save();ctx.translate(a.x,a.y);ctx.rotate(a.rot);
    ctx.beginPath();ctx.moveTo(a.pts[0][0],a.pts[0][1]);
    for(let i=1;i<a.pts.length;i++)ctx.lineTo(a.pts[i][0],a.pts[i][1]);
    ctx.closePath();
    ctx.fillStyle='rgba(150,230,240,.05)';ctx.fill();
    ctx.lineWidth=5;ctx.strokeStyle='rgba(143,230,240,.14)';ctx.stroke();
    ctx.lineWidth=1.6;ctx.strokeStyle=ASTCLR[a.size];ctx.stroke();
    ctx.restore();
  }
}
function drawPows(){
  for(const p of pows){
    const T=PTYPES[p.type],pu=.5+.5*Math.sin(time*4+p.pulse);
    const blink=p.t<3&&Math.floor(time*8)%2===0;
    if(blink)continue;
    ctx.save();ctx.translate(p.x,p.y);
    ctx.globalCompositeOperation='lighter';
    ctx.rotate(time*1.2+p.pulse);
    const r=(12+pu*2)*S;
    ctx.beginPath();
    for(let i=0;i<6;i++){const a=i/6*TAU;i?ctx.lineTo(Math.cos(a)*r,Math.sin(a)*r):ctx.moveTo(r,0);}
    ctx.closePath();
    ctx.strokeStyle='rgba('+T.rgb+',.9)';ctx.lineWidth=2;ctx.stroke();
    ctx.fillStyle='rgba('+T.rgb+','+(.1+.08*pu)+')';ctx.fill();
    ctx.restore();
    ctx.font='700 '+Math.round(12*Math.max(.8,S))+'px '+CMONO;
    ctx.textAlign='center';ctx.textBaseline='middle';
    ctx.fillStyle='rgb('+T.rgb+')';ctx.fillText(T.chr,p.x,p.y+.5);
  }
}
function drawShip(){
  if(ship.dead||mode!=='play')return;
  const blink=ship.invuln>0&&Math.floor(time*14)%2===0;
  ctx.save();ctx.translate(ship.x,ship.y);
  if(ship.shield>0){
    const p=.5+.5*Math.sin(time*6);
    ctx.globalCompositeOperation='lighter';
    ctx.beginPath();ctx.arc(0,0,(24+2*p)*S,0,TAU);
    ctx.strokeStyle='rgba(126,240,255,'+(.45+.3*p)+')';ctx.lineWidth=2;ctx.stroke();
    ctx.fillStyle='rgba(126,240,255,.07)';ctx.fill();
    ctx.globalCompositeOperation='source-over';
  }
  if(!blink){
    ctx.rotate(ship.a);ctx.scale(S,S);
    if(ship.thrust){
      const L=13+Math.random()*9;
      ctx.beginPath();ctx.moveTo(-8,4.5);ctx.lineTo(-8-L,0);ctx.lineTo(-8,-4.5);ctx.closePath();
      ctx.fillStyle='rgba(255,139,61,.9)';ctx.fill();
      ctx.beginPath();ctx.moveTo(-8,2.4);ctx.lineTo(-8-L*.55,0);ctx.lineTo(-8,-2.4);ctx.closePath();
      ctx.fillStyle='rgba(255,243,196,.95)';ctx.fill();
    }
    ctx.beginPath();ctx.moveTo(16,0);ctx.lineTo(-10,9);ctx.lineTo(-6,0);ctx.lineTo(-10,-9);ctx.closePath();
    ctx.fillStyle='rgba(255,179,71,.12)';ctx.fill();
    ctx.lineJoin='round';ctx.lineWidth=2;ctx.strokeStyle='#ffcf7a';ctx.stroke();
  }
  ctx.restore();
}
function drawPops(){
  for(const p of pops){
    const a=Math.max(0,p.t/p.max);
    ctx.save();ctx.globalAlpha=a;
    ctx.font='700 '+p.fs+'px '+CMONO;
    ctx.textAlign='center';ctx.textBaseline='middle';
    ctx.shadowColor=p.clr;ctx.shadowBlur=8;
    ctx.fillStyle=p.clr;ctx.fillText(p.txt,p.x,p.y);
    ctx.restore();
  }
}
function drawBanner(){
  if(!bannerMsg)return;
  const b=bannerMsg,p=b.t/b.dur;
  let a=p<.12?p/.12:(p>.78?(1-p)/.22:1);a=Math.max(0,a);
  const sc=1+Math.max(0,.18-p)*2.2,fs=Math.round(58*Math.max(.8,S));
  ctx.save();ctx.globalAlpha=a;ctx.translate(W/2,H*.36);ctx.scale(sc,sc);
  ctx.textAlign='center';ctx.textBaseline='middle';
  ctx.font='900 '+fs+'px '+CDISP;
  ctx.shadowColor='rgba(143,230,240,.9)';ctx.shadowBlur=26;
  ctx.fillStyle='#eafcff';ctx.fillText(b.txt,0,0);
  if(b.sub){ctx.shadowBlur=8;ctx.font='600 '+Math.round(14*Math.max(.8,S))+'px '+CMONO;
    ctx.fillStyle='rgba(255,207,122,.92)';ctx.fillText(b.sub.split('').join(' '),0,fs*.62);}
  ctx.restore();
}
function render(){
  ctx.drawImage(bgCv,0,0,W,H);
  let ox=0,oy=0;
  if(shakeT>0){const m=shakeMag*shakeT;ox=(Math.random()-.5)*m*2;oy=(Math.random()-.5)*m*2;}
  ctx.save();ctx.translate(ox,oy);
  drawStars();
  ctx.globalCompositeOperation='lighter';
  drawMeteors();drawRings();
  ctx.globalCompositeOperation='source-over';
  drawPows();drawAsteroids();
  ctx.globalCompositeOperation='lighter';
  drawBullets();drawParts();
  ctx.globalCompositeOperation='source-over';
  drawShip();drawPops();drawBanner();
  ctx.restore();
}
