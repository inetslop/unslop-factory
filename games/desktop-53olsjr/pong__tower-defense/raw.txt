}

function updatePaddle(dt){
  let dir=0;
  if(held.has('w')||held.has('ArrowUp'))dir--;
  if(held.has('s')||held.has('ArrowDown'))dir++;
  if(dir){paddle.y+=dir*950*dt;paddle.ty=paddle.y;}
  else paddle.y+=(paddle.ty-paddle.y)*Math.min(1,dt*16);
  paddle.y=clamp(paddle.y,arena.y+paddle.h/2+4,arena.y+arena.h-paddle.h/2-4);
}
function paddleHit(b,demo){
  const px=arena.x+arena.w-24;
  if(b.vx>0&&b.x+b.r>=px&&b.x<px+paddle.w+8&&Math.abs(b.y-paddle.y)<paddle.h/2+b.r){
    const off=clamp((b.y-paddle.y)/(paddle.h/2),-1,1);
    let sp=Math.min(460,Math.hypot(b.vx,b.vy)*1.07+10);
    const a=off*1.15;
    b.vx=-Math.cos(a)*sp;b.vy=Math.sin(a)*sp;b.x=px-b.r-.5;
    paddleFlash=.18;shakeM=Math.max(shakeM,2.5);
    burst(b.x,b.y,'#ffd9a0',6,120,2);
    if(demo){beep(230+Math.abs(off)*170,.08,'square',.4);beep(75,.09,'sine',.5);}
    else{combo++;sfxPaddle(off);hurt(b,(7+sp*.045)*(1+Math.min(combo,12)*.05),'#ffc46b');}
  }
}
function applyAuras(dt){
  for(const b of balls)b.slow=1;
  for(const t of towers){
    if(t.type!=='cryo')continue;
    const s=tstat(t),p=tpos(t);let active=false;
    for(const b of balls){
      if(b.dead)continue;
      if(Math.hypot(b.x-p.x,b.y-p.y)<s.range+b.r){
        active=true;b.slow=Math.min(b.slow,TT.cryo.slow);b.hp-=s.dps*dt;
        if(b.hp<=0)kill(b);
      }
    }
    t.active=active;
  }
}
function stepBalls(dt,demo){
  for(const b of balls){
    if(b.dead)continue;
    if(b.type==='diver')b.vy=Math.sin(gt*b.wf+b.ph)*b.wa;
    const s=b.slow||1;
    b.x+=b.vx*s*dt;b.y+=b.vy*s*dt;
    if(b.y-b.r<arena.y){b.y=arena.y+b.r;b.vy=Math.abs(b.vy);}
    if(b.y+b.r>arena.y+arena.h){b.y=arena.y+arena.h-b.r;b.vy=-Math.abs(b.vy);}
    if(demo){if(b.x-b.r<arena.x){b.x=arena.x+b.r;b.vx=Math.abs(b.vx);}}
    else{
      if(b.x-b.r>arena.x+arena.w-2){leak(b);continue;}
      if(b.x+b.r<arena.x-4){clearLeft(b);continue;}
    }
    paddleHit(b,demo);
    b.trail.push({x:b.x,y:b.y});if(b.trail.length>6)b.trail.shift();
  }
  balls=balls.filter(b=>!b.dead);
}
function updateTowers(dt){
  for(const t of towers){
    const s=tstat(t),p=tpos(t);
    if(t.type==='pulse'){
      let best=null,bd=1e18;
      for(const b of balls){if(b.dead)continue;const d=(b.x-p.x)**2+(b.y-p.y)**2;if(d<s.range*s.range&&d<bd){bd=d;best=b;}}
      if(best){t.ang=Math.atan2(best.y-p.y,best.x-p.x);
        t.cd-=dt;
        if(t.cd<=0){t.cd=s.cd;t.flash=.08;
          shots.push({x:p.x+Math.cos(t.ang)*16,y:p.y+Math.sin(t.ang)*16,vx:Math.cos(t.ang)*540,vy:Math.sin(t.ang)*540,dmg:s.dmg,life:.9,dead:false});
          sfxShot();}
      }
    }else if(t.type==='arc'){
      t.cd-=dt;
      if(t.cd<=0){
        const tg=[];
        for(const b of balls){if(b.dead)continue;const d=Math.hypot(b.x-p.x,b.y-p.y);if(d<s.range+b.r)tg.push({b,d});}
        if(tg.length){
          tg.sort((a,c)=>a.d-c.d);
          const pts=[{x:p.x,y:p.y}];
          for(let i=0;i<Math.min(s.chains,tg.length);i++){const b=tg[i].b;pts.push({x:b.x,y:b.y});hurt(b,s.dmg,'#ffe14d');}
          zaps.push({pts,t:.14,max:.14});t.cd=s.cd;t.flash=.15;sfxZap();
        }else t.cd=.05;
      }
    }else if(t.type==='nova'){
      t.cd-=dt;
      if(t.cd<=0){
        let any=false;
        for(const b of balls){if(b.dead)continue;if(Math.hypot(b.x-p.x,b.y-p.y)<s.range+b.r){any=true;break;}}
        if(any){rings.push({x:p.x,y:p.y,r:10,max:s.range,dmg:s.dmg,hit:new Set(),dead:false});t.cd=s.cd;t.flash=.25;sfxNova();}
        else t.cd=.05;
      }
    }
    if(t.flash>0)t.flash-=dt;
  }
}
function updateEffects(dt){
  for(const s of shots){
    s.x+=s.vx*dt;s.y+=s.vy*dt;s.life-=dt;
    if(s.life<=0||s.x<arena.x-20||s.x>arena.x+arena.w+20||s.y<arena.y-20||s.y>arena.y+arena.h+20){s.dead=true;continue;}
    for(const b of balls){if(b.dead)continue;
      if(Math.hypot(b.x-s.x,b.y-s.y)<b.r+4){hurt(b,s.dmg,'#ffd166');burst(s.x,s.y,'#ffd166',5,120,2);s.dead=true;break;}}
  }
  shots=shots.filter(s=>!s.dead);
  for(const r of rings){
    r.r+=430*dt;if(r.r>r.max){r.dead=true;continue;}
    for(const b of balls){if(b.dead||r.hit.has(b))continue;
      if(Math.hypot(b.x-r.x,b.y-r.y)<=r.r+b.r){r.hit.add(b);hurt(b,r.dmg,'#8aff9e');}}
  }
  rings=rings.filter(r=>!r.dead);
  for(const z of zaps)z.t-=dt;
  zaps=zaps.filter(z=>z.t>0);
  for(const p of parts){p.x+=p.vx*dt;p.y+=p.vy*dt;p.vx*=.94;p.vy*=.94;p.life-=dt;}
  parts=parts.filter(p=>p.life>0);
  for(const f of floaters){f.y-=28*dt;f.life-=dt;}
  floaters=floaters.filter(f=>f.life>0);
  wallFlash=Math.max(0,wallFlash-dt*1.6);
  paddleFlash=Math.max(0,paddleFlash-dt*3);
  shakeM=Math.max(0,shakeM-dt*28);
}
function burst(x,y,color,n,sp,size){
  for(let i=0;i<n;i++){
    const a=Math.random()*6.28,v=rand(.3,1)*sp,ml=rand(.35,.8);
    parts.push({x,y,vx:Math.cos(a)*v,vy:Math.sin(a)*v,life:ml,max:ml,c:color,s:size*rand(.5,1.4)});
  }
  if(parts.length>360)parts.splice(0,parts.length-360);
}

function update(dt){
  gt+=dt;
  if(state==='build'){buildTimer-=dt;if(buildTimer<=0)startWave();}
  else if(state==='wave'){
    if(spawnQueue.length){spawnT-=dt;if(spawnT<=0){spawnEnemy(spawnQueue.shift());spawnT=spawnInterval();}}
    else if(balls.length===0)waveCleared();
  }
  updatePaddle(dt);
  applyAuras(dt);
  stepBalls(dt,false);
  updateTowers(dt);
  updateEffects(dt);
  refreshHUD();
}

    balls.push({type:'drifter',x:fromL?arena.x-14:arena.x+arena.w+14,y:rand(arena.y+30,arena.y+arena.h-30),r:T.r*mul,
      hp:99,maxHp:99,vx:(fromL?1:-1)*rand(90,170),vy:rand(-60,60),slow:1,flash:0,trail:[],dead:false,ph:Math.random()*6.28});
    menuSpawnT=1.1;
  }
  let near=null,bd=-1e9;
  for(const b of balls)if(b.x>bd){bd=b.x;near=b;}
  paddle.ty=near?near.y:arena.y+arena.h/2+Math.sin(gt*.7)*arena.h*.28;
  updatePaddle(dt);
  stepBalls(dt,true);
  updateEffects(dt);
}

function drawBalls(){
  for(const b of balls){
    const T=ET[b.type];
    // trail
    ctx.save();ctx.globalCompositeOperation='lighter';
    for(let i=0;i<b.trail.length;i++){
      const tr=b.trail[i],f=(i+1)/b.trail.length;
      ctx.fillStyle=T.color;ctx.globalAlpha=f*.10;
      ctx.beginPath();ctx.arc(tr.x,tr.y,b.r*(.4+.5*f),0,7);ctx.fill();
    }
    ctx.globalAlpha=1;
    // glow
    const gg=ctx.createRadialGradient(b.x,b.y,b.r*.3,b.x,b.y,b.r*2.1);
    gg.addColorStop(0,T.color+'55');gg.addColorStop(1,T.color+'00');
    ctx.fillStyle=gg;ctx.beginPath();ctx.arc(b.x,b.y,b.r*2.1,0,7);ctx.fill();
    ctx.restore();
    // body
    const bg=ctx.createRadialGradient(b.x-b.r*.35,b.y-b.r*.4,b.r*.2,b.x,b.y,b.r);
    bg.addColorStop(0,'#ffffff');bg.addColorStop(.35,T.color);bg.addColorStop(1,shade(T.color));
    ctx.fillStyle=bg;ctx.beginPath();ctx.arc(b.x,b.y,b.r,0,7);ctx.fill();
    // type markers
    if(b.type==='diver'){
      const a=Math.atan2(b.vy,b.vx);
      ctx.strokeStyle='rgba(255,255,255,.7)';ctx.lineWidth=2;
      ctx.beginPath();
      ctx.moveTo(b.x-Math.cos(a)*b.r*1.4-Math.sin(a)*6,b.y-Math.sin(a)*b.r*1.4+Math.cos(a)*6);
      ctx.lineTo(b.x-Math.cos(a)*b.r*.8,b.y-Math.sin(a)*b.r*.8);
      ctx.lineTo(b.x-Math.cos(a)*b.r*1.4+Math.sin(a)*6,b.y-Math.sin(a)*b.r*1.4-Math.cos(a)*6);
      ctx.stroke();
    }else if(b.type==='splitter'){
      for(let i=0;i<2;i++){
        const a=gt*3.5+b.ph+i*Math.PI;
        ctx.fillStyle='#fff';ctx.beginPath();
        ctx.arc(b.x+Math.cos(a)*b.r*.45,b.y+Math.sin(a)*b.r*.45,2.6,0,7);ctx.fill();
      }
    }else if(b.type==='brute'){
      ctx.save();ctx.translate(b.x,b.y);ctx.rotate(gt*.9+b.ph);
      ctx.strokeStyle='rgba(255,255,255,.55)';ctx.lineWidth=2;
      ctx.beginPath();
      for(let i=0;i<6;i++){const a=i*Math.PI/3;i?ctx.lineTo(Math.cos(a)*b.r*.8,Math.sin(a)*b.r*.8):ctx.moveTo(Math.cos(a)*b.r*.8,Math.sin(a)*b.r*.8);}
      ctx.closePath();ctx.stroke();ctx.restore();
    }
    if(b.slow<1){
      ctx.strokeStyle='rgba(127,232,255,.6)';ctx.lineWidth=1.5;
      ctx.beginPath();ctx.arc(b.x,b.y,b.r+4,gt*2,gt*2+4.5);ctx.stroke();
    }
    if(b.flash>0){ctx.fillStyle='rgba(255,255,255,'+Math.min(1,b.flash*6)+')';ctx.beginPath();ctx.arc(b.x,b.y,b.r,0,7);ctx.fill();}

    if(state!=='menu'&&b.hp<b.maxHp){

      ctx.fillStyle='rgba(4,18,28,.8)';ctx.fillRect(b.x-b.r,b.y-b.r-9,w,3.5);

      ctx.fillRect(b.x-b.r,b.y-b.r-9,w*pct,3.5);
    }
  }
}
function shade(hex){
  const n=parseInt(hex.slice(1),16);
  const r=(n>>16)&255,g=(n>>8)&255,b=n&255;
  return 'rgb('+Math.floor(r*.35)+','+Math.floor(g*.35)+','+Math.floor(b*.4)+')';
}

const elEnergy=$('energyNum'),elWave=$('waveNum'),elHost=$('hostNum'),elCore=$('coreFill'),elPct=$('corePct'),rightHud=$('rightHud');
function refreshHUD(){
  if(energy!==lastEnergy){
    elEnergy.textContent=energy;
    if(energy>lastEnergy){elEnergy.classList.remove('pop');void elEnergy.offsetWidth;elEnergy.classList.add('pop');}
    lastEnergy=energy;
    for(const k in cards)cards[k].classList.toggle('broke',energy<TT[k].cost);
  }
  elWave.textContent=pad2(wave);
  elHost.textContent=state==='wave'?((spawnQueue.length+balls.length)+' LEFT'):(state==='build'?'BUILD':'—');
  const f=Math.max(0,coreHp);
  elCore.style.width=f+'%';elPct.textContent=Math.round(f)+'%';
  rightHud.classList.toggle('low',f<35&&f>0);
}
function updateHover(){
  hoverTower=null;hoverCell=null;
  if(state==='menu'||paused){tip.style.display='none';return;}
  for(const t of towers){
    const x=grid.x+t.col*grid.cell,y=grid.y+t.row*grid.cell;
    if(mouse.x>=x&&mouse.x<x+grid.cell&&mouse.y>=y&&mouse.y<y+grid.cell){hoverTower=t;break;}
  }
  if(!hoverTower){
    const c=Math.floor((mouse.x-grid.x)/grid.cell),r=Math.floor((mouse.y-grid.y)/grid.cell);
    if(c>=0&&c<grid.cols&&r>=0&&r<grid.rows)hoverCell={col:c,row:r};
  }
  if(hoverTower){
    const t=hoverTower,B=TT[t.type],up=upgradeCost(t),p=tpos(t);
    tip.innerHTML='<div class="tt1" style="color:'+B.color+'">'+B.name+' · LV '+t.lvl+'</div>'+
      '<div style="color:var(--dim)">'+B.desc+'</div>'+
      '<div>LMB — '+(up?('UPGRADE '+up+' ⬢'):'MAX LEVEL')+'</div>'+
      '<div>RMB — SELL +'+sellVal(t)+' ⬢</div>';
    tip.style.display='block';
    const tw=tip.offsetWidth;
    tip.style.left=(p.x+grid.cell/2+14>W-tw-8?p.x-grid.cell/2-tw-14:p.x+grid.cell/2+14)+'px';
    tip.style.top=clamp(p.y-30,8,H-120)+'px';
  }else tip.style.display='none';
}

let last=0;
function frame(ts){
  requestAnimationFrame(frame);
  const now=ts/1000;
  let dt=Math.min(.05,now-last||.016);last=now;
  if(hitStopT>0){hitStopT-=dt;dt*=.18;}
  if(paused){render();return;}
  if(state==='menu')updateMenu(dt);
  else if(state!=='over')update(dt);
  else updateEffects(dt);
  render();
}
