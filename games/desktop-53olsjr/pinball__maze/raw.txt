function readInput(){

function physics(dt,demo){
  const b=ball; let ax=0,ay=0;
  if(!demo)[ax,ay]=readInput();
  b.vx+=ax*ACCEL*dt; b.vy+=ay*ACCEL*dt;
  if(demo){demoT-=dt; if(demoT<=0){demoT=1.4+Math.random()*2.2; const a=Math.random()*6.283;
    b.vx+=Math.cos(a)*(260+Math.random()*180); b.vy+=Math.sin(a)*(260+Math.random()*180);}}
  boosting=false;
  if(!demo&&keys.shift&&fuel>2){
    const sp=Math.hypot(b.vx,b.vy);
    if(sp>50){const ux=b.vx/sp,uy=b.vy/sp; b.vx+=ux*BOOST_ACC*dt; b.vy+=uy*BOOST_ACC*dt;
      fuel=Math.max(0,fuel-BURN*dt); boosting=true;
      for(let i=0;i<2;i++)particles.push({t:'dot',x:b.x-ux*10,y:b.y-uy*10,
        vx:-ux*(80+Math.random()*160)+(Math.random()-.5)*70,vy:-uy*(80+Math.random()*160)+(Math.random()-.5)*70,
        life:.28,max:.28,size:2+Math.random()*2,c:i?'#35e0c8':'#ffce54'});}
    if(boosting&&Math.random()<.5)sBoost();
  } else fuel=Math.min(100,fuel+REGEN*dt);
  const dr=Math.exp(-1.35*dt); b.vx*=dr; b.vy*=dr;
  let sp=Math.hypot(b.vx,b.vy);
  if(!demo&&sp<4&&ax===0&&ay===0){b.vx=0;b.vy=0;}
  const cap=(boosting||demo)?BMAXV:MAXV;
  if(sp>cap){b.vx*=cap/sp; b.vy*=cap/sp;}
  const steps=Math.min(10,Math.max(1,Math.ceil(Math.hypot(b.vx,b.vy)*dt/5))), sdt=dt/steps;
  for(let i=0;i<steps;i++){
    b.x+=b.vx*sdt; b.y+=b.vy*sdt;
    collideWalls(demo); collideBumpers(demo);
    if(!demo&&state==='playing'){checkHoles(); if(state!=='playing')break; checkGoal();}
  }
  trail.push({x:b.x,y:b.y}); if(trail.length>16)trail.shift();
}
