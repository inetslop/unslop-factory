/* ============================== LEVEL GEN ============================== */
function genLevel(level,preview){
  const aspect=(W>2&&H>2)?W/H:1.6;
  cols=clamp(Math.round(Math.sqrt(125*aspect)),8,20);
  rows=clamp(Math.round(125/cols),7,14);
  const C=[]; for(let r=0;r<rows;r++){C[r]=[];for(let c=0;c<cols;c++)C[r][c]=[1,1,1,1];} // t r b l
  // carve (recursive backtracker)
  const seen=new Set(), st=[[irand(0,cols-1),irand(0,rows-1)]]; seen.add(st[0][0]+','+st[0][1]);
  while(st.length){
    const [c,r]=st[st.length-1], opts=[];
    if(r>0&&!seen.has(c+','+(r-1)))opts.push([0,c,r-1]);
    if(c<cols-1&&!seen.has((c+1)+','+r))opts.push([1,c+1,r]);
    if(r<rows-1&&!seen.has(c+','+(r+1)))opts.push([2,c,r+1]);
    if(c>0&&!seen.has((c-1)+','+r))opts.push([3,c-1,r]);
    if(!opts.length){st.pop();continue;}
    const [d,nc,nr]=opts[irand(0,opts.length-1)];
    C[r][c][d]=0; C[nr][nc][[2,3,0,1][d]]=0; seen.add(nc+','+nr); st.push([nc,nr]);
  }
  // braid dead ends into loops (more pinball flow)
  const bp=preview?0.3:Math.min(.32,.12+level*.02);
  for(let r=0;r<rows;r++)for(let c=0;c<cols;c++){const w=C[r][c];
    if(w[0]+w[1]+w[2]+w[3]===3&&Math.random()<bp){
      const cl=[]; if(w[0]&&r>0)cl.push(0); if(w[1]&&c<cols-1)cl.push(1);
      if(w[2]&&r<rows-1)cl.push(2); if(w[3]&&c>0)cl.push(3);
      const d=cl[irand(0,cl.length-1)], nr=r+[-1,0,1,0][d], nc=c+[0,1,-1,0][d];
      C[r][c][d]=0; C[nr][nc][[2,3,0,1][d]]=0;
    }}
  // start + farthest exit (BFS)
  const sc={c:irand(0,Math.floor(cols/3)),r:irand(0,Math.floor(rows/3))};
  const dist=[...Array(rows)].map(()=>new Array(cols).fill(-1));
  const q=[[sc.c,sc.r]]; dist[sc.r][sc.c]=0; let pc=sc.c,pr=sc.r,best=0;
  while(q.length){const [c,r]=q.shift(),w=C[r][c];
    if(!w[0]&&dist[r-1][c]<0){dist[r-1][c]=dist[r][c]+1;q.push([c,r-1]);}
    if(!w[1]&&dist[r][c+1]<0){dist[r][c+1]=dist[r][c]+1;q.push([c+1,r]);}
    if(!w[2]&&dist[r+1][c]<0){dist[r+1][c]=dist[r][c]+1;q.push([c,r+1]);}
    if(!w[3]&&dist[r][c-1]<0){dist[r][c-1]=dist[r][c]+1;q.push([c-1,r]);}
    if(dist[r][c]>best){best=dist[r][c];pc=c;pr=r;}
  }
  // wall rectangles (extended at ends for clean corners)
  walls=[];
  for(let r=0;r<=rows;r++)for(let c=0;c<cols;c++){ if(r===0||C[r-1][c][2])walls.push({x:c-T/2,y:r-T/2,w:1+T,h:T}); }
  for(let c=0;c<=cols;c++)for(let r=0;r<rows;r++){ if(c===0||C[r][c-1][1])walls.push({x:c-T/2,y:r-T/2,w:T,h:1+T}); }
  // entities
  const used=new Set([sc.c+','+sc.r,pc+','+pr]);
  const all=[]; for(let r=0;r<rows;r++)for(let c=0;c<cols;c++){ if(used.has(c+','+r))continue;
    const w=C[r][c]; all.push({c,r,dead:w[0]+w[1]+w[2]+w[3]===3}); }
  orbs=[]; G.total=0;
  if(!preview){
    shuffle(all);
    const pool=shuffle(all.filter(a=>a.dead)).concat(shuffle(all.filter(a=>!a.dead)));
    const nOrb=Math.min(8,4+Math.min(4,level-1));
    for(let i=0;i<nOrb&&i<pool.length;i++){const p=pool[i];
      orbs.push({x:p.c+.5,y:p.r+.5,got:false,seed:Math.random()*7}); used.add(p.c+','+p.r);}
    G.total=orbs.length;
  }
  bumpers=[];
  const free=shuffle(all.filter(a=>!used.has(a.c+','+a.r)));
  const nB=Math.min(8,preview?6:3+Math.floor(level/2));
  for(let i=0;i<nB&&i<free.length;i++){const p=free[i];
    bumpers.push({x:p.c+.5,y:p.r+.5,r:.2,flash:0,cool:0}); used.add(p.c+','+p.r);}
  portal={x:pc+.5,y:pr+.5,active:false};
  ball.x=sc.c+.5; ball.y=sc.r+.5; ball.vx=ball.vy=0; ball.rest=true; ball.trail.length=0;
  G.time=preview?9999:(16+G.total*6); G.got=0; G.charge=0; G.charging=false; chargeSndStop();
  parts=[]; pops=[]; computeView(); buildOrbDots(); refreshHUD();
}

function computeView(){ const pad=.62; scale=Math.min(W/(cols+2*pad),H/(rows+2*pad));
  ox=(W-cols*scale)/2; oy=(H-rows*scale)/2; }

/* ============================== PHYSICS ============================== */
function hitRect(b,r){
  const cx=clamp(b.x,r.x,r.x+r.w), cy=clamp(b.y,r.y,r.y+r.h);
  let dx=b.x-cx, dy=b.y-cy; const d2=dx*dx+dy*dy;
  if(d2>=BR*BR)return 0;
  let d=Math.sqrt(d2),nx,ny;
  if(d<1e-5){ const l=b.x-r.x,rr=r.x+r.w-b.x,t2=b.y-r.y,bb=r.y+r.h-b.y,m=Math.min(l,rr,t2,bb);
    if(m===l){nx=-1;ny=0;}else if(m===rr){nx=1;ny=0;}else if(m===t2){nx=0;ny=-1;}else{nx=0;ny=1;}
    b.x=cx+nx*BR; b.y=cy+ny*BR;
  } else { nx=dx/d; ny=dy/d; b.x=cx+nx*BR; b.y=cy+ny*BR; }
  const vn=b.vx*nx+b.vy*ny;
  if(vn<0){ b.vx-=1.94*vn*nx; b.vy-=1.94*vn*ny; return -vn; }
  return 0;
}
function stepBall(dt){
  if(ball.rest)return;
  let sp=Math.hypot(ball.vx,ball.vy);
  if(sp<0.35){ball.vx=ball.vy=0;ball.rest=true;onRest();return;}
  const steps=clamp(Math.ceil(sp*dt/(BR*.5)),1,10), h=dt/steps; let wallHit=0;
  for(let i=0;i<steps;i++){
    ball.x+=ball.vx*h; ball.y+=ball.vy*h;
    for(const w of walls){const imp=hitRect(ball,w); if(imp>wallHit)wallHit=imp;}
    for(const b of bumpers){
      const dx=ball.x-b.x,dy=ball.y-b.y,d=Math.hypot(dx,dy),rr=BR+b.r;
      if(d<rr){ let nx,ny; if(d<1e-5){nx=1;ny=0;}else{nx=dx/d;ny=dy/d;}
        ball.x=b.x+nx*rr; ball.y=b.y+ny*rr;
        if(b.cool<=0){ ball.vx=nx*9.5; ball.vy=ny*9.5; b.flash=1; b.cool=.22;
          G.shake=Math.max(G.shake,.45); sparks(ball.x,ball.y,8,'#ff7aa8',6);
          S.bumper(); if(G.state==='play')addScore(10); }
      }
    }
    const dmp=1-0.5*h; ball.vx*=dmp; ball.vy*=dmp;
  }
  sp=Math.hypot(ball.vx,ball.vy);
  if(sp<1.6){const ns=Math.max(0,sp-1.5*dt),f=ns/Math.max(sp,1e-4); ball.vx*=f; ball.vy*=f;}
  if(wallHit>1.2&&Math.random()<.7)S.wall(Math.min(1,wallHit/9));
  ball.trail.push({x:ball.x,y:ball.y}); if(ball.trail.length>26)ball.trail.shift();
}

/* ============================== UPDATE ============================== */
function update(dt){
  tNow+=dt;
  G.shake=Math.max(0,G.shake-2.6*dt);
  for(const b of bumpers){b.flash=Math.max(0,b.flash-3.2*dt); b.cool-=dt;}
  for(let i=parts.length-1;i>=0;i--){const p=parts[i];
    p.x+=p.vx*dt; p.y+=p.vy*dt; p.vx*=1-1.6*dt; p.vy*=1-1.6*dt; p.life-=dt;
    if(p.life<=0)parts.splice(i,1);}
  for(let i=pops.length-1;i>=0;i--){const p=pops[i]; p.y-=dt*.8; p.life-=dt; if(p.life<=0)pops.splice(i,1);}

  if(G.state==='menu'){
    if(ball.rest){ G.demoWait+=dt;
      if(G.demoWait>.9){ const sp=rnd(7,11);
        ball.vx=Math.cos(G.aim)*sp; ball.vy=Math.sin(G.aim)*sp; ball.rest=false; } }
    else stepBall(dt);
  }
  else if(G.state==='play'&&!G.paused){
    if(ball.rest){
      const L=keys['ArrowLeft']||keys['KeyA'], R=keys['ArrowRight']||keys['KeyD'];
      if(L)G.aim-=3.4*dt; if(R)G.aim+=3.4*dt;
      if(G.charging){ G.charge+=G.chgDir*1.25*dt;
        if(G.charge>1){G.charge=1;G.chgDir=-1;} if(G.charge<0){G.charge=0;G.chgDir=1;}
        chargeSndSet(G.charge); }
    }
    stepBall(dt);
    if(!ball.rest){ G.time-=dt; if(G.time<=0){G.time=0;gameOver();} }
    else if(ball.trail.length)ball.trail.splice(0,2);
    if(G.state==='play')checkPickups();
    if(portal&&portal.active&&Math.random()<dt*7){
      const a=rnd(0,7); parts.push({x:portal.x+Math.cos(a)*.4,y:portal.y+Math.sin(a)*.4,
        vx:Math.cos(a)*1.2,vy:Math.sin(a)*1.2,life:.5,max:.5,size:.03,col:'#7ef29a'});}
  }
  else if(G.state==='clear'){ G.clearT-=dt; if(G.clearT<=0)nextLevel(); }
  hudTime();
}

/* ============================== RENDER ============================== */
function rrp(c,x,y,w,h,r){ c.moveTo(x+r,y); c.arcTo(x+w,y,x+w,y+h,r);
  c.arcTo(x+w,y+h,x,y+h,r); c.arcTo(x,y+h,x,y,r); c.arcTo(x,y,x+w,y,r); c.closePath(); }

function render(){
  ctx.setTransform(DPR,0,0,DPR,0,0);
  ctx.clearRect(0,0,W,H);
  const shx=(Math.random()*2-1)*G.shake*7, shy=(Math.random()*2-1)*G.shake*7;
  ctx.save(); ctx.translate(ox+shx,oy+shy); ctx.scale(scale,scale);

  // playfield panel + dot grid
  ctx.beginPath(); rrp(ctx,-.55,-.55,cols+1.1,rows+1.1,.3); ctx.fillStyle='#101738'; ctx.fill();
  for(let i=0;i<cols;i++)for(let j=0;j<rows;j++){
    ctx.globalAlpha=.09+.07*Math.sin(tNow*1.3+i*.7+j*1.1);
    ctx.fillStyle='#7896e6'; ctx.fillRect(i+.48,j+.48,.05,.05);}
  ctx.globalAlpha=1;

  // portal
  if(portal){ const p=portal, act=p.active;
    const rr2=.34+(act?.05*Math.sin(tNow*6):0);
    const g=ctx.createRadialGradient(p.x,p.y,0,p.x,p.y,.62);
    g.addColorStop(0,act?'rgba(126,242,154,.5)':'rgba(90,109,168,.3)'); g.addColorStop(1,'rgba(0,0,0,0)');
    ctx.fillStyle=g; ctx.beginPath(); ctx.arc(p.x,p.y,.62,0,7); ctx.fill();
    ctx.setLineDash([.14,.1]); ctx.lineDashOffset=-tNow*(act?.9:.35);
    ctx.lineWidth=.08; ctx.strokeStyle=act?'#7ef29a':'#5a6da8';
    ctx.beginPath(); ctx.arc(p.x,p.y,rr2,0,7); ctx.stroke();
    if(act){ctx.lineDashOffset=tNow*1.3; ctx.lineWidth=.04;
      ctx.beginPath(); ctx.arc(p.x,p.y,rr2*.6,0,7); ctx.stroke();}
    ctx.setLineDash([]);
  }

  // orbs
  for(const o of orbs){ if(o.got)continue;
    const y=o.y+Math.sin(tNow*3+o.seed)*.05;
    const g=ctx.createRadialGradient(o.x,y,0,o.x,y,.34);
    g.addColorStop(0,'rgba(76,201,240,.55)'); g.addColorStop(1,'rgba(76,201,240,0)');
    ctx.fillStyle=g; ctx.beginPath(); ctx.arc(o.x,y,.34,0,7); ctx.fill();
    ctx.fillStyle='#eafaff'; ctx.beginPath(); ctx.arc(o.x,y,.13,0,7); ctx.fill();
    ctx.lineWidth=.035; ctx.strokeStyle='#4cc9f0'; ctx.beginPath(); ctx.arc(o.x,y,.19,0,7); ctx.stroke();
    const a=tNow*4+o.seed;
    ctx.fillStyle='#fff'; ctx.beginPath(); ctx.arc(o.x+Math.cos(a)*.24,y+Math.sin(a)*.24,.035,0,7); ctx.fill();
  }

  // walls (one combined path)
  ctx.beginPath(); for(const w of walls)rrp(ctx,w.x,w.y,w.w,w.h,T*.42);
  ctx.fillStyle='#3a4a86'; ctx.fill();
  ctx.lineWidth=.03; ctx.strokeStyle='rgba(255,209,102,'+(.2+.1*Math.sin(tNow*2))+')'; ctx.stroke();

  // bumpers
  for(const b of bumpers){
    const r=b.r*(1+b.flash*.5);
    const g=ctx.createRadialGradient(b.x-r*.3,b.y-r*.3,r*.1,b.x,b.y,r);
    g.addColorStop(0,'#ffd0e2'); g.addColorStop(.55,'#ff5d8f'); g.addColorStop(1,'#a32457');
    ctx.fillStyle=g; ctx.beginPath(); ctx.arc(b.x,b.y,r,0,7); ctx.fill();
    ctx.lineWidth=.03; ctx.strokeStyle='rgba(255,255,255,'+(.35+b.flash*.6)+')';
    ctx.beginPath(); ctx.arc(b.x,b.y,r*.85,0,7); ctx.stroke();
    if(b.flash>0){ ctx.lineWidth=.06; ctx.strokeStyle='rgba(255,140,180,'+b.flash+')';
      ctx.beginPath(); ctx.arc(b.x,b.y,r+(1-b.flash)*.5,0,7); ctx.stroke(); }
  }

  // frame + screws
  ctx.lineWidth=.06; ctx.strokeStyle='#f5b942';
  ctx.beginPath(); rrp(ctx,-.55,-.55,cols+1.1,rows+1.1,.3); ctx.stroke();
  ctx.lineWidth=.14; ctx.strokeStyle='rgba(245,185,66,.12)';
  ctx.beginPath(); rrp(ctx,-.62,-.62,cols+1.24,rows+1.24,.35); ctx.stroke();
  const cs=[[-.37,-.37],[cols+.37,-.37],[-.37,rows+.37],[cols+.37,rows+.37]];
  for(const [x,y] of cs){ ctx.fillStyle='#c99b3f'; ctx.beginPath(); ctx.arc(x,y,.09,0,7); ctx.fill();
    ctx.lineWidth=.02; ctx.strokeStyle='#6e4d15';
    ctx.beginPath(); ctx.moveTo(x-.05,y); ctx.lineTo(x+.05,y); ctx.moveTo(x,y-.05); ctx.lineTo(x,y+.05); ctx.stroke(); }

  // trail
  const tr=ball.trail, n=tr.length;
  for(let i=1;i<n;i++){ const a=i/n;
    ctx.lineWidth=.04+.16*a; ctx.strokeStyle='rgba(255,224,150,'+(a*.5)+')';
    ctx.beginPath(); ctx.moveTo(tr[i-1].x,tr[i-1].y); ctx.lineTo(tr[i].x,tr[i].y); ctx.stroke(); }

  // ball
  { const g=ctx.createRadialGradient(ball.x,ball.y,0,ball.x,ball.y,BR*3.2);
    g.addColorStop(0,'rgba(255,190,80,.5)'); g.addColorStop(1,'rgba(255,190,80,0)');
    ctx.fillStyle=g; ctx.beginPath(); ctx.arc(ball.x,ball.y,BR*3.2,0,7); ctx.fill();
    const c=ctx.createRadialGradient(ball.x-BR*.3,ball.y-BR*.35,BR*.1,ball.x,ball.y,BR);
    c.addColorStop(0,'#ffffff'); c.addColorStop(.6,'#ffe9b0'); c.addColorStop(1,'#ffb703');
    ctx.fillStyle=c; ctx.beginPath(); ctx.arc(ball.x,ball.y,BR,0,7); ctx.fill();
    ctx.lineWidth=.02; ctx.strokeStyle='rgba(255,255,255,.6)';
    ctx.beginPath(); ctx.arc(ball.x-BR*.3,ball.y-BR*.35,BR*.28,-2.4,-1.1); ctx.stroke(); }

  // aim + charge ring
  if(G.state==='play'&&!G.paused&&ball.rest){
    const a=G.aim, L=.75+G.charge*1.5, x0=ball.x+Math.cos(a)*(BR+.08), y0=ball.y+Math.sin(a)*(BR+.08);
    const x1=ball.x+Math.cos(a)*(BR+.08+L), y1=ball.y+Math.sin(a)*(BR+.08+L);
    ctx.setLineDash([.12,.1]); ctx.lineWidth=.05; ctx.strokeStyle='rgba(255,209,102,.9)';
    ctx.beginPath(); ctx.moveTo(x0,y0); ctx.lineTo(x1,y1); ctx.stroke(); ctx.setLineDash([]);
    ctx.fillStyle='#ffd166'; ctx.save(); ctx.translate(x1,y1); ctx.rotate(a);
    ctx.beginPath(); ctx.moveTo(.18,0); ctx.lineTo(-.09,.11); ctx.lineTo(-.09,-.11); ctx.closePath(); ctx.fill();
    ctx.restore();
    if(G.charging){ ctx.lineWidth=.07; ctx.strokeStyle='#ff5d8f';
      ctx.beginPath(); ctx.arc(ball.x,ball.y,BR+.16,-Math.PI/2,-Math.PI/2+G.charge*Math.PI*2); ctx.stroke(); }
  }

  // particles + score pops
  for(const p of parts){ ctx.globalAlpha=Math.max(0,p.life/p.max);
    ctx.fillStyle=p.col; ctx.beginPath(); ctx.arc(p.x,p.y,p.size,0,7); ctx.fill(); }
  ctx.globalAlpha=1;
  ctx.font='700 .26px ui-monospace,Menlo,Consolas,monospace'; ctx.textAlign='center';
  for(const p of pops){ ctx.globalAlpha=Math.max(0,p.life/p.max); ctx.fillStyle=p.col;
    ctx.fillText(p.txt,p.x,p.y); }
  ctx.globalAlpha=1; ctx.restore();

  // vignette
  if(vig){ctx.fillStyle=vig; ctx.fillRect(0,0,W,H);}
}
