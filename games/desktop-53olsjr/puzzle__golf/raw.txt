Collision routine per solid rect (walls + current mover rects):

```js
function collideRect(b, rx,ry,rw,rh){
  const cx=clamp(b.x,rx,rx+rw), cy=clamp(b.y,ry,ry+rh);
  let dx=b.x-cx, dy=b.y-cy; const d2=dx*dx+dy*dy;
  if(d2 > BALL_R*BALL_R) return 0;
  let d=Math.sqrt(d2), nx,ny;
  if(d<1e-6){ // center inside: push out along smallest axis
    const l=b.x-rx, r=rx+rw-b.x, t=b.y-ry, bo=ry+rh-b.y;
    const m=Math.min(l,r,t,bo);
    if(m===l){nx=-1;ny=0;b.x=rx-BALL_R} else if(m===r){nx=1;ny=0;b.x=rx+rw+BALL_R}
    else if(m===t){nx=0;ny=-1;b.y=ry-BALL_R} else {nx=0;ny=1;b.y=ry+rh+BALL_R}
  } else { nx=dx/d; ny=dy/d; b.x=cx+nx*BALL_R; b.y=cy+ny*BALL_R; }
  const vn=b.vx*nx+b.vy*ny;
  if(vn<0){ b.vx-=(1+REST_WALL)*vn*nx; b.vy-=(1+REST_WALL)*vn*ny; return -vn; }
  return 0;
}
```

```html
function inRect(x,y,s){return x>=s[0]&&x<=s[0]+s[4]&&y>=s[1]&&y<=s[1]+s[5];}

let stepHits=[];
function collideRect(b,rx,ry,rw,rh){
  const cx=clamp(b.x,rx,rx+rw),cy=clamp(b.y,ry,ry+rh);
  let dx=b.x-cx,dy=b.y-cy;const d2=dx*dx+dy*dy;
  if(d2>BALL_R*BALL_R)return;
  let d=Math.sqrt(d2),nx,ny;
  if(d<1e-6){
    const l=b.x-rx,r=rx+rw-b.x,t=b.y-ry,bt=ry+rh-b.y,m=Math.min(l,r,t,bt);
    if(m===l){nx=-1;ny=0;b.x=rx-BALL_R;}else if(m===r){nx=1;ny=0;b.x=rx+rw+BALL_R;}
    else if(m===t){nx=0;ny=-1;b.y=ry-BALL_R;}else{nx=0;ny=1;b.y=ry+rh+BALL_R;}
  }else{nx=dx/d;ny=dy/d;b.x=cx+nx*BALL_R;b.y=cy+ny*BALL_R;}
  const vn=b.vx*nx+b.vy*ny;
  if(vn<0){b.vx-=(1+REST_WALL)*vn*nx;b.vy-=(1+REST_WALL)*vn*ny;if(-vn>45)stepHits.push({x:cx,y:cy,nx,ny,i:-vn});}
}

function physics(h){
  stepHits=[];
  for(const s of cur.slopes)if(inRect(ball.x,ball.y,s)){ball.vx+=s[4]*s[6]*h;ball.vy+=s[5]*s[6]*h;}
  ball.x+=ball.vx*h;ball.y+=ball.vy*h;
  const f=Math.exp(-FRICTION*h);ball.vx*=f;ball.vy*=f;
  let sp=hyp(ball.vx,ball.vy);
  if(sp>0){sp=Math.max(0,sp-LIN_DECEL*h);const k=sp/hyp(ball.vx,ball.vy)||0;ball.vx*=k;ball.vy*=k;}
  // boundary
  if(ball.x<BALL_R){ball.x=BALL_R;if(ball.vx<0){if(-ball.vx>45)stepHits.push({x:0,y:ball.y,nx:1,ny:0,i:-ball.vx});ball.vx*=-REST_WALL;}}
  if(ball.x>CW-BALL_R){ball.x=CW-BALL_R;if(ball.vx>0){if(ball.vx>45)stepHits.push({x:CW,y:ball.y,nx:-1,ny:0,i:ball.vx});ball.vx*=-REST_WALL;}}
  if(ball.y<BALL_R){ball.y=BALL_R;if(ball.vy<0){if(-ball.vy>45)stepHits.push({x:ball.x,y:0,nx:0,ny:1,i:-ball.vy});ball.vy*=-REST_WALL;}}
  if(ball.y>CH-BALL_R){ball.y=CH-BALL_R;if(ball.vy>0){if(ball.vy>45)stepHits.push({x:ball.x,y:CH,nx:0,ny:-1,i:ball.vy});ball.vy*=-REST_WALL;}}
  for(const w of allRects())collideRect(ball,w.x,w.y,w.w,w.h);
  // bumpers
  for(const b of cur.bumpers){
    let dx=ball.x-b.x,dy=ball.y-b.y,d=hyp(dx,dy),min=BALL_R+b.r;
    if(d<min){
      if(d<1e-6){dx=1;d=1;}
      const nx=dx/d,ny=dy/d;ball.x=b.x+nx*min;ball.y=b.y+ny*min;
      const vn=ball.vx*nx+ball.vy*ny;
      if(vn<0){ball.vx-=(1+REST_BUMP)*vn*nx;ball.vy-=(1+REST_BUMP)*vn*ny;}
      let s2=hyp(ball.vx,ball.vy);if(s2<240){const k=240/(s2||1);ball.vx*=k;ball.vy*=k;}
      b.pulse=1;
      if(T-lastBumpT<.9)bumperChain++;else bumperChain=0;lastBumpT=T;
      Snd.bump(Math.min(bumperChain,8));
      shake=Math.min(13,shake+Math.min(5,60/ (b.r)) * .9);
      sparkFx(ball.x,ball.y,nx,ny,7,'#ffc94b');ringFx(b.x,b.y,b.r*.6,b.r*2.1,'rgba(255,201,75,.85)',3);
    }
  }
  // portals
  if(portalCd>0)portalCd-=h;
  else for(const p of cur.portals){
    const dx=ball.x-p[0],dy=ball.y-p[1];
    if(hyp(dx,dy)<PORT_R-4){
      const d=hyp(dx,dy)||1,nx=dx/d,ny=dy/d;
      ringFx(p[0],p[1],6,40,'rgba(255,201,75,.9)',3);ringFx(p[2],p[3],6,40,'rgba(63,216,194,.9)',3);
      ball.x=p[2]+nx*10;ball.y=p[3]+ny*10;portalCd=.35;Snd.portal();shake=Math.min(13,shake+3);
      break;
    }
  }
  // water
  for(const w of cur.water){
    if(ball.x>w[0]&&ball.x<w[0]+w[2]&&ball.y>w[1]&&ball.y<w[1]+w[3]){splash();break;}
  }
  // cup
  const cdx=ball.x-cur.cup[0],cdy=ball.y-cur.cup[1],d=hyp(cdx,cdy),spd=hyp(ball.vx,ball.vy);
  if(d<30&&d>0.001){const a=700*(1-d/30)*h;ball.vx-=cdx/d*a;ball.vy-=cdy/d*a;}
  if(d<13&&spd<240)sink();
  // rest
  const s3=hyp(ball.vx,ball.vy);
  if(s3<7){ball.vx=ball.vy=0;if(bumperChain>0)bumperChain=0;
    if(strokes>0&&strokes>=CAP)finishHole(true);}
}
function splash(){
  Snd.splash();shake=Math.min(13,shake+5);
  for(let i=0;i<18;i++){const a=Math.random()*TAU,s=60+Math.random()*220;
    parts.push({x:ball.x,y:ball.y,vx:Math.cos(a)*s,vy:Math.sin(a)*s-140,l:.7,c:'#7fc4ff',r:2+Math.random()*3,g:520});}
  ringFx(ball.x,ball.y,8,46,'rgba(127,196,255,.9)',3);
  strokes++;lockT=.5;ball.x=lastShot.x;ball.y=lastShot.y;ball.vx=ball.vy=0;
  showToast('SPLASH!','+1 penalty stroke');updateHUD();
}
function sink(){
  if(state!=='play')return;
  state='sinking';sinkT=0;sunkFrom={x:ball.x,y:ball.y};flagPulse=1;
  Snd.sink(strokes<cur.par);shake=Math.min(13,shake+5);
}

/* ============================== fx ============================== */
function sparkFx(x,y,nx,ny,n,col){for(let i=0;i<n;i++){const a=Math.atan2(ny,nx)+(Math.random()-.5)*2.2,s=90+Math.random()*260;parts.push({x,y,vx:Math.cos(a)*s,vy:Math.sin(a)*s,l:.45,c:col,r:1.6+Math.random()*2,g:0});}}
function ringFx(x,y,r0,r1,col,w){rings.push({x,y,r0,r1,l:1,col,w});}
const CONF_COLS=['#ffc94b','#ff5340','#3fd8c2','#f6ecd4'];
function spawnConfetti(x,y,n){for(let i=0;i<n;i++){const a=Math.random()*TAU,s=120+Math.random()*340;confetti.push({x,y,vx:Math.cos(a)*s,vy:Math.sin(a)*s-260,rot:Math.random()*TAU,vr:(Math.random()-.5)*12,c:CONF_COLS[i%4],w:4+Math.random()*4,h:3+Math.random()*3,l:1.4});}}
function updateFx(dt){
  shake*=Math.exp(-6.5*dt);if(shake<.05)shake=0;
  flagPulse*=Math.exp(-2.4*dt);
  lockT-=dt;
  for(const b of cur.bumpers)b.pulse*=Math.exp(-7*dt);
  for(const m of cur.movers)m.cy=moverY(m);
  parts=parts.filter(p=>(p.l-=dt*1.8)>0);
  for(const p of parts){p.x+=p.vx*dt;p.y+=p.vy*dt;p.vy+=(p.g||0)*dt;}
  rings=rings.filter(r=>(r.l-=dt*2.4)>0);
  confetti=confetti.filter(c=>(c.l-=dt*.85)>0);
  for(const c of confetti){c.x+=c.vx*dt;c.y+=c.vy*dt;c.vy+=560*dt;c.rot+=c.vr*dt;c.vx*=Math.exp(-.9*dt);}
  const sp=hyp(ball.vx,ball.vy);
  if(sp>90)trail.push({x:ball.x,y:ball.y,l:1});
  trail=trail.filter(t=>(t.l-=dt*2.4)>0);
  for(const m of motes){m.y-=m.s*dt;m.x+=Math.sin(T*.7+m.ph)*6*dt;if(m.y<-8){m.y=CH+8;m.x=Math.random()*CW;}}
  if(state==='sinking'){sinkT+=dt;if(sinkT>=.5)finishHole(false);}
}

/* ============================== preview sim ============================== */
function simPath(x,y,vx,vy,tMax){
  const pts=[];let hit=false,cupEnd=false,h=1/90,n=Math.ceil(tMax/h),px=x,py=y,pvx=vx,pvy=vy;
  for(let i=0;i<n&&!hit;i++){
    for(const s of cur.slopes)if(inRect(px,py,s)){pvx+=s[4]*s[6]*h;pvy+=s[5]*s[6]*h;}
    const f=Math.exp(-FRICTION*h);pvx*=f;pvy*=f;px+=pvx*h;py+=pvy*h;
    if(px<BALL_R){px=BALL_R;pvx*=-REST_WALL;hit=true;}if(px>CW-BALL_R){px=CW-BALL_R;pvx*=-REST_WALL;hit=true;}
    if(py<BALL_R){py=BALL_R;pvy*=-REST_WALL;hit=true;}if(py>CH-BALL_R){py=CH-BALL_R;pvy*=-REST_WALL;hit=true;}
    for(const w of allRects()){
      const cx=clamp(px,w.x,w.x+w.w),cy=clamp(py,w.y,w.y+w.h);
      let dx=px-cx,dy=py-cy,d=hyp(dx,dy);
      if(d<BALL_R){if(d<1e-6)dx=1;const nx=dx/(d||1),ny=dy/(d||1);px=cx+nx*BALL_R;py=cy+ny*BALL_R;const vn=pvx*nx+pvy*ny;if(vn<0){pvx-=(1+REST_WALL)*vn*nx;pvy-=(1+REST_WALL)*vn*ny;}hit=true;}
    }
    for(const b of cur.bumpers){let dx=px-b.x,dy=py-b.y,d=hyp(dx,dy);if(d<BALL_R+b.r){const nx=dx/(d||1),ny=dy/(d||1);px=b.x+nx*(BALL_R+b.r);py=b.y+ny*(BALL_R+b.r);const vn=pvx*nx+pvy*ny;if(vn<0){pvx-=(1+REST_BUMP)*vn*nx;pvy-=(1+REST_BUMP)*vn*ny;}hit=true;}}
    const cd=hyp(px-cur.cup[0],py-cur.cup[1]);
    if(cd<14&&hyp(pvx,pvy)<280){cupEnd=true;hit=true;}
    if(i%3===0)pts.push({x:px,y:py});
  }
  return{pts,hit,cupEnd};
}

/* ============================== update ============================== */
function update(dt){
  T+=dt;
  updateFx(dt);
  for(const m of cur.movers)m.cy=moverY(m);
  if(state==='play'){
    let acc=Math.min(dt,.05),hh=1/120;
    while(acc>1e-6){const s=Math.min(hh,acc);physics(s);acc-=s;}
  }
  // flag height
  const dCup=hyp(ball.x-cur.cup[0],ball.y-cur.cup[1]);
  const target=(state==='sinking'||state==='result')?20:(dCup<75?32:54);
  flagH+=(target-flagH)*Math.min(1,dt*6);
}

function drawBall(){
  if(state==='result')return;
  let x=ball.x,y=ball.y,scl=1;
  if(state==='sinking'){const k=clamp(sinkT/.42,0,1);x=lerp(sunkFrom.x,cur.cup[0],k);y=lerp(sunkFrom.y,cur.cup[1],k);scl=1-k*.9;}

  ctx.fillStyle='rgba(0,0,0,.3)';ctx.beginPath();ctx.ellipse(x+3,y+5,BALL_R*scl,BALL_R*.6*scl,0,0,TAU);ctx.fill();
  ctx.save();ctx.translate(x,y);
  if(sp>160&&state!=='sinking'){const a=Math.atan2(ball.vy,ball.vx),k=Math.min(.16,sp/4500);ctx.rotate(a);ctx.scale(1+k,1-k*.6);}
  ctx.scale(scl,scl);
  const g=ctx.createRadialGradient(-3.5,-4,1,0,0,BALL_R+1);
  g.addColorStop(0,'#ffffff');g.addColorStop(.75,'#f2efe6');g.addColorStop(1,'#c9c4b4');
  ctx.fillStyle=g;ctx.beginPath();ctx.arc(0,0,BALL_R,0,TAU);ctx.fill();
  ctx.strokeStyle='rgba(60,50,30,.35)';ctx.lineWidth=1;ctx.stroke();
  ctx.restore();
}

function drawAim(){
  if(!aiming||!ballAtRest())return;
  const dx=ball.x-aimPt.x,dy=ball.y-aimPt.y,len=hyp(dx,dy);
  if(len<6)return;
  const nx=dx/len,ny=dy/len,pow=MIN_POW+(MAX_POW-MIN_POW)*Math.pow(clamp(len/DRAG_MAX,0,1),1.2),frac=pow/MAX_POW;
  // pull band
  ctx.strokeStyle='rgba(246,236,212,.5)';ctx.lineWidth=3;ctx.setLineDash([7,7]);
  ctx.beginPath();ctx.moveTo(aimPt.x,aimPt.y);ctx.lineTo(ball.x,ball.y);ctx.stroke();ctx.setLineDash([]);
  // preview path
  const p=simPath(ball.x,ball.y,nx*pow,ny*pow,.35+.5*frac);
  const n=p.pts.length;
  for(let i=0;i<n;i++){const t=i/Math.max(1,n-1);
    ctx.globalAlpha=(1-t)*.75;ctx.fillStyle='#f6ecd4';
    ctx.beginPath();ctx.arc(p.pts[i].x,p.pts[i].y,4-t*2.2,0,TAU);ctx.fill();}
  ctx.globalAlpha=1;

  else if(p.hit&&n>2){const e=p.pts[n-1],s=5;ctx.strokeStyle='rgba(255,83,64,.9)';ctx.lineWidth=2.5;
    ctx.beginPath();ctx.moveTo(e.x-s,e.y-s);ctx.lineTo(e.x+s,e.y+s);ctx.moveTo(e.x+s,e.y-s);ctx.lineTo(e.x-s,e.y+s);ctx.stroke();}
  // arrow head

  ctx.fillStyle='#f6ecd4';ctx.beginPath();
  ctx.moveTo(ax+nx*9,ay+ny*9);ctx.lineTo(ax-px*5.5,ay-py*5.5);ctx.lineTo(ax+px*5.5,ay+py*5.5);ctx.closePath();ctx.fill();

}
```
