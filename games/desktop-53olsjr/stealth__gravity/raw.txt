Segment vs rect intersection (Liang-Barsky):
```js
function segRect(x1,y1,x2,y2,r){
  let t0=0,t1=1; const dx=x2-x1, dy=y2-y1;
  const p=[-dx,dx,-dy,dy], q=[x1-r.x, r.x+r.w-x1, y1-r.y, r.y+r.h-y1];
  for(let i=0;i<4;i++){
    if(p[i]===0){ if(q[i]<0) return true; }
    else{ const t=q[i]/p[i];
      if(p[i]<0){ if(t>t1) return true; if(t>t0) t0=t; }
      else { if(t<t0) return true; if(t<t1) t1=t; }
    }
  }
  return false; // no intersection → LOS clear
}
```
Careful: this returns true if segment intersects the interior of rect (t range overlaps [0,1]). Good.

  const nx = clamp(c.x, r.x, r.x+r.w), ny = clamp(c.y, r.y, r.y+r.h);

function losBlocked(x1,y1,x2,y2,walls){
  const dx=x2-x1, dy=y2-y1;
  for(const w of walls){
    // Liang-Barsky: clip segment against rect [w.x..w.x+w.w] etc. If t interval empty → no intersection.
    let t0=0,t1=1;
    const p=[-dx,dx,-dy,dy];
    const q=[x1-w.x, (w.x+w.w)-x1, y1-w.y, (w.y+w.h)-y1];
    let ok=true;
    for(let i=0;i<4;i++){
      if(p[i]===0){ if(q[i]<0){ok=false;break;} }
      else{ const t=q[i]/p[i];
        if(p[i]<0){ if(t>t1){ok=false;break;} if(t>t0)t0=t; }
        else{ if(t<t0){ok=false;break;} if(t<t1)t1=t; }
      }
    }
    if(ok) return true; // segment passes through rect interior
  }
  return false;
}

/* ============================== PHYSICS HELPERS ============================== */
function collideRect(c,r,rest){
  const nx=clamp(c.x,r.x,r.x+r.w), ny=clamp(c.y,r.y,r.y+r.h);
  let dx=c.x-nx, dy=c.y-ny; const d2=dx*dx+dy*dy;
  if(d2>c.r*c.r)return false;
  if(d2>1e-9){
    const d=Math.sqrt(d2),ux=dx/d,uy=dy/d;
    c.x=nx+ux*c.r; c.y=ny+uy*c.r;
    const vn=c.vx*ux+c.vy*uy;
    if(vn<0){c.vx-=(1+rest)*vn*ux; c.vy-=(1+rest)*vn*uy;}
    return true;
  }
  const l=c.x-r.x,rr=r.x+r.w-c.x,t=c.y-r.y,b=r.y+r.h-c.y,m=Math.min(l,rr,t,b);
  if(m===l)c.x=r.x-c.r; else if(m===rr)c.x=r.x+r.w+c.r; else if(m===t)c.y=r.y-c.r; else c.y=r.y+r.h+c.r;
  return true;
}
function losBlocked(x1,y1,x2,y2,walls){
  const dx=x2-x1,dy=y2-y1;
  for(const w of walls){
    let t0=0,t1=1; const p=[-dx,dx,-dy,dy];
    const q=[x1-w.x,(w.x+w.w)-x1,y1-w.y,(w.y+w.h)-y1];
    let ok=true;
    for(let i=0;i<4;i++){
      if(p[i]===0){ if(q[i]<0){ok=false;break;} }
      else{ const t=q[i]/p[i];
        if(p[i]<0){ if(t>t1){ok=false;break;} if(t>t0)t0=t; }
        else{ if(t<t0){ok=false;break;} if(t<t1)t1=t; } }
    }
    if(ok)return true;
  }
  return false;
}

function burst(x,y,n,col){
  for(let i=0;i<n;i++){const a=rand(0,Math.PI*2),s=rand(40,260);
    G.L.sparks.push({x,y,vx:Math.cos(a)*s,vy:Math.sin(a)*s,life:rand(.3,.8),max:.8,col});}
}
function respawnDebris(d){let x,y,t=0;do{x=rand(30,W-30);y=rand(30,H-30);t++;}while(pointInWalls(x,y,G.L.walls)&&t<20);d.x=x;d.y=y;d.vx=d.vy=0;}

function tick(dt){
  G.t+=dt; const L=G.L; if(!L)return;
  if(G.msgT>0)G.msgT-=dt;
  if(G.flash>0)G.flash=Math.max(0,G.flash-dt*1.6);

  // ambient: dust, debris, turret sweep
  for(const d of L.dust){d.x+=Math.sin(G.t*.25+d.seed)*7*d.z*dt; d.y+=Math.cos(G.t*.2+d.seed)*5*d.z*dt;
    if(d.x<0)d.x+=W; if(d.x>W)d.x-=W; if(d.y<0)d.y+=H; if(d.y>H)d.y-=H;}
  for(const t of L.turrets)t.ang=t.base+Math.sin((G.t+t.seed)*t.speed)*t.amp;

  const p=L.player, inPod=L.pods.some(pd=>{const dx=p.x-pd.x,dy=p.y-pd.y;return dx*dx+dy*dy<pd.r*pd.r;});
  const wellActive=G.state==='play'&&!(inPod)&&(keys.has('Space')||mouse.down);

  for(const d of L.debris){
    d.vx+=Math.sin(G.t*.5+d.seed)*10*dt; d.vy+=Math.cos(G.t*.42+d.seed*1.3)*8*dt;
    if(wellActive){const dx=p.x-d.x,dy=p.y-d.y,dd=Math.hypot(dx,dy);
      if(dd<210&&dd>6){d.vx+=dx/dd*950*dt;d.vy+=dy/dd*950*dt;if(dd<13)respawnDebris(d);}}
    for(const a of L.anchors){const dx=a.x-d.x,dy=a.y-d.y,dd=Math.hypot(dx,dy);
      if(dd<a.r&&dd>4){d.vx+=dx/dd*320*dt;d.vy+=dy/dd*320*dt;}}
    d.vx*=Math.exp(-2.4*dt); d.vy*=Math.exp(-2.4*dt);
    const sp=Math.hypot(d.vx,d.vy); if(sp>280){d.vx*=280/sp;d.vy*=280/sp;}
    d.x+=d.vx*dt; d.y+=d.vy*dt;
    if(d.x<20||d.x>W-20){d.vx*=-1;d.x=clamp(d.x,20,W-20);}
    if(d.y<20||d.y>H-20){d.vy*=-1;d.y=clamp(d.y,20,H-20);}
  }
  for(let i=L.sparks.length-1;i>=0;i--){const s=L.sparks[i];
    s.x+=s.vx*dt;s.y+=s.vy*dt;s.vx*=Math.exp(-3*dt);s.vy*=Math.exp(-3*dt);s.life-=dt;
    if(s.life<=0)L.sparks.splice(i,1);}

  // guards: full AI in play, patrol-only otherwise
  for(const g of L.guards){
    let mvx=0,mvy=0;
    if(G.state==='play'&&g.state==='HOT'){
      const t=g.target,dx=t.x-g.x,dy=t.y-g.y,dd=Math.hypot(dx,dy);
      const sp=g.seen>0?150:110;
      if(dd>8){mvx=dx/dd*sp;mvy=dy/dd*sp;} else g.face+=Math.sin(G.t*2+g.seed)*dt*1.3;
    } else {
      const wp=g.pts[g.i],dx=wp[0]-g.x,dy=wp[1]-g.y,dd=Math.hypot(dx,dy);
      if(dd<6){ g.i+=g.dir;
        if(g.i>=g.pts.length){g.i=g.pts.length-2;g.dir=-1;} else if(g.i<0){g.i=1;g.dir=1;} }
      else {mvx=dx/dd*g.speed;mvy=dy/dd*g.speed;}
    }
    if(mvx||mvy)g.face=lerpAngle(g.face,Math.atan2(mvy,mvx),1-Math.exp(-8*dt));
    g.x+=mvx*dt; g.y+=mvy*dt;
    for(const w of L.walls)collideRect(g,w,0.1);
    if(G.state==='play'){
      // sight test
      const dx=p.x-g.x,dy=p.y-g.y,dist=Math.hypot(dx,dy);
      let sees=false;
      if(!inPod&&dist<g.range){
        let da=Math.atan2(dy,dx)-g.face; da=(da+Math.PI*3)%(Math.PI*2)-Math.PI;
        if(Math.abs(da)<g.fov/2 && !losBlocked(g.x,g.y,p.x,p.y,L.walls))sees=true;
      }
      if(sees){ g.seen+=dt; g.lastSeen={x:p.x,y:p.y};
        if(g.state!=='HOT'){g.state='HOT';L.silent=false;}
        g.target=g.lastSeen; g.hotT=3.0;
      } else { g.seen=0; if(g.state==='HOT'){ g.hotT-=dt;
          if(g.hotT<=0){g.state='PATROL';let bi=0,bd=1e18;
            for(let k=0;k<g.pts.length;k++){const q=(g.x-g.pts[k][0])**2+(g.y-g.pts[k][1])**2;if(q<bd){bd=q;bi=k;}}
            g.i=bi;} } }
    } else g.seen=0;
  }

  if(G.state!=='play')return;

  /* ---- player physics ---- */
  let ax=0,ay=0;
  if(wellActive){
    const dx=mouse.x-p.x,dy=mouse.y-p.y,d=Math.hypot(dx,dy)||1;
    const s=760*Math.min(1,d/240)+140;
    ax+=dx/d*s; ay+=dy/d*s;
    p.rippleT-=dt;
    if(p.rippleT<=0){ p.rippleT=RIPPLE_EVERY;
      L.ripples.push({x:p.x,y:p.y,ox:p.x,oy:p.y,r:6,hit:new Set()});
      blip(740,0.09,'triangle',0.045,470); }
  } else p.rippleT=Math.min(p.rippleT,RIPPLE_EVERY-0.12);

  let dirx=0,diry=0;
  if(keys.has('KeyA')||keys.has('ArrowLeft'))dirx-=1;
  if(keys.has('KeyD')||keys.has('ArrowRight'))dirx+=1;
  if(keys.has('KeyW')||keys.has('ArrowUp'))diry-=1;
  if(keys.has('KeyS')||keys.has('ArrowDown'))diry+=1;
  const drifting=(dirx||diry)&&p.charge>0.5;
  if(drifting){ const m=Math.hypot(dirx,diry);
    ax+=dirx/m*260*(inPod?0.55:1); ay+=diry/m*260*(inPod?0.55:1);
    p.charge=Math.max(0,p.charge-55*dt); }
  else p.charge=Math.min(100,p.charge+22*dt);

  for(const a of L.anchors){const dx=a.x-p.x,dy=a.y-p.y,d=Math.hypot(dx,dy);
    if(d<a.r&&d>2){ax+=dx/d*520;ay+=dy/d*520;}}

  p.vx+=ax*dt; p.vy+=ay*dt;
  const drag=Math.exp(-2.1*dt); p.vx*=drag; p.vy*=drag;
  const sp=Math.hypot(p.vx,p.vy); if(sp>330){p.vx*=330/sp;p.vy*=330/sp;}
  p.x+=p.vx*dt; p.y+=p.vy*dt;
  for(let it=0;it<2;it++)for(const w of L.walls)collideRect(p,w,0.25);
  if(p.x<PR+19){p.x=PR+19;p.vx*=-0.3;} if(p.x>W-PR-19){p.x=W-PR-19;p.vx*=-0.3;}
  if(p.y<PR+19){p.y=PR+19;p.vy*=-0.3;} if(p.y>H-PR-19){p.y=H-PR-19;p.vy*=-0.3;}

  p.trail.push({x:p.x,y:p.y}); if(p.trail.length>12)p.trail.shift();

  // ripples
  for(let i=L.ripples.length-1;i>=0;i--){ const r=L.ripples[i]; r.r+=RIP_SPEED*dt;
    if(r.r>HEAR){L.ripples.splice(i,1);continue;}
    for(const g of L.guards){ if(r.hit.has(g.id))continue;
      const d=Math.hypot(g.x-r.x,g.y-r.y);
      if(d<=r.r){ r.hit.add(g.id);
        if(g.state!=='HOT'){g.state='HOT';g.hotT=2.2;g.target={x:r.ox,y:r.oy};L.silent=false;}
        blip(300,0.1,'square',0.05);} } }

  // detection / catch (turrets)
  let det=0;
  for(const g of L.guards){ det=Math.max(det,g.seen/CATCH_TIME); if(g.seen>=CATCH_TIME){onCaught();return;} }
  for(const t of L.turrets){
    const dx=p.x-t.x,dy=p.y-t.y,dist=Math.hypot(dx,dy); let sees=false;
    if(!inPod&&dist<t.range){
      let da=Math.atan2(dy,dx)-t.ang; da=(da+Math.PI*3)%(Math.PI*2)-Math.PI;
      if(Math.abs(da)<t.fov/2 && !losBlocked(t.x,t.y,p.x,p.y,L.walls))sees=true; }
    t.seen=sees?t.seen+dt:0;
    det=Math.max(det,t.seen/CATCH_TIME);
    if(t.seen>=CATCH_TIME){onCaught();return;}
  }
  G.det=det;

  // shards
  const done=L.shards.every(s=>s.got);
  for(const s of L.shards){ if(s.got)continue;
    const d=Math.hypot(p.x-s.x,p.y-s.y);
    if(d<PR+13){ s.got=true; burst(s.x,s.y,14,'#ffd23f'); blip(880,0.1,'sine',0.12); setTimeout(()=>blip(1318,0.14,'sine',0.1),70);
      if(L.shards.every(q=>q.got)){ L.exit.pulse=1; G.msg='EXIT GATE ENERGIZED'; G.msgT=2.2; chord([660,990],80,0.1);} } }
  buildPips();

  // exit
  if(done&&Math.hypot(p.x-L.exit.x,p.y-L.exit.y)<L.exit.r+PR-4){onClear();return;}

  // audio envelopes
  if(AC){ humGain.gain.setTargetAtTime(wellActive?0.13:0,AC.currentTime,0.06);
          sirenGain.gain.setTargetAtTime(det>0.25?0.08:0,AC.currentTime,0.09); }

  L.time+=dt;
}

function drawCone(x,y,ang,fov,range,r,g,b,base){
  const grd=ctx.createRadialGradient(x,y,8,x,y,range);
  grd.addColorStop(0,`rgba(${r},${g},${b},${base})`);
  grd.addColorStop(1,`rgba(${r},${g},${b},0)`);
  ctx.fillStyle=grd; ctx.beginPath();
  ctx.moveTo(x,y); ctx.arc(x,y,range,ang-fov/2,ang+fov/2); ctx.closePath(); ctx.fill();
}

  for(const tr of L.turrets){
    ctx.fillStyle='#14303f'; ctx.strokeStyle=tr.seen>0?'#ff4b2e':'#3a6b82'; ctx.lineWidth=2;
    ctx.beginPath(); ctx.arc(tr.x,tr.y,13,0,7); ctx.fill(); ctx.stroke();
    ctx.save(); ctx.translate(tr.x,tr.y); ctx.rotate(tr.ang);
    ctx.strokeStyle='#ffd23f'; ctx.lineWidth=3;
    ctx.beginPath(); ctx.moveTo(4,0); ctx.lineTo(16,0); ctx.stroke(); ctx.restore();
    ctx.fillStyle='#ffd23f'; ctx.beginPath(); ctx.arc(tr.x,tr.y,3,0,7); ctx.fill();
  }
  // guards
  for(const g of L.guards){
    const col=g.seen>0?'#ff4b2e':(g.state==='HOT'?'#ff8a3c':'#ffb347');
    ctx.fillStyle='rgba(0,0,0,.35)';
    ctx.beginPath(); ctx.ellipse(g.x+3,g.y+6,13,6,0,0,7); ctx.fill();
    ctx.save(); ctx.translate(g.x,g.y); ctx.rotate(g.face);
    ctx.fillStyle=col; ctx.strokeStyle='rgba(0,0,0,.45)'; ctx.lineWidth=2;
    ctx.beginPath(); ctx.moveTo(16,0); ctx.lineTo(-9,10); ctx.lineTo(-5,0); ctx.lineTo(-9,-10); ctx.closePath();
    ctx.fill(); ctx.stroke();
    ctx.fillStyle='#fff7e0'; ctx.fillRect(4,-2,7,4);
    ctx.restore();
    if(g.state==='HOT'){ ctx.fillStyle=col; ctx.font='700 17px '+getComputedStyle(document.body).getPropertyValue('--display');
      ctx.textAlign='center'; ctx.fillText(g.seen>0?'!!':'!',g.x,g.y-20); }
  }

  // player
  { const p=L.player, demo=G.state==='title';
    const px=demo?L.player.x:p.x, py=demo?L.player.y+Math.sin(t*2)*4:p.y;
    if(!demo){ for(let i=0;i<p.trail.length-1;i++){ const tr=p.trail[i];
      ctx.fillStyle=`rgba(159,232,255,${i/p.trail.length*0.22})`;
      ctx.beginPath(); ctx.arc(tr.x,tr.y,3+i/p.trail.length*4,0,7); ctx.fill(); } }
    const inPod=!demo&&L.pods.some(pd=>{const dx=px-pd.x,dy=py-pd.y;return dx*dx+dy*dy<pd.r*pd.r;});
    if(inPod){ ctx.strokeStyle='rgba(75,227,160,.8)'; ctx.lineWidth=1.5;
      ctx.setLineDash([4,5]); ctx.lineDashOffset=-t*24;
      ctx.beginPath(); ctx.arc(px,py,16,0,7); ctx.stroke(); ctx.setLineDash([]); }
    const grd=ctx.createRadialGradient(px,py,1,px,py,13);
    grd.addColorStop(0,'#ffffff'); grd.addColorStop(.5,'#9fe8ff'); grd.addColorStop(1,'rgba(95,232,216,0)');
    ctx.fillStyle=grd; ctx.beginPath(); ctx.arc(px,py,13,0,7); ctx.fill();
    ctx.fillStyle='#fff'; ctx.beginPath(); ctx.arc(px,py,4.5,0,7); ctx.fill();
    if(!demo&&G.det>0){ ctx.strokeStyle=`rgba(255,75,46,${.5+.4*Math.sin(t*18)})`; ctx.lineWidth=3;
      ctx.beginPath(); ctx.arc(px,py,18,-Math.PI/2,-Math.PI/2+G.det*Math.PI*2); ctx.stroke(); }

    // field FX + reticle (play only)
    if(G.state==='play'){
      const wellActive=inPod?false:(keys.has('Space')||mouse.down);
      if(wellActive){
        const dx=mouse.x-px,dy=mouse.y-py,d=Math.hypot(dx,dy)||1,ux=dx/d,uy=dy/d;
        ctx.strokeStyle='rgba(159,232,255,.5)'; ctx.lineWidth=1.5;
        ctx.setLineDash([10,8]); ctx.lineDashOffset=-t*60;
        ctx.beginPath(); ctx.moveTo(px+ux*14,py+uy*14); ctx.lineTo(mouse.x-ux*12,mouse.y-uy*12); ctx.stroke(); ctx.setLineDash([]);
        for(let i=0;i<3;i++){ const f=0.3+i*0.26, cx=px+ux*d*f, cy=py+uy*d*f;
          ctx.fillStyle='rgba(159,232,255,.6)';
          ctx.save(); ctx.translate(cx,cy); ctx.rotate(Math.atan2(uy,ux));
          ctx.beginPath(); ctx.moveTo(7,0); ctx.lineTo(-4,5); ctx.lineTo(-4,-5); ctx.closePath(); ctx.fill(); ctx.restore(); }
        const baseA=Math.atan2(uy,ux);
        for(let i=0;i<3;i++){ ctx.strokeStyle=`rgba(159,232,255,${.34-i*.08})`; ctx.lineWidth=1.5;
          ctx.beginPath(); ctx.arc(px,py,15+i*7,baseA-1.6+t*3,baseA+1.6+t*3); ctx.stroke(); }
      }
      // reticle
      const rr=wellActive?9+2*Math.sin(t*12):9;
      const rc=wellActive?'rgba(255,179,71,.9)':'rgba(159,232,255,.8)';
      ctx.strokeStyle=rc; ctx.lineWidth=1.5;
      ctx.beginPath(); ctx.arc(mouse.x,mouse.y,rr,0,7); ctx.stroke();
      for(let i=0;i<4;i++){ const a=i*Math.PI/2+t*(wellActive?2:0.4);
        ctx.beginPath(); ctx.moveTo(mouse.x+Math.cos(a)*(rr+3),mouse.y+Math.sin(a)*(rr+3));
        ctx.lineTo(mouse.x+Math.cos(a)*(rr+8),mouse.y+Math.sin(a)*(rr+8)); ctx.stroke(); }
      ctx.fillStyle=rc; ctx.fillRect(mouse.x-1,mouse.y-1,2,2);
    }
  }

  // ripples
  for(const r of L.ripples){ const a=Math.max(0,1-r.r/HEAR);
    ctx.strokeStyle=`rgba(255,120,73,${a*0.55})`; ctx.lineWidth=2;
    ctx.beginPath(); ctx.arc(r.x,r.y,r.r,0,7); ctx.stroke();
    ctx.strokeStyle=`rgba(255,120,73,${a*0.25})`;
    ctx.beginPath(); ctx.arc(r.x,r.y,r.r*0.75,0,7); ctx.stroke(); }
  // sparks
  for(const s of L.sparks){ ctx.globalAlpha=Math.max(0,s.life/s.max);
    ctx.fillStyle=s.col; ctx.fillRect(s.x-2,s.y-2,4,4); ctx.globalAlpha=1; }

  // message toast
  if(G.msgT>0&&G.state!=='title'){ const a=Math.min(1,G.msgT);
    ctx.font='700 15px '+getComputedStyle(document.body).getPropertyValue('--display');
    ctx.textAlign='center'; ctx.fillStyle=`rgba(95,232,216,${a})`;
    ctx.fillText(G.msg.split('').join('\u200a'),W/2,74); }

  // vignette + detection flash
  const vg=ctx.createRadialGradient(W/2,H/2,H*0.42,W/2,H/2,H*0.85);
  vg.addColorStop(0,'rgba(0,0,0,0)'); vg.addColorStop(1,'rgba(0,0,0,.42)');
  ctx.fillStyle=vg; ctx.fillRect(0,0,W,H);
  if(G.state==='play'&&G.det>0){ ctx.fillStyle=`rgba(255,60,30,${G.det*0.22*(0.7+0.3*Math.sin(t*14))})`;
    ctx.fillRect(0,0,W,8); ctx.fillRect(0,H-8,W,8); ctx.fillRect(0,0,8,H); ctx.fillRect(W-8,0,8,H); }
  if(G.flash>0){ ctx.fillStyle=`rgba(255,240,235,${G.flash*0.75})`; ctx.fillRect(0,0,W,H); }

  // HUD DOM sync
  if(G.state==='play'){
    const p=L.player;
    chargeFill.style.width=p.charge+'%';
    chargeFill.classList.toggle('low',p.charge<28);
    document.getElementById('hud-score').textContent=String(G.run.score).padStart(4,'0');
    detBanner.classList.toggle('on',G.det>0.14);
  } else detBanner.classList.remove('on');
}
function buildPips(){
  const el=document.getElementById('hud-pips'); if(!el||!G.L)return;
  el.innerHTML=G.L.shards.map(s=>`<span class="pip${s.got?' got':''}"></span>`).join('');
}

/* ============================== MAIN LOOP ============================== */
let last=0,acc=0; const STEP=1/60;
function frame(ts){
  requestAnimationFrame(frame);
  const now=ts/1000; let dt=Math.min(0.1,now-last); last=now;
  if(G.state==='play'||G.state==='title'){ acc+=dt;
    while(acc>=STEP){ tick(STEP); acc-=STEP; } }
  render();
}
