function hitBrick(ball){
 for(const b of bricks){ if(b.dead)continue;
  const cx=b.x+b.w/2, cy=b.y+b.h/2;
  const ox=(b.w/2+ball.r)-Math.abs(ball.x-cx); if(ox<0)continue;
  const oy=(b.h/2+ball.r)-Math.abs(ball.y-cy); if(oy<0)continue;
  // resolve
  if(ox<oy){ ball.vx = ball.x<cx? -Math.abs(ball.vx):Math.abs(ball.vx); ball.x += (ball.x<cx?-ox:ox); }
  else { ball.vy = ball.y<cy? -Math.abs(ball.vy):Math.abs(ball.vy); ball.y += (ball.y<cy?-oy:oy); }
  onBrickHit(b,ball);
  return;
 }
}

  if(n>=3){
    const k=Math.min(n-2,6), idx=bricks.map((_,i)=>i);
    for(let i=idx.length-1;i>0;i--){const j=(Math.random()*(i+1))|0;[idx[i],idx[j]]=[idx[j],idx[i]];}
    for(let i=0;i<k;i++) bricks[idx[i]].joker=true;
  }
  G.total=bricks.length; G.remaining=G.total;
  G.handSuit=HAND_SUITS[(n-1)%4];
  const base=(410+(n-1)*26)*clamp(worldH/850,.8,1.12);
  G.speed=Math.min(base,730); G.speedCap=G.speed*1.35;
  layout();
}
function resetEffects(){ effects.wide=0;effects.bomb=0;effects.slow=0;effects.shield=false; }
function serve(){
  resetEffects();
  const r=ballR();
  balls=[{x:paddle.x,y:paddle.y-paddle.h/2-r-4,vx:0,vy:0,r,stuck:true,trail:[]}];
}
function launch(){
  let did=false;
  for(const b of balls){ if(!b.stuck)continue;
    const a=rand(-.38,.38); b.vx=Math.sin(a)*G.speed; b.vy=-Math.cos(a)*G.speed; b.stuck=false; did=true; }
  if(did) sSplit();
}

/* ============================== shatter / bombs ============================== */
function onBrickHit(b,ball){
  b.flash=1;
  if(effects.bomb>0){ explode(ball.x,ball.y); removeBall(ball); return; }
  if(b.hp>1){ b.hp--; b.cracked=true; b.crackSeed=Math.random()*10; sCrack(); spawnScrap(b,3); G.shake+=1.2; return; }
  shatter(b,ball);
}
function shatter(b,ball){
  if(b.dead)return;
  b.dead=true; G.remaining--;
  const cx=b.x+b.w/2, cy=b.y+b.h/2;
  let base=VAL[b.rank];
  if(b.suit==='H')base*=1.5; if(b.suit==='C')base*=1.2; if(b.joker)base*=2;
  G.streak++; G.mult=Math.min(8,1+((G.streak/4)|0)); G.streakPop=1;
  const pts=Math.round(base*G.mult/5)*5; G.score+=pts;
  texts.push({x:cx,y:cy-6,txt:'+'+pts,t:0,life:.95,color:b.joker?'#f6dfa0':'#f2ead2',size:clamp(pts>200?23:17,14,24)});
  if(G.mult>1&&G.streak%4===0) texts.push({x:cx,y:cy-28,txt:'STREAK ×'+G.mult,t:0,life:1.1,color:'#e5c15c',size:15});
  ghosts.push({x:b.x,y:b.y,w:b.w,h:b.h,suit:b.suit,rank:b.rank,joker:b.joker,t:0,dir:Math.random()<.5?-1:1});
  spawnScrap(b,8); spawnSparks(cx,cy,b.joker?14:6,b.joker?'#f2d98b':(RED[b.suit]?'#e0574a':'#e5c15c'));
  G.shake+=b.joker?4.5:2.2; sShatter(G.streak);
  if(ball&&b.suit==='D') splitBall(ball,cx,cy);
  const ch=b.joker?1:(b.rank==='A'||b.rank==='K'||b.rank==='Q'||b.rank==='J')?.3:.07;
  if(Math.random()<ch) items.push({x:cx,y:cy,type:pick(['wide','bomb','shield','slow']),t:Math.random()*6});
}
function splitBall(ball,cx,cy){
  if(!ball||balls.length>=6)return;
  const a=Math.atan2(ball.vy,ball.vx); let made=0;
  for(const off of [-.5,.5]){
    if(balls.length+made>=6)break;
    const na=a+off;
    balls.push({x:ball.x,y:ball.y,vx:Math.cos(na)*G.speed,vy:Math.sin(na)*G.speed,r:ball.r,stuck:false,trail:[]});
    made++;
  }
  if(made){ sSplit(); texts.push({x:cx,y:cy-24,txt:'SPLIT!',t:0,life:.9,color:'#f2d98b',size:15}); }
}
function removeBall(ball){
  const i=balls.indexOf(ball); if(i>=0)balls.splice(i,1);
  if(effects.bomb>0&&balls.length===0)G.freeServe=true;
}
function explode(x,y){
  const R=Math.max(L.cardW*2.4,95);
  G.shake+=9; sBomb();
  parts.push({type:'ring',x,y,vx:0,vy:0,t:0,life:.5,size:R});
  for(let i=0;i<26;i++){ const a=rand(0,TAU),sp=rand(60,340);
    parts.push({type:'spark',x,y,vx:Math.cos(a)*sp,vy:Math.sin(a)*sp,t:0,life:rand(.25,.6),size:rand(1.5,3),color:pick(['#ffd76a','#ff9d4d','#e0574a'])}); }
  for(const b of bricks){ if(b.dead)continue;
    const d=Math.hypot(b.x+b.w/2-x,b.y+b.h/2-y);
    if(d<R) pending.push({b,delay:(d/R)*.3+Math.random()*.05}); }
}
function applyEffect(type,x){
  if(type==='wide')effects.wide=12; else if(type==='bomb')effects.bomb=8;
  else if(type==='slow')effects.slow=6; else effects.shield=true;
  const names={wide:'WIDE RAKE',bomb:'JOKER BOMB',shield:'DEUCE GUARD',slow:'SLOW BURN'};
  texts.push({x,y:paddle.y-42,txt:names[type],t:0,life:1.2,color:'#f2d98b',size:16});
  sItem(); spawnSparks(x,paddle.y-10,10,'#f2d98b');
}

/* ============================== particles / texts ============================== */
function spawnScrap(b,n){
  const cx=b.x+b.w/2, cy=b.y+b.h/2, red=RED[b.suit];
  for(let i=0;i<n;i++) parts.push({type:'scrap',x:cx+rand(-b.w/3,b.w/3),y:cy+rand(-b.h/3,b.h/3),
    vx:rand(-140,140),vy:rand(-260,-40),t:0,life:rand(.5,.9),size:rand(4,8)*clamp(b.w/70,.7,1.2),
    rot:rand(0,TAU),vr:rand(-9,9),color:i%3===0?(red?'#d96a5f':'#e5c15c'):'#f2ecd9'});
}
function spawnSparks(x,y,n,color){
  for(let i=0;i<n;i++){ const a=rand(0,TAU),sp=rand(40,240);
    parts.push({type:'spark',x,y,vx:Math.cos(a)*sp,vy:Math.sin(a)*sp,t:0,life:rand(.3,.65),size:rand(1.2,2.6),color}); }
}

/* ============================== update ============================== */
function update(dt){
  const wdt = G.state==='playing' ? dt*G.ts : dt;
  G.shake*=Math.exp(-6.5*dt); if(G.shake<.08)G.shake=0;
  G.hurtT=Math.max(0,G.hurtT-dt*1.4);
  G.streakPop=Math.max(0,G.streakPop-dt*2.4);

  // dust
  for(const d of dust){ d.y-=d.spd*dt; d.x+=Math.sin(G.time*.5+d.ph)*6*dt;
    if(d.y<-6){ d.y=worldH+6; d.x=Math.random()*WORLD_W; } }
  // particles
  for(let i=parts.length-1;i>=0;i--){ const p=parts[i]; p.t+=wdt;
    if(p.t>=p.life){ parts.splice(i,1); continue; }
    if(p.type==='scrap'){ p.vy+=560*wdt; p.rot+=p.vr*wdt; }
    else if(p.type==='spark'){ p.vy+=140*wdt; }
    p.x+=p.vx*wdt; p.y+=p.vy*wdt; }
  if(parts.length>420) parts.splice(0,parts.length-420);
  // ghosts (flipping cards)
  for(let i=ghosts.length-1;i>=0;i--){ const g=ghosts[i]; g.t+=wdt/.32; if(g.t>=1)ghosts.splice(i,1); }
  // floating texts
  for(let i=texts.length-1;i>=0;i--){ const t=texts[i]; t.t+=dt; t.y-=36*dt; if(t.t>=t.life)texts.splice(i,1); }
  for(const b of bricks) if(b.flash>0)b.flash=Math.max(0,b.flash-wdt*5);

  if(G.state!=='playing'){
    if(G.state==='handover'){ G.handOverT+=dt; if(G.handOverT>7)nextHand(); }
    return;
  }

  /* ---- playing ---- */
  const tT=effects.slow>0?.55:1; G.ts+=(tT-G.ts)*Math.min(1,dt*6);

  // paddle
  const dir=(keys.has('ArrowRight')||keys.has('KeyD')?1:0)-(keys.has('ArrowLeft')||keys.has('KeyA')?1:0);
  if(dir) paddle.targetX=clamp(paddle.targetX+dir*980*dt,minPX(),maxPX());
  paddle.x+=(paddle.targetX-paddle.x)*Math.min(1,dt*24);
  const wT=paddleWBase()*(effects.wide>0?1.55:1);
  paddle.w+=(wT-paddle.w)*Math.min(1,dt*8);
  paddle.squish=Math.max(0,paddle.squish-dt*3.2);

  effects.wide=Math.max(0,effects.wide-dt);
  effects.bomb=Math.max(0,effects.bomb-dt);
  effects.slow=Math.max(0,effects.slow-dt);

  // joker cycling
  G.jClock+=dt;
  for(const b of bricks) if(b.joker&&!b.dead){
    const ph=Math.floor((G.jClock+b.jPhase)/1.35);
    if(ph!==b.jTick){ b.jTick=ph; b.suit=pick(SUIT_KEYS); b.rank=pick(RANKS); b.flash=.7; }
  }

  // balls
  for(const ball of balls){
    if(ball.stuck){ ball.x=paddle.x; ball.y=paddle.y-paddle.h/2-ball.r-4; continue; }
    for(let s=0;s<2;s++){
      ball.x+=ball.vx*wdt/2; ball.y+=ball.vy*wdt/2;
      if(ball.x-ball.r<8){ ball.x=8+ball.r; ball.vx=Math.abs(ball.vx); sWall(); }
      else if(ball.x+ball.r>WORLD_W-8){ ball.x=WORLD_W-8-ball.r; ball.vx=-Math.abs(ball.vx); sWall(); }
      if(ball.y-ball.r<6){ ball.y=6+ball.r; ball.vy=Math.abs(ball.vy); sWall(); }
      // paddle
      if(ball.vy>0 && ball.y+ball.r>=paddle.y-paddle.h/2 && ball.y-ball.r<=paddle.y+paddle.h/2 &&
         Math.abs(ball.x-paddle.x)<=paddle.w/2+ball.r*.6){
        const rel=clamp((ball.x-paddle.x)/(paddle.w/2),-1,1);
        G.speed=Math.min(G.speed*1.012,G.speedCap);
        const a=rel*.95;
        ball.vx=Math.sin(a)*G.speed; ball.vy=-Math.cos(a)*G.speed;
        ball.y=paddle.y-paddle.h/2-ball.r-.5;
        paddle.squish=1; sPaddle();
      }
      // bricks (one hit per substep)
      for(const b of bricks){ if(b.dead)continue;
        const cx=b.x+b.w/2, cy=b.y+b.h/2;
        const ox=(b.w/2+ball.r)-Math.abs(ball.x-cx); if(ox<0)continue;
        const oy=(b.h/2+ball.r)-Math.abs(ball.y-cy); if(oy<0)continue;
        if(ox<oy){ ball.vx=ball.x<cx?-Math.abs(ball.vx):Math.abs(ball.vx); ball.x+=(ball.x<cx?-ox:ox); }
        else     { ball.vy=ball.y<cy?-Math.abs(ball.vy):Math.abs(ball.vy); ball.y+=(ball.y<cy?-oy:oy); }
        onBrickHit(b,ball); break;
      }
    }
    // bottom
    if(!ball.dead && ball.y-ball.r>worldH+10){
      if(effects.shield){ effects.shield=false; ball.y=worldH-10-ball.r; ball.vy=-Math.abs(ball.vy);
        sShield(); spawnSparks(ball.x,worldH-14,12,'#f2d98b');
        texts.push({x:ball.x,y:worldH-60,txt:'GUARD!',t:0,life:1,color:'#f2d98b',size:16}); }
      else ball.dead=true;
    }
    // trail + bomb sparks
    if(!ball.stuck){
      ball.trail.push({x:ball.x,y:ball.y}); if(ball.trail.length>12)ball.trail.shift();
      if(effects.bomb>0&&Math.random()<.6) parts.push({type:'spark',x:ball.x,y:ball.y,vx:rand(-40,40),vy:rand(-40,40),t:0,life:.3,size:2,color:pick(['#ffd76a','#ff9d4d'])});
    }
  }
  const before=balls.length;
  balls=balls.filter(b=>!b.dead);
  if(before>0&&balls.length===0){
    if(G.freeServe){ G.freeServe=false; serve(); texts.push({x:WORLD_W/2,y:paddle.y-70,txt:'FREE DEAL — NO CARD LOST',t:0,life:1.3,color:'#f2d98b',size:16}); }
    else{
      G.lives--; G.streak=0; G.mult=1; resetEffects(); sLose(); G.shake+=7; G.hurtT=.7;
      if(G.lives<=0) gameOver();
      else{ serve(); texts.push({x:WORLD_W/2,y:paddle.y-70,txt:'CARD DISCARDED',t:0,life:1.3,color:'#e0574a',size:16}); }
    }
  }

  // chained bomb breaks
  for(let i=pending.length-1;i>=0;i--){ const p=pending[i]; p.delay-=wdt;
    if(p.delay<=0){ pending.splice(i,1); if(!p.b.dead) shatter(p.b,null); } }

  // power-up cards
  for(let i=items.length-1;i>=0;i--){ const it=items[i];
    it.y+=150*wdt; it.t+=wdt;
    if(it.y+20>=paddle.y-paddle.h/2 && it.y-20<=paddle.y+paddle.h/2 && Math.abs(it.x-paddle.x)<=paddle.w/2+16){
      applyEffect(it.type,paddle.x); items.splice(i,1); continue; }
    if(it.y>worldH+30) items.splice(i,1);
  }

  // hand cleared?
  if(G.remaining<=0&&G.state==='playing'){
    G.state='handover'; G.handOverT=0;
    const bonus=500*G.hand; G.score+=bonus; G.lastBonus=bonus;
    sHand(); document.body.classList.remove('playing');
    for(let i=0;i<40;i++) parts.push({type:'spark',x:rand(100,WORLD_W-100),y:rand(worldH*.2,worldH*.45),
      vx:rand(-60,60),vy:rand(-120,-30),t:0,life:rand(.5,1.1),size:rand(1.5,3),color:pick(['#f2d98b','#e5c15c','#fff3cf'])});
    if(G.score>G.best){ G.best=G.score; saveBest(); }
  }
}

/* ============================== drawing ============================== */
function rr(x,y,w,h,r){
  r=Math.min(r,w/2,h/2);
  ctx.beginPath();
  ctx.moveTo(x+r,y); ctx.arcTo(x+w,y,x+w,y+h,r); ctx.arcTo(x+w,y+h,x,y+h,r);
  ctx.arcTo(x,y+h,x,y,r); ctx.arcTo(x,y,x+w,y,r); ctx.closePath();
}
function drawCardFace(x,y,w,h,suit,rank,o){
  o=o||{};
  const red=RED[suit];
  rr(x,y,w,h,Math.min(6,w*.14));
  if(o.joker){ ctx.fillStyle='#fdf3cf'; }
  else{ const g=ctx.createLinearGradient(x,y,x,y+h); g.addColorStop(0,'#faf5e7'); g.addColorStop(1,'#efe6cd'); ctx.fillStyle=g; }
  ctx.fill();
  ctx.lineWidth=o.joker?1.8:1;
  ctx.strokeStyle=o.joker?'#b98d2f':'rgba(40,35,25,.35)';
  ctx.stroke();
  const ink=red?'#c23a2e':'#2a2620';
  if(w>30){
    ctx.fillStyle=ink; ctx.font='700 '+(w*.42)+'px '+F_B;
    ctx.textAlign='left'; ctx.textBaseline='top';
    ctx.fillText(rank,x+w*.1,y+h*.05);
    ctx.save(); ctx.translate(x+w,y+h); ctx.rotate(Math.PI); ctx.fillText(rank,w*.1,h*.05); ctx.restore();
  }
  const gs=h*(w>30?.42:.6);
  ctx.font='700 '+gs+'px '+F_B;
  ctx.textAlign='center'; ctx.textBaseline='middle';
  ctx.fillStyle=ink;
  ctx.fillText(GLYPH[suit],x+w/2,y+h*(w>30?.58:.5)+1);
  if(o.joker){
    ctx.globalAlpha*=.9;
    rr(x+2,y+2,w-4,h-4,Math.min(5,(w-4)*.14));
    ctx.strokeStyle='rgba(185,141,47,'+(0.35+0.25*Math.sin(G.time*4)).toFixed(3)+')';
    ctx.lineWidth=1; ctx.stroke();
  }
  if(o.cracked){
    const seed=o.seed||7;
    ctx.strokeStyle='rgba(40,35,25,.55)'; ctx.lineWidth=1; ctx.beginPath();
    let px=x+w*.15, py=y+h*(.3+rnd(seed,0)*.4);
    ctx.moveTo(px,py);
    for(let i=1;i<=4;i++){ px+=w*.2; py+=(rnd(seed,i)-.5)*h*.3; ctx.lineTo(px,py); }
    ctx.stroke();
  }
}

  ctx.lineWidth=2.2; ctx.lineCap='round'; ctx.lineJoin='round';

    ctx.beginPath(); ctx.moveTo(-9,0); ctx.lineTo(9,0);

  }else if(type==='bomb'){

    ctx.beginPath(); ctx.moveTo(3,-2); ctx.quadraticCurveTo(6,-6,8,-7); ctx.stroke();
    ctx.beginPath(); ctx.arc(9,-8,1.6,0,TAU); ctx.fill();

    ctx.beginPath(); ctx.moveTo(-6,-8); ctx.lineTo(6,-8); ctx.lineTo(0,0); ctx.closePath();
    ctx.moveTo(-6,8); ctx.lineTo(6,8); ctx.lineTo(0,0); ctx.closePath(); ctx.stroke();
  }
  ctx.restore();
}

}

    for(const b of bricks){ if(b.dead)continue;
      drawCardFace(b.x,b.y,b.w,b.h,b.suit,b.rank,{joker:b.joker,cracked:b.cracked,seed:b.crackSeed});
      if(b.flash>0){ ctx.fillStyle='rgba(255,255,240,'+(b.flash*.6).toFixed(3)+')'; rr(b.x,b.y,b.w,b.h,Math.min(6,b.w*.14)); ctx.fill(); }
    }
    // flipping ghosts
    for(const g of ghosts){ const sx=Math.cos(g.t*Math.PI)*g.dir;
      ctx.save(); ctx.translate(g.x+g.w/2,g.y+g.h/2-g.t*26); ctx.rotate(g.t*.35*g.dir);
      ctx.scale(Math.abs(sx)||.001,1); ctx.globalAlpha=1-g.t;
      drawCardFace(-g.w/2,-g.h/2,g.w,g.h,g.suit,g.rank,{joker:g.joker});
      ctx.restore(); }
    // power-up cards
    for(const it of items){ ctx.save(); ctx.translate(it.x,it.y); ctx.rotate(Math.sin(it.t*3)*.18);
      ctx.shadowColor='rgba(229,193,92,.8)'; ctx.shadowBlur=10;
      rr(-15,-21,30,42,4); ctx.fillStyle='#f8f3e2'; ctx.fill(); ctx.shadowBlur=0;
      ctx.strokeStyle='#b98d2f'; ctx.lineWidth=1.5; ctx.stroke();
      drawIcon(it.type,0,0,'#8a6a1e');
      ctx.restore(); }
    // paddle
    (function(){
      const w=paddle.w,h=paddle.h;
      ctx.save(); ctx.translate(paddle.x,paddle.y);
      ctx.scale(1+paddle.squish*.06,1-paddle.squish*.3);
      if(effects.wide>0){ ctx.shadowColor='rgba(246,223,138,.9)'; ctx.shadowBlur=16; }
      const g=ctx.createLinearGradient(0,-h/2,0,h/2);
      g.addColorStop(0,'#f0d78e'); g.addColorStop(1,'#a97e24');
      rr(-w/2,-h/2,w,h,8); ctx.fillStyle=g; ctx.fill(); ctx.shadowBlur=0;
      ctx.strokeStyle='rgba(60,40,8,.6)'; ctx.lineWidth=1.5; ctx.stroke();
      ctx.strokeStyle='rgba(255,255,255,.35)'; ctx.lineWidth=1;
      ctx.beginPath(); ctx.moveTo(-w/2+8,-h/2+3); ctx.lineTo(w/2-8,-h/2+3); ctx.stroke();
      if(w>110){ ctx.font='9px '+F_B; ctx.textAlign='center'; ctx.textBaseline='middle';
        ctx.fillStyle='rgba(40,35,25,.5)'; ctx.fillText('♠',-w/2+12,1);
        ctx.fillStyle='rgba(194,58,46,.55)'; ctx.fillText('♥',w/2-12,1); }
      ctx.restore();
    })();
    // balls + trails
    for(const b of balls){
      const bomb=effects.bomb>0;
      for(let i=0;i<b.trail.length;i++){ const p=b.trail[i], f=(i+1)/b.trail.length;
        ctx.globalAlpha=f*.28; ctx.fillStyle=bomb?'#ff9d4d':'#f6efdc';
        ctx.beginPath(); ctx.arc(p.x,p.y,b.r*(.35+.6*f),0,TAU); ctx.fill(); }
      ctx.globalAlpha=1; drawChip(b);
    }
  } else {
    // attract fan of cards behind the start screen
    const faces=[['S','A'],['H','K'],['D','Q'],['C','J'],['S','10']];
    const fx=WORLD_W*.74, fy=worldH-140;
    for(let i=0;i<5;i++){
      ctx.save(); ctx.translate(fx,fy+Math.sin(G.time*.9+i)*5); ctx.rotate((i-2)*.28+Math.sin(G.time*1.1+i)*.03);
      ctx.shadowColor='rgba(0,0,0,.4)'; ctx.shadowBlur=18; ctx.shadowOffsetY=8;
      drawCardFace(-43,-118,86,120,faces[i][0],faces[i][1]);
      ctx.restore();
    }
  }

  // particles
  for(const p of parts){ const f=1-p.t/p.life;
    if(p.type==='scrap'){ ctx.save(); ctx.translate(p.x,p.y); ctx.rotate(p.rot);
      ctx.globalAlpha=f; ctx.fillStyle=p.color; ctx.fillRect(-p.size/2,-p.size*.35,p.size,p.size*.7); ctx.restore(); }
    else if(p.type==='spark'){ ctx.globalCompositeOperation='lighter'; ctx.globalAlpha=f;
      ctx.fillStyle=p.color; ctx.beginPath(); ctx.arc(p.x,p.y,p.size,0,TAU); ctx.fill();
      ctx.globalCompositeOperation='source-over'; }
    else if(p.type==='ring'){ ctx.globalAlpha=f*.8; ctx.strokeStyle='#f2d98b'; ctx.lineWidth=3;
      ctx.beginPath(); ctx.arc(p.x,p.y,p.size*(p.t/p.life),0,TAU); ctx.stroke(); }
  }
  ctx.globalAlpha=1;

  // shield line
  if(effects.shield){
    const a=.45+.3*Math.sin(G.time*6);
    ctx.strokeStyle='rgba(246,223,138,'+a.toFixed(3)+')'; ctx.lineWidth=3;
    ctx.beginPath(); ctx.moveTo(10,worldH-8); ctx.lineTo(WORLD_W-10,worldH-8); ctx.stroke();
  }
  // slow-burn tint
  if(effects.slow>0){ ctx.fillStyle='rgba(255,176,64,.06)'; ctx.fillRect(0,0,WORLD_W,worldH); }

  // floating texts
  for(const t of texts){ const f=1-t.t/t.life;
    ctx.globalAlpha=Math.min(1,f*1.6); ctx.fillStyle=t.color;
    ctx.font='700 '+t.size+'px '+F_B; ctx.textAlign='center'; ctx.textBaseline='middle';
    ctx.fillText(t.txt,t.x,t.y); }
  ctx.globalAlpha=1;

  /* ---------- HUD ---------- */
  if(G.state!=='start'){
    // your hand (lives)
    const lsuits=['H','D','C'];
    for(let i=0;i<3;i++){
      const x=18+i*44, y=16, alive=i<G.lives;
      ctx.save(); ctx.translate(x+17,y+23); ctx.rotate(alive?0:.18);
      if(alive){ ctx.shadowColor='rgba(0,0,0,.35)'; ctx.shadowBlur=6; drawCardFace(-17,-23,34,46,lsuits[i],'A'); }
      else{ ctx.globalAlpha=.4; rr(-17,-23,34,46,4); ctx.fillStyle='#17453a'; ctx.fill();
        ctx.strokeStyle='rgba(229,193,92,.4)'; ctx.lineWidth=1.5; ctx.stroke();
        ctx.globalAlpha=.5; ctx.fillStyle='#e5c15c'; ctx.font='700 16px '+F_B;
        ctx.textAlign='center'; ctx.textBaseline='middle'; ctx.fillText('✕',0,1); }
      ctx.restore();
    }
    ctx.globalAlpha=.7; ctx.fillStyle='#e5c15c'; ctx.font='9px '+F_B;
    ctx.textAlign='left'; ctx.textBaseline='top'; ctx.fillText('Y O U R   H A N D',18,72);
    ctx.globalAlpha=1;

    // hand label + progress
    const accent=(G.handSuit==='♥'||G.handSuit==='♦')?'#e0574a':'#e5c15c';
    ctx.textAlign='center'; ctx.fillStyle='#e5c15c'; ctx.font='700 20px '+F_D;
    ctx.fillText('HAND '+G.hand,400,26);
    rr(320,52,160,5,2.5); ctx.fillStyle='rgba(0,0,0,.4)'; ctx.fill();
    const pw=clamp(G.remaining/G.total,0,1)*160;
    if(pw>2){ rr(320,52,pw,5,2.5); ctx.fillStyle=accent; ctx.fill(); }
    ctx.fillStyle='rgba(233,242,234,.55)'; ctx.font='10px '+F_B;
    ctx.fillText(G.remaining+' / '+G.total+' CARDS',400,63);

    // score
    ctx.textAlign='right'; ctx.fillStyle='#e5c15c'; ctx.font='700 30px '+F_D;
    ctx.fillText(fmt(G.score),782,30);
    ctx.fillStyle='rgba(233,242,234,.6)'; ctx.font='11px '+F_B;
    ctx.fillText('HOUSE RECORD '+fmt(Math.max(G.best,G.score)),782,58);
    if(G.mult>1){ const sc=1+G.streakPop*.5;
      ctx.save(); ctx.translate(782,84); ctx.scale(sc,sc);
      ctx.fillStyle='#f2d98b'; ctx.font='700 15px '+F_B; ctx.fillText('STREAK ×'+G.mult,0,0); ctx.restore(); }

    // effect chips
    const chips=[];
    if(effects.wide>0)chips.push(['wide',effects.wide]);
    if(effects.bomb>0)chips.push(['bomb',effects.bomb]);
    if(effects.slow>0)chips.push(['slow',effects.slow]);
    if(effects.shield)chips.push(['shield',null]);
    let cx0=400-(chips.length*60-8)/2;
    for(const c of chips){
      const low=c[1]!=null&&c[1]<3, blink=low&&(G.time*4|0)%2===0;
      ctx.globalAlpha=blink?.5:1;
      rr(cx0,84,60,22,4); ctx.fillStyle='rgba(4,26,19,.6)'; ctx.fill();
      ctx.strokeStyle='rgba(229,193,92,.5)'; ctx.lineWidth=1; ctx.stroke();
      drawIcon(c[0],cx0+13,95,'#e5c15c');
      ctx.fillStyle='#f2ead2'; ctx.font='700 11px '+F_B; ctx.textAlign='right'; ctx.textBaseline='middle';
      ctx.fillText(c[1]==null?'ON':Math.ceil(c[1])+'s',cx0+52,96);
      cx0+=68; ctx.globalAlpha=1;
    }

    // serve hint
    if(G.state==='playing'&&balls.some(b=>b.stuck)){
      const a=.45+.3*Math.sin(G.time*4);
      ctx.textAlign='center'; ctx.fillStyle='rgba(246,223,138,'+a.toFixed(3)+')';
      ctx.font='700 13px '+F_B;
      ctx.fillText('SPACE / CLICK — DEAL THE CHIP',paddle.x,paddle.y-52);
    }
  }

  // hand-cleared banner
  if(G.state==='handover'){
    const by=worldH*.34;
    ctx.fillStyle='rgba(3,20,14,.8)'; ctx.fillRect(0,by-72,WORLD_W,150);
    ctx.strokeStyle='rgba(229,193,92,.8)'; ctx.lineWidth=2;
    ctx.beginPath(); ctx.moveTo(0,by-72); ctx.lineTo(WORLD_W,by-72); ctx.moveTo(0,by+78); ctx.lineTo(WORLD_W,by+78); ctx.stroke();
    ctx.textAlign='center'; ctx.fillStyle='#e5c15c'; ctx.font='700 46px '+F_D;
    ctx.fillText('HAND '+G.hand+' CLEARED',WORLD_W/2,by-14);
    ctx.fillStyle='#f2ead2'; ctx.font='700 18px '+F_B;
    ctx.fillText('+'+fmt(G.lastBonus)+' BONUS',WORLD_W/2,by+16);
    const a=.5+.4*Math.sin(G.time*4);
    ctx.fillStyle='rgba(233,242,234,'+a.toFixed(3)+')'; ctx.font='700 12px '+F_B;
    ctx.fillText('SPACE / CLICK — DEAL HAND '+(G.hand+1),WORLD_W/2,by+48);
  }

  // hurt vignette (screen space)
  if(G.hurtT>0){
    const g=ctx.createRadialGradient(cv.width/2,cv.height/2,Math.min(cv.width,cv.height)*.35,cv.width/2,cv.height/2,Math.max(cv.width,cv.height)*.7);
    g.addColorStop(0,'rgba(190,40,30,0)'); g.addColorStop(1,'rgba(190,40,30,'+(G.hurtT*.5).toFixed(3)+')');
    ctx.setTransform(1,0,0,1,0,0); ctx.fillStyle=g; ctx.fillRect(0,0,cv.width,cv.height);
  }
}

window.addEventListener('keydown',e=>{
  ensureAudio();
  const c=e.code;
  if(c==='Space'||c==='ArrowLeft'||c==='ArrowRight'||c==='ArrowUp'||c==='ArrowDown') e.preventDefault();
  keys.add(c);
  if(c==='KeyM'){ toggleMute(); return; }
  if(c==='KeyR'){ startGame(); return; }
  if(c==='Space'||c==='Enter'){
    if(G.state==='start') startGame();
    else if(G.state==='playing'&&balls.some(b=>b.stuck)) launch();
    else if(G.state==='handover') nextHand();
    else if(G.state==='gameover') startGame();
  }
  if(c==='KeyP'||c==='Escape'){
    if(G.state==='playing') setPause(true);
    else if(G.state==='paused') setPause(false);
  }
});
