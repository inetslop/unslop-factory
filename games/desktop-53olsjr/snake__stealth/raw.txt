function computeSight(g){

/* ============================== STEP (SNAKE) ============================== */
function doStep(){
  if(dirQueue.length)dir=dirQueue.shift();
  const h=snake[0], nx=h.x+dir.x, ny=h.y+dir.y;
  if(wallAt(nx,ny))return die('wall');
  const fi=foods.findIndex(f=>f.x===nx&&f.y===ny);
  const eatA=!grid.art.taken&&grid.art.x===nx&&grid.art.y===ny;
  const grow=fi>=0||eatA;
  const check=snake.slice(0,grow?snake.length:snake.length-1);
  if(check.some(s=>s.x===nx&&s.y===ny))return die('tail');
  prevSnake=snake.map(s=>({x:s.x,y:s.y}));
  snake.unshift({x:nx,y:ny});
  if(fi>=0){
    foods.splice(fi,1); pendingFood++; score+=25; stamina=Math.min(100,stamina+8);
    sEat(); burstPx(ox+(nx+.5)*cell,oy+(ny+.5)*cell,'#7ef2a8',14,2.4,false); popScore();
  }
  if(eatA){
    grid.art.taken=true; carrying=true; grid.exit.active=true; score+=100; sPickup();
    burstPx(ox+(nx+.5)*cell,oy+(ny+.5)*cell,'#ffd75e',26,3,false); popScore();
  }
  if(!grow)snake.pop();
}

/* ============================== GUARDS ============================== */
function guardUpdate(g,dt){
  if(g.pause>0){g.pause-=dt;return;}
  const n=g.path.length, bx=g.path[(g.seg+1)%n][0], by=g.path[(g.seg+1)%n][1];
  const dx=bx-g.px, dy=by-g.py, d=Math.hypot(dx,dy), step=g.speed*dt;
  if(d<=step+.0001){ g.px=bx; g.py=by; g.seg=(g.seg+1)%n; g.pause=.4; }
  else{ g.px+=dx/d*step; g.py+=dy/d*step; }
  const ndx=bx-g.px, ndy=by-g.py;
  if(Math.abs(ndx)+Math.abs(ndy)>.01){g.fx=Math.sign(ndx);g.fy=Math.sign(ndy);}
  const cx=Math.round(g.px), cy=Math.round(g.py);
  if(cx!==g.cx||cy!==g.cy)computeSight(g);
}

let parts=[];
function burstPx(x,y,color,n,spd,ring){
  for(let i=0;i<n;i++){
    const a=rand(0,TAU), v=rand(.3,1)*spd*cell*.9;
    parts.push({x,y,vx:Math.cos(a)*v,vy:Math.sin(a)*v,life:rand(.4,.9),max:.9,color,size:rand(1.5,3.5)});
  }
  if(ring)parts.push({x,y,ring:true,r:4,life:.8,max:.8,color});
}
const flies=Array.from({length:26},()=>({fx:Math.random(),fy:Math.random(),a:rand(0,TAU),sp:rand(6,16),ph:rand(0,TAU)}));

/* ============================== UPDATE ============================== */
function update(dt){
  tGlobal+=dt;
  // fireflies always drift
  for(const f of flies){
    f.a+=rand(-2.4,2.4)*dt;
    f.fx+=Math.cos(f.a)*f.sp*dt/(cell*COLS); f.fy+=Math.sin(f.a)*f.sp*dt/(cell*ROWS);
    if(f.fx<0)f.fx=1; if(f.fx>1)f.fx=0; if(f.fy<0)f.fy=1; if(f.fy>1)f.fy=0;
  }
  const worldAlive=state==='menu'||state==='playing'||state==='clear';
  if(worldAlive)for(const g of guards)guardUpdate(g,dt);
  for(let i=parts.length-1;i>=0;i--){
    const p=parts[i]; p.life-=dt;
    if(p.ring)p.r+=dt*cell*7; else{p.x+=p.vx*dt;p.y+=p.vy*dt;p.vx*=.94;p.vy*=.94;}
    if(p.life<=0)parts.splice(i,1);
  }
  if(flashT>0)flashT=Math.max(0,flashT-dt*2.2);
  if(shake>0)shake=Math.max(0,shake-dt*2.4);

  if(state!=='playing')return;

  // snake ticks
  stepTimer-=dt;
  while(stepTimer<=0){ doStep(); stepTimer+=tickSec; if(state!=='playing')break; }
  if(state!=='playing')return;

  // cloak
  const head=snake[0], hk=head.x+','+head.y, onShadow=grid.shadows.has(hk);
  const want=cloakHeld&&!exhausted;
  if(want&&stamina>0){ if(!cloaked){cloaked=true;sCloakOn();} }
  else{
    if(cloaked){ cloaked=false; sCloakOff(); if(stamina<=0)exhausted=true; }
  }
  if(cloaked)stamina-=30*dt; else stamina+=(onShadow?34:16)*dt;
  stamina=clamp(stamina,0,100);
  if(exhausted&&stamina>=30)exhausted=false;

  // detection (any scale in a light)
  seer=null; let exposed=false;
  if(!cloaked){
    outer:
    for(const g of guards){
      if(!g.cells.length)continue;
      for(const s of snake){
        const k=s.x+','+s.y;
        if(g.sight.has(k)&&!grid.shadows.has(k)){exposed=true;seer=g;break outer;}
      }
    }
  }
  if(exposed){
    detect+=dt*(carrying?0.8:0.55);
    const iv=lerp(.45,.12,detect);
    if(tGlobal-lastBeep>iv){lastBeep=tGlobal;sBeep(detect);}
    if(detect>=1){detect=1;return die('seen');}
  }else detect=Math.max(0,detect-dt*.5);

  // food respawn
  if(pendingFood>0){ foodTimer-=dt; if(foodTimer<=0){foodTimer=.8;pendingFood--;spawnFood();} }

  // win check
  if(carrying&&head.x===grid.exit.x&&head.y===grid.exit.y)deliver();

  updateHud(false);
}

    const glow=ctx.createRadialGradient(gx+Math.cos(ang)*r,gy+Math.sin(ang)*r,0,gx+Math.cos(ang)*r,gy+Math.sin(ang)*r,cell*1.1);
    glow.addColorStop(0,isSeer?'rgba(255,90,70,0.3)':'rgba(255,214,140,0.28)'); glow.addColorStop(1,'transparent');
    ctx.fillStyle=glow; ctx.beginPath(); ctx.arc(gx+Math.cos(ang)*r,gy+Math.sin(ang)*r,cell*1.1,0,TAU); ctx.fill();
    ctx.fillStyle='#26313d'; ctx.strokeStyle=isSeer?'#ff5d4d':'#0d141b'; ctx.lineWidth=2;
    ctx.beginPath(); ctx.arc(gx,gy,r,0,TAU); ctx.fill(); ctx.stroke();
    ctx.strokeStyle=isSeer?'#ff8a7a':'#ffd27a'; ctx.lineWidth=4.5;
    ctx.beginPath(); ctx.arc(gx,gy,r*.66,ang-.55,ang+.55); ctx.stroke();
    ctx.fillStyle='#ffe9b0';
    ctx.beginPath(); ctx.arc(gx+Math.cos(ang)*r*.95,gy+Math.sin(ang)*r*.95,r*.24,0,TAU); ctx.fill();
    if(isSeer){
      const b=1+.18*Math.sin(t*14);
      ctx.font='700 '+(cell*.72*b)+'px Impact,"Arial Narrow",sans-serif';
      ctx.textAlign='center'; ctx.fillStyle='#ff6a5e';
      ctx.fillText('!',gx,gy-r-cell*.18);
    }
  }

  // snake
  if(state!=='menu'){
    const rt=clamp(1-stepTimer/tickSec,0,1);
    const pts=[];
    for(let i=0;i<snake.length;i++){
      const p0=prevSnake[Math.min(i,prevSnake.length-1)]||snake[i], p1=snake[i];
      pts.push({x:ox+(lerp(p0.x,p1.x,rt)+.5)*cell, y:oy+(lerp(p0.y,p1.y,rt)+.5)*cell});
    }
    ctx.save();
    if(cloaked)ctx.globalAlpha=.32;
    const r=cell*.34;
    for(let i=snake.length-1;i>=0;i--){ // outline pass
      ctx.fillStyle='#0b2417';
      ctx.beginPath(); ctx.arc(pts[i].x,pts[i].y,r+2.5,0,TAU); ctx.fill();
    }
    for(let i=snake.length-1;i>=0;i--){ // body pass
      const f=1-i/snake.length;
      ctx.fillStyle='hsl(152,'+(58+f*8)+'%,'+(32+f*24)+'%)';
      ctx.beginPath(); ctx.arc(pts[i].x,pts[i].y,r,0,TAU); ctx.fill();
      if(i%2===0){ctx.fillStyle='rgba(255,255,255,0.08)';ctx.beginPath();ctx.arc(pts[i].x,pts[i].y,r*.55,0,TAU);ctx.fill();}
    }
    // head
    const hp=pts[0], hr=r*1.16;
    ctx.fillStyle='#a9f7c6';
    ctx.beginPath(); ctx.arc(hp.x,hp.y,hr,0,TAU); ctx.fill();
    ctx.fillStyle='#0b2417'; ctx.lineWidth=2; ctx.stroke();
    const pdx=dir.x,pdy=dir.y,qx=-pdy,qy=pdx;
    for(const s of[-1,1]){
      const exx=hp.x+pdx*hr*.35+qx*s*hr*.48, eyy=hp.y+pdy*hr*.35+qy*s*hr*.48;
      ctx.fillStyle=cloaked?'#d9fff0':'#ffffff';
      ctx.beginPath(); ctx.arc(exx,eyy,hr*.26,0,TAU); ctx.fill();
      ctx.fillStyle='#123';
      ctx.beginPath(); ctx.arc(exx+pdx*hr*.1,eyy+pdy*hr*.1,hr*.13,0,TAU); ctx.fill();
    }
    if((t%2.6)<.3&&!cloaked){ // tongue flick
      const mx=hp.x+pdx*hr, my=hp.y+pdy*hr, tl=cell*.3;
      ctx.strokeStyle='#ff7d6e'; ctx.lineWidth=2; ctx.lineCap='round';
      ctx.beginPath(); ctx.moveTo(mx,my); ctx.lineTo(mx+pdx*tl,my+pdy*tl);
      ctx.moveTo(mx+pdx*tl,my+pdy*tl); ctx.lineTo(mx+pdx*tl+qx*3,my+pdy*tl+qy*3);
      ctx.moveTo(mx+pdx*tl,my+pdy*tl); ctx.lineTo(mx+pdx*tl-qx*3,my+pdy*tl-qy*3);
      ctx.stroke();
    }
    if(cloaked){ // ghost shimmer
      ctx.globalCompositeOperation='lighter'; ctx.globalAlpha=.4+.15*Math.sin(t*9);
      ctx.strokeStyle='#6ef2d8'; ctx.lineWidth=2; ctx.lineJoin='round';
      ctx.beginPath(); ctx.moveTo(pts[pts.length-1].x,pts[pts.length-1].y);
      for(const p of pts)ctx.lineTo(p.x,p.y);
      ctx.stroke(); ctx.globalCompositeOperation='source-over';
    }
    if(carrying){ // held artifact above head
      const dy2=hp.y-cell*.62+Math.sin(t*5)*2, s=cell*.16;
      const g=ctx.createRadialGradient(hp.x,dy2,0,hp.x,dy2,cell*.7);
      g.addColorStop(0,'rgba(255,215,94,0.35)'); g.addColorStop(1,'transparent');
      ctx.fillStyle=g; ctx.beginPath(); ctx.arc(hp.x,dy2,cell*.7,0,TAU); ctx.fill();
      ctx.save(); ctx.translate(hp.x,dy2); ctx.rotate(Math.PI/4+t*.8);
      ctx.fillStyle='#ffd75e'; ctx.fillRect(-s,-s,s*2,s*2);
      ctx.strokeStyle='#8a5a12'; ctx.lineWidth=1.5; ctx.strokeRect(-s,-s,s*2,s*2);
      ctx.restore();
    }
    ctx.restore();
  }

  // particles
  for(const p of parts){
    const a=clamp(p.life/p.max,0,1);
    if(p.ring){ctx.strokeStyle=p.color;ctx.globalAlpha=a;ctx.lineWidth=2.5;
      ctx.beginPath();ctx.arc(p.x,p.y,p.r,0,TAU);ctx.stroke();ctx.globalAlpha=1;continue;}
    ctx.globalAlpha=a; ctx.fillStyle=p.color;
    ctx.beginPath(); ctx.arc(p.x,p.y,p.size,0,TAU); ctx.fill(); ctx.globalAlpha=1;
  }

  // fireflies
  ctx.globalCompositeOperation='lighter';
  for(const f of flies){
    const x=ox+f.fx*cell*COLS, y=oy+f.fy*cell*ROWS;
    const b=.35+.65*(Math.sin(t*1.3+f.ph)*.5+.5);
    const g=ctx.createRadialGradient(x,y,0,x,y,6);
    g.addColorStop(0,'rgba(216,255,154,'+(0.5*b)+')'); g.addColorStop(1,'transparent');
    ctx.fillStyle=g; ctx.beginPath(); ctx.arc(x,y,6,0,TAU); ctx.fill();
    ctx.fillStyle='rgba(244,255,208,'+b+')'; ctx.fillRect(x-1,y-1,2,2);
  }
  ctx.globalCompositeOperation='source-over';
  ctx.restore(); // end shake

  // vignette
  const vg=ctx.createRadialGradient(W/2,H/2,Math.min(W,H)*.32,W/2,H/2,Math.max(W,H)*.72);
  vg.addColorStop(0,'transparent'); vg.addColorStop(1,'rgba(3,8,6,0.6)');
  ctx.fillStyle=vg; ctx.fillRect(0,0,W,H);

  // detection red pulse
  if(detect>0&&state==='playing'){
    const a=detect*(0.45+0.55*Math.abs(Math.sin(t*8)))*.4;
    const rg=ctx.createRadialGradient(W/2,H/2,Math.min(W,H)*.3,W/2,H/2,Math.max(W,H)*.7);
    rg.addColorStop(0,'transparent'); rg.addColorStop(1,'rgba(255,60,40,'+a+')');
    ctx.fillStyle=rg; ctx.fillRect(0,0,W,H);
  }
  if(flashT>0){ctx.fillStyle='rgba(255,80,60,'+flashT*.35+')';ctx.fillRect(0,0,W,H);}
}
