function findHint(){
  const {w,h,walls,targets,boxes,worker}=G;
  const key=(wx,wy,bs)=> wx+','+wy+'|'+bs.slice().sort((a,b)=>a-b).join(',');
  const startBoxes = boxes.map(b=>b.y*w+b.x);
  const startKey = key(worker.x,worker.y,startBoxes);
  // BFS with parent map
  const q=[{wx:worker.x,wy:worker.y,bx:startBoxes}]; let qi=0;
  const par=new Map(); par.set(startKey,{pk:null,d:-1});
  const targetSet=new Set(targets);
  let found=null;
  while(qi<q.length){
    if(par.size>300000) return null; // unknown
    const s=q[qi++];
    const sk=... key(s.wx,s.wy,s.bx);
    // goal?
    if([...targetSet].every(t=>s.bx.includes(t))){ found=s; break; }  // includes is O(n), fine n≤5
    for(d of 0..3){
      nx=s.wx+DX[d], ny=s.wy+DY[d];
      if(wall(nx,ny)) continue;
      const ni=ny*w+nx; let nb=s.bx;
      const bi=s.bx.indexOf(ni);
      if(bi>=0){
        px=nx+DX[d], py=ny+DY[d];
        if(wall(px,py)) continue;
        const pi=py*w+px;
        if(s.bx.includes(pi)) continue;
        // corner-dead pruning
        if(!targetSet.has(pi) && isCornerDeadCell(px,py)) continue;
        nb=s.bx.slice(); nb[bi]=pi;
      }
      const nk=key(nx,ny,nb);
      if(par.has(nk)) continue;
      par.set(nk,{pk:sk,d}); q.push({wx:nx,wy:ny,bx:nb});
    }
  }
  if(!found) return null;
  // walk back to first move
  let fk=key(found.wx,found.wy,found.bx);
  while(true){
    const e=par.get(fk);
    if(e.pk===startKey) return e.d;
    if(e.pk===null) return null;
    fk=e.pk;
  }
}

    if(floors.length<N*3+8)continue;
    const targets=[];let ok=false;
    for(let t=0;t<40&&!ok;t++){targets.length=0;
      for(let n=0;n<N;n++){let placed=false;
        for(let tr=0;tr<250;tr++){const c=floors[(rng()*floors.length)|0],cx=c%w,cy=(c/w)|0;
          if(targets.some(o=>Math.max(Math.abs(o.x-cx),Math.abs(o.y-cy))<2))continue;targets.push({x:cx,y:cy});placed=true;break;}
        if(!placed)for(let tr=0;tr<500;tr++){const c=floors[(rng()*floors.length)|0],cx=c%w,cy=(c/w)|0;
          if(targets.some(o=>o.x===cx&&o.y===cy))continue;targets.push({x:cx,y:cy});placed=true;break;}
        if(!placed)break;}
      ok=targets.length===N;}
    if(!ok)continue;
    const tSet=new Set(targets.map(t=>t.y*w+t.x));
    let bestC=-1,bestD=-1;
    for(const c of floors){const cx=c%w,cy=(c/w)|0;if(tSet.has(c))continue;let md=1e9;
      for(const t of targets)md=Math.min(md,Math.abs(t.x-cx)+Math.abs(t.y-cy));if(md>bestD){bestD=md;bestC=c;}}
    if(bestC<0)continue;
    const worker={x:bestC%w,y:(bestC/w)|0};
    let boxes=targets.map(t=>({x:t.x,y:t.y}));
    const biAt=(x,y)=>boxes.findIndex(b=>b.x===x&&b.y===y);
    const inB=(x,y)=>x>=0&&y>=0&&x<w&&y<h;
    const K=Math.min(16+L*3,72),sol=[];let fails=0;
    for(let i=0;i<K;i++){
      const opts=[];
      for(let d=0;d<4;d++){
        const px=worker.x+DX[d],py=worker.y+DY[d];
        if(!inB(px,py)||walls[py*w+px])continue;
        const bi=biAt(px,py);if(bi<0)continue;
        const qx=worker.x-DX[d],qy=worker.y-DY[d];
        if(!inB(qx,qy)||walls[qy*w+qx]||biAt(qx,qy)>=0)continue;
        opts.push({d,bi});}
      if(!opts.length){if(++fails>8)break;continue;}
      fails=0;const o=opts[(rng()*opts.length)|0];
      boxes[o.bi]={x:worker.x,y:worker.y};
      worker={x:worker.x-DX[o.d],y:worker.y-DY[o.d]};
      sol.push(o.d);}
    if(sol.length<5)continue;
    if(boxes.every(b=>tSet.has(b.y*w+b.x)))continue;
    if(staticDeadlock(walls,w,h,tSet,boxes)){if(!fallback)fallback={w,h,walls,targets,boxes:boxes.map(b=>({...b})),worker:{...worker},solution:sol.slice(),seed};continue;}
    return {w,h,walls,targets,boxes:boxes.map(b=>({x:b.x,y:b.y})),worker:{...worker},solution:sol,seed};
  }
  return fallback||tinyFallback();
}

/* ================= solver (hints) ================= */
function findHint(){
  try{
    const w=G.w,h=G.h,walls=G.walls,targetSet=G.targetSet;
    const wall=(x,y)=>(x<0||y<0||x>=w||y>=h)?true:walls[y*w+x]===1;
    const cornerDead=(x,y)=>(wall(x,y-1)||wall(x,y+1))&&(wall(x-1,y)||wall(x+1,y));
    const keyOf=(wx,wy,bx)=>{const s=bx.slice().sort((a,b)=>a-b);return wx+','+wy+'|'+s.join(',');};
    const startB=G.boxes.map(b=>b.y*w+b.x);
    if(targetSet.every(t=>startB.indexOf(t)>=0))return null;
    const sKey=keyOf(G.worker.x,G.worker.y,startB);
    const q=[{x:G.worker.x,y:G.worker.y,b:startB}];let qi=0,found=null;
    const par=new Map();par.set(sKey,{p:null,d:-1});
    while(qi<q.length){
      if(par.size>250000)return null;
      const s=q[qi++],sk=keyOf(s.x,s.y,s.b);
      if(targetSet.every(t=>s.b.indexOf(t)>=0)){found=s;break;}
      for(let d=0;d<4;d++){
        const nx=s.x+DX[d],ny=s.y+DY[d];if(wall(nx,ny))continue;
        const ni=ny*w+nx,bi=s.b.indexOf(ni);let nb;
        if(bi>=0){const px=nx+DX[d],py=ny+DY[d];if(wall(px,py))continue;
          const pi=py*w+px;if(s.b.indexOf(pi)>=0)continue;
          if(!targetSet.has(pi)&&cornerDead(px,py))continue;
          nb=s.b.slice();nb[bi]=pi;}
        else nb=s.b;
        const nk=keyOf(nx,ny,nb);if(par.has(nk))continue;
        par.set(nk,{p:sk,d});q.push({x:nx,y:ny,b:nb});}
    }
    if(!found)return -1;
    let fk=keyOf(found.x,found.y,found.b);
    while(true){const e=par.get(fk);if(e.p===sKey)return e.d;if(e.p==null)return -1;fk=e.p;}
  }catch(e){return null;}
}

function tryMove(d){
  if(G.state!=='play'||G.anim.active)return false;
  const w=G.worker,nx=w.x+DX[d],ny=w.y+DY[d];G.facing=d;
  if(isWall(nx,ny)){doBump(d);return true;}
  const bi=boxAt(nx,ny);
  const dur=Math.max(95,140-G.level);
  if(bi>=0){
    const px=nx+DX[d],py=ny+DY[d];
    if(isWall(px,py)||boxAt(px,py)>=0){doBump(d);return true;}
    pushUndo();G.moves++;
    const wf={x:w.x,y:w.y},bf={x:G.boxes[bi].x,y:G.boxes[bi].y};
    G.boxes[bi]={x:px,y:py};w.x=nx;w.y=ny;
    G.anim={active:true,t:0,dur,bi,wf,bf};
    sPush();spawnDust(bf.x,bf.y);
    if(G.targetSet.has(py*G.w+px)){sTarget();spawnSparkle(px,py);spawnRing(px,py);}
    syncHUD();
    if(staticDeadlock(G.walls,G.w,G.h,G.targetSet,G.boxes)&&!G.stuck){G.stuck=true;syncHUD();sDead();toast('CRATE DEADLOCKED','press Z to undo · R to reset the sector',1);}
    if(G.boxes.every(b=>G.targetSet.has(b.y*G.w+b.x)))levelClear();
  }else{
    pushUndo();G.moves++;
    const wf={x:w.x,y:w.y};w.x=nx;w.y=ny;
    G.anim={active:true,t:0,dur,bi:-1,wf,bf:null};
    sMove();if(G.moves%3===0)spawnDust(wf.x,wf.y);
    syncHUD();
  }
  return true;}

  for(let y=1;y<G.h;y++){ctx.moveTo(ox,oy+y*cs);ctx.lineTo(ox+G.w*cs,oy+y*cs);}
  ctx.stroke();
  // walls (raised)
  for(let y=0;y<G.h;y++)for(let x=0;x<G.w;x++){
    if(!G.walls[y*G.w+x])continue;
    const px=ox+x*cs,py=oy+y*cs-lift;
    const fg=ctx.createLinearGradient(0,py+lift,0,py+cs);
    fg.addColorStop(0,'#2c3a61');fg.addColorStop(1,'#1d2745');
    ctx.fillStyle=fg;ctx.fillRect(px,py+lift,cs,cs);
    ctx.fillStyle='#3d4d7c';ctx.fillRect(px,py,cs,lift);
    ctx.fillStyle='rgba(190,215,255,.26)';ctx.fillRect(px,py,cs,Math.max(1,cs*.05));
    ctx.fillStyle='rgba(0,0,0,.4)';ctx.fillRect(px,py+cs-Math.max(1,cs*.06),cs,Math.max(1,cs*.06));
    const m=hash2(x,y);if(m%7<2){ctx.fillStyle='rgba(89,230,207,.14)';
      ctx.beginPath();ctx.arc(px+((m>>3)%100/100)*cs*.8+cs*.1,py+lift+((m>>5)%100/100)*cs*.7,cs*.06,0,7);ctx.fill();}}
  // targets
  for(const t of G.targets){
    const px=ox+t.x*cs+cs/2,py=oy+t.y*cs+cs/2;
    const tw=(Math.sin(now/460+t.x*31+t.y*17)+1)/2;
    const occ=G.boxes.some(b=>b.x===t.x&&b.y===t.y);
    ctx.save();
    ctx.strokeStyle='rgba(89,230,207,'+(occ?.95:.4+.4*tw)+')';
    ctx.lineWidth=Math.max(1.5,cs*.05);ctx.shadowColor='#59e6cf';ctx.shadowBlur=occ?18:8;
    ctx.beginPath();ctx.arc(px,py,cs*(.3+tw*.04),0,7);ctx.stroke();
    if(occ){ctx.setLineDash([cs*.07,cs*.09]);ctx.lineDashOffset=-now/60;ctx.strokeStyle='rgba(124,245,223,.5)';
      ctx.beginPath();ctx.arc(px,py,cs*.42,0,7);ctx.stroke();ctx.setLineDash([]);}
    const ds=cs*(.11+tw*.03);
    ctx.fillStyle='rgba(89,230,207,'+(occ?.95:.45+.35*tw)+')';
    ctx.save();ctx.translate(px,py);ctx.rotate(Math.PI/4+(occ?now/900:0));ctx.fillRect(-ds/2,-ds/2,ds,ds);ctx.restore();
    ctx.restore();}
  // hint arrow
  if(G.hint&&now<G.hint.until){
    const a=.5+.4*Math.sin(now/120),d=G.hint.d;
    const hx=ox+G.worker.x*cs+cs/2+DX[d]*cs*.68,hy=oy+G.worker.y*cs+cs/2+DY[d]*cs*.68;
    ctx.save();ctx.translate(hx,hy);ctx.rotate(Math.atan2(DY[d],DX[d]));
    ctx.fillStyle='rgba(255,207,125,'+a+')';ctx.shadowColor='#ffcf7d';ctx.shadowBlur=10;
    ctx.beginPath();ctx.moveTo(cs*.24,0);ctx.lineTo(-cs*.1,-cs*.16);ctx.lineTo(-cs*.03,0);ctx.lineTo(-cs*.1,cs*.16);ctx.closePath();ctx.fill();
    ctx.restore();}
  // boxes then worker
  for(let i=0;i<G.boxes.length;i++)drawBox(G.boxes[i],i,ap,now);
  drawWorker(ap,now);
  // cell-space particles
  for(const p of parts){const t=p.life/p.max,al=1-t,x=ox+p.x*cs,y=oy+p.y*cs;
    if(p.t==='d'){ctx.fillStyle='rgba(205,195,175,'+(al*.45)+')';ctx.beginPath();ctx.arc(x,y,p.r*cs,0,7);ctx.fill();}
    else if(p.t==='s'){ctx.globalCompositeOperation='lighter';ctx.fillStyle='rgba('+p.c+','+(al*.9)+')';ctx.beginPath();ctx.arc(x,y,p.r*cs,0,7);ctx.fill();ctx.globalCompositeOperation='source-over';}
    else if(p.t==='r'){const r=lerp(p.r0,p.r1,t)*cs;ctx.strokeStyle='rgba(89,230,207,'+(al*.8)+')';ctx.lineWidth=Math.max(1.5,cs*.06);ctx.beginPath();ctx.arc(x,y,r,0,7);ctx.stroke();}}
  // clear flash
  if(G.flash>0){ctx.fillStyle='rgba(140,255,230,'+(.22*G.flash)+')';ctx.fillRect(ox-6,oy-lift-6,G.w*cs+12,G.h*cs+lift+12);}
  ctx.restore();
  // confetti (screen space)
  for(const c of confs){const al=c.life>c.max*.7?1-(c.life/c.max-.7)/.3:1;
    ctx.save();ctx.translate(c.x,c.y);ctx.rotate(c.rot);ctx.globalAlpha=al;ctx.fillStyle=c.c;
    ctx.fillRect(-c.w/2,-c.h/2,c.w,c.h);ctx.restore();}
  ctx.globalAlpha=1;
  // vignette
  const vg=ctx.createRadialGradient(W/2,H/2,Math.min(W,H)*.35,W/2,H/2,Math.max(W,H)*.72);
  vg.addColorStop(0,'rgba(0,0,0,0)');vg.addColorStop(1,'rgba(5,8,16,.5)');
  ctx.fillStyle=vg;ctx.fillRect(0,0,W,H);
}
