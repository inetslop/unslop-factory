function collideWalls(p,b){
  const l=p.s*HALF+WALL, r=(p.s+1)*HALF-WALL;
  if(b.x<l+BALL_R){ b.x=l+BALL_R; b.vx=Math.abs(b.vx); sfx.wall(); }
  if(b.x>r-BALL_R){ b.x=r-BALL_R; b.vx=-Math.abs(b.vx); sfx.wall(); }
  if(b.y<TOP+BALL_R){ b.y=TOP+BALL_R; b.vy=Math.abs(b.vy); sfx.wall(); }
}
function collidePaddle(p,b){
  const py=PADDLE_Y-PADDLE_H/2;
  if(b.vy>0 && b.y+BALL_R>=py && b.y-BALL_R<=PADDLE_Y+PADDLE_H/2 && Math.abs(b.x-p.x)<=p.w/2+BALL_R){
    const rel=Math.max(-1,Math.min(1,(b.x-p.x)/(p.w/2)));
    const ang=rel*1.05, sp=Math.min(MAX_SPEED, Math.hypot(b.vx,b.vy)*1.035+6);
    b.vx=Math.cos(ang)*sp; b.vy=-Math.sin(ang)*sp; b.y=py-BALL_R-0.5;
    burst(b.x,py,p.color,3,140); sfx.paddle(rel);
  }
}
function destroyBrick(p,bk){
  bk.alive=false; p.broken++;
  const pts=bk.pts*(p.doubleT>0?2:1); p.score+=pts;
  popup(bk.x+bk.w/2,bk.y,'+'+pts,p.color,13);
  burst(bk.x+bk.w/2,bk.y+bk.h/2,bk.col,9,220);
  sfx.brick(bk.row);
  if(Math.random()<0.16){
    const r=Math.random(), type=r<0.4?'W':(r<0.7?'M':'D');
    p.drops.push({x:bk.x+bk.w/2,y:bk.y+bk.h/2,type,t:0});
  }
  if(!p.bricks.some(b=>b.alive)){
    p.level++; buildBricks(p);
    popup(p.s*HALF+HALF/2,H*0.42,'LEVEL '+p.level,'#ffffff',30); sfx.level();
    for(let i=0;i<14;i++) burst(p.s*HALF+60+Math.random()*(HALF-120),TOP+50,POWER.W.col,2,200);
  }
}
function collideBricks(p,b){
  for(const bk of p.bricks){ if(!bk.alive)continue;
    const cx=Math.max(bk.x,Math.min(b.x,bk.x+bk.w)), cy=Math.max(bk.y,Math.min(b.y,bk.y+bk.h));
    const dx=b.x-cx, dy=b.y-cy;
    if(dx*dx+dy*dy<=BALL_R*BALL_R){
      if(Math.abs(dx)>Math.abs(dy)) b.vx=dx>0?Math.abs(b.vx):-Math.abs(b.vx);
      else if(Math.abs(dy)>Math.abs(dx)) b.vy=dy>0?Math.abs(b.vy):-Math.abs(b.vy);
      else b.vy=-b.vy;
      destroyBrick(p,bk); return true;
    }
  }
  return false;
}
function applyDrop(p,d){
  if(d.type==='W') p.wideT=12;
  if(d.type==='D') p.doubleT=10;
  if(d.type==='M'){
    const src=p.balls.slice(0,2);
    for(const b of src) for(const off of [-0.55,0.55]){
      if(p.balls.length>=5)break;
      const a=Math.atan2(b.vy,b.vx)+off, sp=Math.hypot(b.vx,b.vy)||startSpeed(p.level);
      let vy=Math.sin(a)*sp; if(b.stuck||vy>0) vy=-Math.abs(vy)-30;
      p.balls.push({x:b.x,y:b.y,vx:Math.cos(a)*sp,vy,stuck:b.stuck,trail:[]});
    }
  }
  popup(p.x,PADDLE_Y-26,POWER[d.type].label,POWER[d.type].col,17); sfx.power();
}
function updatePlayer(p,dt){
  if(p.dead)return;
  let dir=0;
  if(keys['KeyA'])dir--; if(keys['KeyD']&&p.s===0)dir++;
  if(keys['ArrowLeft']&&p.s===1)dir--; if(keys['ArrowRight'])dir+= (p.s===1)?1:0;
  if(p.s===0 && keys['ArrowLeft'])dir-=0; // keep roles strict
  p.x+=dir*520*dt;
  const targetW=p.wideT>0?BASE_W*1.55:BASE_W;
  p.w+=(targetW-p.w)*Math.min(1,dt*12);
  const minX=p.s*HALF+WALL+p.w/2+3, maxX=(p.s+1)*HALF-WALL-p.w/2-3;
  p.x=Math.max(minX,Math.min(maxX,p.x));
  p.wideT=Math.max(0,p.wideT-dt); p.doubleT=Math.max(0,p.doubleT-dt); p.flash=Math.max(0,p.flash-dt*1.6);

  // power-up drops
  for(let i=p.drops.length-1;i>=0;i--){ const d=p.drops[i]; d.y+=132*dt; d.t+=dt;
    if(d.y>H+24){ p.drops.splice(i,1); continue; }
    const cx=Math.max(p.x-p.w/2,Math.min(d.x,p.x+p.w/2)), cy=Math.max(PADDLE_Y-7,Math.min(d.y,PADDLE_Y+7));
    if((d.x-cx)**2+(d.y-cy)**2<=14*14){ applyDrop(p,d); p.drops.splice(i,1); }
  }

  // balls (sub-stepped to avoid tunneling)
  for(const b of p.balls){
    if(b.stuck){ b.x=p.x; b.y=PADDLE_Y-PADDLE_H/2-BALL_R-1; continue; }
    const sp=Math.hypot(b.vx,b.vy), n=Math.max(1,Math.ceil(sp*dt/4)), sdt=dt/n;
    for(let i=0;i<n;i++){
      b.x+=b.vx*sdt; b.y+=b.vy*sdt;
      collideWalls(p,b); collideBricks(p,b); collidePaddle(p,b);
    }
    b.trail.push({x:b.x,y:b.y}); if(b.trail.length>12)b.trail.shift();
    if(b.y-BALL_R>H+30) b.dead=true;
  }
  p.balls=p.balls.filter(b=>!b.dead);
  if(p.balls.length===0 && !p.dead) loseLife(p);
}
function update(dt){ for(const p of players) updatePlayer(p,dt); }

/* ---------- fx / ambient ---------- */
function fxUpdate(dt,freeze){
  tGlobal+=dt; shake.m=Math.max(0,shake.m-dt*30);
  for(const a of amb){ a.y-=a.s*dt; if(a.y<TOP-6){ a.y=H-4; a.x=Math.random()*W; } }
  if(freeze)return;
  for(let i=particles.length-1;i>=0;i--){ const p=particles[i]; p.life-=dt;
    if(p.life<=0){ particles.splice(i,1); continue; }
    p.vy+=p.g*dt; p.x+=p.vx*dt; p.y+=p.vy*dt; }
  for(let i=popups.length-1;i>=0;i--){ const q=popups[i]; q.life-=dt; q.y-=28*dt; if(q.life<=0)popups.splice(i,1); }
}

/* ---------- drawing ---------- */
function rr(x,y,w,h,r){ r=Math.min(r,w/2,h/2); ctx.beginPath();
  ctx.moveTo(x+r,y); ctx.arcTo(x+w,y,x+w,y+h,r); ctx.arcTo(x+w,y+h,x,y+h,r); ctx.arcTo(x,y+h,x,y,r); ctx.arcTo(x,y,x+w,y,r); ctx.closePath(); }

function render(){

  const sx=shake.m?(Math.random()-0.5)*shake.m:0, sy=shake.m?(Math.random()-0.5)*shake.m:0;
  ctx.save(); ctx.translate(sx,sy);

  // arena base + side tints

  for(const a of amb){ ctx.globalAlpha=0.06+0.05*Math.sin(tGlobal+a.ph); ctx.fillStyle='#a8ccff';

  ctx.fillStyle=g; ctx.shadowColor='rgba(160,220,255,.5)'; ctx.shadowBlur=10; ctx.fillRect(HALF-3,TOP,6,H-TOP); ctx.shadowBlur=0;

  for(const p of players) for(const b of p.balls){

      ctx.beginPath(); ctx.arc(t.x,t.y,BALL_R*(i/b.trail.length),0,7); ctx.fill(); }
    ctx.globalAlpha=1; ctx.shadowColor=p.color; ctx.shadowBlur=14;
    ctx.fillStyle='#ffffff'; ctx.beginPath(); ctx.arc(b.x,b.y,BALL_R,0,7); ctx.fill(); ctx.shadowBlur=0;
    ctx.strokeStyle=p.color; ctx.lineWidth=2; ctx.beginPath(); ctx.arc(b.x,b.y,BALL_R+1.5,0,7); ctx.stroke();
  }

  // particles & popups
  for(const p of particles){ ctx.globalAlpha=Math.max(0,p.life/p.max); ctx.fillStyle=p.color;

  ctx.globalAlpha=1;
  for(const q of popups){ ctx.globalAlpha=Math.max(0,q.life/q.max); ctx.fillStyle=q.color;

      ctx.fillText(p.s===0?'W / SPACE':'↑ / ENTER',p.x,PADDLE_Y-32); ctx.globalAlpha=1; }

}
