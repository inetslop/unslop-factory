function seedEmbers(){embers=[];for(let i=0;i<70;i++)embers.push({x:Math.random()*W,y:Math.random()*H,s:.6+Math.random()*1.9,v:12+Math.random()*32,p:Math.random()*6.28,gold:Math.random()<.3});}
function sigil(x,y,r,rot){
 ctx.save();ctx.translate(x,y);ctx.rotate(rot);
 ctx.strokeStyle='rgba(233,180,92,.07)';ctx.lineWidth=1;
 ctx.beginPath();ctx.arc(0,0,r,0,6.284);ctx.stroke();
 ctx.setLineDash([5,11]);ctx.beginPath();ctx.arc(0,0,r*.76,0,6.284);ctx.stroke();ctx.setLineDash([]);
 for(let i=0;i<24;i++){const a=i/24*6.283,l=i%2?r*.05:r*.1;
  ctx.beginPath();ctx.moveTo(Math.cos(a)*r,Math.sin(a)*r);ctx.lineTo(Math.cos(a)*(r+l),Math.sin(a)*(r+l));ctx.stroke();}
 ctx.rotate(-rot*2.4);ctx.strokeStyle='rgba(255,122,61,.06)';
 ctx.beginPath();for(let i=0;i<3;i++){const a=i/3*6.283-1.57;const cx=Math.cos(a)*r*.4,cy=Math.sin(a)*r*.4;i?ctx.lineTo(cx,cy):ctx.moveTo(cx,cy);}ctx.closePath();ctx.stroke();
 ctx.restore();
}
