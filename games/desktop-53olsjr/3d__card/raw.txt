function tryAutoFoundation(loc){

  if(!cd)return;
  for(let f=0;f<4;f++)if(canFound(cd,f)){moveRun(loc,{pile:'found',f});return;}
  sBad(); toast('No home for that yet');
}
function autoFinish(){
  if(busy||won){return;}
  if(!canAuto()){sBad();toast('Not ready to auto-finish yet');return;}
  busy=true; startTimer();
  (function step(){
    let br=99,bs=null,bf=-1;
    S.tab.forEach((col,c)=>{const cd=col[col.length-1]; if(cd&&cd.faceUp)
      for(let f=0;f<4;f++)if(canFound(cd,S.found[f])&&cd.rank<br){br=cd.rank;bs={pile:'tab',c,i:col.length-1};bf=f;}});
    if(bf<0){busy=false;sBad();toast('Stuck — try UNDO');return;}
    snap(); S.moves++; doMove(bs,{pile:'found',f:bf});
    if(!won)setTimeout(step,175);
  })();
}

function onMove(e){
  if(!drag)return;
  const dx=e.clientX-drag.sx, dy=e.clientY-drag.sy;
  if(!drag.moved&&Math.hypot(dx,dy)>6){drag.moved=true;document.body.classList.add('grabbing');sPick();}
  if(!drag.moved)return;
  const now=performance.now(), dt=Math.max(8,now-drag.lt);
  const vx=(e.clientX-drag.lx)/dt*16, vy=(e.clientY-drag.ly)/dt*16;
  drag.rx=lerp(drag.rx,clamp(vy*1.1,-16,16),.35);
  drag.ry=lerp(drag.ry,clamp(-vx*1.1,-16,16),.35);
  const s=scale; let zTop=9000;
  for(const c of drag.cards){ setT(c.el,c.x+dx/s,c.y+dy/s,84,drag.rx,drag.ry); c.el.style.zIndex=zTop++; }
  drag.lx=e.clientX; drag.ly=e.clientY; drag.lt=now;
}
function onUp(e){

  window.removeEventListener('pointercancel',onCancel);

}

function handleClick(loc){

function inRect(x,y,r,pad,extB=0){ return x>r.left-pad&&x<r.right+pad&&y>r.top-pad&&y<r.bottom+extB+pad; }
function hitTest(x,y,single){
  let best=null,bd=1e9;
  const test=(r,obj)=>{ if(inRect(x,y,r,14,obj.extra)) {const d=Math.abs(x-(r.left+r.right)/2); if(d<bd){bd=d;best=obj.target;}} };
  if(single)for(let f=0;f<4;f++)test(slotEls.found[f].getBoundingClientRect(),{target:{pile:'found',f}});

    test(r,{target:{pile:'tab',c},extra:{extra:ext}}); }
  return best;
}

function hint(){
  if(busy||won||drag)return; clearHints(); let best=null;
  const consider=(w,src,tgt)=>{ if(w>(best?best.w:-1e9))best={w,src,tgt}; };
  if(S.waste.length){const cd=S.waste[S.waste.length-1];
    for(let f=0;f<4;f++)if(canFound(cd,S.found[f]))consider(95,{pile:'waste',i:S.waste.length-1},{pile:'found',f});}
  for(let a=0;a<7;a++){const col=S.tab[a]; if(!col.length)continue; const cd=col[col.length-1];
    if(cd.faceUp)for(let f=0;f<4;f++)if(canFound(cd,S.found[f]))consider(85,{pile:'tab',c:a,i:col.length-1},{pile:'found',f});}
  for(let a=0;a<7;a++){const col=S.tab[a];
    for(let i=0;i<col.length;i++){const cd=col[i]; if(!cd.faceUp)continue;
      for(let b=0;b<7;b++){ if(b===a)continue; const tc=S.tab[b]; let w=-1e9;
        if(!tc.length){ if(cd.rank===13&&i>0)w=25; }
        else{const t=tc[tc.length-1];
          if(t.faceUp&&colDiff(t,cd)&&t.rank===cd.rank+1){ w=30+(cd.rank<13?12:0)+(i>0?8:0); if(cd.rank===13&&i===0)w=-1e9; }}
        if(w>-1e8)consider(w,{pile:'tab',c:a,i},{pile:'tab',c:b}); } } }
  if(!best&&S.stock.length)best={stock:true};
  if(!best){sBad();toast('No moves — try UNDO or a NEW DEAL');return;}
  if(best.stock)slotEls.stock.classList.add('hint');

}
