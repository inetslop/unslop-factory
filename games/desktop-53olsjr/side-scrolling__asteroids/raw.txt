    parts.push({t:'puff',x:x+rnd(-10,10)*U,y:y+rnd(-10,10)*U,vx:rnd(-30,30)*U,vy:rnd(-30,30)*U,
      r:rnd(6,14)*U,grow:26*U,life:rnd(.4,.9),dur:.9});
  for(let i=0;i<size+2;i++){
    const a=rnd(TAU), v=rnd(50,150)*U;
    parts.push({t:'chunk',x,y,vx:Math.cos(a)*v,vy:Math.sin(a)*v,rot:rnd(TAU),vr:rnd(-7,7),
      s:rnd(2.5,5.5)*U,life:rnd(.5,1.1),dur:1.1});
  }
  parts.push({t:'ring',x,y,r0:6*U,r1:(30+size*26)*U,life:.4,dur:.4,c:col,lw:3});
  shake=Math.min(shake+size*3.5, 26);
}
function popup(x,y,txt,col){ pops.push({x,y,txt,col,life:.9,dur:.9}); }

function startRun(){
  bullets.length=0; rocks.length=0; parts.length=0; pops.length=0; pows.length=0;
  score=0; combo=0; mult=1; comboT=0; bestCombo=0; kills=0; escaped=0; newRec=false;
  sector=1; toSpawn=sectorCount(1); spawnT=1.4; clearing=false; clearT=0;
  ship.x=W*.18; ship.y=H/2; ship.vx=ship.vy=0; ship.tilt=0; ship.hull=3;
  ship.inv=1.5; ship.cool=0; ship.shieldT=ship.tripleT=ship.rapidT=0; ship.dead=false;
  shake=0; flashT=0; ts=1; dying=false; paused=false; state='play';
  banner={t1:'SECTOR 01', t2:'HOSTILES INBOUND', life:2, dur:2};
  showScr(null);
  SFX.start();
}

  const k=(e.key||'').toLowerCase();

  audio();

      if((k==='p'||k==='escape') && !dying){ paused=!paused; showScr(paused?'pause':null); }

});

/* ============ update ============ */
function killRock(i){
  const r=rocks[i]; rocks.splice(i,1); kills++;
  combo++; comboT=2.4; bestCombo=Math.max(bestCombo,combo);
  mult=Math.min(8, 1+Math.floor(combo/5));
  const gain=[0,20,50,100][r.size]*mult;
  score+=gain;
  popup(r.x,r.y,'+'+gain+(mult>1?' ×'+mult:''), mult>1?'#ffd27a':'#eef4ff');
  explode(r.x,r.y,r.size, r.size===3?'rgba(150,190,255,':'rgba(255,200,130,');
  SFX.boom(r.size);
  if(r.size>1){
    const a=rnd(TAU), dx=Math.cos(a), dy=Math.sin(a);
    for(let j=0;j<2;j++){
      rocks.push(makeRock(r.size-1, r.x+dx*r.r*.5, r.y+dy*r.r*.5,
        r.vx*.7+dx*rnd(60,160)*U+(j?40*U:-40*U), r.vy*.6+dy*rnd(60,160)*U));
    }
  }
  const drop=[0,.05,.14,.28][r.size];
  if(Math.random()<drop){
    const roll=Math.random();
    pows.push({x:r.x,y:r.y,vx:-52*U,type: roll<.4?'shield':roll<.75?'triple':'rapid',t:rnd(TAU)});
  }
}
function hitShip(){
  if(ship.inv>0||dying) return;
  ship.hull--; combo=0; mult=1; comboT=0; flashT=.6; shake+=16; SFX.hurt();
  explode(ship.x,ship.y,2,'rgba(255,140,90,');
  if(ship.hull<=0){
    ship.dead=true; dying=true; deathT=1.5; ts=.3;
    explode(ship.x,ship.y,3,'rgba(255,170,90,');
    SFX.boom(3); SFX.over();
  } else ship.inv=2.2;
}
function gameOver(){
  dying=false; state='over';
  if(score>hi){ hi=score|0; newRec=true; saveHi(); }
  $('stScore').textContent=pad7(score);
  $('stSector').textContent=pad2(sector);
  $('stKills').textContent=kills;
  $('stCombo').textContent=bestCombo+' KILLS';
  $('hiOver').textContent=pad7(hi);
  $('recBadge').classList.toggle('hidden',!newRec);
  showScr('over');
}

function update(dt){
  rt+=dt;
  const tsTarget = dying? .28 : 1;
  ts += (tsTarget-ts)*Math.min(1,dt*6);
  const dtG = dt*ts;

  // stars always drift
  const sMul = state==='play' ? 1+(sector-1)*.05 : 1;
  for(const s of stars){ s.x -= s.sp*dt*sMul*(paused?0:1); if(s.x<-3){s.x=W+3;s.y=rnd(H);} }

  if(paused||state==='over') { /* world frozen, over keeps ambient below */ }
  const run = state==='play' && !paused;
  if(run || dying) gt+=dtG;

  // ---- playing / dying simulation ----
  if(state==='play' && !paused){
    if(!dying){
      // input
      let ax=0,ay=0;
      if(keys.has('a')||keys.has('arrowleft')) ax-=1;
      if(keys.has('d')||keys.has('arrowright')) ax+=1;
      if(keys.has('w')||keys.has('arrowup')) ay-=1;
      if(keys.has('s')||keys.has('arrowdown')) ay+=1;
      const len=Math.hypot(ax,ay)||1, A=1500*U;
      ship.vx+=ax/len*A*dtG; ship.vy+=ay/len*A*dtG;
      const damp=Math.exp(-dtG*3.1); ship.vx*=damp; ship.vy*=damp;
      const sp=Math.hypot(ship.vx,ship.vy), mx=470*U;
      if(sp>mx){ ship.vx*=mx/sp; ship.vy*=mx/sp; }
      ship.x=clamp(ship.x+ship.vx*dtG, 28, W-30);
      ship.y=clamp(ship.y+ship.vy*dtG, 30, H-34);
      ship.tilt=lerp(ship.tilt, clamp(ship.vy/520,-.6,.6), Math.min(1,dt*8));

      // timers
      ship.cool-=dtG; ship.inv=Math.max(0,ship.inv-dtG);
      ship.shieldT=Math.max(0,ship.shieldT-dtG);
      ship.tripleT=Math.max(0,ship.tripleT-dtG);
      ship.rapidT=Math.max(0,ship.rapidT-dtG);

      // fire
      if(keys.has(' ') && ship.cool<=0){
        ship.cool = ship.rapidT>0 ? .1 : .19;
        const angs = ship.tripleT>0 ? [-.21,0,.21] : [0];
        for(const a of angs){
          bullets.push({x:ship.x+16*U*Math.cos(a), y:ship.y+16*U*Math.sin(a),
            vx:Math.cos(a)*950*U+ship.vx*.25, vy:Math.sin(a)*950*U+ship.vy*.25, life:1.3});
        }
        for(let i=0;i<3;i++){
          const a=rnd(-.4,.4);
          parts.push({t:'spark',x:ship.x+16*U,y:ship.y,vx:Math.cos(a)*rnd(120,260)*U,vy:Math.sin(a)*rnd(120,260)*U,
            life:.15,dur:.15,c:'rgba(255,210,140,'});
        }
        ship.vx-=26*U*(angs.length>1?.6:1);
        shake+=1.1; SFX.shoot();
      }

      // sector spawning / clearing
      if(!clearing){
        if(toSpawn>0){ spawnT-=dtG; if(spawnT<=0){ toSpawn--; spawnRock();
          spawnT=clamp(1.7-sector*.09,.45,2)*rnd(.6,1.3); } }
        else if(rocks.length===0){ clearing=true; clearT=1.7;
          const bonus=200+sector*100; score+=bonus;
          banner={t1:'SECTOR '+pad2(sector)+' CLEARED', t2:'BONUS +'+bonus, life:1.6, dur:1.6};
          SFX.clear(); shake+=4;
        }
      } else {
        clearT-=dtG;
        if(clearT<=0){ clearing=false; sector++; toSpawn=sectorCount(sector); spawnT=rnd(.8,1.5);
          banner={t1:'SECTOR '+pad2(sector), t2:'HOSTILES INBOUND', life:1.8, dur:1.8}; }
      }

      // combo decay
      if(comboT>0){ comboT-=dtG; if(comboT<=0){ combo=0; mult=1; } }
    } else { deathT-=dtG; if(deathT<=0) gameOver(); }

    // bullets
    for(let i=bullets.length-1;i>=0;i--){
      const b=bullets[i]; b.x+=b.vx*dtG; b.y+=b.vy*dtG; b.life-=dtG;
      if(b.x>W+40||b.y<-40||b.y>H+40||b.life<=0) bullets.splice(i,1);
    }
    // rocks
    for(let i=rocks.length-1;i>=0;i--){
      const r=rocks[i];
      r.x+=r.vx*dtG; r.y+=r.vy*dtG; r.rot+=r.vr*dtG;
      if(!dying) r.vy += (H/2-r.y)*.05*dtG;   // gentle lane pull
      if(r.x < -r.r*1.6-40){ rocks.splice(i,1); escaped++; continue; }

      // bullet hits
      let hit=false;
      for(let j=bullets.length-1;j>=0;j--){
        const b=bullets[j];
        if((b.x-r.x)**2+(b.y-r.y)**2 < (r.r*.9)**2){ bullets.splice(j,1); hit=true; break; }
      }
      if(hit){ killRock(i); continue; }

      // ship collision
      if(!dying && !ship.dead){
        const rr=r.r*.78+12*U;
        if((r.x-ship.x)**2+(r.y-ship.y)**2 < rr*rr){
          if(ship.shieldT>0){ explode(r.x,r.y,r.size,'rgba(90,230,255,'); SFX.shieldHit();
            if(r.size>1){ const a=rnd(TAU);
              for(let j=0;j<2;j++) rocks.push(makeRock(r.size-1,r.x,r.y,
                r.vx*.7+Math.cos(a)*rnd(60,150)*U, r.vy*.6+Math.sin(a)*rnd(60,150)*U)); }
            rocks.splice(i,1); kills++; combo++; comboT=2.4; bestCombo=Math.max(bestCombo,combo);
            mult=Math.min(8,1+Math.floor(combo/5)); score+=[0,20,50,100][r.size]*mult;
          } else hitShip();
        }
      }
    }
    // powerups
    for(let i=pows.length-1;i>=0;i--){
      const p=pows[i]; p.t+=dtG; p.x+=p.vx*dtG; p.y+=Math.sin(p.t*2.4)*18*U*dtG;
      if(p.x<-50){ pows.splice(i,1); continue; }
      if(!dying && !ship.dead && (p.x-ship.x)**2+(p.y-ship.y)**2 < (30*U)**2){
        ship[p.type+'T'] = p.type==='shield'?5:8;
        popup(ship.x,ship.y-30*U,{shield:'SHIELD ONLINE',triple:'TRIPLE SHOT',rapid:'RAPID CYCLE'}[p.type],
          {shield:'#57e6ff',triple:'#ffc857',rapid:'#8dff9a'}[p.type]);
        parts.push({t:'ring',x:p.x,y:p.y,r0:4*U,r1:44*U,life:.35,dur:.35,c:'rgba(255,255,255,',lw:2});
        SFX.power(); pows.splice(i,1);
      }
    }
  }

  // ---- ambient (menu / over backdrop) ----
  if(state==='menu'||state==='over'){
    if(rocks.length<4 && Math.random()<dt*.8){
      const s=irnd2(1,3);
      rocks.push(makeRock(s, W+rnd(60,240)+58*U, rnd(H*.08,H*.92), -rnd(30,70)*U, rnd(-25,25)*U));
    }
    for(let i=rocks.length-1;i>=0;i--){ const r=rocks[i];
      r.x+=r.vx*dt; r.rot+=r.vr*dt; if(r.x<-r.r*2) rocks.splice(i,1); }
    if(Math.random()<dt*.3){ // distant flare
      const x=rnd(W), y=rnd(H);
      parts.push({t:'ring',x,y,r0:2,r1:rnd(30,70),life:.6,dur:.6,c:'rgba(150,190,255,',lw:1.5});
    }
  }

  // particles / popups (tick in play, dying and ambience)
  const tick = run||dying||(state==='menu'||state==='over');
  if(tick){
    const pdt = (run||dying)? dtG : dt;
    for(let i=parts.length-1;i>=0;i--){
      const p=parts[i]; p.life-=pdt;
      if(p.life<=0){ parts.splice(i,1); continue; }
      if(p.t==='spark'){ p.x+=p.vx*pdt; p.y+=p.vy*pdt; p.vx*=Math.exp(-pdt*2.4); p.vy*=Math.exp(-pdt*2.4); }
      else if(p.t==='puff'){ p.x+=p.vx*pdt; p.y+=p.vy*pdt; p.r+=p.grow*pdt; }
      else if(p.t==='chunk'){ p.x+=p.vx*pdt; p.y+=p.vy*pdt; p.rot+=p.vr*pdt; p.vy+=120*U*pdt; }
    }
    if(parts.length>420) parts.splice(0,parts.length-420);
    for(let i=pops.length-1;i>=0;i--){ const p=pops[i]; p.life-=pdt; p.y-=32*pdt; if(p.life<=0) pops.splice(i,1); }
  }

  shake*=Math.exp(-dt*5.5); flashT=Math.max(0,flashT-dt);
  if(banner){ banner.life-= (run||dying)? dtG : 0; if(banner.life<=0) banner=null; }
}

/* ============ draw ============ */
function spaced(txt,x,y,ls){ let cx=x; for(const ch of txt){ ctx.fillText(ch,cx,y); cx+=ctx.measureText(ch).width+ls; } return cx; }

function drawRock(r,alpha=1){
  ctx.save(); ctx.globalAlpha=alpha; ctx.translate(r.x,r.y); ctx.rotate(r.rot);
  ctx.beginPath();
  const n=r.pts.length;
  for(let i=0;i<n;i++){ const a=i/n*TAU, rad=r.r*r.pts[i];
    i?ctx.lineTo(Math.cos(a)*rad,Math.sin(a)*rad):ctx.moveTo(Math.cos(a)*rad,Math.sin(a)*rad); }
  ctx.closePath();
  ctx.fillStyle=r.fill; ctx.fill();
  ctx.lineWidth=4.5; ctx.globalAlpha=alpha*.18; ctx.strokeStyle=r.stroke; ctx.stroke();
  ctx.globalAlpha=alpha; ctx.lineWidth=1.8; ctx.stroke();
  ctx.globalAlpha=alpha*.35; ctx.lineWidth=1; ctx.beginPath();
  for(const c of r.cr){ ctx.moveTo(c[0]*r.r,c[1]*r.r); ctx.lineTo(c[2]*r.r,c[3]*r.r); }
  ctx.stroke(); ctx.restore();
}

function drawShip(){
  if(ship.dead) return;
  const blink = ship.inv>0 && Math.floor(gt*18)%2===0;
  ctx.save(); ctx.translate(ship.x,ship.y);
  // shield ring
  if(ship.shieldT>0){
    const fade=Math.min(1,ship.shieldT/.5);
    ctx.save(); ctx.rotate(gt*.9);
    ctx.setLineDash([12*U,8*U]); ctx.lineDashOffset=-gt*46;
    ctx.strokeStyle=`rgba(90,230,255,${.85*fade})`; ctx.lineWidth=2;
    ctx.beginPath(); ctx.arc(0,0,27*U,0,TAU); ctx.stroke();
    ctx.setLineDash([]); ctx.restore();
  }
  if(blink) ctx.globalAlpha=.3;
  // engine flame
  const moving=Math.hypot(ship.vx,ship.vy)>40;
  if(moving){
    const fl=(12+rnd()*14)*U*(.6+Math.min(1,Math.hypot(ship.vx,ship.vy)/(470*U)));
    ctx.save(); ctx.rotate(ship.tilt*.55); ctx.globalCompositeOperation='lighter';
    const g=ctx.createLinearGradient(-14*U,0,-14*U-fl,0);
    g.addColorStop(0,'rgba(255,190,90,.95)'); g.addColorStop(.5,'rgba(255,120,50,.5)'); g.addColorStop(1,'rgba(255,80,30,0)');
    ctx.fillStyle=g; ctx.beginPath();
    ctx.moveTo(-13*U,-4.5*U); ctx.lineTo((-16-fl)*U,rnd(-2,2)); ctx.lineTo(-13*U,4.5*U);
    ctx.closePath(); ctx.fill(); ctx.restore();
  }
  ctx.rotate(ship.tilt*.55);
  const P=[[18,0],[2,-6],[-13,-15],[-8,-4],[-15,0],[-8,4],[-13,15],[2,6]];
  ctx.beginPath();
  P.forEach((p,i)=> i?ctx.lineTo(p[0]*U,p[1]*U):ctx.moveTo(p[0]*U,p[1]*U));
  ctx.closePath();
  ctx.fillStyle='rgba(220,232,255,.14)'; ctx.fill();
  ctx.lineWidth=2; ctx.strokeStyle='#e8f0ff'; ctx.stroke();
  // cockpit accent
  ctx.beginPath(); ctx.moveTo(11*U,0); ctx.lineTo(3*U,-3.5*U); ctx.lineTo(3*U,3.5*U); ctx.closePath();
  ctx.fillStyle='#ffb347'; ctx.fill();
  ctx.restore();
}

function drawParts(){
  ctx.save(); ctx.globalCompositeOperation='lighter';
  for(const p of parts){
    const k=p.life/p.dur;
    if(p.t==='spark'){
      ctx.strokeStyle=p.c+(k*.9)+')'; ctx.lineWidth=1.6;
      ctx.beginPath(); ctx.moveTo(p.x,p.y); ctx.lineTo(p.x-p.vx*.03,p.y-p.vy*.03); ctx.stroke();
    } else if(p.t==='puff'){
      ctx.fillStyle=`rgba(130,150,200,${k*.28})`;
      ctx.beginPath(); ctx.arc(p.x,p.y,p.r,0,TAU); ctx.fill();
    } else if(p.t==='chunk'){
      ctx.save(); ctx.translate(p.x,p.y); ctx.rotate(p.rot);
      ctx.strokeStyle=`rgba(170,190,235,${k*.8})`; ctx.lineWidth=1.4;
      ctx.beginPath(); ctx.moveTo(p.s,0); ctx.lineTo(-p.s*.6,p.s*.7); ctx.lineTo(-p.s*.6,-p.s*.7); ctx.closePath();
      ctx.stroke(); ctx.restore();
    } else if(p.t==='ring'){
      const rr=lerp(p.r0,p.r1,1-k*k), a=k;
      ctx.strokeStyle=p.c+(a*.8)+')'; ctx.lineWidth=p.lw;
      ctx.beginPath(); ctx.arc(p.x,p.y,rr,0,TAU); ctx.stroke();
    }
  }
  // bullets
  for(const b of bullets){
    const bx=b.x-b.vx*.016, by=b.y-b.vy*.016;
    ctx.strokeStyle='rgba(255,190,110,.55)'; ctx.lineWidth=4.5;
    ctx.beginPath(); ctx.moveTo(bx,by); ctx.lineTo(b.x,b.y); ctx.stroke();
    ctx.strokeStyle='#fff3dd'; ctx.lineWidth=1.8;
    ctx.beginPath(); ctx.moveTo(bx,by); ctx.lineTo(b.x,b.y); ctx.stroke();
  }
  ctx.restore();
}

function drawPows(){
  for(const p of pows){
    const col={shield:'#57e6ff',triple:'#ffc857',rapid:'#8dff9a'}[p.type];
    ctx.save(); ctx.translate(p.x,p.y);
    ctx.strokeStyle=col; ctx.globalAlpha=.25+.15*Math.sin(p.t*5); ctx.lineWidth=5;
    ctx.beginPath(); ctx.arc(0,0,(17+Math.sin(p.t*5)*2)*U,0,TAU); ctx.stroke();
    ctx.globalAlpha=1; ctx.rotate(p.t*.8);
    ctx.beginPath();
    for(let i=0;i<6;i++){ const a=i/6*TAU, r=14*U;
      i?ctx.lineTo(Math.cos(a)*r,Math.sin(a)*r):ctx.moveTo(Math.cos(a)*r,Math.sin(a)*r); }
    ctx.closePath(); ctx.fillStyle='rgba(8,12,26,.9)'; ctx.fill();
    ctx.lineWidth=2; ctx.stroke();
    ctx.rotate(-p.t*.8);
    ctx.fillStyle=col; ctx.font=`700 ${13*U}px ${MONO}`; ctx.textAlign='center'; ctx.textBaseline='middle';
    ctx.fillText({shield:'S',triple:'T',rapid:'R'}[p.type],0,1);
    ctx.restore();
  }
}

  if(ship.tripleT>0) chip('TRIPLE','#ffc857',ship.tripleT/8);
  if(ship.rapidT>0) chip('RAPID','#8dff9a',ship.rapidT/8);
  // hint / mute
  ctx.textAlign='right'; ctx.fillStyle='rgba(160,180,220,.4)'; ctx.font=`600 9px ${MONO}`;
  ctx.fillText((muted?'MUTED · ':'')+'P PAUSE · R RESTART · M SOUND', W-M, H-18);
}

function drawBanner(){
  if(!banner) return;
  const k=banner.life/banner.dur;
  let a=1;
  if(k>.82) a=(1-k)/.18; else if(k<.3) a=k/.3;
  ctx.save(); ctx.globalAlpha=clamp(a,0,1);
  const y=H*.34 - (1-clamp(a,0,1))*14;
  ctx.textAlign='center';
  ctx.fillStyle='#fff'; ctx.font=`900 ${Math.min(56,W*.06)|0}px ${DISP}`;
  ctx.fillText(banner.t1, W/2, y);
  const tw=ctx.measureText(banner.t1).width;
  ctx.strokeStyle='rgba(140,200,255,.6)'; ctx.lineWidth=1.5;
  ctx.beginPath(); ctx.moveTo(W/2-tw/2-70,y-8); ctx.lineTo(W/2-tw/2-24,y-8);
  ctx.moveTo(W/2+tw/2+24,y-8); ctx.lineTo(W/2+tw/2+70,y-8); ctx.stroke();
  ctx.fillStyle='#8fd8ff'; ctx.font=`600 13px ${MONO}`;
  spaced(banner.t2, W/2-ctx.measureText(banner.t2).width/2, y+24, 5);
  ctx.restore();
}

function draw(){
  // base sky
  const g=ctx.createLinearGradient(0,0,0,H);
  g.addColorStop(0,'#050917'); g.addColorStop(.6,'#0a1230'); g.addColorStop(1,'#0d1830');
  ctx.fillStyle=g; ctx.fillRect(0,0,W,H);

  // shaken world
  ctx.save();
  if(shake>.3) ctx.translate(rnd(-shake,shake), rnd(-shake,shake));

  // nebula (slow parallax, wrapped)
  const off=(rt*9)%W;
  ctx.globalAlpha=.95;
  ctx.drawImage(neb,-off,0,W,H); ctx.drawImage(neb,W-off,0,W,H);
  ctx.globalAlpha=1;

  // planet
  if(planet){
    const ps=Math.max(240,H*.55);
    let px=(rt*4)%(W+ps*2.4); px=W-ps-px;
    ctx.globalAlpha=.75; ctx.drawImage(planet, px, H*.06, ps, ps); ctx.globalAlpha=1;
  }

  // stars
  for(const s of stars){
    const a=(.35+.45*(.5+.5*Math.sin(rt*2+s.ph)));
    ctx.fillStyle=s.c+a.toFixed(2)+')';
    ctx.fillRect(s.x,s.y,s.r,s.r);
  }

  // gameplay layer
  if(state!=='menu'){ drawPows(); }
  for(const r of rocks) drawRock(r, state==='menu'?.8:1);
  drawParts();
  if(state==='play' && !ship.dead) drawShip();

  // popups
  ctx.textAlign='center';
  for(const p of pops){
    ctx.globalAlpha=clamp(p.life/p.dur*1.4,0,1);
    ctx.fillStyle=p.col; ctx.font=`700 ${13*U+2}px ${MONO}`;
    ctx.fillText(p.txt,p.x,p.y);
  }
  ctx.globalAlpha=1;

  drawBanner();
  ctx.restore(); // end shake

  if(state==='play') drawHUD();

  // vignette + damage flash
  ctx.fillStyle=vig; ctx.fillRect(0,0,W,H);
  if(flashT>0){ ctx.fillStyle=`rgba(255,64,40,${flashT*.3})`; ctx.fillRect(0,0,W,H); }
}

/* ============ main loop ============ */
let last=performance.now();
function loop(now){
  const dt=Math.min(.05,(now-last)/1000); last=now;
  update(dt); draw();
  requestAnimationFrame(loop);
}
resize();
$('hiMenu').textContent=pad7(hi);
requestAnimationFrame(loop);
