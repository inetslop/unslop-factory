const U=12, MW=29, MH=17;
function makeMaze(W,H){
  const vW=[],hW=[];
  for(let x=0;x<W-1;x++)vW[x]=new Array(H).fill(true);
  for(let y=0;y<H-1;y++)hW[y]=new Array(W).fill(true);
  const vis=new Uint8Array(W*H),st=[[0,0]];vis[0]=1;
  while(st.length){
    const c=st[st.length-1],cx=c[0],cy=c[1],opts=[];
    if(cx>0&&!vis[cy*W+cx-1])opts.push(0);
    if(cx<W-1&&!vis[cy*W+cx+1])opts.push(1);
    if(cy>0&&!vis[(cy-1)*W+cx])opts.push(2);
    if(cy<H-1&&!vis[(cy+1)*W+cx])opts.push(3);
    if(!opts.length){st.pop();continue}
    const d=opts[Math.floor(Math.random()*opts.length)];let nx=cx,ny=cy;
    if(d===0){nx--;vW[cx-1][cy]=false}
    else if(d===1){nx++;vW[cx][cy]=false}
    else if(d===2){ny--;hW[cy-1][cx]=false}
    else{ny++;hW[cy][cx]=false}
    vis[ny*W+nx]=1;st.push([nx,ny]);
  }
  for(let x=0;x<W-1;x++)for(let y=0;y<H;y++)if(vW[x][y]&&Math.random()<.09)vW[x][y]=false;
  for(let x=0;x<W;x++)for(let y=0;y<H-1;y++)if(hW[y][x]&&Math.random()<.09)hW[y][x]=false;
  const chambers=[];let tries=0;
  while(chambers.length<3&&tries++<300){
    const cx=randi(1,W-4),cy=randi(1,H-4);
    if(chambers.some(c=>Math.abs(c.x-cx)<5&&Math.abs(c.y-cy)<4))continue;
    chambers.push({x:cx,y:cy});
    for(let x=0;x<2;x++)for(let y=0;y<3;y++)vW[cx+x][cy+y]=false;
    for(let y=0;y<2;y++)for(let x=0;x<2;x++)hW[cy+y][cx+x]=false;
  }
  const FW=W*4+1,FH=H*4+1,solid=new Uint8Array(FW*FH).fill(1);
  for(let y=0;y<H;y++)for(let x=0;x<W;x++){
    const sx=x*4,sy=y*4;
    for(let i=1;i<=3;i++)for(let j=1;j<=3;j++)solid[(sy+j)*FW+(sx+i)]=0;
  }
  for(let x=0;x<W-1;x++)for(let y=0;y<H;y++)if(!vW[x][y])solid[(y*4+2)*FW+(x*4+4)]=0;
  for(let x=0;x<W;x++)for(let y=0;y<H-1;y++)if(!hW[y][x])solid[(y*4+4)*FW+(x*4+2)]=0;
  return {W,H,U,FW,FH,solid,chambers,vW,hW,worldW:FW*U,worldH:FH*U,
    openR:(x,y)=>x<W-1&&!vW[x][y],openD:(x,y)=>y<H-1&&!hW[y][x]};
}

function hitsSolid(M,x,y,r){
  const FW=M.FW;
  const x0=Math.floor((x-r)/U),x1=Math.floor((x+r)/U),y0=Math.floor((y-r)/U),y1=Math.floor((y+r)/U);
  for(let sy=y0;sy<=y1;sy++){
    if(sy<0||sy>=M.FH)continue;const row=sy*FW;
    for(let sx=x0;sx<=x1;sx++){
      if(sx<0||sx>=FW||!M.solid[row+sx])continue;
      const cx=clamp(x,sx*U,(sx+1)*U),cy=clamp(y,sy*U,(sy+1)*U);
      const dx=x-cx,dy=y-cy;if(dx*dx+dy*dy<r*r)return true;
    }
  }
  return false;
}
function moveBody(M,b,dx,dy){
  let d=dx;while(Math.abs(d)>0){const s=Math.sign(d)*Math.min(6,Math.abs(d));if(!hitsSolid(M,b.x+s,b.y,b.r))b.x+=s;else break;d-=s}
  d=dy;while(Math.abs(d)>0){const s=Math.sign(d)*Math.min(6,Math.abs(d));if(!hitsSolid(M,b.x,b.y+s,b.r))b.y+=s;else break;d-=s}
}
function cellOf(P){return {x:clamp(Math.floor(P.x/(U*4)),0,MW-1),y:clamp(Math.floor(P.y/(U*4)),0,MH-1)}}
function cellDist(M,sx,sy){
  const D=new Int16Array(M.W*M.H).fill(-1);const q=[sy*M.W+sx];D[sx]=0;let h=0;
  while(h<q.length){
    const c=q[h++],x=c%M.W,y=(c/M.W)|0,d=D[c]+1;
    if(x>0&&M.openR(x-1,y)&&D[c-1]<0){D[c-1]=d;q.push(c-1)}
    if(x<M.W-1&&M.openR(x,y)&&D[c+1]<0){D[c+1]=d;q.push(c+1)}
    if(y>0&&M.openD(x,y-1)&&D[c-M.W]<0){D[c-M.W]=d;q.push(c-M.W)}
    if(y<M.H-1&&M.openD(x,y)&&D[c+M.W]<0){D[c+M.W]=d;q.push(c+M.W)}
  }
  return D;
}

function pickCells(n,needChamber){
  const M=G.maze,pc=cellOf(G.player),D=cellDist(M,pc.x,pc.y);
  if(needChamber){
    const list=M.chambers.map(ch=>({x:ch.x+1,y:ch.y+1,d:D[(ch.y+1)*M.W+(ch.x+1)]||0}));
    shuffle(list);return list.slice(0,n).map(o=>({x:o.x,y:o.y}));
  }
  const cand=[];
  for(let y=0;y<M.H;y++)for(let x=0;x<M.W;x++){const d=D[y*M.W+x];if(d>=5)cand.push({x,y,d:d+rand(0,3)})}

    G.parts.push({type:'dot',x:P.x+Math.cos(a)*14,y:P.y+Math.sin(a)*14,vx:0,vy:0,l:.12,ml:.12,size:3,c:'#ffe9a8'});
  }
  P.trailT-=dt;
  if(P.trailT<=0){P.trailT=.035;G.trail.push({x:P.x,y:P.y,l:.4});if(G.trail.length>26)G.trail.shift()}
}
function bossUpdate(e,dt){
  const P=G.player,frac=e.hp/e.maxHp;
  e.phase=frac>.6?1:frac>.3?2:3;
  if(e.phase!==e.prevPhase){e.prevPhase=e.phase;sfx.phase();ringFx(e.x,e.y,e.r,'#ff3d5a');G.hitT=.15;G.shake=Math.max(G.shake,8)}
  const mult=e.phase===1?1:e.phase===2?1.3:1.7;
  e.fanT-=dt;e.spirT-=dt;e.ringT-=dt;
  if(e.fanT<=0){
    const a=Math.atan2(P.y-e.y,P.x-e.x),n=e.phase===3?7:5;
    for(let i=0;i<n;i++)spawnEB(e.x,e.y,a+(i-(n-1)/2)*.28,(185+G.wave*4)*G.bSpeed,'#ffb03b',4.6);
    e.fanT=1.1/mult;
  }
  if(e.spirT<=0){
    e.sAng+=.52;spawnEB(e.x,e.y,e.sAng,150*G.bSpeed,'#ff7a59',4);
    if(e.phase>1)spawnEB(e.x,e.y,e.sAng+Math.PI,150*G.bSpeed,'#ff7a59',4);
    e.spirT=.22/mult;
  }
  if(e.ringT<=0){
    const N=16+e.phase*4;
    for(let i=0;i<N;i++)spawnEB(e.x,e.y,i*TAU/N+(e.phase>1?Math.PI/N:0),135*G.bSpeed,'#e05cff',4.5);
    e.ringT=2.6/mult;ringFx(e.x,e.y,e.r+8,'#e05cff');
  }
  if(!e.sum1&&frac<=.7){e.sum1=true;summon(1)}
  if(!e.sum2&&frac<=.4){e.sum2=true;summon(2);ringFx(e.x,e.y,e.r+16,'#ff3d5a')}
}
function summon(n){
  const cells=pickCells(n,false);
  for(const cell of cells){
    const c=cellCenter(cell.x,cell.y);
    G.pending.push({e:makeEnemy('drone',c.x,c.y,G.wave),t:.7,t0:.7});
  }
}
function updEnemy(e,dt){
  e.rot+=dt*(e.type==='ring'?1.2:.6);e.t+=dt;
  if(e.flash>0)e.flash-=dt;
  if(e.spawnT>0){e.spawnT-=dt;return}
  const P=G.player;
  if(P.dead)return;
  if(e.type==='boss'){bossUpdate(e,dt);return}
  if(e.type==='orbiter'){e.ang+=dt*1.1;e.x=e.ax+Math.cos(e.ang)*e.orbitR;e.y=e.ay+Math.sin(e.ang)*e.orbitR}
  else{e.x=e.ax+Math.cos(e.t*1.7)*3;e.y=e.ay+Math.sin(e.t*2.1)*4}
  if(e.type==='spiker'){
    if(e.tele>0){e.tele-=dt;if(e.tele<=0){spawnEB(e.x,e.y,Math.atan2(P.y-e.y,P.x-e.x),360*G.bSpeed,'#ff3d5a',4.2,true);e.fireT=rand(1.7,2.5)}}
    else if((e.fireT-=dt)<=0)e.tele=.55;
    return;
  }
  e.fireT-=dt;
  if(e.fireT<=0){
    if(e.type==='drone'){
      const a=Math.atan2(P.y-e.y,P.x-e.x);
      for(let i=-1;i<=1;i++)spawnEB(e.x,e.y,a+i*.24,165*G.bSpeed,'#ff7a59',4);
      e.fireT=rand(1.35,1.85);
    }else if(e.type==='ring'){
      const N=9+Math.min(9,G.wave);
      for(let i=0;i<N;i++)spawnEB(e.x,e.y,e.ringOff+i*TAU/N,125*G.bSpeed,'#e05cff',4.5);
      e.ringOff+=Math.PI/N;e.fireT=rand(2.1,2.6);
    }else if(e.type==='orbiter'){
      const a=e.t*3.1;
      spawnEB(e.x,e.y,a,140*G.bSpeed,'#9dff57',4);
      spawnEB(e.x,e.y,a+Math.PI,140*G.bSpeed,'#9dff57',4);
      e.fireT=.16;
    }
  }
}
function offWorld(b,m){return b.x<-m||b.y<-m||b.x>G.maze.worldW+m||b.y>G.maze.worldH+m}
function updBullets(dt){
  const PB=G.pB,EB=G.eB,P=G.player;
  for(let i=PB.length-1;i>=0;i--){const b=PB[i];b.x+=b.vx*dt;b.y+=b.vy*dt;b.l-=dt;if(b.l<=0||offWorld(b,20))PB.splice(i,1)}
  for(let i=EB.length-1;i>=0;i--){const b=EB[i];b.x+=b.vx*dt;b.y+=b.vy*dt;b.l-=dt;if(b.l<=0||offWorld(b,40))EB.splice(i,1)}
  for(let i=PB.length-1;i>=0;i--){
    const b=PB[i];let hit=false;
    for(const e of G.enemies){
      if(e.dead||e.spawnT>0)continue;
      const rr2=e.r+4;
      if(dist2(b.x,b.y,e.x,e.y)<rr2*rr2){
        e.hp--;e.flash=.09;sfx.eHit();burst(b.x,b.y,ECOLOR[e.type],4,80);
        if(e.hp<=0)killEnemy(e,true);
        hit=true;break;
      }
    }
    if(hit)PB.splice(i,1);
  }
  if(P.inv<=0&&!P.dead){
    for(let i=EB.length-1;i>=0;i--){
      const b=EB[i],rr2=b.r+5;
      if(dist2(b.x,b.y,P.x,P.y)<rr2*rr2){EB.splice(i,1);hurtPlayer(b.x,b.y);break}
    }
    for(const e of G.enemies){
      if(e.dead||e.spawnT>0)continue;
      const rr2=e.r+5;
      if(dist2(e.x,e.y,P.x,P.y)<rr2*rr2){hurtPlayer(e.x,e.y);break}
    }
  }
}
function updGems(dt){
  const P=G.player;
  for(let i=G.gems.length-1;i>=0;i--){
    const g=G.gems[i];g.ph+=dt*3;
    const d2=dist2(g.x,g.y,P.x,P.y);
    if(!P.dead&&d2<70*70){const a=Math.atan2(P.y-g.y,P.x-g.x),s=lerp(60,340,1-Math.sqrt(d2)/70);g.x+=Math.cos(a)*s*dt;g.y+=Math.sin(a)*s*dt}
    if(!P.dead&&d2<15*15){G.gems.splice(i,1);G.score+=15;sfx.gem();burst(P.x,P.y,'#ffe9a8',5,70);floatTxt(P.x,P.y-14,'+15','#ffe9a8')}
  }
  for(let i=G.pickups.length-1;i>=0;i--){
    const p=G.pickups[i];p.ph+=dt*2;
    if(!P.dead&&dist2(p.x,p.y,P.x,P.y)<16*16){
      G.pickups.splice(i,1);
      if(p.type==='shield'){if(P.shield<1)P.shield=1;else G.score+=40;sfx.shield();floatTxt(P.x,P.y-18,'SHIELD','#4fd8ff')}
      else{if(P.hp<3)P.hp++;else G.score+=60;sfx.heart();floatTxt(P.x,P.y-18,'+1 LIGHT','#ff5c7a')}
    }
  }
}
function updParts(dt){
  for(let i=G.parts.length-1;i>=0;i--){
    const p=G.parts[i];p.l-=dt;
    if(p.l<=0){G.parts.splice(i,1);continue}
    if(p.type==='ring'){p.r+=p.vr*dt;continue}
    p.x+=p.vx*dt;p.y+=p.vy*dt;p.vx*=1-dt*2.4;p.vy=p.vy*(1-dt*2.4)-6*dt;
  }
  for(let i=G.floats.length-1;i>=0;i--){const f=G.floats[i];f.l-=dt;f.y-=26*dt;if(f.l<=0)G.floats.splice(i,1)}
}
function clampCam(){
  const M=G.maze,vw=view.w,vh=view.h;
  G.cam.x=vw>=M.worldW?(M.worldW-vw)/2:clamp(G.cam.x,0,M.worldW-vw);
  G.cam.y=vh>=M.worldH?(M.worldH-vh)/2:clamp(G.cam.y,0,M.worldH-vh);
}
function updateGame(dt){
  const M=G.maze,P=G.player;
  if(G.hintT>0)G.hintT-=dt;
  if(G.dieT>0){
    G.dieT-=dt;
    updBullets(dt*.4);updParts(dt);
    if(G.dieT<=0){
      state='OVER';
      if(G.score>G.best){G.best=G.score;saveBest(G.best);G.newBest=true}
      AU.ambT=.015;
    }
    return;
  }
  updPlayer(dt);
  for(let i=G.pending.length-1;i>=0;i--){
    const p=G.pending[i];p.t-=dt;
    if(p.t<=0){G.enemies.push(p.e);sfx.spawn();ringFx(p.e.x,p.e.y,26,ECOLOR[p.e.type]);G.pending.splice(i,1)}
  }
  if(G.phase==='banner'){G.phaseT-=dt;if(G.phaseT<=0)G.phase='play'}
  else if(G.phase==='clear'){G.phaseT-=dt;if(G.phaseT<=0)startWave(G.wave+1)}
  else if(G.phase==='bossdown'){G.phaseT-=dt;if(G.phaseT<=0){if(G.wave===10&&!G.endless)goWin();else startWave(G.wave+1)}}
  for(const e of G.enemies)if(!e.dead)updEnemy(e,dt);
  updBullets(dt);updGems(dt);updParts(dt);
  for(let i=G.trail.length-1;i>=0;i--){G.trail[i].l-=dt;if(G.trail[i].l<=0)G.trail.splice(i,1)}
  if(G.phase==='play'){
    G.waveT+=dt;
    const alive=G.enemies.some(e=>!e.dead);
    if(!alive&&G.pending.length===0){
      if(G.bossRef){G.phase='bossdown';G.phaseT=1.6}
      else{
        G.phase='clear';G.phaseT=1.5;
        const bonus=30+G.wave*10;G.score+=bonus;
        floatTxt(P.x,P.y-24,'SECTOR CLEARED +'+bonus,'#8ffbe8');sfx.ui();
      }
    }else if(alive&&G.waveT>45){
      for(const e of G.enemies)if(!e.dead)killEnemy(e,false);
    }
  }
  const bossAlive=G.bossRef&&!G.bossRef.dead;
  if(!bossAlive&&G.phase!=='banner'){AU.ambT=.045;AU.lpT=300}
  // camera
  const tx=P.x+P.fx*46-view.w/2,ty=P.y+P.fy*46-view.h/2;
  G.cam.x+=(tx-G.cam.x)*Math.min(1,dt*5);
  G.cam.y+=(ty-G.cam.y)*Math.min(1,dt*5);
  clampCam();
  G.shake*=Math.pow(.001,dt);if(G.shake<.2)G.shake=0;
  G.shx=rand(-1,1)*G.shake;G.shy=rand(-1,1)*G.shake;
  G.flash=Math.max(0,G.flash-dt*1.6);
}
function goWin(){
  state='WIN';AU.ambT=.02;sfx.win();
  if(G.score>G.best){G.best=G.score;saveBest(G.best);G.newBest=true}
}
