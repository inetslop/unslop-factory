  if(d.t==='wide'){G.fx.wide=10;popup(p.x,paddleY()-46,'WIDE WING','#5ee6ff');}

    if(live===0){const s=ballSpeed();G.balls.push({x:p.x,y:paddleY()-9,vx:0,vy:0,r:7,stuck:false,trail:[],ang:-.5});
      const b=G.balls[G.balls.length-1];b.stuck=false;b.vx=Math.cos(-Math.PI/2-.5)*s;b.vy=Math.sin(-Math.PI/2-.5)*s;live=1;}
    for(const b of G.balls){if(b.stuck)continue;if(G.balls.length>=7)break;const s=Math.hypot(b.vx,b.vy);
      for(const da of[-.45,.45]){const a=Math.atan2(b.vy,b.vx)+da;
        G.balls.push({x:b.x,y:b.y,vx:Math.cos(a)*s,vy:Math.sin(a)*s,r:7,stuck:false,trail:[],ang:0});}}
    popup(p.x,paddleY()-46,'SPLIT SHOT','#9dff57');}

      G.drops.push({x:br.x+br.w/2,bx:br.x+br.w/2,y:br.y+6,vy:105,t:tp,seed:rand(9)});}

function update(dt){
  G.t+=dt;
  for(const f in G.fx)if(G.fx[f]>0)G.fx[f]=Math.max(0,G.fx[f]-dt);
  if(G.invuln>0)G.invuln-=dt;
  G.shake=Math.max(0,G.shake-dt*42);G.flash=Math.max(0,G.flash-dt*1.6);
  G.comboPop=Math.max(0,G.comboPop-dt*3);

  /* paddle */
  const p=G.paddle;const prevX=p.x;
  let kdir=(keys.ArrowLeft||keys.KeyA?-1:0)+(keys.ArrowRight||keys.KeyD?1:0);
  if(kdir){p.x+=kdir*940*dt;mouse.on=false;}
  else if(mouse.on)p.x+=(mouse.x-p.x)*Math.min(1,dt*22);
  p.x=clamp(p.x,FIELD.x+p.w/2+6,FIELD.x+FIELD.w-p.w/2-6);
  p.vx=(p.x-prevX)/Math.max(dt,1e-4);
  const tw=120*(G.fx.wide>0?1.65:1);p.w+=(tw-p.w)*Math.min(1,dt*8);

  /* balls */
  if(G.mode==='playing'){
    for(const b of G.balls){
      if(b.stuck){b.x=p.x;b.y=paddleY()-9;continue;}
      const s=ballSpeed();
      for(let step=0;step<2;step++){
        b.x+=b.vx*dt/2;b.y+=b.vy*dt/2;
        if(b.x<FIELD.x+b.r){b.x=FIELD.x+b.r;b.vx=Math.abs(b.vx);sWall();spark(b.x,b.y,'#cfe6f4',3,110);}
        else if(b.x>FIELD.x+FIELD.w-b.r){b.x=FIELD.x+FIELD.w-b.r;b.vx=-Math.abs(b.vx);sWall();spark(b.x,b.y,'#cfe6f4',3,110);}
        if(b.y<FIELD.y+b.r){b.y=FIELD.y+b.r;b.vy=Math.abs(b.vy);sWall();spark(b.x,b.y,'#cfe6f4',3,110);}
        /* paddle bounce */
        const pt=paddleY();
        if(b.vy>0&&b.y+b.r>=pt&&b.y<b.pt||false){}
        if(b.vy>0&&Math.abs(b.x-p.x)<p.w/2+b.r*.6&&b.y+b.r>pt&&b.y<pt+16){
          const rel=clamp((b.x-p.x)/(p.w/2),-1,1),a=rel*1.05;
          b.vx=Math.sin(a)*s+p.vx*.14;b.vy=-Math.cos(a)*s;
          if(b.vy>-s*.3)b.vy=-s*.3;
          const nn=Math.hypot(b.vx,b.vy)||1;b.vx*=s/nn;b.vy*=s/nn;
          b.y=pt-b.r-0.5;G.combo=0;sPaddle(rel);spark(b.x,pt,'#9fe8ff',4,130);}
        /* bricks */
        for(let i=G.bricks.length-1;i>=0;i--){const br=G.bricks[i];
          const cx2=clamp(b.x,br.x,br.x+br.w),cy2=clamp(b.y,br.y,br.y+br.h);
          const dx=b.x-cx2,dy=b.y-cy2;
          if(dx*dx+dy*dy<b.r*b.r){
            if(Math.abs(dx)>Math.abs(dy)){b.vx=dx>0?Math.abs(b.vx):-Math.abs(b.vx);b.x=cx2+(dx>0?b.r:-b.r);}
            else{b.vy=dy>0?Math.abs(b.vy):-Math.abs(b.vy);b.y=cy2+(dy>0?b.r:-b.r);}
            hitBrick(i,b);break;}}
      }
      b.trail.push({x:b.x,y:b.y});if(b.trail.length>11)b.trail.shift();
    }
    const before=G.balls.length;
    G.balls=G.balls.filter(b=>b.stuck||b.y-b.r<FIELD.y+FIELD.h+46);
    if(before>G.balls.length&&G.balls.length===0)loseLife();

    /* bricks: turrets + emitters */
    for(const br of G.bricks){br.flash=Math.max(0,br.flash-dt*5);
      if(br.t===3){br.cd-=dt;if(br.cd<=0){fireAimed(br);br.cd=rand(1.5,2.3)*coolMul();}}
      else if(br.t===4){br.em-=dt;
        if(br.em<=0){br.em=.3*coolMul()*.9+.02;pushBullet(br.x+br.w/2,br.y+br.h/2,br.a,clamp(115+(G.level-1)*6,115,240),'spiral');br.a+=.42;}
        br.seed+=dt*3;}}

    /* drops */
    for(let i=G.drops.length-1;i>=0;i--){const d=G.drops[i];d.y+=d.vy*dt;d.x=d.bx+Math.sin(G.t*3+d.seed)*14;
      if(d.y>FIELD.y+FIELD.h+20){G.drops.splice(i,1);continue;}
      const pt=paddleY();
      if(d.y+11>pt&&d.y<pt+18&&Math.abs(d.x-p.x)<p.w/2+14){applyDrop(d);G.drops.splice(i,1);}}
  }

  /* bullets (also drift during level-clear) */
  const ts=G.fx.slow>0?.45:1;
  for(let i=G.bullets.length-1;i>=0;i--){const bl=G.bullets[i];bl.t+=dt;
    bl.x+=bl.vx*dt*ts;bl.y+=bl.vy*dt*ts;
    let wall=0;
    if(bl.x<FIELD.x+5){bl.x=FIELD.x+5;bl.vx=Math.abs(bl.vx);wall=1;}
    else if(bl.x>FIELD.x+FIELD.w-5){bl.x=FIELD.x+FIELD.w-5;bl.vx=-Math.abs(bl.vx);wall=1;}
    if(bl.y<FIELD.y+5){bl.y=FIELD.y+5;bl.vy=Math.abs(bl.vy);wall=1;}
    if(wall&&bl.k==='split'&&!bl.bounced){
      const a=Math.atan2(bl.vy,bl.vx),sp=Math.hypot(bl.vx,bl.vy)*1.05;
      pushBullet(bl.x,bl.y,a-.55,sp,'shard');pushBullet(bl.x,bl.y,a+.55,sp,'shard');
      spark(bl.x,bl.y,'#ffb347',6,150);G.bullets.splice(i,1);continue;}
    if(bl.y>FIELD.y+FIELD.h+40){G.bullets.splice(i,1);continue;}
    if(G.mode==='playing'){
      /* paddle: deflect or damage */
      const pt=paddleY();
      if(Math.abs(bl.x-p.x)<p.w/2+5&&bl.y>pt-5&&bl.y<pt+18){
        if(G.fx.deflect>0){bl.vy=-Math.abs(bl.vy);const rel=clamp((bl.x-p.x)/(p.w/2),-1,1);
          bl.vx+=rel*160;G.score+=5;sDeflect();spark(bl.x,pt,'#ff9dbb',3,120);}
        else if(G.invuln<=0){loseLife();break;}
        continue;}
      /* balls pop bullets */
      for(const b of G.balls){if(b.stuck)continue;
        const dx=b.x-bl.x,dy=b.y-bl.y,rr=b.r+5;
        if(dx*dx+dy*dy<rr*rr){G.stats.pops++;G.score+=10;sPopB();spark(bl.x,bl.y,'#fff2d0',4,140);
          G.bullets.splice(i,1);break;}}
    }}

  /* particles */
  for(let i=G.parts.length-1;i>=0;i--){const q=G.parts[i];q.life-=dt;
    if(q.life<=0){G.parts.splice(i,1);continue;}
    q.x+=q.vx*dt;q.y+=q.vy*dt;
    if(q.t==='sh'){q.vy+=560*dt;q.rot+=q.vr*dt;}
    else if(q.t==='sp'){q.vx*=1-2.4*dt;q.vy*=1-2.4*dt;}}
  for(let i=G.waves.length-1;i>=0;i--){const w=G.waves[i];w.life-=dt;
    w.r+=(w.rmax-w.r)*Math.min(1,dt*7);if(w.life<=0)G.waves.splice(i,1);}

  if(G.mode==='levelclear'){G.clearT-=dt;
    if(G.clearT<=0){G.level++;buildLevel(G.level);spawnStuckBall();G.mode='playing';}}
  if(G.mode==='over')G.overT+=dt;
}

const BCOL={aim:['#ff4f7e','#fff0f5'],spiral:['#9dff57','#f2ffe0'],split:['#ffb347','#fff3d6'],shard:['#ff8b3d','#ffe3c4']};

let lastT=performance.now();
function frame(now){
  requestAnimationFrame(frame);
  const dt=Math.min(.033,(now-lastT)/1000);lastT=now;
  if(G.mode!=='paused')update(dt);
  /* screen space */
  ctx.setTransform(dpr,0,0,dpr,0,0);

  /* world space */
  const shx=G.shake>0?rand(-G.shake,G.shake):0,shy=G.shake>0?rand(-G.shake,G.shake):0;

  /* flash */
  if(G.flash>0){ctx.fillStyle='rgba('+G.flashCol+','+(G.flash*.5)+')';ctx.fillRect(FIELD.x,FIELD.y,FIELD.w,FIELD.h);}
  /* vignette (screen space) */

}
