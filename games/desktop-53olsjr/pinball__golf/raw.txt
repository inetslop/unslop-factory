function collideSeg(b, x1,y1,x2,y2, rest){
  const dx=x2-x1, dy=y2-y1; const l2=dx*dx+dy*dy;
  let t=((b.x-x1)*dx+(b.y-y1)*dy)/l2; t=Math.max(0,Math.min(1,t));
  const px=x1+t*dx, py=y1+t*dy;
  let nx=b.x-px, ny=b.y-py; const d=Math.hypot(nx,ny);
  if(d>=b.r || d===0) return false;
  nx/=d; ny/=d;
  b.x=px+nx*b.r; b.y=py+ny*b.r;
  const vn=b.vx*nx+b.vy*ny;
  if(vn<0){ b.vx-=(1+rest)*vn*nx; b.vy-=(1+rest)*vn*ny;
    // tangential friction
    b.vx*=0.995; b.vy*=0.995; }
  return true;
}

```
const dx=x2-x1,dy=y2-y1,l2=dx*dx+dy*dy;
let t=((b.x-x1)*dx+(b.y-y1)*dy)/l2;t=t<0?0:t>1?1:t;
const cx=x1+t*dx, cy=y1+t*dy;
let nx=b.x-cx, ny=b.y-cy; const d=Math.hypot(nx,ny);
if(d>=rsum||d===0)return false;
nx/=d;ny/=d;
b.x=cx+nx*rsum;b.y=cy+ny*rsum;
const rxp=cx-px,ryp=cy-py;
const vfx=-om*ryp, vfy=om*rxp;
const rvx=b.vx-vfx, rvy=b.vy-vfy;
const vn=rvx*nx+rvy*ny;

return true;

/* ================= PHYSICS ================= */
function collideSeg(b,s){
  var dx=s.x2-s.x1, dy=s.y2-s.y1, l2=dx*dx+dy*dy;
  var t=((b.x-s.x1)*dx+(b.y-s.y1)*dy)/l2; t=t<0?0:(t>1?1:t);
  var px=s.x1+t*dx, py=s.y1+t*dy, nx=b.x-px, ny=b.y-py, d=Math.sqrt(nx*nx+ny*ny);
  if(d>=b.r||d===0) return false;
  nx/=d; ny/=d; b.x=px+nx*b.r; b.y=py+ny*b.r;
  var vn=b.vx*nx+b.vy*ny;
  if(vn<0){ b.vx-=(1+s.rest)*vn*nx; b.vy-=(1+s.rest)*vn*ny; b.vx*=0.995; b.vy*=0.995; }
  return true;
}
function collideFlipper(b,f){
  var tx=f.px+Math.cos(f.ang)*f.len, ty=f.py+Math.sin(f.ang)*f.len;
  var dx=tx-f.px, dy=ty-f.py, l2=dx*dx+dy*dy;
  var t=((b.x-f.px)*dx+(b.y-f.py)*dy)/l2; t=t<0?0:(t>1?1:t);
  var cx=f.px+t*dx, cy=f.py+t*dy, nx=b.x-cx, ny=b.y-cy, d=Math.sqrt(nx*nx+ny*ny), R=b.r+f.rad;
  if(d>=R||d===0) return false;
  nx/=d; ny/=d; b.x=cx+nx*R; b.y=cy+ny*R;
  var rxp=cx-f.px, ryp=cy-f.py, vfx=-f.om*ryp, vfy=f.om*rxp;
  var rvx=b.vx-vfx, rvy=b.vy-vfy, vn=rvx*nx+rvy*ny;
  if(vn<0){ b.vx-=1.32*vn*nx; b.vy-=1.32*vn*ny; }
  return true;
}
function hitSling(b,s){
  var dx=s.x2-s.x1, dy=s.y2-s.y1, l2=dx*dx+dy*dy;
  var t=((b.x-s.x1)*dx+(b.y-s.y1)*dy)/l2; t=t<0?0:(t>1?1:t);
  var px=s.x1+t*dx, py=s.y1+t*dy, nx=b.x-px, ny=b.y-py, d=Math.sqrt(nx*nx+ny*ny);
  if(d>=b.r||d===0) return;
  nx/=d; ny/=d; b.x=px+nx*b.r; b.y=py+ny*b.r;
  var vn=b.vx*s.nx+b.vy*s.ny, hit=vn<0;
  if(hit){ b.vx-=1.4*vn*s.nx; b.vy-=1.4*vn*s.ny; }
  var v2=b.vx*s.nx+b.vy*s.ny;
  if(v2<620){ b.vx+=(620-v2)*s.nx; b.vy+=(620-v2)*s.ny; }
  if(hit&&s.cd<=0){ s.cd=0.14; s.t=0.18; score+=10; addPop((s.x1+s.x2)/2,(s.y1+s.y2)/2,'+10','#ffd97a'); sfx.sling(); }
}
function hitBumper(b,c){
  var dx=b.x-c.x, dy=b.y-c.y, R=b.r+c.r, d2=dx*dx+dy*dy;
  if(d2>=R*R) return;
  var d=Math.sqrt(d2)||1, nx=dx/d, ny=dy/d;
  b.x=c.x+nx*R; b.y=c.y+ny*R;
  var vn=b.vx*nx+b.vy*ny, hit=vn<0;
  if(hit){ b.vx-=1.55*vn*nx; b.vy-=1.55*vn*ny; }
  var v2=b.vx*nx+b.vy*ny;
  if(v2<640){ b.vx+=(640-v2)*nx; b.vy+=(640-v2)*ny; }
  if(hit&&c.cd<=0){ c.cd=0.13; c.hitT=0.22; score+=25; addPop(b.x,b.y-c.r-10,'+25','#d7f5df'); sfx.bump(c.r); leaves(c); }
}
function hitTarget(b,g){
  var dx=b.x-g.x, dy=b.y-g.y, R=b.r+g.r, d2=dx*dx+dy*dy;
  if(d2>=R*R) return;
  var d=Math.sqrt(d2)||1, nx=dx/d, ny=dy/d;
  b.x=g.x+nx*R; b.y=g.y+ny*R;
  var vn=b.vx*nx+b.vy*ny, hit=vn<0;
  if(hit){ b.vx-=1.4*vn*nx; b.vy-=1.4*vn*ny; }
  var v2=b.vx*nx+b.vy*ny;
  if(v2<300){ b.vx+=(300-v2)*nx; b.vy+=(300-v2)*ny; }
  if(hit&&g.cd<=0){
    g.cd=0.15;
    if(!g.lit){
      g.lit=true; g.t=0.4; score+=100; addPop(g.x,g.y-34,'+100','#ffd97a'); sfx.target();
      var all=true; for(var i=0;i<TARGETS.length;i++) if(!TARGETS[i].lit) all=false;
      if(all) eagleMode();
    }
  }
}
function cupPhysics(b,h){
  var dx=CUP.x-b.x, dy=CUP.y-b.y, d=Math.sqrt(dx*dx+dy*dy);
  if(d<52&&d>0.1){
    var pull=(1-d/52)*1400;
    b.vx+=dx/d*pull*h; b.vy+=dy/d*pull*h;
    var sp=Math.sqrt(b.vx*b.vx+b.vy*b.vy);
    if((d<26&&sp<560)||d<15) sinkBall(b);
  }
}
function updFlip(f,pressed,h){
  var target=pressed?f.act:f.rest, dir=target>f.ang?1:(target<f.ang?-1:0);
  if(dir!==0){ f.om=dir*34; f.ang+=dir*Math.min(Math.abs(target-f.ang),34*h); } else f.om=0;
}
function step(h,kL,kR){
  updFlip(FLIPS[0],kL,h); updFlip(FLIPS[1],kR,h);
  for(var i=0;i<balls.length;i++){
    var b=balls[i]; if(b.state!=='play') continue;
    if(b.ready) continue;                       // resting on plunger
    b.vy+=G*h;
    for(var bi=0;bi<2;bi++){ var B=BUNKERS[bi], ex=(b.x-B.cx)/B.rx, ey=(b.y-B.cy)/B.ry;
      if(ex*ex+ey*ey<1){ b.vx*=(1-2.6*h); b.vy*=(1-2.6*h);
        var sp=Math.sqrt(b.vx*b.vx+b.vy*b.vy); if(sp>220&&Math.random()<8*h) sandPuff(b); } }
    cupPhysics(b,h);
    b.x+=b.vx*h; b.y+=b.vy*h;
    for(var w=0;w<WALLS.length;w++) collideSeg(b,WALLS[w]);
    for(var s=0;s<2;s++) hitSling(b,SLINGS[s]);
    for(var f=0;f<2;f++) collideFlipper(b,FLIPS[f]);
    for(var c=0;c<BUMPERS.length;c++) hitBumper(b,BUMPERS[c]);
    for(var g=0;g<TARGETS.length;g++) hitTarget(b,TARGETS[g]);
    var sp2=Math.sqrt(b.vx*b.vx+b.vy*b.vy);
    if(sp2>MAXV){ b.vx*=MAXV/sp2; b.vy*=MAXV/sp2; }
  }
}

/* ================= UPDATE ================= */
function update(dt){
  T+=dt;
  shake=Math.max(0,shake-30*dt); nudgeCd=Math.max(0,nudgeCd-dt);
  for(var i=annos.length-1;i>=0;i--){ annos[i].t+=dt; if(annos[i].t>annos[i].dur) annos.splice(i,1); }
  for(i=pops.length-1;i>=0;i--){ pops[i].t+=dt; if(pops[i].t>0.85) pops.splice(i,1); }
  for(i=particles.length-1;i>=0;i--){ var p=particles[i]; p.life+=dt;
    if(p.life>p.max){ particles.splice(i,1); continue; }
    p.vy+=(p.shape==='dot'?200:700)*dt; p.x+=p.vx*dt; p.y+=p.vy*dt; p.rot+=p.vr*dt; }
  BUMPERS.forEach(function(c){ c.hitT=Math.max(0,c.hitT-dt); c.cd-=dt; });
  SLINGS.forEach(function(s){ s.t=Math.max(0,s.t-dt); s.cd-=dt; });
  TARGETS.forEach(function(g){ g.t=Math.max(0,g.t-dt); g.cd-=dt; });

  if(mode!=='play'||paused) return;

  if(charging&&readyBall){ power=Math.min(1,power+dt*1.15); chargeUpd(); }
  if(spawnTimer>0){ spawnTimer-=dt; if(spawnTimer<=0) spawnPlungerBall(); }

  var steps=Math.min(8,Math.max(1,Math.ceil(dt/0.0042))), h=dt/steps;
  for(var s=0;s<steps;s++) step(h,kLdown,kRdown);

  for(i=balls.length-1;i>=0;i--){
    var b=balls[i];
    if(b.state==='sunk'){
      b.sunk+=dt; var k=Math.min(1,b.sunk/0.45);
      b.x=b.sx+(CUP.x-b.sx)*k; b.y=b.sy+(CUP.y-b.sy)*k;
      if(k>=1){ b.state='dead'; }
    } else if(b.state==='play'&&!b.ready){
      var sp=Math.sqrt(b.vx*b.vx+b.vy*b.vy);
      if(sp<12) b.slowT+=dt; else b.slowT=0;
      if(b.slowT>3){ b.vy-=420; b.vx+=(Math.random()-0.5)*200; b.slowT=0; }
      if(sp>140){ b.trail.push({x:b.x,y:b.y}); if(b.trail.length>12) b.trail.shift(); }
      else if(b.trail.length) b.trail.shift();
      if(b.y>1238) drainBall(b);
    }
  }
  balls=balls.filter(function(b){ return b.state!=='dead'; });
  if(advancing && !balls.some(function(b){return b.state==='sunk'||b.state==='play';})){
    advancing=false; holesSunk++; holeNum++; par=Math.min(7,3+Math.floor((holeNum-1)/2)); strokes=0;
    anno('HOLE '+holeNum,'PAR '+par,'#ffd97a',2); spawnTimer=1.0;
  }
}

var ptrs={};
stage.addEventListener('pointerdown',function(e){
  initAudio();
  if(e.target.closest('.overlay')) return;
  if(mode!=='play'||paused) return;
  var r=canvas.getBoundingClientRect(), x=e.clientX-r.left, y=e.clientY-r.top;
  var zone = (y < r.height*0.45) ? 'charge' : (x < r.width/2 ? 'L':'R');
  ptrs[e.pointerId]=zone;
  if(zone==='L') kLdown=true; else if(zone==='R') kRdown=true;
  else if(readyBall){ charging=true; power=0; chargeStart(); }
  e.preventDefault();
});
function ptrEnd(e){
  var zone=ptrs[e.pointerId]; delete ptrs[e.pointerId];
  if(zone==='L'){ var any=false; for(var id in ptrs) if(ptrs[id]==='L')any=true; if(!any) kLdown=false; }
  else if(zone==='R'){ any=false; for(var j in ptrs) if(ptrs[j]==='R')any=true; if(!any) kRdown=false; }
  else if(charging){ launchBall(); charging=false; chargeStop(); power=0; }
}
stage.addEventListener('pointerup',ptrEnd);
stage.addEventListener('pointercancel',ptrEnd);
