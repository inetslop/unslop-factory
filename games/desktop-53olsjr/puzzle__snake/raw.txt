function cellAt(e){if(!board)return null;const x=e.clientX-bx,y=e.clientY-by;if(x<0||y<0||x>=bw||y>=bh)return null;return Math.floor(y/cell)*board.cols+Math.floor(x/cell);}

function makeBoard(){

  layout();

}

function adjacentToEnd(i){const l=path.length?path[path.length-1]:board.head;
  const dr=Math.abs(((i/board.cols)|0)-((l/board.cols)|0)),dc=Math.abs((i%board.cols)-(l%board.cols));return dr+dc===1;}
function afterChange(){
  pathSet=new Set(path);checkReachable();
  routeValid=path.length===board.openCount&&path[path.length-1]===board.apple;
  if(routeValid&&!wasValid){sfx.ready();const p=cellCenter(board.apple);puff(p.x,p.y);}
  wasValid=routeValid;
}
function checkReachable(){
  if(path.length>=board.openCount){deadEnd=false;return;}
  const last=path[path.length-1],seen=new Set([last]),st=[last];let count=0;
  while(st.length){const i=st.pop();const r=(i/board.cols)|0,c=i%board.cols;
    const push=n=>{if(!board.rocks.has(n)&&!seen.has(n)){seen.add(n);if(!pathSet.has(n))count++;st.push(n);}};
    if(r>0)push(i-board.cols);if(r+1<board.rows)push(i+board.cols);if(c>0)push(i-1);if(c+1<board.cols)push(i+1);}
  deadEnd=count!==board.openCount-path.length;
}
function pressCell(i){
  if(board.rocks.has(i)){sfx.bump();return;}
  if(path.length===0){if(i===board.head){path.push(i);sfx.extend(1);afterChange();}else sfx.bump();return;}
  const k=path.indexOf(i);
  if(k>0){path.length=k+1;sfx.retract();afterChange();}
  else if(k<0&&adjacentToEnd(i)){path.push(i);sfx.extend(path.length);afterChange();}
  else sfx.bump();
}
function dragCell(i){
  if(board.rocks.has(i)||!path.length)return;
  const last=path[path.length-1];if(i===last)return;
  if(i===path[path.length-2]){path.pop();sfx.retract();afterChange();return;}
  const k=path.indexOf(i);
  if(k>0&&k<path.length-1){path.length=k+1;sfx.retract();afterChange();return;}
  if(k<0&&adjacentToEnd(i)){path.push(i);sfx.extend(path.length);afterChange();}
}
function keyExtend(dr,dc){
  const base=path.length?path[path.length-1]:board.head;
  const r=((base/board.cols)|0)+dr,c=(base%board.cols)+dc;
  if(r<0||c<0||r>=board.rows||c>=board.cols)return sfx.bump();
  const i=r*board.cols+c;
  if(board.rocks.has(i)||pathSet.has(i))return sfx.bump();
  path.push(i);sfx.extend(path.length);afterChange();
}
function clearPath(){if(path.length){path=[];redraws++;sfx.clearAll();afterChange();}}

function angLerp(a,b,t){let d=b-a;while(d>Math.PI)d-=2*Math.PI;while(d<-Math.PI)d+=2*Math.PI;return a+d*t;}

function drawBoard(t){

let lastT=performance.now();
function frame(now){
  requestAnimationFrame(frame);
  const dt=clamp((now-lastT)/1000,0,.05);lastT=now;const t=now/1000;

  if(state==='run'){
    const total=runPath.length,sp=5+Math.min(level*.3,4.5);
    runT=Math.min(total-1,runT+sp*dt);
    const f=Math.floor(runT);
    if(f>lastStep){lastStep=f;sfx.tick(f);const p=cellCenter(runPath[f]);puff(p.x,p.y);}
    const p0=pointAt(runT),p1=pointAt(Math.max(0,runT-.8));
    headAng=angLerp(headAng,Math.atan2(p0.y-p1.y,p0.x-p1.x),.4);
    if(runT>=total-1)complete();
  }
  if(state==='clear'&&!clearShown&&performance.now()-tClear>650){clearShown=true;showOv(clearOv);}

  for(let i=confetti.length-1;i>=0;i--){const c=confetti[i];c.t+=dt;if(c.t>=c.life){confetti.splice(i,1);continue;}c.vy+=c.g*dt;c.x+=c.vx*dt;c.y+=c.vy*dt;c.r+=c.vr*dt;}
  flash=Math.max(0,flash-dt*2.2);

  /* background */
  const g=ctx.createLinearGradient(0,0,0,H);
  g.addColorStop(0,'#10311f');g.addColorStop(.55,'#0c2417');g.addColorStop(1,'#081a10');
  ctx.fillStyle=g;ctx.fillRect(0,0,W,H);
  if(board){
    const cx=W/2,cy=by+bh/2,rg=ctx.createRadialGradient(cx,cy,10,cx,cy,Math.max(bw,bh)*.8);
    rg.addColorStop(0,'rgba(96,214,150,.1)');rg.addColorStop(1,'rgba(96,214,150,0)');

  }
  for(const f of flies){
    const x=f.x*W+Math.sin(t*f.sp+f.ph)*14,y=(((f.y-t*.006*f.sp)%1)+1)%1*H,a=.22+.3*(.5+.5*Math.sin(t*1.5+f.ph));
    ctx.fillStyle=f.c;ctx.globalAlpha=a*.25;ctx.beginPath();ctx.arc(x,y,f.r*3,0,7);ctx.fill();

  }
  const vg=ctx.createRadialGradient(W/2,H/2,Math.min(W,H)*.35,W/2,H/2,Math.max(W,H)*.75);
  vg.addColorStop(0,'rgba(0,0,0,0)');vg.addColorStop(1,'rgba(3,10,6,.5)');
  ctx.fillStyle=vg;ctx.fillRect(0,0,W,H);

  if(board)drawBoard(t);

  for(const c of confetti){const a=1-c.t/c.life;ctx.save();ctx.translate(c.x,c.y);ctx.rotate(c.r);ctx.globalAlpha=a;ctx.fillStyle=c.c;ctx.fillRect(-c.s/2,-c.s/2,c.s,c.s*.7);ctx.restore();}
  ctx.globalAlpha=1;
  if(flash>0){ctx.fillStyle='rgba(255,255,240,'+(flash*.3).toFixed(3)+')';ctx.fillRect(0,0,W,H);}

  updateHUD();
}

    else if(path.length===0){h="Drag from the snake's head (or use arrow keys) and fill every square once, ending on the apple.";}
    else{h='Thread every open square exactly once — '+(board.openCount-path.length)+' to go.';}
  }
  else if(state==='run')h='The serpent is loose…';
  else h='Garden devoured! SPACE for the next level.';
  setHint(h,c);
  cv.style.cursor=state==='plan'?'crosshair':'default';
}
