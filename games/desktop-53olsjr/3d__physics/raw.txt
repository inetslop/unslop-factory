const SINKR=.82,WALL_YMAX=.95,G_ACC=22;
function physStep(h){
  for(const b of balls){ if(b.state!==0)continue;
    b.v.y-=G_ACC*h;
    b.p.x+=b.v.x*h;b.p.y+=b.v.y*h;b.p.z+=b.v.z*h;
    if(b.p.y<R){b.p.y=R;
      if(b.v.y<0){if(b.v.y<-1.8)SFX.clack(-b.v.y/10);b.v.y=-b.v.y*.45;if(b.v.y<.9)b.v.y=0;}
      const sp=Math.hypot(b.v.x,b.v.z);
      if(sp>0){let ns=sp*Math.exp(-.55*h)-1.1*h;if(ns<.05)ns=0;const k=ns/sp;b.v.x*=k;b.v.z*=k;}}
    for(const pk of pockets){const dx=b.p.x-pk.x,dz=b.p.z-pk.z;
      if(dx*dx+dz*dz<SINKR*SINKR&&b.p.y<1.05){startSink(b,pk);break;}}
    if(b.state!==0)continue;
    if(b.p.y<WALL_YMAX){
      let near=false;for(const pk of pockets){const dx=b.p.x-pk.x,dz=b.p.z-pk.z;if(dx*dx+dz*dz<(SINKR+.3)*(SINKR+.3)){near=true;break;}}
      if(!near){
        if(b.p.x>HX){b.p.x=HX;if(b.v.x>0){if(b.v.x>1.5)SFX.clack(b.v.x/8);b.v.x*=-.7;}}
        else if(b.p.x<-HX){b.p.x=-HX;if(b.v.x<0){if(-b.v.x>1.5)SFX.clack(-b.v.x/8);b.v.x*=-.7;}}
        if(b.p.z>HZ){b.p.z=HZ;if(b.v.z>0){if(b.v.z>1.5)SFX.clack(b.v.z/8);b.v.z*=-.7;}}
        else if(b.p.z<-HZ){b.p.z=-HZ;if(b.v.z<0){if(-b.v.z>1.5)SFX.clack(-b.v.z/8);b.v.z*=-.7;}}
      }else{b.p.x=clamp(b.p.x,-HX-1.1,HX+1.1);b.p.z=clamp(b.p.z,-HZ-1.1,HZ+1.1);}
    }
    if(b.p.y<-4){voidDrop(b);continue;}
    const s2=b.v.x*b.v.x+b.v.y*b.v.y+b.v.z*b.v.z;
    if(s2>46*46){const k=46/Math.sqrt(s2);b.v.x*=k;b.v.y*=k;b.v.z*=k;}
  }
  for(let it=0;it<2;it++)for(let i=0;i<balls.length;i++){const a=balls[i];if(a.state!==0)continue;
    for(let j=i+1;j<balls.length;j++){const b=balls[j];if(b.state!==0)continue;
      const dx=b.p.x-a.p.x,dy=b.p.y-a.p.y,dz=b.p.z-a.p.z;const md=a.r+b.r;const d2=dx*dx+dy*dy+dz*dz;
      if(d2>=md*md||d2<1e-9)continue;const d=Math.sqrt(d2);const nx=dx/d,ny=dy/d,nz=dz/d;
      const ov=(md-d)*.5;a.p.x-=nx*ov;a.p.y-=ny*ov;a.p.z-=nz*ov;b.p.x+=nx*ov;b.p.y+=ny*ov;b.p.z+=nz*ov;
      const vn=(b.v.x-a.v.x)*nx+(b.v.y-a.v.y)*ny+(b.v.z-a.v.z)*nz;
      if(vn<0){const jm=-(1+.93)*vn/2;
        a.v.x-=jm*nx;a.v.y-=jm*ny;a.v.z-=jm*nz;b.v.x+=jm*nx;b.v.y+=jm*ny;b.v.z+=jm*nz;
        if(it===0&&-vn>1.4){SFX.thock(-vn/9);spark((a.p.x+b.p.x)/2,(a.p.y+b.y? (a.p.y+b.p.y)/2:0),(a.p.z+b.p.z)/2,[.8,.95,1],3,2.2);
          if(-vn>7)G.shake=Math.max(G.shake,.05);}}
    }}
  for(const bar of bars){
    const ang=barAng(bar),c=Math.cos(ang),s=Math.sin(ang);
    const ax=bar.cx+bar.hl*c,az=bar.cz+bar.hl*s,bx=bar.cx-bar.hl*c,bz=bar.cz-bar.hl*s;
    const abx=bx-ax,abz=bz-az,len2=abx*abx+abz*abz;const w=barOm(bar);
    for(const b of balls){if(b.state!==0)continue;
      let t=((b.p.x-ax)*abx+(b.p.z-az)*abz)/len2;t=clamp(t,0,1);
      const px=ax+abx*t,pz=az+abz*t;
      const dx=b.p.x-px,dy=b.p.y-bar.y,dz=b.p.z-pz;const md=b.r+bar.r;const d2=dx*dx+dy*dy+dz*dz;
      if(d2>=md*md||d2<1e-9)continue;const d=Math.sqrt(d2);const nx=dx/d,ny=dy/d,nz=dz/d;
      b.p.x=px+nx*md;b.p.y=bar.y+ny*md;b.p.z=pz+nz*md;

    }
  }
}

function computeAim(){

  for(const b of balls){if(b===cue||b.state!==0)continue;

    const per2=ox*ox+oz*oz-pr*pr,rr=4*R*R;if(per2>rr)continue;

let lastT=performance.now(),acc=0;
function tick(now){
  requestAnimationFrame(tick);
  let dt=(now-lastT)/1000;lastT=now;if(dt>.05)dt=.05;
  if(G.state!=='paused'){
    G.simT+=dt;
    if(G.state==='menu')CAM.yaw+=.07*dt;
    if(keys.has('KeyQ'))CAM.yaw+=1.6*dt;
    if(keys.has('KeyE'))CAM.yaw-=1.6*dt;
    if(['aiming','charging'].includes(G.state)){const sp=fine?.45:1.6;
      if(keys.has('KeyA')||keys.has('ArrowLeft'))G.angle+=sp*dt;
      if(keys.has('KeyD')||keys.has('ArrowRight'))G.angle-=sp*dt;}
    if(G.state==='charging')G.powerT+=dt/1.15;
    /* physics */
    acc+=dt;const h=1/240;let steps=0;
    while(acc>=h&&steps<12){physStep(h);acc-=h;steps++;}
    if(acc>h*12)acc=0;
    updParticles(dt);
    G.shake*=Math.exp(-5.5*dt);
    /* sinking animation + settle */
    let anySink=false,maxS=0;
    for(const b of balls){if(b.state===1){anySink=true;b.sinkT+=dt;
        const pk=b.pk;b.p.x+=(pk.x-b.p.x)*8*dt;b.p.z+=(pk.z-b.p.z)*8*dt;b.p.y-=5*dt;
        if(b.sinkT>.45)b.state=2;}
      else if(b.state===0){const s=Math.hypot(b.v.x,b.v.y,b.v.z);if(s>maxS)maxS=s;}}

      if(G.rollT>18){for(const b of balls)if(b.state===0){b.v.x=0;b.v.y=0;b.v.z=0;}settleShot();}
      else if(!anySink&&maxS<.06&&G.rollT>.45)settleShot();}
    hudSync();
  }
  render();
}
