(function(){'use strict';

var AC=null, master=null;
function ensureAudio(){
  if(AC){ if(AC.state==='suspended'){try{AC.resume();}catch(e){}} return; }
  try{
    var Ctx=window.AudioContext||window.webkitAudioContext; if(!Ctx)return;
    AC=new Ctx(); master=AC.createGain(); master.gain.value=0.16; master.connect(AC.destination);
  }catch(e){}
}
function tone(type,f0,f1,dur,vol,delay){
  if(!AC||muted)return; vol=vol||0.4; delay=delay||0;
  try{
    var t0=AC.currentTime+delay, o=AC.createOscillator(), g=AC.createGain();
    o.type=type; o.frequency.setValueAtTime(Math.max(20,f0),t0);
    if(f1&&f1!==f0)o.frequency.exponentialRampToValueAtTime(Math.max(20,f1),t0+dur);
    g.gain.setValueAtTime(0.0001,t0);
    g.gain.exponentialRampToValueAtTime(vol,t0+0.01);
    g.gain.exponentialRampToValueAtTime(0.0001,t0+dur);
    o.connect(g); g.connect(master); o.start(t0); o.stop(t0+dur+0.06);
  }catch(e){}
}
var wakaFlip=false;

/* ============================== MAZE ============================== */
function buildMaze(){
  grid=[]; var y,x;
  for(y=0;y<ROWS;y++){grid.push(new Array(COLS).fill(0));}
  for(x=0;x<COLS;x++){grid[0][x]=1;grid[ROWS-1][x]=1;}
  for(y=0;y<ROWS;y++){grid[y][0]=1;grid[y][COLS-1]=1;}
  grid[TUNNEL][0]=0; grid[TUNNEL][COLS-1]=0;
  var A=[3,6,9,12,15], B=[4,7,10,13,16], i;
  for(i=0;i<A.length;i++){grid[2][A[i]]=1; grid[10][A[i]]=1;}
  for(i=0;i<B.length;i++){grid[4][B[i]]=1; grid[8][B[i]]=1;}
  for(x=8;x<=10;x++){grid[5][x]=1;grid[7][x]=1;}
}
function nextCell(x,y,d){
  var nx=x+d.x; if(nx<0)nx=COLS-1; else if(nx>=COLS)nx=0;
  return {x:nx,y:y+d.y};
}
function reachable(){
  var seen=new Uint8Array(COLS*ROWS), stack=[TUNNEL*COLS+9], k, d, c, j, i;
  seen[stack[0]]=1;
  while(stack.length){
    var p=stack.pop(), x=p%COLS, y=(p/COLS)|0;
    for(k in DIRS){ d=DIRS[k]; c=nextCell(x,y,d); if(grid[c.y][c.x])continue;
      j=c.y*COLS+c.x; if(!seen[j]){seen[j]=1;stack.push(j);} }
  }
  return seen;
}
function fillDots(){
  dots=new Int8Array(COLS*ROWS); dotsLeft=0;
  var seen=reachable(), pellets=[[TUNNEL,2],[TUNNEL,COLS-3],[1,9],[11,9]], y,x,i,p,isP;
  for(y=0;y<ROWS;y++)for(x=0;x<COLS;x++){
    i=y*COLS+x;
    if(grid[y][x]||!seen[i])continue;
    isP=false; for(p=0;p<pellets.length;p++)if(pellets[p][0]===y&&pellets[p][1]===x)isP=true;
    dots[i]=isP?2:1; dotsLeft++;
  }
}

/* ============================== SPAWNS / RESETS ============================== */
function spawnSnake(){
  snake=[{x:9,y:TUNNEL},{x:8,y:TUNNEL},{x:7,y:TUNNEL}];
  prevSnake=snake.map(function(s){return {x:s.x,y:s.y};});
  dir='right'; inputQ=[]; grow=0;
}
function openDirs(x,y,excludeOpp){
  var out=[], k, d, c;
  for(k in DIRS){
    if(excludeOpp&&k===OPP[excludeOpp])continue;
    d=DIRS[k]; c=nextCell(x,y,d);
    if(!grid[c.y][c.x])out.push(k);
  }
  return out;
}
function spawnGhosts(){
  var defs=[[1,3,'#ff3355'],[1,15,'#ff7ae0'],[11,5,'#35e0d8']];
  var count=level>=3?3:2;
  ghosts=[];
  for(var i=0;i<count;i++){
    var d=defs[i], od=openDirs(d[0],d[1],'')||['right'];
    ghosts.push({x:d[0],y:d[1],px:d[0],py:d[1],sx:d[0],sy:d[1],dir:od[(Math.random()*od.length)|0],
      color:d[2],dead:false,respawnAt:0,invulnUntil:0,fright:false,phase:i*2.1});
  }
}

/* ============================== GAME LOGIC ============================== */
function queueDir(n){
  if(state!=='play'&&state!=='ready')return;
  var last=inputQ.length?inputQ[inputQ.length-1]:dir;
  if(n===last||OPP[last]===n)return;
  if(inputQ.length<3)inputQ.push(n);
}
function distTo(x,y,tx,ty){var dx=Math.abs(x-tx);dx=Math.min(dx,COLS-dx);return dx+Math.abs(y-ty);}
function isDeadEnd(x,y){return openDirs(x,y,'').length<=1;}

function demoDir(){
  if(demoTarget<0||!dots[demoTarget]){
    var idxs=[],i; for(i=0;i<dots.length;i++)if(dots[i])idxs.push(i);
    demoTarget=idxs.length?idxs[(Math.random()*idxs.length)|0]:-1;
  }
  if(demoTarget<0)return dir;
  var tx=demoTarget%COLS, ty=(demoTarget/COLS)|0, h=snake[0], best=1e9, pick=null, opts=openDirs(h.x,h.y,dir), k,d,c,s;
  for(k=0;k<opts.length;k++){ d=DIRS[opts[k]]; c=nextCell(h.x,h.y,d); s=distTo(c.x,c.y,tx,ty)+Math.random()*0.5; if(s<best){best=s;pick=opts[k];} }
  return pick||dir;
}

function stepSnake(){
  if(demo&&inputQ.length===0)dir=demoDir();
  if(inputQ.length){var nd=inputQ.shift(); if(nd!==OPP[dir])dir=nd;}
  var h=snake[0], d=DIRS[dir];
  var nx=h.x+d.x, ny=h.y+d.y;
  if(nx<0)nx=COLS-1; else if(nx>=COLS)nx=0;
  if(grid[ny][nx]){ prevSnake=snake.map(function(s){return {x:s.x,y:s.y};}); return; } // wall blocks
  var willGrow=grow>0, i;
  for(i=1;i<snake.length;i++){ if(!willGrow&&i===snake.length-1)continue; if(snake[i].x===nx&&snake[i].y===ny){kill();return;} }
  prevSnake=snake.map(function(s){return {x:s.x,y:s.y};});
  snake.unshift({x:nx,y:ny});
  if(willGrow)grow--; else snake.pop();
  var idx=ny*COLS+nx, dv=dots[idx];
  if(dv===1){
    dots[idx]=0; dotsLeft--;
    score+=feastActive?20:10; sDot();
    burst(nx,ny,'#f4f7ff',5,cell*0.03);
    if(snake.length<MAXLEN)grow++;
    else if(!feastShown){ feastShown=true; feastActive=true; popup(nx,ny,'FEAST x2','#ffd91f'); }
  }else if(dv===2){
    dots[idx]=0; dotsLeft--;
    score+=50; grow+=3; frightT=frightMsFor(level); chain=0;
    ghosts.forEach(function(g){if(!g.dead)g.fright=true;});
    sPellet(); popup(nx,ny,'POWER','#7df9ff'); shake=4; burst(nx,ny,'#7df9ff',14,cell*0.05);
  }
  if(dv&&!demo){ updateHUD(); if(score>hi)hi=score; hiV.textContent=String(Math.max(hi,score)).padStart(6,'0'); }
  if(dotsLeft<=0)levelClear();
}

function levelClear(){
  if(demo){ fillDots(); return; }
  var bonus=300+level*150; score+=bonus; updateHUD();
  if(score>hi){hi=score; saveLS('pacSnakeHi',String(hi));}
  sLevel(); setState('levelup'); levelT=1700;
  banner.text='LEVEL '+level+' CLEAR  +'+bonus; banner.t=banner.dur=1600; banner.color='#58f27c';
}

function stepGhost(g){
  if(g.dead)return;
  var opts=openDirs(g.x,g.y,g.dir);
  if(!opts.length){ g.px=g.x; g.py=g.y; return; }
  var pick,k,d,c,h=snake[0];
  if(frightT>0&&Math.random()<0.5){ pick=opts[(Math.random()*opts.length)|0]; }
  else{
    var best=1e9, bl=[];
    for(k=0;k<opts.length;k++){
      d=DIRS[opts[k]]; c=nextCell(g.x,g.y,d);
      var s=distTo(c.x,c.y,h.x,h.y); if(isDeadEnd(c.x,c.y))s+=30;
      if(s<best){best=s;bl=[opts[k]];} else if(s===best)bl.push(opts[k]);
    }
    pick=bl[(Math.random()*bl.length)|0];
  }
  g.px=g.x; g.py=g.y; g.dir=pick;
  var nc=nextCell(g.x,g.y,DIRS[pick]); g.x=nc.x; g.y=nc.y;
}

function checkHits(){
  var h=snake[0];
  for(var i=0;i<ghosts.length;i++){
    var g=ghosts[i];
    if(g.dead||tGlobal<g.invulnUntil)continue;
    if(g.x===h.x&&g.y===h.y){
      if(g.fright)eatGhost(g); else { kill(); return; }
    }
  }
}
function eatGhost(g){
  g.dead=true; g.respawnAt=tGlobal+5500; chain++;
  var pts=200*Math.pow(2,Math.min(chain-1,3)); score+=pts; grow+=4;
  sGhost(chain); popup(g.x,g.y,'+'+pts,'#7df9ff'); burst(g.x,g.y,g.color,20,cell*0.06); shake=6;
  if(!demo){ updateHUD(); if(score>hi)hi=score; hiV.textContent=String(Math.max(hi,score)).padStart(6,'0'); }
}
function revive(g){
  g.dead=false; g.x=g.sx; g.y=g.sy; g.px=g.x; g.py=g.y;
  g.fright=frightT>0; g.invulnUntil=tGlobal+1500;
  var od=openDirs(g.x,g.y,''); if(od.length)g.dir=od[(Math.random()*od.length)|0];
}

function kill(){
  if(demo){ resetPositions(); return; }
  lives--; updateHUD();
  setState('dying'); dieT=1400; shake=12; sDeath();
  var h=snake[0]; burst(h.x,h.y,'#ffd91f',28,cell*0.07);
}

  $('overHi').textContent=String(hi).padStart(6,'0');

/* ============================== UPDATE ============================== */
function update(dt){
  tGlobal+=dt;
  shake=Math.max(0,shake-dt*0.012);
  if(banner.t>0)banner.t-=dt;
  if(state!=='paused'){
    var i,p;
    for(i=particles.length-1;i>=0;i--){ p=particles[i]; p.x+=p.vx*dt; p.y+=p.vy*dt; p.vx*=Math.pow(0.94,dt/16); p.vy*=Math.pow(0.94,dt/16); p.life-=dt; if(p.life<=0)particles.splice(i,1); }
    for(i=popups.length-1;i>=0;i--){ popups[i].t+=dt; if(popups[i].t>popups[i].dur)popups.splice(i,1); }
  }
  switch(state){
    case 'menu': case 'over': demoStep(dt); break;
    case 'ready': readyT-=dt; if(readyT<=0)setState('play'); break;
    case 'levelup': levelT-=dt;
      if(levelT<=0){ level++; snakeInt=intFor(level); fillDots(); resetPositions(); updateHUD(); setState('ready'); readyT=1200; banner.text='LEVEL '+level; banner.t=banner.dur=1100; banner.color='#5d79ff'; }
      break;
    case 'dying': dieT-=dt; if(dieT<=0){ if(lives<=0)gameOver(); else { resetPositions(); setState('ready'); readyT=1000; banner.text='READY?'; banner.t=banner.dur=950; banner.color='#ffd91f'; } }
      break;
    case 'play': playStep(dt); break;
  }
}
function playStep(dt){
  if(frightT>0){ frightT-=dt; if(frightT<=0){ frightT=0; chain=0; ghosts.forEach(function(g){g.fright=false;}); } }
  for(var r=0;r<ghosts.length;r++){ var g=ghosts[r]; if(g.dead&&tGlobal>=g.respawnAt)revive(g); }
  accS+=dt; var n=0;
  while(accS>=snakeInt&&n++<3){ accS-=snakeInt; stepSnake(); if(state!=='play')return; }
  var gi=frightT>0?snakeInt*1.9:snakeInt*1.35;
  accG+=dt; n=0;
  while(accG>=gi&&n++<3){ accG-=gi; for(var i=0;i<ghosts.length;i++)stepGhost(ghosts[i]); checkHits(); if(state!=='play')return; }
}
function demoStep(dt){
  if(demoTarget<0||!dots[demoTarget])demoTarget=-1;
  accS+=dt; var n=0;
  while(accS>=115&&n++<3){ accS-=115; stepSnake(); }
  accG+=dt; n=0;
  while(accG>=155&&n++<3){ accG-=155; for(var i=0;i<ghosts.length;i++)stepGhost(ghosts[i]); checkHits(); }
}

var W=0,H=0,DPR=1,cell=24,ox=0,oy=0,boardW=0,boardH=0,padW=0;
var bgCv=document.createElement('canvas'), wallCv=document.createElement('canvas');
var dotSpr=document.createElement('canvas'), pelletSpr=document.createElement('canvas');
var stars=[];
(function(){ for(var i=0;i<90;i++)stars.push({x:Math.random(),y:Math.random(),r:Math.random()*1.6+0.4,v:Math.random()*0.012+0.005,tw:Math.random()*2+0.5,ph:Math.random()*6.28,a:Math.random()*0.45+0.12}); })();

function rr(c,x,y,w,h,r){ r=Math.min(r,w/2,h/2); c.beginPath(); c.moveTo(x+r,y); c.arcTo(x+w,y,x+w,y+h,r); c.arcTo(x+w,y+h,x,y+h,r); c.arcTo(x,y+h,x,y,r); c.arcTo(x,y,x+w,y,r); c.closePath(); }

function buildBg(){
  bgCv.width=Math.max(1,W*DPR); bgCv.height=Math.max(1,H*DPR);
  var c=bgCv.getContext('2d'); c.setTransform(DPR,0,0,DPR,0,0);
  c.fillStyle='#04060f'; c.fillRect(0,0,W,H);
  var g=c.createRadialGradient(ox+boardW/2,oy+boardH/2,10,ox+boardW/2,oy+boardH/2,Math.max(W,H)*0.75);
  g.addColorStop(0,'rgba(38,62,170,0.24)'); g.addColorStop(1,'rgba(0,0,0,0)');
  c.fillStyle=g; c.fillRect(0,0,W,H);
  var g2=c.createRadialGradient(0,0,0,0,0,W*0.7); g2.addColorStop(0,'rgba(43,70,255,0.10)'); g2.addColorStop(1,'rgba(0,0,0,0)');
  c.fillStyle=g2; c.fillRect(0,0,W,H);
  var g3=c.createRadialGradient(W,H,0,W,H,W*0.7); g3.addColorStop(0,'rgba(53,224,216,0.08)'); g3.addColorStop(1,'rgba(0,0,0,0)');
  c.fillStyle=g3; c.fillRect(0,0,W,H);
}
function buildWalls(){
  padW=Math.ceil(cell*0.6);
  wallCv.width=(boardW+padW*2)*DPR; wallCv.height=(boardH+padW*2)*DPR;
  var c=wallCv.getContext('2d'); c.setTransform(DPR,0,0,DPR,0,0);
  c.clearRect(0,0,boardW+padW*2,boardH+padW*2);
  rr(c,1,1,boardW+padW*2-2,boardH+padW*2-2,cell*0.35);
  c.fillStyle='rgba(7,12,34,0.88)'; c.fill();
  c.strokeStyle='rgba(93,121,255,0.4)'; c.lineWidth=2; c.stroke();
  var inset=cell*0.05, r=cell*0.2, x,y;
  c.save(); c.shadowColor='rgba(80,120,255,0.9)'; c.shadowBlur=cell*0.45; c.fillStyle='#1b2fae';
  for(y=0;y<ROWS;y++)for(x=0;x<COLS;x++){
    if(!grid[y][x])continue;
    rr(c,padW+x*cell+inset,padW+y*cell+inset,cell-inset*2,cell-inset*2,r); c.fill();
  }
  c.restore();
  c.fillStyle='rgba(140,170,255,0.35)';
  for(y=0;y<ROWS;y++)for(x=0;x<COLS;x++){
    if(!grid[y][x])continue;
    rr(c,padW+x*cell+cell*0.3,padW+y*cell+cell*0.28,cell*0.4,cell*0.16,cell*0.08); c.fill();
  }
  // tunnel chevrons
  var ty=padW+TUNNEL*cell+cell/2;
  c.fillStyle='rgba(255,217,31,0.85)'; c.shadowColor='rgba(255,217,31,0.8)'; c.shadowBlur=8;
  function chev(xp,dirn){ c.beginPath(); c.moveTo(xp,ty-cell*0.22); c.lineTo(xp+dirn*cell*0.3,ty); c.lineTo(xp,ty+cell*0.22); c.closePath(); c.fill(); }
  chev(padW+cell*0.4,-1); chev(padW+cell*0.95,-1);
  chev(padW+boardW-cell*0.4,1); chev(padW+boardW-cell*0.95,1);
  c.shadowBlur=0;
}
function makeSprites(){
  [dotSpr,pelletSpr].forEach(function(s){s.width=cell*DPR;s.height=cell*DPR;});
  var c=dotSpr.getContext('2d'); c.setTransform(DPR,0,0,DPR,0,0);
  var g=c.createRadialGradient(cell/2,cell/2,0,cell/2,cell/2,cell*0.45);
  g.addColorStop(0,'#ffffff'); g.addColorStop(0.35,'rgba(255,255,255,0.9)'); g.addColorStop(1,'rgba(255,255,255,0)');
  c.fillStyle=g; c.fillRect(0,0,cell,cell);
  c= pelletSpr.getContext('2d'); c.setTransform(DPR,0,0,DPR,0,0);
  var g2=c.createRadialGradient(cell/2,cell/2,0,cell/2,cell/2,cell*0.5);
  g2.addColorStop(0,'#ffffff'); g2.addColorStop(0.4,'rgba(230,240,255,0.95)'); g2.addColorStop(1,'rgba(200,220,255,0)');
  c.fillStyle=g2; c.fillRect(0,0,cell,cell);
}
function fit(){
  DPR=Math.min(2,window.devicePixelRatio||1);
  W=window.innerWidth; H=window.innerHeight;
  cv.width=W*DPR; cv.height=H*DPR; cv.style.width=W+'px'; cv.style.height=H+'px';
  ctx.setTransform(DPR,0,0,DPR,0,0);
  var hudH=(hudEl.getBoundingClientRect().height||60)+6;
  cell=Math.max(10,Math.floor(Math.min((W-20)/COLS,(H-hudH-24)/ROWS)));
  boardW=cell*COLS; boardH=cell*ROWS;
  ox=Math.round((W-boardW)/2);
  oy=Math.max(hudH,Math.round(hudH+(H-hudH-boardH)/2));
  buildBg(); buildWalls(); makeSprites();
}

/* ============================== RENDER ============================== */
function lerp(a,b,t){return a+(b-a)*t;}
function wrapLerp(pv,c,p){
  var dx=c.x-pv.x, dy=c.y-pv.y;
  if(dx>COLS/2)pv={x:pv.x-COLS,y:pv.y}; else if(dx<-COLS/2)pv={x:pv.x+COLS,y:pv.y};
  return {x:ox+(lerp(pv.x,c.x,p))*cell+cell/2, y:oy+(lerp(pv.y,c.y,p))*cell+cell/2};
}
function drawStars(){
  for(var i=0;i<stars.length;i++){
    var s=stars[i], off=(tGlobal*s.v*0.00002)%1;
    var yy=((s.y-off)%1+1)%1;
    ctx.globalAlpha=s.a*(0.55+0.45*Math.sin(tGlobal*0.001*s.tw+s.ph));
    ctx.fillStyle='#9fb4ff';
    ctx.fillRect(s.x*W,yy*H,s.r,s.r);
  }
  ctx.globalAlpha=1;
}
function drawDots(){
  if(!dots)return;
  var i,x,y;
  for(i=0;i<dots.length;i++){
    if(dots[i]===1){ x=(i%COLS)*cell, y=((i/COLS)|0)*cell; ctx.drawImage(dotSpr,ox+x,oy+y); }
  }
  for(i=0;i<dots.length;i++){
    if(dots[i]!==2)continue;
    x=(i%COLS), y=(i/COLS)|0;
    var p=0.9+0.18*Math.sin(tGlobal*0.008+i), a=0.65+0.35*Math.sin(tGlobal*0.01+i);
    ctx.save(); ctx.globalAlpha=a;
    ctx.translate(ox+x*cell+cell/2,oy+y*cell+cell/2); ctx.scale(p,p);
    ctx.drawImage(pelletSpr,-cell/2,-cell/2); ctx.restore();
  }
  ctx.globalAlpha=1;
}
function drawPac(cx,cy,r,ang,m){
  ctx.beginPath(); ctx.moveTo(cx,cy);
  ctx.arc(cx,cy,r,ang+m,ang-m+Math.PI*2);
  ctx.closePath(); ctx.fillStyle='#ffd91f'; ctx.fill();
}
function drawSnake(){
  var p=(state==='play')?Math.min(1,accS/snakeInt):1;
  var len=snake.length, i;
  for(i=len-1;i>=1;i--){
    var c=snake[i], pv=prevSnake[i]||c, pos=wrapLerp(pv,c,p);
    var t=i/Math.max(1,len-1), r=cell*(0.40-0.22*t);
    ctx.fillStyle='rgba(255,217,31,0.10)';
    ctx.beginPath(); ctx.arc(pos.x,pos.y,r*1.6,0,Math.PI*2); ctx.fill();
    ctx.fillStyle='hsl('+(48-20*t)+' 100% '+(58-10*t)+'%)';
    ctx.beginPath(); ctx.arc(pos.x,pos.y,r,0,Math.PI*2); ctx.fill();
  }
  var h=snake[0], hp=prevSnake[0]||h, pos2=wrapLerp(hp,h,p);
  var ang=DIRS[dir].a, m=(Math.sin(tGlobal*0.02)+1)/2*0.75+0.08;
  var sc=1, rot=0;
  if(state==='dying'){ var q=1-dieT/1400; sc=Math.max(0,1-q); rot=q*q*6; m=Math.min(1.5,0.2+q*1.3); }
  if(sc>0.03){
    ctx.fillStyle='rgba(255,217,31,0.18)';
    ctx.beginPath(); ctx.arc(pos2.x,pos2.y,cell*0.46*sc*1.55,0,Math.PI*2); ctx.fill();
    drawPac(pos2.x,pos2.y,cell*0.46*sc,ang+rot,m);
  }
}
function drawGhost(g,cx,cy,r){
  var fright=g.fright&&frightT>0;
  var blink=fright&&frightT<1600&&(Math.floor(tGlobal/240)%2===0);
  var bodyC=fright?(blink?'#eef2ff':'#3940e8'):g.color;
  ctx.globalAlpha=0.15; ctx.fillStyle=fright?'#7d86ff':g.color;
  ctx.beginPath(); ctx.arc(cx,cy,r*1.5,0,Math.PI*2); ctx.fill(); ctx.globalAlpha=1;
  var o=Math.sin(tGlobal*0.012+g.phase)*r*0.10, by=cy+r*0.86, n=3, wdt=2*r/n, i;
  ctx.beginPath(); ctx.arc(cx,cy-r*0.1,r,Math.PI,0);
  ctx.lineTo(cx+r,by);
  for(i=0;i<n;i++){
    var xR=cx+r-i*wdt;
    ctx.quadraticCurveTo(xR-wdt/2, by-r*0.55+(i%2?o:-o), xR-wdt, by);
  }
  ctx.closePath(); ctx.fillStyle=bodyC; ctx.fill();
  var ex=r*0.34, ey=cy-r*0.18, d=DIRS[g.dir];
  if(!fright){
    ctx.fillStyle='#fff';
    ctx.beginPath(); ctx.arc(cx-ex,ey,r*0.27,0,Math.PI*2); ctx.fill();
    ctx.beginPath(); ctx.arc(cx+ex,ey,r*0.27,0,Math.PI*2); ctx.fill();
    ctx.fillStyle='#101c4a';
    ctx.beginPath(); ctx.arc(cx-ex+d.x*r*0.12,ey+d.y*r*0.12,r*0.13,0,Math.PI*2); ctx.fill();
    ctx.beginPath(); ctx.arc(cx+ex+d.x*r*0.12,ey+d.y*r*0.12,r*0.13,0,Math.PI*2); ctx.fill();
  }else{
    ctx.fillStyle='#f4f7ff';
    ctx.beginPath(); ctx.arc(cx-ex,ey,r*0.09,0,Math.PI*2); ctx.fill();
    ctx.beginPath(); ctx.arc(cx+ex,ey,r*0.09,0,Math.PI*2); ctx.fill();
    ctx.strokeStyle='#f4f7ff'; ctx.lineWidth=Math.max(2,r*0.09); ctx.lineJoin='round';
    ctx.beginPath();
    for(i=0;i<=4;i++){
      var mx=cx-r*0.5+i*r*0.25, my=cy+r*0.38+(i%2?-r*0.1:r*0.1);
      if(i===0)ctx.moveTo(mx,my); else ctx.lineTo(mx,my);
    }
    ctx.stroke();
  }
}
function drawGhosts(){
  var p=(state==='play')?Math.min(1,accG/(frightT>0?snakeInt*1.9:snakeInt*1.35)):1;
  for(var i=0;i<ghosts.length;i++){
    var g=ghosts[i]; if(g.dead)continue;
    var pos=wrapLerp({x:g.px,y:g.py},g,p);
    drawGhost(g,pos.x,pos.y,cell*0.42);
  }
}
function drawFx(){
  var i,p;
  for(i=0;i<particles.length;i++){
    p=particles[i]; var a=p.life/p.max;
    ctx.globalAlpha=a; ctx.fillStyle=p.c;
    ctx.beginPath(); ctx.arc(p.x,p.y,Math.max(0.4,p.r*a),0,Math.PI*2); ctx.fill();
  }
  ctx.globalAlpha=1;
  for(i=0;i<popups.length;i++){
    p=popups[i]; var a=1-p.t/p.dur;
    ctx.globalAlpha=Math.min(1,a*1.4);
    ctx.font='700 '+Math.max(12,cell*0.5)+'px ui-monospace,monospace';
    ctx.textAlign='center'; ctx.fillStyle=p.c;
    ctx.shadowColor=p.c; ctx.shadowBlur=8;
    ctx.fillText(p.txt,p.x,p.y-p.t*0.03);
    ctx.shadowBlur=0;
  }
  ctx.globalAlpha=1;
}
function drawBanner(){
  if(banner.t<=0)return;
  var el=banner.dur-banner.t, a=Math.min(1,el/120)*Math.min(1,banner.t/300);
  var s=0.8+0.2*Math.min(1,el/160);
  ctx.save();
  ctx.translate(ox+boardW/2,oy+boardH*0.5); ctx.scale(s,s);
  ctx.globalAlpha=Math.max(0,a);
  ctx.font='900 '+Math.min(cell*1.35,48)+'px "Arial Black","Segoe UI",system-ui,sans-serif';
  ctx.textAlign='center'; ctx.textBaseline='middle';
  ctx.shadowColor=banner.color; ctx.shadowBlur=22; ctx.fillStyle=banner.color;
  ctx.fillText(banner.text,0,0);
  ctx.restore();
}
function render(){
  ctx.drawImage(bgCv,0,0,W,H);
  drawStars();
  var sx=(Math.random()*2-1)*shake, sy=(Math.random()*2-1)*shake;
  ctx.save(); ctx.translate(sx,sy);
  ctx.drawImage(wallCv,ox-padW,oy-padW,boardW+padW*2,boardH+padW*2);
  drawDots(); drawSnake(); drawGhosts(); drawFx();
  ctx.restore();
  drawBanner();
}
