 const x0=Math.max(0,Math.floor(y-1.5)),x1=Math.min(ROWS-1,Math.ceil(y+1.5));
 const y0=Math.max(0,Math.floor(x-1.5)),y1=Math.min(COLS-1,Math.ceil(x+1.5));
 for(let yy=x0;yy<=x1;yy++)for(let xx=y0;xx<=y1;xx++){
  if(wGrid[yy][xx]<=0)continue;
  if(Math.hypot(xx-x,yy-y)>1.45)continue;
  wGrid[yy][xx]-=dmg; SFX.crack();
  burst(xx,yy,'#cfae72',6,1);
  if(wGrid[yy][xx]<=0){wGrid[yy][xx]=0;any=true;burst(xx,yy,'#8f713d',14,1.8);ring(xx,yy,'#ffd27a',.35);}
 }
 if(any){recompute();shake=Math.max(shake,6);}
}
function leak(e){
 lives-=e.lives||1; flash=1; shake=Math.max(shake,9); SFX.leak();
 addFloat(EXIT.x-0.6,EXIT.y,'-'+(e.lives||1)+' CORE','#ff6b57',true);
 burst(EXIT.x,EXIT.y,'#ff4fa3',18,2);
 if(e.type==='bomber')explodeWalls(e.x,e.y,BOMBDMG);
 if(lives<=0)endGame(false);
}

/* ============================== towers ============================== */
function towerStats(t){ const d=TDEF[t.key]; return {dmg:Math.round(d.dmg*Math.pow(1.45,t.lvl-1)),range:+(d.range*(1+.1*(t.lvl-1))).toFixed(1),rate:+(d.rate*(1+.12*(t.lvl-1))).toFixed(2)}; }
function upCost(t){ return Math.round(TDEF[t.key].cost*.8*t.lvl); }
function fireTower(t,e,def,rng){
 t.cd=1/(def.rate*(1+.12*(t.lvl-1))); t.pulse=1;
 const dmg=def.dmg*Math.pow(1.45,t.lvl-1), cx=t.x, cy=t.y;
 if(t.key==='spark'){ SFX.shoot();
  beams.push({x1:cx+Math.cos(t.ang)*.35,y1:cy+Math.sin(t.ang)*.35,x2:e.x,y2:e.y,col:def.col,w:2,life:.08,max:.08});
  burst(e.x,e.y,'#ffd27a',3,1); damageEnemy(e,dmg);
 }else if(t.key==='cryo'){ SFX.cryo();
  e.slowT=def.slowT; e.slowMul=def.slow;
  beams.push({x1:cx,y1:cy,x2:e.x,y2:e.y,col:def.col,w:1.4,life:.1,max:.1});
  ring(e.x,e.y,'#57d6ff',e.r+.15); damageEnemy(e,dmg);
 }else if(t.key==='mortar'){ SFX.mortar();
  const d=Math.hypot(e.x-cx,e.y-cy);
  projectiles.push({sx:cx,sy:cy,tx:e.x,ty:e.y,t:0,dur:clamp(d/6,.3,.75),dmg,splash:def.splash,col:def.col});
 }else{ SFX.lance(); shake=Math.max(shake,2);
  const dx=e.x-cx,dy=e.y-cy,L=Math.hypot(dx,dy)||1,ux=dx/L,uy=dy/L;
  const ex=cx+ux*rng,ey=cy+uy*rng;
  beams.push({x1:cx,y1:cy,x2:ex,y2:ey,col:def.col,w:3.5,life:.16,max:.16});
  burst(ex,ey,'#ff9dc0',8,1.4);
  for(const o of enemies){ if(o.dead)continue; if(segDist(o.x,o.y,cx,cy,ex,ey)<.3+o.r*.4)damageEnemy(o,dmg); }
 }
}

function fail(x,y){ badCell={x,y,t:.4}; SFX.error(); }
function placeAt(x,y){
 if(tool==='wall'){ if(gold<WALLCOST||!canPlace(x,y))return fail(x,y);
  gold-=WALLCOST; wGrid[y][x]=WALLHP; recompute(); burst(x,y,'#cfae72',8,1); SFX.build();
 }else{ const def=TDEF[tool]; if(!def)return;
  if(gold<def.cost||!canPlace(x,y))return fail(x,y);
  gold-=def.cost; const t={x,y,key:tool,lvl:1,cd:0,ang:-Math.PI/2,pulse:0,spent:def.cost,target:null};
  towers.push(t); tGrid[y][x]=t; recompute(); burst(x,y,def.col,8,1.2); SFX.build();
 }
}
function sellAt(x,y){
 const t=tGrid[y]&&tGrid[y][x];
 if(t){ const ref=Math.floor(t.spent*.7); gold+=ref; towers.splice(towers.indexOf(t),1); tGrid[y][x]=null;
  if(selTower===t)selTower=null;
  addFloat(x,y,'+'+ref,'#ffd27a',false); burst(x,y,'#9fc6d2',6,1); SFX.sell(); recompute(); return true; }
 if(wGrid[y][x]>0){ gold+=7; wGrid[y][x]=0; if(selWall&&selWall.x===x&&selWall.y===y)selWall=null;
  addFloat(x,y,'+7','#ffd27a',false); burst(x,y,'#cfae72',6,1); SFX.sell(); recompute(); return true; }
 return false;
}

function sim(dt){
 // spawning
 if(phase==='wave'&&qi<queue.length){
  spawnT-=dt;
  while(spawnT<=0&&qi<queue.length){
   const type=queue[qi++]; spawnEnemy(type);
   if(type==='boss'){toastTxt='⚠ WARDEN INBOUND';toastT=1.4;SFX.boss();spawnT+=Math.max(spawnGap,1.5);}
   else spawnT+=spawnGap;
  }
 }
 // towers
 for(const t of towers){
  const def=TDEF[t.key]; t.cd-=dt; if(t.pulse>0)t.pulse=Math.max(0,t.pulse-dt*4);
  const st=towerStats(t),R2=st.range*st.range; let best=null,bp=1e9;
  for(const e of enemies){ if(e.dead)continue;
   const dx=e.x-t.x,dy=e.y-t.y,d2=dx*dx+dy*dy; if(d2>R2)continue;
   const cx=Math.round(e.x),cy=Math.round(e.y);
   let p=(cx>=0&&cx<COLS&&cy>=0&&cy<ROWS)?DF[cy*COLS+cx]:1e8;
   if(p>1e8)p=9e7+d2;
   if(p<bp){bp=p;best=e;}
  }
  t.target=best;
  if(best){ const ta=Math.atan2(best.y-t.y,best.x-t.x);
   t.ang+=wrapA(ta-t.ang)*Math.min(1,dt*14);
   if(t.cd<=0)fireTower(t,best,def,st.range);
  }
 }
 // projectiles
 for(let i=projectiles.length-1;i>=0;i--){ const p=projectiles[i]; p.t+=dt/p.dur;
  if(p.t>=1){ SFX.boom(); shake=Math.max(shake,3);
   ring(p.tx,p.ty,p.col,.45); burst(p.tx,p.ty,p.col,12,2.2);
   for(const e of enemies)if(!e.dead&&Math.hypot(e.x-p.tx,e.y-p.ty)<=p.splash)damageEnemy(e,p.dmg);
   projectiles.splice(i,1);
  }else if(Math.random()<dt*40){ const px=lerp(p.sx,p.tx,p.t),py=lerp(p.sy,p.ty,p.t);
   particles.push({x:px,y:py,vx:(Math.random()-.5)*.3,vy:.2,life:.3,max:.3,size:.05,col:'#c9a86a',shape:'dot'}); }
 }
 // enemies
 for(let i=enemies.length-1;i>=0;i--){ const e=enemies[i]; if(e.dead)continue;
  if(e.slowT>0)e.slowT-=dt;
  const sp=e.spd*(e.slowT>0?e.slowMul:1)*dt;
  let dx=e.tx-e.x,dy=e.ty-e.y,d=Math.hypot(dx,dy);
  if(d<=sp||d<.0001){ e.x=e.tx;e.y=e.ty;
   if(e.tx===EXIT.x&&e.ty===EXIT.y){ e.dead=true; leak(e); continue; }
   const n=nextCell(e);
   if(!n){ e.dead=true; continue; }
   e.tx=n[0];e.ty=n[1];e.dx=n[0]-e.x;e.dy=n[1]-e.y;
  }else{ e.x+=dx/d*sp; e.y+=dy/d*sp; }
  const tx2=e.tx-e.x,ty2=e.ty-e.y;
  if(Math.hypot(tx2,ty2)>.01){ const ta=Math.atan2(ty2,tx2); e.a+=wrapA(ta-e.a)*Math.min(1,dt*10); }
 }
 enemies=enemies.filter(e=>!e.dead);
 // wave clear
 if(phase==='wave'&&qi>=queue.length&&enemies.length===0){
  phase='build'; const bonus=30+wave*6; gold+=bonus; score+=40+wave*10;
  toastTxt='WAVE CLEARED  +'+bonus+'g'; toastT=1.8; SFX.clear();
  if(wave===20&&!endless){ pendingEnd=true; endT=1.1; }
  else if(autoNext()) nextT=5;
 }
 // auto-next / pending end
 if(phase==='build'&&autoNext()&&nextT>0){ nextT-=dt; if(nextT<=0)tryStartWave(); }
 if(pendingEnd){ endT-=dt; if(endT<=0){pendingEnd=false;endGame(true);} }
 // ambience
 if(Math.random()<dt*10)particles.push({x:SPAWN.x-.3,y:SPAWN.y+(Math.random()-.5)*.8,vx:.9,vy:(Math.random()-.5)*.4,life:.5,max:.5,size:.06,col:'#ff4fa3',shape:'dot'});
 if(Math.random()<dt*6){const a=Math.random()*6.283;particles.push({x:EXIT.x+Math.cos(a)*.4,y:EXIT.y+Math.sin(a)*.4,vx:Math.cos(a)*.5,vy:Math.sin(a)*.5,life:.5,max:.5,size:.05,col:'#57d6ff',shape:'dot'});}
 if(particles.length<260&&Math.random()<dt*6)particles.push({x:Math.random()*COLS,y:ROWS+Math.random(),vx:0,vy:-.15-Math.random()*.2,life:2,max:2,size:.03,col:'#7fb8c9',shape:'dot'});
 // fx aging
 for(let i=beams.length-1;i>=0;i--){beams[i].life-=dt;if(beams[i].life<=0)beams.splice(i,1);}
 for(let i=particles.length-1;i>=0;i--){const p=particles[i];p.life-=dt;if(p.life<=0){particles.splice(i,1);continue;}
  if(p.shape==='ring')p.size+=dt*2.4; p.x+=p.vx*dt;p.y+=p.vy*dt;}
 for(let i=floaters.length-1;i>=0;i--){const f=floaters[i];f.t+=dt;f.y-=dt*.55;if(f.t>f.dur)floaters.splice(i,1);}
}
