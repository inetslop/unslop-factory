  if(k==='r'||k==='R'){ if(S.mode!=='menu')reset(); return; }
  if(k==='p'||k==='P'||k==='Escape'){
    if(S.mode==='playing'){ S.mode='paused'; pauseEl.classList.remove('hidden'); sUi(); }
    else if(S.mode==='paused'){ S.mode='playing'; pauseEl.classList.add('hidden'); sUi(); }
    return;
  }
  if(S.mode==='menu'){
    if(k==='Enter'||k===' '){ e.preventDefault(); reset(); }
    else if(k.length===1){ const c=k.toLowerCase(); if(c>='a'&&c<='z')reset(); }
    return;
  }
  if(S.mode==='over'){ if(k==='Enter'||k===' '){ e.preventDefault(); reset(); } return; }
  if(S.mode!=='playing')return;
  if(e.repeat)return;
  if(k.length===1){ const c=k.toLowerCase(); if(c>='a'&&c<='z')typeLetter(c); }
});

/* ============================== UPDATE ============================== */
function update(dt){
  S.t+=dt;
  let ts=1; if(S.slow>0){ S.slow-=dt; ts=.42; }
  const d=dt*ts;
  S.shake=Math.max(0,S.shake-dt*2.4);

  if(S.mode==='menu'){
    S.camX+=55*dt; ensureTerrain();
    P.x=S.camX+Math.min(W*.34,430);
    P.y=-52+Math.sin(S.t*1.7)*20; P.bank=Math.cos(S.t*1.7)*.2;
    spawnGlyphs(dt,.8); stepParticles(d); stepPopups(d); return;
  }
  if(S.mode==='paused')return;

  const sp=P.dead?40:baseSpeed()+S.surge;
  S.surge*=Math.exp(-1.9*dt); if(S.surge<3)S.surge=0;
  P.x+=sp*d;

  if(P.dead){ P.vy+=G*d; P.y+=P.vy*d; P.rot+=6*d; }
  else{
    if(!P.grounded){
      P.y+=P.vy*d; P.vy+=G*d;
      if(P.vy>0&&P.y>=0){
        if(overGround(P.x))land();
        else if(P.leaping){
          const np=S.segs.find(s=>s.x>P.x+30);
          if(np){ puff(np.x+26,0); P.x=np.x+26; land(); }
        }
      }
      if(!P.grounded&&P.y>190)pitFall();
    }
    if(P.leaping)P.flip+=d*(Math.PI*2/1.04);
    const bt=clamp(P.vy*.0011,-.5,.7); P.bank+=(bt-P.bank)*Math.min(1,dt*9);
  }
  P.invuln=Math.max(0,P.invuln-dt);

  const tgt=P.x-Math.min(W*.3,380);
  S.camX+=(tgt-S.camX)*Math.min(1,dt*7); if(S.camX<0)S.camX=0;
  ensureTerrain();
  spawnGlyphs(dt,.5);
  if(!P.dead&&(P.leaping||sp>430))pushP({type:'streak',x:P.x-26,y:P.y+5,vx:-sp*.4,vy:0,life:.25,max:.25});
  stepParticles(d); stepPopups(d);
}

/* ============================== RENDER ============================== */
function ridge(par,baseY,amp,col,rim){
  const pts=[];
  for(let x=-24;x<=W+24;x+=24){
    const wx=x+S.camX*par;
    pts.push([x, baseY-(Math.sin(wx*.0021)*amp+Math.sin(wx*.0053+2.7)*amp*.45+Math.sin(wx*.0009+5.1)*amp*.6)]);
  }
  ctx.beginPath(); ctx.moveTo(-24,H);
  for(const p of pts)ctx.lineTo(p[0],p[1]);
  ctx.lineTo(W+24,H); ctx.closePath(); ctx.fillStyle=col; ctx.fill();
  ctx.beginPath();
  for(let i=0;i<pts.length;i++){ const p=pts[i]; i?ctx.lineTo(p[0],p[1]):ctx.moveTo(p[0],p[1]); }
  ctx.strokeStyle=rim; ctx.lineWidth=2; ctx.stroke();
}
function drawClouds(){
  for(const c of CLOUDS){
    const B=2600, sx=((c.x-S.camX*c.par)%B+B)%B-400, y=c.y*H, s=c.s, a=c.par>1?.5:.9;
    for(const ox of [sx,sx+B]){
      if(ox<-280||ox>W+280)continue;
      ctx.globalAlpha=a*.55; ctx.fillStyle='#c478a8';
      ctx.beginPath(); ctx.ellipse(ox,y+7*s,95*s,26*s,0,0,7); ctx.fill();
      ctx.globalAlpha=a; ctx.fillStyle='#ffd9e7';
      ctx.beginPath();
      ctx.ellipse(ox,y,46*s,24*s,0,0,7);
      ctx.ellipse(ox+38*s,-10*s,38*s,20*s,0,0,7);
      ctx.ellipse(ox-40*s,-4*s,32*s,18*s,0,0,7);
      ctx.ellipse(ox+6*s,-20*s,28*s,16*s,0,0,7);
      ctx.fill();
    }
  }
  ctx.globalAlpha=1;
}
function drawWord(gy){
  const wl=S.word; if(!wl||S.mode==='over')return;
  const n=wl.text.length, cw=Math.min(38,Math.max(24,W*.028)), off=Math.min(W*.16,170);
  const bx=P.x+off, wy=gy-wordOff(), fs=Math.round(cw*1.2), totalW=(n-1)*cw;
  ctx.beginPath();
  rr(bx-totalW/2-cw*.7, wy+fs*.58, totalW+cw*1.4, 8, 4);
  ctx.fillStyle='rgba(255,209,102,.15)'; ctx.fill();
  ctx.font='bold '+fs+'px "Courier New",monospace';
  ctx.textAlign='center'; ctx.textBaseline='middle';
  for(let i=0;i<n;i++){
    if(i<wl.typed)continue;
    const x=bx-totalW/2+i*cw, cur=i===wl.typed, s=cur?1+Math.sin(S.t*7)*.08:1;
    ctx.save(); ctx.translate(x,wy); ctx.scale(s,s);
    ctx.shadowColor='rgba(255,170,60,.95)'; ctx.shadowBlur=cur?22:13;
    ctx.fillStyle=cur?'#ffd166':'#fff3e2';
    ctx.fillText(wl.text[i].toUpperCase(),0,0);
    ctx.restore();
  }
  ctx.shadowBlur=0;
}
function rr(x,y,w,h,r){ ctx.beginPath(); ctx.moveTo(x+r,y); ctx.arcTo(x+w,y,x+w,y+h,r); ctx.arcTo(x+w,y+h,x,y+h,r); ctx.arcTo(x,y+h,x,y,r); ctx.arcTo(x,y,x+w,y,r); ctx.closePath(); }
function drawPlane(gy,cam){
  if(P.invuln>0&&Math.floor(S.t*14)%2===0&&!P.dead)return;
  const x=P.x-cam, y=gy+P.y;
  // shadow
  const sa=clamp(1-(-P.y)/280,0,.35);
  if(sa>0.02){ ctx.globalAlpha=sa; ctx.fillStyle='#2a1547';
    ctx.beginPath(); ctx.ellipse(x,gy+10,26,6,0,0,7); ctx.fill(); ctx.globalAlpha=1; }
  ctx.save(); ctx.translate(x,y);
  ctx.rotate(P.bank+(P.leaping?P.flip:0)+(P.dead?P.rot:0));
  ctx.scale(1.55,1.55);
  ctx.beginPath();
  ctx.moveTo(26,0); ctx.lineTo(-22,-13); ctx.lineTo(-9,1); ctx.lineTo(-18,10); ctx.closePath();
  ctx.fillStyle='#fffdf7'; ctx.fill();
  ctx.lineWidth=2.4; ctx.strokeStyle='#2a1547'; ctx.lineJoin='round'; ctx.stroke();
  ctx.beginPath(); ctx.moveTo(26,0); ctx.lineTo(-9,1); ctx.lineWidth=2; ctx.strokeStyle='rgba(42,21,71,.35)'; ctx.stroke();
  ctx.beginPath(); ctx.moveTo(-22,-13); ctx.lineTo(-9,1); ctx.lineTo(-16,1.5); ctx.closePath();
  ctx.fillStyle='#ff5e6c'; ctx.fill();
  ctx.beginPath(); ctx.arc(8,-4,2.3,0,7); ctx.fillStyle='#2a1547'; ctx.fill();
  ctx.restore();
}
function render(){
  const gy=GY(), shx=(Math.random()-.5)*S.shake*16, shy=(Math.random()-.5)*S.shake*12, cam=S.camX+shx;
  ctx.fillStyle=skyGrad; ctx.fillRect(0,0,W,H);
  // stars
  for(const s of stars){
    if(s.y>gy*.7)continue;
    const sx=((s.x-cam*.03)%W+W)%W, a=.25+.45*(.5+.5*Math.sin(S.t*s.s+s.p));
    ctx.globalAlpha=a; ctx.fillStyle='#ffe9c9'; ctx.fillRect(sx,s.y,s.r,s.r);
  }
  ctx.globalAlpha=1;
  // sun
  const sxp=W*.68, syp=gy-46;
  let g=ctx.createRadialGradient(sxp,syp,8,sxp,syp,H*.32);
  g.addColorStop(0,'rgba(255,214,140,.75)'); g.addColorStop(.4,'rgba(255,150,90,.26)'); g.addColorStop(1,'rgba(255,150,90,0)');
  ctx.fillStyle=g; ctx.fillRect(sxp-H*.32,syp-H*.32,H*.64,H*.64);
  ctx.beginPath(); ctx.arc(sxp,syp,30,0,7); ctx.fillStyle='#ffe4a8'; ctx.fill();
  // ridges + clouds
  ridge(cam*.12, gy-70, 58, '#4a2a6e', 'rgba(255,150,120,.3)');
  ridge(cam*.25, gy-24, 40, '#2b1549', 'rgba(255,170,130,.45)');
  drawClouds();

  ctx.save(); ctx.translate(0,shy);
  // abyss gaps + mist
  for(let i=0;i<S.segs.length-1;i++){
    const a=S.segs[i], b=S.segs[i+1];
    const x0=a.x+a.w-cam, x1=b.x-cam, gw=x1-x0;
    if(gw<=0||x1<0||x0>W)continue;
    const gg=ctx.createLinearGradient(0,gy,0,H);
    gg.addColorStop(0,'rgba(70,25,80,.95)'); gg.addColorStop(1,'#0c0418');
    ctx.fillStyle=gg; ctx.fillRect(x0,gy,gw,H-gy+60);

    ctx.globalAlpha=1;

  // cloud platforms
  for(const s of S.segs){
    const x=s.x-cam; if(x>W||x+s.w<0)continue;
    const pg=ctx.createLinearGradient(0,gy,0,H);
    pg.addColorStop(0,'#fff3f4'); pg.addColorStop(.25,'#ffd9e2'); pg.addColorStop(1,'#8a4a7d');
    ctx.fillStyle=pg; ctx.fillRect(x,gy,s.w,H-gy+60);
    ctx.fillStyle='#fff8fa';
    for(let cx=x+16;cx<x+s.w-4;cx+=40){
      const r=18+((cx*13)%11);
      ctx.beginPath(); ctx.arc(cx,gy+4,r,0,7); ctx.fill();
    }
    ctx.fillStyle='rgba(90,35,80,.16)';
    ctx.fillRect(x,gy,10,H-gy); ctx.fillRect(x+s.w-10,gy,10,H-gy);
  }
  // word
  drawWord(gy);
  // particles
  for(const q of S.particles){
    const x=q.x-cam, y=gy+q.y, a=Math.max(0,q.life/q.max);
    if(q.type==='spark'){ ctx.globalAlpha=a; ctx.fillStyle=q.col; ctx.beginPath(); ctx.arc(x,y,q.size*a+.6,0,7); ctx.fill(); }
    else if(q.type==='puff'){ ctx.globalAlpha=a*.5; ctx.fillStyle='#ffe9f2'; ctx.beginPath(); ctx.arc(x,y,q.r*(1.7-a*.7),0,7); ctx.fill(); }
    else if(q.type==='glyph'){ ctx.globalAlpha=a*.5; ctx.fillStyle='#ffd9a8'; ctx.font='bold 20px "Courier New",monospace'; ctx.textAlign='center'; ctx.fillText(q.ch,x,y); }
    else if(q.type==='streak'){ ctx.globalAlpha=a*.65; ctx.strokeStyle='#fff'; ctx.lineWidth=2; ctx.beginPath(); ctx.moveTo(x,y); ctx.lineTo(x+18,y); ctx.stroke(); }
  }
  ctx.globalAlpha=1;
  // plane
  drawPlane(gy,cam);
  // popups
  for(const p of S.popups){

    ctx.fillStyle=p.col; ctx.fillText(p.txt,x,y);
  }
  ctx.globalAlpha=1;
  ctx.restore();
