    const dx=ball.x-b.x,dy=ball.y-b.y,d2=dx*dx+dy*dy,R=b.r+BR;
    if(d2<R*R){
      let d=Math.sqrt(d2);if(d<.001)d=.001;
      const nx=dx/d,ny=dy/d;
      ball.x=b.x+nx*R;ball.y=b.y+ny*R;
      const vn=ball.vx*nx+ball.vy*ny;
      if(vn<0){const k=1.3*vn;ball.vx-=k*nx;ball.vy-=k*ny;}
      let m=Math.hypot(ball.vx,ball.vy);const sp=Math.max(m,800);
      if(m>.001){ball.vx*=sp/m;ball.vy*=sp/m;}else{ball.vx=nx*sp;ball.vy=ny*sp;}
      if(b.cool<=0){
        b.cool=.12;b.flash=1;score+=100;addFloat('+100',b.x,b.y-b.r-8,'#7fe9ff');
        tone(700,1400,.1,'square',.35);tone(1400,2200,.08,'square',.2,.02);noise(.05,.2,2500);
        burst(ball.x,ball.y,'#7fe9ff',8,300);shake=Math.max(shake,2.5);
      }
    }
  }
  /* slingshots */
  for(const s of SLINGS){
    if(s.cool>0)continue;
    let hit=false;
    const pts=s.pts;
    for(let e=0;e<3&&!hit;e++){
      const x1=pts[e][0],y1=pts[e][1],x2=pts[(e+1)%3][0],y2=pts[(e+1)%3][1];
      const p=closestOnSeg(ball.x,ball.y,x1,y1,x2,y2);
      let dx=ball.x-p.x,dy=ball.y-p.y,d2=dx*dx+dy*dy;
      if(d2<BR*BR){
        let d=Math.sqrt(d2);if(d<.001){d=.001;dx=0;dy=-1;}
        const nx=dx/d,ny=dy/d;
        ball.x=p.x+nx*BR;ball.y=p.y+ny*BR;
        const vn=ball.vx*nx+ball.vy*ny;
        if(vn<0){ball.vx-=1.45*vn*nx;ball.vy-=1.45*vn*ny;}
        let kx=ball.vx*.35+s.kick[0]*.65,ky=ball.vy*.35+s.kick[1]*.65;
        const km=Math.hypot(kx,ky)||1;const sp=Math.max(780,Math.hypot(ball.vx,ball.vy));
        ball.vx=kx/km*sp;ball.vy=ky/km*sp;
        s.cool=.1;s.flash=1;hit=true;
      }
    }
    if(hit){score+=50;addFloat('+50',s.cx,s.cy-24,'#ffb0a0');tone(160,90,.09,'triangle',.4);noise(.05,.25,500);burst(ball.x,ball.y,'#ff8a7a',6,220);shake=Math.max(shake,2);}
  }
  /* aliens */
  for(const a of aliens){
    if(a.alive===false)continue;
    const ax=a.bx+form.x,ay=a.by+form.y+Math.sin(ambientT*3+a.seed)*2;
    const dx=ball.x-ax,dy=ball.y-ay,d2=dx*dx+dy*dy,R=17+BR;
    if(d2<R*R){
      let d=Math.sqrt(d2);if(d<.001){d=.001;}
      const nx=dx/d,ny=dy/d;
      ball.x=ax+nx*R;ball.y=ay+ny*R;
      const vn=ball.vx*nx+ball.vy*ny;
      if(vn<0){ball.vx-=1.5*vn*nx;ball.vy-=1.5*vn*ny;}
      let m=Math.hypot(ball.vx,ball.vy);const sp=Math.max(m,460);
      if(m>.001){ball.vx*=sp/m;ball.vy*=sp/m;}else{ball.vx=nx*sp;ball.vy=ny*sp;}
      killAlien(a,false);
    }
  }
  /* ufo */
  if(ufo.on&&state==='play'){
    const dx=ball.x-ufo.x,dy=ball.y-ufo.y;
    if(dx*dx+dy*dy<28*28){
      ufo.on=false;ufo.t=18+Math.random()*14;score+=500;
      addFloat('+500',ufo.x,ufo.y-20,'#7fe9ff');
      burst(ufo.x,ufo.y,'#7fe9ff',22,340);burst(ufo.x,ufo.y,'#ffffff',10,200);
      tone(784,784,.09,'square',.25);tone(988,988,.09,'square',.25,.09);tone(1319,1319,.18,'square',.25,.18);
      shake=Math.max(shake,5);
      if(Math.random()<.3&&ballsLeft<5){ballsLeft++;addFloat('EXTRA BALL!',250,470,'#ff4fd8',2);tone(600,1200,.3,'square',.25);}
    }
  }
  /* clamp speed */
  const sp2=Math.hypot(ball.vx,ball.vy);
  if(sp2>2100){ball.vx*=2100/sp2;ball.vy*=2100/sp2;}
  /* drain */
  if(ball.y>905||ball.x<-30||ball.x>570){loseBall('BALL LOST');return;}
}

/* ============================== UPDATE ============================== */
function updateFX(dt){
  shake*=Math.exp(-6.5*dt);if(shake<.15)shake=0;
  redFlash=Math.max(0,redFlash-2.2*dt);
  for(const b of BUMPERS){b.flash=Math.max(0,b.flash-4*dt);b.cool-=dt;}
  for(const s of SLINGS){s.flash=Math.max(0,s.flash-5*dt);s.cool-=dt;}
  for(let i=parts.length-1;i>=0;i--){const p=parts[i];p.t+=dt;if(p.t>p.life){parts.splice(i,1);continue;}p.x+=p.vx*dt;p.y+=p.vy*dt;p.vy+=500*dt;}
  for(let i=floats.length-1;i>=0;i--){const f=floats[i];f.t+=dt;f.y-=46*dt;if(f.t>f.life)floats.splice(i,1);}
  if(chargeOsc)chargeOsc.frequency.value=90+charge*560;
}
function updatePlay(dt){
  if(state!=='play')return;
  if(!ball.live){if(charging)charge=Math.min(1,charge+dt/0.85);}
  else{
    march(dt);
    trail.push({x:ball.x,y:ball.y});if(trail.length>14)trail.shift();
    for(const a of aliens){
      if(a.alive!==false){
        const ay=a.by+form.y+Math.sin(ambientT*3+a.seed)*2;
        if(ay>596){killAlien(a,true);break;}
      }
    }
    if(state!=='play')return;
    const sp=Math.hypot(ball.vx,ball.vy);
    if(sp<30&&ball.y<760){stallT+=dt;if(stallT>1.1){ball.vy-=230;ball.vx+=(Math.random()-.5)*260;stallT=0;}}else stallT=0;
    if(ball.x>478&&ball.y>800&&sp<40){laneRestT+=dt;if(laneRestT>1)resetBall();}else laneRestT=0;
  }
  ufoUpdate(dt);
  if(pendingNext!=null&&aliveCount()===0){
    pendingNext-=dt;
    if(pendingNext<=0){wave++;spawnWave(wave);pendingNext=null;setBanner('WAVE '+wave,'GET READY',2.2);tone(660,880,.15,'square',.2);}
  }
}

let last=performance.now(),acc=0;
function frame(now){
  requestAnimationFrame(frame);
  let dt=(now-last)/1000;last=now;if(dt>0.05)dt=0.05;
  ambientT+=dt;
  if(state!=='paused'){
    updateFX(dt);
    if(state==='play'||state==='ballend'){
      acc+=dt;let n=0;
      while(acc>=HSTEP&&n<10){stepBall(HSTEP);acc-=HSTEP;n++;}
    }else acc=0;
    if(state==='play')updatePlay(dt);
    if(state==='ballend'){endT-=dt;if(endT<=0){resetBall();state='play';}}
  }
  render();
}
