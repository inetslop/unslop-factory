function collideSeg(b,x1,y1,x2,y2,t,rest){
  const dx=x2-x1,dy=y2-y1,L2=dx*dx+dy*dy;
  let u=((b.x-x1)*dx+(b.y-y1)*dy)/L2;u=clamp(u,0,1);
  const px=x1+dx*u,py=y1+dy*u;let ex=b.x-px,ey=b.y-py;const d=Math.hypot(ex,ey),minD=BALL_R+t/2;
  if(d>=minD)return 0;if(d<.001){ex=1;ey=0;}
  const nx=ex/d,ny=ey/d;b.x=px+nx*minD;b.y=py+ny*minD;
  const vn=b.vx*nx+b.vy*ny;if(vn<0){b.vx-=(1+rest)*vn*nx;b.vy-=(1+rest)*vn*ny;return -vn}return 0;
}
function collideCirc(b,cx,cy,cr,rest){
  let ex=b.x-cx,ey=b.y-cy;const d=Math.hypot(ex,ey),minD=BALL_R+cr;if(d>=minD)return 0;

}

      const d2=Math.hypot(b.vx,b.vy),dec=DE*sdt;
      if(d2<=dec){b.vx=b.vy=0}else{const ns=(d2-dec)/d2;b.vx*=ns;b.vy*=ns}}
    let dx=cc[0]-b.x,dy=cc[1]-b.y;const d=Math.hypot(dx,dy);
    if(d<CUP_R*2.6&&d>.001){const a=750*(1-d/(CUP_R*2.6));b.vx+=dx/d*a*sdt;b.vy+=dy/d*a*sdt}
    b.x+=b.vx*sdt;b.y+=b.vy*sdt;
    let impW=0,impT=0;
    for(const w of course.walls)impW=Math.max(impW,collideSeg(b,w[0],w[1],w[2],w[3],w[4],.72));
    for(const t of course.trees)impT=Math.max(impT,collideCirc(b,t[0],t[1],t[2],.6));
    const mn=RIM+BALL_R,mx=W-RIM-BALL_R,my=H-RIM-BALL_R;
    if(b.x<mn){b.x=mn;if(b.vx<0){impW=Math.max(impW,-b.vx);b.vx*=-.72}}
    if(b.x>mx){b.x=mx;if(b.vx>0){impW=Math.max(impW,b.vx);b.vx*=-.72}}
    if(b.y<mn){b.y=mn;if(b.vy<0){impW=Math.max(impW,-b.vy);b.vy*=-.72}}
    if(b.y>my){b.y=my;if(b.vy>0){impW=Math.max(impW,b.vy);b.vy*=-.72}}
    if(impW>70&&time-lastWallSfx>.1){sfx.wall(impW);lastWallSfx=time}
    if(impT>70&&time-lastTreeSfx>.1){sfx.tree(impT);lastTreeSfx=time}
    const imp=Math.max(impW,impT);if(imp>240)shakeMag=Math.max(shakeMag,Math.min(9,(imp-200)/150));
    if(waterCool<=0&&terrainAt(b.x,b.y)==='water'){
      if(Math.hypot(b.vx,b.vy)>25){doSplash();return}
      b.x=lastSafe.x;b.y=lastSafe.y;b.vx=b.vy=0;phase='aim';return;}
    const d2=Math.hypot(cc[0]-b.x,cc[1]-b.y);
    if(d2<CUP_R-1&&Math.hypot(b.vx,b.vy)<340){startSink();return}
  }
  if(terrainAt(b.x,b.y)!=='water'){lastSafe.x=b.x;lastSafe.y=b.y}
}

function update(dt){
  time+=dt;
  if(waterCool>0)waterCool-=dt;
  shakeMag=Math.max(0,shakeMag-dt*26);
  flagDrop+=clamp(flagTarget-flagDrop,-dt*1.3,dt*1.3);
  let spd=0;
  if(state==='play'){
    if(phase==='aim'&&!mouseAim){
      let r=(keys.right?1:0)-(keys.left?1:0);
      aim+=r*2.4*dt;
      if(keys.up)aim-=dt*.55;if(keys.down)aim+=dt*.55;
    }
    if(charging){
      power+=chargeDir*dt/.62;
      if(power>=1){power=1;chargeDir=-1}
      if(power<=0){power=0;chargeDir=1}
      updCharge();
    }
    if(phase==='roll'){
      stepBall(dt);
      spd=Math.hypot(ball.vx,ball.vy);
      if(spd>60){trail.push({x:ball.x,y:ball.y,t:0});if(trail.length>26)trail.shift();}
    }
    if(phase==='sink'){sinkT+=dt/.55;if(sinkT>=1){sinkT=1;holeComplete()}}
  }
  setRoll(state==='play'&&phase==='roll'?spd:0);
  for(let i=parts.length-1;i>=0;i--){const p=parts[i];p.t+=dt;
    if(p.ty==='c'){p.vy+=420*dt;p.x+=(p.vx+Math.sin(p.t*13)*36)*dt;p.y+=p.vy*dt;p.rot+=p.vr*dt;
      if(p.t>p.life||p.y>H+30)parts.splice(i,1);}
    else if(p.ty==='d'){p.x+=p.vx*dt;p.y+=p.vy*dt;if(p.t>p.life)parts.splice(i,1);}
    else if(p.ty==='p'){p.vy+=520*dt;p.x+=p.vx*dt;p.y+=p.vy*dt;if(p.t>p.life)parts.splice(i,1);}
    else if(p.ty==='r'&&p.t>p.life)parts.splice(i,1);
  }
  for(let i=trail.length-1;i>=0;i--){trail[i].t+=dt*1.4;if(trail[i].t>.5)trail.splice(i,1)}
}

  if(state!=='play'||phase!=='aim')return;
  const active=charging||mouseAim;
  ctx.save();
  if(!active){ // idle stub
    ctx.strokeStyle='rgba(244,234,210,.5)';ctx.lineWidth=2;ctx.setLineDash([4,7]);
    ctx.beginPath();
    ctx.moveTo(ball.x+Math.cos(aim)*14,ball.y+Math.sin(aim)*14);
    ctx.lineTo(ball.x+Math.cos(aim)*38,ball.y+Math.sin(aim)*38);
    ctx.stroke();ctx.setLineDash([]);
  }else if(power>.02){
    const n=Math.floor(4+power*15),sp=15;
    for(let i=0;i<n;i++){
      const d=20+i*sp,a=(1-i/n)*.9;
      ctx.fillStyle=`rgba(244,234,210,${a})`;
      ctx.beginPath();ctx.arc(ball.x+Math.cos(aim)*d,ball.y+Math.sin(aim)*d,3.5-i/n*1.8,0,TAU);ctx.fill();
    }
    const ex=ball.x+Math.cos(aim)*(20+n*sp),ey=ball.y+Math.sin(aim)*(20+n*sp);
    ctx.fillStyle='#e8c15c';ctx.save();ctx.translate(ex,ey);ctx.rotate(aim);
    ctx.beginPath();ctx.moveTo(9,0);ctx.lineTo(-3,-6);ctx.lineTo(-3,6);ctx.closePath();ctx.fill();ctx.restore();
    // power ring
    ctx.lineWidth=5;ctx.strokeStyle='rgba(0,0,0,.3)';
    ctx.beginPath();ctx.arc(ball.x,ball.y,17,0,TAU);ctx.stroke();
    ctx.strokeStyle=pcol(power);
    ctx.beginPath();ctx.arc(ball.x,ball.y,17,-Math.PI/2,-Math.PI/2+power*TAU);ctx.stroke();
    ctx.font='700 14px "Segoe UI",system-ui,sans-serif';ctx.textAlign='center';
    ctx.fillStyle='rgba(0,0,0,.5)';ctx.fillText(Math.round(power*100)+'%',ball.x+1,ball.y-31);
    ctx.fillStyle='#f4ead2';ctx.fillText(Math.round(power*100)+'%',ball.x,ball.y-32);
  }
  ctx.restore();
}
function drawParts(){
  for(const p of parts){
    const a=clamp(1-p.t/p.life,0,1);
    if(p.ty==='c'){ctx.save();ctx.globalAlpha=a;ctx.translate(p.x,p.y);ctx.rotate(p.rot);
      ctx.fillStyle=p.c;ctx.fillRect(-p.w/2,-p.h/2,p.w,p.h);ctx.restore();}
    else if(p.ty==='d'){ctx.globalAlpha=a*.5;ctx.fillStyle=p.c;
      ctx.beginPath();ctx.arc(p.x,p.y,p.r*(1+p.t*2),0,TAU);ctx.fill();ctx.globalAlpha=1;}
    else if(p.ty==='p'){ctx.globalAlpha=a;ctx.fillStyle=p.c;
      ctx.beginPath();ctx.arc(p.x,p.y,p.r,0,TAU);ctx.fill();ctx.globalAlpha=1;}
    else if(p.ty==='r'){ctx.globalAlpha=a*.6;ctx.strokeStyle=p.c;ctx.lineWidth=3;
      ctx.beginPath();ctx.arc(p.x,p.y,p.r0+p.t*46,0,TAU);ctx.stroke();ctx.globalAlpha=1;}
  }
}
function render(){
  ctx.setTransform(dpr,0,0,dpr,0,0);
  ctx.fillStyle=bgGrad;ctx.fillRect(0,0,cw,chh);
  for(const s of stars){
    const a=.2+.4*(.5+.5*Math.sin(time*s.sp+s.p));
    ctx.fillStyle=`rgba(212,232,255,${a})`;
    ctx.beginPath();ctx.arc(s.x,s.y,s.r,0,TAU);ctx.fill();
  }
  const glow=ctx.createRadialGradient(cw/2,chh/2,10,cw/2,chh/2,Math.max(cw,chh)*.6);
  glow.addColorStop(0,'rgba(90,170,120,.1)');glow.addColorStop(1,'rgba(90,170,120,0)');
  ctx.fillStyle=glow;ctx.fillRect(0,0,cw,chh);
  let s=Math.min((cw-24)/W,(chh-24)/H),ox=(cw-W*s)/2,oy=(chh-H*s)/2;
  if(state==='menu'){s*=1.05;ox+=Math.sin(time*.16)*12;oy+=Math.cos(time*.12)*9;}
  const shx=(Math.random()*2-1)*shakeMag,shy=(Math.random()*2-1)*shakeMag;
  ctx.save();ctx.translate(ox+shx,oy+shy);ctx.scale(s,s);
  drawCourse();
  // trail
  for(const t of trail){const a=(1-t.t/.5)*.35;
    ctx.fillStyle=`rgba(255,255,255,${a})`;
    ctx.beginPath();ctx.arc(t.x,t.y,3*(1-t.t/.5)+1,0,TAU);ctx.fill();}
  drawBall();drawFlag();drawAim();drawParts();
  ctx.restore();
  ctx.fillStyle=vigGrad;ctx.fillRect(0,0,cw,chh);
}
