/* ============================== PATH / ROUTE ============================== */
function computeRoute(){
  const S=level.start,F=level.finish;if(!S||!F)return null;
  const key=c=>c.c+level.w*c.r;
  const prev=new Map([[key(S),null]]);const q=[S];let head=0;
  while(head<q.length){
    const c=q[head++];if(c===F)break;
    for(let d=0;d<4;d++){
      if(!connHas(c,d))continue;
      const n=level.cellAt(c.c+DIRS[d].x,c.r+DIRS[d].y);
      if(!n||n.type<1||!connHas(n,(d+2)%4)||prev.has(key(n)))continue;
      prev.set(key(n),c);q.push(n);
    }
  }
  if(!prev.has(key(F)))return null;
  const path=[];let cur=F;while(cur){path.push(cur);cur=prev.get(key(cur));}
  return path.reverse();
}
function routeValid(p){
  if(!p||p.length<2||!level.start)return false;
  if(p[0]!==level.start||p[p.length-1]!==level.finish)return false;
  for(let i=0;i<p.length-1;i++){
    const a=p[i],b=p[i+1],dx=b.c-a.c,dy=b.r-a.r;
    if(Math.abs(dx)+Math.abs(dy)!==1)return false;
    const d=dx===1?1:dx===-1?3:dy===1?2:0;
    if(!connHas(a,d)||!connHas(b,(d+2)%4))return false;
  }return true;
}

function buildRace(cells,opt){
  const loop=!!opt&&opt.loop;
  const n=cells.length;
  const pts=cells.map(c=>({x:c.c+0.5,y:c.r+0.5}));
  let totalPts=pts.slice();if(loop)totalPts.push(pts[0]);
  const segs=[];const cum=[0];
  for(let i=0;i<totalPts.length-1;i++){
    const a=totalPts[i],b=totalPts[i+1];
    const len=Math.hypot(b.x-a.x,b.y-a.y);
    segs.push({ax:a.x,ay:a.y,bx:b.x,by:b.y,len});cum.push(cum[i]+len);
  }
  for(let i=1;i<pts.length-1;i++){
    const d1x=pts[i].x-pts[i-1].x,d1y=pts[i].y-pts[i-1].y;
    const d2x=pts[i+1].x-pts[i].x,d2y=pts[i+1].y-pts[i].y;
    if(d1x*d2y-d1y*d2x!==0){if(segs[i-1])segs[i-1].slow=true;if(segs[i])segs[i].slow=true;}
  }
  const R={cells,segs,cum,total:cum[cum.length-1],dist:0,speed:0,segI:0,boostT:0,loop,done:false,
    heading:Math.atan2(segs[0].by-segs[0].ay,segs[0].bx-segs[0].ax),
    coins:cells.filter(c=>c.mark==='c').map(c=>({x:c.c+0.5,y:c.r+0.5,got:false})),
    pads:cells.filter(c=>c.mark==='b').map(c=>({x:c.c+0.5,y:c.r+0.5,cool:false}))};
  return R;
}

function pointAt(d){

}

function updateRace(dt){
  const prevDist=race.dist;
  const curSeg=race.segs[race.segI];
  let target=race.boostT>0?6.4:(curSeg&&curSeg.slow?2.15:3.7);
  if(!race.loop)target=Math.min(target,Math.max(0,(race.total-race.dist)*3+2));
  race.speed+=clamp(target-race.speed,-7*dt,4.2*dt);
  race.dist+=race.speed*dt;
  if(race.loop){
    if(race.dist>=race.total){race.dist-=race.total;
      race.coins.forEach(c=>c.got=false);}
  }else if(race.dist>=race.total-0.3&&!race.done){doFinish();return;}
  race.segI=segAt(race.dist,race.segI);
  const pos=carPos(),aim=pointAt(race.dist+0.5);
  const ta=Math.atan2(aim.y-pos.y,aim.x-pos.x);
  race.heading+=angDiff(race.heading,ta)*Math.min(1,dt*9);
  race.boostT-=dt;
  // pads
  for(const p of race.pads){const d=Math.hypot(pos.x-p.x,pos.y-p.y);
    if(d<0.5&&!p.cool){p.cool=true;race.boostT=0.9;shake=Math.max(shake,4);sfx.boost();
      spawnBurst(p.x,p.y,'#3fd9ff',12,3);}
    else if(d>1.3)p.cool=false;}
  // coins
  for(const c of race.coins){if(c.got)continue;
    if(Math.hypot(pos.x-c.x,pos.y-c.y)<0.48){c.got=true;sfx.coin();
      floaters.push({x:c.x,y:c.y-0.15,txt:'+100',col:'#ffd23f',life:1});

}

function update(dt){
  // motes always
  for(const m of motes){m.x+=m.vx*dt;m.y+=m.vy*dt;
    if(m.x<-10)m.x=W+10;if(m.x>W+10)m.x=-10;if(m.y<-10)m.y=H+10;if(m.y>H+10)m.y=-10;}
  // particles
  for(let i=particles.length-1;i>=0;i--){const p=particles[i];p.life-=dt;
    if(p.life<=0){particles.splice(i,1);continue;}
    if(p.kind==='dot'||p.kind==='conf'){p.x+=p.vx*dt;p.y+=p.vy*dt;
      if(p.kind==='conf'){p.vy+=3.4*dt;p.rot+=p.vr*dt;p.vx*=1-0.6*dt;}}
    else if(p.kind==='smoke'){p.x+=p.vx*dt;p.y+=p.vy*dt;p.sz+=dt*.25;}
    else if(p.kind==='ring'){p.r+=p.vr*dt;}}
  for(let i=floaters.length-1;i>=0;i--){const f=floaters[i];f.life-=dt*1.1;f.y-=dt*0.9;if(f.life<=0)floaters.splice(i,1);}
  flashT=Math.max(0,flashT-dt*1.6);
  shake=Math.max(0,shake-dt*9);
  shakeX=(Math.random()-.5)*shake;shakeY=(Math.random()-.5)*shake;
  const k=1-Math.exp(-6*dt);
  cam.x+=(camTarget.x-cam.x)*k;cam.y+=(camTarget.y-cam.y)*k;
  zoom+=(zoomT-zoom)*(1-Math.exp(-4.5*dt));

  if(paused)return;
  if(state==='menu'||state==='race')updateRace(dt);
  else if(state==='build'){
    // spinners
    level.cells.forEach(row=>row.forEach(c=>{
      if(c.spin){const before=Math.floor(c.spin.phase/(Math.PI/2))%4;
        c.spin.phase+=c.spin.speed*dt;if(c.spin.phase<0)c.spin.phase+=TAU*1e6;
        const after=((Math.floor(c.spin.phase/(Math.PI/2))%4)+4)%4;
        if(after!==before){c.pulse=Math.max(c.pulse,.7);dirty=true;}}
      c.pulse=Math.max(0,c.pulse-dt*4);}));
    timeLeft-=dt;
    if(timeLeft<=0){timeLeft=0;timeout();return;}
    if(timeLeft<6){const w=Math.ceil(timeLeft);if(w!==lastWholeSec){lastWholeSec=w;sfx.tick();}}
    recomputeT-=dt;
    if(dirty||recomputeT<=0){maybeRecompute();dirty=false;recomputeT=0.12;}
  }
}

function maybeRecompute(){
  if(route&&routeValid(route))return; // sticky
  const was=routeReady;
  const cand=computeRoute();
  if(cand){route=cand;routeReady=true;if(!was){sfx.ready();flashT=1;}}
  else{route=null;routeReady=false;if(was)sfx.off();}
}

  for(const m of motes){ctx.globalAlpha=m.a;ctx.fillStyle=m.col;
    ctx.beginPath();ctx.arc(m.x,m.y,m.r,0,TAU);ctx.fill();}
  ctx.globalAlpha=1;

  if(!level)return;
  unit=computeUnit();
  // world transform (cell units)
  ctx.save();
  ctx.translate(W/2+shakeX,H/2+14+shakeY);
  ctx.scale(unit*zoom,unit*zoom);
  ctx.translate(-cam.x,-cam.y);

  drawFrame();
  for(let r=0;r<level.h;r++)for(let c=0;c<level.w;c++)drawCell(level.cells[r][c]);
  // route glow
  if((routeReady&&state==='build')||state==='race'||state==='menu'){
    const pts=(state==='race'||state==='menu')?race.cells:route;
    if(pts){ctx.lineCap='round';ctx.lineJoin='round';
      ctx.beginPath();ctx.moveTo(pts[0].c+.5,pts[0].r+.5);
      for(let i=1;i<pts.length;i++)ctx.lineTo(pts[i].c+.5,pts[i].r+.5);
      if(race&&race.loop)ctx.closePath();
      const dim=state==='build'?1:0.4;
      ctx.strokeStyle='rgba(61,255,168,'+(0.13*dim)+')';ctx.lineWidth=.52;ctx.stroke();
      ctx.setLineDash([.45,.32]);ctx.lineDashOffset=-performance.now()/1000*2.2;
      ctx.strokeStyle='rgba(61,255,168,'+(0.75*dim*(0.7+flashT*.6))+')';ctx.lineWidth=.16;ctx.stroke();
      ctx.setLineDash([]);}
  }
  // cursor / hover (build only)
  if(state==='build'&&!paused){
    if(hoverCell&&hoverCell.type>=1)drawHover(hoverCell);
    drawCursor();
  }
  // car
  if(race&&(state==='race'||state==='menu'))drawCar();
  // particles (world)
  for(const p of particles){const a=clamp(p.life/p.max,0,1);
    if(p.kind==='dot'){ctx.globalAlpha=a;ctx.fillStyle=p.col;
      ctx.beginPath();ctx.arc(p.x,p.y,p.sz,0,TAU);ctx.fill();}
    else if(p.kind==='smoke'){ctx.globalAlpha=a*.28;ctx.fillStyle='#cfd8dc';
      ctx.beginPath();ctx.arc(p.x,p.y,p.sz,0,TAU);ctx.fill();}
    else if(p.kind==='ring'){ctx.globalAlpha=a*.7;ctx.strokeStyle=p.col;ctx.lineWidth=.05;
      ctx.beginPath();ctx.arc(p.x,p.y,p.r,0,TAU);ctx.stroke();}
    else if(p.kind==='conf'){ctx.save();ctx.translate(p.x,p.y);ctx.rotate(p.rot);
      ctx.globalAlpha=Math.min(1,a*2);ctx.fillStyle=p.col;
      ctx.fillRect(-p.sz/2,-p.sz*.8,p.sz,p.sz*1.6);ctx.restore();}}
  ctx.globalAlpha=1;
  // floaters
  for(const f of floaters){ctx.globalAlpha=clamp(f.life,0,1);
    ctx.font='900 0.45px Impact,"Arial Black",sans-serif';ctx.textAlign='center';
    ctx.lineWidth=.06;ctx.strokeStyle='#0a1a20';ctx.strokeText(f.txt,f.x,f.y);
    ctx.fillStyle=f.col;ctx.fillText(f.txt,f.x,f.y);}
  ctx.globalAlpha=1;
  ctx.restore();

  // speed lines (screen space)
  if(race&&state==='race'&&!paused&&(race.speed>3.8||race.boostT>0)){
    const inten=clamp((race.speed-3.6)/3,0,1)+(race.boostT>0?.5:0);
    ctx.strokeStyle='rgba(190,240,255,'+(0.16*inten)+')';ctx.lineWidth=2;
    for(let i=0;i<16;i++){const a=rand(0,TAU),r0=rand(H*.3,H*.62),len=rand(20,90)*inten;
      const x=Math.cos(a),y=Math.sin(a);
      ctx.beginPath();ctx.moveTo(W/2+x*r0,H/2+y*r0);ctx.lineTo(W/2+x*(r0+len),H/2+y*(r0+len));ctx.stroke();}
  }
}

function drawFrame(){
  const w=level.w,h=level.h;
  rr(ctx,-.5,-.5,w+1,h+1,.35);
  ctx.fillStyle='#082935';ctx.fill();
  ctx.strokeStyle='rgba(61,255,168,.1)';ctx.lineWidth=.22;ctx.stroke();
  ctx.strokeStyle='rgba(61,255,168,.3)';ctx.lineWidth=.05;ctx.stroke();

  for(let r=0;r<h;r++)for(let c=0;c<w;c++){
    if(level.cells[r][c].type===0){ctx.beginPath();ctx.arc(c+.5,r+.5,.035,0,TAU);ctx.fill();}}
  // corner ticks
  ctx.strokeStyle='rgba(61,255,168,.5)';ctx.lineWidth=.09;ctx.lineCap='round';
  const T=[[-.4,-.4,1,1],[w+.4-.0,-.4,-1,1],[-.4,h+.4,1,-1],[w+.4,h+.4,-1,-1]];
  for(const t of T){ctx.beginPath();ctx.moveTo(t[0]+t[2]*.5,t[1]);ctx.lineTo(t[0],t[1]);ctx.lineTo(t[0],t[1]+t[3]*.5);ctx.stroke();}
}

function drawCell(c){
  const now=performance.now()/1000;
  let sc=clamp((now-c.spawnT)/0.3,0,1);if(sc<1)sc=easeOutBack(sc);else sc=1;
  sc*=1+0.1*c.pulse;
  if(c.type===0)return;
  ctx.save();ctx.translate(c.c+.5,c.r+.5);ctx.scale(sc,sc);
  // pad
  const hovered=(state==='build'&&!paused)&&hoverCell===c;
  rr(ctx,-.46,-.46,.92,.92,.12);
  ctx.fillStyle=c.type===9?'#2a101d':(hovered?PAL.padHover:PAL.pad);ctx.fill();
  ctx.strokeStyle='rgba(21,80,100,.7)';ctx.lineWidth=.03;ctx.stroke();

  if(c.type===9){ // spike
    const pu=.5+.5*Math.sin(now*3+c.c);
    ctx.globalAlpha=.55+.4*pu;ctx.strokeStyle=PAL.red;ctx.lineWidth=.09;ctx.lineCap='round';
    ctx.beginPath();ctx.moveTo(-.26,-.26);ctx.lineTo(.26,.26);ctx.moveTo(.26,-.26);ctx.lineTo(-.26,.26);ctx.stroke();
    ctx.fillStyle=PAL.red;
    for(let i=0;i<4;i++){const d=DIRS[i];ctx.beginPath();
      ctx.moveTo(d.x*.44-d.y*.1,d.y*.44+d.x*.1);
      ctx.lineTo(d.x*.44+d.y*.1,d.y*.44-d.x*.1);
      ctx.lineTo(d.x*.26,d.y*.26);ctx.closePath();ctx.fill();}
    ctx.globalAlpha=1;ctx.restore();return;
  }
  // roads
  const cs=connsOf(c.type,effRot(c));
  ctx.lineCap='round';
  ctx.strokeStyle=PAL.asphaltEdge;ctx.lineWidth=.68;
  for(const d of cs){const v=DIRS[d];ctx.beginPath();ctx.moveTo(0,0);ctx.lineTo(v.x*.52,v.y*.52);ctx.stroke();}
  ctx.strokeStyle=PAL.asphalt;ctx.lineWidth=.56;
  for(const d of cs){const v=DIRS[d];ctx.beginPath();ctx.moveTo(0,0);ctx.lineTo(v.x*.52,v.y*.52);ctx.stroke();}
  // center dash on straights
  if(c.type===1&&cs.length===2){const v=DIRS[cs[0]];
    ctx.strokeStyle='rgba(232,80,58,.75)';ctx.lineWidth=.06;
    ctx.setLineDash([.14,.13]);
    ctx.beginPath();ctx.moveTo(-v.x*.4,-v.y*.4);ctx.lineTo(v.x*.4,v.y*.4);ctx.stroke();ctx.setLineDash([]);}

  // start flag / finish band
  if(level&&c===level.start){
    ctx.strokeStyle='#f2ead8';ctx.lineWidth=.05;
    ctx.beginPath();ctx.moveTo(-.36,.3);ctx.lineTo(-.36,-.24);ctx.stroke();
    for(let i=0;i<3;i++)for(let j=0;j<2;j++){
      ctx.fillStyle=(i+j)%2?'#1a1f24':'#f2ead8';
      ctx.fillRect(-.36+i*.1,-.24+j*.09,.1,.09);}
  }else if(level&&c===level.finish){
    const near=race&&!race.loop&&Math.hypot(carPos().x-(c.c+.5),carPos().y-(c.r+.5))<1.6;
    for(let i=0;i<2;i++)for(let j=0;j<6;j++){
      ctx.fillStyle=(i+j)%2?'#1a1f24':'#f2ead8';
      ctx.fillRect(-.42+i*.1,-.3+j*.1,.1,.1);}
    if(near){ctx.strokeStyle='rgba(255,210,63,'+(.4+.4*Math.sin(now*10))+')';ctx.lineWidth=.07;
      rr(ctx,-.5,-.5,1,1,.14);ctx.stroke();}
  }
  // marks
  if(c.mark==='c'&&!isCollected(c)){
    const sx=Math.abs(Math.cos(now*5+c.c*1.7));
    ctx.strokeStyle='rgba(255,210,63,.25)';ctx.lineWidth=.03;ctx.setLineDash([.08,.08]);
    ctx.beginPath();ctx.arc(0,0,.3,0,TAU);ctx.stroke();ctx.setLineDash([]);
    ctx.save();ctx.scale(Math.max(.15,sx),1);
    const cg=ctx.createRadialGradient(-.08,-.08,.02,0,0,.27);
    cg.addColorStop(0,'#fff2b0');cg.addColorStop(.6,'#ffd23f');cg.addColorStop(1,'#c98a12');
    ctx.fillStyle=cg;ctx.beginPath();ctx.arc(0,0,.25,0,TAU);ctx.fill();
    ctx.strokeStyle='#8a5c07';ctx.lineWidth=.04;ctx.stroke();
    ctx.restore();
  }else if(c.mark==='b'){
    const dir=padDir(c),o=((now*1.5)%0.6)-0.3;
    ctx.strokeStyle='rgba(63,217,255,'+(.6+.4*Math.sin(now*6))+')';ctx.lineWidth=.1;ctx.lineCap='round';
    for(const off of [o,o-.3]){
      const bx=dir.x*off,by=dir.y*off,px=-dir.y,py=dir.x;
      ctx.beginPath();ctx.moveTo(bx-px*.14-dir.x*.12,by-py*.14-dir.y*.12);
      ctx.lineTo(bx+px*.02,by+py*.02);
      ctx.lineTo(bx-px*.14+dir.x*.12,by-py*.14+dir.y*.12);ctx.stroke();}
  }
  if(c.spin){
    const a=c.spin.phase;
    ctx.strokeStyle=level.locked?'rgba(255,181,39,.5)':'#ffb527';ctx.lineWidth=.09;ctx.lineCap='round';
    ctx.beginPath();ctx.arc(0,0,.34,a,a+2.1);ctx.stroke();
    const ex=Math.cos(a+2.1)*.34,ey=Math.sin(a+2.1)*.34;
    const tx=-Math.sin(a+2.1),ty=Math.cos(a+2.1);
    ctx.fillStyle=level.locked?'rgba(255,181,39,.5)':'#ffb527';
    ctx.beginPath();ctx.moveTo(ex+tx*.14,ey+ty*.14);
    ctx.lineTo(ex-ty*.1,ey+tx*.1);ctx.lineTo(ex+ty*.1,ey-tx*.1);ctx.closePath();ctx.fill();
  }
  ctx.restore();
}
function isCollected(c){if(!race)return false;
  for(const k of race.coins)if(Math.abs(k.x-(c.c+.5))<.01&&Math.abs(k.y-(c.r+.5))<.01&&k.got)return true;return false;}
function padDir(c){ // flow direction at a boost pad along the route (or ready route)
  const cells=(state==='race'||state==='menu')?race.cells:route;
  if(cells)for(let i=0;i<cells.length-1;i++)if(cells[i]===c){const n=cells[i+1];
    return {x:n.c-c.c,y:n.r-c.r};}
  return {x:1,y:0};
}
