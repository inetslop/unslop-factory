const clamp=(v,a,b)=>v<a?a:v>b?b:v;
function aabb(ax,ay,aw,ah,bx,by,bw,bh){ return Math.abs(ax-bx)<(aw+bw)/2 && ay<by+bh/2 && ay+ah>by-bh/2; }

function spawnZombie(type,demo){
  const t=ZT[type];
  const z={type,x:VW+50+(type==='brute'?40:0),y:GROUND,h:t.h,face:-1,
    speed:rnd(t.sp[0],t.sp[1])*(demo?.8:(1+Math.min(.6,(wave-1)*.07))),
    hp:t.hp,dmg:t.dmg,pts:t.pts,phase:Math.random()*6,flash:0,atkCd:rnd(.3,.9),
    color:t.sk[(Math.random()*t.sk.length)|0],pants:t.pn[(Math.random()*t.pn.length)|0],
    shirt:SHIRTS[(Math.random()*SHIRTS.length)|0],
    eye:type==='runner'?'#ff5a45':(type==='spitter'?'#c9e05a':'#ffd23e'),
    lean:type==='runner'?.3:.12,dead:false,deathT:0,spitCd:rnd(.8,1.6),demo:!!demo,spots:[]};
  for(let i=0;i<3;i++) z.spots.push([rnd(-7,7),rnd(-z.h*.8,-z.h*.45),rnd(1.5,3)]);
  zombies.push(z);
}
function pickType(){
  const r=Math.random();
  if(wave<2) return 'walker';
  let wB = wave>=4 ? Math.min(.22,.06*(wave-3)) : 0;
  let wS = wave>=3 ? .18 : 0;
  let wR = Math.min(.34,.1+wave*.05);
  if(r<wB) return 'brute'; r-=wB;
  if(r<wS) return 'spitter'; r-=wS;
  if(r<wR) return 'runner';
  return 'walker';
}

function reset(){
  zombies.length=bullets.length=globs.length=parts.length=stains.length=texts.length=pickups.length=0;
  Object.assign(player,{x:170,y:GROUND,vx:0,vy:0,onGround:true,facing:1,hp:100,mag:MAG,rel:0,fireCd:0,invuln:0,adren:0,recoil:0,anim:0});
  elapsed=0;dist=0;score=0;kills=0;wave=1;spawnT=1.2;pickupT=7;moanT=5;heartT=0;bannerT=2.2;
  shake=0;hurtFlash=0;muzzleT=0;dieT=0;playerAlive=true;jumpBuf=0;
}
function startRun(){ reset(); state='play';
  startScr.classList.add('hidden'); overScr.classList.add('hidden'); pauseScr.classList.add('hidden');
  hud.classList.add('on'); }
function setPause(v){ if(v&&state==='play'){ state='pause'; pauseScr.classList.remove('hidden'); }
  else if(!v&&state==='pause'){ state='play'; pauseScr.classList.add('hidden'); } }

/* ============================== FX HELPERS ============================== */
function addShake(m){ shake=Math.min(18,shake+m); }
function burst(x,y,n,cols,spd,grav,up){
  for(let i=0;i<n;i++){ if(parts.length>320)parts.shift();
    const a=rnd(0,Math.PI*2), s=rnd(spd*.3,spd);
    parts.push({x,y,vx:Math.cos(a)*s,vy:Math.sin(a)*s-(up?rnd(60,180):0),
      life:0,max:rnd(.4,.9),size:rnd(1.5,4),color:cols[(Math.random()*cols.length)|0],grav}); }
}
const BLOOD=['#a3271f','#c0392b','#7e1d16'];
function ftext(x,y,txt,color,size){ if(texts.length>24)texts.shift();
  texts.push({x,y,txt,c:color||'#e6e1d3',size:size||14,life:0,max:.95}); }

/* ============================== GAMEPLAY ============================== */
function fire(){
  player.mag--; player.fireCd = player.adren>0 ? .075 : .11;
  muzzleT=.055; player.recoil=1; addShake(2.2); sfx.shot();
  const mx=player.x+player.facing*30, my=player.y-32;
  bullets.push({x:mx,y:my,vx:player.facing*1050});
  burst(mx,my,3,['#ffd76a','#fff'],180,0,false);
  if(player.mag<=0) startRel();
}
function startRel(){ if(player.rel<=0 && player.mag<MAG){ player.rel=REL_T; sfx.reload(); ftext(player.x,player.y-64,'RELOAD','#d9a13b',11); } }

function hurtPlayer(d,color){
  if(player.invuln>0||!playerAlive)return;
  player.hp-=d; player.invuln=.6; hurtFlash=1; addShake(7); sfx.hurt();
  burst(player.x,player.y-30,14,color||BLOOD,260,900,true);
  if(player.hp<=0){ player.hp=0; killPlayer(); }
}
function killPlayer(){
  playerAlive=false; state='dying'; dieT=0; addShake(14); sfx.boom();
  burst(player.x,player.y-28,46,BLOOD,420,900,true);
  stains.push({x:player.x,y:GROUND+3,rx:26,a:.7});
}
function killZombie(z,bonus){
  z.dead=true; z.deathT=0; kills++;
  const pts=z.pts+(bonus||0); score+=pts;
  ftext(z.x,z.y-z.h-8,'+'+pts,'#e6e1d3');
  burst(z.x,z.y-z.h*.5,22,BLOOD,340,900,true);
  stains.push({x:z.x,y:GROUND+3,rx:rnd(9,20),a:.55});
  sfx.die(); addShake(2);
  if(Math.random()<.12) dropPickup(z.x, z.y-z.h*.6, Math.random()<.6?'med':'adr');
}
function dropPickup(x,y,type){
  pickups.push({type,x,y,vy:-rnd(120,220),seed:Math.random()*7});
}

/* ============================== UPDATE ============================== */
function update(dt){
  T+=dt;
  const live = state==='play';
  const worldDt = (state==='dying') ? dt*.3 : dt;
  if(state!=='pause'&&state!=='over'){
    /* scroll & distance */
    const sp = state==='start' ? 55 : 70+Math.min(65,elapsed*.8);
    scrollX+=sp*worldDt;
    if(live) dist+=sp*worldDt/9;

    /* waves */
    if(live){
      elapsed+=dt;
      const nw=1+Math.floor(elapsed/25);
      if(nw>wave){ wave=nw; bannerT=2.2; sfx.wave(); }
      spawnT-=dt;
      if(spawnT<=0 && zombies.length<16){
        spawnZombie(pickType());
        if(wave>=5&&Math.random()<.3) spawnZombie('runner');
        spawnT=Math.max(.4,1.7-wave*.13)*rnd(.7,1.2);
      }
      pickupT-=dt;
      if(pickupT<=0){ dropPickup(VW+30,GROUND-16,Math.random()<.6?'med':'adr'); pickupT=rnd(7,12); }
      moanT-=dt; if(moanT<=0){ distantMoan(); moanT=rnd(4,9); }
      if(player.hp<30&&player.hp>0){ heartT-=dt; if(heartT<=0){ sfx.heart(); heartT=.85; } }
    } else if(state==='start'){
      demoT-=dt;
      if(demoT<=0 && zombies.length<6){ spawnZombie('walker',true); demoT=rnd(1.2,2.4); }
    }

    /* ---- player ---- */
    if(live&&playerAlive){
      const sp=player.adren>0?430:320;
      let ax=0;
      if(keys.KeyA||keys.ArrowLeft)ax-=1;
      if(keys.KeyD||keys.ArrowRight)ax+=1;
      player.vx += ax*sp/dt*.016*8; /* smooth accel */
      player.vx = clamp(player.vx, -sp, sp);
      if(ax===0) player.vx*=Math.pow(.0008,dt);
      if(Math.abs(player.vx)<4&&ax===0)player.vx=0;
      if(ax!==0){ player.facing=ax>0?1:-1; player.anim+=Math.abs(player.vx)*worldDt*.055; }
      jumpBuf-=dt;
      if(jumpBuf>0&&player.onGround){ player.vy=-740; player.onGround=false; jumpBuf=0; sfx.jump(); }
      player.vy+=2000*worldDt;
      player.y+=player.vy*worldDt;
      if(player.y>=GROUND){ player.y=GROUND; player.vy=0; player.onGround=true; }
      player.x=clamp(player.x+player.vx*worldDt,40,VW-60);
      player.fireCd-=dt; player.rel=Math.max(0,player.rel-dt); player.invuln-=dt; player.adren-=dt;
      player.recoil=Math.max(0,player.recoil-dt*8);
      if(player.rel===0&&player.mag<MAG) player.mag=MAG;
      if((keys.Space||keys.KeyJ||mouseDown)&&player.fireCd<=0){
        if(player.rel<=0){ if(player.mag>0)fire(); else { sfx.dry(); startRel(); } }
      }
      /* stomps */
      if(player.vy>120){
        for(const z of zombies){
          if(z.dead)continue;
          const top=z.y-z.h;
          if(player.x>z.x-z.h*.3&&player.x<z.x+z.h*.3 && player.y>=top-4 && player.y<=top+z.h*.45){
            z.hp-=5; z.flash=1; player.vy=-430; player.onGround=false;
            sfx.stomp(); ftext(z.x,top-14,'STOMP!','#8fae5f');
            burst(z.x,top,12,BLOOD,240,900,true); addShake(3);
            if(z.hp<=0) killZombie(z,5);
            break;
          }
        }
      }
      if(player.adren>0&&Math.random()<.5&&parts.length<300)
        parts.push({x:player.x-player.facing*14,y:player.y-rnd(8,40),vx:-player.facing*rnd(120,260),vy:rnd(-30,30),life:0,max:.3,size:2,color:'rgba(255,214,90,.7)',grav:0});
    }

    /* ---- bullets ---- */
    for(let i=bullets.length-1;i>=0;i--){
      const b=bullets[i]; b.x+=b.vx*worldDt;
      if(b.x<-30||b.x>VW+30){ bullets.splice(i,1); continue; }
      let hit=false;
      for(const z of zombies){
        if(z.dead)continue;
        const w2=z.h*.44/2+4;
        if(Math.abs(b.x-z.x)<w2 && b.y>z.y-z.h && b.y<z.y){
          const head = b.y < z.y-z.h*.58;
          z.hp-=head?2:1; z.flash=1; z.x+=(z.x>player.x?2:-2);
          burst(b.x,b.y,7,BLOOD,240*Math.sign(b.vx||1),900,false);
          sfx.zhit();
          if(z.hp<=0){ killZombie(z,head?z.pts:0); if(head)ftext(z.x,z.y-z.h-26,'HEADSHOT ×2','#d9a13b'); }
          hit=true; break;
        }
      }
      if(hit)bullets.splice(i,1);
    }

    /* ---- zombies ---- */
    for(let i=zombies.length-1;i>=0;i--){
      const z=zombies[i];
      if(z.dead){
        z.deathT+=dt; z.x-= (state==='start'?55:70)*worldDt*1.15;
        if(z.x<-80||z.deathT>3.4) zombies.splice(i,1);
        continue;
      }
      z.flash=Math.max(0,z.flash-dt*5); z.atkCd-=dt;
      const dir=playerAlive?(player.x<z.x?-1:1):-1; z.face=dir;
      let mv=true;
      if(z.type==='spitter'&&playerAlive&&!z.demo){
        const d=Math.abs(player.x-z.x);
        if(d<250)mv=false;
        z.spitCd-=dt;
        if(z.spitCd<=0&&d>60){ z.spitCd=rnd(1.7,2.3);
          const tx=player.x, ty=player.y-30, tt=.75, g=1400;
          globs.push({x:z.x+dir*10,y:z.y-z.h*.88,vx:(tx-z.x)/tt,vy:(ty-z.y-.5*g*tt*tt)/tt}); }
      }
      if(mv){ z.x+=dir*z.speed*worldDt; z.phase+=worldDt*(z.speed/30); } else z.phase+=worldDt*2;

      if(playerAlive&&!z.demo&&state==='play'&&z.atkCd<=0){
        const w2=z.h*.44/2+9;
        if(Math.abs(player.x-z.x)<w2 && player.y>z.y-z.h-10 && z.y>player.y-52){
          z.atkCd=.9; hurtPlayer(z.dmg); z.x-=dir*16; addShake(4);
        }
      }
      if(z.x<-90||z.x>VW+160) zombies.splice(i,1);
    }

    /* ---- globs ---- */
    for(let i=globs.length-1;i>=0;i--){
      const g=globs[i]; g.vy+=1400*worldDt; g.x+=g.vx*worldDt; g.y+=g.vy*worldDt;
      let dead=false;
      if(playerAlive&&Math.abs(g.x-player.x)<13&&g.y>player.y-52&&g.y<player.y){
        hurtPlayer(10,['#8fae3f','#b7d94e']); burst(g.x,g.y,10,['#8fae3f','#b7d94e'],220,900,true); dead=true;
      }
      if(g.y>=GROUND){ stains.push({x:g.x,y:GROUND+3,rx:rnd(6,12),a:.4}); burst(g.x,GROUND-2,8,['#8fae3f','#b7d94e'],180,900,true); dead=true; }
      if(dead||g.x<-40||g.x>VW+40) globs.splice(i,1);
    }

    /* ---- pickups ---- */
    for(let i=pickups.length-1;i>=0;i--){
      const p=pickups[i];
      p.vy+=1200*worldDt; p.y+=p.vy*worldDt;
      if(p.y>GROUND-8){ p.y=GROUND-8; p.vy=0; }
      p.x-= (state==='start'?55:70)*worldDt*.9;
      if(playerAlive&&state==='play'&&Math.abs(p.x-player.x)<22&&p.y>player.y-56){
        if(p.type==='med'){ player.hp=Math.min(100,player.hp+35); ftext(player.x,player.y-64,'+35 HP','#e89a9a'); burst(p.x,p.y,10,['#e8e4d8','#c0392b'],160,400,true); }
        else{ player.adren=6; ftext(player.x,player.y-64,'ADRENALINE!','#ffd75a'); burst(p.x,p.y,10,['#ffd75a','#fff'],180,300,true); }
        sfx.pickup(); pickups.splice(i,1); continue;
      }
      if(p.x<-40)pickups.splice(i,1);
    }

    /* ---- particles / stains / texts ---- */
    for(let i=parts.length-1;i>=0;i--){ const p=parts[i];
      p.life+=dt; if(p.life>p.max){ parts.splice(i,1); continue; }
      p.vy+=p.grav*dt; p.x+=p.vx*dt; p.y+=p.vy*dt;
      if(p.y>GROUND+2&&p.grav>0){ p.y=GROUND+2; p.vy*=-.3; p.vx*=.6; } }
    for(let i=stains.length-1;i>=0;i--){ const s=stains[i];
      s.x-= (state==='start'?55:70)*worldDt; s.a-=dt*.045;
      if(s.a<=0||s.x<-40) stains.splice(i,1); }
    for(let i=texts.length-1;i>=0;i--){ const t=texts[i];
      t.life+=dt; t.y-=38*dt; if(t.life>t.max)texts.splice(i,1); }

    shake=Math.max(0,shake-dt*26); hurtFlash=Math.max(0,hurtFlash-dt*2.4); muzzleT-=dt; bannerT-=dt;

    /* dying transition */
    if(state==='dying'){ dieT+=dt;
      if(dieT>1.5){ state='over'; overAt=T; showOver(); } }
  }
}

function showOver(){
  const nb = score>best;
  if(nb){ best=score; saveBest(best); }
  $('stScore').textContent=score.toLocaleString('en-US');
  $('stBest').textContent=best.toLocaleString('en-US');
  $('stKills').textContent=kills;
  $('stWave').textContent=wave;
  $('stLine').textContent='You walked '+Math.floor(dist)+' m on wave '+wave+' before the horde caught up.';
  $('newBest').classList.toggle('hidden',!nb);
  overScr.classList.remove('hidden'); hud.classList.remove('on');
}

/* ============================== DRAW: WORLD ============================== */
function line(x1,y1,x2,y2){ ctx.beginPath();ctx.moveTo(x1,y1);ctx.lineTo(x2,y2);ctx.stroke(); }
function dot(x,y,r,c){ if(c)ctx.fillStyle=c; ctx.beginPath();ctx.arc(x,y,r,0,7);ctx.fill(); }
function rr(x,y,w,h,r){ ctx.beginPath();ctx.moveTo(x+r,y);ctx.arcTo(x+w,y,x+w,y+h,r);ctx.arcTo(x+w,y+h,x,y+h,r);ctx.arcTo(x,y+h,x,y,r);ctx.arcTo(x,y,x+w,y,r);ctx.closePath();ctx.fill(); }

function drawSky(){
  const g=ctx.createLinearGradient(0,-24,0,VH);
  g.addColorStop(0,'#070d14'); g.addColorStop(.55,'#0c1a1a'); g.addColorStop(1,'#16281f');
  ctx.fillStyle=g; ctx.fillRect(-24,-24,VW+48,VH+48);
  for(let i=0;i<50;i++){ const x=h1(i)*VW, y=h1(i*3.1)*(GROUND-90);
    ctx.globalAlpha=.15+.4*Math.abs(Math.sin(T*(.6+h1(i*7))+i));
    ctx.fillStyle='#cfe0d8'; ctx.fillRect(x,y,h1(i*5)>.8?2:1,1); }
  ctx.globalAlpha=1;
  const mx=VW-185-scrollX*.015, my=112;
  const rg=ctx.createRadialGradient(mx,my,10,mx,my,140);
  rg.addColorStop(0,'rgba(205,230,190,.22)'); rg.addColorStop(1,'rgba(205,230,190,0)');
  ctx.fillStyle=rg; ctx.fillRect(mx-140,my-140,280,280);
  dot(mx,my,46,'#cfe0b8');
  ctx.fillStyle='rgba(150,170,130,.5)';
  dot(mx-14,my-10,9); dot(mx+12,my+14,6); dot(mx+4,my-22,4); dot(mx-8,my+24,4);
}
function layer(f,sp,fn){
  const off=scrollX*f; let i=Math.floor(-off/sp)-1;
  for(;;i++){ const x=i*sp-off; if(x>VW+sp)break; fn(i,x); }
}
function drawFar(){
  ctx.fillStyle='#0d1a22';
  layer(.12,96,(i,x)=>{
    const w=40+h1(i)*70, hgt=50+h1(i*3.7)*115;
    if(h1(i*7)>.8){ ctx.fillRect(x,GROUND-30-hgt,w,hgt);
      ctx.beginPath();ctx.moveTo(x,GROUND-30-hgt);ctx.lineTo(x+w*.4,GROUND-30-hgt-12);ctx.lineTo(x+w*.7,GROUND-30-hgt);ctx.fill();
    } else ctx.fillRect(x,GROUND-30-hgt,w,hgt);
    if(h1(i*11)>.92){ line(x+w/2,GROUND-30-hgt,x+w/2,GROUND-58-hgt);
      ctx.globalAlpha=.4+.6*Math.abs(Math.sin(T*2.4+i)); dot(x+w/2,GROUND-60-hgt,2,'#ff5a45'); ctx.globalAlpha=1; }
  });
}

}

/* ============================== DRAW: ENTITIES ============================== */
function drawZombie(z){
  ctx.save(); ctx.translate(z.x,z.y);
  if(z.dead){ const t=Math.min(1,z.deathT/.34); ctx.rotate(-z.face*t*1.52);
    if(z.deathT>2.1)ctx.globalAlpha=Math.max(0,1-(z.deathT-2.1)/1.1); }
  else ctx.rotate(z.face*z.lean*(.6+.4*Math.sin(z.phase)));
  const h=z.h,w=h*.44,lr=h*.17;
  const swg=Math.sin(z.phase)*(z.dead?0:1)*h*.13;
  ctx.lineCap='round'; ctx.strokeStyle=z.pants; ctx.lineWidth=Math.max(3,h*.085);
