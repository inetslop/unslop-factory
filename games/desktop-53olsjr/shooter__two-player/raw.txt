  else if(Math.random()<.14)dropPickup(e.x,e.y);
}

/* ============ update ============ */
function updateFx(dt){
  for(let i=parts.length-1;i>=0;i--){const p=parts[i];p.life-=dt;
    if(p.life<=0){parts.splice(i,1);continue;}
    if(p.kind==='ring'){p.r+=p.vr*dt;}else{p.x+=p.vx*dt;p.y+=p.vy*dt;p.vx*=Math.pow(.02,dt);p.vy*=Math.pow(.02,dt);}}
  for(let i=pops.length-1;i>=0;i--){const p=pops[i];p.t+=dt;p.y-=46*dt;if(p.t>p.life)pops.splice(i,1);}
  shake*=Math.exp(-6*dt);hurtFlash=Math.max(0,hurtFlash-dt*1.3);
}
function updatePlay(dt){
  /* players */
  const M1=[[keys.has('KeyW'),-1],[keys.has('KeyS'),1],[keys.has('KeyA'),-1],[keys.has('KeyD'),1]];
  players.forEach((p,i)=>{
    p.dhp=Math.max(p.hp,p.dhp-50*dt);
    if(!p.alive)return;
    p.iframes=Math.max(0,p.iframes-dt);p.triple=Math.max(0,p.triple-dt);
    p.rapid=Math.max(0,p.rapid-dt);p.shield=Math.max(0,p.shield-dt);p.fireT-=dt;
    let dx=0,dy=0;
    if(i===0){if(M1[2][0])dx--;if(M1[3][0])dx++;if(M1[0][0])dy--;if(M1[1][0])dy++;}
    else{if(keys.has('ArrowLeft'))dx--;if(keys.has('ArrowRight'))dx++;if(keys.has('ArrowUp'))dy--;if(keys.has('ArrowDown'))dy++;}
    if(dx||dy){const l=Math.hypot(dx,dy);dx/=l;dy/=l;p.face=Math.atan2(dy,dx);}
    const sp=265,k=1-Math.exp(-14*dt);
    p.vx+=(dx*sp-p.vx)*k;p.vy+=(dy*sp-p.vy)*k;
    p.x+=p.vx*dt;p.y+=p.vy*dt;
    p.x=Math.min(W-16,Math.max(16,p.x));p.y=Math.min(H-16,Math.max(16,p.y));
    const firing=i===0?(keys.has('Space')||keys.has('KeyF')):(keys.has('Enter')||keys.has('Numpad0'));
    if(firing&&p.fireT<=0)doFire(p);
  });
  /* revive / death */
  let anyAlive=false;players.forEach(p=>{if(p.alive){anyAlive=true;return;}
    p.reviveP=Math.max(0,p.reviveP-dt*2);
    for(const o of players)if(o.alive&&Math.hypot(o.x-p.deadX,o.y-p.deadY)<52){p.reviveP+=dt/1.1;break;}
    if(p.reviveP>=1){p.alive=true;p.hp=50;p.dhp=50;p.iframes=2;p.x=p.deadX;p.y=p.deadY;p.vx=0;p.vy=0;
      boom(p.x,p.y,PCOL[p.id],22,300);pushPop(p.x,p.y-26,'REVIVED',PCOL[p.id]);sfx.revive();}
  });
  if(!anyAlive){pendingOver+=dt;if(pendingOver>1.4){gameOver();return;}}else pendingOver=0;
  /* telegraphs + spawn */
  for(let i=telegraphs.length-1;i>=0;i--){const t=telegraphs[i];t.t-=dt;
    if(t.t<=0){enemies.push(makeEnemy(t.type,t.x,t.y));sparks(t.x,t.y,ETYPES[t.type].col,6,150);telegraphs.splice(i,1);}}
  if(waveState==='spawning'){
    if(spawnQueue.length&&enemies.length<26){spawnT-=dt;
      if(spawnT<=0){const pt=edgePoint();telegraphs.push({x:pt.x,y:pt.y,t:.7,type:spawnQueue.pop()});spawnT=Math.max(.35,.95-wave*.04);}}
    if(!spawnQueue.length&&!telegraphs.length&&!enemies.length){
      waveState='clear';clearT=2.6;score+=wave*100;setBanner('WAVE '+pad2(wave)+' CLEARED','BONUS +'+(wave*100));
      players.forEach(p=>{if(p.alive){p.hp=Math.min(p.maxHp,p.hp+15);pushPop(p.x,p.y-24,'+15','#5dff9d');}});
    }
  }else if(waveState==='clear'){clearT-=dt;if(clearT<=0)nextWave();}
  /* enemies */
  for(const e of enemies){
    e.t+=dt;e.hit=Math.max(0,e.hit-dt*6);e.scale=Math.min(1,e.scale+dt*4);
    let tx=W/2,ty=H/2,bd=1e9,target=null;
    for(const p of players)if(p.alive){const d=Math.hypot(p.x-e.x,p.y-e.y);if(d<bd){bd=d;target=p;tx=p.x;ty=p.y;}}
    const c=ETYPES[e.type],dx=target?tx-e.x:W/2-e.x,dy=target?ty-e.y:H/2-e.y,d=Math.hypot(dx,dy)||1;
    let sp=c.speed*(1+(wave-1)*.03),svx,svy;
    if(e.type==='chaser'){svx=dx/d*sp;svy=dy/d*sp;}
    else if(e.type==='runner'){const px=-dy/d,py=dx/d,w=Math.sin(e.t*7+e.sin)*130;svx=dx/d*sp+px*w;svy=dy/d*sp+py*w;}
    else if(e.type==='spitter'){
      let mx=0,my=0;
      if(d>380){mx=dx/d*sp;my=dy/d*sp;}else if(d<240){mx=-dx/d*sp;my=-dy/d*sp;}
      else{const px=-dy/d,py=dx/d,w=Math.sin(e.t*1.5+e.sin)*46;mx=px*w;my=py*w;}
      svx=mx;svy=my;
      if(e.charge<=0){e.shootT-=dt;if(e.shootT<=0&&target&&d<580)e.charge=.35;}
      else{e.charge-=dt;if(e.charge<=0){if(target)eb(e,tx,ty,250,c.shotDmg);sfx.eshoot();e.shootT=1.7+Math.random()*.8;}}
    }
    else{svx=dx/d*sp;svy=dy/d*sp;}
    const k=1-Math.exp(-(e.type==='runner'?6:e.type==='brute'?2.5:4)*dt);
    e.vx+=(svx-e.vx)*k;e.vy+=(svy-e.vy)*k;
    e.x+=e.vx*dt;e.y+=e.vy*dt;
    e.x=Math.min(W-14,Math.max(14,e.x));e.y=Math.min(H-14,Math.max(14,e.y));
  }
  for(let i=0;i<enemies.length;i++)for(let j=i+1;j<enemies.length;j++){
    const a=enemies[i],b=enemies[j];let dx=b.x-a.x,dy=b.y-a.y;const dd=Math.hypot(dx,dy),min=a.r+b.r-4;
    if(dd>0&&dd<min){const nx=dx/dd,ov=(min-dd)*.35;a.x-=nx*ov;a.y-=ny2(dy,dd)*ov;b.x+=nx*ov;b.y+=ny2(dy,dd)*ov;}
  }
  /* player bullets */
  for(let i=bullets.length-1;i>=0;i--){const b=bullets[i];
    b.x+=b.vx*dt;b.y+=b.vy*dt;b.life-=dt;
    if(b.life<=0||b.x<4||b.x>W-4||b.y<4||b.y>H-4){if(b.life>0)sparks(b.x,b.y,PCOL[b.pid],3,90);bullets.splice(i,1);continue;}
    for(let j=enemies.length-1;j>=0;j--){const e=enemies[j];
      if(Math.hypot(e.x-b.x,e.y-b.y)<e.r+4){
        e.hp-=b.dmg;e.hit=.12;sparks(b.x,b.y,ETYPES[e.type].col,5,180);sfx.hit();bullets.splice(i,1);
        if(e.hp<=0)killEnemy(e,j,b.pid);
        break;}}
  }
  /* enemy bullets */
  for(let i=ebullets.length-1;i>=0;i--){const b=ebullets[i];
    b.x+=b.vx*dt;b.y+=b.vy*dt;b.life-=dt;
    if(b.life<=0||b.x<4||b.x>W-4||b.y<4||b.y>H-4){sparks(b.x,b.y,'#ff7ad9',3,90);ebullets.splice(i,1);continue;}
    for(const p of players)if(p.alive&&Math.hypot(p.x-b.x,p.y-b.y)<14){
      if(p.shield>0){sparks(b.x,b.y,'#bfe8ff',5,150);sfx.block();}
      else damagePlayer(p,b.dmg,-b.vx*.25,-b.vy*.25);
      ebullets.splice(i,1);break;}
  }
  /* contact */
  for(const e of enemies)for(const p of players){
    if(!p.alive)continue;
    const d=Math.hypot(p.x-e.x,p.y-e.y);
    if(d<e.r+12&&d>0){const nx=(p.x-e.x)/d,ny=(p.y-e.y)/d;
      if(p.shield>0){e.vx-=nx*260;e.vy-=ny*260;sparks((p.x+e.x)/2,(p.y+e.y)/2,'#bfe8ff',4,140);}
      else damagePlayer(p,ETYPES[e.type].dmg,nx*300,ny*300);}}
  /* pickups */
  for(let i=pickups.length-1;i>=0;i--){const pk=pickups[i];pk.t-=dt;pk.ph+=dt*2;
    if(pk.t<=0){pickups.splice(i,1);continue;}
    let near=null,bd=1e9;for(const p of players)if(p.alive){const d=Math.hypot(p.x-pk.x,p.y-pk.y);if(d<bd){bd=d;near=p;}}
    if(near&&bd<130){pk.x+=(near.x-pk.x)/bd*260*dt;pk.y+=(near.y-pk.y)/bd*260*dt;}
    if(near&&bd<17){applyPickup(near,pk.type);pickups.splice(i,1);}
  }
  /* combo / banner / heartbeat */
  if(comboT>0){comboT-=dt;if(comboT<=0)combo=0;}
  comboPop=Math.max(0,comboPop-dt*3);
  if(bannerObj){bannerObj.t+=dt;if(bannerObj.t>bannerObj.dur)bannerObj=null;}
  const low=players.some(p=>p.alive&&p.hp<25);
  if(low){hbT-=dt;if(hbT<=0){sfx.heart();hbT=.85;}}else hbT=0;
  updateFx(dt);
}

function shapePath(e){
  const t=e.type,r=ETYPES[t].r;ctx.beginPath();
  if(t==='chaser'){ctx.moveTo(r+4,0);ctx.lineTo(-r,-r*.75);ctx.lineTo(-r*.5,0);ctx.lineTo(-r,r*.75);}
  else if(t==='runner'){ctx.moveTo(r+3,0);ctx.lineTo(-r,-r*.62);ctx.lineTo(-r*.45,0);ctx.lineTo(-r,r*.62);}
  else{const n=t==='spitter'?6:8,off=t==='spitter'?0:Math.PI/8;
    for(let i=0;i<n;i++){const a=i*6.283/n+off,x=Math.cos(a)*r,y=Math.sin(a)*r;i?ctx.lineTo(x,y):ctx.moveTo(x,y);}}
  ctx.closePath();
}
