  let pts=f.pts*(combo?2:1); G.score+=pts;
  if(G.score>G.best){G.best=G.score; newBest=true;}
  scorePop=0.35;
  s.growth=Math.min(s.growth+f.grow,40-s.cells.length);
  const x=ox+f.x*cell,y=oy+f.y*cell;
  texts.push({x,y:y-12,str:'+'+pts+(combo?'  ×2!':''),life:1,max:1,color:combo?'#ffb648':'#f2ead8',size:combo?19:15});
  for(let k=0;k<10;k++)parts.push({shape:'dot',x,y,vx:rand(-90,90),vy:rand(-130,-20),g:340,life:rand(.3,.6),max:.6,size:rand(1.5,3),color:f.type==='fish'?'#7fd8f0':'#8fe05a'});
  if(f.type==='fish')SFX.eatFish(); else SFX.eatFrog();
}
function inBounds(x,y){ return x>=0&&x<COLS&&y>=0&&y<ROWS; }

function updateSnake(dt){
  const s=snake;
  // carrying (only when not mid-hop, so gap-jumps are safe)
  if(s.hopT>=1){
    const sup=supportAtPx(s.px,s.py);
    if(sup){
      s.px+=sup.vx*spdMul()*dt; s.py+=sup.vy*spdMul()*dt;
      if(s.px-(s.cx+0.5)>=0.52){s.cx++;commitMove();}
      else if(s.px-(s.cx+0.5)<=-0.52){s.cx--;commitMove();}
      if(s.py-(s.cy+0.5)>=0.52){s.cy++;commitMove();}
      else if(s.py-(s.cy+0.5)<=-0.52){s.cy--;commitMove();}
    }else{
      const wr=Math.round(s.py-0.5);
      if(wr>=1&&wr<=6){ death('splash'); return; }
    }
    if(!buffer.length&&held.length)buffer.push(held[held.length-1]);
    const d=buffer[0];
    if(d){ const v=DVEC[d];
      if(inBounds(s.cx+v.x,s.cy+v.y))doHop(v);
      buffer.shift(); }
  }
  if(!s.alive||G.mode!=='play')return;
  // hop progression
  s.phase+=dt*(s.hopT<1?16:6);
  if(s.hopT<1){
    s.hopT+=dt/HOP_MS;
    const k=Math.min(1,s.hopT), e=k*(2-k);
    s.px=lerp(s.fx,s.tx,e);
    s.py=lerp(s.fy,s.ty,e)-Math.sin(e*Math.PI)*0.14;
    if(s.hopT>=1){ s.hopT=1; s.px=s.tx; s.py=s.ty;
      s.cx=Math.round(s.tx-0.5); s.cy=Math.round(s.ty-0.5); commitMove(); }
  }
  if(!s.alive||G.mode!=='play')return;
  // car collisions (pixel-level, works mid-hop too)
  for(const v of vehicles){ if(v.type==='log')continue;
    if(s.px>v.x+v.inset-0.28 && s.px<v.x+v.len-v.inset+0.28 && s.py>v.row+0.16-0.28 && s.py<v.row+0.84+0.28){ death('squish'); return; }
  }
  checkEat();
}

// ---------- food ----------
function spawnFood(){
  const maxF=3+Math.min(2,Math.floor(G.level/2));
  if(foods.length>=maxF)return;
  for(let tries=0;tries<6;tries++){
    if(Math.random()<0.4){ // fish riding a log
      const r=1+Math.floor(Math.random()*6);
      const logs=vehicles.filter(v=>v.row===r&&v.x>-v.len+1&&v.x<COLS-1);
      if(!logs.length)continue;
      const v=logs[Math.floor(Math.random()*logs.length)];
      if(v.len<2.5)continue;
      const x=rand(v.x+0.7,v.x+v.len-0.7);
      if(snake&&Math.abs(x-snake.px)<1.2)continue;
      foods.push({type:'fish',lane:r,x,y:r+0.5,life:8,max:8,seed:Math.random()*9,grow:2,pts:100*G.level,dying:false});
    }else{ // prey frog on road / median
      const r=Math.random()<0.3?7:8+Math.floor(Math.random()*6);
      const x=1+Math.floor(Math.random()*(COLS-2))+0.5;
      if(snake&&Math.abs(x-snake.px)<1.4&&r===snake.cy)continue;
      foods.push({type:'frog',lane:r,x,y:r+0.5,life:9,max:9,seed:Math.random()*9,grow:1,pts:50*G.level,dying:false});
    }
    return;
  }
}
function updateFoods(dt){
  const interval=Math.max(1.1,2.4-G.level*0.12);
  foodTimer-=dt; if(foodTimer<=0){ spawnFood(); foodTimer=interval; }
  for(let i=foods.length-1;i>=0;i--){ const f=foods[i]; f.life-=dt;
    if(f.type==='fish'){
      let sup=null;
      for(const v of vehicles)if(v.row===f.lane&&f.x>=v.x+0.1&&f.x<=v.x+v.len-0.1){sup=v;break;}
      if(sup)f.x+=sup.vx*spdMul()*dt; else f.dying=true;
    }
    if(f.life<=0||f.dying){ if(f.dying)parts.push({shape:'ring',x:ox+f.x*cell,y:oy+f.y*cell,life:.4,max:.4,maxR:cell*.8}); foods.splice(i,1); }
  }
}

// ---------- world update ----------
function updateWorld(dt,rdt){
  // vehicles
  const m=spdMul();
  for(const v of vehicles){
    v.x+=v.vx*m*dt;
    if(v.vx>0&&v.x>COLS+0.5)v.x-=v.cycle;
    else if(v.vx<0&&v.x<-(v.len+1.5))v.x+=v.cycle;
    const nowIn=v.x>-1&&v.x<COLS+1;
    if(v.type==='bike'&&nowIn&&!v.wasIn)SFX.whoosh();
    v.wasIn=nowIn;
  }
  // turtles
  for(const t of turtles){
    t.y+=t.vy*m*dt; t.a=Math.min(1,t.a+dt*2.5);
    if(t.y<=1.05){t.y=6.75;t.a=0;} else if(t.y>=6.75){t.y=1.05;t.a=0;}
  }
  // fireflies drift (unscaled ambience)
  cricketT-=rdt; if(cricketT<=0){ SFX.chirp(); cricketT=rand(.6,2.0); }

  if(G.mode==='play'){
    G.timeLeft-=dt;
    if(G.timeLeft<=0){ G.timeLeft=0; death('time'); }
    const ti=Math.ceil(G.timeLeft);
    if(G.timeLeft<10&&ti!==lastTickInt&&G.timeLeft>0){ lastTickInt=ti; SFX.tick(); }
    updateFoods(dt);
    if(snake.alive)updateSnake(dt);
  }else if(G.mode==='dead'){
    deathT-=rdt;
    if(deathT<=0){
      if(G.lives>0){ snake=newSnake(); buffer.length=0; G.mode='play'; TS=1; tsT=0;
        texts.push({x:ox+7.5*cell,y:oy+13*cell,str:'GO!',life:.9,max:.9,color:'#6fe05a',size:26}); }
      else{ G.mode='over'; saveBest(G.best); SFX.over(); }
    }
  }else if(G.mode==='clear'){
    clearT+=dt; if(clearT>2.4)nextLevel();
  }
  // fx (unscaled so drama timers feel right)
  for(let i=parts.length-1;i>=0;i--){ const p=parts[i]; p.life-=rdt;
    if(p.life<=0){parts.splice(i,1);continue;}
    if(p.shape!=='ring'){ p.x+=p.vx*rdt; p.y+=p.vy*rdt; p.vy+=(p.g||0)*rdt; p.rot=(p.rot||0)+(p.vr||0)*rdt; } }
  for(let i=texts.length-1;i>=0;i--){ const t=texts[i]; t.life-=rdt; t.y-=20*rdt; if(t.life<=0)texts.splice(i,1); }
  flash.a=Math.max(0,flash.a-2.4*rdt);
  shake=Math.max(0,shake-shake*7*rdt-2*rdt);
  scorePop=Math.max(0,scorePop-rdt);
  if(tsT>0){ tsT-=rdt; if(tsT<=0)TS=1; }
}

function pressDir(dk,repeat){
  if(G.mode==='start'||G.mode==='over')startGame();
  if(!held.includes(dk))held.push(dk);
  if(!repeat){ buffer.push(dk); if(buffer.length>3)buffer.shift(); }

  const k=e.key;
  if(['ArrowUp','ArrowDown','ArrowLeft','ArrowRight',' '].includes(k))e.preventDefault();
  initAudio();
  const dk=keyDir(k);
  if(dk){ pressDir(dk,e.repeat); return; }
  if(k==='Enter'){ if(G.mode==='start'||G.mode==='over')startGame(); }
  else if(k===' '){
    if(G.mode==='play'||G.mode==='pause')togglePause();
    else if(G.mode==='start'||G.mode==='over')startGame();
  }
  else if(k==='p'||k==='P'){ if(G.mode==='play'||G.mode==='pause')togglePause(); }
  else if(k==='m'||k==='M')toggleMute();
  else if(k==='r'||k==='R'){ if(G.mode!=='start')startGame(); }
});
