function trace(x0,y0,ang,pierce){ const dx=Math.cos(ang),dy=Math.sin(ang),hits=[]; let endT=1e9,blocked=false;
 const maxLen=(COLS+ROWS)*cell*1.6;
 outer: for(let t=4;t<maxLen;t+=3){ const X=x0+dx*t,Y=y0+dy*t;
  const gx=Math.floor((X-ox)/cell),gy=Math.floor((Y-oy)/cell);
  if(gx<0||gy<0||gx>=COLS||gy>=ROWS){endT=t;break}
  if(astSet.has(gx+','+gy)){endT=t;blocked=true;break}
  for(const s of ships){ if(!s.alive)continue; const ddx=s.px-X,ddy=s.py-Y,rr=s.r*cell*.92;
   if(ddx*ddx+ddy*ddy<rr*rr){ hits.push({t,ship:s}); if(!pierce)break outer; } } }
 return {endT,hits,blocked}; }
function los(ax,ay,ar,bx,by,br){ const dx=bx-ax,dy=by-ay,d=Math.hypot(dx,dy); if(d<1)return true;
 const ang=Math.atan2(dy,dx),off=ar+6,sx=ax+Math.cos(ang)*off,sy=ay+Math.sin(ang)*off;
 const tr=trace(sx,sy,ang,false),lim=d-off-br*.9; for(const h of tr.hits) if(h.t<lim)return false; return true; }
function canOccupy(s,nx,ny){ return nx>=0&&ny>=0&&nx<COLS&&ny<ROWS&&!astSet.has(nx+','+ny)&&
 !ships.some(o=>o.alive&&o!==s&&o.x===nx&&o.y===ny); }

 for(let i=0;i<cnt;i++){ const a=cnt>1?ang+(i-(cnt-1)/2)*spr:ang;
  const tr=trace(sx,sy,a,pierce); let endT=tr.endT; const queue=[];
  for(const h of tr.hits){ queue.push({t:h.t,ship:h.ship,dmg:wpn.dmg}); if(!pierce){endT=h.t;break;} }
  bolts.push({ox:sx,oy:sy,dx:Math.cos(a),dy:Math.sin(a),t:0,speed:wpn.speed,endT,color:wpn.color,w:wpn.w||3,pierce,queue,trail:[],blocked:tr.blocked,team:s.team});
  if(s.team==='p')shots++;
  for(let j=0;j<6;j++){ const va=a+(Math.random()-.5)*.9,vs=80+Math.random()*160;
   parts.push({x:sx,y:sy,vx:Math.cos(va)*vs,vy:Math.sin(va)*vs,life:.14+Math.random()*.12,max:.26,size:1+Math.random()*1.6,col:wpn.color}); } }

   let best=null,bs=1e9;
   for(const[dx,dy]of[[1,0],[-1,0],[0,1],[0,-1]]){ const nx=sx+dx,ny=sy+dy,key=nx+','+ny;
    if(!canOccupy(e,nx,ny)||reserved.has(key))continue;
    let sc=Math.abs(nx-player.x)+Math.abs(ny-player.y);
    if(!los(cx(nx),cy(ny),e.r*cell,cx(player.x),cy(player.y),player.r*cell))sc+=3;
    if(sc<bs){bs=sc;best=[dx,dy,key];} }
   if(!best)break; script.push({type:'move',ship:e,dx:best[0],dy:best[1]}); reserved.add(best[2]); sx+=best[0];sy+=best[1]; }
