/* ================= physics ================= */
function collideSeg(b,s,e){
  const dx=s.x2-s.x1,dy=s.y2-s.y1,L2=dx*dx+dy*dy||1;
  let t=((b.x-s.x1)*dx+(b.y-s.y1)*dy)/L2;t=t<0?0:t>1?1:t;
  const cx=s.x1+dx*t,cy=s.y1+dy*t;
  let nx=b.x-cx,ny=b.y-cy,d=Math.hypot(nx,ny);
  if(d>=b.r)return null;
  if(d<1e-4){nx=0;ny=-1;d=1;}else{nx/=d;ny/=d;}
  b.x=cx+nx*b.r;b.y=cy+ny*b.r;
  const vn=b.vx*nx+b.vy*ny;
  if(vn<0){b.vx-=(1+e)*vn*nx;b.vy-=(1+e)*vn*ny;}
  return{nx,ny,vn};
}
function collideFlip(f){
  const tx=f.px+f.len*Math.cos(f.ang),ty=f.py+f.len*Math.sin(f.ang);
  const dx=tx-f.px,dy=ty-f.py,L2=dx*dx+dy*dy;
  let t=((ball.x-f.px)*dx+(ball.y-f.py)*dy)/L2;t=t<0?0:t>1?1:t;
  const cx=f.px+dx*t,cy=f.py+dy*t;
  let nx=ball.x-cx,ny=ball.y-cy,d=Math.hypot(nx,ny),rr=ball.r+f.r;
  if(d>=rr)return;
  if(d<1e-4){nx=0;ny=-1;d=1;}else{nx/=d;ny/=d;}
  ball.x=cx+nx*rr;ball.y=cy+ny*rr;
  const rx=cx-f.px,ry=cy-f.py,svx=-f.av*ry,svy=f.av*rx;
  const rvx=ball.vx-svx,rvy=ball.vy-svy,vn=rvx*nx+rvy*ny;
  if(vn<0){const e=.55;ball.vx-=(1+e)*vn*nx;ball.vy-=(1+e)*vn*ny;}
}
function updFlip(f,pressed){
  const target=pressed?f.up:f.rest;
  const spd=pressed?Math.min(34,24+1.6*G.up.flipper):30;
  const d=target-f.ang,st=spd*STEP;
  const na=d>0?Math.min(target,f.ang+st):Math.max(target,f.ang-st);
  f.av=(na-f.ang)/STEP;f.ang=na;
}
function stepBall(dt){
  ball.vy+=GRAV*dt;
  if(G.up.magnet>0){
    const dx=240-ball.x,dy=380-ball.y,d=Math.hypot(dx,dy);
    if(d>2&&d<360){const a=(70+55*G.up.magnet)*(1-d/420);ball.vx+=dx/d*a*dt;ball.vy+=dy/d*a*dt;}
  }
  const sp=Math.hypot(ball.vx,ball.vy);
  if(sp>2000){ball.vx*=2000/sp;ball.vy*=2000/sp;}
  ball.x+=ball.vx*dt;ball.y+=ball.vy*dt;
  for(const s of WALLS)collideSeg(ball,s,s.e);
  for(const e of SLEDGES){
    const hit=collideSeg(ball,e,.45);
    if(hit&&e.kick){
      const sl=SLINGDEFS[e.sling];
      if(sl.cd<=0){sl.cd=.12;sl.flash=1;
        const vt=(ball.vx*-hit.ny)+(ball.vy*hit.nx);
        ball.vx=hit.nx*460+(-hit.ny)*vt*.3;
        ball.vy=hit.ny*460+(hit.nx)*vt*.3;
        addScore(25,e.x1,e.y1-10);sfx.sling();
      }
    }
  }
  for(const t of TARGETS){
    const cx=clamp(ball.x,t.x,t.x+t.w),cy=clamp(ball.y,t.y,t.y+t.h);
    let nx=ball.x-cx,ny=ball.y-cy,d=Math.hypot(nx,ny);
    if(d<ball.r){
      if(d<1e-4){nx=-1;ny=0;d=1;}else{nx/=d;ny/=d;}
      ball.x=cx+nx*ball.r;ball.y=cy+ny*ball.r;
      const vn=ball.vx*nx+ball.vy*ny;
      if(vn<0){ball.vx-=1.6*vn*nx;ball.vy-=1.6*vn*ny;}
      if(!t.lit&&Math.abs(vn)>40){
        t.lit=true;addScore(250,t.x+8,t.y-8);sfx.target();
        for(let i=0;i<7;i++)parts.push({x:cx,y:cy,vx:rand(-160,160),vy:rand(-200,40),t:0,dur:.5,col:'#3fe0c5',r:rand(1.5,3)});
        if(TARGETS.every(q=>q.lit)){G.targetResetT=1.4;addScore(2000,240,380);sfx.bonus();}
      }
    }
  }
  for(let i=0;i<BUMPERS.length;i++){
    const b=BUMPERS[i];
    let nx=ball.x-b.x,ny=ball.y-b.y,d=Math.hypot(nx,ny),rr=b.r+ball.r;
    if(d<rr){
      if(d<1e-4){nx=0;ny=-1;d=1;}else{nx/=d;ny/=d;}
      ball.x=b.x+nx*rr;ball.y=b.y+ny*rr;
      const f=(520+28*G.up.bumper)*rand(.95,1.05);
      ball.vx=nx*f;ball.vy=ny*f;
      b.flash=1;addScore(75+25*G.up.bumper,b.x-nx*b.r-8,b.y-ny*b.r-8);
      G.stats.bumps++;sfx.bump();G.shake=Math.min(.4,G.shake+.06);
      for(let j=0;j<8;j++)parts.push({x:b.x+nx*b.r,y:b.y+ny*b.r,vx:rand(-220,220),vy:rand(-260,60),t:0,dur:.45,col:b.col,r:rand(1.5,3)});
    }
  }
  collideFlip(FL.L);collideFlip(FL.R);
  if(ball.x>441&&ball.y>700){G.state='ready';banner('BALL RETURNED','hold to charge');}
  else if(ball.y>H+24)drain();
}
function tick(dt){
  G.t+=dt;
  updFlip(FL.L,K.left);updFlip(FL.R,K.right);
  G.shake=Math.max(0,G.shake-dt*2.4);G.zapCd-=dt;G.nudgeCd-=dt;
  G.comboT-=dt;if(G.comboT<=0)G.combo=0;
  BUMPERS.forEach(b=>b.flash=Math.max(0,b.flash-dt*3));
  SLINGDEFS.forEach(s=>{s.flash=Math.max(0,s.flash-dt*4);s.cd-=dt;});
  if(G.targetResetT>0){G.targetResetT-=dt;if(G.targetResetT<=0)TARGETS.forEach(t=>t.lit=false);}
  for(let i=parts.length-1;i>=0;i--){const p=parts[i];p.t+=dt;p.x+=p.vx*dt;p.y+=p.vy*dt;p.vy+=700*dt;if(p.t>p.dur)parts.splice(i,1);}
  for(let i=rings.length-1;i>=0;i--){rings[i].t+=dt;if(rings[i].t>rings[i].dur)rings.splice(i,1);}
  for(let i=popups.length-1;i>=0;i--){const p=popups[i];p.t+=dt;p.y-=45*dt;if(p.t>p.dur)popups.splice(i,1);}
  if(G.charging&&G.state==='ready'){
    G.charge=clamp(G.charge+dt/.85,0,1);
    if(chOsc)chOsc.frequency.value=80+G.charge*540;
  }
  switch(G.state){
    case 'play':
      stepBall(dt);
      ball.trail.push({x:ball.x,y:ball.y});if(ball.trail.length>13)ball.trail.shift();
      break;
    case 'ready':ball.x=453;ball.y=707;ball.vx=0;ball.vy=0;break;
    case 'dead':G.deadT-=dt;if(G.deadT<=0){if(G.balls>0)spawnBall();else gameOver();}break;
  }
}
