function updFlips(dt){for(const f of flips){const t=flipTarget(f,f.pressed),d=t-f.a,st=FL.om*dt;if(Math.abs(d)<=st){f.om=(t-f.a)/dt;f.a=t;}else{f.om=Math.sign(d)*FL.om;f.a+=Math.sign(d)*st;}}}
function hitSeg(s){const dx=s.x2-s.x1,dy=s.y2-s.y1,L2=dx*dx+dy*dy;let t=((ball.x-s.x1)*dx+(ball.y-s.y1)*dy)/L2;t=t<0?0:t>1?1:t;const cx=s.x1+t*dx,cy=s.y1+t*dy;let nx=ball.x-cx,ny=ball.y-cy;const d2=nx*nx+ny*ny;if(d2>=BR*BR||d2<1e-9)return false;const d=Math.sqrt(d2);nx/=d;ny/=d;ball.x=cx+nx*BR;ball.y=cy+ny*BR;const vn=ball.vx*nx+ball.vy*ny;s._vn=vn;if(vn<0){const j=(1+s.e)*vn;ball.vx-=j*nx;ball.vy-=j*ny;}if(s.kick&&vn<-60){const vn2=ball.vx*nx+ball.vy*ny;if(vn2<s.kick){ball.vx+=nx*(s.kick-vn2);ball.vy+=ny*(s.kick-vn2);}}return vn<0;}
function circ(x,y,R){let nx=ball.x-x,ny=ball.y-y;const d2=nx*nx+ny*ny;if(d2>=R*R||d2<1e-9)return null;const d=Math.sqrt(d2);nx/=d;ny/=d;ball.x=x+nx*R;ball.y=y+ny*R;return{nx,ny,vn:ball.vx*nx+ball.vy*ny};}
function bumperHit(b,c,big,idx){if(c.vn<0){const j=1.35*c.vn;ball.vx-=j*c.nx;ball.vy-=j*c.ny;}if(-c.vn>120){const K=big?640:520,vn2=ball.vx*c.nx+ball.vy*c.ny;if(vn2<K){ball.vx+=c.nx*(K-vn2);ball.vy+=c.ny*(K-vn2);}}if(-c.vn>60){b.f=1;S.bump(idx);shake=Math.max(shake,big?3:1.6);goo(ball.x,ball.y,big?8:4,'#7dff5e');bumpCombo();addScore(big?100:50,ball.x,ball.y-b.r-8,big?'#8dff6e':'#a5e88a',big);addPower(big?9:6);if(big&&stormLit)doStorm();}}
function dropBrain(b){b.alive=false;goo(b.x,b.y,14,'#ff8fb0');S.brain();bumpCombo();addScore(500,b.x,b.y-20,'#ff8fb0',true);addPower(12);if(!brains.some(x=>x.alive)){score+=3000;addPop(230,430,'BRAIN FEAST +3000','#ff8fb0',16);bankT=2.5;syncHUD();}}
function flipHit(f){const tx=f.px+Math.cos(f.a)*FL.L,ty=f.py+Math.sin(f.a)*FL.L,dx=tx-f.px,dy=ty-f.py,L2=dx*dx+dy*dy;let t=clamp(((ball.x-f.px)*dx+(ball.y-f.py)*dy)/L2,0,1);const cx=f.px+t*dx,cy=f.py+t*dy;let nx=ball.x-cx,ny=ball.y-cy;const R=BR+FL.wid,d2=nx*nx+ny*ny;if(d2>=R*R||d2<1e-9)return;const d=Math.sqrt(d2);nx/=d;ny/=d;ball.x=cx+nx*R;ball.y=cy+ny*R;const rx=cx-f.px,ry=cy-f.py,cvx=-f.om*ry,cvy=f.om*rx;let rvx=ball.vx-cvx,rvy=ball.vy-cvy;const vn=rvx*nx+rvy*ny;if(vn<0){const j=1.15*vn;rvx-=j*nx;rvy-=j*ny;ball.vx=rvx+cvx;ball.vy=rvy+cvy;}const sp=Math.hypot(ball.vx,ball.vy);if(sp>MAXV){ball.vx*=MAXV/sp;ball.vy*=MAXV/sp;}}
function physics(h){updFlips(h);
 ball.vy+=GRAV*h;let sp=Math.hypot(ball.vx,ball.vy);if(sp>MAXV){const k=MAXV/sp;ball.vx*=k;ball.vy*=k;}
 ball.x+=ball.vx*h;ball.y+=ball.vy*h;
 for(const s of walls)hitSeg(s);
 for(const sl of slings){sl.cool-=h;let fh=false;for(const e of sl.edges){if(hitSeg(e)&&e===sl.face)fh=true;}if(fh&&sl.face._vn<-60&&sl.cool<=0){sl.flash=1;sl.cool=.18;S.sling();shake=Math.max(shake,2);bumpCombo();addScore(25,(sl.pts[0][0]+sl.pts[1][0])/2,(sl.pts[0][1]+sl.pts[1][1])/2-14,'#ffd23d');addPower(5);sparks((sl.pts[0][0]+sl.pts[1][0])/2,(sl.pts[0][1]+sl.pts[1][1])/2,6);}}

 minis.forEach((b,i)=>{const c=circ(b.x,b.y,BR+b.r);if(c)bumperHit(b,c,false,i+3);});
 for(const w of walkers){if(!w.alive)continue;const c=circ(w.x,w.y,BR+w.r);if(c&&-c.vn>40){if(c.vn<0){const j=1.5*c.vn;ball.vx-=j*c.nx;ball.vy-=j*c.ny;}const vn2=ball.vx*c.nx+ball.vy*c.ny;if(vn2<460){ball.vx+=c.nx*(460-vn2);ball.vy+=c.ny*(460-vn2);}w.f=1;S.bonk();shake=Math.max(shake,2.5);bumpCombo();addScore(150,w.x,w.y-24,'#a5e88a');addPower(7);goo(ball.x,ball.y,5,'#7dff5e');}}
 for(const b of brains){if(b.alive&&b.popT<0){const dx=ball.x-b.x,dy=ball.y-b.y;if(dx*dx+dy*dy<(BR+b.r)*(BR+b.r))dropBrain(b);}}
 flipHit(flips[0]);flipHit(flips[1]);
 sp=Math.hypot(ball.vx,ball.vy);if(sp>MAXV){const k=MAXV/sp;ball.vx*=k;ball.vy*=k;}
}

function tick(dt){tGlob+=dt;nudgeCd-=dt;
 for(const f of fog){f.x+=f.s*dt;if(f.x>W+200)f.x=-200;if(f.x<-200)f.x=W+200;}

 for(let i=bats.length-1;i>=0;i--){bats[i].x+=bats[i].vx*dt;if(bats[i].x<-40||bats[i].x>W+40)bats.splice(i,1);}
 for(let i=parts.length-1;i>=0;i--){const p=parts[i];p.life-=dt;if(p.life<=0){parts.splice(i,1);continue;}p.vy+=p.g*dt;p.x+=p.vx*dt;p.y+=p.vy*dt;}
 for(let i=pops.length-1;i>=0;i--){pops[i].t+=dt;if(pops[i].t>.95)pops.splice(i,1);}
 for(let i=msgs.length-1;i>=0;i--){msgs[i].t+=dt;if(msgs[i].t>msgs[i].dur)msgs.splice(i,1);}
 heads.forEach(b=>b.f=Math.max(0,b.f-dt*3.2));minis.forEach(b=>b.f=Math.max(0,b.f-dt*3.2));walkers.forEach(w=>w.f=Math.max(0,w.f-dt*4));slings.forEach(s=>s.flash=Math.max(0,s.flash-dt*5));
 shake=Math.max(0,shake-dt*26);shx=rnd(-1,1)*shake;shy=rnd(-1,1)*shake;stormFlash=Math.max(0,stormFlash-dt*1.5);
 comboT-=dt;if(comboT<=0)combo=0;
 if(bankT>0){bankT-=dt;if(bankT<=0)resetBrains();}
 brains.forEach(b=>{if(!b.alive&&b.popT>0){b.popT-=dt;if(b.popT<=0)dropBrain(b);}else if(b.alive&&b.scale<1)b.scale=Math.min(1,b.scale+dt*3);});
 if(state!=='pause'){

  const sp=Math.hypot(ball.vx,ball.vy);
  trail.push({x:ball.x,y:ball.y});if(trail.length>11)trail.shift();
  if(sp<25)lowT+=dt;else lowT=0;
  if(lowT>2.4){ball.vy-=rnd(140,240);ball.vx+=rnd(-90,90);lowT=0;}
  if(ball.x>466&&ball.y>712&&sp<70){setBallInLane();state='plunge';acc=0;charging=false;}

 }else if(state!=='pause'){updFlips(dt);
  if(state==='plunge'){const want=keys.Space||keys.ArrowDown||keys.KeyS||ptrMode('P');

  }else if(state==='drain'){ball.vy+=GRAV*dt;ball.x+=ball.vx*dt;ball.y+=ball.vy*dt;drainT-=dt;if(drainT<=0)nextBall();}
 }
 syncHUD(false);
}
