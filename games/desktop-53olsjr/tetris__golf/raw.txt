function spawnShards(c,r,ci){for(let i=0;i<3;i++)particles.push({type:'shard',x:c+.5,y:r+.5,vx:(Math.random()-.5)*9,vy:4+Math.random()*8,
  rot:Math.random()*6.28,vr:(Math.random()-.5)*14,sz:.12+Math.random()*.1,col:COLORS[ci],life:.7,max:.7});}
function spawnPuff(x,y,n){for(let i=0;i<n;i++)particles.push({type:'puff',x,y,vx:(Math.random()-.5)*3,vy:(Math.random()-.5)*3,life:.35,max:.35,r0:.08+Math.random()*.06});}
function spawnConfetti(){const cols=['#ffd23f','#ffffff','#5ce06b','#ff5d5d','#45d8f0'];
 for(let i=0;i<80;i++)particles.push({type:'conf',x:cupX()+(Math.random()-.5)*1.4,y:cupY()+.2,vx:(Math.random()-.5)*11,vy:3+Math.random()*9,
  rot:Math.random()*6.28,vr:(Math.random()-.5)*16,sz:.09+Math.random()*.07,col:cols[i%5],life:1.4+Math.random()*.8,max:2,seed:Math.random()*9});}
function updFx(dt){
  for(let i=particles.length-1;i>=0;i--){const p=particles[i];p.life-=dt;
    if(p.life<=0){particles.splice(i,1);continue;}
    if(p.type==='shard'){p.vy-=38*dt;p.x+=p.vx*dt;p.y+=p.vy*dt;p.rot+=p.vr*dt;}
    else if(p.type==='puff'){p.x+=p.vx*dt;p.y+=p.vy*dt;}
    else{p.vy-=15*dt;p.x+=(p.vx+Math.sin(T*7+p.seed)*2.4)*dt;p.y+=p.vy*dt;p.rot+=p.vr*dt;}}
  if(particles.length>420)particles.splice(0,particles.length-420);
  for(let i=floaters.length-1;i>=0;i--){floaters[i].t+=dt;if(floaters[i].t>floaters[i].dur)floaters.splice(i,1);}
  for(let i=clearFx.length-1;i>=0;i--){clearFx[i].t+=dt;if(clearFx[i].t>.32)clearFx.splice(i,1);}
  shake=Math.max(0,shake-dt*16);
}

function drawAim(){

    if(i%3===0){const px=ox+s.x*cell,py2=oy+(GH-s.y)*cell;if(py2>oy+GH*cell)break;

}

        aimAng=Math.atan2(by-mouse.y,mouse.x-bx);}

        if(keys.ArrowLeft)aimAng+=rate*dt;if(keys.ArrowRight)aimAng-=rate*dt;}

    }

  // render

  drawSky(state==='paused'?0:dt);

  drawBoard();
  if(state!=='menu'){drawCup();drawBall();if(state==='playing')drawAim();}
  drawFx();
  ctx.restore();
}
