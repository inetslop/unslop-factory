let AC=null, master=null, noiseBuf=null, muted=false;
let engOsc1=null,engOsc2=null,engGain=null;
function audioInit(){
  if(AC){ if(AC.state==='suspended')AC.resume(); return; }
  try{
    AC=new (window.AudioContext||window.webkitAudioContext)();
    master=AC.createGain(); master.gain.value=muted?0:0.85; master.connect(AC.destination);
    const n=AC.sampleRate|0; noiseBuf=AC.createBuffer(1,n,AC.sampleRate);
    const d=noiseBuf.getChannelData(0); for(let i=0;i<n;i++)d[i]=Math.random()*2-1;
    engGain=AC.createGain(); engGain.gain.value=0;
    const f=AC.createBiquadFilter(); f.type='lowpass'; f.frequency.value=520;
    engOsc1=AC.createOscillator(); engOsc1.type='sawtooth'; engOsc1.frequency.value=60;
    engOsc2=AC.createOscillator(); engOsc2.type='square'; engOsc2.frequency.value=30;
    const g2=AC.createGain(); g2.gain.value=0.4;
    engOsc1.connect(f); engOsc2.connect(g2); g2.connect(f); f.connect(engGain); engGain.connect(master);
    engOsc1.start(); engOsc2.start();
  }catch(e){ AC=null; }
}
function tone(freq,dur,type,vol,slide,delay){ if(!AC)return; type=type||'square'; vol=vol||.2; slide=slide||0; delay=delay||0;
  const t0=AC.currentTime+delay, o=AC.createOscillator(), g=AC.createGain();
  o.type=type; o.frequency.setValueAtTime(freq,t0);
  if(slide) o.frequency.exponentialRampToValueAtTime(Math.max(30,freq+slide),t0+dur);
  g.gain.setValueAtTime(vol,t0); g.gain.exponentialRampToValueAtTime(.001,t0+dur);
  o.connect(g); g.connect(master); o.start(t0); o.stop(t0+dur+.03);
}
function noiseHit(dur,vol,freq,delay){ if(!AC||!noiseBuf)return; delay=delay||0;
  const t0=AC.currentTime+delay, s=AC.createBufferSource(); s.buffer=noiseBuf; s.loop=true;
  const f=AC.createBiquadFilter(); f.type='lowpass';
  f.frequency.setValueAtTime(freq,t0); f.frequency.exponentialRampToValueAtTime(Math.max(60,freq*.25),t0+dur);
  const g=AC.createGain(); g.gain.setValueAtTime(vol,t0); g.gain.exponentialRampToValueAtTime(.001,t0+dur);
  s.connect(f); f.connect(g); g.connect(master); s.start(t0); s.stop(t0+dur+.05);
}

  tick(){tone(1150,.05,'square',.09)},

/* ============ update ============ */
function bump(){ car.v*=-0.28; shake=Math.max(shake,4);
  spawnParts(clamp(car.x,.3,COLS-.3),clamp(car.y,.3,ROWS-.3),5,['#ffd23f','#f4f1e8'],2,.3,.06,0);
  const s=performance.now()/1000; if(s-lastBumpSnd>.15){ lastBumpSnd=s; sfx.bump(); } }

function update(dt){
  if(paused) return;
  gt+=dt; shake*=Math.exp(-6*dt); if(shake<.3)shake=0; wrongCd=Math.max(0,wrongCd-dt);
  for(let i=parts.length-1;i>=0;i--){ const p=parts[i]; p.life-=dt; if(p.life<=0){parts.splice(i,1);continue;}
    p.x+=p.vx*dt; p.y+=p.vy*dt; p.vy+=(p.grav||0)*dt; p.size+= (p.grow||0)*dt; }
  for(let i=floats.length-1;i>=0;i--){ const f=floats[i]; f.t+=dt; if(f.t>.9)floats.splice(i,1); }
  for(let i=trail.length-1;i>=0;i--){ trail[i].a-=dt*.12; if(trail[i].a<=0)trail.splice(i,1); }

  if(state==='count'){
    stateT+=dt;
    const n=Math.ceil(3-stateT);
    if(n!==update._last && n>0 && n<=3){ update._last=n; sfx.count(); }
    if(stateT>=3){ update._last=0; state='play'; sfx.go(); }
    return;
  }
  if(state==='end'){
    stateT+=dt;
    if(endReason==='mine' && stateT<1.4 && Math.random()<.5)
      parts.push({x:car.x+(Math.random()-.5)*.7,y:car.y+(Math.random()-.5)*.7,vx:(Math.random()-.5)*.6,vy:-.8-Math.random()*.6,
        life:1.1,max:1.1,size:.18,col:'#6a6f75',grav:0,grow:.45});
    const delay = endReason==='mine'?1.35 : endReason==='time'?.75 : 1.0;
    if(!uiShown && stateT>delay) showEndUI();
    return;
  }
  if(state!=='play') return;

  raceTime+=dt; timeLeft-=dt;
  if(timeLeft<=0){ timeLeft=0; endRace('time'); return; }
  const s=Math.ceil(timeLeft); if(timeLeft<5 && s!==lowTick){ lowTick=s; sfx.tick(); }

  /* physics */
  const up=keys.has('ArrowUp')||keys.has('KeyW'), dn=keys.has('ArrowDown')||keys.has('KeyS');
  const lf=keys.has('ArrowLeft')||keys.has('KeyA'), rt=keys.has('ArrowRight')||keys.has('KeyD');
  if(up) car.v+=9*dt; else if(dn) car.v-=11*dt; else car.v*=Math.max(0,1-2.4*dt);
  car.v=clamp(car.v,-1.9,5.4);
  const steer=(rt?1:0)-(lf?1:0);
  car.steer+=(steer-car.steer)*Math.min(1,dt*12);
  if(Math.abs(car.v)>.05) car.ang+=steer*3.6*Math.min(1,Math.abs(car.v)/1.4)*dt*Math.sign(car.v);
  car.x+=Math.cos(car.ang)*car.v*dt; car.y+=Math.sin(car.ang)*car.v*dt;
  const R=0.32;
  if(car.x<R){car.x=R;bump();} if(car.x>COLS-R){car.x=COLS-R;bump();}
  if(car.y<R){car.y=R;bump();} if(car.y>ROWS-R){car.y=ROWS-R;bump();}

  /* skid marks */
  const skidding=(Math.abs(steer)>0&&Math.abs(car.v)>2.4)||(dn&&car.v>2);
  if(skidding){ const dx=Math.cos(car.ang),dy=Math.sin(car.ang),px=-dy,py=dx;
    trail.push({x:car.x-dx*.18+px*.13,y:car.y-dy*.18+py*.13,a:.45},{x:car.x-dx*.18-px*.13,y:car.y-dy*.18-py*.13,a:.45});
    if(trail.length>600) trail.splice(0,trail.length-600); }

  markVisited();

  /* mine hit */
  const cc=Math.floor(car.x), cr=Math.floor(car.y);
  if(mineSet.has(cc+','+cr)){ doBoom(); return; }

  /* gates */
  for(let i=0;i<gates.length;i++){ const g=gates[i];
    if(g.c!==cc||g.r!==cr) continue;
    if(i===gateIdx){ g.hit=true; gateIdx++; timeLeft+=GATE_BONUS; sfx.gate(); shake=Math.max(shake,3);
      floats.push({x:g.c+.5,y:g.r+.1,text:'+'+GATE_BONUS+'s',col:'#58c04a',t:0});
      spawnParts(g.c+.5,g.r+.5,16,['#ffd23f','#fff','#4f8fe2','#e23b2e','#58c04a'],3.2,.8,.06,5);
      if(gateIdx>=gates.length){ startWin(); return; }
    } else if(!g.hit && wrongCd<=0){ timeLeft=Math.max(.1,timeLeft-PENALTY); wrongCd=1; g.flash=gt; sfx.bad();
      floats.push({x:g.c+.5,y:g.r+.1,text:'-'+PENALTY+'s',col:'#ff4b3e',t:0}); }
  }
}

function drawField(){
  for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++){
    const k=c+','+r;
    if(visited.has(k)) continue;
    ctx.fillStyle=(c+r)%2?'#4f9e3f':'#58a947';
    ctx.fillRect(ox+c*cell,oy+r*cell,cell+.5,cell+.5);
  }
  ctx.fillStyle='rgba(20,60,20,.35)';
  for(const s of speckles){ ctx.beginPath(); ctx.arc(ox+s.x*cell,oy+s.y*cell,cell*.05,0,7); ctx.fill(); }
  /* start flag */
  const fx=ox+1.4*cell, fy=oy+5.5*cell;
  ctx.strokeStyle='#e8e6da'; ctx.lineWidth=2; ctx.beginPath(); ctx.moveTo(fx,fy); ctx.lineTo(fx,fy-cell*.55); ctx.stroke();
  ctx.fillStyle=patSmall; ctx.fillRect(fx,fy-cell*.55,cell*.34,cell*.22);
  ctx.strokeStyle='#14161a'; ctx.lineWidth=1.5; ctx.strokeRect(fx,fy-cell*.55,cell*.34,cell*.22);
  /* skid marks */
  for(const t of trail){ ctx.fillStyle='rgba(40,55,28,'+t.a+')';
    ctx.beginPath(); ctx.arc(ox+t.x*cell,oy+t.y*cell,cell*.08,0,7); ctx.fill(); }
  /* revealed minesweeper tiles */
  for(const k of visited){ const [c,r]=k.split(',').map(Number);
    const x=ox+c*cell+1.5, y=oy+r*cell+1.5, s=cell-3;
    ctx.fillStyle='#d7d4c9'; ctx.fillRect(x,y,s,s);
    ctx.fillStyle='#f3f1e8'; ctx.fillRect(x,y,s,2.5); ctx.fillRect(x,y,2.5,s);
    ctx.fillStyle='#a09c8f'; ctx.fillRect(x,y+s-2.5,s,2.5); ctx.fillRect(x+s-2.5,y,2.5,s);
    const n=mineCount(c,r);
    if(n>0){ ctx.fillStyle=['','#155bd4','#1d9e3f','#e23b2e','#3a3aa8','#8c3b12','#0f7f7f','#222','#666'][n];
      ctx.font='bold '+(cell*.52)+'px '+FM; ctx.textAlign='center'; ctx.textBaseline='middle';
      ctx.fillText(String(n),x+s/2,y+s/2+1); }
  }
  /* ping: mines + ring */
  if(state==='play'&&gt-pingT<PING_DUR){
    const cc=Math.floor(car.x), cr=Math.floor(car.y);
    ctx.strokeStyle='rgba(255,90,60,.3)'; ctx.lineWidth=2; ctx.setLineDash([4,6]);
    ctx.beginPath(); ctx.arc(ox+car.x*cell,oy+car.y*cell,PING_R*cell,0,7); ctx.stroke(); ctx.setLineDash([]);
    const al=.7+.3*Math.sin(gt*14);
    for(const m of mineList){ if(Math.max(Math.abs(m.c-cc),Math.abs(m.r-cr))<=PING_R) drawMine(m.c+.5,m.r+.5,1,al); }
  }
  /* boom reveal / win reveal */
  if(state==='end'&&endReason==='mine'){ for(const m of mineList){
    if(Math.max(Math.abs(m.c-Math.floor(car.x)),Math.abs(m.r-Math.floor(car.y)))<=2.5) drawMine(m.c+.5,m.r+.5,1,1); } }
  if(state==='end'&&endReason==='win'){ mineList.forEach((m,i)=>{ const p=clamp((stateT-i*.03)*4,0,1);
    if(p>0) drawMine(m.c+.5,m.r+.5,.6+.4*p,Math.min(1,p)); }); }
  /* ping shock ring */
  if(gt-pingT<.45){ const k=(gt-pingT)/.45; ctx.strokeStyle='rgba(255,235,150,'+(1-k)+')';
    ctx.lineWidth=cell*.08*(1-k)+2; ctx.beginPath(); ctx.arc(ox+car.x*cell,oy+car.y*cell,k*PING_R*cell,0,7); ctx.stroke(); }

  drawGates();

  /* arrow to next gate */
  if(state==='play'&&gates[gateIdx]){ const g=gates[gateIdx];
    const dx=(g.c+.5)-car.x, dy=(g.r+.5)-car.y, d=Math.hypot(dx,dy);
    if(d>1.6){ const a=Math.atan2(dy,dx), al=.45+.3*Math.sin(gt*6);
      ctx.save(); ctx.translate(ox+(car.x+Math.cos(a)*.95)*cell,oy+(car.y+Math.sin(a)*.95)*cell); ctx.rotate(a);
      ctx.strokeStyle='rgba(255,210,63,'+al+')'; ctx.lineWidth=cell*.07; ctx.lineCap='round'; ctx.lineJoin='round';
      ctx.beginPath(); ctx.moveTo(-cell*.18,-cell*.14); ctx.lineTo(cell*.12,0); ctx.lineTo(-cell*.18,cell*.14); ctx.stroke(); ctx.restore(); } }

  /* car / wreck */
  if(state==='end'&&endReason==='mine') drawWreck(); else drawCar();

  /* particles */
  for(const p of parts){ const a=clamp(p.life/p.max,0,1); ctx.globalAlpha=a; ctx.fillStyle=p.col;
    ctx.beginPath(); ctx.arc(ox+p.x*cell,oy+p.y*cell,p.size*cell,0,7); ctx.fill(); }
  ctx.globalAlpha=1;
  /* floats */
  for(const f of floats){ const a=1-f.t/.9; ctx.globalAlpha=a; ctx.font='bold '+(cell*.5)+'px '+FM;
    ctx.textAlign='center'; ctx.fillStyle=f.col;
    ctx.fillText(f.text,ox+f.x*cell,oy+(f.y-f.t*.8)*cell); }
  ctx.globalAlpha=1;
}

function drawHUD(){
  ctx.fillStyle='rgba(16,20,17,.92)'; ctx.fillRect(0,0,W,hudH);
  ctx.fillStyle='#ffd23f'; ctx.fillRect(0,hudH-3,W,3);
  const fs=Math.min(hudH*.48,24), cy=hudH/2;
  /* left: gate progress + mini flag */
  ctx.strokeStyle='#e8e6da'; ctx.lineWidth=2;
  ctx.beginPath(); ctx.moveTo(ox+4,cy+9); ctx.lineTo(ox+4,cy-9); ctx.stroke();
  ctx.fillStyle=patSmall; ctx.fillRect(ox+4,cy-10,15,10);
  ctx.strokeStyle='#14161a'; ctx.lineWidth=1; ctx.strokeRect(ox+4,cy-10,15,10);
  ctx.font='bold '+fs+'px '+FM; ctx.fillStyle='#f6f3ea'; ctx.textAlign='left'; ctx.textBaseline='middle';
  ctx.fillText(gateIdx>=gates.length?'FINISH':'GATE '+(gateIdx+1)+'/'+gates.length, ox+28,cy+1);
  /* center: timer */
  const t=Math.max(0,timeLeft), low=t<5&&state==='play';
  ctx.font='italic 900 '+(fs*1.6)+'px '+FD; ctx.textAlign='center';
  if(low){ ctx.fillStyle='rgba(255,75,62,'+(0.7+0.3*Math.sin(gt*10))+')'; } else ctx.fillStyle='#f6f3ea';
  ctx.fillText(t.toFixed(1), W/2, cy+2);
  /* right: ping pill + mute */
  const pw=Math.min(158,W*.3), ph=Math.min(28,hudH*.44), px2=ox+fw-pw, py2=cy-ph/2;
  const ready=state==='play'&&(gt-pingT>=PING_CD);
  if(ready){ ctx.fillStyle='#ffd23f'; rr(px2,py2,pw,ph,5); ctx.fill(); }
  else { ctx.fillStyle='#343a36'; rr(px2,py2,pw,ph,5); ctx.fill();
    const fr=1-clamp((gt-pingT)/PING_CD,0,1);
    if(state!=='menu'){ ctx.fillStyle='rgba(255,210,63,.8)'; rr(px2+2,py2+2,(pw-4)*fr,ph-4,4); ctx.fill(); } }
  ctx.font='bold '+(fs*.7)+'px '+FM; ctx.textAlign='center';
  ctx.fillStyle=ready?'#141414':'#cfd6dd';
  ctx.fillText(ready?'SPACE · SCAN READY':(state==='play'?'SCANNING…':'SCAN'), px2+pw/2, py2+ph/2+1);
  /* mute icon */
  const mx=px2-20;
  ctx.save(); ctx.translate(mx,cy); ctx.fillStyle=muted?'#7a838c':'#f6f3ea';
  ctx.beginPath(); ctx.moveTo(-6,-3); ctx.lineTo(-2,-3); ctx.lineTo(3,-7); ctx.lineTo(3,7); ctx.lineTo(-2,3); ctx.lineTo(-6,3); ctx.closePath(); ctx.fill();
  if(muted){ ctx.strokeStyle='#ff4b3e'; ctx.lineWidth=2.5; ctx.beginPath(); ctx.moveTo(-8,-9); ctx.lineTo(10,9); ctx.stroke(); }
  else { ctx.strokeStyle='#f6f3ea'; ctx.lineWidth=1.8; ctx.beginPath(); ctx.arc(3,0,7,-.9,.9); ctx.stroke(); }
  ctx.restore();
}

function render(){
  ctx.fillStyle='#181d19'; ctx.fillRect(0,0,W,H);
  let g1=ctx.createRadialGradient(W*.15,H*.08,0,W*.15,H*.08,W*.6);
  g1.addColorStop(0,'rgba(255,210,63,.05)'); g1.addColorStop(1,'rgba(0,0,0,0)'); ctx.fillStyle=g1; ctx.fillRect(0,0,W,H);
  let g2=ctx.createRadialGradient(W*.9,H*.95,0,W*.9,H*.95,W*.6);
  g2.addColorStop(0,'rgba(90,200,90,.05)'); g2.addColorStop(1,'rgba(0,0,0,0)'); ctx.fillStyle=g2; ctx.fillRect(0,0,W,H);

  ctx.save();
  if(shake>0.3) ctx.translate((Math.random()*2-1)*shake,(Math.random()*2-1)*shake);
  drawFrame(); drawField();

  /* menu ambient scan */
  if(state==='menu'){ const sx=ox+((Math.sin(gt*.45)+1)/2)*fw;
    ctx.fillStyle='rgba(255,255,255,.05)'; ctx.fillRect(sx-fw*.09,oy,fw*.18,fh);
    ctx.fillStyle='rgba(255,210,63,.18)'; ctx.fillRect(sx,oy,3,fh); }

  /* countdown */
  if(state==='count'){
    ctx.fillStyle='rgba(10,14,10,.45)'; ctx.fillRect(ox-B,oy-B,fw+2*B,fh+2*B);
    const cx=ox+fw/2, cyy=oy+fh/2;
    ctx.strokeStyle='#ffd23f'; ctx.lineWidth=5; ctx.beginPath();
    ctx.arc(cx,cyy,cell*1.15,-Math.PI/2,-Math.PI/2+(stateT/3)*Math.PI*2); ctx.stroke();
    const n=Math.max(1,Math.ceil(3-stateT)), p=stateT%1, sc=1.7-.7*Math.min(1,p*3);
    ctx.font='italic 900 '+(cell*1.5*sc)+'px '+FD; ctx.textAlign='center'; ctx.textBaseline='middle';
    ctx.fillStyle='#ffd23f'; ctx.fillText(String(n),cx,cyy+4);
  }
  /* GO flash */
  if(state==='play'&&raceTime<.6){ const k=raceTime/.6;
    ctx.font='italic 900 '+(cell*(1+k))+'px '+FD; ctx.textAlign='center'; ctx.textBaseline='middle';
    ctx.globalAlpha=1-k; ctx.fillStyle='#58c04a'; ctx.fillText('GO!!',ox+fw/2,oy+fh/2+3); ctx.globalAlpha=1; }

  /* paused */
  if(paused&&(state==='play'||state==='count')){
    ctx.fillStyle='rgba(10,14,10,.55)'; ctx.fillRect(ox-B,oy-B,fw+2*B,fh+2*B);
    ctx.font='italic 900 '+(cell*1.1)+'px '+FD; ctx.textAlign='center'; ctx.textBaseline='middle';
    ctx.fillStyle='#f6f3ea'; ctx.fillText('PAUSED',ox+fw/2,oy+fh/2-cell*.4);
    ctx.font='bold '+(cell*.35)+'px '+FM; ctx.fillStyle='#ffd23f'; ctx.fillText('press P to resume',ox+fw/2,oy+fh/2+cell*.35);
  }

  /* end word (before DOM report) */
  if(state==='end'&&!uiShown){
    ctx.fillStyle='rgba(10,12,10,.5)'; ctx.fillRect(ox-B,oy-B,fw+2*B,fh+2*B);
    const pop=Math.min(1,stateT*3), word=endReason==='mine'?'BOOM!':endReason==='time'?"TIME'S UP!":"CLEAN SWEEP!";
    ctx.font='italic 900 '+(cell*(0.6+.5*pop))+'px '+FD; ctx.textAlign='center'; ctx.textBaseline='middle';
    ctx.fillStyle=endReason==='mine'?'#ff4b3e':endReason==='time'?'#ffd23f':'#58c04a';
    ctx.fillText(word,ox+fw/2,oy+fh/2+3);
  }

  /* low-time vignette */
  if(state==='play'&&timeLeft<5){ const a=.12+.1*Math.sin(gt*8);
    ctx.strokeStyle='rgba(255,60,40,'+a+')'; ctx.lineWidth=10; ctx.strokeRect(ox-B/2,oy-B/2,fw+B,fh+B); }

  ctx.restore();
  drawHUD();
}

resize(); makePatterns(); genBoard(); toMenu();
requestAnimationFrame(frame);
})();
