    if(cell.drop){const d=cell.drop;d.t=Math.min(1,d.t+dt/Math.max(.13,d.dist*.05));if(d.t>=1)cell.drop=null}}
  if(phase==='clear'){clearT+=dt;if(clearT>.15)removeAndDrop()}
  else if(phase==='fall'){
    let done=true;
    for(let r=0;r<LVL.h&&done;r++)for(let c=0;c<LVL.w;c++){if(grid[r][c]&&grid[r][c].drop){done=false;break}}
    if(done){const m=findMatches();
      if(m.length){combo++;startClear(m);
        if(combo>=2){showBanner('CHAIN ×'+Math.min(combo,9),'COMBO '+combo,900);sfx.chain(combo)}}
      else{phase='stable';combo=0}}}
}

function attractStep(dt){
  if(phase!=='stable')return;attractT-=dt;if(attractT>0)return;
  attractT=.8+Math.random()*.9;
  const ok=(r,c)=>{const x=r>=0&&c>=0&&r<LVL.h&&c<LVL.w?grid[r][c]:null;return x&&!x.x};
  for(let t=0;t<70;t++){
    const r=(Math.random()*LVL.h)|0,c=(Math.random()*LVL.w)|0,cell=grid[r][c];
    if(!cell||cell.x)continue;const nc=LVL.next[cell.c];
    const m=(rr2,cc)=>{const x=ok(rr2,cc)?grid[rr2][cc]:null;return x&&x.c===nc};
    if((m(r,c-1)&&m(r,c-2))||(m(r,c+1)&&m(r,c+2))||(m(r-1,c)&&m(r-2,c))||(m(r+1,c)&&m(r+2,c))||(m(r,c-1)&&m(r,c+1))){doBump(r,c,true);return}
  }
  const r=(Math.random()*LVL.h)|0,c=(Math.random()*LVL.w)|0;
  if(grid[r][c]&&!grid[r][c].x)doBump(r,c,true);
}

function resetPlayer(atSpawn){
  const pos=atSpawn?spawnCell:(cpActive&&cpCell?cpCell:spawnCell);
  pl.x=pos.x+.5;pl.y=(pos.y+1)-pl.h/2-.001;pl.vx=0;pl.vy=0;pl.onGround=true;pl.sq=1;pl.baseY=pl.y;
}

function updatePlayer(dt){
  const left=keys['arrowleft']||keys['a'],right=keys['arrowright']||keys['d'],down=keys['arrowdown']||keys['s'];
  const dir=(right?1:0)-(left?1:0);
  if(dir){pl.vx+=dir*ACC*dt;pl.vx=clamp(pl.vx,-SPD,SPD);pl.facing=dir}
  else{const f=FRIC*dt;if(Math.abs(pl.vx)<=f)pl.vx=0;else pl.vx-=Math.sign(pl.vx)*f}
  if(pl.onGround)pl.coyote=.09;else pl.coyote-=dt;
  pl.jbuf-=dt;
  if(pl.jbuf>0&&pl.coyote>0){pl.vy=-JUMPV;pl.jbuf=0;pl.coyote=0;pl.sq=1.25;sfx.jump();spawnDust(pl.x,pl.y+pl.h/2,3)}
  const gg=(down&&!pl.onGround)?GRAV*2.4:GRAV;
  pl.vy=Math.min(pl.vy+gg*dt,VMAX);
  moveX(dt);moveY(dt);
  if(cpCell&&pl.x>cpCell.x+.5)cpActive=true;
  const pcx=portalCell.x+.5,pcy=portalCell.y+.5;
  if(Math.abs(pl.x-pcx)<.7&&Math.abs(pl.y-pcy)<.9)completeLevel();
  if(pl.y>LVL.h+1.5)damage();
}

function moveX(dt){
  const prevVx=pl.vx;if(!prevVx)return;
  pl.x+=prevVx*dt;
  const hw=pl.w/2,hh=pl.h/2;
  const r0=Math.floor(pl.y-hh+1e-4),r1=Math.floor(pl.y+hh-1e-4);
  let bumpR=-1,bumpC=-1,bestD=1e9;
  if(prevVx>0){const c=Math.floor(pl.x+hw);
    for(let r=r0;r<=r1;r++)if(solidAt(r,c)){pl.x=c-hw-1e-3;
      const d=Math.abs(r+.5-pl.y);if(d<bestD){bestD=d;bumpR=r;bumpC=c}}}
  else{const c=Math.floor(pl.x-hw);
    for(let r=r0;r<=r1;r++)if(solidAt(r,c)){pl.x=c+1+hw+1e-3;
      const d=Math.abs(r+.5-pl.y);if(d<bestD){bestD=d;bumpR=r;bumpC=c}}}
  if(bumpC>=0){
    pl.vx=0;
    const held=(prevVx>0&&(keys['arrowright']||keys['d']))||(prevVx<0&&(keys['arrowleft']||keys['a']));
    if(held)doBump(bumpR,bumpC,false);
    spawnDust(prevVx>0?pl.x+hw:pl.x-hw,bumpR+.5,2);
  }
}

function moveY(dt){
  const prevVy=pl.vy;pl.y+=prevVy*dt;
  const hw=pl.w/2,hh=pl.h/2;
  const c0=Math.floor(pl.x-hw+1e-4),c1=Math.floor(pl.x+hw-1e-4);
  if(prevVy>0){const r=Math.floor(pl.y+hh);let hit=false,hitC=-1,bd=1e9;
    for(let c=c0;c<=c1;c++)if(solidAt(r,c)){hit=true;const d=Math.abs(c+.5-pl.x);if(d<bd){bd=d;hitC=c}}
    if(hit){pl.y=r-hh-1e-3;pl.onGround=true;
      if(prevVy>14){pl.sq=.62;sfx.land();spawnDust(pl.x,pl.y+hh,5)}else pl.sq=Math.min(1,pl.sq+.3);
      pl.vy=0}}
  else if(prevVy<0){const r=Math.floor(pl.y-hh);let hit=false,hitC=-1,bd=1e9;
    for(let c=c0;c<=c1;c++)if(solidAt(r,c)){hit=true;const d=Math.abs(c+.5-pl.x);if(d<bd){bd=d;hitC=c}}
    if(hit){pl.y=r+1+hh+1e-3;pl.vy=0;if(hitC>=0)doBump(r,hitC,false)}}
}

  hearts--;updateHUD();sfx.hurt();shake=11;
  const hurt=$('hurt');hurt.classList.add('on');setTimeout(()=>hurt.classList.remove('on'),280);
  if(hearts<=0){gameOver();return}
  resetPlayer(false);
}

function updateGems(dt){
  gemChainT-=dt;if(gemChainT<=0)gemChain=0;
  for(let i=gems.length-1;i>=0;i--){const gm=gems[i];gm.age+=dt;
    if(state==='play'){
      const dx=pl.x-gm.x,dy=pl.y-gm.y,d=Math.hypot(dx,dy);
      if(d<1.6&&d>.01){gm.x+=dx/d*dt*9;gm.vy-=dt*26}
      if(d<.55){gems.splice(i,1);addScore(25);gemChain++;gemChainT=1;sfx.gem(gemChain);spawnSparks(gm.x,gm.y,'#ffd94a',6);continue}}
    gm.vy=Math.min(gm.vy+34*dt,16);gm.y+=gm.vy*dt;
    const bot=gm.y+.24,r=Math.floor(bot),cc=Math.floor(gm.x);
    if(gm.vy>=0&&solidAt(r,cc)&&bot-r<.35){gm.y=r-.24;gm.vy=0}
    if(gm.y>LVL.h+3||gm.age>16)gems.splice(i,1);}
}
function updateParts(dt){
  for(let i=parts.length-1;i>=0;i--){const p=parts[i];p.age+=dt;
    if(p.age>=p.life){parts.splice(i,1);continue}
    if(p.t==='shard'){p.vy+=26*dt;p.x+=p.vx*dt;p.y+=p.vy*dt;p.rot+=p.vr*dt}
    else if(p.t==='dust'){p.x+=p.vx*dt;p.y+=p.vy*dt;p.vy-=dt*2}
    else if(p.t==='ring'){p.r+=dt*7}}
  if(parts.length>400)parts.splice(0,parts.length-400);
  for(let i=texts.length-1;i>=0;i--){const t=texts[i];t.age+=dt;t.y-=dt*.9;if(t.age>=t.life)texts.splice(i,1)}
}
function spawnDust(x,y,n){for(let i=0;i<n;i++)parts.push({t:'dust',x,y,vx:(Math.random()-.5)*3,vy:-Math.random()*1.5,life:.5,age:0,s:.06+Math.random()*.07})}
function spawnSparks(x,y,c,n){for(let i=0;i<n;i++){const a=Math.random()*6.28;
  parts.push({t:'spark',x,y,vx:Math.cos(a)*(1+Math.random()*4),vy:Math.sin(a)*(1+Math.random()*4),life:.5,age:0,c,s:.05+Math.random()*.06})}}
function spawnRing(x,y,c){parts.push({t:'ring',x,y,r:.3,life:.4,age:0,c})}

function palOf(cell){
  if(cell.anim&&cell.anim.t<1){const t=1-Math.pow(1-cell.anim.t,2),A=PAL_RGB[cell.anim.from],B=PAL_RGB[cell.c];
    return {m:mixRGB(A.m,B.m,t),hi:mixRGB(A.hi,B.hi,t),lo:mixRGB(A.lo,B.lo,t)}}
  const p=PAL[cell.c];return {m:p.m,hi:p.hi,lo:p.lo};
}
function drawBlockPix(x,y,s,pal,flash,pop){
  const sc=1+(pop||0)*.2;
  if(sc!==1||flash>0){s*=sc;x-= (sc-1)*s/2;y-=(sc-1)*s/2}
  const g=1.5;
  ctx.fillStyle=pal.lo;rr(x+g,y+g,s-g*2,s-g*2,s*.18);ctx.fill();
  ctx.fillStyle=pal.m;rr(x+g+2,y+g+2,s-g*2-4,s-g*2-4,s*.15);ctx.fill();
  ctx.globalAlpha=.55;ctx.fillStyle=pal.hi;
  rr(x+g+2,y+g+2,s-g*2-4,(s-g*2-4)*.42,s*.15);ctx.fill();
  ctx.globalAlpha=.7;ctx.fillStyle='#fff';
  ctx.beginPath();ctx.arc(x+s*.26,y+s*.24,s*.07,0,7);ctx.fill();
  ctx.globalAlpha=1;
  if(flash>0){ctx.globalAlpha=flash;ctx.fillStyle='#fff';rr(x+g,y+g,s-g*2,s-g*2,s*.18);ctx.fill();ctx.globalAlpha=1}
}
function drawXBlockPix(x,y,s){
  ctx.fillStyle='#1b2a37';rr(x+1.5,y+1.5,s-3,s-3,s*.18);ctx.fill();
  ctx.fillStyle='#2f4557';rr(x+3.5,y+3.5,s-7,s-7,s*.14);ctx.fill();
  ctx.strokeStyle='rgba(150,220,235,.28)';ctx.lineWidth=1.5;
  ctx.beginPath();ctx.moveTo(x+s*.2,y+s*.75);ctx.lineTo(x+s*.5,y+s*.2);ctx.lineTo(x+s*.8,y+s*.75);ctx.closePath();ctx.stroke();
}
function drawBackdropLayer(arr,par,alpha,baseOff){
  for(const c of arr){
    const sx=(c.x-cam.x*par)*TILE;if(sx<-160||sx>W+160)continue;
    const hgt=c.h*TILE*(.8+par*.4),w=c.w*TILE*.5,sk=c.sk*TILE;
    ctx.globalAlpha=alpha;
    ctx.fillStyle=PAL[c.c].m;
    if(c.ceil){const y0=6;ctx.beginPath();ctx.moveTo(sx-w,y0);ctx.lineTo(sx+w,y0);ctx.lineTo(sx+sk,y0+hgt);ctx.closePath()}
    else{const y0=H-4;ctx.beginPath();ctx.moveTo(sx-w,y0);ctx.lineTo(sx+w,y0);ctx.lineTo(sx+sk,y0-hgt);ctx.closePath}
    ctx.fill();
    ctx.globalAlpha=alpha*1.6;ctx.strokeStyle=PAL[c.c].hi;ctx.lineWidth=1.5;
    if(c.ceil){ctx.beginPath();ctx.moveTo(sx+sk*.3,y0+hgt*.85);ctx.lineTo(sx+sk,y0+hgt)}
    else{ctx.beginPath();ctx.moveTo(sx+sk*.3,y0-hgt*.85);ctx.lineTo(sx+sk,y0-hgt)}
    ctx.stroke();
  }
  ctx.globalAlpha=1;
}
function diamond(x,y,s){ctx.beginPath();ctx.moveTo(x,y-s);ctx.lineTo(x+s*.72,y);ctx.lineTo(x,y+s);ctx.lineTo(x-s*.72,y);ctx.closePath()}
