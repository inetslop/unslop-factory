function effStep(){let s=Math.max(70,140-eaten*1.3);if(performance.now()<slowUntil)s*=1.65;return s;}
function pushDir(d){

  if((d.x===-lastD.x&&d.y===-lastD.y)||(d.x===lastD.x&&d.y===lastD.y))return;

}
function freeze(){snake.forEach(s=>{s.px=s.x;s.py=s.y;});}
function consumeShield(){shieldUntil=0;}
function step(){
  const now=performance.now();
  while(queue.length){const d=queue.shift();
    if((d.x===-dir.x&&d.y===-dir.y)||(d.x===dir.x&&d.y===dir.y))continue; dir=d;break;}
  const head=snake[0], nx=head.x+dir.x, ny=head.y+dir.y;
  if(nx<0||ny<0||nx>=COLS||ny>=ROWS){
    if(now<shieldUntil||now<phaseUntil){consumeShield();dir.x*=-1;dir.y*=-1;freeze();sfx.bounce();shake=Math.max(shake,4);return;}
    return die('wall');
  }
  let ci=-1;for(let i=0;i<cards.length;i++)if(cards[i].x===nx&&cards[i].y===ny){ci=i;break;}
  const willGrow=ci>=0;
  const lim=snake.length-(willGrow?0:1);let hitBody=false;
  for(let i=0;i<lim;i++)if(snake[i].x===nx&&snake[i].y===ny){hitBody=true;break;}
  if(hitBody){
    if(now<phaseUntil){/* phasing through */}
    else if(now<shieldUntil){consumeShield();phaseUntil=now+900;freeze();sfx.bounce();shake=Math.max(shake,4);return;}
    else return die('self');
  }
  const oldTail={x:snake[snake.length-1].x,y:snake[snake.length-1].y};
  snake.forEach(s=>{s.px=s.x;s.py=s.y;});
  head.x=nx;head.y=ny;
  for(let i=snake.length-1;i>0;i--){const s=snake[i];s.x=snake[i-1].x;s.y=snake[i-1].y;}
  if(willGrow){
    snake.push({x:oldTail.x,y:oldTail.y,px:oldTail.x,py:oldTail.y});
    handleEat(cards.splice(ci,1)[0]);
  }else snake.pop();
}
function handleEat(c){

  let pts,glyph=SUITSCol(c),col;
  if(c.joker){
    combo=combo; // wild: chain preserved
    pts=25*Math.max(1,Math.min(Math.max(combo,1),8));

    sfx.joker();col=GOLD;glyph='★';
    popups.push({x:c.x,y:c.y,txt:'JOKER +'+pts,col,t:0});
  }else{
    if(streakSuit===-1||c.suit===streakSuit)combo++;else{streakSuit=c.suit;combo=1;}
    maxStreak=Math.max(maxStreak,combo);
    const m=Math.min(combo,8);pts=basePts(c.rank)*m;col=RED[c.suit]?REDC:GOLD;glyph=SUITS[c.suit];
    if(c.rank===10){shieldUntil=now+6000;sfx.shield();}
    else if(c.rank===11){slowUntil=now+3500;sfx.queen();}
    else if(c.rank===12){sfx.king();shake=Math.max(shake,6);for(let i=0;i<2&&cards.length<7;i++)spawnCard();}
    else sfx.eat(combo);
    if(combo>=3)sfx.streak(combo);
    popups.push({x:c.x,y:c.y,txt:'+'+pts+(combo>=2?' '+glyph+'×'+m:''),col,t:0});
  }
  score+=pts;hist.push({suit:c.suit,rank:c.rank,joker:c.joker});
  headPulse=1;shake=Math.max(shake,c.joker?7:(c.rank===12?5:2.5));

  updateHUD();
}

  if(now<shieldUntil){

    ctx.lineWidth=s*.05;ctx.beginPath();ctx.arc(0,0,w*.72,0,7);ctx.stroke();
  }
  ctx.restore();
}

function render(ts){

  let sx=0,sy=0;if(shake>.2){sx=(Math.random()*2-1)*shake;sy=(Math.random()*2-1)*shake;}

function frame(ts){
  requestAnimationFrame(frame);
  const dt=Math.min(50,ts-last);last=ts;
  if(state==='playing'){
    acc+=dt;const eff=effStep();
    while(acc>=eff&&state==='playing'){acc-=eff;step();maintain();}
  }
  if(state!=='paused'){
    const s=dt/1000;
    for(let i=parts.length-1;i>=0;i--){const p=parts[i];p.t+=s;if(p.t>p.life){parts.splice(i,1);continue;}
      p.vy+=p.g*s;p.x+=p.vx*s;p.y+=p.vy*s;p.rot+=p.vr*s;}
    for(let i=popups.length-1;i>=0;i--){popups[i].t+=s;if(popups[i].t>.9)popups.splice(i,1);}
  }
  headPulse=Math.max(0,headPulse-dt*.004);
  shake=Math.max(0,shake-dt*.03);
  if(state==='over'&&!overShown&&ts-overAt>650){overShown=true;showOver();}
  syncBadges();
  render(ts);
}
