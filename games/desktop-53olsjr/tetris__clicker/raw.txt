function update(dt){
  // fx always (unless paused)
  shake.m = shake.m>0 ? Math.max(0, shake.m*Math.pow(.86, dt/16)) : 0;
  if(inv.t>0) inv.t-=dt;
  for(let i=particles.length-1;i>=0;i--){ const p=particles[i]; p.t+=dt; if(p.t>=p.life){particles.splice(i,1);continue;}
    p.vy+=p.g*dt/1000; p.x+=p.vx*dt/1000; p.y+=p.vy*dt/1000; p.rot+=p.vr*dt/1000; }
  for(let i=popups.length-1;i>=0;i--){ const p=popups[i]; p.t+=dt; if(p.t>=p.life){popups.splice(i,1);continue;} p.y+=p.vy*dt/1000; }

  if(state!=='playing'||!cur) return;
  if(clearing){
    clearing.t+=dt;
    if(clearing.t>=CLEAR_MS){
      const rows=clearing.rows.slice().sort((a,b)=>b-a);
      for(const r of rows){ grid.splice(r,1); grid.unshift(emptyRow()); }
      clearing=null; nextPiece();
    }
    return;
  }
  gravityAcc+=dt;
  const iv=gravMs();
  while(gravityAcc>=iv){ gravityAcc-=iv; if(!collide(cur.x,cur.y+1,cur.m)) cur.y++; else break; }
  if(collide(cur.x,cur.y+1,cur.m)){
    lockTimer+=dt;
    if(lockTimer>=LOCK_DELAY) doLock();
  } else lockTimer=0;
  displayX += (cur.x-displayX)*Math.min(1, dt*0.02);
  if(Math.abs(cur.x-displayX)<0.01) displayX=cur.x;
}

    }

function drawMini(g,cv,type,dim){
  const s=cv.width/dpr; g.setTransform(dpr,0,0,dpr,0,0); g.clearRect(0,0,s,s);
  if(!type){ g.globalAlpha=.35; g.fillStyle='#7fa8b2'; g.font='.9rem '+FONT_D; g.textAlign='center'; g.textBaseline='middle'; g.fillText('·',s/2,s/2); g.globalAlpha=1; return; }
  const m=SHAPES[type].m, c=SHAPES[type].c;
  let minR=9,maxR=-1,minC=9,maxC=-1;
  for(let r=0;r<m.length;r++)for(let cc=0;cc<m[r].length;cc++) if(m[r][cc]){ minR=Math.min(minR,r);maxR=Math.max(maxR,r);minC=Math.min(minC,cc);maxC=Math.max(maxC,cc);}
  const bw=maxC-minC+1, bh=maxR-minR+1, bs=Math.min(s/ (Math.max(bw,bh)+.6));
  const ox=(s-bw*bs)/2 + minC*0, oy=(s-bh*bs)/2;
  g.globalAlpha=dim?.35:1;
  for(let r=minR;r<=maxR;r++)for(let cc=minC;cc<=maxC;cc++) if(m[r][cc]) block(g,(cc-minC)*bs+ox,(r-minR)*bs+oy,bs,c);
  g.globalAlpha=1;
}
