    c:['#f4d68b','#e0a33c','#b3272c','#fff2cf'][i%4],s:1.5+Math.random()*2.5}); } }
let lastT=performance.now();
function fxLoop(t){ const dt=Math.min(50,t-lastT)/16.7; lastT=t;
  ctx.clearRect(0,0,innerWidth,innerHeight);
  for(const m of motes){ m.y-=m.vy*dt*.06; m.ph+=.02*dt*m.sp; if(m.y<-4){m.y=innerHeight+4;m.x=Math.random()*innerWidth;}
    const a=.05+.09*(0.5+0.5*Math.sin(m.ph)); ctx.fillStyle='rgba(244,214,139,'+a.toFixed(3)+')';
    ctx.beginPath(); ctx.arc(m.x+Math.sin(m.ph)*8,m.y,m.r,0,6.28); ctx.fill(); }
  for(let i=sparks.length-1;i>=0;i--){ const p=sparks[i]; p.x+=p.vx*dt; p.y+=p.vy*dt; p.vy+=.11*dt; p.life-=dt;
    if(p.life<=0){sparks.splice(i,1);continue;} ctx.globalAlpha=Math.max(0,p.life/p.max); ctx.fillStyle=p.c;
    ctx.fillRect(p.x,p.y,p.s,p.s); }
  ctx.globalAlpha=1; requestAnimationFrame(fxLoop);
}

const rankOf=id=>cards.get(id).rank, suitOf=id=>cards.get(id).suit;
function getRun(i,j){ const col=cols[i],run=[];

function canDrop(ids,dst,src){
  if(src&&dst.t===src.t&&dst.i===src.i)return false;
  const r=rankOf(ids[0]);
  if(dst.t==='found')return ids.length===1&&found[suitOf(ids[0])]===r-1;
  if(dst.t==='cell')return ids.length===1&&cells[dst.i]===null;
  const c=cols[dst.i]; if(c.length===0)return true;
  return rankOf(c[c.length-1])===r+1;
}

function doMove(ids,from,to){
  pushUndo();

  if(to.t==='col')cols[to.i].push(...ids);
  else if(to.t==='cell')cells[to.i]=ids[0];
  else{found[suitOf(ids[0])]++;foundTop[suitOf(ids[0])]=ids[0];}
  rebuildPos(); moves++; clearSel();
  (to.t==='found'?sfx.chime:sfx.place)();
  layout(); hud(); checkWin();
}

function findHint(){
  const tops=[]; cells.forEach((id,i)=>{if(id)tops.push({ids:[id],from:{t:'cell',i}});});
  cols.forEach((c,i)=>{ if(c.length&&canDrop([c[c.length-1]],{t:'found',i:suitOf(c[c.length-1])},{t:'col',i}))
    return tops; }); // checked below properly
  // 1) foundation moves
  const tryF=id=>canDrop([id],{t:'found',i:suitOf(id)},cardPos[id]);
  for(let i=0;i<4;i++)if(cells[i]&&tryF(cells[i]))return{ids:[cells[i]],from:{t:'cell',i},dst:{t:'found',i:suitOf(cells[i])}};
  cols.forEach((c,i)=>{const top=c[c.length-1]; if(top&&tryF(top))return null;});
  const foundMove=[]; cells.forEach((id,i)=>{if(id&&tryF(id))foundMove.push({ids:[id],from:{t:'cell',i},dst:{t:'found',i:suitOf(id)}});});
  cols.forEach((c,i)=>{const top=c[c.length-1]; if(top&&tryF(top))foundMove.push({ids:[top],from:{t:'col',i,j:c.length-1},dst:{t:'found',i:suitOf(top)}});});
  if(foundMove.length)return foundMove[0];
  // 2) run moves onto non-empty columns (prefer same suit)
  let any=null;
  const scan=(ids,from)=>{ for(let i=0;i<8;i++){ if(!cols[i].length||i===from.i)continue;
      const top=cols[i][cols[i].length-1];
      if(rankOf(top)===rankOf(ids[0])+1){ const same=suitOf(top)===suitOf(ids[0]);
        if(same)return{ids,from,dst:{t:'col',i}}; if(!any)any={ids,from,dst:{t:'col',i}}; } } return null; };
  for(let i=0;i<8;i++){ const c=cols[i];
    for(let j=c.length-1;j>=0;j--){ const run=getRun(i,j); if(!run)continue;
      const r=scan(run,{t:'col',i,j}); if(r)return r; } }
  cells.forEach((id,i)=>{ if(id){const r=scan([id],{t:'cell',i}); if(r)return r;} });
  if(any)return any;
  // 3) free a card into an empty column
  const empty=cols.findIndex(c=>!c.length);
  if(empty>=0)for(let i=0;i<8;i++){ const c=cols[i];
    for(let j=0;j<c.length-1;j++){ if(getRun(i,j)&&getRun(i,j).length===1)
      return{ids:[c[j]],from:{t:'col',i,j},dst:{t:'col',i:empty}}; } }
  return null;
}

  if(!h){toast('No moves to be found');sfx.err();return;}

  const src=cardEls[h.ids[0]]; let dst;
  if(h.dst.t==='col')dst=cols[h.dst.i].length?cardEls[cols[h.dst.i][cols[h.dst.i].length-1]]:slotCols[h.dst.i];
  else if(h.dst.t==='cell')dst=slotCells[h.dst.i];
  else dst=foundTop[h.dst.i]?cardEls[foundTop[h.dst.i]]:slotFound[SUITS.indexOf(h.dst.i)];
  src.classList.add('hint'); dst.classList.add('hint');
  setTimeout(()=>{src.classList.remove('hint');dst.classList.remove('hint');},1600);
}
