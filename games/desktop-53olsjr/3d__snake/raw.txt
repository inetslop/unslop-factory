}

    let fatal = nx<0||ny<0||nz<0||nx>=GX||ny>=GY||nz>=GZ;
    if(!fatal){ for(let i=0;i<snake.length-1;i++)
      if(snake[i].x===nx&&snake[i].y===ny&&snake[i].z===nz){ fatal=true; break; } }
    let s = Math.random()*3 - (Math.abs(food.x-nx)+Math.abs(food.y-ny)+Math.abs(food.z-nz))*0.5;
    if(nx===food.x&&ny===food.y&&nz===food.z) s+=40;
    if(fatal) s-=100;
    opts.push({d,s});
  }
  opts.sort((a,b)=>b.s-a.s);
  return opts[0].d;
}

  if(state==='MENU'){ attractDeadT=1.1; return; }
  won=false; gameOver();
}

function drawCube(wx,wy,wz,h,col,opt){
  opt = opt||{};
  const v=[];
  for(let i=0;i<8;i++)
    v.push(toView(wx+((i&1)?h:-h), wy+((i&2)?h:-h), wz+((i&4)?h:-h)));
  for(let f=0;f<6;f++){
    const F=FACES[f];
    if(F.n[0]*(cam.px-wx)+F.n[1]*(cam.py-wy)+F.n[2]*(cam.pz-wz) <= 0.02) continue;
    let pts = clipPoly([v[F.i[0]],v[F.i[1]],v[F.i[2]],v[F.i[3]]]);
    if(pts.length<3) continue;
    const b = opt.flat ? 0.95 : (0.42+0.6*Math.max(0, F.n[0]*LIGHT[0]+F.n[1]*LIGHT[1]+F.n[2]*LIGHT[2]));
    let r=col[0]*b, g=col[1]*b, bl=col[2]*b;
    if(deadFlash>0){ const m=deadFlash*0.6; r+=(255-r)*m; g+=(70-g)*m; bl+=(60-bl)*m; }
    ctx.beginPath();
    for(let i=0;i<pts.length;i++){ const p=proj(pts[i]); if(i)ctx.lineTo(p[0],p[1]); else ctx.moveTo(p[0],p[1]); }
    ctx.closePath();
    ctx.fillStyle='rgb('+(r|0)+','+(g|0)+','+(bl|0)+')';
    ctx.fill();
    if(opt.edge!==false){
      ctx.strokeStyle='rgba(4,18,20,'+(opt.edgeAlpha!==undefined?opt.edgeAlpha:0.5)+')';
      ctx.lineWidth=1; ctx.stroke();
    }
  }
}
function drawShellFace(f){
  const pts = clipPoly(f.pts.map(p=>toView(p.x,p.y,p.z)));
  if(pts.length<3) return;
  ctx.beginPath();
  for(let i=0;i<pts.length;i++){ const p=proj(pts[i]); if(i)ctx.lineTo(p[0],p[1]); else ctx.moveTo(p[0],p[1]); }
  ctx.closePath();
  ctx.fillStyle=f.col; ctx.fill();
  ctx.strokeStyle='rgba(160,230,240,.1)'; ctx.lineWidth=1; ctx.stroke();
}
function drawFloorDetail(){
  // soft central glow on the floor
  const c = toView(0,-hy,0);
  if(c.z>NEAR){
    const p=proj(c), r=focal*6.5/c.z;
    const g=ctx.createRadialGradient(p[0],p[1],0,p[0],p[1],r);
    g.addColorStop(0,'rgba(70,190,200,.13)'); g.addColorStop(1,'rgba(70,190,200,0)');
    ctx.globalCompositeOperation='lighter';
    ctx.fillStyle=g; ctx.beginPath(); ctx.arc(p[0],p[1],r,0,TAU); ctx.fill();
    ctx.globalCompositeOperation='source-over';
  }
  ctx.strokeStyle='rgba(95,195,205,.13)'; ctx.lineWidth=1;
  ctx.beginPath();
  for(let i=0;i<=GX;i++){
    const a=toView(-hx+i,-hy,-hz), b=toView(-hx+i,-hy,hz);
    const s=clipSeg(a,b); if(s){ const p0=proj(s[0]),p1=proj(s[1]); ctx.moveTo(p0[0],p0[1]); ctx.lineTo(p1[0],p1[1]); }
  }
  for(let j=0;j<=GZ;j++){
    const a=toView(-hx,-hy,-hz+j), b=toView(hx,-hy,-hz+j);
    const s=clipSeg(a,b); if(s){ const p0=proj(s[0]),p1=proj(s[1]); ctx.moveTo(p0[0],p0[1]); ctx.lineTo(p1[0],p1[1]); }
  }
  ctx.stroke();
}
function drawShellWire(){
  ctx.strokeStyle='rgba(150,235,245,.14)'; ctx.lineWidth=1;
  ctx.beginPath();
  for(const e of edges){
    const a=toView(corners[e[0]].x,corners[e[0]].y,corners[e[0]].z);
    const b=toView(corners[e[1]].x,corners[e[1]].y,corners[e[1]].z);
    const s=clipSeg(a,b); if(s){ const p0=proj(s[0]),p1=proj(s[1]); ctx.moveTo(p0[0],p0[1]); ctx.lineTo(p1[0],p1[1]); }
  }
  ctx.stroke();
}
function drawDust(){
  for(const p of dust){
    const v=toView(p.x,p.y,p.z); if(v.z<NEAR) continue;
    const s=focal/v.z, q=proj(v);
    const r=Math.min(2.4, Math.max(0.5, s*0.03));
    ctx.fillStyle='rgba(190,235,240,'+(0.08+0.09*Math.sin(tSec*1.3+p.ph)).toFixed(3)+')';
    ctx.beginPath(); ctx.arc(q[0],q[1],r,0,TAU); ctx.fill();
  }
}
function easeOutBack(t){ const c1=1.70158,c3=c1+1; return 1+c3*Math.pow(t-1,3)+c1*Math.pow(t-1,2); }

function renderPositions(){
  const n=snake.length;
  const t=(state==='PLAY'||state==='MENU') ? Math.min(1, acc/intervalMs()) : 1;
  const out=[];
  for(let i=0;i<n;i++){
    const c=snake[i], p=prevSnake[i]||c;
    out.push({ x:lerp(p.x,c.x,t)-hx+0.5, y:lerp(p.y,c.y,t)-hy+0.5, z:lerp(p.z,c.z,t)-hz+0.5 });
  }
  return out;
}
function lerp(a,b,t){ return a+(b-a)*t; }

function drawShadows(rp){
  const ryF = 0.32 + 0.45*Math.max(0,Math.sin(Math.min(1.57,Math.max(0,cam.pitch))));
  for(const p of rp){
    const v=toView(p.x,-hy+0.02,p.z); if(v.z<NEAR) continue;
    const s=focal/v.z, q=proj(v);
    const a=Math.max(0.05, 0.2*(1-p.y/(GY+3)));
    ctx.fillStyle='rgba(0,10,12,'+a.toFixed(3)+')';
    ctx.beginPath(); ctx.ellipse(q[0],q[1],s*0.42,s*0.42*ryF,0,0,TAU); ctx.fill();
  }
}
function drawEyes(wx,wy,wz){
  const d=dir;
  const up = Math.abs(d.y)>0.9 ? [1,0,0] : [0,1,0];
  let rx=d.y*up[2]-d.z*up[1], ry=d.z*up[0]-d.x*up[2], rz=d.x*up[1]-d.y*up[0];
  const rl=Math.hypot(rx,ry,rz)||1; rx/=rl; ry/=rl; rz/=rl;
  for(const sgn of [1,-1]){
    const ex=wx+d.x*0.5+rx*0.24*sgn+up[0]*0.17;
    const ey=wy+d.y*0.5+ry*0.24*sgn+up[1]*0.17;
    const ez=wz+d.z*0.5+rz*0.24*sgn+up[2]*0.17;
    const v=toView(ex,ey,ez); if(v.z<NEAR) continue;
    const s=focal/v.z, q=proj(v);
    ctx.fillStyle='#0a2320';
    ctx.beginPath(); ctx.arc(q[0],q[1],s*0.085,0,TAU); ctx.fill();
    ctx.fillStyle='rgba(240,255,250,.9)';
    ctx.beginPath(); ctx.arc(q[0]-s*0.02,q[1]-s*0.025,s*0.03,0,TAU); ctx.fill();
  }
}
function drawCubesSorted(rp){
  const n=rp.length, items=[];
  for(let i=n-1;i>=0;i--){
    const p=rp[i];
    const z=toView(p.x,p.y,p.z).z;
    const isHead = i===0;
    items.push({z, draw:()=>{
      if(isHead) headPulse=Math.max(headPulse,0); // no-op guard
      const h=(isHead? 0.47*(1+0.22*headPulse) : 0.47);
      drawCube(p.x,p.y,p.z,h,segColor(i,n));
      if(isHead){
        drawEyes(p.x,p.y,p.z);
        const v=toView(p.x,p.y,p.z);
        if(v.z>NEAR){
          const s=focal/v.z, q=proj(v), r=s*1.15;
          const g=ctx.createRadialGradient(q[0],q[1],0,q[0],q[1],r);
          g.addColorStop(0,'rgba(198,255,122,.2)'); g.addColorStop(1,'rgba(198,255,122,0)');
          ctx.globalCompositeOperation='lighter';
          ctx.fillStyle=g; ctx.beginPath(); ctx.arc(q[0],q[1],r,0,TAU); ctx.fill();
          ctx.globalCompositeOperation='source-over';
        }
      }
    }});
  }
  if(food){
    const pop=easeOutBack(Math.min(1,(tSec-food.born)/0.35));
    const fy=food.y-hy+0.5+0.1*Math.sin(tSec*2.1);
    const fx=food.x-hx+0.5, fz=food.z-hz+0.5;
    const h=(0.26+0.045*Math.sin(tSec*5))*Math.max(0.01,pop);
    items.push({z:toView(fx,fy,fz).z, draw:()=>{
      drawCube(fx,fy,fz,h, food.bonus?[255,208,80]:[255,140,66], {flat:true, edge:false});
      const v=toView(fx,fy,fz); if(v.z<NEAR) return;
      const s=focal/v.z, q=proj(v), r=s*0.9*(1+0.12*Math.sin(tSec*6));
      const g=ctx.createRadialGradient(q[0],q[1],0,q[0],q[1],r);
      g.addColorStop(0, food.bonus?'rgba(255,214,90,.5)':'rgba(255,140,60,.42)');
      g.addColorStop(1,'rgba(255,140,60,0)');
      ctx.globalCompositeOperation='lighter';
      ctx.fillStyle=g; ctx.beginPath(); ctx.arc(q[0],q[1],r,0,TAU); ctx.fill();
      ctx.globalCompositeOperation='source-over';
      if(food.bonus){
        const left=food.dieAt-tSec;
        if(!(left<1.5 && Math.sin(tSec*20)>0)){
          ctx.strokeStyle='rgba(255,214,90,.85)'; ctx.lineWidth=2.5;
          ctx.beginPath(); ctx.arc(q[0],q[1],s*0.6,-Math.PI/2,-Math.PI/2+TAU*(left/6)); ctx.stroke();
        }
      }
    }});
  }
  items.sort((a,b)=>b.z-a.z);
  for(const it of items) it.draw();
}

  const age=tSec-deathFX.born;
  if(age>0.9){ deathFX=null; return; }
  const v=toView(deathFX.wx,deathFX.wy,deathFX.wz);
  if(v.z>NEAR){
    const s=focal/v.z, q=proj(v), r=s*(0.25+age*2.6);
    ctx.strokeStyle='rgba(255,97,87,'+(1-age/0.9).toFixed(3)+')';
    ctx.lineWidth=3;
    ctx.beginPath(); ctx.arc(q[0],q[1],r,0,TAU); ctx.stroke();
  }
  ctx.fillStyle='rgba(255,60,45,'+Math.max(0,0.2-age*0.28).toFixed(3)+')';
  ctx.fillRect(0,0,W,H);
}

    const s=focal/v.z, q=proj(v);
    ctx.font='700 '+Math.min(26,Math.max(11,s*0.17))+'px '+'ui-monospace,Consolas,monospace';

    ctx.globalAlpha=1;
  }
}
function drawMinimap(){
  const mmW=Math.min(138,W*0.3), mmH=mmW*(GZ/GX);
  const x0=W-mmW-14, y0=86;
  ctx.fillStyle='rgba(4,15,20,.6)';
  ctx.fillRect(x0,y0,mmW,mmH);
  ctx.strokeStyle='rgba(150,225,235,.25)'; ctx.lineWidth=1;
  ctx.strokeRect(x0+0.5,y0+0.5,mmW-1,mmH-1);
  const px=c=>x0+(c+0.5)/GX*mmW, py=c=>y0+(c+0.5)/GZ*mmH;
  if(food){
    ctx.fillStyle=food.bonus?'#ffd34d':'#ff9a4d';
    ctx.beginPath(); ctx.arc(px(food.x),py(food.z),2.6+Math.sin(tSec*6)*1,0,TAU); ctx.fill();
  }
  const n=snake.length;
  for(let i=n-1;i>=0;i--){
    const f=i/Math.max(1,n-1);
    ctx.globalAlpha=0.25+0.75*f;
    ctx.fillStyle='#8fe8a8';

  ctx.font='700 9px ui-monospace,Consolas,monospace';

function render(){
  ctx.clearRect(0,0,W,H);
  updateCamBasis();
  if(shakeAmp>0){ W2+= (Math.random()*2-1)*shakeAmp; H2+=(Math.random()*2-1)*shakeAmp; }
  for(const f of shellFaces) if(f.n[0]*cam.px+f.n[1]*cam.py+f.n[2]*cam.pz < 0) drawShellFace(f);
  if(cam.py>0.25) drawFloorDetail();
  drawDust();
  const rp=renderPositions();
  if(cam.py>0.25) drawShadows(rp);
  drawCubesSorted(rp);
  drawShellWire();
  drawDeathFX();
  drawPopups();
  if(state!=='MENU') drawMinimap();
}

/* ============================== MAIN LOOP ============================== */
let last=performance.now(), lastUserCam=performance.now();
function frame(now){
  requestAnimationFrame(frame);
  const dt=Math.min(48, now-last); last=now; tSec+=dt/1000;
  const k=1-Math.exp(-dt/140);
  cam.yaw+=(cam.tYaw-cam.yaw)*k;
  cam.pitch+=(cam.tPitch-cam.pitch)*k;
  cam.dist+=(cam.tDist-cam.dist)*k;
  if(state==='MENU' && now-lastUserCam>2600) cam.tYaw+=dt*0.00018;

  shakeAmp*=Math.exp(-dt/110); if(shakeAmp<0.05) shakeAmp=0;
  deadFlash=Math.max(0,deadFlash-dt/700);
  headPulse=Math.max(0,headPulse-dt/260);
  updateDust(dt);

  if(state==='PLAY' && !paused){
    acc+=dt; const iv=intervalMs();
    while(acc>=iv){ acc-=iv; doStep(); }
    if(food && food.bonus && tSec>food.dieAt){ food=null; spawnFood(); }
  } else if(state==='MENU'){
    if(attractDeadT>0){ attractDeadT-=dt/1000; if(attractDeadT<=0) resetWorld(); }
    else{
      acc+=dt;
      while(acc>=150){ acc-=150; dir=aiDir(); doStep(); }
    }
  }
  render();
}
