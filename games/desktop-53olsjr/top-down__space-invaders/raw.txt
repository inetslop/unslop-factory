function makeSprite(rows, fill, glowColor, s, glow){

  const h = rows.length, w = rows[0].length;
  const c = document.createElement('canvas'); c.width = w * s; c.height = h * s;
  const g = c.getContext('2d');
  for(let pass = 0; pass < 2; pass++){
    g.fillStyle = fill;
    g.shadowColor = glowColor; g.shadowBlur = pass ? 0 : glow;
    for(let y = 0; y < h; y++) for(let x = 0; x < w; x++)
      if(rows[y][x] === '#') g.fillRect(x * s, y * s, s, s);
  }
  return c;
}

const Snd = {
  ctx: null, master: null, muted: false, noiseBuf: null, siren: null,
  ensure(){
    if(this.ctx) return true;
    try{
      const AC = window.AudioContext || window.webkitAudioContext;
      if(!AC) return false;
      this.ctx = new AC();
      this.master = this.ctx.createGain();
      this.master.gain.value = 0.45;
      this.master.connect(this.ctx.destination);
      const len = this.ctx.sampleRate, b = this.ctx.createBuffer(1, len, this.ctx.sampleRate), d = b.getChannelData(0);
      for(let i = 0; i < len; i++) d[i] = Math.random() * 2 - 1;
      this.noiseBuf = b;
      return true;
    }catch(e){ return false; }
  },
  resume(){ if(this.ctx && this.ctx.state === 'suspended') this.ctx.resume(); },
  tone(f, f2, dur, type, vol, delay){
    if(!this.ctx || this.muted) return;
    type = type || 'square'; vol = (vol === undefined) ? 0.2 : vol; delay = delay || 0;
    const t0 = this.ctx.currentTime + delay;
    const o = this.ctx.createOscillator(), g = this.ctx.createGain();
    o.type = type;
    o.frequency.setValueAtTime(f, t0);
    if(f2) o.frequency.exponentialRampToValueAtTime(Math.max(20, f2), t0 + dur);
    g.gain.setValueAtTime(vol, t0);
    g.gain.exponentialRampToValueAtTime(0.001, t0 + dur);
    o.connect(g); g.connect(this.master);
    o.start(t0); o.stop(t0 + dur + 0.05);
  },
  noise(dur, vol, freq, type, delay){
    if(!this.ctx || this.muted) return;
    vol = (vol === undefined) ? 0.3 : vol; type = type || 'bandpass'; delay = delay || 0;
    const t0 = this.ctx.currentTime + delay;
    const s = this.ctx.createBufferSource(); s.buffer = this.noiseBuf; s.loop = true;
    const f = this.ctx.createBiquadFilter(); f.type = type; f.frequency.value = freq; f.Q.value = 1;
    const g = this.ctx.createGain();
    g.gain.setValueAtTime(vol, t0);
    g.gain.exponentialRampToValueAtTime(0.001, t0 + dur);
    s.connect(f); f.connect(g); g.connect(this.master);
    s.start(t0); s.stop(t0 + dur + 0.05);
  },
  step(i){ this.tone([131,117,104,92][i % 4], null, .08, 'square', .2); },
  shoot(){ this.tone(880, 160, .1, 'square', .15); },
  aShoot(){ this.tone(300, 90, .18, 'sawtooth', .06); },
  alienDie(){ this.noise(.16, .26, 620); this.tone(220, 50, .14, 'square', .15); },
  tick(){ this.tone(1400, 900, .04, 'square', .05); },
  ufoDie(){ this.noise(.3, .32, 900); [660,880,1320].forEach((f,i)=>this.tone(f,null,.09,'triangle',.18,i*.07)); },
  hit(){ this.noise(.5, .5, 240, 'lowpass'); this.tone(320, 40, .5, 'sawtooth', .3); },
  waveClear(){ [523,659,784].forEach((f,i)=>this.tone(f,null,.1,'square',.16,i*.09)); },
  over(){ [392,311,262,196,131].forEach((f,i)=>this.tone(f,f*.9,.22,'square',.18,i*.17)); },
  startJingle(){ [262,392,523,784].forEach((f,i)=>this.tone(f,null,.08,'triangle',.18,i*.06)); },
  sirenOn(){
    if(!this.ctx || this.muted || this.siren) return;
    const o = this.ctx.createOscillator(), g = this.ctx.createGain();
    const l = this.ctx.createOscillator(), lg = this.ctx.createGain();
    o.type = 'triangle'; o.frequency.value = 520;
    l.type = 'square'; l.frequency.value = 6.5; lg.gain.value = 140;
    l.connect(lg); lg.connect(o.frequency);
    g.gain.value = .045;
    o.connect(g); g.connect(this.master);
    o.start(); l.start();
    this.siren = {o, l, g};
  },
  sirenOff(){
    if(!this.siren) return;
    const s = this.siren, t = this.ctx.currentTime;
    try{ s.g.gain.setTargetAtTime(0.0001, t, .05); s.o.stop(t + .3); s.l.stop(t + .3); }catch(e){}
    this.siren = null;
  }
};

}

function renderShield(sh){
  const g = sh.cvs.getContext('2d');
  g.clearRect(0, 0, sh.cvs.width, sh.cvs.height);
  for(let r = 0; r < BR; r++) for(let c = 0; c < BC; c++){
    if(!sh.grid[r * BC + c]) continue;
    const x = c * CELL + 4, y = r * CELL + 4;
    g.shadowColor = '#3ddc84'; g.shadowBlur = 5;
    g.fillStyle = '#2fbf74'; g.fillRect(x, y, CELL - 1, CELL - 1);
    g.shadowBlur = 0;
    g.fillStyle = '#57e89a'; g.fillRect(x, y, CELL - 1, 2);
    g.fillStyle = 'rgba(0,40,22,.55)'; g.fillRect(x, y + CELL - 3, CELL - 1, 2);
  }
}
function buildShields(){
  const arr = [];
  for(let i = 0; i < 4; i++){
    const x = W * (i + 1) / 5 - (BC * CELL) / 2;
    const grid = new Uint8Array(BC * BR);
    for(let r = 0; r < BR; r++) for(let c = 0; c < BC; c++) grid[r * BC + c] = BUNKER[r][c] === 'X' ? 1 : 0;
    const sh = {x, y: H - 190, grid, cvs: document.createElement('canvas')};
    sh.cvs.width = BC * CELL + 8; sh.cvs.height = BR * CELL + 8;
    renderShield(sh);
    arr.push(sh);
  }
  return arr;
}
function erode(sh, cx, cy, up){
  const sgn = up ? -1 : 1;
  const offs = [[0,0,1],[-1,0,.85],[1,0,.85],[0,sgn,.9],[-1,sgn,.6],[1,sgn,.6],[0,2*sgn,.45]];
  let hit = false;
  for(const o of offs){
    const nx = cx + o[0], ny = cy + o[1];
    if(nx < 0 || ny < 0 || nx >= BC || ny >= BR) continue;
    const i = ny * BC + nx;
    if(sh.grid[i] && Math.random() < o[2]){ sh.grid[i] = 0; hit = true; }
  }
  if(hit){
    for(let k = 0; k < 4; k++)
      particles.push({x: sh.x + cx*CELL + 3, y: sh.y + cy*CELL + 3, vx: (Math.random()-.5)*90, vy: sgn*(30+Math.random()*70), t:.35, life:.35, size:2, color:'#3ddc84', grav:140});
    renderShield(sh);
    Snd.tick();
  }
}
function shieldHit(px, py, up){
  for(const sh of shields){
    const lx = Math.floor((px - sh.x) / CELL), ly = Math.floor((py - sh.y) / CELL);
    if(lx < 0 || ly < 0 || lx >= BC || ly >= BR) continue;
    const i = ly * BC + lx;
    if(sh.grid[i]){ erode(sh, lx, ly, up); return true; }
  }
  return false;
}

function spawnFormation(){
  const list = [];
  for(let r = 0; r < ROWS; r++) for(let c = 0; c < COLS; c++){
    list.push({c, r, type: r < 2 ? 0 : (r < 4 ? 1 : 2), alive: true});
  }
  return {list, x: 44, y: 116, dir: 1, t: 0, frame: 0, note: 0, total: list.length, alive: list.length};
}
const alienRect = a => { const w = TW[a.type]; return {x: F.x + a.c * GAPX - w/2, y: F.y + a.r * GAPY, w, h: 16}; };
function interval(){
  const k = F.alive / F.total;
  return Math.max(0.05, (0.075 + 0.545 * Math.pow(k, 1.35)) * Math.pow(0.92, wave - 1));
}
function chewShields(){
  for(const a of F.list){
    if(!a.alive) continue;
    const r = alienRect(a);
    for(const sh of shields){
      if(r.x >= sh.x + BC*CELL || r.x + r.w <= sh.x || r.y >= sh.y + BR*CELL || r.y + r.h <= sh.y) continue;
      const c0 = Math.max(0, ((r.x - sh.x) / CELL) | 0), c1 = Math.min(BC - 1, ((r.x + r.w - 1 - sh.x) / CELL) | 0);
      const r0 = Math.max(0, ((r.y - sh.y) / CELL) | 0), r1 = Math.min(BR - 1, ((r.y + r.h - 1 - sh.y) / CELL) | 0);
      let changed = false;
      for(let ry = r0; ry <= r1; ry++) for(let cx = c0; cx <= c1; cx++){
        const i = ry * BC + cx;
        if(sh.grid[i]){
          sh.grid[i] = 0; changed = true;
          if(Math.random() < .25) particles.push({x: sh.x + cx*CELL + 3, y: sh.y + ry*CELL + 3, vx:(Math.random()-.5)*60, vy:40+Math.random()*60, t:.4, life:.4, size:2, color:'#3ddc84', grav:120});
        }
      }
      if(changed) renderShield(sh);
    }
  }
}
function doStep(){
  F.frame ^= 1; F.note++;
  if(F.alive > 0) Snd.step(F.note);
  if(F.dir > 0 && F.x + (COLS-1) * GAPX + 12 > W - 30){ F.dir = -1; F.y += DROP; }
  else if(F.dir < 0 && F.x - 12 < 30){ F.dir = 1; F.y += DROP; }
  else F.x += STEP * F.dir;
  chewShields();
  if(F.y + (ROWS - 1) * GAPY + 16 >= player.y - 2) breach();
}

function updatePlayer(dt){
  player.cool -= dt; player.flash -= dt; player.invuln -= dt;
  let dir = 0;
  if(keys['ArrowLeft'] || keys['a']) dir--;
  if(keys['ArrowRight'] || keys['d']) dir++;
  if(pointerDown){
    const tx = pointerX - player.w / 2;
    dir = Math.abs(tx - player.x) > 5 ? (tx > player.x ? 1 : -1) : 0;
  }
  player.x += dir * 270 * dt;
  player.x = Math.max(12, Math.min(W - 12 - player.w, player.x));
  if(Math.random() < (dir !== 0 ? .9 : .3)){
    const nx = player.x + (dir < 0 ? 6 : dir > 0 ? player.w - 8 : 4) + Math.random() * 3;
    particles.push({x: nx, y: player.y + player.h + 2, vx: (Math.random()-.5)*24, vy: 80 + Math.random()*70, t: .22, life: .22, size: 2, color: '#4fd6ff', grav: 0});
  }

    pBullets.push({x: player.x + player.w/2, y: player.y - 14});

  }
}

        let best = null, ba = -1;
        for(const a of F.list) if(a.alive && a.c === c && a.r > ba){ ba = a.r; best = a; }
        if(best){
          const r = alienRect(best);
          aBullets.push({x: r.x + r.w/2, y: r.y + 18, vy: Math.min(330, 150 + wave * 12)});
          Snd.aShoot();

}

  if(!ufo){
    if(T > nextUfoAt && F && F.alive > 4){

      ufo = {x: left ? -30 : W + 30, dir: left ? 1 : -1, y: 86};

function updateBullets(dt){
  for(let i = pBullets.length - 1; i >= 0; i--){
    const b = pBullets[i]; let dead = false;
    for(let s = 0; s < 2 && !dead; s++){
      b.y -= 600 * dt / 2;
      if(b.y < 58){ dead = true; particles.push({x: b.x, y: 58, vx:(Math.random()-.5)*40, vy:-40, t:.15, life:.15, size:2, color:'#9fefff', grav:0}); }
      else if(shieldHit(b.x, b.y, true)) dead = true;
    }
    if(!dead && ufo && b.y < ufo.y + 14 && b.y > ufo.y - 10 && Math.abs(b.x - ufo.x) < 17){ killUFO(); dead = true; }
    if(!dead){
      for(const a of F.list){
        if(!a.alive) continue;
        const r = alienRect(a);
        if(b.x > r.x - 2 && b.x < r.x + r.w + 2 && b.y < r.y + r.h && b.y + 8 > r.y){ killAlien(a, r); dead = true; break; }
      }
    }
    if(dead) pBullets.splice(i, 1);
  }
  for(let i = aBullets.length - 1; i >= 0; i--){
    const b = aBullets[i]; let dead = false;
    for(let s = 0; s < 2 && !dead; s++){
      b.y += b.vy * dt / 2;
      if(b.y > H - 56) dead = true;
      else if(shieldHit(b.x, b.y, false)) dead = true;
    }
    if(!dead && player.invuln <= 0){
      const px = player.x + 3, pw = player.w - 6;
      if(b.x > px - 2 && b.x < px + pw + 2 && b.y > player.y - 2 && b.y < player.y + player.h + 4){ dead = true; playerHit(); }
    }
    if(dead) aBullets.splice(i, 1);
  }
}
function updateParticles(dt){
  for(let i = particles.length - 1; i >= 0; i--){
    const p = particles[i];
    p.t -= dt;
    if(p.t <= 0){ particles.splice(i, 1); continue; }
    p.x += p.vx * dt; p.y += p.vy * dt; p.vy += p.grav * dt;
    p.vx *= Math.max(0, 1 - 1.6 * dt);
  }
}
function updateFloats(dt){
  for(let i = floats.length - 1; i >= 0; i--){
    const f = floats[i];
    f.t -= dt; f.y += f.vy * dt;
    if(f.t <= 0) floats.splice(i, 1);
  }
}

function tick(dt){
  T += dt;
  if(shakeT > 0) shakeT = Math.max(0, shakeT - dt);
  if(state === 'paused') return;
  if(state === 'start'){ updateParticles(dt); updateFloats(dt); return; }
  if(state === 'wave'){
    formAlpha = Math.min(1, formAlpha + dt / 0.5);
    waveT -= dt;
    updatePlayer(dt); updateParticles(dt); updateFloats(dt);
    if(waveT <= 0) state = 'playing';
    return;
  }
  if(state === 'over'){ overT -= dt; updateParticles(dt); updateFloats(dt); return; }

  /* --- playing --- */
  updatePlayer(dt);
  const iv = interval();
  F.t += dt;
  while(F.t >= iv){ F.t -= iv; doStep(); if(state !== 'playing') return; }
  if(clearT > 0){
    clearT -= dt;
    if(clearT <= 0){ startWave(wave + 1); return; }
  } else if(F.alive === 0){
    clearT = 1.6;
    const bonus = 100 + (wave - 1) * 50;
    addScore(bonus);
    addFloat('WAVE CLEARED', W/2, 416, '#3ddc84', 3);
    addFloat('+' + bonus, W/2, 452, '#8effb9', 2);
    Snd.waveClear();
  }
  updateAlienFire(dt);
  stepUFO(dt);
  updateBullets(dt);
  updateParticles(dt);
  updateFloats(dt);
}

{
  const cols = ['rgba(205,224,255,1)','rgba(205,224,255,1)','rgba(205,224,255,1)','#7fd8ff','#ffd9a0','#ff9ecb'];
  for(let i = 0; i < 150; i++){
    const l = i % 3;
    stars.push({
      x: Math.random(), y: Math.random(), l,
      v: [.006,.013,.028][l] * (.7 + Math.random() * .6),
      sz: l === 2 ? 2 : (l === 1 ? 1.5 : 1),
      tw: 1 + Math.random() * 3, ph: Math.random() * 6,
      col: cols[(Math.random() * cols.length) | 0]
    });
  }
}
function blob(x, y, r, col){
  const g = ctx.createRadialGradient(x, y, 0, x, y, r);
  g.addColorStop(0, col); g.addColorStop(1, 'rgba(0,0,0,0)');
  ctx.fillStyle = g; ctx.fillRect(x - r, y - r, r * 2, r * 2);
}
function drawBG(){
  ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
  ctx.fillStyle = bgGrad; ctx.fillRect(0, 0, vw, vh);
  ctx.globalCompositeOperation = 'lighter';
  blob(vw * .24 + Math.sin(T * .06) * 70, vh * .3 + Math.cos(T * .05) * 40, Math.max(vw, vh) * .38, 'rgba(36,130,150,.10)');
  blob(vw * .76 + Math.cos(T * .045) * 90, vh * .55 + Math.sin(T * .06) * 50, Math.max(vw, vh) * .42, 'rgba(96,64,168,.10)');
  blob(vw * .5, vh * .85 + Math.sin(T * .07) * 30, Math.max(vw, vh) * .3, 'rgba(170,64,120,.07)');
  for(const s of stars){
    const x = s.x * vw, y = ((s.y + T * s.v) % 1) * vh;
    ctx.globalAlpha = (s.l === 2 ? .9 : s.l === 1 ? .55 : .3) * (.6 + .4 * Math.sin(T * s.tw + s.ph));
    ctx.fillStyle = s.col;
    ctx.fillRect(x, y, s.sz, s.sz);
  }
  ctx.globalAlpha = 1;
  ctx.globalCompositeOperation = 'source-over';
}

}

function render(){
  drawBG();
  let sx = 0, sy = 0;
  if(shakeT > 0){
    const k = shakeT / shakeDur;
    sx = (Math.random() * 2 - 1) * shakeMag * k;
    sy = (Math.random() * 2 - 1) * shakeMag * k;
  }
  ctx.setTransform(dpr * scale, 0, 0, dpr * scale, dpr * (ox + sx), dpr * (oy + sy));
  ctx.fillStyle = 'rgba(8,16,34,.35)'; ctx.fillRect(0, 0, W, H);
  drawFrame();
  if(state !== 'start'){
    drawHUD();
    drawShields();
    drawAliens();
    drawUfo();
    drawBullets();
    drawPlayer();
  }
  drawParticles();
  drawFloats();
  if(state === 'start') drawStart();
  else if(state === 'wave') drawWaveBanner();
  else if(state === 'paused') drawPaused();
  else if(state === 'over') drawOver();

  /* screen-space CRT pass */
  ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
  const sbY = (T * 36) % (vh + 240) - 120;
  const g = ctx.createLinearGradient(0, sbY, 0, sbY + 120);
  g.addColorStop(0, 'rgba(140,215,255,0)');
  g.addColorStop(.5, 'rgba(140,215,255,.045)');
  g.addColorStop(1, 'rgba(140,215,255,0)');
  ctx.fillStyle = g; ctx.fillRect(0, sbY, vw, 120);
  ctx.fillStyle = scanPat; ctx.fillRect(0, 0, vw, vh);
  ctx.fillStyle = vigGrad; ctx.fillRect(0, 0, vw, vh);
}

const keys = {};
let pointerDown = false, pointerFire = false, pointerX = 0;

window.addEventListener('keydown', e => {
  const k = e.key;
  if(k === ' ' || k === 'ArrowLeft' || k === 'ArrowRight' || k === 'ArrowUp' || k === 'ArrowDown') e.preventDefault();
  Snd.ensure(); Snd.resume();
  const key = k.length === 1 ? k.toLowerCase() : k;
  keys[key] = true;
  if(e.repeat) return;
  if(key === 'Enter'){
    if(state === 'start') newGame();
    else if(state === 'over' && overT < 1.2) newGame();
    else if(state === 'paused') state = 'playing';
  }
  if(key === 'r') newGame();
  if(key === 'p'){
    if(state === 'playing') state = 'paused';
    else if(state === 'paused') state = 'playing';
  }
  if(key === 'm'){
    Snd.muted = !Snd.muted;
    if(Snd.muted) Snd.sirenOff();
    else if(ufo) Snd.sirenOn();
  }
});
window.addEventListener('keyup', e => {
  const k = e.key;
  keys[k.length === 1 ? k.toLowerCase() : k] = false;
});
cvs.addEventListener('pointerdown', e => {
  Snd.ensure(); Snd.resume();
  const fx = (e.clientX - ox) / scale;
  if(state === 'start'){ newGame(); return; }
  if(state === 'over' && overT < 1.2){ newGame(); return; }
  if(state === 'paused'){ state = 'playing'; return; }
  pointerDown = true; pointerFire = true; pointerX = fx;
});
window.addEventListener('pointermove', e => { if(pointerDown) pointerX = (e.clientX - ox) / scale; });
['pointerup','pointercancel'].forEach(t => window.addEventListener(t, () => { pointerDown = false; pointerFire = false; }));
window.addEventListener('blur', () => {
  for(const k in keys) keys[k] = false;
  if(state === 'playing') state = 'paused';
});
document.addEventListener('visibilitychange', () => { if(document.hidden && state === 'playing') state = 'paused'; });

function resize(){
  dpr = Math.min(window.devicePixelRatio || 1, 2);
  vw = window.innerWidth; vh = window.innerHeight;
  cvs.width = Math.round(vw * dpr); cvs.height = Math.round(vh * dpr);
  scale = Math.min(vw / W, vh / H);
  ox = (vw - W * scale) / 2; oy = (vh - H * scale) / 2;
  ctx.imageSmoothingEnabled = false;
  bgGrad = ctx.createLinearGradient(0, 0, 0, vh);
  bgGrad.addColorStop(0, '#0b1630');
  bgGrad.addColorStop(.55, '#071022');
  bgGrad.addColorStop(1, '#04070f');
  vigGrad = ctx.createRadialGradient(vw/2, vh/2, Math.min(vw, vh) * .42, vw/2, vh/2, Math.hypot(vw, vh) * .6);
  vigGrad.addColorStop(0, 'rgba(0,0,0,0)');
  vigGrad.addColorStop(1, 'rgba(2,4,12,.55)');
}
window.addEventListener('resize', resize);
