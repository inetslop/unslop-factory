  tracers.push({x0:p0.x,y0:p0.y,x1:p1.x,y1:p1.y,t0:T,dur:.13,hit:!!target,rock:target||null,done:false});
  updateHud();
}
function brace(){
  if(!canAct()) return;
  const b=$('#bBrace');
  if(game.braced) return deny('ALREADY BRACED THIS TURN',b);
  if(game.ap<1) return deny('OUT OF ENERGY — END TURN [E]',b);
  game.ap--; game.braced=true; sfx.brace();
  const p=cellPt(game.ship.x,game.ship.y);
  parts.push({k:'ring',x:p.x,y:p.y,r:cell*.6,t0:T,life:.4,color:'#5cffc0'});
  say('BRACING — NEXT IMPACT ABSORBED'); updateHud();
}
function endTurn(){
  if(mode!=='play'||phase!=='player'||paused||!game||game.over) return;
  phase='ending'; sfx.phase(); updateHud(); later(260,enemyPhase);
}

/* ============================== ENEMY PHASE ============================== */
function stepRock(r){
  const s=game.ship, dx=Math.sign(s.x-r.x), dy=Math.sign(s.y-r.y);
  if(!dx&&!dy) return;
  const cands=[];
  if(Math.abs(s.x-r.x)>=Math.abs(s.y-r.y)){if(dx)cands.push([dx,0]);if(dy)cands.push([0,dy]);}
  else {if(dy)cands.push([0,dy]);if(dx)cands.push([dx,0]);}
  for(const [mx,my] of cands){
    const nx=r.x+mx,ny=r.y+my;
    if(nx<0||ny<0||nx>=COLS||ny>=ROWS) continue;
    if(rockAt(nx,ny)) continue; // ship cell is allowed → collision
    r.x=nx;r.y=ny;return;
  }
}
function enemyPhase(){
  if(mode!=='play'||game.over) return;
  phase='enemy'; updateHud(); say('ROCKS ADVANCE…');
  const order=game.rocks.slice().sort(()=>Math.random()-.5);
  let slot=430, end=0;
  for(const r of order){
    const extra=r.size===1?280:0;
    end=Math.max(end,slot+extra);
    later(slot,()=>{
      if(mode!=='play'||phase!=='enemy'||game.over) return;
      stepRock(r);
      if(r.size===1&&!(r.x===game.ship.x&&r.y===game.ship.y))
        later(150,()=>{if(mode==='play'&&phase==='enemy'&&!game.over)stepRock(r);});
    });
    slot+=95;
  }
  later(end+320,resolvePhase);
}
function resolvePhase(){
  if(mode!=='play'||game.over) return;
  const s=game.ship;
  const victims=game.rocks.filter(r=>r.x===s.x&&r.y===s.y);
  if(victims.length){
    for(const v of victims) destroyRock(v,true);
    addShake(15);
    if(game.braced){banner('SHIELD HELD','gold');sfx.shield();say('BRACE ABSORBED THE IMPACT');}
    else{
      game.hull--; sfx.breach(); banner('HULL BREACH','red');
      const ht=$('#hudTop');ht.classList.remove('hit');void ht.offsetWidth;ht.classList.add('hit');
      say(victims.length>1?'MULTIPLE IMPACTS — ONE BREACH':'IMPACT! HULL -1');
    }
    updateHud();
    if(game.hull<=0){gameOver();return;}
  }
  game.braced=false;
  maybeWaveClear();
  if(mode==='play'&&!game.over&&phase!=='interlude'){phase='player';game.ap=AP_MAX;updateHud();say('YOUR MOVE');}
}
function maybeWaveClear(){
  if(!game||game.over||mode!=='play') return;
  if(game.rocks.length>0||phase==='interlude') return;
  phase='interlude'; updateHud();
  const bonus=150*game.wave; game.score+=bonus;
  if(game.hull<HULL_MAX) game.hull++;
  banner('WAVE '+String(game.wave).padStart(2,'0')+' CLEAR','gold');
  say('BONUS +'+bonus+' · HULL PATCHED +1'); sfx.waveClear(); updateHud();
  later(1900,nextWave);
}
function nextWave(){
  if(mode!=='play'||game.over) return;
  game.wave++; spawnWave(2+Math.ceil(game.wave/2));
  phase='player'; game.ap=AP_MAX; updateHud();
  banner('WAVE '+String(game.wave).padStart(2,'0'),'cyan'); sfx.waveStart();
  say('NEW ROCKS INBOUND — YOUR MOVE');
}

/* ============================== DAMAGE / EFFECTS ============================== */
function addShake(m){shake=Math.max(shake,m);}
function hitRock(r){
  r.hp--;
  if(r.hp>0){r.dmg=true;sfx.impact(2);addShake(3);return;}
  destroyRock(r,false);
}
function destroyRock(r,ram){
  const i=game.rocks.indexOf(r); if(i<0) return;
  game.rocks.splice(i,1);
  const p=cellPt(r.x,r.y), c=ROCKC[r.size];
  if(!ram){
    const pts=PTS[r.size]; game.score+=pts; game.kills++;
    popups.push({x:p.x,y:p.y-cell*.2,txt:'+'+pts,t0:T,life:.95,color:c});
  }
  boomFX(p.x,p.y,c,r.size===3?2.4:r.size===2?1.6:1);
  sfx.boom(r.size); addShake(2+r.size*1.6);
  if(r.size>1){placeChild(r,r.size-1);placeChild(r,r.size-1);}
  updateHud(); maybeWaveClear();
}
function boomFX(x,y,color,scale){
  parts.push({k:'ring',x,y,r:cell*(.25+.3*scale),t0:T,life:.45,color});
  const n=Math.round(7*scale);
  for(let i=0;i<n;i++){const a=rand(0,TAU),sp=rand(40,150)*Math.min(scale,2);
    parts.push({k:'shard',x,y,vx:Math.cos(a)*sp,vy:Math.sin(a)*sp,rot:a,vr:rand(-6,6),
      len:rand(6,16)*Math.min(scale,2),t0:T,life:rand(.4,.85),color});}
  for(let i=0;i<Math.round(5*scale);i++){const a=rand(0,TAU),sp=rand(30,120)*scale;
    parts.push({k:'spark',x,y,vx:Math.cos(a)*sp,vy:Math.sin(a)*sp,size:rand(1.5,3),t0:T,life:rand(.25,.5),
      color:i%2?'#ffffff':color});}
  if(parts.length>420) parts.splice(0,parts.length-420);
}

/* ============================== UPDATE ============================== */
function update(dt){
  if(mode==='menu'||mode==='over'&&!game){
    for(const a of ambient){a.x+=a.vx*dt;a.y+=a.vy*dt;a.rot+=a.vr*dt;
      if(a.x<-90)a.x=W+80;if(a.x>W+90)a.x=-80;if(a.y<-90)a.y=H+80;if(a.y>H+90)a.y=-80;}
  }
  if(game&&(mode==='play'||mode==='over')){
    const s=game.ship;
    const k=1-Math.exp(-12*dt);
    s.vx+=(s.x-s.vx)*k; s.vy+=(s.y-s.vy)*k;
    let da=((s.facing*Math.PI/4)-s.ang)%TAU;
    if(da>Math.PI)da-=TAU; if(da<-Math.PI)da+=TAU;
    s.ang+=da*Math.min(1,dt*14);
    const kr=1-Math.exp(-9*dt);
    for(const r of game.rocks){r.vx+=(r.x-r.vx)*kr;r.vy+=(r.y-r.vy)*kr;r.rot+=r.vrot*dt;}
  }
  if(mode==='play'){
    for(const t of tracers){
      if(!t.done&&T-t.t0>=t.dur){
        t.done=true;
        parts.push({k:'ring',x:t.x1,y:t.y1,r:cell*.3,t0:T,life:.25,color:'#ffe9a8'});
        for(let i=0;i<6;i++){const a=rand(0,TAU),sp=rand(40,140);
          parts.push({k:'spark',x:t.x1,y:t.y1,vx:Math.cos(a)*sp,vy:Math.sin(a)*sp,size:rand(1.2,2.6),t0:T,life:.3,color:'#ffe9a8'});}
        if(t.rock&&game.rocks.includes(t.rock)) hitRock(t.rock);
      }
    }
    tracers=tracers.filter(t=>T-t.t0<t.dur+.12);
  }
  const dr=Math.exp(-3*dt);
  for(const p of parts){if(p.k!=='ring'){p.x+=p.vx*dt;p.y+=p.vy*dt;p.vx*=dr;p.vy*=dr;} if(p.rot!==undefined)p.rot+=p.vr*dt;}
  parts=parts.filter(p=>T-p.t0<p.life);
  popups=popups.filter(p=>T-p.t0<p.life);
  shake=Math.max(0,shake-dt*26);
  if(game&&mode==='play'){
    scoreDisp+=(game.score-scoreDisp)*(1-Math.exp(-8*dt));
    if(game.score===Math.round(scoreDisp))scoreDisp=game.score;
    const sstr=pad6(scoreDisp);
    if(sstr!==lastScoreStr){lastScoreStr=sstr;scoreEl.textContent=sstr;}
  }
}
