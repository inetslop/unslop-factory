function step(dt){
  const g=2600*S, damp=0.995, L=segLen();
  // integrate
  for(const s of segs){
    let vx=(s.x-s.px)*damp, vy=(s.y-s.py)*damp;
    s.px=s.x; s.py=s.y;
    s.x+=vx+grav.x*g*dt*dt;
    s.y+=vy+grav.y*g*dt*dt;
  }
  const iters=12;
  for(let k=0;k<iters;k++){
    for(let i=0;i<segs.length-1;i++) constrain(segs[i],segs[i+1],L);
    for(const s of segs) walls(s, s===segs[0]?0.45:0.2, 0.86);
  }
  // soft separation body-body
  const r=bodyR(), minD=r*1.75, md2=minD*minD;
  for(let i=1;i<segs.length;i++){const A=segs[i];
    for(let j=i+2;j<segs.length;j++){const B=segs[j];
      const dx=B.x-A.x, dy=B.y-A.y, d2=dx*dx+dy*dy;
      if(d2<md2&&d2>1e-9){const d=Math.sqrt(d2), push=(minD-d)/d*0.5*0.4; A.x-=dx*push;A.y-=dy*push;B.x+=dx*push;B.y+=dy*push;}
    }}
  if(state==='play'){
    // eat
    const hd=segs[0];
    if(food&&dist2(hd.x,hd.y,food.x,food.y)<(hd.r+food.r)**2) eat();
    // expire

}

    const t=segs[segs.length-1];
    for(let i=0;i<2;i++) segs.push({x:t.x+rand(-1,1),y:t.y+rand(-1,1),px:t.x,py:t.y});
  }
  spawnFood(); sfxEat(foodsEaten); updateHUD(true);
}
function die(){
  if(state!=='play')return;
  state='over'; overShown=false;
  shake=16; flash=1; flashCol='rgba(255,90,70,1)';
  for(let i=0;i<segs.length;i+=2) burst(segs[i].x,segs[i].y,3,'cyan',.5);
  sfxDie();
  newBestFlag = score>best;
  if(newBestFlag){ best=score; try{localStorage.setItem('gravitySnakeBest',String(best));}catch(e){} }
  updateHUD(false);
  setTimeout(()=>{ if(state==='over'&&!overShown){ overShown=true; showOver(); } },800);
}
function showOver(){
  finalScore.textContent=pad(score);
  statMotes.textContent=foodsEaten;
  statLen.textContent=segs.length;
  statBest.textContent=pad(best);
  newBest.classList.toggle('hidden',!newBestFlag);
  overOv.classList.remove('hidden');
  startLockUntil=time+0.6;
  if(newBestFlag) sfxBest();
}

function startGame(){
  if(state==='play')return;
  buildSnake(8);
  score=0; foodsEaten=0; newBestFlag=false;
  particles.length=0; pops.length=0; trail.length=0;
  shake=0; flash=0; eatPulse=0; overShown=false;
  spawnFood();
  const h=segs[0]; let dx=food.x-h.x, dy=food.y-h.y;
  if(Math.hypot(dx,dy)<1){dx=0;dy=1;}
  grav = (Math.abs(dx)>Math.abs(dy)) ? {x:Math.sign(dx),y:0} : {x:0,y:Math.sign(dy)||1};
  compassAngle();
  state='play';
  menu.classList.add('hidden'); pauseOv.classList.add('hidden'); overOv.classList.add('hidden');
  hud.classList.remove('hidden');
  updateHUD(false); sfxUI();
}

}
function setGravity(dx,dy){

  grav={x:dx,y:dy}; compassAngle(); sfxFlip();
  const h=segs[0]; if(h) ring(h.x,h.y);
  shake=Math.min(shake+1.5,6);
}

let actx=null, master=null, muted=false, noiseBuf=null;
function initAudio(){
  if(actx)return;
  try{
    const AC=window.AudioContext||window.webkitAudioContext; if(!AC)return;
    actx=new AC(); master=actx.createGain(); master.gain.value=muted?0:0.55; master.connect(actx.destination);
    noiseBuf=actx.createBuffer(1,actx.sampleRate*0.5,actx.sampleRate);
    const d=noiseBuf.getChannelData(0); for(let i=0;i<d.length;i++)d[i]=Math.random()*2-1;
  }catch(e){}
}
function resumeAC(){ if(actx&&actx.state==='suspended') actx.resume(); }
function tone(f0,f1,dur,type,vol,delay=0){
  if(!actx||muted)return;
  const t=actx.currentTime+delay, o=actx.createOscillator(), g=actx.createGain();
  o.type=type; o.frequency.setValueAtTime(Math.max(1,f0),t);
  if(f1) o.frequency.exponentialRampToValueAtTime(Math.max(1,f1),t+dur);
  g.gain.setValueAtTime(0.0001,t);
  g.gain.exponentialRampToValueAtTime(vol,t+0.012);
  g.gain.exponentialRampToValueAtTime(0.0001,t+dur);
  o.connect(g).connect(master); o.start(t); o.stop(t+dur+.05);
}
function noiseHit(dur,vol,fFrom,fTo){
  if(!actx||muted||!noiseBuf)return;
  const t=actx.currentTime, s=actx.createBufferSource(), f=actx.createBiquadFilter(), g=actx.createGain();
  s.buffer=noiseBuf; f.type='lowpass';
  f.frequency.setValueAtTime(fFrom,t); f.frequency.exponentialRampToValueAtTime(Math.max(40,fTo),t+dur);
  g.gain.setValueAtTime(vol,t); g.gain.exponentialRampToValueAtTime(0.0001,t+dur);
  s.connect(f).connect(g).connect(master); s.start(t); s.stop(t+dur+.05);
}
const sfxFlip =()=>{ tone(160,70,.1,'triangle',.5); tone(900,300,.05,'square',.1); };
const sfxEat  =n=>{ const b=440*Math.pow(1.0595,n%8); tone(b,b*1.5,.09,'sine',.5); tone(b*1.5,b*2,.12,'sine',.32,.06); };
const sfxExpire=()=>{ tone(240,140,.25,'sine',.22); };
const sfxDie  =()=>{ noiseHit(.5,.55,2200,120); tone(320,40,.7,'sawtooth',.45); tone(160,30,.9,'triangle',.35,.08); };
const sfxUI   =()=>{ tone(600,900,.06,'square',.18); };
const sfxBest =()=>{ [523,659,784].forEach((f,i)=>tone(f,f,.14,'sine',.38,i*.09)); };

/* ================= UPDATE ================= */
function update(dt){
  const tgt={t:grav.y<0?1:0,b:grav.y>0?1:0,l:grav.x<0?1:0,r:grav.x>0?1:0};
  for(const k of 'tblr') wallHeat[k]+=(tgt[k]-wallHeat[k])*Math.min(1,dt*9);
  shake*=Math.pow(0.02,dt); if(shake<.05)shake=0;
  flash=Math.max(0,flash-dt*2.4);
  eatPulse=Math.max(0,eatPulse-dt*3);
  for(let i=dust.length-1;i>=0;i--){const d=dust[i];
    d.x+=grav.x*34*d.z*dt+Math.sin(time*.5+d.z*9)*6*dt;
    d.y+=grav.y*34*d.z*dt+Math.cos(time*.4+d.z*7)*6*dt;
    if(d.x<-10)d.x=W+10; if(d.x>W+10)d.x=-10;
    if(d.y<-10)d.y=H+10; if(d.y>H+10)d.y=-10;
  }
  for(let i=particles.length-1;i>=0;i--){const p=particles[i];
    p.life-=dt; if(p.life<=0){particles.splice(i,1);continue;}
    if(p.ring){p.r+=320*S*dt;continue;}
    p.vx*=0.985; p.vy=p.vy*0.985+grav.y*620*S*dt;
    p.x+=p.vx*dt; p.y+=p.vy*dt;
  }
  for(let i=pops.length-1;i>=0;i--){const p=pops[i];
    p.t+=dt; p.y-=44*S*dt; if(p.t>0.9)pops.splice(i,1);
  }
}

/* ================= RENDER ================= */
function render(){
  ctx.fillStyle=bgGrad; ctx.fillRect(0,0,W,H);
  const sx=(Math.random()*2-1)*shake, sy=(Math.random()*2-1)*shake;
  ctx.save(); ctx.translate(sx,sy);

  // grid
  const cell=54*S;
  ctx.strokeStyle='rgba(77,245,201,.045)'; ctx.lineWidth=1; ctx.beginPath();
  for(let x=arena.x+cell;x<arena.x+arena.w;x+=cell){ctx.moveTo(x,arena.y);ctx.lineTo(x,arena.y+arena.h);}
  for(let y=arena.y+cell;y<arena.y+arena.h;y+=cell){ctx.moveTo(arena.x,y);ctx.lineTo(arena.x+arena.w,y);}
  ctx.stroke();

  // arena tint
  ctx.fillStyle='rgba(10,25,38,.35)'; ctx.fillRect(arena.x,arena.y,arena.w,arena.h);

  // gravity glow from the active wall
  const pulse=.7+.3*Math.sin(time*1.4), depth=Math.min(arena.w,arena.h)*0.45;
  let gg;
  if(grav.y>0)   gg=ctx.createLinearGradient(0,arena.y+arena.h,0,arena.y+arena.h-depth);
  else if(grav.y<0)gg=ctx.createLinearGradient(0,arena.y,0,arena.y+depth);
  else if(grav.x>0)gg=ctx.createLinearGradient(arena.x+arena.w,0,arena.x+arena.w-depth,0);
  else             gg=ctx.createLinearGradient(arena.x,0,arena.x+depth,0);
  gg.addColorStop(0,`rgba(77,245,201,${0.10*pulse})`); gg.addColorStop(1,'rgba(77,245,201,0)');
  ctx.save(); ctx.globalCompositeOperation='lighter';
  ctx.fillStyle=gg; ctx.fillRect(arena.x,arena.y,arena.w,arena.h);
  // dust
  for(const d of dust){ ctx.globalAlpha=.05+.13*d.z; ctx.fillStyle='#9be8d8'; circ(d.x,d.y,1.6*d.z); }
  ctx.restore();

  // walls
  ctx.lineWidth=2; ctx.strokeStyle='rgba(77,245,201,.18)'; rr(arena.x-6,arena.y-6,arena.w+12,arena.h+12,14); ctx.stroke();
  const sides=[[arena.x,arena.y,arena.x+arena.w,arena.y,'t'],[arena.x,arena.y+arena.h,arena.x+arena.w,arena.y+arena.h,'b'],
               [arena.x,arena.y,arena.x,arena.y+arena.h,'l'],[arena.x+arena.w,arena.y,arena.x+arena.w,arena.y+arena.h,'r']];
  for(const [ax,ay,bx,by,id] of sides){
    const heat=wallHeat[id];
    if(heat>0.02){ ctx.strokeStyle=`rgba(255,179,71,${0.85*heat})`; ctx.lineWidth=3;
      ctx.beginPath();ctx.moveTo(ax,ay);ctx.lineTo(bx,by);ctx.stroke(); }
  }

  if(food) drawFood();
  drawSnake();

  // particles
  ctx.save(); ctx.globalCompositeOperation='lighter';
  for(const p of particles){
    const a=Math.max(0,p.life/p.max), col=PCOL[p.color]||'#fff';
    if(p.ring){ ctx.globalAlpha=a*.8; ctx.strokeStyle=col; ctx.lineWidth=2;
      ctx.beginPath();ctx.arc(p.x,p.y,p.r,0,TAU);ctx.stroke(); }
    else { ctx.globalAlpha=a; ctx.fillStyle=col; circ(p.x,p.y,p.size*(.5+.5*a)); }
  }
  ctx.restore();

  // pops
  for(const p of pops){
    const a=1-p.t/0.9;
    ctx.globalAlpha=a; ctx.fillStyle='#ffd9a0';
    ctx.font=`700 ${Math.round(13+8*S)}px ui-monospace,Menlo,Consolas,monospace`;
    ctx.textAlign='center'; ctx.fillText(p.txt,p.x,p.y);
  }
  ctx.globalAlpha=1;
  ctx.restore();

  // flash + vignette (screen space)
  if(flash>0){ ctx.globalAlpha=flash*.28; ctx.fillStyle=flashCol; ctx.fillRect(0,0,W,H); ctx.globalAlpha=1; }
  ctx.fillStyle=vign; ctx.fillRect(0,0,W,H);
}
function drawFood(){
  const f=food, pulse=1+0.1*Math.sin(time*5+f.x*.01);
  let alpha=1; if(f.life<2.5) alpha=.35+.65*Math.abs(Math.sin(time*9));
  ctx.save(); ctx.globalCompositeOperation='lighter';
  drawGlow(glowAmber,f.x,f.y,f.r*7*pulse,.8*alpha);
  ctx.restore();
  ctx.globalAlpha=alpha;
  ctx.fillStyle='#ffd9a0'; circ(f.x,f.y,f.r*(.9+.15*Math.sin(time*6)));
  ctx.fillStyle='#fff'; circ(f.x,f.y,f.r*.38);
  const frac=clamp(f.life/f.maxLife,0,1);
  ctx.strokeStyle=frac>0.35?'rgba(77,245,201,.75)':'rgba(255,106,85,.9)';
  ctx.lineWidth=2.5;
  ctx.beginPath(); ctx.arc(f.x,f.y,f.r*1.7,-Math.PI/2,-Math.PI/2+frac*TAU); ctx.stroke();
  ctx.globalAlpha=1;
}
function drawSnake(){
  const n=segs.length; if(!n)return;
  // trail
  ctx.save(); ctx.globalCompositeOperation='lighter';
  for(let i=0;i<trail.length;i++){
    const t=i/trail.length, p=trail[i];
    ctx.globalAlpha=t*.10; ctx.fillStyle='#5ffbe0'; circ(p.x,p.y,2+t*headR()*.6);
  }
  ctx.restore();
  // soft glow pass
  ctx.save(); ctx.globalCompositeOperation='lighter';
  for(let i=n-1;i>0;i-=2){const s=segs[i];
    const rad=(i===0?headR():bodyR()*(0.62+0.38*(1-i/(n-1))));
    ctx.fillStyle='rgba(77,245,201,.05)'; circ(s.x,s.y,rad*2.4);
  }
  ctx.restore();
  // body tail -> head
  for(let i=n-1;i>=1;i--){
    const s=segs[i], t=i/(n-1), rad=bodyR()*(0.62+0.38*(1-t));
    const hue=170+t*35, sat=90-t*30, lig=58-t*24;
    ctx.fillStyle=`hsl(${hue},${sat}%,${lig}%)`; circ(s.x,s.y,rad);
    ctx.fillStyle=`hsla(${hue},${sat}%,${Math.min(92,lig+26)}%,.85)`; circ(s.x,s.y,rad*.42);
  }
  // head
  const h=segs[0], hr=headR()*(1+0.35*eatPulse);
  ctx.save(); ctx.globalCompositeOperation='lighter';
  drawGlow(glowCyan,h.x,h.y,hr*8,.9); ctx.restore();
  ctx.fillStyle='#eafff7'; circ(h.x,h.y,hr);
  ctx.strokeStyle='rgba(77,245,201,.9)'; ctx.lineWidth=2;
  ctx.beginPath();ctx.arc(h.x,h.y,hr,0,TAU);ctx.stroke();
  // eyes
  let vx=h.x-h.px, vy=h.y-h.py, sp=Math.hypot(vx,vy), lx=0, ly=1;
  if(sp>0.5){lx=vx/sp;ly=vy/sp;}
  else if(food){const dx=food.x-h.x,dy=food.y-h.y,dd=Math.hypot(dx,dy)||1;lx=dx/dd;ly=dy/dd;}
  const px=-ly, py=lx;
  for(const sgn of [-1,1]){
    const ex=h.x+lx*hr*.35+px*sgn*hr*.42, ey=h.y+ly*hr*.35+py*sgn*hr*.42;
    ctx.fillStyle='#fff'; circ(ex,ey,hr*.30);
    ctx.fillStyle='#08222b'; circ(ex+lx*hr*.10,ey+ly*hr*.10,hr*.16);
  }
}

/* ================= LOOP / BOOT ================= */
function frame(t){
  requestAnimationFrame(frame);
  const now=t/1000;
  let dt=Math.min(Math.max(now-lastT,0),.05); lastT=now; time+=dt;
  acc+=dt; let steps=0;
  while(acc>=DT&&steps<4){
    if(state==='play'||state==='menu') step();
    acc-=DT; steps++;
  }
  update(dt);
  render();
}
