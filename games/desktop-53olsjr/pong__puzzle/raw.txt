function tone(f,dur,type,vol,slide,delay){
  if(!AC||S.muted)return;
  try{
    const t0=AC.currentTime+(delay||0),o=AC.createOscillator(),g=AC.createGain();
    o.type=type||'square';o.frequency.setValueAtTime(f,t0);
    if(slide)o.frequency.exponentialRampToValueAtTime(Math.max(24,slide),t0+dur);
    g.gain.setValueAtTime(vol||.3,t0);g.gain.exponentialRampToValueAtTime(.0001,t0+dur);
    o.connect(g);g.connect(master);o.start(t0);o.stop(t0+dur+.03);
  }catch(e){}
}

/* ============ physics / AI ============ */
function movePad(p,sp,dt){
  const ty=zoneY(p.target),d=ty-p.y,m=sp*dt;
  p.y+=Math.abs(d)<=m?d:Math.sign(d)*m;
  p.flash=Math.max(0,(p.flash||0)-dt*3.2);
}
function predictZone(bx,by,vx,vy,faceX){
  if(Math.abs(vx)<1)return 1;
  const t=Math.max(0,(faceX-bx)/vx);
  let y=by+vy*t,top=COURT.y,h=COURT.h,per=2*h;
  y=(y-top)%per;if(y<0)y+=per;if(y>h)y=per-y;
  return clamp(Math.floor((y-top)/(h/3)),0,2);
}
function cpuErr(){return clamp(.2-S.score.you*.02,.07,.2);}
function cpuThink(dt){
  const b=ball;
  if(b.vx>0){
    S.cpuT-=dt;
    if(S.cpuT<=0){
      let z=predictZone(b.x,b.y,b.vx,b.vy,cpuFace);
      if(Math.random()<cpuErr())z=clamp(z+(Math.random()<.5?-1:1),0,2);
      P.cpu.target=z;S.cpuT=rand(.13,.3);
    }
  }else if(P.cpu.target!==1)P.cpu.target=1;
}
function menuAI(dt){
  const b=ball;
  if(b.vx<0){S.leftT-=dt;if(S.leftT<=0){let z=predictZone(b.x,b.y,b.vx,b.vy,playerFace);
    if(Math.random()<.28)z=clamp(z+(Math.random()<.5?-1:1),0,2);P.you.target=z;S.leftT=rand(.2,.45);}}
  else if(P.you.target!==1)P.you.target=1;
  if(b.vx>0){S.cpuT-=dt;if(S.cpuT<=0){let z=predictZone(b.x,b.y,b.vx,b.vy,cpuFace);
    if(Math.random()<.28)z=clamp(z+(Math.random()<.5?-1:1),0,2);P.cpu.target=z;S.cpuT=rand(.2,.45);}}
  else if(P.cpu.target!==1)P.cpu.target=1;
}
function bouncePad(side,yc){
  const p=side==='you'?P.you:P.cpu;
  const off=clamp((yc-p.y)/(PH/2),-1,1);
  if(S.mode!=='menu')S.bounces++;
  ball.speed=Math.min(ball.speed*1.045+6,980);
  const ang=off*1.05,dir=side==='you'?1:-1;
  ball.vx=Math.cos(ang)*ball.speed*dir;ball.vy=Math.sin(ang)*ball.speed;
  if(side==='you')ball.x=playerFace+BALL_R;else ball.x=cpuFace-BALL_R;
  p.flash=1;S.shake=Math.max(S.shake,3);S.rallyPop=1;
  burst(ball.x,ball.y,side==='you'?'#3fe3ff':'#ff6f61',8,260);
  sfx.hit(ball.speed);
}
function hitPad(side,px,py){
  const b=ball,p=side==='you'?P.you:P.cpu;
  if(side==='you'){
    if(b.vx<0&&px-BALL_R>=playerFace&&b.x-BALL_R<playerFace){
      const d=(px-BALL_R)-(b.x-BALL_R);if(d<=0)return;
      const t=((px-BALL_R)-playerFace)/d,yc=py+(b.y-py)*t;
      if(Math.abs(yc-p.y)<=PH/2+BALL_R*.55)bouncePad('you',yc);
    }
  }else{
    if(b.vx>0&&px+BALL_R<=cpuFace&&b.x+BALL_R>=cpuFace){
      const d=(b.x+BALL_R)-(px+BALL_R);if(d<=0)return;
      const t=(cpuFace-(px+BALL_R))/d,yc=py+(b.y-py)*t;
      if(Math.abs(yc-p.y)<=PH/2+BALL_R*.55)bouncePad('cpu',yc);
    }
  }
}
function ballStep(dt){
  const b=ball;if(!b.visible)return;
  const px=b.x,py=b.y;
  b.x+=b.vx*dt;b.y+=b.vy*dt;
  trail.push({x:b.x,y:b.y});if(trail.length>15)trail.shift();
  if((px-SW/2)*(b.x-SW/2)<0)S.netPulse=1;
  if(b.y-BALL_R<COURT.y){b.y=COURT.y+BALL_R;b.vy=Math.abs(b.vy);burst(b.x,COURT.y+4,'#9fe8ff',5,160);sfx.wall();}
  else if(b.y+BALL_R>COURT.y+COURT.h){b.y=COURT.y+COURT.h-BALL_R;b.vy=-Math.abs(b.vy);burst(b.x,COURT.y+COURT.h-4,'#9fe8ff',5,160);sfx.wall();}
  hitPad('you',px,py);hitPad('cpu',px,py);
}

/* ============ update ============ */
function update(dt){
  S.time+=dt;S.dash-=dt*70;
  S.shake=Math.max(0,S.shake-dt*26);
  S.netPulse=Math.max(0,S.netPulse-dt*2.4);
  if(S.flash.t>0)S.flash.t=Math.max(0,S.flash.t-dt*2.8);
  S.pointGlow=Math.max(0,S.pointGlow-dt*1.5);
  S.popY=Math.max(0,S.popY-dt*3);S.popC=Math.max(0,S.popC-dt*3);S.rallyPop=Math.max(0,S.rallyPop-dt*3);
  for(let i=parts.length-1;i>=0;i--){const p=parts[i];p.x+=p.vx*dt;p.y+=p.vy*dt;p.vx*=.985;p.vy*=.985;p.life-=dt;if(p.life<=0)parts.splice(i,1);}
  for(let i=pops.length-1;i>=0;i--){const p=pops[i];p.t+=dt;p.y-=26*dt;if(p.t>1.35)pops.splice(i,1);}

  movePad(P.you,1500,dt);movePad(P.cpu,1200,dt);

  if(S.mode==='menu'){
    menuAI(dt);ballStep(dt);
    if(ball.x<COURT.x-70||ball.x>COURT.x+COURT.w+70)serveAttract();
    return;
  }
  if(['watch','recall','serve'].includes(S.mode))P.cpu.target=1;

  switch(S.mode){
    case 'watch':{
      S.watchT+=dt;
      while(S.fi<S.seq.length&&S.watchT>=S.flashAt[S.fi]){setFlash(S.seq[S.fi],'watch');sfx.zone(S.seq[S.fi]);S.fi++;}
      if(S.fi>=S.seq.length&&S.watchT>=S.flashAt[S.seq.length-1]+S.gap*.95){S.mode='recall';sfx.cue();}
      break;}
    case 'serve':{
      S.serveT-=dt;
      if(S.serveT<=0){launchServe();S.mode='rally';}
      break;}
    case 'rally':{cpuThink(dt);ballStep(dt);checkExit();break;}
    case 'point':{
      ball.x+=ball.vx*dt;ball.y+=ball.vy*dt;
      S.pointT-=dt;if(S.pointT<=0)afterPoint();
      break;}
  }
}

    const a=(.12+.3*Math.abs(Math.sin(t*1.4+s.tw)))*s.z;

  rr(ctx,COURT.x-14,COURT.y-14,COURT.w+28,COURT.h+28,20);

    ctx.fillStyle='rgba(255,111,97,.3)';
    ctx.beginPath();ctx.moveTo(COURT.x+COURT.w-84,y-7);ctx.lineTo(COURT.x+COURT.w-94,y);ctx.lineTo(COURT.x+COURT.w-84,y+7);ctx.closePath();ctx.fill();
  }

  // net (animated dashes + pulse)
  ctx.save();
  const na=.35+S.netPulse*.45;
  ctx.strokeStyle='rgba(150,230,255,'+na.toFixed(3)+')';
  ctx.lineWidth=3+S.netPulse*3;
  if(S.netPulse>0){ctx.shadowColor='#9fe8ff';ctx.shadowBlur=18*S.netPulse;}
  ctx.setLineDash([10,14]);ctx.lineDashOffset=S.dash;

  // puzzle flash band (player side)
  if(S.flash.t>0){
    const c=S.flash.kind==='watch'?'255,201,75':S.flash.kind==='ok'?'63,227,255':'255,84,112';
    const y=zoneY(S.flash.zone),h=PH*1.18;

    rr(ctx,COURT.x+10,y-h/2,250,h,14);ctx.stroke();
    ctx.shadowBlur=0;ctx.fillStyle='rgba('+c+','+(S.flash.t*.8).toFixed(3)+')';
    for(let i=-1;i<=1;i++){ctx.beginPath();ctx.arc(COURT.x+135+i*34,y,5,0,6.283);ctx.fill();}
    ctx.restore();
  }

  drawPad('you');drawPad('cpu');drawBall(t);

  // particles (additive)
  ctx.save();ctx.globalCompositeOperation='lighter';
  for(const p of parts){
    ctx.globalAlpha=Math.max(0,p.life/p.max);
    ctx.fillStyle=p.color;
    ctx.beginPath();ctx.arc(p.x,p.y,p.size,0,6.283);ctx.fill();
  }
  ctx.restore();ctx.globalAlpha=1;

  drawHUD();

  // popups
  for(const p of pops){
    const a=Math.max(0,1-p.t/1.35);

    ctx.font='900 42px Impact,"Arial Narrow",sans-serif';

    ctx.fillStyle=p.color;ctx.shadowColor=p.color;ctx.shadowBlur=18;ctx.fillText(p.text,p.x,p.y);
    ctx.restore();
  }
}
