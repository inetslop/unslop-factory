function tone(o){if(!AC||muted)return;const{f=440,f2=0,type='sine',t=0,a=.006,d=.2,v=.3}=o;
  const osc=AC.createOscillator(),g=AC.createGain(),T=AC.currentTime+t;
  osc.type=type;osc.frequency.setValueAtTime(f,T);
  if(f2)osc.frequency.exponentialRampToValueAtTime(Math.max(30,f2),T+a+d*.85);
  g.gain.setValueAtTime(0,T);g.gain.linearRampToValueAtTime(v,T+a);
  g.gain.exponentialRampToValueAtTime(.0001,T+a+d);
  osc.connect(g);g.connect(master);osc.start(T);osc.stop(T+a+d+.05);}

  const n=AC.createBufferSource(),len=Math.max(64,(d*AC.sampleRate)|0),b=AC.createBuffer(1,len,AC.sampleRate);

  const f=AC.createBiquadFilter();f.type=type;f.frequency.setValueAtTime(fc,AC.currentTime+t);

  g.gain.linearRampToValueAtTime(v,T+.006);g.gain.exponentialRampToValueAtTime(.0001,T+d);
  n.connect(f);f.connect(g);g.connect(master);n.start(T);}

function rr(c,x,y,w,h,r){r=Math.min(r,w/2,h/2);c.beginPath();
  c.moveTo(x+r,y);c.arcTo(x+w,y,x+w,y+h,r);c.arcTo(x+w,y+h,x,y+h,r);
  c.arcTo(x,y+h,x,y,r);c.arcTo(x,y,x+w,y,r);c.closePath();}
function star(c,x,y,R,r,n,rot){n=n||5;rot=(rot===undefined)?-Math.PI/2:rot;c.beginPath();
  for(let i=0;i<n*2;i++){const a=rot+i*Math.PI/n,rad=i%2?r:R;
    const px=x+Math.cos(a)*rad,py=y+Math.sin(a)*rad;i?c.lineTo(px,py):c.moveTo(px,py);}c.closePath();}

function suitPath(c,s,x,y,sz){
  c.beginPath();
  if(s==='heart'){const k=sz/2;
    c.moveTo(x,y+k*.95);
    c.bezierCurveTo(x-k*1.55,y-k*.38,x-k*.62,y-k*1.32,x,y-k*.34);
    c.bezierCurveTo(x+k*.62,y-k*1.32,x+k*1.55,y-k*.38,x,y+k*.95);}
  else if(s==='diamond'){
    c.moveTo(x,y-sz*.56);c.lineTo(x+sz*.4,y);c.lineTo(x,y+sz*.56);c.lineTo(x-sz*.4,y);c.closePath();}
  else if(s==='spade'){const k=sz/2;
    c.moveTo(x,y-k*.95);
    c.bezierCurveTo(x-k*1.55,y+k*.38,x-k*.62,y+k*1.32,x,y+k*.34);
    c.bezierCurveTo(x+k*.62,y+k*1.32,x+k*1.55,y+k*.38,x,y-k*.95);}
  else if(s==='club'){const r=sz*.27;
    c.moveTo(x+r,y-sz*.24);c.arc(x,y-sz*.24,r,0,Math.PI*2);
    c.moveTo(x-sz*.26+r,y+sz*.13);c.arc(x-sz*.26,y+sz*.13,r,0,Math.PI*2);
    c.moveTo(x+sz*.26+r,y+sz*.13);c.arc(x+sz*.26,y+sz*.13,r,0,Math.PI*2);
    c.moveTo(x-sz*.1,y+sz*.4);c.lineTo(x+sz*.1,y+sz*.4);c.lineTo(x+sz*.17,y+sz*.6);c.lineTo(x-sz*.17,y+sz*.6);c.closePath();}
  else if(s==='star'){star(c,x,y,sz*.58,sz*.24);}
}

const heldDirs=[];
function dirOf(key){
  key=(key||'').toLowerCase();
  if(key==='arrowup'||key==='w')return{x:0,y:-1};
  if(key==='arrowdown'||key==='s')return{x:0,y:1};
  if(key==='arrowleft'||key==='a')return{x:-1,y:0};
  if(key==='arrowright'||key==='d')return{x:1,y:0};
  return null;
}

  audio();
  const k=e.key;

  const d=dirOf(k);

  if(k>='1'&&k<='4'){playCard(+k-1);return;}

function bumpFx(){sBump();S.shake=Math.max(S.shake,.14);}

function tryMove(d){
  const p=S.p,fx=Math.floor(p.cx),fy=Math.floor(p.cy),nx=fx+d.x,ny=fy+d.y;
  if(isWall(nx+.5,ny+.5)){bumpFx();return;}
  for(const g of S.guards)if(Math.hypot(g.x-(nx+.5),g.y-(ny+.5))<.45){bumpFx();return;}
  p.fx=p.cx;p.fy=p.cy;p.tx=nx+.5;p.ty=ny+.5;p.k=0;p.moving=true;S.lastDir=d;
}

function playCard(i){
  if(S.mode!=='play')return;
  const p=S.p,def=CARD_DEFS[i];
  S.playFx[i]=.3;
  if(def.id==='dash'){
    if(p.moving){toast('STILL FIRST — THEN BLINK');fizz();return;}
    if(S.cool.dash>0){fizz();return;}
    const d=S.lastDir;if(!d){toast('HOLD A DIRECTION TO DASH');fizz();return;}
    let x=Math.floor(p.cx),y=Math.floor(p.cy),moved=0;
    for(let s=1;s<=2;s++){const nx=x+d.x,ny=y+d.y;
      if(isWall(nx+.5,ny+.5))break;
      let blocked=false;for(const g of S.guards)if(Math.hypot(g.x-(nx+.5),g.y-(ny+.5))<.45){blocked=true;break;}
      if(blocked)break;x=nx;y=ny;moved++;}
    if(!moved){toast('PATH BLOCKED');fizz();return;}
    for(let s=0;s<moved;s++){const ax=p.cx+(x-p.cx)*((s+1)/moved),ay=p.cy+(y-p.cy)*((s+1)/moved);
      S.fx.push({type:'img',x:ax,y:ay,age:0,life:.32});}
    p.cx=x+.5;p.cy=y+.5;p.moving=false;S.cool.dash=DASH_CD;sDash();S.shake=Math.max(S.shake,.18);
  }else if(def.id==='smoke'){
    if(S.cards.smoke<=0){toast('NO SMOKE LEFT');fizz();return;}
    S.cards.smoke--;S.smokes.push({x:Math.floor(p.cx)+.5,y:Math.floor(p.cy)+.5,r:1.6,age:0,life:4.6});
    sSmoke();toast('SMOKE DROPPED');
  }else if(def.id==='ghost'){
    if(S.cards.ghost<=0){toast('NO WILD LEFT');fizz();return;}
    S.cards.ghost--;S.ghostT=3.2;sGhost();toast('WILD — GHOST FOR A BEAT');
  }else if(def.id==='knock'){
    if(S.cards.knock<=0){toast('NO KNOCKOUT LEFT');fizz();return;}
    let best=null,bd=1e9;
    for(const g of S.guards){if(g.stun>0)continue;const d=Math.hypot(g.x-p.cx,g.y-p.cy);
      if(d<=4.5&&d<bd){bd=d;best=g;}}
    if(!best){toast('NO TARGET IN RANGE');fizz();return;}
    best.stun=3;best.det=0;S.cards.knock--;sKnock();S.shake=Math.max(S.shake,.25);
    for(let s=0;s<7;s++){const a=s/7*Math.PI*2;
      S.fx.push({type:'star',x:best.x,y:best.y,vx:Math.cos(a)*1.6,vy:Math.sin(a)*1.6,age:0,life:.7,size:5});}
    toast('GUARD KNOCKED OUT');
  }
}

function caught(g){
  S.mode='caught';S.caughtT=0;S.caughtBy=g;g.det=1;g.pop=.6;
  S.lives--;S.shake=.7;sSpotted();
}
function respawn(){
  const p=S.p;p.cx=S.P.x+.5;p.cy=S.P.y+.5;p.moving=false;
  for(const g of S.guards){g.x=g.pts[0].x;g.y=g.pts[0].y;g.pi=0;g.dir=1;g.det=0;g.stun=0;g.w=false;
    g.face=Math.atan2(g.pts[1].y-g.pts[0].y,g.pts[1].x-g.pts[0].x);}
  S.smokes.length=0;S.cards=Object.assign({},LEVELS[S.level].cards);
  S.cool.dash=0;S.ghostT=0;S.invuln=1.5;S.mode='play';toast('DECK REDEALT — STEADY NOW');
}

    S.fx.push({type:'spark',x:S.V.x+.5,y:S.V.y+.5,vx:Math.cos(a)*sp,vy:Math.sin(a)*sp,age:0,life:.9,size:2.5});}
}
function nextAfterClear(){

}

function seesPlayer(g){
  const p=S.p;if(S.ghostT>0||S.invuln>0)return false;
  const dx=p.cx-g.x,dy=p.cy-g.y,d=Math.hypot(dx,dy);g.d=d;
  if(d>g.range||d<.01)return false;
  if(Math.abs(angDiff(Math.atan2(dy,dx),g.face))>g.fovR/2+.12)return false;
  const steps=Math.ceil(d/.13);
  for(let i=1;i<steps;i++){const t=i/steps,x=g.x+dx*t,y=g.y+dy*t;
    if(isWall(x,y))return false;
    for(const s of S.smokes)if(Math.hypot(x-s.x,y-s.y)<s.r*.85)return false;}
  return true;
}
function updGuard(g,dt,detect){
  if(g.stun>0)g.stun=Math.max(0,g.stun-dt);
  else{
    let n=g.dir>0?g.pi+1:g.pi-1;
    if(n<0||n>=g.pts.length){g.dir*=-1;n=g.dir>0?g.pi+1:g.pi-1;}
    const tp=g.pts[n],dx=tp.x-g.x,dy=tp.y-g.y,d=Math.hypot(dx,dy),mv=g.spd*dt;
    if(d<=mv||d<.001){g.x=tp.x;g.y=tp.y;g.pi=n;if(n===0||n===g.pts.length-1)g.dir*=-1;}
    else{g.x+=dx/d*mv;g.y+=dy/d*mv;}
  }
  const tn=g.dir>0?Math.min(g.pi+1,g.pts.length-1):Math.max(g.pi-1,0);
  const tp2=g.pts[tn];if(Math.hypot(tp2.x-g.x,tp2.y-g.y)>.05){
    const ta=Math.atan2(tp2.y-g.y,tp2.x-g.x);g.face+=angDiff(ta,g.face)*Math.min(1,dt*8);}
  g.pop=Math.max(0,g.pop-dt);
  if(detect&&S.mode==='play'){
    const sees=seesPlayer(g);
    if(sees){if(!g.w){sAlert();g.pop=.6;}
      g.det=Math.min(1,g.det+dt*(.7+2.6*Math.max(0,1-g.d/g.range)));}
    else g.det=Math.max(0,g.det-dt*1.8);
    g.w=sees;
  }else if(!detect)g.det=0;
}

const drifters=[];
for(let i=0;i<20;i++)drifters.push({x:Math.random(),y:Math.random(),
  vx:(Math.random()-.5)*.008,vy:.006+Math.random()*.01,rot:Math.random()*6,vr:(Math.random()-.5)*.4,
  ch:'♠♥♦♣'[i%4],size:16+Math.random()*26,a:.04+Math.random()*.045,
  col:i%3===0?'#e6c26a':(i%3===1?'#d0524a':'#efe6c8')});

function update(dt){
  for(const d of drifters){d.x+=d.vx*dt;d.y+=d.vy*dt;d.rot+=d.vr*dt;
    if(d.y>1.1){d.y=-.1;d.x=Math.random();}if(d.x>1.1)d.x=-.1;if(d.x<-.1)d.x=1.1;}
  S.playFx=S.playFx.map(v=>Math.max(0,v-dt));
  if(S.mode==='paused')return;

  S.bannerT-=dt;S.cool.dash=Math.max(0,S.cool.dash-dt);
  S.ghostT=Math.max(0,S.ghostT-dt);S.invuln=Math.max(0,S.invuln-dt);
  S.shake=Math.max(0,S.shake-dt*2.4);
  if(S.toast){S.toast.t-=dt;if(S.toast.t<=0)S.toast=null;}

  for(let i=S.smokes.length-1;i>=0;i--){const s=S.smokes[i];s.age+=dt;
    if(s.age>s.life)S.smokes.splice(i,1);}
  for(let i=S.fx.length-1;i>=0;i--){const f=S.fx[i];f.age+=dt;
    if(f.vx){f.x+=f.vx*dt;f.y+=f.vy*dt;}if(f.age>f.life)S.fx.splice(i,1);}
  for(let i=S.conf.length-1;i>=0;i--){const c=S.conf[i];c.y+=c.vy*dt;c.x+=c.vx*dt;c.rot+=c.vr*dt;
    if(c.y>H+30)S.conf.splice(i,1);}

  const L=LEVELS[S.level];
  S.sparkT-=dt;
  if((S.mode==='play'||S.mode==='clearing')&&S.sparkT<=0){S.sparkT=.14;
    const a=Math.random()*Math.PI*2;
    S.fx.push({type:'spark',x:S.V.x+.5,y:S.V.y+.5,vx:Math.cos(a)*.9,vy:Math.sin(a)*.9-.4,age:0,life:.8,size:1.6});}

  if(S.mode==='menu'){
    for(const g of S.guards)updGuard(g,dt,false);return;}

  if(S.mode==='play'){
    S.time+=dt;
    const p=S.p;p.bob+=dt*9;
    if(!p.moving){const d=heldDirs.length?heldDirs[heldDirs.length-1]:null;if(d)tryMove(d);}
    if(p.moving){p.k+=dt*P_SPD;
      if(p.k>=1){p.cx=p.tx;p.cy=p.ty;p.moving=false;sStep();
        S.fx.push({type:'dot',x:p.cx,y:p.cy+.35,vx:0,vy:-.4,age:0,life:.4,size:2});
        if(Math.floor(p.cx)===S.V.x&&Math.floor(p.cy)===S.V.y)startClear();}}
    let any=false;
    for(const g of S.guards){updGuard(g,dt,true);if(g.det>=1&&!any){any=true;caught(g);}}
    S.maxDet=0;for(const g of S.guards)S.maxDet=Math.max(S.maxDet,g.det);
    if(S.maxDet>.15&&S.mode==='play'){S.hbT-=dt;if(S.hbT<=0){sHeart(S.maxDet);S.hbT=.6-.28*S.maxDet;}}
  }
  else if(S.mode==='caught'){
    S.caughtT+=dt;
    if(S.caughtT>1.2){if(S.lives<=0)endRun(false);else respawn();}
  }
  else if(S.mode==='clearing'){S.clearT+=dt;if(S.clearT>1.05)nextAfterClear();}
  else if(S.mode==='victory'||S.mode==='gameover'){
    if(S.conf.length<40&&Math.random()<.3)S.conf.push({x:Math.random()*W,y:-20,
      vy:60+Math.random()*70,vx:(Math.random()-.5)*26,rot:Math.random()*6,vr:(Math.random()-.5)*3,
      ch:'♠♥♦♣'[(Math.random()*4)|0],col:['#e6c26a','#d0524a','#efe6c8'][(Math.random()*3)|0],size:12+Math.random()*12});}
}

function drawCone(c,g,t){
  if(g.stun>0)return;
  const gx=PX(g.x),gy=PY(g.y),R=g.range*TILE,fov=g.fovR;
  const danger=S.maxDet>.5&&g.det===S.maxDet;
  const pts=[];const N=26;
  for(let i=0;i<=N;i++){
    const a=g.face-fov/2+fov*i/N;let end=R;
    for(let s=.14;s<R;s+=.14){if(isWall(g.x+Math.cos(a)*s,g.y+Math.sin(a)*s)){end=s;break;}
      let sm=false;for(const k of S.smokes)if(Math.hypot(g.x+Math.cos(a)*s-k.x,g.y+Math.sin(a)*s-k.y)<k.r*.85){sm=true;break;}
      if(sm){end=s;break;}}
    pts.push([gx+Math.cos(a)*end,gy+Math.sin(a)*end]);}
  const pul=.82+.18*Math.sin(t*6);
  const grad=c.createRadialGradient(gx,gy,4,gx,gy,R);
  if(danger){grad.addColorStop(0,'rgba(255,70,50,'+(.34*pul)+')');grad.addColorStop(1,'rgba(255,70,50,0)');}
  else{grad.addColorStop(0,'rgba(230,190,90,'+(.26*pul)+')');grad.addColorStop(1,'rgba(230,190,90,0)');}
  c.fillStyle=grad;c.beginPath();c.moveTo(gx,gy);
  for(const p of pts)c.lineTo(p[0],p[1]);c.closePath();c.fill();
}

function drawHand(c,t){
  S.handRects.length=0;
  const gap=CW*.82,cx=W/2+(S.hover>=0?0:0);
  for(let i=0;i<4;i++){
    const def=CARD_DEFS[i];
    let x=cx+(i-1.5)*gap,y=H-handH+16;
    const disabled=(def.id==='dash')?(S.cool.dash>0):(S.cards[def.id]<=0);
    if(S.hover===i&&!disabled)y-=9;
    c.save();c.translate(x,y+CH/2);
    if(i!==S.hover)c.globalAlpha=1;
    // shadow
    c.fillStyle='rgba(0,0,0,.4)';rr(c,-CW/2+3,-CH/2+5,CW,CH,CW*.12);c.fill();
    let dim=false;
    if(def.id==='dash'){
      drawFace(c,0,0,CW,CH,def.face,{glow:S.hover===i&&!disabled?16:0});
      const f=S.cool.dash/DASH_CD;
      if(f>0){c.save();rr(c,-CW/2,-CH/2,CW,CH,CW*.12);c.clip();
        c.fillStyle='rgba(8,13,10,.62)';c.fillRect(-CW/2,-CH/2,CW,CH*f);c.restore();}
    }else if(def.id==='smoke'){drawFace(c,0,0,CW,CH,def.face,{glow:S.hover===i&&!disabled?16:0});}
    else if(def.id==='ghost'){drawFace(c,0,0,CW,CH,def.face,{glow:S.hover===i&&!disabled?16:0});}
    else{drawFace(c,0,0,CW,CH,def.face,{glow:S.hover===i&&!disabled?16:0});dim=S.cards.knock<=0;}
    // name strip
    c.fillStyle='rgba(14,20,17,.88)';rr(c,-CW/2+3,CH/2-15,CW-6,12,3);c.fill();
    c.font='700 8.5px '+BODY;c.textAlign='center';c.textBaseline='middle';
    try{c.letterSpacing='2px';}catch(e){}
    c.fillStyle=disabled?'rgba(239,230,200,.4)':'#e6c26a';c.fillText(def.name,0,CH/2-8.5);
    try{c.letterSpacing='0px';}catch(e){}
    // key badge
    c.fillStyle='#1b2d23';rr(c,-CW/2+4,-CH/2+4,13,13,3);c.fill();
    c.strokeStyle='rgba(230,194,106,.5)';c.lineWidth=1;rr(c,-CW/2+4,-CH/2+4,13,13,3);c.stroke();
    c.fillStyle='#efe6c8';c.font='700 9px '+BODY;c.fillText(def.key,0,-CH/2+10.5);
    // count badge (non-dash)
    if(def.id!=='dash'){
      const n=S.cards[def.id];
      c.beginPath();c.arc(CW/2-10,-CH/2+10,8.5,0,7);
      c.fillStyle=n>0?'#f4dd9b':'rgba(60,40,35,.9)';c.fill();
      c.strokeStyle='rgba(30,28,20,.5)';c.stroke();
      c.fillStyle=n>0?'#17130c':'#e08a7a';c.font='700 10px '+BODY;
      c.fillText(n>0?String(n):'×',CW/2-10,-CH/2+9.5);}
    if(disabled){c.globalAlpha=.4;c.fillStyle='rgba(10,14,12,.5)';rr(c,-CW/2,-CH/2,CW,CH,CW*.12);c.fill();}
    if(S.playFx[i]>0){c.strokeStyle='rgba(246,221,154,'+S.playFx[i]*3+')';c.lineWidth=2.5;
      rr(c,-CW/2-2,-CH/2-2,CW+4,CH+4,CW*.13);c.stroke();}
    c.restore();
    S.handRects.push({x:x-CW/2,y:y,w:CW,h:CH,i});
  }
}
