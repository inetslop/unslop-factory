function getTarget(){return state.rocks.find(r=>r.id===state.target)||null;}
function retargetNearest(){if(!state.rocks.length){state.target=null;return;}

function rockPos(r){const t=Math.max(0,Math.min(1.06,r.animSeg/SEG));

function rockR(r){const t=Math.max(0,Math.min(1,r.animSeg/SEG));return ROCK[r.size].r*k*(.6+.8*(1-t));}

function dealDamage(r,amt,type,quiet){if(!r)return;
  const nose={x:ship.x,y:ship.y-18*k},p=rockPos(r);
  if(type==='laser')beam(nose.x,nose.y,p.x,p.y,'#4de3ff',2*k);
  else if(type==='twin'){beam(nose.x-4*k,nose.y+4,p.x-5*k,p.y,'#8fd7ff',2.2*k);beam(nose.x+4*k,nose.y+4,p.x+5*k,p.y,'#8fd7ff',2.2*k);}
  else if(type==='missile'){beam(nose.x,nose.y,p.x,p.y,'#ffb347',2.6*k,.22);boom(p.x,p.y,'S','#ffb347');shake=Math.min(14,shake+1.5);}
  else if(type==='nuke'){beam(nose.x,nose.y,p.x,p.y,'#ff5d5d',6*k,.28);boom(p.x,p.y,'M','#ff5d5d');shake=Math.min(14,shake+4);}
  else beam(nose.x,nose.y,p.x,p.y,'rgba(255,255,255,.8)',1.5*k,.12);
  if(!quiet){if(type==='laser')sfx.laser();else if(type==='twin')sfx.twin();else if(type==='missile')sfx.missile();else if(type==='nuke')sfx.nuke();else sfx.turret();}
  r.hp-=amt;r.flash=1;
  if(r.hp<=0)destroyRock(r);else floater(p.x,p.y-14*k,'-'+amt,'#ffffff');}

function destroyRock(r){const p=rockPos(r),i=state.rocks.indexOf(r);if(i<0)return;
  state.score+=ROCK[r.size].score;floater(p.x,p.y,'+'+ROCK[r.size].score,ROCK[r.size].col);
  boom(p.x,p.y,r.size,ROCK[r.size].col);sfx.boom(r.size);
  if(r.size!=='S'){const child=r.size==='L'?'M':'S';
    const pa=Math.atan2(r.spawn.y-ship.y,r.spawn.x-ship.x);
    addRock(child,r.seg,pa-rand(.15,.3));addRock(child,r.seg,pa+rand(.15,.3));}
  state.rocks.splice(i,1);
  if(state.target===r.id)retargetNearest();}

function impact(r){const i=state.rocks.indexOf(r);if(i<0)return;
  const dmg=ROCK[r.size].dmg,abs=Math.min(state.shield,dmg);state.shield-=abs;const rest=dmg-abs;state.hull-=rest;
  boom(ship.x,ship.y-10*k,r.size,'#ff5d5d');shake=13;sfx.impact();
  if(rest>0){floater(ship.x,ship.y-46*k,'-'+rest+' HULL','#ff6b5d');flashVign();}
  else floater(ship.x,ship.y-46*k,'SHIELD -'+abs,'#8fc6ff');
  state.rocks.splice(i,1);if(state.target===r.id)retargetNearest();}

function playCard(i){if(!state||state.phase!=='play')return;const t=state.hand[i];if(!t)return;
  const c=CARDS[t];
  if(state.energy<c.cost){toast('NOT ENOUGH ENERGY');sfx.err();return;}
  if(c.kind==='dmg'||c.kind==='burn'){if(!getTarget()){toast('SELECT A TARGET — CLICK A ROCK');sfx.err();return;}}
  state.energy-=c.cost;state.discard.push(state.hand.splice(i,1)[0]);
  const r=getTarget();
  if(c.kind==='dmg')dealDamage(r,c.val+state.upg.overclock,t);
  else if(c.kind==='burn'){r.seg=Math.min(SEG,r.seg+c.val);sfx.burn();floater(rockPos(r).x,rockPos(r).y,'REVERSE','#ffd84d');}
  else if(c.kind==='shield'){state.shield=Math.min(12,state.shield+c.val+state.upg.aegis*2);sfx.shield();floater(ship.x,ship.y-46*k,'SHIELD UP','#8fc6ff');}
  else if(c.kind==='repair'){state.hull=Math.min(state.maxHull,state.hull+c.val);sfx.repair();floater(ship.x,ship.y-46*k,'+'+c.val+' HULL','#6bff9e');}
  else if(c.kind==='over'){state.energy+=c.val;drawToHand(1);sfx.over();floater(ship.x,ship.y-46*k,'+'+c.val+' ENERGY','#ff9de4');}
  renderAll();checkWaveClear();}

function checkWaveClear(){if(state.phase==='play'&&state.rocks.length===0)waveClearFlow();}

function applyUpgrade(u){state.upg[u.id]++;

  if(u.id==='hull'){state.maxHull+=3;state.hull=Math.min(state.maxHull,state.hull+3);}
  if(u.id==='array')state.handCap++;
  renderAll();}
function proceedNext(){$('#ov-up').classList.add('hidden');state.phase='play';nextWave();}

function endTurn(){if(!state||state.phase!=='play')return;
  if(state.upg.turret&&state.rocks.length){const r=[...state.rocks].sort((a,b)=>a.seg-b.seg)[0];dealDamage(r,1,'turret',true);}
  for(const r of[...state.rocks]){r.seg-=1;if(r.seg<=0)impact(r);
    if(state.hull<=0){gameOver();return;}}
  sfx.tick();
  if(!state.rocks.length){checkWaveClear();renderAll();return;}
  state.energy=state.maxEnergy;drawToHand();renderAll();}

function update(dt){shake=Math.max(0,shake-dt*22);
  for(const r of(state?state.rocks:[])){r.animSeg+=(r.seg-r.animSeg)*Math.min(1,dt*7);r.rot+=r.vr*dt;r.flash=Math.max(0,r.flash-dt*4);}
  for(let i=beams.length-1;i>=0;i--){beams[i].t+=dt;if(beams[i].t>beams[i].life)beams.splice(i,1);}
  for(let i=parts.length-1;i>=0;i--){const p=parts[i];p.t+=dt;p.x+=p.vx*dt;p.y+=p.vy*dt;p.vx*=.96;p.vy*=.96;if(p.t>p.max)parts.splice(i,1);}
  for(let i=ringsFx.length-1;i>=0;i--){ringsFx[i].t+=dt;if(ringsFx[i].t>ringsFx[i].dur)ringsFx.splice(i,1);}
  for(let i=floaters.length-1;i>=0;i--){floaters[i].t+=dt;if(floaters[i].t>floaters[i].max)floaters.splice(i,1);}
  const showAmb=!state||state.phase==='over';
  if(showAmb){for(const a of amb){a.y+=a.vy*dt;a.rot+=a.vr*dt;if(a.y>H+80){a.y=-80;a.x=Math.random()*W;}}}
  for(const s of stars){s.y+=dt*(.015+.06*s.z);if(s.y>1)s.y-=1;}}
