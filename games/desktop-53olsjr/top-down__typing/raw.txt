function tone(type,f0,f1,dur,vol,delay){
  if(!AC||muted)return;delay=delay||0;
  const t=AC.currentTime+delay,o=AC.createOscillator(),g=AC.createGain();
  o.type=type;o.frequency.setValueAtTime(f0,t);

function step(dt){
  tNow+=dt;
  G.shake=Math.max(0,G.shake-42*dt);
  G.lockout-=dt;sweepA+=dt*.85;
  G.pingT-=dt;
  if(G.pingT<=0){rings.push({x:CX,y:CY,r:CARR_R+10,vr:95,life:1.6,max:1.6,col:'99,220,255'});G.pingT=2.7;}

  /* fx always */
  for(let i=tracers.length-1;i>=0;i--){const t=tracers[i];t.t+=dt/t.dur;
    if(t.t>=1){spark(t.x1,t.y1,t.col);tracers.splice(i,1);}}
  for(let i=parts.length-1;i>=0;i--){const p=parts[i];p.life-=dt/p.max;
    if(p.life<=0){parts.splice(i,1);continue;}
    p.x+=p.vx*dt;p.y+=p.vy*dt;p.vx*=(1-3.2*dt);p.vy*=(1-3.2*dt);}
  for(let i=rings.length-1;i>=0;i--){const r=rings[i];r.life-=dt;r.r+=r.vr*dt;if(r.life<=0)rings.splice(i,1);}
  for(let i=texts.length-1;i>=0;i--){const t=texts[i];t.life-=dt;t.y-=26*dt;if(t.life<=0)texts.splice(i,1);}

  if(G.state!=='play')return;
  G.time+=dt;

  /* spawning */
  const maxC=Math.min(6,2+Math.floor((G.wave-1)));
  G.spawnT-=dt;
  if(G.spawnT<=0&&enemies.filter(e=>!e.friendly).length<maxC){
    spawnEnemy(false);
    G.spawnT=Math.max(.85,3.0-(G.wave-1)*.32)*(.7+Math.random()*.6);
  }
  const hasAlly=enemies.some(e=>e.friendly);
  if(G.hull<65&&!hasAlly){
    G.healT-=dt;
    if(G.healT<=0){spawnEnemy(true);G.healT=13;}
  }else if(G.hull>=65)G.healT=Math.max(G.healT,4);

  /* enemies */
  for(let i=enemies.length-1;i>=0;i--){const e=enemies[i];
    e.dist-=e.speed*dt;e.scale=Math.min(1,e.scale+dt*3.5);if(e.flash>0)e.flash-=dt;
    e.x=CX+Math.cos(e.a)*e.dist;e.y=CY+Math.sin(e.a)*e.dist;
    if(e.dist<=CARR_R){enemies.splice(i,1);if(G.target===e)G.target=null;breach(e);}
  }

  /* targeting */
  let bestE=null,bd=1e18;
  for(const e of enemies){const d=(mouse.x-e.x)*(mouse.x-e.x)+(mouse.y-e.y)*(mouse.y-e.y);if(d<bd){bd=d;bestE=e;}}
  G.target=bestE;
  refreshTP();

  /* damage smoke */
  if(G.hull<45){G.smokeT-=dt;
    if(G.smokeT<=0){G.smokeT=.13;
      parts.push({x:CX+(Math.random()-.5)*70,y:CY+(Math.random()-.5)*26,
        vx:(Math.random()-.5)*14,vy:-12-Math.random()*18,life:1,max:1.6,size:4+Math.random()*5,col:'rgba(90,100,110,.5)'});}}
}

  ctx.font='700 '+fs+'px ui-monospace,Consolas,Menlo,monospace';
  ctx.textBaseline='middle';ctx.textAlign='left';
  for(let i=0;i<e.word.length;i++){
    const cx2=x+8+i*cw;
    ctx.fillStyle=i<e.idx?COL.green:(i===e.idx?(e.friendly?COL.green:COL.amber):'#cfe0ea');
    ctx.fillText(e.word[i],cx2,y+h/2+1);
    if(i===e.idx&&Math.floor(tNow*3)%2===0){
      ctx.fillStyle=e.friendly?COL.green:COL.amber;
      ctx.fillRect(cx2-2,y+h-5,cw,2);}
  }
}
function draw(){
  /* water */
  const g=ctx.createRadialGradient(CX,CY,60,CX,CY,Math.max(W,H)*.78);
  g.addColorStop(0,'#0e2a3d');g.addColorStop(.55,'#0a1e2e');g.addColorStop(1,'#060f18');
  ctx.fillStyle=g;ctx.fillRect(0,0,W,H);
  /* shimmer drift */
  ctx.save();ctx.translate((tNow*9)%240,(tNow*5)%240);ctx.fillStyle=shimA;ctx.fillRect(-240,-240,W+480,H+480);ctx.restore();
  ctx.save();ctx.translate(-(tNow*6)%240,(tNow*3.5)%240);ctx.fillStyle=shimB;ctx.fillRect(-240,-240,W+480,H+480);ctx.restore();

  /* shake world */
  ctx.save();
  if(G.shake>0)ctx.translate((Math.random()*2-1)*G.shake,(Math.random()*2-1)*G.shake);

  /* radar grid */
  ctx.strokeStyle='rgba(99,220,255,.10)';ctx.lineWidth=1;
  [CARR_R+36,CARR_R+124,CARR_R+212,CARR_R+300].forEach(r=>{ctx.beginPath();ctx.arc(CX,CY,r,0,6.283);ctx.stroke();});
  ctx.strokeStyle='rgba(99,220,255,.05)';
  ctx.beginPath();ctx.moveTo(CX-MAXR*.8,CY);ctx.lineTo(CX+MAXR*.8,CY);
  ctx.moveTo(CX,CY-MAXR*.8);ctx.lineTo(CX,CY+MAXR*.8);ctx.stroke();
  /* sweep */
  if(ctx.createConicGradient){
    const cg=ctx.createConicGradient(sweepA,CX,CY);
    cg.addColorStop(0,'rgba(99,220,255,.17)');cg.addColorStop(.14,'rgba(99,220,255,0)');
    cg.addColorStop(1,'rgba(99,220,255,0)');
    ctx.fillStyle=cg;ctx.beginPath();ctx.arc(CX,CY,MAXR*.8,0,6.283);ctx.fill();
  }

  /* rings */
  for(const r of rings){const a=Math.max(0,r.life/r.max);
    ctx.strokeStyle='rgba('+r.col+','+(a*.5)+')';ctx.lineWidth=1.5;
    ctx.beginPath();ctx.arc(r.x,r.y,r.r,0,6.283);ctx.stroke();}

  drawCarrier();

  /* tracers */
  for(const t of tracers){
    const tx=t.x0+(t.x1-t.x0)*Math.min(1,t.t),ty=t.y0+(t.y1-t.y0)*Math.min(1,t.t);
    ctx.strokeStyle=t.col;ctx.lineWidth=2;ctx.globalAlpha=.9;
    ctx.beginPath();ctx.moveTo(t.x0,t.y0);ctx.lineTo(tx,ty);ctx.stroke();
    ctx.fillStyle='#fff';ctx.beginPath();ctx.arc(tx,ty,2.4,0,6.283);ctx.fill();
    ctx.globalAlpha=1;
  }

  /* lock line + target */
  const T=G.target;
  if(T&&G.state==='play'){
    const n=nose();
    ctx.strokeStyle='rgba(99,220,255,.18)';ctx.lineWidth=1;ctx.setLineDash([6,7]);
    ctx.beginPath();ctx.moveTo(n.x,n.y);ctx.lineTo(T.x,T.y);ctx.stroke();ctx.setLineDash([]);
    /* brackets */
    const b=24,L=9;ctx.strokeStyle=T.friendly?COL.green:COL.amber;ctx.lineWidth=2;
    [[-1,-1],[1,-1],[-1,1],[1,1]].forEach(c=>{
      ctx.beginPath();
      ctx.moveTo(T.x+c[0]*b,T.y+c[0]*b+c[1]*L);
      ctx.lineTo(T.x+c[0]*b,T.y+c[1]*b);
      ctx.lineTo(T.x+c[0]*b+c[0]*L,T.y+c[1]*b);
      ctx.stroke();});
  }

  for(const e of enemies)drawEnemy(e);

  /* particles */
  for(const p of parts){ctx.globalAlpha=Math.max(0,p.life);
    ctx.fillStyle=p.col;ctx.beginPath();ctx.arc(p.x,p.y,p.size*(.5+p.life*.5),0,6.283);ctx.fill();}
  ctx.globalAlpha=1;

  /* floating text */
  ctx.textAlign='center';ctx.font='700 15px ui-monospace,Consolas,Menlo,monospace';
  for(const t of texts){ctx.globalAlpha=Math.min(1,t.life);
    ctx.fillStyle=t.col;ctx.fillText(t.txt,t.x,t.y);}
  ctx.globalAlpha=1;

  /* reticle */
  if(G.state==='play'){
    ctx.strokeStyle='rgba(99,220,255,.75)';ctx.lineWidth=1.4;
    ctx.beginPath();ctx.arc(mouse.x,mouse.y,10,0,6.283);ctx.stroke();
    for(let i=0;i<4;i++){const a=i*Math.PI/2;
      ctx.beginPath();
      ctx.moveTo(mouse.x+Math.cos(a)*13,mouse.y+Math.sin(a)*13);
      ctx.lineTo(mouse.x+Math.cos(a)*19,mouse.y+Math.sin(a)*19);ctx.stroke();}
    ctx.fillStyle='rgba(99,220,255,.9)';ctx.beginPath();ctx.arc(mouse.x,mouse.y,1.6,0,6.283);ctx.fill();
  }
  ctx.restore();
}

/* ============================== LOOP ============================== */
let last=performance.now();
function frame(now){
  const dt=Math.min(.05,(now-last)/1000);last=now;
  step(dt);draw();
  requestAnimationFrame(frame);
}
requestAnimationFrame(frame);
