function updBalls(dt){

/* ================= step ================= */
function step(dt){
  t+=dt;
  if(state==='menu'){
    menuHopT-=dt;
    if(menuHopT<=0){doHop(Math.random()<.5?-1:1);menuHopT=1.3+Math.random()*1.4;}
    updFrog(dt);updFx(dt);return;
  }
  if(state==='paused')return;
  if(state==='over'){updFx(dt);return;}
  if(state==='clear'){
    clearT-=dt;updDrops(dt);updFx(dt);
    if(clearT<=0)startLevel(level+1);
    return;
  }
  if(bannerT>0)bannerT-=dt;
  power.t=Math.max(0,power.t-dt);power.s=Math.max(0,power.s-dt);
  for(const br of bricks)br.flash=Math.max(0,br.flash-dt*4);
  updFrog(dt);updBalls(dt);updDrops(dt);updFx(dt);
}

function render(){

  ctx.setTransform(dpr*sc,0,0,dpr*sc,dpr*(ox+sx),dpr*(oy+sy));
  ctx.save();ctx.beginPath();ctx.rect(-6,-6,W+12,H+12);ctx.clip();

  drawDropsDraw();drawPads();drawFrog();drawBalls();drawParts();drawPops();

  rr(3,3,W-6,H-6,10);ctx.strokeStyle='rgba(120,255,190,.22)';ctx.lineWidth=2.5;ctx.stroke();
}
