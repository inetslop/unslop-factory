    if(parts.length<420)parts.push({x:p.x,y:p.y,vx:0,vy:0,t:0,life:.3,sz:9*S,col:p.main});}
  else{if(ix||iy){const l=Math.hypot(ix,iy),sp=340*S;p.x+=ix/l*sp*dt;p.y+=iy/l*sp*dt;
      const want=Math.atan2(iy,ix);p.face+=wrapA(want-p.face)*Math.min(1,dt*14);
      if(Math.random()<dt*18&&parts.length<420)
        parts.push({x:p.x-Math.cos(p.face)*12,y:p.y-Math.sin(p.face)*12,vx:-Math.cos(p.face)*60*S,vy:-Math.sin(p.face)*60*S,t:0,life:.22,sz:2.4*S,col:p.main});}}

function updPB(dt){for(let i=pB.length-1;i>=0;i--){const b=pB[i];
  b.x+=b.vx*dt;b.y+=b.vy*dt;b.life-=dt;
  let dead=b.life<=0||b.x<-20||b.x>W+20||b.y<-20||b.y>H+20;
  if(!dead)for(let j=enemies.length-1;j>=0;j--){const e=enemies[j];if(e.die||e.pop<.5)continue;
    const rr=e.r*.92+b.r;if(d2(b.x,b.y,e.x,e.y)<rr*rr){
      e.hp-=1;e.flash=1;spark(b.x,b.y,'#ffffff',3,60);sHit();dead=true;
      if(e.hp<=0)killEnemy(e);break;}}
  if(dead)pB.splice(i,1);}
  enemies=enemies.filter(e=>!e.die);}

function updBullets(dt){for(let i=eB.length-1;i>=0;i--){const b=eB[i];b.t+=dt;
  if(b.kind==='seek'){const p=nearestAlive(b.x,b.y);
    if(p){const want=Math.atan2(p.y-b.y,p.x-b.x),cur=Math.atan2(b.vy,b.vx);
      const nc=cur+clamp(wrapA(want-cur),-2.4*dt,2.4*dt),sp=Math.hypot(b.vx,b.vy)||1;
      b.vx=Math.cos(nc)*sp;b.vy=Math.sin(nc)*sp;}}
  b.x+=b.vx*dt;b.y+=b.vy*dt;
  if(b.x<-30||b.x>W+30||b.y<-30||b.y>H+30){eB.splice(i,1);continue;}
  for(const p of players){if(!p.alive||p.inv>0)continue;const rr=b.r+3.6;
    if(d2(b.x,b.y,p.x,p.y)<rr*rr){eB.splice(i,1);
      if(p.shield){p.shield=false;p.inv=.5;sShieldS();boom(p.x,p.y,'#7ab8ff',10,false);
        floats.push({x:p.x,y:p.y-24,txt:'BLOCKED',t:0,col:'#7ab8ff',sz:13});}
      else playerHit(p);break;}}}}

function applyPow(p,type){sPick();boom(p.x,p.y,POWS[type].col,8,false);
  if(type==='heal'){p.lives=Math.min(5,p.lives+1);floats.push({x:p.x,y:p.y-26,txt:'+LIFE',t:0,col:'#6dff9e',sz:14});}
  else if(type==='aegis'){p.shield=true;floats.push({x:p.x,y:p.y-26,txt:'AEGIS',t:0,col:'#7ab8ff',sz:14});}
  else{run.slowT=5;floats.push({x:p.x,y:p.y-26,txt:'SLOWFIELD',t:0,col:'#c78bff',sz:14});}}
function updPows(dt){for(let i=pows.length-1;i>=0;i--){const o=pows[i];o.t+=dt;o.life-=dt;
  if(o.life<=0){pows.splice(i,1);continue;}
  for(const p of players)if(p.alive&&d2(o.x,o.y,p.x,p.y)<(20*S)*(20*S)){applyPow(p,o.type);pows.splice(i,1);break;}}}}

/* ================= main update ================= */
function update(dt){run.time+=dt;run.hintT-=dt;if(run.slowT>0)run.slowT-=dt;
  const dte=dt*(run.slowT>0?.45:1);
  run.comboT-=dt;if(run.comboT<=0)run.combo=0;
  if(run.overT>=0){run.overT-=dt;if(run.overT<=0){showOver();return;}}
  if(run.clearT>0){run.clearT-=dt;if(run.clearT<=0)nextWave();}
  else{if(run.introT>0)run.introT-=dt;
    else{if(run.spawnQ.length){run.spawnT-=dt;
        if(run.spawnT<=0){queueSpawn();const base=Math.max(.4,1.5-run.wave*.07);run.spawnT=rand(base*.7,base*1.3);}}
      else if(!enemies.length){const bonus=400+run.wave*150;run.score+=bonus;
        floats.push({x:W/2,y:H*.42,txt:'WAVE CLEARED +'+bonus,t:0,col:'#ffd24d',sz:20});
        pows.push({x:W/2,y:H*.55,type:pick(['heal','aegis','slow']),t:0,life:12});
        run.clearT=2.1;sPick();}}}
  updWarnings(dte);updEnemies(dte);
  for(const p of players)updPlayer(p,dt);
  updPB(dt);updBullets(dte);updPows(dt);updParts(dt);updFloats(dt);
  shake=Math.max(0,shake-dt*46);redFlash=Math.max(0,redFlash-dt*1.6);}
