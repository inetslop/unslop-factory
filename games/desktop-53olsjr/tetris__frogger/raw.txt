function tick(dt){
  t+=dt;
  if(shake>0)shake=Math.max(0,shake-36*dt);
  if(redFlash>0)redFlash=Math.max(0,redFlash-dt*1.6);
  updateFireflies(dt);
  if(state==='start'||state==='over')updateDecor(dt);
  for(let i=particles.length-1;i>=0;i--){const p=particles[i];p.t+=dt;p.x+=p.vx*dt;p.y+=p.vy*dt;p.vy+=p.gr*dt;if(p.t>p.life)particles.splice(i,1);}
  for(let i=popups.length-1;i>=0;i--){popups[i].t+=dt;if(popups[i].t>popups[i].life)popups.splice(i,1);}
  if(state==='play'){
    hintT-=dt;if(spawnWarn>0)spawnWarn-=dt;
    if(frog.inv>0)frog.inv-=dt;
    frog.hopT=Math.min(frog.hopT+dt/frog.hopDur,2);
    if(held.length&&frog.hopT>=1){const d=held[held.length-1];tryHop(d[0],d[1]);}
    updatePiece(dt);
    if(state==='play')updateCars(dt);
    processClearFlashes(dt);
    if(state==='play'&&frog.inv<=0&&(pieceHas(frog.c,frog.r)||carCovers(frog.c,frog.r)))hitFrog();
    if(state==='play'&&frog.r===0)startLevelUp();
  }else if(state==='level'){
    levelT+=dt;processClearFlashes(dt);
    if(levelT<0.9&&Math.random()<dt*40)bubble(rand(1,COLS-1),ROWS-rand(0,5));
    if(washFx)washFx.t=levelT;
    if(levelT>=1.9)finishLevel();
  }else if(state==='over'){overT+=dt;}
}

  ctx.save();
  if(shake>0)ctx.translate((Math.random()*2-1)*shake*0.5,(Math.random()*2-1)*shake*0.5);
  uiPause=null;uiMute=null;
  drawBG();

  else{
    drawBoardScene();
    drawHUD();
    if(state==='pause')drawPause();
    if(state==='level')drawLevelBanner();
    if(state==='over')drawOver();
  }

    ctx.fillStyle=g;ctx.fillRect(0,0,W,H);
  }
}
