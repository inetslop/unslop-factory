function segSeg(ax,ay,bx,by,cx,cy,dx,dy){
 const c1=(bx-ax)*(cy-ay)-(by-ay)*(cx-ax),c2=(bx-ax)*(dy-ay)-(by-ay)*(dx-ax);
 const c3=(dx-cx)*(ay-cy)-(dy-cy)*(ax-cx),c4=(dx-cx)*(by-cy)-(dy-cy)*(bx-cx);

function segRect(x1,y1,x2,y2,o){
 const a=o.x,b=o.y,c2=o.x+o.w,d=o.y+o.h;
 if(Math.max(x1,x2)<a||Math.min(x1,x2)>c2||Math.max(y1,y2)<b||Math.min(y1,y2)>d)return false;

 return segSeg(x1,y1,x2,y2,a,b,c2,b)||segSeg(x1,y1,x2,y2,c2,b,c2,d)||segSeg(x1,y1,x2,y2,c2,d,a,d)||segSeg(x1,y1,x2,y2,a,d,a,b);}

function burst(x,y,n,col,sp){for(let i=0;i<n;i++){const a=Math.random()*TAU,v=sp*(.3+Math.random()*.7);
 S.parts.push({t:'spark',x,y,vx:Math.cos(a)*v,vy:Math.sin(a)*v,life:.4,max:.4,col});}}
function ring(x,y,col){S.parts.push({t:'ring',x,y,life:.6,max:.6,col});}
function puff(x,y){S.parts.push({t:'smoke',x,y,vx:(Math.random()-.5)*20,vy:(Math.random()-.5)*20,life:.7,max:.7,r:5+Math.random()*4});}
function floatTxt(x,y,txt){S.floats.push({x,y,txt,life:1});}
function updParts(dt){
 for(let i=S.parts.length-1;i>=0;i--){const p=S.parts[i];p.life-=dt;
  if(p.t!=='ring'){p.x+=p.vx*dt;p.y+=p.vy*dt;}
  if(p.life<=0)S.parts.splice(i,1);}
 for(let i=S.floats.length-1;i>=0;i--){const f=S.floats[i];f.life-=dt*.8;f.y-=42*dt;if(f.life<=0)S.floats.splice(i,1);}}

/* ================= simulation ================= */
function seenBy(g,sway){const c=S.car,dx=c.x-g.x,dy=c.y-g.y,d=Math.hypot(dx,dy);
 if(d>GUARD_RANGE)return false;
 const da=Math.abs(angDiff(Math.atan2(dy,dx),g.aim+sway));
 return da<GUARD_HALF&&los(g.x,g.y,c.x,c.y);}

function guardStep(g,dt,live){
 const mult=1+(live?S.threat/100*.7:0);
 if(live&&g.locked){
  g.aim=lerpAng2(g.aim,Math.atan2(S.car.y-g.y,S.car.x-g.x),dt*8);
  if(!seenBy(g,0)){g.locked=false;g.susp=.35;setWarn('SIGNAL LOST','ok');}
  else{g.lockT-=dt;if(g.lockT<=0)doBust('SPOTTED BY PATROL');}
  return;}
 let mvx=0,mvy=0,spd=62*mult;
 if(g.mode==='run'){const dx=g.tx-g.x,dy=g.ty-g.y,d=Math.hypot(dx,dy);spd=135*mult;
  if(d<14){g.mode='scan';g.scanT=3.5;}else{mvx=dx/d;mvy=dy/d;}}
 else if(g.mode==='scan'){g.scanT-=dt;g.aim+=dt*2.6;
  if(g.scanT<=0){g.mode='patrol';let bi=0,bd=1e18;g.pts.forEach((p,i)=>{const dd=(p.x-g.x)**2+(p.y-g.y)**2;if(dd<bd){bd=dd;bi=i;}});g.i=bi;}}
 else{const t=g.pts[g.i],dx=t.x-g.x,dy=t.y-g.y,d=Math.hypot(dx,dy);
  if(d<10){g.i=(g.i+g.dir+2)%2;g.dir*=-1;}else{mvx=dx/d;mvy=dy/d;}}
 if(mvx||mvy){g.x+=mvx*spd*dt;g.y+=mvy*spd*dt;g.aim=lerpAng2(g.aim,Math.atan2(mvy,mvx),dt*6);}
 if(live){
  const sway=g.mode==='patrol'?Math.sin(T*1.7+g.ph)*.38:0;
  const c=S.car,sp=Math.hypot(c.vx,c.vy);
  if(seenBy(g,sway)){const d=Math.hypot(c.x-g.x,c.y-g.y),df=lerp(1.45,.7,d/GUARD_RANGE);
   g.susp+=.95*(.55+.65*clamp(sp/380,0,1))*df*dt;
   if(g.susp>=1){g.locked=true;g.lockT=2.6;S.locks++;S.threat=Math.min(100,S.threat+30);
    Audio.alarm();S.shake=9;setWarn('SIGNAL ACQUIRED — BREAK LINE OF SIGHT','bad');alertGuards(c.x,c.y,g);}}
  else g.susp=Math.max(0,g.susp-1.15*dt);}
}
function lerpAng2(a,b,t){const d=angDiff(b,a);return a+d*clamp(t,0,1);}

function droneStep(d,dt,live){
 d.t+=dt;
 if(!live){d.st='patrol';orbitDrone(d);return;}
 const c=S.car,dist=Math.hypot(c.x-d.x,c.y-d.y),range=180+S.threat*.5;
 const seen=dist<range&&los(d.x,d.y,c.x,c.y);
 if(d.st==='patrol'){orbitDrone(d);
  if(seen){d.st='hunt';d.lose=0;S.threat=Math.max(S.threat,40);Audio.hunt();setWarn('DRONE HUNTING — SHED IT','bad');}}
 else if(d.st==='hunt'){
  if(seen)d.lose=0;else d.lose+=dt;
  const sp=175+S.threat*.35,dx=c.x-d.x,dy=c.y-d.y,dd=Math.hypot(dx,dy)||1;
  d.x+=dx/dd*sp*dt;d.y+=dy/dd*sp*dt;
  if(d.lose>2.5)d.st='return';
  if(dist<28)doBust('ZAPPED BY DRONE');}
 else{const dx=d.ax-d.x,dy=d.ay-d.y,dd=Math.hypot(dx,dy);
  if(dd<30){d.st='patrol';d.t=(Math.atan2((d.y-d.ay)/(d.r*.8),(d.x-d.ax)/d.r)-d.ph*2)/.5;}
  else{d.x+=dx/dd*140*dt;d.y+=dy/dd*140*dt;}
  if(seen){d.st='hunt';d.lose=0;}}
}
function orbitDrone(d){const a=d.t*.5+d.ph*2;d.x=d.ax+Math.cos(a)*d.r;d.y=d.ay+Math.sin(a)*d.r*.8;}

function camStep(c,dt,live){
 c.a=c.mid+c.amp*Math.sin(T*.85+c.x*.01);c.ping=Math.max(0,c.ping-2*dt);
 if(!live)return;
 const dx=S.car.x-c.x,dy=S.car.y-c.y,d=Math.hypot(dx,dy);
 const seen=d<300&&Math.abs(angDiff(Math.atan2(dy,dx),c.a))<.5&&los(c.x,c.y,S.car.x,S.car.y);
 if(seen){c.susp+=2*dt;c.ping=1;
  if(!c.flagged&&c.susp>.75){c.flagged=true;S.threat=Math.max(S.threat,35);
   setWarn('TURRET HAS YOU ON TAPE','bad');alertGuards(S.car.x,S.car.y,null);}}
 else{c.susp=Math.max(0,c.susp-1.2*dt);if(c.susp<.2)c.flagged=false;}
 if(c.susp>=1)doBust('CAUGHT ON CAMERA');}

function carStep(dt){
 const c=S.car;
 const up=keys['w']||keys['arrowup'],dn=keys['s']||keys['arrowdown'];
 const lf=keys['a']||keys['arrowleft'],rt=keys['d']||keys['arrowright'],hb=!!keys[' '];
 const fx=Math.cos(c.a),fy=Math.sin(c.a),rx=-fy,ry=fx;
 let vf=c.vx*fx+c.vy*fy,vl=c.vx*rx+c.vy*ry;
 if(up)vf+=470*dt*(1-Math.max(0,vf)/380);
 if(dn){if(vf>40)vf-=900*dt;else vf=Math.max(-150,vf-260*dt);}
 vf-=vf*Math.min(1,.6*dt);
 vl*=Math.max(0,1-Math.min(1,(hb?2.2:8)*dt));
 const steer=(rt?1:0)-(lf?1:0),te=clamp(Math.abs(vf)/230,0,1)*(vf<0?-1:1);
 c.a+=steer*3.1*te*dt*(hb?1.25:1);
 c.vx=fx*vf+rx*vl;c.vy=fy*vf+ry*vl;
 c.x+=c.vx*dt;c.y+=c.vy*dt;
 let impact=0;
 for(const off of[[14,0],[0,0],[-14,0]]){
  const px=c.x+fx*off[0]+rx*off[1],py=c.y+fy*off[0]+ry*off[1];
  const res=collide(px,py,13);
  if(res){c.x+=res.nx*res.push;c.y+=res.ny*res.push;
   const vn=c.vx*res.nx+c.vy*res.ny;
   if(vn<0){impact=Math.max(impact,-vn);c.vx-=res.nx*vn*1.45;c.vy-=res.ny*vn*1.45;}}}
 const B=55;
 if(c.x<B){c.x=B;if(c.vx<0){impact=Math.max(impact,-c.vx);c.vx*=-.35;}}
 if(c.x>W-B){c.x=W-B;if(c.vx>0){impact=Math.max(impact,c.vx);c.vx*=-.35;}}
 if(c.y<B){c.y=B;if(c.vy<0){impact=Math.max(impact,-c.vy);c.vy*=-.35;}}
 if(c.y>H-B){c.y=H-B;if(c.vy>0){impact=Math.max(impact,c.vy);c.vy*=-.35;}}
 c.spd=Math.hypot(c.vx,c.vy);c.slip=vl;
 S.thudCd-=dt;
 if(impact>240&&S.thudCd<=0){S.thudCd=.35;Audio.thump();S.shake=Math.max(S.shake,5);burst(c.x+fx*16,c.y+fy*16,8,'#ffb03a',220);}
 const skid=Math.abs(vl)>85&&c.spd>130;
 const rl={x:c.x-fx*12+rx*7,y:c.y-fy*12+ry*7},rrw={x:c.x-fx*12-rx*7,y:c.y-fy*12-ry*7};
 if(skid){sctx.strokeStyle='rgba(3,7,11,.5)';sctx.lineWidth=2.5;sctx.beginPath();
  if(c.prevL){sctx.moveTo(c.prevL.x/2,c.prevL.y/2);sctx.lineTo(rl.x/2,rl.y/2);}
  if(c.prevR){sctx.moveTo(c.prevR.x/2,c.prevR.y/2);sctx.lineTo(rrw.x/2,rrw.y/2);}
  sctx.stroke();
  if(Math.random()<.5)puff(rl.x+(Math.random()-.5)*6,rl.y+(Math.random()-.5)*6);}
 c.prevL=skid?{...rl}:null;c.prevR=skid?{...rrw}:null;
}

function playSim(dt){
 carStep(dt);
 const c=S.car;
 for(const ch of S.chips)if(!ch.got&&(c.x-ch.x)**2+(c.y-ch.y)**2<36*36){
  ch.got=true;S.nChips++;Audio.pickup();burst(ch.x,ch.y,14,'#ffd23f',260);ring(ch.x,ch.y,'#ffd23f');floatTxt(ch.x,ch.y-10,'+1000');
  if(S.nChips===8){S.extractOn=true;S.hold=0;Audio.go();setWarn('EXTRACTOR LIVE — GET TO THE PAD','go');}}
 if(S.extractOn){const d=Math.hypot(c.x-EX.x,c.y-EX.y);
  if(d<74){S.hold+=dt/1.5;if(S.hold>=1){doWin();return;}}
  else S.hold=Math.max(0,S.hold-dt*1.3);}
 for(const g of S.guards)guardStep(g,dt,true);
 for(const d of S.drones)droneStep(d,dt,true);
 for(const cm of S.cams)camStep(cm,dt,true);
 const active=S.guards.some(g=>g.locked)||S.drones.some(d=>d.st==='hunt');
 if(!active)S.threat=Math.max(0,S.threat-6*dt);
 S.time-=dt;
 if(S.time<30&&!S.warned30){S.warned30=true;setWarn('TIME CRITICAL — MOVE','bad');Audio.hunt();}
 if(S.time<=0){doBust('OUT OF TIME');return;}
 let ms=0;for(const g of S.guards)ms=Math.max(ms,g.susp);for(const cm of S.cams)ms=Math.max(ms,cm.susp);
 if(ms>.55){S.hbT-=dt;if(S.hbT<=0){Audio.thump();S.hbT=lerp(.62,.3,clamp(ms,0,1));}}else S.hbT=0;
 updParts(dt);syncHud();
 Audio.engine(clamp(c.spd/380,0,1),keys['w']||keys['arrowup']||keys['s']||keys['arrowdown'],Math.min(1,Math.abs(c.slip)/250)*clamp(c.spd/380,0,1));
}
function ambient(dt){

 T+=dt;
 for(const f of FOG){f.x+=f.vx*dt;f.y+=f.vy*dt;if(f.x<-f.r)f.x=W+f.r;if(f.x>W+f.r)f.x=-f.r;if(f.y<-f.r)f.y=H+f.r;if(f.y>H+f.r)f.y=-f.r;}
 if(S)S.shake=Math.max(0,S.shake-26*dt);

  cam.x=lerp(cam.x,1200+Math.cos(T*.07)*430,dt*1.2);cam.y=lerp(cam.y,900+Math.sin(T*.05)*330,dt*1.2);
  cam.s=lerp(cam.s,viewScale(),dt*2);
  $('clk').textContent=new Date().toTimeString().slice(0,8);return;}

  if(S.endT>0){S.endT-=dt;if(S.endT<=0&&!S.endShown)showEnd();}return;}

}
