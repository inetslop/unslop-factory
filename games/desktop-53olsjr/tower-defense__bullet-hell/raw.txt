function step(dt){
 time+=dt;

 coreRingT-=dt; if(coreRingT<=0){coreRingT=1.8;parts.push({type:'r',x:core.x,y:core.y,r:tile*.2,vr:tile*3,life:.7,max:.7,size:2,col:'#3ae7ff'});}

 if(waveActive)while(spawnQ.length&&spawnQ[0].at<=time)enemies.push(makeFoe(spawnQ.shift().type));
 // enemies: reset slow, apply vortex fields
 for(const e of enemies){e.age+=dt;e.slowMul=1;}
 for(const t of towers.values()){if(t.type!==3)continue;const st=towerStats(t);
  for(const e of enemies){const dx=e.x-t.x,dy=e.y-t.y;if(dx*dx+dy*dy<=st.range*st.range){e.slowMul=Math.min(e.slowMul,st.slow);e.hp-=st.dps*dt;}}}
 for(let i=enemies.length-1;i>=0;i--){const e=enemies[i];
  if(e.hp<=0){killFoe(e,i);continue;}
  e.d+=e.spd*tile*e.slowMul*dt;
  if(e.d>=totL){leak(e);enemies.splice(i,1);if(state!=='play')return;continue;}
  const p=posAt(e.d),def=FOES[e.type];let x=p.x,y=p.y;
  if(def.wob){const off=Math.sin(e.age*4.5+e.ph)*.65*tile;x+=Math.cos(p.ang+1.5708)*off;y+=Math.sin(p.ang+1.5708)*off;}
  e.x=x;e.y=y;e.ang=p.ang;if(e.flash>0)e.flash-=dt*6;
  if(def.shoot==='ring'){e.shootT-=dt;if(e.shootT<=0)shootRing(e);}
  else if(def.shoot==='spiral'){e.spirA+=dt*2.4;e.spirT-=dt;
   if(e.spirT<=0){e.spirT=.11;eB(x,y,e.spirA,3.6,8,'#ff2e63',tile*.1+2);eB(x,y,e.spirA+Math.PI,3.6,8,'#ff2e63',tile*.1+2);}

 // towers

  if(t.hurtT>0)t.hurtT-=dt;else t.hp=Math.min(t.maxhp,t.hp+6*dt);

   if(tgt)t.aim=Math.atan2(tgt.y-t.y,tgt.x-t.x);
   if(t.cd<=0&&tgt){fireTower(t,b,st,tgt);t.cd=1/st.rof;}}
  else t.spin+=dt*2;}
 // bullets
 for(let i=bullets.length-1;i>=0;i--){const b=bullets[i];b.x+=b.vx*dt;b.y+=b.vy*dt;b.life-=dt;
  let dead=b.life<=0||b.x<-40||b.y<-40||b.x>W+40||b.y>H+40;
  if(!dead&&b.fly){
   for(const [k,t] of towers){const dx=t.x-b.x,dy=t.y-b.y,rr=tile*.4+b.r;if(dx*dx+dy*dy<rr*rr){
    t.hp-=b.dmg;t.hurtT=5;spark(b.x,b.y,'#ff8fa8',3);dead=true;if(t.hp<=0)destroyTower(k,t);break;}}
   if(!dead){const dx=core.x-b.x,dy=core.y-b.y,rr=tile*.55+b.r;if(dx*dx+dy*dy<rr*rr){spark(b.x,b.y,'#ff5c7a',3);coreDamage(b.dmg);dead=true;}}
  }else if(!dead){
   for(const e of enemies){if(e.hp<=0)continue;const rr=e.rw*tile*.85+b.r,dx=e.x-b.x,dy=e.y-b.y;
    if(dx*dx+dy*dy<rr*rr){e.hp-=b.dmg;e.flash=1;spark(b.x,b.y,b.col,2);sfx.hit();dead=true;break;}}
  }
  if(dead)bullets.splice(i,1);}
 // beams / parts / floats
 for(let i=beams.length-1;i>=0;i--){beams[i].life-=dt;if(beams[i].life<=0)beams.splice(i,1);}
 for(let i=parts.length-1;i>=0;i--){const p=parts[i];p.life-=dt;
  if(p.type==='s'){p.x+=p.vx*dt;p.y+=p.vy*dt;p.vx*=Math.max(0,1-3.5*dt);p.vy*=Math.max(0,1-3.5*dt);}

 if(parts.length>420)parts.splice(0,parts.length-420);
