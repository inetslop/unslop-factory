  d=clamp(d,0,totalLen-.001);
  for(let i=0;i<segLens.length;i++){
    if(d<=segLens[i]){const t=d/segLens[i];
      return {x:lerp(wpx[i][0],wpx[i+1][0],t),y:lerp(wpx[i][1],wpx[i+1][1],t),ang:Math.atan2(wpx[i+1][1]-wpx[i][1],wpx[i+1][0]-wpx[i][0])};}

function acquire(t,x,y){
  const D=TOWERS[t.type],R=D.range*ts,R2=R*R;let best=null,bd=-1;
  for(const e of enemies){
    if(e.dead)continue;
    if(D.noFly&&ETYPES[e.type].flying)continue;
    const p=posAt(e.dist),dx=p.x-x,dy=p.y-y;
    if(dx*dx+dy*dy<=R2&&e.dist>bd){bd=e.dist;best=e;}
  }
  return best;
}
function fireTower(t){
  const D=TOWERS[t.type],x=ox+(t.c+.5)*ts,y=oy+(t.r+.5)*ts,dmg=D.dmg[t.level];
  if(D.kind==='needle'){
    const e=acquire(t,x,y);if(!e)return;
    t.aim=Math.atan2(enemyXY(e).y-y,enemyXY(e).x-x);t.recoil=.12;t.cd=D.rate;
    projs.push({kind:'needle',x,y,tgt:e,sp:10*ts,dmg,ang:t.aim});SFX.thorn();
  }else if(D.kind==='bomb'){
    const e=acquire(t,x,y);if(!e)return;
    const p0=posAt(e.dist),spd=ETYPES[e.type].sp*ts*(e.slowT>0?e.slowF:1);
    const tof=Math.hypot(p0.x-x,p0.y-y)/(5.2*ts);
    const ip=posAt(e.dist+spd*tof);t.aim=Math.atan2(ip.y-y,ip.x-x);t.mflash=.15;t.cd=D.rate;
    projs.push({kind:'bomb',sx:x,sy:y,tx:ip.x,ty:ip.y,t:0,tof,dmg,splash:D.splash*ts});SFX.mortar();
  }else if(D.kind==='pulse'){
    const R=D.range*ts,R2=R*R;let any=false;
    for(const e of enemies){if(e.dead)continue;const p=posAt(e.dist),dx=p.x-x,dy=p.y-y;
      if(dx*dx+dy*dy<=R2){any=true;hit(e,dmg);e.slowT=D.slowDur;e.slowF=1-D.slow[t.level];}}
    if(!any)return;t.cd=D.rate;
    fx.push({kind:'ring',x,y,r0:ts*.15,r1:R,color:D.color,life:.45,tot:.45,lw:2.5});SFX.cryo();
  }else if(D.kind==='chain'){
    const e=acquire(t,x,y);if(!e)return;
    t.cd=D.rate;t.aim=Math.atan2(enemyXY(e).y-y,enemyXY(e).x-x);
    const pts=[[x,y]];const hitList=[];let cur=e,d=dmg;
    for(let i=0;i<D.chains[t.level];i++){
      if(!cur)break;hit(cur,d);hitList.push(cur);d*=.75;
      const cp=enemyXY(cur);pts.push([cp.x,cp.y]);
      let nxt=null,nd=1.5*ts;
      for(const o of enemies){if(o.dead||o===cur||hitList.indexOf(o)>=0)continue;
        const op=enemyXY(o),dd=Math.hypot(op.x-cp.x,op.y-cp.y);if(dd<nd){nd=dd;nxt=o;}}
      cur=nxt;
    }
    const jag=[];for(let i=0;i<pts.length-1;i++){
      const [x1,y1]=pts[i],[x2,y2]=pts[i+1];
      const mx=(x1+x2)/2+rand(-7,7),my=(y1+y2)/2+rand(-7,7);
      jag.push([x1,y1],[mx,my],[x2,y2]);}
    fx.push({kind:'chain',pts:jag,color:D.color,life:.18,tot:.18});SFX.arc();
  }else if(D.kind==='beam'){
    const e=acquire(t,x,y);if(!e)return;
    t.cd=D.rate;t.aim=Math.atan2(enemyXY(e).y-y,enemyXY(e).x-x);
    const ep=enemyXY(e);
    fx.push({kind:'beam',x1:x,y1:y,x2:ep.x,y2:ep.y,color:D.color,life:.24,tot:.24});
    fx.push({kind:'ring',x:ep.x,y:ep.y,r0:2,r1:ts*.35,color:D.color,life:.25,tot:.25});
    hit(e,dmg);SFX.prism();
  }
}

/* ============================== UPDATE ============================== */
function update(dt){
  shake=Math.max(0,shake-dt*22);flashBase=Math.max(0,flashBase-dt*1.8);
  G.drawT+=dt;if(G.drawT>=6){G.drawT=0;drawCard();}

  if(G.state==='build'){
    G.phaseT-=dt;if(G.phaseT<=0){beginWave();return;}
  }else if(G.state==='wave'){
    if(G.queue.length){
      G.spawnT-=dt;
      if(G.spawnT<=0){
        const t=G.queue.shift();
        G.spawnT=WAVES[G.wave-1].gap*rand(.85,1.15)+(G.lastSpawnType&&t!==G.lastSpawnType?.7:0);
        G.lastSpawnType=t;spawnEnemy(t,0);
      }
    }
  }

  // enemies
  for(const e of enemies){
    if(e.dead)continue;
    if(e.slowT>0)e.slowT-=dt;
    const sp=ETYPES[e.type].sp*ts*(e.slowT>0?e.slowF:1);
    e.dist+=sp*dt;e.flash=Math.max(0,e.flash-dt);
    if(e.type==='boss'){
      e.summonT+=dt;
      if(e.summonT>=7){e.summonT=0;spawnEnemy('grub',Math.max(0,e.dist-18));spawnEnemy('grub',Math.max(0,e.dist-46));SFX.zap();}
    }
    if(e.dist>=totalLen){
      e.dead=true;G.hp-=ETYPES[e.type].dmg;flashBase=1;shake=Math.max(shake,7);
      const b=posAt(totalLen);floatText(b.x,b.y-ts*.7,'-'+ETYPES[e.type].dmg,'#ff5d5d');SFX.baseHit();
      if(G.hp<=0){G.hp=0;renderHand();endGame(false);return;}
    }
  }
  enemies=enemies.filter(e=>!e.dead);

  // towers
  for(const t of towers){
    t.recoil=Math.max(0,t.recoil-dt*1.4);t.mflash=Math.max(0,t.mflash-dt*5);
    if(t.cd>0){t.cd-=dt;continue;}
    fireTower(t);
  }

  // projectiles
  for(const p of projs){
    if(p.kind==='needle'){
      let tx,ty;
      if(p.tgt&&!p.tgt.dead){const ep=enemyXY(p.tgt);tx=ep.x;ty=ep.y;p.ang=Math.atan2(ty-p.y,tx-p.x);}
      else{tx=p.x+Math.cos(p.ang)*10;ty=p.y+Math.sin(p.ang)*10;}
      const dx=tx-p.x,dy=ty-p.y,dd=Math.hypot(dx,dy),step=p.sp*dt;
      if(dd<=step+4){
        p.done=true;
        if(p.tgt&&!p.tgt.dead){hit(p.tgt,p.dmg);burst(tx,ty,'#d8f0c8',3);}
        else burst(tx,ty,'#8a9a8a',2);
      }else{p.x+=dx/dd*step;p.y+=dy/dd*step;}
    }else if(p.kind==='bomb'){
      p.t+=dt/p.tof;
      if(p.t>=1){
        p.done=true;fx.push({kind:'boom',x:p.tx,y:p.ty,r:p.splash,life:.4,tot:.4});
        for(const e of enemies){if(e.dead)continue;const ep=enemyXY(e),dx=ep.x-p.tx,dy=ep.y-p.ty;
          if(dx*dx+dy*dy<=p.splash*p.splash)hit(e,p.dmg);}
        SFX.boom();shake=Math.max(shake,2.5);
      }
    }
  }
  projs=projs.filter(p=>!p.done);

  // fx & particles
  for(const f of fx)f.life-=dt;
  fx=fx.filter(f=>f.life>0);
  for(const p of parts){p.x+=p.vx*dt;p.y+=p.vy*dt;p.vy+=(p.g||0)*dt;p.life-=dt;}
  parts=parts.filter(p=>p.life>0);

  // wave cleared?
  if(G.state==='wave'&&!G.queue.length&&!enemies.length){
    if(G.wave>=WAVES.length)endGame(true);
    else buildPhase(false);
  }
}

    ctx.fillStyle=pct>.5?'#8fe08f':pct>.25?'#e8b64c':'#ff5d5d';
    ctx.fillRect(bx,by,bw*pct,T.boss?4:2.5);
  }
  if(T.boss){
    ctx.font='bold 10px Georgia,serif';ctx.textAlign='center';ctx.fillStyle='#e8d4ff';
    ctx.fillText('THE WARDEN',x,y-r-20);
  }
}
function drawTowerShape(t){
  const D=TOWERS[t.type],x=ox+(t.c+.5)*ts,y=oy+(t.r+.5)*ts;
  ctx.save();
  ctx.fillStyle='#121a1f';rr(x-ts*.36,y-ts*.36,ts*.72,ts*.72,5);ctx.fill();
  ctx.strokeStyle=D.color+'55';ctx.lineWidth=1.5;rr(x-ts*.36,y-ts*.36,ts*.72,ts*.72,5);ctx.stroke();
  const s=ts*.3*(1+t.level*.08);
  ctx.translate(x,y);
  if(t.type==='thorn'){
    ctx.rotate((t.aim??-Math.PI/2)+Math.PI/2);
    if(t.recoil>0)ctx.translate(0,t.recoil*ts*.25);
    ctx.fillStyle=D.color;
    for(let i=-1;i<=1;i++){const w=i===0?s*.4:s*.36,h=i===0?s*1.15:s*.85;
      ctx.beginPath();ctx.moveTo(i*s*.42-s*.19,s*.6);ctx.lineTo(i*s*.3, -h);ctx.lineTo(i*s*.42+s*.19,s*.6);ctx.closePath();ctx.fill();}
    ctx.fillStyle='#c9f0b8';ctx.beginPath();ctx.arc(0,-s*1.05,s*.09,0,TAU);ctx.fill();
  }else if(t.type==='mortar'){
    ctx.rotate((t.aim??-Math.PI/2)+Math.PI/2);
    ctx.fillStyle='#3a2c22';ctx.beginPath();ctx.arc(0,s*.15,s*.68,0,TAU);ctx.fill();
    ctx.strokeStyle=D.color;ctx.lineWidth=2;ctx.stroke();
    ctx.fillStyle=D.color;ctx.fillRect(-s*.17,-s*1.05,s*.34,s*.95);
    if(t.mflash>0){ctx.fillStyle='rgba(255,220,140,'+t.mflash*6+')';ctx.beginPath();ctx.arc(0,-s*1.05,s*.4,0,TAU);ctx.fill();}
  }else if(t.type==='cryo'){
    const br=1+.08*Math.sin(nowT*3+t.seed);
    ctx.rotate(Math.PI/4);
    ctx.fillStyle='rgba(95,215,255,.85)';ctx.fillRect(-s*.6*br,-s*.6*br,s*1.2*br,s*1.2*br);
    ctx.fillStyle='rgba(255,255,255,'+(.5+.3*Math.sin(nowT*4+t.seed))+')';
    ctx.fillRect(-s*.28,-s*.28,s*.56,s*.56);
  }else if(t.type==='arc'){
    ctx.strokeStyle=D.color;ctx.lineWidth=2;
    ctx.beginPath();ctx.arc(0,0,s*.72,0,TAU);ctx.stroke();
    ctx.fillStyle='#e8dcff';ctx.beginPath();ctx.arc(0,0,s*.26,0,TAU);ctx.fill();
    for(let i=0;i<2;i++){ctx.strokeStyle='rgba(176,122,255,.9)';ctx.lineWidth=2.4;
      ctx.beginPath();ctx.arc(0,0,s*.5,nowT*2.2+i*Math.PI,nowT*2.2+i*Math.PI+1.5);ctx.stroke();}
    const sa=nowT*4+t.seed;
    ctx.fillStyle='#fff';ctx.beginPath();ctx.arc(Math.cos(sa)*s*.5,Math.sin(sa)*s*.5,2,0,TAU);ctx.fill();
  }else if(t.type==='prism'){
    ctx.rotate(nowT*.8);
    ctx.fillStyle=D.color;
    ctx.beginPath();ctx.moveTo(0,-s*1.05);ctx.lineTo(s*.3,0);ctx.lineTo(0,s*1.05);ctx.lineTo(-s*.3,0);ctx.closePath();ctx.fill();
    ctx.beginPath();ctx.moveTo(-s*1.05,0);ctx.lineTo(0,-s*.3);ctx.lineTo(s*1.05,0);ctx.lineTo(0,s*.3);ctx.closePath();ctx.fill();
    ctx.fillStyle='#2a1c06';ctx.beginPath();ctx.arc(0,0,s*.2,0,TAU);ctx.fill();
  }
  ctx.restore();
  for(let i=0;i<3;i++){
    ctx.fillStyle=i<=t.level?'#ffd76a':'#33403c';
    ctx.beginPath();ctx.arc(x-8+i*8,y+ts*.3,2.4,0,TAU);ctx.fill();
  }
}
