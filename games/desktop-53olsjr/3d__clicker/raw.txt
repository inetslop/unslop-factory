const G=(function(){
  const V=[],F=[];
  const top={x:0,y:1.85,z:0},bot={x:0,y:-1.15,z:0};
  const U=[],Lr=[];
  for(let i=0;i<6;i++){const a=i/6*TAU-Math.PI/2;
    U.push({x:Math.cos(a)*.6,y:.42,z:Math.sin(a)*.6});
    Lr.push({x:Math.cos(a+Math.PI/6)*.78,y:-.38,z:Math.sin(a+Math.PI/6)*.78});}
  V.push(top,bot);U.forEach(v=>V.push(v));Lr.forEach(v=>V.push(v));
  const t=0,u0=2,l0=8; // indices in V
  function addFace(idx){
    const p=idx.map(i=>V[i]);
    let nx=(p[1].y-p[0].y)*(p[2].z-p[0].z)-(p[1].z-p[0].z)*(p[2].y-p[0].y);
    let ny=(p[1].z-p[0].z)*(p[2].x-p[0].x)-(p[1].x-p[0].x)*(p[2].z-p[0].z);
    let nz=(p[1].x-p[0].x)*(p[2].y-p[0].y)-(p[1].y-p[0].y)*(p[2].x-p[0].x);
    const cx=(p[0].x+p[1].x+(p.length>3?p[3].x:0))/ (p.length>3?4:3),
          cy=(p[0].y+p[1].y+(p.length>3?p[3].y:0))/ (p.length>3?4:3),
          cz=(p[0].z+p[1].z+(p.length>3?p[3].z:0))/ (p.length>3?4:3);
    if(nx*cx+ny*cy+nz*cz<0){idx=[idx[0],idx[idx.length-1],idx[idx.length-2]].concat(idx.slice(1,idx.length-2));}
    // recompute normal after possible flip
    const q=idx.map(i=>V[i]);
    nx=(q[1].y-q[0].y)*(q[2].z-q[0].z)-(q[1].z-q[0].z)*(q[2].y-q[0].y);
    ny=(q[1].z-q[0].z)*(q[2].x-q[0].x)-(q[1].x-q[0].x)*(q[2].z-q[0].z);
    nz=(q[1].x-q[0].x)*(q[2].y-q[0].y)-(q[1].y-q[0].y)*(q[2].x-q[0].x);
    const l=Math.hypot(nx,ny,nz)||1;
    F.push({idx:n:[nx/l,ny/l,nz/l],tint:(R()-.5)*12,spec:.6+R()*.8,flash:0});
  }
  for(let i=0;i<6;i++){const j=(i+1)%6;
    addFace([t,u0+i,u0+j]);
    addFace([u0+i,u0+j,l0+j]);addFace([u0+i,l0+j,l0+i]);
    addFace([bot,l0+j,l0+i]);}
  return {V,F};
})();
