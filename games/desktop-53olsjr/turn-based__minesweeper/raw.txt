  for(let i=pool.length-1;i>0;i--){ const j=(Math.random()*(i+1))|0; const t=pool[i];pool[i]=pool[j];pool[j]=t; }
  for(let k=0;k<S.mines;k++) S.cells[pool[k]].mine=true;
  S.cells.forEach((c,i)=>{ c.num=0; if(c.mine)return;
    nbrs(i).forEach(n=>{ if(S.cells[n].mine)c.num++; }); });
  S.started=true;
}

/* ---------- actions ---------- */
function openCell(i,by,delay){
  const c=S.cells[i]; c.st='o'; c.by=by; S.revealedCount++;
  paint(i); pop(c.el,delay);
}
function openFlood(i0,by){
  const list=[], q=[[i0,0]], seen=new Set([i0]);
  while(q.length){
    const [i,d]=q.shift(), c=S.cells[i];
    if(c.st!=='h') continue;
    openCell(i,by,d*24); list.push(i);
    if(c.num===0) for(const n of nbrs(i)) if(!seen.has(n)){ seen.add(n); q.push([n,d+1]); }
  }
  SFX.open(Math.min(8,S.cells[i0].num||1));
  return list;
}
function doReveal(i){
  if(!S||S.over)return;
  const c=S.cells[i];
  if(S.mode==='duel'&&(S.locked||S.turn!=='you')){ SFX.deny(); stripDeny(); return; }
  if(c.st==='f'){ SFX.deny(); return; }
  if(c.st==='o'){ if(c.num>0) doChord(i,'you'); return; }
  if(!S.started) placeMines(i);
  if(c.mine){ lose(i,'you'); return; }
  openFlood(i,'you'); S.totalMoves++; S.youMoves++;
  log('T'+String(S.mode==='duel'?S.turnNo:S.totalMoves).padStart(2,'0')+' · YOU OPEN '+coord(i)+(c.num>0?' → '+c.num:' → BUST'));
  afterMove('you');
}
function doFlag(i,by){
  if(!S||S.over)return; by=by||'you';
  const c=S.cells[i];
  if(c.st==='o')return;
  if(S.mode==='duel'&&by==='you'&&(S.locked||S.turn!=='you')){ SFX.deny(); stripDeny(); return; }
  if(c.st==='f'){
    if(S.mode==='duel'&&c.by!==by)return;
    c.st='h'; c.by=null; paint(i); S.flagCount--; SFX.unflag();
    if(by==='you'&&S.mode==='duel')S.flagsYou++;
    updateHUD(); log((by==='you'?'YOU UNFLAG ':'ENEMY UNFLAG ')+coord(i)); return;
  }
  if(S.mode==='duel'&&by==='you'&&S.flagsYou<=0){ SFX.deny(); stripDeny(); return; }
  c.st='f'; c.by=by; paint(i); S.flagCount++; SFX.flag(); vibrate(25);
  if(by==='you') log('YOU FLAG '+coord(i));
  updateHUD();
}
function doChord(i,by){
  const c=S.cells[i]; if(c.st!=='o'||c.num===0)return;
  const ns=nbrs(i), flg=ns.filter(x=>S.cells[x].st==='f').length;
  if(flg!==c.num)return;
  let opened=0, danger=null;
  for(const n of ns){ const nc=S.cells[n]; if(nc.st!=='h')continue;
    if(nc.mine)danger=n; else openCell(n,by); opened++; }
  if(danger!=null){ lose(danger,by); return; }
  if(opened>0){ SFX.chord(); S.totalMoves++; if(by==='you')S.youMoves++;
    log((by==='you'?'CHORD ':'ENEMY CHORD ')+coord(i)+' → +'+opened); afterMove(by); }
}

/* ---------- turn engine ---------- */
function hasSafeMove(){ return S.cells.some(c=>(c.st==='h'||c.st==='f')&&!c.mine); }
function afterMove(by){
  if(S.over)return;
  if(S.mode==='solo'){
    if(S.revealedCount===S.cols*S.rows-S.mines) winGame();
    return;
  }
  const next=other(by);
  if(!hasSafeMove()){ endDuel(by,'elim'); return; }
  if(next==='ai') startAITurn();
  else{ S.turn='you'; S.locked=false; S.turnNo++; updateHUD(); log('TURN '+S.turnNo+' · YOUR MOVE'); }
}
function startAITurn(){
  S.turn='ai'; S.locked=true; updateHUD(); log('ENEMY TURN '+S.turnNo); SFX.ai();
  S.aiT=setTimeout(aiAct, 900+Math.random()*850);
}
function aiAct(){
  if(!S||S.over)return;
  if(S.flagsAI>0){ const f=forcedFlag();
    if(f!=null){ doFlag(f,'ai'); S.flagsAI--; log('ENEMY FLAGS '+coord(f)); } }
  const i=pickCell(false);
  if(i==null){ endDuel('you','elim'); return; }
  const el=S.cells[i].el; el.classList.add('aiTgt'); beep(760,520,.1,'sine',.06);
  setTimeout(()=>{
    if(!S||S.over)return; el.classList.remove('aiTgt');
    if(S.cells[i].mine){ lose(i,'ai'); return; }
    openFlood(i,'ai'); S.totalMoves++;
    log('ENEMY PROBES '+coord(i)+(S.cells[i].num>0?' → '+S.cells[i].num:' → BUST'));
    afterMove('ai');
  },460);
}
function forcedFlag(){
  for(let i=0;i<S.cells.length;i++){ const c=S.cells[i]; if(c.st!=='o'||c.num===0)continue;
    const ns=nbrs(i), hid=ns.filter(x=>S.cells[x].st==='h'), flg=ns.filter(x=>S.cells[x].st==='f').length;
    if(hid.length>0&&c.num-flg===hid.length) return hid[0]; }
  return null;
}
function pickCell(det){
  const opts=[];
  for(let i=0;i<S.cells.length;i++){
    const c=S.cells[i]; if(c.st!=='h')continue;
    let r=0, any=false;
    for(const n of nbrs(i)){ const nc=S.cells[n]; if(nc.st!=='o'||nc.num===0)continue;
      any=true;
      const nn=nbrs(n), hid=nn.filter(x=>S.cells[x].st==='h').length, flg=nn.filter(x=>S.cells[x].st==='f').length;
      const left=nc.num-flg;
      if(hid<=0)r+=.01; else if(left<=0)r+=.02; else if(left>=hid)r+=45; else r+=(left/hid)*9; }
    if(!any) r=(S.mines/(S.cols*S.rows))*7*(det?1:(0.6+Math.random()*.9));
    if(!det) r+=Math.random()*.35;
    opts.push({i,r});
  }
  if(!opts.length)return null;
  opts.sort((a,b)=>a.r-b.r);
  const k=opts.slice(0,Math.min(3,opts.length));
  if(det)return k[0].i;
  let tot=k.reduce((s,o)=>s+1/(1+o.r*.2),0), t=Math.random()*tot;
  for(const o of k){ t-=1/(1+o.r*.2); if(t<=0)return o.i; }
  return k[k.length-1].i;
}

/* ---------- endings ---------- */
function lose(mi,by){
  S.over=true; S.winner=other(by); setFace('dead'); SFX.boom(); vibrate(120);
  S.endInfo={mine:mi,by:by};
  wellEl.classList.add('shake'); setTimeout(()=>wellEl.classList.remove('shake'),550);
  const mines=[]; S.cells.forEach((c,i)=>{ if(c.mine)mines.push(i); });
  mines.forEach((i,k)=>setTimeout(()=>{
    if(!S||!S.over)return;
    const c=S.cells[i]; c.st='o'; paint(i);
    if(i===mi)c.el.classList.add('boom');
    else pop(c.el,0);
  },140+k*26));
  S.cells.forEach(c=>{ if(c.st==='f'&&!c.mine)c.el.classList.add('bad'); });
  log(by==='you'?'YOU HIT A MINE AT '+coord(mi):'ENEMY HIT A MINE AT '+coord(mi));
  updateHUD();
  setTimeout(showEnd,1600);
}
function winGame(){
  S.over=true; S.winner='you'; setFace('win'); SFX.win();
  S.cells.forEach(c=>{ if(c.mine&&c.st==='h'){ c.st='f'; c.by=null; paint(i_of(c)); } });
  function i_of(c){ return S.cells.indexOf(c); }
  let secs=Math.ceil(S.time), rec=false;
  const b=best.solo||(best.solo={});
  if(b[S.mode+'_'+S.diff]==null || secs<b[S.mode+'_'+S.diff]){ b[S.mode+'_'+S.diff]=secs; saveBest(); rec=true; }
  S.newRecord=rec; S.endInfo={win:true,secs:secs};
  confetti(); log('FIELD CLEARED IN '+fmt(S.time)); updateHUD();
  setTimeout(showEnd,1100);
}
function endDuel(winner,how){
  S.over=true; S.winner=winner;
  setFace(winner==='you'?'win':'dead');
  if(winner==='you'){ SFX.win(); confetti(); } else { SFX.boom(); wellEl.classList.add('shake'); setTimeout(()=>wellEl.classList.remove('shake'),550); }
  let rec=false;
  if(winner==='you'){
    const b=best.duel||(best.duel={});
    if(b['duel_'+S.diff]==null || S.youMoves<b['duel_'+S.diff]){ b['duel_'+S.diff]=S.youMoves; saveBest(); rec=true; }
  }
  S.newRecord=rec; S.endInfo={duelWin:winner==='you',how:how};
  log(winner==='you'?'ENEMY WIPED OUT — VICTORY':'YOU ARE K.I.A.');
  updateHUD();
  setTimeout(showEnd,1300);
}
