window.addEventListener('keydown',e=>{
  if(e.ctrlKey||e.metaKey||e.altKey||e.isComposing)return;
  const k=e.key;
  if(k===' '){e.preventDefault();}
  if(e.repeat){return;}
  if(G.state==='playerTurn'){
    if(k==='Backspace'){
      if(G.typed.length){G.typed.pop();sfx.back();renderWord();}
      e.preventDefault();return;
    }
    if(k.length===1&&/[a-zA-Z0-9 ]/.test(k))handleChar(k.toLowerCase());
    return;
  }
  if(k==='m'||k==='M'){toggleMute();return;}
  if(k==='r'||k==='R'){restart();return;}
  if(G.state==='menu'&&(k==='Enter'||k===' ')){startGame();return;}
  if((G.state==='victory'||G.state==='defeat')&&(k==='Enter'||k===' ')){restart();}
});
window.addEventListener('pointerdown',()=>{audio();},{once:false});

function handleChar(ch){
  G.stats.keys++;
  const i=G.typed.length,tgt=G.word;
  if(i<tgt.length&&ch===tgt[i]){
    G.typed.push(ch);G.stats.correct++;sfx.tick(G.combo);renderWord();
    if(G.typed.length===tgt.length)onWordComplete();
  }else{
    G.stats.mistakes++;G.combo=0;updateCombo();sfx.err();
    flashErrChar(Math.min(i,tgt.length-1));
    setHint('Steady hands — the streak is broken');
  }
}

function update(dt,t){
  const now=performance.now();
  G.shake=Math.max(0,G.shake-dt*34);
  G.flashE=Math.max(0,G.flashE-dt*4);
  G.knock=Math.max(0,G.knock-dt*3.2);
  G.hitFlash=Math.max(0,G.hitFlash-dt*1.5);
  hitFlashEl.style.opacity=G.hitFlash.toFixed(3);
  const sx=(Math.random()-.5)*G.shake,sy=(Math.random()-.5)*G.shake;
  wrapEl.style.transform=G.shake>.2?`translate(${sx}px,${sy}px)`:'';

  if(G.state==='intro'&&now>=G.until)startPlayerTurn();
  else if(G.state==='resolving'&&now>=G.until){
    if(!G.enemy||G.enemy.hp<=0)killEnemy();else startWindup();
  }
  else if(G.state==='dying'&&now>=G.until)afterDeath();
  else if(G.state==='windup'&&now>=G.until)enemyStrike();
  else if(G.state==='victory'){
    G.confT-=dt;
    if(G.confT<=0){G.confT=.12;
      burst(W/2+(Math.random()*W*.5-W*.25),H*.86,['#f2b94b','#7ee081','#63d8d2','#ff8a4a'],6,300,true);
    }
    if(now>=G.until&&!overlayEl.hidden===false)showEnd(true);
  }
  else if(G.state==='defeat'&&now>=G.until)showEnd(false);

  /* timer bar */
  if(G.state==='playerTurn'){
    const pct=Math.max(0,(G.deadline-now)/(G.turnTotal*1000)*100);
    tfill.style.width=pct+'%';
    tfill.className=pct<22?'crit':pct<50?'warn':'';
  }else{tfill.style.width='0%';}

  /* particles */
  for(let i=G.parts.length-1;i>=0;i--){const p=G.parts[i];
    p.life-=dt;if(p.life<=0){G.parts.splice(i,1);continue;}
    if(p.type==='ring'){p.r+=p.vr*dt;continue;}
    p.x+=p.vx*dt;p.y+=p.vy*dt;p.vy+=(p.g||0)*dt;
  }
  for(let i=G.texts.length-1;i>=0;i--){const p=G.texts[i];
    p.life-=dt;p.y+=p.vy*dt;if(p.life<=0)G.texts.splice(i,1);}

  /* ambient embers */
  if(G.embers.length<38&&Math.random()<.5){
    G.embers.push({x:Math.random()*W,y:H+8,vy:-(10+Math.random()*26),ph:Math.random()*7,
      sz:1+Math.random()*2,a:.14+Math.random()*.2,
      col:['#f2b94b','#ff8a4a','#63d8d2'][Math.floor(Math.random()*3)]});
  }
  for(let i=G.embers.length-1;i>=0;i--){const e=G.embers[i];
    e.y+=e.vy*dt;e.x+=Math.sin(t*1.2+e.ph)*.4;
    if(e.y<-10)G.embers.splice(i,1);
  }

  /* menu previews */
  if(G.state==='menu'){
    for(const f of G.foePreviews){
      const g=f.cv.getContext('2d');g.clearRect(0,0,96,78);
      g.save();g.translate(48,66);
      f.def.paint(g,t,{s:.3,flash:false,lunge:0,blink:Math.sin(t*1.4+f.def.phase)>.97,death:0});
      g.restore();
    }
  }
}

function draw(t){
  ctx.clearRect(0,0,W,H);
  /* embers */
  for(const e of G.embers){
    ctx.globalAlpha=e.a;ctx.fillStyle=e.col;
    ctx.beginPath();ctx.arc(e.x,e.y,e.sz,0,7);ctx.fill();
  }
  ctx.globalAlpha=1;

  if(G.state==='menu')return;
  if(!G.enemy)return;

  const p=epos(),def=G.enemy.def,s=p.s;
  let alpha=1,scale=1,lunge=0,death=0;
  if(G.state==='intro'){
    const a=Math.min(1,(performance.now()-(G.until-1600))/700);
    alpha=a;scale=.8+.2*(a*a*(3-2*a));
  }
  if(G.state==='windup')lunge=1-Math.max(0,G.until-performance.now())/750;
  if(G.state==='dying')death=Math.min(1,(performance.now()-G.deathT)/900);

  /* arcane ring + glow */
  ctx.save();ctx.translate(p.x,p.y);
  const gl=ctx.createRadialGradient(0,-90*s,10,0,-90*s,240*s);
  gl.addColorStop(0,'rgba(242,185,75,.12)');gl.addColorStop(1,'rgba(242,185,75,0)');
  ctx.fillStyle=gl;ctx.beginPath();ctx.arc(0,-90*s,240*s,0,7);ctx.fill();
  ctx.strokeStyle='rgba(242,185,75,.2)';ctx.lineWidth=1.5;ctx.setLineDash([3,11]);
  ctx.lineDashOffset=-t*30;ctx.beginPath();ctx.arc(0,-90*s,165*s,0,7);ctx.stroke();
  ctx.strokeStyle='rgba(99,216,210,.14)';ctx.lineDashOffset=t*22;
  ctx.beginPath();ctx.arc(0,-90*s,120*s,0,7);ctx.stroke();
  ctx.setLineDash([]);
  /* ground shadow */
  ctx.fillStyle=`rgba(0,0,0,${def.ground*alpha})`;
  ctx.beginPath();ctx.ellipse(0,4,95*s*scale,16*s,0,0,7);ctx.fill();
  /* enemy */
  ctx.save();
  ctx.globalAlpha=alpha*(death?1-death:1);
  if(death)ctx.translate(0,-def.boss? -death*30 : -death*30);
  const ds=death?(1-death*.5):1;
  ctx.scale(scale*ds,scale*ds*(1-death*.2));
  if(death&&G.state==='dying')ctx.translate(0,-death*70);
  def.paint(ctx,t,{s,flash:G.flashE>0.4,lunge,blink:Math.sin(t*1.4+def.phase)>.97,death});
  ctx.restore();ctx.restore();

  /* particles */
  for(const q of G.parts){
    const a=Math.max(0,q.life/q.max);
    if(q.type==='ring'){
      ctx.globalAlpha=a*.8;ctx.strokeStyle=q.col;ctx.lineWidth=3;
      ctx.beginPath();ctx.arc(q.x,q.y,q.r,0,7);ctx.stroke();
    }else{
      ctx.globalAlpha=a;ctx.fillStyle=q.col;
      ctx.beginPath();ctx.arc(q.x,q.y,q.size*a+1,0,7);ctx.fill();
    }
  }
  /* damage texts */
  ctx.textAlign='center';
  for(const q of G.texts){
    const a=Math.max(0,q.life/q.max);
    ctx.globalAlpha=a;
    ctx.font=`900 ${q.size}px Impact,"Arial Black",sans-serif`;
    ctx.lineWidth=5;ctx.strokeStyle='rgba(8,16,20,.85)';
    ctx.strokeText(q.txt,q.x,q.y);
    ctx.fillStyle=q.col;ctx.fillText(q.txt,q.x,q.y);
  }
  ctx.globalAlpha=1;
}

/* ============================== LOOP ============================== */
function frame(ts){
  const t=ts/1000,dt=Math.min(.05,lastTs?(ts-lastTs)/1000:.016);lastTs=ts;
  update(dt,t);draw(t);
  requestAnimationFrame(frame);
}
