  for(let i=0;i<46;i++)shard(rand(W*0.1,W*0.9),rand(playTop+30,playTop+H*0.3),PAL[String(1+(Math.random()*6|0))],1,0.7);
  showBanner('Level Clear','+'+fmt(bonus)+' bonus');}

function stepBall(b,dt){
  const dist=b.spd*dt,steps=Math.max(1,Math.ceil(dist/b.r)),sdt=dt/steps;
  for(let i=0;i<steps;i++){
    b.x+=b.dirx*b.spd*sdt;b.y+=b.diry*b.spd*sdt;
    if(b.x<b.r+2){b.x=b.r+2;b.dirx=Math.abs(b.dirx);SFX.wall();spark(b.x,b.y,'#cbb691',3);}
    else if(b.x>W-b.r-2){b.x=W-b.r-2;b.dirx=-Math.abs(b.dirx);SFX.wall();spark(b.x,b.y,'#cbb691',3);}
    if(b.y<playTop+b.r){b.y=playTop+b.r;b.diry=Math.abs(b.diry);SFX.wall();}
    if(b.diry>0){const px=paddle.x-paddle.w/2,py=paddle.y;
      if(circleHit(b.x,b.y,b.r,px,py,paddle.w,paddle.h)){
        const rel=clamp((b.x-paddle.x)/(paddle.w/2),-1,1);
        const a=-Math.PI/2+rel*1.05;b.dirx=Math.cos(a);b.diry=Math.sin(a);
        b.y=py-b.r-0.5;paddle.squash=1;hits++;combo=0;comboChip();SFX.paddle();spark(b.x,py,'#f4d98b',4);}}
    for(const br of bricks){if(!br.alive)continue;
      if(Math.abs(b.x-(br.x+br.w/2))>b.r+br.w/2)continue;
      if(Math.abs(b.y-(br.y+br.h/2))>b.r+br.h/2)continue;
      const nx=clamp(b.x,br.x,br.x+br.w),ny=clamp(b.y,br.y,br.y+br.h),dx=b.x-nx,dy=b.y-ny;
      if(dx*dx+dy*dy>b.r*b.r)continue;
      const bcx=br.x+br.w/2,bcy=br.y+br.h/2;
      const ox=(b.r+br.w/2)-Math.abs(b.x-bcx),oy=(b.r+br.h/2)-Math.abs(b.y-bcy);
      if(ox<oy)b.dirx=b.x<bcx?-Math.abs(b.dirx):Math.abs(b.dirx);
      else b.diry=b.y<bcy?-Math.abs(b.diry):Math.abs(b.diry);
      if(Math.abs(b.diry)<0.12){b.diry=(b.diry<=0?-1:1)*0.16;const m=Math.hypot(b.dirx,b.diry);b.dirx/=m;b.diry/=m;}
      nudge(b,rand(-0.03,0.03));hitBrick(b,br,false);break;}
  }
}

/* ============================== update ============================== */
function update(dt){
  animT+=dt;
  if(state==='pause'){return;}
  // ambient embers
  if(embers.length<36&&Math.random()<dt*10)
    embers.push({x:rand(0,W),y:H+8,vy:-rand(14,42),ph:rand(0,6.3),size:rand(1,2.6),life:rand(4,9)});
  for(let i=embers.length-1;i>=0;i--){const e=embers[i];e.y+=e.vy*dt;e.x+=Math.sin(animT*0.8+e.ph)*12*dt;e.life-=dt;
    if(e.life<0||e.y<-10)embers.splice(i,1);}

  if(state==='menu'){ // idle demo ball
    const mb=menuBall;mb.dx*=W;mb.dy*=H;mb.x+=mb.dx*0.5*dt*2;mb.y+=mb.dy*0.5*dt*2;
    if(mb.x<40||mb.x>W-40)mb.dx*=-1;if(mb.y<70||mb.y>H-80)mb.dy*=-1;
    mb.trail.push({x:mb.x,y:mb.y});if(mb.trail.length>15)mb.trail.shift();return;}

  if(state==='serve'||state==='play'){T+=dt;
    // paddle
    const pw=basePaddleW()*(wideUntil>T?1.45:1);paddle.w=lerp(paddle.w,pw,Math.min(1,dt*8));
    const pspd=W*1.1;
    if(keyL)paddle.x-=pspd*dt;if(keyR)paddle.x+=pspd*dt;
    else if(pointerX!=null)paddle.x=lerp(paddle.x,pointerX,Math.min(1,dt*18));
    paddle.x=clamp(paddle.x,paddle.w/2+4,W-paddle.w/2-4);
    paddle.squash=Math.max(0,paddle.squash-dt*5);

    // balls
    for(const b of balls){
      if(b.stuck){b.x=paddle.x;b.y=paddle.y-paddle.h/2-b.r-3;continue;}
      b.spd=lerp(b.spd,targetSpeed(),Math.min(1,dt*3));
      stepBall(b,dt);
      b.trail.push({x:b.x,y:b.y});if(b.trail.length>13)b.trail.shift();}
    // fell below
    for(let i=balls.length-1;i>=0;i--)if(balls[i].y>H+ballR*3)balls.splice(i,1);
    if(state==='play'&&balls.length===0){
      lives--;renderLives();shakeV=16;flashRed();SFX.lose();
      if(lives<=0)gameOver();else serve();}

    // lasers
    if(laserUntil>T&&state==='play'){laserCd-=dt;
      if(laserCd<=0){lasers.push({x:paddle.x-paddle.w*0.38,y:paddle.y-4},{x:paddle.x+paddle.w*0.38,y:paddle.y-4});laserCd=0.38;}}
    for(let i=lasers.length-1;i>=0;i--){const l=lasers[i];l.y-=H*1.5*dt;let dead=l.y<playTop-24;
      if(!dead)for(const br of bricks){if(!br.alive)continue;
        if(l.x>br.x&&l.x<br.x+br.w&&l.y>br.y&&l.y<br.y+br.h){hitBrick({diry:-1},br,true);spark(l.x,l.y,'#ff8b96',3);dead=true;break;}}
      if(dead)lasers.splice(i,1);}

    // powerups
    for(let i=powerups.length-1;i>=0;i--){const p=powerups[i];p.y+=p.vy*dt;p.ph+=dt;
      const px=paddle.x-paddle.w/2;
      if(p.y>paddle.y-14&&p.y<paddle.y+paddle.h+14&&Math.abs(p.x-paddle.x)<paddle.w/2+16){applyPower(p.kind);powerups.splice(i,1);continue;}
      if(p.y>H+20)powerups.splice(i,1);}

    // bricks fx timers + shimmer
    for(const br of bricks){if(!br.alive)continue;br.flash=Math.max(0,br.flash-dt*4);
      if(br.type!=='stone'&&br.sparkT<=0&&Math.random()<dt*0.25)br.sparkT=1;}
    for(const br of bricks)if(br.sparkT>0)br.sparkT-=dt*2.2;
  }

  if(state==='clear'){T+=dt;if(T>=clearEnd){
    if(level+1<LEVELS.length){buildLevel(level+1);serve();showBanner('Level '+ROMAN[level],LEVELS[level].name);}
    else winGame();}}

  // particles / floaters
  const g=900*(H/700);
  for(let i=particles.length-1;i>=0;i--){const p=particles[i];p.life-=dt;
    if(p.life<=0){particles.splice(i,1);continue;}
    p.x+=p.vx*dt;p.y+=p.vy*dt;if(p.t==='shard'){p.vy+=g*dt;p.rot+=p.vr*dt;}else{p.vx*=0.96;p.vy*=0.96;}}
  for(let i=floaters.length-1;i>=0;i--){const f=floaters[i];f.life-=dt*0.9;f.y-=28*dt;if(f.life<=0)floaters.splice(i,1);}

  shakeV=Math.max(0,shakeV-46*dt);
  flashA=Math.max(0,flashA-dt*1.3);$('flash').style.opacity=flashA.toFixed(3);
  dispScore+=(score-dispScore)*Math.min(1,dt*8);
  const ds=Math.round(dispScore);if($('score').textContent!==fmt(ds))$('score').textContent=fmt(ds);
}

function rr(x,y,w,h,r){ctx.beginPath();ctx.moveTo(x+r,y);ctx.arcTo(x+w,y,x+w,y+h,r);ctx.arcTo(x+w,y+h,x,y+h,r);
  ctx.arcTo(x,y+h,x,y,r);ctx.arcTo(x,y,x+w,y,r);ctx.closePath();}

function drawBrick(b){
  const{x,y,w,h}=b;let base='#888';
  if(b.type==='glass'){base=PAL[b.ch];
    const g=ctx.createLinearGradient(x,y,x,y+h);g.addColorStop(0,shade(base,0.35));g.addColorStop(.45,base);g.addColorStop(1,shade(base,-0.38));
    ctx.fillStyle=g;ctx.fillRect(x,y,w,h);
    ctx.strokeStyle='rgba(255,255,255,.3)';ctx.lineWidth=1;
    ctx.beginPath();ctx.moveTo(x+1,y+h-1);ctx.lineTo(x+1,y+1);ctx.lineTo(x+w-1,y+1);ctx.stroke();
  }else if(b.type==='tough'){
    const g=ctx.createLinearGradient(x,y,x,y+h);g.addColorStop(0,'#b98c4a');g.addColorStop(.5,'#8a6231');g.addColorStop(1,'#5c401d');
    ctx.fillStyle=g;ctx.fillRect(x,y,w,h);
    ctx.fillStyle='rgba(255,230,180,.5)';
    const rv=Math.min(2.2,w*0.09);[[x+rv+2,y+rv+2],[x+w-rv-2,y+rv+2],[x+rv+2,y+h-rv-2],[x+w-rv-2,y+h-rv-2]]
      .forEach(p=>{ctx.beginPath();ctx.arc(p[0],p[1],rv,0,7);ctx.fill();});
    if(b.hp===1){let s=b.c*9301+b.r*49297+7;const rnd=()=>((s=(s*9301+49297)%233280)/233280);
      ctx.strokeStyle='rgba(20,10,5,.75)';ctx.lineWidth=1.4;ctx.beginPath();
      let cx=x+rnd()*w*0.3,cy=y;ctx.moveTo(cx,cy);
      for(let i=0;i<4;i++){cx+=rnd()*w*0.25+w*0.08;cy+=h/4*(0.6+rnd()*0.8);ctx.lineTo(clamp(cx,x+2,x+w-2),clamp(cy,y+2,y+h-2));}
      ctx.stroke();}
  }else if(b.type==='stone'){
    const g=ctx.createLinearGradient(x,y,x,y+h);g.addColorStop(0,'#4a5262');g.addColorStop(.5,'#343b49');g.addColorStop(1,'#232833');
    ctx.fillStyle=g;ctx.fillRect(x,y,w,h);
    ctx.strokeStyle='rgba(255,255,255,.16)';ctx.lineWidth=1;
    ctx.beginPath();ctx.moveTo(x+1,y+1.5);ctx.lineTo(x+w-1,y+1.5);ctx.stroke();
    ctx.strokeStyle='rgba(0,0,0,.4)';
    ctx.beginPath();ctx.moveTo(x+3,y+h*0.62);ctx.lineTo(x+w-3,y+h*0.62);ctx.stroke();
  }else if(b.type==='gem'){
    const pulse=0.5+0.5*Math.sin(animT*3+b.c);
    ctx.save();ctx.globalCompositeOperation='lighter';
    const gl=ctx.createRadialGradient(x+w/2,y+h/2,1,x+w/2,y+h/2,w*0.7);
    gl.addColorStop(0,'rgba(255,230,150,'+(0.25+pulse*0.3)+')');gl.addColorStop(1,'rgba(255,230,150,0)');
    ctx.fillStyle=gl;ctx.fillRect(x-w*0.4,y-h*0.6,w*1.8,h*2.2);ctx.restore();
    const g=ctx.createLinearGradient(x,y,x,y+h);g.addColorStop(0,'#fff3c8');g.addColorStop(.5,'#f0c96a');g.addColorStop(1,'#a87a1e');
    ctx.fillStyle=g;ctx.fillRect(x,y,w,h);
    const cx=x+w/2,cy=y+h/2,s=h*0.34*(1+0.14*pulse);
    ctx.fillStyle='rgba(255,255,255,.9)';ctx.beginPath();
    ctx.moveTo(cx,cy-s);ctx.lineTo(cx+s*0.36,cy);ctx.lineTo(cx,cy+s);ctx.lineTo(cx-s*0.36,cy);ctx.closePath();ctx.fill();
  }else if(b.type==='rune'){
    const g=ctx.createLinearGradient(x,y,x,y+h);g.addColorStop(0,'#3f8f7a');g.addColorStop(1,'#1d4a3e');
    ctx.fillStyle=g;ctx.fillRect(x,y,w,h);
    ctx.strokeStyle='rgba(255,217,139,.9)';ctx.lineWidth=Math.max(1.6,h*0.12);ctx.lineJoin='round';
    for(let k=0;k<2;k++){const yy=y+h*0.68-k*h*0.3;
      ctx.beginPath();ctx.moveTo(x+w*0.3,yy);ctx.lineTo(x+w/2,yy-h*0.2);ctx.lineTo(x+w*0.7,yy);ctx.stroke();}
  }
  if(b.type!=='gem'&&b.sparkT>0){ // specular sweep
    const t=1-b.sparkT,off=t*(w+h)-h;
    ctx.save();ctx.beginPath();ctx.rect(x,y,w,h);ctx.clip();ctx.globalCompositeOperation='lighter';
    const sg=ctx.createLinearGradient(x+off-h,y,x+off+h,y+h);
    sg.addColorStop(0,'rgba(255,255,255,0)');sg.addColorStop(.5,'rgba(255,255,255,.35)');sg.addColorStop(1,'rgba(255,255,255,0)');
    ctx.fillStyle=sg;ctx.fillRect(x,y,w,h);ctx.restore();}
  if(b.flash>0){ctx.fillStyle='rgba(255,255,255,'+(b.flash*0.65)+')';ctx.fillRect(x,y,w,h);}
  ctx.strokeStyle='rgba(18,9,7,.6)';ctx.lineWidth=1;ctx.strokeRect(x+0.5,y+0.5,w-1,h-1);
}

  for(const e of embers){const a=Math.min(1,e.life/3)*0.28*(0.6+0.4*Math.sin(animT*2+e.ph));

    ctx.fillText('space / click — release the light',W/2,H-clamp(H*0.07,42,70)+paddle.h+26);ctx.restore();}

  ctx.restore(); // shake
}
