function camDir(dx,dy,dz){const x=dx*basisR[0]+dy*basisR[1]+dz*basisR[2],y=dx*basisU[0]+dy*basisU[1]+dz*basisU[2],z=dx*basisF[0]+dy*basisF[1]+dz*basisF[2];return[x,y,z];}
function closestOnSeg(ox,oy,oz,dx,dy,dz,ax,ay,az,bx,by,bz){
  const abx=bx-ax,aby=by-ay,abz=bz-az;const wox=ax-ox,woy=ay-oy,woz=az-oz;
  const a=abx*abx+aby*aby+abz*abz,b=2*(wox*abx+woy*aby+woz*abz),c=wox*wox+woy*woy+woz*woz;
  let t=(a?(-b+Math.sqrt(Math.max(0,b*b-4*a*c)))/(2*a)):0;t=clamp(t,0,1);
  return{t,x:ax+abx*t,y:ay+aby*t,z:az+abz*t};
}
function rayBox(ox,oy,oz,dx,dy,dz,b){
  let tmin=-Infinity,tmax=Infinity;
  const o=[ox,oy,oz],d=[dx,dy,dz],mn=[b.x-b.w/2,b.y-b.h/2,b.z-b.d/2],mx=[b.x+b.w/2,b.y+b.h/2,b.z+b.d/2];
  for(let i=0;i<3;i++){
    if(Math.abs(d[i])<1e-8){if(o[i]<mn[i]||o[i]>mx[i])return Infinity;continue;}
    let t1=(mn[i]-o[i])/d[i],t2=(mx[i]-o[i])/d[i];if(t1>t2){const q=t1;t1=t2;t2=q;}
    if(t1>tmin)tmin=t1;if(t2<tmax)tmax=t2;if(tmin>tmax)return Infinity;
  }
  return tmax>0?tmax:(tmin>0?tmin:Infinity);
}

    const rdx=c.x-ox,rdy=c.y-oy,rdz=c.z-oz;const dist2=rdx*rdx+rdy*rdy+rdz*rdz;

function updatePlayer(dt){
  const p=player;if(p.dead)return;
  if(keys['ArrowLeft'])yaw+=2.3*dt;if(keys['ArrowRight'])yaw-=2.3*dt;
  if(keys['ArrowUp'])pitch=clamp(pitch+1.6*dt,-1.25,1.25);if(keys['ArrowDown'])pitch=clamp(pitch-1.6*dt,-1.25,1.25);
  let f=(keys['KeyW']||keys['ArrowForward'])?1:0,b=(keys['KeyS'])?1:0,l=(keys['KeyA'])?1:0,r=(keys['KeyD'])?1:0;
  const sy=Math.sin(yaw),cy=Math.cos(yaw);
  let mx=(-sy*f+cy*b)*-1*0; // placeholder clarity below
  // forward on ground = (sinYaw, -cosYaw)
  let vx=Math.sin(yaw)*f-Math.cos(yaw)*b+cy*l+sy*r;
  let vz=-Math.cos(yaw)*f-Math.sin(yaw)*b-sy*l+cy*r;
  const vl=Math.hypot(vx,vz);moveAmt=vl>0?1:0;
  if(vl>0){vx/=vl;vz/=vl;}
  const wantSprint=keys['ShiftLeft']||keys['ShiftRight'];
  p.sprintCd-=dt;
  let sprinting=wantSprint&&p.stam>1&&p.sprintCd<=0&&moveAmt>0;
  if(sprinting){p.stam=Math.max(0,p.stam-24*dt);if(p.stam<=0)p.sprintCd=.6;}
  else p.stam=Math.min(100,p.stam+15*dt);
  const spd=sprinting?5.9:4.3;
  p.x+=vx*spd*dt;p.z+=vz*spd*dt;
  collideCircle(p,0.45);p.x=clamp(p.x,-ARENA+.6,ARENA-.6);p.z=clamp(p.z,-ARENA+.6,ARENA-.6);
  if(moveAmt>0)p.bob+=dt*spd*2.3;
  camX=p.x+ (R()-.5)*shakeMag*.1;camY=HZ+Math.sin(p.bob)*.045*moveAmt+(R()-.5)*shakeMag*.1;camZ=p.z;
  // shooting
  p.cd-=dt;p.bloom=Math.max(0,p.bloom-dt*.06);p.kick=Math.max(0,p.kick-dt*7);
  if(p.reloading>0){p.reloading-=dt;if(p.reloading<=0){p.mag=30;}}

  // low hp heartbeat

function collideCircle(o,r){
  for(const c of crates){
    const cx=clamp(o.x,c.x-c.w/2,c.x+c.w/2),cz=clamp(o.z,c.z-c.d/2,c.z+c.d/2);
    let dx=o.x-cx,dz=o.z-cz;const d2=dx*dx+dz*dz;
    if(d2<r*r){if(d2<1e-6){o.x=c.x+(R()-.5)*.1;}else{const d=Math.sqrt(d2);o.x=cx+dx/d*r;o.z=cz+dz/d*r;}}
  }
}
function updateZombies(dt){

    const z=zombies[i];
    if(z.dead){z.deathT+=dt;if(z.deathT>1.4)zombies.splice(i,1);continue;}
    z.spawnT=Math.min(1,z.spawnT+dt/.7);z.flash=Math.max(0,z.flash-dt*6);
    const tdx=(state==='menu'?z.wx:player.x)-z.x,tdz=(state==='menu'?z.wz:player.z)-z.z;
    const dist=Math.hypot(tdx,tdz)||.001,ux=tdx/dist,uz=tdz/dist;
    if(state==='menu'){z.wt-=dt;if(z.wt<=0||dist<1){z.wx=(R()*2-1)*34;z.wz=(R()*2-1)*34;z.wt=5+R()*8;}}
    const spd=z.spd*(state==='menu'?.6:1);

    if(z.lunge>0){z.lunge-=dt;vx+=ux*6;vz+=uz*6;}
    // separation
    for(const o of zombies){if(o===z||o.dead)continue;const dx=z.x-o.x,dz=z.z-o.z,d2=dx*dx+dz*dz;if(d2<.81&&d2>1e-6){const d=Math.sqrt(d2);vx+=dx/d*1.4;vz+=dz/d*1.4;}}
