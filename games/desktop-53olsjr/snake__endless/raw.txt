function tick(dt,now){
  gNow=now;
  updFlies(dt);
  nextTongue-=dt;
  if(nextTongue<=0){ tongueT=0.26; nextTongue=1.4+Math.random()*3; }
  tongueT=Math.max(0,tongueT-dt);
  headPulse=Math.max(0,headPulse-dt*3);
  shake*=Math.exp(-6*dt); if(shake<0.002)shake=0;
  deathFlash=Math.max(0,deathFlash-dt*1.4);

  for(var i=particles.length-1;i>=0;i--){
    var p=particles[i]; p.age+=dt;
    if(p.age>=p.life){ particles.splice(i,1); continue; }
    p.vy+=260*dt; p.x+=p.vx*dt; p.y+=p.vy*dt;
  }
  for(var j=texts.length-1;j>=0;j--){
    var t=texts[j]; t.age+=dt; if(t.age>=t.life) texts.splice(j,1);
  }

  if(state==='play'){
    acc+=dt*1000;
    while(acc>=interval){
      acc-=interval; step();
      if(state!=='play'){ acc=0; break; }
    }
    tNorm=(state==='play')?Math.min(1,acc/interval):1;
    if(golden&&gNow-golden.born>golden.life){ golden=null; }
  }
}

}

  }else{
    ctx.shadowColor='rgba(255,90,71,.75)';
    var g2=ctx.createRadialGradient(x-r*0.35,y-r*0.4,r*0.15,x,y,r);
    g2.addColorStop(0,'#ffb3a0'); g2.addColorStop(0.45,'#ff6a52'); g2.addColorStop(1,'#d92f22');
    ctx.fillStyle=g2;
  }
  ctx.shadowBlur=cell*0.7;
  ctx.beginPath(); ctx.arc(x,y,r,0,7); ctx.fill();
  ctx.shadowBlur=0;
  ctx.strokeStyle='#7a4b23'; ctx.lineWidth=Math.max(1.5,r*0.12);
  ctx.beginPath(); ctx.moveTo(x,y-r*0.9); ctx.lineTo(x+r*0.18,y-r*1.28); ctx.stroke();
  ctx.fillStyle=useGold?'#ffe9a0':'#59d98c';
  ctx.save(); ctx.translate(x+r*0.42,y-r*1.05); ctx.rotate(-0.6);
  ctx.beginPath(); ctx.ellipse(0,0,r*0.42,r*0.2,0,0,7); ctx.fill(); ctx.restore();
  ctx.fillStyle='rgba(255,255,255,.5)';
  ctx.beginPath(); ctx.arc(x-r*0.3,y-r*0.35,r*0.14,0,7); ctx.fill();
  if(useGold){ // sparkle
    var a2=time*3;
    ctx.strokeStyle='rgba(255,246,216,.9)'; ctx.lineWidth=1.5;
    for(var k=0;k<4;k++){
      var an=a2+k*Math.PI/2, sx=x+Math.cos(an)*r*1.5, sy=y+Math.sin(an)*r*1.5;
      ctx.beginPath(); ctx.moveTo(sx-3,sy); ctx.lineTo(sx+3,sy);
      ctx.moveTo(sx,sy-3); ctx.lineTo(sx,sy+3); ctx.stroke();
    }
  }
  ctx.restore();
}
function drawGoldenFruit(time){
  var gd=golden; if(!gd) return;
  var frac=Math.max(0,1-(gNow-gd.born)/gd.life);
  var x=bx+(gd.x+0.5)*cell, y=by+(gd.y+0.5)*cell;
  ctx.save();
  if(frac<0.25) ctx.globalAlpha=0.45+0.55*Math.abs(Math.sin(time*16));
  drawApple(x,y,time,1,true);
  ctx.globalAlpha*=0.9;
  ctx.strokeStyle='rgba(255,210,94,.9)'; ctx.lineWidth=3;
  ctx.shadowColor='rgba(255,210,94,.7)'; ctx.shadowBlur=8;
  ctx.beginPath(); ctx.arc(x,y,cell*0.62,-Math.PI/2,-Math.PI/2+frac*Math.PI*2); ctx.stroke();
  ctx.restore();
}
function drawChain(pts,alpha,pulse){
  var n=pts.length; if(n<2) return;
  ctx.save(); ctx.lineCap='round'; ctx.globalAlpha=alpha||1;
  for(var i=n-1;i>=1;i--){
    var f=1-i/(n-1);
    ctx.strokeStyle=segColor(f);
    ctx.lineWidth=cell*(0.34+0.4*Math.pow(f,0.85));
    ctx.beginPath(); ctx.moveTo(pts[i].x,pts[i].y); ctx.lineTo(pts[i-1].x,pts[i-1].y); ctx.stroke();
  }
  var h=pts[0];
  var dx=h.x-pts[1].x, dy=h.y-pts[1].y; var L=Math.hypot(dx,dy)||1; dx/=L; dy/=L;
  var px=-dy, py=dx;
  var hr=cell*0.46*(1+0.3*(pulse||0)+0.03*Math.sin(gNow*3));
  // tongue
  if(tongueT>0){
    var tx=h.x+dx*hr, ty=h.y+dy*hr, len=cell*0.32;
    ctx.strokeStyle='#ff6b7a'; ctx.lineWidth=2;
    ctx.beginPath(); ctx.moveTo(tx,ty);
    var ex=tx+dx*len, ey=ty+dy*len; ctx.lineTo(ex,ey);
    ctx.moveTo(ex,ey); ctx.lineTo(ex+dx*cell*0.14+px*cell*0.09, ey+dy*cell*0.14+py*cell*0.09);
    ctx.moveTo(ex,ey); ctx.lineTo(ex+dx*cell*0.14-px*cell*0.09, ey+dy*cell*0.14-py*cell*0.09);
    ctx.stroke();
  }
  // head
  ctx.shadowColor='rgba(150,255,190,.6)'; ctx.shadowBlur=cell*0.7;
  ctx.fillStyle='#dbffe9';
  ctx.beginPath(); ctx.arc(h.x,h.y,hr,0,7); ctx.fill();
  ctx.shadowBlur=0;
  // eyes
  var er=Math.max(2,cell*0.11), pr=Math.max(1.2,cell*0.055);
  for(var s=-1;s<=1;s+=2){
    var exx=h.x+dx*cell*0.14+px*s*cell*0.17, eyy=h.y+dy*cell*0.14+py*s*cell*0.17;
    ctx.fillStyle='#ffffff'; ctx.beginPath(); ctx.arc(exx,eyy,er,0,7); ctx.fill();
    ctx.fillStyle='#0b2418'; ctx.beginPath(); ctx.arc(exx+dx*er*0.35,eyy+dy*er*0.35,pr,0,7); ctx.fill();
  }
  ctx.restore();
}
function drawAttract(time){
  var cx=bx+bw/2, cy=by+bh/2, rx=bw*0.40, ry=bh*0.30, N=46, pts=[];
  for(var i=0;i<=N;i++){
    var a=attractA-i*0.085;
    pts.push({x:cx+Math.cos(a)*rx, y:cy+Math.sin(2*a)*ry});
  }
  drawApple(cx,cy,time,1,false);
  drawChain(pts,0.95,0);
}

/* ================= DRAW ================= */
function draw(time){
  // background
  var g=ctx.createRadialGradient(W*0.5,H*0.32,40,W*0.5,H*0.55,Math.max(W,H)*0.85);
  g.addColorStop(0,'#123c2b'); g.addColorStop(0.5,'#0a251b'); g.addColorStop(1,'#04110c');
  ctx.fillStyle=g; ctx.fillRect(0,0,W,H);

  ctx.save();
  if(shake>0) ctx.translate((Math.random()-0.5)*shake*12,(Math.random()-0.5)*shake*12);

  drawFlies(false,time);
  if(boardLayer) ctx.drawImage(boardLayer,bx,by,bw,bh);
  drawFlies(true,time);

  if(state==='menu'){
    drawAttract(time);
  }else{
    if(food) drawApple(bx+(food.x+0.5)*cell, by+(food.y+0.5)*cell, time, 1, false);
    drawGoldenFruit(time);
    var pts=[];
    for(var i=0;i<snake.length;i++){
      var c=snake[i], p=prevSnake[i]||c;
      var ix=p.x+(c.x-p.x)*tNorm, iy=p.y+(c.y-p.y)*tNorm;
      pts.push({x:bx+(ix+0.5)*cell, y:by+(iy+0.5)*cell});
    }
    drawChain(pts,1,headPulse);
  }

  // particles
  if(particles.length){
    ctx.save(); ctx.globalCompositeOperation='lighter';
    for(var j=0;j<particles.length;j++){
      var p=particles[j], a2=1-p.age/p.life;
      ctx.globalAlpha=a2*0.9; ctx.fillStyle=p.col;
      ctx.beginPath(); ctx.arc(p.x,p.y,p.r*(0.5+a2*0.7),0,7); ctx.fill();
    }
    ctx.restore();
  }
  // floating texts
  for(var k=0;k<texts.length;k++){
    var t=texts[k], f=t.age/t.life;
    var tx=bx+(t.x+0.5)*cell, ty=by+(t.y+0.2)*cell-f*cell*1.4;
    ctx.save();
    ctx.globalAlpha=1-f;
    ctx.font='700 '+Math.max(13,Math.min(22,cell*0.62))+'px '+MONO;
    ctx.textAlign='center'; ctx.textBaseline='middle';
    ctx.shadowColor='rgba(0,0,0,.6)'; ctx.shadowBlur=6;
    ctx.fillStyle=t.col; ctx.fillText(t.txt,tx,ty);
    ctx.restore();
  }

  if(state==='pause'){ ctx.fillStyle='rgba(3,12,8,.45)'; ctx.fillRect(-20,-20,W+40,H+40); }
  if(deathFlash>0){ ctx.fillStyle='rgba(255,80,90,'+(deathFlash*0.18)+')'; ctx.fillRect(-20,-20,W+40,H+40); }
  ctx.restore();
}

/* ================= MAIN LOOP ================= */
function frame(){
  var now=performance.now()/1000;
  var dt=Math.min(0.05,now-lastSec); lastSec=now;
  attractA+=dt*1.1;
  tick(dt,now);
  draw(now);
  requestAnimationFrame(frame);
}

})();
