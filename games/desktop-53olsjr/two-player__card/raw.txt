  if(first){const dn=drawUp(d);for(let k=0;k<dn;k++)sfx.deal(k)}
  phase='defense';
  renderAll();
  const canDef=hands[d].some(x=>beats(x,c));
  if(!canDef){
    msg('No defense — '+PNAME[d]+' must take the cards','bad');
    $('#takeBtn').classList.add('urge');
    timer=setTimeout(()=>takeCards(true),1500);
  }else{
    msg(PNAME[d]+' defends — beat the '+cname(c));
  }
}

function playDefense(i){
  if(busy||phase!=='defense')return;
  const b=bot(),c=hands[b][i];if(!c)return;
  const e=trick[trick.length-1];if(!e)return;
  if(!beats(c,e.atk)){shakeCard(i);sfx.err();return}
  clearTimer();$('#takeBtn').classList.remove('urge');
  hands[b].splice(i,1);sel=Math.min(sel,hands[b].length-1);
  e.def=c;
  addDefDom(c,trick.length-1);sfx.defend();
  if(checkEnd())return;
  phase='attack';
  renderAll();
  const a=bot();
  msg(PNAME[b]+' plays '+cname(c)+' — '+PNAME[a]+', add or finish');
  if(!hands[a].some(canAdd)){
    timer=setTimeout(()=>finishAttack(),1150);
  }
}

function takeCards(forced){
  if(busy||phase!=='defense')return;
  clearTimer();$('#takeBtn').classList.remove('urge');
  busy=true;renderBtns();
  const d=bot();
  const total=trick.reduce((s,e)=>s+1+(e.def?1:0),0);
  sfx.take();
  msg(forced?(PNAME[d]+' takes the '+total+' cards'):(PNAME[d]+' swallows the trick — '+total+' cards'),'bad');
  const cards=[...trickEl.querySelectorAll('.pair .card')];
  cards.forEach((el,i)=>{setTimeout(()=>{el.style.transition='transform .55s ease-in, opacity .55s';
    el.style.transform='translateY(-'+(120+i*14)+'px) scale(.3)';el.style.opacity='0'},i*55)});
  setTimeout(()=>{
    hands[d].push(...trick.flatMap(e=>[e.atk,e.def].filter(Boolean)));
    trick=[];renderTrick();
    busy=false;
    newRound(d);
  },cards.length*55+640);
}

function finishAttack(){
  if(busy||phase!=='attack'||trick.length===0)return;
  clearTimer();busy=true;renderBtns();
  sfx.clear();
  const n=trick.length*2;
  msg('Trick stands — '+n+' cards to the discard');
  const cards=[...trickEl.querySelectorAll('.pair .card')];
  cards.forEach((el,i)=>{setTimeout(()=>{el.style.transition='transform .5s ease-in, opacity .5s';
    el.style.transform='translate('+(90+i*16)+'px,40px) rotate(24deg) scale(.4)';el.style.opacity='0'},i*45)});
  setTimeout(()=>{
    discardN+=n;trick=[];renderTrick();renderStock();
    busy=false;
    newRound(1-attacker);
  },cards.length*45+600);
}

function checkEnd(){
  const a=hands[0].length,b=hands[1].length;
  if(a===0||b===0){endGame((a===0&&b===0)?-1:(a===0?0:1));return true}
  return false;
}

}

**How it plays:** Durak is a hot-seat duel on one screen — the bottom hand always belongs to whoever's turn it is (the other player's cards stay face-down for secrecy). The attacker slams cards; the defender beats with a higher card of the same suit or any trump, or swallows the whole trick. You can stack up to 5 attacks per trick, draw back to 6 each time your turn starts, and whoever empties their hand first wins — with session scores remembered between rematches. Fully playable by mouse (click cards, TAKE / FINISH buttons) or keyboard (`←` `→` + `Enter`, `T`, `F`).
