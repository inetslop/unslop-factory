function tryThrow(aim){if(state!=='playing'||spares<=0||throwCd>0)return;if(balls.length>1)return;
  const cx=P.x+P.w/2,cy=P.y-4;let dx,dy;
  if(aim){dx=aim.x-cx;dy=aim.y-cy;const L=Math.hypot(dx,dy);if(L<10){dx=P.facing*.35;dy=-1;}else{dx/=L;dy/=L;}}
  else{dx=P.facing*.4;dy=-1;const L=Math.hypot(dx,dy);dx/=L;dy/=L;}
  spawnBall(cx,cy,dx*780,dy*780);spares--;throwCd=.25;handTimer=0;SFX.throw();dust(cx,cy-4,3);}
function spawnBall(x,y,vx,vy){if(balls.length>=12)return;balls.push({x,y,vx,vy,r:8,trail:[]});}

function rectHit(a,b){return a.x<b.x+b.w&&a.x+a.w>b.x&&a.y<b.y+b.h&&a.y+a.h>b.y;}
function circResolve(b,q,e){const cx=clamp(b.x,q.x,q.x+q.w),cy=clamp(b.y,q.y,q.y+q.h);
  let dx=b.x-cx,dy=b.y-cy,d=Math.hypot(dx,dy);
  if(d>b.r)return null;
  let nx,ny;if(d<.0001){nx=0;ny=-1;d=0;}else{nx=dx/d;ny=dy/d;}
  b.x=cx+nx*(b.r+.5);b.y=cy+ny*(b.r+.5);
  const dot=b.vx*nx+b.vy*ny;
  if(dot<0){b.vx-=(1+e)*dot*nx;b.vy-=(1+e)*dot*ny;}
  return{nx,ny};}

function updPlayer(dt){const p=P;
  let ax=((keys.ArrowRight||keys.KeyD)?1:0)-((keys.ArrowLeft||keys.KeyA)?1:0);
  if(ax)p.facing=ax;
  p.vx+=ax*2600*dt;p.vx-=p.vx*Math.min(1,(p.onGround?18:5)*dt);p.vx=clamp(p.vx,-340,340);
  jumpBuf-=dt;p.coyote-=dt;
  if(jumpBuf>0&&(p.onGround||p.coyote>0)){p.vy=-650;jumpBuf=0;p.coyote=0;p.onGround=false;dust(p.x+p.w/2,p.y+p.h,6);SFX.jump();p.squash=-.18;}
  p.vy+=1500*dt;if(p.vy>950)p.vy=950;
  const wasAir=!p.onGround,fallV=p.vy;
  p.x+=p.vx*dt;
  for(const q of plats){if(q.one||!rectHit(p,q))continue;if(p.vx>0)p.x=q.x-p.w;else if(p.vx<0)p.x=q.x+q.w;p.vx=0;}
  p.x=clamp(p.x,4,W-4-p.w);
  const prevB=p.y+p.h;p.y+=p.vy*dt;p.onGround=false;
  for(const q of plats){if(!rectHit(p,q))continue;
    if(q.one){if(p.vy>0&&prevB<=q.y+8){p.y=q.y-p.h;p.vy=0;p.onGround=true;}}
    else{const ovX=Math.min(p.x+p.w,q.x+q.w)-Math.max(p.x,q.x),ovY=Math.min(p.y+p.h,q.y+q.h)-Math.max(p.y,q.y);
      if(ovY<ovX){if(p.vy>0&&prevB<=q.y+14){p.y=q.y-p.h;p.vy=0;p.onGround=true;}else if(p.vy<0){p.y=q.y+q.h;p.vy=0;}}
      else{if(p.x+p.w/2<q.x+q.w/2)p.x=q.x-p.w;else p.x=q.x+q.w;}}}
  if(p.onGround){p.coyote=.1;if(wasAir&&fallV>320){dust(p.x+p.w/2,p.y+p.h,8);p.squash=.25;if(fallV>700)SFX.land();}}
  p.squash-=p.squash*9*dt;
  if(p.onGround&&Math.abs(p.vx)>240){runDustT-=dt;if(runDustT<=0){dust(p.x+p.w/2-p.facing*10,p.y+p.h,2);runDustT=.13;}}
  if(p.y>H+60)playerFell();}

function playerFell(){if(invuln>0)return;hurt('LOST TO THE MOLTEN DEPTHS');}
function hurt(msg){lives--;shake=12;SFX.life();combo=0;mult=1;comboT=0;
  if(lives<=0){gameOver();return;}
  spares=3;handTimer=1.4;P.x=LEVELS[levelIdx].spawn[0];P.y=H-80;P.vx=0;P.vy=0;invuln=2;
  addFloater(W/2,H/2-30,msg,'#ff9a86',20);}

function ballLost(){SFX.lost();combo=0;mult=1;comboT=0;shake=Math.max(shake,4);addFloater(clamp(balls.length?balls[0].x:W/2,40,W-40),H-36,'EMBER LOST','#ff8a7a',15);}

function updBalls(dt){const ts=slowT>0?.45:1,g=(gravT>0?-1000:1000);
  for(let i=balls.length-1;i>=0;i--){const b=balls[i],d2=dt*ts;
    const steps=Math.max(1,Math.ceil(Math.hypot(b.vx,b.vy)*d2/13)),sd=d2/steps;
    let lost=false;
    for(let s=0;s<steps&&!lost;s++){
      b.vy+=g*sd;b.x+=b.vx*sd;b.y+=b.vy*sd;
      if(b.x<b.r){b.x=b.r;b.vx=Math.abs(b.vx)*.98;hitTick();}
      if(b.x>W-b.r){b.x=W-b.r;b.vx=-Math.abs(b.vx)*.98;hitTick();}
      if(b.y<b.r){b.y=b.r;b.vy=Math.abs(b.vy)*.98;hitTick();}
      for(const q of plats){const res=circResolve(b,q,.96);if(res){hitTick();sparks(b.x,b.y,'#ffd9a0',1);}}
      const pr=circResolve(b,P,.85);
      if(pr){if(pr.ny<-.7){b.y=P.y-b.r;const off=clamp((b.x-(P.x+P.w/2))/(P.w/2),-1,1);
          b.vy=-Math.max(580,Math.abs(b.vy)*.92);b.vx=off*430+P.vx*.35;P.squash=.15;SFX.head();dust(b.x,P.y,3);}
        else SFX.tick(Math.hypot(b.vx,b.vy));}
      for(const k of blocks){if(k.hp<=0)continue;if(circResolve(b,k,.96)){damageBlock(k);break;}}
    }
    const sp=Math.hypot(b.vx,b.vy);
    if(sp>1250){b.vx*=1250/sp;b.vy*=1250/sp;}else if(sp<230&&sp>1){const f=230/sp;b.vx*=f;b.vy*=f;}
    b.trail.push({x:b.x,y:b.y});if(b.trail.length>9)b.trail.shift();
    if(b.y-b.r>H+50){balls.splice(i,1);ballLost();}}
}
function hitTick(){if(time-lastTick>.06){lastTick=time;SFX.tick(Math.hypot(P.vx,P.vy));}}

function damageBlock(k){if(k.type==='M'){k.jit=.2;sparks(k.x+k.w/2,k.y+k.h/2,'#e6edfb',5);SFX.metal();shake=Math.max(shake,2.5);return;}
  k.hp--;k.flash=1;k.jit=.18;
  if(k.hp>0){if(k.type==='2')SFX.stoneHit();else tone(700,.04,'square',.06);return;}
  destroyBlock(k);}
function destroyBlock(k){const cx=k.x+k.w/2,cy=k.y+k.h/2;
  const base={1:50,2:80,G:60,X:100}[k.type]||50;const pts=base*mult;score+=pts;
  addFloater(cx,cy-6,'+'+pts,'#ffd166',14);
  combo++;comboT=4;const nm=Math.min(8,1+((combo-1)/4|0));if(nm>mult){mult=nm;addFloater(cx,cy-26,'COMBO x'+nm,'#ff7a3c',18);}
  if(k.type==='1')shards(cx,cy,'#d9603c',9);
  else if(k.type==='2'){shards(cx,cy,'#8d97bd',10);SFX.stoneBrk(mult);}
  else if(k.type==='G'){shards(cx,cy,'#3fe8d6',8);gems.push({x:cx,y:cy,vx:R(-60,60),vy:-160,rot:0});}
  else{SFX.brick(mult);}
  sparks(cx,cy,'#ffd166',5);ringP(cx,cy,k.type==='G'?'#3fe8d6':'#ffb347',10);
  if(k.type==='X')booms.push({x:cx,y:cy,r:80,t:.12});
  else if(Math.random()<.13){const r=Math.random();drops.push({x:cx,y:cy,vx:R(-50,50),vy:-130,life:12,type:r<.4?'MULTI':(r<.7?'GRAV':'SLOW')});}
  cleared++;shake=Math.max(shake,1.6);checkClear();}
function explode(x,y,r){shake=10;SFX.boom();ringP(x,y,'#ffb347',20);sparks(x,y,'#ffd166',20);
  for(let i=0;i<8;i++)addP({x:x+R(-r/2,r/2),y:y+R(-r/2,r/2),vx:R(-40,40),vy:R(-90,-20),size:R(5,10),life:R(.6,1.1),col:'#5a4a4a',kind:'dust'});
  for(const k of blocks){if(k.hp<=0||k.type==='M')continue;const dx=k.x+k.w/2-x,dy=k.y+k.h/2-y;
    if(Math.hypot(dx,dy)<r){destroyBlock(k);}}
  const push=(o)=>{const dx=o.x-x,dy=o.y-y,d=Math.hypot(dx,dy)||1;if(d<130){o.vx+=dx/d*380;o.vy+=dy/d*380-60;}};
  gems.forEach(push);drops.forEach(push);
  const pdx=(P.x+P.w/2)-x,pdy=(P.y+P.h/2)-y;if(Math.hypot(pdx,pdy)<150){P.vx+=pdx/150*260;P.vy+=pdy/150*260-80;}}
function checkClear(){if(state!=='playing')return;if(!blocks.some(k=>k.hp>0&&k.type!=='M')){state='clear';stateT=0;SFX.clear();
  for(let i=0;i<44;i++)shards(R(100,W-100),R(60,180),['#ffd166','#3fe8d6','#ff8b2e','#c3cbec'][i%4],1);}}

function updGems(dt){for(let i=gems.length-1;i>=0;i--){const g=gems[i];g.vy+=1300*dt;g.x+=g.vx*dt;g.y+=g.vy*dt;g.rot+=4*dt;
  if(g.x<8){g.x=8;g.vx*=-.7}if(g.x>W-8){g.x=W-8;g.vx*=-.7}
  for(const q of plats)if(!q.one||true){if(g.x>q.x&&g.x<q.x+q.w&&g.y+8>q.y&&g.y<q.y+q.h&&g.vy>0){g.y=q.y-8;g.vy*=-.45;g.vx*=.8;}}
  const dx=(P.x+P.w/2)-g.x,dy=(P.y+P.h/2)-g.y;
  if(Math.hypot(dx,dy)<32){score+=150;gemTotal++;addFloater(g.x,g.y-8,'+150','#7ef0df',14);SFX.gem();sparks(g.x,g.y,'#7ef0df',6);
    if(gemTotal%3===0){spares=Math.min(4,spares+1);addFloater(P.x+P.w/2,P.y-16,'+1 EMBER','#7ef0df',15);}
    gems.splice(i,1);continue;}
  if(g.y>H+40)gems.splice(i,1);}}
function updDrops(dt){for(let i=drops.length-1;i>=0;i--){const d=drops[i];d.vy+=800*dt;d.x+=d.vx*dt;d.y+=d.vy*dt;d.life-=dt;
  if(d.x<12){d.x=12;d.vx*=-.6}if(d.x>W-12){d.x=W-12;d.vx*=-.6}
  for(const q of plats)if(d.x>q.x&&d.x<q.x+q.w&&d.y+11>q.y&&d.y<q.y+q.h&&d.vy>0){d.y=q.y-11;d.vy*=-.5;d.vx*=.8;}
  const dx=(P.x+P.w/2)-d.x,dy=(P.y+P.h/2)-d.y;
  if(Math.hypot(dx,dy)<36){SFX.power();ringP(d.x,d.y,'#fff',14);
    if(d.type==='MULTI'){const src=balls[0];
      if(src){for(const a of[-.5,.5]){const c=Math.cos(a),s=Math.sin(a);spawnBall(src.x,src.y,src.vx*c-src.vy*s,src.vx*s+src.vy*c);}}
      else for(const a of[-.4,0,.4])spawnBall(P.x+P.w/2,P.y-4,Math.sin(a)*780,-Math.cos(a)*780);
      addFloater(d.x,d.y,'MULTI-BALL','#ffb347',16);}
    else if(d.type==='GRAV'){gravT=6;addFloater(d.x,d.y,'ANTI-GRAVITY','#7ef0df',16);}
    else{slowT=5;addFloater(d.x,d.y,'SLOW-MO','#9ad0ff',16);}
    drops.splice(i,1);continue;}
  if(d.life<=0||d.y>H+40)drops.splice(i,1);}}
function updBooms(dt){for(let i=booms.length-1;i>=0;i--){const b=booms[i];b.t-=dt;if(b.t<=0){explode(b.x,b.y,b.r);booms.splice(i,1);}}}

function updParticles(dt){for(let i=particles.length-1;i>=0;i--){const p=particles[i];p.t+=dt;
  if(p.t>p.life){particles.splice(i,1);continue;}p.vy+=p.g*dt;p.x+=p.vx*dt;p.y+=p.vy*dt;p.rot+=p.vr*dt;}}
function updFloaters(dt){for(let i=floaters.length-1;i>=0;i--){const f=floaters[i];f.t+=dt;f.y-=46*dt;if(f.t>f.life)floaters.splice(i,1);}}

function fullUpdate(dt){throwCd-=dt;handTimer-=dt;slowT-=dt;gravT-=dt;invuln-=dt;
  comboT-=dt;if(comboT<=0&&mult>1){combo=0;mult=1;}
  for(const k of blocks){k.jit=Math.max(0,k.jit-dt*3);k.flash=Math.max(0,k.flash-dt*4);}
  updPlayer(dt);
  if(state!=='playing')return;
  if(balls.length===0){if(spares>0){if(handTimer<=0)handTimer=.9;}else hurt('OUT OF EMBERS');}
  if(handTimer<=0&&balls.length===0&&spares>0&&state==='playing'){
    spawnBall(P.x+P.w/2,P.y-4,P.facing*140,-780);spares--;SFX.throw();dust(P.x+P.w/2,P.y-4,3);}
  updBalls(dt);if(state!=='playing')return;
  updGems(dt);updDrops(dt);updBooms(dt);
  pitT-=dt;if(pitT<=0){pitT=.25;for(const p of pits)addP({x:p.x+R(6,p.w-6),y:H-4,vx:R(-12,12),vy:R(-90,-30),size:R(1.5,3),life:R(.8,1.6),col:'#ff8b4a',kind:'spark'});}
  updParticles(dt);updFloaters(dt);
  shake-=shake*7*dt;if(shake<0)shake=0;shx=R(-1,1)*shake;shy=R(-1,1)*shake;}

  draw();}
loadLevel(0);
requestAnimationFrame(frame);
})();
