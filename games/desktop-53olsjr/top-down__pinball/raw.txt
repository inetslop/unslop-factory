function chip(x,y,label){ ctx.font='12px '+MONO; const w=ctx.measureText(label).width+20; rrect(x,y,w,24,7); ctx.fillStyle='#152650'; ctx.fill(); ctx.strokeStyle='rgba(110,190,255,.5)'; ctx.lineWidth=1.5; ctx.stroke(); ctx.fillStyle='#dcecff'; ctx.textAlign='center'; ctx.fillText(label,x+w/2,y+16); return w; }
function drawStar(x,y,r,fill){ ctx.beginPath(); for(let i=0;i<10;i++){ const a=-Math.PI/2+i*Math.PI/5, rr=i%2?r*0.45:r; ctx.lineTo(x+rr*Math.cos(a), y+rr*Math.sin(a)); } ctx.closePath(); ctx.fillStyle=fill; ctx.fill(); }

  for(const s of stars){

/* ============================== MAIN RENDER ============================== */
function render(){
  ctx.setTransform(dpr,0,0,dpr,0,0);
  drawBG();
  const scale=Math.min(cw/(VW+56), ch/(VH+36));
  const ox=(cw-VW*scale)/2, oy=(ch-VH*scale)/2;
  ctx.setTransform(dpr,0,0,dpr,0,0);
  ctx.translate(ox+shakeX*scale, oy-HUD_H*scale+shakeY*scale);
  ctx.scale(scale,scale);
  drawTable();
  drawHUD();
  if(state==='menu') drawMenu();
  else if(state==='over') drawOver();
  else if(state==='lost') drawLost();
  if(paused&&(state==='play'||state==='ready')) drawPaused();
}

/* ============================== INPUT ============================== */
function spaceEdge(){ if(state==='menu'||state==='over') newGame(); }
window.addEventListener('keydown',e=>{
  initAudio();
  const c=e.code;
  if(c==='Space'||c==='ArrowLeft'||c==='ArrowRight'||c==='ArrowUp'||c==='ArrowDown') e.preventDefault();
  if(c==='ArrowLeft'||c==='KeyA'){ kLeft=true; }
  else if(c==='ArrowRight'||c==='KeyD'){ kRight=true; }
  else if(c==='Space'){ if(!e.repeat) spaceEdge(); spaceDown=true; }
  else if((c==='ArrowUp'||c==='KeyW')&&!e.repeat){ nudge(); }
  else if(c==='KeyR'&&!e.repeat){ newGame(); }
  else if(c==='KeyP'&&!e.repeat){ if(state==='play'||state==='ready') paused=!paused; }
});
window.addEventListener('keyup',e=>{
  const c=e.code;
  if(c==='ArrowLeft'||c==='KeyA') kLeft=false;
  else if(c==='ArrowRight'||c==='KeyD') kRight=false;
  else if(c==='Space'){ spaceDown=false; if(state==='ready'&&power>0.02) launch(); }
});
function recalcTouch(){ let L=false,R=false; for(const v of ptrs.values()){ if(v==='L')L=true; else R=true; } tLeft=L; tRight=R; }
cv.addEventListener('pointerdown',e=>{
  initAudio(); try{ cv.setPointerCapture(e.pointerId); }catch(err){}
  if(state==='menu'||state==='over'){ newGame(); return; }
  if(state==='ready'){ touchCharge=true; chargePtr=e.pointerId; ptrs.set(e.pointerId,'C'); return; }

cv.addEventListener('pointermove',e=>{ if(!ptrs.has(e.pointerId))return; const v=ptrs.get(e.pointerId); if(v!=='C'){ ptrs.set(e.pointerId,(e.clientX<cw/2)?'L':'R'); recalcTouch(); } });
function endPtr(e){
  const v=ptrs.get(e.pointerId);
  if(v==='C'){ touchCharge=false; chargePtr=null; if(state==='ready'&&power>0.02) launch(); }
  ptrs.delete(e.pointerId); recalcTouch();
}
cv.addEventListener('pointerup',endPtr);
cv.addEventListener('pointercancel',endPtr);

/* ============================== LOOP ============================== */
let last=performance.now();
function frame(ts){
  let dt=(ts-last)/1000; last=ts; if(dt>0.05)dt=0.05; if(dt<0)dt=0;
  update(dt); render(); requestAnimationFrame(frame);
}
requestAnimationFrame(frame);

Score on pop bumpers (100), slingshots (50) and the drop-target bank (250 each, +1000 for clearing it). Three balls per game; **P** pauses, **R** restarts, and your high score is saved locally.
