/* ============================== WAVES ============================== */
function compFor(w){
  const c = [];
  c.push(['drifter', Math.min(2 + Math.floor((w-1)/2), 5)]);
  if (w>=2) c.push(['bouncer', w<6 ? 1 : 2]);
  if (w>=3) c.push(['shuttle', Math.min(1 + Math.floor((w-3)/3), 3)]);
  if (w>=3) c.push(['spinner', Math.min(1 + Math.floor((w-3)/2), 3)]);
  if (w>=4 && (w-4)%2===0) c.push(['attractor', w>=10 ? 2 : 1]);
  return c;
}
function waveSub(n){
  switch(n){
    case 1: return 'FIRST CONTACT — BULLETS FALL, MOVE OR DROP';
    case 2: return 'RICOCHET FLOOR — MINES WILL BOUNCE';
    case 3: return 'SPIN FIELD + SHUTTLE LINE ENGAGED';
    case 4: return '⚠ GRAVITY WELL DETECTED';
    case 5: return 'WELL PULSE ONLINE — STAY OFF CENTER';
    default: return null;
  }
}
function nextWave(){
  wave++;
  pending.length = 0;
  const comp = compFor(wave);
  let total = 0;
  comp.forEach(([type,n]) => {
    for (let j=0;j<n;j++) pending.push({ t: .7 + Math.floor(j/3)*.5, type, si:j, sn:n });
    total += n;
  });
  const sub = waveSub(wave);
  banner('WAVE ' + String(wave).padStart(2,'0'), sub || ('HOSTILES: ' + total));
  AudioSys.wave();
}

function spawnEnemy(type, si, sn){
  const m = Math.max(70, W*.1);
  let x = sn>1 ? m + (W-2*m)*(si+.5)/sn + rand(-30,30) : rand(m, W-m);
  let y;
  if (type==='drifter')      y = rand(80,140);
  else if (type==='bouncer') y = rand(90,130);
  else if (type==='shuttle'){ y = rand(200,270); x = rand(m,W-m); }
  else if (type==='spinner') y = rand(150,215);
  else { x = W/2 + rand(-60,60) * (sn>1 ? si : 1); y = H*.42; }

  const e = { type, x, y, hx:x, hy:y, t:rand(0,9), ph:rand(0,TAU),
    inT:.7, in:0, cool:rand(1.2,2.4), theta:rand(0,TAU), dir:Math.random()<.5?-1:1,
    aimA:0, burstT:0, restT:rand(.6,1.4), shotT:0, R:240, hitF:0, ax:x, ay:y };
  e.r = type==='attractor'?30 : type==='bouncer'?18 : type==='spinner'?16 : type==='shuttle'?13 : 12;
  const base = {drifter:3, bouncer:5, shuttle:4, spinner:7, attractor:16}[type];
  e.hp = e.maxhp = Math.round(base * (1 + (wave-1)*.06));
  enemies.push(e);
}

/* ============================== BULLETS ============================== */
function spawnEB(x,y,vx,vy,o){
  if (ebullets.length > 430) ebullets[0].life = Math.min(ebullets[0].life, .3);
  const life = o.life || 10;
  ebullets.push({ x,y,vx,vy,r:o.r||5,gF:(o.gF!==undefined?o.gF:1),rest:o.rest||.75,
    type:o.type||'lob',life,maxLife:life,bounces:0,rot:rand(0,TAU),px:x,py:y });
}
function spawnPB(x,y,vx,vy){ pbullets.push({x,y,vx,vy,r:3.5,life:1.7,bounces:0,px:x,py:y}); }

function fireLob(e, tx, ty, flight){
  const gF = 1;
  const vx = (tx - e.x)/flight;
  const vy = (ty - e.y)/flight - .5*(G*GDIR*gF)*flight;
  spawnEB(e.x, e.y+8, vx, vy, { r:5, gF, rest:.75, type:'lob', life:12 });
}

function updateEBullets(dt){
  const Gv = G*GDIR;
  for (let i=ebullets.length-1;i>=0;i--){
    const b = ebullets[i];
    b.px=b.x; b.py=b.y;
    b.vy += Gv*b.gF*dt;
    for (let a=0;a<atts.length;a++){
      const w = atts[a];
      const dx=w.x-b.x, dy=w.y-b.y, d2=dx*dx+dy*dy;
      if (d2 < w.R*w.R && d2 > 144){
        const d = Math.sqrt(d2);
        const f = 900*(1 - d/w.R)*dt;
        b.vx += dx/d*f; b.vy += dy/d*f;
        if (d < w.R*.75){ const s = Math.hypot(b.vx,b.vy); if (s>330){ b.vx*=330/s; b.vy*=330/s; } }
      }
    }
    b.x += b.vx*dt; b.y += b.vy*dt;
    b.rot += dt*(2 + Math.hypot(b.vx,b.vy)*.01);
    if (b.x < b.r){ b.x=b.r; b.vx=-b.vx*b.rest; b.bounces++; wallFx(b); }
    else if (b.x > W-b.r){ b.x=W-b.r; b.vx=-b.vx*b.rest; b.bounces++; wallFx(b); }
    if (b.y < b.r){ b.y=b.r; b.vy=-b.vy*b.rest; b.bounces++; wallFx(b); }
    else if (b.y > H-b.r){
      b.y=H-b.r; b.vy=-b.vy*b.rest; b.bounces++; wallFx(b);
      if (Math.hypot(b.vx,b.vy) < 50) b.life = Math.min(b.life, .7);
    }
    b.life -= dt;
    if (b.life <= 0) ebullets.splice(i,1);
  }
}
function wallFx(b){ addSpark(b.x,b.y,2,'#9fc4e8',110); AudioSys.tick(); }

function updatePBullets(dt){
  const Gv = G*GDIR;
  for (let i=pbullets.length-1;i>=0;i--){
    const p = pbullets[i];
    p.px=p.x; p.py=p.y;
    p.vy += Gv*.5*dt;
    p.x += p.vx*dt; p.y += p.vy*dt;
    if (p.x < p.r){ p.x=p.r; p.vx=-p.vx*.8; p.bounces++; wallFx(p); if(p.bounces>2) p.life=Math.min(p.life,.4); }
    else if (p.x > W-p.r){ p.x=W-p.r; p.vx=-p.vx*.8; p.bounces++; wallFx(p); if(p.bounces>2) p.life=Math.min(p.life,.4); }
    if (p.y < p.r){ p.y=p.r; p.vy=-p.vy*.8; p.bounces++; wallFx(p); if(p.bounces>2) p.life=Math.min(p.life,.4); }
    else if (p.y > H-p.r){ p.y=H-p.r; p.vy=-p.vy*.8; p.bounces++; wallFx(p); if(p.bounces>2) p.life=Math.min(p.life,.4); }
    p.life -= dt;
    let dead = p.life <= 0;
    if (!dead){
      for (let j=enemies.length-1;j>=0;j--){
        const e = enemies[j];
        if (e.in < .9) continue;
        const dx=e.x-p.x, dy=e.y-p.y, rr=e.r+p.r;
        if (dx*dx+dy*dy < rr*rr){
          e.hp--; e.hitF=.12; dead=true;
          addSpark(p.x,p.y,4,'#bff3ff',180); AudioSys.ehit();
          if (e.hp<=0) killEnemy(j);
          break;
        }
      }
    }
    if (dead) pbullets.splice(i,1);
  }
}

/* ============================== ENEMIES ============================== */
function updateEnemy(e, dt){
  if (e.inT > 0) e.inT -= dt;
  e.in = clamp(1 - e.inT/.7, 0, 1);
  e.hitF = Math.max(0, e.hitF - dt);
  const ready = e.inT <= 0;
  e.t += dt;

  switch (e.type){
    case 'drifter': {
      e.x = e.hx + Math.sin(e.t*.8 + e.ph)*70;
      e.y = e.hy + Math.sin(e.t*1.3 + e.ph)*16;
      if (!ready) return;
      e.cool -= dt;
      if (e.cool <= 0){
        const iv = Math.max(1.35, 2.8 - wave*.07);
        e.cool = iv * rand(.85,1.15);
        if (wave>=3 && Math.random()<.3){
          const a = Math.atan2(P.y-e.y, P.x-e.x), sp = Math.min(460, 290 + wave*7);
          spawnEB(e.x, e.y+8, Math.cos(a)*sp, Math.sin(a)*sp, {r:4,gF:0,rest:.7,type:'aim',life:8});
        } else {
          fireLob(e, P.x+rand(-20,20), P.y+rand(-20,20), rand(1.15,1.5));
          if (wave>=6 && Math.random()<.4) fireLob(e, P.x+rand(-70,70), P.y+rand(-30,30), rand(1.4,1.8));
        }
      }
    } break;

    case 'bouncer': {
      e.y = e.hy + Math.sin(e.t*1.6 + e.ph)*10;
      if (!ready) return;
      e.cool -= dt;
      if (e.cool <= 0){
        e.cool = Math.max(2, 3.4 - wave*.06);
        const n = 1 + Math.min(2, Math.floor((wave-2)/3));
        for (let k=0;k<n;k++){
          spawnEB(e.x+rand(-8,8), e.y+12, rand(-1,1)*Math.min(320,120+wave*6), rand(70,180),
                  {r:7,gF:.9,rest:.85,type:'mine',life:9});
        }
        AudioSys.launch();
      }
    } break;

    case 'spinner': {
      e.x += (e.hx-e.x)*Math.min(1,dt*2);
      e.y += (e.hy-e.y)*Math.min(1,dt*2);
      e.theta += dt*(2 + wave*.05);
      if (!ready) return;
      if (e.burstT > 0){
        e.burstT -= dt; e.shotT -= dt;
        if (e.shotT <= 0){
          e.shotT = .16;
          const sp = Math.min(420, 235 + wave*7);
          for (let k=0;k<3;k++){
            const a = e.theta + k*TAU/3;
            spawnEB(e.x, e.y, Math.cos(a)*sp, Math.sin(a)*sp, {r:4,gF:.3,rest:.8,type:'spiral',life:9});
          }
        }
        if (e.burstT <= 0) e.restT = 1.9;
      } else {
        e.restT -= dt;
        if (e.restT <= 0){ e.burstT = 1.5; e.shotT = 0; }
      }
    } break;

    case 'shuttle': {
      const sp = Math.min(520, 300 + wave*8);
      e.x += e.dir*sp*dt;
      if (e.x < 70){ e.x=70; e.dir=1; } else if (e.x > W-70){ e.x=W-70; e.dir=-1; }
      if (!ready) return;
      e.cool -= dt;
      if (e.cool < .45 && e.cool > 0) e.aimA = Math.atan2(P.y-e.y, P.x-e.x);
      if (e.cool <= 0){
        e.cool = Math.max(.9, 1.8 - wave*.05);
        const sp2 = Math.min(620, 430 + wave*7), a = e.aimA;
        spawnEB(e.x+Math.cos(a)*16, e.y+Math.sin(a)*16, Math.cos(a)*sp2, Math.sin(a)*sp2, {r:4.5,gF:0,rest:.7,type:'aim',life:8});
      }
    } break;

    case 'attractor': {
      e.x = e.ax + Math.cos(e.t*.6)*50;
      e.y = e.ay + Math.sin(e.t*.8)*30;
      if (!ready) return;
      e.cool -= dt;
      if (e.cool <= 0){
        e.cool = Math.max(3.2, 4.6 - wave*.05);
        const s0 = Math.min(700, 420 + wave*9);
        for (const b of ebullets){
          const dx=b.x-e.x, dy=b.y-e.y, d=Math.hypot(dx,dy);
          if (d < e.R && d > 1){ b.vx = dx/d*s0; b.vy = dy/d*s0; b.type='burst'; b.life = Math.max(b.life,3); }
        }
        ringsFx.push({x:e.x,y:e.y,r:20,max:e.R,life:.5,t:.5,color:'#ff5c8a',lw:3});
        shake = Math.min(shake+7, 16);
        AudioSys.pulse();
      }
    } break;
  }
}

/* ============================== KILLS / PICKUPS ============================== */
const baseScore = {drifter:100, bouncer:150, shuttle:180, spinner:200, attractor:500};
function mult(){ return Math.min(5, 1 + combo*.1); }

function killEnemy(j){
  const e = enemies[j]; enemies.splice(j,1);
  kills++; combo++;
  const pts = Math.round(baseScore[e.type]*mult()/10)*10;
  score += pts;
  floaters.push({x:e.x, y:e.y-14, txt:'+'+pts, color:'#ffd35c', t:1});
  const col = e.type==='attractor' ? '#ff5c8a' : e.type==='bouncer' ? '#ffb84d' : '#9fc4e8';
  addSpark(e.x,e.y, e.type==='attractor'?36:16, col, 320);
  addSpark(e.x,e.y, 8, '#ffffff', 200);
  ringsFx.push({x:e.x,y:e.y,r:6,max:e.r*3.2,life:.4,t:.4,color:col,lw:2});
  shake = Math.min(shake + (e.type==='attractor'?10:4), 16);
  AudioSys.boom(e.type==='attractor');
  if (Math.random() < .15) dropPickup(e.x, e.y);
}

function dropPickup(x,y){
  let type;
  const r = Math.random();
  if (hulls>=5) type = r<.6 ? 'gem' : (r<.8 ? 'flip' : 'slow');
  else          type = r<.38 ? 'gem' : (r<.62 ? 'hull' : (r<.82 ? 'flip' : 'slow'));
  pickups.push({x, y, vx:rand(-70,70), vy:-90, t:0, life:9, type});
}

function updatePickups(dt){
  const Gv = G*GDIR;
  for (let i=pickups.length-1;i>=0;i--){
    const p = pickups[i];
    p.t += dt; p.life -= dt;
    p.vy += Gv*.7*dt;
    if (!P.dead){
      const dx=P.x-p.x, dy=P.y-p.y, d=Math.hypot(dx,dy);
      if (d < 70 && d > 1){ const f = 260*(1-d/70)*dt; p.vx += dx/d*f; p.vy += dy/d*f; }
      if (d < 20){ applyPickup(p); pickups.splice(i,1); continue; }
    }
    p.x += p.vx*dt; p.y += p.vy*dt;
    p.vx *= Math.exp(-1.4*dt);
    if (p.x<12){p.x=12;p.vx*=-.5;} else if (p.x>W-12){p.x=W-12;p.vx*=-.5;}
    if (p.y<12){p.y=12;p.vy*=-.45;}
    else if (p.y>H-12){ p.y=H-12; p.vy*=-.45; p.vx*=.9; }
    if (p.life<=0) pickups.splice(i,1);
  }
}
function applyPickup(p){
  AudioSys.pickup();
  switch(p.type){
    case 'hull': hulls = Math.min(5, hulls+1); toast('HULL RESTORED','var(--green)'); break;
    case 'flip': flipT = 6; GDIR = -1; toast('GRAVITY REVERSED','var(--amber)'); AudioSys.flip(); break;
    case 'slow': slowT = 4; toast('SLOW FIELD ACTIVE','var(--cyan)'); break;
    case 'gem': { const v = Math.round(250*mult()); score += v; floaters.push({x:p.x,y:p.y-12,txt:'+'+v,color:'#ffd35c',t:1}); } break;
  }
  addSpark(p.x,p.y,10,'#ffffff',180);
}

/* ============================== PLAYER ============================== */
function tryDash(){
  if (state!=='play' || P.dead || P.dashCd>0) return;
  let dx = (keys['d']||keys['arrowright']?1:0) - (keys['a']||keys['arrowleft']?1:0);
  let dy = (keys['s']||keys['arrowdown']?1:0) - (keys['w']||keys['arrowup']?1:0);
  if (dx||dy){ const l=Math.hypot(dx,dy); dx/=l; dy/=l; } else { dx=Math.cos(P.ang); dy=Math.sin(P.ang); }
  P.vx += dx*860; P.vy += dy*860;
  P.dashT = .16; P.dashCd = 2.4; P.inv = Math.max(P.inv, .3);
  ghosts.push({x:P.x, y:P.y, a:P.ang, t:.35});
  AudioSys.dash(); shake = Math.min(shake+3, 16);
}

function updatePlayer(dt){
  P.inv = Math.max(0, P.inv - dt);
  P.dashT -= dt; P.dashCd = Math.max(0, P.dashCd - dt); P.fireCd -= dt;
  let ix = (keys['d']||keys['arrowright']?1:0) - (keys['a']||keys['arrowleft']?1:0);
  let iy = (keys['s']||keys['arrowdown']?1:0) - (keys['w']||keys['arrowup']?1:0);
  if (ix&&iy){ ix*=.7071; iy*=.7071; }
  IX=ix; IY=iy;
  const focus = !!keys['shift'];
  P.vx += ix*1650*dt; P.vy += iy*1650*dt;
  P.vy += G*GDIR*.42*dt;
  const f = Math.exp(-(focus?4.6:1.9)*dt);
  P.vx *= f; P.vy *= f;
  const sp = Math.hypot(P.vx,P.vy), mx = focus?175 : (P.dashT>0?950:480);
  if (sp > mx){ P.vx*=mx/sp; P.vy*=mx/sp; }
  P.x += P.vx*dt; P.y += P.vy*dt;
  if (P.x<14){P.x=14;if(P.vx<0)P.vx*=-.4;} else if (P.x>W-14){P.x=W-14;if(P.vx>0)P.vx*=-.4;}
  if (P.y<16){P.y=16;if(P.vy<0)P.vy*=-.4;} else if (P.y>H-14){P.y=H-14;if(P.vy>0)P.vy*=-.4;}
  let ta = Math.atan2(mouse.y-P.y, mouse.x-P.x);
  let da = ta - P.ang; while (da > Math.PI) da-=TAU; while (da < -Math.PI) da+=TAU;
  P.ang += da*Math.min(1, dt*16);
  if ((mouse.down || keys[' ']) && P.fireCd <= 0){
    const a = P.ang, nx = P.x+Math.cos(a)*16, ny = P.y+Math.sin(a)*16;
    spawnPB(nx, ny, Math.cos(a)*640 + P.vx*.25, Math.sin(a)*640 + P.vy*.25);
    P.vx -= Math.cos(a)*9; P.vy -= Math.sin(a)*9;   // recoil — you feel it
    P.fireCd = 1/8;
    addSpark(nx,ny,2,'#bff3ff',90);
    AudioSys.fire();
  }
  if (P.inv <= 0){
    for (let i=ebullets.length-1;i>=0;i--){
      const b = ebullets[i], dx=b.x-P.x, dy=b.y-P.y, rr=b.r+3.5;
      if (dx*dx+dy*dy < rr*rr){ playerHit(); break; }
    }
  }
}

function playerHit(){
  hulls--; combo = 0; P.inv = 2.4; hitFlash = 1; shake = 14;
  AudioSys.hurt();
  const n = Math.min(ebullets.length, 90);
  for (let i=0;i<n;i++){ const b=ebullets[i]; addSpark(b.x,b.y,2,'#ff8a7a',160); }
  ebullets.length = 0;
  floaters.push({x:P.x, y:P.y-20, txt:'HULL -1', color:'#ff4d5e', t:1.2});
  if (hulls <= 0){
    P.dead = true; deadT = 1.4; shake = 22;
    addSpark(P.x,P.y,60,'#ffd35c',430);
    addSpark(P.x,P.y,30,'#63e6ff',300);
    ringsFx.push({x:P.x,y:P.y,r:8,max:110,life:.7,t:.7,color:'#ffd35c',lw:3});
    AudioSys.boom(true);
  }
}

/* ============================== FX ============================== */
function addSpark(x,y,n,color,sp){
  for (let i=0;i<n;i++){
    if (parts.length > 520) return;
    const a = rand(0,TAU), s = rand(sp*.3, sp);
    parts.push({x,y,vx:Math.cos(a)*s,vy:Math.sin(a)*s,t:rand(.25,.6),color,r:rand(1,2.4)});
  }
}
function updateFX(dt, rdt){
  const Gv = G*GDIR;
  for (let i=parts.length-1;i>=0;i--){
    const p = parts[i];
    p.vy += Gv*.3*dt;
    const f = Math.exp(-2*dt); p.vx*=f; p.vy*=f;
    p.x += p.vx*dt; p.y += p.vy*dt;
    p.t -= dt;
    if (p.t<=0) parts.splice(i,1);
  }
  for (let i=ringsFx.length-1;i>=0;i--){
    const r = ringsFx[i]; r.t -= dt;
    if (r.t<=0) ringsFx.splice(i,1);
  }
  for (let i=floaters.length-1;i>=0;i--){
    const fl = floaters[i]; fl.y -= 32*dt; fl.t -= dt*.9;
    if (fl.t<=0) floaters.splice(i,1);
  }
  for (let i=ghosts.length-1;i>=0;i--){ ghosts[i].t -= rdt; if (ghosts[i].t<=0) ghosts.splice(i,1); }
  shake *= Math.exp(-5*rdt);
  hitFlash *= Math.exp(-4*rdt);
  for (const d of dust){
    d.y += GDIR*13*d.z*dt;
    d.x += Math.sin(tGlobal*.5 + d.ph)*4*dt*d.z;
    if (d.y > H+10) d.y = -10; else if (d.y < -10) d.y = H+10;
  }
}

function updatePlay(dt, rdt){
  GDIR = flipT>0 ? -1 : 1;
  if (flipT>0) flipT = Math.max(0, flipT-dt);
  if (slowT>0) slowT = Math.max(0, slowT-dt);

  for (let i=pending.length-1;i>=0;i--){
    const p = pending[i]; p.t -= dt;
    if (p.t <= 0){ spawnEnemy(p.type, p.si, p.sn); pending.splice(i,1); }
  }
  atts.length = 0;
  for (const e of enemies) if (e.type==='attractor') atts.push(e);
  for (const e of enemies) updateEnemy(e, dt);
  updateEBullets(dt);
  updatePBullets(dt);

  if (!P.dead){
    updatePlayer(dt);
    updatePickups(dt);
    if (!P.dead && enemies.length===0 && pending.length===0){
      if (waveClearT <= 0){
        const b = Math.round(150*wave/10)*10;
        score += b;
        floaters.push({x:W/2, y:H*.4, txt:'FIELD CLEARED  +'+b, color:'#6dff9e', t:1.6});
        dropPickup(W/2, 130);
        toast('FIELD CLEARED','var(--green)');
        waveClearT = 1.9;
      } else {
        waveClearT -= dt;
        if (waveClearT <= 0) nextWave();
      }
    }
  } else {
    updatePickups(dt);
    deadT -= rdt;
    if (deadT <= 0) gameOver();
  }
}

**NEWTON'S HELL** is a bullet-hell shooter where every projectile obeys Newton: enemy lobs are true ballistic arcs (solved for flight time, so they drop), mines ricochet off the walls with restitution, spinners droop under partial gravity, and your own rounds fall too — lead your targets and bank shots off walls (they bounce twice). The ship itself has mass: thrust, drift, recoil, and a constant downward pull you fight.

Highlights:
- **5 enemy archetypes** across escalating waves, including the Gravity Well that *captures* nearby bullets in orbit then flings them outward as a telegraphed ring pulse.
