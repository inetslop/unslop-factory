function spawn(kind){
  var w=G.wave,hpMul=1+(w-1)*.16;
  var e={kind:kind,x:W+80*U,y:rand(H*.12,H*.88),vx:0,vy:0,t:0,rot:rand(0,TAU),fireT:rand(.5,1.4),
    seed:rand(0,TAU),flash:0,p:0,data:{},dead:false};
  switch(kind){
    case 'drone':e.r=13*U;e.hp=3*hpMul;e.vx=-120*U;e.score=100;break;
    case 'weaver':e.r=15*U;e.hp=6*hpMul;e.vx=-150*U;e.data.by=e.y;e.data.amp=rand(H*.1,H*.2);e.score=150;break;
    case 'spinner':e.r=20*U;e.hp=9*hpMul;e.data.tx=rand(W*.42,W*.78);e.data.dur=rand(5,7.5);e.vx=-160*U;e.score=250;break;
    case 'turret':e.r=22*U;e.hp=14*hpMul;e.data.tx=W-95*U;e.vx=-130*U;e.fireT=rand(1.2,2);e.score=300;break;
    case 'kami':{var dx=(player?player.x:W*.2)-e.x,dy=(player?player.y:H/2)-e.y,d=Math.hypot(dx,dy)||1,sp=(430+w*8)*U;
      e.vx=dx/d*sp;e.vy=dy/d*sp;e.r=9*U;e.hp=2*hpMul;e.score=80;break;}
    case 'big':e.x=W+120*U;e.y=H/2;e.r=42*U;e.hp=60+w*10;e.data.tx=W-170*U;e.vx=-90*U;e.score=2500;e.fireT=1.5;break;
    case 'boss':e.x=W+160*U;e.y=H/2;e.r=64*U;e.hp=320+w*45;e.data.tx=W-170*U;e.vx=-80*U;e.score=6000+w*250;e.fireT=1.2;e.p=0;break;
  }
  e.maxhp=e.hp;enemies.push(e);return e;
}
function eShot(x,y,vx,vy,r,col,extra){
  if(bullets.length>=EBMAX)return;
  var b={x:x,y:y,vx:vx,vy:vy,r:r,col:col,t:0,grazed:false};
  if(extra)for(var k in extra)b[k]=extra[k];
  bullets.push(b);
}
function aimed(x,y,sp,n,spread,col,r){
  var base=Math.atan2(player.y-y,player.x-x);
  for(var i=0;i<n;i++){var a=base+(n>1?(i-(n-1)/2)*spread:0);
    eShot(x,y,Math.cos(a)*sp,Math.sin(a)*sp,r||5*U,col);}
}
function ringShot(x,y,n,sp,col,r){
  var t0=rand(0,TAU);
  for(var i=0;i<n;i++){var a=t0+i/n*TAU;eShot(x,y,Math.cos(a)*sp,Math.sin(a)*sp,r||5*U,col);}
}
function wallFrom(x,gapY){
  var gh=Math.max(52*U,H*.075),step=30*U,sp=215*G.spdMul*U;
  for(var y=-step;y<H+step;y+=step){if(Math.abs(y-gapY)<gh)continue;eShot(x,y,-sp,0,5*U,COL.amber);}
}

  G.kills++;

  boom(e.x,e.y,s);

  shake(clamp(2+s*3,2,10));

  else if(e.kind==='big'){dropPickup(e.x-20*U,e.y,'P');dropPickup(e.x+20*U,e.y,'B');}
  else{var r=Math.random();if(r<.07)dropPickup(e.x,e.y,'P');else if(r<.10)dropPickup(e.x,e.y,'B');}
  if(e.kind==='boss')bossDeath(e);
}

function updatePlayer(dt){
  var p=player;if(p.dead)return;
  p.invuln=Math.max(0,p.invuln-dt);
  var sp=(KEY.ShiftLeft||KEY.ShiftRight)?150*U:380*U;
  var ax=0,ay=0;
  if(ptr.on){var dx=ptr.x-p.x,dy=ptr.y-p.y,d=Math.hypot(dx,dy);if(d>6){ax=dx/d;ay=dy/d;}}
  else{
    if(KEY.ArrowLeft||KEY.KeyA)ax-=1;if(KEY.ArrowRight||KEY.KeyD)ax+=1;
    if(KEY.ArrowUp||KEY.KeyW)ay-=1;if(KEY.ArrowDown||KEY.KeyS)ay+=1;
    var m=Math.hypot(ax,ay);if(m>0){ax/=m;ay/=m;}
  }
  p.x=clamp(p.x+ax*sp*dt,14,W-14);p.y=clamp(p.y+ay*sp*dt,14,H-14);
  // engine trail
  p.engT-=dt;if(p.engT<=0){p.engT=.05;
    parts.push({type:'spark',x:p.x-16*U,y:p.y+rand(-3,3)*U,vx:-rand(80,140)*U,vy:rand(-15,15)*U,t:0,life:.25,s:2.2*U,col:'#7fd8ff'});}
  // bomb fx
  if(p.bombing){
    var bf=G.bombFx;bf.t+=dt;
    G.bombR=easeOut(Math.min(1,bf.t/.55))*Math.hypot(W,H)*.75;
    for(var i=enemies.length-1;i>=0;i--){var e=enemies[i],ddx=e.x-bf.x,ddy=e.y-bf.y;
      if(ddx*ddx+ddy*ddy<G.bombR*G.bombR)e.hp-=40*dt*(e.kind==='boss'?.35:1);}
    if(bf.t>=.55){p.bombing=false;G.bombFx=null;}
  }
  // fire
  p.shootT-=dt;
  if(p.shootT<=0){
    p.shootT=.085;var bsp=950*U,x=p.x+16*U,y=p.y;
    var P=[[[0,0]],[[-6,0],[6,0]],[[0,0],[-9,0],[9,0]],
      [[-5,-.03],[5,.03],[-14,.12],[14,-.12]],
      [[0,0],[-7,0],[7,0],[-15,.16],[15,-.16]]][p.power-1];
    for(var k=0;k<P.length;k++){var dy=P[k][0]*U,a=P[k][1];
      shots.push({x:x,y:y+dy,vx:Math.cos(a)*bsp,vy:Math.sin(a)*bsp,r:4*U,dmg:1});}
    sfx.shoot();
  }
}
function easeOut(t){return 1-Math.pow(1-t,3);}

function updEnemy(e,dt){
  e.t+=dt;e.flash=Math.max(0,e.flash-dt*5);
  var sp=G.spdMul,fi=G.fiMul,inPlay=(G.state==='play'&&player&&!player.dead);
  switch(e.kind){
    case 'drone':
      e.x+=e.vx*dt;e.y+=Math.sin(e.t*2.1+e.seed)*55*U*dt;
      e.fireT-=dt;if(inPlay&&e.fireT<=0&&e.x<W-30*U){aimed(e.x-e.r,e.y,200*sp*U,1,.1,COL.pink);e.fireT=rand(1.5,2.1)*fi;}break;
    case 'weaver':
      e.x+=e.vx*dt;e.y=e.data.by+Math.sin(e.t*1.4+e.seed)*e.data.amp;
      e.fireT-=dt;if(inPlay&&e.fireT<=0){aimed(e.x-e.r,e.y,185*sp*U,3,.26,COL.violet);e.fireT=rand(1.7,2.3)*fi;}break;
    case 'spinner':
      if(e.phase===0){e.x+=e.vx*dt;if(e.x<=e.data.tx){e.x=e.data.tx;e.phase=1;e.fireT=.3;}}
      else if(e.phase===1){
        e.rot+=2.6*dt;e.y+=Math.sin(e.t*1.2+e.seed)*20*U*dt;
        e.fireT-=dt;while(e.fireT<=0){e.fireT+=.17;
          for(var k=0;k<3;k++){var a=e.rot+k*TAU/3;
            eShot(e.x,e.y,Math.cos(a)*148*sp*U,Math.sin(a)*148*sp*U,5*U,k%2?COL.violet:COL.pink);}}
        if(e.t>e.data.dur){e.phase=2;e.vx=-90*U;}
      }else e.x+=e.vx*dt;break;
    case 'turret':
      if(e.x>e.data.tx)e.x+=e.vx*dt;else e.y+=Math.sin(e.t*.8+e.seed)*14*U*dt;
      e.fireT-=dt;
      if(inPlay&&e.fireT<=0){
        wallFrom(e.x-e.r,clamp(player.y+rand(-H*.16,H*.16),H*.12,H*.88));
        pendingFx.push({t:.9,fn:(function(tt){return function(){if(!tt.dead&&G.state==='play')aimed(tt.x-tt.r,tt.y,240*G.spdMul*U,1,.05,COL.amber,);};})(e)});
        e.fireT=Math.max(1.6,2.7*fi);
      }break;
    case 'kami':
      e.x+=e.vx*dt;e.y+=e.vy*dt;
      if(Math.random()<.5)parts.push({type:'spark',x:e.x-e.vx*.03,y:e.y-e.vy*.03,vx:0,vy:0,t:0,life:.25,s:2*U,col:COL.red});break;
    case 'big':
      if(e.x>e.data.tx)e.x+=e.vx*dt;
      else{
        e.y=H/2+Math.sin(e.t*.9)*H*.16;
        e.fireT-=dt;if(inPlay&&e.fireT<=0){
          e.phase=(e.phase+1)%2;
          if(e.phase===0)ringShot(e.x-e.r*.4,e.y,14,150*sp*U,COL.violet,6*U);
          else aimed(e.x-e.r*.5,e.y,210*sp*U,7,.5,COL.pink);
          e.fireT=2.1*fi;
          if(enemies.length<9&&Math.random()<.5)spawn('drone');
        }break;}break;
    case 'boss':updBoss(e,dt);break;
  }
  // contact with player
  if(inPlay&&e.x<W){var ddx=e.x-player.x,ddy=e.y-player.y,rrr=e.r+9*U;
    if(ddx*ddx+ddy*ddy<rrr*rrr){hurtPlayer();e.hp-=10;if(e.kind==='kami'||e.hp<=0)killEnemy(e);}}
  if(e.x<-160*U||e.y<-240*U||e.y>H+240*U)e.dead=true;
}

/* ============================== UPDATE ============================== */
function update(dt,rdt){
  G.time+=dt;G.bgT+=rdt;
  if(G.state==='menu'){updMenu(dt);updBullets(dt,true);updParts(dt);return;}
  if(G.state!=='play'&&G.state!=='over')return;
  G.spdMul=clamp(1+(G.wave-1)*.03,1,1.4);G.fiMul=clamp(1-(G.wave-1)*.028,.6,1);
  if(G.slow.t>0)G.slow.t-=rdt;else G.slow.v=lerp(G.slow.v,1,Math.min(1,rdt*4));
  G.shake=Math.max(0,G.shake-rdt*30*U);G.flash=Math.max(0,G.flash-rdt*2.2);G.flashRed=Math.max(0,G.flashRed-rdt*.8);
  if(G.banner){G.banner.t+=rdt;if(G.banner.t>G.banner.dur)G.banner=null;}
  for(var i=pendingFx.length-1;i>=0;i--){pendingFx[i].t-=dt;if(pendingFx[i].t<=0){var fn=pendingFx[i].fn;pendingFx.splice(i,1);fn();}}
  if(G.state==='play'){
    G.waveTime+=dt;
    while(G.timeline.length&&G.timeline[0].t<=G.waveTime)G.timeline.shift().fn();
    updatePlayer(dt);
    // shots vs enemies
    for(var s=shots.length-1;s>=0;s--){var sh=shots[s];sh.x+=sh.vx*dt;sh.y+=sh.vy*dt;
      if(sh.x>W+40||sh.y<-40||sh.y>H+40){shots.splice(s,1);continue;}
      for(var en=enemies.length-1;en>=0;en--){var e2=enemies[en];if(e2.dead)continue;
        var dx=sh.x-e2.x,dy=sh.y-e2.y;if(dx*dx+dy*dy<(e2.r+sh.r)*(e2.r+sh.r)){
          e2.hp-=sh.dmg;e2.flash=.6;sfx.ehit();
          parts.push({type:'spark',x:sh.x,y:sh.y,vx:-80*U,vy:rand(-40,40)*U,t:0,life:.2,s:2.5*U,col:'#bffaff'});
          shots.splice(s,1);if(e2.hp<=0)killEnemy(e2);break;}}}
    // wave clear
    if(!G.bossRef&&G.wave>0){
      if(!G.cleared&&G.timeline.length===0&&enemies.length===0){
        G.cleared=true;G.clearT=1.4;var bonus=250*G.wave;G.score+=bonus;
        banner('SECTOR CLEAR','+'+bonus+' BONUS');sfx.clear();}
      else if(G.cleared){G.clearT-=dt;if(G.clearT<=0)nextWave();}
    }
  }
  for(var e=enemies.length-1;e>=0;e--){if(enemies[e].dead){enemies.splice(e,1);continue;}updEnemy(enemies[e],dt);}
  if(G.bossRef&&G.bossRef.dead)G.bossRef=null;
  updBullets(dt,false);
  // pickups
  for(var pk=pickups.length-1;pk>=0;pk--){var u2=pickups[pk];u2.t+=dt;
    if(G.state==='play'&&!player.dead){var ddx=player.x-u2.x,ddy=player.y-u2.y,d=Math.hypot(ddx,ddy);
      if(d<120*U){u2.vx+=ddx/d*900*U*dt;u2.vy+=ddy/d*900*U*dt;}else{u2.vx*=1-dt*.6;}
      if(d<26*U){collect(u2);pickups.splice(pk,1);continue;}}
    u2.x+=u2.vx*dt;u2.y+=(u2.vy+Math.sin(u2.t*4)*14*U)*dt;
    if(u2.x<-30)pickups.splice(pk,1);}
  updParts(dt);
  for(var tx=texts.length-1;tx>=0;tx--){texts[tx].t+=dt;if(texts[tx].t>=texts[tx].life)texts.splice(tx,1);}
  if(player&&player.dead){G.overT-=rdt;if(G.overT<=0)finalizeOver();}
}

}

Every 5th sector brings Dreadnought KAIROS with three escalating attack phases — all rendered on canvas with pre-baked glow sprites, particle explosions, screen shake, slow-mo death cams, a procedural chiptune loop that intensifies during boss fights, and Web Audio SFX that only starts after your first keypress. Shift for fine move, P to pause, R to restart anytime, M to mute (high score persists).
