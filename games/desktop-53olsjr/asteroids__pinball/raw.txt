function tone(f0,f1,dur,type,vol,delay){
  if(!AC||muted)return; delay=delay||0; const t=AC.currentTime+delay;
  const o=AC.createOscillator(), g=AC.createGain();
  o.type=type; o.frequency.setValueAtTime(Math.max(1,f0),t);
  if(f1) o.frequency.exponentialRampToValueAtTime(Math.max(1,f1),t+dur);
  g.gain.setValueAtTime(vol,t); g.gain.exponentialRampToValueAtTime(.0001,t+dur);
  o.connect(g); g.connect(master); o.start(t); o.stop(t+dur+.03);
}

function noiseHit(dur,vol,freq,type,delay){

  const g=AC.createGain(); g.gain.setValueAtTime(vol,t); g.gain.exponentialRampToValueAtTime(.0001,t+dur);
  s.connect(f); f.connect(g); g.connect(master); s.start(t); s.stop(t+dur+.03);
}

function collideSeg(o,sx,sy,ex,ey,e){
  const dx=ex-sx, dy=ey-sy, L2=dx*dx+dy*dy;
  let t=((o.x-sx)*dx+(o.y-sy)*dy)/L2; t=clamp(t,0,1);
  const cx=sx+dx*t, cy=sy+dy*t, ddx=o.x-cx, ddy=o.y-cy, d2=ddx*ddx+ddy*ddy;
  if(d2>o.r*o.r) return -1;
  let d=Math.sqrt(d2), nx, ny;
  if(d<1e-6){ const L=Math.sqrt(L2); nx=-dy/L; ny=dx/L; } else { nx=ddx/d; ny=ddy/d; }
  o.x=cx+nx*(o.r+.05); o.y=cy+ny*(o.r+.05);
  const vn=o.vx*nx+o.vy*ny;
  if(vn<0){ o.vx-=(1+e)*vn*nx; o.vy-=(1+e)*vn*ny; return -vn; }
  return 0;
}

function stepBall(h){
  if(!ball.active) return;
  if(ball.jit>0) ball.jit-=h;
  if(ballOnPlunger()){ ball.y=plunger.topY-9; ball.vx=0; ball.vy=0; return; }
  ball.vy+=820*h; ball.x+=ball.vx*h; ball.y+=ball.vy*h;

  for(const s of segs){ if(s.gate && !(ball.vy>0)) continue;
    const imp=collideSeg(ball,s.a,s.b,s.c,s.d,.72);
    if(imp>120) sWall(imp);
    if(s.gate&&imp>50) gateFlash=1; }

  for(let i=0;i<bumpers.length;i++){ const b=bumpers[i];
    let dx=ball.x-b.x, dy=ball.y-b.y, d=Math.hypot(dx,dy), min=b.r+ball.r;
    if(d<min){ if(d<1e-4){dx=0;dy=-1;d=1;} const nx=dx/d, ny=dy/d;
      ball.x=b.x+nx*(min+.1); ball.y=b.y+ny*(min+.1);
      const vn=ball.vx*nx+ball.vy*ny; if(vn<0){ ball.vx-=1.5*vn*nx; ball.vy-=1.5*vn*ny; }
      ball.vx+=nx*540; ball.vy+=ny*540;
      b.flash=1; addScore(100,ball.x-ball.r*2,b.y-b.r*2,b.css); sparks(ball.x,ball.y,8,b.css,320);
      shake=Math.max(shake,2.6); sBump(i); } }

  for(const sl of slings){ if(sl.cd>0) continue;
    const cx=sl.pts[0][0]+sl.pts[1][0]+sl.pts[2][0], cy=(sl.pts[0][0]+sl.pts[1][0]+sl.pts[2][0])/3,
          cyy=(sl.pts[0][1]+sl.pts[1][1]+sl.pts[2][1])/3;
    let dx=ball.x-cx, dy=ball.y-cyy, d=Math.hypot(dx,dy), min=52+ball.r;
    if(d<min){ if(d<1e-4){dx=0;dy=-1;d=1;} const nx=dx/d, ny=dy/d;
      ball.x=cx+nx*(min+.1); ball.y=cy+ny*(min+.1);
      const vn=ball.vx*nx+ball.vy*ny; if(vn<0){ ball.vx-=1.5*vn*nx; ball.vy-=1.5*vn*ny; }
      ball.vx+=sl.kick[0]*470; ball.vy+=sl.kick[1]*470;
      sl.cd=.55; sl.flash=1; addScore(25,ball.x,ball.y-14,'#ffd27a',13);
      sparks(ball.x,ball.y,6,'#ffb14f',280); shake=Math.max(shake,1.6); sSling(); } }

  for(const f of flips){
    const tx=f.px+Math.cos(f.a)*f.len, ty=f.py+Math.sin(f.a)*f.len;
    const dx=tx-f.px, dy=ty-f.py, L2=dx*dx+dy*dy;
    let t=((ball.x-f.px)*dx+(ball.y-f.py)*dy)/L2; t=clamp(t,0,1);
    const cx=f.px+dx*t, cy=f.py+dy*t, ddx=ball.x-cx, ddy=ball.y-cy, min=ball.r+f.fr;
    let d=Math.hypot(ddx,ddy);
    if(d<min){ if(d<1e-4){ ball.x=cx; ball.y=cy-min; }
      const nx=d<1e-4?0:ddx/d, ny=d<1e-4?-1:ddy/d;
      ball.x=cx+nx*(min+.05); ball.y=cy+ny*(min+.05);
      const rx=cx-f.px, ry=cy-f.py, svx=-f.om*ry, svy=f.om*rx;
      const rvx=ball.vx-svx, rvy=ball.vy-svy, vn=rvx*nx+rvy*ny;
      if(vn<0){ ball.vx-=1.45*vn*nx; ball.vy-=1.45*vn*ny; } } }

  for(let i=rocks.length-1;i>=0;i--){ const r=rocks[i];
    let dx=ball.x-r.x, dy=ball.y-r.y, d=Math.hypot(dx,dy), min=ball.r+r.r;
    if(d<min&&d>1e-4){ const nx=dx/d, ny=dy/d, mr=r.m;
      ball.x+=nx*(min-d)*(mr/(1+mr)); r.x-=nx*(min-d)*(1/(1+mr));
      const rvx=ball.vx-r.vx, rvy=ball.vy-r.vy, vn=rvx*nx+rvy*ny;
      if(vn<0){ const j=-1.75*vn/(1+1/mr); ball.vx+=j*nx; ball.vy+=j*ny; r.vx-=j/mr*nx; r.vy-=j/mr*ny; }
      shatter(i,nx,ny); } }

  const sp=Math.hypot(ball.vx,ball.vy);
  if(sp>1350){ ball.vx*=1350/sp; ball.vy*=1350/sp; }
  if(sp<20){ ball.still=(ball.still||0)+h; if(ball.still>2.2){ ball.vy-=150; ball.vx+=R(-70,70); ball.still=0; } }
  else ball.still=0;

  if(ball.y>865) onDrain();
}
function shatter(i,nx,ny){
  const r=rocks[i]; rocks.splice(i,1);
  addScore(TIERS[r.tier].pts,r.x,r.y,TIERS[r.tier].rim, 14+r.tier*4);
  sRock(r.tier); shake=Math.max(shake,[6,3.5,2][r.tier]);
  ring(r.x,r.y,r.r*.6,TIERS[r.tier].rim); shards(r.x,r.y,r.r,r.tier);
  r.hitFlash=0;
  if(r.tier<2){ const ct=r.tier+1, cr=TIERS[ct].r;
    for(let k=0;k<2;k++){ const ang=Math.atan2(-ny,-nx)+R(-.75,.75), sp=R(90,170)*sf;
      let cxp=r.x+Math.cos(ang)*cr*.9, cyp=r.y+Math.sin(ang)*cr*.9;
      cxp=clamp(cxp,cr+26,472-cr-8); cyp=clamp(cyp,cr+45,TH-cr-10);
      rocks.push(makeRock(ct,cxp,cyp,{x:Math.cos(ang)*sp+r.vx*.35,y:Math.sin(ang)*sp+r.vy*.35})); } }
  if(rocks.length>12) rocks.shift();
}
function stepRocks(h){
  for(const r of rocks){
    r.x+=r.vx*h; r.y+=r.vy*h; r.rot+=r.spin*h; if(r.hitFlash>0) r.hitFlash-=h*3;
    for(const s of segs){ const imp=collideSeg(r,s.a,s.b,s.c,s.d,.82); if(imp>300) noiseHit(.04,.06,700); }
    for(const b of bumpers){ let dx=r.x-b.x, dy=r.y-b.y, d=Math.hypot(dx,dy), min=b.r+r.r;
      if(d<min&&d>1e-4){ const nx=dx/d, ny=dy/d; r.x=b.x+nx*(min+.1); r.y=b.y+ny*(min+.1);
        const vn=r.vx*nx+r.vy*ny; if(vn<0){ r.vx-=1.8*vn*nx; r.vy-=1.8*vn*ny; b.flash=Math.max(b.flash,.7); noiseHit(.06,.1,500);} } }
    for(const f of flips){
      const tx=f.px+Math.cos(f.a)*f.len, ty=f.py+Math.sin(f.a)*f.len;
      const dx=tx-f.px, dy=ty-f.py, L2=dx*dx+dy*dy;
      let t=((r.x-f.px)*dx+(r.y-f.py)*dy)/L2; t=clamp(t,0,1);
      const cx=f.px+dx*t, cy=f.py+dy*t, ddx=r.x-cx, ddy=r.y-cy, min=r.r+f.fr;
      let d=Math.hypot(ddx,ddy);
      if(d<min&&d>1e-4){ const nx=ddx/d, ny=ddy/d; r.x=cx+nx*(min+.05); r.y=cy+ny*(min+.05);
        const rx=cx-f.px, ry=cy-f.py, svx=-f.om*ry, svy=f.om*rx;
        const rvx=r.vx-svx, rvy=r.vy-svy, vn=rvx*nx+rvy*ny;
        if(vn<0){ r.vx-=1.55*vn*nx; r.vy-=1.55*vn*ny; } } }
    const ms=320*sf; if(r.vx*r.vx+r.vy*r.vy>ms*ms){ const s=ms/Math.hypot(r.vx,r.vy); r.vx*=s; r.vy*=s; }
  }
  for(let i=0;i<rocks.length;i++) for(let j=i+1;j<rocks.length;j++){
    const a=rocks[i], b=rocks[j];
    let dx=b.x-a.x, dy=b.y-a.y, d=Math.hypot(dx,dy), min=a.r+b.r;
    if(d<min&&d>1e-4){ const nx=dx/d, ny=dy/d, ov=min-d;
      a.x-=nx*ov*(b.m/(a.m+b.m)); a.y-=ny*ov*(b.m/(a.m+b.m));
      b.x+=nx*ov*(a.m/(a.m+b.m)); b.y+=ny*ov*(a.m/(a.m+b.m));
      const rvx=b.vx-a.vx, rvy=b.vy-a.vy, vn=rvx*nx+rvy*ny;
      if(vn<0){ const j2=-1.6*vn/(1/a.m+1/b.m); a.vx-=j2/a.m*nx; a.vy-=j2/a.m*ny; b.vx+=j2/b.m*nx; b.vy+=j2/b.m*ny; } } }
}
function physStep(h){
  for(const f of flips){ const tgt=f.on?f.act:f.rest, p=f.a;
    f.a+=(tgt-f.a)*(1-Math.exp(-h*38));
    if(Math.abs(tgt-f.a)<.003){ f.a=tgt; f.om=0; } else f.om=(f.a-p)/h; }
  stepBall(h); stepRocks(h);
}

/* ============================== FX UPDATE ============================== */
function updateFx(dt){
  T+=dt; shake=Math.max(0,shake-dt*9);
  gateOpen=Math.max(0,gateOpen-dt*2.4); gateFlash=Math.max(0,gateFlash-dt*3);
  for(const b of bumpers) b.flash=Math.max(0,b.flash-dt*3);
  for(const s of slings){ s.cd-=dt; s.flash=Math.max(0,s.flash-dt*4); }
  if(plunger.kickT<9) plunger.kickT+=dt;
  if(plunger.charging&&state==='play'){ plunger.chRaw+=dt*1.5;
    plunger.charge=1-Math.abs(1-(plunger.chRaw%2)); }
  else if(!plunger.charging) plunger.charge=Math.max(0,plunger.charge-dt*4);
  plunger.topY = plunger.kickT<.3 ? 786 : (800+(plunger.charging?plunger.charge*14:0));

  if(drainDelay>0){ drainDelay-=dt; if(drainDelay<=0&&state==='play'&&ballsLeft>0) serveBall(); }
  if(state==='play'&&!paused){
    time+=dt; const nl=1+Math.floor(time/40);
    if(nl!==level){ const osf=sf; level=nl; sf=Math.min(2.2,1+(level-1)*.16); targetRocks=Math.min(2+level,7);
      for(const r of rocks){ r.vx*=sf/osf; r.vy*=sf/osf; }
      addFloat(246,430,'LEVEL '+level,'#57e8ff',30); sLevel(); shake=Math.max(shake,2); }
    if(rocks.length<targetRocks){ spawnT+=dt; if(spawnT>1.0){ spawnT=0;
      const p=rockSpawnPos(level>2&&Math.random()<.4?1:0), sp=R(45,95)*sf;
      rocks.push(makeRock(level>2&&Math.random()<.4?1:0,p.x,p.y,{x:R(-sp,sp),y:R(-sp,sp)})); } }
  }
  if(state!=='play'){ ambT-=dt; if(ambT<=0){ ambT=R(1.2,2.4); bumpers[RI(0,3)].flash=Math.max(bumpers[RI(0,3)].flash,.5);} }

  for(let i=parts.length-1;i>=0;i--){ const p=parts[i]; p.life-=dt;
    if(p.life<=0){ parts.splice(i,1); continue; }
    if(p.t==='sp'){ p.x+=p.vx*dt; p.y+=p.vy*dt; p.vx*=1-3*dt; p.vy*=1-3*dt; }
    else if(p.t==='sh'){ p.x+=p.vx*dt; p.y+=p.vy*dt; p.vy+=520*dt; p.rot+=p.vr*dt; }
    else if(p.t==='ri'){ p.r+=p.vr*dt; p.vr*=1-2.4*dt; }
    else { p.x+=p.vx*dt; p.y+=p.vy*dt; } }
  if(parts.length>320) parts.splice(0,parts.length-320);

  for(let i=floats.length-1;i>=0;i--){ const f=floats[i]; f.life-=dt; f.y-=34*dt;
    if(f.life<=0) floats.splice(i,1); }

  shootT-=dt;
  if(shootT<=0&&!shoot){ shoot={x:R(.1,.7)*vw,y:R(.05,.3)*vh,vx:R(380,620),vy:R(140,260),life:.8}; shootT=R(6,13); }
  if(shoot){ shoot.x+=shoot.vx*dt; shoot.y+=shoot.vy*dt; shoot.life-=dt; if(shoot.life<=0) shoot=null; }

  const prevY=ball.y;
  if(ball.active&&ball.x>472&&prevY>=165&&ball.y<165) gateOpen=1;
  if(state==='play'&&ball.active&&!ballOnPlunger()){
    ball.trail.push({x:ball.x,y:ball.y}); if(ball.trail.length>16) ball.trail.shift(); }
}
