    const cell=grid[r][c];
    if(cell&&cell.color>=0&&!cell.clearing){let e=r+1;
      while(e<ROWS&&grid[e][c]&&grid[e][c].color===cell.color&&!grid[e][c].clearing)e++;
      if(e-r>=3){const cells=[];for(let i=r;i<e;i++)cells.push({r:i,c});runs.push({cells,color:cell.color,len:e-r,horiz:false});}
      r=e;}else r++;
  }}
  return runs;
}
function hasMove(){
  for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++){
    for(const d of [[0,1],[1,0]]){
      const r2=r+d[0],c2=c+d[1];if(r2>=ROWS||c2>=COLS)continue;
      const t=grid[r][c];grid[r][c]=grid[r2][c2];grid[r2][c2]=t;
      const found=findRuns().length>0;
      grid[r][c]=grid[r2][c2];grid[r2][c2]=t;
      if(found)return true;
    }
  }
  return false;
}
function reshuffle(){
  for(let tries=0;tries<40;tries++){
    for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++){
      const cell=grid[r][c];if(cell.color<0)continue;
      let col;do{col=irand(0,4);}while((c>=2&&grid[r][c-1].color===col&&grid[r][c-2].color===col)||(r>=2&&grid[r-1][c].color===col&&grid[r-2][c].color===col));
      cell.color=col;cell.s=.35;
    }
    if(findRuns().length===0&&hasMove())break;
  }
  addText(W/2,H*.42,'RECALIBRATING','#3fe0ff',26);S.shake=Math.max(S.shake,7);sfx.recalc();
}
function checkMoves(){if(!hasMove())reshuffle();}

function attemptSwap(r1,c1,r2,c2){
  if(phase!=='idle')return;
  if(Math.abs(r1-r2)+Math.abs(c1-c2)!==1)return;
  phase='swap';swapAnim={a:[r1,c1],b:[r2,c2],t:0,back:false};sfx.swap();
}

function startClear(runs){
  phase='clear';clearT=0;
  const set=new Set();
  for(const run of runs){
    for(const p of run.cells)set.add(p.r*COLS+p.c);
    if(run.len>=5){S.storm={t:0,dmg:14*(1+S.t/300),done:false};sfx.storm();addText(W/2,H*.34,'PRISM STORM','#bff4ff',30);}
    else if(run.len===4){for(const p of run.cells)fireLaser(p.c,run.color);}
    else{for(const p of run.cells)fireBolt(p.c,run.color);}
  }
  let gems=0;
  for(const k of set){const r=(k/COLS)|0,c=k%COLS;const cell=grid[r][c];if(cell){cell.clearing=true;cell.ct=0;gems++;}}
  S.score+=gems*10*chain;
  healShield(gems);
  if(chain>=2)addText(W/2,L.by-56,'CHAIN ×'+chain,chain>3?'#ffc53d':'#3fe0ff',18+Math.min(14,chain*3));
  S.maxChain=Math.max(S.maxChain,chain);
  sfx.match(chain);
}
function applyGravity(){
  for(let c=0;c<COLS;c++){
    let write=ROWS-1;
    for(let r=ROWS-1;r>=0;r--){
      const cell=grid[r][c];
      if(cell){if(write!==r){grid[write][c]=cell;grid[r][c]=null;cell.dy=(r-write)*L.cell;}write--;}
    }
    for(let r=write;r>=0;r--)grid[r][c]={color:irand(0,4),dy:(r-(write+1))*L.cell-6,s:1,clearing:false,ct:0};
  }
}

}

/* ============================== UPDATE ============================== */
function update(dt){
  CLOCK+=dt;
  S.shake*=Math.exp(-6*dt);if(S.shake<.1)S.shake=0;
  S.flash=Math.max(0,S.flash-dt*2.2);
  S.shieldFlash=Math.max(0,S.shieldFlash-dt*3);
  for(let c=0;c<COLS;c++)colFlash[c]=Math.max(0,colFlash[c]-dt*3);
  for(const st of STARS){st.y+=st.z*.02*dt*8;if(st.y>1.02){st.y=-.02;st.x=Math.random();}}
  for(let i=parts.length-1;i>=0;i--){const p=parts[i];p.t+=dt;p.x+=p.vx*dt;p.y+=p.vy*dt;p.vy+=210*dt;if(p.t>p.life)parts.splice(i,1);}
  for(let i=texts.length-1;i>=0;i--){texts[i].t+=dt;texts[i].y-=24*dt;if(texts[i].t>texts[i].life)texts.splice(i,1);}
  // shots always tick (harmless without enemies)
  for(let i=shots.length-1;i>=0;i--){
    const s=shots[i];s.y+=s.vy*dt;
    if(s.y<-40){shots.splice(i,1);continue;}
    let dead=false;
    for(let j=enemies.length-1;j>=0;j--){
      const e=enemies[j],dx=e.x-s.x,dy=e.y-s.y,rr=e.r+s.w;
      if(dx*dx+dy*dy<=rr*rr){
        const crit=e.color===s.color,dmg=s.dmg*(crit?2:1);
        e.hp-=dmg;e.flash=.12;sfx.hit();
        burst(s.x,s.y,PAL[s.color].hex,4,crit?3:1.8);
        if(crit)addText(e.x,e.y-e.r-8,'CRIT ×2','#ffc53d',13);
        if(e.hp<=0)killEnemy(j);
        if(s.pierce>0)s.pierce--;else{dead=true;break;}
      }
    }
    if(dead)shots.splice(i,1);
  }
  if(S.storm){
    S.storm.t+=dt;
    if(!S.storm.done&&S.storm.t>.22){
      S.storm.done=true;S.flash=1;S.shake=Math.max(S.shake,16);sfx.boom();
      for(let j=enemies.length-1;j>=0;j--){const e=enemies[j];e.hp-=S.storm.dmg;e.flash=.2;if(e.hp<=0)killEnemy(j);}
    }
    if(S.storm.t>.9)S.storm=null;
  }

  if(S.mode!=='playing')return;
  S.t+=dt;

  // threat levels
  const lvl=Math.floor(S.t/30);
  if(lvl>S.threatLvl){S.threatLvl=lvl;addText(W/2,H*.3,'THREAT '+lvl,'#ff4d6d',26);sfx.horn();}

  // spawn
  spawnT-=dt;
  if(spawnT<=0&&enemies.length<14){spawnT=lerp(2.3,.75,clamp(S.t/180,0,1))*rand(.8,1.2);spawnEnemy();}

  // enemies
  for(let i=enemies.length-1;i>=0;i--){
    const e=enemies[i];e.t+=dt;e.flash=Math.max(0,e.flash-dt);
    if(e.type==='weaver')e.x=clamp((e.ax||e.x)+Math.sin(e.t*e.wf)*e.amp,20,W-20);
    e.y+=e.vy*dt;
    if(e.type!=='weaver'&&e.y>10&&e.y<L.by-60){
      e.fireT-=dt;
      if(e.fireT<=0){
        const spd=rand(130,175)+S.t*.6;
        if(e.type==='drone'){aimAt(e,spd);e.fireT=rand(2.8,4.2);}
        else{aimSpread(e,spd);e.fireT=rand(3.5,5);}
      }
    }
    if(e.y+e.r>=L.by){
      burst(e.x,L.by-6,PAL[e.color].hex,20,3.5);
      damageShield(e.type==='brute'?26:e.type==='weaver'?14:18);
      S.shake=Math.max(S.shake,e.type==='brute'?14:9);sfx.boom();
      if(e.type==='brute')relics.push({x:e.x,y:L.by-10,t:0});
      enemies.splice(i,1);
    }
  }

  // enemy bullets
  for(let i=ebullets.length-1;i>=0;i--){
    const b=ebullets[i];b.x+=b.vx*dt;b.y+=b.vy*dt;
    if(b.y>L.by-2&&b.x>L.bx-4&&b.x<L.bx+L.bw+4){
      burst(b.x,L.by-4,'#ff8a5c',6,1.6);
      damageShield(b.dmg);ebullets.splice(i,1);continue;
    }
    if(b.y>H+30||b.x<-30||b.x>W+30)ebullets.splice(i,1);
  }

  // relics
  for(let i=relics.length-1;i>=0;i--){
    const r=relics[i];r.t+=dt;r.y+=52*dt;
    if(r.y>=L.by-6){healShield(12);addText(r.x,L.by-14,'+SHIELD','#41e6c8',14);sfx.relic();relics.splice(i,1);}
  }

  // board phases
  if(phase==='swap'){
    swapAnim.t+=dt/.13;
    if(swapAnim.t>=1){
      const[r1,c1]=swapAnim.a,[r2,c2]=swapAnim.b;
      const t=grid[r1][c1];grid[r1][c1]=grid[r2][c2];grid[r2][c2]=t;
      if(swapAnim.back){phase='idle';swapAnim=null;checkMoves();}
      else{
        const runs=findRuns();
        if(runs.length){chain=1;startClear(runs);}
        else{swapAnim={a:[r1,c1],b:[r2,c2],t:0,back:true};sfx.invalid();}
      }
    }
  }else if(phase==='clear'){
    clearT+=dt;
    for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++){const cl=grid[r][c];if(cl&&cl.clearing)cl.ct+=dt;}
    if(clearT>=.18){
      const toKill=new Set();
      for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++){const cl=grid[r][c];if(cl&&cl.clearing)toKill.add(r*COLS+c);}
      for(const k of[...toKill]){
        const r=(k/COLS)|0,c=k%COLS;
        for(const d of[[1,0],[-1,0],[0,1],[0,-1]]){
          const nr=r+d[0],nc=c+d[1];
          if(nr>=0&&nr<ROWS&&nc>=0&&nc<COLS){const n=grid[nr][nc];
            if(n&&n.color<0&&!toKill.has(nr*COLS+nc))toKill.add(nr*COLS+nc);}
        }
      }
      for(const k of toKill){
        const r=(k/COLS)|0,c=k%COLS,cell=grid[r][c];if(!cell)continue;
        const x=L.bx+c*L.cell+L.cell/2,y=L.by+r*L.cell+L.cell/2;
        if(cell.color<0){burst(x,y,'#8f9ab5',12,2.4);S.score+=25;addText(x,y,'+25','#ff4d6d',12);}
        else burst(x,y,PAL[cell.color].hex,7,2);
        grid[r][c]=null;
      }
      applyGravity();phase='fall';
    }
  }else if(phase==='fall'){
    let done=true;
    for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++){
      const cl=grid[r][c];if(!cl)continue;
      if(cl.dy<0){cl.dy*=Math.exp(-dt*14);if(cl.dy>-.6)cl.dy=0;else done=false;}
    }
    if(done){phase='idle';const runs=findRuns();
      if(runs.length){chain++;startClear(runs);}
      else{chain=0;checkMoves();}}
  }

  // cell pop-in (reshuffle / virus)
  for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++){const cl=grid[r][c];if(cl&&cl.s<1)cl.s=Math.min(1,cl.s+dt*6);}

  // corruption
  let viruses=0;
  for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++)if(grid[r][c].color<0)viruses++;
  if(viruses>0)damageShield(dt*Math.min(3,viruses)*.8,false);
  virusT-=dt;
  if(virusT<=0&&viruses<4&&phase==='idle'){
    const open=[];
    for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++)if(grid[r][c].color>=0)open.push([r,c]);
    if(open.length){
      const[r,c]=open[irand(0,open.length-1)];
      grid[r][c].color=-1;grid[r][c].s=.3;
      addText(L.bx+c*L.cell+L.cell/2,L.by+r*L.cell,'CORRUPTED','#ff4d6d',12);
      sfx.virus();S.shake=Math.max(S.shake,5);
    }
    virusT=rand(10,15)-Math.min(4,S.t/90);
  }

  // regen
  if(S.t-S.lastHit>4)healShield(dt*3);
}

function rrect(x,y,w,h,r){ctx.beginPath();ctx.moveTo(x+r,y);ctx.arcTo(x+w,y,x+w,y+h,r);ctx.arcTo(x+w,y+h,x,y+h,r);ctx.arcTo(x,y+h,x,y,r);ctx.arcTo(x,y,x+w,y,r);ctx.closePath();}
function poly(x,y,s,n,rot){for(let i=0;i<n;i++){const a=rot+i*TAU/n;const px=x+Math.cos(a)*s,py=y+Math.sin(a)*s;if(i)ctx.lineTo(px,py);else ctx.moveTo(px,py);}ctx.closePath();}
function gemPath(col,x,y,s){
  if(col===0){ctx.arc(x,y,s,0,TAU);}
  else if(col===1)poly(x,y+s*.12,s*1.15,3,-Math.PI/2);
  else if(col===2)poly(x,y,s,4,-Math.PI/2);
  else if(col===3)poly(x,y,s,5,-Math.PI/2);
  else poly(x,y,s,6,Math.PI/6);
}
function drawGem(col,x,y,s){
  const p=PAL[col];
  ctx.beginPath();gemPath(col,x,y,s*1.45);ctx.fillStyle=p.glow;ctx.fill();
  ctx.beginPath();gemPath(col,x,y,s);ctx.fillStyle=p.fill;ctx.fill();
  ctx.lineWidth=Math.max(1.2,s*.16);ctx.strokeStyle=p.light;ctx.stroke();
  ctx.beginPath();ctx.arc(x-s*.3,y-s*.38,s*.17,0,TAU);ctx.fillStyle='rgba(255,255,255,.8)';ctx.fill();
}
function drawVirus(x,y,s){
  const n=9;ctx.beginPath();
  for(let i=0;i<n*2;i++){const a=i*Math.PI/n+CLOCK*.5,rr=(i%2?s*.5:s);
    const px=x+Math.cos(a)*rr,py=y+Math.sin(a)*rr;if(i)ctx.lineTo(px,py);else ctx.moveTo(px,py);}
  ctx.closePath();
  ctx.fillStyle='#343a4d';ctx.fill();ctx.strokeStyle='#5b6580';ctx.lineWidth=1.5;ctx.stroke();
  const pr=s*.32*(1+.25*Math.sin(CLOCK*7));
  ctx.beginPath();ctx.arc(x,y,pr,0,TAU);
  ctx.fillStyle='rgba(255,77,109,'+(.6+.3*Math.sin(CLOCK*7)).toFixed(2)+')';ctx.fill();
}
