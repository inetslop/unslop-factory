  if(!clouds.length)for(let i=0;i<9;i++)clouds.push({x:Math.random(),y:Math.random(),w:rand(130,320),sp:rand(5,16)});
  G.pal={top:[...ZONES[0].c.top],mid:[...ZONES[0].c.mid],bot:[...ZONES[0].c.bot],r1:[...ZONES[0].c.r1],r2:[...ZONES[0].c.r2],acc:[...ZONES[0].c.acc]};
  generate();
}

/* ============================== generation =========================== */
function generate(){
  while(G.genY>G.camTop-560){
    const tier=clamp(Math.floor((-G.genY/PXM)/35),0,24);
    const gap=rand(88,Math.min(96+tier*5,134));
    const y=G.genY-gap;
    const w=Math.max(62,rand(88,150)-tier*2.2);
    let type='norm';const r=Math.random();
    const pS=Math.max(.10-tier*.008,.04),pC=Math.min(.05+tier*.035,.30),pM=Math.min(.07+tier*.045,.36);
    if(r<pS)type='spring';else if(r<pS+pC)type='crumb';else if(r<pS+pC+pM)type='move';
    const x=G.genPrevX+rand(-175,175);
    const pl=mkPlat(x,y,w,type);plats.push(pl);
    // gems
    if(plats.length>2&&Math.random()<.5){
      if(w>110&&type==='norm'){for(let k=-1;k<=1;k++)gemsArr.push({x:pl.x+k*30,y:pl.y-46,ph:rand(0,TAU),taken:false});}
      else{const mx=clamp((G.genPrevX+pl.x)/2+rand(-20,20),-(SHAFT/2-WALL-14),SHAFT/2-WALL-14);
        gemsArr.push({x:mx,y:(G.genY+y)/2,ph:rand(0,TAU),taken:false});}
    }
    G.genY=y;G.genPrevX=pl.x;
  }
}

function burst(x,y,n,cols,o){o=o||{};
  for(let i=0;i<n;i++){const a=rand(0,TAU),sp=rand((o.spd||260)*.3,o.spd||260);
    parts.push({x,y,vx:Math.cos(a)*sp,vy:Math.sin(a)*sp-(o.up||80),t:0,life:rand(.4,o.life||.9),
      s:rand(2,o.size||5),col:cols[i%cols.length],grav:o.grav==null?700:o.grav,front:o.front!==false,
      rect:!!o.rect,rot:rand(0,TAU),vr:rand(-8,8)});}
  if(parts.length>420)parts.splice(0,parts.length-420);
}

function loseHeart(p){
  if(G.mode!=='playing')return;
  G.hearts--;G.ts=.22;G.shake=13;G.flash=Math.max(G.flash,.75);sFall();
  burst(clamp(p.x,-SHAFT/2+30,SHAFT/2-30),clamp(p.y,G.camTop+VW*.4,G.camTop+VW*.95),30,[PC[p.i].flB,PC[p.i].flA,'#ffffff'],{spd:380,up:160,life:1});
  if(G.hearts<=0){gameOver();return;}

  let best=null;
  for(const pl of plats){if(pl.broken)continue;
    if(pl.y>G.camTop+VW*.32&&pl.y<G.camTop+VW*.75){if(!best||pl.y<best.y)best=pl;}}
  if(!best){best=mkPlat(0,G.camTop+VW*.55,150,'norm');plats.push(best);}

}

function upPlayer(p,dt,m){
  p.coyote-=dt;p.buf-=dt;p.blinkT-=dt;if(p.blinkT<-.12)p.blinkT=rand(1.5,4);
  p.sq+=(0-p.sq)*Math.min(1,dt*9);
  if(p.grounded&&p.standOn&&!p.standOn.broken)p.x+=p.standOn.x-p.standOn.px;
  let ax=0;if(keys.has(m.l))ax--;if(keys.has(m.r))ax++;
  p.vx+=ax*(p.grounded?4300:2700)*dt;
  if(!ax)p.vx*=p.grounded?Math.exp(-11*dt):Math.exp(-1.8*dt);
  p.vx=clamp(p.vx,-MAXVX,MAXVX);
  if(p.buf>0&&p.coyote>0){p.vy=-JUMP;p.buf=0;p.coyote=0;p.grounded=false;p.standOn=null;p.sq=-.3;
    sJump();burst(p.x,p.y+p.h/2,5,[css(mixC(G.pal.acc,[255,255,255],.4))],{spd:110,up:30,life:.4,grav:300});}
  p.vy=Math.min(p.vy+GRAV*dt,MAXFALL);
  const prevBot=p.y+p.h/2;
  p.x+=p.vx*dt;p.y+=p.vy*dt;
  const lim=SHAFT/2-WALL-p.w/2-1;
  if(p.x<-lim){p.x=-lim;if(p.vx<0)p.vx*=-.18;}
  if(p.x> lim){p.x= lim;if(p.vx>0)p.vx*=-.18;}
  if(ax)p.face=ax;else if(Math.abs(p.vx)>40)p.face=Math.sign(p.vx);
  const wasG=p.grounded;p.grounded=false;const keepOn=p.standOn;p.standOn=null;
  let landed=null;
  for(const pl of plats){
    if(pl.broken)continue;
    if(Math.abs(p.x-pl.x)>pl.w/2+p.w/2-4)continue;
    const bot=p.y+p.h/2,top=pl.y;
    if(p.vy>=-60&&prevBot<=top+1&&bot>=top){if(!landed||top<landed.top)landed={pl,top};}
  }
  if(landed){
    const pl=landed.pl;
    if(pl.type==='spring'){
      p.y=landed.top-p.h/2;p.vy=-JUMP*1.82;pl.bounce=1;sSpring();p.sq=-.35;
      burst(p.x,landed.top,10,['#7dffab','#e9ffe f'.replace(' ','')],{spd:240,up:240,life:.5,grav:500});
    }else{
      const imp=p.vy;p.y=landed.top-p.h/2;p.vy=0;p.grounded=true;p.standOn=pl;p.coyote=.09;
      if(!wasG&&imp>430){p.sq=Math.min(.5,imp/1500);sLand(imp);
        burst(p.x,landed.top,6,[css(mixC(G.pal.acc,[255,255,255],.4))],{spd:imp*.25,up:30,life:.4,grav:300});}
      if(pl.type==='crumb')pl.standNow=true;
    }
  }
  // gem pickup
  for(const g of gemsArr){
    if(g.taken)continue;
    const gy=g.y+Math.sin(G.t*3+g.ph)*4,dx=p.x-g.x,dy=p.y-gy;
    if(dx*dx+dy*dy<26*26){g.taken=true;G.gems++;sGem();addFloat(g.x,gy-10,'+1','#ffffff');
      burst(g.x,gy,8,['#ffd9a0','#bff0ff','#ffffff'],{spd:160,up:60,life:.5,grav:0});}
  }
  if(Math.abs(p.vy)>620){p.trail.push({x:p.x,y:p.y,t:.32});if(p.trail.length>12)p.trail.shift();}
  for(const tr of p.trail)tr.t+=dt;
  p.trail=p.trail.filter(tr=>tr.t<.32);
  const a=-p.y;if(a>p.maxAlt)p.maxAlt=a;
}
function updateWorld(dt){
  generate();
  for(const p of plats){p.px=p.x;
    if(p.type==='move')p.x=p.baseX+Math.sin(G.t*p.spd+p.ph)*p.amp;
    p.bounce=Math.max(0,p.bounce-dt*4);p.standNow=false;}
  upPlayer(P[0],dt,K1);upPlayer(P[1],dt,K2);
  // gentle separation so the pair doesn't fully overlap
  const ddx=P[1].x-P[0].x;
  if(Math.abs(ddx)<24&&Math.abs(P[0].y-P[1].y)<30){
    const f=(24-Math.abs(ddx))*6*dt,sg=Math.sign(ddx)||1;
    P[0].x-=sg*f;P[1].x+=sg*f;}
  for(const p of plats)if(p.type==='crumb'&&!p.broken){
    if(p.standNow)p.stand+=dt;else p.stand=Math.max(0,p.stand-dt*.5);
    if(p.stand>=.5)breakPlat(p);}
  // camera (only ever rises)
  const lead=Math.min(P[0].y,P[1].y),tgt=lead-VW*.28;
  if(tgt<G.camTop)G.camTop+=(tgt-G.camTop)*Math.min(1,dt*6.5);
  const alt=-lead;if(alt>G.maxAlt)G.maxAlt=alt;
  const m=Math.floor(G.maxAlt/PXM);
  if(m>=G.nextMile){G.nextMile+=50;G.altPop=1;tone(880,1174,.18,'sine',.12);}
  const zi=zoneIndex(m);
  if(zi>G.zone){G.zone=zi;G.banner={t:2.6,txt:'ZONE '+String(zi+1).padStart(2,'0')+' · '+ZONES[zi].name};sZone();droneSet(zi);}
  // death line
  for(const p of P)if(p.y+p.h/2>G.camTop+VW*1.02)loseHeart(p);
  if(G.mode!=='playing')return; // a fall may have ended the run
  // cull + particles
  plats=plats.filter(p=>!p.broken&&p.y<G.camTop+VW+260);
  gemsArr=gemsArr.filter(g=>!g.taken&&g.y<G.camTop+VW+240);
  for(const q of parts){q.t+=dt;q.x+=q.vx*dt;q.y+=q.vy*dt;q.vy+=q.grav*dt;}
  parts=parts.filter(q=>q.t<q.life);
  for(const f of floats)f.t+=dt;
  floats=floats.filter(f=>f.t<f.life);
}
function triggerBolt(){
  const x0=rand(W/2-(SHAFT/2)*S*.8,W/2+(SHAFT/2)*S*.8);
  let x=x0,y=-10;const pts=[[x,y]];
  for(let i=0;i<7;i++){x+=rand(-46,46);y+=H*.09;pts.push([x,y]);}
  G.bolt={t:.15,pts};G.flash=Math.max(G.flash,.85);thunder();
}
function ambient(dt){
  const zc=ZONES[G.zone].c,k=1-Math.exp(-dt*2.2);
  ['top','mid','bot','r1','r2','acc'].forEach(key=>G.pal[key]=mixC(G.pal[key],zc[key],k));
  G.starsA=lerp(G.starsA,ZONES[G.zone].stars,k);
  G.orbA=lerp(G.orbA,ZONES[G.zone].orbA,k);
  for(const c of clouds)c.x=(c.x+dt*c.sp/W)%1;
  if(G.mode==='playing'||G.mode==='menu'){
    if(Math.random()<dt*6&&motes.length<70)
      motes.push({x:rand(W/2-(SHAFT/2)*S,W/2+(SHAFT/2)*S),y:H+10,vy:rand(24,64),sw:rand(0,TAU),t:0,life:rand(3,6),s:rand(1.4,2.8)});
    for(const q of motes){q.t+=dt;q.y-=q.vy*dt;q.x+=Math.sin(q.t*2+q.sw)*9*dt;}
    motes=motes.filter(q=>q.t<q.life&&q.y>-10);
  }
  if(G.banner){G.banner.t-=dt;if(G.banner.t<=0)G.banner=null;}
  G.introT-=dt;G.hintT-=dt;G.sndToast-=dt;
  G.flash=Math.max(0,G.flash-dt*2.1);
  if(G.bolt)G.bolt.t-=dt;
  G.shake*=Math.exp(-5*dt);
  G.altPop=Math.max(0,G.altPop-dt*2.2);
  if(ZONES[G.zone]===ZONES[2]&&G.mode!=='paused'&&Math.random()<dt/3.5)triggerBolt();
}
