function countMines(cx,cy){let n=0;for(let dy=-1;dy<=1;dy++)for(let dx=-1;dx<=1;dx++){if(!dx&&!dy)continue;if(inB(cx+dx,cy+dy)&&grid[ci(cx+dx,cy+dy)].m)n++;}return n;}
function genLevel(L){
  const c=L.c,r=L.r;
  for(let att=0;att<80;att++){
    const g=new Array(c*r);for(let i=0;i<c*r;i++)g[i]={m:false,rev:false,flag:false,n:0};
    const t={x:1,y:clamp(Math.floor(r/2)+Math.round(rnd(-1,1)),1,r-2)};
    const h={x:c-2,y:clamp(Math.floor(r/2)+Math.round(rnd(-1,1)),1,r-2)};
    const pool=[];
    for(let y=0;y<r;y++)for(let x=0;x<c;x++){
      if(Math.max(Math.abs(x-t.x),Math.abs(y-t.y))<=1)continue;
      if(Math.max(Math.abs(x-h.x),Math.abs(y-h.y))<=1)continue;
      pool.push(y*c+x);
    }
    for(let i=pool.length-1;i>0;i--){const j=Math.random()*(i+1)|0;const tmp=pool[i];pool[i]=pool[j];pool[j]=tmp;}
    const mines=new Set(pool.slice(0,L.mines));
    let ok=true;for(const m of mines)g[m].m=true;
    // solvability: BFS over non-mine cells (8-way) from tee to hole
    const seen=new Uint8Array(c*r),q=[ci(t.x,t.y)];seen[q[0]]=1;let found=ci(h.x,h.y)===q[0];
    while(q.length&&!found){const i=q.pop();const x=i%c,y=(i/c)|0;
      for(let dy=-1;dy<=1;dy++)for(let dx=-1;dx<=1;dx++){if(!dx&&!dy)continue;const nx=x+dx,ny=y+dy;
        if(nx<0||ny<0||nx>=c||ny>=r)continue;const ni=ny*c+nx;if(seen[ni]||g[ni].m)continue;seen[ni]=1;q.push(ni);if(ni===found)found=true;}}
    if(found)return {grid:g,tee:t,hole:h};
  }
  // fallback: clear an L-shaped corridor
  const g=new Array(c*r);for(let i=0;i<c*r;i++)g[i]={m:false,rev:false,flag:false,n:0};
  const t={x:1,y:Math.floor(r/2)},h={x:c-2,y:Math.floor(r/2)};
  let n=0;while(n<L.mines){const x=Math.random()*c|0,y=Math.random()*r|0,i=y*c+x;
    if(g[i].m)continue;if(Math.max(Math.abs(x-t.x),Math.abs(y-t.y))<=1)continue;if(Math.max(Math.abs(x-h.x),Math.abs(y-h.y))<=1)continue;
    g[i].m=true;n++;}
  for(let x=Math.min(t.x,h.x);x<=Math.max(t.x,h.x);x++)g[ci(x,t.y)].m=false;
  return {grid:g,tee:t,hole:h};
}

/* ============================== reveal ============================== */
function cascade(sx,sy){
  const stack=[[sx,sy]];let count=0;
  while(stack.length){
    const p=stack.pop(),cx=p[0],cy=p[1];if(!inB(cx,cy))continue;
    const i=ci(cx,cy),g=grid[i];
    if(g.rev)continue;
    if(g.m){g.rev=true;g.exposed=true;baseCell(i);continue;}
    g.rev=true;g.n=countMines(cx,cy);flashes[i]=1;count++;baseCell(i);
    if(g.n===0){for(let dy=-1;dy<=1;dy++)for(let dx=-1;dx<=1;dx++){if(!dx&&!dy)continue;stack.push([cx+dx,cy+dy]);}}
  }
  if(count>0)S.cascade(count);
}
function reveal3x3(cx,cy){
  for(let dy=-1;dy<=1;dy++)for(let dx=-1;dx<=1;dx++){

    if(g.rev)continue;
    g.rev=true;
    if(g.m){g.exposed=true;}else{g.n=countMines(nx,ny);flashes[i]=1;}

  }
}

/* ============================== physics ============================== */
function decelAt(b){
  const cx=clamp(Math.floor(b.x),0,cur.c-1),cy=clamp(Math.floor(b.y),0,cur.r-1),g=grid[ci(cx,cy)];
  if(!g.rev)return 7.6;

}
function integrate(b,dt){ // returns true on wall bounce
  const a=decelAt(b),sp=Math.hypot(b.vx,b.vy);let hit=false;
  if(sp>0){const ns=Math.max(0,sp-a*dt),k=ns/sp;b.vx*=k;b.vy*=k;if(ns<.12&&a>0){b.vx=0;b.vy=0;}}
  b.x+=b.vx*dt;b.y+=b.vy*dt;
  const c=cur.c,r=cur.r;
  if(b.x<BR){b.x=BR;if(b.vx<0){hit=Math.abs(b.vx)>1;b.vx=-b.vx*.72;}}
  else if(b.x>c-BR){b.x=c-BR;if(b.vx>0){hit=Math.abs(b.vx)>1;b.vx=-b.vx*.72;}}
  if(b.y<BR){b.y=BR;if(b.vy<0){hit=Math.abs(b.vy)>1;b.vy=-b.vy*.72;}}
  else if(b.y>r-BR){b.y=r-BR;if(b.vy>0){hit=Math.abs(b.vy)>1;b.vy=-b.vy*.72;}}
  return hit;
}

function update(dt){
  T+=dt;
  for(let i=0;i<flashes.length;i++)if(flashes[i]>0)flashes[i]=Math.max(0,flashes[i]-dt/0.35);
  for(let i=particles.length-1;i>=0;i--){const p=particles[i];p.life-=dt;
    if(p.life<=0){particles.splice(i,1);continue;}
    p.x+=p.vx*dt;p.y+=p.vy*dt;if(p.grav)p.vy+=p.grav*dt;if(p.vr!=null)p.rot+=p.vr*dt;}
  for(let i=popups.length-1;i>=0;i--){popups[i].t+=dt;if(popups[i].t>popups[i].life)popups.splice(i,1);}
  if(shake>0)shake*=Math.exp(-5*dt);
  if(AU.rollG&&AU.ac){const sp=(state==='roll')?Math.hypot(ball.vx,ball.vy):0;
    AU.rollG.gain.setTargetAtTime(Math.min(.2,sp*.03),AU.ac.currentTime,.08);}

  if(state==='play'){
    if(kL.on)aimAngle-=2.3*dt;
    if(kR.on)aimAngle+=2.3*dt;
    if(kUp.on)powerT=clamp(powerT+1.8*dt,0,1);
    if(kDn.on)powerT=clamp(powerT-1.8*dt,.05,1);
  }
  else if(state==='roll'){
    let hit=false;const sp0=Math.hypot(ball.vx,ball.vy);
    for(let s=0;s<3&&state==='roll';s++){
      const pcx=Math.floor(ball.x),pcy=Math.floor(ball.y);
      if(integrate(ball,dt/3))hit=true;
      const cx=clamp(Math.floor(ball.x),0,cur.c-1),cy=clamp(Math.floor(ball.y),0,cur.r-1);
      if(cx!==pcx||cy!==pcy){onEnterCell(cx,cy);if(state!=='roll')break;}
      const d=Math.hypot(ball.x-holeC.x,ball.y-holeC.y),sp=Math.hypot(ball.vx,ball.vy);
      if(d<.42&&sp<5.8){startSink();break;}
      if(sp<=.13){ball.vx=0;ball.vy=0;
        if(Math.hypot(ball.x-holeC.x,ball.y-holeC.y)<.5)startSink();else state='play';}
    }
    if(state==='roll'){
      puffT-=dt;const sp=Math.hypot(ball.vx,ball.vy);
      if(puffT<=0&&sp>2.5){puffT=.08;
        particles.push({x:ball.x-ball.vx/sp*.3,y:ball.y-ball.vy/sp*.3,vx:0,vy:0,grav:0,life:.4,max:.4,sz:.06,col:'puff',type:'puff'});}
      if(hit&&T-clackT>.12){clackT=T;S.clack();}
    }
  }
  else if(state==='boom'){
    boomT+=dt;
    if(boomT>1.05){
      ball.x=strokeStart.x;ball.y=strokeStart.y;ball.vx=0;ball.vy=0;ballScale=1;
      aimAngle=Math.atan2(holeC.y-ball.y,holeC.x-ball.x);powerT=.4;
      state='play';puff(ball.x,ball.y,8,'puff',1.5,.4,.06);
    }
  }
  else if(state==='sink'){
    sinkT+=dt;
    const k=clamp(sinkT/.45,0,1),e=k*k;
    ball.x=lerp(sinkFrom.x,holeC.x,e);ball.y=lerp(sinkFrom.y,holeC.y,e);
    ballScale=Math.max(.12,1-.88*e);
    if(sinkT>1.35)nextLevel();
  }
}
