function parseLevel(idx){
  const def = LEVELS[idx];
  const rows = def.rows;
  const H = rows.length, W = Math.max(...rows.map(r=>r.length));
  const grid = []; // 0 void,1 floor,2 target,3 wall
  const boxes = new Map();
  let player = null;
  for(let y=0;y<H;y++){
    const row = rows[y];
    grid[y]=[];
    for(let x=0;x<W;x++){
      const c = row[x] ?? ' ';
      if(c==='#') grid[y][x]=3;
      else if(c==='.'||c==='*'||c==='+'){grid[y][x]=2;}
      else {grid[y][x]=(x<row.length && (c===' '||c==='@'||c==='$'))?1:0;}
      if(c==='$'||c==='*') boxes.set(x+','+y,{x,y,rx:x,ry:y,pulse:Math.random()*6});
      if(c==='@'||c==='+') player={x,y,rx:x,ry:y,fx:1,fy:0,squash:0};
    }
  }
  return {def,W,H,grid,boxes,player,name:def.name};
}

function tryMove(d){
 if(state!=='play'||busy)return;
 const p=level.player;p.fx=d[0];p.fy=d[1];
 const nx=p.x+d[0],ny=p.y+d[1];
 const cell=(level.grid[ny]&&level.grid[ny][nx]!==undefined)?level.grid[ny][nx]:0;
 if(cell===3||cell===0){doBump(d);return;}
 const b=level.boxes.get(nx+','+ny);
 if(b){
  const bx=nx+d[0],by=ny+d[1];
  const bc=(level.grid[by]&&level.grid[by][bx]!==undefined)?level.grid[by][bx]:0;
  if(bc===3||bc===0||level.boxes.has(bx+','+by)){doBump(d);return;}
  undoStack.push({px:p.x,py:p.y,b1:[b.x,b.y],b2:[bx,by]});
  const fromT=level.grid[b.y][b.x]===2;
  p.x=nx;p.y=ny;b.x=bx;b.y=by;
  startAnim(p,.13);startAnim(b,.16);
  lastPushInfo={box:b,fromT:fromT};
  moves++;pushes++;busy=true;p.squash=1;
  Sfx.push();puff(nx-d[0]*.5,ny-d[1]*.5,'rgba(201,178,138,.55)',4,.9);
  if(undoStack.length>600)undoStack.shift();
 }else{
  undoStack.push({px:p.x,py:p.y});
  p.x=nx;p.y=ny;startAnim(p,.105);

  Sfx.step();puff(nx-d[0]*.4,ny-d[1]*.4,'rgba(159,182,198,.45)',3,.7);

}
function finalize(){

 }
}
