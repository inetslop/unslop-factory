  if(actx){if(actx.state==='suspended')actx.resume();return;}
  try{
    actx=new (window.AudioContext||window.webkitAudioContext)();
    master=actx.createGain();master.gain.value=G.muted?0:0.85;master.connect(actx.destination);
    noiseBuf=actx.createBuffer(1,actx.sampleRate*0.6,actx.sampleRate);
    const d=noiseBuf.getChannelData(0);for(let i=0;i<d.length;i++)d[i]=Math.random()*2-1;
  }catch(e){}
}
function tone(f0,f1,dur,type,vol,when){if(!actx)return;type=type||'square';vol=vol||.15;when=when||0;
  const t=actx.currentTime+when,o=actx.createOscillator(),g=actx.createGain();
  o.type=type;o.frequency.setValueAtTime(Math.max(1,f0),t);o.frequency.exponentialRampToValueAtTime(Math.max(1,f1),t+dur);
  g.gain.setValueAtTime(vol,t);g.gain.exponentialRampToValueAtTime(.0001,t+dur);
  o.connect(g);g.connect(master);o.start(t);o.stop(t+dur+.03);}
function noise(dur,lp0,lp1,vol,when){if(!actx)return;vol=vol||.3;when=when||0;
  const t=actx.currentTime+when,s=actx.createBufferSource();s.buffer=noiseBuf;
  const f=actx.createBiquadFilter();f.type='lowpass';f.frequency.setValueAtTime(lp0,t);f.frequency.exponentialRampToValueAtTime(Math.max(20,lp1),t+dur);
  const g=actx.createGain();g.gain.setValueAtTime(vol,t);g.gain.exponentialRampToValueAtTime(.0001,t+dur);
  s.connect(f);f.connect(g);g.connect(master);s.start(t);s.stop(t+dur+.03);}
const lastS={};
function sfx(n){const now=performance.now();
  const gap={shoot:45,eshoot:90,boom:70,small:70,hit:40,hurt:120,pick:60,ding:140,rumble:95,chime:0,go:0,tick:0}[n]||0;
  if(now-(lastS[n]||0)<gap)return;lastS[n]=now;
  switch(n){
    case 'shoot':tone(760,240,.08,'square',.09);break;
    case 'eshoot':tone(320,150,.1,'sawtooth',.06);break;
    case 'boom':noise(.4,1400,90,.5);tone(150,40,.35,'sine',.5);break;
    case 'small':noise(.12,2600,500,.22);break;
    case 'hit':tone(520,300,.05,'square',.08);break;
    case 'hurt':noise(.25,900,120,.4);tone(220,55,.25,'sawtooth',.3);break;
    case 'pick':tone(620,620,.07,'square',.12);tone(930,930,.1,'square',.12,.07);break;
    case 'ding':tone(1400,880,.2,'triangle',.2);break;
    case 'chime':tone(523,523,.09,'square',.12);tone(784,784,.16,'square',.12,.09);break;
    case 'go':tone(660,990,.4,'sawtooth',.22);noise(.5,800,3200,.14);break;
    case 'rumble':noise(.09,700,300,.12);break;
    case 'tick':tone(880,880,.06,'square',.14);break;
  }}
function engineStart(){if(!actx||engine)return;
  const o1=actx.createOscillator(),o2=actx.createOscillator(),f=actx.createBiquadFilter(),g=actx.createGain();
  o1.type='sawtooth';o2.type='square';f.type='lowpass';g.gain.value=0;
  o1.connect(f);o2.connect(f);f.connect(g);g.connect(master);o1.start();o2.start();engine={o1,o2,f,g};}
function engineSet(s01,nitro,active){if(!engine)return;const t=actx.currentTime;
  const base=46+s01*95+(nitro?38:0);
  engine.o1.frequency.setTargetAtTime(base,t,.06);
  engine.o2.frequency.setTargetAtTime(base*1.5+3,t,.06);
  engine.f.frequency.setTargetAtTime(300+s01*900+(nitro?700:0),t,.08);
  engine.g.gain.setTargetAtTime(active?(.035+s01*.045):0,t,.12);}

  tileA=mkCanvas(Math.max(4,W),160);{const g=tileA.getContext('2d');

  // asphalt pattern
  const at=mkCanvas(90,90);{const g=at.getContext('2d');g.fillStyle=C.road;g.fillRect(0,0,90,90);

function wave(){
  const opts=[['cones',1],['car',1]];
  if(D>1.12)opts.push(['drones',.85]);
  if(D>1.45)opts.push(['barrier',.7]);
  if(D>1.75)opts.push(['convoy',.9]);
  if(D>2.6)opts.push(['car',1]);
  let tot=0;for(const o of opts)tot+=o[1];let r=Math.random()*tot,kind='cones';
  for(const o of opts){if(r<o[1]){kind=o[0];break;}r-=o[1];}
  const free=freeLanes();
  if(kind==='cones'){const n=3+((Math.random()*3)|0),ls=[0,1,2,3].sort(()=>Math.random()-.5).slice(0,n);
    for(let i=0;i<ls.length;i++)enemies.push(mkCone(laneX(ls[i])+rand(-laneW*.18,laneW*.18),-60-i*74-rand(0,50)));}
  else if(kind==='car'){enemies.push(mkCar(pick(free.length?free:[0,1,2,3])));}
  else if(kind==='convoy'){const ls=[0,1,2,3].sort(()=>Math.random()-.5);
    enemies.push(mkCar(ls[0]));const c=mkCar(ls[1]);c.y=-190;enemies.push(c);}
  else if(kind==='drones'){enemies.push(mkDrone(0));enemies.push(mkDrone(1));}
  else{const gap=(Math.random()*4)|0,gap2=D<2.2?(gap+1+((Math.random()*3)|0))%4:-1;
    for(let l=0;l<4;l++){if(l===gap||l===gap2)continue;enemies.push(mkBlock(l));}}

      if(Math.abs(b.x-e.x)<ew/2+3&&Math.abs(b.y-e.y)<eh/2+6){b.dead=true;damageEnemy(e,1,b.x,b.y);break;}}}

    if(px<e.x+ew/2&&px+pw>e.x-ew/2&&py<e.y+eh/2&&py+ph>e.y-eh/2){
      damagePlayer(e.type==='car'?34:e.type==='drone'?18:e.type==='block'?26:12);
      if(e.hp>0)damageEnemy(e,99,e.x,e.y-Y(0));break;}}

    if(Math.abs(b.x-P.x)<P.w/2+b.r&&Math.abs(b.y-P.y)<P.h/2+b.r){b.dead=true;damagePlayer(10);}}

function updatePowerups(dt){
  for(const pu of powerups){pu.t+=dt;pu.life-=dt;pu.y+=(60+G.speed*.5)*dt;
    if(Math.abs(pu.x-P.x)<P.w/2+16&&Math.abs(pu.y-P.y)<P.h/2+16){pu.life=0;sfx('pick');
      if(pu.type==='repair'){P.hp=Math.min(100,P.hp+30);addFloat(P.x,P.y-40,'+30 HULL',C.green,15);}
      else if(pu.type==='rapid'){P.rapid=9;addFloat(P.x,P.y-40,'RAPID FIRE',C.cyan,14);}
      else if(pu.type==='spread'){if(P.spreadLvl<2){P.spreadLvl++;addFloat(P.x,P.y-40,'SPREAD ×'+(P.spreadLvl*2+1),C.yellow,14);}else{P.spreadT=9;addFloat(P.x,P.y-40,'SPREAD +9s',C.yellow,13);}}
      else{P.shield=6;addFloat(P.x,P.y-40,'SHIELD 6s','#bfffff',15);}
      P.spreadT=Math.max(P.spreadT,pu.type==='spread'?9:P.spreadT);}}
  powerups=powerups.filter(pu=>pu.life>0&&pu.y<H+40);
}

function update(dt){
  if(G.state==='paused'){engineSet(0,0,false);return;}
  G.dmg=Math.max(0,G.dmg-dt*2.2);dmgEl.style.opacity=G.dmg>0?Math.min(.85,G.dmg):0;
  G.shake=Math.max(0,G.shake*(1-6*dt)-2*dt);

  if(G.state==='menu'){G.scroll+=300*dt;P.x+=(W/2-P.x)*dt;spawnExhaust(dt*0.7);updateDecos(dt,300);updateParts(dt);engineSet(0,0,false);return;}
  if(G.state==='lights'){G.lightsT+=dt;
    const n=Math.min(5,Math.floor(G.lightsT/.42));
    if(n!==litShown){litShown=n;lampEls[n-1].classList.add('on');sfx('tick');}
    engineSet(.16+n*.08+Math.sin(G.lightsT*30)*.02,false,true);
    if(G.lightsT>=2.35){lampEls.forEach(l=>l.classList.remove('on'));lightsEl.style.display='none';
      G.state='playing';banner('GO!');sfx('go');G.speed=140;}
    return;}
  if(G.state==='dying'){G.dieT-=dt;G.speed=Math.max(0,G.speed-700*dt);G.scroll+=G.speed*dt;
    updateParts(dt);updateFloats(dt);
    for(const b of bullets){b.x+=b.vx*dt;b.y+=b.vy*dt;}for(const b of ebullets){b.x+=b.vx*dt;b.y+=b.vy*dt;}
    if(G.dieT<=0)showOver();return;}
  if(G.state!=='playing')return;

  G.t+=dt;D=Math.min(4,1+G.t/50);

  /* nitro + speed */
  P.nitroOn=(K.nitro||nt)&&P.nitro>0.5;
  if(P.nitroOn){P.nitro=Math.max(0,P.nitro-32*dt);if(P.nitro<=0)P.nitroOn=false;}
  else P.nitro=Math.min(100,P.nitro+11*dt);
  let target=K.u?760:(K.d?300:480+D*25);if(P.nitroOn)target=980;
  const rate=target>G.speed?340:520;
  G.speed+=clamp(target-G.speed,-rate*dt,rate*dt);
  if(P.rubbing)G.speed=Math.max(260,G.speed-420*dt);
  G.scroll+=G.speed*dt;
  const kmh=Math.round(G.speed*.32);if(kmh>G.topKmh)G.topKmh=kmh;

  /* sector */
  const sec=Math.floor(G.scroll/SECTOR)+1;
  if(sec!==G.sector){G.sector=sec;banner('SECTOR '+String(sec).padStart(2,'0'));sfx('chime');
    const g=Math.round(400*G.mult);G.score+=g;addFloat(W/2,H*.3,'+'+g,C.yellow,24);}

  /* steering */
  const mx=560*(P.nitroOn?.9:1);
  let ix=(K.r-K.l)+(touchX!==null?clamp((touchX-P.x)/60,-1,1):0);
  P.vx+=clamp(ix*mx-P.vx,-6000*dt,6000*dt);
  P.x+=P.vx*dt;
  const minX=roadX+P.w/2-8,maxX=roadX+roadW-P.w/2+8;let rub=false;
  if(P.x<minX){P.x=minX;P.vx=Math.max(0,P.vx);rub=true;}
  if(P.x>maxX){P.x=maxX;P.vx=Math.min(0,P.vx);rub=true;}
  P.rubbing=rub;P.brake=!!K.d;
  if(rub){G.shake=Math.max(G.shake,3.2);sfx('rumble');
    addPart({type:'spark',x:P.x+(P.vx>0?1:-1)*P.w*.5,y:P.y+rand(-10,10),vx:rand(-60,60),vy:rand(0,180)+G.speed*.4,life:.2,max:.2,size:3,col:C.yellow});}

  /* timers + fire */
  P.inv=Math.max(0,P.inv-dt);P.fireCd-=dt;P.rapid=Math.max(0,P.rapid-dt);P.shield=Math.max(0,P.shield-dt);
  if(P.spreadT>0){P.spreadT-=dt;if(P.spreadT<=0)P.spreadLvl=0;}
  if((K.fire||touchFire)&&P.fireCd<=0)fire();
  spawnExhaust(dt);

  /* director + entities */
  G.spawnT-=dt;
  if(G.spawnT<=0){wave();G.spawnT=lerp(1.15,.5,D/3)*rand(.8,1.2);}
  updateDecos(dt,G.speed);
  updateEnemies(dt);
  for(const b of bullets){b.x+=b.vx*dt;b.y+=b.vy*dt;}
  bullets=bullets.filter(b=>!b.dead&&b.y>-30&&b.x>-30&&b.x<W+30);
  for(const b of ebullets){b.x+=b.vx*dt;b.y+=b.vy*dt;}
  ebullets=ebullets.filter(b=>b.y<H+40&&b.y<-60?false:b.y<H+40&&b.x>-40&&b.x<W+40);
  if(P.alive)collide();
  updatePowerups(dt);updateParts(dt);updateFloats(dt);

  G.score+=G.speed*dt*.02*G.mult;
  engineSet(clamp(G.speed/980,0,1),P.nitroOn,true);
  syncHud();
}

function drawTile(tile,off,th){if(!tile)return;for(let y=off-th;y<H;y+=th)ctx.drawImage(tile,0,y);}
function rr(x,y,w,h,r){ctx.beginPath();ctx.moveTo(x+r,y);ctx.arcTo(x+w,y,x+w,y+h,r);ctx.arcTo(x+w,y+h,x,y+h,r);ctx.arcTo(x,y+h,x,y,r);ctx.arcTo(x,y,x+w,y,r);ctx.closePath();}

function drawDeco(dc){
  if(dc.type==='rock'){ctx.save();ctx.translate(dc.x,dc.y);
    ctx.fillStyle='#1c2a35';ctx.strokeStyle='#0a1118';ctx.lineWidth=2;ctx.beginPath();
    dc.pts.forEach((p,i)=>i?ctx.lineTo(p.dx,p.dy):ctx.moveTo(p.dx,p.dy));ctx.closePath();ctx.fill();ctx.stroke();ctx.restore();}
  else if(dc.type==='tire'){ctx.save();ctx.translate(dc.x,dc.y);ctx.rotate(dc.rot);
    ctx.fillStyle='#0d1319';ctx.beginPath();ctx.arc(0,0,dc.r,0,7);ctx.fill();
    ctx.strokeStyle='#2b3a48';ctx.lineWidth=3;ctx.beginPath();ctx.arc(0,0,dc.r*.62,0,7);ctx.stroke();ctx.restore();}
  else if(dc.type==='sign'){ctx.save();ctx.translate(dc.x,dc.y);
    ctx.fillStyle='#152230';ctx.fillRect(-34,-15,68,30);
    ctx.strokeStyle=dc.col;ctx.lineWidth=2;ctx.strokeRect(-34,-15,68,30);
    ctx.fillStyle=dc.col;ctx.font='13px '+DISP;ctx.textAlign='center';ctx.textBaseline='middle';
    ctx.globalAlpha=.9;ctx.fillText(dc.word,0,1);ctx.restore();}
  else if(dc.type==='pole'){ctx.save();ctx.globalCompositeOperation='lighter';
    const g=ctx.createRadialGradient(dc.x,dc.y,0,dc.x,dc.y,16);
    g.addColorStop(0,'rgba(160,235,255,.5)');g.addColorStop(1,'rgba(160,235,255,0)');
    ctx.fillStyle=g;ctx.beginPath();ctx.arc(dc.x,dc.y,16,0,7);ctx.fill();
    ctx.fillStyle='#eaffff';ctx.beginPath();ctx.arc(dc.x,dc.y,3,0,7);ctx.fill();ctx.restore();}
}
function carBody(w,h,body,dark,stripe){
  const x=-w/2,y=-h/2;
  // wheels
  ctx.fillStyle='#0a0f14';
  [[x-3,y+h*.16],[x+w-2,y+h*.16],[x-3,y+h*.62],[x+w-2,y+h*.62]].forEach(p=>ctx.fillRect(p[0],p[1],w*.16,h*.18));
  // spoiler
  ctx.fillStyle=dark;ctx.fillRect(-w*.56,y+h*.9,w*1.12,h*.07);
  // body
  ctx.beginPath();
  ctx.moveTo(x+w*.16,y);ctx.lineTo(x+w*.84,y);
  ctx.quadraticCurveTo(x+w,y+h*.1,x+w,y+h*.22);
  ctx.lineTo(x+w,y+h*.92);ctx.quadraticCurveTo(x+w,y+h,x+w*.88,y+h);
  ctx.lineTo(x+w*.12,y+h);ctx.quadraticCurveTo(x,y+h,x,y+h*.92);
  ctx.lineTo(x,y+h*.22);ctx.quadraticCurveTo(x,y+h*.1,x+w*.16,y);
  ctx.closePath();
  ctx.fillStyle=body;ctx.fill();ctx.strokeStyle=dark;ctx.lineWidth=3;ctx.stroke();
  // canopy + windshield
  ctx.fillStyle='#0e1a24';rr(-w*.3,y+h*.2,w*.6,h*.34,5);ctx.fill();
  ctx.fillStyle='rgba(120,200,255,.3)';ctx.fillRect(-w*.26,y+h*.2,w*.52,h*.08);
  // racing stripe
  ctx.fillStyle=stripe;ctx.globalAlpha=.95;
  ctx.fillRect(-w*.14,y+h*.06,w*.1,h*.82);ctx.fillRect(w*.04,y+h*.06,w*.1,h*.82);ctx.globalAlpha=1;
}

function drawPowerup(pu){
  if(pu.life<2&&Math.floor(pu.life*8)%2)return;
  const cols={repair:C.green,rapid:C.cyan,spread:C.yellow,shield:'#bfffff'}[pu.type];
  const letters={repair:'+',rapid:'R',spread:'W',shield:'S'}[pu.type];
  const s=1+Math.sin(pu.t*6)*.12;
  ctx.save();ctx.translate(pu.x,pu.y);ctx.scale(s,s);
  ctx.save();ctx.globalCompositeOperation='lighter';
  const g=ctx.createRadialGradient(0,0,2,0,0,20);g.addColorStop(0,'rgba(255,255,255,.2)');g.addColorStop(1,'rgba(255,255,255,0)');
  ctx.fillStyle=g;ctx.beginPath();ctx.arc(0,0,20,0,7);ctx.fill();ctx.restore();
  ctx.rotate(Math.sin(pu.t*3)*.15);
  ctx.fillStyle='#0b141c';rr(-13,-13,26,26,5);ctx.fill();
  ctx.strokeStyle=cols;ctx.lineWidth=2.5;rr(-13,-13,26,26,5);ctx.stroke();
  ctx.fillStyle=cols;ctx.font='700 16px '+FMONO;ctx.textAlign='center';ctx.textBaseline='middle';ctx.fillText(letters,0,1);
  ctx.restore();
}
function drawParts(){
  for(const p of parts){const a=clamp(p.life/p.max,0,1);
    if(p.type==='smoke'){ctx.globalCompositeOperation='source-over';ctx.globalAlpha=a*.28;
      ctx.fillStyle=p.col;ctx.beginPath();ctx.arc(p.x,p.y,p.size*(1+(1-a)*.8),0,7);ctx.fill();}
    else{ctx.globalCompositeOperation='lighter';
      if(p.type==='spark'){ctx.globalAlpha=a;ctx.fillStyle=p.col;ctx.fillRect(p.x-1.5,p.y-1.5,p.size,p.size);}
      else if(p.type==='flame'){ctx.globalAlpha=a*.85;ctx.fillStyle=p.col;
        ctx.beginPath();ctx.arc(p.x,p.y,Math.max(.5,p.size*Math.pow(a,.7)),0,7);ctx.fill();}
      else if(p.type==='ring'){const prog=1-a;ctx.globalAlpha=a*.9;ctx.strokeStyle=p.col;ctx.lineWidth=3;
        ctx.beginPath();ctx.arc(p.x,p.y,p.r+prog*p.vr,0,7);ctx.stroke();}
      else if(p.type==='flash'){ctx.globalAlpha=a;ctx.fillStyle=p.col;
        ctx.beginPath();ctx.arc(p.x,p.y,p.size*a,0,7);ctx.fill();}}}
  ctx.globalCompositeOperation='source-over';ctx.globalAlpha=1;
}
function draw(){
  ctx.clearRect(0,0,W,H);
  ctx.fillStyle=C.off;ctx.fillRect(0,0,W,H);
  const so=G.scroll;
  drawTile(tileB,(so*.45)%320,320);
  drawTile(tileA,so%160,160);

  ctx.save();
  ctx.translate((Math.random()*2-1)*G.shake,(Math.random()*2-1)*G.shake);

  /* road */
  ctx.fillStyle=patAsphalt||C.road;ctx.fillRect(roadX,0,roadW,H);
  let gl=ctx.createLinearGradient(roadX,0,roadX+30,0);gl.addColorStop(0,'rgba(0,0,0,.4)');gl.addColorStop(1,'rgba(0,0,0,0)');
  ctx.fillStyle=gl;ctx.fillRect(roadX,0,30,H);
  gl=ctx.createLinearGradient(roadX+roadW,0,roadX+roadW-30,0);gl.addColorStop(0,'rgba(0,0,0,.4)');gl.addColorStop(1,'rgba(0,0,0,0)');
  ctx.fillStyle=gl;ctx.fillRect(roadX+roadW-30,0,30,H);
  /* rumble */
  const rb=42;
  for(let y=-(so%rb);y<H;y+=rb){const idx=Math.floor((y+so)/rb);
    ctx.fillStyle=idx%2?C.red:'#e8f1f5';
    ctx.fillRect(roadX-16,y,16,rb);ctx.fillRect(roadX+roadW,y,16,rb);}
  /* edge lines + lane dashes */
  ctx.fillStyle='rgba(232,241,245,.9)';ctx.fillRect(roadX-4,0,3,H);ctx.fillRect(roadX+roadW+1,0,3,H);
  ctx.fillStyle=C.dash;const gp=84;
  for(let l=1;l<4;l++){const x=roadX+l*laneW;
    for(let y=-(so%gp);y<H;y+=gp)ctx.fillRect(x-2,y,4,46);}
  /* sector checker strips */
  const SH=26,cs=16;
  for(let k=Math.floor(so/SECTOR);k*SECTOR<so+H+SH;k++){const y=H+so-k*SECTOR;if(y>H||y<-SH)continue;
    for(let i=0;i*cs<roadW;i++){const xw=Math.min(cs,roadX+ (i+1)*cs - roadX, roadW);
      ctx.fillStyle=(i)%2?'#e8f1f5':'#0d141c';ctx.fillRect(roadX+i*cs,y,xw,SH/2);
      ctx.fillStyle=((i+1))%2?'#e8f1f5':'#0d141c';ctx.fillRect(roadX+i*cs,y+SH/2,xw,SH/2);}}

  for(const dc of decos)drawDeco(dc);
  for(const pu of powerups)drawPowerup(pu);
  for(const e of enemies)drawEnemy(e);

  /* bullets */
  ctx.save();ctx.globalCompositeOperation='lighter';
  for(const b of ebullets){
    const g=ctx.createRadialGradient(b.x,b.y,0,b.x,b.y,9);
    g.addColorStop(0,'rgba(255,160,120,.9)');g.addColorStop(.4,'rgba(255,80,60,.5)');g.addColorStop(1,'rgba(255,80,60,0)');
    ctx.fillStyle=g;ctx.beginPath();ctx.arc(b.x,b.y,9,0,7);ctx.fill();
    ctx.fillStyle='#ffd9c4';ctx.beginPath();ctx.arc(b.x,b.y,3,0,7);ctx.fill();}
  for(const b of bullets){
    ctx.strokeStyle='rgba(255,210,63,.5)';ctx.lineWidth=5;
    ctx.beginPath();ctx.moveTo(b.x-b.vx*.012,b.y-b.vy*.012);ctx.lineTo(b.x,b.y);ctx.stroke();
    ctx.fillStyle='#fff';ctx.fillRect(b.x-1.5,b.y-7,3,14);}
  ctx.restore();

  if(P.alive&&!(P.inv>0&&Math.floor(G.t*14)%2===0))drawPlayer();
  drawParts();

  /* speed streaks */
  if((G.speed>680||P.nitroOn)&&(G.state==='playing'||G.state==='lights')){
    const n=P.nitroOn?10:5;ctx.save();ctx.globalCompositeOperation='lighter';
    for(let i=0;i<n;i++){const x=Math.random()*W,y=Math.random()*H,len=rand(30,60)+G.speed*.07;
      ctx.strokeStyle=(P.nitroOn?'rgba(53,224,255,':'rgba(255,255,255,')+rand(.04,.12)+')';ctx.lineWidth=2;
      ctx.beginPath();ctx.moveTo(x,y);ctx.lineTo(x,y+len);ctx.stroke();}
    ctx.restore();}

  /* floats */
  for(const f of floats){ctx.globalAlpha=clamp(f.life*1.6,0,1);
    ctx.font='700 '+f.size+'px '+FMONO;ctx.textAlign='center';
    ctx.lineWidth=3;ctx.strokeStyle='rgba(5,10,14,.8)';ctx.strokeText(f.txt,f.x,f.y);
    ctx.fillStyle=f.col;ctx.fillText(f.txt,f.x,f.y);}
  ctx.globalAlpha=1;
  ctx.restore();
}
