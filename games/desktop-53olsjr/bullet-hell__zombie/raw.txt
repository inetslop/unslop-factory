  p.bombs--; p.bombCd=1.2; slow.amt=.3; slow.t=.8; shake=Math.max(shake,18); sBombS();
  for(let i=0;i<ebullets.length;i++){ const b=ebullets[i]; addPart(b.x,b.y,rand(-40,40),rand(-40,40),.4,2.5,'#ffd76a','dot'); }
  ebullets.length=0;
  for(let i=enemies.length-1;i>=0;i--){ const e=enemies[i]; if(e.dead)continue; e.hp-=12; e.flash=1; if(e.hp<=0){e.dead=true; killEnemy(e);} }
  if(boss){ boss.hp-=45; boss.flash=1; if(boss.hp<=0) bossDie(); }
  rings.push({x:p.x,y:p.y,r:20,vr:900,life:.6,t:0,c:'#aef3ff',lw:7});
  rings.push({x:p.x,y:p.y,r:5,vr:560,life:.8,t:0,c:'#ffd76a',lw:4});
}
function hurt(){
  if(p.inv>0||p.dead)return;
  p.lives--; p.inv=2.2; slow.amt=.35; slow.t=.35; shake=Math.max(shake,15); hitFlash=1; sHurt();
  for(let i=ebullets.length-1;i>=0;i--){ const b=ebullets[i];
    if((b.x-p.x)**2+(b.y-p.y)**2<90*90){ addPart(b.x,b.y,rand(-60,60),rand(-60,60),.35,2,'#ff8080','dot'); ebullets.splice(i,1);} }
  if(p.lives<0) die();
}
function die(){
  p.dead=true; deathT=1.5; shake=24; slow.amt=.25; slow.t=1.1; sBombS(); sHurt();
  for(let i=0;i<36;i++){ const a=rand(0,TAU),s=rand(40,300); addPart(p.x,p.y,Math.cos(a)*s,Math.sin(a)*s,rand(.4,.9),rand(2,5),pick(['#ffb347','#a4ff5e','#fff','']),'dot'); }

  ebullets.length=0;
}

function snap8(dx,dy){ const a=Math.round(Math.atan2(dy,dx)/(Math.PI/4))*(Math.PI/4); return {x:Math.cos(a),y:Math.sin(a)}; }
function playerUpdate(dt){
  if(p.dead)return;
  p.inv=Math.max(0,p.inv-dt); p.fireCd-=dt;
  let dx=(keys['arrowright']||keys['d']?1:0)-(keys['arrowleft']||keys['a']?1:0);
  let dy=(keys['arrowdown']||keys['s']?1:0)-(keys['arrowup']||keys['w']?1:0);
  if(dx&&dy){dx*=.7071;dy*=.7071;}
  if(dx||dy) p.aim=snap8(dx,dy);
  const focus=!!keys['shift'];
  const sp=p.speed*(focus?.45:1);
  p.x+=dx*sp*dt; p.y+=dy*sp*dt;
  if(touch.on){
    const tx=touch.x, ty=touch.y-70, ddx=tx-p.x, ddy=ty-p.y, d=Math.hypot(ddx,ddy);
    if(d>3){ const m=Math.min(d,sp*dt); p.x+=ddx/d*m; p.y+=ddy/d*m; }
  }
  p.x=clamp(p.x,16,LW-16); p.y=clamp(p.y,52,LH-24);
  if((dx||dy)&&!touch.on){ p.trailT-=dt; if(p.trailT<=0){ p.trailT=.035; addPart(p.x-p.aim.x*15,p.y-p.aim.y*15,-p.aim.x*30+rand(-8,8),-p.aim.y*30+rand(-8,8),.3,2.4,'#ff9a3d','dot'); } }
  const firing=touch.on||keys[' ']||keys['j'];
  if(firing&&p.fireCd<=0){
    p.fireCd=.105;
    const n=1+p.spread*2, base=Math.atan2(p.aim.y,p.aim.x);
    for(let i=0;i<n;i++){
      const a=base+(i-(n-1)/2)*.13, px=p.x+p.aim.x*17, py=p.y+p.aim.y*17;
      pbullets.push({x:px,y:py,px,py,vx:Math.cos(a)*640,vy:Math.sin(a)*640});
      shots++; addPart(px,py,rand(-30,30),rand(-30,30),.12,2,'#fff','dot');
    }

  }
}
function pickType(n){
  const w=[['shambler',10]];
  if(n>=2)w.push(['spitter',6+n]);
  if(n>=3)w.push(['ghoul',5+n*.6]);
  if(n>=4)w.push(['wraith',4+n*.5]);
  if(n>=6)w.push(['brute',2+n*.3]);
  let tot=0; for(const t of w)tot+=t[1]; let r=Math.random()*tot;
  for(const t of w){ r-=t[1]; if(r<=0)return t[0]; } return 'shambler';
}
function waveUpdate(dt){
  if(wave.bannerT>0)wave.bannerT-=dt;
  if(p.dead||boss)return;
  if(wave.toSpawn>0&&wave.bannerT<=0){
    wave.t-=dt;
    if(wave.t<=0){ spawnEnemy(pickType(wave.n),rand(30,LW-30)); wave.toSpawn--; wave.t=wave.interval*rand(.7,1.35); }
  }
  if(wave.toSpawn===0&&enemies.length===0){ wave.gap+=dt; if(wave.gap>1.4) startWave(wave.n+1); }
  else wave.gap=0;
}
function updateEnemies(dt){
  for(const e of enemies){
    if(e.dead||e.gone)continue;
    e.t+=dt; e.flash=Math.max(0,e.flash-dt*6);
    if(e.type==='shambler'){
      e.y+=e.vy*dt; e.x+=Math.sin(e.t*1.6+e.ph)*22*dt;
    }else if(e.type==='spitter'){
      if(e.y<e.ty)e.y+=60*dt; else e.x=e.baseX+Math.sin(e.t*.9+e.ph)*44;
      e.cd-=dt;
      if(e.cd<=0&&p&&!p.dead){ e.cd=Math.max(1.3,2.3-wave.n*.05);
        const a=Math.atan2(p.y-e.y,p.x-e.x); fanB(e.x,e.y+e.r*.5,a,3,.55,175,'glob'); }
    }else if(e.type==='ghoul'){
      if(e.state==='wind'){ e.t+=0; e.y+=26*dt;
        if(e.t>=e.windT){ e.state='dash'; const a=Math.atan2(p.y-e.y,p.x-e.x); e.vx=Math.cos(a)*520; e.vy=Math.sin(a)*520; }
      }else{ e.x+=e.vx*dt; e.y+=e.vy*dt; }
    }else if(e.type==='wraith'){
      e.y+=34*dt; e.x=e.baseX+Math.sin(e.t*e.freq+e.ph)*e.amp;
      if(Math.random()<dt*12) addPart(e.x+rand(-4,4),e.y-e.r*1.3,0,-25,.5,2,'#9fe8ff','dot');
      e.cd-=dt;
      if(e.cd<=0&&p&&!p.dead){ e.cd=1.7; e.alt^=1;
        if(e.alt) for(let i=0;i<10;i++) eb(e.x,e.y,i*TAU/10+e.t,150,'ring');
        else fanB(e.x,e.y,Math.atan2(p.y-e.y,p.x-e.x),3,.22,265,'bolt'); }
    }else if(e.type==='brute'){
      e.y+=e.vy*dt; e.cd-=dt;
      if(e.cd<=0&&p&&!p.dead){ e.cd=2.4;
        eb(e.x,e.y,Math.PI/2,120,'blob'); eb(e.x,e.y,-Math.PI/2,120,'blob');
        eb(e.x,e.y,0,120,'blob'); eb(e.x,e.y,Math.PI,120,'blob');
        fanB(e.x,e.y,Math.atan2(p.y-e.y,p.x-e.x),3,.5,140,'glob'); }
    }
    if(e.y>LH+60||e.x<-70||e.x>LW+70) e.gone=true;
  }
}
function updateBoss(dt){
  const b=boss; b.t+=dt; b.flash=Math.max(0,b.flash-dt*6);
  if(b.y<120){ b.y+=55*dt; return; }
  const frac=b.hp/b.hpMax; b.phase=frac>.6?1:frac>.3?2:3;
  const amp=b.phase===3?LW*.32:LW*.26;
  b.x=LW/2+Math.sin(b.t*(b.phase===3?.6:.4))*amp;
  if(p.dead)return;
  b.cd-=dt;
  if(b.cd<=0){
    const ta=Math.atan2(p.y-b.y,p.x-b.x);
    if(b.phase===1){ b.cd=1.5; b.alt^=1;
      if(b.alt) fanB(b.x,b.y-24,ta,5,.5,230,'pink');
      else { for(let i=0;i<12;i++)eb(b.x,b.y,b.spin+i*TAU/12,160,'gold'); b.spin+=.5; }
    }else if(b.phase===2){ b.cd=1.15; b.alt^=1;
      if(b.alt) fanB(b.x,b.y-24,ta,7,.55,240,'pink');
      else { for(let i=0;i<16;i++)eb(b.x,b.y,b.spin+i*TAU/16,175,'gold'); b.spin+=.42; }
    }else{ b.cd=.95; b.alt=(b.alt+1)%3;
      if(b.alt===0) fanB(b.x,b.y-24,ta,7,.6,255,'pink');
      else if(b.alt===1){ for(let i=0;i<18;i++)eb(b.x,b.y,b.spin+i*TAU/18,190,'gold'); b.spin+=.35; }
      else fanB(b.x,b.y-24,ta,3,.14,275,'bolt');
    }
  }
  if(b.phase>=2){ b.rt-=dt; if(b.rt<=0){ b.rt=2.6; for(let i=0;i<18;i++)eb(b.x,b.y,i*TAU/18+b.t*.3,150,'ring'); } }
  if(b.phase===3){ b.st-=dt; if(b.st<=0&&enemies.length<9){ b.st=6; spawnEnemy('ghoul',rand(40,LW-40)); spawnEnemy('ghoul',rand(40,LW-40)); } }
}

function collisions(){
  for(let i=pbullets.length-1;i>=0;i--){
    const b=pbullets[i]; let hitAny=false;
    for(const e of enemies){ if(e.dead||e.gone)continue;
      const rr=e.r+4; if((b.x-e.x)**2+(b.y-e.y)**2<rr*rr){
        hitsCt++; e.hp--; e.flash=1; hitAny=true; sHitE(); addPart(b.x,b.y,rand(-50,50),rand(-50,50),.2,2,'#fff','dot');
        if(e.hp<=0){e.dead=true; killEnemy(e);} break; } }
    if(!hitAny&&boss){ const rr=boss.r+4;
      if((b.x-boss.x)**2+(b.y-boss.y)**2<rr*rr){ hitsCt++; boss.hp--; boss.flash=1; hitAny=true; sHitE(); addPart(b.x,b.y,rand(-50,50),rand(-50,50),.2,2,'#fff','dot');
        if(boss.hp<=0)bossDie(); } }
    if(hitAny)pbullets.splice(i,1);
  }
  if(p.inv<=0&&!p.dead){
    for(let i=ebullets.length-1;i>=0;i--){ const b=ebullets[i], rr=BT[b.type].r+p.r+1;
      if((b.x-p.x)**2+(b.y-p.y)**2<rr*rr){ ebullets.splice(i,1); hurt(); break; } }
    if(p.inv<=0&&!p.dead) for(const e of enemies){ if(e.dead||e.gone)continue;
      const rr=e.r+p.r+2; if((b=>(b.x-p.x)**2+(b.y-p.y)**2<rr*rr)(e)){ hurt(); break; } }
    if(p.inv<=0&&!p.dead&&boss){ const rr=boss.r+p.r+2; if((boss.x-p.x)**2+(boss.y-p.y)**2<rr*rr)hurt(); }
  }
}

function update(rdt){
  if(state==='paused')return;
  if(slow.t>0)slow.t-=rdt;
  ts=slow.t>0?slow.amt:Math.min(1,ts+rdt*4);
  const dt=rdt*ts;
  updateAmbient(rdt,dt);
  if(state==='menu'){ for(const z of menuZ){ z.y+=z.vy*dt; if(z.y>LH+30){z.y=-30;z.x=rand(30,LW-30);} } return; }
  updateParticles(dt);
  if(state==='playing'){ playerUpdate(dt); waveUpdate(dt); }
  updateEnemies(dt);
  if(boss)updateBoss(dt);
  updateBullets(dt);
  if(state==='playing'&&!p.dead)collisions();
  updatePickups(dt);
  for(let i=enemies.length-1;i>=0;i--)if(enemies[i].dead||enemies[i].gone)enemies.splice(i,1);
  comboT=Math.max(0,comboT-rdt); if(comboT===0)combo=0;
  p.bombCd=Math.max(0,p.bombCd-dt);
  hitFlash=Math.max(0,hitFlash-rdt*2.2);
  shake=Math.max(0,shake-rdt*(shake*3+6));
  if(p.dead){ deathT-=rdt; if(deathT<=0&&state==='playing')showGameOver(); }
}

  ctx.fillStyle=c2; ctx.beginPath(); ctx.ellipse(0,-r*.15,r*.48,r*.6,0,0,TAU); ctx.fill();

function drawEnemy(e){
  if(e.type==='shambler') zbody(e.x,e.y,e.r,e.t,e.ph,e.c1,e.c2);
  else if(e.type==='spitter'){
    ctx.save(); ctx.translate(e.x,e.y);
    const g=e.cd<.45?Math.min(1,(.45-e.cd)/.45):0;
    ctx.fillStyle=e.c1; ctx.beginPath(); ctx.ellipse(0,0,e.r*1.05,e.r*.82,0,0,TAU); ctx.fill();
    ctx.fillStyle=e.c2; ctx.beginPath(); ctx.ellipse(0,-e.r*.2,e.r*.6,e.r*.45,0,0,TAU); ctx.fill();
    if(g>0){ ctx.fillStyle=`rgba(190,255,90,${.3+.6*g})`; circle(0,e.r*.55,3+4*g); }
    else { ctx.fillStyle='#1d2c18'; circle(0,e.r*.55,3); }
    ctx.restore();
  }
  else if(e.type==='ghoul'){
    const ang=e.state==='dash'?Math.atan2(e.vy,e.vx)+Math.PI/2:Math.PI/2;
    ctx.save(); ctx.translate(e.x+(e.state==='wind'?rand(-1.5,1.5):0),e.y); ctx.rotate(ang);
    if(e.state==='dash'){ ctx.strokeStyle='rgba(255,130,80,.3)'; ctx.lineWidth=3;
      for(const o of [-6,0,6]){ ctx.beginPath(); ctx.moveTo(o,e.r*1.4); ctx.lineTo(o,e.r*3); ctx.stroke(); } }
    ctx.fillStyle='#c96a3a'; ctx.beginPath(); ctx.ellipse(0,0,e.r*.72,e.r*1.35,0,0,TAU); ctx.fill();
    ctx.fillStyle='#8f3b2a'; circle(0,-e.r*1.05,e.r*.4);
    ctx.fillStyle='#ffe066'; circle(-2.5,-e.r*1.1,1.7); circle(2.5,-e.r*1.1,1.7);
    if(e.state==='wind'){ ctx.strokeStyle=`rgba(255,70,60,${.4+.4*Math.sin(tAmb*20)})`; ctx.lineWidth=2;
      ctx.beginPath(); ctx.arc(0,0,e.r+7,0,TAU); ctx.stroke(); }
    ctx.restore();
  }
  else if(e.type==='wraith'){
    const wv=Math.sin(e.t*9);
    ctx.save(); ctx.translate(e.x,e.y);
    ctx.fillStyle='rgba(150,235,255,.8)'; ctx.beginPath();
    ctx.moveTo(0,-e.r*1.5);
    ctx.bezierCurveTo(e.r*.9,-e.r*.8,e.r*1.1,e.r*.2,e.r*.7,e.r*.6);
    ctx.quadraticCurveTo(e.r*.45,e.r*1.25+ wv*e.r*.1,e.r*.26,e.r*.95);
    ctx.quadraticCurveTo(0,e.r*1.35-wv*e.r*.1,-e.r*.26,e.r*.95);
    ctx.quadraticCurveTo(-e.r*.45,e.r*1.25+wv*e.r*.1,-e.r*.7,e.r*.6);
    ctx.bezierCurveTo(-e.r*1.1,e.r*.2,-e.r*.9,-e.r*.8,0,-e.r*1.5);
    ctx.closePath(); ctx.fill();
    ctx.fillStyle='rgba(234,255,255,.9)'; circle(0,-e.r*.3,e.r*.4);
    ctx.fillStyle='#0b2530'; circle(-3.5,-e.r*.42,1.8); circle(3.5,-e.r*.42,1.8);
    ctx.restore();
  }
  else if(e.type==='brute'){
    const g=e.cd<.6?Math.min(1,(.6-e.cd)/.6):0;
    ctx.save(); ctx.translate(e.x,e.y);
    ctx.fillStyle='#3d5c31'; circle(0,0,e.r);
    ctx.strokeStyle='#263c1f'; ctx.lineWidth=5; ctx.beginPath(); ctx.arc(0,0,e.r-2.5,0,TAU); ctx.stroke();
    ctx.strokeStyle=`rgba(255,140,60,${.25+g*.6})`; ctx.lineWidth=2;
    for(const a of e.cr){ ctx.beginPath(); ctx.moveTo(0,0); ctx.lineTo(Math.cos(a)*e.r*.8,Math.sin(a)*e.r*.8); ctx.stroke(); }
    ctx.fillStyle='#2c4527'; circle(0,e.r*.68,e.r*.36);
    ctx.fillStyle='#ffa03d'; circle(-5,e.r*.72,2.2); circle(5,e.r*.72,2.2);
    ctx.restore();
  }
  if(e.flash>0){ ctx.fillStyle=`rgba(255,255,255,${e.flash*.8})`; circle(e.x,e.y,e.r*1.08); }
}
function drawBoss(b){
  const ph=b.phase;
  ctx.save(); ctx.translate(b.x,b.y);
  ctx.strokeStyle=ph===3?`rgba(255,61,61,${.14+.1*Math.sin(tAmb*4)})`:`rgba(255,111,176,${.12+.08*Math.sin(tAmb*3)})`;
  ctx.lineWidth=9; ctx.beginPath(); ctx.arc(0,0,b.r+10,0,TAU); ctx.stroke();
  for(let i=0;i<6;i++){ const a=tAmb*.9+i*TAU/6;
    ctx.fillStyle=i%2?'#ffd76a':'#ff6fb0'; ctx.globalAlpha=.5+.4*Math.sin(tAmb*4+i);
    circle(Math.cos(a)*(b.r+17),Math.sin(a)*(b.r+17),3); }
  ctx.globalAlpha=1;
  const path=()=>{ ctx.beginPath(); ctx.moveTo(0,-48); ctx.lineTo(27,-30); ctx.lineTo(27,28); ctx.lineTo(15,48); ctx.lineTo(-15,48); ctx.lineTo(-27,28); ctx.lineTo(-27,-30); ctx.closePath(); };
  path(); ctx.fillStyle='#2b1e14'; ctx.fill(); ctx.strokeStyle='#c9a24b'; ctx.lineWidth=2.5; ctx.stroke();
  ctx.save(); ctx.scale(.78,.78); path(); ctx.strokeStyle='rgba(201,162,75,.4)'; ctx.lineWidth=1.4; ctx.stroke(); ctx.restore();
  ctx.fillStyle='#c9a24b'; ctx.fillRect(-3,-22,6,32); ctx.fillRect(-11,-13,22,6);
  const ec=ph===3?'#ff3d3d':'#ff4d8f';
  ctx.fillStyle=`rgba(255,77,143,${.3+.25*Math.sin(tAmb*5)})`; circle(-9,-38,6); circle(9,-38,6);

  ctx.restore();
}

}

function render(){
  /* --- screen-space backdrop --- */
  ctx.setTransform(dpr,0,0,dpr,0,0);
  const sky=ctx.createLinearGradient(0,0,0,H);
  sky.addColorStop(0,'#0a1410'); sky.addColorStop(.55,'#0c1712'); sky.addColorStop(1,'#081009');
  ctx.fillStyle=sky; ctx.fillRect(0,0,W,H);
  let g=ctx.createRadialGradient(W*.5,H*.12,10,W*.5,H*.12,H*.65);
  g.addColorStop(0,'rgba(200,230,210,.09)'); g.addColorStop(1,'rgba(200,230,210,0)');
  ctx.fillStyle=g; ctx.fillRect(0,0,W,H);
  for(const f of fog){ const fg=ctx.createRadialGradient(f.x,f.y,10,f.x,f.y,f.r);
    fg.addColorStop(0,`rgba(140,190,160,${f.a})`); fg.addColorStop(1,'rgba(140,190,160,0)');
    ctx.fillStyle=fg; ctx.beginPath(); ctx.arc(f.x,f.y,f.r,0,TAU); ctx.fill(); }

  /* --- world space (with shake) --- */
  const sx=shake?rand(-shake,shake):0, sy=shake?rand(-shake,shake)*.6:0;
  ctx.setTransform(dpr*scale,0,0,dpr*scale,dpr*(ox+sx),dpr*sy);
  if(!groundPat)groundPat=ctx.createPattern(tile,'repeat');
  const off=((gOff%240)+240)%240;
  ctx.save(); ctx.translate(0,off); ctx.fillStyle=groundPat; ctx.fillRect(-12,-252,LW+24,LH+264); ctx.restore();
  /* stones */
  for(const s of stones){
    ctx.save(); ctx.translate(s.x,s.y); ctx.rotate(s.rot); ctx.scale(s.s,s.s);
    ctx.fillStyle='#1b2620';
    if(s.type===0){ ctx.fillRect(-3,-14,6,28); ctx.fillRect(-11,-7,22,6); }
    else if(s.type===1){ rr(-11,-14,22,28,9); ctx.fill(); }
    else { rr(-10,-6,20,14,3); ctx.fill(); }
    ctx.fillStyle='rgba(190,230,200,.1)'; ctx.fillRect(s.type===0?-11:-10,s.type===1?-14:s.type===0?-14:-6,s.type===0?22:20,2);
    ctx.restore();
  }
  /* outside-fence void + fence */
  ctx.fillStyle='rgba(2,5,7,.62)'; ctx.fillRect(-900,-20,892,LH+40); ctx.fillRect(LW+10,-20,900,LH+40);
  const poff=((gOff%64)+64)%64;
  for(const fx of [-8,LW+8]){
    ctx.strokeStyle='#0c1114'; ctx.lineWidth=5; ctx.beginPath(); ctx.moveTo(fx,-10); ctx.lineTo(fx,LH+10); ctx.stroke();
    for(let y=-64+poff;y<LH+64;y+=64){
      ctx.fillStyle='#10171b'; ctx.fillRect(fx-3,y,6,15);
      ctx.fillStyle='rgba(200,230,210,.12)'; ctx.fillRect(fx-3,y,6,2);
      if(((y/64)|0)%4===0){ const la=.4+.3*Math.sin(tAmb*2+y);
        ctx.fillStyle=`rgba(255,155,70,${la})`; circle(fx+8,y+4,2.4);
        ctx.fillStyle=`rgba(255,155,70,${la*.25})`; circle(fx+8,y+4,7); }
    }
  }
  /* crows in the void */
  for(const c of crows){ const f=Math.sin(c.t*9)*4;
    ctx.strokeStyle='rgba(36,48,62,.9)'; ctx.lineWidth=2; ctx.lineCap='round';
    ctx.beginPath(); ctx.moveTo(c.x-7,c.y); ctx.quadraticCurveTo(c.x-3,c.y-4-f,c.x,c.y);
    ctx.quadraticCurveTo(c.x+3,c.y-4-f,c.x+7,c.y); ctx.stroke(); }
  /* motes */
  for(const m of motes){ ctx.fillStyle=`rgba(159,216,176,${.12+.14*Math.sin(tAmb*2+m.ph)})`; circle(m.x,m.y,1.4); }
  /* rings */
  for(const r of rings){ const f=1-r.t/r.life;
    ctx.strokeStyle=r.c; ctx.globalAlpha=f*.6; ctx.lineWidth=Math.max(.5,r.lw*f);
    ctx.beginPath(); ctx.arc(r.x,r.y,r.r,0,TAU); ctx.stroke(); }
  ctx.globalAlpha=1;
  for(const k of pickups)drawPickup(k);
  if(boss)drawBoss(boss);
  for(const e of enemies)if(!e.dead&&!e.gone)drawEnemy(e);
  if(state==='menu')for(const z of menuZ)zbody(z.x,z.y,13,tAmb+z.ph,'#4a7238','#6ea84e');
  drawPlayer();
  /* player bullets */
  for(const b of pbullets){
    ctx.strokeStyle='rgba(200,240,255,.4)'; ctx.lineWidth=2;
    ctx.beginPath(); ctx.moveTo(b.px,b.py); ctx.lineTo(b.x,b.y); ctx.stroke();
    ctx.fillStyle='rgba(170,225,255,.3)'; circle(b.x,b.y,6.5);
    ctx.fillStyle='#fff'; circle(b.x,b.y,3);
  }
  /* enemy bullets */
  for(const b of ebullets){ const c=BT[b.type];
    if(c.trail){ ctx.strokeStyle=c.o; ctx.lineWidth=3; ctx.beginPath(); ctx.moveTo(b.px,b.py); ctx.lineTo(b.x,b.y); ctx.stroke(); }
    ctx.fillStyle=c.o; circle(b.x,b.y,c.r*1.9);
    ctx.fillStyle=c.c; circle(b.x,b.y,c.r);
    if(c.ring){ ctx.strokeStyle=c.c; ctx.lineWidth=2; ctx.beginPath(); ctx.arc(b.x,b.y,c.r*.5,0,TAU); ctx.stroke(); }
  }
  /* particles */
  for(const q of parts){ const f=q.t/q.life;
    if(q.type==='smoke'){ ctx.fillStyle=`rgba(40,60,45,${.25*(1-f)})`; circle(q.x,q.y,q.size*(1+f*2)); }
    else { ctx.globalAlpha=(1-f)*.9; ctx.fillStyle=q.c||'#fff'; circle(q.x,q.y,q.size*(1-f*.5)); ctx.globalAlpha=1; } }
  /* popups */
  for(const q of pops){ const f=q.t/q.life;
    ctx.globalAlpha=1-f; ctx.font='bold '+q.size+'px Impact,"Arial Narrow",sans-serif'; ctx.textAlign='center';
    ctx.strokeStyle='rgba(0,0,0,.7)'; ctx.lineWidth=3; ctx.strokeText(q.txt,q.x,q.y-q.t*42);
    ctx.fillStyle=q.c; ctx.fillText(q.txt,q.x,q.y-q.t*42); ctx.globalAlpha=1; }
  /* hit flash */
  if(hitFlash>0){ ctx.setTransform(dpr,0,0,dpr,0,0);
    ctx.fillStyle=`rgba(255,45,60,${hitFlash*.3})`; ctx.fillRect(0,0,W,H); }
  /* HUD + banner (unshaken) */
  if(state!=='menu'){
    ctx.setTransform(dpr*scale,0,0,dpr*scale,dpr*ox,0);
    drawHUD(); drawBanner();
  }
}

  requestAnimationFrame(frame);
  const rdt=Math.min(.033,(t-last)/1000||.016); last=t;
  update(rdt); render();
}
