let last=performance.now();
function frame(now){
  requestAnimationFrame(frame);
  let dt=Math.min(.05,(now-last)/1000); last=now;
  if(state==='playing'||state==='dying'||state==='levelclear')busyT=Math.max(0,busyT-dt);
  [player,...crates].forEach(e=>{if(e&&e.at<1)e.at=Math.min(1,e.at+dt/.13);});
  moving=!!(player&&player.at<1);
  if(moving)walkPhase=(walkPhase+dt*6)%1; else walkPhase=0;
  boardT=Math.min(1,boardT+dt*3);
  shakeAmp=Math.max(0,shakeAmp-dt*2.1);
  for(let i=particles.length-1;i>=0;i--){
    const p=particles[i]; p.t+=dt;
    if(p.t>=p.life){particles.splice(i,1);continue;}
    p.x+=p.vx*dt; p.y+=p.vy*dt;
    const dr=p.type==='smoke'?1.6:3;
    p.vx*=(1-dt*dr); p.vy*=(1-dt*dr);
  }
  for(let i=floats.length-1;i>=0;i--){floats[i].t+=dt/.95; if(floats[i].t>=1)floats.splice(i,1);}
  for(let i=flashes.length-1;i>=0;i--){flashes[i].t+=dt/.45; if(flashes[i].t>=1)flashes.splice(i,1);}
  if(state==='dying'){
    dyingT+=dt;
    if(dyingT>.85&&!minesRevealed){
      minesRevealed=true;
      let i=0; for(const k of mines)minePop[k]=performance.now()+(i++)*45;
    }
    if(dyingT>1.75){
      lives--; updateHUD();
      if(lives>0){loadLevel(levelIdx);state='playing';banner('LEVEL '+(levelIdx+1),LEVELS[levelIdx].name);}
      else{state='gameover';sfx.over();showScreen('gameover');}
    }
  }
  if(state==='levelclear'){
    clearT+=dt;
    if(clearT>1.5){
      if(levelIdx+1>=LEVELS.length){state='victory';showScreen('victory');}
      else{levelIdx++;loadLevel(levelIdx);state='playing';banner('LEVEL '+(levelIdx+1),LEVELS[levelIdx].name);}
    }
  }
  draw(now);
}

loadLevel(0);
