function tone(f0,f1,dur,type,vol,delay){
  if(!AC||muted)return;delay=delay||0;
  const t0=AC.currentTime+delay,o=AC.createOscillator(),g=AC.createGain();
  o.type=type;o.frequency.setValueAtTime(Math.max(1,f0),t0);
  if(f1)o.frequency.exponentialRampToValueAtTime(Math.max(1,f1),t0+dur);
  g.gain.setValueAtTime(vol,t0);g.gain.exponentialRampToValueAtTime(.0001,t0+dur);
  o.connect(g).connect(master);o.start(t0);o.stop(t0+dur+.03);
}
function noiseB(dur,vol,f,delay){

  for(let i=0;i<n;i++)d[i]=(Math.random()*2-1)*(1-i/n);
  const s=AC.createBufferSource();s.buffer=buf;
  const flt=AC.createBiquadFilter();flt.type='lowpass';flt.frequency.value=f;
  const g=AC.createGain();g.gain.value=vol;s.connect(flt).connect(g).connect(master);s.start(t0);
}

function tileOpen(t,d){if(!t)return false;
  if(t.open)return t.open(d);
  if(t.kind==='ground')return false;
  const base=t.kind==='straight'?[0,2]:[0,1];
  return base.some(b=>(b+t.rot)&3===d);
}
function cellOpen(x,y,d){if(x<0||y<0||x>=cols||y>=rows)return false;
  const k=key(x,y);
  if(trapMap[k]||hubMap[k])return true;
  const g=gateMap[k];if(g)return d===g.dir;
  return tileOpen(tiles[y][x],d);
}
function nextMove(z){
  const entry=(z.dir+2)&3,t=tiles[z.cy][z.cx];
  if(!t||t.kind==='ground')return -1;
  for(let d=0;d<4;d++){

    if(!tileOpen(t,d))continue;
    const nx=z.cx+DX[d],ny=z.cy+DY[d];
    if(cellOpen(nx,ny,(d+2)&3))return d;
  }
  return -1;
}

/* ============================== UPDATE ============================== */
function update(dt){
  time+=dt;
  if(bannerT>0){bannerT-=dt;if(bannerT<=0){const cb=bannerCb;hideBanner();if(cb)cb();}}
  // tiles spin
  for(let y=0;y<rows;y++)for(let x=0;x<cols;x++){
    const t=tiles[y][x];if(t.kind==='ground')continue;
    if(Math.abs(t.tgt-t.spin)>.001)t.spin+=(t.tgt-t.spin)*Math.min(1,dt*16);else t.spin=t.tgt;
  }
  // gates
  if(state==='playing'){
    for(const g of gates){
      if(g.closed)continue;
      g.mistT-=dt;
      if(g.mistT<=0){g.mistT=rand(.3,.6);
        parts.push({k:'smoke',x:ox+(g.x+.5)*ts+rand(-6,6),y:oy+(g.y+.5)*ts,vx:rand(-6,6),vy:rand(-22,-10),g:-8,r:ts*.12,life:rand(.9,1.5),t:0,c:'#3f7a2c'});}
      g.timer-=dt;
      if(g.timer<=0&&g.remaining>0){spawnZombie(g);g.remaining--;g.timer=g.interval;}
      if(g.remaining===0)g.closed=true;
    }
  }
  // zombies
  const spd=levelSpeed*(state==='dying'?.35:1);
  for(const z of zombies){
    if(z.dead)continue;
    z.phase+=dt*(z.st==='move'?z.type==='runner'?11:7:4)*spd;
    if(z.st==='stuck'){
      const d=nextMove(z);
      if(d>=0){z.dir=d;z.next=[z.cx+DX[d],z.cy+DY[d]];z.st='move';}
      continue;
    }
    const tx=z.next[0],ty=z.next[1];
    const dx=tx-z.x,dy=ty-z.y,dist=Math.hypot(dx,dy),step=ZTYPES[z.type].speed*spd*dt;
    if(dist<=step){z.x=tx;z.y=ty;z.cx=tx;z.cy=ty;arrive(z);}
    else{z.x+=dx/dist*step;z.y+=dy/dist*step;}
    if(z.type==='brute'&&z.st==='move'){z.dustT-=dt;
      if(z.dustT<=0){z.dustT=.5;parts.push({k:'smoke',x:ox+(z.x+.5)*ts,y:oy+(z.y+.6)*ts,vx:rand(-8,8),vy:-10,g:-4,r:ts*.1,life:.7,t:0,c:'#3a4232'});}}
  }
  zombies=zombies.filter(z=>!z.dead);
  // level end
  if(state==='playing'&&gates.every(g=>g.closed)&&zombies.length===0)beginClear();
  if(state==='dying'){overT-=dt;if(overT<=0)gameOver();}
  // particles / decals / pops / fx decay
  for(const p of parts){p.t+=dt;
    if(p.k!=='ring'){p.vy+=(p.g||0)*dt;p.x+=p.vx*dt;p.y+=p.vy*dt;}
  }
  parts=parts.filter(p=>p.t<p.life);
  for(const d of decals)d.a-=dt*.015;
  decals=decals.filter(d=>d.a>.05);
  for(const p of pops)p.t+=dt;
  pops=pops.filter(p=>p.t<.95);
  traps.forEach(t=>t.flash=Math.max(0,t.flash-dt*2.6));
  hubs.forEach(h=>h.flash=Math.max(0,h.flash-dt*2.2));
  shakeAmp=Math.max(0,shakeAmp-dt*38);flashRed=Math.max(0,flashRed-dt*2.4);
  // long-press CCW
  if(state==='playing'&&downCell&&!longFired&&time-downAt>.35){rotateAt(downCell.x,downCell.y,-1);longFired=true;}
  // ambient
  for(const f of fog){f.x+=f.vx*dt;if(f.x>1.2)f.x=-.2;if(f.x<-.2)f.x=1.2;}
  for(const s of spores){s.y-=s.spd*dt;s.x+=Math.sin(time*.8+s.ph)*.0004;
    if(s.y<-.05){s.y=1.05;s.x=Math.random();}}
  for(const w of walkers){w.x+=w.dir*w.spd*dt;if(w.x>W+60)w.x=-60;if(w.x<-60)w.x=W+60;w.ph+=dt*3;}
}

function rrect(x,y,w,h,r){ctx.beginPath();ctx.moveTo(x+r,y);ctx.arcTo(x+w,y,x+w,y+h,r);ctx.arcTo(x+w,y+h,x,y+h,r);ctx.arcTo(x,y+h,x,y,r);ctx.arcTo(x,y,x+w,y,r);ctx.closePath();}
function heartPath(s){ctx.beginPath();ctx.moveTo(0,s*.3);
  ctx.bezierCurveTo(s*.5,-s*.42,s*1.12,s*.3,0,s);ctx.bezierCurveTo(-s*1.12,s*.3,-s*.5,-s*.42,0,s*.3);ctx.closePath();}
function strokePts(pts,w,c){ctx.strokeStyle=c;ctx.lineWidth=w;ctx.lineCap='round';ctx.lineJoin='round';
  ctx.beginPath();ctx.moveTo(pts[0][0],pts[0][1]);for(let i=1;i<pts.length;i++)ctx.lineTo(pts[i][0],pts[i][1]);ctx.stroke();}

/* ============================== SCENE: BOARD ============================== */
function drawPipe(t){
  const x=ox+(t.x+.5)*ts,y=oy+(t.y+.5)*ts,h=ts*.5;
  ctx.save();ctx.translate(x,y);ctx.rotate(t.spin);
  const pts=t.kind==='straight'?[[0,-h],[0,h]]:[[0,-h],[0,0],[h,0]];
  strokePts(pts,ts*.56,'#0e120c');
  strokePts(pts,ts*.44,'#48533d');
  strokePts(pts,ts*.22,'#333d2b');
  ctx.setLineDash([ts*.1,ts*.16]);ctx.globalAlpha=.16;
  strokePts(pts,ts*.035,'#cfe86e');ctx.setLineDash([]);ctx.globalAlpha=1;
  ctx.restore();
}
function drawGate(g){
  const x=ox+(g.x+.5)*ts,y=oy+(g.y+.5)*ts,r=ts*.4,pulse=.5+.5*Math.sin(time*2.2+g.x*3);
  ctx.fillStyle='#0b0f0a';ctx.beginPath();ctx.arc(x,y,r,0,7);ctx.fill();
  ctx.strokeStyle='#39452f';ctx.lineWidth=3;ctx.stroke();
  if(!g.closed){
    const gl=ctx.createRadialGradient(x,y,1,x,y,r*.85);
    gl.addColorStop(0,'rgba(158,240,26,'+(.35+pulse*.25)+')');gl.addColorStop(1,'rgba(158,240,26,0)');
    ctx.fillStyle=gl;ctx.beginPath();ctx.arc(x,y,r*.85,0,7);ctx.fill();
  }else{ctx.strokeStyle='#2c3a26';ctx.lineWidth=3;
    ctx.beginPath();ctx.moveTo(x-r*.45,y-r*.45);ctx.lineTo(x+r*.45,y+r*.45);
    ctx.moveTo(x+r*.45,y-r*.45);ctx.lineTo(x-r*.45,y+r*.45);ctx.stroke();}
  // badge
  const bw=ts*.42,bh=ts*.26;
  rrect(x+ts*.08,y-ts*.44,bw,bh,3);ctx.fillStyle=g.closed?'#141a12':'#1c2418';ctx.fill();
  ctx.strokeStyle='#39452f';ctx.lineWidth=1.5;ctx.stroke();
  ctx.fillStyle=g.closed?'#4a5540':'#d8e4c0';ctx.font=Math.round(ts*.17)+'px Consolas,monospace';
  ctx.textAlign='center';ctx.textBaseline='middle';
  ctx.fillText(g.closed?'✓':g.remaining,x+ts*.08+bw/2,y-ts*.44+bh/2+.5);
}
function drawTrap(t){
  const x=ox+(t.x+.5)*ts,y=oy+(t.y+.5)*ts,r=ts*.4,f=t.flash;
  ctx.save();ctx.translate(x,y);
  const gl=ctx.createRadialGradient(0,0,2,0,0,r*1.5);
  gl.addColorStop(0,'rgba(255,70,48,'+(.1+f*.35)+')');gl.addColorStop(1,'rgba(255,70,48,0)');
  ctx.fillStyle=gl;ctx.beginPath();ctx.arc(0,0,r*1.5,0,7);ctx.fill();
  const pg=ctx.createRadialGradient(-r*.3,-r*.3,2,0,0,r);
  pg.addColorStop(0,'#3a4234');pg.addColorStop(1,'#161b12');
  ctx.fillStyle=pg;ctx.beginPath();ctx.arc(0,0,r,0,7);ctx.fill();
  ctx.strokeStyle='#39452f';ctx.lineWidth=2.5;ctx.stroke();
  ctx.setLineDash([ts*.09,ts*.08]);ctx.strokeStyle='rgba(255,80,52,'+(.45+f*.4)+')';ctx.lineWidth=2.5;
  ctx.beginPath();ctx.arc(0,0,r*.78,0,7);ctx.stroke();ctx.setLineDash([]);
  const sc=1+f*.3;
  for(let i=0;i<8;i++){const a=i*Math.PI/4+Math.PI/8;
    ctx.save();ctx.rotate(a);ctx.scale(sc,sc);
    ctx.beginPath();ctx.moveTo(r*.16,-r*.07);ctx.lineTo(r*.5,0);ctx.lineTo(r*.16,r*.07);ctx.closePath();
    ctx.fillStyle='#9aa39b';ctx.fill();ctx.strokeStyle='#4a5248';ctx.lineWidth=1;ctx.stroke();
    ctx.restore();}
  ctx.fillStyle=f>0?'#ffffff':'#c8d0c4';ctx.beginPath();ctx.arc(0,0,r*.14*f+1.5,0,7);ctx.fill();
  ctx.restore();
}
function drawHub(h){
  const x=ox+(h.x+.5)*ts,y=oy+(h.y+.5)*ts,s=ts,flick=.8+.2*Math.sin(time*7)+.1*Math.sin(time*13);
  ctx.save();ctx.translate(x,y+ s*.04);
  const gl=ctx.createRadialGradient(0,0,2,0,0,s*.62);
  gl.addColorStop(0,'rgba(255,190,90,'+(.14*flick)+')');gl.addColorStop(1,'rgba(255,190,90,0)');
  ctx.fillStyle=gl;ctx.beginPath();ctx.arc(0,0,s*.62,0,7);ctx.fill();
  rrect(-s*.3,-s*.14,s*.6,s*.38,2);ctx.fillStyle='#5d452c';ctx.fill();ctx.strokeStyle='#2e2114';ctx.lineWidth=2;ctx.stroke();
  ctx.beginPath();ctx.moveTo(-s*.4,-s*.13);ctx.lineTo(0,-s*.4);ctx.lineTo(s*.4,-s*.13);ctx.closePath();
  ctx.fillStyle='#3b2c1a';ctx.fill();ctx.stroke();
  rrect(-s*.085,s*.02,s*.17,s*.24,1.5);ctx.fillStyle='#171008';ctx.fill();
  ctx.fillStyle='rgba(255,196,100,'+(.85*flick)+')';ctx.beginPath();ctx.arc(s*.18,-s*.02,s*.045,0,7);ctx.fill();
  // heart sign
  ctx.save();ctx.translate(-s*.19,-s*.03);const hp=1+.1*Math.sin(time*3.2);ctx.scale(hp,hp);
  heartPath(s*.055);ctx.fillStyle=h.flash>0?'#ffffff':'#ff4d4d';ctx.fill();
  ctx.strokeStyle='#5c130c';ctx.lineWidth=1;ctx.stroke();ctx.restore();
  if(h.flash>0){ctx.fillStyle='rgba(255,60,40,'+h.flash*.4+')';rrect(-s*.36,-s*.44,s*.72,s*.8,4);ctx.fill();}
  ctx.restore();
}
function drawZombie(z){
  const x=ox+(z.x+.5)*ts,y=oy+(z.y+.5)*ts,zd=ZTYPES[z.type],r=zd.r*ts;
  const ang=[-Math.PI/2,0,Math.PI/2,Math.PI][z.dir];
  const wob=z.st==='stuck'?rand(-.12,.12):Math.sin(z.phase)*.13;
  ctx.save();ctx.translate(x,y);
  ctx.fillStyle='rgba(0,0,0,.32)';ctx.beginPath();ctx.ellipse(0,r*.72,r*.85,r*.3,0,0,7);ctx.fill();
  if(z.st==='stuck'){ctx.setLineDash([4,5]);ctx.strokeStyle='rgba(207,232,110,.4)';ctx.lineWidth=1.5;
    ctx.beginPath();ctx.arc(0,0,r*1.25+Math.sin(time*4)*2,0,7);ctx.stroke();ctx.setLineDash([]);}
  ctx.rotate(ang+wob);ctx.translate(0,Math.sin(z.phase*2)*r*.06);
  ctx.strokeStyle=zd.dark;ctx.lineCap='round';
  // legs
  const st=Math.sin(z.phase)*(z.st==='move'?1:.15);
  ctx.lineWidth=r*.3;
  ctx.beginPath();ctx.moveTo(-r*.25,r*.2);ctx.lineTo(-r*.25+st*r*.4,r*.85);ctx.stroke();
  ctx.beginPath();ctx.moveTo(r*.25,r*.2);ctx.lineTo(r*.25-st*r*.4,r*.85);ctx.stroke();
  // arms
  const reach=z.st==='stuck'?-1.9+.3*Math.sin(z.phase*2):.55+Math.sin(z.phase*1.3)*.22;
  ctx.lineWidth=r*.27;
  ctx.beginPath();ctx.moveTo(r*.1,-r*.05);ctx.lineTo(r*.1+Math.cos(reach-.18)*r*.95,-r*.05+Math.sin(reach-.18)*r*.95);ctx.stroke();
  ctx.beginPath();ctx.moveTo(-r*.1,r*.02);ctx.lineTo(-r*.1+Math.cos(reach+.18)*r*.85,r*.02+Math.sin(reach+.18)*r*.85);ctx.stroke();
  // body
  ctx.fillStyle=zd.color;ctx.strokeStyle=zd.dark;ctx.lineWidth=r*.16;
  ctx.beginPath();ctx.arc(0,r*.05,r*.72,0,7);ctx.fill();ctx.stroke();
  // head
  ctx.beginPath();ctx.arc(r*.42,-r*.18,r*.5,0,7);ctx.fillStyle=zd.color;ctx.fill();ctx.stroke();
  if(z.type==='brute'){ctx.strokeStyle=zd.dark;ctx.lineWidth=r*.1;
    ctx.beginPath();ctx.moveTo(r*.3,-r*.45);ctx.lineTo(r*.6,-r*.28);ctx.stroke();}
  // eyes
  ctx.fillStyle=zd.eye;ctx.shadowColor=zd.eye;ctx.shadowBlur=6;
  ctx.beginPath();ctx.arc(r*.72,-r*.3,r*.09,0,7);ctx.fill();
  ctx.beginPath();ctx.arc(r*.58,-r*.42,r*.08,0,7);ctx.fill();ctx.shadowBlur=0;
  ctx.restore();
}
function drawCursorRing(x,y,sign,strong){
  const s=ts*.46;
  ctx.strokeStyle=strong?'#e6e1d2':'rgba(230,225,210,.5)';ctx.lineWidth=2.5;
  [[-1,-1],[1,-1],[-1,1],[1,1]].forEach(c=>{
    ctx.beginPath();ctx.moveTo(x+c[0]*s,y+c[0]*s*.55);ctx.lineTo(x+c[0]*s,y+c[0]*s);ctx.lineTo(x+c[0]*s*.55,y+c[1]*s);ctx.stroke();});
  ctx.save();ctx.translate(x,y);if(sign<0)ctx.scale(-1,1);
  ctx.strokeStyle='#9ef01a';ctx.lineWidth=2;
  ctx.beginPath();ctx.arc(0,0,ts*.34,-2.1,.55);ctx.stroke();
  const a=.55;ctx.fillStyle='#9ef01a';
  ctx.save();ctx.translate(Math.cos(a)*ts*.34,Math.sin(a)*ts*.34);ctx.rotate(a+Math.PI/2);
  ctx.beginPath();ctx.moveTo(0,-ts*.07);ctx.lineTo(ts*.05,ts*.04);ctx.lineTo(-ts*.05,ts*.04);ctx.closePath();ctx.fill();
  ctx.restore();ctx.restore();
}

/* ============================== SCENE: AMBIENT ============================== */
function drawFog(){for(const f of fog){
  const r=f.r*Math.max(W,H),x=f.x*W,y=f.y*H;
  const g=ctx.createRadialGradient(x,y,1,x,y,r);
  g.addColorStop(0,'rgba(140,175,140,'+f.a+')');g.addColorStop(1,'rgba(140,175,140,0)');
  ctx.fillStyle=g;ctx.beginPath();ctx.arc(x,y,r,0,7);ctx.fill();}}
function drawSpores(){ctx.fillStyle='#9ef0a8';

    ctx.fillRect(s.x*W,s.y*H,s.size,s.size);}}
ctx.globalAlpha=1;

function drawBG(){
  const g=ctx.createLinearGradient(0,0,0,H);g.addColorStop(0,'#0d140e');g.addColorStop(1,'#060a07');
  ctx.fillStyle=g;ctx.fillRect(0,0,W,H);
  if(noisePat){ctx.fillStyle=noisePat;ctx.fillRect(0,0,W,H);}
}
function drawBoardScene(){
  drawBG();
  const shx=(Math.random()*2-1)*shakeAmp,shy=(Math.random()*2-1)*shakeAmp;
  ctx.save();ctx.translate(shx,shy);
  // board frame + glow
  const gl=ctx.createRadialGradient(W/2,H/2,10,W/2,H/2,Math.max(W,H)*.7);
  gl.addColorStop(0,'rgba(120,200,80,.06)');gl.addColorStop(1,'rgba(0,0,0,0)');
  ctx.fillStyle=gl;ctx.fillRect(0,0,W,H);
  rrect(ox-9,oy-9,cols*ts+18,rows*ts+18,8);ctx.fillStyle='#0f150e';ctx.fill();
  ctx.strokeStyle='#232c1e';ctx.lineWidth=2;ctx.stroke();
  [[ox-4,oy-4],[ox+cols*ts+4,oy-4],[ox-4,oy+rows*ts+4],[ox+cols*ts+4,oy+rows*ts+4]].forEach(p=>{
    ctx.fillStyle='#2c3526';ctx.beginPath();ctx.arc(p[0],p[1],2.5,0,7);ctx.fill();});
  // slabs
  for(let y=0;y<rows;y++)for(let x=0;x<cols;x++){
    const t=tiles[y][x];rrect(ox+x*ts+2,oy+y*ts+2,ts-4,ts-4,3);
    ctx.fillStyle=t.tint;ctx.fill();ctx.strokeStyle='rgba(0,0,0,.35)';ctx.lineWidth=1;ctx.stroke();
  }
  // pipes
  for(let y=0;y<rows;y++)for(let x=0;x<cols;x++){
    const t=tiles[y][x];if(t.kind!=='ground'){t.x=x;t.y=y;drawPipe(t);}}
  // decals
  for(const d of decals){ctx.globalAlpha=d.a;ctx.fillStyle='#2f5c1c';
    for(const l of d.lobes){ctx.beginPath();ctx.arc(d.x+l.dx,d.y+l.dy,l.r,0,7);ctx.fill();}
    ctx.globalAlpha=1;}
  traps.forEach(drawTrap);gates.forEach(drawGate);hubs.forEach(drawHub);
  // zombies (depth sort)
  const zs=zombies.slice().sort((a,b)=>a.y-b.y);zs.forEach(drawZombie);
  // particles
  for(const p of parts){const k=1-p.t/p.life;
    if(p.k==='ring'){ctx.strokeStyle=p.c;ctx.globalAlpha=k;ctx.lineWidth=2.5;
      ctx.beginPath();ctx.arc(p.x,p.y,p.r0+(p.r1-p.r0)*(1-k),0,7);ctx.stroke();}
    else{ctx.fillStyle=p.c;ctx.globalAlpha=k*(p.k==='smoke'?.35:.9);
      const r=p.k==='smoke'?p.r*(1+p.t*1.6):p.r*k+.6;
      ctx.beginPath();ctx.arc(p.x,p.y,r,0,7);ctx.fill();}
    ctx.globalAlpha=1;}
  // pops
  ctx.textAlign='center';ctx.textBaseline='middle';
  for(const p of pops){const k=p.t/.95,scl=1+(1-Math.min(1,p.t*4))*.6;
    ctx.globalAlpha=1-k;ctx.fillStyle=p.c;
    ctx.font='bold '+Math.round(p.s*scl)+"px Impact,'Arial Black',sans-serif";
    ctx.fillText(p.txt,p.x,p.y-k*ts*.7);}
  ctx.globalAlpha=1;
  // hover / keyboard cursor
  const hc=(kbOn)?{x:kbx,y:kby}:(hover&&state==='playing'?hover:null);
  if(hc){const t=tiles[hc.y]&&tiles[hc.y][hc.x];
    if(t&&t.kind!=='ground')drawCursorRing(ox+(hc.x+.5)*ts,oy+(hc.y+.5)*ts,rotMode,kbOn);}
  ctx.restore();
  drawFog();drawSpores();
  if(flashRed>0){ctx.fillStyle='rgba(255,45,30,'+flashRed*.26+')';ctx.fillRect(0,0,W,H);}
  if(state==='dying'){const a=.2+.18*Math.sin(time*10);
    const rg=ctx.createRadialGradient(W/2,H/2,Math.min(W,H)*.3,W/2,H/2,Math.max(W,H)*.75);
    rg.addColorStop(0,'rgba(160,20,10,0)');rg.addColorStop(1,'rgba(160,20,10,'+a+')');
    ctx.fillStyle=rg;ctx.fillRect(0,0,W,H);}
}
function drawAttract(){
  const g=ctx.createLinearGradient(0,0,0,H);g.addColorStop(0,'#0a110c');g.addColorStop(.62,'#141b11');g.addColorStop(1,'#080c08');
  ctx.fillStyle=g;ctx.fillRect(0,0,W,H);
  if(noisePat){ctx.fillStyle=noisePat;ctx.fillRect(0,0,W,H);}
  // stars
  for(const s of stars){ctx.globalAlpha=.25+.45*Math.abs(Math.sin(time*s.tw+s.ph));
    ctx.fillStyle='#d8e4c0';ctx.fillRect(s.x*W,s.y*H*.55,s.s,s.s);}
  ctx.globalAlpha=1;
  // moon
  const mx=W*.76,my=H*.2,mr=Math.min(W,H)*.09;
  const mg=ctx.createRadialGradient(mx,my,mr*.4,mx,my,mr*3);
  mg.addColorStop(0,'rgba(215,225,190,.5)');mg.addColorStop(.35,'rgba(215,225,190,.08)');mg.addColorStop(1,'rgba(215,225,190,0)');
  ctx.fillStyle=mg;ctx.beginPath();ctx.arc(mx,my,mr*3,0,7);ctx.fill();
  ctx.fillStyle='#cfd6b8';ctx.beginPath();ctx.arc(mx,my,mr,0,7);ctx.fill();
  ctx.fillStyle='rgba(120,135,100,.4)';
  [[-.3,-.2,.22],[.25,.15,.16],[-.05,.32,.13]].forEach(c=>{ctx.beginPath();ctx.arc(mx+c[0]*mr,my+c[1]*mr,c[2]*mr,0,7);ctx.fill();});
  // ground + fence
  const hy=H*.7;
  ctx.fillStyle='#0b0f0a';ctx.fillRect(0,hy,W,H-hy);
  ctx.strokeStyle='#1e2618';ctx.lineWidth=3;
  for(let x=14;x<W;x+=58){ctx.beginPath();ctx.moveTo(x,hy-4);ctx.lineTo(x,hy+30);ctx.stroke();}
  ctx.lineWidth=2.5;
  ctx.beginPath();ctx.moveTo(0,hy+6);ctx.lineTo(W,hy+6);ctx.stroke();
  ctx.beginPath();ctx.moveTo(0,hy+20);ctx.lineTo(W,hy+20);ctx.stroke();
  // silhouette walkers
  for(const w of walkers)drawWalkSil(w.x,H*.86,w.s,w.ph,w.dir);
  drawFog();drawSpores();
}

/* ============================== LOOP / RESIZE ============================== */
function fit(){
  DPR=Math.min(2,window.devicePixelRatio||1);
  W=stage.clientWidth;H=stage.clientHeight;
  cv.width=Math.max(1,W*DPR);cv.height=Math.max(1,H*DPR);
  if(cols&&rows){
    ts=Math.max(14,Math.min(96,Math.floor(Math.min((W-28)/cols,(H-30)/rows))));
    ox=Math.round((W-ts*cols)/2);oy=Math.round((H-ts*rows)/2)+2;
  }
}
function makeNoise(){
  const c=document.createElement('canvas');c.width=c.height=160;const x=c.getContext('2d');
  for(let i=0;i<900;i++){x.fillStyle='rgba(255,255,255,'+rand(.008,.03)+')';
    x.fillRect(Math.random()*160,Math.random()*160,rand(1,2.4),rand(1,2.4));}
  for(let i=0;i<500;i++){x.fillStyle='rgba(0,0,0,'+rand(.03,.09)+')';
    x.fillRect(Math.random()*160,Math.random()*160,rand(1,3),rand(1,3));}
  noisePat=ctx.createPattern(c,'repeat');
}
function initAmbient(){
  for(let i=0;i<4;i++)fog.push({x:Math.random(),y:rand(.2,.9),r:rand(.22,.5),vx:rand(.008,.02)*(Math.random()<.5?-1:1),a:rand(.04,.08)});
  for(let i=0;i<26;i++)spores.push({x:Math.random(),y:Math.random(),spd:rand(.02,.05),ph:rand(0,9),size:rand(1,2.6)});
  for(let i=0;i<42;i++)stars.push({x:Math.random(),y:Math.random(),tw:rand(.5,2),ph:rand(0,9),s:rand(1,2)});
  for(let i=0;i<4;i++)walkers.push({x:Math.random()*W,spd:rand(18,42)*(Math.random()<.5?-1:1)*0+rand(16,38),dir:Math.random()<.5?-1:1,s:rand(.7,1.3),ph:rand(0,9)});
}
function frame(now){
  const dt=Math.min(.05,(now-last)/1000);last=now;
  if(state!=='paused')update(dt);
  ctx.setTransform(DPR,0,0,DPR,0,0);
  if(state==='menu'||state==='gameover'||state==='victory')drawAttract();
  else drawBoardScene();
  requestAnimationFrame(frame);
}

makeNoise();initAmbient();updateHUD();fit();

})();
