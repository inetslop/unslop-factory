  // fx decay
  for(let i=S.beams.length-1;i>=0;i--){S.beams[i].t-=dt;if(S.beams[i].t<=0)S.beams.splice(i,1);}
  for(let i=S.parts.length-1;i>=0;i--){const p=S.parts[i];p.t+=dt;if(p.t>p.max){S.parts.splice(i,1);continue;}p.x+=p.vx*dt;p.y+=p.vy*dt;p.vy+=(p.g||0)*dt;if(!p.smoke)p.vx*=Math.max(0,1-2.2*dt);}
  for(let i=S.rings.length-1;i>=0;i--){S.rings[i].t+=dt;if(S.rings[i].t>S.rings[i].max)S.rings.splice(i,1);}
  for(let i=S.floats.length-1;i>=0;i--){const f=S.floats[i];f.t+=dt;f.y-=dt*.85;if(f.t>f.max)S.floats.splice(i,1);}
  for(let i=S.pending.length-1;i>=0;i--){const p=S.pending[i];p.t-=dt;if(p.t<=0){S.pending.splice(i,1);detonate(p.r,p.c);}}
}
