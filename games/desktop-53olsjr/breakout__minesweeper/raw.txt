function hitTile(t,b){
  if(t.state==='destroyed'||t.state==='exploded')return;
  if(G.flagMode){
    t.flagged=!t.flagged;t.revealAt=0;SFX.flag(t.flagged);
    pop(GRID_X+t.c*(T+GAP)+T/2,GRID_Y+t.r*(T+GAP),'FLAGGED','#e0141c',12);return;
  }
  if(t.flagged){t.flagged=false;SFX.flag(false);return;}
  if(t.state==='hidden'){
    if(!G.firstHit){G.firstHit=true;if(t.mine)moveMineAway(t);}
    if(t.mine){explodeAt(t);b.dead=true;loseLife('mine');return;}
    t.revealAt=0;reveal(t,false);
  }else{destroyTile(t,'ball');}
}
function spawnPower(x,y){
  const r=Math.random();
  const type=r<.24?'wide':r<.46?'slow':r<.66?'multi':r<.86?'scan':'life';
  G.pows.push({bx:x,x,y,vy:150,type,t:rand(0,6)});
}
function applyPower(type){
  SFX.power();pop(PAD.x,PAD.y-34,POWTXT[type],POWCOL[type],17);
  if(type==='wide')G.status.wide=12;
  else if(type==='slow')G.status.slow=8;
  else if(type==='scan')G.status.scan=7;
  else if(type==='life'){G.lives=Math.min(5,G.lives+1);updateBalls();}
  else if(type==='multi'){
    for(const a of[-.9,.9]){
      if(G.balls.length>=6)break;
      const sp=levelBase();
      G.balls.push({x:PAD.x,y:PAD.y-BR-4,vx:Math.sin(a)*sp,vy:-Math.cos(a)*sp,speed:levelBase(),boost:1,stuck:false,trail:[],dead:false});
    }
  }
}
function checkClear(){
  if(G.state!=='play')return;
  for(const t of G.tiles)if(!t.mine&&t.state!=='destroyed')return;
  levelClear();
}
function levelClear(){
  G.state='levelclear';G.clearT=2.1;
  const bonus=250*G.level;addScore(bonus);
  pop(LW/2,LH/2,'+'+bonus+' BONUS','#7dffa8',24);
  setFace('cool',1.9);SFX.clear();
  clearBonus.textContent='+'+bonus+' LEVEL BONUS';
  clearOvl.classList.add('show');
}
function loseLife(reason){
  if(G.state!=='play')return;
  G.lives--;G.combo=0;updateBalls();
  G.lastDeath=reason;
  if(reason==='mine'){setFace('dead',1);}else{setFace('wow',.9);SFX.lifeLost();}
  if(G.lives<=0){G.state='dying';G.dieT=.95;G.balls=[];}
  else G.balls.push(makeBall(true));
}
function gameOver(){
  G.state='gameover';setFace('dead',999);SFX.over();
  const nb=G.score>G.best;
  if(nb){G.best=G.score;saveBest(G.best);}
  overReason.textContent=G.lastDeath==='mine'?'A mine got you. Click that again, never.':'Out of balls. The board still has secrets.';
  finalLed.textContent=pad7(G.score);bestLed.textContent=pad7(G.best);
  newBest.classList.toggle('hide',!nb);
  overOvl.classList.add('show');
}
function newRun(){
  menuOvl.classList.remove('show');overOvl.classList.remove('show');pauseOvl.classList.remove('show');clearOvl.classList.remove('show');
  G.score=0;G.shownScore=0;G.lives=3;G.level=1;genLevel(1);
  G.state='play';updateBalls();SFX.start();
}
function nextLevel(){G.level++;genLevel(G.level);G.state='play';clearOvl.classList.remove('show');}
function toMenu(){
  overOvl.classList.remove('show');pauseOvl.classList.remove('show');clearOvl.classList.remove('show');
  G.state='menu';G.balls=[];genLevel(1);setFace('smile',0);
  if(G.best>0){menuBest.textContent=pad7(G.best);menuBestWrap.classList.remove('hide');}
  menuOvl.classList.add('show');
}
function togglePause(){
  if(G.state==='play'){G.state='paused';pauseOvl.classList.add('show');}
  else if(G.state==='paused'){G.state='play';pauseOvl.classList.remove('show');}
}
function launch(){
  let did=false;
  for(const b of G.balls){
    if(!b.stuck)continue;did=true;
    const a=-Math.PI/2+rand(-.35,.35),sp=levelBase();
    b.vx=Math.cos(a)*sp;b.vy=Math.sin(a)*sp;b.stuck=false;
  }
  if(did){SFX.launch();}
}
function updateBalls(){
  const dots=ballsBox.children;
  for(let i=0;i<dots.length;i++)dots[i].classList.toggle('on',i<G.lives);
}

function stepBall(b,dt){
  const sp=Math.min(780,b.speed*b.boost*(G.status.slow>0?.62:1));
  const cur=Math.hypot(b.vx,b.vy)||1;b.vx*=sp/cur;b.vy*=sp/cur;
  const steps=Math.max(1,Math.ceil(sp*dt/9));
  for(let i=0;i<steps;i++){
    b.x+=b.vx*dt/steps;b.y+=b.vy*dt/steps;
    if(b.x-BR<0){b.x=BR;b.vx=Math.abs(b.vx);SFX.wall();}
    if(b.x+BR>LW){b.x=LW-BR;b.vx=-Math.abs(b.vx);SFX.wall();}
    if(b.y-BR<0){b.y=BR;b.vy=Math.abs(b.vy);SFX.wall();}
    // tiles
    const c0=clamp(Math.floor((b.x-BR-GRID_X)/(T+GAP)),0,COLS-1);
    const c1=clamp(Math.floor((b.x+BR-GRID_X)/(T+GAP)),0,COLS-1);
    const r0=clamp(Math.floor((b.y-BR-GRID_Y)/(T+GAP)),0,G.rows-1);
    const r1=clamp(Math.floor((b.y+BR-GRID_Y)/(T+GAP)),0,G.rows-1);
    let hit=false;
    for(let r=r0;r<=r1&&!hit;r++)for(let c=c0;c<=c1&&!hit;c++){
      const t=G.tiles[r*COLS+c];
      if(t.state!=='hidden'&&t.state!=='revealed')continue;
      const rx=GRID_X+c*(T+GAP),ry=GRID_Y+r*(T+GAP);
      const nx=clamp(b.x,rx,rx+T),ny=clamp(b.y,ry,ry+T);
      const dx=b.x-nx,dy=b.y-ny;
      if(dx*dx+dy*dy>BR*BR)continue;
      hit=true;
      if(Math.abs(dx)>Math.abs(dy)){b.vx=dx>=0?Math.abs(b.vx):-Math.abs(b.vx);b.x=nx+(dx>=0?BR+.2:-BR-.2);}
      else{b.vy=dy>=0?Math.abs(b.vy):-Math.abs(b.vy);b.y=ny+(dy>=0?BR+.2:-BR-.2);}
      if(Math.abs(b.vy)<70)b.vy=(b.vy<=0?-1:1)*85;
      const m=Math.hypot(b.vx,b.vy)||1;b.vx*=sp/m;b.vy*=sp/m;
      hitTile(t,b);
    }
    if(G.state!=='play')return;
    // paddle
    if(b.vy>0&&b.y+BR>=PAD.y&&b.y<PAD.y+PAD.h&&Math.abs(b.x-PAD.x)<=PAD.w/2+BR*.7){
      const t=clamp((b.x-PAD.x)/(PAD.w/2),-1,1),a=t*1.0;
      b.vx=Math.sin(a)*sp;b.vy=-Math.cos(a)*sp;b.y=PAD.y-BR;
      b.boost=Math.min(1.4,b.boost+.03);G.combo=0;SFX.paddle();
    }
    if(b.y-BR>LH){b.dead=true;break;}
  }
  b.trail.push({x:b.x,y:b.y});if(b.trail.length>11)b.trail.shift();
}
function update(dt){
  G.time+=dt;
  if(G.face.t>0)G.face.t-=dt;
  G.nextBlink-=dt;if(G.nextBlink<0){G.blinkT=.12;G.nextBlink=rand(2.4,4.6);}
  if(G.blinkT>0)G.blinkT-=dt;
  G.shake=Math.max(0,G.shake-58*dt);
  G.flash=Math.max(0,G.flash-2.3*dt);
  G.comboPop=Math.max(0,G.comboPop-dt);
  // fx (always)
  for(const p of G.parts){p.t+=dt;p.x+=p.vx*dt;p.y+=p.vy*dt;p.vy+=p.g*dt;}
  G.parts=G.parts.filter(p=>p.t<p.life);
  for(const p of G.pops){p.t+=dt;p.y-=42*dt;}
  G.pops=G.pops.filter(p=>p.t<p.life);
  for(const r of G.rings){r.r+=r.vr*dt;r.a-=2.1*dt;}
  G.rings=G.rings.filter(r=>r.a>0);
  for(const t of G.tiles)if(t.anim<1)t.anim=Math.min(1,t.anim+dt*6);
  // game logic
  if(G.state==='play'){
    for(const k in G.status)G.status[k]=Math.max(0,G.status[k]-dt);
    const tw=G.status.wide>0?235:150;
    PAD.w+=clamp(tw-PAD.w,-420*dt,420*dt);
    if(keys.l||keys.r){PAD.x+=(keys.r-keys.l)*980*dt;mouseT=null;}
    else if(mouseT!=null)PAD.x+=(mouseT-PAD.x)*Math.min(1,dt*22);
    PAD.x=clamp(PAD.x,PAD.w/2+6,LW-PAD.w/2-6);
    // cascade reveals
    for(const t of G.tiles)if(t.revealAt>0&&G.time>=t.revealAt){t.revealAt=0;reveal(t,true);}
    // powerups
    for(const p of G.pows){
      p.t+=dt;p.y+=p.vy*dt;p.x=p.bx+Math.sin(p.t*3.1)*7;
      if(p.y+13>=PAD.y&&p.y-13<=PAD.y+PAD.h&&Math.abs(p.x-PAD.x)<=PAD.w/2+10){applyPower(p.type);p.dead=true;}
      if(p.y>LH+24)p.dead=true;
    }
    G.pows=G.pows.filter(p=>!p.dead);
    // balls
    for(const b of G.balls){
      if(b.dead)continue;
      if(b.stuck){b.x=PAD.x;b.y=PAD.y-BR-2;b.trail.length=0;continue;}
      stepBall(b,dt);
    }
    const lost=G.balls.filter(b=>b.dead);
    G.balls=G.balls.filter(b=>!b.dead);
    if(lost.length&&G.state==='play'){if(G.balls.length===0)loseLife('fall');}
  }
  else if(G.state==='levelclear'){
    for(const t of G.tiles)if(t.revealAt>0&&G.time>=t.revealAt){t.revealAt=0;reveal(t,true);}
    G.clearT-=dt;if(G.clearT<=0)nextLevel();
  }
  else if(G.state==='dying'){G.dieT-=dt;if(G.dieT<=0)gameOver();}
  // hud score lerp
  const d=G.score-G.shownScore;
  G.shownScore+=d*Math.min(1,dt*8);
  if(Math.abs(d)<.6)G.shownScore=G.score;
}

/* ============================== drawing ============================== */
function bevelOut(x,y,w,h,t){
  ctx.fillStyle='#ffffff';ctx.fillRect(x,y,w,t);ctx.fillRect(x,y,t,h);
  ctx.fillStyle='#8b8b8b';ctx.fillRect(x+w-t,y,t,h);ctx.fillRect(x,y+h-t,w,t);
}
function bevelIn(x,y,w,h,t){
  ctx.fillStyle='#07494b';ctx.fillRect(x,y,w,t);ctx.fillRect(x,y,t,h);
  ctx.fillStyle='#3fbdbd';ctx.fillRect(x+w-t,y,t,h);ctx.fillRect(x,y+h-t,w,t);
}
function drawMine(cx,cy,r){
  ctx.strokeStyle='#1c1c1c';ctx.fillStyle='#1c1c1c';
  ctx.lineWidth=Math.max(2,r*.34);ctx.lineCap='round';
  const L=r*1.5;
  ctx.beginPath();
  ctx.moveTo(cx-L,cy);ctx.lineTo(cx+L,cy);
  ctx.moveTo(cx,cy-L);ctx.lineTo(cx,cy+L);
  const d=L*.62;
  ctx.moveTo(cx-d,cy-d);ctx.lineTo(cx+d,cy+d);
  ctx.moveTo(cx+d,cy-d);ctx.lineTo(cx-d,cy+d);
  ctx.stroke();
  ctx.beginPath();ctx.arc(cx,cy,r,0,Math.PI*2);ctx.fill();
  ctx.fillStyle='rgba(255,255,255,.75)';
  ctx.beginPath();ctx.arc(cx-r*.32,cy-r*.32,r*.18,0,Math.PI*2);ctx.fill();
}
function drawFlag(x,y,s){
  ctx.strokeStyle='#2b2b2b';ctx.lineWidth=2;ctx.lineCap='round';
  ctx.beginPath();ctx.moveTo(x+s*.35,y+s*.62);ctx.lineTo(x+s*.35,y-s*.7);ctx.stroke();
  ctx.fillStyle='#e0141c';
  ctx.beginPath();ctx.moveTo(x+s*.35,y-s*.7);ctx.lineTo(x-s*.55,y-s*.28);ctx.lineTo(x+s*.35,y-s*.12);ctx.closePath();ctx.fill();
  ctx.fillStyle='#2b2b2b';ctx.fillRect(x-s*.42,y+s*.6,s*.95,s*.16);
}
function drawTile(t){
  const x=GRID_X+t.c*(T+GAP),y=GRID_Y+t.r*(T+GAP);
  if(t.state==='destroyed'){
    ctx.fillStyle='#0a6666';ctx.fillRect(x,y,T,T);bevelIn(x,y,T,T,2);return;
  }
  if(t.state==='exploded'){
    ctx.fillStyle='#ff4632';ctx.fillRect(x,y,T,T);bevelOut(x,y,T,T,3);
    drawMine(x+T/2,y+T/2,T*.27);return;
  }
  const cx=x+T/2,cy=y+T/2;
  if(t.state==='hidden'){
    ctx.fillStyle='#c9c9c9';ctx.fillRect(x,y,T,T);bevelOut(x,y,T,T,3);
    if(t.flagged)drawFlag(cx,cy,T*.42);
  }else{
    const sy=.35+.65*(1-Math.pow(1-t.anim,3));
    ctx.save();ctx.translate(cx,cy);ctx.scale(1,sy);
    ctx.fillStyle='#cfcfcf';ctx.fillRect(-T/2,-T/2,T,T);
    ctx.strokeStyle='#8b8b8b';ctx.lineWidth=2;
    ctx.beginPath();ctx.moveTo(-T/2+1.5,T/2-1);ctx.lineTo(-T/2+1.5,-T/2+1.5);ctx.lineTo(T/2-1,-T/2+1.5);ctx.stroke();
    if(t.count>0){
      ctx.fillStyle=NUMCOL[t.count];
      ctx.font='bold '+Math.round(T*.56)+'px "Courier New",monospace';
      ctx.textAlign='center';ctx.textBaseline='middle';
      ctx.fillText(String(t.count),0,2);
    }
    if(t.flagged)drawFlag(0,0,T*.42);
    ctx.restore();
  }
}
function chip(x,y,label,col,frac){
  const w=86,h=21;
  ctx.fillStyle='#bcbcbc';ctx.fillRect(x,y,w,h);bevelIn(x,y,w,h,2);
  ctx.fillStyle=col;ctx.font='bold 10px Tahoma,Verdana,sans-serif';
  ctx.textAlign='left';ctx.textBaseline='middle';
  ctx.fillText(label,x+7,y+h/2+.5);
  ctx.fillStyle='#7c7c7c';ctx.fillRect(x+w-34,y+h-7,26,4);
  ctx.fillStyle=col;ctx.fillRect(x+w-34,y+h-7,26*clamp(frac,0,1),4);
}
function draw(){
  const dpr=view.dpr;
  ctx.setTransform(dpr,0,0,dpr,0,0);
  ctx.clearRect(0,0,cv.width/dpr,cv.height/dpr);
  const shx=(Math.random()-.5)*G.shake,shy=(Math.random()-.5)*G.shake;
  ctx.translate(view.ox+shx,view.oy+shy);ctx.scale(view.s,view.s);
  // frame
  ctx.strokeStyle='#085a5a';ctx.lineWidth=2;ctx.strokeRect(1,1,LW-2,LH-2);
  ctx.strokeStyle='rgba(90,220,220,.35)';ctx.strokeRect(4,4,LW-8,LH-8);
  // tiles
  for(const t of G.tiles)drawTile(t);
  // scan glow over hidden mines
  if(G.status.scan>0){
    const p=.4+.3*Math.sin(G.time*7);
    for(const t of G.tiles)if(t.mine&&t.state==='hidden'){
      const cx=GRID_X+t.c*(T+GAP)+T/2,cy=GRID_Y+t.r*(T+GAP)+T/2;
      ctx.fillStyle='rgba(255,60,50,'+(p*.5).toFixed(3)+')';
      ctx.beginPath();ctx.arc(cx,cy,T*.3,0,Math.PI*2);ctx.fill();
      ctx.fillStyle='rgba(255,230,220,'+p.toFixed(3)+')';
      ctx.beginPath();ctx.arc(cx,cy,T*.09,0,Math.PI*2);ctx.fill();
    }
  }
  // powerups
  for(const p of G.pows){
    const s=26,x=p.x-s/2,y=p.y-s/2;
    ctx.fillStyle='#cfcfcf';ctx.fillRect(x,y,s,s);bevelOut(x,y,s,s,2);
    ctx.textAlign='center';ctx.textBaseline='middle';
    if(p.type==='scan'){
      ctx.strokeStyle='#141414';ctx.lineWidth=2;
      ctx.beginPath();ctx.arc(p.x,p.y,7,0,Math.PI*2);ctx.stroke();
      ctx.fillStyle='#d90f14';ctx.beginPath();ctx.arc(p.x,p.y,2.6,0,Math.PI*2);ctx.fill();
    }else if(p.type==='life'){
      ctx.fillStyle='#d90f14';const hx=p.x,hy=p.y+2,r=5;
      ctx.beginPath();ctx.moveTo(hx,hy+r);
      ctx.bezierCurveTo(hx-r*1.6,hy-r*.4,hx-r*.8,hy-r*1.3,hx,hy-r*.4);
      ctx.bezierCurveTo(hx+r*.8,hy-r*1.3,hx+r*1.6,hy-r*.4,hx,hy+r);ctx.fill();
    }else{
      ctx.fillStyle=POWCOL[p.type];ctx.font='bold 15px "Courier New",monospace';
      ctx.fillText(p.type[0].toUpperCase(),p.x,p.y+1);
    }
  }
  // particles
  for(const p of G.parts){
    ctx.globalAlpha=1-p.t/p.life;ctx.fillStyle=p.col;
    ctx.fillRect(p.x-p.size/2,p.y-p.size/2,p.size,p.size);
  }
  ctx.globalAlpha=1;
  // rings
  for(const r of G.rings){
    ctx.strokeStyle='rgba('+r.col+','+Math.max(0,r.a).toFixed(3)+')';ctx.lineWidth=r.wd;
    ctx.beginPath();ctx.arc(r.x,r.y,r.r,0,Math.PI*2);ctx.stroke();
  }
  // balls
  if(G.state!=='menu'){
    for(const b of G.balls){
      for(let i=0;i<b.trail.length;i++){
        const tr=b.trail[i],a=(i/b.trail.length)*.3;
        ctx.fillStyle='rgba(190,235,255,'+a.toFixed(3)+')';
        ctx.beginPath();ctx.arc(tr.x,tr.y,BR*(.35+.6*i/b.trail.length),0,Math.PI*2);ctx.fill();
      }
      ctx.save();
      ctx.shadowColor='rgba(130,220,255,.85)';ctx.shadowBlur=12;
      const g=ctx.createRadialGradient(b.x-3,b.y-4,1,b.x,b.y,BR);
      g.addColorStop(0,'#ffffff');g.addColorStop(.7,'#e8f2f6');g.addColorStop(1,'#b9c6cd');
      ctx.fillStyle=g;ctx.beginPath();ctx.arc(b.x,b.y,BR,0,Math.PI*2);ctx.fill();
      ctx.restore();
      ctx.strokeStyle='#3a4249';ctx.lineWidth=2;
      ctx.beginPath();ctx.arc(b.x,b.y,BR,0,Math.PI*2);ctx.stroke();
    }
  }
  // paddle
  if(G.state!=='menu'){
    const px=PAD.x-PAD.w/2,py=PAD.y;
    ctx.fillStyle=G.flagMode?'#d9b3ad':'#cfcfcf';
    ctx.fillRect(px,py,PAD.w,PAD.h);bevelOut(px,py,PAD.w,PAD.h,3);
    if(G.flagMode){ctx.fillStyle='#e0141c';ctx.fillRect(px+3,py,PAD.w-6,2.5);}
    if(G.flagMode)drawFlag(PAD.x,py+PAD.h/2,10);
    const stuck=G.balls.some(b=>b.stuck);
    if(stuck&&G.state==='play'){

  const blink=G.blinkT>0;
  if(s==='dead'){
    c.lineWidth=2.2;
    for(const ex of[cx-6,cx+6]){
      c.beginPath();c.moveTo(ex-3.4,cy-7.4);c.lineTo(ex+3.4,cy-.6);
      c.moveTo(ex+3.4,cy-7.4);c.lineTo(ex-3.4,cy-.6);c.stroke();
    }
    c.beginPath();c.arc(cx,cy+10,6.5,Math.PI*1.15,Math.PI*1.85);c.lineWidth=2.2;c.stroke();
  }else if(s==='cool'){
    c.fillRect(cx-11.5,cy-7.5,9,6);c.fillRect(cx+2.5,cy-7.5,9,6);
    c.lineWidth=2;c.beginPath();c.moveTo(cx-2.5,cy-4.5);c.lineTo(cx+2.5,cy-4.5);c.stroke();
    c.beginPath();c.arc(cx,cy+4.5,7,Math.PI*.18,Math.PI*.82);c.lineWidth=2;c.stroke();
  }else{
    if(blink){
      c.lineWidth=2;

      c.moveTo(cx+3.6,cy-4);c.lineTo(cx+8.4,cy-4);c.stroke();

    }
  }
}
