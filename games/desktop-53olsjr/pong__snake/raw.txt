function gtext(t,cx,y,u,color,alpha,align){
  t=(''+t).toUpperCase();

  const wFull=t.length*6*u-u;
  let x=(align==='center')?cx-wFull/2:(align==='right'?cx-wFull:cx);
  ctx.globalAlpha=alpha;ctx.fillStyle=color;
  const g=Math.max(0.4,u*0.1);
  for(let i=0;i<t.length;i++){

  }

let AC=null,muted=false,unlocked=false;
function ac(){if(!AC){try{AC=new (window.AudioContext||window.webkitAudioContext)();}catch(e){}}
  if(AC&&AC.state==='suspended')AC.resume();return AC;}
function beep(f,d,type,v,dl,slideTo){
  if(!unlocked||muted)return;const a=ac();if(!a)return;
  d=d||0.07;type=type||'square';v=v||0.14;dl=dl||0;
  const t0=a.currentTime+dl,o=a.createOscillator(),g=a.createGain();
  o.type=type;o.frequency.setValueAtTime(f,t0);
  if(slideTo)o.frequency.exponentialRampToValueAtTime(Math.max(30,slideTo),t0+d);
  g.gain.setValueAtTime(v,t0);g.gain.exponentialRampToValueAtTime(0.0001,t0+d);
  o.connect(g);g.connect(a.destination);o.start(t0);o.stop(t0+d+0.03);
}
const sPaddle=o=>beep(280+520*Math.abs(o),0.05,'square',0.16);
const sWall=()=>beep(150,0.04,'triangle',0.1);
const sEat=()=>{beep(523,0.06,'square',0.16);beep(784,0.07,'square',0.16,0.06);beep(1047,0.09,'square',0.14,0.12);};
const sOver=()=>[392,330,262,196].forEach((f,i)=>beep(f,0.14,'square',0.16,i*0.13));
const sStart=()=>[523,659,784,1047].forEach((f,i)=>beep(f,0.08,'square',0.15,i*0.07));
const sLevel=()=>{beep(880,0.06,'square',0.12);beep(1318,0.09,'square',0.12,0.06);};
const sDeny=()=>beep(90,0.04,'square',0.05);

function resize(){
  DPR=Math.min(2,window.devicePixelRatio||1);
  W=window.innerWidth;H=window.innerHeight;minD=Math.min(W,H);
  cv.width=Math.round(W*DPR);cv.height=Math.round(H*DPR);
  ctx.setTransform(DPR,0,0,DPR,0,0);
  frame=Math.max(12,Math.round(minD*0.03));
  bw=Math.max(4,Math.round(frame*0.42));
  cx0=frame+bw/2;cy0=frame+bw/2;cw=W-2*(frame+bw/2);ch=H-2*(frame+bw/2);
  cell=Math.max(15,Math.floor(minD/26));
  cols=Math.max(10,Math.floor((W-frame*2-30)/cell));
  rows=Math.max(8,Math.floor((H-frame*2-70)/cell));
  ox=(W-cols*cell)/2;oy=(H-rows*cell)/2+cell*0.6;
  ball.sz=clamp(cell*0.4,7,14);
  padH=clamp(minD*0.16,56,120);padW=Math.max(8,frame*0.5);
  layoutPads();buildBg();
  for(const s of snake){s.x=((s.x%cols)+cols)%cols;s.y=((s.y%rows)+rows)%rows;}
  ball.x=clamp(ball.x,cx0+ball.sz/2,cx0+cw-ball.sz/2);
  ball.y=clamp(ball.y,cy0+ball.sz/2,cy0+ch-ball.sz/2);
}
function layoutPads(){
  pads=[{side:'L',x:cx0+14,w:padW,h:padH,cy:H/2},
        {side:'R',x:cx0+cw-14-padW,w:padW,h:padH,cy:H/2}];
  for(const p of pads)p.cy=clamp(p.cy,cy0+p.h/2+2,cy0+ch-p.h/2-2);
}

/* ---------- game setup ---------- */
function makeSnake(len){
  const hx=Math.floor(cols*0.3),hy=Math.floor(rows/2),a=[];
  for(let i=0;i<len;i++)a.push({x:hx-i,y:hy});return a;
}
function serveBall(){
  const base=Math.max(190,minD*0.42);
  ball.speed=Math.min(base*1.85,base+score*6);
  ball.x=W/2;ball.y=H/2;
  const hx=ox+(snake[0].x+0.5)*cell;
  const right=hx<W/2?1:-1;
  const a=(0.42+Math.random()*0.55)*(Math.random()<0.5?1:-1);
  ball.vx=Math.cos(a)*ball.speed*right;
  ball.vy=Math.sin(a)*ball.speed;
  ball.trail.length=0;
}
function startGame(){
  score=0;newBest=false;grow=0;dirQueue=[];parts.length=0;pops.length=0;
  snake=makeSnake(4);preStep=snake.map(s=>({x:s.x,y:s.y}));
  dir={x:1,y:0};stepT=0;serveBall();state='playing';sStart();
}
function resetAttract(){
  grow=0;dirQueue=[];snake=makeSnake(5);preStep=snake.map(s=>({x:s.x,y:s.y}));
  dir={x:1,y:0};stepT=0;serveBall();
}

/* ---------- snake logic ---------- */
function queueDir(x,y){
  if(!x&&!y)return;
  const ref=dirQueue.length?dirQueue[dirQueue.length-1]:dir;
  if(x===-ref.x&&y===-ref.y){sDeny();return;}
  if(x===ref.x&&y===ref.y)return;
  if(dirQueue.length<3)dirQueue.push({x,y});
}
function step(){
  while(dirQueue.length){const d=dirQueue.shift();
    if(!(d.x===-dir.x&&d.y===-dir.y)){dir=d;break;}}
  const hx=(snake[0].x+dir.x+cols)%cols,hy=(snake[0].y+dir.y+rows)%rows;
  const body=grow>0?snake:snake.slice(0,-1);
  for(const s of body)if(s.x===hx&&s.y===hy){die();return;}
  preStep=snake.map(s=>({x:s.x,y:s.y}));
  snake.unshift({x:hx,y:hy});
  if(grow>0)grow--;else snake.pop();
}
function aiPick(){
  const head=snake[0];
  const bx=clamp(Math.floor((ball.x-ox)/cell),0,cols-1);
  const by=clamp(Math.floor((ball.y-oy)/cell),0,rows-1);
  let bd=1e9,bestD=null;
  for(const d of [{x:1,y:0},{x:-1,y:0},{x:0,y:1},{x:0,y:-1}]){
    if(d.x===-dir.x&&d.y===-dir.y)continue;
    const nx=(head.x+d.x+cols)%cols,ny=(head.y+d.y+rows)%rows;
    let dist=Math.abs(nx-bx)+Math.abs(ny-by);
    if(snake.some((s,i)=>i<snake.length-1&&s.x===nx&&s.y===ny))dist+=1000;
    if(dist<bd){bd=dist;bestD=d;}
  }
  if(bestD)dir=bestD;
}

/* ---------- ball / paddles ---------- */
function updatePads(dt){
  const sp=(state==='menu'?0.75:Math.min(1.9,1+score*0.04))*minD*0.5;
  for(const p of pads){
    const threat=(p.side==='L'&&ball.vx<0)||(p.side==='R'&&ball.vx>0);
    let target=H/2+Math.sin(tNow*0.7+p.x)*30;
    if(threat)target=ball.y;
    p.cy+=clamp(target-p.cy,-sp*dt,sp*dt);
    p.cy=clamp(p.cy,cy0+p.h/2+2,cy0+ch-p.h/2-2);
  }
}
function hitPad(p,away){
  const off=clamp((ball.y-p.cy)/(p.h/2),-1,1);
  ball.speed=Math.min(minD*0.95,ball.speed*1.045+2);
  const a=off*1.0;
  ball.vx=Math.cos(a)*ball.speed*away;
  ball.vy=Math.sin(a)*ball.speed;
  if(away>0)ball.x=p.x+p.w+ball.sz/2+0.5;else ball.x=p.x-ball.sz/2-0.5;
  sPaddle(off);shake=Math.max(shake,1.6);
  const fx=away>0?p.x+p.w:p.x;
  for(let i=0;i<7;i++)parts.push({x:fx,y:ball.y+(Math.random()-0.5)*18,
    vx:away*(60+Math.random()*200),vy:(Math.random()-0.5)*260,life:0.5,size:3,color:'#fff'});
}
function spark(x,y,n){
  for(let i=0;i<n;i++)parts.push({x,y,vx:(Math.random()-0.5)*240,vy:(Math.random()-0.5)*240,
    life:0.4,size:2+Math.random()*2,color:'#cfd8ea'});
}
function eat(){
  score++;if(score>best){best=score;newBest=true;saveBest();}
  grow+=1;popT=1;shake=Math.max(shake,4);flashW=0.15;
  const hdx=ox+(snake[0].x+0.5)*cell,hdy=oy+(snake[0].y+0.5)*cell;
  for(let i=0;i<26;i++){const a=Math.random()*Math.PI*2,sp2=40+Math.random()*220;
    parts.push({x:ball.x,y:ball.y,vx:Math.cos(a)*sp2,vy:Math.sin(a)*sp2,life:1,
      size:2+Math.random()*3,color:Math.random()<0.5?'#ffffff':'#ffd23f'});}
  parts.push({ring:true,x:ball.x,y:ball.y,r:6,life:0.7,size:3,color:'#fff'});
  pops.push({text:'+1',x:hdx,y:hdy-cell*0.6,life:1,color:'#ffd23f',u:Math.max(5,cell*0.28)});
  sEat();
  if(score%5===0){pops.push({text:'FASTER!',x:W/2,y:H*0.32,life:1.4,color:'#ffd23f',u:Math.max(6,cell*0.34)});sLevel();}
  serveBall();
}
function die(){
  if(state==='menu'){resetAttract();return;}
  state='dying';dieT=0;shake=13;flashR=0.55;sOver();
  const hdx=ox+(snake[0].x+0.5)*cell,hdy=oy+(snake[0].y+0.5)*cell;
  for(let i=0;i<30;i++){const a=Math.random()*Math.PI*2,sp2=60+Math.random()*260;
    parts.push({x:hdx,y:hdy,vx:Math.cos(a)*sp2,vy:Math.sin(a)*sp2,life:1,
      size:2+Math.random()*4,color:Math.random()<0.5?'#ff5a4e':'#ffffff'});}
}
function updateBall(dt){
  const half=ball.sz/2;
  const dist=ball.speed*dt,n=Math.max(1,Math.ceil(dist/(half*1.1)));
  const hdx=ox+(snake[0].x+0.5)*cell,hdy=oy+(snake[0].y+0.5)*cell;
  const eatR=(cell*0.62)*(cell*0.62);
  for(let i=0;i<n;i++){
    ball.x+=ball.vx*dt/n;ball.y+=ball.vy*dt/n;
    if(ball.y-half<cy0){ball.y=cy0+half;ball.vy=Math.abs(ball.vy);sWall();spark(ball.x,cy0,4);}
    else if(ball.y+half>cy0+ch){ball.y=cy0+ch-half;ball.vy=-Math.abs(ball.vy);sWall();spark(ball.x,cy0+ch,4);}
    for(const p of pads){
      const top=p.cy-p.h/2,bot=p.cy+p.h/2;
      if(p.side==='L'){if(ball.vx<0&&ball.x-half<=p.x+p.w&&ball.x+half>=p.x&&ball.y+half>top&&ball.y-half<bot)hitPad(p,1);}
      else{if(ball.vx>0&&ball.x+half>=p.x&&ball.x-half<=p.x+p.w&&ball.y+half>top&&ball.y-half<bot)hitPad(p,-1);}
    }
    const dx=ball.x-hdx,dy=ball.y-hdy;
    if(dx*dx+dy*dy<eatR&&(state==='playing'||state==='menu')){eat();return;}
  }
  ball.trail.push({x:ball.x,y:ball.y});if(ball.trail.length>14)ball.trail.shift();
}

/* ---------- update ---------- */
function update(dt){
  if(state==='playing'||state==='menu'){
    stepT+=dt*1000;let guard=0;
    while(stepT>=stepTime&&guard++<4&&(state==='playing'||state==='menu')){
      stepT-=stepTime;
      if(state==='menu')aiPick();
      step();
    }
  }else if(state==='dying'){
    dieT+=dt*1000;if(dieT>850)state='over';
  }
  if(state!=='paused'&&state!=='over'){updatePads(dt);updateBall(dt);}
  for(let i=parts.length-1;i>=0;i--){const p=parts[i];p.life-=dt*1.6;
    if(p.life<=0){parts.splice(i,1);continue;}
    if(p.ring)p.r+=dt*320;
    else{p.x+=p.vx*dt;p.y+=p.vy*dt;p.vx*=Math.max(0,1-2.5*dt);p.vy*=Math.max(0,1-2.5*dt);}}
  for(let i=pops.length-1;i>=0;i--){const p=pops[i];p.life-=dt*0.9;p.y-=dt*26;if(p.life<=0)pops.splice(i,1);}
  shake=Math.max(0,shake-dt*26);
  flashR=Math.max(0,flashR-dt*1.3);
  flashW=Math.max(0,flashW-dt*2.2);
  popT=Math.max(0,popT-dt*3);
}
