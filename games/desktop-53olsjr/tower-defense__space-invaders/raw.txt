function stepFX(dt){
 for(const p of parts){p.t+=dt;p.x+=p.vx*dt;p.y+=p.vy*dt;p.vy+=70*dt;}parts=parts.filter(p=>p.t<p.life);
 for(const f of floats){f.t+=dt;f.y-=28*dt;}floats=floats.filter(f=>f.t<f.life);
 for(const b of beams)b.t+=dt;beams=beams.filter(b=>b.t<b.dur);
 for(const r of rings)r.t+=dt;rings=rings.filter(r=>r.t<r.dur);
 G.shake*=Math.exp(-6*dt);if(G.shake<.05)G.shake=0;
 G.flash=Math.max(0,G.flash-dt*1.6);}
function step(dt){
 if(G.counting){G.countdown-=dt;if(G.countdown<=0)launchWave(0);}
 if(G.waveActive){G.qT+=dt;while(G.queue.length&&G.queue[0].t<=G.qT){const s=G.queue.shift();createEnemy(s.type,s.fx);}}
 for(const e of enemies){if(e.dead)continue;e.animT+=dt;
  const sf=e.slowT>0?(1-e.slowF):1;e.y+=e.spdF*H*sf*dt;e.x=clamp(e.fx*W+Math.sin(G.t*e.freq+e.ph)*e.ampF*W,e.rF*U,W-e.rF*U);
  e.hitT-=dt;if(e.slowT>0){e.slowT-=dt;if(e.slowT<=0)e.slowF=0;}
  if(e.type==='boss'){e.dropT-=dt;if(e.dropT<=0){e.dropT=7;const kids=enemies.filter(o=>o.parent===e&&!o.dead).length;if(kids<4)createEnemy('drone',clamp(e.fx+R(-.12,.12),.05,.95),{y:e.y+e.szF*U*.3,hpMul:.4,reward:5,parent:e});}}
  if(e.y>=H*.885)baseHit(e);}
 enemies=enemies.filter(e=>!e.dead);
 for(const p of PADS){const t=p.tower;if(!t)continue;t.cd-=dt;t.recoil=Math.max(0,t.recoil-dt*5);t.flashT=Math.max(0,(t.flashT||0)-dt);fireTower(t);}
 for(const b of bullets){b.x+=b.vx*dt;b.y+=b.vy*dt;let d=Math.hypot(b.vx,b.vy)*dt;if(b.d!==undefined)d=b.d;d-=Math.hypot(b.vx,b.vy)*dt*0; 
  let trav=0;for(const e of enemies){if(e.dead)continue;const rr=e.rF*U+3;if(Math.abs(e.x-b.x)<rr&&Math.abs(e.y-b.y)<rr){damageEnemy(e,b.dmg);boom(b.x,b.y,b.col,3,.5);b.dead=true;break;}}
  if(!b.dead){trav=Math.hypot(b.vx,b.vy)*dt;(b.trav=(b.trav||0)+trav);if((b.trav||0)>b.maxD||b.y<-90||b.y>H+90||b.x<-40||b.x>W+40)b.dead=true;}}
 bullets=bullets.filter(b=>!b.dead);
 if(G.waveActive&&!G.queue.length&&!enemies.length){const bonus=25+G.wave*4;G.credits+=bonus;G.earned+=bonus;banner('WAVE CLEARED  +'+bonus+' SCRAP','ok');sfx.clear();G.waveActive=false;startCountdown(12);}
 stepFX(dt);syncHUD();}
