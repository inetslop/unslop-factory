/* ================= physics ================= */
function collideSeg(s,e,boost){const dx=s.bx-s.ax,dy=s.by-s.ay,L2=dx*dx+dy*dy;
  let t=((ball.x-s.ax)*dx+(ball.y-s.ay)*dy)/L2;t=clamp(t,0,1);
  const px=s.ax+t*dx,py=s.ay+t*dy;let nx=ball.x-px,ny=ball.y-py;const d=Math.hypot(nx,ny);
  if(d>=BALL_R)return 0;
  if(d<1e-4){nx=-dy/Math.sqrt(L2);ny=dx/Math.sqrt(L2);}else{nx/=d;ny/=d;}
  ball.x=px+nx*BALL_R;ball.y=py+ny*BALL_R;
  const vn=ball.vx*nx+ball.vy*ny;if(vn<0){ball.vx-=(1+e)*vn*nx;ball.vy-=(1+e)*vn*ny;if(boost>0){ball.vx+=nx*boost;ball.vy+=ny*boost;}return -vn;}return 0;}
function collideWalls(){for(const w of WALLS){const imp=collideSeg(w,.72,0);if(imp>420)tone(180,120,.05,'triangle',.08);}}
function collideSlings(){for(const s of SLINGS){const imp=collideSeg(s,.85,430);if(imp>60){s.flash=1;addScore(25,s.bx,s.by-6,'+25','#59d8ff');SFX.sling();sparks((s.ax+s.bx)/2,(s.ay+s.by)/2,'#59d8ff',8,200);}}}
function collideBumpers(){for(const b of BUMPERS){let nx=ball.x-b.x,ny=ball.y-b.y;const d=Math.hypot(nx,ny);
  if(d<BALL_R+b.r){if(d<1e-4){nx=0;ny=-1;}else{nx/=d;ny/=d;}
    ball.x=b.x+nx*(BALL_R+b.r);ball.y=b.y+ny*(BALL_R+b.r);
    const sp=Math.max(Math.hypot(ball.vx,ball.vy),760);ball.vx=nx*sp+rand(-40,40);ball.vy=ny*sp+rand(-40,40);
    b.f=1;bumperCombo=Math.min(bumperCombo+1,8);comboT=3.5;
    addScore(50*bumperCombo,b.x,b.y-b.r-16,'+'+(50*bumperCombo)+(bumperCombo>1?'  \u00D7'+bumperCombo:''),'#ffd76a');
    SFX.bump(bumperCombo);shake=Math.max(shake,3);sparks(ball.x,ball.y,'#ff9d8a',8,240);}}}
function collideCrates(){for(const cr of crates){const dx=ball.x-cr.px,dy=ball.y-cr.py,hw=34+BALL_R;
  if(Math.abs(dx)>=hw||Math.abs(dy)>=hw)continue;
  let nx,ny;const adx=Math.abs(dx),ady=Math.abs(dy);
  if(adx>ady){nx=dx>0?1:-1;ny=0;}else{nx=0;ny=dy>0?1:-1;}
  if(nx)ball.x=cr.px+nx*hw;else ball.y=cr.py+ny*hw;
  const pc=-nx,pr=-ny;let canPush=!cr.pushing&&validPush(cr,pc,pr);
  const e=canPush?.42:.75;
  const vn=ball.vx*nx+ball.vy*ny;if(vn<0){ball.vx-=(1+e)*vn*nx;ball.vy-=(1+e)*vn*ny;}
  if(canPush)startPush(cr,pc,pr);}}
function collideFlip(p,o){const tx=p.x+Math.cos(o.ang)*FLIP.len,ty=p.y+Math.sin(o.ang)*FLIP.len;
  const dx=tx-p.x,dy=ty-p.y,L2=dx*dx+dy*dy;let t=((ball.x-p.x)*dx+(ball.y-p.y)*dy)/L2;t=clamp(t,0,1);
  const qx=p.x+t*dx,qy=p.y+t*dy;let nx=ball.x-qx,ny=ball.y-qy;const d=Math.hypot(nx,ny),R=BALL_R+FLIP.r;
  if(d>=R)return;if(d<1e-4){nx=0;ny=-1;}else{nx/=d;ny/=d;}
  ball.x=qx+nx*R;ball.y=qy+ny*R;
  const rx=qx-p.x,ry=qy-p.y,vsx=-o.omega*ry,vsy=o.omega*rx;
  const vn=(ball.vx-vsx)*nx+(ball.vy-vsy)*ny;if(vn<0){ball.vx-=1.45*vn*nx;ball.vy-=1.45*vn*ny;}}
function collideFlippers(){collideFlip(FLIP.pivL,flipL);collideFlip(FLIP.pivR,flipR);}

function physics(dt){const steps=clamp(Math.ceil(Math.hypot(ball.vx,ball.vy)*dt/9),1,5),sdt=dt/steps;
  for(let i=0;i<steps;i++){
    ball.vy+=980*sdt;const sp=Math.hypot(ball.vx,ball.vy);if(sp>1450){ball.vx*=1450/sp;ball.vy*=1450/sp;}
    ball.x+=ball.vx*sdt;ball.y+=ball.vy*sdt;
    collideWalls();collideSlings();collideBumpers();collideCrates();collideFlippers();
    if(ball.y>H+30)return loseBall();}
  const sp2=Math.hypot(ball.vx,ball.vy);
  if(sp2<45){ball.stuckT+=dt;if(ball.stuckT>1.2){ball.vy-=280;ball.vx+=rand(-140,140);ball.stuckT=0;}}else ball.stuckT=0;
  ball.trail.push({x:ball.x,y:ball.y});if(ball.trail.length>14)ball.trail.shift();}
function loseBall(){SFX.drain();shake=8;sparks(400,952,'#ff4757',22,340);popup(400,900,'BALL LOST','#ff6b7a');
  ball.active=false;balls--;updateHUD();if(balls<=0){mode='over';$('overScore').textContent=pad(score);saveHi();SFX.over();showOv('ovOver');}else spawnBall();}

function updFlips(dt){
  const tL=input.left?FLIP.actL:FLIP.restL;flipL.ang+=clamp(tL-flipL.ang,-26*dt,26*dt);flipL.omega=(flipL.ang-flipL.prev)/Math.max(dt,1e-4);flipL.prev=flipL.ang;
  const tR=input.right?FLIP.actR:FLIP.restR;flipR.ang+=clamp(tR-flipR.ang,-26*dt,26*dt);flipR.omega=(flipR.ang-flipR.prev)/Math.max(dt,1e-4);flipR.prev=flipR.ang;
  if(input.left&&!prevL)SFX.flip();prevL=input.left;
  if(input.right&&!prevR)SFX.flip();prevR=input.right;}

/* ================= update ================= */
function update(dt){
  tGlobal+=dt;
  shake=Math.max(0,shake-dt*24);shakeX=rand(-1,1)*shake;shakeY=rand(-1,1)*shake;
  gridFlash=Math.max(0,gridFlash-dt*1.8);
  for(const b of BUMPERS)b.f=Math.max(0,b.f-dt*2.6);
  for(const s of SLINGS)s.flash=Math.max(0,s.flash-dt*3);
  for(let i=particles.length-1;i>=0;i--){const p=particles[i];p.t+=dt;if(p.t>p.life){particles.splice(i,1);continue;}p.vy+=(p.g||0)*dt;p.x+=p.vx*dt;p.y+=p.vy*dt;if(p.rot!==undefined)p.rot+=p.vr*dt;}
  for(let i=popups.length-1;i>=0;i--){popups[i].t+=dt;if(popups[i].t>popups[i].dur)popups.splice(i,1);}
  for(const d of dust){d.y-=d.v*dt;if(d.y<-4){d.y=ih+4;d.x=rand(iw);}}
  if(mode==='play'){
    if(ball.ready&&charging)charge=Math.min(1,charge+dt/1.1);
    updFlips(dt);
    occ=occSet();
    for(const cr of crates){if(!cr.pushing)continue;cr.t+=dt/.11;const k=cr.t>=1?1:1-(1-cr.t)*(1-cr.t);
      cr.px=lerp(cr.fromX,cr.toX,k);cr.py=lerp(cr.fromY,cr.toY,k);
      if(cr.t>=1){cr.pushing=false;cr.c+=cr.dc;cr.r+=cr.dr;cr.px=cr.toX;cr.py=cr.toY;onPushDone(cr);if(mode!=='play')break;}}
    if(mode==='play'&&ball.active)physics(dt);
    if(comboT>0){comboT-=dt;if(comboT<=0)bumperCombo=0;}
  }
  const want=(mode==='play'&&ball.ready)?'block':'none';
  if($('btnLaunch').style.display!==want)$('btnLaunch').style.display=want;
}
