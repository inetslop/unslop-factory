}

function tryFire(){
  if(fireCd>0||state!=='playing')return;
  fireCd=.17;P.mflash=.05;
  const a=P.aimAng,gx=P.x+Math.cos(a)*24,gy=P.y-6*GRAV+Math.sin(a)*24;
  B.push({x:gx,y:gy,vx:Math.cos(a)*1350+P.vx*.3,vy:Math.sin(a)*1350,life:.9});
  sfx.shoot();
}

/* ============================== physics ============================== */
function overlap(a,b){return Math.abs(a.x-b.x)<(a.w+b.w)/2-4&&Math.abs(a.y-b.y)<(a.h+b.h)/2-4}

function movePlayer(dt,canInput){
  if(canInput){
    let L=keys.has('KeyA')||keys.has('ArrowLeft'),Rr=keys.has('KeyD')||keys.has('ArrowRight');
    if(touchMove!==null){const wx=(touchMove.x-ox)/sc;if(wx<P.x-14)L=true;if(wx>P.x+14)Rr=true;}
    P.vx+=((Rr?1:0)-(L?1:0))*2600*dt;
  }
  P.vx*=Math.exp(-(P.grounded?14:3)*dt);
  P.vx=clamp(P.vx,-340,340);
  // aim
  const wx=(mouse.x-ox)/sc,wy=(mouse.y-oy)/sc;
  P.aimAng=Math.atan2(wy-(P.y-6*GRAV),wx-P.x);
  P.face=Math.cos(P.aimAng)>=0?1:-1;
  // integrate
  P.x+=P.vx*dt;P.y+=P.vy*dt;
  if(P.x<T+13){P.x=T+13;if(P.vx<0)P.vx=0}
  if(P.x>W-T-13){P.x=W-T-13;if(P.vx>0)P.vx=0}
  let impact=0;
  if(GRAV>0){
    if(P.y+22>=H-T){if(!P.grounded)impact=P.vy;P.y=H-T-22;P.vy=0;P.grounded=true;}else P.grounded=false;
    if(P.y-22<=T){P.y=T+22;if(P.vy<0)P.vy=0;}
  }else{
    if(P.y-22<=T){if(!P.grounded)impact=-P.vy;P.y=T+22;P.vy=0;P.grounded=true;}else P.grounded=false;
    if(P.y+22>=H-T){P.y=H-T-22;if(P.vy>0)P.vy=0;}
  }
  if(impact>1400&&P.invuln<=0)hurt(1,0);
  if(impact>150){const fy=GRAV>0?H-T:T;dust(P.x,fy,6);sh=Math.max(sh,Math.min(8,impact/260));P.sq=.7;sfx.thud(impact/4000);}
  P.coyote=P.grounded?.09:P.coyote-dt;
  P.jumpBuf-=dt;
  if(P.jumpBuf>0&&(P.grounded||P.coyote>0)){P.vy=-730*GRAV;P.grounded=false;P.sq=-.5;P.jumpBuf=0;P.coyote=0;sfx.jump();}
  P.sq*=Math.exp(-7*dt);
  P.walkPh+=P.vx*dt*.06;
  P.mflash-=dt;P.invuln-=dt;
  if(P.grounded&&Math.abs(P.vx)>200){stepT-=dt;if(stepT<=0){stepT=.18;dust(P.x,GRAV>0?H-T:T,1);}}
}

function updZ(z,dt){
  z.flash-=dt;z.stagger-=dt;
  if(z.type==='spectre'){
    const dx=P.x-z.x,dy=P.y-z.y,d=Math.hypot(dx,dy)||1,k=clamp(dt*2.4,0,1);
    z.vx=lerp(z.vx,dx/d*z.spd,k);z.vy=lerp(z.vy,dy/d*z.spd,k);
  }else{
    if(!z.grounded){z.vy+=G*GRAV*dt;}
    else{
      const dir=(P.x-z.x)>=0?1:-1;z.dir=dir;
      z.vx=lerp(z.vx,dir*z.spd*(z.stagger>0?.35:1),clamp(dt*8,0,1));
      if(Math.abs(z.vx)>4)z.phase+=dt*(6+z.spd*.05)*Math.sign(z.vx);
    }
  }
  z.x+=z.vx*dt;z.y+=z.vy*dt;
  if(z.x<T+14){z.x=T+14;if(z.vx<0)z.vx=0}
  if(z.x>W-T-14){z.x=W-T-14;if(z.vx>0)z.vx=0}
  let impact=0;
  if(z.type!=='spectre'){
    if(GRAV>0){
      if(z.y+z.hh>=H-T){if(!z.grounded)impact=z.vy;z.y=H-T-z.hh;z.vy=0;z.grounded=true;}else z.grounded=false;
      if(z.y-z.hh<=T){z.y=T+z.hh;if(z.vy<0)z.vy=0;}
    }else{
      if(z.y-z.hh<=T){if(!z.grounded)impact=-z.vy;z.y=T+z.hh;z.vy=0;z.grounded=true;}else z.grounded=false;
      if(z.y+z.hh>=H-T){z.y=H-T-z.hh;if(z.vy>0)z.vy=0;}
    }
    if(impact>0){
      const fy=GRAV>0?H-T:T;dust(z.x,fy,4);
      if(impact>1400)killZ(z,'crush');
      else if(impact>900){z.hp-=3;z.flash=.15;if(z.hp<=0)killZ(z,'crush');else z.stagger=1.1;}
      else if(impact>650){z.hp-=1;z.flash=.12;if(z.hp<=0)killZ(z,'crush');else z.stagger=.7;}
    }
  }
}

/* ============================== sim ============================== */
function sim(dt,canInput){
  flipCd-=dt;fireCd-=dt;
  if(comboT>0){comboT-=dt;if(comboT<=0)combo=0;}
  groanT-=dt;
  if(groanT<=0&&Z.length&&AC){const z=Z[irand(0,Z.length-1)];
    sfx.groan(clamp((z.x-W/2)/(W/2),-1,1)*.8);groanT=rand(4,9);}
  if(canInput&&(fireHeld||touchFire!==null))tryFire();

  movePlayer(dt,canInput);

  // stomp + contact
  for(const z of Z){
    if(z.dead)continue;
    if(canInput&&P.invuln<=0&&!z.grounded===false&&false){} // noop guard
    if(!z.grounded===true&&z.type!=='spectre'&&canInput){
      if(!P.grounded&&GRAV*P.vy>250&&overlap(P,z)){
        const fromTop=GRAV>0?P.y<z.y:P.y>z.y;
        if(fromTop){killZ(z,'stomp');P.vy=-GRAV*640;P.grounded=false;sfx.stomp();sh=Math.max(sh,6);continue;}
      }
    }
    if(canInput&&overlap(P,z)&&P.invuln<=0)hurt(1,(P.x>=z.x?1:-1));
  }

  for(const z of Z)if(!z.dead)updZ(z,dt);
  Z=Z.filter(z=>!z.dead);

  // telegraphs
  for(const tg of TELE){tg.t-=dt;if(tg.t<=0)spawnZ(tg.type);}
  TELE=TELE.filter(tg=>tg.t>0);

  // bullets
  for(const b of B){
    b.x+=b.vx*dt;b.y+=b.vy*dt;b.life-=dt;
    if(b.x<T||b.x>W-T||b.y<T||b.y>H-T){b.life=0;burst(b.x,b.y,4,['#d8ff6a','#fff'],120,.3,0);}
    for(const z of Z){
      if(z.dead)continue;
      if(Math.hypot(b.x-z.x,b.y-z.y)<z.w*.55+3){
        b.life=0;z.hp--;z.flash=.12;z.vx+=Math.sign(b.vx)*90;
        burst(b.x,b.y,4,['#7fae62','#5d8f4a'],140,.4,.5);
        sfx.thock();
        if(z.hp<=0)killZ(z,'shot');
        break;
      }
    }
  }
  B=B.filter(b=>b.life>0);

  // wave manager
  if(canInput){
    if(phase==='inter'){
      interT-=dt;
      if(interT<=0){waveN++;queue=buildQueue(waveN);phase='wave';spawnT=.5;
        showBanner('WAVE '+waveN,'HOSTILES INBOUND');sfx.wave();}
    }else{
      if(queue.length){spawnT-=dt;
        if(spawnT<=0){spawnT=Math.max(.4,1.15-waveN*.06);addTelegraph(queue.pop());}}
      else if(!Z.length&&!TELE.length){
        const bonus=200*waveN;score+=bonus;
        POP.push({x:W/2,y:H/2-30,txt:'WAVE BONUS +'+bonus,t:0,dur:1.4,col:'#d9ff4a',size:20});
        burst(W/2,H/2,20,['#d9ff4a','#9dff5a'],260,.8,.3);
        if(waveN%3===0&&hp<maxHp){hp++;sfx.heal();POP.push({x:P.x,y:P.y-34,txt:'+1 ♥',t:0,dur:1.1,col:'#ff5566',size:18});}
        phase='inter';interT=2.4;showBanner('WAVE CLEARED','+'+bonus+' PTS');
      }
    }
  }

  if(state==='dying'){dieT-=dt;if(dieT<=0)finishOver();}
}

function attract(rdt,dt){
  aFlip-=rdt;if(aFlip<=0){aFlip=rand(3,4.2);flipGravity(true);}
  aSpawn-=rdt;
  if(aSpawn<=0&&Z.length<8){aSpawn=rand(.9,1.6);spawnZ(Math.random()<.75?'walker':'runner');}
  P.x=W/2;P.y=H-T-22;
  for(const z of Z)updZ(z,dt);
  Z=Z.filter(z=>!z.dead);
  for(const tg of TELE){tg.t-=dt;if(tg.t<=0)spawnZ(tg.type);}
  TELE=TELE.filter(tg=>tg.t>0);
}

function updateParticles(dt){
  for(const p of PT){
    p.vy+=G*GRAV*(p.g||0)*dt;
    const dr=Math.exp(-(p.dr||0)*dt);p.vx*=dr;if((p.g||0)<.5)p.vy*=dr;
    p.x+=p.vx*dt;p.y+=p.vy*dt;p.t+=dt;
    if(p.rock){
      const fy=GRAV>0?H-T:T;
      if((GRAV>0&&p.y>=fy)||(GRAV<0&&p.y<=fy)){p.t=p.dur;dust(p.x,fy,2);}
      p.x=clamp(p.x,T,W-T);
    }
  }
  PT=PT.filter(p=>p.t<p.dur);
  for(const r of RINGS)r.t+=dt;
  RINGS=RINGS.filter(r=>r.t<r.dur);
}

function update(rdt,dt){
  sh*=Math.exp(-5*rdt);if(sh<.05)sh=0;
  flipFlash=Math.max(0,flipFlash-rdt*3);
  hitFlash=Math.max(0,hitFlash-rdt*2.2);
  BAN.t+=rdt;
  for(const p of POP){p.t+=dt;p.y-=42*dt;}
  POP=POP.filter(p=>p.t<p.dur);
  const adt=state==='paused'?rdt*.25:dt;
  updateMotes(adt);
  for(const d of DEC)d.age+=adt;
  DEC=DEC.filter(d=>d.age<d.dur);
  if(state==='menu')attract(rdt,dt);
  else if(state==='playing'||state==='dying'){sim(dt,state==='playing');updateParticles(dt);}
}

function drawPlayer(t){

function drawBullets(){

function drawParticles(){
  for(const p of PT){const al=1-p.t/p.dur;ctx.globalAlpha=al;
    ctx.fillStyle=p.col;ctx.beginPath();ctx.arc(p.x,p.y,p.r*(.5+al*.5),0,7);ctx.fill();}
  ctx.globalAlpha=1;
  for(const r of RINGS){const p=r.t/r.dur,e=1-Math.pow(1-p,3);
    ctx.strokeStyle='rgba('+r.col+','+(1-p)*.8+')';ctx.lineWidth=r.lw*(1-p)+.5;
    ctx.beginPath();ctx.arc(r.x,r.y,lerp(r.r0,r.r1,e),0,7);ctx.stroke();}
}

function drawHeart(x,y,s,filled){

  if(filled){ctx.fillStyle='#ff5566';ctx.fill();
    ctx.strokeStyle='rgba(255,120,130,.6)';ctx.lineWidth=1;ctx.stroke();}
  else{ctx.fillStyle='rgba(20,30,28,.7)';ctx.fill();
    ctx.strokeStyle='rgba(255,85,102,.35)';ctx.lineWidth=1.5;ctx.stroke();}
}
