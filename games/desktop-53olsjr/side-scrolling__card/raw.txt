function blip(f0,f1,dur,type,vol,delay){
  if(!AC)return; delay=delay||0;

  g.gain.exponentialRampToValueAtTime(.0001,t0+dur);
  o.connect(g);g.connect(master);o.start(t0);o.stop(t0+dur+.06);
}

}

function gen(){
  const sf=Math.min(1,(G.speed-360)/420);
  while(genX < G.camX+LW*1.7){
    const gap=(70+Math.random()*50)+sf*(30+Math.random()*90);
    let w=(150+Math.random()*100)*(1-sf*.15);
    const nx=genX+gap;
    let type='normal';const r=Math.random();
    if(r<.10)type='down';
    else if(r<.18)type='joker';
    else if(r<.30&&genX>420&&!lastBurn)type='burn';
    const c={x:nx,w,type,rank:pick('A23456789JQK'),suit:pick(['sp','he','di','cl']),fl:0,revealed:type!=='down'};
    if(type==='joker'||type==='burn')c.rank=null;
    cards.push(c); lastBurn=(type==='burn');
    if(genX>520&&gap>=105&&Math.random()<.38+sf*.3)
      hazards.push({x:genX+gap/2,y:-108-Math.random()*46,rot:rand(0,6),dead:false});
    if(Math.random()<.5){ // chips on the card
      const n=Math.min(3+((Math.random()*3)|0),Math.max(2,((w-44)/34)|0));
      for(let i=0;i<n;i++)pickups.push({type:'chip',x:nx+26+i*34,y:-48,taken:false,ph:rand(0,6)});
    }else{ // arc over the gap
      for(let k=0;k<5;k++){const f=k/4;

}

function updateParts(dt){
  for(let i=parts.length-1;i>=0;i--){const p=parts[i];p.life-=dt;
    if(p.life<=0){parts.splice(i,1);continue;}
    p.vy+=(p.g||0)*dt;p.x+=p.vx*dt;p.y+=p.vy*dt;if(p.vr)p.rot+=p.vr*dt;}
}

}
function loseLife(src){
  if(G.state!=='play')return;
  G.lives--; setLives(G.lives,true); sfx.hurt();
  G.shake=16; G.hitStop=.13; G.flash=.55;
  shardBurst(P.x,Math.min(P.y,-10)-24);
  if(G.lives<=0){die();return;}
  P.invuln=2;
  if(src==='fall'){
    let tx=G.camX+LW*.4;
    for(const c of cards)if(c.x+70>P.x&&c.type!=='burn'){tx=c.x+60;break;}
    P.x=tx;P.y=0;P.vy=0;G.camX=P.x-LW*.32;addText(P.x,-70,'LIFE LOST','#ffd9d5');
  }else{P.vy=-480;}
}
function die(){
  G.state='over';G.overT=0;G.slowT=0;
  const sc=Math.floor(G.score);
  if(sc>G.best){G.best=sc;G.newBest=true;saveBest(G.best);}
  sfx.over(); if(G.newBest)confetti();
  P.vy=-320;P.spin=1;
  setTimeout(()=>{if(G.state==='over')showOver();},950);
}
function collectAce(p){
  p.taken=true;G.slowT=3.5;G.score+=100;sfx.ace();
  burst(p.x,p.y,14,i=>{const s=pick(['sp','he','di','cl']),a=rand(0,6.28),s2=rand(80,260);
    return {type:'suit',x:p.x,y:p.y,vx:Math.cos(a)*s2,vy:Math.sin(a)*s2-80,g:420,size:rand(5,10),rot:rand(0,6),vr:rand(-7,7),life:.9,max:.9,col:suitColor(s),suit:s};});
  addText(p.x,p.y-30,'SLOW-ACE! +100','#ffe9ad');
}

function update(dt){
  G.t+=dt;
  const tgt=G.state==='over'?.28:(G.slowT>0?.45:1);
  G.ts+=(tgt-G.ts)*Math.min(1,dt*6);
  if(G.flash>0)G.flash-=dt*2.2;
  G.shake*=Math.pow(.003,dt);
  if(G.hitStop>0){G.hitStop-=dt;return;}
  const wdt=dt*G.ts;
  G.slowT=Math.max(0,G.slowT-dt);

  for(const p of pickups)p.ph+=wdt*3;
  // face-down reveals (approach)
  for(const c of cards){
    if(c.type==='down'&&!c.revealed&&G.state!=='menu'){
      if(c.x<P.x+430&&c.x>P.x-120){
        c.fl+=wdt/0.22;
        if(c.fl>=.5){c.revealed=true;c.type='normal';G.score+=50;sfx.flip();
          addText(c.x+c.w/2,-70,'GAMBLE +50','#bfe8cf');
          burst(c.x+c.w/2,-30,6,()=>({type:'dot',x:c.x+rand(0,c.w),y:-rand(4,40),vx:rand(-40,40),vy:rand(-120,-30),g:500,size:rand(1.5,3),life:.5,max:.5,col:'#9fe0b8'}));}
      }
    }
  }

  if(G.state==='menu'){
    G.camX+=260*dt;gen();
    for(const h of hazards)h.rot+=wdt*9;
    emberT-=dt;
    if(emberT<=0){emberT=.3;const c=pick(cards);if(c&&c.x>G.camX&&c.x<G.camX+LW)
      addPart({type:'dot',x:rand(G.camX,G.camX+LW),y:rand(CARD_H+10,150),vx:rand(-8,8),vy:-rand(14,30),g:-6,size:rand(1.5,2.6),life:rand(.8,1.6),max:1.6,col:'#ff9a3d'});}
    updateParts(wdt);
  }
  else if(G.state==='play'){
    G.speed=Math.min(780,360+G.t*9);
    const dx=G.speed*wdt; P.x+=dx; G.dist+=dx; G.score+=dx*.08;

    // buffered jump
    if(P.press>G.t-.12){
      if(P.grounded||P.coyote>0)doJump(1);
      else if(P.jumps<2)doJump(2);
    }
    const prevY=P.y;
    P.vy+=2600*wdt; P.y+=P.vy*wdt;
    // land (one-way, all card tops at y=0)
    if(!P.grounded&&P.vy>=0&&prevY<=1.5&&P.y>=0){
      let c=null;for(const k of cards)if(P.x>k.x+4&&P.x<k.x+k.w-4){c=k;break;}
      if(c){
        if(c.type==='joker'){P.y=0;P.vy=-1280;P.jumps=2;sfx.bounce();
          burst(P.x,-6,10,i=>{const a=rand(0,6.28),s=rand(70,240);return {type:'dot',x:P.x,y:-6,vx:Math.cos(a)*s,vy:Math.sin(a)*s-90,g:500,size:rand(2,3.5),life:.5,max:.5,col:'#8fe0a8'};});}
        else if(c.type==='burn'){P.y=0;P.vy=-560;if(P.invuln<=0)loseLife('burn');}
        else{
          const imp=Math.min(1,Math.abs(P.vy)/1200);
          P.y=0;P.vy=0;P.grounded=true;P.jumps=0;P.squash=1;
          if(imp>.3){sfx.land(imp);dust(P.x,0);}
        }
      }
    }
    if(P.grounded){P.coyote=.09;P.leg+=dt*(6+G.speed*.02);}
    else{P.coyote-=wdt;}
    // rotation
    if(P.spin>0){P.spin-=wdt/.42;P.rot=(1-Math.max(0,P.spin))*Math.PI*3;}
    else{const tr=P.grounded?0:clamp(P.vy*.0009,-.5,.6);P.rot+=(tr-P.rot)*Math.min(1,dt*12);}
    P.squash=Math.max(0,P.squash-dt*5);
    P.invuln=Math.max(0,P.invuln-wdt);
    P.blinkT-=dt;if(P.blinkT<0){P.blinkT=rand(2,5);P.blink=.13;}P.blink=Math.max(0,P.blink-dt);
    // trail
    if((P.spin>0||G.slowT>0)&&Math.abs(P.y)<200)
      P.trail.push({x:P.x,y:P.y,rot:P.rot,a:.45});
    for(let i=P.trail.length-1;i>=0;i--){P.trail[i].a-=dt*1.6;if(P.trail[i].a<=0)P.trail.splice(i,1);}

    if(P.y>240)loseLife('fall');

    // pickups
    const cx=P.x,cy=P.y-24;
    for(const p of pickups){
      if(p.taken)continue;
      const dxp=cx-p.x,dyp=cy-(p.y),d=Math.hypot(dxp,dyp);
      if(p.type==='chip'){
        if(d<80&&d>1){p.x+=dxp/d*640*wdt;p.y+=dyp/d*640*wdt;}
        if(d<20){p.taken=true;G.chips++;G.score+=25;sfx.chip();sparksGold(p.x,p.y);}
      }else if(d<34)collectAce(p);
    }
    // hazards
    for(const h of hazards){
      if(h.dead)continue;
      h.rot+=wdt*9;
      const hy=h.y+Math.sin(G.t*3+h.x)*7;
      if(Math.abs(h.x-P.x)<32&&hy>P.y-54&&hy<P.y+8){
        h.dead=true;shardBurst(h.x,hy);
        if(P.invuln<=0)loseLife('hit');
      }
    }
    G.camX=P.x-LW*.32;
    gen();
    emberT-=dt;
    if(emberT<=0){emberT=.28;
      addPart({type:'dot',x:rand(G.camX,G.camX+LW),y:rand(CARD_H+10,150),vx:rand(-8,8),vy:-rand(14,30),g:-6,size:rand(1.5,2.6),life:rand(.8,1.6),max:1.6,col:'#ff9a3d'});}
    updateParts(wdt);
    // prune
    cards=cards.filter(c=>c.x+c.w>G.camX-500);
    pickups=pickups.filter(p=>!p.taken&&p.x>G.camX-400);
    hazards=hazards.filter(h=>!h.dead&&h.x>G.camX-400);
  }
  else if(G.state==='over'){
    G.overT+=dt;
    for(const h of hazards)h.rot+=wdt*9;
    updateParts(wdt);
  }
}

/* ============================== render ============================== */
const bgCards=[];for(let i=0;i<6;i++)bgCards.push({x:rand(0,1800),y:rand(30,300),w:rand(90,170),h:rand(126,240),suit:pick(['sp','he','di','cl']),ph:rand(0,6)});
const motes=[];for(let i=0;i<42;i++)motes.push({x:rand(0,1320),y:rand(20,500),z:rand(.3,.7),s:rand(1.2,2.8),ph:rand(0,6)});

function render(){
  ctx.setTransform(dpr,0,0,dpr,0,0);
  ctx.fillStyle='#03150e';ctx.fillRect(0,0,W,H);
  ctx.setTransform(dpr*scale,0,0,dpr*scale,dpr*offX,dpr*offY);
  ctx.save();ctx.beginPath();ctx.rect(0,0,LW,LH);ctx.clip();

  const shx=(Math.random()*2-1)*G.shake, shy=(Math.random()*2-1)*G.shake;
  ctx.translate(shx,shy);
  const t=G.t,camX=G.camX;

  // sky
  ctx.fillStyle=skyGrad;ctx.fillRect(-20,-20,LW+40,LH+40);
  // spotlights
  let sg=ctx.createRadialGradient(310+Math.sin(t*.13)*80,150,20,310+Math.sin(t*.13)*80,150,430);
  sg.addColorStop(0,'rgba(255,226,150,.11)');sg.addColorStop(1,'rgba(255,226,150,0)');
  ctx.fillStyle=sg;ctx.fillRect(-20,-20,LW+40,LH+40);
  sg=ctx.createRadialGradient(950+Math.cos(t*.11)*100,120,20,950+Math.cos(t*.11)*100,120,380);
  sg.addColorStop(0,'rgba(255,240,190,.08)');sg.addColorStop(1,'rgba(255,240,190,0)');
  ctx.fillStyle=sg;ctx.fillRect(-20,-20,LW+40,LH+40);
  // giant drifting cards
  for(const b of bgCards){
    const m=LW+520,sx=((b.x-camX*.12)%m+m)%m-260;
    ctx.save();ctx.translate(sx,b.y+Math.sin(t*.4+b.ph)*10);ctx.rotate(Math.sin(b.ph)*.08);
    rr(ctx,-b.w/2,-b.h/2,b.w,b.h,12);ctx.fillStyle='rgba(247,240,223,.05)';ctx.fill();
    suitPath(ctx,b.suit,0,0,b.w*.3);ctx.fillStyle='rgba(247,240,223,.05)';ctx.fill();
    ctx.restore();
  }
  // watermark rail
  const off=-(camX*.35)%90;
  for(let x=off-90,i=0;x<LW+90;x+=90,i++){
    suitPath(ctx,['sp','he','di','cl'][i%4],x,GROUND-26,15);
    ctx.fillStyle='rgba(247,240,223,.05)';ctx.fill();
  }
  // dust motes
  for(const mo of motes){
    const m=LW+40,sx=((mo.x-camX*mo.z)%m+m)%m-20;
    ctx.globalAlpha=.08+.08*Math.sin(t*2+mo.ph);
    ctx.fillStyle='#ffe9ad';ctx.beginPath();ctx.arc(sx,mo.y+Math.sin(t+mo.ph)*8,mo.s,0,7);ctx.fill();
  }
  ctx.globalAlpha=1;
  // pit / table below
  ctx.fillStyle=pitGrad;ctx.fillRect(-20,GROUND,LW+40,LH-GROUND+40);
  ctx.fillStyle='rgba(233,196,106,.16)';ctx.fillRect(-20,GROUND-1,LW+40,2.5);
  const st=-((camX)%46);
  ctx.fillStyle='rgba(233,196,106,.12)';
  for(let x=st;x<LW+46;x+=46)ctx.fillRect(x,GROUND+7,18,3);

  // back-layer particles (embers)
  for(const p of parts)if(p.type==='dot'&&p.y>0){
    ctx.globalAlpha=Math.max(0,p.life/p.max)*.7;ctx.fillStyle=p.col;
    ctx.beginPath();ctx.arc(p.x-camX,GROUND+p.y,p.size,0,7);ctx.fill();
  }
  ctx.globalAlpha=1;

  // cards
  for(const c of cards){
    const sx=c.x-camX;if(sx>LW+40||sx+c.w<-40)continue;
    let sh=ctx.createLinearGradient(0,GROUND+CARD_H,0,GROUND+CARD_H+52);
    sh.addColorStop(0,'rgba(0,0,0,.38)');sh.addColorStop(1,'rgba(0,0,0,0)');
    ctx.fillStyle=sh;ctx.fillRect(sx,GROUND+CARD_H,c.w,52);
    let sc=1;
    if(c.type==='down'&&!c.revealed)sc=Math.max(.06,Math.abs(Math.cos(c.fl*Math.PI)));
    ctx.save();ctx.translate(sx+c.w/2,GROUND);ctx.scale(sc,1);
    const bx=-c.w/2;
    if(!c.revealed&&c.type==='down')drawCardBack(ctx,bx,0,c.w,CARD_H);
    else drawCardFace(ctx,bx,0,c.w,CARD_H,c.type,c.rank||'A',c.suit,t);
    ctx.restore();
  }

  // pickups
  for(const p of pickups){
    if(p.taken)continue;
    const sx=p.x-camX;if(sx<-40||sx>LW+40)continue;
    const sy=GROUND+p.y+Math.sin(p.ph)*6;
    if(p.type==='chip'){
      ctx.save();ctx.translate(sx,sy);ctx.scale(Math.cos(t*5+p.ph),1);
      ctx.beginPath();ctx.arc(0,0,9.5,0,7);ctx.fillStyle='#ffd25e';ctx.fill();
      ctx.lineWidth=2;ctx.strokeStyle='#a8741c';ctx.stroke();
      ctx.setLineDash([3.5,3]);ctx.lineWidth=2;ctx.strokeStyle='rgba(255,255,255,.85)';
      ctx.beginPath();ctx.arc(0,0,5.6,0,7);ctx.stroke();ctx.setLineDash([]);
      ctx.restore();
    }else{
      const pl=1+Math.sin(t*4)*.12;
      let hg=ctx.createRadialGradient(sx,sy,2,sx,sy,30*pl);
      hg.addColorStop(0,'rgba(255,222,120,.5)');hg.addColorStop(1,'rgba(255,222,120,0)');
      ctx.fillStyle=hg;ctx.beginPath();ctx.arc(sx,sy,30*pl,0,7);ctx.fill();
      ctx.save();ctx.translate(sx,sy);ctx.rotate(t*.8);
      ctx.fillStyle='rgba(255,233,173,.14)';
      for(let i=0;i<4;i++){ctx.rotate(Math.PI/2);ctx.beginPath();ctx.moveTo(0,0);ctx.lineTo(-6,-34);ctx.lineTo(6,-34);ctx.closePath();ctx.fill();}
      ctx.restore();
      ctx.save();ctx.translate(sx,sy);
      rr(ctx,-13,-18,26,36,4);ctx.fillStyle='#fdf8ec';ctx.fill();ctx.strokeStyle='#caa14d';ctx.lineWidth=1.6;ctx.stroke();
      ctx.fillStyle='#c23b30';ctx.font='700 15px Georgia,serif';ctx.textAlign='center';ctx.textBaseline='middle';
      ctx.fillText('A',0,-6);suitPath(ctx,'di',0,7,7);ctx.fill();
      ctx.restore();
    }
  }

  // hazards (spinning wilds)
  for(const h of hazards){
    const sx=h.x-camX;if(sx<-50||sx>LW+50)continue;
    const sy=GROUND+h.y+Math.sin(t*3+h.x)*7;
    ctx.save();ctx.translate(sx,sy);ctx.rotate(h.rot);
    drawCardBack(ctx,-20,-28,40,56,'#9c1f2b');
    ctx.strokeStyle='rgba(255,90,60,.55)';ctx.lineWidth=3;rr(ctx,-23,-31,46,62,8);ctx.stroke();
    ctx.restore();
  }

  // player
  if(G.state==='play'||G.state==='over'){
    const px=P.x-camX,py=GROUND+P.y;
    for(const tr of P.trail){
      ctx.save();ctx.translate(tr.x-camX,GROUND+tr.y);ctx.rotate(tr.rot);
      suitPath(ctx,'sp',0,-30,17);ctx.globalAlpha=tr.a*.4;ctx.fillStyle='#e9c46a';ctx.fill();
      ctx.restore();ctx.globalAlpha=1;
    }
    const ghost=P.invuln>0&&Math.floor(t*14)%2===0;
    if(!ghost){
      ctx.save();ctx.translate(px,py);
      let sy2=1,sx2=1;
      if(P.grounded){sy2=1-.28*P.squash;sx2=1+.2*P.squash;}
      else{const st2=clamp(Math.abs(P.vy)/3200,0,.16);sy2=1+st2;sx2=1-st2*.6;}
      ctx.rotate(P.rot);ctx.scale(sx2,sy2);
      // legs
      ctx.strokeStyle='#1b1b22';ctx.lineWidth=3.6;ctx.lineCap='round';
      if(P.grounded){
        const a=Math.sin(P.leg)*8;
        ctx.beginPath();ctx.moveTo(-5,-9);ctx.lineTo(-5+a,-1);ctx.stroke();
        ctx.beginPath();ctx.moveTo(5,-9);ctx.lineTo(5-a,-1);ctx.stroke();
      }else{
        ctx.beginPath();ctx.moveTo(-5,-9);ctx.lineTo(-8,-3);ctx.stroke();
        ctx.beginPath();ctx.moveTo(5,-9);ctx.lineTo(4,-2);ctx.stroke();
      }
      // body spade
      suitPath(ctx,'sp',0,-30,17);
      ctx.fillStyle='#1b1b22';ctx.fill();
      suitPath(ctx,'sp',0,-30,17);ctx.strokeStyle='#e9c46a';ctx.lineWidth=2.4;ctx.stroke();
      // eyes
      const bl=P.blink>0?.15:1;
      for(const ex of[-5.5,5.5]){
        ctx.save();ctx.translate(ex+1.5,-35);ctx.scale(1,bl);
        ctx.beginPath();ctx.arc(0,0,3.4,0,7);ctx.fillStyle='#fff';ctx.fill();
        ctx.beginPath();ctx.arc(1.2,0,1.6,0,7);ctx.fillStyle='#1b1b22';ctx.fill();
        ctx.restore();
      }
      ctx.restore();
    }
    if(G.slowT>0&&G.state==='play'){
      ctx.save();ctx.translate(px,py-24);
      ctx.strokeStyle='rgba(255,233,173,.55)';ctx.lineWidth=2.5;
      ctx.setLineDash([9,7]);ctx.lineDashOffset=-t*60;
      ctx.beginPath();ctx.arc(0,0,36+Math.sin(t*8)*3,0,7);ctx.stroke();
      ctx.restore();
    }
  }

  // front particles
  for(const p of parts){
    if(p.type==='dot'&&p.y>0)continue;
    const sx=p.x-camX,sy=GROUND+p.y,a=Math.max(0,p.life/p.max);
    if(p.type==='dot'){ctx.globalAlpha=a;ctx.fillStyle=p.col;ctx.beginPath();ctx.arc(sx,sy,p.size,0,7);ctx.fill();}
    else if(p.type==='shard'){ctx.save();ctx.translate(sx,sy);ctx.rotate(p.rot||0);
      ctx.globalAlpha=a;ctx.fillStyle='#f8f1df';ctx.strokeStyle='#c23b30';ctx.lineWidth=1.4;
      ctx.fillRect(-p.size/2,-p.size*.7,p.size,p.size*1.4);ctx.strokeRect(-p.size/2,-p.size*.7,p.size,p.size*1.4);ctx.restore();}
    else if(p.type==='suit'){ctx.save();ctx.translate(sx,sy);ctx.rotate(p.rot||0);
      ctx.globalAlpha=a;suitPath(ctx,p.suit,0,0,p.size);ctx.fillStyle=p.col;ctx.fill();ctx.restore();}
    else if(p.type==='ring'){ctx.globalAlpha=a;ctx.strokeStyle=p.col;ctx.lineWidth=3;
      ctx.beginPath();ctx.arc(sx,sy,(1-a)*46+8,0,7);ctx.stroke();}
    else if(p.type==='text'){ctx.globalAlpha=Math.min(1,a*1.6);
      ctx.font='700 19px "Trebuchet MS",sans-serif';ctx.textAlign='center';
      ctx.fillStyle=p.col;ctx.fillText(p.str,sx,sy);}
    ctx.globalAlpha=1;
  }

  // overlays
  if(G.flash>0){ctx.fillStyle='rgba(220,60,50,'+(Math.max(0,G.flash)*.35)+')';ctx.fillRect(-20,-20,LW+40,LH+40);}
  if(G.slowT>0&&G.state==='play'){ctx.fillStyle='rgba(90,180,255,.06)';ctx.fillRect(-20,-20,LW+40,LH+40);}
  ctx.fillStyle=vig;ctx.fillRect(-20,-20,LW+40,LH+40);
  ctx.restore();
}
