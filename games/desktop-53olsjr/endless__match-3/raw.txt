  else if(t===3){ const s=r*.8; rrPath(ctx,x-s,y-s,s*2,s*2,r*.34); }
  else if(t===4){ poly(t5(6),x,y,r,-90); }
  else if(t===5){ poly(5,x,y,r,-90); }

}

  g.beginPath();
  if(t===0){ g.arc(x,y,r,0,6.2832); }
  else if(t===1){ g.moveTo(x,y-r); g.lineTo(x+r*.85,y); g.lineTo(x,y+r); g.lineTo(x-r*.85,y); g.closePath(); }
  else if(t===2){ g.moveTo(x,y-r*.95); g.lineTo(x+r*.92,y+r*.72); g.lineTo(x-r*.92,y+r*.72); g.closePath(); }
  else if(t===3){ const s=r*.8; rrPathT(g,x-s,y-s,s*2,s*2,r*.34); }
  else if(t===4){ polyT(6,g,x,y,r,-90); }
  else if(t===5){ polyT(5,g,x,y,r,-90); }
  else { starT(g,x,y,r,r*.45); }
}

function makeGem(t,sp){ return {t:t, sp:sp||0, fall:null, clear:null, born:T, wigT:-9e9, phase:rnd()*6.28}; }
function randType(){ return (rnd()*colors)|0; }

function newBoard(){
  for(let tries=0;tries<60;tries++){
    grid=[];
    for(let r=0;r<ROWS;r++){ grid[r]=[];
      for(let c=0;c<COLS;c++){ let t,nt=0;
        do{ t=randType(); nt++; }while(nt<40 && (
          (c>=2&&grid[r][c-1].t===t&&grid[r][c-2].t===t)||(r>=2&&grid[r-1][c].t===t&&grid[r-2][c].t===t)));
        grid[r][c]=makeGem(t); } }
    if(hasAnyMove()) return;
  }
}
function matchAt(r,c){ const g=grid[r][c]; if(!g)return false; const t=g.t;
  let h=1; for(let c2=c-1;c2>=0&&grid[r][c2]&&grid[r][c2].t===t;c2--)h++;
  for(let c2=c+1;c2<COLS&&grid[r][c2]&&grid[r][c2].t===t;c2++)h++;
  if(h>=3)return true;
  let v=1; for(let r2=r-1;r2>=0&&grid[r2][c]&&grid[r2][c].t===t;r2--)v++;
  for(let r2=r+1;r2<ROWS&&grid[r2][c]&&grid[r2][c].t===t;r2++)v++;
  return v>=3; }
function hasAnyMove(){
  for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++){
    const A=grid[r][c]; if(!A)continue;
    if(c+1<COLS){ const B=grid[r][c+1]; grid[r][c]=B; grid[r][c+1]=A;
      const ok=matchAt(r,c)||matchAt(r,c+1); grid[r][c]=A; grid[r][c+1]=B; if(ok)return true; }
    if(r+1<ROWS){ const B=grid[r+1][c]; grid[r][c]=B; grid[r+1][c]=A;
      const ok=matchAt(r,c)||matchAt(r+1,c); grid[r][c]=A; grid[r+1][c]=B; if(ok)return true; } }
  return false; }
function aiMove(){
  for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++){
    const A=grid[r][c]; if(!A)continue;
    if(c+1<COLS){ const B=grid[r][c+1]; grid[r][c]=B; grid[r][c+1]=A;
      const ok=matchAt(r,c)||matchAt(r,c+1); grid[r][c]=A; grid[r][c+1]=B; if(ok)return [{r:r,c:c},{r:r,c:c+1}]; }
    if(r+1<ROWS){ const B=grid[r+1][c]; grid[r][c]=B; grid[r+1][c]=A;
      const ok=matchAt(r,c)||matchAt(r+1,c); grid[r][c]=A; grid[r+1][c]=B; if(ok)return [{r:r,c:c},{r:r+1,c:c}]; } }
  return null; }

function findRuns(){
  const runs=[];
  for(let r=0;r<ROWS;r++){ let c=0; while(c<COLS){ const g=grid[r][c]; if(!g){c++;continue;}
    let l=1; while(c+l<COLS&&grid[r][c+l]&&grid[r][c+l].t===g.t)l++;
    if(l>=3){ const cells=[]; for(let i=0;i<l;i++)cells.push({r:r,c:c+i}); runs.push({cells:cells,color:g.t,len:l}); }
    c+=l; } }
  for(let c=0;c<COLS;c++){ let r=0; while(r<ROWS){ const g=grid[r][c]; if(!g){r++;continue;}
    let l=1; while(r+l<ROWS&&grid[r+l][c]&&grid[r+l][c].t===g.t)l++;
    if(l>=3){ const cells=[]; for(let i=0;i<l;i++)cells.push({r:r+i,c:c}); runs.push({cells:cells,color:g.t,len:l}); }
    r+=l; } }
  return runs; }

/* ============================== RESOLVE CYCLE ============================== */
function choosePos(run){
  if(lastSwap) for(const c of run.cells)
    if((c.r===lastSwap.a.r&&c.c===lastSwap.a.c)||(c.r===lastSwap.b.r&&c.c===lastSwap.b.c)) return c;
  return run.cells[(run.len/2)|0];
}
function pickColor(counts){
  let bestV=-1,cand=[];
  for(const k in counts){ if(counts[k]>bestV){bestV=counts[k];cand=[+k];} else if(counts[k]===bestV)cand.push(+k); }
  return cand.length?cand[(rnd()*cand.length)|0]:randType();
}
let pendingSpecials=[];

function beginClear(runs){
  const matched=new Map(), counts={};
  for(const run of runs) for(const c of run.cells){ const k=key(c.r,c.c);
    if(!matched.has(k)) matched.set(k,c); counts[run.color]=(counts[run.color]||0)+1; }

  const spawn={};
  for(const run of runs){ const p=choosePos(run), k=key(p.r,p.c), sp=run.len>=5?2:1;
    if(!(spawn[k]&&spawn[k].sp===2)) spawn[k]={sp:sp,t:run.color}; }
  const runCnt=new Map(), ovCol={};
  runs.forEach(run=>run.cells.forEach(cel=>{ const k=key(cel.r,cel.c);
    runCnt.set(k,(runCnt.get(k)||0)+1); if(!(k in ovCol))ovCol[k]=run.color; }));
  runCnt.forEach((n,k)=>{ if(n>=2) spawn[k]={sp:2,t:ovCol[k]}; });

  const clearSet=new Set(matched.keys()), queue=[];
  matched.forEach((v,k)=>{ const g=grid[v.r][v.c]; if(g&&g.sp)queue.push(k); });
  const consumed=new Set(), booms=[], prisms=[]; let prismCol=null;
  while(queue.length){
    const k=queue.pop(); if(consumed.has(k))continue;
    const r=k>>3, c=k&7, g=grid[r][c]; if(!g||!g.sp)continue; consumed.add(k);
    if(g.sp===1){ booms.push({x:cellX(c),y:cellY(r)});
      for(let dr=-1;dr<=1;dr++)for(let dc=-1;dc<=1;dc++){ const rr2=r+dr,cc=c+dc;
        if(rr2<0||rr2>=ROWS||cc<0||cc>=COLS)continue; const gg=grid[rr2][cc]; if(!gg)continue;
        const nk=key(rr2,cc); clearSet.add(nk); if(gg.sp&&!consumed.has(nk))queue.push(nk); } }
    else { prisms.push({x:cellX(c),y:cellY(r)});
      if(prismCol==null)prismCol=pickColor(counts);
      for(let rr2=0;rr2<ROWS;rr2++)for(let cc=0;cc<COLS;cc++){ const gg=grid[rr2][cc];
        if(gg&&gg.t===prismCol){ const nk=key(rr2,cc); clearSet.add(nk);
          if(gg.sp&&!consumed.has(nk))queue.push(nk); } } }
  }

  pendingSpecials=[];
  for(const k in spawn){ const r=k>>3,c=k&7;
    pendingSpecials.push({r:r,c:c,t:spawn[k].t,sp:spawn[k].sp}); clearSet.delete(k); }

  let maxDur=190, sx=0, sy=0;
  clearSet.forEach(k=>{ const r=k>>3,c=k&7,g=grid[r][c]; if(!g)return;
    const bigFx=consumed.has(k)&&g.sp; g.clear={t0:T,dur:bigFx?300:190};
    maxDur=Math.max(maxDur,g.clear.dur); sx+=cellX(c); sy+=cellY(r); spawnBurst(cellX(c),cellY(r),g.t,bigFx?14:7); });

  if(state==='playing'){
    const pts=clearSet.size*10*chainCount*streak + booms.length*60 + prisms.length*200;
    score+=pts; levelCheck(); bumpScore();
    popups.push({x:sx/clearSet.size,y:sy/clearSet.size,t0:T,
      text:'+'+fmt(pts)+(chainCount>1?'  ×'+chainCount:''),col:'#ffe9b0',big:false});
  }
  if(booms.length){ sfx.boom(); shakeAmp=Math.min(14,shakeAmp+6);
    rings.push({x:booms[0].x,y:booms[0].y,t0:T,col:'#ffd94a',maxR:cell*2.4});
    popups.push({x:bx+boardPx/2,y:by+boardPx/2-cell,t0:T,text:'BOMB!',col:'#ffd94a',big:true}); }
  if(prisms.length){ sfx.prism(); shakeAmp=Math.min(16,shakeAmp+8);
    rings.push({x:prisms[0].x,y:prisms[0].y,t0:T,col:PAL[prismCol],maxR:cell*2.8});
    popups.push({x:bx+boardPx/2,y:by+boardPx/2-cell,t0:T,text:'PRISM!',col:'#ff9ae2',big:true}); }
  sfx.pop(chainCount,clearSet.size);
  phase='clearing'; clearUntil=T+maxDur+90;
}

function applyGravity(){
  for(const s of pendingSpecials){ grid[s.r][s.c]=makeGem(s.t,s.sp); }
  pendingSpecials=[];
  for(let c=0;c<COLS;c++){
    let write=ROWS-1;
    for(let r=ROWS-1;r>=0;r--){ const g=grid[r][c];
      if(!g||g.clear){ grid[r][c]=null; continue; }
      if(r!==write){ grid[write][c]=g; grid[r][c]=null;
        const d=write-r; g.fall={off:-(d*cell),t0:T,dur:clamp(130+d*26,140,430)}; }
      write--; }
    const nSp=write+1;
    for(let r=write;r>=0;r--){ const g=makeGem(randType());
      g.fall={off:-(nSp*cell),t0:T,dur:clamp(140+nSp*26,150,430)}; grid[r][c]=g; }
  }
  phase='falling'; fallSettleAt=null;
}

function permuteBoard(){
  const arr=[]; for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++)arr.push(grid[r][c]);
  let ok=false;
  for(let tr=0;tr<80&&!ok;tr++){

    for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++)grid[r][c]=arr[key(r,c)];
    ok = findRuns().length===0 && hasAnyMove();
  }
  if(!ok){ newBoard(); }
  for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++){ grid[r][c].born=T; grid[r][c].clear=null; grid[r][c].fall=null; }
}

function dust(x,y){ for(let i=0;i<3;i++)
  particles.push({x:x+(rnd()-.5)*cell*.5,y:y,vx:(rnd()-.5)*60,vy:-30-rnd()*40,g:300,
    t0:T,life:320,col:'rgba(160,200,210,.7)',size:cell*.05,rot:0,vr:0,shard:false}); }

/* ============================== UPDATE ============================== */
function update(dt){
  while(toastQ.length&&T>=toastQ[0].at) showT(toastQ.shift());
  if(shakeAmp>0.05) shakeAmp*=Math.exp(-dt*.006); else shakeAmp=0;
  if(state==='playing'&&streak>1&&T>streakEndAt) streak=1;
  if(T>tipNextAt){ tipIdx=(tipIdx+1)%TIPS.length;
    tipsEl.style.opacity=0; setTimeout(()=>{tipsEl.textContent=TIPS[tipIdx];tipsEl.style.opacity=1;},300);
    tipNextAt=T+7000; }

  for(let i=particles.length-1;i>=0;i--){ const p=particles[i], age=T-p.t0;
    if(age>p.life){ particles.splice(i,1); continue; }
    p.vy+=p.g*dt/1000; p.x+=p.vx*dt/1000; p.y+=p.vy*dt/1000; p.rot+=p.vr*dt/1000; }
  for(let i=rings.length-1;i>=0;i--) if(T-rings[i].t0>420) rings.splice(i,1);
  for(let i=popups.length-1;i>=0;i--) if(T-popups[i].t0>(popups[i].big?1200:950)) popups.splice(i,1);

  switch(phase){
    case 'swap': {
      const p=(T-swap.t0)/swap.dur;
      if(p>=1){
        if(!swap.back){
          const A=grid[swap.a.r][swap.a.c], B=grid[swap.b.r][swap.b.c];
          grid[swap.a.r][swap.a.c]=B; grid[swap.b.r][swap.b.c]=A;
          const runs=findRuns();
          if(runs.length){
            if(state==='playing'){
              if(T-lastScoreT<STREAK_MS) streak=Math.min(streak+1,6); else streak=1;
              lastScoreT=T; streakEndAt=T+STREAK_MS; }
            chainCount=1; swap=null; beginClear(runs);
          } else { swap.back=true; swap.t0=T; sfx.invalid(); A.wigT=T; B.wigT=T; }
        } else { phase='idle'; swap=null; }
      } break; }
    case 'clearing': if(T>=clearUntil) applyGravity(); break;
    case 'falling': {
      let done=true;
      for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++){ const g=grid[r][c];
        if(g&&g.fall){ const e=g.fall.t0+g.fall.dur;
          if(T>=e){ const dcells=Math.abs(g.fall.off)/cell; g.fall=null;
            if(dcells>1.3){ dust(cellX(c),by+r*cell+cell*.8); sfx.tick(); } }
          else done=false; } }
      if(done){ if(fallSettleAt===null) fallSettleAt=T+80;
        else if(T>=fallSettleAt){ fallSettleAt=null; chainCount++;
          const runs=findRuns();
          if(runs.length&&chainCount<40) beginClear(runs); else finishResolve(); } }
      break; }
    case 'shuffle': {
      const p=(T-shufT0)/560;
      if(p>=.5&&!shufDone){ permuteBoard(); shufDone=true; }
      if(p>=1) phase='idle';
      break; }
    case 'idle':
      if(state==='menu'&&T>nextDemoAt){ nextDemoAt=T+1300;
        const m=aiMove(); if(m) trySwap(m[0],m[1]); else doShuffle(true); }
      break;
  }
}

/* ============================== RENDER ============================== */
const GL=[]; for(let i=0;i<5;i++)GL.push({x:rnd(),y:rnd(),sp:.04+rnd()*.08,ph:rnd()*6.28,col:[1,3,4,6][i%4],r:170+rnd()*160});
const STARS=[]; for(let i=0;i<70;i++)STARS.push({x:rnd(),y:rnd(),r:.6+rnd()*1.5,sp:.5+rnd()*2,ph:rnd()*6.28});

function render(){
  ctx.clearRect(0,0,Ww,Hh);
  // drifting color glows
  ctx.save(); ctx.globalCompositeOperation='screen';
  for(const g of GL){ const gx=Ww/2+Math.cos(T*.0001*g.sp*60+g.ph)*Ww*.45,
    gy=Hh/2+Math.sin(T*.00013*g.sp*60+g.ph)*Hh*.4;
    const gr=ctx.createRadialGradient(gx,gy,0,gx,gy,g.r);
    gr.addColorStop(0,PAL[g.col]+'2e'); gr.addColorStop(1,'rgba(0,0,0,0)');
    ctx.fillStyle=gr; ctx.fillRect(gx-g.r,gy-g.r,g.r*2,g.r*2); }
  ctx.restore();
  for(const s of STARS){ const a=.15+.3*(.5+.5*Math.sin(T*.001*s.sp+s.ph));
    ctx.fillStyle='rgba(190,230,240,'+a.toFixed(3)+')';
    ctx.fillRect(s.x*Ww,s.y*Hh,s.r,s.r); }

  // shake
  let shx=0,shy=0; if(shakeAmp>0){ shx=(rnd()-.5)*shakeAmp; shy=(rnd()-.5)*shakeAmp; }
  ctx.save(); ctx.translate(shx,shy);

  drawFrame();
  // cells
  for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++){
    rr(bx+c*cell+2,by+r*cell+2,cell-4,cell-4,cell*.18);
    ctx.fillStyle=(r+c)%2?'#0f2c35':'#12333d'; ctx.fill(); }

  // gems
  const ka=swap?key(swap.a.r,swap.a.c):-1, kb=swap?key(swap.b.r,swap.b.c):-1;
  let shufScale=1;
  if(phase==='shuffle'){ const p=(T-shufT0)/560; shufScale=p<.5?1-p*2:(p-.5)*2; }
  for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++){
    const g=grid[r][c]; if(!g)continue; const k=key(r,c);
    if(k===ka||k===kb)continue;
    drawGem(g,cellX(c),cellY(r),1,k);
  }
  if(swap){
    const p=swap.back?1-easeIO(Math.min(1,(T-swap.t0)/swap.dur)):easeIO(Math.min(1,(T-swap.t0)/swap.dur));
    const ax=cellX(swap.a.c),ay=cellY(swap.a.r),px2=cellX(swap.b.c),py2=cellY(swap.b.r);
    drawGem(grid[swap.a.r][swap.a.c],ax+(px2-ax)*p,ay+(py2-ay)*p,1.08,-1);
    drawGem(grid[swap.b.r][swap.b.c],px2+(ax-px2)*p,py2+(ay-py2)*p,1.08,-1);
  }

  // rings
  for(const rg of rings){ const p=(T-rg.t0)/420; if(p>=1)continue;
    ctx.beginPath(); ctx.arc(rg.x,rg.y,easeOC(p)*rg.maxR,0,6.2832);
    ctx.strokeStyle=rg.col; ctx.globalAlpha=1-p; ctx.lineWidth=5*(1-p)+1; ctx.stroke(); ctx.globalAlpha=1; }

  // particles
  for(const p of particles){ const age=(T-p.t0)/p.life, a=age<.6?1:1-(age-.6)/.4;
    ctx.globalAlpha=Math.max(0,a); ctx.fillStyle=p.col;
    if(p.shard){ ctx.save(); ctx.translate(p.x,p.y); ctx.rotate(p.rot);
      ctx.fillRect(-p.size/2,-p.size/2,p.size,p.size); ctx.restore(); }
    else { ctx.beginPath(); ctx.arc(p.x,p.y,p.size/2,0,6.2832); ctx.fill(); } }
  ctx.globalAlpha=1;

  drawCursorAndSel();
  drawPopups();
  ctx.restore();
}

function drawGem(g,x,y,baseScale,k){
  let sc=baseScale*shufScale, al=1, yo=Math.sin(T*.002+g.phase)*cell*.018, xo=0;
  const bp=Math.min(1,(T-g.born)/240); if(bp<1) sc*=easeOB(bp);
  if(g.clear){ const p=(T-g.clear.t0)/g.clear.dur;
    if(p>=1)return;
    sc*= p<.3?1+(p/.3)*.45:Math.max(0,1.45*(1-(p-.3)/.7));
    al=1-Math.max(0,(p-.25)/.75); }
  if(T-g.wigT<300) xo=Math.sin((T-g.wigT)*.09)*cell*.07;
  const hov = hoverCell&&k===key(hoverCell.r,hoverCell.c)&&phase==='idle'&&state==='playing';
  const sel = selected&&k===key(selected.r,selected.c);
  if(hov)sc*=1.06; if(sel)sc*=1.05+Math.sin(T*.008)*.02;

  const s=cell*.86*sc;
  ctx.globalAlpha=al;
  // ground shadow
  ctx.beginPath(); ctx.ellipse(x,y+cell*.22,cell*.3*sc,cell*.12*sc,0,0,6.2832);
  ctx.fillStyle='rgba(0,0,0,.28)'; ctx.fill();
  // sprite
  ctx.drawImage(SPR[g.t],x-s/2+xo,y-s/2+yo,s,s);
  // gloss sweep
  const sw=(T*.0006+g.phase)%3.2;
  if(sw<1){ ctx.save(); shapePath(g.t,x+xo,y+yo,cell*.4*sc); ctx.clip();
    ctx.globalAlpha=al*(1-sw)*.35; ctx.fillStyle='#ffffff';
    ctx.fillRect(x-s/2+sw*s*1.4-6+xo,y-s/2,s*.28,s*1.2); ctx.restore(); }
  // specials
  if(g.sp===1){ const pu=.8+.2*Math.sin(T*.01+g.phase);
    ctx.save(); ctx.shadowColor='#ffd94a'; ctx.shadowBlur=10*pu;
    ctx.fillStyle='#fffbe6'; ctx.strokeStyle='rgba(70,35,5,.9)'; ctx.lineWidth=cell*.045;
    const R=cell*.28;
    ctx.beginPath(); ctx.moveTo(x+R*.25+xo,y-R); ctx.lineTo(x-R*.45+xo,y+R*.12); ctx.lineTo(x-R*.02+xo,y+R*.12);
    ctx.lineTo(x-R*.25+xo,y+R); ctx.lineTo(x+R*.45+xo,y-R*.12); ctx.lineTo(x+R*.02+xo,y-R*.12);
    ctx.closePath(); ctx.fill(); ctx.stroke(); ctx.restore(); }
  else if(g.sp===2){ const hue=(T*.12+g.phase*57)%360;
    shapePath(g.t,x+xo,y+yo,cell*.4*sc);
    ctx.strokeStyle='hsl('+hue+' 95% 68%)'; ctx.lineWidth=cell*.075; ctx.stroke();
    const rot=T*.003+g.phase, R2=cell*.3*sc;
    ctx.save(); ctx.translate(x+xo,y+yo); ctx.rotate(rot);
    ctx.strokeStyle='rgba(255,255,255,.9)'; ctx.lineWidth=cell*.04;
    ctx.beginPath(); ctx.moveTo(-R2,0); ctx.lineTo(R2,0); ctx.moveTo(0,-R2); ctx.lineTo(0,R2); ctx.stroke();
    ctx.restore(); }
  if(sel){ ctx.beginPath(); ctx.arc(x,y,cell*.5*(1+.04*Math.sin(T*.008)),0,6.2832);
    ctx.strokeStyle='rgba(255,255,255,.9)'; ctx.lineWidth=2.5; ctx.stroke(); }
  ctx.globalAlpha=1;
}

function drawFrame(){
  const pad=14;
  rr(bx-pad,by-pad,boardPx+pad*2,boardPx+pad*2,18);
  ctx.save(); ctx.shadowColor='rgba(0,0,0,.55)'; ctx.shadowBlur=30; ctx.shadowOffsetY=12;
  ctx.fillStyle='#0a1c23'; ctx.fill(); ctx.restore();
  const fg=ctx.createLinearGradient(0,by-pad,0,by+boardPx+pad);
  fg.addColorStop(0,'#f0cd7e'); fg.addColorStop(.5,'#b98a35'); fg.addColorStop(1,'#8a5f22');
  ctx.strokeStyle=fg; ctx.lineWidth=3.5; ctx.stroke();
  rr(bx-pad+5,by-pad+5,boardPx+pad*2-10,boardPx+pad*2-10,14);
  ctx.strokeStyle='rgba(0,0,0,.4)'; ctx.lineWidth=1.5; ctx.stroke();
  for(const [cx,cy] of [[bx-pad/2,by-pad/2],[bx+boardPx+pad/2,by-pad/2],[bx-pad/2,by+boardPx+pad/2],[bx+boardPx+pad/2,by+boardPx+pad/2]]){
    ctx.beginPath(); ctx.arc(cx,cy,4.5,0,6.2832); ctx.fillStyle='#d9b45c'; ctx.fill();
    ctx.beginPath(); ctx.arc(cx,cy,1.8,0,6.2832); ctx.fillStyle='#6e4a15'; ctx.fill(); }
  if(T-frameFlashT<600){ const p=(T-frameFlashT)/600;
    rr(bx-pad,by-pad,boardPx+pad*2,boardPx+pad*2,18);
    ctx.strokeStyle='rgba(255,217,74,'+(1-p).toFixed(3)+')'; ctx.lineWidth=6*(1-p)+1; ctx.stroke(); }
}

function drawCursorAndSel(){
  if(kbMode&&state==='playing'&&phase!=='shuffle'){
    rr(bx+cursor.c*cell+3,by+cursor.r*cell+3,cell-6,cell-6,cell*.2);
    ctx.setLineDash([7,6]); ctx.lineDashOffset=-T*.02;
    ctx.strokeStyle='#ffd94a'; ctx.lineWidth=2.5; ctx.stroke(); ctx.setLineDash([]); }
}

function drawPopups(){
  for(const p of popups){ const dur=p.big?1200:950, age=(T-p.t0)/dur, q=age/dur;
    if(q>=1)continue;
    const y=p.y-q*38, a=q<.7?1:1-(q-.7)/.3;
    const size=p.big?cell*.95:clamp(cell*.42,14,26);
    const sc=p.big?1+Math.max(0,.25-q)*2:1;
    ctx.save(); ctx.translate(p.x,y); ctx.scale(sc,sc); ctx.globalAlpha=a;
    ctx.font='900 '+size+'px "Arial Rounded MT Bold","Trebuchet MS",sans-serif';
    ctx.textAlign='center'; ctx.textBaseline='middle';
    ctx.lineWidth=size*.22; ctx.strokeStyle='rgba(10,20,26,.85)'; ctx.strokeText(p.text,0,0);
    ctx.fillStyle=p.col; ctx.fillText(p.text,0,0);
    ctx.restore(); }
}

}

function layout(){
  const hudH=hudEl.offsetHeight||64, footH=footEl.offsetHeight||30;
  const availW=Ww-28, availH=Hh-hudH-footH-16;
  cell=Math.max(15,Math.floor(Math.min(availW/ROWS,availH/ROWS)));
  boardPx=cell*ROWS;
  bx=(Ww-boardPx)/2;
  by=hudH+(Hh-hudH-footH-boardPx)/2+4;
}
function resize(){
  dpr=Math.min(2,window.devicePixelRatio||1);
  Ww=window.innerWidth; Hh=window.innerHeight;
  cv.width=Ww*dpr; cv.height=Hh*dpr;
  cv.style.width=Ww+'px'; cv.style.height=Hh+'px';
  ctx.setTransform(dpr,0,0,dpr,0,0);
  layout();
}
window.addEventListener('resize',resize);
