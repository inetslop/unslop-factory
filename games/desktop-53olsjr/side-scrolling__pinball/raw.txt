function collideSeg(b,x1,y1,x2,y2,e,surf){
 const dx=x2-x1,dy=y2-y1,L2=dx*dx+dy*dy; let t=( (b.x-x1)*dx+(b.y-y1)*dy )/L2; if(t<0)t=0;else if(t>1)t=1;

 if(d2>=R*R) return null;
 let d=Math.sqrt(d2); if(d<1e-6){nx=0;ny=-1;d=1;} else {nx/=d;ny/=d;}

/* ============================== PHYSICS ============================== */
function collideSeg(b,x1,y1,x2,y2,e,surf){
  const dx=x2-x1, dy=y2-y1, L2=dx*dx+dy*dy;
  let t=((b.x-x1)*dx+(b.y-y1)*dy)/L2; t = t<0?0 : t>1?1 : t;
  const px=x1+t*dx, py=y1+t*dy;
  let nx=b.x-px, ny=b.y-py;
  const R=b.r+WALLR, d2=nx*nx+ny*ny;
  if(d2>=R*R) return null;
  let d=Math.sqrt(d2);
  if(d<1e-6){ nx=0; ny=-1; } else { nx/=d; ny/=d; }
  b.x=px+nx*R; b.y=py+ny*R;
  const sv = surf ? surf(px,py) : [0,0];
  const rvx=b.vx-sv[0], rvy=b.vy-sv[1], vn=rvx*nx+rvy*ny;
  if(vn<0){ b.vx-=(1+e)*vn*nx; b.vy-=(1+e)*vn*ny; }
  return { vn, px, py };
}

function overFloor(x){ for(const s of floorSegs) if(x>=s[0]&&x<=s[1]) return true; return false; }

function physBall(b,h){
  b.vy += G*h;
  const sp0 = Math.hypot(b.vx,b.vy);
  if(sp0>MAXV){ b.vx*=MAXV/sp0; b.vy*=MAXV/sp0; }
  b.x += b.vx*h; b.y += b.vy*h;

  // static solids (walls, kickers, floor)
  for(let i=0;i<solids.length;i++){
    const s=solids[i];
    const hit = collideSeg(b,s[0],s[1],s[2],s[3],rest[i]);
    if(hit && Math.abs(hit.vn)>340) sWall(Math.abs(hit.vn));
  }

  // pop bumpers
  for(const bp of bumpers){
    const dx=b.x-bp.x, dy=b.y-bp.y, R=bp.r+b.r, d2=dx*dx+dy*dy;
    if(d2<R*R){
      const d=Math.sqrt(d2)||1, nx=dx/d, ny=dy/d;
      b.x=bp.x+nx*R; b.y=bp.y+ny*R;
      const s=Math.max(900, Math.hypot(b.vx,b.vy)*0.85);
      b.vx=nx*s; b.vy=ny*s;
      bp.flash=1; shake=Math.min(shake+2.5,10);
      gain(100,b.x,b.y-bp.r-6,bp.col); sBump(chain);
      ring(bp.x,bp.y,bp.r*.8,bp.col,1); burst(b.x,b.y,bp.col,7);
    }
  }

  // slingshots
  for(const sg of slings){
    let hit=false;
    for(const e of sg.edges) if(collideSeg(b,e[0],e[1],e[2],e[3],0.9)) hit=true;
    if(hit){
      const sp=Math.hypot(b.vx,b.vy);
      if(sp<1450 && sp>1){ b.vx*=1450/sp; b.vy*=1450/sp; }
      sg.flash=1; shake+=2; gain(75,b.x,b.y-16,sg.col); sSling(); burst(b.x,b.y,sg.col,5);
    }
  }

  // rotating bars (moving obstacles)
  for(const br of bars){
    const c=Math.cos(br.a), s=Math.sin(br.a);
    const surf=(px,py)=>[br.w*(br.cy-py), br.w*(px-br.cx)];
    const hit=collideSeg(b, br.cx+c*br.half, br.cy+s*br.half, br.cx-c*br.half, br.cy-s*br.half, 0.5, surf);
    if(hit && Math.abs(hit.vn)>220){ sWall(320); burst(hit.px,hit.py,'#9fc6ff',4); }
  }

  // pop targets (rects)
  for(const tg of targets){
    const cx=clamp(b.x,tg.x,tg.x+tg.w), cy=clamp(b.y,tg.y,tg.y+tg.h);
    let nx=b.x-cx, ny=b.y-cy; const d2=nx*nx+ny*ny, R=b.r+4;
    if(d2<R*R){
      let d=Math.sqrt(d2);
      if(d<1e-6){ nx=0; ny=-1; } else { nx/=d; ny/=d; }
      b.x=cx+nx*R; b.y=cy+ny*R;
      const vn=b.vx*nx+b.vy*ny;
      if(vn<0){ b.vx-=1.7*vn*nx; b.vy-=1.7*vn*ny; }
      if(!tg.lit){ tg.lit=true; tg.t=4; gain(500,cx,tg.y-12,'#ffb300'); sTarget(); burst(cx,tg.y+13,'#ffd23f',10); }
      else { gain(25,cx,tg.y-8,'#8fb7ff'); tone(500,420,.05,'sine',.12); }
    }
  }

  // rocket pads
  for(const rk of rocks){
    if(rk.cd>0) continue;
    if(b.x>rk.x1-6 && b.x<rk.x2+6 && b.y>FLOOR-b.r-6 && b.y<FLOOR+8 && b.vy>140){
      b.vy = -(1500+rnd()*260); b.vx += (rnd()-.5)*280; rk.cd=.4;
      sRocket(); shake+=9; gain(50,(rk.x1+rk.x2)/2,FLOOR-34,'#ffb300','LAUNCH!');
      ring((rk.x1+rk.x2)/2,FLOOR-6,26,'#ffb300',1.5); burst(b.x,FLOOR-8,'#ffd23f',12);
    }
  }

  // flippers (kinematic capsules with surface velocity)
  const rx = rigX();
  if(Math.abs(b.x-rx)<360){
    for(const f of flippers){
      const g=fgeom(f);
      const surf=(px,py)=>{
        const dx=g.x2-g.x1, dy=g.y2-g.y1, L2=dx*dx+dy*dy;
        let t=((px-g.x1)*dx+(py-g.y1)*dy)/L2; t=t<0?0:t>1?1:t;
        const d=t*Math.sqrt(L2);
        return [-d*g.om*Math.sin(g.th), d*g.om*Math.cos(g.th)];
      };
      collideSeg(b,g.x1,g.y1,g.x2,g.y2,0.45,surf);
    }
  }

  // rolling friction on floor
  if(b.y>FLOOR-b.r-3 && overFloor(b.x)) b.vx *= (1-2.2*h);

  // anti-stall nudge
  const spd=Math.hypot(b.vx,b.vy);
  if(spd<30 && b.y>FLOOR-64){
    b.still+=h;
    if(b.still>1.15){ b.still=0; b.vy=-430; b.vx=clamp((rigX()-b.x)*1.4,-150,150); sNudge(); burst(b.x,b.y,'#9fc6ff',5); }
  } else b.still=0;

  // drain?
  if(b.y>FLOOR+26 || b.y>VH+60){ drainBall(b); return; }
}

function ballBall(){
  for(let i=0;i<balls.length;i++){ const a=balls[i]; if(!a.alive) continue;
    for(let j=i+1;j<balls.length;j++){ const c=balls[j]; if(!c.alive) continue;
      let nx=c.x-a.x, ny=c.y-a.y; const d2=nx*nx+ny*ny, R=a.r+c.r;
      if(d2<R*R && d2>1e-6){
        const d=Math.sqrt(d2); nx/=d; ny/=d;
        const ov=(R-d)/2; a.x-=nx*ov; a.y-=ny*ov; c.x+=nx*ov; c.y+=ny*ov;
        const vr=(a.vx-c.vx)*nx+(a.vy-c.vy)*ny;
        if(vr>0){ const jimp=vr*(1+0.85)/2; a.vx-=jimp*nx; a.vy-=jimp*ny; c.vx+=jimp*nx; c.vy+=jimp*ny;
          if(vr>260) sWall(vr); }
      }
    }
  }
}

function drainBall(b){
  b.alive=false; chain=0; chainT=0;
  sDrain(); shake+=8; ring(b.x,FLOOR+18,'#ff3d81',1.6); burst(b.x,FLOOR+10,'#ff3d81',14);
  if(aliveCount()>0){
    if(multiball && aliveCount()===1){ multiball=false; popups.push({x:rigX(),y:FLOOR-200,txt:'MULTIBALL END',col:'#8fb7ff',age:0,dur:1.2,big:true}); }
    return;
  }
  ballsLeft--;
  if(ballsLeft<=0){ gameOver(); return; }
  const nb = makeBall(rigX(),150);
  balls.push(nb);
  popups.push({x:rigX(),y:FLOOR-96,txt:'FLIP  A/◀  D/▶',col:'#9fc6ff',age:0,dur:1.5});
  tone(440,660,.12,'triangle',.2);
}

function step(h){

  for(const rk of rocks) if(rk.cd>0) rk.cd-=h;
  for(const b of balls) if(b.alive) physBall(b,h);
  ballBall();
}

function camera(dt){
  let p=null; for(const b of balls) if(b.alive){ p=b; break; }
  if(!p) return;
  const target = clamp(p.x - VW/2, 0, WORLD_W-VW);
  const d=target-cam.x;
  const rate = Math.abs(d)>300 ? 18 : 8;
  cam.x += d*(1-Math.exp(-rate*dt));
  cam.x = clamp(cam.x, 0, WORLD_W-VW);
}
function effects(dt){
  chainT-=dt; if(chainT<=0) chain=0;
  if(multiball){ mbT-=dt; if(mbT<=0){ multiball=false; } }
  for(const st of stars){
    if(!st.dead) continue;
    st.rt-=dt;
    if(st.rt<=0){
      const used=new Set(stars.filter(s=>!s.dead).map(s=>s.x+','+s.y));
      let slot=starSlots[0], guard=0;
      do{ slot=starSlots[(rnd()*6)|0]; guard++; }while(used.has(slot[0]+','+slot[1]) && guard<20);
      st.x=slot[0]; st.y=slot[1]; st.dead=false; st.phase=rnd()*6.28;
    }
  }
  // star collection (generous radius)
  for(const b of balls){ if(!b.alive) continue;
    for(const st of stars){ if(st.dead) continue;
      const dx=b.x-st.x, dy=b.y-(st.y+Math.sin(tGlobal*2+st.phase)*6);
      if(dx*dx+dy*dy < 28*28){
        st.dead=true; st.rt=2.6; starProg++;
        gain(750,st.x,st.y-14,'#ffd23f','★'); sStar(); burst(st.x,st.y,'#ffd23f',14); ring(st.x,st.y,10,'#ffd23f',1.2);
        if(starProg>=4) triggerMultiball();
      }
    }
  }
  for(const bp of bumpers) bp.flash=Math.max(0,bp.flash-4*dt);
  for(const sg of slings) sg.flash=Math.max(0,sg.flash-4.5*dt);
  for(const tg of targets){ if(tg.lit){ tg.t-=dt; if(tg.t<=0) tg.lit=false; } }
  for(const br of bars) br.a+=br.w*dt;
  for(let i=popups.length-1;i>=0;i--){ const p=popups[i]; p.age+=dt; if(p.age>=p.dur) popups.splice(i,1); }
  for(let i=parts.length-1;i>=0;i--){ const p=parts[i]; p.life-=dt; if(p.life<=0){ parts.splice(i,1); continue; }
    p.x+=p.vx*dt; p.y+=p.vy*dt; p.vy+=500*dt; }
  for(let i=rings.length-1;i>=0;i--){ const r=rings[i]; r.life-=dt; if(r.life<=0){ rings.splice(i,1); continue; } r.r+=r.vr*dt; }
  shake*=Math.exp(-7*dt); if(shake<.08) shake=0;
  for(const b of balls){ if(!b.alive) continue;
    b.trail.push({x:b.x,y:b.y}); if(b.trail.length>14) b.trail.shift(); }
}

function starPath(cx,cy,R,rot){
  ctx.beginPath();
  for(let i=0;i<10;i++){

    ctx.strokeStyle='rgba(255,255,255,.45)'; ctx.lineWidth=1.5; ctx.stroke();

  }

  ctx.globalCompositeOperation='lighter';
  for(const p of parts){

  for(const b of balls){ if(!b.alive) continue;

let last=performance.now();
function frame(now){
  requestAnimationFrame(frame);
  let dt=(now-last)/1000; last=now;
  if(dt>0.05) dt=0.05;
  tGlobal+=dt;

  if(state==='play'){
    acc+=dt; let n=0;
    while(acc>=HSTEP && n<12){ step(HSTEP); acc-=HSTEP; n++; }
    camera(dt); effects(dt);
  } else if(state==='menu'){
    cam.x=(Math.sin(tGlobal*0.13)*0.5+0.5)*(WORLD_W-VW);
    for(const br of bars) br.a+=br.w*dt;
  }

  ctx.setTransform(dpr,0,0,dpr,0,0);
  drawBezel(tGlobal);
  const shx=(rnd()*2-1)*shake*scale, shy=(rnd()*2-1)*shake*0.6*scale;
  ctx.save();
  ctx.beginPath(); ctx.rect(ox,oy,VW*scale,VH*scale); ctx.clip();
  ctx.translate(ox+shx,oy+shy); ctx.scale(scale,scale);
  drawScene(tGlobal);
  ctx.restore();
}

menuBest.innerHTML='BEST — <b>'+best.toLocaleString()+'</b>';
initStars();
requestAnimationFrame(frame);
})();
