  if(targets.length>14)return;
  var roll=Math.random(),type='popper';
  if(roll<.10&&elapsed>8)type='gold';
  else if(roll<.10+Math.min(.16,.03+prog*.13)&&elapsed-lastJoker>1.4){type='joker';lastJoker=elapsed;}
  else if(roll<.32+prog*.2)type=Math.random()<.5?'fast':'slider';
  var x,y,ok=false;
  for(var k=0;k<8;k++){x=rand(56,W-56);y=rand(railH+78,H*.46);ok=true;
    for(var i=0;i<targets.length;i++){var t2=targets[i],dx=t2.x-x,dy=t2.by-y;if(dx*dx+dy*dy<4900){ok=false;break;}}
    if(ok)break;}

  if(type==='popper'){b.r=26;b.val=100;b.life=rand(2.4,3.4);}

/* ================= shooting / scoring ================= */
function resetStreak(){
  if(mult>1){comboEl.classList.remove('lost');void comboEl.offsetWidth;comboEl.classList.add('lost');}
  streak=0;mult=1;
}
function finishCard(p){if(p.hitCount===0&&!p.neutral)resetStreak();}
function hitTarget(t,p){
  t.dead=true;hits++;p.hitCount++;
  streak++;if(streak>maxStreak)maxStreak=streak;
  var nm=Math.min(1+((streak/4)|0),5);
  if(nm>mult){mult=nm;SFX.comboUp(mult);floater(gun.x,gun.y-84,'COMBO ×'+mult,'#e8c15a',19);
    pipsEl.classList.remove('pop');void pipsEl.offsetWidth;pipsEl.classList.add('pop');}
  var pts=t.val*mult;
  if(p.suit==='c')pts*=2;
  var ace=p.card.r==='A';if(ace)pts*=2;
  score+=pts;
  floater(t.x,t.by-6,'+'+pts,t.type==='gold'?'#ffd75e':'#ffffff',pts>=400?23:17);
  if(ace&&!p.aceShown){p.aceShown=true;floater(p.x,p.y-22,'ACE ×2','#ffd75e',16);}
  burst(t.x,t.by,t.type==='gold'?GOLD_C:SUIT_COLORS[p.suit],14,230);
  if(t.type==='gold'){SFX.gold();shake=Math.min(shake+3,10);}
  else{SFX.hit(mult);shake=Math.min(shake+1.5,8);}
}
function explodeAt(x,y,p){

  waves.push({x:x,y:y,ttl:.42,life:.42,max:R});
  SFX.explode();shake=Math.min(shake+8,12);
  burst(x,y,['#ffd75e','#fff3c4','#ff9d5c','#ffffff'],26,260);
  for(var i=0;i<targets.length;i++){var t=targets[i];if(t.dead)continue;
    var dx=t.x-x,dy=t.by-y,rr2=R+t.r*.8;
    if(dx*dx+dy*dy<rr2*rr2){
      if(t.type==='joker'){t.dead=true;p.neutral=true;score+=25;floater(t.x,t.by,'DEFUSED +25','#ffd75e',14);SFX.defuse();}
      else hitTarget(t,p);}}
}
function jokerHit(t){

  floater(t.x,t.by,'-150','#ff6a5e',24);floater(gun.x,gun.y-84,'JOKER!','#ff8d7e',17);

}
function shoot(){
  if(state!=='play'||gun.stun>0)return;
  var c=hand.shift();refillHand();
  var spd=Math.max(820,Math.min(W,H)*1.5),a=gun.ang;
  var mx=gun.x+Math.cos(a)*(56-gun.rec),my=gun.y+Math.sin(a)*(56-gun.rec);
  cards.push({x:mx,y:my,px:mx,py:my,vx:Math.cos(a)*spd,vy:Math.sin(a)*spd,rot:a+Math.PI/2,
    spin:(Math.random()<.5?-1:1)*rand(3,6),card:c,suit:c.s,hitCount:0,neutral:false,bounces:0,trail:[],dead:false,aceShown:false});
  shots++;gun.flash=.06;crossPulse=1;gun.rec=7;SFX.shoot();
}

/* ================= update ================= */
function update(dt){
  elapsed+=dt;timeLeft-=dt;
  var tc=Math.ceil(timeLeft);
  if(tc!==tPrevCeil){if(tc<=10&&tc>0)SFX.tick();tPrevCeil=tc;}
  if(timeLeft<=0){timeLeft=0;endGame();return;}
  var prog=elapsed/60;
  spawnT-=dt;
  if(spawnT<=0){doSpawn(prog);spawnT=lerp(.95,.42,prog)*rand(.7,1.35);if(Math.random()<lerp(0,.35,prog))doSpawn(prog);}
  /* gun */
  gun.x=W/2;gun.y=H-64+Math.sin(elapsed*2)*1.5;
  var ta=Math.atan2(mouse.y-gun.y,mouse.x-gun.x);ta=clamp(ta,-Math.PI+.18,-.18);
  var d=ta-gun.ang;while(d>Math.PI)d-=TAU;while(d<-Math.PI)d+=TAU;
  gun.ang+=d*Math.min(1,dt*16);
  fireT-=dt;gun.stun=Math.max(0,gun.stun-dt);gun.flash=Math.max(0,gun.flash-dt);
  gun.rec*=Math.exp(-12*dt);crossPulse*=Math.exp(-9*dt);
  if(firing&&gun.stun<=0&&fireT<=0){shoot();fireT=.14;}
  /* targets */
  for(var i=0;i<targets.length;i++){var t=targets[i];t.age+=dt;
    if(t.vx){t.x+=t.vx*dt;if(t.x<46){t.x=46;t.vx*=-1;}else if(t.x>W-46){t.x=W-46;t.vx*=-1;}}
    t.by=t.baseY+(t.type==='slider'||t.type==='fast'?0:Math.sin(t.age*2.6+t.seed)*2.5);
    if(t.type==='gold'&&Math.random()<dt*10)burst(t.x+rand(-t.r,t.r)*.6,t.by+rand(-t.r,t.r)*.6,GOLD_C,1,60);
    if(t.age>t.life&&!t.dead){t.dead=true;burst(t.x,t.by,['rgba(200,210,205,.8)','#8fa09a'],4,70);}
  }
  targets=targets.filter(function(tt){return !tt.dead;});
  /* cards */
  for(var j=0;j<cards.length;j++){var p=cards[j];if(p.dead)continue;
    p.px=p.x;p.py=p.y;p.x+=p.vx*dt;p.y+=p.vy*dt;p.rot+=p.spin*dt;
    p.trail.push({x:p.px,y:p.py});if(p.trail.length>5)p.trail.shift();
    if(p.suit==='d'&&p.bounces<3){
      if(p.x<14&&p.vx<0){p.x=14;p.vx*=-1;p.bounces++;burst(14,p.y,['#ffe9a8','#fff'],3,120);}
      else if(p.x>W-14&&p.vx>0){p.x=W-14;p.vx*=-1;p.bounces++;burst(W-14,p.y,['#ffe9a8','#fff'],3,120);}
      else if(p.y<railH+6&&p.vy<0){p.y=railH+6;p.vy*=-1;p.bounces++;}
    }
    for(var k=0;k<targets.length;k++){var tt=targets[k];if(tt.dead)continue;
      var cr=tt.r*(tt.type==='joker'?.8:1)+8;
      if(segCircle(p.px,p.py,p.x,p.y,tt.x,tt.by,cr)){
        if(tt.type==='joker'){
          if(p.suit==='h'){explodeAt(p.x,p.y,p);p.dead=true;break;}
          else{p.hitCount=Math.max(p.hitCount,1);jokerHit(tt);if(p.suit!=='s')p.dead=true;if(p.dead)break;}
        }else{
          hitTarget(tt,p);
          if(p.suit==='s'){/* pierce */}
          else if(p.suit==='h'){explodeAt(p.x,p.y,p);p.dead=true;break;}
          else{p.dead=true;break;}
        }
      }
    }
    if(p.dead){finishCard(p);continue;}
    if(p.x<-50||p.x>W+50||p.y>H+60||p.y<-70){
      if(p.suit==='h')explodeAt(clamp(p.x,30,W-30),clamp(p.y,railH+30,H-40),p);
      else finishCard(p);
      p.dead=true;
    }
  }
  cards=cards.filter(function(cc){return !cc.dead;});
  /* fx */
  for(var q=parts.length-1;q>=0;q--){var pp=parts[q];pp.life-=dt;
    if(pp.life<=0){parts.splice(q,1);continue;}
    pp.vy+=pp.g*dt;pp.vx*=Math.exp(-1.6*dt);pp.x+=pp.vx*dt;pp.y+=pp.vy*dt;pp.rot+=pp.vr*dt;}
  for(var w2=waves.length-1;w2>=0;w2--){waves[w2].life-=dt;if(waves[w2].life<=0)waves.splice(w2,1);}
  for(var f=floats.length-1;f>=0;f--){floats[f].life-=dt;floats[f].y-=46*dt;if(floats[f].life<=0)floats.splice(f,1);}
  shake*=Math.exp(-7*dt);if(shake<.15)shake=0;
  flashRed=Math.max(0,flashRed-dt*1.4);
  syncHUD();
}

    ctx.lineTo(mouse.x+Math.cos(a)*(r+7),mouse.y+Math.sin(a)*(r+7));}
  ctx.stroke();
  ctx.fillStyle=stun?'rgba(255,110,94,.95)':'rgba(247,239,220,.95)';
  ctx.beginPath();ctx.arc(mouse.x,mouse.y,1.8,0,TAU);ctx.fill();

function draw(){
  ctx.save();
  var shx=0,shy=0;
  if(shake>.1){shx=(Math.random()*2-1)*shake;shy=(Math.random()*2-1)*shake;}
  ctx.translate(shx,shy);
  if(bgC)ctx.drawImage(bgC,-M,-M,W+2*M,H+2*M);
  if(state==='menu'||state==='over')drawAmbient();
  if(state!=='menu'){
    drawRails();
    for(var i=0;i<targets.length;i++)drawTarget(targets[i]);
    for(var j=0;j<cards.length;j++)drawCardProj(cards[j]);
    drawWaves();drawParts();
    drawGun();drawHand();drawDeckStack();
    drawFloats();
    if(state==='play')drawCrosshair();
  }
  ctx.restore();
  if(flashRed>0){ctx.fillStyle='rgba(210,40,30,'+(flashRed*.35)+')';ctx.fillRect(0,0,W,H);}
  if(state==='play'&&timeLeft<=10&&timeLeft>0){
    ctx.fillStyle='rgba(180,30,20,'+(.04+.04*Math.sin(elapsed*6))+')';ctx.fillRect(0,0,W,H);}
}
