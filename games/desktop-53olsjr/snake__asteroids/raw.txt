```js
function tone({f=440,f2,type='square',d=0.15,v=0.3,delay=0}){
 if(!AC||muted) return;
 const t0=AC.currentTime+delay;
 const o=AC.createOscillator(), g=AC.createGain();
 o.type=type; o.frequency.setValueAtTime(f,t0);
 if(f2) o.frequency.exponentialRampToValueAtTime(Math.max(1,f2),t0+d);
 g.gain.setValueAtTime(v,t0);
 g.gain.exponentialRampToValueAtTime(0.001,t0+d);
 o.connect(g).connect(master);
 o.start(t0); o.stop(t0+d+0.05);
}
function noiseBurst({d=0.2,v=0.3,f=1000,q=1,type='lowpass',delay=0}){
 if(!AC||muted) return;
 const t0=AC.currentTime+delay;
 const n=Math.floor(AC.sampleRate*d);
 const buf=AC.createBuffer(1,n,AC.sampleRate);
 const ch=buf.getChannelData(0);
 for(let i=0;i<n;i++) ch[i]=(Math.random()*2-1)*(1-i/n);
 const src=AC.createBufferSource(); src.buffer=buf;
 const flt=AC.createBiquadFilter(); flt.type=type; flt.frequency.value=f; flt.Q.value=q;
 const g=AC.createGain(); g.gain.value=v;
 src.connect(flt).connect(g).connect(master);
 src.start(t0);
}
```

  for(let i=0;i<n;i++){const a=rand(TAU),v=rand(.2,1)*sp;parts.push({x,y,vx:Math.cos(a)*v,vy:Math.sin(a)*v,life:rand(.35,.8),ml:.8,size:rand(1,size||3),col});}
}
function popup(x,y,txt,col,size){pops.push({x:wx(x),y:wy(y),txt,t:0,dur:1.05,col,size:size||14});}
function toast(txt,col){const s=document.createElement('span');s.textContent=txt;s.style.color=col||'#cfe6ff';toasts.appendChild(s);setTimeout(()=>s.remove(),1750);}

/* ================= game flow ================= */
function setMode(m){
  S.mode=m;
  ovMenu.classList.toggle('hidden',m!=='menu');
  ovDead.classList.toggle('hidden',m!=='dead');
  ovPause.classList.toggle('hidden',m!=='paused');
  document.body.classList.toggle('playing',m!=='menu');
}
function newGame(){
  S.score=0;S.combo=0;S.comboT=0;S.wave=1;S.waveT=0;S.shieldT=0;S.hurtT=0;S.dashCD=0;S.boostT=0;S.ts=1;S.dieT=0;
  S.msg={txt:'WAVE 1',t:0,dur:1.8};
  rocks=[];picks=[];parts=[];pops=[];
  const cx=W/2,cy=H/2,a=-Math.PI/2;
  snake={x:cx,y:cy,a,vx:0,vy:0,segs:[],pending:0,growT:0};
  for(let i=1;i<6;i++) snake.segs.push({x:wx(cx-Math.cos(a)*SPACING*i),y:wy(cy-Math.sin(a)*SPACING*i)});
  for(let i=0;i<6;i++) spawnRock();
  elScore._v=null;elLen._v=null;elWave._v=null;
  setMode('playing');
}
function startGame(){
  if(S.mode==='menu'){initAudio();sfx.ui();newGame();return true;}
  if(S.mode==='dead'&&S.t-S.deadAt>.4){initAudio();sfx.ui();newGame();return true;}
  return false;
}
function setPause(p){ if(p&&S.mode==='playing')setMode('paused'); else if(!p&&S.mode==='paused'){setMode('playing');} }
function toggleMute(){ muted=!muted; if(master)master.gain.value=muted?0:.5; lastEng=-1; toast(muted?'SOUND OFF':'SOUND ON','#9db8e0'); }

/* ================= rules ================= */
function registerEat(base){
  S.combo = S.comboT>0 ? S.combo+1 : 1;
  S.comboT=4; S.comboPop=1;
  return base*S.combo;
}
function dropRoll(size,x,y){
  const r=Math.random(); let type=null;
  if(r<.08)type='shield';
  else if(r < .08+(size===3?.4:size===2?.26:.12))type='crystal';
  if(type) picks.push({type,x:wx(x),y:wy(y),vx:rand(-15,15),vy:rand(-15,15),life:type==='shield'?10:12});
}
function eatRock(r,i){
  rocks.splice(i,1);
  const pts=registerEat(r.size===1?50:100); S.score+=pts;
  snake.pending+=r.size;
  popup(r.x,r.y,'+'+pts,r.size===1?'#ffd9a0':'#ffb35c',14+r.size*2);
  burst(r.x,r.y,12+r.size*8,r.size===1?'#ffcf8a':'#f7a852',130+r.size*30,3.2);
  if(r.size===2){
    for(let k=0;k<2;k++){const a=rand(TAU),sp=rand(40,90);const m=makeRock(1,r.x+rand(-6,6),r.y+rand(-6,6),1);m.vx=r.vx*.5+Math.cos(a)*sp;m.vy=r.vy*.5+Math.sin(a)*sp;rocks.push(m);}
  }
  S.shake=Math.min(S.shake+(r.size===1?3:6),14);
  sfx.shatter(r.size); sfx.eat(S.combo);
  dropRoll(r.size,r.x,r.y);
}
function shatterBig(r,i){
  rocks.splice(i,1);
  const pts=registerEat(150); S.score+=pts;
  popup(r.x,r.y,'+'+pts,'#ff8d3c',18);
  for(let k=0;k<2;k++){const a=rand(TAU),sp=rand(45,95);const m=makeRock(2,r.x,r.y,1);m.vx=r.vx*.4+Math.cos(a)*sp;m.vy=r.vy*.4+Math.sin(a)*sp;rocks.push(m);}
  burst(r.x,r.y,30,'#f7a852',230,5);
  S.shake=Math.min(S.shake+9,20);
  sfx.shatter(3); sfx.eat(S.combo);
  dropRoll(3,r.x,r.y);
}
function hitBig(r,i){
  const dx=wdx(snake.x,r.x),dy=wdy(snake.y,r.y),d=Math.hypot(dx,dy)||1e-4,nx=dx/d,ny=dy/d;
  if(S.shieldT>0){ S.shieldT=0; toast('SHIELD SHATTERED','#8fe8ff'); shatterBig(r,i); return; }
  const vn=snake.vx*nx+snake.vy*ny;
  if(vn<0){snake.vx-=2*vn*nx;snake.vy-=2*vn*ny;}
  snake.vx*=.9; snake.vy*=.9;
  snake.x=wx(r.x-nx*(r.r*.82+HEAD_R+2)); snake.y=wy(r.y-ny*(r.r*.82+HEAD_R+2));
  S.hurtT=1; S.comboT=0; S.combo=0; S.shake=Math.min(S.shake+13,18); sfx.hit();
  const lose=Math.min(2,Math.max(0,snake.segs.length-(MIN_LEN-1)));
  if(lose>0){ for(let k=0;k<lose;k++){const ts=snake.segs.pop();burst(ts.x,ts.y,6,'#3df2cf',90,2.5);} toast('−'+lose+' SEGMENT'+(lose>1?'S':''),'#ff8d6a'); }
  r.hp--;
  if(r.hp<=0) shatterBig(r,i); else burst(snake.x,snake.y,10,'#ffb35c',120,3);
}
function collect(p){
  if(p.type==='crystal'){
    const pts=registerEat(250); S.score+=pts;
    popup(p.x,p.y,'+'+pts,'#ff7ae0',16); burst(p.x,p.y,18,'#ff4fd8',170,3.5); sfx.crystal();
  }else{
    S.shieldT=SHIELD_DUR;
    popup(p.x,p.y,'SHIELD UP','#8fe8ff',15); burst(p.x,p.y,22,'#8fe8ff',180,3.5); sfx.shield();
    toast('SHIELD ACTIVE · '+SHIELD_DUR+'s','#8fe8ff');
  }
}
function doDash(){
  if(S.mode!=='playing'||S.dashCD>0||!snake)return;
  S.dashCD=DASH_CD; S.boostT=.45;
  snake.vx+=Math.cos(snake.a)*320; snake.vy+=Math.sin(snake.a)*320;
  burst(snake.x-Math.cos(snake.a)*14,snake.y-Math.sin(snake.a)*14,12,'#8fe8ff',150,3);
  sfx.dash();
}
function die(){
  S.finalLen=snake.segs.length+1;
  setMode('dying'); S.dieT=0; S.shake=22; sfx.die();
  burst(snake.x,snake.y,26,'#bfffee',260,4);
  for(const s of snake.segs) burst(s.x,s.y,3,'#3df2cf',120,3);
  snake=null;
}
function finishDeath(){
  const nb=S.score>S.best;
  if(nb){S.best=S.score;saveBest(S.best);elBest.textContent=S.best;}
  $('stScore').textContent=S.score;
  $('stLen').textContent=S.finalLen;
  $('stWave').textContent='WAVE '+S.wave;
  $('newBest').classList.toggle('hidden',!nb);
  setMode('dead'); S.deadAt=S.t;
  if(nb)sfx.best(); else tone({type:'sine',f:180,f2:70,d:.5,v:.25});
}

/* ================= input (window keys) ================= */
window.addEventListener('keydown',e=>{
  const kk=e.key.toLowerCase();
  if(['arrowup','arrowdown','arrowleft','arrowright',' '].includes(kk)) e.preventDefault();
  initAudio();
  keys[kk]=true;
  if(e.repeat)return;
  if(kk==='enter'||kk===' '){ if(startGame())return; }
  else if((kk==='r')&&S.mode==='dead'){ if(S.t-S.deadAt>.4){initAudio();sfx.ui();newGame();} return; }
  else if(kk==='p'||kk==='escape'){ setPause(S.mode==='playing'); return; }
  else if(kk==='m'){ toggleMute(); return; }
  if(kk===' '&&S.mode==='playing') doDash();
});
window.addEventListener('keyup',e=>{ keys[e.key.toLowerCase()]=false; });
window.addEventListener('pointerdown',()=>{ initAudio(); if(AC&&AC.state==='suspended')AC.resume(); });
ovMenu.addEventListener('click',()=>startGame());
$('btnStart').addEventListener('click',e=>{e.stopPropagation();startGame();});
ovDead.addEventListener('click',()=>{ if(S.t-S.deadAt>.4) startGame(); });
$('btnRetry').addEventListener('click',e=>{e.stopPropagation();if(S.t-S.deadAt>.4)startGame();});
document.addEventListener('visibilitychange',()=>{ if(document.hidden&&S.mode==='playing')setPause(true); });

/* ================= update ================= */
function update(rdt){
  S.t+=rdt;
  const target=S.mode==='dying'?.3:1;
  S.ts+=(target-S.ts)*Math.min(1,rdt*5);
  const dt=rdt*S.ts;
  const playing=S.mode==='playing';

  if(S.msg&&(S.msg.t+=rdt)>S.msg.dur) S.msg=null;
  S.shake*=Math.exp(-rdt*5.5); if(S.shake<.1)S.shake=0;
  S.hurtT=Math.max(0,S.hurtT-rdt);
  S.comboPop=Math.max(0,S.comboPop-rdt*3);
  S.dashCD=Math.max(0,S.dashCD-rdt);
  S.boostT=Math.max(0,S.boostT-dt);

  if(playing&&snake){
    const L=!!(keys['arrowleft']||keys['a']), R=!!(keys['arrowright']||keys['d']), U=!!(keys['arrowup']||keys['w']);
    S.thrusting=U;
    snake.a+=((R?1:0)-(L?1:0))*3.4*dt;
    if(U){snake.vx+=Math.cos(snake.a)*420*dt;snake.vy+=Math.sin(snake.a)*420*dt;}
    const f=Math.exp(-dt*.55); snake.vx*=f; snake.vy*=f;
    const maxSp=S.boostT>0?470:265+Math.min((S.wave-1)*8,70);
    const sp=Math.hypot(snake.vx,snake.vy); if(sp>maxSp){const k=maxSp/sp;snake.vx*=k;snake.vy*=k;}
    snake.x=wx(snake.x+snake.vx*dt); snake.y=wy(snake.y+snake.vy*dt);

    let px=snake.x,py=snake.y;
    for(const s of snake.segs){
      const dx=wdx(px,s.x),dy=wdy(py,s.y),d=Math.hypot(dx,dy);
      if(d>SPACING){const t=(d-SPACING)/d;s.x=wx(px-dx*t);s.y=wy(py-dy*t);}
      px=s.x;py=s.y;
    }
    if(snake.pending>0){
      snake.growT-=dt;
      while(snake.pending>0&&snake.growT<=0){
        snake.growT+=.12;
        const ls=snake.segs[snake.segs.length-1]||{x:snake.x,y:snake.y};
        if(snake.segs.length<MAX_LEN) snake.segs.push({x:ls.x,y:ls.y});
        snake.pending--;
      }
    }
    S.shieldT=Math.max(0,S.shieldT-dt);
    if(S.comboT>0){S.comboT-=dt;if(S.comboT<=0)S.combo=0;}
    S.waveT+=dt;
    if(S.waveT>=20){S.waveT=0;S.wave++;S.msg={txt:'WAVE '+S.wave,t:0,dur:1.8};sfx.ui();}
    const tgt=Math.min(4+S.wave*2,13); while(rocks.length<tgt) spawnRock();
  } else S.thrusting=false;

  if(S.mode!=='paused'){
    for(const r of rocks){r.x=wx(r.x+r.vx*dt);r.y=wy(r.y+r.vy*dt);r.rot+=r.vr*dt;}
    for(let i=picks.length-1;i>=0;i--){const p=picks[i];p.x=wx(p.x+p.vx*dt);p.y=wy(p.y+p.vy*dt);p.life-=dt;if(p.life<=0)picks.splice(i,1);}
    for(let i=parts.length-1;i>=0;i--){const p=parts[i];p.x+=p.vx*dt;p.y+=p.vy*dt;const f=Math.exp(-dt*1.6);p.vx*=f;p.vy*=f;p.life-=dt;if(p.life<=0)parts.splice(i,1);}
    for(let i=pops.length-1;i>=0;i--){pops[i].t+=rdt;if(pops[i].t>pops[i].dur)pops.splice(i,1);}
  }

  if(playing&&snake){
    for(let i=picks.length-1;i>=0;i--){const p=picks[i];if(wd(snake.x,snake.y,p.x,p.y)<HEAD_R+13){collect(p);picks.splice(i,1);}}
    for(let i=rocks.length-1;i>=0;i--){const r=rocks[i];if(wd(snake.x,snake.y,r.x,r.y)<HEAD_R+r.r*.82){ if(r.size===3)hitBig(r,i); else eatRock(r,i); }}
    if(S.mode==='playing'){
      for(let i=3;i<snake.segs.length;i++){const s=snake.segs[i];if(wd(snake.x,snake.y,s.x,s.y)<HEAD_R+bodyR(i)-2){die();break;}}
    }
  }

  if(engGain&&AC){ const on=(playing&&S.thrusting&&!muted)?.13:0; if(on!==lastEng){engGain.gain.setTargetAtTime(on,AC.currentTime,.08);lastEng=on;} }

  if(S.mode==='dying'){ S.dieT+=rdt; if(S.dieT>1.4) finishDeath(); }
  hud();
}

/* ================= HUD ================= */
function hud(){
  if(elScore._v!==S.score){elScore.textContent=S.score;elScore._v=S.score;}
  if(snake){const L=snake.segs.length+1;if(elLen._v!==L){elLen.textContent=L;elLen._v=L;}}
  if(elWave._v!==S.wave){elWave.textContent=S.wave;elWave._v=S.wave;}
  const showC=S.combo>=2&&S.comboT>0&&snake;
  elCombo.style.opacity=showC?1:0;
  if(showC)elCombo.textContent='COMBO ×'+S.combo;
  elCombo.style.transform='scale('+(1+S.comboPop*.4)+')';
  const dc=clamp(1-S.dashCD/DASH_CD,0,1);
  elDash.style.transform='scaleX('+dc+')';
  elDashBar.classList.toggle('ready',dc>=1);
}

/* ================= render ================= */
function neb(x,y,r,c){const g=ctx.createRadialGradient(x,y,0,x,y,r);g.addColorStop(0,c);g.addColorStop(1,'rgba(0,0,0,0)');ctx.fillStyle=g;ctx.fillRect(x-r,y-r,r*2,r*2);}

function drawPicks(){
  for(const p of picks){
    if(p.life<3&&Math.floor(S.t*8)%2)continue;
    ctx.save(); ctx.translate(p.x,p.y);
    if(p.type==='crystal'){
      const pulse=.7+.3*Math.sin(S.t*6);
      ctx.globalCompositeOperation='lighter';
      ctx.fillStyle='rgba(255,79,216,'+(.22*pulse)+')';
      ctx.beginPath();ctx.arc(0,0,15,0,TAU);ctx.fill();
      ctx.globalCompositeOperation='source-over';
      ctx.rotate(S.t*1.6);
      ctx.beginPath();ctx.moveTo(0,-9);ctx.lineTo(6,0);ctx.lineTo(0,9);ctx.lineTo(-6,0);ctx.closePath();
      ctx.fillStyle='#ff4fd8';ctx.fill();ctx.strokeStyle='#ffd0f4';ctx.lineWidth=1.5;ctx.stroke();
      ctx.beginPath();ctx.moveTo(0,-4);ctx.lineTo(2.6,0);ctx.lineTo(0,4);ctx.lineTo(-2.6,0);ctx.closePath();ctx.fillStyle='#fff';ctx.fill();
    }else{
      ctx.rotate(S.t*.8);
      ctx.strokeStyle='rgba(143,232,255,.9)';ctx.lineWidth=2;
      ctx.setLineDash([7,5]);ctx.lineDashOffset=-S.t*30;
      ctx.beginPath();ctx.arc(0,0,9.5,0,TAU);ctx.stroke();ctx.setLineDash([]);
      ctx.fillStyle='#8fe8ff';ctx.beginPath();ctx.arc(0,0,2.6,0,TAU);ctx.fill();
    }
    ctx.restore();
  }
}
function drawRocks(){
  for(const r of rocks){
    ctx.save();ctx.translate(r.x,r.y);ctx.rotate(r.rot);
    ctx.beginPath();const n=r.verts.length;
    for(let i=0;i<n;i++){const a=i/n*TAU,rad=r.r*r.verts[i];const x=Math.cos(a)*rad,y=Math.sin(a)*rad;i?ctx.lineTo(x,y):ctx.moveTo(x,y);}
    ctx.closePath();
    ctx.fillStyle='rgba(43,28,15,.93)';ctx.fill();
    ctx.lineWidth=r.size===3?3:2;ctx.strokeStyle=STROKE[r.size];ctx.stroke();
    if(r.size===3&&r.hp<3){
      ctx.strokeStyle='rgba(15,8,3,.65)';ctx.lineWidth=1.5;
      for(let c=0;c<3-r.hp;c++){const a=c*2.4+1;
        ctx.beginPath();
        ctx.moveTo(Math.cos(a)*r.r*.1,Math.sin(a)*r.r*.1);
        ctx.lineTo(Math.cos(a+.35)*r.r*.55,Math.sin(a+.35)*r.r*.55);
        ctx.lineTo(Math.cos(a-.2)*r.r*.9,Math.sin(a-.2)*r.r*.9);
        ctx.stroke();}
    }
    ctx.restore();
  }
}
function drawSnake(){
  const sn=snake; if(!sn)return;
  const n=sn.segs.length;
  if(S.hurtT>0&&Math.floor(S.t*16)%2)ctx.globalAlpha=.35;
  ctx.globalCompositeOperation='lighter';
  for(let i=n-1;i>=0;i--){const s=sn.segs[i];ctx.fillStyle='rgba(45,230,190,.07)';ctx.beginPath();ctx.arc(s.x,s.y,bodyR(i)*2.4,0,TAU);ctx.fill();}
  ctx.globalCompositeOperation='source-over';
  for(let i=n-1;i>=0;i--){const s=sn.segs[i],t=n>1?i/(n-1):0;
    ctx.fillStyle='hsl('+(168-t*38)+' 85% '+(62-t*18)+'%)';
    ctx.beginPath();ctx.arc(s.x,s.y,bodyR(i),0,TAU);ctx.fill();}
  ctx.save();ctx.translate(sn.x,sn.y);ctx.rotate(sn.a);
  if(S.thrusting&&S.mode==='playing'){
    const fl=10+7*Math.sin(S.t*42)+rand(0,5);
    ctx.fillStyle='rgba(255,170,60,.85)';
    ctx.beginPath();ctx.moveTo(-6,4.5);ctx.lineTo(-6-fl,0);ctx.lineTo(-6,-4.5);ctx.closePath();ctx.fill();
    ctx.fillStyle='rgba(255,240,200,.9)';
    ctx.beginPath();ctx.moveTo(-6,2.4);ctx.lineTo(-6-fl*.55,0);ctx.lineTo(-6,-2.4);ctx.closePath();ctx.fill();
  }
  ctx.beginPath();ctx.moveTo(13,0);ctx.lineTo(-8,7.5);ctx.lineTo(-4,0);ctx.lineTo(-8,-7.5);ctx.closePath();
  ctx.fillStyle='#eafffb';ctx.fill();ctx.strokeStyle='rgba(61,242,207,.9)';ctx.lineWidth=1.5;ctx.stroke();
  ctx.fillStyle='#3df2cf';ctx.beginPath();ctx.arc(3,0,2,0,TAU);ctx.fill();
  ctx.restore();
  if(S.shieldT>0){
    ctx.save();ctx.translate(sn.x,sn.y);
    ctx.strokeStyle='rgba(143,232,255,.22)';ctx.lineWidth=2;
    ctx.beginPath();ctx.arc(0,0,17,0,TAU);ctx.stroke();
    ctx.strokeStyle='#8fe8ff';ctx.setLineDash([6,6]);ctx.lineDashOffset=-S.t*40;
    ctx.beginPath();ctx.arc(0,0,17,-Math.PI/2,-Math.PI/2+TAU*(S.shieldT/SHIELD_DUR));ctx.stroke();
    ctx.setLineDash([]);ctx.restore();
  }
  ctx.globalAlpha=1;
}
function drawParts(){
  ctx.globalCompositeOperation='lighter';
  for(const p of parts){const a=clamp(p.life/p.ml,0,1);ctx.globalAlpha=a;ctx.fillStyle=p.col;ctx.beginPath();ctx.arc(p.x,p.y,p.size*a+.4,0,TAU);ctx.fill();}
  ctx.globalAlpha=1;ctx.globalCompositeOperation='source-over';
}
function drawPops(){
  ctx.textAlign='center';
  for(const p of pops){const k=p.t/p.dur,a=k<.6?1:1-(k-.6)/.4;
    ctx.globalAlpha=Math.max(0,a);ctx.font='700 '+p.size+'px '+FMONO;
    ctx.shadowColor='rgba(0,0,0,.85)';ctx.shadowBlur=5;ctx.fillStyle=p.col;
    ctx.fillText(p.txt,p.x,p.y-34*k);ctx.shadowBlur=0;}
  ctx.globalAlpha=1;
}
function drawMsg(){
  if(!S.msg)return;const m=S.msg;
  const a=Math.min(clamp(m.t/.25,0,1),clamp((m.dur-m.t)/.5,0,1));
  const sc=.9+.1*clamp(m.t/.25,0,1);
  ctx.save();ctx.globalAlpha=a;ctx.translate(W/2,H*.34);ctx.scale(sc,sc);
  ctx.font='46px '+FDISP;ctx.textAlign='center';
  try{ctx.letterSpacing='8px';}catch(e){}
  ctx.shadowColor='rgba(255,179,92,.8)';ctx.shadowBlur=24;
  ctx.fillStyle='#ffb35c';ctx.fillText(m.txt,0,0);
  ctx.restore();
}
function render(){
  ctx.fillStyle=bgGrad;ctx.fillRect(0,0,W,H);
  ctx.globalCompositeOperation='lighter';
  neb(W*.3+Math.sin(S.t*.03)*60,H*.35+Math.cos(S.t*.021)*40,Math.max(W,H)*.5,'rgba(45,190,170,.045)');
  neb(W*.7+Math.cos(S.t*.026)*70,H*.65+Math.sin(S.t*.031)*50,Math.max(W,H)*.45,'rgba(255,150,80,.035)');
  neb(W*.55,H*.2+Math.sin(S.t*.02)*30,Math.max(W,H)*.4,'rgba(90,120,255,.04)');
  ctx.globalCompositeOperation='source-over';
  for(const s of stars){
    const x=(s.x+S.t*7*s.z)%W;
    ctx.globalAlpha=(.25+.55*(.5+.5*Math.sin(S.t*s.sp+s.ph)))*(.35+.65*s.z);
    ctx.fillStyle=s.r>1?'#cfe8ff':'#9db8e0';
    if(s.r>1){ctx.beginPath();ctx.arc(x,s.y,1.4,0,TAU);ctx.fill();}else ctx.fillRect(x,s.y,1.5,1.5);
  }
  ctx.globalAlpha=1;
  ctx.save();
  if(S.shake>0)ctx.translate(rand(-S.shake,S.shake)*.5,rand(-S.shake,S.shake)*.5);
  drawPicks();drawRocks();
  if(snake&&(S.mode==='playing'||S.mode==='paused'))drawSnake();
  drawParts();drawPops();
  ctx.restore();
  if(vig)ctx.drawImage(vig,0,0,W,H);
  if(S.hurtT>0&&vigR){ctx.globalAlpha=Math.min(1,S.hurtT)*.5;ctx.drawImage(vigR,0,0,W,H);ctx.globalAlpha=1;}
  drawMsg();
}
