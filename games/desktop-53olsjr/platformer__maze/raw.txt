  const cols=13+d*2,rows=9+(d-1)*2;
  const grid=Array.from({length:rows},()=>new Array(cols).fill(SOLID));
  // carve (iterative backtracker)
  const st=[[1,1]];grid[1][1]=EMPTY;const D=[[2,0],[-2,0],[0,2],[0,-2]];
  while(st.length){const[x,y]=st[st.length-1];const opts=[];
    for(const[dx,dy]of D){const nx=x+dx,ny=y+dy;
      if(nx>0&&nx<cols-1&&ny>0&&ny<rows-1&&grid[ny][nx]===SOLID)opts.push([nx,ny,dx,dy]);}
    if(!opts.length){st.pop();continue;}
    const[nx,ny,dx,dy]=opts[(Math.random()*opts.length)|0];
    grid[y+dy/2][x+dx/2]=EMPTY;grid[ny][nx]=EMPTY;st.push([nx,ny]);}
  function bfs(sx,sy){const dist=Array.from({length:rows},()=>new Array(cols).fill(-1));
    const q=[[sx,sy]];dist[sy][sx]=0;
    for(let i=0;i<q.length;i++){const[x,y]=q[i];
      for(const[dx,dy]of[[1,0],[-1,0],[0,1],[0,-1]]){const nx=x+dx,ny=y+dy;
        if(nx>=0&&nx<cols&&ny>=0&&ny<rows&&grid[ny][nx]===EMPTY&&dist[ny][nx]<0){
          dist[ny][nx]=dist[y][x]+1;q.push([nx,ny]);}}}
    return dist;}
  // dead ends → pick farthest pair for start / exit
  const dead=[];
  for(let y=1;y<rows-1;y++)for(let x=1;x<cols-1;x++){if(grid[y][x]!==EMPTY)continue;let n=0;
    for(const[dx,dy]of[[1,0],[-1,0],[0,1],[0,-1]])if(grid[y+dy]&&grid[y+dy][x+dx]===EMPTY)n++;
    if(n===1)dead.push([x,y]);}
  const d0=bfs(1,1);let A=[1,1],bd=-1;
  for(const[x,y]of dead)if(d0[y][x]>bd){bd=d0[y][x];A=[x,y];}
  const dB=bfs(A[0],A[1]);let B=A;bd=-1;const pool=dead.length?dead:[[cols-2,rows-2]];
  for(const[x,y]of pool)if(dB[y][x]>bd){bd=dB[y][x];B=[x,y];}
  const start=A,exit=B,dS=bfs(start[0],start[1]);
  // lava pits (jumpable 1-tile holes)
  {const k=Math.min(2+((d*.7)|0),99);const cands=[];
   for(let y=1;y<rows-2;y++)for(let x=1;x<cols-1;x++){
     if(grid[y][x]!==EMPTY||grid[y+1][x]!==SOLID)continue;
     if(Math.abs(x-start[0])<=2&&Math.abs(y-start[1])<=2)continue;
     if(Math.abs(x-exit[0])<=2&&Math.abs(y-exit[1])<=2)continue;
     if(dS[y][x]>=6)cands.push([x,y]);}
   shuffle(cands);const pits=[];
   for(const[x,y]of cands){if(pits.length>=k)break;
     if(pits.some(p=>Math.abs(p[0]-x)<=1&&Math.abs(p[1]-y)<=1))continue;
     pits.push([x,y]);grid[y+1][x]=LAVA;}}
  // low (jumpable / step-able) wall tiles — never under a corridor ceiling
  {const p=.10+d*.012;
   for(let y=1;y<rows-1;y++)for(let x=1;x<cols-1;x++)
     if(grid[y][x]===SOLID&&grid[y+1][x]!==EMPTY&&Math.random()<p)grid[y][x]=LOW;}
  // coals
  let coals=[];{const n=4+d;const cands=[];
   for(let y=1;y<rows-1;y++)for(let x=1;x<cols-1;x++){
     if(grid[y][x]!==EMPTY)continue;
     if(Math.abs(x-start[0])<=1&&Math.abs(y-start[1])<=1)continue;
     if(Math.abs(x-exit[0])<=1&&Math.abs(y-exit[1])<=1)continue;
     if(grid[y+1][x]===LAVA||dS[y][x]<3)continue;cands.push([x,y]);}
   shuffle(cands);const picked=[];
   for(const c of cands){if(picked.length>=n)break;
     if(picked.some(p=>Math.abs(p[0]-c[0])+Math.abs(p[1]-c[1])<2))continue;picked.push(c);}
   coals=picked.map(([x,y],i)=>({x:x*T+T/2,y:y*T+T/2,got:false,ph:i*1.37}));}
  // wraiths
  let enemies=[];{const n=1+Math.ceil(d*.9);const cands=[];
   for(let y=1;y<rows-1;y++)for(let x=2;x<cols-2;x++){
     if(grid[y][x]!==EMPTY||dS[y][x]<7)continue;
     if(grid[y+1][x]!==SOLID||grid[y+1][x-1]!==SOLID||grid[y+1][x+1]!==SOLID)continue;
     if(Math.abs(x-start[0])<=2&&Math.abs(y-start[1])<=2)continue;cands.push([x,y]);}
   shuffle(cands);const picked=[];
   for(const c of cands){if(picked.length>=n)break;
     if(picked.some(p=>Math.max(Math.abs(p[0]-c[0]),Math.abs(p[1]-c[1]))<3))continue;picked.push(c);}
   enemies=picked.map(([x,y])=>({x:x*T+(T-30)/2,y:y*T+T-24,w:30,h:24,
     dir:Math.random()<.5?-1:1,sp:52+d*7,ph:rand(6.28),dead:false}));}
  const pl={x:start[0]*T+(T-22)/2,y:start[1]*T+T-26,w:22,h:26,vx:0,vy:0,onGround:true,face:1,squash:0};
  G={cols,rows,grid,start,exit,coals,enemies,pl,gateOpen:false,got:0,time:0,zoom:1,
     cam:{x:pl.x+11,y:pl.y+13}};
  computeZoom();parts.length=0;invuln=1.2;jumpBuf=0;coyoteT=0;flash=0;shake=0;
}
function computeZoom(){const ww=G.cols*T,wh=G.rows*T;
  G.zoom=clamp(Math.max(vw/ww,vh/wh),.5,2.4);}

/* ================= collision ================= */
function solidRect(tx,ty){if(!G)return null;
  if(tx<0||ty<0||tx>=G.cols||ty>=G.rows)return{x:tx*T,y:ty*T,w:T,h:T};
  const g=G.grid[ty][tx];
  if(g===SOLID)return{x:tx*T,y:ty*T,w:T,h:T};
  if(g===LOW)return{x:tx*T,y:ty*T+T/2,w:T,h:T/2};
  return null;}
const overlap=(a,b)=>a.x<b.x+b.w&&a.x+a.w>b.x&&a.y<b.y+b.h&&a.y+a.h>b.y;
function freeAt(x,y,w,h){for(let ty=Math.floor(y/T);ty<=Math.floor((y+h-.01)/T);ty++)
  for(let tx=Math.floor(x/T);tx<=Math.floor((x+w-.01)/T);tx++){
    const b=solidRect(tx,ty);if(b&&overlap({x,y,w,h},b))return false;}return true;}
function moveX(p,dx){if(!dx)return;p.x+=dx;let g=0;
  while(g++<4){const r={x:p.x,y:p.y,w:p.w,h:p.h};let res=false;
    for(let ty=Math.floor(r.y/T);ty<=Math.floor((r.y+r.h-.01)/T);ty++)
      for(let tx=Math.floor(r.x/T);tx<=Math.floor((r.x+r.w-.01)/T);tx++){
        const b=solidRect(tx,ty);if(!b||!overlap(r,b))continue;
        const stepH=(p.y+p.h)-b.y;
        if(stepH>.5&&stepH<=26&&p.vy>-40&&(p.onGround||coyoteT>0)&&freeAt(p.x,b.y-p.h,p.w,p.h)){
          p.y=b.y-p.h;res=true;continue;}
        if(dx>0)p.x=b.x-p.w;else p.x=b.x+b.w;p.vx=0;res=true;break;}
    if(!res)break;}}
function moveY(p,dy){if(!dy)return;p.y+=dy;let g=0;
  while(g++<4){const r={x:p.x,y:p.y,w:p.w,h:p.h};let res=false;
    for(let ty=Math.floor(r.y/T);ty<=Math.floor((r.y+r.h-.01)/T);ty++)
      for(let tx=Math.floor(r.x/T);tx<=Math.floor((r.x+r.w-.01)/T);tx++){
        const b=solidRect(tx,ty);if(!b||!overlap(r,b))continue;
        if(dy>0){p.y=b.y-p.h;p.onGround=true;p.vy=0;}else{p.y=b.y+b.h;p.vy=0;}
        res=true;break;}
    if(!res)break;}}
function tileSolidAt(px,py){const tx=Math.floor(px/T),ty=Math.floor(py/T);
  if(tx<0||ty<0||tx>=G.cols||ty>=G.rows)return true;
  const g=G.grid[ty][tx];if(g===SOLID)return true;if(g===LOW)return py>ty*T+T/2;return false;}
const fullSolid=(tx,ty)=>tx>=0&&ty>=0&&tx<G.cols&&ty<G.rows&&G.grid[ty][tx]===SOLID;
function touchesLava(p){const r={x:p.x,y:p.y,w:p.w,h:p.h};
  for(let ty=Math.floor(r.y/T);ty<=Math.floor((r.y+r.h-.01)/T);ty++)
    for(let tx=Math.floor(r.x/T);tx<=Math.floor((r.x+r.w-.01)/T);tx++){
      if(tx<0||ty<0||tx>=G.cols||ty>=G.rows)continue;
      if(G.grid[ty][tx]===LAVA&&overlap(r,{x:tx*T,y:ty*T,w:T,h:T}))return true;}
  return false;}

/* ================= particles ================= */
function burst(x,y,hex,n){const c=hexRgb(hex);
  for(let i=0;i<n;i++)parts.push({x,y,vx:rand(-150,150),vy:rand(-200,60),
    life:rand(.4,.8),max:.8,s:rand(1.5,3.5),col:c,g:240});}
function dust(x,y,n){for(let i=0;i<n;i++)parts.push({x:x+rand(-8,8),y:y-2,
  vx:rand(-50,50),vy:rand(-70,-10),life:rand(.3,.6),max:.6,s:rand(1.5,3),col:'150,170,200',g:120});}
function popup(x,y,txt){parts.push({x,y,vx:0,vy:-46,life:.9,max:.9,txt,col:'255,210,122',s:0,g:0});}
function updateParts(dt){for(let i=parts.length-1;i>=0;i--){const q=parts[i];
  q.life-=dt;if(q.life<=0){parts.splice(i,1);continue;}
  q.x+=q.vx*dt;q.y+=q.vy*dt;q.vy+=(q.g||0)*dt;}
  if(parts.length>500)parts.splice(0,parts.length-500);}

/* ================= gameplay ================= */
const inputL=()=>keys.ArrowLeft||keys.KeyA||touchL;
const inputR=()=>keys.ArrowRight||keys.KeyD||touchR;
const inputJ=()=>keys.Space||keys.ArrowUp||keys.KeyW||touchJ;
function resetPlayerPos(){const p=G.pl;p.x=G.start[0]*T+(T-22)/2;p.y=G.start[1]*T+T-26;p.vx=0;p.vy=0;}

function hurt(src,e){lives--;syncHUD(true);flash=.4;shake=Math.max(shake,12);
  const p=G.pl,cx=p.x+p.w/2,cy=p.y+p.h/2;
  if(lives>0){
    if(src==='lava'){sfx.lava();burst(cx,cy,'#ff8a2e',22);resetPlayerPos();invuln=2.5;}
    else{sfx.hurt();p.vx=(cx<(e.x+e.w/2)?-1:1)*360;p.vy=-330;invuln=2;}}
  else{mode='dying';deathT=1.15;(src==='lava'?sfx.lava():sfx.hurt)();burst(cx,cy,'#ffb43c',36);}}

function levelClear(){const par=40+depth*25;
  const bonus=250+depth*100+Math.max(0,Math.round(par-G.time))*5;
  score+=bonus;sfx.clear();mode='clear';clearT=1.6;
  toast('DEPTH '+depth+' CLEARED','bonus +'+bonus);
  burst(G.exit[0]*T+T/2,G.exit[1]*T+T/2,'#3ef2b0',30);syncHUD(true);}

function sim(dt){const p=G.pl;G.time+=dt;invuln=Math.max(0,invuln-dt);
  // horizontal
  const acc=p.onGround?2600:1500;
  if(inputL()&&!inputR()){p.vx-=acc*dt;p.face=-1;}
  else if(inputR()&&!inputL()){p.vx+=acc*dt;p.face=1;}
  else p.vx-=p.vx*Math.min(1,(p.onGround?14:3)*dt);
  p.vx=clamp(p.vx,-300,300);
  // jump (buffer + coyote)
  jumpBuf=Math.max(0,jumpBuf-dt);
  if(jumpBuf>0&&(p.onGround||coyoteT>0)){p.vy=-520;p.onGround=false;coyoteT=0;jumpBuf=0;
    p.squash=-.3;sfx.jump();dust(p.x+p.w/2,p.y+p.h,6);}
  // gravity + integrate
  p.vy=Math.min(900,p.vy+1500*dt);
  const wasG=p.onGround,fallV=p.vy;
  moveX(p,p.vx*dt);p.onGround=false;moveY(p,p.vy*dt);
  coyoteT=p.onGround?.1:Math.max(0,coyoteT-dt);
  if(!wasG&&p.onGround){p.squash=.32;
    if(fallV>480){dust(p.x+p.w/2,p.y+p.h,10);sfx.land();shake=Math.max(shake,3);}}
  p.squash-=p.squash*Math.min(1,12*dt);
  // trail
  if((Math.abs(p.vx)>80||!p.onGround)&&Math.random()<dt*40)
    parts.push({x:p.x+p.w/2-p.face*8,y:p.y+p.h-rand(10),vx:-p.face*rand(20,50),
      vy:rand(-30,10),life:.5,max:.5,s:rand(1.5,3),col:'255,170,60',g:80});
  // lava
  if(touchesLava(p)){hurt('lava',null);return;}
  // coals
  for(const c of G.coals){if(c.got)continue;
    if(Math.abs(p.x+p.w/2-c.x)<26&&Math.abs(p.y+p.h/2-c.y)<30){
      c.got=true;G.got++;score+=100;runCoals++;sfx.coin();
      burst(c.x,c.y,'#ffcf5e',14);popup(c.x,c.y-14,'+100');syncHUD(true);
      if(G.got===G.coals.length&&!G.gateOpen){G.gateOpen=true;sfx.gate();
        toast('THE GATE OPENS','make for it — the depths are watching');
        burst(G.exit[0]*T+T/2,G.exit[1]*T+T/2,'#3ef2b0',26);}}}
  // gate
  const gx=G.exit[0]*T,gy=G.exit[1]*T;
  if(G.gateOpen){
    if(Math.random()<dt*10)parts.push({x:gx+T/2+rand(-14,14),y:gy+T-6,
      vx:rand(-8,8),vy:-rand(30,70),life:.9,max:.9,s:rand(1.5,3),col:'120,255,200',g:0});
    if(overlap({x:p.x,y:p.y,w:p.w,h:p.h},{x:gx+6,y:gy+6,w:T-12,h:T-12})){levelClear();return;}}
  // wraiths
  for(const e of G.enemies){if(e.dead)continue;e.ph+=dt*7;e.x+=e.dir*e.sp*dt;
    const front=e.dir>0?e.x+e.w+3:e.x-3;
    const blocked=tileSolidAt(front,e.y+e.h*.5);
    const groundOk=fullSolid(Math.floor(front/T),Math.floor((e.y+e.h+4)/T));
    if(blocked||!groundOk){e.dir*=-1;e.x+=e.dir*4;}
    if(Math.random()<dt*2)parts.push({x:e.x+rand(4,26),y:e.y+e.h-3,vx:0,vy:rand(20,50),
      life:.5,max:.5,s:1.6,col:'255,120,40',g:100});
    if(overlap({x:p.x,y:p.y,w:p.w,h:p.h},e)){
      if(p.vy>120&&(p.y+p.h)-e.y<16){e.dead=true;score+=150;runStomps++;sfx.stomp();
        burst(e.x+e.w/2,e.y+e.h/2,'#8fa3c4',16);popup(e.x+e.w/2,e.y-10,'+150');
        p.vy=-400;shake=Math.max(shake,5);syncHUD(true);}
      else if(invuln<=0)hurt('wraith',e);}}}

function camFollow(dt){const p=G.pl,z=G.zoom,vwz=vw/z,vhz=vh/z,ww=G.cols*T,wh=G.rows*T;
  let tx=p.x+p.w/2,ty=p.y+p.h/2-14;
  tx=ww>vwz?clamp(tx,vwz/2,ww-vwz/2):ww/2;
  ty=wh>vhz?clamp(ty,vhz/2,wh-vhz/2):wh/2;
  const k=Math.min(1,dt*8);G.cam.x+=(tx-G.cam.x)*k;G.cam.y+=(ty-G.cam.y)*k;}

const STONE=['rgb(36,45,60)','rgb(40,49,66)','rgb(43,52,70)','rgb(38,47,63)','rgb(41,50,68)'];
function nearLava(tx,ty){for(const[dx,dy]of[[1,0],[-1,0],[0,1],[0,-1]]){
  const nx=tx+dx,ny=ty+dy;if(nx<0||ny<0||nx>=G.cols||ny>=G.rows)continue;
  if(G.grid[ny][nx]===LAVA)return true;}return false;}
function drawStone(tx,ty,g){const x=tx*T,y=g===LOW?ty*T+T/2:ty*T,h=g===LOW?T/2:T;
  ctx.fillStyle=STONE[hash2(tx,ty)%5];ctx.fillRect(x,y,T,h);
  const up=ty>0?G.grid[ty-1][tx]:EMPTY;
  if(g===LOW||!(up===SOLID||up===LOW)){ctx.fillStyle='rgba(120,150,200,.32)';ctx.fillRect(x,y,T,3);}

function drawLavaTile(tx,ty){const x=tx*T,y=ty*T,t=animT;

    life:.6,max:.6,s:rand(1.5,3),col:'255,190,90',g:60});}
function drawGate(){const gx=G.exit[0]*T,gy=G.exit[1]*T,cx=gx+T/2,cy=gy+T/2,r=T*.32;

  if(Math.random()<dt*7&&amb.length<80)
    amb.push({x:rand(vw),y:vh+8,vy:-rand(14,42),ph:rand(6.28),s:rand(.8,2.6),t:0});
  for(let i=amb.length-1;i>=0;i--){const a=amb[i];a.t+=dt;a.y+=a.vy*dt;
    a.x+=Math.sin(a.t*1.7+a.ph)*14*dt;if(a.y<-12)amb.splice(i,1);}
  hudTick-=dt;if(hudTick<=0){hudTick=.15;syncHUD(false);}
  if(mode==='playing'){sim(dt);updateParts(dt);camFollow(dt);}
  else if(mode==='dying'){deathT-=dt;updateParts(dt);
