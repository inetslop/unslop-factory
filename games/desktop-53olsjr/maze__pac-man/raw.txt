/* ================= movement core ================= */
function isWall(c,r){
  if(r===TUNNEL&&(c<0||c>=MW))return false;
  if(c<0||c>=MW||r<0||r>=MH)return true;
  const ch=MAZE[r][c]; return ch==='#'||ch==='-';
}
function wrapX(m){ if(m.x<-T/2)m.x+=MW*T; else if(m.x>LW+T/2)m.x-=MW*T; }
const wrapC=c=>((c%MW)+MW)%MW;

function advance(m,dist,cb){
  let rem=dist, guard=8;
  while(rem>1e-4&&guard--){
    if(!m.dir)return;
    const c=Math.floor(m.x/T), r=Math.floor(m.y/T);
    let d;
    if(m.dir.dx===1)d=(c+1.5)*T-m.x;
    else if(m.dir.dx===-1)d=m.x-(c-0.5)*T;
    else if(m.dir.dy===1)d=(r+1.5)*T-m.y;
    else d=m.y-(r-0.5)*T;
    if(rem<d){ if(m.dir.dx)m.x+=m.dir.dx*rem; else m.y+=m.dir.dy*rem; rem=0; }
    else{
      if(m.dir.dx)m.x=(c+m.dir.dx+0.5)*T; else m.y=(r+m.dir.dy+0.5)*T;
      rem-=d;
      const nc=c+m.dir.dx, nr=r+m.dir.dy;
      wrapX(m);
      if(cb(m,nc,nr)===false){m.dir=null;return;}
    }
  }
}
function pacCenter(m,c,r){
  const w=pac.want;
  if(w&&w.dx!==pac.dir.dx&&w.dy!==pac.dir.dy!==true){/* noop guard */}
  if(w&&(w.dx!==pac.dir.dx||w.dy!==pac.dir.dy)){
    if(!isWall(c+w.dx,r+w.dy)){pac.dir={dx:w.dx,dy:w.dy};return true;}
  }
  return isWall(c+pac.dir.dx,r+pac.dir.dy)?false:true;
}
function pickDir(opts,c,r,g,target){
  if(!target||!g.dir)return opts[(Math.random()*opts.length)|0];
  const rx=-g.dir.dx, ry=-g.dir.dy;
  let cand=opts.filter(d=>!(d.dx===rx&&d.dy===ry));
  if(!cand.length)cand=opts;
  let best=cand[0],bd=1e18;
  for(const d of cand){const a=(c+d.dx-target.c)*(c+d.dx-target.c)+(r+d.dy-target.r)*(r+d.dy-target.r);if(a<bd){bd=a;best=d;}}
  return best;
}
function ghostTarget(g){
  if(g.fright)return null;
  const pc={c:wrapC(Math.floor(pac.x/T)),r:Math.floor(pac.y/T)};
  const ahead=n=>({c:wrapC(pc.c+pac.dir.dx*n),r:pc.r+pac.dir.dy*n});
  switch(g.name){
    case 'BLINKY': return ghostMode==='scatter'?g.sc:pc;
    case 'PINKY':  return ghostMode==='scatter'?g.sc:ahead(4);
    case 'INKY':{
      if(ghostMode==='scatter')return g.sc;
      const p=ahead(2), b=ghosts[0];
      return {c:2*p.c-Math.floor(b.x/T), r:2*p.r-Math.floor(b.y/T)};
    }
    case 'CLYDE':{
      const gc={c:wrapC(Math.floor(g.x/T)),r:Math.floor(g.y/T)};
      return (Math.abs(gc.c-pc.c)+Math.abs(gc.r-pc.r))>8?pc:g.sc;
    }
  }
}

function checkExtra(){
  if(!extraGiven&&score>=10000){extraGiven=true;lives++;sfx.extra();shake=Math.max(shake,4);
    popups.push({x:pac.x,y:pac.y-16,t:0,text:'EXTRA LIFE!',col:'#ffe14d'});}
}
function powerUp(){
  frightT=frightDur();chain=0;shake=Math.max(shake,4);sfx.power();
  rings.push({x:pac.x,y:pac.y,t:0});
  for(const g of ghosts){ if(g.state==='eyes'||g.state==='return')continue;
    g.fright=true; if(g.state==='out'&&g.dir)g.dir={dx:-g.dir.dx,dy:-g.dir.dy}; }
}
function endFright(){
  for(const g of ghosts){g.fright=false;if(g.state==='out'&&g.dir)g.dir={dx:-g.dir.dx,dy:-g.dir.dy};}
  sfx.blip(320);
}
function updatePac(dt){
  const w=pac.want;
  if(pac.dir&&w&&w.dx===-pac.dir.dx&&w.dy===-pac.dir.dy)pac.dir={dx:w.dx,dy:w.dy};
  if(!pac.dir&&w&&!isWall(Math.floor(pac.x/T)+w.dx,Math.floor(pac.y/T)+w.dy))pac.dir={dx:w.dx,dy:w.dy};
  if(pac.dir){pac.face=Math.atan2(pac.dir.dy,pac.dir.dx);advance(pac,pacSp()*dt,pacCenter);}
  wrapX(pac);
  const c=Math.floor(pac.x/T),r=Math.floor(pac.y/T);
  if(c>=0&&c<MW&&r>=0&&r<MH){
    const v=dotGrid[r][c];
    if(v){dotGrid[r][c]=0;dots--;
      if(v===1){score+=10;sfx.waka();}else{score+=50;powerUp();}
      checkExtra();}
  }
}
function updateGhosts(dt){
  for(const g of ghosts){
    if(freezeT>0)break;
    g.phase+=dt*(g.state==='out'?(g.fright?7:10):6);
    switch(g.state){
      case 'house': g.bob+=dt*4; g.y=12.5*T+Math.sin(g.bob)*3.5; g.release-=dt; if(g.release<=0)g.state='exit'; break;
      case 'exit':{
        const sp=gSp()*0.9,hx=9.5*T;
        if(Math.abs(g.x-hx)>1){g.x+=Math.sign(hx-g.x)*sp*dt;}
        else{g.x=hx;g.y-=sp*dt;if(g.y<=10.5*T){g.y=10.5*T;g.state='out';g.dir={dx:0,dy:-1};}}
        break;}
      case 'return':{
        const sp=eyesSp*0.8,hx=9.5*T;
        if(Math.abs(g.x-hx)>1)g.x+=Math.sign(hx-g.x)*sp*dt;
        else{g.x=hx;g.y+=sp*dt;if(g.y>=12.5*T){g.y=12.5*T;g.state='house';g.release=0.8;g.fright=false;}}
        break;}
      case 'eyes':
        advance(g,eyesSp*dt,(m,c,r)=>{
          if(c===9&&r===TUNNEL){m.dir=null;return false;}
          const opts=DIRS.filter(d=>!isWall(c+d.dx,r+d.dy));
          m.dir=pickDir(opts,c,r,g,{c:9,r:TUNNEL}); return true;});
        break;
      case 'out':{
        const sp=g.fright?frightSp:gSp();
        advance(g,sp*dt,(m,c,r)=>{
          if(!g.dir){m.dir=DIRS[(Math.random()*4)|0];return true;}
          const opts=DIRS.filter(d=>!isWall(c+d.dx,r+d.dy));
          if(!opts.length){m.dir={dx:-m.dir.dx,dy:-m.dir.dy};return true;}
          m.dir=pickDir(opts,c,r,g,ghostTarget(g)); return true;});
        break;}
    }
  }
}
function burst(x,y,col,n){
  for(let i=0;i<n;i++){const a=Math.random()*TAU,v=40+Math.random()*120;
    parts.push({x,y,vx:Math.cos(a)*v,vy:Math.sin(a)*v,t:0,life:0.45+Math.random()*0.35,col});}
}
function collisions(){
  for(const g of ghosts){
    if(freezeT>0)break;
    if(g.state!=='out'&&g.state!=='exit')continue;
    const dx=g.x-pac.x,dy=g.y-pac.y;
    if(dx*dx+dy*dy<(T*0.55)*(T*0.55)){
      if(g.fright){
        g.fright=false;g.state='eyes';freezeT=0.9;
        const pts=[200,400,800,1600][Math.min(3,chain++)];
        score+=pts;shake=Math.max(shake,5);sfx.ghost();burst(g.x,g.y,g.color,14);
        popups.push({x:g.x,y:g.y-12,t:0,text:''+pts,col:'#7df9ff'});checkExtra();
      }else{
        state='dying';stateT=0;sfx.death();shake=7;burst(pac.x,pac.y,'#ffe14d',18);break;
      }
    }
  }
}
function sirenUpdate(){
  const A=AudioSys; if(!A.ctx)return;
  let f=0,rate=0,depth=0,vol=0;
  if(!paused){
    if(state==='play'){
      if(frightT>0){f=150;rate=9;depth=60;vol=0.05;}
      else if(ghostMode==='chase'){f=430;rate=3.8;depth=170;vol=0.042;}
      else{f=300;rate=2.4;depth=110;vol=0.035;}
    }else if(state==='ready'){f=240;rate=2;depth=60;vol=0.025;}
  }
  A.siren(f,rate,depth,vol);
}
function update(dt){
  time+=dt;
  for(const s of stars){s.y+=s.z*7*dt;s.x-=s.z*2*dt;if(s.y>LH)s.y-=LH;if(s.x<0)s.x+=LW;}
  shake=Math.max(0,shake-dt*16);
  for(let i=popups.length-1;i>=0;i--){popups[i].t+=dt;if(popups[i].t>0.95)popups.splice(i,1);}
  for(let i=rings.length-1;i>=0;i--){rings[i].t+=dt;if(rings[i].t>0.45)rings.splice(i,1);}
  for(let i=parts.length-1;i>=0;i--){const p=parts[i];p.t+=dt;p.x+=p.vx*dt;p.y+=p.vy*dt;
    const dr=Math.max(0,1-2.5*dt);p.vx*=dr;p.vy*=dr;if(p.t>p.life)parts.splice(i,1);}
  if(paused){sirenUpdate();return;}
  switch(state){
    case 'ready': stateT-=dt; if(stateT<=0)state='play'; break;
    case 'play':{
      pacInvuln=Math.max(0,pacInvuln-dt);
      updatePac(dt);
      if(freezeT>0)freezeT-=dt;
      else if(frightT>0){frightT-=dt;if(frightT<=0)endFright();}
      else{
        modeT+=dt; const seg=schedule[modeIdx];
        if(isFinite(seg.d)&&modeT>=seg.d){modeT=0;modeIdx++;ghostMode=schedule[modeIdx].m;
          for(const g of ghosts)if(g.state==='out'&&g.dir)g.dir={dx:-g.dir.dx,dy:-g.dir.dy};}
      }
      updateGhosts(dt);
      collisions();
      if(state==='play'&&dots===0){state='clear';stateT=0;sfx.clear();}
      break;}
    case 'dying':
      stateT+=dt;
      if(stateT>=2.4){
        lives--;
        if(lives<=0){state='over';if(score>hi){hi=score;try{localStorage.setItem('neonpac_hi',String(hi))}catch(e){}}sfx.gameover();}
        else{resetPositions();state='ready';stateT=1.6;}
      }
      break;
    case 'clear': stateT+=dt; if(stateT>=3)nextLevel(); break;
  }
  sirenUpdate();
}

function drawPac(x,y,ang,mouth,r,alpha){
  if(r<0.5)return;
  ctx.save();ctx.translate(x,y);ctx.rotate(ang);ctx.globalAlpha*=alpha;
  ctx.fillStyle='#ffe14d';ctx.shadowColor='rgba(255,220,90,.85)';ctx.shadowBlur=14;

function ghostEyes(x,y,r,g){
  const d=g.dir||{dx:0,dy:0};
  for(const sgn of [-1,1]){
    ctx.fillStyle='#fff';
    ctx.beginPath();ctx.ellipse(x+sgn*r*0.4,y-3,r*0.30,r*0.38,0,0,TAU);ctx.fill();
    ctx.fillStyle='#2030c8';
    ctx.beginPath();ctx.arc(x+sgn*r*0.4+d.dx*r*0.16,y-3+d.dy*r*0.16,r*0.17,0,TAU);ctx.fill();
  }
}
function drawGhost(g){
  const x=g.x,y=g.y,r=T*0.46;
  if(g.state==='eyes'){
    ctx.save();ctx.shadowColor='rgba(255,255,255,.8)';ctx.shadowBlur=6;
    ghostEyes(x,y-2,r*1.1,g);ctx.restore();return;
  }
  let body=g.color;
  const flash=frightT<2&&g.fright&&(Math.floor(frightT*(frightT<0.75?14:8))%2===0);
  if(g.fright)body=flash?'#eef3ff':'#2b3cff';
  ctx.save();
  ctx.shadowColor=g.fright?(flash?'#ffffff':'#3a52ff'):g.color;ctx.shadowBlur=10;
  ctx.fillStyle=body;
  const domeY=y-3,by=y+r*0.98,tw=(2*r)/3;
  ctx.beginPath();
  ctx.arc(x,domeY,r,Math.PI,0);
  ctx.lineTo(x+r,by);
  for(let i=0;i<3;i++){const x1=x+r-tw*i,lift=((i+Math.floor(g.phase))%2===0)?5:1.5;
    ctx.lineTo(x1-tw/2,by-lift);ctx.lineTo(x1-tw,by);}
  ctx.closePath();ctx.fill();
  ctx.shadowBlur=0;
  if(g.fright){
    const fc=flash?'#5a60c8':'#ffe2c4';
    ctx.fillStyle=fc;
    ctx.beginPath();ctx.arc(x-r*0.38,y-4,2.1,0,TAU);ctx.fill();
    ctx.beginPath();ctx.arc(x+r*0.38,y-4,2.1,0,TAU);ctx.fill();
    ctx.strokeStyle=fc;ctx.lineWidth=1.6;ctx.beginPath();
    const m0=x-r*0.5,m1=x+r*0.5,seg=(m1-m0)/4;
    let px=m0,py=y+4;ctx.moveTo(px,py);
    for(let i=1;i<=4;i++){px=m0+seg*i;py=(i%2===1)?y+7:y+3;ctx.lineTo(px,py);}
    ctx.stroke();
  }else ghostEyes(x,y-2,r,g);
  ctx.restore();
}
function drawPacEntity(){
  let ang=pac.face,mouth=0.35,r=T*0.46,alpha=1;
  if(state==='play'){if(!paused)mouth=0.12+0.75*Math.abs(Math.sin(time*9));
    if(pacInvuln>0)alpha=Math.floor(time*12)%2?0.35:1;}
  if(state==='dying'){
    const t=stateT;
    if(t<0.9){mouth=0.3+(t/0.9)*2.4;}
    else{const k=Math.min(1,(t-0.9)/1.4);r=T*0.46*(1-k);alpha=1-k;mouth=0.05;ang=t*7;}
  }
  drawPac(pac.x,pac.y,ang,mouth,r,alpha);
}

  }
  if(state==='over'){
    ctx.fillStyle='rgba(3,0,14,.6)';ctx.fillRect(0,HUD_TOP,LW,MH*T);
    const fl=0.7+0.3*Math.sin(time*9);

  if(paused){
    ctx.fillStyle='rgba(3,0,14,.62)';ctx.fillRect(0,HUD_TOP,LW,MH*T);

  }
}
