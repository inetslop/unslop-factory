**Match detection code**:
```js
function findMatches(){
  const seen = Array.from({length:H},()=>Array(W).fill(false));
  const groups=[];
  for(let r=0;r<H;r++)for(let c=0;c<W;c++){
    const v=grid[r][c];
    if(v==null||seen[r][c])continue;
    // BFS
    const stack=[[r,c]]; seen[r][c]=true; const cells=[]; let centroid=[0,0];
    while(stack.length){
      const [cr,cc]=stack.pop(); cells.push([cr,cc]);
      for(const [dr,dc] of [[1,0],[-1,0],[0,1],[0,-1]]){
        const nr=cr+dr,nc=cc+dc;
        if(nr<0||nr>=H||nc<0||nc>=W)continue;
        if(seen[nr][nc]||grid[nr][nc]!==v)continue;
        seen[nr][nc]=true; stack.push([nr,nc]);
      }
    }
    if(cells.length>=3) groups.push({color:v,cells});
  }
  return groups;
}
```

```js
function stepGame(dt){
```

```html
    dropTimer+=dt;
    while(dropTimer>=iv&&phase==='drop'&&piece){ dropTimer-=iv; if(!tryMove(0,1)){ lockPiece(); break; } }
  } else if(phase==='clear'){ phaseT+=dt; if(phaseT>=CLEAR_MS) endClear(); }
  else if(phase==='fall'){ phaseT+=dt; if(phaseT>=fallDur) endFall(); }
}
function aiStep(dt){
  if(!piece) return;
  aiT-=dt; if(aiT>0) return; aiT=240+Math.random()*300;
  if(Math.random()<.16) tryRotate(1);
  const dx=Math.sign(aiCol-piece.x); if(dx&&Math.random()<.85) tryMove(dx,0);
  if(piece.y>=7&&Math.abs(piece.x-aiCol)<=1&&Math.random()<.4) hardDrop();
  else if(piece.y>=H-3) hardDrop();
}

}

function drawBlock(g,x,y,s,ci,alpha,sc){
  const p=PAL[ci];
  if(sc!==1){ const cs=s*sc; x+=(s-cs)/2; y+=(s-cs)/2; s=cs; }
  g.globalAlpha=alpha==null?1:alpha;
  rr(g,x+.5,y+.5,s-1,s-1,Math.max(2,s*.16)); g.fillStyle=p.base; g.fill();
  g.strokeStyle='rgba(0,0,0,.32)'; g.lineWidth=1; g.stroke();
  g.fillStyle=p.light; rr(g,x+2,y+2,s-4,(s-4)*.3,Math.max(2,s*.14)); g.fill();
  g.fillStyle=p.dark; rr(g,x+2,y+s-(s-4)*.26-2,s-4,(s-4)*.26,Math.max(2,s*.12)); g.fill();
  g.fillStyle='rgba(255,255,255,.7)'; g.beginPath(); g.arc(x+s*.3,y+s*.24,Math.max(1.2,s*.06),0,7); g.fill();
  g.globalAlpha=1;
}

function renderBoard(){
  const g=bctx; g.setTransform(RS,0,0,RS,0,0); g.clearRect(0,0,BW,BH);
  g.translate((Math.random()*2-1)*shake,(Math.random()*2-1)*shake);

  g.fillStyle='#0d1a2b'; g.fillRect(-4,-4,BW+8,BH+8);
  for(let r=0;r<H;r++)for(let c=0;c<W;c++){ if((r+c)&1){ g.fillStyle='rgba(255,255,255,.022)'; g.fillRect(c*CELL,r*CELL,CELL,CELL);} }
  g.strokeStyle='rgba(255,255,255,.04)'; g.lineWidth=1; g.beginPath();
  for(let c=1;c<W;c++){ g.moveTo(c*CELL+.5,0); g.lineTo(c*CELL+.5,BH); }
  for(let r=1;r<H;r++){ g.moveTo(0,r*CELL+.5); g.lineTo(BW,r*CELL+.5); }
  g.stroke();
  if(dangerOn){ const a=.09+.06*Math.sin(tGlobal*.009);
    const gr=g.createLinearGradient(0,0,0,CELL*4);
    gr.addColorStop(0,'rgba(255,93,82,'+(a+.1).toFixed(3)+')'); gr.addColorStop(1,'rgba(255,93,82,0)');
    g.fillStyle=gr; g.fillRect(0,0,BW,CELL*4); }
  // static cells
  for(let r=0;r<H;r++)for(let c=0;c<W;c++){ const v=grid[r][c]; if(v==null) continue;
    if(skipStatic.has(r*W+c)) continue; drawBlock(g,c*CELL,r*CELL,CELL,v); }
  // clearing
  if(phase==='clear'){
    const q=Math.min(1,phaseT/CLEAR_MS);
    for(let i=0;i<clearKeys.length;i++){ const k=clearKeys[i], r=(k/W)|0, c=k%W;
      let sc,al;
      if(q<.35){ sc=1+(q/.35)*.28; al=1; } else { sc=Math.max(.01,1-(q-.35)/.65); al=sc; }
      drawBlock(g,c*CELL,r*CELL,CELL,clearCols[i],al,sc);
      if(q<.35){ g.globalAlpha=(1-q/.35)*.75; g.fillStyle='#fff'; rr(g,c*CELL+2,r*CELL+2,CELL-4,CELL-4,5); g.fill(); g.globalAlpha=1; } }
  }
  // falling
  if(phase==='fall'){
    fallList.forEach(f=>{ const p=Math.min(1,phaseT/f.dur), e=p*p;
      drawBlock(g,f.c*CELL,(f.from+(f.to-f.from)*e)*CELL,CELL,f.color); });
  }
  // ghost + piece
  if(piece&&phase==='drop'){
    const gy=ghostY();
    if(gy!==piece.y){ pieceCells({...piece,y:gy}).forEach(rc=>{ const p=PAL[piece.color];
      g.globalAlpha=.09; g.fillStyle=p.base; rr(g,rc[1]*CELL+2,rc[0]*CELL+2,CELL-4,CELL-4,5); g.fill();
      g.globalAlpha=.4; g.strokeStyle=p.base; g.lineWidth=2; rr(g,rc[1]*CELL+3,rc[0]*CELL+3,CELL-6,CELL-6,4); g.stroke(); g.globalAlpha=1; }); }
    g.save(); g.shadowColor='rgba(0,0,0,.5)'; g.shadowBlur=10; g.shadowOffsetY=5;
    pieceCells(piece).forEach(rc=>drawBlock(g,rc[1]*CELL,rc[0]*CELL,CELL,piece.color));
    g.restore();
  }
  // particles
  parts.forEach(p=>{ const a=1-p.t/p.life; g.save(); g.translate(p.x,p.y); g.rotate(p.rot);
    g.globalAlpha=a; g.fillStyle=PAL[p.ci].base; const s=p.sz*(.35+.65*a); g.fillRect(-s/2,-s/2,s,s); g.restore(); });
  g.globalAlpha=1;
  // popups

    g.fillStyle=p.col; g.fillText(p.txt,p.x,p.y); });
  g.globalAlpha=1; g.textAlign='left';
}
```

**Tetris Match-3** — real tetrominoes, but instead of clearing lines you link up 3+ same-colour blocks to blast them. Cascades multiply your score, streaks pay bonuses, and the level ramps as you place pieces. The menu even runs a live AI demo behind it.

Controls: **← →** move · **↑/X** rotate · **↓** soft drop · **SPACE** hard drop · **P** pause · **M** sound · **R** restart. Best score persists locally (safely).
