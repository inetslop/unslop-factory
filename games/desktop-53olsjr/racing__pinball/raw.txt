function cpSeg(px,py,x1,y1,x2,y2){
  const dx=x2-x1,dy=y2-y1,L2=dx*dx+dy*dy;let t=((px-x1)*dx+(py-y1)*dy)/L2;
  t=t<0?0:t>1?1:t;return{x:x1+t*dx,y:y1+t*dy};
}
function orient(ax,ay,bx,by,cx,cy){return (bx-ax)*(cy-ay)-(by-ay)*(cx-ax);}
function segHit(p1x,p1y,p2x,p2y,g){
  const o1=orient(p1x,p1y,p2x,p2y,g.a.x,g.a.y),o2=orient(p1x,p1y,p2x,p2y,g.b.x,g.b.y);
  const o3=orient(g.a.x,g.a.y,g.b.x,g.b.y,p1x,p1y),o4=orient(g.a.x,g.a.y,g.b.x,g.b.y,p2x,p2y);
  return ((o1>0&&o2<0)||(o1<0&&o2>0))&&((o3>0&&o4<0)||(o3<0&&o4>0));
}
function moveFlipper(f,down,dt){
  const target=down?f.actA:f.restA;
  const rate=down?32:26;
  f.omega=0;
  if(Math.abs(target-f.ang)>1e-4){
    const step=Math.sign(target-f.ang)*Math.min(rate*dt,Math.abs(target-f.ang));
    f.omega=step/dt;f.ang+=step;
  }
}
function collideWalls(dt){
  for(const w of walls){
    const c=cpSeg(ball.x,ball.y,w.x1,w.y1,w.x2,w.y2);
    let dx=ball.x-c.x,dy=ball.y-c.y,d2=dx*dx+dy*dy;
    if(d2<BALL_R*BALL_R){
      const d=Math.sqrt(d2)||.0001,nx=dx/d,ny=dy/d;
      ball.x+=nx*(BALL_R-d);ball.y+=ny*(BALL_R-d);
      const vn=ball.vx*nx+ball.vy*ny;
      if(vn<0){ball.vx-=1.5*vn*nx;ball.vy-=1.5*vn*ny;}
    }
  }
}
function collideSlings(){
  for(const sl of slings){
    const c=cpSeg(ball.x,ball.y,sl.A.x,sl.A.y,sl.B.x,sl.B.y);
    let dx=ball.x-c.x,dy=ball.y-c.y,d2=dx*dx+dy*dy;
    if(d2<(BALL_R+2)*(BALL_R+2)){
      const d=Math.sqrt(d2)||.0001,nx=dx/d,ny=dy/d;
      ball.x+=nx*(BALL_R+2-d);ball.y+=ny*(BALL_R+2-d);
      const vn=ball.vx*nx+ball.vy*ny;
      if(vn<0){
        if(-vn>240){ // kick!
          const tx=-ny,ty=nx,vt=ball.vx*tx+ball.vy*ty;
          ball.vx=nx*520+tx*vt*.3;ball.vy=ny*520+ty*vt*.3;
          sl.flash=1;if(gameTime-sl.lastS>.1){sSling();sl.lastS=gameTime;}
          spawn(c.x,c.y,8,'#35e0ff',220);
        }else{ball.vx-=1.5*vn*nx;ball.vy-=1.5*vn*ny;}
      }
    }
  }
}
function collideBumpers(){
  for(const b of bumpers){
    let dx=ball.x-b.x,dy=ball.y-b.y,d=Math.hypot(dx,dy);
    if(d<BALL_R+b.r){
      const nx=dx/(d||.001),ny=dy/(d||.001);
      ball.x=b.x+nx*(BALL_R+b.r);ball.y=b.y+ny*(BALL_R+b.r);
      const tx=-ny,ty=nx,vt=ball.vx*tx+ball.vy*ty;
      ball.vx=nx*640+tx*vt*.35;ball.vy=ny*640+ty*vt*.35;
      b.flash=1;if(gameTime-b.lastS>.1){sBump();b.lastS=gameTime;}
      spawn(ball.x,ball.y,6,'#ff8fa3',200);
    }
  }
}
function collideFlipper(f){
  const tx=f.px+Math.cos(f.ang)*f.len,ty=f.py+Math.sin(f.ang)*f.len;
  const c=cpSeg(ball.x,ball.y,f.px,f.py,tx,ty);
  let dx=ball.x-c.x,dy=ball.y-c.y,d=Math.hypot(dx,dy);
  if(d<BALL_R+FLIP_R){
    let nx,ny;
    if(d<.001){nx=0;ny=-1;}else{nx=dx/d;ny=dy/d;}
    ball.x+=nx*(BALL_R+FLIP_R-d);ball.y+=ny*(BALL_R+FLIP_R-d);
    const rx=c.x-f.px,ry=c.y-f.py;
    const vsx=-f.omega*ry,vsy=f.omega*rx;
    const rvx=ball.vx-vsx,rvy=ball.vy-vsy,vn=rvx*nx+rvy*ny;
    if(vn<0){ball.vx-=1.45*vn*nx;ball.vy-=1.45*vn*ny;}
  }
}
function inZone(z,p){return p.x>z.x0&&p.x<z.x1&&p.y>z.y0&&p.y<z.y1;}

let gameTime=0;
function stepPhysics(dt){
  moveFlipper(flipL,keys.l,dt);moveFlipper(flipR,keys.r,dt);
  if(state!=='play')return;
  const ppx=ball.x,ppy=ball.y;
  ball.vy+=GRAV*dt;
  const dr=Math.exp(-.35*dt);ball.vx*=dr;ball.vy*=dr;
  let sp=Math.hypot(ball.vx,ball.vy);
  if(sp>1600){ball.vx*=1600/sp;ball.vy*=1600/sp;}
  ball.x+=ball.vx*dt;ball.y+=ball.vy*dt;
  collideWalls(dt);collideSlings();collideBumpers();collideFlipper(flipL);collideFlipper(flipR);
  // turbo lanes (frame-level entry)
  const now={x:ball.x,y:ball.y};
  if(inZone(zones[0],now)&&!inL){ball.vy=Math.max(ball.vy,820);ball.vx+=50;zones[0].flash=1;sBoost();spawn(ball.x,ball.y,10,'#ffd23f',240);}
  if(inZone(zones[1],now)&&!inR){ball.vy=Math.min(ball.vy,-820);ball.vx-=50;zones[1].flash=1;sBoost();spawn(ball.x,ball.y,10,'#ffd23f',240);}
  inL=inZone(zones[0],now);inR=inZone(zones[1],now);
  if(inL)ball.vx+=90*dt; if(inR)ball.vx-=90*dt;
  // gates
  if(wrongCd>0)wrongCd-=dt;
  for(let i=0;i<gates.length;i++){
    const g=gates[i];
    if(segHit(ppx,ppy,ball.x,ball.y,g)){
      if(i===nextGate){passGate(g);}
      else if(wrongCd<=0&&state==='play'){wrongCd=.9;wrongGate=i;sWrong();}
    }
  }
  // stall nudge
  if(Math.hypot(ball.vx,ball.vy)<45){stallT+=dt;}else stallT=0;
  if(stallT>2.2){ball.vy-=280;ball.vx+=(Math.random()-.5)*90;stallT=0;}
  // drain
  if(ball.y>H-8)drain();
}

/* ============================== frame update ============================== */
function update(dt){
  gameTime+=dt;
  if(state==='play'||state==='launch'){
    totalT+=dt;if(raceStarted)lapT+=dt;
    // charge
    if(charging&&state==='launch'){
      const prev=charge;charge=Math.min(1,charge+dt/1.1);
      const t8=Math.floor(charge*8);if(t8>Math.floor(prev*8))sTick(charge);
      if(charge>=1){holdFull+=dt;if(holdFull>.12)release();}else holdFull=0;
    }
    // ghost + record
    if(ghostTrail&&raceStarted)ghostT=(ghostT+dt)%Math.max(.5,ghostDur||bestLapSession);
    if(raceStarted){sampleAcc+=dt;while(sampleAcc>=1/120){lapRecord.push(ball.x,ball.y);sampleAcc-=1/120;}}
    // trail
    trail.push({x:ball.x,y:ball.y});if(trail.length>13)trail.shift();
  }
  if(state!=='pause'){
    for(let i=particles.length-1;i>=0;i--){const p=particles[i];
      p.vy+=p.g*dt;p.x+=p.vx*dt;p.y+=p.vy*dt;p.life-=dt;if(p.life<=0)particles.splice(i,1);}
    for(let i=msgs.length-1;i>=0;i--){msgs[i].t+=dt;if(msgs[i].t>msgs[i].dur)msgs.splice(i,1);}
    bumpers.forEach(b=>b.flash=Math.max(0,b.flash-dt*4));
    slings.forEach(s=>s.flash=Math.max(0,s.flash-dt*4));
    zones.forEach(z=>z.flash=Math.max(0,z.flash-dt*3));
    flipL.glow=Math.max(0,flipL.glow-dt*5);flipR.glow=Math.max(0,flipR.glow-dt*5);
  }
}

    const cx=(z.x0+z.x1)/2;
    ctx.strokeStyle='rgba(255,210,63,'+z.flash+')';ctx.lineWidth=4;ctx.lineCap='round';
    [490,515,540].forEach(y=>{
      const yy=z.dir>0?y:y; // arrows always drawn at their lane positions
      ctx.beginPath();
      if(z.dir>0){ctx.moveTo(cx-9,y-7);ctx.lineTo(cx,y+6);ctx.lineTo(cx+9,y-7);}
      else{ctx.moveTo(cx-9,y+7);ctx.lineTo(cx,y-6);ctx.lineTo(cx+9,y+7);}
      ctx.stroke();});
  });
}
function drawPlunger(){
  const headY=702+34*charge;
  // charge meter on casing bar
  if(state==='launch'){

  ctx.strokeStyle='#9aa6d8';ctx.lineWidth=3;ctx.lineJoin='round';ctx.beginPath();ctx.moveTo(450,top);

  ctx.stroke();

  ctx.fillStyle='#ff4d4d';ctx.strokeStyle='#ffd7d7';ctx.lineWidth=2;
  ctx.beginPath();ctx.roundRect(436,headY,28,10,4);ctx.fill();ctx.stroke();
}
function drawFlipper(f){
  const tx=f.px+Math.cos(f.ang)*f.len,ty=f.py+Math.sin(f.ang)*f.len;
  ctx.save();
  if(f.glow>0){ctx.shadowColor='#ffb14d';ctx.shadowBlur=20*f.glow;}
  ctx.lineCap='round';
  ctx.strokeStyle='#a83c0e';ctx.lineWidth=FLIP_R*2+1;
  ctx.beginPath();ctx.moveTo(f.px,f.py);ctx.lineTo(tx,ty);ctx.stroke();
  ctx.strokeStyle='#ff8c3a';ctx.lineWidth=FLIP_R*2-4;
  ctx.beginPath();ctx.moveTo(f.px,f.py);ctx.lineTo(tx,ty);ctx.stroke();
  ctx.fillStyle='#fff';ctx.beginPath();ctx.arc(tx,ty,3.6,0,7);ctx.fill();
  ctx.restore();
}
function drawGhost(){
  if(!ghostTrail||ghostTrail.length<4)return;
  const i=Math.floor(ghostT*120)%Math.floor(ghostTrail.length/2)*2;
  for(let k=3;k>=0;k--){
    const j=(i-k*2+ghostTrail.length)%ghostTrail.length;
    ctx.beginPath();ctx.arc(ghostTrail[j],ghostTrail[j+1],BALL_R*(1-k*.15),0,7);
    ctx.fillStyle='rgba(255,255,255,'+(k===0?.4:.12)+')';ctx.fill();
  }
  ctx.beginPath();ctx.arc(ghostTrail[i],ghostTrail[i+1],BALL_R,0,7);
  ctx.strokeStyle='rgba(255,255,255,.6)';ctx.lineWidth=2;ctx.stroke();
}
function drawBall(){
  for(let i=0;i<trail.length;i++){
    const p=trail[i],a=(i/trail.length);
    ctx.beginPath();ctx.arc(p.x,p.y,BALL_R*a*.85,0,7);
    ctx.fillStyle='rgba(190,205,255,'+(a*.16)+')';ctx.fill();
  }
  const g=ctx.createRadialGradient(ball.x-3,ball.y-4,1,ball.x,ball.y,BALL_R+1);
  g.addColorStop(0,'#ffffff');g.addColorStop(.35,'#dfe6ff');g.addColorStop(.72,'#8fa0d8');g.addColorStop(1,'#3c4a86');
  ctx.beginPath();ctx.arc(ball.x,ball.y,BALL_R,0,7);ctx.fillStyle=g;ctx.fill();
  ctx.strokeStyle='rgba(255,255,255,.5)';ctx.lineWidth=1;ctx.stroke();
}
function drawParticles(){
  for(const p of particles){
    ctx.globalAlpha=Math.max(0,p.life/p.max);
    ctx.fillStyle=p.col;
    ctx.fillRect(p.x-p.size/2,p.y-p.size/2,p.size,p.size);
  }
  ctx.globalAlpha=1;
}
function drawMessages(){
  msgs.forEach((m,i)=>{
    const pop=easeOutBack(Math.min(1,m.t/.18));
    const a=m.t>m.dur-.4?Math.max(0,(m.dur-m.t)/.4):1;

    ctx.font="700 "+m.size+"px Impact,'Arial Black',sans-serif";
    ctx.textAlign='center';ctx.textBaseline='middle';
    ctx.lineWidth=7;ctx.strokeStyle='rgba(9,11,28,.9)';ctx.strokeText(m.text,0,0);
    ctx.fillStyle=m.color;ctx.fillText(m.text,0,0);
    ctx.restore();
  });
  ctx.globalAlpha=1;
}
function drawLaunchHint(t){
  if(state!=='launch')return;
  const b=Math.sin(t*6)*4;
  ctx.strokeStyle='rgba(255,210,63,'+(charging?.9:.55)+')';ctx.lineWidth=4;ctx.lineCap='round';
  for(let k=0;k<2;k++){const y=636+k*20-b%10;
    ctx.beginPath();ctx.moveTo(440,y+8);ctx.lineTo(450,y-2);ctx.lineTo(460,y+8);ctx.stroke();}
  if(!charging){
    ctx.font="10px Consolas,monospace";ctx.textAlign='center';ctx.fillStyle='rgba(255,210,63,.8)';
    ctx.fillText('SPACE',450,610);
  }
}
function render(t){
  drawBackground(t);
  ctx.setTransform(dpr*scale,0,0,dpr*scale,dpr*ox,dpr*oy);
  ctx.drawImage(SC,0,0,W,H);
  drawGates(t);drawZones();drawSlings();drawBumpers();
  if(state==='launch'||state==='menu')drawPlunger();
  // drain pulse
  if(state==='play'&&ball.y>640){
    const k=(ball.y-640)/112;
    ctx.fillStyle='rgba(255,60,80,'+(k*.25*(0.7+0.3*Math.sin(t*14)))+')';
    ctx.fillRect(170,690,140,70);
  }
  drawGhost();
  if(state!=='menu')drawBall();else{ball.x=450;ball.y=693;drawBall();}
  drawFlipper(flipL);drawFlipper(flipR);
  drawParticles();drawLaunchHint(t);drawMessages();
}

let last=performance.now(),acc=0;
function frame(now){
  let dt=(now-last)/1000;last=now;
  if(dt>.05)dt=.05;
  update(dt);
  acc+=dt;
  while(acc>=STEP){stepPhysics(STEP);acc-=STEP;}
  render(now/1000);
  updateHUD();
  requestAnimationFrame(frame);
}

**How to play:** hold **SPACE** to charge the plunger, release to launch. Use **← / A** and **→ / D** to flip. Race through gates **1 → 2 → 3 → 4** in order — three laps before your last ball drains. The yellow turbo lanes boost you; bumpers and slingshots are chaos. Your fastest lap becomes a ghost to chase, and records persist between sessions. **R** restarts, **P** pauses, and touch zones (left / middle / right of the table) work on mobile.
