function triDist(px,py,ax,ay,bx,by,cx2,cy2){

  [[60,.10,1.1],[42,.22,1.7],[24,.45,2.5]].forEach(([n,sp,sz],li)=>{ for(let i=0;i<n;i++)

/* ---------- helpers ---------- */
const clamp=(v,a,b)=>Math.max(a,Math.min(b,v));
function rrect(x,y,w,h,r){ ctx.beginPath(); ctx.moveTo(x+r,y);
  ctx.arcTo(x+w,y,x+w,y+h,r);ctx.arcTo(x+w,y+h,x,y+h,r);ctx.arcTo(x,y+h,x,y,r);ctx.arcTo(x,y,x+w,y,r);ctx.closePath(); }
function segClosest(px,py,x1,y1,x2,y2){ const dx=x2-x1,dy=y2-y1,l2=dx*dx+dy*dy;
  let t=l2?((px-x1)*dx+(py-y1)*dy)/l2:0; t=clamp(t,0,1); return Math.hypot(px-(x1+t*dx),py-(y1+t*dy)); }
function triDist(px,py,ax,ay,bx,by,cx,cy){
  const s1=(bx-ax)*(py-ay)-(by-ay)*(px-ax), s2=(cx-bx)*(py-by)-(cy-by)*(px-bx), s3=(ax-cx)*(py-cy)-(ay-cy)*(px-cx);
  if(!((s1<0||s2<0)&&(s1>0||s2>0)) && !((s2<0||s3<0)&&(s2>0||s3>0)) && !((s3<0||s1<0)&&(s3>0||s1>0))) return 0;
  return Math.min(segClosest(px,py,ax,ay,bx,by),segClosest(px,py,bx,by,cx,cy),segClosest(px,py,cx,cy,ax,ay)); }
function rectDist(px,py,x,y,w,h){ const cx=clamp(px,x,x+w),cy=clamp(py,y,y+h); return Math.hypot(px-cx,py-cy); }

/* ---------- obstacles ---------- */
const difficulty=()=>clamp(score/2000,0,1);
function spawnPattern(){
  const d=difficulty(), x=W+70*S, r=Math.random();
  let type='spikes';
  if(score>700 && r<.18) type='hole';
  else if(score>350 && r<.42) type='gate';
  else if(score>120 && r<.68) type='pillar';

  if(type==='spikes'){ const n=1+Math.floor(Math.random()*(2+d*2)); addSpikes(x,n,Math.random()<.5); }
  else if(type==='pillar') addPillar(x,Math.random()<.5);
  else if(type==='gate') addGate(x,d);
  else addHole(x,Math.random()<.5);

  spawnIn=speed*(1.35-.5*d)*(.85+Math.random()*.4)+240*S;
}
function addSpikes(x,n,onFloor){ obstacles.push({type:'spikes',x,w:n*30*S,n,sw:30*S,sh:46*S,onFloor,minD:null,passed:false}); }
function addPillar(x,fromFloor){ const playH=floorY-ceilY,h=playH*.62,w=46*S;
  obstacles.push({type:'pillar',x,y:fromFloor?floorY-h:ceilY,w,h,fromFloor,minD:null,passed:false}); }
function addGate(x,d){ const gap=(170-74*d)*S*(.9+Math.random()*.2), playH=floorY-ceilY;
  const yc=ceilY+playH*(.35+Math.random()*.3);
  let topH=yc-gap/2-ceilY, botH=floorY-(yc+gap/2);
  if(topH<18*S){topH=18*S;} if(botH<18*S){botH=18*S;}
  obstacles.push({type:'pillar',x,y:ceilY,w:46*S,h:topH,fromFloor:false,minD:null,passed:false});
  obstacles.push({type:'pillar',x,y:floorY-botH,w:46*S,h:botH,fromFloor:true,minD:null,passed:false}); }
function addHole(x,onFloor){ const w=(90+Math.random()*60)*S*(1-difficulty()*.3);
  obstacles.push({type:'hole',x,w,onFloor,minD:null,passed:false,y:0,h:0}); }

function overHoleAt(x,onFloor){ for(const o of obstacles) if(o.type==='hole'&&o.onFloor===onFloor&&x>o.x+4*S&&x<o.x+o.w-4*S) return true; return false; }
function distToObstacle(o,px,py){
  if(o.type==='spikes'){ let m=1e9; const base=o.onFloor?floorY:ceilY, dir=o.onFloor?-1:1;
    for(let i=0;i<o.n;i++){ const bx=o.x+i*o.sw; m=Math.min(m,triDist(px,py,bx,base,bx+o.sw/2,base+dir*o.sh,bx+o.sw,base)); } return m; }
  if(o.type==='pillar') return rectDist(px,py,o.x,o.y,o.w,o.h);
  const band=o.onFloor?[o.x,floorY,o.w,H-floorY]:[o.x,0,o.w,ceilY];
  return rectDist(px,py,band[0],band[1],band[2],band[3]);
}
function hitObstacle(o,px,py,r){
  if(o.type==='spikes'){ const base=o.onFloor?floorY:ceilY, dir=o.onFloor?-1:1;
    for(let i=0;i<o.n;i++){ const bx=o.x+i*o.sw; if(triDist(px,py,bx,base,bx+o.sw/2,base+dir*o.sh,bx+o.sw,base)<r*.8) return true; } return false; }
  if(o.type==='pillar') return rectDist(px,py,o.x,o.y,o.w,o.h)<r*.88;
  if(o.onFloor){ if(gravDir>0&&px>o.x+4*S&&px<o.x+o.w-4*S&&py+r>.5*r+floorY-r*.5) { } return gravDir>0&&px>o.x+6*S&&px<o.x+o.w-6*S&&(py+r*0.5)>floorY; }
  return gravDir<0&&px>o.x+6*S&&px<o.x+o.w-6*S&&(py-r*.5)<ceilY;
}

/* ---------- particles / fx ---------- */
function ringAt(x,y,r0,r1,col,life){ particles.push({type:'ring',x,y,r0,r1,t:0,life:life||.5,col}); }
function landDust(surfY,dir,impact){ const n=3+Math.min(9,impact/(160*S));
  for(let i=0;i<n;i++) particles.push({type:'dust',x:player.x+(Math.random()-.5)*player.r*2,y:surfY,
    vx:(Math.random()-.5)*280*S-speed*.12,vy:-dir*(30+Math.random()*90)*S,t:0,life:.35+Math.random()*.35,size:(2+Math.random()*3.2)*S}); }
function flipBurst(){ const col=gravDir>0?AMB:CYN;
  for(let i=0;i<7;i++){ const a=Math.random()*TAU,v=(140+Math.random()*380)*S;
    particles.push({type:'spark',x:player.x,y:player.y,vx:Math.cos(a)*v,vy:Math.sin(a)*v,t:0,life:.25+Math.random()*.25,col}); }
  ringAt(player.x,player.y,player.r*1.2,player.r*3,col,.4); }
function explode(x,y){ const cols=['#eef4fb','#16233c',AMB,CYN,DNG];
  for(let i=0;i<16;i++){ const a=Math.random()*TAU,v=(120+Math.random()*520)*S;
    particles.push({type:'debris',x,y,vx:Math.cos(a)*v-speed*.25,vy:Math.sin(a)*v,t:0,life:.8+Math.random()*.7,size:(3+Math.random()*6.5)*S,rot:Math.random()*TAU,vr:Math.random()*14-7,col:cols[i%cols.length]}); }
  for(let i=0;i<26;i++){ const a=Math.random()*TAU,v=(200+Math.random()*680)*S;
    particles.push({type:'spark',x,y,vx:Math.cos(a)*v,vy:Math.sin(a)*v,t:0,life:.25+Math.random()*.35,col:i%3?'#ffd9a0':'#ffffff'}); }
  ringAt(x,y,6*S,130*S,'#ffffff',.6); ringAt(x,y,4*S,90*S,DNG,.5); }

function updateFx(dt){
  for(let i=particles.length-1;i>=0;i--){ const p=particles[i]; p.t+=dt;
    if(p.t>=p.life){ particles.splice(i,1); continue; }
    if(p.type==='dust'){ p.x+=p.vx*dt; p.y+=p.vy*dt; const dr=Math.exp(-3*dt); p.vx*=dr; p.vy*=dr; }
    else if(p.type==='spark'){ p.x+=p.vx*dt; p.y+=p.vy*dt; }
    else if(p.type==='debris'){ p.x+=p.vx*dt; p.y+=p.vy*dt; p.vy+=1500*S*dt; p.rot+=p.vr*dt; } }
  if(particles.length>420) particles.splice(0,particles.length-420);
  for(let i=popups.length-1;i>=0;i--){ const q=popups[i]; q.t+=dt; q.y-=34*S*dt; if(q.t>q.life) popups.splice(i,1); }
  for(let i=ghosts.length-1;i>=0;i--){ const g=ghosts[i]; g.x-=(80*S+speed*.15)*dt; if(g.x<-260*S) ghosts.splice(i,1); }
}

/* ---------- core actions ---------- */
function doFlip(){ gravDir*=-1; player.vy+=gravDir*430*S; player.grounded=false;
  Snd.flip(); flipBurst(); shake=Math.max(shake,2.5*S); edgeFlash[gravDir>0?1:0]=1; }

function stepPlayer(dt){
  const p=player,g=2800*S,vmax=1350*S;
  if(p.grounded&&((gravDir>0&&p.y+p.r<floorY-2)||(gravDir<0&&p.y-p.r>ceilY+2))) p.grounded=false;
  p.vy=clamp(p.vy+g*gravDir*dt,-vmax,vmax);
  p.y+=p.vy*dt;
  if(gravDir>0){ const oh=overHoleAt(p.x,true);
    if(!oh&&p.y+p.r>=floorY) land(floorY-p.r,1);
    else if(oh&&state==='play'&&p.y-p.r>floorY) die(); }
  else { const oh=overHoleAt(p.x,false);
    if(!oh&&p.y-p.r<=ceilY) land(ceilY+p.r,-1);
    else if(oh&&state==='play'&&p.y+p.r<ceilY) die(); }
  p.squash*=Math.exp(-7*dt);
  const target=p.grounded?0:clamp(p.vy/(2800*S),-.55,.55);
  p.angle+=(target-p.angle)*Math.min(1,dt*14);
}
function land(y,dir){ const impact=Math.abs(player.vy);
  if(!player.grounded&&impact>90*S){ landDust(dir>0?floorY:ceilY,dir,impact); Snd.land(impact/(1300*S));
    if(impact>760*S) shake=Math.max(shake,Math.min(5*S,impact/(260*S))); }
  player.y=y; player.vy=0; player.grounded=true;
  player.squash=Math.min(.4,impact/(1500*S)); }

function die(){ if(state!=='play')return; state='dying'; deathT=0;
  Snd.boom(); shake=15*S; flash=.8; explode(player.x,player.y);
  const sc=Math.floor(score); if(sc>best){ best=sc; saveLS('graviton.best',best); newBest=true; } }

function stepWorld(dt){ scroll+=speed*dt;
  for(let i=obstacles.length-1;i>=0;i--){ obstacles[i].x-=speed*dt;
    const o=obstacles[i]; if(o.x+(o.w||o.sw*o.n)<-90*S) obstacles.splice(i,1); } }

/* ---------- state machine ---------- */
function reset(){ score=0; nearMisses=0; milestoneNext=100; speedMaxSeen=baseSpeed;
  obstacles.length=0; particles.length=0; popups.length=0; ghosts.length=0;
  gravDir=1; player.y=floorY-player.r; player.vy=0; player.grounded=true; player.angle=0; player.squash=0;
  spawnIn=baseSpeed*1.9+320*S; flash=.25; shake=0; edgeFlash[0]=edgeFlash[1]=0; menuFlipT=1.6; }

function startGame(){ Snd.ensure(); Snd.duck(.045); reset(); state='play';
  startScreen.classList.add('hidden'); overScreen.classList.add('hidden'); pauseScreen.classList.add('hidden'); }
function resumeGame(){ if(state!=='pause')return; state='play'; pauseScreen.classList.add('hidden'); }
function togglePause(){ if(state==='play'){ state='pause'; pauseScreen.classList.remove('hidden'); }
  else if(state==='pause') resumeGame(); }
function showOver(){ $('finalScore').textContent=Math.floor(score);
  $('statBest').textContent=best+'m'; $('statNear').textContent=nearMisses;
  $('statSpeed').textContent=(speedMaxSeen/baseSpeed).toFixed(2)+'×';
  $('newBest').classList.toggle('hidden',!newBest);
  $('startBest').textContent=best; overScreen.classList.remove('hidden'); }

function primaryAction(){
  if(state==='menu') startGame();
  else if(state==='play') doFlip();
  else if(state==='pause') resumeGame();
  else if(state==='over'&&tGlobal-overAt>.45) startGame();
}

/* ---------- updates ---------- */
function updatePlay(dt){
  speed=baseSpeed+(maxSpeed-baseSpeed)*Math.pow(Math.min(1,score/1800),.65)+Math.sin(tGlobal*1.7)*8*S;
  speedMaxSeen=Math.max(speedMaxSeen,speed);
  score+=speed*dt/(38*S);
  while(score>=milestoneNext){ ghosts.push({txt:milestoneNext,x:W+140*S,y:H*.42}); Snd.tick(milestoneNext/100); milestoneNext+=100; }
  spawnIn-=speed*dt; if(spawnIn<=0) spawnPattern();
  stepWorld(dt); stepPlayer(dt);
  if(state!=='play') return;
  const px=player.x,py=player.y,r=player.r;
  for(const o of obstacles){
    if(!o.passed&&o.x+(o.w||o.sw*o.n)<px-r*.5){ o.passed=true;
      if(o.type!=='hole'&&o.minD!==null&&o.minD>0&&o.minD<r*2.3){
        nearMisses++; score+=2; Snd.near();
        popups.push({x:px+r+14*S,y:py-36*S,txt:'CLOSE +2',col:CYN,t:0,life:.9});
        ringAt(px,py,r*1.5,r*3.2,CYN,.45); } }
    const ow=o.w||o.sw*o.n;
    if(o.x<px+r+ow&&o.x+ow>px-r){
      const d=distToObstacle(o,px,py);
      if(o.type!=='hole'&&(o.minD===null||d<o.minD)) o.minD=d;
      if(hitObstacle(o,px,py,r)){ die(); return; } } }
  updateFx(dt);
  shake*=Math.exp(-5.5*dt); flash=Math.max(0,flash-dt*1.6);
  edgeFlash[0]*=Math.exp(-4*dt); edgeFlash[1]*=Math.exp(-4*dt);
}
function updateMenu(dt){ speed=baseSpeed*.42; stepWorld(dt); stepPlayer(dt);
  menuFlipT-=dt; if(menuFlipT<=0){ doFlip(); menuFlipT=1.9+Math.random()*.8; }
  updateFx(dt); shake*=Math.exp(-5.5*dt); flash=Math.max(0,flash-dt*1.6);
  edgeFlash[0]*=Math.exp(-4*dt); edgeFlash[1]*=Math.exp(-4*dt); }
function updateDying(dt){ deathT+=dt; const sdt=dt*.25;
  speed*=Math.exp(-4*dt); stepWorld(sdt); updateFx(sdt);
  shake*=Math.exp(-4.5*dt); flash=Math.max(0,flash-dt*1.4);
  if(deathT>1.15){ state='over'; overAt=tGlobal; showOver(); } }

  ctx.beginPath(); ctx.arc(ox,oy,R*.68,a0,a0+1.1); ctx.stroke(); }

function drawParallax(){
  for(const s of stars){ const x=((s.u*W-scroll*s.sp)%W+W)%W, y=s.v*H;
    const a=.25+.45*(.5+.5*Math.sin(tGlobal*s.tw+s.ph));
    ctx.fillStyle='rgba(190,220,255,'+a.toFixed(3)+')';
    const rr=(speed-baseSpeed)/(maxSpeed-baseSpeed);
    if(rr>.7&&s.sp>=.4){ const l=(rr-.7)*90*S; ctx.fillRect(x,y,l,s.s*.8); }
    else { ctx.beginPath(); ctx.arc(x,y,s.s,0,TAU); ctx.fill(); } }
  ctx.strokeStyle='rgba(120,165,205,.16)'; ctx.lineWidth=1.2;
  for(const d of debrisF){ const x=((d.u*W-scroll*.45)%W+W)%W, y=d.v*H, r=d.rot+tGlobal*d.vr;
    ctx.save(); ctx.translate(x,y); ctx.rotate(r); const s=d.sz;
    if(d.sh===0) ctx.strokeRect(-s/2,-s/2,s,s);
    else if(d.sh===1){ ctx.beginPath(); ctx.moveTo(0,-s*.6); ctx.lineTo(s*.55,s*.45); ctx.lineTo(-s*.55,s*.45); ctx.closePath(); ctx.stroke(); }
    else { ctx.beginPath(); ctx.arc(0,0,s*.5,0,TAU); ctx.stroke(); }
    ctx.restore(); } }

function drawHoles(){ for(const o of obstacles){ if(o.type!=='hole')continue;

function drawPillar(o){

      ctx.restore(); } }
