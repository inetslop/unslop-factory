function collide(){const a=P[0],b=P[1]; if(a.fallT>=0||b.fallT>=0)return;
  const dx=b.x-a.x,dy=b.y-a.y,d=Math.hypot(dx,dy),rr=a.r+b.r;
  if(d>=rr||d===0)return;
  const nx=dx/d,ny=dy/d,ma=a.r*a.r,mb=b.r*b.r,tm=ma+mb,ov=rr-d;
  a.x-=nx*ov*(mb/tm); a.y-=ny*ov*(mb/tm); b.x+=nx*ov*(ma/tm); b.y+=ny*ov*(ma/tm);
  const rel=(b.vx-a.vx)*nx+(b.vy-a.vy)*ny;
  if(rel<0){const e=.42, boost=(a.dashT>0||b.dashT>0)?1.5:1;
    const j=-(1+e)*rel*boost/(1/ma+1/mb);
    a.vx-=j*nx/ma; a.vy-=j*ny/ma; b.vx+=j*nx/mb; b.vy+=j*ny/mb;
    const cx=a.x+nx*a.r,cy=a.y+ny*a.r,strength=Math.min(1,-rel/700);
    if(-rel>90){SFX.thud(strength); burst(cx,cy,'#ffffff',6+strength*8,'spark'); ring(cx,cy,20,'#ffffff');
      a.hitT=b.hitT=.25; G.shake=Math.max(G.shake,4+strength*10);
      if(-rel>520){G.hitstop=.06; addText(cx,cy-34,'POW!','#ffd23f',34);}}}}
