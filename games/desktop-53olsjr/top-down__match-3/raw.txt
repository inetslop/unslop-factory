function tracePath(ctx,type,r){
  ctx.beginPath();
  switch(type){
   case 0: // diamond (tall)
     ctx.moveTo(0,-r); ctx.lineTo(r*0.72,0); ctx.lineTo(0,r); ctx.lineTo(-r*0.72,0); ctx.closePath(); break;
   case 1: // orb
     ctx.arc(0,0,r*0.85,0,Math.PI*2); break;
   case 2: // triangle up
     const s=r*1.06; ctx.moveTo(0,-s); ctx.lineTo(s*0.92,s*0.72); ctx.lineTo(-s*0.92,s*0.72); ctx.closePath(); break;
   case 3: // hexagon
     for(let i=0;i<6;i++){ const a=Math.PI/6 + i*Math.PI/3; const x=Math.cos(a)*r*0.95, y=Math.sin(a)*r*0.95; i?ctx.lineTo(x,y):ctx.moveTo(x,y);} ctx.closePath(); break;
   case 4: // four-point star
     const o=r*1.05, q=r*0.38;
     ctx.moveTo(0,-o); ctx.lineTo(q, -q); ctx.lineTo(o,0); ctx.lineTo(q,q); ctx.lineTo(0,o); ctx.lineTo(-q,q); ctx.lineTo(-o,0); ctx.lineTo(-q,-q); ctx.closePath(); break;
  }
}

  let pts=(plan.set.size*10+plan.actCount*80)*mult;
  score+=pts;totalCleared+=plan.set.size;
  let cx=0,cy=0;plan.set.forEach(k=>{cx+=(k%COLS);cy+=((k/COLS)|0);});cx/=plan.set.size;cy/=plan.set.size;
  floaters.push({x:cx,y:cy,txt:'+'+fmt(pts),size:cell*0.52,color:'#ffffff',t:0,dur:0.9,vy:-1.1});
  if(mult>=2)floaters.push({x:cx,y:cy-0.72,txt:'COMBO ×'+mult,size:cell*0.46,color:'#ffd76a',t:0,dur:1.1,vy:-0.8});
  else if(plan.set.size>=6)floaters.push({x:cx,y:cy-0.65,txt:'NICE!',size:cell*0.32,color:'#8fe6ff',t:0,dur:0.9,vy:-0.9});
  let crossS=false,novaS=false;
  plan.set.forEach(k=>{const r=(k/COLS)|0,c=k%COLS,g=grid[r][c];if(!g)return;
    g.dying=true;burst(c,r,GEMS[g.type],g.special?16:7);
    if(g.special===1)crossS=true;if(g.special===2)novaS=true;});
  if(crossS){shakeMag=Math.max(shakeMag,9);sCross();}
  if(novaS){shakeMag=Math.max(shakeMag,15);sNova();burst(cx,cy,GOLD,26);}
  if(!crossS&&!novaS)sPop(mult);
  clearInfo={set:plan.set,t:0,dur:0.3,spawns:plan.spawns};
  state='clear';updateHUD();pulse(scoreEl);
}
function finishClear(){
  clearInfo.set.forEach(k=>{grid[(k/COLS)|0][k%COLS]=null;});
  clearInfo.spawns.forEach(s=>{const g=makeGem(s.type,s.r,s.c);g.special=s.sp;g.sc=1.9;grid[s.r][s.c]=g;burst(s.c,s.r,s.sp===2?GOLD:GEMS[s.type],14);});
  applyGravity();clearInfo=null;state='fall';
}
function applyGravity(){
  for(let c=0;c<COLS;c++){
    let write=ROWS-1;
    for(let r=ROWS-1;r>=0;r--){const g=grid[r][c];if(g){if(write!==r){grid[write][c]=g;grid[r][c]=null;}write--;}}
    const m=write+1;
    for(let r=write;r>=0;r--)grid[r][c]=makeGem((Math.random()*TYPES)|0,r,c,-(m-r)-0.5);
  }
}
function afterSettle(){
  const runs=findRuns();
  if(runs.length){cascade++;startClear(runs,null);return;}
  cascade=0;
  if(score>=target){levelUp();return;}
  if(moves<=0){gameOver();return;}
  if(!hasMove()){doShuffle();return;}
  state='idle';
}
function levelUp(){
  const bonus=moves*40;
  if(bonus>0){score+=bonus;floaters.push({x:COLS/2,y:ROWS/2-1,txt:'MOVE BONUS +'+fmt(bonus),size:cell*0.38,color:'#ffd76a',t:0,dur:1.2,vy:-0.8});}
  level++;target=calcTarget(level);moves=16;
  banner('LEVEL '+level,'TARGET '+fmt(target));sLevel();confetti();
  state='levelup';levelT=0;updateHUD();pulse(scoreEl);
}
function doShuffle(){banner('NO MOVES','RESHUFFLING…');sInvalid();state='shuffle';shuffleT=0;shuffled=false;}
function typesBad(t){
  for(let r=0;r<ROWS;r++)for(let c=0;c<COLS-2;c++){const i=r*COLS+c;if(t[i]===t[i+1]&&t[i]===t[i+2])return true;}
  for(let c=0;c<COLS;c++)for(let r=0;r<ROWS-2;r++){const i=r*COLS+c;if(t[i]===t[i+COLS]&&t[i]===t[i+2*COLS])return true;}
  return false;
}
function reassign(){
  for(let a=0;a<60;a++){
    const t=[];for(let i=0;i<ROWS*COLS;i++)t.push((Math.random()*TYPES)|0);
    if(typesBad(t))continue;
    let i=0;for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++)grid[r][c].type=t[i++];
    if(hasMove())return;
  }
  for(let t2=0;t2<300&&(!hasMove());t2++){const g=grid[(Math.random()*ROWS)|0][(Math.random()*COLS)|0];g.type=(g.type+1)%TYPES;}
}
function gameOver(){
  state='over';sOver();
  const nb=score>best;
  if(nb){best=score;lsSet('shardfall_best',String(best));}
  $('finalScore').textContent=fmt(score);$('finalBest').textContent=fmt(best);
  $('finalLevel').textContent=level;$('finalCleared').textContent=fmt(totalCleared);
  $('newBest').classList.toggle('hidden',!nb);
  $('bestStart').textContent=fmt(best);
  startPanel.classList.add('hidden');overPanel.classList.remove('hidden');
  overlay.classList.remove('hidden');
}

function burst(x,y,col,n){
  for(let i=0;i<n;i++){const a=Math.random()*Math.PI*2,sp=1.2+Math.random()*4.2;
    particles.push({x:x,y:y,vx:Math.cos(a)*sp,vy:Math.sin(a)*sp-0.8,life:0,ttl:0.45+Math.random()*0.4,size:cell*(0.05+Math.random()*0.08),color:Math.random()<0.25?'#ffffff':col.main,rot:Math.random()*6,vr:(Math.random()-0.5)*10});}
  if(particles.length>430)particles.splice(0,particles.length-430);
}
function dust(c,r){
  for(let i=0;i<4;i++)particles.push({x:c+(Math.random()-0.5)*0.6,y:r+0.35,vx:(Math.random()-0.5)*2,vy:-0.6-Math.random(),life:0,ttl:0.35,size:cell*0.07,color:'#9fb6d8',rot:0,vr:4});
}
function confetti(){
  for(let i=0;i<56;i++){const col=GEMS[(Math.random()*TYPES)|0];
    particles.push({x:Math.random()*COLS,y:-0.5-Math.random()*1.5,vx:(Math.random()-0.5)*1.5,vy:2+Math.random()*3,life:0,ttl:1.1+Math.random()*0.7,size:cell*(0.09+Math.random()*0.08),color:col.main,rot:Math.random()*6,vr:(Math.random()-0.5)*10});}
}

/* ================= UPDATE ================= */
function lerp(a,b,t){return a+(b-a)*t}
function easeBack(t){const c1=1.70158,c3=c1+1;return 1+c3*Math.pow(t-1,3)+c1*Math.pow(t-1,2)}
function update(dt){
  time+=dt;
  for(const m of motes){m.fy+=m.spd*dt*2.2;if(m.fy>1.03){m.fy=-0.03;m.fx=Math.random();}}
  for(let i=particles.length-1;i>=0;i--){const p=particles[i];p.life+=dt;
    if(p.life>=p.ttl){particles.splice(i,1);continue;}
    p.x+=p.vx*dt;p.y+=p.vy*dt;p.vy+=5*dt;p.rot+=p.vr*dt;}
  for(let i=floaters.length-1;i>=0;i--){const f=floaters[i];f.t+=dt;
    if(f.t>=f.dur)floaters.splice(i,1);else f.y+=f.vy*dt*0.6;}
  shakeMag*=Math.exp(-dt*5.5);if(shakeMag<0.15)shakeMag=0;
  sx=(Math.random()*2-1)*shakeMag;sy=(Math.random()*2-1)*shakeMag;
  /* gem scale springs */
  for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++){const g=grid[r][c];if(!g)continue;
    let ts=1;if(time-g.landT<0.16)ts=1-0.22*Math.sin((time-g.landT)/0.16*Math.PI);
    g.sc+=(ts-g.sc)*Math.min(1,dt*13);}
  switch(state){
    case 'swap':case 'revert':{
      const si=swapInfo;si.t+=dt;const k=easeBack(Math.min(1,si.t/si.dur));
      si.ga.px=lerp(si.fa.x,si.endA.c,k);si.ga.py=lerp(si.fa.y,si.endA.r,k);
      si.gb.px=lerp(si.fb.x,si.endB.c,k);si.gb.py=lerp(si.fb.y,si.endB.r,k);
      if(si.t>=si.dur){si.ga.px=si.endA.c;si.ga.py=si.endA.r;si.gb.px=si.endB.c;si.gb.py=si.endB.r;finishSwap();}
      break;}
    case 'clear':{clearInfo.t+=dt;if(clearInfo.t>=clearInfo.dur)finishClear();break;}
    case 'fall':{
      const G=42;let moving=false;
      for(let c=0;c<COLS;c++)for(let r=0;r<ROWS;r++){const g=grid[r][c];if(!g||g.py>=r-1e-4)continue;
        g.vy=Math.min(30,g.vy+G*dt);g.py+=g.vy*dt;
        if(g.py>=r){g.py=r;if(g.vy>7&&time-lastTick>0.05){lastTick=time;sLand();dust(c,r);}g.landT=time;g.vy=0;}
        moving=true;}
      if(!moving)afterSettle();
      break;}
    case 'levelup':{levelT+=dt;if(levelT>1.35)state='idle';break;}
    case 'shuffle':{
      shuffleT+=dt;
      if(shuffleT>0.45&&!shuffled){shuffled=true;reassign();
        for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++){const g=grid[r][c];if(g)g.sc=0.25;}sPop(1);}
      if(shuffleT>1.25){if(!hasMove())reassign();state='idle';}
      break;}
  }
}

function tracePath(type,r){
  ctx.beginPath();
  switch(type){
    case 0:ctx.moveTo(0,-r);ctx.lineTo(r*0.72,0);ctx.lineTo(0,r);ctx.lineTo(-r*0.72,0);ctx.closePath();break;
    case 1:ctx.arc(0,0,r*0.85,0,Math.PI*2);break;
    case 2:{const s=r*1.06;ctx.moveTo(0,-s);ctx.lineTo(s*0.95,s*0.62);ctx.lineTo(-s*0.95,s*0.62);ctx.closePath();break;}
    case 3:for(let i=0;i<6;i++){const a=Math.PI/6+i*Math.PI/3,x=Math.cos(a)*r*0.95,y=Math.sin(a)*r*0.95;if(i)ctx.lineTo(x,y);else ctx.moveTo(x,y);}ctx.closePath();break;
    case 4:{const o=r*1.05,q=r*0.38;ctx.moveTo(0,-o);ctx.lineTo(q,-q);ctx.lineTo(o,0);ctx.lineTo(q,q);ctx.lineTo(0,o);ctx.lineTo(-q,q);ctx.lineTo(-o,0);ctx.lineTo(-q,-q);ctx.closePath();break;}
  }
}
function drawGem(g,r,c){
  const x=ox+g.px*cell,y=oy+g.py*cell;
  if(y<oy-cell*2.4)return;
  let sc=g.sc,alpha=g.alpha;
  if(g.dying&&clearInfo){const k=Math.min(1,clearInfo.t/clearInfo.dur);sc*=1+0.5*Math.sin(k*Math.PI);alpha=1-k;}
  if(state==='menu')sc*=1+0.02*Math.sin(time*2+r*7+c*13);
  if(state==='idle'&&selected&&selected.r===r&&selected.c===c)sc*=1.08+0.03*Math.sin(time*7);
  const R=cell*0.36*sc,col=g.special===2?GOLD:GEMS[g.type],ph=r*7+c*13;
  ctx.globalAlpha=alpha*0.85;
  ctx.drawImage(haloCache[g.special===2?5:g.type],x-R*2.1,y-R*2.1,R*4.2,R*4.2);
  ctx.globalAlpha=alpha;
  ctx.save();ctx.translate(x,y);
  if(g.type===0||g.type===4)ctx.rotate(Math.sin(time*1.3+ph)*0.08);
  tracePath(g.type,R);
  const lg=ctx.createLinearGradient(-R,-R,R,R);lg.addColorStop(0,col.hi);lg.addColorStop(0.5,col.main);lg.addColorStop(1,col.lo);
  ctx.fillStyle=lg;ctx.fill();
  ctx.lineWidth=Math.max(1.2,R*0.13);ctx.strokeStyle=col.lo;ctx.stroke();
  ctx.save();ctx.scale(0.52,0.52);tracePath(g.type,R);
  ctx.strokeStyle='rgba(255,255,255,0.4)';ctx.lineWidth=Math.max(1,R*0.1);ctx.stroke();ctx.restore();
  ctx.fillStyle='rgba(255,255,255,'+(0.3+0.28*Math.sin(time*2+ph)).toFixed(3)+')';

  }
