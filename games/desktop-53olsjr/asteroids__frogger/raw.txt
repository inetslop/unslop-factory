function updateRocks(dt){
  for(let li=0;li<LANES;li++){
    const L=lanes[li],dir=li%2?-1:1,spd=getSpd(li);
    for(const rk of L){
      rk.x+=dir*spd*dt;rk.rot+=rk.vr*dt;
      if(rk.wob>0)rk.wob=Math.max(0,rk.wob-dt*2.4);
      if((dir>0&&rk.x>SW+95)||(dir<0&&rk.x<-95)){
        const g=gapRange();
        rk.x=dir>0?-60-Math.random()*g:SW+60+Math.random()*g;

          let bad=false;for(const o of L)if(o!==rk&&Math.abs(o.x-rk.x)<o.r+rk.r+24){bad=true;break;}

      }

function updateFX(dt){
  for(let i=particles.length-1;i>=0;i--){const p=particles[i];p.t+=dt;
    if(p.t>p.life){particles.splice(i,1);continue;}
    if(p.kind!=='ring'){p.vy+=(p.g||0)*dt;p.x+=p.vx*dt;p.y+=p.vy*dt;if(p.rot!==undefined)p.rot+=p.vr*dt;}
  }
  for(let i=beams.length-1;i>=0;i--){beams[i].t-=dt;if(beams[i].t<=0)beams.splice(i,1);}
  for(let i=floats.length-1;i>=0;i--){floats[i].t+=dt;if(floats[i].t>.85)floats.splice(i,1);}
  if(banner){banner.age+=dt;if(banner.age>banner.dur)banner=null;}
}
function frogUpdate(dt){
  zapCd=Math.max(0,zapCd-dt);
  if(frog.alive){
    frog.invuln=Math.max(0,frog.invuln-dt);
    frog.sq=Math.max(0,frog.sq-dt*6);
    if(frog.p<1){
      frog.p=Math.min(1,frog.p+dt/.13);const e=frog.p;
      frog.rx=frog.fx+(frog.tx-frog.fx)*e;
      frog.ry=frog.fy+(frog.ty-frog.fy)*e-Math.sin(e*Math.PI)*16;
      if(frog.p>=1){
        frog.rx=frog.tx;frog.ry=frog.ty;frog.sq=1;
        burst(frog.rx,frog.ry+8,'rgba(190,230,255,.7)',3,50);
        if(frog.row===0)padCheck();
        else if(frog.row<frog.crossBest){score+=15;float(frog.rx,frog.ry-22,'+15','#7fd3c7');frog.crossBest=frog.row;}
        if(pending&&frog.alive){const d=pending;pending=null;hop(d[0],d[1]);}
      }
    }
    if(frog.alive&&frog.invuln<=0&&frog.row>=1&&frog.row<=LANES){
      for(const rk of lanes[frog.row-1]){
        if(Math.abs(rk.x-frog.rx)>rk.r+20)continue;
        const dx=rk.x-frog.rx,dy=laneCY(frog.row)-frog.ry,r2=(rk.r*.8+13)*(rk.r*.8+13);
        if(dx*dx+dy*dy<r2){die('crush');break;}
      }
    }
  }else{
    deathT-=dt;
    if(frog.fall){frog.rx+=frog.vx*dt;frog.ry+=frog.vy*dt;frog.vy+=700*dt;frog.rot+=7*dt;}
    if(deathT<=0){
      lives--;
      if(lives<=0)gameOver();else resetFrog(1.2);
    }
  }
}

  }

function drawWorld(){
  // lane tints + dividers
  for(let i=0;i<LANES;i++){
    if(i%2){ctx.fillStyle='rgba(255,255,255,.014)';ctx.fillRect(0,LANE_Y0+i*LANE_H,SW,LANE_H);}
  }
  ctx.setLineDash([6,10]);ctx.strokeStyle='rgba(130,180,255,.09)';ctx.lineWidth=1;
  for(let i=1;i<LANES;i++){ctx.beginPath();ctx.moveTo(0,LANE_Y0+i*LANE_H);ctx.lineTo(SW,LANE_Y0+i*LANE_H);ctx.stroke();}
  ctx.setLineDash([]);
  drawPads();
  drawPlatform();
  for(const L of lanes)for(const rk of L)drawRock(rk);
  // beams
  for(const b of beams){
    const a=b.t/.12;
    ctx.strokeStyle='rgba(186,255,176,'+(a*.3)+')';ctx.lineWidth=9;
    ctx.beginPath();ctx.moveTo(b.x1,b.y1);ctx.lineTo(b.x2,b.y2);ctx.stroke();
    ctx.strokeStyle='rgba(230,255,220,'+a+')';ctx.lineWidth=3.5;
    ctx.beginPath();ctx.moveTo(b.x1,b.y1);ctx.lineTo(b.x2,b.y2);ctx.stroke();
    ctx.fillStyle='rgba(255,255,255,'+a+')';ctx.beginPath();ctx.arc(b.x2,b.y2,7*a+3,0,6.29);ctx.fill();
  }
  // frog
  if(state==='menu'){
    drawFrog(colX(5),rowTargetY(7)+Math.sin(t*2.6)*3,0,'up',1,0);
  }else if(frog.alive){
    let al=1;if(frog.invuln>0)al=Math.sin(t*28)>0?1:.3;
    drawFrog(frog.rx,frog.ry,frog.sq,frog.face,al,0);
  }else if(frog.fall){
    drawFrog(frog.rx,frog.ry,0,'up',Math.max(0,deathT/.95),frog.rot);
  }
  // particles
  for(const p of particles){
    const k=1-p.t/p.life;
    if(p.kind==='spark'){ctx.globalAlpha=k;ctx.fillStyle=p.color;ctx.beginPath();ctx.arc(p.x,p.y,p.size*k+.4,0,6.29);ctx.fill();}

    else if(p.kind==='conf'){ctx.save();ctx.globalAlpha=k;ctx.translate(p.x,p.y);ctx.rotate(p.rot);ctx.fillStyle=p.color;ctx.fillRect(-p.size/2,-p.size/2,p.size,p.size*.6);ctx.restore();}
    else if(p.kind==='ring'){const r=(1-k)*p.maxR;ctx.globalAlpha=k;ctx.strokeStyle=p.color;ctx.lineWidth=3*k+1;ctx.beginPath();ctx.arc(p.x,p.y,r,0,6.29);ctx.stroke();}
  }
  ctx.globalAlpha=1;
  // void edges
  for(const side of[0,1]){
    const x0=side?SW-40:0,g=ctx.createLinearGradient(side?SW-40:0,0,side?SW:40,0);
    g.addColorStop(0,'rgba(2,4,10,.65)');g.addColorStop(1,'rgba(2,4,10,0)');

  }
  ctx.setLineDash([4,8]);ctx.strokeStyle='rgba(255,190,80,.28)';ctx.lineWidth=1.5;
  ctx.beginPath();ctx.moveTo(2,HUD_H);ctx.lineTo(2,SH);ctx.stroke();
  ctx.beginPath();ctx.moveTo(SW-2,HUD_H);ctx.lineTo(SW-2,SH);ctx.stroke();
  ctx.setLineDash([]);
  // floats
  for(const f of floats){
    const k=f.t/.85;

  }
}

function frame(now){
  const dt=Math.min(.033,(now-lastT)/1000);lastT=now;t+=dt;
  if(!paused){
    if(state!=='over')updateRocks(dt);
    updateFX(dt);
    if(state==='play')frogUpdate(dt);
  }
  if(shake>0){shake*=Math.max(0,1-6*dt);if(shake<.15)shake=0;}
  shakeX=(Math.random()*2-1)*shake;shakeY=(Math.random()*2-1)*shake;

  drawBG();
  ctx.setTransform(DPR,0,0,DPR,0,0);
  ctx.translate(ox+shakeX,oy+shakeY);ctx.scale(scale,scale);
  drawWorld();drawHUD();drawBanner();
  ctx.strokeStyle='rgba(120,180,255,.1)';ctx.lineWidth=2;ctx.strokeRect(1,1,SW-2,SH-2);
  requestAnimationFrame(frame);
}

  if(state!=='play')return;

}
