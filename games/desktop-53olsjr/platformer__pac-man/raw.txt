    g.fade=Math.min(1,g.fade+dt*2);
    g.jumpCd-=dt;
    const maxS=g.frightened?165:Math.min(290,230+G.heightM*.4);
    let dir;
    if(g.frightened)dir=p.x<g.x?-1:1;
    else{const dx=p.x-g.x;dir=dx===0?(g.vx>=0?1:-1):Math.sign(dx)}
    for(const o of ghosts){
      if(o===g)continue;
      const dx=g.x-o.x,dy=g.y-o.y;
      if(Math.abs(dx)<34&&Math.abs(dy)<30)g.vx+=(dx<0?-1:1)*900*dt;
    }
    g.vx=clamp(g.vx+dir*1500*dt,-maxS,maxS);
    if(!g.frightened){
      if(g.onGround&&p.y<g.y-50&&Math.abs(p.x-g.x)<75&&g.jumpCd<=0){
        g.vy=-JUMPV*(.88+Math.random()*.12);g.jumpCd=.9;
      }else if(g.onGround&&g.jumpCd<=0&&Math.random()<dt*.5){
        g.vy=-420;g.jumpCd=1.2;
      }
    }
    g.vy=Math.min(g.vy+GRAV*dt,MAXFALL);
    const pb=g.y+g.r;
    g.x=clamp(g.x+g.vx*dt,g.r,WORLD_W-g.r);
    g.y+=g.vy*dt;
    const lp=landOn(g,pb);
    if(lp)g.x+=lp.dxMove;
    if(g.y>G.camY+viewH+260&&!rehome(g))ghosts.splice(i,1);
  }
}
function ghostVsPlayer(){
  if(G.state!=='playing'&&G.state!=='menu')return;
  const p=player;
  for(let i=ghosts.length-1;i>=0;i--){
    const g=ghosts[i];
    const dx=g.x-p.x,dy=g.y-p.y;
    if(dx*dx+dy*dy<27*27){
      if(g.frightened){
        const val=200*G.combo;
        G.score+=val;G.ghostsEaten++;G.combo++;
        floatFx(g.x,g.y-24,'+'+val,g.col,22);
        burst(g.x,g.y,18,[g.col,'#ffffff'],300,500);
        ringFx(g.x,g.y,'#ffffff',50);
        G.shake=Math.max(G.shake,7);
        sfx.eatGhost();
        ghosts.splice(i,1);
      }else if(p.invuln<=0){
        die('ghost');return;
      }
    }
  }
}

      const d=Math.abs((p.x+p.w/2)-lastSafe.x)+Math.abs(p.y-lastSafe.y)*.3;

  }

  player.x=clamp(lastSafe.x,best.x+12,best.x+best.w-12);
  player.y=best.y-player.r-2;
  player.vx=0;player.vy=0;player.jumps=1;player.invuln=2.2;
}

/* ============================== BOT (attract mode) ============================== */
function botInput(){
  const p=player;let L=false,R=false,J=false,best=null,bs=1e9;
  for(const d of dots){
    if(d.y<p.y-520||d.y>p.y+60)continue;
    const dx=d.x-p.x,dy=d.y-p.y;
    const s=dx*dx+dy*dy*(d.t==='p'?.4:1);
    if(s<bs){bs=s;best=d}
  }
  if(!best){L=p.x>470;R=p.x<430;J=p.onGround&&Math.random()<.02;return{L,R,J}}
  const dx=best.x-p.x;
  if(dx<-16)L=true;else if(dx>16)R=true;
  if(p.onGround&&best.y<p.y-48)J=true;
  else if(!p.onGround&&p.vy>0&&p.jumps>0&&best.y<p.y-100)J=true;
  for(const g of ghosts){
    if(!g.frightened&&Math.abs(g.x-p.x)<55&&Math.abs(g.y-p.y)<65){
      J=true;if(p.onGround){L=p.x>g.x;R=p.x<g.x}
    }
  }
  return{L,R,J};
}

/* ============================== FX ============================== */
function burst(x,y,n,cols,sp,grav){
  for(let i=0;i<n;i++){
    const a=rand(0,TAU),s=rand(sp*.3,sp);
    parts.push({x,y,vx:Math.cos(a)*s,vy:Math.sin(a)*s-sp*.25,life:rand(.4,.9),max:.9,s:rand(2,4.5),col:cols[i%cols.length],grav:grav||0});
  }
  if(parts.length>320)parts.splice(0,parts.length-320);
}
function ringFx(x,y,col,r1){parts.push({type:'ring',x,y,life:.5,max:.5,r0:6,r1,col})}
function floatFx(x,y,text,col,size){floats.push({x,y,text,col,size:size||20,life:1.1,max:1.1})}
function updateParts(dt){
  for(let i=parts.length-1;i>=0;i--){
    const q=parts[i];q.life-=dt;
    if(q.life<=0){parts.splice(i,1);continue}
    if(!q.type==='ring'||true){}
    if(q.type!=='ring'){q.x+=q.vx*dt;q.y+=q.vy*dt;q.vy+=(q.grav||0)*dt}
  }
  for(let i=floats.length-1;i>=0;i--){
    const f=floats[i];f.life-=dt;f.y-=42*dt;
    if(f.life<=0)floats.splice(i,1);
  }
}

/* ============================== TICK ============================== */
function camera(dt){
  const target=player.y-viewH*.62+(player.vy<0?-30:0);
  if(target<G.camY)G.camY+=(target-G.camY)*Math.min(1,dt*9);
}
function tick(dt){
  G.t+=dt;
  G.shake=Math.max(0,G.shake-30*dt);
  updatePlats(dt);
  if(G.state==='playing'||G.state==='menu'){
    let ctl;
    if(G.state==='menu'){
      const b=botInput();
      if(b.J&&!G.botPrevJ)player.bufferT=BUFFER;
      G.botPrevJ=b.J;
      ctl={left:b.L,right:b.R,held:b.J};
    }else ctl=heldCtl();
    updatePlayer(dt,ctl);
    eatDots();
    ghostVsPlayer();
    addMilestones();
  }else if(G.state==='dying'){
    G.dieT-=dt;
    if(G.dieT<=0)afterDeath();
  }
  updateGhosts(dt);
  if(G.frightT>0){
    G.frightT-=dt;
    if(G.frightT<=0){G.frightT=0;G.combo=1;for(const g of ghosts)g.frightened=false;stopSiren()}
  }
  spawnLogic(dt);
  if(G.state==='playing'||G.state==='menu'){
    camera(dt);
    if(player.y>G.camY+viewH+70)die('fall');
  }
  updateParts(dt);
  ensure();
}

/* ============================== RENDER ============================== */
function circle(x,y,r,col){ctx.fillStyle=col||ctx.fillStyle;ctx.beginPath();ctx.arc(x,y,r,0,TAU);ctx.fill()}
function rrPath(x,y,w,h,r){
  r=Math.min(r,h/2,w/2);
  ctx.beginPath();
  ctx.moveTo(x+r,y);ctx.lineTo(x+w-r,y);ctx.arcTo(x+w,y,x+w,y+r,r);
  ctx.lineTo(x+w,y+h-r);ctx.arcTo(x+w,y+h,x+w-r,y+h,r);
  ctx.lineTo(x+r,y+h);ctx.arcTo(x,y+h,x,y+h-r,r);
  ctx.lineTo(x,y+r);ctx.arcTo(x,y,x+r,y,r);ctx.closePath();
}
function drawStars(){
  for(const s of stars){
    const sy=G.camY+mod(s.yy-G.camY*s.f,viewH+40)-20;
    if(sy<G.camY-25||sy>G.camY+viewH+25)continue;
    ctx.globalAlpha=s.a;ctx.fillStyle=s.c;ctx.fillRect(s.x,sy,s.r,s.r);
  }
  ctx.globalAlpha=1;
}

  }
}

function drawGhost(g,t){

  ctx.lineTo(x+r,yb);
  for(let i=0;i<3;i++)ctx.quadraticCurveTo(x+r-w*i-w/2,yb-7,x+r-w*(i+1),yb);
  ctx.closePath();
  ctx.fillStyle=bodyCol;
  if(!g.frightened){ctx.shadowColor=bodyCol;ctx.shadowBlur=10}
  ctx.fill();ctx.shadowBlur=0;

  let dt=(now-last)/1000;last=now;

    acc+=dt;
    while(acc>=STEP){tick(STEP);acc-=STEP}

}
