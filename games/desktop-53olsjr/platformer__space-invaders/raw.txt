    }

  if(G.mode==='menu')return;

  /* ---- playing / over sim ---- */
  const gy=groundY();

  // player
  if(!P.dead){
    P.invuln=Math.max(0,P.invuln-dt);P.dropT=Math.max(0,P.dropT-dt);
    P.fireCd-=dt;P.recoil*=Math.exp(-12*dt);
    P.triple=Math.max(0,P.triple-dt);P.rapid=Math.max(0,P.rapid-dt);
    G.comboT=Math.max(0,G.comboT-dt);if(G.comboT<=0)resetCombo();
    G.hintT=Math.max(0,G.hintT-dt);

    const L=keys.has('arrowleft')||keys.has('a'),Rt=keys.has('arrowright')||keys.has('d');
    const target=(Rt?1:0)-(L?1:0);
    if(target)P.face=target;
    const acc=P.onGround?2600:1700;
    P.vx+=clamp(target*340-P.vx,-acc*dt,acc*dt);
    if(!target&&Math.abs(P.vx)<20)P.vx=0;
    P.runPhase+=Math.abs(P.vx)*dt*0.055;

    const jHold=keys.has('arrowup')||keys.has('w');
    P.coyote=P.onGround?0.1:Math.max(0,P.coyote-dt);
    P.jumpBuf=Math.max(0,P.jumpBuf-dt);
    if(P.jumpBuf>0&&P.coyote>0){P.vy=-560;P.jumpBuf=0;P.coyote=0;P.onGround=false;P.squash=0.18;sfx.jump();}
    P.vy=Math.min(920,P.vy+(jHold&&P.vy<0?1500:2600)*dt);
    const prevB=P.y+P.h;
    P.x=clamp(P.x+P.vx*dt,0,W-P.w);
    P.y+=P.vy*dt;

    // one-way platforms
    P.onGround=false;
    if(P.vy>=0){
      for(const pl of platforms){
        if(!pl.ground&&P.dropT>0)continue;
        if(prevB<=pl.y+1&&P.y+P.h>=pl.y&&P.x+P.w>pl.x+2&&P.x<pl.x+pl.w-2){
          const imp=P.vy;P.y=pl.y-P.h;P.vy=0;P.onGround=true;P.standPlat=pl.ground?'ground':'float';
          if(imp>320){P.squash=-0.24;sfx.land(imp);addShakeFx(Math.min(5,imp/220));dust(P.x+P.w/2,pl.y);}
        }
      }
    }
    P.squash*=Math.exp(-9*dt);

    // fire
    if(G.mode==='playing'&&(keys.has('space')||keys.has('x')||keys.has('j')||mouseDown)&&P.fireCd<=0)shoot();

    // jet flame particles
    if(!P.onGround&&P.vy<40)parts.push({type:'p',x:P.x+P.w/2-P.face*13,y:P.y+P.h-6,vx:rand(-15,15),vy:rand(60,140),g:0,life:.22,t:0,size:rand(2,4),col:pick(['#ffd166','#ff8a3d'])});
  }else{
    G.overT-=dt;if(G.overT<=0&&!G.overShown)showOver();
  }

  if(G.mode!=='playing')return;

  // bullets
  for(let i=bullets.length-1;i>=0;i--){const b=bullets[i];
    b.x+=b.vx*dt;b.y+=b.vy*dt;
    if(b.y<-30||b.x<-20||b.x>W+20){if(!b.hit)resetCombo();bullets.splice(i,1);continue;}
    let used=false;
    for(let j=aliens.length-1;j>=0;j--){const a=aliens[j],ax=F.fx+a.ox,ay=F.y+a.oy;
      if(b.x>ax-3&&b.x<ax+F.aw+3&&b.y>ay-3&&b.y<ay+F.ah+3){
        b.hit=true;used=true;a.hp--;a.flash=.12;sfx.hit();
        parts.push({type:'p',x:b.x,y:b.y,vx:rand(-60,60),vy:rand(-90,-20),g:100,life:.25,t:0,size:3,col:'#fff'});
        if(a.hp<=0){G.combo++;G.comboT=3;killAlien(j,true);}
        break;}}
    if(!used&&raider&&b.x>raider.x-4&&b.x<raider.x+raider.w+4&&b.y>raider.y-4&&b.y<raider.y+raider.h+4){
      b.hit=true;used=true;raider.hp--;sfx.hit();
      parts.push({type:'p',x:b.x,y:b.y,vx:rand(-60,60),vy:rand(-90,-20),g:100,life:.25,t:0,size:3,col:'#ffd166'});
      if(raider.hp<=0){boom(raider.x+raider.w/2,raider.y+raider.h/2,'#ffd166',true);
        addScore(500,raider.x+raider.w/2,raider.y,'#ffd166');G.kills++;raider=null;}}
    if(used)bullets.splice(i,1);
  }

  // aliens: breach line + player contact
  const gyNow=groundY();
  for(let i=aliens.length-1;i>=0;i--){const a=aliens[i];a.flash=Math.max(0,a.flash-dt*4);
    const ax=F.fx+a.ox,ay=F.y+a.oy;
    if(ay+F.ah>=gyNow-2){aliens.splice(i,1);boom(ax+F.aw/2,ay+F.ah,TYPES[a.type].col,true);
      G.flashA=.35;G.flashCol='rgba(255,70,90,';hurtPlayer(ax);}
    else if(!P.dead&&P.invuln<=0&&P.x<ax+F.aw&&P.x+P.w>ax&&P.y<ay+F.ah&&P.y+P.h>ay){
      G.combo++;G.comboT=3;killAlien(i,true);hurtPlayer(ax);}
  }

  // raider
  if(!raider&&aliens.length&&G.phase==='fight'&&G.wave>=3){
    G.raiderT-=dt;
    if(G.raiderT<=0){const fromL=Math.random()<.5;
      raider={x:fromL?-60:W+10,y:rand(H*0.2,H*0.38),dir:fromL?1:-1,spd:240+G.wave*18,hp:2,w:52,h:26,fireT:.7};}}
  if(raider){raider.x+=raider.dir*raider.spd*dt;raider.fireT-=dt;
    if(raider.fireT<=0){raider.fireT=.9;bombs.push({x:raider.x+raider.w/2,y:raider.y+raider.h,vy:180,vx:0,smoke:0});}
    if(raider.x<-90||raider.x>W+90)raider=null;
    else if(!P.dead&&P.invuln<=0&&P.x<raider.x+raider.w&&P.x+P.w>raider.x&&P.y<raider.y+raider.h&&P.y+P.h>raider.y){
      boom(raider.x+raider.w/2,raider.y+raider.h/2,'#ffd166',true);raider=null;hurtPlayer(raider?0:P.x);}}

  // bombs
  for(let i=bombs.length-1;i>=0;i--){const b=bombs[i];
    b.y+=b.vy*dt;b.x+=b.vx*dt;b.smoke-=dt;
    if(b.smoke<=0){b.smoke=.05;parts.push({type:'p',x:b.x,y:b.y-4,vx:rand(-8,8),vy:rand(-30,-10),g:0,life:.3,t:0,size:2.5,col:'#7a6a8a'});}
    let dead=false;
    if(!P.dead&&b.x>P.x-6&&b.x<P.x+P.w+6&&b.y>P.y-6&&b.y<P.y+P.h+6){hurtPlayer(b.x);dead=true;}
    for(const pl of platforms){if(b.y>=pl.y&&b.y-b.vy*dt<=pl.y+4&&b.x>pl.x&&b.x<pl.x+pl.w){
      boom(b.x,pl.y,'#ff5d6d',false);sfx.bomb();dead=true;break;}}
    if(dead)bombs.splice(i,1);}

  // pickups
  for(let i=pickups.length-1;i>=0;i--){const k=pickups[i];k.t+=dt;
    if(!k.landed){k.vy=Math.min(260,k.vy+500*dt);k.y+=k.vy*dt;
      for(const pl of platforms){if(k.x+13>pl.x&&k.x<pl.x+pl.w&&k.y+26>=pl.y&&k.y+26-k.vy*dt<=pl.y+4){
        k.y=pl.y-26;k.landed=true;break;}}}
    if(k.t>7){pickups.splice(i,1);continue;}
    if(!P.dead&&P.x<k.x+26&&P.x+P.w>k.x&&P.y<k.y+26&&P.y+P.h>k.y){applyPickup(k);pickups.splice(i,1);}}

  // wave flow
  if(aliens.length===0&&!F.intro){
    if(G.phase==='fight'){G.phase='clear';G.clearT=1.8;const bonus=G.wave*100;G.score+=bonus;
      sfx.waveClear();banner('WAVE '+String(G.wave).padStart(2,'0')+' CLEAR','BONUS +'+bonus);}
    else{G.clearT-=dt;if(G.clearT<=0){G.wave++;spawnWave(G.wave);sfx.waveStart();
      banner('WAVE '+String(G.wave).padStart(2,'0'),'HOSTILES INBOUND');}}
  }
}

function render(){
  ctx.setTransform(DPR,0,0,DPR,0,0);
  const shx=(Math.random()*2-1)*G.shake,shy=(Math.random()*2-1)*G.shake;
  ctx.translate(shx,shy);
  if(bgCv)ctx.drawImage(bgCv,0,0,W,H);

  // stars w/ parallax
  const cam=P.dead?0:(P.x-W/2)*0.05;
  for(let i=0;i<stars.length;i++){const s=stars[i];const L=i%3;
    const x=((s.x-cam*[0.15,0.3,0.5][L])%W+W)%W;
    const a=0.35+0.45*(0.5+0.5*Math.sin(G.t*s.sp+s.tw));
    ctx.globalAlpha=a;ctx.fillStyle=s.col;ctx.fillRect(x,s.y,s.r,s.r);}
  ctx.globalAlpha=1;
  for(const m of meteors){const a=1-m.t/m.life;
    const g=ctx.createLinearGradient(m.x,m.y,m.x-m.vx*0.12,m.y-m.vy*0.12);
    g.addColorStop(0,'rgba(207,232,255,'+(0.8*a)+')');g.addColorStop(1,'rgba(207,232,255,0)');
    ctx.strokeStyle=g;ctx.lineWidth=2;ctx.beginPath();
    ctx.moveTo(m.x,m.y);ctx.lineTo(m.x-m.vx*0.12,m.y-m.vy*0.12);ctx.stroke();}

  drawDeck();drawPlatforms();drawDangerLine();
  drawPickups();drawAliens();drawRaider();
  drawBombs();drawBullets();
  if(!P.dead)drawPlayer();
  drawParts();drawTexts();

  if(G.flashA>0){ctx.fillStyle=G.flashCol+G.flashA+')';ctx.fillRect(-20,-20,W+40,H+40);}

  if(G.mode==='menu')drawMinis();
}
