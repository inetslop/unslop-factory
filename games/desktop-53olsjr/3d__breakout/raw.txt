function proj(x,y,z){
  const X=x-cam.x-cam.sx, Y=y-cam.y-cam.sy, Z=z-cam.z;
  const yv=Y*cP+Z*sP, zv=Z*cP-Y*sP;

}

let actx=null,master=null,noiseBuf=null,muted=false;
function initAudio(){
  if(actx){ if(actx.resume)actx.resume(); return; }
  try{
    actx=new (window.AudioContext||window.webkitAudioContext)();
    master=actx.createGain(); master.gain.value=muted?0:0.5; master.connect(actx.destination);
    noiseBuf=actx.createBuffer(1,actx.sampleRate*0.6,actx.sampleRate);
    const d=noiseBuf.getChannelData(0); for(let i=0;i<d.length;i++)d[i]=R()*2-1;
  }catch(e){}
}
function tone(type,f0,f1,dur,vol,delay=0){
  if(!actx||muted)return; const t=actx.currentTime+delay;
  const o=actx.createOscillator(),g=actx.createGain();
  o.type=type; o.frequency.setValueAtTime(Math.max(1,f0),t);
  if(f1&&f1!==f0)o.frequency.exponentialRampToValueAtTime(Math.max(1,f1),t+dur);
  g.gain.setValueAtTime(.0001,t); g.gain.exponentialRampToValueAtTime(vol,t+.012);
  g.gain.exponentialRampToValueAtTime(.0001,t+dur);
  o.connect(g);g.connect(master);o.start(t);o.stop(t+dur+.05);
}
function noise(dur,vol,freq,delay=0,type='bandpass',q=1){
  if(!actx||muted)return; const t=actx.currentTime+delay;
  const s=actx.createBufferSource();s.buffer=noiseBuf;s.loop=true;
  const fl=actx.createBiquadFilter();fl.type=type;fl.frequency.value=freq;fl.Q.value=q;
  const g=actx.createGain();g.gain.setValueAtTime(vol,t);g.gain.exponentialRampToValueAtTime(.0001,t+dur);
  s.connect(fl);fl.connect(g);g.connect(master);s.start(t);s.stop(t+dur+.05);
}

function makeBall(stuck){return {x:0,y:PADTOP+BR+.03,vx:0,vy:0,stuck:!!stuck,boost:1,trail:[],tc:[120,225,255],dead:false};}
function genLevel(n){
  bricks=[]; const p=(n-1)%4, cols=10, rows=6;
  for(let r=0;r<rows;r++)for(let c=0;c<cols;c++){
    let on=false,hp=1;
    if(p===0){on=true; if(n>=3&&r>3&&(c+r)%3===0)hp=2;}
    else if(p===1){on=(r+c)%2===0||r===0; if(n>=3&&r===5&&c%2===0)hp=2;}
    else if(p===2){on=R()<.78||r<1; if(r>=4&&c%3===1)hp=2;}
    else {const d=Math.abs(c-4.5)+Math.abs(r-2.5); on=d<=4.4; if(d<1.8)hp=2;}
    if(!on)continue;
    const cell=17.2/cols;
    bricks.push({x:-17.2/2+cell*(c+.5),y:9.6+r*1.18,w:1.54,h:1.02,hp,maxHp:hp,
      col:ROWC[r],flash:0,solid:false,delay:GT+(r+c)*.045+R()*.05,anim:0,seed:R()*6.3,dead:false});

}

function hitBrick(br,cx,cy,b){
  br.flash=1; b.tc=br.col.slice();
  if(br.hp>1){
    br.hp--; sCrack(); spawnChips(cx,cy,br.col,6); addScore(15); popup(br.x,br.y,'+15',[220,235,255]); shake=Math.max(shake,.12);
  }else{
    br.dead=true; brickAlive--; statBricks++;
    combo++; maxCombo=Math.max(maxCombo,combo); bumpCombo();
    const mult=1+Math.min(combo-1,8)*.25;
    const pts=Math.round(((br.maxHp>1?80:50)*mult)/5)*5;
    addScore(pts); popup(br.x,br.y,'+'+pts,br.col);
    spawnChips(br.x,br.y,br.col,13); spawnSparks(cx,cy,[255,255,255],7,8);
    sBrick(combo); shake=Math.max(shake,br.maxHp>1?.2:.09);
    if(R()<.16)spawnCapsule(br.x,br.y);
    if(combo===3||combo===5||combo===8)toast('COMBO ×'+combo,'pink');
    if(brickAlive<=0&&bricks.every(x=>x.dead))levelCleared();
  }
}
function applyEffect(t,x){
  sPower();
  if(t==='wide'){wideT=12;popup(pad.x,3.4,'WIDE',[69,216,255]);}
  else if(t==='slow'){
    if(!slowActive)for(const b of balls){b.vx*=.78;b.vy*=.78;}
    slowActive=true;slowUntil=GT+8;popup(pad.x,3.4,'SLOW',[255,210,74]);
  }
  else if(t==='multi'){
    const flying=balls.filter(b=>!b.stuck);
    if(flying.length===0){addScore(150);popup(x,PADTOP+2,'+150',[255,92,158]);return;}
    for(const b of flying.slice()){
      if(balls.length>=6)break;
      const sp=Math.hypot(b.vx,b.vy),a=Math.atan2(b.vx,b.vy);
      for(const da of [.5,-.5]){
        if(balls.length>=6)break;
        balls.push({x:b.x,y:b.y,vx:sp*Math.sin(a+da),vy:sp*Math.cos(a+da),stuck:false,boost:b.boost,trail:[],tc:b.tc.slice(),dead:false});
      }
    }
  }
  else if(t==='life'){lives=Math.min(5,lives+1);updatePips();toast('+1 LIFE','green');}
}

/* ============================== UPDATE ============================== */
function camUpdate(dt){
  let tx;
  if(state==='play'||state==='pause'){
    const avg=balls.reduce((s,b)=>s+b.x,0)/Math.max(1,balls.length);
    tx=clamp(avg*.5,-3.4,3.4);
  }else tx=Math.sin(T*.22)*1.9+(state==='menu'?mouseNX*1.1:0);
  cam.x+=(tx-cam.x)*(1-Math.exp(-2.6*dt));
  shake*=Math.exp(-3.2*dt); if(shake<.002)shake=0;
  cam.sx=(R()*2-1)*shake*.55; cam.sy=(R()*2-1)*shake*.4;
}
function tickAmbient(dt){
  for(const d of dust){d.z+=d.v*dt;if(d.z>3.9)d.z=-6.5;}
  for(const b of bricks){
    if(!b.solid){if(GT>=b.delay)b.solid=true;continue;}
    if(b.anim<1)b.anim=Math.min(1,b.anim+dt*4);
    if(b.flash>0)b.flash=Math.max(0,b.flash-dt*3.5);
  }
}
function updatePaddle(dt){
  let ax=0; if(keys.l)ax-=1; if(keys.r)ax+=1;
  const before=pad.x;
  if(ax!==0){pad.vx=clamp(pad.vx+ax*170*dt,-24,24);}
  else{pad.vx*=Math.exp(-10*dt);if(Math.abs(pad.vx)<.05)pad.vx=0;}
  pad.x+=pad.vx*dt;
  if(ax===0&&inputMode==='mouse')pad.x+=(padTargetX-pad.x)*(1-Math.exp(-18*dt));
  pad.x=clamp(pad.x,-(XW2-pad.pw/2-.15),XW2-pad.pw/2-.15);
  pad.vx=(pad.x-before)/Math.max(dt,1e-4);
  const pwT=BASE_PW*(wideT>0?1.62:1);
  pad.pw+=(pwT-pad.pw)*(1-Math.exp(-8*dt));
  if(pad.flash>0)pad.flash=Math.max(0,pad.flash-dt*4);
}
function stepBall(b,h){
  if(b.stuck){b.x=pad.x;b.y=PADTOP+BR+.03;return;}
  b.x+=b.vx*h; b.y+=b.vy*h;
  if(b.x-BR<-XW2){b.x=-XW2+BR;b.vx=Math.abs(b.vx);sWall();spawnSparks(-XW2+.1,b.y,[140,200,255],3,4);}
  else if(b.x+BR>XW2){b.x=XW2-BR;b.vx=-Math.abs(b.vx);sWall();spawnSparks(XW2-.1,b.y,[140,200,255],3,4);}
  if(b.y+BR>YMAX&&b.vy>0){b.y=YMAX-BR;b.vy=-Math.abs(b.vy);sWall();spawnSparks(b.x,YMAX-.1,[140,200,255],3,4);}
  if(b.vy<0&&b.y-BR<=PADTOP&&b.y-BR>=PADTOP-1.4&&Math.abs(b.x-pad.x)<=pad.pw/2+BR*.6){
    b.y=PADTOP+BR;
    const rel=clamp((b.x-pad.x)/(pad.pw/2),-1,1);
    const ang=rel*1.05+clamp(pad.vx*.045,-.35,.35);
    b.boost=Math.min(b.boost+.03,1.42);
    const sp=speedBase()*b.boost*slowFac();
    b.vx=sp*Math.sin(ang);b.vy=sp*Math.cos(ang);
    combo=0;bumpCombo();pad.flash=1;sPaddle();spawnSparks(b.x,PADTOP,[69,216,255],5,5);
  }
  for(const br of bricks){
    if(!br.solid||br.dead)continue;
    if(Math.abs(b.x-br.x)>br.w/2+BR+.2||Math.abs(b.y-br.y)>br.h/2+BR+.2)continue;
    const cx=clamp(b.x,br.x-br.w/2,br.x+br.w/2),cy=clamp(b.y,br.y-br.h/2,br.y+br.h/2);
    let dx=b.x-cx,dy=b.y-cy;const d2=dx*dx+dy*dy;
    if(d2>BR*BR)continue;
    let nx,ny;const d=Math.sqrt(d2);
    if(d<1e-4){nx=0;ny=b.vy>0?-1:1;}else{nx=dx/d;ny=dy/d;}
    const dot=b.vx*nx+b.vy*ny;
    if(dot<0){b.vx-=2*dot*nx;b.vy-=2*dot*ny;}
    b.x=cx+nx*(BR+.01);b.y=cy+ny*(BR+.01);
    hitBrick(br,cx,cy,b);break;
  }
  if(b.y<KILLY)killBall(b);
}
function updateGame(dt){
  camUpdate(dt);
  flashA=Math.max(0,flashA-dt*1.8);
  dangerFlash=Math.max(0,dangerFlash-dt*1.6);
  if(state!=='play')return;
  GT+=dt;
  const sdt=dt*timeScale;
  wideT=Math.max(0,wideT-dt);
  if(slowActive&&GT>slowUntil){slowActive=false;for(const b of balls){b.vx/= .78;b.vy/=.78;}}
  updatePaddle(dt);
  if(clearing){clearT-=dt;if(clearT<=0)finishLevel();}
  bricks=bricks.filter(b=>!b.dead);
  const steps=3,h=sdt/steps;
  for(const b of balls)if(!b.dead)for(let i=0;i<steps;i++)stepBall(b,h);
  for(const b of balls){
    if(b.stuck||b.dead)continue;
    b.trail.push({x:b.x,y:b.y});if(b.trail.length>15)b.trail.shift();
  }
  for(const c of capsules){
    c.y-=c.vy*sdt;c.x+=Math.sin(T*2+c.ph)*.6*sdt;
    if(c.y< PADTOP+.3&&c.y>-0.5&&Math.abs(c.x-pad.x)<=pad.pw/2+.45){c.dead=true;applyEffect(c.t,c.x);spawnSparks(c.x,PADTOP+1,c.c,9,6);}
    else if(c.y<-1.2)c.dead=true;
  }
  capsules=capsules.filter(c=>!c.dead);
  for(const s of shards){
    s.x+=s.vx*sdt;s.y+=s.vy*sdt;s.z+=s.vz*sdt;
    if(s.k==='c'){s.vy-=13*sdt;s.rot+=s.vr*sdt;}
    s.life-=sdt;
  }
  shards=shards.filter(s=>s.life>0&&s.z>-9);
  for(const p of popups){p.t+=sdt;p.y+=2.2*sdt;}
  popups=popups.filter(p=>p.t<.95);
}

  }
}

function drawShadow(x,y,z,scale){
  const p=proj(x,FLOORY+.02,z);if(!p)return;
  const rx=.5*p[2]*(1+y*.05)*scale, a=.3/(1+y*.14);
  ctx.save();ctx.translate(p[0],p[1]);ctx.scale(1,.36);
  ctx.beginPath();ctx.arc(0,0,rx,0,6.283);ctx.fillStyle=`rgba(0,0,0,${a})`;ctx.fill();ctx.restore();
}
function drawDust(front){
  for(const d of dust){
    const isFront=d.z<ZBR_F; if(isFront!==front)continue;
    const p=proj(d.x,d.y,d.z);if(!p)continue;
    const t=clamp(1-(p[2]-9)/18,0,1);
    ctx.globalAlpha=(.05+t*.3)*d.s;ctx.fillStyle='#bfe6ff';
    const sz=Math.max(.6,.045*p[2]*d.s);
    ctx.fillRect(p[0],p[1],sz,sz);
  }
  ctx.globalAlpha=1;
}
function drawBricks(){
  for(const b of bricks){
    if(!b.solid)continue;
    const k=eob(b.anim);if(k<=.02)continue;
    const w=b.w*k,hh=b.h*k,x0=b.x-w/2,x1=b.x+w/2,y0=b.y-hh/2,y1=b.y+hh/2;
    const F0=proj(x0,y0,ZBR_F),F1=proj(x1,y0,ZBR_F),F2=proj(x1,y1,ZBR_F),F3=proj(x0,y1,ZBR_F);
    if(!F0||!F1||!F2||!F3)continue;
    const B0=proj(x0,y0,ZB),B1=proj(x1,y0,ZB),B2=proj(x1,y1,ZB),B3=proj(x0,y1,ZB);
    if(!B0||!B1||!B2||!B3)continue;
    const c=b.col;
    // visible side face
    if(b.x>cam.x){pathQ(F0,F3,B3,B0);ctx.fillStyle=css(c,.5);ctx.fill();}
    else{pathQ(F1,F2,B2,B1);ctx.fillStyle=css(c,.5);ctx.fill();}
    // visible top/bottom face
    if(b.y<cam.y){pathQ(F0,F3,B3,B0);ctx.fillStyle=css(c,.78);ctx.fill();}
    else{pathQ(F0,F1,B1,B0);ctx.fillStyle=css(c,.42);ctx.fill();}
    // front face
    pathQ(F0,F1,F2,F3);ctx.fillStyle=css(c,1,k);ctx.fill();
    line2(F3,F2,Math.max(1.5,F2[2]*.04),'rgba(255,255,255,.4)');
    line2(F0,F1,Math.max(1,F1[2]*.03),'rgba(0,0,0,.35)');
    if(b.maxHp>1){
      const m=proj(b.x,b.y,ZBR_F+.02);
      if(m){ctx.save();ctx.translate(m[0],m[1]);ctx.rotate(Math.PI/4);
        const s=m[2]*.14;ctx.fillStyle=`rgba(255,255,255,${.75*k*(b.hp>1?1:.35)})`;ctx.fillRect(-s/2,-s/2,s,s);ctx.restore();}
      if(b.hp===1){ // cracks
        const u=.25+b.seed%0.3,v=.4+((b.seed*1.7)%0.3);
        const pA=proj(lerp(x0,x1,u),y0,ZBR_F),pB=proj(lerp(x0,x1,clamp(u+.2,-.1,1.1)),lerp(y0,y1,v),ZBR_F),pC=proj(lerp(x0,x1,clamp(u+.45,-.1,1.1)),y1,ZBR_F);
        if(pA&&pB&&pC){ctx.strokeStyle='rgba(6,10,24,.7)';ctx.lineWidth=Math.max(1,m[2]*.03);
          ctx.beginPath();ctx.moveTo(pA[0],pA[1]);ctx.lineTo(pB[0],pB[1]);ctx.lineTo(pC[0],pC[1]);ctx.stroke();}
      }
    }
    if(b.flash>0){pathQ(F0,F1,F2,F3);ctx.fillStyle=`rgba(255,255,255,${b.flash*.7})`;ctx.fill();}
  }
}

function drawBalls(){

    for(let i=0;i<n;i++){

function drawCapsules(){
  for(const c of capsules){
    const p=proj(c.x,c.y,-1.5);if(!p)continue;
    const w=.95*p[2],h=.62*p[2];
    ctx.save();ctx.translate(p[0],p[1]);ctx.rotate(Math.sin(T*3+c.ph)*.45);
    ctx.globalCompositeOperation='lighter';
    let g=ctx.createRadialGradient(0,0,0,0,0,w*1.6);
    g.addColorStop(0,css(c.c,1,.3));g.addColorStop(1,css(c.c,1,0));
    ctx.fillStyle=g;ctx.beginPath();ctx.arc(0,0,w*1.6,0,6.283);ctx.fill();
    ctx.globalCompositeOperation='source-over';
    ctx.beginPath();

    ctx.moveTo(x+r,y);ctx.arcTo(x+w,y,x+w,y+h,r);ctx.arcTo(x+w,y+h,x,y+h,r);ctx.arcTo(x,y+h,x,y,r);ctx.arcTo(x,y,x+w,y,r);ctx.closePath();
    ctx.fillStyle='rgba(8,12,24,.9)';ctx.fill();
    ctx.strokeStyle=css(c.c,1,1);ctx.lineWidth=Math.max(1.5,p[2]*.05);ctx.stroke();

    ctx.textAlign='center';ctx.textBaseline='middle';ctx.fillText(c.l,0,p[2]*.03);
    ctx.restore();
  }
}
function drawShards(){
  for(const s of shards){
    const p=proj(s.x,s.y,s.z);if(!p)continue;
    const a=clamp(s.life/s.ml,0,1);

      ctx.fillRect(-sz/2,-sz*.35,sz,sz*.7);ctx.restore();

      if(p2)line2(p2,p,Math.max(1,p[2]*.05),css(s.c,1,a));
    }
  }
}
