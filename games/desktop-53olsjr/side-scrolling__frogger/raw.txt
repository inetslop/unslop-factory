function rr(c,x,y,w,h,r){r=Math.min(r,Math.abs(w)/2,Math.abs(h)/2);c.beginPath();c.moveTo(x+r,y);c.arcTo(x+w,y,x+w,y+h,r);c.arcTo(x+w,y+h,x,y+h,r);c.arcTo(x,y+h,x,y,r);c.arcTo(x,y,x+w,y,r);c.closePath();}
function mulberry32(a){return function(){a|=0;a=a+0x6D2B79F5|0;let t=Math.imul(a^a>>>15,1|a);t=t+Math.imul(t^t>>>7,61|t)^t;return((t^t>>>14)>>>0)/4294967296;};}

function getNoise(){ if(!noiseBuf){noiseBuf=AC.createBuffer(1,AC.sampleRate*.5,AC.sampleRate);const d=noiseBuf.getChannelData(0);for(let i=0;i<d.length;i++)d[i]=Math.random()*2-1;} return noiseBuf; }

function makeLanes(lv){
  const r=mulberry32(lv*7919+17), m=speedMult(), L=[];
  const defs=[
    {x:1,type:'road',veh:'car',  base:2.2},
    {x:2,type:'road',veh:'truck',base:1.5},
    {x:3,type:'water',logs:[3,4],base:.8},
    {x:4,type:'water',logs:[3],  base:1.4},
    {x:5,type:'water',logs:[4,2],base:.6},
    {x:6,type:'road',veh:'bike', base:3.0},
    {x:7,type:'road',veh:'car',  base:2.0}];
  for(const d of defs){
    const dir=r()<.5?1:-1;
    const sp=d.base*(.9+r()*.2)*m*(d.type==='water'?.85:1);
    const items=[]; let cur=0,i=0;
    if(d.type==='road'){
      const len=d.veh==='truck'?2:d.veh==='bike'?.9:1;
      const gmin=d.veh==='bike'?1.6:d.veh==='truck'?3:2.4, gmax=d.veh==='bike'?2.7:d.veh==='truck'?5:4;
      while(cur<PERIOD){ if(cur+len<=PERIOD)items.push({y:cur,len,ci:(r()*CARCOL.length)|0}); cur+=len+gmin+r()*(gmax-gmin); }

  }
  return L;
}

/* ================= gameplay ================= */
const HOPD=()=>Math.max(.1,.14-(level-1)*.003);
function tryHop(dx,dy){
  if(state!=='play'||!frog||!frog.alive||frog.hop)return;
  const bx=Math.round(frog.x),by=Math.round(frog.y),tx=bx+dx,ty=by+dy;
  if(tx<0||tx>=COLS||ty<0||ty>=ROWS)return;
  frog.aT = dy<0?0 : dx>0?Math.PI/2 : dy>0?Math.PI : -Math.PI/2;
  const laneAt=(bx>=1&&bx<=7)?lanes[bx-1]:null;
  if(!(laneAt&&laneAt.type==='water')){ for(let k=0;k<4;k++){const a=Math.random()*6.28;addP({x:bx+.5,y:by+.8,vx:Math.cos(a)*.8,vy:Math.sin(a)*.5-.7,g:3,type:'dust',life:.4,size:.05,color:'#cbb88a'});} }
  frog.hop={fx:bx,fy:by,tx,ty,t:0,dur:HOPD()};
  SFX.hop();
}
function logUnder(ln,y){ for(const it of ln.items){ const p=norm(it.y+ln.off,PERIOD); if(y>p-.2&&y<p+it.len+.2)return it; const p2=p-PERIOD; if(y>p2-.2&&y<p2+it.len+.2)return it; } return null; }
function onLand(){
  const li=Math.floor(frog.x), ln=(li>=1&&li<=7)?lanes[li-1]:null, w=ln&&ln.type==='water';
  if(w&&!logUnder(ln,frog.y)){die('drown');return;}
  if(w&&!frog.inWater){SFX.splash();splashAt(frog.x+.5,frog.y+.5,8);}
  frog.inWater=w;
}
function updateFrog(dt){
  if(frog.hop){
    const h=frog.hop; h.t+=dt; const k=Math.min(1,h.t/h.dur);
    frog.x=h.fx+(h.tx-h.fx)*k; frog.y=h.fy+(h.ty-h.fy)*k;
    if(k>=1){frog.x=h.tx;frog.y=h.ty;frog.hop=null;onLand();return;}
  }else{
    const li=Math.floor(frog.x);
    if(li>=1&&li<=7){
      const ln=lanes[li-1];
      if(ln.type==='water'){
        if(!logUnder(ln,frog.y)){die('drown');return;}
        frog.y+=ln.dir*ln.speed*dt;
        if(frog.y<-.45||frog.y>ROWS-.55){die('drown');return;}
      }
    }
  }
  const col=Math.round(frog.x);
  if(col>maxCol&&col<=7){maxCol=col;score+=5;SFX.blip();}
  for(const ln of lanes){
    if(ln.type!=='road')continue;
    if(Math.abs(frog.x-(ln.x+.5))>.54)continue;
    for(const it of ln.items){
      const p=norm(it.y+ln.off,PERIOD);
      if(frog.y>p-.2&&frog.y<p+it.len+.2){die('splat');return;}
      const p2=p-PERIOD;
      if(frog.y>p2-.2&&frog.y<p2+it.len+.2){die('splat');return;}
    }
  }
  if(Math.round(frog.x)===PADX&&!frog.hop){
    for(const pd of pads){ if(!pd.filled&&Math.round(frog.y)===pd.y){doClear(pd);return;} }
  }
}

  pd.filled=true;
  const pts=25+Math.round(timeLeft)*3; score+=pts;
  pops.push({x:PADX+.5,y:pd.y+.5,txt:'+'+pts,t:0,life:1.2,color:'#ffe9a0',size:.6});
  state='clear'; clearT=0; SFX.clear();
  for(let k=0;k<16;k++){const a=Math.random()*6.28,s=.8+Math.random()*1.8;addP({x:PADX+.5,y:pd.y+.5,vx:Math.cos(a)*s,vy:Math.sin(a)*s,type:'spark',life:.7,size:.05,color:k%2?'#ffd76a':'#7be85f'});}
}
function die(cause){
  if(state!=='play')return;
  lastCause=cause; state='dying'; frog.deadT=0; lives--; shake=9;
  if(cause==='splat'){
    frog.alive=false; SFX.splat();

    frog.alive=false; SFX.splash(); splashAt(frog.x+.5,Math.min(Math.max(frog.y,.2),ROWS-.2)+.5,14);
  }else{
    SFX.life(); pops.push({x:frog.x+.5,y:frog.y-.4,txt:'TIME UP!',t:0,life:1.2,color:'#f2c94c',size:.5});
  }
}

/* ================= update ================= */
function updateLanes(dt){ for(const ln of lanes) ln.off=norm(ln.off+ln.dir*ln.speed*dt,PERIOD); }
function update(dt){
  if(state!=='paused'){ tG+=dt; updateLanes(dt); }
  if(state==='play'){
    timeLeft-=dt;
    const c=Math.ceil(timeLeft);
    if(c!==lastTick&&c<=5&&c>0)SFX.tick();
    lastTick=c;
    if(timeLeft<=0){timeLeft=0;die('time');}
    else{
      if(frog.alive)updateFrog(dt);
      frog.angle+=shortAngle(frog.aT-frog.angle)*Math.min(1,dt*14);
      frog.nextBlink-=dt; if(frog.nextBlink<0){frog.blink=.13;frog.nextBlink=2+Math.random()*3;}
      if(frog.blink>0)frog.blink-=dt;
      frog.alpha=Math.min(1,frog.alpha+dt*2.5);
      if(introT>0)introT-=dt;
    }
  }else if(state==='dying'){
    frog.deadT+=dt;
    if(frog.deadT>1){
      if(lives<=0)gameOver();
      else{frog=makeFrog(true);maxCol=0;state='play';}
    }
  }else if(state==='clear'){
    clearT+=dt;
    if(clearT>1.6){level++;setupLevel();state='play';}
  }
  for(let i=parts.length-1;i>=0;i--){const p=parts[i];p.t+=dt;if(p.t>p.life){parts.splice(i,1);continue;}p.vy+=(p.g||0)*dt;p.x+=p.vx*dt;p.y+=p.vy*dt;}
  for(let i=pops.length-1;i>=0;i--){const p=pops[i];p.t+=dt;if(p.t>p.life)pops.splice(i,1);}
  shake=Math.max(0,shake-dt*26);
}

function drawGutters(){
  const gutL=fx0, gutR=W-fx0-fieldW;
  if(gutL>70||gutR>70){
    ctx.lineCap='round';
    for(const rd of reeds){
      const gutter=rd.side?gutR:gutL; if(gutter<70)continue;
      const inner=rd.side?(fx0+fieldW)+14:fx0-14;
      const outer=rd.side?W-gutter*.35:gutter*.35;
      const ax=rd.inner?inner:outer, yb=H0+rd.ny*(H-H0), hh=rd.h*cell*.9;
      const sway=Math.sin(tG*1.1+rd.ph)*4;
      ctx.strokeStyle='rgba(24,74,51,.85)';ctx.lineWidth=3;
      ctx.beginPath();ctx.moveTo(ax,yb);ctx.quadraticCurveTo(ax+sway*.3,yb-hh*.6,ax+rd.lean*14+sway,yb-hh);ctx.stroke();
      if(rd.head){

        ctx.fillStyle='#5b3a1e';rr(ctx,hx-4,hy-6,8,20,4);ctx.fill();
        ctx.fillStyle='#3c2412';dot(hx,hy-8,2.5);
      }
    }
  }
  ctx.save();ctx.globalCompositeOperation='lighter';
  for(const f of flies){
    const x=f.nx*W+Math.sin(tG*f.sp+f.ph)*26, y=H0+f.ny*(H-H0)+Math.cos(tG*f.sp*.7+f.ph*1.9)*18;

  }

  ctx.save();if(!on)ctx.globalAlpha=.32;
  ctx.fillStyle=on?'#4cc43e':'#3a5a72';
  ctx.beginPath();ctx.arc(x,y,r,0,6.283);ctx.fill();
  ctx.strokeStyle='rgba(0,0,0,.45)';ctx.lineWidth=1.5;ctx.stroke();
  for(const s of [-1,1]){
    ctx.fillStyle=on?'#ffdf6e':'#2c4258';
    ctx.beginPath();ctx.arc(x+s*r*.62,y-r*.8,r*.5,0,6.283);ctx.fill();

}

function drawFrogState(){
  if(!frog)return;
  const u=cell*.1;
  let px=fx0+(frog.x+.5)*cell, py=fy0+(frog.y+.5)*cell;
  if(frog.alive){
    const hopK=frog.hop?Math.sin(Math.PI*Math.min(1,frog.hop.t/frog.hop.dur)):0;
    ctx.fillStyle='rgba(0,0,0,'+(.3*(1-hopK*.5))+')';
    ctx.beginPath();ctx.ellipse(px,py+u*(1.2+hopK*1.4),u*1.6,u*.7,0,0,6.283);ctx.fill();
    const sc=(1+hopK*.25)*(state==='clear'?1+Math.sin(tG*9)*.05:1);
    drawFrogBody(px,py-u*hopK*.8,frog.angle,sc*frog.alpha===0?1:sc,frog.alpha,u,frog.blink>0);
    if(state!=='play'&&lastCause==='time'){
      for(let i=0;i<3;i++){const a=tG*4+i*2.09;star4(px+Math.cos(a)*cell*.5,py-cell*.95+Math.sin(a)*cell*.18,cell*.09);}
    }
  }else if(lastCause==='splat'){
    const k=Math.min(1,frog.deadT/.35),R=u*(1.2+2.4*k);
    ctx.strokeStyle='#2f9e33';ctx.lineWidth=u*.55;ctx.lineCap='round';
    for(let i=0;i<4;i++){const a=i*Math.PI/2+.6+k*.4;
      ctx.beginPath();ctx.moveTo(px+Math.cos(a)*R*.6,py+Math.sin(a)*R*.4);ctx.lineTo(px+Math.cos(a)*(R+u*1.3*k),py+Math.sin(a)*(R*.5+u*1.3*k));ctx.stroke();}
    ctx.fillStyle='#3aa23e';ctx.strokeStyle='rgba(8,40,10,.5)';ctx.lineWidth=u*.3;
    ctx.beginPath();ctx.ellipse(px,py,R*1.15,R*.6,0,0,6.283);ctx.fill();ctx.stroke();
    for(const s of [-1,1]){
      dot(px+s*u*.9,py-u*.15,u*.3);
      ctx.strokeStyle='#131313';ctx.lineWidth=2;
      ctx.beginPath();ctx.moveTo(px+s*u*.9-u*.16,py-u*.15-u*.16);ctx.lineTo(px+s*u*.9+u*.16,py-u*.15+u*.16);
      ctx.moveTo(px+s*u*.9+u*.16,py-u*.15-u*.16);ctx.lineTo(px+s*u*.9-u*.16,py-u*.15+u*.16);ctx.stroke();
    }
  }else{ // drowned
    const k=Math.min(1,frog.deadT/.9);
    py=fy0+(Math.min(Math.max(frog.y,-.3),ROWS-.7)+.5)*cell;
    ctx.strokeStyle='rgba(191,226,255,'+((1-k)*.6)+')';ctx.lineWidth=2.5;
    ctx.beginPath();ctx.arc(px,py,u*(1+3*k),0,6.283);ctx.stroke();
    drawFrogBody(px,py,frog.angle,Math.max(.05,1-k*.7),1-k,u,false);
  }
  ctx.globalAlpha=1;
}

function dirFromKey(k){
  switch(k){
    case 'ArrowUp':case 'w':case 'W':return[0,-1];
    case 'ArrowDown':case 's':case 'S':return[0,1];
    case 'ArrowLeft':case 'a':case 'A':return[-1,0];
    case 'ArrowRight':case 'd':case 'D':return[1,0];
  }
  return null;
}

  const k=e.key;
  if(k==='ArrowUp'||k==='ArrowDown'||k==='ArrowLeft'||k==='ArrowRight'||k===' ')e.preventDefault();
  unlockAudio();
  if(k==='m'||k==='M'){muted=!muted;return;}

  const d=dirFromKey(k);
  if(d&&state==='play')tryHop(d[0],d[1]);
});
