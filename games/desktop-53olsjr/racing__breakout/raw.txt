      ball.y=cy0-ball.r-0.5;combo=0;sfx.bounce();
      for(let i=0;i<4;i++)spawnPart('spark',ball.x,ball.y+6,(Math.random()-0.5)*200,-120-Math.random()*120,0.3,2,'#ffffff');
    }
    // bricks
    for(let bi=0;bi<bricks.length;bi++){
      const b=bricks[bi];if(!b.alive)continue;
      const nx=clamp(ball.x,b.x,b.x+b.w),ny=clamp(ball.y,b.y,b.y+b.h);
      let dx=ball.x-nx,dy=ball.y-ny;
      if(dx*dx+dy*dy>ball.r*ball.r)continue;
      const d=Math.hypot(dx,dy),pen=ball.r-d;
      if(dx===0&&dy===0){ball.y=b.y-ball.r-0.1;ball.vy=-Math.abs(ball.vy);}
      else if(Math.abs(dx)>Math.abs(dy)){ball.x+=Math.sign(dx)*pen;ball.vx=(dx>0?1:-1)*Math.abs(ball.vx);}
      else{ball.y+=Math.sign(dy)*pen;ball.vy=(dy>0?1:-1)*Math.abs(ball.vy);}
      hitBrick(b);break;
    }
    if(!ball.stuck&&ball.y>GH+30){loseLife();return;}
  }
  normalizeSpeed();
}
function update(dt){
  t+=dt;
  // engine sound
  if(AC&&!muted){
    const running=state==='play'&&!paused;
    const tg=running?0.028+car.thr*0.035+(nitroT>0?0.02:0):0;
    engGain.gain.setTargetAtTime(tg,AC.currentTime,0.08);
    const f=46+car.thr*150+(nitroT>0?70:0);
    engOsc.frequency.setTargetAtTime(f,AC.currentTime,0.06);
    engOsc2.frequency.setTargetAtTime(f/2,AC.currentTime,0.06);
  }
  if(state==='menu'){
    off+=520*dt;
    car.x=GW/2+Math.sin(t*0.7)*GW*0.28;car.tilt=Math.cos(t*0.7)*0.14;car.thr=0.45;
    if(Math.random()<dt*6)spawnPart('smoke',car.x+(Math.random()-0.5)*14,car.y+46,(Math.random()-0.5)*30,120,0.8,9,'#7d8598');
    updateFx(dt);return;
  }
  if(state==='win'){gameOverWinConfetti();updateFx(dt);return;}
  if(state==='over'||paused){if(paused)return;return;}
  if(state!=='play')return;

  off+=(300+car.thr*900+(nitroT>0?600:0))*dt;
  nitroT=Math.max(0,nitroT-dt);wideT=Math.max(0,wideT-dt);
  updateCar(dt);

  if(phase==='count'){
    countT-=dt;
    const lit=Math.min(3,Math.floor(3-countT));
    if(lit!==lastLight){if(lit>0)sfx.light(lit);lastLight=lit;}
    if(countT<=0){phase='serve';serveT=5;goT=0.6;sfx.go();}
    followStuck();
  }else if(phase==='serve'){
    serveT-=dt;if(serveT<=0)launch();
    followStuck();
  }else if(phase==='clear'){
    clearT-=dt;if(clearT<=0)nextLap();
    followStuck();
  }else if(phase==='live'){
    ballStep(dt);
    if(aliveCount<=0&&phase==='live')startClear();
  }
  goT=Math.max(0,goT-dt);

  // powerups
  for(let i=powerups.length-1;i>=0;i--){
    const p=powerups[i];p.y+=150*dt;
    if(p.y>GH+40){powerups.splice(i,1);continue;}
    const cw=carW();
    if(Math.abs(p.x-car.x)<cw/2+18&&Math.abs(p.y-car.y)<60){applyPU(p);powerups.splice(i,1);}
  }
  // trail
  if(!ball.stuck){trail.push({x:ball.x,y:ball.y});if(trail.length>14)trail.shift();}
  updateFx(dt);
}
function updateFx(dt){
  for(let i=particles.length-1;i>=0;i--){
    const p=particles[i];p.life-=dt;
    if(p.life<=0){particles.splice(i,1);continue;}
    p.x+=p.vx*dt;p.y+=p.vy*dt;p.rot+=p.vr*dt;
    if(p.type==='smoke'){p.vy-=30*dt;p.size+=26*dt;}
    if(p.type==='conf')p.vy+=140*dt;
  }
  for(let i=floats.length-1;i>=0;i--){const f=floats[i];f.life-=dt;f.y-=42*dt;if(f.life<=0)floats.splice(i,1);}
  bricks.forEach(b=>{if(b.alive)b.flash=Math.max(0,b.flash-dt*5);});
  shake=Math.max(0,shake-dt*30);
}

let vig=null;
function drawMargins(s,ox,oy){
  if(ox>8){
    for(const mx of [0,W-ox]){
      ctx.fillStyle='#0d1119';ctx.fillRect(mx,0,ox,H);
      ctx.save();ctx.beginPath();ctx.rect(mx,0,ox,H);ctx.clip();
      const per=52*s,o=(t*40*s)%per;
      for(let y=-per;y<H+per;y+=per){
        ctx.fillStyle=((y/per)|0)%2?'rgba(230,72,61,.4)':'rgba(242,237,228,.18)';
        ctx.beginPath();ctx.moveTo(mx,y+o);ctx.lineTo(mx+ox,y+o-24*s);ctx.lineTo(mx+ox,y+o+16*s-24*s);ctx.lineTo(mx,y+o+16*s);ctx.closePath();ctx.fill();
      }
      ctx.restore();
    }
    ctx.strokeStyle='rgba(255,90,70,.28)';ctx.lineWidth=6;ctx.strokeRect(ox-3,oy-3,GW*s+6,GH*s+6);
    ctx.strokeStyle='rgba(255,90,70,.6)';ctx.lineWidth=2;ctx.strokeRect(ox,oy,GW*s,GH*s);
  }
}
function render(){
  if(!vig){vig=ctx.createRadialGradient(GW/2,GH*0.55,GH*0.28,GW/2,GH*0.55,GH*0.78);vig.addColorStop(0,'rgba(0,0,0,0)');vig.addColorStop(1,'rgba(0,0,0,.4)');}
  ctx.setTransform(dpr,0,0,dpr,0,0);
  ctx.fillStyle='#0a0d14';ctx.fillRect(0,0,W,H);
  const s=Math.min(W/GW,H/GH),ox=(W-GW*s)/2,oy=(H-GH*s)/2;
  drawMargins(s,ox,oy);
  ctx.save();
  ctx.translate(ox,oy);ctx.scale(s,s);
  ctx.beginPath();ctx.rect(0,0,GW,GH);ctx.clip();
  if(shake>0)ctx.translate((Math.random()-0.5)*shake,(Math.random()-0.5)*shake);
  drawRoad();
  powerups.forEach(drawPU);
  bricks.forEach(drawBrick);
  drawParticles();
  if(state!=='menu')drawCar(car.x,car.y,car.tilt,{steer:clamp(car.vx/780,-1,1),thr:car.thr,nitro:nitroT>0,wide:wideT>0});
  else drawCar(car.x,car.y,car.tilt,{steer:Math.sin(t*0.7)*0.5,thr:0.5,nitro:false,wide:false});
  if(state==='play')drawBall();
  drawFloats();
  const tg=ctx.createLinearGradient(0,HUD_H,0,HUD_H+60);tg.addColorStop(0,'rgba(0,0,0,.4)');tg.addColorStop(1,'rgba(0,0,0,0)');
  ctx.fillStyle=tg;ctx.fillRect(0,HUD_H,GW,60);
  if(state!=='menu')drawHUD();
  ctx.fillStyle=vig;ctx.fillRect(0,0,GW,GH);
  if(state==='play'){if(phase==='count'||goT>0)drawCountdown();}
  if(paused&&state==='play')drawPause();
  if(state==='menu')drawMenu();
  if(state==='over')drawEnd(false);
  if(state==='win')drawEnd(true);
  ctx.restore();
}
