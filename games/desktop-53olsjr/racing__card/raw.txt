function frame(ts){const dt=Math.min(.05,(ts-last)/1000||.016);last=ts;tick(dt,ts/1000);draw(ts/1000);requestAnimationFrame(frame);}
function tick(dt,t){
  const idle=phase==='start';
  let maxV=0;
  for(const r of[ {o:C,idle},{o:P,idle} ]){const o=r.o;
    if(idle){o.anim=2;o.vel=0;}else{o.prev=o.anim;o.anim+=(o.dist-o.anim)*Math.min(1,dt*5.5);o.vel=(o.anim-o.prev)/dt;}
    const vpx=Math.abs(o.vel)*(W-32-carW())/RACE;maxV=Math.max(maxV,vpx);
    o.boost=Math.max(0,o.boost-dt*1.1);o.spinT=Math.max(0,o.spinT-dt/.9);o.flashT=Math.max(0,o.flashT-dt);}
  scroll+=(8+maxV)*dt;shake=Math.max(0,shake-dt*1.7);
  for(let i=parts.length-1;i>=0;i--){const p=parts[i];p.life-=dt;if(p.life<=0){parts.splice(i,1);continue;}
    if(p.type==='spark'){p.vy+=170*dt;p.x+=p.vx*dt;p.y+=p.vy*dt;}
    else if(p.type==='smoke'){p.x+=p.vx*dt;p.y+=p.vy*dt;}
    else if(p.type==='ring')p.r+=p.vr*dt;
    else if(p.type==='txt')p.y+=p.vy*dt;}
  if(parts.length>420)parts.splice(0,parts.length-420);
}

function resetGame(){runId++;P=newRacer();C=newRacer();round=1;stats={maxGain:0,megas:0};
  drawCards(P,START_HAND);drawCards(C,START_HAND);parts.length=0;shake=0;sel=0;phase='pick';
  $('#startOvl').classList.add('off');$('#endOvl').classList.add('off');
  vsReset();renderAll();setPhaseUI();}

function cpuChoose(){const h=C.hand,has=k=>h.includes(k),gap=P.dist-C.dist;
  if(gap>12&&has('OIL')&&Math.random()<.7)return 'OIL';
  if(C.nitro>=MAXNITRO){const f=h.filter(k=>TYPES[k].f);if(f.length)return f.sort((a,b)=>TYPES[b].g-TYPES[a].g)[0];}
  if(has('SHIELD')&&gap<8&&Math.random()<.35)return 'SHIELD';
  const fwd=h.filter(k=>TYPES[k].f);
  if(!fwd.length)return has('OIL')&&gap>0?'OIL':'SHIELD';
  const score=k=>{const t=TYPES[k];return t.g+(t.ch?2:0)+(t.dr?(h.length<=3?3:0):0)+Math.random()*6;};
  return fwd.sort((a,b)=>score(b)-score(a))[0];}

function launch(){if(phase!=='pick')return;ensureAC();
  phase='reveal';setPhaseUI();sfx.go();
  const pk=P.hand[sel];P.disc.push(P.hand.splice(sel,1)[0]);
  const ck=cpuChoose();C.disc.push(C.hand.splice(C.hand.indexOf(ck),1)[0]);
  renderAll();
  const pp=$('#pPlayed'),cp=$('#cpuPlayed');
  pp.className='card mini vscard inR';pp.style.setProperty('--tc',TYPES[pk].c);pp.innerHTML=cardHTML(pk);
  cp.className='card mini vscard down inL';cp.dataset.k=ck;
  $('#vs').classList.remove('dim');$('#vs').classList.add('on');
  $('#pRes').innerHTML='';$('#cpuRes').innerHTML='';$('#vsLine').textContent='';
  const id=runId;
  setTimeout(()=>{if(id!==runId)return;sfx.flip();
    cp.className='card mini vscard flip';cp.style.setProperty('--tc',TYPES[ck].c);cp.innerHTML=cardHTML(ck);},720);
  setTimeout(()=>{if(id!==runId)return;resolveRound(pk,ck);},1300);}

function carXY(r){const g=geo();return{x:cxFor(r.anim),y:r===P?g.cy2:g.cy1};}
function applyFx(r,gain,mega,key,pen,hitByOil,shieldUsed){
  r.boost=Math.min(2.4,.6+gain/28)+(mega?1:0);
  const pos=carXY(r);
  if(mega){shake=Math.max(shake,.9);ringAt(pos.x,pos.y,r===P?'#ffb400':'#2fd6ff');
    setTimeout(()=>ringAt(pos.x,pos.y,'#fff'),120);textAt(pos.x,pos.y-34,'MEGA!','#ffd21f',22);sfx.mega();}
  else if(gain>0){sfx.burst(gain);textAt(pos.x,pos.y-30,'+'+gain+' m',TYPES[key].c,18);
    if(key==='DRIFT'){smokeAt(pos.x-carW()*.55,pos.y,7,'rgba(155,225,93,.5)');sfx.drift();}
    if(key==='PIT'){burst(pos.x,pos.y-10,10,'#ffd21f',110);sfx.pit();}}
  if(hitByOil){r.spinT=1;r.flashT=.4;shake=Math.max(shake,.5);
    burst(pos.x,pos.y,22,'#c04dff',200);burst(pos.x,pos.y,10,'#3a1d5e',140);
    textAt(pos.x,pos.y-30,'−'+pen+' m','#c04dff',18);sfx.hit();}
  if(shieldUsed){burst(pos.x,pos.y-8,14,'#ffffff',150);ringAt(pos.x,pos.y,'#a8b8cf');sfx.ping();}}

function resolveRound(pk,ck){phase='resolve';
  const A=TYPES[pk],B=TYPES[ck];
  let pg=A.g||0,cg=B.g||0,pPen=0,cPen=0,line='',pMega=false,cMega=false,bP=false,bC=false;
  if(A.f&&P.nitro>=MAXNITRO){pg*=2;P.nitro=0;pMega=true;stats.megas++;}
  if(B.f&&C.nitro>=MAXNITRO){cg*=2;C.nitro=0;cMega=true;}
  const oilP=pk==='OIL',oilC=ck==='OIL';
  if(oilP&&oilC){pPen=cPen=7;line='OIL STORM — BOTH CARS SPIN!';}
  else{
    if(oilP){if(ck==='SHIELD')bC=true;else cPen=A.atk;}
    if(oilC){if(pk==='SHIELD')bP=true;else pPen=B.atk;}
    if(bC)line='YOUR OIL WAS BLOCKED!';
    if(bP)line="VOLT'S OIL WAS BLOCKED!";}
  if(pMega&&!line)line='MEGA NITRO! ×2 POWER';
  if(cMega&&!line)line="VOLT'S MEGA NITRO!";
  if(A.ch){P.nitro=Math.min(MAXNITRO,P.nitro+1);}
  if(B.ch){C.nitro=Math.min(MAXNITRO,C.nitro+1);}
  pg=Math.max(0,pg-pPen);cg=Math.max(0,cg-cPen);
  P.dist=Math.min(RACE,P.dist+pg);C.dist=Math.min(RACE,C.dist+cg);
  stats.maxGain=Math.max(stats.maxGain,pg);
  applyFx(P,pg,pMega,pk,pPen,oilC&&!bP,bP&&pk==='SHIELD');
  applyFx(C,cg,cMega,ck,cPen,oilP&&!bC,bC&&ck==='SHIELD');
  $('#pRes').innerHTML='<b>'+(bP&&pk==='SHIELD'?'BLK! ':'')+'+'+pg+' m</b>';
  $('#cpuRes').innerHTML='<b style="color:var(--cyan)">'+(bC&&ck==='SHIELD'?'BLK! ':'')+'+'+cg+' m</b>';
  if(pPen>0)$('#pRes').innerHTML+='<br><span style="color:#c04dff">−'+pPen+'</span>';
  if(cPen>0)$('#cpuRes').innerHTML+='<br><span style="color:#c04dff">−'+cPen+'</span>';
  $('#vsLine').textContent=line;
  updateHUD();
  const id=runId,roundsPlayed=round;
  setTimeout(()=>{if(id!==runId)return;afterRound(pk,ck,roundsPlayed);},1550);}

function afterRound(pk,ck,rp){
  drawCards(P,1+(pk==='PIT'?1:0));drawCards(C,1+(ck==='PIT'?1:0));
  round=rp+1;renderAll();updateHUD();
  if(P.dist>=RACE||C.dist>=RACE){endRace(rp);return;}
  phase='pick';sel=0;renderHand();setSelInfo();$('#vs').classList.add('dim');sfx.draw();setPhaseUI();}

function endRace(rp){phase='end';setPhaseUI();
  let verdict;
  if(P.dist>=RACE&&C.dist>=RACE)verdict=P.dist===C.dist?'tie':(P.dist>C.dist?'win':'lose');
  else verdict=P.dist>=RACE?'win':'lose';
  const id=runId;
  setTimeout(()=>{if(id!==runId)return;verdict==='win'?sfx.fanfare():sfx.lose();},350);
  setTimeout(()=>{if(id!==runId)return;
    const et=$('#endTitle');et.className=verdict;
    et.textContent=verdict==='win'?'CHECKERED FLAG — YOU WIN!':verdict==='lose'?'VOLT TAKES THE FLAG':'PHOTOFINISH DEAD HEAT';
    $('#stRounds').textContent=rp;$('#stRun').textContent=stats.maxGain+' m';$('#stMega').textContent=stats.megas;
    $('#stMarks').innerHTML='<span style="color:var(--amber)">YOU '+Math.round(P.dist)+'m</span> · <span style="color:var(--cyan)">VOLT '+Math.round(C.dist)+'m</span>';
    let bl='';
    if(verdict==='win'){const rec=setBest(rp);bl=rec?'★ NEW RECORD — FASTEST WIN EVER! ★':'BEST WIN: '+getBest()+' ROUNDS';}
    else if(getBest())bl='BEST WIN: '+getBest()+' ROUNDS';
    $('#endBest').textContent=bl;$('#bestChip').textContent=getBest()?('BEST '+getBest()+' RND'):'BEST —';
    $('#endOvl').classList.remove('off');},1000);}
