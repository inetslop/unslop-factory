function raiseAlert(g,pos){g.state='alert';g.lastSeen=pos;S.alerts++;sAlert();ring(g.x,g.y,'255,84,73',34,.4,2);}
function moveGuard(g,a,st){let nx=g.x+Math.cos(a)*st,ny=g.y+Math.sin(a)*st;
  if(!solidAt(nx,ny)){g.x=nx;g.y=ny;return;}
  for(const off of [.9,-.9,1.8,-1.8]){const b=a+off;nx=g.x+Math.cos(b)*st;ny=g.y+Math.sin(b)*st;
    if(!solidAt(nx,ny)){g.x=nx;g.y=ny;return;}}}
function updateGuards(dt){
  S.maxM=0;const p=S.player;
  for(const g of S.guards){
    if(g.state==='patrol'){
      const wp=g.wps[g.wi],d=dist(g.x,g.y,wp[0],wp[1]);
      if(d<4){g.wait-=dt;g.face+=dt*.85;
        if(g.wait<=0){g.wi=(g.wi+1)%g.wps.length;g.wait=.5+Math.random()*.8;}}
      else{const a=Math.atan2(wp[1]-g.y,wp[0]-g.x);g.face=turnToward(g.face,a,5,dt);moveGuard(g,a,85*dt);}
    }else{
      const ls=g.lastSeen,d=dist(g.x,g.y,ls.x,ls.y);
      if(d>14){const a=Math.atan2(ls.y-g.y,ls.x-g.x);g.face=turnToward(g.face,a,7,dt);moveGuard(g,a,132*dt);}
      else g.face+=dt*1.8;
    }
    if(g.interest&&S.t<g.interest.until)
      g.face=turnToward(g.face,Math.atan2(g.interest.y-g.y,g.interest.x-g.x),6,dt);
    if(S.mode==='play'){
      const d=dist(g.x,g.y,p.x,p.y);
      const range=g.state==='alert'?250:215,half=g.state==='alert'?.8:.62;
      let vis=false;
      if(d<45)vis=true;
      else if(!p.hidden&&d<=range){const a=Math.atan2(p.y-g.y,p.x-g.x);
        if(Math.abs(angDiff(g.face,a))<half&&rayFree(g.x,g.y,p.x,p.y))vis=true;}
      if(vis)g.meter+=dt*(0.75+1.8*(1-clamp(d/range,0,1)));
      else g.meter-=dt*(g.state==='alert'?.34:.5);
      g.meter=clamp(g.meter,0,1);
      S.maxM=Math.max(S.maxM,g.meter);
      if(g.state==='patrol'&&g.meter>=1)raiseAlert(g,{x:p.x,y:p.y});
      else if(g.state==='alert'&&g.meter<.3)g.state='patrol';
      if(dist(g.x,g.y,p.x,p.y)<42){doCaught(S.guards.indexOf(g));}
    }
  }
}

function update(dt){
  S.t+=dt;S.shake=Math.max(0,S.shake-dt*22);S.flashA=Math.max(0,S.flashA-dt*1.5);
  for(let i=S.pings.length-1;i>=0;i--){const p=S.pings[i];p.r+=400*dt;if(p.r>=p.maxR)S.pings.splice(i,1);}
  for(let i=S.parts.length-1;i>=0;i--){const q=S.parts[i];q.life-=dt;
    if(q.life<=0){S.parts.splice(i,1);continue;}
    if(q.tp==='ring')q.r+=q.vr*dt;
    if(q.tp==='dot'){q.x+=q.vx*dt;q.y+=q.vy*dt;q.vx*=.92;q.vy*=.92;}}
  for(const d of DUST){d.x+=d.vx*dt;d.y+=d.vy*dt;
    if(d.x<0)d.x+=WW;if(d.x>WW)d.x-=WW;if(d.y<0)d.y+=WH;if(d.y>WH)d.y-=WH;}
  updateGuards(dt);
  const p=S.player;
  if(S.mode==='play'){S.runT+=dt;
    if(p.walking&&p.target){const dx=p.target.x-p.x,dy=p.target.y-p.y,d=Math.hypot(dx,dy),step=185*dt;
      if(d<=step){p.x=p.target.x;p.y=p.target.y;p.walking=false;p.target=null;}
      else{p.x+=dx/d*step;p.y+=dy/d*step;p.traveled+=step;}}
    if(AC&&!muted){const mm=S.maxM;
      if(mm>.42&&S.t>nextBeat){thump();nextBeat=S.t+lerp(1.0,.45,(mm-.42)/.58);}
      else if(mm<=.42)nextBeat=Math.max(nextBeat,S.t+.15);}
  }else if(S.mode==='cracking'){S.crackT+=dt;if(S.crackT>1.6)finishWin();}
  else if(S.mode==='dead'){S.deadT+=dt;
    if(S.deadT>.9&&!S.endShown){S.endShown=true;const acc=S.keys?Math.round(S.good/S.keys*100):100;showEnd(false,acc);setMode('win');}}
  $('danger').style.opacity=S.mode==='play'?clamp((S.maxM-.3)*1.4,0,.8):0;
  hud();
}
