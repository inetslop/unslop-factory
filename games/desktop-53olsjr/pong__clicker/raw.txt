function foldY(y,top,bot){const span=bot-top;if(span<=0)return (top+bot)/2;let d=(y-top)%(2*span);if(d<0)d+=2*span;return d<=span?top+d:top+(2*span-d);}
function predictY(faceX,errAmp,seed){

 let t=(faceX-ball.x)/ball.vx;if(t<0||t>3.5)return ball.y;
 const top=court.y+ball.r,bot=court.y+court.h-ball.r;
 return foldY(ball.y+ball.vy*t,top,bot)+Math.sin(nowSec*2.6+seed)*errAmp*(court.h/700);
}

function paddleCollide(p,dir,demo){
 const face=p.cx+(dir===1?p.w/2:-p.w/2);
 const toward=dir===1?ball.vx<0:ball.vx>0;
 if(!toward)return false;
 if(dir===1){if(ball.x-ball.r>face||ball.x+ball.r<p.cx-p.w/2)return false;}
 else{if(ball.x+ball.r<face||ball.x-ball.r>p.cx+p.w/2)return false;}
 if(Math.abs(ball.y-p.y)>p.h/2+ball.r*0.95)return false;
 const off=clamp((ball.y-p.y)/(p.h/2+ball.r),-1,1);
 const ang=off*1.05+rand(-0.04,0.04);
 ball.sp=Math.min(2.4,ball.sp*1.035);
 const sp=baseSpeed()*ball.sp;
 ball.vx=dir*Math.cos(ang)*sp;ball.vy=Math.sin(ang)*sp;
 ball.x=dir===1?face+ball.r+0.5:face-ball.r-0.5;
 p.squash=1;ball.rally++;
 S.longestRally=Math.max(S.longestRally,ball.rally);

 if(!demo){sfx.hit(ball.rally);
  if(dir===1)earn(hitVal()*comboMult()*(ball.golden?5:1),ball.x+8,ball.y,false,demo);}
 return true;
}
function step(dt,demo){
 /* paddles */
 const h=court.h,ph=padL.h;
 if(demo){
  const fxL=padL.cx+padL.w/2,fxR=padR.cx-padR.w/2;
  let tL=h/2,tR=h/2,sp=h*0.7;
  if(ball.vx<0&&ball.freeze<=0)tL=predictY(fxL,12,3);
  if(ball.vx>0&&ball.freeze<=0)tR=predictY(fxR,12,9);
  moveToward(padL,tL,sp,dt);moveToward(padR,tR,sp,dt);
 }else{
  // player (left)
  const servo=U('servo').lvl;
  padL.auto=servo>0&&(nowSec-lastManual>2.5);
  if(padL.auto){
   const fx=padL.cx+padL.w/2;
   const ty=ball.vx<0&&ball.freeze<=0?predictY(fx,Math.max(4,30-5*servo),1.7):h/2;
   moveToward(padL,court.y+court.h/2+(ty-h/2),h*(0.5+0.09*servo),dt);
  }else if(keyUp||KeyDown){
   const d=(keyUp?-1:0)+(KeyDown?1:0);
   padL.y+=d*h*1.7*dt;lastManual=nowSec;
  }else if(pointerY!==null){
   let ty=clamp(pointerY,court.y+ph/2,court.y+h-ph/2);
   const s=reflexS()*(surgeUntil>nowSec?2.6:1);
   padL.y+=(ty-padL.y)*(1-Math.exp(-s*dt));
  }
  // AI (right)
  const fxR=padR.cx-padR.w/2;
  let ty=h/2,sp=h*[0.48,0.62,0.78][S.diff]*(1+Math.min(0.9,S.rank*0.05));
  if(ball.vx>0&&ball.freeze<=0)ty=predictY(fxR,Math.max(3,[26,14,7][S.diff]-S.rank*0.4),5);
  moveToward(padR,court.y+court.h/2+(ty-h/2),sp*(ball.vx>0?1:0.45),dt);
 }
 padL.y=clamp(padL.y,court.y+ph/2,court.y+h-ph/2);
 padR.y=clamp(padR.y,court.y+ph/2,court.y+h-ph/2);
 padL.squash=Math.max(0,padL.squash-dt*5);padR.squash=Math.max(0,padR.squash-dt*5);

 /* ball */
 if(ball.freeze>0){ball.freeze-=dt;}
 else{
  const speed=Math.hypot(ball.vx,ball.vy);
  const steps=Math.max(1,Math.ceil(speed*dt/(ball.r*0.7)));
  const sdt=dt/steps;
  for(let i=0;i<steps;i++){
   ball.x+=ball.vx*sdt;ball.y+=ball.vy*sdt;
   if(ball.y-ball.r<=court.y){ball.y=court.y+ball.r;ball.vy=Math.abs(ball.vy);
    if(nowSec-lastWallSfx>0.05){lastWallSfx=nowSec;if(!demo)sfx.wall();}
    burst(ball.x,court.y+2,'#3fe0c8',4,120,0,.3);}
   else if(ball.y+ball.r>=court.y+court.h){ball.y=court.y+court.h-ball.r;ball.vy=-Math.abs(ball.vy);
    if(nowSec-lastWallSfx>0.05){lastWallSfx=nowSec;if(!demo)sfx.wall();}
    burst(ball.x,court.y+court.h-2,'#3fe0c8',4,120,0,.3);}
   paddleCollide(padL,1,demo);paddleCollide(padR,-1,demo);
  }
  ball.trail.push({x:ball.x,y:ball.y});if(ball.trail.length>15)ball.trail.shift();
  if(ball.golden&&Math.random()<0.2)burst(ball.x,ball.y,'#ffe9a0',1,60,0,.4);

  if(ball.x<court.x-28){ // you missed
   S.scoreAI++;const r=ball.rally;ball.rally=0;
   burst(court.x+6,clamp(ball.y,court.y,court.y+court.h),'#ff5c7a',16,260,0,.6);
   if(!demo){sfx.concede();shake(4);spawnFloat('MISSED',court.x+40,clamp(ball.y,court.y+20,court.y+court.h-20),false);}
   serve(-1);
  }else if(ball.x>court.x+court.w+28){ // you scored
   const mult=comboMult(),gm=ball.golden?5:1;
   S.scoreYou++;S.goals++;
   burst(court.x+court.w-6,clamp(ball.y,court.y,court.y+court.h),'#ffc94d',26,340,0,.8);
   if(!demo){sfx.score();shake(7);earn(goalVal()*mult*gm,court.x+court.w-50,clamp(ball.y,court.y+30,court.y+court.h-30),true);}
   serve(1);
  }
 }

 /* fx decay */
 for(let i=particles.length-1;i>=0;i--){const p=particles[i];
  p.x+=p.vx*dt;p.y+=p.vy*dt;p.vy+=(p.g||0)*dt;p.life-=dt;if(p.life<=0)particles.splice(i,1);}
 for(let i=floats.length-1;i>=0;i--){const f=floats[i];f.y-=42*dt;f.life-=dt;if(f.life<=0)floats.splice(i,1);}
 shakeAmp*=Math.exp(-6*dt);if(shakeAmp<0.15)shakeAmp=0;
 for(const d of dust){d.y-=d.s*dt;d.x+=Math.sin(nowSec+d.ph)*8*dt;if(d.y<-4){d.y=H+4;d.x=rand(0,W);}}
 if(!demo)S.playTime+=dt;
}

 case 'hit':return `${hitVal()} → ${hitVal()+2} pts / hit`;

 case 'gold':return `${u.lvl*6}% → ${(u.lvl+1)*6}% golden chance`;}
 return '';}
