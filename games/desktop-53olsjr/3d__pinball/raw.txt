  o.connect(g); g.connect(master); o.start(t); o.stop(t+dur+.03); }

  s.connect(f); f.connect(g); g.connect(master); s.start(t); s.stop(t+dur+.03); }

function stepFlipper(f,h){
  const want=(f.side==='L'?(leftHeld||aiL):(rightHeld||aiR));
  const target=want?f.active:f.rest;
  const prev=f.angle, maxd=38*h;
  f.angle+=Math.max(-maxd,Math.min(maxd,target-f.angle));
  f.omega=(f.angle-prev)/h;
  if(want&&!f.on){f.on=true; snd.flip();} else if(!want)f.on=false;
}
function hitSeg(o,ax,ay,bx,by,r,e,out){ // circle vs segment; mutates ball via o ref
  const ex=bx-ax,ey=by-ay,l2=ex*ex+ey*ey;
  let t=((o.x-ax)*ex+(o.y-ay)*ey)/l2; t=t<0?0:(t>1?1:t);
  const qx=ax+t*ex,qy=ay+t*ey, dx=o.x-qx,dy=o.y-qy,d2=dx*dx+dy*dy;
  if(d2>r*r)return null;
  const d=Math.sqrt(Math.max(d2,1e-9)),nx=dx/d,ny=dy/d;
  o.x=qx+nx*r; o.y=qy+ny*r;
  const vn=o.vx*nx+o.vy*ny;
  if(vn<0){ o.vx-=(1+e)*vn*nx; o.vy-=(1+e)*vn*ny; }
  return {nx:nx,ny:ny,impact:-vn};
}
function physics(h){
  // plunger charge
  if(charging)charge=Math.min(1,charge+h/0.85);
  dispCharge+=( (charging?charge:0)-dispCharge )*Math.min(1,h*14);

  stepFlipper(flippers[0],h); stepFlipper(flippers[1],h);

  if(state==='ready'){ ball.x=44.5; ball.y=20+Math.sin(tAnim*2)*0.15; return; }
  if(state!=='playing'&&state!=='lost')return;

  const falling=(state==='lost');
  ball.vy-=GRAV*h; const m=Math.exp(-DRAG*h); ball.vx*=m; ball.vy*=m;
  const sp=Math.hypot(ball.vx,ball.vy); if(sp>1500){ball.vx*=1500/sp;ball.vy*=1500/sp;}
  ball.x+=ball.vx*h; ball.y+=ball.vy*h;

  if(falling){ return; } // let it drop out

  // walls
  for(const w of walls){ const r=hitSeg(ball,w.ax,w.ay,w.bx,w.by,BALL_R,.45);
    if(r&&r.impact>130&&wallSndCd<=0){snd.wall(r.impact); wallSndCd=.06;} }
  // bumpers
  for(let i=0;i<bumpers.length;i++){ const b=bumpers[i]; b.cd-=h;
    const dx=ball.x-b.x,dy=ball.y-b.y,d=Math.hypot(dx,dy);
    if(d<b.r+BALL_R){ const nx=dx/d,ny=dy/d; ball.x=b.x+nx*(b.r+BALL_R); ball.y=b.y+ny*(b.r+BALL_R);
      const vn=ball.vx*nx+ball.vy*ny; if(vn<0){ball.vx-=1.3*vn*nx; ball.vy-=1.3*vn*ny;}
      if(b.cd<=0){ b.cd=.09; b.flash=1; ball.vx+=380*nx; ball.vy+=380*ny; shake=Math.max(shake,3);
        snd.bump(i); if(!attract){const s=100*mult; addScore(s); popupAt(b.x,b.y,'+'+s,C[b.col]);} } } }
  // slings
  for(const s of slings){ s.cd-=h; const r=hitSeg(ball,s.ax,s.ay,s.bx,s.by,BALL_R,.5);
    if(r&&s.cd<=0){ s.cd=.1; s.flash=1; ball.vx+=320*r.nx; ball.vy+=320*r.ny; shake=Math.max(shake,2);
      snd.sling(); if(!attract){const v=50*mult; addScore(v); popupAt((s.ax+s.bx)/2,(s.ay+s.by)/2,'+'+v,C.mag);} } }
  // targets
  for(const t of targets){ t.cd-=h; if(t.lit)continue; const r=hitSeg(ball,t.ax,t.ay,t.bx,t.by,BALL_R,.4);
    if(r&&t.cd<=0){ t.cd=.15; t.lit=true; snd.target();
      if(!attract){ const v=500*mult; addScore(v); popupAt((t.ax+t.bx)/2,(t.ay+t.by)/2,'+'+v,C.amb); }
      if(targets.every(tt=>tt.lit)){ bankRelight=1.7; snd.bank(); shake=Math.max(shake,4);
        if(!attract){ addScore(2500); mult=Math.min(8,mult*2); showToast('BANK BONUS +2500',1300); } } } }
  // flippers
  for(const f of flippers){ const tipx=f.px+f.len*Math.cos(f.angle),tipy=f.py+f.len*Math.sin(f.angle);
    const ex=tipx-f.px,ey=tipy-f.py,l2=ex*ex+ey*ey;
    let t=((ball.x-f.px)*ex+(ball.y-f.py)*ey)/l2; t=t<0?0:(t>1?1:t);
    const qx=f.px+t*ex,qy=f.py+t*ey, dx=ball.x-qx,dy=ball.y-qy,d=Math.hypot(dx,dy),rr=BALL_R+f.r;
    if(d<rr){ let nx,ny; if(d>1e-4){nx=dx/d;ny=dy/d;}else{nx=-ey/Math.sqrt(l2);ny=ex/Math.sqrt(l2);}
      ball.x=qx+nx*rr; ball.y=qy+ny*rr;
      const vsx=-f.omega*(qy-f.py),vsy=f.omega*(qx-f.px);
      const rvx=ball.vx-vsx,rvy=ball.vy-vsy,vn=rvx*nx+rvy*ny;
      if(vn<0){ ball.vx-=1.38*vn*nx; ball.vy-=1.38*vn*ny; } } }

  // drain + stuck guard
  if(ball.y<-7){ loseBall(); return; }
  const s2=Math.hypot(ball.vx,ball.vy);
  if(s2<12)stillT+=h; else stillT=0;
  if(stillT>3.5){ stillT=0; ball.vx+=(Math.random()*2-1)*60+ (ball.x>8?-60:ball.x<-8?60:0); ball.vy+=90; }
}

function drawWall(w,accent){
  const A=proj(w.ax,w.ay,0),B=proj(w.bx,w.by,0),At=proj(w.ax,w.ay,w.h),Bt=proj(w.bx,w.by,w.h);
  let nx=w.by-w.ay,ny=w.ax-w.bx; const nl=Math.hypot(nx,ny)||1; nx/=nl; ny/=nl;
  const mx=(w.ax+w.bx)/2-CAM.x? (w.ax+w.bx)/2:0, my=(w.ay+w.by)/2;
  if(nx*(CAM.x-mx)+ny*(CAM.y-my)<0){nx=-nx;ny=-ny;}
  const lam=Math.abs((nx*LX+ny*LY)/LN), fog=Math.max(0,Math.min(1,(A.zc+B.zc)/2-150))/280;
  ctx.beginPath(); ctx.moveTo(A.x,A.y); ctx.lineTo(B.x,B.y); ctx.lineTo(Bt.x,Bt.y); ctx.lineTo(At.x,At.y); ctx.closePath();
  ctx.fillStyle=shade('gun',(.35+.65*lam)*(1-.32*fog)); ctx.fill();
  // neon top strip
  const k=(At.k+Bt.k)/2;
  ctx.save(); ctx.globalCompositeOperation='lighter';
  ctx.strokeStyle=accent; ctx.lineWidth=6*k; ctx.globalAlpha=.22; line(At,Bt); ctx.stroke();
  ctx.lineWidth=2.1*k; ctx.globalAlpha=.95; line(At,Bt); ctx.stroke(); ctx.restore();
}

function drawSolid(pts,h,topFn,baseCol,accent){
  const n=pts.length; let A=0;
  for(let i=0;i<n;i++){const p=pts[i],q=pts[(i+1)%n]; A+=p.x*q.y-q.x*p.y;} A*=.5;
  const s=A>0?1:-1;
  const B=[],T=[]; for(let i=0;i<n;i++){B.push(proj(pts[i].x,pts[i].y,0)); T.push(proj(pts[i].x,pts[i].y,h));}
  const far=[],near=[];
  for(let i=0;i<n;i++){ const j=(i+1)%n; let nx=s*(pts[j].y-pts[i].y), ny=-s*(pts[j].x-pts[i].x);
    const nl=Math.hypot(nx,ny)||1; nx/=nl; ny/=nl;
    const mx=(pts[i].x+pts[j].x)/2,my=(pts[i].y+pts[j].y)/2;
    const facing=nx*(CAM.x-mx)+ny*(CAM.y-my)>0;
    const it={i:j,nx:nx,ny:ny,depth:(B[i].zc+B[j].zc)/2};
    (facing?near:far).push(it); }
  const quad=it=>{ctx.beginPath();ctx.moveTo(B[it.i].x,B[it.i].y);ctx.lineTo(B[(it.i+1)%n].x,B[(it.i+1)%n].y);
    ctx.lineTo(T[(it.i+1)%n].x,T[(it.i+1)%n].y);ctx.lineTo(T[it.i].x,T[it.i].y);ctx.closePath();
    const lam=Math.abs((it.nx*LX+it.ny*LY)/LN),fog=Math.max(0,Math.min(1,it.depth-150))/280;
    ctx.fillStyle=shade(baseCol||'gun',(.3+.62*lam)*(1-.3*fog)); ctx.fill(); };
  far.sort((a,b)=>b.depth-a.depth).forEach(quad);
  path(T); if(topFn)topFn(T); else {ctx.fillStyle=shade('gun',.95);ctx.fill();}
  near.sort((a,b)=>b.depth-a.depth).forEach(quad);
  if(accent){ const k=T[0].k; ctx.save(); ctx.globalCompositeOperation='lighter';
    ctx.strokeStyle=accent; ctx.lineWidth=1.6*k; ctx.globalAlpha=.5; path(T); ctx.stroke(); ctx.restore(); }
}
function capsulePts(ax,ay,bx,by,r){ const dir=Math.atan2(by-ay,bx-ax),pts=[]; const N1=7,N2=8;
  for(let i=0;i<=N1;i++){const th=dir+Math.PI/2+Math.PI*i/N1; pts.push({x:ax+r*Math.cos(th),y:ay+r*Math.sin(th)});}
  for(let i=1;i<=N2;i++){const u=i/(N2+1); pts.push({x:bx-r*Math.sin(dir)+u*(0), y:0}); } // placeholder, rebuilt below
  pts.length=0;
  for(let i=0;i<=N1;i++){const th=dir+Math.PI/2+Math.PI*i/N1; pts.push({x:ax+r*Math.cos(th),y:ay+r*Math.sin(th)});}
  for(let i=1;i<N2;i++){const u=i/N2, px=Math.cos(dir),py=Math.sin(dir), qx=-py,qy=px;
    pts.push({x:(ax+u*(bx-ax)) - r*qx? (ax+u*(bx-ax))-r*(-Math.sin(dir))*0 : 0,y:0}); } // keep it simple: straight edges
  pts.length=0;
  const px=Math.cos(dir),py=Math.sin(dir), qx=-py,qy=px;
  for(let i=0;i<=N1;i++){const th=dir+Math.PI/2+Math.PI*i/N1; pts.push({x:ax+r*Math.cos(th),y:ay+r*Math.sin(th)});}
  pts.push({x:bx-r*qx,y:by-r*qy});
  for(let i=0;i<=N1;i++){const th=dir-Math.PI/2+Math.PI*i/N1; pts.push({x:bx+r*Math.cos(th),y:by+r*Math.sin(th)});}
  pts.push({x:ax+r*qx,y:ay+r*qy});

function render(dt){
  ctx.setTransform(DPR,0,0,DPR,0,0);
  drawBackground();
  ctx.save();
  if(shake>.05){ ctx.translate((Math.random()*2-1)*shake*S,(Math.random()*2-1)*shake*S); }
  else shake=0;
  drawArt();

  const items=[];
  for(const w of walls)items.push({d:(proj(w.ax,w.ay,0).zc+proj(w.bx,w.by,0).zc)/2, f:()=>drawWall(w,C.cyan)});
  slings.forEach((s,i)=>items.push({d:proj((s.ax+s.bx)/2,(s.ay+s.by)/2,0).zc, f:()=>drawSling(s)}));
  targets.forEach(t=>items.push({d:proj((t.ax+t.bx)/2,(t.ay+t.by)/2,0).zc, f:()=>drawTarget(t)}));
  bumpers.forEach((b,i)=>items.push({d:proj(b.x,b.y,0).zc, f:()=>drawBumper(b,i)}));
  flippers.forEach(f=>items.push({d:proj(f.px,f.py,0).zc, f:()=>drawFlipper(f)}));
  if(state!=='over')items.push({d:proj(44.5,12,0).zc, f:drawPlunger});
  items.push({d:proj(ball.x,ball.y,0).zc, f:drawBall});
  items.sort((a,b)=>b.d-a.d); for(const it of items)it.f();

  // popups (screen space, inside shake)
  for(let i=popups.length-1;i>=0;i--){ const p=popups[i]; p.t+=dt;
    if(p.t>p.life){popups.splice(i,1);continue;}
    const a=p.t<p.life*.6?1:1-(p.t-p.life*.6)/(p.life*.4);
    ctx.font='700 '+Math.round(15*S)+'px ui-monospace,Menlo,Consolas,monospace';

    ctx.strokeStyle='rgba(0,0,10,.8)'; ctx.lineWidth=4; ctx.strokeText(p.txt,p.x,p.y-p.t*52*S);
    ctx.fillStyle=p.col; ctx.fillText(p.txt,p.x,p.y-p.t*52*S); ctx.globalAlpha=1; }
  ctx.restore();
  ctx.fillStyle=vig; ctx.fillRect(0,0,W,H);
}
