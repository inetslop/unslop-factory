function physStep(b,dt){

  let sp=Math.hypot(b.vx,b.vy);
  if(sp>0 && SURF[surf]){
    const ns=Math.max(0, sp*Math.exp(-surf.k*dt)-surf.L*dt);
    b.vx*=ns/sp; b.vy*=ns/sp; sp=ns;
  }
  if(surf!=='sand' && surf!=='water'){
    const wf=surf==='green'?0.25:1;
    b.vx+=wind.x*wf*dt; b.vy+=wind.y*wf*dt;
  }
  const dx=h.hole[0]-b.x, dy=h.hole[1]-b.y, d=Math.hypot(dx,dy);
  if(d<h.greenR*1.5 && sp>10 && d>0.5){
    const g=1500*(1-d/(h.greenR*1.5));
    b.vx+=dx/d*g*dt; b.vy+=dy/d*g*dt;
  }
  if(d<HOLE_R && sp>340){ const f=Math.exp(-2.5*dt); b.vx*=f; b.vy*=f; }
  b.x+=b.vx*dt; b.y+=b.vy*dt;
  b.spin+=Math.hypot(b.vx,b.vy)*dt/BALL_R*0.6;

  const res={};
  // rails
  if(b.x<48+BALL_R){ b.x=48+BALL_R; if(b.vx<0){ res.hit=Math.max(res.hit||0,-b.vx); b.vx=-b.vx*REST; } }
  if(b.x>W-48-BALL_R){ b.x=W-48-BALL_R; if(b.vx>0){ res.hit=Math.max(res.hit||0,b.vx); b.vx=-b.vx*REST; } }
  if(b.y<48+BALL_R){ b.y=48+BALL_R; if(b.vy<0){ res.hit=Math.max(res.hit||0,-b.vy); b.vy=-b.vy*REST; } }
  if(b.y>H-48-BALL_R){ b.y=H-48-BALL_R; if(b.vy>0){ res.hit=Math.max(res.hit||0,b.vy); b.vy=-b.vy*REST; } }
  // hedge walls
  for(const w of h.walls){
    const cx=clamp(b.x,w.x,w.x+w.w), cy=clamp(b.y,w.y,w.y+w.h);
    let ddx=b.x-cx, ddy=b.y-cy, dd=Math.hypot(ddx,ddy);
    if(dd<BALL_R){
      if(dd<0.001){ ddx=0; ddy=-1; dd=1; }
      const nx=ddx/dd, ny=ddy/dd;
      b.x=cx+nx*BALL_R; b.y=cy+ny*BALL_R;
      const vn=b.vx*nx+b.vy*ny;
      if(vn<0){ res.hit=Math.max(res.hit||0,-vn); b.vx-=(1+REST)*vn*nx; b.vy-=(1+REST)*vn*ny; }
    }
  }
  // trees
  for(const t of h.trees){

    let dd=Math.hypot(ddx,ddy);
    if(dd<R2 && dd>0.001){
      const nx=ddx/dd, ny=ddy/dd;
      b.x=t.x+nx*R2; b.y=t.y+ny*R2;
      const vn=b.vx*nx+b.vy*ny;
      if(vn<0){ res.hit=Math.max(res.hit||0,-vn); b.vx-=(1+REST)*vn*nx; b.vy-=(1+REST)*vn*ny; }
    }
  }
  sp=Math.hypot(b.vx,b.vy);
  const d2=Math.hypot(h.hole[0]-b.x,h.hole[1]-b.y);
  if((d2<HOLE_R-3 && sp<330)||d2<6) res.sink=true;
  else if(surfaceAt(h,b.x,b.y)==='water') res.water=true;
  return res;
}

function simulate(ang,power){
  const h=curHole();
  const sp=MIN_SP+(MAX_SP-MIN_SP)*power;
  const b={x:ball.x,y:ball.y,vx:Math.cos(ang)*sp,vy:Math.sin(ang)*sp,spin:0};
  const pts=[]; let end='stop';
  for(let i=0;i<360;i++){
    const res=physStep(b,1/60);
    if(i%3===0) pts.push({x:b.x,y:b.y});
    if(res.sink){ end='sink'; break; }
    if(res.water){ end='water'; pts.push({x:b.x,y:b.y}); break; }
    if(res.hit && res.hit>40){ end='wall'; break; }
    if(Math.hypot(b.vx,b.vy)<8) break;
  }
  return {pts,end};
}

function rr(g,x,y,w,h,r){
  g.beginPath();
  g.moveTo(x+r,y); g.arcTo(x+w,y,x+w,y+h,r); g.arcTo(x+w,y+h,x,y+h,r);
  g.arcTo(x,y+h,x,y,r); g.arcTo(x,y,x+w,y,r); g.closePath();
}
function waterPath(g,w){
  if(w.shape==='circle'){ g.beginPath(); g.arc(w.x,w.y,w.r,0,Math.PI*2); }
  else rr(g,w.x,w.y,w.w,w.h,14);
}

function makeBase(hi){
  const h=HOLES[hi];
  const c=document.createElement('canvas'); c.width=W; c.height=H;
  const g=c.getContext('2d');
  const rand=mulberry32(hi*991+7);

  // rough + mow stripes
  rr(g,30,30,W-60,H-60,48); g.fillStyle='#2e7d4f'; g.fill();

  g.transform(1,0,-0.45,1,500,0);
  g.fillStyle='rgba(255,255,255,0.05)';
  for(let x=-900;x<W+900;x+=84){ if(((x/84)|0)%2===0) g.fillRect(x,-100,84,H+200); }
  g.restore();

  // fairway two-tone
  const fpath=()=>{ g.beginPath(); g.moveTo(h.fairway[0][0],h.fairway[0][1]); for(let i=1;i<h.fairway.length;i++) g.lineTo(h.fairway[i][0],h.fairway[i][1]); };
  g.lineJoin='round'; g.lineCap='round';

  g.strokeStyle='rgba(255,255,255,.07)'; g.lineWidth=3;

  // hole cup
  g.beginPath(); g.arc(hx,hy,HOLE_R,0,Math.PI*2); g.fillStyle='#0d231a'; g.fill();

/* ============================== UPDATE ============================== */
function update(dt){
  T+=dt;
  shake-=shake*8*dt; if(shake<0.2) shake=0;

  if(mode==='play' && !paused){
    // keyboard aim
    if(ball.state==='ready'){
      if(keys.left||keys.right){
        kbActive=true;
        kbAngle+=((keys.right?1:0)-(keys.left?1:0))*(keys.shift?1.1:2.8)*dt;
      }
      if(charging) chargeT+=dt*1.5;
    }
    // physics
    if(ball.state==='rolling'){
      acc+=dt; const hs=1/120; let n=0;
      while(acc>=hs && n<6){
        acc-=hs; n++;
        const res=physStep(ball,hs);
        if(res.hit && res.hit>70){ AU.bounce(res.hit); shake=Math.max(shake,Math.min(5,res.hit/180)); }
        const ns=surfaceAt(curHole(),ball.x,ball.y);
        if(ns==='sand' && prevSurf!=='sand' && Math.hypot(ball.vx,ball.vy)>140){ AU.sand(); puffBurst(ball.x,ball.y,8,'rgba(236,217,160,.8)'); }
        prevSurf=ns;
        if(res.water){ onWater(); break; }
        if(res.sink){ onSink(); break; }
        if(Math.hypot(ball.vx,ball.vy)<6){ ball.vx=0; ball.vy=0; ball.state='ready'; refreshHud(); }
      }
    } else if(ball.state==='sinking'){
      ball.sinkT+=dt;
      const t=Math.min(1,ball.sinkT/0.55), e=t*t*(3-2*t);
      const h=curHole();
      ball.x=lerp(ball.from.x,h.hole[0],e);
      ball.y=lerp(ball.from.y,h.hole[1],e);
      ball.scale=1-0.85*e;
      if(t>=1) finishSink();
    } else if(ball.state==='splash'){
      ball.splashT+=dt;
      if(ball.splashT>0.95){
        ball.x=lastSafe.x; ball.y=lastSafe.y; ball.vx=ball.vy=0;
        ball.hidden=false; ball.scale=1; ball.state='ready';
        puffBurst(ball.x,ball.y,10,'rgba(246,241,222,.7)');
        prevSurf=surfaceAt(curHole(),ball.x,ball.y);
      }
    } else if(ball.state==='done' && T>=resultAt){
      showResult();
    }
  }

  if(flagDropping && flagDrop<1) flagDrop=Math.min(1,flagDrop+dt*2.6);

  // trail
  const spNow=ball.state==='rolling'?Math.hypot(ball.vx,ball.vy):0;
  if(spNow>140){ trail.push({x:ball.x,y:ball.y,l:1}); if(trail.length>26) trail.shift(); }
  for(const t of trail) t.l-=dt*2.4;
  trail=trail.filter(t=>t.l>0);

  // particles
  for(const p of parts){
    p.l-=dt/p.max; p.x+=p.vx*dt; p.y+=p.vy*dt;
    if(p.type==='ring') p.r+=90*dt;
    else if(p.type==='puff'){ p.r+=18*dt; p.vx*=0.92; p.vy*=0.92; }
    else { p.vx*=0.94; p.vy*=0.94; }
  }
  parts=parts.filter(p=>p.l>0);

  // confetti (screen space)
  if(mode==='scorecard'){
    for(const c of confetti){
      c.x+=c.vx*dt; c.y+=c.vy*dt; c.a+=c.va*dt; c.vy+=60*dt;
      if(c.y>innerHeight+30){ c.y=-20; c.x=rnd()*innerWidth; }
    }
  }

  updateRollAudio(dt);
}

function updateRollAudio(dt){
  if(!AU.ac) return;
  let tgt=0, fq=400;
  if(mode==='play' && !paused && ball.state==='rolling'){
    const sp=Math.hypot(ball.vx,ball.vy);
    const s=surfaceAt(curHole(),ball.x,ball.y);
    tgt=Math.min(0.15,sp/950*0.15)*(s==='sand'?0.5:1);
    fq=260+sp*0.45;
  }
  const g=AU.rollGain.gain;
  g.value+=(tgt-g.value)*Math.min(1,dt*8);
  AU.rollFilt.frequency.value+=(fq-AU.rollFilt.frequency.value)*Math.min(1,dt*6);
}

/* ============================== RENDER ============================== */
function render(){
  const cw=innerWidth, chh=innerHeight;
  ctx.setTransform(dpr,0,0,dpr,0,0);
  computeView();

  // letterbox ambient background
  ctx.fillStyle='#07130d'; ctx.fillRect(0,0,cw,chh);
  ctx.save(); ctx.globalAlpha=0.05; ctx.strokeStyle='#f2ead2'; ctx.lineWidth=2;
  const off=(TA*14)%48;
  for(let x=-chh;x<cw+chh;x+=48){ ctx.beginPath(); ctx.moveTo(x+off,0); ctx.lineTo(x+off-chh,chh); ctx.stroke(); }
  ctx.restore();
  const vg=ctx.createRadialGradient(cw/2,chh/2,Math.min(cw,chh)*0.35,cw/2,chh/2,Math.max(cw,chh)*0.78);
  vg.addColorStop(0,'rgba(0,0,0,0)'); vg.addColorStop(1,'rgba(0,0,0,0.45)');
  ctx.fillStyle=vg; ctx.fillRect(0,0,cw,chh);

  const h=curHole();
  const bob=(mode==='menu')?Math.sin(TA*0.6)*4:0;
  let tx=ox, ty=oy+bob;
  if(shake>0){ tx+=(rnd()-.5)*shake; ty+=(rnd()-.5)*shake; }

  ctx.save();
  ctx.translate(tx,ty); ctx.scale(scale,scale);

  if(base) ctx.drawImage(base,0,0);

  // animated water
  for(const w of h.water){
    const bx=w.shape==='circle'?w.x-w.r:w.x, by=w.shape==='circle'?w.y-w.r:w.y;
    const bw=w.shape==='circle'?w.r*2:w.w, bh=w.shape==='circle'?w.r*2:w.h;
    ctx.save(); waterPath(ctx,w); ctx.clip();
    for(let i=0;i<Math.max(2,Math.floor(bh/46));i++){
      const yy=by+((TA*18+i*46)%(bh+24))-12;
      ctx.beginPath();
      for(let x=bx-12;x<=bx+bw+12;x+=14){
        const y=yy+Math.sin(x*0.06+TA*2.2+i)*3;
        if(x===bx-12) ctx.moveTo(x,y); else ctx.lineTo(x,y);
      }
      ctx.strokeStyle='rgba(255,255,255,.15)'; ctx.lineWidth=3; ctx.stroke();
    }
    for(let i=0;i<6;i++){
      const px=bx+bw*(0.15+0.7*((Math.sin(i*7.3+h.hole[0])*43758)%1));
      const py=by+bh*(0.15+0.7*((Math.sin(i*3.1+h.hole[1])*12543)%1));
      ctx.globalAlpha=0.08+0.08*Math.sin(TA*3+i*2);
      ctx.beginPath(); ctx.arc(px,py,2.2,0,Math.PI*2); ctx.fillStyle='#fff'; ctx.fill();
    }
    ctx.globalAlpha=1; ctx.restore();
    waterPath(ctx,w); ctx.strokeStyle='#16498c'; ctx.lineWidth=6; ctx.stroke();
  }

  // trees (sway)
  h.trees.forEach((t,i)=>{
    const sw=Math.sin(TA*1.3+i*2.1)*2;
    ctx.beginPath(); ctx.ellipse(t.x+7,t.y+9,34,30,0,0,Math.PI*2); ctx.fillStyle='rgba(0,0,0,.18)'; ctx.fill();
    const gr=ctx.createRadialGradient(t.x+sw-10,t.y-10,4,t.x+sw,t.y,36);
    gr.addColorStop(0,'#3aa85e'); gr.addColorStop(1,'#1a5c33');
    ctx.beginPath(); ctx.arc(t.x+sw,t.y,34,0,Math.PI*2); ctx.fillStyle=gr; ctx.fill();
    ctx.beginPath(); ctx.arc(t.x+sw-8,t.y-9,10,0,Math.PI*2); ctx.fillStyle='rgba(255,255,255,.1)'; ctx.fill();
  });

  // aim preview
  if(mode==='play' && !paused && ball.state==='ready'){
    const aiming=dragging||charging||kbActive;
    if(aiming){
      const ang=dragging?aimAngle:kbAngle;
      const pw=dragging?aimPower:(charging?chargePower():0);
      const sim=simulate(ang,Math.max(pw,0.12));
      const n=sim.pts.length;
      for(let i=0;i<n;i++){
        const p=sim.pts[i], f=1-i/n;
        ctx.globalAlpha=0.15+0.5*f;
        ctx.beginPath(); ctx.arc(p.x,p.y,1.5+3*f,0,Math.PI*2);
        ctx.fillStyle=(i===n-1 && sim.end==='water')?'#ff6a5c':(sim.end==='sink'&&i===n-1?'#ffd24a':'#f6f1de');
        ctx.fill();
      }
      ctx.globalAlpha=1;
      if(sim.end==='water'){
        const p=sim.pts[n-1];
        ctx.beginPath(); ctx.arc(p.x,p.y,10+Math.sin(TA*6)*2,0,Math.PI*2); ctx.strokeStyle='#ff6a5c'; ctx.lineWidth=3; ctx.stroke();
      }
      // arrow
      const L=26+pw*46, ax=Math.cos(ang), ay=Math.sin(ang);
      ctx.strokeStyle='rgba(246,241,222,.9)'; ctx.lineWidth=4; ctx.lineCap='round';
      ctx.beginPath(); ctx.moveTo(ball.x+ax*(BALL_R+4),ball.y+ay*(BALL_R+4)); ctx.lineTo(ball.x+ax*L,ball.y+ay*L); ctx.stroke();
      const hx2=ball.x+ax*(L+8), hy2=ball.y+ay*(L+8);
      ctx.beginPath(); ctx.moveTo(hx2,hy2);
      ctx.lineTo(hx2-ax*14-ay*7,hy2-ay*14+ax*7);
      ctx.lineTo(hx2-ax*14+ay*7,hy2-ay*14-ax*7);
      ctx.closePath(); ctx.fillStyle='rgba(246,241,222,.9)'; ctx.fill();
      // power ring
      if(pw>0.01){
        ctx.beginPath(); ctx.arc(ball.x,ball.y,20,0,Math.PI*2); ctx.strokeStyle='rgba(0,0,0,.3)'; ctx.lineWidth=5; ctx.stroke();
        const col=pw<0.4?'#7fe08f':(pw<0.75?'#ffd24a':'#ff6a5c');
        ctx.beginPath(); ctx.arc(ball.x,ball.y,20,-Math.PI/2,-Math.PI/2+pw*Math.PI*2); ctx.strokeStyle=col; ctx.lineWidth=5; ctx.stroke();
        ctx.font='700 18px Impact, "Arial Black", sans-serif';
        ctx.fillStyle='#f6f1de'; ctx.textAlign='center';
        ctx.fillText(Math.round(pw*100)+'%', ball.x, ball.y-30);
      }
    } else {
      // idle pulse invite
      ctx.beginPath(); ctx.arc(ball.x,ball.y,BALL_R+8+Math.sin(TA*4)*3,0,Math.PI*2);
      ctx.strokeStyle='rgba(246,241,222,.5)'; ctx.lineWidth=2.5; ctx.setLineDash([5,6]); ctx.stroke(); ctx.setLineDash([]);
    }
  }

  // trail
  for(const t of trail){
    ctx.globalAlpha=t.l*0.3;
    ctx.beginPath(); ctx.arc(t.x,t.y,1+4*t.l,0,Math.PI*2); ctx.fillStyle='#fff'; ctx.fill();
  }
  ctx.globalAlpha=1;

  // ball
  if(!ball.hidden){
    const r=BALL_R*ball.scale;
    ctx.beginPath(); ctx.ellipse(ball.x+4,ball.y+5,r*0.95,r*0.62,0,0,Math.PI*2); ctx.fillStyle='rgba(0,0,0,.25)'; ctx.fill();
    const bg=ctx.createRadialGradient(ball.x-r*.35,ball.y-r*.4,1,ball.x,ball.y,r);
    bg.addColorStop(0,'#ffffff'); bg.addColorStop(1,'#d5ded8');
    ctx.beginPath(); ctx.arc(ball.x,ball.y,r,0,Math.PI*2); ctx.fillStyle=bg; ctx.fill();
    ctx.strokeStyle='rgba(0,0,0,.15)'; ctx.lineWidth=1; ctx.stroke();
    if(r>3){
      const sx=ball.x+Math.cos(ball.spin)*r*.5, sy=ball.y+Math.sin(ball.spin)*r*.5;
      ctx.beginPath(); ctx.arc(sx,sy,r*.16,0,Math.PI*2); ctx.fillStyle='rgba(0,0,0,.2)'; ctx.fill();
    }
  }

  // flag (top-down cloth pointing with wind)
  {
    const [hx,hy]=h.hole;
    const wm=Math.hypot(wind.x,wind.y);
    let ux,uy;
    if(wm<1){ ux=0.8+Math.sin(TA*0.8)*0.2; uy=0.4; } else { ux=wind.x/wm; uy=wind.y/wm; }
    const ul=Math.hypot(ux,uy); ux/=ul; uy/=ul;
    const vx=-uy, vy=ux;
    const drop=flagDrop;
    const len=(26*(1-drop*0.9))+4, wid=(13*(1-drop*0.85))+2;
    const wav=Math.sin(TA*7)*3;
    ctx.beginPath(); ctx.ellipse(hx+6,hy+7,8,5,0,0,Math.PI*2); ctx.fillStyle='rgba(0,0,0,.2)'; ctx.fill();
    ctx.beginPath();
    ctx.moveTo(hx+ux*5, hy+uy*5);
    ctx.lineTo(hx+ux*len+vx*(wid/2+wav), hy+uy*len+vy*(wid/2+wav));
    ctx.lineTo(hx+ux*(len+4), hy+uy*(len+4));
    ctx.lineTo(hx+ux*len+vx*(-(wid/2-wav)), hy+uy*len+vy*(-(wid/2-wav)));
    ctx.closePath();
    ctx.fillStyle='#e8483f'; ctx.fill();
    ctx.strokeStyle='#9c1f16'; ctx.lineWidth=1.5; ctx.stroke();
    ctx.beginPath(); ctx.arc(hx,hy,3.2,0,Math.PI*2); ctx.fillStyle='#20301f'; ctx.fill();
  }

  // particles
  for(const p of parts){
    const a=Math.max(0,p.l);
    if(p.type==='ring'){
      ctx.globalAlpha=a*0.8; ctx.beginPath(); ctx.arc(p.x,p.y,p.r,0,Math.PI*2); ctx.strokeStyle=p.c; ctx.lineWidth=3*a+1; ctx.stroke();
    } else if(p.type==='puff'){
      ctx.globalAlpha=a*0.6; ctx.beginPath(); ctx.arc(p.x,p.y,p.r,0,Math.PI*2); ctx.fillStyle=p.c; ctx.fill();
    } else if(p.type==='spark'){
      ctx.globalAlpha=Math.abs(Math.sin(a*Math.PI*3)); ctx.beginPath(); ctx.arc(p.x,p.y,p.r,0,Math.PI*2); ctx.fillStyle=p.c; ctx.fill();
    } else {
      ctx.globalAlpha=a*0.9; ctx.beginPath(); ctx.arc(p.x,p.y,p.r*a+0.5,0,Math.PI*2); ctx.fillStyle=p.c; ctx.fill();
    }
  }
  ctx.globalAlpha=1;

  ctx.restore();

  // confetti (screen space)
  if(mode==='scorecard' && confetti.length){
    for(const c of confetti){
      ctx.save(); ctx.translate(c.x,c.y); ctx.rotate(c.a);
      ctx.fillStyle=c.c; ctx.fillRect(-4,-6,8,12); ctx.restore();
    }
  }
}

  const dt=Math.min((t-last)/1000||0, 1/30); last=t;
  TA+=dt;
  if(!paused) update(dt);
  render();
  requestAnimationFrame(frame);
}
