function tone(f,d,type,v,fEnd,delay){
  if(!actx||muted) return; type=type||'square'; v=v==null?.1:v; delay=delay||0;
  const t0=actx.currentTime+delay, o=actx.createOscillator(), g=actx.createGain();
  o.type=type; o.frequency.setValueAtTime(f,t0);
  if(fEnd) o.frequency.exponentialRampToValueAtTime(Math.max(30,fEnd), t0+d);
  g.gain.setValueAtTime(v,t0); g.gain.exponentialRampToValueAtTime(.001, t0+d);
  o.connect(g); g.connect(master); o.start(t0); o.stop(t0+d+.03);
}

}

/* ================= UPDATE ================= */
function update(dt){
  t+=dt;
  /* lanes always drift (attract mode too) */
  LANES.forEach(ln=>{ if(!ln.es) return; ln.es.forEach(e=>{
    e.x+=ln.sp*CELL*dt;
    if(ln.sp>0 && e.x>W+80) e.x=-e.len-90-Math.random()*160;
    else if(ln.sp<0 && e.x+e.len<-80) e.x=W+90+Math.random()*160; }); });
  rippleT-=dt;
  if(rippleT<=0){ rippleT=.45; ripples.push({x:Math.random()*W, y:(1+Math.random()*4)*CELL+CELL/2, r:3, life:1}); }
  ripples.forEach(r=>{ r.r+=26*dt; r.life-=dt*.8; }); ripples=ripples.filter(r=>r.life>0);

  if(frog) frog.jumpT=Math.max(0,frog.jumpT-dt/.13);
  particles.forEach(p=>{ p.life-=dt;
    if(p.type==='dot'||p.type==='spark'){ const f=Math.pow(p.fric,dt*60); p.x+=p.vx*dt; p.y+=p.vy*dt; p.vx*=f; p.vy*=f; }
    else if(p.type==='leaf'){ p.vy+=260*dt; p.x+=p.vx*dt; p.y+=p.vy*dt; p.rot+=p.spin*dt; }
    else if(p.type==='smoke'){ p.size+=14*dt; p.y+=(p.vy||-18)*dt; }
    else if(p.type==='ring'){ p.r+=p.vr*dt; } });
  particles=particles.filter(p=>p.life>0);
  floats.forEach(f=>f.age+=dt); for(let i=floats.length;i--;) if(floats[i].age>floats[i].dur) floats.splice(i,1);
  shake=Math.max(0,shake-dt*2.4);

  if(S.mode!=='playing') return;

  if(frog.state==='ready'||frog.state==='home'){
    S.time-=dt;
    if(S.time<=0){ S.time=0;
      if(frog.state==='home'){ const p=PADS.find(q=>q.col===frog.col); die('time',(p?p.col:6.5)*CELL, CELL*.5); }
      else die('time'); }
  }

  if(frog.state==='ready'){
    hopCd-=dt;
    let dir=null; ['up','down','left','right'].forEach(d=>{ if(held[d]) dir=d; });
    if(dir && hopCd<=0){ doHop(dir); hopCd=HOP_CD; }
    /* hazards */
    const r=frog.row;
    if(r>=1&&r<=5){
      const ln=LANES[r]; let onLog=false, croc=false;
      for(const e of ln.es){ const cx=e.x+e.len/2;
        if(ln.k==='croc'){ if(Math.abs(frog.x-cx)<e.len/2+.2*CELL) croc=true; }
        else { if(frog.x>e.x-.16*CELL && frog.x<e.x+e.len+.16*CELL){ onLog=true; frog.x+=ln.sp*CELL*dt; } } }
      if(croc) die('croc');
      else if(!onLog && frog.jumpT<.7) die('splash');
      if(frog.state==='ready'&&(frog.x<-.5*CELL||frog.x>W+.5*CELL)) die('splash');
    } else if(r>=7&&r<=10){
      const ln=LANES[r];
      for(const e of ln.es){ const cx=e.x+e.len/2;
        if(Math.abs(frog.x-cx)<e.len/2+.24*CELL){ die('crash'); break; } }
    }
  }
  else if(frog.state==='dead'){
    frog.deadT+=dt;
    if(frog.deadT>=(S.lives>0?1.0:1.35)){
      if(S.lives>0){ clearPads(); resetFrog(true); updateHUD(); }
      else doGameOver();
    }
  }
  else if(frog.state==='cleared'){
    frog.clearT-=dt;
    if(frog.clearT<=0){ clearPads(); resetFrog(true); }
  }
  updateHUD();
}

function draw(){
  ctx.save();
  if(shake>0) ctx.translate((Math.random()*2-1)*shake*8,(Math.random()*2-1)*shake*8);
  if(bg) ctx.drawImage(bg,0,0,W,H);
  /* water shimmer + ripples */
  ctx.fillStyle='#dff4ff';
  SHIM.forEach(d=>{ const x=((d.x+t*14)%(W+20))-10;
    ctx.globalAlpha=d.a; ctx.beginPath(); ctx.ellipse(x,d.y,d.w,2,0,0,7); ctx.fill(); });
  ctx.globalAlpha=1;
  ripples.forEach(r=>{ ctx.strokeStyle='rgba(190,230,250,'+(.3*r.life)+')'; ctx.lineWidth=1.5;
    ctx.beginPath(); ctx.ellipse(r.x,r.y,r.r,r.r*.55,0,0,7); ctx.stroke(); });
  /* entities */
  LANES.forEach(ln=>{ if(!ln.es) return; ln.es.forEach(e=>{
    if(ln.t==='water') (ln.k==='croc'?drawCroc:drawLog)(ctx,ln,e); else drawVehicle(ctx,ln,e); }); });
  drawPads(ctx);
  drawFrog(ctx);
  /* particles */
  particles.forEach(p=>{ const a=Math.max(0,p.life/p.max);
    if(p.type==='dot'){ ctx.globalAlpha=a; ctx.fillStyle=p.c; ctx.beginPath(); ctx.arc(p.x,p.y,p.size,0,7); ctx.fill(); }
    else if(p.type==='spark'){ ctx.globalAlpha=a; ctx.strokeStyle=p.c; ctx.lineWidth=p.size; ctx.lineCap='round';
      ctx.beginPath(); ctx.moveTo(p.x,p.y); ctx.lineTo(p.x-p.vx*.03,p.y-p.vy*.03); ctx.stroke(); }
    else if(p.type==='ring'){ ctx.globalAlpha=a; ctx.strokeStyle=p.c; ctx.lineWidth=2.5;
      ctx.beginPath(); ctx.ellipse(p.x,p.y,p.r,p.r*.6,0,0,7); ctx.stroke(); }
    else if(p.type==='smoke'){ ctx.globalAlpha=a*.4; ctx.fillStyle='#9aa0a0';
      ctx.beginPath(); ctx.arc(p.x,p.y,p.size,0,7); ctx.fill(); }
    else if(p.type==='leaf'){ ctx.globalAlpha=Math.min(1,a*1.6); ctx.save(); ctx.translate(p.x,p.y); ctx.rotate(p.rot);
      ctx.fillStyle=p.c; ctx.beginPath(); ctx.ellipse(0,0,5,2.6,0,0,7); ctx.fill(); ctx.restore(); } });
  ctx.globalAlpha=1;
  /* floating text */
  ctx.font='bold 15px "Courier New",monospace'; ctx.textAlign='center';
  floats.forEach(f=>{ const a=1-f.age/f.dur, y=f.y-(f.age/f.dur)*34;
    ctx.globalAlpha=Math.max(0,a);
    ctx.lineWidth=3.5; ctx.strokeStyle='rgba(0,0,0,.7)'; ctx.strokeText(f.txt,f.x,y);
    ctx.fillStyle=f.c; ctx.fillText(f.txt,f.x,y); });
  ctx.globalAlpha=1;
  /* vignette */
  ctx.fillStyle=vg; ctx.fillRect(0,0,W,H);
  ctx.restore();
}
