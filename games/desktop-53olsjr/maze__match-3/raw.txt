  for(let c=0;c<COLS;c++)for(let r=0;r<ROWS-2;r++){const v=a[r*COLS+c];if(v>=0&&a[(r+1)*COLS+c]===v&&a[(r+2)*COLS+c]===v)return false;}
  return true;
}

/* ============================== PHASE MACHINE ============================== */
const ease=t=>t*(2-t);
function stepBoard(dt,logic){
  if(shake>0)shake=Math.max(0,shake-dt*26);
  switch(phase){
    case 'swapping':{
      swap.t+=dt/.14;const e=ease(Math.min(1,swap.t));
      gems[swap.a].ox=swap.dx*e;gems[swap.a].oy=swap.dy*e;
      gems[swap.b].ox=-swap.dx*e;gems[swap.b].oy=-swap.dy*e;
      if(swap.t>=1){
        const a=swap.a,b=swap.b;[gems[a],gems[b]]=[gems[b],gems[a]];
        gems[a].ox=gems[a].oy=gems[b].ox=gems[b].oy=0;
        const groups=findMatches();
        if(groups.length){moves--;updateHUD();beginClear(groups,1);}
        else{[gems[a],gems[b]]=[gems[b],gems[a]];swap.t=1;phase='reverting';sfx.bad();}
      }break;}
    case 'reverting':{
      swap.t-=dt/.14;const e=ease(Math.max(0,swap.t));
      gems[swap.a].ox=swap.dx*e;gems[swap.a].oy=swap.dy*e;
      gems[swap.b].ox=-swap.dx*e;gems[swap.b].oy=-swap.dy*e;
      if(swap.t<=0){gems[swap.a].ox=gems[swap.a].oy=gems[swap.b].ox=gems[swap.b].oy=0;phase='idle';}
      break;}
    case 'clearing':
      clearT+=dt/.3;for(const d of dying)d.t=clearT;
      if(clearT>=1){dying=[];spawnRefills();}break;
    case 'refilling':{
      let done=true;
      for(let i=0;i<N;i++){const g=gems[i];if(!g){done=false;continue;}
        if(g.delay>0){g.delay-=dt;done=false;continue;}
        if(g.oy<0||g.vy>0){g.vy+=34*dt;g.oy+=g.vy*dt;done=false;
          if(g.oy>=0){g.oy=0;g.vy=0;landFx(i);}}}
      if(done)settle(logic);break;}
    case 'shuffling':{
      shuffleT+=dt/.55;const st2=Math.min(1,shuffleT);
      for(const g of gems)if(g)g.sc=1-Math.sin(Math.PI*st2)*.85;
      if(shuffleT>=.5&&!shuffleCommitted){for(let i=0;i<N;i++)gems[i].c=newCols[i];shuffleCommitted=true;}
      if(st2>=1){for(const g of gems)if(g)g.sc=1;phase='idle';
        if(logic&&!hasMove(gemGet))doShuffle();}
      break;}
  }
}

}

function bfs(s,tgt){
  const prev=new Int32Array(N).fill(-1),q=[s];prev[s]=s;let head=0;
  const dirs=[[0,-1,1],[1,0,2],[0,1,4],[-1,0,8]];
  while(head<q.length){const i=q[head++];if(i===tgt)break;
    const x=i%COLS,y=(i/COLS)|0;
    for(const d of dirs){if(walls[i]&d[2])continue;const nx=x+d[0],ny=y+d[1];
      if(nx<0||nx>=COLS||ny<0||ny>=ROWS)continue;const j=ny*COLS+nx;
      if(prev[j]===-1){prev[j]=i;q.push(j);}}}
  if(prev[tgt]===-1)return[];
  const path=[];let c=tgt;while(c!==s){path.push(c);c=prev[c];}return path.reverse();
}
function wispUpd(dt){
  if(wisp.path.length){
    const t=wisp.path[0],tx=(t%COLS)+.5,ty=((t/COLS)|0)+.5;
    let dx=tx-wisp.x,dy=ty-wisp.y;const d=Math.hypot(dx,dy)||1,st2=3.4*dt;
    if(d<=st2){wisp.x=tx;wisp.y=ty;wisp.path.shift();}
    else{wisp.x+=dx/d*st2;wisp.y+=dy/d*st2;}
  }else{wisp.wait-=dt;if(wisp.wait<=0){wisp.path=bfs(Math.round(wisp.x-.5)*1+Math.round((wisp.y-.5))*COLS,(Math.random()*N)|0);
    wisp.path=bfs((Math.round(wisp.y-.5))*COLS+(Math.round(wisp.x-.5)),(Math.random()*N)|0);if(!wisp.path.length)wisp.wait=.4;}}
  const lt=wisp.trail[wisp.trail.length-1];
  if(!lt||Math.hypot(lt.x-wisp.x,lt.y-wisp.y)>.18){wisp.trail.push({x:wisp.x,y:wisp.y,l:1});if(wisp.trail.length>24)wisp.trail.shift();}
  for(const t of wisp.trail)t.l-=dt*1.5;
  while(wisp.trail.length&&wisp.trail[0].l<=0)wisp.trail.shift();
}

function drawGem(x,y,rad,ci,alpha){
  const g=GEMS[ci];ctx.save();ctx.globalAlpha=alpha;
  const gl=ctx.createRadialGradient(x,y,0,x,y,rad*1.6);
  gl.addColorStop(0,'rgba('+g.rgb+',.45)');gl.addColorStop(1,'rgba('+g.rgb+',0)');
  ctx.fillStyle=gl;ctx.beginPath();ctx.arc(x,y,rad*1.6,0,7);ctx.fill();
  const bg2=ctx.createLinearGradient(x,y-rad,x,y+rad);
  bg2.addColorStop(0,g.hi);bg2.addColorStop(.55,g.c);bg2.addColorStop(1,g.lo);
  ctx.fillStyle=bg2;ctx.beginPath();ctx.arc(x,y,rad,0,7);ctx.fill();
  ctx.strokeStyle='rgba(0,0,0,.28)';ctx.lineWidth=1.5;ctx.stroke();
  const s=rad*.52;ctx.fillStyle='rgba(255,255,255,.92)';ctx.strokeStyle='rgba(255,255,255,.92)';
  if(g.s==='diamond'){polyPts(x,y,[[0,-1],[.85,0],[0,1],[-.85,0]],s);}
  else if(g.s==='tri'){polyPts(x,y,[[0,-.95],[.9,.75],[-.9,.75]],s);}
  else if(g.s==='hex'){polyPts(x,y,HEXPTS,s*.95);}
  else if(g.s==='sq'){ctx.fillRect(x-s*.68,y-s*.68,s*1.36,s*1.36);}
  else if(g.s==='ring'){ctx.lineWidth=s*.5;ctx.beginPath();ctx.arc(x,y,s*.62,0,7);ctx.stroke();}
  else{ctx.fillRect(x-s*.3,y-s,s*.6,s*2);ctx.fillRect(x-s,y-s*.3,s*2,s*.6);}
  ctx.fillStyle='rgba(255,255,255,.5)';
  ctx.beginPath();ctx.ellipse(x-rad*.35,y-rad*.42,rad*.22,rad*.13,-.6,0,7);ctx.fill();
  ctx.restore();
}
function polyPts(x,y,pts,s){ctx.beginPath();for(let i=0;i<pts.length;i++){const p=pts[i];i?ctx.lineTo(x+p[0]*s,y+p[1]*s):ctx.moveTo(x+p[0]*s,y+p[1]*s);}ctx.closePath();ctx.fill();}
function segFor(a,b){

  return[bx+x2*S+S,by+y2*S+.5,bx+x1*S,by+y1*S+.5];
}
