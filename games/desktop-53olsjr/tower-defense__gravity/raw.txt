/* ============ game flow ============ */
function startGame(){
  towers=[];enemies=[];shots=[];parts=[];floats=[];occ.clear();
  Object.assign(game,{state:'play',phase:'build',paused:false,speed:1,flux:140,lives:20,wave:0,score:0,kills:0,buildT:12,spawnT:0});
  spawnList=[];placing=null;selected=null;shake=0;coreFlash=0;gatePulse=0;
  btnSpeed.textContent='1×';btnPause.textContent='❚❚';
  document.body.classList.remove('menu');
  screen.classList.add('hidden');pauser.classList.add('hidden');
  elChipLives.classList.remove('low');hideInspector();updateShop();setBuildBanner();hudUpdate();
}
function callWave(){
  if(game.state!=='play'||game.phase!=='build')return;
  const bonus=Math.max(0,Math.floor(game.buildT)*2);
  if(bonus>0){game.flux+=bonus;addFloat(W/2,150,'+'+bonus+'◈ EARLY','#ffd23f');}
  game.wave++;spawnList=makeWave(game.wave);game.spawnT=0;game.phase='wave';
  setWaveBanner();sfxWave();hudUpdate();
}
function waveComplete(){
  const bonus=28+6*game.wave;
  game.flux+=bonus;game.score+=game.wave*100;
  addFloat(1250,270,'+'+bonus+'◈','#ffd23f');sfxDone();
  if(game.wave>=MAXW){endGame(true);return;}
  game.phase='build';game.buildT=9;setBuildBanner();hudUpdate();
}
function endGame(won){
  game.state=won?'won':'over';shots.length=0;
  const isBest=game.score>best.score;
  if(isBest){best={score:game.score,wave:game.wave};saveBest();}
  won?sfxWin():sfxLose();showEnd(won,isBest);
}
function spawnEnemy(type){
  const s=ETYPES[type],w=game.wave,hp=Math.round(s.hp*(1+0.13*(w-1)));
  const poly=[],cracks=[];
  const n=type==='boss'?9:8;
  for(let i=0;i<n;i++)poly.push({a:i/n*TAU+rand(-.25,.25),r:s.r*(0.8+Math.random()*.35)});
  for(let i=0;i<3;i++){const a=rand(TAU);cracks.push({x1:rand(-.2,.2),y1:rand(-.2,.2),x2:Math.cos(a)*.9,y2:Math.sin(a)*.9});}
  enemies.push({type,hp,maxhp:hp,sp:s.sp,r:s.r,dmgLives:s.dmg,reward:s.reward,wres:s.wres,
    dist:-rand(20,90),rot:rand(TAU),flash:0,slow:1,push:0,near:null,x:-60,y:PATH[0][1],a:0,dead:false,poly,cracks});
  gatePulse=Math.min(1,gatePulse+.5);
}
function kill(e){
  e.dead=true;game.kills++;
  const rw=Math.round(e.reward*(1+0.05*(game.wave-1)));
  game.flux+=rw;game.score+=rw*10;
  addFloat(e.x,e.y-e.r-6,'+'+rw+'◈','#ffd23f');
  chunkBurst(e.x,e.y,e.r>24?16:9,ETYPES[e.type].col);
  sparks(e.x,e.y,'#ff9d3f',5);
  parts.push({type:'ring',x:e.x,y:e.y,r0:e.r*.4,r1:e.r*1.9,life:.3,max:.3,col:'#ff9d3f'});
  sfxPop();
}
function coreHit(e){
  e.dead=true;game.lives=Math.max(0,game.lives-e.dmgLives);
  coreFlash=.5;shake=Math.max(shake,9);
  parts.push({type:'ring',x:1250,y:320,r0:14,r1:64,life:.4,max:.4,col:'#ff5468'});
  addFloat(1250,270,'-'+e.dmgLives+' CORE','#ff5468');sfxCore();
  elChipLives.classList.toggle('low',game.lives<=6);
  if(game.lives<=0)endGame(false);
}

/* ============ simulation ============ */
function solve(sx,sy,tx,ty,arc){
  const dx=tx-sx,dy=ty-sy,D=Math.hypot(dx,dy)||1;
  let T=Math.sqrt(2*D/G)*arc;if(T<.18)T=.18;
  return{vx:dx/T,vy:dy/T+.5*G*T,T};
}
function simTowers(dt){
  for(const tw of towers){
    const c=TOWERS[tw.type];
    if(tw.type==='well'){wellAmbient(tw,dt);continue;}
    tw.cd-=dt;tw.recoil*=Math.exp(-7*dt);
    let best=null,bd=-1e9;const r2=c.range*c.range;
    for(const e of enemies){if(e.dead)continue;const dx=e.x-tw.x,dy=e.y-tw.y;if(dx*dx+dy*dy<=r2&&e.dist>bd){bd=e.dist;best=e;}}
    tw.target=best;
    if(best){
      let sol=solve(tw.x,tw.y,best.x,best.y,c.arc);
      const eff=Math.max(8,best.sp*(1+0.012*(game.wave-1))*best.slow-best.push);
      const pr=posAt(clamp(best.dist+eff*sol.T,0,totalLen));
      sol=solve(tw.x,tw.y,pr.x,pr.y,c.arc);
      tw.angle=lerpAngle(tw.angle,Math.atan2(sol.vy,sol.vx),Math.min(1,dt*12));
      if(tw.cd<=0){
        tw.cd=1/c.rate;tw.recoil=1;
        const mx=tw.x+Math.cos(tw.angle)*c.barrel,my=tw.y+Math.sin(tw.angle)*c.barrel;
        shots.push({x:mx,y:my,vx:sol.vx,vy:sol.vy,r:c.pr,dmg:c.dmg,splash:c.splash,type:tw.type,col:c.color,big:c.big,life:6,trail:[],dead:false});
        parts.push({type:'flash',x:mx,y:my,r:c.big?10:6,life:.07,max:.07,col:c.color});
        sfxShoot(tw.type);
      }
    }
  }
}
function simEnemies(dt){
  const wl=towers.filter(x=>x.type==='well');
  for(const e of enemies){
    if(e.dead)continue;
    let slow=1,push=0,near=null;const R=TOWERS.well.range;
    for(const w of wl){
      const dx=w.x-e.x,dy=w.y-e.y,d=Math.hypot(dx,dy);
      if(d<R&&d>.001){
        const f=1-d/R,s=1-TOWERS.well.slow*(1-e.wres);
        if(s<slow){slow=s;near=w;}
        push=Math.max(push,TOWERS.well.push*f*(1-e.wres));
      }
    }
    e.slow=slow;e.push=push;e.near=near;
    const net=e.sp*(1+0.012*(game.wave-1))*slow-push;
    e.dist+=net*dt;if(e.dist<-140)e.dist=-140;
    const p=posAt(e.dist);e.x=p.x;e.y=p.y;e.a=p.a;
    e.rot+=(net*dt)/e.r;
    e.flash=Math.max(0,e.flash-dt);
    if(slow<1&&Math.random()<dt*7)spawnSwirl(near,e.x,e.y);
    if(e.dist>=totalLen)coreHit(e);
  }
}
function hitShot(p,e){
  e.hp-=p.dmg;e.flash=.12;p.dead=true;
  sparks(p.x,p.y,p.col,4);
  if(p.splash>0){
    for(const o of enemies){
      if(o===e||o.dead)continue;
      const d=Math.hypot(o.x-p.x,o.y-p.y);
      if(d<p.splash){o.hp-=p.dmg*(.35+.65*(1-d/p.splash));o.flash=.12;}
    }
    parts.push({type:'ring',x:p.x,y:p.y,r0:p.r+2,r1:p.splash,life:.35,max:.35,col:p.col});
    chunkBurst(p.x,p.y,p.big?12:7,'#39415a');
    shake=Math.max(shake,p.big?6:3);sfxBoom(p.big);
  }else sfxTick();
}
function updateShots(dt,demo){
  const wells=demo?DEMO_WELLS:towers.filter(x=>x.type==='well');
  for(const p of shots){
    if(p.dead)continue;
    p.vy+=G*dt;
    for(const w of wells){
      const dx=w.x-p.x,dy=w.y-p.y,d=Math.hypot(dx,dy);
      if(d<w.range&&d>1){const f=(1-d/w.range)*300/d;p.vx+=dx*f*dt;p.vy+=dy*f*dt;}
    }
    const sp=Math.hypot(p.vx,p.vy);if(sp>980){p.vx*=980/sp;p.vy*=980/sp;}
    p.x+=p.vx*dt;p.y+=p.vy*dt;p.life-=dt;
    p.trail.push({x:p.x,y:p.y});if(p.trail.length>7)p.trail.shift();
    if(!demo){
      for(const e of enemies){
        if(e.dead)continue;
        const dx=e.x-p.x,dy=e.y-p.y,rr=p.r+e.r;
        if(dx*dx+dy*dy<rr*rr){hitShot(p,e);break;}
      }
    }
  }
  shots=shots.filter(p=>!p.dead&&p.life>0&&p.y<H+90&&p.x>-140&&p.x<W+140);
}
function simUpdate(dt){
  if(game.phase==='build'){game.buildT-=dt;if(game.buildT<=0)callWave();}
  else{
    game.spawnT+=dt;
    while(spawnList.length&&game.spawnT>=spawnList[0].at)spawnEnemy(spawnList.shift().t);
    if(!spawnList.length&&!enemies.length){waveComplete();if(game.state!=='play')return;}
  }
  simTowers(dt);simEnemies(dt);updateShots(dt,false);
  for(const e of enemies)if(!e.dead&&e.hp<=0)kill(e);
  enemies=enemies.filter(e=>!e.dead);
  hudUpdate();
}

function addFloat(x,y,txt,col){floats.push({x,y,txt,col,life:1.1,max:1.1});}
function sparks(x,y,col,n){for(let i=0;i<n;i++){const a=rand(TAU),s=rand(60,230);parts.push({type:'spark',x,y,vx:Math.cos(a)*s,vy:Math.sin(a)*s,life:rand(.25,.5),max:.5,col});}}
function chunkBurst(x,y,n,col){for(let i=0;i<n;i++)parts.push({type:'chunk',x,y,vx:rand(-130,130),vy:rand(-270,-40),g:G*.85,size:rand(3,7),rot:rand(TAU),vr:rand(-9,9),life:rand(.6,1.1),max:1.1,col:i%4===0?'#ff9d3f':(col||'#4a5468')});}
function wellAmbient(w,dt){w.sw-=dt;if(w.sw<=0){w.sw=rand(.12,.3);const a=rand(TAU),rad=TOWERS.well.range*rand(.45,.85);parts.push({type:'swirl',cx:w.x,cy:w.y,a,rad,col:'#c07bff',life:.9,max:.9});}}
function spawnSwirl(w,x,y){if(!w)return;const d=Math.hypot(w.x-x,w.y-y)||1;parts.push({type:'swirl',cx:w.x,cy:w.y,a:Math.atan2(y-w.y,x-w.x),rad:d,col:'#c07bff',life:.8,max:.8});}
function updateParts(dt){
  for(const p of parts){
    p.life-=dt;
    if(p.type==='chunk'){p.vy+=p.g*dt;p.x+=p.vx*dt;p.y+=p.vy*dt;p.rot+=p.vr*dt;}
    else if(p.type==='spark'){p.x+=p.vx*dt;p.y+=p.vy*dt;p.vy+=300*dt;}
    else if(p.type==='swirl'){p.rad-=150*dt;p.a+=4.5*dt;}
  }
  parts=parts.filter(p=>p.life>0&&!(p.type==='swirl'&&p.rad<5));
  floats=floats.filter(f=>(f.life-=dt)>0);
  shake*=Math.exp(-5*dt);if(shake<.1)shake=0;
  coreFlash=Math.max(0,coreFlash-dt*1.2);gatePulse=Math.max(0,gatePulse-dt*1.4);
}

  ctx.strokeStyle='rgba(0,0,0,.45)';ctx.lineWidth=2;ctx.stroke();
  if(pct<.66){
    ctx.strokeStyle=`rgba(255,157,63,${pct<.33?.95:.55})`;ctx.lineWidth=pct<.33?1.8:1.2;
    for(const c of e.cracks){ctx.beginPath();ctx.moveTo(c.x1*e.r,c.y1*e.r);ctx.lineTo(c.x2*e.r,c.y2*e.r);ctx.stroke();}
  }
  ctx.restore();
  if(e.flash>0){ctx.globalAlpha=e.flash/.12*.7;ctx.fillStyle='#fff';ctx.beginPath();ctx.arc(e.x,e.y,e.r,0,TAU);ctx.fill();ctx.globalAlpha=1;}
  if(pct<1){
    const bw=e.r*2,bx=e.x-e.r,by=e.y-e.r-9;
    ctx.fillStyle='rgba(5,10,20,.8)';ctx.fillRect(bx-1,by-1,bw+2,5.5);
    ctx.fillStyle=pct>.5?'#7dff8a':pct>.25?'#ffd23f':'#ff5468';
    ctx.fillRect(bx,by,bw*pct,3.5);
  }
}
function drawShots(){
  ctx.globalCompositeOperation='lighter';
  for(const p of shots){
    const n=p.trail.length;
    for(let i=1;i<n;i++){
      ctx.globalAlpha=(i/n)*.45;ctx.strokeStyle=p.col;ctx.lineWidth=p.r*(i/n);
      ctx.beginPath();ctx.moveTo(p.trail[i-1].x,p.trail[i-1].y);ctx.lineTo(p.trail[i].x,p.trail[i].y);ctx.stroke();
    }
    ctx.globalAlpha=.3;ctx.fillStyle=p.col;ctx.beginPath();ctx.arc(p.x,p.y,p.r*2.4,0,TAU);ctx.fill();
    ctx.globalAlpha=1;ctx.beginPath();ctx.arc(p.x,p.y,p.r,0,TAU);ctx.fill();
    ctx.fillStyle='#fff';ctx.beginPath();ctx.arc(p.x,p.y,p.r*.5,0,TAU);ctx.fill();
  }
  ctx.globalAlpha=1;ctx.globalCompositeOperation='source-over';
}
function drawParts(){
  for(const p of parts){
    const k=Math.max(0,p.life/p.max);
    if(p.type==='chunk'){
      ctx.globalAlpha=k;ctx.save();ctx.translate(p.x,p.y);ctx.rotate(p.rot);
      ctx.fillStyle=p.col;ctx.fillRect(-p.size/2,-p.size/2,p.size,p.size);ctx.restore();
    }else{
      ctx.globalCompositeOperation='lighter';
      if(p.type==='spark'){
        ctx.globalAlpha=k;ctx.strokeStyle=p.col;ctx.lineWidth=2;
        ctx.beginPath();ctx.moveTo(p.x,p.y);ctx.lineTo(p.x-p.vx*.03,p.y-p.vy*.03);ctx.stroke();
      }else if(p.type==='ring'){
        const r=lerp(p.r0,p.r1,1-k);
        ctx.globalAlpha=k*.8;ctx.strokeStyle=p.col;ctx.lineWidth=3*k+1;
        ctx.beginPath();ctx.arc(p.x,p.y,r,0,TAU);ctx.stroke();
      }else if(p.type==='flash'){
        const r=p.r*(1+(1-k)*1.6);
        ctx.globalAlpha=k*.9;ctx.fillStyle=p.col;ctx.beginPath();ctx.arc(p.x,p.y,r,0,TAU);ctx.fill();
      }else if(p.type==='swirl'){
        const x=p.cx+Math.cos(p.a)*p.rad,y=p.cy+Math.sin(p.a)*p.rad;
        ctx.globalAlpha=k*.8;ctx.fillStyle=p.col;ctx.beginPath();ctx.arc(x,y,2,0,TAU);ctx.fill();
      }
      ctx.globalCompositeOperation='source-over';
    }
  }
  ctx.globalAlpha=1;
}
function drawFloats(){
  ctx.font='700 15px Consolas,Menlo,monospace';ctx.textAlign='center';
  for(const f of floats){
    const k=f.life/f.max;
    ctx.globalAlpha=Math.min(1,k*1.4);
    ctx.fillStyle='#050a14';ctx.fillText(f.txt,f.x+1,f.y-(1-k)*26+1);
    ctx.fillStyle=f.col;ctx.fillText(f.txt,f.x,f.y-(1-k)*26);
  }
  ctx.globalAlpha=1;ctx.textAlign='left';
}
function drawRange(x,y,r,col,a){
  ctx.globalAlpha=a*.15;ctx.fillStyle=col;ctx.beginPath();ctx.arc(x,y,r,0,TAU);ctx.fill();
  ctx.globalAlpha=a;ctx.strokeStyle=col;ctx.lineWidth=1.5;ctx.stroke();ctx.globalAlpha=1;
}
function drawArc(sx,sy,col){
  let x=sx,y=sy,vx=190,vy=-170;
  ctx.fillStyle=col;
  for(let i=0;i<46;i++){
    vy+=G*.045;x+=vx*.045;y+=vy*.045;
    if(i%3===0){ctx.globalAlpha=Math.max(0,.8-i*.016);ctx.beginPath();ctx.arc(x,y,2,0,TAU);ctx.fill();}
    if(y>H)break;
  }
  ctx.globalAlpha=1;
}
function drawGhost(){
  const c=TOWERS[placing],x=hover.gx*CELL+20,y=hover.gy*CELL+20;
  const ok=!blocked.has(hover.gx+','+hover.gy)&&!occ.has(hover.gx+','+hover.gy)&&game.flux>=c.cost;
  drawRange(x,y,c.range,ok?'#5fe8ff':'#ff5468',.35);
  ctx.globalAlpha=.75;
  if(placing==='well')drawWell(x,y);
  else{
    ctx.fillStyle='#16233c';ctx.beginPath();ctx.arc(x,y,14,0,TAU);ctx.fill();
    ctx.strokeStyle=c.color;ctx.lineWidth=2;ctx.stroke();
    ctx.save();ctx.translate(x,y);ctx.rotate(-.5);ctx.fillStyle='#9fb6d8';ctx.fillRect(0,-c.bw/2,c.barrel,c.bw);ctx.restore();
  }
  ctx.globalAlpha=1;
  if(ok&&placing!=='well')drawArc(x,y-6,c.color);
  ctx.strokeStyle=ok?'rgba(95,232,255,.8)':'rgba(255,84,104,.8)';ctx.lineWidth=1.5;
  ctx.strokeRect(hover.gx*CELL+1,hover.gy*CELL+1,CELL-2,CELL-2);
}
function render(){
  ctx.setTransform(dpr,0,0,dpr,0,0);
  const w=cv.width/dpr,h=cv.height/dpr;
  ctx.fillStyle='#04070f';ctx.fillRect(0,0,w,h);
  ctx.save();
  ctx.translate(ox+(Math.random()*2-1)*shake,oy+(Math.random()*2-1)*shake);
  ctx.scale(scale,scale);
  ctx.drawImage(bg,-200,-200);
  ctx.globalCompositeOperation='lighter';
  for(const s of twinkles){
    ctx.globalAlpha=.2+.45*(.5+.5*Math.sin(t*s.s+s.ph));
    ctx.fillStyle='#cfe3ff';ctx.beginPath();ctx.arc(s.x,s.y,s.r,0,TAU);ctx.fill();
  }
  ctx.globalAlpha=1;ctx.globalCompositeOperation='source-over';
  drawPath();drawGate();drawCore();
  if(game.state==='menu'){for(const dw of DEMO_WELLS)drawWell(dw.x,dw.y);}
  else for(const tw of towers){tw.type==='well'?drawWell(tw.x,tw.y):drawTower(tw);}

  if(selected&&game.state==='play')drawRange(selected.x,selected.y,TOWERS[selected.type].range,TOWERS[selected.type].color,.5);
  ctx.drawImage(vig,0,0);
  ctx.restore();
}

  ambT-=dt;
  if(ambT<=0){
    ambT=rand(.5,1);

  }
  wellAmbient(DEMO_WELLS[0],dt);wellAmbient(DEMO_WELLS[1],dt);
}
function frame(now){

  const raw=Math.min(.05,(now-last)/1000);last=now;t+=raw;

  const tick=(game.state==='play'&&!game.paused)?raw*game.speed:raw;
  updateParts(tick);
  render();
}
