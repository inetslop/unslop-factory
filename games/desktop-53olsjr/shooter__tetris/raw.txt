  ship.vx+=((dir*9.2)-ship.vx)*Math.min(1,dt*16);
  ship.x=clamp(ship.x+ship.vx*dt,0,COLS-1);
  ship.recoil=Math.max(0,ship.recoil-dt*7);
  cool-=dt;
  if((held.fire||tHold.fire)&&cool<=0){
    if(energy>0||powers.od>0)fire();
    else{cool=0.3;drainFlash=1;if(tGlobal-lastSput>0.4){lastSput=tGlobal;sfx.sputter();}}
  }

  // pieces fall
  const spd=Math.min(7.5,1.5+(lvl-1)*0.42)*sm;
  for(const p of pieces){
    if(p.dead)continue;
    p.acc+=spd*dt;p.flash=Math.max(0,p.flash-dt*3);
    while(p.acc>=1&&!p.dead){p.acc-=1;p.row++;if(collides(p)){p.row--;lockPiece(p);break;}}
  }
  pieces=pieces.filter(p=>!p.dead&&p.row<ROWS+4);
  if(state!=='play')return;

  // bolts
  let destroyedAny=false;
  for(const b of bolts){
    const y0=b.y;b.y-=26*dt;
    const rEnd=Math.floor(y0),rStart=Math.floor(b.y);
    for(let r=rEnd;r>=rStart;r--){
      if(r<0||r>=ROWS)continue;
      const cell=grid[r][b.col];
      if(cell&&!cell.clearing){destroyCell(r,b.col);destroyedAny=true;}
    }
    for(const p of pieces){if(p.dead)continue;
      for(let i=0;i<p.cells.length;i++){const c=p.cells[i];if(!c.alive)continue;
        const cx=p.x+c.dc,cy=p.row+c.dr;if(cy<0||cx!==b.col)continue;
        if(cy<=y0&&cy+1>=b.y){hitPiece(p,i);destroyedAny=true;}}}
    for(const cp of caps){if(cp.dead)continue;
      if(Math.abs(cp.col-b.col)<0.55&&cp.y<=y0&&cp.y+1>=b.y)collectCap(cp);}
  }
  bolts=bolts.filter(b=>b.y>-2);
  if(destroyedAny){settleAll();pendingLine=0.2;}
  if(pendingLine>0){pendingLine-=dt;if(pendingLine<=0)checkLines('shot');}

  // line clear animation
  for(const a of clearAnims)a.ct+=dt;
  const done=clearAnims.filter(a=>a.ct>0.34);
  if(done.length){
    const set=new Set(done.map(a=>a.r));
    clearAnims=clearAnims.filter(a=>!set.has(a.r));
    for(let r=ROWS-1;r>=0;r--){
      if(set.has(r))continue;
      let above=0;for(const x of set)if(x>r)above++;
      if(!above)continue;
      for(let c=0;c<COLS;c++){const cell=grid[r][c];if(!cell||cell.clearing)continue;
        grid[r][c]=null;const nr=r+above;cell.drop=(r+(cell.drop||0))-nr;grid[nr][c]=cell;dustAt(c,nr);
        burst(px(c)+L.cell/2,py(nr), '#ffffff',1,1.2,2);}
    }
  }

  // capsules
  for(const cp of caps){if(cp.dead)continue;cp.y+=2.3*dt*sm;
    const c=Math.floor(cp.col),r=Math.floor(cp.y);
    if(r>=ROWS||grid[r][c]){cp.dead=true;burst(px(cp.col)+L.cell/2,py(Math.min(r,ROWS-1))+L.cell/2,'#8fa3c8',5,2,2.5);}
  }
  caps=caps.filter(c=>!c.dead);

  // grid cell animation
  for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++){const cell=grid[r][c];if(!cell)continue;
    if(cell.drop<0)cell.drop=Math.min(0,cell.drop+dt*26);
    if(cell.flash>0)cell.flash-=dt*3;}

  // danger heartbeat
  const h=stackHeight();
  if(h>=17){hbT-=dt;if(hbT<=0){hbT=0.9;sfx.warn();}}

  updateFx(dt);
}
function updateFx(dt){
  for(const p of parts){p.t+=dt;p.vy+=p.g*dt;p.x+=p.vx*dt;p.y+=p.vy*dt;p.rot+=p.vr*dt;}
  parts=parts.filter(p=>p.t<p.max);
  for(const r of rings){r.r+=r.vr*dt;r.a-=dt*2.4;}
  rings=rings.filter(r=>r.a>0);
  for(const t of texts){t.t+=dt;t.y-=26*dt;}
  texts=texts.filter(t=>t.t<t.max);
}

  const shx=(Math.random()-0.5)*shake,shy=(Math.random()-0.5)*shake;
  ctx.save();ctx.translate(shx,shy);
  drawFieldBack();
  if(state!=='menu'){drawGuides();drawGrid();}
  else{ctx.globalAlpha=0.32;}
  // pieces (clipped to field for falling parts)
  ctx.save();ctx.beginPath();ctx.rect(L.fx-1,L.fy-1,L.fw+2,L.fh+2);ctx.clip();
  for(const p of pieces){if(p.dead)continue;
    for(const c of p.cells){if(!c.alive)continue;drawBlock(px(p.x+c.dc),py(p.row),L.cell,p.t,clamp(p.flash||0,0,1));}}
  ctx.restore();
  if(state==='menu')ctx.globalAlpha=1;
  drawCaps();
  if(state!=='menu'){drawBolts();drawShip();}
  drawFX();
  if(state!=='menu')drawTexts();
  drawFrame();
  ctx.restore();

  if(state!=='menu')drawHUD();

  // slow-mo tint
  if(state==='play'&&powers.sm>0){ctx.fillStyle='rgba(90,140,255,0.06)';ctx.fillRect(0,0,W,H);}
  // danger vignette
  if((state==='play'||state==='pause')&&grid){
    const h=stackHeight();
    if(h>=13){
      const a=(0.12+0.1*Math.sin(tGlobal*7))*Math.min(1,(h-12)/4);
      const g=ctx.createLinearGradient(0,L.fy,0,L.fy+L.cell*4);
      g.addColorStop(0,`rgba(255,60,80,${a})`);g.addColorStop(1,'rgba(255,60,80,0)');
      ctx.fillStyle=g;ctx.fillRect(L.fx,L.fy,L.fw,L.cell*4);
    }
  }
}
