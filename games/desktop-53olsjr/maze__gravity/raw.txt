function update(dt){
  t+=dt; shake=Math.max(0,shake-dt*2.2); flashRed=Math.max(0,flashRed-dt*1.4); flashCyan=Math.max(0,flashCyan-dt*1.2);
  doors.forEach(d=>{ if(d.open) d.anim=Math.min(1,d.anim+dt*3); });
  // ambient
  const gs=p.grav;
  stars.forEach(s=>{ s.y+=gs*s.z*.5*dt*(s.z===2?1.8:1); if(s.y>H)s.y-=H; if(s.y<0)s.y+=H; });
  motes.forEach(m=>{ m.y+=gs*m.s*.45*dt; if(m.y>ROWS-1)m.y=1; if(m.y<1)m.y=ROWS-1; });
  // particles
  for(let i=parts.length-1;i>=0;i--){ const q=parts[i]; q.life-=dt;
    if(q.life<=0){ parts.splice(i,1); continue; }
    if(q.kind==='ring'){ q.r+=q.vr*dt; continue; }
    q.x+=q.vx*dt; q.y+=q.vy*dt;
    if(q.grav) q.vy+=q.grav*dt; else { q.vx*= (1-2.2*dt); q.vy*=(1-2.2*dt); } }

  if(mode==='play'){
    runTime+=dt; lvlTime+=dt; p.flipCd-=dt; p.squash=Math.max(0,p.squash-dt*5);
    const ax=(keys.right?1:0)-(keys.left?1:0);
    if(ax) p.face=ax;
    p.vx+=(ax*MAXRUN-p.vx)*Math.min(1,dt*(p.grounded?16:9));
    if(!ax&&p.grounded&&Math.abs(p.vx)<.2) p.vx=0;
    p.vy+=G*p.grav*dt; if(p.vy>MAXFALL)p.vy=MAXFALL; if(p.vy<-MAXFALL)p.vy=-MAXFALL;
    wasGnd=p.grounded;
    p.x+=p.vx*dt; collideX();
    p.y+=p.vy*dt; collideY();
    if(p.grounded&&buffer&&t-buffer.at<.25&&buffer.dir!==p.grav&&p.flipCd<=0) doFlip(buffer.dir);
    // hazards
    for(const h of hazards){ if(circleRect(p.x,p.y,.3,h.x,h.y)){ die(); break; } }
    if(mode!=='play') return;
    // orbs
    for(const o of orbs){ if(o.got)continue; const dx=o.x-p.x,dy=o.y-p.y;
      if(dx*dx+dy*dy<.78*.78){ o.got=true; totalOrbs++; sfx.orb(); burst(o.x,o.y,12,4,'#5fe8ff',0); ring(o.x,o.y,'rgba(95,232,255,.9)');
        addP({kind:'text',x:o.x,y:o.y-.6,vx:0,vy:-1.3,life:.9,max:.9,txt:'+100'}); } }
    // switches
    for(const s of switches){ if(s.used)continue; const dx=s.x-p.x,dy=s.y-p.y;
      if(dx*dx+dy*dy<.62*.62){ s.used=true; sfx.sw(); burst(s.x,s.y,14,5,'#7dffb0',0); ring(s.x,s.y,'rgba(125,255,176,.9)');
        doors.forEach(d=>{ if(d.t===s.t&&!d.open){ d.open=true; burst(d.x+.5,d.y+.5,14,5,'#5fe8ff',0); ring(d.x+.5,d.y+.5,'rgba(95,232,255,.9)'); } }); } }
    // exit
    const dx=exit.x-p.x,dy=exit.y-p.y;
    if(dx*dx+dy*dy<.85*.85){ setMode('clear'); clearT=0; flashCyan=.35; sfx.clear(); }
    trail.unshift({x:p.x,y:p.y}); if(trail.length>12) trail.pop();
  }
  else if(mode==='dead'){
    deadT+=dt; if(deadT>.55){ respawn(); setMode('play'); }
  }
  else if(mode==='clear'){
    clearT+=dt;
    p.x+=(exit.x-p.x)*Math.min(1,dt*9); p.y+=(exit.y-p.y)*Math.min(1,dt*9);
    if(clearT<.5&&Math.random()<.6){ const a=Math.random()*6.283,r=.7;
      addP({kind:'spark',x:exit.x+Math.cos(a)*r,y:exit.y+Math.sin(a)*r,vx:-Math.cos(a)*2,vy:-Math.sin(a)*2,life:.4,max:.4,size:2,col:'#bfefff',grav:0}); }
    if(clearT>1.35) nextLevel();
  }
}

/* ================= RENDER ================= */
const wallCols=['#16323f','#18374a','#15303c','#193a4b'];
function render(){
  ctx.clearRect(0,0,W,H);
  ctx.fillStyle=bgGrad; ctx.fillRect(0,0,W,H);
  ctx.fillStyle=glowA; ctx.fillRect(0,0,W,H);
  ctx.fillStyle=glowB; ctx.fillRect(0,0,W,H);
  // stars
  for(const s of stars){ const a=(s.z===2?.5:.3)*(0.6+0.4*Math.sin(t*s.tw+s.x));
    ctx.fillStyle='rgba(190,225,240,'+a.toFixed(3)+')';
    ctx.fillRect(s.x,s.y,s.z===2?2:1,s.z===2?2:1); }

  const shx=(Math.random()-.5)*shake*tile*.6, shy=(Math.random()-.5)*shake*tile*.6;
  ctx.save(); ctx.translate(ox+shx,oy+shy);
  // maze frame glow
  ctx.strokeStyle='rgba(95,232,255,.10)'; ctx.lineWidth=1; ctx.strokeRect(-8,-8,COLS*tile+16,ROWS*tile+16);
  ctx.strokeStyle='rgba(95,232,255,.04)'; ctx.strokeRect(-16,-16,COLS*tile+32,ROWS*tile+32);
  // motes
  for(const m of motes){ ctx.fillStyle='rgba(220,240,250,.06)'; ctx.fillRect(m.x*tile,m.y*tile,2,2); }
  // walls
  for(let y=0;y<ROWS;y++)for(let x=0;x<COLS;x++){ if(!grid[y*COLS+x])continue;
    const px=x*tile,py=y*tile;
    ctx.fillStyle=wallCols[(x*31+y*57)&3]; ctx.fillRect(px,py,tile+.5,tile+.5);
    if(y>0&&!grid[(y-1)*COLS+x]){ ctx.fillStyle='rgba(96,190,220,.35)'; ctx.fillRect(px,py,tile,2); }
    if(y<ROWS-1&&!grid[(y+1)*COLS+x]){ ctx.fillStyle='rgba(3,10,14,.5)'; ctx.fillRect(px,py+tile-2,tile,2); }
    if(x>0&&!grid[y*COLS+x-1]){ ctx.fillStyle='rgba(96,190,220,.14)'; ctx.fillRect(px,py,1.5,tile); }
    if(x<COLS-1&&!grid[y*COLS+x+1]){ ctx.fillStyle='rgba(3,10,14,.35)'; ctx.fillRect(px+tile-1.5,py,1.5,tile); } }
  // doors
  for(const d of doors){ const px=d.x*tile,py=d.y*tile;
    if(!d.open){ const a=.5+.3*Math.sin(t*6+d.x);
      ctx.fillStyle='rgba(95,232,255,'+(.10*a).toFixed(3)+')'; ctx.fillRect(px,py,tile,tile);
      ctx.strokeStyle='rgba(140,240,255,'+a.toFixed(3)+')'; ctx.lineWidth=2; ctx.beginPath();
      for(let i=1;i<=3;i++){ const lx=px+tile*i/4; ctx.moveTo(lx,py+3); ctx.lineTo(lx,py+tile-3); } ctx.stroke(); }
    else if(d.anim<1){ const off=d.anim*tile*.5, a=1-d.anim;
      ctx.strokeStyle='rgba(140,240,255,'+(a*.8).toFixed(3)+')'; ctx.lineWidth=2; ctx.beginPath();
      for(let i=1;i<=3;i++){ const lx=px+tile*i/4+off*(i<2?-1:1); ctx.moveTo(lx,py+3); ctx.lineTo(lx,py+tile-3); } ctx.stroke(); }
    else { ctx.strokeStyle='rgba(95,232,255,.10)'; ctx.lineWidth=1; ctx.strokeRect(px+2,py+2,tile-4,tile-4); } }
  // switches
  for(const s of switches){ const px=s.x*tile,py=s.y*tile, pulse=1+.15*Math.sin(t*4+s.x), dim=s.used?.3:1;
    ctx.save(); ctx.translate(px,py); ctx.scale(pulse,pulse);
    ctx.strokeStyle='rgba(125,255,176,'+(.85*dim).toFixed(3)+')'; ctx.lineWidth=2;
    ctx.beginPath(); ctx.arc(0,0,tile*.24,0,6.283); ctx.stroke();
    ctx.fillStyle='rgba(125,255,176,'+(dim).toFixed(3)+')'; ctx.beginPath(); ctx.arc(0,0,tile*.1,0,6.283); ctx.fill();
    ctx.restore(); }
  // hazards
  for(const h of hazards){ const cx=(h.x+.5)*tile,cy=(h.y+.5)*tile, r=tile*.3, rot=t*1.2+h.ph;
    const g=ctx.createRadialGradient(cx,cy,0,cx,cy,r*2.4);
    g.addColorStop(0,'rgba(255,93,93,.35)'); g.addColorStop(1,'rgba(255,93,93,0)');
    ctx.fillStyle=g; ctx.beginPath(); ctx.arc(cx,cy,r*2.4,0,6.283); ctx.fill();
    ctx.save(); ctx.translate(cx,cy); ctx.rotate(rot);
    ctx.fillStyle='#ff5d5d'; ctx.strokeStyle='#ffd7c9'; ctx.lineWidth=1.5;
    ctx.beginPath(); ctx.moveTo(0,-r); ctx.lineTo(r*.62,0); ctx.lineTo(0,r); ctx.lineTo(-r*.62,0); ctx.closePath();
    ctx.fill(); ctx.stroke();
    ctx.fillStyle='rgba(255,255,255,.85)'; ctx.beginPath(); ctx.arc(0,-r*.3,r*.14,0,6.283); ctx.fill();
    ctx.restore(); }
  // orbs
  for(const o of orbs){ if(o.got)continue; const bob=Math.sin(t*2+o.ph)*tile*.06;
    const cx=o.x*tile,cy=o.y*tile+bob, r=tile*.17;
    const g=ctx.createRadialGradient(cx,cy,0,cx,cy,r*3);
    g.addColorStop(0,'rgba(95,232,255,.5)'); g.addColorStop(1,'rgba(95,232,255,0)');
    ctx.fillStyle=g; ctx.beginPath(); ctx.arc(cx,cy,r*3,0,6.283); ctx.fill();
    ctx.fillStyle='#eaffff'; ctx.beginPath(); ctx.arc(cx,cy,r*.55,0,6.283); ctx.fill();
    ctx.strokeStyle='rgba(95,232,255,.9)'; ctx.lineWidth=1.5; ctx.beginPath(); ctx.arc(cx,cy,r,0,6.283); ctx.stroke(); }
  // exit portal
  { const cx=exit.x*tile,cy=exit.y*tile,R=tile*.42;
    const g=ctx.createRadialGradient(cx,cy,0,cx,cy,R*2.6);
    g.addColorStop(0,'rgba(191,239,255,.5)'); g.addColorStop(.4,'rgba(95,232,255,.18)'); g.addColorStop(1,'rgba(95,232,255,0)');
    ctx.fillStyle=g; ctx.beginPath(); ctx.arc(cx,cy,R*2.6,0,6.283); ctx.fill();
    const inner=ctx.createRadialGradient(cx,cy,0,cx,cy,R);
    inner.addColorStop(0,'#0a2733'); inner.addColorStop(.8,'#12455c'); inner.addColorStop(1,'#bfefff');
    ctx.fillStyle=inner; ctx.beginPath(); ctx.arc(cx,cy,R,0,6.283); ctx.fill();
    ctx.strokeStyle='rgba(191,239,255,.9)'; ctx.lineWidth=2.5;
    for(let k=0;k<2;k++){ const a0=t*2+k*Math.PI; ctx.beginPath(); ctx.arc(cx,cy,R*1.35,a0,a0+1.9); ctx.stroke(); } }
  // trail
  if(mode!=='dead'&&mode!=='menu'){ ctx.globalCompositeOperation='lighter';
    for(let i=trail.length-1;i>=0;i--){ const q=trail[i],f=(i+1)/trail.length;
      ctx.fillStyle='rgba(255,190,80,'+(f*.16).toFixed(3)+')';
      ctx.beginPath(); ctx.arc(q.x*tile,q.y*tile,PR*tile*(.3+.7*f),0,6.283); ctx.fill(); }
    ctx.globalCompositeOperation='source-over'; }
  // player
  if(mode==='play'||mode==='clear'){
    const shrink=mode==='clear'?Math.max(0,1-clearT/.5):1;
    if(shrink>.04){
      const px=p.x*tile,py=p.y*tile,r=PR*tile;
      const st=Math.min(1,Math.abs(p.vy)/MAXFALL);
      let sx,sy; if(!p.grounded){ sy=1+st*.35; sx=1-st*.22; } else { sy=1-p.squash*.4; sx=1+p.squash*.3; }
      ctx.save(); ctx.translate(px,py); ctx.scale(sx*shrink,sy*shrink);
      const g=ctx.createRadialGradient(0,0,0,0,0,r*2.4);
      g.addColorStop(0,'rgba(255,190,80,.35)'); g.addColorStop(1,'rgba(255,190,80,0)');
      ctx.fillStyle=g; ctx.beginPath(); ctx.arc(0,0,r*2.4,0,6.283); ctx.fill();
      const b=ctx.createRadialGradient(-r*.3,-p.grav*r*.3,r*.15,0,0,r);
      b.addColorStop(0,'#fff6dc'); b.addColorStop(.55,'#ffc247'); b.addColorStop(1,'#b06a10');
      ctx.fillStyle=b; ctx.beginPath(); ctx.arc(0,0,r,0,6.283); ctx.fill();
      ctx.strokeStyle='rgba(255,217,138,.5)'; ctx.lineWidth=1; ctx.stroke();
      const blink=(t%3.2)<.12? .15:1;
      ctx.fillStyle='#4a2c07';
      ctx.beginPath(); ctx.ellipse(-r*.3+p.face*r*.12, p.grav*r*.22, r*.16, r*.16*blink,0,0,6.283); ctx.fill();
      ctx.beginPath(); ctx.ellipse( r*.3+p.face*r*.12, p.grav*r*.22, r*.16, r*.16*blink,0,0,6.283); ctx.fill();
      // gravity chevrons
      const ch=Math.sin(t*5)*.15+.4;
      ctx.strokeStyle='rgba(255,210,120,'+ch.toFixed(2)+')'; ctx.lineWidth=2; ctx.beginPath();
      for(let k=0;k<2;k++){ const d=r*(1.7+k*.4), s=p.grav;
        ctx.moveTo(-r*.3,d*s); ctx.lineTo(0,d*s+r*.35*s); ctx.lineTo(r*.3,d*s); }
      ctx.stroke();
      ctx.restore(); } }
  // particles
  for(const q of parts){ const f=Math.max(0,q.life/q.max), qx=q.x*tile, qy=q.y*tile;
    if(q.kind==='spark'){ ctx.globalCompositeOperation='lighter'; ctx.fillStyle=q.col;
      ctx.globalAlpha=f; ctx.beginPath(); ctx.arc(qx,qy,q.size*f+.4,0,6.283); ctx.fill();
      ctx.globalAlpha=1; ctx.globalCompositeOperation='source-over'; }
    else if(q.kind==='dust'){ ctx.fillStyle=q.col; ctx.globalAlpha=f*.5;
      ctx.beginPath(); ctx.arc(qx,qy,q.size,0,6.283); ctx.fill(); ctx.globalAlpha=1; }
    else if(q.kind==='ring'){ ctx.strokeStyle=q.col; ctx.globalAlpha=f; ctx.lineWidth=2.5*f+.5;
      ctx.beginPath(); ctx.arc(qx,qy,q.r,0,6.283); ctx.stroke(); ctx.globalAlpha=1; }
    else if(q.kind==='text'){ ctx.font='700 '+Math.max(10,tile*.4)+'px '+'ui-monospace,Menlo,Consolas,monospace';
      ctx.fillStyle='rgba(95,232,255,'+f.toFixed(2)+')'; ctx.textAlign='center'; ctx.fillText(q.txt,qx,qy); } }
  ctx.restore();

  // speedlines
  if(mode==='play'&&Math.abs(p.vy)>11){ const n=7,a=(Math.abs(p.vy)-8)*.014;
    ctx.strokeStyle='rgba(200,235,255,'+Math.min(.16,a).toFixed(3)+')'; ctx.lineWidth=1.5; ctx.beginPath();
    for(let i=0;i<n;i++){ const x=Math.random()*W,y=Math.random()*H,L=(Math.abs(p.vy)-8)*7*p.grav;
      ctx.moveTo(x,y); ctx.lineTo(x+(Math.random()-.5)*6,y+L); } ctx.stroke(); }

  if(flashRed>0){ ctx.fillStyle='rgba(255,60,60,'+(flashRed*.4).toFixed(3)+')'; ctx.fillRect(0,0,W,H); }
  if(flashCyan>0){ ctx.fillStyle='rgba(95,232,255,'+(flashCyan*.35).toFixed(3)+')'; ctx.fillRect(0,0,W,H); }
  ctx.fillStyle=vigGrad; ctx.fillRect(0,0,W,H);

  // HUD text
  if(mode!=='menu'&&mode!=='end'){
    const s=LEVELS[level];
    el.lvlNum.textContent=(level+1)+'/5'; el.lvlName.textContent=s.name;
    const got=orbs.filter(o=>o.got).length;
    el.orbCount.textContent=got+'/'+s.orbs.length;
    el.timeVal.textContent=fmtTime(runTime);
    el.flipVal.textContent=totalFlips;
    el.deathVal.textContent=deaths;
  }
}
