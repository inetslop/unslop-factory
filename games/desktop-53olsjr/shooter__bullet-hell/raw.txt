function updatePlayer(gdt){const p=player;p.inv=Math.max(0,p.inv-gdt);
  if(p.bombing>0){p.bombing-=gdt;const k=1-p.bombing/1.4;p.bombR=70+k*Math.max(W,H)*.62;
    for(let i=bullets.length-1;i>=0;i--){const b=bullets[i],dx=b.x-p.x,dy=b.y-p.y;
      if(dx*dx+dy*dy<p.bombR*p.bombR){sparkle(b);rmB(i);}}
    if(boss&&!boss.dying)bossDmg(120*gdt);}
  else p.bombR=0;
  let sx=0,sy=0;
  if(keys.KeyW||keys.ArrowUp)sy-=1;if(keys.KeyS||keys.ArrowDown)sy+=1;
  if(keys.KeyA||keys.ArrowLeft)sx-=1;if(keys.KeyD||keys.ArrowRight)sx+=1;
  if(ptr.on){const dx=ptr.x-p.x,dy=ptr.y-p.y,d=Math.hypot(dx,dy)||1,m=Math.min(d,520*gdt);p.x+=dx/d*m;p.y+=dy/d*m;}
  else{p.focus=!!(keys.ShiftLeft||keys.ShiftRight);const sp=p.focus?160:360;
    const m=Math.hypot(sx,sy)||1;p.x+=sx/m*sp*gdt;p.y+=sy/m*sp*gdt;}
  p.x=Math.max(16,Math.min(W-16,p.x));p.y=Math.max(70,Math.min(H-16,p.y));}

function updBullets(gdt,collide){for(let i=bullets.length-1;i>=0;i--){const b=bullets[i];
  if(b.h&&b.life>0){b.life-=gdt;
    const a=Math.atan2(b.vy,b.vx),ta=Math.atan2(player.y-b.y,player.x-b.x);
    let da=ta-a;while(da>Math.PI)da-=TAU;while(da<-Math.PI)da+=TAU;
    const mx=b.turn*gdt;if(Math.abs(da)<mx)a=ta;else a+=Math.sign(da)*mx;
    const sp=Math.hypot(b.vx,b.vy);b.vx=Math.cos(a)*sp;b.vy=Math.sin(a)*sp;}
  b.x+=b.vx*gdt;b.y+=b.vy*gdt;b.rot+=b.vr*gdt;
  if(b.x<-40||b.x>W+40||b.y<-60||b.y>H+40){rmB(i);continue;}
  if(collide&&state==='play'&&player.inv<=0&&player.bombing<=0){
    const dx=b.x-player.x,dy=b.y-player.y,d2=dx*dx+dy*dy,rr=b.r+player.r;
    if(d2<rr*rr){killPlayer();continue;}
    else if(!b.gr&&d2<(26+b.r)*(26+b.r)){b.gr=true;grazes++;score+=30;spark(b.x,b.y,'#9be8ff',2);}}}}

function updEnemies(gdt){const d=diff();
  for(let i=enemies.length-1;i>=0;i--){const e=enemies[i];e.t+=gdt;
    if(e.k===0){e.y+=e.vy*gdt;e.x=e.bx+Math.sin(e.t*e.fw)*e.fa;e.ft-=gdt;
      if(e.ft<=0&&e.y>50&&e.y<H*.72){e.ft=Math.max(1,1.9-stage*.09);
        const sp=Math.min(310,(150+stage*16)*d),dx=player.x-e.x,dy=player.y-e.y,dd=Math.hypot(dx,dy)||1;
        addB(e.x,e.y+8,dx/dd*sp,dy/dd*sp,'#ff8fb0','dot',4.5);
        if(stage>2)addB(e.x-6,e.y+6,dx/dd*sp*.9,dy/dd*sp*.9,'#ffb63d','dot',4.5);}
      if(e.y>H+50){rmE(i);continue;}}
    else{if(!e.exit&&e.y<e.ty)e.y+=130*gdt;
      else{e.ht+=gdt;e.x+=Math.sin(e.t*.7)*24*gdt;e.ft-=gdt;
        if(e.ft<=0&&e.ht<5.6){e.ft=1.15;e.rot+=.5;const n=10+stage*2,sp=Math.min(300,(125+stage*12)*d);
          for(let q=0;q<n;q++){const a=e.rot+q/n*TAU;addB(e.x,e.y,Math.cos(a)*sp,Math.sin(a)*sp,q%2?'#ff8fb0':'#9be8ff','dot',4.5);}}
        if(e.ht>6)e.exit=true;}
      if(e.exit){e.y+=150*gdt;if(e.y>H+60){rmE(i);continue;}}}}}

function updBoss(gdt){const b=boss,d=diff(),R=bossR();b.t+=gdt;b.rot+=gdt*(.6+b.ph*.35);
  if(b.dying){b.dieT-=gdt;b.sub-=gdt;
    if(b.sub<=0){b.sub=.12;const a=RND()*TAU,rr=RND()*R*.9;
      boom(b.x+Math.cos(a)*rr,b.y+Math.sin(a)*rr,PHCOL[b.ph],8,.7);shake=Math.max(shake,6);}
    if(b.dieT<=0)bossDie();return;}
  const tx=W/2+Math.sin(b.t*.45)*W*.3,ty=H*.21+Math.sin(b.t*.85)*H*.04;
  const k=Math.min(1,gdt*(b.fight?1.8:1.2));b.x+=(tx-b.x)*k;b.y+=(ty-b.y)*k;
  if(!b.fight){if(b.t>1.8){b.fight=true;nextPattern();}return;}
  b.patT-=gdt;b.sub-=gdt;
  switch(b.pat){
    case 'spiral':{b.rot2+=gdt*(1.9+d*.5);
      if(b.sub<=0){b.sub=.05;const arms=5+Math.min(4,stage-1),sp=Math.min(300,(115+stage*10)*d);
        for(let i=0;i<arms;i++){const a=b.rot2+i/arms*TAU;addB(b.x,b.y,Math.cos(a)*sp,Math.sin(a)*sp,'#ff8fb0','dot',4.5);}}break;}
    case 'spiralFast':{b.rot2+=gdt*(3+d*.6);
      if(b.sub<=0){b.sub=.045;const arms=8,sp=Math.min(320,(135+stage*8)*d);
        for(let i=0;i<arms;i++){const a=b.rot2+i/arms*TAU+(i%2?TAU/16:0);addB(b.x,b.y,Math.cos(a)*sp,Math.sin(a)*sp,'#ffb63d','dot',4.5);}}break;}
    case 'ring':{if(b.sub<=0&&b.cnt<3){b.sub=.55;b.cnt++;const n=14+stage*2,ro=RND()*TAU,sp=Math.min(300,(135+stage*12)*d);
      for(let i=0;i<n;i++){const a=ro+i/n*TAU;addB(b.x,b.y,Math.cos(a)*sp,Math.sin(a)*sp,i%2?'#ff8fb0':'#9be8ff','dot',4.5);}}break;}
    case 'fan':{if(b.sub<=0&&b.cnt<3){b.sub=.22;b.cnt++;const n=6+Math.min(4,stage),
      base=Math.atan2(player.y-b.y,player.x-b.x),sp=Math.min(330,(215+stage*12)*d);
      for(let i=0;i<n;i++){const a=base+(i-(n-1)/2)*.17;addB(b.x,b.y,Math.cos(a)*sp,Math.sin(a)*sp,'#ffb63d','dot',4.5);}}break;}
    case 'wall':{if(b.sub<=0&&b.cnt<4){b.sub=.5;b.cnt++;
      const gx=Math.max(70,Math.min(W-70,player.x+(RND()-.5)*W*.12)),gw=Math.max(96,185-stage*9),sp=Math.min(320,(140+stage*16)*d);
      for(let x=17;x<W;x+=34)if(Math.abs(x-gx)>gw/2)addB(x,-14,0,sp,'#9be8ff','dot',5);}break;}
    case 'rain':{if(b.sub<=0){b.sub=.085;const sp=Math.min(300,(110+stage*10)*d)+RND()*60;
      addB(RND()*W,-16,(RND()-.5)*46,sp,'#ff8fb0','petal',5);}break;}
    case 'seeker':{if(b.sub<=0&&b.cnt<4){b.sub=.62;b.cnt++;
      const base=Math.atan2(player.y-b.y,player.x-b.x),sp=Math.min(270,(160+stage*8)*d);
      for(let i=-1;i<=1;i++){const a=base+i*.55;
        addB(b.x,b.y,Math.cos(a)*sp,Math.sin(a)*sp,'#ffd166','dot',5,{h:1,life:3.6,turn:1.7+d*.25});}}break;}}
  if(b.patT<=0)nextPattern();}

function update(gdt,rdt){
  shake*=Math.exp(-6*rdt);if(shake<.1)shake=0;
  hitFlash=Math.max(0,hitFlash-2.4*rdt);bombFlash=Math.max(0,bombFlash-2.2*rdt);
  for(const s of stars){s.ny+=s.sp*rdt/H;if(s.ny>1)s.ny-=1;}
  for(let i=petals.length-1;i>=0;i--){const p=petals[i];
    p.x+=p.vx*rdt+Math.sin(tG*1.3+i)*8*rdt;p.y+=p.vy*rdt;p.rot+=p.vr*rdt;
    if(p.y>H+26||p.x<-40||p.x>W+40)petals[i]=petalNew(true);}
  if(banner){banner.t+=gdt;if(banner.t>=banner.dur)banner=null;}
  updParts(gdt);updTexts(gdt);

  if(state==='menu'){menuT-=rdt;
    if(menuT<=0){menuT=1.25;const cx=W*.5,cy=H*.3,n=18+((RND()*10)|0);
      for(let i=0;i<n;i++){const a=i/n*TAU+RND()*.25,sp=60+RND()*70;
        addB(cx,cy,Math.cos(a)*sp,Math.sin(a)*sp,i%3?'#ff8fb0':'#9be8ff','dot',4);}}
    updBullets(gdt,false);return;}
  if(state!=='play')return;

  updatePlayer(gdt);
  player.fireT-=gdt;while(player.fireT<=0){player.fireT+=.072;fireShots();if((shotN++&1)===0)sfx.shoot();}
  for(let i=pshots.length-1;i>=0;i--){const p=pshots[i];p.y+=p.vy*gdt;p.x+=p.vx*gdt;
    if(p.y<-30||p.x<-20||p.x>W+20){pshots[i]=pshots[pshots.length-1];pshots.pop();}}
  for(let i=pshots.length-1;i>=0;i--){const p=pshots[i];let hit=false;
    for(let j=enemies.length-1;j>=0&&!hit;j--){const e=enemies[j],dx=e.x-p.x,dy=e.y-p.y;
      if(dx*dx+dy*dy<(e.r+4)*(e.r+4)){e.hp--;spark(p.x,p.y,'#bfffd9');sfx.hit();hit=true;if(e.hp<=0)killEnemy(j);}}
    if(!hit&&boss&&!boss.dying){const R=bossR()*.85,dx=boss.x-p.x,dy=boss.y-p.y;
      if(dx*dx+dy*dy<R*R){bossDmg(1);spark(p.x,p.y,'#ffd7e4');hit=true;}}
    if(hit){pshots[i]=pshots[pshots.length-1];pshots.pop();}}
  updEnemies(gdt);
  phaseT+=gdt;
  if(phase==='enemies'){spawnT-=gdt;
    if(spawnT<=0&&enemies.length<7){formation();spawnT=Math.max(1.05,2.4-stage*.13);}
    if(phaseT>=enemyDur()){boss=makeBoss();phase='boss';banner(boss.name,'it blooms — fight');sfx.phase();}}
  else if(phase==='boss')updBoss(gdt);
  else if(phase==='clear'){for(let k2=0;k2<36&&bullets.length>0;k2++){const i=(RND()*bullets.length)|0;sparkle(bullets[i]);rmB(i);}
    if(phaseT>2.8){stage++;phase='enemies';phaseT=0;spawnT=1.7;banner('STAGE '+stage,STAGETAG[stage%STAGETAG.length]);}}
  updBullets(gdt,true);
  updItems(gdt);}
