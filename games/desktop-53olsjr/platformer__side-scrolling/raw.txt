  if(dir){P.vx=clamp(P.vx+dir*acc*h,-MOVE,MOVE)}
  else{const fr=(P.onGround?2500:430)*h;P.vx=P.vx>0?Math.max(0,P.vx-fr):Math.min(0,P.vx+fr)}
  P.coyote=P.onGround?COYOTE:Math.max(0,P.coyote-h);
  P.jbuf=Math.max(0,P.jbuf-h);
  if(P.jbuf>0&&P.coyote>0){P.vy=JV;P.onGround=false;P.coyote=0;P.jbuf=0;P.sq=.3;sfx.jump();
    puff(P.x+P.w/2,P.y+P.h,5,'180,195,235',{spd:70,g:260,life:.4,r:2})}
  P.glide=false;
  const hold=JUMPKEYS.some(k=>keys[k]);
  if(!P.onGround&&hold&&P.vy>30)P.glide=true;
  P.vy+=G*h;
  if(P.glide){if(P.vy>GLIDE_MAX)P.vy=lerp(P.vy,GLIDE_MAX,1-Math.pow(.02,h));
    if(Math.random()<h*26)puff(P.x+P.w/2-P.facing*8,P.y+P.h*.4,1,'190,215,255',{spd:30,g:-40,life:.5,r:1.8})}
  if(P.glide&&!wasGlide)sfx.glide();
  wasGlide=P.glide;
  // X
  P.x+=P.vx*h;
  for(const s of solids)if(aabb(P,s)){if(P.vx>0)P.x=s.x-P.w;else if(P.vx<0)P.x=s.x+s.w;P.vx=0}
  P.x=clamp(P.x,2,LEVEL_W-2-P.w);
  // Y
  const wasAir=!P.onGround,impact=P.vy;P.y+=P.vy*h;P.onGround=false;
  for(const s of solids)if(aabb(P,s)){
    if(P.vy>0){P.y=s.y-P.h;P.vy=0;P.onGround=true}
    else{P.y=s.y+s.h;P.vy=0}}
  if(wasAir&&P.onGround&&impact>250){
    P.sq=-clamp(impact/1900,0,.42);sfx.land(impact);
    puff(P.x+P.w/2,P.y+P.h,6+Math.floor(impact/220),'170,185,230',{spd:90,g:320,life:.45,r:2.2});
    if(impact>950)shake(2.5);
  }
}
function hurt(){
  if(P.inv>0)return;
  hearts--;sfx.hurt();shake(6);P.inv=1.6;P.vx=-P.facing*280;P.vy=-320;
  puff(P.x+P.w/2,P.y+P.h/2,16,'255,240,200',{spd:220,g:120,life:.7,r:2.4});
  if(hearts<=0){state='dead';deadT=1.15;P.spin=0}
}
function respawn(){
  P.x=spawnPt.x;P.y=spawnPt.y;P.vx=0;P.vy=0;P.inv=1.4;shake(3);
  toast('✦ back at the lantern');
}
function dieToScreen(){state='gameover';sfx.lose();
  const lines=['The meadow swallows your light…','A bat\u2019s shadow wins this time.','Even moths rest. Try again.','The dark is patient. You should be too.'];

  $('over-stat').textContent=fliesGot+' / '+fliesTotal;
  show('scr-over');blurActive();
}
function doWin(){
  state='win';winT=0;sfx.win();
  const nbT=best.t==null||runTime<best.t,nbF=best.f==null||fliesGot>best.f;
  if(nbT)best.t=runTime;if(nbF)best.f=fliesGot;saveBest();
  $('win-time').textContent=fmt(runTime);$('win-flies').textContent=fliesGot+' / '+fliesTotal;
  $('win-badge').classList.toggle('show',nbT||nbF);
  puff(goal.x,390,40,'255,215,106',{spd:260,g:-60,life:1.1,r:2.8});
}

/* ============================== update ============================== */
function update(dt){
  tG+=dt;
  flyPop=Math.max(0,flyPop-dt*3);toastT=Math.max(0,toastT-dt);
  shT=Math.max(0,shT-dt);if(shT<=0)shMag=0;
  P.sq=lerp(P.sq,0,1-Math.pow(.0004,dt));

  if(state==='menu'){
    const span=Math.max(1,LEVEL_W-viewW);
    const p=(tG*26)%(span*2);cam.x=p<span?p:span-p;
    ambChime-=dt;if(ambChime<=0){ambChime=.3+Math.random()*.4;
      puff(cam.x+Math.random()*viewW,VIEW_H-40-Math.random()*160,1,'255,215,120',{spd:14,g:-30,life:2.2,r:1.7})}
    updParts(dt);return;
  }
  if(state==='dead'){
    deadT-=dt;const s=dt*.4;
    P.vy+=G*s;P.y+=P.vy*s;P.spin+=dt*9;
    updParts(s);camFollow(dt*.5);
    if(deadT<=0)dieToScreen();return;
  }
  if(state==='win'){
    winT+=dt;P.alpha=Math.max(0,1-winT*1.4);
    if(Math.random()<dt*22){const cols=['255,215,106','126,230,168','150,190,255','255,150,170'];
      puff(goal.x+(Math.random()-.5)*140,330+Math.random()*120,2,cols[Math.floor(Math.random()*4)],{spd:160,g:-80,life:1,r:2.4})}
    updParts(dt);camFollow(dt);
    if(winT>1&&!winShown){winShown=true;show('scr-win');blurActive()}
    return;
  }
  if(state!=='playing')return;

  runTime+=dt;
  // physics substeps
  let rem=dt;const h=1/120;let n=0;
  while(rem>1e-6&&n<8){stepPlayer(Math.min(h,rem));rem-=h;n++}
  P.inv=Math.max(0,P.inv-dt);

  // run dust
  if(P.onGround&&Math.abs(P.vx)>190){dustT-=dt;
    if(dustT<=0){dustT=.06;puff(P.x+P.w/2-P.facing*8,P.y+P.h,1,'150,165,215',{spd:36,g:140,life:.35,r:1.8})}}

  // fireflies
  for(const f of FL){if(f.taken)continue;
    if(Math.abs(P.x+P.w/2-f.x)<27&&Math.abs(P.y+P.h/2-(f.y+Math.sin(tG*1.6+f.ph)*4))<29){
      f.taken=true;fliesGot++;flyPop=1;sfx.collect();
      puff(f.x,f.y,12,'255,215,120',{spd:170,g:-40,life:.7,r:2.2});}}

  // bats
  for(const b of bats){if(!b.alive){b.squish=Math.max(0,b.squish-dt);continue}
    const bx=b.x0+Math.sin(tG*b.sp+b.ph)*b.range,by=b.y+Math.sin(tG*2.3+b.ph)*9;
    if(P.inv<=0&&P.x<bx+17&&P.x+P.w>bx-17&&P.y<by+11&&P.y+P.h>by-11){
      if(P.vy>0&&(P.y+P.h)<by+4){ // stomp
        b.alive=false;b.squish=.4;P.vy=-460;sfx.stomp();shake(2.5);
        puff(bx,by,14,'150,130,220',{spd:190,g:480,life:.6,r:2.2});puff(bx,by,6,'255,215,120',{spd:120,g:300,life:.5,r:1.8});
      }else hurt();}}

  // spikes
  if(P.inv<=0)for(const sp of spikes){
    const r={x:sp.x+4,y:452,w:sp.w-8,h:18};
    if(aabb(P,r)){hurt();break}}

  // checkpoints
  for(let i=0;i<checkpoints.length;i++){const c=checkpoints[i];
    if(!c.lit&&Math.abs(P.x+P.w/2-c.x)<34&&Math.abs(P.y+P.h-c.y)<80){
      c.lit=true;cpIndex=i;Object.assign(spawnPt,{x:c.x-15,y:c.y-70});
      sfx.checkpoint();toast('✦ Checkpoint');shake(1.5);
      puff(c.x,c.y-42,20,'255,190,90',{spd:130,g:-60,life:.9,r:2.2});}}

  // lantern embers
  for(const c of checkpoints)if(c.lit&&Math.random()<dt*2.4)
    puff(c.x+(Math.random()-.5)*8,c.y-46,1,'255,170,70',{spd:12,g:-90,life:1,r:1.5});

  // goal sparkles
  if(Math.random()<dt*8)puff(goal.x+(Math.random()-.5)*90,320+Math.random()*120,1,'255,235,190',{spd:20,g:-50,life:.9,r:1.6});

  // goal
  if(aabb(P,goal.rect)){doWin();return}

  // fell out
  if(P.y>720){
    hearts--;sfx.hurt();shake(5);
    if(hearts<=0){state='dead';deadT=.9;P.spin=0}else respawn()}

  // ambient distant chime
  ambChime-=dt;if(ambChime<=0){ambChime=1.2+Math.random()*3;
    tone({f0:1300+Math.random()*900,d:1.4,g:.016});}

  updParts(dt);camFollow(dt);
}
function camFollow(dt){
  const max=Math.max(0,LEVEL_W-viewW);

  const target=clamp(P.x-viewW*.4+P.facing*55,0,max);
  cam.x=lerp(cam.x,target,1-Math.pow(.0008,dt));
}
