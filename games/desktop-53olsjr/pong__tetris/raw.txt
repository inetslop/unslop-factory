function predictY(side){
 const face = side==='L'?FACE_L:FACE_R;
 if(ball.vx===0) return null;
 const dxTo = (side==='L') ? (face - ball.x) : (ball.x - face);
 const v = (side==='L') ? ball.vx : -ball.vx; // distance rate
 if(v<=0 || dxTo<0) return null;
 const t=dxTo/v; if(t<0||t>4) return null;
 let y=ball.y+ball.vy*t;
 const lo=WALL+BALLR, hi=VH-WALL-BALLR, span=2*(hi-lo);
 let m=(y-lo)%span; if(m<0)m+=span; if(m>hi-lo) m=span-m; return lo+m;
}

/* ================= STAMPING (the tetris core) ================= */
function stamp(side,pieceId){
  const g=side==='L'?gridL:gridR, fr=side==='L'?frL:frR, b=side==='L'?birthL:birthR;
  const offs=PIECES[pieceId]; let placed=0, burst=false;
  for(const o of offs){
    const rr2=clamp(Math.round((ball.y-WALL)/CELL)+o[1],0,ROWS-1);
    let cc=fr[rr2]+1-o[0];
    if(cc<0) continue;
    if(cc>=COLS-1){ burst=true; break; }
    while(cc>0&&g[rr2][cc]) cc--;
    if(!g[rr2][cc]){ g[rr2][cc]=pieceId+1; fr[rr2]=Math.max(fr[rr2],cc); b[rr2][cc]=performance.now()/1000; placed++;
      addP(cellX(side,cc)+8, cellY(rr2)+8,(Math.random()*2-1)*40,-60-Math.random()*60,.5,PIECE_COLORS[pieceId],5); }
  }
  if(burst){ doBurst(side); return; }
  sStamp(placed); syncMeters();
}
function clearDeepest(side){
  const fr=side==='L'?frL:frR, g=side==='L'?gridL:gridR, b=side==='L'?birthL:birthR;
  let best=-1, br=-1; for(let r=0;r<ROWS;r++) if(fr[r]>best){ best=fr[r]; br=r; }
  if(br<0) return 0;
  let n=0; for(let c=0;c<COLS-1;c++){ if(g[br][c]){ const col=PIECE_COLORS[g[br][c]-1];

    g[br][c]=0; b[br][c]=0; n++; } }
  fr[br]=-1; sClear(); syncMeters(); return n;
}

function predictY(side){
  if(ball.vx===0) return null;
  const face=side==='L'?FACE_L:FACE_R;
  const dxTo = side==='L' ? face-ball.x : ball.x-face;
  const v = side==='L' ? ball.vx : -ball.vx;
  if(v<=0||dxTo<0) return null;
  const t=dxTo/v; if(t>4) return null;
  let y=ball.y+ball.vy*t;
  const lo=WALL+BALLR, hi=VH-WALL-BALLR, span=2*(hi-lo);
  let m=(y-lo)%span; if(m<0)m+=span; if(m>hi-lo)m=span-m; return lo+m;
}
function cpuMove(side,dt){
  const p=side==='L'?padL:padR, c=cpus[side];
  c.errT-=dt; if(c.errT<=0){ c.errT=.12; c.err=(Math.random()*2-1)*c.mag; }
  const heading = side==='L' ? ball.vx<0 : ball.vx>0;
  let target=VH/2, sp=c.speed;
  if(heading){ const pr=predictY(side); if(pr!=null) target=pr+c.err; }
  else target=p.y+(VH/2-p.y)*Math.min(1,dt*1.5);
  if(mode===0 && side==='R' && state!=='menu'){ // CPU plays smart: sacrifice a point to clear a row when in danger
    const danger=maxFrontier('R')>=COLS-2;
    if(heading && !c.approach){ c.approach=true; c.give=danger&&Math.random()<.55; }
    if(!heading) c.approach=false;
    if(c.give){ target=VH/2; sp*=.32; }
  }
  const d=target-p.y, step=Math.sign(d)*Math.min(Math.abs(d), sp*dt);
  p.y=clamp(p.y+step, WALL+p.h/2, VH-WALL-p.h/2);
}

function hit(side){

  stamp(side, ball.piece);                       // incoming piece gets stamped on your wall
  if(state!=='play'&&state!=='menu') return;     // burst ended the rally
  const rel=clamp((ball.y-p.y)/(p.h/2),-1,1);
  speed=Math.min(speed*1.05+6, MAX_SPEED);
  let ang=rel*MAXANG+(Math.random()*.08-.04);
  const dir=side==='L'?1:-1;
  ball.vx=Math.cos(ang)*speed*dir; ball.vy=Math.sin(ang)*speed;
  if(Math.abs(ball.vy)<speed*.14) ball.vy=(Math.random()<.5?-1:1)*speed*.16;
  ball.piece=randPiece();
  p.flash=1; rally++; (side==='L'?hitsL++:hitsR++);
  sparkFx(side==='L'?FACE_L:FACE_R, ball.y, colorOf(side), 10); sPaddle(speed);
  if(rally>=4){ rallyEl.textContent='RALLY ×'+rally; rallyEl.classList.remove('hidden','pop'); void rallyEl.offsetWidth; rallyEl.classList.add('pop'); }
}
function moveBall(dt){
  const steps=Math.max(1, Math.ceil((speed*dt)/18));
  for(let i=0;i<steps;i++){
    const pdt=dt/steps, px=ball.x;
    ball.x+=ball.vx*pdt; ball.y+=ball.vy*pdt;
    if(ball.y<WALL+BALLR){ ball.y=WALL+BALLR; ball.vy=Math.abs(ball.vy); sWall(); sparkFx(ball.x,WALL,'#cfe9ee',4); }
    else if(ball.y>VH-WALL-BALLR){ ball.y=VH-WALL-BALLR; ball.vy=-Math.abs(ball.vy); sWall(); sparkFx(ball.x,VH-WALL,'#cfe9ee',4); }
    if(ball.vx<0){ const pe=px-BALLR, ne=ball.x-BALLR;
      if(pe>=FACE_L&&ne<FACE_L && Math.abs(ball.y-padL.y)<=padL.h/2+10){ ball.x=FACE_L+BALLR; hit('L'); } }
    else if(ball.vx>0){ const pe=px+BALLR, ne=ball.x+BALLR;
      if(pe<=FACE_R&&ne>FACE_R && Math.abs(ball.y-padR.y)<=padR.h/2+10){ ball.x=FACE_R-BALLR; hit('R'); } }
    if(state!=='play'&&state!=='menu') break;
  }
  ball.rot+=ball.spin*dt*(ball.vx<0?-1:1);
  ball.trail.push({x:ball.x,y:ball.y}); if(ball.trail.length>16) ball.trail.shift();
}
function checkCross(){
  if(ball.x<-50||ball.x>VW+50){
    const scorer=ball.x<0?'R':'L';
    if(state==='menu'){
      if(maxFrontier('L')>=COLS-3) clearDeepest('L');
      if(maxFrontier('R')>=COLS-3) clearDeepest('R');
      serve(Math.random()<.5?-1:1);
    } else point(scorer);
  }
}

/* ================= UPDATE ================= */
function update(dt){
  if(state==='pause') return;
  shake=Math.max(0,shake-dt*30); flash=Math.max(0,flash-dt*2.2);
  padL.flash=Math.max(0,padL.flash-dt*4); padR.flash=Math.max(0,padR.flash-dt*4);
  banner.t+=dt; if(banner.t>banner.max) banner.max=0;
  updateParticles(dt);
  if(state==='menu'){ setCPUParams(); cpuMove('L',dt); cpuMove('R',dt); moveBall(dt); checkCross(); }
  else if(state==='count'){ stateT+=dt;
    const n=3-Math.floor(stateT/.45);
    if(n!==lastCount&&n>=1){ lastCount=n; sCount(); }
    if(stateT>=1.35){ serve(serveDir); state='play'; sGo(); bannerSet('GO','','#eefcff',.45); } }
  else if(state==='play'){
    humanMove('L',dt);
    if(mode===0){ setCPUParams(); cpuMove('R',dt); } else humanMove('R',dt);
    moveBall(dt); checkCross(); }
  else if(state==='point'){ stateT+=dt;
    if(stateT>=pointDur){ if(pendingOver) showOver(); else { serve(serveDir); state='play'; bannerSet('','','#fff',0); } } }
}

function draw(t,dtA){

  const W=innerWidth,H=innerHeight;
  // layered ambient background

  ctx.fillStyle=g; ctx.fillRect(0,0,W,H);
  const wx=x=>OX+x*SC, wy=y=>OY+y*SC;
  for(const s of [['L',C_AQUA,FACE_L],['R',C_AMBER,FACE_R]]){
