function audio(){if(muted)return null;if(!AC){try{AC=new (window.AudioContext||window.webkitAudioContext)()}catch(e){return null}}if(AC.state==='suspended')AC.resume();return AC}
function tone(f,o={}){const a=audio();if(!a)return;const{dur=.15,type='sine',vol=.2,slide=0,delay=0}=o;const t=a.currentTime+delay;const osc=a.createOscillator(),g=a.createGain();osc.type=type;osc.frequency.setValueAtTime(f,t);if(slide)osc.frequency.exponentialRampToValueAtTime(Math.max(30,f+slide),t+dur);g.gain.setValueAtTime(0,t);g.gain.linearRampToValueAtTime(vol,t+.006);g.gain.exponentialRampToValueAtTime(.0001,t+dur);osc.connect(g).connect(a.destination);osc.start(t);osc.stop(t+dur+.05)}
function noiseS(o={}){const a=audio();if(!a)return;const{dur=.2,vol=.25,f=800,q=1,type='bandpass',delay=0}=o;const t=a.currentTime+delay;const n=Math.max(1,Math.floor(a.sampleRate*dur));const buf=a.createBuffer(1,n,a.sampleRate);const d=buf.getChannelData(0);for(let i=0;i<n;i++)d[i]=(Math.random()*2-1)*(1-i/n);const src=a.createBufferSource();src.buffer=buf;const fl=a.createBiquadFilter();fl.type=type;fl.frequency.value=f;fl.Q.value=q;const g=a.createGain();g.gain.value=vol;src.connect(fl).connect(g).connect(a.destination);src.start(t)}

function shuffle(a){for(let i=a.length-1;i>0;i--){const j=Math.floor(Math.random()*(i+1));[a[i],a[j]]=[a[j],a[i]]}return a}
function newDeck(){const d=[];for(const[s,cls]of SUITS)for(const r of RANKS)d.push({r,s,cls});return shuffle(d)}
function draw(n){if(!S.deck.length)S.deck=newDeck();const h=[];while(n--)h.push(S.deck.pop());return h}

function startHole(i){S.hole=i;const h=S.holes[i];
 S.deck=newDeck();S.hand=draw(5);S.strokes=0;S.remaining=h.dist;S.onGreen=false;
 S.phase='idle';S.flight=null;S.sink=null;trail.length=0;ripples.length=0;

function playCard(i){if(S.screen!=='playing'||S.phase!=='idle')return;
 const isP=i===5;if(!isP&&i>=S.hand.length)return;
 if(isP&&!S.onGreen)return;
 const h=S.holes[S.hole];
 const eff=isP?Math.round(S.remaining):YARDS[S.hand[i].r]+h.wind;
 if(!isP&&eff>S.remaining){badCard(i);return}
 const token=S.run;
 S.strokes++;sfxSwing();
 const el=cardEls[i];if(el)el.classList.add('away');
 if(!isP)S.hand.splice(i,1);
 S.phase='fly';updateHUD();
 setTimeout(()=>{if(token!==S.run)return;
   const p0=h.dist-S.remaining;
   S.flight={p0,p1:p0+eff,dur:isP?520:clamp(450+eff*1.3,500,1350),arc:isP?3:clamp(26+eff*.28,30,150),putt:isP,el:0};
   trail.length=0;
 },270)}

function drawHand(){if(S.screen!=='playing'||S.phase!=='idle')return;

function landShot(){const f=S.flight;S.flight=null;
 if(f.putt){doSink();return}
 S.remaining-=f.eff;
 const g=geom(),h=S.holes[S.hole];
 puff(g.teeX+(h.dist-S.remaining)/h.dist*g.len,g.fy-2);sfxLand();
 if(S.remaining<=LIP){doSink()}
 else if(S.remaining<=GREEN){S.onGreen=true;S.flagPulse=1;renderHand(true);updateHUD();
   tone(392,{dur:.1,vol:.14,type:'triangle'});tone(523,{dur:.14,delay:.08,vol:.14,type:'triangle'});
   toast('ON THE GREEN — play the PUTT card (P)');S.phase='idle'}
 else{S.hand.push(draw(1)[0]);renderHand(false);updateHUD();S.phase='idle'}}

function doSink(){S.phase='sink';S.sink={t:0,done:false};
