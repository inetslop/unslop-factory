    const d=nbuf.getChannelData(0);for(let i=0;i<d.length;i++)d[i]=Math.random()*2-1;
  }catch(e){AC=null;}}
  if(AC&&AC.state==='suspended')AC.resume();
}
function tone(f,dur,type,vol,at,slideTo){
  if(!AC||muted)return;
  const t=AC.currentTime+(at||0);
  const o=AC.createOscillator(),g=AC.createGain();
  o.type=type;o.frequency.setValueAtTime(f,t);
  if(slideTo)o.frequency.exponentialRampToValueAtTime(Math.max(30,slideTo),t+dur);
  g.gain.setValueAtTime(vol,t);g.gain.exponentialRampToValueAtTime(.001,t+dur);
  o.connect(g);g.connect(master);o.start(t);o.stop(t+dur+.03);
}
function noiz(dur,vol,fq,at){
  if(!AC||muted)return;
  const t=AC.currentTime+(at||0);
  const s=AC.createBufferSource();s.buffer=nbuf;

  const g=AC.createGain();g.gain.setValueAtTime(vol,t);g.gain.exponentialRampToValueAtTime(.001,t+dur);
  s.connect(f);f.connect(g);g.connect(master);s.start(t);s.stop(t+dur+.03);
}

/* ================= controllers ================= */
function humanMove(p,dt){
  let dir=0;
  if(p.side===-1){if(keys['w'])dir--;if(keys['s'])dir++;}
  else{if(keys['ArrowUp'])dir--;if(keys['ArrowDown'])dir++;}
  let rawV=0;
  if(dir!==0){p.y+=dir*PSPEED*dt;rawV=dir*PSPEED;if(p.side===-1)mouseY=p.y;}
  else if(p.side===-1&&mode==='solo'&&mouseY!=null){const mv=clamp(mouseY-p.y,-MSPEED*dt,MSPEED*dt);p.y+=mv;rawV=mv/dt;}
  p.y=clamp(p.y,PH/2+8,CH-PH/2-8);
  p.vy+=(rawV-p.vy)*(1-Math.exp(-24*dt));
}
function predictY(p){
  const plane=p.side===1?padR.x:padL.x+PW;
  let x=ball.x,y=ball.y,vx=ball.vx,vy=ball.vy,sp=ball.spin,t=0;const h=1/60;
  while(t<3.5){
    const s=Math.hypot(vx,vy)||1;
    vx+=SPINK*sp*(-vy/s)*h;vy+=SPINK*sp*(vx/s)*h;
    x+=vx*h;y+=vy*h;sp*=Math.exp(-SPINDEC*h);t+=h;
    if(p.side===1&&x+R>=plane)return clamp(y,PH/2,CH-PH/2);
    if(p.side===-1&&x-R<=plane)return clamp(y,PH/2,CH-PH/2);
    if(y<R){y=R;vy=-vy;}else if(y>CH-R){y=CH-R;vy=-vy;}
  }
  return CH/2;
}
function aiMove(p,dt,cfg){
  const toward=p.side===1?ball.vx>0:ball.vx<0;
  if(toward&&!p._wasToward)p.err=rand(-cfg.err,cfg.err);
  p._wasToward=toward;
  let target=CH/2+Math.sin(time*1.5+p.side*2)*42*cfg.bob;
  if(toward&&state===SPLAY){
    const py=cfg.predict?predictY(p):ball.y+ball.vy*0.1;
    target=clamp(py+p.err,PH/2+6,CH-PH/2-6);
  }
  const mv=clamp(target-p.y,-cfg.speed*dt,cfg.speed*dt);
  p.y=clamp(p.y+mv,PH/2+8,CH-PH/2-8);
  p.vy+=(mv/dt-p.vy)*(1-Math.exp(-16*dt));
}

/* ================= physics ================= */
function launch(){
  state=SPLAY;
  const a=rand(-0.42,0.42);
  ball.vx=Math.cos(a)*BASE*serveDir;ball.vy=Math.sin(a)*BASE;
  ball.spin=0;rally=0;trail.length=0;
  sfx.serve();
}
function hitPad(p,dir,face){
  const off=clamp((ball.y-p.y)/(PH/2),-1,1);
  const ang=off*MAXANG;
  let sp=Math.min(MAXS,Math.hypot(ball.vx,ball.vy)*GAIN+Math.abs(p.vy)*0.05);
  sp=Math.max(sp,BASE*0.95);
  if(Math.abs(off)>0.82)sp=Math.min(MAXS,sp*1.06);           // sharp-edge bonus
  ball.vx=Math.cos(ang)*sp*dir;
  ball.vy=Math.sin(ang)*sp;
  ball.x=dir>0?face+R:face-R;
  const flick=Math.abs(p.vy)>430;
  ball.spin=clamp(p.vy*SPIN_TR*dir-ball.spin*0.18,-SPIN_MAX,SPIN_MAX);
  if(Math.abs(off)>0.82)ball.spin=clamp(ball.spin*1.3,-SPIN_MAX,SPIN_MAX);
  rally++;rallyPulse=1;
  if(mode!=='attract'){stats.rally=Math.max(stats.rally,rally);stats.top=Math.max(stats.top,sp);if(flick)stats.spin++;}
  const c=p.side===-1?COL.p1:COL.p2;
  hitFx(face,c,sp);sfx.hit(sp);
  shake=Math.min(18,shake+(sp>1150?5:2));
}
function wallFx(){
  rings.push({x:ball.x,y:ball.y,r:R,vr:380,t:0,life:.3,c:'rgba(170,205,230,1)'});
  spawnBurst(ball.x,ball.y,['#a9cddd','#ffffff'],6,300);
  sfx.wall();shake=Math.min(shake+1.2,18);
}
function hitFx(x,c,sp){
  rings.push({x,y:ball.y,r:R+2,vr:560,t:0,life:.34,c});
  spawnBurst(x,ball.y,[c,'#ffffff'],(sp>1100?18:10),430);
}
function integrateBall(dt){
  const spd=Math.hypot(ball.vx,ball.vy)||1;
  const n=clamp(Math.ceil(spd*dt/(R*0.9)),1,6);   // substeps prevent tunnelling
  const h=dt/n;
  for(let i=0;i<n;i++){
    const s=Math.hypot(ball.vx,ball.vy)||1;
    const ax=SPINK*ball.spin*(-ball.vy/s),ay=SPINK*ball.spin*(ball.vx/s); // Magnus: curves without changing speed
    ball.vx+=ax*h;ball.vy+=ay*h;
    ball.x+=ball.vx*h;ball.y+=ball.vy*h;
    ball.spin*=Math.exp(-SPINDEC*h);
    ball.rot+=ball.spin*ROTK*h;
    if(ball.y<R){ball.y=R;if(ball.vy<0){ball.vy=-ball.vy;wallFx();}}
    else if(ball.y>CH-R){ball.y=CH-R;if(ball.vy>0){ball.vy=-ball.vy;wallFx();}}
    const lFace=padL.x+PW;
    if(ball.vx<0&&ball.x-R<=lFace&&ball.x>=padL.x-30&&Math.abs(ball.y-padL.y)<=PH/2+R)hitPad(padL,1,lFace);
    else if(ball.vx>0){
      const rFace=padR.x;
      if(ball.x+R>=rFace&&ball.x<=padR.x+PW+30&&Math.abs(ball.y-padR.y)<=PH/2+R)hitPad(padR,-1,rFace);
    }
  }
  if(ball.x<-R*4)point('R');
  else if(ball.x>CW+R*4)point('L');
}
function spawnBurst(x,y,cols,n,pow){
  for(let i=0;i<n;i++){const a=rand(0,TAU),s=rand(pow*.25,pow);
    parts.push({x,y,vx:Math.cos(a)*s,vy:Math.sin(a)*s,t:0,life:rand(.5,1.3),c:pick(cols),sz:rand(2,5)});}
  if(parts.length>420)parts.splice(0,parts.length-420);
}
function pushTrail(){trail.push({x:ball.x,y:ball.y});if(trail.length>16)trail.shift();}

/* ================= fixed-step simulation ================= */
function step(dt){
  time+=dt;
  if(state!==SOVER){
    if(mode==='attract'){aiMove(padL,dt,AICHILL);aiMove(padR,dt,AISHARP);}
    else if(mode==='solo'){humanMove(padL,dt);aiMove(padR,dt,AICFG[diff]);}
    else{humanMove(padL,dt);humanMove(padR,dt);}
    if(state===SSERVE){serveT-=dt;if(serveT<=0)launch();}
    else if(state===SPLAY){integrateBall(dt);pushTrail();}
    else if(state===SPOINT){
      ball.x+=ball.vx*dt;ball.y+=ball.vy*dt;ptT-=dt;
      if(ptT<=0){
        if(scoreL>=WIN||scoreR>=WIN)gameOver();
        else{resetBall();state=SSERVE;serveT=1.05;}
      }
    }
  }
  if(flash.t>0)flash.t=Math.max(0,flash.t-dt*1.3);
  if(pop.t>0)pop.t=Math.max(0,pop.t-dt*1.4);
  if(rallyPulse>0)rallyPulse=Math.max(0,rallyPulse-dt*2.5);
  for(let i=parts.length-1;i>=0;i--){const q=parts[i];q.t+=dt;
    if(q.t>=q.life){parts.splice(i,1);continue;}
    const d=Math.exp(-1.7*dt);q.vx*=d;q.vy*=d;q.x+=q.vx*dt;q.y+=q.vy*dt;}
  for(let i=rings.length-1;i>=0;i--){const q=rings[i];q.t+=dt;
    if(q.t>=q.life)rings.splice(i,1);else q.r+=q.vr*dt;}
}
