function update(dt){time+=dt;shake=Math.max(0,shake-22*dt);exitFlash=Math.max(0,exitFlash-dt*1.6);

 if(state!=='playing')return;

  if(e.slowT>0)e.slowT-=dt;
  const spd=e.d.spd*spdScale*(e.slowT>0?0.5:1);e.dist+=spd*dt;
  if(e.dist>=PLEN){enemies.splice(i,1);leak(e);continue;}
  const p=pathPos(e.dist);const dx=p.x-e.x,dz=p.z-e.z,dd=Math.hypot(dx,dz);
  if(dd>0.35){e.seg=p.ang;e.x+=dx/dd*Math.min(spd*dt,dd);e.z+=dz/dd*Math.min(spd*dt,dd);}else{e.x=p.x;e.z=p.z;}
 }
 // bullets
 for(let i=bullets.length-1;i>=0;i--){const b=bullets[i];b.px=b.x;b.py=b.y;b.pz=b.z;
  let ex=b.end.x,ez=b.end.z;if(b.tgt&&enemies.includes(b.tgt)){ex=b.tgt.x;ez=b.tgt.z;b.end={x:ex,z:ez};}
  const dx=ex-b.x,dz=ez-b.z,dd=Math.hypot(dx,dz),step=b.spd*dt;
  if(dd<=step+0.12){if(b.tgt&&enemies.includes(b.tgt))damage(b.tgt,b.dmg);puff(ex,0.4,ez,4,b.col,1.6,0.04,0.25,0,true);bullets.splice(i,1);}
  else{b.x+=dx/dd*step;b.z+=dz/dd*step;}}
 // shells

  if(parts.length<420)parts.push({x:s.x0+(s.x1-s.x0)*(s.t/s.T),y:0.5+Math.sin(Math.PI*s.t/s.T)*1.9,z:s.z0+(s.z1-s.z0)*(s.t/s.T),

   for(let k=enemies.length-1;k>=0;k--){const e=enemies[k],dx=e.x-s.x1,dz=e.z-s.z1,dd2=dx*dx+dz*dz;
    if(dd2<=s.splash*s.splash)damage(e,s.dmg*(1-0.5*Math.sqrt(dd2)/s.splash));}
   SFX.impact();shells.splice(i,1);}}
 updateTowers(dt);
 // fx timers
 for(let i=ringsFx.length-1;i>=0;i--){ringsFx[i].t+=dt;if(ringsFx[i].t>0.45)ringsFx.splice(i,1);}
 for(let i=beams.length-1;i>=0;i--){beams[i].ttl-=dt;if(beams[i].ttl<=0)beams.splice(i,1);}

  p.vy+=p.g*dt;p.x+=p.vx*dt;p.y+=p.vy*dt;p.z+=p.vz*dt;if(p.y<0.02){p.y=0.02;p.vy*=-0.3;}}
