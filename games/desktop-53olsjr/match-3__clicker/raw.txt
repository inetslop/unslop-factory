 const s=swapAnim;if(!s){phase='idle';return;}
 s.t=Math.min(1,s.t+dt/.16);
 const e=s.t<.5?2*s.t*s.t:1-Math.pow(-2*s.t+2,2)/2;
 s.items.forEach(it=>{it.g.x=it.fx+(it.tx-it.fx)*e;it.g.y=it.fy+(it.ty-it.fy)*e;});
 if(s.t>=1){if(s.back)finishRevert();else finishSwap();}
}
function finishRevert(){
 const s=swapAnim,A=s.a,B=s.b;
 grid[A.r][A.c]=s.gemB;grid[B.r][B.c]=s.gemA;
 s.gemA.r=A.r;s.gemA.c=A.c;s.gemB.r=B.r;s.gemB.c=B.c;
 swapAnim=null;phase='idle';
}
function finishSwap(){
 const s=swapAnim,A=s.a,B=s.b;
 if(s.fx){handlePrismFx(s.fx);return;}
 const gs=findLines().filter(g=>g.cells.some(p=>(p.r===A.r&&p.c===A.c)||(p.r===B.r&&p.c===B.c)));
 if(!gs.length){
  grid[A.r][A.c]=s.gemB;grid[B.r][B.c]=s.gemA;
  s.gemA.r=A.r;s.gemA.c=A.c;s.gemB.r=B.r;s.gemB.c=B.c;
  swapAnim={t:0,back:true,a:A,b:B,gemA:s.gemA,gemB:s.gemB,fx:null,
   items:[{g:s.gemA,fx:s.gemA.x,fy:s.gemA.y,tx:A.c,ty:A.r},{g:s.gemB,fx:s.gemB.x,fy:s.gemB.y,tx:B.c,ty:B.r}]};
  sNope();return;
 }
 const spAt={};
 const h4=gs.filter(g=>g.dir==='h'&&g.len>=4),v4=gs.filter(g=>g.dir==='v'&&g.len>=4);
 for(const gh of h4)for(const gv of v4){const it=gh.cells.find(p=>gv.cells.some(q=>q.r===p.r&&q.c===p.c));if(it)spAt[it.r+','+it.c]=3;}
 gs.forEach(g=>{
  let cell=null;
  for(const p of g.cells)if((p.r===A.r&&p.c===A.c)||(p.r===B.r&&p.c===B.c))cell=p;
  if(!cell)cell=g.cells[Math.floor(g.len/2)];
  const sp=g.len>=5?2:(g.len===4?1:0);
  if(sp){const k=cell.r+','+cell.c;spAt[k]=Math.max(spAt[k]||0,sp);}
 });
 const list=[];
 gs.forEach(g=>g.cells.forEach(p=>{if(!spAt[p.r+','+p.c])list.push(p);}));
 survivors=Object.entries(spAt).map(([k,sp])=>{const[r,c]=k.split(',').map(Number);return{r,c,sp,t:grid[r][c].t};});
 if(!startPops(list)){survivors=null;phase='idle';}
 swapAnim=null;
}
function handlePrismFx(fx){

 if(other.sp===2){for(let r=0;r<N;r++)for(let c=0;c<N;c++){const g=grid[r][c];if(g&&g!==prism)list.push({r,c});}}
 else{const t=other.t;for(let r=0;r<N;r++)for(let c=0;c<N;c++){const g=grid[r][c];if(g&&g!==prism&&g.t===t)list.push({r,c});}}
 if(!startPops(list))phase='idle';
}

function updateFall(dt){

  if(g.y<g.r-1e-4){done=false;g.vy+=38*dt;g.y+=g.vy*dt;
   if(g.y>=g.r){g.y=g.r;if(g.vy>6){g.vy*=-.25;g.squash=1;}else g.vy=0;}}
  else if(Math.abs(g.x-g.c)>1e-4){done=false;g.x+=(g.c-g.x)*Math.min(1,dt*18);}
  g.squash=Math.max(0,g.squash-dt*5);
  g.born=Math.min(1,g.born+dt*3.2);
 }
 if(done){phase='idle';afterSettle();}
}
