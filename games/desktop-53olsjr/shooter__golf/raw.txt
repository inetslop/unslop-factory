function collideRect(b, r){ // r {x,y,w,h} top-left
  const cx=clamp(b.x,r.x,r.x+r.w), cy=clamp(b.y,r.y,r.y+r.h);
  let dx=b.x-cx, dy=b.y-cy; const d2=dx*dx+dy*dy;
  if(d2 > b.r*b.r) return 0;
  let d=Math.sqrt(d2);
  if(d<0.0001){ // center inside rect: push out along min penetration axis
     const l=b.x-r.x, rr=r.x+r.w-b.x, t=b.y-r.y, bb=r.y+r.h-b.y;
     const m=Math.min(l,rr,t,bb);
     if(m===l){b.x=r.x-b.r; b.vx=-Math.abs(b.vx)*REST;} else if(m===rr){b.x=r.x+r.w+b.r;b.vx=Math.abs(b.vx)*REST;} else if(m===t){b.y=r.y-b.r;b.vy=-Math.abs(b.vy)*REST;} else {b.y=r.y+r.h+b.r;b.vy=Math.abs(b.vy)*REST;}
     return Math.hypot(b.vx,b.vy);
  }
  const nx=dx/d, ny=dy/d; b.x=cx+nx*b.r; b.y=cy+ny*b.r;
  const vn=b.vx*nx+b.vy*ny; if(vn<0){ b.vx-=(1+REST)*vn*nx; b.vy-=(1+REST)*vn*ny; return -vn;}
  return 0;
}

function solids(){
  const s=[{x:0,y:0,w:VW,h:BOUND,ref:null},{x:0,y:VH-BOUND,w:VW,h:BOUND,ref:null},
           {x:0,y:0,w:BOUND,h:VH,ref:null},{x:VW-BOUND,y:0,w:BOUND,h:VH,ref:null}];

  for(const c of G.crates)if(c.hp>0)s.push({x:c.x,y:c.y,w:c.w,h:c.h,ref:c,crate:true});
  for(const m of G.mowers)s.push({x:m.x,y:m.y,w:m.w,h:m.h,ref:m,mower:true});
  return s;
}
function collideRect(b,r,rest){
  const cx=clamp(b.x,r.x,r.x+r.w),cy=clamp(b.y,r.y,r.y+r.h);
  let dx=b.x-cx,dy=b.y-cy;const d2=dx*dx+dy*dy;
  if(d2>b.r*b.r)return 0;
  const sp=Math.hypot(b.vx,b.vy);let d=Math.sqrt(d2);
  if(d<.0001){
    const l=b.x-r.x,rt=r.x+r.w-b.x,t=b.y-r.y,bb=r.y+r.h-b.y,m=Math.min(l,rt,t,bb);
    if(m===l){b.x=r.x-b.r;if(b.vx>0)b.vx=-b.vx*rest;}else if(m===rt){b.x=r.x+r.w+b.r;if(b.vx<0)b.vx=-b.vx*rest;}
    else if(m===t){b.y=r.y-b.r;if(b.vy>0)b.vy=-b.vy*rest;}else{b.y=r.y+r.h+b.r;if(b.vy<0)b.vy=-b.vy*rest;}
    return sp;
  }
  const nx=dx/d,ny=dy/d;b.x=cx+nx*b.r;b.y=cy+ny*b.r;
  const vn=b.vx*nx+b.vy*ny;
  if(vn<0){b.vx-=(1+rest)*vn*nx;b.vy-=(1+rest)*vn*ny;return -vn;}
  return 0;
}

/* ================= update ================= */
function update(dt){
  time+=dt;shake=Math.max(0,shake-dt*26);
  if(state==='menu'||state==='play'){
    /* mowers */
    for(const m of G.mowers){const mid=(m.min+m.max)/2,amp=(m.max-m.min)/2,px=m.x,py=m.y,p=mid+amp*Math.sin(time*m.speed+m.phase);
      if(m.axis==='y')m.y=p;else m.x=p;m.vx=(m.x-px)/Math.max(dt,.001);m.vy=(m.y-py)/Math.max(dt,.001);}
    /* turrets */
    for(const t of G.turrets){if(t.hp<=0)continue;t.flash=Math.max(0,t.flash-dt);
      const bs=Math.hypot(ball.vx,ball.vy),bd=Math.hypot(ball.x-t.x,ball.y-t.y);
      let want=t.base+Math.sin(time*.7+t.x)*.25,engaged=false;
      if(state==='play'&&ball.alive&&!ball.sunk&&bs>60&&bd<340){const tt=bd/380;want=Math.atan2(ball.y+ball.vy*tt-t.y,ball.x+ball.vx*tt-t.x);engaged=true;}
      const da=normAng(want-t.a);t.a+=clamp(da,-3*dt,3*dt);
      if(engaged){t.cd-=dt;if(t.cd<=0&&Math.abs(normAng(want-t.a))<.2)fireTurret(t);}else t.cd=Math.max(t.cd-dt*.5,.3);
    }
  }
  if(state==='play'){
    /* charging */
    if(charging){charge=Math.min(1,charge+dt/1.05);if(chOsc)chOsc.frequency.value=120+charge*480;}
    /* blaster auto-fire */
    fireCd-=dt;
    if(mouse.rmb&&ball.alive&&!ball.sunk&&fireCd<=0){shoot();fireCd=.16;}
    /* ball */
    if(ball.alive&&!ball.sunk){
      const sol=solids(),steps=3,dt2=dt/steps;
      for(let i=0;i<steps;i++){
        ball.x+=ball.vx*dt2;ball.y+=ball.vy*dt2;
        for(const r of sol){
          const imp=collideRect(ball,r,r.crate?.45:.8);
          if(imp>90&&!r.mower){sWall(Math.min(1,imp/700));burst(ball.x,ball.y,4,{spd:120,life:.3,size:2.5,colors:['#5cd65c','#a9e5b1'],shape:'c'});shake=Math.max(shake,Math.min(6,imp/140));}
          if(r.mower&&imp>0){ball.vx+=r.ref.vx*.55;ball.vy+=r.ref.vy*.55;}
          if(r.crate&&imp>50){r.ref.hp--;burst(ball.x,ball.y,6,{spd:180,life:.4,size:3,colors:['#c98a3d'],shape:'s'});sCrate();if(r.ref.hp<=0)crateBoom(r.ref);}
        }
      }
      /* friction zones */
      let k=1.7;
      for(const z of G.ice)if(ball.x>z[0]&&ball.x<z[0]+z[2]&&ball.y>z[1]&&ball.y<z[1]+z[3]){k=.28;break;}
      if(k===1.7)for(const z of G.sand)if(ball.x>z[0]&&ball.x<z[0]+z[2]&&ball.y>z[1]&&ball.y<z[1]+z[3]){k=4.6;break;}
      const f=Math.exp(-k*dt);ball.vx*=f;ball.vy*=f;
      /* cup */
      const dx=G.cup.x-ball.x,dy=G.cup.y-ball.y,d=Math.hypot(dx,dy),sp=Math.hypot(ball.vx,ball.vy);
      if(d<26&&d>.01&&sp<210){const pull=520*(1-d/26);ball.vx+=dx/d*pull*dt;ball.vy+=dy/d*pull*dt;}
      if(d<CUP_R+BR-4&&sp>1){
        if(sp<235)sink();
        else{const nx=dx/d,ny=dy/d,vn=ball.vx*nx+ball.vy*ny;ball.vx-=1.45*vn*nx;ball.vy-=1.45*vn*ny;ball.x=G.cup.x-nx*(CUP_R+BR-3);ball.y=G.cup.y-ny*(CUP_R+BR-3);sLip();addFloat(ball.x,ball.y-20,'LIPPED!','#ffd43b',15);}
      }
      /* water */
      for(const w of G.water)if(circleRect(ball.x,ball.y,BR,w)){splash();break;}
      /* stop / trail / cap */
      const sp2=Math.hypot(ball.vx,ball.vy);
      if(sp2>1100){const s=1100/sp2;ball.vx*=s;ball.vy*=s;}
      if(sp2<8){ball.vx=ball.vy=0;ball.rest={x:ball.x,y:ball.y};}
      if(sp2>140){ball.trail.push({x:ball.x,y:ball.y,t:0});if(ball.trail.length>26)ball.trail.shift();}
    }else{for(const tr of ball.trail)tr.t+=dt;}
    while(ball.trail.length&&ball.trail[0].t>.25)ball.trail.shift();
    /* respawn */
    if(!ball.alive&&!ball.sunk){respawnT-=dt;if(respawnT<=0){ball.x=ball.rest.x;ball.y=ball.rest.y;ball.vx=ball.vy=0;ball.alive=true;ball.alpha=0;ring(ball.x,ball.y,'rgba(255,255,255,.7)',24);}}
    ball.alpha=Math.min(1,ball.alpha+dt*3);
    /* sunk animation */
    if(ball.sunk){
      ball.sinkT+=dt;const t=clamp(ball.sinkT/.55,0,1);
      const r0=Math.hypot(ball.sx-G.cup.x,ball.sy-G.cup.y),a0=Math.atan2(ball.sy-G.cup.y,ball.sx-G.cup.x);
      ball.x=G.cup.x+Math.cos(a0+t*9)*r0*(1-t);ball.y=G.cup.y+Math.sin(a0+t*9)*r0*(1-t);
      if(t>=1&&!ball.confDone){ball.confDone=true;burst(G.cup.x,G.cup.y-20,40,{spd:340,life:.9,size:4,colors:['#ffd43b','#69db7c','#ff8787','#74c0fc','#ffa94d'],shape:'s'});shake=Math.max(shake,5);}
      if(ball.sinkT>=.7&&!bannerShown){const[w,c]=resWord(strokes-G.par);showBanner(w,c);bannerShown=true;}
      if(ball.sinkT>=advanceT){
        holeIdx++;
        if(holeIdx>=HOLES.length)showFinal();
        else{loadHole(holeIdx);hud();showBanner('HOLE '+(holeIdx+1)+' · '+G.name,'b-title');}
      }
    }
    /* player bullets */
    const sol=solids();
    for(let i=B.length-1;i>=0;i--){const b=B[i];b.x+=b.vx*dt;b.y+=b.vy*dt;b.life-=dt;let dead=b.life<=0;
      if(!dead)for(const r of sol)if(circleRect(b.x,b.y,3,r)){burst(b.x,b.y,4,{spd:150,life:.2,size:2,colors:['#ffe08a'],shape:'c'});noiseBurst(.03,.06,2500,700);dead=true;break;}
      if(!dead)for(const c of G.crates)if(c.hp>0&&circleRect(b.x,b.y,5,c)){c.hp--;burst(b.x,b.y,5,{spd:170,life:.35,size:3,colors:['#c98a3d'],shape:'s'});sCrate();if(c.hp<=0)crateBoom(c);dead=true;break;}
      if(!dead)for(const t of G.turrets)if(t.hp>0&&Math.hypot(b.x-t.x,b.y-t.y)<19){t.hp--;t.flash=.15;burst(b.x,b.y,6,{spd:200,life:.3,size:2.5,colors:['#ffb020','#ffd43b'],shape:'c'});tone(240,130,.07,'square',.22);if(t.hp<=0)turretDown(t);dead=true;break;}
      if(dead)B.splice(i,1);
    }
    /* turret bullets */
    for(let i=TB.length-1;i>=0;i--){const b=TB[i];b.x+=b.vx*dt;b.y+=b.vy*dt;b.life-=dt;let dead=b.life<=0;
      if(!dead)for(const r of sol)if(circleRect(b.x,b.y,3,r)){burst(b.x,b.y,4,{spd:140,life:.2,size:2,colors:['#ff8787'],shape:'c'});dead=true;break;}
      if(!dead&&ball.alive&&!ball.sunk&&Math.hypot(b.x-ball.x,b.y-ball.y)<BR+3){
        ball.vx+=b.vx/380*160;ball.vy+=b.vy/380*160;sHitBall();shake=Math.max(shake,4);
        burst(ball.x,ball.y,7,{spd:200,life:.3,size:2.5,colors:['#ff8787','#ffd43b'],shape:'c'});addFloat(ball.x,ball.y-20,'TAGGED!','#ff8787',14);dead=true;
      }
      if(dead)TB.splice(i,1);
    }
  }
  /* particles & floats */
  for(let i=parts.length-1;i>=0;i--){const p=parts[i];p.t+=dt;if(p.t>p.life){parts.splice(i,1);continue;}
    const dr=Math.exp(-3*dt);p.x+=p.vx*dt;p.y+=p.vy*dt;p.vx*=dr;p.vy*=dr;p.rot+=p.vr*dt;}
  for(let i=floats.length-1;i>=0;i--){const f=floats[i];f.t+=dt;if(f.t>f.life)floats.splice(i,1);}
}
