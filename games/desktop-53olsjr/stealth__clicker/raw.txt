function segRectHit(x1,y1,x2,y2,rx,ry,rw,rh){
  const dx=x2-x1,dy=y2-y1,p=[-dx,dx,-dy,dy],q=[x1-rx,rx+rw-x1,y1-ry,ry+rh-y1];
  let t0=0,t1=1;
  for(let i=0;i<4;i++){
    if(p[i]===0){if(q[i]<0)return false;}
    else{const t=q[i]/p[i];
      if(p[i]<0){if(t>t1)return false;if(t>t0)t0=t;}
      else{if(t<t0)return false;if(t<t1)t1=t;}}
  }return true;}
function rayWall(x,y,dx,dy,maxT){
  let best=maxT;
  for(const w of level.walls){
    let tn=0,tf=maxT;
    if(Math.abs(dx)<1e-9){if(x<w[0]||x>w[0]+w[2])continue;}
    else{const inv=1/dx;let a=(w[0]-x)*inv,b=(w[0]+w[2]-x)*inv;if(a>b){const t=a;a=b;b=t;}tn=Math.max(tn,a);tf=Math.min(tf,b);}
    if(Math.abs(dy)<1e-9){if(y<w[1]||y>w[1]+w[3])continue;}
    else{const inv=1/dy;let a=(w[1]-y)*inv,b=(w[1]+w[3]-y)*inv;if(a>b){const t=a;a=b;b=t;}tn=Math.max(tn,a);tf=Math.min(tf,b);}
    if(tf>tn&&tn>=0&&tn<best)best=tn;
  }return best;}
function losBlocked(ax,ay,bx,by){
  for(const w of level.walls)if(segRectHit(ax,ay,bx,by,w[0],w[1],w[2],w[3]))return true;
  return false;}
function collide(x,y,r){
  for(let it=0;it<2;it++)for(const w of level.walls){
    const cx=clamp(x,w[0],w[0]+w[2]),cy=clamp(y,w[1],w[1]+w[3]);
    let dx=x-cx,dy=y-cy;const d2=dx*dx+dy*dy;
    if(d2<r*r){
      const d=Math.sqrt(d2);
      if(d<1e-6){const l=x-w[0],rr=w[0]+w[2]-x,t=y-w[1],b=w[1]+w[3]-y,m=Math.min(l,rr,t,b);
        if(m===l)x=w[0]-r;else if(m===rr)x=w[0]+w[2]+r;else if(m===t)y=w[1]-r;else y=w[1]+w[3]+r;}
      else{x=cx+dx/d*r;y=cy+dy/d*r;}
    }}
  return [x,y];}

function moveWatcher(w,dt,live){
  if(w.type==='cam'){w.t+=dt;if(!w.staring)w.angle=w.base+Math.sin(w.t*w.speed)*w.amp;return;}
  if(w.state==='inv'&&live){
    w.invT-=dt;
    const ta=Math.atan2(w.ny-w.y,w.nx-w.x);
    w.angle=angLerp(w.angle,ta,10*dt);
    if(w.invT<=0)w.state='patrol';
  }else{
    const tp=w.path[w.idx];let dx=tp.x-w.x,dy=tp.y-w.y;const d=Math.hypot(dx,dy);
    if(d<3){w.idx=(w.idx+1)%w.path.length;}
    else if(!w.staring){
      w.x+=dx/d*w.speed*dt;w.y+=dy/d*w.speed*dt;
      const c=collide(w.x,w.y,9);w.x=c[0];w.y=c[1];}
    const ta=Math.atan2(dy,dx);
    w.angle=angLerp(w.angle,ta,(d<3?1:5)*dt);
  }
  if(w.staring&&live){const ta=Math.atan2(player.y-w.y,player.x-w.x);w.angle=angLerp(w.angle,ta,14*dt);}
}

function update(dt){
  gtime+=dt;
  alarmT=Math.max(0,alarmT-dt*1.3);
  shake*=Math.max(0,1-6*dt);if(shake<.05)shake=0;
  updateParts(dt);

  if(mode==='play'){
    levelTime+=dt;runTime+=dt;
    invuln=Math.max(0,invuln-dt);
    player.stepCool=Math.max(0,player.stepCool-dt);
    if(player.moving){
      player.ft+=dt/DASH_DUR;const t=1-(1-Math.min(1,player.ft))*(1-player.ft);
      const px=lerp(player.fx,player.tx,t),py=lerp(player.fy,player.ty,t);
      const c=collide(px,py,player.r);player.x=c[0];player.y=c[1];
      if(player.ft>=1){player.moving=false;player.lastStep=gtime;}
    }
    // pickups
    let taken=0;
    for(const g of gems){if(g.taken)continue;
      if(dist(g.x,g.y,player.x,player.y)<player.r+14){g.taken=true;score+=100;sparkle(g.x,g.y);SFX.pickup();updateGemHUD();}}
    taken=gems.filter(g=>g.taken).length;
    if(taken===gems.length&&!exitP.open){exitP.open=true;SFX.unlock();
      if(!exitAnnounced){exitAnnounced=true;showToast('<b>LOCKER OPEN</b><span>REACH THE EXIT →</span>',1700);}}
    if(exitP.open&&dist(exitP.x,exitP.y,player.x,player.y)<28){clearLevel();}

    // watchers: move then detect
    for(const w of watchers)moveWatcher(w,dt,true);
    spotted=false;let maxRate=0;
    const active=invuln<=0;
    for(const w of watchers){
      w.sees=false;w.staring=false;
      if(!active)continue;
      const d=dist(w.x,w.y,player.x,player.y);
      const angTo=Math.atan2(player.y-w.y,player.x-w.x);
      if(d<=w.range&&angDiff(w.angle,angTo)<=w.fov/2&&!losBlocked(w.x,w.y,player.x,player.y)){
        w.sees=true;let r=(1-d/w.range)*.95+.35;
        if(player.moving)r*=1.45;else r*=.5;
        if(r>maxRate)maxRate=r;}
    }
    spotted=maxRate>0;
    if(spotted){suspicion=clamp(suspicion+dt*maxRate,0,1);
      for(const w of watchers)if(w.sees&&suspicion>.55)w.staring=true;
      tickTimer+=dt;const iv=lerp(.38,.12,suspicion);
      if(tickTimer>iv){tickTimer=0;SFX.tick(suspicion);}
    }else{suspicion=Math.max(0,suspicion-dt*.55);tickTimer=0;}
    if(suspicion>=1&&invuln<=0)caught();

  }else if(mode==='menu'){
    for(const w of watchers){w.staring=false;moveWatcher(w,dt,false);}
  }else if(mode==='caught'){
    caughtT-=dt;
    if(caughtT<=0){
      if(cover<=0)showGameOver();
      else{player.x=level.start[0];player.y=level.start[1];player.moving=false;
        player.ft=0;suspicion=0;invuln=2;mode='play';}
    }
  }else if(mode==='clear'){
    clearT-=dt;
    if(clearT<=0){if(levelIdx+1<LEVELS.length)loadLevel(levelIdx+1,true);else showVictory();}
  }

  if(droneG&&AC){const target=(mode==='play'&&spotted)?0.04+suspicion*.1:0;
    droneG.gain.setTargetAtTime(target,AC.currentTime,.08);}

  // HUD refresh
  $('#susFill').style.width=(suspicion*100)+'%';
  const gg=Math.round(lerp(190,64,suspicion)),bb=Math.round(lerp(74,50,suspicion));
  $('#susFill').style.background='rgb(255,'+gg+','+bb+')';
  $('#vignette').style.opacity=clamp(suspicion*.5+(alarmT>0?alarmT*.35:0),0,.85);
}

  for(const p of parts){const a=p.life/p.max;

}
