  if(!AC||muted) return; const t=AC.currentTime;
  const o=AC.createOscillator(), g=AC.createGain();
  o.type=type; o.frequency.setValueAtTime(f,t);
  if(f2) o.frequency.exponentialRampToValueAtTime(Math.max(30,f2),t+dur);

}

  if(!AC||muted) return; const t=AC.currentTime;
  const len=Math.max(1,(dur*AC.sampleRate)|0), buf=AC.createBuffer(1,len,AC.sampleRate), d=buf.getChannelData(0);

  if(f2) f.frequency.exponentialRampToValueAtTime(f2,t+dur);
  const g=AC.createGain(); g.gain.value=vol;
  s.connect(f); f.connect(g); g.connect(master); s.start(t);
}

  }
  else if(type==='QUEEN'){ opp.sizeMul*=0.6; beep(500,'triangle',.12,.25,180); addFloat(opp.x,opp.y-paddleH(opp)/2-24*U,'SHRUNK!','#ff97a2',20); burst(opp.x,opp.y,'#ff97a2',10); }
  else if(type==='JACK'){ by.sizeMul*=1.5; by.speedMul*=1.35; beep(220,'triangle',.14,.28,440); addFloat(by.x,by.y-paddleH(by)/2-24*U,'GROWN!','#c9f0a9',20); burst(by.x,by.y,'#c9f0a9',10); }
  else if(type==='TEN'){
    const dir=(opp.y < F.y+F.h/2)?1:-1;
    ball.spinA=dir*760*U; sfx.pop(); addFloat(ball.x,ball.y-BR*2,'CURL!','#f5e9c8',20); burst(ball.x,ball.y,'#f5e9c8',10);
  }
  else if(type==='JOKER'){ opp.frozen=1.5; sfx.freeze(); addFloat(opp.x,opp.y,'FROZEN!','#bfe3ff',24,true); burst(opp.x,opp.y,'#bfe3ff',16); }
}

function playCard(p,slot){
  const type=p.hand[slot]; if(!type) return;
  if(!(S.phase==='rally'||S.phase==='serve')) return;
  p.hand[slot]=null; p.cardsPlayed++; if(p.id===0) stats.c1++; else stats.c2++;
  let x0, y0;
  const sr=p.slotRects[slot];
  if(sr){ x0=sr.x+sr.w/2; y0=sr.y+sr.h/2; }
  else { x0=W/2+(slot-1)*CARDW*1.5; y0=p.id===0? H-BOTM*0.5 : TOPM*0.5; }
  flies.push({x0,y0, t:0, dur:0.34, type, by:p});
  sfx.deal();
}

/* ============================== AI ============================== */
function predictY(xTarget){
  if(!ball.vx) return ball.y;
  let t=(xTarget-ball.x)/ball.vx;
  if(t<0||t>4) return ball.y;
  let y=ball.y+ball.vy*t+0.5*ball.spinA*t*t;
  const top=F.y+BR, bot=F.y+F.h-BR, span=bot-top;
  let m=(y-top)%(2*span); if(m<0)m+=2*span;
  return m<span? top+m : top+2*span-m;
}
function aiThink(p){
  p.thinkCd-=0.016; const b=ball;
  if(!b||b.dead) return;
  const opp=other(p);
  const incoming = Math.sign(b.vx)===p.side && Math.abs(b.vx)>1;
  let t=9; if(incoming){ t=(p.faceX-b.x)/b.vx; }
  if(!(t>0&&t<3)) return;
  const yP=predictY(p.faceX), ph=paddleH(p)/2;
  const danger=Math.abs(yP-p.y)>ph+8*U;
  if(p.thinkCd>0) return;
  const has=ty=>p.hand.includes(ty);
  let choice=null;
  if(incoming && t<0.95){
    if(danger){
      if(has('JOKER')) choice='JOKER';
      else if(has('QUEEN')) choice='QUEEN';
      else if(has('JACK')) choice='JACK';
    } else if(t>0.45 && has('TEN') && Math.random()<0.5) choice='TEN';
  } else if(!incoming){
    if(b.speed*b.boost<BASE*1.3 && has('ACE') && Math.random()<0.3) choice='ACE';
  }
  if(!choice && has('KING') && Math.abs(b.x-(F.x+F.w/2))<F.w*0.14){
    if(Math.abs(b.y-p.y) > Math.abs(b.y-opp.y)+70*U) choice='KING';
  }
  if(choice){ playCard(p, p.hand.indexOf(choice)); p.thinkCd=0.9; }
}

const keys={};
function movePlayer(p,dt){
  const ph=paddleH(p)/2;
  if(p.frozen>0){ p.frozen-=dt; p.vy*=0.8; p.y+=p.vy*dt*0.15; }
  else{
    let dir=0; const maxSp=(p.ai? 470:560)*U*p.speedMul;
    if(p.ai){
      p.aiMem.t-=dt;
      if(p.aiMem.t<=0){
        p.aiMem.t = S.attract? 0.07 : 0.1;
        const b=ball;
        if(b && !b.dead && Math.sign(b.vx)===p.side){
          const err=(S.attract? 26:44)*U*clamp(b.speed*b.boost/MAXS,0.3,1.4);
          p.aiMem.target=predictY(p.faceX)+rand(-err,err);
        } else p.aiMem.target=F.y+F.h/2+Math.sin(S.t*0.7+p.id*2)*F.h*0.15;
      }
      dir=clamp((p.aiMem.target-p.y)/(24*U),-1,1);
    } else {
      if(keys[p.kUp]) dir-=1;
      if(keys[p.kDown]) dir+=1;
      if(p.id===0 && !S.attract && ptr.down) dir=clamp((ptr.y-p.y)/(18*U),-1.25,1.25);
    }
    p.vy=lerp(p.vy, dir*maxSp, 1-Math.pow(0.0002,dt));
    p.y+=p.vy*dt;
  }
  p.x = p.side===-1 ? F.x+30*U : F.x+F.w-30*U;
  p.faceX = p.side===-1 ? p.x+PW/2 : p.x-PW/2;
  p.y=clamp(p.y, F.y+ph+4*U, F.y+F.h-ph-4*U);
}
function updateBall(dt){
  const b=ball; if(b.dead) return;
  b.px=b.x; b.py=b.y;
  b.vy+=b.spinA*dt;
  const vyMax=Math.abs(b.vx)*1.2+60*U;
  b.vy=clamp(b.vy,-vyMax,vyMax);
  const eff=Math.min(b.speed*b.boost, MAXS*1.45);
  const m=Math.hypot(b.vx,b.vy)||1;
  b.vx=b.vx/m*eff; b.vy=b.vy/m*eff;
  b.x+=b.vx*dt; b.y+=b.vy*dt;
  b.rot += (b.vx>0?1:-1)*(3.2+Math.min(5,eff/(MAXS*U))*2)*dt;
  b.trail.unshift({x:b.x,y:b.y,r:b.rot}); if(b.trail.length>12) b.trail.pop();

  if(b.y-BR<F.y){ b.y=F.y+BR; b.vy=Math.abs(b.vy); sfx.wall(); burst(b.x,F.y+4,'#e9dfc4',4); S.shake=Math.max(S.shake,1.5*U); }
  if(b.y+BR>F.y+F.h){ b.y=F.y+F.h-BR; b.vy=-Math.abs(b.vy); sfx.wall(); burst(b.x,F.y+F.h-4,'#e9dfc4',4); S.shake=Math.max(S.shake,1.5*U); }

  for(const p of players){
    if(Math.sign(b.vx)!==p.side) continue;
    const ph=paddleH(p)/2, fx=p.faceX;
    const within=b.y>p.y-ph-BR && b.y<p.y+ph+BR;
    const crossed = p.side===-1 ? (b.x-BR<=fx && b.px-BR>fx) : (b.x+BR>=fx && b.px+BR<fx);
    if(within&&crossed){
      const off=clamp((b.y-p.y)/(ph+BR),-1,1), ang=off*1.05;
      b.speed=Math.min(b.speed*1.045,b.MAX=b.MAX||MAXS, MAXS);
      const e2=Math.min(b.speed*b.boost,MAXS*1.45);
      b.vx=Math.cos(ang)*e2*p.side; b.vy=Math.sin(ang)*e2;
      b.x = p.side===-1? fx+BR : fx-BR;
      S.rallyHits++; if(S.rallyHits>S.longestRally) S.longestRally=S.rallyHits;
      sfx.hit(b.speed/MAXS); burst(fx,b.y,p.lite,7); S.shake=Math.max(S.shake,(2+b.speed/MAXS*3)*U);
    }
  }
  if(b.x < F.x-50*U) scorePoint(players.find(p=>p.side===1));
  else if(b.x > F.x+F.w+50*U) scorePoint(players.find(p=>p.side===-1));
}
function scorePoint(w){
  if(!w) return;
  w.score++; ball.dead=true; S.phase='point'; S.pointT=0;
  const ex = w.side===1 ? F.x+30 : F.x+F.w-30;
  trickFlies.push({x:clamp(ball.x,F.x+20*U,F.x+F.w-20*U), y:clamp(ball.y,F.y+40*U,F.y+F.h-40*U), t:0, by:w});
  burst(ex, clamp(ball.y,F.y+30*U,F.y+F.h-30*U), w.lite, 26, true);
  S.shake=7*U; sfx.flip(); setTimeout(()=>sfx.score(),120);
  if(!S.attract) addFloat(F.x+F.w/2, F.y+F.h*0.4, 'POINT · '+w.name, w.lite, 30, true);
  if(w.score>=5){
    if(S.attract){ setTimeout(()=>{ if(S.attract){ players[0].score=0; players[1].score=0; resetPoint(); } },900); }
    else { setTimeout(()=>{ if(S.phase==='point') endMatch(w); }, 1200); }
  } else {
    setTimeout(()=>{ if(S.phase==='point') resetPoint(); }, S.attract? 700 : 1300);
  }
}
function update(dt){
  S.t+=dt; S.shake*=Math.pow(0.02,dt); if(S.shake<0.2) S.shake=0;
  if(S.paused) return;
  for(const p of players) movePlayer(p,dt);

  if(S.phase==='dealing'){
    S.dealT+=dt;
    let done=true;
    for(let i=0;i<3;i++) if(clamp((S.dealT-i*0.09)/0.3,0,1)<1) done=false;
    if(done && S.dealT>0.65){
      S.phase='serve'; S.serveT=0;
      ball.x=F.x+F.w/2; ball.y=F.y+F.h*0.5+rand(-F.h*0.18,F.h*0.18);
      ball.vx=0; ball.vy=0; ball.speed=BASE; ball.boost=1; ball.spinA=0; ball.trail.length=0;
    }
  }
  else if(S.phase==='serve'){
    S.serveT+=dt;
    if(S.serveT>0.85){
      const a=rand(-0.32,0.32);
      ball.vx=Math.cos(a)*BASE*S.serveDir; ball.vy=Math.sin(a)*BASE;
      ball.dead=false; S.phase='rally'; sfx.pop(); burst(ball.x,ball.y,'#f6efdc',8);
    }
  }
  else if(S.phase==='rally'){
    updateBall(dt);
    for(const p of players) if(p.ai && (S.attract||true)) aiThink(p);
  }

  for(let i=flies.length-1;i>=0;i--){ const f=flies[i]; f.t+=dt/f.dur;
    if(f.t>=1){ flies.splice(i,1);
      if(S.phase==='rally'||S.phase==='serve') applyFx(f.by,f.type); else burst(F.x+F.w/2,F.y+F.h/2,'#f6efdc',8); } }
  for(let i=trickFlies.length-1;i>=0;i--){ const f=trickFlies[i]; f.t+=dt/0.55; if(f.t>=1) trickFlies.splice(i,1); }
  for(let i=floats.length-1;i>=0;i--){ const f=floats[i]; f.t+=dt/f.dur; f.y-=28*U*dt; if(f.t>=1) floats.splice(i,1); }
  for(let i=particles.length-1;i>=0;i--){ const p=particles[i]; p.t+=dt/p.life;
    if(p.t>=1){ particles.splice(i,1); continue; }
    p.vy+=520*U*dt; p.x+=p.vx*dt; p.y+=p.vy*dt; p.rot+=p.vr*dt; }
}

}

function drawHandRow(p,bottom){
  const scale = bottom?1:0.78;
  const cw=CARDW*scale, ch=CARDH*scale;
  const gap=cw+14*U;
  const x0=W/2-(gap)/1*0 - (gap*1); // center 3 slots
  const cx0=W/2-gap;
  const cy = bottom? H-BOTM+16*U+ch/2 : TOPM-14*U*scale-ch/2;
  p.slotRects=[];
  for(let i=0;i<3;i++){
    const x=cx0+i*gap, type=p.hand[i];
    const prog = clamp((S.dealT - i*0.09)/0.3,0,1);
    let yy=cy, alpha=1;
    if(prog<1){
      const e=1-Math.pow(1-prog,3);
      yy = bottom? cy+(1-e)*(-BOTM*0.8) : cy+(1-e)*(TOPM*0.9);
      alpha=prog*1.4;
    }
    if(type){
      ctx.save(); ctx.globalAlpha=Math.min(1,alpha);
      drawPlayingCard(x,yy,cw,ch,type,{rot:(1-prog)*-0.3*(i%2?1:-1)});
      ctx.restore();
      // hover shine when playable
      if(S.phase==='rally'||S.phase==='serve'){
        const s=1+Math.sin(S.t*4+i)*0.015;
        ctx.save(); ctx.translate(x,yy); ctx.scale(s,s); ctx.translate(-x,-yy);
        rrect(x-cw/2-3,yy-ch/2-3,cw+6,ch+6,7*U);
        ctx.strokeStyle='rgba(246,239,220,'+(0.18+Math.sin(S.t*4+i)*0.1)+')'; ctx.lineWidth=1.5; ctx.stroke();
        ctx.restore();
      }
    } else {
      rrect(x-cw/2,yy-ch/2,cw,ch,6*U);
      ctx.strokeStyle='rgba(246,239,220,.18)'; ctx.setLineDash([5,5]); ctx.lineWidth=1.5; ctx.stroke(); ctx.setLineDash([]);
    }
    p.slotRects[i]={x:x-cw/2,y:yy-ch/2,w:cw,h:ch};
    if(!p.ai && !S.attract){
      const label = bottom? ['1','2','3'][i] : ['8','9','0'][i];
      drawKeycap(x, bottom? yy-ch/2-15*U : yy+ch/2+15*U, label);
    }
  }
  if(!p.ai){
    ctx.font='700 '+(10*U)+'px Georgia, serif'; ctx.textAlign='center'; ctx.fillStyle=p.lite; ctx.globalAlpha=0.8;
    ctx.fillText('YOUR CARDS', W/2 - (gap*1) - cw*0.9 - 34*U, bottom? cy : cy+4);
    ctx.globalAlpha=1;
  }
}
function drawFlies(){
  for(const f of flies){
    const e=f.t<.5? 2*f.t*f.t : 1-Math.pow(-2*f.t+2,2)/2;
    const x=lerp(f.x0,F.x+F.w/2,e), y=lerp(f.y0,F.y+F.h*0.45,e);
    const sc=1+e*0.6;
    ctx.save(); ctx.globalAlpha=1-Math.max(0,f.t-0.85)/0.15;
    drawPlayingCard(x,y,CARDW*0.8*sc,CARDH*0.8*sc,f.type,{rot:(1-e)*0.9});
    ctx.restore();
  }
}
function drawTrickFlies(){
  for(const f of trickFlies){
    const e=f.t*f.t*(3-2*f.t);
    const dir=f.by.id===0?1:-1;
    const tx=f.by.id===0? F.x+52*U+dir*f.by.score*20*U : F.x+F.w-52*U+dir*f.by.score*20*U;
    const ty=F.y+64*U;
    const x=lerp(f.x,tx,e), y=lerp(f.y,ty,e)-Math.sin(e*Math.PI)*70*U;
    drawPlayingCardMini(x,y,30*U*(1.4-e*0.4),42*U*(1.4-e*0.4),f.by);
  }
}
function drawServeCard(){
  const b=ball, cx=b.x, cy=b.y;
  const w=BR*2, h=BR*2.7;
  if(S.serveT<0.45){
    const e=S.serveT/0.45;
    drawPlayingCard(cx,cy,w,h,pick&&'ACE',{back:true,rot:Math.sin(e*3)*0.1});
  } else {
    const e=clamp((S.serveT-0.45)/0.25,0,1);
    const sx=Math.abs(Math.cos(e*Math.PI));
    ctx.save(); ctx.translate(cx,cy); ctx.scale(sx,1);
    rrect(-w/2,-h/2,w,h,4*U);
    const g=ctx.createLinearGradient(0,-h/2,0,h/2); g.addColorStop(0,'#fdfaf0'); g.addColorStop(1,'#e6dcc2');
    ctx.fillStyle=g; ctx.fill(); ctx.strokeStyle='#c9bd9f'; ctx.stroke();
    ctx.fillStyle=(b.suit==='♥'||b.suit==='♦')?'#c23a4e':'#2a2731';
    ctx.font=(h*0.5)+'px Georgia, serif'; ctx.textAlign='center'; ctx.textBaseline='middle';
    ctx.fillText(b.suit,0,h*0.06);
    ctx.restore();
  }
}
function drawFloats(){
  for(const f of floats){
    const a=1-f.t*f.t;
    ctx.save(); ctx.globalAlpha=a;
    ctx.font='700 '+(f.size*U)+'px '+(f.disp?'Georgia, serif':"'Segoe UI', Arial, sans-serif");
    ctx.textAlign='center'; ctx.textBaseline='middle';
    ctx.lineWidth=4*U; ctx.strokeStyle='rgba(8,20,12,.7)'; ctx.strokeText(f.txt,f.x,f.y);
    ctx.fillStyle=f.col; ctx.fillText(f.txt,f.x,f.y);
    ctx.restore();
  }
}
function drawParticles(){
  for(const p of particles){
    const a=1-p.t;
    ctx.save(); ctx.translate(p.x,p.y); ctx.rotate(p.rot); ctx.globalAlpha=a*0.9;
    ctx.fillStyle=p.col;
    ctx.beginPath(); ctx.moveTo(0,-p.sz); ctx.lineTo(p.sz*0.8,0); ctx.lineTo(0,p.sz); ctx.lineTo(-p.sz*0.8,0); ctx.closePath(); ctx.fill();
    ctx.restore();
  }
}
function render(){
  // room
  const rg=ctx.createRadialGradient(W/2,H*0.35,60,W/2,H*0.5,Math.max(W,H)*0.8);
  rg.addColorStop(0,'#2a1a10'); rg.addColorStop(1,'#0f0704');
  ctx.fillStyle=rg; ctx.fillRect(0,0,W,H);

  ctx.save();
  if(S.shake>0) ctx.translate(rand(-S.shake,S.shake),rand(-S.shake,S.shake));

  // table rim
  const rx=F.x-RIM, ry=F.y-RIM, rw=F.w+RIM*2, rh=F.h+RIM*2;
  rrect(rx-4,ry-4,rw+8,rh+8,RIM*1.6);
  ctx.fillStyle='rgba(0,0,0,.5)'; ctx.fill();
  const wg=ctx.createLinearGradient(rx,ry,rx,ry+rh);
  wg.addColorStop(0,'#9a6a3c'); wg.addColorStop(0.5,'#7a4f28'); wg.addColorStop(1,'#5d3a1e');
  rrect(rx,ry,rw,rh,RIM*1.4); ctx.fillStyle=wg; ctx.fill();
  ctx.strokeStyle='#3a2410'; ctx.lineWidth=2; ctx.stroke();

  // felt
  const fg=ctx.createRadialGradient(F.x+F.w/2,F.y+F.h*0.42,40,F.x+F.w/2,F.y+F.h/2,Math.max(F.w,F.h)*0.75);
  fg.addColorStop(0,'#2f7d4e'); fg.addColorStop(0.6,'#1c5c38'); fg.addColorStop(1,'#0f3f25');
  rrect(F.x,F.y,F.w,F.h,10*U); ctx.fillStyle=fg; ctx.fill();
  if(noisePat){ ctx.save(); rrect(F.x,F.y,F.w,F.h,10*U); ctx.clip(); ctx.globalAlpha=0.5; ctx.fillStyle=noisePat; ctx.fillRect(F.x,F.y,F.w,F.h); ctx.restore(); }
  // inner shadow
  ctx.save(); rrect(F.x,F.y,F.w,F.h,10*U); ctx.clip();
  ctx.strokeStyle='rgba(0,0,0,.4)'; ctx.lineWidth=10*U; rrect(F.x+5*U,F.y+5*U,F.w-10*U,F.h-10*U,8*U); ctx.stroke();
  ctx.restore();

  drawNet();
  if(S.phase==='serve' && !ball.dead===false) {} // (ball drawn below per phase)
  drawTrickPiles();
  for(const p of players) drawPaddle(p);
  if(S.phase==='rally') drawBall();
  if(S.phase==='serve') drawServeCard();
  if(ball.boost>1 && !ball.dead && S.phase==='rally') drawBadge('turbo', ball.x, ball.y-BR*2.6-10*U);
  drawParticles();
  drawTrickFlies();
  drawFlies();
  drawHand();
  drawFloats();

  if(S.paused){
    ctx.fillStyle='rgba(5,14,9,.6)'; ctx.fillRect(0,0,W,H);
    ctx.font='700 '+(34*U)+'px Georgia, serif'; ctx.textAlign='center'; ctx.textBaseline='middle';
    ctx.fillStyle='#f6efdc'; ctx.fillText('PAUSED', W/2, H/2-14*U);
    ctx.font=(15*U)+"px 'Segoe UI', Arial, sans-serif"; ctx.fillStyle='#9fc4a8';
    ctx.fillText('press P to resume · ESC for the main table', W/2, H/2+26*U);
  }
  ctx.restore();
}
