function segPos(i){
  let d=i*SPACING, px=snake.trail[0].x, py=snake.trail[0].y;
  for(let k=1;k<snake.trail.length&&d>0;k++){
    const a=snake.trail[k-1], b=snake.trail[k];
    const l=Math.hypot(b.x-a.x,b.y-a.y);
    if(l===0)continue;
    if(d<=l){ const t=d/l; return {x:a.x+(b.x-a.x)*t, y:a.y+(b.y-a.y)*t}; }
    d-=l;
  }
  const last=snake.trail[snake.trail.length-1];
  return {x:last.x,y:last.y};
}
function radiusFor(i){
  if(i===0)return HEAD_R;
  return Math.max(3.5, SEG_R*(1-0.5*i/Math.max(snake.segs,6)));
}
function circleRect(cx,cy,r,rx,ry,hw,hh){
  const px=Math.max(rx-hw,Math.min(cx,rx+hw)), py=Math.max(ry-hh,Math.min(cy,ry+hh));
  const dx=cx-px, dy=cy-py; return dx*dx+dy*dy<r*r;
}
function burst(x,y,col,n){
  for(let i=0;i<n;i++){ const a=R(0,6.283), sp=R(30,190);
    particles.push({x,y,vx:Math.cos(a)*sp,vy:Math.sin(a)*sp,life:R(0.35,0.8),max:0.8,col,r:R(1.5,3.5)}); }
}

function resetSnake(){
  const x=W/2, y=Math.min(H-84,dangerY-34);
  snake={x,y,ang:-Math.PI/2,speed:235,segs:BASE_SEGS,invuln:2.2,trail:[]};
  for(let i=0;i<96;i++)snake.trail.push({x:x-Math.cos(snake.ang)*i*6,y:y-Math.sin(snake.ang)*i*6});
}
function startGame(){
  score=0; lives=3; wave=1; combo=1; comboT=0; newBest=false;
  stats={eat:0,shot:0}; pbullets=[]; ebullets=[]; particles=[]; popups=[];
  dying=0; fireCd=0; wavePending=false; banner=null;
  resetSnake(); spawnWave(1);
  setBanner('WAVE 1','THE FLEET APPROACHES');
  state='playing'; SFX.wave();
}
const SUBS=['THE FLEET APPROACHES','REINFORCEMENTS INBOUND','DEEPER INTO THE SWARM','THE SKY KEEPS FALLING','NO QUARTER GIVEN'];
function spawnWave(w){
  const rows=Math.min(3+((w-1)>>1),5), cols=5;
  invaders=[];
  for(let r=0;r<rows;r++)for(let c=0;c<cols;c++){
    const type=r===0?0:(r===1?1:2);
    invaders.push({bx:(c-(cols-1)/2)*58, by:r*46, type, val:VALS[type], x:0,y:-300});
  }
  form={ox:0,dir:w%2?1:-1,topY:-96,speed:34+w*7,fall:4+w*1.3,
        interval:Math.max(0.55,1.5-w*0.1),fireT:2.6,frameT:0,total:rows*cols,
        maxOx:(cols-1)/2*58+30};
}
function killInvader(i,how){
  const inv=invaders[i]; invaders.splice(i,1);
  const mult=combo;
  const base=how==='eat'?inv.val*2:inv.val;
  const gained=base*mult;
  score+=gained; stats[how]++;
  combo=Math.min(8,combo+1); comboT=2.5; scorePop=0.25;
  if(how==='eat'){
    if(snake.segs<MAX_SEGS)snake.segs+=2;
    else{ score+=25; popup(inv.x,inv.y-14,'FULL +25','#ffd23f',12); }
    SFX.eat();
  }else{ SFX.boom(); shakeFx(0.18,4); }
  popup(inv.x,inv.y,(how==='eat'?'EAT ':'')+'+'+gained+(mult>1?' ×'+mult:''), how==='eat'?'#8dff57':'#eaffb0', how==='eat'?16:13);
  burst(inv.x,inv.y,TYPE_COL[inv.type],how==='eat'?26:16);
}
function damage(){
  if(snake&&snake.invuln>0)return;
  lives--; combo=1; comboT=0;
  SFX.hurt(); flashA=0.5; shakeFx(0.6,14);
  if(snake)for(let i=0;i<snake.segs;i++){
    const p=i===0?{x:snake.x,y:snake.y}:segPos(i);
    burst(p.x,p.y,'#8dff57',3);
  }
  dying=1.1;
}
function gameOver(){
  state='gameover'; SFX.over();
  try{
    const h=+localStorage.getItem(HIKEY)||0;
    if(score>h){ localStorage.setItem(HIKEY,String(score)); newBest=true; hi=Math.max(hi,score); }
  }catch(e){ if(score>hi){ hi=score; newBest=true; } }
}

/* ============================== UPDATE ============================== */
function snakeMove(dt){
  const s=snake;
  s.invuln=Math.max(0,s.invuln-dt);
  let tv=235; if(keys.up)tv=345; else if(keys.down)tv=135;
  s.speed+=(tv-s.speed)*Math.min(1,dt*6);
  const turn=3.4*(0.7+0.5*s.speed/345);
  if(keys.left)s.ang-=turn*dt;
  if(keys.right)s.ang+=turn*dt;
  let nx=s.x+Math.cos(s.ang)*s.speed*dt, ny=s.y+Math.sin(s.ang)*s.speed*dt;
  if(nx<HEAD_R){nx=HEAD_R;s.ang=Math.PI-s.ang;}
  else if(nx>W-HEAD_R){nx=W-HEAD_R;s.ang=Math.PI-s.ang;}
  if(ny<HEAD_R){ny=HEAD_R;s.ang=-s.ang;}
  else if(ny>H-HEAD_R){ny=H-HEAD_R;s.ang=-s.ang;}
  s.x=nx; s.y=ny;
  s.trail.unshift({x:s.x,y:s.y});
  const maxLen=(s.segs+3)*SPACING+60; let acc=0;
  for(let i=1;i<s.trail.length;i++){
    acc+=Math.hypot(s.trail[i-1].x-s.trail[i].x,s.trail[i-1].y-s.trail[i].y);
    if(acc>maxLen){ s.trail.length=i; break; }
  }
  if(s.invuln<=0){
    for(let i=4;i<s.segs;i++){
      const p=segPos(i), rr=HEAD_R+radiusFor(i)-5;
      if(Math.hypot(p.x-s.x,p.y-s.y)<rr){ damage(); break; }
    }
  }
}
function tryFire(){
  if(state!=='playing'||dying>0||!snake||fireCd>0)return;
  fireCd=0.17;
  const a=snake.ang, mx=snake.x+Math.cos(a)*(HEAD_R+9), my=snake.y+Math.sin(a)*(HEAD_R+9);
  pbullets.push({x:mx,y:my,vx:Math.cos(a)*560,vy:Math.sin(a)*560});
  SFX.fire(); burst(mx,my,'#d8ffb0',2);
}
function formationUpdate(dt){
  const rem=invaders.length, tot=form.total||1;
  const sf=1+0.9*(1-rem/tot);
  form.ox+=form.dir*form.speed*sf*dt;
  form.topY+=form.fall*dt;
  if(Math.abs(form.ox)>form.maxOx){
    form.ox=Math.sign(form.ox)*form.maxOx; form.dir*=-1; form.topY+=26; SFX.tick();
  }
  form.frameT+=dt*(1.6+2.2*(1-rem/tot));
  for(const inv of invaders){ inv.x=W/2+form.ox+inv.bx; inv.y=form.topY+inv.by; }
}
function enemyFire(dt){
  form.fireT-=dt;
  if(form.fireT>0||!snake)return;
  const vis=invaders.filter(v=>v.y>-6);
  if(!vis.length){ form.fireT=0.35; return; }
  const inv=vis[(Math.random()*vis.length)|0];
  const sp=Math.min(460,235+wave*15);
  const d=Math.hypot(snake.x-inv.x,snake.y-inv.y)||1;
  const tt=d/sp;
  let tx=snake.x+Math.cos(snake.ang)*snake.speed*tt*0.85;
  let ty=snake.y+Math.sin(snake.ang)*snake.speed*tt*0.85;
  tx=Math.min(Math.max(tx,20),W-20); ty=Math.min(ty,H-20);
  let ax=tx-inv.x, ay=ty-(inv.y+12); const al=Math.hypot(ax,ay)||1;
  ebullets.push({x:inv.x,y:inv.y+13,vx:ax/al*sp,vy:ay/al*sp});
  form.fireT=form.interval*R(0.75,1.35);
  SFX.efire();
}
function breachCheck(){
  for(let i=invaders.length-1;i>=0;i--){
    const inv=invaders[i];
    if(inv.y+14>=dangerY){
      invaders.splice(i,1);
      burst(inv.x,dangerY,'#ff3b5c',26);
      popup(inv.x,dangerY-16,'BREACH!','#ff3b5c',16);
      shakeFx(0.5,10); SFX.boom(); damage(); return;
    }
  }
}
function eatCheck(){
  if(dying>0||!snake)return;
  for(let i=invaders.length-1;i>=0;i--){
    const inv=invaders[i]; let eaten=false;
    if(circleRect(snake.x,snake.y,HEAD_R+3,inv.x,inv.y,17,13))eaten=true;
    else{
      for(let s=2;s<snake.segs&&!eaten;s+=2){
        const p=segPos(s);
        if(circleRect(p.x,p.y,SEG_R+1,inv.x,inv.y,17,13))eaten=true;
      }
    }
    if(eaten)killInvader(i,'eat');
  }
}
function update(dt){
  tGlobal+=dt;
  const sp=(state==='playing'?26:14)*dt;
  for(const st of stars){ st.y+=st.z*sp; if(st.y>H+2){st.y=-2;st.x=R(0,W);} }
  shakeT=Math.max(0,shakeT-dt);
  flashA=Math.max(0,flashA-dt*1.4);
  scorePop=Math.max(0,scorePop-dt);
  for(let i=particles.length-1;i>=0;i--){ const p=particles[i];
    p.life-=dt; if(p.life<=0){particles.splice(i,1);continue;}
    p.x+=p.vx*dt; p.y+=p.vy*dt; p.vx*=(1-2.4*dt); p.vy*=(1-2.4*dt); }
  for(let i=popups.length-1;i>=0;i--){ const p=popups[i];
    p.t-=dt; if(p.t<=0){popups.splice(i,1);continue;} p.y-=30*dt; }

  if(state!=='playing')return;

  if(banner){ banner.t-=dt; if(banner.t<=0)banner=null; }
  comboT-=dt; if(comboT<=0)combo=1;

  if(dying>0){
    dying-=dt;
    if(dying<=0){
      if(lives>0){ resetSnake(); ebullets.length=0; }
      else gameOver();
    }
  }else{
    snakeMove(dt);
    if(fireCd>0)fireCd-=dt;
    if(keys.fire)tryFire();
    if(dying<=0){
      formationUpdate(dt); enemyFire(dt); breachCheck(); eatCheck();
    }
  }

  // player bullets
  for(let i=pbullets.length-1;i>=0;i--){
    const b=pbullets[i]; b.x+=b.vx*dt; b.y+=b.vy*dt;
    if(b.x<-24||b.x>W+24||b.y<-24||b.y>H+24){pbullets.splice(i,1);continue;}
    let hit=false;
    for(let j=invaders.length-1;j>=0;j--){
      const inv=invaders[j];
      if(Math.abs(b.x-inv.x)<19&&Math.abs(b.y-inv.y)<15){ killInvader(j,'shot'); hit=true; break; }
    }
    if(hit)pbullets.splice(i,1);
  }
  // enemy bullets
  for(let i=ebullets.length-1;i>=0;i--){
    const b=ebullets[i]; b.x+=b.vx*dt; b.y+=b.vy*dt;
    if(b.x<-24||b.x>W+24||b.y>H+34||b.y<-34){ebullets.splice(i,1);continue;}
    if(dying<=0&&snake.invuln<=0){
      let hit=Math.hypot(b.x-snake.x,b.y-snake.y)<HEAD_R+4;
      for(let s=2;s<snake.segs&&!hit;s++){
        const p=segPos(s);
        if(Math.hypot(b.x-p.x,b.y-p.y)<SEG_R+3)hit=true;
      }
      if(hit){ ebullets.splice(i,1); burst(b.x,b.y,'#ff5f7a',10); damage(); }
    }
  }
  // wave clear
  if(invaders.length===0&&dying<=0){
    if(!wavePending){
      wavePending=true; waveT=1.5;
      const bonus=100+(wave-1)*50; score+=bonus;
      popup(W/2,H*0.4,'WAVE CLEAR +'+bonus,'#ffd23f',22); SFX.ui();
    }else{
      waveT-=dt;
      if(waveT<=0){
        wavePending=false; wave++; spawnWave(wave);
        setBanner('WAVE '+wave,SUBS[(wave-1)%SUBS.length]); SFX.wave();
      }
    }
  }
}

function drawSnake(){

  drawSnakeBody(pts,ang,s.segs);
  ctx.globalAlpha=1;
}

function togglePause(){
  if(state==='playing'){ state='paused'; SFX.ui(); }

}
window.addEventListener('keydown',e=>{
  const c=e.code;
  if(['ArrowLeft','ArrowRight','ArrowUp','ArrowDown','Space'].includes(c))e.preventDefault();
  initAudio();
  switch(c){
    case 'ArrowLeft': case 'KeyA': keys.left=true; break;
    case 'ArrowRight': case 'KeyD': keys.right=true; break;
    case 'ArrowUp': case 'KeyW': keys.up=true; break;
    case 'ArrowDown': case 'KeyS': keys.down=true; break;
    case 'Space': keys.fire=true; if(state==='menu')startGame(); else if(state==='gameover')startGame(); break;
    case 'Enter': onPrimary(); break;
    case 'KeyR': if(state==='playing'||state==='gameover')startGame(); break;
    case 'KeyP': case 'Escape': togglePause(); break;
    case 'KeyM': muted=!muted; break;
  }
});
window.addEventListener('keyup',e=>{
  switch(e.code){
    case 'ArrowLeft': case 'KeyA': keys.left=false; break;
    case 'ArrowRight': case 'KeyD': keys.right=false; break;
    case 'ArrowUp': case 'KeyW': keys.up=false; break;
    case 'ArrowDown': case 'KeyS': keys.down=false; break;
    case 'Space': keys.fire=false; break;
  }
});
