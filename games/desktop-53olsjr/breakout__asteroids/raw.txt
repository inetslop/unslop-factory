function updatePaddle(dt){
  const MAXV=980*S;let dir=0;
  if(keys['arrowleft']||keys['a'])dir--;if(keys['arrowright']||keys['d'])dir++;
  if(dir!==0){pad.vx+=(dir*MAXV-pad.vx)*Math.min(1,dt*12);pad.x+=pad.vx*dt;}
  else if(pointerX!=null&&amb-lastPointerT<.25){
    const dx=pointerX-pad.x;pad.vx=clamp(dx*14,-MAXV,MAXV);pad.x+=clamp(dx,-MAXV*dt,MAXV*dt);}
  else{pad.vx*=Math.exp(-8*dt);pad.x+=pad.vx*dt;}
  pad.x=clamp(pad.x,pad.w/2+8,W-pad.w/2-8);
  const tw=(132*S)*(wideT>0?1.6:1);pad.w+=(tw-pad.w)*Math.min(1,dt*8);
  const tt=clamp(pad.vx/MAXV,-1,1)*.3;pad.tilt+=(tt-pad.tilt)*Math.min(1,dt*10);
  if(Math.abs(pad.vx)>90&&Math.random()<dt*26)
    parts.push({type:'spark',x:pad.x-Math.sign(pad.vx)*pad.w*.42,y:pad.y+pad.h*.3,
      vx:-Math.sign(pad.vx)*rand(60,150)*S,vy:rand(10,50)*S,size:rand(1.5,3)*S,color:'#ffb14d',life:.3,max:.3});
}
function collideBall(b){
  if(b.x-b.r<0){b.x=b.r;b.vx=Math.abs(b.vx);sfxWall();sparkBurst(2,b.y,3,'#cfe8ff',90*S);}
  if(b.x+b.r>W){b.x=W-b.r;b.vx=-Math.abs(b.vx);sfxWall();sparkBurst(W-2,b.y,3,'#cfe8ff',90*S);}
  if(b.y-b.r<TOPWALL){b.y=TOPWALL+b.r;b.vy=Math.abs(b.vy);sfxWall();wallFx={x:b.x,t:1};}
  if(shield&&b.vy>0){const sy=pad.y-pad.h/2-30*S;
    if(b.y+b.r>=sy&&Math.abs(b.x-pad.x)<pad.w/2+14*S){b.y=sy-b.r;b.vy=-Math.abs(b.vy);
      shield=false;sfxPaddle();sparkBurst(b.x,sy,10,'#b28dff',180*S);}}
  const top=pad.y-pad.h/2;
  if(b.vy>0&&b.y+b.r>=top&&b.y<pad.y+pad.h&&Math.abs(b.x-pad.x)<=pad.w/2+b.r){
    const rel=clamp((b.x-pad.x)/(pad.w/2),-1,1);
    let sp=Math.min(Math.hypot(b.vx,b.vy)*1.035+4*S,maxBallSpd());
    const a=-Math.PI/2+rel*.86;
    b.vx=Math.cos(a)*sp+pad.vx*.18;b.vy=Math.sin(a)*sp;
    if(Math.abs(b.vy)<sp*.42){b.vy=sp*.42*(b.vy<0?-1:1);
      b.vx=(b.vx>=0?1:-1)*Math.sqrt(Math.max(0,sp*sp-b.vy*b.vy));}
    b.y=top-b.r;combo=0;sfxPaddle();sparkBurst(b.x,top,6,'#7ff3ff',140*S);}
  if(b.y-b.r>H){b.dead=true;return;}
  for(const rk of rocks.slice()){
    if(rk.dead)continue;
    const dx=b.x-rk.x,dy=b.y-rk.y,rr=b.r+rk.r,d2=dx*dx+dy*dy;
    if(d2<rr*rr){
      const d=Math.sqrt(d2)||.01,nx=dx/d,ny=dy/d;
      b.x=rk.x+nx*rr;b.y=rk.y+ny*rr;
      let rvx=b.vx-rk.vx,rvy=b.vy-rk.vy;const dot=rvx*nx+rvy*ny;
      if(dot<0){rvx-=2*dot*nx;rvy-=2*dot*ny;}
      b.vx=rvx+rk.vx;b.vy=rvy+rk.vy;
      rk.flash=1;rk.vr+=rand(-2,2);
      damageRock(rk);
    }
  }
}
function updateBall(b,dt){
  if(b.stuck){b.x=pad.x;b.y=pad.y-pad.h/2-b.r-1;return;}
  const sf=slowT>0?.62:1,sp=Math.hypot(b.vx,b.vy);
  const n=Math.max(1,Math.ceil(sp*sf*dt/7)),sdt=dt/n*sf;
  for(let i=0;i<n;i++){b.x+=b.vx*sdt;b.y+=b.vy*sdt;collideBall(b);if(b.dead)break;}
  b.trail.push({x:b.x,y:b.y});if(b.trail.length>12)b.trail.shift();
}
function updateLasers(dt){
  const sp=880*S;
  for(const l of lasers){
    if(l.dead)continue;
    const n=Math.max(1,Math.ceil(sp*dt/7));
    for(let i=0;i<n&&!l.dead;i++){
      l.y-=sp/n*dt;
      if(l.y<TOPWALL-4){l.dead=true;sparkBurst(l.x,TOPWALL,3,'#ff8fa3',100*S);continue;}
      for(const rk of rocks.slice()){
        if(rk.dead)continue;
        const dx=l.x-rk.x,dy=l.y-rk.y;
        if(dx*dx+dy*dy<(rk.r+4)*(rk.r+4)){damageRock(rk);sparkBurst(l.x,l.y,6,'#ff8fa3',150*S);l.dead=true;break;}
      }
    }
  }
  lasers=lasers.filter(l=>!l.dead);
}
function update(dt){
  shake*=Math.exp(-6*dt);if(shake<.1)shake=0;
  redFlash=Math.max(0,redFlash-dt*1.4);whiteFlash=Math.max(0,whiteFlash-dt*2);
  wallFx.t=Math.max(0,wallFx.t-dt*3);laserCd-=dt;
  wideT=Math.max(0,wideT-dt);slowT=Math.max(0,slowT-dt);
  meteorT-=dt;
  if(!meteor&&meteorT<=0){meteor={x:rand(W*.15,W*.95),y:-30,vx:(rand(-160,160)-90)*S,vy:rand(320,480)*S,life:1.6};meteorT=rand(6,13);}
  if(meteor){meteor.x+=meteor.vx*dt;meteor.y+=meteor.vy*dt;meteor.life-=dt;if(meteor.life<=0||meteor.y>H+40)meteor=null;}
  for(let i=parts.length-1;i>=0;i--){const p=parts[i];p.life-=dt;
    if(p.life<=0){parts.splice(i,1);continue;}
    p.x+=p.vx*dt;p.y+=p.vy*dt;
    if(p.type==='chunk'){p.rot+=p.vr*dt;p.vx*=Math.exp(-1.6*dt);p.vy*=Math.exp(-1.6*dt);}
    else if(p.type==='spark'){p.vx*=Math.exp(-3*dt);p.vy*=Math.exp(-3*dt);}}
  if(parts.length>480)parts.splice(0,parts.length-480);
  for(let i=popups.length-1;i>=0;i--){const p=popups[i];p.life-=dt;p.y-=34*S*dt;if(p.life<=0)popups.splice(i,1);}

  if(state==='menu'||state==='over'){for(const rk of rocks)moveRock(rk,dt);return;}
  if(state==='clear'){clearT-=dt;
    if(clearT<=0){level++;buildLevel(level);balls=[makeStuck()];wideT=slowT=0;shield=false;state='serving';}
    return;}

  updatePaddle(dt);
  for(const b of balls)updateBall(b,dt);
  const before=balls.length;balls=balls.filter(b=>!b.dead);
  if(state==='play'&&before>0&&balls.length===0){loseLife();return;}
  updateLasers(dt);
  for(let i=drops.length-1;i>=0;i--){const d=drops[i];d.y+=d.vy*dt;d.x+=Math.sin(amb*3+d.ph)*14*S*dt;
    if(Math.abs(d.x-pad.x)<pad.w/2+15*S&&d.y>pad.y-pad.h/2-8*S&&d.y<pad.y+pad.h){applyDrop(d);drops.splice(i,1);continue;}
    if(d.y>H+24)drops.splice(i,1);}
  for(const rk of rocks)moveRock(rk,dt);
  /* soft separation */
  for(let i=0;i<rocks.length;i++)for(let j=i+1;j<rocks.length;j++){
    const A=rocks[i],B=rocks[j],min=(A.r+B.r)*.9;
    let dx=B.x-A.x,dy=B.y-A.y,d2=dx*dx+dy*dy;
    if(d2<min*min){const d=Math.sqrt(d2)||.01,nx=dx/d,ny=dy/d,f=(min-d)*dt*3;
      A.x-=nx*f;A.y-=ny*f;B.x+=nx*f;B.y+=ny*f;}}
  rocks=rocks.filter(r=>!r.dead);
  if(state==='play'&&rocks.length===0){
    state='clear';clearT=2.1;const bonus=250+level*150;score+=bonus;
    showBanner('SECTOR '+level+' CLEAR   +'+bonus);sfxClear();whiteFlash=Math.max(whiteFlash,.3);
  }
}
