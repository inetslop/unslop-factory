  msgText='P'+(s+1)+' SCORES!';msgColor=COLS[s];shake=15;
  burst(s===0?W-12:12,clamp(B.y,40,H-40),COLS[s],36,H*0.9);
  sndScore();
  if(score[s]>=target){winner=s;wins[s]++;saveWins();state=S.OVER;jingleWin();}
  else{serveDir=(s===0)?1:-1;state=S.POINT;pointT=1.25;}
}
function menuPoint(s){
  burst(s===0?W-12:12,clamp(B.y,40,H-40),'rgba(200,230,245,0.8)',14,H*0.5);
  beep(240,0.06,'triangle',0.3);
  resetBall(Math.random()<0.5?-1:1);
}

/* ============================== PADDLES / AI ============================== */
function movePaddle(i,up,dn,dt){
  const v=((dn?1:0)-(up?1:0))*L.psp;
  P[i].y=clamp(P[i].y+v*dt,L.top+L.padH/2,L.bot-L.padH/2);
}
function randErr(){return (Math.random()*2-1)*L.padH*(0.3+Math.random()*0.45);}
function aiMove(i,dt){
  const p=P[i],toward=i===0?B.vx<0:B.vx>0;
  const ty=toward?B.y+aiErr[i]:H*0.5;
  const sp=L.psp*0.8,d=ty-p.y;
  if(Math.abs(d)>3)p.y=clamp(p.y+Math.sign(d)*Math.min(sp*dt,Math.abs(d)),L.top+L.padH/2,L.bot-L.padH/2);
}

/* ============================== PARTICLES / STARS ============================== */
const parts=[];
function updParts(dt){
  for(let i=parts.length-1;i>=0;i--){const p=parts[i];p.t+=dt;
    if(p.t>p.life){parts.splice(i,1);continue;}
    p.vy+=(p.g||0)*dt;p.x+=p.vx*dt;p.y+=p.vy*dt;
    const dr=Math.exp(-1.5*dt);p.vx*=dr;}
}
const stars=[];
for(let i=0;i<80;i++)stars.push({x:Math.random(),y:Math.random(),z:Math.random()*0.7+0.3,ph:Math.random()*TAU});
function drawStars(){
  for(const s of stars){
    const sx=(((s.x+T*0.006*s.z+(B.x/W-0.5)*0.03*s.z)%1)+1)%1*W;
    const sy=s.y*H,tw=0.5+0.5*Math.sin(T*(0.8+s.z)+s.ph);
    ctx.fillStyle='rgba(170,220,240,'+((0.08+0.3*tw)*s.z).toFixed(3)+')';
    ctx.fillRect(sx,sy,s.z>0.75?2:1,s.z>0.75?2:1);
  }
}

function startMatch(){score=[0,0];winner=-1;bestRally=0;sfxVol=1;
  P[0].y=P[1].y=H/2;serveDir=Math.random()<0.5?-1:1;startCountdown();sndUI();}
function rematch(){score=[0,0];winner=-1;bestRally=0;sfxVol=1;
  P[0].y=P[1].y=H/2;serveDir=Math.random()<0.5?-1:1;startCountdown();sndUI();}
function toMenu(){state=S.MENU;score=[0,0];winner=-1;bestRally=0;sfxVol=0.3;
  P[0].y=P[1].y=H/2;resetBall(Math.random()<0.5?-1:1);aiErr=[randErr(),randErr()];sndUI();}
function togglePause(){
  if(state===S.PLAY||state===S.COUNT||state===S.POINT){prevState=state;state=S.PAUSE;sndUI();}
  else if(state===S.PAUSE){state=prevState;sndUI();}
}
function toggleMute(){muted=!muted;if(master)master.gain.value=muted?0:0.18;saveMute();
  sndMsg=muted?'SOUND OFF':'SOUND ON';sndMsgT=1.4;if(!muted)beep(660,0.07,'square',0.7);}
function primary(){
  if(state===S.MENU)startMatch();
  else if(state===S.OVER)rematch();
  else if(state===S.PAUSE)togglePause();
}

const keys=new Set();
window.addEventListener('keydown',e=>{
  const k=e.key;
  if(k==='ArrowUp'||k==='ArrowDown'||k===' '||k==='ArrowLeft'||k==='ArrowRight')e.preventDefault();
  audio();
  const lk=k.length===1?k.toLowerCase():k;
  keys.add(lk);
  if(e.repeat)return;
  if(k==='Enter'||k===' ')primary();
  else if(k==='Escape'||lk==='p')togglePause();
  else if(lk==='r')toMenu();
  else if(lk==='m')toggleMute();
});

window.addEventListener('blur',()=>{keys.clear();
  if(state===S.PLAY||state===S.COUNT||state===S.POINT){prevState=state;state=S.PAUSE;}});

/* ============================== UPDATE ============================== */
function update(dt){
  T+=dt;sfxVol=(state===S.MENU)?0.3:1;
  shake*=Math.exp(-5*dt);if(shake<0.05)shake=0;
  if(sndMsgT>0)sndMsgT-=dt;
  P[0].flash=Math.max(0,P[0].flash-dt*4);
  P[1].flash=Math.max(0,P[1].flash-dt*4);
  updParts(dt);
  switch(state){
    case S.MENU:
      aiMove(0,dt);aiMove(1,dt);stepBall(dt,true);break;
    case S.COUNT:
      cdT-=dt;const n=Math.ceil(Math.max(0,cdT));
      if(n!==lastCd&&n>0){lastCd=n;sndCount();}
      if(cdT<=0){state=S.PLAY;goT=0.6;launch();sndGo();}
      break;
    case S.PLAY:
      movePaddle(0,keys.has('w'),keys.has('s'),dt);
      movePaddle(1,keys.has('arrowup'),keys.has('arrowdown'),dt);
      stepBall(dt,false);
      if(goT>0)goT-=dt;
      break;
    case S.POINT:
      pointT-=dt;if(pointT<=0)startCountdown();break;
    case S.OVER:
      if(winner>=0){for(let k=0;k<2;k++)if(Math.random()<0.55)
        parts.push({x:Math.random()*W,y:-12,vx:(Math.random()-0.5)*90,vy:60+Math.random()*130,g:340,t:0,life:2+Math.random()*1.5,c:[COLS[winner],'#ffffff','#ffd76a'][k%3],sz:3+Math.random()*4});}
      break;
    case S.PAUSE:break;
  }
}

let bgGrad,rgGrad,gL,gR,bgKey='';
function makeBg(){
  const key=W+'|'+H;if(key===bgKey)return;bgKey=key;
  bgGrad=ctx.createLinearGradient(0,0,0,H);
  bgGrad.addColorStop(0,'#0c1a29');bgGrad.addColorStop(0.55,'#0e2334');bgGrad.addColorStop(1,'#081120');
  rgGrad=ctx.createRadialGradient(W/2,H*0.5,0,W/2,H*0.5,Math.max(W,H)*0.75);
  rgGrad.addColorStop(0,'rgba(90,170,200,0.10)');rgGrad.addColorStop(1,'rgba(0,0,0,0)');
  gL=ctx.createLinearGradient(0,0,W*0.16,0);
  gL.addColorStop(0,'rgba(63,227,255,0.08)');gL.addColorStop(1,'rgba(63,227,255,0)');
  gR=ctx.createLinearGradient(W,0,W*0.84,0);
  gR.addColorStop(0,'rgba(255,162,69,0.08)');gR.addColorStop(1,'rgba(255,162,69,0)');
}
function rr(x,y,w,h,r){r=Math.min(r,w/2,h/2);ctx.beginPath();
  ctx.moveTo(x+r,y);ctx.arcTo(x+w,y,x+w,y+h,r);ctx.arcTo(x+w,y+h,x,y+h,r);
  ctx.arcTo(x,y+h,x,y,r);ctx.arcTo(x,y,x+w,y,r);ctx.closePath();}
function ticks(x,y,w,h,c){
  const l=Math.min(16,w*0.15);ctx.strokeStyle=c;ctx.lineWidth=2;ctx.beginPath();
  [[x,y,1,1],[x+w,y,-1,1],[x,y+h,1,-1],[x+w,y+h,-1,-1]].forEach(p=>{
    ctx.moveTo(p[0]+p[2]*l,p[1]);ctx.lineTo(p[0],p[1]);ctx.lineTo(p[0],p[1]+p[3]*l);});
  ctx.stroke();
}
function drawNet(){
  ctx.fillStyle='rgba(150,215,235,0.22)';
  const dw=Math.max(3,W*0.004),dh=H*0.035,gap=H*0.026;
  for(let y=L.top+gap;y<L.bot-gap-dh;y+=dh+gap)ctx.fillRect(W/2-dw/2,y,dw,dh);
}
function drawPaddle(i){
  const x=i===0?L.padX:W-L.padX-L.padW,y=P[i].y-L.padH/2,fl=P[i].flash;
  ctx.save();ctx.shadowColor=COLS[i];ctx.shadowBlur=16+30*fl;
  rr(x,y,L.padW,L.padH,L.padW/2);ctx.fillStyle=COLS[i];ctx.fill();
  if(fl>0){ctx.shadowBlur=0;ctx.globalAlpha=fl*0.85;
    rr(x-2,y-2,L.padW+4,L.padH+4,(L.padW+4)/2);ctx.fillStyle='#ffffff';ctx.fill();}
  ctx.restore();
}
function ballColor(){return 'rgb(255,'+Math.round(lerp(255,168,B.hot))+','+Math.round(lerp(255,84,B.hot))+')';}
function drawBall(x,y,r){
  const col=ballColor();
  for(let i=0;i<B.trail.length;i++){const t=B.trail[i],k=i/B.trail.length;
    ctx.globalAlpha=k*k*0.35*(0.5+B.hot);ctx.fillStyle=col;
    ctx.beginPath();ctx.arc(t.x,t.y,B.r*(0.3+0.7*k),0,TAU);ctx.fill();}
  ctx.globalAlpha=1;ctx.save();
  ctx.shadowColor=col;ctx.shadowBlur=18+26*B.hot;ctx.fillStyle=col;
  ctx.beginPath();ctx.arc(x,y,r,0,TAU);ctx.fill();
  ctx.shadowBlur=0;ctx.fillStyle='rgba(255,255,255,0.9)';
  ctx.beginPath();ctx.arc(x-r*0.25,y-r*0.3,r*0.45,0,TAU);ctx.fill();
  ctx.restore();
}
