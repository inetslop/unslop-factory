  for(let i=0;i<p.grid.length;i++){
    p.grid[i]={mine:false,flagged:false,revealed:false,num:0};
    const el=document.createElement('div'); el.className='cell'; el.dataset.i=i;
    el.style.setProperty('--sd',(((i%p.cols)+((i/p.cols)|0))*7)+'ms');
    el.innerHTML='<span class="n"></span>'+FLAG_SVG+MINE_SVG;
    p.cellEls.push(el); frag.appendChild(el);
  }
  b.appendChild(frag);
  const idxs=[...Array(p.grid.length).keys()];
  for(let i=idxs.length-1;i>0;i--){const j=rand(i+1);[idxs[i],idxs[j]]=[idxs[j],idxs[i]];}
  for(let k=0;k<p.mines;k++)p.grid[idxs[k]].mine=true;
  recompute(p);
  p.cursor={x:p.cols>>1,y:p.rows>>1}; setCursor(p,true);
}
function inb(p,x,y){return x>=0&&y>=0&&x<p.cols&&y<p.rows;}
function recompute(p){
  for(let y=0;y<p.rows;y++)for(let x=0;x<p.cols;x++){
    let n=0;
    for(const[dx,dy]of DIRS){const nx=x+dx,ny=y+dy; if(inb(p,nx,ny)&&p.grid[ny*p.cols+nx].mine)n++;}
    p.grid[y*p.cols+x].num=n;
  }
}
function ensureSafe(p,x,y){
  const i=y*p.cols+x; if(!p.grid[i].mine)return;
  let j; do{ j=rand(p.grid.length); }while(p.grid[j].mine);
  p.grid[i].mine=false; p.grid[j].mine=true; recompute(p);
}
function flood(p,sx,sy){
  const out=[],seen=new Set(),q=[[sx,sy,0]]; seen.add(sy*p.cols+sx);
  while(q.length){
    const[x,y,d]=q.shift(), i=y*p.cols+x, c=p.grid[i];
    if(c.revealed||c.flagged)continue;
    out.push({c,x,y,d,i});
    if(c.mine||c.num>0)continue;
    for(const[dx,dy]of DIRS){
      const nx=x+dx,ny=y+dy; if(!inb(p,nx,ny))continue;
      const ni=ny*p.cols+nx; if(seen.has(ni))continue;
      seen.add(ni); q.push([nx,ny,d+1]);
    }
  }
  return out;
}
function applyFlood(p,list){
  let dead=false,deadI=-1;
  for(const o of list){
    const c=o.c; if(c.revealed)continue;
    c.revealed=true; if(!c.mine)p.revealedSafe++;
    const el=p.cellEls[o.i]; el.classList.add('rev');
    if(c.mine){el.classList.add('boom');dead=true;deadI=o.i;}
    else{
      if(c.num>0){el.classList.add('n'+c.num);el.querySelector('.n').textContent=c.num;}
      el.style.setProperty('--pd',Math.min(o.d*26,420)+'ms');
    }
  }
  return {dead,deadI};
}
function revealAt(p,x,y){
  if(state!=='playing'||p.dead)return;
  setCursorXY(p,x,y);
  const i=y*p.cols+x,c=p.grid[i];
  if(c.revealed){ if(!c.mine&&c.num>0)chordAt(p,x,y); return; }
  if(c.flagged){SFX.ui();return;}
  if(p.firstClick){p.firstClick=false;ensureSafe(p,x,y);}
  const r=applyFlood(p,flood(p,x,y));
  if(!r.dead)SFX.tick(p.grid[i].num||0);
  refreshHUD(p);
  if(r.dead){die(p,r.deadI%p.cols,(r.deadI/p.cols)|0);return;}
  if(p.revealedSafe>=p.totalSafe)win(p);
}
function chordAt(p,x,y){
  const c=p.grid[y*p.cols+x]; let fl=0;
  for(const[dx,dy]of DIRS){const nx=x+dx,ny=y+dy;if(inb(p,nx,ny)&&p.grid[ny*p.cols+nx].flagged)fl++;}
  if(fl!==c.num)return;
  SFX.chord();
  let r={dead:false,deadI:-1};
  for(const[dx,dy]of DIRS){
    const nx=x+dx,ny=y+dy; if(!inb(p,nx,ny))continue;
    const n=p.grid[ny*p.cols+nx]; if(n.flagged||n.revealed)continue;
    const fr=applyFlood(p,flood(p,nx,ny)); if(fr.dead)r=fr;
  }
  refreshHUD(p);
  if(r.dead){die(p,r.deadI%p.cols,(r.deadI/p.cols)|0);}
  else if(p.revealedSafe>=p.totalSafe)win(p);
}
function toggleFlagIdx(p,i){
  if(state!=='playing'||p.dead)return;
  const c=p.grid[i]; if(c.revealed)return;
  c.flagged=!c.flagged; p.flags+=c.flagged?1:-1;
  p.cellEls[i].classList.toggle('flg',c.flagged);
  c.flagged?SFX.flag():SFX.unflag();
  refreshHUD(p);
}

/* ================= round flow ================= */
function die(p,bx,by){
  p.dead=true; p.finalTime=performance.now()-roundStart;
  p.boardEl.classList.add('shake','dead');
  SFX.boom(); explosionFX(p,bx,by);
  let k=0;
  for(let i=0;i<p.grid.length;i++){
    const c=p.grid[i]; if(!c.mine||c.revealed)continue;
    c.revealed=true;
    const el=p.cellEls[i]; el.style.setProperty('--pd',(k*38)+'ms');
    el.classList.add('rev','boom'); k++;
  }
  setTimeout(()=>{
    if(state!=='playing')return;
    const o=other(p);
    if(o.dead){
      const w=p.revealedSafe===o.revealedSafe?null:(p.revealedSafe>o.revealedSafe?p:o);
      endRound(w,'both');
    } else endRound(o,'mine');
  },950);
}
function win(p){
  if(state!=='playing')return;
  p.won=true; p.finalTime=performance.now()-roundStart;
  const o=other(p); o.boardEl.classList.add('dead');
  let k=0;
  for(let i=0;i<o.grid.length;i++){
    const c=o.grid[i]; if(!c.mine||c.revealed)continue;
    c.revealed=true;
    const el=o.cellEls[i]; el.style.setProperty('--pd',(k*30)+'ms');
    el.classList.add('rev','dmg'); k++;
  }
  endRound(p,'clear');
}
function endRound(winner,reason){
  state='over';
  const now=performance.now();
  players.forEach(pl=>{if(pl.finalTime==null)pl.finalTime=now-roundStart;});
  refreshHUD(P1);refreshHUD(P2);
  P1.panel.classList.remove('lead');P2.panel.classList.remove('lead');
  if(winner){
    score[winner.id]++; save('s'+winner.num,score[winner.id]);
    winner.panel.classList.add('won'); SFX.win(); confetti(winner);
  } else SFX.draw();
  $('#sc1').textContent=score.p1; $('#sc2').textContent=score.p2;
  setTimeout(()=>{ if(state!=='over')return; fillOver(winner,reason); show($('#over')); },750);
}
function fillOver(winner,reason){
  const t=pl=>(pl.finalTime/1000).toFixed(1)+'s';
  const pct=pl=>Math.round(pl.revealedSafe/pl.totalSafe*100)+'%';
  const title=$('#resTitle');
  if(!winner){title.textContent='STALEMATE';title.className='wd';}
  else{title.textContent=winner.name+' WINS';title.className=winner===P1?'w1':'w2';}
  let msg;
  if(reason==='clear') msg=`${winner.name} swept the board clean in ${t(winner)}.`;
  else if(reason==='mine'){const l=other(winner);msg=`${l.name} kissed a mine — ${winner.name} takes the round.`;}
  else msg=winner?`Both boards blew up. ${winner.name} had revealed more (${pct(winner)} vs ${pct(other(winner))}).`
                :`Both boards blew up at identical coverage.`;
  $('#resReason').textContent=msg;
  $('#stT1').textContent=t(P1); $('#stT2').textContent=t(P2);
  $('#stC1').textContent=pct(P1); $('#stC2').textContent=pct(P2);
  $('#stF1').textContent=P1.flags; $('#stF2').textContent=P2.flags;
  $('#osl1').textContent=score.p1; $('#osl2').textContent=score.p2;
}

const parts=[],rings=[],embers=[];
for(let i=0;i<16;i++)embers.push({x:Math.random()*innerWidth,y:Math.random()*innerHeight,
  vy:8+Math.random()*14,r:1+Math.random()*1.5,a:.05+Math.random()*.09,

function confetti(w){
  const r=w.boardEl.getBoundingClientRect();
  const cx=r.left+r.width/2, cy=r.top+r.height*.4;
  const cols=[getComputedStyle(w.panel).getPropertyValue('--pc').trim()||'#fff','#ffffff','#ffe08a'];
  for(let i=0;i<130;i++){
    const a=Math.random()*6.28, sp=120+Math.random()*420;
    parts.push({x:cx,y:cy,vx:Math.cos(a)*sp,vy:Math.sin(a)*sp-220,g:900,life:1.6+Math.random()*1.4,
      t:0,s:3.5+Math.random()*4,c:cols[i%3],rot:Math.random()*6.28,vr:(Math.random()-.5)*12});
  }
  rings.push({x:cx,y:cy,r:10,vr:700,a:.9,c:'#ffffff'});
}
function explosionFX(p,bx,by){
  const el=p.cellEls[by*p.cols+bx]; if(!el)return;
  const r=el.getBoundingClientRect(), cx=r.left+r.width/2, cy=r.top+r.height/2;
  const cols=['#ff5b6a','#ffb52e','#fff3d6'];
  for(let i=0;i<48;i++){
    const a=Math.random()*6.28, sp=60+Math.random()*380;
    parts.push({x:cx,y:cy,vx:Math.cos(a)*sp,vy:Math.sin(a)*sp,g:700,life:.5+Math.random()*.8,t:0,
      s:2+Math.random()*3.5,c:cols[i%3],rot:0,vr:0,spark:true});
  }
  rings.push({x:cx,y:cy,r:6,vr:900,a:.95,c:'#ff5b6a'});
}
function stepFX(dt,now){
  for(let i=parts.length-1;i>=0;i--){const q=parts[i];q.t+=dt;
    if(q.t>q.life){parts.splice(i,1);continue;}
    q.vy+=(q.g||0)*dt;q.x+=q.vx*dt;q.y+=q.vy*dt;q.rot+=q.vr*dt;}
  for(let i=rings.length-1;i>=0;i--){const r=rings[i];r.r+=r.vr*dt;r.a-=dt*1.6;
    if(r.a<=0)rings.splice(i,1);}
  for(const e of embers){e.y-=e.vy*dt;if(e.y<-12){e.y=innerHeight+12;e.x=Math.random()*innerWidth;}}
}
function drawFX(now){
  ctx.clearRect(0,0,innerWidth,innerHeight);
  const ts=now/1000;
  for(const e of embers){
    ctx.globalAlpha=e.a*(.6+.4*Math.sin(ts*2+e.ph));
    ctx.fillStyle=e.c;ctx.beginPath();ctx.arc(e.x,e.y,e.r,0,7);ctx.fill();
  }
  for(const r of rings){
    ctx.globalAlpha=Math.max(0,r.a);ctx.strokeStyle=r.c;ctx.lineWidth=3;
    ctx.beginPath();ctx.arc(r.x,r.y,r.r,0,7);ctx.stroke();
  }
  for(const q of parts){
    const a=1-q.t/q.life;
    ctx.globalAlpha=a;ctx.fillStyle=q.c;
    if(q.spark){ctx.fillRect(q.x,q.y,q.s,q.s);}
    else{ctx.save();ctx.translate(q.x,q.y);ctx.rotate(q.rot);ctx.fillRect(-q.s/2,-q.s/2,q.s,q.s*.6);ctx.restore();}
  }
  ctx.globalAlpha=1;
}
