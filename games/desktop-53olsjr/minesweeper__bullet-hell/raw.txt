function step(rdt){
  if(mode==='pause')return;
  const dt=rdt*timeScale; tGlobal+=dt;
  shake=Math.max(0,shake-shake*5.5*dt-3*dt);

  for(let i=waves.length-1;i>=0;i--){const w=waves[i];w.r+=w.spd*dt;if(w.r>w.maxR)waves.splice(i,1);}

    while(e.acc>=e.interval&&g++<8){e.acc-=e.interval;
      bullets.push(bullet(e.x,e.y,e.a,e.spd,'spiral'));bullets.push(bullet(e.x,e.y,e.a+Math.PI,e.spd,'spiral'));e.a+=e.spin;}
    if(e.left<=0)emitters.splice(i,1);}
  for(let i=parts.length-1;i>=0;i--){const p=parts[i];p.t+=dt;if(p.t>p.dur){parts.splice(i,1);continue;}
    p.x+=p.vx*dt;p.y+=p.vy*dt;const dr=Math.max(0,1-2.5*dt);p.vx*=dr;p.vy*=dr;}
  for(let i=floats.length-1;i>=0;i--){floats[i].t+=dt;if(floats[i].t>.95)floats.splice(i,1);}
  if(sonarPulse){sonarPulse.r+=sonarPulse.spd*dt;if(sonarPulse.r>sonarPulse.maxR)sonarPulse=null;}

  if(mode==='play'&&!ship.dead){
    elapsed+=dt;
    const f=!!keys['shift'];
    let ax=((keys['d']||keys['arrowright'])?1:0)-((keys['a']||keys['arrowleft'])?1:0);
    let ay=((keys['s']||keys['arrowdown'])?1:0)-((keys['w']||keys['arrowup'])?1:0);
    if(ax&&ay){ax*=.7071;ay*=.7071;}
    const sp=f?2.05:4.6,acc=Math.min(1,dt*(f?14:9));
    ship.vx+=(ax*sp-ship.vx)*acc;ship.vy+=(ay*sp-ship.vy)*acc;
    ship.x=clamp(ship.x+ship.vx*dt,.28,GW-.28);ship.y=clamp(ship.y+ship.vy*dt,.28,GH-.28);
    if(Math.hypot(ship.vx,ship.vy)>.4)ship.dir=Math.atan2(ship.vy,ship.vx);
    ship.trail.unshift({x:ship.x,y:ship.y});if(ship.trail.length>15)ship.trail.pop();

    const c=cells[ship.y|0][ship.x|0];

      else if(!c.mine)revealCell(c);
    }
    for(const row of cells)for(const cc of row){
      if(cc.markT>0)cc.markT-=dt;
      if(cc.anim>0)cc.anim=Math.max(0,cc.anim-dt*3);
      if(cc.crackT>0&&cc.crackT<2)cc.crackT=Math.min(2,Math.max(0,cc.crackT-dt*.8));
      if(cc.arming){cc.fuse-=dt;const p=1-cc.fuse/cc.fuseMax;cc.ct+=dt;
        const iv=Math.max(.05,.16-p*.09);if(cc.ct>=iv){cc.ct=0;sArm(p);}
        if(cc.fuse<=0)explode(cc);}
    }
    for(let i=pulsers.length-1;i>=0;i--){const p=pulsers[i];
      if(p.c.detonated){pulsers.splice(i,1);continue;}
      p.t-=dt;if(p.t<=0&&!p.c.arming){p.c.arming=true;p.c.fuse=PULSE_FUSE;p.c.fuseMax=PULSE_FUSE;p.c.ct=0;}}
    sonarCh=Math.min(3,sonarCh+dt/7);
    if(invuln>0)invuln-=dt;

    const hitR=f?.09:.16;
    for(let i=bullets.length-1;i>=0;i--){const b=bullets[i];b.x+=b.vx*dt;b.y+=b.vy*dt;
      if(b.x<-.4||b.x>GW+.4||b.y<-.4||b.y>GH+.4){bullets.splice(i,1);continue;}
      if(invuln<=0){const dx=b.x-ship.x,dy=b.y-ship.y,rr=b.r+hitR;

    for(const w of waves){const d=Math.hypot(ship.x-w.x,ship.y-w.y);
      if(Math.abs(d-w.r)<.15+hitR*.6){die('MINE BLAST');break;}}
  }
  else if(mode==='clear'||mode==='over'||ship.dead){ // let loose bullets fly out
    for(let i=bullets.length-1;i>=0;i--){const b=bullets[i];b.x+=b.vx*dt;b.y+=b.vy*dt;
      if(b.x<-.4||b.x>GW+.4||b.y<-.4||b.y>GH+.4)bullets.splice(i,1);}
  }
  if(bullets.length>420)bullets.splice(0,bullets.length-420);

  if(mode==='clear'){ship.vx*=Math.max(0,1-2*dt);ship.vy*=Math.max(0,1-2*dt);

  else timeScale+=(1-timeScale)*Math.min(1,rdt*3);
}
