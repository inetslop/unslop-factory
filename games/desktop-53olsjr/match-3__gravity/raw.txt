function roundRectPath(g,x,y,w,h,r){g.moveTo(x+r,y);g.arcTo(x+w,y,x+w,y+h,r);g.arcTo(x+w,y+h,x,y+h,r);g.arcTo(x,y+h,x,y,r);g.arcTo(x,y,x+w,y,r);g.closePath();}
function shapePath(g,i,cx,cy,r){g.beginPath();
  if(i===0){g.arc(cx,cy,r,0,Math.PI*2);}
  else if(i===1){g.moveTo(cx,cy-r);g.lineTo(cx+r*0.92,cy);g.lineTo(cx,cy+r);g.lineTo(cx-r*0.92,cy);g.closePath();}
  else if(i===2){const a=r*0.86;roundRectPath(g,cx-a,cy-a,a*2,a*2,a*0.32);}
  else if(i===3){for(let k=0;k<6;k++){const ang=-Math.PI/2+k*Math.PI/3,px=cx+Math.cos(ang)*r,py=cy+Math.sin(ang)*r;k?g.lineTo(px,py):g.moveTo(px,py);}g.closePath();}
  else{g.moveTo(cx,cy-r);g.lineTo(cx+r*0.95,cy+r*0.72);g.lineTo(cx-r*0.95,cy+r*0.72);g.closePath();}}
function buildSprites(){
  sprites=[];
  const s=Math.ceil(cell*1.3),size=cell*0.8,r=size/2*0.94;
  for(let i=0;i<NCOLORS;i++){
    const col=COLORS[i],cv=document.createElement('canvas');cv.width=cv.height=s*dpr;
    const g=cv.getContext('2d');g.scale(dpr,dpr);g.translate(s/2,s/2);g.lineJoin='round';
    g.shadowColor=col.base;g.shadowBlur=cell*0.18;
    const grad=g.createRadialGradient(-r*0.35,-r*0.4,r*0.1,0,0,r*1.05);
    grad.addColorStop(0,col.light);grad.addColorStop(0.55,col.base);grad.addColorStop(1,col.dark);
    g.fillStyle=grad;shapePath(g,i,0,0,r);g.fill();
    g.shadowBlur=0;
    g.lineWidth=Math.max(1.5,cell*0.03);g.strokeStyle='rgba(255,255,255,0.35)';shapePath(g,i,0,0,r*0.96);g.stroke();
    g.fillStyle='rgba(255,255,255,0.35)';g.beginPath();g.ellipse(-r*0.3,-r*0.42,r*0.34,r*0.2,-0.6,0,Math.PI*2);g.fill();
    sprites[i]={cv,s};
  }
}

/* ============ board logic ============ */
function makeGem(color){return{color,special:0,x:0,y:0,vy:0,r:0,c:0,spawn:1,sq:0};}
function snapGems(){for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++){const g=grid[r]&&grid[r][c];if(g){g.x=colCenter(c);g.y=rowCenter(r);g.vy=0;}}}
function newBoard(){
  grid=[];particles.length=0;rings.length=0;popups.length=0;shake=0;
  for(let r=0;r<ROWS;r++){const row=[];
    for(let c=0;c<COLS;c++){let col;
      do{col=rand(NCOLORS);}while(
        (c>=2&&row[c-1]&&row[c-2]&&row[c-1].color===col&&row[c-2].color===col)||
        (r>=2&&grid[r-1][c]&&grid[r-2][c]&&grid[r-1][c].color===col&&grid[r-2][c].color===col));
      row.push(makeGem(col));}
    grid.push(row);}
  snapGems();
}
function findRuns(){
  const runs=[];
  for(let r=0;r<ROWS;r++){let c=0;while(c<COLS){const g=grid[r][c];if(!g){c++;continue;}let e=c+1;while(e<COLS&&grid[r][e]&&grid[r][e].color===g.color)e++;if(e-c>=3)runs.push({dir:0,fixed:r,start:c,len:e-c});c=e;}}
  for(let c=0;c<COLS;c++){let r=0;while(r<ROWS){const g=grid[r][c];if(!g){r++;continue;}let e=r+1;while(e<ROWS&&grid[e][c]&&grid[e][c].color===g.color)e++;if(e-r>=3)runs.push({dir:1,fixed:c,start:r,len:e-r});r=e;}}
  return runs;
}
function expandClear(initial){
  const set=new Set(initial),queue=[...set];
  while(queue.length){
    const k=queue.pop(),r=(k/COLS)|0,c=k%COLS,g=grid[r][c];if(!g)continue;
    if(g.special===1){
      for(let i=0;i<COLS;i++){const kk=r*COLS+i;if(grid[r][i]&&!set.has(kk)){set.add(kk);queue.push(kk);}}
      for(let i=0;i<ROWS;i++){const kk=i*COLS+c;if(grid[i][c]&&!set.has(kk)){set.add(kk);queue.push(kk);}}
    }else if(g.special===2){
      for(let rr=0;rr<ROWS;rr++)for(let cc=0;cc<COLS;cc++){const gg=grid[rr][cc],kk=rr*COLS+cc;if(gg&&gg.color===g.color&&!set.has(kk)){set.add(kk);queue.push(kk);}}
    }
  }
  return set;
}
function planCreations(runs,a,b){
  const out=[],used=new Array(runs.length).fill(false);
  const inRun=(run,r,c)=>run.dir===0?(r===run.fixed&&c>=run.start&&c<run.start+run.len):(c===run.fixed&&r>=run.start&&r<run.start+run.len);
  for(let i=0;i<runs.length;i++)for(let j=i+1;j<runs.length;j++){
    const A=runs[i],B=runs[j];if(A.dir===B.dir)continue;
    const h=A.dir===0?A:B,v=A.dir===0?B:A,r=h.fixed,c=v.fixed;
    if(c>=h.start&&c<h.start+h.len&&r>=v.start&&r<v.start+v.len&&h.len+v.len-1>=5){out.push({r,c,type:2});used[i]=true;used[j]=true;}
  }
  runs.forEach((run,i)=>{if(used[i])return;let pos=null;
    for(const p of[a,b])if(p&&inRun(run,p.r,p.c))pos={r:p.r,c:p.c};
    if(!pos){if(run.dir===0)pos={r:run.fixed,c:run.start+((run.len-1)>>1)};else pos={c:run.fixed,r:run.start+((run.len-1)>>1)};}
    out.push({r:pos.r,c:pos.c,type:run.len>=5?2:1});});
  const map=new Map();out.forEach(o=>{const k=o.r*COLS+o.c,ex=map.get(k);if(!ex||o.type>ex.type)map.set(k,o);});
  return[...map.values()];
}
function startClearFromRuns(runs){
  const keys=new Set();
  runs.forEach(run=>{for(let i=0;i<run.len;i++){const r=run.dir===0?run.fixed:run.start+i,c=run.dir===1?run.fixed:run.start+i;keys.add(r*COLS+c);}});
  beginClear(keys,runs);
}
function startClearFromSeeds(gems){
  const keys=new Set();gems.forEach(g=>{if(g)keys.add(g.r*COLS+g.c);});
  beginClear(keys,null);
}
function beginClear(keys,runs){
  combo++;
  const set=expandClear(keys);
  let creations=[];
  if(runs)creations=planCreations(runs,swapAnim&&swapAnim.a,swapAnim&&swapAnim.b);
  const createdKeys=new Set();
  creations.forEach(cr=>{const k=cr.r*COLS+cr.c;if(set.has(k)){set.delete(k);createdKeys.add(k);}});
  pendingCreations=creations.filter(cr=>createdKeys.has(cr.r*COLS+cr.c));
  clearCells=[...set].map(k=>({r:(k/COLS)|0,c:k%COLS}));
  clearKeySet=new Set([...set]);
  let sp=0;clearCells.forEach(cl=>{const g=grid[cl.r][cl.c];if(g&&g.special)sp++;});
  pendingGain=clearCells.length*20*combo+sp*150;
  phase='clear';clearT=0;
  if(state==='play'){
    S.clear(combo);
    if(sp){S.boom();shake=Math.min(20,shake+6*sp);}
    if(clearCells.length>=8)shake=Math.min(20,shake+4);
    if(combo>=2)addPopup('COMBO ×'+combo,boardX+boardW/2,boardY+cell*1.15,22+Math.min(16,combo*3),'#ffd23f');
  }
}
function finishClear(){
  clearCells.forEach(cl=>{const g=grid[cl.r][cl.c];if(g){burst(g.x,g.y,COLORS[g.color].base);grid[cl.r][cl.c]=null;}});
  pendingCreations.forEach(cr=>{const g=grid[cr.r][cr.c];if(g){g.special=cr.type;g.spawn=0;rings.push({x:g.x,y:g.y,r0:cell*0.2,maxR:cell*1.15,t:0,life:0.5,col:'#ffffff'});}});
  if(pendingCreations.length&&state==='play')S.create();
  if(state==='play'){score+=pendingGain;addPopup('+'+pendingGain,boardX+boardW/2,boardY+cell*0.45,18,'#ffe9c9');}
  applyGravity();phase='fall';clearCells=null;clearKeySet=null;
}
function applyGravity(){
  for(let c=0;c<COLS;c++){
    const stack=[];
    for(let r=ROWS-1;r>=0;r--)if(grid[r][c])stack.push(grid[r][c]);
    const n=ROWS-stack.length;
    for(let r=ROWS-1,i=0;r>=0;r--,i++){
      if(i<stack.length){const g=stack[i];grid[r][c]=g;g.r=r;g.c=c;}
      else{const g=makeGem(rand(NCOLORS));grid[r][c]=g;g.r=r;g.c=c;g.x=colCenter(c);g.y=boardY-(n-r)*cell;g.vy=0;g.spawn=0;}
    }
  }
}
function updateFall(dt){
  let settled=true;const G=3600+cell*14;
  for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++){
    const g=grid[r][c];if(!g)continue;
    const ty=rowCenter(r);
    if(g.y<ty-0.5){settled=false;g.vy+=G*dt;g.y+=g.vy*dt;if(g.y>=ty){g.y=ty;g.vy=0;g.sq=1;}}
    else{g.y=ty;g.vy=0;}
    g.x=colCenter(c);
  }
  return settled;
}
function endTurn(){
  phase='idle';
  if(state!=='play'){combo=0;return;}
  if(score>=target){levelClear();return;}
  if(moves<=0){gameOver();return;}
  combo=0;
}

/* ============ turns / flow ============ */
function trySwap(r1,c1,r2,c2){
  if((state!=='play'&&state!=='menu')||phase!=='idle')return;
  if(Math.abs(r1-r2)+Math.abs(c1-c2)!==1)return;
  const A=grid[r1][c1],B=grid[r2][c2];if(!A||!B)return;
  grid[r1][c1]=B;grid[r2][c2]=A;A.r=r2;A.c=c2;B.r=r1;B.c=c1;
  swapAnim={a:A,b:B,f0:[A.x,A.y],f1:[B.x,B.y],t0:[colCenter(c2),rowCenter(r2)],t1:[colCenter(c1),rowCenter(r1)],t:0,dur:0.16,back:false};
  phase='swap';selected=null;
  if(state==='play')S.swap();
}
function afterSwap(){
  const runs=findRuns();
  if(runs.length){if(state==='play')moves--;startClearFromRuns(runs);}
  else if(swapAnim.a.special||swapAnim.b.special){if(state==='play')moves--;startClearFromSeeds([swapAnim.a,swapAnim.b]);}
  else{swapAnim.back=true;swapAnim.t=0;phase='swapback';if(state==='play')S.bad();}
}
function matchesAt(r,c){
  const col=grid[r][c].color;let n=1;
  for(let i=c-1;i>=0&&grid[r][i]&&grid[r][i].color===col;i--)n++;
  for(let i=c+1;i<COLS&&grid[r][i]&&grid[r][i].color===col;i++)n++;
  if(n>=3)return true;n=1;
  for(let i=r-1;i>=0&&grid[i][c]&&grid[i][c].color===col;i--)n++;
  for(let i=r+1;i<ROWS&&grid[i][c]&&grid[i][c].color===col;i++)n++;
  return n>=3;
}
function wouldMatch(r1,c1,r2,c2){
  const A=grid[r1][c1],B=grid[r2][c2];
  grid[r1][c1]=B;grid[r2][c2]=A;
  const ok=matchesAt(r1,c1)||matchesAt(r2,c2);
  grid[r1][c1]=A;grid[r2][c2]=B;return ok;
}
function autoSwap(){
  for(let i=0;i<40;i++){
    const c=rand(COLS),r=rand(ROWS),useR=Math.random()<0.5,dr=Math.random()<0.5?1:-1,dc=Math.random()<0.5?1:-1;
    const t={r:r+(useR?dr:0),c:c+(useR?0:dc)};
    if(t.r<0||t.r>=ROWS||t.c<0||t.c>=COLS)continue;
    if(wouldMatch(r,c,t.r,t.c)){trySwap(r,c,t.r,t.c);return;}
  }
}
function startLevel(){
  moves=20;target=1400+(level-1)*900;combo=0;selected=null;swapAnim=null;clearCells=null;clearKeySet=null;
  newBoard();phase='idle';paused=false;state='play';hideAllScreens();
}
function newGame(){score=0;level=1;startLevel();}
function levelClear(){
  const bonus=moves*40;score+=bonus;saveBest();state='levelup';
  $('winLvl').textContent=level;
  $('winStats').innerHTML='SCORE <b>'+score+'</b><br>MOVE BONUS <b>+'+bonus+'</b>';
  show($('scr-win'));S.win();shake=10;
}
function gameOver(){
  saveBest();state='over';$('finalScore').textContent=score;
  $('newBest').classList.toggle('hidden',!(score>=best&&score>0));
  show($('scr-over'));S.lose();
}
function showMenu(){
  state='menu';paused=false;combo=0;selected=null;swapAnim=null;clearCells=null;clearKeySet=null;menuTimer=0.5;
  $('bestVal').textContent=best;newBoard();phase='idle';show($('scr-start'));
}

function addPopup(text,x,y,size,col){popups.push({text,x,y,size,col,t:0,life:0.95});}
function burst(x,y,col){
  rings.push({x,y,r0:cell*0.15,maxR:cell*(0.8+Math.random()*0.4),t:0,life:0.35,col});
  for(let i=0;i<9;i++){const a=Math.random()*Math.PI*2,sp=60+Math.random()*220;
    particles.push({x,y,vx:Math.cos(a)*sp,vy:Math.sin(a)*sp-40,t:0,life:0.45+Math.random()*0.35,size:1.5+Math.random()*2.6,col});}
  if(particles.length>350)particles.splice(0,particles.length-350);
}
const embers=[];for(let i=0;i<48;i++)embers.push({x:Math.random(),y:Math.random(),r:1+Math.random()*2.4,s:8+Math.random()*26,a:0.14+Math.random()*0.3,hue:18+Math.random()*26,seed:Math.random()*10});

/* ============ update ============ */
function update(dt){
  embers.forEach(e=>{e.y-=e.s*dt/H;e.x+=Math.sin(timeAcc*0.7+e.seed)*8*dt/W;if(e.y<-0.02){e.y=1.02;e.x=Math.random();}});
  for(let i=particles.length-1;i>=0;i--){const p=particles[i];p.t+=dt;if(p.t>=p.life){particles.splice(i,1);continue;}p.x+=p.vx*dt;p.y+=p.vy*dt;p.vy+=520*dt;}
  for(let i=rings.length-1;i>=0;i--){rings[i].t+=dt;if(rings[i].t>=rings[i].life)rings.splice(i,1);}
  for(let i=popups.length-1;i>=0;i--){popups[i].t+=dt;if(popups[i].t>=popups[i].life)popups.splice(i,1);}
  shake=Math.max(0,shake-dt*36);
  if(paused||(state!=='play'&&state!=='menu'))return;
  for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++){const g=grid[r]&&grid[r][c];if(!g)continue;if(g.spawn<1)g.spawn=Math.min(1,g.spawn+dt/0.24);if(g.sq>0)g.sq=Math.max(0,g.sq-dt*5);}
  if(phase==='swap'||phase==='swapback'){
    const an=swapAnim;an.t+=dt/an.dur;const p=smooth(Math.min(1,an.t));
    an.a.x=lerp(an.f0[0],an.t0[0],p);an.a.y=lerp(an.f0[1],an.t0[1],p);
    an.b.x=lerp(an.f1[0],an.t1[0],p);an.b.y=lerp(an.f1[1],an.t1[1],p);
    if(an.t>=1){
      if(!an.back){
        an.a.x=colCenter(an.a.c);an.a.y=rowCenter(an.a.r);an.b.x=colCenter(an.b.c);an.b.y=rowCenter(an.b.r);
        afterSwap();
      }else{
        const A=an.a,B=an.b;
        grid[A.r][A.c]=B;grid[B.r][B.c]=A;
        const tr=A.r,tc=A.c;A.r=B.r;A.c=B.c;B.r=tr;B.c=tc;
        A.x=colCenter(A.c);A.y=rowCenter(A.r);B.x=colCenter(B.c);B.y=rowCenter(B.r);
        swapAnim=null;phase='idle';
      }
    }
  }else if(phase==='clear'){clearT+=dt;if(clearT>=CLEAR_DUR)finishClear();}
  else if(phase==='fall'){if(updateFall(dt)){const runs=findRuns();if(runs.length)startClearFromRuns(runs);else endTurn();}}
  else if(phase==='idle'&&state==='menu'){menuTimer+=dt;if(menuTimer>1.0){menuTimer=0;autoSwap();}}
}

function drawBG(){
  const g=ctx.createLinearGradient(0,0,0,H);
  g.addColorStop(0,'#1b1114');g.addColorStop(0.55,'#120c10');g.addColorStop(1,'#0b0709');
  ctx.fillStyle=g;ctx.fillRect(0,0,W,H);
  const hg=ctx.createLinearGradient(0,H-140,0,H);
  hg.addColorStop(0,'rgba(255,80,30,0)');hg.addColorStop(1,'rgba(255,80,30,0.10)');
  ctx.fillStyle=hg;ctx.fillRect(0,H-140,W,140);
  ctx.globalCompositeOperation='lighter';
  embers.forEach(e=>{
    const fl=0.6+0.4*Math.sin(timeAcc*3+e.seed*7);
    ctx.fillStyle='hsla('+e.hue+',95%,62%,'+(e.a*fl).toFixed(3)+')';
    ctx.beginPath();ctx.arc(e.x*W,e.y*H,e.r,0,Math.PI*2);ctx.fill();
  });
  ctx.globalCompositeOperation='source-over';
  const v=ctx.createRadialGradient(W/2,H*0.45,Math.min(W,H)*0.3,W/2,H*0.5,Math.max(W,H)*0.75);
  v.addColorStop(0,'rgba(0,0,0,0)');v.addColorStop(1,'rgba(0,0,0,0.5)');
  ctx.fillStyle=v;ctx.fillRect(0,0,W,H);
}

}
function drawGemAt(g){
  const sp=sprites[g.color];if(!sp)return;
  const inClear=phase==='clear'&&clearKeySet&&clearKeySet.has(g.r*COLS+g.c);
  let sc=1,sx=1,sy=1,flash=0;
  if(g.spawn<1)sc*=easeBack(Math.min(1,g.spawn));
  sc*=1+0.02*Math.sin(timeAcc*2.3+(g.r*3.7+g.c*1.9));
  const isSel=selected&&selected.r===g.r&&selected.c===g.c;
  const isHov=hoverCell&&phase==='idle'&&state==='play'&&!paused&&hoverCell.r===g.r&&hoverCell.c===g.c;
  if(isSel)sc*=1.08;else if(isHov)sc*=1.05;
  if(g.sq>0){sy=sc*(1-0.24*g.sq);sx=sc*(1+0.2*g.sq);}else{sx=sy=sc;}
  if(inClear){const p=Math.min(1,clearT/CLEAR_DUR);sx*=(1-p*p);sy*=(1-p*p);flash=Math.max(0,0.9-p*2.4);}
  ctx.save();ctx.translate(g.x,g.y);
  if(g.special){
    ctx.globalCompositeOperation='lighter';
    const gg=ctx.createRadialGradient(0,0,0,0,0,cell*0.55);
    gg.addColorStop(0,g.special===2?'rgba(255,255,255,0.3)':'rgba(255,180,90,0.3)');
    gg.addColorStop(1,'rgba(255,150,60,0)');
    ctx.fillStyle=gg;ctx.beginPath();ctx.arc(0,0,cell*0.55,0,Math.PI*2);ctx.fill();
    ctx.globalCompositeOperation='source-over';
  }
  ctx.scale(sx,sy);
  ctx.drawImage(sp.cv,-sp.s/2,-sp.s/2,sp.s,sp.s);
  if(g.special===1){
    ctx.strokeStyle='rgba(255,255,255,0.9)';ctx.lineWidth=Math.max(1.5,cell*0.045);
    ctx.setLineDash([cell*0.18,cell*0.12]);ctx.lineDashOffset=-timeAcc*cell*0.6;
    ctx.beginPath();ctx.arc(0,0,cell*0.33,0,Math.PI*2);ctx.stroke();ctx.setLineDash([]);
    ctx.strokeStyle='rgba(255,240,200,0.75)';ctx.lineCap='round';
    ctx.beginPath();ctx.moveTo(-cell*0.26,0);ctx.lineTo(cell*0.26,0);ctx.moveTo(0,-cell*0.26);ctx.lineTo(0,cell*0.26);ctx.stroke();
  }else if(g.special===2){
    ctx.lineCap='round';
    for(let i=0;i<NCOLORS;i++){
      const a0=i*(Math.PI*2/NCOLORS)+timeAcc*1.8;
      ctx.strokeStyle=COLORS[i].base;ctx.lineWidth=Math.max(2,cell*0.05);
      ctx.beginPath();ctx.arc(0,0,cell*0.36,a0,a0+(Math.PI*2/NCOLORS)*0.72);ctx.stroke();
    }
  }
  if(flash>0){
    ctx.globalCompositeOperation='lighter';
    ctx.fillStyle='rgba(255,240,210,'+flash.toFixed(3)+')';
    ctx.beginPath();ctx.arc(0,0,cell*0.42,0,Math.PI*2);ctx.fill();
    ctx.globalCompositeOperation='source-over';
  }
  ctx.restore();
}
function drawGems(){
  ctx.save();
  roundRectPath(ctx,boardX-8,boardY-10,boardW+16,boardH+26,14);ctx.clip();
  for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++){const g=grid[r][c];if(g)drawGemAt(g);}
  if(swapAnim&&(phase==='swap'||phase==='swapback')){drawGemAt(swapAnim.a);drawGemAt(swapAnim.b);}
  ctx.restore();
  if(state==='play'&&!paused){
    if(selected&&phase==='idle')drawRing(selected.r,selected.c,'rgba(255,210,63,',0.95);
    if(kbActive)drawRing(cursor.r,cursor.c,'rgba(72,224,255,',0.85);
  }
}
function drawRing(r,c,col,alpha){
  const x=boardX+c*cell+2.5,y=boardY+r*cell+2.5,w=cell-5;
  ctx.save();ctx.strokeStyle=col+alpha+')';ctx.lineWidth=3;ctx.shadowColor=col+'0.9)';ctx.shadowBlur=10;
  roundRectPath(ctx,x,y,w,w,Math.max(6,cell*0.14));ctx.stroke();ctx.restore();
}
function drawRings(){
  rings.forEach(rg=>{
    const p=rg.t/rg.life,r=lerp(rg.r0,rg.maxR,smooth(p));
    ctx.strokeStyle=rg.col;ctx.globalAlpha=(1-p)*0.8;ctx.lineWidth=Math.max(1.5,cell*0.06*(1-p)+1);
    ctx.beginPath();ctx.arc(rg.x,rg.y,r,0,Math.PI*2);ctx.stroke();ctx.globalAlpha=1;
  });
}
function drawParticles(){
  ctx.save();ctx.globalCompositeOperation='lighter';
  particles.forEach(p=>{
    const q=p.t/p.life;
    ctx.globalAlpha=(1-q)*0.9;ctx.fillStyle=p.col;
    ctx.beginPath();ctx.arc(p.x,p.y,Math.max(0.5,p.size*(1-q*0.5)),0,Math.PI*2);ctx.fill();
  });
  ctx.restore();ctx.globalAlpha=1;
}
function drawPopups(){
  popups.forEach(p=>{
    const q=p.t/p.life,y=p.y-q*38;
    const a=q<0.12?q/0.12:1-Math.max(0,(q-0.55)/0.45);
    if(a<=0)return;
    const s=p.size*(1+Math.max(0,0.5-q*2)*0.6);
    ctx.font='900 '+s.toFixed(1)+'px '+F_DISPLAY;ctx.textAlign='center';
    ctx.globalAlpha=a;
    ctx.lineWidth=4;ctx.strokeStyle='rgba(20,8,6,0.9)';ctx.strokeText(p.text,p.x,y);
    ctx.fillStyle=p.col;ctx.fillText(p.text,p.x,y);
    ctx.globalAlpha=1;
  });
}
function hudLabel(t,x,y,size,col,align){
  ctx.font='700 '+size+'px '+F_BODY;ctx.fillStyle=col;ctx.textAlign=align||'left';
  try{ctx.letterSpacing='2px';}catch(e){}
  ctx.fillText(t,x,y);
  try{ctx.letterSpacing='0px';}catch(e){}
}
function drawHUD(){
  const f=clamp(W/980,0.72,1);
  hudLabel('SCORE',18,hudH*0.36,(10*f).toFixed(0),'#c99f84');
  ctx.save();ctx.font='900 '+(26*f).toFixed(0)+'px '+F_DISPLAY;ctx.fillStyle='#ffd23f';
  ctx.shadowColor='rgba(255,150,40,.5)';ctx.shadowBlur=10;ctx.textAlign='left';
  ctx.fillText(''+score,16,hudH*0.8);ctx.restore();
  const bw=Math.min(300,W*0.34),bx=(W-bw)/2,by=hudH/2-5,bh=10;
  roundRectPath(ctx,bx,by,bw,bh,5);ctx.fillStyle='rgba(255,255,255,0.07)';ctx.fill();
  ctx.lineWidth=1;ctx.strokeStyle='rgba(255,140,70,.3)';ctx.stroke();
  const pct=clamp(score/target,0,1);
  if(pct>0){
    const fg=ctx.createLinearGradient(bx,0,bx+bw,0);fg.addColorStop(0,'#ff5a22');fg.addColorStop(1,'#ffd23f');
    roundRectPath(ctx,bx,by,Math.max(6,bw*pct),bh,5);ctx.fillStyle=fg;ctx.fill();
  }
  hudLabel('TARGET '+target,bx+bw+10,by+bh-1,(11*f).toFixed(0),'#c99f84');
  const low=moves<=5;
  hudLabel('MOVES',W-18,hudH*0.36,(10*f).toFixed(0),'#c99f84','right');
  ctx.save();ctx.font='900 '+(26*f).toFixed(0)+'px '+F_DISPLAY;ctx.textAlign='right';
  if(low){ctx.globalAlpha=0.55+0.45*Math.sin(timeAcc*6);ctx.fillStyle='#ff5a3c';}else ctx.fillStyle='#f6ead9';
  ctx.shadowColor='rgba(255,120,40,.3)';ctx.shadowBlur=8;
  ctx.fillText(''+moves,W-18,hudH*0.8);ctx.restore();
  hudLabel('LEVEL',W-18-(64*f),hudH*0.36,(10*f).toFixed(0),'#c99f84','right');
  ctx.save();ctx.font='900 '+(17*f).toFixed(0)+'px '+F_DISPLAY;ctx.fillStyle='#ffb35c';ctx.textAlign='right';
  ctx.fillText(''+level,W-18-(64*f),hudH*0.8);ctx.restore();
}

function render(){
  ctx.clearRect(0,0,W,H);
  drawBG();
  const sx=(Math.random()-0.5)*shake,sy=(Math.random()-0.5)*shake;
  ctx.save();ctx.translate(sx,sy);
  if(grid.length){drawBoardFrame();drawGems();drawRings();drawParticles();drawPopups();}
  ctx.restore();
  if(state!=='menu'){drawHUD();drawFooter();}
  if(paused&&state==='play')drawPause();
}

/* ============ input ============ */
let drag=null;
function ptr(e){const r=cvs.getBoundingClientRect();return{x:e.clientX-r.left,y:e.clientY-r.top};}
function cellAt(x,y){
  if(x<boardX||y<boardY||x>=boardX+boardW||y>=boardY+boardH)return null;
  return{r:clamp(((y-boardY)/cell)|0,0,ROWS-1),c:clamp(((x-boardX)/cell)|0,0,COLS-1)};
}
cvs.addEventListener('pointerdown',e=>{
  ac();
  if(state!=='play'||paused||phase!=='idle')return;
  const p=ptr(e);kbActive=false;hoverCell=cellAt(p.x,p.y);
  if(!hoverCell)return;
  const cl=hoverCell;
  if(selected&&Math.abs(selected.r-cl.r)+Math.abs(selected.c-cl.c)===1)trySwap(selected.r,selected.c,cl.r,cl.c);
  else{selected=cl;S.select();}
  drag={x:p.x,y:p.y,r:cl.r,c:cl.c};
});
window.addEventListener('pointermove',e=>{
  const p=ptr(e);hoverCell=cellAt(p.x,p.y);
  if(drag&&state==='play'&&!paused&&phase==='idle'){
    const dx=p.x-drag.x,dy=p.y-drag.y;
    if(Math.max(Math.abs(dx),Math.abs(dy))>cell*0.35){
      const d=Math.abs(dx)>Math.abs(dy)?{r:0,c:Math.sign(dx)}:{r:Math.sign(dy),c:0};
      const t={r:drag.r+d.r,c:drag.c+d.c};
      if(t.r>=0&&t.r<ROWS&&t.c>=0&&t.c<COLS)trySwap(drag.r,drag.c,t.r,t.c);
      selected=null;drag=null;
    }
  }
});
window.addEventListener('pointerup',()=>{drag=null;});
cvs.addEventListener('pointerleave',()=>{hoverCell=null;});
cvs.addEventListener('contextmenu',e=>e.preventDefault());

function kbMove(dr,dc){
  if(state!=='play'||paused)return;
  kbActive=true;
  cursor.r=clamp(cursor.r+dr,0,ROWS-1);cursor.c=clamp(cursor.c+dc,0,COLS-1);
  if(selected&&phase==='idle'&&Math.abs(selected.r-cursor.r)+Math.abs(selected.c-cursor.c)===1){
    trySwap(selected.r,selected.c,cursor.r,cursor.c);
  }else S.select();
}
function kbSelect(){
  if(state!=='play'||paused||phase!=='idle')return;
  kbActive=true;
  if(selected&&selected.r===cursor.r&&selected.c===cursor.c)selected=null;
  else{selected={r:cursor.r,c:cursor.c};S.select();}
}
function togglePause(){paused=!paused;if(!paused)S.select();}
function toggleMute(){muted=!muted;try{localStorage.setItem('mm_muted',muted?'1':'0');}catch(e){}if(!muted)S.select();}

window.addEventListener('keydown',e=>{
  ac();const k=e.key;
  if(k==='ArrowUp'||k==='w'||k==='W'){e.preventDefault();kbMove(-1,0);}
  else if(k==='ArrowDown'||k==='s'||k==='S'){e.preventDefault();kbMove(1,0);}
  else if(k==='ArrowLeft'||k==='a'||k==='A'){e.preventDefault();kbMove(0,-1);}
  else if(k==='ArrowRight'||k==='d'||k==='D'){e.preventDefault();kbMove(0,1);}
  else if(k===' '||k==='Enter'){
    e.preventDefault();
    if(state==='menu')newGame();
    else if(state==='levelup'){level++;startLevel();}
    else if(state==='over')newGame();
    else kbSelect();
  }
  else if(k==='r'||k==='R'){newGame();}
  else if(k==='p'||k==='P'){if(state==='play')togglePause();}
  else if(k==='m'||k==='M'){toggleMute();}
});
document.addEventListener('visibilitychange',()=>{if(document.hidden&&state==='play'&&!paused)paused=true;});
