function getRow(r){return rows[r]||null;}
function cellX(c){return trackX+(c+0.5)*cell;}
function cellY(r){return carY-(r-dist)*cell;}
function carPX(){return trackX+(carX+0.5)*cell;}
function hash(n){const s=Math.sin(n*127.1+311.7)*43758.5453;return s-Math.floor(s);}

/* ============ board generation (no pre-made matches) ============ */
function noRun(r,c,col,row){
  if(c>=2&&row[c-1]===col&&row[c-2]===col)return true;
  const a=getRow(r-1),b=getRow(r-2);
  return !!(a&&a[c]===col&&b&&b[c]===col);
}
function genRow(r){
  const row=new Array(CFG.cols).fill(null);
  if(r<5)return row;                       // clean runway at the start
  const p3=Math.min(0.8,0.25+r*0.006);      // wall density ramps with distance
  let k=1+(Math.random()<0.6?1:0)+(Math.random()<p3?1:0);if(k>3)k=3;
  const cs=[0,1,2,3,4].sort(()=>Math.random()-0.5).slice(0,k).sort((a,b)=>a-b);
  for(const c of cs){let col,g2=0;do{col=CFG.colors[(Math.random()*CFG.cols)|0];g2++;}while(g2<10&&noRun(r,c,col,row));row[c]=col;}
  return row;
}
function ensureRows(upTo){while(nextRow<=upTo){rows[nextRow]=genRow(nextRow);nextRow++;}}

/* ============ matching ============ */
function addC(seen,set,r,c){const id=r*32+c;if(!seen.has(id)){seen.add(id);set.push({r,c});}}
function findMatches(){
  const lo=Math.floor(dist)-1,hi=Math.ceil(dist+rowsAbove+1),set=[],seen=new Set();
  for(let r=lo;r<=hi;r++){const row=getRow(r);if(!row)continue;let run=1;
    for(let c=1;c<=CFG.cols;c++){
      if(c<CFG.cols&&row[c]!=null&&row[c]===row[c-1])run++;
      else{if(run>=3)for(let j=c-run;j<c;j++)addC(seen,set,r,j);run=1;}
    }}
  for(let c=0;c<CFG.cols;c++){let run=1;
    for(let r2=lo+1;r2<=hi+1;r2++){
      const cur=getRow(r2),prev=getRow(r2-1);
      if(cur&&cur[c]!=null&&prev&&prev[c]===cur[c])run++;
      else{if(run>=3)for(let j=r2-run;j<r2;j++)addC(seen,set,j,c);run=1;}
    }}
  return set;
}
function resolve(m){
  const n=m.length;let sx=0,sy=0;
  for(const p of m){
    const row=getRow(p.r);if(!row||row[p.c]==null)continue;
    const col=row[p.c];row[p.c]=null;pops.push({r:p.r,c:p.c,col,t:0});
    const x=cellX(p.c),y=cellY(p.r);sx+=x;sy+=y;
    for(let i=0;i<7;i++){const a=Math.random()*6.283,s=40+Math.random()*160;
      particles.push({x,y,vx:Math.cos(a)*s,vy:Math.sin(a)*s-60,g:500,life:0,max:.45+Math.random()*.3,size:2+Math.random()*2.5,col,type:'spark'});}
  }
  score+=n*15;turbo=Math.min(1,turbo+n*0.075);
  const tx=sx/n,ty=sy/n;
  addText(tx,ty,'+'+(n*15),n>=4?'#ffd23f':'#cfe0ff',n>=4?26:18,n>=4?.9:.7);
  if(n>=4)addText(tx,ty-26,n>=6?'GREAT!':'NICE!','#ffffff',20,.9);
  sfx('pop',n);shake=Math.max(shake,2+n*0.35);
  if(selected){const sr=getRow(selected.r);if(!sr||sr[selected.c]==null)selected=null;}
  hideHint();
}
function trySwap(a,b){
  const ra=getRow(a.r),rb=getRow(b.r);if(!ra||!rb)return;
  const va=ra[a.c],vb=rb[b.c];if(va==null||vb==null)return;
  ra[a.c]=vb;rb[b.c]=va;
  const m=findMatches();
  fx={a,b,oa:va,ob:vb,t:0,revert:m.length===0,didR:false};
  if(m.length)resolve(m);else{sfx('bad');shake=Math.max(shake,3);}
}

/* ============ fx helpers ============ */
function addText(x,y,txt,col,size,dur){if(texts.length>14)texts.shift();texts.push({x,y,txt,col,size,t:0,dur:dur||.8});}
function updBursts(dt){
  for(let i=particles.length-1;i>=0;i--){const p=particles[i];p.life+=dt;
    if(p.life>p.max){particles.splice(i,1);continue;}
    p.vy+=(p.g||0)*dt;p.x+=p.vx*dt;p.y+=p.vy*dt;
    if(p.type==='smoke'){p.size+=14*dt;p.vx*=.98;}}
  for(let i=texts.length-1;i>=0;i--){const t=texts[i];t.t+=dt;if(t.t>t.dur)texts.splice(i,1);}
}
function updFx(dt){
  if(!fx)return;fx.t+=dt/0.3;
  if(fx.revert&&!fx.didR&&fx.t>=0.5){const ra=getRow(fx.a.r),rb=getRow(fx.b.r);
    if(ra&&rb){const t=ra[fx.a.c];ra[fx.a.c]=rb[fx.b.c];rb[fx.b.c]=t;}fx.didR=true;}
  if(fx.t>=1)fx=null;
}
function updPops(dt){for(let i=pops.length-1;i>=0;i--){pops[i].t+=dt/0.28;if(pops[i].t>=1)pops.splice(i,1);}}

function reset(){dist=0;meters=0;score=0;turbo=0.55;carX=2;steerVis=0;curSp=2.6;
  rows={};nextRow=5;doneRows={};selected=null;hoverCell=null;fx=null;pops=[];particles=[];texts=[];
  shake=0;cpFlash=0;warnT=0;lastCp=0;animT=0;}
function setScr(s){[menuScr,overScr,pauseScr].forEach(el=>el.classList.toggle('show',el===s));}
function start(){reset();state='play';setScr(null);stage.dataset.state='play';
  ensureRows(rowsAbove+3);hintT=6;hintEl.classList.add('show');sfx('go');}
function toMenu(){state='menu';setScr(menuScr);stage.dataset.state='menu';
  $('menuBest').hidden=best<=0;$('menuBestVal').textContent=best+' m';sparkT=0;}
function togglePause(){if(state==='play'){state='pause';setScr(pauseScr);stage.dataset.state='pause';}
  else if(state==='pause'){state='play';setScr(null);stage.dataset.state='play';}}
function hideHint(){hintT=0;hintEl.classList.remove('show');}
function crash(){
  if(state!=='play')return;
  state='over_anim';animT=0;shake=16;hideHint();
  const x=carPX(),y=carY,cols=['#ffd23f','#ff4d5e','#ffffff','#ffb830'];
  for(let i=0;i<30;i++){const a=Math.random()*6.283,s=80+Math.random()*320;
    particles.push({x,y,vx:Math.cos(a)*s,vy:Math.sin(a)*s,g:560,life:0,max:.5+Math.random()*.5,size:2+Math.random()*3,col:cols[i&3],type:'spark'});}
  for(let i=0;i<14;i++){const a=Math.random()*6.283,s=60+Math.random()*200;
    particles.push({x,y,vx:Math.cos(a)*s,vy:Math.sin(a)*s-120,g:520,life:0,max:.7+Math.random()*.4,size:3+Math.random()*3,col:i%2?'#e0243c':'#20294a',type:'chip'});}
  for(let i=0;i<8;i++)particles.push({x:x+(Math.random()*30-15),y:y+(Math.random()*20-10),vx:Math.random()*40-20,vy:-30-Math.random()*40,g:0,life:0,max:1.6,size:9,col:'#5a627a',type:'smoke'});
  sfx('crash');
  const m=Math.floor(meters);overNew=m>best;
  if(overNew){best=m;try{localStorage.setItem('gridracer_best_m',''+m);}catch(e){}}
}

  $('overDist').textContent=Math.floor(meters)+' m';
  $('overScore').textContent=Math.floor(score);
  $('overBest').textContent=best+' m';
  $('newBadge').hidden=!overNew;

}

/* ============ update ============ */
function update(dt){
  gt+=dt;
  if(state==='play'){
    const base=2.2+Math.min(2.8,meters*0.0035);
    let sp=base*(0.55+turbo*1.0);if(brake)sp*=0.45;
    curSp=sp;dist+=sp*dt;meters=dist*10;score+=sp*dt*10;
    turbo=Math.max(0,turbo-(0.065+Math.min(0.09,meters*0.00012))*dt);
    const steer=(keys.r||touchR?1:0)-(keys.l||touchL?1:0);
    carX=Math.max(0,Math.min(CFG.cols-1,carX+steer*6.4*dt));
    steerVis+=(steer-steerVis)*Math.min(1,dt*12);
    ensureRows(Math.ceil(dist)+rowsAbove+2);
    Object.keys(rows).forEach(k=>{if(+k<dist-3)delete rows[+k];});
    Object.keys(doneRows).forEach(k=>{if(+k<dist-4)delete doneRows[+k];});
    if(selected){const sr=getRow(selected.r);if(!sr||sr[selected.c]==null)selected=null;}
    updFx(dt);updPops(dt);updBursts(dt);
    // dodge bonus for blocks you steered past
    for(let r=Math.floor(dist)-2;r<dist-0.5;r++){
      if(doneRows[r])continue;const row=getRow(r);if(!row)continue;
      if(r-dist+1<-0.45){let d=0;for(let c=0;c<CFG.cols;c++)if(row[c]!=null)d++;
        doneRows[r]=1;if(d>0){score+=d*5;addText(trackX-16,carY+8,'+'+(d*5)+' DODGE','#9fe8c9',13,.7);}}
    }
    const cp=Math.floor(meters/250);
    if(cp>lastCp){lastCp=cp;score+=100;turbo=Math.min(1,turbo+0.35);cpFlash=1;sfx('check');
      addText(W/2,carY-rowsAbove*cell*0.55,'CHECKPOINT +100','#ffd23f',Math.min(34,W/16),1.4);}
    cpFlash=Math.max(0,cpFlash-dt*1.6);
    if(turbo<0.18){warnT+=dt;if(warnT>0.5){warnT=0;sfx('warn');}}else warnT=0;
    if(hintT>0){hintT-=dt;if(hintT<=0)hideHint();}
    // collision
    for(let r=Math.floor(dist)-1;r<=Math.floor(dist)+1;r++){
      const row=getRow(r);if(!row)continue;const rel=r-dist;
      if(rel>=0.45||rel+1<=-0.45)continue;
      for(let c=0;c<CFG.cols;c++)if(row[c]!=null&&Math.abs(carX-(c+0.5))<0.72){crash();break;}
    }
  }
  else if(state==='menu'){
    curSp=2.6;dist+=curSp*dt;meters=dist*10;
    ensureRows(Math.ceil(dist)+rowsAbove+2);
    Object.keys(rows).forEach(k=>{if(+k<dist-3)delete rows[+k];});
    sparkT-=dt;if(sparkT<=0){sparkT=0.45;
      const r=Math.floor(dist)+1+((Math.random()*rowsAbove)|0),c=(Math.random()*CFG.cols)|0,row=getRow(r);
      if(row&&row[c]!=null)particles.push({x:cellX(c),y:cellY(r),vx:0,vy:-20,g:0,life:0,max:.5,size:3,col:'#fff',type:'glint'});}
    updBursts(dt);
  }
  else if(state==='over_anim'){
    animT+=dt;updBursts(dt);
    if(Math.random()<0.5)particles.push({x:carPX()+(Math.random()*22-11),y:carY-10,vx:Math.random()*30-15,vy:-40-Math.random()*30,g:0,life:0,max:1.4,size:8+Math.random()*6,col:'#5a627a',type:'smoke'});
    if(animT>1.05)showOver();
  }
  else if(state==='over'){updBursts(dt);}
  shake=Math.max(0,shake-dt*(state==='over_anim'?12:3));
}

/* ============ drawing ============ */
function rr(x,y,w,h,r){ctx.beginPath();ctx.moveTo(x+r,y);ctx.arcTo(x+w,y,x+w,y+h,r);ctx.arcTo(x+w,y+h,x,y+h,r);ctx.arcTo(x,y+h,x,y,r);ctx.arcTo(x,y,x+w,y,r);ctx.closePath();}
function drawShoulders(){
  const tw=CFG.cols*cell;
  ctx.fillStyle='#0b101e';
  ctx.fillRect(-8,-8,Math.max(0,trackX-16)+8,H+16);
  ctx.fillRect(trackX+tw+16,-8,W-(trackX+tw+16)+8,H+16);
  const band=76,off=(dist*cell)%band,kMax=Math.ceil(H/band)+2,bId=Math.floor(dist*cell/band);
  for(let k=-1;k<kMax;k++){const y=k*band-off,id=bId+k;
    for(let j=0;j<3;j++){
      const h1=hash(id*17.3+j*3.1),h2=hash(id*5.7+j*9.2),h3=hash(id*11.9+j*7.7),s=2+h1*3;
      ctx.fillStyle='rgba(160,180,220,'+(0.03+h2*0.05).toFixed(3)+')';
      if(trackX-40>20)ctx.fillRect(h2*(trackX-40),y+h3*band,s,s);
      const rw=W-trackX-tw-52;
      if(rw>20)ctx.fillRect(trackX+tw+26+h2*rw,y+(1-h3)*band,s,s);
    }}
}
function drawTrack(){
  const tw=CFG.cols*cell;
  ctx.fillStyle='#101627';ctx.fillRect(trackX-18,-8,18,H+16);ctx.fillRect(trackX+tw,-8,18,H+16);
  ctx.fillStyle=trackGrad;ctx.fillRect(trackX,-8,tw,H+16);
  ctx.fillStyle='rgba(255,255,255,0.07)';
  for(let r=Math.floor(dist)-1;r<=dist+rowsAbove+1;r++){const y=carY-(r-dist)*cell;
    if(y>H+cell||y<-cell)continue;
    for(let b=1;b<CFG.cols;b++)ctx.fillRect(trackX+b*cell-2,y,4,cell*0.5);}
  for(let r=Math.floor(dist)-1;r<=dist+rowsAbove+1;r++){const y=carY-(r-dist)*cell;
    if(y>H+cell||y<-cell)continue;
    ctx.fillStyle=(r&1)?'#d9304a':'#e8ecf5';
    ctx.fillRect(trackX-16,y,12,cell);ctx.fillRect(trackX+tw+4,y,12,cell);}
  ctx.fillStyle='rgba(0,0,0,0.35)';ctx.fillRect(trackX-4,-8,4,H+16);ctx.fillRect(trackX+tw,-8,4,H+16);
  ctx.font='700 10px "Trebuchet MS",sans-serif';ctx.textAlign='right';ctx.fillStyle='rgba(200,215,255,0.4)';
  const m0=Math.max(0,Math.ceil((dist*10)/100)*100);
  for(let m=m0;m<(dist+rowsAbove)*10+60;m+=100){const y=carY-(m/10-dist)*cell;if(y<-10||y>H+10)continue;ctx.fillText(m+'m',trackX-24,y+3);}
}
function drawTile(x,y,col,r,c,moving){
  const s=cell-6,x0=x-s/2,y0=y-s/2,rd=s*0.18;
  ctx.save();
  if(!moving&&state==='play'&&r!=null){const rel=r-dist;
    if(rel<2.2&&Math.abs(carX-(c+0.5))<0.95){const n=Math.max(0,1-rel/2.2);
      ctx.shadowColor='#ff4d5e';ctx.shadowBlur=14*n*(0.6+0.4*Math.sin(gt*10));}}
  ctx.fillStyle=col;rr(x0,y0,s,s,rd);ctx.fill();
  ctx.shadowBlur=0;
  ctx.fillStyle='rgba(255,255,255,.28)';rr(x0+3,y0+3,s-6,s*0.34,rd*0.7);ctx.fill();
  ctx.fillStyle='rgba(0,0,0,.18)';rr(x0+3,y0+s*0.62,s-6,s*0.34-3,rd*0.7);ctx.fill();
  ctx.strokeStyle='rgba(0,0,0,0.4)';ctx.lineWidth=2;rr(x0,y0,s,s,rd);ctx.stroke();
  ctx.restore();
}
function drawTiles(){
  const fxCells={};if(fx){fxCells[fx.a.r*32+fx.a.c]=1;fxCells[fx.b.r*32+fx.b.c]=1;}
  for(let r=Math.floor(dist)-1;r<=dist+rowsAbove+1;r++){
    const row=getRow(r);if(!row)continue;const y=carY-(r-dist)*cell;
    if(y>H+cell||y<-cell)continue;
    for(let c=0;c<CFG.cols;c++){const col=row[c];if(col==null||fxCells[r*32+c])continue;drawTile(cellX(c),y,col,r,c,false);}
  }
  if(fx){
    const raw=fx.revert?(fx.t<0.5?fx.t*2:(1-fx.t)*2):Math.min(1,fx.t/0.6);
    const k=raw*raw*(3-2*raw);
    const ax=cellX(fx.a.c),ay=carY-(fx.a.r-dist)*cell,bx=cellX(fx.b.c),by=carY-(fx.b.r-dist)*cell;
    let ca,cb;if(fx.revert&&fx.t>=0.5){ca=fx.oa;cb=fx.ob;}else{ca=fx.ob;cb=fx.oa;}
    drawTile(ax+(bx-ax)*(1-k),ay+(by-ay)*(1-k),ca,null,null,true);
    drawTile(bx+(ax-bx)*(1-k),by+(ay-by)*(1-k),cb,null,null,true);
  }
  const ring=(p,a,col)=>{const x=cellX(p.c),y=carY-(p.r-dist)*cell;ctx.save();ctx.translate(x,y);
    ctx.lineWidth=3;ctx.strokeStyle=col;ctx.globalAlpha=a;ctx.shadowColor='#fff';ctx.shadowBlur=8;
    rr(-cell/2+2,-cell/2+2,cell-4,cell-4,cell*0.16);ctx.stroke();ctx.restore();};
  if(hoverCell&&!fx)ring(hoverCell,0.3,'#ffffff');
  if(selected&&!fx)ring(selected,0.55+0.4*Math.sin(gt*9),'#ffffff');
}
function drawPops(){
  for(const p of pops){const x=cellX(p.c),y=carY-(p.r-dist)*cell,s=(cell-6)*(1+p.t*0.9);
    ctx.globalAlpha=1-p.t;ctx.fillStyle=p.col;rr(x-s/2,y-s/2,s,s,8);ctx.fill();
    ctx.globalAlpha=(1-p.t)*0.8;ctx.fillStyle='#fff';rr(x-s/2,y-s/2,s,s*0.4,8);ctx.fill();}
  ctx.globalAlpha=1;
}
function drawCar(){
  const wreck=state!=='play',x=carPX(),y=carY,w=cell*0.62,h=cell*1.02;
  ctx.save();ctx.translate(x,y);ctx.rotate(steerVis*0.14);
  ctx.fillStyle='rgba(255,244,200,0.07)';
  ctx.beginPath();ctx.moveTo(-w*0.35,-h/2);ctx.lineTo(w*0.35,-h/2);ctx.lineTo(w*0.6,-h/2-cell*1.4);ctx.lineTo(-w*0.6,-h/2-cell*1.4);ctx.closePath();ctx.fill();
  ctx.fillStyle='rgba(0,0,0,0.4)';ctx.beginPath();ctx.ellipse(3,7,w*0.62,h*0.55,0,0,6.283);ctx.fill();
  const ww=cell*0.15,wh=cell*0.3;ctx.fillStyle='#0a0d16';
  rr(-w/2-ww*0.8,-h*0.34,ww,wh,3);ctx.fill();rr(w/2-ww*0.2,-h*0.34,ww,wh,3);ctx.fill();
  rr(-w/2-ww*0.8,h*0.12,ww,wh,3);ctx.fill();rr(w/2-ww*0.2,h*0.12,ww,wh,3);ctx.fill();
  const flame=(0.25+turbo*1.1)*cell*0.45*(0.7+Math.random()*0.6)*(wreck?0:1);
  if(flame>3){ctx.save();ctx.globalCompositeOperation='lighter';
    for(let i=0;i<2;i++){const fxp=i?w*0.18:-w*0.18,fh=flame*(0.8+Math.random()*0.4);
      ctx.fillStyle=i?'#ffd23f':'rgba(255,77,94,0.9)';
      ctx.beginPath();ctx.moveTo(fxp-5,h/2);ctx.lineTo(fxp+5,h/2);ctx.lineTo(fxp,h/2+fh);ctx.closePath();ctx.fill();}
    ctx.restore();}
  const g=ctx.createLinearGradient(0,-h/2,0,h/2);g.addColorStop(0,'#ff9a4d');g.addColorStop(0.5,'#f43b52');g.addColorStop(1,'#c81e3c');

function initAudio(){
  if(!AC){try{

    master=AC.createGain();master.gain.value=muted?0:0.55;master.connect(AC.destination);
    engOsc=AC.createOscillator();engOsc.type='sawtooth';
    const f=AC.createBiquadFilter();f.type='lowpass';f.frequency.value=420;f.Q.value=1.5;
    engGain=AC.createGain();engGain.gain.value=0;

  }catch(e){AC=null;}}

function tone(f,d,type,vol,slide,delay){if(!AC||muted)return;delay=delay||0;
  const t0=AC.currentTime+delay,o=AC.createOscillator(),g=AC.createGain();
  o.type=type;o.frequency.setValueAtTime(f,t0);
  if(slide)o.frequency.exponentialRampToValueAtTime(Math.max(30,slide),t0+d);
  g.gain.setValueAtTime(vol,t0);g.gain.exponentialRampToValueAtTime(.0001,t0+d);
  o.connect(g);g.connect(master);o.start(t0);o.stop(t0+d+.05);}
function noiseBurst(d,freq,vol,q,delay){if(!AC||muted)return;delay=delay||0;
  const t0=AC.currentTime+delay,n=Math.floor(AC.sampleRate*d),buf=AC.createBuffer(1,n,AC.sampleRate),ch=buf.getChannelData(0);
  for(let i=0;i<n;i++)ch[i]=(Math.random()*2-1)*(1-i/n);
  const src=AC.createBufferSource();src.buffer=buf;
  const f=AC.createBiquadFilter();f.type='bandpass';f.frequency.value=freq;f.Q.value=q||1;
  const g=AC.createGain();g.gain.value=vol;
  src.connect(f);f.connect(g);g.connect(master);src.start(t0);}
function sfx(name,n){

**How to play:** Your car auto-drives up an endless grid of colored blocks. Steer with **← → / A D**, brake with **Space**, and click/tap (or drag) two adjacent tiles to swap them — matching 3+ paves that road open *and* charges your turbo, which drives your speed. Dodge what you can't clear, grab checkpoints every 250 m, and touch a block at your own peril. **P** pauses, **M** mutes, **R** restarts.
