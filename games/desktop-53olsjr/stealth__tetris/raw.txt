/* ============================== COLLISION / MOVEMENT ============================== */
function collide(x,y,type,rot){
  type=type||G.cur.type; rot=rot==null?G.cur.rot:rot;
  for(const [dx,dy] of SHAPES[type][rot]){
    const cx=x+dx, cy=y+dy;
    if(cx<0||cx>=COLS||cy>=ROWS) return true;
    if(cy>=0 && G.grid[cy][cx]) return true;
  }
  return false;
}
function ghostY(){
  let gy=G.cur.y; while(!collide(G.cur.x,gy+1)) gy++; return gy;
}
function placeType(t){
  G.cur={type:t,rot:0,x:3,y:-3}; G.dropAcc=0; G.lockT=.5; G.resets=0; G.spawnT=.18;
  if(collide(3,-3,t,0)) gameOver('overflow');
}
function tryMove(dx){
  if(!G.cur||G.busted) return false;
  const nx=G.cur.x+dx;
  if(!collide(nx,G.cur.y)){ G.cur.x=nx;
    if(G.resets<15){ G.lockT=.5; G.resets++; }
    SFX.move(); return true; }
  return false;
}
function tryRotate(dir){
  if(!G.cur||G.busted) return;
  const from=G.cur.rot, to=(from+dir+4)%4;
  const table=G.cur.type==='I'?KICK.I:KICK.JLSTZ;
  for(const [kx,ky] of table[from+''+to]){
    if(!collide(G.cur.x+kx,G.cur.y+ky)){
      G.cur.rot=to; G.cur.x+=kx; G.cur.y+=ky;
      if(G.resets<15){ G.lockT=.5; G.resets++; }
      SFX.rotate(); return;
    }
  }
}
function hardDrop(){
  if(!G.cur||G.busted) return;
  let d=0; while(!collide(G.cur.x,G.cur.y+1)){ G.cur.y++; d++; }
  if(d>0){ G.score+=2*d; G.shake=Math.max(G.shake,d>6?5:3); dustAtPiece(); SFX.hard(); }
  lockPiece();
}
function holdSwap(){
  if(!G.cur||!G.canHold||G.busted) return;
  const t=G.cur.type;
  if(G.hold==null){ G.hold=t; placeType(G.queue.shift()); G.queue.push(drawBag()); }
  else { const h=G.hold; G.hold=t; placeType(h); }
  G.canHold=false; SFX.hold();
}
function spawnNext(){
  G.cur=null; G.canHold=true;
  placeType(G.queue.shift()); G.queue.push(drawBag());
}

/* ============================== LOCK / CLEARS / SCORE ============================== */
function lockPiece(){
  const cur=G.cur; G.cur=null;
  const idx=TYPES.indexOf(cur.type)+1; let topOut=false;
  for(const [dx,dy] of SHAPES[cur.type][cur.rot]){
    const cx=cur.x+dx, cy=cur.y+dy;
    if(cy<0){ topOut=true; continue; }
    G.grid[cy][cx]=idx;
    spawnSparks(BOARD_X+(cx+.5)*C, BOARD_Y+(cy+.5)*C, PCOL[idx-1], 3, .5);
  }
  SFX.lock();
  if(topOut){ gameOver('overflow'); return; }
  const full=[]; for(let r=0;r<ROWS;r++) if(G.grid[r].every(v=>v)) full.push(r);
  if(full.length) doClears(full); else spawnNext();
}
function doClears(rows){

  const n=rows.length, base=[0,100,300,500,800][n-1]*G.level;
  const bonus=Math.min(200,Math.round(G.streak*5));
  G.score+=base+bonus; G.lines+=n; G.jam=Math.min(1,G.jam+n*.25);
  SFX.clear(n); G.shake=Math.max(G.shake,n>=3?6:3);
  for(const r of rows){

function jammer(){
  if(!G.cur||G.busted||G.jammed>0||G.jam<.25) return;
  G.jammed=.7+G.jam*1.5; G.jam=0; SFX.jam();

function detection(dt){
  const cur=G.cur; if(!cur) return;
  let any=false;
  for(const cam of G.cams){
    const c=Math.max(0,Math.min(9,Math.round(cam.colF))); let hit=false;
    for(const [dx,dy] of SHAPES[cur.type][cur.rot]){
      if(cur.y+dy>=0 && cur.x+dx===c) hit=true;
    }
    cam.hit=hit; any=any||hit;
  }
  const grace=G.time<2.5;
  if(any&&!grace){
    G.det=Math.min(1,G.det+dt/.85); G.streak=0;
    G.warnT-=dt; if(G.warnT<=0){ SFX.warn(); G.warnT=.16; }
    G.shake=Math.max(G.shake,1.2);
  } else {
    for(const cam of G.cams) cam.hit=false;
    G.det=Math.max(0,G.det-dt/1.7); G.streak+=dt;
  }
  if(G.det>=1) triggerBust();
}
function triggerBust(){
  G.busted=true; G.bustT=0; G.shake=12; SFX.bust();
  for(let i=0;i<46;i++) G.parts.push({x:BOARD_X+BW/2,y:BOARD_Y+BH*.35,
    vx:(Math.random()-.5)*520,vy:(Math.random()-.5)*520,g:160,life:.9,max:.9,size:2+Math.random()*4,c:'#ff3b5c'});
}

  showOver();
}

  }
}

function update(dt){
  tGlobal+=dt;
  // background rain
  for(const p of rain){ p.y+=p.sp*dt; if(p.y>VH+p.len){ p.y=-p.len-10; p.x=Math.random()*VW; } }
  G.shake=Math.max(0,G.shake-dt*26);

    if(G.busted){
      G.bustT+=dt;
      if(G.bustT>1.15){ G.reason='busted'; state='over'; if(G.score>best){best=G.score;saveBest();} showOver(); }
    } else {
      G.time+=dt;
      if(G.jammed>0) G.jammed-=dt; else G.camTime+=dt;
      for(const cam of G.cams) cam.colF=4.5+cam.amp*Math.sin(G.camTime*cam.w+cam.ph);

      const live=(!G.clearRows.length && G.cur);
      if(live){
        // auto-run / soft-drop DAS
        for(const key of ['L','R']){
          const d=das[key]; if(!d.on) continue; d.t+=dt;
          if(d.t>.17){ d.r+=dt; while(d.r>.055){ d.r-=.055; tryMove(key==='L'?-1:1); } }
        }
        const soft=das.D.on; das.D.t+=dt;
        let iv=soft? .045 : Math.max(.06,.72*Math.pow(.85,G.level-1));
        G.dropAcc+=dt;
        while(G.dropAcc>=iv){
          G.dropAcc-=iv;
          if(!collide(G.cur.x,G.cur.y+1)){ G.cur.y++; if(soft) G.score+=1; } else break;
        }
        const grounded=collide(G.cur.x,G.cur.y+1);
        if(grounded){ G.lockT-=dt*(soft?5:1); if(G.lockT<=0) lockPiece(); }
        else G.lockT=Math.min(.5,G.lockT+dt);
        detection(dt);
      } else if(G.clearRows.length){
        G.det=Math.max(0,G.det-dt/1.7);
        G.clearT-=dt; if(G.clearT<=0) collapseRows();
      } else {
        G.det=Math.max(0,G.det-dt/1.7);
        if(!G.cur) spawnNext();
      }
    }
  } else if(state==='menu'){
    G.camTime+=dt;
    for(const cam of G.cams) cam.colF=4.5+cam.amp*Math.sin(G.camTime*cam.w+cam.ph);
  }

  // fx (always run so menu/over also feel alive)
  for(let i=G.parts.length-1;i>=0;i--){ const p=G.parts[i];
    p.life-=dt; if(p.life<=0){ G.parts.splice(i,1); continue; }
    p.vy+=p.g*dt; p.x+=p.vx*dt; p.y+=p.vy*dt; }
  for(let i=G.floats.length-1;i>=0;i--){ const f=G.floats[i];
    f.t+=dt; f.y-=22*dt; if(f.t>f.life) G.floats.splice(i,1); }
}
