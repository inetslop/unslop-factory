function genMaze(seed){

  const st=[[0,0]]; vis[0][0]=true; g[1][1]=0;
  while(st.length){
    const [cx,cy]=st[st.length-1];
    const dirs=shuffle([[1,0],[-1,0],[0,1],[0,-1]],rnd).filter(([dx,dy])=>{const nx=cx+dx,ny=cy+dy;return nx>=0&&nx<CW2&&ny>=0&&ny<CH2&&!vis[ny][nx];});
    if(!dirs.length){st.pop();continue;}
    const [dx,dy]=dirs[0];
    const nx=cx+dx,ny=cy+dy;

  }
  // loops
  let removed=0, guard=0;

  while(c!==sy*W+sx&&guard++<400){out.push([c%W,(c/W)|0]);c=prev[c];if(c<0)break;}
  return out.reverse();}

function los(x0,y0,x1,y1){const d=Math.hypot(x1-x0,y1-y0),n=Math.max(2,Math.ceil(d*4));
  for(let i=1;i<n;i++){const t=i/n;if(grid[(y0+(y1-y0)*t)|0][(x0+(x1-x0)*t)|0]===1)return false;}return true;}

function buildLevel(){const L=LEVELS[level];const seed=(runSeed+level*1013904223)>>>0;grid=genMaze(seed);
  floors=[];for(let y=0;y<H;y++)for(let x=0;x<W;x++)if(grid[y][x]===0)floors.push([x,y]);

  wallPath=new Path2D();dotsPath=new Path2D();edgePath=new Path2D();
  const visWall=(x,y)=>{for(const[dx,dy]of DIRS){const nx=x+dx,ny=y+dy;
    if(nx>=0&&nx<W&&ny>=0&&ny<H&&grid[ny][nx]===0)return true;}return false;};
  for(let y=0;y<H;y++)for(let x=0;x<W;x++){
    if(grid[y][x]===0){dotsPath.arc((x+.5)*T,(y+.5)*T,1.7,0,TAU);}
    else if(visWall(x,y)){wallPath.rect(x*T+1.5,y*T+1.5,T-3,T-3);
      if(grid[y][x+1]===0){edgePath.moveTo((x+1)*T-2,y*T+3);edgePath.lineTo((x+1)*T-2,(y+1)*T-3);}
      if(grid[y][x-1]===0){edgePath.moveTo(x*T+2,y*T+3);edgePath.lineTo(x*T+2,(y+1)*T-3);}
      if(grid[y+1]&&grid[y+1][x]===0){edgePath.moveTo(x*T+3,(y+1)*T-2);edgePath.lineTo((x+1)*T-3,(y+1)*T-2);}
      if(grid[y-1]&&grid[y-1][x]===0){edgePath.moveTo(x*T+3,y*T+2);edgePath.lineTo((x+1)*T-3,y*T+2);}}}

  const b=bfsDist(1,1);let maxD=0;for(const[x,y]of floors)maxD=Math.max(maxD,b[y*W+x]);
  let best=-1;exitT=null;for(const[x,y]of floors){if(b[y*W+x]>best){best=b[y*W+x];exitT=[x,y];}}

  const pool=floors.filter(([x,y])=>b[y*W+x]>maxD*.3).sort((p,q)=>b[q[1]*W+q[0]]-b[p[1]*W+p[0]]);
  let coreTiles=[];
  for(const[x,y]of pool){if(coreTiles.length>=3)break;
    if(x===exitT[0]&&y===exitT[1])continue;
    if(coreTiles.every(c=>Math.abs(c[0]-x)+Math.abs(c[1]-y)>=5))coreTiles.push([x,y]);}
  let guard=0;
  while(coreTiles.length<3&&guard++<60){const[x,y]=pool[(coreTiles.length*7+guard)%pool.length];
    if(!coreTiles.some(c=>c[0]===x&&c[1]===y)&&!(x===exitT[0]&&y===exitT[1]))coreTiles.push([x,y]);}

  const cand=floors.filter(([x,y])=>b[y*W+x]>8).sort((p,q)=>b[q[1]*W+q[0]]-b[p[1]*W+p[0]]);
  let spawns=[];
  for(const[x,y]of cand){if(spawns.length>=L.ghosts)break;
    if(coreTiles.some(c=>c[0]===x&&c[1]===y)||(x===exitT[0]&&y===exitT[1]))continue;
    if(spawns.every(s=>Math.abs(s[0]-x)+Math.abs(s[1]-y)>=6))spawns.push([x,y]);}
  guard=0;
  while(spawns.length<L.ghosts&&guard++<60){const[x,y]=cand[(spawns.length*5+2+guard)%cand.length]||[W-2,1];
    if(!spawns.some(s=>s[0]===x&&s[1]===y))spawns.push([x,y]);}

  P={px:1,py:1,fx:1,fy:1,dirx:0,diry:0,faceAng:0,decoys:3,lockedT:0,trail:[],moving:false};
  if(grid[1][2]===0)P.faceAng=0;else if(grid[2][1]===0)P.faceAng=Math.PI/2;

  ghosts=spawns.map((s,i)=>({id:i+1,px:s[0],py:s[1],faceAng:Math.random()*TAU,state:'patrol',
    suspicion:0,flashT:0,path:[],idx:0,waitT:.4+Math.random(),alertT:0,seenT:9,
    lastSeen:{x:1,y:1},decoy:null,wasAlert:false,inspected:false,phase:Math.random()*TAU,color:GCOL[i%4]}));

  cores=coreTiles.map(([x,y],i)=>({x,y,i,taken:false}));
  decoys=[];ripples=[];gateOpen=false;gotCores=0;levelTime=0;caughtBy=null;

  dust=[];for(let i=0;i<46;i++)dust.push({x:Math.random()*W*T,y:Math.random()*H*T,
    vx:(Math.random()-.5)*6,vy:-2-Math.random()*4,r:.8+Math.random()*1.4,a:.05+Math.random()*.1});

  hudSector.textContent=`SECTOR 0${level+1} · ${L.name}`;
  updatePips();hudObj.textContent=`RECOVER CORES · 0/${cores.length}`;}

function updatePlayer(dt){const w=wantDir();
  const atC=Math.round(P.px)===P.fx&&Math.round(P.py)===P.fy;
  if(atC){if(w&&grid[P.fy+w.y]&&grid[P.fy+w.y][P.fx+w.x]===0){P.dirx=w.x;P.diry=w.y;}else{P.dirx=0;P.diry=0;}}
  else if(w&&w.x===-P.dirx&&w.y===-P.diry){P.dirx=-P.dirx;P.diry=-P.diry;}
  const sp=(sneakHeld?SNEAK:RUN)*dt;let stepped=false;
  if(P.dirx!==0){const t=P.fx+P.dirx;
    if((P.dirx>0&&P.px>=t)||(P.dirx<0&&P.px<=t)){P.px=t;P.fx=t;stepped=true;}else P.px+=P.dirx*sp;}
  if(P.diry!==0){const t=P.fy+P.diry;
    if((P.diry>0&&P.py>=t)||(P.diry<0&&P.py<=t)){P.py=t;P.fy=t;stepped=true;}else P.py+=P.diry*sp;}
  P.moving=!!(P.dirx||P.diry);
  if(P.moving)P.faceAng=Math.atan2(P.diry,P.dirx);
  if(P.moving&&!sneakHeld)P.trail.push({x:P.px,y:P.py,a:1});
  for(const t of P.trail)t.a-=dt*2.6;
  P.trail=P.trail.filter(t=>t.a>0);if(P.trail.length>14)P.trail.shift();
  if(stepped)onStep();
  P.lockedT=Math.max(0,P.lockedT-dt);}

function onStep(){if(sneakHeld)SFX.stepSneak();else{SFX.stepRun();noise(P.px,P.py);}
  const tx=Math.round(P.px),ty=Math.round(P.py);
  for(const c of cores){if(!c.taken&&c.x===tx&&c.y===ty){c.taken=true;gotCores++;SFX.core();
    flash={c:'255,217,59',a:.26};updatePips();
    if(gotCores===cores.length){gateOpen=true;SFX.gate();banner('GATE OPEN','HEAD FOR THE EXIT');}}}
  if(exitT&&tx===exitT[0]&&ty===exitT[1]){
    if(gateOpen)levelClear();else if(P.lockedT<=0){P.lockedT=1;SFX.buzz();}}}

function noise(x,y){ripples.push({x,y,age:0,life:.5,max:1.7,c:'150,214,255'});
  let heard=false;
  for(const g of ghosts){if(g.state==='alert'||g.decoy)continue;
    if(Math.hypot(g.px-x,g.py-y)<NOISE_R){makeSusp(g,x,y);heard=true;}}
  if(heard)SFX.hear();}

function makeSusp(g,x,y){g.state='susp';g.suspicion=Math.max(g.suspicion,.4);
  g.lastSeen={x,y};g.seenT=0;g.waitT=2.4;g.inspected=false;
  g.path=findPath(Math.round(g.px),Math.round(g.py),Math.round(x),Math.round(y));g.idx=0;}

function alertGhost(g){if(g.state==='alert')return;
  g.state='alert';g.suspicion=1;g.flashT=.9;g.decoy=null;g.inspected=false;g.alertT=0;
  SFX.sting();
  g.path=findPath(Math.round(g.px),Math.round(g.py),Math.round(P.px),Math.round(P.py));g.idx=0;}

function throwDecoy(){if(!P)return;if(P.decoys<=0){SFX.buzz();return;}
  let tx=null,ty=null;
  const dx=Math.round(Math.cos(P.faceAng)),dy=Math.round(Math.sin(P.faceAng));
  for(let i=1;i<=2&&tx==null;i++){const nx=P.fx+dx*i,ny=P.fy+dy*i;
    if(grid[ny]&&grid[ny][nx]===0){tx=nx;ty=ny;}}
  if(tx==null)for(const[ddx,ddy]of DIRS){const nx=P.fx+ddx,ny=P.fy+ddy;
    if(grid[ny]&&grid[ny][nx]===0){tx=nx;ty=ny;break;}}
  if(tx==null)return;
  P.decoys--;const d={x:tx,y:ty,age:0,alive:true,ping:0};decoys.push(d);SFX.throwD();updatePips();
  for(const g of ghosts){if(g.decoy)continue;
    if(Math.hypot(g.px-d.x,g.py-d.y)<6.5){g.wasAlert=g.state==='alert';g.decoy=d;g.state='invest';
      g.suspicion=0;g.inspected=false;
      g.path=findPath(Math.round(g.px),Math.round(g.py),d.x,d.y);g.idx=0;}}}

function updateGhosts(dt,detect){const L=LEVELS[level];let anyAlert=false,anyHeard=false;
  for(const g of ghosts){
    if(detect){
      const dx=P.px-g.px,dy=P.py-g.py,d=Math.hypot(dx,dy);
      const range=(g.state==='alert'?L.sight*1.25:(sneakHeld?L.sight*.82:L.sight));
      let seen=false;
      if(d<range){const a=Math.atan2(dy,dx);
        if(Math.abs(angDiff(a,g.faceAng))<(g.state==='alert'?ALERT_FOV:FOV)/2&&los(g.px,g.py,P.px,P.py))seen=true;}
      if(d<1.05)seen=true;
      if(seen){g.lastSeen={x:P.px,y:P.py};g.seenT=0;
        if(!sneakHeld)alertGhost(g);
        else{g.suspicion=Math.min(1,g.suspicion+dt/1.4);if(g.suspicion>=1)alertGhost(g);}}
      else{g.seenT+=dt;g.suspicion=Math.max(0,g.suspicion-dt/2.2);}}
    g.flashT=Math.max(0,(g.flashT||0)-dt);

    if(g.state==='invest'&&g.decoy&&!g.decoy.alive){
      const was=g.wasAlert&&!g.inspected;g.decoy=null;
      if(was)alertGhost(g);else{g.state='patrol';g.waitT=.6;g.suspicion=0;}}

    if(g.state==='alert'){
      g.alertT+=dt;
      if(g.alertT>0.45){g.alertT=0;
        g.path=findPath(Math.round(g.px),Math.round(g.py),Math.round(P.px),Math.round(P.py));g.idx=0;}
      if(detect&&g.seenT>4){g.state='susp';g.suspicion=.6;g.waitT=2.2;
        g.path=findPath(Math.round(g.px),Math.round(g.py),Math.round(g.lastSeen.x),Math.round(g.lastSeen.y));g.idx=0;}}

    if(!g.path.length){
      g.faceAng+=dt*(g.state==='invest'?2.2:.7);
      g.waitT-=dt;
      if(g.waitT<=0){
        if(g.state==='patrol'){const[tx,ty]=floors[(Math.random()*floors.length)|0];
          if(Math.abs(tx-g.px)+Math.abs(ty-g.py)>3){g.path=findPath(Math.round(g.px),Math.round(g.py),tx,ty);g.idx=0;}else g.waitT=.5;}
        else if(g.state==='susp'){g.state='patrol';g.suspicion=0;g.waitT=.6;}
        else if(g.state==='invest'){g.state='patrol';g.suspicion=0;g.waitT=.5;}}}

    const mult=g.state==='alert'?2.05:g.state==='susp'?1.35:g.state==='invest'?1.5:1;
    const sp=L.patrol*mult*dt;
    if(g.path.length){const n=g.path[g.idx];let dxn=n[0]-g.px,dyn=n[1]-g.py;const dd=Math.hypot(dxn,dyn);
      if(dd<=sp){g.px=n[0];g.py=n[1];g.idx++;
        if(g.idx>=g.path.length){g.path.length=0;g.inspected=(g.state==='invest');
          g.waitT=g.state==='patrol'?.5+Math.random()*1.4:g.state==='susp'?2.2:1.3;}}
      else{g.px+=dxn/dd*sp;g.py+=dyn/dd*sp;g.faceAng=Math.atan2(dyn,dxn);}}
    if(g.state==='alert')anyAlert=true;
    if((g.state!=='patrol'||g.suspicion>.15)&&g.state!=='invest')anyHeard=true;}
  return{anyAlert,anyHeard};}
