/* ============ update ============ */
function carrierUnder(bx,by){
  for(const L of layout.river){
    if(by<L.y||by>L.y+L.h)continue;
    const cy=L.y+L.h/2;
    if(L.island&&bx>=L.island.x-R*0.4&&bx<=L.island.x+L.island.w+R*0.4&&Math.abs(by-cy)<18+R*.4)return {carry:0};
    for(let i=0;i<L.N;i++){
      const x=mod(L.off+i*L.spacing,L.span)-L.len;
      if(x>VW||x+L.len<0)continue;
      const w=L.type==='duck'?34:L.len,h=L.type==='duck'?18:28;
      if(hitRect(bx,by,6,x,cy-h/2,w,h))return {carry:L.d*L.s};}
  }return null;}

function update(dt){
  time+=dt;shake=Math.max(0,shake-dt*30);
  // hazards
  for(const L of layout.road)L.off=mod(L.off+L.d*L.s*dt,L.span);
  for(const L of layout.river)L.off=mod(L.off+L.d*L.s*dt,L.span);
  for(const c of clouds){c.x+=c.v*dt;if(c.x<-c.r)c.x=VW+c.r;if(c.x>VW+c.r)c.x=-c.r;}
  rippleTimer-=dt;
  if(rippleTimer<=0){rippleTimer=.45+Math.random()*.4;
    const ry=layout.riverTop+10+Math.random()*(layout.walkY-layout.riverTop-20);
    ripples.push({x:20+Math.random()*(VW-40),y:ry,r:2,max:26,c:'255,255,255'});}
  for(let i=ripples.length-1;i>=0;i--){const r=ripples[i];r.r+=34*dt;if(r.r>=r.max)ripples.splice(i,1);}
  for(let i=parts.length-1;i>=0;i--){const p=parts[i];p.life-=dt;
    if(p.life<=0){parts.splice(i,1);continue;}p.vy+=(p.g||0)*dt;p.x+=p.vx*dt;p.y+=p.vy*dt;p.rot+=(p.vr||0)*dt;}

  if(phase!=='play')return;
  // aim smoothing from keys
  let has=false;const dx=(keys.has('R')?1:0)-(keys.has('L')?1:0),dy=(keys.has('D')?1:0)-(keys.has('U')?1:0);
  if(dx||dy){has=true;const t=Math.atan2(dy,dx);let d=mod(t-aim+Math.PI,2*Math.PI)-Math.PI;
    aim+=clamp(d,-16*dt,16*dt);}
  if(charging)power=Math.min(1,power+dt/.85);

  if(mode==='intro'){introT+=dt;if(introT>1.5)mode='aim';return;}
  if(mode==='dead'){deadT+=dt;
    if(deadT>1){ball.x=VW/2;ball.y=layout.teeY+TEE_H*.52;ball.vx=0;ball.vy=0;drownT=0;mode='aim';}return;}
  if(mode==='sunk'){sunkT+=dt;
    if(sunkT>.4&&!confettiDone){confettiDone=true;shake=5;sfx[HOLES[holeIndex].par-strokes<=-1?'good':'click']();
      const cols=['#ffd23f','#ff5d5d','#4fc3ff','#7dff8a','#fff'];
      for(let i=0;i<46;i++)parts.push({x:layout.holeX,y:layout.holeY,vx:(Math.random()-.5)*480,
        vy:-80-Math.random()*260,g:430,life:1.1+Math.random()*.7,max:1.8,size:3+Math.random()*4,
        color:cols[i%5],shape:'rect',rot:Math.random()*6,vr:(Math.random()-.5)*14});}
    if(sunkT>2.35){if(holeIndex<HOLES.length-1){holeIndex++;setupHole(holeIndex);}else showEnd();}return;}

  // ---- ball physics (aim = resting, roll = moving) ----
  const speed=Math.hypot(ball.vx,ball.vy);
  if(mode==='roll'){
    let k;
    if(ball.y>=layout.roadTop&&ball.y<layout.walkY)k=FR.asphalt;
    else if(ball.y>=layout.walkY&&ball.y<layout.riverTop)k=FR.walk;
    else if(ball.y>=layout.riverTop&&ball.y<layout.greenY)k=FR.water;
    else if(ball.y>=layout.greenY)k=FR.green;else k=FR.grass;
    const f=Math.exp(-k*dt);ball.vx*=f;ball.vy*=f;
    const sp2=Math.hypot(ball.vx,ball.vy);
    if(sp2>0){const dec=Math.min(sp2,26*dt)/sp2;ball.vx-=ball.vx*dec;ball.vy-=ball.vy*dec;}

    // water: carry or drown
    if(ball.y>=layout.riverTop&&ball.y<layout.greenY){
      const c=carrierUnder(ball.x,ball.y);
      if(c){drownT=0;ball.x+=c.carry*dt;
        if(time-lastWake>.1){lastWake=time;ripples.push({x:ball.x-ball.vx*.02,y:ball.y,r:3,max:26,c:'255,255,255'});}}
      else{drownT+=dt;if(drownT> .45){die('water');return;}}}
    else drownT=0;

    // green: attraction + sink
    if(ball.y>=layout.greenY){
      const ddx=layout.holeX-ball.x,ddy=layout.holeY-ball.y,d=Math.hypot(ddx,ddy);
      if(d<26&&d>1&&speed<380){ball.vx+=ddx/d*520*dt;ball.vy+=ddy/d*520*dt;}
      if(d<11.5&&(Math.hypot(ball.vx,ball.vy)<360)){sink();return;}}

    // walls
    const inWater=ball.y>=layout.riverTop&&ball.y<layout.greenY;
    if(ball.x-R<BMIN){ball.x=BMIN+R;if(inWater)ball.vx=0;else{if(Math.abs(ball.vx)>50)sfx.bounce();ball.vx=-ball.vx*.55;}}
    if(ball.x+R>BMAX){ball.x=BMAX-R;if(inWater)ball.vx=0;else{if(Math.abs(ball.vx)>50)sfx.bounce();ball.vx=-ball.vx*.55;}}
    if(ball.y-R<layout.greenY){ball.y=layout.greenY+R;if(Math.abs(ball.vy)>50)sfx.bounce();ball.vy=-ball.vy*.5;}
    if(ball.y+R>layout.teeY+TEE_H){ball.y=layout.teeY+TEE_H-R;if(Math.abs(ball.vy)>50)sfx.bounce();ball.vy=-ball.vy*.5;}

    // cars hit
    for(const L of layout.road)for(let i=0;i<L.N;i++){
      const x=mod(L.off+i*L.spacing,L.span)-L.len;
      if(x>VW||x+L.len<0)continue;
      if(hitRect(ball.x,ball.y,7,x,L.y+8,L.len,L.h-16)){die('car');return;}}

    ball.rot+=Math.max(0,Math.hypot(ball.vx,ball.vy))/R*dt*(ball.vx>=0?1:-1);
    if(Math.hypot(ball.vx,ball.vy)<12){ball.vx=0;ball.vy=0;mode='aim';power=0;}
  }else{ // aim: resting — can still be run over or fall in lip
    for(const L of layout.road)for(let i=0;i<L.N;i++){
      const x=mod(L.off+i*L.spacing,L.span)-L.len;
      if(x>VW||x+L.len<0)continue;
      if(hitRect(ball.x,ball.y,7,x,L.y+8,L.len,L.h-16)){die('car');return;}}
    if(ball.y>=layout.greenY){const d=Math.hypot(layout.holeX-ball.x,layout.holeY-ball.y);
      if(d<11){sink();return;}}
  }
}

function rr(x,y,w,h,r){ctx.beginPath();ctx.moveTo(x+r,y);ctx.arcTo(x+w,y,x+w,y+h,r);

      for(let i=0;i<3;i++){const a=ball.rot+i*2.09;
        ctx.beginPath();ctx.arc(bx+Math.cos(a)*rr2*.5,by+Math.sin(a)*rr2*.5,1.4*sc,0,7);ctx.fill();}}
  }

  // ---- aim UI ----
  if(phase==='play'&&mode==='aim'){
    let dir=aim,len=26,pow=power;
    if(dragging){const dx=ball.x-dragP.x,dy=ball.y-dragP.y,d=Math.hypot(dx,dy);
      if(d>4){dir=Math.atan2(dy,dx);len=30+70*clamp(d/150,0,1);pow=clamp(d/150,0,1);
        ctx.strokeStyle='rgba(255,255,255,.35)';ctx.lineWidth=2;ctx.setLineDash([4,6]);
        ctx.beginPath();ctx.moveTo(dragP.x,dragP.y);ctx.lineTo(ball.x,ball.y);ctx.stroke();ctx.setLineDash([]);}}
    const col=pow<.01?'rgba(255,255,255,.5)':pow<.4?'#7dff8a':pow<.75?'#ffd23f':'#ff6b4a';
    ctx.strokeStyle=col;ctx.lineWidth=3;ctx.setLineDash([7,7]);
    const ex=ball.x+Math.cos(dir)*(R+6+len),ey=ball.y+Math.sin(dir)*(R+6+len);
    ctx.beginPath();ctx.moveTo(ball.x+Math.cos(dir)*(R+6),ball.y+Math.sin(dir)*(R+6));ctx.lineTo(ex,ey);ctx.stroke();
    ctx.setLineDash([]);
    ctx.fillStyle=col;ctx.save();ctx.translate(ex,ey);ctx.rotate(dir);
    ctx.beginPath();ctx.moveTo(9,0);ctx.lineTo(-4,-7);ctx.lineTo(-4,7);ctx.closePath();ctx.fill();ctx.restore();
    if(charging||dragging){
      ctx.strokeStyle='rgba(255,255,255,.18)';ctx.lineWidth=4;
      ctx.beginPath();ctx.arc(ball.x,ball.y,17,0,7);ctx.stroke();
      ctx.strokeStyle=col;ctx.beginPath();ctx.arc(ball.x,ball.y,17,-Math.PI/2,-Math.PI/2+pow*2*Math.PI);ctx.stroke();
      ttxt(Math.round(pow*100)+'%',ball.x+24,ball.y-20,13,'m','#fff','left','middle',4);}
  }

  // particles
  for(const p of parts){const a=Math.max(0,p.life/p.max);
    ctx.globalAlpha=a;ctx.fillStyle=p.color;
    if(p.shape==='rect'){ctx.save();ctx.translate(p.x,p.y);ctx.rotate(p.rot);
      ctx.fillRect(-p.size/2,-p.size/2,p.size,p.size*.7);ctx.restore();}
    else if(p.shape==='spark'){ctx.save();ctx.translate(p.x,p.y);ctx.rotate(p.rot+Math.PI/4);
      ctx.fillRect(-p.size,-p.size*.35,p.size*2,p.size*.7);ctx.restore();}
    else{ctx.beginPath();ctx.arc(p.x,p.y,p.size,0,7);ctx.fill();}}
  ctx.globalAlpha=1;

  // dead splat star
  if(mode==='dead'){const p=Math.min(1,deadT/.2),s=(30+14*(1-p))*Math.min(1,p*3);
    const a=deadT<.55?1:Math.max(0,1-(deadT-.55)/.4);
    ctx.save();ctx.translate(ball.x,ball.y);ctx.globalAlpha=a;
    ctx.fillStyle='#fff';ctx.beginPath();
    for(let i=0;i<10;i++){const ang=i/10*Math.PI*2,rad=i%2?s*.45:s;
      ctx[i?'lineTo':'moveTo'](Math.cos(ang)*rad,Math.sin(ang)*rad);}
    ctx.closePath();ctx.fill();
    ctx.fillStyle='#ffd23f';ctx.beginPath();ctx.arc(0,0,s*.3,0,7);ctx.fill();ctx.restore();}

  // ---- HUD ----
  if(phase!=='menu'){
    ctx.fillStyle='#0e2418';ctx.fillRect(0,0,VW,HUD_H);
    ctx.fillStyle='#f4c542';ctx.fillRect(0,HUD_H-3,VW,3);
    ttxt('HOLE',16,16,10,'m','#9fdcb2');
    ttxt((holeIndex+1)+'/'+HOLES.length,14,38,24,'d','#fff');
    ttxt('STROKES',VW/2-52,16,10,'m','#9fdcb2','center');
    ttxt(''+strokes,VW/2-52,39,27,'d','#fff','center');
    ttxt('PAR',VW/2+52,16,10,'m','#9fdcb2','center');
    ttxt(''+HOLES[holeIndex].par,VW/2+52,39,27,'d','#ffd23f','center');
    const done=scores.length,diff=done?scores.reduce((a,b,i)=>a+(b-HOLES[i].par),0):null;
    ttxt('TOTAL',VW-14,16,10,'m','#9fdcb2','right');
    ttxt(fmt(diff),VW-14,39,24,'d',diff===null?'#8aa795':(diff>0?'#ff8a7a':diff<0?'#8dff9e':'#fff'),'right');

    ctx.fillStyle='#0e2418';ctx.fillRect(0,VH-BOT_H,VW,BOT_H);
    ctx.fillStyle='#f4c542';ctx.fillRect(0,VH-BOT_H,VW,2);
    let hint='DRAG TO SLINGSHOT · OR  AIM + HOLD SPACE';
    if(mode==='intro')hint='GET READY…';
    else if(charging)hint='POWER — RELEASE!';
    else if(mode==='roll')hint=strokes? 'ROLLING…':'…';
    else if(mode==='dead')hint='SPLAT! PENALTY +1';
    ttxt(hint,VW/2,VH-BOT_H/2+1,11,'m','#bfe8cd','center');
    if(lastMsg)ttxt(lastMsg,14,VH-BOT_H/2+1,11,'m',lastMsgCol);
    ttxt('R REDO · M SND '+(muted?'OFF':'ON'),VW-14,VH-BOT_H/2+1,10,'m','rgba(191,232,205,.55)','right');

    // banners
    if(mode==='intro')bannerDraw('HOLE '+(holeIndex+1),'PAR '+HOLES[holeIndex].par,introT,1.5,'#fff','#ffd23f');
    if(mode==='sunk'&&sunkT>.55){const rel=scores[scores.length-1]-HOLES[holeIndex].par;
      bannerDraw(termFor(rel)+'!',scores[scores.length-1]+' stroke'+(scores[scores.length-1]===1?'':'s'),Math.min(1,(sunkT-.55)/.28),9,rel<=0?'#ffd23f':'#ff8a7a','#fff');}
  }
  ctx.restore();
}
