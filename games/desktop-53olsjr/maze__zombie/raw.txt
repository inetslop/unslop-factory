function genMaze(){
  const t=54;let w=Math.max(11,Math.min(23,Math.round(W/t))),h=Math.max(9,Math.min(17,Math.round(H/t)));
  if(w%2===0)w++;if(h%2===0)h++;gw=w;gh=h;
  walls=new Uint8Array(gw*gh).fill(1);passCells=[];
  const st=[[1,1]];walls[1+gw]=0;
  while(st.length){const [x,y]=st[st.length-1];const o=[];
    for(const [dx,dy] of [[0,-2],[2,0],[0,2],[-2,0]]){const nx=x+dx,ny=y+dy;
      if(nx>0&&nx<gw-1&&ny>0&&ny<gh-1&&!walls[nx+ny*gw])o.push([nx,ny,dx,dy]);}
    if(o.length){const [nx,ny,dx,dy]=o[(Math.random()*o.length)|0];
      walls[x+(y+dy/2)*gw+dx/2]=0;walls[nx+ny*gw]=0;st.push([nx,ny]);}
    else st.pop();}
  for(let y=1;y<gh-1;y++)for(let x=1;x<gw-1;x++){const i=x+y*gw;if(!walls[i])continue;
    const hO=!walls[i-1]&&!walls[i+1],vO=!walls[i-gw]&&!walls[i+gw];
    if(hO&&vO){if(Math.random()<.02)walls[i]=0;}else if(hO||vO){if(Math.random()<.06)walls[i]=0;}}
  exX=gw-2;exY=gh-2;passCells.length=0;
  for(let y=0;y<gh;y++)for(let x=0;x<gw;x++)if(!walls[x+y*gw])passCells.push({x,y});
}
const wallAt=(x,y)=>(x<0||y<0||x>=gw||y>=gh)||!!walls[x+y*gw];
function bfs(sx,sy,tx,ty){if(sx===tx&&sy===ty)return[];
  const prev=new Int32Array(gw*gh).fill(-1),q=[sx+sy*gw];let qi=0;prev[q[0]]=q[0];
  while(qi<q.length){const cur=q[qi++],cx=cur%gw,cy=(cur/gw)|0;
    if(cx===tx&&cy===ty)break;
    const nb=[[1,0],[-1,0],[0,1],[0,-1]];
    for(const [dx,dy] of nb){const nx=cx+dx,ny=cy+dy;if(nx<0||ny<0||nx>=gw||ny>=gh)continue;
      const ni=nx+ny*gw;if(walls[ni]||prev[ni]!==-1)continue;prev[ni]=cur;q.push(ni);}}
  const t=tx+ty*gw;if(prev[t]===-1)return[];const path=[];let c=t;const s0=sx+sy*gw;
  while(c!==s0){path.push({x:c%gw,y:(c/gw)|0});c=prev[c];}
  path.reverse();return path;}
function bfsField(sx,sy){const dist=new Int16Array(gw*gh).fill(-1);const q=[sx+sy*gw];let qi=0;dist[q[0]]=0;
  while(qi<q.length){const cur=q[qi++],cx=cur%gw,cy=(cur/gw)|0,d=dist[cur]+1;
    const nb=[[1,0],[-1,0],[0,1],[0,-1]];
    for(const [dx,dy] of nb){const nx=cx+dx,ny=cy+dy;if(nx<0||ny<0||nx>=gw||ny>=gh)continue;
      const ni=nx+ny*gw;if(walls[ni]||dist[ni]!==-1)continue;dist[ni]=d;q.push(ni);}}
  return dist;}

function placeItems(){items=[];const cells=passCells.filter(c=>!(c.x===exX&&c.y===exY)&&!(c.x===1&&c.y===1));
  for(let i=cells.length-1;i>0;i--){const j=(Math.random()*(i+1))|0;[cells[i],cells[j]]=[cells[j],cells[i]];}
  const take=(mS,mSep,seps,count)=>{const out=[];
    outer:for(const c of cells){if(Math.abs(c.x-1)+Math.abs(c.y-1)<mS)continue;let ok=true;
      for(const s of seps.concat(out)){if(Math.abs(c.x-s.x)+Math.abs(c.y-s.y)<mSep){ok=false;break;}}
      if(ok){out.push(c);if(out.length===count)break outer;}}return out;};
  const v=take(5,4,[],6),sv=take(3,2,v,3);
  totalVials=v.length||1;
  v.forEach((c,i)=>items.push({x:c.x,y:c.y,kind:'vial',ph:i}));
  sv.forEach((c,i)=>items.push({x:c.x,y:c.y,kind:'shiv',ph:i}));}
