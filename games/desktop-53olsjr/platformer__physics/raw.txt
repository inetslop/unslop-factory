  dashT:0,dashCd:0,dashDir:1,landSquash:0,runPh:0,alpha:1,spawnFade:0,dead:0,ghosts:[],groundRef:null};
const K={l:false,r:false,jump:false};

function setState(s){state=s;document.body.dataset.state=s;}

/* ============================== FX ============================== */
const parts=[];
function addP(o){if(parts.length<450)parts.push(o);}
function dustBurst(x,y,n,up){for(let i=0;i<n;i++)addP({tp:'dust',x:x+(Math.random()-.5)*22,y,vx:(Math.random()-.5)*130,vy:up?-(20+Math.random()*60):-(10+Math.random()*40),g:-60,life:.35+Math.random()*.35,max:.7,size:3+Math.random()*3.5,col:'232,201,160'});}
function ringFx(x,y,col,R){addP({tp:'ring',x,y,r:8,r1:R,life:.32,max:.32,col});}
function sparks(x,y,n,col,spd){for(let i=0;i<n;i++){const a=Math.random()*Math.PI*2,v=spd*(.4+Math.random()*.8);
  addP({tp:'spark',x,y,vx:Math.cos(a)*v,vy:Math.sin(a)*v-40,g:520,life:.3+Math.random()*.3,max:.6,size:1.5+Math.random()*2,col});}}
function shards(x,y,n,col){for(let i=0;i<n;i++)addP({tp:'shard',x,y,vx:(Math.random()-.5)*420,vy:-120-Math.random()*380,g:1500,life:.6+Math.random()*.5,max:1.1,size:4+Math.random()*5,col,rot:Math.random()*7,vr:(Math.random()-.5)*14});}
function confetti(x,y,n){const cols=['#ffd27a','#7ef0d4','#ff5d73','#ffe9c9'];for(let i=0;i<n;i++)addP({tp:'conf',x,y,vx:(Math.random()-.5)*380,vy:-160-Math.random()*320,g:900,life:.9+Math.random()*.7,max:1.6,size:4+Math.random()*4,col:cols[i%4],rot:Math.random()*7,vr:(Math.random()-.5)*18});}
function txtFx(x,y,str,col){addP({tp:'txt',x,y,vx:0,vy:-42,g:0,life:.9,max:.9,size:16,col,txt:str});}
function updateParticles(dt){for(let i=parts.length-1;i>=0;i--){const p=parts[i];p.life-=dt;
  if(p.life<=0){parts.splice(i,1);continue;}
  p.vy+=(p.g||0)*dt;p.x+=p.vx*dt;p.y+=p.vy*dt;if(p.rot!==undefined)p.rot+=p.vr*dt;}}
function updateGhosts(dt){for(let i=P.ghosts.length-1;i>=0;i--){P.ghosts[i].life-=dt;if(P.ghosts[i].life<=0)P.ghosts.splice(i,1);}}

/* ============================== WORLD UPDATE ============================== */
function updateWorld(dt){
  for(const s of solids){
    if(s.t==='moveH'){const p=(s.axis==='x'?s.x:s.y)+s.dir*s.sp*dt;let np;
      if(p>s.to){np=s.to;s.dir=-1;}else if(p<s.from){np=s.from;s.dir=1;}else np=p;
      if(s.axis==='x'){s.dx=np-s.x;s.x=np;}else{s.dy=np-s.y;s.y=np;}}
    else if(s.t==='crumble'){
      if(s.state===1){s.timer-=dt;if(s.timer<=0){s.state=2;s.vy=30;shards(s.x+s.w/2,s.y+8,6,'#a8603a');noiseS(.15,.14,900,'highpass');}}
      else if(s.state===2){s.vy+=1800*dt;s.y+=s.vy*dt;s.alpha-=dt*1.5;if(s.alpha<=0)s.dead=true;}}
  }
  for(const sp of springs)sp.anim=Math.max(0,sp.anim-dt*3.2);
  for(const e of embers)e.t+=dt;
  for(const c of checks)if(c.on&&Math.random()<dt*1.6)sparks(c.x+18,c.y-62,1,'255,210,122',60);
  if(Math.random()<dt*3){const gx=goal.x,gy=goal.y;addP({tp:'spark',x:gx+(Math.random()-.5)*60,y:gy-20-Math.random()*90,vx:(Math.random()-.5)*24,vy:-30,g:0,life:.7,max:.7,size:1.6+Math.random()*1.6,col:'180,240,255'});}
}

/* ============================== PLAYER PHYSICS ============================== */
function tryDash(){if(state!=='play'||P.dead>0||P.dashCd>0||P.dashT>0)return;
  P.dashT=.13;P.dashCd=.9;P.dashDir=(K.l&&!K.r)?-1:(K.r&&!K.l)?1:P.face;
  SFX.dash();shake=Math.max(shake,3);ringFx(P.x+16,P.y+15,'rgba(126,240,212,.7)',34);}
function doJump(){P.vy=-760;P.grounded=false;P.coyote=0;P.buf=0;SFX.jump();dustBurst(P.x+16,P.y+30,5,false);ringFx(P.x+16,P.y+28,'rgba(255,233,201,.4)',22);}
function doDouble(){P.vy=-700;P.buf=0;P.usedDouble=true;SFX.dbl();sparks(P.x+16,P.y+24,8,'126,240,212',140);ringFx(P.x+16,P.y+15,'rgba(126,240,212,.7)',30);}
function die(){if(state!=='play'||P.dead>0)return;deaths++;P.dead=.75;SFX.death();shake=13;
  const cx=P.x+16,cy=P.y+15;shards(cx,cy,18,'#8df5dd');shards(cx,cy,8,'#3fbfa0');ringFx(cx,cy,'rgba(255,93,115,.8)',70);}
function respawn(){P.x=respPt.x;P.y=respPt.y;P.vx=P.vy=0;P.usedDouble=false;P.dashT=0;P.dashCd=0;P.spawnFade=.35;
  for(const s of solids)if(s.t==='crumble'){s.state=0;s.timer=.45;s.y=s.oy;s.vy=0;s.alpha=1;s.dead=false;}
  ringFx(P.x+16,P.y+15,'rgba(255,210,122,.7)',36);}

let gAcc=0,runDust=0;
function physStep(h){
  const active=(state==='play');
  P.coyote-=h;P.buf-=h;P.dashCd-=h;if(P.landSquash>0)P.landSquash-=h;
  if(P.spawnFade>0)P.spawnFade-=h;
  P.alpha=clamp((.35-P.spawnFade)/.2,0,1);
  const ax=active?((K.r?1:0)-(K.l?1:0)):0;
  if(ax!==0)P.face=ax;

  if(P.dashT>0){
    P.dashT-=h;P.vx=P.dashDir*640;P.vy=0;
    gAcc+=h;if(gAcc>.02){gAcc=0;if(P.ghosts.length<12)P.ghosts.push({x:P.x,y:P.y,life:.25});}
    addP({tp:'streak',x:P.x+16-P.dashDir*14,y:P.y+8+Math.random()*14,vx:-P.dashDir*300,vy:0,g:0,life:.22,max:.22,size:1.5,col:'126,240,212'});
  }else{
    const acc=P.grounded?2400:1500,max=340;
    if(ax!==0){P.vx+=ax*acc*h;
      if(Math.abs(P.vx)>max)P.vx-=Math.sign(P.vx)*Math.min(Math.abs(P.vx)-max,(P.grounded?900:140)*h);}
    else{const dec=P.grounded?1900:260;P.vx-=Math.sign(P.vx)*Math.min(Math.abs(P.vx),dec*h);}
    const gmul=(P.vy<0&&!K.jump)?1.5:1;
    P.vy=Math.min(980,P.vy+2200*gmul*h);
    if(active&&P.buf>0){
      if(P.grounded||P.coyote>0)doJump();
      else if(!P.usedDouble)doDouble();
    }
  }
  // carry by moving platform
  const gr=P.groundRef;
  if(gr&&gr.t==='moveH'){P.x+=gr.dx;P.y+=gr.dy;}

  // X
  P.x+=P.vx*h;
  for(const s of solids){if(s.dead||s.t==='wall'&&false)continue;if(hit(P,s)){
    if(P.vx>0||(P.x+P.w/2)<(s.x+s.w/2))P.x=s.x-P.w;else P.x=s.x+s.w;
    if(P.dashT>0){P.dashT=0;P.vx=-P.dashDir*240;SFX.bonk();dustBurst(P.x+(P.vx<0?0:P.w),P.y+15,6,false);}
    else if(Math.abs(P.vx)>560){shake=Math.max(shake,3);dustBurst(P.x+(P.vx<0?0:P.w),P.y+15,5,false);}
    P.vx=0;}}
  // Y
  const prevG=P.grounded;const fallV=P.vy;
  P.y+=P.vy*h;P.grounded=false;let ref=null;
  for(const s of solids){if(s.dead)continue;if(hit(P,s)){
    if(P.vy>=0&&(P.y+P.h-P.vy*h)<=s.y+10){P.y=s.y-P.h;const v=P.vy;P.vy=0;P.grounded=true;ref=s;}
    else if(P.vy<0){P.y=s.y+s.h;if(-P.vy>320)SFX.bonk();P.vy=0;}}
  }
  if(P.grounded){P.coyote=.1;P.usedDouble=false;P.groundRef=ref;
    if(!prevG&&fallV>520){P.landSquash=.16;SFX.land(fallV);dustBurst(P.x+16,P.y+30,Math.min(12,4+fallV/180),false);shake=Math.max(shake,Math.min(6,fallV/260));}
    if(ref&&ref.t==='crumble'&&ref.state===0){ref.state=1;SFX.crumble();}
    if(Math.abs(P.vx)>240){runDust-=h;if(runDust<=0){runDust=.09;dustBurst(P.x+16-Math.sign(P.vx)*10,P.y+30,1,false);}}
    P.runPh+=Math.abs(P.vx)*h*.09;
  }else P.groundRef=null;

  // springs
  for(const sp of springs){const box={x:sp.x,y:sp.y-sp.h,w:sp.w,h:sp.h};
    if(hit(P,box)&&P.vy>=-50){P.y=sp.y-sp.h-P.h;P.vy=-1250;P.grounded=false;P.usedDouble=false;
      sp.anim=1;SFX.spring();shake=Math.max(shake,4);sparks(sp.x+sp.w/2,sp.y-8,10,'255,93,115',220);ringFx(sp.x+sp.w/2,sp.y-6,'rgba(255,210,122,.7)',44);}}
  if(!active)return;
  // hazards & pickups
  for(const sp of spikes)if(hit(P,{x:sp.x+8,y:sp.y-22,w:sp.w-16,h:22})){die();return;}
  if(P.y>KILL_Y){die();return;}
  const pcx=P.x+P.w/2,pcy=P.y+P.h/2;
  for(const e of embers){if(e.got)continue;const dx=pcx-e.x,dy=pcy-e.y;
    if(dx*dx+dy*dy<900){e.got=true;emberCount++;SFX.ember();sparks(e.x,e.y,8,'255,210,122',160);txtFx(e.x,e.y-14,'+1','#ffd27a');}}
  for(const c of checks){if(c.on)continue;
    if(hit(P,{x:c.x-30,y:c.y-110,w:60,h:110})){c.on=true;respPt={x:c.x-16,y:c.y-30};SFX.check();
      confetti(c.x,c.y-70,18);txtFx(c.x,c.y-96,'CHECKPOINT!','#ffd27a');shake=Math.max(shake,3);}}
  if(hit(P,{x:goal.x-36,y:goal.y-118,w:72,h:118}))winGame();
}
function updatePlayer(dt){
  if(P.dead>0){P.dead-=dt;if(P.dead<=0)respawn();return;}
  const n=Math.min(4,Math.ceil(dt/.0083)),h=dt/n;
  for(let i=0;i<n;i++)physStep(h);
}

function camUpdate(dt){let tx,ty;
  if(state==='menu'){tx=250+Math.sin(AT*.3)*70;ty=CAMY;}
  else{tx=P.x+P.w/2+P.vx*.16;ty=P.y+P.h/2-40;}
  const k=1-Math.exp(-dt*7);cam.x=lerp(cam.x,tx,k);cam.y=lerp(cam.y,ty,k*.8);
  const vw=VW/ZOOM/2,vh=VH/ZOOM/2;
  cam.x=vw<LW/2?clamp(cam.x,vw,LW-vw):LW/2;
  cam.y=vh<LH/2?clamp(cam.y,vh,LH-vh):LH/2;}

function rr(x,y,w,h,r){r=Math.min(r,w/2,h/2);ctx.beginPath();
  ctx.moveTo(x+r,y);ctx.arcTo(x+w,y,x+w,y+h,r);ctx.arcTo(x+w,y+h,x,y+h,r);ctx.arcTo(x,y+h,x,y,r);ctx.arcTo(x,y,x+w,y,r);ctx.closePath();}
