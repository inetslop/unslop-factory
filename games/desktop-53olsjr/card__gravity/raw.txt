function hitPlane(c,px,py,nx,ny,pen,e){

  const j=-(1+e)*vn/(1+invI*crn*crn);

    lx=clamp(lx,-hw,hw);lyl=clamp(lyl,-hh,hh);
    const wx=c.x+lx*ca-lyl*sa,wy=c.y+lx*sa+lyl*ca;
    let nx=wx-bx,ny=wy-by;const d=Math.hypot(nx,ny),rb=9*S;
    if(d<.001){nx=0;ny=-1;}else{nx/=d;ny/=d;}
    const pen=rb-d;
    if(pen>0)imp=Math.max(imp,hitPlane(c,wx,wy,nx,ny,pen,.5));
  }
  if(imp>420*S)sBounce(Math.min(1600,imp));
}
function pairHit(a,b){
  const dx=b.x-a.x,dy=b.y-a.y,minD=rc*2,d2=dx*dx+dy*dy;
  if(d2>=minD*minD||d2===0)return;
  const d=Math.sqrt(d2),nx=dx/d,ny=dy/d,ov=(minD-d);
  a.x-=nx*ov/2;a.y-=ny*ov/2;b.x+=nx*ov/2;b.y+=ny*ov/2;
  const rvx=b.vx-a.vx,rvy=b.vy-a.vy,vn=rvx*nx+rvy*ny;
  if(vn<0){
    const j=-(1+.3)*vn/2;
    a.vx-=j*nx;a.vy-=j*ny;b.vx+=j*nx;b.vy+=j*ny;
    const tx=-ny,ty=nx,vt=rvx*tx+rvy*ty;
    a.w=clamp(a.w-vt*.0011,-9,9);b.w=clamp(b.w+vt*.0011,-9,9);
  }
}
function physStep(h){
  const tX=((keys.right?TILT:0)-(keys.left?TILT:0))*(zgOn?.2:1);
  grav.x+=(tX-grav.x)*(1-Math.exp(-h*9));
  const gY=zgOn?G*.04:G;
  for(const c of cards){
    if(c.state!=='live')continue;
    c.vx+=grav.x*h;c.vy+=gY*h;
    const dr=Math.exp(-h*.4);c.vx*=dr;c.vy*=dr;
    c.w*=Math.exp(-h*.55);c.w=clamp(c.w,-9,9);
    c.x+=c.vx*h;c.y+=c.vy*h;c.a+=c.w*h;
  }
  for(const c of cards)if(c.state==='live')collideCard(c);
  for(let i=0;i<cards.length;i++)for(let j=i+1;j<cards.length;j++){
    if(cards[i].state==='live'&&cards[j].state==='live')pairHit(cards[i],cards[j]);
  }
}

function sparks(x,y,col,n,spd){
  for(let i=0;i<n;i++){
    const a=rnd(0,6.28),sp=rnd(.3,1)*spd;
    parts.push({x,y,vx:Math.cos(a)*sp,vy:Math.sin(a)*sp-90*S,t:0,life:rnd(.4,.95),col,sz:rnd(2,5)*S,conf:Math.random()<.35,a:rnd(0,6.28),spin:rnd(-8,8)});
  }
}
function bankCard(c){
  c.state='bank';c.t=0;c.sx=c.x;c.sy=c.y;c.sa=c.a;
  combo=(lastSuit===c.suit)?combo+1:1;lastSuit=c.suit;
  maxCombo=Math.max(maxCombo,combo);
  const mult=Math.min(combo,8),pts=(c.gold?400:100)*mult;
  score+=pts;banked++;
  bankStack.push({suit:c.suit,gold:c.gold});if(bankStack.length>6)bankStack.shift();
  floats.push({x:c.x,y:c.y-24*S,txt:'+'+fmt(pts),col:c.gold?'#ffd970':'#f4ecd7',size:(c.gold?30:24)*S,t:0,life:.95});
  if(combo>=2)floats.push({x:W/2,y:H*.34,txt:SNAMES[c.suit].toUpperCase()+' \u00d7'+mult,col:(c.suit===1||c.suit===2)?'#ff8b97':'#f7d97c',size:clamp(W*.055,26*S,44*S),t:0,life:1.15});
  sparks(c.x,floorY-18*S,c.gold?'#ffd970':(c.suit===1||c.suit===2?'#ff6b7a':'#f4ecd7'),c.gold?28:14,340*S);
  sBank(combo);
  const nl=1+Math.floor(banked/7);
  if(nl>level){
    level=nl;wavePulse=1;sLevel();
    rackW=Math.max(96*S,176*S-(level-1)*12*S);
    banner={txt:'WAVE '+level,t:0,life:1.5};
  }
  updateHUD(false);
}
function missCard(c){
  c.state='lost';c.t=0;hearts--;
  updateHearts(true);shake=Math.max(shake,13*S);
  sparks(c.x,floorY-10*S,'#ff5a68',12,260*S);
  sMiss();
  if(hearts<=0){pendingOver=true;overTimer=setTimeout(showOver,850);}
}

/* ============================== update ============================== */
function makeDeco(anyY){return{x:rnd(40,W-40),y:anyY?rnd(-H*.6,H): -ch*1.2,a:rnd(0,6.28),va:rnd(-.5,.5),vy:rnd(34,72)*S,vx:rnd(-16,16)*S,suit:(Math.random()*4)|0,rank:RANKS[(Math.random()*13)|0],gold:false};}
function update(dt){
  shake*=Math.exp(-dt*7);if(shake<.3)shake=0;
  wavePulse*=Math.exp(-dt*2.2);
  if(banner){banner.t+=dt;if(banner.t>banner.life)banner=null;}
  for(let i=pips.length-1;i>=0;i--){const p=pips[i];p.y+=p.vy*dt;if(p.y>H+60*S){pips[i]=makePip(false);}}
  for(let i=parts.length-1;i>=0;i--){const p=parts[i];p.t+=dt;if(p.t>p.life){parts.splice(i,1);continue;}p.x+=p.vx*dt;p.vy+=(p.conf?900:620)*S*dt*.6;p.y+=p.vy*dt;p.a+=p.spin*dt;}
  for(let i=floats.length-1;i>=0;i--){const f=floats[i];f.t+=dt;f.y-=42*S*dt;if(f.t>f.life)floats.splice(i,1);}

  if(state!=='play'){
    if(decos.length<5&&Math.random()<.05)decos.push(makeDeco(false));
    for(const d of decos){d.x+=d.vx*dt;d.y+=d.vy*dt;d.a+=d.va*dt;if(d.y>H+ch*1.2){Object.assign(d,makeDeco(false));}}
    return;
  }
  // zero-G meter
  const want=(keys.zg||zgPointer);
  if(want&&zgMeter>0){if(!zgOn){zgOn=true;sZg(true);}zgMeter=Math.max(0,zgMeter-dt*.45);}
  else{if(zgOn){zgOn=false;sZg(false);}zgMeter=Math.min(1,zgMeter+dt*.22);}
  zgFill.style.transform='scaleX('+zgMeter.toFixed(3)+')';
  zgBar.classList.toggle('low',zgMeter<.25);

  // spawner
  const maxC=Math.min(6,2+((level/2)|0));
  let live=0;for(const c of cards)if(c.state==='live')live++;
  if(live<maxC){spawnT-=dt;if(spawnT<=0){spawnCard();spawnT=Math.max(.5,1.6-level*.09)*rnd(.75,1.25);}}

  // physics (2 substeps)
  physStep(dt/2);physStep(dt/2);

  // rest / resolve
  if(!zgOn)for(const c of cards){
    if(c.state!=='live')continue;
    c.age+=dt;
    const sp=Math.hypot(c.vx,c.vy);
    if(sp<45*S&&Math.abs(c.w)<1.8&&c.y>floorY-ch*1.3)c.restT+=dt;else c.restT=0;
    if(c.restT>.5||c.age>16){
      if(Math.abs(c.x-rackX)<=rackW/2-4*S&&c.y>floorY-ch*1.7)bankCard(c);else missCard(c);
    }
  }
  // anim states
  for(let i=cards.length-1;i>=0;i--){
    const c=cards[i];
    if(c.state==='bank'){c.t+=dt/.45;if(c.t>=1)cards.splice(i,1);}
    else if(c.state==='lost'){c.t+=dt/.6;c.y+=(280*S+1400*S*c.t)*dt;c.a+=dt*2.2;if(c.t>=1)cards.splice(i,1);}
  }
}

function corner(w,h,rank,suit,col,mirror){
  ctx.save();if(mirror)ctx.rotate(Math.PI);

  ctx.fillStyle=col;ctx.textAlign='left';ctx.textBaseline='top';

  ctx.font=(14*S)+'px Georgia, serif';ctx.fillText(suit,x,-h/2+6*S+21*S);
  ctx.restore();
}
function drawCard(x,y,a,sc,alpha,suit,rank,gold,tint){
  ctx.save();ctx.translate(x,y);ctx.rotate(a);ctx.scale(sc,sc);ctx.globalAlpha=alpha;
  const w=cw,h=ch,r=8*S;
  rr(-w/2,-h/2,w,h,r);ctx.fillStyle=gold?goldFace:faceGrad;ctx.fill();
  ctx.lineWidth=1.6*S;ctx.strokeStyle=gold?'#c99b3f':'#b7ab8e';ctx.stroke();
  if(gold){rr(-w/2+4*S,-h/2+4*S,w-8*S,h-8*S,r*.7);ctx.strokeStyle='rgba(160,120,30,.5)';ctx.lineWidth=S;ctx.stroke();}
  const col=gold?'#9a741d':SCOL[suit];
  ctx.fillStyle=col;ctx.font=(46*S)+'px Georgia, serif';ctx.textAlign='center';ctx.textBaseline='middle';

}

function drawRail(){

function drawCompass(){
  const cx=rackX,cy=floorY-ch*2.6;
  ctx.save();
  if(zgOn){ctx.setLineDash([4*S,5*S]);ctx.strokeStyle='rgba(160,220,255,.7)';}
  else ctx.strokeStyle='rgba(244,236,215,.4)';
  ctx.lineWidth=1.5*S;ctx.beginPath();ctx.arc(cx,cy,15*S,0,7);ctx.stroke();ctx.setLineDash([]);
  if(!zgOn){
    const L=Math.hypot(grav.x,G)||1,ux=grav.x/L,uy=G/L;
    ctx.strokeStyle='rgba(240,205,110,.9)';ctx.lineWidth=2.5*S;ctx.lineCap='round';
    ctx.beginPath();ctx.moveTo(cx-ux*8*S,cy-uy*8*S);ctx.lineTo(cx+ux*10*S,cy+uy*10*S);ctx.stroke();
    const ang=Math.atan2(uy,ux),tx=cx+ux*10*S,ty=cy+uy*10*S;
    ctx.beginPath();ctx.moveTo(tx,ty);ctx.lineTo(tx-6*S*Math.cos(ang-.5),ty-6*S*Math.sin(ang-.5));
    ctx.moveTo(tx,ty);ctx.lineTo(tx-6*S*Math.cos(ang+.5),ty-6*S*Math.sin(ang+.5));ctx.stroke();
  }else{ctx.fillStyle='rgba(160,220,255,.8)';ctx.beginPath();ctx.arc(cx,cy,3.5*S,0,7);ctx.fill();}

}
function drawChevrons(){
  const amt=grav.x/TILT;
  if(Math.abs(amt)<.12)return;
  const dir=Math.sign(amt),x0=dir>0?W-22*S:22*S;
  ctx.fillStyle='rgba(240,205,110,'+(.3*Math.abs(amt)).toFixed(3)+')';
  for(let i=0;i<3;i++){

    ctx.beginPath();ctx.moveTo(x0-dir*8*S,y-10*S);ctx.lineTo(x0-dir*8*S,y+10*S);ctx.lineTo(x0+dir*8*S,y);ctx.closePath();ctx.fill();
  }
}
function render(){
  ctx.fillStyle=feltGrad;ctx.fillRect(0,0,W,H);
  if(noisePat){ctx.globalAlpha=.5;ctx.fillStyle=noisePat;ctx.fillRect(0,0,W,H);ctx.globalAlpha=1;}
  ctx.fillStyle=glowGold;ctx.fillRect(0,0,W,H);
  ctx.fillStyle=glowRed;ctx.fillRect(0,0,W,H);
  // drifting pips
  ctx.textAlign='center';ctx.textBaseline='middle';
  for(const p of pips){
    const dx=p.x+Math.sin(tGlobal*.6+p.ph)*18*S;
    ctx.globalAlpha=p.a*(1+.3*Math.sin(tGlobal+p.ph));
    ctx.fillStyle='#f4ecd7';ctx.font=p.size+'px Georgia, serif';
    ctx.fillText(SUITS[p.suit],dx,p.y);
  }
  ctx.globalAlpha=1;

  const shx=(Math.random()-.5)*shake,shy=(Math.random()-.5)*shake;
  ctx.save();ctx.translate(shx,shy);
  drawRail();
  if(state!=='menu')drawRack();
  if(state==='play'&&!paused)drawCompass();
  if(state==='play'&&zgOn){

  }
  if(state==='play'&&!paused)drawChevrons();

  // shadows
  for(const c of cards){
    if(c.state==='lost')continue;
    const hgt=floorY-(c.y+ch/2),k=clamp(1-hgt/(H*.55),0,1);
    if(k<=0)continue;
    ctx.globalAlpha=.3*k;ctx.fillStyle='#000';
    ctx.beginPath();ctx.ellipse(c.x,floorY+7*S,cw*.44*(.5+.5*k)+6*S,7*S,0,0,7);ctx.fill();
  }
  ctx.globalAlpha=1;

  for(const d of decos)drawCard(d.x,d.y,d.a,.92,.85,d.suit,d.rank,false,0);

    if(c.state==='live')drawCard(c.x,c.y,c.a,1,1,c.suit,c.rank,c.gold,0);
    else if(c.state==='bank'){
      const e=easeInCubic(Math.min(1,c.t));
      const x=lerp(c.sx,rackX,e),y=lerp(c.sy,floorY+14*S,e);
      const al=c.t>.7?1-(c.t-.7)/.3:1;
      drawCard(x,y,c.sa*(1-e),1-.62*e,al,c.suit,c.rank,c.gold,0);
    }else if(c.state==='lost')drawCard(c.x,c.y,c.a,1,Math.max(0,1-c.t),c.suit,c.rank,c.gold,1-c.t);
  }

  // particles
  for(const p of parts){
    const f=1-p.t/p.life;ctx.globalAlpha=f;
    if(p.conf){ctx.save();ctx.translate(p.x,p.y);ctx.rotate(p.a);ctx.fillStyle=p.col;ctx.fillRect(-p.sz,-p.sz*.6,p.sz*2,p.sz*1.2);ctx.restore();}
    else{ctx.fillStyle=p.col;ctx.beginPath();ctx.arc(p.x,p.y,p.sz*f,0,7);ctx.fill();}
  }
  ctx.globalAlpha=1;

  // floaters
  for(const f of floats){

    ctx.font='bold '+(f.size*sc)+'px Georgia, serif';
    ctx.textAlign='center';ctx.textBaseline='middle';

  }

  // banner
  if(banner){

  }
  ctx.restore();

  if(zgOn&&state==='play'){ctx.fillStyle='rgba(150,210,255,.05)';ctx.fillRect(0,0,W,H);}
  ctx.fillStyle=vigGrad;ctx.fillRect(0,0,W,H);
}
