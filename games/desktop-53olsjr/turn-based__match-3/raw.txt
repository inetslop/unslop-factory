  bx=Math.round((W-cell*COLS)/2);by=hud+Math.max(10,Math.round((H-hud-16-cell*ROWS)/2));
  buildSprites();buildBg();
}

  const cx=bx+cell*COLS/2,cy=by+cell*ROWS/2;
  warmGlow=ctx.createRadialGradient(cx,cy,0,cx,cy,Math.max(cell*6.5,300));warmGlow.addColorStop(0,'rgba(246,196,83,.10)');warmGlow.addColorStop(1,'rgba(246,196,83,0)');
  coolGlow=ctx.createRadialGradient(W*.15,H*.9,0,W*.15,H*.9,H*.7);coolGlow.addColorStop(0,'rgba(64,180,200,.08)');coolGlow.addColorStop(1,'rgba(64,180,200,0)');
  vigGrad=ctx.createRadialGradient(W/2,H/2,Math.min(W,H)*.35,W/2,H/2,Math.max(W,H)*.75);vigGrad.addColorStop(0,'rgba(0,0,0,0)');vigGrad.addColorStop(1,'rgba(3,9,13,.6)');
  crystals=[];const defs=[{x:.07,y:1.02,s:.42,n:3},{x:.2,y:1.05,s:.3,n:2},{x:.87,y:1.03,s:.4,n:3},{x:.72,y:1.06,s:.24,n:2}];
  for(const d of defs)for(let i=0;i<d.n;i++)crystals.push({x:d.x*W+rand(-46,46),y:d.y*H,w:H*d.s*.13*rand(.7,1.4),h:H*d.s*rand(.45,1),ph:rand(6)});

function buildSprites(){
  for(let t=0;t<NTYPES;t++){
    const sw=Math.ceil(cell*1.75*dpr),c=document.createElement('canvas');c.width=c.height=sw;
    const g=c.getContext('2d'),C=sw/2,R=sw*.31,gc=GEMS[t];
    const halo=g.createRadialGradient(C,C,R*.4,C,C,R*1.6);halo.addColorStop(0,gc.main+'aa');halo.addColorStop(.55,gc.main+'33');halo.addColorStop(1,gc.main+'00');
    g.fillStyle=halo;g.beginPath();g.arc(C,C,R*1.6,0,7);g.fill();
    const body=g.createLinearGradient(C-R,C-R,C+R,C+R);body.addColorStop(0,gc.hi);body.addColorStop(.45,gc.main);body.addColorStop(1,gc.lo);
    g.translate(C,C);
    if(t===1){g.strokeStyle=gc.main;g.lineCap='round';g.lineWidth=R*.16;for(let i=0;i<8;i++){const a=i*Math.PI/4+Math.PI/8;g.beginPath();g.moveTo(Math.cos(a)*R*.72,Math.sin(a)*R*.72);g.lineTo(Math.cos(a)*R*1.05,Math.sin(a)*R*1.05);g.stroke();}
      g.fillStyle=body;g.beginPath();g.arc(0,0,R*.8,0,7);g.fill();g.strokeStyle='rgba(255,255,255,.3)';g.lineWidth=R*.09;g.beginPath();g.arc(0,0,R*.5,0,7);g.stroke();}
    else{
      const path=(sc)=>{g.beginPath();
        if(t===0){g.moveTo(0,-R*sc);g.bezierCurveTo(R*.98*sc,-R*.36*sc,R*.98*sc,R*.36*sc,0,R*sc);g.bezierCurveTo(-R*.98*sc,R*.36*sc,-R*.98*sc,-R*.36*sc,0,-R*sc);}
        else if(t===2){for(let i=0;i<6;i++){const a=-Math.PI/2+i*Math.PI/3;const x=Math.cos(a)*R*.98*sc,y=Math.sin(a)*R*.98*sc;i?g.lineTo(x,y):g.moveTo(x,y);}g.closePath();}
        else if(t===3){g.moveTo(0,-R);g.lineTo(R*.72*sc,-R*.15*sc);g.lineTo(R*.45*sc,R*.85*sc);g.lineTo(-R*.45*sc,R*.85*sc);g.lineTo(-R*.72*sc,-R*.15*sc);g.closePath();}
        else if(t===4){g.moveTo(0,-R*.96*sc);g.lineTo(R*.96*sc,0);g.lineTo(0,R*.96*sc);g.lineTo(-R*.96*sc,0);g.closePath();}
        else{for(let i=0;i<5;i++){const a=-Math.PI/2+i*2*Math.PI/5;const x=Math.cos(a)*R*.94*sc,y=Math.sin(a)*R*.94*sc;i?g.lineTo(x,y):g.moveTo(x,y);}g.closePath();}
      };
      path(1);g.fillStyle=body;g.fill();g.strokeStyle=gc.lo;g.lineWidth=R*.08;g.stroke();
      g.strokeStyle='rgba(255,255,255,.26)';g.lineWidth=R*.07;
      if(t===3){g.beginPath();g.moveTo(0,-R*.7);g.lineTo(0,R*.6);g.moveTo(-R*.4,-R*.1);g.lineTo(0,R*.6);g.moveTo(R*.4,-R*.1);g.lineTo(0,R*.6);g.stroke();}
      else if(t===4){g.beginPath();g.moveTo(0,-R*.96);g.lineTo(0,R*.96);g.moveTo(-R*.96,0);g.lineTo(R*.96,0);g.stroke();}

    }

/* ================= board data ================= */
function newGem(t){return{t,s:0,fx:0,fy:0,sq:1,flipT:0,flash:0,shake:0,dying:false,dieT0:0,popK:-1,burst:false,dead:false,anim:null,willSquash:false};}
function genBoard(attempts=0){
  grid=[];
  for(let r=0;r<ROWS;r++){const row=[];for(let c=0;c<COLS;c++){let t,g2;do{t=Math.random()*NTYPES|0;g2=(c>=2&&row[c-1].t===t&&row[c-2].t===t)||(r>=2&&grid[r-1][c].t===t&&grid[r-2][c].t===t);}while(g2);const g=newGem(t);g.fx=c;g.fy=r;row.push(g);}grid.push(row);}
  if(!findHint()&&attempts<40)genBoard(attempts+1);
}
function findMatches(){
  const cells=new Set(),runs=[];
  for(let r=0;r<ROWS;r++){let c=0;while(c<COLS){const g=grid[r][c];if(g&&g.s===0&&!g.dying){let e=c+1;while(e<COLS){const h=grid[r][e];if(h&&h.s===0&&!h.dying&&h.t===g.t)e++;else break;}
    if(e-c>=3){for(let i=c;i<e;i++)cells.add(r*COLS+i);runs.push({axis:0,i:r,a0:c,a1:e});}c=e;}else c++;}}
  for(let c=0;c<COLS;c++){let r=0;while(r<ROWS){const g=grid[r][c];if(g&&g.s===0&&!g.dying){let e=r+1;while(e<ROWS){const h=grid[e][c];if(h&&h.s===0&&!h.dying&&h.t===g.t)e++;else break;}
    if(e-r>=3){for(let i=r;i<e;i++)cells.add(i*COLS+c);runs.push({axis:1,i:c,a0:r,a1:e});}r=e;}else r++;}}
  const spawns=new Map();
  for(const a of runs)for(const b of runs){if(a.axis===b.axis)continue;
    const row=a.axis===0?a:b,col=a.axis===0?b:a;
    if(col.a0<=row.i&&row.i<col.a1&&row.a0<=col.i&&col.i<row.a1)spawns.set(row.i*COLS+col.i,2);}
  const inRun=(run,k)=>{const r=(k/COLS)|0,c=k%COLS;return run.axis===0?(r===run.i&&run.a0<=c&&c<run.a1):(c===run.i&&run.a0<=r&&r<run.a1);};
  for(const run of runs){let covered=false;for(const k of spawns.keys())if(inRun(run,k)){covered=true;break;}if(covered)continue;
    const len=run.a1-run.a0,sp=len>=5?2:1;
    let k=-1;if(lastSwap)for(const s of lastSwap)if(inRun(run,s)){k=s;break;}
    if(k<0)k=run.axis===0?run.i*COLS+((run.a0+run.a1)>>1):((run.a0+run.a1)>>1)*COLS+run.i;
    const old=spawns.get(k);if(old!==2)spawns.set(k,sp);}
  for(const k of spawns.keys())cells.delete(k);
  return{cells,spawns};
}
function frequentType(){const cnt=new Array(NTYPES).fill(0);for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++){const g=grid[r][c];if(g&&g.s===0&&!g.dying)cnt[g.t]++;}let m=0;for(const v of cnt)m=Math.max(m,v);if(!m)return 0;const out=[];cnt.forEach((v,i)=>{if(v===m)out.push(i);});return out[Math.random()*out.length|0];}
function findHint(){
  for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++){const g=grid[r][c];if(g&&g.s>0){const n=c<COLS-1?{r,c:c+1}:{r:r+1,c};return[{r,c},n];}}
  for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++){
    const A=grid[r][c];if(!A)continue;
    if(c<COLS-1){const B=grid[r][c+1];grid[r][c]=B;grid[r][c+1]=A;const m=findMatches().cells.size>0;grid[r][c]=A;grid[r][c+1]=B;if(m)return[{r,c},{r,c:c+1}];}
    if(r<ROWS-1){const B=grid[r+1][c];grid[r][c]=B;grid[r+1][c]=A;const m=findMatches().cells.size>0;grid[r][c]=A;grid[r+1][c]=B;if(m)return[{r,c},{r:r+1,c}];}}
  return null;
}

  const gc=GEMS[t]||{hi:'#fff',main:'#fff'};
  for(let i=0;i<n;i++){const a=rand(Math.PI*2),sp=rand(40,300);

  if(parts.length>450)parts.splice(0,parts.length-450);
}

/* ================= animation core ================= */
function tweenGem(g,x2,y2,ms,ease){return new Promise(res=>{g.anim={x0:g.fx,y0:g.fy,x2,y2,t0:performance.now(),ms,ease,res};});}
function stepAnims(now){
  for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++){const g=grid[r][c];if(!g||!g.anim)continue;
    const a=g.anim,k=(now-a.t0)/a.ms;
    if(k>=1){g.fx=a.x2;g.fy=a.y2;g.anim=null;a.res();}
    else{const e=a.ease(clamp(k,0,1));g.fx=lerp(a.x0,a.x2,e);g.fy=lerp(a.y0,a.y2,e);}}
}
function flushAnims(){for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++){const g=grid[r][c];if(g&&g.anim){const a=g.anim;g.anim=null;a.res();}}}
function clearFx(){parts.length=0;pops.length=0;beams.length=0;rings.length=0;bannerMsg=null;}

/* ================= cascade pipeline ================= */
function expandSpecials(set){
  const q=[...set],seen=new Set(q),now=performance.now();
  const add=k=>{if(!seen.has(k)){seen.add(k);set.add(k);q.push(k);}};
  while(q.length){const k=q.pop(),r=(k/COLS)|0,c=k%COLS,g=grid[r][c];if(!g||!g.s)continue;
    if(g.s===1){for(let i=0;i<COLS;i++)add(r*COLS+i);for(let i=0;i<ROWS;i++)add(i*COLS+c);
      beams.push({axis:0,i:r,t0:now},{axis:1,i:c,t0:now});sfx.prism();popup(cellX(c),cellY(r),'PRISM!',cell*.5);}
    else{const t=frequentType();for(let rr=0;rr<ROWS;rr++)for(let cc=0;cc<COLS;cc++){const h=grid[rr][cc];if(h&&h.s===0&&h.t===t)add(rr*COLS+cc);}
      rings.push({x:cellX(c),y:cellY(r),t0:now});sfx.nova();popup(cellX(c),cellY(r),'NOVA!',cell*.5);}}
}
function startPops(set,spawns){
  const now=performance.now();
  set.forEach(k=>{const r=(k/COLS)|0,c=k%COLS,g=grid[r][c];if(!g)return;
    if(spawns.has(k)){g.s=spawns.get(k);g.flash=now;burst(cellX(c),cellY(r),Math.random()*NTYPES|0,14);sfx.special();}
    else{g.dying=true;g.dieT0=now+rand(0,50);}});
}
function addScore(pts,set,chain){
  score+=pts;
  let sx=0,sy=0;set.forEach(k=>{sx+=(k%COLS);sy+=(k/COLS|0);});
  popup(cellX(sx/set.size),cellY(sy/set.size),'+'+fmt(pts)+(chain>1?' ×'+chain:''),clamp(cell*.42,16,32));
  const sv=$('scoreVal');sv.classList.remove('bump');void sv.offsetWidth;sv.classList.add('bump');
  if(!quotaFlag&&score>=target){quotaFlag=true;sfx.ding();banner('QUOTA MET!');}
  updateHUD();
}
function collapseRefill(){
  const ps=[];
  for(let c=0;c<COLS;c++){let w=ROWS-1;
    for(let r=ROWS-1;r>=0;r--){const g=grid[r][c];if(g){if(w!==r){grid[w][c]=g;grid[r][c]=null;}w--;}}
    const e=w+1;
    for(let r=w;r>=0;r--){const g=newGem(Math.random()*NTYPES|0);g.fx=c;g.fy=r-e;grid[r][c]=g;}}
  for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++){const g=grid[r][c];
    if(g&&(g.fx!==c||g.fy!==r)){const d=Math.abs(r-g.fy);g.willSquash=d>.6;ps.push(tweenGem(g,c,r,clamp(140+d*80,160,620),easeDrop));}}
  return Promise.all(ps).then(()=>{let landed=false;
    for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++){const g=grid[r][c];if(g&&g.willSquash){g.sq=.72;landed=true;}}
    if(landed&&performance.now()-lastLandSfx>90){sfx.land();lastLandSfx=performance.now();}});
}
async function runCascade(my,initSet,initSpawns){
  let chain=1,set=initSet;curChain=1;
  while(true){
    if(!set){const m=findMatches();if(m.cells.size===0)break;set=m.cells;var curSpawns=m.spawns;}
    else{curSpawns=initSpawns||new Map();initSpawns=null;}
    lastSwap=null;
    expandSpecials(set);
    addScore(set.size*20*chain+curSpawns.size*50,set,chain);
    if(chain>=2)banner('CHAIN ×'+chain);
    startPops(set,curSpawns);
    await sleep(400);if(!alive(my))return;
    for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++){const g=grid[r][c];if(g&&g.dying&&g.dead)grid[r][c]=null;}
    await collapseRefill();if(!alive(my))return;
    chain++;curChain=chain;}
  curChain=1;
}

/* ================= turn flow ================= */
const alive=id=>id===runId;
function swapAnim(A,B,pA,pB){return Promise.all([tweenGem(A,pB.c,pB.r,170,easeOutBack),tweenGem(B,pA.c,pA.r,170,easeOutBack)]);}
function trySwap(a,b){
  if(phase!=='idle')return;
  if(Math.abs(a.r-b.r)+Math.abs(a.c-b.c)!==1)return;
  const my=runId;phase='busy';sel=null;hint=null;lastAct=performance.now();
  moves--;updateHUD();
  const A=grid[a.r][a.c],B=grid[b.r][b.c];if(!A||!B){phase='idle';return;}
  grid[a.r][a.c]=B;grid[b.r][b.c]=A;
  lastSwap=[b.r*COLS+b.c,a.r*COLS+a.c];sfx.swap();
  swapAnim(A,B,a,b).then(()=>{
    if(!alive(my))return;
    if(A.s>0||B.s>0){
      const set=new Set([a.r*COLS+a.c,b.r*COLS+b.c]);const now=performance.now();
      const blast=(g,pr,pc,partner)=>{
        if(g.s===1){for(let i=0;i<COLS;i++)set.add(pr*COLS+i);for(let i=0;i<ROWS;i++)set.add(i*COLS+pc);beams.push({axis:0,i:pr,t0:now},{axis:1,i:pc,t0:now});sfx.prism();}
        else{const t=(partner&&partner.s===0)?partner.t:frequentType();for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++){const h=grid[r][c];if(h&&h.s===0&&h.t===t)set.add(r*COLS+c);}rings.push({x:cellX(pc),y:cellY(pr),t0:now});sfx.nova();}};
      if(A.s>0)blast(A,b.r,b.c,B);
      if(B.s>0)blast(B,a.r,a.c,A);
      runCascade(my,set,new Map()).then(()=>endTurn(my));
      return;}
    const m=findMatches();lastSwap=null;
    if(m.cells.size===0){
      grid[a.r][a.c]=A;grid[b.r][b.c]=B;sfx.bad();const now=performance.now();A.shake=now;B.shake=now;
      swapAnim(A,B,b,a).then(()=>{if(!alive(my))return;moves++;updateHUD();phase='idle';lastAct=performance.now();});
    }else runCascade(my,m.cells,m.spawns).then(()=>endTurn(my));
  });
}
async function endTurn(my){
  if(!alive(my))return;phase='busy';
  if(score>=target){levelUp();return;}
  if(moves<=0){gameOver();return;}
  if(!findHint()){await reshuffle(my);if(!alive(my))return;}
  phase='idle';lastAct=performance.now();
}
async function reshuffle(my){
  banner('NO MOVES — RESHUFFLING');sfx.special();
  for(let att=0;att<40;att++){
    const plain=[],types=[];
    for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++){const g=grid[r][c];if(g&&g.s===0){plain.push(g);types.push(g.t);}}
    for(let i=types.length-1;i>0;i--){const j=Math.random()*(i+1)|0;[types[i],types[j]]=[types[j],types[i]];}
    plain.forEach((g,i)=>g.t=types[i]);
    if(findMatches().cells.size===0&&findHint())break;}
  const now=performance.now();
  for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++){const g=grid[r][c];if(g)g.flipT=now+rand(0,120);}
  await sleep(650);
}

function updateDying(now){
  frameBursts=0;
  for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++){const g=grid[r][c];if(!g)continue;
    if(g.sq<1)g.sq=Math.min(1,g.sq+(1-g.sq)*.16);
    if(g.dying&&!g.dead){const k=(now-g.dieT0)/320;g.popK=clamp(k,0,1);
      if(k>=.3&&!g.burst){g.burst=true;burst(cellX(c),cellY(r),g.t,9);frameBursts++;}
      if(k>=1)g.dead=true;}}
  if(frameBursts&&now-lastPopSfx>50){sfx.pop(curChain);lastPopSfx=now;}
}
function updateParts(dt){
  const s=dt/1000,now=performance.now();
  for(let i=parts.length-1;i>=0;i--){const p=parts[i];
    if(now-p.t0>p.life){parts.splice(i,1);continue;}
    p.vy+=900*s;p.x+=p.vx*s;p.y+=p.vy*s;p.rot+=p.vr*s;}
  for(let i=pops.length-1;i>=0;i--)if(now-pops[i].t0>950)pops.splice(i,1);
  for(let i=beams.length-1;i>=0;i--)if(now-beams[i].t0>420)beams.splice(i,1);
  for(let i=rings.length-1;i>=0;i--)if(now-rings[i].t0>650)rings.splice(i,1);
}
