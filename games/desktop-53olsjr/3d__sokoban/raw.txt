function flatQuad(x0,z0,x1,z1,y,fill,opts){const pts=[[x0,y,z0],[x1,y,z0],[x1,y,z1],[x0,y,z1]].map(p=>xf(p[0],p[1],p[2]));

function addBox(cx,cy,cz,hx,hy,hz,yaw,col,o){o=o||{};
 const cs=Math.cos(yaw),sn=Math.sin(yaw);
 const Wp=(lx,ly,lz)=>[cx+cs*lx+sn*lz,cy+ly,cz-sn*lx+cs*lz];
 const c=[Wp(-hx,-hy,-hz),Wp(hx,-hy,-hz),Wp(hx,hy,-hz),Wp(-hx,hy,-hz),Wp(-hx,-hy,hz),Wp(hx,-hy,hz),Wp(hx,hy,hz),Wp(-hx,hy,hz)];
 const facesDef=[
  {i:[3,2,6,7],n:[0,1,0],m:1.12,det:'top'},
  {i:[0,4,5,1],n:[0,-1,0],m:.5},
  {i:[1,2,6,5],n:[1,0,0],m:1},
  {i:[4,7,3,0],n:[-1,0,0],m:.92},
  {i:[4,5,6,7],n:[0,0,1],m:1.04},
  {i:[0,1,2,3],n:[0,0,-1],m:.85}];
 for(const fd of facesDef){
  const nx=cs*fd.n[0]+sn*fd.n[2],nz=-sn*fd.n[0]+cs*fd.n[2],ny=fd.n[1];
  if((ex-cx)*nx+(ey-cy)*ny+(ez-cz)*nz<=0)continue;
  const pts=fd.i.map(i=>xf(c[i][0],c[i][1],c[i][2]));
  if(pts.some(p=>p[2]<.4))continue;
  let det=null;
  if(o.det==='crate')det=fd.det==='top'?'crateTop':'crateSide';
  else if(o.det==='wall'&&fd.det==='top')det='wallTop';
  FACES.push({p:pts.map(q=>[W/2+F*q[0]/q[2],H/2-F*q[1]/q[2]]),
   d:(pts[0][2]+pts[1][2]+pts[2][2]+pts[3][2])/4,
   fill:o.em?'rgb('+col[0]+','+col[1]+','+col[2]+')':shade(col,nx,ny,nz,fd.m),
   stroke:dark(col,.85),lw:1,det});}}
