function genHole(n){
  const rnd=rng(n*9176+41);
  emitters=[];bullets=[];geese=[];gooseQ=[];bunkers=[];particles=[];floaters=[];
  cup={x:80+rnd()*(FW-160),y:105+rnd()*35}; start={x:FW/2,y:FH-70};
  holeTime=0; nearMisses=0; hoConfetti=false;
  spdM=Math.min(1+(n-1)*.06,1.5); rateM=Math.min(1+(n-1)*.07,1.6);
  // bunkers (slow zones)
  const nb=n>=2?((rnd()<.75?1:0)+(n>=5&&rnd()<.5?1:0)):0;
  for(let i=0;i<nb;i++)for(let tr=0;tr<30;tr++){
    const b={x:60+rnd()*(FW-120),y:280+rnd()*(FH-620),rx:55+rnd()*45,ry:36+rnd()*26};
    if(Math.hypot(b.x-start.x,b.y-start.y)>190&&Math.hypot(b.x-cup.x,b.y-cup.y)>170){bunkers.push(b);break}
  }
  const place=(minD,extra)=>{ for(let tr=0;tr<60;tr++){
      const x=70+rnd()*(FW-140), y=(extra||230)+rnd()*(FH-(extra||230)-450);
      let ok=Math.hypot(x-cup.x,y-cup.y)>170&&Math.hypot(x-start.x,y-start.y)>260;
      if(ok)for(const e of emitters)if(Math.hypot(x-e.x,y-e.y)<minD){ok=false;break}
      if(ok)return{x,y};
    } return {x:FW/2,y:FH/2-150}; };
  // windmills
  const nw=1+(n>=4?1:0)+(n>=9?1:0);
  for(let i=0;i<nw;i++){const p=place(170,230); emitters.push({type:'wm',x:p.x,y:p.y,
    arms:(n>=6&&rnd()<.5)?4:3,len:72+rnd()*30,rot:rnd()*TAU,om:(.9+rnd()*.5)*(rnd()<.5?-1:1),
    fireI:clamp(1.9-n*.07,1.0,1.9),t:rnd(),ringN:7+Math.min(n-1,8)});}
  // gnomes
  const ng=1+(n>=3?1:0)+(n>=6?1:0)+(n>=11?1:0);
  for(let i=0;i<ng;i++){const p=place(150,180); emitters.push({type:'gn',x:p.x,y:p.y,i,
    T:2.6-Math.min(n*.08,1.0),t:rnd()*1.2});}
  // sprinklers (edge spirals)
  const ns=n>=2?1+(n>=5?1:0):0;
  const corners=[[75,205],[FW-75,205],[75,FH-380],[FW-75,FH-380]];
  for(let i=0;i<ns&&i<corners.length;i++){const c=corners[(rnd()*4)|0];
    const x=c[0]+(rnd()-.5)*36,y=c[1]+(rnd()-.5)*36;
    emitters.push({type:'sp',x,y,th:rnd()*TAU,acc:0});}
  // ricochet balls
  const nb2=n>=3?Math.min(1+((n-3)/3|0),3):0;
  for(let i=0;i<nb2;i++){const a=rnd()*TAU,sp=150*spdM;
    bullets.push({x:100+rnd()*(FW-200),y:200+rnd()*(FH-380),vx:Math.cos(a)*sp,vy:Math.sin(a)*sp,
      r:8,type:'bb',nm:0,bounce:1,rot:rnd()*TAU});}
  // goose waves
  const gw=n>=2?Math.min(1+((n-2)/3|0),3):0; let gt=5+rnd()*4;
  for(let i=0;i<gw;i++){gooseQ.push({t:gt,y:160+rnd()*(FH-480),dir:rnd()<.5?1:-1,done:false});gt+=7+rnd()*3;}
}

function updateEmitters(wdt){
  const aim=aimPoint();
  for(const e of emitters){
    if(e.type==='wm'){
      e.rot+=e.om*wdt; e.t+=wdt;
      if(e.t>=e.fireI){e.t=0;const n=e.ringN;
        for(let i=0;i<n;i++){const a=e.rot+i*TAU/n,sp=130*spdM;
          spawnB(e.x,e.y,Math.cos(a)*sp,Math.sin(a)*sp,5.5,'wm');}}
    }else if(e.type==='gn'){
      e.t+=wdt;
      if(e.t>=e.T){e.t=0;
        const base=Math.atan2(aim.y-e.y,aim.x-e.x);
        const s=3+(hole>=4?1:0)+(hole>=8?1:0)+(hole>=12?1:0), sp=205*spdM;
        for(let i=0;i<s;i++){const a=base+(i-(s-1)/2)*.17;
          spawnB(e.x,e.y,Math.cos(a)*sp,Math.sin(a)*sp,5,'gn');}
        sfx.shot();
      }
    }else{ // sprinkler spiral
      e.acc+=wdt; const step=.115/rateM;
      while(e.acc>step){e.acc-=step;e.th+=.44;
        spawnB(e.x,e.y,Math.cos(e.th)*110*spdM,Math.sin(e.th)*110*spdM,6,'sp');}
    }
  }
}

      geese.push({x:q.dir>0?-30-back:FW+30+back,y:q.y+yo,vx:q.dir*95*spdM,ph:i*1.3,dt:Math.random()*.25});}
    sfx.honk();}

  for(let i=0;i<geese.length;i++){const g=geese[i];
    g.x+=g.vx*wdt; g.dt-=wdt;

    if(g.x>-70&&g.x<FW+70)geese[w++]=g;}

function updateBullets(wdt){
  let w=0;
  for(let i=0;i<bullets.length;i++){const b=bullets[i];
    b.x+=b.vx*wdt; b.y+=b.vy*wdt;
    if(b.bounce){
      if(b.x<b.r){b.x=b.r;b.vx=Math.abs(b.vx)} if(b.x>FW-b.r){b.x=FW-b.r;b.vx=-Math.abs(b.vx)}
      if(b.y<b.r){b.y=b.r;b.vy=Math.abs(b.vy)} if(b.y>FH-b.r){b.y=FH-b.r;b.vy=-Math.abs(b.vy)}
      b.rot+=wdt*6; bullets[w++]=b; continue;}
    if(b.x>-40&&b.x<FW+40&&b.y>-40&&b.y<FH+40)bullets[w++]=b;}
  bullets.length=w;
}
function inBunker(x,y){for(const b of bunkers){const dx=(x-b.x)/b.rx,dy=(y-b.y)/b.ry;if(dx*dx+dy*dy<1)return true}return false}

function updatePlayer(rdt){

function segDist(px,py,ax,ay,bx,by){const dx=bx-ax,dy=by-ay,l2=dx*dx+dy*dy;
  let t=((px-ax)*dx+(py-ay)*dy)/l2;t=clamp(t,0,1);
  return Math.hypot(px-(ax+t*dx),py-(ay+t*dy));}

function hit(){

}
function collide(){
  if(!player.alive)return;
  const pr=4.2;
  if(player.inv<=0){
    for(const b of bullets){const dx=b.x-player.x,dy=b.y-player.y;
      if(dx*dx+dy*dy<(b.r+pr)*(b.r+pr)){hit();return}}
    for(const e of emitters)if(e.type==='wm'){
      for(let a=0;a<e.arms;a++){const ang=e.rot+a*TAU/e.arms;
        if(segDist(player.x,player.y,e.x,e.y,e.x+Math.cos(ang)*e.len,e.y+Math.sin(ang)*e.len)<pr+4){hit();return}}}
    for(const g of geese){const dx=g.x-player.x,dy=g.y-player.y;
      if(dx*dx+dy*dy<(13+pr)*(13+pr)){hit();return}}
  }
  // near misses
  for(const b of bullets){if(b.nm)continue;
    const dx=b.x-player.x,dy=b.y-player.y,d2=dx*dx+dy*dy,rr=b.r+16;
    if(d2<rr*rr){b.nm=1;score+=5;nearMisses++;sfx.tick();
      particles.push({x:b.x,y:b.y,vx:0,vy:0,t:0,life:.3,size:14,col:'#ffffff',ring:1});}}
}

  }
}

  w=0;for(let i=0;i<floaters.length;i++){const f=floaters[i];f.t+=rdt;if(f.t<f.life)floaters[w++]=f}floaters.length=w;

  else if(state==='dying'){dieT+=rdt;const s2=wdt*.6;updateEmitters(s2);updateGeese(s2);updateBullets(s2);

  else if(state==='holeout'){hoT+=rdt;
    if(!hoConfetti&&hoT>1.05){hoConfetti=true;

    if(hoT>=1.7)nextHole();}
  hudUpdate(focus);
}

const GN=['#d94b5a','#e0a53c','#7a4fbf','#3fa35c'];

    ctx.beginPath();ctx.moveTo(6,4);ctx.lineTo(6+Math.cos(ang)*14,4+Math.sin(ang)*14);ctx.stroke();}
  ctx.fillStyle=c;ctx.beginPath();ctx.moveTo(-9,14);ctx.lineTo(9,14);ctx.lineTo(6,-2);ctx.lineTo(-6,-2);ctx.closePath();ctx.fill();
  ctx.strokeStyle='rgba(0,0,0,.25)';ctx.lineWidth=1.5;ctx.stroke();

function drawBullet(b){
  const c=BCOL[b.type];
  if(b.type==='bb'){drawBB(b);return}
  ctx.globalAlpha=.2;ctx.fillStyle=c;circle(b.x,b.y,b.r*1.85);
  ctx.globalAlpha=1;circle(b.x,b.y,b.r);
  ctx.fillStyle='#fff';circle(b.x,b.y,b.r*.45);
}
function drawPlayerBody(x,y,sc,alpha,rot){
  ctx.save();ctx.translate(x,y);ctx.scale(sc,sc);ctx.rotate(rot||0);ctx.globalAlpha=alpha;

function drawPlayer(focus){
  if(state==='dying'){
    drawPlayerBody(player.x,player.y,1+dieT*.9,Math.max(0,1-dieT),dieT*6);return;}
  for(let i=0;i<player.trail.length;i++){const t=player.trail[i];
    ctx.globalAlpha=i/player.trail.length*.12;ctx.fillStyle='#fff';circle(t.x,t.y,7*i/player.trail.length);}
  ctx.globalAlpha=1;
  const blink=player.inv>0&&Math.floor(T*18)%2===0;
  drawPlayerBody(player.x,player.y,1,blink?.35:1);
  if(focus){ctx.fillStyle='#ff4757';circle(player.x,player.y,3.6);
    ctx.strokeStyle='#fff';ctx.lineWidth=1.6;ctx.beginPath();ctx.arc(player.x,player.y,3.6,0,TAU);ctx.stroke();
    ctx.globalAlpha=.4+.25*Math.sin(T*9);ctx.strokeStyle='#8fe3ff';ctx.beginPath();ctx.arc(player.x,player.y,8.5,0,TAU);ctx.stroke();ctx.globalAlpha=1;}
}

function render(){
  ctx.setTransform(dpr,0,0,dpr,0,0);
  ctx.fillStyle='#123b22';ctx.fillRect(0,0,W,H);
  ctx.fillStyle=oobVign;ctx.fillRect(0,0,W,H);
  const shx=(Math.random()*2-1)*shake, shy=(Math.random()*2-1)*shake;
  ctx.save();ctx.translate(view.ox+shx,view.oy+shy);ctx.scale(view.s,view.s);
  // apron + striped fairway
  ctx.fillStyle='#8fbf72';ctx.fillRect(-16,-16,FW+32,FH+32);
  for(let i=0;i<9;i++){ctx.fillStyle=i%2?'#3d9a58':'#46a963';ctx.fillRect(i*64,0,64,FH)}
  ctx.strokeStyle='rgba(0,0,0,.2)';ctx.lineWidth=10;ctx.strokeRect(-7,-7,FW+14,FH+14);
  ctx.strokeStyle='rgba(255,255,255,.55)';ctx.lineWidth=2.5;ctx.strokeRect(1.5,1.5,FW-3,FH-3);
  for(const b of bunkers)drawBunker(b);
  // cup
  ctx.fillStyle='rgba(0,0,0,.3)';ctx.beginPath();ctx.ellipse(cup.x+4,cup.y+5,14,7,0,0,TAU);ctx.fill();
  ctx.fillStyle='#0b2413';circle(cup.x,cup.y,13);
  ctx.fillStyle='#06140b';circle(cup.x,cup.y,9);
  ctx.strokeStyle='rgba(255,255,255,.35)';ctx.lineWidth=1.5;ctx.beginPath();ctx.arc(cup.x,cup.y,13,0,TAU);ctx.stroke();
  for(const e of emitters){if(e.type==='wm')drawWindmill(e);else if(e.type==='gn')drawGnome(e);else drawSprinkler(e)}
  for(const g of geese)drawGoose(g);
  for(const b of bullets)drawBullet(b);
  if(state==='play'||state==='dying')drawPlayer(state==='play'&&player.alive&&keys['shift']);
  else if(state==='holeout'){
    const t=clamp(hoT/1.2,0,1),k=t*t;
    drawPlayerBody(lerp(player.x,cup.x,k),lerp(player.y,cup.y,k),Math.max(.05,1-k),1,t*8);}
  drawFlag();
  // particles
  for(const p of particles){const a=1-p.t/p.life;ctx.globalAlpha=a;
    if(p.ring){ctx.strokeStyle=p.col;ctx.lineWidth=2.5;ctx.beginPath();
      ctx.arc(p.x,p.y,p.size*(p.t/p.life),0,TAU);ctx.stroke();}
    else{ctx.fillStyle=p.col;circle(p.x,p.y,p.size)}}
  ctx.globalAlpha=1;
  // floaters
  ctx.textAlign='center';
  for(const f of floaters){const a=1-f.t/f.life;ctx.globalAlpha=a;
    ctx.font='700 '+f.size+'px "Trebuchet MS",sans-serif';
    ctx.strokeStyle='rgba(10,30,16,.8)';ctx.lineWidth=4;
    ctx.strokeText(f.txt,f.x,f.y-f.t*26);
    ctx.fillStyle=f.col;ctx.fillText(f.txt,f.x,f.y-f.t*26);}
  ctx.globalAlpha=1;
  // vignettes
  ctx.fillStyle=fieldVign;ctx.fillRect(0,0,FW,FH);
  if(state==='play'&&player.alive&&keys['shift']){ctx.fillStyle=focusGrad;ctx.fillRect(0,0,FW,FH)}
  ctx.restore();
  // leaves (screen space)
  for(const l of leaves){ctx.save();ctx.translate(l.x,l.y);ctx.rotate(l.rot);
    ctx.globalAlpha=.85;ctx.fillStyle=l.col;
    ctx.beginPath();ctx.ellipse(0,0,l.size,l.size*.45,0,0,TAU);ctx.fill();ctx.restore();}
  ctx.globalAlpha=1;
  if(flash>0){ctx.fillStyle='rgba(255,40,60,'+(flash*.3).toFixed(3)+')';ctx.fillRect(0,0,W,H)}
  if(lives===1&&state==='play'){const p=.1+.08*Math.sin(T*6);
    ctx.globalAlpha=p;ctx.fillStyle=dangerGrad;ctx.fillRect(0,0,W,H);ctx.globalAlpha=1}
}
