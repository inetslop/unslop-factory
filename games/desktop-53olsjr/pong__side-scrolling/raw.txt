  G.serveT=0;G.serveTo=Math.random()<0.5?-1:1;enemy.err=rand(-0.4,0.4)*enemy.vh*0.9;
  G.state='serve';
  banner("LET'S ROLL",'FIRST TO '+WIN,C.cream,1.1);SFX.start();
}
function beginServe(){ball.x=W/2;ball.y=H/2;ball.vx=ball.vy=0;ball.alive=false;G.trail.length=0;G.serveT=0;G.state='serve';}
function launch(attract){
  const dir=G.serveTo,ang=rand(-0.35,0.35);
  ball.speed=baseSpeed;ball.x=W/2;ball.y=H/2;ball.alive=true;
  ball.vx=Math.cos(ang)*ball.speed*dir;ball.vy=Math.sin(ang)*ball.speed;
  if(!attract)G.state='play';
}
function onPoint(who,attract){
  if(attract){ball.x=W/2;ball.y=H/2;ball.alive=false;G.serveT=0;G.serveTo=Math.random()<0.5?-1:1;return;}
  G.score[who]++;if(who==='p')G.popP=1;else G.popE=1;
  G.boost=1;G.shake=Math.min(12,W*0.012);ball.alive=false;
  burst(who==='p'?W+ball.r:-ball.r,clamp(ball.y,20,H-20),who==='p'?C.mint:C.coral,30,W*0.28);
  who==='p'?SFX.scoreP():SFX.scoreE();
  G.willEnd=G.score[who]>=WIN;G.serveTo=who==='p'?1:-1;
  G.state='point';G.pointT=0;
  banner(who==='p'?'POINT · YOU':'POINT · CPU',G.score.p+' — '+G.score.e,who==='p'?C.mint:C.coral,1.2);
}
function endMatch(){
  const won=G.score.p>=WIN;G.state='over';G.overAt=G.time;ball.speed=0;
  if(won)SFX.win();else SFX.lose();
  const prevR=best.bestRally||0,prevD=best.farthest||0;
  const nr=G.matchBestRally>prevD*0+G.matchBestRally>prevR?true:G.matchBestRally>prevR; // (kept explicit below)
  const newRec=(G.matchBestRally>prevR)||(G.meters>prevD);
  store.set({bestRally:Math.max(prevR,G.matchBestRally),farthest:Math.max(prevD,Math.floor(G.meters))});
  best=store.get();
  const rw=$('resultWord');rw.textContent=won?'VICTORY':'DEFEAT';
  rw.className='t-big '+(won?'c-mint':'c-coral');
  $('fsP').textContent=G.score.p;$('fsE').textContent=G.score.e;
  $('statDist').textContent=fmtDist(G.meters);$('statRally').textContent='×'+G.matchBestRally;
  $('recordBadge').classList.toggle('hidden',!newRec);
  $('endScreen').classList.remove('hidden');
}

/* ============ controls / AI ============ */
const keys={};
function humanControl(dt){
  const up=keys['ArrowUp']||keys['KeyW'],dn=keys['ArrowDown']||keys['KeyS'];
  let dy=0;
  if(up||dn){dy=((dn?1:0)-(up?1:0))*H*1.5*dt;player.target=null;}
  else if(player.target!=null){dy=clamp(player.target-player.y,-H*1.5*dt,H*1.5*dt);}
  player.vy=lerp(player.vy,dt>0?dy/dt:0,0.4);
  player.y=clamp(player.y+dy,player.vh/2,H-player.vh/2);
}
function foldY(y){const t=ball.r,b=H-ball.r,span=(b-t)*2;let m=(y-t)%span;if(m<0)m+=span;return m<=(b-t)?t+m:b-(m-(b-t));}
function controlAI(p,dt,isRight){
  let target=H*0.5;
  const toward=ball.alive&&(isRight?ball.vx>0:ball.vx<0);
  if(toward){
    const faceX=isRight?enemy.x:player.x+player.vw;
    const t=Math.abs((faceX-ball.x)/Math.max(60,Math.abs(ball.vx)));
    target=foldY(ball.y+ball.vy*t)+p.err;
  }
  const maxSp=(isRight?H*Math.min(1.28,0.95+G.rally*0.02):H*1.15)*dt;
  const step=clamp(target-p.y,-maxSp,maxSp);
  p.vy=lerp(p.vy,step/dt,0.3);
  p.y=clamp(p.y+step,p.vh/2,H-p.vh/2);
}

/* ============ ball physics ============ */
function hitPaddle(p,isRight){
  G.rally++;G.matchBestRally=Math.max(G.matchBestRally,G.rally);G.rallyPop=1;
  p.squash=1;
  const rel=clamp((ball.y-p.y)/(p.vh/2),-1,1),ang=rel*0.95;
  ball.speed=Math.min(capSpeed,ball.speed*1.045+baseSpeed*0.004);
  const dir=isRight?-1:1;
  let vx=Math.cos(ang)*ball.speed*dir,vy=Math.sin(ang)*ball.speed+p.vy*0.16;
  const m=Math.hypot(vx,vy)||1;let nx=vx/m,ny=vy/m;
  if(Math.abs(nx)<0.38){nx=(nx<0?-1:1)*0.38;ny=(ny<0?-1:1)*Math.sqrt(1-0.38*0.38);}
  ball.vx=nx*ball.speed;ball.vy=ny*ball.speed;
  ball.x=isRight?p.x-ball.r:p.x+p.vw+ball.r;
  G.shake=Math.min(6,2+G.rally*0.15);
  burst(ball.x,ball.y,isRight?C.coral:C.mint,10,W*0.16);
  SFX.hit(G.rally);
  enemy.err=(Math.random()-0.5)*enemy.vh*0.9*(G.rally>9&&Math.random()<0.35?2:1);
}
function moveBall(dt,attract){
  const px=ball.x,py=ball.y;
  ball.x+=ball.vx*dt;ball.y+=ball.vy*dt;
  G.trail.push({x:ball.x,y:ball.y});if(G.trail.length>15)G.trail.shift();
  if(ball.y<ball.r){ball.y=ball.r;ball.vy=Math.abs(ball.vy);G.wallFlash.t=1;burst(ball.x,4,'#dfeaff',6,W*0.08);SFX.wall();}
  else if(ball.y>H-ball.r){ball.y=H-ball.r;ball.vy=-Math.abs(ball.vy);G.wallFlash.b=1;burst(ball.x,H-4,'#dfeaff',6,W*0.08);SFX.wall();}
  // left paddle (player)
  if(ball.vx<0&&px-ball.r>player.x+player.vw&&ball.x-ball.r<=player.x+player.vw){
    const f=((px-ball.r)-(player.x+player.vw))/((px-ball.r)-(ball.x-ball.r));
    if(Math.abs(lerp(py,ball.y,f)-player.y)<=player.vh/2+ball.r*0.9)hitPaddle(player,false);
  }
  // right paddle (enemy)
  if(ball.vx>0&&px+ball.r<enemy.x&&ball.x+ball.r>=enemy.x){
    const f=(enemy.x-(px+ball.r))/((ball.x+ball.r)-(px+ball.r));
    if(Math.abs(lerp(py,ball.y,f)-enemy.y)<=enemy.vh/2+ball.r*0.9)hitPaddle(enemy,true);
  }
  if(ball.x<-ball.r*3)onPoint('e',attract);
  else if(ball.x>W+ball.r*3)onPoint('p',attract);
}

/* ============ update ============ */
function update(dt){
  G.time+=dt;
  G.shake*=Math.exp(-dt*7);G.boost*=Math.exp(-dt*2.1);
  G.popP*=Math.exp(-dt*5);G.popE*=Math.exp(-dt*5);G.rallyPop*=Math.exp(-dt*6);
  G.wallFlash.t*=Math.exp(-dt*4);G.wallFlash.b*=Math.exp(-dt*4);
  player.squash*=Math.exp(-dt*6);enemy.squash*=Math.exp(-dt*6);
  if(G.banner){G.banner.t+=dt;if(G.banner.t>G.banner.dur)G.banner=null;}

  const attract=G.state==='start';
  const sp=attract?W*0.16:(W*0.07+ball.speed*0.32);
  const sSpeed=sp*(1+G.boost*2.6);
  G.scroll+=sSpeed*dt;
  if(!attract)G.meters+=sSpeed*dt/55;

  // paddles
  if(G.state==='over'){player.y=lerp(player.y,H/2,Math.min(1,dt*3));enemy.y=lerp(enemy.y,H/2,Math.min(1,dt*3));player.vy=enemy.vy=0;}
  else if(attract){controlAI(player,dt,false);controlAI(enemy,dt,true);}
  else{humanControl(dt);controlAI(enemy,dt,true);}

  // ball phases
  if(G.state==='point'){G.pointT+=dt;if(G.pointT>=1.25){if(G.willEnd)endMatch();else beginServe();}}
  else if(G.state==='serve'){G.serveT+=dt;if(G.serveT>=0.75)launch(false);}
  else if(attract||G.state==='play'){
    if(ball.alive)moveBall(dt,attract);
    else{G.serveT+=dt;if(G.serveT>=0.8)launch(true);}
  }

  // speed streaks
  const rate=(!attract&&G.state==='play')?clamp((ball.speed/baseSpeed-1.05)*26,0,46):0;
  G.streakAcc+=dt*rate;
  while(G.streakAcc>1){G.streakAcc--;G.streaks.push({x:W+60,y:rand(H*0.08,H*0.9),len:rand(50,220),sp:sSpeed*(1.6+rand(0,1.8)),a:rand(0.04,0.13)});}
  for(let i=G.streaks.length-1;i>=0;i--){const s=G.streaks[i];s.x-=s.sp*dt;if(s.x+s.len<0)G.streaks.splice(i,1);}
  if(G.streaks.length>90)G.streaks.splice(0,G.streaks.length-90);

  // particles
  for(let i=G.particles.length-1;i>=0;i--){const p=G.particles[i];p.t+=dt;if(p.t>p.life){G.particles.splice(i,1);continue;}
    p.x+=p.vx*dt;p.y+=p.vy*dt;p.vx*=Math.exp(-dt*3);p.vy*=Math.exp(-dt*3);}
}

function ridge(x,s){return Math.sin(x*0.0013+s)*0.55+Math.sin(x*0.0034+s*1.7)*0.3+Math.sin(x*0.0083+s*2.9)*0.15;}
