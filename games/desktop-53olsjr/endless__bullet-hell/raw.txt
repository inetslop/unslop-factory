function step(dt){
  G.t+=dt; G.score+=dt*12;
  const sec=1+((G.t/30)|0);
  if(sec!==G.sector){G.sector=sec; toast('SECTOR '+sec); sfx.sector();}
  director(dt);
  updatePlayer(dt);
  for(let i=enemies.length-1;i>=0;i--){
    const e=enemies[i]; e.t+=dt; updEnemy(e,dt);
    if(e.dead||e.y>H+80)enemies.splice(i,1);
  }
  for(let i=pbullets.length-1;i>=0;i--){
    const b=pbullets[i]; b.x+=b.vx*dt; b.y+=b.vy*dt;
    if(b.y<-30||b.x<-30||b.x>W+30)pbullets.splice(i,1);
  }
  for(let i=ebullets.length-1;i>=0;i--){
    const b=ebullets[i]; b.t+=dt;
    if(b.wf)b.x+=(b.vx+Math.sin(b.t*b.wf+b.ph)*b.wa)*dt; else b.x+=b.vx*dt;
    b.y+=b.vy*dt;
    if(b.y>H+30||b.y<-70||b.x<-50||b.x>W+50)ebullets.splice(i,1);
  }
  // player shots vs enemies
  for(let i=pbullets.length-1;i>=0;i--){
    const b=pbullets[i]; let hit=false;
    for(const e of enemies){
      if(e.dead)continue;
      const dx=e.x-b.x, dy=e.y-b.y, rr=e.r+b.r;
      if(dx*dx+dy*dy<rr*rr){
        e.hp--; spark(b.x,b.y,e.c); pbullets.splice(i,1); hit=true;
        if(e.hp<=0)killEnemy(e);
        break;
      }
    }
    if(hit)continue;
  }
  // player vs hazards
  const p=player;
  if(!p.dead&&p.invuln<=0){
    const hr=p.focus?3.5:5;
    for(let i=ebullets.length-1;i>=0;i--){
      const b=ebullets[i], dx=b.x-p.x, dy=b.y-p.y, rr=b.r+hr;
      if(dx*dx+dy*dy<rr*rr){ ebullets.splice(i,1); onPlayerHit(); break; }
    }
    if(!p.dead&&p.invuln<=0)for(const e of enemies){
      const dx=e.x-p.x, dy=e.y-p.y, rr=e.r+hr-2;
      if(dx*dx+dy*dy<rr*rr){ onPlayerHit(); break; }
    }
  }
  // pickups
  for(let i=pickups.length-1;i>=0;i--){
    const k=pickups[i]; k.t+=dt; k.vy=Math.min(120,k.vy+40*dt);
    if(!p.dead){
      const dx=p.x-k.x, dy=p.y-k.y, d2=dx*dx+dy*dy;
      if(d2<130*130){const d=Math.sqrt(d2)||1; k.x+=dx/d*420*dt; k.y+=dy/d*420*dt;}
      if(d2<26*26){
        pickups.splice(i,1); sfx.power();
        boom(k.x,k.y,'#ffffff',8,120,2,.35);
        if(k.type==='power'){ if(p.power<5)p.power++; else G.score+=200; floater(k.x,k.y-16,p.power>1?'POWER '+p.power:'+200','#59e6ff'); }
        else if(k.type==='shield'){ p.shield=true; floater(k.x,k.y-16,'SHIELD ONLINE','#ffd23c'); }
        else{ p.bombs=Math.min(5,p.bombs+1); lastB=-1; floater(k.x,k.y-16,'BOMB +1','#ff4fd8'); }
      }
    }
    if(k.y>H+30)pickups.splice(i,1);
  }
  // bomb shockwaves
  for(let i=waves.length-1;i>=0;i--){
    const w=waves[i]; w.r+=1500*dt;
    for(let j=ebullets.length-1;j>=0;j--){
      const b=ebullets[j], dx=b.x-w.x, dy=b.y-w.y;
      if(dx*dx+dy*dy<w.r*w.r){ spark(b.x,b.y,'#ffd23c'); ebullets.splice(j,1); G.score+=10; }
    }
    if(w.r>w.maxR)waves.splice(i,1);
  }
  // particles & floaters
  for(let i=parts.length-1;i>=0;i--){
    const q=parts[i]; q.life-=dt;
    if(q.life<=0){parts.splice(i,1);continue;}
    q.x+=q.vx*dt; q.y+=q.vy*dt;
    q.vx*=Math.max(0,1-2.2*dt); q.vy*=Math.max(0,1-2.2*dt);
  }
  for(let i=floaters.length-1;i>=0;i--){ floaters[i].t+=dt; if(floaters[i].t>floaters[i].max)floaters.splice(i,1); }
  if(G.comboT>0){ G.comboT-=dt; if(G.comboT<=0){G.combo=0;G.mult=1;} }
  if(p.dead){ deathT+=dt; if(deathT>1.1)gameOver(); }
}
