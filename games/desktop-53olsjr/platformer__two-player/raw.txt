      burst(p.x+PW/2,p.y+PH*.5,7,['#bdf3ff','#e6fbff'],2.6,4,.4,.06);p.sy=1.3;p.sx=.76;}}
  // gravity + wall slide (NOVA)
  p.vy+=GRAV*dt;
  if(p.id===1&&!p.grounded&&p.vy>0){const toward=(p.wallR&&inp.r)||(p.wallL&&inp.l);
    if(toward){p.vy=Math.min(p.vy,3.2);if(Math.random()<dt*26)part({type:'dot',x:p.x+PW/2+(p.wallR?PW*.4:-PW*.4),y:p.y+PH*.6,vx:(p.wallR?-1:1)*rand(.5,1.4),vy:rand(-.4,.8),g:2,life:.35,max:.35,size:.05,col:'#bdf3ff'});}}
  p.vy=Math.min(p.vy,MAXFALL);
  if(!inp.j&&p.vy<-4.6)p.vy=-4.6;
  // integrate + resolve X
  p.x+=p.vx*dt;p.wallL=false;p.wallR=false;
  {const r0=Math.floor(p.y+.06),r1=Math.floor(p.y+PH-.06);
   if(p.vx>0){let c=Math.floor(p.x+PW-.001);for(let r=r0;r<=r1;r++)if(solidAt(c,r)){p.x=c-PW;p.vx=0;p.wallR=true;c=Math.floor(p.x+PW-.001);}}
   else if(p.vx<0){let c=Math.floor(p.x+.001);for(let r=r0;r<=r1;r++)if(solidAt(c,r)){p.x=c+1;p.vx=0;p.wallL=true;c=Math.floor(p.x+.001);}}}
  // integrate + resolve Y
  const prevB=p.y+PH,fs=p.vy;
  p.y+=p.vy*dt;p.grounded=false;p.rider=null;p.groundKind='';
  {const c0=Math.floor(p.x+.06),c1=Math.floor(p.x+PW-.06);
   if(p.vy>0){let r=Math.floor(p.y+PH-.001);
     for(let c=c0;c<=c1;c++){const ch=chAt(c,r);
       if(ch==='#'||(ch==='G'&&!cur.gateOpen)||(ch==='-'&&p.dropT<=0&&prevB<=r+.14)){p.y=r-PH;land(p,ch==='-'?'slab':'tile',fs);r=Math.floor(p.y+PH-.001);}}}
   else if(p.vy<0){const r=Math.floor(p.y+.001);
     for(let c=c0;c<=c1;c++){const ch=chAt(c,r);if(ch==='#'||(ch==='G'&&!cur.gateOpen)){p.y=r+1;p.vy=0;}}}}
  // movers
  for(const m of cur.movers){
    if(p.x+PW>m.x&&p.x<m.x+m.w&&p.y+PH>m.y&&p.y<m.y+m.h){
      const dl=p.x+PW-m.x,dr=m.x+m.w-p.x,dtp=p.y+PH-m.y,db=m.y+m.h-p.y;
      if(dtp<=dl&&dtp<=dr&&dtp<=db&&p.vy>=-.5){p.y=m.y-PH;land(p,'plat',fs);p.rider=m;}
      else if(dl<dr&&dl<dtp&&dl<db){p.x=p.vx>0?m.x-PW:m.x+m.w;p.vx=0;}
      else if(dr<dl&&dr<=dtp&&dr<=db){p.x=p.vx<0?m.x+m.w:m.x-PW;p.vx=0;}
      else{p.y=m.y+m.h;if(p.vy<0)p.vy=0;}}}
  // cosmetics
  p.sx+=(1-p.sx)*Math.min(1,dt*12);p.sy+=(1-p.sy)*Math.min(1,dt*12);
  if(p.grounded&&Math.abs(p.vx)>.3)p.walk+=Math.abs(p.vx)*dt*6.5;
  // trail
  if(Math.abs(p.vx)>4.4||(!p.grounded&&Math.abs(p.vy)>9))if(Math.random()<dt*30)
    part({type:'dot',x:p.x+PW/2,y:p.y+PH*.8,vx:0,vy:.5,g:0,life:.3,max:.3,size:.06,col:p.id===0?'rgba(255,154,92,.7)':'rgba(140,225,245,.7)'});
}

function feetIn(p,c,r){const fy=p.y+PH;return fy>=r-.05&&fy<=r+1.3&&p.x+PW>c+.05&&p.x<c+.95;}
function platesGate(dt){if(!cur.plates.length)return;let any=false,was=cur.prevPlate;
  for(const pl of cur.plates){const pr=players.some(p=>feetIn(p,pl.x,pl.y));any=any||pr;
    pl.press+=((pr?1:0)-pl.press)*Math.min(1,dt*10);}
  if(any){cur.gateOpen=true;cur.holdT=.75;if(!was)SFX.plate();}
  else{cur.holdT-=dt;if(cur.holdT<=0)cur.gateOpen=false;}
  cur.prevPlate=any;
  cur.gateLerp+=((cur.gateOpen?1:0)-cur.gateLerp)*Math.min(1,dt*5.5);
  if(cur.gateOpen&&!cur.prevGate){SFX.door();for(const g of cur.gates)burst(g[0]+.5,g[1]+.5,4,['#9db1d6','#ffd166'],2,4,.35,.05);}
  if(!cur.gateOpen&&cur.prevGate){SFX.gclose();shake=Math.max(shake,2.5);}
  cur.prevGate=cur.gateOpen;}
function exitCheck(dt){if(!cur.exit)return;const e=cur.exit;
  players[0].inPad=feetIn(players[0],e.x,e.y);players[1].inPad=feetIn(players[1],e.x,e.y);
  if(players[0].inPad&&players[1].inPad)cur.exitT+=dt;else cur.exitT=Math.max(0,cur.exitT-dt*3);
  if(cur.exitT>=.6)clearLevel();}
function hazards(){for(const p of players){
    if(p.y>ROWS+1.5){die();return;}
    for(const s of cur.spikes){const hx=s[0]+.25,hw=.5,hy=s[1]+.42,hh=.58;
      if(p.x+PW>hx&&p.x<hx+hw&&p.y+PH>hy&&p.y<hy+hh){die();return;}}}}
function die(){if(dying>0)return;dying=.95;falls++;score=Math.max(0,score-20);
  SFX.die();shake=13;
  players.forEach(p=>{burst(p.x+PW/2,p.y+PH/2,26,[PCOL[p.id].base,PCOL[p.id].light,'#ffffff'],5.5,9,.8,.09);});
  textPop((players[0].x+players[1].x)/2+PW/2,(players[0].y+players[1].y)/2,'-20','#ff8d7a');}
function collectGems(){for(const g of cur.gems){if(g.taken)continue;
  for(const p of players){const gx=g.x+.5,gy=g.y+.5;
    if(p.x+PW>gx-.42&&p.x<gx+.42&&p.y+PH>gy-.42&&p.y<gy+.42){g.taken=true;score+=150;runGems++;
      SFX.gem();burst(gx,gy,10,['#ffd166','#fff3c9'],3,3,.5,.07);textPop(gx,gy-.3,'+150','#ffd166');break;}}}}

const STEP=1/60;
function step(dt){t+=dt;
  if(state==='playing'){runTime+=dt;levelTime+=dt;
    for(const m of cur.movers){const ox=m.x;m.x+=m.dir*m.sp*dt;
      if(m.x<=m.x1){m.x=m.x1;m.dir=1;}else if(m.x>=m.x2){m.x=m.x2;m.dir=-1;}m.dx=m.x-ox;
      part({type:'dot',x:m.x+(m.dir>0?-.05:m.w+.05),y:m.y+.17,vx:-m.dir*1.2,vy:0,g:0,life:.3,max:.3,size:.05,col:'rgba(159,232,192,.55)'});}
    if(dying>0){dying-=dt;if(dying<=0)respawn();}
    else{for(const p of players)updatePlayer(p,dt);hazards();if(dying<=0){collectGems();platesGate(dt);exitCheck(dt);}}}
  else if(state==='clear'){platesGate(dt);clearTimer+=dt;if(clearTimer>2.6)nextLevelOrWin();}
  updateParts(dt);shake=Math.max(0,shake-dt*30);}
