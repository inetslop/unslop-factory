  const pct=c/(G.n*G.n);

function spawnSpark(r,c,st){ const g=G.geo,x=g.bx+c*g.step+g.cell/2,y=g.by+r*g.step+g.cell/2;
  const cc=[['#ff8a6a','#ffd98a'],['#ff7a5c','#ffc46b','#8feaff']][G.m===2?0:1];

function doFire(isRow,idx,demo){ if(!demo&&G.mode!=='PLAYING')return;
  if(!demo){ G.moves++; G.started=true; }
  let changed=0; const g=G.geo;
  for(let k=0;k<G.n;k++){ const r=isRow?idx:k,c=isRow?k:idx,o=G.board[r][c],nv=(o+1)%G.m;
    G.board[r][c]=nv; if(nv!==o){ changed++; G.flashes[r*G.n+c]=nowT; spawnSpark(r,c,nv); } }
  G.effects.push({type:isRow?'row':'col',idx,t:0});
  if(!demo){ sfxFire(isRow,changed); G.shake=Math.min(3+changed*.5,8); } else G.shake=2;
  updatePower();
  if(!demo&&isSolved())beginClear(); }

function beginClear(){ const moves=G.moves,par=G.par;

  starsArr[G.lvl]=Math.max(starsArr[G.lvl],stars); lsSet('sc.stars',JSON.stringify(starsArr));
  G.lastStats={time:G.time,moves,par,stars,newBest};

let last=performance.now(),confT=0;
function update(dt){ nowT+=dt; G.shake=Math.max(0,G.shake-dt*26);
  for(let i=G.effects.length-1;i>=0;i--){ G.effects[i].t+=dt; if(G.effects[i].t>.45)G.effects.splice(i,1); }
  const stepP=(p,d)=>{ p.x+=p.vx*d; p.y+=p.vy*d; p.vy+=(p.g||0)*d; p.vx*=.985; p.life-=d; };
  for(let i=G.parts.length-1;i>=0;i--){ stepP(G.parts[i],dt); if(G.parts[i].life<=0)G.parts.splice(i,1); }
  for(let i=G.conf.length-1;i>=0;i--){ stepP(G.conf[i],dt); if(G.conf[i].life<=0||G.conf[i].y>h+40)G.conf.splice(i,1); }
  for(const b of G.bgParts){ b.x+=b.vx*dt; b.y-=b.vy*dt;
    if(b.y<-10){b.y=h+10;b.x=Math.random()*w;} if(b.x<-10)b.x=w+10; if(b.x>w+10)b.x=-10; }
  G.dispPower+=(G.power-G.dispPower)*Math.min(1,dt*8);

  switch(G.mode){
    case 'PLAYING': if(G.started)G.time+=dt; break;
    case 'CLEARING': G.clearT+=dt;
      if(G.clearT>=G.clearDur){ if(G.lvl>=LEVELS.length-1)enterVictory(); else { G.mode='CLEAR'; G.panelT=0; } } break;
    case 'CLEAR': case 'VICTORY': G.panelT+=dt; break;
    case 'START':
      G.demoT-=dt;
      if(G.demoT<=0){ G.demoT=.4+Math.random()*.75; const r=Math.random();
        if(r<.3)G.rowCur=mod(G.rowCur+(Math.random()<.5?1:-1),G.n);
        else if(r<.6)G.colCur=mod(G.colCur+(Math.random()<.5?1:-1),G.n);
        else doFire(Math.random()<.5,(Math.random()*G.n)|0,true); }
      if(isSolved())setupLevel(0); break;
    case 'VICTORY': break;
  }
  if(G.mode==='VICTORY'){ confT-=dt; if(confT<=0){ confT=.7;
    confBurst(Math.random()<.5?w*.12:w*.88,h*.85,16,-(Math.random()<.5?.4:Math.PI+.4)); } } }

        case 'ArrowUp': G.colCur=mod(G.colCur-1,G.n); if(!e.repeat)sfxMove(1); break;
        case 'ArrowDown': G.colCur=mod(G.colCur+1,G.n); if(!e.repeat)sfxMove(1); break;
