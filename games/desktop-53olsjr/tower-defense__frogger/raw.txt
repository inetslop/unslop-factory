function sim(dt){
  G.t+=dt;
  if(G.waveState==='prep'){G.prepT-=dt;if(G.prepT<=0)beginWave()}
  else{G.waveT+=dt;while(G.spawnQ.length&&G.spawnQ[0].t<=G.waveT)spawnEnemy(G.spawnQ.shift().type)}
  for(const e of G.enemies)e.slow=1;
  for(const t of G.towers){if(t.kind!=='slime')continue;const s=tstat(t),R=s.range*L.cell,[tx,ty]=towerXY(t);t.pulse=(t.pulse||0)+dt;
    for(const e of G.enemies){if(e.gone)continue;const dx=eCx(e)-tx,dy=eCy(e)-ty;
      if(dx*dx+dy*dy<R*R){e.slow=Math.min(e.slow,1-s.slow);damage(e,s.dps*dt,false)}}}
  for(const e of G.enemies)if(!e.gone)updateEnemy(e,dt);
  G.enemies=G.enemies.filter(e=>!e.gone);
  for(const t of G.towers){t.phase=(t.phase||0)+dt;
    if(t.kind==='slime')continue;
    if(t.cd>0)t.cd-=dt;const tgt=pickTarget(t);
    if(tgt){t.aim=Math.atan2(eCy(tgt)-towerXY(t)[1],eCx(tgt)-towerXY(t)[0]);if(t.cd<=0)fireTower(t,tgt)}}
  // projectiles
  for(const p of G.projs){
    if(p.k===0){let tx,ty;
      if(p.tgt&&!p.tgt.gone){tx=eCx(p.tgt);ty=eCy(p.tgt)}else{p.life-=dt;if(p.life<=0){p.dead=true;continue}tx=p.x+p.dx*999;ty=p.y+p.dy*999}
      const dx=tx-p.x,dy=ty-p.y,d=Math.hypot(dx,dy)||1;p.dx=dx/d;p.dy=dy/d;
      p.x+=p.dx*p.spd*L.cell*dt;p.y+=p.dy*p.spd*L.cell*dt;
      if(p.tgt&&!p.tgt.gone&&d<L.cell*.3){damage(p.tgt,p.dmg);burst(p.x,p.y,'#eaffcf',4,60);p.dead=true}}
    else{p.t+=dt;const k=Math.min(1,p.t/p.T);p.px=p.sx+(p.tx-p.sx)*k;p.py=p.sy+(p.ty-p.sy)*k;
      if(k>=1){p.dead=true;sfx.boom();G.parts.push({k:'ring',x:p.tx,y:p.ty,t:0,life:.4,size:p.r});
        burst(p.tx,p.ty,'#8fd46a',10,90);
        for(const e of G.enemies){if(e.gone)continue;const dd=Math.hypot(eCx(e)-p.tx,eCy(e)-p.ty);
          if(dd<p.r)damage(e,p.dmg)}}}}
  G.projs=G.projs.filter(p=>!p.dead);
  for(const b of G.beams)b.t+=dt;G.beams=G.beams.filter(b=>b.t<b.life);
  updFx(dt);
  if(G.waveState==='active'&&!G.spawnQ.length&&!G.enemies.length)waveCleared();

function updFx(dt){
  for(const p of G.parts){p.t+=dt;
    if(p.k==='dot'){p.x+=p.vx*dt;p.y+=p.vy*dt;p.vx*=1-dt*3;p.vy*=1-dt*3}
    else if(p.k==='smoke'){p.x+=p.vx*dt;p.y+=p.vy*dt}}
  G.parts=G.parts.filter(p=>p.t<p.life);
  for(const p of G.pops)p.t+=dt;G.pops=G.pops.filter(p=>p.t<p.life);
  G.shake*=Math.max(0,1-dt*4.5);
}
