  const w={b,bricks:bricks2,pad:{x:paddle.x,y:paddle.y,w:paddle.w,h:paddle.h},B};
  const pts=[{x:b.x,y:b.y}];let first=null,total=0,cn=0,golds=0,bnc=0,steps=0;
  const dt=1/90,maxS=Math.ceil(MAX_FLIGHT*90);
  while(b.alive&&steps<maxS&&bnc<=MAX_BOUNCES){
    steps++;
    const evs=stepBall(w,dt);
    if(steps%3===0)pts.push({x:b.x,y:b.y});
    for(const ev of evs){
      if(typeof ev==='string'){if(ev==='wall'||ev==='paddle')bnc++;if(ev==='dead')b.alive=false;}
      else{bnc++;const br=bricks2[ev.idx];if(!first)first={x:b.x,y:b.y};
        br.hp--;if(br.hp<=0){br.alive=false;cn++;total+=brickPoints(br.type,br.row,cn,VMULT[velIdx]);if(br.type==='g')golds++;}}
    }
  }
  pts.push({x:b.x,y:b.y});
  preview={pts,first,total,golds,broken:cn};
}

function fire(){
  if(mode!=='aim')return;
  const lp=launchPoint(),sp=SPEEDS[velIdx];
  ball.x=lp.x;ball.y=lp.y;ball.vx=Math.sin(aimAngle)*sp;ball.vy=-Math.cos(aimAngle)*sp;ball.alive=true;
  shotsLeft--;totalTurns++;combo=0;flightTime=0;bounces=0;trail.length=0;holding=false;
  mode='flight';sLaunch();
  for(let i=0;i<8;i++)particles.push({x:lp.x,y:lp.y,vx:(Math.random()-.5)*160,vy:(Math.random()-.5)*160,
    rot:Math.random()*6.28,vr:(Math.random()-.5)*8,len:4,color:'#eaf6ff',t:0,life:.35,g:0});
  updateHUD();
}
function handleEvent(ev){
  if(typeof ev==='string'){
    if(ev==='wall'){bounces++;sWall();shake=Math.max(shake,1.4);}
    else if(ev==='paddle'){bounces++;sPaddle();shake=Math.max(shake,2);}
    else if(ev==='dead'){ball.alive=false;sDrop();}
    return;
  }
  bounces++;
  const br=bricks[ev.idx];br.hp--;
  if(br.hp<=0){
    br.alive=false;combo++;maxComboRun=Math.max(maxComboRun,combo);
    const pts=brickPoints(br.type,br.row,combo,VMULT[velIdx]);score+=pts;
    addFloat('+'+pts,br.x+br.w/2,br.y-4,combo>1?'×'+combo:null,br.type==='g'?'#ffd23f':'#eaf6ff');
    burst(br.x+br.w/2,br.y+br.h/2,brickFill(br),10+Math.min(combo*2,14));
    shake=Math.max(shake,br.type==='g'?7:3.5);
    if(combo>=2)hitstop=Math.max(hitstop,combo>4?.09:.05);
    sBrick(combo,br.type);
    if(br.type==='g'){shotsLeft=Math.min(shotsLeft+1,maxShots+4);addFloat('+1 SHOT',br.x+br.w/2,br.y-22,null,'#7dffb8');}
  }else{
    sClang();shake=Math.max(shake,2.5);
    burst(br.x+br.w/2,br.y+br.h/2,'#9db8d4',6);
    addFloat('CRACK',br.x+br.w/2,br.y-4,null,'#9fb6cc');
  }
  updateHUD();
}
function endFlight(){
  if(!bricks.some(b=>b.alive)){
    const bonus=shotsLeft*150;score+=bonus;
    mode='clear';clearT=2.8;stamp={t:performance.now(),bonus,shots:shotsLeft};
    sClear();shake=8;confetti();
  }else if(shotsLeft<=0){
    const wasNew=score>best&&score>0;saveBest();
    el('oScore').textContent=score.toLocaleString();
    el('oSheets').textContent=pad2(level-1);
    el('oCombo').textContent='×'+maxComboRun;
    el('oTurns').textContent=totalTurns;
    el('oBest').textContent=best.toLocaleString();
    el('oNew').classList.toggle('hidden',!wasNew);
    el('oSub').textContent='Sheet '+pad2(level)+' void — '+bricks.filter(b=>b.alive).length+' bricks still stand.';
    mode='over';showScreen('scr-over','solid');sOver();
  }else{mode='aim';}
  updateHUD();
}

/* ============================== UPDATE ============================== */
function updateFlight(dt){
  const w={b:ball,bricks,pad:paddle,B};
  const sp=Math.hypot(ball.vx,ball.vy);
  const n=Math.max(1,Math.ceil(sp*dt/(BALL_R*.7)));
  for(let i=0;i<n;i++){
    if(!ball.alive)break;
    for(const ev of stepBall(w,dt/n))handleEvent(ev);
  }
  flightTime+=dt;
  trail.push({x:ball.x,y:ball.y});if(trail.length>16)trail.shift();
  if(!ball.alive||flightTime>MAX_FLIGHT||bounces>MAX_BOUNCES){ball.alive=false;endFlight();}
}
function update(dt){
  shake=Math.max(0,shake-shake*6*dt-3*dt);
  for(let i=particles.length-1;i>=0;i--){const p=particles[i];p.t+=dt;
    if(p.t>p.life){particles.splice(i,1);continue;}
    p.vy+=p.g*dt;p.x+=p.vx*dt;p.y+=p.vy*dt;p.rot+=p.vr*dt;}
  for(let i=floats.length-1;i>=0;i--){const f=floats[i];f.t+=dt;if(f.t>f.life)floats.splice(i,1);}
  if(mode==='menu'){aimAngle=-.45+Math.sin(dashT*.6)*.4;computePreview();}
  else if(mode==='aim'){
    const mv=(keys.ArrowLeft||keys.a?-1:0)+(keys.ArrowRight||keys.d?1:0);
    if(mv)paddle.x=clamp(paddle.x+mv*520*dt,B.l+4,B.r-4-paddle.w);
    const sw=(keys.w||keys.ArrowUp?-1:0)+(keys.s||keys.ArrowDown?1:0);
    if(sw)aimAngle=clamp(aimAngle+sw*2.4*dt,-AIMMAX,AIMMAX);
    computePreview();
  }
  else if(mode==='flight')updateFlight(dt);
  else if(mode==='clear'){clearT-=dt;if(clearT<=0)nextLevel();}
  updateCalc();
}
