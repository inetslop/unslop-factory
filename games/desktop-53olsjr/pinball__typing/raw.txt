/* ============================== physics ============================== */
function collideSeg(s,e,kick){
  const b=ball, dx=s.x2-s.x1, dy=s.y2-s.y1, L2=dx*dx+dy*dy||1;
  let t=((b.x-s.x1)*dx+(b.y-s.y1)*dy)/L2; t=t<0?0:t>1?1:t;
  const px=s.x1+dx*t, py=s.y1+dy*t;
  let nx=b.x-px, ny=b.y-py; const d=Math.hypot(nx,ny);
  if(d>=b.r||d===0) return false;
  nx/=d; ny/=d; b.x=px+nx*b.r; b.y=py+ny*b.r;
  const vn=b.vx*nx+b.vy*ny;
  if(vn<0){ b.vx-=(1+e)*vn*nx; b.vy-=(1+e)*vn*ny; }
  if(kick){ b.vx+=nx*kick; b.vy+=ny*kick; }
  return true;
}
function hitCircle(c,r,e,minSpd){
  const b=ball; let nx=b.x-c.x, ny=b.y-c.y; const d=Math.hypot(nx,ny);
  if(d>=r||d===0) return false;
  nx/=d; ny/=d; b.x=c.x+nx*r; b.y=c.y+ny*r;
  const vn=b.vx*nx+b.vy*ny;
  if(vn<0){ b.vx-=(1+e)*vn*nx; b.vy-=(1+e)*vn*ny; }
  if(minSpd){ const sp=Math.hypot(b.vx,b.vy)||1; if(sp<minSpd){ b.vx*=minSpd/sp; b.vy*=minSpd/sp; } }
  return true;
}
function collideFlipper(f,dt){
  const b=ball, tx=f.px+FL.len*Math.cos(f.th), ty=f.py+FL.len*Math.sin(f.th);
  const dx=tx-f.px, dy=ty-f.py, L2=dx*dx+dy*dy||1;
  let t=((b.x-f.px)*dx+(b.y-f.py)*dy)/L2; t=t<0?0:t>1?1:t;
  const px=f.px+dx*t, py=f.py+dy*t;
  let nx=b.x-px, ny=b.y-py; const d=Math.hypot(nx,ny), R=b.r+FL.r;
  if(d>=R||d===0) return;
  nx/=d; ny/=d; b.x=px+nx*R; b.y=py+ny*R;
  const om=(f.th-f.prev)/dt;
  const rx=px-f.px, ry=py-f.py, pvx=-om*ry, pvy=om*rx;
  const vn=(b.vx-pvx)*nx+(b.vy-pvy)*ny;
  if(vn<0){ const e=0.35; b.vx-=(1+e)*vn*nx; b.vy-=(1+e)*vn*ny; }
}
function updateFlippers(dt){
  for(const f of flippers){
    f.prev=f.th;
    let target;
    if(f===flippers[0]) target=keys.left?FL.upL:FL.restL;
    else target=keys.right?Math.PI-FL.upL:Math.PI-FL.restL;
    const d=target-f.th, step=FL.speed*dt;
    f.th=Math.abs(d)<=step?target:f.th+Math.sign(d)*step;
  }
}
function hitPin(p){
  if(state!=='playing'||paused) return;
  if(p.state==='idle'&&!active){
    p.state='active';
    const tm=2.0+0.4*p.word.length;
    active={pin:p,word:p.word,idx:0,errors:0,timeMax:tm,timeLeft:tm};
    SFX.wordHit(); setPanelActive();
  }
}
function physStep(dt){
  const b=ball;
  b.vy+=1500*dt;
  b.x+=b.vx*dt; b.y+=b.vy*dt;
  for(const w of WALLS) collideSeg(w,w.e);
  for(const s of SLINGS){
    if(collideSeg(s,1.05,300)){
      if(s.flash<0.4&&state==='playing') SFX.sling();
      s.flash=1; shake=Math.max(shake,3);
      spawnBurst(b.x,b.y,'#ffb63d',5);
      if(state==='playing'){ score+=5; updateHUD(); }
    }
  }
  for(const bp of BUMPERS){
    if(hitCircle(bp,bp.r+b.r,0.55,470)){
      if(bp.flash<0.4&&state==='playing') SFX.bumper();
      bp.flash=1; shake=Math.max(shake,2.5);
      spawnBurst(b.x,b.y,'#ffb63d',7);
      if(state==='playing'){ score+=25; updateHUD(); }
    }
  }
  for(const dp of DOTS){
    if(hitCircle(dp,dp.r+b.r,0.8,0)){
      if(dp.flash<0.4&&state==='playing') SFX.dot();
      dp.flash=1;
      if(state==='playing'){ score+=10; updateHUD(); }
    }
  }
  for(const p of pins){
    if(hitCircle(p,p.r+b.r,0.92,160)){
      p.flash=Math.max(p.flash,0.7); hitPin(p);
      if(state==='playing'&&p.state!=='active'){ score+=15; updateHUD(); }
    }
  }
  for(const f of flippers) collideFlipper(f,dt);
  const sp=Math.hypot(b.vx,b.vy);
  if(sp>1500){ b.vx*=1500/sp; b.vy*=1500/sp; }
}

function update(rdt){
  frameDt=rdt;
  if(burstT>0) burstT-=rdt;
  shake*=Math.exp(-7*rdt); if(shake<0.1) shake=0;
  redFlash=Math.max(0,redFlash-rdt*1.6);
  nudgeCd-=rdt;

  const tgt=(state==='playing'&&active)?0.55:(burstT>0?1.45:1);
  ts+=(tgt-ts)*Math.min(1,rdt*12);

  for(const p of pins){
    p.flash=Math.max(0,p.flash-rdt*2.6);
    p.pulse+=rdt*(p.state==='active'?9:3);
    if(p.state==='cooldown'){ p.cd-=rdt*ts; if(p.cd<=0){ p.state='idle'; p.word=nextWord(); } }
  }
  for(const b of BUMPERS) b.flash=Math.max(0,b.flash-rdt*4);
  for(const d of DOTS) d.flash=Math.max(0,d.flash-rdt*6);
  for(const s of SLINGS) s.flash=Math.max(0,s.flash-rdt*3.5);
  for(const g of ghosts){ g.y-=g.v*rdt; if(g.y<140){ g.y=660; g.x=rand(50,430); } }

  if(state==='playing'&&!paused){
    playTime+=rdt;
    const total=rdt*ts;
    const SUBS=Math.min(16,Math.max(4,Math.ceil(total*1350/7)));
    const sdt=total/SUBS;
    for(let i=0;i<SUBS;i++){
      updateFlippers(sdt);
      if(ball.state==='live') physStep(sdt);
    }
    if(active){ active.timeLeft-=rdt*ts; if(active.timeLeft<=0) resolveWord(false); }

    if(ball.state==='ready'){
      if(charging) ball.power=clamp(ball.power+rdt*1.25,0.12,1);
    }else if(ball.state==='live'&&ball.x>472&&ball.y>800&&Math.hypot(ball.vx,ball.vy)<60){
      loadBall(); // weak launch rolled back — reload plunger
    }

    if(ball.state==='live'&&ball.y>H+50){
      ball.state='drained'; drainT=0.9; lives--; redFlash=0.7; shake=13; SFX.drain(); updateHUD();
      if(active){ active=null; combo=0; mult=1; updateMult(); panelLock=0; setPanelIdle(); }
      for(const p of pins) if(p.state==='active'){ p.state='cooldown'; p.cd=1; }
    }
    if(ball.state==='drained'){
      drainT-=rdt; ball.y+=340*rdt; ball.vy=0;
      if(drainT<=0){ if(lives>0) loadBall(); else gameOver(); }
    }
    if(ball.state==='live'){
      const t0=trail[0];
      if(!t0||Math.hypot(ball.x-t0.x,ball.y-t0.y)>2.5){ trail.unshift({x:ball.x,y:ball.y}); if(trail.length>14) trail.pop(); }
      const sp=Math.hypot(ball.vx,ball.vy);
      if(sp<30) stallT+=rdt; else stallT=0;
      if(stallT>2.4){ ball.vy-=340; ball.vx+=rand(-80,80); stallT=0; SFX.nudge(); }
    }
    wpmTimer+=rdt; if(wpmTimer>=0.5){ wpmTimer=0; updateWPM(); }
    if(active){
      const f=clamp(active.timeLeft/active.timeMax,0,1);
      const tf=$('timerFill');
      tf.style.width=(f*100)+'%';
      tf.style.background=f>0.5?'#3fd9ff':f>0.25?'#ffb63d':'#ff5c49';
    }
    if(panelLock>0){ panelLock-=rdt; if(panelLock<=0) setPanelIdle(); }
  }
  else if(state==='start'&&!paused){
    const sdt=rdt/4;
    for(let i=0;i<4;i++){ updateFlippers(sdt); if(ball.state==='demo') physStep(sdt); }
    if(ball.state!=='demo'){ ball.state='demo'; trail.length=0; }
    if(ball.y>H+60||ball.x<30||ball.x>500){
      ball.x=rand(80,400); ball.y=110; ball.vx=rand(-250,250); ball.vy=rand(100,260);
    }
    const t0=trail[0];
    if(!t0||Math.hypot(ball.x-t0.x,ball.y-t0.y)>2.5){ trail.unshift({x:ball.x,y:ball.y}); if(trail.length>14) trail.pop(); }
  }

  for(let i=particles.length-1;i>=0;i--){
    const p=particles[i]; p.life-=rdt;
    if(p.life<=0){ particles.splice(i,1); continue; }
    p.x+=p.vx*rdt; p.y+=p.vy*rdt; p.vy+=p.grav*rdt;
  }
  for(let i=floaters.length-1;i>=0;i--){
    const f=floaters[i]; f.life-=rdt*0.85; f.y-=38*rdt;
    if(f.life<=0) floaters.splice(i,1);
  }
}
