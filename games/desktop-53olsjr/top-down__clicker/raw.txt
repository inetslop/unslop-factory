    else if(!isSweep&&!g.suck&&d<=R*1.9+g.r){ g.suck={x0:g.x,y0:g.y,tx:px,ty:py,t:0}; }
  }

    S.combo=Math.min(40,S.combo+1); S.comboT=tNow; S.bestCombo=Math.max(S.bestCombo,S.combo);

}

function update(dt){
  tNow=performance.now();
  // ambient always
  for(const c of clouds){ c.x+=c.vx*dt; c.y+=c.vy*dt;
    if(c.x<-c.r)c.x=W+c.r; if(c.x>W+c.r)c.x=-c.r; if(c.y<-c.r)c.y=H+c.r; if(c.y>H+c.r)c.y=-c.r; }
  for(const s of specks){ s.x+=s.vx*dt; s.y+=s.vy*dt;
    if(s.y<-4){s.y=H+4;s.x=rnd(W);} if(s.x<-4)s.x=W+4; if(s.x>W+4)s.x=-4; }

  if(S.mode==='play'){
    S.playTime+=dt;
    updateSpawning(dt);
    if(tNow-S.comboT>1000&&S.combo>0){S.combo=0;hudDirty=true;}
    if(pressing&&mouse.in&&tNow-lastSweep>130){ lastSweep=tNow; harvest(mouse.x,mouse.y,true); }
    // drones
    for(const d of drones){
      let best=null,bd=1e9;
      for(const g of gems){ if(g.dead||g.state==='out')continue; const dd=Math.hypot(g.x-d.x,g.y-d.y); if(dd<bd){bd=dd;best=g;} }
      if(best){ const a=Math.atan2(best.y-d.y,best.x-d.x); d.x+=Math.cos(a)*120*dt; d.y+=Math.sin(a)*120*dt;
        if(bd<best.r+9)collectGem(best,'drone');
      }else{ d.wa-=dt; if(d.wa<0){d.wa=rnd(1,3);d.seed=rnd(TAU);} d.x+=Math.cos(d.seed)*26*dt; d.y+=Math.sin(d.seed)*26*dt; }
      d.x=clamp(d.x,14,W-14); d.y=clamp(d.y,90,H-14);
    }
  }
  // gem aging (not in menu — decorative)
  if(S.mode!=='menu'){
    for(const g of gems){
      if(g.dead)continue;
      g.t+=dt*1000;
      if(g.state==='in'&&g.t>250)g.state='idle';
      if(!g.suck&&g.t>g.life-300)g.state='out';
      if(g.suck){ g.suck.t+=dt/0.18; const t=clamp(g.suck.t,0,1),e=t*t*(3-2*t);
        g.x=g.suck.x0+(g.suck.tx-g.suck.x0)*e; g.y=g.suck.y0+(g.suck.ty-g.suck.y0)*e;
        if(t>=1){collectGem(g,'player');} }
      if(!g.suck&&g.state==='out'&&g.t>g.life)g.dead=true;
    }
    gems=gems.filter(g=>!g.dead);
  }
  // particles / ripples / texts
  for(const p of parts){ p.life-=dt; p.vy+=(p.grav||0)*dt; p.x+=p.vx*dt; p.y+=p.vy*dt; p.rot+=(p.vr||0)*dt; }
  parts=parts.filter(p=>p.life>0); if(parts.length>420)parts.splice(0,parts.length-420);
  for(const r of rips){ r.r+=(r.max-r.r)*Math.min(1,dt*9)+40*dt; r.a-=dt*1.6; }
  rips=rips.filter(r=>r.a>0&&r.r<r.max); if(rips.length>20)rips.splice(0,rips.length-20);
  for(const t of texts){ t.life-=dt; t.y+=t.vy*dt; t.vy*=Math.pow(.4,dt); }
  texts=texts.filter(t=>t.life>0); if(texts.length>26)texts.splice(0,texts.length-26);

  shake*=Math.pow(.03,dt); if(shake<.15)shake=0;
  if(armedRestart&&tNow-armedRestart>2500){armedRestart=0;$('btnRestart').classList.remove('armed');}

  saveT+=dt*1000;
  if(saveT>6000&&(S.mode==='play'||S.mode==='won')){saveT=0;save();}

  shopClock+=dt*1000;
  if(shopDirty||shopClock>300){ shopClock=0; if(shopDirty){refreshShop();shopDirty=false;} }
  if(hudDirty){refreshHUD();hudDirty=false;}
}

  if(g.dead)return;
  const i=g.tier,T=TIERS[i];
  let s=easeBack(Math.min(1,g.t/250)), al=1;
  if(g.state==='out'){const o=(g.t-(g.life-300))/300;s*=Math.max(0,1-o);al=Math.max(0,1-o);}
  const bob=Math.sin(tNow*0.002+g.seed)*2;
  ctx.save(); ctx.translate(g.x,g.y+bob); ctx.globalAlpha=al;
  // ground shadow
  ctx.fillStyle='rgba(0,0,0,.28)'; ctx.beginPath(); ctx.ellipse(0,g.r*.95*s,g.r*.9*s,g.r*.36*s,0,0,TAU); ctx.fill();
  const R=g.r*s; if(R<1){ctx.restore();return;}
  let rot=Math.sin(tNow*0.001+g.seed)*.25;
  if(T.sh==='tri'||T.sh==='star')rot=tNow*0.0012+g.seed;
  ctx.rotate(rot);
  // body path
  ctx.beginPath();
  if(T.sh==='circle')ctx.arc(0,0,R,0,TAU);
  else if(T.sh==='diamond'){ctx.moveTo(0,-R);ctx.lineTo(R*.74,0);ctx.lineTo(0,R);ctx.lineTo(-R*.74,0);ctx.closePath();}
  else if(T.sh==='hex')polyPath(6,R,Math.PI/6);
  else if(T.sh==='oct')polyPath(8,R,Math.PI/8);
  else if(T.sh==='tri')polyPath(3,R,-Math.PI/2);
  else if(T.sh==='star')starPath(R*1.15);
  else if(T.sh==='prism'){ctx.moveTo(0,-R);ctx.lineTo(R*.74,0);ctx.lineTo(0,R);ctx.lineTo(-R*.74,0);ctx.closePath();}
  let fill=T.c;
  if(T.sh==='prism'){
    const h=(tNow*0.12+g.seed*57)%360;
    const gr=ctx.createLinearGradient(-R,-R,R,R);
    gr.addColorStop(0,'hsl('+h+',95%,66%)');gr.addColorStop(.5,'hsl('+((h+120)%360)+',95%,62%)');gr.addColorStop(1,'hsl('+((h+240)%360)+',95%,68%)');
    fill=gr; ctx.shadowColor='hsla('+h+',95%,60%,.9)'; ctx.shadowBlur=16;
  }
  ctx.fillStyle=fill; ctx.fill();
  ctx.shadowBlur=0;
  ctx.lineWidth=Math.max(2,R*.16); ctx.strokeStyle=T.d; ctx.stroke();
  // highlight
  ctx.globalAlpha=al*.55; ctx.fillStyle='#fff';
  ctx.beginPath(); ctx.ellipse(-R*.3,-R*.34,R*.2,R*.13,-.7,0,TAU); ctx.fill();
  ctx.restore();
  // value label for rarer gems
  if(i>=2&&!g.suck){
    const lbl=fmt(T.v*gildMult());
    ctx.font='bold 11px "Trebuchet MS",sans-serif'; ctx.textAlign='center';
    ctx.lineWidth=3; ctx.strokeStyle='rgba(0,0,0,.5)'; ctx.strokeText(lbl,g.x,g.y+bob+g.r+14);
    ctx.fillStyle='#f6ead8'; ctx.fillText(lbl,g.x,g.y+bob+g.r+14);
  }
}

function drawDrone(d){
  const bob=Math.sin(tNow*0.004+d.seed)*3;
  ctx.save(); ctx.translate(d.x,d.y+bob);
  ctx.fillStyle='rgba(0,0,0,.25)'; ctx.beginPath(); ctx.ellipse(0,10,7,3,0,0,TAU); ctx.fill();
  // rotor
  ctx.save(); ctx.rotate(tNow*0.04+d.seed);
  ctx.strokeStyle='rgba(230,238,250,.75)'; ctx.lineWidth=2; ctx.lineCap='round';
  ctx.beginPath(); ctx.moveTo(-11,0); ctx.lineTo(11,0); ctx.moveTo(0,-11); ctx.lineTo(0,11); ctx.stroke();
  ctx.restore();
  // body
  ctx.fillStyle='#2c313c'; ctx.strokeStyle='#14161c'; ctx.lineWidth=2;
  ctx.beginPath(); ctx.arc(0,0,7,0,TAU); ctx.fill(); ctx.stroke();
  ctx.fillStyle=(Math.sin(tNow*0.008+d.seed)>0)?'#ffb84d':'#7a5a26';
  dot(ctx,0,0,2.4);
  ctx.restore();
}

function drawCursor(){
  if(!mouse.in||S.mode!=='play')return;
  const R=magnetR();
  ctx.save(); ctx.translate(mouse.x,mouse.y);
  ctx.setLineDash([6,8]); ctx.lineDashOffset=-tNow*0.02*(pressing?2:1);
  ctx.strokeStyle=pressing?'rgba(255,210,90,.9)':'rgba(255,210,90,.45)'; ctx.lineWidth=1.6;
  ctx.beginPath(); ctx.arc(0,0,R,0,TAU); ctx.stroke();
  if(pressing){ctx.setLineDash([]);ctx.fillStyle='rgba(255,190,80,.07)';dot(ctx,0,0,R);}
  ctx.setLineDash([]);
  const s=pressing?.88:1;
  ctx.scale(s,s); ctx.rotate(-0.9+(pressing?-0.25:0)); ctx.lineCap='round';
  ctx.strokeStyle='#7a4a22'; ctx.lineWidth=6; ctx.beginPath(); ctx.moveTo(-16,10); ctx.lineTo(8,-10); ctx.stroke();
  ctx.strokeStyle='#a86c34'; ctx.lineWidth=3; ctx.beginPath(); ctx.moveTo(-15,9); ctx.lineTo(7,-9); ctx.stroke();
  ctx.strokeStyle='#8b93a1'; ctx.lineWidth=7.5; ctx.beginPath(); ctx.arc(8,-10,16,-0.5,2.0); ctx.stroke();
  ctx.strokeStyle='#e2e8f2'; ctx.lineWidth=4.5; ctx.beginPath(); ctx.arc(8,-10,16,-0.5,2.0); ctx.stroke();
  ctx.fillStyle='#fff'; dot(ctx,8+Math.cos(2)*16,-10+Math.sin(2)*16,2.4);
  ctx.restore();
}

function render(){
  ctx.setTransform(dpr,0,0,dpr,0,0);
  const sx=shake?(Math.random()-.5)*shake:0, sy=shake?(Math.random()-.5)*shake:0;
  ctx.translate(sx,sy);
  if(bg)ctx.drawImage(bg,0,0,W,H);
  // cloud shadows
  for(const c of clouds){
    const gr=ctx.createRadialGradient(c.x,c.y,c.r*.15,c.x,c.y,c.r);
    gr.addColorStop(0,'rgba(0,0,0,'+c.a+')'); gr.addColorStop(1,'rgba(0,0,0,0)');
    ctx.fillStyle=gr; dot(ctx,c.x,c.y,c.r);
  }
  // specks
  for(const s of specks){
    ctx.globalAlpha=.22+.3*(.5+.5*Math.sin(tNow*.002+s.tw));
    ctx.fillStyle=s.c; dot(ctx,s.x,s.y,s.s);
  }
  ctx.globalAlpha=1;
  // ripples
  for(const r of rips){
    ctx.globalAlpha=Math.max(0,r.a); ctx.strokeStyle=r.c; ctx.lineWidth=2.5;
    ctx.beginPath(); ctx.arc(r.x,r.y,r.r,0,TAU); ctx.stroke();
  }
  ctx.globalAlpha=1;
  // gems (depth sorted)
  const sg=gems.slice().sort((a,b)=>a.y-b.y);
  for(const g of sg)drawGem(g);
  for(const d of drones)drawDrone(d);
  // particles
  for(const p of parts){
    ctx.globalAlpha=clamp(p.life/p.max,0,1); ctx.fillStyle=p.c;
    if(p.s>3){ ctx.save(); ctx.translate(p.x,p.y); ctx.rotate(p.rot); ctx.beginPath(); ctx.moveTo(0,-p.s); ctx.lineTo(p.s*.8,p.s*.7); ctx.lineTo(-p.s*.8,p.s*.7); ctx.closePath(); ctx.fill(); ctx.restore(); }
    else dot(ctx,p.x,p.y,p.s);
  }
  ctx.globalAlpha=1;
  // floating text
  for(const t of texts){
    ctx.globalAlpha=clamp(t.life/t.max,0,1);
    ctx.font='bold '+t.sz+'px Impact,"Arial Black",sans-serif'; ctx.textAlign='center';
    ctx.lineWidth=4; ctx.strokeStyle='rgba(0,0,0,.6)'; ctx.strokeText(t.txt,t.x,t.y);
    ctx.fillStyle=t.c; ctx.fillText(t.txt,t.x,t.y);
  }
  ctx.globalAlpha=1;
  drawCursor();
}
