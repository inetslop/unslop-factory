function update(dt,active){
  t+=dt;
  par.x+=(par.tx-par.x)*Math.min(1,dt*4);par.y+=(par.ty-par.y)*Math.min(1,dt*4);
  if(hitstop>0)hitstop-=dt;
  const d=dt*(hitstop>0?.12:1);
  gridOff+=speed*d*(active?1:.5);
  for(const s of stars){s.z-=speed*1.05*d;if(s.z<34)stars[stars.indexOf(s)]=mkStar(1700);}
  if(active){
    runT+=dt;spawnT-=d;
    if(spawnT<=0&&words.length<maxWords()){spawn(false);spawnT=spawnInt*(.7+Math.random()*.6)/1000;}
  }else if(state==='start'||state==='over'){
    demoT-=dt;let ghosts=0;for(const w of words)if(w.demo)ghosts++;
    if(demoT<=0&&ghosts<4){spawn(true);demoT=1.5;}
  }
  for(let i=words.length-1;i>=0;i--){const w=words[i];
    w.z-=speed*d*(w.demo?.6:1);
    w.pop=Math.max(0,w.pop-d*4.5);
    if(!w.demo){
      if(!w.warned&&w.z<DANGER_Z){w.warned=true;sPing();}
      if(w.z<=HIT_Z){impact(i);continue;}
      if(w.z<640){w.trailT-=d;if(w.trailT<=0){w.trailT=.05;
        const p=proj(w.nx,w.ny,w.z);
        parts.push({x:p.x,y:p.y,vx:(cx-p.x)*.28,vy:(cy-p.y)*.28,life:.34,age:0,size:Math.max(1,fontPx(w)*.05),c:'#8fe8ff',dim:true});}}
    }else if(w.z<=150){words.splice(i,1);continue;}
  }
  for(let i=parts.length-1;i>=0;i--){const p=parts[i];p.age+=d;
    const dr=Math.exp(-2.2*d);p.vx*=dr;p.vy*=dr;p.x+=p.vx*d;p.y+=p.vy*d;
    if(p.age>=p.life)parts.splice(i,1);}
  for(let i=pops.length-1;i>=0;i--){const p=pops[i];p.life-=d*.9;p.y-=42*d;if(p.life<=0)pops.splice(i,1);}
  shake=Math.max(0,shake-shake*6*dt-2*dt);
  wpmT+=dt;
  if(wpmT>=.5){wpmT=0;
    if(runT>3){wpm=Math.round((correct/5)/(runT/60));topWpm=Math.max(topWpm,wpm);}
    elWpm.textContent=wpm;}
}

function drawWord(w){
  const fpx=fontPx(w);
  const p=proj(w.nx,w.ny,w.z);
  if(fpx<7){ // distant glimmer before it resolves into text
    ctx.globalCompositeOperation='lighter';ctx.globalAlpha=.5;ctx.fillStyle='#bfffe e'.replace(' ','');
    ctx.beginPath();ctx.arc(p.x,p.y,1.8,0,TAU);ctx.fill();
    ctx.globalAlpha=1;ctx.globalCompositeOperation='source-over';return;}
  const danger=w.demo?0:clamp((DANGER_Z-w.z)/(DANGER_Z-HIT_Z),0,1);
  let a=clamp((1650-w.z)/260,0,1);
  if(w.demo)a*=clamp((w.z-150)/410,0,1);
  const rot=Math.sin(t*1.4+w.phase)*.05*Math.min(1,(w.z-200)/900);
  ctx.save();ctx.translate(p.x,p.y);ctx.rotate(rot);
  const sc=1+w.pop*.09;ctx.scale(sc,sc);
  ctx.font='700 '+fpx.toFixed(1)+'px '+WFONT;
  ctx.textBaseline='middle';ctx.textAlign='left';
  const cs=w.text.split(''),ws=cs.map(c=>ctx.measureText(c).width);
  let tot=0;for(const x of ws)tot+=x;let x=-tot/2;
  for(let i=0;i<cs.length;i++){
    if(i<w.idx){ctx.fillStyle='#5ff2d6';ctx.shadowColor='rgba(95,242,214,.9)';ctx.shadowBlur=fpx*.24;}
    else{
      const r=(234+(255-234)*danger)|0,g=(246+(93-246)*danger)|0,b=(255+(85-255)*danger)|0;
      ctx.fillStyle='rgb('+r+','+g+','+b+')';
      if(danger>0){ctx.shadowColor='rgba(255,93,85,'+(.7*danger)+')';ctx.shadowBlur=fpx*.22*danger;}
      else{ctx.shadowColor='rgba(180,230,255,.45)';ctx.shadowBlur=fpx*.1;}
    }
    ctx.globalAlpha=a*(i>=w.idx&&danger>0?(.8+.2*Math.sin(t*12)):1);
    ctx.fillText(cs[i],x,0);x+=ws[i];
  }
  ctx.restore();ctx.globalAlpha=1;
}
function drawGrid(){
  ctx.lineWidth=1;
  for(let i=-9;i<=9;i++){
    const a=proj(i*GSX,GY,80),b=proj(i*GSX,GY,1520);
    if(a[1]<cy-2||a[1]>H+80)continue;
    ctx.strokeStyle='rgba(64,215,195,.07)';ctx.beginPath();ctx.moveTo(a[0],a[1]);ctx.lineTo(b[0],b[1]);ctx.stroke();
  }
  const GAP=95,SPAN=1450;
  for(let i=0;i<16;i++){
    const zr=1520-((i*GAP+gridOff)%SPAN);
    const a=proj(-focal*4,GY,zr),b=proj(focal*4,GY,zr);
    if(a[1]>H+80)continue;
    ctx.strokeStyle='rgba(64,215,195,'+(.03+.2*(1-zr/1520)).toFixed(3)+')';
    ctx.beginPath();ctx.moveTo(a[0],a[1]);ctx.lineTo(b[0],b[1]);ctx.stroke();
  }
}
function render(){
  ctx.clearRect(0,0,W,H);
  ctx.save();
  if(shake>.1)ctx.translate((Math.random()-.5)*shake,(Math.random()-.5)*shake);
  ctx.drawImage(bgCv,-W*.075+par.x*.35,-H*.075+par.y*.35,bgCv.width,bgCv.height);
  // gate glow at the horizon point
  const gx=cx+par.x,gy=cy+par.y,m=Math.min(W,H);
  ctx.globalCompositeOperation='lighter';
  let g=ctx.createRadialGradient(gx,gy,0,gx,gy,m*.3);
  g.addColorStop(0,'rgba(190,255,240,'+(.3+.08*Math.sin(t*2)).toFixed(3)+')');
  g.addColorStop(.25,'rgba(63,232,200,.15)');g.addColorStop(.6,'rgba(255,170,90,.045)');g.addColorStop(1,'rgba(0,0,0,0)');
  ctx.fillStyle=g;ctx.fillRect(gx-m*.3,gy-m*.3,m*.6,m*.6);
  ctx.globalAlpha=.9;ctx.fillStyle='#eafffb';ctx.beginPath();ctx.arc(gx,gy,3+m*.004,0,TAU);ctx.fill();ctx.globalAlpha=1;
  // stars streaming out of the gate
  for(const s of stars){
    const z2=s.z*1.06+7;
    const x1=cx+par.x+s.nx*focal/s.z,y1=cy+par.y+s.ny*focal/s.z;
    if(x1<-40||x1>W+40||y1<-40||y1>H+40)continue;
    const x2=cx+par.x+s.nx*focal/z2,y2=cy+par.y+s.ny*focal/z2;
    let a=(.55+.45*Math.sin(t*2.1+s.tw))*clamp((1700-s.z)/900,.25,1);
    if(s.z<80)a*=s.z/80;
    ctx.globalAlpha=a;ctx.strokeStyle=s.c;ctx.lineWidth=Math.min(3.2,s.sz*(700/s.z));
    ctx.beginPath();ctx.moveTo(x1,y1);ctx.lineTo(x2,y2);ctx.stroke();
  }
  ctx.globalAlpha=1;ctx.globalCompositeOperation='source-over';
  drawGrid();
  const ws=[...words].sort((a,b)=>b.z-a.z);
  for(const w of ws)drawWord(w);
  // particles
  ctx.globalCompositeOperation='lighter';
  for(const p of parts){
    const a=1-p.age/p.life;ctx.globalAlpha=a*(p.dim?.45:.9);
    ctx.fillStyle=p.c;ctx.beginPath();ctx.arc(p.x,p.y,Math.max(.4,p.size*(.4+.6*a)),0,TAU);ctx.fill();
  }
  ctx.globalAlpha=1;ctx.globalCompositeOperation='source-over';
  // score popups
  for(const p of pops){
    ctx.globalAlpha=clamp(p.life,0,1);ctx.fillStyle=p.c;
    ctx.font='700 '+p.size+'px '+MONO;ctx.textAlign='center';ctx.textBaseline='middle';
    ctx.fillText(p.text,p.x,p.y);
  }
  ctx.globalAlpha=1;
  ctx.restore();
}
