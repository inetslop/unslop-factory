function updatePlay(h){
  // mode schedule
  modeT-=h;
  if(modeT<=0&&sched[0].m!=='chase'||(modeT<=0&&sched[(sched.length-1)].t<1e9)){
    const idx=sched.indexOf(sched.find(s=>s.m===mode&&s.t===Math.max(modeT,0)));
  }
  if(modeT<=0){
    let i=sched.findIndex(s=>s.m===mode);
    // find current index robustly: track by pointer instead
  }
  // (handled via modeIdx below)
  modeT2();
  function modeT2(){}
  frightT>0&&(frightT-=h);
  if(frightT<=0&&frightT>-99){ for(const g of ghosts)if(g.state!=='eyes')g.fright=false; }
  for(let i=1;i<4;i++)if(relT[i]>0)relT[i]-=h;

  // pac
  if(pac.dir>=0&&pac.wantDir===OPP[pac.dir])pac.dir=pac.wantDir;
  const pcol=Math.floor(((pac.x%COLS)+COLS)%COLS);
  pac.speed=(7.2+Math.min(1.1,(level-1)*0.14))*(pcol===0||pcol===COLS-1?0.7:1);
  stepMover(pac,h,pacDecide);
  if(pac.dir>=0)pac.face=pac.dir;
  pac.phase+=h*(pac.dir<0?6:13);

  // pickups
  const pcx=((Math.floor(pac.x))%COLS+COLS)%COLS, pcy=Math.floor(pac.y);
  if(pcy>=0&&pcy<ROWS){
    const cell=grid[pcy][pcx];
    if(cell===DOT){ grid[pcy][pcx]=EMPTY; dotsLeft--; dotsEaten++; addScore(10); SFX.dot(wakaFlip=!wakaFlip); }
    else if(cell===PELLET){
      grid[pcy][pcx]=EMPTY; dotsLeft--; dotsEaten++; addScore(50);
      frightT=frightDur(); frightChain=0; SFX.pellet(); ring(pac.x*t,pac.y*t,'#ffe600');
      for(const g of ghosts){
        if(g.state==='eyes')continue;
        g.fright=true;
        if(g.state==='normal')g.dir=OPP[g.dir];
      }
    }
    if(fruitActive&&pcx===10&&pcy===13){
      fruitActive=false; addScore(FRUIT_VAL[(level-1)%5]);
      addPopup(10.5*t,13*t,String(FRUIT_VAL[(level-1)%5]),'#ff9f1a');
      burst(10.5*t,13.5*t,14,'#ff9f1a',5); SFX.fruit();
    }
  }
  if(!fruitActive&&dotsEaten>=nextFruitAt){ fruitActive=true; fruitT=9; nextFruitAt+=100; }
  if(fruitActive){ fruitT-=h; if(fruitT<=0)fruitActive=false; }

  // ghosts
  for(const g of ghosts){
    const gc=Math.floor(((g.x%COLS)+COLS)%COLS);
    if(g.state==='house'){ updateHouse(g,h); continue; }
    if(g.state==='exiting'){ updateExiting(g,h); continue; }
    let sp=6.5+Math.min(1.9,(level-1)*0.28);
    if(g.state==='eyes')sp=13; else if(g.fright)sp=5;
    if(g.state!=='eyes'&&(gc===0||gc===COLS-1))sp*=0.55;
    g.speed=sp;
    const eyes=g.state==='eyes'?{isEyes:true}:null;
    stepMover(g,h,()=>ghostDecide(g));
    if(g.state==='eyes'&&(g.path.length===0||Math.hypot(g.x-10.5,g.y-11.5)<0.45))g.state='exiting';
  }

  // collisions
  for(const g of ghosts){
    if(g.state==='house'||g.state==='eyes')continue;
    if(Math.hypot(pac.x-g.x,pac.y-g.y)<0.55){
      if(g.fright)eatGhost(g); else { die(); return; }
    }
  }

  updateFx(h);
  if(dotsLeft===0){ state='clear'; clearT=0; setSiren('off'); SFX.clear(); }
}

  modeT-=h;
  if(modeT<=0&&modeIdx<sched.length-1){
    modeIdx++; mode=sched[modeIdx].m; modeT=sched[modeIdx].t;

/* ---------- main update ---------- */
function update(dt){
  t+=dt;
  if(state==='start')return;
  if(state==='ready'){
    readyT-=dt;
    for(const g of ghosts)if(g.state==='house')updateHouse(g,dt);
    if(readyT<=0){ state='play'; }
    return;
  }
  if(state==='dying'){
    deathT+=dt;
    if(deathT>1.7){
      lives--;
      if(lives<=0){
        for(const g of ghosts)g.fright=false;
        state='gameover'; saveHS(); SFX.over(); setSiren('off');
      } else { resetPositions(); state='ready'; readyT=2; SFX.ready(); }
    }
    return;
  }
  if(state==='clear'){ clearT+=dt; if(clearT>2.3)nextLevel(); return; }
  if(state==='pause')return;
  if(state==='gameover'){
    for(const g of ghosts){
      if(g.state==='house'){updateHouse(g,dt);continue;}
      if(g.state==='exiting'){updateExiting(g,dt);continue;}
      g.speed=6.5; stepMover(g,dt,()=>ghostDecide(g));
      if(g.state==='eyes'&&(g.path.length===0||Math.hypot(g.x-10.5,g.y-11.5)<0.45))g.state='exiting';
    }
    updateFx(dt); return;
  }
  // play (substepped for reliable collisions)
  const n=Math.max(1,Math.ceil(dt/0.016)),h=dt/n;
  for(let i=0;i<n&&state==='play';i++)updatePlay(h);
  syncSiren();
}

function rr(x,y,w,h,r){
  ctx.beginPath();
  ctx.moveTo(x+r,y); ctx.arcTo(x+w,y,x+w,y+h,r); ctx.arcTo(x+w,y+h,x,y+h,r);
  ctx.arcTo(x,y+h,x,y,r); ctx.arcTo(x,y,x+w,y,r); ctx.closePath();
}

function ghostBodyPath(x,y,r,frame){
  const p=new Path2D(), skirtY=y+0.72*r;
  p.arc(x,y-0.1*r,r,Math.PI,0);
  p.lineTo(x+r,skirtY);
  if(frame===0){ const rb=r/3; for(let i=0;i<3;i++){ const cxp=x+r-rb*(2*i+1); p.arc(cxp,skirtY,rb,0,Math.PI); } }
  else { const rb=r/4; for(let i=0;i<4;i++){ const cxp=x+r-rb*(2*i+1); p.arc(cxp,skirtY,rb,0,Math.PI); } }
  p.closePath(); return p;
}
function drawGhostSprite(g){
  const x=g.x*t,y=g.y*t,r=0.42*t;
  const eyesOnly=g.state==='eyes';

  const frame=Math.floor(t*8+g.id*0.7)%2;
  if(!eyesOnly){
    ctx.save();
    ctx.shadowColor=fright?'rgba(90,90,255,.8)':'rgba(255,255,255,.35)';
    ctx.shadowBlur=t*0.3;
    ctx.fillStyle=fright?(flash?'#ffffff':'#2b2bd8'):GCOL[g.id];
    ctx.fill(ghostBodyPath(x,y,r,frame));
    ctx.restore();
  }
  if(fright){
    const c=flash?'#d81e1e':'#ffd9b3';
    ctx.fillStyle=c;
    const es=r*0.24;
    ctx.fillRect(x-r*0.5,y-r*0.45,es,es);

    }
    ctx.lineTo(x+mw/2,my); ctx.stroke();

/* ---------- input ---------- */
function keyDir(k){
  switch(k){
    case 'ArrowUp':case 'w':case 'W':return UP;
    case 'ArrowLeft':case 'a':case 'A':return LEFT;
    case 'ArrowDown':case 's':case 'S':return DOWN;
    case 'ArrowRight':case 'd':case 'D':return RIGHT;
  }
  return null;
}
function togglePause(){
  if(state==='play'){ state='pause'; setSiren('off'); }
  else if(state==='pause'){ state='play'; }
}
window.addEventListener('keydown',e=>{
  initAudio();
  const k=e.key;
  if(['ArrowUp','ArrowDown','ArrowLeft','ArrowRight',' '].indexOf(k)>=0)e.preventDefault();
  const d=keyDir(k);
  if(state==='start'){
    if(d!=null&&pac)pac.wantDir=d;
    if(k==='Enter'||k===' '||d!=null){ newGame(); SFX.jingle(); }
  } else if(state==='gameover'){
    if(k==='Enter'||k===' '||k==='r'||k==='R'){ newGame(); SFX.jingle(); }
  } else {
    if(d!=null&&state==='play')pac.wantDir=d;
    if(k==='p'||k==='P')togglePause();
    if(k==='m'||k==='M'){ muted=!muted; if(master)master.gain.value=muted?0:0.5; saveMute(); }
    if(k==='r'||k==='R'){ newGame(); SFX.jingle(); }
  }
});
let pDown=null;
cv.addEventListener('pointerdown',e=>{
  initAudio();
  pDown={x:e.clientX,y:e.clientY};
  if(state==='start'||state==='gameover'){ newGame(); SFX.jingle(); pDown=null; }
});
cv.addEventListener('pointermove',e=>{
  if(!pDown||state!=='play')return;
  const dx=e.clientX-pDown.x, dy=e.clientY-pDown.y;
  if(Math.max(Math.abs(dx),Math.abs(dy))>26){
    pac.wantDir=Math.abs(dx)>Math.abs(dy)?(dx>0?RIGHT:LEFT):(dy>0?DOWN:UP);
    pDown={x:e.clientX,y:e.clientY};
  }
});
cv.addEventListener('pointerup',e=>{
  if(pDown&&Math.hypot(e.clientX-pDown.x,e.clientY-pDown.y)<12){
    if(state==='start'||state==='gameover'){ newGame(); SFX.jingle(); }
    else if(state==='pause')togglePause();
  }
  pDown=null;
});
cv.addEventListener('contextmenu',e=>e.preventDefault());
document.addEventListener('visibilitychange',()=>{ if(document.hidden&&state==='play')togglePause(); });
window.addEventListener('blur',()=>{ if(state==='play')togglePause(); });
