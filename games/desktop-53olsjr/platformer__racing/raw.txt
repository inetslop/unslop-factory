    L.items.push({x,w,h:rand(hMin,hMax),y:yFn(),s:Math.random()});
    L.genTo=x+w;
  }
  while(L.items.length && L.items[0].x+L.items[0].w < camX*L.f-320) L.items.shift();
}

function puff(x,y,n,col,sp){
  for(let i=0;i<n;i++){ const a=rand(0,Math.PI*2),v=rand(30,sp||160);
    parts.push({k:'d',x,y,vx:Math.cos(a)*v,vy:Math.sin(a)*v-40,g:700,life:rand(.25,.55),max:.55,r:rand(2,4.5),col}); }
}
function confetti(){
  const cols=['#ffcf3f','#ff5e5b','#2ec4b6','#f4f7ff','#ff6fb0'];
  for(let i=0;i<130;i++) parts.push({k:'c',x:camX+rand(0,VW),y:rand(-40,60),vx:rand(-50,50),vy:rand(40,160),
      g:260,life:rand(1.6,3),max:3,r:rand(3,6),col:cols[i%cols.length],rot:rand(0,9),vr:rand(-7,7)});
}
function stepParts(dt){
  for(let i=parts.length-1;i>=0;i--){ const p=parts[i];
    p.life-=dt; if(p.life<=0){ parts.splice(i,1); continue; }
    p.vy+=(p.g||0)*dt; p.x+=p.vx*dt; p.y+=p.vy*dt;
    if(p.k==='c'){ p.rot+=p.vr*dt; p.x+=Math.sin(p.life*6)*24*dt; }
  }
  if(parts.length>380) parts.splice(0,parts.length-380);
}

function respawn(e,big){
  let i=e.pi; while(i+1<platforms.length && platforms[i].end < e.x) i++;
  const np=platforms[Math.min(i,platforms.length-1)];
  e.x=np.x+18; e.y=np.y; e.vy=0; e.grounded=true; e.pi=i;
  return np;
}

function updateRace(dt){
  raceT+=dt; ensureTrack();
  const p=player;
  const racing = state==='RACING';

  /* ---- player input (only while actively racing) ---- */
  if(racing){
    jumpBuf-=dt; coyote-=dt;
    if(jumpBuf>0){
      if(p.grounded||coyote>0){ p.vy=JUMP_V; p.grounded=false; p.jumps=0; sfx.jump(); puff(p.x-6,p.y,5,'#cfd8f2',120); }
      else if(p.jumps<1){ p.jumps=1; p.vy=DJUMP_V; sfx.djump(); puff(p.x,p.y-4,8,'#ffe9b0',170); }
    }
  }

  /* ---- player physics ---- */
  const ff = (keys.ArrowDown||keys.KeyS) && !p.grounded;
  p.boostT=Math.max(0,p.boostT-dt);
  p.stunT=Math.max(0,p.stunT-dt);
  p.invulnT=Math.max(0,p.invulnT-dt);
  p.ringBoost=Math.max(0,p.ringBoost-6*dt);
  let target = p.boostT>0 ? BOOST_SPD : BASE_SPD*(p.stunT>0?.45:1)+p.ringBoost;
  if(!racing) target=Math.min(target,260);
  p.spd += (target-p.spd)*Math.min(1,dt*5);
  p.x += p.spd*dt;
  const prevY=p.y;
  p.vy += GRAV*(ff?2.0:1)*dt;
  p.y += p.vy*dt;
  p.ph += p.spd*dt*.055;

  const g=groundFor(p);
  if(g && p.vy>=0 && p.y>=g.y-2 && p.y<=g.y+34){
    if(!p.grounded){ p.squash=.4; sfx.land(); puff(p.x,p.y,7,'#cfd8f2',150); }
    p.y=g.y; p.vy=0; p.grounded=true; p.jumps=0;
  } else {
    if(p.grounded){ p.grounded=false; coyote=.09; }
  }

  /* ---- features on current platform ---- */
  const gp=groundFor(p);
  if(gp && p.grounded){
    for(const s of (gp.spikes||[])){
      if(p.x>s.x+4 && p.x<s.x+s.w-2 && p.invulnT<=0){
        p.stunT=.75; shake=Math.max(shake,6); sfx.stumble(); puff(p.x,p.y-8,10,'#ff8a5b',200); break;
      }
    }
    if(gp.boost && !gp.boost.hit && p.x>gp.boost.x-4 && p.x<gp.boost.x+gp.boost.w+10){
      gp.boost.hit=true; p.boostT=1.35; sfx.boost(); puff(p.x-8,p.y,12,'#2ec4b6',220);
    }
  }
  for(let k=p.pi-1;k<=p.pi+2;k++){ const pl=platforms[k]; if(pl&&pl.rings)
    for(const rg of pl.rings){ if(rg.got)continue;
      const dx=rg.x-p.x, dy=rg.y-(p.y-18);
      if(dx*dx+dy*dy<900){ rg.got=true; p.rings++; p.ringBoost=Math.min(48,p.ringBoost+7); sfx.ring(); puff(rg.x,rg.y,8,'#ffd75e',180); } } }

  /* ---- pit fall ---- */
  if(p.y>KILL_Y){
    p.crashes++; shake=18; flashT=.5; sfx.crash();
    respawn(p,true); p.invulnT=1.1; p.spd=240; puff(p.x,p.y,14,'#ffcf6e',220);
  }

  /* ---- gates / laps ---- */
  if(!p.finished && p.nextGate<=LAPS && (p.x-START_X)>=p.nextGate*LAP){
    const split=raceT-p.lastSplitT; p.lastSplitT=raceT; p.splits.push(split);
    p.bestLap=p.bestLap? Math.min(p.bestLap,split):split;
    if(p.nextGate===1) banner('LAP 2','#ffcf6e','split '+fmtTime(split));
    else if(p.nextGate===2) banner('FINAL LAP!','#ff5e5b','split '+fmtTime(split),1.6);
    sfx.lap(); p.nextGate++;
    if(p.nextGate>LAPS) startFinish();
  }

  /* ---- boost trail & run dust ---- */
  if(p.boostT>0 && Math.random()<.9)
    parts.push({k:'d',x:p.x-14,y:p.y-5,vx:-rand(220,430),vy:rand(-60,60),g:0,life:.3,max:.3,r:rand(2.5,5),
      col:['#ffcf6e','#ff8a5b','#2ec4b6'][irand(0,2)]});
  if(p.grounded && Math.random()<dt*11)
    parts.push({k:'d',x:p.x-9,y:p.y,vx:-rand(40,110),vy:rand(-50,-10),g:300,life:.4,max:.4,r:rand(1.6,3),col:'#cfd8f2'});

  /* ---- rivals ---- */
  for(const r of rivals){
    if(r.finished && r.alpha>0){ // coast off after crossing
      r.x+=r.spd*dt; r.spd*=Math.pow(.4,dt); r.alpha-=dt*.8; continue;
    }
    if(r.state==='down'){
      r.down-=dt;
      if(r.down<=0){ const np=respawn(r,false); r.y=np.y; r.vy=0; r.grounded=true; r.state='run'; puff(r.x,r.y,8,'#cfd8f2',160); }
      continue;
    }
    r.stun=Math.max(0,r.stun-dt);
    const d=r.x-p.x;
    let band=1; if(d>800)band=.965; else if(d<-800)band=1.04;
    const sp=r.base*band*(1+r.amp*Math.sin(tGlobal*r.freq+r.ph))*(r.stun>0?.5:1);
    r.spd=sp; r.x+=sp*dt; r.vy+=GRAV*dt; r.y+=r.vy*dt; r.ph+=sp*dt*.055;
    const rg=groundFor(r);
    if(rg && r.vy>=0 && r.y>=rg.y-2 && r.y<=rg.y+34){
      r.y=rg.y; r.vy=0; r.grounded=true;
      for(const s of (rg.spikes||[])) if(r.x>s.x&&r.x<s.x+s.w && r.stun<=0 && Math.random()<r.stumRate*dt){
        r.stun=.7; puff(r.x,r.y-8,6,'#ff8a5b',150); break; }
    } else {
      if(r.grounded){ // reached an edge — decide like a real racer
        r.grounded=false;
        if(Math.random()<r.skill){ r.vy=JUMP_V*rand(.95,1.03); puff(r.x,r.y,4,'#cfd8f2',110); }
      }
    }
    if(r.y>KILL_Y){ r.state='down'; r.down=rand(1.2,1.8); puff(r.x,520,10,'#ff8a5b',200); }
    if(!r.finished && (r.x-START_X)>=RACE){
      r.finished=true; r.finishT=racing? raceT : 0; sfx.rivalFin();
    }
  }

  /* ---- camera & fx decay ---- */
  const camT=p.x-VW*.3+(p.spd-BASE_SPD)*.22;
  camX+=(camT-camX)*Math.min(1,dt*12);
}

function stepAmbient(dt){
  genLayer(LAY.far ,90,220,150,320,40,()=>0);
  genLayer(LAY.mid ,120,300,80,190,60,()=>0);
  genLayer(LAY.cldN,90,220,20,60,220,()=>rand(70,400));
  genLayer(LAY.wisp,260,520,30,80,900,()=>rand(50,500));
  birdT-=dt;
  if(birdT<0){ birdT=rand(5,11); const y=rand(60,210),bx=camX*.5+VW+40,n=irand(3,5);
    for(let i=0;i<n;i++) birds.push({bx:bx+i*26,by:y+(i%2)*14-Math.floor(i/2)*11,ph:i}); }
  while(birds.length&&birds[0].bx<camX*.5-80) birds.shift();
  stepParts(dt);
  if(bannerObj){ bannerObj.t+=dt; if(bannerObj.t>bannerObj.dur) bannerObj=null; }
  if(toast){ toast.t-=dt; if(toast.t<=0)toast=null; }
}

function step(dt){
  tGlobal+=dt;
  shake*=Math.exp(-dt*6); flashT=Math.max(0,flashT-dt);
  switch(state){
    case 'MENU': camX+=90*dt; ensureTrack(); stepAmbient(dt); setHum(.25,48); break;
    case 'COUNTDOWN':
      stepAmbient(dt);
      cdT-=dt; const num=Math.ceil(cdT/.8);
      if(num!==lastNum){ lastNum=num; if(num>0)sfx.beep(false); }
      if(cdT<=0){ state='RACING'; sfx.beep(true); banner('GO!','#7ef0a0','',.8); }
      setHum(.3,52); break;
    case 'RACING': updateRace(dt); stepAmbient(dt); setHum(Math.min(1,player.spd/700),42+player.spd*.16); break;
    case 'FINISHING':
      updateRace(dt*.45); stepAmbient(dt); finishTimer-=dt; setHum(.1,60);
      if(finishTimer<=0 && state==='FINISHING'){} // overlay handled by timeout
      break;
    case 'FINISHED': stepAmbient(dt); break;
    case 'PAUSED': break;
  }
}

/* ================= DRAW HELPERS ================= */
function xform(f,shx,shy){ ctx.setTransform(DPR*SC,0,0,DPR*SC,DPR*(shx*f),DPR*(shy*f)); ctx.translate(-camX*f); }
function rr(x,y,w,h,r){
  r=Math.min(r,w/2,h/2);
  ctx.beginPath();
  ctx.moveTo(x+r,y); ctx.arcTo(x+w,y,x+w,y+h,r); ctx.arcTo(x+w,y+h,x,y+h,r);
  ctx.arcTo(x,y+h,x,y,r); ctx.arcTo(x,y,x+w,y,r); ctx.closePath();
}
function drawRunner(x,y,o){
  ctx.save(); ctx.translate(x,y);
  const sq=o.squash||0, st=o.stretch||0;
  ctx.scale(1+sq*.8-st,1-sq+st);
  if(o.rot)ctx.rotate(o.rot);
  ctx.globalAlpha=(o.alpha==null?1:o.alpha);
  const ph=o.ph, sw=Math.sin(ph)*(o.grounded?5:1.5);
  // legs
  ctx.strokeStyle='#1c2749'; ctx.lineWidth=3.5; ctx.lineCap='round';
  if(o.grounded){
    ctx.beginPath();ctx.moveTo(0,-10);ctx.lineTo(-3+sw,0);ctx.stroke();
    ctx.beginPath();ctx.moveTo(0,-10);ctx.lineTo(4-sw*.8,-1);ctx.stroke();
  }else{
    ctx.beginPath();ctx.moveTo(0,-10);ctx.lineTo(-4,-2);ctx.stroke();
    ctx.beginPath();ctx.moveTo(0,-10);ctx.lineTo(5,-4);ctx.stroke();
  }
  // scarf / cape
  if(o.scarf){
    ctx.strokeStyle=o.scarf; ctx.lineWidth=5; ctx.lineCap='round'; ctx.beginPath();
    let sx=-3,sy=-24; ctx.moveTo(sx,sy);
    for(let i=0;i<6;i++){ sx-=7.2; sy+=Math.sin(tGlobal*12+i*1.3)*2.4+i*.5; ctx.lineTo(sx,sy); }
    ctx.stroke();
  } else if(o.cape){
    const flap=Math.sin(tGlobal*10+ (o.ph||0))*2.5;
    ctx.fillStyle=o.cape; ctx.beginPath();
    ctx.moveTo(-4,-23); ctx.lineTo(-17,-15+flap); ctx.lineTo(-4,-9); ctx.closePath(); ctx.fill();
  }
  // torso
  ctx.fillStyle=o.col; rr(-6,-25,12,16,5); ctx.fill();
  // arm
  ctx.strokeStyle=o.col; ctx.lineWidth=3.5;
  const aw=Math.sin(ph+Math.PI)*(o.grounded?4:2);
  ctx.beginPath();ctx.moveTo(0,-20);ctx.lineTo(aw+2,-13);ctx.stroke();
  // head + cap
  ctx.fillStyle='#f6e8d8'; ctx.beginPath();ctx.arc(3,-30,6.5,0,Math.PI*2);ctx.fill();
  ctx.fillStyle=o.cap; ctx.beginPath();ctx.arc(3,-31,6.5,Math.PI,0);ctx.closePath();ctx.fill();
  ctx.fillRect(3,-32,8,2.6);
  ctx.fillStyle='#1b2340'; ctx.fillRect(5,-30,5.5,3); // visor
  ctx.restore();
}

      ctx.fillStyle='#8f2f38'; ctx.fillRect(s.x-2,p.y-4,s.w+4,5);

function standingsInfo(){
  const p=player;
  let ahead=rivals.filter(r=> (r.finished&&(!p.finished||r.finishT<p.finishT)) ||
                              (!r.finished&&!p.finished&&r.x>p.x)).length;
  const pos=ahead+1;
  const xs=[{v:p.x,me:1}].concat(rivals.map(r=>({v:r.x}))).sort((a,b)=>b.v-a.v);
  let lead=0,toBeat=0;
  if(pos===1)lead=Math.max(0,xs[0].v-xs[1].v); else toBeat=Math.max(0,xs[0].v-p.x);
  return {pos,lead,toBeat};
}
