function hitPlayer(){
  const p=player;if(!p.alive||p.invuln>0)return;
  if(G.power.s>0){

    AU.shieldHit();return;
  }

/* ================= formation stepping ================= */
function beatInterval(){
  const base=Math.max(0.14,0.6*Math.pow(0.95,G.wave-1));
  return base*(0.5+0.5*G.fCount/Math.max(1,G.fTotal));
}
function stepFormation(dt,silent){
  G.stepT-=dt;
  if(G.stepT>0)return;
  G.stepT=silent?Math.max(beatInterval(),0.5):beatInterval();
  const sx=8*L.S;
  G.fx+=G.dir*sx;
  if(G.fx>L.maxShift){G.fx=L.maxShift;G.dir=-1;G.fy+=12*L.S;}
  else if(G.fx<-L.maxShift){G.fx=-L.maxShift;G.dir=1;G.fy+=12*L.S;}
  G.flip=!G.flip;
  if(!silent&&G.state==='playing'){
    const fr=G.fCount/G.fTotal;
    AU.march(fr>0.6?0:fr>0.35?1:fr>0.15?2:3);
  }
  if(silent||!player.alive)return;
  // collisions with bunkers & base line
  for(let i=aliens.length-1;i>=0;i--){
    const a=aliens[i],T=TIER[a.tier],[ax,ay]=alienPos(a);
    const by=ay+SPR[T.spr][0].length*L.PX/2;
    let eaten=false;
    for(const bk of bunkers){
      const r=bunkerRect(bk);
      if(ax>r.x-6&&ax<r.x+r.w+6&&by>=r.y-2&&ay<r.y+r.h){
        damageRadius(bk,clamp(Math.round((ax-r.x)/(r.w/bk.gw)),0,bk.gw-1),clamp(Math.round((by-r.y)/(r.h/bk.gh)),0,bk.gh-1),2.6);
        burst(ax,r.y,T.col,10,150*L.S,0.5);AU.bunker();
        aliens.splice(i,1);G.fCount--;eaten=true;break;
      }
    }
    if(eaten)continue;
    if(by>=L.baseY-4*L.PX){
      if(G.power.s>0){G.power.s=0;G.fy=Math.max(0,G.fy-70);AU.shieldHit();pop('SHIELD SAVED YOU',W/2,H*0.5,COL.shield);}
      else{hitPlayer();}
      break;
    }
  }
}

function updateStars(dt){for(const s of stars){s.y+=(6+s.z*9)*L.S*dt;if(s.y>H+2){s.y=-2;s.x=rand(W);}}}
function updateFX(dt){
  G.shake*=Math.exp(-5.5*dt);if(G.shake<0.1)G.shake=0;
  G.flash=Math.max(0,G.flash-dt*2.2);
  player.recoil=Math.max(0,player.recoil-dt*14);
  for(let i=particles.length-1;i>=0;i--){const p=particles[i];
    p.x+=p.vx*dt;p.y+=p.vy*dt;p.vy+=130*L.S*dt;p.life-=dt;if(p.life<=0)particles.splice(i,1);}
  for(let i=rings.length-1;i>=0;i--){const r=rings[i];r.r+=r.vr*dt;r.life-=dt;if(r.life<=0)rings.splice(i,1);}
  for(let i=popups.length-1;i>=0;i--){const p=popups[i];p.y-=28*L.S*dt;p.life-=dt;if(p.life<=0)popups.splice(i,1);}
}

function updatePlayer(dt){
  const p=player;
  p.invuln=Math.max(0,p.invuln-dt);
  for(const k of['d','r','s'])G.power[k]=Math.max(0,G.power[k]-dt);
  if(!p.alive)return;
  const mv=(keys.r?1:0)-(keys.l?1:0);
  if(mv)p.x+=mv*430*L.S*dt;
  if(touch.active)p.x+=(touch.x-p.x)*Math.min(1,dt*14);
  p.x=clamp(p.x,20*L.S+8,W-20*L.S-8);
  G.fireCd=Math.max(0,G.fireCd-dt);
  if((keys.f||touch.active)&&G.fireCd<=0&&!G.waveTransition)shoot();
}

function updateBullets(dt){
  for(let i=bullets.length-1;i>=0;i--){
    const b=bullets[i];b.y-=760*L.S*dt;
    if(b.y<-24){bullets.splice(i,1);continue;}
    let used=false;
    for(let j=aliens.length-1;j>=0&&!used;j--){
      const a=aliens[j],T=TIER[a.tier],[ax,ay]=alienPos(a);
      if(Math.abs(b.x-ax)<SPR[T.spr][0].length*L.PX*0.3&&Math.abs(b.y-ay)<SPR[T.spr][0].length*L.PX*0.32){killAlien(j);used=true;}
    }
    if(used){bullets.splice(i,1);continue;}
    for(let j=raiders.length-1;j>=0&&!used;j--){const r=raiders[j];
      if(Math.abs(b.x-r.x)<6*L.PX&&Math.abs(b.y-r.y)<5*L.PX){
        addScore(60*pmult(),r.x,r.y);pop('RAIDER DOWN',r.x,r.y-14,COL.cyan,Math.max(9,11*L.S));
        burst(r.x,r.y,COL.tall,16,220*L.S,0.7);G.shake=Math.min(G.shake+3,12);AU.alienHit();raiders.splice(j,1);used=true;}}
    if(used){bullets.splice(i,1);continue;}
    if(G.saucer){const s=G.saucer;
      if(Math.abs(b.x-s.x)<8*L.PX&&Math.abs(b.y-s.y)<3.5*L.PX){
        addScore(s.pts,s.x,s.y);pop('BONUS '+s.pts,s.x,s.y,COL.pink,Math.max(12,15*L.S));
        burst(s.x,s.y,COL.saucer,24,260*L.S,0.8);burst(s.x,s.y,'#ffffff',10,200*L.S,0.5);
        rings.push({x:s.x,y:s.y,r:6,vr:380*L.S,life:0.45,max:0.45,color:COL.saucer,lw:4});
        G.shake=Math.min(G.shake+8,14);AU.bigBoom();G.saucer=null;AU.warbleStop();G.saucerT=rand(8,14);used=true;}}
    if(used){bullets.splice(i,1);continue;}
    for(const bk of bunkers){const r=bunkerRect(bk);
      if(b.x>r.x&&b.x<r.x+r.w&&b.y>r.y&&b.y<r.y+r.h){
        const cx=Math.floor((b.x-r.x)/(r.w/bk.gw)),cy=Math.floor((b.y-r.y)/(r.h/bk.gh));
        if(cx>=0&&cx<bk.gw&&cy>=0&&cy<bk.gh&&bk.cells[cy*bk.gw+cx]){
          damageRadius(bk,cx,cy,1.5);burst(b.x,b.y,'#63e6a0',4,90*L.S,0.3);AU.bunker();used=true;break;}}}
    if(used)bullets.splice(i,1);
  }
}

function updateBombs(dt){
  const p=player;
  for(let i=bombs.length-1;i>=0;i--){
    const b=bombs[i];b.x+=b.vx*dt;b.y+=b.vy*dt;
    if(b.y>H+24||b.x<-24||b.x>W+24){bombs.splice(i,1);continue;}
    if(p.alive&&p.invuln<=0&&Math.abs(b.x-p.x)<5*L.PX&&b.y>L.playerY-3*L.PX&&b.y<L.playerY+4*L.PX){
      bombs.splice(i,1);hitPlayer();continue;}
    let used=false;
    for(const bk of bunkers){const r=bunkerRect(bk);
      if(b.x>r.x&&b.x<r.x+r.w&&b.y>r.y&&b.y<r.y+r.h){
        const cx=Math.floor((b.x-r.x)/(r.w/bk.gw)),cy=Math.floor((b.y-r.y)/(r.h/bk.gh));
        if(cx>=0&&cx<bk.gw&&cy>=0&&cy<bk.gh&&bk.cells[cy*bk.gw+cx]){
          damageRadius(bk,cx,cy,2.4);burst(b.x,b.y,'#63e6a0',7,110*L.S,0.35);AU.bunker();used=true;break;}}}
    if(used)bombs.splice(i,1);
  }
}

function updateSaucer(dt,silent){
  if(G.saucer){const s=G.saucer;s.x+=s.dir*150*L.S*dt;
    if(s.x<-80||s.x>W+80){G.saucer=null;AU.warbleStop();G.saucerT=rand(8,14);}}
  else{G.saucerT-=dt;if(G.saucerT<=0)spawnSaucer(silent);}
}

function updateRaiders(dt){
  for(let i=raiders.length-1;i>=0;i--){
    const r=raiders[i];r.t+=dt;
    r.x+=r.vx*dt*(r.phase==='out'?1.25:1);
    if(r.phase==='in'&&((r.vx>0&&r.x>=r.tx)||(r.vx<0&&r.x<=r.tx))){
      r.phase='out';
      bombs.push({x:r.x,y:r.y+16*L.PX,vx:clamp((player.x-r.x)*rand(0.2,0.45),-150*L.S,150*L.S),vy:(200+G.wave*6)*L.S});
    }
    if(r.x<-80||r.x>W+80){raiders.splice(i,1);continue;}
    const ry=r.y+Math.sin(r.t*5)*10*L.S;
    if(player.alive&&player.invuln<=0&&Math.abs(r.x-player.x)<8*L.PX&&Math.abs(ry-L.playerY)<6*L.PX){
      burst(r.x,ry,COL.tall,14,200*L.S,0.6);raiders.splice(i,1);hitPlayer();}
  }
}

function updatePowerups(dt){
  const p=player;
  for(let i=powerups.length-1;i>=0;i--){
    const u=powerups[i];u.y+=u.vy*dt;u.x+=Math.sin(G.t*3+u.seed)*22*L.S*dt;
    if(u.y>H+24){powerups.splice(i,1);continue;}
    if(p.alive&&Math.abs(u.x-p.x)<Math.max(24,8*L.PX)&&Math.abs(u.y-L.playerY)<Math.max(20,6*L.PX)){
      powerups.splice(i,1);
      if(u.type==='D')G.power.d=10;else if(u.type==='R')G.power.r=10;else G.power.s=15;
      pop(POW_NAME[u.type],p.x,L.playerY-24*L.PX,POW_COL[u.type],Math.max(11,13*L.S));AU.power();}
  }
}

function update(dt){
  G.t+=dt;G.dt=dt;
  updateStars(dt);updateFX(dt);
  if(G.state==='menu'){
    G.demoT+=dt;if(G.demoT>22)setupWave(1);
    stepFormation(dt,true);updateSaucer(dt,true);return;
  }
  if(G.state!=='playing')return;

  updatePlayer(dt);
  updateBullets(dt);

  if(!player.alive){
    G.deadT-=dt;
    if(G.deadT<=0){
      if(G.willDie)doGameOver();
      else{player.alive=true;player.invuln=2.5;player.x=W/2;G.fx=0;G.fy=0;G.freeze=false;AU.ui();}
    }
  }else if(!G.waveTransition){
    stepFormation(dt,false);
    G.bombT-=dt;if(G.bombT<=0){fireBomb();G.bombT=clamp(1.7-G.wave*0.05,0.38,1.7)*rand(0.6,1.4);}
    if(G.wave>=2){G.raiderT-=dt;if(G.raiderT<=0){spawnRaider();G.raiderT=Math.max(2.5,rand(5,9)-G.wave*0.1);}}
  }
  updateBombs(dt);updateSaucer(dt,false);updateRaiders(dt);updatePowerups(dt);

  if(G.waveTransition){G.nextWaveT-=dt;if(G.nextWaveT<=0)setupWave(G.wave+1);}
  G.banner=Math.max(0,G.banner-dt);
}

  let ix=W/2-(icons.length-1)*46*L.S;
  for(const[nm,c,off]of icons){const spr=getSpr(nm,Math.floor(G.t*2)%2?1:0,c);
    const dw=spr.width*L.PX,dh=spr.height*L.PX;
    ctx.drawImage(spr,ix-dw/2,y+u*5.4-dh/2+Math.sin(G.t*3+off)*3,dw,dh);ix+=92*L.S;}
}
function drawBanner(){
  const t=2.4-G.banner,a=clamp(Math.min(t*3,G.banner*2.5),0,1);
  ctx.globalAlpha=a;
  const uB=clamp(Math.floor(titleUnit()*0.62),2,7);
  pxText('WAVE '+G.wave,W/2,H*0.3,uB,'#bfffe9');
  ctx.fillStyle=COL.dim;ctx.font=`${Math.max(10,12*L.S)}px ${FONT_M}`;
  ctx.textAlign='center';ctx.fillText(FLAVOR[(G.wave-1)%FLAVOR.length],W/2,H*0.3+uB*6);
  ctx.globalAlpha=1;
}

function draw(){
  const shx=(Math.random()*2-1)*G.shake,shy=(Math.random()*2-1)*G.shake;
  ctx.save();ctx.translate(shx,shy);
  drawBG();
  if(G.state!=='menu')drawBunkers();
  drawAliens();drawSaucer();drawRaiders();
  if(G.state!=='menu'){drawBombs();drawPowerups();drawBullets();drawPlayerShip();}
  drawFX();
  if(G.state!=='menu'){drawBaseline();drawHUD();}
  ctx.restore();

  if(G.state==='gameover'){ctx.fillStyle='rgba(4,10,18,0.5)';ctx.fillRect(0,0,W,H);}
  if(G.state==='menu')drawTitle();
  else{
    if(G.banner>0&&G.state==='playing')drawBanner();
    if(G.state==='paused')pxText('PAUSED',W/2,H*0.4,titleUnit()*0.8,COL.cyan);
    if(G.state==='gameover')pxText('GAME OVER',W/2,H*0.26,Math.min(titleUnit(),Math.floor(W*0.9/(53))),COL.pink);
  }
  if(G.flash>0){ctx.fillStyle=`rgba(255,255,255,${Math.min(0.45,G.flash)})`;ctx.fillRect(0,0,W,H);}
}

function startGame(){
  if(AU.ctx)AU.resume();
  G.score=0;G.lives=3;G.extraIdx=0;G.power={d:0,r:0,s:0};
  player.x=W/2;player.alive=true;player.invuln=1.2;
  particles=[];rings=[];popups=[];bullets=[];

  G.state='playing';hide(ovMenu);hide(ovOver);hide(ovPause);blurActive();AU.ui();
}
function doGameOver(){
  G.state='gameover';
  const rec=G.score>G.hi;
  if(rec){G.hi=G.score;store.set('nf_hi',G.hi);}
  document.getElementById('st-score').textContent=pad6(G.score);
  document.getElementById('st-wave').textContent=G.wave;
  document.getElementById('st-hi').textContent=pad6(G.hi);
  if(rec)show(document.getElementById('st-rec'));else hide(document.getElementById('st-rec'));
  show(ovOver);blurActive();
}

    else if(G.state==='gameover')startGame();

  if(k==='p'||k==='P'||k==='Escape')togglePause();
  if(k==='m'||k==='M')toggleMute();

});
