    const comp=AC.createDynamicsCompressor();master.connect(comp);comp.connect(AC.destination);
    noiseBuf=AC.createBuffer(1,AC.sampleRate,AC.sampleRate);
    const d=noiseBuf.getChannelData(0);for(let i=0;i<d.length;i++)d[i]=Math.random()*2-1;
  }catch(e){AC=null;}
}
function tone(f,f2,dur,type='square',vol=.2,delay=0){
  if(!AC||muted)return;
  try{
    const t0=AC.currentTime+delay,o=AC.createOscillator(),g=AC.createGain();
    o.type=type;o.frequency.setValueAtTime(Math.max(30,f),t0);
    if(f2)o.frequency.exponentialRampToValueAtTime(Math.max(30,f2),t0+dur);
    g.gain.setValueAtTime(vol,t0);g.gain.exponentialRampToValueAtTime(.0008,t0+dur);
    o.connect(g).connect(master);o.start(t0);o.stop(t0+dur+.03);
  }catch(e){}
}
function noise(dur,vol,freq=1000,delay=0){
  if(!AC||muted)return;
  try{
    const t0=AC.currentTime+delay,s=AC.createBufferSource();s.buffer=noiseBuf;
    const f=AC.createBiquadFilter();f.type='lowpass';f.frequency.value=freq;
    const g=AC.createGain();g.gain.setValueAtTime(vol,t0);g.gain.exponentialRampToValueAtTime(.0008,t0+dur);
    s.connect(f).connect(g).connect(master);s.start(t0);s.stop(t0+dur+.03);
  }catch(e){}
}

function updatePlay(dt){
  G.t+=dt;const p=G.p,f=G.f;
  /* player */
  const[ix,iy]=moveInput();const SP=250,k=Math.min(1,dt*14);
  p.vx=lerp(p.vx,ix*SP,k);p.vy=lerp(p.vy,iy*SP,k);
  p.x=clamp(p.x+p.vx*dt,PANEL.x+16,W-PANEL.x-16);
  p.y=clamp(p.y+p.vy*dt,HUD_H+20,H-PANEL.x-14);
  const spd=Math.hypot(p.vx,p.vy);
  if(spd>30)p.face=Math.atan2(p.vy,p.vx);
  p.mouth+=dt*(p.chomp>0?15:(spd>30?10:4.5));
  if(p.invuln>0)p.invuln-=dt;
  if(p.chomp>0){
    p.chomp-=dt;
    if(p.chomp<=0){p.chomp=0;sfx.sputter();}
  }
  /* dot economy */
  G.dotT-=dt;
  if(G.dotT<=0){if(G.dots.length<74)spawnDot();G.dotT=.85;}
  G.pearlT-=dt;
  if(G.pearlT<=0){
    const cands=G.dots.filter(d=>d.type==='d');
    if(cands.length){cands[Math.floor(rand(cands.length))].type='p';sfx.pearlSpawn();G.pearlT=rand(11,17);}
    else G.pearlT=1;
  }
  /* eat dots & pearls */
  for(let i=G.dots.length-1;i>=0;i--){
    const d=G.dots[i];
    if(d.type==='d'){
      if(Math.hypot(p.x-d.x,p.y-d.y)<20){
        G.score+=10;G.waka=!G.waka;sfx.dot();
        burst(d.x,d.y,4,'#ffd75e',60,2,.35);G.dots.splice(i,1);
      }
    }else{
      if(Math.hypot(p.x-d.x,p.y-d.y)<24){
        G.score+=50;popup('50',d.x,d.y,'#ff4fd8');sfx.pearl();
        p.chomp=G.CHOMP_T;p.chompKills=0;
        burst(d.x,d.y,26,'#ff4fd8',220);
        G.banner={text:'CHOMP MODE!',sub:'',t:0,dur:.9,color:'#ff4fd8'};
        G.dots.splice(i,1);
      }
    }
  }
  /* formation */
  let alive=0;for(const v of G.invs)if(!v.dead)alive++;
  if(alive>0){
    f.t+=dt;if(f.t>.42){f.t=0;f.frame^=1;}
    const prog=1-alive/(ROWS*COLS);
    const sp=(30+(G.wave-1)*7)*(0.6+0.95*prog);
    f.x+=f.dir*sp*dt;
    const minX=PANEL.x+18,maxX=W-PANEL.x-18-blockW;
    if(f.x<minX){f.x=minX;f.dir=1;f.y+=14;}
    else if(f.x>maxX){f.x=maxX;f.dir=-1;f.y+=14;}
    /* shooting */
    G.shotT-=dt;
    if(G.shotT<=0&&G.lasers.length<36){
      const arr=[];for(const v of G.invs)if(!v.dead)arr.push(v);
      arr.sort((a,b)=>b.r-a.r);
      const v=arr[Math.floor(rand(Math.max(1,Math.ceil(arr.length*.45))))];
      const cx=f.x+v.c*SPX+16,cy=f.y+v.r*SPY+12;
      const ls=Math.min(430,265+G.wave*22);
      const dist=Math.hypot(p.x-cx,p.y-cy),tt=dist/ls;
      const a=Math.atan2((p.y+p.vy*tt*.6)-cy,(p.x+p.vx*tt*.6)-cx)+rand(-.1,.1);
      G.lasers.push({x:cx,y:cy,vx:Math.cos(a)*ls,vy:Math.sin(a)*ls});
      sfx.laser();
      G.shotT=rand(.55,1.2)/(1+(G.wave-1)*.13);
    }
    /* breach */
    if(G.breachCd>0)G.breachCd-=dt;
    let crossed=false;
    for(const v of G.invs){if(!v.dead&&f.y+v.r*SPY+24>H-38){crossed=true;break;}}
    if(crossed&&G.breachCd<=0){
      for(const v of G.invs){
        if(v.dead)continue;
        const vy=f.y+v.r*SPY+24;
        if(vy>H-58){v.dead=true;burst(f.x+v.c*SPX+16,vy-10,22,ROWCOL[v.r],260);}
      }
      G.breachCd=2.2;G.shake=13;G.flash=.55;sfx.boom();
      G.lives--;
      if(G.lives<1){gameOver();return;}
    }
  }
  /* lasers */
  for(let i=G.lasers.length-1;i>=0;i--){
    const L=G.lasers[i];L.x+=L.vx*dt;L.y+=L.vy*dt;
    if(L.y>H-14||L.x<PANEL.x+4||L.x>W-PANEL.x-4){
      burst(clamp(L.x,PANEL.x+8,W-PANEL.x-8),H-16,7,'#7defff',90,2,.4);
      if(Math.random()<.5&&G.dots.length<80)G.dots.push({x:clamp(L.x,30,W-30),y:H-18,type:'d',ph:rand(TAU)});
      G.lasers.splice(i,1);continue;
    }
    if(p.invuln<=0&&Math.hypot(L.x-p.x,L.y-p.y)<17){G.lasers.splice(i,1);kill();return;}
  }
  /* ghosts */
  for(let i=G.respawnQ.length-1;i>=0;i--){
    const q=G.respawnQ[i];q.t-=dt;
    if(q.t<=0){
      G.ghosts.push({x:Math.random()<.5?PANEL.x-26:W-PANEL.x+26,y:rand(H*.45,H-40),vx:0,vy:0,ci:q.ci,eats:0,wob:rand(TAU)});
      G.respawnQ.splice(i,1);
    }
  }
  const gsp=Math.min(235,140+G.wave*14);
  for(const g of G.ghosts){
    let dx=p.x-g.x,dy=p.y-g.y;const d=Math.hypot(dx,dy)||1;dx/=d;dy/=d;
    if(p.chomp>0){dx=-dx;dy=-dy;}
    const s=gsp*(p.chomp>0?.78:1);
    g.vx=lerp(g.vx,dx*s,Math.min(1,dt*3.2));
    g.vy=lerp(g.vy,dy*s,Math.min(1,dt*3.2));
    g.wob+=dt*5;
    const wm=Math.hypot(g.vx,g.vy)||1;
    g.x+=(g.vx-g.vy/wm*Math.sin(g.wob)*26)*dt;
    g.y+=(g.vy+g.vx/wm*Math.sin(g.wob)*26)*dt;
    if(g.x<PANEL.x+16){g.x=PANEL.x+16;if(g.vx<0)g.vx*=-.5;}
    if(g.x>W-PANEL.x-16){g.x=W-PANEL.x-16;if(g.vx>0)g.vx*=-.5;}
    if(g.y<HUD_H+26){g.y=HUD_H+26;if(g.vy<0)g.vy*=-.5;}
    if(g.y>H-PANEL.x-14){g.y=H-PANEL.x-14;if(g.vy>0)g.vy*=-.5;}
  }
  /* chomp eats / contact death */
  if(p.chomp>0){
    for(const v of G.invs){
      if(v.dead)continue;
      const cx=f.x+v.c*SPX+16,cy=f.y+v.r*SPY+12;
      if(Math.hypot(cx-p.x,cy-p.y)<34){
        v.dead=true;p.chompKills++;
        G.score+=200;popup('200',cx,cy-8,ROWCOL[v.r]);
        burst(cx,cy,20,ROWCOL[v.r],240);sfx.eat(false);
        if(p.chompKills===1)spawnPearlNear(p.x,p.y-90);
        if(p.chompKills%5===0){G.score+=250;popup('FEAST +250',p.x,p.y-34,'#ffe14d');sfx.feast();}
        let aliveNow=0;for(const u of G.invs)if(!u.dead)aliveNow++;
        if(aliveNow===0){waveClear();return;}
      }
    }
    for(let i=G.ghosts.length-1;i>=0;i--){
      const g=G.ghosts[i];
      if(Math.hypot(g.x-p.x,g.y-p.y)<27){
        G.score+=300;popup('300',g.x,g.y-10,GHOSTCOL[g.ci]);
        burst(g.x,g.y,24,'#5f8dff',260);sfx.eat(true);
        g.eats++;if(g.eats<2)G.respawnQ.push({t:5,ci:g.ci});
        G.ghosts.splice(i,1);
      }
    }
  }else{
    if(p.invuln<=0){
      for(const v of G.invs){
        if(v.dead)continue;
        const cx=f.x+v.c*SPX+16,cy=f.y+v.r*SPY+12;
        if(Math.hypot(cx-p.x,cy-p.y)<30){kill();return;}
      }
      for(const g of G.ghosts)if(Math.hypot(g.x-p.x,g.y-p.y)<26){kill();return;}
    }
  }
  /* threat alarm */
  let threat=false;
  for(const v of G.invs)if(!v.dead&&f.y+v.r*SPY+24>DANGER_Y-50){threat=true;break;}
  if(threat){G.alarmT-=dt;if(G.alarmT<=0){sfx.alarm();G.alarmT=.5;}}
  G._threat=threat;
}

    if(pt.life<=0){G.parts.splice(i,1);continue;}
    pt.x+=pt.vx*dt;pt.y+=pt.vy*dt;
    pt.vx*=(1-2.4*dt);pt.vy*=(1-2.4*dt);
  }
  for(let i=G.pops.length-1;i>=0;i--){
    const p=G.pops[i];p.t+=dt;p.y-=28*dt;
    if(p.t>p.life)G.pops.splice(i,1);
  }
  G.shake=Math.max(0,G.shake-dt*30);
  G.flash=Math.max(0,G.flash-dt*1.4);
  if(G.ring){G.ring.t+=dt;if(G.ring.t>.6)G.ring=null;}
  if(G.banner){G.banner.t+=dt;if(G.banner.t>G.banner.dur)G.banner=null;}
}

  ctx.beginPath();
  ctx.arc(0,-3,r,Math.PI,0);
  ctx.lineTo(r,yb);
  for(let i=0;i<3;i++){const cx=r-(r/3)*(i*2+1);ctx.arc(cx,yb,r/3,0,Math.PI);}
  ctx.closePath();ctx.fill();
  ctx.shadowBlur=0;
  if(fr){
    ctx.fillStyle='#0b1030';
    ctx.beginPath();ctx.arc(-4.5,-3,2.2,0,TAU);ctx.arc(4.5,-3,2.2,0,TAU);ctx.fill();
    ctx.strokeStyle='#0b1030';ctx.lineWidth=1.6;ctx.beginPath();
    ctx.moveTo(-8,4);ctx.lineTo(-5,1);ctx.lineTo(-2,4);ctx.lineTo(1,1);ctx.lineTo(4,4);ctx.lineTo(7,1);ctx.stroke();
  }else{
    const m=Math.hypot(g.vx,g.vy)||1,eox=g.vx/m*2,eoy=g.vy/m*2;
    ctx.fillStyle='#fff';
    ctx.beginPath();ctx.ellipse(-4.6,-4,3.8,4.6,0,0,TAU);ctx.ellipse(4.6,-4,3.8,4.6,0,0,TAU);ctx.fill();
    ctx.fillStyle='#10173a';
    ctx.beginPath();ctx.arc(-4.6+eox,-4+eoy,2,0,TAU);ctx.arc(4.6+eox,-4+eoy,2,0,TAU);ctx.fill();
  }
  ctx.restore();
}
function drawInv(v){
  const f=G.f;
  const x=f.x+v.c*SPX,y=f.y+v.r*SPY+Math.sin(G.t*3+v.bob)*2;
  let col=ROWCOL[v.r];
  if(G.p.chomp>0)col=(G.p.chomp<1.6&&Math.floor(CLOCK*8)%2===0)?'#eaf2ff':'#4f7dff';
  ctx.drawImage(invSprite(f.frame,col),x,y);
}
function drawDots(){
  for(const d of G.dots){
    if(d.type==='d'){
      const a=.7+.3*Math.sin(CLOCK*3+d.ph);
      ctx.globalAlpha=a*.25;ctx.fillStyle='#ffd75e';
      ctx.beginPath();ctx.arc(d.x,d.y,6.5,0,TAU);ctx.fill();
      ctx.globalAlpha=a;
      ctx.beginPath();ctx.arc(d.x,d.y,3.2,0,TAU);ctx.fill();
    }else{
      const rr2=7+Math.sin(CLOCK*5+d.ph)*1.6;
      ctx.globalAlpha=.22;ctx.fillStyle='#ff4fd8';
      ctx.beginPath();ctx.arc(d.x,d.y,rr2+7,0,TAU);ctx.fill();
      ctx.globalAlpha=1;
      ctx.beginPath();ctx.arc(d.x,d.y,rr2,0,TAU);ctx.fill();
      ctx.fillStyle='#fff';
      ctx.beginPath();ctx.arc(d.x,d.y,rr2*.45,0,TAU);ctx.fill();
      ctx.strokeStyle='rgba(255,255,255,.8)';ctx.lineWidth=1.5;
      const a=CLOCK*2+d.ph;
      for(let i=0;i<4;i++){
        const an=a+i*Math.PI/2;
        ctx.beginPath();
        ctx.moveTo(d.x+Math.cos(an)*(rr2+3),d.y+Math.sin(an)*(rr2+3));
        ctx.lineTo(d.x+Math.cos(an)*(rr2+8),d.y+Math.sin(an)*(rr2+8));
        ctx.stroke();
      }
    }
  }
  ctx.globalAlpha=1;
}
function drawLasers(){
  for(const L of G.lasers){
    const a=Math.atan2(L.vy,L.vx)+Math.PI/2;
    ctx.save();ctx.translate(L.x,L.y);ctx.rotate(a);
    ctx.shadowColor='#7defff';ctx.shadowBlur=8;
    ctx.fillStyle='rgba(125,239,255,.4)';ctx.fillRect(-3,-10,6,20);
    ctx.fillStyle='#ffffff';ctx.fillRect(-1.4,-9,2.8,18);
    ctx.restore();
  }
}
function drawPlayer(){
  const p=G.p;
  if(G.mode==='dying')return;
  if(p.invuln>0&&Math.floor(CLOCK*12)%2===0)return;
  drawPac(p.x,p.y,15,p.face,p.mouth,p.chomp>0);
  if(p.chomp>0){
    const frac=p.chomp/G.CHOMP_T;
    const col=frac<.3?'#ff4d5e':'#ff4fd8';
    ctx.save();ctx.strokeStyle=col;ctx.lineWidth=3;
    ctx.shadowColor=col;ctx.shadowBlur=10;
    ctx.beginPath();ctx.arc(p.x,p.y,26,-Math.PI/2,-Math.PI/2+frac*TAU);ctx.stroke();
    ctx.restore();
  }
}
function drawFx(){

}

function frame(ts){
  requestAnimationFrame(frame);
  let dt=(ts-lastTs)/1000;lastTs=ts;
  if(!(dt>0))dt=.016;if(dt>.033)dt=.033;
  CLOCK+=dt;

  if(G.mode!=='pause')updateFx(dt);
  if(G.mode==='play')updatePlay(dt);
  else if(G.mode==='dying'){G.dieT+=dt;if(G.dieT>1.15)afterDeath();}
  else if(G.mode==='waveclear'){
    G.wcT+=dt;G.f.t+=dt;if(G.f.t>.42){G.f.t=0;G.f.frame^=1;}
    if(G.wcT>2.3){G.wave++;startWave(G.wave);G.mode='play';}
  }

  draw();
}
