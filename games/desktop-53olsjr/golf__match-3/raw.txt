function findRuns(){

function mAt(r,c){
  const t=grid[r][c].t;let n=1;
  for(let cc=c-1;cc>=0&&grid[r][cc]&&grid[r][cc].t===t;cc--)n++;
  for(let cc=c+1;cc<COLS&&grid[r][cc]&&grid[r][cc].t===t;cc++)n++;
  if(n>=3)return true;n=1;
  for(let rr=r-1;rr>=0&&grid[rr][c]&&grid[rr][c].t===t;rr--)n++;
  for(let rr=r+1;rr<ROWS&&grid[rr][c]&&grid[rr][c].t===t;rr++)n++;
  return n>=3;
}
function hasMove(){
  for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++){
    const b=grid[r][c];
    const dirs=[[0,1],[1,0]];
    for(const d of dirs){const nr=r+d[0],nc=c+d[1];if(nr>=ROWS||nc>=COLS)continue;
      const o=grid[nr][nc];if(o.t===b.t)continue;
      const tmp=b.t;b.t=o.t;o.t=tmp;
      const ok=mAt(r,c)||mAt(nr,nc);
      o.t=b.t;b.t=tmp;if(ok)return true}
  }return false;
}

const key=(r,c)=>r*COLS+c;
function analyze(runs,origin){
  const clear=new Set(),pending={},beams=[],bSeen=new Set();let ace=false;
  for(const run of runs)for(const p of run.cells)clear.add(key(p[0],p[1]));
  // coverage map for cross detection
  const cov={};
  for(const run of runs)for(const p of run.cells){const k=key(p[0],p[1]);cov[k]=cov[k]||{};cov[k][run.dir]=(cov[k][run.dir]||0)+1}
  for(const run of runs){
    if(run.len<4)continue;
    let pos=null;
    for(const o of(origin||[]))if(run.cells.some(p=>p[0]===o.r&&p[1]===o.c)){pos=[o.r,o.c];break}
    if(!pos)pos=run.cells[(run.len/2)|0];
    const k=key(pos[0],pos[1]);
    pending[k]=run.len>=5||(cov[k].h&&cov[k].v)?3:(run.dir==='h'?1:2);
  }
  for(const k in pending)clear.delete(+k);
  // BFS special triggers
  const trig=new Set(),q=[...clear];
  while(q.length){
    const k=q.pop();if(trig.has(k))continue;
    const r=(k/COLS)|0,c=k%COLS;
    if(pending[k])continue;
    const b=grid[r][c];if(!b||!b.sp)continue;
    trig.add(k);
    if(b.sp===1){for(let cc=0;cc<COLS;cc++){const k2=key(r,cc);if(!clear.has(k2)){clear.add(k2);q.push(k2)}}
      const bk='r'+r;if(!bSeen.has(bk)){bSeen.add(bk);beams.push({kind:'row',i:r})}}
    else if(b.sp===2){for(let rr=0;rr<ROWS;rr++){const k2=key(rr,c);if(!clear.has(k2)){clear.add(k2);q.push(k2)}}
      const bk='c'+c;if(!bSeen.has(bk)){bSeen.add(bk);beams.push({kind:'col',i:c})}}
    else{ // ace: wipe most common color among non-cleared
      const cnt=[0,0,0,0,0];
      for(let rr=0;rr<ROWS;rr++)for(let cc=0;cc<COLS;cc++){const bb=grid[rr][cc];if(bb&&!clear.has(key(rr,cc)))cnt[bb.t]++}
      let best=-1,bi=0;for(let t=0;t<5;t++)if(cnt[t]>best){best=cnt[t];bi=t}
      if(best>0){ace=true;
        for(let rr=0;rr<ROWS;rr++)for(let cc=0;cc<COLS;cc++){const bb=grid[rr][cc];if(bb&&bb.t===bi){const k2=key(rr,cc);if(!clear.has(k2)){clear.add(k2);q.push(k2)}}}}
    }
  }
  return{clear,pending,beams,ace};
}
function collapse(){
  for(let c=0;c<COLS;c++){
    let w=ROWS-1;
    for(let r=ROWS-1;r>=0;r--){const b=grid[r][c];if(b){
      if(w!==r){grid[w][c]=b;grid[r][c]=null;b.r=w}
      b.c=c;b.fy=b.y;b.ty=rowY(w);w--;
    }}
    for(let r=w;r>=0;r--){
      const b=mkBall(ri(5));b.r=r;b.c=c;grid[r][c]=b;
      b.x=colX(c);b.y=rowY(r)-(w+1)*cell;b.fy=b.y;b.ty=rowY(r);b.vy=0;
    }
  }
}

async function attemptSwap(a,b){
  if(phase!=='idle')return;
  phase='swap';
  const ba=grid[a.r][a.c],bb=grid[b.r][b.c];if(!ba||!bb){phase='idle';return}
  S.swap();
  const ax=colX(a.c),ay=rowY(a.r),bx=colX(b.c),by=rowY(b.r);
  ba.ty=null;bb.ty=null;ba.vy=0;bb.vy=0;
  swapAnim={ba,bb,ax,ay,bx,by,t0:performance.now(),dur:150,rev:false};
  await sleep(165);
  grid[a.r][a.c]=bb;grid[b.r][b.c]=ba;ba.r=b.r;ba.c=b.c;bb.r=a.r;bb.c=a.c;
  ba.ty=rowY(b.r);bb.ty=rowY(a.r);

async function resolve(runs,origin){
  let chain=0;
  while(runs.length){
    chain++;
    const an=analyze(runs,origin);origin=[];
    const now=performance.now();
    // centroid for score text
    let cx=0,cy=0;for(const k of an.clear){cx+=colX(k%COLS);cy+=rowY((k/COLS)|0)}
    cx/=an.clear.size;cy/=an.clear.size;
    for(const k of an.clear){const b=grid[(k/COLS)|0][k%COLS];if(b&&!b.cl)b.cl={t0:now+rand(0,60)}}
    an.beams.forEach(b=>beams.push(Object.assign({t0:now},b)));
    if(an.ace)aceFlash={t0:now};
    S.pop(chain);if(an.beams.length)S.line();if(an.ace)S.ace();
    if(chain>=2)comboText(chain);
    await sleep(310);
    // commit removals + transforms
    sunk+=an.clear.size;
    for(const k of an.clear)grid[(k/COLS)|0][k%COLS]=null;
    for(const k in an.pending){const r=(+k/COLS)|0,c=+k%COLS;
      if(grid[r][c]){grid[r][c].sp=an.pending[k];rings.push({x:colX(c),y:rowY(r),t0:performance.now(),col:'#d9a441'});
        floats.push({x:colX(c),y:rowY(r)-cell*.5,t0:performance.now(),txt:an.pending[k]===3?'ACE!':'STRIPE!',size:cell*.3,col:'#b8842a'})}

    floats.push({x:cx,y:cy,t0:performance.now(),txt:'+'+an.clear.size,size:cell*.55,col:'#fffbe8'});
    updateMeter();
    collapse();
    await waitSettle();
    runs=findRuns();
  }
}

function update(dt,now){
  if(swapAnim){
    const k=easeIO(clamp((now-swapAnim.t0)/swapAnim.dur,0,1));
    const A=swapAnim;
    if(!A.rev){A.ba.x=lerp(A.ax,A.bx,k);A.ba.y=lerp(A.ay,A.by,k);A.bb.x=lerp(A.bx,A.ax,k);A.bb.y=lerp(A.by,A.ay,k)}
    else{A.ba.x=lerp(A.bx,A.ax,k);A.ba.y=lerp(A.by,A.ay,k);A.bb.x=lerp(A.ax,A.bx,k);A.bb.y=lerp(A.ay,A.by,k)}
    if(k>=1)swapAnim=null;
  }
  const G=cell*52,VMAX=cell*46;let lands=0;
  allBalls(b=>{
    if(b.ty!=null&&b.y<b.ty-0.4){
      b.vy=Math.min(VMAX,b.vy+G*dt);b.y+=b.vy*dt;
      if(b.y>=b.ty){b.y=b.ty;b.vy=0;if((b.ty-b.fy)>cell*1.6)b.sq=1;lands++}
    }
  });
  if(lands)S.land();
  allBalls(b=>{if(b.sq>0)b.sq=Math.max(0,b.sq-dt*5)});
  if(settleRes){
    let ok=true;
    allBalls(b=>{if(b.vy!==0||b.y<(b.ty==null?b.y:b.ty)-0.6)ok=false});
    if(ok){const s=settleRes;settleRes=null;s.res()}
  }
  for(let i=particles.length-1;i>=0;i--){const p=particles[i];
    if((now-p.t0)/1000>p.life){particles.splice(i,1);continue}
    p.vy+=p.g*dt;p.x+=p.vx*dt;p.y+=p.vy*dt;if(p.shape==='r')p.rot+=p.vr*dt}
  for(let i=beams.length-1;i>=0;i--)if(now-beams[i].t0>340)beams.splice(i,1);
  for(let i=rings.length-1;i>=0;i--)if(now-rings[i].t0>380)rings.splice(i,1);
  for(let i=floats.length-1;i>=0;i--)if(now-floats[i].t0>750)floats.splice(i,1);
}
