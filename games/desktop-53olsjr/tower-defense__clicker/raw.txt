        if(T-lastShotS>.07){sShoot();lastShotS=T;}}}
    else t.face+=angDiff(t.a+Math.PI/2,t.face)*Math.min(1,dt*4);
  }

    if(!del)for(const e of enemies){if(e.dead)continue;const dx=e.x-s.x,dy=e.y-s.y;if(dx*dx+dy*dy<(e.r+4)*(e.r+4)){hitEnemy(e,shotDmg()*(Math.random()<.1?2:1),Math.random()<.1);del=true;break;}}
    if(del)shots.splice(i,1);}

}
function ambient(dt){

  for(let i=floats.length-1;i>=0;i--){const f=floats[i];f.life-=dt;f.y-=30*dt;if(f.life<=0)floats.splice(i,1);}

}
