        a.x-=dx*push;a.y-=dy*push;b.x+=dx*push;b.y+=dy*push;}}}

  // goo
  for(let i=goo.length-1;i>=0;i--){const g=goo[i];g.life-=dt;
    g.x+=g.vx*dt;g.y+=g.vy*dt;
    if(Math.random()<dt*30)parts.push({t:'goo',x:g.x,y:g.y,vx:0,vy:0,life:.25,max:.25,size:2.4});
    let dead=g.life<=0||g.x<-40||g.x>W+40||g.y<-40||g.y>H+40;
    if(!dead&&aggro&&P&&Math.hypot(g.x-P.x,g.y-P.y)<P.r+g.r){
      damagePlayer(10,g.x,g.y);parts.push({t:'goo',x:g.x,y:g.y,vx:60,vy:0,life:.4,max:.4,size:5});dead=true;}
    if(dead){if(g.life<=0||!aggro)gore(g.x,g.y,16,5,'70,120,40');goo.splice(i,1);}}

  // bullets
  for(let i=bullets.length-1;i>=0;i--){const b=bullets[i];b.life-=dt;
    b.x+=b.vx*dt;b.y+=b.vy*dt;
    if(b.life<=0||b.x<-40||b.x>W+40||b.y<-40||b.y>H+40){bullets.splice(i,1);continue;}
    let hitB=false;
    for(let gi=goo.length-1;gi>=0;gi--){const g=goo[gi];
      if((g.x-b.x)*(g.x-b.x)+(g.y-b.y)*(g.y-b.y)<(g.r+4)*(g.r+4)){
        parts.push({t:'goo',x:g.x,y:g.y,vx:80,vy:-30,life:.35,max:.35,size:4});goo.splice(gi,1);hitB=true;break;}}
    if(hitB){bullets.splice(i,1);continue;}
    for(const z of zombies){const dx=z.x-b.x,dy=z.y-b.y;
      if(dx*dx+dy*dy<(z.r+3)*(z.r+3)){
        z.hp-=b.dmg;z.flash=.09;hits++;
        const a=Math.atan2(b.vy,b.vx);z.kx+=Math.cos(a)*b.kb;z.ky+=Math.sin(a)*b.kb;
        bloodBurst(b.x,b.y,3,120*S);
        if(texts.length<60)addText(z.x+(Math.random()-.5)*8,z.y-z.r-4,''+Math.round(b.dmg),'#ffd9a0',10);
        sfx.meat();bullets.splice(i,1);
        if(z.hp<=0)killZombie(z);
        break;}}
  }

  // drops
  for(let i=drops.length-1;i>=0;i--){const d=drops[i];d.bobT+=dt;
    if(aggro&&P){const dx=P.x-d.x,dy=P.y-d.y,dist=Math.hypot(dx,dy);
      if(dist<95*S){d.x+=dx/dist*340*dt;d.y+=dy/dist*340*dt;}
      if(dist<P.r+13){collectDrop(d);drops.splice(i,1);continue;}}
    d.life=(d.life||8)-dt;if(d.life<=0)drops.splice(i,1);}

  // player
  if(state==='playing'&&P){
    P.invuln-=dt;P.hurtFlash-=dt;P.dashCd-=dt;P.muzzle-=dt;P.recoil=Math.max(0,P.recoil-22*dt);
    const tx=((keys.KeyD||keys.ArrowRight?1:0)-(keys.KeyA||keys.ArrowLeft?1:0))*275*S;
    const ty=((keys.KeyS||keys.ArrowDown?1:0)-(keys.KeyW||keys.ArrowUp?1:0))*275*S;
    if(P.dashT>0){P.dashT-=dt;P.vx=P.dashDx*840*S;P.vy=P.dashDy*840*S;
      parts.push({t:'ghost',x:P.x,y:P.y,life:.3,max:.3,size:P.r});}
    else{P.vx+=(tx-P.vx)*Math.min(1,dt*12);P.vy+=(ty-P.vy)*Math.min(1,dt*12);}
    P.x=clamp(P.x+P.vx*dt,P.r,W-P.r);P.y=clamp(P.y+P.vy*dt,P.r,H-P.r);
    P.aim=Math.atan2(my-P.y,mx-P.x);
    tryFire(dt);
    if(P.hp<30){hbT-=rdt;if(hbT<=0){hbT=.9-(P.hp/30)*.4;sfx.thump();}}
  }

  // wave flow
  if(state==='playing'){
    if(phase==='banner'){bannerT-=dt;if(bannerT<=0)phase='wave';}
    else if(phase==='wave'){
      spawnT-=dt;
      if(toSpawn>0&&spawnT<=0){
        const k=Math.min(toSpawn,1+(Math.random()<.35?1:0)+(wave>=6&&Math.random()<.25?1:0));
        for(let i=0;i<k;i++)spawnZombie(rollType());
        toSpawn-=k;spawnT=Math.max(.3,1.45-wave*.07)*(.55+Math.random()*.9);}
      if(toSpawn===0&&zombies.length===0){
        const b=wave*25;score+=b;P.hp=Math.min(P.maxhp,P.hp+12);
        addText(W/2,H/2-64,'+'+b+' BONUS','#d8a13c',20);
        addText(W/2,H/2-38,'+12 VITALITY','#8fd05f',15);
        phase='clear';clearT=1.7;sfx.pickup(true);}
    }
    else if(phase==='clear'){clearT-=dt;if(clearT<=0)startWave(wave+1);}
  }
  else if(state==='dying'){dieT-=rdt;if(dieT<=0)showOver();}
}
