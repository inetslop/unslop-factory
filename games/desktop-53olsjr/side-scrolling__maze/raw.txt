  if(picked.length<TOTAL_SHARDS){for(const c of [...dead].sort((a,b)=>b.d-a.d)){
    if(picked.length>=TOTAL_SHARDS)break;if(!picked.includes(c))picked.push(c);}}
  G.shards=picked.map(c=>({bx:c.bx,by:c.by,x:(c.bx+.5)*CS,y:(c.by+.5)*CS,ph:rnd()*TAU,taken:false}));
  // wraiths
  const cands=[];
  for(let by=1;by<BH-1;by++)for(let bx=1;bx<BW-1;bx++){
    const id=by*BW+bx;if(g[id]!==0)continue;
    let n=0;if(g[id-1]===0)n++;if(g[id+1]===0)n++;if(g[id-BW]===0)n++;if(g[id+BW]===0)n++;
    if(n<=1||d[id]<14||d[id]>maxD-8)continue;
    if(Math.abs(bx-(BW-2))+Math.abs(by-13)<6)continue;
    if(G.shards.some(s=>Math.abs(s.bx-bx)+Math.abs(s.by-by)<4))continue;
    cands.push({bx,by});
  }
  for(let i=cands.length-1;i>0;i--){const j=(rnd()*(i+1))|0;[cands[i],cands[j]]=[cands[j],cands[i]];}
  const wr=[];let gap=10;
  while(wr.length<5&&gap>=2){for(const c of cands){if(wr.length>=5)break;
    if(wr.every(p=>Math.abs(p.bx-c.bx)+Math.abs(p.by-c.by)>=gap))wr.push(c);}gap-=4;}
  G.wraiths=wr.slice(0,5).map(c=>({bx:c.bx,by:c.by,x:(c.bx+.5)*CS,y:(c.by+.5)*CS,
    tx:0,ty:0,lx:0,ly:0,hunt:0,seed:rnd()*10}));
  for(const w of G.wraiths)chooseNext(w);
  return d;
}
function bfs(g,sx,sy){const d=new Int32Array(BW*BH).fill(-1);d[sy*BW+sx]=0;
  const q=[[sx,sy]];let i=0;
  while(i<q.length){const [x,y]=q[i++],dd=d[y*BW+x]+1;
    for(const[dx,dy]of DIRS){const nx=x+dx,ny=y+dy;
      if(nx<0||ny<0||nx>=BW||ny>=BH)continue;const id=ny*BW+nx;
      if(g[id]===0&&d[id]<0){d[id]=dd;q.push([nx,ny]);}}}
  return d;}

/* ============================== PHYSICS ============================== */
function solid(bx,by){return bx<0||by<0||bx>=BW||by>=BH?true:G.grid[by*BW+bx]===1;}
function moveBody(b,dx,dy){
  const steps=Math.max(1,Math.ceil(Math.hypot(dx,dy)/8));
  for(let i=0;i<steps;i++){
    const sx=dx/steps,sy=dy/steps;
    if(sx!==0){b.x+=sx;resolveAxis(b,'x',sx);}
    if(sy!==0){b.y+=sy;resolveAxis(b,'y',sy);}
  }
}
function resolveAxis(b,axis,sign){
  const x0=Math.floor((b.x-b.r)/CS),x1=Math.floor((b.x+b.r)/CS);
  const y0=Math.floor((b.y-b.r)/CS),y1=Math.floor((b.y+b.r)/CS);
  let hit=false,hitB=axis==='x'?-999:999;
  for(let by=y0;by<=y1;by++)for(let bx=x0;bx<=x1;bx++){
    if(!solid(bx,by))continue;hit=true;
    const id=by*BW+bx;
    hitB=Math.min(Math.max(hitB,id),axis==='x'?999:999); // placeholder replaced below
  }
  // recompute properly for the leading edge
  if(!hit)return;
  let bestEdge=axis==='x'?(sign>0?-1e9:1e9):(sign>0?-1e9:1e9);
  for(let by=y0;by<=y1;by++)for(let bx=x0;bx<=x1;bx++){
    if(!solid(bx,by))continue;
    if(axis==='x')bestEdge=sign>0?Math.max(bestEdge,bx):Math.min(bestEdge,bx);
    else bestEdge=sign>0?Math.max(bestEdge,by):Math.min(bestEdge,by);
  }
  if(axis==='x'){if(sign>0)b.x=bestEdge*CS-b.r-.01;else b.x=(bestEdge+1)*CS+b.r+.01;}
  else{if(sign>0)b.y=bestEdge*CS-b.r-.01;else b.y=(bestEdge+1)*CS+b.r+.01;}
}

/* ============================== WRAITHS ============================== */
function chooseNext(w){
  const bx=Math.floor(w.x/CS),by=Math.floor(w.y/CS);
  let opts=[];
  for(const[dx,dy]of DIRS){const nx=bx+dx,ny=by+dy;
    if(nx<0||ny<0||nx>=BW||ny>=BH)continue;if(G.grid[ny*BW+nx]===0)
      opts.push({dx,dy,x:(nx+.5)*CS,y:(ny+.5)*CS});}
  if(!opts.length)return;
  let pick=null;
  if(w.hunt>0.5&&rnd()<0.78){
    const p=G.player;let bs=1e12;
    for(const o of opts){const d2=(o.x-p.x)**2+(o.y-p.y)**2;if(d2<bs){bs=d2;pick=o;}}
  }else{
    const straight=opts.find(o=>o.dx===w.lx&&o.dy===w.ly);
    if(straight&&rnd()<0.6)pick=straight;
    else{let pool=opts.length>1?opts.filter(o=>!(o.dx===-w.lx&&o.dy===-w.ly)):opts;
      pick=pool[(rnd()*pool.length)|0];}
  }
  if(!pick)pick=opts[(rnd()*opts.length)|0];
  w.lx=pick.dx;w.ly=pick.dy;w.tx=pick.x;w.ty=pick.y;
}
function updateWraiths(dt,huntAllowed){
  const p=G.player;let nearest=1e9;
  for(const w of G.wraiths){
    const sp=w.hunt>0.3?188:92;
    let dx=w.tx-w.x,dy=w.ty-w.y,d=Math.hypot(dx,dy),step=sp*dt;
    if(d<=step||d<0.001){w.x=w.tx;w.y=w.ty;chooseNext(w);}
    else{w.x+=dx/d*step;w.y+=dy/d*step;}
    const pd=Math.hypot(p.x-w.x,p.y-w.y);nearest=Math.min(nearest,pd);
    if(huntAllowed&&pd<CS*7.5)w.hunt=Math.min(1,w.hunt+dt*3);
    else w.hunt=Math.max(0,w.hunt-dt*2);
  }
  if(G.state==='playing'){
    const target=nearest<CS*8?(1-nearest/(CS*8))*.55:0;
    G.threat+=(target-G.threat)*Math.min(1,dt*4);
    if(G.threat>0.25&&G.t>G.nextThump){AU.thump(.06+G.threat*.3);
      G.nextThump=G.t+lerp(1.1,.42,G.threat);}
  }else G.threat*=Math.exp(-dt*2);
}

function updateParts(dt){

    const dr=Math.exp(-dt*p.drag);p.vx*=dr;p.vy=p.vy*dr+p.grav*dt;}
  if(G.parts.length>320)G.parts.splice(0,G.parts.length-320);
}

/* ============================== UPDATE ============================== */
function update(dt){
  G.t+=dt;
  if(G.t>G.nextDrip){AU.drip();G.nextDrip=G.t+4+rnd()*7;}
  switch(G.state){
    case 'menu':{
      updateWraiths(dt,false);updateParts(dt);
      const p=G.player,maxX=Math.max(0,WORLD_W-vw),maxY=Math.max(0,WORLD_H-vh);
      const tx=clamp(p.x-vw/2+Math.sin(G.t*.13)*160,0,maxX);
      const ty=clamp(p.y-vh/2+Math.sin(G.t*.09)*46,0,maxY);
      G.cam.x+=(tx-G.cam.x)*(1-Math.exp(-dt*1.5));
      G.cam.y+=(ty-G.cam.y)*(1-Math.exp(-dt*1.5));
      break;}
    case 'playing':{
      G.runTime+=dt;const p=G.player;
      const[ix,iy]=inputVec();let m=Math.hypot(ix,iy);if(m>1){ix/=m;iy/=m;}
      const speed=G.dash.t>0?375:235;
      if(G.dash.t>0)G.dash.t-=dt;
      G.dash.cd+=dt;if(G.dash.cd>=DASH_CD&&G.dash.charges<3){G.dash.charges++;G.dash.cd=0;AU.pip();syncPips();}
      const k=1-Math.exp(-dt*16);
      p.vx+=(ix*speed-p.vx)*k;p.vy+=(iy*speed-p.vy)*k;
      p.kx*=Math.exp(-dt*6);p.ky*=Math.exp(-dt*6);
      moveBody(p,(p.vx+p.kx)*dt,(p.vy+p.ky)*dt);
      if(Math.abs(p.vx)>20)p.face=p.vx>0?1:-1;
      p.nextBlink-=dt;if(p.nextBlink<0){p.blink=.14;p.nextBlink=2+rnd()*3;}p.blink-=dt;
      if(p.inv>0)p.inv-=dt;
      const sp=Math.hypot(p.vx,p.vy);
      if(sp>60||G.dash.t>0){
        const n=G.dash.t>0?3:1;
        for(let i=0;i<n;i++)G.parts.push({x:p.x+(rnd()-.5)*8,y:p.y+(rnd()-.5)*8,
          vx:-p.vx*.15+(rnd()-.5)*20,vy:-p.vy*.15+(rnd()-.5)*20,life:0,max:.4+rnd()*.3,
          s:G.dash.t>0?9+rnd()*8:5+rnd()*5,drag:2.6,grav:0,a:.8,ring:false});}
      // shards
      for(const s of G.shards){if(s.taken)continue;
        const d=Math.hypot(p.x-s.x,p.y-s.y);
        if(d<26){s.taken=true;G.got++;AU.shard();syncShards();
          for(let i=0;i<14;i++)spawnPart(s.x,s.y,60+rnd()*150,rnd()*TAU,.4+rnd()*.35,5+rnd()*8);
          G.parts.push({x:s.x,y:s.y,vx:0,vy:0,life:0,max:.5,s:1,drag:0,grav:0,a:.9,ring:true});
          if(G.got>=TOTAL_SHARDS&&!G.unlocked){G.unlocked=true;AU.unlock();G.shake=4;
            addToast('The gate stirs — reach the burning light');
            G.parts.push({x:G.gate.x,y:G.gate.y,vx:0,vy:0,life:0,max:.8,s:1,drag:0,grav:0,a:1,ring:true});}}}
      // wraiths + collision
      updateWraiths(dt,true);
      for(const w of G.wraiths){if(p.inv>0)continue;
        const dx=p.x-w.x,dy=p.y-w.y,d=Math.hypot(dx,dy);
        if(d<22&&d>0.001){damage(dx/d,dy/d);break;}}
      // gate
      const g=G.gate,gd=Math.hypot(p.x-g.x,p.y-g.y);
      if(gd<34){
        if(G.unlocked)winGame();
        else if(G.t-G.lastCold>5){G.lastCold=G.t;AU.knock();
          addToast('The gate is cold — '+(TOTAL_SHARDS-G.got)+' shards remain');}}
      if(G.unlocked&&rnd()<.6)G.parts.push({x:g.x+(rnd()-.5)*16,y:g.y+8,vx:(rnd()-.5)*20,
        vy:-30-rnd()*45,life:0,max:.8+rnd()*.5,s:3+rnd()*5,drag:1,grav:-14,a:.9,ring:false});
      // camera
      const lookX=clamp(p.vx*.28,-90,90),lookY=clamp(p.vy*.22,-70,70);
      const maxX=Math.max(0,WORLD_W-vw),maxY=Math.max(0,WORLD_H-vh);
      const tx=clamp(p.x+lookX-vw/2,0,maxX),ty=clamp(p.y+lookY-vh/2,0,maxY);
      const ck=1-Math.exp(-dt*7);
      G.cam.x+=(tx-G.cam.x)*ck;G.cam.y+=(ty-G.cam.y)*ck;
      break;}
    case 'dying':{
      G.deathT+=dt;updateParts(dt);
      if(G.deathT>1.2){setState('dead');showScreen('dead');
        $('deadStats').textContent=`${fmtTime(G.runTime)} ELAPSED · ${G.got} / ${TOTAL_SHARDS} SHARDS`;}}
      break;
    case 'win':{
      updateParts(dt);
      const g=G.gate,maxX=Math.max(0,WORLD_W-vw),maxY=Math.max(0,WORLD_H-vh);
      const tx=clamp(g.x-vw/2,0,maxX),ty=clamp(g.y-vh/2,0,maxY);
      G.cam.x+=(tx-G.cam.x)*(1-Math.exp(-dt*3));G.cam.y+=(ty-G.cam.y)*(1-Math.exp(-dt*3));
      if(rnd()<.8)G.parts.push({x:g.x+(rnd()-.5)*20,y:g.y+10,vx:(rnd()-.5)*26,
        vy:-40-rnd()*60,life:0,max:.9+rnd()*.6,s:3+rnd()*6,drag:1,grav:-18,a:1,ring:false});}
      break;
    case 'dead':updateParts(dt);break;
    case 'paused':break;
  }
  G.shake*=Math.exp(-dt*5);G.flash*=Math.exp(-dt*3.2);
  updateDust(dt);
}
function updateDust(dt){
  const m=140,cx=G.cam.x,cy=G.cam.y;
  for(const d of G.dust){d.y-=d.sp*dt;d.x+=Math.sin(G.t*.6+d.ph)*dt*8;
    if(d.x<cx-m||d.x>cx+vw+m||d.y<cy-m||d.y>cy+vh+m){
      d.x=cx+rnd()*(vw+2*m)-m;d.y=cy+rnd()*(vh+2*m)-m;}}
}

  const y0=Math.max(0,Math.floor(camY/CS)-1),y1=Math.min(BH-1,Math.ceil((camY+vh)/CS)+1);
  for(let by=y0;by<=y1;by++)for(let bx=x0;bx<=x1;bx++){
    const id=by*BW+bx,px=bx*CS,py=by*CS;
    if(G.grid[id]===1){
      ctx.fillStyle=WALLS[G.wallShade[id]&3];ctx.fillRect(px,py,CS,CS);
      ctx.fillStyle=RIM;
      if(bx>0&&G.grid[id-1]===0)ctx.fillRect(px,py,2,CS);
      if(bx<BW-1&&G.grid[id+1]===0)ctx.fillRect(px+CS-2,py,2,CS);
      if(by>0&&G.grid[id-BW]===0)ctx.fillRect(px,py,CS,2);
      if(by<BH-1&&G.grid[id+BW]===0)ctx.fillRect(px,py+CS-2,CS,2);
    }else{
      ctx.fillStyle=FLOORS[G.floorShade[id]&3];ctx.fillRect(px,py,CS,CS);
    }
  }
  for(const d of G.decor){
    if(d.x<camX-CS||d.x>camX+vw+CS||d.y<camY-CS||d.y>camY+vh+CS)continue;
    if(d.type===0){ctx.fillStyle='rgba(70,190,175,.10)';ctx.beginPath();
      ctx.ellipse(d.x,d.y,d.rx,d.ry,d.rot,0,TAU);ctx.fill();}

  }
}
