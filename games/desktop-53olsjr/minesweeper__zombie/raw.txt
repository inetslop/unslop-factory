    sg.addColorStop(0,`rgba(255,170,80,${.5*d})`);sg.addColorStop(1,'rgba(255,170,80,0)');
    bctx.fillStyle=sg;bctx.fillRect(0,0,W,H);}
  // ground + silhouettes
  const gy=H-H*.09;
  bctx.fillStyle='#04080a';bctx.fillRect(0,gy,W,H-gy);
  bctx.fillStyle='rgba(120,255,170,.06)';bctx.fillRect(0,gy,W,1.5);
  const drawTomb=(tb,cy,col)=>{
    bctx.fillStyle=col;bctx.save();bctx.translate(tb.x,cy);
    if(tb.t===0){bctx.beginPath();bctx.moveTo(0,0);bctx.lineTo(0,-tb.h*.62);bctx.arc(tb.w/2,-tb.h*.62,tb.w/2,Math.PI,0);bctx.lineTo(tb.w,0);bctx.closePath();bctx.fill();}
    else if(tb.t===1){const vw=tb.w*.34;bctx.fillRect((tb.w-vw)/2,-tb.h,vw,tb.h);bctx.fillRect(0,-tb.h*.72,tb.w,tb.h*.2);}
    else{bctx.rotate(.09);bctx.fillRect(0,-tb.h*.85,tb.w,tb.h*.85);}
    bctx.restore();
  };
  farTomb.forEach(tb=>drawTomb(tb,gy-3,'#0a1216'));
  trees.forEach(segs=>{bctx.strokeStyle='#050a0c';bctx.lineCap='round';segs.forEach(s=>{bctx.lineWidth=s.w;bctx.beginPath();bctx.moveTo(s.x1,s.y1);bctx.lineTo(s.x2,s.y2);bctx.stroke();});});
  nearTomb.forEach(tb=>drawTomb(tb,gy+H*.035,'#04080a'));
  // fog
  const fr=hexLerp('#96d7b4','#ffbe85',d).slice(4,-1); // "r,g,b"
  for(const f of fogBlobs){
    f.x+=f.v*dt;if(f.x<-f.r)f.x=W+f.r;if(f.x>W+f.r)f.x=-f.r;
    const fy=f.y+Math.sin(t*.0002+f.a*40)*8;
    let fg=bctx.createRadialGradient(f.x,fy,1,f.x,fy,f.r);
    fg.addColorStop(0,`rgba(${fr},${f.a})`);fg.addColorStop(1,`rgba(${fr},0)`);
    bctx.fillStyle=fg;bctx.fillRect(f.x-f.r,fy-f.r,f.r*2,f.r*2);
  }
  // spores
  const sr=hexLerp('#9fe8b0','#ffcf8a',d).slice(4,-1);
  for(const s of spores){
    s.y-=s.vy*dt;if(s.y<-6){s.y=H+6;s.bx=rand(0,W);}
    const sx=s.bx+Math.sin(t*.001*s.sp+s.ph)*s.amp;
    const tw=.3+.5*Math.abs(Math.sin(t*.002*s.sp+s.ph));
    bctx.globalAlpha=tw;bctx.fillStyle=`rgb(${sr})`;bctx.beginPath();bctx.arc(sx,s.y,s.r,0,7);bctx.fill();
  }
  bctx.globalAlpha=1;
  // lightning
  if(d<.4&&t>nextBolt){flashA=1;flickerAt=t+rand(90,180);nextBolt=t+rand(7000,16000);sfx.thunder();}
  if(t>flickerAt&&flashA>.2)flashA=.65;
  flashA=Math.max(0,flashA-dt*3.2);
  if(flashA>0){bctx.fillStyle=`rgba(210,230,255,${flashA*.16})`;bctx.fillRect(0,0,W,H);}
}

/* ================= fx particles ================= */
let parts=[];
function burst(x,y,type,n){
  for(let i=0;i<n;i++){
    let p;
    if(type==='dirt')p={x,y,vx:rand(-60,60),vy:rand(-150,-30),g:520,size:rand(1.5,3.5),life:rand(.35,.7),c:['#6b4e33','#4a3624','#7d5c3d'][irand(0,2)],glow:false};
    else if(type==='goo'){const a=rand(0,6.28),sp=rand(40,260);p={x,y,vx:Math.cos(a)*sp,vy:Math.sin(a)*sp-60,g:600,size:rand(2,5),life:rand(.5,.95),c:['#5fae4f','#7cd36a','#3f8a3c','#8dffab'][irand(0,3)],glow:false};}
    else p={x,y,vx:rand(-22,22),vy:rand(-95,-30),g:-30,size:rand(1.5,3),life:rand(.6,1.2),c:['#ffb347','#ff8f3d','#ffd97a'][irand(0,2)],glow:true};
    p.max=p.life;parts.push(p);
  }
  if(parts.length>600)parts.splice(0,parts.length-600);
}
function drawFX(dt){
  fctx.clearRect(0,0,W,H);
  for(let i=parts.length-1;i>=0;i--){
    const p=parts[i];p.life-=dt;
    if(p.life<=0){parts.splice(i,1);continue;}
    p.vy+=p.g*dt;p.x+=p.vx*dt;p.y+=p.vy*dt;
    const a=p.life/p.max;
    fctx.globalCompositeOperation=p.glow?'lighter':'source-over';
    fctx.globalAlpha=a;fctx.fillStyle=p.c;
    fctx.beginPath();fctx.arc(p.x,p.y,p.size*(p.glow?a+.3:1),0,7);fctx.fill();
  }
  fctx.globalAlpha=1;fctx.globalCompositeOperation='source-over';
}

  const dt=Math.min(.05,(t-lastT)/1000);lastT=t;
  drawBG(t,dt);drawFX(dt);requestAnimationFrame(loop);
}

/* ================= board logic ================= */
const board=$('#board'),boardWrap=$('#boardWrap');
function faceOf(el){return el.querySelector('.face');}
function nbrs(i){
  const r=(i/cols)|0,c=i%cols,out=[];
  for(let dr=-1;dr<=1;dr++)for(let dc=-1;dc<=1;dc++){
    if(!dr&&!dc)continue;const rr=r+dr,cc=c+dc;
    if(rr>=0&&rr<rows&&cc>=0&&cc<cols)out.push(rr*cols+cc);
  }return out;
}
function fit(){
  if(!rows||!cols)return;
  const aw=boardWrap.clientWidth-10,ah=boardWrap.clientHeight-10,gap=2,pad=12;
  const c=Math.floor(Math.min((aw-pad*2-(cols-1)*gap)/cols,(ah-pad*2-(rows-1)*gap)/rows));
  board.style.setProperty('--cell',clamp(c,9,54)+'px');
  board.style.setProperty('--cols',cols);
}
function newGame(key){
  if(key)selectDiff(key);
  diffKey=selected;const d=DIFFS[diffKey];
  rows=d.rows;cols=d.cols;mines=d.mines;
  grid=[];for(let i=0;i<rows*cols;i++)grid.push({mine:false,adj:0,state:'hidden',flagged:false});
  minesPlaced=false;revealedCount=0;flagCount=0;digs=0;elapsed=0;kIdx=-1;kbActive=false;
  stopTimer();document.body.classList.remove('dawn');
  board.innerHTML='';cells=[];
  for(let i=0;i<rows*cols;i++){
    const el=document.createElement('div');
    el.className='cell hidden in';el.dataset.i=i;
    el.style.animationDelay=((i%cols)+((i/cols)|0))*3+'ms';
    el.innerHTML='<span class="face"></span>';
    board.appendChild(el);cells[i]=el;
  }
  phase='play';
  hide($('#menu'));hide($('#over'));
  fit();updateHUD();
}
function placeMines(safe){
  const forbid=new Set([safe,...nbrs(safe)]),pool=[];
  for(let i=0;i<rows*cols;i++)if(!forbid.has(i))pool.push(i);
  for(let k=0;k<mines&&k<pool.length;k++){const j=k+irand(0,pool.length-1-k);[pool[k],pool[j]]=[pool[j],pool[k]];}
  for(let k=0;k<Math.min(mines,pool.length);k++)grid[pool[k]].mine=true;
  for(let i=0;i<rows*cols;i++){let n=0;for(const m of nbrs(i))if(grid[m].mine)n++;grid[i].adj=n;}
  minesPlaced=true;
}
function revealSet(list){
  const dist=new Map();
  for(const i of list){const g=grid[i];if(g.state==='hidden'&&!g.flagged&&!g.mine)dist.set(i,0);}
  const arr=[...dist.keys()];let head=0;
  while(head<arr.length){
    const i=arr[head++];
    if(grid[i].adj===0){
      for(const n of nbrs(i)){
        const ng=grid[n];
        if(ng.state==='hidden'&&!ng.flagged&&!ng.mine&&!dist.has(n)){dist.set(n,dist.get(i)+1);arr.push(n);}
      }
    }
  }
  for(const [i,dd] of dist){
    const g=grid[i];g.state='revealed';
    const el=cells[i],f=faceOf(el);
    f.innerHTML=g.adj?`<span class="num" style="color:${NUMCOLORS[g.adj-1]}">${g.adj}</span>`:'';
    const delay=Math.min(dd,18)*15;
    if(delay<20){el.classList.remove('hidden');el.classList.add('revealed');}
    else{el.classList.add('pending');setTimeout(()=>{el.classList.remove('hidden','pending');el.classList.add('revealed');},delay);}
  }
  revealedCount+=dist.size;
}
function doDig(i){
  if(phase!=='play'||i<0||i>=rows*cols)return;
  const g=grid[i];
  if(g.state==='hidden'){
    if(g.flagged)return;
    if(!minesPlaced){placeMines(i);startTimer();}
    if(g.mine){lose(i);return;}
    digs++;sfx.pop();
    const r=cells[i].getBoundingClientRect();burst(r.left+r.width/2,r.top+r.height/2,'dirt',4);
    revealSet([i]);
  }else if(g.state==='revealed'&&g.adj>0){chord(i);}
  checkWin();
}
function chord(i){
  const g=grid[i];if(g.state!=='revealed'||g.adj===0)return;
  const ns=nbrs(i);
  const flagged=ns.filter(n=>grid[n].flagged).length;
  if(flagged!==g.adj)return;
  const targets=ns.filter(n=>grid[n].state==='hidden'&&!grid[n].flagged);
  if(!targets.length)return;
  const boom=targets.find(n=>grid[n].mine);
  if(boom){lose(boom);return;}
  digs++;sfx.sweep();revealSet(targets);checkWin();
}
function toggleFlag(i){
  if(phase!=='play')return;
  const g=grid[i];if(g.state!=='hidden')return;
  const el=cells[i],f=faceOf(el);
  if(!g.flagged){g.flagged=true;flagCount++;el.classList.add('flag');f.innerHTML=STAKE;sfx.thock();}
  else{g.flagged=false;flagCount--;el.classList.remove('flag');f.innerHTML='';sfx.unflag();}
  updateHUD();
}
function checkWin(){if(phase==='play'&&revealedCount===rows*cols-mines)win();}
function lose(i){
  phase='lost';stopTimer();
  document.body.classList.add('shake');setTimeout(()=>document.body.classList.remove('shake'),650);
  const fl=$('#flash');fl.classList.add('show');setTimeout(()=>fl.classList.remove('show'),900);
  grid[i].state='revealed';
  const el=cells[i];el.classList.remove('hidden','flag','pending');el.classList.add('revealed','boom');
  faceOf(el).innerHTML=ZBOOM;
  sfx.death();
  const r=el.getBoundingClientRect();burst(r.left+r.width/2,r.top+r.height/2,'goo',30);
  // staggered zombie reveal by distance from the bite
  const dm=new Map([[i,0]]),arr=[i];let head=0;
  while(head<arr.length){const c=arr[head++];for(const n of nbrs(c))if(!dm.has(n)){dm.set(n,dm.get(c)+1);arr.push(n);}}
  for(let m=0;m<rows*cols;m++){
    if(m===i)continue;const g=grid[m];
    if(g.mine){g.state='revealed';
      setTimeout(()=>{const e2=cells[m];e2.classList.remove('hidden','flag','pending');e2.classList.add('revealed','mine');faceOf(e2).innerHTML=ZCALM;},Math.min(dm.get(m),20)*45);
    }else if(g.flagged){setTimeout(()=>cells[m].classList.add('wrong'),250+irand(0,350));}
  }
  setTimeout(()=>showEnd(false,false),1300);
}
function win(){
  phase='won';stopTimer();document.body.classList.add('dawn');
  let rec=false;
  if(!best[diffKey]||elapsed<best[diffKey]){best[diffKey]=elapsed;rec=true;persistBest();}
  for(let m=0;m<rows*cols;m++){
    if(grid[m].mine){
      setTimeout(()=>{const e2=cells[m];e2.classList.remove('hidden','flag','pending');e2.classList.add('revealed','mine','burn');faceOf(e2).innerHTML=ZCALM;
        const r=e2.getBoundingClientRect();burst(r.left+r.width/2,r.top+r.height/2,'ember',8);},irand(0,900));
    }
  }
  sfx.win();refreshBest();setTimeout(()=>showEnd(true,rec),1500);
}
