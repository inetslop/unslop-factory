  if(['ArrowUp','ArrowDown','Space'].includes(e.code)) e.preventDefault();
  firstInput(); keys[e.code]=true; if(e.repeat) return;
  switch(e.code){
    case 'Enter':
      if(mode==='menu'||mode==='over') startGame(); break;
    case 'KeyR':
      if(mode!=='menu') startGame(); break;
    case 'KeyP':
      if(mode==='play') pauseGame(); else if(mode==='pause') resumeGame(); break;
    case 'Escape':
      if(mode==='play') pauseGame(); else if(mode==='pause') resumeGame();
      else if(mode==='over') toMenu(); break;
    case 'KeyM':{
      muted=!muted; try{ localStorage.setItem('sp_mute',muted?'1':'0'); }catch(_){}
      if(master) master.gain.value=muted?0:.5; toast(muted?'SOUND OFF':'SOUND ON'); break; }
  }
});

/* ============================== CONTROLS / AI ============================== */
function humanControl(pad,dt){
  const maxV=H*1.05; let dir=0;
  if(keys.KeyW||keys.ArrowUp) dir-=1; if(keys.KeyS||keys.ArrowDown) dir+=1;
  let step;
  if(mouseY!=null && tGlobal-mouseT<.3){ const d=clamp(mouseY-pad.y,-maxV*dt,maxV*dt); step=d; }
  else step=dir*maxV*dt;
  pad.vy=dt>0?step/dt:0; pad.y+=step; clampPad(pad);
}
function predictY(x,y,vx,vy,tx){ if(Math.abs(vx)<1)return H/2;
  const t=(tx-x)/vx; if(t<0||t>3) return y;
  const yy=y+vy*t, lo=M.br, hi=H-M.br, span=hi-lo;
  let m=((yy-lo)%(2*span)+2*span)%(2*span);
  return lo+(m<=span?m:2*span-m); }
function aiControl(pad,dt){
  const b=G.ball, isLeft=pad.side==='p'; let target;
  if(G.serveT>0||G.end) target=H/2;
  else{ const movingIn=isLeft?b.vx<0:b.vx>0;
    if(movingIn){ const faceX=isLeft?pad.x+pad.w:pad.x-pad.w; target=predictY(b.x,b.y,b.vx,b.vy,faceX)+pad.aiErr; }
    else target=H*.5+(b.y-H*.5)*.3+Math.sin(tGlobal*1.3+(isLeft?0:2))*M.ph*.18; }
  const diff=mode==='menu'?1:.72+Math.min(.5,(G.sp+G.sc)*.05);
  const maxV=H*(.42+.30*diff), dy=target-pad.y, step=clamp(dy,-maxV*dt,maxV*dt);
  pad.vy=dt>0?step/dt:0; pad.y+=step; clampPad(pad);
  if(Math.random()<dt*.5) pad.aiErr=(Math.random()-.5)*M.ph*(.24-.1*Math.min(1,diff));
}
function fire(pad,tx,ty){
  if(pad.cd>0||pad.dead) return;
  pad.cd=pad.power===2?.3:.24;
  const sp=Math.max(760,W*.75);
  const mx=pad.side==='p'?pad.x+pad.w:pad.x;
  const my=pad.y+clamp((pad.vy||0)*.08,-30,30);
  const angs=[];
  if(pad.power===1) angs.push(-.17,0,.17);
  else{ let a=(tx!=null)?Math.atan2(ty-my,tx-mx):(pad.side==='p'?0:Math.PI);
        a+=clamp((pad.vy||0)*.0005,-.12,.12); angs.push(a); }
  for(const a of angs){
    G.bullets.push({x:mx,y:my+(angs.length>1?a*40:0),vx:Math.cos(a)*sp,vy:Math.sin(a)*sp,
      owner:pad.side,r:pad.power===2?6.5*S+2:3.6,dmg:pad.power===2?2:1,life:2.5}); }
  if(pad.side==='p') G.stats.shots+=angs.length;
  spark(mx,my,pad.side==='p'?COL.p:COL.c,3,220,.25,2.4);
  sFire();
}
function aiFire(pad,dt){
  if(pad.cd>0||pad.dead) return;
  const b=G.ball, isLeft=pad.side==='p';
  const muzzleX=isLeft?pad.x+pad.w:pad.x;
  const incoming=(isLeft?b.vx<0:b.vx>0)&&G.serveT<=0&&!G.end;
  if(incoming&&b.speed>M.baseSpeed*1.2){
    const sp=Math.max(760,W*.75), d=Math.abs(b.x-muzzleX)/sp;
    fire(pad,b.x+b.vx*d,b.y+b.vy*d); pad.cd=rand(.45,.8); return; }
  if(Math.random()<dt*.55){
    const foe=isLeft?G.c:G.p;
    fire(pad,foe.x+(isLeft?20:-20),foe.y+rand(-.4,.4)*foe.h); pad.cd=rand(.9,1.5); }
}

/* ============================== COMBAT / SCORING ============================== */
function pointInPaddle(x,y,p,r){ return x>=p.x-r&&x<=p.x+p.w+r&&y>=p.y-p.h/2-r&&y<=p.y+p.h/2+r; }
function damagePaddle(pad,dmg,x,y){
  pad.hp-=dmg; pad.hurtT=.35;
  spark(x,y,COL.red,14+dmg*8,380,.5,3); ring(x,y,pad.side==='p'?COL.p:COL.c,6,420);
  G.shake=Math.max(G.shake,4+dmg*2.5); sHit(dmg);
  pushText(pad.x+(pad.side==='p'?-46:46),pad.y,'-'+dmg,COL.red,Math.round(18*S+8),.7);
  if(pad.hp<=0){
    if(mode==='menu'){ pad.hp=pad.maxHp; spark(x,y,pad.side==='p'?COL.p:COL.c,30,420,.6,3); return; }
    destroySide(pad); }
}
function destroySide(pad){
  pad.dead=true;
  const cx=pad.x+pad.w/2;
  spark(cx,pad.y,pad.side==='p'?COL.p:COL.c,50,560,.9,4);
  spark(cx,pad.y,'#ffffff',24,420,.7,3); ring(cx,pad.y,pad.side==='p'?COL.p:COL.c,10,900);
  G.shake=18; sBoom();
  pushText(W/2,H*.4,'PADDLE DESTROYED',COL.red,Math.round(26*S+12),1.3);
  endMatch(pad.side==='p'?'c':'p','destroy');
}
function slowBall(bl){
  const b=G.ball; const ns=Math.max(b.speed*.8,M.baseSpeed);
  if(ns<b.speed){ const k=ns/b.speed; b.vx*=k; b.vy*=k; b.speed=ns; }
  G.stats.slow++; spark(b.x,b.y,COL.mint,10,260,.4,2.5);
  pushText(b.x,b.y-18,'SLOWED',COL.mint,Math.round(13*S+7),.8); sSlow(); G.shake=Math.max(G.shake,2);
}
function grantPower(side,type,cx,cy){
  const pad=side==='p'?G.p:G.c; const fresh=pad.power!==type;
  pad.power=type; pad.powerT=7;
  pushText(cx,cy-20,type===1?'TRI-SHOT!':'HEAVY ROUNDS!',type===1?COL.mint:COL.gold,Math.round(18*S+9),1.2);
  ring(cx,cy,type===1?COL.mint:COL.gold,8,520); spark(cx,cy,COL.mint,16,320,.5,2.6); sPower();
}
function hitPaddle(pad,dir){
  const b=G.ball; G.rallyHits++;
  const rel=clamp((b.y-pad.y)/(pad.h/2+M.br),-1,1), ang=rel*.96;
  const sp=Math.min(b.speed*1.05,M.baseSpeed*1.95);
  b.vx=Math.cos(ang)*sp*dir; b.vy=Math.sin(ang)*sp; b.speed=sp;
  b.x=dir>0?pad.x+pad.w+M.br:pad.x-M.br; b.tint=pad.side; pad.hitT=.18;
  spark(b.x,b.y,pad.side==='p'?COL.p:COL.c,10,320,.4,2.6);
  G.shake=Math.max(G.shake,1.5+G.rallyHits*.25); sPaddle(sp/M.baseSpeed);
}
function wallFx(b){ spark(b.x,b.y,'#bfe6f5',6,220,.3,2); G.shake=Math.max(G.shake,1); sWall(); }
function score(side){
  if(G.end) return;
  if(side==='p') G.sp++; else G.sc++;
  const win=side==='p';
  G.bullets.length=0;
  G.shake=Math.max(G.shake,7); if(win) G.flashP=.5; else G.flashC=.5;
  pushText(W/2,H*.3,win?'POINT — YOU':'POINT — CPU',win?COL.p:COL.c,Math.round(24*S+13),1.1);
  sScore(win);
  if(G.sp>=WIN_SCORE||G.sc>=WIN_SCORE) endMatch(G.sp>=WIN_SCORE?'p':'c','points');
  else{ G.serveT=1.05; G.serveDir=win?-1:1; G.rallyHits=0; serveBall(G.serveDir); }
}

/* ============================== UPDATE ============================== */
function update(dt){
  G.shake=Math.max(0,G.shake-dt*34);
  G.flashP=Math.max(0,G.flashP-dt*2.2); G.flashC=Math.max(0,G.flashC-dt*2.2);
  if(G.end){ endUpdate(dt); return; }
  G.playT+=dt; if(G.serveT>0) G.serveT-=dt;

  const p=G.p,c=G.c,human=mode==='play';
  p.cd=Math.max(0,p.cd-dt); c.cd=Math.max(0,c.cd-dt);
  for(const pad of [p,c]){
    if(pad.hitT>0) pad.hitT-=dt; if(pad.hurtT>0) pad.hurtT-=dt;
    if(pad.powerT>0){ pad.powerT-=dt;
      if(pad.powerT<=0){ pad.power=0; pushText(pad.x+(pad.side==='p'?50:-50),pad.y,'POWER OFF','#9fb7c8',Math.round(11*S+6),.8); } }
  }
  if(human) humanControl(p,dt); else aiControl(p,dt);
  aiControl(c,dt);
  if(human&&(keys.Space||mouseDown)) fire(p,null,null);
  aiFire(c,dt); if(!human) aiFire(p,dt);

  if(G.serveT<=0) moveBall(dt);
  updateBullets(dt); updateCells(dt); updateParts(dt); updateTexts(dt);
}
function endUpdate(dt){
  G.end.t-=dt; updateParts(dt); updateTexts(dt);
  if(G.end.t<=0) doGameOver();
}
function moveBall(dt){
  const b=G.ball, px=b.x, py=b.y;
  b.x+=b.vx*dt; b.y+=b.vy*dt;
  if(b.y-M.br<0&&b.vy<0){ b.y=M.br; b.vy*=-1; wallFx(b); }
  else if(b.y+M.br>H&&b.vy>0){ b.y=H-M.br; b.vy*=-1; wallFx(b); }
  const p=G.p,c=G.c;
  if(b.vx<0){ const face=p.x+p.w;
    if(px-M.br>=face&&b.x-M.br<=face){ const dx=px-b.x;
      if(Math.abs(dx)>1e-4){ const tH=(px-face)/dx, yAt=py+(b.y-py)*tH;
        if(Math.abs(yAt-p.y)<=p.h/2+M.br) hitPaddle(p,1); } } }
  if(b.vx>0){ const face=c.x;
    if(px+M.br<=face&&b.x+M.br>=face){ const dx=b.x-px;
      if(Math.abs(dx)>1e-4){ const tH=(face-px)/dx, yAt=py+(b.y-py)*tH;
        if(Math.abs(yAt-c.y)<=c.h/2+M.br) hitPaddle(c,-1); } } }
  b.trail.push({x:b.x,y:b.y}); if(b.trail.length>16) b.trail.shift();
  G.stats.top=Math.max(G.stats.top,b.speed);
  if(b.x<-M.br*4) score('c'); else if(b.x>W+M.br*4) score('p');
}
function updateBullets(dt){
  for(let i=G.bullets.length-1;i>=0;i--){
    const bl=G.bullets[i]; bl.x+=bl.vx*dt; bl.y+=bl.vy*dt; bl.life-=dt;
    if(bl.x<-50||bl.x>W+50||bl.y<-50||bl.y>H+50||bl.life<=0){ G.bullets.splice(i,1); continue; }
    const enemy=bl.owner==='p'?G.c:G.p;
    if(!enemy.dead&&pointInPaddle(bl.x,bl.y,enemy,bl.r*.6)){
      if(bl.owner==='p') G.stats.hits++;
      damagePaddle(enemy,bl.dmg,bl.x,bl.y); G.bullets.splice(i,1); continue; }
    const b=G.ball;
    if(G.serveT<=0){ const dx=bl.x-b.x,dy=bl.y-b.y,r=M.br+bl.r;
      if(dx*dx+dy*dy<r*r){ slowBall(bl); G.bullets.splice(i,1); continue; } }
    for(let j=G.cells.length-1;j>=0;j--){ const c=G.cells[j], dx=bl.x-c.x,dy=bl.y-c.y,r=c.r+bl.r;
      if(dx*dx+dy*dy<r*r){ grantPower(bl.owner,c.type,c.x,c.y); G.cells.splice(j,1); G.bullets.splice(i,1); break; } }
  }
}
function updateCells(dt){
  if(tGlobal>G.nextCell&&G.cells.length<1){
    const type=Math.random()<.5?1:2;
    G.cells.push({x:rand(W*.3,W*.7),y:rand(H*.2,H*.8),r:clamp(16*S+4,14,26),rot:0,type,life:9});
    ring(rand(W*.3,W*.7),G.cells[0].y,COL.mint,4,300); G.nextCell=tGlobal+rand(8,15); }
  for(let i=G.cells.length-1;i>=0;i--){ const c=G.cells[i]; c.rot+=dt*2; c.life-=dt;
    if(c.life<=0){ spark(c.x,c.y,COL.mint,8,160,.4,2); G.cells.splice(i,1); } }
}
function updateParts(dt){
  for(let i=G.parts.length-1;i>=0;i--){ const p=G.parts[i]; p.life-=dt;
    if(p.ring){ p.r+=p.vr*dt; } else { p.x+=p.vx*dt; p.y+=p.vy*dt; p.vx*=(1-2.6*dt); p.vy*=(1-2.6*dt); }
    if(p.life<=0) G.parts.splice(i,1); }
}
function updateTexts(dt){
  for(let i=G.texts.length-1;i>=0;i--){ const t=G.texts[i]; t.y-=26*S*dt; t.life-=dt;
    if(t.life<=0) G.texts.splice(i,1); }
}

/* ============================== RENDER ============================== */
function render(){
  ctx.fillStyle=COL.bg; ctx.fillRect(0,0,W,H);
  // ambient side glows
  let g=ctx.createRadialGradient(0,H/2,0,0,H/2,W*.55);
  g.addColorStop(0,'rgba(255,180,84,'+(0.05+Math.sin(tGlobal*1.2)*0.015)+')'); g.addColorStop(1,'rgba(0,0,0,0)');
  ctx.fillStyle=g; ctx.fillRect(0,0,W*.6,H);
  g=ctx.createRadialGradient(W,H/2,0,W,H/2,W*.55);
  g.addColorStop(0,'rgba(95,219,232,'+(0.045+Math.cos(tGlobal*1.2)*0.015)+')'); g.addColorStop(1,'rgba(0,0,0,0)');
  ctx.fillStyle=g; ctx.fillRect(W*.4,0,W*.6,H);
  // drifting grid
  const gap=72*S, ox=(tGlobal*10)%gap;
  ctx.strokeStyle=COL.grid; ctx.lineWidth=1; ctx.beginPath();
  for(let x=-ox;x<W;x+=gap){ ctx.moveTo(x,0); ctx.lineTo(x,H); }
  for(let y=0;y<H;y+=gap){ ctx.moveTo(0,y); ctx.lineTo(W,y); }
  ctx.stroke();

  ctx.save();
  if(G.shake>0) ctx.translate(rand(-G.shake,G.shake),rand(-G.shake,G.shake));

  // net
  const pulse=.22+Math.sin(tGlobal*2)*.06;
  ctx.strokeStyle='rgba(140,205,235,'+(pulse*.5)+')'; ctx.lineWidth=6*S;
  ctx.setLineDash([12*S,18*S]); ctx.lineDashOffset=-tGlobal*30;
  ctx.beginPath(); ctx.moveTo(W/2,0); ctx.lineTo(W/2,H); ctx.stroke();
  ctx.strokeStyle='rgba(170,225,245,'+pulse+')'; ctx.lineWidth=2; ctx.stroke();
  ctx.setLineDash([]);

  // power cells
  for(const c of G.cells){
    const blink=c.life<2.5?(Math.sin(tGlobal*18)>0?.9:.3):.95;
    ctx.save(); ctx.translate(c.x,c.y); ctx.rotate(c.rot); ctx.globalAlpha=blink;
    ctx.shadowColor=COL.mint; ctx.shadowBlur=16; ctx.strokeStyle=COL.mint; ctx.lineWidth=2.4;
    const r=c.r*(1+Math.sin(tGlobal*5)*.06);
    ctx.beginPath();
    for(let i=0;i<6;i++){ const a=i*Math.PI/3; i?ctx.lineTo(Math.cos(a)*r,Math.sin(a)*r):ctx.moveTo(r,0); }
    ctx.closePath(); ctx.stroke();
    if(c.type===1){ ctx.fillStyle=COL.mint;
      for(let i=-1;i<=1;i++){ const a=i*.5; ctx.beginPath();
        ctx.arc(Math.cos(a)*r*.45,Math.sin(a)*r*.45,r*.13,0,7); ctx.fill(); } }
    else{ ctx.strokeStyle=COL.gold; ctx.beginPath(); ctx.arc(0,0,r*.42,0,7); ctx.stroke();
      ctx.fillStyle=COL.gold; ctx.beginPath(); ctx.arc(0,0,r*.12,0,7); ctx.fill(); }
    ctx.restore();
  }

  // ball trail + serve ring + ball
  const b=G.ball;
  if(G.end) ctx.globalAlpha=Math.max(0,G.end.t/.95);
  const tintCol=b.tint==='p'?COL.p:(b.tint==='c'?COL.c:'#ffffff');
  for(let i=0;i<b.trail.length;i++){ const t=b.trail[i], f=i/b.trail.length;
    ctx.globalAlpha=(G.end?Math.max(0,G.end.t/.95):1)*f*.3;
    ctx.fillStyle=tintCol; ctx.beginPath(); ctx.arc(t.x,t.y,M.br*f*.9,0,7); ctx.fill(); }
  ctx.globalAlpha=G.end?Math.max(0,G.end.t/.95):1;
  if(G.serveT>0&&!G.end){
    const f=1-G.serveT/1.2;
    ctx.strokeStyle='rgba(255,246,224,'+(.5-.3*f)+')'; ctx.lineWidth=2;
    ctx.beginPath(); ctx.arc(b.x,b.y,M.br+10+f*26,0,7); ctx.stroke(); }
  { const ang=Math.atan2(b.vy,b.vx), k=clamp(1+b.speed/M.baseSpeed*.3,1,1.8);
    ctx.save(); ctx.translate(b.x,b.y); ctx.rotate(ang); ctx.scale(k,1);
    ctx.shadowColor=tintCol; ctx.shadowBlur=18;
    const rg=ctx.createRadialGradient(0,0,0,0,0,M.br);
    rg.addColorStop(0,'#ffffff'); rg.addColorStop(.6,COL.ball); rg.addColorStop(1,tintCol);
    ctx.fillStyle=rg; ctx.beginPath(); ctx.arc(0,0,M.br,0,7); ctx.fill(); ctx.restore(); }
  ctx.globalAlpha=1;

  // bullets
  for(const bl of G.bullets){
    const a=Math.atan2(bl.vy,bl.vx), len=bl.r*3.4, col=bl.owner==='p'?COL.p:COL.c;
    ctx.save(); ctx.shadowColor=col; ctx.shadowBlur=10; ctx.lineCap='round';
    ctx.strokeStyle=col; ctx.lineWidth=bl.r*1.8;
    ctx.beginPath(); ctx.moveTo(bl.x-Math.cos(a)*len,bl.y-Math.sin(a)*len); ctx.lineTo(bl.x,bl.y); ctx.stroke();
    ctx.strokeStyle='#ffffff'; ctx.lineWidth=bl.r*.7;
    ctx.beginPath(); ctx.moveTo(bl.x-Math.cos(a)*len*.5,bl.y-Math.sin(a)*len*.5); ctx.lineTo(bl.x,bl.y); ctx.stroke();
    ctx.restore(); }

  // paddles
  drawPad(G.p); drawPad(G.c);

  // particles (additive)
  ctx.save(); ctx.globalCompositeOperation='lighter';
  for(const p of G.parts){ const f=p.life/p.max;
    if(p.ring){ ctx.globalAlpha=f*.8; ctx.strokeStyle=p.color; ctx.lineWidth=2.5;
      ctx.beginPath(); ctx.arc(p.x,p.y,p.r,0,7); ctx.stroke(); }
    else{ ctx.globalAlpha=f; ctx.fillStyle=p.color;
      ctx.beginPath(); ctx.arc(p.x,p.y,Math.max(.4,p.size*f),0,7); ctx.fill(); } }
  ctx.restore(); ctx.globalAlpha=1;

  // floating texts
  for(const t of G.texts){ const f=t.life/t.max;
    ctx.globalAlpha=Math.min(1,f*1.6); ctx.fillStyle='#04121c';
    ctx.font='700 '+t.size+'px Consolas, Menlo, monospace'; ctx.textAlign='center';
    ctx.fillText(t.txt,t.x+2,t.y+2); ctx.fillStyle=t.color; ctx.fillText(t.txt,t.x,t.y); }
  ctx.globalAlpha=1;
  ctx.restore(); // shake

  drawHUD();
  // score flashes
  if(G.flashP>0){ ctx.fillStyle='rgba(255,180,84,'+(G.flashP*.16)+')'; ctx.fillRect(W/2,0,W/2,H); }
  if(G.flashC>0){ ctx.fillStyle='rgba(95,219,232,'+(G.flashC*.16)+')'; ctx.fillRect(0,0,W/2,H); }
}
function drawPad(pad){
  if(pad.dead) return;
  const x=pad.x,y=pad.y-pad.h/2,w=pad.w,h=pad.h,col=pad.side==='p'?COL.p:COL.c;
  ctx.save();
  ctx.shadowColor=col; ctx.shadowBlur=14+(pad.hitT>0?20:0)+(pad.power?12:0);
  const g=ctx.createLinearGradient(x,0,x+w,0);
  g.addColorStop(0,col); g.addColorStop(.5,'#ffffff'); g.addColorStop(1,col);
  ctx.fillStyle=g; rr(x,y,w,h,w*.45); ctx.fill();
  ctx.restore();
  if(pad.hitT>0){ ctx.fillStyle='rgba(255,255,255,'+(pad.hitT*3)+')'; rr(x,y,w,h,w*.45); ctx.fill(); }
  if(pad.hurtT>0){ ctx.fillStyle='rgba(255,84,112,'+clamp(pad.hurtT*2.4,0,.9)+')'; rr(x,y,w,h,w*.45); ctx.fill(); }
  // integrity ticks above paddle
  const tw=w*1.1, gap=3;
  for(let i=0;i<pad.maxHp;i++){
    const on=i<pad.hp, bx=x+w/2-(pad.maxHp*(tw+gap)-gap)/2+i*(tw+gap);
    ctx.fillStyle=on?(pad.hurtT>0?COL.red:col):'rgba(255,255,255,.14)';
    ctx.fillRect(bx,y-13,tw,4); }
  // power sigil on face
  if(pad.power){ const cx=pad.x+w/2;
    ctx.save(); ctx.translate(cx,pad.y); ctx.rotate(tGlobal*3);
    ctx.strokeStyle=pad.power===1?COL.mint:COL.gold; ctx.lineWidth=2; ctx.globalAlpha=.9;
    if(pad.power===1){ for(let i=0;i<3;i++){ const a=i*Math.PI*2/3;
        ctx.beginPath(); ctx.moveTo(0,0); ctx.lineTo(Math.cos(a)*10,Math.sin(a)*10); ctx.stroke(); } }
    else{ ctx.beginPath(); ctx.arc(0,0,8,0,7); ctx.stroke();
      ctx.fillStyle=COL.gold; ctx.beginPath(); ctx.arc(0,0,2.5,0,7); ctx.fill(); }
    ctx.restore(); }
}

function drawInteg(x,y,pad,right,color){
  const seg=Math.max(12,16*S+4), gap=5, total=pad.maxHp*(seg+gap)-gap;
  ctx.font='700 '+(9*S+4)+'px Consolas,Menlo,monospace';
  ctx.fillStyle=COL.dim; ctx.textAlign=right?'right':'left';

    const idx=right?pad.maxHp-1-i:i, bx=right?x-(i+1)*(seg+gap)+gap:x+i*(seg+gap);
    ctx.fillStyle=idx<pad.hp?(pad.hurtT>0?COL.red:color):'rgba(255,255,255,.12)';
    ctx.fillRect(bx,y,seg,7); }
}

}
