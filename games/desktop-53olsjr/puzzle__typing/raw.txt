function anagram(word){const a=word.split('');let s,g=0;
  do{for(let i=a.length-1;i>0;i--){const j=Math.floor(Math.random()*(i+1));[a[i],a[j]]=[a[j],a[i]];}s=a.join('');}
  while(s===word&&++g<24);return s;}

function makeBlockObj(word,lane,w,tile,h,speed){
  return {lane,y:-70,word,shown:anagram(word),ci:S.ciCounter++%PALETTE.length,
    speed,seed:Math.random()*6.28,w,tile,h};}

function spawnBlock(){
  for(let t=0;t<5;t++){
    const minLen=clamp(4+Math.floor((S.level-3)/3),4,7);
    const maxLen=clamp(4+S.level,5,9);
    let len=irand(minLen,maxLen);
    const used=new Set(S.blocks.map(b=>b.lane)),free=[];
    for(let i=0;i<laneCount;i++)if(!used.has(i))free.push(i);
    const lane=free.length?pick(free):irand(0,laneCount-1);
    const lw=LANES[lane].w;
    const fitLen=Math.max(3,Math.floor((lw-14)/18));
    if(len>fitLen)len=fitLen;
    let pool=(WORDS[len]||[]).filter(w=>!S.blocks.some(b=>b.word===w));
    if(!pool.length)pool=WORDS[len]||[];
    if(!pool.length)continue;
    const word=pick(pool);
    const w=Math.min(lw,36+word.length*27);
    const tile=clamp((w-14-(word.length-1)*5)/word.length,13,40);
    const base=(H*.1)*(1+(S.level-1)*.13)*rand(.9,1.12)*clamp(1.15-word.length*.03,.82,1.12);
    S.blocks.push(makeBlockObj(word,lane,w,tile,tile+16,Math.min(base,H*.4)));
    return;
  }
  /* absolute fallback */
  const lane=irand(0,laneCount-1),word=pick(WORDS[4]||['word']),lw=LANES[lane].w;
  const w=Math.min(lw,36+word.length*27),tile=clamp((w-14-(word.length-1)*5)/word.length,10,40);
  S.blocks.push(makeBlockObj(word,lane,w,tile,tile+16,H*.12));
}

/* ================= update ================= */
function update(dt){
  S.t+=dt;
  for(const r of S.rain){r.y+=r.sp*dt;if(r.y>H+20){r.y=-20;r.x=rand(0,W);}}
  if(S.mode!=='play'){
    for(let i=0;i<S.ambient.length;i++){const a=S.ambient[i];a.y+=a.sp*dt;
      if(a.y>H+90)S.ambient[i]=makeAmbient(false);}
  }else{
    S.spawnTimer-=dt;
    const target=clamp(1+Math.floor(S.level/2),1,laneCount);
    if(S.blocks.length<target&&S.spawnTimer<=0){
      const used=new Set(S.blocks.map(b=>b.lane));let free=[];
      for(let i=0;i<laneCount;i++)if(!used.has(i))free.push(i);
      const lane=free.length?pick(free):-1;
      if(lane>=0||!S.blocks.some(b=>b.y<130)){spawnBlock();S.spawnTimer=clamp(2-S.level*.12,.7,2)*rand(.85,1.15);}
      else S.spawnTimer=.3;
    }
    for(const b of [...S.blocks]){b.y+=b.speed*dt;if(b.y+b.h/2>=floorY)lifeLost(b);}
  }
  for(let i=S.particles.length-1;i>=0;i--){const p=S.particles[i];p.life-=dt;
    if(p.life<=0){S.particles.splice(i,1);continue;}
    if(p.type==='ring'){p.r+=p.vr*dt;continue;}
    p.x+=p.vx*dt;p.y+=p.vy*dt;p.vy+=760*dt;if(p.rot!==undefined)p.rot+=p.vr*dt;}
  for(let i=S.floaters.length-1;i>=0;i--){const f=S.floaters[i];f.life-=dt;f.y+=f.vy*dt;
    if(f.life<=0)S.floaters.splice(i,1);}
  S.shake=Math.max(0,S.shake-dt*42);
  S.bufFlash=Math.max(0,S.bufFlash-dt*3);
  S.dmgFlash=Math.max(0,S.dmgFlash-dt*1.8);
  S.levelFlash=Math.max(0,S.levelFlash-dt*2.2);
}

  [['BRAIN','d-amber'],['CHAOS','d-coral'],['PIXEL','d-sky']].forEach(([word,cls])=>{

  const demos=[...row.children];
  const reshuffle=()=>{if(document.hidden)return;

  reshuffle();setInterval(reshuffle,1400);
}
