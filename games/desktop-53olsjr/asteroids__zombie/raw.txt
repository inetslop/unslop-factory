function makeLayer(n, sMin, sMax, aMin, aMax){
  const arr = [];
  for (let i=0;i<n;i++) arr.push({ x:Math.random()*W, y:Math.random()*H,
    s:rand(sMin,sMax), a:rand(aMin,aMax), tw:Math.random()<.35, ph:rand(0,TAU), sp:rand(.6,2.2) });
  return arr;
}
function buildStars(){
  starsFar  = makeLayer(Math.max(60, Math.round(W*H/9000)),  .7, 1.3, .25, .5);
  starsMid  = makeLayer(Math.max(34, Math.round(W*H/15000)), 1.1, 1.8, .4, .7);

}
function drawStars(arr, speed, col){
  ctx.fillStyle = col;
  for (const s of arr){
    s.x -= speed * 0.28 * (1/60); s.y += speed * (1/60);   // slow diagonal drift per frame-rate approx
    if (s.x < -3) s.x = W + 3; if (s.y > H + 3) s.y = -3;
    let a = s.a;
    if (s.tw) a *= .55 + .45 * Math.sin(t * s.sp + s.ph);
    ctx.globalAlpha = a;
    ctx.fillRect(s.x, s.y, s.s, s.s);
  }
  ctx.globalAlpha = 1;
}
function nebula(x, y, r, rgb, a){
  const g = ctx.createRadialGradient(x, y, 0, x, y, r);
  g.addColorStop(0, 'rgba(' + rgb + ',' + a.toFixed(3) + ')');
  g.addColorStop(1, 'rgba(' + rgb + ',0)');
  ctx.fillStyle = g; ctx.fillRect(x - r, y - r, r * 2, r * 2);
}

let AC = null, master = null, noiseBuf = null, thrGain = null, muted = false;
try { muted = localStorage.getItem('zombie-asteroids-muted') === '1'; } catch(e){}
function setMuted(m){
  muted = m;
  if (AC && master) master.gain.setTargetAtTime(m ? 0 : 0.5, AC.currentTime, .02);
  $('muteTag').classList.toggle('hidden', !m);
  try { localStorage.setItem('zombie-asteroids-muted', m ? '1' : '0'); } catch(e){}
}
function initAudio(){
  if (AC) { if (AC.state === 'suspended') AC.resume(); return; }
  try {
    AC = new (window.AudioContext || window.webkitAudioContext)();
    master = AC.createGain(); master.gain.value = muted ? 0 : .5; master.connect(AC.destination);
    const len = AC.sampleRate; noiseBuf = AC.createBuffer(1, len, AC.sampleRate);
    const d = noiseBuf.getChannelData(0);
    for (let i=0;i<len;i++) d[i] = Math.random()*2 - 1;
    // thruster loop
    const src = AC.createBufferSource(); src.buffer = noiseBuf; src.loop = true;
    const lp = AC.createBiquadFilter(); lp.type = 'lowpass'; lp.frequency.value = 380;
    thrGain = AC.createGain(); thrGain.gain.value = 0;
    src.connect(lp); lp.connect(thrGain); thrGain.connect(master); src.start();
  } catch(e){ AC = null; }
}
function beep(type, f0, f1, dur, vol, when){
  if (!AC) return;
  const t0 = AC.currentTime + (when || 0);
  const o = AC.createOscillator(), g = AC.createGain();
  o.type = type; o.frequency.setValueAtTime(f0, t0);
  if (f1) o.frequency.exponentialRampToValueAtTime(Math.max(1, f1), t0 + dur);
  g.gain.setValueAtTime(vol, t0);
  g.gain.exponentialRampToValueAtTime(.0001, t0 + dur);
  o.connect(g); g.connect(master); o.start(t0); o.stop(t0 + dur + .03);
}
function noiseHit(dur, vol, f0, f1){
  if (!AC) return;
  const t0 = AC.currentTime;
  const s = AC.createBufferSource(); s.buffer = noiseBuf;
  const f = AC.createBiquadFilter(); f.type = 'lowpass';
  f.frequency.setValueAtTime(f0, t0); f.frequency.exponentialRampToValueAtTime(Math.max(40, f1), t0 + dur);
  const g = AC.createGain(); g.gain.setValueAtTime(vol, t0);
  g.gain.exponentialRampToValueAtTime(.0001, t0 + dur);
  s.connect(f); f.connect(g); g.connect(master); s.start(t0); s.stop(t0 + dur + .03);
}
function groan(f, v){
  if (!AC) return;
  const t0 = AC.currentTime;
  const o = AC.createOscillator(); o.type = 'sawtooth';
  o.frequency.setValueAtTime(64*f, t0); o.frequency.linearRampToValueAtTime(50*f, t0 + .5);
  const lfo = AC.createOscillator(); lfo.frequency.value = 5.5;
  const lg = AC.createGain(); lg.gain.value = 9 * f;
  lfo.connect(lg); lg.connect(o.frequency);
  const bp = AC.createBiquadFilter(); bp.type = 'bandpass'; bp.frequency.value = 280*f; bp.Q.value = 1.4;
  const g = AC.createGain(); g.gain.setValueAtTime(.0001, t0);
  g.gain.linearRampToValueAtTime(v, t0 + .12);
  g.gain.exponentialRampToValueAtTime(.0001, t0 + .55);
  o.connect(bp); bp.connect(g); g.connect(master);
  o.start(t0); lfo.start(t0); o.stop(t0 + .6); lfo.stop(t0 + .6);
}

const ROCK_R = [48, 27, 15], ROCK_SPD = [34, 64, 104], ROCK_PTS = [20, 50, 100];
function spawnRock(size, x, y, vx, vy){
  if (x === undefined){ // random edge, away from ship
    for (let i=0;i<8;i++){
      const side = Math.floor(Math.random()*4);
      x = side===0 ? -40 : side===1 ? W+40 : rand(0,W);
      y = side===2 ? -40 : side===3 ? H+40 : rand(0,H);
      if (Math.hypot(x-ship.x, y-ship.y) > 260) break;
    }
  }
  const R = ROCK_R[size];
  const n = 8 + Math.floor(Math.random()*5), pts = [];
  for (let i=0;i<n;i++){
    const a = i/n*TAU, r = R * rand(.72, 1.18);
    pts.push([Math.cos(a)*r, Math.sin(a)*r]);
  }
  const cr = [];
  const cn = 1 + Math.floor(Math.random()*3);
  for (let i=0;i<cn;i++) cr.push({ a:rand(0,TAU), d:R*rand(.2,.55), r:R*rand(.12,.24) });
  if (vx === undefined){
    const a = rand(0,TAU), sp = ROCK_SPD[size] * rand(.7,1.4) + S.wave*3;
    vx = Math.cos(a)*sp; vy = Math.sin(a)*sp;
  }
  asteroids.push({ size, x, y, vx, vy, rot:rand(0,TAU), vr:rand(-1.4,1.4), pts, cr });
}

/* ============================== UPDATE ============================== */
function turnToward(h, target, max){
  let d = target - h;
  while (d > Math.PI) d -= TAU; while (d < -Math.PI) d += TAU;
  return h + Math.max(-max, Math.min(max, d));
}
function update(dt){
  t += dt;
  S.shake *= Math.exp(-5.5 * dt);
  S.shX = (Math.random()*2-1) * S.shake; S.shY = (Math.random()*2-1) * S.shake;

  // hi-score persistence throttle
  if (hiDirty){ hiTimer -= dt; if (hiTimer <= 0){ saveHi(); hiDirty = false; hiTimer = 2; } }

  // asteroids always drift
  for (const a of asteroids){ a.x += a.vx*dt; a.y += a.vy*dt; a.rot += a.vr*dt; wrap(a, a.size===0?60:35); }

  // zombies
  for (let i = zombies.length-1; i >= 0; i--){
    const z = zombies[i], T = ZTYPES[z.type];
    z.t += dt; if (z.flash > 0) z.flash -= dt;
    if (z.entry > 0){
      z.entry -= dt;
      if (z.entry <= 0){ sfx.spawnZ(); burst(z.x, z.y, 6, ['#7dff5e','#aef29a'], 70, 2.4, .4); }
      continue;
    }
    let tx, ty;
    if ((S.mode === 'playing' || (S.mode === 'start')) && !(S.mode==='playing' && !ship.alive)){
      tx = ship.x; ty = ship.y;
    } else {
      z.wt -= dt;
      if (z.wt <= 0){ z.wx = rand(40, W-40); z.wy = rand(40, H-40); z.wt = rand(2,4.5); }
      tx = z.wx; ty = z.wy;
    }
    const want = Math.atan2(ty - z.y, tx - z.x);
    const head = Math.atan2(z.vy, z.vx);
    const nh = turnToward(head, want, T.turn * dt);
    const spT = T.speed * (1 + S.wave*.04) * (S.mode === 'start' ? .5 : 1);
    let sp = Math.hypot(z.vx, z.vy);
    sp += (spT - sp) * (1 - Math.exp(-3*dt));
    const wob = Math.sin(z.t*3.2 + z.ph) * 16;
    z.vx = Math.cos(nh)*sp + Math.cos(nh + Math.PI/2)*wob*.5;
    z.vy = Math.sin(nh)*sp + Math.sin(nh + Math.PI/2)*wob*.5;
    z.x += z.vx*dt; z.y += z.vy*dt; wrap(z, z.r);
  }

  // pickups
  for (let i = pickups.length-1; i >= 0; i--){
    const p = pickups[i];
    p.x += p.vx*dt; p.y += p.vy*dt; p.life -= dt; wrap(p, 24);
    if (p.life <= 0){ pickups.splice(i,1); continue; }
    if (ship.alive && Math.hypot(p.x-ship.x, p.y-ship.y) < 24){ applyPickup(p); pickups.splice(i,1); }
  }

  // particles & texts (all modes)
  for (let i = parts.length-1; i >= 0; i--){
    const p = parts[i];
    p.life -= dt; if (p.life <= 0){ parts.splice(i,1); continue; }
    const dmp = Math.exp(-1.6*dt);
    p.vx *= dmp; p.vy *= dmp; p.x += p.vx*dt; p.y += p.vy*dt;
  }
  for (let i = texts.length-1; i >= 0; i--){
    const x = texts[i]; x.life -= dt; x.y -= 28*dt;
    if (x.life <= 0) texts.splice(i,1);
  }

  if (S.mode !== 'playing') return;

  /* ---- ship ---- */
  if (ship.alive){
    if (keys.left)  ship.a -= 3.8 * dt;
    if (keys.right) ship.a += 3.8 * dt;
    if (keys.thrust){
      ship.vx += Math.cos(ship.a) * 460 * dt;
      ship.vy += Math.sin(ship.a) * 460 * dt;
      if (Math.random() < .55) parts.push({ x:ship.x - Math.cos(ship.a)*12 + rand(-3,3),
        y:ship.y - Math.sin(ship.a)*12 + rand(-3,3),
        vx:-Math.cos(ship.a)*rand(60,140) + rand(-20,20), vy:-Math.sin(ship.a)*rand(60,140) + rand(-20,20),
        life:.3, max:.3, size:2.4, color: Math.random()<.5 ? '#ffb347' : '#ff7a1a' });
    }
    const dmp = Math.exp(-.7*dt); ship.vx *= dmp; ship.vy *= dmp;
    const sp = Math.hypot(ship.vx, ship.vy);
    if (sp > 430){ ship.vx *= 430/sp; ship.vy *= 430/sp; }
    ship.x += ship.vx*dt; ship.y += ship.vy*dt; wrap(ship, 16);
    if (ship.invuln > 0) ship.invuln -= dt;
    if (ship.shield > 0) ship.shield -= dt;
    if (ship.triple > 0) ship.triple -= dt;
    if (ship.rapid > 0) ship.rapid -= dt;

    // fire
    ship.fireCd -= dt;
    if (keys.fire && ship.fireCd <= 0 && bullets.length < 8){
      const n = ship.triple > 0 ? 3 : 1;
      for (let k=0;k<n;k++){
        const off = n === 3 ? (k-1)*.2 : 0;
        const a = ship.a + off;
        bullets.push({ x:ship.x + Math.cos(ship.a)*16, y:ship.y + Math.sin(ship.a)*16,
          vx:Math.cos(a)*540 + ship.vx*.4, vy:Math.sin(a)*540 + ship.vy*.4, life:.8 });
      }
      ship.fireCd = ship.rapid > 0 ? .07 : .16;
      sfx.shoot();
    }

    // collisions
    if (ship.invuln <= 0){
      let dead = false;
      for (const z of zombies){
        if (z.entry > 0) continue;
        if (Math.hypot(z.x-ship.x, z.y-ship.y) < z.r + 8){
          if (ship.shield > 0){ ship.shield = 0; z.hp--; z.flash=.15; sfx.boom(.4);
            burst(ship.x, ship.y, 20, ['#7dff5e','#baffa0'], 200, 3, .6);
            if (z.hp <= 0){ zombies.splice(zombies.indexOf(z),1); killZombie(z); }
          } else { dead = true; }
          break;
        }
      }
      if (!dead) for (const a of asteroids){
        if (Math.hypot(a.x-ship.x, a.y-ship.y) < ROCK_R[a.size]*.82 + 8){
          if (ship.shield > 0){ ship.shield = 0; sfx.boom(.4);
            burst(ship.x, ship.y, 20, ['#7dff5e','#baffa0'], 200, 3, .6);
          } else { dead = true; }
          break;
        }
      }
      if (dead) killShip();
    }
  } else {
    if (S.lives > 0){
      ship.respawn -= dt;
      if (ship.respawn <= 0){
        for (let i=0;i<14;i++){
          const x = W/2 + rand(-130,130), y = H/2 + rand(-130,130);
          let ok = true;
          for (const a of asteroids) if (Math.hypot(a.x-x,a.y-y) < ROCK_R[a.size]+70){ ok=false; break; }
          for (const z of zombies) if (z.entry<=0 && Math.hypot(z.x-x,z.y-y) < 90){ ok=false; break; }
          if (ok){ ship.x = x; ship.y = y; break; }
        }
        ship.vx = 0; ship.vy = 0; ship.alive = true; ship.invuln = 2.5;
        burst(ship.x, ship.y, 14, ['#6fd8ff','#dcf3ff'], 120, 2.6, .5);
      }
    } else {
      S.overTimer -= dt;
      if (S.overTimer <= 0){ gameOver(); return; }
    }
  }

  /* ---- bullets ---- */
  for (let i = bullets.length-1; i >= 0; i--){
    const b = bullets[i];
    b.x += b.vx*dt; b.y += b.vy*dt; b.life -= dt;
    if (b.life <= 0 || b.x < -20 || b.x > W+20 || b.y < -20 || b.y > H+20){ bullets.splice(i,1); continue; }
    let hit = false;
    for (let j = zombies.length-1; j >= 0 && !hit; j--){
      const z = zombies[j];
      if (z.entry > 0) continue;
      if (Math.hypot(z.x-b.x, z.y-b.y) < z.r + 3){
        hit = true; z.hp--; z.flash = .12;
        burst(b.x, b.y, 5, ['#7dff5e','#cdd6c4'], 90, 2.2, .35);
        sfx.hitZ();
        if (z.hp <= 0){ zombies.splice(j,1); killZombie(z); }
      }
    }
    if (!hit) for (let j = asteroids.length-1; j >= 0 && !hit; j--){
      const a = asteroids[j];
      if (Math.hypot(a.x-b.x, a.y-b.y) < ROCK_R[a.size]*.92 + 3){
        hit = true; asteroids.splice(j,1); destroyRock(a);
      }
    }
    if (hit) bullets.splice(i,1);
  }

  // wave clear
  if (asteroids.length === 0 && zombies.length === 0 && ship.alive && S.lives > 0){
    if (S.waveT === null) S.waveT = 1.5;
    else {
      S.waveT -= dt;
      if (S.waveT <= 0){
        S.waveT = null; S.wave++;
        spawnWave(S.wave);
        banner('WAVE ' + S.wave, FLAVOR[(S.wave-2) % FLAVOR.length]);
        sfx.wave();
      }
    }
  }

  // thruster audio level
  if (AC && thrGain){
    const target = (ship.alive && keys.thrust) ? .13 : 0;
    thrGain.gain.setTargetAtTime(target, AC.currentTime, .06);
  }
}

function drawRock(a){
  ctx.save(); ctx.translate(a.x, a.y); ctx.rotate(a.rot);
  const p = a.pts;
  ctx.beginPath(); ctx.moveTo(p[0][0], p[0][1]);
  for (let i=1;i<p.length;i++) ctx.lineTo(p[i][0], p[i][1]);
  ctx.closePath();
  ctx.fillStyle = '#1d232e'; ctx.fill();
  ctx.lineWidth = a.size === 0 ? 2 : 1.5;
  ctx.strokeStyle = 'rgba(148,160,184,.9)'; ctx.stroke();
  ctx.fillStyle = 'rgba(9,12,17,.85)';
  for (const c of a.cr){ ctx.beginPath(); ctx.arc(Math.cos(c.a)*c.d, Math.sin(c.a)*c.d, c.r, 0, TAU); ctx.fill(); }
  ctx.restore();
}
function drawZombie(z){
  const T = ZTYPES[z.type];
  ctx.save(); ctx.translate(z.x, z.y);
  ctx.rotate(Math.atan2(z.vy, z.vx) + Math.PI/2 + Math.sin(z.t*2.6 + z.ph)*.18);
  ctx.scale(T.scale, T.scale);
  const wig = Math.sin(z.t*5.4 + z.ph);
  // backpack
  ctx.fillStyle = '#33402f'; ctx.fillRect(-4, 5, 8, 6);
  // legs
  ctx.strokeStyle = T.bone; ctx.lineWidth = T.lw; ctx.lineCap = 'round';
  ctx.beginPath();
  ctx.moveTo(-3, 8); ctx.lineTo(-4.5 + Math.sin(z.t*5.4 + z.ph + 1)*2, 16);
  ctx.moveTo(3, 8);   ctx.lineTo(4.5 - Math.sin(z.t*5.4 + z.ph + 1)*2, 16);
  ctx.stroke();
  // torso
  ctx.beginPath();
  ctx.moveTo(-7,-3); ctx.lineTo(7,-3); ctx.lineTo(5.5,8); ctx.lineTo(-5.5,8); ctx.closePath();
  ctx.fillStyle = '#38452f'; ctx.fill();
  ctx.lineWidth = T.lw*.6; ctx.stroke();
  // arms reaching forward
  ctx.strokeStyle = T.bone; ctx.lineWidth = T.lw;
  ctx.beginPath();
  ctx.moveTo(-7,-1); ctx.lineTo(-9 + wig*1.5, -11 + wig*2);
  ctx.moveTo(7,-1);   ctx.lineTo(9 - wig*1.5, -11 - wig*2);
  ctx.stroke();
  // helmet
  ctx.beginPath(); ctx.arc(0,-9,6.4,0,TAU);
  ctx.fillStyle = '#151d16'; ctx.fill();
  ctx.lineWidth = T.lw*.7; ctx.strokeStyle = T.bone; ctx.stroke();
  // visor crack
  ctx.strokeStyle = 'rgba(220,255,230,.28)'; ctx.lineWidth = .8;
  ctx.beginPath(); ctx.moveTo(-3.6,-12); ctx.lineTo(2.6,-7); ctx.stroke();
  // glowing eyes
  ctx.shadowColor = '#8dff5e'; ctx.shadowBlur = 8;
  ctx.fillStyle = '#8dff5e';
  ctx.beginPath(); ctx.arc(-2.3,-9.4,1.7,0,TAU); ctx.fill();
  ctx.beginPath(); ctx.arc(2.3,-9.4,1.7,0,TAU); ctx.fill();
  ctx.shadowBlur = 0;
  // chest light
  ctx.fillStyle = 'rgba(125,255,94,' + (.45 + .4*Math.sin(z.t*4 + z.ph)).toFixed(2) + ')';
  ctx.beginPath(); ctx.arc(0,1.5,1.4,0,TAU); ctx.fill();
  // hit flash
  if (z.flash > 0){
    ctx.globalCompositeOperation = 'lighter';
    ctx.fillStyle = 'rgba(255,255,255,' + Math.min(.75, z.flash*6).toFixed(2) + ')';
    ctx.beginPath(); ctx.arc(0,-1,17,0,TAU); ctx.fill();
    ctx.globalCompositeOperation = 'source-over';
  }
  ctx.restore();
}
function drawShip(){
  if (!ship.alive) return;
  const blink = ship.invuln > 0 && (Math.floor(t*12) % 2 === 0);
  ctx.save(); ctx.translate(ship.x, ship.y);
  if (blink) ctx.globalAlpha = .35;
  // shield
  if (ship.shield > 0){
    ctx.save(); ctx.rotate(t*.9);
    ctx.setLineDash([7,9]); ctx.lineDashOffset = -t*34;
    ctx.strokeStyle = ship.shield < 2 ? 'rgba(255,179,71,.8)' : 'rgba(125,255,94,.75)';
    ctx.lineWidth = 2; ctx.shadowColor = '#7dff5e'; ctx.shadowBlur = 10;
    ctx.beginPath(); ctx.arc(0,0,21,0,TAU); ctx.stroke();
    ctx.restore();
  }
  ctx.rotate(ship.a + Math.PI/2);
  // thruster flame
  if (keys.thrust){
    const f = 10 + Math.random()*9;
    ctx.fillStyle = 'rgba(255,140,40,.9)';
    ctx.beginPath(); ctx.moveTo(-4.5,8); ctx.lineTo(4.5,8); ctx.lineTo(0,8+f); ctx.closePath(); ctx.fill();
    ctx.fillStyle = 'rgba(255,236,180,.95)';
    ctx.beginPath(); ctx.moveTo(-2.2,8); ctx.lineTo(2.2,8); ctx.lineTo(0,8+f*.55); ctx.closePath(); ctx.fill();
  }
  // hull
  ctx.lineJoin = 'round';
  ctx.beginPath();
  ctx.moveTo(0,-15); ctx.lineTo(10,10); ctx.lineTo(0,5); ctx.lineTo(-10,10); ctx.closePath();
  ctx.fillStyle = 'rgba(150,220,255,.10)'; ctx.fill();
  ctx.strokeStyle = '#dcf3ff'; ctx.lineWidth = 2; ctx.stroke();
  ctx.fillStyle = 'rgba(160,230,255,.85)';
  ctx.beginPath(); ctx.arc(0,-4,2,0,TAU); ctx.fill();
  ctx.restore();
}
function drawPickup(p){
  const bob = Math.sin(t*3 + p.ph)*3;
  const blinkA = p.life < 2.5 ? (Math.floor(t*8) % 2 ? .3 : 1) : 1;
  const col = p.type === 'shield' ? '#7dff5e' : p.type === 'triple' ? '#ffb347' : '#6fd8ff';
  ctx.save(); ctx.translate(p.x, p.y + bob); ctx.globalAlpha = blinkA;
  ctx.shadowColor = col; ctx.shadowBlur = 12;
  ctx.strokeStyle = col; ctx.lineWidth = 1.6;
  ctx.beginPath(); ctx.arc(0,0,11,0,TAU); ctx.stroke();
  ctx.fillStyle = col; ctx.shadowBlur = 0;
  if (p.type === 'shield'){ ctx.fillRect(-5,-1.6,10,3.2); ctx.fillRect(-1.6,-5,3.2,10); }
  else if (p.type === 'triple'){
    for (let k=-1;k<=1;k++){ ctx.save(); ctx.rotate(k*.5);
      ctx.beginPath(); ctx.moveTo(0,-6); ctx.lineTo(2.4,2); ctx.lineTo(-2.4,2); ctx.closePath(); ctx.fill(); ctx.restore(); }
  } else {
    ctx.beginPath();
    ctx.moveTo(1.5,-6.5); ctx.lineTo(-3.5,1); ctx.lineTo(-.5,1); ctx.lineTo(-1.5,6.5);
    ctx.lineTo(3.5,-1); ctx.lineTo(.5,-1); ctx.closePath(); ctx.fill();
  }
  ctx.restore();
}
function render(){
  // background
  ctx.fillStyle = '#04070c'; ctx.fillRect(0,0,W,H);
  ctx.globalCompositeOperation = 'lighter';
  nebula(W*.25 + Math.sin(t*.05)*W*.1, H*.3 + Math.cos(t*.04)*H*.1, Math.min(W,H)*.7, '60,255,120', .05 + .018*Math.sin(t*.11));
  nebula(W*.75 + Math.cos(t*.037)*W*.08, H*.68 + Math.sin(t*.045)*H*.09, Math.min(W,H)*.6, '40,190,210', .045);
  nebula(W*.5, H*1.05 + Math.sin(t*.03)*40, Math.min(W,H)*.55, '255,95,60', .035);
  ctx.globalCompositeOperation = 'source-over';
  drawStars(starsFar, 7, '#8fa3c8');
  drawStars(starsMid, 15, '#b9c9e6');
  drawStars(starsNear, 27, '#e8f2ff');

  // world (shaken)
  ctx.save(); ctx.translate(S.shX, S.shY);
  for (const p of pickups) drawPickup(p);
  for (const a of asteroids) drawRock(a);
  for (const z of zombies) if (z.entry <= 0) drawZombie(z);
  // bullets
  ctx.save(); ctx.shadowColor = '#ffb347'; ctx.shadowBlur = 8;
  for (const b of bullets){
    ctx.strokeStyle = 'rgba(255,190,110,.4)'; ctx.lineWidth = 2;
    ctx.beginPath(); ctx.moveTo(b.x - b.vx*.028, b.y - b.vy*.028); ctx.lineTo(b.x, b.y); ctx.stroke();
    ctx.fillStyle = '#ffd9a0';
    ctx.beginPath(); ctx.arc(b.x, b.y, 2.6, 0, TAU); ctx.fill();
  }
  ctx.restore();
  drawShip();
  // particles (additive glow)
  ctx.globalCompositeOperation = 'lighter';
  for (const p of parts){
    const a = Math.max(0, p.life / p.max);
    ctx.globalAlpha = a;
    ctx.fillStyle = p.color;
    ctx.beginPath(); ctx.arc(p.x, p.y, p.size * (.4 + a*.6), 0, TAU); ctx.fill();
  }
  ctx.globalAlpha = 1; ctx.globalCompositeOperation = 'source-over';
  // floating score texts
  ctx.font = '700 13px ' + getComputedStyle(document.body).getPropertyValue('--mono');
  ctx.textAlign = 'center';
  for (const x of texts){
    ctx.globalAlpha = Math.min(1, x.life * 1.6);
    ctx.fillStyle = x.color;
    ctx.fillText(x.str, x.x, x.y);
  }
  ctx.globalAlpha = 1; ctx.textAlign = 'left';
  ctx.restore();
}

function syncChips(){
  const map = [['chipShield','SHIELD ', ship.shield], ['chipTriple','TRIPLE ', ship.triple], ['chipRapid','RAPID ', ship.rapid]];
  for (const [id, label, v] of map){

    if (v > 0){ el.style.display = 'inline-block'; el.textContent = label + Math.ceil(v); }
    else el.style.display = 'none';
  }
}
