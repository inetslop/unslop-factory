function collide(){

 let cx=clamp((P.x/CELL)|0,0,M.c-1), cy=clamp((P.y/CELL)|0,0,M.r-1);

  const w=M.walls[gy*M.c+gx], x0=gx*CELL, y0=gy*CELL;
  if(w&1) pushSeg(x0,y0,x0+CELL,y0);
  if(w&4) pushSeg(x0,y0+CELL,x0+CELL,y0+CELL);
  if(w&8) pushSeg(x0,y0,x0,y0+CELL);
  if(w&2) pushSeg(x0+CELL,y0,x0+CELL,y0+CELL);
 }}

function pushSeg(ax,ay,bx,by){ const P=G.player; const dx=bx-ax,dy=by-ay; const L2=dx*dx+dy*dy; let t=((P.x-ax)*dx+(P.y-ay)*dy)/L2; t=t<0?0:t>1?1:t; const px=ax+dx*t,py=ay+dy*t; let ex=P.x-px,ey=P.y-py; const d2=ex*ex+ey*ey, R=P.r+2.5; if(d2<R*R&&d2>0.0001){const d=Math.sqrt(d2); const push=(R-d)/d; P.x+=ex*push; P.y+=ey*push;} }

const CELL=48;
// [dx,dy,bitToMe,bitToNeighbor]  N=1 E=2 S=4 W=8
const DIRS=[[1,-1,1,4],[1,0,2,8],[-1,0,8,2],[0,1,4,1]];
function makeMaze(c,r,braid){
  const walls=new Uint8Array(c*r).fill(15),vis=new Uint8Array(c*r);
  const stack=[0];vis[0]=1;
  while(stack.length){
    const cur=stack[stack.length-1],cx=cur%c,cy=(cur/c)|0,opts=[];
    for(const d of DIRS){const nx=cx+d[0],ny=cy+d[1];if(nx>=0&&ny>=0&&nx<c&&ny<r&&!vis[ny*c+nx])opts.push(d);}
    if(!opts.length){stack.pop();continue;}
    const d=opts[irand(opts.length)],nb=(cy+d[1])*c+(cx+d[0]);
    walls[cur]&=~d[2];walls[nb]&=~d[3];vis[nb]=1;stack.push(nb);
  }
  // braid: knock out interior walls to create loops
  let want=Math.max(1,Math.floor(c*r*braid)),done=0,guard=c*r*4;
  while(done<want&&guard-->0){
    const cx=irand(c),cy=irand(r);
    const d=Math.random()<.5?[1,0,2,8]:[0,1,4,1];
    const nx=cx+d[0],ny=cy+d[1];if(nx>=c||ny>=r)continue;
    const cur=cy*c+cx,nb=ny*c+nx;
    if(walls[cur]&d[2]){walls[cur]&=~d[2];walls[nb]&=~d[3];done++;}
  }
  return {c,r,walls};
}
function bfsDist(m,from){
  const n=m.c*m.r,d=new Int32Array(n).fill(-1);d[from]=0;
  const q=[from];let qi=0;
  while(qi<q.length){const cur=q[qi++],cx=cur%m.c,cy=(cur/m.c)|0;
    for(const dd of DIRS){if(m.walls[cur]&dd[2])continue;
      const nx=cx+dd[0],ny=cy+dd[1];if(nx<0||ny<0||nx>=m.c||ny>=m.r)continue;
      const nb=ny*m.c+nx;if(d[nb]===-1){d[nb]=d[cur]+1;q.push(nb);}}}
  return d;
}
function bfsPath(m,from,to){
  if(from===to)return[to];
  const n=m.c*m.r,prev=new Int32Array(n).fill(-1);prev[from]=from;
  const q=[from];let qi=0;
  while(qi<q.length){const cur=q[qi++];if(cur===to)break;
    const cx=cur%m.c,cy=(cur/m.c)|0;
    for(const dd of DIRS){if(m.walls[cur]&dd[2])continue;
      const nx=cx+dd[0],ny=cy+dd[1];if(nx<0||ny<0||nx>=m.c||ny>=m.r)continue;
      const nb=ny*m.c+nx;if(prev[nb]===-1){prev[nb]=cur;q.push(nb);}}}
  if(prev[to]===-1)return null;
  const path=[];let cur=to;
  while(cur!==from){path.push(cur);cur=prev[cur];}
  path.push(from);path.reverse();return path;
}
function paintMaze(m){
  const w=m.c*CELL,h=m.r*CELL,mc=document.createElement('canvas');
  mc.width=w;mc.height=h;const g=mc.getContext('2d');
  g.fillStyle='#0f1e29';g.fillRect(0,0,w,h);
  for(let y=0;y<m.r;y++)for(let x=0;x<m.c;x++){
    if((x+y)&1){g.fillStyle='rgba(150,190,210,0.03)';g.fillRect(x*CELL,y*CELL,CELL,CELL);}
  }
  g.fillStyle='rgba(180,215,230,0.05)';
  for(let i=0;i<m.c*m.r*2;i++)g.fillRect(irand(w),irand(h),1.5,1.5);
  const segs=[];
  for(let y=0;y<m.r;y++)for(let x=0;x<m.c;x++){
    const ww=m.walls[y*m.c+x],px=x*CELL,py=y*CELL;
    if(ww&1)segs.push(px,py,px+CELL,py);
    if(ww&8)segs.push(px,py,px,py+CELL);
    if(x===m.c-1&&(ww&2))segs.push(px+CELL,py,px+CELL,py+CELL);
    if(y===m.r-1&&(ww&4))segs.push(px,py+CELL,px+CELL,py+CELL);
  }
  const pass=(lw,st)=>{g.lineWidth=lw;g.strokeStyle=st;g.lineCap='round';g.beginPath();
    for(let i=0;i<segs.length;i+=4){g.moveTo(segs[i],segs[i+1]);g.lineTo(segs[i+2],segs[i+3]);}g.stroke();};
  pass(9,'#1c3140');pass(5,'#4f7286');pass(1.6,'rgba(150,195,215,0.35)');
  return {canvas:mc,w,h};
}

function pushSeg(ax,ay,bx,by){
  const dx=bx-ax,dy=by-ay,L2=dx*dx+dy*dy;
  let t=((P.x-ax)*dx+(P.y-ay)*dy)/L2;t=t<0?0:t>1?1:t;
  const px=ax+dx*t,py=ay+dy*t,ex=P.x-px,ey=P.y-py,d2=ex*ex+ey*ey,R=P.r+2.5;
  if(d2<R*R&&d2>0.0001){const push=(R-Math.sqrt(d2))/Math.sqrt(d2);P.x+=ex*push;P.y+=ey*push;}
}
function collide(){
  const cx=clamp((P.x/CELL)|0,0,M.c-1),cy=clamp((P.y/CELL)|0,0,M.r-1);
  for(let gy=cy-1;gy<=cy+1;gy++){if(gy<0||gy>=M.r)continue;
    for(let gx=cx-1;gx<=cx+1;gx++){if(gx<0||gx>=M.c)continue;
      const w=M.walls[gy*M.c+gx],x0=gx*CELL,y0=gy*CELL;
      if(w&1)pushSeg(x0,y0,x0+CELL,y0);
      if(w&4)pushSeg(x0,y0+CELL,x0+CELL,y0+CELL);
      if(w&8)pushSeg(x0,y0,x0,y0+CELL);
      if(w&2)pushSeg(x0+CELL,y0,x0+CELL,y0+CELL);}}
  P.x=clamp(P.x,P.r,M.c*CELL-P.r);P.y=clamp(P.y,P.r,M.r*CELL-P.r);
}
const cellOf=(x,y)=>{const cx=clamp((x/CELL)|0,0,M.c-1),cy=clamp((y/CELL)|0,0,M.r-1);return cy*M.c+cx;};

function updateWraith(w,ts){
  w.sway+=ts*4;w.wy=Math.sin(w.sway)*3;
  if(w.state==='spawn'){w.st-=ts;if(w.st<=0)w.state='active';return;}

  if(w.pt<=0){w.pt=0.35+Math.random()*0.1;
    const p=bfsPath(M,cellOf(w.x,w.y),cellOf(P.x,P.y));
    w.path=p||[cellOf(w.x,w.y)];w.pi=1;}
  let tx,ty;
  if(w.pi<w.path.length){
    const ci=w.path[w.pi];tx=(ci%M.c+.5)*CELL;ty=((ci/M.c)|0+.5)*CELL;
    if(dist2(tx,ty,w.x,w.y)<16)w.pi++;

function worldUpdate(dt){
  const ts=dt*G.timeScale;
  G.dashCd=Math.max(0,G.dashCd-dt);G.dashT=Math.max(0,G.dashT-dt);
  G.reveal=Math.max(0,G.reveal-dt);G.boostT=Math.max(0,G.boostT-dt);
  G.invuln=Math.max(0,G.invuln-dt);
  if(G.streakT>0){G.streakT-=dt;if(G.streakT<=0)G.streak=0;}
  G.shake=Math.max(0,G.shake-dt*26);
  G.ambT-=dt;
  if(G.ambT<0&&G.state==='playing'){G.ambT=rand(3,7);tone({type:'sine',f0:rand(180,420),f1:90,t:.5,v:.04});}

  if(G.trans){
    G.trans.t+=dt;const p=G.trans.t/G.trans.dur;
    if(G.trans.type==='descend'&&p>=0.5&&!G.trans.done){
      G.trans.done=true;buildFloor();
      splash('FLOOR '+String(G.floor).padStart(2,'0'),flavor());
    }
    if(p>=1)G.trans=null;
  }

  if(G.state==='playing'&&!G.trans){
    let ix=((keys['arrowright']||keys['d'])?1:0)-((keys['arrowleft']||keys['a'])?1:0);
    let iy=((keys['arrowdown']||keys['s'])?1:0)-((keys['arrowup']||keys['w'])?1:0);
    const il=Math.hypot(ix,iy)||1;ix/=il;iy/=il;
    const spd=playerSpeed()*(G.dashT>0?2.7:1)*(G.boostT>0?1.25:1);
    const vx=ix*spd+P.kbx,vy=iy*spd+P.kby;
    P.kbx*=Math.exp(-8*dt);P.kby*=Math.exp(-8*dt);
    if(ix||iy)P.face=Math.atan2(iy,ix);
    const stepLen=Math.hypot(vx,vy)*ts,n=Math.max(1,Math.ceil(stepLen/4));
    for(let i=0;i<n;i++){P.x+=vx*ts/n;P.y+=vy*ts/n;collide();}
    if(G.dashT>0)P.trail.push({x:P.x,y:P.y,t:0});
    for(const tr of P.trail)tr.t+=dt;
    P.trail=P.trail.filter(tr=>tr.t<0.35);

    for(const o of G.orbsArr){if(o.got)continue;o.t+=ts;
      if(dist2(o.x,o.y,P.x,P.y)<(P.r+10)*(P.r+10))collectOrb(o);}
    G.orbsArr=G.orbsArr.filter(o=>!o.got);
    const F=G.flareOrb;
    if(F&&!F.got){F.t+=ts;if(dist2(F.x,F.y,P.x,P.y)<(P.r+14)*(P.r+14))collectFlare();}
    G.portal.t+=ts;
    if(dist2(G.portal.x,G.portal.y,P.x,P.y)<(P.r+15)*(P.r+15))floorClear();
    for(const w of G.wraiths)updateWraith(w,ts);
  }

  const focus=G.state==='menu'?menuLight():P;
  const ww=M.c*CELL,wh=M.r*CELL;
  const tx=ww>vw?clamp(focus.x-vw/2,0,ww-vw):(ww-vw)/2;
  const ty=wh>vh?clamp(focus.y-vh/2,0,wh-vh):(wh-vh)/2;
  const k=1-Math.exp(-dt*5);
  G.cam.x+=(tx-G.cam.x)*k;G.cam.y+=(ty-G.cam.y)*k;
}
function updateFx(ts){
  for(const p of G.parts){p.t+=ts;p.x+=p.vx*ts;p.y+=p.vy*ts;p.vx*=Math.exp(-2.5*ts);p.vy*=Math.exp(-2.5*ts);}
  G.parts=G.parts.filter(p=>p.t<p.life);
  if(G.parts.length>380)G.parts.splice(0,G.parts.length-380);
  for(const f of G.floats)f.t+=ts;
  G.floats=G.floats.filter(f=>f.t<f.life);
  // portal embers
  if(M&&G.state!=='gameover'&&Math.random()<0.5){
    const pp=G.portal;
    G.parts.push({x:pp.x+rand(-14,14),y:pp.y+rand(-6,10),vx:rand(-8,8),vy:rand(-42,-18),t:0,life:rand(.8,1.6),size:rand(1,2.4),cr:120,cg:240,cb:220});
  }
  // dust motes
  const focus=G.state==='menu'?menuLight():P;
  if(!G.dust.length)for(let i=0;i<34;i++)G.dust.push({x:rand(0,M.c*CELL),y:rand(0,M.r*CELL),ph:rand(0,TAU)});
  const Rr=lightBase*1.35;
  for(const m of G.dust){
    m.ph+=ts;m.x+=Math.cos(m.ph*.7)*9*ts;m.y+=Math.sin(m.ph*.9)*9*ts-5*ts;
    if(dist2(m.x,m.y,focus.x,focus.y)>Rr*Rr){const a=rand(0,TAU),rr=rand(Rr*.4,Rr*.95);
      m.x=focus.x+Math.cos(a)*rr;m.y=focus.y+Math.sin(a)*rr;}
  }
}

}
