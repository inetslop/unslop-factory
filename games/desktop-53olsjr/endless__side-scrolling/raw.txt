      score+=speed*dt*.1*mult;
      parts.push({type:'ember',x:pxw-2.5,y:bird.y+rnd(-1,1),vx:-speed*.45,vy:rnd(-3,3),life:.4,max:.4,r:rnd(.5,1)});
    }
  }else if(state==='dying'){
    deathT+=rdt;shake=Math.max(0,shake-rdt*12);
    if(deathT>.9)showOver();
  }else{ // over — world keeps drifting behind the card
    dayP+=dt*.006;camX+=6*dt;}
  updateParts(dt);
  for(const p of pillars){if(p.x>camX-10&&p.x<camX+VWw+10&&Math.random()<dt*4){
    const[gt,gb]=gapYs(p),up=Math.random()<.5;
    parts.push({type:'mote',x:p.x+rnd(-4,4),y:up?gt-1:gb+1,vx:rnd(-1,1),vy:(up?-1:1)*rnd(2,6),life:rnd(.6,1.3),max:1.3,r:rnd(.4,.8)});}}
  pillars=pillars.filter(p=>p.x>camX-30);
  sparks=sparks.filter(s=>!s.got&&s.x>camX-12);
  gusts=gusts.filter(g=>g.x>camX-40);
  parts=parts.filter(p=>p.life>0);if(parts.length>500)parts.splice(0,parts.length-500);
  if(windGain){const tgt=state==='play'?windTarget:0;windGain.gain.value+=(tgt-windGain.gain.value)*Math.min(1,rdt*5);}
}
