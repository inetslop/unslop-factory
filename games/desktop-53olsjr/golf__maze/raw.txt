 for(let i=0;i<3;i++){ let bt=Infinity,bn=null;
  for(const w of G.maze.walls){ const r=rayAABB(x,y,dx,dy,w); if(r&&r.t>1e-4&&r.t<bt){bt=r.t;bn=r.n;} }
  const rem=total-used;
  if(bt>rem || bt===Infinity){ out.push({x:x+dx*rem,y:y+dy*rem}); break; }
  x+=dx*bt;y+=dy*bt;used+=bt; out.push({x,y});

/* ================= physics ================= */
function collideWall(w){
 const b=G.ball,cx=clamp(b.x,w.x,w.x+w.w),cy=clamp(b.y,w.y,w.y+w.h);
 let dx=b.x-cx,dy=b.y-cy;const d2=dx*dx+dy*dy;if(d2>=BR*BR)return;
 let nx,ny,pen;
 if(d2>1e-9){const d=Math.sqrt(d2);nx=dx/d;ny=dy/d;pen=BR-d;}
 else{const l=b.x-w.x,r=w.x+w.w-b.x,tp=b.y-w.y,bt=w.y+w.h-b.y,mn=Math.min(l,r,tp,bt);
  if(mn===l){nx=-1;ny=0;pen=l+BR;}else if(mn===r){nx=1;ny=0;pen=r+BR;}
  else if(mn===tp){nx=0;ny=-1;pen=tp+BR;}else{nx=0;ny=1;pen=bt+BR;}}
 b.x+=nx*pen;b.y+=ny*pen;
 const vn=b.vx*nx+b.vy*ny;
 if(vn<0){b.vx-=(1+REST)*vn*nx;b.vy-=(1+REST)*vn*ny;const imp=-vn;
  if(imp>90&&G.time-G.lastTick>.035){Snd.tick(imp/G.maxShot);G.lastTick=G.time;}
  if(imp>620){G.shake=Math.min(13,G.shake+imp*.012);spawnDust(cx,cy,nx,ny,imp);}}}
function stepBall(dt){
 const b=G.ball;b.x+=b.vx*dt;b.y+=b.vy*dt;
 for(const w of G.maze.walls)collideWall(w);
 if(G.phase!=='rolling'&&G.phase!=='start')return;
 const c=G.cup,dx=c.x-b.x,dy=c.y-b.y,d=Math.hypot(dx,dy)||1e-6;
 let sp=Math.hypot(b.vx,b.vy);
 if(d<c.r+46&&sp<CAPTURE_V*1.05&&sp>0){const pull=300*(1-d/(c.r+46));b.vx+=dx/d*pull*dt;b.vy+=dy/d*pull*dt;sp=Math.hypot(b.vx,b.vy);}
 if(d<c.r-8&&sp<CAPTURE_V){startSink();return;}
 if(d<c.r*1.5&&sp>CAPTURE_V+260&&G.time>G.lipCd){G.lipCd=G.time+.35;
  const sgn=Math.random()<.5?-1:1,tx=-dy/d*sgn,ty=dx/d*sgn;
  b.vx+=tx*140;b.vy+=ty*140;Snd.lip();addText('LIP!',c.x,c.y-64);}
 if(sp>0){const dec=(FA+FB*sp)*dt;
  if(dec>=sp){b.vx=0;b.vy=0;}else{const k=1-dec/sp;b.vx*=k;b.vy*=k;}}
}
function shotDist(p){const v=p*G.maxShot;if(v<5)return 0;return(v-(FA/FB)*Math.log(1+FB*v/FA))/FB;}

let G_lipCd=0;Object.defineProperty(G,'lipCd',{get(){return G_lipCd;},set(v){G_lipCd=v;}});
function update(dt){
 G.shake*=Math.exp(-7*dt);if(G.shake<.1)G.shake=0;
 G.flash=Math.max(0,G.flash-3*dt);
 if(G.toastT>0){G.toastT-=dt;if(G.toastT<=0)toastEl.classList.remove('show');}
 // particles
 for(let i=G.parts.length-1;i>=0;i--){const p=G.parts[i];p.t+=dt;
  if(p.t>p.life){G.parts.splice(i,1);continue;}
  p.x+=p.vx*dt;p.y+=p.vy*dt;if(p.grav)p.vy+=980*dt;p.vx*=(1-1.4*dt);p.rot+=p.vr*dt;}
 for(let i=G.texts.length-1;i>=0;i--){const t=G.texts[i];t.t+=dt;if(t.t>t.life)G.texts.splice(i,1);}
 const b=G.ball,moving=Math.abs(b.vx)>1||Math.abs(b.vy)>1;
 if(G.phase==='rolling'||G.phase==='start'){
  if(moving){
   G.rollT+=dt;G.acc+=dt;const h=1/240;let n=0;
   while(G.acc>=h&&n++<30&&(G.phase==='rolling'||G.phase==='start')){stepBall(h);G.acc-=h;}
   if(moving&&Math.hypot(b.vx,b.vy)>80){G.trail.push({x:b.x,y:b.y});if(G.trail.length>22)G.trail.shift();}
  }else{
   if(G.phase==='start'){G.aiT-=dt;if(G.aiT<=0){
     G.aim={ang:Math.atan2(G.cup.y-b.y,G.cup.x-b.x)+rand(-.45,.45),power:rand(.5,.95)};fireShot();}}
   else{G.phase='aim';refreshUI();}
  }
  if(G.rollT>12){b.vx=0;b.vy=0;}
 }else if(G.phase==='sinking'){
  G.sinkT+=dt/.55;const t=Math.min(1,G.sinkT),e=t*t,sp2=G.sinkT*9;
  b.x=lerp(G.sinkFrom.x,G.cup.x,e)+Math.cos(sp2)*(1-e)*30;
  b.y=lerp(G.sinkFrom.y,G.cup.y,e)+Math.sin(sp2)*(1-e)*30;
  G.flagDown=Math.min(1,G.flagDown+dt*2.2);
  if(t>=1&&!G.sinkDone){G.sinkDone=true;onSunk();}
 }else if(G.phase==='banner'){
  G.bannerTimer-=dt;if(G.bannerTimer<=0){bannerEl.classList.remove('show');advanceHole();}
 }
}

function rayAABB(px,py,dx,dy,r){
 let tx1=-Infinity,tx2=Infinity;
 if(Math.abs(dx)<1e-9){if(px<r.x||px>r.x+r.w)return null;}
 else{tx1=(r.x-px)/dx;tx2=(r.x+r.w-px)/dx;if(tx1>tx2){const t=tx1;tx1=tx2;tx2=t;}}
 let ty1=-Infinity,ty2=Infinity;
 if(Math.abs(dy)<1e-9){if(py<r.y||py>r.y+r.h)return null;}
 else{ty1=(r.y-py)/dy;ty2=(r.y+r.h-py)/dy;if(ty1>ty2){const t=ty1;ty1=ty2;ty2=t;}}
 const tmin=Math.max(tx1,ty1),tmax=Math.min(tx2,ty2);
 if(tmax<Math.max(tmin,0))return null;
 let t;if(tmin>1e-4)t=tmin;else if(tmax>1e-4)t=tmax;else return null;
 const n=tx1>=ty1?[dx>0?-1:1,0]:[0,dy>0?-1:1];
 return{t,n};
}
function tracePath(ang,p){
 const b=G.ball;let x=b.x,y=b.y,dx=Math.cos(ang),dy=Math.sin(ang);
 const total=shotDist(p)*.95;let used=0;const out=[{x,y}];
 for(let i=0;i<3;i++){
  let bt=Infinity,bn=null;
  for(const w of G.maze.walls){const r=rayAABB(x,y,dx,dy,w);if(r&&r.t>1e-4&&r.t<bt){bt=r.t;bn=r.n;}}
  const rem=total-used;
  if(!bn||bt>rem){out.push({x:x+dx*rem,y:y+dy*rem});break;}
  x+=dx*bt;y+=dy*bt;used+=bt;out.push({x,y});
  if(bn[0])dx=-dx;else dy=-dy;}
 const res=[out[0]];
 for(let i=1;i<out.length;i++){const a0=out[i-1],a1=out[i],d=Math.hypot(a1.x-a0.x,a1.y-a0.y);
  const n=Math.max(1,Math.floor(d/18));
  for(let k=1;k<=n;k++)res.push({x:lerp(a0.x,a1.x,k/n),y:lerp(a0.y,a1.y,k/n)});}
 return res;
}
