/* ================= PHYSICS (deterministic core) ================= */
function microStep(b,pads){
  b.prevX=b.x;
  b.x+=b.vx/N; b.y+=b.vy/N;
  let ev=null;
  const top=COURT.y+BR,bot=COURT.y+COURT.h-BR;
  if(b.y<top){b.y=top+(top-b.y);b.vy=-b.vy;ev='wall';}
  else if(b.y>bot){b.y=bot+(b.y-bot);b.vy=-b.vy;ev='wall';}
  if(b.vx<0 && b.x-BR<=FL && b.prevX-BR>FL && Math.abs(b.y-pads[0].y)<=PHH+BR){
    const rel=clamp((b.y-pads[0].y)/PHH,-1,1);
    b.speed=Math.min(b.speed*SPD_UP,MAX_SPD);
    b.x=FL+BR+(FL+BR-b.x);
    const a=rel*MAX_ANG; b.vx=Math.cos(a)*b.speed; b.vy=Math.sin(a)*b.speed; ev='p0';
  }else if(b.vx>0 && b.x+BR>=FR && b.prevX+BR<FR && Math.abs(b.y-pads[1].y)<=PHH+BR){
    const rel=clamp((b.y-pads[1].y)/PHH,-1,1);
    b.speed=Math.min(b.speed*SPD_UP,MAX_SPD);
    b.x=FR-BR-(FR-BR-b.x);
    const a=rel*MAX_ANG; b.vx=-Math.cos(a)*b.speed; b.vy=Math.sin(a)*b.speed; ev='p1';
  }
  if(b.x<COURT.x-14)return 's1';
  if(b.x>COURT.x+COURT.w+14)return 's0';
  return ev;
}
function simulateTicks(b,n,pads){
  const pts=[];let sc=null;
  for(let t=0;t<n;t++){
    pts.push({x:b.x,y:b.y});
    for(let i=0;i<N;i++){const ev=microStep(b,pads);if(ev==='s0'||ev==='s1'){sc=ev;break;}}
    if(sc)break;
  }
  return {pts,sc};
}
function buildTickPath(){
  const b={x:ball.x,y:ball.y,vx:ball.vx,vy:ball.vy,speed:ball.speed};
  const pads={0:{y:P[0].y},1:{y:P[1].y}};
  const path=[{x:b.x,y:b.y}],events=[];let sc=null;
  for(let i=0;i<N;i++){
    const ev=microStep(b,pads);
    path.push({x:b.x,y:b.y});
    if(ev)events.push({i:i+1,ev});
    if(ev==='s0'||ev==='s1'){sc=(ev==='s0'?0:1);break;}
  }
  return {path,events,final:{x:b.x,y:b.y,vx:b.vx,vy:b.vy,speed:b.speed},score:sc};
}

/* ================= AI ================= */
function isAI(m){return G.attract||(G.mode==='cpu'&&m===1);}
function aiChoose(m){
  const d=DIFFS[G.attract?'normal':G.diff];
  const b={x:ball.x,y:ball.y,vx:ball.vx,vy:ball.vy,speed:ball.speed};
  const pads={0:{y:P[0].y},1:{y:P[1].y}};
  const toward = m===1 ? ball.vx>0 : ball.vx<0;
  let target;
  if(toward){
    let intercept=null;
    for(let i=0;i<80*N;i++){
      const ev=microStep(b,pads);
      if(ev==='s0'||ev==='s1')break;
      if(m===1? (b.vx>0&&b.x+BR>=FR) : (b.vx<0&&b.x-BR<=FL)){intercept=b.y;break;}
    }
    if(intercept==null)intercept=clamp(b.y,YMIN,YMAX);
    const CTR=COURT.y+COURT.h/2;
    let r=clamp((CTR-P[1-m].y)/(COURT.h/2),-1,1)*0.75*d.aim+(Math.random()-.5)*.3;
    r=clamp(r,-.85,.85);
    intercept+=(Math.random()-.5)*2*d.err;
    target=intercept-r*PHH;
  }else{
    target=clamp(b.y+ball.vy*3,YMIN,YMAX);
    if(Math.random()<d.err/60)target=P[m].y;
  }
  const diff=target-P[m].y;
  return Math.abs(diff)<14?0:Math.sign(diff);
}

/* ================= UPDATE ================= */
function update(dt){
  tGlobal+=dt;
  for(const d of dust){d.y-=d.v*dt;d.x+=Math.sin(tGlobal*.001*d.wf+d.ph)*.06;if(d.y<-20){d.y=H+20;d.x=Math.random()*(W+300)-150;}}
  G.shake*=Math.pow(.9,dt/16);if(G.shake<.1)G.shake=0;
  for(let i=0;i<2;i++){
    G.scoreAnim[i]=Math.max(0,G.scoreAnim[i]-dt*.0022);
    P[i].flash=Math.max(0,P[i].flash-dt*.004);
    if(P[i].ghostA>0)P[i].ghostA-=dt*.003;
  }
  G.flashL=Math.max(0,G.flashL-dt*.0025);G.flashR=Math.max(0,G.flashR-dt*.0025);
  G.wallTop=Math.max(0,G.wallTop-dt*.004);G.wallBot=Math.max(0,G.wallBot-dt*.004);
  for(let i=parts.length-1;i>=0;i--){const p=parts[i];
    p.life+=dt;if(p.life>p.ttl){parts.splice(i,1);continue;}
    p.x+=p.vx*dt/1000;p.y+=p.vy*dt/1000;p.vy+=(p.grav||0)*dt/1000;
    if(p.rot!=null)p.rot+=p.vr*dt/1000;
  }
  if(G.paused)return;

  switch(G.phase){
    case 'menu': break; // attract flows through preTick/moveAnim/ballTick below
    case 'preTick':
      if(isAI(G.mover)){G.thinkT+=dt;if(G.thinkT>=G.thinkDelay)commitMove(aiChoose(G.mover));}
      break;
    case 'moveAnim':{
      G.moveT+=dt;const p=Math.min(1,G.moveT/95),e=1-Math.pow(1-p,3);
      P[G.mover].y=P[G.mover].from+(P[G.mover].to-P[G.mover].from)*e;
      if(p>=1){P[G.mover].y=P[G.mover].to;advanceAfterMove();}
      break;}
    case 'ballTick':{
      const tk=G.tick;tk.t+=dt;const p=Math.min(1,tk.t/tk.dur);
      const f=p*(tk.path.length-1),i=Math.floor(f),fr=f-i;
      const a=tk.path[i],b=tk.path[Math.min(i+1,tk.path.length-1)];
      dispX=a.x+(b.x-a.x)*fr;dispY=a.y+(b.y-a.y)*fr;
      trail.push({x:dispX,y:dispY});if(trail.length>16)trail.shift();
      while(tk.i<tk.events.length&&tk.events[tk.i].i<=f+.001){onTickEvent(tk.events[tk.i]);tk.i++;}
      if(p>=1){ball={...tk.final};dispX=ball.x;dispY=ball.y;
        if(tk.score!=null)doScore(tk.score);else{G.mover=0;startPreTick();}}
      break;}
    case 'point':
      G.pointT+=dt;if(G.pointT>980)endPoint();break;
    case 'over':
      confettiT-=dt;if(confettiT<0){confettiBurst(false);confettiT=380;}
      break;
  }
}

function spawnSparks(x,y,col,n,spd){
  for(let i=0;i<n;i++){const a=Math.random()*Math.PI*2,s=spd*(.4+Math.random()*.8);
    parts.push({kind:0,x,y,vx:Math.cos(a)*s,vy:Math.sin(a)*s,life:0,ttl:300+Math.random()*250,size:1.5+Math.random()*1.5,col});}

function confettiBurst(big){
  const side=G.pointSide,n=big?90:14;
  for(let i=0;i<n;i++){
    const inHalf=Math.random()<.7;
    const x=inHalf?(side===0?COURT.x+COURT.w/2:COURT.x)+Math.random()*(COURT.w/2):COURT.x+Math.random()*COURT.w;

  }
}

/* ================= RENDER ================= */
function fit(){
  const w=innerWidth,h=innerHeight,d=Math.min(2,devicePixelRatio||1);
  if(cv.width!==Math.round(w*d)||cv.height!==Math.round(h*d)){cv.width=Math.round(w*d);cv.height=Math.round(h*d);}
  return {w,h,d};
}
function rr(x,y,w,h,r){ctx.beginPath();ctx.moveTo(x+r,y);ctx.arcTo(x+w,y,x+w,y+h,r);ctx.arcTo(x+w,y+h,x,y+h,r);ctx.arcTo(x,y+h,x,y,r);ctx.arcTo(x,y,x+w,y,r);ctx.closePath();}
function tri(x,y,s,dir){ctx.beginPath();ctx.moveTo(x,y+dir*s);ctx.lineTo(x-s*.9,y-dir*s*.7);ctx.lineTo(x+s*.9,y-dir*s*.7);ctx.closePath();ctx.fill();}

function render(){
  const {w,h,d}=fit();
  ctx.setTransform(d,0,0,d,0,0);
  // screen-space backdrop
  let g=ctx.createLinearGradient(0,0,0,h);g.addColorStop(0,'#0a1020');g.addColorStop(1,'#060a12');
  ctx.fillStyle=g;ctx.fillRect(0,0,w,h);
  const pulse=.5+.5*Math.sin(tGlobal*.0012);
  let gl=ctx.createRadialGradient(w*.1,h*.55,0,w*.1,h*.55,Math.max(w,h)*.45);
  gl.addColorStop(0,'rgba(255,176,58,'+(0.07+pulse*0.03)+')');gl.addColorStop(1,'rgba(255,176,58,0)');
  ctx.fillStyle=gl;ctx.fillRect(0,0,w,h);
  let gr=ctx.createRadialGradient(w*.9,h*.45,0,w*.9,h*.45,Math.max(w,h)*.45);
  gr.addColorStop(0,'rgba(69,214,255,'+(0.07+(1-pulse)*0.03)+')');gr.addColorStop(1,'rgba(69,214,255,0)');
  ctx.fillStyle=gr;ctx.fillRect(0,0,w,h);

  // logical transform + shake
  const s=Math.min(w/W,h/H),ox=(w-W*s)/2,oy=(h-H*s)/2;
  const shx=(Math.random()-.5)*2*G.shake,shy=(Math.random()-.5)*2*G.shake;
  ctx.setTransform(d*s,0,0,d*s,(ox+shx)*d,(oy+shy)*d);

  // dust
  for(const p of dust){ctx.globalAlpha=p.a;ctx.fillStyle='#aac4ff';ctx.beginPath();ctx.arc(p.x,p.y,p.r,0,7);ctx.fill();}
  ctx.globalAlpha=1;

  drawCourt();drawXray();drawScores();drawPaddles();drawBall();drawParticles();drawTurnUI();

  if(G.paused){
    ctx.fillStyle='rgba(7,11,20,.62)';ctx.fillRect(-40,-30,W+80,H+60);
    ctx.textAlign='center';ctx.textBaseline='middle';
    ctx.fillStyle='#eef4ff';ctx.font='52px '+FDISP;ctx.fillText('PAUSED',W/2,H/2-16);
    ctx.font='11px '+FMONO;ctx.fillStyle='#7d8ca6';ctx.fillText('P — RESUME · R — REMATCH · ESC — MENU',W/2,H/2+24);
  }
}
