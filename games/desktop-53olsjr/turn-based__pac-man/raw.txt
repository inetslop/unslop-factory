let pac={c:PAC_START.c,r:PAC_START.r,dir:0,pending:0,mv:null,mouthT:0};
let ghosts=[];
let frightTurns=0, chain=0, allowRev=false;
let mode='scatter', modeIdx=0, modeLeft=4;
const MODE_SEQ=[['scatter',4],['chase',8],['scatter',5],['chase',10],['scatter',5],['chase',1e9]];
let chargeEvery=0, chargeLeft=0;

let T=0;                 // game clock (pauses cleanly)
let particles=[], popups=[], banner=null;
let shake={t:0,dur:0,mag:0};
let flashA=0, flashCol='255,60,80';
let attractT=0, attractDeadT=0, eatingList=[];

const heldKeys={1:false,2:false,3:false,4:false};

/* ---------- helpers ---------- */
function cellAt(c,r){ if(r===TUNNEL_R) c=((c%W)+W)%W; return MAZE[r][c]; }
function isWallFor(c,r,ghost){ const ch=cellAt(c,r); if(ch==='#')return true; if(ch==='-')return !ghost; return false; }
function wrapStep(c,r,d){
  let nc=c+DIRS[d].x, nr=r+DIRS[d].y;
  if(nr===TUNNEL_R) nc=((nc%W)+W)%W;
  else if(nc<0||nc>=W||nr<0||nr>=H) return null;
  return {c:nc,r:nr};
}
function pacCan(c,r,d){ const s=wrapStep(c,r,d); return s&&!isWallFor(s.c,s.r,false); }
function ghostCan(c,r,d){ const s=wrapStep(c,r,d); return s&&!isWallFor(s.c,s.r,true); }
function isWrap(f,t){ return f.r===TUNNEL_R&&t.r===TUNNEL_R&&Math.abs(f.c-t.c)>1; }
function frightFor(lv){ return Math.max(3,7-(lv-1)); }
function hesFor(lv){ return lv<=2?0.25: lv<=4?0.12:0; }
function chargeEveryFor(lv){ return lv===2?8: lv===3?7: lv===4?6: lv>=5?5:0; }

/* ---------- BFS field (ghost passability) ---------- */
function field(tc,tr){
  const d=new Int16Array(W*H); d.fill(-1);
  tc=Math.max(0,Math.min(W-1,tc)); tr=Math.max(0,Math.min(H-1,tr));
  const q=[tr*W+tc]; let qi=0; d[q[0]]=0;
  while(qi<q.length){
    const i=q[qi++], c=i%W, r=(i/W)|0;
    for(let k=0;k<4;k++){
      const dd=DIR_ORDER[k], nc=c+DIRS[dd].x, nr=r+DIRS[dd].y;
      let ncc=nc;
      if(nr===TUNNEL_R) ncc=((ncc%W)+W)%W;
      else if(ncc<0||ncc>=W||nr<0||nr>=H) continue;
      if(isWallFor(ncc,nr,true)) continue;
      const j=nr*W+ncc;
      if(d[j]===-1){ d[j]=d[i]+1; q.push(j); }
    }
  }
  return d;
}

/* ---------- ghost AI ---------- */
function aheadCell(c,r,d,n){
  let nc=c+DIRS[d].x*n, nr=r+DIRS[d].y*n;
  if(nr===TUNNEL_R) nc=((nc%W)+W)%W;
  return {c:Math.max(0,Math.min(W-1,nc)), r:Math.max(0,Math.min(H-1,nr))};
}
function chaseTarget(g){
  const p=pac;
  if(g.k===0) return {c:p.c,r:p.r};
  if(g.k===1) return aheadCell(p.c,p.r,p.dir||4,2);
  if(g.k===2){
    const a=aheadCell(p.c,p.r,p.dir||4,2), b=ghosts[0];
    let c=a.c*2-b.c, r=a.r*2-b.r;
    if(c<0||c>=W||r<0||r>=H) return {c:p.c,r:p.r};
    return {c:c,r:r};
  }
  const dx=g.c-p.c, dy=g.r-p.r;
  return (dx*dx+dy*dy>36) ? {c:p.c,r:p.r} : g.corner;
}
function ghostStep(g){
  if(g.state==='eaten') return null;
  let opts=[];
  for(const d of DIR_ORDER) if(ghostCan(g.c,g.r,d)){ const s=wrapStep(g.c,g.r,d); opts.push({c:s.c,r:s.r,d}); }
  if(!opts.length) return null;
  if(frightTurns>0){
    let pool=opts;
    if(g.dir&&opts.length>1){ const nr=opts.filter(o=>o.d!==OPP[g.dir]); if(nr.length)pool=nr; }
    return pool[(Math.random()*pool.length)|0];
  }
  let t = (g.state==='house') ? EXIT_CELL : chaseTarget(g);
  const f=field(t.c,t.r);
  let pool=opts;
  if(g.dir&&!allowRev&&opts.length>1){
    const nr=opts.filter(o=>o.d!==OPP[g.dir]); if(nr.length)pool=nr;
  }
  let best=null,bd=1e9;
  for(const o of pool){
    const dd=f[o.r*W+o.c]<0?1e9:f[o.r*W+o.c];
    if(dd<bd){ bd=dd; best=o; }
  }
  const hes=hesFor(level);
  if(best&&hes>0&&Math.random()<hes&&pool.length>1){
    const alt=pool.filter(o=>o!==best);
    best=alt[(Math.random()*alt.length)|0];
  }
  return best;
}

/* ---------- setup / reset ---------- */
function buildDots(){
  dots=[]; let n=0;
  for(let r=0;r<H;r++){ dots[r]=[];
    for(let c=0;c<W;c++){
      const ch=MAZE[r][c];
      if(ch==='.'){dots[r][c]=1;n++;}
      else if(ch==='o'){dots[r][c]=2;n++;}
      else dots[r][c]=0;
    }
  }
  dotsLeft=n;
}
function resetPositions(){
  pac.c=PAC_START.c; pac.r=PAC_START.r; pac.dir=0; pac.pending=0; pac.mv=null; pac.mouthT=0;
  ghosts=GHOST_DEFS.map((d,i)=>({
    k:i,name:d.name,color:d.color,corner:d.corner,
    c:d.spawn.c,r:d.spawn.r,dir:0,state:d.outside?'out':'house',mv:null,fly:null,alp:1,bobT:Math.random()*6
  }));
}
function resetMaze(lv){
  buildDots();
  frightTurns=0; chain=0; allowRev=false;
  mode='scatter'; modeIdx=0; modeLeft=MODE_SEQ[0][1];
  chargeEvery=chargeEveryFor(lv); chargeLeft=chargeEvery;
}
function startRun(){
  score=0; lives=3; level=1;
  resetMaze(1); resetPositions();
  particles.length=0; popups.length=0; banner=null; eatingList=[];
  hideOverlay(); paused=false;
  sfx.start();
  setState('ready');
  hudAll();
}
function toMenu(){
  paused=false; state='menu'; sub='await'; attractT=0; attractDeadT=0;
  buildDots(); resetPositions(); particles.length=0; popups.length=0; banner=null; eatingList=[];
  showStartOverlay(); hudAll();
}
function setState(s){
  state=s;
  if(s==='ready'){ sub='await'; }
  else if(s==='play'){ sub='await'; setChip('you'); }
  else if(s==='dying'){ sub='dead'; setChip('dim','OUCH'); }
  else if(s==='levelclear'){ sub='clear'; setChip('dim','CLEAR!'); }
  else if(s==='gameover'){ setChip('dim','GAME OVER'); showGameOverOverlay(); }
}

/* ---------- turn flow ---------- */
function modeTick(){
  modeLeft--;
  if(modeLeft<=0){
    modeIdx=Math.min(modeIdx+1,MODE_SEQ.length-1);
    mode=MODE_SEQ[modeIdx][0]; modeLeft=MODE_SEQ[modeIdx][1];
    allowRev=true;
  }
}
function startPacMove(dir){
  const s=wrapStep(pac.c,pac.r,dir);
  if(!s||isWallFor(s.c,s.r,false)) return false;
  pac.dir=dir;
  pac.mv={fc:pac.c,fr:pac.r,tc:s.c,tr:s.r,t0:T,dur:PAC_MS,wrap:isWrap({c:pac.c,r:pac.r},s)};
  sub='pacM';
  if(pac.pending===dir) pac.pending=0;
  return true;
}
function pacArrived(){
  pac.c=pac.mv.tc; pac.r=pac.mv.tr; pac.mv=null;
  const d=dots[pac.r][pac.c];
  if(d){
    dots[pac.r][pac.c]=0; dotsLeft--;
    if(state!=='menu'){
      if(d===1){ score+=10; sfx.dot(); spark(pac.c,pac.r,5,'#ffd9a3',.6); }
      else { score+=50; frightTurns=frightFor(level); chain=0; allowRev=true; sfx.power(); burst(pac.c,pac.r,14,'#fff1cf'); }
      if(score>hiScore){ hiScore=score; elHi.textContent=hiScore; bumpEl(elHi); }
    } else spark(pac.c,pac.r,3,'#ffd9a3',.4);
  }
  // collisions at destination
  for(const g of ghosts){
    if(g.state==='eaten') continue;
    if(g.c===pac.c&&g.r===pac.r){
      if(state==='menu'){ attractHit(); return; }
      if(frightTurns>0){ eatGhost(g); return; }
      die(); return;
    }
  }
  if(state==='play'&&dotsLeft<=0){ levelClearNow(); return; }
  if(state==='menu'&&dotsLeft<=0){ buildDots(); resetPositions(); sub='await'; return; }
  startGhostPhase();
}
function startGhostPhase(){
  sub='ghostM'; setChip('ghosts');
  if(state==='play') sfx.tick();
  for(const g of ghosts){
    const m=ghostStep(g);
    if(m){
      g.mv={fc:g.c,fr:g.r,tc:m.c,tr:m.r,t0:T,dur:GHOST_MS,wrap:isWrap({c:g.c,r:g.r},m)};
      g.dir=m.d;
    } else g.mv=null;
  }
}
function ghostsArrived(){
  let ateList=[];
  for(const g of ghosts){
    if(g.mv){ g.c=g.mv.tc; g.r=g.mv.tr; g.mv=null; }
    if(g.state==='house'&&g.c===EXIT_CELL.c&&g.r===EXIT_CELL.r) g.state='out';
  }
  for(const g of ghosts){
    if(g.state==='eaten') continue;
    if(g.c===pac.c&&g.r===pac.r){
      if(frightTurns>0) ateList.push(g); else { die(); return; }
    }
  }
  if(state==='play'){
    if(frightTurns>0){ frightTurns--; if(frightTurns===0) chain=0; }
    else modeTick();
    if(chargeEvery>0&&chargeLeft>0){
      chargeLeft--;
      if(chargeLeft===0){
        chargeLeft=chargeEvery;
        banner={text:'⚡ GHOSTS CHARGE!',color:'#ff5d5d',t0:T,dur:1.3,y:1.9};
        shake={t:T,dur:.4,mag:5}; sfx.charge();
      }
    }
    hudCharge();
  }
  if(ateList.length){ for(const g of ateList) eatGhost(g); sub='eat'; return; }
  setChip('you'); sub='await';
}
function eatGhost(g){
  const pts=Math.min(1600,200*Math.pow(2,Math.min(chain,3)));
  chain++;
  if(state!=='menu'){ score+=pts; sfx.eatGhost(); }
  popup(g.c,g.r,'+'+pts,g.color);
  burst(g.c,g.r,16,g.color);
  g.state='eaten';
  g.fly={fc:g.c,fr:g.r,t0:T,dur:EAT_FLY_MS};
  eatingList.push(g);
  sub='eat'; setChip('ghosts');
}
function finishEat(){
  for(const g of eatingList){
    if(g.fly){ g.c=HOUSE_C.c; g.r=HOUSE_C.r; g.dir=1; g.state='house'; g.fly=null; }
  }
  eatingList.length=0;
  startGhostPhase();
}
function die(){
  if(state==='menu'){ attractHit(); return; }
  shake={t:T,dur:.5,mag:6}; flashA=.45; flashCol='255,60,80'; sfx.death();
  setState('dying');
}
function afterDeath(){
  lives--;
  hudLives();
  if(lives<=0){
    try{ localStorage.setItem('tpac_hi',String(hiScore)); }catch(e){}
    setState('gameover');
  } else {
    resetPositions(); frightTurns=0; chain=0;
    setState('ready');
  }
}
function levelClearNow(){
  sfx.levelup(); flashA=.35; flashCol='120,240,255';
  banner={text:'LEVEL CLEAR',color:'#7dfcff',t0:T,dur:1.6,y:10.6};
  for(let i=0;i<40;i++) spark(Math.random()*W|0,Math.random()*H|0,1,['#ffe14d','#3df2ff','#ff9de0','#ffa14a'][i%4],1.4);
  setState('levelclear');
}
function afterClear(){
  level++;
  resetMaze(level); resetPositions();
  banner={text:'LEVEL '+level,color:'#ffe14d',t0:T,dur:1.2,y:3};
  hudAll();
  setState('ready');
}

/* ---------- attract mode (menu) ---------- */
function nearestDotDist(c,r){
  let best=1e9;
  for(let rr=0;rr<H;rr++)for(let cc=0;cc<W;cc++){
    if(dots[rr][cc]){ const dd=Math.abs(cc-c)+Math.abs(rr-r); if(dd<best)best=dd; }
  }
  return best;
}
function attractTick(){
  if(attractDeadT){ if(T>attractDeadT){ attractDeadT=0; buildDots(); resetPositions(); sub='await'; } return; }
  if(sub!=='await') return;
  attractT+=1/60;
  if(attractT<0.55) return;
  attractT=0;
  let opts=[];
  for(const d of DIR_ORDER) if(pacCan(pac.c,pac.r,d)) opts.push(d);
  if(!opts.length) return;
  let best=opts[0], bs=1e9;
  for(const d of opts){
    const s=wrapStep(pac.c,pac.r,d);
    const nd=nearestDotDist(s.c,s.r);
    if(nd<bs-2){ bs=nd; best=d; }
  }
  pac.pending=0;
  startPacMove(best);
}
function attractHit(){ sub='dead'; attractDeadT=T+1.0; }

/* ---------- particles / popups ---------- */
function spark(c,r,n,col,spd){
  for(let i=0;i<n;i++){
    const a=Math.random()*Math.PI*2, v=(0.4+Math.random())*(spd||1);
    particles.push({x:c+.5,y:r+.5,vx:Math.cos(a)*v,vy:Math.sin(a)*v,t:T,dur:.35+Math.random()*.3,col,sz:.06+Math.random()*.07});
  }
}
function burst(c,r,n,col){
  for(let i=0;i<n;i++){
    const a=(i/n)*Math.PI*2+Math.random()*.4, v=1.2+Math.random()*2;
    particles.push({x:c+.5,y:r+.5,vx:Math.cos(a)*v,vy:Math.sin(a)*v,t:T,dur:.5+Math.random()*.4,col,sz:.08+Math.random()*.1});
  }
}
function popup(c,r,text,col){ popups.push({x:c+.5,y:r+.2,text,col,t:T,dur:.95}); }

/* ---------- canvas / rendering ---------- */
const cvs=document.getElementById('game'), ctx=cvs.getContext('2d');
const mcv=document.createElement('canvas'), mctx=mcv.getContext('2d');
let S=20, OX=20, OY=20, DPR=1;

function fit(){
  const wrap=document.getElementById('board');
  const rw=wrap.clientWidth-14, rh=wrap.clientHeight-14;
  DPR=Math.min(2,window.devicePixelRatio||1);
  S=Math.max(6,Math.min((rw)/W,(rh)/H));
  OX=S*0.55; OY=S*0.55;
  const cw=W*S+OX*2, ch=H*S+OY*2;
  cvs.style.width=cw+'px'; cvs.style.height=ch+'px';
  cvs.width=Math.round(cw*DPR); cvs.height=Math.round(ch*DPR);
  ctx.setTransform(DPR,0,0,DPR,0,0);
  buildMazeLayer();
}
function buildMazeLayer(){
  mcv.width=cvs.width; mcv.height=cvs.height;
  mctx.setTransform(DPR,0,0,DPR,0,0);
  mctx.clearRect(0,0,cvs.width,cvs.height);
  // wall edges
  const segs=[];
  for(let r=0;r<H;r++)for(let c=0;c<W;c++){
    if(MAZE[r][c]!=='#') continue;
    if(c+1<W&&MAZE[r][c+1]==='#') segs.push([c,r,c+1,r]);
    if(r+1<H&&MAZE[r+1][c]==='#') segs.push([c,r,c,r+1]);
  }
  const px=(c)=>OX+(c+.5)*S, py=(r)=>OY+(r+.5)*S;
  mctx.lineCap='round'; mctx.lineJoin='round';
  mctx.save();
  mctx.shadowColor='rgba(70,130,255,.85)'; mctx.shadowBlur=S*.6;
  mctx.strokeStyle='#1b2f8a'; mctx.lineWidth=S*.34;
  mctx.beginPath();
  for(const s of segs){ mctx.moveTo(px(s[0]),py(s[1])); mctx.lineTo(px(s[2]),py(s[3])); }
  mctx.stroke();
  mctx.restore();
  mctx.strokeStyle='#5b7cff'; mctx.lineWidth=S*.14;
  mctx.beginPath();
  for(const s of segs){ mctx.moveTo(px(s[0]),py(s[1])); mctx.lineTo(px(s[2]),py(s[3])); }
  mctx.stroke();
  // ghost door
  const dx=OX+(DOOR.c+.5)*S, dy=OY+(DOOR.r+.5)*S;
  mctx.save();
  mctx.shadowColor='rgba(255,184,222,.9)'; mctx.shadowBlur=S*.4;
  mctx.fillStyle='#ffb8de';
  mctx.fillRect(dx-S*.42,dy-S*.09,S*.84,S*.18);
  mctx.restore();
}

function drawBg(now){
  const g=ctx.createLinearGradient(0,0,0,cvs.height/DPR);
  g.addColorStop(0,'#0a1030'); g.addColorStop(1,'#060a1c');
  ctx.fillStyle=g; ctx.fillRect(0,0,cvs.width/DPR,cvs.height/DPR);
  // ambient pulse behind maze
  const a=0.10+0.05*Math.sin(now/900);
  const rg=ctx.createRadialGradient(OX+W*S/2,OY+H*S/2,S*.5,OX+W*S/2,OY+H*S/2,W*S*.75);
  rg.addColorStop(0,'rgba(61,107,255,'+a.toFixed(3)+')'); rg.addColorStop(1,'rgba(61,107,255,0)');
  ctx.fillStyle=rg; ctx.fillRect(0,0,cvs.width/DPR,cvs.height/DPR);
}
function drawDots(now){
  let shrink=1;
  if(state==='levelclear'){ const p=Math.min(1,(T-READY_T)/0.7); shrink=1-p; }
  if(shrink<=0) return;
  ctx.fillStyle='#ffd9a3';
  for(let r=0;r<H;r++)for(let c=0;c<W;c++){
    const d=dots[r][c]; if(!d) continue;
    const x=OX+(c+.5)*S, y=OY+(r+.5)*S;
    if(d===1){ ctx.beginPath(); ctx.arc(x,y,S*.10*shrink,0,7); ctx.fill(); }
  }
  for(let r=0;r<H;r++)for(let c=0;c<W;c++){
    if(dots[r][c]!==2) continue;
    const pr=S*(.30+.05*Math.sin(now/140))*shrink;
    ctx.save();
    ctx.shadowColor='rgba(255,184,77,.95)'; ctx.shadowBlur=S*.5;
    ctx.fillStyle='#fff1cf';
    ctx.beginPath(); ctx.arc(OX+(c+.5)*S,OY+(r+.5)*S,pr,0,7); ctx.fill();
    ctx.restore();
  }
}
let READY_T=0;
function drawPac(now){
  if(state==='dying'){
    const p=Math.min(1,(T-READY_T)/(DIE_MS/1000));
    const cx=OX+(pac.c+.5)*S, cy=OY+(pac.r+.5)*S;
    if(p<0.55){
      const m=0.25+ (p/0.55)*Math.PI*0.75;
      drawPacShape(cx,cy,S*.46,pac.dir?angOf(pac.dir):Math.PI,m,1);
    } else {
      const qp=(p-0.55)/0.45;
      if(!pac.burstDone){ pac.burstDone=true; burst(pac.c,pac.r,26,'#ffe14d'); shake={t:T,dur:.4,mag:4}; }
      ctx.save(); ctx.globalAlpha=1-qp;
      drawPacShape(cx,cy,S*.46*(1-qp),Math.PI*2*p,0.3,1);
      ctx.restore();
    }
    return;
  }
  let x=pac.c+.5,y=pac.r+.5;
  if(pac.mv){ const p=ease((T-pac.mv.t0)/pac.mv.dur); const q=posFor(pac.mv,p); x=q.x+.5; y=q.y+.5; }
  const moving = pac.mv!==null;
  if(moving) pac.mouthT+=1/60*9; else pac.mouthT*=0.9;
  const m=0.18+0.30*Math.abs(Math.sin(pac.mouthT));
  const a=pac.dir?angOf(pac.dir):Math.PI;
  let alp=1;
  if(pac.mv&&pac.mv.wrap){ const p=ease((T-pac.mv.t0)/pac.mv.dur); alp=p<.5?1-2*p:2*p-1; }
  ctx.save(); ctx.globalAlpha=Math.max(0,alp);
  drawPacShape(OX+x*S,OY+y*S,S*.46,a,m,1);
  // queued direction indicator
  if(pac.pending&&sub!=='await'){
    const d=DIRS[pac.pending];
    const qx=OX+(x+d.x*.72)*S, qy=OY+(y+d.y*.72)*S;
    ctx.globalAlpha=.5+.4*Math.sin(now/120);
    ctx.fillStyle='#ffe14d';
    ctx.beginPath(); ctx.arc(qx,qy,S*.13,0,7); ctx.fill();
  }
  ctx.restore();
}
function angOf(d){ return d===1?-Math.PI/2: d===3?Math.PI/2: d===2?Math.PI:0; }
function drawPacShape(cx,cy,r,a,m,alpha){
  ctx.save();
  ctx.shadowColor='rgba(255,225,77,.8)'; ctx.shadowBlur=S*.45;
  const g=ctx.createRadialGradient(cx-r*.3,cy-r*.3,r*.1,cx,cy,r);
  g.addColorStop(0,'#fff3a6'); g.addColorStop(1,'#ffe14d');
  ctx.fillStyle=g;
  ctx.beginPath();
  ctx.moveTo(cx,cy);
  ctx.arc(cx,cy,r,a+m,a-m,false);
  ctx.closePath(); ctx.fill();
  ctx.restore();
}
function posFor(mv,p){
  if(!mv.wrap) return {x:mv.fc+(mv.tc-mv.fc)*p, y:mv.fr+(mv.tr-mv.fr)*p};
  const dx = mv.tc>mv.fc ? -1 : 1; // 0→18 means leftward exit (tc=18 from fc=0 is wrap-left)
  const dirx = (mv.fc===0&&mv.tc===W-1)? -1 : 1;
  if(p<.5){ const q=p*2; return {x:mv.fc+dirx*q*.7, y:mv.fr}; }
  else { const q=(p-.5)*2; return {x:mv.tc-dirx*(1-q)*.7, y:mv.tr}; }
}
function drawGhosts(now){
  for(const g of ghosts){
    let x=g.c+.5,y=g.r+.5,alp=1,flying=false;
    if(g.fly){
      const p=ease((T-g.fly.t0)/g.fly.dur); flying=true;
      x=g.fly.fc+(HOUSE_C.c-g.fly.fc)*p+.5; y=g.fly.fr+(HOUSE_C.r-g.fly.fr)*p+.5;
    } else if(g.mv){
      const p=ease((T-g.mv.t0)/g.mv.dur); const q=posFor(g.mv,p); x=q.x+.5; y=q.y+.5;
      if(g.mv.wrap){ const pp=ease((T-g.mv.t0)/g.mv.dur); alp=pp<.5?1-2*pp:2*pp-1; }
    } else g.bobT+=1/60;
    const cx=OX+x*S, cy=OY+y*S;
    ctx.save(); ctx.globalAlpha=Math.max(0,alp);
    if(flying){ drawEyesOnly(cx,cy,g.dir); ctx.restore(); continue; }
    const fright = frightTurns>0 && g.state!=='eaten';
    let body=g.color;
    if(fright){
      if(frightTurns<=2 && (now/130|0)%2===0) body='#f4f7ff';
      else body='#2f45ff';
    }
    const bob=Math.sin(g.bobT*6)*S*.03;
    drawGhostBody(cx,cy+bob,S*.46,body,g);
    if(fright){
      // frightened face
      ctx.fillStyle= body==='#f4f7ff' ? '#e2556b' : '#ffb8de';
      const ex=S*.13, ey=-S*.04;
      for(const s of [-1,1]){ ctx.beginPath(); ctx.arc(cx+s*ex,cy+bob+ey,S*.055,0,7); ctx.fill(); }
      ctx.strokeStyle=ctx.fillStyle; ctx.lineWidth=S*.05; ctx.lineJoin='round';
      const my=cy+bob+S*.14;
      ctx.beginPath();
      for(let i=0;i<6;i++){ const mx=cx-S*.2+i*(S*.066); ctx.lineTo(mx,my+(i%2? S*.05:-S*.03)); }
      ctx.stroke();
    } else {
      drawEyes(cx,cy+bob,g.dir,body);
    }
    ctx.restore();
  }
}
function drawGhostBody(cx,cy,r,body,g){
  const skirt=cy+r*.92, bump=r*.34;
  const phase=((g.bobT*7)|0)%2;
  ctx.save();
  ctx.shadowColor=g.color; ctx.shadowBlur=S*.5;
  ctx.fillStyle=body;
  ctx.beginPath();
  ctx.arc(cx,cy,r,Math.PI,0);
  ctx.lineTo(cx+r,skirt);
  const n=4, seg=(r*2)/n;
  for(let i=0;i<n;i++){
    const x1=cx-r+(i+1)*seg, yv=( (i+phase)%2===0 ) ? skirt : skirt-bump;
    ctx.lineTo(x1,yv);
  }
  ctx.lineTo(cx-r,skirt);
  ctx.closePath(); ctx.fill();
  ctx.restore();
}
function drawEyes(cx,cy,dir){
  const ex=S*.14, ey=-S*.05, off=dir?{x:DIRS[dir].x*S*.07,y:DIRS[dir].y*S*.07}:{x:0,y:0};
  for(const s of [-1,1]){
    ctx.fillStyle='#fff';
    ctx.beginPath(); ctx.ellipse(cx+s*ex,cy+ey,S*.13,S*.16,0,0,7); ctx.fill();
    ctx.fillStyle='#2038a8';
    ctx.beginPath(); ctx.arc(cx+s*ex+off.x,cy+ey+off.y,S*.07,0,7); ctx.fill();
  }
}
function drawEyesOnly(cx,cy,dir){
  const off=dir?{x:DIRS[dir].x*S*.09,y:DIRS[dir].y*S*.09}:{x:0,y:0};
  for(const s of [-1,1]){
    ctx.save(); ctx.shadowColor='#fff'; ctx.shadowBlur=S*.3;
    ctx.fillStyle='#fff';
    ctx.beginPath(); ctx.ellipse(cx+s*S*.14,cy,S*.15,S*.19,0,0,7); ctx.fill();
    ctx.restore();
    ctx.fillStyle='#2038a8';
    ctx.beginPath(); ctx.arc(cx+s*S*.14+off.x,cy+off.y,S*.08,0,7); ctx.fill();
  }
}
function drawParticles(){
  for(let i=particles.length-1;i>=0;i--){
    const p=particles[i], age=(T-p.t)/p.dur;
    if(age>=1){ particles.splice(i,1); continue; }
    ctx.save(); ctx.globalAlpha=1-age;
    ctx.fillStyle=p.col;
    ctx.beginPath(); ctx.arc(OX+p.x*S,OY+p.y*S,p.sz*S*(1-age*.5),0,7); ctx.fill();
    ctx.restore();
  }
}
function drawPopups(){
  for(let i=popups.length-1;i>=0;i--){
    const p=popups[i], age=(T-p.t)/p.dur;
    if(age>=1){ popups.splice(i,1); continue; }
    ctx.save();
    ctx.globalAlpha=1-age*age;
    ctx.font='800 '+(S*.72)+'px ui-monospace,Consolas,monospace';

    const x=OX+p.x*S, y=OY+(p.y-age*.9)*S;

    ctx.strokeText(p.text,x,y);
    ctx.fillStyle=p.col; ctx.fillText(p.text,x,y);
    ctx.restore();
  }
}
function drawBanner(){
  if(!banner) return;
  const p=(T-banner.t0)/banner.dur;
  if(p>=1){ banner=null; return; }
  const a = p<.15? p/.15 : p>.8? (1-p)/.2 : 1;
  const sc = 1+Math.max(0,.35-p)*1.4;
  ctx.save();
  ctx.globalAlpha=Math.max(0,a);
  ctx.translate(OX+W*S/2, OY+banner.y*S);
  ctx.scale(sc,sc);
  ctx.font='900 '+(S*1.15)+'px "Arial Black",Arial,sans-serif';

  ctx.lineWidth=S*.22; ctx.strokeStyle='rgba(3,5,14,.9)';
  ctx.strokeText(banner.text,0,0);
  ctx.fillStyle=banner.color; ctx.fillText(banner.text,0,0);

function drawReady(){

function render(now){
  drawBg(now);
  ctx.save();
  if(shake.t!==0 && T-shake.t<shake.dur){
    const k=1-(T-shake.t)/shake.dur;
    ctx.translate((Math.random()-.5)*2*shake.mag*k,(Math.random()-.5)*2*shake.mag*k);
  }
  ctx.drawImage(mcv,0,0,cvs.width/DPR,cvs.height/DPR);
  drawDots(now);
  drawGhosts(now);
  drawPac(now);
  drawParticles();
  drawPopups();
  drawReady();
  drawBanner();
  ctx.restore();
  if(flashA>0){
    ctx.save();
    ctx.fillStyle='rgba('+flashCol+','+(flashA*.5).toFixed(3)+')';
    ctx.fillRect(0,0,cvs.width/DPR,cvs.height/DPR);
    ctx.restore();
    flashA=Math.max(0,flashA-1/60*1.2);
  }
}

/* ---------- main loop ---------- */
let lastTs=0;
function frame(ts){
  requestAnimationFrame(frame);
  const dt=Math.min(.05,(ts-lastTs)/1000||0); lastTs=ts;
  if(!paused){
    T+=dt;
    // particles physics
    for(const p of particles){ p.x+=p.vx*dt; p.y+=p.vy*dt; p.vx*=.94; p.vy*=.94; }

    if(state==='menu'){
      attractTick();
      playSub(dt);
    }
    else if(state==='play') playSub(dt);
    else if(state==='ready'){ if(T-READY_T>=1.25) setState('play'); }
    else if(state==='dying'){ if(T-READY_T>=DIE_MS/1000) afterDeath(); }
    else if(state==='levelclear'){ if(T-READY_T>=CLEAR_MS/1000) afterClear(); }
  }
  render(ts);
}
function playSub(){
  if(sub==='await'&&state==='play'){
    // deliberate turn: queued input first, then held key, then cruise straight
    let d=0;
    if(pac.pending&&pacCan(pac.c,pac.r,pac.pending)) d=pac.pending;
    else {
      for(const k of DIR_ORDER) if(heldKeys[k]&&pacCan(pac.c,pac.r,k)){ d=k; break; }
      if(!d&&pac.dir&&pacCan(pac.c,pac.r,pac.dir)) d=pac.dir;
    }
    if(d){ pac.pending=0; startPacMove(d); }
  }
  else if(sub==='pacM'&&pac.mv){
    const p=(T-pac.mv.t0)/pac.mv.dur;
    if(p>=1) pacArrived();
  }
  else if(sub==='ghostM'){
    let done=true;
    for(const g of ghosts) if(g.mv && (T-g.mv.t0)<g.mv.dur) done=false;
    if(done) ghostsArrived();
  }
  else if(sub==='eat'){
    let all=true;
    for(const g of eatingList){ if(!g.fly|| (T-g.fly.t0)<g.fly.dur) { if(g.fly&&(T-g.fly.t0)>=g.fly.dur) {} if(g.fly&&all &&(T-g.fly.t0)<g.fly.dur) all=false; } else if(g.fly){ /* done */ } }
    // simpler explicit check:
    all=true;
    for(const g of eatingList){ if(!g.fly || (T-g.fly.t0)>=g.fly.dur) continue; all=false; }
    if(eatingList.length===0||all) finishEat();
  }
  else if(sub==='dead'&&state==='menu'){
    if(attractDeadT&&T>attractDeadT){ attractDeadT=0; buildDots(); resetPositions(); sub='await'; }
  }
}
