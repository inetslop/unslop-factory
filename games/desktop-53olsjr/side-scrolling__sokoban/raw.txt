const WALK_MS=115, PUSH_MS=135;
let G=null, state='title', camX=0, camY=0, shakeT=0, tGlobal=0;
let particles=[];

function loadLevel(i){
  const def=LEVELS[i];
  const rows=def.map.length;
  let w=0; for(const r of def.map) w=Math.max(w,r.length);
  const walls=new Set(), targets=new Set(), boxes=[];
  let px=1,py=1;
  for(let y=0;y<rows;y++){
    for(let x=0;x<w;x++){
      const c=def.map[y][x]||'#';
      if(c==='#')walls.add(x+','+y);
      if(c==='.'||c==='*'||c==='+')targets.add(x+','+y);
      if(c==='$'||c==='*')boxes.push({x,y,px:x,py:y,squash:0});
      if(c==='@'||c==='+'){px=x;py=y;}
    }
  }
  G={i,rows,w,walls,targets,boxes,boxAt:new Map(),player:{x:px,y:py,px,py,facing:'R',phase:0},
     moves:0,pushes:0,startTime:performance.now(),undoStack:[],move:null,bumpT:0,bumpDir:{x:0,y:0},won:false};
  for(const b of G.boxes)G.boxAt.set(b.x+','+b.y,b);
  particles=[];
  layoutLevel(true);
  camX=Math.max(0,Math.min(G.worldW-W,px*G.ts-W*0.42));
  camY=G.offY;
  updateHUD();
  showBanner(i);
}
function layoutLevel(resnap){
  const ts=Math.round(Math.max(20,Math.min(78,
    H/(G.rows+1.9), W/Math.max(9,G.w*0.56))));
  G.ts=ts; G.worldW=G.w*ts; G.offY=Math.floor((H-G.rows*ts)/2);
  if(resnap){
    for(const b of G.boxes){b.px=b.x;b.py=b.y;}
    G.player.px=G.player.x;G.player.py=G.player.y;
  }
}

/* ============================== INPUT / LOGIC ============================== */
const DIRS={up:{x:0,y:-1},down:{x:0,y:1},left:{x:-1,y:0},right:{x:1,y:0}};
let held=[]; // stack of directions, last = priority
function pushDir(d){ if(held.includes(d))held.splice(held.indexOf(d),1); held.push(d); }
function releaseDir(d){ const i=held.indexOf(d); if(i>=0)held.splice(i,1); }

function step(){
  if(state!=='play'||G.move)return;
  const d=held[held.length-1]; if(!d)return;
  const {x:dx,y:dy}=DIRS[d], P=G.player;
  const nx=P.x+dx, ny=P.y+dy;
  if(G.walls.has(nx+','+ny)){ G.bumpT=130;G.bumpDir={x:dx,y:dy}; sfx('bump'); return; }
  const box=G.boxAt.get(nx+','+ny);
  let kind='walk', bx=null, by=null;
  if(box){
    const ex=nx+dx, ey=ny+dy;
    if(G.walls.has(ex+','+ey)||G.boxAt.has(ex+','+ey)){ G.bumpT=130;G.bumpDir={x:dx,y:dy}; sfx('bump'); return; }
    kind='push'; bx=ex; by=ey;
  }
  G.undoStack.push({px:P.x,py:P.y,bx:box?box.x:null,by:box?box.y:null});
  if(G.undoStack.length>600)G.undoStack.shift();
  if(kind==='push'){
    G.boxAt.delete(box.x+','+box.y);
    box.px=box.x;box.py=box.y; box.x=bx;box.y=by; box.squash=1;
    G.boxAt.set(bx+','+by,box);
    G.pushes++;
  }
  P.px=P.x;P.py=P.y; P.x=nx;P.y=ny;
  P.facing = dx!==0 ? (dx>0?'R':'L') : (dy<0?'U':'D');
  G.moves++;
  G.move={t:0,dur:kind==='push'?PUSH_MS:WALK_MS,box:box||null,kind};
  sfx(kind==='push'?'push':'step');
}
function finishMove(){
  const m=G.move; G.move=null;
  const P=G.player;
  if(m.box){
    const on=G.targets.has(m.box.x+','+m.box.y);
    if(on){ sfx('ding'); sparks(m.box); shakeT=120; }
    if(G.boxes.every(b=>G.targets.has(b.x+','+b.y))) winLevel();
  } else {
    puff(P.x,P.y,2);
  }
  updateHUD();
}
function undo(){
  if(state!=='play'||G.move||!G.undoStack.length)return;
  const s=G.undoStack.pop(), P=G.player;
  if(s.bx!=null){
    const b=G.boxAt.get(s.bx+','+s.by);
    G.boxAt.delete(s.bx+','+s.by); b.x=s.bx;b.y=s.by;b.px=b.x;b.py=b.y;
    G.boxAt.set(b.x+','+b.y,b);
  }
  P.x=s.px;P.y=s.py;P.px=P.x;P.py=P.y;
  sfx('undo'); updateHUD();
}
function winLevel(){
  G.won=true; state='won';
  const t=(performance.now()-G.startTime)/1000;
  let rec=false;
  if(best[G.i]==null||G.moves<best[G.i]){ best[G.i]=G.moves; LS.set('dockhaul_best',best); rec=true; }
  if(G.i+1<LEVELS.length && unlocked<G.i+1){ unlocked=G.i+1; LS.set('dockhaul_unlocked',unlocked); }
  sfx('win'); confetti();
  setTimeout(()=>{
    if(state!=='won')return;
    const last=G.i===LEVELS.length-1;
    document.getElementById('wTag').textContent=last?'Final dock cleared':'Dock cleared';
    document.getElementById('wTitle').textContent=last?'ALL DOCKS CLEARED':'SHIFT COMPLETE';
    document.getElementById('wMoves').textContent=G.moves;
    document.getElementById('wPushes').textContent=G.pushes;
    document.getElementById('wTime').textContent=fmt(t);
    document.getElementById('wBest').textContent=best[G.i]==null?'—':best[G.i];
    const nr=document.getElementById('newRec'); nr.classList.toggle('gone',!rec);
    document.getElementById('nextBtn').textContent=last?'CLOCK OUT → TITLE':'NEXT DOCK →';
    show('winScreen'); state='won2';
  },900);
}
function fmt(s){s=Math.floor(s);return Math.floor(s/60)+':'+String(s%60).padStart(2,'0')}

let AC=null,master=null,noiseBuf=null;
function audio(){
  if(!AC){
    const Ctx=window.AudioContext||window.webkitAudioContext; if(!Ctx)return;
    AC=new Ctx(); master=AC.createGain(); master.gain.value=muted?0:0.55; master.connect(AC.destination);
    noiseBuf=AC.createBuffer(1,AC.sampleRate*2,AC.sampleRate);
    const d=noiseBuf.getChannelData(0); let last=0;
    for(let i=0;i<d.length;i++){const w=Math.random()*2-1; last=(last+0.02*w)/1.02; d[i]=w*0.5+last*3;}
    // ambient: low drone + wind
    const dg=AC.createGain();dg.gain.value=0.018;const lp=AC.createBiquadFilter();lp.type='lowpass';lp.frequency.value=160;
    const o1=AC.createOscillator();o1.type='sine';o1.frequency.value=54;
    const o2=AC.createOscillator();o2.type='sine';o2.frequency.value=54.7;
    o1.connect(lp);o2.connect(lp);lp.connect(dg);dg.connect(master);o1.start();o2.start();
    const wg=AC.createGain();wg.gain.value=0.012;
    const wl=AC.createBiquadFilter();wl.type='lowpass';wl.frequency.value=420;
    const src=AC.createBufferSource();src.buffer=noiseBuf;src.loop=true;
    const lfo=AC.createOscillator();lfo.frequency.value=0.07;const lg=AC.createGain();lg.gain.value=0.008;
    lfo.connect(lg);lg.connect(wg.gain);src.connect(wl);wl.connect(wg);wg.connect(master);src.start();lfo.start();
  }
  if(AC.state==='suspended')AC.resume();
}
function tone(f,dur,type,vol,f2,delay){
  if(!AC||muted)return; const t=AC.currentTime+(delay||0);
  const o=AC.createOscillator(),g=AC.createGain();
  o.type=type;o.frequency.setValueAtTime(f,t);
  if(f2)o.frequency.exponentialRampToValueAtTime(Math.max(30,f2),t+dur);
  g.gain.setValueAtTime(vol,t);g.gain.exponentialRampToValueAtTime(0.001,t+dur);
  o.connect(g);g.connect(master);o.start(t);o.stop(t+dur+0.02);
}
function noiseS(dur,freq,vol,type){
  if(!AC||muted)return; const t=AC.currentTime;
  const s=AC.createBufferSource();s.buffer=noiseBuf;s.loop=true;
  const f=AC.createBiquadFilter();f.type=type||'lowpass';f.frequency.value=freq;
  const g=AC.createGain();g.gain.setValueAtTime(vol,t);g.gain.exponentialRampToValueAtTime(0.001,t+dur);
  s.connect(f);f.connect(g);g.connect(master);s.start(t);s.stop(t+dur+0.02);
}
function sfx(n){
  if(!AC||muted)return;
  switch(n){
    case 'step': noiseS(0.05,900,0.12,'lowpass'); break;
    case 'push': noiseS(0.14,320,0.3,'bandpass'); noiseS(0.1,180,0.18,'lowpass'); break;
    case 'bump': tone(72,0.09,'sine',0.3,50); noiseS(0.04,500,0.12); break;
    case 'ding': tone(660,0.14,'sine',0.28); tone(990,0.22,'sine',0.22,null,0.07); break;
    case 'undo': tone(300,0.12,'square',0.12,190); break;
    case 'click': tone(1250,0.04,'square',0.1); break;
    case 'sweep': tone(220,0.28,'sine',0.16,640); break;
    case 'win':[523,659,784,1047].forEach((f,k)=>tone(f,0.22,'triangle',0.2,null,k*0.09)); noiseS(0.5,4000,0.06,'highpass'); break;
  }
}
function toggleMute(){

  if(master)master.gain.value=muted?0:0.55;
  document.getElementById('bMute').style.opacity=muted?0.45:1;

/* ============================== PARTICLES ============================== */
function puff(x,y,n){for(let i=0;i<n;i++)particles.push({x:x*G.ts+G.ts/2+(Math.random()*8-4),y:y*G.ts+G.ts*0.9,
  vx:(Math.random()-0.5)*30,vy:-Math.random()*22,g:14,life:0,max:450+rnd(250),size:2+rnd(3),col:'rgba(160,150,130,',type:'dust'});}
function sparks(b){const cx=b.x*G.ts+G.ts/2,cy=b.y*G.ts+G.ts/2;
  for(let i=0;i<16;i++){const a=Math.random()*Math.PI*2,s=40+rnd(90);
    particles.push({x:cx,y:cy,vx:Math.cos(a)*s,vy:Math.sin(a)*s-30,g:120,life:0,max:500+rnd(300),size:1.5+rnd(2),col:'rgba(255,190,80,',type:'spark'});}}
function confetti(){const P=G.player;
  for(let i=0;i<70;i++){const a=Math.random()*Math.PI*2,s=60+rnd(180);
    particles.push({x:P.px*G.ts+G.ts/2,y:P.py*G.ts+G.ts/2,vx:Math.cos(a)*s,vy:Math.sin(a)*s-140,g:260,life:0,max:1200+rnd(900),
    size:3+rnd(4),col:['rgba(255,181,77,','rgba(126,214,223,','rgba(245,185,61,','rgba(232,242,255,'][i%4],type:'conf',rot:Math.random()*6,vr:(Math.random()-0.5)*12});}}
function rnd(n){return Math.random()*n}

/* ============================== UPDATE ============================== */
let last=performance.now();
function frame(now){
  const dt=Math.min(50,now-last); last=now; tGlobal+=dt/1000;
  if(G)update(dt,now);
  render(now);
  requestAnimationFrame(frame);
}
function update(dt,now){
  // movement tween
  if(G.move){
    G.move.t+=dt;
    const k=Math.min(1,G.move.t/G.move.dur), P=G.player;
    if(!G.move.sx){G.move.sx=P.px;G.move.sy=P.py;}
    P.px=G.move.sx+(P.x-G.move.sx)*k;
    P.py=G.move.sy+(P.y-G.move.sy)*k;
    if(G.move.box){const b=G.move.box;b.px=b.px0!==undefined?b.px0+(b.x-b.px0)*k:b.px;b.py=b.py0+(b.y-b.py0)*k;}
    P.phase+=dt*0.02;
    if(k>=1)finishMove();
  } else if(state==='play' && held.length){ step(); }
  // box pixel snap when no move (undo/restart)
  for(const b of G.boxes){ b.px+=(b.x-b.px)*Math.min(1,dt*0.02); b.py+=(b.y-b.py)*Math.min(1,dt*0.02);
    if(Math.abs(b.x-b.px)<0.002)b.px=b.x; if(Math.abs(b.y-b.py)<0.002)b.py=b.y;
    b.squash=Math.max(0,b.squash-dt/260); }
  if(G.bumpT>0)G.bumpT-=dt;
  if(shakeT>0)shakeT-=dt;
  // camera
  const P=G.player, look=(P.facing==='R'?1:P.facing==='L'?-1:0)*1.3*G.ts;
  const tx=Math.max(0,Math.min(G.worldW-W,P.px*G.ts+G.ts/2-W/2+look));
  camX+=(tx-camX)*Math.min(1,dt*0.008);
  camY=G.offY;
  // particles
  for(let i=particles.length-1;i>=0;i--){const p=particles[i];p.life+=dt;
    if(p.life>p.max){particles.splice(i,1);continue;}
    p.vy+=p.g*dt/1000;p.x+=p.vx*dt/1000;p.y+=p.vy*dt/1000;
    if(p.type==='conf')p.rot+=p.vr*dt/1000;}
  // timer
  if(state==='play'||state==='won'){
    const t=(now-G.startTime)/1000;
    document.getElementById('hTime').textContent=fmt(t);
  }
}

/* ============================== RENDER ============================== */
function hash(n){n=Math.sin(n*127.1+311.7)*43758.5453;return n-Math.floor(n);}
function render(now){
  const t=tGlobal;
  // sky
  const sky=ctx.createLinearGradient(0,0,0,H);
  sky.addColorStop(0,'#060b16');sky.addColorStop(0.55,'#0c1a2e');sky.addColorStop(1,'#132741');
  ctx.fillStyle=sky;ctx.fillRect(0,0,W,H);
  if(!G)return;
  const shx=shakeT>0?(Math.random()-0.5)*shakeT*0.06:0;
  const shy=shakeT>0?(Math.random()-0.5)*shakeT*0.04:0;
  const cx=camX+shx, cy=camY+shy;

  // stars (parallax .1)
  ctx.save();
  const sp=95;
  for(let k=Math.floor(cx*0.1/sp)-1;k<Math.floor((cx*0.1+W)/sp)+2;k++){
    const x=k*sp-cx*0.1, r=hash(k*3.7);
    const sx=x+r*sp*0.8, sy=H*0.06+hash(k*9.1)*H*0.42;
    if(sy>G.offY-20)continue;
    const tw=0.35+0.65*Math.abs(Math.sin(t*(0.5+r)+k));
    ctx.fillStyle='rgba(210,228,255,'+(0.25+0.45*tw*hash(k*5))+')';
    ctx.fillRect(sx,sy,r>0.8?2:1.3,r>0.8?2:1.3);
  }
  // moon
  const mx=W*0.78-cx*0.05, my=H*0.16;
  const mg=ctx.createRadialGradient(mx,my,4,mx,my,90);
  mg.addColorStop(0,'rgba(220,235,255,.5)');mg.addColorStop(1,'rgba(220,235,255,0)');
  ctx.fillStyle=mg;ctx.beginPath();ctx.arc(mx,my,90,0,7);ctx.fill();
  ctx.fillStyle='#dce8f7';ctx.beginPath();ctx.arc(mx,my,22,0,7);ctx.fill();
  ctx.fillStyle='rgba(160,180,210,.5)';ctx.beginPath();ctx.arc(mx-7,my+4,5,0,7);ctx.arc(mx+6,my-6,3.5,0,7);ctx.fill();

  // skyline (parallax .28)
  const baseY=G.offY+G.rows*G.ts;
  ctx.fillStyle='#0e1c30';
  const bp=140;
  for(let k=Math.floor(cx*0.28/bp)-1;k<Math.floor((cx*0.28+W)/bp)+2;k++){
    const x=k*bp-cx*0.28, bw=bp*(0.5+hash(k*1.3)*0.4), bh=H*0.10+hash(k*7.7)*H*0.2;
    ctx.fillRect(x,baseY-bh,bw,bh);
    // windows
    for(let wy=0;wy<Math.floor(bh/26);wy++)for(let wx=0;wx<Math.floor(bw/20);wx++){
      if(hash(k*31+wy*7+wx*13)>0.78){ctx.fillStyle='rgba(255,190,100,.35)';
        ctx.fillRect(x+6+wx*20,baseY-bh+8+wy*26,4,6);ctx.fillStyle='#0e1c30';}
    }
  }
  // cranes (parallax .55)
  const cp=620;
  for(let k=Math.floor(cx*0.55/cp)-1;k<Math.floor((cx*0.55+W)/cp)+2;k++){
    if(hash(k*3.3)<0.45)continue;
    const x=k*cp+hash(k)*200-cx*0.55, top=baseY-H*0.34-hash(k*8)*H*0.1, wcol='#15263e';
    ctx.strokeStyle=wcol;ctx.lineWidth=Math.max(3,H*0.006);
    ctx.beginPath();
    ctx.moveTo(x,baseY);ctx.lineTo(x,top);
    ctx.moveTo(x+H*0.06,baseY);ctx.lineTo(x+H*0.06,top);
    ctx.moveTo(x- H*0.05,top);ctx.lineTo(x+H*0.16,top);
    ctx.stroke();
    ctx.beginPath();ctx.moveTo(x,top+(top-baseY)*0.25);ctx.lineTo(x+H*0.06,top+(top-baseY)*0.45);ctx.stroke();
    // trolley + hanging cable
    const tx=x+H*(0.08+0.05*Math.sin(t*0.3+k));
    ctx.beginPath();ctx.moveTo(tx,top);ctx.lineTo(tx,top+H*0.06+Math.sin(t*0.7+k)*4);ctx.stroke();
    // beacon
    if(Math.sin(t*2.2+k)>0.4){ctx.fillStyle='rgba(255,70,60,.9)';ctx.beginPath();ctx.arc(x,top-4,3,0,7);ctx.fill();
      const bg=ctx.createRadialGradient(x,top-4,1,x,top-4,14);bg.addColorStop(0,'rgba(255,70,60,.5)');bg.addColorStop(1,'rgba(255,70,60,0)');
      ctx.fillStyle=bg;ctx.beginPath();ctx.arc(x,top-4,14,0,7);ctx.fill();}
  }
  // searchlight sweep (parallax .4)
  const slx=W*0.2-cx*0.15, sly=-30, ang=Math.sin(t*0.16)*0.5+0.9;
  ctx.save();ctx.translate(slx,sly);ctx.rotate(ang);
  const beam=ctx.createLinearGradient(0,0,0,H*1.2);
  beam.addColorStop(0,'rgba(255,220,160,.10)');beam.addColorStop(1,'rgba(255,220,160,0)');
  ctx.fillStyle=beam;ctx.beginPath();ctx.moveTo(0,0);ctx.lineTo(-H*0.09,H*1.2);ctx.lineTo(H*0.09,H*1.2);ctx.closePath();ctx.fill();
  ctx.restore();

  /* ---- world ---- */
  const ts=G.ts, x0=Math.max(0,Math.floor(cx/ts)), x1=Math.min(G.w-1,Math.ceil((cx+W)/ts));
  for(let y=0;y<G.rows;y++){
    for(let x=x0;x<=x1;x++){
      const sx=x*ts-cx, sy=y*ts-cy;
      if(G.walls.has(x+','+y)){ drawWall(sx,sy,ts,x,y); continue; }
      // floor
      const h=hash(x*17+y*311);
      ctx.fillStyle=h>0.5?'#1c2a3d':'#1a2839';
      ctx.fillRect(sx,sy,ts,ts);
      ctx.strokeStyle='rgba(0,0,0,.22)';ctx.lineWidth=1;
      ctx.strokeRect(sx+0.5,sy+0.5,ts-1,ts-1);
      if(h>0.87){ctx.fillStyle='rgba(10,14,20,.35)';ctx.beginPath();
        ctx.ellipse(sx+ts*hash(x+y),sy+ts*(0.4+h*0.3),ts*0.16,ts*0.09,0,0,7);ctx.fill();}
      if(G.targets.has(x+','+y))drawTarget(sx,sy,ts,x,y,t);
    }
  }
  // boxes
  for(const b of G.boxes)drawBox(b,b.px*ts-cx,b.py*ts-cy,ts,t);
  // player
  drawPlayer(cx,cy,ts,t);
  // particles
  for(const p of particles){
    const a=1-p.life/p.max;
    if(p.type==='conf'){ctx.save();ctx.translate(p.x-cx,p.y-cy);ctx.rotate(p.rot);
      ctx.fillStyle=p.col+a+')';ctx.fillRect(-p.size/2,-p.size/4,p.size,p.size/2);ctx.restore();}
    else{ctx.fillStyle=p.col+(a*0.7)+')';ctx.beginPath();ctx.arc(p.x-cx,p.y-cy,p.size*(1+p.life/p.max),0,7);ctx.fill();}
  }
  // fog wisps (parallax .85)
  ctx.save();
  const fp=340;
  for(let k=Math.floor((cx*0.85-t*12)/fp)-1;k<Math.floor(((cx*0.85-t*12)+W)/fp)+2;k++){
    const x=k*fp+t*12-cx*0.85, y=baseY-30-hash(k*4.4)*H*0.2;
    const fg=ctx.createRadialGradient(x,y,4,x,y,110);
    fg.addColorStop(0,'rgba(120,150,190,.05)');fg.addColorStop(1,'rgba(120,150,190,0)');
    ctx.fillStyle=fg;ctx.beginPath();ctx.arc(x,y,110,0,7);ctx.fill();
  }
  ctx.restore();
  // vignette
  const vg=ctx.createRadialGradient(W/2,H/2,Math.min(W,H)*0.45,W/2,H/2,Math.max(W,H)*0.75);
  vg.addColorStop(0,'rgba(0,0,0,0)');vg.addColorStop(1,'rgba(3,6,12,.55)');
  ctx.fillStyle=vg;ctx.fillRect(0,0,W,H);
}
function drawWall(sx,sy,ts,x,y){
  const g=ctx.createLinearGradient(sx,sy,sx,sy+ts);
  g.addColorStop(0,'#31465f');g.addColorStop(1,'#22344a');
  ctx.fillStyle=g;ctx.fillRect(sx,sy,ts,ts);
  const openAbove=!G.walls.has(x+','+(y-1));
  if(openAbove){ctx.fillStyle='#4d688a';ctx.fillRect(sx,sy,ts,Math.max(3,ts*0.12));}
  ctx.fillStyle='rgba(0,0,0,.3)';ctx.fillRect(sx,sy+ts-Math.max(3,ts*0.1),ts,Math.max(3,ts*0.1));
  ctx.strokeStyle='rgba(8,14,22,.6)';ctx.lineWidth=1;ctx.strokeRect(sx+0.5,sy+0.5,ts-1,ts-1);
  ctx.fillStyle='rgba(190,215,245,.25)';
  const r=Math.max(1.5,ts*0.05);
  ctx.beginPath();ctx.arc(sx+r*1.6,sy+r*1.6,r,0,7);ctx.arc(sx+ts-r*1.6,sy+r*1.6,r,0,7);
  ctx.arc(sx+r*1.6,sy+ts-r*1.6,r,0,7);ctx.arc(sx+ts-r*1.6,sy+ts-r*1.6,r,0,7);ctx.fill();
}
function drawTarget(sx,sy,ts,x,y,t){
  const m=ts*0.12, on=G.boxAt.has(x+','+y);
  ctx.fillStyle='rgba(8,13,20,.55)';ctx.fillRect(sx+m,sy+m,ts-2*m,ts-2*m);
  // hazard tape edges
  const th=Math.max(3,ts*0.09);
  for(let i=0;i<Math.floor((ts-2*m)/th);i++){
    ctx.fillStyle=i%2?'#141a24':'rgba(245,185,61,.75)';
    ctx.fillRect(sx+m+i*th,sy+m,th,th*0.6);
    ctx.fillRect(sx+m+((i+1)%2?0:th),sy+ts-m-th*0.6,th,th*0.6);
  }
  const pulse=on?1:(0.45+0.3*Math.sin(t*3+x+y));
  ctx.strokeStyle='rgba(126,214,223,'+(on?0.9:pulse)+')';
  ctx.lineWidth=Math.max(2,ts*0.06);
  const cs=ts*(on?0.2:0.13+0.03*Math.sin(t*3+x+y));
  ctx.strokeRect(sx+ts/2-cs,sy+ts/2-cs,cs*2,cs*2);
  if(on){const gg=ctx.createRadialGradient(sx+ts/2,sy+ts/2,2,sx+ts/2,sy+ts/2,ts*0.75);
    gg.addColorStop(0,'rgba(126,214,223,.18)');gg.addColorStop(1,'rgba(126,214,223,0)');
    ctx.fillStyle=gg;ctx.fillRect(sx,sy,ts,ts);}
}
function drawBox(b,sx,sy,ts,t){
  const on=G.targets.has(b.x+','+b.y);
  const sq=b.squash;
  const w=ts*0.84*(1+sq*0.06), h=ts*0.84*(1-sq*0.12);
  const x=sx+(ts-w)/2, y=sy+(ts-h)/2+h*sq*0.5;
  ctx.save();
  // shadow
  ctx.fillStyle='rgba(0,0,0,.35)';
  ctx.beginPath();ctx.ellipse(sx+ts/2,sy+ts*0.92,w*0.45,h*0.1,0,0,7);ctx.fill();
  if(on){ctx.shadowColor='rgba(255,190,80,.9)';ctx.shadowBlur=16;}
  const g=ctx.createLinearGradient(x,y,x,y+h);
  g.addColorStop(0,on?'#c98d54':'#b57a44');g.addColorStop(1,on?'#9a6234':'#8a552c');
  ctx.fillStyle=g;ctx.fillRect(x,y,w,h);
  ctx.shadowBlur=0;
  // planks
  ctx.strokeStyle='rgba(60,32,12,.55)';ctx.lineWidth=Math.max(2,ts*0.05);
  ctx.strokeRect(x+ctx.lineWidth,y+ctx.lineWidth,w-ctx.lineWidth*2,h-ctx.lineWidth*2);
  ctx.beginPath();ctx.moveTo(x,y+h*0.32);ctx.lineTo(x+w,y+h*0.32);
  ctx.moveTo(x,y+h*0.68);ctx.lineTo(x+w,y+h*0.68);ctx.stroke();
  // stencil arrow
  ctx.fillStyle=on?'rgba(255,225,170,.9)':'rgba(30,16,6,.5)';
  const ax=x+w/2,ay=y+h/2;
  ctx.beginPath();ctx.moveTo(ax-w*0.14,ay-h*0.09);ctx.lineTo(ax+w*0.02,ay-h*0.09);
  ctx.lineTo(ax+w*0.02,ay-h*0.16);ctx.lineTo(ax+w*0.16,ay);ctx.lineTo(ax+w*0.02,ay+h*0.16);
  ctx.lineTo(ax+w*0.02,ay+h*0.09);ctx.lineTo(ax-w*0.14,ay+h*0.09);ctx.closePath();ctx.fill();
  // bolts
  ctx.fillStyle='rgba(40,22,8,.7)';
  const r=Math.max(1.5,ts*0.045);
  ctx.beginPath();ctx.arc(x+r*2,y+r*2,r,0,7);ctx.arc(x+w-r*2,y+r*2,r,0,7);
  ctx.arc(x+r*2,y+h-r*2,r,0,7);ctx.arc(x+w-r*2,y+h-r*2,r,0,7);ctx.fill();
  if(on&&Math.sin(t*4+b.x)>0.6){ctx.fillStyle='rgba(255,230,160,.9)';
    ctx.fillRect(x+w-6,y+4,2,2);}
  ctx.restore();
}
function drawPlayer(cx,cy,ts,t){
  const P=G.player;
  let ox=0,oy=0;
  if(G.bumpT>0){const k=Math.sin((1-G.bumpT/130)*Math.PI)*3;ox=G.bumpDir.x*k;oy=G.bumpDir.y*k;}
  const pushing=!!G.move&&G.move.kind==='push';
  let x=P.px*ts-cx+ts/2+ox, y=P.py*ts-cy+ts/2+oy;
  const bob=(G.move||state!=='play')?0:Math.sin(t*3)*1.5;
  y+=bob*0.4;
  ctx.save();
  // shadow
  ctx.fillStyle='rgba(0,0,0,.35)';
  ctx.beginPath();ctx.ellipse(x,P.py*ts-cy+ts*0.9,ts*0.3,ts*0.08,0,0,7);ctx.fill();
  const dir=P.facing;
  const flip=dir==='L'?-1:1;
  ctx.translate(x,y);ctx.scale(flip,1);
  if(pushing)ctx.rotate(0.12);
  const u=ts/32; // unit scale
  const stepA=G.move?Math.sin(P.phase*4)*6*u:0;
  // legs
  ctx.fillStyle='#2b3a52';
  ctx.fillRect(-7*u,-8*u+stepA*0.4,5*u,10*u);
  ctx.fillRect(2*u,-8*u-stepA*0.4,5*u,10*u);
  ctx.fillStyle='#1c2636';
  ctx.fillRect(-8*u,1*u+stepA*0.4,7*u,3*u);
  ctx.fillRect(1*u,1*u-stepA*0.4,7*u,3*u);
  // torso (overalls)
  const tg=ctx.createLinearGradient(0,-22*u,0,-6*u);
  tg.addColorStop(0,'#e8842f');tg.addColorStop(1,'#c96a1d');
  ctx.fillStyle=tg;
  ctx.fillRect(-8*u,-22*u,16*u,15*u);
  ctx.fillStyle='#33507a';ctx.fillRect(-8*u,-12*u,16*u,5*u); // bib band
  // arms
  ctx.strokeStyle='#c96a1d';ctx.lineWidth=4.5*u;ctx.lineCap='round';
  if(pushing){
    ctx.beginPath();ctx.moveTo(2*u,-19*u);ctx.lineTo(13*u,-17*u);ctx.stroke();
    ctx.beginPath();ctx.moveTo(-2*u,-18*u);ctx.lineTo(12*u,-15*u);ctx.stroke();
    ctx.fillStyle='#e8b58c';ctx.beginPath();ctx.arc(14*u,-17*u,3*u,0,7);ctx.fill();
  } else if(dir==='D'){
    ctx.beginPath();ctx.moveTo(-8*u,-20*u);ctx.lineTo(-9*u,-11*u+stepA*0.3);ctx.stroke();
    ctx.beginPath();ctx.moveTo(8*u,-20*u);ctx.lineTo(9*u,-11*u-stepA*0.3);ctx.stroke();
  } else {
    ctx.beginPath();ctx.moveTo(0,-19*u);ctx.lineTo(dir==='U'?0:5*u,-10*u+stepA*0.3);ctx.stroke();
  }
  // head
  ctx.fillStyle='#e8b58c';ctx.beginPath();ctx.arc(0,-27*u,6.5*u,0,7);ctx.fill();
  // cap
  ctx.fillStyle='#ffb54d';
  ctx.beginPath();ctx.arc(0,-28.5*u,6.8*u,Math.PI,0);ctx.fill();
  ctx.fillRect(-7*u,-29.5*u,14*u,2.5*u);
  if(dir!=='U'){ // face
    ctx.fillStyle='#20242c';
    const ex=dir==='D'?0:1.8*u;
    ctx.beginPath();ctx.arc(ex-2.4*u,-26.5*u,1.1*u,0,7);ctx.arc(ex+2.4*u,-26.5*u,1.1*u,0,7);ctx.fill();
  } else {
    ctx.fillStyle='#ffb54d';ctx.fillRect(-3*u,-31*u,6*u,2*u); // back of cap
  }
  ctx.restore();
}
