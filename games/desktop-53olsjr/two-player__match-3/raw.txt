/* ================= EASING / TWEENS ================= */
const easeInQuad=t=>t*t, easeInOut=t=>t<.5?2*t*t:1-Math.pow(-2*t+2,2)/2;
function tween(g,toX,toY,dur,ease,del){

}

function arrRun(a){
  for(let r=0;r<N;r++){ let run=1;
    for(let c=1;c<=N;c++){ if(c<N&&a[r][c]===a[r][c-1]) run++; else { if(run>=3) return true; run=1; } } }
  for(let c=0;c<N;c++){ let run=1;
    for(let r=1;r<=N;r++){ if(r<N&&a[r][c]===a[r-1][c]) run++; else { if(run>=3) return true; run=1; } } }
  return false;
}
function arrMove(a){
  const chk=(r1,c1,r2,c2)=>{ const t=a[r1][c1]; a[r1][c1]=a[r2][c2]; a[r2][c2]=t;

  for(let r=0;r<N;r++)for(let c=0;c<N;c++){
    if(c+1<N&&chk(r,c,r,c+1)) return true;
    if(r+1<N&&chk(r,c,r+1,c)) return true; }
  return false;
}

function findRuns(){

  for(let c=0;c<N;c++){ let run=1;

}

async function doSwap(p,r,c,dr,dc){
  state='busy';
  const r2=r+dr, c2=c+dc;
  if(r2<0||r2>=N||c2<0||c2>=N){ sDeny(); shake=Math.max(shake,4); state='idle'; return; }

  let res=findRuns();
  if(res.cells.size===0){
    sBad(); shake=Math.max(shake,5);
    await Promise.all([tween(A,c,r,160,easeInOut),tween(B,c2,r2,160,easeInOut)]);
    current=1-current; uiTurn(); pulseChip(); state='idle'; return;
  }
  let chain=0;
  while(res.cells.size){
    chain++;
    const pts=res.cells.size*10*chain+(res.maxRun>=4?30*chain:0);
    scores[p]+=pts; hudScore(p);
    floatText((res.cx+.5)*cell,(res.cy+.5)*cell,'+'+pts+(chain>1?' ×'+chain:''),P[p].col);
    waves.push({x:(res.cx+.5)*cell,y:(res.cy+.5)*cell,t0:performance.now(),col:P[p].col});
    res.cells.forEach(k=>{ const rr=(k/N)|0, cc=k%N, g=grid[rr][cc];
      g.popT=performance.now(); burst((cc+.5)*cell,(rr+.5)*cell,GEMS[g.color].base,P[p].col); });
    sPop(chain); if(chain>=2) shake=Math.min(14,6+chain*2);
    await wait(280);
    res.cells.forEach(k=>{ const rr=(k/N)|0, cc=k%N; grid[rr][cc]=null; });
    await gravity();
    res=findRuns();
  }
  if(scores[p]>=TARGET){ winGame(p); return; }
  state='busy'; await maybeReshuffle();
  if(state==='busy'){ state='idle'; pulseChip(); }
}

    for(let r=write;r>=0;r--){ const g=newGem((Math.random()*NCOLORS)|0,c,r-e);
      grid[r][c]=g; ps.push(tween(g,c,r,60+e*52,easeInQuad)); }
  }
  if(ps.length){ await Promise.all(ps); sLand(); }
}

}

function rrPath(x,y,w,h,r){
  ctx.beginPath();
  ctx.moveTo(x+r,y); ctx.lineTo(x+w-r,y); ctx.arcTo(x+w,y,x+w,y+r,r);
  ctx.lineTo(x+w,y+h-r); ctx.arcTo(x+w,y+h,x+w-r,y+h,r);
  ctx.lineTo(x+r,y+h); ctx.arcTo(x,y+h,x,y+h-r,r);
  ctx.lineTo(x,y+r); ctx.arcTo(x,y,x+r,y,r); ctx.closePath();
}
function gemPath(shape,cx,cy,r){
  ctx.beginPath();
  if(shape==='diamond'){
    ctx.moveTo(cx,cy-r); ctx.lineTo(cx+r*.72,cy); ctx.lineTo(cx,cy+r); ctx.lineTo(cx-r*.72,cy); ctx.closePath();
  } else if(shape==='star'){
    for(let i=0;i<8;i++){ const a=-Math.PI/2+i*Math.PI/4, rad=i%2?r*.44:r;
      const px=cx+Math.cos(a)*rad, py=cy+Math.sin(a)*rad; i?ctx.lineTo(px,py):ctx.moveTo(px,py); }
    ctx.closePath();
  } else if(shape==='square'){
    const s=r*.92; rrPath(cx-s,cy-s,s*2,s*2,r*.34);
  } else if(shape==='triangle'){
    ctx.moveTo(cx,cy-r*.85); ctx.lineTo(cx+r*.95,cy+r*.7); ctx.lineTo(cx-r*.95,cy+r*.7); ctx.closePath();
  } else { // pentagon
    for(let i=0;i<5;i++){ const a=-Math.PI/2+i*6.283/5;
      const px=cx+Math.cos(a)*r, py=cy+Math.sin(a)*r; i?ctx.lineTo(px,py):ctx.moveTo(px,py); }
    ctx.closePath();
  }
}
function drawGem(ci,cx,cy,r,alpha){
  if(r<=.5) return;
  const gm=GEMS[ci];
  ctx.save(); ctx.globalAlpha=alpha;
  ctx.fillStyle='rgba(0,0,0,.28)';
  ctx.beginPath(); ctx.ellipse(cx,cy+r*.74,r*.72,r*.3,0,0,6.283); ctx.fill();
  gemPath(gm.shape,cx,cy,r);
  const grad=ctx.createRadialGradient(cx-r*.35,cy-r*.42,r*.12,cx,cy,r*1.2);
  grad.addColorStop(0,gm.light); grad.addColorStop(.55,gm.base); grad.addColorStop(1,gm.dark);
  ctx.fillStyle=grad; ctx.fill();
  ctx.lineJoin='round'; ctx.lineWidth=Math.max(1.5,r*.14); ctx.strokeStyle=gm.dark; ctx.stroke();
  ctx.fillStyle='rgba(255,255,255,.5)';
  ctx.beginPath(); ctx.ellipse(cx-r*.3,cy-r*.38,r*.3,r*.16,-.55,0,6.283); ctx.fill();
  ctx.restore();
}
function drawCursor(p,now){
  const [r,c]=cursors[p], col=P[p].col;
  const active=(current===p)&&(state==='idle'||state==='busy');
  const a=active? .75+.25*Math.sin(now*.008) : .26;
  ctx.save(); ctx.globalAlpha=a;
  const ins=cell*.07, x=c*cell+ins, y=r*cell+ins, s=cell-ins*2;
  if(active){ ctx.shadowColor=col; ctx.shadowBlur=cell*.34; }
  ctx.strokeStyle=col; ctx.lineWidth=cell*.055;
  rrPath(x,y,s,s,cell*.16); ctx.stroke();
  if(active){ const [dr,dc]=dirs[p], sz=cell*.13, ox=dc*cell*.27, oy=dr*cell*.27;
    ctx.shadowBlur=0; ctx.translate((c+.5)*cell+ox,(r+.5)*cell+oy);
    ctx.rotate(Math.atan2(dr,dc));
    ctx.fillStyle=col; ctx.beginPath();
    ctx.moveTo(sz,0); ctx.lineTo(-sz*.7,sz*.8); ctx.lineTo(-sz*.7,-sz*.8); ctx.closePath(); ctx.fill(); }
  ctx.restore();
}

let lastFrame=performance.now();
function frame(now){
  requestAnimationFrame(frame);
  const dt=Math.min(50,now-lastFrame); lastFrame=now;
  stepTweens(now);
  if(shake>.1) shake*=Math.exp(-dt*.009); else shake=0;

  const W=cell*N;
  ctx.clearRect(0,0,W,W);
  ctx.save();
  if(shake>0) ctx.translate((Math.random()-.5)*shake,(Math.random()-.5)*shake);

  // board base + checker cells
  ctx.fillStyle='#0c2531'; ctx.fillRect(-8,-8,W+16,W+16);
  for(let r=0;r<N;r++)for(let c=0;c<N;c++){
    ctx.fillStyle=((r+c)&1)?'#123445':'#0e2b3a';
    ctx.fillRect(c*cell,r*cell,cell,cell);
  }

  // shockwaves
  for(let i=waves.length-1;i>=0;i--){ const wv=waves[i], k=(now-wv.t0)/480;
    if(k>=1){ waves.splice(i,1); continue; }
    ctx.save(); ctx.globalAlpha=(1-k)*.55; ctx.strokeStyle=wv.col; ctx.lineWidth=cell*.09;
    ctx.beginPath(); ctx.arc(wv.x,wv.y,cell*(.25+1.15*k),0,6.283); ctx.stroke(); ctx.restore(); }

  // gems
  for(let r=0;r<N;r++)for(let c=0;c<N;c++){ const g=grid[r][c]; if(!g) continue;
    let scale=1, alpha=1;
    if(g.popT){ const k=(now-g.popT)/260;
      if(k>=1){ scale=0; } else { scale=Math.max(0,1-k*k); alpha=1-k*.9; } }
    if(scale<=0) continue;
    const bob=g.popT? 0 : Math.sin(now*.0022+g.seed)*cell*.02;
    drawGem(g.color,(g.x+.5)*cell,(g.y+.5)*cell+bob,cell*.34*scale,alpha);
  }

  // cursors
  if(state==='idle'||state==='busy'){ drawCursor(1-current,now); drawCursor(current,now); }

  // particles
  for(let i=particles.length-1;i>=0;i--){ const p=particles[i], k=(now-p.t0)/p.life;
    if(k>=1){ particles.splice(i,1); continue; }
    p.vy+=p.grav*dt/1000; p.x+=p.vx*dt/1000; p.y+=p.vy*dt/1000; p.rot+=p.vr*dt/1000;
    ctx.save(); ctx.globalAlpha=1-k; ctx.translate(p.x,p.y); ctx.rotate(p.rot);
    ctx.fillStyle=p.col; ctx.fillRect(-p.size/2,-p.size/2,p.size,p.size); ctx.restore(); }

  // floating score text
  for(let i=floats.length-1;i>=0;i--){ const f=floats[i], k=(now-f.t0)/850;
    if(k>=1){ floats.splice(i,1); continue; }
    ctx.save(); ctx.globalAlpha=1-k;
    ctx.font='900 '+Math.round(cell*.42)+"px Impact,'Arial Black',sans-serif";
    ctx.textAlign='center'; ctx.lineJoin='round';
    const y=f.y-k*cell*.95;
    ctx.lineWidth=5; ctx.strokeStyle='rgba(0,0,0,.6)'; ctx.strokeText(f.text,f.x,y);
    ctx.fillStyle=f.col; ctx.fillText(f.text,f.x,y); ctx.restore(); }

  ctx.restore();
}

resetHud(); uiTurn();
fit(); requestAnimationFrame(()=>fit());
