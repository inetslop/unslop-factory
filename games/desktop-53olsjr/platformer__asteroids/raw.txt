const secName=n=> n<=4 ? SECT[n] : 'DEEP SPACE '+String(n-4).padStart(2,'0');

function buildSector(n){
  const W=2100+n*260;
  G.platforms=[]; G.crystals=[]; G.powerups=[]; G.asteroids=[]; G.bullets=[]; G.popups=[]; G.particles=[];
  let x=300,i=0; const pat=[356,296,244,318,262,212];
  while(x<W-560){
    const w=rand(104,168), y=pat[i%pat.length]+rand(-14,14);
    G.platforms.push({x,y,w,h:16,id:i*7+3});
    if(Math.random()<.5) G.crystals.push(mkCrystal(x+w/2,y-46));
    x+=w+rand(96,168); i++;
  }
  const extra=Math.min(4+n,7);
  for(let a=0;a<extra;a++) G.crystals.push(mkCrystal(rand(500,W-520),rand(230,330)));
  G.portal={x:W-130,y:GROUND-62,r:46,t:0};
  G.levelW=W;
  P.x=90; P.y=GROUND-P.h; P.vx=0; P.vy=0; P.face=1; P.grounded=true; P.airJumps=0;
  P.inv=0; P.shieldT=0; P.tripleT=0; P.sq=1; P.flash=0; P.walk=0; P.jbuf=0;
  G.camX=0; G.spawnT=1.5; G.clearing=false; G.clearT=0;
}
function resetGame(){
  G={mode:'menu',time:0,score:0,sector:1,lives:3,camX:0,shake:0,endless:false,
     clearing:false,clearT:0,spawnT:1.5,levelW:2400,portal:{x:9999,y:400,r:46,t:0},
     platforms:[],crystals:[],powerups:[],asteroids:[],bullets:[],particles:[],popups:[]};
  P={w:26,h:34,x:90,y:GROUND-34,vx:0,vy:0,face:1,grounded:true,coyote:0,airJumps:0,jbuf:0,
     inv:0,shieldT:0,tripleT:0,sq:1,flash:0,walk:0,cd:0};
  buildSector(1);
  syncHUD(true);
}

/* ================= particles / popups ================= */
function burst(x,y,n,cols,spd,grav){
  for(let i=0;i<n;i++){ const a=rand(0,TAU),s=rand(spd*.3,spd);
    G.particles.push({x,y,vx:Math.cos(a)*s,vy:Math.sin(a)*s,life:rand(.3,.8),max:.8,
      size:rand(1.5,4),col:cols[i%cols.length],grav:grav===undefined?900:grav}); }
  if(G.particles.length>420) G.particles.splice(0,G.particles.length-420);
}
function ring(x,y,r,col){ G.particles.push({ring:true,x,y,size:r,col,life:.4,max:.4}); }
function addPop(x,y,txt,col){ G.popups.push({x,y,txt,col,t:0}); }
function addScore(n,x,y,col){ G.score+=n; if(G.score>best){best=G.score;} if(x!==undefined)addPop(x,y,'+'+n,col||'#ffb454'); }

/* ================= spawning ================= */
function spawnAst(){
  const n=G.sector, sf=1+(n-1)*.1+Math.max(0,n-4)*.05;
  const maxAir=Math.min(4+n*2,13);
  if(G.asteroids.filter(a=>!a.boulder).length>=maxAir) return;
  let type='L', r=Math.random();
  if(r<.45)type='L'; else if(r<.78)type='M'; else type='S';
  const shard = n>=2 && Math.random()<Math.min(.3,.1+(n-2)*.04);
  let x,y,vx,vy;
  if(shard){
    type='shard';
    const side=Math.random()<.5?-1:1;
    x=side<0?G.camX-90:G.camX+VIEWW+90; y=rand(80,360);
    const tx=P.x+rand(-120,120), ty=rand(200,400);
    const d=Math.hypot(tx-x,ty-y)||1, sp=300*sf;
    vx=(tx-x)/d*sp; vy=(ty-y)/d*sp;
  }else if(Math.random()<.55){
    x=rand(G.camX-80,G.camX+VIEWW+80); y=-70;
    vx=rand(-70,70)*sf; vy=rand(70,130)*sf;
  }else{
    const side=Math.random()<.5?-1:1;
    x=side<0?G.camX-90:G.camX+VIEWW+90; y=rand(90,360);
    vx=(side<0?1:-1)*rand(80,170)*sf; vy=rand(-40,40);
  }
  G.asteroids.push(mkAst(type,clamp(x,-80,G.levelW+80),y,vx,vy));
}

/* ================= combat ================= */
function killAst(a){
  const SC={L:20,M:50,S:100,shard:75}, szIdx={L:2,M:1,S:0,shard:0};
  addScore(SC[a.type],a.x,a.y-a.r-6,a.type==='shard'?'#ff8d7e':'#ffb454');
  burst(a.x,a.y,14+szIdx[a.type]*8,['#8a6f5c','#c9a37e','#ffcf7e'],220);
  ring(a.x,a.y,a.r*1.7,a.type==='shard'?'rgba(255,107,125,':'rgba(255,190,110,');
  sBoom(szIdx[a.type]); G.shake=Math.max(G.shake,4+szIdx[a.type]*3);
  if(a.type==='L'||a.type==='M'){
    const child=a.type==='L'?'M':'S';
    for(let k=0;k<2;k++){ const an=rand(0,TAU), sp=Math.min(250,Math.hypot(a.vx,a.vy)*1.3+rand(30,90));
      G.asteroids.push(mkAst(child,a.x+Math.cos(an)*8,a.y+Math.sin(an)*8,Math.cos(an)*sp,Math.sin(an)*sp*.55)); }
    if(Math.random()<.12) G.powerups.push({x:a.x,y:a.y,vy:60,type:Math.random()<.5?'T':'S',life:15});
  }
  if(Math.random()<.3 && G.crystals.length<12) G.crystals.push(Object.assign(mkCrystal(a.x,a.y),{drop:true}));
  a.dead=true;
}

/* ================= update ================= */
function circleRect(cx,cy,r,rx,ry,rw,rh){
  const nx=clamp(cx,rx,rx+rw), ny=clamp(cy,ry,ry+rh);
  return (cx-nx)*(cx-nx)+(cy-ny)*(cy-ny) < r*r;
}

function update(dt){
  G.time+=dt; G.shake=Math.max(0,G.shake-dt*34);
  G.portal.t+=dt; menuAst.ang+=dt*.25;
  // particles & popups always
  for(const p of G.particles){ p.life-=dt; if(!p.ring){p.vy+=(p.grav||0)*dt;p.x+=p.vx*dt;p.y+=p.vy*dt;} }
  G.particles=G.particles.filter(p=>p.life>0);
  for(const q of G.popups) q.t+=dt;
  G.popups=G.popups.filter(q=>q.t<1.1);

  if(G.mode!=='playing') return;

  /* --- clearing sequence --- */
  if(G.clearing){
    G.clearT-=dt;
    if(G.clearT<=0){
      G.clearing=false;
      if(G.sector===4 && !G.endless){ setMode('win'); }
      else startSector(G.sector+1);
    }
  }

  /* --- player --- */
  const fire=FIRES.some(c=>K[c]);
  let target=0;
  if(K['ArrowLeft']||K['KeyA'])target-=340;
  if(K['ArrowRight']||K['KeyD'])target+=340;
  P.vx=lerp(P.vx,target,1-Math.exp(-14*dt));
  if(target!==0)P.face=target>0?1:-1;

  P.jbuf-=dt;
  if(P.grounded){P.coyote=.1;P.airJumps=0;}else P.coyote-=dt;
  if(P.jbuf>0){
    if(P.grounded||P.coyote>0){ P.vy=-780; P.grounded=false; P.sq=1.3; P.jbuf=0; sJump(); }
    else if(P.airJumps<1){ P.vy=-700; P.airJumps=1; P.sq=1.28; P.jbuf=0; sDJump();
      burst(P.x+P.w/2,P.y+P.h,9,['#ffd9a0','#fff'],150,300); }
  }
  P.vy=Math.min(P.vy+2300*dt,1250);
  const prevBottom=P.y+P.h;
  P.x=clamp(P.x+P.vx*dt,8,G.levelW-8-P.w);
  P.y+=P.vy*dt;
  const wasG=P.grounded; P.grounded=false;
  if(P.y+P.h>=GROUND && P.vy>=0){ P.y=GROUND-P.h; P.vy=0; P.grounded=true; }
  if(P.vy>=0) for(const pl of G.platforms){
    if(P.x+P.w>pl.x-2 && P.x<pl.x+pl.w+2 && prevBottom<=pl.y+6 && P.y+P.h>=pl.y){
      P.y=pl.y-P.h; P.vy=0; P.grounded=true; }
  }
  for(const a of G.asteroids) if(a.boulder){
    const top=a.y-a.r;
    if(P.vy>=0 && P.x+P.w>a.x-a.r*.8 && P.x<a.x+a.r*.8 && prevBottom<=top+9 && P.y+P.h>=top){
      P.y=top-P.h; P.vy=0; P.grounded=true; }
  }
  if(P.grounded&&!wasG){ P.sq=.72; burst(P.x+P.w/2,P.y+P.h,6,['#6b5142','#8a6f5c'],90,500); }

  /* --- boulder side push --- */
  for(const a of G.asteroids) if(a.boulder){
    const dx=(P.x+P.w/2)-a.x;
    if(Math.abs(dx)<a.r+P.w/2 && P.y+P.h>a.y-a.r+8 && P.y<a.y+a.r){
      P.x+=(dx>0?1:-1)*((a.r+P.w/2)-Math.abs(dx)); }
  }

  /* --- shooting --- */
  P.cd-=dt; P.flash-=dt;
  if(fire&&P.cd<=0&&!G.clearing){
    P.cd=.17; P.flash=.06; sShoot();
    const by=P.y+17, bx=P.face>0?P.x+P.w+6:P.x-6;
    if(P.tripleT>0){ for(const vy of[-150,0,150]) G.bullets.push({x:bx,y:by,vx:P.face*950,vy,life:0}); }
    else G.bullets.push({x:bx,y:by,vx:P.face*950,vy:0,life:0});
  }

  /* --- bullets --- */
  for(const b of G.bullets){
    b.x+=b.vx*dt; b.y+=b.vy*dt; b.life+=dt;
    if(b.x<G.camX-140||b.x>G.camX+VIEWW+140||b.life>1.6)b.dead=true;
    if(b.dead)continue;
    for(const a of G.asteroids){
      if(a.dead)continue;
      const dx=b.x-a.x,dy=b.y-a.y;
      if(dx*dx+dy*dy<(a.r+4)*(a.r+4)){
        b.dead=true; a.hp--; a.flash=.08;
        burst(b.x,b.y,5,['#ffe08a','#fff'],180,200);
        a.vx+=Math.sign(b.vx)*12;
        if(a.hp<=0)killAst(a); else sThud();
        break;
      }
    }
  }
  G.bullets=G.bullets.filter(b=>!b.dead);
  G.asteroids=G.asteroids.filter(a=>!a.dead);

  /* --- asteroids --- */
  const sf=1+(G.sector-1)*.1+Math.max(0,G.sector-4)*.05;
  for(const a of G.asteroids){
    if(a.boulder)continue;
    a.flash=Math.max(0,a.flash-dt);
    a.x+=a.vx*dt; a.y+=a.vy*dt; a.ang+=a.spin*dt;
    if(a.y+a.r>GROUND && a.vy>0){
      a.y=GROUND-a.r; a.vy=-a.vy*.5; a.vx*=.85; sThud();
      burst(a.x,GROUND,5,['#6b5142'],70,400);
      if(Math.abs(a.vy)<42&&Math.abs(a.vx)<52){
        a.boulder=true;a.vx=0;a.vy=0;
        const bs=G.asteroids.filter(b=>b.boulder);
        if(bs.length>7){ const old=bs[0]; burst(old.x,old.y,12,['#6b5142','#8a6f5c'],140); old.dead=true; }
      }
    }
    if(a.x<-170||a.x>G.levelW+170||a.y>WH+90)a.dead=true;
  }
  G.asteroids=G.asteroids.filter(a=>!a.dead);

  /* --- spawn --- */
  if(!G.clearing){
    G.spawnT-=dt;
    if(G.spawnT<=0){
      spawnAst();
      const base=Math.max(.55,2.3-(G.sector-1)*.28)*(G.sector>4?Math.max(.6,1-(G.sector-4)*.05):1);
      G.spawnT=base*rand(.75,1.3);
    }
  }

  /* --- damage --- */
  P.inv-=dt; P.shieldT-=dt; P.tripleT-=dt;
  if(!G.clearing&&P.inv<=0){
    for(const a of G.asteroids){
      if(a.boulder||a.dead)continue;
      if(circleRect(a.x,a.y,a.r*.9,P.x+3,P.y+3,P.w-6,P.h-6)){
        if(P.shieldT>0){
          P.shieldT=0; P.inv=.8; sShield();
          ring(P.x+P.w/2,P.y+P.h/2,40,'rgba(126,232,255,');
          burst(P.x+P.w/2,P.y+P.h/2,14,['#7ee8ff','#fff'],240);
        }else{
          G.lives--; P.inv=1.6; P.vx=Math.sign(P.x+P.w/2-a.x)*260; P.vy=-380;
          G.shake=13; sHurt();
          ring(P.x+P.w/2,P.y+P.h/2,46,'rgba(255,93,93,');
          burst(P.x+P.w/2,P.y+P.h/2,16,['#ff5d5d','#ffb454'],260);
          syncLives();
          if(G.lives<=0){ setMode('over'); return; }
        }
        break;
      }
    }
  }

  /* --- crystals --- */
  for(const c of G.crystals){
    if(c.drop&&c.y<GROUND-12)c.y+=60*dt;
    const cy=c.y+Math.sin(G.time*2+c.phase)*5;
    if(Math.abs(P.x+P.w/2-c.x)<24&&Math.abs(P.y+P.h/2-cy)<30){
      c.dead=true; addScore(50,c.x,cy-14,'#7ef0dc'); sCrystal();
      burst(c.x,cy,10,['#7ef0dc','#fff'],180);
    }
  }
  G.crystals=G.crystals.filter(c=>!c.dead);

  /* --- powerups --- */
  for(const u of G.powerups){
    u.life-=dt;
    if(u.y<GROUND-14)u.y+=60*dt;
    const uy=u.y+Math.sin(G.time*3)*4;
    if(Math.abs(P.x+P.w/2-u.x)<26&&Math.abs(P.y+P.h/2-uy)<32){
      u.dead=true; sPower();
      if(u.type==='T'){ P.tripleT=8; addPop(u.x,uy-14,'TRIPLE SHOT!','#ffb454'); }
      else{ P.shieldT=6; addPop(u.x,uy-14,'SHIELD UP!','#7ee8ff'); }
      ring(u.x,uy,30,u.type==='T'?'rgba(255,180,84,':'rgba(126,232,255,');
    }
  }
  G.powerups=G.powerups.filter(u=>!u.dead&&u.life>0);

  /* --- portal --- */
  if(!G.clearing&&circleRect(G.portal.x,G.portal.y,G.portal.r,P.x,P.y,P.w,P.h)){
    G.clearing=true; G.clearT=1.0;
    const bonus=200+G.sector*100;
    addScore(bonus,G.portal.x,G.portal.y-70,'#8fe9ff');
    sPortal(); G.shake=6;
    ring(G.portal.x,G.portal.y,110,'rgba(143,233,255,');
    ring(G.portal.x,G.portal.y,70,'rgba(255,180,84,');
    burst(G.portal.x,G.portal.y,26,['#8fe9ff','#ffb454','#fff'],300);
    saveBest();
  }

  /* --- timers / squash --- */
  P.sq=lerp(P.sq,1,Math.min(1,dt*9));
  if(Math.abs(P.vx)>30)P.walk+=Math.abs(P.vx)*dt*.05;

  /* --- camera --- */
  const tgt=clamp(P.x-VIEWW*.38+P.face*50,0,Math.max(0,G.levelW-VIEWW));
  G.camX=lerp(G.camX,tgt,Math.min(1,dt*6));
}

/* ================= HUD sync ================= */
let lastLives=-1;
function syncLives(){ livesEl.innerHTML=[0,1,2].map(i=>`<span class="life${i<G.lives?'':' off'}"></span>`).join(''); lastLives=G.lives; }
function syncHUD(force){
  if(force)lastLives=-1;
  $('score').textContent=G.score;
  $('best').textContent='BEST '+best;
  $('secName').textContent=`SECTOR ${G.sector} · ${secName(G.sector)}`;
  $('prog').style.width=clamp(P.x/G.portal.x*100,0,100)+'%';
  if(lastLives!==G.lives)syncLives();
  const b=$('buffs');
  if(G.tripleT>0){ b.style.display='inline-block'; b.textContent='TRIPLE '+Math.ceil(G.tripleT)+'s'; b.style.borderColor='#ffb454'; b.style.color='#ffb454'; }
  else if(G.shieldT>0){ b.style.display='inline-block'; b.textContent='SHIELD '+Math.ceil(G.shieldT)+'s'; b.style.borderColor='#7ee8ff'; b.style.color='#7ee8ff'; }
  else b.style.display='none';
}

const stars=[[],[],[]];
(function genStars(){
  const cfg=[[90,.9,.14],[55,1.5,.32],[30,2.3,.6]];
  for(let l=0;l<3;l++)for(let i=0;i<cfg[l][0];i++)
    stars[l].push({x:Math.random()*2200,y:Math.random()*(WH-40),r:cfg[l][1]*rand(.7,1.3),
      tw:rand(1.5,4),ph:rand(0,TAU),warm:Math.random()<.12});
})();
const nebulas=[
  {x:200,y:120,r:300,c:'64,196,184',a:.09,par:.08},
  {x:900,y:300,r:340,c:'255,150,70',a:.07,par:.1},
  {x:1500,y:90,r:260,c:'90,130,220',a:.08,par:.12},
  {x:400,y:380,r:280,c:'64,196,184',a:.05,par:.14}
];
function wrap(x,span){ const v=((x%span)+span)%span; return v-(span-VIEWW)/2; }

function drawBackground(){
  const g=ctx.createLinearGradient(0,0,0,WH);
  g.addColorStop(0,'#081724'); g.addColorStop(.65,'#0d2434'); g.addColorStop(1,'#123043');
  ctx.fillStyle=g; ctx.fillRect(0,0,VIEWW,WH);
  const hz=ctx.createLinearGradient(0,GROUND-160,0,WH);
  hz.addColorStop(0,'rgba(255,150,70,0)'); hz.addColorStop(1,'rgba(255,150,70,.1)');
  ctx.fillStyle=hz; ctx.fillRect(0,GROUND-160,VIEWW,160);
  // nebulas
  for(const nb of nebulas){
    const nx=wrap(nb.x-G.camX*nb.par,VIEWW+900), ny=nb.y;
    const rg=ctx.createRadialGradient(nx,ny,0,nx,ny,nb.r);
    rg.addColorStop(0,`rgba(${nb.c},${nb.a})`); rg.addColorStop(1,`rgba(${nb.c},0)`);
    ctx.fillStyle=rg; ctx.fillRect(nx-nb.r,ny-nb.r,nb.r*2,nb.r*2);
  }
  // planet
  {
    const px=wrap(1500-G.camX*.12,VIEWW+900), py=118, pr=64;
    ctx.save();
    ctx.translate(px,py);
    const pg=ctx.createRadialGradient(-pr*.4,-pr*.4,pr*.1,0,0,pr);
    pg.addColorStop(0,'#3d5f7d'); pg.addColorStop(1,'#16283a');
    ctx.fillStyle=pg; ctx.beginPath(); ctx.arc(0,0,pr,0,TAU); ctx.fill();
    ctx.strokeStyle='rgba(240,190,140,.5)'; ctx.lineWidth=2.5;
    ctx.beginPath(); ctx.arc(-pr*.15,-pr*.1,pr*.96,Math.PI*1.05,Math.PI*1.7); ctx.stroke();
    ctx.rotate(-.32);
    ctx.strokeStyle='rgba(230,180,130,.4)'; ctx.lineWidth=4;
    ctx.beginPath(); ctx.ellipse(0,0,pr*1.7,pr*.36,0,0,TAU); ctx.stroke();
    ctx.strokeStyle='rgba(230,180,130,.18)'; ctx.lineWidth=9;
    ctx.beginPath(); ctx.ellipse(0,0,pr*1.95,pr*.44,0,0,TAU); ctx.stroke();
    ctx.restore();
  }
  // stars
  for(let l=0;l<3;l++){
    const par=[.14,.32,.6][l], span=VIEWW+80;
    for(const s of stars[l]){
      const sx=wrap(s.x-G.camX*par,span);
      if(sx<-20||sx>VIEWW+20)continue;
      const a=.35+.55*Math.abs(Math.sin(G.time*s.tw*.4+s.ph));
      ctx.fillStyle=s.warm?`rgba(255,217,160,${a})`:`rgba(207,230,239,${a})`;
      ctx.fillRect(sx,s.y,s.r,s.r);
    }
  }
}

/* ================= world drawing ================= */
function rr(x,y,w,h,r){
  ctx.beginPath();
  ctx.moveTo(x+r,y); ctx.lineTo(x+w-r,y); ctx.arcTo(x+w,y,x+w,y+r,r);
  ctx.lineTo(x+w,y+h-r); ctx.arcTo(x+w,y+h,x+w-r,y+h,r);
  ctx.lineTo(x+r,y+h); ctx.arcTo(x,y+h,x,y+h-r,r);
  ctx.lineTo(x,y+r); ctx.arcTo(x,y,x+r,y,r); ctx.closePath();
}
function drawGround(){
  const x0=G.camX-60,w=VIEWW+120;
  const g=ctx.createLinearGradient(0,GROUND,0,WH);
  g.addColorStop(0,'#2a1e17'); g.addColorStop(1,'#150d07');
  ctx.fillStyle=g; ctx.fillRect(x0,GROUND,w,WH-GROUND);
  const i0=Math.floor((G.camX-60)/24), i1=Math.ceil((G.camX+VIEWW+60)/24);
  for(let i=i0;i<i1;i++){
    const hx=hash(i), x=i*24;
    ctx.fillStyle=`rgba(255,170,90,${.18+.3*hx})`;
    ctx.fillRect(x,GROUND,24,2.5);
    if(hash(i*7)>.58){
      const cx=x+hash(i*3)*24, cy=GROUND+16+hash(i*5)*38, cr=5+hash(i*11)*9;
      ctx.fillStyle='rgba(0,0,0,.28)';
      ctx.beginPath(); ctx.ellipse(cx,cy,cr,cr*.45,0,0,TAU); ctx.fill();
    }
  }
}
function drawPlatforms(){
  for(const p of G.platforms){
    if(p.x+p.w<G.camX-40||p.x>G.camX+VIEWW+40)continue;
    const g=ctx.createLinearGradient(0,p.y,0,p.y+p.h);
    g.addColorStop(0,'#7a5c48'); g.addColorStop(1,'#463529');
    ctx.fillStyle=g; rr(p.x,p.y,p.w,p.h,5); ctx.fill();
    ctx.fillStyle='rgba(234,177,121,.95)'; ctx.fillRect(p.x+2,p.y,p.w-4,3);
    const n=Math.floor(p.w/34);
    for(let k=0;k<n;k++){
      if(hash(p.id*13+k)>.45){
        ctx.fillStyle='rgba(0,0,0,.3)';
        ctx.beginPath(); ctx.arc(p.x+8+hash(p.id+k)* (p.w-16), p.y+9, 2.6,0,TAU); ctx.fill();
      }
    }
  }
}
function drawPortal(){
  const pt=G.portal, t=pt.t;
  const pulse=1+.08*Math.sin(t*3)+(G.clearing?(1-G.clearT)*.5:0);
  let rg=ctx.createRadialGradient(pt.x,pt.y,4,pt.x,pt.y,72*pulse);
  rg.addColorStop(0,'rgba(143,233,255,.4)'); rg.addColorStop(1,'rgba(143,233,255,0)');
  ctx.fillStyle=rg; ctx.fillRect(pt.x-80,pt.y-80,160,160);
  rg=ctx.createRadialGradient(pt.x,pt.y,2,pt.x,pt.y,pt.r*.9);
  rg.addColorStop(0,'rgba(10,40,54,.95)'); rg.addColorStop(1,'rgba(18,70,88,.6)');
  ctx.fillStyle=rg; ctx.beginPath(); ctx.arc(pt.x,pt.y,pt.r*.9,0,TAU); ctx.fill();
  ctx.lineWidth=3.5; ctx.strokeStyle='#8fe9ff';
  ctx.beginPath(); ctx.arc(pt.x,pt.y,pt.r,t*2.2,t*2.2+1.9); ctx.stroke();
  ctx.lineWidth=3; ctx.strokeStyle='#ffb454';
  ctx.beginPath(); ctx.arc(pt.x,pt.y,pt.r*.78,-t*1.7+2,-t*1.7+3.6); ctx.stroke();
  for(let i=0;i<5;i++){
    const a=t*2+i/5*TAU, r=pt.r*(.4+.16*Math.sin(t*3+i));
    ctx.fillStyle='rgba(255,255,255,.8)';
    ctx.beginPath(); ctx.arc(pt.x+Math.cos(a)*r,pt.y+Math.sin(a)*r,2,0,TAU); ctx.fill();
  }
}
function drawCrystals(){
  for(const c of G.crystals){
    if(c.x<G.camX-40||c.x>G.camX+VIEWW+40)continue;
    const y=c.y+Math.sin(G.time*2+c.phase)*5;
    const rg=ctx.createRadialGradient(c.x,y,1,c.x,y,18);
    rg.addColorStop(0,'rgba(126,240,220,.4)'); rg.addColorStop(1,'rgba(126,240,220,0)');
    ctx.fillStyle=rg; ctx.fillRect(c.x-18,y-18,36,36);
    ctx.save(); ctx.translate(c.x,y); ctx.rotate(Math.sin(G.time+c.phase)*.25);
    ctx.beginPath(); ctx.moveTo(0,-9); ctx.lineTo(6,0); ctx.lineTo(0,9); ctx.lineTo(-6,0); ctx.closePath();
    ctx.fillStyle='#7ef0dc'; ctx.fill();
    ctx.strokeStyle='rgba(234,255,251,.9)'; ctx.lineWidth=1.4; ctx.stroke();
    ctx.restore();
    if(Math.sin(G.time*6+c.phase)>.9){
      ctx.strokeStyle='rgba(255,255,255,.9)'; ctx.lineWidth=1;
      ctx.beginPath(); ctx.moveTo(c.x-8,y); ctx.lineTo(c.x+8,y); ctx.moveTo(c.x,y-8); ctx.lineTo(c.x,y+8); ctx.stroke();
    }
  }
}
function drawPowerups(){
  for(const u of G.powerups){
    if(u.x<G.camX-40||u.x>G.camX+VIEWW+40)continue;
    const y=u.y+Math.sin(G.time*3)*4;
    let a=1; if(u.life<3)a=.3+.7*Math.abs(Math.sin(G.time*8));
    ctx.save(); ctx.globalAlpha=a; ctx.translate(u.x,y);
    const col=u.type==='T'?'#ffb454':'#7ee8ff';
    ctx.beginPath();
    for(let i=0;i<6;i++){const an=i/6*TAU-Math.PI/2;ctx.lineTo(Math.cos(an)*13,Math.sin(an)*13);}
    ctx.closePath(); ctx.fillStyle='#14242c'; ctx.fill();
    ctx.strokeStyle=col; ctx.lineWidth=2; ctx.stroke();
    ctx.fillStyle=col; ctx.font='700 13px Impact, Arial, sans-serif';
    ctx.textAlign='center'; ctx.textBaseline='middle'; ctx.fillText(u.type,0,1);
    ctx.restore();
  }
}
function drawAsteroid(a){
  if(a.x+a.r*1.4<G.camX||a.x-a.r*1.4>G.camX+VIEWW)return;
  if(a.type==='shard'){
    const rg=ctx.createRadialGradient(a.x,a.y,1,a.x,a.y,a.r*2.4);
    rg.addColorStop(0,'rgba(255,107,125,.3)'); rg.addColorStop(1,'rgba(255,107,125,0)');
    ctx.fillStyle=rg; ctx.fillRect(a.x-a.r*2.4,a.y-a.r*2.4,a.r*4.8,a.r*4.8);
  }
  ctx.save(); ctx.translate(a.x,a.y); ctx.rotate(a.ang);
  ctx.beginPath();
  if(a.type==='shard'){
    for(let i=0;i<16;i++){const rr2=i%2?a.r:a.r*1.9,an=i/16*TAU;ctx.lineTo(Math.cos(an)*rr2,Math.sin(an)*rr2);}
  }else{
    a.verts.forEach((v,i)=>{const x=Math.cos(v.a)*v.d*a.r,y=Math.sin(v.a)*v.d*a.r;i?ctx.lineTo(x,y):ctx.moveTo(x,y);});
  }
  ctx.closePath();
  ctx.fillStyle=a.boulder?'#6d5546':'#7d6553'; ctx.fill();
  ctx.strokeStyle='#41322a'; ctx.lineWidth=2.5; ctx.stroke();
  if(a.type!=='shard'){
    ctx.fillStyle='rgba(0,0,0,.28)';
    ctx.beginPath(); ctx.arc(a.c1.dx,a.c1.dy,a.c1.rr,0,TAU); ctx.fill();
  }
  if(a.flash>0){ ctx.globalAlpha=a.flash*9; ctx.fillStyle='#fff'; ctx.fill(); ctx.globalAlpha=1; }
  ctx.restore();
  if(a.type!=='shard'){
    ctx.strokeStyle='rgba(240,190,140,.4)'; ctx.lineWidth=2;
    ctx.beginPath(); ctx.arc(a.x-a.r*.28,a.y-a.r*.3,a.r*.72,Math.PI*1.05,Math.PI*1.6); ctx.stroke();
  }
}
function drawPlayer(){
  if(G.mode==='menu')return;
  const cx=P.x+P.w/2, feet=P.y+P.h;
  if(P.inv>0&&(G.time*14|0)%2===0)ctx.globalAlpha=.35;
  ctx.save();
  ctx.translate(cx,feet); ctx.scale(1+(1-P.sq)*.6,P.sq); ctx.translate(-cx,-feet);
  const swing=P.grounded?Math.sin(P.walk)*4:3;
  ctx.fillStyle='#d8cbb4';
  ctx.fillRect(P.x+6-(P.grounded?swing:2),feet-10,5,10);
  ctx.fillRect(P.x+P.w-11+(P.grounded?swing:2),feet-10,5,10);
  ctx.fillStyle='#ff7a2f';
  ctx.fillRect(P.x+6-(P.grounded?swing:2),feet-3,5,3);
  ctx.fillRect(P.x+P.w-11+(P.grounded?swing:2),feet-3,5,3);
  ctx.fillStyle='#e0642c';
  ctx.fillRect(P.face>0?P.x-5:P.x+P.w-1,P.y+13,6,12);
  ctx.fillStyle='#efe7d6'; rr(P.x+3,P.y+11,P.w-6,P.h-21,4); ctx.fill();
  ctx.fillStyle='rgba(255,122,47,.9)'; ctx.fillRect(P.x+3,P.y+18,P.w-6,4);
  ctx.fillStyle='#f7f2e6'; ctx.beginPath(); ctx.arc(cx,P.y+9,9.5,0,TAU); ctx.fill();
  ctx.strokeStyle='rgba(0,0,0,.15)'; ctx.lineWidth=1; ctx.stroke();
  ctx.fillStyle='#8fe9ff'; rr(cx-5+P.face*1.5,P.y+6,10,6,3); ctx.fill();
  ctx.fillStyle='rgba(255,255,255,.9)'; ctx.beginPath(); ctx.arc(cx-2+P.face*1.5,P.y+8,1.4,0,TAU); ctx.fill();
  ctx.restore();
  // gun
  const gx=P.face>0?P.x+P.w-2:P.x-10;
  ctx.fillStyle='#9fb0bd'; ctx.fillRect(gx,P.y+17,12,5);
  ctx.fillStyle='#5b6a76'; ctx.fillRect(P.face>0?gx+9:gx,P.y+17,3,5);
  if(P.flash>0){
    const mx=P.face>0?P.x+P.w+8:P.x-8;
    ctx.fillStyle='rgba(255,220,120,.9)';
    ctx.beginPath(); ctx.arc(mx,P.y+19.5,6,0,TAU); ctx.fill();
    ctx.fillStyle='#fff'; ctx.beginPath(); ctx.arc(mx,P.y+19.5,2.6,0,TAU); ctx.fill();
  }
  ctx.globalAlpha=1;
  if(P.shieldT>0){
    const r2=24+Math.sin(G.time*9)*1.5;
    ctx.strokeStyle=`rgba(126,232,255,${.5+.25*Math.sin(G.time*9)})`;
    ctx.lineWidth=2.2; ctx.beginPath(); ctx.arc(cx,P.y+P.h/2,r2,0,TAU); ctx.stroke();
    ctx.fillStyle='rgba(126,232,255,.07)'; ctx.fill();
  }
}
function drawBullets(){
  for(const b of G.bullets){
    const an=Math.atan2(b.vy,b.vx);
    ctx.save(); ctx.translate(b.x,b.y); ctx.rotate(an);
    ctx.fillStyle='rgba(255,215,94,.35)';
    ctx.beginPath(); ctx.arc(0,0,7,0,TAU); ctx.fill();
    ctx.fillStyle='#ffd75e'; ctx.fillRect(-9,-2.5,16,5);
    ctx.fillStyle='#fff8e6'; ctx.fillRect(-4,-1,10,2);
    ctx.restore();
  }
}
function drawParticles(){
  for(const p of G.particles){
    const a=clamp(p.life/p.max,0,1);
    if(p.ring){
      const r=p.size*(1-a)+6;
      ctx.strokeStyle=p.col+(a*.8)+')'; ctx.lineWidth=3*a+1;
      ctx.beginPath(); ctx.arc(p.x,p.y,r,0,TAU); ctx.stroke();
    }else{
      ctx.globalAlpha=a; ctx.fillStyle=p.col;
      ctx.beginPath(); ctx.arc(p.x,p.y,p.size*a+.5,0,TAU); ctx.fill();
      ctx.globalAlpha=1;
    }
  }
}
function drawPopups(){
  ctx.font='700 15px ui-monospace, Menlo, Consolas, monospace';
  ctx.textAlign='center'; ctx.textBaseline='middle';
  for(const q of G.popups){
    const a=q.t<.6?1:1-(q.t-.6)/.5;
    ctx.globalAlpha=clamp(a,0,1);
    ctx.strokeStyle='rgba(0,0,0,.6)'; ctx.lineWidth=3;
    ctx.strokeText(q.txt,q.x,q.y-q.t*34);
    ctx.fillStyle=q.col; ctx.fillText(q.txt,q.x,q.y-q.t*34);
  }
  ctx.globalAlpha=1;
}

/* ================= render ================= */
function render(){
  ctx.setTransform(1,0,0,1,0,0);
  ctx.clearRect(0,0,cv.width,cv.height);
  ctx.scale(DPR*SCALE,DPR*SCALE);
  drawBackground();
  const shx=(Math.random()*2-1)*G.shake, shy=(Math.random()*2-1)*G.shake;
  ctx.save(); ctx.translate(-G.camX+shx,shy);
  if(G.mode==='menu'){
    // attract asteroid
    const mx=VIEWW/2,my=WH*.55;
    ctx.save(); ctx.translate(mx,my); ctx.rotate(menuAst.ang);
    ctx.beginPath();
    menuAst.verts.forEach((v,i)=>{const x=Math.cos(v.a)*v.d*menuAst.r,y=Math.sin(v.a)*v.d*menuAst.r;i?ctx.lineTo(x,y):ctx.moveTo(x,y);});
    ctx.closePath(); ctx.fillStyle='#7d6553'; ctx.fill();
    ctx.strokeStyle='#41322a'; ctx.lineWidth=3; ctx.stroke();
    ctx.fillStyle='rgba(0,0,0,.28)'; ctx.beginPath(); ctx.arc(-14,-8,16,0,TAU); ctx.fill();
    ctx.beginPath(); ctx.arc(22,18,10,0,TAU); ctx.fill();
    ctx.restore();
    ctx.strokeStyle='rgba(240,190,140,.35)'; ctx.lineWidth=2.5;
    ctx.beginPath(); ctx.arc(mx-menuAst.r*.28,my-menuAst.r*.3,menuAst.r*.72,Math.PI*1.05,Math.PI*1.6); ctx.stroke();
  }else{
    drawGround(); drawPlatforms(); drawPortal();
    drawCrystals(); drawPowerups();
    for(const a of G.asteroids)if(a.boulder)drawAsteroid(a);
    drawPlayer();
    for(const a of G.asteroids)if(!a.boulder)drawAsteroid(a);
    drawBullets(); drawParticles(); drawPopups();
  }
  ctx.restore();
}

/* ================= main loop ================= */
let last=performance.now();
function loop(ts){
  const dt=Math.min((ts-last)/1000,.033); last=ts;
  update(dt); render();
  if(G.mode!=='menu')syncHUD(false);
  requestAnimationFrame(loop);
}

resetGame(); setMode('menu');
requestAnimationFrame(loop);
})();
