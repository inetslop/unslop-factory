  cvs.width=Math.round(W*DPR);cvs.height=Math.round(H*DPR);if(gw)layout()}
window.addEventListener('resize',resize);
function layout(){
  cell=Math.max(6,Math.floor(Math.min((W-24)/gw,(H-HUD_H-20)/gh)));
  offX=Math.round((W-gw*cell)/2);offY=HUD_H+Math.round((H-HUD_H-gh*cell)/2);
  buildMazeCanvas();buildExpCanvas();
  fogCvs=document.createElement('canvas');fogCvs.width=W*DPR;fogCvs.height=H*DPR;fogCtx=fogCvs.getContext('2d');
}
function stampBlob(c,cx,cy){const g=c.createRadialGradient(cx,cy,0,cx,cy,cell*1.25);
  g.addColorStop(0,'rgba(255,255,255,.8)');g.addColorStop(1,'rgba(255,255,255,0)');
  c.fillStyle=g;c.beginPath();c.arc(cx,cy,cell*1.25,0,7);c.fill()}
function buildExpCanvas(){expCvs=document.createElement('canvas');expCvs.width=gw*cell*DPR;expCvs.height=gh*cell*DPR;
  expCtx=expCvs.getContext('2d');expCtx.setTransform(DPR,0,0,DPR,0,0);
  for(const i of exploredList)stampBlob(expCtx,((i%gw)+.5)*cell,(((i/gw)|0)+.5)*cell)}
function buildMazeCanvas(){mazeCvs=document.createElement('canvas');mazeCvs.width=gw*cell*DPR;mazeCvs.height=gh*cell*DPR;
  const c=mazeCvs.getContext('2d');c.setTransform(DPR,0,0,DPR,0,0);const C=cell;
  c.fillStyle='#0a141f';c.fillRect(0,0,gw*C,gh*C);
  for(let y=0;y<gh;y++)for(let x=0;x<gw;x++){const i=y*gw+x,px=x*C,py=y*C;
    if(data[i]===0){c.fillStyle=(x+y)%2?'#101d2b':'#0e1926';c.fillRect(px+1,py+1,C-2,C-2);
      const h=h2(x,y);
      if(h<.05){c.fillStyle='rgba(255,255,255,.045)';c.beginPath();c.arc(px+C*(.3+h*6),py+C*(.35+h*4),C*.07,0,7);c.fill()}
      else if(h>.97){c.strokeStyle='rgba(120,160,200,.08)';c.beginPath();c.moveTo(px+C*.25,py+C*.7);c.lineTo(px+C*.7,py+C*.3);c.stroke()}
    }else{c.fillStyle='#2c4159';c.fillRect(px,py,C,C);
      c.strokeStyle='rgba(0,0,0,.32)';c.lineWidth=1;c.strokeRect(px+.5,py+.5,C-1,C-1);
      if(y>0&&data[i-gw]===0){c.fillStyle='#4a6a8c';c.fillRect(px,py,C,Math.max(2,C*.14))}
      if(y<gh-1&&data[i+gw]===0){const h2h=Math.max(2,C*.16);c.fillStyle='#14212f';c.fillRect(px,py+C-h2h,C,h2h)}
      const h=h2(x+7,y+3);if(h<.04&&y>0&&data[i-gw]===1){c.fillStyle='rgba(90,220,180,.10)';c.beginPath();c.arc(px+C*(.3+h*5),py+C*.3,C*.07,0,7);c.fill()}}}
  c.save();c.strokeStyle='rgba(255,184,92,.3)';c.lineWidth=2;c.setLineDash([3,4]);
  c.beginPath();c.arc(1.5*C,1.5*C,C*.32,0,7);c.stroke();c.restore();
}

/* ============ GAME STEP ============ */
function inputDir(){let dx=0,dy=0;
  if(tvec){dx=tvec.x;dy=tvec.y}else{dx=(keys.right?1:0)-(keys.left?1:0);dy=(keys.down?1:0)-(keys.up?1:0)}
  const m=Math.hypot(dx,dy);if(m>1){dx/=m;dy/=m}return[dx,dy]}
function markExplored(){const i=((player.y|0)*gw)+(player.x|0);if(explored[i])return;
  explored[i]=1;exploredList.push(i);stampBlob(expCtx,((i%gw)+.5)*cell,(((i/gw)|0)+.5)*cell)}
function burst(x,y,n,cols,sp){for(let i=0;i<n;i++){const a=Math.random()*6.283,v=sp*(.4+Math.random()*.8);
  particles.push({x,y,vx:Math.cos(a)*v,vy:Math.sin(a)*v,tot:.5+Math.random()*.5,t:0,sz:1.5+Math.random()*2,col:cols[(Math.random()*cols.length)|0]})}}
function onHit(){lives--;updateLives();sfx.hit();shake=13;flash=.85;invulnT=2.4;
  player.x=1.5;player.y=1.5;player.fx=0;player.fy=1;
  for(const w of wraiths){w.path=[];w.chase=false;w.loseT=0}
  burst(offX+player.x*cell,offY+player.y*cell,18,[[255,80,110],[255,160,90]],140);
  if(lives<=0)gameOver()}
function step(dt){
  runTime+=dt;levelTime+=dt;invulnT=Math.max(0,invulnT-dt);
  shake=Math.max(0,shake-46*dt);flash=Math.max(0,flash-2.6*dt);
  const[dx,dy]=inputDir();const moving=!!(dx||dy);
  sprinting=((keys.sprint||(tvec&&tvec.mag>85))&&moving);
  if(moving){const m=Math.hypot(dx,dy)||1;player.fx=dx/m;player.fy=dy/m;
    const v=PS*(sprinting?1.45:1)*dt;moveEntity(player,dx/m*v,dy/m*v);
    footT-=dt;if(footT<=0){footT=sprinting?.17:.26;tone(95,.05,'sine',.035,58)}}
  markExplored();
  for(const r of relics){if(r.got)continue;
    const d=Math.hypot(player.x-(r.bx+.5),player.y-(r.by+.5));
    if(d<.6){r.got=true;score+=100;sfx.pickup();updatePips();
      burst(offX+(r.bx+.5)*cell,offY+(r.by+.5)*cell,16,[[125,243,214],[255,255,255]],150);
      if(relics.every(q=>q.got)){exitC.open=true;sfx.unlock();toast('THE GREEN DOOR UNSEALS')}}}
  if(exitC.open){const d=Math.hypot(player.x-(exitC.bx+.5),player.y-(exitC.by+.5));if(d<.62)levelClear()}
  updateWraiths(dt);updateHUD();
}
function updateWraiths(dt){
  const sf=1+(level-1)*.045;
  for(const w of wraiths){
    const dxp=player.x-w.x,dyp=player.y-w.y,d=Math.hypot(dxp,dyp);
    const see=d<6.5&&los(w.x,w.y,player.x,player.y);
    if(see&&!w.chase){w.chase=true;sfx.aggro()}
    if(w.chase){
      if(!see){w.loseT+=dt;if(w.loseT>1.8||d>9.5){w.chase=false;w.path=[]}}else w.loseT=0;
      w.rt-=dt;if(w.rt<=0){w.path=bfsPath(w.x|0,w.y|0,player.x|0,player.y|0);w.rt=.4}
    }else{const t=w.path[0];
      if(!t||Math.hypot(t[0]+.5-w.x,t[1]+.5-w.y)<.06){
        const bx=w.x|0,by=w.y|0;
        let pool=DIRS.map(v=>[bx+v[0],by+v[1]]).filter(p=>p[0]>=0&&p[1]>=0&&p[0]<gw&&p[1]<gh&&data[p[1]*gw+p[0]]===0);
        const noRev=pool.filter(p=>!(p[0]===bx-w.last[0]&&p[1]===by-w.last[1]));
        if(noRev.length)pool=noRev;
        w.path=pool.length?[pool[(Math.random()*pool.length)|0]]:[]}}
    const tgt=w.path[0];
    if(tgt){const tx=tgt[0]+.5,ty=tgt[1]+.5;let vx=tx-w.x,vy=ty-w.y;const dd=Math.hypot(vx,vy);
      const sp=(w.chase?2.85:1.7)*sf*dt;
      if(dd<sp+.02){w.last=[Math.sign(tx-w.x)||0,Math.sign(ty-w.y)||0];w.x=tx;w.y=ty;w.path.shift()}
      else{w.x+=vx/dd*sp;w.y+=vy/dd*sp}}
    if(Math.random()<dt*13)particles.push({x:offX+w.x*cell,y:offY+w.y*cell,vx:(Math.random()-.5)*14,vy:(Math.random()-.5)*14-6,
      tot:.4+Math.random()*.4,t:0,sz:cell*.1,col:w.chase?[255,80,110]:[255,140,160]});
    if(invulnT<=0&&d<(player.r+w.r)*.9)onHit();
  }
}
function ambient(dt){
  for(let i=particles.length-1;i>=0;i--){const p=particles[i];p.t+=dt;
    if(p.t>=p.tot){particles.splice(i,1);continue}
    p.x+=p.vx*dt;p.y+=p.vy*dt;p.vx*=(1-2.2*dt);p.vy*=(1-2.2*dt)}
  for(let i=dust.length-1;i>=0;i--){const d=dust[i];d.t+=dt;
    if(d.t>=d.tot){dust.splice(i,1);continue}
    d.x+=(d.vx+Math.sin(T*2+d.ph)*6)*dt;d.y+=d.vy*dt}
  if(state!=='pause'&&state!=='over'&&state!=='win'){
    if(dust.length<70&&Math.random()<dt*14){const px=offX+player.x*cell,py=offY+player.y*cell;
      const a=Math.random()*6.283,rr=Math.random()*cell*2.6;
      dust.push({x:px+Math.cos(a)*rr,y:py+Math.sin(a)*rr,vx:(Math.random()-.5)*4,vy:-6-Math.random()*8,tot:1.4+Math.random()*1.8,t:0,ph:Math.random()*7,sz:.8+Math.random()*1.4})}
    if(exitC&&exitC.open&&state==='play'&&Math.random()<dt*5)
      particles.push({x:offX+(exitC.bx+.5)*cell+(Math.random()-.5)*cell*.6,y:offY+(exitC.by+.5)*cell,
        vx:(Math.random()-.5)*10,vy:-24-Math.random()*20,tot:.7+Math.random()*.5,t:0,sz:1.6,col:[125,255,176]})}
}

/* ============ RENDER ============ */
function flick(){return 1+.05*Math.sin(T*9+seed)+.035*Math.sin(T*23+seed*1.7)}
function drawExit(){const x=offX+(exitC.bx+.5)*cell,y=offY+(exitC.by+.5)*cell,R=cell*.5,open=exitC.open;
  ctx.fillStyle='#04121c';ctx.beginPath();ctx.arc(x,y,R*1.15,0,7);ctx.fill();
  const sp=open?2.4:.6,a0=open?.85:.3;
  for(let i=0;i<3;i++){const rr=R*(1.05-i*.24),st=T*sp+i*2.1;
    ctx.strokeStyle=`rgba(125,255,176,${a0*(open?(.6+.4*Math.sin(T*5+i*2)):.5)})`;
    ctx.lineWidth=Math.max(1.5,cell*.07);ctx.beginPath();ctx.arc(x,y,rr,st,st+4.3);ctx.stroke()}
  if(open){ctx.save();ctx.globalCompositeOperation='lighter';
    const g=ctx.createRadialGradient(x,y,0,x,y,R*1.6);g.addColorStop(0,'rgba(125,255,176,.35)');g.addColorStop(1,'rgba(125,255,176,0)');
    ctx.fillStyle=g;ctx.beginPath();ctx.arc(x,y,R*1.6,0,7);ctx.fill();ctx.restore()}
  else{ctx.strokeStyle='rgba(170,255,215,.18)';ctx.lineWidth=Math.max(2,cell*.09);
    for(let i=-1;i<=1;i++){ctx.beginPath();ctx.moveTo(x+i*R*.5,y-R*.75);ctx.lineTo(x+i*R*.5,y+R*.75);ctx.stroke()}}}

  ctx.fillStyle='#aefee4';ctx.beginPath();ctx.moveTo(0,-s);ctx.lineTo(s*.68,0);ctx.lineTo(0,s);ctx.lineTo(-s*.68,0);ctx.closePath();ctx.fill();
  ctx.strokeStyle='rgba(255,255,255,.75)';ctx.lineWidth=1.5;ctx.stroke();
  ctx.fillStyle='#fff';ctx.beginPath();ctx.arc(0,0,s*.28,0,7);ctx.fill();ctx.restore()}

  ctx.quadraticCurveTo(x,y+r*1.5+Math.sin(T*7+w.phase)*r*.2,x+r*.55,y+r*.3);ctx.closePath();ctx.fill();ctx.globalAlpha=1;
  const dxp=player.x-w.x,dyp=player.y-w.y,dd=Math.hypot(dxp,dyp)||1;
  const fxo=x+dxp/dd*r*.28,fyo=y+dyp/dd*r*.28;
  ctx.fillStyle='#26091a';ctx.beginPath();ctx.arc(fxo,fyo,r*.4,0,7);ctx.fill();

  for(const s of[-1,1]){const px=-dyp/dd,py=dxp/dd;ctx.beginPath();ctx.arc(fxo+px*r*.16*s,fyo+py*r*.16*s,r*.11,0,7);ctx.fill()}}}

  if(invulnT>0&&((T*14)|0)%2===0)ctx.globalAlpha=.35;

function drawParticles(){if(!particles.length)return;ctx.save();ctx.globalCompositeOperation='lighter';
  for(const p of particles){const a=1-p.t/p.tot;
    ctx.fillStyle=`rgba(${p.col[0]},${p.col[1]},${p.col[2]},${a*.85})`;

  for(const d of dust){const a=Math.sin(Math.PI*Math.min(1,d.t/d.tot));
    ctx.fillStyle=`rgba(255,214,150,${a*.4})`;ctx.beginPath();ctx.arc(d.x,d.y,d.sz,0,7);ctx.fill()}}
function drawFog(){const f=fogCtx;f.setTransform(DPR,0,0,DPR,0,0);
  f.globalCompositeOperation='source-over';f.clearRect(0,0,W,H);
  f.fillStyle='rgba(4,9,18,.965)';f.fillRect(0,0,W,H);
  const px=offX+player.x*cell,py=offY+player.y*cell;
  const Rr=cell*3.3*flick()*(sprinting?.8:1);
  let g=f.createRadialGradient(px,py,Rr*.1,px,py,Rr);
  g.addColorStop(0,'rgba(0,0,0,1)');g.addColorStop(.55,'rgba(0,0,0,.92)');g.addColorStop(1,'rgba(0,0,0,0)');
  f.fillStyle=g;f.beginPath();f.arc(px,py,Rr,0,7);f.fill();
