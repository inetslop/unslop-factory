var actx=null,master=null,wallNoise=null,rumbVal=0;
function ensureAudio(){
  if(actx){if(actx.state==='suspended')actx.resume();return;}
  var AC=window.AudioContext||window.webkitAudioContext;if(!AC)return;
  actx=new AC();
  master=actx.createGain();master.gain.value=muted?0:0.55;master.connect(actx.destination);
  // ambient pad
  var amb=actx.createGain();amb.gain.value=0.032;
  var lp=actx.createBiquadFilter();lp.type='lowpass';lp.frequency.value=320;
  amb.connect(lp);lp.connect(master);
  [54,81.6].forEach(function(f,i){var o=actx.createOscillator();o.type='sine';o.frequency.value=f;o.detune.value=i?8:-5;o.connect(amb);o.start();});
  var lfo=actx.createOscillator(),lg=actx.createGain();lfo.frequency.value=0.13;lg.gain.value=0.012;lfo.connect(lg);lg.connect(amb.gain);lfo.start();
  // wall rumble (looped noise)

  var f=actx.createBiquadFilter();f.type='lowpass';f.frequency.value=100;

function noiseBuf(sec){var len=Math.floor(actx.sampleRate*sec),b=actx.createBuffer(1,len,actx.sampleRate),d=b.getChannelData(0);for(var i=0;i<len;i++)d[i]=Math.random()*2-1;return b;}
function beep(type,f0,f1,dur,vol,delay){
  if(!actx)return;delay=delay||0;var t=actx.currentTime+delay;
  var o=actx.createOscillator(),g=actx.createGain();o.type=type;
  o.frequency.setValueAtTime(Math.max(1,f0),t);o.frequency.exponentialRampToValueAtTime(Math.max(1,f1),t+dur);
  g.gain.setValueAtTime(0.0001,t);g.gain.exponentialRampToValueAtTime(vol,t+0.012);g.gain.exponentialRampToValueAtTime(0.0001,t+dur);
  o.connect(g);g.connect(master);o.start(t);o.stop(t+dur+0.05);
}
function noiseHit(dur,freq,vol,delay){
  if(!actx)return;delay=delay||0;var t=actx.currentTime+delay;
  var s=actx.createBufferSource();s.buffer=noiseBuf(dur+0.1);
  var f=actx.createBiquadFilter();f.type='lowpass';f.frequency.value=freq;
  var g=actx.createGain();g.gain.setValueAtTime(vol,t);g.gain.exponentialRampToValueAtTime(0.0001,t+dur);
  s.connect(f);f.connect(g);g.connect(master);s.start(t);s.stop(t+dur+0.1);
}

/* ================= physics ================= */
function circleRectHit(b,rx,ry,rw,rh){
  var cx=clamp(b.x,rx,rx+rw),cy=clamp(b.y,ry,ry+rh),dx=b.x-cx,dy=b.y-cy,d2=dx*dx+dy*dy;
  if(d2>b.r*b.r)return null;
  var d=Math.sqrt(d2),nx,ny,pen;
  if(d>0.0001){nx=dx/d;ny=dy/d;pen=b.r-d;}
  else{var l=b.x-rx,r2=rx+rw-b.x,tp=b.y-ry,bo=ry+rh-b.y,m=Math.min(l,r2,tp,bo);
    if(m===l){nx=-1;ny=0;pen=b.r+l;}else if(m===r2){nx=1;ny=0;pen=b.r+r2;}else if(m===tp){nx=0;ny=-1;pen=b.r+tp;}else{nx=0;ny=1;pen=b.r+bo;}}
  return {nx:nx,ny:ny,pen:pen};
}
function resolve(b,hit,e){
  b.x+=hit.nx*hit.pen;b.y+=hit.ny*hit.pen;
  var vn=b.vx*hit.nx+b.vy*hit.ny;
  if(vn<0){b.vx-=(1+e)*vn*hit.nx;b.vy-=(1+e)*vn*hit.ny;return -vn;}
  return 0;
}
function pistonPose(p){
  var rest=p.y-64,down=p.y+58,mount=p.y-294;
  if(!p.armed)return{bottom:rest,ext:false,desc:false};
  var pr=(p.t%p.period)/p.period,bottom=rest,desc=false;
  if(pr<0.30)bottom=rest;
  else if(pr<0.52){var k=(pr-0.30)/0.22;k=k*k;bottom=rest+(down-rest)*k;desc=true;}
  else if(pr<0.60)bottom=down;
  else if(pr<0.85){var k2=(pr-0.60)/0.25;k2=1-(1-k2)*(1-k2);bottom=down-(down-rest)*k2;}
  else bottom=rest;
  return{bottom:bottom,ext:bottom>p.y-40,desc:desc,mount:mount};
}

function stepBody(h){
  var b=ball;
  b.coyote-=h;b.jbuf-=h;
  if(b.jbuf>0&&(b.grounded||b.coyote>0)){b.vy=-760;b.grounded=false;b.coyote=0;b.jbuf=0;sJump();puf(b.x,b.y+b.r,4,'rgba(150,175,200,',80,50,0,.4);}
  var g=2300;
  if(!b.grounded){if(input.jump&&b.vy<0)g*=0.45;if(input.down)g*=1.85;}
  b.vy+=g*h;if(b.vy>1500)b.vy=1500;
  var dir=(input.right?1:0)-(input.left?1:0);
  b.vx+=dir*(b.grounded?1500:820)*h;
  if(!dir&&b.grounded)b.vx-=b.vx*Math.min(1,3.0*h);
  b.vx-=b.vx*0.06*h;
  b.vx=clamp(b.vx,-940,940);
  b.x+=b.vx*h;b.y+=b.vy*h;b.ang+=(b.vx/b.r)*h*(b.grounded?1:0.55);

  var wasG=b.grounded;b.grounded=false;
  for(var i=0;i<plats.length;i++){var p=plats[i];if(p.x>b.x+220||p.x+p.w<b.x-220)continue;
    var hit=circleRectHit(b,p.x,p.y,p.w,Math.min(p.h,400));
    if(hit){var imp=resolve(b,hit,hit.ny<-0.7?0.12:0.4);
      if(hit.ny<-0.6){b.grounded=true;b.coyote=0.09;if(!wasG&&imp>320)landFX(imp);}}
  }
  for(var sI=0;sI<springs.length;sI++){var s=springs[sI];if(s.x>b.x+160||s.x+s.w<b.x-160)continue;
    var sh=circleRectHit(b,s.x,s.y-34,56,34);
    if(sh){
      if(sh.ny<-0.7&&b.vy>140){b.y=s.y-34-b.r;b.vy=-1150;s.anim=1;sSpring();sparks(b.x,s.y-30,8,'rgba(255,190,90,');}
      else{var imp2=resolve(b,sh,hit_e(sh));if(sh.ny<-0.6){b.grounded=true;b.coyote=0.09;if(!wasG&&imp2>320)landFX(imp2);}}
    }
  }
  function hit_e(sh){return sh.ny<-0.7?0.12:0.4;}

  for(var k2=0;k2<spikes.length;k2++){var sk=spikes[k2];if(sk.x>b.x+120||sk.x+sk.w<b.x-120)continue;
    if(circleRectHit(b,sk.x+sk.w*0.18,sk.y+sk.h*0.35,sk.w*0.64,sk.h*0.62)){die('spikes');return;}}
  for(var q=0;q<pistons.length;q++){var ps=pistons[q],pose=pistonPose(ps);if(!pose.ext)continue;
    if(circleRectHit(b,ps.x,pose.mount,ps.w,pose.bottom-pose.mount)){die('piston');return;}}
  for(var oI=0;oI<orbs.length;oI++){var o=orbs[oI];if(o.got||o.x>b.x+120||o.x<b.x-120)continue;
    var oy=o.y+Math.sin(T*3+o.ph)*5,dx=o.o?0:0,dxo=o.x-b.x,dyo=oy-b.y;
    if(dxo*dxo+dyo*dyo<(b.r+15)*(b.r+15)){o.got=true;orbsGot++;sOrb();sparks(o.x,oy,8,'rgba(255,200,110,');
      texts.push({x:o.x,y:oy-16,txt:'+25',life:.8,max:.8});}}

  if(wall.x>=b.x-b.r*0.6){die('wall');return;}
  if(b.y>1430){die('pits');return;}

  trailT-=h;
  var spd=Math.abs(b.vx)+Math.abs(b.vy)*0.5;
  if(trailT<=0&&spd>240){parts.push({t:'ghost',x:b.x,y:b.y,life:.3,max:.3,r:b.r*0.9});trailT=0.03;}
  if(b.grounded&&Math.abs(b.vx)>520&&Math.random()<.5)sparks(b.x+(Math.random()-.5)*10,b.y+b.r-2,1,'rgba(255,170,80,');
}

/* ================= update ================= */
function update(dt){
  if(state==='playing'){
    elapsed+=dt;
    genUntil(ball.x+VW*2);
    physAcc=Math.min(physAcc+dt,STEP*6);
    while(physAcc>=STEP){stepBody(STEP);physAcc-=STEP;if(state!=='playing')break;}
    if(state==='playing'){
      var gap=ball.x-wall.x;
      var sp=Math.min(380,60+elapsed*4.2);
      if(gap>760)sp+=(gap-760)*0.9;
      wall.x+=sp*dt;wall.t+=dt;
      distMax=Math.max(distMax,ball.x);
    }
  }else if(state==='dying'){
    dieT+=dt;
    ball.vy+=1400*dt;ball.x+=ball.vx*dt;ball.y+=ball.vy*dt;ball.ang+=(ball.vx/ball.r)*dt;
    if(dieT>1.0)finishDeath();
  }
  // hazards timers
  for(var i=0;i<pistons.length;i++){var p=pistons[i],c=p.x+p.w/2,d=Math.abs(ball.x-c);
    if(d<430&&!p.armed){p.armed=true;p.t=0;}else if(!p.armed&&d>620){p.armed=false;p.t=0;}
    if(p.armed)p.t+=dt;}
  for(var sI=0;sI<springs.length;sI++)springs[sI].anim=Math.max(0,springs[sI].anim-dt*5);

  // cleanup behind camera
  var cut=camX-900;
  if(plats.length>40)plats=plats.filter(function(p){return p.x+p.w>cut;});
  spikes=spikes.filter(function(s){return s.x+s.w>cut;});
  springs=springs.filter(function(s){return s.x+s.w>cut;});
  pistons=pistons.filter(function(p){return p.x+p.w>cut;});
  orbs=orbs.filter(function(o){return o.x>cut&&!o.got;});

  // particles / texts
  for(var k=parts.length-1;k>=0;k--){var pt=parts[k];pt.life-=dt;
    if(pt.life<=0){parts.splice(k,1);continue;}
    if(pt.t!=='ghost'){pt.vy=(pt.vy||0)+(pt.g||0)*dt;pt.x+=(pt.vx||0)*dt;pt.y+=pt.vy*dt;if(pt.av)pt.ang+=pt.av*dt;
      if(pt.t==='ring')pt.r+=260*dt;}}
  if(parts.length>340)parts.splice(0,parts.length-340);
  for(var tI=texts.length-1;tI>=0;tI--){var tx=texts[tI];tx.life-=dt;tx.y-=44*dt;if(tx.life<=0)texts.splice(tI,1);}

  // wall embers
  emberT-=dt;
  if(emberT<=0&&(state==='playing'||state==='menu')&&wall.x>camX-60){
    emberT=0.06;
    parts.push({t:'spark',x:wall.x+Math.random()*14,y:camY+Math.random()*VH,vx:-30-Math.random()*70,vy:-20-Math.random()*80,g:0,life:.5,max:.5,r:1.6,c:'rgba(120,255,236,'});
  }

  // camera / fx decay
  var lead=clamp((ball.vx-150)/900,0,1)*0.14;
  var tx=ball.x-VW*(0.32+lead);
  camX+=(tx-camX)*Math.min(1,dt*6);
  var ty=clamp(ball.y-VH*0.55,80,950);
  camY+=(ty-camY)*Math.min(1,dt*5);
  shake=Math.max(0,shake-dt*34);flash=Math.max(0,flash-dt*2.4);ball.sq=Math.max(0,ball.sq-dt*3.4);

  // motes drift
  var push=(state==='playing'?Math.abs(ball.vx)*0.25:0);
  for(var mI=0;mI<motes.length;mI++){var mo=motes[mI];mo.x-=(12+mo.z*26+push*mo.z)*dt;
    if(mo.x<-4){mo.x=W+4;mo.y=Math.random()*H;}}

  // wall rumble gain
  if(actx&&wallNoise){var gp=9999;if(state==='playing')gp=ball.x-wall.x;
    var tgt=Math.max(0,1-gp/620);tgt=tgt*tgt*0.3;
    rumbVal+=(tgt-rumbVal)*Math.min(1,dt*8);wallNoise.gain.value=rumbVal;}
}

function drawBall(){
  var b=ball;
  // contact shadow
  for(var i=0;i<plats.length;i++){var p=plats[i];if(p.x<=b.x&&p.x+p.w>=b.x&&p.y>=b.y-b.r&&p.y<b.y+260){
    var d=p.y-(b.y+b.r),sh=1-d/260;if(sh>0){
      ctx.fillStyle='rgba(0,0,0,'+(sh*0.35)+')';ctx.beginPath();
      ctx.ellipse(b.x,p.y+4,b.r*(1.5-sh*.7),b.r*0.32*(1.5-sh*.7),0,0,7);ctx.fill();break;}}
  }
  var sx=1+b.sq,sy=1-b.sq;
  ctx.save();ctx.translate(b.x,b.y);ctx.scale(sx,sy);
  var g=ctx.createRadialGradient(-b.r*0.35,-b.r*0.42,b.r*0.15,0,0,b.r*1.15);
  g.addColorStop(0,'#f2f8ff');g.addColorStop(.45,'#9fb4c6');g.addColorStop(.8,'#3c4f61');g.addColorStop(1,'#22313f');
  ctx.fillStyle=g;ctx.beginPath();ctx.arc(0,0,b.r,0,7);ctx.fill();
  // wall rim light (left)
  ctx.strokeStyle='rgba(65,240,223,.65)';ctx.lineWidth=3;
  ctx.beginPath();ctx.arc(0,0,b.r-1.5,Math.PI*0.55,Math.PI*1.45);ctx.stroke();
  // roll marker
  ctx.save();ctx.rotate(b.ang);
  ctx.fillStyle='#2b3a49';ctx.beginPath();ctx.arc(b.r*0.55,0,3.6,0,7);ctx.fill();
  ctx.fillStyle='rgba(255,179,71,.8)';ctx.beginPath();ctx.arc(-b.r*0.5,0,2.4,0,7);ctx.fill();
  ctx.restore();
  ctx.fillStyle='rgba(255,255,255,.85)';ctx.beginPath();ctx.arc(-5,-7,3,0,7);ctx.fill();
  ctx.restore();
}

function drawParts(front){
  for(var i=0;i<parts.length;i++){var p=parts[i];
    if(front===(p.t==='ghost'||p.t==='dust'))continue;
    var a=clamp(p.life/p.max,0,1);
    if(p.t==='ghost'){ctx.fillStyle='rgba(140,200,255,'+(a*0.2)+')';ctx.beginPath();ctx.arc(p.x,p.y,p.r*a+3,0,7);ctx.fill();}
    else if(p.t==='dust'){ctx.fillStyle=(p.c||'rgba(150,175,200,')+(a*0.4)+')';ctx.beginPath();ctx.arc(p.x,p.y,p.r*(2-a),0,7);ctx.fill();}
    else if(p.t==='spark'){ctx.strokeStyle=(p.c||'rgba(255,190,90,')+a+')';ctx.lineWidth=p.r;
      ctx.beginPath();ctx.moveTo(p.x,p.y);ctx.lineTo(p.x-(p.vx||0)*0.03,p.y-(p.vy||0)*0.03);ctx.stroke();}
    else if(p.t==='shard'){ctx.save();ctx.translate(p.x,p.y);ctx.rotate(p.ang||0);
      ctx.fillStyle='rgba(174,191,210,'+a+')';ctx.beginPath();ctx.moveTo(-6,-5);ctx.lineTo(7,-2);ctx.lineTo(-3,6);ctx.closePath();ctx.fill();ctx.restore();}
    else if(p.t==='ring'){ctx.strokeStyle='rgba(216,255,251,'+(a*0.8)+')';ctx.lineWidth=3;ctx.beginPath();ctx.arc(p.x,p.y,p.r,0,7);ctx.stroke();}
  }
}
function render(){
  ctx.setTransform(DPR,0,0,DPR,0,0);
  var sky=ctx.createLinearGradient(0,0,0,H);
  sky.addColorStop(0,'#0e1a28');sky.addColorStop(.55,'#0b131d');sky.addColorStop(1,'#171014');
  ctx.fillStyle=sky;ctx.fillRect(0,0,W,H);
  strata(0.16,0.22,96,52,'#111e2c',3);
  strata(0.34,0.40,72,66,'#0d1824',11);
  for(var i=0;i<motes.length;i++){var m=motes[i];ctx.fillStyle='rgba(159,196,224,'+(0.05+m.z*0.08)+')';
    ctx.beginPath();ctx.arc(m.x,m.y,0.7+m.z,0,7);ctx.fill();}

  var ox=(Math.random()*2-1)*shake,oy=(Math.random()*2-1)*shake;
  ctx.save();ctx.scale(SC,SC);ctx.translate(-camX+ox,-camY+oy);
  drawLava();drawPlatforms();drawSprings();drawPistons();drawSpikes();drawOrbs();

  drawWall();drawParts(true);

    ctx.fillStyle='rgba(255,210,122,'+clamp(tx.life/tx.max,0,1)+')';ctx.fillText(tx.txt,tx.x,tx.y);}
  ctx.restore();

  // speed streaks
  if(state==='playing'&&Math.abs(ball.vx)>640){

    for(var sI=0;sI<n;sI++){var yy=Math.random()*H,ln=40+Math.random()*120;

}

function pressJump(){
  if(state==='menu'){start();return;}
  if(state==='dead'){if(dieLock<=0)start();return;}
  if(state==='paused'){setState('playing');return;}
  if(state==='playing'){input.jump=true;ball.jbuf=0.12;}
}
window.addEventListener('keydown',function(e){
  var k=e.key;
  if(k===' '||k==='ArrowUp'||k==='ArrowDown'||k==='ArrowLeft'||k==='ArrowRight')e.preventDefault();
  ensureAudio();
  switch(k){
    case ' ':case 'ArrowUp':case 'w':case 'W':if(!e.repeat)pressJump();break;
    case 'ArrowLeft':case 'a':case 'A':input.left=true;break;
    case 'ArrowRight':case 'd':case 'D':input.right=true;break;
    case 'ArrowDown':case 's':case 'S':input.down=true;break;
    case 'p':case 'P':case 'Escape':
      if(state==='playing')setState('paused');else if(state==='paused')setState('playing');break;
    case 'r':case 'R':if(state==='playing'||state==='dead'||state==='paused')start();break;
    case 'm':case 'M':toggleMute();break;
  }
});
window.addEventListener('keyup',function(e){
  switch(e.key){
    case ' ':case 'ArrowUp':case 'w':case 'W':input.jump=false;break;
    case 'ArrowLeft':case 'a':case 'A':input.left=false;break;
    case 'ArrowRight':case 'd':case 'D':input.right=false;break;
    case 'ArrowDown':case 's':case 'S':input.down=false;break;
  }
});
