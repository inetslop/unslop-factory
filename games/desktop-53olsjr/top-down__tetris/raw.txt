  piece=p;gravT=0;lockT=0;resets=0;arrMode=false;dasT=0;stepT=0;
  renderPreviews();return p;
}
function ghostY(){let gy=piece.y;while(!collide(piece.type,piece.rot,piece.x,gy+1))gy++;return gy;}
const grounded=()=>piece&&collide(piece.type,piece.rot,piece.x,piece.y+1);
const levelMs=l=>Math.max(35,Math.floor(1000*Math.pow(0.8-(l-1)*0.007,l-1)));

function postShift(){if(grounded()&&resets<15){lockT=0;resets++;}}
function moveX(d){
  if(!piece||clearing)return false;
  if(!collide(piece.type,piece.rot,piece.x+d,piece.y)){piece.x+=d;sMove();postShift();return true;}
  return false;
}
function rotate(dir){
  if(!piece||clearing)return;
  const nr=(piece.rot+dir+4)%4,key=piece.rot+'>'+nr;
  const table=(piece.type==='I'?KICKS_I:KICKS_JLSTZ)[key];
  for(const[kx,ky]of table){
    if(!collide(piece.type,nr,piece.x+kx,piece.y+ky)){
      piece.x+=kx;piece.y+=ky;piece.rot=nr;sRot();postShift();return;
    }
  }
}
function hardDrop(){
  if(!piece||clearing)return;
  const gy=ghostY(),d=gy-piece.y,oldCells=cellsOf(piece.type,piece.rot,piece.x,piece.y);
  piece.y=gy;score+=2*d;updateHUD();
  dropFx={cells:oldCells.map(([x,y])=>({x,y0:y,y1:gy})),t:0,color:COLORS[piece.type]};
  shake(4,160);sHard();doLock(true);
}
function holdSwap(){
  if(!piece||clearing)return;
  if(!canHold){sDeny();return;}
  const cur=piece.type;
  if(holdType===cur){sDeny();return;}
  let nt=holdType;
  if(nt==null){refillQueue();nt=queue.shift();}
  holdType=cur;piece=makePiece(nt);canHold=false;
  gravT=0;lockT=0;resets=0;
  if(collide(piece.type,piece.rot,piece.x,piece.y)){gameOver();return;}
  sHold();renderPreviews();
}
function doLock(fromHard){
  const cells=cellsOf(piece.type,piece.rot,piece.x,piece.y);
  let allHidden=true;
  for(const[x,y]of cells){if(y>=0&&y<TOTAL)grid[y][x]=piece.type;if(y>=HIDDEN)allHidden=false;}
  lockFx={set:cells.map(([x,y])=>y*COLS+x),t:0};
  if(!fromHard)sLock();
  canHold=true;
  const full=[];
  for(let r=0;r<TOTAL;r++)if(grid[r].every(v=>v))full.push(r);
  piece=null;
  if(full.length){
    combo++;
    const base=[0,100,300,500,800][full.length];
    let pts=base*level;let b2bHit=false;
    if(full.length===4&&b2b){pts=Math.floor(pts*1.5);b2bHit=true;}
    b2b=(full.length===4);
    const cb=combo>1?(combo-1)*50*level:0;pts+=cb;
    score+=pts;lines+=full.length;
    const topY=PAD+(Math.min(...full)-HIDDEN)*CELL;
    popup('+'+pts,'',BW/2,topY);
    if(full.length===4){popup(b2bHit?'B2B TETRASTORM!':'TETRASTORM!','gold',BW/2,topY-34);flash('#ffcf45');}
    else if(combo>=2)popup('COMBO ×'+combo,'cyan',BW/2,topY+26);
    for(const r of full)for(let c=0;c<COLS;c++)burst(PAD+c*CELL+CELL/2,PAD+(r-HIDDEN)*CELL+CELL/2,COLORS[grid[r][c]]);
    sClear(full.length);
    const nl=Math.floor(lines/10)+1;
    if(nl>level){level=nl;popup('LEVEL '+level,'gold',BW/2,BH/2-40);flash('#3ee6d8');sLevel();}
    clearing=true;clearT=0;clearRows=full;
  }else{
    combo=0;updateHUD();
    if(!spawnNext()){gameOver();return;}
  }
  updateHUD();
}

/* ============================== UPDATE ============================== */
function update(dt){
  tGlobal+=dt;
  if(shakeT>0)shakeT=Math.max(0,shakeT-dt);else shakeMag=0;
  for(let i=particles.length-1;i>=0;i--){const p=particles[i];p.life+=dt;
    if(p.life>p.max){particles.splice(i,1);continue;}
    p.vy+=p.g*dt/1000;p.x+=p.vx*dt/1000;p.y+=p.vy*dt/1000;}
  if(dropFx){dropFx.t+=dt;if(dropFx.t>140)dropFx=null;}
  if(lockFx){lockFx.t+=dt;if(lockFx.t>130)lockFx=null;}

  if(state!=='playing')return;

  if(clearing){
    clearT+=dt;
    if(clearT>=CLEAR_MS){
      for(const r of clearRows){grid.splice(r,1);grid.unshift(Array(COLS).fill(0));}
      clearing=false;clearRows=[];updateHUD();
      if(!spawnNext()){gameOver();return;}
    }
    return;
  }
  if(!piece)return;

  // DAS auto-shift
  if(lastDir){
    if(!arrMode){dasT+=dt;if(dasT>=DAS){arrMode=true;stepT=0;moveX(lastDir);}}
    else{stepT+=dt;while(stepT>=ARR){if(moveX(lastDir))stepT-=ARR;else{stepT=0;break;}}}
  }
  // gravity
  const g=softHeld?SOFT_MS:levelMs(level);
  gravT+=dt;
  while(gravT>=g){
    gravT-=g;
    if(!collide(piece.type,piece.rot,piece.x,piece.y+1)){piece.y++;if(softHeld)score+=1;}
    else break;
  }
  if(grounded()){lockT+=dt;if(lockT>=LOCK_MS){doLock(false);return;}}
  else{lockT=0;resets=0;}
  if(softHeld)updateHUD();
}

function renderBoard(){
  const c=bctx;c.clearRect(0,0,BW,BH);
  c.save();
  if(shakeMag>0){const k=shakeT/shakeDur;c.translate((Math.random()-.5)*2*shakeMag*k,(Math.random()-.5)*2*shakeMag*k);}
  const g=c.createLinearGradient(0,0,0,BH);g.addColorStop(0,'#0e1c2a');g.addColorStop(1,'#0a141d');
  c.fillStyle=g;c.fillRect(0,0,BW,BH);
  c.strokeStyle='rgba(130,180,220,.055)';c.lineWidth=1;c.beginPath();
  for(let i=0;i<=COLS;i++){c.moveTo(PAD+i*CELL+.5,PAD);c.lineTo(PAD+i*CELL+.5,BH-PAD);}
  for(let i=0;i<=VISIBLE;i++){c.moveTo(PAD,PAD+i*CELL+.5);c.lineTo(BW-PAD,PAD+i*CELL+.5);}
  c.stroke();

  // danger pulse
  let topRow=TOTAL;for(let r=0;r<TOTAL;r++)if(grid[r].some(v=>v)){topRow=r;break;}
  if(topRow<=HIDDEN+3){
    const a=.12+.1*Math.sin(tGlobal/130);
    const dg=c.createLinearGradient(0,PAD,0,PAD+64);
    dg.addColorStop(0,`rgba(255,80,80,${a})`);dg.addColorStop(1,'rgba(255,80,80,0)');
    c.fillStyle=dg;c.fillRect(PAD,PAD,COLS*CELL,64);
  }

  // locked cells
  const lf=lockFx?new Set(lockFx.set):null;
  for(let r=HIDDEN;r<TOTAL;r++)for(let x=0;x<COLS;x++){
    const t=grid[r][x];if(!t)continue;
    drawBlock(c,PAD+x*CELL,PAD+(r-HIDDEN)*CELL,CELL,COLORS[t]);
    if(lf&&lf.has(r*COLS+x)){c.fillStyle=`rgba(255,255,255,${.5*(1-lockFx.t/130)})`;c.fillRect(PAD+x*CELL+1,PAD+(r-HIDDEN)*CELL+1,CELL-2,CELL-2);}
  }

  // hard-drop streaks
  if(dropFx){const a=1-dropFx.t/140;
    c.fillStyle=dropFx.color;c.globalAlpha=.3*a;
    for(const s of dropFx.cells){const y0=PAD+(Math.min(s.y0,s.y1)-HIDDEN)*CELL,y1=PAD+(Math.max(s.y0,s.y1)-HIDDEN)*CELL+CELL;
      c.fillRect(PAD+s.x*CELL+7,y0,CELL-14,Math.max(2,y1-y0));}
    c.globalAlpha=1;}

  // active piece + ghost
  if(piece){
    const gy=ghostY();
    for(const[x,y]of cellsOf(piece.type,piece.rot,piece.x,gy)){
      if(y<HIDDEN)continue;
      const px=PAD+x*CELL,py=PAD+(y-HIDDEN)*CELL;
      c.fillStyle='rgba(255,255,255,.05)';c.fillRect(px+3,py+3,CELL-6,CELL-6);
      c.strokeStyle=COLORS[piece.type];c.globalAlpha=.5;c.lineWidth=2;
      c.strokeRect(px+3.5,py+3.5,CELL-7,CELL-7);c.globalAlpha=1;
    }
    c.save();c.shadowColor=COLORS[piece.type];c.shadowBlur=14;
    for(const[x,y]of cellsOf(piece.type,piece.rot,piece.x,piece.y)){
      if(y<HIDDEN)continue;
      drawBlock(c,PAD+x*CELL,PAD+(y-HIDDEN)*CELL,CELL,COLORS[piece.type]);
    }
    c.restore();
  }

  // clearing flash
  if(clearing){
    const p=clearT/CLEAR_MS,a=Math.sin(p*Math.PI);
    for(const r of clearRows){
      const y=PAD+(r-HIDDEN)*CELL;
      c.fillStyle=`rgba(255,255,255,${.85*a})`;c.fillRect(PAD,y,COLS*CELL,CELL);
      c.fillStyle=`rgba(255,207,69,${.5*a})`;c.fillRect(PAD,y+CELL/2-1,COLS*CELL,2);
    }
  }

  // particles
  for(const p of particles){
    c.globalAlpha=Math.max(0,1-p.life/p.max);c.fillStyle=p.color;
    c.fillRect(p.x-p.size/2,p.y-p.size/2,p.size,p.size);
  }
  c.globalAlpha=1;c.restore();
}
