LOS:

```js
function los(x0,y0,x1,y1){
  let dx=Math.abs(x1-x0), dy=Math.abs(y1-y0), sx=x0<x1?1:-1, sy=y0<y1?1:-1, err=dx-dy;
  let x=x0,y=y0;
  while(!(x===x1&&y===y1)){
    const e2=2*err;
    if(e2>-dy){err-=dy;x+=sx;}
    if(e2<dx){err+=dx;y+=sy;}
    if(x===x1&&y===y1) break;
    if(map[y][x]===1) return false;
  }
  return true;
}
```

```js
function stepToward(e,tx,ty,prefer){
  const dx=Math.sign(tx-e.x), dy=Math.sign(ty-e.y);
  const order = Math.abs(tx-e.x)>=Math.abs(ty-e.y)?[[dx,0],[0,dy]]:[[0,dy],[dx,0]];
```

```html
function wander(e){
  const dirs=[[1,0],[-1,0],[0,1],[0,-1]];
  for(let i=0;i<4;i++){const d=dirs[R(0,3)];if(canOccupy(e.x+d[0],e.y+d[1],e)){e.x+=d[0];e.y+=d[1];return;}}
}
function enemiesTurn(){
  if(state!=='play')return;
  for(const e of enemies){
    if(e.hp<=0)continue;
    const dist=Math.abs(e.x-player.x)+Math.abs(e.y-player.y);
    if(e.type==='boss'){
      if(dist===1)enemyHitPlayer(e);
      else if(turn%3===0&&dist<=6&&los(e.x,e.y,player.x,player.y))bossSpit(e);
      else stepToward(e,player.x,player.y);
    }else{
      if(dist===1)enemyHitPlayer(e);
      else if(dist<=6&&los(e.x,e.y,player.x,player.y))stepToward(e,player.x,player.y);
      else if(Math.random()<0.3)wander(e);
    }
    if(player.hp<=0)break;
  }
}

/* ============ combat ============ */
function gate(){const n=performance.now();if(n-lastAct<90)return false;lastAct=n;return true;}
function attackEnemy(e){
  player.lunge=1;player.lx=player.dx*0.3;player.ly=player.dy*0.3;
  let dmg=4+player.lvl*2+R(0,3);const crit=Math.random()<0.15;if(crit)dmg*=2;
  e.hp-=dmg;e.flash=1;e.lunge=1;e.lx=-player.dx*0.28;e.ly=-player.dy*0.28;
  burst(e.x,e.y,crit?'#ffd27a':'#ffe9c9',crit?16:8,crit?3:2,false);
  floatText(e.x,e.y-0.2,(crit?'CRIT ':'')+dmg,crit?'#ffd27a':'#fff3df',crit);
  if(crit){SFX.crit();shake=Math.max(shake,4);}else SFX.hit();
  log('You strike the '+ENAME[e.type]+' for <b>'+dmg+'</b>'+(crit?' — critical!':'.'),'dmg');
  if(e.hp<=0)killEnemy(e);
}
function killEnemy(e){
  kills++;SFX.die();
  burst(e.x,e.y,e.type==='boss'?'#9dff7a':'#8fd6a0',e.type==='boss'?48:14,e.type==='boss'?3.4:2,true);
  if(e.type==='boss'){log('<b>The Gloom Matriarch unravels into drifting ash.</b> The seal shudders.','magic');beginWin();return;}
  floatText(e.x,e.y-0.5,'+'+e.xp+' xp','#8ce8a0');
  const g=R(2,6);player.gold+=g;floatText(e.x,e.y-1.0,'+'+g+'g','#ffd45e');
  log('The '+ENAME[e.type]+' dissolves. <b>'+g+'g</b> spills free.','loot');
  if(Math.random()<0.3)pickups.push({x:e.x,y:e.y,type:Math.random()<0.6?'flask':'shard'});
  gainXp(e.xp);
}
function gainXp(n){
  player.xp+=n;
  while(player.xp>=player.xpNeed){
    player.xp-=player.xpNeed;player.lvl++;player.xpNeed=18+(player.lvl-1)*14;
    player.maxhp+=6;player.hp=Math.min(player.maxhp,player.hp+8);
    player.maxmp+=3;player.mp=Math.min(player.maxmp,player.mp+4);
    SFX.levelup();burst(player.x,player.y,'#7fbf6a',20,2.4,true);
    floatText(player.x,player.y-0.7,'LEVEL '+player.lvl+'!','#b8ff9e',true);
    log('<b>Level up!</b> The ember grows in your lantern. (Lv '+player.lvl+')','magic');
  }
}
function enemyHitPlayer(e){
  const d=R(e.dmg[0],e.dmg[1]);player.hp-=d;
  e.lunge=1;e.lx=Math.sign(player.x-e.x)*0.35;e.ly=Math.sign(player.y-e.y)*0.35;
  SFX.hurt();shake=Math.max(shake,5);hurtT=1;
  burst(player.x,player.y,'#ff8d7a',8,2,false);
  floatText(player.x,player.y-0.4,'-'+d,'#ff8d7a');
  log('The '+ENAME[e.type]+' hits you for <b>'+d+'</b>.','hurt');
}
function bossSpit(e){
  const d=R(8,10);player.hp-=d;
  boltAnim={x0:e.x+0.5,y0:e.y+0.3,x1:player.x+0.5,y1:player.y+0.4,t:0,dur:0.28,col:'#9dff7a',drop:true};
  SFX.spit();shake=Math.max(shake,6);hurtT=1;
  floatText(player.x,player.y-0.4,'-'+d,'#c6ff8a');
  log('The Matriarch spits <b>gloom</b> across the dark — you take <b>'+d+'</b>.','hurt');
}

/* ============ player actions ============ */
function onStairs(){return !!(stairs&&player.x===stairs.x&&player.y===stairs.y);}
function tryMove(mx,my){
  if(state!=='play'||!gate())return;
  player.dx=mx;player.dy=my;
  const nx=player.x+mx,ny=player.y+my;
  const foe=enemies.find(e=>e.hp>0&&e.x===nx&&e.y===ny);
  if(foe){attackEnemy(foe);endTurn();return;}
  if(map[ny][nx]!==0){SFX.bump();shake=Math.max(shake,1.5);return;}
  player.x=nx;player.y=ny;SFX.move();
  collectAt(nx,ny);
  if(onStairs()&&!stairMsgDone){stairMsgDone=true;log('A stairway spirals into the dark. <kbd>Enter</kbd> to descend · <kbd>G</kbd> to barter a flask (15g).','sys');}
  endTurn();
}
function fireBolt(){
  if(state!=='play'||!gate())return;
  if(player.mp<3){SFX.deny();log('Not enough ember for a bolt.','sys');return;}
  player.mp-=3;
  const dx=player.dx,dy=player.dy;let hx=player.x+0.5+dx*0.6,hy=player.y+0.45+dy*0.6,hit=null;
  for(let i=1;i<=4;i++){
    const x=player.x+dx*i,y=player.y+dy*i;
    if(x<0||y<0||x>=COLS||y>=ROWS)break;
    if(map[y][x]===1){hx=x+0.5-dx*0.2;hy=y+0.5-dy*0.2;break;}
    const f=enemies.find(o=>o.hp>0&&o.x===x&&o.y===y);
    if(f){hit=f;hx=x+0.5;hy=y+0.5;break;}
  }
  boltAnim={x0:player.x+0.5,y0:player.y+0.45,x1:hx,y1:hy,t:0,dur:0.2,col:'#ffcf8a',drop:false};
  burst(player.x,player.y,'#ffcf8a',5,1.2,false);SFX.bolt();
  if(hit){
    const dmg=7+player.lvl*2+R(0,3);hit.hp-=dmg;hit.flash=1;
    burst(hit.x,hit.y,'#ffcf8a',14,2.6,false);
    floatText(hit.x,hit.y-0.3,dmg,'#ffe9b0',true);SFX.hit();
    log('Your ember bolt sears the '+ENAME[hit.type]+' for <b>'+dmg+'</b>.','magic');
    if(hit.hp<=0)killEnemy(hit);
  }else burst(hx-0.5,hy-0.5,'#8a9a92',6,1.3,false);
  endTurn();
}
function useFlask(){
  if(state!=='play'||!gate())return;
  if(player.flasks<=0){SFX.deny();log('No flasks left in your pack.','sys');return;}
  if(player.hp>=player.maxhp){log('You are already at full health.','sys');return;}
  const heal=12+player.lvl*2;player.hp=Math.min(player.maxhp,player.hp+heal);player.flasks--;
  SFX.flask();burst(player.x,player.y,'#8ce8a0',14,1.6,false);
  floatText(player.x,player.y-0.5,'+'+heal,'#8ce8a0');
  log('You drink a flask. <b>+'+heal+'</b> vitality.','heal');endTurn();
}
function buyFlask(){
  if(state!=='play')return;
  if(!onStairs())return;
  if(player.gold<15){SFX.deny();log('The stairway-echo wants 15g. You carry '+player.gold+'g.','sys');return;}
  player.gold-=15;player.flasks++;SFX.pickup();
  floatText(player.x,player.y-0.6,'+1 flask','#9fe8ff');
  log('You barter <b>15g</b> for a flask from the stairway-echo.','loot');updateHUD();
}
function descend(){
  if(state!=='play'||!onStairs())return;
  floorN++;genFloor(floorN);descF=1;SFX.stairs();
  showBanner('FLOOR '+ROM[floorN],FLOOR_NAMES[floorN]);
  log('You descend to <b>Floor '+ROM[floorN]+' — '+FLOOR_NAMES[floorN]+'</b>.','sys');
  if(floorN===3)setTimeout(()=>{if(state==='play'){SFX.roar();shake=9;log('<b>The Gloom Matriarch</b> unfolds from the dark. Break her to break the seal.','magic');}},600);
  updateHUD();
}
function collectAt(x,y){
  for(let i=pickups.length-1;i>=0;i--){
    const p=pickups[i];if(p.x!==x||p.y!==y)continue;pickups.splice(i,1);
    if(p.type==='flask'){player.flasks++;SFX.pickup();floatText(x,y-0.4,'+1 flask','#9fe8ff');burst(x,y,'#9fe8ff',8,1.5,false);log('You pick up a <b>vitality flask</b>.','loot');}
    else if(p.type==='shard'){player.mp=Math.min(player.maxmp,player.mp+4);SFX.pickup();floatText(x,y-0.4,'+4 MP','#7fd6e8');burst(x,y,'#7fd6e8',10,1.8,false);log('An <b>ember shard</b> hums against your palm. +4 MP.','loot');}
    else{player.gold+=p.amt;SFX.pickup();floatText(x,y-0.4,'+'+p.amt+'g','#ffd45e');burst(x,y,'#ffd45e',10,1.6,true);log('You gather <b>'+p.amt+'g</b> of old gold.','loot');}
    updateHUD();
  }
}
function endTurn(){turn++;enemiesTurn();if(player.hp<=0)gameOver();updateHUD();}

function startGame(){
  player=freshPlayer();floorN=1;turn=0;kills=0;startTime=Date.now();
  particles.length=0;floats.length=0;shake=0;hurtT=0;descF=0;boltAnim=null;
  genFloor(1);state='play';hideOvAll();
  $('#logLines').innerHTML='';
  log('You light the lantern and step into <b>Floor I — '+FLOOR_NAMES[1]+'</b>.','sys');
  log('Bump into foes to strike. Find the stairway below.','sys');
  showBanner('FLOOR I',FLOOR_NAMES[1]);updateHUD();SFX.ui();
}
function gameOver(){
  if(state!=='play')return;state='over';
  best.deepest=Math.max(best.deepest,floorN);saveBest();
  SFX.over();shake=10;hurtT=1.2;burst(player.x,player.y,'#ff8d7a',26,2.5,true);
  setTimeout(()=>{if(state!=='over')return;fillEnd(false);showOv(ovEnd);},900);
}
function beginWin(){
  if(state!=='play')return;state='won';
  lastNewRec=!best.bestTurns||turn<best.bestTurns;
  if(lastNewRec)best.bestTurns=turn;
  best.wins++;best.deepest=Math.max(best.deepest,3);saveBest();shake=8;
  setTimeout(()=>{if(state!=='won')return;SFX.win();fillEnd(true);showOv(ovEnd);},1000);
}
function fillEnd(win){
  $('#endCard').classList.toggle('win',win);
  $('#endOver').textContent=win?'THE SEAL IS BROKEN':'THE DESCENT ENDS';
  $('#endTitle').textContent=win?'Dawn Breaks Above':'The Depths Claim You';
  $('#endSub').textContent=win?'The Matriarch is ash, and cold light falls through three floors of stone.':'Your lantern gutters out on Floor '+ROM[floorN]+'.';
  const t=Math.max(1,Math.round((Date.now()-startTime)/1000));
  const fmt=m=>Math.floor(m/60)+':'+String(m%60).padStart(2,'0');
  const st=(l,v)=>'<div class="stat"><b>'+v+'</b><small>'+l+'</small></div>';
  $('#statGrid').innerHTML=st('Floor',ROM[floorN])+st('Turns',turn)+st('Foes Felled',kills)+st('Gold',player.gold)+st('Level',player.lvl)+st('Time',fmt(t));
  $('#endBest').textContent=win
    ?(lastNewRec?'✦ New fastest descent — '+turn+' turns! ✦':'Fastest: '+best.bestTurns+' turns · Victories: '+best.wins)
    :'Deepest reached: Floor '+ROM[best.deepest]+' · Victories so far: '+best.wins;
}

/* ============ particles / floats ============ */
function burst(tx,ty,col,n,sp,size,grav){
  for(let i=0;i<n;i++){const a=Math.random()*6.283,v=(0.3+Math.random()*0.7)*sp;
    particles.push({x:tx*T+T/2,y:ty*T+T/2,vx:Math.cos(a)*v*T,vy:Math.sin(a)*v*T-(grav?sp*T*0.12:0),life:0.4+Math.random()*0.35,max:0.75,col,size:size*T*0.05*(0.6+Math.random()*0.8),g:grav?260:0});}
  if(particles.length>520)particles.splice(0,particles.length-520);
}
function floatText(tx,ty,txt,col,big){floats.push({x:tx,y:ty,txt,col,t:0,dur:big?1.3:1.0,big});}

  const r=stage.getBoundingClientRect();
  dpr=Math.min(2,window.devicePixelRatio||1);
  T=Math.max(12,Math.floor(Math.min((r.width-8)/COLS,(r.height-8)/ROWS)));
  W=COLS*T;H=ROWS*T;
  cv.style.width=W+'px';cv.style.height=H+'px';
  cv.width=W*dpr;cv.height=H*dpr;ctx.setTransform(dpr,0,0,dpr,0,0);
  lc.width=cv.width;lc.height=cv.height;lctx.setTransform(dpr,0,0,dpr,0,0);
}

const FLOOR_COLS=[[22,36,30],[26,41,34],[24,39,32],[30,45,37]];
const WALL_TOPS=[[35,58,50],[31,52,45],[38,62,53]];

function ell(g,x,y,rx,ry,col){g.fillStyle=col;g.beginPath();g.ellipse(x,y,rx,ry,0,0,6.283);g.fill();}
function rrect(g,x,y,w,h,r,col){g.fillStyle=col;g.beginPath();if(g.roundRect)g.roundRect(x,y,w,h,r);else g.rect(x,y,w,h);g.fill();}

function drawStairs(g,X,Y,t){
  g.fillStyle='#050d0a';g.fillRect(X+3,Y+3,T-6,T-6);
  const pulse=0.5+0.5*Math.sin(t*2.4);
  for(let i=0;i<4;i++){const r=(i+1)*T*0.11+pulse*(i===3?1.5:0.5);
    g.strokeStyle='rgba('+(i%2?'90,220,200':'60,180,170')+','+(0.5-i*0.09)+')';g.lineWidth=1.5;g.strokeRect(X+r,Y+r,T-2*r,T-2*r);}
  g.fillStyle='rgba(140,255,235,'+(0.25+0.2*pulse)+')';g.beginPath();g.arc(X+T/2,Y+T/2,T*0.07,0,6.283);g.fill();
}
function drawTile(g,x,y,t){
  const X=x*T,Y=y*T,v=map[y][x];
  if(v===0){
    const c=FLOOR_COLS[shadeMap?shadeMap[y][x]:0];
    g.fillStyle='rgb('+c[0]+','+c[1]+','+c[2]+')';g.fillRect(X,Y,T,T);
    g.strokeStyle='rgba(0,0,0,0.25)';g.lineWidth=1;g.strokeRect(X+0.5,Y+0.5,T-1,T-1);
    const h=hash2(x,y,floorN*31);
    if(h<0.2){g.fillStyle='rgba(98,150,84,0.4)';const mx=X+T*(0.2+h*2),my=Y+T*(0.3+(h*7%1)*0.4);g.beginPath();g.arc(mx,my,T*0.07,0,6.283);g.arc(mx+T*0.12,my+T*0.08,T*0.05,0,6.283);g.fill();}
    else if(h>0.8){g.fillStyle='rgba(190,200,190,0.1)';g.fillRect(X+T*(h*10%1),Y+T*((h*17)%1)*0.9,T*0.06,T*0.06);}
  }else if(v===2){
    g.fillStyle='#0e2b32';g.fillRect(X,Y,T,T);
    g.strokeStyle='rgba(0,0,0,0.3)';g.lineWidth=1;g.strokeRect(X+0.5,Y+0.5,T-1,T-1);
    const w1=Math.sin(t*1.7+x*1.1+y*0.7),w2=Math.sin(t*2.3-y*1.3+x*0.4);
    g.strokeStyle='rgba(120,215,225,0.16)';g.lineWidth=1.5;
    g.beginPath();g.moveTo(X+T*0.15,Y+T*(0.38+w1*0.06));g.lineTo(X+T*0.85,Y+T*(0.34+w2*0.06));g.stroke();
    g.beginPath();g.moveTo(X+T*0.2,Y+T*(0.68+w2*0.05));g.lineTo(X+T*0.8,Y+T*(0.72+w1*0.05));g.stroke();
    if(hash2(x,y,77)<0.3){const a=Math.max(0,0.3+0.3*Math.sin(t*3+x*5+y*3));g.fillStyle='rgba(190,240,245,'+a+')';g.fillRect(X+T*(hash2(x,y,5)*0.7+0.1),Y+T*(hash2(x,y,6)*0.6+0.2),2,2);}
  }else{
    const c=WALL_TOPS[shadeMap?shadeMap[y][x]:0];
    g.fillStyle='#0a1310';g.fillRect(X,Y,T,T);
    g.fillStyle='rgb('+c[0]+','+c[1]+','+c[2]+')';g.fillRect(X+2,Y+2,T-4,T-9);
    g.fillStyle='rgba(120,170,150,0.18)';g.fillRect(X+2,Y+2,T-4,2);
    if(y<ROWS-1&&map[y+1][x]!==1){g.fillStyle='#071009';g.fillRect(X+2,Y+T-7,T-4,5);}
    const h=hash2(x,y,floorN*13);
    if(h<0.25){g.fillStyle='rgba(80,130,70,'+(0.15+h)+')';g.fillRect(X+4+h*T*0.5,Y+T-9,T*0.16,3);}
  }
  if(stairs&&stairs.x===x&&stairs.y===y)drawStairs(g,X,Y,t);
}

function drawPickup(g,p,t){
  const x=p.x*T+T/2,y=p.y*T+T/2,bob=Math.sin(t*3+p.x)*T*0.05;
  ell(g,x,y+T*0.28,T*0.16,T*0.06,'rgba(0,0,0,0.3)');
  if(p.type==='flask'){
    const fy=y-T*0.05+bob;
    rrect(g,x-T*0.09,fy-T*0.02,T*0.18,T*0.2,T*0.04,'#2c4a52');
    rrect(g,x-T*0.065,fy-T*0.005,T*0.13,T*0.15,T*0.03,'#ff9a4d');
    rrect(g,x-T*0.03,fy-T*0.12,T*0.06,T*0.11,T*0.02,'#5c7a84');
  }else if(p.type==='shard'){
    g.save();g.translate(x,y-T*0.08+bob);g.rotate(t*1.2+p.y);
    const s=T*0.16*(1+0.08*Math.sin(t*4));
    g.fillStyle='#6fe3ea';g.beginPath();g.moveTo(0,-s);g.lineTo(s*0.6,0);g.lineTo(0,s);g.lineTo(-s*0.6,0);g.closePath();g.fill();
    g.fillStyle='rgba(255,255,255,0.7)';g.beginPath();g.moveTo(0,-s*0.7);g.lineTo(s*0.25,0);g.lineTo(0,s*0.3);g.closePath();g.fill();
    g.restore();
  }else{
    const fy=y+T*0.1;
    ell(g,x-T*0.08,fy,T*0.09,T*0.07,'#c98a2e');ell(g,x+T*0.07,fy+T*0.02,T*0.09,T*0.07,'#e0a53f');ell(g,x,fy-T*0.05,T*0.09,T*0.07,'#ffc857');
    const a=Math.max(0,Math.sin(t*4+p.x*2));g.fillStyle='rgba(255,240,190,'+a*0.8+')';g.fillRect(x-T*0.03,fy-T*0.09,T*0.05,T*0.05);
  }
}

function drawHero(g,cx,cy,S,t,p){

  rrect(g,cx-S*0.16,by-S*0.02,S*0.32,S*0.07,S*0.03,'#c98a3e');

function drawSlime(g,cx,cy,S,t,e){
  const w=Math.sin(t*4+e.wob),rx=S*0.32*(1+0.07*w),ry=S*0.24*(1-0.09*w);

  const by=cy+S*0.30-ry;
  ell(g,cx,by,rx,ry,'#4f9152');
  ell(g,cx+rx*0.1,by+ry*0.25,rx*0.6,ry*0.5,'rgba(124,191,116,0.7)');
  g.fillStyle='#16301a';g.beginPath();g.arc(cx-rx*0.35,by-ry*0.1,S*0.035,0,6.283);g.arc(cx+rx*0.35,by-ry*0.1,S*0.035,0,6.283);g.fill();

function drawWisp(g,cx,cy,S,t,e){

  if(e.flash>0){g.globalAlpha=e.flash*0.8;ell(g,cx,fy,S*0.16,S*0.18,'#fff');g.globalAlpha=1;}

function drawReaver(g,cx,cy,S,t,e){

function drawBoss(g,cx,cy,S,t,e){
  const bT=S*1.32;
  g.strokeStyle='#141f1a';g.lineWidth=bT*0.07;g.lineCap='round';
  for(const side of[-1,1])for(let i=0;i<4;i++){
    const hipX=cx+side*bT*0.30,hipY=cy-bT*0.05+(i-1.5)*bT*0.16;
    const kneeX=cx+side*bT*(0.58+i*0.06),kneeY=hipY-bT*0.28+Math.sin(t*7+i+e.wob)*bT*0.05;
    const footX=cx+side*bT*(0.82+(i%2)*0.10),footY=cy+bT*0.40+Math.cos(t*5+i*2+e.wob)*bT*0.04;
    g.beginPath();g.moveTo(hipX,hipY);g.lineTo(kneeX,kneeY);g.lineTo(footX,footY);g.stroke();

  g.strokeStyle='#0e1613';g.lineWidth=3;g.beginPath();g.ellipse(cx,cy-bT*0.02,bT*0.42,bT*0.38,0,0,6.283);g.stroke();

  ell(g,cx,cy+bT*0.02,bT*0.13*cp,bT*0.13*cp,'#7dff9a');
  g.fillStyle='#ffb454';g.beginPath();g.arc(cx-bT*0.16,cy-bT*0.24,bT*0.035,0,6.283);g.arc(cx+bT*0.16,cy-bT*0.24,bT*0.035,0,6.283);g.fill();
  g.fillStyle='#ff5d5d';g.beginPath();g.arc(cx,cy-bT*0.28,bT*0.04,0,6.283);g.fill();
  if(e.flash>0){g.globalAlpha=e.flash*0.8;ell(g,cx,cy-bT*0.02,bT*0.44,bT*0.4,'#fff');g.globalAlpha=1;}

function drawEnemy(g,e,t){
  const cx=(e.px+e.lx*e.lunge)*T+T/2,cy=(e.py+e.ly*e.lunge)*T+T/2;
  if(e.type==='slime')drawSlime(g,cx,cy,T,t,e);
  else if(e.type==='wisp')drawWisp(g,cx,cy,T,t,e);
  else if(e.type==='reaver')drawReaver(g,cx,cy,T,t,e);
  else drawBoss(g,cx,cy,T,t,e);
  if(e.hp<e.maxhp){
    const w=e.type==='boss'?T*1.1:T*0.62,bx=cx-w/2,by=cy-(e.type==='boss'?T*0.95:T*0.62);
    g.fillStyle='rgba(5,10,8,.8)';g.fillRect(bx-1,by-1,w+2,4);
    g.fillStyle='#ff6b52';g.fillRect(bx,by,w*clamp(e.hp/e.maxhp,0,1),2);
  }
}
function drawBolt(g,b){
  const p=Math.min(1,b.t/b.dur);
  g.globalCompositeOperation='lighter';
  for(let i=4;i>=0;i--){const q=p-i*0.06;if(q<=0)continue;
    const x=lerp(b.x0,b.x1,q)*T,y=lerp(b.y0,b.y1,q)*T;
    g.globalAlpha=(1-i*0.17)*(1-p*0.3);g.fillStyle=b.col;
    g.beginPath();g.arc(x,y,T*(0.18-i*0.025),0,6.283);g.fill();}
  g.globalAlpha=1;g.globalCompositeOperation='source-over';
}
function punch(x,y,r,a){
  const gr=lctx.createRadialGradient(x,y,r*0.1,x,y,r);
  gr.addColorStop(0,'rgba(0,0,0,'+a+')');gr.addColorStop(0.55,'rgba(0,0,0,'+(a*0.8)+')');gr.addColorStop(1,'rgba(0,0,0,0)');
  lctx.fillStyle=gr;lctx.beginPath();lctx.arc(x,y,r,0,6.283);lctx.fill();
}
function warm(x,y,r,rgb,a){

}
function drawLighting(g,t){

  for(const p of pickups){if(p.type==='gold')continue;punch((p.x+0.5)*T,(p.y+0.5)*T,T*1.1,0.7);}
  for(const e of enemies)if(e.hp>0&&(e.type==='boss'||e.type==='reaver'))punch((e.px+0.5)*T,(e.py+0.5)*T,e.type==='boss'?T*2.4:T*1.3,e.type==='boss'?0.9:0.6);
  if(boltAnim){const p=Math.min(1,boltAnim.t/boltAnim.dur);punch(lerp(boltAnim.x0,boltAnim.x1,p)*T,lerp(boltAnim.y0,boltAnim.y1,p)*T,T*1.4,0.9);}
  g.drawImage(lc,0,0,W,H);
}
function render(){

  const t=tGlobal;
  for(let y=0;y<ROWS;y++)for(let x=0;x<COLS;x++)drawTile(g,x,y,t);
  for(const p of pickups)drawPickup(g,p,t);
  const ents=enemies.filter(e=>e.hp>0).sort((a,b)=>a.py-b.py);
  for(const e of ents)drawEnemy(g,e,t);

  g.restore();
}

let lastT=performance.now();
function frame(now){
  const dt=Math.min(0.05,(now-lastT)/1000);lastT=now;
  tGlobal+=dt;
  const k=Math.min(1,dt*13);
  player.px+=(player.x-player.px)*k;player.py+=(player.y-player.py)*k;
  player.lunge=Math.max(0,player.lunge-dt*3.2);
  for(const e of enemies){e.px+=(e.x-e.px)*k;e.py+=(e.y-e.py)*k;e.flash=Math.max(0,e.flash-dt*5);e.lunge=Math.max(0,e.lunge-dt*3.2);}
  shake=Math.max(0,shake-dt*26);hurtT=Math.max(0,hurtT-dt*1.6);descF=Math.max(0,descF-dt*2.4);
  for(let i=particles.length-1;i>=0;i--){const p=particles[i];p.life-=dt;if(p.life<=0){particles.splice(i,1);continue;}p.x+=p.vx*dt;p.y+=p.vy*dt;p.vy+=(p.g||0)*dt;}
  for(let i=floats.length-1;i>=0;i--){floats[i].t+=dt;if(floats[i].t>floats[i].dur)floats.splice(i,1);}
  if(boltAnim){
    boltAnim.t+=dt;
    if(boltAnim.t<boltAnim.dur){
      const p=Math.min(1,boltAnim.t/boltAnim.dur),hx=lerp(boltAnim.x0,boltAnim.x1,p),hy=lerp(boltAnim.y0,boltAnim.y1,p);
      for(let i=0;i<2;i++)particles.push({x:hx*T+T/2+(Math.random()-0.5)*T*0.3,y:hy*T+T/2+(Math.random()-0.5)*T*0.3,vx:(Math.random()-0.5)*20,vy:(Math.random()-0.5)*20-(boltAnim.drop?14:0),life:0.25,max:0.25,col:boltAnim.col,size:T*0.05,g:boltAnim.drop?90:0});
    }
    if(boltAnim.t>boltAnim.dur+0.06)boltAnim=null;
  }
  render();
  requestAnimationFrame(frame);
}
```
