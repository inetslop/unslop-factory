function predictY(side){ // side +1 = right paddle (AI usually) 
  const px = side>0 ? pads[1].x : pads[0].x+pads[0].w; // crossing plane
  let x=ball.x,y=ball.y,vx=ball.vx,vy=ball.vy,t=0; const sdt=1/120;
  const gx=Math.cos(grav.a)*GMAG, gy=Math.sin(grav.a)*GMAG;
  while(t<3){ 
    x+=vx*sdt; y+=vy*sdt; vx+=gx*sdt; vy+=gy*sdt; t+=sdt;
    if(y-ball.r<WALL_T && vy<0){y=WALL_T+ball.r; vy=-vy;}
    if(y+ball.r>WALL_B && vy>0){y=WALL_B-ball.r; vy=-vy;}
    if(side>0){ if(vx>0 && x>=px) return y; } else { if(vx<0 && x<=px) return y; }
    if(x<-100||x>W+100) return H/2; // going away, shouldn't happen
  }
  return y;
}

/* ============ physics ============ */
function predictY(side){
  const p=pads[side>0?1:0], plane=side>0?p.x:p.x+p.w;
  let x=ball.x,y=ball.y,vx=ball.vx,vy=ball.vy,t=0; const sdt=1/120;
  const gx=Math.cos(grav.a)*GMAG, gy=Math.sin(grav.a)*GMAG;
  while(t<3){
    x+=vx*sdt; y+=vy*sdt; vx+=gx*sdt; vy+=gy*sdt; t+=sdt;
    if(y-ball.r<WALL_T&&vy<0){y=WALL_T+ball.r;vy=-vy;}
    if(y+ball.r>WALL_B&&vy>0){y=WALL_B-ball.r;vy=-vy;}
    if(side>0){if(vx>0&&x>=plane)return y;} else {if(vx<0&&x<=plane)return y;}
  }
  return H/2;
}
function stepPad(p,dt){

  if(p.ctrl==='ai'){
    const d=DIFFS[p.diff];
    const toward=p.side>0?ball.vx>0:ball.vx<0;
    if(G.phase==='rally'&&(toward||Math.abs(ball.x-W/2)<W*0.4)){
      if(G.time>=p.nextThink){
        p.targetY=(d.predict?predictY(p.side):ball.y)+rand(-d.err,d.err);
        p.nextThink=G.time+d.think;
      }
    } else p.targetY=H/2+Math.sin(G.time*1.4+p.side)*70;
    const maxV=d.speed*dt, dy=p.targetY-p.cy;
    v=clamp(dy,-maxV,maxV)/dt;
  } else {
    const idx=p.ctrl==='p1'?0:1;
    const tch=Object.values(touches).find(t=>t.pad===idx&&G.state==='play');
    if(tch){ const maxV=p.speed*dt, dy=tch.y-p.h/2-p.cy; v=clamp(dy,-maxV,maxV)/dt; }
    else{
      let nv=0;
      if(p.ctrl==='p1'){if(keys.KeyW)nv-=1;if(keys.KeyS)nv+=1;}
      else{if(keys.ArrowUp)nv-=1;if(keys.ArrowDown)nv+=1;}
      v=nv*p.speed;
    }
  }
  p.vel=v;
  p.cy=clamp(p.cy+v*dt, WALL_T+4, WALL_B-p.h-4);
}
function collidePaddle(p){

    hit(p,1); ball.x=face+ball.r;
  } else if(p.side>0&&ball.vx>0&&ball.x+ball.r>=face&&ball.x+ball.r<p.x+p.w+26&&Math.abs(ball.y-p.cy)<p.h/2+ball.r){
    hit(p,-1); ball.x=face-ball.r;
  }
}
function hit(p,dir){

  const ang=off*(Math.PI/3);

}

function gravUpdate(dt){
  grav.timer-=dt;
  if(grav.timer<1.5&&!grav.warned){grav.warned=true; SFX.warn();}
  if(grav.timer<=0){
    let n; do{n=(Math.random()*8)|0;}while(n===grav.dir);
    grav.from=grav.a; grav.to=n*Math.PI/4; grav.dir=n; grav.ease=0; grav.fired=false;
    grav.timer=rand(8,12); grav.warned=false;
  }
  if(grav.ease<1){
    grav.ease=Math.min(1,grav.ease+dt/0.9);
    grav.a=lerpAngle(grav.from,grav.to,easeOut(grav.ease));
    if(grav.ease===1&&!grav.fired){
      grav.fired=true;
      ripples.push({x:W/2,y:84,t:0,col:GOLD});
      SFX.shift();
    }
  }
  grav.gx=Math.cos(grav.a)*GMAG; grav.gy=Math.sin(grav.a)*GMAG;
}
function physics(dt){
  G.time+=dt;
  gravUpdate(dt);
  stepPad(pads[0],dt); stepPad(pads[1],dt);
  if(G.phase==='serve'){
    ball.x=W/2; ball.y=H/2; ball.vx=0; ball.vy=0;
    G.serveT-=dt;
    if(G.serveT<=0){launch(); G.phase='rally';}
    return;
  }
  ball.x+=ball.vx*dt; ball.y+=ball.vy*dt;
  ball.vx+=grav.gx*dt; ball.vy+=grav.gy*dt;
  let sp=Math.hypot(ball.vx,ball.vy);
  if(sp>980){ball.vx*=980/sp; ball.vy*=980/sp; sp=980;}
  if(G.state==='play'&&sp>G.topSpeed)G.topSpeed=sp;
  if(ball.y-ball.r<WALL_T&&ball.vy<0){ball.y=WALL_T+ball.r;ball.vy=-ball.vy;SFX.wall();burst(ball.x,WALL_T,5,'#bcd6ff',140,0.3);}
  if(ball.y+ball.r>WALL_B&&ball.vy>0){ball.y=WALL_B-ball.r;ball.vy=-ball.vy;SFX.wall();burst(ball.x,WALL_B,5,'#bcd6ff',140,0.3);}
  collidePaddle(pads[0]); collidePaddle(pads[1]);
  if(ball.x<-40)scorePoint(1);
  else if(ball.x>W+40)scorePoint(0);
}

/* ============ fx ============ */
function burst(x,y,n,col,spd,life){
  for(let i=0;i<n;i++){
    const a=Math.random()*Math.PI*2, s=rand(spd*0.3,spd);
    particles.push({x,y,vx:Math.cos(a)*s,vy:Math.sin(a)*s,t:0,life:rand(life*0.6,life),size:rand(1.5,4),col});
  }
  if(particles.length>320)particles.splice(0,particles.length-320);
}
function updateFX(dt){
  G.shake*=Math.pow(0.0008,dt); if(G.shake<0.1)G.shake=0;
  ball.pulse=Math.max(0,ball.pulse-dt*4);
  pads.forEach(p=>{p.flash=Math.max(0,p.flash-dt*3);});
  G.pop[0]=Math.max(0,G.pop[0]-dt*2.4); G.pop[1]=Math.max(0,G.pop[1]-dt*2.4);
  G.rallyPop=Math.max(0,G.rallyPop-dt*3);
  G.flashT=Math.max(0,G.flashT-dt*1.6);
  if(G.toast){G.toast.t-=dt; if(G.toast.t<=0)G.toast=null;}
  for(const d of dust){
    const v=8+d.d*26;
    d.x+=grav.gx/GMAG*v*dt + Math.sin(d.y*0.013+G.time)*4*dt;
    d.y+=grav.gy/GMAG*v*dt + Math.cos(d.x*0.011+G.time)*4*dt;
    if(d.x<-90)d.x=W+85; if(d.x>W+90)d.x=-85; if(d.y<-90)d.y=H+85; if(d.y>H+90)d.y=-85;
  }
  for(let i=particles.length-1;i>=0;i--){const p=particles[i];p.t+=dt;
    if(p.t>p.life){particles.splice(i,1);continue;}
    p.x+=p.vx*dt;p.y+=p.vy*dt;p.vx*=Math.pow(0.15,dt);p.vy*=Math.pow(0.15,dt);}
  for(let i=ripples.length-1;i>=0;i--){const r=ripples[i];r.t+=dt;if(r.t>0.8)ripples.splice(i,1);}
  for(let i=popups.length-1;i>=0;i--){const p=popups[i];p.t+=dt;p.y-=42*dt;if(p.t>1.1)popups.splice(i,1);}
}

function drawCompass(rt){
  const cx=W/2,cy=84;
  ctx.save();
  ctx.fillStyle='rgba(7,12,22,0.75)';
  ctx.beginPath();ctx.arc(cx,cy,34,0,Math.PI*2);ctx.fill();
  ctx.strokeStyle='rgba(255,209,102,'+(grav.warned?0.4+0.5*Math.abs(Math.sin(rt*10)):0.35)+')';
  ctx.lineWidth=2;ctx.stroke();
  for(let i=0;i<8;i++){
    const a=i*Math.PI/4, on=i===grav.dir;
    ctx.strokeStyle=on?GOLD:'rgba(200,225,240,0.3)';ctx.lineWidth=on?3:1.5;
    ctx.beginPath();
    ctx.moveTo(cx+Math.cos(a)*26,cy+Math.sin(a)*26);
    ctx.lineTo(cx+Math.cos(a)*(on?31:30),cy+Math.sin(a)*(on?31:30));
    ctx.stroke();
  }
  let ang=grav.a;
  if(grav.warned)ang+=Math.sin(rt*16)*0.15;
  const ax=Math.cos(ang),ay=Math.sin(ang);
  ctx.strokeStyle=GOLD;ctx.fillStyle=GOLD;
  ctx.shadowColor='rgba(255,209,102,0.8)';ctx.shadowBlur=8;
  ctx.lineWidth=3;
  ctx.beginPath();ctx.moveTo(cx-ax*14,cy-ay*14);ctx.lineTo(cx+ax*16,cy+ay*16);ctx.stroke();

  ctx.moveTo(cx+ax*24,cy+ay*24);
  ctx.lineTo(cx+ax*12-ay*7,cy+ay*12+ax*7);
  ctx.lineTo(cx+ax*12+ay*7,cy+ay*12-ax*7);
  ctx.closePath();ctx.fill();
  ctx.shadowBlur=0;
  if(grav.warned){
    ctx.setLineDash([5,7]);ctx.lineDashOffset=-rt*40;

  ctx.fillStyle='rgba(255,209,102,0.5)';ctx.fillText('GRAVITY',cx,cy+52);
  ctx.restore();
}

let last=performance.now(),acc=0;const STEP=1/240;
function frame(now){
  requestAnimationFrame(frame);
  let dt=(now-last)/1000; last=now; dt=Math.min(dt,0.1);
  if(G.state==='play'||G.state==='menu'){
    acc+=dt;
    while(acc>=STEP){physics(STEP);acc-=STEP;}
  } else acc=0;
  if(G.state!=='pause')updateFX(dt);
  render(now);
}
