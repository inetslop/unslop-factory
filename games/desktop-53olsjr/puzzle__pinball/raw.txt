/* ============ physics ============ */
function updFlip(f,dt){const tgt=f.down?f.act:f.rest,sp=(f.down?21:14)*dt,prev=f.a,d=tgt-f.a;f.a+=Math.abs(d)<=sp?d:(d>0?sp:-sp);f.om=dt>0?(f.a-prev)/dt:0;}
function collideSeg(s){const dx=s.x2-s.x1,dy=s.y2-s.y1,L2=dx*dx+dy*dy||1;let t=((ball.x-s.x1)*dx+(ball.y-s.y1)*dy)/L2;t=t<0?0:t>1?1:t;const px=s.x1+t*dx,py=s.y1+t*dy;let nx=ball.x-px,ny=ball.y-py;const d2=nx*nx+ny*ny,R=ball.r;if(d2>R*R)return;const d=Math.sqrt(d2)||.0001;nx/=d;ny/=d;ball.x=px+nx*R;ball.y=py+ny*R;const vn=ball.vx*nx+ball.vy*ny;if(vn<0){ball.vx-=(1+s.e)*vn*nx;ball.vy-=(1+s.e)*vn*ny;if(s.kick){ball.vx+=nx*s.kick;ball.vy+=ny*s.kick;s.flash=1;score+=10;sSling();spawnParts(ball.x,ball.y,4,'#ffb545',180);}}}
function collideFlip(f){const dx=Math.cos(f.a),dy=Math.sin(f.a);let t=((ball.x-f.px)*dx+(ball.y-f.py)*dy)/(f.len*f.len);t=t<0?0:t>1?1:t;const cx=f.px+dx*f.len*t,cy=f.py+dy*f.len*t;let nx=ball.x-cx,ny=ball.y-cy;let d=Math.hypot(nx,ny);const R=ball.r+f.r;if(d>=R)return;if(d<.0001){nx=0;ny=-1;d=1;}else{nx/=d;ny/=d;}ball.x=cx+nx*R;ball.y=cy+ny*R;const rx=cx-f.px,ry=cy-f.py,pvx=-f.om*ry,pvy=f.om*rx;const vn=(ball.vx-pvx)*nx+(ball.vy-pvy)*ny;if(vn<0){ball.vx-=1.35*vn*nx;ball.vy-=1.35*vn*ny;}}
function collideBump(b){let nx=ball.x-b.x,ny=ball.y-b.y;const R=ball.r+b.r,d2=nx*nx+ny*ny;if(d2>R*R)return;const d=Math.sqrt(d2)||.0001;nx/=d;ny/=d;ball.x=b.x+nx*R;ball.y=b.y+ny*R;const vn=ball.vx*nx+ball.vy*ny;if(vn<0){ball.vx-=1.5*vn*nx;ball.vy-=1.5*vn*ny;}ball.vx+=nx*300+b.vx*.6;ball.vy+=ny*300+b.vy*.6;b.hit=1;score+=25;sBump();spawnParts(ball.x,ball.y,7,'#ff5d8f',260);}
function collideNode(nd){let nx=ball.x-nd.x,ny=ball.y-nd.y;const R=ball.r+nd.r,d2=nx*nx+ny*ny;if(d2>R*R)return;const d=Math.sqrt(d2)||.0001;nx/=d;ny/=d;ball.x=nd.x+nx*R;ball.y=nd.y+ny*R;const vn=ball.vx*nx+ball.vy*ny;if(vn<0){ball.vx-=1.45*vn*nx;ball.vy-=1.45*vn*ny;}ball.vx+=nx*190;ball.vy+=ny*190+nd.vy*.5;
  if(nd.cd>0)return;nd.cd=.3;
  if(nd.lit){score+=5;sTick();spawnParts(ball.x,ball.y,4,'#9dff57',120);}
  else if(nodes.indexOf(nd)===nextIdx){nd.lit=true;const pts=100*chain;score+=pts;popup(nd.x,nd.y-26,'+'+pts,'#9dff57');sNode(chain);spawnParts(nd.x,nd.y,14,'#ffd257',300);chain++;chainPulse=1;nextIdx++;
    if(nextIdx>=nodes.length)startClear();else if(chain%3===0)setBanner('CHAIN ×'+chain,'KEEP IT ROLLING','#ffd257',.9);}
  else{nd.wrongT=1;chain=1;sWrong();shake=4;popup(nd.x,nd.y-26,'WRONG NODE','#ff5c5c');spawnParts(nd.x,nd.y,8,'#ff5c5c',220);}}
function physics(dt){const spd=Math.hypot(ball.vx,ball.vy),steps=Math.max(3,Math.min(9,Math.ceil(spd*dt/7))),h=dt/steps;
  for(let i=0;i<steps;i++){ball.vy+=G*h;ball.x+=ball.vx*h;ball.y+=ball.vy*h;
    for(const s of SEGMENTS)collideSeg(s);
    collideFlip(flL);collideFlip(flR);
    for(const b of bumpers)collideBump(b);
    for(const nd of nodes)collideNode(nd);
    const sp2=Math.hypot(ball.vx,ball.vy);if(sp2>2000){ball.vx*=2000/sp2;ball.vy*=2000/sp2;}}
  ball.trail.push({x:ball.x,y:ball.y});if(ball.trail.length>14)ball.trail.shift();}
function launch(){if(!ball.ready)return;ball.ready=false;launchedEver=true;const p=power;power=0;charging=false;ball.vy=-(980+820*p);ball.vx=(Math.random()-.5)*10;sLaunch(p);spawnParts(ball.x,862,10,'#7fe8ff',300);}
function nudge(){if(state==='playing'&&!ball.ready&&!ball.dead){ball.vy-=230;ball.vx+=(Math.random()-.5)*80;sNudge();shake=Math.max(shake,1.5);}}

/* ============ step ============ */
function step(dt){
  updFlip(flL,dt);updFlip(flR,dt);
  if(state==='playing'&&charging&&ball.ready)power=Math.min(1,power+dt*1.35);
  for(const b of bumpers){if(b.mv){const nx=b.bx+Math.sin(time*b.mv.s+b.mv.p)*b.mv.a;b.vx=nx-b.x;b.x=nx;}else b.vx=0;if(b.hit>0)b.hit=Math.max(0,b.hit-4*dt);}
  for(const nd of nodes){if(nd.mv){const ny=nd.by+Math.sin(time*nd.mv.s+nd.mv.p)*nd.mv.a;nd.vy=ny-nd.y;nd.y=ny;}else nd.vy=0;if(nd.cd>0)nd.cd-=dt;if(nd.wrongT>0)nd.wrongT-=dt;}
  if(SLING_L.flash>0)SLING_L.flash=Math.max(0,SLING_L.flash-5*dt);
  if(SLING_R.flash>0)SLING_R.flash=Math.max(0,SLING_R.flash-5*dt);
  if(slingCd>0)slingCd-=dt;
  if(banner.on){banner.t+=dt;if(banner.t>banner.dur)banner.on=false;}
  for(let i=particles.length-1;i>=0;i--){const p=particles[i];p.life-=dt;if(p.life<=0){particles.splice(i,1);continue;}p.vy+=G*.3*dt;p.x+=p.vx*dt;p.y+=p.vy*dt;}
  for(let i=popups.length-1;i>=0;i--){const p=popups[i];p.t+=dt;if(p.t>p.dur)popups.splice(i,1);}
  shake=Math.max(0,shake-14*dt);chainPulse=Math.max(0,chainPulse-3*dt);
  if(state==='clearing'){clearT+=dt;
    if(Math.random()<.5)particles.push({x:ball.x+(Math.random()-.5)*40,y:ball.y+(Math.random()-.5)*40,vx:(Math.random()-.5)*60,vy:-40-Math.random()*60,life:.6,max:.6,col:'#9dff57',sz:2});
    if(clearT>1.7)advanceLevel();return;}
  if(pendingOver>0){pendingOver-=dt;if(pendingOver<=0)showGameOver();}
  if(state==='playing'&&!ball.dead&&!ball.ready){physics(dt);
    if(ball.x>500&&ball.y>838&&Math.hypot(ball.vx,ball.vy)<30)resetBall();
    else{const sp=Math.hypot(ball.vx,ball.vy);if(sp<16)slowT+=dt;else slowT=0;if(slowT>2.6){ball.vy-=190;ball.vx+=(Math.random()-.5)*80;slowT=0;}
    if(ball.y>LH+30)onDrain();}}
}
