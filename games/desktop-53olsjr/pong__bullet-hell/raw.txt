function tone(f,d,type,v,f2,when){
  if(!AC)return;type=type||'square';v=v===undefined?0.4:v;when=when||0;
  const t0=AC.currentTime+when,o=AC.createOscillator(),g=AC.createGain();
  o.type=type;o.frequency.setValueAtTime(f,t0);
  if(f2)o.frequency.exponentialRampToValueAtTime(Math.max(30,f2),t0+d);
  g.gain.setValueAtTime(0.0001,t0);g.gain.exponentialRampToValueAtTime(v,t0+0.012);
  g.gain.exponentialRampToValueAtTime(0.0001,t0+d);
  o.connect(g);g.connect(master);o.start(t0);o.stop(t0+d+0.03);
}
function noiseBurst(d,v){
  if(!AC)return;const n=AC.createBufferSource();
  const buf=AC.createBuffer(1,Math.max(1,(AC.sampleRate*d)|0),AC.sampleRate),ch=buf.getChannelData(0);
  for(let i=0;i<ch.length;i++)ch[i]=(Math.random()*2-1)*(1-i/ch.length);
  n.buffer=buf;const g=AC.createGain();g.gain.value=v;
  const f=AC.createBiquadFilter();f.type='lowpass';f.frequency.value=900;
  n.connect(f);f.connect(g);g.connect(master);n.start();
}

function step(dt,mode){
  updatePlayer(dt);updateEnemy(dt);
  pPaddle.hitT=Math.max(0,pPaddle.hitT-dt*3);ePaddle.hitT=Math.max(0,ePaddle.hitT-dt*3);

  S.fireT-=dt;
  if(S.fireT<=0){spawnPattern(mode);const w=mode==='menu'?1:S.wave;
    S.fireT=(mode==='menu'?2.4:Math.max(0.55,2.1-(w-1)*0.17))*rand(0.85,1.15);}

  if(mode==='playing'){
    S.invuln=Math.max(0,S.invuln-dt);
    S.waveT+=dt;
    if(S.waveT>=WAVE_LEN){S.waveT=0;S.wave++;
      addFloat(FIELD.w/2,FIELD.h*0.34,'WAVE '+S.wave,C.cyan,42,1.7,true);
      const nm=S.wave<=PATTERN_NAMES.length?PATTERN_NAMES[S.wave-1]:'FULL CHAOS';
      addFloat(FIELD.w/2,FIELD.h*0.34+44,nm,'#cfe8ff',15,1.7,false);sfx.wave();}
  }

  if(S.serveT>0){S.serveT-=dt;
    if(S.serveT<=0){const a=rand(-0.55,0.55);
      ball.vx=Math.sin(a)*ball.speed;ball.vy=Math.cos(a)*ball.speed*ball.dir;}
  }else{
    const n=clamp(Math.ceil(ball.speed*dt/10),1,6);
    for(let i=0;i<n&&S.mode===mode;i++){
      const s=dt/n;ball.x+=ball.vx*s;ball.y+=ball.vy*s;
      if(ball.x-ball.r<0){ball.x=ball.r;ball.vx=Math.abs(ball.vx);wallFlash.l=1;sfx.wall();}
      if(ball.x+ball.r>FIELD.w){ball.x=FIELD.w-ball.r;ball.vx=-Math.abs(ball.vx);wallFlash.r=1;sfx.wall();}
      if(ball.vy>0&&circleRect(ball.x,ball.y,ball.r,pPaddle.x-pPaddle.w/2,pPaddle.y,pPaddle.w,pPaddle.h))bounce(pPaddle,true);
      if(ball.vy<0&&circleRect(ball.x,ball.y,ball.r,ePaddle.x-ePaddle.w/2,ePaddle.y,ePaddle.w,ePaddle.h))bounce(ePaddle,false);
    }
    ball.trail.push({x:ball.x,y:ball.y});if(ball.trail.length>10)ball.trail.shift();
    if(ball.y+ball.r<-6)playerGoal();
    else if(ball.y-ball.r>FIELD.h+6)missBall();
  }

  // bullets
  for(let i=bullets.length-1;i>=0;i--){const bl=bullets[i];bl.t+=dt;
    if(bl.type==='wig')bl.x+=Math.sin(bl.t*4.5+bl.ph)*150*dt;
    bl.x+=bl.vx*dt;bl.y+=bl.vy*dt;
    if(bl.y>FIELD.h+30||bl.y<-40||bl.x<-30||bl.x>FIELD.w+30){bullets.splice(i,1);continue;}
    if(mode==='playing'&&circleRect(bl.x,bl.y,bl.r,pPaddle.x-pPaddle.w/2,pPaddle.y,pPaddle.w,pPaddle.h)){
      bullets.splice(i,1);
      if(S.invuln>0){burst(bl.x,bl.y,7,C.cyan,200,0.35);sfx.shatter();}
      else loseLife('bullet');
    }
  }

  // lasers
  for(let i=lasers.length-1;i>=0;i--){const l=lasers[i];l.t-=dt;
    if(l.state==='warn'&&l.t<=0){l.state='fire';l.t=0.5;sfx.laserFire();}
    else if(l.state==='fire'){
      if(!l.hit&&mode==='playing'&&(pPaddle.x+pPaddle.w/2>l.x-l.w/2)&&(pPaddle.x-pPaddle.w/2<l.x+l.w/2)){
        l.hit=true;if(S.invuln>0){burst(pPaddle.x,pPaddle.y,10,C.cyan,220,0.4);sfx.shatter();}else loseLife('laser');}
      if(l.t<=0)lasers.splice(i,1);
    }
  }
  for(let i=emitters.length-1;i>=0;i--){emitters[i].t-=dt;if(emitters[i].t<=0)emitters.splice(i,1);}
}

  ctx.strokeStyle='rgba(255,255,255,0.03)';ctx.lineWidth=1;ctx.beginPath();
  for(let x=65;x<FIELD.w;x+=65){ctx.moveTo(x,0);ctx.lineTo(x,FIELD.h);}
  for(let y=65;y<FIELD.h;y+=65){ctx.moveTo(0,y);ctx.lineTo(FIELD.w,y);}
  ctx.stroke();

  // center line
  ctx.setLineDash([10,14]);ctx.strokeStyle='rgba(255,255,255,0.08)';ctx.lineWidth=2;
  ctx.beginPath();ctx.moveTo(0,FIELD.h/2);ctx.lineTo(FIELD.w,FIELD.h/2);ctx.stroke();ctx.setLineDash([]);
  ctx.strokeStyle='rgba(255,255,255,0.06)';
  ctx.beginPath();ctx.arc(FIELD.w/2,FIELD.h/2,34,0,TAU);ctx.stroke();

  // goal glow (top)
  if(S.topGlow>0){const gg=ctx.createLinearGradient(0,0,0,140);
    gg.addColorStop(0,'rgba(255,210,87,'+(0.5*S.topGlow)+')');gg.addColorStop(1,'rgba(255,210,87,0)');
    ctx.fillStyle=gg;ctx.fillRect(0,0,FIELD.w,140);}

  // lasers
  for(const l of lasers){
    if(l.state==='warn'){
      ctx.fillStyle='rgba(255,93,107,'+(0.10+0.06*Math.sin(timeNow*20))+')';
      ctx.fillRect(l.x-l.w/2,0,l.w,FIELD.h);
      ctx.strokeStyle='rgba(255,93,107,0.55)';ctx.lineWidth=2;ctx.setLineDash([14,18]);
      ctx.lineDashOffset=-timeNow*260;
      ctx.beginPath();ctx.moveTo(l.x,0);ctx.lineTo(l.x,FIELD.h);ctx.stroke();ctx.setLineDash([]);
    }else{
      ctx.fillStyle='rgba(255,93,107,0.5)';ctx.fillRect(l.x-l.w/2,0,l.w,FIELD.h);
      ctx.fillStyle='rgba(255,240,240,0.85)';ctx.fillRect(l.x-l.w*0.16,0,l.w*0.32,FIELD.h);
    }
  }

  // emitter sparks
  for(const e of emitters){const k=e.t/e.max;
    ctx.save();ctx.translate(e.x,e.y);ctx.rotate(PI/4);
    ctx.globalAlpha=k;ctx.fillStyle=e.color;const s=12*(1.6-k);
    ctx.fillRect(-s/2,-s/2,s,s);ctx.restore();}
  ctx.globalAlpha=1;

  drawPaddle(ePaddle,C.enemy,C.enemyCore);
  drawPaddle(pPaddle,C.player,C.playerCore);

  // ball (additive section)
  ctx.globalCompositeOperation='lighter';
  if(S.serveT>0){
    const pr=ball.r*(1+0.3*Math.sin(timeNow*8)),prog=1-clamp(S.serveT/0.9,0,1);
    ctx.fillStyle='rgba(255,246,220,0.9)';ctx.beginPath();ctx.arc(ball.x,ball.y,pr,0,TAU);ctx.fill();
    ctx.strokeStyle=C.player;ctx.lineWidth=3;ctx.globalAlpha=0.8;
    ctx.beginPath();ctx.arc(ball.x,ball.y,20,-PI/2,-PI/2+prog*TAU);ctx.stroke();ctx.globalAlpha=1;
  }else{
    for(let i=0;i<ball.trail.length;i++){const t=ball.trail[i],k=i/ball.trail.length;
      ctx.globalAlpha=k*0.22;ctx.fillStyle='#ffd8a0';
      ctx.beginPath();ctx.arc(t.x,t.y,ball.r*k,0,TAU);ctx.fill();}
    ctx.globalAlpha=1;
    ctx.fillStyle='rgba(255,220,140,0.3)';ctx.beginPath();ctx.arc(ball.x,ball.y,ball.r*2.6,0,TAU);ctx.fill();
    ctx.fillStyle='#ffffff';ctx.beginPath();ctx.arc(ball.x,ball.y,ball.r,0,TAU);ctx.fill();
    ctx.strokeStyle='rgba(255,210,87,0.9)';ctx.lineWidth=2;
    ctx.beginPath();ctx.arc(ball.x,ball.y,ball.r+3,0,TAU);ctx.stroke();
  }
  // bullets
  for(const bl of bullets){
    ctx.globalAlpha=0.16;ctx.fillStyle=bl.color;
    ctx.beginPath();ctx.arc(bl.x,bl.y,bl.r*2.1,0,TAU);ctx.fill();
    ctx.globalAlpha=1;ctx.beginPath();ctx.arc(bl.x,bl.y,bl.r,0,TAU);ctx.fill();
    ctx.fillStyle='rgba(255,255,255,0.9)';
    ctx.beginPath();ctx.arc(bl.x,bl.y,bl.r*0.42,0,TAU);ctx.fill();
  }
  // particles
  for(const p of parts){ctx.globalAlpha=clamp(p.t/p.max,0,1);ctx.fillStyle=p.color;
    ctx.fillRect(p.x-p.size/2,p.y-p.size/2,p.size,p.size);}
  ctx.globalAlpha=1;ctx.globalCompositeOperation='source-over';

  // floats
  for(const f of floats){const k=clamp(f.t/f.life,0,1);
    ctx.globalAlpha=k<0.85?1:k/0.85;ctx.fillStyle=f.color;ctx.textAlign='center';ctx.textBaseline='middle';
    if(f.big){ctx.font=f.size+'px '+F_DISP;
      ctx.shadowColor=f.color;ctx.shadowBlur=14;ctx.fillText(f.text,f.x,f.y);ctx.shadowBlur=0;}
    else{ctx.font=f.size+'px '+F_MONO;ctx.fillText(f.text,f.x,f.y);}
  }
  ctx.globalAlpha=1;

  // wall flashes
  if(wallFlash.l>0){const wg=ctx.createLinearGradient(0,0,70,0);
    wg.addColorStop(0,'rgba(255,210,87,'+(0.5*wallFlash.l)+')');wg.addColorStop(1,'rgba(255,210,87,0)');
    ctx.fillStyle=wg;ctx.fillRect(0,0,70,FIELD.h);}
  if(wallFlash.r>0){const wg=ctx.createLinearGradient(FIELD.w,0,FIELD.w-70,0);
    wg.addColorStop(0,'rgba(255,210,87,'+(0.5*wallFlash.r)+')');wg.addColorStop(1,'rgba(255,210,87,0)');
    ctx.fillStyle=wg;ctx.fillRect(FIELD.w-70,0,70,FIELD.h);}

  // HUD
  if(S.mode==='playing'||S.mode==='paused'){
    ctx.fillStyle='rgba(4,12,20,0.55)';ctx.fillRect(0,0,FIELD.w,30);
    ctx.textBaseline='middle';
    ctx.textAlign='left';ctx.font='24px '+F_DISP;ctx.fillStyle=C.player;
    ctx.fillText(S.score,14,16);
    ctx.font='11px '+F_MONO;ctx.fillStyle='#7fa8c9';ctx.fillText('/'+GOAL_TARGET,52,16);
    // wave center
    const nm=S.wave<=PATTERN_NAMES.length?PATTERN_NAMES[S.wave-1]:'FULL CHAOS';
    ctx.textAlign='center';ctx.font='11px '+F_MONO;ctx.fillStyle='#9fd8ff';
    ctx.fillText('WAVE '+S.wave+' · '+nm,FIELD.w/2,16);
    // lives orbs
    for(let i=0;i<3;i++){const ox=FIELD.w-18-i*22,oy=15;
      if(i<S.lives){ctx.fillStyle='#ff5d7e';ctx.beginPath();ctx.arc(ox,oy,7,0,TAU);ctx.fill();
        ctx.fillStyle='rgba(255,255,255,0.85)';ctx.beginPath();ctx.arc(ox-2,oy-2,2.4,0,TAU);ctx.fill();}
      else{ctx.strokeStyle='rgba(255,93,126,0.4)';ctx.lineWidth=1.5;
        ctx.beginPath();ctx.arc(ox,oy,7,0,TAU);ctx.stroke();}}
    // rally counter
    if(ball.rally>=4&&S.serveT<=0){
      ctx.textAlign='center';ctx.font='12px '+F_MONO;
      ctx.globalAlpha=0.6+0.4*Math.sin(timeNow*8);ctx.fillStyle=C.player;
      ctx.fillText('RALLY ×'+ball.rally,pPaddle.x,pPaddle.y-18);ctx.globalAlpha=1;}
    if(muted){ctx.textAlign='left';ctx.font='10px '+F_MONO;ctx.globalAlpha=0.5;
      ctx.fillStyle='#9fc3dd';ctx.fillText('SOUND OFF [M]',8,FIELD.h-10);ctx.globalAlpha=1;}
  }

  // frame + corner brackets
  ctx.strokeStyle='#2f5b80';ctx.lineWidth=2;ctx.strokeRect(1,1,FIELD.w-2,FIELD.h-2);
  const br=[[0,0,1,1,C.enemy],[FIELD.w,0,-1,1,C.enemy],[0,FIELD.h,1,-1,C.player],[FIELD.w,FIELD.h,-1,-1,C.player]];
  ctx.lineWidth=3;
  for(const c of br){ctx.strokeStyle=c[4];ctx.beginPath();
    ctx.moveTo(c[0]+c[2]*22,c[1]);ctx.lineTo(c[0],c[1]);ctx.lineTo(c[0],c[1]+c[3]*22);ctx.stroke();}

  ctx.restore();

  // screen-space overlays
  ctx.fillStyle=vig;ctx.fillRect(0,0,W,H);
  if(S.flash>0){ctx.fillStyle='rgba(255,60,70,'+(S.flash*0.38)+')';ctx.fillRect(0,0,W,H);}
  const dim=S.mode==='menu'?0.4:S.mode==='paused'?0.45:S.mode==='over'?0.5:0;
  if(dim){ctx.fillStyle='rgba(3,8,15,'+dim+')';ctx.fillRect(0,0,W,H);}
}

function frame(now){
  let dt=Math.min(0.05,(now-last)/1000);last=now;timeNow=now/1000;
  updateBG(dt);
  if(S.mode==='playing'||S.mode==='menu')step(dt*S.timeScale,S.mode);
  if(S.timeScale<1)S.timeScale=Math.min(1,S.timeScale+dt*2.2);
  updateFX(dt);
  draw();
  requestAnimationFrame(frame);
}
