let AC=null,masterG=null,nbuf=null,ambSrc=null,ambLfo=null,muted=false;
function audioInit(){if(AC){if(AC.state==='suspended')AC.resume();return;}try{AC=new (window.AudioContext||window.webkitAudioContext)();masterG=AC.createGain();masterG.gain.value=muted?0:0.5;masterG.connect(AC.destination);}catch(e){}}
function setMaster(v){if(masterG)masterG.gain.setTargetAtTime(v,AC.currentTime,0.05);}
function getNoise(){if(nbuf)return nbuf;const len=AC.sampleRate;nbuf=AC.createBuffer(1,len,AC.sampleRate);const d=nbuf.getChannelData(0);let l=0;for(let i=0;i<len;i++){const w=Math.random()*2-1;l=(l+0.02*w)/1.02;d[i]=l*3.5;}return nbuf;}
function tone(type,f0,f1,dur,vol,delay){if(!AC||muted)return;delay=delay||0;const t=AC.currentTime+delay,o=AC.createOscillator(),g=AC.createGain();o.type=type;o.frequency.setValueAtTime(f0,t);if(f1&&f1!==f0)o.frequency.exponentialRampToValueAtTime(Math.max(1,f1),t+dur);g.gain.setValueAtTime(0.0001,t);g.gain.exponentialRampToValueAtTime(vol,t+0.012);g.gain.exponentialRampToValueAtTime(0.0001,t+dur);o.connect(g);g.connect(masterG);o.start(t);o.stop(t+dur+0.06);}
function noiseHit(dur,vol,freq,delay){if(!AC||muted)return;delay=delay||0;const t=AC.currentTime+delay,src=AC.createBufferSource();src.buffer=getNoise();const f=AC.createBiquadFilter();f.type='lowpass';f.frequency.value=freq;const g=AC.createGain();g.gain.setValueAtTime(vol,t);g.gain.exponentialRampToValueAtTime(0.0001,t+dur);src.connect(f);f.connect(g);g.connect(masterG);src.start(t);src.stop(t+dur+0.05);}
function startAmbient(){if(!AC||ambSrc)return;const src=AC.createBufferSource();src.buffer=getNoise();src.loop=true;const f=AC.createBiquadFilter();f.type='lowpass';f.frequency.value=320;const g=AC.createGain();g.gain.value=0.05;const lfo=AC.createOscillator();lfo.frequency.value=0.35;const lg=AC.createGain();lg.gain.value=0.02;lfo.connect(lg);lg.connect(g.gain);src.connect(f);f.connect(g);g.connect(masterG);src.start();lfo.start();ambSrc=src;ambLfo=lfo;}
function stopAmbient(){if(ambSrc){try{ambSrc.stop()}catch(e){}try{ambLfo.stop()}catch(e){}ambSrc=null;ambLfo=null;}}

const playCrit=()=>{tone('triangle',700,1250,0.18,0.35);tone('sine',1400,1900,0.12,0.2,0.03);noiseHit(0.09,0.3,4200);};

function strike(px,py){if(S.mode!=='play')return;const dx=px-core.cx,dy=py-core.cy;if(dx*dx+dy*dy>Math.pow(core.R*1.45,2)){miss(px,py);return;}

 S.embers+=g;S.total+=g;S.heat=clamp(S.heat+3.2,0,100);hit=1;

 if(crit){playCrit();shake=Math.max(shake,8);}else playStrike(S.combo);
 checkAscend();}
function miss(x,y){S.combo=0;S.lastStrike=-9;playMiss();for(let i=0;i<3;i++)parts.push({x:x+rand(-6,6),y:y+rand(-6,6),vx:rand(-20,20),vy:rand(-45,-10),life:rand(.3,.6),max:.6,size:rand(1.5,2.5),col:'hsl(30,20%,45%)',g:0,ly:1});}
function checkAscend(){let up=false;while(S.total>=S.nextNeed){S.s++;S.heat=clamp(S.heat+28,0,100);S.nextNeed=60*Math.pow(1.45,S.s-1);up=true;}if(up){playAscend();rings.push({x:core.cx,y:core.cy,r:core.R*0.5,vr:750,a:.9,col:'hsla(38,100%,68%,'});shake=Math.max(shake,5);computeCore();$('stratumChip').textContent='STRATUM '+roman(S.s)+' · ×'+fmtMult(stratM());}}
function update(dt){S.t+=dt;const wr=S.ups.wisp*0.6*cacheM()*stratM();if(wr>0){S.embers+=wr*dt;S.total+=wr*dt;}
 S.heat+=(wispHeat()-decayRate())*dt;
 if(S.heat<=0){S.heat=0;gameOver();return;}
 if(S.heat<25){S.hb-=dt;if(S.hb<=0){playHeart();S.hb=0.85;}}
 checkAscend();}
function gameOver(){S.mode='over';stopAmbient();playOver();

 $('newBest').classList.toggle('hidden',!nb);showScreen('over');updateBestPanel();}
