function initAudio(){
  if(AC) { if(AC.state==='suspended') AC.resume(); return; }
  try{
    AC = new (window.AudioContext||window.webkitAudioContext)();
    master = AC.createGain(); master.gain.value = muted?0:0.18; master.connect(AC.destination);
  }catch(e){}
}
function tone(type,f0,f1,dur,vol=1,delay=0){
  if(!AC||muted) return;
  const t0=AC.currentTime+delay;
  const o=AC.createOscillator(), g=AC.createGain();
  o.type=type; o.frequency.setValueAtTime(Math.max(1,f0),t0);
  if(f1&&f1!==f0) o.frequency.exponentialRampToValueAtTime(Math.max(1,f1),t0+dur);
  g.gain.setValueAtTime(vol,t0);
  g.gain.exponentialRampToValueAtTime(0.0008,t0+dur);
  o.connect(g); g.connect(master);
  o.start(t0); o.stop(t0+dur+0.03);
}
function noiseBurst(dur,vol,f0,f1,delay=0){
  if(!AC||muted) return;
  const t0=AC.currentTime+delay;
  const n=Math.floor(AC.sampleRate*dur);
  const buf=AC.createBuffer(1,n,AC.sampleRate);
  const d=buf.getChannelData(0);
  for(let i=0;i<n;i++) d[i]=(Math.random()*2-1)*(1-i/n);
  const src=AC.createBufferSource(); src.buffer=buf;
  const f=AC.createBiquadFilter(); f.type='lowpass';
  f.frequency.setValueAtTime(Math.max(40,f0),t0);
  f.frequency.exponentialRampToValueAtTime(Math.max(40,f1),t0+dur);
  const g=AC.createGain(); g.gain.value=vol;
  src.connect(f); f.connect(g); g.connect(master);
  src.start(t0);
}

/* ================= ACTIONS ================= */
function tryMove(dx){
 if(state!=='play'||!piece)return false;
 const nx=piece.x+dx;
 if(!collides(ROT[piece.id][piece.r],nx,piece.y)){piece.x=nx;S.move();return true;}
 S.bump();return false;
}
function rotate(dir){
 if(state!=='play'||!piece)return;
 const nr=(piece.r+dir+4)%4,m=ROT[piece.id][nr];
 const kicks=[[0,0],[-1,0],[1,0],[-2,0],[2,0],[0,-1]];
 for(const k of kicks){
  if(!collides(m,piece.x+k[0],piece.y+k[1])){piece.r=nr;piece.x+=k[0];piece.y+=k[1];S.rot();return;}}
 S.bump();
}
function ghostY(){let gy=piece.y;while(!collides(ROT[piece.id][piece.r],piece.x,gy+1))gy++;return gy;}
function hardDrop(){
 if(state!=='play'||!piece)return;
 const gy=ghostY(),d=gy-piece.y;
 piece.dropPts+=2*d;piece.y=gy;
 S.hard();shake(4,150);lockPiece();
}

/* ================= UPDATE ================= */
function update(dt,t){
 if(hold&&t-hold.t0>160&&t-hold.last>=45){hold.last=t;tryMove(hold.dir);}
 if(touchHold&&t-touchHold.t0>170&&t-touchHold.last>=60){touchHold.last=t;tryMove(touchHold.dir);}
 const interval=softDown?Math.max(36,gravBase/22):gravBase;
 acc+=dt;
 while(acc>=interval&&state==='play'&&piece){
  acc-=interval;
  if(!collides(ROT[piece.id][piece.r],piece.x,piece.y+1)){piece.y++;if(softDown)piece.dropPts++;}
  else{lockPiece();break;}
 }
 if(state==='clear'&&t>=clearUntil){
  hideOverlay();levelIndex++;buildLevel(levelIndex);state='play';acc=0;spawnNext();
 }
}

function spawnBurst(x,y,col,n){
 for(let i=0;i<n;i++){const a=Math.random()*6.283,sp=40+Math.random()*170;
  const life=350+Math.random()*350;
  particles.push({x,y,vx:Math.cos(a)*sp,vy:Math.sin(a)*sp-70,life,max:life,col,r:1.5+Math.random()*2.5});}
 if(particles.length>420)particles.splice(0,particles.length-420);
}
function initDust(){dust=[];for(let i=0;i<34;i++)dust.push({x:Math.random()*W,y:Math.random()*H,v:.15+Math.random()*.35,ph:Math.random()*6.28,r:.6+Math.random()*1.4,a:.05+Math.random()*.09});}
function stepDust(dt,t){for(const d of dust){d.y-=d.v*dt/1000;d.x+=Math.sin(t/900+d.ph)*dt*.00028;if(d.y<-1){d.y=H+1;d.x=Math.random()*W;}}}
function stepParticles(dt){
 for(let i=particles.length-1;i>=0;i--){const p=particles[i];p.life-=dt;if(p.life<=0){particles.splice(i,1);continue;}
  p.vy+=420*dt/1000;p.x+=p.vx*dt/1000;p.y+=p.vy*dt/1000;}
 for(let i=floaters.length-1;i>=0;i--){const f=floaters[i];f.life-=dt;f.y-=34*dt/1000;if(f.life<=0)floaters.splice(i,1);}
 shake.t+=dt;clearFlash=Math.max(0,clearFlash-dt/650);introAge+=dt;
}

function buildWalls(){
 if(!grid)return;
 if(!wallsCv)wallsCv=document.createElement('canvas');
 wallsCv.width=game.width;wallsCv.height=game.height;
 const g=wallsCv.getContext('2d');g.setTransform(dpr,0,0,dpr,0,0);
 const s=cellPx,BW=W*s,BH=H*s;
 g.fillStyle='#0a1119';g.fillRect(0,0,BW,BH);
 for(let y=0;y<H;y++)for(let x=0;x<W;x++)if(grid[y][x]===0){
  g.fillStyle='rgba(150,200,225,0.035)';g.fillRect(x*s+1,y*s+1,s-2,s-2);}
 const isOpen=(y,x)=>{if(y<0)return true;if(y>=H||x<0||x>=W)return false;return grid[y][x]===0;};
 for(let y=0;y<H;y++)for(let x=0;x<W;x++){
  if(grid[y][x]!==1)continue;
  const n=hash(x*3.7,y*5.1),b=46+n*16,px=x*s,py=y*s;
  g.fillStyle='rgb('+((b+16)|0)+','+((b+7)|0)+','+((b-8)|0)+')';g.fillRect(px,py,s,s);
  g.fillStyle='rgba(0,0,0,.25)';g.fillRect(px,py+(s*.55)|0,s,1);
  if((x+y)%2===0)g.fillRect(px+(s*.5)|0,py,1,(s*.55)|0);else g.fillRect(px+(s*.5)|0,py+(s*.55)|0,1,(s*.45)|0);
  if(isOpen(y-1,x)){g.fillStyle='rgba(255,214,150,.20)';g.fillRect(px,py,s,Math.max(2,(s*.16)|0));}
  if(isOpen(y+1,x)){g.fillStyle='rgba(0,0,0,.45)';const h=Math.max(2,(s*.18)|0);g.fillRect(px,py+s-h,s,h);}
  if(isOpen(y,x-1)){g.fillStyle='rgba(255,230,190,.10)';g.fillRect(px,py,2,s);}
  if(isOpen(y,x+1)){g.fillStyle='rgba(0,0,0,.30)';g.fillRect(px+s-2,py,2,s);}
 }
}
function drawCell(g,px,py,col,glow){
 const s=cellPx;g.save();
 if(glow){g.shadowColor=col;g.shadowBlur=s*.5;}
 const x=px+1,y=py+1,w=s-2,h=s-2,r=Math.max(2,s*.14);
 g.fillStyle=col;rr(g,x,y,w,h,r);g.fill();
 g.shadowBlur=0;
 g.fillStyle='rgba(255,255,255,.28)';rr(g,x+1.5,y+1.5,w-3,h*.3,r*.8);g.fill();
 g.fillStyle='rgba(0,0,0,.22)';rr(g,x+1.5,y+h-h*.32,w-3,h*.32,r*.8);g.fill();
 g.strokeStyle='rgba(0,0,0,.35)';g.lineWidth=1;rr(g,x+.5,y+.5,w-1,h-1,r);g.stroke();
 g.restore();
}

 portals.forEach((p,i)=>{
