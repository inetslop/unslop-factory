function cellsOf(p,dx,dy,rot){
  const r=rot===undefined?p.rot:rot;
  return SHAPES[p.type][r].map(c=>[p.x+c[0]+(dx||0), p.y+c[1]+(dy||0)]);
}
function fits(cells){
  for(const c of cells){
    const x=c[0],y=c[1];
    if(x<0||x>=COLS||y>=ROWS) return false;
    if(y>=0 && G.grid[y][x]) return false;
  }
  return true;
}
function grounded(){ return !fits(cellsOf(G.piece,0,1)); }
function onGroundReset(){ const p=G.piece; if(grounded()&&p.resets<15){p.lockT=0;p.resets++;} }
function doMove(dx,dy){
  const p=G.piece; if(!p) return false;
  if(fits(cellsOf(p,dx,dy))){ p.x+=dx; p.y+=dy; onGroundReset(); return true; }
  return false;
}
function doRotate(dir){
  const p=G.piece; if(!p||p.type==='O'){ if(p&&p.type==='O')SFX.rot(); return; }
  const nr=(p.rot+dir+4)%4, tbl=p.type==='I'?KI:KJ;
  for(const k of (tbl[p.rot+'-'+nr]||[[0,0]])){
    const test=SHAPES[p.type][nr].map(c=>[p.x+c[0]+k[0], p.y+c[1]+k[1]]);
    if(fits(test)){ p.x+=k[0]; p.y+=k[1]; p.rot=nr; p.lastAct='rotate'; onGroundReset(); SFX.rot(); return; }
  }
}
function ghostY(){ const p=G.piece; let gy=p.y; while(fits(cellsOf(p,0,gy-p.y+1)))gy++; return gy; }
function spawnNext(){
  const t=G.queue.shift(); G.queue.push(drawBag());
  G.piece={type:t,x:3,y:-2,rot:0,lockT:0,resets:0,lastAct:'move'};
  G.holdUsed=false; G.dropAcc=0; G.lastSpin='none';
  if(!fits(cellsOf(G.piece))) crash('top');
}
function doHold(){
  const p=G.piece; if(!p||G.holdUsed) return;
  const cur=p.type, nt=G.hold!==null?G.hold:G.queue.shift();
  G.hold=cur; p.type=nt; p.x=3; p.y=-2; p.rot=0; p.lockT=0; p.resets=0; p.lastAct='move';
  G.dropAcc=0; G.holdUsed=true; SFX.hold();
  if(!fits(cellsOf(p))) crash('top');
}
function hardDrop(){
  const p=G.piece; let d=0;
  while(fits(cellsOf(p,0,1))){p.y++;d++;}
  G.score+=d*2;
  for(let i=0;i<6;i++)G.parts.push(pt(BX+p.x*CS+8+Math.random()*p.type*4,BY+ (p.y+3)*CS, (Math.random()-0.5)*60, -30-Math.random()*60, 0.35, 3, '#cfd6e4'));
  G.shake=Math.max(G.shake,3); SFX.drop(); lockPiece();
}
function detectSpin(){
  const p=G.piece;
  if(p.type!=='T'||p.lastAct!=='rotate') return 'none';
  const cx=p.x+1,cy=p.y+1; let c=0;
  for(const d of [[-1,-1],[1,-1],[-1,1],[1,1]]){
    const x=cx+d[0],y=cy+d[1];
    if(x<0||x>=COLS||y<0||y>=ROWS||G.grid[y][x])c++;
  }
  return c>=3?'full':(c===2?'mini':'none');
}
function lockPiece(){
  const p=G.piece; let topOut=false;
  for(const c of cellsOf(p)){ if(c[1]<0)topOut=true; if(c[1]>=0)G.grid[c[1]][c[0]]=p.type; }
  G.lastSpin=detectSpin();
  SFX.land(); G.shake=Math.max(G.shake,2);
  if(topOut){ crash('top'); return; }
  const rows=[]; for(let y=0;y<ROWS;y++) if(G.grid[y].every(v=>v))rows.push(y);
  if(rows.length){ G.clearRows=rows; G.clearT=0; G.state='clearing'; }
  else afterLockNoClear();
}
function afterLockNoClear(){
  if(G.combo>=0)G.combo=-1;
  G.b2b=false;
  spawnNext();
}
function resolveClear(){
  const n=G.clearRows.length;
  for(const y of G.clearRows){
    for(let x=0;x<COLS;x++){
      const t=G.grid[y][x]; if(!t)continue;
      for(let i=0;i<2;i++)G.parts.push(pt(BX+x*CS+13,BY+y*CS+13,(Math.random()-0.5)*260,-60-Math.random()*160,0.5,4,PCOL[t],true));
    }
    G.grid.splice(y,1); G.grid.unshift(Array(COLS).fill(null));
  }
  let base=0,name='',diff=false;
  const spin=G.lastSpin;
  if(spin==='full'){
    base=[400,800,1200,1600][n]; diff=true;
    name='T-SPIN '+(n===4?'TETRIS!!':(n===3?'TRIPLE!':(n===2?'DOUBLE!':''))).trim();
  }else if(spin==='mini'){
    base=[100,200,400,0][n];
    name='T-SPIN MINI'+(n>0?' +'+['','SINGLE','DOUBLE','TRIPLE'][n]:'');
  }else{
    base=[0,100,300,500,800][n]; diff=(n===4);
    name=['','SINGLE','DOUBLE','TRIPLE!','TETRIS!!'][n];
  }
  if(diff&&G.b2b){base=Math.floor(base*1.5); G.pops.push(mkPop(BX+130,BY+250,'B2B ×1.5','#ffd23f',15));}
  G.b2b=diff;
  const comboBefore=Math.max(0,G.combo);
  const pts=(base*G.level)+(comboBefore>0?50*comboBefore*G.level:0);
  G.score+=pts; G.lines+=n;
  G.combo=(n>0||spin!=='none')?G.combo+1:-1;
  if(G.combo>0)G.topCombo=Math.max(G.topCombo,G.combo);
  const gain=[0,30,68,115,190][n]+(spin==='full'?15:spin==='mini'?6:0)+(comboBefore>0?Math.min(30,comboBefore*6):0);
  G.dist=Math.min(RACE_LEN,G.dist+gain);
  G.nitro=Math.min(100,G.nitro+[0,10,16,24,40][n]+(spin==='full'?18:spin==='mini'?8:0));
  if(name){
    const col=n===4?'#ffd23f':(spin!=='none'?'#b06bff':'#ffffff');
    G.pops.push(mkPop(BX+130,BY+215,name,col, n===4?30:22));

  }
  if(G.combo>=2)G.pops.push(mkPop(BX+130,BY+282,'COMBO ×'+(G.combo+1),'#7dff5a',15));
  const lbl=name||['','SINGLE','DOUBLE','TRIPLE','TETRIS'][n];
  if(lbl)G.bestClear=lbl;
  if(n===4){SFX.tetris();G.shake=Math.max(G.shake,6);}else{SFX.clear(n);if(spin!=='none')SFX.spin();}
  const nl=1+Math.floor(G.lines/10);
  if(nl>G.level){G.level=nl;G.pops.push(mkPop(BX+130,BY+160,'LEVEL '+nl+' · SPEED UP!','#ff9e3d',18));SFX.levelup();}
  spawnNext();
}

function updateAI(dt){
  for(const a of G.ais){
    if(a.fin)continue;
    a.bt-=dt;
    if(!a.burst&&a.bt<=0){a.burst=true;a.bt=1.4;}
    else if(a.burst&&a.bt<=0){a.burst=false;a.bt=3+Math.random()*4;}
    let pace=a.base+Math.sin(G.raceTime*0.6+a.ph)*1.1+(a.burst?3.5:0);
    const diff=a.dist-G.dist;
    if(diff<G.dist-140&&diff<-140)pace+=1.4;
    else if(diff>180)pace-=1.2;
    pace=Math.max(1,pace);
    a.spd=pace; a.dist=Math.min(RACE_LEN,a.dist+pace*dt);
    if(a.dist>=RACE_LEN){a.fin=true;a.time=G.raceTime;}
  }
}

function handlePlayKey(k){
  const kl=k.toLowerCase();
  switch(true){
    case k==='ArrowLeft'||kl==='a': if(doMove(-1,0))SFX.move(); break;
    case k==='ArrowRight'||kl==='d': if(doMove(1,0))SFX.move(); break;
    case k==='ArrowDown'||kl==='s': G.soft=true; break;
    case k==='ArrowUp'||kl==='x'||kl==='w': doRotate(1); break;
    case kl==='z'||kl==='q': doRotate(-1); break;
    case k===' ': hardDrop(); break;
    case kl==='c'||k==='Shift': doHold(); break;
    case kl==='e'||kl==='f':
      if(!G.nitroOn&&G.nitro>0){G.nitroOn=true;SFX.nitro();}
      G.nitroOn=true; break;
    case k==='p'||k==='P'||k==='Escape': pauseGame(); break;
    case kl==='m': toggleMute(); break;
    case kl==='r': newGame(); break;
  }
}

function gravMs(){return Math.max(70,800-(G.level-1)*65);}
function update(dt){
  if(!G)return;
  G.shake=Math.max(0,G.shake-dt*30);
  G.flash=Math.max(0,G.flash-dt*1.4);
  if(G.goT>0)G.goT-=dt;
  /* particles & popups always */
  for(let i=G.parts.length-1;i>=0;i--){const p=G.parts[i];p.t+=dt;p.x+=p.vx*dt;p.y+=p.vy*dt;p.vy+=(p.g||0)*dt;if(p.t>=p.life)G.parts.splice(i,1);}
  for(let i=G.pops.length-1;i>=0;i--){const p=G.pops[i];p.t+=dt;p.y-=28*dt;if(p.t>=p.life)G.pops.splice(i,1);}

  switch(G.state){
    case 'countdown':{
      G.cdT+=dt;
      const lit=Math.min(5,Math.floor((G.cdT-0.7)/0.6));
      if(lit>G.lastLight&&lit>0){G.lastLight=lit;SFX.light();}
      if(G.cdT>=0.7+5*0.6+0.3&&!goFlag){goFlag=true;G.state='play';G.goT=1;G.raceTime=0;spawnNext();SFX.go();}
      break;}
    case 'play':{
      goFlag=false;
      G.raceTime+=dt;
      /* gravity */
      const iv=G.soft?Math.min(gravMs(),45):gravMs();
      G.dropAcc+=dt*1000;
      while(G.dropAcc>=iv){
        if(doMove(0,1)){if(G.soft)G.score+=1;G.dropAcc-=iv;}else{G.dropAcc=0;break;}
      }
      if(grounded()){G.piece.lockT+=dt;if(G.piece.lockT>=0.5)lockPiece();}
      else G.piece.lockT=0;
      /* race movement */
      let sp=2.5;
      if(G.nitroOn&&G.nitro>0){sp+=30;G.nitro=Math.max(0,G.nitro-34*dt);
        const y=trkY(G.disp);
        G.parts.push(pt(RCX+YOU_LANE+(Math.random()-0.5)*8,y+22,(Math.random()-0.5)*40,120+Math.random()*80,0.3,4,['#ffd23f','#ff9e3d','#fff'][ (Math.random()*3)|0 ]));
      }
      G.dist=Math.min(RACE_LEN,G.dist+sp*dt);
      if(!G.fin&&G.dist>=RACE_LEN)playerFinish();
      updateAI(dt);
      break;}
    case 'clearing':{
      goFlag=false;
      G.clearT+=dt;
      if(G.clearT>=0.24){G.clearRows=[];resolveClear();}
      updateAI(dt);
      break;}
    case 'finish':{
      G.finT+=dt;
      if(G.finT>=1.5)doOver();
      break;}
    case 'over':
      G.overT+=dt; break;
  }
  /* smooth display distances */
  const k=Math.min(1,dt*6);
  G.disp+=(G.dist-G.disp)*k;
  for(const a of G.ais)a.disp+=(a.dist-a.disp)*k;
  G.scrollT+= (G.state==='play'?(2.5+(G.nitroOn&&G.nitro>0?30:0)):2)*dt*0.14;
  /* engine sound */
  if(AC&&engGain){
    let g=0,f=70;
    if(G.state==='play'){const sp=2.5+(G.nitroOn&&G.nitro>0?30:0);g=0.045+Math.min(sp,35)/35*0.05;f=62+sp*2.4+Math.sin(performance.now()*0.02)*4;}
    else if(G.state==='countdown'){g=0.03+flick(0.01);f=90+Math.sin(performance.now()*0.03)*10;}
    engGain.gain.setTargetAtTime(g,AC.currentTime,0.08);
    engO1.frequency.setTargetAtTime(f,AC.currentTime,0.05);
    engO2.frequency.setTargetAtTime(f*0.5,AC.currentTime,0.05);
  }
}

  ctx.fillStyle='rgba(0,0,0,0.35)';ctx.fillRect(x+1,y+2,s-1,s-1);
  ctx.fillStyle=col;ctx.fillRect(x,y,s-1,s-1);
  ctx.fillStyle='rgba(255,255,255,0.42)';ctx.fillRect(x,y,s-1,Math.max(2,s*0.28));
  ctx.fillStyle='rgba(0,0,0,0.28)';ctx.fillRect(x,y+s*0.68,s-1,s*0.3-1);
  ctx.strokeStyle='rgba(0,0,0,0.5)';ctx.lineWidth=1;ctx.strokeRect(x+0.5,y+0.5,s-2,s-2);
}

let vign=null;
function render(){
  ctx.clearRect(0,0,LW,LH);
  if(!G){drawMenuBG(0);return;}
  if(G.state==='menu'){drawMenu();return;}
  ctx.save();
  if(G.shake>0)ctx.translate((Math.random()-0.5)*G.shake,(Math.random()-0.5)*G.shake);
  drawBG();
  drawTopBar();
  drawLeftPanel();
  drawBoard();
  drawNextCol();
  drawTrack();
  drawParts();
  drawPops();
  ctx.restore();
  /* overlays */
  if(G.state==='countdown')drawCountdown();
  if(G.goT>0)drawGO();
  if(G.state==='pause')drawPause();
  if(G.state==='finish')drawFinishBanner();
  if(G.state==='over')drawOverPanel();

}

/* ---- board ---- */
function drawBoard(){
  /* racing stripes beside board */
  for(const sx of [BX-14,BX+COLS*CS+8]){
    const off=(G.scrollT*20)%36;
    for(let i=-1;i<ROWS*CS/18+1;i++){
      ctx.fillStyle=(i%2===0)?'#ff3b30':'#e8ecf5';
      ctx.fillRect(sx,BY+i*18-off,6,18);
    }
  }
  /* frame */
  ctx.fillStyle='#0d1220';rr(BX-4,BY-4,COLS*CS+8,ROWS*CS+8,6);ctx.fill();
  const fg=ctx.createLinearGradient(0,BY,0,BY+ROWS*CS);
  fg.addColorStop(0,'#0e1526');fg.addColorStop(1,'#131b30');
  ctx.fillStyle=fg;ctx.fillRect(BX,BY,COLS*CS,ROWS*CS);
  /* danger glow */
  let topRow=ROWS;for(let y=0;y<ROWS;y++){if(G.grid[y].some(v=>v)){topRow=y;break;}}
  if(topRow<=3){const p=0.25+0.2*Math.sin(performance.now()*0.01);
    const dg=ctx.createLinearGradient(0,BY,0,BY+120);dg.addColorStop(0,'rgba(255,40,30,'+p+')');dg.addColorStop(1,'rgba(255,40,30,0)');
    ctx.fillStyle=dg;ctx.fillRect(BX,BY,COLS*CS,120);}
  /* grid */
  ctx.strokeStyle='rgba(160,190,255,0.06)';ctx.lineWidth=1;
  for(let x=1;x<COLS;x++){ctx.beginPath();ctx.moveTo(BX+x*CS,BY);ctx.lineTo(BX+x*CS,BY+ROWS*CS);ctx.stroke();}
  for(let y=1;y<ROWS;y++){ctx.beginPath();ctx.moveTo(BX,BY+y*CS);ctx.lineTo(BX+COLS*CS,BY+y*CS);ctx.stroke();}
  /* stack */
  for(let y=0;y<ROWS;y++)for(let x=0;x<COLS;x++)if(G.grid[y][x])drawBlock(BX+x*CS,BY+y*CS,CS,G.grid[y][x]);
  /* clear flash */
  if(G.state==='clearing'){
    const a=G.clearT/0.24;
    for(const y of G.clearRows){
      ctx.fillStyle='rgba(255,255,255,'+(0.85*(1-a))+')';
      ctx.fillRect(BX,BY+y*CS,COLS*CS,CS);
      ctx.fillStyle='rgba(255,210,63,'+a+')';
      ctx.fillRect(BX,BY+y*CS+(1-a)*4,COLS*CS,(a*2));
    }
  }
  /* ghost + active */
  const p=G.piece;
  if(p&&(G.state==='play')){
    const gy=ghostY();
    for(const c of SHAPES[p.type][p.rot]){
      const x=p.x+c[0],y=gy+c[1];
      if(y<0)continue;
      ctx.strokeStyle=PCOL[p.type];ctx.globalAlpha=0.45;ctx.lineWidth=2;
      ctx.strokeRect(BX+x*CS+3,BY+y*CS+3,CS-6,CS-6);ctx.globalAlpha=1;
    }
    ctx.save();ctx.shadowColor=PCOL[p.type];ctx.shadowBlur=10;
    for(const c of SHAPES[p.type][p.rot]){
      const x=p.x+c[0],y=p.y+c[1]; if(y<0)continue;
      drawBlock(BX+x*CS,BY+y*CS,CS,p.type);
    }
    ctx.restore();
  }
  /* frame border */
  ctx.strokeStyle='rgba(232,236,245,0.85)';ctx.lineWidth=2;rr(BX-4,BY-4,COLS*CS+8,ROWS*CS+8,6);ctx.stroke();
}

function drawTrack(){

  const rx=RCX-RH,rw=RH*2;
  ctx.fillStyle='#2a3040';ctx.fillRect(rx,TRK_TOP-6,rw,TRK_BOT-TRK_TOP+12);
  /* curbs */
  const off=(G.scrollT*120)%28;
  for(const cx of [rx-10,rx+rw]){
    for(let i=-1;i<(TRK_BOT-TRK_TOP)/14+1;i++){
      ctx.fillStyle=((i+Math.floor(off/14))%2===0)?'#ff3b30':'#e8ecf5';
