  const hp=Math.round(base.hp*(1+0.35*(d-1))*(base.boss?1.2:1));
  G.enemy={type:base,name:base.name,hp,maxhp:hp,flash:0,spawnT:0,dying:false,lungeT0:0,burn:{d:0,t:0},pi:0,intent:null};
  rollIntent(G.enemy);
}
function rollIntent(en){
  const d=G.depth,sc=1+0.16*(d-1);
  let kind=en.type.cycle?en.type.cycle[en.pi++%en.type.cycle.length]:pick(en.type.pool);
  let hits=1,mult=1;
  if(kind==='big')mult=1.8;else if(kind==='aa'){hits=2;mult=.6}else if(kind==='aaa'){hits=3;mult=.45}
  const dmg=Math.max(1,Math.round(rnd(en.type.atk[0]*sc,en.type.atk[1]*sc)*mult));
  const labels={a:'STRIKE',big:'HEAVY BLOW',aa:'TWIN FANGS',aaa:'SWEEP',d:'SCORCHING BREATH',h:'REKINDLE'};
  en.intent={kind,hits,dmg,label:labels[kind]};
}

function lungeOff(en){const s=SCALE();if(!en.lungeT0)return 0;
  const p=(tNowS-en.lungeT0)/.5;if(p>=1){en.lungeT0=0;return 0;}
  if(p<.28)return 16*s*(p/.28);
  return -74*s*Math.sin(((p-.28)/.72)*Math.PI);}
function enemyCenter(){const en=G.enemy,s=SCALE();
  return{x:W*0.73+lungeOff(en),y:gy()-en.type.hgt*s*.55};}

function enemyTurn(){
  if(G.mode!=='player'||!G.enemy)return;

  if(en.burn.t>0){en.hp-=en.burn.d;en.burn.t--;

  if(it.kind==='d'){after(base,applyScorch);}

    const c=enemyCenter();addFloat(c.x,c.y-20,'+'+h,'#7ee787',18);SFX.heal();});}
  else{for(let i=0;i<it.hits;i++)after(base+i*260,()=>hitHero(it.dmg));}
  after(base+(Math.max(1,it.hits)-1)*260+780,startPlayerTurn);
}

}

    after(240,()=>{if(!G.enemy||G.enemy.dying)return;const e=G.enemy;

function dealToEnemy(base,kind){
  const en=G.enemy;if(!en||en.dying)return;
  if(tNowS-G.lastHit>0.9)G.combo=0;
  G.combo++;G.lastHit=tNowS;
  const frenzy=G.combo%10===0;
  let dmg=base*(0.85+Math.random()*0.3);
  const crit=Math.random()<critChance();if(crit)dmg*=1.7;if(frenzy)dmg*=1.5;
  dmg=Math.max(1,Math.round(dmg));
  en.hp-=dmg;en.flash=1;
  G.stats.hits++;G.stats.dmg+=dmg;if(crit)G.stats.crits++;
  const c=enemyCenter();
  addFloat(c.x+rnd(-16,16),c.y,'-'+dmg,crit?'#ffd166':'#fff3e0',crit?27:19);
  if(frenzy){addFloat(c.x,c.y-38,'FRENZY ×'+G.combo,'#ff7a1a',22);SFX.frenzy();}
  spawnParts(c.x,c.y,crit?18:10,kind==='fire'?'#ff9a2e':'#ffd9b0',crit?190:130);
  const g=(1+UP().greed)*(frenzy?2:1);G.gold+=g;G.stats.gold+=g;
  addFloat(c.x+rnd(-8,8),c.y-20,'+'+g+'g','#ffd166',13,.75);popGold();
  if(UP().ember>0&&kind!=='fire'){en.burn.d=Math.max(en.burn.d,1+UP().ember);en.burn.t=Math.max(en.burn.t,2);}
  if(kind==='heavy')SFX.heavy();else if(crit)SFX.crit();else SFX.hit();
  G.shake=Math.max(G.shake,crit?9:4);
  updateTop();
  if(en.hp<=0)killEnemy();
}

    const bw=150*s,bx=ex,by=gyy-en.type.hgt*s-24*s;

    if(pct>0){RR(bx-bw/2,by,bw*pct,9,4);
      let g=ctx.createLinearGradient(0,by,0,by+9);
      g.addColorStop(0,pct>.35?'#ffb347':'#ff7a5a');g.addColorStop(1,pct>.35?'#e05a10':'#b02a14');

  }
}
function drawIntent(en,x,y,s){
  const it=en.intent;if(!it)return;
  const cols={a:'#ffb347',big:'#ff5340',aa:'#ffd166',aaa:'#ffd166',d:'#ff8a5c',h:'#7ee787'};
  const col=cols[it.kind];
  const w=(it.kind==='d'||it.kind==='h')?108:88;
  const sc=it.kind==='big'?1+Math.sin(tNowS*9)*.05:1;
  ctx.save();ctx.translate(x,y);ctx.scale(sc,sc);
  RR(-w/2,-17,w,34,8);ctx.fillStyle='rgba(16,8,5,.88)';ctx.fill();
  if(it.kind==='big'){ctx.shadowColor=col;ctx.shadowBlur=10;}
  ctx.strokeStyle=col;ctx.lineWidth=1.5;ctx.stroke();ctx.shadowBlur=0;
  const ix=-w/2+17;
  ctx.strokeStyle=col;ctx.fillStyle=col;ctx.lineCap='round';
  if(it.kind==='a'||it.kind==='big'){ctx.lineWidth=3;ctx.beginPath();ctx.moveTo(ix-6,8);ctx.lineTo(ix+7,-9);ctx.stroke();
    if(it.kind==='big'){ctx.beginPath();ctx.moveTo(ix-1,10);ctx.lineTo(ix+12,-7);ctx.stroke();}}
  else if(it.kind==='aa'||it.kind==='aaa'){ctx.font='bold 13px Georgia';ctx.textAlign='center';
    ctx.fillText(it.kind==='aa'?'×2':'×3',ix,5);}
  else if(it.kind==='d'){const fl=Math.sin(tNowS*10)*2;
    ctx.beginPath();ctx.moveTo(ix-5,9);ctx.quadraticCurveTo(ix-7,-1+fl*.4,ix,-9-fl);
    ctx.quadraticCurveTo(ix+7,0,ix+5,9);ctx.closePath();ctx.fill();}
  else{ctx.lineWidth=4;ctx.beginPath();ctx.moveTo(ix,-8);ctx.lineTo(ix,8);ctx.moveTo(ix-8,0);ctx.lineTo(ix+8,0);ctx.stroke();}
  const tx=w/2-14;
  if(it.kind==='d'||it.kind==='h'){ctx.font='bold 13px Copperplate,Georgia';ctx.textAlign='center';
    ctx.fillStyle=col;ctx.fillText(it.kind==='d'?'SCORCH':'REKINDLE',tx,5);}
  else{ctx.font='bold '+(it.hits>1?16:19)+'px Georgia';ctx.textAlign='center';
    ctx.fillStyle=col;ctx.fillText(it.dmg*it.hits,tx,7);
    if(it.hits>1){ctx.font='9px Georgia';ctx.fillStyle='rgba(243,230,207,.6)';ctx.fillText('per hit',tx,-8);}}
  ctx.restore();
}
function drawProjectiles(){
  for(let i=projs.length-1;i>=0;i--){const p=projs[i];
    const q=clamp((tNowS-p.t0)/p.dur,0,1);
    if(q>=1){projs.splice(i,1);continue;}
    const e=easeOutCubic(q),x=lerp(p.x0,p.x1,e),y=lerp(p.y0,p.y1,e)-Math.sin(e*Math.PI)*40;
    parts.push({x,y,vx:rnd(-20,20),vy:rnd(-20,20),g:-30,t:0,life:.3,c:'#ff9a2e',r:rnd(1.5,3)});
    let g=ctx.createRadialGradient(x,y,1,x,y,16);

    ctx.fillStyle=g;ctx.beginPath();ctx.arc(x,y,16,0,7);ctx.fill();}
}
function drawFx(){

  for(const p of parts){const a=clamp(1-p.t/p.life,0,1);
    ctx.save();ctx.globalAlpha=a;ctx.fillStyle=p.c;

}

let lastT=performance.now();
function loop(now){
  const dt=Math.min(.05,(now-lastT)/1000);lastT=now;tNowS=now/1000;
  for(let i=timers.length-1;i>=0;i--){if(now>=timers[i].t){const fn=timers[i].fn;timers.splice(i,1);fn();}}
  G.shake=Math.max(0,G.shake-dt*26);
  G.hurtT=Math.max(0,G.hurtT-dt*2.2);
  G.winFlash=Math.max(0,G.winFlash-dt*1.2);
  if(G.hero.guardT>0)G.hero.guardT=Math.max(0,G.hero.guardT-dt*1.4);
  const en=G.enemy;
  if(en){if(en.flash>0)en.flash=Math.max(0,en.flash-dt*4);
    if((en.type.id==='drake'||en.type.boss)&&!en.dying&&Math.random()<dt*8){
      const c=enemyCenter();parts.push({x:c.x+rnd(-14,26),y:c.y-rnd(10,40),vx:rnd(-15,15),vy:rnd(-40,-10),g:-40,t:0,life:.8,c:'#ff9a2e',r:rnd(1.5,3)});}}
  // ambient embers
  if(parts.length<240&&Math.random()<dt*7){
    const fromTorch=Math.random()<.6;
    const x=fromTorch?(Math.random()<.5?W*0.055:W*0.945)+rnd(-8,8):rnd(0,W);
    const y=fromTorch?H*0.36:gy()+rnd(0,(H-gy())*.6);
    parts.push({x,y,vx:rnd(-8,8),vy:rnd(-34,-16),g:-12,t:0,life:rnd(1.5,3),c:'#ff9a2e',r:rnd(1,2.4)});}
  for(let i=parts.length-1;i>=0;i--){const p=parts[i];p.t+=dt;
    if(p.t>=p.life){parts.splice(i,1);continue;}
    p.vy+=p.g*dt;p.x+=p.vx*dt;p.y+=p.vy*dt;}
  for(let i=floats.length-1;i>=0;i--){const f=floats[i];f.t+=dt;f.y-=46*dt;
    if(f.t>=f.life)floats.splice(i,1);}
  for(let i=rings.length-1;i>=0;i--){rings[i].t+=dt;if(rings[i].t>=rings[i].dur)rings.splice(i,1);}
  // cursor
  const want=G.mode==='player'&&G.enemy&&!G.enemy.dying&&mouseInEnemy();
  if(cv.style.cursor!==(want?'pointer':'default'))cv.style.cursor=want?'pointer':'default';
  render();
  requestAnimationFrame(loop);
}
