    else if(S.phase==='clear'){S.clearT+=dt;if(S.clearT>=1.7)nextLevel()}
    if(S.phase!=='clear'){S.timeLeft-=dt;
      if(S.timeLeft<=10){const s=Math.ceil(S.timeLeft);if(s!==S.tickSec){S.tickSec=s;sfx.tick()}}
      if(S.timeLeft<=0){S.timeLeft=0;
        if(S.phase==='alive')die('splash');
        else if(S.phase==='respawn'){placeStart();S.lives--;updLives();
          if(S.lives<=0)gameOver();else{S.phase='alive';frog.grace=.7}}}}
    frog.grace=Math.max(0,frog.grace-dt);
    if(S.phase==='alive')checkHazards(dt)}

  S.shake=Math.max(0,S.shake-dt*26);S.flash=Math.max(0,S.flash-dt*1.5);
  pads.forEach(p=>p.shake=Math.max(0,p.shake-dt*3));
  updTimerBar()}

function burst(gx,gy,kind,n){const px=ox+(gx+.5)*cell,py=oy+(gy+.5)*cell;

**Hop Logic** is a puzzle-Frogger: each level's far bank shows three lotus pads in a scrambled order, and you must cross the road and rapids to fill them **1 → 2 → 3** before the timer drains — riding logs and ducks, dodging cars, trucks, taxis and (from level 2) a pike lane. Wrong pad? You bounce back into the water and risk it. Includes lives, level scaling, time bonuses, one-ups, an attract-mode background behind the menu, WebAudio sound effects (M to mute), pause (P), restart (R), touch d-pad on mobile, and a persisted best score.
