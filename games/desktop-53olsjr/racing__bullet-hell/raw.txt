    spawnT-=dt;if(spawnT<=0&&enem.length<2+Math.floor(d*5)){spawnT=(2.1-1.2*d)*rand(.7,1.3);const r=Math.random();if(r<.5)spawnMite();else if(r<.8)spawnWeaver();else spawnVan();}
    gateT-=dt;if(gateT<=0){gateT=lerp(9.5,6,d)*rand(.8,1.2);spawnGate();}
  }

  updateEnemies(dt);updateBoss(dt);
  /* bullets */
  for(let i=ebul.length-1;i>=0;i--){const b=ebul[i];b.x+=b.vx*dt;b.y+=b.vy*dt;if(b.x<-60||b.x>W+60||b.y<-90||b.y>H+60)ebul.splice(i,1);}
  for(let i=pbul.length-1;i>=0;i--){
    const b=pbul[i];b.x+=b.vx*dt;b.y+=b.vy*dt;
    if(b.y<-40||b.x<-20||b.x>W+20){pbul.splice(i,1);continue;}
    let hit=false;
    for(let j=enem.length-1;j>=0;j--){const e=enem[j],dx=b.x-e.x,dy=b.y-e.y,rr2=(e.r+4)*(e.r+4);
      if(dx*dx+dy*dy<rr2){hit=true;e.hp--;e.flash=1;spark(b.x,b.y,'#ffffff');sfx.tick();if(e.hp<=0)killEnemy(j);break;}}
    if(!hit&&boss&&!boss.dying&&boss.inOut){const dx=b.x-boss.x,dy=b.y-boss.y;
      if(Math.abs(dx)<96&&Math.abs(dy)<44){hit=true;boss.hp--;boss.flash=1;spark(b.x,b.y,'#ffb35c');sfx.tick();
        if(boss.hp<=0){boss.dying=true;boss.dieT=1.6;hide(el.bossWrap);}}}
    if(hit)pbul.splice(i,1);
  }
  updateDrops(dt);updateParts(dt);updateFloats(dt);

  /* player collisions */
  if(!P.dead){
    for(let i=ebul.length-1;i>=0;i--){const b=ebul[i],dx=b.x-P.x,dy=b.y-P.y,rr2=(b.r+7)*(b.r+7);
      if(dx*dx+dy*dy<rr2){ebul.splice(i,1);hitPlayer();if(P.dead)break;}}
    if(!P.dead)for(const e of enem){const dx=e.x-P.x,dy=e.y-P.y,rr2=(e.r+11)*(e.r+11);if(dx*dx+dy*dy<rr2){hitPlayer();break;}}
    if(!P.dead&&boss&&boss.inOut&&!boss.dying){const dx=P.x-boss.x,dy=P.y-boss.y;if(Math.abs(dx)<106&&Math.abs(dy)<54)hitPlayer();}
  }

  shake*=Math.pow(.001,dt);if(shake<.15)shake=0;
  flash=Math.max(0,flash-dt*1.8);
  el.dmg.style.opacity=(flash+(P.shields===0&&!P.dead?.12+.08*Math.sin(T*6):0)).toFixed(3);
  el.boostFx.style.opacity=(boosting&&!P.dead)?1:0;
  engineSet(true,clamp((scroll-300)/520,0,1),boosting);
}
