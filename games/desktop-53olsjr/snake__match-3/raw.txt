 {base:'#ff5d6c',lt:'#ffc4ca',dk:'#b0283a'}, // triangle

let AC=null,master=null;
function audio(){
  if(!AC){const Ctx=window.AudioContext||window.webkitAudioContext;if(!Ctx)return;
    AC=new Ctx();master=AC.createGain();master.gain.value=muted?0:.22;master.connect(AC.destination);}
  if(AC.state==='suspended')AC.resume();
}
function tone(f,dur,type,vol,delay,slideTo){
  if(!AC||muted)return;const t0=AC.currentTime+(delay||0);
  const o=AC.createOscillator(),g=AC.createGain();
  o.type=type||'square';o.frequency.setValueAtTime(f,t0);
  if(slideTo)o.frequency.exponentialRampToValueAtTime(Math.max(30,slideTo),t0+dur);
  g.gain.setValueAtTime(vol||.3,t0);g.gain.exponentialRampToValueAtTime(.001,t0+dur);
  o.connect(g);g.connect(master);o.start(t0);o.stop(t0+dur+.05);
}
function noiseBurst(dur,vol,freq){
  if(!AC||muted)return;const n=(AC.sampleRate*dur)|0,buf=AC.createBuffer(1,n,AC.sampleRate),d=buf.getChannelData(0);
  for(let i=0;i<n;i++)d[i]=(Math.random()*2-1)*(1-i/n);
  const s=AC.createBufferSource();s.buffer=buf;
  const f=AC.createBiquadFilter();f.type='bandpass';f.frequency.value=freq||2400;f.Q.value=.8;
  const g=AC.createGain();g.gain.value=vol||.2;
  s.connect(f);f.connect(g);g.connect(master);s.start();
}

/* ================= board / snake ================= */
function mkGem(c,fall){return {c,fall:fall||0,land:0,h:rand(10)};}
function newBoard(){
  const n=colorsForLevel(level);board=[];
  for(let x=0;x<COLS;x++){board[x]=[];
    for(let y=0;y<ROWS;y++)board[x][y]=mkGem(irand(n),(y+1)*rand(.4,.95));}
}
function reset(){
  score=0;combo=1;comboT=0;level=1;eats=0;matched=0;
  newBoard();
  const cy=Math.floor(ROWS/2);
  snake=[{x:3,y:cy,c:0},{x:2,y:cy,c:0},{x:1,y:cy,c:0}];
  prevSnake=snake.map(s=>({x:s.x,y:s.y}));
  dir={x:1,y:0};queue=[];
  particles.length=texts.length=rings.length=0;
  shakeT=0;flashA=0;acc=0;ph=0;paused=false;newBestRun=false;
  overTok++;
  updateHUD();
}
function refill(){
  const n=colorsForLevel(level);
  const occ=new Set(snake.map(s=>s.x+','+s.y));
  for(let x=0;x<COLS;x++){
    let y=0;
    while(y<ROWS){
      if(occ.has(x+','+y)){y++;continue;}
      const a=y;while(y<ROWS&&!occ.has(x+','+y))y++;
      const b=y-1,gems=[];
      for(let r=a;r<=b;r++){const g=board[x][r];if(g){gems.push({g,r});board[x][r]=null;}}
      let slot=b;
      for(const it of gems){it.g.fall=slot-it.r;it.g.land=0;board[x][slot]=it.g;slot--;}
      while(slot>=a){board[x][slot]=mkGem(irand(n),(slot-a)+1);slot--;}
    }
  }
}
function tailRun(){
  let i=snake.length-1,c=snake[i].c;if(c<0)return 0;
  let r=0;while(i>=0&&snake[i].c===c){r++;i--;}return r;
}

  for(let i=0;i<n;i++){
    const a=rand(6.28),sp=rand(power*.35,power);
    particles.push({x:px,y:py,vx:Math.cos(a)*sp,vy:Math.sin(a)*sp-40,g:520,size:rand(2.5,6.5),
      life:rand(.4,.85),max:.85,rot:rand(6.3),vr:rand(-9,9),color:colors[irand(colors.length)],shape:i%3?'rect':'dot'});
  }
}

/* ================= game step ================= */
function step(){
  if(queue.length){const d=queue.shift();
    if(!(d.x===dir.x&&d.y===dir.y)&&!(d.x===-dir.x&&d.y===-dir.y))dir=d;}
  const hd=snake[0];
  const nx=(hd.x+dir.x+COLS)%COLS,ny=(hd.y+dir.y+ROWS)%ROWS;
  const gem=board[nx][ny];
  for(let i=0;i<snake.length;i++){
    if(i===snake.length-1&&!gem)continue; // tail vacates this tick
    if(snake[i].x===nx&&snake[i].y===ny){die();return;}
  }
  prevSnake=snake.map(s=>({x:s.x,y:s.y}));
  const nb=[{x:nx,y:ny,c:-1}];
  if(gem){for(let i=0;i<snake.length;i++)nb.push(snake[i]);nb[nb.length-1].c=gem.c;}
  else{for(let i=0;i<snake.length-1;i++)nb.push(snake[i]);}
  snake=nb;

  if(gem){
    board[nx][ny]=null;refill();eats++;score+=10;
    const p=cellPx(nx,ny);
    sfx.eat(tailRun());
    addText(p.x,p.y-cell*.4,'+10','#bfe0da',Math.max(10,cell*.32),.8,-40);
    burstParticles(p.x,p.y,[GEMS[gem.c].base,GEMS[gem.c].lt],4,90);
    shake(.12,2);
  }
  // match-3 on the tail
  let i=snake.length-1,c=snake[i].c,run=0;
  if(c>=0){while(i>=0&&snake[i].c===c){run++;i--;}
    if(run>=3)doBurst(Math.min(run,snake.length-3));}
  const nl=levelFor(score);
  if(nl>level){level=nl;onLevel();}
  updateHUD();
}
function doBurst(n){
  if(n<1)return;
  const segs=snake.splice(snake.length-n);
  matched+=n;
  const pts=40*n*combo;
  combo=Math.min(9,combo+1);comboT=COMBO_MS;
  score+=pts;
  let cx=0,cy2=0;segs.forEach(s=>{const p=cellPx(s.x,s.y);cx+=p.x;cy2+=p.y;});
  cx/=n;cy2/=n;
  segs.forEach((s,k)=>{const G=GEMS[s.c],p=cellPx(s.x,s.y);
    burstParticles(p.x,p.y,[G.base,G.lt,'#ffffff'],8,150+combo*14);});
  rings.push({x:cx,y:cy2,r:6,vr:cell*(6+combo),life:1,dur:.5,color:GEMS[segs[0].c].base});
  addText(cx,cy2,'+'+pts,combo>1?'#ffb627':'#eef8f5',Math.max(14,cell*.5)*(1+.08*Math.min(combo,6)),1.2,-34);
  if(combo>1)addText(cx,cy2-cell*.9,'CHAIN ×'+combo,'#ff8d99',Math.max(11,cell*.34),1,-26);
  sfx.burst(combo,n);
  shake(.32,3+combo*1.3);
  if(combo>=3){flashA=Math.max(flashA,.12);flashC='90,220,255';}
}
function onLevel(){
  sfx.level();
  const cx=ox+cell*COLS/2,cy=oy+cell*ROWS/2;
  addText(cx,cy-cell*.6,'LEVEL '+level,'#7fe7ff',Math.max(18,cell*1.05),1.6,-12);
  rings.push({x:cx,y:cy,r:10,vr:cell*9,life:1,dur:.8,color:'#3fd8ff'});
  if(colorsForLevel(level)>colorsForLevel(level-1)){
    addText(cx,cy+cell*.5,'NEW GEM COLOR!','#ffb627',Math.max(12,cell*.5),1.6,-10);
    const nc=colorsForLevel(level)-1;
    const occ=new Set(snake.map(s=>s.x+','+s.y));
    let dropped=0,tries=0;
    while(dropped<6&&tries<50){const x=irand(COLS),y=irand(2);
      if(!board[x][y]&&!occ.has(x+','+y)){board[x][y]=mkGem(nc,y+1.5+rand(.8));dropped++;}tries++;}
  }
}
function die(){
  mode='over';deathT=performance.now();

  burstParticles(p.x,p.y,['#eef8f5','#ff5d6c','#ffffff'],26,300);

  sfx.over();

  if(newBestRun){best=score;try{localStorage.setItem('serpentGems.best',best);}catch(e){}}
  updateHUD();

  setTimeout(()=>{if(tok!==overTok||mode!=='over')return;populateOver();},800);
}

function gemPath(c,x,y,r){
  ctx.beginPath();
  if(c===0){ctx.moveTo(x,y-r);ctx.lineTo(x+r*.72,y);ctx.lineTo(x,y+r);ctx.lineTo(x-r*.72,y);ctx.closePath();}
  else if(c===1){ctx.arc(x,y,r*.8,0,6.283);}
  else if(c===2){ctx.moveTo(x,y-r*.95);ctx.lineTo(x+r*.9,y+r*.72);ctx.lineTo(x-r*.9,y+r*.72);ctx.closePath();}
  else if(c===3){rr(ctx,x-r*.72,y-r*.72,r*1.44,r*1.44,r*.28);}
  else{for(let k=0;k<6;k++){const a=Math.PI/6+k*Math.PI/3,px=x+Math.cos(a)*r*.85,py=y+Math.sin(a)*r*.85;
    k?ctx.lineTo(px,py):ctx.moveTo(px,py);}ctx.closePath();}
}
function sparkle(x,y,s,a){
  ctx.save();ctx.translate(x,y);ctx.globalAlpha=a;ctx.fillStyle='#fff';
  for(let k=0;k<2;k++){ctx.rotate(k*Math.PI/2);
    ctx.beginPath();ctx.moveTo(0,-s);ctx.lineTo(s*.28,0);ctx.lineTo(0,s);ctx.lineTo(-s*.28,0);ctx.closePath();ctx.fill();}
  ctx.restore();
}
function drawGem(g,x,y,time){
  const G=GEMS[g.c],r=cell*.4;
  let sy=1,sx=1;
  if(g.land>0&&g.land<1){sy=1-.3*Math.exp(-g.land*7)*Math.cos(g.land*14);sx=2-sy;}
  ctx.fillStyle='rgba(0,0,0,.25)';
  ctx.beginPath();ctx.ellipse(x,y+r*.6,r*.6,sx*r*.2,0,0,6.283);ctx.fill();
  ctx.save();ctx.translate(x,y);ctx.scale(sx,sy);
  const gr=ctx.createLinearGradient(0,-r,0,r);
  gr.addColorStop(0,G.lt);gr.addColorStop(.55,G.base);gr.addColorStop(1,G.dk);
  gemPath(g.c,0,0,r);ctx.fillStyle=gr;ctx.fill();
  ctx.lineWidth=1.4;ctx.strokeStyle='rgba(255,255,255,.35)';ctx.stroke();
  ctx.fillStyle='rgba(255,255,255,.75)';
  ctx.beginPath();ctx.ellipse(-r*.26,-r*.34,r*.17,r*.09,-.6,0,6.283);ctx.fill();
  const sp=(g.h+time*.5)%1;
  if(sp<.12)sparkle(r*.3,-r*.28,r*.3,(1-sp/.12)*.8);
  ctx.restore();
}
function drawBG(time){
  const g=ctx.createLinearGradient(0,0,0,H);
  g.addColorStop(0,'#0a1c25');g.addColorStop(.55,'#0d2833');g.addColorStop(1,'#081720');
  ctx.fillStyle=g;ctx.fillRect(0,0,W,H);
  let gl=ctx.createRadialGradient(W*.18,H*.22,0,W*.18,H*.22,Math.max(W,H)*.5);
  gl.addColorStop(0,'rgba(63,216,255,.06)');gl.addColorStop(1,'rgba(63,216,255,0)');
  ctx.fillStyle=gl;ctx.fillRect(0,0,W,H);
  gl=ctx.createRadialGradient(W*.85,H*.85,0,W*.85,H*.85,Math.max(W,H)*.5);
  gl.addColorStop(0,'rgba(255,182,39,.05)');gl.addColorStop(1,'rgba(255,182,39,0)');
  ctx.fillStyle=gl;ctx.fillRect(0,0,W,H);
  for(const d of drifters){
    const px=((d.x+time*d.sp)%1.1-0.05)*W,py=d.y*H+Math.sin(time*.4+d.r*10)*8;
    ctx.save();ctx.translate(px,py);ctx.rotate(d.r+time*.15);

    const s=d.s*cell*.4;
    ctx.strokeRect(-s/2,-s/2,s,s);
    ctx.restore();
  }
}
function panelRect(){const p=7;return {x:ox-p,y:oy-p,w:cell*COLS+p*2,h:cell*ROWS+p*2};}

  ctx.lineWidth=1.5;ctx.strokeStyle='rgba(150,230,220,.16)';ctx.stroke();
  ctx.fillStyle='rgba(255,255,255,.016)';
  for(let x=0;x<COLS;x++)for(let y=0;y<ROWS;y++)
    if((x+y)%2===0)ctx.fillRect(ox+x*cell,oy+y*cell,cell,cell);
}

function render(time){
  drawBG(time);
  const sh=shakeT>0?Math.min(1,shakeT*6):0;
  const sx=(Math.random()-.5)*2*shakeMag*sh,sy=(Math.random()-.5)*2*shakeMag*sh;
  ctx.save();ctx.translate(sx,sy);

  drawPanel();

  // interpolated centers (wrap-safe)
  const stepMs=1000/speedForLevel(level);
  const t=mode==='play'?clamp(acc/stepMs,0,1):1;
  const n=snake.length,ctr=[];
  for(let i=0;i<n;i++){
    const s=snake[i],p=prevSnake[i]||s;
    let dx=s.x-p.x,dy=s.y-p.y;
    if(Math.abs(dx)>1)dx=0;if(Math.abs(dy)>1)dy=0;
    ctr.push({x:ox+(p.x+dx*t+.5)*cell,y:oy+(p.y+dy*t+.5)*cell});
  }
  const amp=(mode==='play'&&!paused)?cell*.045:cell*.02;
  for(let i=1;i<n;i++){
    const a=ctr[Math.max(0,i-1)],b=ctr[Math.min(n-1,i+1)];
    let vx=b.x-a.x,vy=b.y-a.y,L=Math.hypot(vx,vy);
    if(L<.001||L>cell*2)continue;
    vx/=L;vy/=L;
    const off=Math.sin(ph-i*.85)*amp;
    ctr[i].x+=-vy*off;ctr[i].y+=vx*off;
  }

  // gems (clipped so falling ones emerge from behind the frame)
  ctx.save();ctx.beginPath();ctx.rect(ox,oy,cell*COLS,cell*ROWS);ctx.clip();
  for(let x=0;x<COLS;x++)for(let y=0;y<ROWS;y++){
    const g=board[x][y];if(!g)continue;
    drawGem(g,ox+(x+.5)*cell,oy+(y-g.fall+.5)*cell,time);
  }
  // headlight
  const hl=ctx.createRadialGradient(ctr[0].x,ctr[0].y,0,ctr[0].x,ctr[0].y,cell*1.5);
  hl.addColorStop(0,'rgba(255,244,214,.10)');hl.addColorStop(1,'rgba(255,244,214,0)');
  ctx.fillStyle=hl;ctx.fillRect(ox,oy,cell*COLS,cell*ROWS);
  // body (tail → head), grouped by color runs
  ctx.lineCap='round';ctx.lineJoin='round';
  for(let i=n-1;i>=1;){
    let j=i,c=snake[i].c;
    while(j>0&&snake[j-1].c===c)j--;
    if(c>=0){
      const G=GEMS[c];
      ctx.beginPath();ctx.moveTo(ctr[j].x,ctr[j].y);
      for(let k=j+1;k<=i;k++)ctx.lineTo(ctr[k].x,ctr[k].y);
      ctx.lineTo(ctr[i].x+.02,ctr[i].y);
      ctx.strokeStyle=G.base;ctx.lineWidth=cell*.6;ctx.stroke();
      ctx.strokeStyle='rgba(255,255,255,.18)';ctx.lineWidth=cell*.24;ctx.stroke();
    }
    i=j-1;
  }
  // pending-match glow on the tail run
  const tr=tailRun();
  if(tr>=2&&mode==='play'){
    ctx.beginPath();ctx.moveTo(ctr[n-tr].x,ctr[n-tr].y);
    for(let k=n-tr+1;k<n;k++)ctx.lineTo(ctr[k].x,ctr[k].y);
    ctx.strokeStyle='rgba(255,255,255,'+(.3+.3*Math.sin(time*9)).toFixed(3)+')';
    ctx.lineWidth=cell*.68;ctx.stroke();
  }
  // head
  const hx=ctr[0].x,hy=ctr[0].y,r=cell*.47;
  if(mode==='play'&&!paused&&(time%2.8)<.3){ // tongue flick
    ctx.strokeStyle='#ff5d6c';ctx.lineWidth=2;ctx.lineCap='round';
    const tx=hx+dir.x*(r+cell*.1),ty=hy+dir.y*(r+cell*.1);
    const fx=hx+dir.x*(r+cell*.4),fy=hy+dir.y*(r+cell*.4);
    ctx.beginPath();ctx.moveTo(tx,ty);ctx.lineTo(fx,fy);
    ctx.moveTo(fx,fy);ctx.lineTo(fx-dir.y*cell*.12-dir.x*cell*.08,fy+dir.x*cell*.12-dir.y*cell*.08);
    ctx.moveTo(fx,fy);ctx.lineTo(fx+dir.y*cell*.12-dir.x*cell*.08,fy-dir.x*cell*.12-dir.y*cell*.08);
    ctx.stroke();
  }
  const hg=ctx.createRadialGradient(hx-r*.3,hy-r*.3,r*.1,hx,hy,r);
  hg.addColorStop(0,'#ffffff');hg.addColorStop(1,'#c3d9d5');
  ctx.beginPath();ctx.arc(hx,hy,r,0,6.283);ctx.fillStyle=hg;ctx.fill();
  ctx.lineWidth=1.5;ctx.strokeStyle='rgba(20,40,48,.35)';ctx.stroke();
  const px=-dir.y,py=dir.x;
  for(const s of[-1,1]){
    const ex=hx+dir.x*r*.32+px*s*r*.4,ey=hy+dir.y*r*.32+py*s*r*.4;
    if(mode==='over'){
      ctx.strokeStyle='#1a2f36';ctx.lineWidth=2.5;ctx.lineCap='round';
      const k=r*.17;
      ctx.beginPath();
      ctx.moveTo(ex-k,ey-k);ctx.lineTo(ex+k,ey+k);
      ctx.moveTo(ex+k,ey-k);ctx.lineTo(ex-k,ey+k);ctx.stroke();
    }else{
      ctx.fillStyle='#152b33';
      ctx.beginPath();ctx.arc(ex,ey,r*.16,0,6.283);ctx.fill();
      ctx.fillStyle='#fff';
      ctx.beginPath();ctx.arc(ex-r*.05,ey-r*.06,r*.055,0,6.283);ctx.fill();
    }
  }
  ctx.restore(); // un-clip gems/snake

  // rings
  for(const rg of rings){
    ctx.globalAlpha=clamp(rg.life,0,1)*.8;
    ctx.strokeStyle=rg.color;ctx.lineWidth=2+3*rg.life;
    ctx.beginPath();ctx.arc(rg.x,rg.y,rg.r,0,6.283);ctx.stroke();
  }
  ctx.globalAlpha=1;
  // particles
  for(const p of particles){
    const a=clamp(p.life/p.max,0,1);
    ctx.globalAlpha=a;ctx.fillStyle=p.color;
    if(p.shape==='rect'){
      ctx.save();ctx.translate(p.x,p.y);ctx.rotate(p.rot);
      ctx.fillRect(-p.size/2,-p.size/2,p.size,p.size);ctx.restore();
    }else{
      ctx.beginPath();ctx.arc(p.x,p.y,p.size*.6,0,6.283);ctx.fill();
    }
  }
  ctx.globalAlpha=1;
  // floating texts
  for(const tx of texts){
    const a=clamp(tx.life*1.5,0,1);
    ctx.globalAlpha=a;
    ctx.font='900 '+Math.round(tx.size)+'px '+"'Trebuchet MS','Segoe UI',sans-serif";
    ctx.textAlign='center';ctx.textBaseline='middle';
    ctx.lineWidth=4;ctx.strokeStyle='rgba(4,16,22,.7)';
    ctx.strokeText(tx.str,tx.x,tx.y);
    ctx.fillStyle=tx.color;ctx.fillText(tx.str,tx.x,tx.y);
  }
  ctx.globalAlpha=1;
  ctx.restore(); // un-shake

  if(vig){ctx.fillStyle=vig;ctx.fillRect(0,0,W,H);}
  if(flashA>0){ctx.fillStyle='rgba('+flashC+','+clamp(flashA,0,1).toFixed(3)+')';ctx.fillRect(0,0,W,H);}
}

/* ================= fx update ================= */
function updateFx(dt){
  for(let x=0;x<COLS;x++)for(let y=0;y<ROWS;y++){
    const g=board[x][y];if(!g)continue;
    if(g.fall>0){
      g.fall-=dt*24;
      if(g.fall<=0){g.fall=0;g.land=.001;
        const p=cellPx(x,y);
        particles.push({x:p.x+rand(-6,6),y:p.y+cell*.35,vx:rand(-14,14),vy:-26,g:60,size:2.4,life:.35,max:.35,rot:0,vr:0,color:'rgba(210,238,242,.7)',shape:'dot'});
      }
    }
    if(g.land>0&&g.land<1){g.land+=dt*4;if(g.land>=1)g.land=0;}
  }
  for(let i=particles.length-1;i>=0;i--){const p=particles[i];
    p.x+=p.vx*dt;p.y+=p.vy*dt;p.vy+=p.g*dt;p.rot+=p.vr*dt;p.life-=dt;
    if(p.life<=0)particles.splice(i,1);}
  for(let i=texts.length-1;i>=0;i--){const tx=texts[i];
    tx.y+=tx.vy*dt;tx.life-=dt/tx.dur;if(tx.life<=0)texts.splice(i,1);}
  for(let i=rings.length-1;i>=0;i--){const r2=rings[i];
    r2.r+=r2.vr*dt;r2.life-=dt/r2.dur;if(r2.life<=0)rings.splice(i,1);}
  shakeT=Math.max(0,shakeT-dt);
  flashA=Math.max(0,flashA-dt*1.3);
}

/* ================= main loop ================= */
function frame(t){
  requestAnimationFrame(frame);
  let dt=(t-lastT)/1000;lastT=t;dt=Math.min(dt,.1);
  if(mode==='play'&&!paused){
    acc+=dt*1000;ph+=dt*speedForLevel(level)*.9;
    const stepMs=1000/speedForLevel(level);
    let guard=0;
    while(acc>=stepMs&&guard++<5&&mode==='play'){acc-=stepMs;step();}
    if(comboT>0){comboT-=dt*1000;if(comboT<=0){combo=1;updateHUD();}}
  }
  updateFx(dt);
  render(t/1000);
}

function setDir(d){
  if(mode!=='play'||paused)return;
  const last=queue.length?queue[queue.length-1]:dir;
  if(d.x===last.x&&d.y===last.y)return;
  if(d.x===-last.x&&d.y===-last.y)return;
  if(queue.length<3)queue.push(d);
  sfx.tick();
}
