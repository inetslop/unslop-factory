    if(o.y>padY-6&&o.y<padY+padH+10&&Math.abs(ox-paddle.x)<paddle.w/2+14){
      o.dead=true;catchOrb(o,ox);
    }else if(o.y>CORE.y-6){o.dead=true;burst(ox,o.y,'#5a8f7a',4,90);}
  }
  orbs=orbs.filter(o=>!o.dead);
}
function catchOrb(o,ox){
  if(o.type==='coin'){credits+=20;stats.earned+=20;credFlash=.3;addText(ox,o.y,'+20 ⌁','#ffd94a');SFX.coin();burst(ox,o.y,'#ffd94a',12);}
  else if(o.type==='ball'){lives=Math.min(lives+1,4);addText(ox,o.y,'+1 ORB','#7ef0ff');SFX.orbBall();burst(ox,o.y,'#7ef0ff',12);}
  else{overdriveT=8;addText(ox,o.y,'OVERDRIVE!','#ff7ad9');SFX.overdrive();burst(ox,o.y,'#ff7ad9',14);}
}

function updateFx(dt){
  for(const p of particles){p.x+=p.vx*dt;p.y+=p.vy*dt;p.vy+=(p.grav||0)*dt;p.life-=dt;}
  particles=particles.filter(p=>p.life>0);if(particles.length>420)particles.splice(0,particles.length-420);
  for(const t of texts){t.y-=30*dt;t.life-=dt;}
  texts=texts.filter(t=>t.life>0);
  for(const r of rings){r.r+=r.vr*dt;r.life-=dt;}
  rings=rings.filter(r=>r.life>0);
  for(const b of beams)b.life-=dt;
  beams=beams.filter(b=>b.life>0);
}

function waveCleared(){
  const cB=40+wave*15,sB=200+wave*50;
  credits+=cB;stats.earned+=cB;score+=sB;credFlash=.4;
  banner={text:'WAVE '+wave+' CLEARED',sub:'+⌁'+cB+'   +'+sB+' PTS',t:1.8};
  wave++;state='wavebreak';breakT=6;SFX.clear();
}

function updateAttract(dt){
  aSpawnT-=dt;
  if(aSpawnT<=0&&aBricks.length<7){
    const types=['iron','rapid','armor','bomber','shield'];
    aBricks.push(makeBrick(types[(Math.random()*5)|0],40+Math.random()*(W-120),-40,26+Math.random()*22));
    aSpawnT=1.1+Math.random();
  }
  for(const b of aBricks){b.y+=b.vy*dt;if(b.flash>0)b.flash-=dt;}
  aBricks=aBricks.filter(b=>!b.dead&&b.y<H+50);
  aBeamT-=dt;
  if(aBeamT<=0){
    aBeamT=.45+Math.random()*.9;
    const p=pads[(Math.random()*pads.length)|0];
    let tb=aBricks.length?aBricks[(Math.random()*aBricks.length)|0]:null;
    const tx=tb?tb.x+tb.w/2:80+Math.random()*(W-160);
    const ty=tb?tb.y+tb.h/2:H*.5+Math.random()*H*.3;
    const a=Math.atan2(ty-p.y,tx-p.x);p.angle=a;
    beams.push({x1:p.x+Math.cos(a)*24,y1:p.y+Math.sin(a)*24,x2:tx,y2:ty,life:.14,maxLife:.14,col:'#8ff0ff'});
    burst(tx,ty,'#bff6ff',6,150);
    if(tb){tb.dead=true;burst(tx,ty,BTYPES[tb.type].col,10);}
  }
}

/* ------------------------------ update --------------------------------- */
function update(dt){
  time+=dt;shake=Math.max(0,shake-28*dt);
  if(overdriveT>0)overdriveT-=dt;
  if(credFlash>0)credFlash-=dt;
  if(hintT>0)hintT-=dt;
  if(banner){banner.t-=dt;if(banner.t<=0)banner=null;}
  switch(state){
    case 'menu':updateAttract(dt);break;
    case 'wavebreak':
      breakT-=dt;updatePaddle(dt);updateBall(dt);updateShots(dt);updateOrbs(dt);updateTurrets(dt);
      if(breakT<=0){state='playing';spawnWave();SFX.wave();}
      break;
    case 'playing':case 'over':
      updatePaddle(dt);updateBall(dt);updateBricks(dt);updateTurrets(dt);updateShots(dt);updateOrbs(dt);
      if(state==='playing'&&bricks.length===0)waveCleared();
      break;
  }
  updateFx(dt);
}

    ctx.globalAlpha=a*.4;ctx.strokeStyle=s.col;ctx.lineWidth=5;
    ctx.beginPath();ctx.moveTo(s.x1,s.y1);ctx.lineTo(s.x2,s.y2);ctx.stroke();
    ctx.globalAlpha=a;ctx.lineWidth=1.6;ctx.strokeStyle='#ffffff';
    ctx.beginPath();ctx.moveTo(s.x1,s.y1);ctx.lineTo(s.x2,s.y2);ctx.stroke();
  }
  ctx.globalAlpha=1;
  for(const s of shots){
    const l=Math.hypot(s.vx,s.vy)||1,ux=s.vx/l,uy=s.vy/l;
    ctx.strokeStyle=s.col;ctx.globalAlpha=.35;ctx.lineWidth=4;
    ctx.beginPath();ctx.moveTo(s.x-ux*10,s.y-uy*10);ctx.lineTo(s.x,s.y);ctx.stroke();
    ctx.globalAlpha=1;ctx.lineWidth=1.6;
    ctx.beginPath();ctx.moveTo(s.x-ux*9,s.y-uy*9);ctx.lineTo(s.x,s.y);ctx.stroke();
  }
}

function drawFx(){
  for(const p of particles){
    ctx.globalAlpha=clamp(p.life/p.maxLife,0,1);
    ctx.fillStyle=p.col;ctx.fillRect(p.x-p.size/2,p.y-p.size/2,p.size,p.size);
  }
  ctx.globalAlpha=1;
  for(const r of rings){
    const a=r.life/r.maxLife;
    ctx.globalAlpha=a;ctx.strokeStyle=r.col;ctx.lineWidth=Math.max(1,r.lw*a);
    ctx.beginPath();ctx.arc(r.x,r.y,r.r,0,7);ctx.stroke();
  }
  ctx.globalAlpha=1;
  for(const t of texts)txt(t.str,t.x,t.y,14,t.col,'mono','center',t.life/t.maxLife);
}

function drawTooltip(p){
  if(!p)return;
  const up=p.level<3;
  let title,action,aCol;
  if(p.level===0){title='EMPTY PAD';
    if(credits>=TURRET.cost[0]){action='CLICK — BUILD MK I · ⌁60';aCol='#ffb347';}
    else{action='NEED ⌁'+(TURRET.cost[0]-credits);aCol='#ff8a8a';}}
  else if(up){const c=TURRET.cost[p.level];title=TURRET.name[p.level-1]+' · DMG '+TURRET.dmg[p.level-1];
    if(credits>=c){action='CLICK — UPGRADE · ⌁'+c;aCol='#ffb347';}
    else{action='NEED ⌁'+c;aCol='#ff8a8a';}}
  else{title=TURRET.name[2]+' · MAX';action='R-CLICK SELL · +⌁'+Math.round(p.invested*.6);aCol='#9fc6d4';}

  if(p.level>0){
    ctx.setLineDash([6,6]);ctx.strokeStyle='rgba(120,230,255,.2)';ctx.lineWidth=1.5;
    ctx.beginPath();ctx.arc(p.x,p.y,TURRET.range,0,7);ctx.stroke();ctx.setLineDash([]);
  }
  const pw=224,ph=46;
  let px=p.x-pw/2,py=p.y<110?p.y+42:p.y-90;
  px=clamp(px,8,W-pw-8);py=clamp(py,70,H-ph-8);
  ctx.fillStyle='rgba(5,18,26,.94)';rr(px,py,pw,ph,4);ctx.fill();
  ctx.strokeStyle='rgba(87,230,255,.45)';ctx.lineWidth=1;ctx.stroke();
  txt(title,px+pw/2,py+15,12,'#ecfdff','mono','center');
  txt(action,px+pw/2,py+33,12,aCol,'mono','center');
}

function draw(){
  ctx.setTransform(dpr,0,0,dpr,0,0);
  const w=canvas.width/dpr,h=canvas.height/dpr;
  ctx.fillStyle='#04090f';ctx.fillRect(0,0,w,h);
  for(const s of stars){
    ctx.globalAlpha=.2+.35*Math.abs(Math.sin(time*s.sp+s.tw));
    ctx.fillStyle='#9fd8ff';ctx.fillRect(s.x*w,s.y*h,s.r,s.r);
  }
  ctx.globalAlpha=1;
  const vg=ctx.createRadialGradient(w/2,h/2,Math.min(w,h)*.35,w/2,h/2,Math.max(w,h)*.75);
  vg.addColorStop(0,'rgba(0,0,0,0)');vg.addColorStop(1,'rgba(0,0,0,.55)');
  ctx.fillStyle=vg;ctx.fillRect(0,0,w,h);

  const sx=(Math.random()*2-1)*shake,sy=(Math.random()*2-1)*shake;
  ctx.setTransform(dpr*scale,0,0,dpr*scale,dpr*(ox+sx*scale),dpr*(oy+sy*scale));
  ctx.save();ctx.beginPath();ctx.rect(-2,-2,W+4,H+4);ctx.clip();

  drawArenaBG();
  drawCore();
  let hoverPad=null;
  if(state==='playing'||state==='wavebreak')hoverPad=padAt(mouse.wx,mouse.wy);
  drawPads(hoverPad);
  drawOrbs();
  for(const b of (state==='menu'?aBricks:bricks))drawBrick(b);
  drawShotsBeams();
  drawBall();
  drawPaddle();
  drawFx();

  if(state!=='menu'){
    drawHUD();
    if(state==='wavebreak'){
      txt('WAVE '+String(wave).padStart(2,'0'),W/2,H*.34,58,'#ecfdff','disp','center');
      txt('INCOMING IN '+Math.max(0,breakT).toFixed(1),W/2,H*.34+46,18,'#ffb347','mono','center');
      if(wave===1&&breakT>1.5){
        txt('CLICK A WALL PAD TO BUILD TURRETS — CLICK AGAIN TO UPGRADE',W/2,H*.34+94,14,'#9fc6d4','mono','center');
        txt('MOVE MOUSE TO AIM PADDLE · CLICK OR SPACE LAUNCHES THE ORB',W/2,H*.34+118,14,'#9fc6d4','mono','center');
      }
    }
    if(banner){
      const a=clamp(banner.t/.4,0,1);
      txt(banner.text,W/2,H*.42,44,'#3fe0b6','disp','center',a);
      txt(banner.sub,W/2,H*.42+42,16,'#ffd94a','mono','center',a);
    }
    if(ball.stuck&&(hintT>0||state==='wavebreak')){
      txt('CLICK / SPACE TO LAUNCH',paddle.x,padY-58,14,'#7ef0ff','mono','center',.5+.5*Math.sin(time*6));
    }
    drawTooltip(hoverPad);
  }else{
    txt('SYSTEM STANDBY — AWAITING DEPLOYMENT',W/2,34,13,'#5f93a6','mono','center',.5+.4*Math.sin(time*2));
  }
  ctx.restore();

  // arena frame
  ctx.setTransform(dpr*scale,0,0,dpr*scale,dpr*(ox+sx*scale),dpr*(oy+sy*scale));
  ctx.strokeStyle='rgba(87,230,255,.4)';ctx.lineWidth=2;ctx.strokeRect(1,1,W-2,H-2);
  ctx.strokeStyle='rgba(87,230,255,.12)';ctx.lineWidth=6;ctx.strokeRect(-4,-4,W+8,H+8);

  canvas.style.cursor=hoverPad?'pointer':((state==='playing'||state==='wavebreak')?'crosshair':'default');
}

let last=performance.now();
function frame(now){
  requestAnimationFrame(frame);
  let dt=(now-last)/1000;last=now;
  if(dt>0.05)dt=0.05;
  if(state!=='paused')update(dt*(state==='over'?0.35:1));
  draw();
}
