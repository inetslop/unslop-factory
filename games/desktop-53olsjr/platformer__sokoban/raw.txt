function clearance(b,dir){

  const r0=Math.floor((b.y+2)/T), r1=Math.floor((b.y+b.h-2)/T);
  if(dir>0){
    for(let r=r0;r<=r1;r++){

      while(c<cols && !solidAt(c,r)) c++;
      if(c<cols) best=Math.min(best,c*T-(b.x+b.w));
    }
    best=Math.min(best,WW-b.w-b.x);
  }else{
    for(let r=r0;r<=r1;r++){
      let c=Math.floor((b.x-0.5)/T)-1;
      while(c>=0 && !solidAt(c,r)) c--;
      if(c>=0) best=Math.min(best,b.x-(c+1)*T);
    }
    best=Math.min(best,b.x);
  }
  for(const o of boxes){
    if(o===b) continue;
    const yOv=Math.min(b.y+b.h,o.y+o.h)-Math.max(b.y,o.y);
    if(yOv<3) continue;

    else if(dir<0 && o.x+o.w<=b.x+1) best=Math.min(best,b.x-(o.x+o.w));
  }
  return Math.max(0,best);
}

}

function stepBoxes(dt){
  for(const b of boxes){
    b.snapT=Math.max(0,b.snapT-dt);
    const wasG=b.grounded, vy0=b.vy;
    if(!b.pushedThis){
      b.vx*=Math.pow(b.grounded?0.0001:0.000006,dt);
      if(Math.abs(b.vx)<3&&b.grounded) b.vx=0;
      const dx=b.vx*dt;
      if(dx!==0){
        const dir=Math.sign(dx), free=clearance(b,dir);
        let m=Math.min(Math.abs(dx),free)*dir;
        if(m!==dx){
          if(Math.abs(b.vx)>150 && b.grounded){
            for(const o of boxes){
              if(o===b) continue;
              const yOv=Math.min(b.y+b.h,o.y+o.h)-Math.max(b.y,o.y);
              if(yOv<4) continue;
              if(dir>0&&o.x>=b.x+b.w-1&&(o.x-(b.x+b.w))<8){ o.vx=dir*Math.abs(b.vx)*0.55; sfx.bonk(); }
              else if(dir<0&&o.x+o.w<=b.x+1&&(b.x-(o.x+o.w))<8){ o.vx=dir*Math.abs(b.vx)*0.55; sfx.bonk(); }
            }
          }
          b.vx=0; b.x+=m;
        }else b.x+=dx;
      }
    }
    if(b.grounded&&Math.abs(b.vx)<12) b.pushArmed=false;
    // vertical
    const dy=b.vy*dt, ny=b.y+dy; let hitY=null,bonkY=null;
    const gs=gridStopY(b,ny,dy);
    if(gs){ if(gs.dir==='down') hitY=(hitY===null)?gs.y:Math.min(hitY,gs.y); else bonkY=(bonkY===null)?gs.y:Math.max(bonkY,gs.y); }
    for(const o of boxes){
      if(o===b) continue;
      const ovx=Math.min(b.x+b.w,o.x+o.w)-Math.max(b.x,o.x);
      if(ovx<=2) continue;
      const topD=(b.y+b.h)-o.y, botD=o.y+o.h-b.y;
      if(dy>=0&&topD>0&&topD<14) hitY=(hitY===null)?(o.y-b.h):Math.min(hitY,o.y-b.h);
      else if(dy<0&&botD>0&&botD<14) bonkY=(bonkY===null)?(o.y+o.h):Math.max(bonkY,o.y+o.h);
    }
    b.grounded=false;
    if(dy>=0&&hitY!==null){
      ny=hitY; b.vy=0; b.grounded=true;
      if(!wasG&&vy0>280){ sfx.thud(); dust(b.x+b.w/2,b.y+b.h,7); if(vy0>520) shake(3); }
    }else if(dy<0&&bonkY!==null){ ny=bonkY; b.vy=0; }
    b.y=ny;
  }
}

function stepPlayer(dt){
  const p=player;
  p.squashT=Math.max(0,p.squashT-dt);
  p.blink-=dt; if(p.blink<-0.12) p.blink=2+Math.random()*3;
  blockCd=Math.max(0,blockCd-dt);
  const canAct=(state==='play');
  const dir=canAct?((keys.right?1:0)-(keys.left?1:0)):0;
  if(dir!==0){ p.face=dir; p.vx+=(dir*RUN-p.vx)*Math.min(1,dt*(p.grounded?16:10)); }
  else{ p.vx*=Math.pow(p.grounded?0.00005:0.03,dt); if(Math.abs(p.vx)<4)p.vx=0; }
  if(canAct&&p.jbuf>0&&(p.grounded||p.coyote>0)){
    p.vy=-JV; p.grounded=false; p.coyote=0; p.jbuf=0; sfx.jump(); dust(p.x+p.w/2,p.y+p.h,5);
  }
  if(canAct&&!keys.jump&&p.vy<-170) p.vy=-170;
  p.vy=Math.min(900,p.vy+G*dt);
  // carried by a box
  if(p.carryBox&&Math.abs(p.carryBox.vx)>1){ moveXGrid(p,p.carryBox.vx*dt*0.85); }
  // X movement vs grid
  const dx=p.vx*dt; moveXGrid(p,dx);
  p.pushing=null;
  for(const b of boxes){
    const ovx=Math.min(p.x+p.w,b.x+b.w)-Math.max(p.x,b.x);
    const ovy=Math.min(p.y+p.h,b.y+b.h)-Math.max(p.y,b.y);
    if(ovx<=0.3||ovy<=2) continue;
    const side=(b.x+b.w/2)>(p.x+p.w/2)?1:-1;
    const canPush=canAct&&dir===side&&Math.abs(p.vx)>30&&b.grounded;
    if(canPush){
      const free=clearance(b,side);
      const m=Math.min(ovx,free);
      if(m>0.05){
        if(Math.abs(b.vx)<12&&!b.pushArmed){ b.pushArmed=true; snapshot(); pushes++; sfx.pushStart(); updateHUD(true); }
        b.x+=side*m; b.pushedThis=true; b.vx=side*Math.max(80,Math.abs(p.vx)); p.pushing=b;
        const d2=ovx-m;
        if(d2>0.5){ p.x-=side*d2; p.vx=0; }
      }else{
        p.x-=side*ovx; p.vx=0;
        if(blockCd<=0){ sfx.block(); blockCd=0.25; dust(p.x+(dir>0?p.w:0),p.y+p.h,3,20); }
      }
    }else{
      p.x-=side*ovx;
      if(canAct&&side===dir&&Math.abs(p.vx)>30&&blockCd<=0){ sfx.block(); blockCd=0.25; }
    }
  }
  // Y movement
  const vy0=p.vy, dy=p.vy*dt; let ny=p.y+dy, hitY=null,bonkY=null,carry=null;
  const gs=gridStopY(p,ny,dy);
  if(gs){ if(gs.dir==='down') hitY=(hitY===null)?gs.y:Math.min(hitY,gs.y); else bonkY=(bonkY===null)?gs.y:Math.max(bonkY,gs.y); }
  for(const b of boxes){
    const ovx=Math.min(p.x+p.w,b.x+b.w)-Math.max(p.x,b.x);
    if(ovx<=2) continue;
    const topD=(p.y+p.h)-b.y, botD=b.y+b.h-p.y;
    if(dy>=0&&topD>0&&topD<14){ const yv=b.y-p.h; hitY=(hitY===null)?yv:Math.min(hitY,yv); carry=b; }
    else if(dy<0&&botD>0&&botD<14) bonkY=(bonkY===null)?(b.y+b.h):Math.max(bonkY,b.y+b.h);
  }
  p.grounded=false;
  if(dy>=0&&hitY!==null){
    ny=hitY; p.vy=0; p.grounded=true; p.carryBox=carry;
    if(vy0>260){ sfx.land(vy0); dust(p.x+p.w/2,p.y+p.h,Math.min(9,vy0/120|0)); if(vy0>640) shake(5); p.squashT=0.16; }
  }else if(dy<0&&bonkY!==null){ ny=bonkY; p.vy=0; sfx.bonk(); p.carryBox=null; }
  else{ p.grounded=false; p.carryBox=null; }
  p.y=ny;
  if(p.grounded) p.runPhase+=Math.abs(p.vx)*dt*0.09;
  if(p.grounded&&Math.abs(p.vx)>140){ p.dustT-=dt; if(p.dustT<=0){ p.dustT=0.13; dust(p.x+p.w/2-p.face*8,p.y+p.h,2,20); } }
}

function padLogic(dt){
  for(const b of boxes){
    b.onPadNow=null;
    if(b.grounded&&Math.abs(b.vx)<60){
      const cx=b.x+b.w/2, bottom=b.y+b.h;
      for(const pad of pads){
        const pcx=pad.c*T+T/2, ptop=(pad.r+1)*T;
        if(Math.abs(cx-pcx)<17&&Math.abs(bottom-ptop)<9){ b.onPadNow=pad; break; }
      }
    }
    if(b.onPadNow){
      const tx=b.onPadNow.c*T+3, ddx=tx-b.x;
      b.x+=ddx*Math.min(1,dt*16);
      if(Math.abs(ddx)<0.4) b.x=tx;
    }
  }
  for(const pad of pads){
    let has=false; for(const b of boxes) if(b.onPadNow===pad) has=true;
    const was=pad.has; pad.has=has;
    if(has&&!was){
      sfx.snap(); sparkle(pad);
      const b=boxes.find(x=>x.onPadNow===pad); if(b) b.snapT=0.28;
    }
  }
}

function checkDeath(msg){
  player=null; // temp guard not needed
  if(state!=='play') return;
  const p=player;
  let dead = p.y>WH+120;
  if(!dead) for(const b of boxes) if(b.y>WH+300){dead=true;break;}
  if(dead){ sfx.splat(); shake(8); deathFlash=0.5; loadLevel(li,true); state='play'; toast(msg); }
}

function step(dt){
  if(state==='play') time+=dt;
  for(const b of boxes) b.pushedThis=false;
  stepBoxes(dt);
  stepPlayer(dt);
  padLogic(dt);
  checkDeath('SPLAT!');
  if(state==='play'&&pads.length&&pads.every(p=>p.has)){
    state='winAnim'; winT=0.95; sfx.win(); confettiBursts();
  }
  if(state==='winAnim'){ winT-=dt; if(winT<=0){ showClear(); state='clear'; } }
  if(AC&&pushGain){
    const tgt=(state==='play'&&player.pushing)?Math.min(0.26,0.1+Math.abs(player.vx)/900):0;
    pushGain.gain.setTargetAtTime(tgt,AC.currentTime,0.06);
  }
}

function drawTiles(){
  const tints=['#3a4f60','#3b5163','#394e5f','#3a4f62','#384d5e'];
  for(let r=0;r<rows;r++) for(let c=0;c<cols;c++){
    if(!solidAt(c,r)) continue;
    const x=c*T,y=r*T;
    ctx.fillStyle=tints[(c*7+r*13)%5]; ctx.fillRect(x,y,T,T);
    ctx.strokeStyle='rgba(0,0,0,.2)'; ctx.lineWidth=1; ctx.strokeRect(x+3.5,y+3.5,T-7,T-7);
    if(!solidAt(c,r-1)){
      ctx.fillStyle='#5c7890'; ctx.fillRect(x,y,T,5);
      ctx.fillStyle='rgba(255,255,255,.16)'; ctx.fillRect(x,y,T,1.5);
    }else{
      ctx.fillStyle='rgba(0,0,0,.18)'; ctx.fillRect(x,y,T,1);
    }
    if(!solidAt(c,r+1)){
      ctx.save(); ctx.beginPath(); ctx.rect(x,y+T-6,T,6); ctx.clip();
      ctx.fillStyle='#232f38'; ctx.fillRect(x,y+T-6,T,6);
      ctx.fillStyle='#ffd23e';
      for(let i=-1;i<T/8+1;i++){ const sx=x+i*8;
        ctx.beginPath(); ctx.moveTo(sx,y+T); ctx.lineTo(sx+3,y+T); ctx.lineTo(sx+9,y+T-6); ctx.lineTo(sx+6,y+T-6); ctx.closePath(); ctx.fill(); }
      ctx.restore();
    }
    ctx.fillStyle='rgba(0,0,0,.15)';
    if(!solidAt(c-1,r)) ctx.fillRect(x,y,1.5,T);
    if(!solidAt(c+1,r)) ctx.fillRect(x+T-1.5,y,1.5,T);
    ctx.fillStyle='#22303c';
    const riv=[[x+4.5,y+4.5],[x+T-4.5,y+4.5]];
    if(solidAt(c,r+1)){ riv.push([x+4.5,y+T-4.5],[x+T-4.5,y+T-4.5]); }
    for(const p of riv){ ctx.beginPath(); ctx.arc(p[0],p[1],1.6,0,6.29); ctx.fill(); }
  }
}
function drawPads(t){
  pads.forEach((pad,i)=>{
    const cx=pad.c*T+T/2, cy=pad.r*T+T/2;
    const pulse=0.5+0.5*Math.sin(t*3+i*1.7);
    const col=pad.has?'111,232,169':'255,179,71';
    const g=ctx.createRadialGradient(cx,cy,2,cx,cy,18);
    g.addColorStop(0,'rgba('+col+','+(0.16+0.2*pulse)+')'); g.addColorStop(1,'rgba('+col+',0)');
    ctx.fillStyle=g; ctx.beginPath(); ctx.arc(cx,cy,18,0,6.29); ctx.fill();
    ctx.strokeStyle='rgba('+col+','+(0.55+0.4*pulse)+')'; ctx.lineWidth=2;
    ctx.setLineDash([5,4]); ctx.lineDashOffset=-t*14;
    ctx.beginPath(); ctx.arc(cx,cy,9.5,0,6.29); ctx.stroke();
    ctx.setLineDash([]);
    ctx.fillStyle='rgba('+col+',.85)'; ctx.beginPath(); ctx.arc(cx,cy,3,0,6.29); ctx.fill();
  });
}
function drawBox(b,t){
  if(b.grounded){ ctx.fillStyle='rgba(0,0,0,.28)'; ctx.fillRect(b.x-2,b.y+b.h-2.5,34,4); }
  const s=1+(b.snapT>0?Math.sin((0.28-b.snapT)/0.28*Math.PI)*0.12:0);
  ctx.save();
  ctx.translate(b.x+15,b.y+15); ctx.scale(s,s); ctx.translate(-15,-15);
  ctx.fillStyle='#c85a28'; rr(ctx,0,0,30,30,3); ctx.fill();
  ctx.fillStyle='rgba(90,32,8,.35)';
  for(let i=1;i<6;i++) ctx.fillRect(i*5-1,4,2,22);
  ctx.fillStyle='#e8793c'; ctx.fillRect(0,0,30,5);
  ctx.fillStyle='rgba(60,20,5,.4)'; ctx.fillRect(0,26,30,4);
  ctx.strokeStyle='#5f2a10'; ctx.lineWidth=2; rr(ctx,1,1,28,28,2.5); ctx.stroke();
  ctx.fillStyle='#f2b45c';
  [[2,2],[25,2],[2,25],[25,25]].forEach(p=>ctx.fillRect(p[0],p[1],3,3));
  ctx.fillStyle='rgba(255,236,200,.9)';
  ctx.font='bold 8px "Courier New",monospace'; ctx.textAlign='center'; ctx.textBaseline='middle';
  ctx.fillText('SOKO',15,21);
  ctx.fillStyle='#ffd23e'; ctx.beginPath(); ctx.arc(7,8,4.6,0,6.29); ctx.fill();
  ctx.fillStyle='#5f2a10'; ctx.font='bold 7px "Courier New",monospace';
  ctx.fillText(b.tag,7,8.5);
  if(b.onPadNow){
    ctx.strokeStyle='rgba(111,232,169,'+(0.45+0.4*Math.sin(t*5))+')';
    ctx.lineWidth=2.5; rr(ctx,-3,-3,36,36,5); ctx.stroke();
  }
  ctx.restore();
}
function drawPlayer(t){
  const p=player; if(!p) return;
  const f=p.face;
  let sx=1,sy=1;
  if(p.squashT>0){ const k=p.squashT/0.16; sy=1-0.26*k; sx=1+0.3*k; }
  else if(!p.grounded){ const k=Math.min(0.14,Math.abs(p.vy)/2800); sy=1+k; sx=1-k*0.7; }
  if(state==='menu') sy*=1+0.02*Math.sin(t*2.4);
  const cx=p.x+p.w/2, feet=p.y+p.h;
  ctx.fillStyle='rgba(0,0,0,.3)'; ctx.beginPath(); ctx.ellipse(cx,feets===p.y+p.h?feet:feet,9,2.6,0,0,6.29); ctx.fill();
  ctx.save();
  ctx.translate(cx,feet); ctx.scale(sx,sy);
  if(p.pushing) ctx.rotate(f*0.12);
  const run=p.grounded&&Math.abs(p.vx)>30?Math.sin(p.runPhase):0;
  // legs + boots
  ctx.fillStyle='#24313c';
  if(p.grounded){
    ctx.fillRect(-7+run*3,-8,4.5,8); ctx.fillRect(2.5-run*3,-8,4.5,8);
    ctx.fillStyle='#151f28'; ctx.fillRect(-7+run*3-1,-2.5,6,2.5); ctx.fillRect(2.5-run*3,-2.5,6,2.5);
  }else{
    ctx.fillRect(-6,-7,4.5,6); ctx.fillRect(1.5,-9,4.5,6);
  }
  // arms
  ctx.fillStyle='#c9611f';
  if(p.pushing){ ctx.fillRect(f>0?6:-15,-19,9,3.2); ctx.fillRect(f>0?6:-15,-14.5,9,3.2); }
  else{ const sw=p.grounded?run*3:2; ctx.fillRect(f>0?4:-8+sw,-18,4,8); }
  // vest
  ctx.fillStyle='#ff7f2a'; rr(ctx,-9,-20,18,12.5,3); ctx.fill();
  ctx.fillStyle='rgba(255,224,138,.9)'; ctx.fillRect(-9,-14.5,18,3);
  // head
  ctx.fillStyle='#f2c9a0'; rr(ctx,-7,-30,14,10.5,2.5); ctx.fill();
  const blink=p.blink<0;
  ctx.fillStyle='#1c2a33';
  if(blink) ctx.fillRect(f>0?1:-5,-26,f>0?4:4,0.8); else ctx.fillRect(f>0?2.5:-4.5,-27,2.4,2.6);
  // hard hat
  ctx.fillStyle='#ffd23e';
  ctx.beginPath(); ctx.arc(0,-29.5,7.6,Math.PI,0); ctx.fill();
  ctx.fillRect(f>0?-2:-8,-30,10,2.6);
  ctx.fillStyle='rgba(140,90,10,.35)'; ctx.fillRect(-7,-29.5,14,1.4);
  ctx.restore();
}
function drawParticles(){
  for(const p of particles){
    const a=Math.max(0,p.life/p.max);
    if(p.t==='d'){ ctx.fillStyle='rgba(190,215,225,'+(a*0.5)+')'; ctx.beginPath(); ctx.arc(p.x,p.y,p.sz*a+0.5,0,6.29); ctx.fill(); }
    else if(p.t==='s'){ ctx.fillStyle=p.col; ctx.globalAlpha=a; ctx.fillRect(p.x-1,p.y-1,2.4,2.4); ctx.globalAlpha=1; }
    else{ ctx.save(); ctx.translate(p.x,p.y); ctx.rotate(p.rot); ctx.globalAlpha=Math.min(1,a*2); ctx.fillStyle=p.col; ctx.fillRect(-p.sz/2,-p.sz/3,p.sz,p.sz*0.66); ctx.restore(); ctx.globalAlpha=1; }
  }
}
function render(t){
  ctx.setTransform(dpr,0,0,dpr,0,0);
  drawBG(t);
  // world space
  let shx=0,shy=0;
  if(shakeT>0){ shakeT-=1/60; const m=shakeMag*Math.max(0,shakeT)/0.3; shx=(Math.random()-0.5)*2*m; shy=(Math.random()-0.5)*2*m; }
  ctx.save();
  ctx.translate(ox+shx,oy+shy); ctx.scale(S,S);
  // dock frame
  ctx.fillStyle='rgba(5,13,18,.78)'; rr(ctx,-16,-16,WW+32,WH+32,10); ctx.fill();
  ctx.strokeStyle='rgba(255,179,71,.14)'; ctx.lineWidth=1.5; rr(ctx,-16,-16,WW+32,WH+32,10); ctx.stroke();
  ctx.fillStyle='#2c4351';
  [[-8,-8],[WW+8,-8],[-8,WH+8],[WW+8,WH+8]].forEach(p=>{ctx.beginPath();ctx.arc(p[0],p[1],3.2,0,6.29);ctx.fill();});
  drawTiles();
  drawPads(t);
  for(const b of boxes) drawBox(b,t);
  drawPlayer(t);
  // ambient motes
  ctx.fillStyle='rgba(200,230,240,.14)';
  for(const m of motes){
    const mx=m.x+Math.sin(t*0.5+m.ph)*6, my=(m.y - t*m.sp%WH+WH)%WH;
    ctx.beginPath(); ctx.arc(mx,my,m.sz,0,6.29); ctx.fill();
  }
  drawParticles();
  ctx.restore();
  if(deathFlash>0){ deathFlash=Math.max(0,deathFlash-1/60*1.4); ctx.fillStyle='rgba(255,70,50,'+(deathFlash*0.35)+')'; ctx.fillRect(0,0,cw,ch); }
  // hud time tick
  if(state==='play'){ if(Math.floor(time)!==lastSec){ lastSec=Math.floor(time); hTime.textContent=fmt(time); } }
}
let lastSec=-1;

/* ================= MAIN LOOP ================= */
let last=performance.now(), acc=0; const STEP=1/60;
function frame(now){
  requestAnimationFrame(frame);
  let dt=(now-last)/1000; last=now; if(dt>0.1) dt=0.1;
  if(state==='play'||state==='winAnim'){ acc+=dt; let n=0; while(acc>=STEP&&n<4){ step(STEP); acc-=STEP; n++; } }
  updateParticles(Math.min(dt,0.05));
  render(now/1000);
}

loadLevel(0,true);
state='menu'; showOv('ovStart'); updateRecLine();
requestAnimationFrame(frame);
})();
