function burst(x,y){const cols=['#ff8c42','#ffe6c9','#ff5a5f','#ffffff'];
  for(let i=0;i<28;i++){const a=rand(0,Math.PI*2),v=rand(120,430);
    parts.push({x,y,vx:Math.cos(a)*v,vy:Math.sin(a)*v-120,t:0,life:rand(.5,.95),r:rand(3,6.5),c:cols[i%4],layer:1});}}
function spark(x,y,n,col){for(let i=0;i<n;i++){const a=rand(0,Math.PI*2),v=rand(60,220);
  parts.push({x,y,vx:Math.cos(a)*v,vy:Math.sin(a)*v,t:0,life:rand(.3,.55),r:rand(1.5,3.4),c:col,layer:1});}}
function updateParts(wdt){for(let i=parts.length-1;i>=0;i--){const p=parts[i];p.t+=wdt;
  if(p.c==='dust')p.vy+=260*wdt;else if(p.c!=='fire')p.vy+=520*wdt;
  p.x+=p.vx*wdt;p.y+=p.vy*wdt;if(p.rot!==undefined)p.rot+=(p.vr||0)*wdt;
  if(p.t>p.life||p.x<dist-PSX-120)parts.splice(i,1);}
  for(let i=floats.length-1;i>=0;i--){const f=floats[i];f.t+=wdt;f.y-=34*wdt;if(f.t>.85)floats.splice(i,1);}}
function ambient(wdt){ambT-=wdt;if(ambT>0)return;ambT=rand(.12,.3);
  const x=dist+VW-PSX+50;
  if(PAL.night>.5)parts.push({x,y:rand(60,GY-40),vx:-(speed*.4)+rand(-15,5),vy:rand(-9,9),t:0,life:rand(2,3.6),r:rand(1.6,3),c:'fire',layer:1});
  else parts.push({x,y:rand(30,GY-70),vx:-(speed*.55)+rand(-20,8),vy:rand(-8,12),t:0,life:rand(2,3.6),r:rand(2,3.4),c:'leaf',rot:rand(0,6),vr:rand(-2.5,2.5),layer:1});}

function boxHit(x,y,w,h){return dist-14<x+w&&dist+14>x&&P.y+6<y+h&&P.y+FH-1>y;}
function circRect(cx,cy,r,rx,ry,rw,rh){const nx=clamp(cx,rx,rx+rw),ny=clamp(cy,ry,ry+rh),dx=cx-nx,dy=cy-ny;return dx*dx+dy*dy<r*r;}

function update(wdt,rdt){time+=rdt;animT+=wdt;calcBiome();
  if(state==='menu'){dist+=speed*wdt;ensureGround(dist+VW+800);P.ph+=speed*wdt*.045;runDust(wdt);ambient(wdt);updateParts(wdt);return;}
  if(state==='over'){dist+=90*wdt;ensureGround(dist+VW+800);ambient(wdt);updateParts(wdt);flash=Math.max(0,flash-rdt*2);shake*=Math.max(0,1-rdt*5);return;}
  if(state==='dying'){dieT+=rdt;ts=Math.max(.12,1-dieT*2.6);dist+=speed*wdt;updateParts(wdt);
    flash=Math.max(0,flash-rdt*2.4);shake*=Math.max(0,1-rdt*5);
    if(dieT>1.05)finish();return;}

  /* --- play --- */
  const target=Math.min(760,330+meters*.42);speed+=(target-speed)*Math.min(1,wdt*1.5);
  dist+=speed*wdt;meters=dist/12;
  if(meters>=nextMs){showBanner(nextMs.toLocaleString('en-US')+' m!');sfx.ms();nextMs+=500;}

  P.buf-=wdt;P.coyote-=wdt;
  if(P.buf>0){if(P.onG||P.coyote>0)jump();else if(P.jumps<2)djump();}
  const impact=P.vy;
  P.vy=Math.min(1500,P.vy+GRAV*wdt);
  const prevFoot=P.y+FH;P.y+=P.vy*wdt;const foot=P.y+FH;
  let landY=null;
  if(P.vy>=0){const gt=groundAt(dist);
    if(gt!==null&&prevFoot<=gt+8&&foot>=gt)landY=gt;
    if(landY===null)for(const p of plats)if(dist>p.x-4&&dist<p.x+p.w+4&&prevFoot<=p.y+10&&foot>=p.y){landY=p.y;break;}}
  if(landY!==null){
    if(!P.onG&&impact>430){P.sq=.6;dust(dist,landY,9,Math.min(1.4,impact/900));sfx.land(Math.min(1,impact/1200));
      if(impact>950)shake=Math.min(6,(impact-850)/140);}
    P.y=landY-FH;P.vy=0;P.onG=true;P.jumps=0;}
  else if(P.onG){let sup=false;const gt=groundAt(dist);
    if(gt!==null&&Math.abs(foot-gt)<6)sup=true;
    if(!sup)for(const p of plats)if(dist>p.x-4&&dist<p.x+p.w+4&&Math.abs(foot-p.y)<6){sup=true;break;}
    if(!sup){P.onG=false;P.coyote=.1;}}

  P.ph+=speed*wdt*.045;P.sq+=(1-P.sq)*Math.min(1,wdt*9);
  P.blink-=wdt;if(P.blink<=0){P.blink=rand(2,4.2);P.blk=.13;}P.blk=Math.max(0,P.blk-wdt);
  runDust(wdt);ambient(wdt);updateParts(wdt);musicTick(rdt);

  while(cursorX<dist+VW+700)spawnPattern();
  const cut=dist-520;
  segs=segs.filter(s=>s.x+s.w>cut);plats=plats.filter(p=>p.x+p.w>cut);
  spikes=spikes.filter(s=>s.x>cut);drones=drones.filter(d=>d.x>cut);gems=gems.filter(g=>g.x>cut-150);

  for(let i=gems.length-1;i>=0;i--){const g=gems[i],dx=dist-g.x,dy=(P.y+FH/2)-g.y;
    if(dx*dx+dy*dy<34*34){gems.splice(i,1);gemCount++;sfx.coin();spark(g.x,g.y,7,PAL.acc);
      floats.push({x:g.x,y:g.y-14,t:0,txt:'+25'});}}

  for(const s of spikes)if(boxHit(s.x-9,GY-s.h+6,18,s.h-6)){die('hit');return;}
  for(const d of drones){const dy=d.y+Math.sin(animT*2+d.bob)*d.amp,
      dx=d.x+(d.dx?Math.sin(animT*.7+d.bob)*d.dx:0);
    if(circRect(dx,dy,15,dist-14,P.y+6,28,FH-7)){die('hit');return;}}
  for(const p of plats)if(p.solid&&boxHit(p.x+3,p.y+10,p.w-6,GY-p.y-10)&&P.y+FH>p.y+12){die('hit');return;}
  if(foot>VH+80){die('pit');return;}
  flash=Math.max(0,flash-rdt*2.4);shake*=Math.max(0,1-rdt*5);}
