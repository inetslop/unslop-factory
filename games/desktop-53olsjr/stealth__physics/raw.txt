/* ============ geometry ============ */
function segRect(x1,y1,x2,y2,r){
 const dx=x2-x1,dy=y2-y1;let tmin=0,tmax=1;
 const test=(pp,q)=>{if(Math.abs(pp)<1e-9)return q>=0;const t=q/pp;
  if(pp>0){if(t>tmin)tmin=t;}else{if(t<tmax)tmax=t;}return tmin<=tmax;};
 return test(-dx,x1-r.x)&&test(dx,r.x+r.w-x1)&&test(-dy,y1-r.y)&&test(dy,r.y+r.h-y1);
}
function segBlocked(x1,y1,x2,y2){for(const w of walls)if(segRect(x1,y1,x2,y2,w))return true;return false;}
function closestD2(px,py,w){const cx=clamp(px,w.x,w.x+w.w),cy=clamp(py,w.y,w.y+w.h);const dx=px-cx,dy=py-cy;return dx*dx+dy*dy;}
function visPoly(x,y,R,cn,half){
 const pts=[[-half,R],[half,R]];
 for(const w of walls){
  if(closestD2(x,y,w)>(R+60)*(R+60))continue;
  const cs=[[w.x,w.y],[w.x+w.w,w.y],[w.x+w.w,w.y+w.h],[w.x,w.y+w.h]];
  for(const cp of cs){const dx=cp[0]-x,dy=cp[1]-y,d=Math.hypot(dx,dy);
   if(d>R+50)continue;
   const rel=angDiff(cn,Math.atan2(dy,dx));
   if(Math.abs(rel)<=half+.3)pts.push([rel,Math.min(d,R)]);}
 }
 pts.sort((a,b)=>a[0]-b[0]);
 return pts.map(pt=>[x+Math.cos(cn+pt[0])*pt[1],y+Math.sin(cn+pt[0])*pt[1]]);
}
function collide(o){
 let imp=0;
 for(let pass=0;pass<2;pass++)for(const w of walls){
  const cx=clamp(o.x,w.x,w.x+w.w),cy=clamp(o.y,w.y,w.y+w.h);
  let dx=o.x-cx,dy=o.y-cy;const d2=dx*dx+dy*dy;
  if(d2<o.r*o.r){
   const d=Math.sqrt(d2);
   if(d<.0001){
    const l=o.x-w.x,r2=w.x+w.w-o.x,t=o.y-w.y,b=w.y+w.h-o.y,m=Math.min(l,r2,t,b);
    if(m===l)o.x=w.x-o.r;else if(m===r2)o.x=w.x+w.w+o.r;else if(m===t)o.y=w.y-o.r;else o.y=w.y+w.h+o.r;
   }else{
    const nx=dx/d,ny=dy/d;o.x=cx+nx*o.r;o.y=cy+ny*o.r;
    const vn=o.vx*nx+o.vy*ny;
    if(vn<0){o.vx-=vn*nx*1.35;o.vy-=vn*ny*1.35;if(-vn>imp)imp=-vn;}
   }
  }
 }
 return imp>0?imp:null;
}
function lightFactor(x,y,t){
 let l=0;
 for(const L of lights){const dx=x-L.x,dy=y-L.y,d2=dx*dx+dy*dy,R=L.r;
  if(d2>R*R)continue;
  if(segBlocked(L.x,L.y,x,y))continue;
  const d=Math.sqrt(d2);l+=L._i*Math.pow(1-d/R,1.4)*1.25;}
 return clamp(.25+clamp(l,0,1)*1.05,.25,1);
}

function updatePlayer(dt){
 let ax=0,ay=0;

   pingAt(p.x-p.vx/(sp||1)*p.cr,p.y-p.vy/(sp||1)*p.cr,clamp((hit-80)/600,.15,1));}

function updateGuards(dt){
 for(const g of guards){
  g.cool=Math.max(0,g.cool-dt);
  let tx=g.x,ty=g.y,spd=0,turn=2.4;
  if(g.state==='patrol'){
   const w=g.wps[g.wi];tx=w[0];ty=w[1];spd=85;turn=2.2;
   if(Math.hypot(tx-g.x,ty-g.y)<16){
    if(g.wps.length===2){g.dir*=-1;g.wi=(g.wi+g.dir+2)%2;}else g.wi=(g.wi+1)%g.wps.length;}
  }else if(g.state==='investigate'){
   tx=g.invest.x;ty=g.invest.y;spd=150;turn=3.4;
   if(Math.hypot(tx-g.x,ty-g.y)<20){g.state='search';g.st=1.7;}
  }else if(g.state==='search'){
   g.st-=dt;g.ang+=2.6*dt;spd=0;
   if(g.st<=0){g.state='patrol';let bi=0,bd=1e9;
    g.wps.forEach((w,i)=>{const d=Math.hypot(w[0]-g.x,w[1]-g.y);if(d<bd){bd=d;bi=i;}});g.wi=bi;}
  }else if(g.state==='chase'){tx=p.x;ty=p.y;spd=252;turn=4.2;}
  if(spd>0){
   const des=Math.atan2(ty-g.y,tx-g.x);

   g.x+=Math.cos(g.ang)*spd*dt;g.y+=Math.sin(g.ang)*spd*dt;collide(g);
  }
 }
 for(let i=0;i<guards.length;i++)for(let j=i+1;j<guards.length;j++){
  const a=guards[i],b=guards[j];let dx=b.x-a.x,dy=b.y-a.y,d=Math.hypot(dx,dy);

 }
}
function detection(dt){

 for(const g of guards)g.seen=false;
 const sp=Math.hypot(p.vx,p.vy);
 for(const g of guards){
  const dx=p.x-g.x,dy=p.y-g.y,d=Math.hypot(dx,dy);
  if(d<VIS.range&&d>1){
   const da=Math.abs(angDiff(g.ang,Math.atan2(dy,dx)));
   if(da<VIS.half&&!segBlocked(g.x,g.y,p.x,p.y)){
    g.seen=true;seenNow=true;
    const lf=lightFactor(p.x,p.y,sceneT);
    rate=Math.max(rate,(0.8+0.9*(1-d/VIS.range))*lf*(p.crouchNow?.45:1)*(.85+.35*Math.min(1,sp/260)));
   }
  }
  if(d<g.r+p.cr+3){catchGame('tagged');return;}
 }
 if(seenNow){
  meter=Math.min(1,meter+rate*dt);
  if(!th35&&meter>.35){warnBeep(false);th35=true;}
  if(!th7&&meter>.7){warnBeep(true);th7=true;}
  hbT-=dt;if(hbT<=0){thudS(.22+meter*.15);hbT=lerp(.5,.16,meter);}
 }else{
  meter=Math.max(0,meter-.55*dt);
  if(meter<.2)th35=false;if(meter<.5)th7=false;
 }
 if(meter>.7&&!chasing){chasing=true;guards.forEach(g=>{if(g.state!=='chase')g.state='chase';});}
 else if(meter<.32&&chasing){chasing=false;guards.forEach(g=>{if(g.state==='chase'){g.state='patrol';
  let bi=0,bd=1e9;g.wps.forEach((w,i)=>{const d=Math.hypot(w[0]-g.x,w[1]-g.y);if(d<bd){bd=d;bi=i;}});g.wi=bi;}});}
 if(meter>=1)catchGame('detected');
}
