```js
function genMaze(cols,rows){
  const W=[]; for(let i=0;i<cols*rows;i++) W.push([true,true,true,true]); // N E S W
  const vis=new Uint8Array(cols*rows);
  const D=[[0,-1,0,2],[1,0,1,3],[0,1,2,0],[-1,0,3,1]];
  let cx=0,cy=0; vis[0]=1; let count=1;
  while(count<cols*rows){
    const opts=[];
    for(const d of D){const nx=cx+d[0],ny=cy+d[1]; if(nx>=0&&ny>=0&&nx<cols&&ny<rows&&!vis[ny*cols+nx]) opts.push(d);}
    if(opts.length){ const d=opts[(Math.random()*opts.length)|0];
      const nx=cx+d[0],ny=cy+d[1];
      W[cy*cols+cx][d[2]]=false; W[ny*cols+nx][d[3]]=false;
      vis[ny*cols+nx]=1; count++; cx=nx; cy=ny;
    } else { // backtrack: find previous visited neighbor
      let moved=false;
      for(const d of D){const px=cx-d[0],py=cy-d[1]; if(px>=0&&py>=0&&px<cols&&py<rows&&vis[py*cols+px]){cx=px;cy=py;moved=true;break;}}
      if(!moved) break; // safety (shouldn't happen in connected gen)
    }
  }
  // loops: remove ~10% of internal walls
  for(let y=0;y<rows;y++)for(let x=0;x<cols;x++){
    const i=y*cols+x;
    if(x<cols-1 && Math.random()<0.09) W[i][1]=W[i+1][3]=false;
    if(y<rows-1 && Math.random()<0.09) W[i][0]=W[i+cols][2]=false;
  }
  return {cols,rows,W};
}
```

Wait — the DFS with explicit backtracking by scanning visited neighbors: standard iterative approach uses a stack of positions; my "find previous visited neighbor" works because in a spanning tree each unvisited expansion creates exactly one parent link... but with loops carved later, during generation it's still a tree so each cell has exactly one visited neighbor (its parent) — yes this works: the current cell always has exactly one visited neighbor until we pop. But careful: when we backtrack to parent and later expand from there to another unvisited child, fine. The "previous visited neighbor" is unique at that moment in a tree. ✔

function bfsPath(m, sx,sy, tx,ty){
  const {cols,rows,W}=m;
  const prev=new Int32Array(cols*rows).fill(-1);
  const q=[sy*cols+sx]; prev[sy*cols+sx]= -2; let qi=0;
  while(qi<q.length){
    const cur=q[qi++]; if(cur===ty*cols+tx) break;
    const cx=cur%cols, cy=(cur/cols)|0;
    for(let d=0;d<4;d++){ if(W[cur][d]) continue; const nx=cx+[0,1,0,-1][d], ny=cy+[-1,0,1,0][d];
      if(nx<0||ny<0||nx>=cols||ny>=rows) continue; const ni=ny*cols+nx;
      if(prev[ni]===-1){prev[ni]=cur;q.push(ni);} }
  }
  const ti=ty*cols+tx; if(prev[ti]===-1) return null;
  const path=[]; let c=ti; while(c!==-2){path.push(c); c=prev[c];}
  path.reverse(); return path; // array of cell indices
}

/* ================= level building ================= */

function addPickup(i,type,used,n){ if(used.has(i))i=clamp(i+1,1,n-1); used.add(i); G.pickups.push({i,type,taken:false}); }
function spawnWisp(){
  const d=distField(G.maze,G.player.x,G.player.y);let mx=0;
  for(let i=0;i<d.length;i++)mx=Math.max(mx,d[i]);
  let cands=[];for(let i=0;i<d.length;i++)if(d[i]>=Math.max(6,mx*.5))cands.push(i);
  if(!cands.length)cands=[d.length-1];
  const ni=cands[(Math.random()*cands.length)|0];
  G.wisp.x=ni%G.maze.cols;G.wisp.y=(ni/G.maze.cols)|0;G.wisp.trail.length=0;
}
function startLevel(L){
  G.level=L;const{cols,rows}=levelDims(L);G.maze=genMaze(cols,rows);
  const d=distField(G.maze,0,0);let maxD=1;for(let i=0;i<d.length;i++)maxD=Math.max(maxD,d[i]);
  const target=clamp(7+L*2,9,maxD),cands=[];
  for(let i=0;i<d.length;i++)if(d[i]===target)cands.push(i);
  G.exitI=cands[(Math.random()*cands.length)|0];
  G.path=bfsPath(G.maze,0,0,G.exitI%cols,(G.exitI/cols)|0)||[0];
  G.pIdx=0;
  const P=G.player;P.x=0;P.y=0;P.fx=0;P.fy=0;P.boost=0;
  G.moving=false;G.queue=false;G.animT=0;G.trail.length=0;
  const n=G.path.length-1,used=new Set([0,G.exitI]);
  addPickup(clamp(Math.round(n*.38),1,n-1),'ink',used,n);
  if(L>=4)addPickup(clamp(Math.round(n*.58),1,n-1),'ink',used,n);
  addPickup(clamp(Math.round(n*.78),1,n-1),'skip',used,n);
  spawnWisp();G.wisp.timer=2.6;G.invuln=1.6;
  G.word=pickWord(L,'');G.typed=0;G.misses=0;
  computeLayout();buildMazeLayer();
  buildKeys();updateHUD();updateProgress();
  G.banner={txt:'MAZE '+L,sub:L===1?'type to move — reach the green gate':'the glitch is faster now',t:0};
  setHint('exit '+n+' steps ahead');
}

/* ================= gameplay actions ================= */
function beginMove(){ if(G.pIdx+1>=G.path.length)return;
  const m=G.maze,a=G.path[G.pIdx],b=G.path[G.pIdx+1];
  G.fromX=a%m.cols;G.fromY=(a/m.cols)|0;G.toX=b%m.cols;G.toY=(b/m.cols)|0;
  G.animT=0;G.moving=true; }
function arrive(){
  G.moving=false;G.pIdx++;const m=G.maze,c=G.path[G.pIdx];
  const P=G.player;P.x=c%m.cols;P.y=(c/m.cols)|0;P.fx=P.x;P.fy=P.y;
  const pk=G.pickups.find(p=>!p.taken&&p.i===c);
  if(pk){pk.taken=true;collect(pk,c);}
  updateProgress();
  if(c===G.exitI){levelClear();return;}
  if(G.queue){G.queue=false;beginMove();}
}
function collect(pk,ci){
  const m=G.maze,x=(ci%m.cols+.5)*cs,y=((ci/m.cols)|0+.5)*cs;
  if(pk.type==='ink'){G.ink=Math.min(5,G.ink+1);updateInk();popup(x,y,'+INK','#5fd4e6');burst(x,y,'#5fd4e6',14,1);}
  else{G.tokens=Math.min(3,G.tokens+1);updateTokens();popup(x,y,'+SKIP KEY','#ffd166');burst(x,y,'#ffd166',14,1);}
  AudioSys.pickup();
}
function completeWord(){
  if(G.state!=='play'||G.paused)return;
  const bonus=40+G.level*10;G.score+=bonus;
  const px=(G.player.fx+.5)*cs,py=(G.player.fy+.5)*cs;
  popup(px,py-cs*.5,'+'+bonus,'#ffd166');burst(px,py,'#ffb454',12,.9);ring(px,py,'#ffb454');AudioSys.word();
  if(G.moving)G.queue=true;else beginMove();
  if((G.pIdx+1)<(G.path.length-1)){G.word=pickWord(G.level,G.word);G.typed=0;G.misses=0;buildKeys();setHint('keep typing — the gate is '+ (G.path.length-2-G.pIdx)+' steps away');}
  else setHint('final step —');
}
function typeLetter(ch){
  if(G.state!=='play'||G.paused)return;
  const w=G.word;if(!w)return;
  if(ch===w[G.typed]){
    G.correct++;G.typed++;AudioSys.note(G.typed);G.player.boost=.25;
    const ks=keysEl.children;
    if(ks[G.typed-1]){ks[G.typed-1].classList.add('done');ks[G.typed-1].classList.remove('cur');}
    if(ks[G.typed])ks[G.typed].classList.add('cur');
    if(G.typed===w.length)completeWord();
  }else{
    G.wrong++;G.misses++;AudioSys.err();G.flash=Math.max(G.flash,.18);
    const sp=keysEl.children[G.typed];
    if(sp){sp.classList.add('err');setTimeout(()=>sp.classList.remove('err'),200);}
    if(G.misses>=4){resetWord();setHint('ink splatter — word reshuffled',true);}
  }
}
function resetWord(){G.word=pickWord(G.level,G.word);G.typed=0;G.misses=0;buildKeys();}
function useSkip(){
  if(G.state!=='play'||G.paused||G.tokens<=0||G.moving)return;
  G.tokens--;updateTokens();AudioSys.slam();G.shake=Math.max(G.shake,6);
  const bonus=40+G.level*10;G.score+=bonus;
  const px=(G.player.fx+.5)*cs,py=(G.player.fy+.5)*cs;
  popup(px,py-cs*.5,'SLAM +'+bonus,'#ffd166');burst(px,py,'#ffb454',18,1.2);ring(px,py,'#ffd166');
  [...keysEl.children].forEach(s=>{s.classList.remove('cur');s.classList.add('done');});
  beginMove();
  if((G.pIdx+1)<(G.path.length-1)){G.word=pickWord(G.level,G.word);G.typed=0;G.misses=0;buildKeys();}
}
function hitPlayer(){
  if(G.invuln>0||G.state!=='play')return;
  G.ink--;updateInk();G.shake=14;G.flash=.55;AudioSys.hit();
  const px=(G.player.fx+.5)*cs,py=(G.player.fy+.5)*cs;

  if(G.ink<=0){gameOver();return;}
  const d=distField(G.maze,G.player.x,G.player.y);let mx=0;
  for(let i=0;i<d.length;i++)mx=Math.max(mx,d[i]);
  let cands=[];for(let i=0;i<d.length;i++)if(d[i]>=mx*.6)cands.push(i);
  const ni=cands[(Math.random()*cands.length)|0];
  G.wisp.x=ni%G.maze.cols;G.wisp.y=(ni/G.maze.cols)|0;G.wisp.trail.length=0;
  G.invuln=1.3;resetWord();setHint('the glitch bit your ribbon — word reshuffled',true);
}

  cv.width=Math.max(1,r.width*dpr);cv.height=Math.max(1,r.height*dpr);

}
function computeLayout(){
  if(!G.maze)return;
  const w=cv.width/dpr,h=cv.height/dpr;
  cs=Math.max(8,Math.floor(Math.min((w-26)/G.maze.cols,(h-26)/G.maze.rows)));
  ox=(w-cs*G.maze.cols)/2;oy=(h-cs*G.maze.rows)/2;
}
function buildMazeLayer(){
  const m=G.maze;if(!m||cs<8)return;
  mL=document.createElement('canvas');mL.width=m.cols*cs*dpr;mL.height=m.rows*cs*dpr;
  const c=mL.getContext('2d');c.scale(dpr,dpr);
  for(let y=0;y<m.rows;y++)for(let x=0;x<m.cols;x++){
    c.fillStyle=((x+y)&1)?'#13212c':'#111d27';c.fillRect(x*cs,y*cs,cs+1,cs+1);
    c.fillStyle='rgba(120,160,200,.07)';c.beginPath();c.arc((x+.5)*cs,(y+.5)*cs,Math.max(1,cs*.045),0,TAU);c.fill();
  }
  const lw=Math.max(3,cs*.16);c.lineCap='round';
  for(let pass=0;pass<2;pass++){
    c.strokeStyle=pass?'#4a6b8c':'rgba(96,150,205,.16)';c.lineWidth=pass?lw:lw*2.8;c.beginPath();
    for(let y=0;y<m.rows;y++)for(let x=0;x<m.cols;x++){
      const i=y*m.cols+x,W=m.W[i];
      if(W[0]){c.moveTo(x*cs,y*cs);c.lineTo((x+1)*cs,y*cs);}
      if(W[1]){c.moveTo((x+1)*cs,y*cs);c.lineTo((x+1)*cs,(y+1)*cs);}
      if(y===m.rows-1&&W[2]){c.moveTo(x*cs,(y+1)*cs);c.lineTo((x+1)*cs,(y+1)*cs);}
      if(x===0&&W[3]){c.moveTo(x*cs,y*cs);c.lineTo(x*cs,(y+1)*cs);}
    }
    c.stroke();
  }
  c.strokeStyle='rgba(95,212,230,.3)';c.lineWidth=2;c.strokeRect(-4,-4,m.cols*cs+8,m.rows*cs+8);
  c.strokeStyle='rgba(95,212,230,.12)';c.lineWidth=6;c.strokeRect(-7,-7,m.cols*cs+14,m.rows*cs+14);
}

function dropPath(c,x,y,s){c.beginPath();c.moveTo(x,y-s);c.bezierCurveTo(x+s*.95,y-s*.1,x+s*.78,y+s*.85,x,y+s*.85);c.bezierCurveTo(x-s*.78,y+s*.85,x-s*.95,y-s*.1,x,y-s);c.closePath();}

    else{ctx.globalAlpha=a;ctx.fillStyle=p.col;ctx.beginPath();ctx.arc(p.x,p.y,p.size,0,TAU);ctx.fill();}
  }
  ctx.globalAlpha=1;
}
function drawPopups(){
  const sz=clamp(cs*.5,12,22);ctx.font='700 '+sz+'px '+MONO;ctx.textAlign='center';
  for(const p of G.popups){const a=clamp(1-p.t/.9,0,1);
    ctx.globalAlpha=a;ctx.lineWidth=3;ctx.strokeStyle='rgba(8,12,16,.8)';
    ctx.strokeText(p.txt,p.x,p.y);ctx.fillStyle=p.col;ctx.fillText(p.txt,p.x,p.y);}
  ctx.globalAlpha=1;ctx.textAlign='left';
}
function drawBanner(w,h){
  const b=G.banner;if(!b||b.t>1.5||G.state==='menu')return;
  const p=b.t/1.5,a=p<.15?p/.15:p>.8?(1-p)/.2:1,y=h*.3-p*10;
  ctx.save();ctx.globalAlpha=a;ctx.textAlign='center';
  ctx.font='800 '+clamp(cs*1.15,26,58)+'px '+MONO;
  ctx.fillStyle='#ffd9a0';ctx.fillText(b.txt,w/2,y);
  ctx.font='400 13px '+MONO;ctx.fillStyle='#8aa0b2';ctx.fillText(b.sub,w/2,y+24);
  ctx.restore();ctx.textAlign='left';
}
function render(t){
  const w=cv.width/dpr,h=cv.height/dpr;
  ctx.setTransform(dpr,0,0,dpr,0,0);
  const g=ctx.createLinearGradient(0,0,0,h);
  g.addColorStop(0,'#0d1620');g.addColorStop(.55,'#0f1b26');g.addColorStop(1,'#0a1118');
  ctx.fillStyle=g;ctx.fillRect(0,0,w,h);
  drawMotes(w,h);
  if(!G.maze)return;
  let sx=0,sy=0;if(G.shake>0){sx=(Math.random()*2-1)*G.shake*.5;sy=(Math.random()*2-1)*G.shake*.5;}
  ctx.save();ctx.translate(ox+sx,oy+sy);
  if(mL)ctx.drawImage(mL,0,0,G.maze.cols*cs,G.maze.rows*cs);
  drawRouteDots();drawExit(t);drawPickups(t);drawWisp(t);drawPlayer(t);

  if(G.flash>0){ctx.fillStyle='rgba(255,64,96,'+(G.flash*.35)+')';ctx.fillRect(0,0,w,h);}
}

/* ================= update loop ================= */
function wispInterval(){
  const base=G.state==='play'?Math.max(640,1500-(G.level-1)*110):820;
  return (base+Math.random()*160)/1000;
}
function wispStep(){
  const m=G.maze,d=distField(m,G.player.x,G.player.y),ci=G.wisp.y*m.cols+G.wisp.x,opts=[];
  for(let dd=0;dd<4;dd++){if(m.W[ci][dd])continue;const nx=G.wisp.x+DX[dd],ny=G.wisp.y+DY[dd];
    if(nx<0||ny<0||nx>=m.cols||ny>=m.rows)continue;opts.push({x:nx,y:ny,dist:d[ny*m.cols+nx]});}
  if(!opts.length)return;
  const chase=G.state==='play'?Math.min(.35+G.level*.06,.82):.15;let pick;
  if(Math.random()<chase){opts.sort((a,b)=>a.dist-b.dist);pick=opts[0];}
  else pick=opts[(Math.random()*opts.length)|0];
  G.wisp.trail.push({x:G.wisp.x,y:G.wisp.y,t:0});
  if(G.wisp.trail.length>7)G.wisp.trail.shift();
  G.wisp.x=pick.x;G.wisp.y=pick.y;
  if(pick.x===G.player.x&&pick.y===G.player.y)hitPlayer();
}
function update(dt){
  tGlobal+=dt;
  for(const mo of motes){mo.y-=mo.sp*dt*8;if(mo.y<0){mo.y=1;mo.x=Math.random();}}
  if(G.paused)return;
  if(G.shake>0)G.shake=Math.max(0,G.shake-dt*40);
  if(G.flash>0)G.flash=Math.max(0,G.flash-dt*1.6);
  if(G.invuln>0)G.invuln-=dt;
  if(G.banner)G.banner.t+=dt;
  for(let i=G.particles.length-1;i>=0;i--){const p=G.particles[i];p.life-=dt;
    if(p.ring)p.r+=p.vr*dt;else{p.x+=p.vx*dt;p.y+=p.vy*dt;p.vy+=p.g*dt;}
    if(p.life<=0)G.particles.splice(i,1);}
  for(let i=G.popups.length-1;i>=0;i--){const p=G.popups[i];p.t+=dt;p.y-=26*dt;if(p.t>1)G.popups.splice(i,1);}
  for(const tr of G.wisp.trail)tr.t+=dt;
  while(G.wisp.trail.length&&G.wisp.trail[0].t>.4)G.wisp.trail.shift();
  if(G.state!=='clear'){G.wisp.timer-=dt;if(G.wisp.timer<=0){G.wisp.timer=wispInterval();wispStep();}}
  if(G.state==='play'){
    G.time+=dt;updateTime();
    if(G.player.boost>0)G.player.boost=Math.max(0,G.player.boost-dt);
    if(G.moving){
      G.animT+=dt/STEP_DUR;const k=Math.min(G.animT,1),e=1-(1-k)*(1-k);
      G.player.fx=lerp(G.fromX,G.toX,e);G.player.fy=lerp(G.fromY,G.toY,e);
      G.trailT+=dt;if(G.trailT>.03){G.trailT=0;G.trail.push({x:G.player.fx,y:G.player.fy,t:0});if(G.trail.length>14)G.trail.shift();}
      for(const tr of G.trail)tr.t+=dt;
      while(G.trail.length&&G.trail[0].t>.3)G.trail.shift();
      if(G.animT>=1)arrive();
    }
  }else if(G.state==='clear'){
    G.clearT+=dt;
    $('nextFill').style.width=clamp(G.clearT/2.1*100,0,100)+'%';
    if(G.clearT>2.1)nextLevel();
  }
}
let last=performance.now();
function loop(now){const dt=Math.min(.05,(now-last)/1000);last=now;update(dt);render(now/1000);requestAnimationFrame(loop);}
