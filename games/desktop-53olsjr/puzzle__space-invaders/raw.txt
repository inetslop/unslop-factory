/* ============ grid core ============ */
function colX(c){return gridX+c*cell+cell/2;}
function rowY(r){return gridY+r*cell;}
function inBounds(c,r){return c>=0&&c<COLS&&r>=0&&r<ROWS;}
function adjacent(a,b){return Math.abs(a.c-b.c)+Math.abs(a.r-b.r)===1;}
function makesRun(r,c,col){
  let n=1;for(let i=c-1;i>=0&&grid[r][i]&&grid[r][i].color===col;i--)n++;
  for(let i=c+1;i<COLS&&grid[r][i]&&grid[r][i].color===col;i++)n++;if(n>=3)return true;
  n=1;for(let i=r-1;i>=0&&grid[i][c]&&grid[i][c].color===col;i--)n++;
  for(let i=r+1;i<ROWS&&grid[i][c]&&grid[i][c].color===col;i++)n++;return n>=3;}
function initGrid(dropIn){
  grid=[];
  for(let r=0;r<ROWS;r++){const rowArr=[];
    for(let c=0;c<COLS;c++){let col;do{col=(Math.random()*4)|0;}while(makesRun(r,c,col));
      rowArr.push({color:col,offY:dropIn?-(ROWS-r)*cell-30-c*9:0,vy:0,pop:null,sx:0,sy:0});}
    grid.push(rowArr);}
  phase=dropIn?'fall':'idle';cascade=1;
}
function reassignColors(){
  for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++){let col;do{col=(Math.random()*4)|0;}while(makesRun(r,c,col));grid[r][c].color=col;}
}
function findRuns(){
  const runs=[];
  for(let r=0;r<ROWS;r++){let c=0;while(c<COLS){const tl=grid[r][c];if(!tl){c++;continue;}
    let len=1;while(c+len<COLS&&grid[r][c+len]&&grid[r][c+len].color===tl.color)len++;
    if(len>=3){const cells=[];for(let i=0;i<len;i++)cells.push(r*COLS+c+i);runs.push({len,cells});}c+=len;}}
  for(let c=0;c<COLS;c++){let r=0;while(r<ROWS){const tl=grid[r][c];if(!tl){r++;continue;}
    let len=1;while(r+len<ROWS&&grid[r+len][c]&&grid[r+len][c].color===tl.color)len++;
    if(len>=3){const cells=[];for(let i=0;i<len;i++)cells.push((r+i)*COLS+c);runs.push({len,cells});}r+=len;}}
  return runs;
}
function matchAt(c,r){return makesRun(r,c,grid[r][c].color);}
function swapData(r1,c1,r2,c2){const tmp=grid[r1][c1];grid[r1][c1]=grid[r2][c2];grid[r2][c2]=tmp;}
function hasAnyMove(){
  for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++){
    if(c+1<COLS){swapData(r,c,r,c+1);const ok=matchAt(c,r)||matchAt(c+1,r);swapData(r,c,r,c+1);if(ok)return true;}
    if(r+1<ROWS){swapData(r,c,r+1,c);const ok=matchAt(c,r)||matchAt(c,r+1);swapData(r,c,r+1,c);if(ok)return true;}}
  return false;
}
function trySwap(a,b){
  if(phase!=='idle'||!adjacent(a,b))return false;
  phase='swapout';swapAnim={a:{c:a.c,r:a.r},b:{c:b.c,r:b.r},t:0};
  selected=null;wanted=null;sfx.swap();return true;
}
function tapTile(c,r){
  if(!inBounds(c,r))return;
  if(selected&&selected.c===c&&selected.r===r){selected=null;sfx.select();return;}
  if(selected&&adjacent(selected,{c,r})){
    if(phase==='idle')trySwap(selected,{c,r});else{wanted={c,r};sfx.select();}return;}
  selected={c,r};cursor={c,r};wanted=null;sfx.select();
}
function startResolve(runs,casc){
  phase='resolve';cascade=casc;gridFlash=1;
  const set=new Set();let bestRun=null;
  for(const run of runs){for(const k of run.cells)set.add(k);if(!bestRun||run.len>bestRun.len)bestRun=run;}
  let novaCell=-1;
  if(bestRun&&bestRun.len>=4)novaCell=bestRun.cells[(bestRun.cells.length/2)|0];
  for(const k of set){const r=(k/COLS)|0,c=k%COLS,tile=grid[r][c];tile.pop=0;
    bolts.push({x:colX(c),y:gridY-14,v:600+casc*35,col:tile.color,combo:casc,nova:false});
    for(let i=0;i<3;i++)particles.push({kind:'shard',x:colX(c),y:rowY(r)+cell/2,
      vx:(Math.random()-0.5)*160,vy:-80-Math.random()*120,rot:Math.random()*6,vr:(Math.random()-0.5)*14,
      size:3+Math.random()*4,t:0,life:0.6,col:PAL[tile.color].main});}
  if(novaCell>=0){const c=novaCell%COLS;
    bolts.push({x:colX(c),y:gridY-14,v:470,col:-1,combo:casc,nova:true});
    floats.push({x:colX(c),y:gridY-cell*ROWS-22,text:'NOVA!',color:'#ffd98a',size:22,t:0,life:1.2,vy:-36});
    sfx.nova();}
  sfx.match(casc);sfx.laser();
  if(casc>1)floats.push({x:W/2,y:gridY-cell*ROWS-48,text:'COMBO ×'+casc,color:'#ffffff',size:17+casc*3,t:0,life:1.2,vy:-30});
}
function finalizeRemove(){
  for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++){if(grid[r][c]&&grid[r][c].pop!=null)grid[r][c]=null;}
  for(let c=0;c<COLS;c++){let wr=ROWS-1;
    for(let r=ROWS-1;r>=0;r--){const tl=grid[r][c];if(tl){
      if(wr!==r){grid[wr][c]=tl;grid[r][c]=null;tl.offY=(r-wr)*cell;tl.vy=0;}wr--;}}
    for(let r=wr;r>=0;r--){grid[r][c]={color:(Math.random()*4)|0,offY:-((wr-r+1)*cell+34),vy:0,pop:null,sx:0,sy:0};}}
  phase='fall';
}
function afterChainEnd(){
  if(!hasAnyMove()){doShuffle();return;}
  if(wanted&&selected&&adjacent(selected,wanted)){trySwap(selected,wanted);wanted=null;}
}
function doShuffle(){
  phase='shuffle';shuffleT=0.85;sfx.shuffle();
  floats.push({x:W/2,y:gridY-cell*ROWS-24,text:'NO MOVES — SHUFFLE',color:'#ffb63d',size:15,t:0,life:1.4,vy:-18});
  let ok=false;for(let i=0;i<80&&!ok;i++){reassignColors();ok=findRuns().length===0&&hasAnyMove();}
}
function updateGrid(dt){
  for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++){const tl=grid[r][c];if(!tl)continue;
    if(tl.pop!=null)tl.pop+=dt/0.16;
    if(tl.offY<0){tl.vy+=cell*32*dt;tl.offY+=tl.vy*dt;if(tl.offY>=0){tl.offY=0;tl.vy=0;}}}
  if(phase==='swapout'||phase==='swapback'){
    const sw=swapAnim;sw.t+=dt/0.11;
    const p=(phase==='swapout')?easeInOut(clamp(sw.t,0,1)):1-easeInOut(clamp(sw.t,0,1));
    const A=grid[sw.a.r][sw.a.c],B=grid[sw.b.r][sw.b.c];
    const dx=(sw.b.c-sw.a.c)*cell*p,dy=(sw.b.r-sw.a.r)*cell*p;
    A.sx=dx;A.sy=dy;B.sx=-dx;B.sy=-dy;
    if(sw.t>=1){swapData(sw.a.r,sw.a.c,sw.b.r,sw.b.c);A.sx=A.sy=B.sx=B.sy=0;
      if(phase==='swapout'){const runs=findRuns();
        if(runs.length)startResolve(runs,1);
        else{phase='swapback';swapAnim={a:sw.a,b:sw.b,t:0};sfx.invalid();}}
      else{phase='idle';afterChainEnd();}}}
  else if(phase==='resolve'){let done=true;
    for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++){const tl=grid[r][c];if(tl&&tl.pop!=null&&tl.pop<1)done=false;}
    if(done)finalizeRemove();}
  else if(phase==='fall'){let done=true;
    for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++){const tl=grid[r][c];if(tl&&tl.offY<0)done=false;}
    if(done){const runs=findRuns();
      if(runs.length){cascade++;startResolve(runs,cascade);}
      else{phase='idle';cascade=1;afterChainEnd();}}}
  else if(phase==='shuffle'){shuffleT-=dt;if(shuffleT<=0)phase='idle';}
}

/* ============ waves / invaders ============ */
function spawnWave(n){
  const cols=Math.min(7+((n-1)>>1),9),rows=Math.min(3+(((n-1)/3)|0),5),hpB=((n-1)/4)|0;
  invaders=[];let idx=0;
  for(let r=0;r<rows;r++)for(let c=0;c<cols;c++,idx++){
    const type=r===0?'tank':(r===rows-1?'scuttle':'warden');
    let color=(Math.random()*4)|0;if(idx===0)color=0;else if(idx===1)color=1;else if(idx===2)color=2;
    const hp=(type==='tank'?4:type==='warden'?3:2)+hpB;
    invaders.push({type,color,hp,maxhp:hp,col:c,row:r,inT:-((c*0.06)+(r*0.1)),flash:0,dead:false,px:0,py:-300});}
  F={cols,rows,fx:(W-(cols-1)*invCell)/2,fy:hudH+64,dir:1,timer:1.3,frame:0,total:invaders.length,alive:invaders.length};
}
function stepFormation(){
  F.frame^=1;const step=invCell*0.85;F.fx+=step*F.dir;
  let minX=1e9,maxX=-1e9;
  for(const inv of invaders)if(!inv.dead){const x=F.fx+inv.col*invCell;if(x<minX)minX=x;if(x>maxX)maxX=x;}
  const half=invWD/2;
  if(maxX+half>W-8||minX-half<8){F.fx-=step*F.dir;F.fy+=rowH*0.62;F.dir*=-1;}
}
function updateFormation(dt){
  if(!F||!invaders.length)return;
  for(const inv of invaders){if(inv.inT<1.5)inv.inT+=dt;inv.flash=Math.max(0,inv.flash-dt*4);}
  if(state==='playing'&&F.alive>0){
    F.timer-=dt;const base=Math.max(0.24,0.6-wave*0.04),minI=0.075,prog=1-F.alive/F.total;
    if(F.timer<=0){stepFormation();F.timer=minI+(base-minI)*(1-prog);}
    if(F.fy+(F.rows-1)*rowH+invHT/2+8>=gridY-4)startDying();}
  else if(state==='dying'){
    dieT-=dt;dieAllT+=dt;
    if(dieT<=0){const a=invaders.find(i=>!i.dead);if(a){a.dead=true;F.alive--;explode(a.px,a.py,PAL[a.color],1.2);sfx.kill();}dieT=0.07;}
    if(dieAllT>2.6||F.alive<=0)endGame();}
  for(const inv of invaders){const p=clamp(inv.inT,0,1),e=easeOutCubic(p);
    inv.px=F.fx+inv.col*invCell;inv.py=(F.fy+inv.row*rowH)-(1-e)*260;}
}
function killInvader(inv,combo){
  if(inv.dead)return;inv.dead=true;F.alive--;
  const pts=PTS[inv.type]*combo;score+=pts;if(score>best)best=score;
  floats.push({x:inv.px,y:inv.py,text:'+'+pts,color:'#ffffff',size:14,t:0,life:1.1,vy:-46});
  explode(inv.px,inv.py,PAL[inv.color],inv.type==='tank'?1.5:inv.type==='warden'?1.15:0.9);
  shake+=inv.type==='tank'?7:3.5;sfx.kill();
  if(F.alive<=0&&state==='playing'&&!interlude){
    const b=250*wave;score+=b;interlude={t:2.1};whiteFlash=0.45;
    banners.push({text:'WAVE '+wave+' CLEARED',sub:'BONUS +'+b,t:0,life:2.2});sfx.clear();}
}

/* ============ bolts / particles / floats / banners ============ */
function updateBolts(dt){
  for(let i=bolts.length-1;i>=0;i--){const b=bolts[i];b.y-=b.v*dt;
    if(b.y<hudH+4){bolts.splice(i,1);continue;}
    if(state!=='playing')continue;
    for(const inv of invaders){if(inv.dead)continue;
      if(Math.abs(b.x-inv.px)<invWD/2+3&&b.y>inv.py-invHT/2-4&&b.y<inv.py+invHT/2+8){
        const dmg=b.nova?4:(b.col===inv.color?3:1);inv.hp-=dmg;inv.flash=1;
        for(let s=0;s<5;s++)particles.push({kind:'spark',x:b.x,y:b.y,
          vx:(Math.random()-0.5)*260,vy:(Math.random()-0.5)*260,t:0,life:0.3,col:b.nova?'#ffffff':PAL[b.col].main});
        if(inv.hp<=0)killInvader(inv,b.combo||1);else{sfx.hit();shake=Math.max(shake,2);}
        if(b.nova){for(const o of invaders){if(o.dead||o===inv)continue;
            const d=Math.hypot(o.px-inv.px,o.py-inv.py);
            if(d<invCell*1.6){o.hp-=2;o.flash=1;if(o.hp<=0)killInvader(o,b.combo||1);}}
          particles.push({kind:'ring',x:inv.px,y:inv.py,r:6,vr:520,t:0,life:0.4,col:'#ffffff'});
          shake+=8;redFlash=0.25;}
        bolts.splice(i,1);break;}}}
}
function explode(x,y,pal,s){
  particles.push({kind:'ring',x,y,r:4,vr:300*s,t:0,life:0.35,col:pal.main});
  const n=Math.round(12*s);
  for(let i=0;i<n;i++){const a=Math.random()*6.283,sp=(70+Math.random()*240)*s;
    particles.push({kind:'spark',x,y,vx:Math.cos(a)*sp,vy:Math.sin(a)*sp,t:0,life:0.35+Math.random()*0.3,col:i%3?pal.main:'#ffffff'});}
  for(let i=0;i<5;i++)particles.push({kind:'shard',x,y,vx:(Math.random()-0.5)*260*s,vy:-100-Math.random()*160*s,
    rot:Math.random()*6,vr:(Math.random()-0.5)*14,size:3+Math.random()*5*s,t:0,life:0.7,col:pal.main});
}
function updateParticles(dt){
  for(let i=particles.length-1;i>=0;i--){const p=particles[i];p.t+=dt;
    if(p.t>=p.life){particles.splice(i,1);continue;}
    if(p.kind==='ring'){p.r+=p.vr*dt;continue;}
    if(p.kind==='shoot'){p.x+=p.vx*dt;p.y+=p.vy*dt;continue;}
    if(p.kind==='shard')p.vy+=520*dt;
    p.x+=p.vx*dt;p.y+=p.vy*dt;if(p.rot!=null)p.rot+=p.vr*dt;}}
function updateFloats(dt){for(let i=floats.length-1;i>=0;i--){const f=floats[i];f.t+=dt;f.y+=f.vy*dt;if(f.t>=f.life)floats.splice(i,1);}}
function updateBanners(dt){for(let i=banners.length-1;i>=0;i--){banners[i].t+=dt;if(banners[i].t>=banners[i].life)banners.splice(i,1);}}

  for(const inv of invaders){inv.inT=Math.max(inv.inT,1);inv.px=F.fx+inv.col*invCell;inv.py=F.fy+inv.row*rowH;}
  if(F.fy+(F.rows-1)*rowH+invHT/2>gridY-30){F.fy=hudH+64;F.fx=(W-(F.cols-1)*invCell)/2;F.timer=0.8;}
}

/* ============ update ============ */
function updateStars(dt){
  for(const s of stars){s.y+=s.sp*dt;if(s.y>H+2){s.y=-2;s.x=Math.random()*W;}}
  shootT-=dt;if(shootT<=0){shootT=4+Math.random()*7;
    particles.push({kind:'shoot',x:W*(0.15+Math.random()*0.7),y:hudH+10+Math.random()*90,
      vx:-260-Math.random()*240,vy:130+Math.random()*80,t:0,life:0.8});}}
function update(dt){
  t+=dt;updateStars(dt);
  if(shake>0)shake=Math.max(0,shake-dt*26);
  whiteFlash=Math.max(0,whiteFlash-dt*2.2);redFlash=Math.max(0,redFlash-dt*1.4);gridFlash=Math.max(0,gridFlash-dt*3);
  if(kbT>0)kbT-=dt;
  updateParticles(dt);updateFloats(dt);updateBanners(dt);
  if(state==='menu'){menuStep(dt);return;}
  if(state==='paused')return;
  updateGrid(dt);updateFormation(dt);updateBolts(dt);
  if(interlude){interlude.t-=dt;if(interlude.t<=0){interlude=null;wave++;spawnWave(wave);
    banners.push({text:'WAVE '+wave,sub:'SWARM INBOUND',t:0,life:2.2});sfx.ui();}}
}

function gem(x,y,s,color,alpha){
  const p=PAL[color],m=s*0.1;ctx.globalAlpha=alpha;
  rr(x+m,y+m,s-2*m,s-2*m,s*0.2);
  const g=ctx.createLinearGradient(0,y,0,y+s);g.addColorStop(0,p.main);g.addColorStop(1,p.dark);
  ctx.fillStyle=g;ctx.fill();
  ctx.fillStyle='rgba(255,255,255,0.4)';ctx.beginPath();ctx.arc(x+s*0.34,y+s*0.3,s*0.1,0,7);ctx.fill();
  rr(x+m+1,y+m+1,s-2*m-2,s-2*m-2,s*0.18);
  ctx.strokeStyle=p.light;ctx.globalAlpha=alpha*0.85;ctx.lineWidth=Math.max(1.5,s*0.05);ctx.stroke();
  ctx.globalAlpha=1;
}
function drawBackground(){
  const g=ctx.createLinearGradient(0,0,0,H);g.addColorStop(0,'#050a1c');g.addColorStop(0.55,'#081226');g.addColorStop(1,'#0a0f24');
  ctx.fillStyle=g;ctx.fillRect(0,0,W,H);
  function neb(x,y,r,c){const ng=ctx.createRadialGradient(x,y,0,x,y,r);ng.addColorStop(0,c);ng.addColorStop(1,'rgba(0,0,0,0)');
    ctx.fillStyle=ng;ctx.fillRect(x-r,y-r,r*2,r*2);}
  neb(W*0.22+Math.sin(t*0.07)*34,H*0.26,W*0.36,'rgba(96,64,210,0.10)');
  neb(W*0.82+Math.cos(t*0.05)*42,H*0.46,W*0.3,'rgba(32,170,200,0.08)');
  neb(W*0.55,H*0.92,W*0.4,'rgba(255,70,150,0.05)');
  for(const s of stars){const a=(0.3+0.6*(0.5+0.5*Math.sin(t*s.tw+s.ph)))*(0.35+0.28*s.z);
    ctx.fillStyle='rgba(210,230,255,'+a.toFixed(3)+')';ctx.fillRect(s.x,s.y,s.r,s.r);}
}

      rr(x+2,y+2,cell-4,cell-4,6);ctx.fill();ctx.globalAlpha=1;}
