  for(let i=G.flashes.length-1;i>=0;i--){const f=G.flashes[i];f.t+=dt;if(f.t>.45)G.flashes.splice(i,1);}
  if(G.flash){G.flash.a-=dt*.28;if(G.flash.a<=0)G.flash=null;}
  G.shake*=Math.exp(-7*dt);if(G.shake<.3)G.shake=0;
  for(const d of dust){d.y-=d.sp*dt;if(d.y<-40){d.y=LH+40;d.x=Math.random()*(LW+200)-100;}}}
