function teleport(a,b){
  const dx=ball.x-a.x,dy=ball.y-a.y,d=Math.hypot(dx,dy)||1,off=Math.min(d,PORTR*0.55);
  ball.x=b.bx+dx/d*off;ball.y=b.by+dy/d*off;
  const c=Math.cos(b.rot),s=Math.sin(b.rot),vx=ball.vx,vy=ball.vy;
  ball.vx=vx*c-vy*s;ball.vy=vx*s+vy*c;
  ball.cd=0.4;a.flash=1;b.flash=1;
  for(let i=0;i<12;i++){const an=rnd(0,6.28),sp=rnd(40,220);
    parts.push({k:'dot',x:a.x+Math.cos(an)*PORTR,y:a.y+Math.sin(an)*PORTR,vx:Math.cos(an)*sp,vy:Math.sin(an)*sp,t:0,max:.6,size:rnd(1.5,3),col:'#ff8be0'});
    parts.push({k:'dot',x:b.bx+Math.cos(an)*PORTR,y:b.by+Math.sin(an)*PORTR,vx:Math.cos(an)*sp,vy:Math.sin(an)*sp,t:0,max:.6,size:rnd(1.5,3),col:'#ff8be0'});}
  pops.push({x:b.bx,y:b.by-PORTR-14,t:0,max:.9,text:'WARP',col:'#ffb7ec',size:15});
  sfxPortal();
}

  parts.push({k:'ring',x:r.x,y:r.y,vx:0,vy:0,t:0,max:.5,size:r.r*.7,col:'#dfe9ff'});
}
function splitRock(r,nx,ny,i){

  for(const sgn of[-1,1]){
    const cr=Math.max(16,r.r*0.58);
    const c=makeRock(r.x+px*sgn*cr*.9+r.vx*0.02,r.y+py*sgn*cr*.9+r.vy*0.02,cr,
      r.vx+nx*rnd(40,90)*sgn+px*sgn*rnd(10,35),r.vy+ny*rnd(40,90)*sgn+py*sgn*rnd(10,35));

    parts.push({k:'spark',x:r.x,y:r.y,vx:Math.cos(an)*sp,vy:Math.sin(an)*sp,t:0,max:rnd(.4,.8),size:rnd(1.5,3),col:'#cdd8ea'});}
}

function substep(h){
  // gravity / ion wells
  for(const w of wells){
    const dx=w.x-ball.x,dy=w.y-ball.y,d=Math.hypot(dx,dy)||1;
    const m=Math.min(Math.abs(w.pow)/Math.max(d*d,2400),1100)*(w.pow>0?1:-1);
    ball.vx+=dx/d*m*h;ball.vy+=dy/d*m*h;
  }
  const k=Math.pow(DRAG,h);ball.vx*=k;ball.vy*=k;
  ball.x+=ball.vx*h;ball.y+=ball.vy*h;
  ball.rot+=Math.hypot(ball.vx,ball.vy)*h/BR*0.5;
  // portals
  ball.cd-=h;
  if(ball.cd<=0&&Math.hypot(ball.vx,ball.vy)>40){
    for(const p of portals){
      if(Math.hypot(ball.x-p.ax,ball.y-p.ay)<PORTR-4){teleport({x:p.ax,y:p.ay},p);break;}
      if(Math.hypot(ball.x-p.bx,ball.y-p.by)<PORTR-4){teleport({x:p.bx,y:p.by},p);break;}
    }
  }
  // walls
  const R=0.9;let hitWall=false;
  if(ball.x<BR){ball.x=BR;ball.vx=-ball.vx*R;hitWall=true}
  if(ball.x>W-BR){ball.x=W-BR;ball.vx=-ball.vx*R;hitWall=true}
  if(ball.y<BR){ball.y=BR;ball.vy=-ball.vy*R;hitWall=true}
  if(ball.y>H-BR){ball.y=H-BR;ball.vy=-ball.vy*R;hitWall=true}
  if(hitWall){sfxBounce(Math.hypot(ball.vx,ball.vy)*1.2);
    for(let j=0;j<5;j++)parts.push({k:'dot',x:ball.x,y:ball.y,vx:rnd(-90,90),vy:rnd(-90,90),t:0,max:.4,size:2,col:'#7fdcff'});}
  // rocks
  for(let i=rocks.length-1;i>=0;i--){
    const r=rocks[i],dx=ball.x-r.x,dy=ball.y-r.y,d=Math.hypot(dx,dy),rr=r.r+BR;
    if(d<rr&&d>0.001){
      const nx=dx/d,ny=dy/d;
      ball.x=r.x+nx*rr;ball.y=r.y+ny*rr;
      let rvx=ball.vx-r.vx,rvy=ball.vy-r.vy,dot=rvx*nx+rvy*ny;
      if(dot<0){
        const impact=-dot;
        rvx-=2*dot*nx;rvy-=2*dot*ny;
        ball.vx=r.vx+rvx*0.84;ball.vy=r.vy+rvy*0.84;
        sfxBounce(impact);
        for(let j=0;j<6;j++)parts.push({k:'spark',x:ball.x,y:ball.y,vx:rnd(-120,120)+ball.vx*.3,vy:rnd(-120,120)+ball.vy*.3,t:0,max:.35,size:2,col:'#d7e6ff'});
        if(impact>430){
          if(r.r<=30)destroyRock(r,i);
          else if(r.r<=62)splitRock(r,nx,ny,i);
          else{r.flash=1;sfxClang();shake=Math.max(shake,3);}
        }else if(impact>140)shake=Math.max(shake,1.5);
      }
    }
  }
  // cup
  const dx=cup.x-ball.x,dy=cup.y-ball.y,d=Math.hypot(dx,dy),sp=Math.hypot(ball.vx,ball.vy);
  if(d<44&&sp<CUPCAP&&d>0.1){const m=(1-d/44)*300;ball.vx+=dx/d*m*h;ball.vy+=dy/d*m*h;}
  if(d<21){
    if(sp<CUPCAP){startSink();return;}
    const nx=ball.x-cup.x/(d||1),ny=ball.y-cup.y/(d||1);
    ball.x=cup.x+nx*21;ball.y=cup.y+ny*21;
    const dot=ball.vx*nx+ball.vy*ny;
    if(dot<0){ball.vx-=2*dot*nx*0.65;ball.vy-=2*dot*ny*0.65;}
    sfxClink();pops.push({x:cup.x,y:cup.y-40,t:0,max:.8,text:'TOO FAST!',col:'#ffc94d',size:13});
  }
}
function physics(dt){
  const sp=Math.hypot(ball.vx,ball.vy);
  const n=clamp(Math.ceil(sp*dt/7),1,5),h=dt/n;
  for(let i=0;i<n;i++){substep(h);if(state!=='rolling')return;}
  if(state==='rolling'){
    const s2=Math.hypot(ball.vx,ball.vy);
    if(s2>30){trail.push({x:ball.x,y:ball.y});if(trail.length>26)trail.shift();}
    if(s2<STOPV){lowT+=dt;
      if(lowT>0.45){ball.vx=ball.vy=0;state='aiming';
        if(strokes>=maxS)finishHole(false);}}
    else lowT=0;
  }
}

function update(dt){
  time+=dt;shake*=Math.exp(-6*dt);if(shake<0.05)shake=0;
  // ambient rocks
  for(const r of rocks){
    r.x+=r.vx*dt;r.y+=r.vy*dt;r.rot+=r.spin*dt;if(r.flash>0)r.flash=Math.max(0,r.flash-dt*3);
    if(r.x<r.r&&r.vx<0)r.vx*=-1;if(r.x>W-r.r&&r.vx>0)r.vx*=-1;
    if(r.y<r.r&&r.vy<0)r.vy*=-1;if(r.y>H-r.r&&r.vy>0)r.vy*=-1;
  }
  for(const p of portals)if(p.flash>0)p.flash=Math.max(0,p.flash-dt*2);
  // particles / popups
  for(let i=parts.length-1;i>=0;i--){const p=parts[i];p.t+=dt;
    if(p.t>p.max){parts.splice(i,1);continue;}
    const k=Math.pow(0.25,dt);p.vx*=k;p.vy*=k;p.x+=p.vx*dt;p.y+=p.vy*dt;}
  for(let i=pops.length-1;i>=0;i--){pops[i].t+=dt;if(pops[i].t>pops[i].max)pops.splice(i,1);}
  // dissolve trail when not rolling
  if(state!=='rolling'&&trail.length)trail.splice(0,2);

  if(state==='aiming'||state==='charging'){
    if(keys.ArrowLeft)aim.ang-=2.3*dt;
    if(keys.ArrowRight)aim.ang+=2.3*dt;
    if(charging){
      aim.power+=aim.dir*dt*0.9;
      if(aim.power>1){aim.power=1;aim.dir=-1}
      if(aim.power<0){aim.power=0;aim.dir=1}
      const b=Math.floor(aim.power*8);if(b!==lastB){sfxTick(aim.power);lastB=b;}
      if(chOsc)chOsc.o.frequency.value=60+aim.power*340;
    }
  }
  else if(state==='rolling')physics(dt);
  else if(state==='sinking'){
    ball.sinkT+=dt/0.55;ball.rot+=dt*22;
    if(ball.sinkT>=1&&state==='sinking'){finishHole(true);}
  }
}

        for(let j=0;j<3;j++){a+=rnd(-1.4,1.4);const rr=rnd(10,62);
          x=w.x+Math.cos(a)*rr;y=w.y+Math.sin(a)*rr;ctx.lineTo(x,y);}ctx.stroke();}
      ctx.setLineDash([10,14]);ctx.lineDashOffset=-time*40;
      ctx.strokeStyle=`rgba(120,220,255,${.35*fl})`;ctx.lineWidth=2;
      ctx.beginPath();ctx.arc(w.x,w.y,78,0,6.28);ctx.stroke();ctx.setLineDash([]);
      const cg=ctx.createRadialGradient(w.x-4,w.y-4,1,w.x,w.y,22);
      cg.addColorStop(0,'#ffffff');cg.addColorStop(.6,'#9fe4ff');cg.addColorStop(1,'#3f8fd8');
      ctx.fillStyle=cg;ctx.beginPath();ctx.arc(w.x,w.y,20,0,6.28);ctx.fill();
      ctx.globalCompositeOperation='source-over';
    }
  }
}
function drawPortals(){
  for(const p of portals){
    const ends=[[p.ax,p.ay,1],[p.bx,p.by,-1]];
    for(const [x,y,dir] of ends){
      ctx.globalCompositeOperation='lighter';
      const g=ctx.createRadialGradient(x,y,0,x,y,PORTR*2.3);
      g.addColorStop(0,'rgba(255,95,210,.28)');g.addColorStop(1,'rgba(255,95,210,0)');
      ctx.fillStyle=g;ctx.beginPath();ctx.arc(x,y,PORTR*2.3,0,6.28);ctx.fill();
      ctx.globalCompositeOperation='source-over';
      ctx.fillStyle='rgba(24,4,34,.85)';ctx.beginPath();ctx.arc(x,y,PORTR*.72,0,6.28);ctx.fill();
      ctx.setLineDash([16,13]);ctx.lineDashOffset=time*55*dir;
      ctx.strokeStyle='rgba(255,95,210,.85)';ctx.lineWidth=3;
      ctx.beginPath();ctx.arc(x,y,PORTR,0,6.28);ctx.stroke();
      ctx.setLineDash([8,20]);ctx.lineDashOffset=-time*40*dir;
      ctx.strokeStyle='rgba(255,170,235,.5)';ctx.lineWidth=2;
      ctx.beginPath();ctx.arc(x,y,PORTR*.62,0,6.28);ctx.stroke();ctx.setLineDash([]);
      const oa=time*2.4*dir;
      ctx.fillStyle='#ffd1f2';ctx.beginPath();
      ctx.arc(x+Math.cos(oa)*PORTR*.82,y+Math.sin(oa)*PORTR*.82,3,0,6.28);ctx.fill();
      if(p.flash>0){ctx.strokeStyle=`rgba(255,180,240,${p.flash})`;ctx.lineWidth=3;
        ctx.beginPath();ctx.arc(x,y,PORTR*(1+(1-p.flash)*1.7),0,6.28);ctx.stroke();}
    }
  }
}
function drawCup(){
  const near=Math.hypot(ball.x-cup.x,ball.y-cup.y)<140&&(state==='rolling');
  let g=ctx.createRadialGradient(cup.x,cup.y,0,cup.x,cup.y,46);
  g.addColorStop(0,'rgba(77,225,255,.12)');g.addColorStop(1,'rgba(77,225,255,0)');
  ctx.fillStyle=g;ctx.beginPath();ctx.arc(cup.x,cup.y,46,0,6.28);ctx.fill();
  g=ctx.createRadialGradient(cup.x-6,cup.y-6,2,cup.x,cup.y,26);
  g.addColorStop(0,'#2b3a55');g.addColorStop(1,'#141d30');
  ctx.fillStyle=g;ctx.beginPath();ctx.arc(cup.x,cup.y,26,0,6.28);ctx.fill();
  ctx.strokeStyle='#3b5578';ctx.lineWidth=2;ctx.stroke();
  g=ctx.createRadialGradient(cup.x,cup.y,0,cup.x,cup.y,14);
  g.addColorStop(0,'#020409');g.addColorStop(.75,'#050a14');g.addColorStop(1,'rgba(77,225,255,.5)');
  ctx.fillStyle=g;ctx.beginPath();ctx.arc(cup.x,cup.y,13,0,6.28);ctx.fill();
  // flag
  const px=cup.x,py0=cup.y-20,ph=64,wy=py0-ph,wave=Math.sin(time*(near?9:4.5))* (near?6:3.5);
  ctx.strokeStyle='#cfd8e6';ctx.lineWidth=3;ctx.beginPath();ctx.moveTo(px,py0);ctx.lineTo(px,wy);ctx.stroke();
  ctx.fillStyle='#ff5c5c';ctx.beginPath();ctx.moveTo(px+1,wy);
  ctx.quadraticCurveTo(px+20,wy+4+wave,px+34,wy+9+wave*1.6);
  ctx.quadraticCurveTo(px+20,wy+13+wave*.6,px+1,wy+18);ctx.closePath();ctx.fill();
  ctx.strokeStyle='rgba(120,10,20,.5)';ctx.lineWidth=1;ctx.stroke();
}
function drawRock(r){
  ctx.save();ctx.translate(r.x,r.y);ctx.rotate(r.rot);
  const L=r.r>60?34:r.r>30?46:60;
  const g=ctx.createLinearGradient(-r.r,-r.r,r.r,r.r);
  g.addColorStop(0,`hsl(218,10%,${L+14}%)`);g.addColorStop(1,`hsl(222,14%,${Math.max(12,L-16)}%)`);
  ctx.beginPath();
  r.verts.forEach((v,i)=>i?ctx.lineTo(v[0],v[1]):ctx.moveTo(v[0],v[1]));
  ctx.closePath();ctx.fillStyle=g;ctx.fill();
  ctx.strokeStyle='rgba(190,215,255,.28)';ctx.lineWidth=1.5;ctx.stroke();
  for(const c of r.craters){
    const cx=Math.cos(c.a)*c.d*r.r,cy=Math.sin(c.a)*c.d*r.r,cr=c.s*r.r;
    ctx.fillStyle='rgba(4,8,20,.3)';ctx.beginPath();ctx.arc(cx,cy,cr,0,6.28);ctx.fill();
    ctx.strokeStyle='rgba(200,225,255,.14)';ctx.lineWidth=1;
    ctx.beginPath();ctx.arc(cx,cy-cr*.15,cr*.8,Math.PI*1.1,Math.PI*1.9);ctx.stroke();
  }
  if(r.flash>0){ctx.fillStyle=`rgba(255,255,255,${r.flash*.45})`;
    ctx.beginPath();r.verts.forEach((v,i)=>i?ctx.lineTo(v[0],v[1]):ctx.moveTo(v[0],v[1]));ctx.closePath();ctx.fill();}
  ctx.restore();
}
function drawTrail(){
  if(trail.length<2)return;
  ctx.globalCompositeOperation='lighter';ctx.lineCap='round';
  for(let i=1;i<trail.length;i++){
    const f=i/trail.length;
    ctx.strokeStyle=`rgba(120,230,255,${f*f*.5})`;ctx.lineWidth=1+f*6;
    ctx.beginPath();ctx.moveTo(trail[i-1].x,trail[i-1].y);ctx.lineTo(trail[i].x,trail[i].y);ctx.stroke();
  }
  ctx.globalCompositeOperation='source-over';
}
function drawBall(x,y,sc){
  ctx.globalCompositeOperation='lighter';
  const g=ctx.createRadialGradient(x,y,0,x,y,30*sc);
  g.addColorStop(0,'rgba(140,235,255,.3)');g.addColorStop(1,'rgba(140,235,255,0)');
  ctx.fillStyle=g;ctx.beginPath();ctx.arc(x,y,30*sc,0,6.28);ctx.fill();
  ctx.globalCompositeOperation='source-over';
  const b=BR*sc,bg=ctx.createRadialGradient(x-b*.4,y-b*.45,b*.15,x,y,b);
  bg.addColorStop(0,'#ffffff');bg.addColorStop(.6,'#d8edff');bg.addColorStop(1,'#8fb8dc');
  ctx.fillStyle=bg;ctx.beginPath();ctx.arc(x,y,b,0,6.28);ctx.fill();
  if(sc>0.4){ctx.save();ctx.translate(x,y);ctx.rotate(ball.rot);
    ctx.fillStyle='rgba(90,130,170,.5)';
    for(let i=0;i<3;i++){const a=i*2.09;ctx.beginPath();ctx.arc(Math.cos(a)*b*.48,Math.sin(a)*b*.48,b*.2,0,6.28);ctx.fill();}
    ctx.restore();}
}
function drawAim(){
  const bob=Math.sin(time*2.1)*3,bx=ball.x,by=ball.y+bob;
  const p=charging?aim.power:0,len=charging?60+aim.power*320:120;
  const ex=bx+Math.cos(aim.ang)*len,ey=by+Math.sin(aim.ang)*len;
  ctx.setLineDash([10,10]);ctx.lineDashOffset=-time*45;
  ctx.strokeStyle=charging?lerpColor(aim.power):'rgba(120,255,230,.5)';
  ctx.lineWidth=charging?3:2;
  ctx.beginPath();ctx.moveTo(bx+Math.cos(aim.ang)*16,by+Math.sin(aim.ang)*16);ctx.lineTo(ex,ey);ctx.stroke();
  ctx.setLineDash([]);
  ctx.fillStyle=ctx.strokeStyle;ctx.save();ctx.translate(ex,ey);ctx.rotate(aim.ang);
  ctx.beginPath();ctx.moveTo(12,0);ctx.lineTo(-4,-7);ctx.lineTo(-4,7);ctx.closePath();ctx.fill();ctx.restore();
  if(charging){
    ctx.strokeStyle=lerpColor(aim.power);ctx.lineWidth=4;
    ctx.beginPath();ctx.arc(bx,by,18,-Math.PI/2,-Math.PI/2+aim.power*Math.PI*2);ctx.stroke();
  }
  // trajectory preview
  const pw=charging?Math.max(aim.power,.05):.5,sp=MINV+pw*(MAXV-MINV),k60=Math.pow(DRAG,1/60);
  let px=ball.x,py=ball.y,pvx=Math.cos(aim.ang)*sp,pvy=Math.sin(aim.ang)*sp,hit=null;
  ctx.fillStyle='#9fe8ff';
  for(let i=0;i<84;i++){
    for(const w of wells){const dx=w.x-px,dy=w.y-py,d=Math.hypot(dx,dy)||1;
      const m=Math.min(Math.abs(w.pow)/Math.max(d*d,2400),1100)*(w.pow>0?1:-1);
      pvx+=dx/d*m/60;pvy+=dy/d*m/60;}
    pvx*=k60;pvy*=k60;px+=pvx/60;py+=pvy/60;
    if(px<BR||px>W-BR||py<BR||py>H-BR){hit=[px,py];break;}
    let blocked=false;for(const r of rocks)if(Math.hypot(px-r.x,py-r.y)<r.r+BR){blocked=true;break;}
    if(blocked){hit=[px,py];break;}
    if(i%4===0){ctx.globalAlpha=.5*(1-i/84);
      ctx.beginPath();ctx.arc(px,py,2.6-1.4*i/84,0,6.28);ctx.fill();}
  }
  ctx.globalAlpha=1;
  if(hit){ctx.strokeStyle='rgba(255,200,80,.65)';ctx.lineWidth=2;
    ctx.beginPath();ctx.arc(hit[0],hit[1],7,0,6.28);ctx.stroke();}
}
function drawFx(){
  for(const p of parts){
    const a=1-p.t/p.max;
    if(p.k==='ring'){ctx.strokeStyle=p.col.replace(')',`,${a*.8})`).replace('rgb','rgba');
      ctx.lineWidth=2.5;ctx.beginPath();ctx.arc(p.x,p.y,p.size+p.t*260,0,6.28);ctx.stroke();}
    else if(p.k==='spark'){ctx.strokeStyle=p.col;ctx.globalAlpha=a;ctx.lineWidth=p.size;
      const l=Math.max(4,Math.hypot(p.vx,p.vy)*.03);
      ctx.beginPath();ctx.moveTo(p.x,p.y);ctx.lineTo(p.x-p.vx*.035,p.y-p.vy*.035);ctx.stroke();ctx.globalAlpha=1;}
    else{ctx.fillStyle=p.col;ctx.globalAlpha=a;
      ctx.beginPath();ctx.arc(p.x,p.y,p.size*a+0.4,0,6.28);ctx.fill();ctx.globalAlpha=1;}
  }
  for(const p of pops){
    const a=1-p.t/p.max,s=p.t<.15?1+(1-p.t/.15)*.5:1;
    ctx.font=`700 ${Math.round(p.size*s)}px ui-monospace,Consolas,monospace`;
    ctx.textAlign='center';ctx.globalAlpha=a;
    ctx.fillStyle=p.col;ctx.fillText(p.text,p.x,p.y-p.t*36);ctx.globalAlpha=1;
  }
}
function draw(){
  ctx.setTransform(DPR,0,0,DPR,0,0);
  drawBackground();
  const shx=(Math.random()*2-1)*shake,shy=(Math.random()*2-1)*shake;
  ctx.save();ctx.translate(OX+shx,OY+shy);ctx.scale(S,S);
  drawBoundary();drawWells();drawPortals();drawCup();
  rocks.forEach(drawRock);drawTrail();
  if(state==='sinking'){
    const t=Math.min(1,ball.sinkT),e=t*t*(3-2*t);
    drawBall(ball.sx+(cup.x-ball.sx)*e,ball.sy+(cup.y-ball.sy)*e,1-e*.92);
  }else if(state!=='holeDone'&&state!=='roundEnd'){
    const bob=(state==='aiming'||state==='charging')?Math.sin(time*2.1)*3:0;
    drawBall(ball.x,ball.y+bob,1);
    if(state==='aiming'||state==='charging')drawAim();
  }
  drawFx();
  ctx.restore();
  // vignette + scanlines
  ctx.fillStyle=vigGrad;ctx.fillRect(0,0,cw,chh);
  if(scanPat){ctx.globalAlpha=.16;ctx.fillStyle=scanPat;ctx.fillRect(0,0,cw,chh);ctx.globalAlpha=1;}
}
