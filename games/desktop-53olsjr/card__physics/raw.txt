function collide(a,b){
  // axes
  const ca=Math.cos(a.angle), sa=Math.sin(a.angle);
  const cb=Math.cos(b.angle), sb=Math.sin(b.angle);
  const ax=[ [ca,sa], [-sa,ca], [cb,sb], [-sb,cb] ];
  let minPen=Infinity, nx=0,ny=0;
  const dx=b.x-a.x, dy=b.y-a.y;
  for (const [ux,uy] of ax){
    // project both boxes onto axis
    const ra = a.w/2*Math.abs(ux*ca+uy*sa) + a.h/2*Math.abs(ux*(-sa)+uy*ca);
    const rb = b.w/2*Math.abs(ux*cb+uy*sb) + b.h/2*Math.abs(ux*(-sb)+uy*cb);
    const d = dx*ux+dy*uy;
    const pen = ra+rb-Math.abs(d);
    if (pen<=0) return null;
    if (pen<minPen){ minPen=pen; nx=ux; ny=uy; }
  }
  // orient normal from a to b
  if (dx*nx+dy*ny<0){nx=-nx;ny=-ny;}
  // contacts: vertices of a inside b, or b inside a
  const eps=0.35;
  const contacts=[];
  for(const v of boxVerts(a)) if(pointInBox(v,b,eps)) contacts.push({x:v.x,y:v.y,pen:minPen});
  for(const v of boxVerts(b)) if(pointInBox(v,a,eps)) contacts.push({x:v.x,y:v.y,pen:minPen});
  if(!contacts.length){ // fallback: midpoint contact (shouldn't happen much)
    contacts.push({x:(a.x+b.x)/2,y:(a.y+b.y)/2,pen:minPen});
  }
  if(contacts.length>2){
    // keep the two with max separation? just take first two… better: pick most "opposite" pair. Simple: sort by penetration along normal? take first two.
    contacts.length=2;
  }
  return {a,b,nx,ny,pen:minPen,contacts};
}

  let minPen=Infinity,nx=0,ny=0;
  for(let i=0;i<4;i++){
    const ux=axes[i][0],uy=axes[i][1];
    const ra=a.w/2*Math.abs(ux*ca+uy*sa)+a.h/2*Math.abs(-ux*sa+uy*ca);
    const rb=b.w/2*Math.abs(ux*cb+uy*sb)+b.h/2*Math.abs(-ux*sb+uy*cb);
    const pen=ra+rb-Math.abs(dx*ux+dy*uy);
    if(pen<=0)return null;
    if(pen<minPen){minPen=pen;nx=ux;ny=uy;}
  }
  if(dx*nx+dy*ny<0){nx=-nx;ny=-ny;}
  const eps=.35,contacts=[];
  for(const v of boxVerts(a))if(pointInBox(v,b,eps))contacts.push({x:v.x,y:v.y});
  for(const v of boxVerts(b))if(pointInBox(v,a,eps))contacts.push({x:v.x,y:v.y});
  if(!contacts.length)contacts.push({x:(a.x+b.x)/2,y:(a.y+b.y)/2});
  if(contacts.length>2)contacts.length=2;
  return {a,b,nx,ny,pen:minPen,contacts};
}
function resolvePair(m){
  const a=m.a,b=m.b,nx=m.nx,ny=m.ny;
  for(const ct of m.contacts){
    const rax=ct.x-a.x,ray=ct.y-a.y,rbx=ct.x-b.x,rby=ct.y-b.y;
    let rvx=(b.vx-b.va*rby)-(a.vx-a.va*ray);
    let rvy=(b.vy+b.va*rbx)-(a.vy+a.va*rax);
    const vn=rvx*nx+rvy*ny;
    if(vn<0){
      const raCn=rax*ny-ray*nx,rbCn=rbx*ny-rby*nx;
      const invN=a.invMass+b.invMass+raCn*raCn*a.invI+rbCn*rbCn*b.invI;
      const j=-(1.02)*vn/invN;
      a.vx-=j*nx*a.invMass;a.vy-=j*ny*a.invMass;a.va-=raCn*j*a.invI;
      b.vx+=j*nx*b.invMass;b.vy+=j*ny*b.invMass;b.va+=rbCn*j*b.invI;
      rvx=(b.vx-b.va*rby)-(a.vx-a.va*ray);
      rvy=(b.vy+b.va*rbx)-(a.vy+a.va*rax);
      const vn2=rvx*nx+rvy*ny;
      let tx=rvx-nx*vn2,ty=rvy-ny*vn2;
      const tl=Math.hypot(tx,ty);
      if(tl>1e-4){
        tx/=tl;ty/=tl;
        const raCt=rax*ty-ray*tx,rbCt=rbx*ty-rby*tx;
        const invT=a.invMass+b.invMass+raCt*raCt*a.invI+rbCt*rbCt*b.invI;
        let jt=-(rvx*tx+rvy*ty)/invT;
        const mx=MU*Math.abs(j);
        jt=clamp(jt,-mx,mx);
        a.vx-=jt*tx*a.invMass;a.vy-=jt*ty*a.invMass;a.va-=raCt*jt*a.invI;
        b.vx+=jt*tx*b.invMass;b.vy+=jt*ty*b.invMass;b.va+=rbCt*jt*b.invI;
      }
    }
  }
}
function correctPos(m){
  const a=m.a,b=m.b,inv=a.invMass+b.invMass;
  if(!inv)return;
  const c=Math.max(m.pen-.5,0)*.32/inv;
  a.x-=m.nx*c*a.invMass;a.y-=m.ny*c*a.invMass;
  b.x+=m.nx*c*b.invMass;b.y+=m.ny*c*b.invMass;
}
function step(dt){
  for(const b of bodies){
    if(b.static||b.sleeping)continue;
    b.vy+=GRAV*dt;
    const ld=Math.exp(-.15*dt),ad=Math.exp(-.9*dt);
    b.vx*=ld;b.vy*=ld;b.va*=ad;
  }
  const ms=[];
  for(let i=0;i<bodies.length;i++)for(let j=i+1;j<bodies.length;j++){
    const a=bodies[i],b=bodies[j];
    if((a.static&&b.static)||(a.sleeping&&b.sleeping))continue;
    if(!aabbOverlap(a,b))continue;
    const m=collide(a,b);if(!m)continue;
    const relN=(b.vx-a.vx)*m.nx+(b.vy-a.vy)*m.ny;
    const aSpd=a.static?0:Math.hypot(a.vx,a.vy),bSpd=b.static?0:Math.hypot(b.vx,b.vy);
    if(relN<-26){wake(a);wake(b);}
    else{
      if(!a.sleeping&&(bSpd>45||Math.abs(b.va)>.6))wake(b);
      if(!b.sleeping&&(aSpd>45||Math.abs(a.va)>.6))wake(a);
    }
    if(relN<-60&&impactQueue.length<8)impactQueue.push({x:(a.x+b.x)/2,y:(a.y+b.y)/2,s:-relN});
    ms.push(m);
  }
  for(let it=0;it<10;it++)for(const m of ms)resolvePair(m);
  for(const b of bodies){if(b.static||b.sleeping)continue;b.x+=b.vx*dt;b.y+=b.vy*dt;b.angle+=b.va*dt;}
  for(let p=0;p<2;p++)for(const m of ms)correctPos(m);
  for(const b of bodies){
    if(b.static||b.sleeping)continue;
    if(Math.hypot(b.vx,b.vy)<7&&Math.abs(b.va)<.1){
      b.sleepT+=dt;if(b.sleepT>.45){b.sleeping=true;b.vx=b.vy=b.va=0;}
    }else b.sleepT=0;
  }
}

function update(dt){
  tSec+=dt;
  shake*=Math.exp(-5*dt);if(shake<.02)shake=0;
  for(let i=particles.length-1;i>=0;i--){const p=particles[i];p.life-=dt;
    if(p.life<=0){particles.splice(i,1);continue;}
    p.vy+=p.g*dt;p.x+=p.vx*dt;p.y+=p.vy*dt;}
  for(let i=floaters.length-1;i>=0;i--){const f=floaters[i];f.t+=dt;f.y-=26*dt;if(f.t>f.dur)floaters.splice(i,1);}
  if(impactQueue.length){
    for(const im of impactQueue){
      if(mode==='play'&&!paused){
        if(im.s>150){sfxThud(im.s);shake=Math.min(shake+im.s/260,8);spawnDust(im.x,im.y,Math.min(2+(im.s/120|0),7));}
        else if(im.s>55&&tSec-lastTick>.1){sfxTick();lastTick=tSec;}
      }
    }
    impactQueue.length=0;
  }
  if(paused)return;
  // removals
  for(let i=bodies.length-1;i>=0;i--){const b=bodies[i];
    if(!b.static&&b.y>H+140){
      if(b===droppedRef&&!droppedLost){droppedLost=true;dropLastX=clamp(b.x,80,W-80);}
      else if(mode==='play'&&(phase==='fall'||phase==='result'))toppledCount++;
      removeBody(b);
    }}
  // menu attract mode
  if(mode==='menu'){
    attractT-=dt;
    if(attractT<=0){
      attractT=rand(1.2,2.2);
      const b=makeBody(rand(300,660),70,CARD_W,CARD_H,false);
      b.angle=rand(-.25,.25);bodies.push(b);
      if(bodies.length>34){const old=bodies[1];if(old)removeBody(old);}
    }
  }
  if(mode!=='play')return;
  if(phase==='aim'){
    let mv=0;
    if(keys.ArrowLeft||keys.KeyA)mv--;
    if(keys.ArrowRight||keys.KeyD)mv++;
    const spd=(keys.ShiftLeft||keys.ShiftRight)?215:560;
    if(mv){aimX+=mv*spd*dt;pointerX=null;}
    else if(pointerX!=null)aimX+=(pointerX-aimX)*Math.min(1,dt*14);
    aimX=clamp(aimX,CARD_W/2+6,W-CARD_W/2-6);
    aimSpawnT=Math.min(1,aimSpawnT+dt*3.2);
  }else if(phase==='fall'){
    tSinceDrop+=dt;
    if(tSinceDrop>6||(tSinceDrop>.7&&bodiesSettled())||(droppedLost&&tSinceDrop>3&&bodiesSettled()))finishTurn();
  }else if(phase==='result'){
    resultT-=dt;
    if(resultT<=0){if(lives<=0)gameOver();else enterAim();}
  }
}

  const vg=ctx.createRadialGradient(W/2,H/2,H*.45,W/2,H/2,H*.95);
  vg.addColorStop(0,'rgba(0,0,0,0)');vg.addColorStop(1,'rgba(0,0,0,.42)');
  ctx.fillStyle=vg;ctx.fillRect(-30,-30,W+60,H+60);
  ctx.strokeStyle='rgba(224,178,92,.12)';ctx.lineWidth=1.5;ctx.strokeRect(1,1,W-2,H-2);
  ctx.restore();
}

/* ============ main loop ============ */
let last=performance.now(),acc=0;
function frame(t){
  requestAnimationFrame(frame);
  const dt=Math.min((t-last)/1000,.05);last=t;
  if(!paused&&(mode==='play'||mode==='menu')){
    acc+=dt;let n=0;
    while(acc>=STEP&&n<5){step(STEP);acc-=STEP;n++;}
    if(n===5)acc=0;
  }
  update(dt);
  render(dt);
}
