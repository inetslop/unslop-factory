/* ============================== WAVE LOGIC ============================== */
function hps(){ return 1+(wave-1)*.16+Math.max(0,wave-8)*.06; }
function spdScale(){ return Math.min(1.45,1+(wave-1)*.02); }
function buildWave(n){
  const q=[],total=6+Math.round(n*1.9);
  for(let i=0;i<total;i++){ let t='s'; const r=Math.random();
    if(n>=2&&r<clamp(.34-n*.02,.1,.34))t='m';
    if(n>=3&&Math.random()<clamp((n-2)*.05,.05,.24))t='l';
    q.push(t); }
  if(n%5===0)q.push('g');
  for(let i=q.length-1;i>0;i--){const j=irand(0,i);[q[i],q[j]]=[q[j],q[i]];}
  return q;
}

}

function makeRock(type,x,y,spd,hpsm){
  const R=ROCKS[type],r=R.r*rand(.92,1.08),n=irand(8,12),verts=[],craters=[];
  for(let i=0;i<n;i++)verts.push({a:i/n*TAU+rand(-.15,.15),f:rand(.74,1.16)});
  const kc=irand(1,3); for(let k=0;k<kc;k++)craters.push({x:rand(-.42,.42)*r,y:rand(-.42,.42)*r,rr:rand(.12,.28)*r});
  return {type,x,y,r,hp:R.hp*hpsm,maxHp:R.hp*hpsm,spd,rot:rand(0,TAU),vr:rand(-1.3,1.3)*(R.r>25?.5:1),
    wob:rand(0,TAU),wf:rand(.4,1.1),slowFrac:0,hitT:0,dead:false,verts,craters,hps:hpsm,cr:R.cr,dmg:R.dmg,name:R.name||null,born:0};
}
function spawnRock(type){
  const ang=rand(0,TAU),rr=Math.hypot(W,H)/2+80;
  const a=makeRock(type,cx+Math.cos(ang)*rr,cy+Math.sin(ang)*rr,ROCKS[type].spd*spdScale(),hps());
  asteroids.push(a);
  if(ROCKS[type].name){ flashMsg('!! GOLIATH DETECTED !!','alert',2.6); SFX.goliath(); shake=Math.max(shake,10); }
}
function killRock(a){
  if(a.dead)return; a.dead=true;
  const c=Math.round(a.cr*(1+(wave-1)*.04)); credits+=c; earned+=c; kills++;
  floater(a.x,a.y,'+'+c,'#ffd27a',13);
  boomFx(a.x,a.y,a.r,a.type==='g'?'#ff6a4a':'#ffb454');
  const sub=ROCKS[a.type].split;
  if(sub){ for(let k=0;k<2;k++){ const ang=Math.atan2(a.y-cy,a.x-cx)+(k?.55:-.55)+rand(-.2,.2);
    asteroids.push(makeRock(sub,a.x+Math.cos(ang)*a.r*.7,a.y+Math.sin(ang)*a.r*.7,ROCKS[sub].spd*spdScale(),a.hps)); } }
}
function damageRock(a,d){ if(a.dead)return; a.hp-=d; a.hitT=Math.max(a.hitT,.1); if(a.hp<=0)killRock(a); }
function impactBase(a){
  a.dead=true; hull-=a.dmg; hurtT=1; shake=Math.max(shake,8+a.r*.3);
  const d=Math.max(1,Math.hypot(a.x-cx,a.y-cy)),nx=(a.x-cx)/d,ny=(a.y-cy)/d;
  boomFx(cx+nx*(baseR+6),cy+ny*(baseR+6),Math.max(20,a.r),'#ff7a5e'); SFX.hit();
  if(hull<=0&&!dying){ hull=0; dying=true; dieT=1.9; shake=30;
    boomFx(cx,cy,90,'#7fd8ff'); boomFx(cx,cy,46,'#ffd9a0'); SFX.over(); }
}
function tstats(t){
  const b=TOWERS[t.type],l=t.lv;
  return { range:b.range*Math.pow(1.12,l-1), dmg:(b.dmg||0)*Math.pow(1.35,l-1), rate:(b.rate||1)*Math.pow(1.1,l-1),
    splash:(b.splash||0)*(1+.15*(l-1)), slow:b.slow?Math.min(.7,b.slow+.08*(l-1)):0, dps:(b.dps||0)*Math.pow(1.4,l-1), spd:b.speed||0 };
}
function upCost(t){ return Math.round(TOWERS[t.type].cost*(0.9+0.8*(t.lv-1))); }

  for(let i=0;i<n;i++){ const a=rand(0,TAU),s=rand(25,90+r*1.6);
    parts.push({type:'sh',x,y,vx:Math.cos(a)*s,vy:Math.sin(a)*s,rot:rand(0,TAU),vr:rand(-6,6),life:rand(.5,1.1),maxLife:1.1,size:rand(2,3+r*.14),c:['#8fa3c4','#6b7d9c','#a98f78'][irand(0,2)]}); }

  shocks.push({x,y,r0:5,maxR:r*2.6,t:0,life:.42,c:c||'#ffb454'});

}
function explode(x,y,r,dmg){
  for(const a of asteroids){ if(a.dead)continue; const d=Math.hypot(a.x-x,a.y-y);
    if(d<r+a.r*.5)damageRock(a,dmg*(1-.45*clamp(d/r,0,1))); }
  shocks.push({x,y,r0:6,maxR:r,t:0,life:.38,c:'#ffb454'});
  sparkFx(x,y,'#ffd9a0',10); SFX.boom(r*.7); shake=Math.max(shake,2.5);
}

function fireShot(t,st,tgt){
  const ang=t.ang,mx2=t.x+Math.cos(ang)*20,my2=t.y+Math.sin(ang)*20;
  if(t.type==='pulse'){ shots.push({type:'pulse',x:mx2,y:my2,vx:Math.cos(ang)*st.spd,vy:Math.sin(ang)*st.spd,dmg:st.dmg,life:0}); SFX.shoot(); sparkFx(mx2,my2,'#bff1ff',2); }
  else { shots.push({type:'flak',x:mx2,y:my2,ang,tgt,dmg:st.dmg,splash:st.splash,spd:st.spd,life:0}); SFX.flak(); }
}
function fireRail(t,st){
  const ang=t.lockAng,x2=t.x+Math.cos(ang)*st.range,y2=t.y+Math.sin(ang)*st.range;
  beams.push({x1:t.x,y1:t.y,x2,y2,t:0,life:.15,c:TOWERS.rail.color});
  for(const a of asteroids){ if(a.dead)continue; const dx=a.x-t.x,dy=a.y-t.y;
    const pr=dx*Math.cos(ang)+dy*Math.sin(ang); if(pr<0||pr>st.range)continue;
    if(Math.abs(-dx*Math.sin(ang)+dy*Math.cos(ang))<a.r*.85+9){ damageRock(a,st.dmg); sparkFx(a.x,a.y,'#ffc2de',6); } }
  SFX.rzap(); shake=Math.max(shake,3);
}
function updateTowers(dt){
  for(const t of towers){
    if(t.type==='cryo')continue;
    const st=tstats(t); t.cd-=dt;
    let tgt=null,best=-1e18;
    for(const a of asteroids){ if(a.dead)continue;
      if(dist2(t.x,t.y,a.x,a.y)>st.range*st.range)continue;
      const sc=t.type==='rail'?(a.hp+a.r):-dist2(a.x,a.y,cx,cy);
      if(sc>best){best=sc;tgt=a;} }
    t.tgt=tgt;
    if(tgt){ const da=Math.atan2(tgt.y-t.y,tgt.x-t.x);
      t.ang+=clamp(angDiff(da,t.ang),-TOWERS[t.type].rotSpd*dt,TOWERS[t.type].rotSpd*dt); }
    if(!tgt)continue;
    const ta=Math.atan2(tgt.y-t.y,tgt.x-t.x);
    if(t.type==='rail'){
      if(t.fireT>0){ t.fireT-=dt; t.lockAng=Math.atan2(tgt.y-t.y,tgt.x-t.x);
        if(t.fireT<=0){ fireRail(t,st); t.cd=1/st.rate; } }
      else if(t.cd<=0&&Math.abs(angDiff(ta,t.ang))<.15){ t.fireT=.34; t.lockAng=ta; SFX.rcharge(); }
    } else if(t.cd<=0&&Math.abs(angDiff(ta,t.ang))<.3){ fireShot(t,st,tgt); t.cd=1/st.rate; }
  }
}

/* ============================== UPDATE ============================== */
function updateAmbient(raw){
  for(const s of stars){ s.x+=s.vx*raw; s.y+=s.vy*raw;
    if(s.x<-5)s.x=W+5; if(s.x>W+5)s.x=-5; if(s.y>H+5)s.y=-5; }
  for(const r of ambRocks){ r.x+=r.vx*raw; r.y+=r.vy*raw; r.rot+=r.vr*raw;
    if(r.x<-70)r.x=W+70; if(r.y>H+70)r.y=-70; }
  shT-=raw;
  if(shT<=0){ const x=rand(0,W); shooters.push({x,y:-20,vx:rand(-260,260),vy:rand(420,680),t:0,life:.9}); shT=rand(5,12); }
  for(let i=shooters.length-1;i>=0;i--){ const s=shooters[i]; s.t+=raw; s.x+=s.vx*raw; s.y+=s.vy*raw; if(s.t>s.life)shooters.splice(i,1); }
  shake=Math.max(0,shake-40*raw); hurtT=Math.max(0,hurtT-1.6*raw); flashT=Math.max(0,flashT-raw);
}
function updateWorld(dt){
  if(dying){ dieT-=dt; if(dieT<=0&&mode==='playing'){ mode='gameover'; showOver(); } }
  if(!dying){
    if(phase==='prep'){ prepT-=dt; if(prepT<=0)startWave(); }
    else{
      if(spawnQueue.length&&asteroids.length<48){ spawnT-=dt;
        if(spawnT<=0){ spawnRock(spawnQueue.shift()); spawnT=Math.max(.45,1.5-wave*.04)*rand(.7,1.3); } }
      else if(!spawnQueue.length&&!asteroids.length)waveClear();
    }
  }
  // stasis pass
  for(const a of asteroids)if(!a.dead)a.slowFrac=0;
  for(const t of towers){ if(t.type!=='cryo')continue; const st=tstats(t);
    for(const a of asteroids){ if(a.dead)continue;
      if(dist2(t.x,t.y,a.x,a.y)<st.range*st.range){ a.slowFrac=Math.max(a.slowFrac,st.slow); damageRock(a,st.dps*dt); } } }
  // rocks
  for(const a of asteroids){ if(a.dead)continue;
    a.born+=dt; a.hitT-=dt;
    const dirA=Math.atan2(cy-a.y,cx-a.x),wa=dirA+Math.sin(tGlobal*a.wf+a.wob)*.3,sp=a.spd*(1-a.slowFrac);
    a.x+=Math.cos(wa)*sp*dt; a.y+=Math.sin(wa)*sp*dt; a.rot+=a.vr*dt;
    if(a.slowFrac>0&&Math.random()<dt*8){ const ra=rand(0,TAU);
      parts.push({type:'sp',x:a.x+Math.cos(ra)*a.r,y:a.y+Math.sin(ra)*a.r,vx:rand(-15,15),vy:rand(-15,15),life:.4,maxLife:.4,size:1.6,c:'#8affc1'}); }
    if(Math.hypot(a.x-cx,a.y-cy)<baseR+a.r*.42)impactBase(a);
  }
  updateTowers(dt);
  // shots
  for(let i=shots.length-1;i>=0;i--){ const s=shots[i]; s.life+=dt; let done=false;
    if(s.type==='flak'){
      if(s.tgt&&!s.tgt.dead)s.ang+=clamp(angDiff(Math.atan2(s.tgt.y-s.y,s.tgt.x-s.x),s.ang),-4*dt,4*dt);
      s.x+=Math.cos(s.ang)*s.spd*dt; s.y+=Math.sin(s.ang)*s.spd*dt;
      for(const a of asteroids){ if(a.dead)continue; if(dist2(s.x,s.y,a.x,a.y)<Math.pow(a.r*.75+8,2)){ explode(s.x,s.y,s.splash,s.dmg); done=true; break; } }
    } else {
      s.x+=s.vx*dt; s.y+=s.vy*dt;
      for(const a of asteroids){ if(a.dead)continue; if(dist2(s.x,s.y,a.x,a.y)<Math.pow(a.r*.9+4,2)){ damageRock(a,s.dmg); sparkFx(s.x,s.y,'#bff1ff',3); done=true; break; } }
    }
    if(done||s.life>2.5||s.x<-60||s.x>W+60||s.y<-60||s.y>H+60)shots.splice(i,1);
  }
  // beams / shocks / parts / floaters
  for(let i=beams.length-1;i>=0;i--){ beams[i].t+=dt; if(beams[i].t>beams[i].life)beams.splice(i,1); }
  for(let i=shocks.length-1;i>=0;i--){ shocks[i].t+=dt; if(shocks[i].t>shocks[i].life)shocks.splice(i,1); }
  for(let i=parts.length-1;i>=0;i--){ const p=parts[i]; p.life-=dt;
    if(p.life<=0){ parts.splice(i,1); continue; }
    p.x+=p.vx*dt; p.y+=p.vy*dt; p.vx*=Math.max(0,1-2*dt); p.vy*=Math.max(0,1-2*dt);
    if(p.type==='sh')p.rot+=p.vr*dt; }
  while(parts.length>450)parts.shift();
  for(let i=floaters.length-1;i>=0;i--){ floaters[i].t+=dt; if(floaters[i].t>floaters[i].life)floaters.splice(i,1); }
  for(let i=asteroids.length-1;i>=0;i--)if(asteroids[i].dead)asteroids.splice(i,1);
}

function drawStation(){

  for(const a of asteroids)if(!a.dead&&Math.hypot(a.x-cx,a.y-cy)<210){near=true;break;}

function drawRock(a){
  const hpF=clamp(a.hp/a.maxHp,0,1);
  ctx.save(); ctx.translate(a.x,a.y); ctx.rotate(a.rot);
  const gr=ctx.createRadialGradient(-a.r*.3,-a.r*.3,a.r*.2,0,0,a.r*1.15);
  if(a.type==='g'){ gr.addColorStop(0,'#6b4a52'); gr.addColorStop(1,'#241d33'); }
  else { gr.addColorStop(0,'#5c6d90'); gr.addColorStop(1,'#232c42'); }
  ctx.beginPath();
  for(let i=0;i<a.verts.length;i++){ const v=a.verts[i],x=Math.cos(v.a)*a.r*v.f,y=Math.sin(v.a)*a.r*v.f; i?ctx.lineTo(x,y):ctx.moveTo(x,y); }
  ctx.closePath(); ctx.fillStyle=gr; ctx.fill();
  if(a.type==='g'){ ctx.fillStyle=`rgba(255,90,60,${.22+.14*Math.sin(tGlobal*5)})`; ctx.beginPath(); ctx.arc(0,0,a.r*.45,0,TAU); ctx.fill(); }
  const hot=1-hpF;
  ctx.strokeStyle=`rgba(${70+hot*185},${90-hot*20},${130-hot*70},.9)`; ctx.lineWidth=a.type==='g'?3:2; ctx.stroke();
  ctx.fillStyle='rgba(0,0,0,.25)';
  for(const c of a.craters){ ctx.beginPath(); ctx.arc(c.x,c.y,c.rr,0,TAU); ctx.fill(); }
  if(a.hitT>0){ ctx.globalCompositeOperation='lighter'; ctx.fillStyle=`rgba(255,255,255,${clamp(a.hitT*4,0,.7)})`;
    ctx.beginPath(); ctx.arc(0,0,a.r,0,TAU); ctx.fill(); ctx.globalCompositeOperation='source-over'; }
  ctx.restore();
  if(hpF<1){ ctx.strokeStyle=hpF>.5?'#8affc1':hpF>.25?'#ffb454':'#ff5a5a'; ctx.globalAlpha=.75; ctx.lineWidth=2;
    ctx.beginPath(); ctx.arc(a.x,a.y,a.r+7,-Math.PI/2,-Math.PI/2+hpF*TAU); ctx.stroke(); ctx.globalAlpha=1; }
  if(a.slowFrac>0){ ctx.strokeStyle='rgba(138,255,193,.5)'; ctx.lineWidth=1.5;
    ctx.beginPath(); ctx.arc(a.x,a.y,a.r+4,0,TAU); ctx.stroke(); }

    ctx.fillStyle=`rgba(255,110,90,${clamp(3-a.born,0,1)})`; ctx.fillText('GOLIATH',a.x,a.y-a.r-16); }
}

    ctx.strokeStyle='#fff'; ctx.globalAlpha=f; ctx.lineWidth=2.5*f;
    ctx.beginPath(); ctx.moveTo(b.x1,b.y1); ctx.lineTo(b.x2,b.y2); ctx.stroke(); ctx.restore(); }

    else if(p.type==='sp'){ ctx.save(); ctx.globalCompositeOperation='lighter'; ctx.globalAlpha=f;
      ctx.fillStyle=p.c; ctx.beginPath(); ctx.arc(p.x,p.y,p.size,0,TAU); ctx.fill(); ctx.restore(); }
    else { ctx.save(); ctx.translate(p.x,p.y); ctx.rotate(p.rot); ctx.globalAlpha=f; ctx.fillStyle=p.c;
      const s2=p.size; ctx.beginPath(); ctx.moveTo(s2,0); ctx.lineTo(-s2*.7,s2*.6); ctx.lineTo(-s2*.5,-s2*.6); ctx.closePath(); ctx.fill(); ctx.restore(); } }

    ctx.fillStyle=f.c; ctx.fillText(f.txt,f.x,f.y-28*f.t); }
  ctx.globalAlpha=1;
  // placement ghost

    rangeCircle(mx,my,TOWERS[buildSel].range,col,false);

    ctx.fillStyle=g; ctx.fillRect(0,0,W,H); }
}
