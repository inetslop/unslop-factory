    case 'voids':emitters.push({kind:'voids',x:W/2,t:0,dur:2.8,tick:.3,ph:rnd(TAU)});break;
    case 'blooms':{const n=1+(wave>6?1:0);for(let i=0;i<n;i++)pushB(rnd(W*.2,W*.8),-30,rnd(-30,30),55*spdMul(),'bloom',2.4);}break;
    case 'drone':spawnDrone('turret');break;
    case 'caster':spawnDrone('caster');break;
  }
}

function popAt(px,py){
  if(state!=='play'||clickT>0){if(state==='play')sFizzle();return;}
  clickT=.2;
  shocks.push({x:px,y:py,t:0,maxR:90});
  let popped=0,gain=0,hitVoid=false,fx=px,fy=py;
  for(let i=bullets.length-1;i>=0;i--){
    const b=bullets[i],dx=b.x-px,dy=b.y-py,rr=90+b.r;
    if(dx*dx+dy*dy>rr*rr)continue;
    if(b.type==='void'){hitVoid=true;puff(b.x,b.y,'rgba(177,76,255,');spark(b.x,b.y,'rgba(120,40,180,',3,60);bullets.splice(i,1);continue;}
    bullets.splice(i,1);popped++;poppedTotal++;
    combo++;comboT=1.1;const m=Math.min(9,1+(combo/8|0));
    if(m>mult){mult=m;bumpCombo();}
    maxCombo=Math.max(maxCombo,m);gain+=BTYPE[b.type].score*m;
    spark(b.x,b.y,BTYPE[b.type].col,BTYPE[b.type].r>8?12:7);ringFx(b.x,b.y,b.r*3.4,BTYPE[b.type].col,1.5);
    if(popped===1){fx=b.x;fy=b.y;}
  }
  for(let i=drones.length-1;i>=0;i--){
    const d=drones[i],dx=d.x-px,dy=d.y-py;
    if(dx*dx+dy*dy>(120+d.r)*(120+d.r))continue;
    d.hp--;d.flash=.14;spark(d.x,d.y,'rgba(255,210,63,',8);sDroneHit();
    if(d.hp<=0)killDrone(d,i,true);
  }
  if(popped>0){score+=gain;popup(fx,fy-14,'+'+gain,popped>1?20:14,popped>3?'#ffd23f':'#fff');sPop(combo);shake=Math.min(7,2+popped*.9);}
  else if(hitVoid)sDeny();
  else{sFizzle;ringFx(px,py,26,'rgba(53,224,255,',1.5);}
}

  const pts=byPop?150*mult:150;score+=pts;popup(d.x,d.y-20,'+'+pts,18,'#ffd23f');
  if(Math.random()<.45)pickups.push({x:d.x,y:d.y,t:0});
}

function updPlayer(dt){
  const k=Math.min(1,dt*11);
  const nx=clamp(mouse.x,14,W-14),ny=clamp(mouse.y,14,H-14);
  player.tilt+=((nx-player.x)*.02-player.tilt)*Math.min(1,dt*8);
  player.x+=(nx-player.x)*k;player.y+=(ny-player.y)*k;
  if(Math.random()<.7)parts.push({k:3,x:player.x+rnd(-3,3),y:player.y+9,t:0,life:.3,sz:2.5,col:'rgba(53,224,255,'});
}
function updDrones(dt,hostile){
  for(let i=drones.length-1;i>=0;i--){const d=drones[i];d.t+=dt;d.flash-=dt;
    const dx=d.tx-d.x,dy=d.ty-d.y,ds=Math.hypot(dx,dy)||1,sp=60*dt;
    if(ds>2){d.x+=dx/ds*Math.min(sp,ds);d.y+=dy/ds*Math.min(sp,ds);}
    d.moveT+=dt;if(d.moveT>4){d.moveT=0;d.tx=rnd(W*.15,W*.85);d.ty=rnd(H*.12,H*.32);}
    if(hostile){d.fireT-=dt;
      if(d.fireT<=0){
        if(d.kind==='turret'){const base=Math.atan2(player.y-d.y,player.x-d.x);
          for(let k=-1;k<=1;k++)pushB(d.x,d.y+8,Math.cos(base+k*.16)*170*spdMul(),Math.sin(base+k*.16)*170*spdMul(),'ember');
          d.fireT=Math.max(.9,1.35-wave*.04);}
        else{ring(d.x,d.y+8,10+Math.min(6,wave),112*spdMul(),'ember');d.fireT=2.1;}
      }}
  }
}
function updBullets(dt){
  for(let i=bullets.length-1;i>=0;i--){const b=bullets[i];
    b.x+=b.vx*dt;b.y+=b.vy*dt;
    if(b.type==='bloom'){b.fuse-=dt;
      if(b.fuse<=0){ringFx(b.x,b.y,60,'rgba(255,79,163,',3);spark(b.x,b.y,'rgba(255,79,163,',14);
        ring(b.x,b.y,12,130*spdMul(),'ember');bullets.splice(i,1);continue;}}
    if(b.x<-70||b.x>W+70||b.y<-90||b.y>H+70)bullets.splice(i,1);
  }
}
function collisions(){
  if(player.inv>0)return;
  for(const b of bullets){const dx=b.x-player.x,dy=b.y-player.y,rr=b.r*.8+player.r;
    if(dx*dx+dy*dy<rr*rr){damage();return;}}
  for(const d of drones){const dx=d.x-player.x,dy=d.y-player.y,rr=d.r+player.r-3;
    if(dx*dx+dy*dy<rr*rr){damage();return;}}
  for(let i=pickups.length-1;i>=0;i--){const p=pickups[i],dx=p.x-player.x,dy=p.y-player.y;
    if(dx*dx+dy*dy<26*26){pickups.splice(i,1);if(bombs<5)bombs++;uiBombs();sPickup();popup(player.x,player.y-24,'+1 BOMB',15,'#ffd23f');}}
}
function updFx(dt){
  for(let i=parts.length-1;i>=0;i--){const p=parts[i];p.t+=dt;
    if(p.t>p.life){parts.splice(i,1);continue;}
    if(p.k===0||p.k===3){p.x+=p.vx*dt;p.y+=p.vy*dt;p.vx*=(1-3*dt);p.vy*=(1-3*dt);}
  }
  for(let i=pops.length-1;i>=0;i--){const p=pops[i];p.t+=dt;p.y-=34*dt;if(p.t>p.life)pops.splice(i,1);}
  for(let i=shocks.length-1;i>=0;i--){shocks[i].t+=dt;if(shocks[i].t>.32)shocks.splice(i,1);}
  for(let i=pickups.length-1;i>=0;i--){const p=pickups[i];p.t+=dt;p.y+=45*dt;if(p.y>H+30)pickups.splice(i,1);}
}
function updEmitters(dt){
  for(let i=emitters.length-1;i>=0;i--){const em=emitters[i];em.t+=dt;
    if(em.kind==='spiral'){em.tick-=dt;
      while(em.tick<=0){em.tick+=em.rate;em.shots++;
        const dartB=em.shots%4===3,sp=(dartB?165:108)*spdMul();
        for(let k=0;k<2;k++){const a=em.ang+k*Math.PI;pushB(em.x,em.y,Math.cos(a)*sp,Math.sin(a)*sp,dartB?'dart':'ember');}
        em.ang+=.47;}
    }else if(em.kind==='voids'){em.tick-=dt;
      while(em.tick<=0){em.tick+=em.rate||.3;
        pushB(em.x+Math.sin(em.t*1.5+em.ph)*W*.32,-20,0,140*spdMul(),'void');}}
    if(em.t>em.dur)emitters.splice(i,1);
  }
}
function attract(dt){
  attractT-=dt;if(attractT>0)return;attractT=1.6;
  const r=Math.random();
  if(r<.45)ring(rnd(W*.2,W*.8),rnd(H*.2,H*.7),14,rnd(50,90),pick(['ember','dart','bloom']));
  else if(r<.75)emitters.push({kind:'spiral',x:rnd(W*.3,W*.7),y:rnd(H*.2,H*.6),t:0,dur:2.4,tick:0,ang:rnd(TAU),rate:.1,shots:0});
  else pushB(rnd(0,W),-20,rnd(-20,20),rnd(60,110),pick(['ember','void','dart']));
}
function step(dt){
  vt+=dt;
  if(state==='play'){
    tGame+=dt;
    const w=1+(tGame/20|0);if(w>wave){wave=w;onWave();}
    ambT-=dt;if(ambT<=0){ambT=Math.max(.35,.95-wave*.06)*rnd(.7,1.4);volley(1);}
    evT-=dt;if(evT<=0)scheduleEvent();
    updEmitters(dt);updPlayer(dt);updDrones(dt,true);updBullets(dt);collisions();
    comboT-=dt;if(comboT<0&&combo>0){combo=0;mult=1;bumpCombo(true);}
    clickT=Math.max(0,clickT-dt);bombCD=Math.max(0,bombCD-dt);player.inv-=dt;
  }else{
    attract(dt);updEmitters(dt);updDrones(dt,false);updBullets(dt);
  }
  updFx(dt);
  shake=Math.max(0,shake-shake*7*dt-2*dt);
  shakeX=rnd(-1,1)*shake;shakeY=rnd(-1,1)*shake;
}
