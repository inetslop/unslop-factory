function updatePlaying(dt){
  const p=player;
  if(p.inv>0)p.inv-=dt;p.fireCd-=dt;p.kickCd-=dt;
  if(p.kickT>0)p.kickT-=dt;if(p.muzT>0)p.muzT-=dt;
  p.recoil=Math.max(0,p.recoil-dt*8);if(p.kbT>0)p.kbT-=dt;
  const ax=(keys.has('d')||keys.has('arrowright')?1:0)-(keys.has('a')||keys.has('arrowleft')?1:0);
  if(ax){p.facing=ax;if(p.kbT<=0){p.vx+=ax*2600*dt;p.vx=clamp(p.vx,-272,272);}}
  else{p.vx*=Math.exp(-(p.onGround?10:2.5)*dt);if(Math.abs(p.vx)<4)p.vx=0;}
  p.vy=Math.min(950,p.vy+GRAV*dt);
  if(p.jbuf>0){
    p.jbuf-=dt;
    if(p.onGround||p.coyote>0){p.vy=-585;p.jbuf=0;p.coyote=0;p.airJ=1;sfx.jump();dust(p.x+p.w/2,p.y+p.h,6);}
    else if(p.airJ<2){p.vy=-520;p.jbuf=0;p.airJ=2;sfx.djump();ring(p.x+p.w/2,p.y+p.h);}
  }
  movePhys(p,dt);
  if(p.onGround){p.coyote=0.09;p.airJ=0;lastSafe={x:p.x,y:p.y};
    if(Math.abs(p.vx)>140&&Math.random()<dt*7)dust(p.x+p.w/2-p.facing*8,p.y+p.h,1);}
  else p.coyote-=dt;
  p.anim+=dt*(2+Math.abs(p.vx)*0.05);
  if(p.x<30){p.x=30;if(p.vx<0)p.vx=0;}
  if(p.x+p.w/2>p.maxX)p.maxX=p.x+p.w/2;
  if(p.y>VIEW_H+140){
    hurt(35,p.x-60,true);
    p.x=lastSafe.x;p.y=lastSafe.y-4;p.vx=0;p.vy=0;p.inv=Math.max(p.inv,1.3);
  }
  genWorld(camX+VW*2.4);
  const sh=keys.has('x')||keys.has('z')||keys.has('j')||mouseDown;
  if(sh){if(p.ammo>0&&p.fireCd<=0)fire();else if(p.ammo<=0&&p.kickCd<=0)doKick();}
  spawnT-=dt;
  const target=Math.min(4+Math.floor(distM()/240),15);
  if(spawnT<=0){
    if(zombies.length<target)
      spawnZombie(Math.random()<0.7?camX+VW*(0.9+Math.random()*0.35):Math.max(60,camX-80-Math.random()*200));
    spawnT=Math.max(0.45,1.15-distM()/3000);
  }
  const dm=distM();
  if(dm>=nextMile){nextMile+=500;showBanner(nextMile+' m — THE HORDE GROWS');sfx.mile();}
  if(p.hp<30&&!lowWarned){lowWarned=true;showBanner('WOUNDS CRITICAL');}
  updateZombies(dt,false);
  updateBullets(dt);updatePickups(dt);updateParts(dt);
  const tx=p.x+p.w/2-VW*0.38+p.facing*VW*0.08;
  camX+=(tx-camX)*Math.min(1,dt*5);
  if(camX<-140)camX=-140;
  updateHUD();
}
function update(dt){
  t+=dt;updateAmbient(dt);
  if(state==='menu'){
    camX+=dt*36;genWorld(camX+VW*2.5);
    mSpawnT-=dt;if(mSpawnT<=0&&zombies.length<6){spawnZombie(camX+VW+70);mSpawnT=1.6;}
    updateZombies(dt,true);updateParts(dt);return;
  }
  if(state==='paused')return;
  if(state==='dying'){
    deathT-=dt;updateZombies(dt,false);updateParts(dt);
    if(deathT<=0)finishOver();return;
  }
  if(state==='over'){updateParts(dt);return;}
  updatePlaying(dt);
}

function branch(x,y,a,len,d,seed){
  if(d===0||len<5)return;
  const x2=x+Math.cos(a)*len,y2=y+Math.sin(a)*len;
  ctx.lineWidth=Math.max(1,d*1.7);
  ctx.beginPath();ctx.moveTo(x,y);ctx.lineTo(x2,y2);ctx.stroke();

    branch(x2,y2,a+(h(seed*1.3+i)-0.5)*1.5,len*(0.6+h(seed+i)*0.14),d-1,seed*1.7+i);
}

    const tx=b.x-Math.sign(b.vx)*14;
    ctx.globalCompositeOperation='lighter';
    ctx.strokeStyle='rgba(255,200,130,.4)';ctx.lineWidth=6;
    ctx.beginPath();ctx.moveTo(tx,b.y);ctx.lineTo(b.x,b.y);ctx.stroke();
    ctx.strokeStyle='#ffe9c8';ctx.lineWidth=2.5;
    ctx.beginPath();ctx.moveTo(tx,b.y);ctx.lineTo(b.x,b.y);ctx.stroke();
    ctx.globalCompositeOperation='source-over';
  }
}
function drawParts(){
  for(const q of parts){
    const a=Math.max(0,q.life/q.maxLife);
    if(q.type==='dot'){ctx.globalAlpha=a;ctx.fillStyle=q.col;
      ctx.beginPath();ctx.arc(q.x,q.y,q.size,0,7);ctx.fill();}
    else if(q.type==='case'){ctx.globalAlpha=a;ctx.save();ctx.translate(q.x,q.y);ctx.rotate(q.rot||0);
      ctx.fillStyle='#d8b45a';ctx.fillRect(-2.5,-1.5,5,3);ctx.restore();}
    else if(q.type==='ring'){ctx.globalAlpha=a*0.8;ctx.strokeStyle=q.col;ctx.lineWidth=2;
      ctx.beginPath();ctx.arc(q.x,q.y,q.size+(1-a)*26,0,7);ctx.stroke();}
    else if(q.type==='wisp'){ctx.globalAlpha=a*0.5;ctx.globalCompositeOperation='lighter';
      ctx.fillStyle=q.col;ctx.beginPath();
      ctx.arc(q.x+Math.sin(q.life*9)*4,q.y,q.size*a+1,0,7);ctx.fill();
      ctx.globalCompositeOperation='source-over';}
  }
  ctx.globalAlpha=1;
}
function drawFloaters(){
  ctx.font='20px Impact,"Arial Black",sans-serif';ctx.textAlign='center';
  for(const f of floaters){
    ctx.globalAlpha=Math.max(0,f.life);
    ctx.fillStyle='#06110a';ctx.fillText(f.txt,f.x+1,f.y+1);
    ctx.fillStyle=f.col;ctx.fillText(f.txt,f.x,f.y);
  }
  ctx.globalAlpha=1;ctx.textAlign='left';
}
function drawFog(){
  for(let i=0;i<6;i++){
    const span=VW*1.5;
    const fx=camX+(((i*span/6+t*(12+i*4))%span)+span)%span-VW*0.2;
    ctx.fillStyle='rgba(140,210,185,'+(0.035+(i%3)*0.012).toFixed(3)+')';
    ctx.beginPath();ctx.ellipse(fx,GY-30-(i%3)*26,190+i*32,26,0,0,7);ctx.fill();
  }
}
function render(){
  const shx=(Math.random()*2-1)*shakeM,shy=(Math.random()*2-1)*shakeM;
  ctx.setTransform(DPR,0,0,DPR,0,0);
  drawSky();drawSkyline();drawMidground();
  ctx.save();ctx.scale(SCALE,SCALE);ctx.translate(-camX+shx,shy);
  drawGround();drawDecals();drawCrates();drawDecos();drawPickups();
  for(const z of zombies)if(z.x+z.w>camX-60&&z.x<camX+VW+60)drawZombie(z);
  if(state!=='over'&&state!=='dying')drawPlayer();
  drawBullets();drawParts();drawFog();
  if(state==='playing')drawLight();
  drawFloaters();
  ctx.restore();
  // screen-space overlays
  let g=ctx.createLinearGradient(0,H*0.7,0,H);
  g.addColorStop(0,'rgba(140,210,185,0)');g.addColorStop(1,'rgba(140,210,185,.09)');
  ctx.fillStyle=g;ctx.fillRect(0,0,W,H);
  g=ctx.createRadialGradient(W/2,H*0.45,Math.min(W,H)*0.35,W/2,H*0.5,Math.max(W,H)*0.75);
  g.addColorStop(0,'rgba(2,4,8,0)');g.addColorStop(1,'rgba(2,4,8,.6)');
  ctx.fillStyle=g;ctx.fillRect(0,0,W,H);
  if(flashT>0.01){ctx.fillStyle='rgba(205,228,255,'+(flashT*0.3).toFixed(3)+')';ctx.fillRect(0,0,W,H);}
  if(hurtFlash>0.02){ctx.fillStyle='rgba(140,20,16,'+(hurtFlash*0.32).toFixed(3)+')';ctx.fillRect(0,0,W,H);}
  if(state==='playing'&&player.hp<30&&player.hp>0){
    ctx.fillStyle='rgba(160,25,18,'+(0.1+0.07*Math.sin(t*6)).toFixed(3)+')';ctx.fillRect(0,0,W,H);
  }
}
