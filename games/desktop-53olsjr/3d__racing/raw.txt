let segments=[],trackLength=0;
function lastY(){return segments.length?segments[segments.length-1].p2.world.y:0}
function addSegment(curve,y){const i=segments.length;
  segments.push({index:i,curve,sprites:[],cars:[],fog:1,clip:0,looped:false,
    p1:{world:{y:lastY(),z:i*SEG},screen:{}},p2:{world:{y:y,z:(i+1)*SEG},screen:{}}})}
function addRoad(enter,hold,leave,curve,hill){const sY=lastY(),eY=sY+hill*SEG,total=enter+hold+leave;
  for(let i=0;i<enter;i++)addSegment(easeIn(0,curve,i/enter),easeInOut(sY,eY,i/total));
  for(let i=0;i<hold;i++)addSegment(curve,easeInOut(sY,eY,(enter+i)/total));
  for(let i=0;i<leave;i++)addSegment(easeInOut(curve,0,i/leave),easeInOut(sY,eY,(enter+hold+i)/total))}

function buildTrack(){
  segments=[];
  addRoad(40,60,40, 0, 0);      // launch straight
  addRoad(50,90,50, 2.5, 20);   // gentle right, climb
  addRoad(40,60,40,-3.0,-10);   // left dip
  addRoad(30,80,30, 0, 40);     // crest straight
  addRoad(60,120,60, 4.0,-30);  // long fast right downhill
  addRoad(20,40,20, 0, 0);      // breather
  addRoad(50,70,50,-4.5, 15);   // left climb
  addRoad(40,40,40, 3.5,-25);   // chicane right down
  addRoad(30,60,30,-3.0, -5);   // left (closes elevation: sum = 0)
  addRoad(20,80,20, 0, 0);      // back straight
  addRoad(50,100,50, 4.5, 10);  // sweeper right
  addRoad(40,60,40,-2.5,-15);   // cool-down left
  addRoad(30,30,30, 0, 0);      // final straight to line
  trackLength=segments.length*SEG;
}

function findSegment(z){z=((z%trackLength)+trackLength)%trackLength;
  return segments[Math.floor(z/SEG)%segments.length]}
function updateCars(dt){
  for(const s of segments)s.cars.length=0;
  for(const c of cars){
    let closest=null,gapBest=1e9;
    for(const o of cars){if(o===c)continue;
      const gap=((o.z-c.z)%trackLength+trackLength)%trackLength;
      if(gap<SEG*14&&Math.abs(o.offset-c.offset)<0.42&&gap<gapBest){gapBest=gap;closest=o}}
    let target=c.base;
    if(closest&&gapBest<(c.speed-closest.speed)*1.6+SEG*4)target=closest.speed*.97;
    c.speed+=(target-c.speed)*Math.min(1,dt*2.5);
    c.z=(c.z+c.speed*dt)%trackLength;
    c.percent=(c.z%SEG)/SEG;
    findSegment(c.z).cars.push(c);
  }
}
function spawn(x,y,vx,vy,life,size,col){if(parts.length<120)parts.push({x,y,vx,vy,t:0,life,size,col})}

    if(p.t>p.life){parts.splice(i,1);continue}
    p.x+=p.vx*dt;p.y+=p.vy*dt;p.vy-=20*dt}
}

function update(dt){
  time+=dt;
  const pSeg=findSegment(position+PLAYER_Z);
  const sp=speed/MAX_SPEED;
  hitTimer=Math.max(0,hitTimer-dt);
  shake=Math.max(0,shake-shake*7*dt-2*dt);

  if(state==='countdown'){
    countT-=dt;
    const c=Math.ceil(countT);
    if(c<lastCount&&c>=1)beep(440,.14,'square',.2);
    lastCount=c;
    updateCars(dt);updateParticles(dt);updateAudio();return;
  }
  if(state==='racing'&&countT<=-1){/*noop*/}
  if(countT<=0&&state==='countdown'){state='racing';raceT=0;beep(880,.4,'square',.22);toast('GO!',110,1)}

  // ---- drive ----
  let throttle=K.u||autoGas,brake=K.d,hand=K.hb;
  if(state==='menu'){throttle=true;brake=false;hand=false}
  if(state==='finished'){throttle=false}

  if(throttle)speed+=ACCEL*dt*(1-sp*.35);
  else speed+=DECEL*dt;
  if(brake)speed+=BRAKE*dt;
  if(hand)speed+=(BRAKE*1.6)*dt;
  const offroad=Math.abs(playerX)>1.02;
  if(offroad&&speed>OFF_LIMIT){speed+=OFF_DECEL*dt;
    shake=Math.max(shake,sp*3);
    if(Math.random()<.5)spawn(W/2+(Math.random()<.5?-1:1)*W*.1,H*.94,(Math.random()-.5)*80,-(30+Math.random()*60),
      .5,5+Math.random()*5,'rgba(214,160,100,.8)')}
  speed=clamp(speed,0,MAX_SPEED);

  // ---- steer ----
  const dx=dt*STEER_RATE*sp;
  let steer=(K.l?-1:0)+(K.r?1:0);
  if(state==='menu'){ // autopilot
    let target=-pSeg.curve*.14*sp;

    if(state==='racing'){

    }
  }

  // ---- collisions (racing only) ----
  if(state==='racing'){
    const pWorldZ=position+PLAYER_Z;
    const ps=findSegment(pWorldZ);
    if(hitTimer<=0){
      for(const c of ps.cars){if(speed>c.speed&&Math.abs(playerX-c.offset)<0.3)crash(c)}
      for(const s of ps.sprites){const d=SPR[s.type];
        if(d.noCollide)continue;

      }
    }
  }

  bgX+=pSeg.curve*sp*dt*90;

  // ---- HUD ----
  if(state==='racing'||state==='countdown'){
    speedVal.textContent=Math.round(sp*320);
    speedBar.style.transform='scaleX('+sp.toFixed(3)+')';
    lapNum.textContent=Math.min(currentLap,TOTAL_LAPS)+'/'+TOTAL_LAPS;
    timeVal.textContent=fmt(state==='racing'?raceT:0);
    lastLapEl.textContent=laps.length?('LAST '+fmt(laps[laps.length-1])):'LAST —:——.——';
  }
  updateAudio();
}
