}

function audio(){

function tone(f,d,type,vol,f2,delay=0){
  const ac=audio();if(!ac||S.muted)return;
  const t=ac.currentTime+delay,o=ac.createOscillator(),g=ac.createGain();
  o.type=type;o.frequency.setValueAtTime(f,t);

let w=0,h=0,dpr=1,time=0;
let L={};
let clouds=[],flowers=[];
function resize(){
  w=innerWidth;h=innerHeight;dpr=Math.min(2,devicePixelRatio||1);
  cv.width=w*dpr;cv.height=h*dpr;ctx.setTransform(dpr,0,0,dpr,0,0);
  layout();
  $('#toast').style.left=((w-L.shopW)/2)+'px';
}
function layout(){
  const mobile=w<720,sheet=mobile&&document.body.classList.contains('shopOpen');
  const shopW=mobile?0:Math.min(340,w*.34);
  const playW=w-shopW;
  const playH=(mobile&&sheet)?h*.56:h;
  L={shopW,playW,playH,
     horizon:playH*.44,
     fwBot:playH*.80,fringeBot:playH*.87,
     teeX:playW*.32,teeY:playH*.72,
     holeX:playW*.70,holeY:playH*.60};
  L.R=clamp(Math.min(playW,playH)*.035,13,30);
  if(!clouds.length)for(let i=0;i<6;i++)clouds.push({x:rand(0,w+200),u:rand(.08,.55),s:rand(.55,1.35),v:rand(4,13)});
  flowers=[];for(let i=0;i<20;i++)flowers.push({u:Math.random(),v:rand(.1,.92),c:['#ffffff','#ffe9a8','#ffb3c1'][i%3],r:rand(1.6,3)});
}
addEventListener('resize',resize);

/* entities */
let shots=[],parts=[],texts=[],conf=[],ripples=[];
let putterT=99,flagKick=0,shake=0,ghostTimer=0,px=0,py=0,tx=0,ty=0,hintT=0;

function launchBall(crit,amt,ghost){
  if(shots.length>16)return;
  const g=ghost?rand(-70,70):0;
  shots.push({t:0,dur:ghost?rand(.85,1.1):.5,crit,amt,ghost,
    sx:L.teeX+g,sy:L.teeY+(ghost?rand(-12,26):0),
    tx:L.holeX,ty:L.holeY});
}
function plop(s){
  flagKick=Math.max(flagKick,s.ghost?.4:1);
  ripples.push({x:s.tx,y:s.ty,r:L.R*.5,life:1});
  if(s.ghost){burstCoins(s.tx,s.ty,3,false);return;}
  const n=s.crit?26:8+Math.min(12,Math.floor(Math.log10(perClick()+1)*3));
  burstCoins(s.tx,s.ty,n,s.crit);
  texts.push({x:s.tx,y:s.ty-24,v:-48,life:1.15,age:0,text:'+$'+fmt(s.amt),
    color:s.crit?'#ffd24a':'#ffffff',size:s.crit?30:Math.min(26,15+L.R*.3)});
  if(s.crit){shake=9;texts.push({x:L.playW/2,y:L.horizon+40,v:-26,life:1.6,age:0,text:'HOLE IN ONE! ×10',color:'#ffd24a',size:44});}
  SFX.tock();SFX.coin(s.crit);
}
function burstCoins(x,y,n,crit){
  for(let i=0;i<n;i++){const a=rand(-Math.PI*.95,-Math.PI*.05),sp=rand(90,crit?340:230);
    parts.push({x,y,vx:Math.cos(a)*sp,vy:Math.sin(a)*sp,r:rand(3.5,crit?8:6.5),life:rand(.6,1)});}
}
function confetti(){
  const cols=['#ffd24a','#ff5347','#4fc3f7','#8fe39a','#ffffff'];
  for(let i=0;i<90;i++)conf.push({x:rand(0,w),y:rand(-h*.2,0),vx:rand(-60,60),vy:rand(40,150),
    rot:rand(0,7),vr:rand(-6,6),w:rand(5,9),h:rand(8,14),c:cols[i%cols.length],life:rand(1.8,2.8)});
}

/* drawing helpers */
function drawBall(x,y,r,ghost){
  ctx.save();if(ghost)ctx.globalAlpha=.55;
  const g=ctx.createRadialGradient(x-r*.35,y-r*.4,r*.15,x,y,r);
  if(ghost){g.addColorStop(0,'#eaf7ff');g.addColorStop(1,'#8fb8d8');}
  else{g.addColorStop(0,'#ffffff');g.addColorStop(.75,'#f2f7fa');g.addColorStop(1,'#c9dbe6');}
  ctx.fillStyle=g;ctx.beginPath();ctx.arc(x,y,r,0,7);ctx.fill();
  ctx.strokeStyle='rgba(40,80,110,.35)';ctx.lineWidth=1;ctx.stroke();
  if(r>8){ctx.fillStyle='rgba(110,150,175,.35)';
    for(let i=0;i<6;i++){const a=i*1.05+.5,dx=Math.cos(a)*r*.45,dy=Math.sin(a)*r*.45;
      ctx.beginPath();ctx.arc(x+dx,y+dy,r*.09,0,7);ctx.fill();}}
  ctx.restore();
}
function drawShadow(x,y,rx,ry,a){ctx.fillStyle=`rgba(10,40,25,${a})`;ctx.beginPath();ctx.ellipse(x,y,rx,ry,0,0,7);ctx.fill();}

const farHillY=x=>L.horizon-30-Math.sin(x*.005)*14-Math.sin(x*.013+2)*8;
const nearHillY=x=>L.horizon-5-Math.sin(x*.007+1)*10-Math.sin(x*.02)*5;
const TREES=[.05,.14,.23,.36,.47,.58,.69,.8,.9];

function render(){
  ctx.clearRect(0,0,w,h);
  ctx.save();
  if(shake>0)ctx.translate(rand(-shake,shake),rand(-shake,shake));
  /* sky */
  const sky=ctx.createLinearGradient(0,0,0,L.horizon+40);
  sky.addColorStop(0,'#59bdf0');sky.addColorStop(.7,'#a8e2fb');sky.addColorStop(1,'#e6fafc');
  ctx.fillStyle=sky;ctx.fillRect(-20,-20,w+40,L.horizon+60);
  /* sun */
  const sx=L.playW*.14,sy=L.horizon*.3,sp=1+Math.sin(time*1.4)*.05;
  const sg=ctx.createRadialGradient(sx,sy,4,sx,sy,90*sp);
  sg.addColorStop(0,'rgba(255,247,207,.95)');sg.addColorStop(.35,'rgba(255,236,150,.5)');sg.addColorStop(1,'rgba(255,236,150,0)');
  ctx.fillStyle=sg;ctx.beginPath();ctx.arc(sx,sy,90*sp,0,7);ctx.fill();
  ctx.strokeStyle='rgba(255,240,180,.35)';ctx.lineWidth=3;
  for(let i=0;i<8;i++){const a=time*.15+i*Math.PI/4,r1=38*sp,r2=54*sp;
    ctx.beginPath();ctx.moveTo(sx+Math.cos(a)*r1,sy+Math.sin(a)*r1);ctx.lineTo(sx+Math.cos(a)*r2,sy+Math.sin(a)*r2);ctx.stroke();}
  /* clouds */
  ctx.save();ctx.translate(px*4,0);ctx.fillStyle='rgba(255,255,255,.9)';
  for(const c of clouds){const cy=L.horizon*c.u;
    ctx.beginPath();
    ctx.ellipse(c.x,cy,46*c.s,15*c.s,0,0,7);
    ctx.ellipse(c.x-30*c.s,cy+6*c.s,26*c.s,11*c.s,0,0,7);
    ctx.ellipse(c.x+32*c.s,cy+5*c.s,28*c.s,12*c.s,0,0,7);
    ctx.fill();}
  ctx.restore();
  /* far hill + trees */
  ctx.save();ctx.translate(px*6,0);
  ctx.fillStyle='#9adfae';ctx.beginPath();ctx.moveTo(-40,L.horizon+50);
  for(let x=-40;x<=w+40;x+=16)ctx.lineTo(x,farHillY(x));
  ctx.lineTo(w+40,L.horizon+50);ctx.closePath();ctx.fill();
  TREES.forEach((u,i)=>{const x=u*(w+80)-40,y=farHillY(x)+2,sw=Math.sin(time*1.5+i)*2,s=13+((i*7)%9);
    ctx.fillStyle='#37945c';ctx.beginPath();
    ctx.moveTo(x+sw*.6,y-s*1.7);ctx.lineTo(x-s,y);ctx.lineTo(x+s,y);ctx.closePath();ctx.fill();
    ctx.beginPath();ctx.moveTo(x+sw,y-s*2.2);ctx.lineTo(x-s*.8,y-s*.8);ctx.lineTo(x+s*.8,y-s*.8);ctx.closePath();ctx.fill();});
  ctx.restore();
  /* near hill */
  ctx.save();ctx.translate(px*12,0);
  ctx.fillStyle='#6cc47f';ctx.beginPath();ctx.moveTo(-40,L.horizon+50);
  for(let x=-40;x<=w+40;x+=16)ctx.lineTo(x,nearHillY(x));
  ctx.lineTo(w+40,L.horizon+50);ctx.closePath();ctx.fill();
  ctx.restore();
  /* fairway mow stripes (perspective trapezoid) */
  const topL=L.playW*.16,topR=L.playW*.84,N=9;
  for(let i=0;i<N;i++){const t0=i/N,t1=(i+1)/N;
    ctx.fillStyle=i%2?'#5ec66e':'#4db35d';
    ctx.beginPath();
    ctx.moveTo(lerp(topL,0,t0),L.horizon);ctx.lineTo(lerp(topL,0,t1),L.horizon);
    ctx.lineTo(lerp(topR,L.playW,t1),L.fwBot);ctx.lineTo(lerp(topR,L.playW,t0),L.fwBot);
    ctx.closePath();ctx.fill();}
  /* fringe + front green */
  ctx.fillStyle='#3f9e51';ctx.fillRect(-20,L.fwBot,L.playW+40,L.fringeBot-L.fwBot);
  ctx.fillStyle='#6fd07b';ctx.fillRect(-20,L.fringeBot,L.playW+40,h-L.fringeBot+20);
  ctx.fillStyle='rgba(180,255,190,.5)';ctx.fillRect(-20,L.fringeBot,L.playW+40,3);
  for(const f of flowers){const x=f.u*L.playW,y=lerp(L.fringeBot+12,h-10,f.v);
    ctx.fillStyle=f.c;ctx.beginPath();ctx.arc(x,y,f.r,0,7);ctx.fill();}
  /* hole + flag */
  const hx=L.holeX,hy=L.holeY,hr=L.R*.46;
  drawShadow(hx,hy+3,hr*1.5,hr*.7,.28);
  ctx.fillStyle='#08130c';ctx.beginPath();ctx.ellipse(hx,hy,hr*1.15,hr*.5,0,0,7);ctx.fill();
  ctx.strokeStyle='rgba(255,255,255,.85)';ctx.lineWidth=2;ctx.beginPath();ctx.ellipse(hx,hy,hr*1.15,hr*.5,0,0,7);ctx.stroke();
  for(const rp of ripples){ctx.strokeStyle=`rgba(255,255,255,${rp.life*.6})`;ctx.lineWidth=2;
    ctx.beginPath();ctx.ellipse(rp.x,rp.y,rp.r,rp.r*.45,0,0,7);ctx.stroke();}
  const poleL=L.R*4.6,wave=Math.sin(time*6)*(1+flagKick)*hr*.3;
  ctx.strokeStyle='#f2f2f2';ctx.lineWidth=2.5;ctx.beginPath();ctx.moveTo(hx,hy-2);ctx.lineTo(hx,hy-poleL);ctx.stroke();
  ctx.fillStyle='#ff5347';ctx.beginPath();
  ctx.moveTo(hx+1,hy-poleL);ctx.quadraticCurveTo(hx+hr*1.6+wave,hy-poleL+hr*.3,hx+1,hy-poleL+hr*.75);ctx.closePath();ctx.fill();
  ctx.fillStyle='rgba(0,0,0,.15)';ctx.beginPath();
  ctx.moveTo(hx+1,hy-poleL+hr*.25);ctx.quadraticCurveTo(hx+hr*1.4+wave,hy-poleL+hr*.4,hx+1,hy-poleL+hr*.7);ctx.closePath();ctx.fill();
  /* tee + idle ball */
  const R=L.R,showIdle=!shots.some(s=>!s.ghost&&s.t<.3);
  if(showIdle){
    const bob=Math.sin(time*2.4)*2;
    ctx.fillStyle='#e2543f';ctx.beginPath();
    ctx.moveTo(L.teeX-R*.5,L.teeY+R*.95);ctx.lineTo(L.teeX+R*.5,L.teeY+R*.95);
    ctx.lineTo(L.teeX+R*.3,L.teeY+R*.35);ctx.lineTo(L.teeX-R*.3,L.teeY+R*.35);ctx.closePath();ctx.fill();
    drawShadow(L.teeX,L.teeY+R*.9,R*1.05,R*.4,.32);
    drawBall(L.teeX,L.teeY-R+bob*.4,R,false);
  }
  /* flying balls */
  for(const s of shots){
    const t=s.t,arc=Math.hypot(s.tx-s.sx,s.ty-s.sy)*.38;
    const x=lerp(s.sx,s.tx,t),gy=lerp(s.sy,s.ty,t),y=gy-Math.sin(Math.PI*t)*arc;
    const r=R*(s.ghost?.72:1)*(1-.58*t);
    drawShadow(x,gy,r*1.05*(1-t*.4),r*.42*(1-t*.35),.3*(1-t*.5));
    drawBall(x,y,r,s.ghost);
  }
  /* putter */
  const P={x:L.teeX-3.6*R,y:L.teeY+.9*R},len=3.4*R;
  let a=-1.15+Math.sin(time*2)*.02,swingPh=0;
  if(putterT<1){const t=putterT;swingPh=t<.4?t/.4:1-(t-.4)/.6;
    a=-1.15+1.7*(1-Math.pow(1-clamp(swingPh,0,1),3));}
  const H={x:P.x+Math.cos(a)*len,y:P.y+Math.sin(a)*len};
  if(swingPh>.25&&putterT<1){ctx.strokeStyle=`rgba(255,255,255,${(1-putterT)*.5})`;ctx.lineWidth=4;
    ctx.beginPath();ctx.arc(P.x,P.y,len,-1.0,a-.1);ctx.stroke();}
  ctx.lineCap='round';
  ctx.strokeStyle='#8a5a2b';ctx.lineWidth=R*.2;
  ctx.beginPath();ctx.moveTo(P.x,P.y);ctx.lineTo(H.x,H.y);ctx.stroke();
  ctx.fillStyle='#ffd24a';ctx.strokeStyle='#b8860b';ctx.lineWidth=2;
  ctx.beginPath();ctx.arc(H.x,H.y,R*.42,0,7);ctx.fill();ctx.stroke();
  ctx.lineCap='butt';
  /* coins */
  for(const p of parts){const al=clamp(p.life/.6,0,1);
    ctx.globalAlpha=al;ctx.fillStyle='#ffd24a';ctx.strokeStyle='#c98f1b';ctx.lineWidth=1.5;
    ctx.beginPath();ctx.arc(p.x,p.y,p.r,0,7);ctx.fill();ctx.stroke();
    if(p.r>5){ctx.fillStyle='#7a5200';ctx.font=`bold ${p.r*1.3}px Trebuchet MS`;ctx.textAlign='center';ctx.fillText('$',p.x,p.y+p.r*.45);}
    ctx.globalAlpha=1;}
  /* floating texts */
  for(const t of texts){const al=clamp(t.life,0,1),sc=t.age<.12?1.6-5*t.age:1;
    ctx.globalAlpha=al;ctx.font=`900 ${t.size*sc}px Impact,'Arial Black',sans-serif`;ctx.textAlign='center';
    ctx.lineWidth=4;ctx.strokeStyle='rgba(8,30,20,.8)';ctx.strokeText(t.text,t.x,t.y);
    ctx.fillStyle=t.color;ctx.fillText(t.text,t.x,t.y);ctx.globalAlpha=1;}
  /* hint */
  if(playing&&S.swings===0){const al=.55+Math.sin(time*4)*.35;
    ctx.globalAlpha=al;ctx.font=`900 ${Math.max(15,L.R*.8)}px Impact,'Arial Black',sans-serif`;ctx.textAlign='center';
    ctx.lineWidth=4;ctx.strokeStyle='rgba(8,30,20,.7)';
    const hy=L.teeY-R*3.4+Math.sin(time*3)*4;
    ctx.strokeText('CLICK / TAP TO SWING!',L.teeX+L.R*2,hy);
    ctx.fillStyle='#fff';ctx.fillText('CLICK / TAP TO SWING!',L.teeX+L.R*2,hy);ctx.globalAlpha=1;}
  /* confetti */
  for(const c of conf){ctx.save();ctx.globalAlpha=clamp(c.life,0,1);
    ctx.translate(c.x,c.y);ctx.rotate(c.rot);ctx.fillStyle=c.c;
    ctx.fillRect(-c.w/2,-c.h/2,c.w,c.h);ctx.restore();}
  ctx.restore();
}

/* ================= game loop ================= */
let last=performance.now(),hudAcc=0;
function frame(now){
  const dt=Math.min(.05,(now-last)/1000);last=now;time+=dt;
  if(playing){
    const inc=perSec()*dt;if(inc>0){S.cash+=inc;S.total+=inc;}
    checkMS();
    if(perSec()>=1){ghostTimer-=dt;if(ghostTimer<=0){launchBall(false,0,true);ghostTimer=rand(.9,1.6);}}
  }
  px+=(tx-px)*Math.min(1,dt*3);py+=(ty-py)*Math.min(1,dt*3);
  for(const c of clouds){c.x+=c.v*dt;if(c.x>w+220)c.x=-220;}
  putterT=Math.min(99,putterT+dt/.24);
  flagKick=Math.max(0,flagKick-dt*1.5);
  shake=Math.max(0,shake-dt*22);
  for(let i=shots.length-1;i>=0;i--){const s=shots[i];s.t+=dt/s.dur;
    if(s.t>=1){plop(s);shots.splice(i,1);}}
  for(let i=parts.length-1;i>=0;i--){const p=parts[i];p.vy+=900*dt;p.x+=p.vx*dt;p.y+=p.vy*dt;p.life-=dt;if(p.life<=0)parts.splice(i,1);}
  if(parts.length>320)parts.splice(0,parts.length-320);
  for(let i=texts.length-1;i>=0;i--){const t=texts[i];t.age+=dt;t.y+=t.v*dt;t.life-=dt*.9;if(t.life<=0)texts.splice(i,1);}
  for(let i=ripples.length-1;i>=0;i--){const r=ripples[i];r.r+=70*dt;r.life-=dt*1.4;if(r.life<=0)ripples.splice(i,1);}
  for(let i=conf.length-1;i>=0;i--){const c=conf[i];c.vy+=520*dt;c.x+=c.vx*dt+Math.sin(time*3+c.rot)*30*dt;c.y+=c.vy*dt;c.rot+=c.vr*dt;c.life-=dt;if(c.life<=0||c.y>h+30)conf.splice(i,1);}
  render();
  hudAcc+=dt;if(hudAcc>.12){hudAcc=0;updateHUD();refreshShop();}
  requestAnimationFrame(frame);
}

function swing(){
  if(!playing)return;
  const crit=Math.random()<critChance();
  const amt=perClick()*(crit?10:1);
  S.cash+=amt;S.total+=amt;S.swings++;if(amt>S.best)S.best=amt;
  putterT=0;launchBall(crit,amt,false);
  if(crit){SFX.crit();SFX.swing();}else SFX.swing();
  updateHUD();saveThrottled();
}
