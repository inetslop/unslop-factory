  if(!AC||muted||!musicOn)return;
  const spb=(G.state==='wave')?60/138/2:60/92/2;
  while(nextNoteT<AC.currentTime+.16){
    const i=mStep%16,f=BASS[i];
    if(f){const t0=nextNoteT,o=AC.createOscillator(),g=AC.createGain();o.type='triangle';
      o.frequency.value=f;g.gain.setValueAtTime(.0001,t0);g.gain.exponentialRampToValueAtTime(.2,t0+.02);
      g.gain.exponentialRampToValueAtTime(.0001,t0+spb*.95);o.connect(g).connect(musG);o.start(t0);o.stop(t0+spb);}
    nBurstM(i%4===2?(G.state==='wave'?.1:.04):.02,i);
    if(G.state==='wave'&&(i===4||i===12)){const t0=nextNoteT,src=AC.createBufferSource(),fl=AC.createBiquadFilter(),g=AC.createGain();
      src.buffer=noiseBuf;fl.type='bandpass';fl.frequency.value=1900;
      g.gain.setValueAtTime(.14,t0);g.gain.exponentialRampToValueAtTime(.0001,t0+.1);
      src.connect(fl).connect(g).connect(musG);src.start(t0);src.stop(t0+.12);}
    nextNoteT+=spb;mStep++;
  }
}
function nBurstM(g,at){ if(!AC||muted)return;const t0=nextNoteT,src=AC.createBufferSource(),fl=AC.createBiquadFilter(),gg=AC.createGain();
  src.buffer=noiseBuf;fl.type='highpass';fl.frequency.value=6500;gg.gain.setValueAtTime(g,t0);
  gg.gain.exponentialRampToValueAtTime(.0001,t0+.04);src.connect(fl).connect(gg).connect(musG);src.start(t0);src.stop(t0+.05);}

let W=0,H=0,dpr=1,cell=40,ox=0,oy=0;
function resize(){
  const r=cv.getBoundingClientRect();W=r.width;H=r.height;dpr=Math.min(2,window.devicePixelRatio||1);
  cv.width=Math.round(W*dpr);cv.height=Math.round(H*dpr);
  ctx.setTransform(dpr,0,0,dpr,0,0);

  ox=Math.round((W-cell*COLS)/2);oy=Math.round((H-cell*ROWS)/2);
  refreshPath();
}
const cx=c=>ox+c*cell+cell/2, cy=r=>oy+r*cell+cell/2;

/* ================= PATH LOGIC ================= */
function connMask(t){return t.kind==='s'?(t.rot%2?E|W:N|S):[[N|E],[E|S],[S|W],[W|N]][t.rot];}
function currentPorts(){ if(G.wave<=10)return PORTS[G.wave-1]; return [(G.wave*3)%ROWS,(G.wave*5+2)%ROWS]; }
function computePath(){
  const[sr,er]=currentPorts();
  const res={pts:[],idx:{},valid:false,breakAt:null,sr,er};
  let c=0,r=sr,entry=W;const visited=new Set();
  while(true){
    if(c<0||c>=COLS||r<0||r>=ROWS){res.breakAt={c:Math.max(0,Math.min(COLS-1,c)),r:Math.max(0,Math.min(ROWS-1,r))};break;}
    const k=c+','+r;
    if(visited.has(k)){res.breakAt={c,r};break;}
    visited.add(k);
    const t=G.grid[r][c];
    if(!t||ROCKS.has(k)){res.breakAt={c,r};break;}
    const m=connMask(t);
    if(!(m&entry)){res.breakAt={c,r};break;}
    res.idx[k]=res.pts.length;res.pts.push({c,r,t});
    let out=0;for(let d=N;d<=W;d<<=1)if((m&d)&&d!==entry)out=d;
    c+=DX[out];r+=DY[out];entry=OPP[out];
    if(c===COLS){res.valid=(r===er);if(!res.valid)res.breakAt={c:COLS-1,r};break;}
  }
  return res;
}
function refreshPath(){
  const p=computePath();G.path=p;
  const px=[{x:ox-cell*.5,y:cy(p.sr),gate:0,virt:true}];
  for(const q of p.pts)px.push({x:cx(q.c),y:cy(q.r),gate:q.t.gate?1:0,tile:q.t});
  if(p.valid)px.push({x:ox+cell*COLS+cell*.5,y:cy(p.er),gate:0,virt:true});
  G.pathPx=px;
}
function pathStats(){ // gates + est damage of current route
  let g=0,d=0;if(G.path&&G.path.valid)for(const q of G.path.pts)if(q.t.gate){g++;d+=GATE_DMG[q.t.level];}
  return{g,d};
}

/* ================= ACTIONS ================= */
function setMsg(t,ms){G.msgOv=t;G.msgOvT=ms||0;}
function flashMsg(t){setMsg(t,1400);sfx.err();}
function placeTile(c,r){
  const tool=TOOLS[G.tool];
  if(tool.cost>G.scrap){flashMsg('NEED '+tool.cost+' SCRAP');return;}
  G.grid[r][c]={kind:tool.kind,rot:G.ghostRot,gate:!!tool.gate,level:1,spent:tool.cost,flash:0};
  if(tool.cost)G.scrap-=tool.cost;
  sfx.place();refreshPath();
}
function removeTile(c,r){
  const t=G.grid[r][c];if(!t)return;
  G.grid[r][c]=null;
  if(t.spent){G.scrap+=t.spent;G.earned+=0;addText(cx(c),cy(r),'+'+t.spent,'#ffd166');}
  sfx.pop();refreshPath();
}
function upgradeGate(c,r){
  const t=G.grid[r][c];if(!t||!t.gate)return;
  if(t.level>=5){flashMsg('GATE AT MAX CHARGE');return;}
  const cost=10+5*t.level;
  if(cost>G.scrap){flashMsg('NEED '+cost+' SCRAP TO CHARGE');return;}
  G.scrap-=cost;t.spent+=cost;t.level++;t.flash=1;
  sfx.charge();ring(cx(c),cy(r),'#ffd166',cell*.9);addText(cx(c),cy(r)-cell*.3,'LV'+t.level,'#ffd166');
  refreshPath();
}
function clickCell(c,r){
  if(G.state!=='build'){if(!G.paused)flashMsg('WAVE ACTIVE — BUILDING LOCKED');return;}
  const k=c+','+r;
  if(ROCKS.has(k)){flashMsg('DEAD ROCK — UNBUILDABLE');return;}
  const t=G.grid[r][c],ti=TOOLS[G.tool];
  if(ti.id==='x'){t?removeTile(c,r):sfx.tick();return;}
  if(ti.id==='u'){if(t&&t.gate)upgradeGate(c,r);else flashMsg('CHARGE TARGETS A GATE');return;}
  if(t){t.rot=(t.rot+1)%4;sfx.tick();refreshPath();}
  else placeTile(c,r);
}
function buildQueue(){
  const w=G.wave,groups=w<=10?WAVES[w-1]:endlessWave(w),mul=1+.08*(w-1);
  G.queue=[];let at=.6;
  for(const g of groups)for(let i=0;i<g.n;i++){G.queue.push({t:g.t,at,hp:Math.ceil(ETYPES[g.t].hp*(g.hp||mul))});at+=g.gap;}
  G.qIdx=0;G.qT=0;
}
function endlessWave(w){
  const m=1+.25*(w-10);
  const g=[{t:'grunt',n:6+w%5,gap:.5},{t:'runner',n:4+(w>>1)%7,gap:.42},{t:'tank',n:1+(w>>1)%4,gap:1.6}];
  if(w%3===0)g.push({t:'boss',n:1,gap:1,hp:m});
  return g;
}
function launch(){
  if(G.state!=='build'||!G.path.valid)return;
  refreshPath();buildQueue();G.state='wave';musicOn=true;
  sfx.waveStart();setMsg('WAVE '+G.wave+' INCOMING',1800);updateHUD(true);
}
function endWave(){
  const bonus=12+2*G.wave;G.scrap+=bonus;G.earned+=bonus;
  addText(W/2,oy-cell*.4,'WAVE CLEARED  +'+bonus,'#3df2c6',true);
  sfx.clear();G.wave++;saveBest(G.wave-1);
  if(G.wave>10&&!G.endless){victory();return;}
  G.state='build';refreshPath();setMsg('PORTS SHIFTED — RE-ROUTE YOUR CONDUITS',2400);updateHUD(true);
}
function victory(){G.state='victory';musicOn=false;sfx.victory();showEnd(true);}
function gameOver(){G.state='over';musicOn=false;G.shake=16;saveBest(G.wave-1);sfx.boom();sfx.defeat();setTimeout(()=>showEnd(false),700);}

/* ================= ENEMIES ================= */
function spawnEnemy(q){
  const d=ETYPES[q.t];
  G.enemies.push({type:q.t,hp:q.hp,maxhp:q.hp,spd:d.spd,seg:0,t:Math.random()*.15,ph:Math.random()*7,dead:false});
}
function killEnemy(e,i){
  e.dead=true;G.enemies.splice(i,1);
  const p=enemyPos(e),d=ETYPES[e.type];
  G.scrap+=d.reward;G.earned+=d.reward;G.kills++;
  burst(p.x,p.y,d.color,e.type==='boss'?36:14,e.type==='boss'?260:150);
  ring(p.x,p.y,d.color,cell*1.4);addText(p.x,p.y,'+'+d.reward,'#ffd166');
  if(e.type==='boss'){G.shake=14;sfx.boom();}else sfx.pop();
}
function enemyPos(e){
  const a=G.pathPx[e.seg],b=G.pathPx[Math.min(e.seg+1,G.pathPx.length-1)];
  return{x:a.x+(b.x-a.x)*e.t,y:a.y+(b.y-a.y)*e.t,ang:Math.atan2(b.y-a.y,b.x-a.x)};
}
function updateEnemies(dt){
  const pts=G.pathPx;
  for(let i=G.enemies.length-1;i>=0;i--){
    const e=G.enemies[i];
    e.t+=e.spd*dt;
    while(e.t>=1&&!e.dead){
      e.seg++;e.t-=1;
      if(e.seg>=pts.length-1){ // leak
        G.enemies.splice(i,1);G.coreHP-=ETYPES[e.type].leak;G.coreFlash=1;G.shake=Math.max(G.shake,e.type==='boss'?16:8);
        const cc={x:cx(COLS-1),y:cy(currentPorts()[1])};
        burst(cc.x,cc.y,'#ff4d5e',20,200);sfx.boom();sfx.alarm();
        addText(cc.x,cc.y-cell*.7,'-'+ETYPES[e.type].leak+' CORE','#ff4d5e');
        if(G.coreHP<=0)gameOver();
        break;
      }
      const p=pts[e.seg];
      if(p.gate){ // gate discharge
        e.hp-=GATE_DMG[p.tile.level];p.tile.flash=1;sfx.zap();
        burst(p.x,p.y,'#ffd166',7,180);ring(p.x,p.y,'#ffd166',cell*.7);
        if(e.hp<=0){killEnemy(e,i);}
      }
    }
    if(e.type==='boss'&&!e.dead&&Math.random()<dt*24)G.parts.push({x:enemyPos(e).x,y:enemyPos(e).y,vx:(Math.random()-.5)*30,vy:(Math.random()-.5)*30,life:.5,max:.5,c:'#ff3b5c',s:cell*.08});
  }
}

function drawGhost(){
  const{c,r}=G.hover,k=c+','+r;
  if(ROCKS.has(k))return;
  const t=G.grid[r][c],ti=TOOLS[G.tool];
  ctx.fillStyle='rgba(61,242,198,.07)';ctx.fillRect(ox+c*cell,oy+r*cell,cell,cell);
  const x=cx(c),y=cy(r);
  if(ti.id==='u'){ // charge preview on gate
    if(t&&t.gate){ctx.strokeStyle='rgba(255,209,102,.8)';ctx.setLineDash([4,4]);ctx.lineDashOffset=-G.time*20;
      ctx.lineWidth=2;ctx.strokeRect(ox+c*cell+3,oy+r*cell+3,cell-6,cell-6);ctx.setLineDash([]);}
    return;}
  if(ti.id==='x'){if(t){ctx.strokeStyle='rgba(255,77,94,.8)';ctx.lineWidth=2;ctx.strokeRect(ox+c*cell+3,oy+r*cell+3,cell-6,cell-6);
      ctx.beginPath();ctx.moveTo(x-6,y-6);ctx.lineTo(x+6,y+6);ctx.moveTo(x+6,y-6);ctx.lineTo(x-6,y+6);ctx.stroke();}return;}
  // conduit ghost
  const rot=t?t.rot:G.ghostRot,kind=t?t.kind:ti.kind;
  const m=kind==='s'?(rot%2?E|W:N|S):[[N|E],[E|S],[S|W],[W|N]][rot];
  ctx.lineCap='round';
  ctx.strokeStyle=ti.gate?'rgba(255,209,102,.75)':'rgba(61,242,198,.65)';
  ctx.lineWidth=cell*.16;tileStroke(x,y,m);
  // connector nubs + compatibility color
  const[sr,er]=currentPorts();
  for(const d of[N,E,S,W]){if(!(m&d))continue;
    const nx=c+DX[d],ny=r+DY[d];let col='#4f7d84';
    if(nx<0||nx>=COLS||ny<0||ny>=ROWS){
      if(nx===-1&&c===0&&r===sr)col='#3df2c6';
      else if(nx===COLS&&c===COLS-1&&r===er)col='#3df2c6';}
    else{const nt=G.grid[ny][nx];
      if(nt&&(connMask(nt)&OPP[d]))col='#3df2c6';else col='#ff4d5e';}
    ctx.fillStyle=col;
    const px=x+DX[d]*cell/2,py=y+DY[d]*cell/2,sz=Math.max(4,cell*.14);
    ctx.fillRect(px-sz/2,py-sz/2,sz,sz);}
  if(ti.gate){ctx.fillStyle='rgba(255,209,102,.8)';const s=cell*.16;
    ctx.save();ctx.translate(x,y);ctx.rotate(Math.PI/4);ctx.fillRect(-s/2,-s/2,s,s);ctx.restore();}
  // would this complete the route?
  if(!t){
    const old=null;G.grid[r][c]={kind:ti.kind,rot:G.ghostRot,gate:!!ti.gate,level:1,spent:0};
    const v=computePath().valid;G.grid[r][c]=null;
    if(v){ctx.strokeStyle='rgba(61,242,198,'+(.5+.4*Math.sin(G.time*8))+')';ctx.lineWidth=2.5;
      ctx.strokeRect(ox+c*cell+2,oy+r*cell+2,cell-4,cell-4);}
  }
}
