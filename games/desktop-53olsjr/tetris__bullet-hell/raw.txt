function resolveMatches(chain){
  if(chain>7)return;
  const seen=new Int8Array(cols*rows),groups=[];
  for(let r=0;r<rows;r++)for(let c=0;c<cols;c++){
    const i=r*cols+c;

    const col=grid[i],st=[[c,r]];seen[i]=1;const g=[];
    while(st.length){
      const [cc,rr]=st.pop();g.push([cc,rr]);
      for(const [dx,dy] of DIRS4){
        const nc=cc+dx,nr=rr+dy;
        if(nc<0||nr<0||nc>=cols||nr>=rows)continue;
        const ni=nr*cols+nc;
        if(grid[ni]===col&&!seen[ni]){seen[ni]=1;st.push([nc,nr]);}
      }
    }
    if(g.length>=3)groups.push({col,g});
  }
  if(!groups.length)return;
  let n=0,cx=0,cy=0;
  for(const grp of groups){
    for(const [c,r] of grp.g){
      grid[r*cols+c]=-1;n++;
      const px=offX+c*cell+cell/2,py=r*cell+cell/2;
      cx+=px;cy+=py;

}

    const dx=ship.x-x,dy=shipY-y,a=Math.atan2(dy,dx)+rand(-.05,.05);
    pushBullet({x,y,vx:Math.cos(a)*bs,vy:Math.sin(a)*bs,r:Math.max(4,cell*.15),color:C_AIM});
  }else if(tg.type==='fan'){
    const dx=ship.x-x,dy=shipY-y,a=Math.atan2(dy,dx);
    for(const off of[-.27,0,.27])
      pushBullet({x,y,vx:Math.cos(a+off)*bs*.92,vy:Math.sin(a+off)*bs*.92,r:Math.max(3.5,cell*.13),color:C_AIM});
  }else{
    pushBullet({x,y:-12,vx:0,vy:cell*6.2+level*3.5,r:Math.max(3.5,cell*.13),color:C_RAIN});
  }
}
function updateEmitters(dt){
  fireT-=dt;
  if(fireT<=0){
    fireT=Math.max(.3,1.0-level*.055);
    const cells=occupiedCells(.7);
    if(cells.length){
      const [c,r]=cells[ri(0,cells.length-1)];
      const roll=Math.random();
      let type='aim';
      if(level>=4&&roll<.3)type='rain';
      else if(level>=2&&roll<.65)type='fan';
      scheduleTelegraph(c,r,type);
    }
  }
  for(let i=telegraphs.length-1;i>=0;i--){
    const tg=telegraphs[i];tg.t-=dt*slowF;
    if(tg.t<=0){fireFromTelegraph(tg);telegraphs.splice(i,1);}
  }
}
function updateWall(dt){
  if(level<3)return;
  if(wallWarn>0){
    wallWarn-=dt;
    if(wallWarn<=0)spawnWall();
    return;
  }
  wallT-=dt;
  if(wallT<=0){wallWarn=.95;wallT=Math.max(8,15-level*.6);}
}
function spawnWall(){
  const gc=ri(2,cols-3);
  for(let c=0;c<cols;c++){
    if(Math.abs(c-gc)<=1)continue;
    pushBullet({x:offX+c*cell+cell/2,y:-14,vx:0,vy:H/1.6,r:Math.max(4,cell*.15),color:C_WALL});
  }
  SFX.levelup();shake=Math.min(shake+3,16);
}
function updateBullets(dt){
  const collide=state==='play';
  for(let i=bullets.length-1;i>=0;i--){
    const b=bullets[i];
    const py=b.y;
    b.x+=b.vx*dt*slowF;b.y+=b.vy*dt*slowF;
    if(b.x<-40||b.x>W+40||b.y>H+40||b.y<-260){bullets.splice(i,1);continue;}
    // near-miss dodge
    if(collide&&!shipDead&&!b.dodged&&b.vy>0&&py<shipY&&b.y>=shipY){
      const dx=Math.abs(b.x-ship.x);
      if(dx>b.r+4&&dx<cell*2.3){
        b.dodged=true;stats.dodge++;
        if(state==='play'){score+=6;popText(ship.x,shipY-20,'dodge +6','#9fd8ff',10);}
      }
    }
    if(collide&&!shipDead&&ship.invuln<=0){
      const dx=b.x-ship.x,dy=b.y-shipY,d2=dx*dx+dy*dy,rr=(b.r+9)* (b.r+9);
      if(d2<rr){bullets.splice(i,1);hitShip();return;}
    }
  }
}

/* ============================== ship ============================== */
function updateShip(dt){
  const sp=Math.max(380,cell*14);
  let dir=0;
  if(keys.ArrowLeft||keys.KeyA)dir-=1;
  if(keys.ArrowRight||keys.KeyD)dir+=1;
  if(pointer.active)ship.x+=(pointer.x-ship.x)*Math.min(1,dt*14);
  ship.x=clamp(ship.x+dir*sp*dt,offX+14,W-offX-14);
  ship.invuln=Math.max(0,ship.invuln-dt);
  ship.fireCd-=dt;
  const firing=keys.Space||keys.ArrowUp||keys.KeyW||(pointer.active&&pointer.firing);
  if(firing&&ship.fireCd<=0){
    ship.fireCd=.14;SFX.shoot();
    const mx=ship.x,my=shipY-13;
    sBullets.push({x:mx,y:my,vx:0,vy:-560,px:mx,py:my});
    if(powers.spread>0){
      for(const a of[-.2,.2])
        sBullets.push({x:mx,y:my,vx:Math.sin(a)*560,vy:-Math.cos(a)*560,px:mx,py:my});
    }
  }
}
function updateShipBullets(dt){
  for(let i=sBullets.length-1;i>=0;i--){
    const b=sBullets[i];b.px=b.x;b.py=b.y;
    b.x+=b.vx*dt;b.y+=b.vy*dt;
    let dead=b.y<-24||b.x<offX-20||b.x>W-offX+20;
    if(!dead){
      for(const [bx,by] of [[b.px,b.py],[b.x,b.y]]){
        const c=Math.floor((bx-offX)/cell),r=Math.floor(by/cell);
        if(c>=0&&r>=0&&c<cols&&r<rows&&grid[r*cols+c]>=0){
          grid[r*cols+c]=-1;dead=true;
          const px=offX+c*cell+cell/2,py=r*cell+cell/2;
          burst(px,py,PAL[gridHitColor(grid,r,c)]||PAL[0].m,7,160,.45,3);
          if(state==='play'){
            const gain=addScore(25);stats.blocks++;
            popText(px,py,'+'+gain,'#fff',12);SFX.hitBlock();
            resolveMatches(0);
          }else SFX.hitBlock();
          break;
        }
      }
    }
    if(!dead){
      for(let pi=pieces.length-1;pi>=0&&!dead;pi--){
        const p=pieces[pi];
        for(let ci=p.cells.length-1;ci>=0;ci--){
          const cl=p.cells[ci];
          const cx=offX+(p.x+cl.dx)*cell+cell/2,cy=(p.y+cl.dy)*cell+cell/2;
          const dx=b.x-cx,dy=b.y-cy;
          if(dx*dx+dy*dy<cell*cell*.25){
            p.cells.splice(ci,1);dead=true;
            burst(cx,cy,PAL[p.color].m,7,160,.45,3);
            if(state==='play'){
              const gain=addScore(40);stats.blocks++;
              popText(cx,cy,'+'+gain,'#fff',12);SFX.hitBlock();
            }else SFX.hitBlock();
            if(!p.cells.length){
              pieces.splice(pi,1);
              if(state==='play'){const gain=addScore(150);popText(cx,cy-18,'SHATTERED +'+gain,'#7dff6a',18);}
            }
            break;
          }
        }
      }
    }
    if(dead)sBullets.splice(i,1);
  }
}
function gridHitColor(g2,r,c){return g2[r*cols+c];}

function hitShip(){
  lives--;combo=0;comboT=0;ship.invuln=2;
  shake=14;flash=.5;SFX.hitShip();
  for(let i=bullets.length-1;i>=0;i--){
    const b=bullets[i],dx=b.x-ship.x,dy=b.y-shipY;
    if(dx*dx+dy*dy<(cell*3)*(cell*3)){burst(b.x,b.y,b.color,4,150,.4,2);bullets.splice(i,1);}
  }
  popText(ship.x,shipY-30,'HULL HIT!','#ff5470',20);
  if(lives<=0)die('SHIP DESTROYED');
}
function doOverflow(){die('STACK OVERFLOW');}
function die(why){
  if(state!=='play')return;
  cause=why;state='dying';deathT=1.2;shipDead=true;
  burst(ship.x,shipY,'#eaf7ff',40,320,.9,4);
  burst(ship.x,shipY,C_AIM,30,260,.8,3);
  burst(ship.x,shipY,C_RAIN,25,220,.8,3);
  shake=18;flash=.7;SFX.gameover();
}

  pieces=[];bullets=[];sBullets=[];parts=[];texts=[];caps=[];telegraphs=[];
  score=0;combo=0;comboT=0;lives=3;level=1;prevLevel=1;time=0;

}

/* ============================== update loops ============================== */
function updateFX(dt){
  for(let i=parts.length-1;i>=0;i--){
    const p=parts[i];p.t+=dt;
    if(p.t>=p.life){parts.splice(i,1);continue;}
    p.x+=p.vx*dt;p.y+=p.vy*dt;p.vy+=(p.grav||0)*dt;
  }
  for(let i=texts.length-1;i>=0;i--){
    const t=texts[i];t.t+=dt;t.y-=32*dt;
    if(t.t>=t.life)texts.splice(i,1);
  }
  for(const d of dust){d.y-=d.v*dt;if(d.y<-4){d.y=H+4;d.x=rand(0,W);}}
  shake=Math.max(0,shake-dt*26);
  flash=Math.max(0,flash-dt*1.8);
}
function updatePlay(dt){
  time+=dt;
  level=Math.min(12,1+Math.floor(time/24));
  if(level>prevLevel){prevLevel=level;SFX.levelup();popText(W/2,H*.3,'LEVEL '+level,'#ffc233',30);}
  powers.spread=Math.max(0,powers.spread-dt);
  powers.slow=Math.max(0,powers.slow-dt);
  slowF=powers.slow>0?.55:1;
  comboT-=dt;if(comboT<=0)combo=0;
  spawnT-=dt;
  if(spawnT<=0){spawnT=Math.max(.75,2.4-level*.15);spawnPiece();}
  updateShip(dt);
  updatePieces(dt);
  updateShipBullets(dt);
  updateEmitters(dt);
  updateWall(dt);
  updateBullets(dt);
  updateCapsules(dt);
  updateFX(dt);
}
function updateDying(dt){
  const s=dt*.25;
  updatePieces(s);updateBullets(s);updateCapsules(s);updateFX(dt);
}
function updateMenu(dt){
  menuSpawnT-=dt;
  if(menuSpawnT<=0){menuSpawnT=1.3;spawnPiece();}
  for(const p of pieces)p.y+=3*dt;
  // snap like play (silent)
  for(let i=pieces.length-1;i>=0;i--){
    const p=pieces[i];let snap=false;
    for(const cl of p.cells){
      const below=gGet(p.x+cl.dx,Math.floor(p.y)+cl.dy+1);
      if(below===-2||below>=0){snap=true;break;}
    }
    if(snap){
      const yr=Math.floor(p.y);
      for(const cl of p.cells){const rr=yr+cl.dy;if(rr>=0&&rr<rows)grid[rr*cols+p.x+cl.dx]=p.color;}
      pieces.splice(i,1);
    }
  }
  resolveMatches(0);
  if(findMinRow()<5){
    for(let i=0;i<grid.length;i++)if(grid[i]>=0)burst(offX+(i%cols)*cell+cell/2,(Math.floor(i/cols))*cell+cell/2,PAL[grid[i]].m,3,120,.5,2);
    grid.fill(-1);
  }
  menuBulletT-=dt;
  if(menuBulletT<=0&&bullets.length<60){
    menuBulletT=.8;
    pushBullet({x:rand(offX,W-offX),y:-10,vx:rand(-12,12),vy:H/5.5,r:Math.max(3,cell*.13),color:C_RAIN});
  }
  const cells=occupiedCells(.6);
  menuFireT-=dt;
  if(menuFireT<=0&&cells.length){
    menuFireT=2.4;
    const [c,r]=cells[ri(0,cells.length-1)];scheduleTelegraph(c,r,'aim');
  }
  updateEmitters(dt);
  // ghost ship
  const gx=W/2+Math.sin(tGlobal*.6)*W*.32;
  menuGhostFire-=dt;
  if(menuGhostFire<=0){menuGhostFire=.45;sBullets.push({x:gx,y:shipY-13,vx:0,vy:-560,px:gx,py:shipY-13});}
  updateShipBullets(dt);
  updateBullets(dt);
  updateFX(dt);
  ship.x=gx; // reuse for drawing ghost
}
let menuGhostFire=0;

/* ============================== render ============================== */
function drawBlock(x,y,col,glow=false){
  const p=PAL[col],inset=Math.max(1,cell*.06);
  if(glow){ctx.shadowColor=p.m;ctx.shadowBlur=10;}
  ctx.fillStyle=p.m;ctx.fillRect(x+inset,y+inset,cell-inset*2,cell-inset*2);
  ctx.shadowBlur=0;
  ctx.fillStyle='rgba(255,255,255,.32)';ctx.fillRect(x+inset,y+inset,cell-inset*2,Math.max(2,cell*.16));
  ctx.fillStyle='rgba(0,0,0,.3)';ctx.fillRect(x+inset,y+cell-inset-Math.max(2,cell*.16),cell-inset*2,Math.max(2,cell*.16));
  ctx.strokeStyle=p.d;ctx.lineWidth=1.5;
  ctx.strokeRect(x+inset+.75,y+inset+.75,cell-inset*2-1.5,cell-inset*2-1.5);
}
function drawShip(x,y,ghost){
  const sc=clamp(cell/26,.8,1.4);
  ctx.save();ctx.translate(x,y);ctx.scale(sc,sc);
  if(!ghost&&ship.invuln>0&&(Math.floor(tGlobal*14)%2))ctx.globalAlpha=.3;
  if(ghost)ctx.globalAlpha*=.55;
  const fl=rand(8,17);
  ctx.fillStyle='rgba(53,224,255,.55)';
  ctx.beginPath();ctx.moveTo(-5,11);ctx.lineTo(0,11+fl);ctx.lineTo(5,11);ctx.closePath();ctx.fill();
  ctx.fillStyle='#fff';
  ctx.beginPath();ctx.moveTo(-2.5,11);ctx.lineTo(0,11+fl*.55);ctx.lineTo(2.5,11);ctx.closePath();ctx.fill();
  ctx.fillStyle=ghost?'#9fd8ff':'#eaf7ff';
  ctx.strokeStyle='#35e0ff';ctx.lineWidth=1.6;
  ctx.beginPath();ctx.moveTo(0,-14);ctx.lineTo(11,8);ctx.lineTo(5,12);ctx.lineTo(-5,12);ctx.lineTo(-11,8);ctx.closePath();
  ctx.fill();ctx.stroke();
  ctx.fillStyle='#ff2fa0';ctx.beginPath();ctx.arc(0,-3,3,0,TAU);ctx.fill();
  ctx.restore();
}
function hexPath(x,y,r){
  ctx.beginPath();
  for(let i=0;i<6;i++){const a=Math.PI/6+i*TAU/6;
    i?ctx.lineTo(x+Math.cos(a)*r,y+Math.sin(a)*r):ctx.moveTo(x+Math.cos(a)*r,y+Math.sin(a)*r);}
  ctx.closePath();
}
function render(){
  if(cv.clientWidth!==W||cv.clientHeight!==H)resize();
  ctx.setTransform(DPR,0,0,DPR,0,0);
  const sx=(Math.random()*2-1)*shake,sy=(Math.random()*2-1)*shake;
  if(bgCv)ctx.drawImage(bgCv,0,0,W,H);
  // dust parallax
  ctx.fillStyle='#9fd8ff';
  for(const d of dust){ctx.globalAlpha=d.a;ctx.fillRect(d.x,d.y,d.s,d.s*2.5);}
  ctx.globalAlpha=1;
  ctx.save();ctx.translate(sx,sy);

  // danger line
  const mr=findMinRow();
  if(mr<=3&&state!=='menu'){
    const a=.14+.1*Math.sin(tGlobal*9);
    const g2=ctx.createLinearGradient(0,0,0,cell*3.5);
    g2.addColorStop(0,'rgba(255,84,112,'+a+')');g2.addColorStop(1,'rgba(255,84,112,0)');
    ctx.fillStyle=g2;ctx.fillRect(offX,0,cols*cell,cell*3.5);
    ctx.strokeStyle='rgba(255,84,112,'+(a+.25)+')';ctx.setLineDash([8,6]);ctx.lineWidth=1.5;
    ctx.beginPath();ctx.moveTo(offX,cell*3+2);ctx.lineTo(offX+cols*cell,cell*3+2);ctx.stroke();ctx.setLineDash([]);
  }
  // grid blocks
  for(let r=0;r<rows;r++)for(let c=0;c<cols;c++){
    const v=grid[r*cols+c];
    if(v>=0)drawBlock(offX+c*cell,r*cell,v);
  }
  // telegraphs
  for(const tg of telegraphs){
    const x=offX+tg.c*cell+cell/2,y=tg.r*cell+cell/2,k=1-tg.t/tg.T;
    const col=tg.type==='rain'?C_RAIN:C_AIM;
    ctx.strokeStyle=col;ctx.globalAlpha=.4+.5*Math.sin(tGlobal*20);ctx.lineWidth=2;
    ctx.beginPath();ctx.arc(x,y,cell*(.3+.6*k),0,TAU);ctx.stroke();
    ctx.fillStyle=col;ctx.beginPath();ctx.arc(x,y,cell*.14,0,TAU);ctx.fill();
    ctx.globalAlpha=1;
  }
  // pieces
  for(const p of pieces)for(const cl of p.cells){
    const cy=(p.y+cl.dy)*cell;
    if(cy> -cell*2)drawBlock(offX+(p.x+cl.dx)*cell,cy,p.color,true);
  }
  // capsules
  for(const k of caps){
    const px=k.x+Math.sin(k.t*3)*6,py=k.y,s=cell*.45;
    ctx.save();ctx.translate(px,py);
    ctx.strokeStyle=k.type.c;ctx.globalAlpha=.5+.4*Math.sin(tGlobal*8);ctx.lineWidth=2;
    ctx.beginPath();ctx.arc(0,0,s+5+2*Math.sin(tGlobal*6),0,TAU);ctx.stroke();
    ctx.globalAlpha=1;ctx.fillStyle='#0d1428';
    ctx.fillRect(-s,-s,s*2,s*2);
    ctx.strokeStyle=k.type.c;ctx.lineWidth=2;ctx.strokeRect(-s,-s,s*2,s*2);
    ctx.fillStyle=k.type.c;ctx.font='700 '+Math.round(s*1.1)+'px '+DISPLAY;
    ctx.textAlign='center';ctx.textBaseline='middle';
    ctx.fillText(k.type.ch,0,1);
    ctx.restore();
  }
  // bullets (additive)
  ctx.globalCompositeOperation='lighter';
  for(const b of bullets){
    ctx.fillStyle=b.color;ctx.globalAlpha=.22;
    ctx.beginPath();ctx.arc(b.x,b.y,b.r*2.3,0,TAU);ctx.fill();
    ctx.globalAlpha=.85;
    ctx.beginPath();ctx.arc(b.x,b.y,b.r,0,TAU);ctx.fill();
    ctx.fillStyle='#fff';ctx.globalAlpha=1;
    ctx.beginPath();ctx.arc(b.x,b.y,b.r*.45,0,TAU);ctx.fill();
  }
  for(const b of sBullets){
    ctx.strokeStyle='rgba(160,240,255,.9)';ctx.lineWidth=3;
    ctx.beginPath();ctx.moveTo(b.x,b.y+7);ctx.lineTo(b.x,b.y-7);ctx.stroke();
    ctx.strokeStyle='rgba(53,224,255,.35)';ctx.lineWidth=6;
    ctx.beginPath();ctx.moveTo(b.x,b.y+9);ctx.lineTo(b.x,b.y-9);ctx.stroke();
  }
  ctx.globalCompositeOperation='source-over';ctx.globalAlpha=1;
  // particles
  for(const p of parts){
    const a=1-p.t/p.life;
    ctx.globalAlpha=a;ctx.fillStyle=p.color;
    ctx.fillRect(p.x-p.size/2,p.y-p.size/2,p.size,p.size);
  }
  ctx.globalAlpha=1;
  // ship / ghost
  if(state==='menu')drawShip(ship.x,shipY,true);
  else if(!shipDead)drawShip(ship.x,shipY,false);
  // floating texts
  for(const t of texts){
    const a=t.t<.15?t.t/.15:1-Math.max(0,(t.t-.5)/(t.life-.5));
    ctx.globalAlpha=clamp(a,0,1);
    ctx.font='700 '+t.size+'px '+DISPLAY;ctx.textAlign='center';
    ctx.fillStyle=t.color;
    ctx.shadowColor=t.color;ctx.shadowBlur=8;
    ctx.fillText(t.txt,t.x,t.y);ctx.shadowBlur=0;
  }
  ctx.globalAlpha=1;
  ctx.restore();

  // slow-mo tint
  if(powers.slow>0&&state!=='menu'){
    ctx.fillStyle='rgba(200,107,255,.06)';ctx.fillRect(0,0,W,H);
  }
  drawHUD();
  if(wallWarn>0&&state==='play'){
    const a=.4+.5*Math.sin(tGlobal*16);
    ctx.globalAlpha=a;ctx.fillStyle=C_WALL;
    ctx.font='700 '+Math.round(clamp(W*.05,22,40))+'px '+DISPLAY;ctx.textAlign='center';
    ctx.fillText('!! WALL INBOUND !!',W/2,H*.3);
    ctx.fillRect(0,H*.3-8,W,2);ctx.fillRect(0,H*.3+10,W,2);
    ctx.globalAlpha=1;
  }
  if(flash>0){ctx.fillStyle='rgba(255,255,255,'+flash+')';ctx.fillRect(0,0,W,H);}
}

  if(powers.slow>0){ctx.font='700 10px '+MONO;ctx.fillStyle='#c86bff';ctx.fillText('SLOW-MO '+powers.slow.toFixed(1)+'s',16,py);}
