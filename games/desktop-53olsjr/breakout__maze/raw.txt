function genMaze(lvl){

  // rooms at odd coords
  const stack=[[1,1]]; g[1][1]=0;
  while(stack.length){
    const [r,c]=stack[stack.length-1];
    const dirs=shuffle([[ -2,0],[2,0],[0,-2],[0,2]]);
    let moved=false;
    for(const [dr,dc] of dirs){
      const nr=r+dr,nc=c+dc;
      if(nr>0&&nr<ROWS-? ... bounds: nr>=1&&nr<=ROWS-2 (odd max 7) && nc>=1&&nc<=COLS-2 && g[nr][nc]===1){
        g[r+dr/2][c+dc/2]=0; g[nr][nc]=0; stack.push([nr,nc]); moved=true; break;
      }
    }
    if(!moved) stack.pop();
  }
  // build blocks
  for r,c:
    let t=0,hp=1,power=null;
    const boundary = r===0||r===ROWS-1||c===0||c===COLS-1;
    if(boundary){
      if(r===ROWS-1) { t=2; hp=1; } // gate
      else t=1; // wall (top & sides)
    } else if(g[r][c]===1){ t=1; }
    else {
      const room = (r%2===1&&c%2===1);
      if(room && Math.random()<0.92){
        t=2;
        // hp roll
        const r3=Math.min(0.04+0.035*lvl,0.28), r2=Math.min(0.16+0.04*lvl,0.42);

        hp = rr<r3?3: rr<r3+r2?2:1;
        if(Math.random()<0.14) power=pickPower();

    }

}

function burst(x,y,col,n,pow){for(let i=0;i<n;i++){const a=Math.random()*Math.PI*2,v=(40+Math.random()*160)*(pow||1);parts.push({x,y,vx:Math.cos(a)*v,vy:Math.sin(a)*v-40,life:.5+Math.random()*.5,max:1,col,sz:1.5+Math.random()*2.5})}}
function updateParts(dt){
  for(const p of parts){p.x+=p.vx*dt;p.y+=p.vy*dt;p.vy+=260*dt;p.life-=dt}
  parts=parts.filter(p=>p.life>0);
  for(const p of pops){p.y-=30*dt;p.t-=dt*.85}
  pops=pops.filter(p=>p.t>0);
}

/* ---------- brick damage ---------- */
function hitBrick(r,c){
  const bl=blocks[r][c];if(!bl||bl.t!==2)return;
  bl.hp--;bl.flash=1;
  const cx=fx+c*s+s/2,cy=fy+r*s+s/2;
  if(bl.hp<=0){
    blocks[r][c]={t:0,hp:0,max:0,pw:null,flash:0};leftCount--;
    S.combo++;const pts=10*Math.min(S.combo,10);S.score+=pts;
    const col=bl.pw?'#19d8c0':HP_COL[bl.max];
    pops.push({x:cx,y:cy,txt:'+'+pts,col:'#ffe9b0',t:1,size:13});
    burst(cx,cy,col,12+(bl.max>1?6:0),1.2);
    shake=Math.min(shake+(bl.max>=2?4:2),11);
    sBrick(bl.max,S.combo);
    if(bl.pw)drops.push({x:cx,y:cy,vy:150+25*S.level,type:bl.pw,t:Math.random()*6});
    if(comboEl.classList.contains('hidden')===false||S.combo>=2){comboEl.classList.remove('pop');void comboEl.offsetWidth;comboEl.classList.add('pop')}
    if(leftCount<=0)levelClear();
  }else{shake=Math.min(shake+1.5,8);burst(cx,cy,'#ffffff',4,.7);beep(900,600,.05,'triangle',.06)}
}
const HP_COL=[null,'#ffb020','#ff4d6d','#c94dff'];

/* ---------- power-ups ---------- */
const PLETTER={wide:'W',multi:'M',slow:'S',laser:'L',life:'+'};
const PCOL={wide:'#59f2ff',multi:'#e05aff',slow:'#6dff9e',laser:'#ffb020',life:'#ff5e7e'};
function applyPower(type){
  if(type==='wide')fxT.wide=12;
  else if(type==='slow')fxT.slow=8;
  else if(type==='laser')fxT.laser=9;
  else if(type==='life'){S.lives=Math.min(5,S.lives+1)}
  else if(type==='multi'){
    const srcs=balls.filter(b=>!b.stuck);
    for(const b of srcs){for(const da of[-.6,.6]){if(balls.length>=6)break;const a=Math.atan2(b.vy,b.vx)+da;balls.push({x:b.x,y:b.y,vx:Math.cos(a)*b.sp,vy:Math.sin(a)*b.sp,r:b.r,trail:[],stuck:false,sp:b.sp})}}
    if(srcs.length===0)fxT.slow=8;
  }
  sPower();shake=Math.min(shake+3,10);
  pops.push({x:paddle.x,y:paddleY-26,txt:type.toUpperCase()+(type==='life'?' +1':' PADDLE').slice(0,9),col:PCOL[type],t:1.2,size:15});
  syncHUD(true);
}

/* ---------- updates ---------- */
function updatePaddle(dt){
  const spd=fieldW*2.6;
  if(keys.l)paddle.x-=spd*dt;
  if(keys.r)paddle.x+=spd*dt;
  if(mouseTarget!=null&&!keys.l&&!keys.r)paddle.x+=(mouseTarget-paddle.x)*Math.min(1,dt*18);
  const tw=basePaddleW()*(fxT.wide>0?1.5:1);
  paddle.w+=(tw-paddle.w)*Math.min(1,dt*8);
  clampPaddle();
}
function collideBlocks(b){
  const c0=Math.max(0,Math.floor((b.x-b.r-fx)/s)),c1=Math.min(COLS-1,Math.floor((b.x+b.r-fx)/s));
  const r0=Math.max(0,Math.floor((b.y-b.r-fy)/s)),r1=Math.min(ROWS-1,Math.floor((b.y+b.r-fy)/s));
  for(let r=r0;r<=r1;r++)for(let c=c0;c<=c1;c++){
    const bl=blocks[r][c];if(!bl||bl.t===0)continue;
    const bx=fx+c*s+1,by=fy+r*s+1,bw=s-2,bh=s-2;
    const cx=Math.max(bx,Math.min(b.x,bx+bw)),cy=Math.max(by,Math.min(b.y,by+bh));
    const dx=b.x-cx,dy=b.y-cy;
    if(dx*dx+dy*dy>b.r*b.r)continue;
    const bxc=bx+bw/2,byc=by+bh/2;
    const ox=(bw/2+b.r)-Math.abs(b.x-bxc),oy=(bh/2+b.r)-Math.abs(b.y-byc);
    if(ox<oy){
      const left=b.x<bxc;b.x+=left?-ox:ox;b.vx=left?-Math.abs(b.vx):Math.abs(b.vx);
      if(bl.t===1){sWall();burst(b.x,b.y,'#9fd8ff',3,.6)}else hitBrick(r,c);
    }else{
      const up=b.y<byc;b.y+=up?-oy:oy;b.vy=up?-Math.abs(b.vy):Math.abs(b.vy);
      if(bl.t===1){sWall();burst(b.x,b.y,'#9fd8ff',3,.6)}else hitBrick(r,c);
    }
  }
}
function collidePaddle(b){
  if(b.vy>0&&b.y+b.r>=paddleY&&b.y-b.r<=paddleY+14&&Math.abs(b.x-paddle.x)<=paddle.w/2+b.r){
    const rel=Math.max(-1,Math.min(1,(b.x-paddle.x)/(paddle.w/2)));
    const a=rel*1.05;
    b.vy=-Math.cos(a)*b.sp;b.vx=Math.sin(a)*b.sp+(keys.l?-40:keys.r?40:0);
    const m=Math.hypot(b.vx,b.vy)||1;b.vx=b.vx/m*b.sp;b.vy=b.vy/m*b.sp;
    b.y=paddleY-b.r-.1;
    S.combo=0;S.speedUp=Math.min(.3,S.speedUp+.025);
    sPaddle();burst(b.x,paddleY,'#8ff5ff',4,.7);
  }
}
function updateBalls(dt){
  for(const b of balls){
    if(b.stuck){b.x=paddle.x;b.y=paddleY-b.r-1;continue}
    const target=ballSpeed();b.sp+=(target-b.sp)*Math.min(1,dt*3);
    const cur=Math.hypot(b.vx,b.vy)||1;b.vx*=b.sp/cur;b.vy*=b.sp/cur;
    const steps=Math.max(1,Math.ceil((b.sp*dt)/(b.r*.7)));
    for(let i=0;i<steps;i++){
      b.x+=b.vx*dt/steps;b.y+=b.vy*dt/steps;
      if(b.x-b.r<fx){b.x=fx+b.r;b.vx=Math.abs(b.vx);sWall();burst(fx,b.y,'#9fd8ff',3,.6)}
      else if(b.x+b.r>fx+fieldW){b.x=fx+fieldW-b.r;b.vx=-Math.abs(b.vx);sWall();burst(fx+fieldW,b.y,'#9fd8ff',3,.6)}
      if(b.y-b.r<py0){b.y=py0+b.r;b.vy=Math.abs(b.vy);sWall();burst(b.x,py0,'#9fd8ff',3,.6)}
      if(b.y-b.r>py1+40){b.dead=true;break}
      collideBlocks(b);collidePaddle(b);
    }
    b.trail.push({x:b.x,y:b.y});if(b.trail.length>12)b.trail.shift();
  }
  balls=balls.filter(b=>!b.dead);
  if(!balls.length&&S.mode==='playing')loseLife();
}
function updateLasers(dt){
  for(const L of lasers){
    const steps=5;let hit=false;
    for(let i=0;i<steps&&!hit;i++){
      L.y-=920*dt/steps;
      if(L.y<py0-14){L.dead=true;break}
      const c=Math.floor((L.x-fx)/s),r=Math.floor((L.y-fy)/s);
      if(c>=0&&c<COLS&&r>=0&&r<ROWS){const bl=blocks[r][c];
        if(bl.t===1){hit=true;burst(L.x,L.y,'#9fd8ff',4,.6);sWall()}
        else if(bl.t===2){hit=true;hitBrick(r,c)}}
    }
    if(hit)L.dead=true;
  }
  lasers=lasers.filter(L=>!L.dead);
}
function updateDrops(dt){
  for(const d of drops){
    d.y+=d.vy*dt;d.t+=dt;
    if(Math.abs(d.x-paddle.x)<paddle.w/2+14&&Math.abs(d.y-(paddleY-8))<20){applyPower(d.type);d.dead=true}
    if(d.y>py1+20)d.dead=true;
  }
  drops=drops.filter(d=>!d.dead);
}
function updateDemo(dt){
  const b=demoBall;b.x+=b.vx*dt;b.y+=b.vy*dt;
  if(b.x<fx){b.x=fx;b.vx=Math.abs(b.vx)}if(b.x>fx+fieldW){b.x=fx+fieldW;b.vx=-Math.abs(b.vx)}
  if(b.y<py0){b.y=py0;b.vy=Math.abs(b.vy)}if(b.y>py1-20){b.y=py1-20;b.vy=-Math.abs(b.vy)}
}

function update(dt){
  time+=dt;
  shake=Math.max(0,shake-dt*30);redFlash=Math.max(0,redFlash-dt*1.4);fireCd=Math.max(0,fireCd-dt);
  for(const d of dust){d.y-=d.v*dt;d.x-=d.v*.35*dt;if(d.y<-4){d.y=H+4;d.x=Math.random()*W}if(d.x<-4)d.x=W+4}
  if(S.mode==='playing'){
    fxT.wide=Math.max(0,fxT.wide-dt);fxT.slow=Math.max(0,fxT.slow-dt);fxT.laser=Math.max(0,fxT.laser-dt);
    updatePaddle(dt);
    if(fxT.laser>0&&spaceHeld)tryFire();
    updateBalls(dt);updateLasers(dt);updateDrops(dt);
  }else if(S.mode==='clearing'){
    clearT-=dt;
    if(Math.random()<.4)burst(fx+Math.random()*fieldW,fy+Math.random()*fieldH*.7,['#59f2ff','#ffb020','#e05aff'][Math.floor(Math.random()*3)],2,1);
    if(clearT<=0){S.level++;startLevel()}
  }else if(S.mode==='menu'&&demoBall){updateDemo(dt)}
  for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++){const b=blocks[r]&&blocks[r][c];if(b&&b.flash>0)b.flash=Math.max(0,b.flash-dt*5)}
  updateParts(dt);
  syncHUD(false);
}

/* ---------- rendering ---------- */
function rr(x,y,w,h,r){ctx.beginPath();ctx.moveTo(x+r,y);ctx.arcTo(x+w,y,x+w,y+h,r);ctx.arcTo(x+w,y+h,x,y+h,r);ctx.arcTo(x,y+h,x,y,r);ctx.arcTo(x,y,x+w,y,r);ctx.closePath()}
function drawFrame(){
  ctx.save();
  ctx.strokeStyle='rgba(120,180,255,.05)';ctx.lineWidth=1;
  for(let c=1;c<COLS;c++){const x=fx+c*s;ctx.beginPath();ctx.moveTo(x,fy);ctx.lineTo(x,fy+fieldH);ctx.stroke()}
  ctx.strokeStyle='rgba(46,230,255,.55)';ctx.lineWidth=2;ctx.shadowColor='rgba(46,230,255,.7)';ctx.shadowBlur=10;
  ctx.beginPath();ctx.moveTo(fx,py1);ctx.lineTo(fx,py0);ctx.lineTo(fx+fieldW,py0);ctx.lineTo(fx+fieldW,py1);ctx.stroke();
  ctx.shadowBlur=0;
  const L=16;ctx.strokeStyle='#7fe9ff';ctx.lineWidth=3;
  [[fx,py0,1,1],[fx+fieldW,py0,-1,1],[fx,py1,1,-1],[fx+fieldW,py1,-1,-1]].forEach(p=>{
    ctx.beginPath();ctx.moveTo(p[0]+p[2]*L,p[1]);ctx.lineTo(p[0],p[1]);ctx.lineTo(p[0],p[1]+p[3]*L);ctx.stroke()});
  ctx.setLineDash([4,10]);ctx.lineDashOffset=-time*30;ctx.strokeStyle='rgba(46,230,255,.25)';ctx.lineWidth=1;
  ctx.beginPath();ctx.moveTo(fx,py1);ctx.lineTo(fx+fieldW,py1);ctx.stroke();ctx.setLineDash([]);
  ctx.restore();
}
function drawBlocks(){
  for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++){
    const bl=blocks[r][c];if(!bl||bl.t===0)continue;
    const x=fx+c*s+1,y=fy+r*s+1,w=s-2,h=s-2;

      ctx.fillStyle='rgba(140,180,240,.35)';ctx.fillRect(x,y,w,3);

      if(s>=26){ctx.fillStyle='rgba(190,220,255,.3)';ctx.beginPath();ctx.arc(x+5,y+5,1.4,0,7);ctx.arc(x+w-5,y+h-5,1.4,0,7);ctx.fill()}

      ctx.fillStyle=col;ctx.fillRect(x,y,w,h);

      if(bl.hp<bl.max){

        ctx.moveTo(x+w*.2,y+h*.3);ctx.lineTo(x+w*(.4+seed*.08),y+h*.55);ctx.lineTo(x+w*.75,y+h*.4);

      }
      if(r===ROWS-1){const a=.35+.3*Math.sin(time*3+c*.7);ctx.fillStyle='rgba(255,220,140,'+a.toFixed(2)+')';ctx.fillRect(x+2,y+1,w-4,2)}
      if(bl.flash>0){ctx.fillStyle='rgba(255,255,255,'+(bl.flash*.8).toFixed(2)+')';ctx.fillRect(x,y,w,h)}
    }
  }
}

}
function render(){
