  chain=0;fx.wide=fx.zeroG=fx.slow=0;shield=false;netFlash=0;
  if(attract){setState('serve');serveDelay=.9;return;}
  lives--;SFX.life();shake(.4,10);addText(clamp(ball.x,60,W-60),H-34,'BALL LOST','#ff5a45',20);
  updateLivesUI();
  if(lives<=0)gameOver();else{ball.stuck=true;ball.trail.length=0;setState('serve');}
}
function gameOver(){
  setState('over');overLock=tGlobal+.7;SFX.over();
  const rec=score>bestAtStart&&score>0;
  if(score>best)best=score;saveBest();
  $('oScore').textContent=score;$('oLevel').textContent=level;$('oBricks').textContent=bricksTotal;
  document.querySelector('#oBest .v').textContent=best;
  $('newRec').style.display=rec?'block':'none';
}
function levelClear(){
  SFX.level();
  if(!attract)showBanner('LEVEL '+(level+1),'GRAVITY RISING // FIELD REBUILT');
  buildLevel(level+1);ball.stuck=true;ball.trail.length=0;setState('serve');
}

function hitBrick(br){
  br.hp--;chain++;
  if(br.hp>0){SFX.crack();spark(ball.x,ball.y,'#cfe6ff',4,H*.25);return;}
  br.alive=false;bricksLeft--;bricksTotal++;
  const mult=Math.min(2,1+.1*(chain-1)),pts=Math.round(br.sc*mult);
  score+=pts;if(score>best)best=score;
  addText(br.x+br.w/2,br.y+br.h/2,'+'+pts+(chain>=3?' ×'+chain:''),chain>=3?'#ffc247':'#dcebf7',13+Math.min(chain*.8,6));
  shardBurst(br);SFX.brick(br.r);if(level>=2)shake(.12,3);
  dropPU(br.x+br.w/2,br.y+br.h/2,br.type==='crys');
  if(bricksLeft<=0)levelClear();
}
function stepBall(h){
  const b=ball;
  b.vy+=(fx.zeroG>0?0:G)*h;
  const sp=Math.hypot(b.vx,b.vy);if(sp>maxSpd){b.vx*=maxSpd/sp;b.vy*=maxSpd/sp;}
  b.x+=b.vx*h;b.y+=b.vy*h;
  if(b.x<b.r){b.x=b.r;b.vx=Math.abs(b.vx);SFX.wall();spark(b.x,b.y,'#9fd8ff',5,H*.3);}
  else if(b.x>W-b.r){b.x=W-b.r;b.vx=-Math.abs(b.vx);SFX.wall();spark(b.x,b.y,'#9fd8ff',5,H*.3);}
  if(b.vy<0){
    const pL=pad.cx-pad.w/2,pR=pad.cx+pad.w/2;
    if(b.y-b.r<=faceY&&b.y>faceY-40&&b.x>pL-b.r*.5&&b.x<pR+b.r*.5){
      const o=clamp((b.x-pad.cx)/(pad.w/2),-1,1),a=o*1.0;
      b.vx=Math.sin(a)*launchS;b.vy=-Math.cos(a)*launchS;b.y=faceY+b.r;
      chain=0;pad.bounce=1;SFX.paddle();spark(b.x,faceY,'#ffc247',6,H*.3);
    }else if(b.y-b.r<=beamH){
      b.y=beamH+b.r;const keep=fx.zeroG>0?1:.5;b.vy=-b.vy*keep;
      SFX.ceiling();spark(b.x,beamH,'#8fb3d9',6,H*.25);
      if(keep<1)addText(b.x,beamH+16,'-ALT','#ff8a5c',11);
    }
  }
  for(const br of bricks){
    if(!br.alive)continue;
    if(b.x+b.r<br.x||b.x-b.r>br.x+br.w||b.y+b.r<br.y||b.y-b.r>br.y+br.h)continue;
    const cx=clamp(b.x,br.x,br.x+br.w),cy=clamp(b.y,br.y,br.y+br.h);
    let dx=b.x-cx,dy=b.y-cy;const d2=dx*dx+dy*dy;
    if(d2>b.r*b.r)continue;
    const d=Math.sqrt(d2);
    if(d<.001){b.y=br.y-b.r;b.vy=-Math.max(Math.abs(b.vy),H*.3);}
    else if(Math.abs(dx)>Math.abs(dy)){const s=dx>0?1:-1;b.x+=s*(b.r-d);b.vx=Math.abs(b.vx)*s;}
    else{const s=dy>0?1:-1;b.y+=s*(b.r-d);let nv=Math.abs(b.vy)*s;if(s<0)nv=-Math.max(Math.abs(nv),H*.3);b.vy=nv;}
    hitBrick(br);break;
  }
  if(state!=='play')return;
  if(shield&&b.vy>0&&b.y-b.r>=voidY){
    b.y=voidY+b.r;b.vy=-Math.max(H*.8,Math.abs(b.vy)*.85);shield=false;netFlash=1;
    SFX.shieldHit();addText(b.x,voidY-18,'SHIELD!','#6fe3c2',16);spark(b.x,voidY,'#6fe3c2',14,H*.5);return;
  }
  if(b.y-b.r>H+10)loseLife();
}
function movePaddle(dt){
  const ksp=Math.max(760,W*1.05);
  pad.tw=basePadW()*(fx.wide>0?1.7:1);
  if(keys.l)pad.cx-=ksp*dt;
  if(keys.r)pad.cx+=ksp*dt;
  if(pointerX!=null&&!keys.l&&!keys.r)pad.cx+=(pointerX-pad.cx)*Math.min(1,18*dt);
  pad.w+=(pad.tw-pad.w)*Math.min(1,14*dt);
  pad.bounce=Math.max(0,pad.bounce-6*dt);
  const half=pad.w/2+8;pad.cx=clamp(pad.cx,half,W-half);
}
function aiPaddle(dt){
  const target=ball.stuck?W/2:ball.x+Math.sin(tGlobal*1.3)*W*.045;
  const maxd=W*1.1*dt;
  pad.cx+=clamp(target-pad.cx,-maxd,maxd);
  pad.tw=basePadW();pad.w+=(pad.tw-pad.w)*Math.min(1,14*dt);
  pad.bounce=Math.max(0,pad.bounce-6*dt);
  const half=pad.w/2+8;pad.cx=clamp(pad.cx,half,W-half);
}
function updatePUs(dt){
  for(let i=pus.length-1;i>=0;i--){const p=pus[i];p.t+=dt;p.y+=p.vy*dt;p.x+=Math.sin(p.t*2.4)*16*dt;
    const pL=pad.cx-pad.w/2,pR=pad.cx+pad.w/2;
    if(p.x>pL-10&&p.x<pR+10&&p.y>beamH-4&&p.y<faceY+ball.r+8){applyPU(p);pus.splice(i,1);continue;}
    if(p.y<beamH+10){spark(p.x,p.y,p.c,6,H*.2);SFX.pop();pus.splice(i,1);}
  }
}
function updateParts(dt){for(let i=parts.length-1;i>=0;i--){const p=parts[i];p.t+=dt;if(p.t>p.life){parts.splice(i,1);continue;}p.vy+=p.g*dt;p.x+=p.vx*dt;p.y+=p.vy*dt;if(p.rot!==undefined)p.rot+=p.vr*dt;}}
function updateTexts(dt){for(let i=texts.length-1;i>=0;i--){const t=texts[i];t.t+=dt;t.y-=H*.05*dt;if(t.t>t.life)texts.splice(i,1);}}
function updateStars(dt){
  for(const s of starsFar){s.y+=s.v*dt;if(s.y>H+4){s.y=-4;s.x=Math.random()*W;}}
  for(const s of starsNear){s.y+=s.v*dt;if(s.y>H+4){s.y=-4;s.x=Math.random()*W;}}
}
function simUpdate(dt){
  fx.wide=Math.max(0,fx.wide-dt);fx.zeroG=Math.max(0,fx.zeroG-dt);fx.slow=Math.max(0,fx.slow-dt);
  shakeT=Math.max(0,shakeT-dt);netFlash=Math.max(0,netFlash-3*dt);
  updateStars(dt);
  const ts=fx.slow>0?.55:1,sdt=dt*ts;
  if(state==='serve'||state==='play'){
    if(attract)aiPaddle(dt);else movePaddle(dt);
    if(state==='serve'){
      ball.x=pad.cx;ball.y=faceY+ball.r+2;
      if(attract){serveDelay-=dt;if(serveDelay<=0)launch();}
    }else{
      let rem=sdt;const step=1/240;
      while(rem>1e-6&&state==='play'&&!ball.stuck){const hh=Math.min(step,rem);rem-=hh;stepBall(hh);}
      ball.trail.push({x:ball.x,y:ball.y});if(ball.trail.length>15)ball.trail.shift();
    }
  }
  updatePUs(sdt);updateParts(sdt);updateTexts(dt);
}

let last=performance.now();
function frame(now){
  requestAnimationFrame(frame);
