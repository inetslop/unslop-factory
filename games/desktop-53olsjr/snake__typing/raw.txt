/* ================= PATHFINDING ================= */
function findPath(sx,sy,tx,ty,blocked){
  const N=COLS*ROWS, prev=new Int32Array(N).fill(-1);
  const si=sy*COLS+sx, ti=ty*COLS+tx;
  if(si===ti) return [];
  const q=[si]; prev[si]=-2; let qi=0;
  while(qi<q.length){
    const cur=q[qi++], cx=cur%COLS, cy=(cur/COLS)|0;
    if(cur===ti) break;
    for(let d=0;d<4;d++){
      const nx=cx+(d===0?1:d===1?-1:0), ny=cy+(d===2?1:d===3?-1:0);
      if(nx<0||ny<0||nx>=COLS||ny>=ROWS) continue;
      const ni=ny*COLS+nx;
      if(prev[ni]!==-1 || (ni!==ti && blocked.has(ni))) continue;
      prev[ni]=cur; q.push(ni);
    }
  }
  if(prev[ti]===-1) return null;
  const path=[]; let cur=ti;
  while(cur!==-2 && cur!==-1){ path.push({x:cur%COLS,y:(cur/COLS)|0}); cur=prev[cur]; }
  path.reverse(); return path;
}
function pickSafe(){
  const blocked=new Set(snake.map(p=>p.y*COLS+p.x));
  let bestCell=null, bestScore=-1;
  for(let d=0;d<4;d++){
    const nx=dir.x? -dir.x : (d===0?1:d===2?-1:0); // placeholder, replaced below
  }
  const dirs=[{x:1,y:0},{x:-1,y:0},{x:0,y:1},{x:0,y:-1}];
  for(const dd of dirs){
    if(snake.length>1 && dd.x===-dir.x && dd.y===-dir.y) continue; // no reversing
    const nx=snake[0].x+dd.x, ny=snake[0].y+dd.y;
    if(nx<0||ny<0||nx>=COLS||ny>=ROWS) continue;
    if(blocked.has(ny*COLS+nx)) continue;

    for(const e of dirs){ const ex=nx+e.x, ey=ny+e.y; if(ex>=0&&ey>=0&&ex<COLS&&ey<ROWS && !blocked.has(ey*COLS+ex)) sc++; }

  }
  return bestCell;
}

/* ================= SIMULATION STEP ================= */
function doStep(){
  const h=snake[0];
  let next=null;
  const blocked=new Set(); for(let i=1;i<snake.length;i++) blocked.add(snake[i].y*COLS+snake[i].x);
  if(target){
    const path=findPath(h.x,h.y,target.x,target.y,blocked);
    if(path && path.length) next=path[0]; else next=pickSafe();
  }
  if(!next){ state==='menu' ? initDemo() : gameOver('self'); return; }

  const nx=h.x+next.x, ny=h.y+next.y;
  dir=next;
  if(nx<0||ny<0||nx>=COLS||ny>=ROWS){ state==='menu' ? initDemo() : gameOver('wall'); return; }
  const willGrow = growPending>0;
  for(let i=0;i<snake.length-(willGrow?0:1);i++){ if(snake[i].x===nx && snake[i].y===ny){ state==='menu' ? initDemo() : gameOver('self'); return; } }

  snake.unshift({x:nx,y:ny});
  if(willGrow) growPending--; else snake.pop();

  if(state==='play' && target && nx===target.x && ny===target.y){ damage(); }
  else if(state==='menu'){ newDemoTarget(); }
}

    SFX.wrong(); shake(4);

  }
}

}

/* ================= FX ================= */
function burst(x,y,color,n,spd){
  for(let i=0;i<n;i++){
    const a=Math.random()*Math.PI*2, v=(60+Math.random()*180)*spd/3;
    parts.push({x,y,vx:Math.cos(a)*v,vy:Math.sin(a)*v-40,life:400+Math.random()*350,ttl:750,size:2+Math.random()*3,color,grav:320});
  }
}
function initAmbient(){
  ambient.length=0;
  const chars='ABCDEFGHIJKLMNOPQRSTUVWXYZ', cols=['rgba(87,215,255,','rgba(93,242,160,','rgba(255,182,56,'];
  for(let i=0;i<26;i++) ambient.push({
    x:Math.random()*W, y:Math.random()*H, ch:chars[(Math.random()*26)|0],
    s:11+Math.random()*17, v:7+Math.random()*14, ph:Math.random()*6.28,
    a:0.04+Math.random()*0.05, c:cols[i%3]
  });
}

/* ================= LAYOUT / RESIZE ================= */
function resize(){
  W=window.innerWidth; H=window.innerHeight;
  dpr=Math.min(2, window.devicePixelRatio||1);
  cv.width=W*dpr; cv.height=H*dpr;
  cv.style.width=W+'px'; cv.style.height=H+'px';
  ctx.setTransform(dpr,0,0,dpr,0,0);
  if(!ambient.length) initAmbient();
}
window.addEventListener('resize', resize);
function layout(){
  const top = (state==='menu') ? 26 : hud.offsetHeight+16;
  cell=Math.max(8, Math.floor(Math.min((W-36)/COLS, (H-top-30)/ROWS)));
  bx=Math.round((W-COLS*cell)/2);
  by=Math.round(top+(H-top-ROWS*cell)/2);
  metricsOK=true;
}

/* ================= UPDATE ================= */
function effTick(){ return (state==='menu'?120:tickMs)*(boostTicks>0?0.55:1); }
function stepAccum(dt){
  const et=effTick(); let steps=0;
  acc+=dt;
  while(acc>=et && steps<6){ doStep(); acc-=et; if(boostTicks>0) boostTicks--; steps++; if(state!=='play'&&state!=='menu') break; }
  if(steps===6 || acc>et*3) acc=0;
}
function updateFx(dt){
  for(let i=parts.length-1;i>=0;i--){ const p=parts[i]; p.life-=dt; if(p.life<=0){ parts.splice(i,1); continue; }
    p.x+=p.vx*dt/1000; p.y+=p.vy*dt/1000; p.vy+=p.grav*dt/1000; }
  for(let i=floats.length-1;i>=0;i--){ const f=floats[i]; f.life-=dt; if(f.life<=0){ floats.splice(i,1); continue; } f.y-=42*dt/1000; }
  shakeMag=Math.max(0, shakeMag-dt*0.035);
}
function updateAmbient(dt,t){
  for(const a of ambient){
    a.y-=a.v*dt/1000;
    if(a.y<-30){ a.y=H+30; a.x=Math.random()*W; }
    a.dx=Math.sin(t/1400+a.ph)*9;
  }
}

/* ================= RENDER ================= */
function drawBoard(){
  ctx.fillStyle='rgba(6,13,24,0.82)';
  rr(bx-9,by-9,COLS*cell+18,ROWS*cell+18,10); ctx.fill();
  for(let y=0;y<ROWS;y++)for(let x=0;x<COLS;x++){
    if((x+y)&1){ ctx.fillStyle='rgba(125,200,255,0.03)'; ctx.fillRect(bx+x*cell,by+y*cell,cell,cell); }
  }
  ctx.strokeStyle='rgba(140,200,255,0.05)'; ctx.lineWidth=1; ctx.beginPath();
  for(let x=0;x<=COLS;x++){ ctx.moveTo(bx+x*cell+.5,by); ctx.lineTo(bx+x*cell+.5,by+ROWS*cell); }
  for(let y=0;y<=ROWS;y++){ ctx.moveTo(bx,by+y*cell+.5); ctx.lineTo(bx+COLS*cell,by+y*cell+.5); }
  ctx.stroke();
  ctx.save(); ctx.shadowColor='rgba(255,182,56,0.5)'; ctx.shadowBlur=16;
  ctx.strokeStyle='rgba(255,182,56,0.38)'; ctx.lineWidth=2;
  rr(bx-9,by-9,COLS*cell+18,ROWS*cell+18,10); ctx.stroke(); ctx.restore();
}
function drawTarget(t){
  if(!target) return;
  const px=bx+target.x*cell+cell/2, py=by+target.y*cell+cell/2;
  const pulse=(Math.sin(t/170)+1)/2;
  if(state==='menu'){
    ctx.save(); ctx.shadowColor='#5df2a0'; ctx.shadowBlur=14+10*pulse;
    ctx.fillStyle='rgba(93,242,160,'+(0.55+0.3*pulse)+')';
    ctx.beginPath(); ctx.arc(px,py,cell*(0.28+0.07*pulse),0,7); ctx.fill(); ctx.restore();
    return;
  }
  // expanding ring
  const rp=(t/950)%1;
  ctx.strokeStyle='rgba(255,182,56,'+(0.4*(1-rp))+')'; ctx.lineWidth=2;
  ctx.beginPath(); ctx.arc(px,py,cell*(0.5+0.7*rp),0,7); ctx.stroke();
  // tile
  const s=cell*0.88;
  ctx.save(); ctx.shadowColor='#ffb638'; ctx.shadowBlur=12+14*pulse;
  rr(px-s/2,py-s/2,s,s,5);
  ctx.fillStyle='rgba(255,182,56,'+(0.13+0.09*pulse)+')'; ctx.fill();
  ctx.strokeStyle='rgba(255,205,95,0.9)'; ctx.lineWidth=2; ctx.stroke();
  ctx.font='700 '+Math.floor(cell*0.56)+"px 'Courier New',monospace";
  ctx.textAlign='center'; ctx.textBaseline='middle';
  ctx.fillStyle='#ffe9b0';
  ctx.fillText(word[typedIdx], px, py+cell*0.03);
  ctx.restore();
}
function drawSnake(t){
  if(snake.length<2) return;
  const pts=snake.map(p=>({x:bx+p.x*cell+cell/2, y:by+p.y*cell+cell/2}));
  const n=pts.length;
  ctx.lineCap='round'; ctx.lineJoin='round';
  // dark outline
  ctx.beginPath(); ctx.moveTo(pts[0].x,pts[0].y);
  for(let i=1;i<n;i++) ctx.lineTo(pts[i].x,pts[i].y);
  ctx.strokeStyle='rgba(5,12,20,0.85)'; ctx.lineWidth=cell*0.8; ctx.stroke();
  // gradient body (tail -> head)
  for(let i=n-1;i>=1;i--){
    const f=1-i/(n-1);
    ctx.beginPath(); ctx.moveTo(pts[i].x,pts[i].y); ctx.lineTo(pts[i-1].x,pts[i-1].y);
    ctx.strokeStyle=lerpColor(TAIL,HEAD,f);
    ctx.lineWidth=cell*(0.62-0.2*i/n); ctx.stroke();
  }
  // sheen
  ctx.beginPath(); ctx.moveTo(pts[0].x,pts[0].y);
  for(let i=1;i<n;i++) ctx.lineTo(pts[i].x,pts[i].y);
  ctx.strokeStyle='rgba(255,255,255,0.1)'; ctx.lineWidth=cell*0.14; ctx.stroke();
  // head
  const h=pts[0], dx=dir.x, dy=dir.y;
  const flick=Math.sin(t/130);
  if(flick>0.25){ // tongue
    const mx=h.x+dx*cell*0.46, my=h.y+dy*cell*0.46, tl=cell*(0.28+0.14*flick);
    ctx.strokeStyle='#ff5d5d'; ctx.lineWidth=Math.max(1.5,cell*0.07);
    ctx.beginPath(); ctx.moveTo(mx,my); ctx.lineTo(mx+dx*tl,my+dy*tl);
    const ex=mx+dx*tl, ey=my+dy*tl, px=-dy, py2=dx;
    ctx.moveTo(ex,ey); ctx.lineTo(ex+dx*cell*0.12+px*cell*0.09, ey+dy*cell*0.12+py2*cell*0.09);
    ctx.moveTo(ex,ey); ctx.lineTo(ex+dx*cell*0.12-px*cell*0.09, ey+dy*cell*0.12-py2*cell*0.09);
    ctx.stroke();
  }
  const g=ctx.createRadialGradient(h.x-dx*3,h.y-dy*3,cell*0.05,h.x,h.y,cell*0.5);
  g.addColorStop(0,'#ffe27a'); g.addColorStop(1,'#f0a81f');
  ctx.fillStyle=g; ctx.beginPath(); ctx.arc(h.x,h.y,cell*0.47,0,7); ctx.fill();
  // eyes
  const exo=cell*0.16, fxo=cell*0.13;
  for(const s of [-1,1]){
    const ex=h.x+dx*fxo-dy*exo*s, ey=h.y+dy*fxo+dx*exo*s;
    ctx.fillStyle='#fff'; ctx.beginPath(); ctx.arc(ex,ey,cell*0.115,0,7); ctx.fill();
    ctx.fillStyle='#132030'; ctx.beginPath(); ctx.arc(ex+dx*cell*0.045,ey+dy*cell*0.045,cell*0.055,0,7); ctx.fill();
  }
}
function render(t){
  layout();
  ctx.clearRect(0,0,W,H);
  // ambient letters
  for(const a of ambient){
    ctx.font=a.s+"px 'Courier New',monospace";
    ctx.textAlign='center'; ctx.textBaseline='middle';
    ctx.fillStyle=a.c+a.a+')';
    ctx.fillText(a.ch, a.x+(a.dx||0), a.y);
  }
  ctx.save();
  if(shakeMag>0.2) ctx.translate((Math.random()-0.5)*2*shakeMag,(Math.random()-0.5)*2*shakeMag);
  drawBoard();
  drawTarget(t);
  drawSnake(t);
  // particles
  for(const p of parts){
    ctx.globalAlpha=Math.max(0,p.life/p.ttl);
    ctx.fillStyle=p.color;
    ctx.fillRect(p.x-p.size/2,p.y-p.size/2,p.size,p.size);
  }
  ctx.globalAlpha=1;
  // floating text
  for(const f of floats){
    ctx.globalAlpha=Math.max(0,f.life/f.ttl);
    ctx.font='700 '+Math.floor(f.big?Math.max(14,cell*0.62):Math.max(11,cell*0.46))+"px 'Courier New',monospace";
    ctx.textAlign='center'; ctx.textBaseline='middle';
    ctx.shadowColor=f.color; ctx.shadowBlur=8;
    ctx.fillStyle=f.color; ctx.fillText(f.txt,f.x,f.y);
    ctx.shadowBlur=0;
  }
  ctx.globalAlpha=1;
  ctx.restore();
}

/* ================= MAIN LOOP ================= */
let lastTs=0;
function loop(ts){
  requestAnimationFrame(loop);
  const dt=Math.min(50, ts-lastTs||16); lastTs=ts;
  updateAmbient(dt,ts);
  if(!paused && (state==='menu'||state==='play')) stepAccum(dt);
  if(!paused) updateFx(dt);
  render(ts);
}
