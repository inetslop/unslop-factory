function onImpact(a,b,sp,cx,cy){

 }else if(!proj&&(a.type==='iron'||b.type==='iron')&&sp>260)S.clank(clamp(sp/3000,.05,.25));

}

function physStep(dt){
 for(const b of bodies){if(!b.alive)continue;
  b.vy+=GRAV*dt;const f=Math.max(0,1-AIRK*dt);b.vx*=f;b.vy*=f;
  b.x+=b.vx*dt;b.y+=b.vy*dt;b.rot+=b.vr*dt;b.vr*=(1-.6*dt);
  b.flash-=dt;b.hitCd-=dt;b.rico-=dt;b.touched=false;}
 collide();collide();
 for(const b of bodies){if(!b.alive)continue;
  if(b.touched){const s2=b.vx*b.vx+b.vy*b.vy;if(s2<90){b.vx*=.5;b.vy*=.5;}b.vr*=.85;}
  if(b.y>H+200||b.x<-120||b.x>W+120)b.alive=false;}
}
function collide(){
 for(let i=0;i<bodies.length;i++){const a=bodies[i];if(!a.alive)continue;
  for(let j=i+1;j<bodies.length;j++){const b=bodies[j];if(!b.alive)continue;cc(a,b);}}
 for(const b of bodies){if(!b.alive)continue;for(const s of statics)cs(b,s);}
}
function cc(a,b){
 const dx=b.x-a.x,dy=b.y-a.y,rs=a.r+b.r,d2=dx*dx+dy*dy;
 if(d2>=rs*rs)return;
 let d=Math.sqrt(d2),nx,ny,pen;
 if(d<.001){nx=0;ny=-1;pen=rs;}else{nx=dx/d;ny=dy/d;pen=rs-d;}
 const im=a.invM+b.invM,corr=Math.max(pen-.35,0)/im*.8;
 a.x-=nx*corr*a.invM;a.y-=ny*corr*a.invM;b.x+=nx*corr*b.invM;b.y+=ny*corr*b.invM;
 const rvx=b.vx-a.vx,rvy=b.vy-a.vy,vn=rvx*nx+rvy*ny;
 a.touched=true;b.touched=true;
 if(vn<0){
  const e=Math.min(a.rest,b.rest),j=-(1+e)*vn/im;
  a.vx-=j*nx*a.invM;a.vy-=j*ny*a.invM;b.vx+=j*nx*b.invM;b.vy+=j*ny*b.invM;
  const tx=-ny,ty=nx,vt=rvx*tx+rvy*ty,jt=clamp(-vt/im,-.4*Math.abs(j),.4*Math.abs(j));
  a.vx-=jt*tx*a.invM;a.vy-=jt*ty*a.invM;b.vx+=jt*tx*b.invM;b.vy+=jt*ty*b.invM;
  onImpact(a,b,-vn,a.x+nx*a.r,a.y+ny*a.r);
 }
}
function cs(b,s){
 const cx=clamp(b.x,s.x,s.x+s.w),cy=clamp(b.y,s.y,s.y+s.h);
 let dx=b.x-cx,dy=b.y-cy,d2=dx*dx+dy*dy;
 if(d2>b.r*b.r)return;
 let nx,ny,pen;
 if(d2>1e-6){const d=Math.sqrt(d2);nx=dx/d;ny=dy/d;pen=b.r-d;}
 else{const l=b.x-s.x,r=s.x+s.w-b.x,t=b.y-s.y,bo=s.y+s.h-b.y,m=Math.min(l,r,t,bo);
  if(m===l){nx=-1;ny=0;pen=l+b.r;}else if(m===r){nx=1;ny=0;pen=r+b.r;}else if(m===t){nx=0;ny=-1;pen=t+b.r;}else{nx=0;ny=1;pen=bo+b.r;}}
 b.x+=nx*pen;b.y+=ny*pen;b.touched=true;
 const vn=b.vx*nx+b.vy*ny;
 if(vn<0){
  const e=b.rest;b.vx-=(1+e)*vn*nx;b.vy-=(1+e)*vn*ny;
  const tx=-ny,ty=nx,vt=b.vx*tx+b.vy*ty,jt=clamp(-vt,-.35*Math.abs(vn),.35*Math.abs(vn));
  b.vx-=jt*tx;b.vy-=jt*ty;
  if(b.type==='proj'){
   if(-vn>140)S.bump(-vn);
   if(s.f&&-vn>540&&b.rico<=0){addFloat(cx,cy,'RICOCHET',15,'#ffb636');S.tick(.8);b.rico=.6;}
  }
  if(-vn>420)burst(cx,cy,2,'#ffd98a');
 }
}

  for(const b of bodies){if(!b.alive||b.type!=='proj')continue;b.life+=dt;
   const sp=Math.hypot(b.vx,b.vy);if(sp<30)b.slowT+=dt;else b.slowT=0;
   b.trail.push({x:b.x,y:b.y});if(b.trail.length>8)b.trail.shift();

   for(const b of bodies)if(b.alive){if(b.type==='proj')anyP=true;const s=Math.hypot(b.vx,b.vy);if(s>mx)mx=s;}
   if(!anyP){if(mx<14)failT+=dt;else failT=0;if(failT>1.5)loseLife();}else failT=0;}
 }
