function tone(type,f0,f1,dur,vol,delay){
  if(!AC) return; delay=delay||0;

function physStep(h){
  time+=h;
  if(!dead){
    energy=Math.min(E_MAX, energy+REGEN*h);
    score+=12*h;
    if(combo>0 && time-lastCollect>2.4) combo=0;

    /* --- comet forces --- */
    let ax=0, ay=0;
    for(const w of wells){
      const dx=w.x-comet.x, dy=w.y-comet.y, d2=Math.max(dx*dx+dy*dy,MIN_D2), d=Math.sqrt(d2);
      const f=G_WELL/d2; ax+=dx/d*f; ay+=dy/d*f;
    }
    for(const hl of holes){
      if(hl.tel>0) continue;
      const mul=clamp(hl.life/.7,0,1);
      const dx=hl.x-comet.x, dy=hl.y-comet.y, d2=Math.max(dx*dx+dy*dy,(hl.rH+10)*(hl.rH+10)), d=Math.sqrt(d2);
      const f=G_HOLE*hl.str*mul/d2; ax+=dx/d*f; ay+=dy/d*f;
    }
    const A=Math.hypot(ax,ay), cap=A_MAX*S;
    if(A>cap){ ax=ax/A*cap; ay=ay/A*cap; }
    comet.vx+=ax*h; comet.vy+=ay*h;
    const v=Math.hypot(comet.vx,comet.vy), vc=V_MAX*S;
    if(v>vc){ comet.vx=comet.vx/v*vc; comet.vy=comet.vy/v*vc; }
    comet.x+=comet.vx*h; comet.y+=comet.vy*h;

    /* --- walls --- */
    const M=10*S, cr=comet.r;
    let hit=false, hx=comet.x, hy=comet.y;
    if(comet.x<M+cr){ comet.x=M+cr; comet.vx=Math.abs(comet.vx)*REST; hit=true; hy=M; }
    if(comet.x>W-M-cr){ comet.x=W-M-cr; comet.vx=-Math.abs(comet.vx)*REST; hit=true; hy=H-M; }
    if(comet.y<M+cr){ comet.y=M+cr; comet.vy=Math.abs(comet.vy)*REST; hit=true; hx=M; }
    if(comet.y>H-M-cr){ comet.y=H-M-cr; comet.vy=-Math.abs(comet.vy)*REST; hit=true; hx=W-M; }
    if(hit){
      burst(hx,hy,8,{col:'rgba(150,220,255,',sp:140*S,life:.35,size:1.8});
      shake=Math.max(shake,3.5);
      if(wallSndCd<=0){ sWall(); wallSndCd=.12; }
    }

    /* --- holes: collision & close shave --- */
    for(const hl of holes){
      if(hl.tel>0) continue;
      const d2=dist2(comet.x,comet.y,hl.x,hl.y);
      const capR=(hl.rH+cr);
      if(d2<capR*capR){ die(); return; }
      if(!hl.shaved){
        const sr=hl.rH*3.2;
        if(d2<sr*sr){ hl.shaved=true; score+=50; stats.shaves++; shake=Math.max(shake,6);
          pops.push({x:comet.x,y:comet.y-24*S,t:0,max:.9,txt:'CLOSE! +50',size:(17+8*S),color:'#ffb347',disp:true});
          sShave(); }
      }
    }

    /* --- motes --- */
    for(let i=motes.length-1;i>=0;i--){
      const m=motes[i];
      let mx=0,my=0;
      for(const w of wells){
        const dx=w.x-m.x, dy=w.y-m.y, d2=Math.max(dx*dx+dy*dy,MIN_D2), d=Math.sqrt(d2);
        const f=G_WELL*.28/d2; mx+=dx/d*f; my+=dy/d*f;
      }
      for(const hl of holes){
        if(hl.tel>0) continue;
        const dx=hl.x-m.x, dy=hl.y-m.y, d2=Math.max(dx*dx+dy*dy,(hl.rH+6)*(hl.rH+6)), d=Math.sqrt(d2);
        const f=G_HOLE*.5/d2; mx+=dx/d*f; my+=dy/d*f;
      }
      m.vx=(m.vx+mx*h)*Math.max(0,1-.4*h);
      m.vy=(m.vy+my*h)*Math.max(0,1-.4*h);
      const ms=Math.hypot(m.vx,m.vy), mc=320*S;
      if(ms>mc){ m.vx=m.vx/ms*mc; m.vy=m.vy/ms*mc; }
      m.x+=m.vx*h; m.y+=m.vy*h;

      let gone=false;
      for(const hl of holes){
        if(hl.tel<=0 && dist2(m.x,m.y,hl.x,hl.y) < (hl.rH+3)*(hl.rH+3)){
          burst(m.x,m.y,4,{col:'rgba(255,170,90,',sp:80*S,life:.3,size:1.6}); gone=true; break;
        }
      }
      if(gone){ motes.splice(i,1); continue; }
      const R=20*S+cr;
      if(dist2(m.x,m.y,comet.x,comet.y) < R*R){
        motes.splice(i,1);
        stats.motes++;
        combo = (time-lastCollect<2.4) ? Math.min(8,combo+1) : 1;
        lastCollect=time;
        const pts=10*combo; score+=pts;
        energy=Math.min(E_MAX,energy+5);
        burst(m.x,m.y,10,{col:'rgba(255,214,106,',sp:170*S,life:.45,size:2});
        pops.push({x:m.x,y:m.y-14*S,t:0,max:.7,txt:'+'+pts,size:(13+combo*1.6)*S+8,color:combo>=6?'#ffffff':combo>=4?'#ffd76a':'#8fe8ff',disp:false});
        hudCombo.textContent='×'+combo+' COMBO';
        hudCombo.style.color=combo>=6?'#ffffff':combo>=4?'#ffd76a':'#8fe8ff';
        hudCombo.classList.remove('on'); void hudCombo.offsetWidth; hudCombo.classList.add('on');
        sCollect(combo);
      } else if(m.x<-50||m.x>W+50||m.y<-50||m.y>H+50){ motes.splice(i,1); }
    }
  }

  /* --- wells lifetime (always) --- */
  for(let i=wells.length-1;i>=0;i--){
    const w=wells[i]; w.t-=h;
    if(w.t<=0){ burst(w.x,w.y,8,{col:'rgba(83,230,255,',sp:60*S,life:.4,size:1.8}); wells.splice(i,1); }
  }
  /* --- holes lifecycle (always) --- */
  for(let i=holes.length-1;i>=0;i--){
    const hl=holes[i];
    if(hl.tel>0){ hl.tel-=h; continue; }
    hl.life-=h;
    if(hl.life<=0){ burst(hl.x,hl.y,16,{col:'rgba(255,150,80,',sp:120*S,life:.5,size:2}); holes.splice(i,1); }
  }
  /* --- timers (not while dead) --- */
  if(!dead){
    holeTimer-=h;
    if(holeTimer<=0 && holes.length<3){ spawnHole(); holeTimer=Math.max(2.1,4.8-time*.032)*rand(.85,1.2); }
    moteTimer-=h;
    if(moteTimer<=0){ spawnMotes(); moteTimer=rand(1.2,1.9); }
  }
}

  }
  if(parts.length>420) parts.splice(0,parts.length-420);
}
function fxUpdate(dt){
  if(state==='playing' && comet && !dead){
    trail.push({x:comet.x,y:comet.y});
    if(trail.length>36) trail.shift();
  } else if(dead && trail.length){ trail.splice(0,2); }
  const pvx=(comet&&!dead)?comet.vx:0, pvy=(comet&&!dead)?comet.vy:0;
  for(const st of stars){
    const f=[.1,.25,.5][st.L];
    st.x-=pvx*f*dt*.06 + 4*f*dt; st.y-=pvy*f*dt*.06;
    if(st.x<0) st.x+=W; if(st.x>W) st.x-=W;
    if(st.y<0) st.y+=H; if(st.y>H) st.y-=H;
  }
  for(let i=parts.length-1;i>=0;i--){
    const p=parts[i]; p.t+=dt;
    if(p.t>p.max){ parts.splice(i,1); continue; }
    if(p.type==='ring') continue;
    p.x+=p.vx*dt; p.y+=p.vy*dt;
    const dr=Math.max(0,1-2.4*dt); p.vx*=dr; p.vy*=dr;
  }
  for(let i=pops.length-1;i>=0;i--){
    const p=pops[i]; p.t+=dt; p.y-=30*S*dt;
    if(p.t>p.max) pops.splice(i,1);
  }
  shake=Math.max(0,shake-shake*6*dt-2*dt);
}
function ambientSpawn(dt){
  if(Math.random()<dt*14){
    parts.push({x:rand(0,W),y:H+10,vx:rand(-8,8),vy:rand(-34,-14),t:0,max:rand(4,7),size:rand(.8,2),col:'rgba(150,190,255,',a0:.3});
  }
}

/* ================= RENDER ================= */
function render(t){
  ctx.setTransform(DPR,0,0,DPR,0,0);
  ctx.drawImage(nebula,0,0,W,H);

  /* stars */
  for(const st of stars){
    const a=(.22+.45*Math.abs(Math.sin(t*st.sp+st.tw)))*[.5,.75,1][st.L];
    ctx.fillStyle='rgba(200,220,255,'+a.toFixed(3)+')';
    ctx.beginPath(); ctx.arc(st.x,st.y,st.r,0,TAU); ctx.fill();
  }

  ctx.save();
  if(shake>0) ctx.translate(rand(-1,1)*shake, rand(-1,1)*shake);

  /* arena frame */
  ctx.strokeStyle='rgba(90,140,220,.14)'; ctx.lineWidth=1;
  ctx.strokeRect(6.5,6.5,W-13,H-13);
  ctx.strokeStyle='rgba(255,179,71,.3)'; ctx.lineWidth=2;
  const cl=26*S;
  [[8,8,1,1],[W-8,8,-1,1],[8,H-8,1,-1],[W-8,H-8,-1,-1]].forEach(c=>{
    ctx.beginPath();
    ctx.moveTo(c[0]+c[2]*cl,c[1]); ctx.lineTo(c[0],c[1]); ctx.lineTo(c[0],c[1]+c[3]*cl);
    ctx.stroke();
  });

  /* wells */
  ctx.globalCompositeOperation='lighter';
  for(const w of wells){
    const alpha=Math.min(1,w.t/.9), blink=(w.t<1)?(.5+.5*Math.sin(t*20)):1;
    const g=ctx.createRadialGradient(w.x,w.y,0,w.x,w.y,90*S);
    g.addColorStop(0,'rgba(83,230,255,'+(.2*alpha*blink).toFixed(3)+')');
    g.addColorStop(1,'rgba(83,230,255,0)');
    ctx.fillStyle=g; ctx.beginPath(); ctx.arc(w.x,w.y,90*S,0,TAU); ctx.fill();
    ctx.strokeStyle='rgba(140,245,255,'+(.75*alpha*blink).toFixed(3)+')';
    ctx.lineWidth=1.5; ctx.setLineDash([6,7]); ctx.lineDashOffset=-(t*30+w.ph*20);
    ctx.beginPath(); ctx.arc(w.x,w.y,24*S,0,TAU); ctx.stroke(); ctx.setLineDash([]);
    ctx.strokeStyle='rgba(225,252,255,'+(.9*alpha*blink).toFixed(3)+')';
    ctx.lineWidth=2;
    ctx.beginPath(); ctx.arc(w.x,w.y,31*S,-Math.PI/2,-Math.PI/2+TAU*(w.t/W_LIFE)); ctx.stroke();
    ctx.fillStyle='rgba(255,255,255,'+(.9*alpha).toFixed(3)+')';
    ctx.beginPath(); ctx.arc(w.x,w.y,2.6*S,0,TAU); ctx.fill();
  }
  /* motes */
  for(const m of motes){
    const tw=.7+.3*Math.sin(t*4+m.ph);
    ctx.fillStyle='rgba(255,214,106,'+(.12*tw).toFixed(3)+')';
    ctx.beginPath(); ctx.arc(m.x,m.y,m.r*3.2,0,TAU); ctx.fill();
    ctx.fillStyle='rgba(255,228,150,'+(.95*tw).toFixed(3)+')';
    ctx.beginPath(); ctx.arc(m.x,m.y,m.r*(1+.25*Math.sin(t*4+m.ph)),0,TAU); ctx.fill();
  }
  ctx.globalCompositeOperation='source-over';

  /* holes */
  for(const hl of holes){
    if(hl.tel>0){
      const p=1-hl.tel/.8, r=hl.rH*(2.6-1.6*p);
      ctx.strokeStyle='rgba(255,90,94,'+(.35+.4*Math.sin(t*16)).toFixed(3)+')';
      ctx.lineWidth=1.5; ctx.setLineDash([5,6]);
      ctx.beginPath(); ctx.arc(hl.x,hl.y,r,0,TAU); ctx.stroke(); ctx.setLineDash([]);
      ctx.fillStyle='rgba(255,90,94,'+(.8*p).toFixed(3)+')';
      ctx.beginPath(); ctx.arc(hl.x,hl.y,p*4*S,0,TAU); ctx.fill();
      continue;
    }
    const mul=clamp(hl.life/.7,0,1);
    ctx.globalCompositeOperation='lighter';
    const gg=ctx.createRadialGradient(hl.x,hl.y,hl.rH*.6,hl.x,hl.y,hl.rH*3);
    gg.addColorStop(0,'rgba(255,120,60,'+(.3*mul).toFixed(3)+')');
    gg.addColorStop(1,'rgba(255,120,60,0)');
    ctx.fillStyle=gg; ctx.beginPath(); ctx.arc(hl.x,hl.y,hl.rH*3,0,TAU); ctx.fill();
    ctx.lineWidth=2.5;
    ctx.strokeStyle='rgba(255,170,90,'+(.85*mul).toFixed(3)+')';
    ctx.beginPath(); ctx.arc(hl.x,hl.y,hl.rH+4*S,t*2+hl.ph,t*2+hl.ph+4.6); ctx.stroke();
    ctx.strokeStyle='rgba(255,230,180,'+(.5*mul).toFixed(3)+')';
    ctx.beginPath(); ctx.arc(hl.x,hl.y,hl.rH+7*S,-t*2.6+hl.ph,-t*2.6+hl.ph+4.2); ctx.stroke();
    for(let k=0;k<3;k++){
      const a=t*3+k*TAU/3+hl.ph;
      ctx.fillStyle='rgba(255,210,150,'+(.8*mul).toFixed(3)+')';
      ctx.beginPath(); ctx.arc(hl.x+Math.cos(a)*hl.rH*1.75, hl.y+Math.sin(a)*hl.rH*1.75, 1.6*S,0,TAU); ctx.fill();
    }
    ctx.globalCompositeOperation='source-over';
    ctx.fillStyle='#04060c';
    ctx.beginPath(); ctx.arc(hl.x,hl.y,hl.rH,0,TAU); ctx.fill();
    ctx.strokeStyle='rgba(255,120,80,'+(.9*mul).toFixed(3)+')'; ctx.lineWidth=1;
    ctx.beginPath(); ctx.arc(hl.x,hl.y,hl.rH,0,TAU); ctx.stroke();
  }

  /* comet + trail */
  if(comet && !dead){
    ctx.globalCompositeOperation='lighter';
    ctx.lineCap='round';
    for(let i=0;i<trail.length-1;i++){
      const p=trail[i], q=trail[i+1], a=i/trail.length;
      ctx.strokeStyle='rgba(255,180,90,'+(a*.5).toFixed(3)+')';
      ctx.lineWidth=Math.max(.5,a*6*S);
      ctx.beginPath(); ctx.moveTo(p.x,p.y); ctx.lineTo(q.x,q.y); ctx.stroke();
    }
    const cg=ctx.createRadialGradient(comet.x,comet.y,0,comet.x,comet.y,30*S);
    cg.addColorStop(0,'rgba(255,190,90,.6)'); cg.addColorStop(1,'rgba(255,140,50,0)');
    ctx.fillStyle=cg; ctx.beginPath(); ctx.arc(comet.x,comet.y,30*S,0,TAU); ctx.fill();
    ctx.globalCompositeOperation='source-over';
    ctx.fillStyle='#fff7e6';
    ctx.beginPath(); ctx.arc(comet.x,comet.y,5*S,0,TAU); ctx.fill();
    ctx.strokeStyle='rgba(255,179,71,.9)'; ctx.lineWidth=2;
    ctx.beginPath(); ctx.arc(comet.x,comet.y,comet.r,0,TAU); ctx.stroke();
  }

  /* particles */
  ctx.globalCompositeOperation='lighter';
  for(const p of parts){
    const k=p.t/p.max, a=(1-k)*(p.a0||.9);
    if(p.type==='ring'){
      const e=1-Math.pow(1-k,3), r=p.r0+(p.r1-p.r0)*e;
      ctx.strokeStyle='rgba('+p.col+','+(a*.8).toFixed(3)+')';
      ctx.lineWidth=2;
      ctx.beginPath(); ctx.arc(p.x,p.y,r,0,TAU); ctx.stroke();
    } else {
      ctx.fillStyle=p.col+a.toFixed(3)+')';
      ctx.beginPath(); ctx.arc(p.x,p.y,Math.max(.4,p.size*(1-k*.5)),0,TAU); ctx.fill();
    }
  }
  ctx.globalCompositeOperation='source-over';

  /* popups */
  ctx.textAlign='center';
  for(const p of pops){
    const a=1-p.t/p.max;
    ctx.globalAlpha=a;
    ctx.font = p.disp ? (p.size)+'px Impact, "Arial Black", sans-serif' : '700 '+p.size+'px "Segoe UI", sans-serif';
    ctx.fillStyle=p.color;
    ctx.fillText(p.txt,p.x,p.y);
  }
  ctx.globalAlpha=1;
  ctx.restore();

  /* reticle */
  if(state==='playing' && !dead){
    const ok = energy>=W_COST && placeCd<=0;
    const col = ok ? 'rgba(140,245,255,' : 'rgba(255,90,94,';
    ctx.strokeStyle=col+'.85)'; ctx.lineWidth=1.5;
    ctx.beginPath(); ctx.arc(pointer.x,pointer.y,13,0,TAU); ctx.stroke();
    for(let k=0;k<4;k++){
      const a=t*1.2+k*Math.PI/2;
      ctx.beginPath();
      ctx.moveTo(pointer.x+Math.cos(a)*16, pointer.y+Math.sin(a)*16);
      ctx.lineTo(pointer.x+Math.cos(a)*21, pointer.y+Math.sin(a)*21);
      ctx.stroke();
    }
    ctx.fillStyle=col+'.9)';
    ctx.beginPath(); ctx.arc(pointer.x,pointer.y,2,0,TAU); ctx.fill();
    if(placeCd>0){
      ctx.strokeStyle='rgba(255,179,71,.9)'; ctx.lineWidth=2.5;
      ctx.beginPath(); ctx.arc(pointer.x,pointer.y,13,-Math.PI/2,-Math.PI/2+TAU*(placeCd/P_CD)); ctx.stroke();
    }
  }
}

}

const H_STEP=1/120;
let last=performance.now(), acc=0;
function frame(now){
  requestAnimationFrame(frame);
  let dt=(now-last)/1000; last=now; dt=Math.min(dt,.05);
  if(state==='playing'){
    const sdt=dt*timeScale;
    acc+=sdt;
    let n=0;
    while(acc>=H_STEP && n<10){ physStep(H_STEP); acc-=H_STEP; n++; }
    placeCd=Math.max(0,placeCd-dt);
    wallSndCd=Math.max(0,wallSndCd-dt);
    if(pointer.down && !dead) tryPlace(pointer.x,pointer.y);
    fxUpdate(sdt);
    if(dead){ deadT+=dt; if(deadT>.95) finishGame(); }
  } else if(state!=='paused'){
    fxUpdate(dt); ambientSpawn(dt);
  }
  render(now/1000);
  updateHUD();
}

resize();
requestAnimationFrame(frame);
})();
