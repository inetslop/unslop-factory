function step(dt){state.time+=dt;
  if(state.mode==='menu'){
    for(const z of zombies){z.wanderT-=dt;if(z.wanderT<=0){z.dir=rand(0,TAU);z.wanderT=rand(1.5,4);}
      z.x+=Math.cos(z.dir)*z.sp*.4*dt;z.y+=Math.sin(z.dir)*z.sp*.4*dt;collideCircle(z,z.r);}
    const tx=clamp(AW/2-viewW/2+Math.cos(state.time*.07)*160,(AW-viewW)/2,(AW-viewW)/2);

    cam.x+=(tx-cam.x)*Math.min(1,dt*1.5);cam.y+=(((AH-viewH)/2)-cam.y)*Math.min(1,dt*1.5);
  }
  else if(state.mode==='playing'){playerUpdate(dt);waveUpdate(dt);}
  else if(state.mode==='dying'){}
  // bullets
  for(let i=bullets.length-1;i>=0;i--){const b=bullets[i];b.x+=b.vx*dt;b.y+=b.vy*dt;b.life-=dt;
    let dead=b.life<=0||b.x<30||b.x>AW-30||b.y<30||b.y>AH-30;
    if(!dead)for(const o of obstacles){if(o.dead)continue;
      if(dist2(b.x,b.y,o.x,o.y)<o.r*o.r){dead=true;
        for(let k=0;k<4;k++)parts.push({kind:'spark',x:b.x,y:b.y,vx:rand(-160,160),vy:rand(-160,160),t:0,life:.2,size:2,color:'#ffd98a'});
        if(o.kind==='barrel'){o.hp-=14;if(o.hp<=0){o.dead=true;blasts.push({x:o.x,y:o.y,t:.05});}}break;}}
    if(!dead&&!P.dead)for(const z of zombies){if(z.dead||z.wander)continue;
      if(dist2(b.x,b.y,z.x,z.y)<(z.r+3)*(z.r+3)){dead=true;
        damageZombie(z,b.dmg,Math.atan2(b.vy,b.vx),90);break;}}
    if(dead)bullets.splice(i,1);}
  // globs
  for(let i=globs.length-1;i>=0;i--){const g=globs[i];g.t+=dt;const p=g.t/g.dur;
    g.x=g.sx+(g.tx-g.sx)*p;g.y=g.sy+(g.ty-g.sy)*p;let hit=p>=1;
    if(!hit)for(const o of obstacles){if(o.dead)continue;if(dist2(g.x,g.y,o.x,o.y)<(o.r+4)*(o.r+4)){hit=true;break}}
    if(hit)splashGlob(g);globs.splice(i,1);}

    parts.push({kind:'dust',x:g.x,y:g.y,vx:rand(-8,8),vy:rand(-8,8),t:0,life:.35,size:3,color:'#7a8268'});
    let hit=g.life<=0||g.x<34||g.x>AW-34||g.y<34||g.y>AH-34;

    if(!P.dead&&dist2(P.x,P.y,pd.x,pd.y)<(pd.r+P.r)*(pd.r+P.r)&&pd.tick<=0){pd.tick=.5;hurtPlayer(9,'acid');}

      let hit=null;for(const o of obstacles){if(o.dead)continue;if(dist2(px,py,o.x,o.y)<o.r*o.r){hit=o;break}}
      if(hit){const cross=tx*(hit.y-z.y)-ty*(hit.x-z.x);ang+=(cross>0?-1:1)*.8;tx=Math.cos(ang);ty=Math.sin(ang);}}
    z.x+=(tx*sp+z.kx)*dt;z.y+=(ty*sp+z.ky)*dt;
    z.kx*=Math.max(0,1-dt*6);z.ky*=Math.max(0,1-dt*6);collideCircle(z,z.r);
    if(!P.dead&&dist2(z.x,z.y,P.x,P.y)<(z.r+P.r+6)*(z.r+P.r+6)&&z.atkCd<=0){

    for(let j=i+1;j<zombies.length;j++){const b=zombies[j];if(b.dead)continue;

  if(active&&!P.dead)for(let i=pickups.length-1;i>=0;i--){const p=pickups[i];
    if(dist2(P.x,P.y,p.x,p.y)<(P.r+16)*(P.r+16)){pickups.splice(i,1);sPickup();

    if(P.hp<30&&state.hbT<=0){sHeart();state.hbT=P.hp<15?.7:1.1;}

  if(P){P.kick=Math.max(0,P.kick-dt*6);P.hitMark=Math.max(0,P.hitMark-dt);}
  stepFx(dt);
  // camera
  const tx=clamp(P.x-viewW/2,(AW-viewW)/2,(AW-viewV||viewH)/2< (AH-viewH)/2?(AH-viewH)/2:(AH-viewH)/2),ty=clamp(P.y-viewH/2,(AH-viewH)/2,(AH-viewH)/2);

function stepFx(dt){
  for(let i=parts.length-1;i>=0;i--){const p=parts[i];p.t+=dt;
    if(p.t>=p.life){parts.splice(i,1);continue}
    const dr=p.kind==='smoke'?.4:p.kind==='blood'?.12:.2;p.vx*=Math.pow(dr,dt);p.vy*=Math.pow(dr,dt);
    p.x+=p.vx*dt;p.y+=p.vy*dt;if(p.kind==='smoke')p.size+=dt*22;}
  if(parts.length>460)parts.splice(0,parts.length-460);

  for(let i=floats.length-1;i>=0;i--){const f=floats[i];f.t+=dt;f.y-=34*dt;if(f.t>=f.life)floats.splice(i,1);}
  for(let i=corpses.length-1;i>=0;i--){corpses[i].t+=dt;if(corpses[i].t>=corpses[i].life)corpses.splice(i,1);}
  for(let i=lights.length-1;i>=0;i--){lights[i].t-=dt;if(lights[i].t<=0)lights.splice(i,1);}
