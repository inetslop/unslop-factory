  if(!actx||muted)return;type=type||'triangle';vol=vol||.12;delay=delay||0;
  const t0=actx.currentTime+delay,o=actx.createOscillator(),g=actx.createGain();
  o.type=type;o.frequency.setValueAtTime(f,t0);
  if(fEnd)o.frequency.exponentialRampToValueAtTime(Math.max(20,fEnd),t0+dur);
  g.gain.setValueAtTime(vol,t0);g.gain.exponentialRampToValueAtTime(.0001,t0+dur);
  o.connect(g).connect(actx.destination);o.start(t0);o.stop(t0+dur+.03);
}
function noise(dur,vol,fq,delay){
  if(!actx||muted)return;delay=delay||0;
  const t0=actx.currentTime+delay,n=(actx.sampleRate*dur)|0,buf=actx.createBuffer(1,n,actx.sampleRate),d=buf.getChannelData(0);
  for(let i=0;i<n;i++)d[i]=(Math.random()*2-1)*(1-i/n);
  const s=actx.createBufferSource();s.buffer=buf;
  const f=actx.createBiquadFilter();f.type='lowpass';f.frequency.value=fq;
  const g=actx.createGain();g.gain.setValueAtTime(vol,t0);g.gain.exponentialRampToValueAtTime(.0001,t0+dur);
  s.connect(f).connect(g).connect(actx.destination);s.start(t0);
}

function newBoard(){
  minesCount=Math.min(24,8+level*2);
  safeTotal=N*N-minesCount;clearedSafe=0;
  winPending=false;losePending=false;combo=0;selected=null;anim=null;popping=[];pendingSpawns=[];phase='idle';
  grid=[];
  for(let r=0;r<N;r++){const row=[];for(let c=0;c<N;c++)
    row.push({r,c,gem:-1,mine:false,cleared:false,blown:false,special:false,popT:0,fall:0,vy:0,adjMines:0,revealAt:0,bornAt:0});
  grid.push(row);}
  const idx=shuffle([...Array(N*N).keys()]);
  for(let i=0;i<minesCount;i++)grid[(idx[i]/N)|0][idx[i]%N].mine=true;
  fillGems();
  let g=0;while(!hasMoves()&&g++<80)refillGems();
}
function fillGems(){
  for(let r=0;r<N;r++)for(let c=0;c<N;c++){
    const o=grid[r][c],bad=[];
    if(c>=2&&grid[r][c-1].gem===grid[r][c-2].gem)bad.push(grid[r][c-1].gem);
    if(r>=2&&grid[r-1][c].gem===grid[r-2][c].gem)bad.push(grid[r-1][c].gem);
    let t;do{t=ri(6);}while(bad.indexOf(t)>=0);
    o.gem=t;o.special=false;
  }
}
function refillGems(){for(let r=0;r<N;r++)for(let c=0;c<N;c++){grid[r][c].gem=ri(6);grid[r][c].special=false;}}

/* ================= match logic ================= */
function findRuns(){
  const cells=new Set(),runs=[];
  for(let r=0;r<N;r++){let c=0;while(c<N){const t=grid[r][c].gem;
    if(t<0){c++;continue;}let e=c+1;while(e<N&&grid[r][e].gem===t)e++;
    if(e-c>=3){const rc=[];for(let i=c;i<e;i++){cells.add(r+','+i);rc.push([r,i]);}runs.push(rc);}c=e;}}
  for(let c=0;c<N;c++){let r=0;while(r<N){const t=grid[r][c].gem;
    if(t<0){r++;continue;}let e=r+1;while(e<N&&grid[e][c].gem===t)e++;
    if(e-r>=3){const rc=[];for(let i=r;i<e;i++){cells.add(i+','+c);rc.push([i,c]);}runs.push(rc);}r=e;}}
  return {cells,runs};
}
function hasMoves(){
  for(let r=0;r<N;r++)for(let c=0;c<N;c++)if(grid[r][c].special)return true;
  const ok=(r,c)=>{const g=grid[r][c].gem;if(g<0)return false;let n=1;
    for(let i=c-1;i>=0&&grid[r][i].gem===g;i--)n++;for(let i=c+1;i<N&&grid[r][i].gem===g;i++)n++;if(n>=3)return true;
    n=1;for(let i=r-1;i>=0&&grid[i][c].gem===g;i--)n++;for(let i=r+1;i<N&&grid[i][c].gem===g;i++)n++;return n>=3;};
  const sw=(a,b,a2,b2)=>{const v=grid[a][b].gem;grid[a][b].gem=grid[a2][b2].gem;grid[a2][b2].gem=v;};
  for(let r=0;r<N;r++)for(let c=0;c<N;c++){
    if(c+1<N){sw(r,c,r,c+1);const v=ok(r,c)||ok(r,c+1);sw(r,c,r,c+1);if(v)return true;}
    if(r+1<N){sw(r,c,r+1,c);const v=ok(r,c)||ok(r+1,c);sw(r,c,r+1,c);if(v)return true;}}
  return false;
}
function reshuffle(){
  let g=0;do{refillGems();g++;}while(g<80&&(!hasMoves()||findRuns().cells.size>0));
  for(let i=0;i<24;i++)particles.push({x:rand(GAP,boardPx-GAP),y:rand(GAP,boardPx-GAP),vx:rand(-15,15),vy:rand(-30,-8),g:-20,life:rand(.3,.6),max:.6,size:rand(1.4,2.6),col:'#9fb4d8',add:true});
  setMsg('No moves — field rearranged','info');sfx.ui();
}

/* ================= gameplay flow ================= */
function previewRuns(ra,ca,rb,cb){
  const v=grid[ra][ca].gem;grid[ra][ca].gem=grid[rb][cb].gem;grid[rb][cb].gem=v;
  const res=findRuns();
  const w=grid[ra][ca].gem;grid[ra][ca].gem=grid[rb][cb].gem;grid[rb][cb].gem=w;
  return res;
}
function attemptSwap(r,c,r2,c2){
  if(state!=='playing'||phase!=='idle')return;
  if(Math.abs(r-r2)+Math.abs(c-c2)!==1)return;
  const a=grid[r][c],b=grid[r2][c2];
  if(a.gem<0||b.gem<0)return;
  if(a.special||b.special){ // bomb crystal detonation
    const br=a.special?r:r2,bc=a.special?c:c2;
    combo=1;const clearSet=new Set();processBlasts([[br,bc]],clearSet);
    setMsg('BOMB CRYSTAL! 3×3 blast','warn');startPop(clearSet,[]);return;
  }
  const runs=previewRuns(r,c,r2,c2);
  anim={ra:r,ca:c,rb:r2,cb:c2,t:0,valid:runs.cells.size>0,runs};
  phase='swapping';sfx.swap();selected=null;
}
function processBlasts(queue,clearSet){
  const seen=new Set(queue.map(k=>k.join(',')));
  while(queue.length){const b=queue.pop(),r=b[0],c=b[1];
    for(let dr=-1;dr<=1;dr++)for(let dc=-1;dc<=1;dc++){
      const nr=r+dr,nc=c+dc;if(nr<0||nc<0||nr>=N||nc>=N)continue;
      const key=nr+','+nc;clearSet.add(key);
      if(grid[nr][nc].special&&!seen.has(key)){seen.add(key);queue.push([nr,nc]);}
    }}
}
function beginClear(runs){
  const clearSet=new Set(runs.cells),spawns=[],spawnKeys=new Set();
  for(const run of runs.runs)if(run.length>=5){
    const mid=run[(run.length/2)|0],key=mid[0]+','+mid[1];
    if(!spawnKeys.has(key)){spawnKeys.add(key);clearSet.delete(key);spawns.push([mid[0],mid[1]]);}
  }
  const q=[];for(const key of clearSet){const p=key.split(',').map(Number);if(grid[p[0]][p[1]].special)q.push(p);}
  processBlasts(q,clearSet);
  startPop(clearSet,spawns);
}
function startPop(clearSet,spawns){
  popping=[];pendingSpawns=spawns||[];
  for(const key of clearSet){const p=key.split(',').map(Number),o=grid[p[0]][p[1]];
    if(pendingSpawns.some(s=>s[0]===p[0]&&s[1]===p[1]))continue;
    o.popT=.0001;popping.push(o);burstSpark(o.r,o.c,GEMS[o.gem].m,5);}
  phase='popping';sfx.pop(combo);
}
function commitClear(){
  let mineHits=0,gemsCleared=0,msg=null;
  for(const o of popping){
    o.special=false;o.popT=0;o.gem=-1;gemsCleared++;
    const cx=GAP+o.c*pitch+cell/2,cy=GAP+o.r*pitch+cell/2;
    if(o.mine&&!o.blown){o.blown=true;o.revealAt=timeSec;mineHits++;boomFx(cx,cy);}
    else{o.cleared=true;o.adjMines=countAdj(o.r,o.c);o.revealAt=timeSec;clearedSafe++;tilesTotal++;score+=25*combo;dustFx(cx,cy,true);}
  }
  for(const s of pendingSpawns){const o=grid[s[0]][s[1]];o.special=true;o.bornAt=timeSec;
    rings.push({x:GAP+s[1]*pitch+cell/2,y:GAP+s[0]*pitch+cell/2,t:0,dur:.45,maxR:cell*1.1,col:'#ffd23e'});
    msg='BOMB CRYSTAL forged!';}
  score+=gemsCleared*10*combo;
  if(combo>=2)msg='Cascade ×'+combo+'!';
  if(mineHits>0){shields=Math.max(0,shields-mineHits);shake=Math.min(16,shake+5*mineHits);flashT=.4;
    sfx.boom();bumpShields();msg='MINE HIT! −'+mineHits+' shield'+(mineHits>1?'s':'');}
  if(msg)setMsg(msg,mineHits>0?'bad':(combo>=2?'good':'warn'));
  pendingSpawns=[];popping=[];updateHUD();
  if(shields<=0)losePending=true;else if(clearedSafe>=safeTotal)winPending=true;
  applyGravity();
}
function countAdj(r,c){let n=0;for(let dr=-1;dr<=1;dr++)for(let dc=-1;dc<=1;dc++){
  if(!dr&&!dc)continue;const nr=r+dr,nc=c+dc;if(nr<0||nc<0||nr>=N||nc>=N)continue;
  if(grid[nr][nc].mine)n++;}return n;}
function applyGravity(){
  let any=false;
  for(let c=0;c<N;c++){let w=N-1;
    for(let r=N-1;r>=0;r--){const o=grid[r][c];if(o.gem<0)continue;
      if(r!==w){const d=grid[w][c];d.gem=o.gem;d.special=o.special;d.bornAt=o.bornAt;d.fall=(w-r)*pitch;d.vy=0;
        o.gem=-1;o.special=false;o.fall=0;any=true;}
      w--;}
    for(let r=w;r>=0;r--){const o=grid[r][c];o.gem=ri(6);o.special=false;o.fall=(w+1)*pitch;o.vy=0;any=true;}
  }
  phase='falling';if(!any)finishResolve();
}
function finishResolve(){
  if(losePending){gameOver();return;}
  if(winPending){levelComplete();return;}
  const runs=findRuns();
  if(runs.cells.size){combo++;beginClear(runs);return;}
  phase='idle';if(!hasMoves())reshuffle();
}

}

    ctx.closePath();ctx.fillStyle='#fff6d8';ctx.fill();ctx.restore();
    const oa=timeSec*2.6;ctx.fillStyle='#ffd23e';
    ctx.beginPath();ctx.arc(Math.cos(oa)*s*.95,Math.sin(oa)*s*.95,1.8,0,6.283);ctx.fill();
  }
  ctx.restore();
}
function strokeCell(r,c,col,w){rrPath(GAP+c*pitch+1,GAP+r*pitch+1,cell-2,cell-2,5);
  ctx.strokeStyle=col;ctx.lineWidth=w;ctx.stroke();}
function strokeCursor(r,c){
  const x=GAP+c*pitch,y=GAP+r*pitch,L=Math.min(10,cell*.3),m=2.5;
  ctx.strokeStyle='#ffb020';ctx.lineWidth=2;
  const cs=[[x+m,y+m,1,1],[x+cell-m,y+m,-1,1],[x+m,y+cell-m,1,-1],[x+cell-m,y+cell-m,-1,-1]];
  for(const p of cs){ctx.beginPath();ctx.moveTo(p[0]+p[2]*L,p[1]);ctx.lineTo(p[0],p[1]);ctx.lineTo(p[0],p[1]+p[3]*L);ctx.stroke();}
}
function drawBoard(){
  ctx.setTransform(dpr,0,0,dpr,0,0);
  ctx.clearRect(0,0,boardPx,boardPx);
  rrPath(1.5,1.5,boardPx-3,boardPx-3,9);ctx.fillStyle='#141b29';ctx.fill();
  ctx.strokeStyle='#2c3a58';ctx.lineWidth=1;ctx.stroke();
  if(shake>0)ctx.translate(rand(-shake,shake)*.5,rand(-shake,shake)*.5);
  for(let r=0;r<N;r++)for(let c=0;c<N;c++)drawFloor(grid[r][c],GAP+c*pitch,GAP+r*pitch);
  // gem shadows
  ctx.fillStyle='rgba(0,0,0,.22)';
  for(let r=0;r<N;r++)for(let c=0;c<N;c++){const o=grid[r][c];if(o.gem<0||o.fall>2)continue;
    ctx.beginPath();ctx.ellipse(GAP+c*pitch+cell/2,GAP+r*pitch+cell*.82,cell*.28,cell*.09,0,0,6.283);ctx.fill();}
  // swap offsets
  const offs={};
  if(anim){const u=anim.t/(anim.valid?SWAP_T:SWAP_T*2);
    let p=u<=.5?ease(u*2):ease(2-u*2);if(!anim.valid&&u>=1)p=0;
    const ax=GAP+anim.ca*pitch,ay=GAP+anim.ra*pitch,bx=GAP+anim.cb*pitch,by=GAP+anim.rb*pitch;
    offs[anim.ra+','+anim.ca]={dx:(bx-ax)*p,dy:(by-ay)*p};
    offs[anim.rb+','+anim.cb]={dx:(ax-bx)*p,dy:(ay-by)*p};}
  for(let r=0;r<N;r++)for(let c=0;c<N;c++){const o=grid[r][c];if(o.gem<0)continue;
    let dx=0,dy=0;const off=offs[r+','+c];if(off){dx=off.dx;dy=off.dy;}
    const x=GAP+c*pitch+cell/2+dx,y=GAP+r*pitch+cell/2-o.fall+dy;
    let sc=1,al=1;
    if(o.popT>0){const t=o.popT/POP_DUR;sc=1+.3*Math.sin(Math.min(t*1.4,1)*1.57);al=t<.35?1:1-(t-.35)/.65;}
    if(o.bornAt&&timeSec-o.bornAt<.3)sc*=.5+.5*((timeSec-o.bornAt)/.3);
    drawGem(x,y,sc,al,o);}
  if(state==='playing'||state==='paused'){
    if(hover&&!(selected&&selected.r===hover.r&&selected.c===hover.c))strokeCell(hover.r,hover.c,'rgba(255,255,255,.26)',1.5);
    if(selected){const p=.6+.4*Math.sin(timeSec*7);strokeCell(selected.r,selected.c,'rgba(255,176,32,'+(0.5+0.5*p)+')',3);}
    strokeCursor(cur.r,cur.c);}
  // ambient shine sweep
  const sx=((timeSec*45)%(boardPx+220))-110;
  const sg=ctx.createLinearGradient(sx,0,sx+110,0);
  sg.addColorStop(0,'rgba(255,255,255,0)');sg.addColorStop(.5,'rgba(255,255,255,.045)');sg.addColorStop(1,'rgba(255,255,255,0)');
  ctx.fillStyle=sg;ctx.fillRect(0,0,boardPx,boardPx);
  // particles
  for(const p of particles){ctx.globalAlpha=clamp(p.life/p.max,0,1);
    if(p.add)ctx.globalCompositeOperation='lighter';
    ctx.fillStyle=p.col;ctx.beginPath();ctx.arc(p.x,p.y,Math.max(.4,p.size*(p.life/p.max)),0,6.283);ctx.fill();
    ctx.globalCompositeOperation='source-over';}
  ctx.globalAlpha=1;
  for(const rg of rings){const t=rg.t/rg.dur,e=1-Math.pow(1-t,3);
    ctx.globalAlpha=(1-t)*.9;ctx.strokeStyle=rg.col;ctx.lineWidth=2+3*(1-t);
    ctx.beginPath();ctx.arc(rg.x,rg.y,Math.max(.1,rg.maxR*e),0,6.283);ctx.stroke();}
  ctx.globalAlpha=1;
  if(flashT>0){ctx.fillStyle='rgba(255,64,72,'+(flashT*.28)+')';ctx.fillRect(0,0,boardPx,boardPx);}
}

/* ================= update ================= */
function updateGame(dt){
  if(anim){
    anim.t+=dt;const dur=anim.valid?SWAP_T:SWAP_T*2;
    if(anim.t>=dur){
      if(anim.valid){combo=1;
        const A=grid[anim.ra][anim.ca],B=grid[anim.rb][anim.cb];
        ['gem','special'].forEach(k=>{const v=A[k];A[k]=B[k];B[k]=v;});
        beginClear(anim.runs);
      }else{phase='idle';sfx.bad();setMsg('No match','dim');if(!hasMoves())reshuffle();}
      anim=null;
    }
  }else if(phase==='popping'){
    let done=true;for(const o of popping){o.popT+=dt;if(o.popT<POP_DUR)done=false;}
    if(done)commitClear();
  }else if(phase==='falling'){
    let all=true;
    for(let r=0;r<N;r++)for(let c=0;c<N;c++){const o=grid[r][c];if(o.fall<=0)continue;all=false;
      o.vy+=G_FALL*dt;const f=o.fall-o.vy*dt;
      if(f<=0){o.fall=0;dustFx(GAP+c*pitch+cell/2,GAP+r*pitch+cell,false);
        if(timeSec-lastLandSfx>.06){sfx.land();lastLandSfx=timeSec;}}
      else o.fall=f;}
    if(all)finishResolve();
  }
}

}

function tapSelect(r,c){
  if(phase!=='idle')return;
  if(selected&&selected.r===r&&selected.c===c){selected=null;return;}
  if(selected&&Math.abs(selected.r-r)+Math.abs(selected.c-c)===1){attemptSwap(selected.r,selected.c,r,c);return;}
  selected={r,c};sfx.select();
}

  if(state!=='playing'||phase!=='idle')return;
  const p=cellAt(e);if(!p)return;
  drag={r:p.r,c:p.c,x:e.clientX,y:e.clientY,moved:false};
  try{cv.setPointerCapture(e.pointerId);}catch(err){}});

  if(mv){e.preventDefault();
    cur.r=clamp(cur.r+mv[0],0,N-1);cur.c=clamp(cur.c+mv[1],0,N-1);

      if(selected.r===cur.r&&selected.c===cur.c)selected=null;

      else{selected={r:cur.r,c:cur.c};sfx.select();}
    }else{selected={r:cur.r,c:cur.c};sfx.select();}
  }
});
