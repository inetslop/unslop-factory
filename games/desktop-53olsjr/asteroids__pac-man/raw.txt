function wrapVec(ax,ay,bx,by){ // shortest wrapped delta from a to b
  let dx=bx-ax, dy=by-ay;
  if(dx> W/2) dx-=W; else if(dx<-W/2) dx+=W;
  if(dy> H/2) dy-=H; else if(dy<-H/2) dy+=H;
  return [dx,dy];
}
function wrapPos(o){ if(o.x<0)o.x+=W; else if(o.x>=W)o.x-=W; if(o.y<0)o.y+=H; else if(o.y>=H)o.y-=H; }

      g.gain.linearRampToValueAtTime(0,ac.currentTime+0.1); s.stop(ac.currentTime+0.16);
    }
  }
  return {
    init:ensure,
    setMuted(m){ if(master) master.gain.value=m?0:0.5; },
    setThrust,
    shoot(){ tone('square',920,280,0.08,0.06); },
    pellet(){ chompAlt=!chompAlt; tone('triangle',chompAlt?392:540,chompAlt?300:420,0.055,0.055); },
    power(){ [523,659,784,1047].forEach((f,i)=>tone('square',f,f,0.09,0.06,i*0.05)); },
    rock(size){ noiseHit(0.16+size*0.05, 420+size*220, 0.22,'lowpass'); tone('triangle',170-size*30,40,0.24,0.16); },
    ghost(){ [523,784,1047,1568].forEach((f,i)=>tone('square',f,f,0.07,0.06,i*0.04)); },
    death(){ tone('sawtooth',320,40,0.7,0.2); noiseHit(0.5,300,0.28,'lowpass'); },
    wave(){ [392,523,659].forEach((f,i)=>tone('triangle',f,f,0.14,0.08,i*0.07)); },
    gameOver(){ tone('sawtooth',200,55,1.1,0.14); }
  };
})();

/* ============================== ENTITIES ============================== */
function wrapPos(o){ if(o.x<0)o.x+=W; else if(o.x>=W)o.x-=W; if(o.y<0)o.y+=H; else if(o.y>=H)o.y-=H; }
function wrapVec(ax,ay,bx,by){ let dx=bx-ax,dy=by-ay;
  if(dx>W/2)dx-=W; else if(dx<-W/2)dx+=W;
  if(dy>H/2)dy-=H; else if(dy<-H/2)dy+=H; return [dx,dy]; }
function wrapOffs(x,y){ const o=[[0,0]];
  if(x<70)o.push([W,0]); else if(x>W-70)o.push([-W,0]);
  if(y<70)o.push([0,H]); else if(y>H-70)o.push([0,-H]); return o; }

function makeAst(x,y,size,mul){
  mul=mul||1; const R=[0,15,27,48][size];
  const pts=[], n=irand(8,11); for(let i=0;i<n;i++) pts.push(rand(.68,1.22));
  const a=rand(0,TAU), sp=rand(36,70)*[1,1.4,1.9][size-1]*mul;
  return {x,y,vx:Math.cos(a)*sp,vy:Math.sin(a)*sp,r:R,rot:rand(0,TAU),vr:rand(-1.2,1.2)*(3/size),pts,size,born:0};
}
function spawnRock(size){
  if(asteroids.length>=14) return;
  let x,y,t=0;
  do{ x=rand(0,W); y=rand(0,H); t++; }while(player && Math.hypot(x-player.x,y-player.y)<260 && t<40);
  asteroids.push(makeAst(x,y,size,1+wave*0.06));
}
const GHOST_DEFS=[{c:'#ff5f7a'},{c:'#ffa94d'},{c:'#b06bff'},{c:'#6ad7ff'}];
function spawnGhost(){
  let x,y,t=0;
  do{ x=rand(40,W-40); y=rand(60,H-60); t++;
      const [dx,dy]=wrapVec(x,y,player?player.x:W/2, player?player.y:H/2);
  }while(t<50 && dx*dx+dy*dy<300*300);
  ghosts.push({x,y,h:rand(0,TAU),sp:0,def:GHOST_DEFS[ghostSeq++%4],born:0,wob:rand(0,TAU)});
}
function spawnPellets(){

  for(let gy=sp;gy<H-sp/2;gy+=sp) for(let gx=sp;gx<W-sp/2;gx+=sp){

  }
  const pool=pellets.slice();
  for(let i=0;i<4&&pool.length;i++){ pool[irand(0,pool.length-1)].power=true; }

}

function tryFire(){
  if(fireCd>0||lasers.length>=4) return;
  fireCd=0.16; const p=player, dx=Math.cos(p.a), dy=Math.sin(p.a);
  lasers.push({x:p.x+dx*18,y:p.y+dy*18,vx:p.vx+dx*560,vy:p.vy+dy*560,life:0.9});
  Sfx.shoot();
}
function playerPhysics(dt){
  const p=player;
  const rot=((keys.ArrowRight||keys.KeyD)?1:0)-((keys.ArrowLeft||keys.KeyA)?1:0);
  p.a+=rot*3.8*dt;
  let thrusting=false;
  if(touch.on){ const [dx,dy]=wrapVec(p.x,p.y,touch.x,touch.y);
    if(dx*dx+dy*dy>20*20){ p.a=Math.atan2(dy,dx); thrusting=true; } }
  else if(keys.ArrowUp||keys.KeyW) thrusting=true;
  p.thrust=thrusting;
  if(thrusting){
    p.vx+=Math.cos(p.a)*340*dt; p.vy+=Math.sin(p.a)*340*dt;
    if(Math.random()<0.9) parts.push({x:p.x-Math.cos(p.a)*16,y:p.y-Math.sin(p.a)*16,
      vx:-Math.cos(p.a)*rand(80,160)+rand(-25,25),vy:-Math.sin(p.a)*rand(80,160)+rand(-25,25),
      t:0,life:rand(.15,.32),col:'#7ad8ff',r:rand(1.5,3)});
  }
  Sfx.setThrust(thrusting && state==='play');
  const dmp=Math.exp(-0.55*dt); p.vx*=dmp; p.vy*=dmp;
  const sp=Math.hypot(p.vx,p.vy), max=330+wave*4;
  if(sp>max){ p.vx*=max/sp; p.vy*=max/sp; }
  p.x+=p.vx*dt; p.y+=p.vy*dt; wrapPos(p);
  p.mouth+=dt*(sp>30?9:3.5);
  if(p.inv>0) p.inv-=dt;
}
function updRocks(dt){
  for(const a of asteroids){ a.x+=a.vx*dt; a.y+=a.vy*dt; a.rot+=a.vr*dt; a.born=Math.min(1,a.born+dt*3); wrapPos(a); }
}
function updGhosts(dt,chase){
  const p=player;
  for(const g of ghosts){
    g.born=Math.min(1,g.born+dt*2); g.wob+=dt*6;
    let des;
    if(chase && p){ const [dx,dy]=wrapVec(g.x,g.y,p.x,p.y); des=Math.atan2(dy,dx); if(fright>0) des+=Math.PI; }
    else { des=g.h+Math.sin(time*0.7+g.wob)*1.4; }
    let diff=des-g.h; while(diff>Math.PI)diff-=TAU; while(diff<-Math.PI)diff+=TAU;
    const tr=(fright>0&&chase?3.4:2.1)*dt; g.h+=clamp(diff,-tr,tr);
    const target=(fright>0&&chase)?95:(chase?150+wave*6:70);
    g.sp+=(target-g.sp)*Math.min(1,3*dt);
    g.x+=Math.cos(g.h)*g.sp*dt; g.y+=Math.sin(g.h)*g.sp*dt; wrapPos(g);
  }
}
function updLasers(dt){
  for(let i=lasers.length-1;i>=0;i--){ const l=lasers[i];
    l.x+=l.vx*dt; l.y+=l.vy*dt; l.life-=dt; wrapPos(l);
    if(l.life<=0) lasers.splice(i,1); }
}
function updParts(dt){
  for(let i=parts.length-1;i>=0;i--){ const p=parts[i]; p.t+=dt;
    if(p.t>=p.life){ parts.splice(i,1); continue; }
    if(!p.ring){ p.x+=p.vx*dt; p.y+=p.vy*dt; const d=Math.exp(-1.6*dt); p.vx*=d; p.vy*=d; } }
  for(let i=pops.length-1;i>=0;i--){ pops[i].t+=dt; if(pops[i].t>0.9) pops.splice(i,1); }
}
function eatGhost(g,byLaser){
  const pts=100*ghostCombo; ghostCombo=Math.min(ghostCombo*2,8);
  score+=pts; pop(g.x,g.y,'+'+pts, g.def.c);
  burst(g.x,g.y,30,'#ffffff',240,0.6,3); burst(g.x,g.y,20,g.def.c,180,0.7,3);
  parts.push({ring:1,x:g.x,y:g.y,t:0,life:.4,col:g.def.c});
  const i=ghosts.indexOf(g); if(i>=0) ghosts.splice(i,1);
  Sfx.ghost(); shake=Math.min(shake+6,18); updateHUD();
}
function breakRock(a){
  const pts=[0,5,10,20][a.size]; score+=pts; pop(a.x,a.y,'+'+pts,'#cfe4ff');
  burst(a.x,a.y,8+a.size*6,'#bcd9f5',120+a.size*45,0.6,3);
  parts.push({ring:1,x:a.x,y:a.y,t:0,life:.35,col:'#9fd4ff'});
  shake=Math.min(shake+a.size*3,22); Sfx.rock(a.size); updateHUD();
  if(a.size>1) for(let i=0;i<2;i++){
    const b=makeAst(a.x,a.y,a.size-1,1+wave*0.06), ca=rand(0,TAU), cs=rand(50,110);
    b.vx=a.vx+Math.cos(ca)*cs; b.vy=a.vy+Math.sin(ca)*cs; asteroids.push(b);
  }
}
function collisions(){
  const p=player;
  /* pellets */
  for(const pl of pellets){ if(pl.eaten) continue;
    const rr=(pl.power?22:18);
    if(Math.hypot(pl.x-p.x,pl.y-p.y)<rr){
      pl.eaten=true; dotsLeft--; score+=pl.power?50:10; Sfx.pellet();
      burst(pl.x,pl.y,pl.power?14:3, pl.power?'#ff7fe2':'#bfe8ff', pl.power?200:90, pl.power?0.6:0.3, 2.5);
      if(pl.power){ fright=frightMax; ghostCombo=1; pop(p.x,p.y-24,'POWER!','#ff7fe2'); Sfx.power(); }
      if(dotsLeft<=0) nextWave();
    }
  }
  /* lasers vs rocks */
  for(let i=lasers.length-1;i>=0;i--){ const l=lasers[i]; let hit=false;
    for(let j=asteroids.length-1;j>=0;j--){ const a=asteroids[j];
      if(a.born<0.4) continue;
      if(Math.hypot(l.x-a.x,l.y-a.y)<a.r*0.85+3){ asteroids.splice(j,1); breakRock(a); hit=true; break; } }
    if(hit){ lasers.splice(i,1); continue; }
    /* lasers vs ghosts (only when powered) */
    if(fright>0) for(let j=ghosts.length-1;j>=0;j--){ const g=ghosts[j];
      if(g.born<0.6) continue;
      if(Math.hypot(l.x-g.x,l.y-g.y)<g.r+4){ eatGhost(g,true); lasers.splice(i,1); break; } }
  }
  if(state!=='play') return; // may have just advanced a wave etc.
  /* player vs rocks */
  if(p.inv<=0) for(const a of asteroids){
    if(a.born<0.4) continue;
    if(Math.hypot(a.x-p.x,a.y-p.y)<a.r*0.82+p.r*0.7){ die(); return; } }
  /* player vs ghosts */
  if(p.inv<=0) for(let i=ghosts.length-1;i>=0;i--){ const g=ghosts[i];
    if(Math.hypot(g.x-p.x,g.y-p.y)<g.r+p.r*0.8){
      if(fright>0) eatGhost(g,false); else { die(); return; } } }
}
function menuAmbient(dt){ updRocks(dt); updGhosts(dt,false); updParts(dt); }

function update(dt){
  time+=dt;
  shake*=Math.exp(-5*dt); if(shake<0.2)shake=0;
  flash=Math.max(0,flash-2.2*dt); bannerT-=dt; fireCd-=dt;
  updParts(dt);
  if(state==='dying'){
    updRocks(dt); updGhosts(dt,false); updLasers(dt);
    freezeT-=dt;
    if(freezeT<=0){ if(lives>0) respawn(); else gameOver(); }
    return;
  }
  if(state!=='play'){ menuAmbient(dt);
    starOff.x-=8*dt; starOff.y+=3*dt; return; }
  /* play */
  playerPhysics(dt);
  if(keys.Space||touch.on) tryFire();
  updLasers(dt); updRocks(dt); updGhosts(dt,true); collisions();
  if(fright>0){ fright-=dt; if(fright<=0) ghostCombo=1; }
  const p=player; starOff.x-=p.vx*0.05*dt; starOff.y-=p.vy*0.05*dt;
}

}
function drawRocks(){

      ctx.save(); ctx.translate(a.x+ox,a.y+oy); ctx.rotate(a.rot); ctx.globalAlpha=a.born;
      ctx.beginPath(); const n=a.pts.length;
      for(let i=0;i<n;i++){ const ang=i/n*TAU, rr=a.r*a.pts[i];
        const px=Math.cos(ang)*rr, py=Math.sin(ang)*rr; i?ctx.lineTo(px,py):ctx.moveTo(px,py); }
      ctx.closePath();
      ctx.fillStyle='rgba(52,74,110,0.30)'; ctx.fill();

      ctx.restore();

  }
  ctx.restore();
}
function drawGhosts(){
  for(const g of ghosts){
    for(const [ox,oy] of wrapOffs(g.x,g.y)){
      const fr=fright>0 && state==='play';
      const blink=fr && fright<1.6 && Math.floor(time*8)%2===0;
      ctx.save(); ctx.translate(g.x+ox,g.y+oy+Math.sin(g.wob)*2); ctx.rotate(g.h);
      ctx.globalAlpha=g.born;
      const r=13;
      ctx.beginPath(); ctx.arc(0,0,r,Math.PI,0,false);
      let x=r; const steps=4, sw=(2*r)/steps;
      for(let i=0;i<steps;i++){ const nx=x-sw;
        const dip=r*0.55+Math.sin(g.wob*1.6+i*1.9)*2.2;
        ctx.quadraticCurveTo((x+nx)/2,dip,nx,0); x=nx; }
      ctx.closePath();
      const body = blink ? '#eef2ff' : (fr ? '#3040b8' : g.def.c);
      ctx.shadowColor=fr?'#4a5cff':g.def.c; ctx.shadowBlur=14;
      ctx.fillStyle=body; ctx.fill(); ctx.shadowBlur=0;
      if(fr){
        ctx.strokeStyle=blink?'#d21f3c':'#ffffff'; ctx.lineWidth=2;
        ctx.beginPath(); ctx.moveTo(-5,3); ctx.lineTo(-2.5,6); ctx.lineTo(0,3); ctx.lineTo(2.5,6); ctx.lineTo(5,3); ctx.stroke();
        ctx.fillStyle=blink?'#d21f3c':'#ffffff'; circle(3,-4.6,2.6); circle(3,4.6,2.6);
      } else {
        ctx.fillStyle='#fff'; circle(3,-4.6,3.4); circle(3,4.6,3.4);
        ctx.fillStyle='#0a1030'; circle(4.8,-4.6,1.7); circle(4.8,4.6,1.7);
      }
      ctx.restore();
    }
  }
}
function drawPlayer(){
  const p=player; if(!p) return;
  for(const [ox,oy] of wrapOffs(p.x,p.y)){
    ctx.save(); ctx.translate(p.x+ox,p.y+oy); ctx.rotate(p.a);
    if(p.inv>0 && Math.floor(time*12)%2===0) ctx.globalAlpha=0.35;
    if(p.thrust){
      const fl=12+Math.random()*10;
      ctx.beginPath(); ctx.moveTo(-p.r*0.7,-5); ctx.lineTo(-p.r-fl,0); ctx.lineTo(-p.r*0.7,5); ctx.closePath();
      ctx.shadowColor='#7ad8ff'; ctx.shadowBlur=12; ctx.fillStyle='rgba(140,230,255,.9)'; ctx.fill(); ctx.shadowBlur=0;

    const open=Math.sin(p.mouth)*0.5+0.5, m=0.10+0.35*open;

    ctx.fillStyle='#ffe14d'; ctx.fill(); ctx.shadowBlur=0;

    ctx.restore();
  }
}
function render(){
  ctx.save();
  if(shake>0.3) ctx.translate(rand(-shake,shake),rand(-shake,shake));
  const dx=Math.sin(time*0.05)*12, dy=Math.cos(time*0.043)*10;
  if(neb) ctx.drawImage(neb,-80+dx,-80+dy);
  drawStars();

  if(state!=='menu' && pellets.length){
    for(const pl of pellets){ if(pl.eaten) continue;
      if(pl.power){ const s=5.5+Math.sin(time*6+pl.ph)*1.8;
        ctx.shadowColor='#ff4fd8'; ctx.shadowBlur=16; ctx.fillStyle='#ff7fe2'; circle(pl.x,pl.y,s); ctx.shadowBlur=0;
      } else { ctx.fillStyle='rgba(205,232,255,.92)'; circle(pl.x,pl.y,2.3+Math.sin(time*3+pl.ph)*0.4); } }
  }
  drawRocks(); drawGhosts();

  ctx.save(); ctx.globalCompositeOperation='lighter';
  for(const l of lasers){
    ctx.strokeStyle='#9df3ff'; ctx.lineWidth=3; ctx.lineCap='round';
    ctx.shadowColor='#6ad7ff'; ctx.shadowBlur=10;
    ctx.beginPath(); ctx.moveTo(l.x,l.y); ctx.lineTo(l.x-l.vx*0.022,l.y-l.vy*0.022); ctx.stroke();
  }
  for(const p of parts){
    const t=p.t/p.life;
    if(p.ring){ ctx.globalAlpha=(1-t)*0.8; ctx.strokeStyle=p.col; ctx.lineWidth=3*(1-t)+1;
      ctx.beginPath(); ctx.arc(p.x,p.y,6+t*74,0,TAU); ctx.stroke(); continue; }
    ctx.globalAlpha=1-t; ctx.fillStyle=p.col; circle(p.x,p.y,p.r*(1-t*0.5));
  }
  ctx.restore();

  drawPlayer();

  if(pops.length){
    ctx.save(); ctx.textAlign='center';
    ctx.font='700 15px ui-monospace, Menlo, Consolas, monospace';
    for(const p of pops){ ctx.globalAlpha=1-p.t/0.9; ctx.fillStyle=p.col; ctx.fillText(p.txt,p.x,p.y-p.t*34); }
    ctx.restore();
  }

  if(bannerT>0){
    const a=clamp(Math.min((BANNER_DUR-bannerT)*4, bannerT*1.6),0,1);
    ctx.save(); ctx.globalAlpha=a; ctx.textAlign='center';
    const fs=Math.min(72,W*0.09);
    ctx.font='400 '+fs+'px Impact, "Arial Narrow", sans-serif';
    if('letterSpacing' in ctx) ctx.letterSpacing='6px';
    ctx.shadowColor='#6ad7ff'; ctx.shadowBlur=24; ctx.fillStyle='#dff3ff';
    ctx.fillText(banner,W/2,H*0.3);
    if(bannerSub){ ctx.font='12px ui-monospace, Menlo, Consolas, monospace'; ctx.shadowBlur=8;
      ctx.fillStyle='#9fd4ff'; ctx.fillText(bannerSub.toUpperCase(),W/2,H*0.3+fs*0.55); }
    if('letterSpacing' in ctx) ctx.letterSpacing='0px';
    ctx.restore();
  }
  ctx.restore();

  if(flash>0){ ctx.fillStyle='rgba(255,120,120,'+(flash*0.35)+')'; ctx.fillRect(0,0,W,H); }
  if(state==='play'&&lives===1){
    const a=0.05+0.05*Math.sin(time*4);
    const g=ctx.createRadialGradient(W/2,H/2,Math.min(W,H)*0.35,W/2,H/2,Math.max(W,H)*0.7);
    g.addColorStop(0,'rgba(255,60,80,0)'); g.addColorStop(1,'rgba(255,60,80,'+a+')');
    ctx.fillStyle=g; ctx.fillRect(0,0,W,H);
  }
}
