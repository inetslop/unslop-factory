  const mk=(n,r,l)=>{for(let i=0;i<n;i++)stars.push({x:R(0,W),y:R(0,H),r,ph:R(0,TAU),tw:R(.5,2.2),l});};
  mk(Math.min(140,A/9000|0),1,0);
  mk(Math.min(70,A/18000|0),1.6,1);
  mk(Math.min(34,A/42000|0),2.3,2);
}
function updateStars(dt){
  const tv=(G.state==='playing'&&G.ship)?{x:-G.ship.vx*.05,y:-G.ship.vy*.05}:{x:0,y:0};
  par.x+=(tv.x-par.x)*Math.min(1,dt*2);par.y+=(tv.y-par.y)*Math.min(1,dt*2);
  const f=[.3,.7,1.4],v=[3,6,11];
  for(const s of stars){
    s.x+=(-v[s.l]+par.x*f[s.l])*dt;
    s.y+=(v[s.l]*.5+par.y*f[s.l])*dt;
    if(s.x<-3)s.x+=W+6;if(s.x>W+3)s.x-=W+6;
    if(s.y<-3)s.y+=H+6;if(s.y>H+3)s.y-=H+6;
  }
}
function updateComets(dt){
  cometT-=dt;
  if(cometT<=0){cometT=R(4,9);
    comets.push({x:R(0,W),y:-24,vx:R(-80,80)*S,vy:R(260,430)*S});}
  for(let i=comets.length-1;i>=0;i--){const c=comets[i];c.x+=c.vx*dt;c.y+=c.vy*dt;
    if(c.y>H+70||c.x<-90||c.x>W+90)comets.splice(i,1);}
}

/* ============================== audio ============================== */
let AC=null,master=null,nbuf=null,engCur=false,warble=null;
let muted=load('mute','0')==='1';
function resumeAudio(){
  if(!AC){try{
    AC=new (window.AudioContext||window.webkitAudioContext)();
    master=AC.createGain();master.gain.value=muted?0:.55;master.connect(AC.destination);
  }catch(e){}}
  if(AC&&AC.state==='suspended')AC.resume();
  applyMute();
}
function applyMute(){if(master)master.gain.value=muted?0:.55;
  elMute.textContent=muted?'SOUND OFF':'SOUND ON';elMute.classList.toggle('off',muted);}
function toggleMute(){muted=!muted;save('mute',muted?'1':'0');applyMute();}
function noiseBuf(){if(nbuf)return nbuf;const len=(AC.sampleRate*1.2)|0;
  nbuf=AC.createBuffer(1,len,AC.sampleRate);const ch=nbuf.getChannelData(0);
  for(let i=0;i<len;i++)ch[i]=Math.random()*2-1;return nbuf;}

function setEngine(on){
  if(on===engCur)return;engCur=on;
  if(!AC||!master)return;
  if(!window.__eng){
    const g=AC.createGain();g.gain.value=0;
    const f=AC.createBiquadFilter();f.type='lowpass';f.frequency.value=230;f.Q.value=6;
    const src=AC.createBufferSource();src.buffer=noiseBuf();src.loop=true;
    src.connect(f);f.connect(g);g.connect(master);src.start();
    const o=AC.createOscillator();o.type='sawtooth';o.frequency.value=46;
    const og=AC.createGain();og.gain.value=.5;o.connect(og);og.connect(f);o.start();
    window.__eng={g,f};
  }
  const e=window.__eng,t=AC.currentTime;
  e.g.gain.cancelScheduledValues(t);
  e.g.gain.setTargetAtTime(on?.16:0,t,on?.03:.09);
  e.f.frequency.setTargetAtTime(on?430:220,t,.05);
}
function sfxLaser(){if(!AC)return;const t=AC.currentTime;
  const o=AC.createOscillator();o.type='square';
  o.frequency.setValueAtTime(1000,t);o.frequency.exponentialRampToValueAtTime(140,t+.12);
  const g=AC.createGain();g.gain.setValueAtTime(.16,t);g.gain.exponentialRampToValueAtTime(.001,t+.14);
  o.connect(g);g.connect(master);o.start(t);o.stop(t+.16);}
function sfxBoom(sz){if(!AC)return;const t=AC.currentTime,dur=.35+.75*sz;
  const src=AC.createBufferSource();src.buffer=noiseBuf();
  const f=AC.createBiquadFilter();f.type='lowpass';
  f.frequency.setValueAtTime(200+1800*(1-sz*.6),t);f.frequency.exponentialRampToValueAtTime(60,t+dur);
  const g=AC.createGain();g.gain.setValueAtTime(.45+.4*sz,t);g.gain.exponentialRampToValueAtTime(.001,t+dur);
  src.connect(f);f.connect(g);g.connect(master);src.start(t);src.stop(t+dur+.05);
  const o=AC.createOscillator();o.type='sine';
  o.frequency.setValueAtTime(110,t);o.frequency.exponentialRampToValueAtTime(32,t+dur*.7);
  const og=AC.createGain();og.gain.setValueAtTime(.45,t);og.gain.exponentialRampToValueAtTime(.001,t+dur*.8);
  o.connect(og);og.connect(master);o.start(t);o.stop(t+dur);}
function warbleStart(){if(!AC||warble)return;const t=AC.currentTime;
  const o=AC.createOscillator();o.type='triangle';o.frequency.value=520;
  const lfo=AC.createOscillator();lfo.type='square';lfo.frequency.value=5.5;
  const lg=AC.createGain();lg.gain.value=140;lfo.connect(lg);lg.connect(o.frequency);
  const g=AC.createGain();g.gain.setValueAtTime(0,t);g.gain.linearRampToValueAtTime(.05,t+.2);
  o.connect(g);g.connect(master);o.start(t);lfo.start(t);warble={o,lfo,g};}
function warbleStop(){if(!warble)return;const t=AC.currentTime,w=warble;warble=null;
  w.g.gain.setTargetAtTime(0,t,.1);w.o.stop(t+.4);w.lfo.stop(t+.4);}
function blip(f,dt,dur,type,gain){if(!AC)return;const t=AC.currentTime+dt;
  const o=AC.createOscillator();o.type=type||'sine';o.frequency.value=f;
  const g=AC.createGain();g.gain.setValueAtTime(.001,t);g.gain.linearRampToValueAtTime(gain||.14,t+.02);
  g.gain.exponentialRampToValueAtTime(.001,t+dur);o.connect(g);g.connect(master);o.start(t);o.stop(t+dur+.02);}
function sfxExtra(){blip(784,0,.13,'sine',.15);blip(1175,.11,.16,'sine',.15);}
function sfxWave(){[392,523,659].forEach((f,i)=>blip(f,i*.09,.14,'square',.11));}
function sfxOver(){if(!AC)return;const t=AC.currentTime;
  const o=AC.createOscillator();o.type='sawtooth';
  o.frequency.setValueAtTime(200,t);o.frequency.exponentialRampToValueAtTime(38,t+1.1);
  const g=AC.createGain();g.gain.setValueAtTime(.24,t);g.gain.exponentialRampToValueAtTime(.001,t+1.2);
  o.connect(g);g.connect(master);o.start(t);o.stop(t+1.25);}

/* ============================== entities ============================== */
const ROCKS=[{r:72,pts:20,n:3,col:'#ffb45e'},{r:38,pts:50,n:2,col:'#ff9e6b'},{r:19,pts:100,n:0,col:'#ffd08a'}];
function makeRock(sz,x,y,vx,vy){
  const base=ROCKS[sz],r=base.r*S*R(.92,1.08);
  const nv=RI(9,12),pts=[];
  for(let i=0;i<nv;i++){const a=i/nv*TAU;pts.push({a,m:r*R(.76,1.14)});}
  let crack=null;
  if(sz<2){const a=R(0,TAU);
    crack={x1:Math.cos(a)*r*.5,y1:Math.sin(a)*r*.5,x2:Math.cos(a+R(-.6,.6))*r*R(.35,.7),y2:Math.sin(a+R(-.6,.6))*r*R(.35,.7)};}
  return {sz,x,y,vx,vy,r,cr:r*.86,a:R(0,TAU),va:R(-1.3,1.3),pts,col:base.col,ptsV:base.pts,n:base.n,crack};
}
function spawnWaveRock(){
  const r=ROCKS[0].r*S,side=RI(0,3);let x,y;
  if(side===0){x=-r-20;y=R(0,H);}else if(side===1){x=W+r+20;y=R(0,H);}
  else if(side===2){y=-r-20;x=R(0,W);}else{y=H+r+20;x=R(0,W);}
  const tx=R(W*.3,W*.7),ty=R(H*.3,H*.7);
  const ang=Math.atan2(ty-y,tx-x)+R(-.3,.3);
  const sp=R(34,70)*S*Math.min(1.6,1+G.wave*.05);
  G.rocks.push(makeRock(0,x,y,Math.cos(ang)*sp,Math.sin(ang)*sp));
}
function seedAmbient(){
  G.rocks.length=0;
  for(let i=0;i<4;i++){const sz=RI(0,2);
    G.rocks.push(makeRock(sz,R(0,W),R(0,H),R(-30,30)*S,R(-30,30)*S));}
}
function spawnShip(){
  let x=W/2,y=H/2;
  for(let t=0;t<14;t++){
    const ok=G.rocks.every(r=>((r.x-x)**2+(r.y-y)**2)>(r.r+95*S)**2);
    if(ok||t===13)break;
    x=R(W*.2,W*.8);y=R(H*.25,H*.75);
  }
  G.ship={x,y,vx:0,vy:0,a:-Math.PI/2,rv:0,cd:0,inv:2.4,thr:false};
}
function wrapE(o,m){if(o.x<-m)o.x+=W+2*m;if(o.x>W+m)o.x-=W+2*m;if(o.y<-m)o.y+=H+2*m;if(o.y>H+m)o.y-=H+2*m;}

/* ============================== fx ============================== */
function spawnParts(x,y,n,spd,cols,dir){
  for(let i=0;i<n;i++){
    if(G.parts.length>480)G.parts.shift();
    let a;if(dir)a=Math.atan2(dir[1],dir[0])+R(-.7,.7);else a=R(0,TAU);
    const s=R(.3,1)*spd,lf=R(.25,.6);
    G.parts.push({x,y,vx:Math.cos(a)*s,vy:Math.sin(a)*s,life:lf,max:lf,r:R(1,2.6)*S,col:cols[RI(0,cols.length-1)]});
  }
}
function addRing(x,y,col,vr){G.rings.push({x,y,r:4*S,vr:vr||380*S,life:.5,max:.5,col});}
function explosion(x,y,sz,col){
  spawnParts(x,y,Math.round(10+24*sz),(70+170*sz)*S,[col,'#fff3d6','#ffffff']);
  addRing(x,y,col);
  G.shake=Math.min(1.4,G.shake+.12+.32*sz);
  sfxBoom(sz);
}
function updateFx(dt){
  for(let i=G.parts.length-1;i>=0;i--){const p=G.parts[i];p.life-=dt;
    if(p.life<=0){G.parts.splice(i,1);continue;}
    p.x+=p.vx*dt;p.y+=p.vy*dt;const d=Math.exp(-2.2*dt);p.vx*=d;p.vy*=d;}
  for(let i=G.rings.length-1;i>=0;i--){const r=G.rings[i];r.life-=dt;
    if(r.life<=0)G.rings.splice(i,1);else{r.r+=r.vr*dt;r.vr*=Math.exp(-1.6*dt);}}
  for(let i=G.shards.length-1;i>=0;i--){const s=G.shards[i];s.life-=dt;
    if(s.life<=0){G.shards.splice(i,1);continue;}
    s.x+=s.vx*dt;s.y+=s.vy*dt;s.a+=s.va*dt;}
  for(let i=G.pops.length-1;i>=0;i--){const p=G.pops[i];p.life-=dt;p.y-=24*dt;
    if(p.life<=0)G.pops.splice(i,1);}
  G.shake*=Math.exp(-3.2*dt);if(G.shake<.01)G.shake=0;
  G.flash=Math.max(0,G.flash-dt*1.6);
}

/* ============================== gameplay ============================== */
function addScore(p,x,y,col){
  G.score+=p;
  if(x!=null)G.pops.push({x,y,txt:'+'+p,life:1.1,max:1.1,col:col||'#ffd9a0'});
  if(G.score>G.hi)G.hi=G.score;
  if(G.score>=G.nextExtra){
    G.nextExtra+=10000;if(G.lives<9)G.lives++;
    G.pops.push({x:x!=null?x:W/2,y:(y!=null?y:H*.5)-34*S,txt:'EXTRA SHIP',life:1.6,max:1.6,col:'#7df3ff'});
    sfxExtra();syncLives(true);
  }
}
function fire(){
  const sh=G.ship;sh.cd=.16;
  const nx=sh.x+Math.cos(sh.a)*16*S,ny=sh.y+Math.sin(sh.a)*16*S;
  G.bullets.push({x:nx,y:ny,vx:sh.vx+Math.cos(sh.a)*520*S,vy:sh.vy+Math.sin(sh.a)*520*S,life:1.1,r:3});
  spawnParts(nx,ny,4,90*S,['#ffe27a','#ffffff'],[Math.cos(sh.a),Math.sin(sh.a)]);
  sfxLaser();
}
function splitRock(j){
  const r=G.rocks[j];G.rocks.splice(j,1);
  addScore(r.ptsV,r.x,r.y,'#ffd9a0');
  explosion(r.x,r.y,[.3,.55,.85][r.sz],r.col);
  if(r.n>0){const cs=r.sz+1;
    for(let k=0;k<r.n;k++){const a=R(0,TAU),sp=R(50,130)*S*(1+r.sz*.2);
      G.rocks.push(makeRock(cs,r.x,r.y,r.vx*.4+Math.cos(a)*sp,r.vy*.4+Math.sin(a)*sp));}}
}
function shipDie(){
  const sh=G.ship;G.ship=null;setEngine(false);
  explosion(sh.x,sh.y,1,'#9fe8ff');
  const basePts=[{x:9*S,y:0},{x:-7*S,y:6*S},{x:-5*S,y:-6*S}];
  for(let i=0;i<3;i++){const a=R(0,TAU),sp=R(90,210)*S;
    G.shards.push({x:sh.x,y:sh.y,vx:sh.vx*.4+Math.cos(a)*sp,vy:sh.vy*.4+Math.sin(a)*sp,
      a:R(0,TAU),va:R(-5,5),life:1.5,max:1.5,pts:basePts});}
  G.flash=.7;G.shake=1;
  sfxBoom(1);
  G.lives--;syncLives(true);
  if(G.lives>0)G.respawnT=1.3;else G.overT=1.5;
}
function spawnUfo(){
  const small=Math.random()<Math.min(.65,.25+G.wave*.08);
  const r=(small?22:32)*S,dir=Math.random()<.5?1:-1;
  G.ufo={r,small,x:dir>0?-r-20:W+r+20,y:R(H*.2,H*.8),vx:dir*(small?R(70,95):R(50,70))*S,
    t:R(0,TAU),fireT:R(1.1,1.9),pts:small?1000:500};
  warbleStart();
}
function killUfo(score){
  const u=G.ufo;if(!u)return;G.ufo=null;
  if(score)addScore(u.pts,u.x,u.y,'#ff8fc2');
  explosion(u.x,u.y,.6,'#ff7ab8');warbleStop();
  G.ufoT=R(11,20);
}
function nextWave(){
  G.wave++;G.clearT=0;
  const n=Math.min(7,2+G.wave);
  for(let i=0;i<n;i++)spawnWaveRock();
  G.banner={t:2.4,text:'WAVE '+String(G.wave).padStart(2,'0'),sub:n+' ROCKS INBOUND'};
  if(!G.ufo)G.ufoT=R(9,15);
  sfxWave();
}

/* ============================== update ============================== */
function update(dt){
  G.time+=dt;
  updateStars(dt);updateComets(dt);updateFx(dt);

  if(G.state==='menu'||G.state==='gameover'){          // ambient drift
    for(const r of G.rocks){r.a+=r.va*dt;r.x+=r.vx*dt;r.y+=r.vy*dt;wrapE(r,r.r+10);}
    return;
  }
  if(G.state!=='playing')return;

  const sh=G.ship;
  if(!sh){
    if(G.respawnT>0){G.respawnT-=dt;if(G.respawnT<=0)spawnShip();}
    if(G.overT>0){G.overT-=dt;if(G.overT<=0){doGameOver();return;}}
  }
  const s=sh;
  if(s){
    let tv=0;const T=4.3;
    if(keys.ArrowLeft||keys.KeyA)tv-=T;
    if(keys.ArrowRight||keys.KeyD)tv+=T;
    s.rv+=(tv-s.rv)*Math.min(1,dt*13);
    s.a+=s.rv*dt;
    const thr=!!(keys.ArrowUp||keys.KeyW);
    s.thr=thr;
    if(thr){
      s.vx+=Math.cos(s.a)*400*S*dt;s.vy+=Math.sin(s.a)*400*S*dt;setEngine(true);
      spawnParts(s.x-Math.cos(s.a)*12*S,s.y-Math.sin(s.a)*12*S,1,R(60,120)*S,['#ff9b3d','#ffd76a'],[-Math.cos(s.a),-Math.sin(s.a)]);
    }else setEngine(false);
    const dmp=Math.exp(-.5*dt);s.vx*=dmp;s.vy*=dmp;
    const sp=Math.hypot(s.vx,s.vy),mx=480*S;
    if(sp>mx){s.vx*=mx/sp;s.vy*=mx/sp;}
    s.x+=s.vx*dt;s.y+=s.vy*dt;wrapE(s,16);
    s.cd-=dt;
    if(keys.Space&&s.cd<=0&&G.bullets.length<5)fire();
    if(s.inv>0)s.inv-=dt;
  }

  for(let i=G.bullets.length-1;i>=0;i--){const b=G.bullets[i];b.life-=dt;b.x+=b.vx*dt;b.y+=b.vy*dt;wrapE(b,8);
    if(b.life<=0)G.bullets.splice(i,1);}
  for(const r of G.rocks){r.a+=r.va*dt;r.x+=r.vx*dt;r.y+=r.vy*dt;wrapE(r,r.r+10);}
  for(let i=G.ufoBullets.length-1;i>=0;i--){const b=G.ufoBullets[i];b.life-=dt;b.x+=b.vx*dt;b.y+=b.vy*dt;wrapE(b,8);
    if(b.life<=0)G.ufoBullets.splice(i,1);}

  /* UFO */
  if(G.ufo){
    const u=G.ufo;u.t+=dt;u.x+=u.vx*dt;u.y+=Math.sin(u.t*1.7)*36*S*dt;
    if((u.vx>0&&u.x>W+u.r+30)||(u.vx<0&&u.x<-u.r-30)){warbleStop();G.ufo=null;G.ufoT=R(11,20);}
    else{
      u.fireT-=dt;
      if(u.fireT<=0&&G.ship){
        u.fireT=u.small?R(1.0,1.6):R(1.5,2.3);
        const ang=Math.atan2(G.ship.y-u.y,G.ship.x-u.x)+R(-1,1)*(u.small?.14:.5);
        G.ufoBullets.push({x:u.x,y:u.y+u.r*.5,vx:Math.cos(ang)*260*S,vy:Math.sin(ang)*260*S,life:3.2,r:3});
      }
    }
  }

  /* bullets vs everything */
  for(let i=G.bullets.length-1;i>=0;i--){const b=G.bullets[i];let dead=false;
    for(let j=G.ufoBullets.length-1;j>=0;j--){const ub=G.ufoBullets[j];
      const dx=b.x-ub.x,dy=b.y-ub.y;
      if(dx*dx+dy*dy<(7*S)*(7*S)){spawnParts((b.x+ub.x)/2,(b.y+ub.y)/2,6,90*S,['#ffe27a','#ff8fc2']);
        G.ufoBullets.splice(j,1);dead=true;break;}}
    if(!dead)for(let j=G.rocks.length-1;j>=0;j--){const r=G.rocks[j],rr=r.cr+b.r;
      const dx=b.x-r.x,dy=b.y-r.y;
      if(dx*dx+dy*dy<rr*rr){splitRock(j);dead=true;break;}}
    if(!dead&&G.ufo){const u=G.ufo,rr=u.r+4*S,dx=b.x-u.x,dy=b.y-u.y;
      if(dx*dx+dy*dy<rr*rr){killUfo(true);dead=true;}}
    if(dead)G.bullets.splice(i,1);
  }

  /* ship vs */
  const sh2=G.ship;
  if(sh2&&sh2.inv<=0){
    let dead=false;
    for(let j=G.rocks.length-1;j>=0&&!dead;j--){const r=G.rocks[j],rr=r.cr+11*S;
      const dx=sh2.x-r.x,dy=sh2.y-r.y;if(dx*dx+dy*dy<rr*rr)dead=true;}
    if(!dead&&G.ufo){const u=G.ufo,rr=u.r+10*S,dx=sh2.x-u.x,dy=sh2.y-u.y;
      if(dx*dx+dy*dy<rr*rr){killUfo(false);dead=true;}}
    if(!dead)for(let j=G.ufoBullets.length-1;j>=0;j--){const ub=G.ufoBullets[j],rr=ub.r+11*S;
      const dx=sh2.x-ub.x,dy=sh2.y-ub.y;
      if(dx*dx+dy*dy<rr*rr){G.ufoBullets.splice(j,1);dead=true;break;}}
    if(dead)shipDie();
  }

  /* rocks vs UFO */
  if(G.ufo){const u=G.ufo;
    for(let j=G.rocks.length-1;j>=0;j--){const r=G.rocks[j],rr=r.cr+u.r;
      const dx=u.x-r.x,dy=u.y-r.y;
      if(dx*dx+dy*dy<rr*rr){killUfo(false);splitRock(j);break;}}}

  /* wave + saucer timers */
  if(G.rocks.length===0){G.clearT+=dt;if(G.clearT>1.8)nextWave();}else G.clearT=0;
  if(!G.ufo&&G.wave>=1){G.ufoT-=dt;if(G.ufoT<=0)spawnUfo();}

  G.banner.t=Math.max(0,G.banner.t-dt);
  syncHUD();
}

/* ============================== render ============================== */
function render(){
  ctx.drawImage(neb,0,0,W,H);
  for(const s of stars){
    const tw=.55+.45*Math.sin(G.time*s.tw+s.ph);
    ctx.globalAlpha=(s.l===2?.9:s.l===1?.6:.38)*tw;
    ctx.fillStyle=s.l===2?'#dffbff':'#9fc4d8';
    ctx.fillRect(s.x,s.y,s.r,s.r);
  }
  ctx.globalAlpha=1;
  for(const c of comets){ctx.strokeStyle='rgba(190,240,255,.5)';ctx.lineWidth=1.5;
    ctx.beginPath();ctx.moveTo(c.x,c.y);ctx.lineTo(c.x-c.vx*.06,c.y-c.vy*.06);ctx.stroke();}

  const sx=(Math.random()-.5)*G.shake*10*S,sy=(Math.random()-.5)*G.shake*10*S;
  ctx.save();ctx.translate(sx,sy);

  for(const r of G.rocks){
    ctx.save();ctx.translate(r.x,r.y);ctx.rotate(r.a);
    ctx.beginPath();
    for(let i=0;i<r.pts.length;i++){const p=r.pts[i],x=Math.cos(p.a)*p.m,y=Math.sin(p.a)*p.m;
      i?ctx.lineTo(x,y):ctx.moveTo(x,y);}
    ctx.closePath();
    ctx.fillStyle='rgba(255,170,90,.05)';ctx.fill();
    ctx.strokeStyle=r.col;ctx.lineWidth=clamp(1.8*S,1.2,2.4);ctx.stroke();
    if(r.crack){ctx.globalAlpha=.35;ctx.beginPath();ctx.moveTo(r.crack.x1,r.crack.y1);
      ctx.lineTo(r.crack.x2,r.crack.y2);ctx.stroke();ctx.globalAlpha=1;}
    ctx.restore();
  }

  for(const b of G.ufoBullets){
    const pr=b.r+Math.sin(G.time*14)*.6;
    ctx.fillStyle='#ff7ab8';ctx.shadowColor='#ff5fa2';ctx.shadowBlur=9;
    ctx.beginPath();ctx.arc(b.x,b.y,pr,0,TAU);ctx.fill();ctx.shadowBlur=0;
  }
  for(const b of G.bullets){
    ctx.strokeStyle='rgba(255,226,122,.9)';ctx.lineWidth=3;
    ctx.shadowColor='#ffdf7a';ctx.shadowBlur=8;
    ctx.beginPath();ctx.moveTo(b.x-b.vx*.02,b.y-b.vy*.02);ctx.lineTo(b.x,b.y);ctx.stroke();
    ctx.shadowBlur=0;
  }

  const u=G.ufo;
  if(u){
    ctx.save();ctx.translate(u.x,u.y);
    ctx.strokeStyle='#ff6aa8';ctx.lineWidth=2;ctx.shadowColor='#ff5fa2';ctx.shadowBlur=12;
    ctx.beginPath();ctx.ellipse(0,0,u.r,u.r*.42,0,0,TAU);
    ctx.fillStyle='rgba(255,106,168,.08)';ctx.fill();ctx.stroke();
    ctx.beginPath();ctx.arc(0,-u.r*.32,u.r*.45,Math.PI,0);
    ctx.fillStyle='rgba(255,170,210,.12)';ctx.fill();ctx.stroke();
    ctx.shadowBlur=0;
    for(let i=-1;i<=1;i++){const on=Math.sin(G.time*8+i*2)>0;
      ctx.globalAlpha=on?.9:.25;ctx.fillStyle='#ffe3f1';
      ctx.beginPath();ctx.arc(i*u.r*.45,u.r*.16,2.2*S,0,TAU);ctx.fill();}
    ctx.restore();
  }

  const sh=G.ship;
  if(sh){
    let al=1;if(sh.inv>0)al=Math.sin(G.time*22)>.2?.95:.3;
    ctx.save();ctx.translate(sh.x,sh.y);
    if(sh.inv>0){ctx.globalAlpha=.28+.14*Math.sin(G.time*6);ctx.strokeStyle='#7df3ff';ctx.lineWidth=1.5;
      ctx.beginPath();ctx.arc(0,0,20*S,0,TAU);ctx.stroke();}
    ctx.rotate(sh.a);
    if(sh.thr){const fl=Math.random();
      ctx.save();ctx.globalCompositeOperation='lighter';
      ctx.fillStyle='rgba(255,'+(140+fl*60|0)+',60,.85)';
      ctx.beginPath();ctx.moveTo(-7*S,-4.5*S);ctx.lineTo((-16-fl*9)*S,0);ctx.lineTo(-7*S,4.5*S);ctx.closePath();ctx.fill();
      ctx.fillStyle='rgba(255,235,170,.9)';
      ctx.beginPath();ctx.moveTo(-7*S,-2.4*S);ctx.lineTo((-12-fl*5)*S,0);ctx.lineTo(-7*S,2.4*S);ctx.closePath();ctx.fill();
      ctx.restore();}
    ctx.globalAlpha=al;
    ctx.beginPath();ctx.moveTo(15*S,0);ctx.lineTo(-9*S,8*S);ctx.lineTo(-5*S,0);ctx.lineTo(-9*S,-8*S);ctx.closePath();
    ctx.fillStyle='rgba(120,230,255,.09)';ctx.fill();
    ctx.strokeStyle='#eafcff';ctx.lineWidth=2;ctx.shadowColor='#6ee7ff';ctx.shadowBlur=14;ctx.stroke();ctx.shadowBlur=0;
    ctx.restore();ctx.globalAlpha=1;
  }

  for(const s of G.shards){
    ctx.save();ctx.translate(s.x,s.y);ctx.rotate(s.a);
    ctx.globalAlpha=s.life/s.max;ctx.strokeStyle='#cfeeff';ctx.lineWidth=1.6;
    ctx.beginPath();s.pts.forEach((p,i)=>i?ctx.lineTo(p.x,p.y):ctx.moveTo(p.x,p.y));ctx.closePath();ctx.stroke();
    ctx.restore();
  }

  ctx.save();ctx.globalCompositeOperation='lighter';
  for(const p of G.parts){
    ctx.globalAlpha=Math.max(0,p.life/p.max)*.9;
    ctx.fillStyle=p.col;ctx.beginPath();ctx.arc(p.x,p.y,p.r,0,TAU);ctx.fill();
  }
  for(const r of G.rings){
    ctx.globalAlpha=Math.max(0,r.life/r.max)*.8;
    ctx.strokeStyle=r.col;ctx.lineWidth=2;
    ctx.beginPath();ctx.arc(r.x,r.y,r.r,0,TAU);ctx.stroke();
  }
  ctx.restore();ctx.globalAlpha=1;

  ctx.font='700 '+Math.round(14*Math.max(S,.9))+'px ui-monospace,Menlo,Consolas,monospace';
  ctx.textAlign='center';
  for(const p of G.pops){
    ctx.globalAlpha=clamp(p.life/p.max*1.4,0,1);
    ctx.fillStyle=p.col;ctx.shadowColor=p.col;ctx.shadowBlur=8;
    ctx.fillText(p.txt,p.x,p.y);ctx.shadowBlur=0;
  }
  ctx.globalAlpha=1;
  ctx.restore();

  if(G.banner.t>0&&G.state==='playing'){
    const b=G.banner,inA=clamp((2.4-b.t)/.35,0,1),outA=clamp(b.t/.6,0,1);
    const a=Math.min(inA,outA),sc=1+(1-inA)*.12;
    ctx.save();ctx.translate(W/2,H*.36);ctx.scale(sc,sc);
    ctx.font=Math.round(96*S)+'px Impact,"Arial Black",sans-serif';
    ctx.textAlign='center';ctx.lineWidth=2.5;
    ctx.strokeStyle='rgba(158,232,255,'+a+')';
    ctx.shadowColor='rgba(80,210,255,'+(a*.7)+')';ctx.shadowBlur=26;
    ctx.strokeText(b.text,0,0);
    if('letterSpacing' in ctx)ctx.letterSpacing='8px';
    ctx.font='700 '+Math.round(15*S)+'px ui-monospace,Menlo,Consolas,monospace';
    ctx.fillStyle='rgba(191,233,247,'+(a*.85)+')';ctx.shadowBlur=0;
    ctx.fillText(b.sub,0,34*S);
    if('letterSpacing' in ctx)ctx.letterSpacing='0px';
    ctx.restore();
  }

  if(G.flash>0){ctx.fillStyle='rgba(255,240,230,'+(G.flash*.4)+')';ctx.fillRect(0,0,W,H);}
}

function setState(st){
  G.state=st;document.body.dataset.state=st;
  menuOv.classList.toggle('hidden',st!=='menu');
  overOv.classList.toggle('hidden',st!=='gameover');

function startGame(){
  setState('playing');
  G.score=0;G.lives=3;G.wave=0;G.nextExtra=10000;G.startHi=G.hi;
  G.bullets.length=0;G.rocks.length=0;G.ufoBullets.length=0;G.shards.length=0;G.pops.length=0;
  G.ufo=null;warbleStop();setEngine(false);
  G.ship=null;G.respawnT=0;G.overT=-1;G.clearT=.9;G.banner.t=0;G.shake=0;G.flash=0;

}
