      s.flash=1; if(s.cd<=0 && vn<0){ addScore(25,(s.ax+s.bx)/2,(s.ay+s.by)/2); sfx.sling(); spark(b.x,b.y,'#ffd23c',6); shake=Math.max(shake,2); }
    }
  }

  for(const f of flippers){
    const a=f.base+f.ang, tx=f.px+f.len*Math.cos(a), ty=f.py+f.len*Math.sin(a);
    const q=closest(b.x,b.y,f.px,f.py,tx,ty); let dx=b.x-q[0],dy=b.y-q[1]; const d2=dx*dx+dy*dy, rr=BR+FR;
    if(d2<rr*rr){ const d=Math.sqrt(d2)||.0001; const nx=dx/d,ny=dy/d; b.x=q[0]+nx*rr; b.y=q[1]+ny*rr;
      const rcx=q[0]-f.px, rcy=q[1]-f.py, svx=-f.angVel*rcy, svy=f.angVel*rcx;
      let rvx=b.vx-svx, rvy=b.vy-svy, vn=rvx*nx+rvy*ny; if(vn<0){ const j=(1+.35)*vn; rvx-=j*nx; rvy-=j*ny; }
      b.vx=rvx+svx; b.vy=rvy+svy;
    }
  }

  for(const g of ghosts){
    const dx=b.x-g.cx, dy=b.y-g.cy, rr=BR+g.r, d2=dx*dx+dy*dy;
    if(d2<rr*rr){ const d=Math.sqrt(d2)||.0001; const nx=dx/d,ny=dy/d; b.x=g.cx+nx*rr; b.y=g.cy+ny*rr;
      let vn=b.vx*nx+b.vy*ny; if(vn<0){ b.vx-=(1+.9)*vn*nx; b.vy-=(1+.9)*vn*ny; }
      const outv=b.vx*nx+b.vy*ny; if(outv<BUMPER_MIN){ b.vx+=(BUMPER_MIN-outv)*nx; b.vy+=(BUMPER_MIN-outv)*ny; }
      g.flash=1; ring(b.x,b.y,g.color); shake=Math.max(shake,3);
      if(g.cd<=0){ g.cd=.2;
        if(fright>0 && g.eatCd<=0){ g.eatCd=.5; frightComboT=2; addScore(comboVal,b.x,b.y); comboVal=Math.min(1600,comboVal*2); sfx.ghostEat(); poof(g.cx,g.cy,'#ffffff'); shake=Math.max(shake,5); }
        else { addScore(g.score,b.x,b.y); sfx.bumper(); spark(b.x,b.y,g.color,7); }
      }
    }
  }

  let sp=Math.hypot(b.vx,b.vy); if(sp>MAXSPEED){ const k=MAXSPEED/sp; b.vx*=k; b.vy*=k; }
  if(b.x<16+BR){b.x=16+BR;b.vx=Math.abs(b.vx);} else if(b.x>464-BR){b.x=464-BR;b.vx=-Math.abs(b.vx);}

  eatDots();
  if(b.y-BR>704) loseBall();
}

function eatDots(){
  const b=ball;
  for(const d of dots){ if(d.eaten) continue; const dx=b.x-d.x,dy=b.y-d.y; if(dx*dx+dy*dy<(BR+5)*(BR+5)){ d.eaten=true; dotsLeft--; score+=10; if(score>high)high=score; sfx.waka(); spark(d.x,d.y,'#ffbf95',4); if(dotsLeft<=0) startLevelClear(); } }
  for(const p of pellets){ if(p.eaten) continue; const dx=b.x-p.x,dy=b.y-p.y; if(dx*dx+dy*dy<(BR+9)*(BR+9)){ p.eaten=true; fright=frightBase; comboVal=200; frightComboT=2; sfx.pellet(); poof(p.x,p.y,'#ffbf95'); showBanner('FIGHT MODE!','GHOSTS ARE VULNERABLE!',1.3); } }
}

/* ---- update ---- */
function update(dt){
  t+=dt;
  shake-=shake*8*dt; if(shake<.05)shake=0;
  nudgeCd=Math.max(0,nudgeCd-dt);
  for(const g of ghosts){ g.flash=Math.max(0,g.flash-dt*3); g.cd=Math.max(0,g.cd-dt); g.eatCd=Math.max(0,g.eatCd-dt); }
  for(const s of slings){ s.flash=Math.max(0,s.flash-dt*4); s.cd=Math.max(0,s.cd-dt); }
  if(banner.t>0) banner.t-=dt;

  for(let i=particles.length-1;i>=0;i--){ const p=particles[i]; p.t-=dt; if(p.t<=0){particles.splice(i,1);continue;} p.x+=p.vx*dt; p.y+=p.vy*dt; if(p.kind==='s'){p.vy+=600*dt;p.vx*=.98;} else {p.size+=220*dt;} }
  for(let i=popups.length-1;i>=0;i--){ const p=popups[i]; p.t-=dt; p.y-=34*dt; if(p.t<=0) popups.splice(i,1); }

  if(state!=='playing') return;
  if(pause) return;

  if(fright>0){ fright=Math.max(0,fright-dt); frightComboT-=dt; if(frightComboT<=0) comboVal=200; }
  updateFlippers(dt);

  const wantCharge = launchKeys.size>0 && canLaunch();
  if(wantCharge){
    chargePower+=chargeDir*1.5*dt;
    if(chargePower>=1){chargePower=1;chargeDir=-1;} else if(chargePower<=0){chargePower=0;chargeDir=1;}
    ball.y=(685+chargePower*12)-9;
  }

  if(ballDead){ ballDeadTimer-=dt; if(ballDeadTimer<=0){ lives--; if(lives<=0) gameOver(); else resetBall(); } return; }

  if(levelClearTimer>0){ levelClearTimer-=dt; if(levelClearTimer<=0){ level++; buildDots(level); frightBase=Math.max(3.5,6-(level-1)*.5); } }

  if(ball.launched){
    const n=Math.min(8,Math.max(1,Math.ceil(dt/(1/240)))), h=dt/n;
    for(let i=0;i<n && state==='playing' && !ballDead;i++) stepPhysics(h);
    if(ball.launched && !ballDead){
      const sp=Math.hypot(ball.vx,ball.vy);
      ball.mouth+=sp*dt*.05;
      if(sp>170){ trail.push({x:ball.x,y:ball.y,a:Math.atan2(ball.vy,ball.vx)}); if(trail.length>6)trail.shift(); }
    }
  } else { trail.length=0; }
}
