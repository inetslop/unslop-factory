function makeGhosts(n){
  const cols=['#ff6ec7','#4dd8ff','#ffa14d','#ff5252','#c98bff'];
  const types=['hunter','chaser','drifter','ambusher','chaos'];
  const smul=[1.03,1,-.06,.9,.78], arr=[];
  for(let i=0;i<n;i++){
    const s=SHRINES[i%5];
    arr.push({ si:i%5,x:s[0]*TILE+24,y:s[1]*TILE+24,vx:0,vy:0,r:15,color:cols[i],type:types[i],
      state:'cooldown',t:.8+i*.2,seed:Math.random()*9,wt:Math.random()*9,wAng:Math.random()*6.28,
      squash:0,sqNx:1,sqNy:0,chaosT:0,cx:CX,cy:CY,
      spd:Math.max(175,(248+level*9)*smul[i]*Math.min(1+(level-1)*.05,1.35)) });
  }
  return arr;
}
function genLevel(lvl){
  grid=genGrid(); mazeCv=buildMazeLayer();
  pellets=[]; pelGrid.fill(-1); pelletsLeft=0;
  for(let y=0;y<ROWS;y++)for(let x=0;x<COLS;x++){
    if(grid[y][x])continue;
    if(Math.abs(x-12)<=1&&Math.abs(y-7)<=1)continue;
    if(SHRINES.some(s=>s[0]===x&&s[1]===y))continue;
    const i=pellets.length;
    pellets.push({x:x*TILE+24,y:y*TILE+24,pow:false,dead:false});
    pelGrid[y*COLS+x]=i; pelletsLeft++;
  }
  [[2,2],[COLS-3,2],[2,ROWS-3],[COLS-3,ROWS-3]].forEach(t=>{
    let bi=-1,bd=1e9;
    for(let y=Math.max(0,t[1]-4);y<=Math.min(ROWS-1,t[1]+4);y++)
      for(let x=Math.max(0,t[0]-4);x<=Math.min(COLS-1,t[0]+4);x++){
        const i=pelGrid[y*COLS+x];
        if(i>=0){ const d=(x-t[0])**2+(y-t[1])**2; if(d<bd){bd=d;bi=i;} }
      }
    if(bi>=0)pellets[bi].pow=true;
  });
  ghosts=makeGhosts(state==='menu'?3:Math.min(2+lvl,5));
  P={x:CX,y:CY,vx:0,vy:0,r:15,face:0,invuln:state==='menu'?0:1.2,squash:0,sqNx:1,sqNy:0,lat:0,dieA:0};
  powerT=0; combo=1; trail.length=0; skids.length=0; popups.length=0;
  if(state==='menu')setupDemo();
}

/* ============================== PHYSICS ============================== */
function collideWalls(e,rest){
  let hit=null;
  const x0=Math.max(0,Math.floor((e.x-e.r)/TILE)),x1=Math.min(COLS-1,Math.floor((e.x+e.r)/TILE));
  const y0=Math.max(0,Math.floor((e.y-e.r)/TILE)),y1=Math.min(ROWS-1,Math.floor((e.y+e.r)/TILE));
  for(let ty=y0;ty<=y1;ty++)for(let tx=x0;tx<=x1;tx++){
    if(!grid[ty][tx])continue;
    const bx=tx*TILE,by=ty*TILE;
    const px=Math.max(bx,Math.min(e.x,bx+TILE)),py=Math.max(by,Math.min(e.y,by+TILE));
    const dx=e.x-px,dy=e.y-py,d2=dx*dx+dy*dy;
    if(d2>e.r*e.r)continue;
    let d=Math.sqrt(d2),nx,ny;
    if(d<.001){
      const l=e.x-bx,r2=bx+TILE-e.x,t2=e.y-by,b2=by+TILE-e.y,m=Math.min(l,r2,t2,b2);
      if(m===l){nx=-1;ny=0;}else if(m===r2){nx=1;ny=0;}else if(m===t2){nx=0;ny=-1;}else{nx=0;ny=1}
    }else{ nx=dx/d; ny=dy/d; }
    const pen=e.r-d; e.x+=nx*pen; e.y+=ny*pen;
    const vn=e.vx*nx+e.vy*ny;
    if(vn<0){
      e.vx-=(1+rest)*vn*nx; e.vy-=(1+rest)*vn*ny;
      if(!hit||(-vn)>hit.impact)hit={nx,ny,impact:-vn};
    }
  }
  return hit;
}
function lerpAng(a,b,t){ let d=b-a; while(d>Math.PI)d-=6.2832; while(d<-Math.PI)d+=6.2832; return a+d*t; }

/* ============================== PLAYER ============================== */
function updatePlayer(dt){
  let ix=0,iy=0,pb=false;
  if(keys.has('KeyW')||keys.has('ArrowUp'))iy-=1;
  if(keys.has('KeyS')||keys.has('ArrowDown'))iy+=1;
  if(keys.has('KeyA')||keys.has('ArrowLeft'))ix-=1;
  if(keys.has('KeyD')||keys.has('ArrowRight'))ix+=1;
  if(pAct&&P){
    const wx=(pSx-offx)/view,wy=(pSy-offy)/view,dx=wx-P.x,dy=wy-P.y,d=Math.hypot(dx,dy);
    if(d>14){ ix+=dx/d; iy+=dy/d; }
    pb=d>130||pointers.size>1;
  }
  const m=Math.hypot(ix,iy); if(m>1){ix/=m;iy/=m;}
  boosting=(keys.has('Space')||keys.has('ShiftLeft')||keys.has('ShiftRight')||pb)&&boost>2;
  if(boosting&&prevBoosting===false)sfxBoost();
  prevBoosting=boosting;
  if(boosting){ boost=Math.max(0,boost-46*dt); boostCd=.35; }
  else{ boostCd-=dt; if(boostCd<=0)boost=Math.min(100,boost+26*dt); }

  const acc=1150*(1+(level-1)*.03)*(boosting?1.85:1);
  P.lat=0;
  if(m>0){
    P.vx+=ix*acc*dt; P.vy+=iy*acc*dt;
    const f=P.vx*ix+P.vy*iy,lx=P.vx-f*ix,ly=P.vy-f*iy,gk=1-Math.exp(-4.6*dt);
    P.vx-=lx*gk; P.vy-=ly*gk; P.lat=Math.hypot(lx,ly);
  }
  const dr=Math.exp(-.55*dt); P.vx*=dr; P.vy*=dr;
  const sp=Math.hypot(P.vx,P.vy),maxS=(boosting?540:330)*(1+(level-1)*.02);
  if(sp>maxS){ const k=1-Math.exp(-7*dt),ns=maxS+(sp-maxS)*(1-k); P.vx*=ns/sp; P.vy*=ns/sp; }
  P.x+=P.vx*dt; P.y+=P.vy*dt;
  const hit=collideWalls(P,.42);
  if(hit){
    P.squash=1; P.sqNx=hit.nx; P.sqNy=hit.ny;
    if(hit.impact>140){ sfxWall(hit.impact); shake=Math.min(10,shake+hit.impact/70);
      sparks(P.x-hit.nx*P.r,P.y-hit.ny*P.r,5,'#9db4ff',160); }
  }
  P.squash*=Math.exp(-8*dt);
  if(sp>30)P.face=lerpAng(P.face,Math.atan2(P.vy,P.vx),1-Math.exp(-14*dt));
  P.invuln=Math.max(0,P.invuln-dt);
  // trail + skids
  trailAcc+=dt;
  if((sp>260||boosting)&&trailAcc>.024){ trailAcc=0; trail.push({x:P.x,y:P.y,l:.3,b:boosting}); }
  skidAcc+=dt;
  if(P.lat>150&&sp>230&&skidAcc>.03){ skidAcc=0;
    skids.push({x:P.x,y:P.y,a:Math.atan2(P.vy,P.vx),l:2.2,life:2.2}); }
}

/* ============================== GHOSTS ============================== */
function updateGhosts(dt){
  for(const g of ghosts){
    if(g.state==='cooldown'){
      g.t-=dt; g.vx*=Math.exp(-4*dt); g.vy*=Math.exp(-4*dt);
      if(g.t<=0)g.state='chase';
      g.squash*=Math.exp(-8*dt); continue;
    }
    if(g.state==='eyes'){
      const s=SHRINES[g.si],sx=s[0]*TILE+24,sy=s[1]*TILE+24;
      let dx=sx-g.x,dy=sy-g.y,d=Math.hypot(dx,dy)||1;
      g.vx+=(dx/d*430-g.vx)*Math.min(1,5*dt); g.vy+=(dy/d*430-g.vy)*Math.min(1,5*dt);
      g.x+=g.vx*dt; g.y+=g.vy*dt;
      if(Math.random()<.5)particles.push({x:g.x,y:g.y,vx:0,vy:0,l:.3,life:.3,c:'#bfeeff',s:2});
      if(d<14){ g.state='cooldown'; g.t=1; g.vx*=.2; g.vy*=.2; }
      continue;
    }
    const fr=powerT>0;
    let tx=P.x,ty=P.y;
    if(!fr){
      if(g.type==='hunter'){tx=P.x+P.vx*.45;ty=P.y+P.vy*.45;}
      else if(g.type==='ambusher'){tx=P.x+P.vx*.8;ty=P.y+P.vy*.8;}
      else if(g.type==='chaos'){ g.chaosT-=dt;
        if(g.chaosT<=0){ g.chaosT=1+Math.random();
          const a=Math.random()*6.283,r2=120+Math.random()*180;
          g.cx=P.x+Math.cos(a)*r2; g.cy=P.y+Math.sin(a)*r2; }
        tx=g.cx;ty=g.cy; }
    }
    let dx=tx-g.x,dy=ty-g.y,d=Math.hypot(dx,dy)||1;
    if(fr){ dx=g.x-P.x; dy=g.y-P.y; d=Math.hypot(dx,dy)||1; }
    g.wt+=dt;
    const wa=Math.sin(g.wt*2.3+g.seed)*(fr?.95:(g.type==='drifter'?1.4:.35));
    const ang=Math.atan2(dy,dx)+wa, spd=fr?168:g.spd;
    let sxv=Math.cos(ang)*spd-g.vx, syv=Math.sin(ang)*spd-g.vy;
    const F=fr?520:(g.type==='drifter'?360:640), sm=Math.hypot(sxv,syv);
    if(sm>F){sxv*=F/sm;syv*=F/sm;}
    g.vx+=sxv*dt; g.vy+=syv*dt;
    const gd=Math.exp(-.35*dt); g.vx*=gd; g.vy*=gd;
    const sp2=Math.hypot(g.vx,g.vy),cap=spd*1.3;
    if(sp2>cap){const k=cap/sp2;g.vx*=k;g.vy*=k;}
    g.x+=g.vx*dt; g.y+=g.vy*dt;
    const hit=collideWalls(g,.85);
    if(hit){ g.squash=1; g.sqNx=hit.nx; g.sqNy=hit.ny;
      if(hit.impact>230)tone(300,120,.06,'triangle',.04); }
    g.squash*=Math.exp(-8*dt);
  }
  // separation
  for(let i=0;i<ghosts.length;i++)for(let j=i+1;j<ghosts.length;j++){
    const a=ghosts[i],b=ghosts[j];
    if(a.state!=='chase'||b.state!=='chase')continue;
    const dx=b.x-a.x,dy=b.y-a.y,d=Math.hypot(dx,dy);
    if(d>0.01&&d<32){ const p=(32-d)/d*45*dt;
      a.vx-=dx*p;a.vy-=dy*p;b.vx+=dx*p;b.vy+=dy*p; }
  }
}
function collidePlayerGhosts(){
  if(P.invuln>0)return;
  for(const g of ghosts){
    if(g.state!=='chase')continue;
    const d=Math.hypot(g.x-P.x,g.y-P.y);
    if(d<(P.r+g.r)*.8){
      if(powerT>0){ eatGhost(g); } else { die(); return; }
    }
  }
}
function eatGhost(g){
  g.state='eyes'; g.vx=g.vy=0;

}

function tryPellets(e,silent){
  const cx=(e.x/TILE)|0,cy=(e.y/TILE)|0;
  for(let dy=-1;dy<=1;dy++)for(let dx=-1;dx<=1;dx++){
    const gx=cx+dx,gy=cy+dy;
    if(gx<0||gy<0||gx>=COLS||gy>=ROWS)continue;
    const i=pelGrid[gy*COLS+gx]; if(i<0)continue;
    const pl=pellets[i]; if(pl.dead)continue;
    const d2=(e.x-pl.x)**2+(e.y-pl.y)**2;
    if(d2<(e.r+6)*(e.r+6)){
      pl.dead=true; pelGrid[gy*COLS+gx]=-1; pelletsLeft--;
      if(silent){ sparks(pl.x,pl.y,3,'#ffe9c4',60); }
      else if(pl.pow){
        addScore(50); powerMax=Math.max(4.5,8-level*.4); powerT=powerMax; combo=1;
        sfxPower(); shake=Math.min(8,shake+3);
        ringFX(pl.x,pl.y,'#ff5d73'); sparks(pl.x,pl.y,18,'#ff9db0',220);
        popup(pl.x,pl.y-16,'POWER!','#ff8fa5');
        for(const g of ghosts)if(g.state==='chase')ringFX(g.x,g.y,'#8fd6ff');
      }else{ addScore(10); sfxDot(); sparks(pl.x,pl.y,3,'#ffe9c4',70); }
    }
  }
  if(!silent&&pelletsLeft<=0&&state==='playing')startLevelUp();
}

  for(let i=particles.length-1;i>=0;i--){ const p=particles[i]; p.l-=dt;
    if(p.type==='ring'){ p.r+=p.vr*dt; } else { p.x+=p.vx*dt;p.y+=p.vy*dt;
      const k=Math.exp(-2.5*dt);p.vx*=k;p.vy*=k; }
    if(p.l<=0)particles.splice(i,1);
  }
  for(let i=popups.length-1;i>=0;i--){ const p=popups[i]; p.l-=dt; p.y-=28*dt; if(p.l<=0)popups.splice(i,1); }
  for(let i=trail.length-1;i>=0;i--){ trail[i].l-=dt; if(trail[i].l<=0)trail.splice(i,1); }

  shake*=Math.exp(-6*dt); if(shake<.05)shake=0;
}

  lg.addColorStop(0,'#0a0f30'); lg.addColorStop(.55,'#0b1236'); lg.addColorStop(1,'#080c26');
  c.fillStyle=lg; c.fillRect(0,0,vw,vh);
  const blobs=[[.2,.25,'rgba(70,110,255,.07)'],[.85,.3,'rgba(45,220,200,.06)'],[.5,.9,'rgba(255,110,199,.05)']];
  for(const b of blobs){ const g=c.createRadialGradient(b[0]*vw,b[1]*vh,10,b[0]*vw,b[1]*vh,vw*.45);
    g.addColorStop(0,b[2]); g.addColorStop(1,'rgba(0,0,0,0)'); c.fillStyle=g; c.fillRect(0,0,vw,vh); }
}
function resize(){
  vw=innerWidth; vh=innerHeight; dpr=Math.min(2,window.devicePixelRatio||1);
  cv.width=vw*dpr; cv.height=vh*dpr;
  view=Math.min(vw/W,vh/H); offx=(vw-W*view)/2; offy=(vh-H*view)/2;
  makeStars(); makeBG();
}
function dropShadow(x,y,r){
  ctx.fillStyle='rgba(0,0,10,.3)';
  ctx.beginPath(); ctx.ellipse(x,y+r*.72,r*.85,r*.34,0,0,7); ctx.fill();
}
function drawPellets(){
  ctx.fillStyle='rgba(255,236,190,.92)'; ctx.beginPath();
  for(const pl of pellets){ if(pl.dead||pl.pow)continue;
    ctx.moveTo(pl.x+3.1,pl.y); ctx.arc(pl.x,pl.y,3.1,0,7); }
  ctx.fill();
  for(const pl of pellets){ if(pl.dead||!pl.pow)continue;
    const r=6.5+1.8*Math.sin(T*6+pl.x*.05);
    ctx.shadowColor='#ff5d73'; ctx.shadowBlur=14;
    ctx.fillStyle='#ff5d73'; ctx.beginPath(); ctx.arc(pl.x,pl.y,r,0,7); ctx.fill();
    ctx.shadowBlur=0; ctx.fillStyle='rgba(255,255,255,.8)';
    ctx.beginPath(); ctx.arc(pl.x-1.5,pl.y-1.5,1.8,0,7); ctx.fill();
  }
}
function drawPac(e){
  const sp=Math.hypot(e.vx||0,e.vy||0);
  let scl=1;
  if(state==='dying'&&e===P)scl=deathT<.85?1:Math.max(0,1-(deathT-.85)/.6);
  ctx.save(); ctx.translate(e.x,e.y);
  if(e.squash>.02){ const a=Math.atan2(e.sqNy,e.sqNx);
    ctx.rotate(a); ctx.scale(1+.35*e.squash,1-.3*e.squash); ctx.rotate(-a); }
  if(scl<1)ctx.scale(scl,scl);
  if(state==='dying'&&e===P)ctx.rotate(P.dieA); else ctx.rotate(e.face);
  const mh=state==='dying'&&e===P?1.3:(sp>25?.18+.95*(.5+.5*Math.sin(T*(8+sp*.02)+ (e.seed||0))):.25);
  if(e.invuln>0&&(T*14%2<1))ctx.globalAlpha=.35;
  ctx.shadowColor='#ffd93b'; ctx.shadowBlur=boosting&&e===P?24:13;
  ctx.fillStyle='#ffd93b';
  ctx.beginPath(); ctx.moveTo(0,0); ctx.arc(0,0,e.r+1,mh,Math.PI*2-mh); ctx.closePath(); ctx.fill();
  ctx.shadowBlur=0; ctx.strokeStyle='rgba(120,80,0,.3)'; ctx.lineWidth=1.5; ctx.stroke();
  ctx.restore();
}
function drawGhost(g){
  const fr=powerT>0&&g.state==='chase';
  let body=fr?'#3a46a8':g.color;
  if(g.state==='cooldown')body=(T*10%2<1)?'#ffffff':g.color;
  const wob=Math.sin(T*9+g.seed)*1.6, r=g.r;
  ctx.save(); ctx.translate(g.x,g.y);
  if(g.squash>.02){ const a=Math.atan2(g.sqNy,g.sqNx);
    ctx.rotate(a); ctx.scale(1+.3*g.squash,1-.26*g.squash); ctx.rotate(-a); }
  // body
  if(g.state!=='eyes'){
    ctx.shadowColor=body; ctx.shadowBlur=12; ctx.fillStyle=body;
    ctx.beginPath();
    ctx.moveTo(-r,r*.85+wob); ctx.lineTo(-r,-r*.15);
    ctx.arc(0,-r*.15,r,Math.PI,0);
    ctx.lineTo(r,r*.85+wob);
    const w=2*r/4;
    for(let i=0;i<4;i++){ const x0=r-i*w;
      ctx.quadraticCurveTo(x0-w/2,r*.85+wob+7,x0-w,r*.85+wob); }
    ctx.closePath(); ctx.fill(); ctx.shadowBlur=0;
  }
  // eyes / face
  let lx=g.vx,ly=g.vy,ld=Math.hypot(lx,ly);
  if(ld<20){lx=P.x-g.x;ly=P.y-g.y;ld=Math.hypot(lx,ly)||1;}
  lx/=ld;ly/=ld;
  if(fr){
    ctx.fillStyle='#dfe6ff';
    ctx.beginPath();ctx.arc(-r*.34,-r*.2,r*.13,0,7);ctx.fill();
    ctx.beginPath();ctx.arc(r*.34,-r*.2,r*.13,0,7);ctx.fill();
    ctx.strokeStyle='#dfe6ff';ctx.lineWidth=2;ctx.beginPath();
    for(let i=0;i<=6;i++){const x=-r*.5+i*r/6,y=r*.28+(i%2?-4:4);
      i?ctx.lineTo(x,y):ctx.moveTo(x,y);} ctx.stroke();
  }else{
    ctx.fillStyle='#fff';
    ctx.beginPath();ctx.ellipse(-r*.33,-r*.25,r*.27,r*.31,0,0,7);ctx.fill();
    ctx.beginPath();ctx.ellipse(r*.33,-r*.25,r*.27,r*.31,0,0,7);ctx.fill();
    ctx.fillStyle='#10173f';
    ctx.beginPath();ctx.arc(-r*.33+lx*r*.12,-r*.25+ly*r*.12,r*.14,0,7);ctx.fill();
    ctx.beginPath();ctx.arc(r*.33+lx*r*.12,-r*.25+ly*r*.12,r*.14,0,7);ctx.fill();
  }
  ctx.restore();
}
function render(){
  ctx.setTransform(dpr,0,0,dpr,0,0);
  if(bgCv)ctx.drawImage(bgCv,0,0,vw,vh);
  for(const s of stars){ const a=.1+.22*(.5+.5*Math.sin(T*s.tw+s.ph));
    ctx.fillStyle='rgba(200,215,255,'+a.toFixed(3)+')';
    ctx.beginPath();ctx.arc(s.x,s.y,s.r,0,7);ctx.fill(); }
  const shx=(Math.random()*2-1)*shake, shy=(Math.random()*2-1)*shake;
  ctx.save(); ctx.translate(offx+shx,offy+shy); ctx.scale(view,view);
  if(mazeCv)ctx.drawImage(mazeCv,0,0,W,H);
  // skids
  ctx.strokeStyle='rgba(150,180,255,.9)'; ctx.lineCap='round';
  for(const s of skids){ ctx.globalAlpha=.13*(s.l/s.life);
    const dx=Math.cos(s.a)*7,dy=Math.sin(s.a)*7;
    ctx.lineWidth=4; ctx.beginPath(); ctx.moveTo(s.x-dx,s.y-dy); ctx.lineTo(s.x+dx,s.y+dy); ctx.stroke(); }
  ctx.globalAlpha=1;
  drawPellets();
  for(const t of trail){ const k=t.l/.3;
    ctx.globalAlpha=k*.42; ctx.fillStyle=t.b?'#bfeeff':'#ffd93b';
    ctx.beginPath();ctx.arc(t.x,t.y,13*(.4+.6*k),0,7);ctx.fill(); }
  ctx.globalAlpha=1;
  if(powerT>0&&state!=='menu'&&P){
    ctx.strokeStyle='rgba(140,220,255,'+(.25+.2*Math.sin(T*9)).toFixed(3)+')';
    ctx.lineWidth=2.5; ctx.beginPath();
    ctx.arc(P.x,P.y,26+3*Math.sin(T*9),0,7); ctx.stroke();
  }
  for(const g of ghosts){ dropShadow(g.x,g.y,g.r); drawGhost(g); }
  if(state==='menu'){ if(DP){dropShadow(DP.x,DP.y,15); drawPac(DP);} }
  else if(P){ dropShadow(P.x,P.y,15); drawPac(P); }
  for(const p of particles){ const k=p.l/p.life;
    if(p.type==='ring'){ ctx.globalAlpha=k*.8; ctx.strokeStyle=p.c; ctx.lineWidth=3;
      ctx.beginPath();ctx.arc(p.x,p.y,p.r,0,7);ctx.stroke(); }
    else{ ctx.globalAlpha=k; ctx.fillStyle=p.c;
      ctx.beginPath();ctx.arc(p.x,p.y,Math.max(.4,p.s*k+.4),0,7);ctx.fill(); } }
  ctx.globalAlpha=1;
  ctx.font='700 15px ui-monospace,Menlo,Consolas,monospace'; ctx.textAlign='center';
  for(const p of popups){ const k=p.l/p.life; ctx.globalAlpha=k;
    ctx.lineWidth=3; ctx.strokeStyle='rgba(8,12,40,.8)'; ctx.strokeText(p.txt,p.x,p.y);
    ctx.fillStyle=p.c; ctx.fillText(p.txt,p.x,p.y); }
  ctx.globalAlpha=1; ctx.restore();
}
