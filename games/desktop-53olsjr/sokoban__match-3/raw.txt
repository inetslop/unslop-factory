function inBounds(r,c){return r>=0&&c>=0&&r<rows&&c<cols;}
function doBump(){player.bumpT=1;bumpCd=.24;SFX.bump();}
function startPlayerMove(nr,nc,push){
  player.r=nr;player.c=nc;
  player.moving={t:0,dur:.085,sx:player.x,sy:player.y,tx:cx(nc),ty:cy(nr)};
  player.pendingPush=push;

function tryStep(dx,dy){
  if(state!=='play'||player.moving)return;
  player.fx=dx;player.fy=dy;
  const nr=player.r+dy,nc=player.c+dx;
  if(!inBounds(nr,nc)||walls[nr][nc]){doBump();return;}
  const cr=crateAt(nr,nc);
  if(cr){const br=nr+dy,bc=nc+dx;
    if(!inBounds(br,bc)||walls[br][bc]||crateAt(br,bc)){doBump();return;}
    pushUndo();moves++;
    cr.r=br;cr.c=bc;cr.tw={t:0,dur:.085,sx:cr.x,sy:cr.y,tx:cx(bc),ty:cy(br)};
    startPlayerMove(nr,nc,true);SFX.push();dust(cr.x,cr.y+cell*.32,3,'#c9a06a');
  }else{pushUndo();moves++;startPlayerMove(nr,nc,false);SFX.step();}
  updateHUD();
}
function onPlayerArrive(){
  if(player.pendingPush){player.pendingPush=false;resolveBoard();return;}
  let fell=false;
  for(const k of crates)if(k.state==='idle'&&canFall(k)){startFreeFall(k);fell=true;}
  if(fell){state='cascade';cascadePhase='fall';}
}
function canFall(k){const r=k.r+1,c=k.c;if(!inBounds(r,c)||walls[r][c]||crateAt(r,c))return false;
  if(player&&player.r===r&&player.c===c)return false;return true;}
function startFreeFall(k){let tr=k.r+1;
  while(inBounds(tr+1,k.c)&&!walls[tr+1][k.c]&&!crateAt(tr+1,k.c)&&!(player.r===tr+1&&player.c===k.c))tr++;
  k.r=tr;k.state='fall';k.vy=Math.max(k.vy,0);k.hit=false;k.ty=cy(tr);}

/* ============================== match-3 core ============================== */
function findMatches(){
  const hits=new Map();
  function absorb(run){const len=run.length;for(const cr of run){const e=hits.get(cr.id)||{crate:cr,len:0};if(len>e.len)e.len=len;hits.set(cr.id,e);}}
  for(let r=0;r<rows;r++){let run=[];for(let c=0;c<=cols;c++){const cr=(c<cols)?crateAt(r,c):null;
    if(cr)run.push(cr);else{if(run.length>=3)absorb(run);run=[];}}}
  for(let c=0;c<cols;c++){let run=[];for(let r=0;r<=rows;r++){const cr=(r<rows)?crateAt(r,c):null;
    if(cr)run.push(cr);else{if(run.length>=3)absorb(run);run=[];}}}
  return[...hits.values()];
}
function resolveBoard(){
  const hits=findMatches();
  if(!hits.length){combo=0;if(crates.length===0)beginClear();else state='play';return;}
  combo++;cascadePhase='pop';
  let pts=0,sx=0,sy=0;
  for(const h of hits){const k=h.crate;k.state='pop';k.popT=0;
    pts+=10*combo+Math.max(0,h.len-3)*25*combo;sx+=k.x;sy+=k.y;gemBurst(k);}
  score+=pts;flashScore();
  addPopup(sx/hits.length,sy/hits.length,'+'+pts,combo>=2?'#ffd166':'#ffffff');
  if(combo>1)banner={t:0,n:combo};
  shake=Math.min(shake+4+hits.length*1.3+(combo-1)*3,16);
  SFX.clear(combo,hits.length);updateHUD();
}
function applyGravity(){
  for(const k of crates)k.oldR=k.r;
  for(let c=0;c<cols;c++){
    const ranges=[];let a=0;
    for(let r=0;r<=rows;r++){if(r===rows||walls[r][c]){if(a<r)ranges.push([a,r-1]);a=r+1;}}
    for(const[ra,rb]of ranges){
      const inR=crates.filter(k=>k.c===c&&k.r>=ra&&k.r<=rb);
      const pHere=(player.c===c&&player.r>=ra&&player.r<=rb)?player.r:-1;
      let write=rb;
      const below=inR.filter(k=>pHere<0||k.r>pHere).sort((x,y)=>y.r-x.r);
      for(const k of below){if(pHere>0&&write===pHere)break;k.r=write--;}
      if(pHere>0){write=pHere-1;for(const k of inR.filter(x=>x.r<pHere).sort((x,y)=>y.r-x.r))k.r=write--;}
    }
  }
  let any=false;
  for(const k of crates){if(k.state!=='idle')continue;if(k.r!==k.oldR){k.state='fall';k.vy=0;k.hit=false;k.ty=cy(k.r);any=true;}}
  return any;
}

  newBestFlag=score>best;if(newBestFlag){best=score;store.set('gemhaul.best',best);}
  confetti();SFX.win();updateHUD();$('best-start').textContent=best;
}

function update(dt){
  tGlobal+=dt;bumpCd-=dt;undoT=Math.max(0,undoT-dt);
  shake*=Math.exp(-6*dt);if(shake<.05)shake=0;
  // motes
  for(const m of motes){m.y-=m.spd*dt;m.x+=Math.sin(tGlobal*.4+m.ph)*8*dt;if(m.y<-10){m.y=H+10;m.x=Math.random()*W;}}
  // held input
  if(touchDir&&tGlobal>touchExp)touchDir=null;
  if(state==='play'&&!player.moving&&bumpCd<=0){
    let dir=null;
    if(touchDir)dir=touchDir;else{const code=held[held.length-1];if(code)dir=KEYMAP[code];}
    if(dir)tryStep(dir[0],dir[1]);
  }
  // player tween
  if(player.moving){const m=player.moving;m.t+=dt/m.dur;const k=easeOut(Math.min(1,m.t));
    player.x=m.sx+(m.tx-m.sx)*k;player.y=m.sy+(m.ty-m.sy)*k;
    if(m.t>=1){player.x=m.tx;player.y=m.ty;player.moving=null;onPlayerArrive();}}
  player.bumpT=Math.max(0,player.bumpT-dt*5);
  // crates
  for(let i=crates.length-1;i>=0;i--){const k=crates[i];
    if(k.tw){k.tw.t+=dt/k.tw.dur;const e=easeOut(Math.min(1,k.tw.t));k.x=k.tw.sx+(k.tw.tx-k.tw.sx)*e;k.y=k.tw.sy+(k.tw.ty-k.tw.sy)*e;if(k.tw.t>=1){k.x=k.tw.tx;k.y=k.tw.ty;k.tw=null;}}
    if(k.state==='fall'){const drop=Math.max(1,k.ty-(k.y-k.vy*dt));
      k.vy+=46*cell*dt;k.y+=k.vy*dt;
      if(k.y>=k.ty){k.y=k.ty;
        if(!k.hit&&drop>cell*.55){SFX.land();dust(k.x,k.y+cell*.36,4);}
        k.hit=true;k.vy=-Math.abs(k.vy)*.28;k.squash=1;
        if(Math.abs(k.vy)<cell*1.9){k.state='idle';k.vy=0;}}}
    k.squash=Math.max(0,k.squash-dt*5);
  }
  // cascade phases
  if(state==='cascade'){
    if(cascadePhase==='pop'){let done=true;
      for(const k of crates)if(k.state==='pop'){k.popT+=dt/.28;if(k.popT<1)done=false;}
      if(done){crates=crates.filter(k=>k.state!=='pop');updateHUD();
        cascadePhase=applyGravity()?'fall':'check';}}
    else if(cascadePhase==='fall'){if(!crates.some(k=>k.state==='fall'))cascadePhase='check';}
    if(cascadePhase==='check')resolveBoard();
  }
  // level clear timer
  if(state==='clear'){clearTimer+=dt;
    if(clearTimer>1&&!overlayShown){overlayShown=true;SFX.ui();
      const last=LIDX===LEVELS.length-1;
      if(last){$('v-score').textContent=score;$('v-best').textContent=best;
        $('newbest').classList.toggle('hidden',!newBestFlag);show('scr-victory');}
      else{$('clear-title').textContent='VAULT '+(LIDX+1)+' CLEARED';
        $('clear-flavor').textContent='“'+LEVELS[LIDX].flavor+'”';
        $('st-moves').textContent=moves;$('st-par').textContent=par;
        $('st-bonus').textContent='+'+lastBonus;$('st-score').textContent=score;show('scr-clear');}}
  }
  // particles
  for(let i=parts.length-1;i>=0;i--){const p=parts[i];p.life-=dt;if(p.life<=0){parts.splice(i,1);continue;}
    if(p.type==='shard'){p.vy+=34*cell*dt;p.x+=p.vx*dt;p.y+=p.vy*dt;p.rot+=p.vr*dt;}
    else if(p.type==='dust'){p.x+=p.vx*dt;p.y+=p.vy*dt;p.r+=14*dt;}
    else if(p.type==='ring'){p.r+=p.vr*dt;}}
  for(let i=popups.length-1;i>=0;i--){popups[i].t+=dt;if(popups[i].t>1)popups.splice(i,1);}
  if(banner){banner.t+=dt;if(banner.t>.85)banner=null;}
  // hint
  computeHint();
}

function computeHint(){
  hintCrate=null;
  if(state!=='play'||player.moving)return;
  const dx=player.fx,dy=player.fy,nr=player.r+dy,nc=player.c+dx;
  if(!inBounds(nr,nc)||walls[nr][nc])return;
  const cr=crateAt(nr,nc);if(!cr)return;
  const br=nr+dy,bc=nc+dx;if(!inBounds(br,bc)||walls[br][bc]||crateAt(br,bc))return;
  const col=cr.color;
  function scan(r,c,dr,dc){let n=0;while(inBounds(r,c)&&!walls[r][c]){const o=crateAt(r,c);
    if(o&&o.color===col&&!(r===cr.r&&c===cr.c)){n++;r+=dr;c+=dc;}else break;}return n;}
  const h=1+scan(nr,bc+1,0,1)+scan(nr,bc-1,0,-1);
  const v=1+scan(nr+1,bc,1,0)+scan(nr-1,bc,-1,0);
  if(Math.max(h,v)>=3)hintCrate=cr;
}

function diamond(x,y,s){ctx.beginPath();ctx.moveTo(x,y-s);ctx.lineTo(x+s,y);ctx.lineTo(x,y+s);ctx.lineTo(x-s,y);ctx.closePath();}
