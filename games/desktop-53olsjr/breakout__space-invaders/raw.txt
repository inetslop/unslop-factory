function ring(x,y,color){ particles.push({ring:true,x,y,r:4,life:0,max:.45,color}); }
function mkFloat(txt,x,y,color,s,dur){ floaters.push({txt:String(txt),x,y,t:0,dur:dur||1,color:s?color:'#eaf6ff',s:s||2}); if(floaters.length>24)floaters.shift(); }
function addScore(n,x,y,color){ S.score+=n; if(x!=null)mkFloat(n,x,y,color,2); if(S.score>S.hi){S.hi=S.score;S.newHi=true;} }

/* ============ game flow ============ */
const curSpeed=()=>clamp(320+S.wave*22,320,560)*(W<640?0.85:1);
function buildWave(){
  bricks.length=0; shots.length=0; eshots.length=0; invaders.length=0;
  const bw=Math.min(W*0.6,620), cw=bw/22, chh=clamp(cw*0.75,10,18);
  const bx=(W-bw)/2, by=HUD_H+24;
  let cnt=0;
  for(let r=0;r<9;r++)for(let c=0;c<22;c++) if(BOSS[r][c]==='1'){
    bricks.push({x:bx+c*cw+1,y:by+r*chh+1,w:cw-2,h:chh-2,hp:2,kind:'boss',row:r,color:BOSSPAL[r],flash:0}); cnt++; }
  bossTotal=cnt; bossLeft=cnt;
  const wy=by+9*chh+16, ww=Math.min(W*0.78,760), ncol=Math.max(6,Math.floor(ww/52)), gpx=6, wbw=(ww-(ncol-1)*gpx)/ncol;
  for(let r=0;r<3;r++)for(let c=0;c<ncol;c++){
    bricks.push({x:(W-ww)/2+c*(wbw+gpx),y:wy+r*22,w:wbw,h:15,hp:1,kind:'wall',row:r,color:['#9fe870','#ffd166','#ff9f45'][r],flash:0}); }
  S.invLeft=5+S.wave*2; S.maxInv=Math.min(7,3+S.wave); S.spawnT=1.8;
}
function resetBall(){ balls=[{x:pd.x,y:pd.y-13,vx:0,vy:0,stuck:true,r:6,base:curSpeed(),trail:[]}]; }
function tryLaunch(){
  let any=false;
  for(const b of balls) if(b.stuck){ b.stuck=false; const a=-Math.PI/2+rand(-.25,.25),sp=b.base||curSpeed();
    b.vx=Math.sin(a)*sp; b.vy=-Math.cos(a)*sp; any=true; }
  if(any)sLaunch();
}
function spawnExtraBalls(){
  const src=balls.find(b=>!b.stuck); if(!src)return;
  for(const a of [-0.5,0.5]){ if(balls.length>=6)break;
    const sp=src.base||curSpeed(), ang=-Math.PI/2+a*0.9;
    balls.push({x:src.x,y:src.y,vx:Math.sin(ang)*sp,vy:-Math.cos(ang)*sp,stuck:false,r:6,base:sp,trail:[]}); }
}
function updateMult(){ S.mult=1+Math.min(4,Math.floor(S.combo/5)); }
function mkPower(x,y){
  const r=Math.random(); const t=r<.24?'L':r<.44?'W':r<.64?'M':r<.82?'S':'D';
  powerups.push({bx:x,x,y,vy:140+S.wave*6,t:rand(TAU),type:t});
}
function applyPower(t){
  if(t==='L')powers.laser=Math.min(24,powers.laser+12);
  else if(t==='W')powers.wide=12;
  else if(t==='S')powers.slow=8;
  else if(t==='D'){ powers.shield=Math.min(3,powers.shield+1); sShield(); }
  else if(t==='M')spawnExtraBalls();
  sPower(); mkFloat(POWNAME[t],pd.x,pd.y-46,POWCOL[t],2,1.2);
}
function loseLife(kind){
  if(S.invuln>0&&kind!=='ball')return;
  S.lives--; sHurt(); shake=Math.max(shake,kind==='ball'?12:16); S.redFlash=0.9; S.combo=0; updateMult();
  burst(kind==='ball'?W/2:pd.x,pd.y-10,'#ff5d7a',26,320,400,4);
  if(S.lives<=0){ gameOver(); return; }
  S.invuln=kind==='ball'?0:1.4;
  if(kind==='ball')resetBall();
}
function hitBrick(br,byBall){
  br.hp--; br.flash=1;
  if(byBall){ S.combo++; updateMult(); }
  if(br.hp<=0){
    br.dead=true;
    const base=br.kind==='boss'?18:35-br.row*5;
    addScore(base*S.mult,br.x+br.w/2,br.y+br.h/2, br.kind==='boss'?br.color:'#eaf6ff');
    burst(br.x+br.w/2,br.y+br.h/2,br.color,8,200,650,3);
    sBrickBreak(br.y);
    if(br.kind==='boss'){ bossLeft--; }
    const ch=br.kind==='wall'?0.18:0.05;
    if(Math.random()<ch)mkPower(br.x+br.w/2,br.y+br.h/2);
  } else sBrickHit(br.y);
}
function killInvader(inv){
  inv.dead=true; addScore(75+S.wave*5,inv.x,inv.y,'#ff9fd6',2);
  burst(inv.x,inv.y,'#ff7cc8',16,260,300,4); ring(inv.x,inv.y,'#ff7cc8'); shake=Math.max(shake,5); sInvDie();
  if(Math.random()<0.4)mkPower(inv.x,inv.y);
}
function spawnInvader(){
  const top=HUD_H+9*clamp(bossTotal>0?1:0,0,1); // placeholder no-op guard
  const bandTop=HUD_H+ (bricks.length? Math.max(...bricks.filter(b=>b.kind==='boss').map(b=>b.y+b.h)) : 120)+26;
  const hoverY=rand(clamp(bandTop,90,H*0.38), H*0.45);
  const amp=Math.min(W*0.26,170);
  invaders.push({x:rand(60,W-60),y:-24,ax:rand(60+amp? Math.max(40,amp+30):40, W-Math.max(40,amp+30)),hoverY,t:rand(TAU),f:rand(.5,.95),amp,ft:rand(1.2,2.4),tele:0,dead:false,ph:rand(TAU)});
}
function waveComplete(){
  const bonus=400+S.wave*100; addScore(bonus,null,null); mkFloat('BONUS +'+bonus,W/2,H*0.5,'#ffd166',3,1.6);
  for(const inv of invaders){ if(!inv.dead)burst(inv.x,inv.y,'#ff7cc8',8,200,200,3); }
  eshots.length=0; powerups.length=0;
  S.wave++; buildWave(); S.banner=2.4; resetBall(); sWave();
}
function hideScreens(){ document.getElementById('scr-start').classList.remove('on'); document.getElementById('scr-over').classList.remove('on'); }
function startGame(){
  hideScreens();
  S.score=0;S.lives=3;S.wave=1;S.combo=0;S.mult=1;S.invuln=0;S.redFlash=0;S.overT=0;S.panelShown=false;S.newHi=false;
  powers.laser=0;powers.wide=0;powers.slow=0;powers.shield=0;
  shots.length=0;eshots.length=0;powerups.length=0;particles.length=0;floaters.length=0;
  layoutPaddle(); buildWave(); S.banner=2.4; resetBall();
  S.mode='playing'; sWave();
}

}

}

/* ============ collision ============ */
function hitRect(x,y,r,rx,ry,rw,rh){
  const cx=clamp(x,rx,rx+rw), cy=clamp(y,ry,ry+rh);
  const dx=x-cx, dy=y-cy;
  return (dx*dx+dy*dy<=r*r)?{dx,dy}:null;
}

/* ============ update ============ */
function update(dt){
  S.time+=dt;
  shake=Math.max(0,shake-dt*40);
  S.redFlash=Math.max(0,S.redFlash-dt*1.4);
  S.invuln=Math.max(0,S.invuln-dt);
  S.fireCd=Math.max(0,S.fireCd-dt);

  // background
  const spd=S.mode==='playing'?1.7:1;
  for(const st of stars){ st.y+=(6+st.l*16)*spd*dt; if(st.y>H+2){st.y=-2;st.x=Math.random()*W;} }
  shootT-=dt;
  if(shootT<=0){ shootT=rand(3,8); shooters.push({x:rand(W*.1,W*.9),y:-20,vx:(Math.random()<.5?-1:1)*rand(260,420),vy:rand(160,260),life:0,max:rand(.7,1.1)}); }
  for(let i=shooters.length-1;i>=0;i--){ const s=shooters[i]; s.life+=dt; s.x+=s.vx*dt; s.y+=s.vy*dt; if(s.life>s.max)shooters.splice(i,1); }

  // particles & floaters (all modes)
  for(let i=particles.length-1;i>=0;i--){ const p=particles[i]; p.life+=dt;
    if(p.life>p.max){particles.splice(i,1);continue;}
    if(p.ring){p.r+=340*dt;} else {p.x+=p.vx*dt;p.y+=p.vy*dt;p.vy+=(p.grav||0)*dt;p.rot+=p.vr*dt;} }
  for(let i=floaters.length-1;i>=0;i--){ const f=floaters[i]; f.t+=dt; f.y-=32*dt; if(f.t>f.dur)floaters.splice(i,1); }

  if(S.mode==='over'){ S.overT+=dt; if(S.overT>1&&!S.panelShown){S.panelShown=true;showOver();} return; }
  if(S.mode!=='playing')return;

  if(S.banner>0)S.banner-=dt;

  // paddle
  const prevX=pd.x;
  pd.w=lerp(pd.w,pd.pw*(powers.wide>0?1.55:1),Math.min(1,dt*6));
  if(keys.ArrowLeft||keys.KeyA){ usePointer=false; aimX=null; }
  if(keys.ArrowRight||keys.KeyD){ usePointer=false; aimX=null; }
  const dir=((keys.ArrowRight||keys.KeyD)?1:0)-((keys.ArrowLeft||keys.KeyA)?1:0);
  if(dir!==0)pd.x+=dir*760*dt;
  else if(usePointer&&aimX!=null)pd.x+=(aimX-pd.x)*Math.min(1,dt*22);
  pd.x=clamp(pd.x,pd.w/2+8,W-pd.w/2-8);
  pd.vx=(pd.x-prevX)/Math.max(dt,.001);

  // power timers
  powers.wide=Math.max(0,powers.wide-dt);
  powers.slow=Math.max(0,powers.slow-dt);

  // laser fire
  if(powers.laser>0&&keys.Space&&S.fireCd<=0){
    S.fireCd=0.2; powers.laser--;
    const side=(S.fireCd<0?1: (Math.floor(S.time/0.2)%2?1:-1));
    shots.push({x:pd.x+side*pd.w*0.35,y:pd.y-14}); sLaser();
  }

  // balls
  for(let i=balls.length-1;i>=0;i--){
    const b=balls[i];
    if(b.stuck){ b.x=pd.x; b.y=pd.y-13; continue; }
    const want=b.base*(powers.slow>0?0.72:1);
    const m=Math.hypot(b.vx,b.vy)||1; b.vx*=want/m; b.vy*=want/m;
    const sp=Math.hypot(b.vx,b.vy), steps=Math.max(1,Math.ceil(sp*dt/6));
    for(let s=0;s<steps;s++){
      b.x+=b.vx*dt/steps; b.y+=b.vy*dt/steps;
      if(b.x<b.r){b.x=b.r;b.vx=Math.abs(b.vx);sWall();}
      if(b.x>W-b.r){b.x=W-b.r;b.vx=-Math.abs(b.vx);sWall();}
      if(b.y<HUD_H+b.r){b.y=HUD_H+b.r;b.vy=Math.abs(b.vy);sWall();}
      for(const br of bricks){ if(br.dead)continue;
        const h=hitRect(b.x,b.y,b.r,br.x,br.y,br.w,br.h);
        if(h){ if(Math.abs(h.dx)>Math.abs(h.dy)) b.vx=h.dx>0?Math.abs(b.vx):-Math.abs(b.vx);
               else b.vy=h.dy>0?Math.abs(b.vy):-Math.abs(b.vy);
          hitBrick(br,true); break; } }
      for(const inv of invaders){ if(!inv.dead&&hitRect(b.x,b.y,b.r,inv.x-24,inv.y-10,48,20))killInvader(inv); }
      if(b.vy>0&&b.y+b.r>=pd.y-7&&b.y-b.r<=pd.y+9&&Math.abs(b.x-pd.x)<=pd.w/2+b.r){
        const rel=clamp((b.x-pd.x)/(pd.w/2),-1,1), ang=rel*1.05, sp2=Math.hypot(b.vx,b.vy)||want;
        b.vx=Math.sin(ang)*sp2+pd.vx*0.12; b.vy=-Math.abs(Math.cos(ang)*sp2); b.y=pd.y-7-b.r;
        S.combo=0; updateMult(); sPaddle(); burst(b.x,pd.y-8,'#9fe8ff',4,120,300,2);
      }
    }
    b.trail.unshift({x:b.x,y:b.y}); if(b.trail.length>9)b.trail.pop();
    if(b.y-b.r>H+14){ balls.splice(i,1); }
  }
  if(balls.length===0)loseLife('ball');

  // player shots
  for(let i=shots.length-1;i>=0;i--){ const s=shots[i]; s.y-=760*dt;
    let dead=s.y<HUD_H-20;
    if(!dead)for(const inv of invaders){ if(!inv.dead&&Math.abs(s.x-inv.x)<26&&Math.abs(s.y-inv.y)<12){killInvader(inv);dead=true;break;} }
    if(!dead)for(const br of bricks){ if(br.dead)continue;
      if(s.x>br.x&&s.x<br.x+br.w&&s.y>br.y&&s.y<br.y+br.h){hitBrick(br,true);dead=true;break;} }
    if(dead)shots.splice(i,1);
  }

  // invaders
  if(S.invLeft>0&&S.banner<1.2){ S.spawnT-=dt;
    if(S.spawnT<=0&&invaders.length<S.maxInv){ spawnInvader(); S.invLeft--; S.spawnT=rand(1.4,2.8)/(1+S.wave*0.05); } }
  for(let i=invaders.length-1;i>=0;i--){ const inv=invaders[i];
    if(inv.dead){invaders.splice(i,1);continue;}
    if(inv.y<inv.hoverY)inv.y+=95*dt; else { inv.t+=dt; inv.x=clamp(inv.ax+Math.sin(inv.t*inv.f)*inv.amp,40,W-40); }
    if(inv.tele>0){ inv.tele-=dt; if(inv.tele<=0&&inv.y>=inv.hoverY){
        eshots.push({x:inv.x,y:inv.y+14,vx:0,vy:Math.min(560,300+S.wave*16)}); sEShot(); } }
    else { inv.ft-=dt; if(inv.ft<=0&&inv.y>=inv.hoverY){ inv.tele=0.22; inv.ft=rand(1.3,2.6)/(1+S.wave*0.07); } }
    if(inv.y>pd.y-26){
      burst(inv.x,inv.y,'#ff7cc8',14,240,300,4); ring(inv.x,inv.y,'#ff9fd6');
      if(powers.shield>0){ powers.shield--; sShield(); mkFloat('BLOCKED',pd.x,pd.y-52,'#b3c7ff',2); }
      else loseLife('breach');
      invaders.splice(i,1); continue;
    }
  }

  // enemy shots
  for(let i=eshots.length-1;i>=0;i--){ const s=eshots[i];
    s.vy=Math.min(560,s.vy+40*dt); s.y+=s.vy*dt;
    s.vx=lerp(s.vx,clamp((pd.x-s.x)*1.3,-90,90),dt*1.2); s.x+=s.vx*dt;
    let dead=false;
    if(powers.shield>0&&Math.hypot(s.x-pd.x,s.y-(pd.y-4))<38){ powers.shield--; sShield(); mkFloat('BLOCKED',pd.x,pd.y-52,'#b3c7ff',2); burst(s.x,s.y,'#b3c7ff',6,140,200,2); dead=true; }
    if(!dead&&s.y>pd.y-9&&s.y<pd.y+11&&Math.abs(s.x-pd.x)<pd.w/2+5){ loseLife('shot'); burst(s.x,s.y,'#ff5d7a',10,200,300,3); dead=true; }
    if(!dead)for(const br of bricks){ if(br.dead)continue;
      if(s.x>br.x&&s.x<br.x+br.w&&s.y>br.y&&s.y<br.y+br.h){ hitBrick(br,false); burst(s.x,s.y,'#7dff9e',4,120,300,2); dead=true; break; } }
    if(dead||s.y>H+22)eshots.splice(i,1);
  }

  // powerups
  for(let i=powerups.length-1;i>=0;i--){ const p=powerups[i]; p.t+=dt; p.y+=p.vy*dt; p.x=p.bx+Math.sin(p.t*3)*14;
    if(p.y>H+24){powerups.splice(i,1);continue;}
    if(p.y>pd.y-24&&p.y<pd.y+16&&Math.abs(p.x-pd.x)<pd.w/2+16){ applyPower(p.type); powerups.splice(i,1); }
  }

  // bricks housekeeping + wave check
  for(const br of bricks)br.flash=Math.max(0,br.flash-dt*4);
  if(bricks.some(b=>b.dead))bricks=bricks.filter(b=>!b.dead);
  if(bossLeft<=0&&S.mode==='playing')waveComplete();
}

  if(S.invuln>0&&Math.floor(S.time*18)%2)ctx.globalAlpha=.35;
  const g=ctx.createLinearGradient(0,pd.y-7,0,pd.y+7);
  g.addColorStop(0,'#eaf7ff'); g.addColorStop(.45,'#8fb7dd'); g.addColorStop(1,'#2c4a77');

  for(const s of [-1,1]){ ctx.fillStyle='rgba(87,230,255,'+pulse+')';
    ctx.beginPath(); ctx.arc(pd.x+s*(pd.w/2-9),pd.y,2.6,0,TAU); ctx.fill(); }

  if(powers.shield>0){ const a=.4+.25*Math.sin(S.time*5);

function drawParticles(){

      ctx.beginPath(); ctx.arc(p.x,p.y,p.r,0,TAU); ctx.stroke(); }
    else { ctx.save(); ctx.translate(p.x,p.y); ctx.rotate(p.rot); ctx.globalAlpha=a; ctx.fillStyle=p.color;
      ctx.fillRect(-p.size/2,-p.size/2,p.size,p.size); ctx.restore(); } }
  ctx.globalAlpha=1;
}

}

  if(ra>0){ const vg=ctx.createRadialGradient(W/2,H/2,Math.min(W,H)*.3,W/2,H/2,Math.max(W,H)*.72);

      drawPixelText('GAME OVER',W/2,H*0.28,s,'#ff5d7a','center'); ctx.shadowBlur=0; }
  }
}
