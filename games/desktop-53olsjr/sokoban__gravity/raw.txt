/* gravity: every crate falls to the lowest cell in its column */
function settle(delay){
  const occ=new Set(boxes.map(b=>b.x+","+b.y));
  const order=[...boxes].sort((a,b)=>b.y-a.y||a.x-b.x);
  let maxEnd=0;
  for(const b of order){
    occ.delete(b.x+","+b.y);
    let ly=b.y;
    while(ly+1<rows&&walkable(b.x,ly+1)&&!occ.has(b.x+","+ (ly+1)))ly++;
    if(ly>b.y){
      const t0=now()+delay,dur=clamp(95+(ly-b.y)*52,130,430);
      b.fall={fy:b.y,ty:ly,t0,dur};
      occ.add(b.x+","+ly);b.y=ly;maxEnd=Math.max(maxEnd,t0+dur);
      at(t0+dur-delay? t0+dur-now():1,()=>finishFall(b,ly));
      if(player.x===b.x&&player.y===ly&&!player.gone)at(t0+dur-now(),()=>crushNow());
    }}
  if(maxEnd){unlockAt=Math.max(unlockAt,maxEnd+30);at(maxEnd-now()+4,()=>checkWin());}
  return maxEnd||null;
}

function safeSpawn(){

  const q=[[spawn.x,spawn.y]],seen=new Set([spawn.x+","+spawn.y]);
  while(q.length){const[x,y]=q.shift();

  return spawn;
}

function drawParticles(dt){
  for(let i=particles.length-1;i>=0;i--){const p=particles[i];p.t+=dt;
    if(p.t>=p.life){particles.splice(i,1);continue;}
    const q=p.t/p.life,a=1-q;
    if(p.type==="conf"||p.type==="spark"||p.type==="puff"){p.x+=p.vx*dt;p.y+=p.vy*dt;p.vy+=(p.g||0)*dt;}
    if(p.type==="puff")ctx.fillStyle="rgba("+p.col[0]+","+p.col[1]+","+p.col[2]+","+(a*.5).toFixed(3)+")",
      ctx.beginPath(),ctx.arc(p.x,p.y,p.size*(1+q),0,7),ctx.fill();
    else if(p.type==="spark"){ctx.fillStyle="rgba("+p.col[0]+","+p.col[1]+","+p.col[2]+","+a.toFixed(3)+")";ctx.fillRect(p.x-p.size/2,p.y-p.size/2,p.size,p.size);}
    else if(p.type==="ring"){const r=lerp(p.r0,p.r1,q);ctx.strokeStyle="rgba("+p.col[0]+","+p.col[1]+","+p.col[2]+","+(a*.8).toFixed(3)+")";ctx.lineWidth=2.5;ctx.beginPath();ctx.arc(p.x,p.y,r,0,7);ctx.stroke();}
    else if(p.type==="conf"){p.rot+=p.vr*dt;ctx.save();ctx.translate(p.x,p.y);ctx.rotate(p.rot);ctx.fillStyle="rgba("+p.col[0]+","+p.col[1]+","+p.col[2]+","+a.toFixed(3)+")";ctx.fillRect(-p.w/2,-p.h/2,p.w,p.h);ctx.restore();}
  }
  for(let i=floats.length-1;i>=0;i--){const f=floats[i];f.t+=dt;
    if(f.t>=f.life){floats.splice(i,1);continue;}
    const q=f.t/f.life;
    ctx.font="400 "+Math.round(tile*.52)+"px Impact,'Arial Narrow',sans-serif";ctx.textAlign="center";
    ctx.globalAlpha=1-q;ctx.lineWidth=4;ctx.strokeStyle="rgba(5,12,18,.9)";
    ctx.strokeText(f.txt,f.x,f.y-q*tile*.5);ctx.fillStyle=f.col;ctx.fillText(f.txt,f.x,f.y-q*tile*.5);ctx.globalAlpha=1;}
}

function frame(t){
  requestAnimationFrame(frame);
  const dt=Math.min(.05,(t-lastNow)/1000);lastNow=t;
  // events
  for(let i=events.length-1;i>=0;i--)if(t>=events[i].t){const fn=events[i].fn;events.splice(i,1);fn();}
  // buffered input
  if(state==="play"&&!busy()&&bufferDir){const d=bufferDir;bufferDir=null;doMove(d[0],d[1]);}
  // jetpack puffs while sliding
  if(player.slide&&state==="play"){const p=(t-player.slide.t0)/player.slide.dur;
    if(p<1&&Math.random()<.5){particles.push({type:"puff",x:px(player.x)-player.face.dx*tile*.3+rnd(-2,2),y:py(player.y+ (player.face.dy>0?-.2:.2))-player.face.dy*tile*.3,vx:-player.face.dx*30,vy:-player.face.dy*30,g:0,life:.3,t:0,size:rnd(1.5,3),col:[89,214,255]});}}
  // ambient motes
  if((state==="play"||state==="menu")&&Math.random()<dt*.7){
    particles.push({type:"puff",x:ox+rnd(0,cols*tile),y:oy-6,vx:rnd(-6,6),vy:rnd(10,22),g:0,life:rnd(3,5),t:0,size:rnd(1,2),col:[140,200,220]});}
  if(shakeT>0)shakeT=Math.max(0,shakeT-dt);
  // ---- draw ----
  ctx.setTransform(DPR,0,0,DPR,0,0);
  drawBg(t,dt);
  let sx=0,sy=0;
  if(shakeT>0){const k=(shakeT/shakeDur)*shakeMag;sx=rnd(-k,k);sy=rnd(-k,k);}
  ctx.save();ctx.translate(sx,sy);
  if(scv)ctx.drawImage(scv,ox,oy,cols*tile,rows*tile);
  if(rows){for(const tp of targets){const p=tp.split(",");drawPad(+p[0],+p[1],t,!!boxAt(+p[0],+p[1]));}
    [...boxes].sort((a,b)=>(b.fall?1:0)-(a.fall?1:0)||a.x-b.x).forEach(b=>drawBox(b,t));
    drawPlayer(t);}
  drawParticles(dt);
  ctx.restore();
  if(vigCv)ctx.drawImage(vigCv,0,0,W,H);
}

    if(DIRS[k])bufferDir=DIRS[k];
    else if(k==="z"||k==="Z"||k==="Backspace")doUndo();
    else if(k==="r"||k==="R"){Sfx.click();loadLevel(idx);}
    else if(k==="Escape")toMenu();
  }else if(state==="clear"){
    if(k==="n"||k==="N"||k==="Enter")nextFromBanner();
    else if(k==="r"||k==="R"){Sfx.click();$("#banner").classList.add("hidden");loadLevel(idx);}
    else if(k==="Escape")toMenu();
  }else if(state==="complete"){
    if(k==="Enter"||k==="n"||k==="N"){Sfx.click();$("#banner").classList.add("hidden");loadLevel(0);}
    else if(k==="Escape")toMenu();
  }
});
function nextFromBanner(){
  Sfx.click();$("#banner").classList.add("hidden");
  if(idx+1<LEVELS.length)loadLevel(idx+1);else showDone();
}

  $("#sndLabel").textContent=save.muted?"MUTED":"SOUND";
  $("#bSound").classList.toggle("off",save.muted);
  if(!save.muted)Sfx.click();}
