    for(var i=0;i<n;i++) vs.push({x:((off+i*span/n)%span)-(d.len+1.5)});
    return {row:d.row,dir:d.dir,sp:d.sp*scale,len:d.len,col:d.col,span:span,vs:vs};
  });
}
function buildGhosts(){
  var n=[2,2,3,3,4][phase-1];
  ghosts=[];
  for(var i=0;i<n;i++) ghosts.push({x:1+Math.random()*(COLS-2),dir:Math.random()<0.5?-1:1,
    sp:(1.2+phase*0.25)*(0.9+Math.random()*0.3),col:GHOST_COLS[i%4],ph:Math.random()*7});
}
function buildPellets(){
  pellets=[];
  var rows=[1,2,3,4,5,7,8,9];
  for(var r=0;r<rows.length;r++)for(var c=0;c<COLS;c++) if(Math.random()<0.55) pellets.push({c:c,r:rows[r],t:1,alive:true});
  var placed=0,guard=0;
  while(placed<3&&guard++<200){
    var rr2=rows[(Math.random()*rows.length)|0], cc=(Math.random()*COLS)|0, p=null;
    for(var i=0;i<pellets.length;i++) if(pellets[i].c===cc&&pellets[i].r===rr2) p=pellets[i];
    if(!p){p={c:cc,r:rr2,t:2,alive:true};pellets.push(p);} else p.t=2;
    placed++;
  }
}
function initPhase(){
  buildLanes(); buildGhosts(); buildPellets();
  timeLeft=TIME_BY[phase-1]; powerT=0; ghostCombo=0; lastTickSec=-1;
}

/* ---------- gameplay ---------- */
function tryMove(d){
  var dx=d==='left'?-1:d==='right'?1:0, dy=d==='up'?-1:d==='down'?1:0;
  if(dy===-1&&player.y===1){
    if(PAD_COLS.indexOf(player.x)>=0){startPad();return true;}
    return false;
  }
  var nx=player.x+dx, ny=player.y+dy;
  if(nx<0||nx>=COLS||ny<1||ny>WATER) return false;
  player.fromX=player.fx; player.fromY=player.fy;
  player.x=nx; player.y=ny; player.hopT=0;
  player.rot = dx===1?0:dx===-1?Math.PI:dy===-1?-Math.PI/2:Math.PI/2;
  return true;
}
function stepHop(dt){
  if(player.hopT<1){
    player.hopT=Math.min(1,player.hopT+dt/0.09);
    var e=1-(1-player.hopT)*(1-player.hopT);
    player.fx=player.fromX+(player.x-player.fromX)*e;
    player.fy=player.fromY+(player.y-player.fromY)*e;
  }
}
function startPad(){
  var px=player.x, bonus=300+Math.ceil(timeLeft)*10;
  score+=bonus;
  player.fromX=player.fx; player.fromY=player.fy;
  player.y=0; player.hopT=0; player.rot=-Math.PI/2;
  state='stageclear'; clearT=1.9;
  var cx=ox+(px+0.5)*cell, cy=oy+cell*0.5;
  POP.push({x:cx,y:cy,txt:'+'+bonus,life:0.9,max:0.9,col:'#3ee6c1'});
  banner={txt:'STAGE '+phase+' CLEAR',sub:'BONUS +'+bonus,t:1.7,max:1.7,col:'#3ee6c1'};
  sfx.pad(); sfx.clear(); confettiAt(cx,cy,28);
}

}

}

/* ---------- eating & hazards ---------- */
function eatNearbyPellets(){
  var pc=Math.round(player.fx), pr=Math.round(player.fy);
  for(var i=0;i<pellets.length;i++){
    var p=pellets[i];
    if(!p.alive||p.c!==pc||p.r!==pr) continue;
    p.alive=false;
    var px=ox+(p.c+0.5)*cell, py=oy+(p.r+0.5)*cell;
    if(p.t===2){
      score+=50; powerT=6; ghostCombo=0; sfx.power();
      flashCol='#fff3b0'; flashA=0.3;
      POP.push({x:px,y:py,txt:'POWER!',life:0.8,max:0.8,col:'#ffdd57'});
      spawnBurst(px,py,'#ffdd57',14,160);
    }else{
      score+=10; wakaFlip=!wakaFlip; sfx.waka(wakaFlip?640:470);
      spawnBurst(px,py,'#ffd23f',6,90);
    }
  }
}
function checkHazards(){
  if(invuln>0) return;
  var fx=player.fx, fy=player.fy, i, j, v, L, g, a, b;
  for(i=0;i<lanes.length;i++){
    L=lanes[i];
    if(Math.abs(fy-L.row)>=0.48) continue;
    for(j=0;j<L.vs.length;j++){
      v=L.vs[j]; a=v.x+0.06; b=v.x+L.len-0.06;
      if(fx+0.34>a&&fx-0.34<b){
        if(powerT>0) eatCar(L,j); else {die('car');return;}
      }
    }
  }
  for(i=0;i<ghosts.length;i++){
    g=ghosts[i];
    if(Math.abs(fy-MEDIAN)>=0.48) continue;
    if(Math.abs(fx-g.x)<0.62){
      if(powerT>0) eatGhost(i); else {die('ghost');return;}
    }
  }
}
function eatCar(L,i){
  var v=L.vs[i]; L.vs.splice(i,1); score+=50;
  var x=ox+(v.x+L.len/2)*cell, y=oy+L.row*cell+cell/2;
  POP.push({x:x,y:y,txt:'+50',life:0.7,max:0.7,col:'#3ee6c1'});
  spawnBurst(x,y,L.col,14,180); spawnBurst(x,y,'#9aa4c8',8,120);
  sfx.carEat(); shake=Math.max(shake,0.3);
}
function eatGhost(i){
  var g=ghosts[i]; ghosts.splice(i,1); ghostCombo++;
  var pts=[200,400,800,1600][Math.min(3,ghostCombo-1)]; score+=pts;
  var x=ox+(g.x+0.5)*cell, y=oy+(MEDIAN+0.5)*cell;
  POP.push({x:x,y:y,txt:pts,life:0.8,max:0.8,col:'#ff8ae0'});
  spawnBurst(x,y,g.col,16,200); sfx.ghostEat(ghostCombo);
  flashCol='#d8f4ff'; flashA=0.25; shake=Math.max(shake,0.2);
}

    var a=Math.random()*Math.PI*2, s=spd*(0.4+Math.random()*0.8);

  }
}

    life:1+Math.random()*0.6,max:1.6,size:3+Math.random()*3,col:CONF[(Math.random()*CONF.length)|0],grav:380});

  P.push({x:ox+Math.random()*cell*COLS,y:oy-10,vx:(Math.random()-0.5)*60,vy:60+Math.random()*80,

/* ---------- update ---------- */
function updateWorld(dt){
  var i,j,v,L,g;
  for(i=0;i<lanes.length;i++){L=lanes[i];
    for(j=0;j<L.vs.length;j++){v=L.vs[j]; v.x+=L.dir*L.sp*dt;
      if(L.dir>0&&v.x>COLS+0.2) v.x-=L.span;
      if(L.dir<0&&v.x<-(L.len+0.2)) v.x+=L.span;}
  }
  for(i=0;i<ghosts.length;i++){g=ghosts[i];
    var sp=(powerT>0)?g.sp*0.45:g.sp;
    g.x+=g.dir*sp*dt;
    if(g.x<0.4){g.x=0.4;g.dir=1;} else if(g.x>COLS-1.4){g.x=COLS-1.4;g.dir=-1;}
    if(Math.random()<dt*0.25) g.dir*=-1;
  }
}
function updateFX(dt){
  var i;
  for(i=P.length-1;i>=0;i--){var p=P[i];p.x+=p.vx*dt;p.y+=p.vy*dt;p.vy+=p.grav*dt;p.life-=dt;if(p.life<=0)P.splice(i,1);}
  for(i=POP.length-1;i>=0;i--){var q=POP[i];q.y-=28*dt;q.life-=dt;if(q.life<=0)POP.splice(i,1);}
  shake=Math.max(0,shake-dt*1.6);
  flashA=Math.max(0,flashA-dt*1.2);
  if(banner){banner.t-=dt;if(banner.t<=0)banner=null;}
  for(i=0;i<FF.length;i++){var f=FF[i];f.y-=dt*0.012*f.s;f.x+=Math.sin(tGlobal*0.5+f.ph)*dt*0.01;if(f.y<-0.03){f.y=1.03;f.x=Math.random();}}
}
function updateState(dt){
  if(state==='playing'){
    moveCd-=dt;
    if(held.length&&moveCd<=0){ if(tryMove(held[held.length-1])) moveCd=0.105; }
    stepHop(dt);
    invuln=Math.max(0,invuln-dt);
    if(powerT>0){powerT-=dt;if(powerT<=0){powerT=0;ghostCombo=0;}}
    timeLeft-=dt;
    var sec=Math.ceil(timeLeft);
    if(sec!==lastTickSec&&sec<=10&&sec>=1){lastTickSec=sec;sfx.tick();}
    if(timeLeft<=0){die('time');return;}
    eatNearbyPellets();
    checkHazards();
  }else if(state==='dying'){
    stepHop(dt); deathT-=dt;
    if(deathT<=0){ if(lives<=0) gameover(); else respawn(); }
  }else if(state==='stageclear'){
    stepHop(dt); clearT-=dt;
    if(clearT<=0) nextStage();
  }else if(state==='victory'){
    if(Math.random()<dt*4) confettiDrop();
  }
}

function drawPellets(){
  for(var i=0;i<pellets.length;i++){

      ctx.fillStyle='rgba(255,210,63,.22)';ctx.beginPath();ctx.arc(x,y,cell*0.17,0,7);ctx.fill();

      var pr=cell*(0.16+0.04*Math.sin(tGlobal*6+p.c));

      ctx.fillStyle='#ffdd57';ctx.beginPath();ctx.arc(x,y,pr,0,7);ctx.fill();
    }
  }
}

  }
}
