  ship={x:W*.16,y:H*.5,vx:0,vy:0,a:-Math.PI/2,hull:100,cpower:100,cloaked:false,inv:2,cd:0};
  asteroids=[]; let tries=0;
  while(asteroids.length<11&&tries++<500){
    const size=asteroids.length<7?3:(Math.random()<.5?2:1);
    const x=rand(0,W),y=rand(0,H);
    if(dist(x,y,ship.x,ship.y)<210)continue;
    let ok=true;
    for(const a of asteroids){ if(dist(x,y,a.x,a.y)<a.r+AR[size]+10){ok=false;break;} }
    if(ok)asteroids.push(makeAsteroid(x,y,size));
  }
  drones=[];
  for(let i=0;i<3;i++)drones.push({cxn:rand(.35,.65),cyn:rand(.35,.65),axn:rand(.2,.32),ayn:rand(.2,.32),
    fx:rand(.18,.34),fy:rand(.18,.34),p1:rand(0,TAU),p2:rand(0,TAU),sw:rand(.5,.9),sp:rand(0,TAU),
    x:0,y:0,face:rand(0,TAU),cone:0,pingT:rand(0,2.8),scanning:false,shots:[],dead:false});
  shards=[];
  for(let i=0;i<3;i++){ let x=W*.5,y=H*.5,tr=0;
    do{ x=rand(W*.15,W*.9); y=rand(H*.15,H*.85); tr++; }
    while(tr<300&&(dist(x,y,ship.x,ship.y)<260||shards.some(s=>dist(x,y,s.x,s.y)<170)));
    shards.push({x,y,ph:rand(0,TAU),taken:false});
  }
  gate.side=irand(0,3); gate.f=rand(.25,.75);
}

function spawnParts(x,y,n,col,sp,l0,sz){ for(let i=0;i<n;i++){ const a=rand(0,TAU),s=rand(sp*.25,sp);
  parts.push({x,y,vx:Math.cos(a)*s,vy:Math.sin(a)*s,l:rand(l0*.4,l0),ml:l0,sz:rand(sz*.5,sz),col}); } }

function stepAsteroids(dt){ for(const a of asteroids){ a.x+=a.vx*dt;a.y+=a.vy*dt;
  a.x=wrapN(a.x,a.r,W); a.y=wrapN(a.y,a.r,H); a.rot+=a.rv*dt; } }
function dronePos(d,tt){ return { x:(d.cxn+d.axn*Math.sin(tt*d.fx+d.p1))*W, y:(d.cyn+d.ayn*Math.cos(tt*d.fy+d.p2))*H }; }
function stepDrones(dt){ for(const d of drones){ if(d.dead)continue;
  const p=dronePos(d,t),p2=dronePos(d,t+.06);
  const target=(state==='ALARM')?Math.atan2(ship.y-p.y,ship.x-p.x):Math.atan2(p2.y-p.y,p2.x-p.x);
  d.face=lerpA(d.face,target,1-Math.exp(-3*dt));
  d.cone=d.face+.55*Math.sin(t*d.sw+d.sp);
  d.x=p.x; d.y=p.y;
  d.pingT+=dt; if(d.pingT>2.8){d.pingT=0;rings.push({x:d.x,y:d.y,r:12,max:95,col:'255,178,90'});} } }
function stepFx(dt){ for(let i=parts.length-1;i>=0;i--){const p=parts[i];
  p.x+=p.vx*dt;p.y+=p.vy*dt; const dm=Math.exp(-2.2*dt); p.vx*=dm;p.vy*=dm; p.l-=dt; if(p.l<=0)parts.splice(i,1);}
  for(let i=rings.length-1;i>=0;i--){const r=rings[i]; r.r+=dt*95; if(r.r>r.max)rings.splice(i,1);}
  const svx=ship?ship.vx*.04:0, svy=ship?ship.vy*.04:0;
  for(const s of stars){ s.x-=svx*s.z*dt; s.y-=svy*s.z*dt; s.x=wrapN(s.x,0,W); s.y=wrapN(s.y,0,H); } }

function updShip(dt){ const s=ship;
  if(keys.ArrowLeft||keys.KeyA)s.a-=3.8*dt;
  if(keys.ArrowRight||keys.KeyD)s.a+=3.8*dt;
  // cloak
  const want=(keys.ShiftLeft||keys.ShiftRight)&&s.cpower>8;
  if(want&&!s.cloaked){s.cloaked=true;sfxCloak(true);rings.push({x:s.x,y:s.y,r:6,max:44,col:'180,150,255'});}
  if(!want&&s.cloaked){s.cloaked=false;if(s.cpower>0)sfxCloak(false);}
  if(s.cloaked){ s.cpower-=30*dt; if(s.cpower<=0){s.cpower=0;s.cloaked=false;sfxCloak(false);} }
  else s.cpower=Math.min(100,s.cpower+14*dt);
  // thrust
  thrusting=!!(keys.ArrowUp||keys.KeyW||keys.Space);
  if(thrusting){ const acc=s.cloaked?200:360;
    s.vx+=Math.cos(s.a)*acc*dt; s.vy+=Math.sin(s.a)*acc*dt;
    for(let i=0;i<2;i++)parts.push({x:s.x-Math.cos(s.a)*14+rand(-3,3),y:s.y-Math.sin(s.a)*14+rand(-3,3),
      vx:-Math.cos(s.a)*rand(60,140)+s.vx*.4,vy:-Math.sin(s.a)*rand(60,140)+s.vy*.4,
      l:rand(.2,.45),ml:.45,sz:rand(1,2.4),col:s.cloaked?'170,140,255':'255,170,90'}); }
  const vmax=s.cloaked?175:350, sp=Math.hypot(s.vx,s.vy);
  if(sp>vmax){s.vx*=vmax/sp;s.vy*=vmax/sp;}
  const damp=Math.exp(-.55*dt); s.vx*=damp; s.vy*=damp;
  s.x+=s.vx*dt; s.y+=s.vy*dt;
  s.x=wrapN(s.x,14,W); s.y=wrapN(s.y,14,H);
  s.inv-=dt; s.cd-=dt;
  // fire
  if((keys.KeyX||keys.KeyJ||keys.KeyF||keys.MOUSE)&&s.cd<=0){
    if(ammo>0){ ammo--; s.cd=.22;
      bullets.push({x:s.x+Math.cos(s.a)*16,y:s.y+Math.sin(s.a)*16,
        vx:s.vx+Math.cos(s.a)*540,vy:s.vy+Math.sin(s.a)*540,life:1.05});
      parts.push({x:s.x+Math.cos(s.a)*20,y:s.y+Math.sin(s.a)*20,vx:0,vy:0,l:.12,ml:.12,sz:3,col:'190,255,255'});
      sfxFire(); exposure=clamp(exposure+6,0,100);
      if(s.cloaked){s.cloaked=false;sfxCloak(false);} }
    else { s.cd=.28; sfxEmpty(); } }
}

function splitAsteroid(j){ const a=asteroids[j];
  spawnParts(a.x,a.y,10,'150,175,200',140,.5,2); sfxRock();
  if(a.size>1)for(let k=0;k<2;k++){ const na=makeAsteroid(a.x,a.y,a.size-1);
    na.vx+=a.vx*.6+rand(-30,30); na.vy+=a.vy*.6+rand(-30,30); asteroids.push(na); }
  asteroids.splice(j,1);
}
function killDrone(d){ d.dead=true; kills++;
  spawnParts(d.x,d.y,22,'255,92,138',220,.7,2.4); sfxBoom(false); shake=Math.max(shake,5); }

function updBullets(dt){ for(let i=bullets.length-1;i>=0;i--){ const b=bullets[i];
  b.x+=b.vx*dt;b.y+=b.vy*dt; b.x=wrapN(b.x,4,W); b.y=wrapN(b.y,4,H); b.life-=dt;
  if(b.life<=0){bullets.splice(i,1);continue;}
  let hit=false;
  for(let j=asteroids.length-1;j>=0;j--){ const a=asteroids[j];
    if(dist(b.x,b.y,a.x,a.y)<a.r+3){ splitAsteroid(j); hit=true; break; } }
  if(!hit)for(const d of drones){ if(!d.dead&&dist(b.x,b.y,d.x,d.y)<17){ killDrone(d); hit=true; break; } }
  if(hit)bullets.splice(i,1); } }

function damage(n,nx,ny){ const s=ship; if(s.inv>0)return;
  s.hull-=n; s.inv=1.5; shake=Math.max(shake,8);
  s.vx=nx*270+rand(-30,30); s.vy=ny*270+rand(-30,30);
  spawnParts(s.x,s.y,14,'160,220,255',190,.5,2.2); sfxHit();
  if(s.hull<=0){ s.hull=0; startDying(); } }
function startDying(){ state='DYING'; deathT=1.5; shake=12; sfxBoom(true);
  spawnParts(ship.x,ship.y,30,'110,243,255',260,.8,2.6);
  for(let i=0;i<7;i++){ const a=rand(0,TAU);
    frags.push({x:ship.x,y:ship.y,vx:ship.vx+Math.cos(a)*rand(40,160),vy:ship.vy+Math.sin(a)*rand(40,160),
      a:rand(0,TAU),va:rand(-4,4),len:rand(8,20)}); } }

function startAlarm(){ state='ALARM'; alarmT=1.7; sfxAlarm(); shake=7;
  drones.forEach((d,i)=>{ d.shots=[t+.3+i*.12,t+.9+i*.1]; }); }
function droneShot(d){ const lead=.28, tx=ship.x+ship.vx*lead, ty=ship.y+ship.vy*lead;
  const a=Math.atan2(ty-d.y,tx-d.x);
  ebullets.push({x:d.x+Math.cos(a)*16,y:d.y+Math.sin(a)*16,vx:Math.cos(a)*430,vy:Math.sin(a)*430,life:1.2});
  sfxEShot(); }

function updPlay(dt){
  missionT+=dt;
  updShip(dt); stepAsteroids(dt); stepDrones(dt); updBullets(dt);
  // detection
  scannedNow=false; const cl=coneLen();
  for(const d of drones){ d.scanning=false; if(d.dead||ship.cloaked)continue;
    const dx=ship.x-d.x, dy=ship.y-d.y, dd=Math.hypot(dx,dy);
    if(dd<cl+14&&Math.abs(angDiff(Math.atan2(dy,dx),d.cone))<.5+12/dd){ d.scanning=true; scannedNow=true; } }
  if(scannedNow)exposure+=(13+(thrusting?30:0))*dt;
  else exposure-=(ship.cloaked?16:9)*dt;
  exposure=clamp(exposure,0,100); maxExp=Math.max(maxExp,exposure);
  // collisions
  if(state==='PLAYING'){
    for(const a of asteroids){ const dd=dist(ship.x,ship.y,a.x,a.y);
      if(dd<a.r+12&&ship.inv<=0){ damage(34,(ship.x-a.x)/dd,(ship.y-a.y)/dd); } }
    for(const d of drones){ if(d.dead)continue; const dd=dist(ship.x,ship.y,d.x,d.y);
      if(dd<25&&ship.inv<=0){ damage(40,(ship.x-d.x)/(dd||1),(ship.y-d.y)/(dd||1)); } }
    // shards
    for(const s of shards){ if(s.taken)continue;
      if(Math.random()<dt*3)parts.push({x:s.x+rand(-8,8),y:s.y+rand(-8,8),vx:rand(-15,15),vy:rand(-15,15),l:.4,ml:.4,sz:1.4,col:'255,209,102'});
      if(dist(ship.x,ship.y,s.x,s.y)<26){ s.taken=true; shardsGot++; ammo=Math.min(40,ammo+8);
        sfxShard(); spawnParts(s.x,s.y,18,'255,209,102',180,.6,2.2);
        rings.push({x:s.x,y:s.y,r:8,max:70,col:'255,209,102'});
        if(shardsGot>=3){ gateActive=true; sfxGate(); const g=edgePos();
          rings.push({x:g.x,y:g.y,r:10,max:280,col:'89,255,160'}); } } }
    // win
    if(gateActive){ const g=edgePos();
      if(dist(ship.x,ship.y,g.x,g.y)<54){ state='WARP'; warpT=warpMax; sfxWarp(); } } }
  stepFx(dt);
  if(state==='PLAYING'&&exposure>=100)startAlarm();
}

function updAlarm(dt){ alarmT-=dt;
  updShip(dt); stepAsteroids(dt); stepDrones(dt); updBullets(dt);
  for(const d of drones){ if(d.dead||!d.shots.length)continue;
    while(d.shots.length&&t>=d.shots[0]){ d.shots.shift(); droneShot(d); } }
  for(let i=ebullets.length-1;i>=0;i--){ const b=ebullets[i];
    b.x+=b.vx*dt;b.y+=b.vy*dt; b.life-=dt;
    if(b.life<=0){ebullets.splice(i,1);continue;}
    if(dist(b.x,b.y,ship.x,ship.y)<14&&ship.inv<=0){
      damage(24,-b.vx/430,-b.vy/430); ebullets.splice(i,1); } }
  stepFx(dt);
  if(alarmT<=0)endGame('detected');
}
function updDying(dt){ deathT-=dt;
  for(const f of frags){ f.x+=f.vx*dt;f.y+=f.vy*dt;f.a+=f.va*dt;
    const dm=Math.exp(-.5*dt);f.vx*=dm;f.vy*=dm;f.x=wrapN(f.x,0,W);f.y=wrapN(f.y,0,H);} 
  for(let i=ebullets.length-1;i>=0;i--){const b=ebullets[i];b.x+=b.vx*dt;b.y+=b.vy*dt;b.life-=dt;if(b.life<=0)ebullets.splice(i,1);}
  stepFx(dt); if(deathT<=0)endGame('hull');
}
function updWarp(dt){ warpT-=dt;
  for(let i=0;i<4;i++){ const a=rand(0,TAU),r=Math.max(W,H)*.6;
    parts.push({x:ship.x+Math.cos(a)*r,y:ship.y+Math.sin(a)*r,
      vx:-Math.cos(a)*r*2.4,vy:-Math.sin(a)*r*2.4,l:.35,ml:.35,sz:1.6,col:'110,243,255'}); }
  stepFx(dt); if(warpT<=0)endGame('win');
}

    const base=hot?'255,77,94':'255,178,90';
    const al=(.16+.07*Math.sin(t*3+d.sp)+(hot?.12:0)).toFixed(3);
    g.addColorStop(0,`rgba(${base},${al})`); g.addColorStop(1,`rgba(${base},0)`);
    ctx.fillStyle=g;
    ctx.beginPath(); ctx.moveTo(d.x,d.y);
    ctx.arc(d.x,d.y,cl,d.cone-half,d.cone+half); ctx.closePath(); ctx.fill();
    ctx.strokeStyle=`rgba(${base},${(.22+.1*Math.sin(t*3+d.sp)).toFixed(3)})`; ctx.lineWidth=1;
    ctx.beginPath(); ctx.arc(d.x,d.y,cl,d.cone-half,d.cone+half); ctx.stroke();
    const pr=(t*110+d.sp*40)%cl; // sweeping scan band
    ctx.strokeStyle=`rgba(${base},${(.4*(1-pr/cl)).toFixed(3)})`; ctx.lineWidth=2;
    ctx.beginPath(); ctx.arc(d.x,d.y,pr,d.cone-half,d.cone+half); ctx.stroke();
    if(hot){ ctx.setLineDash([4,6]); ctx.lineDashOffset=-t*60;
      ctx.strokeStyle='rgba(255,77,94,.5)'; ctx.lineWidth=1;
      ctx.beginPath();ctx.moveTo(d.x,d.y);ctx.lineTo(ship.x,ship.y);ctx.stroke(); ctx.setLineDash([]); } }

  // drones
  for(const d of drones){ if(d.dead)continue;
    ctx.save(); ctx.translate(d.x,d.y); ctx.rotate(d.face);
    ctx.shadowColor='#ff5c8a'; ctx.shadowBlur=10; ctx.strokeStyle='#ff5c8a'; ctx.lineWidth=1.6;
    ctx.beginPath();
    for(let i=0;i<6;i++){ const a=i/6*TAU; const px=Math.cos(a)*13,py=Math.sin(a)*13;
      i?ctx.lineTo(px,py):ctx.moveTo(px,py); }
    ctx.closePath(); ctx.stroke();
    ctx.rotate(t*6+d.sp);
    ctx.beginPath();ctx.moveTo(-8,0);ctx.lineTo(8,0);ctx.moveTo(0,-8);ctx.lineTo(0,8);ctx.stroke();
    ctx.shadowBlur=0;
    ctx.fillStyle=`rgba(255,220,235,${(.4+.6*Math.abs(Math.sin(t*4+d.sp))).toFixed(3)})`;
    ctx.fillRect(-1.5,-1.5,3,3);
    ctx.restore(); }

  // ship & fragments
  if(state==='PLAYING'||state==='ALARM')drawShip();
  if(state==='WARP'){ const p=1-warpT/warpMax;
    ctx.save(); ctx.translate(ship.x,ship.y); ctx.rotate(ship.a);
    ctx.scale(1+p*5,1+p*5); ctx.globalAlpha=Math.max(0,1-p);
    ctx.shadowColor='#6ef3ff'; ctx.shadowBlur=20; ctx.strokeStyle='#eafcff'; ctx.lineWidth=2;
    ctx.beginPath();ctx.moveTo(18,0);ctx.lineTo(-11,9);ctx.lineTo(-6.5,0);ctx.lineTo(-11,-9);ctx.closePath();ctx.stroke();
    ctx.restore(); ctx.globalAlpha=1; }
  if(state==='DYING'){ const al=clamp(deathT/1.5,0,1);
    ctx.strokeStyle=`rgba(110,243,255,${al.toFixed(3)})`; ctx.lineWidth=2;
    for(const f of frags){ ctx.save();ctx.translate(f.x,f.y);ctx.rotate(f.a);
      ctx.beginPath();ctx.moveTo(-f.len/2,0);ctx.lineTo(f.len/2,0);ctx.stroke();ctx.restore(); } }

  // glow pass: particles + bullets
  ctx.save(); ctx.globalCompositeOperation='lighter';
  for(const p of parts){ const a=clamp(p.l/p.ml,0,1);
    ctx.fillStyle=`rgba(${p.col},${a.toFixed(3)})`;
    ctx.fillRect(p.x-p.sz/2,p.y-p.sz/2,p.sz,p.sz); }
  ctx.lineWidth=2; ctx.strokeStyle='rgba(160,250,255,.9)';
  for(const b of bullets){ ctx.beginPath();ctx.moveTo(b.x-b.vx*.02,b.y-b.vy*.02);ctx.lineTo(b.x,b.y);ctx.stroke(); }
  ctx.lineWidth=2.5; ctx.strokeStyle='rgba(255,90,77,.95)';
  for(const b of ebullets){ ctx.beginPath();ctx.moveTo(b.x-b.vx*.03,b.y-b.vy*.03);ctx.lineTo(b.x,b.y);ctx.stroke(); }
  ctx.restore();

  // danger tints
  if(state==='ALARM'){ ctx.fillStyle=`rgba(255,40,60,${(.1+.07*Math.sin(t*28)).toFixed(3)})`; ctx.fillRect(-40,-40,W+80,H+80); }
  else if(state==='PLAYING'&&exposure>60){ const a=(exposure-60)/40*.16*(.6+.4*Math.sin(t*6));
    const g=ctx.createRadialGradient(W/2,H/2,Math.min(W,H)*.3,W/2,H/2,Math.max(W,H)*.7);
    g.addColorStop(0,'rgba(255,77,94,0)'); g.addColorStop(1,`rgba(255,77,94,${a.toFixed(3)})`);
    ctx.fillStyle=g; ctx.fillRect(0,0,W,H); }
  ctx.restore();
}
