Projection helpers:
```js
function project(p) { // p is world coords; returns null if behind
  // to camera space
  let x = p.x - cam.pos.x, y = p.y - cam.pos.y, z = p.z - cam.pos.z;
  // rotate by view basis (right, up, fwd as rows)
  const cx = dot(x,y,z, cam.right), cy = dot(...,cam.up), cz = dot(...,cam.fwd);
  if (cz < NEAR) return null;
  const s = focal / cz;
  return { x: W/2 + cx*s, y: H/2 - cy*s, s };
}
```

```html
function setCam(pos,look,roll,dt){
  let vx=(pos[0]-prevCam[0]),vy=(pos[1]-prevCam[1]),vz=(pos[2]-prevCam[2]);
  CAMV=[wdelta(vx,0,WX)/Math.max(dt,0.001),vy/Math.max(dt,0.001),vz/Math.max(dt,0.001)];
  prevCam=pos.slice();
  let fx=look[0]-pos[0],fy=look[1]-pos[1],fz=look[2]-pos[2];
  let l=Math.hypot(fx,fy,fz)||1;fx/=l;fy/=l;fz/=l;
  let rx=-fz,ry=0,rz=fx;if(rx*rx+rz*rz<1e-6){rx=1;rz=0;}
  const rl=Math.hypot(rx,rz);rx/=rl;rz/=rl;
  let ux=ry*fz-rz*fy,uy=rz*fx-rx*fz,uz=rx*fy-ry*fx;
  if(roll){const c=Math.cos(roll),s=Math.sin(roll),omc=1-c;
    const rot=(v)=>{const d=v[0]*fx+v[1]*fy+v[2]*fz;
      const cx=fy*v[2]-fz*v[1],cy=fz*v[0]-fx*v[2],cz=fx*v[1]-fy*v[0];
      return[v[0]*c+cx*s+fx*d*omc,v[1]*c+cy*s+fy*d*omc,v[2]*c+cz*s+fz*d*omc];};
    const rr=rot([rx,ry,rz]),uu=rot([ux,uy,uz]);rx=rr[0];ry=rr[1];rz=rr[2];ux=uu[0];uy=uu[1];uz=uu[2];}
  BR0=rx;BR1=ry;BR2=rz;BU0=ux;BU1=uy;BU2=uz;BF0=fx;BF1=fy;BF2=fz;
  BLx=LIGHT[0]*rx+LIGHT[1]*ry+LIGHT[2]*rz;BLy=LIGHT[0]*ux+LIGHT[1]*uy+LIGHT[2]*uz;BLz=LIGHT[0]*fx+LIGHT[1]*fy+LIGHT[2]*fz;
  const na=Math.atan2(fz,fx);let dy=na-viewYaw;dy-=TAU*Math.round(dy/TAU);viewYaw+=dy;
}

function transformShip(){
  const f=[Math.cos(ship.yaw),0,Math.sin(ship.yaw)],rgt=[-f[2],0,f[0]];
  const cp=Math.cos(ship.pitch),sp=Math.sin(ship.pitch),cb=Math.cos(ship.bank),sb=Math.sin(ship.bank);
  const dx=wdelta(ship.pos[0],cam.pos[0],WX),dy=wdelta(ship.pos[1],cam.pos[1],WY),dz=wdelta(ship.pos[2],cam.pos[2],WZ);
  for(let i=0;i<SLV.length;i++){
    const x=SLV[i][0],y=SLV[i][1],z=SLV[i][2];
    const xr=x*cp-y*sp,yr=x*sp+y*cp;
    const y2=yr*cb-z*sb,z2=yr*sb+z*cb;
    shipW[i*3]=dx+f[0]*xr+rgt[2]*z2;
    shipW[i*3+1]=dy+y2+rgt? 0:0; // placeholder replaced below
    shipW[i*3+1]=dy+y2;
    shipW[i*3+2]=dz+f[2]*xr+rgt[2]*z2;
  }
}

function pushFace(z,x0,y0,x1,y1,x2,y2,col,al){
  if(fCount>=FCAP)return;const i=fCount++;fz[i]=z;const o=i*6;

function drawRockFaces(r){
  const dx=wdelta(r.pos[0],cam.pos[0],WX),dy=wdelta(r.pos[1],cam.pos[1],WY),dz=wdelta(r.pos[2],cam.pos[2],WZ);
  const czc=dx*BF0+dy*BF1+dz*BF2;
  if(czc<NEAR+r.radius||czc>FAR)return;
  const c=Math.cos(r.ang),sn=Math.sin(r.ang),omc=1-c,kx=r.axis[0],ky=r.axis[1],kz=r.axis[2],sc=r.scale;
  const V=r.mesh.V,pts=r.pts;
  for(let i=0,nv=V.length;i<nv;i++){
    const vx=V[i][0]*sc,vy=V[i][1]*sc,vz=V[i][2]*sc;
    const d=kx*vx+ky*vy+kz*vz;
    const cxr=ky*vz-kz*vy,cyr=kz*vx-kx*vz,czr=kx*vy-ky*vx;
    pts[i*3]=dx+vx*c+cxr*sn+kx*d*omc;
    pts[i*3+1]=dy+vy*c+cyr*sn+ky*d*omc;
    pts[i*3+2]=dz+vz*c+czr*sn+kz*d*omc;
  }
  const F=r.mesh.F,sh=r.shades;
  for(let i=0;i<F.length;i++){
    const f=F[i],a=f[0]*3,b=f[1]*3,c2=f[2]*3;
    let x0=pts[a]*BR0+pts[a+1]*BR1+pts[a+2]*BR2, y0=pts[a]*BU0+pts[a+1]*BU1+pts[a+2]*BU2, z0=pts[a]*BF0+pts[a+1]*BF1+pts[a+2]*BF2;
    let x1=pts[b]*BR0+pts[b+1]*BR1+pts[b+2]*BR2, y1=pts[b]*BU0+pts[b+1]*BU1+pts[b+2]*BU2, z1=pts[b]*BF0+pts[b+1]*BF1+pts[b+2]*BF2;
    let x2=pts[c2]*BR0+pts[c2+1]*BR1+pts[c2+2]*BR2,y2=pts[c2]*BU0+pts[c2+1]*BU1+pts[c2+2]*BU2,z2=pts[c2]*BF0+pts[c2+1]*BF1+pts[c2+2]*BF2;
    if(z0<NEAR||z1<NEAR||z2<NEAR)continue;
    const nx=(y1-y0)*(z2-z0)-(z1-z0)*(y2-y0);
    const ny=(z1-z0)*(x2-x0)-(x1-x0)*(z2-z0);
    const nz=(x1-x0)*(y2-y0)-(y1-y0)*(x2-x0);
    if(nx*(x0+x1+x2)+ny*(y0+y1+y2)+nz*(z0+z1+z2)>=0)continue;
    const il=1/Math.sqrt(nx*nx+ny*ny+nz*nz)||1;
    let bi=Math.floor((0.22+0.78*Math.max(0,(nx*BLx+ny*BLy+nz*BLz)*il))*7.99);if(bi>7)bi=7;
    const z=(z0+z1+z2)/3,k=focal/z;
    pushFace(z,W2+x0*k,H2-y0*k,W2+x1*k,H2-y1*k,W2+x2*k,H2-y2*k,sh[bi],1);
  }
}

function updThrustSnd(){
  if(!AC||!thrustGain)return;
  const sp=Math.hypot(ship.vel[0],ship.vel[1],ship.vel[2]);
  const g=(state==='play'&&ship.alive&&keys.U)?(0.12+Math.min(0.18,sp*0.0005)):0;
  thrustGain.gain.setTargetAtTime(g,AC.currentTime,0.08);
  if(thrustFilt)thrustFilt.frequency.value=130+sp*0.45;
}
function update(dt){
  shake*=Math.exp(-dt*5);if(shake<0.05)shake=0;
  dmgFlash=Math.max(0,dmgFlash-dt*1.6);
  updThrustSnd();
  for(const r of rocks){
    r.pos[0]+=r.vel[0]*dt;r.pos[1]+=r.vel[1]*dt;r.pos[2]+=r.vel[2]*dt;
    wrapPos(r.pos,WX,WY,WZ);r.ang+=r.w*dt;
  }
  if(state==='menu'){
    menuAng+=dt*0.14;const R=235;
    setCam([Math.cos(menuAng)*R,62+Math.sin(menuAng*0.7)*30,Math.sin(menuAng)*R],[0,-8,0],Math.sin(menuAng*0.5)*0.08,dt);
    updateFx(dt);return;
  }
  if(state==='over'){
    const f=[Math.cos(ship.yaw),0,Math.sin(ship.yaw)];
    cam.pos[0]+=f[0]*14*dt;cam.pos[2]+=f[2]*14*dt;wrapPos(cam.pos,WX,WY,WZ);
    setCam(cam.pos,[cam.pos[0]+f[0]*60,cam.pos[1],cam.pos[2]+f[2]*60],0,dt);
    updateFx(dt);return;
  }
  if(state!=='play')return;

  /* --- ship control --- */
  if(ship.alive){
    const f=[Math.cos(ship.yaw),0,Math.sin(ship.yaw)];
    curTurn=(keys.R?1:0)-(keys.L?1:0);
    ship.yaw+=curTurn*2.7*dt;
    const bt=curTurn*0.6,pt=Math.max(-0.5,Math.min(0.5,ship.vel[1]*0.004));
    ship.bank+=(bt-ship.bank)*(1-Math.exp(-dt*7));
    ship.pitch+=(pt-ship.pitch)*(1-Math.exp(-dt*7));
    const th=keys.U?1:0;ship.thrustOn=!!th;
    if(th){ship.vel[0]+=f[0]*400*dt;ship.vel[2]+=f[2]*400*dt;}
    if(keys.D){
      const vx=ship.vel[0],vz=ship.vel[2],sp=Math.hypot(vx,vz);
      if(sp>2){const d=320*dt/sp;ship.vel[0]-=vx*d;ship.vel[2]-=vz*d;}
      else{ship.vel[0]-=f[0]*170*dt;ship.vel[2]-=f[2]*170*dt;}
    }
    const vv=(keys.E?1:0)-(keys.Q?1:0);if(vv)ship.vel[1]+=vv*320*dt;
    const dh=Math.exp(-dt*0.55);ship.vel[0]*=dh;ship.vel[2]*=dh;ship.vel[1]*=Math.exp(-dt*1.7);
    const sp=Math.hypot(ship.vel[0],ship.vel[1],ship.vel[2]);
    if(sp>340){const d=340/sp;ship.vel[0]*=d;ship.vel[1]*=d;ship.vel[2]*=d;}
    ship.pos[0]+=ship.vel[0]*dt;ship.pos[1]+=ship.vel[1]*dt;ship.pos[2]+=ship.vel[2]*dt;
    wrapPos(ship.pos,WX,WY,WZ);
    ship.inv=Math.max(0,ship.inv-dt);
    ship.cd-=dt;
    if(keys.F&&ship.cd<=0){
      ship.cd=0.14;lastShot=time;
      const dx=f[0]+rnd(-0.006,0.006),dy=rnd(-0.006,0.006),dz=f[2]+rnd(-0.006,0.006);
      const l=Math.hypot(dx,dy,dz)||1;
      lasers.push({pos:[ship.pos[0]+f[0]*3.4,ship.pos[1]+0.2,ship.pos[2]+f[2]*3.4],dir:[dx/l,dy/l,dz/l],age:0});
      sShoot();
    }
    if(ship.inv<=0){
      for(let i=rocks.length-1;i>=0;i--){
        const r=rocks[i],dx=wdelta(ship.pos[0],r.pos[0],WX),dy=wdelta(ship.pos[1],r.pos[1],WY),dz=wdelta(ship.pos[2],r.pos[2],WZ);
        const rr=r.radius*0.92+3;
        if(dx*dx+dy*dy+dz*dz<rr*rr){killShip();break;}
      }
    }
  }else{
    ship.deadT-=dt;
    if(ship.deadT<=0){if(lives>0)respawn();else doGameOver();}
  }

  /* --- lasers --- */
  for(let i=lasers.length-1;i>=0;i--){
    const l=lasers[i];l.age+=dt;
    if(l.age>0.5){lasers.splice(i,1);continue;}
    l.pos[0]+=l.dir[0]*1500*dt;l.pos[1]+=l.dir[1]*1500*dt;l.pos[2]+=l.dir[2]*1500*dt;
    wrapPos(l.pos,WX,WY,WZ);
    let hit=false;
    for(let j=rocks.length-1;j>=0;j--){
      const r=rocks[j];
      const dx=wdelta(l.pos[0],r.pos[0],WX),dy=wdelta(l.pos[1],r.pos[1],WY),dz=wdelta(l.pos[2],r.pos[2],WZ);
      const t2=dx*l.dir[0]+dy*l.dir[1]+dz*l.dir[2];
      if(t2<-4||t2>820)continue;
      const tt=t2<0?0:t2,qx=dx-l.dir[0]*tt,qy=dy-l.dir[1]*tt,qz=dz-l.dir[2]*tt;
      const rr=r.radius+1.6;
      if(qx*qx+qy*qy+qz*qz<rr*rr){
        boom([l.pos[0]+l.dir[0]*tt,l.pos[1]+l.dir[1]*tt,l.pos[2]+l.dir[2]*tt],6,ROCK_BOOM,80,1.4);
        hitRock(j);hit=true;break;
      }
    }
    if(hit)lasers.splice(i,1);
  }

  /* --- waves --- */
  if(rocks.length===0&&clearT<0){clearT=2.4;if(wave>1)sWave();}
  if(clearT>0){clearT-=dt;if(clearT<=0){wave++;spawnWave(wave);banner('WAVE '+pad(wave,2),'HOSTILES INBOUND');}}

  /* --- camera --- */
  const f=[Math.cos(ship.yaw),0,Math.sin(ship.yaw)];
  if(ship.alive){
    const des=[ship.pos[0]-f[0]*15,ship.pos[1]+7,ship.pos[2]-f[2]*15];
    const k=1-Math.exp(-dt*9);
    cam.pos[0]+=wdelta(des[0],cam.pos[0],WX)*k;
    cam.pos[1]+=wdelta(des[1],cam.pos[1],WY)*k;
    cam.pos[2]+=wdelta(des[2],cam.pos[2],WZ)*k;
    wrapPos(cam.pos,WX,WY,WZ);
    const look=[ship.pos[0]+f[0]*14,ship.pos[1]+1.5,ship.pos[2]+f[2]*14];
    camRoll+=((curTurn*0.32)-camRoll)*(1-Math.exp(-dt*6));
    setCam([cam.pos[0]+rnd(-1,1)*shake,cam.pos[1]+rnd(-1,1)*shake,cam.pos[2]+rnd(-1,1)*shake],look,camRoll,dt);
  }else{
    setCam(cam.pos,[cam.pos[0]+f[0]*30,cam.pos[1],cam.pos[2]+f[2]*30],0,dt);
  }
  updateFx(dt);
}

/* ============================== render ============================== */
function render(){
  ctx.globalAlpha=1;ctx.globalCompositeOperation='source-over';
  let offX=(-viewYaw*42)%W;if(offX<0)offX+=W;
  if(neb){ctx.drawImage(neb,offX-W,0);ctx.drawImage(neb,offX,0);}

  /* stars */
  for(let i=0;i<stars.length;i++){
    const st=stars[i];
    const cx=st.x*BR0+st.y*BR1+st.z*BR2,cy=st.x*BU0+st.y*BU1+st.z*BU2,cz=st.x*BF0+st.y*BF1+st.z*BF2;
    if(cz<0.06)continue;
    const k=focal/cz,sx=W2+cx*k,sy=H2-cy*k;
    if(sx<-4||sx>W+4||sy<-4||sy>H+4)continue;
    ctx.globalAlpha=st.a*(0.55+0.45*Math.sin(time*st.sp+st.ph));
    ctx.fillStyle=st.c;ctx.fillRect(sx,sy,st.s,st.s);
  }
  ctx.globalAlpha=1;

  /* dust (speed streaks) */
  const mvx=CAMV[0]*0.06,mvy=CAMV[1]*0.06,mvz=CAMV[2]*0.06;
  for(let i=0;i<dust.length;i++){
    const p=dust[i];
    const dx=wdelta(p[0],cam.pos[0],WX),dy=wdelta(p[1],cam.pos[1],WY),dz=wdelta(p[2],cam.pos[2],WZ);
    const a=projRel(dx,dy,dz);if(!a)continue;
    const b=projRel(dx-mvx,dy-mvy,dz-mvz);
    ctx.globalAlpha=Math.min(0.5,140/a[2]);
    ctx.strokeStyle='#8fb8e6';ctx.lineWidth=1;
    if(b){ctx.beginPath();ctx.moveTo(a[0],a[1]);ctx.lineTo(b[0],b[1]);ctx.stroke();}
    else{ctx.fillStyle='#8fb8e6';ctx.fillRect(a[0],a[1],1.4,1.4);}
  }
  ctx.globalAlpha=1;

  /* faces */
  fCount=0;
  for(let i=0;i<rocks.length;i++)drawRockFaces(rocks[i]);
  if(state==='play'&&ship.alive)drawShipFaces();
  const ord=order.subarray(0,fCount);
  ord.sort((a,b)=>fz[b]-fz[a]);
  ctx.lineWidth=1;
  for(let i=0;i<fCount;i++){
    const idx=ord[i],o=idx*6,c=fcol[idx];
    ctx.globalAlpha=fal[idx];ctx.fillStyle=c;ctx.strokeStyle=c;
    ctx.beginPath();ctx.moveTo(fxy[o],fxy[o+1]);ctx.lineTo(fxy[o+2],fxy[o+3]);ctx.lineTo(fxy[o+4],fxy[o+5]);ctx.closePath();
    ctx.fill();ctx.stroke();
  }
  ctx.globalAlpha=1;

  /* additive pass */
  ctx.globalCompositeOperation='lighter';
  for(let i=0;i<lasers.length;i++){
    const l=lasers[i];
    const dx=wdelta(l.pos[0],cam.pos[0],WX),dy=wdelta(l.pos[1],cam.pos[1],WY),dz=wdelta(l.pos[2],cam.pos[2],WZ);
    const a=projRel(dx,dy,dz),b=projRel(dx-l.dir[0]*30,dy-l.dir[1]*30,dz-l.dir[2]*30);
    if(!a||!b)continue;
    const lw=Math.max(1,Math.min(4,0.5*a[2]));
    ctx.strokeStyle='rgba(255,190,70,.35)';ctx.lineWidth=lw*3;
    ctx.beginPath();ctx.moveTo(a[0],a[1]);ctx.lineTo(b[0],b[1]);ctx.stroke();
    ctx.strokeStyle='#ffe9a0';ctx.lineWidth=lw;
    ctx.beginPath();ctx.moveTo(a[0],a[1]);ctx.lineTo(b[0],b[1]);ctx.stroke();
  }
  if(state==='play'&&ship.alive){
    transformShip();
    const f=[Math.cos(ship.yaw),0,Math.sin(ship.yaw)],rgt=[-f[2],0,f[0]];
    if(ship.thrustOn&&keys.U){
      const ox=(shipW[3]+shipW[12])/2,oy=(shipW[4]+shipW[13])/2,oz=(shipW[5]+shipW[14])/2;
      const fl=4+Math.random()*2.5+Math.hypot(ship.vel[0],ship.vel[2])*0.012;
      flame(ox,oy,oz,f,rgt,fl,'rgba(255,160,50,.7)',1.15);
      flame(ox,oy,oz,f,rgt,fl*0.55,'rgba(255,240,190,.9)',0.7);
    }
    if(time-lastShot<0.05){
      const a=projRel(shipW[0],shipW[1],shipW[2]);
      if(a){const r=Math.max(2,2.4*a[2]);
        ctx.fillStyle='rgba(255,170,60,.35)';ctx.beginPath();ctx.arc(a[0],a[1],r*2.2,0,TAU);ctx.fill();
        ctx.fillStyle='rgba(255,235,170,.9)';ctx.beginPath();ctx.arc(a[0],a[1],r,0,TAU);ctx.fill();}
    }
  }
  for(let i=0;i<parts.length;i++){
    const p=parts[i];
    const a=projRel(wdelta(p.pos[0],cam.pos[0],WX),wdelta(p.pos[1],cam.pos[1],WY),wdelta(p.pos[2],cam.pos[2],WZ));
    if(!a)continue;
    const sz=Math.max(1,Math.min(26,p.sz*a[2])),al=Math.max(0,p.life/p.max);
    ctx.globalAlpha=al*0.9;ctx.fillStyle=p.c;
    ctx.fillRect(a[0]-sz/2,a[1]-sz/2,sz,sz);
  }
  ctx.globalAlpha=1;ctx.globalCompositeOperation='source-over';

  /* score popups */
  for(let i=0;i<popups.length;i++){
    const p=popups[i];
    const a=projRel(wdelta(p.pos[0],cam.pos[0],WX),wdelta(p.pos[1],cam.pos[1],WY),wdelta(p.pos[2],cam.pos[2],WZ));
    if(!a)continue;
    ctx.globalAlpha=Math.max(0,1-p.age/1.3);
    ctx.font='700 '+Math.max(10,Math.min(20,520/a[2]))+'px Consolas,Menlo,monospace';
    ctx.textAlign='center';ctx.fillStyle='#ffd23f';
    ctx.fillText(p.txt,a[0],a[1]-p.age*30);
  }
  ctx.globalAlpha=1;

  if(state!=='menu')updateHUD();
  dmgEl.style.opacity=Math.min(1,dmgFlash);
}
function flame(ox,oy,oz,f,rgt,len,col,w){
  const a=projRel(ox,oy,oz),b=projRel(ox-f[0]*len+rgt[2]*w,oy-f[1]*len,oz-f[2]*len+rgt[2]*w);
  const c=projRel(ox-f[0]*len-rgt[2]*w,oy-f[1]*len,oz-f[2]*len-rgt[2]*w);
  if(!a||!b||!c)return;
  ctx.fillStyle=col;ctx.beginPath();ctx.moveTo(a[0],a[1]);ctx.lineTo(b[0],b[1]);ctx.lineTo(c[0],c[1]);ctx.closePath();ctx.fill();
}
```
