function banner(text,sub){ bannerObj={text,sub,t:0,tot:1.9}; }
function startGame(){
  score=0; balls=3; elapsed=0; wave=1; spawnT=2.2;
  combo=0; comboT=0; kills=0; purged=0; maxCombo=1;
  timeScale=1; tsTarget=1; shake=0; bannerObj=null; paused=false;
  enemies.length=0; bullets.length=0; parts.length=0; rings.length=0; pops.length=0; pickups.length=0;
  resetBall(); hideScreens(); state='playing';
  banner('WAVE 1','ESCAPE THE STORM'); sfx.wave();
}
function resetBall(){
  ball.x=558; ball.y=875; ball.vx=0; ball.vy=0; ball.state='plunger'; ball.shield=false; ball.inv=0;
  ball.trail.length=0; chargeT=0; power=0; charging=false;
}
function loseBall(){
  if(ball.state!=='live')return;
  balls--; ball.state='dying'; deathT=0; tsTarget=.22; charging=false; combo=0;
  burst(ball.x,ball.y,44,'#9fefff',160,560,380);
  rings.push({x:ball.x,y:ball.y,r:8,max:150,color:'#ff4fd8'});
  sfx.lost(); shake=13; banner('BALL LOST',balls>0?'LAUNCH YOUR NEXT BALL':'');
}
function afterDeath(){ if(balls>0){ resetBall(); tsTarget=1; } else gameOver(); }
function gameOver(){
  state='over'; tsTarget=1;
  const rec=score>hi; if(rec){ hi=score; saveHi(); }
  $('finScore').textContent=score.toLocaleString('en-US');
  $('stWave').textContent=wave; $('stKills').textContent=kills;
  $('stPurge').textContent=purged; $('stCombo').textContent='×'+maxCombo;
  $('newRec').classList.toggle('hidden',!rec);
  hideScreens(); scrOver.classList.remove('hidden'); sfx.over();
}
function togglePause(){ if(state!=='playing')return; paused=!paused; charging=false;
  scrPause.classList.toggle('hidden',!paused); }

    parts.push({x,y,vx:Math.cos(a)*s,vy:Math.sin(a)*s,t:0,tot:rand(.35,.8),col,sz:rand(1.5,3.4),g}); }
}

function spawnEnemy(){
  const pool=elapsed>95?4:elapsed>60?3:elapsed>28?2:1;
  const type=Math.floor(rand(0,pool)), T=ETYPES[type], sa=rand(26,60);
  const e={type,color:T.color,sides:T.sides,hp:T.hp,maxHp:T.hp,rotSpd:T.rotSpd,
    bx:rand(70+sa,500-sa),x:0,y:-36,ty:rand(85,180),t:0,life:rand(16,22),sa,sf:rand(.5,.95),
    ft:rand(.8,1.4),ang:rand(0,TAU),rot:rand(0,TAU),flash:0,r:24,retreat:false,fade:1};
  e.x=e.bx; enemies.push(e);
}
function killEnemy(i,e){
  enemies.splice(i,1); kills++;
  const pts=600+e.type*400; score+=pts;
  if(pops.length<30)pops.push({x:e.x,y:e.y,txt:'+'+pts,t:0,col:'#ff9fe8',big:true});
  burst(e.x,e.y,30,e.color,140,520,260);
  rings.push({x:e.x,y:e.y,r:10,max:80,color:e.color});
  const spd=1+Math.min(.55,elapsed/160); radial(e.x,e.y,18,235*spd,e.color);
  if(Math.random()<.18)pickups.push({x:e.x,y:e.y,vy:70,r:13,t:0});
  sfx.boom(); shake=Math.max(shake,8);
}

// ============================ physics ============================
function collideSeg(w){
  const b=ball,dx=w.x2-w.x1,dy=w.y2-w.y1,L2=dx*dx+dy*dy||1;
  let t=((b.x-w.x1)*dx+(b.y-w.y1)*dy)/L2; if(t<0)t=0; else if(t>1)t=1;
  const qx=w.x1+dx*t,qy=w.y1+dy*t;
  let nx=b.x-qx,ny=b.y-qy; const d2=nx*nx+ny*ny;
  if(d2>b.r*b.r)return;
  let d=Math.sqrt(d2); if(d<.001){nx=0;ny=-1;d=1;} else {nx/=d;ny/=d;}
  b.x=qx+nx*b.r; b.y=qy+ny*b.r;
  const vn=b.vx*nx+b.vy*ny; if(vn>=0)return;
  if(w.kick){
    const tx=-w.ny,ty=w.nx,tv=b.vx*tx+b.vy*ty;
    b.vx=w.nx*w.kick+tx*tv*.35; b.vy=w.ny*w.kick+ty*tv*.35;
    w.flash=1; comboHit(); addScore(w.score,b.x,b.y); sfx.sling();
    burst(b.x,b.y,6,'#ffd23e',140,280,200); shake=Math.max(shake,3);
  } else {
    const e=(-vn>90)?w.e:0;
    b.vx-=(1+e)*vn*nx; b.vy-=(1+e)*vn*ny;
    if(-vn>700)sfx.thud();
  }
}
function collideFlipper(f){
  const b=ball,ca=Math.cos(f.a),sa=Math.sin(f.a);
  const tx=f.x+ca*f.len,ty=f.y+sa*f.len;
  let t=((b.x-f.x)*(tx-f.x)+(b.y-f.y)*(ty-f.y))/(f.len*f.len); t=clamp(t,0,1);
  const qx=f.x+(tx-f.x)*t,qy=f.y+(ty-f.y)*t;
  let nx=b.x-qx,ny=b.y-qy; const rr=f.r+b.r,d2=nx*nx+ny*ny;
  if(d2>rr*rr)return;
  let d=Math.sqrt(d2); if(d<.01){nx=0;ny=-1;d=.01;} else {nx/=d;ny/=d;}
  b.x=qx+nx*rr; b.y=qy+ny*rr;
  const rx=qx-f.x,ry=qy-f.y,svx=-f.omega*ry,svy=f.omega*rx;
  const rvx=b.vx-svx,rvy=b.vy-svy,vn=rvx*nx+rvy*ny;
  if(vn<0){ b.vx-=(1+f.e)*vn*nx; b.vy-=(1+f.e)*vn*ny; }
}
function doBumperHit(bp){
  rings.push({x:bp.x,y:bp.y,r:14,max:95,color:'#4de3ff'});
  for(let i=bullets.length-1;i>=0;i--){ const u=bullets[i],dx=u.x-bp.x,dy=u.y-bp.y;
    if(dx*dx+dy*dy<85*85){ bullets.splice(i,1); purged++; score+=5; } }
  comboHit(); addScore(100,bp.x,bp.y-bp.r-6); sfx.bumper(combo);
  burst(bp.x,bp.y,10,'#ff9fe8',120,320,160); shake=Math.max(shake,4);
}
function stepBall(s){
  const b=ball;
  b.vy+=1200*s; b.vx*=1-.025*s; b.vy*=1-.025*s;
  const sp2=b.vx*b.vx+b.vy*b.vy,M=1900;
  if(sp2>M*M){ const k=M/Math.sqrt(sp2); b.vx*=k; b.vy*=k; }
  b.x+=b.vx*s; b.y+=b.vy*s;
  for(const w of walls)collideSeg(w);
  if(b.x<534&&b.vx>0)collideSeg(gateW);
  for(const p of posts){
    let nx=b.x-p.x,ny=b.y-p.y; const rr=p.r+b.r,d2=nx*nx+ny*ny;
    if(d2>rr*rr)continue;
    let d=Math.sqrt(d2)||.01; nx/=d; ny/=d;
    b.x=p.x+nx*rr; b.y=p.y+ny*rr;
    const vn=b.vx*nx+b.vy*ny;
    if(vn<0){ b.vx-=1.75*vn*nx; b.vy-=1.75*vn*ny; p.flash=1; addScore(15,b.x,b.y); sfx.tick(); }
  }
  for(const bp of bumpers){
    let nx=b.x-bp.x,ny=b.y-bp.y; const rr=bp.r+b.r,d2=nx*nx+ny*ny;
    if(d2>rr*rr)continue;
    let d=Math.sqrt(d2)||.01; nx/=d; ny/=d;
    b.x=bp.x+nx*rr; b.y=bp.y+ny*rr;
    const vn=b.vx*nx+b.vy*ny;
    if(vn<0){
      const tx=-ny,ty=nx,tv=b.vx*tx+b.vy*ty;
      b.vx=nx*830+tx*tv*.3; b.vy=ny*830+ty*tv*.3;
      bp.flash=1; if(bp.cd<=0){ bp.cd=.12; doBumperHit(bp); }
    }
  }
  collideFlipper(flippers[0]); collideFlipper(flippers[1]);
  if(b.x>536&&b.y>790){ b.x=558; b.y=875; b.vx=0; b.vy=0; b.state='plunger'; }
  if(b.y>H+26)loseBall();
}
function updFlippers(dt,L,R){
  const fL=flippers[0],fR=flippers[1];
  fL.active=L; fR.active=R;
  if(L&&!fL.prev)sfx.flipper(true); if(R&&!fR.prev)sfx.flipper(true);
  if(!L&&fL.prev)sfx.flipper(false); if(!R&&fR.prev)sfx.flipper(false);
  for(const f of flippers){
    const prev=f.a,tgt=f.active?f.aA:f.aR,d=tgt-prev,st=20*dt;
    let na=Math.abs(d)<=st?tgt:prev+Math.sign(d)*st;
    f.a=na; f.omega=(na-prev)/Math.max(dt,1e-5);
  }
  fL.prev=L; fR.prev=R;
}
function updTrail(){ ball.trail.unshift({x:ball.x,y:ball.y}); if(ball.trail.length>14)ball.trail.pop(); }

function releasePlunger(){
  if(!charging)return; charging=false;
  if(state==='playing'&&!paused&&ball.state==='plunger'){
    ball.state='live'; ball.x=558; ball.y=863;
    ball.vy=-(820+1050*power); ball.vx=rand(-14,14); ball.inv=.9;
    sfx.launch(power); shake=Math.max(shake,2);
    rings.push({x:558,y:870,r:6,max:60,color:'#4de3ff'});
  }
  power=0; chargeT=0;
}
function updEnemies(sdt){
  const spd=1+Math.min(.55,elapsed/160), rate=Math.max(.55,1-elapsed*.004);
  for(let i=enemies.length-1;i>=0;i--){
    const e=enemies[i]; e.t+=sdt; e.flash=Math.max(0,e.flash-sdt*3.5);
    if(e.retreat){ e.y-=160*sdt; e.fade-=sdt/1.4; if(e.y<-70||e.fade<=0)enemies.splice(i,1); continue; }
    if(e.t>=e.life){ e.retreat=true; e.fade=1; continue; }
    e.y+=(e.ty-e.y)*Math.min(1,sdt*2.5);
    e.x=e.bx+Math.sin(e.t*e.sf)*e.sa;
    if(e.y>95){
      switch(e.type){
        case 0: e.ft-=sdt; if(e.ft<=0){ e.ft=rand(1.3,1.7)*rate; radial(e.x,e.y,12+Math.min(8,Math.floor(elapsed/45)),235*spd,e.color); } break;
        case 1: e.ft-=sdt; if(e.ft<=0){ e.ft=.09; e.ang+=.46;
          addB(e.x,e.y,Math.cos(e.ang)*255*spd,Math.sin(e.ang)*255*spd,e.color,4);
          addB(e.x,e.y,Math.cos(e.ang+Math.PI)*255*spd,Math.sin(e.ang+Math.PI)*255*spd,e.color,4); } break;
        case 2: e.ft-=sdt; if(e.ft<=0){ e.ft=1.05*rate;
          const tx=ball.state==='live'?ball.x:300, ty=ball.state==='live'?ball.y:500;
          const a=Math.atan2(ty-e.y,tx-e.x);
          for(const o of[-.26,0,.26])addB(e.x,e.y,Math.cos(a+o)*300*spd,Math.sin(a+o)*300*spd,e.color,5.5); } break;
        case 3: e.ft-=sdt; if(e.ft<=0){ e.ft=1.85*rate; e.rot+=.5;
          for(let k=0;k<26;k++){ if(k%5===4)continue; const a=k/26*TAU+e.rot;
            addB(e.x,e.y,Math.cos(a)*205*spd,Math.sin(a)*205*spd,e.color,4.5); } } break;
      }
    }
    const b=ball;
    if(b.state==='live'){
      let nx=b.x-e.x,ny=b.y-e.y; const rr=e.r+b.r,d2=nx*nx+ny*ny;
      if(d2<rr*rr){
        let d=Math.sqrt(d2)||.01; nx/=d; ny/=d;
        b.x=e.x+nx*rr; b.y=e.y+ny*rr;
        const tx=-ny,ty=nx,tv=b.vx*tx+b.vy*ty;
        b.vx=nx*540+tx*tv*.3; b.vy=ny*540+ty*tv*.3;
        e.hp--; e.flash=1; addScore(50,b.x,b.y-14); sfx.hit();
        burst(b.x,b.y,8,e.color,160,320,220);
        if(e.hp<=0)killEnemy(i,e);
      }
    }
  }
}
function updBullets(sdt){
  for(let i=bullets.length-1;i>=0;i--){
    const u=bullets[i]; u.x+=u.vx*sdt; u.y+=u.vy*sdt;
    if(u.x<-30||u.x>W+30||u.y>H+30||u.y<-70){ bullets.splice(i,1); continue; }
    let dead=false;
    for(const w of walls){
      const dx=w.x2-w.x1,dy=w.y2-w.y1,L2=dx*dx+dy*dy||1;
      let t=((u.x-w.x1)*dx+(u.y-w.y1)*dy)/L2; if(t<0||t>1)continue;
      const qx=w.x1+dx*t,qy=w.y1+dy*t,nx=u.x-qx,ny=u.y-qy,rr=u.r+1.5;
      if(nx*nx+ny*ny<rr*rr){ dead=true; break; }
    }
    if(!dead)for(const bp of bumpers){
      const nx=u.x-bp.x,ny=u.y-bp.y,rr=bp.r+u.r;
      if(nx*nx+ny*ny<rr*rr){ dead=true; purged++; score+=2; burst(u.x,u.y,3,'#ff9fe8',60,140,80); break; }
    }
    if(dead){ bullets.splice(i,1); continue; }
    const b=ball;
    if(b.state==='live'&&b.inv<=0){
      const nx=u.x-b.x,ny=u.y-b.y,rr=u.r+b.r-1.5;
      if(nx*nx+ny*ny<rr*rr){
        if(b.shield){ b.shield=false; shieldClear(); }
        else { loseBall(); break; }
      }
    }
  }
}
function updPickups(sdt){
  for(let i=pickups.length-1;i>=0;i--){
    const p=pickups[i]; p.t+=sdt; p.y+=p.vy*sdt; p.vy+=30*sdt;
    if(p.y>H+20){ pickups.splice(i,1); continue; }
    if(ball.state==='live'){
      const dx=p.x-ball.x,dy=p.y-ball.y,rr=p.r+ball.r;
      if(dx*dx+dy*dy<rr*rr){
        ball.shield=true; pickups.splice(i,1); sfx.pickup();
        if(pops.length<30)pops.push({x:p.x,y:p.y-14,txt:'SHIELD',t:0,col:'#4de3ff'});
        burst(p.x,p.y,12,'#7fd4ff',100,260,100);
      }
    }
  }
}
function updFx(dt){
  for(let i=parts.length-1;i>=0;i--){ const p=parts[i]; p.t+=dt;
    if(p.t>=p.tot){ parts.splice(i,1); continue; }
    p.x+=p.vx*dt; p.y+=p.vy*dt; p.vy+=p.g*dt; p.vx*=1-2.2*dt; }
  for(let i=rings.length-1;i>=0;i--){ const r=rings[i]; r.r+=420*dt; if(r.r>=r.max)rings.splice(i,1); }
  for(let i=pops.length-1;i>=0;i--){ const p=pops[i]; p.t+=dt; p.y-=30*dt; if(p.t>.9)pops.splice(i,1); }
  for(const bp of bumpers){ bp.cd-=dt; bp.flash=Math.max(0,bp.flash-dt*3); }
  for(const p of posts)p.flash=Math.max(0,p.flash-dt*4);
  for(const w of walls)if(w.kick)w.flash=Math.max(0,w.flash-dt*4);
  if(bannerObj){ bannerObj.t+=dt; if(bannerObj.t>=bannerObj.tot)bannerObj=null; }
}
function worldUpdate(dt){
  timeScale+=(tsTarget-timeScale)*Math.min(1,dt*7);
  const sdt=dt*timeScale;
  elapsed+=sdt;
  if(comboT>0){ comboT-=sdt; if(comboT<=0)combo=0; }
  comboPop=Math.max(0,comboPop-dt*3);
  shake=Math.max(0,shake-dt*26);
  const nw=1+Math.floor(elapsed/25);
  if(nw>wave){ wave=nw; score+=250*wave; banner('WAVE '+wave,'THREAT RISING · +'+(250*wave)); sfx.wave(); }
  spawnT-=sdt;
  const maxE=Math.min(3,1+Math.floor(elapsed/35));
  if(spawnT<=0){ if(enemies.length<maxE)spawnEnemy(); spawnT=rand(2.4,4.4)*Math.max(.55,1-elapsed*.003); }
  const L=!!(keys.KeyA||keys.ArrowLeft||holdL), R=!!(keys.KeyD||keys.ArrowRight||holdR);
  updFlippers(sdt,L,R);
  if(charging&&ball.state==='plunger'){ chargeT+=dt; const ph=(chargeT*1.35)%2; power=ph<1?ph:2-ph; }
  if(ball.state==='plunger')ball.y=charging?875+power*9:875;
  if(ball.state==='live'){
    ball.inv=Math.max(0,ball.inv-sdt);
    for(let i=0;i<8;i++)stepBall(sdt/8);
    updTrail();
  } else if(ball.state==='dying'){ deathT+=dt; if(deathT>1.25)afterDeath(); }
  updEnemies(sdt); updBullets(sdt); updPickups(sdt); updFx(sdt);
}

function rr(c,x,y,w,h,r){ c.beginPath(); c.moveTo(x+r,y); c.arcTo(x+w,y,x+w,y+h,r);
  c.arcTo(x+w,y+h,x,y+h,r); c.arcTo(x,y+h,x,y,r); c.arcTo(x,y,x+w,y,r); c.closePath(); }
function polyPath(c,x,y,rad,n,rot){ c.beginPath();
  for(let i=0;i<n;i++){ const a=rot+i/n*TAU,px=x+Math.cos(a)*rad,py=y+Math.sin(a)*rad;
    i?c.lineTo(px,py):c.moveTo(px,py); } c.closePath(); }

  g.addColorStop(0,col); g.addColorStop(1,'rgba(0,0,0,0)');
  ctx.fillStyle=g; ctx.fillRect(x-r,y-r,r*2,r*2);
}

}
function drawBumpersLive(){
  for(const bp of bumpers){
    const p=1+bp.flash*.22, r=bp.r*p;
    ctx.strokeStyle='rgba(255,79,216,.3)'; ctx.lineWidth=8;
    ctx.beginPath(); ctx.arc(bp.x,bp.y,r,0,TAU); ctx.stroke();

    ctx.globalAlpha=tw*(s.z>.5?1:.5);
    ctx.fillStyle=s.z>.5?'#cfe6ff':'#8fa8d8';

  ctx.globalAlpha=1;

  ctx.restore();
}

let last=performance.now();
function frame(now){
  requestAnimationFrame(frame);
  let dt=(now-last)/1000; last=now;
  if(dt>0.05)dt=0.05;
  tG+=dt;
  for(const s of stars){ s.yf+=dt*(.010+.05*s.z); if(s.yf>1)s.yf-=1; }
  if(state==='playing'&&!paused)worldUpdate(dt);
  else if(state==='over')updFx(dt);
  render();
}
requestAnimationFrame(frame);
})();
