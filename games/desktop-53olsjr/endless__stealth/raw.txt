  const o=AC.createOscillator();o.type='sine';o.frequency.value=52;

function tone(fr,dur,type,vol,slide,at){
  if(!AC)return;const t=AC.currentTime+(at||0);
  const o=AC.createOscillator(),g=AC.createGain();
  o.type=type||'sine';o.frequency.setValueAtTime(Math.max(20,fr),t);
  if(slide)o.frequency.exponentialRampToValueAtTime(Math.max(20,slide),t+dur);
  g.gain.setValueAtTime(vol||.15,t);g.gain.exponentialRampToValueAtTime(.0001,t+dur);
  o.connect(g);g.connect(master);o.start(t);o.stop(t+dur+.03);
}
function nz(dur,vol,fType,fFreq,q){

  const f=AC.createBiquadFilter();f.type=fType||'bandpass';f.frequency.value=fFreq||800;f.Q.value=q||1;
  const g=AC.createGain();g.gain.setValueAtTime(vol,t);g.gain.exponentialRampToValueAtTime(.0001,t+dur);
  s.connect(f);f.connect(g);g.connect(master);s.start(t);s.stop(t+dur+.03);
}

function pickChunk(t){
  const opts=[['watch',3],['gravel',1.5],['propcrate',1.3],['laser',t>0.05?2.2+t*2.5:0.9],
    ['patrol',t>0.1?1.4+t*2.4:0],['cover',t>0.14?2+t*2.6:0],['cam',t>0.2?1.3+t*2.2:0],
    ['double',t>0.5?(t-0.5)*4:0]];
  let sum=0;for(const o of opts)sum+=o[1];let r=Math.random()*sum;
  for(const o of opts){if((r-=o[1])<=0)return o[0];}return 'watch';
}

  if(state==='menu')k=Math.random()<.5?'propcrate':'gravel';

  else if(k==='patrol'){const x0=x+150,x1=x0+230;if(lane)ensurePlatform((x0+x1)/2);ents.push(mkGuard(0,lane,'patrol',x0,x1,t));len=400;}
  else if(k==='cover'){const gx=x+320,cr=gx-158;crates.push({x:cr,w:CRW,lane});if(lane)ensurePlatform(gx);ents.push(mkGuard(gx,lane,'watch',0,0,t));len=110;}

function updateWorld(sdt,rdt){
  const playing=state==='playing';
  time+=sdt;vt+=rdt;
  if(playing)speed=Math.min(420,200+time*3.2);

  // dash
  pl.dashCd-=sdt;
  let eff=speed;
  if(pl.dashT>0){pl.dashT-=sdt;eff=speed*2.4;
    ghosts.push({x:pxw,y:pl.py,c:pl.crouch,t:.4});if(ghosts.length>14)ghosts.shift();}
  camX+=eff*sdt;runPhase+=eff*sdt*0.05;

  // lane / vertical
  if(pl.lane===1&&!platformAt(pxw)){pl.lane=0;pl.falling=true;}
  const tgt=laneFootY(pl.lane);
  pl.py+=(tgt-pl.py)*Math.min(1,sdt*(pl.py<tgt?11:13));
  if(pl.falling&&pl.lane===0&&Math.abs(pl.py-GY)<3){
    pl.falling=false;sfx.land();meter=Math.min(100,meter+(input.crouch?2:5));
    shake=Math.max(shake,3);puff(pxw,GY,8,'#8a9aa8',70);}

  if(playing){
    // spawn & cull
    spawnAhead();cull();
    // entities
    for(const g of ents){
      if(g.type==='g'&&g.sub==='patrol'){
        g.x+=g.dir*g.spd*sdt;
        if(g.x<g.x0){g.x=g.x0;g.dir=1;}if(g.x>g.x1){g.x=g.x1;g.dir=-1;}
        g.face=g.dir>0?0:Math.PI;
      }else if(g.type==='c'){g.face=Math.PI+Math.sin(time*g.w+g.ph)*g.spread;}
      else g.face=Math.PI;
    }
    // detection
    const ppy=pl.py-(pl.crouch?28:58);
    let seen=false,fill=0;
    for(const g of ents){
      const ex=g.x,ey=laneFootY(g.lane)-(g.type==='c'?150:64);
      const dx=pxw-ex,dy=ppy-ey,d=Math.hypot(dx,dy);let inside=false;
      if(d>24&&d<g.R){const a=Math.atan2(dy,dx);if(Math.abs(angDiff(a,g.face))<=g.half)inside=true;}
      if(inside){
        let blocked=false;
        for(const c of crates){
          const ov=pxw>c.x-10&&pxw<c.x+c.w+10;if(ov&&!pl.crouch)continue;
          if(segRect(ex,ey,pxw,ppy,{x:c.x,y:laneFootY(c.lane)-CRH,w:c.w,h:CRH})){blocked=true;break;}
        }
        if(!blocked)for(const p of platforms){
          if(segRect(ex,ey,pxw,ppy,{x:p.x,y:WY,w:p.w,h:SLAB})){blocked=true;break;}}
      }
      g.detX=pxw;g.detY=ppy;
      if(inside&&!blocked){seen=true;const prox=1-d/g.R;fill+=(16+40*prox*prox)*(pl.crouch?0.55:1);g.saw=true;}
      else g.saw=false;
    }
    if(seen){fill*= (1+tier()*0.5);meter=Math.min(100,meter+fill*sdt);lastCause='A GUARD CLOCKED YOU';}
    else meter=Math.max(0,meter-(pl.crouch?26:17)*sdt);
    if(seen&&sawCd<=0){sfx.tick();sawCd=.18;}
    for(const g of ents)if(g.sawPrev&&!g.saw)g.saw=false;
    sawCd-=sdt;for(const g of ents)g.sawPrev=g.saw;

    // gravel noise
    for(const gr of gravelSegs){
      if(pl.lane===0&&pxw>gr.x&&pxw<gr.x+gr.w){
        meter=Math.min(100,meter+(input.crouch?3:9)*sdt);
        dustAcc+=sdt*(input.crouch?4:14);
        while(dustAcc>=1){dustAcc--;particles.push({x:pxw-8,y:GY,vx:-speed*.25,vy:-20-Math.random()*30,life:.5,t:0,col:'#6d7f8e',sz:2});}
      }
    }
    // lasers
    for(const l of lasers){
      if(l.hitT>0)l.hitT-=sdt;
      const dy=Math.abs(pxw-l.x);
      if(pl.lane===l.lane&&!l.hit&&dy<9&&!pl.crouch){
        l.hit=true;l.hitT=.35;meter=Math.min(100,meter+70);lastCause='SENSOR GRID TRIPPED';
        redFlash=1;shake=9;sfx.alarm();puff(l.x,laneFootY(l.lane)-58,12,'#ff5468',120);}
      if(!l.hit&&!l.passed&&dy<9&&pl.crouch){l.passed=true;score+=40;addFloat(l.x,laneFootY(l.lane)-92,'+40 CLEAN DUCK','#7fd8cf');sfx.clean();}
    }
    // streak & score
    if(meter>0)streakT=Math.max(0,streakT-sdt*2.5);else streakT+=sdt;
    const nl=1+Math.floor(streakT/8);if(nl>5)nl=5;
    if(nl!==streakLvl){
      if(nl>streakLvl){addFloat(pxw,pl.py-110,'SNEAK ×'+nl,'#ffb454');sfx.streak();streakEl.classList.remove('pop');void streakEl.offsetWidth;streakEl.classList.add('pop');}
      streakLvl=nl;}
    maxLvl=Math.max(maxLvl,streakLvl);
    score+=speed*sdt*0.05+(meter<=0?sdt*8*streakLvl:0);

    // footsteps + heartbeat
    stepT-=sdt*(pl.dashT>0?2.2:1);
    if(stepT<=0){sfx.step(pl.crouch);stepT=pl.crouch?0.42:0.3;}
    hbT-=rdt;
    if(meter>26&&hbT<=0){sfx.heart(meter>75);hbT=lerp(1.05,0.42,(meter-26)/74);}

    // tips
    if(tipIdx<TIPS.length&&time>TIPS[tipIdx].t){
      tipEl.textContent=TIPS[tipIdx].txt;tipEl.classList.remove('off');
      tipHideAt=time+3.8;tipIdx++;}
    if(tipHideAt>0&&time>tipHideAt){tipEl.classList.add('off');tipHideAt=-1;}

    // death
    if(meter>=100){state='dying';dieT=1.05;redFlash=1;sfx.over();}
  }

  // fx decay (both states)
  for(let i=particles.length-1;i>=0;i--){const p=particles[i];p.t+=sdt;p.x+=p.vx*sdt;p.y+=p.vy*sdt;p.vy+=60*sdt;if(p.t>=p.life)particles.splice(i,1);}
  for(let i=floaters.length-1;i>=0;i--){const f=floaters[i];f.t+=rdt;f.y-=28*rdt;if(f.t>1.1)floaters.splice(i,1);}
  for(let i=ghosts.length-1;i>=0;i--){ghosts[i].t-=sdt*2.6;if(ghosts[i].t<=0)ghosts.splice(i,1);}
  shake*=Math.exp(-6*rdt);redFlash=Math.max(0,redFlash-rdt*1.5);

  if(state==='dying'){dieT-=rdt;if(dieT<=0)showOver();}
}
