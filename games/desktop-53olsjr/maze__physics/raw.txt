function collideRect(b,rx,ry,rw,rh){
  const cx=clamp(b.x,rx,rx+rw),cy=clamp(b.y,ry,ry+rh);
  let dx=b.x-cx,dy=b.y-cy;const d2=dx*dx+dy*dy;
  if(d2>b.r*b.r)return null;
  let nx,ny,pen;
  if(d2>1e-9){const d=Math.sqrt(d2);nx=dx/d;ny=dy/d;pen=b.r-d;}
  else{const l=b.x-rx,r2=rx+rw-b.x,t=b.y-ry,bt=ry+rh-b.y,m=Math.min(l,r2,t,bt);
    if(m===l){nx=-1;ny=0;pen=l+b.r;}else if(m===r2){nx=1;ny=0;pen=r2+b.r;}
    else if(m===t){nx=0;ny=-1;pen=t+b.r;}else{nx=0;ny=1;pen=bt+b.r;}}
  b.x+=nx*pen;b.y+=ny*pen;
  const vn=b.vx*nx+b.vy*ny;
  if(vn<0){if(-vn<2.4){b.vx-=vn*nx;b.vy-=vn*ny;return{nx,ny,imp:0};}
    b.vx-=(1+REST)*vn*nx;b.vy-=(1+REST)*vn*ny;return{nx,ny,imp:-vn};}
  return{nx,ny,imp:0};}
