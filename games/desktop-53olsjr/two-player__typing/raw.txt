/* ---------- UPDATE ---------- */
function stepPlayers(dt){
  const d=diff();
  for(const p of players){
    const top=p.words.length?p.words[0]:null;
    if(p.spawnT<=0&&p.words.length<4&&(!top||top.y>.18)){
      p.words.push({text:pickWord(p),y:-.05,idx:0,actT:0,warned:false,bornA:S.A});
      p.spawnT=spawnInt(d,p);
    }
    for(const w of p.words){
      const isAct=p.active===w;
      w.y+=fallSpd(d,p)*dt*(isAct?1:.94);
      if(!w.warned&&w.y>.86){w.warned=true;sWarn();}
    }
    for(let i=p.words.length-1;i>=0;i--){if(p.words[i].y>=1){p.words.splice(i,1);leak(p);}}
    p.active=null;for(const w of p.words)if(!p.active||w.y>p.active.y)p.active=w;
    if(p.active)p.active.actT+=dt;
  }
}

function updParts(dt){

    if(p.life>p.max){parts.splice(i,1);continue;}
    p.vy+=(p.g||0)*dt;p.x+=p.vx*dt;p.y+=p.vy*dt;if(p.rot!==undefined)p.rot+=p.vr*dt;}
}
function updPops(dt){for(let i=pops.length-1;i>=0;i--){const p=pops[i];p.t+=dt;if(p.t>p.life)pops.splice(i,1);}}

function update(dt){
  S.A+=dt;
  S.gshake=Math.max(0,S.gshake-30*dt);
  S.flash=Math.max(0,S.flash-2.4*dt);
  updGlyphs(dt);updParts(dt);updPops(dt);
  for(const p of players){p.shake=Math.max(0,p.shake-42*dt);p.hitT=Math.max(0,p.hitT-dt);p.errT=Math.max(0,p.errT-dt);}
  if(S.state==='countdown'){
    S.countT+=dt;
    const st=Math.min(3,Math.floor(S.countT/.65));
    if(st!==S.lastStage){S.lastStage=st;if(st<3)beep(440,.08,'square',.3);else beep(880,.22,'square',.4);}
    if(S.countT>=2.5){S.state='playing';players.forEach(p=>p.spawnT=.45+Math.random()*.3);}
  }else if(S.state==='playing'){
    S.t+=dt;stepPlayers(dt);checkDeath();
  }else if(S.state==='dying'){
    S.dieT+=dt;const ts=S.dieT<.9?.3:1,d=diff();
    for(const p of players)for(const w of p.words)w.y=Math.min(1,w.y+fallSpd(d,p)*ts*dt);
    if(S.dieT>1.6)finalizeOver();
  }
}

  ctx.font=`${DISP}`;ctx.textAlign='center';ctx.textBaseline='middle';
  ctx.fillStyle=p.hp>0?'#eaf6ff':'#ff8a95';
  ctx.font=`${Math.round(r*.62)}px ${DISP}`;
  ctx.fillText('P'+(p.i+1),x,y-r*.06);
  ctx.font=`700 9px ${MONO}`;ctx.fillStyle='rgba(180,205,230,.55)';
  ctx.fillText('CORE',x,y+r*.4);
}
function drawHUD(p){
  const lx=laneX(p.i),cx=laneCx(p.i),col=p.def.col;
  ctx.textAlign='center';ctx.textBaseline='middle';
  ctx.font=`700 ${clamp(H*.02,10,13)}px ${MONO}`;
  ctx.fillStyle=col;ctx.fillText(p.def.label,cx,H*.034);
  const menu=S.state==='menu';
  ctx.font=`800 ${clamp(H*.045,22,36)}px ${MONO}`;
  ctx.fillStyle='#eef6ff';
  ctx.fillText(menu?'—':String(p.score),cx,H*.078);
  const bw=Math.min(260,LANE*.42);
  rr(cx-bw/2,H*.100,bw,5,3);ctx.fillStyle='rgba(255,255,255,.09)';ctx.fill();
  if(p.hp>0){const hc=p.hp>50?col:(p.hp>25?'#ffd54d':'#ff4757');
    rr(cx-bw/2,H*.100,bw*Math.max(0,p.hp)/100,5,3);ctx.fillStyle=hc;ctx.fill();}

  if(!menu&&S.t>0){ctx.textAlign='left';ctx.fillStyle='rgba(160,185,215,.6)';

  if(p.combo>0){ctx.textAlign='right';
    ctx.fillStyle=multOf(p.combo)>1?col:'rgba(160,185,215,.7)';
    ctx.fillText('COMBO '+p.combo+' ×'+multOf(p.combo),lx+LANE-14,H*.034);}
}
function drawLane(p){
  const lx=laneX(p.i);
  ctx.save();
  if(p.shake>.3)ctx.translate((Math.random()-.5)*p.shake,(Math.random()-.5)*p.shake);
  const g=ctx.createLinearGradient(0,FIELD_TOP*.6,0,H);
  g.addColorStop(0,rgba(p.def.col,0));g.addColorStop(1,rgba(p.def.col,.06));
  ctx.fillStyle=g;ctx.fillRect(lx,0,LANE,H);
  // spawn line
  ctx.strokeStyle=rgba(p.def.col,.25);ctx.setLineDash([8,10]);ctx.lineWidth=1.5;
  ctx.beginPath();ctx.moveTo(lx+10,FIELD_TOP);ctx.lineTo(lx+LANE-10,FIELD_TOP);ctx.stroke();
  ctx.setLineDash([]);
  // danger line
  if(p.words.some(w=>w.y>.72)){
    ctx.strokeStyle=`rgba(255,71,87,${.14+.1*Math.sin(S.A*9)})`;ctx.lineWidth=2;
    ctx.beginPath();ctx.moveTo(lx+10,FIELD_TOP+.9*FIELD_H);ctx.lineTo(lx+LANE-10,FIELD_TOP+.9*FIELD_H);ctx.stroke();
  }
  for(const w of p.words)(w===p.active)?drawActive(p,w):drawChip(p,w);
  drawCore(p);
  drawHUD(p);
  if(S.state==='menu'||S.state==='countdown'){
    ctx.font=`${Math.round(clamp(LANE*.14,30,60))}px ${DISP}`;
    ctx.textAlign='center';ctx.textBaseline='middle';
    ctx.fillStyle=rgba(p.def.col,.16);
    ctx.fillText(p.i?'RIGHT':'LEFT',laneCx(p.i),FIELD_TOP+FIELD_H*.42);
  }
  ctx.restore();
}
function drawGutter(){
  const g0=W/2-GUT/2;
  ctx.fillStyle='rgba(3,7,14,.6)';ctx.fillRect(g0,0,GUT,H);
  ctx.save();ctx.shadowBlur=8;

  ctx.shadowColor='#ffa13d';ctx.strokeStyle='rgba(255,161,61,.7)';
  ctx.beginPath();ctx.moveTo(g0+GUT,0);ctx.lineTo(g0+GUT,H);ctx.stroke();
  ctx.restore();

  ctx.font=`700 ${clamp(GUT*.16,13,19)}px ${MONO}`;

  ctx.font=`${Math.round(vs*.62)}px ${DISP}`;ctx.fillStyle='#eef6ff';
  ctx.fillText('VS',cx,vy+1);

  if(d>0){const fh=(gy1-gy0)*d;

    gg.addColorStop(0,'#ffd54d');gg.addColorStop(1,'#ff4757');

  ctx.font=`700 ${clamp(GUT*.09,8,11)}px ${MONO}`;
  ctx.fillStyle='rgba(160,185,215,.6)';
  ctx.fillText('STORM',cx,gy1+14);
  ctx.fillStyle='#ffd54d';ctx.fillText('LV '+(1+Math.floor(d*8)),cx,gy0-14);
  ctx.font=`700 9px ${MONO}`;ctx.fillStyle='rgba(120,145,175,.5)';
  ctx.fillText(muted?'SOUND OFF':'SOUND ON',cx,H*.68);
}
function drawParts(){
  for(const p of parts){
    const a=1-p.life/p.max;
    if(p.type==='s'){
      ctx.globalCompositeOperation='lighter';
      ctx.fillStyle=rgba(p.c.startsWith('#')?p.c:'#ffffff',a);
      ctx.beginPath();ctx.arc(p.x,p.y,p.sz*a+0.4,0,7);ctx.fill();
      ctx.globalCompositeOperation='source-over';
    }else{
      ctx.save();ctx.globalAlpha=a;ctx.translate(p.x,p.y);ctx.rotate(p.rot||0);
      ctx.font=`800 ${p.sz}px ${MONO}`;ctx.textAlign='center';ctx.textBaseline='middle';
      ctx.fillStyle=p.c;ctx.fillText(p.ch,0,0);ctx.restore();
    }
  }
}

}

/* ---------- KEYCAP DOM ---------- */
const capMap=new Map();
function buildCaps(el,rows){
  for(const row of rows){
    const rd=document.createElement('div');rd.className='row';
    for(const k of row){
      const kb=document.createElement('kbd');kb.className='key';kb.textContent=k.toUpperCase();
      kb.addEventListener('click',()=>{audio();lightCap(k);});
      rd.appendChild(kb);capMap.set(k,kb);
    }
    el.appendChild(rd);
  }
}
buildCaps($('capsL'),ROWS0);buildCaps($('capsR'),ROWS1);
function lightCap(k){
  if(!k||k.length!==1)return;
  const el=capMap.get(k.toLowerCase());if(!el)return;
  el.classList.add('lit');clearTimeout(el._t);
  el._t=setTimeout(()=>el.classList.remove('lit'),140);
}

/* ---------- INPUT ---------- */
window.addEventListener('keydown',e=>{
  const k=e.key;
  if(k===' '||k==='Enter'||k==='Tab')e.preventDefault();
  audio();lightCap(k);
  if(e.repeat)return;
  switch(S.state){
    case 'menu':
      if(k===' '||k==='Enter')startGame();
      else if(k==='`')toggleMute();
      break;
    case 'countdown':
      if(k==='Escape')toMenu();
      break;
    case 'playing':
      if(k===' '||k==='Escape')pauseGame();
      else if(k==='`')toggleMute();
      else if(k.length===1&&/[a-z]/i.test(k))handleType(k.toLowerCase());
      break;
    case 'paused':
      if(k===' '||k==='Enter')resumeGame();
      else if(k==='Escape')toMenu();
      break;
    case 'over':
      if(k===' '||k==='Enter')startGame();
      else if(k==='Escape'||/^m$/i.test(k))toMenu();
      break;
  }
});
window.addEventListener('blur',()=>{if(S.state==='playing')pauseGame();});
