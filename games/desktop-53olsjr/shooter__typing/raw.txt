function tierFor(w){if(w<=1)return[4,5];if(w===2)return[5,6];if(w===3)return[5,7];if(w===4)return[6,8];return[6,9];}
function countFor(w){return Math.min(26,6+w*2);}
function conCap(){return Math.min(6,2+Math.floor(wave*.8));}
function mult(){return Math.min(8,1+Math.floor(chain/4));}
function wpmNow(){const m=playTime/60;return m>.02?Math.round(correct/5/m):0;}
function accNow(){const t=correct+errors;return t?Math.round(100*correct/t):100;}
const breachY=()=>H-130, shipY=()=>H-64;

/* ---------- pods ---------- */
function makePod(word,o={}){
  const len=word.length;
  const fs=Math.max(13,Math.min(o.elite?22:20,Math.floor(Math.min(W*.26,380)/(len*.66))));
  ctx.font='700 '+fs+'px '+MONO;
  const tw=ctx.measureText(word.toUpperCase()).width;
  const d={id:++uid,word,progress:0,elite:!!o.elite,type:o.type||'drone',power:o.power||null,ambient:!!o.ambient,
    x:rand(90,Math.max(120,W-90)),y:-50,w:tw+36+(o.elite?14:0),h:fs*2.5+10+(o.elite?8:0),fs,hitT:0,ph:rand(0,6.28),baseX:0,
    vy:(o.vy!=null?o.vy:H*Math.min(.13,.045+wave*.006))*rand(.85,1.15)*(o.elite?1.15:1)};
  if(o.type==='crate')d.vy*=.72;
  d.baseX=clamp(d.x,d.w/2+24,Math.max(d.w/2+24,W-d.w/2-24));
  drones.push(d);return d;
}
function rollPower(){const r=Math.random();
  if(hull===1)return r<.5?'repair':(r<.8?'overdrive':'vaporize');
  return r<.34?'repair':(r<.72?'overdrive':'vaporize');}

/* ---------- fx ---------- */
function explode(x,y,col,n,s){for(let i=0;i<n;i++){const a=rand(0,6.28),sp=rand(40,240)*s;
  parts.push({kind:'px',x,y,vx:Math.cos(a)*sp,vy:Math.sin(a)*sp,g:120,life:rand(.4,.9),max:.9,size:rand(2,5)*s,
    col:[col,'#ffd9a0','#ffffff'][ri(0,2)],rot:rand(0,6.28),vr:rand(-8,8)});}
  rings.push({x,y,r:6,vr:340*s,life:.4,max:.4,col});
  if(parts.length>500)parts.splice(0,parts.length-500);}
function puff(x,y,col){for(let i=0;i<8;i++){const a=rand(0,6.28),sp=rand(20,90);
  parts.push({kind:'px',x,y,vx:Math.cos(a)*sp,vy:Math.sin(a)*sp-30,g:60,life:rand(.3,.5),max:.5,size:rand(1.5,3),col});}}
function pop(x,y,txt,col,size){pops.push({x,y,txt,col,size:size||18,life:.95,max:.95});}
function flash(c,a){flashC=c;flashA=Math.max(flashA,a);}

/* ---------- combat ---------- */
function activeTarget(){
  let cur=null,low=null;
  for(const d of drones){if(d.ambient)continue;
    if(d.progress>0&&!cur)cur=d;
    else if(!d.progress&&(!low||d.y>low.y))low=d;}
  return cur||low;
}
function typeChar(ch){
  const t=activeTarget();if(!t)return;
  if(t.word[t.progress]===ch){
    t.progress++;correct++;t.hitT=.12;mFlash=.07;S.blip(t.progress,mult());
    bolts.push({x1:shipX,y1:shipY()-26,x2:t.x,y2:t.y,life:.15,max:.15});
    for(let i=0;i<3;i++)parts.push({kind:'px',x:t.x+rand(-8,8),y:t.y-t.h/2+rand(-4,4),vx:rand(-60,60),vy:rand(-90,-20),life:.25,max:.25,size:rand(1,2.5),col:'#9ffbe8'});
    if(t.progress>=t.word.length)killDrone(t,false);
  }else{
    errors++;chain=0;shake=Math.max(shake,7);flash('255,80,80',.16);S.err();
    const c=$('console');c.classList.remove('err');void c.offsetWidth;c.classList.add('err');
    setTimeout(()=>c.classList.remove('err'),200);
  }
}
function killDrone(d,noDrop){
  const i=drones.indexOf(d);if(i>=0)drones.splice(i,1);
  if(target===d)target=null;
  if(d.type==='crate'){applyPower(d);return;}
  const m=mult(),pts=d.word.length*10*(d.elite?2:1)*m;
  score+=pts;chain++;maxChain=Math.max(maxChain,chain);
  explode(d.x,d.y,d.elite?'#ff8a3d':'#59e6cf',d.elite?26:16,d.elite?1.25:.95);
  pop(d.x,d.y-10,'+'+pts+(m>1?' ×'+m:''),m>=4?'#ff8a3d':(m>=2?'#59e6cf':'#dff1ee'),15+m*2);
  shake=Math.max(shake,d.elite?9:4.5);flash('255,255,255',d.elite?.14:.07);S.boom(d.elite?1.3:.8);
  if(!noDrop&&Math.random()<.16&&drones.filter(c=>c.type==='crate').length<2){
    const pw=rollPower();makePod(POWERS[pw].word,{type:'crate',power:pw});}
}
function applyPower(d){
  S.power();flash('89,230,207',.15);explode(d.x,d.y,'#59e6cf',18,.9);
  pop(d.x,d.y-12,d.power.toUpperCase(),'#59e6cf',22);
  if(d.power==='overdrive')ocT=7;
  else if(d.power==='repair'){if(hull<3){hull++;pop(d.x,d.y+14,'HULL +1','#7dffab',18);}
    else{score+=250;pop(d.x,d.y+14,'BONUS +250','#ffb454',18);}}
  else if(d.power==='vaporize'){S.vap();flash('255,255,255',.35);shake=14;
    rings.push({x:d.x,y:d.y,r:10,vr:900,life:.6,max:.6,col:'#ffffff'});
    for(let i=drones.length-1;i>=0;i--){const o=drones[i];if(o.type==='crate')continue;
      score+=Math.floor(o.word.length*10*(o.elite?2:1)/2);explode(o.x,o.y,'#9ffbe8',14,.8);drones.splice(i,1);}}}
function hullHit(d,i){drones.splice(i,1);if(target===d)target=null;
  hull--;chain=0;shake=16;flash('255,70,70',.3);S.hull();
  explode(d.x,d.y,'#8fa8b5',14,.9);pop(d.x,clamp(d.y,H*.5,H-160),'HULL BREACH','#ff5f5f',22);
  if(hull<=0)gameOver();}

/* ---------- flow ---------- */
function startGame(){
  score=0;hull=3;wave=1;chain=0;maxChain=0;correct=0;errors=0;playTime=0;bestWpm=0;
  drones=[];bolts=[];parts=[];pops=[];rings=[];targetId=null;target=null;ocT=0;shake=0;flashA=0;
  shipX=W/2;tilt=0;recent.length=0;
  wavePhase='intro';phaseT=2.2;toSpawn=countFor(1);spawnT=.6;paused=false;
  state='playing';body.dataset.state='playing';
  showBanner('WAVE 01','HOSTILES INBOUND');S.horn();
}
function gameOver(){
  state='over';body.dataset.state='over';S.over();shake=18;flash('255,70,70',.4);
  const nb=score>best;if(nb){best=score;try{localStorage.setItem('typefire_best',String(best))}catch(e){}}
  $('ovScore').textContent=score;$('ovBest').textContent=best;$('ovBestChip').textContent='BEST — '+best;
  $('ovWave').textContent=wave;$('ovAcc').textContent=accNow()+'%';$('ovWpm').textContent=bestWpm;

/* ---------- update ---------- */
function update(dt){
  tG+=dt;updStars(dt);
  streakT-=dt;if(streakT<=0&&state!=='paused'){streakT=rand(3.5,8);
    parts.push({kind:'streak',x:rand(W*.1,W*.9),y:-20,vx:rand(160,420)*(Math.random()<.5?1:-1),vy:rand(420,700),life:.9,max:.9});}
  if(state==='start'){ambT-=dt;
    if(ambT<=0){ambT=rand(1.4,2.8);if(drones.length<6)makePod(pickWord(ri(4,7)),{ambient:true,vy:H*.035});}
    for(let i=drones.length-1;i>=0;i--){const d=drones[i];d.y+=d.vy*dt;if(d.y>H+90)drones.splice(i,1);}
    decayFx(dt);return;}
  if(state==='paused'||state==='over'){decayFx(dt);return;}

  ocT=Math.max(0,ocT-dt);const spd=ocT>0?.45:1;
  if(wavePhase!=='run'){phaseT-=dt;if(phaseT<=0){wavePhase='run';
    if(wave%3===0&&hull<3){hull++;pop(W/2,H*.3,'HULL PATCHED +1','#7dffab',24);}}}
  else{
    playTime+=dt;const w=wpmNow();if(w>bestWpm)bestWpm=w;
    if(toSpawn>0){spawnT-=dt;
      if(spawnT<=0&&drones.filter(d=>!d.ambient&&d.type==='drone').length<conCap()){
        toSpawn--;const t=tierFor(wave);
        makePod(pickWord(ri(t[0],t[1])),{elite:Math.random()<Math.min(.3,.06+wave*.02)});
        spawnT=Math.max(.85,2.4-wave*.16)*rand(.7,1.3);}}
    else if(!drones.some(d=>!d.ambient)){
      const bonus=150+wave*50;score+=bonus;pop(W/2,H*.34,'WAVE CLEAR +'+bonus,'#ffb454',28);S.power();
      wavePhase='break';phaseT=1.9;wave++;toSpawn=countFor(wave);
      showBanner('WAVE '+String(wave).padStart(2,'0'),'HOSTILES INBOUND');S.horn();}
  }
  for(let i=drones.length-1;i>=0;i--){const d=drones[i];if(d.ambient)continue;
    d.y+=d.vy*spd*dt;d.x=d.baseX+Math.sin(tG*.9+d.ph)*Math.min(16,d.w*.15);
    d.hitT=Math.max(0,d.hitT-dt);
    if(d.y>breachY()){if(d.type==='crate'){puff(d.x,d.y,'#59e6cf');drones.splice(i,1);}else hullHit(d,i);}}
  const t=activeTarget();
  if(t){if(targetId!==t.id){targetId=t.id;if(target)relockT=.5;}}
  else targetId=null;
  target=t;
  const tx=t?clamp(t.x,70,W-70):W/2;
  shipX+=(tx-shipX)*Math.min(1,dt*6);
  tilt+=((t?(t.x-shipX)*.0015:0)-tilt)*Math.min(1,dt*8);
  mFlash=Math.max(0,mFlash-dt);
  if(Math.random()<.9)parts.push({kind:'px',x:shipX+rand(-4,4),y:shipY()+16,vx:rand(-12,12),vy:rand(70,130),life:.35,max:.35,size:rand(1.5,3.5),col:Math.random()<.5?'#ffb454':'#ff8a3d'});
  decayFx(dt);
}
function decayFx(dt){
  shake=Math.max(0,shake-shake*7*dt-8*dt);flashA=Math.max(0,flashA-dt*1.3);relockT=Math.max(0,relockT-dt);
  for(let i=bolts.length-1;i>=0;i--){bolts[i].life-=dt;if(bolts[i].life<=0)bolts.splice(i,1);}
  for(let i=parts.length-1;i>=0;i--){const p=parts[i];p.life-=dt;if(p.life<=0){parts.splice(i,1);continue;}
    p.x+=(p.vx||0)*dt;p.y+=(p.vy||0)*dt;p.vy=(p.vy||0)+(p.g||0)*dt;p.rot=(p.rot||0)+(p.vr||0)*dt;}
  for(let i=pops.length-1;i>=0;i--){const p=pops[i];p.life-=dt;p.y-=34*dt;if(p.life<=0)pops.splice(i,1);}
  for(let i=rings.length-1;i>=0;i--){const r=rings[i];r.life-=dt;r.r+=r.vr*dt;if(r.life<=0)rings.splice(i,1);}
}
