  lanes.push({row:row,dir,speed,type:'river',items,span:count*6});
}
function buildLevel(){
  lanes=[]; crypts=CRYPT_COLS.map(c=>({col:c,filled:false,fillT:0}));
  const m=Math.min(2.3, 1+(level-1)*0.16);
  addRiver(1, +1, .85*m, 3);
  addRiver(2, -1, 1.05*m, 3);
  addRoad(7, +1, 1.15*m, 'zombie', 2.6, 6, '#9db28b');
  addRoad(6, -1, 1.65*m, 'zombie', 2.2, 7, '#8aa67a');
  addRoad(5, +1, 3.1*m,  'hearse', 7.5, 2, null);
  addRoad(4, -1, 2.55*m, 'zombie', 1.9, 8, '#7f9c6f');
  addRoad(3, +1, 3.25*m, 'sprinter', 3.2, 5, '#c4756a');
  levelTime=Math.max(45, 60-(level-1)*5); timeLeft=levelTime; tickSec=-1;
  puddles=[]; for(let i=0;i<4;i++) puddles.push({row:(3+Math.floor(Math.random()*5)), x:rand(1,12), rx:rand(.5,.9)});
  lilies=[]; for(let i=0;i<5;i++) lilies.push({row:1+Math.floor(Math.random()*2), x:rand(.6,12.4), r:rand(.14,.2), rot:rand(0,7)});
  tufts=[]; for(let i=0;i<14;i++) tufts.push({x:rand(.2,12.8), o:rand(0,7), shroom:i%5===0});
  resetPlayer();
  floaters.push({x:W/2,y:boardY+cell*4.2,txt:'LEVEL '+level,t:0,life:1.6,size:Math.min(46,cell*.95),color:'#bfffd0'});
}
function resetPlayer(){ player.col=6; player.row=8; player.sx=6; player.sy=8; player.tx=6; player.ty=8; player.hopT=1; player.alive=true; player.face=0; }

function startGame(){
  score=0; lives=3; level=1; totalCrypts=0; charges=2; chargeT=0; chompCd=0;
  parts=[]; floaters=[]; rings=[]; bubbles=[]; paused=false;
  buildLevel(); state='play';
  startEl.classList.add('hidden'); overEl.classList.add('hidden');
  sfxStart();
}
function gameOver(){
  state='over';
  let nb=false;
  if(score>best){ best=score; nb=true; try{ localStorage.setItem('zombieFroggerBest',String(best)); }catch(e){} }
  document.getElementById('fScore').textContent=pad(score);
  document.getElementById('fCrypts').textContent=totalCrypts;
  document.getElementById('fLevel').textContent=level;
  document.getElementById('fBest').textContent=pad(best);
  document.getElementById('newBest').classList.toggle('hidden',!nb);
  document.getElementById('causeLine').textContent='CAUSE OF DEATH — '+lastCauseText;
  overEl.classList.remove('hidden');
}

/* ============================== ACTIONS ============================== */
function playerCenter(){
  const t=player.hopT<1 ? (t2=>t2*t2*(3-2*t2))(player.hopT) : 1;
  return { x: lerp(player.sx,player.tx,t)+.5, y: lerp(player.sy,player.ty,t)+.5 };
}
function tryMove(dc,dr){
  if(state!=='play'||paused||!player.alive) return;
  const nc=player.col+dc, nr=player.row+dr;
  if(nc<0||nc>=COLS||nr<0||nr>=ROWS) return;
  const t=player.hopT<1 ? (t2=>t2*t2*(3-2*t2))(player.hopT) : 1;
  player.sx = lerp(player.sx,player.tx,t); player.sy = lerp(player.sy,player.ty,t);
  player.tx=nc; player.ty=nr; player.hopT=0;
  if(dc!==0) player.face=dc;
  sfxHop();
}
function doChomp(){
  if(state!=='play'||paused||!player.alive||chompCd>0) return;
  if(charges<=0){ sfxDeny(); pipFlash=1; return; }
  charges--; chompCd=.5; chompAnim=1;
  const p=playerCenter(), px=boardX+p.x*cell, py=boardY+p.y*cell;
  rings.push({x:px,y:py,t:0}); sfxChomp();
  let kills=0;
  for(const lane of lanes){
    if(lane.type==='river') continue;
    if(Math.abs(lane.row+.5-p.y)>0.46) continue;
    const kept=[];
    for(const it of lane.items){
      if(Math.abs(it.x-p.x)<2.3){ gooBurst(boardX+it.x*cell, by(lane.row)+cell*.5, lane.tint||'#9db28b'); kills++; sfxGoo(); }
      else kept.push(it);
    }
    lane.items=kept;
  }
  if(kills){ score+=kills*25; floaters.push({x:px,y:py-cell*.6,txt:'GROSS! +'+(kills*25),t:0,life:1.1,size:Math.min(24,cell*.5),color:'#b8ff7d'}); }
}
function gooBurst(x,y,c){
  for(let i=0;i<14;i++){ const a=rand(0,7), s=rand(.5,3.4);
    parts.push({x,y,vx:Math.cos(a)*s*cell*.06,vy:Math.sin(a)*s*cell*.06-cell*.8,g:cell*4.2,life:rand(.4,.9),t:0,size:rand(cell*.05,cell*.13),color:pick([c,'#69c94f','#3f8f3a'])});
  }
}
function die(cause){
  if(!player.alive) return;
  player.alive=false; state='dying'; deathT=0; lives--; shake=.5;
  const p=playerCenter(); deathX=boardX+p.x*cell; deathY=boardY+p.y*cell;
  lastCauseText=pick(CAUSES[cause]||CAUSES.ZOMBIE);
  gooBurst(deathX,deathY,'#7fbf6a'); gooBurst(deathX,deathY,'#c9584a');
  floaters.push({x:W/2,y:boardY+cell*4.5,txt:lastCauseText,t:0,life:1.5,size:Math.min(30,cell*.6),color:'#ff9d7a'});
  sfxDeath();
}

/* ============================== UPDATE ============================== */
function update(dt){
  T+=dt;
  shake=Math.max(0,shake-dt*1.4); flash*=Math.pow(.02,dt); boltT=Math.max(0,boltT-dt);
  chompCd=Math.max(0,chompCd-dt); chompAnim=Math.max(0,chompAnim-dt/0.3); pipFlash=Math.max(0,pipFlash-dt*3);
  landPulse=Math.max(0,landPulse-dt/0.16);

  // lightning
  nextBolt-=dt;
  if(nextBolt<=0){ flash=1; boltT=.28; nextBolt=rand(7,16);
    boltPts=[]; let bxr=W*rand(.08,.9), byr=-10;
    for(let i=0;i<9;i++){ bxr+=rand(-W*.05,W*.05); byr+=H*rand(.06,.1); boltPts.push([bxr,byr]); }
    sfxRumble();
  }
  // fireflies
  for(const f of fireflies){ f.x+=Math.cos(T*f.sp*.2+f.ph)*f.sp*dt; f.y+=Math.sin(T*f.sp*.16+f.ph*2)*f.sp*dt*.7;
    if(f.x<-10)f.x=W+10; if(f.x>W+10)f.x=-10; if(f.y<-10)f.y=H+10; if(f.y>H+10)f.y=-10; }

  // particles / floaters / rings / bubbles
  for(let i=parts.length-1;i>=0;i--){ const p=parts[i]; p.t+=dt; if(p.t>p.life){parts.splice(i,1);continue;} p.x+=p.vx*dt*60; p.y+=p.vy*dt*60; p.vy+=(p.g||0)*dt; }
  for(let i=floaters.length-1;i>=0;i--){ const f=floaters[i]; f.t+=dt; if(f.t>f.life) floaters.splice(i,1); }
  for(let i=rings.length-1;i>=0;i--){ rings[i].t+=dt/0.42; if(rings[i].t>1) rings.splice(i,1); }
  if(bubbles.length<26 && Math.random()<dt*8) bubbles.push({row:1+(Math.random()*2|0), x:rand(.5,12.5), life:rand(1.2,2.4), t:0, r:rand(cell*.03,cell*.07)});
  for(let i=bubbles.length-1;i>=0;i--){ const b=bubbles[i]; b.t+=dt; if(b.t>b.life) bubbles.splice(i,1); }

  // lanes move (ambient in every state)
  const moving = !paused && (state==='play'||state==='start'||state==='dying'||state==='clear');
  if(moving) for(const lane of lanes){
    for(const it of lane.items){ it.x += lane.dir*lane.speed*dt;
      if(lane.dir>0 && it.x>14.5) it.x-=lane.span;
      if(lane.dir<0 && it.x<-1.5) it.x+=lane.span;
    }
  }

  // crypt fill pop + sparkles
  for(const c of crypts){ if(c.filled && c.fillT<1) c.fillT=Math.min(1,c.fillT+dt/0.4);
    if(c.filled && Math.random()<dt*1.1) parts.push({x:bx(c.col+.5)+rand(-cell*.2,cell*.2), y:by(0)+cell*.3, vx:0, vy:-cell*.5, g:0, life:rand(.5,.9), t:0, size:rand(cell*.04,cell*.08), color:pick(['#ffd24a','#7dffa0'])});
  }

  if(state==='play' && !paused){
    if(player.alive){
      if(player.hopT<1){ player.hopT=Math.min(1,player.hopT+dt/0.14); if(player.hopT>=1){ landPulse=1;
        const p=playerCenter(); for(let i=0;i<5;i++) parts.push({x:boardX+p.x*cell+rand(-8,8),y:boardY+p.y*cell+cell*.3,vx:rand(-40,40),vy:rand(-60,-10),g:200,life:.3,t:0,size:rand(1.5,3),color:'rgba(160,200,150,.5)'});
      }}
      player.blinkIn-=dt; if(player.blinkIn<=0){ player.blinkOn=.13; player.blinkIn=rand(1.8,4.5); }
      player.blinkOn=Math.max(0,player.blinkOn-dt);

      chargeT+=dt; if(chargeT>=6){ chargeT=0; charges=Math.min(3,charges+1); }

      timeLeft-=dt;
      const s=Math.ceil(timeLeft);
      if(s!==tickSec){ tickSec=s; if(s<=10 && s>0) sfxTick(); }
      if(timeLeft<=0){ die('TIME'); return; }

      const p=playerCenter(), px=p.x, py=p.y;

      // crypt check (row 0)
      if(Math.abs(py-0.5)<0.42){
        for(const c of crypts){ if(!c.filled && Math.abs(c.col+.5-px)<0.6){
          c.filled=true; totalCrypts++; score+=100; sfxSecure();
          floaters.push({x:bx(c.col+.5),y:by(0)+cell*.2,txt:'CRYPT SECURED +100',t:0,life:1.3,size:Math.min(22,cell*.48),color:'#7dffa0'});
          for(let i=0;i<16;i++){ const a=rand(0,7); parts.push({x:bx(c.col+.5),y:by(0)+cell*.5,vx:Math.cos(a)*90,vy:Math.sin(a)*90-40,g:260,life:rand(.5,.9),t:0,size:rand(2,4),color:pick(['#ffd24a','#7dffa0','#eaffd0'])}); }
          if(crypts.every(cc=>cc.filled)){ lastBonus=Math.ceil(Math.max(0,timeLeft))*5; score+=lastBonus; state='clear'; clearT=0; sfxClear(); }
          break;
        }}
      }

      // road hazard collision (continuous)
      for(const lane of lanes){
        if(lane.type==='river') continue;
        if(Math.abs(lane.row+.5-py)>0.46) continue;
        for(const it of lane.items){
          const hw=(it.type==='hearse'?0.87:0.3)+0.28;
          if(Math.abs(it.x-px)<hw){ die(it.type==='hearse'?'HEARSE':'ZOMBIE'); return; }
        }
      }

      // swamp support (when in water rows)
      if(player.row===1||player.row===2 || (py>0.6&&py<2.9&&player.hopT>=1)){
        let supported=false;
        for(const lane of lanes){ if(lane.type!=='river') continue;
          if(Math.abs(lane.row+.5-py)>0.46) continue;
          for(const it of lane.items) if(Math.abs(it.x-px)<0.72) supported=true;
        }
        if(!supported){ die('SWAMP'); return; }
      }
    }
  }

  if(state==='dying'){ deathT+=dt;
    if(deathT>0.95){ if(lives<=0) gameOver(); else { resetPlayer(); timeLeft=levelTime; tickSec=-1; charges=Math.max(1,charges); state='play'; } }
  }
  if(state==='clear'){ clearT+=dt; if(clearT>2.4){ level++; buildLevel(); state='play'; } }
}

  o=o||{}; const sq=o.sq||1, face=o.face||0;
  g.save(); g.translate(x,y); g.scale(1+(1-sq)*.55, sq);

  // feet
  g.fillStyle='#4aa55c'; g.strokeStyle='#16301f'; g.lineWidth=u*.05;
  for(const s of [-1,1]){ g.beginPath(); g.ellipse(s*u*.34,u*.4,u*.2,u*.1,0,0,7); g.fill(); g.stroke();
    g.beginPath(); for(let i=0;i<3;i++){ const a=s*(0.5+i*0.55); g.moveTo(s*u*.34+Math.cos(a)*u*.18,u*.4+Math.sin(a)*u*.08); g.lineTo(s*u*.34+Math.cos(a)*u*.26,u*.4+Math.sin(a)*u*.13);} g.stroke(); }

  g.strokeStyle='#5cc96f'; g.lineWidth=u*.09;
  g.beginPath(); g.moveTo(-u*.44,u*.02); g.lineTo(-u*.58,u*.3); g.stroke();
  g.fillStyle='#5cc96f'; g.beginPath(); g.arc(-u*.58,u*.3,u*.06,0,7); g.fill();

  g.fillStyle='#f4fff0'; g.strokeStyle='#16301f'; g.lineWidth=u*.05;

  if(o.blink){ g.fillStyle='#5cc96f'; g.beginPath(); g.arc(-u*.22,-u*.32,u*.19,Math.PI,0); g.fill(); }

function drawZombie(g,lane,it){
  const u=cell, x=boardX+it.x*u, y=by(lane.row)+u*.54;
  const t=T*lane.speed*1.8+it.phase, sw=Math.sin(t), dir=lane.dir, skin=lane.tint||'#8aa67a';
  g.save(); g.translate(x,y); if(lane.type==='sprinter') g.rotate(dir*.16);
  g.fillStyle='rgba(0,0,0,.3)'; g.beginPath(); g.ellipse(0,u*.34,u*.28,u*.09,0,0,7); g.fill();
  if(lane.type==='sprinter'){ g.strokeStyle='rgba(255,140,110,.3)'; g.lineWidth=u*.04;
    g.beginPath(); g.moveTo(-dir*u*.5,-u*.1); g.lineTo(-dir*u*.9,-u*.1); g.moveTo(-dir*u*.45,u*.08); g.lineTo(-dir*u*.85,u*.08); g.stroke(); }
  g.lineCap='round';
  // legs
  g.strokeStyle='#3d4c42'; g.lineWidth=u*.1;
  g.beginPath(); g.moveTo(-u*.09,u*.1); g.lineTo(-u*.09+sw*u*.11,u*.34); g.stroke();
  g.beginPath(); g.moveTo(u*.09,u*.1); g.lineTo(u*.09-sw*u*.11,u*.34); g.stroke();
  // torso (tattered)
  g.fillStyle='#2b343a'; rr(g,-u*.21,-u*.17,u*.42,u*.35,u*.1); g.fill();
  g.strokeStyle='rgba(0,0,0,.4)'; g.lineWidth=u*.03;
  g.beginPath(); g.moveTo(-u*.15,u*.16); g.lineTo(-u*.08,u*.1); g.lineTo(0,u*.17); g.lineTo(u*.1,u*.1); g.stroke();
  // arms reaching forward
  g.strokeStyle=skin; g.lineWidth=u*.09;
  const ay=-u*.08, ax=dir*u*.15;
  g.beginPath(); g.moveTo(ax,ay); g.lineTo(ax+dir*u*.26, ay-u*.06+sw*u*.06); g.stroke();
  g.beginPath(); g.moveTo(ax-dir*u*.07,ay+u*.05); g.lineTo(ax+dir*u*.19, ay+u*.15-sw*u*.05); g.stroke();
  g.fillStyle=skin; g.beginPath(); g.arc(ax+dir*u*.26,ay-u*.06+sw*u*.06,u*.05,0,7); g.arc(ax+dir*u*.19,ay+u*.15-sw*u*.05,u*.045,0,7); g.fill();
  // head
  g.fillStyle=skin; g.strokeStyle='rgba(0,0,0,.3)'; g.lineWidth=u*.03;
  g.beginPath(); g.arc(0,-u*.32,u*.18,0,7); g.fill(); g.stroke();
  if(it.band){ g.save(); g.translate(dir*u*.02,-u*.46); g.rotate(dir*.5); g.fillStyle='#d9cfa8'; g.fillRect(-u*.13,-u*.04,u*.26,u*.08); g.restore(); }
  // face: one eye, one X, drool mouth
  g.fillStyle='#101b12'; g.beginPath(); g.arc(dir*u*.07,-u*.34,u*.035,0,7); g.fill();
  g.strokeStyle='#101b12'; g.lineWidth=u*.03;
  g.beginPath(); g.moveTo(-dir*u*.08,-u*.37); g.lineTo(-dir*u*.02,-u*.31); g.moveTo(-dir*u*.02,-u*.37); g.lineTo(-dir*u*.08,-u*.31); g.stroke();
  g.beginPath(); g.arc(dir*u*.05,-u*.24,u*.06,.2,Math.PI-.4); g.stroke();
  g.restore();
}
function drawHearse(g,lane,it){
  const u=cell, x=boardX+it.x*u, y=by(lane.row)+u*.5, L=u*1.7, Hh=u*.52, dir=lane.dir;
  g.save(); g.translate(x,y);
  g.fillStyle='rgba(0,0,0,.35)'; g.beginPath(); g.ellipse(0,Hh*.6,L*.5,u*.09,0,0,7); g.fill();
  // headlight beam
  g.fillStyle='rgba(255,240,180,.1)';
  g.beginPath(); g.moveTo(dir*L/2,-u*.08); g.lineTo(dir*(L/2+u*1.3),-u*.26); g.lineTo(dir*(L/2+u*1.3),u*.18); g.closePath(); g.fill();
  // body

  const g=ctx;
  const bg=g.createLinearGradient(0,0,0,H);
  bg.addColorStop(0,'#03100b'); bg.addColorStop(.5,'#071511'); bg.addColorStop(1,'#0a1a13');
  g.fillStyle=bg; g.fillRect(0,0,W,H);

  drawAmbient(g);

  g.save();
  if(shake>0) g.translate((Math.random()-.5)*12*shake,(Math.random()-.5)*10*shake);
  drawBoardBg(g);
  for(const c of crypts) drawCrypt(g,c);
  for(const lane of lanes){ if(lane.type==='river'){ const row=lane.row+1; // sort: water then road top-down handled naturally
    for(const it of lane.items) drawTurtle(g,lane,it); } }
  const roadLanes=lanes.filter(l=>l.type!=='river').sort((a,b)=>a.row-b.row);
  for(const lane of roadLanes) for(const it of lane.items){
    if(it.type==='hearse') drawHearse(g,lane,it); else drawZombie(g,lane,it); }
  if(state!=='start' && state!=='over') drawPlayer(g);
  drawFx(g);
  g.restore();

  // fog drift over the board (top layer)
  for(let i=0;i<2;i++){ const fy=H*(.35+i*.3)+Math.sin(T*.1+i*4)*14;
    const fx=((T*(14+i*9))%(W+600))-300;
    const fg=g.createRadialGradient(fx,fy,0,fx,fy,W*.5);
    fg.addColorStop(0,'rgba(160,210,180,.05)'); fg.addColorStop(1,'rgba(160,210,180,0)');
    g.fillStyle=fg; g.fillRect(fx-W*.5,fy-W*.4,W*1.1,W*.8); }

  // vignette
  const vg=g.createRadialGradient(W/2,H/2,Math.min(W,H)*.35,W/2,H/2,Math.max(W,H)*.75);
  vg.addColorStop(0,'rgba(0,0,0,0)'); vg.addColorStop(1,'rgba(0,0,0,.42)');
  g.fillStyle=vg; g.fillRect(0,0,W,H);

  // lightning flash + low-time dread tint
  if(flash>0){ g.fillStyle='rgba(210,255,240,'+(flash*.16)+')'; g.fillRect(0,0,W,H); }
  if(state==='play'){ const f=clamp(timeLeft/levelTime,0,1);
    if(f<.35){ g.fillStyle='rgba(255,60,40,'+((.35-f)*.3)+')'; g.fillRect(0,0,W,H); } }

  if(state!=='start') drawHUD(g);
  drawBanners(g);

  // start-screen frog icon
  if(state==='start'){
    ictx.clearRect(0,0,140,140);
    ictx.fillStyle='rgba(0,0,0,.35)'; ictx.beginPath(); ictx.ellipse(70,112,38,9,0,0,7); ictx.fill();
    drawFrog(ictx,70,84+Math.sin(T*3)*4,66,{face:1,sq:1+Math.sin(T*3)*.05,blink:Math.sin(T*1.3)>.97,tongue:(Math.sin(T*.8)>.92)?(Math.sin(T*.8)-.92)*10:0});
  }
}

}
