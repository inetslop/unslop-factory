function tone(type,f0,f1,dur,vol,delay){
  if(!AC||!store.sound)return;delay=delay||0;
  const t=AC.currentTime+delay,o=AC.createOscillator(),g=AC.createGain();
  o.type=type;o.frequency.setValueAtTime(Math.max(1,f0),t);
  if(f1)o.frequency.exponentialRampToValueAtTime(Math.max(1,f1),t+dur);
  g.gain.setValueAtTime(vol,t);g.gain.exponentialRampToValueAtTime(0.0001,t+dur);
  o.connect(g);g.connect(master);o.start(t);o.stop(t+dur+0.03);
}
function noise(dur,vol,freq,delay){
  if(!AC||!store.sound||!noiseBuf)return;delay=delay||0;
  const t=AC.currentTime+delay,s=AC.createBufferSource();s.buffer=noiseBuf;
  const f=AC.createBiquadFilter();f.type='bandpass';f.frequency.value=freq;f.Q.value=0.9;
  const g=AC.createGain();g.gain.setValueAtTime(vol,t);g.gain.exponentialRampToValueAtTime(0.0001,t+dur);
  s.connect(f);f.connect(g);g.connect(master);s.start(t);s.stop(t+dur+0.03);
}

function spawnDust(gx,gy,n,tint){

}

    particles.push({type:'dot',x:X+Math.cos(a)*cell*0.3,y:Y+Math.sin(a)*cell*0.3,vx:Math.cos(a)*42,vy:-30-Math.random()*40,

function confetti(n){
  const cs=['#ffb648','#5ff2c8','#ff7a3d','#e9f4f1'];
  for(let i=0;i<n;i++)particles.push({type:'conf',x:ox+Math.random()*cols*cell,y:oy-10-Math.random()*40,
    vx:(Math.random()-.5)*220,vy:-60-Math.random()*160,g:380,rot:Math.random()*6.28,vr:(Math.random()-.5)*12,
    w:3+Math.random()*5,h:2+Math.random()*4,c:cs[i%4],ttl:1.8+Math.random()*1.2,life:0});
}
function stepParticles(dt){
  for(let i=particles.length-1;i>=0;i--){const p=particles[i];p.life+=dt;
    if(p.life>=p.ttl){particles.splice(i,1);continue;}
    if(p.type==='dot'){p.x+=p.vx*dt;p.y+=p.vy*dt;p.vy+=30*dt;}
    else if(p.type==='conf'){p.x+=p.vx*dt;p.y+=p.vy*dt;p.vy+=p.g*dt;p.rot+=p.vr*dt;p.vx*=0.995;}}
}
function drawParticles(){
  for(const p of particles){const k=1-p.life/p.ttl;
    if(p.type==='dot'){ctx.globalAlpha=k*0.7;ctx.fillStyle=p.c+'1)';ctx.beginPath();ctx.arc(p.x,p.y,p.r*k+0.4,0,7);ctx.fill();}
    else if(p.type==='ring'){const r=lerp(p.r0,p.r1,1-k);ctx.globalAlpha=k*0.9;ctx.strokeStyle=p.c+'1)';ctx.lineWidth=2.5;ctx.beginPath();ctx.arc(p.x,p.y,r,0,7);ctx.stroke();}
    else if(p.type==='conf'){ctx.globalAlpha=Math.min(1,k*1.6);ctx.save();ctx.translate(p.x,p.y);ctx.rotate(p.rot);ctx.fillStyle=p.c;ctx.fillRect(-p.w/2,-p.h/2,p.w,p.h);ctx.restore();}}
  ctx.globalAlpha=1;
}
