  if(shakeM>.1)ctx.translate(rand(-shakeM,shakeM),rand(-shakeM,shakeM));

  const pad=cell*.2,x0=bx-pad,y0=by-pad,w0=cell*COLS+pad*2,h0=cell*ROWS+pad*2;
  ctx.fillStyle='rgba(10,16,48,.78)';rr(x0,y0,w0,h0,14);ctx.fill();
  ctx.strokeStyle='#2c3f9d';ctx.lineWidth=3.5;rr(x0,y0,w0,h0,14);ctx.stroke();
  ctx.strokeStyle='rgba(90,130,255,.18)';ctx.lineWidth=1;rr(x0+6,y0+6,w0-12,h0-12,10);ctx.stroke();
  ctx.fillStyle='#3d55c4';
  [[x0+9,y0+9],[x0+w0-9,y0+9],[x0+9,y0+h0-9],[x0+w0-9,y0+h0-9]].forEach(p=>{ctx.beginPath();ctx.arc(p[0],p[1],3.5,0,TAU);ctx.fill();});

  // sockets
  for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++){
    ctx.fillStyle=(r+c)%2?'#0a1030':'#0c1338';
    rr(bx+c*cell+2.5,by+r*cell+2.5,cell-5,cell-5,5);ctx.fill(); }

  // tiles
  for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++){ const t=grid[r][c]; if(!t)continue;
    drawTile(t,bx+(t.cx+.5)*cell,by+(t.cy+.5)*cell); }

  // raid ghost
  if(raid){ const e=easeInOut(clamp(raid.t,0,1));
    const gx=lerp(raid.x,raid.tc,e), gy=raid.tr+Math.sin(raid.t*20)*.14;
    drawGhostShape(bx+(gx+.5)*cell,by+(gy+.5)*cell,cell*.42,raid.color,raid.tc>gx?.7:-.7); }

  // particles (additive)
  ctx.globalCompositeOperation='lighter';
  for(const p of parts){ const a=clamp(p.life/p.max,0,1);
    ctx.globalAlpha=a;ctx.fillStyle=p.color;
    ctx.beginPath();ctx.arc(p.x,p.y,p.r*(.5+a*.7),0,TAU);ctx.fill(); }
  ctx.globalAlpha=1;ctx.globalCompositeOperation='source-over';

  // cursor + pac (in play)
  if(mode==='play'){
    const pr=.45+.25*Math.sin(time*5);
    ctx.strokeStyle='rgba(255,217,59,'+(pr*.8).toFixed(3)+')';ctx.lineWidth=2.5;
    rr(bx+cursor.c*cell+3,by+cursor.r*cell+3,cell-6,cell-6,7);ctx.stroke();
    ctx.strokeStyle='rgba(255,217,59,.10)';ctx.lineWidth=1.5;
    [[0,1],[0,-1],[1,0],[-1,0]].forEach(d=>{const nr=cursor.r+d[0],nc=cursor.c+d[1];
      if(nr>=0&&nr<ROWS&&nc>=0&&nc<COLS){rr(bx+nc*cell+4,by+nr*cell+4,cell-8,cell-8,6);ctx.stroke();}});
    if(hoverCell&&!(hoverCell.r===cursor.r&&hoverCell.c===cursor.c)){
      ctx.strokeStyle='rgba(127,212,255,.35)';ctx.lineWidth=2;
      rr(bx+hoverCell.c*cell+4,by+hoverCell.r*cell+4,cell-8,cell-8,6);ctx.stroke(); }

    lunge.t+=0; // advanced in update
    const lt=clamp(lunge.t/.18,0,1),lo=Math.sin(lt*Math.PI)*cell*.12;
    const pxp=bx+(pac.x+.5)*cell+lunge.dx*lo, pyp=by+(pac.y+.5)*cell+lunge.dy*lo;
    const chompSpeed=(phase==='swap'||phase==='frenzy')?16:7;
    const chomp=(Math.sin(time*chompSpeed)+1)/2;
    ctx.save();ctx.shadowColor='rgba(255,225,77,.8)';ctx.shadowBlur=14;
    drawPacShape(pxp,pyp,cell*.4*(1+.05*Math.sin(time*6)),pac.face,chomp,'#ffe14d');ctx.restore(); }

  // popups
  for(const p of pops){ const k=p.t/p.dur, a=k<.7?1:1-(k-.7)/.3;
    ctx.globalAlpha=a; ctx.font=(p.big?'900 ':'800 ')+p.size+'px "Arial Black",Impact,sans-serif';
    ctx.textAlign='center'; ctx.lineWidth=p.big?6:3; ctx.strokeStyle='rgba(8,12,40,.9)';
    const yy=p.y-k*30;
    ctx.strokeText(p.txt,p.x,yy); ctx.fillStyle=p.color; ctx.fillText(p.txt,p.x,yy); }
  ctx.globalAlpha=1;

  // combo flash
  if(comboFx){ const k=comboFx.t;
    const sc=easeBack(clamp(k*4,0,1)); ctx.save();ctx.translate(bx+w0/2,by+h0*.5);ctx.scale(sc,sc);
    const a=k<.6?1:clamp(1-(k-.6)/.4,0,1); ctx.globalAlpha=a;
    const col=['#ffd93b','#ffb02e','#ff8ad8','#4de9ff'][Math.min(3,(comboFx.n-2)|0)];
    ctx.font='900 '+Math.round(cell*.7)+'px "Arial Black",Impact,sans-serif';ctx.textAlign='center';
    ctx.lineWidth=7;ctx.strokeStyle='rgba(8,12,40,.95)';
    const txt='COMBO ×'+comboFx.n;
    ctx.strokeText(txt,0,cell*.25);ctx.fillStyle=col;ctx.fillText(txt,0,cell*.25);
    ctx.restore(); }

  ctx.restore(); }

/* ================= main loop ================= */
let last=performance.now();
function frame(now){ requestAnimationFrame(frame);
  let dt=(now-last)/1000; last=now; if(dt>.05)dt=.05; time+=dt;
  shakeM*=Math.pow(.02,dt); if(shakeM<.05)shakeM=0;
  pacChompBoost=Math.max(0,pacChompBoost-dt);
  lunge.t+=dt;
  updateTweens(dt); updateParts(dt); updatePops(dt);
  if(comboFx){comboFx.t+=dt;if(comboFx.t>1)comboFx=null;}

  // pac eases to cursor unless mid-tween (frenzy)
  if(!pac.tweening&&mode==='play'){ const e=Math.min(1,dt*14);
    pac.x+=(cursor.c-pac.x)*e; pac.y+=(cursor.r-pac.y)*e; }

  if(mode==='play'){
    if(phase==='raid'&&raid){ raid.t+=dt/raid.dur; if(raid.t>=1)raidLand(); }
    else if(phase==='frenzy'){ frenzyWait-=dt; if(frenzyWait<=0)frenzyTick(); }
    else if(phase==='levelup'){ levelUpT+=dt; if(levelUpT>1.8)applyLevelUp(); }
    else if(phase==='shuffling'){ shufWait-=dt; if(shufWait<=0){phase='idle';updateHUD();} }
    else if(phase==='idle'){ raidT+=dt; if(raidT>=raidInt()){raidT=0;startRaid();} }
  }
  draw(); updateHUD(); }

fillBoard(); // attract-mode board behind the start screen
updateHUD();
requestAnimationFrame(frame);
