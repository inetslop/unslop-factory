/* ================= maze generation ================= */
function newMaze(){
  G.bricks.length=0;
  const x0=LW*.30,x1=LW*.70,y0=LH*.135,y1=LH*.865, cell=104;
  const cols=Math.max(3,Math.floor((x1-x0)/cell)), rows=Math.max(3,Math.floor((y1-y0)/cell));
  const gx=x0+((x1-x0)-cols*cell)/2, gy=y0+((y1-y0)-rows*cell)/2;
  const cells=[]; for(let c=0;c<cols;c++)for(let r=0;r<rows;r++)cells.push([c,r]);
  for(let i=cells.length-1;i>0;i--){const j=Math.random()*(i+1)|0;[cells[i],cells[j]]=[cells[j],cells[i]];}
  const n=Math.floor(cells.length*.58);
  for(let i=0;i<n;i++){
    const [c,r]=cells[i], cx=gx+c*cell, cy=gy+r*cell;
    const horiz=Math.random()<.5, th=rand(14,20), len=cell*rand(.52,.74);
    let w,h,jx,jy;
    if(horiz){w=len;h=th;jx=(cell-len)/2;jy=(cell-th)/2;}
    else{w=th;h=len;jx=(cell-th)/2;jy=(cell-len)/2;}
    let bx=cx+jx+rand(-1,1)*cell*.16;
    let by=cy+jy+rand(-1,1)*(cell-Math.max(w,h))*.35;
    bx=clamp(bx,cx+2,cx+cell-w-2); by=clamp(by,cy+2,cy+cell-h-2);
    G.bricks.push({x:bx,y:by,w,h,flash:0,born:G.time+i*.02+rand(0,.06)});
  }
  // keep the serve spot clear of bricks
  const b=G.ball;
  G.bricks=G.bricks.filter(k=>!(b.x>k.x-42&&b.x<k.x+k.w+42&&b.y>k.y-42&&b.y<k.y+k.h+42));
}

/* ================= flow ================= */
function primePoint(dir){
  const b=G.ball;
  b.x=LW/2;b.y=LH/2;b.vx=b.vy=0;b.sp=SERVE;b.pulse=0;
  G.trail.length=0; G.rally=0; syncHud();
  G.mazeNo++; newMaze();
  G.serveDir=dir; G.serveT=1.05;
}
function serve(dir){
  const b=G.ball, ang=rand(-.38,.38);
  b.sp=SERVE;
  b.vx=Math.cos(ang)*b.sp*dir; b.vy=Math.sin(ang)*b.sp;
}
function resetMatch(){
  G.scoreL=0;G.scoreR=0;G.mazeNo=0;G.matchBest=0;G.matchTop=0;
  pads.L.y=LH/2-PH/2; pads.R.y=LH/2-PH/2;
  G.shake=0;G.flash=0;G.trail.length=0;
  primePoint(Math.random()<.5?-1:1);
}

function onPoint(scorer){
  if(G.mode==='menu'){ primePoint(Math.random()<.5?-1:1); return; }
  const win=scorer==='L';
  if(win)G.scoreL++; else G.scoreR++;
  syncHud();

  sfx.score(win);
  G.shake=14; G.flash=.35; G.flashRGB=win?'61,245,200':'255,181,69';
  spark(win?LW-14:14, G.ball.y, win?'#3df5c8':'#ffb545', 28, 430, .6);
  if(G.scoreL>=WINS||G.scoreR>=WINS){ endMatch(scorer); return; }
  G.mode='point'; G.pointT=1.05; G.pendingDir=win?1:-1; // serve toward the side that lost
}

function playerControl(dt){
  const p=pads.L, spd=820;
  let dy=0; if(keys.up)dy-=1; if(keys.down)dy+=1;
  if(dy) p.y+=dy*spd*dt;
  else if(G.time-G.pointer.t<1.6){
    const ty=G.pointer.ly-PH/2, d=ty-p.y;
    p.y+=Math.sign(d)*Math.min(Math.abs(d),1500*dt);
  }
  p.y=clamp(p.y,8,LH-PH-8);
}
function predictY(t){
  let y=G.ball.y+G.ball.vy*t; const m=BR, span=LH-2*m;
  y=(y-m)%span; if(y<0)y+=span; return y+m;
}
function aiSide(side,dt){
  const p=pads[side], sk=SKILLS[G.diff], b=G.ball;
  const toward = side==='R' ? b.vx>0 : b.vx<0;
  let target=LH/2 + Math.sin(G.time*1.4+(side==='R'?0:2))*36;
  if(toward && Math.abs(b.vx)>1){
    const px = side==='R' ? LW-PX-PW-BR : PX+PW+BR;
    const t=(px-b.x)/(side==='R'?b.vx:-b.vx);
    target = (sk.pred>.5 && t>0) ? predictY(t*sk.pred) : b.y+b.vy*Math.min(t,.22);
    p.reactT-=dt;
    if(p.reactT<=0){ p.reactT=sk.react; const spF=Math.hypot(b.vx,b.vy)/MAXS; p.err=rand(-1,1)*sk.err*(.4+spF); }
    target+=p.err;
  }
  const d=target-(p.y+PH/2), maxV=sk.spd*(toward?1:.55);
  p.y+=clamp(d,-maxV*dt,maxV*dt);
  p.y=clamp(p.y,8,LH-PH-8);
}

/* ================= physics ================= */
function wallFx(x){ if(G.mode!=='menu')sfx.wall(); spark(x, x?0:LH, '#bfe9ff', 5, 170, .3); G.shake=Math.max(G.shake,2); }
function brickFx(x,y){ if(G.mode!=='menu')sfx.brick(); spark(x,y,'#9fe8ff',7,230,.35); G.shake=Math.max(G.shake,2.5); }
function padFx(x,y,col){ if(G.mode!=='menu')sfx.paddle(G.ball.sp); spark(x,y,col,10,300,.4); G.shake=Math.max(G.shake,4); }

function paddleHit(p,dir){
  const b=G.ball;
  if(dir<0&&b.vx>0)return; if(dir>0&&b.vx<0)return;
  const px=dir<0?PX:LW-PX-PW, py=p.y;
  const nx=clamp(b.x,px,px+PW), ny=clamp(b.y,py,py+PH);
  let dx=b.x-nx,dy=b.y-ny,d2=dx*dx+dy*dy;
  if(d2>BR*BR)return;
  const d=Math.sqrt(d2);
  let ux,uy;
  if(d<.001){ux=dir<0?1:-1;uy=0;}else{ux=dx/d;uy=dy/d;}
  b.x=nx+ux*BR; b.y=ny+uy*BR;
  const rel=clamp((b.y-(py+PH/2))/(PH/2+BR),-1,1);
  const ang=rel*(Math.PI/3.2);
  b.sp=Math.min(b.sp*GROW,MAXS);
  if(dir<0){b.vx=Math.cos(ang)*b.sp;b.vy=Math.sin(ang)*b.sp;}
  else{b.vx=-Math.cos(ang)*b.sp;b.vy=Math.sin(ang)*b.sp;}
  p.flash=1;
  if(G.mode==='rally'){
    G.rally++; syncHud();
    if(G.rally>G.matchBest)G.matchBest=G.rally;
    if(G.ball.sp>G.matchTop)G.matchTop=G.ball.sp;
    if(G.rally>G.bestRally){G.bestRally=G.rally;store.set('mazepong.rally',G.bestRally);}
  }
  padFx(nx,ny,dir<0?'#3df5c8':'#ffb545');
}
function brickHit(k){
  const b=G.ball;
  const nx=clamp(b.x,k.x,k.x+k.w), ny=clamp(b.y,k.y,k.y+k.h);
  let dx=b.x-nx,dy=b.y-ny,d2=dx*dx+dy*dy;
  if(d2>BR*BR)return false;
  const d=Math.sqrt(d2); let ux,uy;
  if(d<.001){
    const l=b.x-k.x,r=k.x+k.w-b.x,t=b.y-k.y,bt=k.y+k.h-b.y,m=Math.min(l,r,t,bt);
    if(m===l){b.x=k.x-BR;ux=1;uy=0;}else if(m===r){b.x=k.x+k.w+BR;ux=-1;uy=0;}
    else if(m===t){b.y=k.y-BR;ux=0;uy=1;}else{b.y=k.y+k.h+BR;ux=0;uy=-1;}
  }else{ux=dx/d;uy=dy/d;b.x=nx+ux*BR;b.y=ny+uy*BR;}
  const dot=b.vx*ux+b.vy*uy;
  if(dot<0){b.vx-=2*dot*ux;b.vy-=2*dot*uy;}
  b.vx*=1+rand(-.008,.008); b.vy*=1+rand(-.008,.008);
  if(Math.abs(b.vy)<35)b.vy=(b.vy===0?(Math.random()<.5?-1:1):Math.sign(b.vy))*rand(35,70);
  k.flash=1; brickHit2(nx,ny); return true;
}
function brickHit2(x,y){ brickFx(x,y); }

function ballPhysics(dt){
  const b=G.ball;
  let sp=Math.hypot(b.vx,b.vy); if(sp>G.matchTop)G.matchTop=sp;
  const n=Math.max(1,Math.ceil(sp*dt/(BR*.8))), sdt=dt/n;
  for(let i=0;i<n;i++){
    b.x+=b.vx*sdt; b.y+=b.vy*sdt;
    if(b.y<BR){b.y=BR;b.vy=Math.abs(b.vy);wallFx(b.x);}
    else if(b.y>LH-BR){b.y=LH-BR;b.vy=-Math.abs(b.vy);wallFx(b.x,1);}
    paddleHit(pads.L,-1); paddleHit(pads.R,1);
    for(const k of G.bricks){
      const g=clamp((G.time-k.born)/.24,0,1); if(g<.5)continue;
      brickHit(k);
    }
    if(b.x<-40){onPoint('R');return;}
    if(b.x>LW+40){onPoint('L');return;}
  }
  G.trail.push({x:b.x,y:b.y}); if(G.trail.length>16)G.trail.shift();
}

/* ================= particles ================= */
function spark(x,y,color,n,spd=260,life=.5){
  for(let i=0;i<n;i++){const a=rand(TAU),s=rand(.3,1)*spd;
    G.parts.push({x,y,vx:Math.cos(a)*s,vy:Math.sin(a)*s,life,max:life,size:rand(2,4.5),color});}
  if(G.parts.length>420)G.parts.splice(0,G.parts.length-420);
}
function confetti(){
  const cols=['#3df5c8','#ffb545','#ff6b4a','#eaf6f4'];
  for(let i=0;i<90;i++){const a=rand(TAU),s=rand(120,620);
    G.parts.push({x:LW/2,y:LH/2,vx:Math.cos(a)*s,vy:Math.sin(a)*s-180,life:rand(.5,1.1),max:1.1,size:rand(3,7),color:cols[i&3]});}
}
function updateParts(dt){
  const dr=Math.max(0,1-3*dt);
  for(let i=G.parts.length-1;i>=0;i--){const p=G.parts[i];
    p.life-=dt; if(p.life<=0){G.parts.splice(i,1);continue;}
    p.x+=p.vx*dt;p.y+=p.vy*dt;p.vx*=dr;p.vy*=dr;}
}

/* ================= update ================= */
function update(dt){
  G.time+=dt;
  G.shake=Math.max(0,G.shake-dt*42);
  G.flash=Math.max(0,G.flash-dt*1.7);
  for(const p of pads)p.flash=Math.max(0,p.flash-dt*5);
  for(const k of G.bricks)k.flash=Math.max(0,k.flash-dt*4);
  updateParts(dt);

  if(G.mode==='menu'){
    aiSide('L',dt); aiSide('R',dt);
    if(G.serveT>0){G.serveT-=dt;G.ball.pulse+=dt*6;} else ballPhysics(dt);
    return;
  }
  if(G.mode==='serve'){
    G.serveT-=dt; G.ball.pulse+=dt*6;
    playerControl(dt); aiSide('R',dt);
    if(G.serveT<=0){serve(G.serveDir);G.mode='rally';sfx.serve();}
    return;
  }
  if(G.mode==='point'){
    G.pointT-=dt; playerControl(dt); aiSide('R',dt);
    if(G.pointT<=0){primePoint(G.pendingDir);G.mode='serve';}
    return;
  }
  if(G.mode==='rally'){ playerControl(dt); aiSide('R',dt); ballPhysics(dt); }
}

  r=Math.min(r,w/2,h/2);
  ctx.beginPath();
  ctx.moveTo(x+r,y);ctx.arcTo(x+w,y,x+w,y+h,r);ctx.arcTo(x+w,y+h,x,y+h,r);
  ctx.arcTo(x,y+h,x,y,r);ctx.arcTo(x,y,x+w,y,r);ctx.closePath();
}

  rr(px+PW/2-3,p.y+10,6,PH-20,3); ctx.fill();

}

  // dust
  for(const d of dust){
    d.y-=d.v/innerHeight*.016; if(d.y<-.02){d.y=1.02;d.x=Math.random();}
    ctx.fillStyle='rgba(150,220,235,'+d.a+')';
    ctx.fillRect(d.x*w,d.y*h,d.s,d.s);
  }
  // soft central glow
  const cg=ctx.createRadialGradient(w/2,h/2,10,w/2,h/2,Math.max(w,h)*.55);
  cg.addColorStop(0,'rgba(46,160,170,.1)');cg.addColorStop(1,'rgba(0,0,0,0)');
  ctx.fillStyle=cg;ctx.fillRect(0,0,w,h);
  // field (with shake)
  const shx=G.shake?rand(-1,1)*G.shake:0, shy=G.shake?rand(-1,1)*G.shake*.6:0;
  ctx.setTransform(dpr*scale,0,0,dpr*scale,dpr*(ox+shx),dpr*(oy+shy));
  drawField();
}
