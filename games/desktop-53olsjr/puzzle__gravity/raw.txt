function loadLevel(i){
  const L=LEVELS[i];
  walls=new Set(); spikes=[]; pads=[]; crystals=[]; gates={}; gateMap=new Map();
  startPos=null; goalPos=null;
  for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++){
    const ch=L.g[r][c];
    if(ch==='#')walls.add(c+','+r);
    else if(ch==='S')startPos={x:c*T+T/2,y:r*T+T/2};
    else if(ch==='G')goalPos={x:c*T+T/2,y:r*T+T/2};
    else if(ch==='o')crystals.push({x:c*T+T/2,y:r*T+T/2,got:false,ph:rand(0,6)});
    else if(ch==='p'||ch==='q'){const g=ch==='p'?'A':'B';pads.push({x:c*T+T/2,y:r*T+T/2,r:r,g,hit:false,flash:0});}

const G=2000,DRAG=3.2,FRIC=14,REST=.12,R=11,SUB=1/240;
function collideAxis(axis){
  let sup=false,hitSpeed=0;
  const c0=Math.floor((pos.x-R)/T),c1=Math.floor((pos.x+R)/T),r0=Math.floor((pos.y-R)/T),r1=Math.floor((pos.y+R)/T);
  for(let r=r0;r<=r1;r++)for(let c=c0;c<=c1;c++){
    if(!solidTile(c,r))continue; const rx=c*T,ry=r*T;
    if(!circRect(pos.x,pos.y,R,rx,ry,T,T))continue;
    if(axis==='x'){
      if(Math.abs(vel.x)>hitSpeed)hitSpeed=Math.abs(vel.x);
      if(pos.x<rx+T/2){ pos.x=Math.min(pos.x,rx-R); if(gravDir.x>0.5)sup=true; }
      else{ pos.x=Math.max(pos.x,rx+T+R); if(gravDir.x<-0.5)sup=true; }
      vel.x=-vel.x*REST;
    }else{
      if(Math.abs(vel.y)>hitSpeed)hitSpeed=Math.abs(vel.y);
      if(pos.y<ry+T/2){ pos.y=Math.min(pos.y,ry-R); if(gravDir.y>0.5)sup=true; }
      else{ pos.y=Math.max(pos.y,ry+T+R); if(gravDir.y<-0.5)sup=true; }
      vel.y=-vel.y*REST;
    }
  }
  return {sup,hitSpeed};
}
function physics(dt){
  vel.x+=gravDir.x*G*dt; vel.y+=gravDir.y*G*dt;
  const dr=Math.exp(-DRAG*dt); vel.x*=dr; vel.y*=dr;
  pos.x+=vel.x*dt; let a=collideAxis('x');
  pos.y+=vel.y*dt; let b=collideAxis('y');
  if(a.sup||b.sup){ const f=Math.exp(-FRIC*dt); if(gravDir.y!==0)vel.x*=f; else vel.y*=f; }
  const hs=Math.max(a.hitSpeed,b.hitSpeed);
  if(hs>420 && performance.now()-lastThud>130){ lastThud=performance.now(); sfx.thud(hs); burst(pos.x-gravDir.x*R,pos.y-gravDir.y*R,'spark'); }
}
function die(){ if(state!=='playing')return; state='dying'; dieT=0; deaths++; sfx.death(); shake=10; redFlash=1; burst(pos.x,pos.y,'death'); }
function frameChecks(dt){
  // spikes
  const bc=Math.floor(pos.x/T),br=Math.floor(pos.y/T);
  for(const s of spikes){ if(Math.abs(s.c-bc)>1||Math.abs(s.r-br)>1)continue;
    if(circRect(pos.x,pos.y,R,s.c*T+8,s.r*T+8,T-16,T-16)){ die(); return; } }
  // pads
  for(const p of pads){ const ov=circRect(pos.x,pos.y,R,p.x-T/2+7,p.y-T/2+7,T-14,T-14);
    if(ov&&!p.hit){ p.hit=true; const g=gates[p.g]; g.open=!g.open; p.flash=1; sfx.pad(g.open); padBurst(p,g.open); buildLEDs(); }
    else if(!ov)p.hit=false; }
  // crystals
  for(const c of crystals){ if(c.got)continue; const d=Math.hypot(pos.x-c.x,pos.y-c.y);
    if(d<30){ c.got=true; runCrys++; sfx.crystal(); burst(c.x,c.y,'crystal'); }
    else if(d<46&&losClear(pos,c)){ vel.x+=(c.x-pos.x)/d*1500*dt; vel.y+=(c.y-pos.y)/d*1500*dt; } }
  // goal
  const gd=Math.hypot(pos.x-goalPos.x,pos.y-goalPos.y);
  if(gd<20){ clearLevel(); return; }
  if(gd<70&&losClear(pos,goalPos)){ vel.x+=(goalPos.x-pos.x)/gd*(1-gd/70)*3000*dt; vel.y+=(goalPos.y-pos.y)/gd*(1-gd/70)*3000*dt;
    const f=Math.exp(-3.2*dt); vel.x*=f; vel.y*=f; }
}

}

function drawBall(t){
  if(state==='dying'||state==='clear'||state==='win')return;
  for(let i=0;i<trail.length;i++){ const p=trail[i],a=i/trail.length;
    ctx.fillStyle='rgba(255,179,71,'+(a*.14)+')'; ctx.beginPath(); ctx.arc(p.x,p.y,R*(.3+.7*a),0,6.29); ctx.fill(); }
  const dx=-gravDir.x,dy=-gravDir.y,sw=Math.sin(t*3.2)*4;

function render(t){

/* ============================== main loop ============================== */
let last=performance.now();
function frame(now){
  requestAnimationFrame(frame);
  let dt=(now-last)/1000; last=now; if(dt>.05)dt=.05; const t=now/1000;

  const d=touchDir||(held.length?held[held.length-1]:null);
  if(state==='playing'&&d&&(d.x!==gravDir.x||d.y!==gravDir.y)){ gravDir=d; onGravChange(); }

  if(state==='playing'){
    acc+=dt; while(acc>=SUB){ physics(SUB); acc-=SUB; }
    frameChecks(dt); time+=dt; levelT+=dt;
    trail.push({x:pos.x,y:pos.y}); if(trail.length>16)trail.shift();
  } else if(state==='start'){ // idle scene behind title
    acc+=dt; while(acc>=SUB){ physics(SUB); acc-=SUB; }
    trail.push({x:pos.x,y:pos.y}); if(trail.length>16)trail.shift();
  } else if(state==='dying'){ dieT+=dt; time+=dt; levelT+=dt; if(dieT>.85){ spawnBall(); resetGates(); buildLEDs(); state='playing'; } }

  for(let i=particles.length-1;i>=0;i--){ const p=particles[i]; p.life-=dt;
    if(p.life<=0){ particles.splice(i,1); continue; }
    if(p.t!=='ring'){ p.x+=p.vx*dt; p.y+=p.vy*dt; p.vx+=gravDir.x*G*.35*dt; p.vy+=gravDir.y*G*.35*dt;
      const f=Math.exp(-2.2*dt); p.vx*=f; p.vy*=f; if(p.t==='shard')p.rot+=p.vr*dt; } }
  for(const k in gates){ const g=gates[k]; g.t+=( (g.open?1:0)-g.t )*Math.min(1,dt*9); }
  pads.forEach(p=>{ if(p.flash>0)p.flash=Math.max(0,p.flash-dt*2.4); });
  shake*=Math.exp(-6*dt); redFlash*=Math.exp(-5*dt); mintFlash*=Math.exp(-4.5*dt);

  render(t); hudUpdate();
}

/* ============================== boot ============================== */
resize(); loadLevel(0); resetGates(); spawnBall(); buildLEDs(); showScreen('scrStart');
requestAnimationFrame(frame);
