function update(dt){time+=dt;
  if(state==='pause')return;
  const slow=(state==='over')?.3:1;dt*=slow;
  if(state==='play')runT+=dt;
  genTo(Math.floor((cam.x+vw/S+400)/8));

  /* --- input (autopilot in menu) --- */
  let dir=0,jumpHeld=false;
  if(state==='menu'){dir=1;jumpHeld=true;
    const reach=ball.vx*.42+90;
    for(const s of spikes){if(s.x>ball.x&&s.x<ball.x+reach&&ball.grounded){ball.jbuf=.13;break;}}
    for(const g of gaps){if(g.x0>ball.x-R&&g.x0<ball.x+reach+40&&ball.grounded){ball.jbuf=.13;break;}}
  }else if(state==='play'){dir=(keys.right?1:0)-(keys.left?1:0);jumpHeld=keys.jumpHeld;}

  /* --- ball physics (fixed-ish substep) --- */
  const sub=Math.max(1,Math.ceil(dt/(1/120)));dt/=sub;
  for(let s=0;s<sub;s++){
    if(ball.dead)break;
    const b=ball;
    if(b.grounded){
      if(dir){if(b.vx*dir<MAX_RUN)b.vx+=dir*3200*dt;else if(b.vx*dir<1500)b.vx+=dir*800*dt;}
      else b.vx*=Math.exp(-.6*dt);
      if(keys.brake&&state==='play'){b.vx*=Math.exp(-4.5*dt);
        if(Math.abs(b.vx)>350&&Math.random()<.6)parts.push({x:b.x-b.vx*.01,y:surfY(b.x)-R*.4,vx:-b.vx*.2,vy:-80,g:500,life:.3,max:1,r:2,c:'#ffd27a'});}
      const sl=(surfY(b.x+20)-surfY(b.x-20))/40; /* gravity along slope */
      b.vx+=GRAV*sl/Math.sqrt(1+sl*sl)*dt*(dir?1:.9);
    }else{ if(dir)b.vx+=clamp(dir*1500*dt,-1500,1500)*(b.vx*dir>0?1:1.4); b.vx*=Math.exp(-.03*dt); }
    b.vx=clamp(b.vx,-1500,1500);
    const held=(state==='menu')?jumpHeld:keys.jumpHeld;
    b.vy+=GRAV*(b.vy<0&&held?.52:1)*dt;b.vy=Math.min(b.vy,1900);
    b.x+=b.vx*dt;b.y+=b.vy*dt;

    /* ground collision */
    const wasG=b.grounded;b.grounded=false;
    if(!inGap(b.x)){
      const gy=surfY(b.x),hL=inGap(b.x-14)?gy:surfY(b.x-14),hR=inGap(b.x+14)?gy:surfY(b.x+14);
      const t=(hR-hL)/28,il=1/Math.sqrt(1+t*t),nx=t*il,ny=-il;
      const pc=(b.y-gy)*ny+(0)*nx;
      if(pc<R){const pen=R-pc;b.x+=nx*pen;b.y+=ny*pen;
        const vn=b.vx*nx+b.vy*ny;
        if(vn<0){const imp=-vn,rest=imp>320?.3:0;
          b.vx-=(1+rest)*vn*nx;b.vy-=(1+rest)*vn*ny;
          if(imp>520){dust(b.x,b.y,Math.min(14,(imp/180)|0));b.sq=Math.min(.42,imp/1600);cam.shake=Math.min(9,imp*.008);
            if(!wasG)noiseB(.15,300,.12);}
          else if(!wasG&&imp>260){dust(b.x,b.y,4);b.sq=Math.min(.25,imp/2200);}
        }
      }
      b.grounded=(R-pc)>-4;
    }
    if(b.grounded){
      b.coyote=.1;b.angV+=(b.vx/R-b.angV)*Math.min(1,dt*14);
      if(Math.abs(b.vx)>650&&Math.random()<.35)dust(b.x-Math.sign(b.vx)*R*.7,b.y,1);
    }else b.angV*=Math.exp(-.25*dt);
    b.rot+=b.angV*dt;

    /* jump */
    b.coyote-=dt;b.jbuf-=dt;
    if(b.jbuf>0&&(b.grounded||b.coyote>0)){b.vy=-JUMP_V;b.grounded=false;b.coyote=0;b.jbuf=0;
      sfx.jump();dust(b.x,b.y,7);}

    /* gems */
    for(const g of gems){if(g.taken)continue;if(Math.abs(g.x-b.x)>60)continue;
      const dx=g.x-b.x,dy=(g.y+Math.sin(time*3+g.x*.05)*4)-b.y;
      if(dx*dx+dy*dy<(R+17)*(R+17)){g.taken=true;gemCount++;
        combo=(time-lastGemT<2.5)?combo+1:1;lastGemT=time;
        ftext(g.x,g.y-14,combo>1?'◆ x'+combo:'◆',combo>1?'#ffd98c':'#59e3c8');
        burst(g.x,g.y,8,'#59e3c8');sfx.gem(combo);updateHUD();}}

    /* spikes */
    if(b.invuln<=0){for(const sp of spikes){if(Math.abs(sp.x-b.x)>sp.w/2+R)continue;
      const x0=sp.x-sp.w/2+5,x1=sp.x+sp.w/2-5,gy=surfY(sp.x),y0=gy-sp.h+7,y1=gy+4;
      const cx=clamp(b.x,x0,x1),cy=clamp(b.y,y0,y1),dx=b.x-cx,dy=b.y-cy;
      if(dx*dx+dy*dy<(R*.82)*(R*.82)){loseLife('spike');break;}}}
    b.invuln-=dt;

    /* fall */
    if(inGap(b.x)&&b.y>1300)loseLife('fall');
  }
  ball.sq=Math.max(0,ball.sq-dt*3.2);
  ball.blinkT-=dt;if(ball.blinkT<-.12)ball.blinkT=2.4+Math.random()*3;

  /* checkpoint */
  if(state==='play'&&ball.grounded&&!inGap(ball.x)){
    const sl=(surfY(ball.x+20)-surfY(ball.x-20))/40;
    if(Math.abs(sl)<.3&&ball.invuln<=0&&time-(respawn._t||0)>.4&&ball.x>safeX+10){safeX=ball.x;respawn._t=time;}}

  /* distance & milestones */
  if(state==='play'){const d=Math.max(0,(ball.x-START_X)/PXM);distM=d;
    if(d>=nextMile){ftext(ball.x,ball.y-70,nextMile+' m','#ffd98c');sfx.mile();mileFlash=1;nextMile+=500;}
    $('distEl').textContent=Math.floor(distM);
    if(mileFlash>0){mileFlash-=dt*1.4;const el=$('distEl');el.classList.remove('flash');void el.offsetWidth;if(mileFlash>.6)el.classList.add('flash');}}

  /* camera */
  const look=clamp(ball.vx*.24,-140,280);
  cam.x+=(ball.x-vw/S*.38+look-cam.x)*Math.min(1,dt*7);
  const falling=inGap(ball.x)&&ball.vy>0;
  cam.y+=((falling?ball.y-160:ball.y-300)-cam.y)*Math.min(1,dt*(falling?12:5));
  cam.shake=Math.max(0,cam.shake-dt*22);

  /* particles / texts */
  for(let i=parts.length-1;i>=0;i--){const p=parts[i];p.life-=dt;if(p.life<=0){parts.splice(i,1);continue;}
    p.vy+=p.g*dt;p.x+=p.vx*dt;p.y+=p.vy*dt;}
  for(let i=texts.length-1;i>=0;i--){const t=texts[i];t.t-=dt*.85;t.y-=42*dt;if(t.t<=0)texts.splice(i,1);}
  if(Math.abs(ball.vx)>950&&!ball.dead)parts.push({x:ball.x-Math.sign(ball.vx)*R,y:ball.y+(Math.random()-.5)*R,
    vx:-ball.vx*.3,vy:(Math.random()-.5)*60,g:0,life:.25,max:1,r:1.5,c:'#ffe9c4',line:true});

  if((trimWorld._t||0)<time-2){trimWorld();trimWorld._t=time;}
  updateRoll();
}

  if(sunE>-.14){const sx=vw*.5+Math.cos(a)*vw*.42-cam.x*.02*S,sy=hz-sunE*vh*.36,al=clamp((sunE+.14)/.3,0,1);
    const sc=mix3([255,140,74],[255,223,154],clamp(sunE*1.6,0,1));
    g=ctx.createRadialGradient(sx,sy,4,sx,sy,190*S);g.addColorStop(0,sc.replace('rgb','rgba').replace(')',','+(.5*al)+')'));
    g.addColorStop(1,'rgba(255,180,94,0)');ctx.fillStyle=g;ctx.fillRect(sx-200*S,sy-200*S,400*S,400*S);
    ctx.globalAlpha=al;ctx.fillStyle=sc;ctx.beginPath();ctx.arc(sx,sy,44*S,0,7);ctx.fill();ctx.globalAlpha=1;}
  if(moonE>0){const mx=vw*.5+Math.cos(a+Math.PI)*vw*.42-cam.x*.02*S,my=hz-moonE*vh*.36;
    g=ctx.createRadialGradient(mx,my,2,mx,my,90*S);g.addColorStop(0,'rgba(223,234,242,.35)');g.addColorStop(1,'rgba(223,234,242,0)');
    ctx.fillStyle=g;ctx.fillRect(mx-100*S,my-100*S,200*S,200*S);
    ctx.globalAlpha=clamp(moonE*1.4,0,1);ctx.fillStyle='#dfeaf2';ctx.beginPath();ctx.arc(mx,my,28*S,0,7);ctx.fill();ctx.globalAlpha=1;}
  const starA=clamp(-sunE,0,1)*.9;
  if(starA>.02){for(const s of stars){const x=((s.u*vw-cam.x*.03*S)%vw+vw)%vw,y=s.v*hz*.85,
    al=starA*(.4+.6*Math.abs(Math.sin(time*s.tw+s.u*9)));
    ctx.fillStyle='rgba(230,240,255,'+al.toFixed(2)+')';ctx.fillRect(x,y,s.r*S,s.r*S);}}

  /* fog band */
  g=ctx.createLinearGradient(0,hz-70*S,0,hz+50*S);g.addColorStop(0,'rgba(0,0,0,0)');
  g.addColorStop(.5,hor.replace('rgb','rgba').replace(')',',.28)'));g.addColorStop(1,'rgba(0,0,0,0)');
  ctx.fillStyle=g;ctx.fillRect(0,hz-70*S,vw,120*S);

  /* far mountains */
  function ridge(par,amp1,amp2,f1,f2,baseY,col){ctx.beginPath();ctx.moveTo(-50,vh+50);
    for(let sx=-50;sx<=vw+60;sx+=26){const wx=cam.x*par+S*sx;
      const y=baseY-(vnoise(wx*f1,7)*amp1+vnoise(wx*f2,11)*amp2)*S;ctx.lineTo(sx,y);}
    ctx.lineTo(vw+60,vh+50);ctx.closePath();ctx.fillStyle=col;ctx.fill();}
  ridge(.12,150,52,.0024,.0085,hz-S*8,mix3(DK[i0][1],DK[i0][0],.55).replace('rgb','rgba').replace(')',',.75)'));
  ridge(.3,80,30,.004,.012,hz+vh*.07,'rgba(10,26,34,.9)');

  /* clouds */
  const cCol=hor.replace('rgb','rgba').replace(')',',.5)');
  for(const c of clouds){const cx=((c.u*(vw+400)+time*c.sp-cam.x*.2*S)%(vw+400)+(vw+400))%(vw+400)-200,
    cy=hz-vh*(.1+c.v)*S*1.2;ctx.fillStyle=cCol;
    for(let k=-1;k<=1;k++){ctx.beginPath();ctx.ellipse(cx+k*38*S*c.s,cy+(k?6:0)*S,44*S*c.s,15*S*c.s,0,0,7);ctx.fill();}}

  /* terrain */
  const x0=cam.x-60,x1=cam.x+vw/S+120,yBot=cam.y+vh/S+300;
  genTo(Math.floor(x1/8)+2);
  ctx.beginPath();let inG=true;
  for(let wx=x0;wx<=x1;wx+=8){const gg=inGap(wx),py=surfY(wx);
    if(gg&&!inG)ctx.lineTo((wx-cam.x)*S,(py-cam.y)*S);
    if(!gg&&inG){ctx.moveTo((wx-cam.x)*S,yBot-cam.y? (yBot-cam.y)*S:(yBot-cam.y)*S);}
    if(!gg){if(inG||wx===x0){ctx.moveTo((wx-cam.x)*S,(yBot-cam.y)*S);ctx.lineTo((wx-cam.x)*S,(py-cam.y)*S);}
      else ctx.lineTo((wx-cam.x)*S,(py-cam.y)*S);}
    inG=gg;}
  /* close & fill */
  const tCol1=mix3(DK[i0][0],[70,42,30],.5),tCol2=mix3(DK[i0][0],[28,16,14],.7);
  g=ctx.createLinearGradient(0,(300-cam.y)*S,0,yBot*S*0+(cam.y+vh/S)*S);g.addColorStop(0,tCol1);g.addColorStop(1,tCol2);
  ctx.fillStyle=g;
  /* rebuild properly for fill: segments */
  ctx.beginPath();
  let seg=[];const segs=[];inG=null;
  for(let wx=x0;wx<=x1;wx+=8){const gg=inGap(wx)?1:0,py=surfY(wx);
    if(gg!==inG){if(seg.length)segs.push(seg);seg=gg?[]:[[(wx-cam.x)*S,(py-cam.y)*S]];inG=gg;}
    else if(!gg)seg.push([(wx-cam.x)*S,(py-cam.y)*S]);}
  if(seg.length)segs.push(seg);
  for(const s of segs){if(s.length<2)continue;ctx.beginPath();ctx.moveTo(s[0][0],(yBot-cam.y)*S);
    for(const p of s)ctx.lineTo(p[0],p[1]);ctx.lineTo(s[s.length-1][0],(yBot-cam.y)*S);ctx.closePath();ctx.fill();}
  /* rim light */
  ctx.lineWidth=3*S;ctx.strokeStyle=hor.replace('rgb','rgba').replace(')',',.9)');
  ctx.shadowColor=hor;ctx.shadowBlur=8*S;
  for(const s of segs){if(s.length<2)continue;ctx.beginPath();ctx.moveTo(s[0][0],s[0][1]);
    for(let k=1;k<s.length;k++)ctx.lineTo(s[k][0],s[k][1]);ctx.stroke();}
  ctx.shadowBlur=0;

  /* gap mist */
  for(const gp of gaps){if(gp.x1<cam.x||gp.x0>cam.x+vw/S)continue;
    const gy=Math.min(surfY(gp.x0-6),surfY(gp.x1+6)),sy=(gy-cam.y)*S;
    g=ctx.createLinearGradient(0,sy,0,vh);g.addColorStop(0,'rgba(191,232,226,0)');
    g.addColorStop(.55,'rgba(191,232,226,.4)');g.addColorStop(1,'rgba(230,250,246,.75)');
    ctx.fillStyle=g;ctx.fillRect((gp.x0-cam.x)*S,sy,(gp.x1-gp.x0)*S,vh-sy);
    for(let k=0;k<3;k++){const fx=((gp.x0+Math.sin(time*.6+k*2.1)*40+ (k*97)%Math.max(60,gp.x1-gp.x0))-cam.x)*S,
      fy=(gy-cam.y)*S+40*S+Math.sin(time*.8+k)*10*S;
      ctx.fillStyle='rgba(230,250,246,.25)';ctx.beginPath();ctx.ellipse(fx,fy,50*S,14*S,0,0,7);ctx.fill();}}

  /* props */
  for(const p of props){const sx=(p.x-cam.x)*S;if(sx<-80||sx>vw+80)continue;
    const gy=surfY(p.x),sy=(gy-cam.y)*S,s=p.s*S;
    if(p.type===0){ctx.fillStyle='rgba(20,14,12,.75)';ctx.beginPath();
      ctx.moveTo(sx-16*s,sy);ctx.lineTo(sx-8*s,sy-20*s);ctx.lineTo(sx+4*s,sy-26*s);ctx.lineTo(sx+15*s,sy-12*s);ctx.lineTo(sx+20*s,sy);ctx.fill();}
    else if(p.type===1){ctx.fillStyle='#123c33';const w=9*s,h2=34*s;
      ctx.beginPath();ctx.roundRect?ctx.roundRect(sx-w/2,sy-h2,w,h2,4*s):ctx.rect(sx-w/2,sy-h2,w,h2);ctx.fill();
      ctx.beginPath();ctx.roundRect?ctx.roundRect(sx-w-8*s,sy-h2*.72,9*s,h2*.45,4*s):ctx.rect(sx-w-8*s,sy-h2*.72,9*s,h2*.45);ctx.fill();
      ctx.beginPath();ctx.roundRect?ctx.roundRect(sx-1*s,sy-h2*.62,9*s,h2*.4,4*s):ctx.rect(sx-1*s,sy-h2*.62,9*s,h2*.4);ctx.fill();}
    else{const pu=.6+.4*Math.sin(time*2+p.x);
      g=ctx.createRadialGradient(sx,sy-20*s,2,sx,sy-20*s,34*s);g.addColorStop(0,'rgba(89,227,200,'+(.4*pu)+')');g.addColorStop(1,'rgba(89,227,200,0)');
      ctx.fillStyle=g;ctx.fillRect(sx-36*s,sy-56*s,72*s,72*s);
      ctx.fillStyle='#59e3c8';ctx.beginPath();ctx.moveTo(sx,sy-40*s);ctx.lineTo(sx+8*s,sy-12*s);ctx.lineTo(sx,sy);
      ctx.lineTo(sx-8*s,sy-12*s);ctx.closePath();ctx.fill();}}

  /* gems */
  for(const gm of gems){if(gm.taken)continue;const sx=(gm.x-cam.x)*S;if(sx<-40||sx>vw+40)continue;
    const bob=Math.sin(time*3+gm.x*.05)*4,sy=(gm.y+bob-cam.y)*S,r=10*S;
    g=ctx.createRadialGradient(sx,sy,1,sx,sy,r*2.4);g.addColorStop(0,'rgba(89,227,200,.5)');g.addColorStop(1,'rgba(89,227,200,0)');
    ctx.fillStyle=g;ctx.fillRect(sx-r*2.4,sy-r*2.4,r*4.8,r*4.8);
    ctx.save();ctx.translate(sx,sy);ctx.rotate(Math.PI/4);
    ctx.fillStyle='#59e3c8';ctx.fillRect(-r*.7,-r*.7,r*1.4,r*1.4);
    ctx.fillStyle='rgba(255,255,255,.75)';ctx.fillRect(-r*.7,-r*.7,r*.6,r*.6);ctx.restore();}

  /* spikes */
  for(const sp of spikes){const sx=(sp.x-cam.x)*S;if(sx<-sp.w*S||sx>vw+sp.w*S)continue;
    const gy=surfY(sp.x),sy=(gy-cam.y)*S,n=Math.max(1,Math.round(sp.w/26)),tw=sp.w/n;
    for(let k=0;k<n;k++){const bx=(sp.x-sp.w/2+k*tw-cam.x)*S,bx2=bx+tw*S,hh=sp.h*(.85+.3*hashi(k*31+sp.x))*S;
      ctx.fillStyle='#241a17';ctx.beginPath();ctx.moveTo(bx,sy);ctx.lineTo(bx+tw*S/2,sy-hh);ctx.lineTo(bx2,sy);ctx.closePath();ctx.fill();
      ctx.strokeStyle='rgba(255,107,61,.8)';ctx.lineWidth=1.6*S;ctx.beginPath();
      ctx.moveTo(bx+tw*S*.3,sy-hh*.45);ctx.lineTo(bx+tw*S/2,sy-hh);ctx.lineTo(bx+tw*S*.7,sy-hh*.45);ctx.stroke();}}

  /* particles */
  for(const p of parts){const al=clamp(p.life/p.max,0,1),sx=(p.x-cam.x)*S,sy=(p.y-cam.y)*S;
    ctx.globalAlpha=al;if(p.line){ctx.strokeStyle=p.c;ctx.lineWidth=p.r*S;
      ctx.beginPath();ctx.moveTo(sx,sy);ctx.lineTo(sx-p.vx*.02*S,sy-p.vy*.02*S);ctx.stroke();}
    else{ctx.fillStyle=p.c;ctx.beginPath();ctx.arc(sx,sy,p.r*S*al+.5,0,7);ctx.fill();}}
  ctx.globalAlpha=1;

  /* ball */
  if(!ball.dead){
    const sx=(ball.x-cam.x)*S,sy=(ball.y-cam.y)*S,r=R*S;
    const flick=ball.invuln>0&&Math.floor(time*14)%2===0;
    ctx.globalAlpha=flick?.35:1;
    if(!inGap(ball.x)){const gy=surfY(ball.x),hgt=(gy-(ball.y+R));
      if(hgt<130){const sa=clamp(1-hgt/130,0,1)*.4;ctx.fillStyle='rgba(0,0,0,'+sa.toFixed(2)+')';
        ctx.beginPath();ctx.ellipse(sx,(gy-cam.y)*S+r*.25,r*(1.1-hgt/400),r*.28,0,0,7);ctx.fill();}}
    ctx.save();ctx.translate(sx,sy);ctx.scale(1+ball.sq*.6,1-ball.sq);
    const sg=ctx.createRadialGradient(-r*.4,-r*.45,r*.15,0,0,r);
    sg.addColorStop(0,'#ffcf9a');sg.addColorStop(.45,'#d97a2e');sg.addColorStop(1,'#5c2b10');
    ctx.fillStyle=sg;ctx.beginPath();ctx.arc(0,0,r,0,7);ctx.fill();
    ctx.strokeStyle='rgba(40,16,4,.8)';ctx.lineWidth=2*S;ctx.stroke();
    /* rotating etchings */
    ctx.save();ctx.rotate(ball.rot);ctx.strokeStyle='rgba(70,30,8,.55)';ctx.lineWidth=2.5*S;
    for(let k=0;k<3;k++){ctx.beginPath();ctx.arc(0,0,r*.62,k*2.094,k*2.094+1.1);ctx.stroke();}
    ctx.restore();
    /* specular */
    ctx.fillStyle='rgba(255,255,255,.5)';ctx.beginPath();ctx.ellipse(-r*.38,-r*.42,r*.26,r*.16,-.6,0,7);ctx.fill();
    /* face */
    const dx=clamp(ball.vx/600,-1,1),blink=ball.blinkT<.12;
    for(const e of [-1,1]){const ex=dx*r*.2+e*r*.3,ey=-r*.08;
      ctx.fillStyle='#2a1408';ctx.beginPath();
      ctx.ellipse(ex,ey,r*.16,blink?r*.03:r*.19,0,0,7);ctx.fill();
      if(!blink){ctx.fillStyle='#fff';ctx.beginPath();ctx.arc(ex+r*.05,ey-r*.06,r*.05,0,7);ctx.fill();}}
    ctx.strokeStyle='#2a1408';ctx.lineWidth=2.2*S;ctx.beginPath();
    if(Math.abs(ball.vx)>700){ctx.arc(dx*r*.15,r*.3,r*.1,0,7);}
    else{ctx.moveTo(-r*.14,r*.26);ctx.quadraticCurveTo(dx*r*.1,r*.4,r*.14+r*dx*.05,r*.26);}
    ctx.stroke();
    ctx.restore();ctx.globalAlpha=1;
  }

  /* floating texts */
  for(const t of texts){const sx=(t.x-cam.x)*S,sy=(t.y-cam.y)*S;
    ctx.globalAlpha=clamp(t.t,0,1);ctx.font='bold '+(22*S)+'px Impact,"Arial Black",sans-serif';
    ctx.textAlign='center';ctx.lineWidth=4*S;ctx.strokeStyle='rgba(10,8,4,.7)';
    ctx.strokeText(t.txt,sx,sy);ctx.fillStyle=t.c;ctx.fillText(t.txt,sx,sy);}
  ctx.globalAlpha=1;

  /* first-run hint */
  if(state==='play'&&runT<6){const sx=(ball.x-cam.x)*S,sy=(ball.y-R-52-cam.y)*S;
    ctx.globalAlpha=clamp(6-runT,0,1);ctx.font='bold '+(15*S)+'px "Trebuchet MS",sans-serif';ctx.textAlign='center';
    ctx.lineWidth=4;ctx.strokeStyle='rgba(10,8,4,.7)';ctx.strokeText('hold SPACE for a higher jump',sx,sy);
    ctx.fillStyle='#ffe9c4';ctx.fillText('hold SPACE for a higher jump',sx,sy);ctx.globalAlpha=1;}

  /* camera shake applied next frame via translate? apply now around world would be ideal — cheap trick: */
}
