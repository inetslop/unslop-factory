function paddleBounce(){const y0=paddle.y;

  const rel=clamp((ball.x-paddle.x)/(paddle.curW/2),-1,1),ang=rel*1.02;
  ball.speed=Math.min(MAXS,ball.speed*1.012+3);
  ball.vx=Math.sin(ang)*ball.speed+pv*.10;ball.vy=-Math.cos(ang)*ball.speed;
  const sp=Math.hypot(ball.vx,ball.vy)||1;ball.vx*=ball.speed/sp;ball.vy*=ball.speed/sp;
  if(-ball.vy<ball.speed*.25){const mn=ball.speed*.3;ball.vy=-mn;
   ball.vx=(ball.vx>=0?1:-1)*Math.sqrt(Math.max(0,ball.speed*ball.speed-mn*mn));}
  ball.y=y0-ball.r-.5;sPaddle();ring(ball.x,y0,'#5ee6ff');}}

function guardHitStep(){for(const g of guards){if(g.dead)continue;
 const nx=clamp(ball.x,g.x,g.x+g.w),ny=clamp(ball.y,g.y,g.y+g.h);
 const dx=ball.x-nx,dy=ball.y-ny,d2=dx*dx+dy*dy;
 if(d2>ball.r*ball.r)continue;
 const loud=phaseT<=0&&inCone(g,ball.x,ball.y);
 if(d2===0){ball.vy=-ball.vy;ball.y=(ball.y<g.cy)?g.y-ball.r-.5:g.y+g.h+ball.r+.5;}
 else{const d=Math.sqrt(d2),px=dx/d,py=dy/d,dot=ball.vx*px+ball.vy*py;
  ball.vx-=2*dot*px;ball.vy-=2*dot*py;ball.x=nx+px*(ball.r+.6);ball.y=ny+py*(ball.r+.6);}
 damage(g,loud,nx,ny);break;}}

  if(keys.ArrowLeft||keys.a||keys.A){paddle.x-=760*dt;moved=true;}
  if(keys.ArrowRight||keys.d||keys.D){paddle.x+=760*dt;moved=true;}

  const sp=Math.hypot(ball.vx,ball.vy)||1,n=sp>430?2:1,sdt=dt/n;
  for(let i=0;i<n;i++){ball.x+=ball.vx*sdt;ball.y+=ball.vy*sdt;walls();paddleBounce();guardHitStep();enforceAngle();}
  ball.trail.push({x:ball.x,y:ball.y});if(ball.trail.length>12)ball.trail.shift();

  if(phaseT<=0){for(const g of guards){if(g.dead)continue;if(inCone(g,ball.x,ball.y)){detecting=true;g.hotT=.3;}}}
  if(detecting){alarm+=dt*(56+lvl*2);sirenTick(dt);}else alarm-=dt*13;

  p.x+=(p.vx||0)*dt;p.y+=(p.vy||0)*dt;if(p.grav)p.vy+=p.grav*dt;if(p.rot!==undefined)p.rot+=p.vr*dt;}
 if(parts.length>320)parts.splice(0,parts.length-320);
 for(let i=texts.length-1;i>=0;i--){const t=texts[i];t.t+=dt;t.y-=28*dt;if(t.t>t.life)texts.splice(i,1);}}
