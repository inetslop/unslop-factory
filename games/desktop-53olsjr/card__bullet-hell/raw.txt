function genWave(n){ const ev=[]; let t=.6;
  const drones=Math.min(5+n,14);
  for(let i=0;i<drones;i++){ev.push({t,type:'drone'});t+=R(.5,.9)-Math.min(.25,n*.02);}
  if(n>=2){const o=Math.min(1+((n/2)|0),3);for(let i=0;i<o;i++)ev.push({t:R(1,4)+i*1.2,type:'orb'});}
  if(n>=3){const u=Math.min(1+(((n-3)/2)|0),3);for(let i=0;i<u;i++)ev.push({t:2+i*1.6,type:'turret',x:[.18,.5,.82][i]});}
  if(n%4===0)ev.push({t:Math.max(3,drones*.5),type:'boss'});
  ev.sort((a,b)=>a.t-b.t); return ev; }

function spawn(e){
  const n=wave.n,bs=Math.min(1.6,1+(n-1)*.04); e.bs=bs;e.t=0;e.flash=0;e.su=RI(0,3);
  if(e.type==='drone'){e.x=R(70,W-70);e.baseX=e.x;e.y=-30;e.vy=R(42,60)+n*2;e.hp=e.max=8+n*1.5;e.r=16;e.fr=R(1.2,2.2);e.ft=e.fr;e.face=RANKS[RI(2,9)];}
  else if(e.type==='orb'){e.cx=R(W*.3,W*.7);e.cy=R(90,150);e.rad=R(50,85);e.a=R(0,TAU);e.w=(Math.random()<.5?-1:1)*R(.6,1);e.x=e.cx;e.y=e.cy;e.hp=e.max=20+n*2;e.r=18;e.ft=2.4;}
  else if(e.type==='turret'){e.x=W*(e.x||.5);e.y=70;e.hp=e.max=34+n*4;e.r=22;e.ang=R(0,TAU);e.ft=.1;e.face='K';}
  else if(e.type==='boss'){e.x=W/2;e.y=-90;e.targetY=115;e.hp=e.max=260+n*45;e.r=40;e.pat=0;e.pt=2.2;e.stream=null;e.su=Math.random()<.5?1:3;e.face=(n%8===4)?'K':'Q';}
  enemies.push(e);
  if(e.type==='boss'){S.boss();banner('THE WARDEN');shake=10;}
}

function playCard(i){ if(state!=='play'||player.dead)return;
  const c=hand[i]; if(!c||flyers.some(f=>f.slot===i))return;
  kf[i]=1;
  const su=c.suit,v=c.v,cx=Math.cos(aim),cy=Math.sin(aim);
  if(su===0){ shots.push({x:player.x+cx*20,y:player.y+cy*20,vx:cx*950,vy:cy*950,r:7,dmg:v*1.8,pierce:24,big:true}); S.rail(); spark(player.x,player.y,'#dfeeff',6); }
  else if(su===1){ const n=3+((v/3)|0); for(let j=0;j<n;j++){const a=aim+(j-(n-1)/2)*.17; shots.push({x:player.x,y:player.y,vx:Math.cos(a)*640,vy:Math.sin(a)*640,r:5,dmg:4,pierce:1});} S.fan(); }
  else if(su===2){ beam={a:aim,t:.8+v*.09,w:12+v*1.6,dps:26+v*7}; humStart(); S.beamOn(); }
  else { const rad=95+v*13; ringFx(player.x,player.y,rad); shake=Math.max(shake,9); freeze=Math.max(freeze,.045); goldFlash=Math.max(goldFlash,.25);
    for(let i2=ebs.length-1;i2>=0;i2--){const b=ebs[i2],dx=b.x-player.x,dy=b.y-player.y; if(dx*dx+dy*dy<rad*rad){ebs.splice(i2,1);score+=Math.round(2*mult()); if(Math.random()<.4)spark(b.x,b.y,BC.amber,1);}}
    for(let i2=enemies.length-1;i2>=0;i2--){const e=enemies[i2],dx=e.x-player.x,dy=e.y-player.y; if(dx*dx+dy*dy<(rad+e.r)*(rad+e.r)){e.hp-=18+v*4;e.flash=.4;}}
    S.bomb(); }
  hand[i]=null; plays++;
  const ace=(plays%6===0)||pendingAce;
  queueDraw(i,ace);
}

  if(ace){ if(pendingAce){pendingAce=false;meter=0;} toast('JACKPOT · ACE DRAWN'); S.ace(); }

  player.hp--; syncHearts(); S.hit(); shake=13; redFlash=.5; meter=0; pendingAce=false;
  for(let i=ebs.length-1;i>=0;i--){const b=ebs[i],dx=b.x-player.x,dy=b.y-player.y; if(dx*dx+dy*dy<12100)ebs.splice(i,1);}

function updPlayer(dt){ const p=player; if(p.dead)return;
  if(touch.on){ const dx=touch.x-p.x,dy=(touch.y-70)-p.y; p.x+=dx*Math.min(1,dt*9); p.y+=dy*Math.min(1,dt*9);
    p.tilt+=((dx*.002)-p.tilt)*Math.min(1,dt*8);
    let nd=1e9,na=-Math.PI/2; for(const e of enemies){const d=(e.x-p.x)**2+(e.y-p.y)**2; if(d<nd){nd=d;na=Math.atan2(e.y-p.y,e.x-p.x);}} aim=na;
  } else { const acc=2600,maxv=340; let ax=0,ay=0;
    if(keys['a']||keys['arrowleft'])ax--; if(keys['d']||keys['arrowright'])ax++;
    if(keys['w']||keys['arrowup'])ay--; if(keys['s']||keys['arrowdown'])ay++;
    const l=Math.hypot(ax,ay)||1; p.vx+=ax/l*acc*dt; p.vy+=ay/l*acc*dt;
    const sp=Math.hypot(p.vx,p.vy); if(sp>maxv){p.vx*=maxv/sp;p.vy*=maxv/sp;}
    p.tilt+=(clamp(p.vx*.0009,-.3,.3)-p.tilt)*Math.min(1,dt*10);
    aim=mouse.used?Math.atan2(mouse.y-p.y,mouse.x-p.x):-Math.PI/2; }
  p.x=clamp(p.x,26,W-26); p.y=clamp(p.y,60,H-150);
  if(p.inv>0)p.inv-=dt;
}
function updEnemies(dt){ const alive=!player.dead;
  for(let i=enemies.length-1;i>=0;i--){ const e=enemies[i]; e.t+=dt; if(e.flash>0)e.flash-=dt*4;
    if(e.type==='drone'){ e.y+=e.vy*dt; e.x=e.baseX+Math.sin(e.t*1.8)*46;
      if(alive){ e.ft-=dt; if(e.ft<=0&&e.y>20&&e.y<H-170){ e.ft=Math.max(1,2-wave.n*.05); const a=Math.atan2(player.y-e.y,player.x-e.x),s=(160+wave.n*8)*e.bs; eb(e.x,e.y,Math.cos(a)*s,Math.sin(a)*s,5,BC.red);} }
      if(e.y>H+40)enemies.splice(i,1); }
    else if(e.type==='orb'){ e.a+=e.w*dt; e.x=e.cx+Math.cos(e.a)*e.rad; e.y=e.cy+Math.sin(e.a)*e.rad*.55;
      if(alive){ e.ft-=dt; if(e.ft<=0){ e.ft=Math.max(1.4,2.4-wave.n*.05); const k=10+Math.min(6,(wave.n/2)|0); for(let j=0;j<k;j++){const a=TAU*j/k+e.a,s=130*e.bs; eb(e.x,e.y,Math.cos(a)*s,Math.sin(a)*s,5,BC.amber);} } } }
    else if(e.type==='turret'){ e.ang+=dt*2.4;
      if(alive){ e.ft-=dt; if(e.ft<=0){ e.ft=.13; const s=150*e.bs; for(const o of [0,Math.PI])eb(e.x,e.y,Math.cos(e.ang+o)*s,Math.sin(e.ang+o)*s,4.5,BC.cyan); } } }
    else if(e.type==='boss'){
      if(e.y<e.targetY)e.y+=60*dt; else e.y=e.targetY+Math.sin(e.t*1.2)*12;
      e.x=W/2+Math.sin(e.t*.55)*(W*.3);
      const en=e.hp<e.max*.5,spd=en?1.3:1;
      if(alive){
        if(e.stream){ const st=e.stream; st.t+=dt; st.n-=dt;
          if(st.type==='fan'&&st.n<=0){ st.n=.32*(en?.7:1); for(let j=-3;j<=3;j++){const a=Math.atan2(player.y-e.y,player.x-e.x)+j*.16; eb(e.x,e.y+20,Math.cos(a)*200*spd*e.bs,Math.sin(a)*200*spd*e.bs,5.5,BC.red);} }
          else if(st.type==='spiral'&&st.n<=0){ st.n=.085*(en?.7:1); st.a+=.46; for(const o of [0,Math.PI])eb(e.x,e.y+10,Math.cos(st.a+o)*170*spd*e.bs,Math.sin(st.a+o)*170*spd*e.bs,4.5,BC.mag); }
          else if(st.type==='ring'&&st.n<=0){ st.n=.4; st.a+=.35; const k=16+(en?4:0); for(let j=0;j<k;j++){const a=TAU*j/k+st.a; eb(e.x,e.y,Math.cos(a)*150*spd*e.bs,Math.sin(a)*150*spd*e.bs,5,BC.amber);} }
          if(st.t>=st.dur)e.stream=null; }
        e.pt-=dt*(en?1.3:1);
        if(e.pt<=0&&!e.stream){ const p=e.pat%4; e.pat++;
          if(p===0){e.pt=2.6;e.stream={type:'ring',t:0,dur:1.2,n:0,a:R(0,TAU)};}
          else if(p===1){e.pt=3.0;e.stream={type:'fan',t:0,dur:1.4,n:0};}
          else if(p===2){e.pt=3.2;e.stream={type:'spiral',t:0,dur:1.8,n:0,a:Math.PI/2};}
          else { e.pt=2.6; const gap=RI(2,10); for(let j=0;j<14;j++){ if(j===gap||j===gap+1)continue; eb(40+j*((W-80)/13),-10,0,120*e.bs,5,BC.cyan);} } }
      } }
  }
}
function updShots(dt){ for(let i=shots.length-1;i>=0;i--){ const s=shots[i]; s.x+=s.vx*dt; s.y+=s.vy*dt;
    if(s.x<-40||s.x>W+40||s.y<-60||s.y>H+40){shots.splice(i,1);continue;}
    let dead=false;
    for(const e of enemies){ const rr2=s.r+e.r*.8,dx=s.x-e.x,dy=s.y-e.y;
      if(dx*dx+dy*dy<rr2*rr2){ e.hp-=s.dmg; e.flash=.35; spark(s.x,s.y,'#ffe9a3',4); s.pierce--; if(s.pierce<=0){dead=true;break;} } }
    if(dead)shots.splice(i,1);
  } }
function updBeam(dt){ if(!beam)return; beam.t-=dt; beam.a+=(aim-beam.a)*Math.min(1,dt*6);
  const bx=Math.cos(beam.a),by=Math.sin(beam.a);
  for(const e of enemies){ const dx=e.x-player.x,dy=e.y-player.y,t=dx*bx+dy*by; if(t<0)continue;
    const px=player.x+bx*t,py=player.y+by*t,rr2=beam.w/2+e.r*.7;
    if((e.x-px)**2+(e.y-py)**2<rr2*rr2){ e.hp-=beam.dps*dt; e.flash=Math.max(e.flash,.15); if(Math.random()<.3)spark(e.x,e.y,'#ffe9a3',1); } }
  if(beam.t<=0){beam=null;humStop();} }
function updEbs(dt){ const gr=26,hr=5;
  for(let i=ebs.length-1;i>=0;i--){ const b=ebs[i]; b.x+=b.vx*dt;b.y+=b.vy*dt;
    if(b.x<-30||b.x>W+30||b.y<-40||b.y>H+30){ebs.splice(i,1);continue;}
    if(player.dead)continue;
    const dx=b.x-player.x,dy=b.y-player.y,d2=dx*dx+dy*dy;
    if(d2<(hr+b.r)*(hr+b.r)){ ebs.splice(i,1); damagePlayer(); continue; }
    if(!b.g&&d2<gr*gr){ b.g=true; const prev=meter; meter=clamp(meter+6,0,100); score+=Math.round(5*mult());
      const d=Math.sqrt(d2)||1; spark(player.x+dx/d*15,player.y+dy/d*15,'#ffffff',1);
      if(hiGrazeT<=0){S.graze();hiGrazeT=.06;}
      if(prev<100&&meter>=100&&!pendingAce){ pendingAce=true; goldFlash=.35; S.ace(); toast('FULL FOCUS · NEXT CARD IS AN ACE'); } }
  } }

  if(wv.phase==='intro'){ wv.introT-=dt; if(wv.introT<=0){wv.phase='run';wv.t=0;banner('WAVE '+wv.n);} return; }

  while(wv.idx<wv.ev.length&&wv.ev[wv.idx].t<=wv.t){ spawn(Object.assign({},wv.ev[wv.idx])); wv.idx++; }
  if(wv.idx>=wv.ev.length&&enemies.length===0&&!player.dead){ const b=200*wv.n; score+=Math.round(b*mult());
    floater(W/2,H*.4,'WAVE CLEAR +'+Math.round(b*mult()).toLocaleString(),'#7ee0ff',20); S.wave();
    wv.n++; wv.phase='intro'; wv.introT=1.8; wv.ev=genWave(wv.n); wv.idx=0; } }

function updFloaters(dt){ for(let i=floaters.length-1;i>=0;i--){ const f=floaters[i]; f.t-=dt*.8; f.y-=28*dt; if(f.t<=0)floaters.splice(i,1); } }

function update(dt){
  hiGrazeT=Math.max(0,hiGrazeT-dt); shufflesCd=Math.max(0,shufflesCd-dt);
  for(let i=0;i<5;i++)kf[i]=Math.max(0,kf[i]-dt*6);
  shake=Math.max(0,shake-dt*38); redFlash=Math.max(0,redFlash-dt*1.6); goldFlash=Math.max(0,goldFlash-dt*1.2);
  for(const a of amb){ a.y+=a.vy*dt; a.r+=a.vr*dt; if(a.y>1.1){a.y=-.1;a.x=Math.random();} }
  updParts(dt); updFloaters(dt);
  if(state==='dying'){ player.dieT-=dt; if(player.dieT<=0)showOver(); }
  if(freeze>0){freeze-=dt;return;}
  if(state!=='play'&&state!=='dying')return;
  maxMult=Math.max(maxMult,mult());
  updPlayer(dt); updWave(dt); updEnemies(dt); updShots(dt); updBeam(dt); updEbs(dt);
  for(let i=enemies.length-1;i>=0;i--)if(enemies[i].hp<=0)killEnemy(i);
  updFlyers(dt);
}
