  for(let ty=ty0;ty<LH;ty++){const c=grid[ty][tx];if(c==='#'||c==='~')return{x:cx,y:ty*T};}

}

/* ================= physics ================= */
function collide(b,dt){
  b.x+=b.vx*dt;let wall=0;
  if(b.vx>0){const tx=Math.floor((b.x+b.w)/T);for(let ty=Math.floor((b.y+2)/T);ty<=Math.floor((b.y+b.h-2)/T);ty++)if(solid(tx,ty)){b.x=tx*T-b.w;b.vx=0;wall=1;break;}}
  else if(b.vx<0){const tx=Math.floor(b.x/T);for(let ty=Math.floor((b.y+2)/T);ty<=Math.floor((b.y+b.h-2)/T);ty++)if(solid(tx,ty)){b.x=(tx+1)*T;b.vx=0;wall=-1;break;}}
  const pb=b.y+b.h;b.y+=b.vy*dt;let gnd=false;
  if(b.vy>0){const ty=Math.floor((b.y+b.h)/T);for(let tx=Math.floor((b.x+2)/T);tx<=Math.floor((b.x+b.w-2)/T);tx++){const c=tileAt(tx,ty);if(c==='#'||(c==='-'&&pb<=ty*T+6)){b.y=ty*T-b.h;b.vy=0;gnd=true;break;}}}
  else if(b.vy<0){const ty=Math.floor(b.y/T);for(let tx=Math.floor((b.x+2)/T);tx<=Math.floor((b.x+b.w-2)/T);tx++)if(solid(tx,ty)){b.y=(ty+1)*T;b.vy=0;break;}}
  return{wall,gnd};
}

function stepWorld(dt){
  runT+=dt;
  const p=player;
  /* player */
  if(!p.dead){
    p.cool-=dt;p.inv-=dt;p.landT-=dt;p.recoil=Math.max(0,p.recoil-dt*6);
    const acc=p.onG?2800:1900,tvx=keys.L?-340:(keys.R?340:0);
    if(tvx)p.face=tvx>0?1:-1;
    p.vx+=clamp(tvx-p.vx,-acc*dt,acc*dt);
    const jP=!!keys.J;
    if(jP&&!prevJump)p.jbuf=.12;p.jbuf-=dt;
    if(p.onG){p.coyote=.1;p.jumps=0;}else p.coyote-=dt;
    if(p.jbuf>0){
      if(p.onG||p.coyote>0){p.vy=-650;p.jumps=1;p.coyote=0;p.jbuf=0;sfx.jump();dust(p.x+p.w/2,p.y+p.h,6);}
      else if(p.jumps<2){p.vy=-600;p.jumps=2;p.jbuf=0;sfx.djump();ring(p.x+p.w/2,p.y+p.h-8);}
    }
    if(prevJump&&!jP&&p.vy<-200)p.vy*=.45;
    prevJump=jP;
    p.vy=Math.min(900,p.vy+2100*dt);
    const wasG=p.onG,r=collide(p,dt);
    if(r.gnd&&!wasG){p.landT=.14;dust(p.x+p.w/2,p.y+p.h,5);}
    p.onG=r.gnd;
    if(p.onG&&Math.abs(p.vx)>60)p.runP+=dt*Math.abs(p.vx)*.055;
    p.dustT-=dt;if(p.onG&&Math.abs(p.vx)>320&&p.dustT<=0){p.dustT=.11;dust(p.x+p.w/2-p.face*8,p.y+p.h,2);}
    /* hazards */
    const x0=Math.floor((p.x+4)/T),x1=Math.floor((p.x+p.w-4)/T),y0=Math.floor((p.y+6)/T),y1=Math.floor((p.y+p.h-2)/T);
    for(let ty=y0;ty<=y1;ty++)for(let tx=x0;tx<=x1;tx++){
      const c=tileAt(tx,ty);
      if(c==='^'&&p.inv<=0){damagePlayer(12,(Math.sign(p.vx)||p.face)*180);if(!p.dead){p.vy=-560;sfx.spike();shake(5,.2);sparks(p.x+p.w/2,p.y+p.h,6,'#ff8c4a');}}
      if(c==='~'&&p.inv<=0){damagePlayer(34,(Math.sign(p.vx)||1)*120);if(!p.dead){p.vy=-950;shake(8,.3);sparks(p.x+p.w/2,p.y+p.h,14,'#ff6a1f');}}
    }
    if(p.y>LH*T+80)die();
    /* aim + fire */
    let ta=p.face>0?0:Math.PI;
    if(mouse.down){ta=Math.atan2(mouse.y+cam.y-(p.y+14),mouse.x+cam.x-(p.x+p.w/2));}
    p.aim=angLerp(p.aim,ta,Math.min(1,dt*14));
    const wantF=keys.ff||keys.fu||mouse.down;
    if(wantF&&p.cool<=0){
      let dx,dy;
      if(mouse.down){dx=Math.cos(ta);dy=Math.sin(ta);}
      else if(keys.fu){dx=0;dy=-1;}else{dx=p.face;dy=0;}
      p.cool=.16;p.recoil=1;
      bullets.push({x:p.x+p.w/2+dx*16,y:p.y+14+dy*14,vx:dx*760,vy:dy*760,t:.9});
      sfx.shoot();muzzle(p.x+p.w/2+dx*18,p.y+14+dy*18,dx,dy);p.vx-=dx*22;shake(1.2,.06);
    }
  }
  /* enemies */
  const px=p.x+p.w/2,py=p.y+18;
  for(const e of enemies){
    e.t+=dt;e.fl-=dt;
    if(e.kind==='drone'){
      e.vx=e.dir*58;e.vy=Math.min(900,e.vy+2100*dt);
      const r=collide(e,dt);
      if(r.wall)e.dir*=-1;
      if(r.gnd){const fx=e.x+(e.dir>0?e.w+3:-3);if(!floorT(Math.floor(fx/T),Math.floor((e.y+e.h+6)/T)))e.dir*=-1;}
    }else if(e.kind==='wisp'){
      let dx=px-e.x,dy=py-e.y,d=Math.hypot(dx,dy)||1;
      if(d<430){const tvx=dx/d*96+Math.sin(e.t*2.1)*36,tvy=dy/d*78+Math.cos(e.t*2.6)*44;e.vx+=(tvx-e.vx)*Math.min(1,dt*2.2);e.vy+=(tvy-e.vy)*Math.min(1,dt*2.2);}
      else{const hx=e.sx+Math.sin(e.t*.7)*46,hy=e.sy+Math.cos(e.t*.9)*30;e.vx+=((hx-e.x)-e.vx)*Math.min(1,dt*1.5);e.vy+=((hy-e.y)-e.vy)*Math.min(1,dt*1.5);}
      e.x=clamp(e.x+e.vx*dt,T,LW*T-T);e.y=clamp(e.y+e.vy*dt,T,LH*T-T);
    }else if(e.kind==='turret'){
      e.cool-=dt;const dx=px-(e.x+15),dy=py-(e.y+14),d=Math.hypot(dx,dy),ta2=Math.atan2(dy,dx);
      if(d<500){
        e.ang=angLerp(e.ang,ta2,Math.min(1,dt*7));
        if(e.cool<=0){const was=e.ch>0;e.ch+=dt;if(!was&&e.ch>0)sfx.tcharge();
          if(e.ch>.55){ebullets.push({x:e.x+15+Math.cos(e.ang)*22,y:e.y+14+Math.sin(e.ang)*22,vx:Math.cos(e.ang)*300,vy:Math.sin(e.ang)*300,t:3});e.cool=1.4;e.ch=0;sfx.tshot();sparks(e.x+15+Math.cos(e.ang)*24,e.y+14+Math.sin(e.ang)*24,5,'#ff8c4a');}}
        else e.ch=Math.max(0,e.ch-dt);
      }else{e.ch=Math.max(0,e.ch-dt*2);e.ang=angLerp(e.ang,-Math.PI/2,Math.min(1,dt*3));}
    }else if(e.kind==='heavy'){
      const moving=e.st==='dash'||e.st==='patrol';
      if(e.st==='patrol'){
        e.vx=e.dir*44;
        if(Math.abs(px-e.x)<310&&Math.abs(py-e.y)<95&&!p.dead){e.st='tele';e.t=.55;sfx.warn();}
      }else if(e.st==='tele'){e.vx=0;if((e.t-=dt)<=0){e.st='dash';e.t=.62;e.dir=Math.sign(px-e.x)||1;sfx.roar();shake(5,.2);}}
      else if(e.st==='dash'){e.vx=e.dir*470;e.aiT-=dt;if(e.aiT<=0){e.aiT=.05;P(e.x+R(4,10),e.y+R(6,e.h-6),0,0,.18,8,'rgba(255,90,40,.35)',0,true);}if((e.t-=dt)<=0){e.st='cool';e.t=1.1;}}
      else{e.vx=0;if((e.t-=dt)<=0)e.st='patrol';}
      e.vy=Math.min(900,e.vy+2100*dt);
      const r=collide(e,dt);
      if(r.wall){if(e.st==='dash'){e.st='cool';e.t=1.1;shake(6,.2);}else e.dir*=-1;}
      if(r.gnd&&moving){const fx=e.x+(e.dir>0?e.w+3:-3);if(!floorT(Math.floor(fx/T),Math.floor((e.y+e.h+6)/T)))e.dir*=-1;}
    }
    /* contact */
    if(!p.dead&&e.fl<=-9){} // noop
    const ecx=e.kind==='wisp'?e.x:e.x+e.w/2,ecy=e.kind==='wisp'?e.y:e.y+e.h/2;
    if(!p.dead&&Math.abs(px-ecx)<(e.kind==='wisp'?11:e.w/2)+p.w/2-4&&Math.abs(py-ecy)<(e.kind==='wisp'?11:e.h/2)+p.h/2-4){
      const dmg=e.kind==='drone'?12:e.kind==='wisp'?10:22;
      if(dmg)damagePlayer(dmg,Math.sign(px-ecx)*260||180);
    }
  }
  /* bullets */
  for(const b of bullets){
    b.t-=dt;b.x+=b.vx*dt;b.y+=b.vy*dt;
    const c=tileAt(Math.floor(b.x/T),Math.floor(b.y/T));
    if(c==='#'||c==='-'){sparks(b.x,b.y,4,'#ffd25e');b.t=0;}
    else for(const e of enemies){
      if(e.dead)continue;
      const ecx=e.kind==='wisp'?e.x:e.x+e.w/2,ecy=e.kind==='wisp'?e.y:e.y+e.h/2;
      if(Math.abs(b.x-ecx)<e.w/2+5&&Math.abs(b.y-ecy)<e.h/2+5){
        e.hp--;e.fl=.09;b.t=0;sfx.hit();sparks(b.x,b.y,5,'#ffe7c2');
        if(e.kind!=='wisp'){e.vx+=Math.sign(b.vx)*40;}
        if(e.hp<=0)killEnemy(e);
        break;
      }
    }
  }
  bullets=bullets.filter(b=>b.t>0);
  for(const b of ebullets){
    b.t-=dt;b.x+=b.vx*dt;b.y+=b.vy*dt;
    const c=tileAt(Math.floor(b.x/T),Math.floor(b.y/T));
    if(c==='#'||c==='-'){sparks(b.x,b.y,4,'#ff6a3d');b.t=0;continue;}
    if(!p.dead&&Math.abs(b.x-(p.x+p.w/2))<p.w/2+5&&Math.abs(b.y-(p.y+p.h/2))<p.h/2+5){damagePlayer(12,Math.sign(b.vx)*200);b.t=0;}
  }
  ebullets=ebullets.filter(b=>b.t>0);
  enemies=enemies.filter(e=>!e.dead);
  /* pickups */
  for(const k of pickups){
    k.t+=dt;
    if(Math.abs(k.x-px)<22&&Math.abs(k.y-(py-4))<30){
      if(k.k==='cell'){gate.got++;score+=150;pop(k.x,k.y-10,'+150');sfx.pick();sparks(k.x,k.y,8,'#ffab3f');
        if(gate.got>=gate.req&&!gate.open){gate.open=true;sfx.gate();showToast('GATE UNLOCKED','reach the ember arch');sparks(gate.x+32,gate.y+40,26,'#ffd25e');shake(6,.3);}}

      k.got=true;
    }
  }
  pickups=pickups.filter(k=>!k.got);
  /* gate */
  if(gate.open&&!p.dead&&p.x+p.w>gate.x+6&&p.x<gate.x+gate.w-6&&p.y+p.h>gate.y&&p.y<gate.y+gate.h){
    score+=500;pop(gate.x+32,gate.y-14,'+500');sparks(gate.x+32,gate.y+40,20,'#ffe7c2');
    if(levelIdx<LEVELS.length-1){transT=0;pendingLv=levelIdx+1;}else winGame();
  }
  /* ambient lava embers */
  const tx0=Math.floor(cam.x/T),tx1=Math.ceil((cam.x+VW)/T);
  for(let i=0;i<3;i++){const tx=tx0+(Math.random()*(tx1-tx0)|0),ty=LH-2;
    if(tileAt(tx,ty)==='~'&&Math.random()<.5)P(tx*T+R(4,28),(LH-2)*T+6,R(-10,10),R(-90,-40),R(.6,1.2),R(2,4),'rgba(255,150,70,.9)',-30,true);}
  partsUpdate(dt);popsUpdate(dt);fxDecay(dt);

  const tx=clamp(px-VW/2+p.face*70,0,LW*T-VW);
  cam.x+=(tx-cam.x)*Math.min(1,dt*6.5);
  cam.y+=(((LH*T-VH)/2)-cam.y)*Math.min(1,dt*4);
}
function killEnemy(e){
  e.dead=true;score+=e.pts;
  const ecx=e.kind==='wisp'?e.x:e.x+e.w/2,ecy=e.kind==='wisp'?e.y:e.y+e.h/2;
  pop(ecx,ecy-14,'+'+e.pts);boomFx(ecx,ecy);sfx.boom();freeze=Math.max(freeze,.05);shake(6,.25);
}
