let AC=null,master=null,muted=false,noiseBuf=null;
function audioInit(){
  if(AC){if(AC.state==='suspended')AC.resume();return;}
  try{AC=new (window.AudioContext||window.webkitAudioContext)();master=AC.createGain();master.gain.value=muted?0:.28;master.connect(AC.destination);}catch(e){}
}
function tone(type,f0,f1,dur,vol,delay){
  if(!AC||muted)return;delay=delay||0;const t0=AC.currentTime+delay;
  const o=AC.createOscillator(),g=AC.createGain();o.type=type;
  o.frequency.setValueAtTime(f0,t0);if(f1&&f1!==f0)o.frequency.exponentialRampToValueAtTime(Math.max(20,f1),t0+dur);
  g.gain.setValueAtTime(vol,t0);g.gain.exponentialRampToValueAtTime(.0001,t0+dur);
  o.connect(g);g.connect(master);o.start(t0);o.stop(t0+dur+.03);
}
function noise(dur,vol,fType,f0,f1,delay){
  if(!AC||muted)return;delay=delay||0;
  if(!noiseBuf){noiseBuf=AC.createBuffer(1,AC.sampleRate,AC.sampleRate);const d=noiseBuf.getChannelData(0);for(let i=0;i<d.length;i++)d[i]=Math.random()*2-1;}
  const t0=AC.currentTime+delay;const src=AC.createBufferSource();src.buffer=noiseBuf;src.loop=true;
  const f=AC.createBiquadFilter();f.type=fType;f.frequency.setValueAtTime(f0,t0);if(f1)f.frequency.exponentialRampToValueAtTime(Math.max(40,f1),t0+dur);
  const g=AC.createGain();g.gain.setValueAtTime(vol,t0);g.gain.exponentialRampToValueAtTime(.0001,t0+dur);
  src.connect(f);f.connect(g);g.connect(master);src.start(t0);src.stop(t0+dur+.03);
}
function sfx(n){
  switch(n){
    case 'move':tone('sine',150,95,.07,.22);noise(.05,.1,'lowpass',700,300);break;
    case 'push':tone('sine',110,55,.14,.38);noise(.1,.2,'bandpass',900,380);break;
    case 'bump':tone('triangle',82,48,.09,.5);break;
    case 'lock':tone('sine',660,660,.14,.3);tone('sine',990,990,.2,.24,.07);noise(.22,.1,'highpass',3000,5200,.05);break;
    case 'win':[523.25,659.25,783.99,1046.5].forEach((f,i)=>tone('triangle',f,f,.34,.28,i*.09));noise(.5,.09,'highpass',2500,6000,.12);break;
    case 'undo':noise(.18,.22,'bandpass',1400,300);tone('sine',300,170,.16,.14);break;
    case 'click':tone('square',950,700,.045,.13);break;
  }
}

function setAnim(e,x,y,dur){e.anim={sx:e.px,sy:e.py,tx:x,ty:y,t0:performance.now(),dur:dur||130};}
const animating=()=>!!(G.lvl.player.anim||G.lvl.cards.some(c=>c.anim));
function tickAnims(now){
  let done=true;
  const fin=e=>{if(!e.anim)return;const a=e.anim;let k=(now-a.t0)/a.dur;
    if(k>=1){e.px=a.tx;e.py=a.ty;e.anim=null;}
    else{k=1-Math.pow(1-k,3);e.px=a.sx+(a.tx-a.sx)*k;e.py=a.sy+(a.ty-a.sy)*k;done=false;}};
  fin(G.lvl.player);G.lvl.cards.forEach(fin);return done;
}
function pushUndo(){
  const P=G.lvl.player;
  G.undo.push({p:{x:P.x,y:P.y},c:G.lvl.cards.map(c=>({x:c.x,y:c.y})),m:G.moves,pu:G.pushes});
  if(G.undo.length>300)G.undo.shift();
}
function tryStep(dx,dy){
  if(G.state!=='play')return;
  if(animating()){G.buffer=[dx,dy];return;}
  const P=G.lvl.player,nx=P.x+dx,ny=P.y+dy;
  G.lastDir={dx,dy};
  if(isWall(nx,ny)){bump(dx,dy);return;}
  const card=cardAt(nx,ny);
  if(card){
    const cx=nx+dx,cy=ny+dy;
    if(isWall(cx,cy)||cardAt(cx,cy)){bump(dx,dy);return;}
    pushUndo();
    P.x=nx;P.y=ny;setAnim(P,nx,ny);
    card.x=cx;card.y=cy;setAnim(card,cx,cy,150);
    G.afterMove={card,wasLocked:card.locked};
    G.pushes++;G.moves++;sfx('push');
  }else{
    pushUndo();P.x=nx;P.y=ny;setAnim(P,nx,ny);G.moves++;sfx('move');
  }
  if(!G.started){G.started=true;G.t0=performance.now();}
  updateHUD();
}
function bump(dx,dy){G.bump={dx,dy,t0:performance.now()};G.shake={t:0,dur:.16,mag:2.5};sfx('bump');}
function cornered(x,y){const b=(a,c)=>isWall(a,c);return (b(x-1,y)&&b(x,y-1))||(b(x+1,y)&&b(x,y-1))||(b(x-1,y)&&b(x,y+1))||(b(x+1,y)&&b(x,y+1));}
function updateStuck(){
  let newly=false;
  G.lvl.cards.forEach(c=>{const was=c.stuck;if(isOnMatch(c)){c.stuck=false;return;}c.stuck=cornered(c.x,c.y);if(c.stuck&&!was)newly=true;});
  if(newly&&!G.stuckWarned){G.stuckWarned=true;toast('A card is pinned in a corner — press Z to undo');}
}
function handleAfter(){
  const a=G.afterMove;G.afterMove=null;
  if(a.card){const now=isOnMatch(a.card);a.card.locked=now;if(now&&!a.wasLocked)lockFX(a.card);}
  updateStuck();checkWin();
  if(G.buffer&&G.state==='play'&&!animating()){const b=G.buffer;G.buffer=null;tryStep(b[0],b[1]);}
}
function checkWin(){
  if(G.state!=='play')return;
  const win=G.lvl.slots.every(s=>{const c=cardAt(s.x,s.y);return c&&(s.suit==='O'||c.suit===s.suit);});
  if(!win)return;
  G.state='won';G.endT=performance.now();
  const prev=save.best[G.li];
  G.newBest=!!(prev&&G.moves<prev.m)||!prev;
  if(G.newBest)save.best[G.li]={m:G.moves,t:Math.round((G.endT-G.t0)/1000)};
  save.unlock=Math.max(save.unlock,Math.min(G.li+1,LEVELS.length-1));persist();updateHUD();
  sfx('win');winFX();
  G.winTimer=setTimeout(()=>{if(G.state==='won')showWinPanel();},950);
}
function undo(){
  if(G.state!=='play'||G.undo.length===0||animating())return;
  const s=G.undo.pop(),P=G.lvl.player;
  P.x=s.p.x;P.y=s.p.y;setAnim(P,P.x,P.y,120);
  G.lvl.cards.forEach((c,i)=>{c.x=s.c[i].x;c.y=s.c[i].y;setAnim(c,c.x,c.y,140);c.locked=isOnMatch(c);});
  G.moves=s.m;G.pushes=s.pu;updateStuck();sfx('undo');updateHUD();
}

  L.cards.forEach(c=>{for(let i=0;i<13;i++){const a=Math.random()*Math.PI*2,sp=90+Math.random()*180;
    confetti.push({x:cxOf(c),y:cyOf(c),vx:Math.cos(a)*sp,vy:Math.sin(a)*sp-160,rot:Math.random()*6,vr:(Math.random()-.5)*9,fl:Math.random()*7,suit:suits[Math.random()*4|0],life:0,max:3.2});}});
  for(let i=0;i<34;i++)confetti.push({x:boardX+Math.random()*L.w*cell,y:boardY-30-Math.random()*180,vx:(Math.random()-.5)*50,vy:60+Math.random()*120,rot:Math.random()*6,vr:(Math.random()-.5)*7,fl:Math.random()*7,suit:suits[Math.random()*4|0],life:0,max:5});
}

/* ============================== DRAW ============================== */
function draw(now,dt){
  ctx.drawImage(bgCv,0,0,W,H);
  // ambient sweep + drifting suits
  ctx.save();ctx.globalAlpha=.9;
  const sw=((now*.02)%(W+500))-250;
  ctx.save();ctx.translate(W/2,H/2);ctx.rotate(-.5);
  const sg=ctx.createLinearGradient(sw-140,0,sw+140,0);
  sg.addColorStop(0,'rgba(255,255,255,0)');sg.addColorStop(.5,'rgba(255,250,230,.035)');sg.addColorStop(1,'rgba(255,255,255,0)');
  ctx.fillStyle=sg;ctx.fillRect(sw-140,-H,280,H*2);ctx.restore();
  ctx.globalAlpha=.035;ctx.fillStyle='#f3ead2';ctx.textAlign='center';ctx.textBaseline='middle';
  ctx.font=Math.round(H*.7)+'px Georgia,serif';
  ctx.save();ctx.translate(W*.16,H*.8+Math.sin(now/4000)*14);ctx.rotate(-.2);ctx.fillText('♠',0,0);ctx.restore();
  ctx.save();ctx.translate(W*.87,H*.2+Math.cos(now/5200)*16);ctx.rotate(.25);ctx.fillText('♦',0,0);ctx.restore();
  ctx.restore();

  // world (shaken)
  let ox=0,oy=0;
  if(G.shake){G.shake.t+=dt;const k=G.shake.t/G.shake.dur;if(k>=1)G.shake=null;else{ox=(Math.random()*2-1)*G.shake.mag*(1-k);oy=(Math.random()*2-1)*G.shake.mag*(1-k);}}
  ctx.save();ctx.translate(ox,oy);
  const L=G.lvl;
  ctx.drawImage(tableCv,boardX-rim,boardY-rim,L.w*cell+rim*2,L.h*cell+rim*2);

  // slots
  const cw=cell*.76,ch=cell*.94;
  L.slots.forEach(s=>{
    const x=boardX+s.x*cell+(cell-cw)/2,y=boardY+s.y*cell+(cell-ch)/2;
    const filled=!!cardAt(s.x,s.y);
    ctx.save();ctx.setLineDash([Math.max(3,cell*.09),Math.max(3,cell*.07)]);
    const pulse=filled?0:Math.sin(now/450+s.x*1.3+s.y*2.1);
    ctx.strokeStyle='rgba(232,196,106,'+(filled?.85:.38+.2*pulse).toFixed(3)+')';
    ctx.lineWidth=Math.max(1.5,cell*.045);rr(ctx,x,y,cw,ch,ch*.12);ctx.stroke();ctx.setLineDash([]);
    if(s.suit!=='O'){ctx.fillStyle=SUIT_RED[s.suit]?'rgba(235,95,82,.9)':'rgba(222,228,240,.8)';
      ctx.font=Math.round(cell*.26)+'px Georgia,serif';ctx.textAlign='left';ctx.textBaseline='top';
      ctx.fillText(SUIT_GLYPH[s.suit],x+cw*.1,y+ch*.07);}
    else{ctx.fillStyle='rgba(232,196,106,.8)';ctx.font=Math.round(cell*.2)+'px Georgia,serif';ctx.textAlign='center';ctx.textBaseline='middle';ctx.fillText('◆',x+cw/2,y+ch/2);}
    ctx.restore();
  });

  // walls (stacks of card backs)
  L.walls.forEach(k=>{const [x,y]=k.split(',').map(Number);ctx.drawImage(stackSpr,boardX+x*cell,boardY+y*cell,cell,cell);});

  // cards
  L.cards.forEach(c=>{
    const x=boardX+c.px*cell+(cell-cw)/2,y=boardY+c.py*cell+(cell-ch)/2;
    ctx.save();
    ctx.fillStyle='rgba(0,0,0,.35)';rr(ctx,x+1.5,y+ch*.09,cw,ch,ch*.12);ctx.fill();
    const pk=(now-c.popT)/350,scale=pk<1?1+Math.sin(pk*Math.PI)*.1:1;
    ctx.translate(x+cw/2,y+ch/2);ctx.scale(scale,scale);ctx.translate(-cw/2,-ch/2);
    if(c.locked){ctx.shadowColor='rgba(240,210,120,.95)';ctx.shadowBlur=cell*.3;}
    ctx.drawImage(getFace(c),0,0,cw,ch);
    ctx.shadowBlur=0;
    if(c.stuck){const a=.4+.3*Math.sin(now/170);
      ctx.strokeStyle='rgba(226,80,74,'+a.toFixed(3)+')';ctx.lineWidth=Math.max(2,cell*.05);rr(ctx,-2,-2,cw+4,ch+4,ch*.14);ctx.stroke();
      const bx=cw/2,by=-ch*.18;ctx.fillStyle='#e2504a';ctx.beginPath();ctx.arc(bx,by,Math.max(7,cell*.13),0,Math.PI*2);ctx.fill();
      ctx.strokeStyle='rgba(0,0,0,.4)';ctx.lineWidth=1.5;ctx.stroke();
      ctx.fillStyle='#fff';ctx.font='bold '+Math.max(9,cell*.17)+'px Trebuchet MS,sans-serif';ctx.textAlign='center';ctx.textBaseline='middle';ctx.fillText('!',bx,by+.5);}
    ctx.restore();
  });

  // player chip
  {
    const P=L.player,r=cell*.42;
    let px=cxOf(P),py=cyOf(P)+Math.sin(now/380)*cell*.03;
    if(G.bump){const k=(now-G.bump.t0)/160;if(k>=1)G.bump=null;else{px+=Math.sin(k*Math.PI)*cell*.12*G.bump.dx;py+=Math.sin(k*Math.PI)*cell*.12*G.bump.dy;}}
    const target=P.anim?(P.anim.tx>P.anim.sx?1:P.anim.tx<P.anim.sx?-1:0):0;
    P.tilt+=(target*.1-P.tilt)*.25;
    ctx.fillStyle='rgba(0,0,0,.4)';ctx.beginPath();ctx.ellipse(px,py+r*.86,r*.9,r*.3,0,0,Math.PI*2);ctx.fill();
    ctx.save();ctx.translate(px,py);ctx.rotate(P.tilt);
    const cs=chipSpr.width/DPR;ctx.drawImage(chipSpr,-cs/2,-cs/2,cs,cs);ctx.restore();
  }

  // rings
  for(let i=rings.length-1;i>=0;i--){const g=rings[i];g.t+=dt;const k=g.t/g.dur;
    if(k>=1){rings.splice(i,1);continue;}
    const e=1-Math.pow(1-k,3);ctx.strokeStyle=g.col+(1-k).toFixed(3)+')';
    ctx.lineWidth=Math.max(2,cell*.09*(1-k));ctx.beginPath();ctx.arc(g.x,g.y,g.r0+(g.r1-g.r0)*e,0,Math.PI*2);ctx.stroke();}
  // dust
  for(let i=dust.length-1;i>=0;i--){const p=dust[i];p.life+=dt;if(p.life>p.max){dust.splice(i,1);continue;}
    p.x+=p.vx*dt;p.y+=p.vy*dt;const a=(1-p.life/p.max)*.5;
    ctx.fillStyle='rgba(190,205,175,'+a.toFixed(3)+')';ctx.beginPath();ctx.arc(p.x,p.y,p.r*(1+p.life*2),0,Math.PI*2);ctx.fill();}
  // sparkles
  for(let i=spark.length-1;i>=0;i--){const p=spark[i];p.life+=dt;if(p.life>p.max){spark.splice(i,1);continue;}
    p.x+=p.vx*dt;p.y+=p.vy*dt;p.vy+=150*dt;p.rot+=dt*8;const a=1-p.life/p.max;
    ctx.save();ctx.translate(p.x,p.y);ctx.rotate(p.rot);ctx.strokeStyle=p.col+a.toFixed(3)+')';ctx.lineWidth=1.6;
    ctx.beginPath();ctx.moveTo(-p.sz,0);ctx.lineTo(p.sz,0);ctx.moveTo(0,-p.sz);ctx.lineTo(0,p.sz);ctx.stroke();ctx.restore();}
  // confetti mini-cards
  for(let i=confetti.length-1;i>=0;i--){const p=confetti[i];p.life+=dt;
    if(p.life>p.max||boardY&&p.y>boardY+L.h*cell+80){confetti.splice(i,1);continue;}
    p.vy=Math.min(520,p.vy+360*dt);p.x+=(p.vx+Math.sin(p.life*4+p.fl)*26)*dt;p.y+=p.vy*dt;p.rot+=p.vr*dt;
    const w=cell*.3,h=cell*.42,a=p.life>p.max-.8?(p.max-p.life)/.8:1;
    ctx.save();ctx.globalAlpha=a;ctx.translate(p.x,p.y);ctx.rotate(p.rot);
    rr(ctx,-w/2,-h/2,w,h,2);ctx.fillStyle='#f7f1de';ctx.fill();ctx.strokeStyle='rgba(95,84,50,.5)';ctx.lineWidth=1;ctx.stroke();
    ctx.fillStyle=SUIT_RED[p.suit]?RED:BLK;ctx.font=Math.round(h*.6)+'px Georgia,serif';ctx.textAlign='center';ctx.textBaseline='middle';
    ctx.fillText(SUIT_GLYPH[p.suit],0,1);ctx.restore();}
  ctx.restore();
}

/* ============================== LOOP ============================== */
function frame(now){
  requestAnimationFrame(frame);
  const dt=Math.min(.05,(now-(lastT||now))/1000);lastT=now;
  if(G.state==='play'||G.state==='won'){
    if(tickAnims(now)&&G.afterMove)handleAfter();
  }
  if(G.state==='play'&&lastHeldStack.length){const d={u:[0,-1],d:[0,1],l:[-1,0],r:[1,0]}[lastHeldStack[lastHeldStack.length-1]];if(d)tryStep(d[0],d[1]);}
  draw(now,dt);
  if(G.started){const ms=G.state==='won'?G.endT-G.t0:now-G.t0;const s=fmtTime(ms);
    if($('stTime').textContent!==s)$('stTime').textContent=s;}
}
