    const sp=P.focus?120:252,k=Math.min(1,dt*12);
    P.vx+=((ix*sp)-P.vx)*k; P.vy+=((iy*sp)-P.vy)*k;
    P.x=clamp(P.x+P.vx*dt,14,WW-14); P.y=clamp(P.y+P.vy*dt,30,WH-16);
    if(P.inv>0)P.inv-=dt;
    P.fireCd-=dt;
    if(fireHeld&&P.fireCd<=0)shoot();
  }

  /* enemies */
  for(const e of enemies){
    e.t+=dt; if(e.flash>0)e.flash-=dt;
    if(e.type==='boss'){ updateBoss(e,dt); continue; }
    const d=diff();
    switch(e.type){
      case 'drone':
        e.x=e.bx+Math.sin(e.t*e.sw)*e.amp; e.y+=e.vy*dt;
        e.ft-=dt; if(e.ft<=0&&e.y>20){ e.ft=Math.max(.7,1.5-d*.6); aimedFire(e.x,e.y+8,95+80*d,wave>3?2:1,.2); }
        break;
      case 'dive': e.x+=e.vx*dt; e.y+=e.vy*dt; break;
      case 'spinner':
        e.y+=(26+20*d)*dt; e.a+=e.spin*dt; e.ft-=dt;
        while(e.ft<=0){ e.ft+=.11;
          for(let i=0;i<e.arms;i++){ const a=e.a+i*TAU/e.arms; eFire(e.x,e.y,Math.cos(a)*120,Math.sin(a)*120,3,'spark'); } }
        break;
      case 'turret':
        if(e.leaving){ e.y-=150*dt; break; }
        if(e.y<e.ty)e.y+=90*dt;
        else{
          e.x=e.tx+Math.sin(e.t*.8)*30; e.life-=dt; e.ft-=dt;
          if(e.ft<=0){ e.ft=1.7-d*.5; ringFire(e.x,e.y,10+Math.floor(d*6)+(Math.random()<.5?2:0),120+40*d,e.t*.7); }
          if(e.life<=0)e.leaving=true;
        }
        break;
      case 'waller':
        if(e.leaving){ e.y-=130*dt; break; }
        if(e.y<130)e.y+=80*dt;
        else{
          e.ft-=dt;
          if(e.ft<=0&&e.shots<3){ e.ft=1.5; e.shots++; fireWall(e.y+16,P.x+(Math.random()-.5)*70); }
          if(e.shots>=3)e.leaving=true;
        }
        break;
    }
    if(e.y>WH+50||e.y<-90||e.x<-70||e.x>WW+70)e.dead=true;
  }

  /* player bullets */
  for(const b of pB){
    if(b.dead)continue;
    b.x+=b.vx*dt; b.y+=b.vy*dt;
    if(b.y<-30||b.x<-20||b.x>WW+20){b.dead=true;continue;}
    for(const e of enemies){
      if(e.dead||e.dying)continue;
      const dx=b.x-e.x,dy=b.y-e.y,rr=e.r+5;
      if(dx*dx+dy*dy<rr*rr){
        b.dead=true; e.hp-=b.dmg; e.flash=.08; sparkAt(b.x,b.y,'#9ff8ff',2);
        if(e.type==='boss'){ if(e.hp<=0&&!e.dying){e.dying=true;freeze=Math.max(freeze,.35);sfx.boomB();} }
        else if(e.hp<=0)killEnemy(e);
        break;
      }
    }
  }

  /* bomb wave */
  if(bombFx){
    bombFx.t+=dt;
    bombFx.r=easeOut(Math.min(1,bombFx.t/.85))*560;
    for(const b of eB){ if(b.dead)continue; const dx=b.x-bombFx.x,dy=b.y-bombFx.y;
      if(dx*dx+dy*dy<bombFx.r*bombFx.r){b.dead=true;addScore(10);sparkAt(b.x,b.y,'#ffd23f',1);} }
    for(const e of enemies){ if(e.dead||e.dying)continue; const dx=e.x-bombFx.x,dy=e.y-bombFx.y;
      if(dx*dx+dy*dy<bombFx.r*bombFx.r){ e.hp-=90*dt; e.flash=.12; if(e.type!=='boss'&&e.hp<=0)killEnemy(e); } }
    shake=Math.max(shake,6*(1-bombFx.t/.85));
    if(bombFx.t>.85)bombFx=null;
  }

  /* enemy bullets + player collision */
  for(const b of eB){
    if(b.dead)continue;
    b.t+=dt; b.x+=b.vx*dt; b.y+=b.vy*dt;
    if(b.x<-30||b.x>WW+30||b.y<-40||b.y>WH+30){b.dead=true;continue;}
    if(P.alive){
      const dx=b.x-P.x,dy=b.y-P.y,d2=dx*dx+dy*dy;
      const hr=b.r+P.r;
      if(d2<hr*hr){ b.dead=true; hitPlayer(); }
      else if(!b.grazed&&d2<256){ b.grazed=true; grazeC++; addScore(25); sfx.graze(); sparkAt(b.x,b.y,'#ffffff',1); }
    }
  }

  /* enemy body vs player */
  if(P.alive&&P.inv<=0){
    for(const e of enemies){
      if(e.dead||e.dying)continue;
      const dx=e.x-P.x,dy=e.y-P.y,rr=e.r+P.r;
      if(dx*dx+dy*dy<rr*rr){ hitPlayer(); e.hp-=30; if(e.type!=='boss'&&e.hp<=0)killEnemy(e); break; }
    }
  }

  /* drops */
  for(const dr of drops){
    dr.y+=dr.vy*dt; dr.t+=dt;
    if(dr.y>WH+24){dr.dead=true;continue;}
    const dx=dr.x-P.x,dy=dr.y-P.y;
    if(P.alive&&dx*dx+dy*dy<19*19){
      dr.dead=true; sfx.pickup(); sparkAt(dr.x,dr.y,'#7df6ff',8);
      if(P.power<8){ P.power++; sfx.power(); } else addScore(750);
    }
  }

  /* particles */
  for(const p of parts){
    p.life-=dt;
    if(p.kind==='spark'){p.x+=p.vx*dt;p.y+=p.vy*dt;p.vx*=1-3*dt;p.vy*=1-3*dt;}
    else if(p.kind==='ring')p.r+=p.vr*dt;
  }

  compact(enemies);compact(pB);compact(eB);compact(drops);
  for(let i=parts.length-1;i>=0;i--)if(parts[i].life<=0)parts.splice(i,1);

  if(bannerObj){bannerObj.t+=dt;if(bannerObj.t>bannerObj.dur)bannerObj=null;}

  if(!P.alive){ overDelay-=dt; if(overDelay<=0)gameOver(); }
}
function compact(a){ let w=0; for(let i=0;i<a.length;i++)if(!a[i].dead)a[w++]=a[i]; a.length=w; }

function updateBoss(e,dt){
  if(e.dying){
    e.dieT-=dt; e.boomT-=dt;
    if(e.boomT<=0){ e.boomT=.13; boom(e.x+(Math.random()-.5)*72,e.y+(Math.random()-.5)*54,.7+Math.random(),'#ffb648',false); }
    if(e.dieT<=0){
      e.dead=true; bossRef=null;
      addScore(5000+wave*400); bombStock=Math.min(4,bombStock+1);
      banner('WARDEN DOWN',2.2,'gold'); sfx.boomB(); shake=Math.max(shake,14);
      for(let i=0;i<3;i++)drops.push({x:e.x-40+i*40,y:e.y,vy:46,t:Math.random()*9,dead:false});
    }
    return;
  }
  if(e.y<110){ e.y=Math.min(110,e.y+80*dt); return; }
  const frac=e.hp/e.maxhp, ph=frac>.62?0:frac>.3?1:2;
  e.x=WW/2+Math.sin(e.t*(0.45+ph*0.12))*(76-ph*8);
  const d=diff();
  if(ph===0){
    e.ft-=dt;
    if(e.ft<=0){ e.ft=.95; e.alt=!e.alt;
      if(e.alt)aimedFire(e.x,e.y+24,175,5,.24);
      else ringFire(e.x,e.y,13+Math.floor(d*4),135,e.t*.6); }
  }else if(ph===1){
    e.sa+=dt*2.6; e.sft-=dt;
    while(e.sft<=0){ e.sft+=.085; const a=e.sa;
      eFire(e.x,e.y+6,Math.cos(a)*150,Math.sin(a)*150,3,'spark');
      eFire(e.x,e.y+6,Math.cos(a+Math.PI)*150,Math.sin(a+Math.PI)*150,3,'spark'); }
    e.wt-=dt; if(e.wt<=0){e.wt=2.7;fireWall(e.y+30,P.x+(Math.random()-.5)*60);}
  }else{
    e.rft-=dt; if(e.rft<=0){e.rft=.75;ringFire(e.x,e.y,18,150+30*d,e.t*.9,'gold');}
    e.aft-=dt; if(e.aft<=0){e.aft=.16;aimedFire(e.x,e.y+24,210,3,.14,'spark');}
    e.ct-=dt;  if(e.ct<=0){e.ct=.5;eFire(20+Math.random()*(WW-40),-10,0,170+80*d,4.5,'gold');}
  }
}

function updateMenu(dt){
  menuSpawnT-=dt;
  if(menuSpawnT<=0&&mB.length<42){
    menuSpawnT=.35;
    const types=['orb','spark','gold'];
    mB.push({x:Math.random()*WW,y:-16,vx:(Math.random()-.5)*30,vy:55+Math.random()*80,t:Math.random()*9,type:types[(Math.random()*3)|0],dead:false});
  }
  for(const b of mB){ b.t+=dt; b.x+=(b.vx+Math.sin(b.t*1.4)*12)*dt; b.y+=b.vy*dt; if(b.y>WH+30)b.dead=true; }
  compact(mB);
  for(const p of parts){p.life-=dt;if(p.kind==='spark'){p.x+=p.vx*dt;p.y+=p.vy*dt;}else if(p.kind==='ring')p.r+=p.vr*dt;}
  for(let i=parts.length-1;i>=0;i--)if(parts[i].life<=0)parts.splice(i,1);
}

function update(dt){
  T+=dt;
  const boost=state==='play'?1+diff()*.5:1;
  for(const s of stars){ s.y+=s.sp*boost*dt; if(s.y>CH+2){s.y=-2;s.x=Math.random()*CW;} }
  shake=Math.max(0,shake-dt*34);
  if(state==='menu'){ updateMenu(dt); return; }
  if(state!=='play')return;
  if(freeze>0){ freeze-=dt; return; }
  updatePlay(dt);
}

  ctx.setTransform(DPR,0,0,DPR,0,0);
  F=fit();

  g.addColorStop(0,'#050713');g.addColorStop(.5,'#070b1d');g.addColorStop(1,'#04060f');

    ctx.fillRect(s.x,s.y,s.r,s.r);
