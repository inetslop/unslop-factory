      if(dir>0&&h.x>W+hw) h.x=-hw; else if(dir<0&&h.x<-hw) h.x=W+hw; } }
}

}

/* ---------------- update ---------------- */
function update(dt){
  tGlobal+=dt;
  const mult=(state==='playing'||state==='clear')?speedMult():1;
  moveHazards(dt*mult);
  for(let i=parts.length-1;i>=0;i--){ const p=parts[i]; p.life-=dt;
    if(p.life<=0){ parts.splice(i,1); continue; }
    p.x+=p.vx*dt; p.y+=p.vy*dt; p.vy+=(p.g||0)*dt; }
  for(let i=pops.length-1;i>=0;i--){ pops[i].t+=dt; if(pops[i].t>0.28)pops.splice(i,1); }
  frog.invuln=Math.max(0,frog.invuln-dt);
  frog.squash=Math.max(0,frog.squash-dt);
  frog.hitT=Math.max(0,frog.hitT-dt);
  flash.t=Math.max(0,flash.t-dt);
  shake=Math.max(0,shake-shake*8*dt-2*dt);

  if(state==='playing'){
    typeTime+=dt; timeLeft-=dt;
    if(timeLeft<5.05){ const s=Math.ceil(Math.max(0,timeLeft));
      if(s!==lowTickPrev&&s>0){ lowTickPrev=s; snd.tick(); } } else lowTickPrev=null;
    if(frog.hop){ frog.hop.t+=dt; if(frog.hop.t>=frog.hop.dur) land(); }
    if(!frog.hop && frog.row>=1 && frog.row<=4 && frog.invuln<=0){
      for(const h of hazards[frog.row])
        if(Math.abs(h.x-frog.x)<hwOf(h)+frogR*0.55){ doHit(); break; }
    }
    if(!frog.hop && pending.length) pressLetter(pending.shift());
    if(state==='playing' && timeLeft<=0) timeUp();
  } else if(state==='clear'){
    clearT+=dt;
    if(clearT>1.7){ level++; startRound(); state='playing'; }
  }
}

/* ---------------- drawing bits ---------------- */
function rr(x,y,w,h,r){ r=Math.min(r,w/2,h/2); ctx.beginPath();
  ctx.moveTo(x+r,y); ctx.arcTo(x+w,y,x+w,y+h,r); ctx.arcTo(x+w,y+h,x,y+h,r);
  ctx.arcTo(x,y+h,x,y,r); ctx.arcTo(x,y,x+w,y,r); ctx.closePath(); }
function heartPath(x,y,s){
  ctx.beginPath(); ctx.moveTo(x,y+s*0.3);
  ctx.bezierCurveTo(x,y, x-s*0.5,y-s*0.12, x-s*0.5,y+s*0.18);
  ctx.bezierCurveTo(x-s*0.5,y+s*0.52, x,y+s*0.72, x,y+s*0.92);
  ctx.bezierCurveTo(x,y+s*0.72, x+s*0.5,y+s*0.52, x+s*0.5,y+s*0.18);
  ctx.bezierCurveTo(x+s*0.5,y-s*0.12, x,y, x,y+s*0.3); ctx.closePath();
}
function drawWater(r){

  if(r===3){ g.addColorStop(0,'#175084'); g.addColorStop(1,'#123c63'); }
  else{ g.addColorStop(0,'#123c63'); g.addColorStop(1,'#0e2f52'); }
  ctx.fillStyle=g; ctx.fillRect(0,topY,W,laneH);
  ctx.strokeStyle='rgba(200,235,255,.14)'; ctx.lineWidth=2;
  const sp=Math.max(46,laneH*0.9), off=((tGlobal*dir*36)%sp+sp)%sp;
  for(let x=-sp+off;x<W+sp;x+=sp)
    for(let j=0;j<3;j++){
      const wy=topY+laneH*(0.24+0.27*j)+Math.sin(tGlobal*1.7+x*0.05+j)*2.5;
      ctx.beginPath(); ctx.arc(x+(j%2?sp*0.3:0),wy,laneH*0.14,Math.PI*1.08,Math.PI*1.92); ctx.stroke(); }
}
function drawCar(h,y){

  for(const a of [0.8,2.3,3.9,5.4]){ const lx=Math.cos(a)*r*0.82, ly=Math.sin(a)*r*0.62;
    ctx.beginPath(); ctx.arc(lx,ly,r*0.18,0,7); ctx.fill(); }
  ctx.beginPath(); ctx.arc(dirX*r*1.08,0,r*0.34,0,7); ctx.fill();
  const g=ctx.createRadialGradient(-r*0.2,-r*0.25,r*0.15,0,0,r);
  g.addColorStop(0,'#4d8f3f'); g.addColorStop(1,'#33622b');

}
function drawHazardLane(r){

function drawTiles(){
  if(state!=='playing'||frog.row>=5)return;
  const target=frog.row+1, typed=laneTyped[target], g=tileGeom(target), word=laneWords[target];
  ctx.fillStyle='rgba(6,14,9,.55)'; rr(g.x0-12,g.y-9,g.total+24,g.ts+18,10); ctx.fill();
  for(let i=typed;i<g.n;i++){
    const cx=g.x0+i*(g.ts+g.gap)+g.ts/2, cy=g.y+g.ts/2;
    const isNext=i===typed, pulse=isNext?1+0.05*Math.sin(tGlobal*7):1, s=g.ts*pulse;
    ctx.save();
    if(isNext){ ctx.shadowColor='rgba(157,255,87,.55)'; ctx.shadowBlur=14; }
    ctx.fillStyle=isNext?'#16351f':'rgba(14,36,22,.85)';
    rr(cx-s/2,cy-s/2,s,s,s*0.2); ctx.fill();
    ctx.lineWidth=2; ctx.strokeStyle=isNext?'#9dff57':'#2c5a36'; ctx.stroke();
    ctx.shadowBlur=0;
    ctx.fillStyle=isNext?'#eaffe6':'#79a884';
    ctx.font='700 '+Math.floor(g.ts*0.58)+'px '+MONO;
    ctx.textAlign='center'; ctx.textBaseline='middle';
    ctx.fillText(word[i],cx,cy+1);
    if(isNext && Math.floor(tGlobal*2.4)%2===0){
      ctx.fillStyle='#9dff57'; ctx.fillRect(cx-s*0.32,cy+s/2+4,s*0.64,3); }
    ctx.restore();
  }
  ctx.fillStyle='rgba(157,255,87,.2)'; ctx.fillRect(g.x0,g.y+g.ts+9,g.total,3);
  ctx.fillStyle='#9dff57'; ctx.fillRect(g.x0,g.y+g.ts+9,g.total*typed/g.n,3);
}
function drawFrog(){
  let rp=frog.row, z=0, s=1;
  if(frog.hop){ const p=frog.hop.t/frog.hop.dur; rp=frog.hop.from+(frog.hop.to-frog.hop.from)*p;
    z=Math.sin(Math.PI*p)*laneH*0.55; s=1+0.22*Math.sin(Math.PI*p); }
  const y=laneCY(rp)-z, x=frog.x;
  let sy=s, sx=s;
  if(frog.squash>0){ const q=frog.squash/0.18; sy*=1-0.35*q; sx*=1+0.28*q; }
  if(frog.hitT>0){ sy*=0.62; sx*=1.25; }
  /* shadow */
  ctx.fillStyle='rgba(0,0,0,.3)';
  ctx.beginPath(); ctx.ellipse(x,laneCY(rp)+laneH*0.28,frogR*0.9*(1-z/(laneH*3)),frogR*0.42*(1-z/(laneH*3)),0,0,7); ctx.fill();
  /* chevron hint */
  if(state==='playing'&&!frog.hop&&frog.row<5){
    const cyy=laneCY(frog.row)-frogR-16+Math.sin(tGlobal*5)*4;
    ctx.strokeStyle='rgba(157,255,87,'+(0.55+0.3*Math.sin(tGlobal*5))+')'; ctx.lineWidth=3; ctx.lineCap='round';
    for(let k=0;k<2;k++){ const yy=cyy-k*9;
      ctx.globalAlpha=k?0.4:1; ctx.beginPath(); ctx.moveTo(x-8,yy+6); ctx.lineTo(x,yy); ctx.lineTo(x+8,yy+6); ctx.stroke(); }
    ctx.globalAlpha=1;
  }
  ctx.save();
  if(frog.invuln>0 && Math.floor(tGlobal*12)%2===0) ctx.globalAlpha=0.45;

    ctx.save(); ctx.translate(lx*R,ly*R); ctx.rotate(a);

  g.addColorStop(0,'#8cf06b'); g.addColorStop(1,'#46a83c');
  ctx.fillStyle=g; ctx.beginPath(); ctx.ellipse(0,0,R*0.95,R*0.78,0,0,7); ctx.fill();
  ctx.strokeStyle='#2c6e2a'; ctx.lineWidth=2; ctx.stroke();
  ctx.fillStyle='rgba(47,125,44,.6)';

    ctx.beginPath(); ctx.arc(px*R,py*R,R*sr,0,7); ctx.fill(); }

    ctx.fillStyle='#fff'; ctx.beginPath(); ctx.arc(ex*R,-R*0.6,R*0.34,0,7); ctx.fill();
    ctx.strokeStyle='#2c6e2a'; ctx.lineWidth=1.5; ctx.stroke();

  if(frog.hitT>0){ ctx.fillStyle='rgba(255,70,50,.3)';
    ctx.beginPath(); ctx.ellipse(0,0,R*0.95,R*0.78,0,0,7); ctx.fill(); }
  ctx.restore();
}

function render(){
  ctx.clearRect(0,0,W,H);
  ctx.save();
  if(shake>0) ctx.translate(rnd(-shake,shake),rnd(-shake,shake));
  if(bg) ctx.drawImage(bg,0,0,W,H);
  drawWater(3); drawWater(4);
  /* marching center dashes */
  ctx.save(); ctx.strokeStyle='#e6b93c'; ctx.lineWidth=3;
  ctx.setLineDash([16,14]); ctx.lineDashOffset=-tGlobal*46;
  ctx.beginPath(); ctx.moveTo(0,hudH+3*laneH); ctx.lineTo(W,hudH+3*laneH); ctx.stroke(); ctx.restore();
  drawHazardLane(1); drawHazardLane(2); drawHazardLane(3); drawHazardLane(4);
  if(state==='clear'){ /* home pad glow */
    const a=0.35+0.3*Math.sin(tGlobal*6);
    ctx.strokeStyle='rgba(255,210,63,'+a+')'; ctx.lineWidth=3;
    ctx.beginPath(); ctx.arc(W/2,hudH+laneH/2,laneH*0.48,0,7); ctx.stroke(); }
  drawTiles(); drawFrog(); drawPops(); drawParts(); drawHUD();
  if(state==='clear') drawBanner();
  if(flash.t>0){ ctx.fillStyle='rgba('+flash.col+','+(0.32*flash.t/flash.max)+')'; ctx.fillRect(0,0,W,H); }
  ctx.restore();
  if(vg) ctx.drawImage(vg,0,0,W,H);
  if(state==='playing'&&timeLeft<5&&rg){
    ctx.globalAlpha=0.35+0.25*Math.sin(tGlobal*9);
    ctx.drawImage(rg,0,0,W,H); ctx.globalAlpha=1; }
}

});
