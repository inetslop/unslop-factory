function placeTower(c,r){
  const d=TOW[S.selType];
  towers.push({type:S.selType,c,r,x:c*CELL+CELL/2,y:r*CELL+CELL/2,level:0,cd:0,angle:-Math.PI/2,flash:0,spent:d.cost});
  S.gold-=d.cost;SFX.place();sparks(c*CELL+CELL/2,r*CELL+CELL/2,10,'#9fe8c4',110);ringFx(c*CELL+CELL/2,r*CELL+CELL/2,34,d.col,2,.35);
}
function upgrade(){
  const t=S.selTower;if(!t||t.level>=2)return;
  const cost=TOW[t.type].up[t.level].cost;
  if(S.gold<cost){SFX.deny();renderInfo();return;}
  S.gold-=cost;t.spent+=cost;t.level++;SFX.up();
  ringFx(t.x,t.y,44,TOW[t.type].col,3,.5);sparks(t.x,t.y,12,'#eafff5',130);renderInfo();
}
function sell(){
  const t=S.selTower;if(!t)return;
  const back=Math.round(t.spent*.7);S.gold+=back;towers.splice(towers.indexOf(t),1);
  popup(t.x,t.y,'+'+back,'#9fe8c4');SFX.sell();S.selTower=null;renderInfo();
}
function firstTarget(t,range){
  let best=null,bd=-1;const rr=range*range;
  for(const e of enemies){if(!e.alive)continue;
    const dx=e.x-t.x,dy=e.y-t.y;if(dx*dx+dy*dy>rr+(e.r*e.r))continue;if(e.dist>bd){bd=e.dist;best=e;}}
  return best;
}
function angLerp(a,b,t){const d=((b-a+Math.PI*3)%(Math.PI*2))-Math.PI;return a+d*Math.min(1,t);}

function fireTower(t,st,tgt){
  t.cd=1/st.rate;t.flash=.09;
  if(t.type==='pulse'){shots.push({kind:'bolt',x:t.x,y:t.y,tgt,ex:tgt.x,ey:tgt.y,spd:560,dmg:st.dmg,col:TOW.pulse.col});SFX.bolt();}
  else if(t.type==='coil'){
    const d=Math.hypot(tgt.x-t.x,tgt.y-t.y),time=d/300;
    const sf=tgt.slowT>0?1-tgt.slowF:1;
    shots.push({kind:'shell',x:t.x,y:t.y,tx:tgt.x+tgt.hx*tgt.spd*sf*time*.85,ty:tgt.y+tgt.hy*tgt.spd*sf*time*.85,spd:300,dmg:st.dmg,splash:st.splash,col:TOW.coil.col});
    SFX.shellF();
  }else if(t.type==='frost'){
    ringFx(t.x,t.y,st.range,TOW.frost.col,4,.6);SFX.frost();
    const rr=st.range*st.range;
    for(const e of enemies){if(!e.alive)continue;const dx=e.x-t.x,dy=e.y-t.y;
      if(dx*dx+dy*dy<=rr+e.r*e.r){e.slowT=Math.max(e.slowT,st.slowT);e.slowF=1-st.slow;hit(e,st.dmg);}}
  }else if(t.type==='arc'){
    const hits=[tgt],used=new Set([tgt]);let cur=tgt;
    for(let i=1;i<st.chains;i++){let nx=null,bd=130*130;
      for(const e of enemies){if(!e.alive||used.has(e))continue;const dx=e.x-cur.x,dy=e.y-cur.y,d2=dx*dx+dy*dy;if(d2<bd){bd=d2;nx=e;}}
      if(!nx)break;hits.push(nx);used.add(nx);cur=nx;}
    const pts=[{x:t.x,y:t.y}].concat(hits.map(e=>({x:e.x,y:e.y})));
    let dmg=st.dmg;
    for(const e of hits){hit(e,dmg);sparks(e.x,e.y,3,'#ff9de4',100);dmg*=.72;}
    fx.push({kind:'chain',pts,t:0,life:.15});SFX.arc();
  }
}

/* ============================== SIM ============================== */
function tickSim(dt){
  S.simT+=dt;
  if(S.mode==='build'){S.countdown-=dt;if(S.countdown<=0)startWave();return;}
  // combat: spawn
  S.waveT+=dt;
  while(S.queue.length&&S.queue[0].t<=S.waveT)spawnEnemy(S.queue.shift().type);
  // enemies
  for(const e of enemies){
    if(!e.alive)continue;
    e.flashT=Math.max(0,e.flashT-dt);if(e.slowT>0)e.slowT-=dt;
    e.rot+=dt*(e.type==='runner'||e.type==='shard'?3.2:1.1);
    let step=e.spd*(e.slowT>0?e.slowF:1)*dt;
    while(step>0&&e.wp<WAY.length){
      const t=WAY[e.wp],dx=t.x-e.x,dy=t.y-e.y,d=Math.hypot(dx,dy);
      if(d<=step){e.x=t.x;e.y=t.y;e.dist+=d;step-=d;e.wp++;}
      else{e.hx=dx/d;e.hy=dy/d;e.x+=e.hx*step;e.y+=e.hy*step;e.dist+=step;step=0;}
    }
    if(e.wp>=WAY.length)leak(e);
  }
  // towers
  for(const t of towers){
    t.cd-=dt;t.flash=Math.max(0,t.flash-dt);
    const st=tStats(t),tgt=firstTarget(t,st.range);
    if(tgt)t.angle=angLerp(t.angle,Math.atan2(tgt.y-t.y,tgt.x-t.x),dt*14);
    if(t.cd<=0){
      if((t.type==='frost')?enemies.some(e=>e.alive&&((e.x-t.x)**2+(e.y-t.y)**2)<=(st.range+e.r)**2):tgt)fireTower(t,st,tgt);
    }
  }
  // shots
  for(const s of shots){
    if(s.kind==='bolt'){
      if(s.tgt&&s.tgt.alive){s.ex=s.tgt.x;s.ey=s.tgt.y;}
      const dx=s.ex-s.x,dy=s.ey-s.y,d=Math.hypot(dx,dy),mv=s.spd*dt;
      if(d<=mv+6){
        if(s.tgt&&s.tgt.alive)hit(s.tgt,s.dmg);else sparks(s.ex,s.ey,3,'rgba(150,220,255,.7)',70);
        s.dead=true;
      }else{s.x+=dx/d*mv;s.y+=dy/d*mv;}
    }else if(s.kind==='shell'){
      const dx=s.tx-s.x,dy=s.ty-s.y,d=Math.hypot(dx,dy),mv=s.spd*dt;
      if(d<=mv+4){
        s.dead=true;SFX.boom();addShake(3);
        ringFx(s.tx,s.ty,s.splash,'#ffb454',4,.4);sparks(s.tx,s.ty,12,'#ffcf8a',190);
        const R=s.splash;
        for(const e of enemies){if(!e.alive)continue;const ex=e.x-s.tx,ey=e.y-s.ty,dist=Math.hypot(ex,ey);
          if(dist<=R+e.r)hit(e,s.dmg*(1-.45*dist/(R+e.r)));}
      }else{s.x+=dx/d*mv;s.y+=dy/d*mv;}
    }
  }
  // wave cleared?
  if(S.mode==='combat'&&S.queue.length===0&&!enemies.some(e=>e.alive))waveCleared();
}
function tickFx(dt){
  for(const f of fx){f.t+=dt;
    if(f.kind==='spark'){f.x+=f.vx*dt;f.y+=f.vy*dt;f.vx*=.92;f.vy*=.92;}}
  fx=fx.filter(f=>f.t<f.life);
  for(const p of pops){p.t+=dt;p.y-=30*dt;}
  pops=pops.filter(p=>p.t<p.life);
  S.coreFlash=Math.max(0,S.coreFlash-dt*1.6);
}
function waveCleared(){
  const bonus=20+S.wave*5;S.gold+=bonus;S.earned+=bonus;saveBest(S.wave);
  popup(W/2,120,'WAVE CLEAR +'+bonus,'#ffd9a0',17);SFX.clear();
  if(!S.endless&&S.wave>=15){winGame();return;}
  S.mode='build';S.countdown=18;S.buildTotal=18;updateHUD(true);
}

  S.mode='over';saveBest(S.wave);addShake(14);SFX.lose();
  sparks(CORE.x,CORE.y,50,'#ff4d5e',320);sparks(CORE.x,CORE.y,30,'#ffb454',260);ringFx(CORE.x,CORE.y,180,'#ff4d5e',6,.9);

  $('stWaveO').textContent=S.wave;$('stKillO').textContent=S.kills;$('stGoldO').textContent=S.earned;
  $('nbOver').style.display=(S.wave>=best&&S.wave>0)?'inline-block':'none';

}
