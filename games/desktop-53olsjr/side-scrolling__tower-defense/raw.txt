function buildWave(w){
  const q=[];const add=(n,t)=>{for(let i=0;i<n;i++)q.push(t);};
  add(4+w,'sham');
  if(w>=2)add(Math.floor(1+w*1.1),'skit');
  if(w>=3)add(Math.floor((w-1)*.8),'wisp');
  if(w>=4)add(Math.floor((w-2)/2)+(w%5===0?1:0),'brut');
  for(let i=q.length-1;i>0;i--){const j=(Math.random()*(i+1))|0;[q[i],q[j]]=[q[j],q[i]];}
  if(w>=5&&w%5===0)q.push('boss');
  return q;
}

function gain(g){gold+=g;goldEarned+=g;panelDirty=true;}
function ey(e){return e.fly?e.baseY:GROUND-e.r;}
function spawnEnemy(type){
  const d=EDEF[type],m=1+(wave-1)*.24+Math.pow(wave,1.45)*.035,hp=d.hp*m;
  enemies.push({type,def:d,x:WORLD_W+80+Math.random()*60,hp,max:hp,r:d.r,fly:!!d.fly,boss:!!d.boss,
    baseY:GROUND-(120+Math.random()*90),phase:Math.random()*6,slowT:0,slowF:0,hitT:0,appear:0,dead:false,
    reward:Math.round(d.reward*(1+wave*.03))});
}
function applyDmg(e,amt,type){
  if(e.dead)return;
  let d=amt;if((type==='ember'||type==='mortar')&&e.def.armor)d=Math.max(1,d-e.def.armor);
  e.hp-=d;e.hitT=.12;
  const top=e.fly?e.baseY-e.r:GROUND-e.r*2.1;
  dnums.push({x:e.x,y:top-6,t:0,txt:String(Math.round(d)),c:DAMGC[type]});
  if(e.hp<=0)killEnemy(e);
}
function killEnemy(e){
  e.dead=true;kills++;gain(e.reward);
  const top=e.fly?e.baseY-e.r*2.1-16:GROUND-e.r*2.3;
  dnums.push({x:e.x,y:top,t:0,txt:'+'+e.reward,c:'#ffd76a'});
  const col={sham:'#cfd6b4',skit:'#d8cfae',wisp:'#9fe8b8',brut:'#8a9aa5',boss:'#b6ff5e'}[e.type];
  burst(e.x,e.fly?e.baseY:GROUND-e.r,col,e.boss?34:12,e.boss?300:170,.6,180,true);
  if(e.boss){shake=Math.max(shake,16);rings.push({x:e.x,y:GROUND-40,r:230,t:0,dur:.7,c:'182,255,94'});banner('THE HOLLOW KING FALLS');sfx('boom');}
  else sfx('die');
}
function baseDamage(d){
  lives-=d;redFlash=1;shake=Math.max(shake,10);sfx('baseHit');
  if(lives<=0){lives=0;gameOver();}
}
function acquire(ax,ay,rng){let best=null,bx=1e9;
  for(const e of enemies){if(e.dead)continue;const dx=e.x-ax,dy=ey(e)-ay;if(dx*dx+dy*dy<=rng*rng&&e.x<bx){best=e;bx=e.x;}}
  return best;}
function nearEx(x,y,used,rng){let b=null,bd=1e9;
  for(const e of enemies){if(e.dead||used.has(e))continue;const d=(e.x-x)*(e.x-x)+(ey(e)-y)*(ey(e)-y);if(d<bd&&d<=rng*rng){b=e;bd=d;}}
  return b;}

  const tw=selTower;if(!tw||tw.lvl>=3)return;const c=upCost(tw);

  burst(padOf(tw).x,GROUND-50,'#8ef0dc',14,130,.5,60,true);

function sellSel(){
  const tw=selTower;if(!tw)return;

  burst(padOf(tw).x,GROUND-10,'rgb(150,160,170)',8,80,.4,200,false);

function padOf(tw){for(const p of pads)if(p.tower===tw)return p;return null;}
function updateTower(p,tw,dt){
  const def=TOWERS[tw.type],st=def.lv[tw.lvl-1];
  tw.cool-=dt;tw.flash=Math.max(0,tw.flash-dt*4.5);
  if(tw.type==='arc'&&Math.random()<dt*1.2)addPart(p.x+(Math.random()*8-4),GROUND-66,(Math.random()*40-20),-30-Math.random()*20,.3,1.6,'#ffe9a8',.9,0,true);
  if(tw.cool>0)return;
  const ax=p.x,ay=GROUND+def.anchor[1];
  if(tw.type==='frost'){
    let any=false;
    for(const e of enemies){if(e.dead)continue;const dx=e.x-ax,dy=ey(e)-ay;
      if(dx*dx+dy*dy<=st.range*st.range){any=true;applyDmg(e,st.dmg,'frost');e.slowT=Math.max(e.slowT,st.sdur);e.slowF=Math.max(e.slowF,st.slow*(1-(e.def.slowRes||0)));}}
    if(!any)return;
    tw.cool=1/st.rate;tw.flash=1;
    rings.push({x:p.x,y:GROUND-40,r:st.range,t:0,dur:.45,c:'120,220,255'});
    for(let i=0;i<7;i++)addPart(p.x+Math.random()*60-30,GROUND-50-Math.random()*20,(Math.random()*40-20),20+Math.random()*30,.7,1.8,'#cfefff',.8,30,false);
    sfx('frost');return;
  }
  const tgt=acquire(ax,ay,st.range);if(!tgt)return;
  tw.cool=1/st.rate;tw.flash=1;
  if(tw.type==='ember'){
    const dx=tgt.x-ax,dy=ey(tgt)-ay,d=Math.hypot(dx,dy)||1;
    projs.push({kind:'bolt',x:ax,y:GROUND-58,vx:dx/d*560,vy:dy/d*560,dmg:st.dmg,tgt,dead:false});
    burst(ax,GROUND-58,'#ffb060',3,90,.25,120,true);sfx('shoot');
  }else if(tw.type==='arc'){
    const segs=[];let cur=tgt,dmg=st.dmg,used=new Set([cur]);
    segs.push({x1:ax,y1:GROUND-64,x2:cur.x,y2:ey(cur)});applyDmg(cur,dmg,'arc');burst(cur.x,ey(cur),'#fff3c0',4,110,.3,80,true);
    dmg*=.65;
    for(let i=0;i<st.chains;i++){const nx=nearEx(cur.x,ey(cur),used,150);if(!nx)break;used.add(nx);
      segs.push({x1:cur.x,y1:ey(cur),x2:nx.x,y2:ey(nx)});applyDmg(nx,dmg,'arc');burst(nx.x,ey(nx),'#fff3c0',3,90,.25,80,true);dmg*=.65;cur=nx;}
    boltsFx.push({segs,t:.14});sfx('arc');
  }else{ // mortar
    const mx=ax,my=GROUND-26,dist=Math.abs(tgt.x-mx);
    tw.aimT=Math.atan2(GROUND-my,tgt.x-mx);
    const tDur=clamp(dist/240,.35,1.7),h=dist*.3+55;
    projs.push({kind:'shell',x0:mx,y0:my,tx:tgt.x,ty:GROUND,k:0,tDur,h,dmg:st.dmg,splash:st.splash,dead:false});
    const bx=mx+Math.cos(tw.aimT)*36,by=my+Math.sin(tw.aimT)*36;
    burst(bx,by,'#c9b8a0',5,60,.4,120,false);sfx('mortarFire');
  }
}
function explodeShell(p){
  rings.push({x:p.tx,y:GROUND,r:p.splash,t:0,dur:.35,c:'255,170,80'});
  for(let i=0;i<14;i++){const a=Math.random()*TAU,s=60+Math.random()*180;
    addPart(p.tx,GROUND-6,Math.cos(a)*s,Math.sin(a)*s-60,.5+Math.random()*.3,2.4,i%3?'#ffb060':'#ffe9c0',1,220,true);}
  for(let i=0;i<6;i++)addPart(p.tx+(Math.random()*30-15),GROUND-10,(Math.random()*20-10),-30-Math.random()*30,.9,6+Math.random()*5,'rgb(125,128,138)',.25,-20,false);
  let hitAny=false;
  for(const e of enemies){if(e.dead)continue;const dx=e.x-p.tx,dy=(e.fly?e.baseY:GROUND)-p.ty,d=Math.hypot(dx,dy);
    if(d<=p.splash+e.r*.5){applyDmg(e,p.dmg*(1-d/p.splash*.45)*(e.fly?.5:1),'mortar');hitAny=true;}}
  sfx('boom');shake=Math.max(shake,hitAny?5:3);
}

/* ============================== UPDATE ============================== */
function update(dt){
  time+=dt;
  for(const e of ambE){e.y-=e.v*dt;e.x-=6*dt;if(e.y<GROUND-280||e.x<-40){e.x=WORLD_W+Math.random()*120;e.y=GROUND-Math.random()*30;}}
  for(const f of fog){f.x-=f.v*dt;if(f.x<-320)f.x=WORLD_W+300;}
  shake=Math.max(0,shake-shake*6*dt-2*dt);
  redFlash=Math.max(0,redFlash-dt*1.3);

  if(state==='build'){interT-=dt;if(interT<=0)startWave();}
  if(state==='wave'&&spawnQ.length){spawnT-=dt;while(spawnT<=0&&spawnQ.length){spawnEnemy(spawnQ.shift());spawnT+=spawnGap*(.7+Math.random()*.6);}}

  for(const e of enemies){
    e.appear=Math.min(1,e.appear+dt*3);
    if(e.slowT>0){e.slowT-=dt;if(e.slowT<=0)e.slowF=0;}
    e.hitT=Math.max(0,e.hitT-dt);
    e.phase+=dt*(e.def.sp/26);
    e.x-=e.def.sp*(1-e.slowF)*dt;
    if(e.x<=HIT_X&&!e.dead){e.dead=true;baseDamage(e.def.dmg);}
  }
  for(const p of pads)if(p.tower)updateTower(p,p.tower,dt);

  for(const p of projs){
    if(p.kind==='bolt'){
      if(p.tgt&&!p.tgt.dead){const dx=p.tgt.x-p.x,dy=ey(p.tgt)-p.y,d=Math.hypot(dx,dy)||1;p.vx=dx/d*560;p.vy=dy/d*560;}
      p.x+=p.vx*dt;p.y+=p.vy*dt;
      if(p.tgt&&!p.tgt.dead&&Math.hypot(p.tgt.x-p.x,ey(p.tgt)-p.y)<p.tgt.r+7){applyDmg(p.tgt,p.dmg,'ember');burst(p.x,p.y,'#ffb060',5,110,.3,140,true);p.dead=true;}
      else if(Math.hypot(p.vx,p.vy)*dt*60>9e9||false){} // noop guard
      p.life=(p.life===undefined?1.2:p.life)-dt;if(p.life<=0)p.dead=true;
    }else{
      p.k+=dt/p.tDur;if(p.k>=1){explodeShell(p);p.dead=true;}
    }
  }
  projs=projs.filter(p=>!p.dead);

  for(const b of boltsFx)b.t-=dt;boltsFx=boltsFx.filter(b=>b.t>0);
  for(const r of rings)r.t+=dt;rings=rings.filter(r=>r.t<r.dur);
  for(const p of parts){p.life-=dt;p.x+=p.vx*dt;p.y+=p.vy*dt;p.vy+=(p.grav||0)*dt;}
  parts=parts.filter(p=>p.life>0);
  for(const d of dnums){d.t+=dt;d.y-=34*dt;}
  dnums=dnums.filter(d=>d.t<.9);

  enemies=enemies.filter(e=>!e.dead);
  if(state==='wave'&&!spawnQ.length&&!enemies.length)waveClear();
  updateCam(dt);
}
function updateCam(dt){
  const maxC=Math.max(0,WORLD_W-viewW);
  if(keys.l&&state!=='menu'&&state!=='over'){cam.x-=640*dt;manualT=1.3;}
  if(keys.r&&state!=='menu'&&state!=='over'){cam.x+=640*dt;manualT=1.3;}
  let focus=BASE_X+250;
  if(enemies.length){let m=1e9;for(const e of enemies)if(e.x<m)m=e.x;focus=m;}
  camTarget=clamp(focus-viewW*.4,0,maxC);
  if(manualT>0)manualT-=dt;else cam.x+=(camTarget-cam.x)*Math.min(1,dt*3.2);
  cam.x=clamp(cam.x,0,maxC);
}
