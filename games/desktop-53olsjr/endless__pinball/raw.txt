function collideSeg(s){
  if(s.one!==undefined&&ball.y>s.one)return;                 // one-way deck
  closest(ball.x,ball.y,s.x1,s.y1,s.x2,s.y2);
  let dx=ball.x-_qx,dy=ball.y-_qy,d=Math.hypot(dx,dy);
  if(d>=R)return;

  if(d>1e-6){nx=dx/d;ny=dy/d;}
  else{const ex=s.x2-s.x1,ey=s.y2-s.y1,L=Math.hypot(ex,ey)||1;nx=-ey/L;ny=ex/L;}

  if(vn<0){const e=s.e!==undefined?s.e:.42;ball.vx-=(1+e)*vn*nx;ball.vy-=(1+e)*vn*ny;}
}
function hitBumper(b,rvx,rvy,isD){

function hitFlip(f){

  closest(ball.x,ball.y,f.px,f.py,tx,ty);

}
function stepPhysics(h){

  for(let i=0;i<walls.length;i++)collideSeg(walls[i]);
  hitSling(slings[0]);hitSling(slings[1]);

}

  S.waveT-=dt;if(S.waveT<=0)waveUp();
  updFlip(FL[0],keyL,L_UP,dt);updFlip(FL[1],keyR,R_UP,dt);

  const canP=ball.alive&&ball.x>534&&ball.y>846&&(Math.abs(ball.vx)+Math.abs(ball.vy))<70;
  S.canP=canP;
  if(S.plHeld&&canP)S.charge=Math.min(1,S.charge+dt*1.6);
  else{if(!S.plHeld&&S.charge>.12&&canP)launch();S.charge=0;}
  if(S.charge>0&&canP){ball.x=554;ball.y=861+S.charge*42;ball.vx=0;ball.vy=0;}

  if(!ball.alive){
    if(S.respawnT>0){S.respawnT-=dt;if(S.respawnT<=0)spawnBall();}
    else if(S.overT>0){S.overT-=dt;if(S.overT<=0)doGameOver();}
  }
  if(ball.alive&&S.charge===0){
    const n=Math.min(10,Math.max(1,Math.ceil(dt/.00385))),h=dt/n;
    for(let i=0;i<n&&ball.alive;i++)stepPhysics(h);
  }
  ball.spin+=ball.vx*dt/R;
  if(ball.alive){ball.trail.push({x:ball.x,y:ball.y});if(ball.trail.length>16)ball.trail.shift();}

  moveDrifters(dt);moveSpinners(dt);decayFx(dt);
  S.popups=S.popups.filter(p=>t-p.t0<p.life);
}
