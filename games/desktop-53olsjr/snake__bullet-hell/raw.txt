  if(snake.prev&&state!=='over'){ const t=Math.min(1,tickT/tickInt),a=snake.prev[Math.min(i,snake.prev.length-1)];

    if(dc>COLS/2)dc-=COLS;if(dc<-COLS/2)dc+=COLS;if(dr>ROWS/2)dr-=ROWS;if(dr<-ROWS/2)dr+=ROWS;
    px=((a.c+dc*t)%COLS+COLS)%COLS*CELL+CELL/2;py=((a.r+dr*t)%ROWS+ROWS)%ROWS*CELL+CELL/2;}
  else{px=cur.c*CELL+CELL/2;py=cur.r*CELL+CELL/2;}
  return{x:px,y:py};}

function snakeTick(){ if(dead)return;
  if(snake.queue.length){ const d=snake.queue.shift();
    if(!(d.x===-snake.dir.x&&d.y===-snake.dir.y)&&!(d.x===snake.dir.x&&d.y===snake.dir.y))snake.dir=d;}
  const h=snake.cells[0],nc=(h.c+snake.dir.x+COLS)%COLS,nr=(h.r+snake.dir.y+ROWS)%ROWS;
  const lim=snake.grow>0?snake.cells.length:snake.cells.length-1;
  for(let i=0;i<lim;i++){ if(snake.cells[i].c===nc&&snake.cells[i].r===nr)return kill('self');}
  snake.prev=snake.cells.map(o=>({c:o.c,r:o.r}));
  snake.cells.unshift({c:nc,r:nr});
  if(snake.grow>0)snake.grow--;else snake.cells.pop();

function sim(gdt){ const P=state==='playing';
  if(P)timeAlive+=gdt;
  if(slowT>0){slowT-=gdt/ts;if(slowT<=0)tsTarget=1;}
  ts+=(tsTarget-ts)*Math.min(1,gdt*7);
  invuln=Math.max(0,invuln-gdt);

  if(P&&!dead){ waveT+=gdt;
    if(waveT>18){wave++;waveT=0;banner('WAVE '+wave,'#3ae0ff');sWave();}
    spawnT-=gdt;
    if(spawnT<=0){pickPattern();spawnT=Math.max(.5,1.9-wave*.12)*(.7+Math.random()*.6);}}

  for(const e of emitters){ e.t-=gdt;
    if(e.kind==='spiral'&&e.shots>0&&!dead){ e.it=(e.it||0)-gdt;
      if(e.it<=0){e.it=e.intv;B(e.x,e.y,Math.cos(e.a)*e.sp,Math.sin(e.a)*e.sp,5.5,C_CREAM);e.a+=e.rot;e.shots--;}}
    if(e.t<=0){if(e.kind==='burst'&&!dead)e.fire();e.done=true;}}
  emitters=emitters.filter(e=>!e.done);
  for(const w of warns){ w.t-=gdt; if(w.t<=0){fireSweep(w);w.done=true;} }
  warns=warns.filter(w=>!w.done);

  const hp=headPx();
  for(const b of bullets){ b.x+=b.vx*gdt;b.y+=b.vy*gdt;
    if(b.x<-40||b.x>FW+40||b.y<-40||b.y>FH+40)b.dead=true;
    if(!P||dead)continue;
    const dd=Math.hypot(b.x-hp.x,b.y-hp.y);
    if(dd<b.r+13){ if(invuln<=0){ if(shield>0){shield--;b.dead=true;invuln=1.2;shake=10;sShieldBrk();burst(hp.x,hp.y,C_CYAN,14,180,3);pop(hp.x,hp.y-26,'SHIELD!','#3ae0ff',15);}
      else{b.dead=true;burst(hp.x,hp.y,C_PINK,10,170,3);kill('bullet');}} }
    else if(!b.near&&dd<b.r+13+16){ b.near=true;streak++;streakT=3;
      mult=Math.min(8,1+Math.floor(streak/4));stats.maxMult=Math.max(stats.maxMult,mult);
      stats.near++;meter=Math.min(100,meter+7);score+=2*mult;sNear();
      if(Math.random()<.5)pop(hp.x,hp.y-30,'+'+(2*mult),'#ffd166',11);} }
  bullets=bullets.filter(b=>!b.dead);

  if(!dead){ tickT+=gdt; while(tickT>=tickInt){tickT-=tickInt;if(state==='menu')aiSteer();snakeTick();} }

  // pickups
  goldT-=gdt; shieldT-=gdt;
  if(P&&goldT<=0&&!pickups.some(p=>p.type==='gold')){spawnPick('gold');goldT=14+Math.random()*8;}
  if(P&&shieldT<=0){if(!pickups.some(p=>p.type==='shield'))spawnPick('shield');shieldT=22+Math.random()*10;}
  for(const p of pickups){ p.t+=gdt;
    if(p.life!=null){p.life-=gdt;if(p.life<=0)p.dead=true;}
    if(!P||dead)continue;
    if(Math.hypot(p.x-hp.x,p.y-hp.y)<24){ p.dead=true;
      if(p.type==='food'){ const maxed=snake.cells.length>=40;
        score+=maxed?25*mult:10*mult; stats.eaten++; meter=Math.min(100,meter+8);
        if(!maxed){snake.grow+=1;tickInt=Math.max(.06,.118-(snake.cells.length-3)*.0022);}
        pop(p.x,p.y,'+'+(maxed?25:10)*mult,'#ffb52e',14);burst(p.x,p.y,'#ffb52e',10,120,2.5);sEat(mult);}
      else if(p.type==='gold'){ const maxed=snake.cells.length>=40;score+=60*mult;stats.eaten++;meter=Math.min(100,meter+30);
        if(!maxed){snake.grow+=2;tickInt=Math.max(.06,.118-(snake.cells.length-3)*.0022);}

}
