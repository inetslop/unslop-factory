  let text=null;
  if(S.level>=4 && Math.random()<0.18) text=PHRASES[irand(0,PHRASES.length-1)];
  else for(let t=0;t<6;t++){ const L=irand(r[0],r[1]), a=POOL[L], w=a[irand(0,a.length-1)]; if(w!==lastText){text=w;break;} }
  if(!text) text='train';
  lastText=text; return text;
}

function clearWord(w){
  w.done=true; w.el.classList.add('clearing');
  const m=S.mult;
  let pts=(w.text.length*12+S.level*8)*m;
  const near=(w.x+w.w < decoderX+130); if(near) pts+=40;
  S.score+=pts; updateScore();
  S.streak++; S.mult=Math.min(5,1+Math.floor(S.streak/3)); updateCombo();
  S.totalCleared++; S.clearedInLevel++; updateLevel();
  const cx=w.x+w.w*0.5, cy=laneY[w.lane]-14;
  popAt(cx,cy,(near?'CLOSE! ':'')+'+'+pts+(m>1?' ×'+m:''), near?'gold':'');
  burst(w.x+w.w*0.5, laneY[w.lane]+6, ['#ffd98a','#7ac74f','#fff3d6'], 240);
  sfxClear(m);
  const i=words.indexOf(w); if(i>=0) words.splice(i,1);
  setTimeout(()=>{ w.el.remove(); markActive(); },520);
  if(S.clearedInLevel>=NEED){ S.level++; S.clearedInLevel=0;
    S.targetSpeed=Math.min(MAXSPD,BASE+(S.level-1)*STEP); updateLevel();
    showBanner('LEVEL '+S.level+' — FASTER!'); sfxLevel(); }
  markActive();
}
function missWord(w){
  w.done=true; w.el.classList.add('missed');
  S.lives--; updateLives(); shake=14; flashDanger();
  popAt(decoderX,laneY[w.lane]+8,'MISSED','stamp');

  if(S.lives<=0) gameOver(); else markActive();
}

function handleKey(k){
  const ch=k===' '?' ':k.toLowerCase();
  const w=words.find(w=>!w.done);
  if(!w){ clack(1,0.35); return; }
  if(ch===w.text[w.progress]){
    S.ok++;
    const spn=w.spans[w.progress]; spn.classList.remove('next'); spn.classList.add('hit');
    w.progress++;
    if(w.progress<w.text.length) w.spans[w.progress].classList.add('next');
    clack(w.progress); inkDot(spn);
    if(w.progress>=w.text.length) clearWord(w);
  }else{
    S.bad++; w.dirty=true; S.streak=0; S.mult=1; updateCombo();
    w.inner.classList.remove('shake'); void w.inner.offsetWidth; w.inner.classList.add('shake');
    const nx=w.spans[w.progress]; nx.classList.remove('want'); void nx.offsetWidth; nx.classList.add('want');
    sfxBad();
  }
}

  if(e.repeat) return;
  if(k===' '||/^[a-zA-Z]$/.test(k)){ e.preventDefault(); handleKey(k); }
});

  ctx.restore();

function frame(now){
  requestAnimationFrame(frame);
  tNow=now/1000;
  let dt=(now-tPrev)/1000; tPrev=now; dt=clamp(dt,0,0.05);
  const frozen=S.mode==='paused';
  if(!frozen){
    gameTime+=dt;
    S.speed+=(S.targetSpeed-S.speed)*Math.min(1,dt*2.4);
    worldX+=S.speed*dt;
  }
  shake=Math.max(0,shake-dt*36);
  if(S.mode==='playing'&&gameTime>=nextSpawnAt&&words.filter(w=>!w.done).length<7) spawnWord();
  for(const w of words){
    const drift=w.done?S.speed*0.45:S.speed;
    if(!frozen) w.x-=drift*dt;
    w.el.style.transform='translate('+w.x.toFixed(1)+'px,'+laneY[w.lane]+'px)';
    if(!w.done&&S.mode==='playing'&&w.x+w.w<decoderX-8) missWord(w);
  }
  let removed=false;
  for(let i=words.length-1;i>=0;i--){ const w=words[i];
    if(w.x<-w.w-80){ w.el.remove(); words.splice(i,1); removed=true; } }
  if(removed) markActive();
  if(!frozen){
    smokeT-=dt;
    if(smokeT<=0){ smokeT=rand(0.13,0.26);
      const F=W*0.17,tw=trainW(),th=trainH();
      smokes.push({x:F-tw*0.16+rand(-2,2),y:groundY-th+th*0.24-42*s,vx:-S.speed*0.22+rand(-8,8),vy:-rand(26,52),r:(4+rand(0,3))*s,life:1.5,max:1.5});
      if(smokes.length>90) smokes.shift(); }
    for(const p of smokes){ p.life-=dt; p.x+=p.vx*dt; p.y+=p.vy*dt; p.r+=9*s*dt; }
    smokes=smokes.filter(p=>p.life>0);
    for(const p of parts){ p.life-=dt; p.vy+=p.g*dt; p.x+=p.vx*dt; p.y+=p.vy*dt; }
    parts=parts.filter(p=>p.life>0);
    if(parts.length>240) parts.splice(0,parts.length-240);
  }
  shakeX=0; shakeY=0;
  if(shake>0&&!frozen){ shakeX=(Math.random()*2-1)*shake; shakeY=(Math.random()*2-1)*shake*0.7; }
  tapeLayer.style.transform='translate('+shakeX.toFixed(1)+'px,'+shakeY.toFixed(1)+'px)';
  render();
}
