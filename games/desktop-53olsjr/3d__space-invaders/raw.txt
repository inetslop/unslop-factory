    // bullets
    for(let i=bullets.length-1;i>=0;i--){ const b=bullets[i]; b.pz=b.z; b.z+=95*dt;
      if(b.z>72) bullets.splice(i,1); }
    // bombs
    enemyFire(dt);
    for(let i=bombs.length-1;i>=0;i--){ const m=bombs[i]; m.pz=m.z; m.x+=m.vx*dt; m.z+=m.vz*dt;
      if(m.pz>=PLAYER_Z&&m.z<PLAYER_Z&&Math.abs(m.x-player.x)<1.6){
        bombs.splice(i,1);
        if(player.shield>0){ explode(m.x,1.2,PLAYER_Z,'#5cffb0',10,3); tone('sine',700,200,.15,.2); continue; }
        if(player.inv>0){ explode(m.x,1.2,PLAYER_Z,'#6ee7ff',8,2.5); continue; }
        diePlayer(); break;
      }
      if(m.z<2) bombs.splice(i,1);
    }
    // drops
    for(let i=drops.length-1;i>=0;i--){ const d=drops[i]; const pz=d.z; d.z-=15*dt;
      if(pz>=PLAYER_Z&&d.z<PLAYER_Z&&Math.abs(d.x-player.x)<2.4){ applyDrop(d); drops.splice(i,1); continue; }
      if(d.z<2) drops.splice(i,1); }
    // ufo
    ufoT-=dt;
    if(!ufo&&ufoT<=0){ ufo={x:Math.random()<.5?-27:27,vx:(Math.random()<.5?1:-1)*8,y:13.5,z:55,ht:0}; ufoT=13+Math.random()*9; }
    if(ufo){ ufo.x+=ufo.vx*dt; ufo.ht-=dt; if(ufo.ht<=0){ufoT>0&&sUfo(); ufo.ht=.35;} if(Math.abs(ufo.x)>28) ufo=null; }

    // bullet collisions
    for(let i=bullets.length-1;i>=0;i--){ const b=bullets[i]; let hit=false;
      for(const e of enemies){ if(e.dead) continue;
        if(b.pz<e.z&&b.z>=e.z&&Math.abs(b.x-e.x)<e.hw){ killEnemy(e); hit=true; break; } }
      if(!hit&&ufo&&b.pz<ufo.z&&b.z>=ufo.z&&Math.abs(b.x-ufo.x)<2.3){
        const pts=[50,100,150,300][(Math.random()*4)|0]; score+=pts;
        pop(ufo.x,ufo.y,ufo.z,'MYSTERY +'+pts,'#c9a1ff'); explode(ufo.x,ufo.y,ufo.z,'#b26bff',22,5);
        sBoom(); shake=6; ufo=null; hit=true; }
      if(hit) bullets.splice(i,1);
    }
    // breach
    if(Z0+formZ<=KILL_Z){ banner('GRID BREACHED','THEY REACHED YOU'); diePlayer(); }
    $('warn').style.display=(Z0+formZ<22)?'block':'none';
  }
  else if(state==='dying'){ dyingT-=dt; if(dyingT<=0){ if(lives<=0) gameOver(); else { spawnWave(wave); resetPlayer(); state='play'; banner('WAVE '+String(wave).padStart(2,'0'),'REDEPLOYED'); } } }
  else if(state==='clear'){ clearT-=dt; if(clearT<=0){ spawnWave(wave+1); resetPlayer(); player.inv=1.2; state='play'; banner('WAVE '+String(wave+1).padStart(2,'0'),'INCOMING'); } }

  // particles & popups
  for(let i=parts.length-1;i>=0;i--){ const p=parts[i]; p.l+=dt; if(p.l>p.max){parts.splice(i,1);continue;}
    p.x+=p.vx*dt; p.y+=p.vy*dt; p.z+=p.vz*dt; p.vx*=.96; p.vy*=.96; p.vz*=.96; }
  for(let i=pops.length-1;i>=0;i--){ const p=pops[i]; p.l+=dt; p.y+=dt*2; if(p.l>p.max) pops.splice(i,1); }

  // camera
  const target=state==='menu'?Math.sin(t*.3)*4:player.x*.32;
  camX+=(target-camX)*Math.min(1,dt*5);
  syncHUD();
}

function renderWorld(c,cx){
  g=c; camXDraw=cx;

  const sx=(Math.random()-.5)*shake, sy=(Math.random()-.5)*shake;
  saveCam(); camXc=camX+sx; camYc=CAMY-sy;
  c.save(); c.setTransform(1,0,0,1,0,0); c.drawImage(bgCv,0,0); c.restore();
  drawStars(c); drawGrid(c);
  // depth-sorted world objects
  const list=[];
  if(ufo) list.push({z:ufo.z,f:()=>drawUfo(c)});
  for(const e of enemies) if(!e.dead) list.push({z:e.z,f:()=>drawSprite(c,e)});
  for(const d of drops) list.push({z:d.z,f:()=>drawDrops(c)});
  for(const m of bombs) list.push({z:m.z,f:()=>drawBombs(c)});
  for(const b of bullets) list.push({z:b.z,f:()=>drawBullets(c)});
  list.sort((a,b)=>b.z-a.z);
  drawDrops(c); // (small, drawn in pass below instead)
  // simpler robust ordering: far groups first
  c.clearRect(0,0,0,0);
  if(ufo) drawUfo(c);
  const sorted=enemies.filter(e=>!e.dead).sort((a,b)=>b.z-a.z);
  for(const e of sorted) drawSprite(c,e);
  drawDrops(c); drawBombs(c); drawBullets(c); drawParts(c);
  drawShip(c); drawPops(c);
}
