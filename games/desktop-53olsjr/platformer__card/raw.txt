function spawnRow(){
  G.rowIdx++;
  const y=G.genY, alt=-y/10;
  let x=clamp(G.prevX+rand(-190,190), -BAND+70, BAND-70);
  let type;
  if(G.rowIdx<=5){ type=Math.random()<.5?'D':'H'; }
  else{ const r=Math.random(); type=r<.42?'D': r<.58?'H': r<.76?'C':'S'; }
  const w = type==='D'?rand(100,150) : type==='H'?rand(92,128) : rand(92,126);
  const pl={x,y,w,h:60,type, baseX:x, amp:type==='C'?rand(45,85):0, speed:rand(.7,1.3),
    phase:rand(0,6.28), dx:0, breakT:-1, broken:false};
  G.plats.push(pl); G.prevX=x;

  if(G.rowIdx-G.lastJok>=6 && Math.random()<.08){
    G.golds.push({x:clamp(x+rand(-170,170),-BAND+40,BAND-40), y:y-rand(55,120), ph:rand(0,6.28)});
    G.lastJok=G.rowIdx;
  }
  if(alt>15 && Math.random()<Math.min(.5,.12+alt*.003)){
    if(Math.random()<.6){
      G.enemies.push({kind:'bat', x:rand(-BAND+60,BAND-60), y:y-rand(24,110), dir:Math.random()<.5?-1:1,
        sp:Math.min(300, rand(110,160)+alt*1.1), ph:rand(0,6.28), frozenT:0});
    } else if(type==='D'||type==='C'){
      G.enemies.push({kind:'chip', plat:pl, x:x, y:y-17, dir:1, sp:Math.min(190, 60+alt*.7), frozenT:0});
    }
  }
  let gap=rand(88,132); if(G.rowIdx<6) gap=rand(78,95);
  G.genY-=gap;
}

  return {type:'spark',x,y,vx:Math.cos(a)*sp,vy:Math.sin(a)*sp-60,g:.8,life:rand(.3,.7),max:.7,size:rand(2,4.5),color}; }
function updParts(dt){
  for(let i=G.parts.length-1;i>=0;i--){ const q=G.parts[i]; q.life-=dt;
    if(q.life<=0){ G.parts.splice(i,1); continue; }
    if(q.type==='ring'){ q.size+=q.gr*dt; continue; }
    q.vy+=(q.g||0)*900*dt; q.x+=q.vx*dt; q.y+=q.vy*dt;
    if(q.vr) q.rot=(q.rot||0)+q.vr*dt;
  }
  if(G.parts.length>420) G.parts.splice(0,G.parts.length-420);
}

function playCard(i){
  if(G.state!=='play'||G.paused) return;
  const s=G.hand[i]; if(!s) return;
  ensureAudio();
  G.hand[i]=null; G.refill[i]=4;
  const card=slotEls[i].querySelector('.card'); card.classList.add('played');
  const rid=G.runId; setTimeout(()=>{ if(G.runId===rid) renderSlot(i,false); },340);
  const p=G.player;
  switch(s){
    case 'H':
      p.vy=-ROCKET; G.shake=Math.max(G.shake,5); sfx.rocket();
      for(let k=0;k<16;k++) G.parts.push({type:'spark',x:p.x,y:p.y+18,vx:rand(-90,90),vy:rand(60,240),g:.3,
        life:rand(.3,.6),max:.6,size:rand(2,5),color:k%2?'#ff9d4d':'#ffd27a'});
      break;
    case 'D':
      G.shieldT=2.5; sfx.shield();
      G.parts.push({type:'ring',x:p.x,y:p.y,size:10,gr:300,life:.35,max:.35,col:'233,184,76',lw:4});
      break;
    case 'C': {
      sfx.freeze(); let n=0;
      for(const e of G.enemies) if(Math.hypot(e.x-p.x,e.y-p.y)<270){ e.frozenT=3.5; frostBurst(e); n++; }
      G.parts.push({type:'ring',x:p.x,y:p.y,size:14,gr:n?620:240,life:.5,max:.5,col:'150,220,245',lw:3});
      break; }
    case 'S': {
      sfx.blast(); G.shake=14; p.vy=Math.min(p.vy,-340);
      G.parts.push({type:'ring',x:p.x,y:p.y,size:16,gr:520,life:.38,max:.38,col:'233,184,76',lw:6});
      for(let k=0;k<18;k++) G.parts.push(spark(p.x,p.y,k%2?'#2a3040':'#e9b84c',rand(120,380)));
      for(let i=G.enemies.length-1;i>=0;i--) if(Math.hypot(G.enemies[i].x-p.x,G.enemies[i].y-p.y)<175) killEnemy(i,false);
      for(let i=G.plats.length-1;i>=0;i--){ const pl=G.plats[i];
        if(pl.type==='S' && Math.hypot(pl.x-p.x,pl.y-p.y)<180) breakPlatIdx(i); }
      break; }
    case 'J':
      G.slowT=2.8; sfx.slow();
      for(let k=0;k<16;k++) G.parts.push({type:'star',x:p.x,y:p.y,vx:rand(-140,140),vy:rand(-160,100),g:.25,
        life:rand(.5,.9),max:.9,size:rand(8,15),color:'#ffd76e'});
      break;
  }
}

function damage(){
  if(G.invulnT>0) return;
  const p=G.player;
  G.lives--; updateLives();
  G.flash=1; G.hitT=.4; G.shake=15; sfx.hurt();
  for(let k=0;k<26;k++) G.parts.push(spark(p.x,p.y,k%2?'#e9b84c':'#f6efdd',rand(80,340)));
  G.parts.push({type:'ring',x:p.x,y:p.y,size:10,gr:500,life:.4,max:.4,col:'200,53,74',lw:5});
  if(G.lives<=0){ gameOver(); return; }
  const sp=G.safePlat;
  p.x=sp.x; p.y=sp.y-22; p.vx=0; p.vy=0; p.stand=null; p.coyote=0;
  G.cam.y=p.y-VH*.45; G.invulnT=2.2;
}

/* ============================== UPDATE ============================== */
function updPlatforms(dtW,live){
  for(let i=G.plats.length-1;i>=0;i--){ const pl=G.plats[i];
    if(pl.type==='C'){ const nx=pl.baseX+Math.sin(G.wt*pl.speed+pl.phase)*pl.amp; pl.dx=nx-pl.x; pl.x=nx; }
    else pl.dx=0;
    if(live && pl.type==='S' && !pl.broken && pl.breakT>=0){
      pl.breakT+=dtW; if(pl.breakT>0.55) breakPlatIdx(i);
    }
  }
}
function updEnemies(dtW,live){
  for(let i=G.enemies.length-1;i>=0;i--){ const e=G.enemies[i];
    if(e.frozenT>0) e.frozenT-=dtW;
    else{
      if(e.kind==='bat'){ e.x+=e.dir*e.sp*dtW; e.ph+=dtW*9;
        if(e.x>BAND-60){e.x=BAND-60;e.dir=-1;} if(e.x<-BAND+60){e.x=-BAND+60;e.dir=1;} }
      else{ const pl=e.plat; e.x+=e.dir*e.sp*dtW;
        const lo=pl.x-pl.w/2+16, hi=pl.x+pl.w/2-16;
        if(e.x<lo){e.x=lo;e.dir=1;} if(e.x>hi){e.x=hi;e.dir=-1;} }
    }
    if(e.kind==='chip') e.y=e.plat.y-17;
    if(e.y>G.cam.y+VH+(live?500:600)){ G.enemies.splice(i,1); continue; }
    if(live){
      const p=G.player, rad=(e.kind==='bat'?14:15)+16;
      if(G.invulnT<=0 && Math.hypot(e.x-p.x,e.y-p.y)<rad){
        if(G.shieldT>0) killEnemy(i,true); else damage();
      }
    }
  }
}

function update(dt){
  const target=(G.slowT>0||G.hitT>0)?.32:1;
  G.ts+=(target-G.ts)*Math.min(1,dt*8);
  const dtW=dt*G.ts;
  G.wt+=dtW;
  if(G.slowT>0)G.slowT-=dt; if(G.hitT>0)G.hitT-=dt;
  if(G.shieldT>0)G.shieldT-=dt; if(G.invulnT>0)G.invulnT-=dt;
  G.flash=Math.max(0,G.flash-dt*2.2);
  G.shake=Math.max(0,G.shake-dt*40);

  for(let i=0;i<5;i++) if(!G.hand[i]){ G.refill[i]-=dt;
    if(G.refill[i]<=0){ G.hand[i]=randSuit(); renderSlot(i,true); sfx.deal(); } }

  updPlatforms(dtW,true);
  for(let i=G.golds.length-1;i>=0;i--){ const g=G.golds[i];
    if(g.y>G.cam.y+VH+400){ G.golds.splice(i,1); continue; }
    if(Math.hypot(g.x-G.player.x,g.y-G.player.y)<42){ G.golds.splice(i,1); collectGold(g); } }
  updEnemies(dtW,true);

  /* player */
  const p=G.player;
  const ax=(keys.r?1:0)-(keys.l?1:0);
  if(ax) p.face=ax;
  p.vx+=(ax*MOVE-p.vx)*Math.min(1,dt*13);
  p.vy=Math.min(MAXFALL,p.vy+GRAV*dt);
  const prevFeet=p.y+20;
  p.x+=p.vx*dt+(p.stand?p.stand.dx:0);
  p.y+=p.vy*dt;
  p.x=clamp(p.x,-BAND+16,BAND-16);

  let grounded=false, lastPlat=null;
  if(p.stand && (p.stand.broken||G.plats.indexOf(p.stand)<0)) p.stand=null;
  if(p.vy>=0){ for(const pl of G.plats){ if(pl.broken) continue;
    const top=pl.y, feet=p.y+20;
    if(prevFeet<=top+12 && feet>=top && Math.abs(p.x-pl.x)<=pl.w/2+6){
      if(pl.type==='H'){ p.vy=-BOUNCE; sfx.bounce();
        for(let k=0;k<6;k++) G.parts.push({type:'heart',x:pl.x+rand(-pl.w/2,pl.w/2),y:top-4,vx:rand(-40,40),
          vy:rand(-190,-90),g:.25,life:rand(.5,.8),max:.8,size:rand(10,16),color:'#e05a72'});
        continue; }
      const impact=p.vy; p.y=top-20; p.vy=0; grounded=true; lastPlat=pl;
      if(impact>760){ sfx.land(); p.squash=.14;
        for(let k=0;k<8;k++) G.parts.push({type:'spark',x:p.x,y:top,vx:rand(-90,90),vy:rand(-70,-10),g:.5,
          life:rand(.2,.4),max:.4,size:rand(1.5,3),color:'rgba(246,239,221,.8)'}); }
      else p.squash=Math.max(p.squash,.06);
      if(pl.type==='S' && pl.breakT<0) pl.breakT=0;
      if(pl.type==='D'||pl.type==='C'||pl.type==='start') G.safePlat=pl;
      break;
    }
  }}
  p.onGround=grounded; p.stand=lastPlat;
  if(grounded) p.coyote=.11; else p.coyote-=dt;
  if(p.buf>0){ p.buf-=dt;
    if(p.coyote>0){ p.vy=-JUMP; p.coyote=0; p.buf=0; sfx.jump(); } }
  p.squash=Math.max(0,p.squash-dt);

  if(p.y-20>G.cam.y+VH+90) damage();

  const altM=-(p.y+20)/10;
  if(altM>G.maxAlt){ G.maxAlt=altM;
    if(G.maxAlt>=G.nextToast){ toast(PHRASES[G.toastIdx++%PHRASES.length]); G.nextToast+=25; sfx.mile(); } }

  /* camera */
  const desY=p.y-VH*.45; G.cam.y+=(desY-G.cam.y)*Math.min(1,dt*7);
  const maxOff=Math.max(0,VW/2-360), desX=clamp(p.x,-maxOff,maxOff)*.9;
  G.cam.x+=(desX-G.cam.x)*Math.min(1,dt*5);

  while(G.genY>G.cam.y-320) spawnRow();
  for(let i=G.plats.length-1;i>=0;i--)
    if(G.plats[i]!==G.safePlat && G.plats[i].y>G.cam.y+VH+450) G.plats.splice(i,1);

  updParts(dtW);
}

/* ============================== RENDER ============================== */

function tone(f0,f1,dur,type,vol,delay){

  const t=AC.currentTime+delay, o=AC.createOscillator(), g=AC.createGain();
  o.type=type; o.frequency.setValueAtTime(Math.max(1,f0),t);
  o.frequency.exponentialRampToValueAtTime(Math.max(1,f1),t+dur);
  g.gain.setValueAtTime(vol,t); g.gain.exponentialRampToValueAtTime(.001,t+dur);
  o.connect(g); g.connect(master); o.start(t); o.stop(t+dur+.03);
}
function noise(dur,vol,f){
  if(!AC||muted) return; vol=vol||.3; f=f||800;
  const n=Math.floor(AC.sampleRate*dur), buf=AC.createBuffer(1,n,AC.sampleRate), d=buf.getChannelData(0);
  for(let i=0;i<n;i++) d[i]=(Math.random()*2-1)*(1-i/n);
  const src=AC.createBufferSource(); src.buffer=buf;
  const fl=AC.createBiquadFilter(); fl.type='lowpass'; fl.frequency.value=f;
  const g=AC.createGain(); g.gain.value=vol;
  src.connect(fl); fl.connect(g); g.connect(master); src.start();
}

function spawnRow(){
  G.rowIdx++;
  const y=G.genY, alt=-y/10;
  let x=clamp(G.prevX+rand(-190,190), -BAND+70, BAND-70);
  let type;
  if(G.rowIdx<=5){ type=Math.random()<.5?'D':'H'; }
  else{ const r=Math.random(); type=r<.42?'D': r<.58?'H': r<.76?'C':'S'; }
  const w = type==='D'?rand(100,150) : type==='H'?rand(92,128) : rand(92,126);
  const pl={x,y,w,h:60,type, baseX:x, amp:type==='C'?rand(45,85):0, speed:rand(.7,1.3),
    phase:rand(0,6.28), dx:0, breakT:-1, broken:false};
  G.plats.push(pl); G.prevX=x;

  if(G.rowIdx-G.lastJok>=6 && Math.random()<.08){
    G.golds.push({x:clamp(x+rand(-170,170),-BAND+40,BAND-40), y:y-rand(55,120), ph:rand(0,6.28)});
    G.lastJok=G.rowIdx;
  }
  if(alt>15 && Math.random()<Math.min(.5,.12+alt*.003)){
    if(Math.random()<.6){
      G.enemies.push({kind:'bat', x:rand(-BAND+60,BAND-60), y:y-rand(24,110), dir:Math.random()<.5?-1:1,
        sp:Math.min(300, rand(110,160)+alt*1.1), ph:rand(0,6.28), frozenT:0});
    } else if(type==='D'||type==='C'){
      G.enemies.push({kind:'chip', plat:pl, x:x, y:y-17, dir:1, sp:Math.min(190, 60+alt*.7), frozenT:0});
    }
  }
  let gap=rand(88,132); if(G.rowIdx<6) gap=rand(78,95);
  G.genY-=gap;
}

  const rid=G.runId; setTimeout(()=>{ if(G.runId===rid) renderSlot(i,false); },340);
  const p=G.player;
  switch(s){
    case 'H':
      p.vy=-ROCKET; G.shake=Math.max(G.shake,5); sfx.rocket();

    case 'D':
      G.shieldT=2.5; sfx.shield();
      G.parts.push({type:'ring',x:p.x,y:p.y,size:10,gr:300,life:.35,max:.35,col:'233,184,76',lw:4});

    case 'C': {
      sfx.freeze(); let n=0;

      G.parts.push({type:'ring',x:p.x,y:p.y,size:14,gr:n?620:240,life:.5,max:.5,col:'150,220,245',lw:3});
      break; }
    case 'S': {

function damage(){
  if(G.invulnT>0) return;
  const p=G.player;
  G.lives--; updateLives();
  G.flash=1; G.hitT=.4; G.shake=15; sfx.hurt();
  for(let k=0;k<26;k++) G.parts.push(spark(p.x,p.y,k%2?'#e9b84c':'#f6efdd',rand(80,340)));
  G.parts.push({type:'ring',x:p.x,y:p.y,size:10,gr:500,life:.4,max:.4,col:'200,53,74',lw:5});
  if(G.lives<=0){ gameOver(); return; }
  const sp=G.safePlat;
  p.x=sp.x; p.y=sp.y-22; p.vx=0; p.vy=0; p.stand=null; p.coyote=0;
  G.cam.y=p.y-VH*.45; G.invulnT=2.2;
}

/* ============================== UPDATE ============================== */
function updPlatforms(dtW,live){
  for(let i=G.plats.length-1;i>=0;i--){ const pl=G.plats[i];
    if(pl.type==='C'){ const nx=pl.baseX+Math.sin(G.wt*pl.speed+pl.phase)*pl.amp; pl.dx=nx-pl.x; pl.x=nx; }
    else pl.dx=0;
    if(live && pl.type==='S' && !pl.broken && pl.breakT>=0){
      pl.breakT+=dtW; if(pl.breakT>0.55) breakPlatIdx(i);
    }
  }
}
function updEnemies(dtW,live){
  for(let i=G.enemies.length-1;i>=0;i--){ const e=G.enemies[i];
    if(e.frozenT>0) e.frozenT-=dtW;
    else{
      if(e.kind==='bat'){ e.x+=e.dir*e.sp*dtW; e.ph+=dtW*9;
        if(e.x>BAND-60){e.x=BAND-60;e.dir=-1;} if(e.x<-BAND+60){e.x=-BAND+60;e.dir=1;} }
      else{ const pl=e.plat; e.x+=e.dir*e.sp*dtW;
        const lo=pl.x-pl.w/2+16, hi=pl.x+pl.w/2-16;
        if(e.x<lo){e.x=lo;e.dir=1;} if(e.x>hi){e.x=hi;e.dir=-1;} }
    }
    if(e.kind==='chip') e.y=e.plat.y-17;
    if(e.y>G.cam.y+VH+(live?500:600)){ G.enemies.splice(i,1); continue; }
    if(live){
      const p=G.player, rad=(e.kind==='bat'?14:15)+16;
      if(G.invulnT<=0 && Math.hypot(e.x-p.x,e.y-p.y)<rad){
        if(G.shieldT>0) killEnemy(i,true); else damage();
      }
    }
  }
}
function updParts(dt){
  for(let i=G.parts.length-1;i>=0;i--){ const q=G.parts[i]; q.life-=dt;
    if(q.life<=0){ G.parts.splice(i,1); continue; }
    if(q.type==='ring'){ q.size+=q.gr*dt; continue; }
    q.vy+=(q.g||0)*900*dt; q.x+=q.vx*dt; q.y+=q.vy*dt;
    if(q.vr) q.rot=(q.rot||0)+q.vr*dt;
  }
  if(G.parts.length>420) G.parts.splice(0,G.parts.length-420);
}

function update(dt){
  const target=(G.slowT>0||G.hitT>0)?.32:1;
  G.ts+=(target-G.ts)*Math.min(1,dt*8);
  const dtW=dt*G.ts;
  G.wt+=dtW;
  if(G.slowT>0)G.slowT-=dt; if(G.hitT>0)G.hitT-=dt;
  if(G.shieldT>0)G.shieldT-=dt; if(G.invulnT>0)G.invulnT-=dt;
  G.flash=Math.max(0,G.flash-dt*2.2);
  G.shake=Math.max(0,G.shake-dt*40);

  for(let i=0;i<5;i++) if(!G.hand[i]){ G.refill[i]-=dt;
    if(G.refill[i]<=0){ G.hand[i]=randSuit(); renderSlot(i,true); sfx.deal(); } }

  updPlatforms(dtW,true);
  for(let i=G.golds.length-1;i>=0;i--){ const g=G.golds[i];
    if(g.y>G.cam.y+VH+400){ G.golds.splice(i,1); continue; }
    if(Math.hypot(g.x-G.player.x,g.y-G.player.y)<42){ G.golds.splice(i,1); collectGold(g); } }
  updEnemies(dtW,true);

  /* player */
  const p=G.player;
  const ax=(keys.r?1:0)-(keys.l?1:0);
  if(ax) p.face=ax;
  p.vx+=(ax*MOVE-p.vx)*Math.min(1,dt*13);
  p.vy=Math.min(MAXFALL,p.vy+GRAV*dt);
  const prevFeet=p.y+20;
  p.x+=p.vx*dt+(p.stand?p.stand.dx:0);
  p.y+=p.vy*dt;
  p.x=clamp(p.x,-BAND+16,BAND-16);

  let grounded=false, lastPlat=null;
  if(p.stand && (p.stand.broken||G.plats.indexOf(p.stand)<0)) p.stand=null;
  if(p.vy>=0){ for(const pl of G.plats){ if(pl.broken) continue;
    const top=pl.y, feet=p.y+20;
    if(prevFeet<=top+12 && feet>=top && Math.abs(p.x-pl.x)<=pl.w/2+6){
      if(pl.type==='H'){ p.vy=-BOUNCE; sfx.bounce();
        for(let k=0;k<6;k++) G.parts.push({type:'heart',x:pl.x+rand(-pl.w/2,pl.w/2),y:top-4,vx:rand(-40,40),
          vy:rand(-190,-90),g:.25,life:rand(.5,.8),max:.8,size:rand(10,16),color:'#e05a72'});
        continue; }
      const impact=p.vy; p.y=top-20; p.vy=0; grounded=true; lastPlat=pl;
      if(impact>760){ sfx.land(); p.squash=.14;
        for(let k=0;k<8;k++) G.parts.push({type:'spark',x:p.x,y:top,vx:rand(-90,90),vy:rand(-70,-10),g:.5,
          life:rand(.2,.4),max:.4,size:rand(1.5,3),color:'#f6efdd'}); }
      else p.squash=Math.max(p.squash,.06);
      if(pl.type==='S' && pl.breakT<0) pl.breakT=0;
      if(pl.type==='D'||pl.type==='C'||pl.type==='start') G.safePlat=pl;
      break;
    }
  }}
  p.onGround=grounded; p.stand=lastPlat;
  if(grounded) p.coyote=.11; else p.coyote-=dt;
  if(p.buf>0){ p.buf-=dt;
    if(p.coyote>0){ p.vy=-JUMP; p.coyote=0; p.buf=0; sfx.jump(); } }
  p.squash=Math.max(0,p.squash-dt);

  if(p.y-20>G.cam.y+VH+90) damage();

  const altM=-(p.y+20)/10;
  if(altM>G.maxAlt){ G.maxAlt=altM;
    if(G.maxAlt>=G.nextToast){ toast(PHRASES[G.toastIdx++%PHRASES.length]); G.nextToast+=25; sfx.mile(); } }

  /* camera */
  const desY=p.y-VH*.45; G.cam.y+=(desY-G.cam.y)*Math.min(1,dt*7);
  const maxOff=Math.max(0,VW/2-360), desX=clamp(p.x,-maxOff,maxOff)*.9;
  G.cam.x+=(desX-G.cam.x)*Math.min(1,dt*5);

  while(G.genY>G.cam.y-320) spawnRow();
  for(let i=G.plats.length-1;i>=0;i--)
    if(G.plats[i]!==G.safePlat && G.plats[i].y>G.cam.y+VH+450) G.plats.splice(i,1);

  updParts(dtW);
}

/* ============================== RENDER ============================== */
const wx=x=>x-G.cam.x+VW/2, wy=y=>y-G.cam.y;

function drawPlat(pl){
  const x=wx(pl.x)-pl.w/2, y=wy(pl.y);
  if(x>VW+60||x<-pl.w-60||y>VH+90||y<-110) return;
  let wob=0; if(pl.type==='S'&&pl.breakT>=0) wob=Math.sin(G.t*46)*Math.min(3,(pl.breakT/.55)*3);
  ctx.save(); ctx.translate(wob,0);

  if(pl.type==='start'){
    for(let k=2;k>=1;k--){ rr(x-k*3,y+k*3,pl.w+k*6,pl.h,10);
      ctx.fillStyle=k===2?'#8a5f14':'#b98a2c'; ctx.fill(); }
    const g=ctx.createLinearGradient(0,y,0,y+pl.h); g.addColorStop(0,'#ffe9a8'); g.addColorStop(1,'#d99b2e');

function drawParts(){
  for(const q of G.parts){
    const a=clamp(q.life/q.max,0,1), x=wx(q.x), y=wy(q.y);
    if(x<-40||x>VW+40||y<-40||y>VH+40) continue;
    ctx.globalAlpha=a;
    switch(q.type){
      case 'spark': ctx.fillStyle=q.color; ctx.beginPath(); ctx.arc(x,y,q.size*a+.5,0,7); ctx.fill(); break;
      case 'ring': ctx.strokeStyle='rgba('+q.col+','+(a*.9)+')'; ctx.lineWidth=(q.lw||3)*a+1;
        ctx.beginPath(); ctx.arc(x,y,q.size,0,7); ctx.stroke(); break;
      case 'shard': ctx.save(); ctx.translate(x,y); ctx.rotate(q.rot||0);
        ctx.fillStyle=q.color; ctx.strokeStyle='#c9bd97'; ctx.lineWidth=1;
        ctx.fillRect(-q.w/2,-q.h/2,q.w,q.h); ctx.strokeRect(-q.w/2,-q.h/2,q.w,q.h); ctx.restore(); break;
      case 'star': case 'heart': ctx.font=q.size+'px Georgia'; ctx.fillStyle=q.color;
        ctx.fillText(q.type==='star'?'✦':'♥',x,y); break;
      case 'text': ctx.font='bold 15px Georgia'; ctx.strokeStyle='rgba(0,0,0,.5)'; ctx.lineWidth=3;
        ctx.strokeText(q.txt,x,y-(1-a)*26); ctx.fillStyle='#f3d98a'; ctx.fillText(q.txt,x,y-(1-a)*26); break;
    }
  }
  ctx.globalAlpha=1;
}

function render(){
  const t=G.t, camX=G.cam.x, camY=G.cam.y;
  let shx=0,shy=0;
  if(G.shake>0){ shx=(Math.random()*2-1)*G.shake; shy=(Math.random()*2-1)*G.shake; }
  ctx.save(); ctx.translate(shx,shy);
  ctx.textAlign='center'; ctx.textBaseline='middle';

  const aF=clamp(-camY/2200,0,1);
  const g=ctx.createLinearGradient(0,0,0,VH);
  g.addColorStop(0,mix(COL.topA,COL.topB,aF)); g.addColorStop(1,mix(COL.botA,COL.botB,aF));
  ctx.fillStyle=g; ctx.fillRect(-24,-24,VW+48,VH+48);

  const fAmt=1-clamp(-camY/900,0,1);
  if(fAmt>.02 && feltPat){ ctx.globalAlpha=fAmt*.8; ctx.save();
    ctx.translate(0,mod(camY*.5,140)); ctx.fillStyle=feltPat;
    ctx.fillRect(-24,-164,VW+48,VH+348); ctx.restore(); ctx.globalAlpha=1; }

  for(const s of bgSuits){
    const yy=mod(s.y-camY*s.f,VH+520)-260;
    const xx=mod(s.x*(VW+480)-camX*s.f,VW+480)-240;
    ctx.save(); ctx.translate(xx,yy); ctx.rotate(s.r+t*s.rs);
    ctx.globalAlpha=s.a; ctx.font=s.s+'px Georgia'; ctx.fillStyle='#f6efdd'; ctx.fillText(s.g,0,0); ctx.restore();
  }
  ctx.globalAlpha=1;
  for(const st of bgStars){
    const yy=mod(st.y-camY*st.f,VH+200)-100;
    const xx=mod(st.x*(VW+240)-camX*st.f,VW+240)-120;
    const a=.15+.3*(.5+.5*Math.sin(t*st.tw+st.ph));
    ctx.fillStyle=st.c+a+')';
    const s2=2+.8*Math.sin(t*st.tw+st.ph);
    ctx.fillRect(xx-s2/2,yy-s2/2,s2,s2);
  }

  for(const pl of G.plats) drawPlat(pl);
  for(const gl of G.golds) drawGold(gl);
  for(const e of G.enemies) drawEnemy(e);
  if(G.state!=='menu') drawPlayer();
  drawParts();
  ctx.restore();

  if(vig) ctx.drawImage(vig,0,0,VW,VH);
  if(G.slowT>0){ ctx.fillStyle='rgba(233,184,76,'+(.1+.05*Math.sin(t*6))+')'; ctx.fillRect(0,0,VW,VH); }
  if(G.flash>0){ ctx.fillStyle='rgba(255,240,235,'+(G.flash*.35)+')'; ctx.fillRect(0,0,VW,VH); }
}
