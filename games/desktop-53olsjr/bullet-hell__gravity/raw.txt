  setPips();syncBest();
}

/* ================= update ================= */
function update(dt){
  time+=dt;
  if(banner){banner.t+=dt;if(banner.t>1.8)banner=null;}
  invuln=Math.max(0,invuln-dt);hitFlash=Math.max(0,hitFlash-dt*1.6);shake*=Math.exp(-5.5*dt);if(shake<.2)shake=0;

  /* target */
  if(state===ST.play&&P.alive){TGT.x=P.x;TGT.y=P.y;}
  else{lastPX=W*.5+Math.sin(time*.4)*W*.3;lastPY=H*.68+Math.sin(time*.9)*H*.07;TGT.x=lastPX;TGT.y=lastPY;}

  /* gravity shift */
  shiftT-=dt;
  if(warnNew===null&&shiftT<=2){warnNew=(gDir+1+Math.floor(rand(3)))%4;sShift();}
  if(shiftT<=0){gDir=warnNew;warnNew=null;shiftDur=rand(10,16);shiftT=shiftDur;shake=Math.max(shake,7);}

  /* wells */
  wellT-=dt;
  if(wellT<=0&&wells.length<2){wellT=rand(8,14);
    let x,y,tries=0;do{x=rand(W*.2,W*.8);y=rand(H*.2,H*.75);tries++;}while(state===ST.play&&P.alive&&Math.hypot(x-P.x,y-P.y)<200*S&&tries<10);
    wells.push({x,y,t:0,life:6,sp:rand(TAU),dir:Math.random()<.5?1:-1});sWell();}
  for(let i=wells.length-1;i>=0;i--){wells[i].t+=dt;if(wells[i].t>wells[i].life)wells.splice(i,1);}

  /* spawning */
  if(state===ST.play&&wave>0){spawnT+=dt;
    while(spawnQ.length&&spawnT>=spawnQ[0].d){const s=spawnQ.shift();spawnFoe(s.t());}}
  if(state===ST.menu&&foes.length<4&&Math.random()<dt*.7)spawnFoe(Math.random()<.6?'mite':'arcer');

  /* wave progression */
  if(state===ST.play){
    if(wave===0){waveGap-=dt;if(waveGap<=0)startWave(1);}
    else if(!spawnQ.length&&!foes.length){waveGap-=dt;if(waveGap<=0){waveGap=2.2;startWave(wave+1);}}
  }

  /* player */
  if(state===ST.play&&P.alive){
    if(P.tx!=null){P.x+=(P.tx-P.x)*Math.min(1,dt*14);P.y+=(P.ty-P.y)*Math.min(1,dt*14);}
    else{let ax=0,ay=0;
      if(keys.has('a')||keys.has('ArrowLeft'))ax-=1;if(keys.has('d')||keys.has('ArrowRight'))ax+=1;
      if(keys.has('w')||keys.has('ArrowUp'))ay-=1;if(keys.has('s')||keys.has('ArrowDown'))ay+=1;
      const l=Math.hypot(ax,ay)||1,A=3400*S;
      P.vx+=ax/l*A*dt;P.vy+=ay/l*A*dt;}
    const g=Gv();P.vx+=g.x*.4*dt;P.vy+=g.y*.4*dt;
    const dmp=Math.exp(-3.4*dt);P.vx*=dmp;P.vy*=dmp;
    const sp=Math.hypot(P.vx,P.vy),mx=520*S;if(sp>mx){P.vx*=mx/sp;P.vy*=mx/sp;}
    P.x=clamp(P.x+P.vx*dt,12,W-12);P.y=clamp(P.y+P.vy*dt,12,H-12);
    /* well tug on player */
    for(const w of wells){const dx=w.x-P.x,dy=w.y-P.y,d2=dx*dx+dy*dy,R=260*S;
      if(d2<R*R){const d=Math.sqrt(d2)||1,a=Math.min(1500*S,7e6*S/Math.max(d2,900))*.14;P.vx+=dx/d*a*dt;P.vy+=dy/d*a*dt;}}

    /* firing */
    P.cool-=dt;
    let want=firing||keys.has(' ')||autofire;
    if(want&&P.cool<=0){
      let vx,vy;
      if(autofire){let bestF=null,bd=1e9;for(const f of foes)if(f.t>.8){const d=Math.hypot(f.x-P.x,f.y-P.y);if(d<bd){bd=d;bestF=f;}}
        if(bestF)[vx,vy]=aimBallistic(P.x,P.y,bestF.x,bestF.y,980*S,14*S);}
      else{const a=Math.atan2(mouse.y-P.y,mouse.x-P.x);vx=Math.cos(a)*980*S;vy=Math.sin(a)*980*S;}
      const bx=P.x+(mouseOn&&!autofire?Math.cos(Math.atan2(mouse.y-P.y,mouse.x-P.x))*14:0),
            by=P.y-(autofire?0:Math.sin(Math.atan2(mouse.y-P.y,mouse.x-P.x))*14);
      bullets.push({x:bx,y:by,vx,vy,r:4.5*S,t:0});P.cool=.15;sShoot();
      parts.push({x:bx,y:by,vx:vx*.06,vy:vy*.06,t:0,life:.08,c:'#bffcf2',r:5*S});}
  }

  /* foes */
  const errBase=(state===ST.play?Math.max(10,64-wave*5):90)*S;
  for(let i=foes.length-1;i>=0;i--){const f=foes[i];f.t+=dt;f.flash=Math.max(0,f.flash-dt*8);
    const ease=1-Math.pow(1-clamp(f.t/1,0,1),3);
    f.y=f.ay+Math.sin(f.t*2+f.ph)*(f.type==='mite'?14:9)*S;
    if(f.type==='mite')f.x+=Math.cos(f.t*.8+f.ph)*26*S*dt;
    if(f.type==='heavy')f.x+=Math.sin(f.t*.35)*18*S*dt;
    f.x=clamp(f.x+(f.ay*0-f.y*0),W*.08,W*.92);
    if(state===ST.menu&&f.t>7)foes.splice(i,1);
    if(f.t<.9||state===ST.over)continue;
    f.cool-=dt;if(f.cool>0)continue;
    const sp=S;
    if(f.type==='mite'){const[vx,vy]=aimBallistic(f.x,f.y,TGT.x,TGT.y,330*sp,errBase);fireShot(f.x,f.y,vx,vy,5,'#ffb14e');f.cool=rand(2,2.8);}
    else if(f.type==='arcer'){fanFire(f,5,.62,300*sp,5.5,'#ff5c49');f.cool=3.1;}
    else if(f.type==='veil'){ringFire(f,12,240*sp,4.5,'#b78cff');f.cool=2.6;}
    else{ /* heavy */
      f.pat=(f.pat+1)%3;
      if(f.pat===0){for(let k=-1;k<=1;k++){const[vx,vy]=aimBallistic(f.x,f.y,TGT.x,TGT.y,340*sp,errBase*.7);fireShot(f.x,f.y,vx+k*40,vy,7,'#ffd166');}f.cool=2.3;}
      else if(f.pat===1){fanFire(f,7,.85,290*sp,6,'#ff5c49');f.cool=2.6;}
      else{ringFire(f,16,220*sp,5.5,'#b78cff');f.cool=2.9;}}
    sEShot();}

  /* wells influence helper */
  const wellPull=(b,fac)=>{for(const w of wells){const dx=w.x-b.x,dy=w.y-b.y,d2=dx*dx+dy*dy,R=260*S;
    if(d2<R*R){const d=Math.sqrt(d2)||1,a=Math.min(1500*S,7e6*S/Math.max(d2,900))*fac;b.vx+=dx/d*a*dt;b.vy+=dy/d*a*dt;}}};

  /* player bullets */
  const g=Gv();
  for(let i=bullets.length-1;i>=0;i--){const b=bullets[i];b.t+=dt;
    b.vx+=g.x*dt;b.vy+=g.y*dt;wellPull(b,1);
    b.px=b.x;b.py=b.y;b.x+=b.vx*dt;b.y+=b.vy*dt;
    let dead=b.t>1.4||b.x<-60||b.x>W+60||b.y<-80||b.y>H+60;
    if(!dead)for(let j=foes.length-1;j>=0;j--){const f=foes[j];
      if(f.t<.3)continue;
      const dx=f.x-b.x,dy=f.y-b.y;
      if(dx*dx+dy*dy<(f.r+b.r)*(f.r+b.r)){f.hp--;f.flash=1;dead=true;sBoom(.4);
        boom(b.x,b.y,'#bffcf2',5,180*S);
        if(f.hp<=0){const p=FT[f.type];kills++;streak++;streakT=2.5;mult=1+Math.min(7,Math.floor(streak/3));
          const pts=p.sc*mult;score+=pts;addFloat(f.x,f.y-f.r,'+'+pts,'#ffe9b0',13);
          boom(f.x,f.y,f.type==='mite'?'#ffb14e':f.type==='arcer'?'#ff5c49':f.type==='veil'?'#b78cff':'#ffd166',clamp(p.r*2,10,26),300*S);
          sBoom(clamp(p.r/15,.5,1.3));shake=Math.max(shake,f.type==='heavy'?14:4);
          if(state===ST.play&&Math.random()<.13)drops.push({x:f.x,y:f.y,vx:rand(-30,30),vy:-40,t:0});
          foes.splice(j,1);}
        break;}}
    if(dead)bullets.splice(i,1);}

  /* enemy shots */
  for(let i=shots.length-1;i>=0;i--){const b=shots[i];
    b.vx+=g.x*dt;b.vy+=g.y*dt;wellPull(b,1);
    b.x+=b.vx*dt;b.y+=b.vy*dt;
    if(b.x<-70||b.x>W+70||b.y<-90||b.y>H+70){shots.splice(i,1);continue;}
    if(state===ST.play&&P.alive){const dx=b.x-P.x,dy=b.y-P.y,d2=dx*dx+dy*dy;
      const hitR=(P.r+b.r*.45)*(P.r+b.r*.45),nearR=38*S;
      if(invuln<=0&&d2<hitR){hitPlayer(b);shots.splice(i,1);continue;}
      /* near miss */
      if(!b.near&&d2<nearR*nearR)b.near=true;
      else if(b.near&&!b.nmDone&&d2>58*S*58*S){b.nmDone=true;if(state===ST.play){score+=15;addFloat(P.x+rand(-20,20),P.y-24,'CLOSE +15','#5df2dd',11);sClose();}}}
  }

  /* streak decay */
  if(streakT>0){streakT-=dt;if(streakT<=0){streak=0;mult=1;}}

  /* drops */
  for(let i=drops.length-1;i>=0;i--){const d=drops[i];d.t+=dt;
    d.vy+=g.y*.35*dt;d.vx*=Math.exp(-.8*dt);
    d.x+=d.vx*dt+Math.sin(d.t*4)*20*dt;d.y+=d.vy*dt;
    if(state===ST.play&&P.alive){const dx=d.x-P.x,dy=d.y-P.y;
      if(dx*dx+dy*dy<30*30){if(shields<3){shields++;setPips();addFloat(P.x,P.y-26,'SHIELD +1','#7dff9e',13);}
        else{score+=250;addFloat(P.x,P.y-26,'+250','#ffe9b0',13);}sHeal();drops.splice(i,1);continue;}}
    if(d.t>9||d.y>H+40||d.x<-40||d.x>W+40)drops.splice(i,1);}

  /* particles / floaters / ambient */
  for(let i=parts.length-1;i>=0;i--){const p=parts[i];p.t+=dt;
    if(p.ring){if(p.t>p.life)parts.splice(i,1);continue;}
    p.vx*=Math.exp(-2.5*dt);p.vy=p.vy*Math.exp(-2.5*dt)+g.y*.3*dt;
    p.x+=p.vx*dt;p.y+=p.vy*dt;if(p.t>p.life)parts.splice(i,1);}
  for(let i=floats.length-1;i>=0;i--){const f=floats[i];f.t+=dt;f.y-=28*S*dt;if(f.t>f.life)floats.splice(i,1);}
  for(const s of stars){s.x+=(g.x*.035*s.z)*dt;s.y+=(g.y*.035*s.z)*dt;
    if(s.x<0)s.x+=W;if(s.x>W)s.x-=W;if(s.y<0)s.y+=H;if(s.y>H)s.y-=H;}
  for(const d of dust){d.vx+=g.x*dt;d.vy+=g.y*dt;const dm=Math.exp(-1.4*dt);d.vx*=dm;d.vy*=dm;
    d.x+=d.vx*dt;d.y+=d.vy*dt;
    if(d.x<-30)d.x=W+20;if(d.x>W+30)d.x=-20;if(d.y<-30)d.y=H+20;if(d.y>H+30)d.y=-20;}

  if(state===ST.over){overT+=dt;if(overT>.9&&!overEl.classList.contains('on'))overEl.classList.add('on');}
}
function hitPlayer(){shields--;invuln=1.2;hitFlash=1;shake=15;sHit();
  setPips();pipAnim(shields);boom(P.x,P.y,'#ff5c49',16,320*S);
  if(shields<=0)gameOver();}

/* ================= draw ================= */
function draw(){
  ctx.clearRect(0,0,W,H);
  const sx=shake?rand(-shake,shake):0,sy=shake?rand(-shake,shake):0;
  ctx.save();ctx.translate(sx,sy);

  /* background + gravity floor glow */
  ctx.drawImage(bg,0,0,W,H);
  const d=DIRS[gDir],eg=140*S;
  let gx=0,gy=0,gw=W,gh=H;
  if(d.y===1){gy=H-eg;gh=eg;}else if(d.y===-1){gy=0;gh=eg;}
  else{gw=eg;if(d.x===-1)gx=0;else gx=W-eg;}
  const fg=ctx.createLinearGradient(gx+(d.x?eg/2:W/2),gy+(d.y?eg/2:H/2),gx+(d.x?-eg/2:W/2),gy+(d.y?-eg/2:H/2));
  fg.addColorStop(0,'rgba(93,242,221,.10)');fg.addColorStop(1,'rgba(93,242,221,0)');
  ctx.fillStyle=fg;ctx.fillRect(gx,gy,gw,gh);
  ctx.strokeStyle='rgba(93,242,221,'+(.18+.1*Math.sin(time*3))+')';ctx.lineWidth=1.5;
  for(let i=0;i<5;i++){const p=((time*.6+i/5)%1)*eg;ctx.beginPath();
    if(d.y===1){ctx.moveTo(W*.2,H-p);ctx.lineTo(W/2,H-p-14*S);ctx.lineTo(W*.8,H-p);}
    else if(d.y===-1){ctx.moveTo(W*.2,p);ctx.lineTo(W/2,p+14*S);ctx.lineTo(W*.8,p);}
    else if(d.x===-1){ctx.moveTo(p,H*.2);ctx.lineTo(p+14*S,H/2);ctx.moveTo(p,H*.8);}ctx.stroke();}

  /* stars + dust */
  for(const s of stars){ctx.globalAlpha=.25+s.z*.18;ctx.fillStyle='#cfe6e2';ctx.fillRect(s.x,s.y,s.s,s.s);}
  ctx.globalAlpha=1;ctx.strokeStyle='rgba(140,200,205,.3)';ctx.lineWidth=1.2;
  for(const d2 of dust){const sp=Math.hypot(d2.vx,d2.vy);if(sp<10)continue;
    ctx.beginPath();ctx.moveTo(d2.x,d2.y);ctx.lineTo(d2.x-d2.vx*.05,d2.y-d2.vy*.05);ctx.stroke();}

  /* wells */
  for(const w of wells){const a=Math.min(1,w.t*2,(w.life-w.t)*1.5),R=260*S;
    ctx.save();ctx.globalAlpha=a;
    const rg=ctx.createRadialGradient(w.x,w.y,0,w.x,w.y,R*.5);
    rg.addColorStop(0,'rgba(183,140,255,.28)');rg.addColorStop(1,'rgba(183,140,255,0)');
    ctx.fillStyle=rg;ctx.fillRect(w.x-R,w.y-R,R*2,R*2);
    ctx.strokeStyle='rgba(183,140,255,.7)';ctx.lineWidth=2;
    for(let i=0;i<3;i++){const rr=(20+i*26)*S+Math.sin(time*2+i)*4;
      ctx.beginPath();ctx.arc(w.x,w.y,rr,w.sp+w.dir*time*(1.5-i*.3),w.sp+w.dir*time*(1.5-i*.3)+TAU*.7);ctx.stroke();}
    ctx.fillStyle='#e6d8ff';ctx.beginPath();ctx.arc(w.x,w.y,4*S,0,TAU);ctx.fill();
    ctx.restore();}

  /* drops */
  for(const d2 of drops){const bl=d2.t>7?(Math.sin(d2.t*14)>0?1:.3):1;
    ctx.save();ctx.globalCompositeOperation='lighter';ctx.globalAlpha=bl;
    ctx.drawImage(spr('#7dff9e'),d2.x-16*S,d2.y-16*S,32*S,32*S);
    ctx.fillStyle='#7dff9e';ctx.translate(d2.x,d2.y);ctx.rotate(Math.PI/4);
    const s2=7*S;ctx.fillRect(-s2/2,-s2/2,s2,s2);ctx.restore();}

  /* foes */
  for(const f of foes)drawFoe(f);

  ctx.globalCompositeOperation='lighter';
  /* player bullets */
  for(const b of bullets){
    ctx.strokeStyle='rgba(93,242,221,.5)';ctx.lineWidth=b.r;
    ctx.beginPath();ctx.moveTo(b.px||b.x,b.py||b.y);ctx.lineTo(b.x,b.y);ctx.stroke();
    ctx.drawImage(spr('#5df2dd'),b.x-b.r*3.2,b.y-b.r*3.2,b.r*6.4,b.r*6.4);
    ctx.fillStyle='#eafffb';ctx.beginPath();ctx.arc(b.x,b.y,b.r*.6,0,TAU);ctx.fill();}
  /* enemy shots */
  for(const b of shots){
    const near=state===ST.play&&P.alive&&((b.x-P.x)**2+(b.y-P.y)**2)<(38*S)*(38*S);
    ctx.drawImage(spr(b.c),b.x-b.r*3.4,b.y-b.r*3.4,b.r*6.8,b.r*6.8);
    ctx.fillStyle=b.c;ctx.beginPath();ctx.arc(b.x,b.y,b.r,0,TAU);ctx.fill();
    ctx.fillStyle='#fff';ctx.beginPath();ctx.arc(b.x,b.y,b.r*.45,0,TAU);ctx.fill();
    if(near){ctx.strokeStyle='rgba(255,92,73,.8)';ctx.lineWidth=1.5;
      ctx.beginPath();ctx.arc(b.x,b.y,b.r+4+Math.sin(time*20)*2,0,TAU);ctx.stroke();}}
  /* particles */
  for(const p of parts){const a=1-p.t/p.life;
    if(p.ring){ctx.globalAlpha=a*.8;ctx.strokeStyle=p.c;ctx.lineWidth=2.5;
      ctx.beginPath();ctx.arc(p.x,p.y,p.r+(p.t/p.life)*60*S,0,TAU);ctx.stroke();}
    else{ctx.globalAlpha=a;ctx.fillStyle=p.c;ctx.beginPath();ctx.arc(p.x,p.y,p.r*a+ .4,0,TAU);ctx.fill();}}
  ctx.globalAlpha=1;ctx.globalCompositeOperation='source-over';

  /* player */
  if(state===ST.play&&P.alive)drawPlayer();

  /* floaters */
  for(const f of floats){const a=1-f.t/f.life;ctx.globalAlpha=a;
    ctx.font='700 '+f.s+'px Consolas,Menlo,monospace';ctx.textAlign='center';
    ctx.fillStyle=f.c;ctx.fillText(f.txt,f.x,f.y);}
  ctx.globalAlpha=1;

  /* gravity shift warning */
  if(warnNew!==null){const a=.45+.4*Math.sin(time*12);
    const d=DIRS[warnNew];
    ctx.save();ctx.translate(W/2,H*.36);ctx.rotate(Math.atan2(d.y,d.x)+Math.PI/2);
    ctx.globalAlpha=a;ctx.strokeStyle='#ff8a7a';ctx.lineWidth=5*S;ctx.lineCap='round';
