function update(dt){
  time+=dt/1000;
  if(shakeT>0){shakeT=Math.max(0,shakeT-dt/260);if(!shakeT)shakeMag=0;}
  for(const m of motes){m.x-=m.v*dt/1000;m.y-=m.v*.35*dt/1000;if(m.x<-4)m.x=W+4;if(m.y<-4){m.y=H+4;m.x=Math.random()*W;}}
  if(player){
    if(player.anim){player.anim.t+=dt/player.anim.dur;if(player.anim.t>=1){player.anim=null;if(pendingDir)tryMove(pendingDir);else refreshConsole();}}
    if(player.bump){player.bump.t+=dt/130;if(player.bump.t>=1)player.bump=null;}
  }
  for(const b of boxes)if(b.anim){b.anim.t+=dt/b.anim.dur;if(b.anim.t>=1){b.anim=null;if(isTarget(b.gx,b.gy)&&!b.sealed)sealBox(b);}}
  parts=parts.filter(p=>{p.t+=dt/1000;if(!p.ring){p.x+=p.vx*dt/1000;p.y+=p.vy*dt/1000;p.vy+=(p.g||0)*dt/1000;}return p.t<p.max;});
  pops=pops.filter(p=>{p.t+=dt/1000;return p.t<p.max;});
}
