  const padT=padGrid[by][bx];

   al.alive=false;score+=PTS[al.type];
   popups.push({x:ox+(bx+.5)*cell,y:oy+(by+.5)*cell,txt:'+'+PTS[al.type],col:COL[al.type],life:1});
   pendingBooms.push(al);
   if(aliens.every(a=>!a.alive))pendingClear=1.0;
  }
 }
 player.ox=player.x;player.oy=player.y;player.x=nx;player.y=ny;
 animT=0;busy=true;sfx.move();
}

/* ================= update ================= */
function update(dt){
 t+=dt;
 if(screen==='playing'&&!paused){
  inputLock=Math.max(0,inputLock-dt);
  if(pendingClear>0){pendingClear-=dt;if(pendingClear<=0)doClear();}
  else{
   timeLeft-=dt;
   const w=Math.ceil(timeLeft);
   if(w!==lastWarnSec){lastWarnSec=w;if(w<=10&&w>0)sfx.tick();}
   if(timeLeft<=0){timeLeft=0;breach();}
  }
  if(!busy&&!paused&&held.length)tryMove(held[held.length-1]);
 }
 if(screen==='breach'){breachT+=dt;if(breachT>=1.5){shields--;updatePips();if(shields<=0)gameOver();else{loadLevel(levelIdx);setScreen('playing');}}}
 if(animT<1){animT=Math.min(1,animT+dt/.09);if(animT>=1&&busy){busy=false;resolveBooms();}}
 if(bump){bump.t-=dt;if(bump.t<=0)bump=null;}
 for(const p of particles){p.life-=dt;if(p.k==='conf'){p.vy+=p.g*dt;p.rot+=p.vr*dt;}p.x+=p.vx*dt;p.y+=p.vy*dt;}
 particles=particles.filter(p=>p.life>0);
 for(const p of popups){p.life-=dt*.9;p.y-=28*dt;}
 popups=popups.filter(p=>p.life>0);
 shake*=Math.exp(-6*dt);if(shake<.05)shake=0;
 if(banner){banner.t+=dt;if(banner.t>2.4)banner=null;}
 // shooting star
 shoot.next-=dt;
 if(!shoot.on&&shoot.next<=0){shoot={on:true,x:W*(.2+Math.random()*.7),y:H*(.05+Math.random()*.3),vx:-(180+Math.random()*220),vy:60+Math.random()*90,life:.7,max:.7,next:0};}
 if(shoot.on){shoot.x+=shoot.vx*dt;shoot.y+=shoot.vy*dt;shoot.life-=dt;if(shoot.life<=0){shoot.on=false;shoot.next=5+Math.random()*6;}}
 syncHUD();
}
