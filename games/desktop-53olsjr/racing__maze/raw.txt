const T=24,C=6,P=C+1; // tile size, interior tiles per cell, pitch
function genMaze(cols,rows){
  const cells=[]; for(let i=0;i<cols*rows;i++)cells.push({t:1,r:1,b:1,l:1});
  const at=(x,y)=>cells[y*cols+x];
  const vis=new Uint8Array(cols*rows);
  const st=[[0,0]]; vis[0]=1;
  while(st.length){
    const [cx,cy]=st[st.length-1], opts=[];
    if(cy>0&&!vis[(cy-1)*cols+cx])opts.push([cx,cy-1,'t','b']);
    if(cx<cols-1&&!vis[cy*cols+cx+1])opts.push([cx+1,cy,'r','l']);
    if(cy<rows-1&&!vis[(cy+1)*cols+cx])opts.push([cx,cy+1,'b','t']);
    if(cx>0&&!vis[cy*cols+cx-1])opts.push([cx-1,cy,'l','r']);
    if(!opts.length){st.pop();continue}
    const [nx,ny,a,b]=opts[(Math.random()*opts.length)|0];
    at(cx,cy)[a]=0; at(nx,ny)[b]=0; vis[ny*cols+nx]=1; st.push([nx,ny]);
  }
  // knock out a few walls → shortcuts & loops
  const loops=Math.round(cols*rows*.08);
  for(let i=0;i<loops;i++){
    const cx=(Math.random()*cols)|0, cy=(Math.random()*rows)|0, c=at(cx,cy);
    const dirs=[];
    if(c.t&&cy>0)dirs.push(['t','b',cx,cy-1]);
    if(c.l&&cx>0)dirs.push(['l','r',cx-1,cy]);
    if(c.r&&cx<cols-1)dirs.push(['r','l',cx+1,cy]);
    if(c.b&&cy<rows-1)dirs.push(['b','t',cx,cy+1]);
    if(dirs.length){const d=dirs[(Math.random()*dirs.length)|0]; at(cx,cy)[d[0]]=0; at(d[2],d[3])[d[1]]=0;}
  }
  // BFS shortest path length (cells)
  const dist=new Int32Array(cols*rows).fill(-1); dist[0]=0;
  const q=[0]; let head=0;
  while(head<q.length){
    const i=q[head++], x=i%cols, y=(i/cols)|0, c=at(x,y);
    const tryN=(dx,dy,w)=>{ if(w)return; const j=i+dy*cols+dx; if(dist[j]<0){dist[j]=dist[i]+1;q.push(j);} };
    tryN(0,-1,c.t); tryN(1,0,c.r); tryN(0,1,c.b); tryN(-1,0,c.l);
  }
  return {cells,cols,rows,path:Math.max(4,dist[cols*rows-1])};
}
function buildTiles(mz){
  const tc=mz.cols*P+1, tr=mz.rows*P+1;
  const s=new Uint8Array(tc*tr);
  for(let r=0;r<tr;r++)for(let c=0;c<tc;c++)s[r*tc+c]=(r%P===0||c%P===0)?1:0;
  for(let cy=0;cy<mz.rows;cy++)for(let cx=0;cx<mz.cols;cx++){
    const w=mz.cells[cy*mz.cols+cx];
    if(!w.t)for(let c=cx*P+1;c<=cx*P+C;c++)s[cy*P*tc+c]=0;
    if(!w.l&&cx>0)for(let r=cy*P+1;r<=cy*P+C;r++)s[r*tc+cx*P]=0;
  }
  return {solid:s,tc,tr};
}

function moveAxis(car,ax){ // resolve circle vs solid tiles along axis 'x' or 'y'
  const r=CARR; let impact=0;
  for(let pass=0;pass<2;pass++){
    const c0=Math.floor((car.x-r)/T),c1=Math.floor((car.x+r)/T),r0=Math.floor((car.y-r)/T),r1=Math.floor((car.y+r)/T);
    for(let rr=r0;rr<=r1;rr++)for(let cc=c0;cc<=c1;cc++){
      if(cc<0||rr<0||cc>=G.tiles.tc||rr>=G.tiles.tr)continue;
      if(!G.tiles.solid[rr*G.tiles.tc+cc])continue;
      const nx=clamp(car.x,cc*T,(cc+1)*T), ny=clamp(car.y,rr*T,(rr+1)*T);
      let dx=car.x-nx, dy=car.y-ny; const d2=dx*dx+dy*dy;
      if(d2<r*r){
        let d=Math.sqrt(d2), ux,uy;
        if(d<1e-4){ // center inside tile: push along moved axis
          if(ax==='x'){ux=car.x<nx?-1:1;uy=0;} else {uy=car.y<ny?-1:1;ux=0;}
          car[ax]+=ux*(r+0.2);
        }else{ ux=dx/d; uy=dy/d; car.x+=ux*(r-d); car.y+=uy*(r-d); }
        const vn=car.vx*ux+car.vy*uy;
        if(vn<0){ car.vx-=(1+.32)*vn*ux; car.vy-=(1+.32)*vn*uy; impact=Math.max(impact,-vn); }
      }
    }
  }
  return impact;
}

function update(dt){
  G.time+=dt;
  const car=G.car, c=G.cam;
  if(G.state==='menu'){ // attract drift
    c.x=lerp(c.x,G.worldW/2+Math.cos(G.time*.1)*G.worldW*.3,dt*1.6);
    c.y=lerp(c.y,G.worldH/2+Math.sin(G.time*.085)*G.worldH*.3,dt*1.6);
    c.zoom=lerp(c.zoom,baseZoom()*.92,dt*2);
    return;
  }
  if(G.state==='count'){
    const pi=Math.ceil(Math.max(0,G.countT));
    if(pi!==G.lastInt&&pi<=3){G.lastInt=pi;Snd.count(pi);}
    G.countT-=dt;
    if(G.countT<=0&&G.lastInt!==0){G.lastInt=0;Snd.count(0);}
    if(G.countT<=-0.8)G.state='race';
  }

  const racing=G.state==='race';
  // --- car physics ---
  let fX=Math.cos(car.a),fY=Math.sin(car.a), rX=-fY,rY=fX;
  let vF=car.vx*fX+car.vy*fY, vL=car.vx*rX+car.vy*rY;
  if(racing){
    if(keys.up)vF+=540*dt;
    if(keys.down){ if(vF>30)vF-=900*dt; else vF=Math.max(vF-320*dt,-170); }
    if(keys.hb)vF*=Math.exp(-.9*dt);
    const dr=keys.up||keys.down?0.45:1.15;
    vF*=Math.exp(-dr*dt);
    if(!keys.up&&!keys.down){const s=Math.sign(vF);vF-=s*60*dt;if(Math.sign(vF)!==s)vF=0;}
    vF=clamp(vF,-170,MAXF);
    const steer=(keys.right?1:0)-(keys.left?1:0);
    const sf=clamp(Math.abs(vF)/150,0,1)*(vF>=0?1:-1);
    car.a+=steer*2.7*sf*dt;
    const grip=keys.hb?1.9:8.5;
    vL*=Math.exp(-grip*dt);
  }else{ vF*=Math.exp(-4*dt); vL*=Math.exp(-4*dt); }

  fX=Math.cos(car.a);fY=Math.sin(car.a);rX=-fY;rY=fX;
  car.vx=fX*vF+rX*vL; car.vy=fY*vF+rY*vL;
  const px=car.x,py=car.y;
  car.x+=car.vx*dt; let imp=moveAxis(car,'x');
  car.y+=car.vy*dt; imp=Math.max(imp,moveAxis(car,'y'));

  const speed=Math.hypot(car.vx,car.vy);
  if(imp>70){
    G.hits++; Snd.thud(imp); c.shake=Math.min(c.shake+imp*.05,14);
    for(let i=0;i<Math.min(14,4+imp*.04);i++){
      const a=Math.random()*TAU,s=60+Math.random()*imp*.9;
      G.sparks.push({x:car.x,y:car.y,vx:Math.cos(a)*s,vy:Math.sin(a)*s,life:.25+Math.random()*.2,max:.45});
    }
  }

  // --- skids & smoke ---
  const slip=Math.abs(vL);
  if(racing&&((keys.hb&&speed>110)||slip>85)){
    for(const off of [7,-7]){
      G.skids.push({x1:px+rX*off,y1:py+rY*off,x2:car.x+rX*off,y2:car.y+rY*off,life:5});
    }
    if(G.skids.length>480)G.skids.splice(0,G.skids.length-480);
    if(speed>140&&Math.random()<.85){
      const off=(Math.random()<.5?1:-1)*7;
      G.smoke.push({x:car.x-rX*off-fX*14,y:car.y-rY*off-fY*14,vx:(Math.random()-.5)*30,vy:(Math.random()-.5)*30,life:.55,max:.55,s:5+Math.random()*5});
    }
  }

  // --- timers / win / lose ---
  if(racing){
    G.timeLeft-=dt;
    const s=Math.ceil(G.timeLeft);
    if(G.timeLeft<=5.2&&s!==G.tickSec&&G.timeLeft>0){G.tickSec=s;Snd.tick();}
    if(G.timeLeft<=0){G.timeLeft=0;timeUp();}
    else if(Math.hypot(car.x-G.goal[0],car.y-G.goal[1])<T*1.7)finishLevel();
  }

  // --- particles decay ---
  for(let i=G.skids.length-1;i>=0;i--){G.skids[i].life-=dt;if(G.skids[i].life<=0)G.skids.splice(i,1);}
  for(let i=G.smoke.length-1;i>=0;i--){const p=G.smoke[i];p.life-=dt;p.x+=p.vx*dt;p.y+=p.vy*dt;p.s+=18*dt;if(p.life<=0)G.smoke.splice(i,1);}
  for(let i=G.sparks.length-1;i>=0;i--){const p=G.sparks[i];p.life-=dt;p.x+=p.vx*dt;p.y+=p.vy*dt;if(p.life<=0)G.sparks.splice(i,1);}
  for(let i=G.confetti.length-1;i>=0;i--){const p=G.confetti[i];p.life-=dt;p.x+=p.vx*dt;p.y+=p.vy*dt;p.vy+=420*dt;p.rot+=p.vr*dt;if(p.life<=0)G.confetti.splice(i,1);}

  // --- camera ---
  const tz=baseZoom()*(1-clamp(speed/MAXF,0,1)*.1);
  c.zoom=lerp(c.zoom,tz,dt*3);
  c.x=lerp(c.x,car.x+car.vx*.22,dt*5);
  c.y=lerp(c.y,car.y+car.vy*.22,dt*5);
  c.shake=Math.max(0,c.shake-40*dt);

  // --- audio live ---
  if(G.state==='race'||G.state==='count'){
    Snd.engine(true,clamp(speed/MAXF,0,1),!!keys.up);
    Snd.skidSet(racing?((keys.hb&&speed>110)||slip>85)?(keys.hb?.22:Math.min(.2,(slip-85)/400)):0:0);
  }
}

function drawWorld(){
  const c=G.cam, t=G.time;
  const shx=(Math.random()-.5)*c.shake, shy=(Math.random()-.5)*c.shake;
  ctx.save();
  ctx.translate(VW/2+shx,VH/2+shy); ctx.scale(c.zoom,c.zoom); ctx.translate(-c.x,-c.y);

  const hw=VW/(2*c.zoom)+40, hh=VH/(2*c.zoom)+40;
  // asphalt base
  ctx.fillStyle='#151d2a';

  const s=G.tiles.solid, tc=G.tiles.tc, tr=G.tiles.tr;
  const c0=Math.max(0,Math.floor((c.x-hw)/T)), c1=Math.min(tc-1,Math.ceil((c.x+hw)/T));
  const r0=Math.max(0,Math.floor((c.y-hh)/T)), r1=Math.min(tr-1,Math.ceil((c.y+hh)/T));

  // floor tiles (subtle checker)
  for(let r=r0;r<=r1;r++)for(let cc=c0;cc<=c1;cc++){
    if(s[r*tc+cc])continue;
    ctx.fillStyle=((r+cc)&1)?'#161f2d':'#141c29';
    ctx.fillRect(cc*T,r*T,T,T);
  }
  // walls + neon edges
  const edge=(x,y,dx,dy)=>{ctx.beginPath();ctx.moveTo(x,y);ctx.lineTo(x+dx,y+dy);ctx.stroke();};
  for(let pass=0;pass<2;pass++){
    ctx.lineWidth=pass?1.8:6;
    ctx.strokeStyle=pass?'rgba(140,250,255,.9)':'rgba(99,242,255,.13)';
    for(let r=r0;r<=r1;r++)for(let cc=c0;cc<=c1;cc++){
      if(!s[r*tc+cc])continue;
      const x=cc*T,y=r*T;
      const oL=cc>0&&!s[r*tc+cc-1], oR=cc<tc-1&&!s[r*tc+cc+1];
      const oT=r>0&&!s[(r-1)*tc+cc], oB=r<tr-1&&!s[(r+1)*tc+cc];
      if(oL)edge(x,y,0,T); if(oR)edge(x+T,y,0,T);
      if(oT)edge(x,y,T,0); if(oB)edge(x,y+T,T,0);
    }
  }

  // start pad (checker strip at spawn)
  const car=G.car;
  ctx.save(); ctx.translate(car.x,car.y); ctx.rotate(car.a);
  for(let i=0;i<3;i++)for(let j=0;j<5;j++){
    ctx.fillStyle=((i+j)&1)?'#e8f1fb':'#22303f';
    ctx.fillRect(26+i*14,-30+j*12,14,12);
  }
  ctx.restore();

  // goal: checker floor + waving flag
  const [gx,gy]=G.goal;
  for(let i=0;i<5;i++)for(let j=0;j<5;j++){
    ctx.fillStyle=((i+j+((t*3)|0))&1)?'#ff5d9e':'#fff';
    ctx.globalAlpha=.85;
    ctx.fillRect(gx-44+i*17.6,gy-44+j*17.6,17.6,17.6);
  }
  ctx.globalAlpha=1;
  const ringR=30+Math.sin(t*3)*5;
  ctx.strokeStyle='rgba(255,93,158,.5)'; ctx.lineWidth=2;
  ctx.beginPath(); ctx.arc(gx,gy,ringR,0,TAU); ctx.stroke();
  // flag
  const fx=gx+46,fy=gy-46;
  ctx.strokeStyle='#cfe2f3'; ctx.lineWidth=3;
  ctx.beginPath(); ctx.moveTo(fx,fy); ctx.lineTo(fx,fy+52); ctx.stroke();
  for(let i=0;i<8;i++)for(let j=0;j<3;j++){
    const wx=fx+i*6, wy=fy+j*9+Math.sin(t*7+i*.8)*3;
    ctx.fillStyle=((i+j)&1)?'#ff2f7e':'#fff';
    ctx.fillRect(wx,wy,6,9);
  }
  ctx.font='italic 15px Impact,"Arial Black",sans-serif';
  ctx.textAlign='center'; ctx.fillStyle='#ff8ab8';
  ctx.fillText('F I N I S H',gx,gy+70);

  // skid marks
  ctx.lineCap='round';
  for(const k of G.skids){
    ctx.strokeStyle='rgba(10,13,18,'+(.5*Math.min(1,k.life/2))+')';
    ctx.lineWidth=4.5;
    ctx.beginPath();ctx.moveTo(k.x1,k.y1);ctx.lineTo(k.x2,k.y2);ctx.stroke();
  }
  // smoke
  for(const p of G.smoke){
    const a=p.life/p.max;
    ctx.fillStyle='rgba(168,184,200,'+(a*.3)+')';
    ctx.beginPath();ctx.arc(p.x,p.y,p.s,0,TAU);ctx.fill();
  }

  // car
  if(G.state!=='menu'){
    ctx.save(); ctx.translate(car.x,car.y); ctx.rotate(car.a);
    // headlight cones
    const hg=ctx.createLinearGradient(12,0,150,0);
    hg.addColorStop(0,'rgba(255,240,190,.16)'); hg.addColorStop(1,'rgba(255,240,190,0)');
    ctx.fillStyle=hg;
    ctx.beginPath();ctx.moveTo(12,-7);ctx.lineTo(150,-38);ctx.lineTo(150,38);ctx.lineTo(12,7);ctx.closePath();ctx.fill();
    // wheels
    ctx.fillStyle='#0b0f14';
    for(const [wx,wy,st] of [[10,-9.5,1],[10,9.5,1],[-10,-9.5,0],[-10,9.5,0]]){
      ctx.save();ctx.translate(wx,wy);if(st)ctx.rotate((keys.right?1:0-(keys.left?1:0))*.42);
      ctx.fillRect(-5,-3,10,6);ctx.restore();
    }
    // body
    const bg=ctx.createLinearGradient(0,-11,0,11);
    bg.addColorStop(0,'#ffa14d');bg.addColorStop(.5,'#ff7a1c');bg.addColorStop(1,'#d9530a');
    ctx.fillStyle=bg;
    roundRect(-16,-10.5,32,21,5); ctx.fill();
    ctx.strokeStyle='rgba(255,255,255,.35)'; ctx.lineWidth=1; ctx.stroke();
    // stripe + cockpit
    ctx.fillStyle='#fff'; ctx.fillRect(-15,-2.4,30,4.8);
    ctx.fillStyle='#0e1620'; roundRect(-4,-7,12,14,3.5); ctx.fill();
    ctx.fillStyle='rgba(99,242,255,.7)'; ctx.fillRect(1.5,-6,4,12);
    // spoiler
    ctx.fillStyle='#c84e0a'; ctx.fillRect(-17,-9,3.5,18);
    // tail lights
    const brake=keys.down&&G.state==='race';
    if(brake||car.vx*Math.cos(car.a)+car.vy*Math.sin(car.a)<-20){
      ctx.fillStyle='rgba(255,60,50,.9)';
      ctx.shadowColor='#ff3c32'; ctx.shadowBlur=10;
      ctx.fillRect(-17.5,-9,3,4); ctx.fillRect(-17.5,5,3,4);
      ctx.shadowBlur=0;
    }else{ ctx.fillStyle='rgba(255,80,60,.5)'; ctx.fillRect(-17.5,-9,3,4); ctx.fillRect(-17.5,5,3,4); }
    // headlights
    ctx.fillStyle='#fff6d8'; ctx.fillRect(15,-8,2.5,4); ctx.fillRect(15,4,2.5,4);
    ctx.restore();

    // compass chevron toward goal
    const dgx=G.goal[0]-car.x, dgy=G.goal[1]-car.y, dd=Math.hypot(dgx,dgy);
    if(dd>190){
      const ang=Math.atan2(dgy,dgx), pulse=.55+.3*Math.sin(t*6);
      ctx.save(); ctx.translate(car.x+Math.cos(ang)*44,car.y+Math.sin(ang)*44); ctx.rotate(ang);
      ctx.globalAlpha=pulse; ctx.fillStyle='#ffd23f';
      ctx.beginPath();ctx.moveTo(9,0);ctx.lineTo(-5,-6);ctx.lineTo(-1,0);ctx.lineTo(-5,6);ctx.closePath();ctx.fill();
      ctx.restore(); ctx.globalAlpha=1;
    }
  }

  // sparks
  for(const p of G.sparks){
    const a=p.life/p.max;
    ctx.strokeStyle='rgba(255,'+(180+((p.x*.7)|0)%60)+',40,'+a+')';
    ctx.lineWidth=2;
    ctx.beginPath();ctx.moveTo(p.x,p.y);ctx.lineTo(p.x-p.vx*.03,p.y-p.vy*.03);ctx.stroke();
  }
  // confetti
  for(const p of G.confetti){
    const a=clamp(p.life/p.max,0,1);
    ctx.save();ctx.translate(p.x,p.y);ctx.rotate(p.rot);ctx.globalAlpha=a;
    ctx.fillStyle=p.c;ctx.fillRect(-p.s/2,-p.s/3,p.s,p.s*.66);ctx.restore();
  }
  ctx.globalAlpha=1;
  ctx.restore();

  // countdown numerals (screen space)
  if(G.state==='count'){
    const ct=G.countT;
    let label, col, lifeIn;
    if(ct>0){label=String(Math.ceil(ct));col=['#ffd23f','#ffb020','#ff5d3b'][Math.ceil(ct)-1]||'#ffd23f';lifeIn=ct-Math.floor(ct);}
    else{label='GO!';col='#5dff9f';lifeIn=Math.max(0,1+ct/.7);}

function roundRect(x,y,w,h,r){
  ctx.beginPath();
  ctx.moveTo(x+r,y);ctx.arcTo(x+w,y,x+w,y+h,r);ctx.arcTo(x+w,y+h,x,y+h,r);
  ctx.arcTo(x,y+h,x,y,r);ctx.arcTo(x,y,x+w,y,r);ctx.closePath();
}

function drawMinimap(){
  const mw=mmC.width/DPR, mh=mmC.height/DPR;
  mmX.setTransform(DPR,0,0,DPR,0,0);
  mmX.clearRect(0,0,mw,mh);
  if(!G.tiles)return;
  const sc=Math.min((mw-8)/G.worldW,(mh-8)/G.worldH);
  const ox=(mw-G.worldW*sc)/2, oy=(mh-G.worldH*sc)/2;
  mmX.strokeStyle='#3d6076';mmX.lineWidth=1;
  mmX.beginPath();
  const m=G.maze;
  for(let cy=0;cy<m.rows;cy++)for(let cx=0;cx<m.cols;cx++){
    const w=m.cells[cy*m.cols+cx], x=ox+(cx*P+1)*T*sc, y=oy+(cy*P+1)*T*sc, L=C*T*sc;
    if(w.t){mmX.moveTo(x,y);mmX.lineTo(x+L,y);}
    if(w.l){mmX.moveTo(x,y);mmX.lineTo(x,y+L);}
    if(cx===m.cols-1&&w.r){mmX.moveTo(x+L,y);mmX.lineTo(x+L,y+L);}
    if(cy===m.rows-1&&w.b){mmX.moveTo(x,y+L);mmX.lineTo(x+L,y+L);}
  }
  mmX.stroke();
  // goal
  const [gx,gy]=G.goal;
  mmX.fillStyle='#ff5d9e';
  const gp=2+Math.sin(G.time*4);
  mmX.fillRect(ox+gx*sc-gp/2-1,oy+gy*sc-gp/2-1,gp+2,gp+2);
  // player
  const car=G.car;
  mmX.fillStyle='#ff7a1c';
  mmX.beginPath();mmX.arc(ox+car.x*sc,oy+car.y*sc,3,0,TAU);mmX.fill();
  mmX.strokeStyle='#ff7a1c';
  mmX.beginPath();mmX.moveTo(ox+car.x*sc,oy+car.y*sc);
  mmX.lineTo(ox+(car.x+Math.cos(car.a)*26)*sc,oy+(car.y+Math.sin(car.a)*26)*sc);mmX.stroke();
