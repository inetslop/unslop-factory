function tone(f,dur,type,vol,when,slideTo){
  if(!actx||muted)return;
  type=type||'square';vol=vol||0.1;when=when||0;
  const t0=actx.currentTime+when,o=actx.createOscillator(),g=actx.createGain();
  o.type=type;o.frequency.setValueAtTime(f,t0);
  if(slideTo)o.frequency.exponentialRampToValueAtTime(slideTo,t0+dur);
  g.gain.setValueAtTime(vol,t0);g.gain.exponentialRampToValueAtTime(0.0001,t0+dur);
  o.connect(g);g.connect(actx.destination);o.start(t0);o.stop(t0+dur+0.03);
}

function rotCW(m){const n=m.length,r=Array.from({length:n},()=>new Array(n).fill(0));
  for(let y=0;y<n;y++)for(let x=0;x<n;x++)r[x][n-1-y]=m[y][x];return r;}
function rotCCW(m){const n=m.length,r=Array.from({length:n},()=>new Array(n).fill(0));
  for(let y=0;y<n;y++)for(let x=0;x<n;x++)r[n-1-x][y]=m[y][x];return r;}
function collide(m,px,py){
  for(let y=0;y<m.length;y++)for(let x=0;x<m.length;x++){
    if(!m[y][x])continue;const bx=px+x,by=py+y;
    if(bx<0||bx>=COLS||by<0||by>=ROWS)return true;
    if(board[by][bx])return true;}
  return false;
}
function pullBag(){if(!bag.length){bag=[1,2,3,4,5,6,7];
  for(let i=bag.length-1;i>0;i--){const j=(Math.random()*(i+1))|0;[bag[i],bag[j]]=[bag[j],bag[i]];}}
  return bag.pop();}

function makePiece(type){
  const m=SHAPES[type].map(r=>r.slice());
  const x=Math.floor((COLS-m.length)/2), y=0;
  if(collide(m,x,y)){gameOver();return;}
  cur={type,m,x,y,rx:x,ry:y-1.6,moves:0};
  holdUsed=false;
  updateHUD();renderMinis();
  if(piecesPlaced===0)floaters.push({text:'PRESS SPACE TO LOCK',size:14,color:'#8ea3cd',
    x:cell*COLS/2,y:cell*4.6,t:0,life:1.7});
}

function tryMove(dx,dy){
  if(state!=='playing'||!cur)return;
  if(!collide(cur.m,cur.x+dx,cur.y+dy)){cur.x+=dx;cur.y+=dy;cur.moves++;sfx.move();updateHUD();}
}
function rotate(dir){
  if(state!=='playing'||!cur)return;
  const nm=dir>0?rotCW(cur.m):rotCCW(cur.m);
  for(const k of [0,-1,1,-2,2]){
    if(!collide(nm,cur.x+k,cur.y)){cur.m=nm;cur.x+=k;cur.moves++;sfx.rotate();updateHUD();return;}
  }
}
function doHold(){
  if(state!=='playing'||!cur||holdUsed)return;
  sfx.hold();
  const t=cur.type;
  if(held){const h=held;held=t;makePiece(h);}
  else{held=t;cur=null;makePiece(queue.shift());queue.push(pullBag());}
  holdUsed=true;renderMinis();updateHUD();
}

function doLock(){
  if(state!=='playing'||!cur)return;
  let topOut=true;
  for(let y=0;y<cur.m.length;y++)for(let x=0;x<cur.m.length;x++){
    if(!cur.m[y][x])continue;
    const bx=cur.x+x,by=cur.y+y;
    board[by][bx]=cur.type;
    if(by>0)topOut=false;
    lockPops[bx+','+by]={t:0};
  }
  piecesPlaced++;
  sfx.lock();shakeMag=Math.max(shakeMag,1.6);
  if(topOut){gameOver();return;}

  const cleared=[];for(let r=0;r<ROWS;r++)if(board[r].every(v=>v))cleared.push(r);
  let gained=0,msgs=[];
  const mult=1+Math.floor(lines/10)*0.5;
  if(cleared.length){
    const n=cleared.length;streak++;maxStreak=Math.max(maxStreak,streak);
    const pts=[0,100,300,500,800][n]*mult;gained+=pts;
    msgs.push({text:LINE_NAMES[n]+' +'+Math.round(pts),
      size:n===4?36:n===3?29:n===2?24:19,
      color:n===4?'#ffd23f':n>=2?'#ffc857':'#9fd8ff'});
    if(streak>1){const cp=50*(streak-1);gained+=cp;
      msgs.push({text:'CHAIN ×'+streak+' +'+cp,size:16,color:'#4fd8ff'});}
    lines+=n;
  }else streak=0;

  const sw=Math.max(0,(PAR-cur.moves))*SWIFT_STEP;
  if(sw>0){gained+=sw;swiftLocks++;msgs.push({text:'SWIFT +'+sw,size:15,color:'#57e07a'});}
  score+=gained;
  msgs.forEach((m,i)=>floaters.push(Object.assign(m,{x:cell*COLS/2+(Math.random()*44-22),
    y:cell*(3.1+i*1.2),t:0,life:1.15})));

  if(gained>0)bump(scoreEl);
  if(cleared.length&&streak>1)bump(chainEl);

  cur=null;
  if(cleared.length){
    state='clearing';animBoard=board.map(r=>r.slice());clearRows=cleared;clearT=0;
    flashA=cleared.length===4?0.5:cleared.length>=3?0.32:0.16;
    spawnParticles(cleared);sfx.clear(cleared.length);
    if(cleared.length===4)shakeMag=Math.max(shakeMag,6);
  }else{
    makePiece(queue.shift());queue.push(pullBag());
  }
  updateHUD();
}

function collapseBoard(){
  const keep=board.filter((r,i)=>clearRows.indexOf(i)<0);
  while(keep.length<ROWS)keep.unshift(new Array(COLS).fill(0));
  board=keep;animBoard=null;clearRows=[];
}
function spawnParticles(rows){
  rows.forEach(r=>{for(let c=0;c<COLS;c++){
    const ci=animBoard[r][c];if(!ci)continue;
    for(let i=0;i<3;i++)particles.push({x:(c+0.5)*cell,y:(r+0.5)*cell,
      vx:(Math.random()-0.5)*270,vy:-60-Math.random()*190,
      s:cell*(0.12+Math.random()*0.15),life:0.5+Math.random()*0.4,max:0,color:COLORS[ci]});
  }});
  particles.forEach(p=>p.max=p.life);
  if(particles.length>420)particles.splice(0,particles.length-420);
}

  setTimeout(()=>{if(rid!==runId)return;

    $('osChain').textContent='×'+maxStreak;$('osSwift').textContent=swiftLocks;

    show($('overOv'));
  },700);
  updateHUD();
}

function roundRect(g,x,y,w,h,r){r=Math.min(r,w/2,h/2);g.beginPath();
  g.moveTo(x+r,y);g.arcTo(x+w,y,x+w,y+h,r);g.arcTo(x+w,y+h,x,y+h,r);
  g.arcTo(x,y+h,x,y,r);g.arcTo(x,y,x+w,y,r);g.closePath();}

function drawBlock(g,px,py,s,ci,alpha,glow){
  alpha=alpha==null?1:alpha;glow=glow||0;
  g.save();g.globalAlpha=alpha;
  if(glow>0){g.shadowColor=COLORS[ci];g.shadowBlur=glow;}
  roundRect(g,px+0.5,py+0.5,s-1,s-1,Math.max(2,s*0.16));
  g.fillStyle=COLORS[ci];g.fill();
  g.shadowBlur=0;
  const lw=Math.max(1,s*0.1);
  g.globalAlpha=alpha*0.5;g.strokeStyle='rgba(255,255,255,.85)';g.lineWidth=lw;
  g.beginPath();g.moveTo(px+2.5,py+s-3);g.lineTo(px+2.5,py+2.5);g.lineTo(px+s-3,py+2.5);g.stroke();
  g.globalAlpha=alpha*0.45;g.strokeStyle='rgba(0,0,0,.6)';
  g.beginPath();g.moveTo(px+s-2.5,py+3);g.lineTo(px+s-2.5,py+s-2.5);g.lineTo(px+3,py+s-2.5);g.stroke();
  g.globalAlpha=alpha*0.4;g.fillStyle='#fff';
  roundRect(g,px+s*0.18,py+s*0.16,s*0.3,s*0.13,s*0.06);g.fill();
  g.restore();
}

function drawWell(g){
  const W=cell*COLS,H=cell*ROWS;
  g.fillStyle='#0a1122';roundRect(g,0,0,W,H,8);g.fill();
  g.strokeStyle='rgba(130,165,255,.07)';g.lineWidth=1;
  for(let x=1;x<COLS;x++){g.beginPath();g.moveTo(x*cell+0.5,4);g.lineTo(x*cell+0.5,H-4);g.stroke();}
  for(let y=1;y<ROWS;y++){g.beginPath();g.moveTo(4,y*cell+0.5);g.lineTo(W-4,y*cell+0.5);g.stroke();}
  const gr=g.createLinearGradient(0,0,0,cell*3);
  gr.addColorStop(0,'rgba(0,0,0,.5)');gr.addColorStop(1,'rgba(0,0,0,0)');
  g.fillStyle=gr;g.fillRect(2,2,W-4,cell*3);
  g.strokeStyle='rgba(96,138,230,.4)';g.lineWidth=2;roundRect(g,1,1,W-2,H-2,8);g.stroke();

function drawStack(){
  for(let y=0;y<ROWS;y++)for(let x=0;x<COLS;x++){
    const v=board[y][x];if(!v)continue;
    const p=lockPops[x+','+y];
    let s=cell,ox=x*cell,oy=y*cell;
    if(p){const e=Math.sin(Math.PI*p.t);s=cell*(1+0.25*e);ox-=(s-cell)/2;oy-=(s-cell)/2;}
    drawBlock(bctx,ox,oy,s,v,1,p?4:0);
    if(p&&p.t<1){bctx.globalAlpha=(1-p.t)*0.55;bctx.fillStyle='#fff';
      roundRect(bctx,ox+1,oy+1,s-2,s-2,s*0.16);bctx.fill();bctx.globalAlpha=1;}
  }
}
function drawClearing(){
  for(let y=0;y<ROWS;y++)for(let x=0;x<COLS;x++){const v=animBoard[y][x];if(v)drawBlock(bctx,x*cell,y*cell,cell,v,1);}
  const sweep=Math.floor(clearT*(COLS+2));
  for(const r of clearRows){
    for(let c=0;c<sweep&&c<COLS;c++){
      bctx.fillStyle='rgba(255,255,255,'+(0.85-clearT*0.2)+')';
      bctx.fillRect(c*cell+1,r*cell+1,cell-2,cell-2);}
    if(sweep<COLS){bctx.fillStyle='rgba(255,255,255,.9)';bctx.fillRect(sweep*cell,r*cell,3,cell);}
  }
}
function drawActive(){
  const grounded=collide(cur.m,cur.x,cur.y+1);
  let gy=cur.y;while(!collide(cur.m,cur.x,gy+1))gy++;
  if(gy>cur.y){
    bctx.save();
    for(let y=0;y<cur.m.length;y++)for(let x=0;x<cur.m.length;x++){
      if(!cur.m[y][x])continue;
      const gx=(cur.x+x)*cell,gyy=(gy+y)*cell;
      roundRect(bctx,gx+1.5,gyy+1.5,cell-3,cell-3,Math.max(2,cell*0.14));
      bctx.fillStyle=rgba(cur.type,0.1);bctx.fill();
      bctx.strokeStyle=rgba(cur.type,0.45);bctx.lineWidth=2;bctx.stroke();
    }
    bctx.restore();
  }else if(grounded){
    const low={};
    for(let y=0;y<cur.m.length;y++)for(let x=0;x<cur.m.length;x++){
      if(cur.m[y][x])low[cur.x+x]=Math.max(low[cur.x+x]||-1,y);}
    const a=0.35+0.35*Math.sin(gt*7);
    bctx.fillStyle='rgba(255,200,87,'+a+')';
    for(const bx in low){roundRect(bctx,(bx+0)*cell+2,(low[bx]+1)*cell-4,cell-4,3,1.5);bctx.fill();}
  }
  const px=cur.rx*cell,py=cur.ry*cell;
  for(let y=0;y<cur.m.length;y++)for(let x=0;x<cur.m.length;x++){
    if(!cur.m[y][x])continue;
    drawBlock(bctx,px+x*cell,py+y*cell,cell,cur.type,1,6+3*Math.sin(gt*4));
  }
}
function drawDanger(){

  if(top>=0&&top<=5&&state!=='menu'&&state!=='over'){
    const a=0.14+0.1*Math.sin(gt*5);
    const gr=bctx.createLinearGradient(0,0,0,cell*5);
    gr.addColorStop(0,'rgba(255,70,90,'+a+')');gr.addColorStop(1,'rgba(255,70,90,0)');

  }
}

function draw(){
  const w=boardC.clientWidth,h=boardC.clientHeight;
  bctx.setTransform(dpr,0,0,dpr,0,0);
  bctx.clearRect(0,0,w,h);
  const ox=(w-cell*COLS)/2,oy=(h-cell*ROWS)/2;
  const shx=(Math.random()*2-1)*shakeMag,shy=(Math.random()*2-1)*shakeMag;
  bctx.save();bctx.translate(ox+shx,oy+shy);
  drawWell(bctx);
  if(state==='clearing')drawClearing();else drawStack();
  if(cur&&state==='playing')drawActive();
  drawDanger();
  if(flashA>0.01){bctx.fillStyle='rgba(255,255,255,'+flashA+')';bctx.fillRect(0,0,cell*COLS,cell*ROWS);}
  for(const p of particles){bctx.globalAlpha=Math.max(0,p.life/p.max);bctx.fillStyle=p.color;
    bctx.fillRect(p.x-p.s/2,p.y-p.s/2,p.s,p.s);}
  bctx.globalAlpha=1;
  for(const f of floaters){
    const k=f.t/f.life;
    const alpha=k<0.7?1:Math.max(0,1-(k-0.7)/0.3);

    bctx.strokeText(f.text,fx,fy);bctx.fillStyle=f.color;bctx.fillText(f.text,fx,fy);
  }
  bctx.globalAlpha=1;bctx.restore();
}

function frame(t){
  const dt=Math.min(0.033,(t-last)/1000);last=t;
  if(state!=='paused'){
    gt+=dt;
    if(cur){cur.rx+=(cur.x-cur.rx)*Math.min(1,dt*16);cur.ry+=(cur.y-cur.ry)*Math.min(1,dt*16);}
    for(const k in lockPops){lockPops[k].t+=dt*4.2;if(lockPops[k].t>=1)delete lockPops[k];}
    if(state==='clearing'){
      clearT+=dt/CLEAR_DUR;
      if(clearT>=1){collapseBoard();state='playing';makePiece(queue.shift());queue.push(pullBag());updateHUD();}
    }
  }
  for(let i=particles.length-1;i>=0;i--){const p=particles[i];
    p.life-=dt;p.vy+=520*dt;p.x+=p.vx*dt;p.y+=p.vy*dt;
    if(p.life<=0)particles.splice(i,1);}
  for(let i=floaters.length-1;i>=0;i--){floaters[i].t+=dt;if(floaters[i].t>floaters[i].life)floaters.splice(i,1);}
  shakeMag*=Math.exp(-6*dt);if(shakeMag<0.05)shakeMag=0;
  flashA*=Math.exp(-6*dt);
  dispScore+=(score-dispScore)*Math.min(1,dt*9);
  if(Math.abs(score-dispScore)<0.6)dispScore=score;

  const ds=Math.round(dispScore);
  if(ds!==lastDs){lastDs=ds;scoreEl.textContent=fmt(ds);}
  const b=Math.max(best,score);
  if(b!==lastBest){lastBest=b;bestEl.textContent=fmt(b);
    bestEl.classList.toggle('hot',score>0&&score>best&&state!=='menu'&&state!=='over');}

  drawBg(state==='paused'?0:dt);
  draw();
  requestAnimationFrame(frame);
}
