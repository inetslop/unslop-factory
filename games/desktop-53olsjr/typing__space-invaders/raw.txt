function tick(dt){
  G.time+=dt;
  // ambient stars always
  for(const s of stars){ s.y+=s.z*14*dt; if(s.y>H+2){ s.y=-2; s.x=Math.random()*W; } }
  if(G.hitstop>0){ G.hitstop-=dt; return; }

  const fx=(st)=>G.state===st||G.state==='paused';
  // fx entities (particles/beams/rings/texts) run in play, over & menu
  if(fx('playing')||G.state==='over'||G.state==='menu'){
    for(let i=G.parts.length-1;i>=0;i--){ const p=G.parts[i]; p.t+=dt;
      if(p.t>p.dur){ G.parts.splice(i,1); continue; }
      p.x+=p.vx*dt; p.y+=p.vy*dt; p.vx-=p.vx*2.4*dt; p.vy-=p.vy*2.4*dt; }
    for(let i=G.beams.length-1;i>=0;i--){ const b=G.beams[i]; b.t+=dt; if(b.t>b.dur) G.beams.splice(i,1); }
    for(let i=G.rings.length-1;i>=0;i--){ const r=G.rings[i]; r.t+=dt; r.r+=r.vr*dt; if(r.t>r.dur) G.rings.splice(i,1); }
    for(let i=G.texts.length-1;i>=0;i--){ const t=G.texts[i]; t.t+=dt; if(t.t>t.dur) G.texts.splice(i,1); }
  }
  G.shake=Math.max(0,G.shake-dt*34);
  G.flashA=Math.max(0,G.flashA-dt*1.5);
  G.recoil*=Math.exp(-dt*10);

  if(G.state==='menu'){
    for(const e of G.enemies){ e.y+=e.vy*dt;
      e.x=e.baseX+Math.sin(G.time*e.swaySpd+e.phase)*e.swayAmp;
      if(e.y>H+90){ e.y=-80-Math.random()*60; e.baseX=90+Math.random()*Math.max(40,W-180);
        G.used.delete(e.word); const w=pickWord(false); G.used.add(w);
        e.word=w; e.type=w.length>=7?2:w.length>=5?1:0; e.rows=SPR[e.type*2];
        e.popT=new Array(w.length).fill(-9); } }
    return;
  }
  if(G.state!=='playing') return;

  G.runT+=dt;
  // chain timer
  if(G.chain>0){ G.chainT-=dt; if(G.chainT<=0) G.chain=0; }
  // spawning
  const cap=Math.min(7,[3,4,5][G.diff]+Math.floor(G.level/3));
  const iv=Math.max(.95, DIFFS[G.diff].spawn/(1+(G.level-1)*.14));
  G.spawnT-=dt;
  if(G.spawnT<=0){
    if(G.bossPending && !G.enemies.some(e=>e.boss)){ makeEnemy(true); G.bossPending=false; G.spawnT=iv*1.8; }
    else if(G.enemies.length<cap){ makeEnemy(false); G.spawnT=iv*(.75+Math.random()*.5); }
    else G.spawnT=.3;
  }
  // enemies
  const shieldY=H-84;
  for(let i=G.enemies.length-1;i>=0;i--){
    const e=G.enemies[i];
    e.y+=e.vy*dt;
    e.x=e.baseX+Math.sin(G.time*e.swaySpd+e.phase)*e.swayAmp;
    const half=Math.min(W*0.36,380)/2+14;
    e.x=Math.max(half,Math.min(W-half,e.x));
    if(e.y+e.spriteH/2>=shieldY) damage(e);
  }
  // hud throttle
  G.hudAcc+=dt;
  if(G.hudAcc>.15){ G.hudAcc=0; updateHUD(false); }
}

/* ================= RENDER ================= */
function drawSprite(rows,cx,cy,cell,color,alpha){
  const w=rows[0].length,h=rows.length,x0=cx-w*cell/2,y0=cy-h*cell/2;
  ctx.globalAlpha=alpha; ctx.fillStyle=color;
  for(let r=0;r<h;r++){ const row=rows[r];
    for(let c=0;c<w;c++) if(row[c]==='X') ctx.fillRect(x0+c*cell,y0+r*cell,cell,cell); }
  ctx.globalAlpha=1;
}
function halo(x,y,r,color,a){
  const g=ctx.createRadialGradient(x,y,0,x,y,r);
  g.addColorStop(0,color.replace(')',','+a+')').replace('rgb','rgba'));
  g.addColorStop(1,'rgba(0,0,0,0)');
  ctx.fillStyle=g; ctx.fillRect(x-r,y-r,r*2,r*2);
}
function render(){
  ctx.clearRect(0,0,W,H);
  if(neb) ctx.drawImage(neb,0,0,W,H);
  // stars
  for(const s of stars){
    const a=(.35+.4*s.z)*(0.7+0.3*Math.sin(G.time*s.tw+s.ph));
    ctx.globalAlpha=a; ctx.fillStyle=s.z>.8?'#cfeaff':'#7fa8cc';
    ctx.fillRect(s.x,s.y,s.sz,s.sz);
  }
  ctx.globalAlpha=1;

  ctx.save();
  if(G.shake>0) ctx.translate((Math.random()-.5)*G.shake,(Math.random()-.5)*G.shake);

  const shieldY=H-84, inGame=G.state==='playing'||G.state==='paused';
  // horizon glow + shield line (only during play/over/pause)
  if(inGame||G.state==='over'){
    const g=ctx.createLinearGradient(0,H,0,shieldY-46);
    g.addColorStop(0,'rgba(53,224,255,.10)'); g.addColorStop(1,'rgba(53,224,255,0)');
    ctx.fillStyle=g; ctx.fillRect(0,shieldY-46,W,46);
    const base=.14+.22*(G.lives/3);
    for(let x=8;x<W-26;x+=36){
      const p=base+ .05*Math.sin(G.time*2+x*.05);
      ctx.fillStyle='rgba(53,224,255,'+p.toFixed(3)+')';
      ctx.fillRect(x,shieldY,24,3);
    }
  }
  // targeting line + reticle
  const act = inGame? activeEnemy():null;
  if(act){
    ctx.save();
    ctx.setLineDash([4,8]); ctx.lineDashOffset=-G.time*40;
    ctx.strokeStyle='rgba(125,249,255,'+(0.16+0.07*Math.sin(G.time*6)).toFixed(3)+')';
    ctx.lineWidth=1.5;
    ctx.beginPath(); ctx.moveTo(W/2,shieldY-30); ctx.lineTo(act.x,act.y); ctx.stroke();
    ctx.restore();
  }
  // enemies
  for(const e of G.enemies){
    const col=e.boss?BOSS_COLOR:TYPE_COLORS[e.type];
    const frame=Math.floor(G.time*4+e.phase)%2;
    const rows=(e.boss?BOSS_SPR:e.rows);
    halo(e.x,e.y,e.spriteW*.85,'rgb(120,200,255)',.13);
    drawSprite(rows[frame],e.x,e.y,e.cell,col,.96);
    // word
    const size=e.wordSize;
    ctx.font='700 '+size+'px '+MONOFONT;
    const cw=ctx.measureText('M').width, total=cw*e.word.length;
    const wy=e.y+e.spriteH/2+8+size*.55;
    let x=e.x-total/2;
    ctx.textAlign='center'; ctx.textBaseline='middle';
    for(let i=0;i<e.word.length;i++){
      let colr,sc=1,dy=0;
      if(e.demo){ colr='rgba(150,185,215,.34)'; }
      else if(i<e.prog){
        colr='#63ff9c';
        const age=G.time-e.popT[i];
        if(age<.18) sc=1+(1-age/.18)*.45;
      } else if(act===e && i===e.prog){
        const pul=.5+.5*Math.sin(G.time*7);
        colr='rgba(200,255,255,'+(0.6+0.4*pul).toFixed(3)+')';
        dy=Math.sin(G.time*7)*1.6;
        if(G.time-e.errT<.18) colr='#ff5d6c';
      } else colr = act? 'rgba(226,240,255,.92)' : 'rgba(160,190,220,.4)';
      ctx.save();
      ctx.translate(x+cw/2,wy+dy); ctx.scale(sc,sc);
      ctx.fillStyle=colr; ctx.fillText(e.word[i],0,0);
      ctx.restore();
      x+=cw;
    }
    // reticle brackets on active
    if(act===e){
      const bx=e.x-total/2-9, by=wy-size*.72, bw=total+18, bh=size*1.45, L=11;
      ctx.strokeStyle='rgba(125,249,255,'+(0.55+0.3*Math.sin(G.time*6)).toFixed(3)+')';
      ctx.lineWidth=2; ctx.beginPath();
      ctx.moveTo(bx,by+L);ctx.lineTo(bx,by);ctx.lineTo(bx+L,by);
      ctx.moveTo(bx+bw-L,by);ctx.lineTo(bx+bw,by);ctx.lineTo(bx+bw,by+L);
      ctx.moveTo(bx+bw,by+bh-L);ctx.lineTo(bx+bw,by+bh);ctx.lineTo(bx+bw-L,by+bh);
      ctx.moveTo(bx+L,by+bh);ctx.lineTo(bx,by+bh);ctx.lineTo(bx,by+bh-L);
      ctx.stroke();
    }
  }
  // cannon
  if(inGame||G.state==='over'){
    const cx=W/2, cy=shieldY-16+G.recoil*5;
    halo(cx,cy,40,'rgb(90,220,255)',.16);
    drawSprite(CANNON,cx,cy,4.5,'#8ff4ff',1);
    // engine flicker
    for(let i=0;i<3;i++){
      const len=3+Math.random()*8;
      ctx.fillStyle=i===0?'rgba(255,181,77,.9)':'rgba(255,224,138,'+(.7-i*.2)+')';
      ctx.fillRect(cx-1.5,cy+11+i*3,len*.6,2);
    }
  }
  // additive fx: beams / rings / particles
  ctx.save(); ctx.globalCompositeOperation='lighter';
  for(const b of G.beams){
    const k=b.t/b.dur, a=1-k;
    ctx.strokeStyle=b.color; ctx.globalAlpha=a*.35; ctx.lineWidth=10*(1-k)+2;
    ctx.beginPath(); ctx.moveTo(b.x1,b.y1); ctx.lineTo(b.x2,b.y2); ctx.stroke();
    ctx.strokeStyle='#ffffff'; ctx.globalAlpha=a; ctx.lineWidth=2.4;
    ctx.beginPath(); ctx.moveTo(b.x1,b.y1); ctx.lineTo(b.x2,b.y2); ctx.stroke();
  }
  ctx.globalAlpha=1;
  for(const r of G.rings){
    const k=r.t/r.dur;
    ctx.strokeStyle=r.color; ctx.globalAlpha=(1-k)*.7; ctx.lineWidth=2;
    ctx.beginPath(); ctx.arc(r.x,r.y,r.r,0,TAU); ctx.stroke();
  }
  for(const p of G.parts){
    const k=p.t/p.dur;
    ctx.globalAlpha=1-k; ctx.fillStyle=p.color;
    ctx.fillRect(p.x-p.sz/2,p.y-p.sz/2,p.sz,p.sz);
  }
  ctx.restore(); ctx.globalAlpha=1;
  // floating score text
  for(const t of G.texts){
    const k=t.t/t.dur;
    ctx.globalAlpha=1-k; ctx.fillStyle=t.color;
    ctx.font='700 '+t.size+'px '+MONOFONT;
    ctx.textAlign='center'; ctx.textBaseline='middle';
    ctx.fillText(t.txt,t.x,t.y-k*46);
  }
  ctx.globalAlpha=1;
  ctx.restore(); // shake

  if(G.flashA>0){
    ctx.fillStyle='rgba('+G.flashC+','+G.flashA.toFixed(3)+')';
    ctx.fillRect(0,0,W,H);
  }
}

const trio=[{c:$('inv0'),t:0},{c:$('inv1'),t:1},{c:$('inv2'),t:2}];
function drawTrio(){
  if(G.state!=='menu') return;
  for(const m of trio){
    const c=m.c, x=c.getContext('2d');
    x.clearRect(0,0,c.width,c.height);
    const frame=Math.floor(G.time*3+m.t)%2;
    const bob=Math.sin(G.time*2+m.t*2)*3;
    drawSpriteCtx(x,SPR[m.t*2+frame],c.width/2,c.height/2+bob,4.6,TYPE_COLORS[m.t]);
  }
}
function drawSpriteCtx(c2,rows,cx,cy,cell,color){
  const w=rows[0].length,h=rows.length,x0=cx-w*cell/2,y0=cy-h*cell/2;
  c2.fillStyle=color;
  for(let r=0;r<h;r++){ const row=rows[r];
    for(let cc=0;cc<w;cc++) if(row[cc]==='X') c2.fillRect(x0+cc*cell,y0+r*cell,cell,cell); }
}

}
