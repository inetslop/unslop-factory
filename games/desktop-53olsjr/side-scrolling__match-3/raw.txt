/* ============ AUDIO (WebAudio, post-gesture) ============ */
let AC=null, master=null;
function ensureAudio(){
  if(AC){ if(AC.state==='suspended') AC.resume(); return; }
  try{
    const A=window.AudioContext||window.webkitAudioContext; if(!A) return;
    AC=new A(); master=AC.createGain(); master.gain.value=0.45; master.connect(AC.destination);
  }catch(e){ AC=null; }
}
function beep(f0,f1,dur,type,vol,delay){
  if(!AC||muted) return; delay=delay||0;
  const t0=AC.currentTime+delay, o=AC.createOscillator(), g=AC.createGain();
  o.type=type; o.frequency.setValueAtTime(f0,t0);
  if(f1) o.frequency.exponentialRampToValueAtTime(Math.max(30,f1), t0+dur);
  g.gain.setValueAtTime(0.0001,t0);
  g.gain.exponentialRampToValueAtTime(vol, t0+0.014);
  g.gain.exponentialRampToValueAtTime(0.0001, t0+dur);
  o.connect(g); g.connect(master); o.start(t0); o.stop(t0+dur+0.06);
}
function noise(dur,vol,freq,delay){
  if(!AC||muted) return; delay=delay||0;
  const t0=AC.currentTime+delay, n=Math.max(1,Math.floor(AC.sampleRate*dur));
  const b=AC.createBuffer(1,n,AC.sampleRate), d=b.getChannelData(0);
  for(let i=0;i<n;i++) d[i]=(Math.random()*2-1)*(1-i/n);
  const s=AC.createBufferSource(); s.buffer=b;
  const f=AC.createBiquadFilter(); f.type='lowpass'; f.frequency.value=freq;
  const g=AC.createGain(); g.gain.value=vol;
  s.connect(f); f.connect(g); g.connect(master); s.start(t0);
}
const sfx={

  match:(ch)=>{ const b=392*Math.pow(1.122,Math.min(ch,8)); [0,4,7].forEach((st,i)=>beep(b*Math.pow(2,st/12),0,.14,'triangle',.2,i*.055)); noise(.09,.05,3200); },
  furnace:()=>{ noise(.24,.22,240); beep(72,44,.3,'sine',.3); },
  speedup:()=>beep(220,460,.18,'square',.14),
  overOn:()=>beep(520,820,.22,'triangle',.18),
  overOff:()=>beep(600,390,.18,'triangle',.1),
  overload:()=>{ noise(.7,.4,320); beep(220,38,.95,'sawtooth',.34); beep(110,28,.95,'square',.24,.06); },
  start:()=>{ [0,4,7,12].forEach((st,i)=>beep(261*Math.pow(2,st/12),0,.16,'triangle',.2,i*.07)); }

/* ============ WORLD ============ */
function randT(){ return (Math.random()*TYPES.length)|0; }
function newGem(t){ return {t:t, dx:0, dy:0, pop:1, uid:++UID}; }
function makeColumn(avoid){
  const col=[];
  for(let r=0;r<ROWS;r++){ let t; let guard=0;
    do{ t=randT(); guard++; }
    while(guard<24 && grid[L-2]&&grid[L-1] && grid[L-2][r]&&grid[L-1][r]
          && grid[L-2][r].t===t && grid[L-1][r].t===t);
    col.push(newGem(t)); }
  return col;
}
function makeGrid(){
  grid=[];
  for(let c=0;c<L;c++){ const col=[];
    for(let r=0;r<ROWS;r++){ let t; let guard=0;
      do{ t=randT(); guard++; }
      while(guard<24 && (
        (c>=2 && grid[c-1][r]&&grid[c-2][r] && grid[c-1][r].t===t && grid[c-2][r].t===t) ||
        (r>0 && col[r-1] && col[r-1].t===t && c>=1 && grid[c-1][r]&&grid[c-1][r-1] && grid[c-1][r].t===t && grid[c-1][r-1].t===t)
      ));
      col.push(newGem(t)); }
    grid.push(col); }
}

}

/* ============ MATCH LOGIC ============ */
function findMatches(){
  const cells=[], seen=new Set(), runs=[];
  function add(c,r){ const k=c*32+r; if(!seen.has(k)){ seen.add(k); cells.push([c,r]); } }
  for(let r=0;r<ROWS;r++){ let c=0; while(c<L){ const g=grid[c][r]; if(!g){c++;continue;}
    let e=c+1; while(e<L && grid[e][r] && grid[e][r].t===g.t) e++;
    if(e-c>=3){ runs.push({len:e-c}); for(let i=c;i<e;i++) add(i,r); } c=e; } }
  for(let c=0;c<L;c++){ let r=0; while(r<ROWS){ const g=grid[c][r]; if(!g){r++;continue;}
    let e=r+1; while(e<ROWS && grid[c][e] && grid[c][e].t===g.t) e++;
    if(e-r>=3){ runs.push({len:e-r}); for(let i=r;i<e;i++) add(c,i); } r=e; } }
  return {has:cells.length>0, cells:cells, runs:runs};
}
function applyGravity(){
  for(let c=0;c<L;c++){
    let write=ROWS-1;
    for(let r=ROWS-1;r>=0;r--){ const g=grid[c][r];
      if(g){ if(write!==r){ grid[c][write]=g; grid[c][r]=null; g.dy=(r-write)*cell; } write--; } }
    for(let r=write;r>=0;r--){ const g=newGem(randT()); g.pop=0; g.dy=-(write+1-r)*cell; grid[c][r]=g; }
  }
}
function beginResolve(){ if(resolving) return; resolving=true; chain=0; resolveStep(); }
function xOf(c){ return boardLeft + c*cell - scroll; }
function resolveStep(){
  const m=findMatches();
  if(!m.has){ resolving=false; chain=0; return; }
  chain++;
  let total=0, sx=0, sy=0, n=0;
  for(const cr of m.cells){ const c=cr[0], r=cr[1], g=grid[c][r]; if(!g) continue;
    grid[c][r]=null;
    const x=xOf(c)+cell/2, y=boardTop+r*cell+cell/2;
    burst(x,y,TYPES[g.t].fill, 7+Math.min(6,chain*2), 55+chain*18);
    total+=10*chain; sx+=x; sy+=y; n++;
  }
  for(const run of m.runs) if(run.len>=4) total+=(run.len-3)*20*chain;
  if(state==='play'){
    heat=Math.max(0, heat-m.cells.length*RELIEF);
    if(overdrive) total=Math.round(total*1.5);
    score+=total;
    const cx=sx/n, cy=sy/n;
    floats.push({x:cx,y:cy,txt:'+'+total,t:0,dur:.9,size:Math.min(26,14+chain*3),color:'#ffd98a'});
    if(chain>=2) floats.push({x:cx,y:cy-28,txt:'COMBO \u00D7'+chain,t:0,dur:1.05,size:Math.min(34,16+chain*4),color:chain>=4?'#ff8a3c':'#ffc93c'});
    sfx.match(chain);
    if(chain>=3){ shakeMag=Math.max(shakeMag, 3+chain*2); flashA=Math.max(flashA,.12); }
  }
  applyGravity();
  dropTimer=0.26+Math.min(.12, chain*.03);
}

/* ============ SWAPS ============ */
function inBounds(c,r){ return c>=0 && c<L && r>=0 && r<ROWS; }
function adjacent(a,b){ return Math.abs(a.c-b.c)+Math.abs(a.r-b.r)===1; }
function doSwap(c1,r1,c2,r2){
  if(!inBounds(c2,r2)) return;
  const A=grid[c1][r1], B=grid[c2][r2];
  if(!A||!B) return;
  grid[c1][r1]=B; grid[c2][r2]=A;
  A.dx=(c1-c2)*cell; A.dy=(r1-r2)*cell;
  B.dx=(c2-c1)*cell; B.dy=(r2-r1)*cell;
  lastSwap={c1:c1,r1:r1,c2:c2,r2:r2};
  swapBusy=true; swapReverting=false; swapT=0.16;
  if(state==='play') sfx.swap();
}
function evalSwap(){
  const m=findMatches();
  if(m.has){ beginResolve(); return; }
  const s=lastSwap, A=grid[s.c2][s.r2], B=grid[s.c1][s.r1];
  grid[s.c1][s.r1]=A; grid[s.c2][s.r2]=B;
  A.dx=(s.c2-s.c1)*cell; A.dy=(s.r2-s.r1)*cell;
  B.dx=(s.c1-s.c2)*cell; B.dy=(s.r1-s.r2)*cell;
  swapBusy=true; swapReverting=true; swapT=0.16;
  if(state==='play') sfx.deny();
}
function tryAutoSwap(){
  const opts=[];
  function test(c1,r1,c2,r2){
    const t=grid[c1][r1]; grid[c1][r1]=grid[c2][r2]; grid[c2][r2]=t;
    const ok=findMatches().has;
    t=grid[c1][r1]; grid[c1][r1]=grid[c2][r2]; grid[c2][r2]=t;
    if(ok) opts.push([c1,r1,c2,r2]);
  }
  for(let c=0;c<L-1;c++) for(let r=0;r<ROWS;r++) if(grid[c][r]&&grid[c+1][r]) test(c,r,c+1,r);
  for(let c=0;c<L;c++)   for(let r=0;r<ROWS-1;r++) if(grid[c][r]&&grid[c][r+1]) test(c,r,c,r+1);
  if(opts.length){ const o=opts[(Math.random()*opts.length)|0]; doSwap(o[0],o[1],o[2],o[3]); }
}

/* ============ CONVEYOR / SHIFT ============ */
function doShift(){
  const consumed=grid[0];
  grid.shift(); grid.push(makeColumn(true)); ticks++; chuteFlash=1;
  if(state==='play'){
    let n=0; for(let r=0;r<ROWS;r++) if(consumed[r]) n++;
    if(n){ heat=Math.min(HEAT_MAX, heat+n*H_PER_GEM); furnaceFlash=1; sfx.furnace(); shakeMag=Math.max(shakeMag,2.5); mouthBurst(); }
    if(ticks%8===0 && tickInterval>MIN_TICK+0.01){ tickInterval=Math.max(MIN_TICK, tickInterval*0.94); banner('BELT SPEED UP','#ff8a3c'); sfx.speedup(); }
    if(heat>=HEAT_MAX) gameOver();
  }
}

/* ============ FX ============ */
function burst(x,y,col,n,pow){
  if(parts.length>420) return;
  for(let i=0;i<n;i++){ const a=Math.random()*6.283, v=pow*(.4+Math.random()*.8);
    parts.push({x:x,y:y,vx:Math.cos(a)*v,vy:Math.sin(a)*v-30,t:0,dur:.5+Math.random()*.4,size:1.5+Math.random()*2.6,col:Math.random()<.25?'#fff3d8':col,g:300}); }
}
function mouthBurst(){
  const fx1=boardLeft+cell*0.2;
  for(let i=0;i<12;i++) parts.push({x:fx1-4,y:boardTop+Math.random()*boardH,vx:-(40+Math.random()*130),vy:(Math.random()-.5)*70,t:0,dur:.5,size:1.6+Math.random()*2,col:'#ff8a3c',g:0});
}
function banner(text,color){ bannerQ.push({text:text,color:color||'#ffd9b0'}); }
function updParts(dt){ for(let i=parts.length-1;i>=0;i--){ const p=parts[i]; p.t+=dt; if(p.t>p.dur){ parts.splice(i,1); continue;} p.vy+=(p.g||0)*dt; p.x+=p.vx*dt; p.y+=p.vy*dt; } }
function updFloats(dt){ for(let i=floats.length-1;i>=0;i--){ const f=floats[i]; f.t+=dt; f.y-=28*dt; if(f.t>f.dur) floats.splice(i,1); } }

  if(bannerObj){ bannerObj.t+=dt; if(bannerObj.t>bannerObj.dur) bannerObj=null; } }

/* ============ UPDATE ============ */
function step(dt){
  if(state==='pause') return;
  time+=dt;
  if(state==='over'){
    overT+=dt; updParts(dt); updFloats(dt);
    shakeMag*=Math.exp(-5*dt); flashA=Math.max(0,flashA-dt*1.8);
    if(overT>0.9 && !overShown) showOver();
    return;
  }
  // conveyor
  scroll += (cell/tickInterval)*dt;
  let guard=0; while(scroll>=cell && guard++<4){ scroll-=cell; doShift(); }

  // swap animation
  if(swapBusy){ swapT-=dt; if(swapT<=0){ if(swapReverting) swapBusy=false; else evalSwap(); } }

  // cascades
  if(resolving && dropTimer>0){ dropTimer-=dt; if(dropTimer<=0) resolveStep(); }

  // ambient matches (conveyor gifts + attract mode life)
  if(!resolving && !swapBusy){ const m=findMatches(); if(m.has) beginResolve(); }

  // gem offsets / pop
  const f=Math.exp(-16*dt);
  for(let c=0;c<L;c++) for(let r=0;r<ROWS;r++){ const g=grid[c][r]; if(!g) continue;
    g.dx*=f; g.dy*=f; if(Math.abs(g.dx)<.5) g.dx=0; if(Math.abs(g.dy)<.5) g.dy=0;
    if(g.pop<1) g.pop=Math.min(1,g.pop+dt*4); }

  // play-only systems
  if(state==='play'){
    heat=Math.max(0, heat-DECAY*dt);
    const od=heat<25;
    if(od!==overdrive){ overdrive=od; banner(od?'OVERDRIVE \u00D71.5':'BELT NORMAL', od?'#2fe0c8':'#ffd9b0'); if(od) sfx.overOn(); else sfx.overOff(); }
    hintT+=dt;
    scoreShown+=(score-scoreShown)*Math.min(1,dt*12);
  }
  if(state==='menu'){ autoSwapT-=dt; if(autoSwapT<=0){ autoSwapT=1+Math.random(); tryAutoSwap(); } }

  bgScroll += (cell/tickInterval)*dt;
  for(const e of embers){ e.y-=e.v*dt; e.x+=Math.sin(time*1.3+e.ph)*10*dt; if(e.y<-8){ e.y=H+8; e.x=Math.random()*W; } }
  updParts(dt); updFloats(dt); updBanner(dt);
  shakeMag*=Math.exp(-5*dt); if(shakeMag<.1) shakeMag=0;
  flashA=Math.max(0,flashA-dt*2.2);
  chuteFlash=Math.max(0,chuteFlash-dt*3);
  furnaceFlash=Math.max(0,furnaceFlash-dt*3);
}

/* ============ DRAW HELPERS ============ */
function rr(x,y,w,h,r){ ctx.beginPath(); ctx.moveTo(x+r,y); ctx.arcTo(x+w,y,x+w,y+h,r);
  ctx.arcTo(x+w,y+h,x,y+h,r); ctx.arcTo(x,y+h,x,y,r); ctx.arcTo(x,y,x+w,y,r); ctx.closePath(); }
function eob(t){ const c1=1.70158,c3=c1+1; return 1+c3*Math.pow(t-1,3)+c1*Math.pow(t-1,2); }

/* ============ GEMS ============ */
function drawGem(g,cx,cy,isSel){
  const s=cell*0.82, T=TYPES[g.t];
  const pop=g.pop<1? eob(Math.max(0,g.pop)) : 1;
  ctx.fillStyle='rgba(0,0,0,.35)';
  ctx.beginPath(); ctx.ellipse(cx+g.dx, cy+cell*0.42+g.dy, s*0.34, s*0.1, 0, 0, 6.283); ctx.fill();
  ctx.save();
  ctx.translate(cx+g.dx, cy+g.dy);
  let sc=pop;
  if(isSel){ const p=1+.06*Math.sin(time*13); sc=pop*p; ctx.rotate(Math.sin(time*11)*.07); }
  ctx.scale(sc,sc);
  switch(g.t){
    case 0:{ // ember flame
      ctx.rotate(Math.sin(time*6+g.uid)*.05); ctx.scale(1, 1+.04*Math.sin(time*9+g.uid));
      const gr=ctx.createLinearGradient(0,-s*.5,0,s*.5);
      gr.addColorStop(0,T.hi); gr.addColorStop(.45,T.fill); gr.addColorStop(1,T.lo);
      ctx.fillStyle=gr; ctx.strokeStyle='rgba(20,4,0,.5)'; ctx.lineWidth=s*.06; ctx.lineJoin='round';
      ctx.beginPath(); ctx.moveTo(0,-s*.5);
      ctx.bezierCurveTo(s*.30,-s*.28, s*.46,s*.02, s*.36,s*.27);
      ctx.bezierCurveTo(s*.26,s*.46, -s*.26,s*.46, -s*.36,s*.27);
      ctx.bezierCurveTo(-s*.46,s*.02, -s*.30,-s*.28, 0,-s*.5);
      ctx.closePath(); ctx.stroke(); ctx.fill();
      ctx.fillStyle='rgba(255,244,200,.85)';
      ctx.beginPath(); ctx.moveTo(0,-s*.16);
      ctx.bezierCurveTo(s*.14,-s*.02, s*.17,s*.14, s*.11,s*.24);
      ctx.bezierCurveTo(s*.05,s*.32, -s*.05,s*.32, -s*.11,s*.24);
      ctx.bezierCurveTo(-s*.17,s*.14, -s*.14,-s*.02, 0,-s*.16); ctx.fill();
      break; }
    case 1:{ // gold hexagon
      const R=s*.48; let p=new Path2D();
      for(let i=0;i<6;i++){ const a=i*Math.PI/3-Math.PI/6, px=Math.cos(a)*R, py=Math.sin(a)*R; i?p.lineTo(px,py):p.moveTo(px,py); } p.closePath();
      const gr=ctx.createLinearGradient(-R,-R,R,R);
      gr.addColorStop(0,T.hi); gr.addColorStop(.5,T.fill); gr.addColorStop(1,T.lo);

**How it plays:** The slag belt scrolls gems leftward toward a glowing furnace maw — every column that reaches the jaws raises the FURNACE LOAD gauge, while every gem you shatter in a match cools it. Swap adjacent gems by dragging, click-click, or keyboard (arrows aim, SPACE grabs, an arrow flings). Cascades stack combo multipliers, keeping the load below 25 triggers **OVERDRIVE** for ×1.5 score, and the belt speeds up every few columns until something gives.
