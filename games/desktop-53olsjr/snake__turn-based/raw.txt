function aiChoose(sn){
 const opts=legalDirs(sn, {allowFatal:false}); // only safe moves
 if(!opts.length) return null;
 let best=null,bestScore=-1e9;
 for(const d of opts){
   const h=sn.cells[0], nx=h.x+DIRS[d][0], ny=h.y+DIRS[d][1];
   const eating = foods.some(f=>f.x===nx&&f.y===ny);
   // distance to nearest food after move
   let bd=99; for(const f of foods){ if(eating && f.x===nx&&f.y===ny) continue; bd=Math.min(bd,Math.abs(f.x-nx)+Math.abs(f.y-ny));}
   const space = openSpace(nx,ny,sn,eating); // count of free neighbors excluding blocked (treat tail as free if !eating)
   let s = eating? 1000 : -bd*3 + space*2 + Math.random()*4;

 }

}

function foodAt(x,y){if(gold&&gold.x===x&&gold.y===y)return'gold';return foods.some(f=>f.x===x&&f.y===y)?'food':null;}

function moveLegal(s,dir){
  const[dx,dy]=DIRS[dir],h=s.cells[0],nx=h.x+dx,ny=h.y+dy;
  if(nx<0||ny<0||nx>=N||ny>=N)return false;
  const eat=!!foodAt(nx,ny);
  for(const o of snakes){
    if(!o.alive)continue;
    const cs=o.cells,L=cs.length;
    for(let i=0;i<L;i++){
      const c=cs[i];
      if(c.x!==nx||c.y!==ny)continue;
      if(o===s){if(i===L-1&&!eat)continue;return false;}
      return false;
    }
  }
  return true;
}
function openSpace(x,y,s){
  let n=0;
  for(const d of['up','down','left','right']){
    const nx=x+DIRS[d][0],ny=y+DIRS[d][1];
    if(nx<0||ny<0||nx>=N||ny>=N)continue;
    let occ=false;
    outer:for(const o of snakes){
      if(!o.alive)continue;
      const cs=o.cells,L=cs.length;
      for(let i=0;i<L;i++){const c=cs[i];
        if(c.x===nx&&c.y===ny){if(o===s&&i===L-1)continue;occ=true;break outer;}}
    }
    if(!occ)n++;
  }
  return n;
}

function resetGame(){
  session++;phase='play';
  snakes=[

  ];
  foods=[];for(let i=0;i<3;i++)spawnFood();
  gold=null;score=0;round=1;turnIdx=0;pendingDir='right';particles=[];shake=0;
  $('log').innerHTML='';
  addLog('The match begins — <b style="color:#3ddc7a">YOU</b> strike first.','sys');
  setScoreText(0,false);
  updateChips();updateBars();updateHud();setTurnLabel('you');
  $('startOv').classList.add('hidden');$('endOv').classList.add('hidden');
}
function schedule(fn,ms){const sid=session;setTimeout(()=>{if(sid===session&&phase==='play')fn();},ms);}

function onRound(){
  addLog(`— ROUND ${round} —`,'sys');
  if(!gold&&round>1&&(round-1)%3===0){
    const p=freeCell();
    if(p){gold={x:p.x,y:p.y,ttl:14};addLog('A <b style="color:#ffd166">GOLDEN BERRY</b> appears on the field!','gold');}
  }
  updateHud();
}

function performMove(s,dir){
  const[dx,dy]=DIRS[dir];s.dir=dir;
  const h=s.cells[0],nx=h.x+dx,ny=h.y+dy;
  const eat=foodAt(nx,ny);let grow=eat?1:0;if(eat==='gold')grow=2;
  const old=s.cells.slice();
  s.cells.unshift({x:nx,y:ny});
  if(!eat)s.cells.pop();
  else{
    if(grow===2)s.cells.push({x:old[old.length-1].x,y:old[old.length-1].y});
    if(eat==='gold')gold=null;else{const i=foods.findIndex(f=>f.x===nx&&f.y===ny);if(i>=0)foods.splice(i,1);}
    onEat(s,eat,nx,ny);
  }
  const L=s.cells.length,from=new Array(L);
  for(let i=0;i<L;i++){const c=old[Math.min(i,old.length-1)];from[i]={x:c.x,y:c.y};}
  s.anim={from,t0:performance.now()};
  if(gold){gold.ttl--;if(gold.ttl<=0){gold=null;addLog('The golden berry faded away.','sys');}}
  updateHud();
}

function onEat(s,type,x,y){
  const p=cellPx({x,y});
  burst(p.x,p.y,type==='gold'?[255,209,102]:[255,90,90],type==='gold'?26:14,3.2);
  if(type==='gold'){SFX.gold();shake=Math.max(shake,5);}
  else if(s.isPlayer)SFX.eatP();else SFX.eatR();
  s.eaten++;
  if(s.isPlayer)addScore(type==='gold'?40:15);
  addLog(`${tag(s)} ${type==='gold'?'STRIKE! golden berry · +2 length':'ate a berry (+1)'}`,type==='gold'?'gold':(s.isPlayer?'good':''));
  if(type==='food')spawnFood();
  updateChips();updateBars();
  if(s.cells.length>=WIN_LEN)endGame(s.isPlayer,s.isPlayer?'win_len':'lose_rival',s);
}

function killSnake(s,cause){
  if(!s.alive)return;
  s.alive=false;shake=13;SFX.die();
  const p=cellPx(s.cells[0]);
  burst(p.x,p.y,s.body,34,4.5);
  setTimeout(()=>burst(p.x,p.y,s.head,18,2.4),120);
  const msg={wall:'smashed into the wall',self:'bit its own coils'}[cause]||'suffocated';
  addLog(`${tag(s)} ${msg} — <b>ELIMINATED</b>`,'bad');
  if(!s.isPlayer)addScore(50);
  updateChips();updateBars();
  const alive=snakes.filter(x=>x.alive);
  if(!s.isPlayer&&alive.length===1&&snakes[0].alive)endGame(true,'win_last');
}

function commitPlayer(){
  if(phase!=='play')return;
  const s=snakes[turnIdx];if(!s||!s.isPlayer)return;
  SFX.tick();
  if(!moveLegal(s,pendingDir)){
    const[dx,dy]=DIRS[pendingDir],h=s.cells[0];
    const wall=h.x+dx<0||h.y+dy<0||h.x+dx>=N||h.y+dy>=N;
    killSnake(s,wall?'wall':'self');
    endGame(false,wall?'lose_wall':'lose_self');return;
  }
  performMove(s,pendingDir);
  setTurnLabel('wait');
  schedule(nextTurn,NEXT_DELAY);
}

function nextTurn(){
  if(phase!=='play')return;
  let i=turnIdx,g=0;
  do{i=(i+1)%snakes.length;g++;}while(!snakes[i].alive&&g<=snakes.length);
  turnIdx=i;
  if(i===0)onRound();
  updateChips();
  const s=snakes[i];
  if(s.isPlayer){
    const has=['up','down','left','right'].some(d=>moveLegal(s,d));
    if(!has){killSnake(s,'self');endGame(false,'lose_stuck');return;}
    setTurnLabel('you');
  }else{
    setTurnLabel('rival',s);
    schedule(()=>{
      if(phase!=='play')return;
      const d=aiChoose(s);
      if(d===null){killSnake(s,'self');if(phase==='play'){setTurnLabel('wait');schedule(nextTurn,650);}return;}
      if(!moveLegal(s,d)){
        const[dx,dy]=DIRS[d],h=s.cells[0];
        killSnake(s,(h.x+dx<0||h.y+dy<0||h.x+dx>=N||h.y+dy>=N)?'wall':'self');
        if(phase==='play'){setTurnLabel('wait');schedule(nextTurn,650);}
        return;
      }
      SFX.rival();
      performMove(s,d);
      setTurnLabel('wait');
      schedule(nextTurn,NEXT_DELAY);
    },RIVAL_DELAY+Math.random()*220);
  }
}

function aiChoose(s){
  const all=Object.keys(DIRS);
  const safe=all.filter(d=>moveLegal(s,d));
  if(!safe.length)return null;
  if(safe.length<=2&&Math.random()<.18)return all[Math.random()*4|0]; // panic blunder
  const targets=foods.slice();if(gold)targets.push(gold);
  let best=safe[0],bs=-1e9;
  for(const d of safe){
    const[dx,dy]=DIRS[d],h=s.cells[0],nx=h.x+dx,ny=h.y+dy;
    const et=foodAt(nx,ny);let bd=99;
    for(const f of targets){if(et&&f.x===nx&&f.y===ny)continue;bd=Math.min(bd,Math.abs(f.x-nx)+Math.abs(f.y-ny));}
    const space=openSpace(nx,ny,s);
    const wallDist=Math.min(nx,ny,N-1-nx,N-1-ny);
    const sc=(et==='gold'?1400:et==='food'?900:-bd*3)+space*2.5+wallDist*.6+Math.random()*5;
    if(sc>bs){bs=sc;best=d;}
  }
  return best;
}

function endGame(win,reason,winner){
  if(phase==='over')return;
  phase='over';session++;
  let final=score;if(win)final+=200;
  if(final>best){best=final;try{localStorage.setItem('sg_best',String(best));}catch(e){}}
  $('best').textContent=best;$('bestLine').textContent='BEST SCORE — '+best;
  setScoreText(final,true);
  const tEl=$('endTitle');
  tEl.textContent=win?'VICTORY':'DEFEAT';
  tEl.className='title result '+(win?'victory':'defeat');
  const F={
    win_len:'You coiled to length 9 — a true serpent rules the garden.',
    win_last:'EMBER and TIDE are no more. Last serpent standing.',
    lose_wall:'Your head met the wall. A gambit poorly played.',
    lose_self:"You bit your own coils. The oldest fool's move.",
    lose_stuck:'Every cell was taken. Suffocated in your own garden.'
  };
  $('endFlavor').textContent=reason==='lose_rival'?`${winner.name} reached length 9 first. The crown changes hands.`:(F[reason]||'');
  const p=snakes[0];
  $('endStats').innerHTML=

    `<div class="stat"><b>${p.cells.length}</b><span>LENGTH</span></div>`+

  $('endOv').classList.remove('hidden');

}

function circle(x,y,r){ctx.beginPath();ctx.arc(x,y,Math.max(.1,r),0,Math.PI*2);ctx.fill();}
function mix(a,b,f){return[a[0]+(b[0]-a[0])*f|0,a[1]+(b[1]-a[1])*f|0,a[2]+(b[2]-a[2])*f|0];}
function rgb(c){return`rgb(${c[0]},${c[1]},${c[2]})`;}

function drawSnake(s,t,active){
  const{M,cell}=metrics(),L=s.cells.length;
  const k=s.anim?Math.min(1,(t-s.anim.t0)/MOVE_MS):1;
  const e=1-Math.pow(1-k,3);
  const pts=[];
  for(let i=0;i<L;i++){
    const c=s.cells[i];let fx=c.x,fy=c.y;
    if(s.anim){const f=s.anim.from[i]||c;fx=f.x+(c.x-f.x)*e;fy=f.y+(c.y-f.y)*e;}
    pts.push({x:M+fx*cell+cell/2,y:M+fy*cell+cell/2});
  }
  const w=cell*.66;
  ctx.lineCap='round';ctx.lineJoin='round';
  for(let i=L-1;i>=1;i--){
    const f=i/(L-1);
    ctx.strokeStyle=rgb(mix(s.body,s.dark,f*.75));
    ctx.lineWidth=w*(1-f*.3);
    ctx.beginPath();ctx.moveTo(pts[i].x,pts[i].y);ctx.lineTo(pts[i-1].x,pts[i-1].y);ctx.stroke();
  }
  const hp=pts[0];
  ctx.save();ctx.shadowColor=s.glow;ctx.shadowBlur=cell*.55;
  ctx.fillStyle=rgb(s.head);circle(hp.x,hp.y,w*.62);ctx.restore();
  const[dx,dy]=DIRS[s.dir],ex=-dy,ey=dx,offF=cell*.16,offS=cell*.2,er=cell*.13;
  for(const sg of[1,-1]){
    const cx=hp.x+dx*offF+ex*offS*sg,cy=hp.y+dy*offF+ey*offS*sg;
    ctx.fillStyle='#fff';circle(cx,cy,er);
    ctx.fillStyle='#0b1410';circle(cx+dx*cell*.05,cy+dy*cell*.05,er*.55);
  }
  if(active&&Math.sin(t/280+(s.isPlayer?0:2))>.55){ // tongue flick
    ctx.strokeStyle='#ff6b6b';ctx.lineWidth=2;
    ctx.beginPath();ctx.moveTo(hp.x+dx*w*.55,hp.y+dy*w*.55);
    ctx.lineTo(hp.x+dx*(w*.55+cell*.4),hp.y+dy*(w*.55+cell*.4));ctx.stroke();
  }
  if(active){ // bobbing chevron over the active head
    const bob=Math.sin(t/260)*3,cx=hp.x,cy=hp.y-cell*1.08+bob;
    ctx.fillStyle=s.isPlayer?'rgba(255,209,102,.95)':'rgba(255,255,255,.75)';
    ctx.beginPath();ctx.moveTo(cx,cy+8);ctx.lineTo(cx-6,cy-3);ctx.lineTo(cx+6,cy-3);ctx.closePath();ctx.fill();
  }
}

function draw(t){
  const dpr=Math.min(2,window.devicePixelRatio||1);
  ctx.setTransform(dpr,0,0,dpr,0,0);
  ctx.clearRect(0,0,S,S);
  if(shake>.3){ctx.translate((Math.random()-.5)*shake,(Math.random()-.5)*shake);shake*=.86;}else shake=0;
  const{M,cell}=metrics();

  // board base + checker
  ctx.fillStyle='#0d2015';ctx.fillRect(M-5,M-5,S-2*(M-5),S-2*(M-5));
  for(let y=0;y<N;y++)for(let x=0;x<N;x++){
    ctx.fillStyle=(x+y)%2?'#143122':'#183a28';
    ctx.fillRect(M+x*cell,M+y*cell,cell-1,cell-1);
  }
  // frame + corner ticks
  ctx.strokeStyle='#3a7d58';ctx.lineWidth=3;
  ctx.strokeRect(M-5,M-5,S-2*(M-5),S-2*(M-5));
  ctx.strokeStyle='rgba(255,209,102,.5)';ctx.lineWidth=2;const tk=cell*.7;
  [[M-5,M-5,1,1],[S-M+5,M-5,-1,1],[M-5,S-M+5,1,-1],[S-M+5,S-M+5,-1,-1]].forEach(([x,y,sx,sy])=>{
    ctx.beginPath();ctx.moveTo(x+sx*tk,y);ctx.lineTo(x,y);ctx.lineTo(x,y+sy*tk);ctx.stroke();
  });

  // berries
  for(const f of foods){
    const p=cellPx(f),sc=Math.min(1,(t-f.born)/250);
    const r=cell*.3*(1+.1*Math.sin(t/240+(f.x*7+f.y*13)))*(.6+.4*sc);
    ctx.save();ctx.shadowColor='rgba(255,90,90,.8)';ctx.shadowBlur=cell*.5;
    ctx.fillStyle='#ff5a5a';circle(p.x,p.y,r);ctx.restore();
    ctx.fillStyle='rgba(255,255,255,.85)';circle(p.x-r*.3,p.y-r*.35,r*.28);
    ctx.save();ctx.translate(p.x+r*.45,p.y-r*.95);ctx.rotate(-.6);
    ctx.fillStyle='#3ddc7a';ctx.beginPath();ctx.ellipse(0,0,cell*.16,cell*.07,0,0,Math.PI*2);ctx.fill();ctx.restore();
  }
  // golden berry
  if(gold){
    const p=cellPx(gold),blink=gold.ttl<=4?.55+.45*Math.sin(t/90):1;
    ctx.save();ctx.globalAlpha=blink;
    ctx.shadowColor='rgba(255,209,102,.9)';ctx.shadowBlur=cell*.8;
    ctx.fillStyle='#ffd166';circle(p.x,p.y,cell*.34);
    ctx.fillStyle='#fff7df';circle(p.x,p.y,cell*.17);
    ctx.strokeStyle='rgba(255,247,223,.9)';ctx.lineWidth=2;
    const rot=t/600,ar=cell*.8;
    for(let i=0;i<4;i++){const a=rot+i*Math.PI/2;
      ctx.beginPath();ctx.moveTo(p.x+Math.cos(a)*cell*.2,p.y+Math.sin(a)*cell*.2);
      ctx.lineTo(p.x+Math.cos(a)*ar,p.y+Math.sin(a)*ar);ctx.stroke();}
    ctx.restore();
  }

  // target highlight (your turn)
  if(phase==='play'&&snakes[turnIdx]&&snakes[turnIdx].isPlayer){
    const s=snakes[turnIdx],[dx,dy]=DIRS[pendingDir],h=s.cells[0];
    const tx=h.x+dx,ty=h.y+dy;
    if(tx>=0&&ty>=0&&tx<N&&ty<N){
      const p=cellPx({x:tx,y:ty}),pr=cell*.34*(1+.1*Math.sin(t/150));
      const ok=moveLegal(s,pendingDir);
      ctx.strokeStyle=ok?'rgba(61,220,122,.9)':'rgba(255,90,90,.95)';ctx.lineWidth=3;
      ctx.beginPath();ctx.arc(p.x,p.y,pr,0,Math.PI*2);ctx.stroke();
      if(!ok){ctx.beginPath();
        ctx.moveTo(p.x-pr*.6,p.y-pr*.6);ctx.lineTo(p.x+pr*.6,p.y+pr*.6);
        ctx.moveTo(p.x+pr*.6,p.y-pr*.6);ctx.lineTo(p.x-pr*.6,p.y+pr*.6);ctx.stroke();}
    }
  }

  // snakes (active drawn last / on top)
  const order=snakes.filter(s=>s.alive).sort((a,b)=>(a===snakes[turnIdx]?1:0)-(b===snakes[turnIdx]?1:0));
  for(const s of order)drawSnake(s,t,phase==='play'&&s===snakes[turnIdx]);

  // particles
  const dt=Math.min(50,t-tPrev);tPrev=t;
  for(let i=particles.length-1;i>=0;i--){
    const p=particles[i];p.life-=dt;
    if(p.life<=0){particles.splice(i,1);continue;}
    p.x+=p.vx*dt/16;p.y+=p.vy*dt/16;p.vy+=.06*dt/16;
    const a=p.life/p.max;
    ctx.fillStyle=`rgba(${p.col[0]},${p.col[1]},${p.col[2]},${a.toFixed(3)})`;
    circle(p.x,p.y,Math.max(.4,p.size*a));
  }
}
