function rr(x,y,w,h,r){ctx.beginPath();ctx.moveTo(x+r,y);ctx.arcTo(x+w,y,x+w,y+h,r);ctx.arcTo(x+w,y+h,x,y+h,r);ctx.arcTo(x,y+h,x,y,r);ctx.arcTo(x,y,x+w,y,r);ctx.closePath()}
function pad6(n){return String(Math.max(0,Math.floor(n))).padStart(6,'0')}
function multFor(c){return c>=15?5:c>=10?4:c>=6?3:c>=3?2:1}
function mkFloat(x,y,txt,col,size){return{x,y,txt,col,size,t:0,life:1.1}}
function mkBanner(txt,sub,d){G.banner={txt,sub,t:d,d}}

/* ================= WAVES & ENTITIES ================= */
function ballSpeed(){return Math.min(560,360+(G.wave-1)*14)}
function buildWave(n){
  bricks.length=0;
  const cols=9,gap=8,side=52,bh=24,vgap=10,top=86;
  const bw=(FIELD.w-side*2-(cols-1)*gap)/cols;
  const rows=Math.min(8,4+Math.floor(n/2));
  const tiers=Math.min(3,1+Math.floor((n-1)/2));
  for(let r=0;r<rows;r++)for(let c=0;c<cols;c++){
    if(Math.random()<.05)continue;
    let hp=1;
    if(tiers>=2&&r<rows*.55)hp=2;
    if(tiers>=3&&r<rows*.34)hp=3;
    if(n>=3&&Math.random()<.07)hp=Math.min(3,hp+1);
    const k={x:side+c*(bw+gap),y:top+r*(bh+vgap),w:bw,h:bh,hp,maxHp:hp,flash:0,moving:false,
      baseX:side+c*(bw+gap),amp:0,ph:rand(0,6.28),sp:rand(.8,1.4)*Math.min(2,.5+n*.12),seed:Math.random()*99};
    if(n>=4&&r>0&&Math.random()<.13){k.moving=true;k.amp=rand(6,10)}
    bricks.push(k);
  }
  G.total=bricks.length;
}
function spawnOnPaddle(){balls.length=0;const b={x:paddle.x,y:paddle.y-12,vx:0,vy:0,r:7,speed:ballSpeed(),trail:[]};balls.push(b);return b}
function launch(){
  const b=balls[0];if(!b)return;
  const a=rand(-.35,.35);
  b.vx=Math.sin(a)*b.speed;b.vy=-Math.cos(a)*b.speed;
  G.state='play';sfx.paddle();burst(b.x,b.y+8,'#6ff2df',6,140);
}

/* capsules */
const CAPS={WIDE:{col:'#5fe8d9',w:24,label:'W'},MULTI:{col:'#a4e34f',w:22,label:'M'},
  SLOW:{col:'#7ab8ff',w:20,label:'S'},LASER:{col:'#ff8a3c',w:20,label:'L'},FAST:{col:'#ff4d6d',w:14,label:'F'}};
function dropCapsule(x,y){
  const names=Object.keys(CAPS);let tot=0;names.forEach(n=>tot+=CAPS[n].w);
  let r=Math.random()*tot,t='WIDE';
  for(const n of names){r-=CAPS[n].w;if(r<=0){t=n;break}}
  capsules.push({x,y,vy:150,type:t,ph:rand(0,6)});
}
function applyCapsule(t){
  if(t==='FAST'){fx.fast=8;sfx.bad();floats.push(mkFloat(paddle.x,paddle.y-34,'BALL SPEED UP!','#ff4d6d',18))}
  else{
    if(t==='WIDE')fx.wide=12;
    if(t==='SLOW')fx.slow=8;
    if(t==='LASER')fx.laser=10;
    if(t==='MULTI'){
      const orig=balls.slice();
      for(const b of orig){
        const hasV=Math.abs(b.vx)+Math.abs(b.vy)>1;
        const base=hasV?Math.atan2(b.vy,b.vx):-Math.PI/2;
        for(const da of[-.5,.5]){
          if(balls.length>=6)break;
          balls.push({x:b.x,y:b.y,vx:Math.cos(base+da)*b.speed,vy:Math.sin(base+da)*b.speed,r:7,speed:b.speed,trail:[]});
        }
      }
    }
    sfx.power();floats.push(mkFloat(paddle.x,paddle.y-34,t,CAPS[t].col,18));
  }
}

/* particles & floats */
function burst(x,y,col,n,spd){
  spd=spd||260;
  for(let i=0;i<n;i++){const a=rand(0,6.28),s=rand(.3,1)*spd;
    parts.push({x,y,vx:Math.cos(a)*s,vy:Math.sin(a)*s-60,g:520,life:rand(.4,.9),t:0,size:rand(2,5),col,rot:rand(0,6.28),vr:rand(-8,8)});}
  if(parts.length>420)parts.splice(0,parts.length-420);
}
function updateParts(dt){for(let i=parts.length-1;i>=0;i--){const p=parts[i];p.t+=dt;if(p.t>p.life){parts.splice(i,1);continue}p.vy+=p.g*dt;p.x+=p.vx*dt;p.y+=p.vy*dt;p.rot+=p.vr*dt}}
function updateFloats(dt){for(let i=floats.length-1;i>=0;i--){const f=floats[i];f.t+=dt;f.y-=34*dt;if(f.t>f.life)floats.splice(i,1)}}

/* ================= FLOW ================= */
function addScore(v){G.score+=v;if(G.score>G.hi){G.hi=G.score;G.hiDirty=true}}
function checkLife(){
  while(G.nextLife<LIFE_AT.length&&G.score>=LIFE_AT[G.nextLife]){
    G.nextLife++;
    if(G.lives<6){G.lives++;sfx.extra();floats.push(mkFloat(FIELD.w/2,300,'EXTRA BALL!','#6ff2df',26))}
  }
}
function brickDestroyed(k){
  G.combo++;killsSinceDrop++;
  const m=multFor(G.combo),base=10*k.maxHp;
  addScore(base*m);
  floats.push(mkFloat(k.x+k.w/2,k.y+8,'+'+(base*m),m>1?'#6ff2df':'#ffd97a',m>1?18:15));
  if(multFor(G.combo)>multFor(G.combo-1))floats.push(mkFloat(paddle.x,paddle.y-42,'COMBO x'+m,'#ffb43a',22));
  sfx.brick(Math.min(20,G.combo));
  burst(k.x+k.w/2,k.y+k.h/2,TIER[k.maxHp],14);
  G.shake=Math.max(G.shake,2.2);
  if(killsSinceDrop>=12||Math.random()<.16){dropCapsule(k.x+k.w/2,k.y+k.h/2);killsSinceDrop=0}
  checkLife();
}
function hitBrick(k){
  k.hp--;k.flash=.14;
  if(k.hp>0){sfx.crack();G.shake=Math.max(G.shake,1)}
  else brickDestroyed(k);
}
function loseLife(){
  G.lives--;G.combo=0;lasers.length=0;
  G.shake=11;G.flash=.9;sfx.life();
  burst(paddle.x,paddle.y+6,'#ff5f45',26,320);
  if(G.lives<=0){gameOver()}
  else{spawnOnPaddle();G.state='ready';mkBanner('BALL LOST',(G.lives)+' LEFT',1.3)}
}
function waveCleared(){
  const bonus=150*G.wave;addScore(bonus);checkLife();
  G.state='waveclear';G.waveTimer=1.7;balls.length=0;
  sfx.wave();burst(FIELD.w/2,FIELD.h*.35,'#ffd97a',40,420);
  mkBanner('WAVE '+G.wave+' CLEARED','BONUS +'+bonus,1.7);
}
function startGame(){
  G.score=0;G.wave=1;G.lives=3;G.combo=0;G.nextLife=0;G.newBest=false;G.shake=0;G.flash=0;G.paused=false;
  G.hiStart=G.hi;
  fx.wide=fx.slow=fx.laser=fx.fast=0;paddle.w=paddle.baseW;
  capsules.length=0;lasers.length=0;parts.length=0;floats.length=0;killsSinceDrop=0;
  buildWave(1);spawnOnPaddle();
  G.state='ready';mkBanner('WAVE 1','CLEAR THE FORGE',1.6);
  hideAll();
}
function toMenu(){
  G.paused=false;hideAll();G.state='menu';
  fx.wide=fx.slow=fx.laser=fx.fast=0;paddle.w=paddle.baseW;
  buildWave(1);capsules.length=0;lasers.length=0;floats.length=0;parts.length=0;
  const b={x:FIELD.w/2,y:FIELD.h*.5,vx:rand(-.6,.6),vy:-1,r:7,speed:380,trail:[]};
  const m=Math.hypot(b.vx,b.vy);b.vx=b.vx/m*b.speed;b.vy=b.vy/m*b.speed;balls=[b];
  document.getElementById('menuHi').textContent=pad6(G.hi);
  show('menu');
}
function gameOver(){
  G.state='gameover';sfx.over();G.shake=12;G.flash=.7;
  G.newBest=G.score>G.hiStart;
  if(G.newBest){G.hi=G.score;saveHi(G.hi);G.hiDirty=false}
  document.getElementById('finalScore').textContent=pad6(G.score);
  document.getElementById('overStats').textContent='You melted through '+G.wave+' wave'+(G.wave>1?'s':'')+' · best '+pad6(G.hi);
  document.getElementById('newBest').classList.toggle('hidden',!G.newBest);
  show('overbox');
}

  if(G.state!=='play'&&G.state!=='ready')return;
  G.paused=!G.paused;
  document.getElementById('pausebox').classList.toggle('hidden',!G.paused);
}

/* ================= UPDATE ================= */
function speedFactor(){return (fx.slow>0?.62:1)*(fx.fast>0?1.35:1)}
function paddleTargetW(){let w=paddle.baseW;if(fx.wide>0)w*=1.55;return w}

function movePaddle(dt){
  paddle.px=paddle.x;
  let dir=0;
  if(keys.ArrowLeft||keys.KeyA)dir-=1;
  if(keys.ArrowRight||keys.KeyD)dir+=1;
  if(dir){paddle.x+=dir*560*dt;pointerX=null}
  else if(pointerX!=null)paddle.x+=(pointerX-paddle.x)*Math.min(1,dt*22);
  paddle.x=clamp(paddle.x,paddle.w/2+WALL.l+4,WALL.r-paddle.w/2-4);
  paddle.vx=(paddle.x-paddle.px)/Math.max(dt,1e-4);
  paddle.w+=(paddleTargetW()-paddle.w)*Math.min(1,dt*10);
}

function fireLaser(){
  if(fx.laser<=0||laserCd>0)return;
  laserCd=.26;sfx.laser();
  lasers.push({x:paddle.x-paddle.w/2+10,y:paddle.y-4},{x:paddle.x+paddle.w/2-10,y:paddle.y-4});
}

function updateLasers(dt){
  laserCd-=dt;
  for(let i=lasers.length-1;i>=0;i--){
    const l=lasers[i];l.y-=980*dt;
    let dead=l.y<WALL.t+4;
    if(!dead)for(const k of bricks){
      if(k.hp<=0)continue;
      if(l.x>k.x&&l.x<k.x+k.w&&l.y>k.y&&l.y<k.y+k.h){hitBrick(k);burst(l.x,l.y,'#ff8a3c',4,120);sfx.laserHit();dead=true;break}
    }
    if(dead)lasers.splice(i,1);
  }
}

function updateBalls(dt){
  const sf=speedFactor();
  for(let i=balls.length-1;i>=0;i--){
    const b=balls[i];
    for(let s=0;s<3;s++){
      b.x+=b.vx*dt*sf/3;b.y+=b.vy*dt*sf/3;
      if(b.x-b.r<WALL.l){b.x=WALL.l+b.r;b.vx=Math.abs(b.vx);sfx.wall()}
      else if(b.x+b.r>WALL.r){b.x=WALL.r-b.r;b.vx=-Math.abs(b.vx);sfx.wall()}
      if(b.y-b.r<WALL.t){b.y=WALL.t+b.r;b.vy=Math.abs(b.vy);sfx.wall()}
      if(b.vy>0&&b.y+b.r>=paddle.y&&b.y-b.r<=paddle.y+paddle.h+8&&Math.abs(b.x-paddle.x)<=paddle.w/2+b.r){
        const rel=clamp((b.x-paddle.x)/(paddle.w/2),-1,1);
        const a=rel*1.02;
        b.vx=Math.sin(a)*b.speed+paddle.vx*.12;
        b.vy=-Math.cos(a)*b.speed;
        b.y=paddle.y-b.r-.5;
        G.combo=0;sfx.paddle();burst(b.x,paddle.y,'#6ff2df',3,90);
      }
      for(const k of bricks){
        if(k.hp<=0)continue;
        const cx=clamp(b.x,k.x,k.x+k.w),cy=clamp(b.y,k.y,k.y+k.h);
        let dx=b.x-cx,dy=b.y-cy;const d2=dx*dx+dy*dy;
        if(d2>b.r*b.r)continue;
        let nx,ny;
        if(d2<1e-6){nx=0;ny=-1;b.y=k.y-b.r}
        else{const d=Math.sqrt(d2);nx=dx/d;ny=dy/d;b.x=cx+nx*(b.r+.5);b.y=cy+ny*(b.r+.5)}
        const dot=b.vx*nx+b.vy*ny;
        if(dot<0){b.vx-=2*dot*nx;b.vy-=2*dot*ny}
        hitBrick(k);break;
      }
    }
    let m=Math.hypot(b.vx,b.vy);
    if(m>0){b.vx=b.vx/m*b.speed;b.vy=b.vy/m*b.speed}
    if(Math.abs(b.vy)<b.speed*.1){
      b.vy=(b.y<FIELD.h/2?-1:1)*b.speed*.1;
      m=Math.hypot(b.vx,b.vy);if(m>0){b.vx*=b.speed/m;b.vy*=b.speed/m}
    }
    b.trail.unshift({x:b.x,y:b.y});if(b.trail.length>12)b.trail.pop();
    if(b.y-b.r>FIELD.h+30)balls.splice(i,1);
  }
  if(G.state==='play'&&balls.length===0)loseLife();
}

function updateCapsules(dt){
  for(let i=capsules.length-1;i>=0;i--){
    const c=capsules[i];c.y+=c.vy*dt;c.ph+=dt*4;
    if(c.y>FIELD.h+20){capsules.splice(i,1);continue}
    if(c.y+9>=paddle.y&&c.y-9<=paddle.y+paddle.h&&Math.abs(c.x-paddle.x)<=paddle.w/2+16){
      applyCapsule(c.type);burst(c.x,c.y,CAPS[c.type].col,12);capsules.splice(i,1);
    }
  }
}

/* menu attract mode */
function movePaddleAuto(dt){
  const b=balls[0];if(!b)return;
  const target=(b.vy>0?b.x:FIELD.w/2+Math.sin(G.time*.7)*160);
  paddle.x+=clamp(target-paddle.x,-420*dt,420*dt);
  paddle.x=clamp(paddle.x,paddle.w/2+WALL.l+4,WALL.r-paddle.w/2-4);
}
function attract(dt){
  movePaddleAuto(dt);
  for(const k of bricks)if(k.moving&&k.hp>0)k.x=k.baseX+Math.sin(G.time*k.sp+k.ph)*k.amp;
  for(let i=balls.length-1;i>=0;i--){
    const b=balls[i];
    b.x+=b.vx*dt;b.y+=b.vy*dt;
    if(b.x-b.r<WALL.l){b.x=WALL.l+b.r;b.vx=Math.abs(b.vx)}
    else if(b.x+b.r>WALL.r){b.x=WALL.r-b.r;b.vx=-Math.abs(b.vx)}
    if(b.y-b.r<WALL.t){b.y=WALL.t+b.r;b.vy=Math.abs(b.vy)}
    if(b.vy>0&&b.y+b.r>=paddle.y&&Math.abs(b.x-paddle.x)<=paddle.w/2+b.r){
      const rel=clamp((b.x-paddle.x)/(paddle.w/2),-1,1),a=rel*1.02;
      b.vx=Math.sin(a)*b.speed;b.vy=-Math.cos(a)*b.speed;b.y=paddle.y-b.r;
    }
    for(const k of bricks){
      if(k.hp<=0)continue;
      const cx=clamp(b.x,k.x,k.x+k.w),cy=clamp(b.y,k.y,k.y+k.h);
      const dx=b.x-cx,dy=b.y-cy,d2=dx*dx+dy*dy;
      if(d2>b.r*b.r)continue;
      let nx,ny;
      if(d2<1e-6){nx=0;ny=-1;b.y=k.y-b.r}else{const d=Math.sqrt(d2);nx=dx/d;ny=dy/d;b.x=cx+nx*(b.r+.5);b.y=cy+ny*(b.r+.5)}
      const dot=b.vx*nx+b.vy*ny;if(dot<0){b.vx-=2*dot*nx;b.vy-=2*dot*ny}
      k.hp--;burst(k.x+k.w/2,k.y+k.h/2,TIER[k.maxHp],8,180);break;
    }
    b.trail.unshift({x:b.x,y:b.y});if(b.trail.length>12)b.trail.pop();
    if(b.y-b.r>FIELD.h+30)balls.splice(i,1);
  }
  let any=false;bricks.forEach(k=>{if(k.hp<=0)any=true});
  if(any){bricks=bricks.filter(k=>k.hp>0);G.total=bricks.length;if(!bricks.length)buildWave(1)}
  if(!balls.length){
    const b={x:FIELD.w/2,y:FIELD.h*.45,vx:rand(-.5,.5),vy:-1,r:7,speed:380,trail:[]};
    const m=Math.hypot(b.vx,b.vy);b.vx=b.vx/m*b.speed;b.vy=b.vy/m*b.speed;balls.push(b);
  }
  updateParts(dt);updateFloats(dt);
}

function update(dt){
  G.time+=dt;
  if(G.hiDirty&&G.time-lastSave>2){saveHi(G.hi);G.hiDirty=false;lastSave=G.time}
  G.shake=Math.max(0,G.shake-dt*30);
  G.flash=Math.max(0,G.flash-dt*1.6);
  if(G.banner){G.banner.t-=dt;if(G.banner.t<=0)G.banner=null}
  if(G.state==='menu'){attract(dt);return}
  if(G.paused)return;
  fx.wide=Math.max(0,fx.wide-dt);fx.slow=Math.max(0,fx.slow-dt);
  fx.laser=Math.max(0,fx.laser-dt);fx.fast=Math.max(0,fx.fast-dt);
  for(const k of bricks)if(k.flash>0)k.flash=Math.max(0,k.flash-dt);

  if(G.state==='ready'){
    movePaddle(dt);
    const b=balls[0];
    if(b){b.x=paddle.x;b.y=paddle.y-12;if(b.trail.length)b.trail.length=0}
  }
  else if(G.state==='play'){
    movePaddle(dt);
    for(const k of bricks)if(k.moving&&k.hp>0)k.x=k.baseX+Math.sin(G.time*k.sp+k.ph)*k.amp;
    updateBalls(dt);updateLasers(dt);
    let any=false;bricks.forEach(k=>{if(k.hp<=0)any=true});
    if(any){bricks=bricks.filter(k=>k.hp>0);G.total=bricks.length;if(bricks.length===0)waveCleared()}
  }
  else if(G.state==='waveclear'){
    G.waveTimer-=dt;
    if(G.waveTimer<=0){
      G.wave++;buildWave(G.wave);spawnOnPaddle();G.state='ready';
      mkBanner('WAVE '+G.wave,WAVE_HINTS[G.wave]||'KEEP GOING',1.6);
    }
  }
  updateCapsules(dt);updateParts(dt);updateFloats(dt);
}

function drawBricks(){
  for(const k of bricks){
    if(k.hp<=0)continue;
    const g=ctx.createLinearGradient(0,k.y,0,k.y+k.h);
    if(k.maxHp===3){const p=.5+.5*Math.sin(G.time*4+k.seed);g.addColorStop(0,'#ff7a5c');g.addColorStop(1,'rgb('+(150+p*40|0)+','+(40+p*12|0)+',40)')}
    else if(k.maxHp===2){g.addColorStop(0,'#ffb066');g.addColorStop(1,'#e05e2b')}
    else{g.addColorStop(0,'#ffe9a8');g.addColorStop(1,'#f0a83a')}
    rr(k.x,k.y,k.w,k.h,5);ctx.fillStyle=g;ctx.fill();
    ctx.fillStyle='rgba(255,255,255,.28)';rr(k.x+2,k.y+2,k.w-4,5,3);ctx.fill();
    rr(k.x,k.y,k.w,k.h,5);ctx.strokeStyle='rgba(0,0,0,.35)';ctx.lineWidth=1.5;ctx.stroke();
    if(k.hp<k.maxHp){
      ctx.strokeStyle='rgba(40,12,8,.55)';ctx.lineWidth=1.5;ctx.beginPath();
      const cx=k.x+k.w*(.3+.4*frac(k.seed));
      ctx.moveTo(cx,k.y+2);ctx.lineTo(cx+8,k.y+k.h*.5);ctx.lineTo(cx-6,k.y+k.h-3);
      if(k.hp<=k.maxHp-2){ctx.moveTo(k.x+k.w*.7,k.y+3);ctx.lineTo(k.x+k.w*.55,k.y+k.h*.6)}
      ctx.stroke();
    }
    if(k.moving){ctx.fillStyle='rgba(10,20,30,.5)';const cx=k.x+k.w/2;tri(cx-8,cx?k.y+k.h/2:k.y+k.h/2,-1);tri(cx+8,k.y+k.h/2,1)}

  }
}

  for(const s of stars){

    ctx.fillStyle=s.warm?'rgba(255,190,120,'+a+')':'rgba(160,215,235,'+a+')';
    ctx.fillRect(s.nx*W,y,s.r,s.r);
  }
  const sh=G.shake,sx=(Math.random()*2-1)*sh,sy=(Math.random()*2-1)*sh;
  ctx.save();

  ctx.restore();
}
