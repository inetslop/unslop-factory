function genLevel(lv){
  const rnd=mulberry32(lv*7919+101);
  const pat=(lv-1)%5;
  const cols=Math.min(17, 9+Math.floor((lv-1)/2));
  const rows=Math.min(11, 6+Math.ceil((lv-1)/3));
  const cc=(cols-1)/2, rc=(rows-1)/2;
  const out=[];
  for(let r=0;r<rows;r++)for(let c=0;c<cols;c++){
    let fill=false, armor=false;
    switch(pat){
      case 0: { // stagger with central lane
        const mid=Math.floor(rows/2);
        fill = (r===mid) ? (c%4===1||c%4===2) : (c+r)%3!==0;
        armor = (r===0||r===rows-1) && c%2===0;
      } break;
      case 1: { const d=Math.abs(c-cc)/cc+Math.abs(r-rc)/rc; fill=d<=1.18; armor=d<0.5; } break;
      case 2: { const border=r===0||c===0||r===rows-1||c===cols-1; fill=border||((c+r)%2===0); armor=border; } break;
      case 3: { fill=(c===2||c===cols-3); const mid=Math.floor(rows/2); if(r===mid&&c%2===0)fill=true; if((r===1||r===rows-2)&&(c===Math.floor(cols/2)))fill=true; armor=(c===2||c===cols-3)&& (r===0||r===rows-1); } break;
      case 4: { const d=Math.hypot(c-cc,r-rc); fill=Math.sin(d*1.5-lv*0.7)>0.15; armor=d>Math.min(cc,rc)*0.8 && fill; } break;
    }
    if(!fill) continue;
    let type='n', hp=1;
    if(armor){type='a';hp=3;}
    else if(rnd()<0.06){type='g';hp=2;}
    out.push({gx:c,gy:r,type,hp,maxHp:hp,seed:rnd()*1000,alive:true,rect:{x:0,y:0,w:0,h:0}});
  }
  if(!out.length) for(let c=2;c<cols-2;c++) out.push({gx:c,gy:Math.floor(rows/2),type:'n',hp:1,maxHp:1,seed:rnd()*99,alive:true,rect:{x:0,y:0,w:0,h:0}});

function genLevel(lv){

function update(dt){

  if(S.mode==='play')updatePad(dt);else if(S.mode==='menu')aiPad(dt);
  if(S.mode==='play'||S.mode==='menu'){stepBalls(sdt)}
  // powerups
  for(let i=pups.length-1;i>=0;i--){const p=pups[i];p.t+=sdt;
    p.x+=p.vx*sdt;p.y+=p.vy*sdt;
    if(p.x<arena.x+12){p.x=arena.x+12;p.vx=Math.abs(p.vx)}if(p.x>arena.x+arena.w-12){p.x=arena.x+arena.w-12;p.vx=-Math.abs(p.vx)}
    if(p.y<arena.y+12){p.y=arena.y+12;p.vy=Math.abs(p.vy)}if(p.y>arena.y+arena.h-12){p.y=arena.y+arena.h-12;p.vy=-Math.abs(p.vy)}
    const hl=pad.curLen/2,dx=Math.cos(pad.a)*hl,dy=Math.sin(pad.a)*hl,cx=pad.x-dx,cy=pad.y-dy;
    const t=clamp(((p.x-cx)*(dx*2)+(p.y-cy)*(dy*2))/(4*hl*hl),0,1);
    const qx=cx+dx*2*t,qy=cy+dy*2*t;
    if(Math.hypot(p.x-qx,p.y-qy)<pad.thick/2+13){applyPup(p.type,p.x,p.y);pups.splice(i,1);continue}
    if(p.t>p.life)pups.splice(i,1)}
  // particles / texts
  for(let i=parts.length-1;i>=0;i--){const p=parts[i];p.t+=sdt;
    if(!p.ring){p.x+=p.vx*sdt;p.y+=p.vy*sdt;p.vx*=Math.exp(-sdt*2.4);p.vy*=Math.exp(-sdt*2.4);p.rot+=p.vr*sdt}
