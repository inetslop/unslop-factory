function ensureAudio(){
  if(AC){if(AC.state==='suspended')AC.resume();return;}
  try{
    AC=new (window.AudioContext||window.webkitAudioContext)();
    master=AC.createGain();master.gain.value=muted?0:0.42;master.connect(AC.destination);
    if(!droneOn){droneOn=true;
      const g=AC.createGain();g.gain.value=0.045;const f=AC.createBiquadFilter();f.type='lowpass';f.frequency.value=220;
      const o1=AC.createOscillator(),o2=AC.createOscillator();o1.type='sine';o2.type='sine';o1.frequency.value=54;o2.frequency.value=54.7;
      o1.connect(f);o2.connect(f);f.connect(g);g.connect(master);o1.start();o2.start();}
  }catch(e){AC=null;}
}
function beep(f0,f1,dur,type,vol,delay){
  if(!AC||muted)return;delay=delay||0;
  try{const t=AC.currentTime+delay,o=AC.createOscillator(),g=AC.createGain();
  o.type=type;o.frequency.setValueAtTime(Math.max(1,f0),t);o.frequency.exponentialRampToValueAtTime(Math.max(1,f1),t+dur);
  g.gain.setValueAtTime(vol,t);g.gain.exponentialRampToValueAtTime(0.001,t+dur);
  o.connect(g);g.connect(master);o.start(t);o.stop(t+dur+0.03);}catch(e){}
}
function noiseB(dur,vol,freq,delay){
  if(!AC||muted)return;delay=delay||0;
  try{const t=AC.currentTime+delay,n=Math.floor(AC.sampleRate*dur),buf=AC.createBuffer(1,n,AC.sampleRate),d=buf.getChannelData(0);
  for(let i=0;i<n;i++)d[i]=(Math.random()*2-1)*(1-i/n);
  const s=AC.createBufferSource();s.buffer=buf;const f=AC.createBiquadFilter();f.type='bandpass';f.frequency.value=freq;f.Q.value=0.8;
  const g=AC.createGain();g.gain.value=vol;s.connect(f);f.connect(g);g.connect(master);s.start(t);}catch(e){}
}

function buildWave(n){
  const q=[];let side=Math.random()<0.5;
  let grunts=Math.min(6+Math.floor(n*1.5),20);
  let wisps=n>=2?Math.min(Math.floor(n*1.2),14):0;
  let brutes=n>=3?Math.floor(n/3):0;
  const base=Math.max(0.45,1.0-n*0.04);
  for(let i=0;i<wisps;i++){q.push({t:'w',d:base*0.9,side:side=!side});}
  for(let i=0;i<brutes;i++){q.push({t:'b',d:2.4,side:side=!side});}
  for(let i=0;i<grunts;i++){q.push({t:'g',d:base*(0.8+Math.random()*0.5),side:side=!side});}

  G.queue=q;
}

  for(const t of G.turrets)if(Math.abs(t.x-p.x)<46&&Math.abs(t.y-p.gy)<12)return false;

function landOnPlats(e,prevBottom){
  let bestY=null;
  for(const p of PLATS){
    if(e.x+e.w/2>p.x+2&&e.x-e.w/2<p.x+p.w-2&&prevBottom<=p.y+6&&e.y+e.h/2>=p.y){
      if(bestY===null||p.y<bestY)bestY=p.y;
    }
  }
  if(bestY!==null){e.y=bestY-e.h/2;e.vy=0;return true;}
  return false;
}
function groundCheck(e){
  e.onG=false;let gy=null;
  const b=e.y+e.h/2;
  for(const p of PLATS){if(e.x+e.w/2>p.x+2&&e.x-e.w/2<p.x+p.w-2&&Math.abs(b-p.y)<=3){e.onG=true;gy=p.y;break;}}
  return gy;
}

      const lead=bd/640,ax2=bt.x+bt.mvx*lead,ay2=bt.y+bt.mvy*lead;
      const want=Math.atan2(ay2-(t.y-14),ax2-t.x);
      t.angle=lerpAng(t.angle,want,Math.min(1,12*dt));
      if(t.cd<=0){
        t.cd=0.5;t.flash=1;SFX.tshoot();
        g.bullets.push({x:t.x+Math.cos(t.angle)*24,y:t.y-14+Math.sin(t.angle)*24,vx:Math.cos(t.angle)*660,vy:Math.sin(t.angle)*660,dmg:1,own:'t',age:0,baseY:t.y});
        P(t.x+Math.cos(t.angle)*26,t.y-14+Math.sin(t.angle)*26,Math.cos(t.angle)*90,Math.sin(t.angle)*90,0.1,4,'#ffe2a6',0,'spark');
      }
    }else t.angle=lerpAng(t.angle,-Math.PI/2,Math.min(1,5*dt));
  }
  // enemies
  for(const e of g.enemies){
    if(e.dead)continue;
    e.t+=dt;e.flash=Math.max(0,e.flash-dt*4);
    const px=e.x,py=e.y;
    if(e.type==='w'){
      const a=Math.atan2(CORE.y-e.y,CORE.x-e.x);
      e.x+=Math.cos(a)*e.spd*dt;
      e.y+=(Math.sin(a)*e.spd*0.55+Math.sin(e.t*3+e.seed)*46)*dt;
      e.y=clamp(e.y,120,620);
    }else{
      let spd=e.spd;if(e.burst>0){spd*=1.8;e.burst-=dt;}
      const dir=CORE.x<e.x?-1:1;
      e.vy=Math.min(900,e.vy+1800*dt);
      e.kvx*=Math.max(0,1-6*dt);
      e.x+=(dir*spd+e.kvx)*dt;
      const pb=e.y+e.h/2;e.y+=e.vy*dt;landOnPlats(e,pb);
      e.onG=groundCheck(e)!==null;
      if(e.type==='b'){e.stompT-=dt;
        if(e.stompT<=0&&e.onG){e.stompT=3+Math.random();e.burst=0.6;shake(2);SFX.hit();

    }
    e.mvx=(e.x-px)/Math.max(dt,0.001);e.mvy=(e.y-py)/Math.max(dt,0.001);
    // core
    if(dist(e.x,e.y,CORE.x,CORE.y)<e.r+42){explodeAtCore(e);continue;}
    // turrets
    for(const t of g.turrets){
      if(t.ifr>0)continue;
      if(dist(e.x,e.y,t.x,t.y-14)<e.r+16){
        t.hp-=e.tdmg;t.ifr=0.6;e.kvx=(e.x<t.x?-1:1)*280;SFX.hit();shake(2);
        for(let i=0;i<5;i++)P(t.x,t.y-14,(Math.random()-0.5)*180,-60-Math.random()*100,0.35,2.5,'#ffb350',300,'spark');
        if(t.hp<=0){g.turrets.splice(g.turrets.indexOf(t),1);text(t.x,t.y-34,'TURRET DOWN','#ff6b6b',14);

      }
    }
    // player
    if(!p.dead&&p.inv<=0&&dist(e.x,e.y,p.x,p.y)<e.r+15)damagePlayer(e);
  }
  g.enemies=g.enemies.filter(e=>!e.dead);
  // bullets
  for(const b of g.bullets){
    b.age+=dt;b.x+=b.vx*dt;b.y+=b.vy*dt;
    if(b.x<-20||b.x>W+20||b.y<-20||b.y>H+20){b.dead=true;continue;}
    for(const pl of PLATS){
      const ownBase=(b.baseY>=0&&Math.abs(pl.y-b.baseY)<1&&b.age<0.18);
      if(!ownBase&&b.x>pl.x&&b.x<pl.x+pl.w&&b.y>pl.y&&b.y<pl.y+pl.h){
        b.dead=true;
        for(let i=0;i<3;i++)P(b.x,b.y,(Math.random()-0.5)*140,-40-Math.random()*80,0.25,2,b.own==='p'?'#9ff0ff':'#ffd27a',300,'spark');
        break;
      }
    }
    if(b.dead)continue;
    for(const e of g.enemies){
      if(e.dead)continue;
      if(dist(b.x,b.y,e.x,e.y)<e.r+5){
        b.dead=true;e.hp-=b.dmg;e.flash=1;e.kvx+=(b.vx>0?1:-1)*70;SFX.hit();

        if(e.hp<=0)killEnemy(e);
        break;
      }
    }
  }
  g.bullets=g.bullets.filter(b=>!b.dead);
  // pickups
  for(const k of g.pickups){
    k.t-=dt;k.vy+=900*dt;
    if(!p.dead){const d=dist(k.x,k.y,p.x,p.y);
      if(d<130&&d>1){k.vx+=(p.x-k.x)/d*1100*dt;k.vy+=(p.y-k.y)/d*1100*dt;}}
    const pb=k.y+4;k.y+=k.vy*dt;k.x+=k.vx*dt;
    if(k.vy>0)for(const pl of PLATS){if(k.x>pl.x&&k.x<pl.x+pl.w&&pb<=pl.y+6&&k.y+4>=pl.y){k.y=pl.y-4;k.vy*=-0.35;k.vx*=0.7;break;}}
    if(!p.dead&&dist(k.x,k.y,p.x,p.y)<22){
      k.t=-1;g.scrap+=k.val;g.scrapEarned+=k.val;SFX.scrap();
      for(let i=0;i<4;i++)P(k.x,k.y,(Math.random()-0.5)*160,-80-Math.random()*120,0.35,2,'#ffd27a',-100,'spark');
    }
  }
  g.pickups=g.pickups.filter(k=>k.t>0);
  // particles / texts
  for(const q of g.parts){q.t-=dt;q.vy+=q.grav*dt;q.x+=q.vx*dt;q.y+=q.vy*dt;q.rot+=q.rs*dt;}
  g.parts=g.parts.filter(q=>q.t>0);
  for(const tx of g.texts){tx.t-=dt;tx.y-=34*dt;}
  g.texts=g.texts.filter(tx=>tx.t>0);
  // core pulse + decay
  g.core.flash=Math.max(0,g.core.flash-dt*2.5);g.core.pulse-=dt;
  if(g.core.pulse<=0&&g.core.hp>0){g.core.pulse=1.8;P(CORE.x,CORE.y,0,0,1,64,'rgba(127,233,220,0.5)',0,'ring');}
  g.shake=Math.max(0,g.shake-(g.shake*6+4)*dt);
}

let last=0;
function frame(ts){
  requestAnimationFrame(frame);
  const rdt=Math.min(0.05,(ts-last)/1000||0);last=ts;T+=rdt;
  let dt=rdt;
  if(G.freeze>0){G.freeze-=rdt;dt=0;}
  if(state==='playing')update(dt,true);
  else if(state==='dying'){
    update(dt*0.3,false);
    G.dieT-=rdt;
    if(G.dieT<=0){
      state='over';G.overT=0;
      G.newBest=G.score>best;
      if(G.newBest){best=G.score;try{localStorage.setItem('corefall_best',String(best));}catch(e){}}
    }
  }else if(state==='over'){G.overT+=rdt;}
  // ---- render ----
  ctx.setTransform(dpr,0,0,dpr,0,0);
  const cw=cv.width/dpr,ch=cv.height/dpr;
  ctx.fillStyle='#030b0e';ctx.fillRect(0,0,cw,ch);
  ctx.save();
  ctx.beginPath();ctx.rect(view.ox,view.oy,W*view.s,H*view.s);ctx.clip();
  ctx.setTransform(dpr*view.s,0,0,dpr*view.s,dpr*view.ox,dpr*view.oy);
  drawBackground();
  if(state==='menu'){drawMenu();}
  else{
    drawWorld(0);
    drawVignette();
    drawHUD();
    if(state==='dying'){
      ctx.fillStyle='rgba(255,60,40,'+(0.12+0.1*Math.sin(T*18)).toFixed(3)+')';ctx.fillRect(0,0,W,H);
    }
    if(state==='paused')drawPause();
    if(state==='over')drawOver();
  }
  ctx.restore();
  // frame line
  ctx.setTransform(dpr,0,0,dpr,0,0);
  ctx.strokeStyle='rgba(255,179,80,0.14)';ctx.lineWidth=1;
  ctx.strokeRect(view.ox+0.5,view.oy+0.5,W*view.s-1,H*view.s-1);
}
