try{ muted = localStorage.getItem('zombieGolfMute')==='1'; }catch(e){}
function ensureAudio(){
  if(actx) { if(actx.state==='suspended') actx.resume(); return; }
  try{
    const AC = window.AudioContext||window.webkitAudioContext;
    actx = new AC(); master = actx.createGain(); master.gain.value = muted?0:0.5; master.connect(actx.destination);
    noiseBuf = actx.createBuffer(1, actx.sampleRate, actx.sampleRate);
    const d = noiseBuf.getChannelData(0); for(let i=0;i<d.length;i++) d[i]=Math.random()*2-1;
  }catch(e){}
}
function tone(f,dur,o){ o=o||{}; if(!actx) return;
  const t0=actx.currentTime+(o.delay||0), osc=actx.createOscillator(), g=actx.createGain();
  osc.type=o.type||'sine'; osc.frequency.setValueAtTime(f,t0);
  if(o.slide) osc.frequency.exponentialRampToValueAtTime(Math.max(30,o.slide), t0+dur);
  g.gain.setValueAtTime(o.vol||.2, t0); g.gain.exponentialRampToValueAtTime(.001, t0+dur);
  osc.connect(g); g.connect(master); osc.start(t0); osc.stop(t0+dur+.03);
}
function noise(dur,o){ o=o||{}; if(!actx) return;
  const t0=actx.currentTime+(o.delay||0), src=actx.createBufferSource(), f=actx.createBiquadFilter(), g=actx.createGain();
  src.buffer=noiseBuf; f.type='bandpass'; f.frequency.setValueAtTime(o.freq||1000,t0); f.Q.value=o.q||1;
  if(o.slide) f.frequency.exponentialRampToValueAtTime(Math.max(40,o.slide), t0+dur);
  g.gain.setValueAtTime(o.vol||.2,t0); g.gain.exponentialRampToValueAtTime(.001,t0+dur);
  src.connect(f); f.connect(g); g.connect(master); src.start(t0); src.stop(t0+dur+.03);
}

  swing:p=>{ noise(.16,{vol:.28*p+.05,freq:900,slide:280}); tone(150,.07,{type:'triangle',vol:.14*p}); },
  bounce:v=>{ const q=clamp(v/6000,.03,.2); tone(120+v/12,.05,{type:'square',vol:q}); noise(.04,{vol:q*.7,freq:2200}); },
  thud:()=>{ tone(88,.1,{type:'sine',vol:.3,slide:48}); noise(.07,{vol:.22,freq:520}); },
  pop:()=>{ tone(420,.12,{type:'square',vol:.2,slide:90}); noise(.09,{vol:.26,freq:850,slide:220}); },
  sink:()=>{ tone(300,.08,{type:'sine',vol:.2,slide:140}); tone(660,.1,{type:'triangle',vol:.24,delay:.1}); tone(990,.15,{type:'triangle',vol:.2,delay:.2}); tone(1320,.18,{type:'sine',vol:.12,delay:.32}); },
  hurt:()=>{ tone(160,.3,{type:'sawtooth',vol:.3,slide:58}); noise(.22,{vol:.3,freq:320}); },
  wave:n=>[392,494,587].forEach((f,i)=>tone(f,.12,{type:'square',vol:.13,delay:i*.09})),
  clearW:()=>[523,659,784,1046].forEach((f,i)=>tone(f,.13,{type:'triangle',vol:.16,delay:i*.08})),
  over:()=>[330,262,196,147].forEach((f,i)=>tone(f,.3,{type:'triangle',vol:.22,delay:i*.2})),

function pickType(w){
  const wSp = w>=2 ? 4+w : 0, wBr = w>=3 ? 2+w*.8 : 0, tot=10+wSp+wBr;
  let r=Math.random()*tot;
  if(r<10) return 'sh'; r-=10;
  if(r<wSp) return 'sp'; r-=wSp;
  return 'br';
}
function spawnZombie(){
  const t=ZTYPES[pickType(G.wave)];
  let x,y,tries=0;
  do{
    const e=Math.floor(rand(0,4));
    if(e===0){x=rand(BND.l+30,BND.r-30); y=BND.t+26;}
    else if(e===1){x=rand(BND.l+30,BND.r-30); y=BND.b-26;}
    else if(e===2){x=BND.l+26; y=rand(BND.t+30,BND.b-30);}
    else {x=BND.r-26; y=rand(BND.t+30,BND.b-30);}
    tries++;
  }while(tries<12 && (Math.hypot(x-B.x,y-B.y)<150 || CRATES.some(c=>x>c.x-34&&x<c.x+c.w+34&&y>c.y-34&&y<c.y+c.h+34) || BUMPERS.some(b=>Math.hypot(x-b.x,y-b.y)<b.r+40)));
  const z = { type:Object.assign({},t), x:x, y:y, vx:0, vy:0, rot:Math.atan2(HOLE.y-y,HOLE.x-x),
    t:rand(0,10), ph:rand(0,TAU), wf:rand(2,4), hp:t.hp, flash:0, spawnT:.3, hitCd:0, bruised:false,
    state:'shamble', knockT:0, dead:false, gone:false, ambient:false, wt:0, wa:0 };
  zombies.push(z);
  for(let i=0;i<8;i++) parts.push({x:x,y:y,vx:rand(-50,50),vy:rand(-50,50),life:.5,max:.5,size:rand(2,4.5),col:'#7cb35a'});
}
function updZombie(z,dt){
  z.t+=dt; if(z.spawnT>0)z.spawnT-=dt; if(z.flash>0)z.flash-=dt; if(z.hitCd>0)z.hitCd-=dt;
  if(z.dead){
    const f=Math.exp(-2.2*dt); z.vx*=f; z.vy*=f; z.x+=z.vx*dt; z.y+=z.vy*dt; z.rot+=z.spin*dt; z.deadT+=dt;
    if(z.deadT>1 || z.x<BND.l-60||z.x>BND.r+60||z.y<BND.t-60||z.y>BND.b+60) z.gone=true;
    return;
  }
  if(z.state==='knocked'){
    const f=Math.exp(-3.5*dt); z.vx*=f; z.vy*=f; z.x+=z.vx*dt; z.y+=z.vy*dt; z.rot+=dt*6;
    z.knockT-=dt; if(z.knockT<=0) z.state='shamble';
    if(!z.ambient && Math.hypot(z.x-HOLE.x,z.y-HOLE.y)<HOLE.r+4+z.r*.3) eatFlag(z);
    return;
  }
  let ang, sp;
  if(z.ambient){
    z.wt-=dt;
    if(z.wt<=0 || z.x<BND.l+50||z.x>BND.r-50||z.y<BND.t+50||z.y>BND.b-50){ z.wa=rand(0,TAU); z.wt=rand(1.2,3); }
    ang=z.wa; sp=26;
  } else {
    const a=Math.atan2(HOLE.y-z.y,HOLE.x-z.x);
    ang=a+Math.sin(z.t*z.wf+z.ph)*.4;
    sp=z.sp*Math.min(1+(G.wave-1)*.06, 1.9);
  }
  z.x+=Math.cos(ang)*sp*dt; z.y+=Math.sin(ang)*sp*dt;
  /* obstacle repulsion */
  for(const c of CRATES){
    const px=clamp(z.x,c.x,c.x+c.w), py=clamp(z.y,c.y,c.y+c.h);
    const dx=z.x-px, dy=z.y-py, d=Math.hypot(dx,dy);
    if(d<z.r+16 && d>0.001){ z.x+=dx/d*(z.r+16-d)*.9; z.y+=dy/d*(z.r+16-d)*.9; }
  }
  for(const b of BUMPERS){
    const dx=z.x-b.x, dy=z.y-b.y, d=Math.hypot(dx,dy);
    if(d<b.r+z.r && d>0.001){ z.x+=dx/d*(b.r+z.r-d)*.9; z.y+=dy/d*(b.r+z.r-d)*.9; }
  }
  if(!z.ambient){ /* don't stack on the ball */
    const dx=z.x-B.x, dy=z.y-B.y, d=Math.hypot(dx,dy), m=z.r+B.r+2;
    if(d<m && d>0.001){ z.x+=dx/d*(m-d); z.y+=dy/d*(m-d); }
  }
  z.rot += angDiff(ang,z.rot)*Math.min(1,dt*5);
  if(!z.ambient && Math.hypot(z.x-HOLE.x,z.y-HOLE.y)<HOLE.r+4+z.r*.3) eatFlag(z);
}
function separate(){
  for(let i=0;i<zombies.length;i++) for(let j=i+1;j<zombies.length;j++){
    const a=zombies[i], b=zombies[j];
    if(a.state!=='shamble'||b.state!=='shamble') continue;
    const dx=b.x-a.x, dy=b.y-a.y, d=Math.hypot(dx,dy), m=a.r+b.r-2;
    if(d<m && d>0.001){ const push=(m-d)*.35; a.x-=dx/d*push; a.y-=dy/d*push; b.x+=dx/d*push; b.y+=dy/d*push; }
  }
}
function eatFlag(z){
  z.gone=true; G.hearts--; updateHearts(); hurtF=1; shake=Math.max(shake,16); flagWob=1.2;
  for(let i=0;i<14;i++) parts.push({x:HOLE.x,y:HOLE.y,vx:rand(-190,190),vy:rand(-190,190),life:rand(.3,.7),max:.7,size:rand(2.5,5.5),col:Math.random()<.5?'#8a2b2b':'#5d8f43'});
  SFX.hurt();
  if(G.hearts<=0) gameOver();
}

/* ============================== BALL PHYSICS ============================== */
const REST=.82;
function hitWalls(o){
  let hit=0, imp=0;
  if(o.x<BND.l+o.r){o.x=BND.l+o.r; if(o.vx>0){imp=Math.abs(o.vx); o.vx=-o.vx*REST;} hit=1;}
  if(o.x>BND.r-o.r){o.x=BND.r-o.r; if(o.vx<0){imp=Math.abs(o.vx); o.vx=-o.vx*REST;} hit=1;}
  if(o.y<BND.t+o.r){o.y=BND.t+o.r; if(o.vy>0){imp=Math.max(imp,Math.abs(o.vy)); o.vy=-o.vy*REST;} hit=1;}
  if(o.y>BND.b-o.r){o.y=BND.b-o.r; if(o.vy<0){imp=Math.max(imp,Math.abs(o.vy)); o.vy=-o.vy*REST;} hit=1;}
  return {hit:hit, imp:imp};
}
function hitCrate(o,c){
  const px=clamp(o.x,c.x,c.x+c.w), py=clamp(o.y,c.y,c.y+c.h);
  let dx=o.x-px, dy=o.y-py; const d2=dx*dx+dy*dy;
  if(d2>=o.r*o.r) return false;
  let d=Math.sqrt(d2), nx, ny;
  if(d<0.001){ /* center inside: push out shortest axis */
    const l=o.x-c.x, r=c.x+c.w-o.x, t=o.y-c.y, b=c.y+c.h-o.y;
    const m=Math.min(l,r,t,b);
    if(m===l){nx=-1;ny=0;} else if(m===r){nx=1;ny=0;} else if(m===t){nx=0;ny=-1;} else {nx=0;ny=1;}
    o.x+=nx*(o.r+1); o.y+=ny*(o.r+1);
  } else { nx=dx/d; ny=dy/d; o.x=px+nx*o.r; o.y=py+ny*o.r; }
  const dot=o.vx*nx+o.vy*ny;
  if(dot<0){ o.vx-=2*dot*nx*.85; o.vy-=2*dot*ny*.85; }
  return true;
}
function hitBumper(o,b){
  const dx=o.x-b.x, dy=o.y-b.y, d=Math.hypot(dx,dy), m=b.r+o.r;
  if(d>=m||d<0.001) return false;
  const nx=dx/d, ny=dy/d; o.x=b.x+nx*m; o.y=b.y+ny*m;
  const dot=o.vx*nx+o.vy*ny;
  if(dot<0){ o.vx-=2*dot*nx*.95; o.vy-=2*dot*ny*.95; }
  return true;
}
function simShot(vx,vy){
  const s={x:B.x,y:B.y,vx:vx,vy:vy,r:B.r}; const pts=[]; const dt=1/80;
  for(let i=0;i<150;i++){
    const f=Math.exp(-(0.55+(inSand(s.x,s.y)?3.2:0))*dt); s.vx*=f; s.vy*=f;
    s.x+=s.vx*dt; s.y+=s.vy*dt;
    hitWalls(s); CRATES.forEach(c=>hitCrate(s,c)); BUMPERS.forEach(b=>hitBumper(s,b));
    if(Math.hypot(s.vx,s.vy)<25) break;
    if(i%3===0) pts.push({x:s.x,y:s.y});
    if(Math.hypot(s.x-HOLE.x,s.y-HOLE.y)<HOLE.r-8 && Math.hypot(s.vx,s.vy)<560){ pts.push({x:HOLE.x,y:HOLE.y}); break; }
  }
  return {pts:pts, ex:s.x, ey:s.y};
}
function updBall(dt){
  if(B.respawn>0){
    B.respawn-=dt;
    if(B.respawn<=0){ B.x=TEE.x; B.y=TEE.y; B.vx=B.vy=0; B.sunk=false; B.rewarded=false;
      for(let i=0;i<10;i++){ const a=rand(0,TAU); parts.push({x:B.x,y:B.y,vx:Math.cos(a)*90,vy:Math.sin(a)*90,life:.4,max:.4,size:3,col:'#ffffff'}); }
    }
    return;
  }
  if(B.sunk){
    B.sinkT+=dt;
    B.x += (HOLE.x-B.x)*Math.min(1,dt*14); B.y += (HOLE.y-B.y)*Math.min(1,dt*14);
    if(B.sinkT>=.38 && !B.rewarded){
      B.rewarded=true;
      const bonus=100+G.wave*25; addScore(bonus, HOLE.x, HOLE.y-46, 'SUNK! +'+bonus, '#ffd43b', 27);
      if(G.hearts<3){ G.hearts++; updateHearts(); }
      for(let i=0;i<18;i++){ const a=rand(0,TAU), s=rand(60,240); parts.push({x:HOLE.x,y:HOLE.y,vx:Math.cos(a)*s,vy:Math.sin(a)*s,life:rand(.35,.7),max:.7,size:rand(2,4.5),col:'#ffd43b'}); }
      SFX.sink();
    }
    if(B.sinkT>1.05) B.respawn=.8;
    return;
  }
  const wasSp=Math.hypot(B.vx,B.vy);
  let f=Math.exp(-0.55*dt);
  if(inSand(B.x,B.y)) f*=Math.exp(-3.4*dt);
  B.vx*=f; B.vy*=f; B.x+=B.vx*dt; B.y+=B.vy*dt;
  const w=hitWalls(B); CRATES.forEach(c=>{if(hitCrate(B,c)) w.hit=1, w.imp=Math.max(w.imp,Math.hypot(B.vx,B.vy));}); BUMPERS.forEach(b=>{if(hitBumper(B,b)) w.hit=1, w.imp=Math.max(w.imp,Math.hypot(B.vx,B.vy));});
  if(w.hit && w.imp>150) SFX.bounce(w.imp);
  const sp2=Math.hypot(B.vx,B.vy);
  if(sp2<8){ B.vx=0; B.vy=0; }
  if(wasSp>140 && Math.hypot(B.vx,B.vy)<8) G.strokeKills=0; /* stroke ended */
  const dh=Math.hypot(B.x-HOLE.x,B.y-HOLE.y);
  if(dh<HOLE.r-8 && sp2<560){ B.sunk=true; B.sinkT=0; B.vx=B.vy=0; tone(300,.08,{type:'sine',vol:.18,slide:140}); }
  if(sp2>160){ B.trail.push({x:B.x,y:B.y}); if(B.trail.length>14) B.trail.shift(); }
  else if(B.trail.length && sp2<80) B.trail.shift();
  /* sand puffs */
  if(inSand(B.x,B.y) && sp2>300){
    parts.push({x:B.x+rand(-6,6), y:B.y+rand(-6,6), vx:-B.vx*.05+rand(-20,20), vy:-B.vy*.05+rand(-20,20), life:.45,max:.45,size:rand(2.5,5),col:'#d9b23a'});
  }
}
function ballHitZombies(dt){
  if(B.sunk||B.respawn>0) return;
  const sp=Math.hypot(B.vx,B.vy);
  if(sp<140) return;
  for(const z of zombies){
    if(z.dead||z.gone||z.hitCd>0) continue;
    const dx=z.x-B.x, dy=z.y-B.y, d=Math.hypot(dx,dy);
    if(d>=B.r+z.r||d<0.001) continue;
    const nx=dx/d, ny=dy/d;
    z.hp--; z.flash=.12; z.hitCd=.3;
    const imp=Math.min(sp,1400)*1.35;
    z.vx=nx*imp+B.vx*.3; z.vy=ny*imp+B.vy*.3;
    B.vx*=.6; B.vy*=.6;
    shake=Math.max(shake,4); SFX.thud();
    if(z.hp<=0) killZombie(z,nx,ny); else { z.state='knocked'; z.knockT=.55; z.bruised=true; }
  }
}
function killZombie(z,nx,ny){
  z.dead=true; z.deadT=0; z.spin=(Math.random()<.5?-1:1)*rand(9,16);
  z.vx=nx*rand(620,1000)+B.vx*.25; z.vy=ny*rand(620,1000)+B.vy*.25;
  G.kills++; G.strokeKills++;
  const mult=Math.min(G.strokeKills,6), pts=z.type.score*mult;
  addScore(pts, z.x, z.y-14, mult>1?('+'+pts+'  x'+mult):('+'+pts), mult>2?'#ffd43b':'#ffffff', 17+Math.min(mult*3,15));
  if(G.strokeKills===3||G.strokeKills===5||(G.strokeKills>=6&&G.strokeKills%4===0))
    pops.push({x:B.x,y:B.y-38,text:'COMBO x'+G.strokeKills+'!',life:1.1,max:1.1,size:34,col:'#ffd43b'});
  splats.push({x:z.x,y:z.y,r:z.r,rot:rand(0,TAU),life:9});
  if(splats.length>40) splats.shift();
  for(let i=0;i<12;i++) parts.push({x:z.x,y:z.y,vx:rand(-220,220),vy:rand(-220,220),life:rand(.3,.7),max:.7,size:rand(2.5,6),col:Math.random()<.6?'#8fce5f':'#4c7a35'});
  shake=Math.max(shake,7); SFX.pop();
}
function addScore(n,x,y,label,col,size){ G.score+=n; setScore(G.score); pops.push({x:x,y:y,text:label,life:1,max:1,size:size||18,col:col||'#fff'}); }

/* ============================== UPDATE ============================== */
function update(dt){
  shake*=Math.exp(-5*dt); if(shake<.2)shake=0;
  hurtF=Math.max(0,hurtF-dt*1.4); flagWob=Math.max(0,flagWob-dt);
  for(const p of parts){ p.life-=dt; p.x+=p.vx*dt; p.y+=p.vy*dt; const f=Math.exp(-3*dt); p.vx*=f; p.vy*=f; }
  parts=parts.filter(p=>p.life>0); if(parts.length>250) parts.splice(0,parts.length-250);
  for(const p of pops){ p.life-=dt; p.y-=42*dt; }
  pops=pops.filter(p=>p.life>0);
  for(const s of splats) s.life-=dt;
  splats=splats.filter(s=>s.life>0);

  if(G.mode==='play'){
    if(G.phase==='intro'){ G.phaseT-=dt; if(G.phaseT<=0){ G.phase='spawning'; G.spawnT=.4; } }
    else if(G.phase==='spawning'){
      const alive=zombies.filter(z=>!z.dead).length;
      if(G.toSpawn>0 && alive<14){
        G.spawnT-=dt;
        if(G.spawnT<=0){ spawnZombie(); G.toSpawn--; G.spawnT=Math.max(.5,1.7-G.wave*.08)*rand(.7,1.3); }
      }
      if(G.toSpawn===0 && zombies.filter(z=>!z.dead).length===0){
        G.phase='clear'; G.phaseT=1.7;
        const bonus=40*G.wave; addScore(bonus,HOLE.x,HOLE.y-90,'WAVE BONUS +'+bonus,'#ffe9a3',22);
        SFX.clearW();
      }
    } else if(G.phase==='clear'){ G.phaseT-=dt; if(G.phaseT<=0) startWave(G.wave+1); }
    for(const z of zombies) updZombie(z,dt);
    separate();
    zombies=zombies.filter(z=>!z.gone);
    updBall(dt); ballHitZombies(dt);
  } else {
    for(const z of zombies){ if(z.ambient||z.dead) updZombie(z,dt); }
    zombies=zombies.filter(z=>!z.gone);
    if(G.mode==='over' && !overShown && tG-overAt>.75){ overShown=true; overReady=true; overOv.classList.remove('hidden'); }
  }
}

function drawZombie(z){
  const s=Math.max(.2, z.spawnT>0 ? 1-z.spawnT/.3 : 1);
  let alpha=1; if(z.dead && z.deadT>.6) alpha=Math.max(0,1-(z.deadT-.6)/.4);
  if(alpha<=0) return;
  ctx.save(); ctx.globalAlpha=alpha;
  /* shadow */
  ctx.fillStyle='rgba(0,0,0,.25)';
  ctx.beginPath(); ctx.ellipse(z.x,z.y+z.r*.55,z.r*.95*s,z.r*.4*s,0,0,TAU); ctx.fill();
  const bob=(z.state==='shamble'&&!z.dead)?Math.sin(z.t*7)*1.6:0;
  ctx.translate(z.x, z.y+bob); ctx.rotate(z.rot); ctx.scale(s,s);
  const r=z.r, c=z.type.c, cd=z.type.cd;
  /* feet */
  const fp=Math.sin(z.t*9+z.ph)*3.5;
  ctx.fillStyle=cd;
  ctx.beginPath(); ctx.ellipse(-r*.72+fp, -r*.42, r*.3, r*.18, .3, 0, TAU); ctx.fill();
  ctx.beginPath(); ctx.ellipse(-r*.72-fp, r*.42, r*.3, r*.18, -.3, 0, TAU); ctx.fill();
  /* arms */
  const aw=Math.sin(z.t*6+z.ph)*3;
  ctx.strokeStyle=cd; ctx.lineWidth=r*.3; ctx.lineCap='round';
  ctx.beginPath(); ctx.moveTo(r*.25,-r*.4); ctx.lineTo(r*1.05,-(r*.42+aw)); ctx.stroke();
  ctx.beginPath(); ctx.moveTo(r*.25, r*.4); ctx.lineTo(r*1.05,(r*.42-aw)); ctx.stroke();
  ctx.fillStyle='#d8e6b0';
  ctx.beginPath(); ctx.arc(r*1.05,-(r*.42+aw), r*.17,0,TAU); ctx.fill();
  ctx.beginPath(); ctx.arc(r*1.05,(r*.42-aw), r*.17,0,TAU); ctx.fill();
  /* body */
  ctx.fillStyle=c; ctx.strokeStyle=cd; ctx.lineWidth=3;
  ctx.beginPath(); ctx.ellipse(0,0,r*.92,r,0,0,TAU); ctx.fill(); ctx.stroke();
  ctx.fillStyle='rgba(255,255,255,.14)';
  ctx.beginPath(); ctx.ellipse(-r*.15,-r*.1,r*.5,r*.6,0,0,TAU); ctx.fill();
  if(z.bruised){ ctx.strokeStyle='rgba(30,50,20,.7)'; ctx.lineWidth=2;
    ctx.beginPath(); ctx.moveTo(-r*.45,-r*.15); ctx.lineTo(-r*.1,0); ctx.lineTo(-r*.4,r*.2); ctx.stroke(); }
  /* face (front = +x) */
  const er=Math.max(3.4, r*.26);
  if(z.dead){
    ctx.strokeStyle='#1c2a14'; ctx.lineWidth=2.4;
    for(const sy of [-r*.3,r*.3]){
      ctx.beginPath(); ctx.moveTo(r*.34-er*.5, sy-er*.5); ctx.lineTo(r*.34+er*.5, sy+er*.5);
      ctx.moveTo(r*.34+er*.5, sy-er*.5); ctx.lineTo(r*.34-er*.5, sy+er*.5); ctx.stroke();
    }
  } else {
    ctx.fillStyle='#fff';
    ctx.beginPath(); ctx.arc(r*.42,-r*.3,er,0,TAU); ctx.fill();
    ctx.beginPath(); ctx.arc(r*.42, r*.3,er,0,TAU); ctx.fill();
    ctx.fillStyle= z.type.hp===2 && z.bruised ? '#ff5252' : '#1c2a14';
    ctx.beginPath(); ctx.arc(r*.42+er*.3,-r*.3,er*.42,0,TAU); ctx.fill();
    ctx.beginPath(); ctx.arc(r*.42+er*.3, r*.3,er*.42,0,TAU); ctx.fill();
    if(z.type.hp===2){ ctx.strokeStyle='#1c2a14'; ctx.lineWidth=2.6;
      ctx.beginPath(); ctx.moveTo(r*.2,-r*.55); ctx.lineTo(r*.62,-r*.42);
      ctx.moveTo(r*.2, r*.55); ctx.lineTo(r*.62, r*.42); ctx.stroke(); }
  }
  if(!z.dead){ ctx.fillStyle='#1c2a14'; rr(r*.62,-r*.1,r*.3,r*.2,r*.08); ctx.fill();
    ctx.fillStyle='#fff'; ctx.fillRect(r*.7,-r*.06,r*.07,r*.09); }
  if(z.flash>0){ ctx.globalAlpha=alpha*clamp(z.flash/.12,0,1)*.85; ctx.fillStyle='#fff';
    ctx.beginPath(); ctx.arc(0,0,r*1.05,0,TAU); ctx.fill(); }
  ctx.restore();
}
