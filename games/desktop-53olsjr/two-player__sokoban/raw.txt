/* ---------------- UPDATE / DRAW ---------------- */
function rrp(x,y,w,h,r){ctx.beginPath();ctx.moveTo(x+r,y);ctx.arcTo(x+w,y,x+w,y+h,r);
  ctx.arcTo(x+w,y+h,x,y+h,r);ctx.arcTo(x,y+h,x,y,r);ctx.arcTo(x,y,x+w,y,r);ctx.closePath();}

function update(dt){
  if(state==='play')levelTime+=dt;
  const k=1-Math.exp(-dt*22);
  P.forEach(w=>{w.rx+=(w.x-w.rx)*k;w.ry+=(w.y-w.ry)*k;});
  boxes.forEach(b=>{b.rx+=(b.x-b.rx)*k;b.ry+=(b.y-b.ry)*k;});
  for(let i=particles.length-1;i>=0;i--){const p=particles[i];p.life-=dt;
    if(p.life<=0){particles.splice(i,1);continue;}
    p.x+=p.vx*dt;p.y+=p.vy*dt;p.vy+=p.g*dt;p.rot+=p.vr*dt;}
  shake=Math.max(0,shake-dt*12);
}

function draw(){
  ctx.setTransform(dpr,0,0,dpr,0,0);
  const g=ctx.createLinearGradient(0,0,0,CH);g.addColorStop(0,'#0d1424');g.addColorStop(1,'#0a0f1a');
  ctx.fillStyle=g;ctx.fillRect(0,0,CW,CH);
  const sx=CW*(.5+.3*Math.sin(T*.13)),sy=CH*(.35+.2*Math.cos(T*.09));
  const rg=ctx.createRadialGradient(sx,sy,0,sx,sy,Math.max(CW,CH)*.5);
  rg.addColorStop(0,'rgba(255,190,110,.07)');rg.addColorStop(1,'rgba(0,0,0,0)');
  ctx.fillStyle=rg;ctx.fillRect(0,0,CW,CH);

  const pad=Math.min(CW,CH)*.06+14;
  const cell=Math.max(10,Math.min((CW-pad*2)/W,(CH-pad*2)/H));
  const ox=(CW-cell*W)/2,oy=(CH-cell*H)/2;
  ctx.save();
  if(shake>0)ctx.translate((Math.random()*2-1)*shake,(Math.random()*2-1)*shake);

  // dock slab
  rrp(ox-10,oy-10,W*cell+20,H*cell+20,14);
  ctx.fillStyle='rgba(3,7,14,.6)';ctx.fill();
  ctx.strokeStyle='rgba(90,120,180,.18)';ctx.lineWidth=1;ctx.stroke();

  // floor
  for(let y=0;y<H;y++)for(let x=0;x<W;x++){
    if(isWall(x,y))continue;
    const s=30+((x*7+y*13)%5)*2;
    ctx.fillStyle='rgb('+s+','+(s+8)+','+(s+22)+')';
    ctx.fillRect(ox+x*cell,oy+y*cell,cell,cell);
    ctx.strokeStyle='rgba(0,0,0,.16)';ctx.lineWidth=1;
    ctx.strokeRect(ox+x*cell+.5,oy+y*cell+.5,cell-1,cell-1);
    for(let k2=0;k2<3;k2++){
      const h=Math.sin(x*127.1+y*311.7+k2*74.7)*43758.5453,hf=h-Math.floor(h);
      ctx.fillStyle='rgba(242,234,216,.05)';
      ctx.fillRect(ox+(x+.15+hf*.7)*cell,oy+(y+.15+((hf*7)%1)*.7)*cell,1.5,1.5);}
  }

  // targets
  for(const key of targetSet){const p=key.split(','),x=+p[0],y=+p[1];
    const cx=ox+(x+.5)*cell,cy=oy+(y+.5)*cell,pulse=.5+.5*Math.sin(T*2.4+x+y);
    ctx.strokeStyle='rgba(255,180,84,'+(.35+.4*pulse)+')';ctx.lineWidth=Math.max(2,cell*.06);
    ctx.beginPath();ctx.arc(cx,cy,cell*(.3+.02*pulse),0,7);ctx.stroke();
    ctx.fillStyle='rgba(255,180,84,'+(.35+.3*pulse)+')';
    ctx.beginPath();ctx.arc(cx,cy,cell*.07,0,7);ctx.fill();}

  // boxes
  for(const b of boxes){
    const cx=ox+(b.rx+.5)*cell,cy=oy+(b.ry+.5)*cell;
    const st=T-b.landT,scl=1+.1*Math.max(0,1-st/.18);
    ctx.fillStyle='rgba(0,0,0,.32)';
    ctx.beginPath();ctx.ellipse(cx,cy+cell*.4,cell*.36,cell*.13,0,0,7);ctx.fill();
    const cs=cell*.8*scl,x=cx-cs/2,y=cy-cs/2;
    rrp(x,y,cs,cs,cs*.12);ctx.fillStyle='#c08a4b';ctx.fill();
    ctx.strokeStyle='rgba(70,42,15,.55)';ctx.lineWidth=Math.max(1,cs*.035);
    for(let i=1;i<3;i++){ctx.beginPath();ctx.moveTo(x+cs*.08,y+cs*i/3);ctx.lineTo(x+cs*.92,y+cs*i/3);ctx.stroke();}
    ctx.globalAlpha=.4;ctx.beginPath();ctx.moveTo(x+cs*.06,y+cs*.06);ctx.lineTo(x+cs*.94,y+cs*.94);
    ctx.moveTo(x+cs*.94,y+cs*.06);ctx.lineTo(x+cs*.06,y+cs*.94);ctx.stroke();ctx.globalAlpha=1;
    rrp(x,y,cs,cs,cs*.12);ctx.strokeStyle='#6b451f';ctx.lineWidth=Math.max(2,cell*.05);ctx.stroke();
    ctx.fillStyle='#f0d9a8';
    [[.14,.14],[.86,.14],[.14,.86],[.86,.86]].forEach(pt=>{ctx.beginPath();ctx.arc(x+cs*pt[0],y+cs*pt[1],cs*.05,0,7);ctx.fill();});
    if(onGoal(b.x,b.y)){
      ctx.save();ctx.shadowColor='rgba(255,190,90,.9)';ctx.shadowBlur=cell*.3;
      rrp(x,y,cs,cs,cs*.12);ctx.strokeStyle='#ffd27a';ctx.lineWidth=Math.max(2,cell*.06);ctx.stroke();ctx.restore();
      ctx.strokeStyle='rgba(58,30,4,.85)';ctx.lineWidth=Math.max(2,cs*.1);
      ctx.beginPath();ctx.moveTo(x+cs*.3,y+cs*.55);ctx.lineTo(x+cs*.45,y+cs*.72);ctx.lineTo(x+cs*.72,y+cs*.32);ctx.stroke();}
    if(st<.5){ctx.strokeStyle='rgba(255,196,110,'+(1-st/.5)+')';ctx.lineWidth=2;
      ctx.beginPath();ctx.arc(cx,cy,cell*(.35+st*1.4),0,7);ctx.stroke();}
  }

  // workers
  const cols=[['#ff5c49','#a3271b'],['#3ad6e8','#0f7c8c']];
  for(const w of P){
    let bxo=0,byo=0;const bt=T-w.bumpT;
    if(bt>=0&&bt<.18){const q=Math.sin(bt/.18*Math.PI);bxo=w.fx*q*cell*.14;byo=w.fy*q*cell*.14;}
    const stp=T-w.stepT,sc=1+.12*Math.max(0,1-stp/.16);
    const bob=(stp>.25)?Math.sin(T*3+w.i*2.7)*cell*.02:0;
    const cx=ox+(w.rx+.5)*cell+bxo,cy=oy+(w.ry+.5)*cell+byo+bob;
    ctx.fillStyle='rgba(0,0,0,.32)';
    ctx.beginPath();ctx.ellipse(cx,oy+(w.ry+.5)*cell+cell*.36,cell*.3,cell*.1,0,0,7);ctx.fill();
    const bw=cell*.56*sc,bh=cell*.6*sc,x=cx-bw/2,y=cy-bh/2;
    rrp(x,y,bw,bh,bw*.38);ctx.fillStyle=cols[w.i][0];ctx.fill();
    ctx.strokeStyle=cols[w.i][1];ctx.lineWidth=Math.max(1.5,cell*.045);ctx.stroke();
    rrp(x+bw*.12,y+bh*.16,bw*.76,bh*.3,bw*.18);ctx.fillStyle='#ffe9c9';ctx.fill();
    ctx.fillStyle='#1b2333';const ex=cx+w.fx*cell*.06,ey=y+bh*.31+w.fy*cell*.04;
    ctx.beginPath();ctx.arc(ex-bw*.15,ey,cell*.045,0,7);ctx.fill();
    ctx.beginPath();ctx.arc(ex+bw*.15,ey,cell*.045,0,7);ctx.fill();
    ctx.fillStyle='rgba(242,234,216,.9)';ctx.fillRect(cx-bw*.18,y-bh*.06,bw*.36,bh*.12);
    const fs=Math.max(9,cell*.17),tag=(w.i?'P2':'P1');
    ctx.font='bold '+fs+'px "Courier New",monospace';ctx.textAlign='center';
    const tw=ctx.measureText(tag).width,ty=y+bh+cell*.16;
    rrp(cx-tw/2-4,ty-fs*.8,tw+8,fs*1.35,3);ctx.fillStyle='rgba(10,14,22,.72)';ctx.fill();
    ctx.fillStyle=cols[w.i][0];ctx.fillText(tag,cx,ty+fs*.25);
  }

  // particles
  for(const p of particles){
    const a=Math.max(0,p.life/p.max),px=ox+p.x*cell,py=oy+p.y*cell;
    ctx.globalAlpha=a;
    if(p.shape==='c'){ctx.save();ctx.translate(px,py);ctx.rotate(p.rot);ctx.fillStyle=p.col;
      const s=p.size*cell;ctx.fillRect(-s/2,-s/2,s,s*.6);ctx.restore();}
    else{ctx.fillStyle=p.col;ctx.beginPath();ctx.arc(px,py,p.size*cell,0,7);ctx.fill();}
  }
  ctx.globalAlpha=1;
  ctx.restore();

  if(state!=='start')hudTime.textContent=fmt(levelTime);
}

let last=performance.now();
function frame(ts){const dt=Math.min(.05,(ts-last)/1000);last=ts;T=ts/1000;update(dt);draw();requestAnimationFrame(frame);}
