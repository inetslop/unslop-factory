let AC=null,master=null,muted=false,noiseBuf=null;
function ensureAudio(){
  if(!AC){const A=window.AudioContext||window.webkitAudioContext;if(!A)return;
    AC=new A();master=AC.createGain();master.gain.value=muted?0:.5;master.connect(AC.destination);
    noiseBuf=AC.createBuffer(1,AC.sampleRate,AC.sampleRate);const d=noiseBuf.getChannelData(0);
    for(let i=0;i<d.length;i++)d[i]=Math.random()*2-1;}
  if(AC.state==='suspended')AC.resume();
}
function tone(f0,f1,dur,type,vol,at){at=at||0;if(!AC||muted)return;const t=AC.currentTime+at,o=AC.createOscillator(),g=AC.createGain();
  o.type=type;o.frequency.setValueAtTime(Math.max(1,f0),t);o.frequency.exponentialRampToValueAtTime(Math.max(1,f1),t+dur);
  g.gain.setValueAtTime(vol,t);g.gain.exponentialRampToValueAtTime(.0001,t+dur);o.connect(g);g.connect(master);o.start(t);o.stop(t+dur+.05);}
function noise(dur,vol,f0,f1,at){at=at||0;if(!AC||muted)return;const t=AC.currentTime+at,s=AC.createBufferSource(),g=AC.createGain(),f=AC.createBiquadFilter();
  s.buffer=noiseBuf;s.loop=true;f.type='bandpass';f.Q.value=1;
  f.frequency.setValueAtTime(f0,t);f.frequency.exponentialRampToValueAtTime(Math.max(20,f1),t+dur);
  g.gain.setValueAtTime(vol,t);g.gain.exponentialRampToValueAtTime(.0001,t+dur);
  s.connect(f);f.connect(g);g.connect(master);s.start(t);s.stop(t+dur+.05);}

/* ================= game flow ================= */
function newGame(){G={state:'serve',hole:1,score:0,lives:5,strokes:0,results:[],holeT:0,ovShown:false,deadT:0};bricks=[];parts=[];pops=[];buildHole();resetBall();hideAll();}
function buildHole(){const def=HOLE_DEFS[G.hole-1];bricks=[];let id=0;
  def.rows.forEach((row,r)=>{for(let c=0;c<13;c++){const ch=row[c]||'.';if(ch==='.')continue;
    let hp=1;if(ch==='T')hp=2;else if(ch==='S')hp=3;
    bricks.push({x:MGX+c*(BRW+GAP),y:TOPY+r*(BRH+GAP),type:ch,hp,alive:true,flash:0,id:id++});}});
  P.w=Math.max(92,126-(G.hole-1)*4);P.x=W/2;}
function resetBall(){B.x=P.x;B.y=P.y-B.r-8;B.vx=0;B.vy=0;B.px=B.x;B.py=B.y;B.trail.length=0;B.glow=0;}
function resWord(d){if(d<=-3)return'ALBATROSS';if(d===-2)return'EAGLE';if(d===-1)return'BIRDIE';if(d===0)return'PAR';if(d===1)return'BOGEY';return'+'+d;}
function fmtDiff(d){return d===0?'E':(d>0?'+'+d:''+d);}

function launch(){const ax=P.x,ay=P.y-14;let ang=-Math.PI/2+rnd(-.15,.15);
  if(pointer.has)ang=clamp(Math.atan2(pointer.y-ay,pointer.x-ax),-Math.PI+.45,-.45);
  const sp=560;B.vx=Math.cos(ang)*sp;B.vy=Math.sin(ang)*sp;
  G.strokes++;strokeFlash=1;G.state='play';SFX.serve();P.swingT=.3;P.hitSwing=true;turf(P.x,P.y);}
function trySwing(){if(P.cd>0)return;P.swingT=.3;P.cd=1.05;P.hitSwing=false;SFX.whoosh();}
function advance(){if(document.activeElement&&document.activeElement.blur)document.activeElement.blur();
  if(!G)return;
  if(G.state==='menu')newGame();
  else if(G.state==='serve')launch();
  else if(G.state==='play')trySwing();
  else if(G.state==='holeout'&&G.ovShown)nextHole();
  else if(G.state==='over'||G.state==='done')newGame();
  else if(G.state==='paused')togglePause();}
function togglePause(){if(G.state==='play'||G.state==='serve'){G.prev=G.state;G.state='paused';showOv($('ov-pause'));}
  else if(G.state==='paused'){G.state=G.prev||'serve';hideAll();}}
function toMenu(){G.state='menu';updateBestLabels();hideAll();showOv($('ov-menu'));}

function holeOut(){if(G.state!=='play')return;const par=PARS[G.hole-1];const d=G.strokes-par;
  const bonus=Math.max(0,-d)*100+200;G.score+=bonus;
  G.results.push({par,st:G.strokes,word:resWord(d),d});
  B.vx*=.25;B.vy=-380;SFX.fanfare();G.state='holeout';G.holeT=0;G.ovShown=false;}
function showHoleOv(){const r=G.results[G.results.length-1];
  const w=$('ho-word');w.textContent=r.word;w.className='result '+(r.d<0?'u':r.d>0?'o':'p');
  $('ho-detail').textContent='Strokes '+r.st+' \u00b7 Par '+r.par+' \u2014 Hole '+G.hole+'/9';
  $('ho-bonus').textContent='+'+(Math.max(0,-r.d)*100+200).toLocaleString()+' hole bonus \u00b7 Total '+G.score.toLocaleString();
  showOv($('ov-hole'));}
function nextHole(){hideAll();G.hole++;if(G.hole>HOLE_DEFS.length){finishRound();return;}
  buildHole();G.strokes=0;resetBall();G.state='serve';}
function finishRound(){G.state='done';const best=loadBest();const nb=G.score>best;if(nb)saveBest(G.score);
  let html='',tp=0,ts=0;
  G.results.forEach((r,i)=>{tp+=r.par;ts+=r.st;

  html+='<tr class="tot"><td>TOT</td><td>'+tp+'</td><td>'+ts+'</td><td class="'+(ts-tp<0?'u':ts-tp>0?'o':'')+'">'+fmtDiff(ts-tp)+'</td></tr>';

/* ================= effects helpers ================= */
function addPop(x,y,txt,col,size){if(pops.length>40)pops.shift();pops.push({x:clamp(x,36,W-36),y,txt,col,t:0,life:.95,size:size||15});}
function shardBurst(bxp,byp,colA,colB,n){for(let i=0;i<n;i++)parts.push({kind:'shard',x:bxp,y:byp,vx:rnd(-180,180),vy:rnd(-240,40),g:900,life:rnd(.4,.75),max:.75,size:rnd(3,6.5),col:Math.random()<.5?colA:colB,rot:rnd(0,6),vr:rnd(-9,9)});}
function dotBurst(x,y,n,col,g,spd){for(let i=0;i<n;i++)parts.push({kind:'dot',x,y,vx:rnd(-spd,spd),vy:rnd(-spd,-30),g:rnd(g*.7,g*1.2),life:rnd(.3,.6),max:.6,size:rnd(2,3.5),col});}
function turf(x,y){for(let i=0;i<8;i++)parts.push({kind:'dot',x:x+rnd(-14,14),y,vx:rnd(-140,140),vy:rnd(-220,-40),g:800,life:rnd(.3,.55),max:.55,size:rnd(2,3.5),col:Math.random()<.5?'#4fae6a':'#3c8f57'});}
function burstConfetti(){const cols=['#f5c84c','#ffffff','#7ed9a0','#ff8d7a','#6fb7ff'];
  for(let i=0;i<90;i++)parts.push({kind:'conf',x:rnd(40,W-40),y:rnd(-70,10),vx:0,vy:rnd(55,140),life:rnd(1.6,2.8),max:2.8,size:rnd(4,7),col:cols[i%5],rot:rnd(0,6),vr:rnd(-7,7),ph:rnd(0,6),conf:true});}
function slowBall(f){let sp=Math.hypot(B.vx,B.vy)*f;if(sp<400)sp=400;const l=Math.hypot(B.vx,B.vy)||1;B.vx=B.vx/l*sp;B.vy=B.vy/l*sp;}

/* ================= physics ================= */
function circleRect(b,r){const cx=clamp(b.x,r.x,r.x+BRW),cy=clamp(b.y,r.y,r.y+BRH);const dx=b.x-cx,dy=b.y-cy;return dx*dx+dy*dy<=b.r*b.r;}
function stepBall(dt){
  const dist=Math.hypot(B.vx,B.vy)*dt,n=Math.max(1,Math.ceil(dist/6));
  for(let i=0;i<n;i++){
    B.px=B.x;B.py=B.y;B.x+=B.vx*dt/n;B.y+=B.vy*dt/n;
    if(B.x<B.r){B.x=B.r;B.vx=Math.abs(B.vx);SFX.wall();}
    else if(B.x>W-B.r){B.x=W-B.r;B.vx=-Math.abs(B.vx);SFX.wall();}
    if(B.y<B.r){B.y=B.r;B.vy=Math.abs(B.vy);SFX.wall();}
    const pt=P.y;
    if(B.vy>0&&B.y+B.r>=pt&&B.py+B.r<=pt+18&&Math.abs(B.x-P.x)<=P.w/2+B.r){
      B.y=pt-B.r;let sp=Math.min(MAXSP,Math.hypot(B.vx,B.vy)*1.035+4);
      const off=clamp((B.x-P.x)/(P.w/2),-1,1),ang=off*1.05;
      B.vx=Math.sin(ang)*sp;B.vy=-Math.cos(ang)*sp;
      G.strokes++;strokeFlash=1;SFX.tock();P.swingT=.3;P.hitSwing=true;turf(B.x,pt);
    }
    for(let k=0;k<bricks.length;k++){const bk=bricks[k];if(bk.alive&&circleRect(B,bk)){resolveBrick(bk);break;}}
    if(G.state!=='play')return;
    if(B.y-B.r>H+12){loseBall();return;}
  }
}
function resolveBrick(b){
  const cx=clamp(B.x,b.x,b.x+BRW),cy=clamp(B.y,b.y,b.y+BRH);
  let dx=B.x-cx,dy=B.y-cy;
  if(dx===0&&dy===0){B.y=b.y-B.r-.1;B.vy=-Math.abs(B.vy);}
  else{const adx=Math.abs(dx),ady=Math.abs(dy);
    if(adx>ady){B.vx=dx>0?Math.abs(B.vx):-Math.abs(B.vx);B.x=dx>0?b.x+BRW+B.r:b.x-B.r;}
    else{B.vy=dy>0?Math.abs(B.vy):-Math.abs(B.vy);B.y=dy>0?b.y+BRH+B.r:b.y-B.r;}}
  if(Math.abs(B.vy)<55)B.vy=(B.vy<=0?-1:1)*55;
  B.vx+=(Math.random()-.5)*26;
  const bxp=b.x+BRW/2,byp=b.y+BRH/2;
  if(b.type==='W'){SFX.brick('W');G.score=Math.max(0,G.score-50);addPop(bxp,byp,'-50','#8fc6ff',14);slowBall(.62);shake=Math.max(shake,5);dotBurst(bxp,byp,10,'#7fb8ea',700,160);return;}
  if(b.type==='R'){SFX.brick('R');dotBurst(B.x,B.y,5,'#cfd8de',600,140);return;}
  b.hp--;b.flash=.12;
  if(b.hp>0){if(b.type==='S'){slowBall(.85);addPop(bxp,byp-10,'SLOW!','#ffe9a8',13);}
    SFX.brick(b.type);shardBurst(B.x,B.y,'#e9cd72','#d4b45c',3);}
  else{b.alive=false;SFX.brick(b.type);
    const pts=b.type==='G'?40:b.type==='T'?70:b.type==='S'?90:200;
    G.score+=pts;addPop(bxp,byp,'+'+pts,b.type==='M'?'#ffd76a':'#eafff0',15);
    if(b.type==='G')shardBurst(bxp,byp,'#4fae6a','#3c8f57',8);
    else if(b.type==='T')shardBurst(bxp,byp,'#b5813f','#96682f',8);
    else if(b.type==='S')shardBurst(bxp,byp,'#e9cd72','#d4b45c',9);
    else{shardBurst(bxp,byp,'#f2b53a','#ffd76a',10);dotBurst(bxp,byp,8,'#fff2c0',400,120);}
    let left=false;for(const bb of bricks)if(bb.alive&&bb.type!=='W'&&bb.type!=='R'){left=true;break;}
    if(!left)holeOut();
  }
}
function swingHit(){if(P.swingT<=0||P.hitSwing)return;
  const dx=B.x-P.x,dy=B.y-(P.y-24);
  if(dx*dx+dy*dy<85*85){P.hitSwing=true;
    const s=Math.min(900,Math.hypot(B.vx,B.vy)*1.32);
    let ux=B.vx,uy=B.vy;const l=Math.hypot(ux,uy)||1;ux/=l;uy/=l;if(uy>-.5)uy=-.5;
    const n2=Math.hypot(ux,uy);B.vx=ux/n2*s;B.vy=uy/n2*s;
    B.glow=.35;shake=9;SFX.smash();addPop(B.x,B.y-16,'SMASH!','#f5c84c',20);
    dotBurst(B.x,B.y,14,'#ffe9a8',700,260);}}
function loseBall(){G.lives--;SFX.lose();shake=7;G.state='dead';G.deadT=1.1;if(B.vy<260)B.vy=260;
  addPop(clamp(B.x,50,W-50),H-90,'BALL LOST','#ff8d7a',16);}

/* ================= update ================= */
function update(dt,t){
  const pad=ox/scale+220;
  for(const c of clouds){c.x+=c.v*dt;if(c.x>W/2+pad)c.x=-W/2-pad+rnd(0,150);}
  shake=Math.max(0,shake-shake*7*dt-2*dt);
  strokeFlash=Math.max(0,strokeFlash-dt*3);
  B.glow=Math.max(0,B.glow-dt*2);
  for(const b of bricks)if(b.flash>0)b.flash-=dt;
  for(let i=parts.length-1;i>=0;i--){const p=parts[i];p.t+=dt;p.life-=dt;
    if(p.life<=0){parts.splice(i,1);continue;}
    if(p.conf){p.x+=Math.sin(p.t*5+p.ph)*70*dt;p.y+=p.vy*dt;p.rot+=p.vr*dt;}
    else{p.x+=p.vx*dt;p.y+=p.vy*dt;if(p.g)p.vy+=p.g*dt;p.rot+=p.vr*dt;}}
  if(parts.length>420)parts.splice(0,parts.length-420);
  for(let i=pops.length-1;i>=0;i--){const q=pops[i];q.t+=dt;q.y-=34*dt;if(q.t>q.life)pops.splice(i,1);}
  if(!G)return;
  const dir=(keys.r?1:0)-(keys.l?1:0);
  P.x+=dir*480*dt;
  if(!dir&&pointer.has)P.x+=(pointer.x-P.x)*Math.min(1,dt*16);
  P.x=clamp(P.x,P.w/2+8,W-P.w/2-8);
  P.swingT=Math.max(0,P.swingT-dt);P.cd=Math.max(0,P.cd-dt);
  if(G.state==='serve'||G.state==='menu'){B.x=P.x;B.y=P.y-B.r-8;}
  else if(G.state==='play'){B.trail.unshift({x:B.x,y:B.y});if(B.trail.length>14)B.trail.pop();stepBall(dt);if(G.state==='play')swingHit();}
  else if(G.state==='dead'){B.vy+=1500*dt;B.y+=B.vy*dt;G.deadT-=dt;
    if(G.deadT<=0){if(G.lives>0){resetBall();G.state='serve';}else gameOver();}}
  else if(G.state==='holeout'){B.x+=B.vx*dt;B.y+=B.vy*dt;G.holeT+=dt;
    if(G.holeT>.5&&!G.ovShown){G.ovShown=true;showHoleOv();burstConfetti();}}
}
