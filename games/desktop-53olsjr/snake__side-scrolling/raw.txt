const diff=()=>Math.min(1,dist/12000);
function addGSpikes(sx){ hazards.push({x:sx,kind:'g',h:74,s:26,circles:[{y:GROUND-26,r:19},{y:GROUND-52,r:12}]}); }
function addTSpikes(sx,h){ hazards.push({x:sx,kind:'t',h,s:26,circles:[{y:h*.42,r:Math.min(23,26)},{y:h*.78,r:14}]}); }
function pickPattern(d){
  const o=[['food',4],['gspike',5]];
  if(d>.1)o.push(['stal',3]);
  if(d>.2)o.push(['pit',4]);
  if(d>.32)o.push(['gate',3.5]);
  if(d>.4)o.push(['thorn',3.5]);
  if(d>.55)o.push(['pitSpike',3.5]);
  let tot=0; for(const x of o)tot+=x[1]; let r=rnd()*tot;
  for(const x of o){ r-=x[1]; if(r<=0)return x[0]; } return 'food';
}
function spawnChunk(){
  const d=diff(); let x=genX, w=0;
  let pat = genX<950 ? (rnd()<.6?'food':'gspike') : pickPattern(d);
  if(pat==='food'){
    const y=240+rnd()*190, n=3+(rnd()<.5?1:0), mid=(n-1)>>1;
    for(let i=0;i<n;i++) orbs.push({x:x+i*64,y:y+Math.sin(i)*8,r:11,gold:false,ph:rnd()*6});
    if(rnd()<.3){ const o=orbs[orbs.length-n+mid]; o.gold=true; o.r=13; }
    w=(n-1)*64;
  } else if(pat==='gspike'){
    let n=1+((d>.25&&rnd()<.7)?1:0)+((d>.5&&rnd()<.5)?1:0);
    for(let i=0;i<n;i++) addGSpikes(x+i*56); w=(n-1)*56+40;
  } else if(pat==='stal'){
    addTSpikes(x,110+rnd()*70+d*40);
    if(d>.5&&rnd()<.4) addTSpikes(x+64,90+rnd()*50); w=64;
  } else if(pat==='pit'){
    const pw=Math.round(lerp(150,270,d)+rnd()*60); pits.push({x,w:pw});
    const n=5, amp=130+d*50;
    for(let i=0;i<n;i++){ const f=i/(n-1);
      orbs.push({x:x+pw*f,y:GROUND-60-Math.sin(f*Math.PI)*amp,r:11,gold:(i===2&&rnd()<.4),ph:rnd()*6}); }
    w=pw;
  } else if(pat==='gate'){
    const gapT=lerp(255,190,d)+rnd()*25; let h=GROUND-74-gapT; if(h<80)h=80;
    addTSpikes(x,h); addGSpikes(x+26);
    orbs.push({x:x+13,y:(h+GROUND-74)/2,r:11,gold:false,ph:rnd()*6}); w=90;
  } else if(pat==='thorn'){
    stingers.push({x:x+50,baseY:230+rnd()*230,amp:44+rnd()*55,sp:1.5+rnd()*1.3,ph:rnd()*6,y:0}); w=100;
  } else { // pitSpike
    const pw=Math.round(140+rnd()*70); pits.push({x,w:pw});
    addGSpikes(x+pw+80); if(d>.75) addGSpikes(x+pw+136);
    w=pw+(d>.75?196:140);
  }
  genX=x+w+lerp(470,330,d)+rnd()*170;
}
function solidAt(x){ for(const p of pits) if(x>p.x&&x<p.x+p.w) return false; return true; }

/* ============================== SIM ============================== */
function stepHead(dt,up,down){
  let a=G-(up?THRUST:0)+(down?DIVE:0);
  head.vy+=a*dt; head.vy-=head.vy*.35*dt;
  head.vy=clamp(head.vy,VU,VD);
  head.y+=head.vy*dt;
  if(head.y<HEAD_R+2){ head.y=HEAD_R+2; if(head.vy<0)head.vy=0; }
  if(solidAt(head.x)&&head.y>=GROUND-HEAD_R-.5){ head.y=GROUND-HEAD_R; if(head.vy>0)head.vy=0; }
  head.x+=speed*dt;
}
function sampleBody(){
  bodyPts.length=0;
  const n=Math.min(48,Math.floor(bodyLen/SPACING)+1);
  bodyPts.push({x:head.x,y:head.y});
  let acc=0, idx=trail.length-1;
  for(let i=1;i<n&&idx>=0;i++){
    const target=i*SPACING;
    while(idx>=0){
      const p=trail[idx], q=(idx===trail.length-1)?head:trail[idx+1];
      const seg=Math.hypot(q.x-p.x,q.y-p.y);
      if(seg<.0001){ idx--; continue; }
      if(acc+seg>=target){ const f=(target-acc)/seg;
        bodyPts.push({x:q.x+(p.x-q.x)*f,y:q.y+(p.y-q.y)*f}); break; }
      acc+=seg; idx--;
    }
  }
  while(bodyPts.length<n){ const l=bodyPts[bodyPts.length-1]; bodyPts.push({x:l.x-SPACING*.6,y:l.y}); }
  // prune trail
  let a2=0;
  for(let i=trail.length-1;i>=0;i--){
    const q=i===trail.length-1?head:trail[i+1];
    a2+=Math.hypot(q.x-trail[i].x,q.y-trail[i].y);
    if(a2>bodyLen+70){ trail.splice(0,i+1); break; }
  }
}
function hitCircle(cx,cy,cr,hx,hy,hr){ const dx=cx-hx,dy=cy-hy,r=cr+hr; return dx*dx+dy*dy<r*r; }
function checkCollisions(){
  for(const h of hazards){ if(h.x<camX-80||h.x>camX+vw+80)continue;
    for(const c of h.circles){ if(hitCircle(head.x,head.y,HEAD_R-2,h.x,c.y,c.r)) return 'spike'; } }
  for(const st of stingers){ const y=st.baseY+Math.sin(t*st.sp+st.ph)*st.amp; st.y=y;
    if(st.x>camX-100&&st.x<camX+vw+100&&hitCircle(head.x,head.y,HEAD_R-2,st.x,y,25)) return 'thorn'; }
  if(!solidAt(head.x)&&head.y+HEAD_R>=LAVA) return 'lava';
  const n=bodyPts.length;
  for(let i=4;i<n;i+=3){
    const p=bodyPts[i], rad=Math.max(6,HEAD_R*(1-.55*i/(n-1)))-3;
    if(!solidAt(p.x)&&p.y+rad>=LAVA) return 'lava';
    for(const h of hazards){ if(h.x<camX-80||h.x>camX+vw+80)continue;
      for(const c of h.circles){ if(hitCircle(p.x,p.y,rad,h.x,c.y,c.r)) return 'spike'; } }
    for(const st of stingers){ if(st.x>camX-120&&st.x<camX+vw+120&&hitCircle(p.x,p.y,rad,st.x,st.y,25)) return 'thorn'; }
  }
  return null;
}
function burst(x,y,col,n,pow){ for(let i=0;i<n;i++){ const a=rnd()*Math.PI*2,v=(.3+rnd()*.7)*pow;
  parts.push({x,y,vx:Math.cos(a)*v,vy:Math.sin(a)*v-80,life:1,col,r:2+rnd()*4}); } }

function die(kind){
  setState('dying'); dyingT=0; shake=16; sfx.die();
  const cols={spike:'#3fe07f',thorn:'#ff5a4e',lava:'#ffb347'}[kind];
  burst(head.x,head.y,cols[kind],26,420); burst(head.x,head.y,'#ffffff',10,260);
}
function finishDeath(){
  const causes={spike:'Impaled on the spikes',thorn:'Pierced by a thornball',lava:'Sizzled in the lava'};
  causeEl.textContent=causes[deathCause]||'The swamp wins';
  oScore.textContent=score; oLen.textContent=bodyPts.length;
  const nb=score>best; if(nb){ best=score; saveBest(); }
  oBest.textContent=best; newBest.classList.toggle('hidden',!nb);
  overAt=Date.now(); setState('over');
}
let deathCause='spike';

function updateParticles(dt){
  for(let i=parts.length-1;i>=0;i--){ const p=parts[i];
    p.vy+=G*.5*dt; p.x+=p.vx*dt; p.y+=p.vy*dt; p.life-=dt*1.5; if(p.life<=0)parts.splice(i,1); }
  for(let i=texts.length-1;i>=0;i--){ const x=texts[i]; x.y-=42*dt; x.life-=dt*.9; if(x.life<=0)texts.splice(i,1); }
  for(let i=wind.length-1;i>=0;i--){ const w=wind[i]; w.x-=w.v*dt; if(w.x<camX-250)wind.splice(i,1); }
}
function cleanWorld(){
  while(pits.length&&pits[0].x+pits[0].w<camX-400)pits.shift();
  while(hazards.length&&hazards[0].x+60<camX-400)hazards.shift();
  while(stingers.length&&stingers[0].x+80<camX-400)stingers.shift();
  while(orbs.length&&orbs[0].x+20<camX-400)orbs.shift();
}

function updatePlaying(dt){
  stepHead(dt,kUp||pUp,kDown);
  dist+=speed*dt; speed=SPD0+(SPD1-SPD0)*diff();
  trail.push({x:head.x,y:head.y}); sampleBody();
  if(genX<camX+vw+600) spawnChunk(); cleanWorld();
  for(const st of stingers){ st.y=st.baseY+Math.sin(t*st.sp+st.ph)*st.amp; }
  // eat
  for(let i=orbs.length-1;i>=0;i--){ const o=orbs[i];
    const dx=o.x-head.x,dy=o.y-head.y,rr=o.r+HEAD_R+4;
    if(dx*dx+dy*dy<rr*rr){
      orbs.splice(i,1); score+=o.gold?25:10; eaten++;
      bodyLen=Math.min(MAX_LEN,bodyLen+GROW*(o.gold?2:1));
      burst(o.x,o.y,o.gold?'#ffe27a':'#ffbe55',o.gold?16:10,240);
      texts.push({x:o.x,y:o.y-14,txt:'+'+(o.gold?25:10),life:1,gold:o.gold});
      o.gold?sfx.gold():sfx.eat();
      if(eaten%10===0){ sfx.chime(); scoreEl.classList.remove('pop'); void scoreEl.offsetWidth; scoreEl.classList.add('pop'); }
    } }
  // wind streaks
  if(speed>380){ windTimer-=dt; if(windTimer<=0){ windTimer=.16+rnd()*.25;
    wind.push({x:camX+vw+50,y:30+rnd()*(GROUND-90),len:60+rnd()*90,v:speed*(1.4+rnd())}); } }
  updateParticles(dt);
  const hit=checkCollisions(); if(hit){ deathCause=hit; die(hit); return; }
  camX=head.x-vw*.3;
}

function updateDying(rdt){
  const dt=rdt*.35;
  stepHead(dt,false,true); head.vy+=G*dt;
  trail.push({x:head.x,y:head.y}); sampleBody();
  updateParticles(dt);
  camX=head.x-vw*.3; dyingT+=rdt;
  if(dyingT>.95) finishDeath();
}

function ridge(par,baseY,amp,period,seed,col){
  const step=26, ox=camX*par;
  ctx.fillStyle=col; ctx.beginPath();
  let wx0=Math.floor(ox/step)*step-step;
  ctx.moveTo(wx0-ox,LH);
  for(let wx=wx0;wx<=ox+vw+step*2;wx+=step){
    const n=wx/period,i=Math.floor(n),f=n-i,u=(1-Math.cos(f*Math.PI))/2;
    const y=baseY+(lerp(hash(i,seed),hash(i+1,seed),u)*2-1)*amp;
    ctx.lineTo(wx-ox,y);
  }
  ctx.lineTo(ox+vw+step*2-ox,LH); ctx.closePath(); ctx.fill();
}

    } }

    ctx.beginPath(); ctx.moveTo(w.x,w.y); ctx.lineTo(w.x+w.len,w.y); ctx.stroke(); }
  for(const p of parts){ ctx.globalAlpha=Math.max(0,p.life); ctx.fillStyle=p.col;

  ctx.font='26px '+getComputedStyle(document.body).getPropertyValue('--disp');
  for(const x of texts){ ctx.globalAlpha=Math.max(0,x.life);
    ctx.fillStyle=x.gold?'#ffe27a':'#fff'; ctx.textAlign='center';

function drawSnake(){
  const n=bodyPts.length; if(n<2)return;
  const L=[],R=[];
  for(let i=0;i<n;i++){
    const p=bodyPts[i], pr=bodyPts[Math.max(0,i-1)], q=bodyPts[Math.min(n-1,i+1)];
    let dx=q.x-pr.x,dy=q.y-pr.y; const len=Math.hypot(dx,dy)||1; const nx=-dy/len,ny=dx/len;
    const rad=i===0?HEAD_R+2:Math.max(6,HEAD_R*(1-.55*i/(n-1)));
    const wob=i>0?Math.sin(t*7-i*.55)*1.8*(i/n):0;
    L.push([p.x+nx*rad,p.y+wob+ny*rad]); R.push([p.x-nx*rad,p.y+wob-ny*rad]);
  }
  ctx.beginPath(); ctx.moveTo(L[0][0],L[0][1]);
  for(let i=1;i<n;i++)ctx.lineTo(L[i][0],L[i][1]);
  for(let i=n-1;i>=0;i--)ctx.lineTo(R[i][0],R[i][1]);
  ctx.closePath();
  const tail=bodyPts[n-1];
  const g=ctx.createLinearGradient(head.x,head.y,tail.x,tail.y);
  g.addColorStop(0,'#4ded96'); g.addColorStop(.55,'#23b46b'); g.addColorStop(1,'#137a49');
  ctx.fillStyle=g; ctx.fill();
  ctx.lineWidth=3.5; ctx.strokeStyle='rgba(8,34,22,.85)'; ctx.stroke();
  // belly sheen
  ctx.beginPath();
  for(let i=0;i<n;i++){ const p=bodyPts[i], rad=i===0?HEAD_R:Math.max(6,HEAD_R*(1-.55*i/(n-1)));
    const y=p.y+rad*.42+(i>0?Math.sin(t*7-i*.55)*1.8*(i/n):0);
    i?ctx.lineTo(p.x,y):ctx.moveTo(p.x,y); }
  ctx.lineWidth=3; ctx.strokeStyle='rgba(255,255,255,.14)'; ctx.stroke();
  // bands
  ctx.strokeStyle='rgba(9,45,28,.3)'; ctx.lineWidth=4;
  for(let i=3;i<n-2;i+=4){ const p=bodyPts[i], rad=Math.max(6,HEAD_R*(1-.55*i/(n-1)));
    ctx.beginPath(); ctx.arc(p.x,p.y+Math.sin(t*7-i*.55)*1.8*(i/n),rad+1,0,7); ctx.stroke(); }
  // head
  const hx=head.x,hy=head.y;
  let hg=ctx.createRadialGradient(hx-4,hy-5,2,hx,hy,HEAD_R+3);
  hg.addColorStop(0,'#7df9b4'); hg.addColorStop(.6,'#2ecb7d'); hg.addColorStop(1,'#1a9a5c');
  ctx.fillStyle=hg; ctx.beginPath(); ctx.arc(hx,hy,HEAD_R+2.5,0,7); ctx.fill();
  ctx.lineWidth=3; ctx.strokeStyle='rgba(8,34,22,.85)'; ctx.stroke();
  // eye (side view: one eye)
  const ea=Math.atan2(head.vy*.4,1)*.5;
  const ex=hx+Math.cos(ea)*6, ey=hy+Math.sin(ea)*6-5;
  const blink=(t%3.4)<.12;
  ctx.save(); ctx.translate(ex,ey); ctx.scale(1,blink?.15:1);
  ctx.fillStyle='#fff'; ctx.beginPath(); ctx.arc(0,0,4.8,0,7); ctx.fill();
  ctx.fillStyle='#101c14'; ctx.beginPath(); ctx.arc(1.6,0,2.5,0,7); ctx.fill();
  ctx.restore();
  // tongue
  let nearFood=false; for(const o of orbs){ if(o.x>hx&&o.x-hx<170&&Math.abs(o.y-hy)<90){nearFood=true;break;} }
  if(nearFood||t%2.7<.32){ const wob=Math.sin(t*30)*2;
    ctx.strokeStyle='#ff5a5a'; ctx.lineWidth=2; ctx.beginPath();
    ctx.moveTo(hx+HEAD_R,hy+3); ctx.lineTo(hx+HEAD_R+13,hy+3+wob*.4);
    ctx.moveTo(hx+HEAD_R+13,hy+3+wob*.4); ctx.lineTo(hx+HEAD_R+19,hy-1+wob);
    ctx.moveTo(hx+HEAD_R+13,hy+3+wob*.4); ctx.lineTo(hx+HEAD_R+19,hy+8-wob);
    ctx.stroke(); }
}
