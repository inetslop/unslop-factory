/* ============ zombie AI (reverse BFS from player) ============ */
function reverseBFS(sx,sy){

  for(let i=0;i<q.length;i++){const c=q[i],d=dist[c[0]+','+c[1]];

      dist[k]=d+1;q.push([x,y]);}}
  return dist;
}
function pickStep(dist,x,y,d,z,taken){
  const cands=[];
  for(const dr of DIRS){const nx=x+dr[0],ny=y+dr[1],nk=nx+','+ny;
    if(dist[nk]===d-1&&!taken.has(nk))cands.push({nx,ny,dx:dr[0],dy:dr[1]});}
  if(!cands.length)return null;
  cands.sort((a,b)=>{
    const as=(z.lastDir&&z.lastDir[0]===a.dx&&z.lastDir[1]===a.dy)?0:1+Math.random()*.5;
    const bs=(z.lastDir&&z.lastDir[0]===b.dx&&z.lastDir[1]===b.dy)?0:1+Math.random()*.5;
    return as-bs;});
  const c=cands[0];taken.add(c.nx+','+c.ny);z.lastDir=[c.dx,c.dy];return [c.nx,c.ny];
}
function computeZombies(){
  const dist=reverseBFS(player.gx,player.gy),taken=new Set();
  const order=zombies.slice().sort((a,b)=>{
    const da=dist[a.gx+','+a.gy],db=dist[b.gx+','+b.gy];
    return (da===undefined?1e9:(db===undefined?-1e9:db-da));});
  for(const z of order){
    z.next.length=0;

    const s1=pickStep(dist,z.gx,z.gy,d,z,taken);if(!s1)continue;
    z.next.push(s1);
    if(z.runner){const s2=pickStep(dist,s1[0],s1[1],d-1,z,taken);if(s2)z.next.push(s2);}
  }
}

/* ============ update loop ============ */
function updEnt(e,dt,onArrive){
  if(e.t>=1)return;
  const prev=e.t;e.t=Math.min(1,e.t+dt/e.dur);
  const k=1-Math.pow(1-e.t,3);
  e.cx=lerp(e.fx,e.gx,k);e.cy=lerp(e.fy,e.gy,k);
  if(e.t>=1&&prev<1&&onArrive)onArrive(e);
}
function tick(dt){
  clock+=dt;
  const frozen=screen==='pause';
  // ambient (always alive, even paused)
  for(const s of spores){s.y+=s.v*dt;if(s.y<-1)s.y=rows+1;if(s.y>rows+1)s.y=-1;}
  for(const f of fogs){f.x+=f.vx*dt;f.y+=f.vy*dt;
    if(f.x<-f.r)f.x=cols+f.r;if(f.x>cols+f.r)f.x=-f.r;if(f.y<-f.r)f.y=rows+f.r;if(f.y>rows+f.r)f.y=-f.r;}
  for(const s of stains)s.a=Math.max(0,s.a-dt*.015);
  if(!frozen){
    updEnt(player,dt,null);
    for(const c of crates)updEnt(c,dt,e=>{if(pads.has(e.gx+','+e.gy)&&screen==='play'){SFX.pad();sparkle(e.gx,e.gy);}});
    for(const z of zombies)updEnt(z,dt,zOnArrive);
  }
  if(!frozen){
    shake=Math.max(0,shake-shake*7*dt-2*dt);
    player.bx=lerp(player.bx,0,1-Math.exp(-16*dt));player.by=lerp(player.by,0,1-Math.exp(-16*dt));
  }
  // particles
  for(let i=parts.length-1;i>=0;i--){const p=parts[i];p.life-=dt;
    if(p.life<=0){parts.splice(i,1);continue;}
    p.x+=p.vx*dt;p.y+=p.vy*dt;p.vy+=(p.g||0)*dt;}
  // zombie turn & input
  if(screen==='play'&&!frozen){
    lvlTime+=dt;
    if(pendingTurn&&!busyAll()){computeZombies();pendingTurn=false;}
    while(inputQ.length&&!busyAll())tryMove(inputQ.shift()[0],inputQ.shift()===undefined?0:inputQ[inputQ.length-1][1]);
  }
  // danger + heartbeat
  dangerNow=false;
  if(screen==='play'){
    for(const z of zombies)if(Math.max(Math.abs(z.gx-player.gx),Math.abs(z.gy-player.gy))<=1){dangerNow=true;break;}
    hbT-=dt;if(dangerNow&&hbT<=0&&AC&&!muted){SFX.heart();hbT=.95;}else if(!dangerNow)hbT=0;
    ambT-=dt;if(ambT<=0){let near=false;for(const z of zombies)if(Math.hypot(z.cx-player.cx,z.cy-player.cy)<=5){near=true;break;}
      if(near)SFX.growl(false);ambT=rand(4,8);}
  }
  // sequences
  if(screen==='dying'){dieT+=dt;if(dieT>1.15)showDead();}
  if(screen==='clearing'){clearT+=dt;if(clearT>1.5)showClearOverlay();}
  // menu attract mode
  if(screen==='menu'&&!frozen){attractT+=dt;if(attractT>.95){attractT=0;attractStep();}}
}
function zOnArrive(z){
  if(screen==='play'&&z.gx===player.gx&&z.gy===player.gy){doDeath();return;}
  const d=Math.hypot(z.cx-player.cx,z.cy-player.cy);
  if(d<=3.2&&Math.random()<.5)SFX.growl(true);
}
function attractStep(){
  const dist=reverseBFS(player.gx,player.gy),taken=new Set();
  for(const z of zombies){
    if(z.t<1||z.next.length)continue;
    const d=dist[z.gx+','+z.gy];if(d===undefined||d<=0)continue;
    const s=pickStep(dist,z.gx,z.gy,d,z,taken);if(!s)continue;
    if(s[0]===player.gx&&s[1]===player.gy){poof(z.cx,z.cy);z.gx=z.sx;z.gy=z.sy;z.cx=z.sx;z.cy=z.sy;z.t=1;continue;}
    startMove(z,s[0],s[1],.55);
  }
}

  ctx.globalAlpha=1;
}
function drawLighting(){
  if(screen!=='dead'&&player){
    const px=ox+(player.cx+.5)*cell,py=oy+(player.cy+.5)*cell;
    const g=ctx.createRadialGradient(px,py,0,px,py,cell*4.6);
    g.addColorStop(0,'rgba(255,238,190,.07)');g.addColorStop(1,'rgba(255,238,190,0)');
    ctx.fillStyle=g;ctx.fillRect(0,0,W,H);
  }
  if(vig){ctx.fillStyle=vig;ctx.fillRect(0,0,W,H);}
  if(dangerNow||screen==='dying'){
    ctx.globalAlpha=.1+.08*Math.sin(clock*7);
    if(redVig){ctx.fillStyle=redVig;ctx.fillRect(0,0,W,H);}
    ctx.globalAlpha=1;
  }
  if(screen==='dying'&&dieT<.2){ctx.fillStyle='rgba(200,30,15,'+(.45*(1-dieT/.2))+')';ctx.fillRect(0,0,W,H);}
  if(screen==='clearing'&&clearT<.3){ctx.fillStyle='rgba(87,255,122,'+(.22*(1-clearT/.3))+')';ctx.fillRect(0,0,W,H);}
}
