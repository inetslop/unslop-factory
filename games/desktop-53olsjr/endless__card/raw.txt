function spawnInterval(){ return Math.max(1.15 - (level-1)*0.06, 0.55); }

function spawn(){
  const L = layout();
  const targetLive = cards.some(c=>c.suit===target && !c.dead);
  let suit;
  if(!targetLive && (sinceTarget>=3 || Math.random()<0.40)) suit = target;
  else {
    const others = SUITS.filter(s=>s!==target);
    suit = others[(Math.random()*others.length)|0];
  }
  sinceTarget = (suit===target) ? 0 : sinceTarget+1;
  let lane = (Math.random()*4)|0;
  if(lane===lastLane) lane=(lane+1+((Math.random()*3)|0))%4;
  lastLane=lane;
  const joker = suit!==target && Math.random()<0.06;
  cards.push({ lane, y:-L.cardH*1.15, suit, rank:RANKS[(Math.random()*13)|0], joker, dead:false, missed:false });
}

function catchCard(c){
  const L = layout();
  const x = L.margin + c.lane*L.laneW + L.laneW/2;
  const y = c.y + L.cardH*.5;
  cards.splice(cards.indexOf(c),1);
  combo++; maxCombo=Math.max(maxCombo,combo);
  const oldMult = multNow();
  let pts = 10*oldMult + (c.joker?75*oldMult:0);
  score += pts; catches++;
  level = Math.min(20, 1+Math.floor(catches/8));
  laneFlash[c.lane]=1;
  fx.push({type:'card', x, y, w:L.cardW, h:L.cardH, suit:c.suit, rank:c.rank, joker:c.joker, t:0, dur:.5, dir:(x<W/2?-1:1)});
  burst(x,y, c.joker?'#f4d789':'#ffe9ad', c.joker?26:13, 190, -1);
  texts.push({x, y:y-6, txt:'+'+pts, color:'#f4d789', t:0, dur:.85, size:Math.max(17,W*0.022)});
  if(c.joker){
    shield=true; flashGold=.6; sJoker();
    texts.push({x, y:y+26, txt:'★ SHIELD', color:'#f4d789', t:0, dur:1.1, size:Math.max(13,W*0.016)});
  } else {
    sCatch(combo);
  }
  if(multNow()>oldMult){
    texts.push({x, y:y-34, txt:'COMBO ×'+multNow(), color:'#ffb454', t:0, dur:1.1, size:Math.max(14,W*0.017)});
  }
  updateHUD();
  if(catches%6===0) newTarget();
}

}

function lanePress(i){
  if(state!=='play') return;
  laneFlash[i]=1;
  let pick=null;
  for(const c of cards){
    if(c.lane!==i || c.dead || c.missed) continue;
    if(!pick || c.y>pick.y) pick=c;
  }
  if(!pick){ sWhiff(); return; }
  if(pick.joker || pick.suit===target) catchCard(pick);
  else wrongCard(pick);
}

function burst(x,y,color,n,speed,grav){
  for(let i=0;i<n;i++){
    const a=Math.random()*Math.PI*2, sp=speed*(.4+Math.random()*.9);
    fx.push({type:'p', x, y, vx:Math.cos(a)*sp, vy:Math.sin(a)*speed*(grav===1?.6:.5)-30, grav:grav, t:0, dur:.5+Math.random()*.4, size:2+Math.random()*3.5, color});
  }
}

/* ============================== UPDATE ============================== */
function update(dt){
  runTime+=dt;
  const L = layout();
  const fallV = 0.20 + Math.min(level-1,20)*0.016;   // field-heights per second
  spawnT-=dt;
  if(spawnT<=0){ spawn(); spawnT = spawnInterval()*(0.85+Math.random()*0.3); }
  for(const c of cards){
    if(c.dead) continue;
    c.y += fallV*H*dt;
    const bottom = c.y + L.cardH;
    if(!c.missed && bottom >= L.lineY){
      if(c.suit===target && !c.joker) missCard(c);
      else c.missed=true;              // decoy / joker slips past, no penalty
    }
    if(c.y > H+50) c.dead=true;
  }
  cards = cards.filter(c=>!c.dead);
}

function updateFx(dt){
  for(const f of fx){ f.t += dt/(f.dur||1); }
  fx = fx.filter(f=>f.t<1);
  for(const t of texts) t.t += dt/t.dur;
  texts = texts.filter(t=>t.t<1);
}

function updateAmbient(dt){
  for(const a of ambient){
    a.y += a.v*dt; a.rot += a.vr*dt;
    if(a.y > H+a.size){ a.y=-a.size; a.x=Math.random()*W; }
  }
}

      ctx.font=`bold ${w*.14}px "Trebuchet MS",sans-serif`;
      ctx.fillText('JOKER',-w*.02,w*.3);
      ctx.restore();
    };
    corner(-w/2+w*.1,-h/2+h*.05);
    ctx.save(); ctx.rotate(Math.PI); corner(-w/2+w*.1,-h/2+h*.05); ctx.restore();
  } else {
    // corners
    const drawCorner=()=>{
      ctx.fillStyle=ink; ctx.textAlign='left'; ctx.textBaseline='top';
      const fs=w*.27;
      ctx.font=`bold ${fs}px Georgia,serif`;
      ctx.fillText(rank, w*.09, h*.045);
      ctx.font=`${w*.19}px Georgia,serif`;
      const sw = ctx.measureText(SUI[suit].ch).width;
      ctx.fillText(SUI[suit].ch, w*.09+(fs-sw)/2, h*.045+fs*1.08);
    };
    drawCorner();
    ctx.save(); ctx.rotate(Math.PI); drawCorner(); ctx.restore();

    ctx.textAlign='center'; ctx.textBaseline='middle';
    if(rank==='A'){
      ctx.strokeStyle=ink; ctx.globalAlpha=alpha*.4; ctx.lineWidth=1;
      rr(ctx,-w/2+w*.17,-h/2+h*.08,w-w*.34,h-h*.16,r*.7); ctx.stroke();
      ctx.globalAlpha=alpha; ctx.fillStyle=ink;
      ctx.font=`${w*.5}px Georgia,serif`; ctx.fillText(SUI[suit].ch,0,0);
    } else if(rank==='J'||rank==='Q'||rank==='K'){
      ctx.strokeStyle=ink; ctx.lineWidth=1.4; ctx.globalAlpha=alpha*.65;
      rr(ctx,-w/2+w*.16,-h/2+h*.075,w-w*.32,h-h*.15,r*.7); ctx.stroke();
      ctx.globalAlpha=alpha*.35; ctx.lineWidth=1;
      rr(ctx,-w/2+w*.21,-h/2+h*.11,w-w*.42,h-h*.22,r*.6); ctx.stroke();
      ctx.globalAlpha=alpha; ctx.fillStyle=ink;
      ctx.font=`bold ${w*.48}px Georgia,serif`; ctx.fillText(rank,0,0);
      ctx.font=`${w*.15}px Georgia,serif`;
      ctx.fillText(SUI[suit].ch,0,-h*.26); ctx.fillText(SUI[suit].ch,0,h*.26);
    } else {
      const iw=w*.62, ih=h*.64, ix=-iw/2, iy=-ih/2;
      const ps = (rank==='9'||rank==='10') ? w*.135 : w*.17;
      ctx.fillStyle=ink; ctx.font=`${ps}px Georgia,serif`;
      for(const p of PIPS[rank]){
        const px=ix+p[0]*iw, py=iy+p[1]*ih;
        if(p[1]>0.5){ ctx.save(); ctx.translate(px,py); ctx.rotate(Math.PI); ctx.fillText(SUI[suit].ch,0,0); ctx.restore(); }
        else ctx.fillText(SUI[suit].ch,px,py);
      }
    }
  }
  ctx.restore();
}

function easeOut(t){ return 1-Math.pow(1-t,3); }

function render(){
  ctx.clearRect(0,0,W,H);
  let ox=0, oy=0;
  if(shake>0.2){ ox=(Math.random()*2-1)*shake; oy=(Math.random()*2-1)*shake; }
  ctx.save(); ctx.translate(ox,oy);
  if(bg) ctx.drawImage(bg,0,0,W,H);
  const L = layout();

  // ambient drifting suits
  for(const a of ambient){
    ctx.save();
    ctx.globalAlpha=a.a; ctx.translate(a.x,a.y); ctx.rotate(a.rot);
    ctx.fillStyle = RED.has(a.suit)?'#d0342c':'#f6efdc';
    ctx.font=`${a.size}px Georgia,serif`; ctx.textAlign='center'; ctx.textBaseline='middle';
    ctx.fillText(SUI[a.suit].ch,0,0);
    ctx.restore();
  }

  // lane press flashes
  for(let i=0;i<4;i++){
    if(laneFlash[i]>0.01){
      const g=ctx.createLinearGradient(0,L.lineY,0,L.lineY-260);
      g.addColorStop(0,`rgba(244,215,137,${laneFlash[i]*.28})`);
      g.addColorStop(1,'rgba(244,215,137,0)');
      ctx.fillStyle=g;
      ctx.fillRect(L.margin+i*L.laneW, L.lineY-260, L.laneW, 260);
    }
  }

  // cards
  for(const c of cards){
    const x = L.margin + c.lane*L.laneW + (L.laneW-L.cardW)/2;
    let alpha=1, rot=0;
    if(c.missed){
      alpha = Math.max(0, 1-(c.y-L.lineY)/(H*.3));
      rot = (c.y-L.lineY)*.004;
    }
    drawFace(x, c.y, L.cardW, L.cardH, c.suit, c.rank, c.joker, alpha, rot);
  }

  // effects
  for(const f of fx){
    if(f.type==='card'){
      const e=easeOut(f.t);
      drawFace(f.x-f.w/2 + f.dir*e*50, f.y-f.h/2 - e*H*.24, f.w, f.h, f.suit, f.rank, f.joker, 1-e, e*(f.dir>0?2.4:-2.4));
    } else if(f.type==='p'){
      const g = (f.grav===1? 380 : -60);
      f.vy += g*0.016; // approx per-frame gravity feel
      ctx.globalAlpha = 1-f.t;
      ctx.fillStyle=f.color;
      ctx.beginPath(); ctx.arc(f.x+f.vx*f.t*.3, f.y+f.vy*f.t*.4 + (f.grav===1? 200*f.t*f.t : -60*f.t*f.t), f.size*(1-f.t*.5), 0, 7); ctx.fill();
      ctx.globalAlpha=1;
    } else if(f.type==='x'){
      const e=easeOut(f.t);
      ctx.save(); ctx.translate(f.x,f.y); ctx.rotate(e*1.2);
      ctx.globalAlpha=1-f.t; ctx.strokeStyle='#e0483f'; ctx.lineWidth=6; ctx.lineCap='round';
      const s=f.w*.5*(0.5+e*.7);
      ctx.beginPath(); ctx.moveTo(-s,-s); ctx.lineTo(s,s); ctx.moveTo(s,-s); ctx.lineTo(-s,s); ctx.stroke();
      ctx.restore(); ctx.globalAlpha=1;
    } else if(f.type==='banner'){
      const a = f.t<.15? f.t/.15 : f.t>.72? (1-f.t)/.28 : 1;
      const bw=Math.min(W*.8,430), bh=64, bx=W/2-bw/2, by=H*.26 - (1-easeOut(Math.min(1,f.t*2)))*30;
      ctx.globalAlpha=a;
      ctx.shadowColor='rgba(0,0,0,.5)'; ctx.shadowBlur=24; ctx.shadowOffsetY=8;
      ctx.fillStyle='#f8f2df'; rr(ctx,bx,by,bw,bh,10); ctx.fill();
      ctx.shadowColor='transparent'; ctx.shadowBlur=0; ctx.shadowOffsetY=0;
      ctx.strokeStyle='#d9b45b'; ctx.lineWidth=2; rr(ctx,bx+3,by+3,bw-6,bh-6,8); ctx.stroke();
      const red = RED.has(f.suit);
      ctx.fillStyle = red?'#d0342c':'#22262e';
      ctx.font=`${bh*.5}px Georgia,serif`; ctx.textAlign='center'; ctx.textBaseline='middle';
      ctx.fillText(SUI[f.suit].ch, bx+bw*.16, by+bh/2+2);
      ctx.fillStyle='#3c3628'; ctx.font=`bold ${bh*.24}px "Trebuchet MS",sans-serif`;
      ctx.textAlign='left';
      if('letterSpacing' in ctx) ctx.letterSpacing='3px';
      ctx.fillText('CATCH THE '+SUI[f.suit].name, bx+bw*.28, by+bh/2+1);
      if('letterSpacing' in ctx) ctx.letterSpacing='0px';
      ctx.globalAlpha=1;
    }
  }

  // floating texts
  for(const t of texts){
    const e=easeOut(t.t);
    ctx.save();
    ctx.globalAlpha = 1-e*e;
    ctx.translate(t.x, t.y - e*42);
    const sc = 1 + (1-Math.min(1,t.t*5))*.35;
    ctx.scale(sc,sc);
    ctx.font=`bold ${t.size}px Georgia,serif`;
    ctx.textAlign='center'; ctx.textBaseline='middle';
    ctx.strokeStyle='rgba(0,0,0,.6)'; ctx.lineWidth=4; ctx.lineJoin='round';
    ctx.strokeText(t.txt,0,0);
    ctx.fillStyle=t.color; ctx.fillText(t.txt,0,0);
    ctx.restore();
  }
  ctx.restore();

  // full-screen flashes (unshaken)
  if(flashRed>0.01){ ctx.fillStyle=`rgba(208,52,44,${flashRed*.32})`; ctx.fillRect(0,0,W,H); }
  if(flashGold>0.01){ ctx.fillStyle=`rgba(244,215,137,${flashGold*.26})`; ctx.fillRect(0,0,W,H); }
}

/* ============================== LOOP ============================== */
let last = performance.now();
function frame(now){
  requestAnimationFrame(frame);
  let dt=(now-last)/1000; last=now;
  if(dt>0.05) dt=0.05;
  if(state==='play') update(dt);
  if(state!=='start') updateFx(dt);
  updateAmbient(dt);
  shake   = Math.max(0, shake - dt*34);
  flashRed  = Math.max(0, flashRed - dt*1.5);
  flashGold = Math.max(0, flashGold - dt*1.2);
  for(let i=0;i<4;i++) laneFlash[i]=Math.max(0,laneFlash[i]-dt*4.5);
  render();
}
