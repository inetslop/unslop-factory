let AC=null,master=null,muted=false;
function ensureAC(){
  if(!AC){try{AC=new (window.AudioContext||window.webkitAudioContext)();master=AC.createGain();master.gain.value=.9;master.connect(AC.destination);}catch(e){}}
  if(AC&&AC.state==='suspended')AC.resume();
}
function tone(f0,f1,dur,type,vol,when){
  if(!AC||muted)return;when=when||0;const t=AC.currentTime+when;
  const o=AC.createOscillator(),g=AC.createGain();o.type=type;
  o.frequency.setValueAtTime(Math.max(1,f0),t);
  if(f1&&f1!==f0)o.frequency.exponentialRampToValueAtTime(Math.max(1,f1),t+dur);
  g.gain.setValueAtTime(vol,t);g.gain.exponentialRampToValueAtTime(.0001,t+dur);
  o.connect(g);g.connect(master);o.start(t);o.stop(t+dur+.03);
}
function noiseS(dur,fc,vol,when){
  if(!AC||muted)return;when=when||0;const t=AC.currentTime+when;
  const n=Math.max(1,Math.floor(AC.sampleRate*dur)),buf=AC.createBuffer(1,n,AC.sampleRate),d=buf.getChannelData(0);
  for(let i=0;i<n;i++)d[i]=(Math.random()*2-1)*(1-i/n);
  const s=AC.createBufferSource();s.buffer=buf;
  const f=AC.createBiquadFilter();f.type='lowpass';f.frequency.value=fc;
  const g=AC.createGain();g.gain.value=vol;
  s.connect(f);f.connect(g);g.connect(master);s.start(t);
}

/* ambient seeds */
let fireflies=[],speckle=[],shimmer=[],nightStars=[];

  fireflies=[];for(let i=0;i<13;i++)fireflies.push({bx:Math.random()*W,by:Math.random()*H,ph:Math.random()*7,sp:.4+Math.random()*.8,r:1.2+Math.random()*1.6});
  speckle=[];for(let i=0;i<26;i++)speckle.push({x:Math.random(),y:Math.random()});
  shimmer=[];for(let r=RIVER_TOP;r<=RIVER_BOT;r++){const a=[];for(let i=0;i<5;i++)a.push({x:Math.random()*W,ph:Math.random()*7,w:30+Math.random()*60});shimmer.push(a);}
  nightStars=[];for(let i=0;i<26;i++)nightStars.push({x:Math.random(),y:Math.random()*.8,r:Math.random()*1.4+.5});

/* ============================== lanes ============================== */
function placeVehicles(lens,minGap){
  for(let a=0;a<50;a++){
    const off=Math.floor(Math.random()*COLS);
    const xs=lens.map((ln,i)=>off+Math.round(i*COLS/lens.length));
    for(let i=1;i<xs.length;i++) if(xs[i]-xs[i-1]<lens[i-1]+minGap) xs[i]=xs[i-1]+lens[i-1]+minGap;
    let ok=true;
    const last=xs.length-1;
    if(((xs[0]-xs[last])%COLS+COLS)%COLS < lens[last]+minGap) ok=false;
    for(let i=0;i<xs.length;i++) xs[i]=norm(xs[i]);
    if(ok)return xs;
  }
  return lens.map((ln,i)=>norm(Math.round(i*COLS/lens.length)));
}
function buildLanes(level){
  const sp=b=>Math.min(b+Math.floor((level-1)/2),3);
  const lanes=[];
  RIVER_DEFS.forEach(d=>{
    const xs=placeVehicles(d.lens,1);
    lanes.push({row:d.row,dir:d.dir,speed:sp(d.speed),type:'log',vehicles:xs.map((x,i)=>({x,len:d.lens[i],prevX:x}))});
  });
  ROAD_DEFS.forEach(d=>{
    const lens=[];for(let i=0;i<d.cars;i++)lens.push(1);for(let i=0;i<d.trucks;i++)lens.push(2);
    for(let i=lens.length-1;i>0;i--){const j=Math.floor(Math.random()*(i+1));[lens[i],lens[j]]=[lens[j],lens[i]];}
    const xs=placeVehicles(lens,1);
    lanes.push({row:d.row,dir:d.dir,speed:sp(d.speed),type:'car',vehicles:xs.map((x,i)=>({x,len:lens[i],prevX:x,color:CAR_COLORS[Math.floor(Math.random()*CAR_COLORS.length)],truck:lens[i]===2}))});
  });
  G.lanes=lanes;G.laneMap={};lanes.forEach(l=>G.laneMap[l.row]=l);
}
const findV=(l,x)=>l.vehicles.find(v=>(((x-v.x)%COLS)+COLS)%COLS<v.len)||null;
const logContains=x=>{for(let r=RIVER_TOP;r<=RIVER_BOT;r++){const l=G.laneMap[r];if(l&&findV(l,x))return true;}return false;};
const carAt=(cx,cy)=>{const l=G.laneMap[cy];return (l&&l.type==='car'&&!!findV(l,cx));};

/* ============================== flow ============================== */
function initGame(){
  G.score=0;G.lives=3;G.level=1;G.crossings=0;G.pads=[false,false,false,false,false];
  G.budget=budget(1);G.turns=G.budget;G.dispFrac=1;G.bestRow=10;
  G.frog={x:6,y:10,dx:0,dy:-1};G.ride=null;
  buildLanes(1);G.particles=[];G.popups=[];G.shake=0;
  hide('start');hide('over');G.phase='idle';
}
function respawn(){
  G.frog={x:6,y:10,dx:0,dy:-1};G.ride=null;G.bestRow=GRASS;
  G.budget=budget(G.level);G.turns=G.budget;
  G.phase='idle';popup(ox+7*S,oy+(GRASS-0.4)*S,'GO!',15,'#d2ffb0');sfx.ready();
}
function startTraffic(){
  G.turns--;
  if(G.frog.y>=RIVER_TOP&&G.frog.y<=RIVER_BOT){
    const l=G.laneMap[G.frog.y],v=findV(l,G.frog.x);
    G.ride=v?{l,v}:null;
  } else G.ride=null;
  G.frogPrevX=G.frog.x;
  stepVisual();G.phase='traffic';
}
function stepVisual(){
  G.lanes.forEach(l=>l.vehicles.forEach(v=>{v.prevX=v.x;v.x=norm(v.x+l.dir*l.speed);}));
  G.lanes.filter(l=>l.type==='car').forEach(l=>l.vehicles.forEach(v=>{
    for(let i=0;i<3;i++)G.particles.push({t:'puff',x:ox+norm(v.x+(l.dir>0?-v.len:.4))*S+S/2,y:oy+l.row*S+S/2,
      vx:-l.dir*(.4+Math.random()*.5),vy:(Math.random()-.5)*.4,a:.32,r:2+Math.random()*2,life:0,max:420});
  }));
  G.trafficT0=performance.now();
}
function finalizeTraffic(){
  if(G.ride){
    const l=G.ride.l,nx=G.frogPrevX+l.dir*l.speed;
    if(nx<0||nx>=COLS){death('water');return;}
    G.frog.x=nx;
  }
  for(const l of G.lanes)if(l.type==='car'&&l.row===G.frog.y){
    let c=G.frogPrevX;let hit=false;
    for(let i=0;i<l.speed;i++){c=norm(c+l.dir);if(c===G.frog.x)hit=true;}
    if(hit){death('car');return;}
  }
  if(G.frog.y>=RIVER_TOP&&G.frog.y<=RIVER_BOT&&!logContains(G.frog.x)){death('water');return;}
  scoreRows();
  if(G.turns<=0){death('night');return;}
  G.phase='idle';
}
function scoreRows(){
  if(G.frog.y<G.bestRow&&G.frog.y>0){
    const pts=(G.bestRow-G.frog.y)*10;G.score+=pts;
    popup(frogPx(),frogPy()-S*.4,'+'+pts,15,'#ffc95e');sfx.row();G.bestRow=G.frog.y;
  }
}
function tryMove(dx,dy){
  if(G.phase!=='idle')return;
  G.frog.dx=dx;G.frog.dy=dy;
  const nx=G.frog.x+dx,ny=G.frog.y+dy;
  if(nx<0||nx>=COLS||ny<0||ny>=ROWS)return bump();
  if(ny===0){
    if(!PAD_COLS.includes(nx))return bump();
    const idx=PAD_COLS.indexOf(nx);
    if(G.pads[idx])return bump();
    hopTo(nx,ny,'home',idx);return;
  }
  if(carAt(nx,ny)){hopTo(nx,ny,'dcar');return;}
  if(ny>=RIVER_TOP&&ny<=RIVER_BOT&&!logContains(nx)){hopTo(nx,ny,'dwater');return;}
  hopTo(nx,ny,'traffic');
}
function hopTo(nx,ny,pending,idx){
  G.pending={kind:pending,idx};
  G.anim={ax0:G.frog.x,ay0:G.frog.y,ax1:nx,ay1:ny,t0:performance.now(),dur:140};
  G.frog.x=nx;G.frog.y=ny;G.phase='player-mov';
  if(pending==='traffic'||pending==='home')sfx.move();
}
function doWait(){
  if(G.phase!=='idle')return;
  G.anim={ax0:G.frog.x,ay0:G.frog.y,ax1:G.frog.x,ay1:G.frog.y,t0:performance.now(),dur:90};
  G.pending={kind:'traffic'};G.phase='player-mov';sfx.wait();
}
function bump(){G.bumpT=performance.now();sfx.bump();}

function executePending(){
  const p=G.pending;G.anim=null;
  if(p.kind==='traffic'){startTraffic();return;}
  if(p.kind==='home'){startHome(p.idx);return;}
  if(p.kind==='dcar'){death('car');return;}
  if(p.kind==='dwater'){death('water');return;}
}

  G.pads[idx]=true;G.crossings++;
  const pts=100+G.turns*3;G.score+=pts;
  popup(frogPx(),frogPy()-S*.5,'+'+pts+' HOME!',18,'#d2ffb0');
  sparkle(frogPx(),frogPy());sfx.home();
  G.phase='home';G.homeT0=performance.now();
}
function afterHome(){
  if(G.pads.every(p=>p))startLevelUp();else respawn();
}
function startLevelUp(){
  const bonus=100*(G.level-1);G.score+=bonus;
  G.level++;popup(W/2,oy+S*3.2,'WAVE BONUS +'+bonus,16,'#ffc95e');

}

function death(cause){
  G.lastDeath=cause;G.lives--;
  const now=performance.now();G.deadT0=now;
  if(cause==='water'){
    let px,py;
    if(G.ride){px=G.frogPrevX<0? ox+ox*0 : (G.frogPrevX+G.ride.l.dir*G.ride.l.speed)<0? ox+S*.2 : ox+(COLS-0.2)*S;}
    else {px=ox+G.frog.x*S+S/2;py=oy+G.frog.y*S+S/2;}
    if(!py)py=oy+G.frog.y*S+S/2;
    G.deathAnchor={x:px,y:py};splash(px,py);sfx.splash();
  } else if(cause==='car'){
    G.deathAnchor={x:frogPx(),y:frogPy()};stars(G.deathAnchor.x,G.deathAnchor.y);
    G.shake=10;sfx.hit();
  } else {G.deathAnchor={x:frogPx(),y:frogPy()};sfx.night();}
  G.phase='dying';
}
function afterDeath(){ if(G.lives<=0)showOver(); else respawn(); }

function drawFrog(px,py,s,o){
  o=o||{};const a=o.alpha==null?1:o.alpha;
  ctx.save();ctx.translate(px,py);if(o.rot)ctx.rotate(o.rot);
  if(o.sx)ctx.scale(o.sx,o.sy||o.sx);
  ctx.globalAlpha=a;
  const g='#4fae4a';
  // legs
  ctx.fillStyle='#3f9440';
  [[-1,1],[1,1],[-1,-1],[1,-1]].forEach(d=>{ctx.beginPath();ctx.ellipse(d[0]*s*.3,d[1]*s*.26,s*.1,s*.055,d[0]*d[1]*.7);ctx.fill();});
  // body + head
  ctx.fillStyle=g;
  ctx.beginPath();ctx.ellipse(0,s*.05,s*.33,s*.26,0,0,7);ctx.fill();
  ctx.beginPath();ctx.arc(0,-s*.15,s*.235,0,7);ctx.fill();
  // belly
  ctx.fillStyle='#8fe36b';ctx.globalAlpha=a*.4;
  ctx.beginPath();ctx.ellipse(0,s*.11,s*.2,s*.14,0,0,7);ctx.fill();
  ctx.globalAlpha=a;
  // spots
  ctx.fillStyle='rgba(30,80,40,.35)';
  [[-.12,-.02],[.14,.06],[.02,.16]].forEach(p=>{ctx.beginPath();ctx.arc(p[0]*s,p[1]*s,s*.045,0,7);ctx.fill();});
  // eyes
  const ex=s*.155,ey=-s*.34;
  ctx.fillStyle='#fff';
  ctx.beginPath();ctx.arc(-ex,ey,s*.105,0,7);ctx.fill();
  ctx.beginPath();ctx.arc(ex,ey,s*.105,0,7);ctx.fill();
  if(o.happy){
    ctx.strokeStyle='#1e4a26';ctx.lineWidth=s*.04;ctx.lineCap='round';
    [[-ex],[ex]].forEach(p=>{ctx.beginPath();ctx.arc(p[0],ey-s*.02,s*.055,Math.PI*1.15,Math.PI*1.85);ctx.stroke();});
  }else{
    ctx.fillStyle='#10240f';
    const px2=(o.dx||0)*s*.035,py2=(o.dy||0)*s*.035;
    ctx.beginPath();ctx.arc(-ex+px2,ey+py2,s*.05,0,7);ctx.fill();
    ctx.beginPath();ctx.arc(ex+px2,ey+py2,s*.05,0,7);ctx.fill();
  }
  // cheeks
  ctx.fillStyle='rgba(242,143,177,.3)';
  ctx.beginPath();ctx.arc(-s*.2,-s*.16,s*.055,0,7);ctx.fill();
  ctx.beginPath();ctx.arc(s*.2,-s*.16,s*.055,0,7);ctx.fill();
  ctx.restore();
}

function drawLog(l,v,pos){
  const y=oy+l.row*S+S*.14,h=S*.72;
  const passes=pos+v.len>COLS?[pos-COLS,pos]:[pos];
  passes.forEach(pc=>{
    const x=ox+pc*S,w=v.len*S;
    ctx.fillStyle='#5d3a1e';rr(x+1,y+h-4,w-2,5,3);ctx.fill();
    ctx.fillStyle='#7a4f2b';rr(x,y,w,h,S*.3);ctx.fill();
    ctx.fillStyle='rgba(190,140,90,.4)';rr(x+2,y+2,w-4,h*.34,S*.28);ctx.fill();
    ctx.strokeStyle='rgba(70,40,15,.4)';ctx.lineWidth=1.5;
    for(let i=1;i<v.len;i++){ctx.beginPath();ctx.arc(x+i*S-h*.2,y+h*.62,h*.3,Math.PI*1.1,Math.PI*1.9);ctx.stroke();}
    ctx.fillStyle='#8a5c33';ctx.beginPath();ctx.arc(x+w-S*.24,y+h/2,S*.2,0,7);ctx.fill();
    ctx.strokeStyle='rgba(70,40,15,.5)';ctx.beginPath();ctx.arc(x+w-S*.24,y+h/2,S*.1,0,7);ctx.stroke();
  });
}
function drawCar(l,v,pos){
  const y=oy+l.row*S,h=S*.62,w=(v.len-0.1)*S;
  const passes=pos+v.len>COLS?[pos-COLS,pos]:[pos];
  passes.forEach(pc=>{
    const x=ox+pc*S+S*.05,yy=y+(S-h)/2;
    // wheels
    ctx.fillStyle='#141a16';
    [x+w*.18,x+w*.82].forEach(wx=>{ctx.beginPath();ctx.arc(wx,yy+h-1,S*.11,0,7);ctx.fill();});
    if(v.truck){
      const cabW=w*(l.dir>0? .32:.68), boxX=l.dir>0? x+cabW : x, boxW=w-cabW;
      ctx.fillStyle='#c9d1c9';rr(boxX,yy+h*.08,boxW,h*.84,3);ctx.fill();
      ctx.strokeStyle='rgba(60,70,60,.5)';ctx.lineWidth=2;
      for(let i=1;i<3;i++){const bx=boxX+boxW*i/3;ctx.beginPath();ctx.moveTo(bx,yy+h*.1);ctx.lineTo(bx,yy+h*.9);ctx.stroke();}
      ctx.fillStyle=v.color;rr(l.dir>0? x : x+cabW,yy,l.dir>0?cabW:boxW,h,4);ctx.fill();
      ctx.fillStyle='rgba(191,227,242,.9)';const wx=l.dir>0?x+3:x+cabW-3;rr(wx+S*.06,yy+h*.18,S*.2,h*.3,2);ctx.fill();
    }else{
      ctx.fillStyle=v.color;rr(x,yy,w,h,5);ctx.fill();
      ctx.fillStyle='rgba(0,0,0,.18)';rr(x,yy+h*.62,w,h*.38,4);ctx.fill();
      ctx.fillStyle='rgba(191,227,242,.9)';rr(x+w*.2,yy+h*.14,w*.6,h*.34,3);ctx.fill();
      ctx.strokeStyle='rgba(0,0,0,.25)';ctx.lineWidth=1.5;
      ctx.beginPath();ctx.moveTo(x+w/2,yy+h*.14);ctx.lineTo(x+w/2,yy+h*.48);ctx.stroke();
    }
    const frontX=l.dir>0? x+w-3 : x+3;
    ctx.fillStyle='#fff3b0';ctx.beginPath();ctx.arc(frontX,yy+h*.2,S*.06,0,7);ctx.fill();
    ctx.fillStyle='#e2574c';const backX=l.dir>0? x+3 : x+w-3;ctx.beginPath();ctx.arc(backX,yy+h*.2,S*.05,0,7);ctx.fill();
  });
}

function drawGhostCells(row,start,len){
  ctx.strokeStyle='rgba(255,255,255,.28)';ctx.lineWidth=1.5;ctx.setLineDash([4,3]);
  for(let i=0;i<len;i++){const c=norm(start+i);rr(ox+c*S+3,oy+row*S+3,S-6,S-6,4);ctx.stroke();}
  ctx.setLineDash([]);
}

function render(now){
  const dt=clamp(now-(render.last||now),0,50);render.last=now;
  // auto traffic in menu / gameover (living backdrop)
  if((G.phase==='menu'||G.phase==='gameover')&&now-G.lastAuto>760){G.lastAuto=now;stepVisual();}

  G.shake=Math.max(0,G.shake-dt*.02);
  const shx=(Math.random()-.5)*G.shake,shy=(Math.random()-.5)*G.shake;

  // background
  const bg=ctx.createLinearGradient(0,0,0,H);bg.addColorStop(0,'#12241a');bg.addColorStop(.6,'#0c1a12');bg.addColorStop(1,'#08110b');
  ctx.fillStyle=bg;ctx.fillRect(0,0,W,H);

  ctx.save();ctx.translate(shx,shy);

  /* ---- field base ---- */
  // home strip
  ctx.fillStyle='#24361f';ctx.fillRect(ox,oy,S*COLS,S);
  // river
  for(let r=RIVER_TOP;r<=RIVER_BOT;r++){
    ctx.fillStyle=r%2?'#0f3441':'#123c4a';ctx.fillRect(ox,oy+r*S,S*COLS,S);
  }
  // median
  ctx.fillStyle='#5d6a66';ctx.fillRect(ox,oy+MEDIAN*S,S*COLS,S);
  // road
  ctx.fillStyle='#2b3034';ctx.fillRect(ox,oy+ROAD_TOP*S,S*COLS,(ROAD_BOT-ROAD_TOP+1)*S);
  for(let r=ROAD_TOP;r<ROAD_BOT;r++){
    ctx.strokeStyle='rgba(255,201,94,.28)';ctx.lineWidth=2;ctx.setLineDash([9,9]);
    ctx.beginPath();ctx.moveTo(ox,oy+(r+1)*S);ctx.lineTo(ox+S*COLS,oy+(r+1)*S);ctx.stroke();ctx.setLineDash([]);
  }
  // grass
  for(let c=0;c<COLS;c++){ctx.fillStyle=(c%2)?'#3e7d3a':'#468a41';ctx.fillRect(ox+c*S,oy+GRASS*S,S,S);}
  speckle.forEach(s=>{
    const inMed=s.y>0.5;
    ctx.fillStyle=inMed?'rgba(255,255,255,.14)':'rgba(20,50,20,.3)';
    if(inMed)ctx.fillRect(ox+s.x*S*COLS,oy+MEDIAN*S+s.y*.5*S,S*.12,S*.12);
    else ctx.fillRect(ox+s.x*S*COLS,oy+GRASS*S+(1-s.y)*.5*S,S*.12,S*.12);
  });

  // water shimmer
  ctx.fillStyle='rgba(200,240,255,.07)';
  for(let r=RIVER_TOP;r<=RIVER_BOT;r++){
    shimmer[r-RIVER_TOP].forEach(st=>{
      const x=ox+((st.x+now*.012)% (S*COLS+S))-S/2;
      ctx.beginPath();ctx.ellipse(x,oy+r*S+S*(.3+.5*Math.abs(Math.sin(now*.001+st.ph))),st.w,S*.09,0,0,7);ctx.fill();
    });
  }

  /* ---- lane flow strips + ghosts ---- */
  G.lanes.forEach(l=>{
    const y=oy+l.row*S+3;
    ctx.fillStyle=l.type==='log'?'rgba(159,232,255,.16)':'rgba(255,210,122,.16)';
    let base=(now*.03*l.speed)%10;if(l.dir<0)base=10-base;
    for(let x=ox-10+base;x<ox+S*COLS;x+=10)ctx.fillRect(x,y,5,2);
  });
  if((G.phase==='idle'||G.phase==='player-mov')&&G.lives>0){
    G.lanes.forEach(l=>{if(Math.abs(l.row-G.frog.y)<=1)l.vehicles.forEach(v=>drawGhostCells(l.row,norm(v.x+l.dir*l.speed),v.len));});
    // valid-step dots
    ctx.fillStyle='rgba(234,255,176,.16)';
    [[0,-1],[0,1],[-1,0],[1,0]].forEach(d=>{
      const nx=G.frog.x+d[0],ny=G.frog.y+d[1];
      if(nx<0||nx>=COLS||ny<0||ny>=ROWS)return;
      if(ny===0&&!PAD_COLS.includes(nx))return;
      ctx.beginPath();ctx.arc(ox+nx*S+S/2,oy+ny*S+S/2,S*.07,0,7);ctx.fill();
    });
  }

  /* ---- vehicles ---- */
  const inT=(now-G.trafficT0<260)&&(G.phase==='traffic'||G.phase==='menu'||G.phase==='gameover');
  const p=inT?easeOut(clamp((now-G.trafficT0)/260,0,1)):1;
  G.lanes.forEach(l=>{
    l.vehicles.forEach(v=>{
      const pos=inT?norm(v.prevX+l.dir*l.speed*p):v.x;
      if(l.type==='log')drawLog(l,v,pos);else drawCar(l,v,pos);
    });
  });

  /* ---- pads ---- */
  PAD_COLS.forEach((c,i)=>{
    const x=ox+c*S+S/2,y=oy+S*.55;
    if(!G.pads[i]){
      const a=.35+.25*Math.sin(now/480+i*1.3);
      ctx.save();ctx.shadowColor='rgba(140,255,160,'+a+')';ctx.shadowBlur=12;
      ctx.fillStyle='#3f8f4a';ctx.beginPath();ctx.arc(x,y,S*.4,0,7);ctx.fill();ctx.restore();
      ctx.strokeStyle='rgba(127,217,126,'+(.5+.3*Math.sin(now/480+i*1.3))+')';ctx.lineWidth=2;
      ctx.beginPath();ctx.arc(x,y,S*.28,0,7);ctx.stroke();
    }else{
      ctx.fillStyle='#356b3c';ctx.beginPath();ctx.arc(x,y,S*.4,0,7);ctx.fill();
      drawFrog(x,y,S*.8,{happy:true});
    }
  });

  /* ---- the frog ---- */
  const showFrog=!(G.phase==='menu'||G.phase==='gameover');
  if(showFrog){
    let fx=frogPx(),fy=frogPy(),rot=0,alpha=1,sx=1,sy=1,hopY=0;
    if(G.phase==='player-mov'&&G.anim){
      const q=easeOut(clamp((now-G.anim.t0)/G.anim.dur,0,1));
      fx=ox+lerp(G.anim.ax0,G.anim.ax1,q)*S+S/2;
      fy=oy+lerp(G.anim.ay0,G.anim.ay1,q)*S+S/2 - Math.sin(q*Math.PI)*S*.3;
      sy=1-.14*Math.sin(q*Math.PI);sx=1+.14*Math.sin(q*Math.PI);
    } else if(G.phase==='traffic'&&G.ride){
      fx=ox+norm(G.frogPrevX+G.ride.l.dir*G.ride.l.speed*p)*S+S/2;
      fy=oy+G.frog.y*S+S/2;
    } else if(G.phase==='dying'){
      const d=G.deathAnchor||{x:fx,y:fy},q=clamp((now-G.deadT0)/DEATH_DUR[G.lastDeath],0,1);
      fx=d.x;fy=d.y;
      if(G.lastDeath==='water'){fy+=q*S*.4;alpha=1-q;sx=sy=1-q*.5;}
      else if(G.lastDeath==='car'){fy-=Math.sin(Math.min(q*1.6,1)*Math.PI)*S*.9;rot=q*Math.PI*2;alpha=q>.75?1-(q-.75)/.25:1;}
    }
    const riding=(G.frog.y>=RIVER_TOP&&G.frog.y<=RIVER_BOT)&&(G.phase==='idle'||G.phase==='traffic');
    if(riding&&G.phase!=='dying'){ctx.fillStyle='rgba(0,0,0,.25)';ctx.beginPath();ctx.ellipse(fx,fy+S*.3,S*.3,S*.1,0,0,7);ctx.fill();}
    let bs=1;
    if(now-G.bumpT<160){const bq=Math.sin((now-G.bumpT)/160*Math.PI);sx*=1+.18*bq;sy*=1-.18*bq;}
    drawFrog(fx,fy+S*.02,S*.86,{dx:G.frog.dx,dy:G.frog.dy,alpha,sx:sx*bs,sy:sy*bs,rot});
    if(G.phase==='idle'){
      const a=.22+.12*Math.sin(now/300);
      ctx.strokeStyle='rgba(234,255,176,'+a+')';ctx.lineWidth=2;
      ctx.beginPath();ctx.arc(fx,fy,S*.55+S*.04*Math.sin(now/300),0,7);ctx.stroke();
    }
  }

  /* ---- particles ---- */
  G.particles=G.particles.filter(pt=>{
    pt.life+=dt;const q=pt.life/pt.max;if(q>=1)return false;
    if(pt.t==='dot'){pt.x+=pt.vx*dt*.06;pt.y+=pt.vy*dt*.06;pt.vy+=(pt.g||0)*dt*.06;}
    ctx.globalAlpha=(1-q)*(pt.a||1);
    if(pt.t==='ring'){ctx.strokeStyle=pt.color;ctx.lineWidth=2.5;ctx.beginPath();ctx.arc(pt.x,pt.y,lerp(pt.r0,pt.r1,q),0,7);ctx.stroke();}
    else{ctx.fillStyle=pt.color||'#fff';ctx.beginPath();ctx.arc(pt.x,pt.y,pt.r*(1-q*.4),0,7);ctx.fill();}
    ctx.globalAlpha=1;return true;
  });

  /* ---- popups ---- */
  G.popups=G.popups.filter(pp=>{
    const q=(now-pp.t0)/pp.dur;if(q>=1)return false;
    ctx.font=FMONO(pp.size);ctx.textAlign='center';
    ctx.globalAlpha=1-q*q;
    ctx.strokeStyle='rgba(6,14,9,.9)';ctx.lineWidth=4;ctx.strokeText(pp.text,pp.x,pp.y-q*26);
    ctx.fillStyle=pp.color;ctx.fillText(pp.text,pp.x,pp.y-q*26);
    ctx.globalAlpha=1;return true;
  });

  /* ---- field frame ---- */
  ctx.strokeStyle='#0a120c';ctx.lineWidth=4;ctx.strokeRect(ox-2,oy-2,S*COLS+4,S*ROWS+4);
  ctx.strokeStyle='rgba(143,227,107,.18)';ctx.lineWidth=1.5;ctx.strokeRect(ox+2,oy+2,S*COLS-4,S*ROWS-4);

  /* ---- fireflies ---- */
  fireflies.forEach(f=>{
    const x=f.bx+Math.sin(now*.0003*f.sp+f.ph)*46,y=f.by+Math.cos(now*.00023*f.sp+f.ph*1.7)*34;
    const a=.25+.3*Math.sin(now*.002+f.ph);
    ctx.save();ctx.shadowColor='rgba(255,230,140,.9)';ctx.shadowBlur=8;
    ctx.fillStyle='rgba(255,236,170,'+Math.max(.05,a)+')';
    ctx.beginPath();ctx.arc(x,y,f.r,0,7);ctx.fill();ctx.restore();
  });

  ctx.restore(); // shake

  /* ---- night overlay ---- */
  if(G.phase==='dying'&&G.lastDeath==='night'){
    const q=clamp((now-G.deadT0)/DEATH_DUR.night,0,1);
    const a=q<.5?q*2:(q>.8?(1-q)*5:1);
    ctx.fillStyle='rgba(3,6,10,'+clamp(a,0,1)*.94+')';ctx.fillRect(0,0,W,H);
    nightStars.forEach(s=>{ctx.globalAlpha=clamp(a,0,1)*(.4+.6*Math.sin(now*.003+s.x*20));ctx.fillStyle='#dfe8ff';
      ctx.beginPath();ctx.arc(s.x*W,s.y*H,s.r,0,7);ctx.fill();});
    ctx.globalAlpha=clamp(a,0,1);
    ctx.font=FDISP(Math.min(46,W*.07));ctx.textAlign='center';
    ctx.fillStyle='#ffc95e';ctx.fillText('NIGHT FALLS',W/2,H*.46);
    ctx.globalAlpha=1;
  }

  /* ---- level banner ---- */
  if(G.phase==='levelup'){
    const q=(now-G.luT0)/1700,a=clamp(Math.min(q*5,(1-q)*4),0,1);
    const bw=Math.min(W-40,560),bh=86,bx=W/2-bw/2+(1-easeOut(clamp(q*3,0,1)))*-70,by=oy+S*ROWS*.42;
    ctx.globalAlpha=a;
    ctx.fillStyle='rgba(10,20,14,.92)';rr(bx,by,bw,bh,6);ctx.fill();
    ctx.strokeStyle='#3c6a4d';ctx.lineWidth=3;rr(bx,by,bw,bh,6);ctx.stroke();
    ctx.fillStyle='#ffc95e';ctx.fillRect(bx,by,7,bh);
    ctx.textAlign='center';ctx.font=FDISP(36);ctx.fillStyle='#d3f0b6';
    ctx.fillText('WAVE '+G.level,W/2+(1-easeOut(clamp(q*3,0,1)))*-70+0,by+bh*.48);
    ctx.font=FMONO(12);ctx.fillStyle='#8fae92';
    ctx.fillText('traffic quickens · daylight shrinks',W/2,by+bh*.78);
    ctx.globalAlpha=1;
  }

  /* ============================== HUD ============================== */
  ctx.fillStyle='#0f1a13';ctx.fillRect(0,0,W,hudH);
  ctx.fillStyle='#2e5c3a';ctx.fillRect(0,hudH-2,W,2);

  ctx.textAlign='left';
  ctx.font=FMONO(10);ctx.fillStyle='#7fae7a';ctx.fillText('SCORE',18,20);
  ctx.font=FDISP(26);ctx.fillStyle='#f2f7e9';ctx.fillText(String(G.score).padStart(5,'0'),18,hudH-14);
  const bw=W>640?110:0;
  if(bw){ctx.font=FMONO(10);ctx.fillStyle='#7fae7a';ctx.fillText('BEST',bw+22,20);
    ctx.font=FDISP(17);ctx.fillStyle='#9db89c';ctx.fillText(String(G.best).padStart(5,'0'),bw+22,hudH-14);}

  // center: wave + daylight bar
  const cx=W/2;
  ctx.textAlign='center';ctx.font=FDISP(19);ctx.fillStyle='#ffc95e';
  ctx.fillText('WAVE '+G.level,cx,hudH*.46);
  const barW=Math.min(120,W*.2),barX=cx-barW/2,barY=hudH-13;
  G.dispFrac+=(clamp(G.turns/G.budget,0,1)-G.dispFrac)*Math.min(1,dt*.012);
  const fr=G.dispFrac;
  ctx.fillStyle='#26352b';rr(barX,barY-4,barW,8,4);ctx.fill();
  const mix=(a,b,t)=>Math.round(lerp(a,b,t));
  const col=fr>.5?`rgb(${mix(255,255,(1-fr)*2)},${mix(210,93,(1-fr)*2)},${mix(122,77,(1-fr)*2)})`
                 :`rgb(${mix(255,255,fr*2)},${mix(93,157,fr*2)},${mix(77,77,fr*2)})`;
  ctx.fillStyle=col;rr(barX,barY-4,Math.max(6,barW*fr),8,4);ctx.fill();
  if(G.turns<=8&&G.phase!=='menu'){ctx.strokeStyle='rgba(255,107,93,'+(.4+.4*Math.sin(now/160))+')';ctx.lineWidth=2;rr(barX-3,barY-7,barW+6,14,6);ctx.stroke();}
  ctx.fillStyle='#fff';ctx.beginPath();ctx.arc(barX+barW*fr,barY,4.5,0,7);ctx.fill();
  ctx.font=FMONO(11);ctx.textAlign='left';ctx.fillStyle=fr<.2?'#ff6b5d':'#cfeeb0';
  ctx.fillText(G.turns+' TURNS',barX+barW+10,barY+4);

  // lives
  for(let i=0;i<3;i++){
    const x=W-24-i*30,y=hudH/2;
    if(i<G.lives)drawFrog(x,y,26,{happy:true});
    else{ctx.strokeStyle='rgba(150,170,155,.5)';ctx.lineWidth=2.5;
      ctx.beginPath();ctx.arc(x,y-4,8,0,7);ctx.stroke();
      ctx.beginPath();ctx.moveTo(x-5,y-9);ctx.lineTo(x+5,y+1);ctx.moveTo(x+5,y-9);ctx.lineTo(x-5,y+1);ctx.stroke();}
  }
}

/* ============================== main loop ============================== */
function loop(now){
  // phase timers
  if(G.phase==='player-mov'&&G.anim&&now-G.anim.t0>=G.anim.dur)executePending();
  else if(G.phase==='traffic'&&now-G.trafficT0>=260)finalizeTraffic();
  else if(G.phase==='dying'&&now-G.deadT0>=DEATH_DUR[G.lastDeath])afterDeath();
  else if(G.phase==='home'&&now-G.homeT0>=950)afterHome();
  else if(G.phase==='levelup'&&now-G.luT0>=1700)G.phase='idle';
  render(now);
  requestAnimationFrame(loop);
}

resize();
buildLanes(1);            // living backdrop behind the start screen
document.getElementById('sBest').textContent=G.best;
requestAnimationFrame(loop);
