    return;
  }else if(p.state==='dead'){
    p.deathT+=dt; if(p.lives>0&&p.deathT>1.15)p.state='respawn'; return;
  }
  /* hazards (ready / moving) */
  const R=p.row;
  if(R>=WATER_R0&&R<=WATER_R1){
    const arr=waterPlats[R], pl=platAt(arr,p.cx);
    if(pl){p.cx+=pl.dx;p.col=Math.round((p.cx-TILE/2)/TILE);if(!platAt(arr,p.cx))kill(p,'splash');}
    else kill(p,'splash');
  }else{
    const rowsChk=[R];
    if(p.state==='moving'&&p.anim&&p.anim.tr!==R)rowsChk.push(p.anim.tr);
    for(const RR of rowsChk){
      if(RR>=ROAD_R0&&RR<=ROAD_R1&&Math.abs(p.cy-(RR*TILE+TILE/2))<=13){
        if(vehHit(roadVehs[RR],p.cx)){kill(p,'boom');break;}
      }
    }
  }
}
function platAt(arr,cx){for(const p of arr)if(cx>p.x+5&&cx<p.x+p.L-5)return p;return null;}
function vehHit(arr,cx){for(const v of arr)if(cx>v.x+3&&cx<v.x+v.L-3)return v;return null;}

function spawnSplash(x,y){

function spawnBoom(x,y){
  for(let i=0;i<20;i++){const a=Math.random()*Math.PI*2,s=60+Math.random()*190;
    particles.push({x:x,y:y,vx:Math.cos(a)*s,vy:Math.sin(a)*s,g:130,r:2+Math.random()*2.5,col:['#ffd166','#ff8c42','#ff5a3c'][i%3],t:0,life:.4+Math.random()*.4});}
  for(let i=0;i<7;i++)particles.push({smoke:true,x:x+(Math.random()-.5)*14,y:y,vx:(Math.random()-.5)*26,vy:-24-Math.random()*22,r:6+Math.random()*5,t:0,life:.8+Math.random()*.3});

function spawnSparkle(x,y){
  for(let i=0;i<14;i++)particles.push({x:x+(Math.random()-.5)*16,y:y,vx:(Math.random()-.5)*70,vy:-30-Math.random()*80,g:70,r:1.5+Math.random()*2,col:['#b6f28a','#ffe9a3','#ffffff'][i%3],t:0,life:.45+Math.random()*.4});
  particles.push({ring:true,x:x,y:y,t:0,life:.55,col:'182,242,138'});

function updateFX(dt){
  shake=Math.max(0,shake-dt*2.6);
  for(let i=particles.length-1;i>=0;i--){const q=particles[i];q.t+=dt;
    if(q.ring||q.smoke){if(q.smoke)q.y+=q.vy*dt;if(q.t>=q.life)particles.splice(i,1);}
    else{q.x+=q.vx*dt;q.y+=q.vy*dt;q.vy+=q.g*dt;if(q.t>=q.life)particles.splice(i,1);}}
  if(particles.length>420)particles.splice(0,particles.length-420);
  for(let i=popups.length-1;i>=0;i--){popups[i].t+=dt;if(popups[i].t>1.15)popups.splice(i,1);}
  rippleT-=dt;
  if(rippleT<=0){ripples.push({x:20+Math.random()*(W-40),y:40+Math.random()*176,t:0,dur:.9});rippleT=.5+Math.random();}
  for(let i=ripples.length-1;i>=0;i--){ripples[i].t+=dt;if(ripples[i].t>ripples[i].dur)ripples.splice(i,1);}
  for(const f of fireflies){
    f.x+=Math.sin(time*.4+f.ph)*dt*15; f.y+=Math.cos(time*.3+f.ph*1.7)*dt*10;
    if(f.x<0)f.x+=W; if(f.x>W)f.x-=W; if(f.y<26){f.y=26;} if(f.y>BOARD_H-24){f.y=BOARD_H-24;}
  }
  if(AC&&!muted){cricketT-=dt;if(cricketT<=0){sfx.chirp();cricketT=1.4+Math.random()*2.6;}}
}

function drawPlayer(p){
  if(p.state==='dead'){
    const t=Math.min(1,p.deathT/.5); if(t>=1)return;
    drawFrog(p.cx,p.cy,{col:p.colors.col,dark:p.colors.dark,belly:p.colors.belly,
      s:1-t*.9,alpha:1-t,rot:t*(p.deadCause==='splash'?3:6),aimx:p.aimX,aimy:p.aimY});
    return;
  }
  if(p.state==='respawn')return;
  let s=1,a=1,rot=0;
  if(p.state==='celebrate'){
    const t=p.celeT/.85,sink=Math.min(1,t/.4);
    s=1-sink*.72; a=t>.5?1-(t-.5)/.5:1;
  }else if(p.state==='ready'){
    const pe=1+((1-p.pop)*(-3.7)+p.pop*0); /* fallback */
    const c=1.7, tt=p.pop-1;
    s*=0.6+0.4*(1+(c+1)*tt*tt*tt+c*tt*tt);
    s*=1+Math.sin(time*3+p.id)*.02;
  }
  drawFrog(p.cx,p.cy,{col:p.colors.col,dark:p.colors.dark,belly:p.colors.belly,
    s:s,alpha:a,rot:rot,squash:p.squash,aimx:p.aimX,aimy:p.aimY,blink:Math.sin(time*1.9+p.id*2.3)>0.965});
}

  if(!map[code])return;
  if(down){p.heldStack.push(code);p.heldDir=map[code];}
  else{const i=p.heldStack.indexOf(code);if(i>=0)p.heldStack.splice(i,1);
    p.heldDir=p.heldStack.length?map[p.heldStack[p.heldStack.length-1]]:null;}
}
