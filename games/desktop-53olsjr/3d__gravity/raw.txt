    ob.push({kind:'block',x,z:ZFAR,w,h});

  }else if(r<0.9&&t>0.28){

function doCollide(){

function burst(x,y,n,cols,sp,life,size,grav){
  for(let i=0;i<n;i++){const a=Math.random()*6.283,v=(0.3+Math.random()*0.7)*sp;

}
function updateBall(dt){
  let dir=0;
  if(keys['ArrowLeft']||keys['KeyA'])dir-=1;
  if(keys['ArrowRight']||keys['KeyD'])dir+=1;
  if(pointer.on){const d=pointer.tx-ball.x;dir=Math.abs(d)<10?0:Math.sign(d)*Math.min(1,Math.abs(d)/70);}
  ball.vx+=(dir*STEER-ball.vx)*Math.min(1,dt*9);
  ball.x+=ball.vx*dt;
  const lim=TRKW/2-R-8;
  if(ball.x<-lim){ball.x=-lim;ball.vx=0;}
  if(ball.x>lim){ball.x=lim;ball.vx=0;}
  ball.rot+=speed*dt*0.012;
  if(ball.flyT<1){
    ball.flyT+=dt/FLIPT;const t=Math.min(1,ball.flyT),e=t*t*(3-2*t);
    ball.y=ball.yFrom+(ball.yTo-ball.yFrom)*e+Math.sin(Math.PI*t)*26;
    if(t>=1){
      ball.y=surfY(ball.g);ball.sq=0.5;sh=Math.max(sh,2);sfx.land();
      const p=P(ball.x,ball.y,ZBALL);burst(p[0],p[1],8,['#9fd4ff','#ffffff'],120,0.5,3,60);
    }
  }else ball.y=surfY(ball.g);
  ball.sq+=(0-ball.sq)*Math.min(1,dt*8);
}
function step(dt){
  time+=dt;droneUpdate();
  const sp=(state==='playing'||state==='dead')?speed:260,maxR=Math.hypot(W,H)*0.75;
  for(const st of stars){st.r+=dt*(18+sp*0.055)*st.d;st.a+=dt*0.02*st.d;if(st.r>maxR)st.r=6+Math.random()*30;}
  sh*=Math.exp(-5.5*dt);flashA=Math.max(0,flashA-dt*1.6);
  if(state==='paused')return;
  if(state==='playing'){
    speed=Math.min(SMAX,speed+ACC*dt);dist+=speed*dt;
    invuln=Math.max(0,invuln-dt);hintT-=dt;
    if(combo>0){comboT-=dt;if(comboT<=0)combo=0;}
    for(const o of ob)o.z-=speed*dt;
    for(const o of orbs)o.z-=speed*dt;
    spawnIn-=speed*dt;if(spawnIn<=0)spawnPattern();
    updateBall(dt);doCollide();doCollect();
  }else if(state==='dead'){
    deadT+=dt;speed*=Math.exp(-2.8*dt);const dz=speed*dt;
    for(const o of ob)o.z-=dz;for(const o of orbs)o.z-=dz;
    if(deadT>0.85&&!deadShown){deadShown=true;showDead();}
  }else if(state==='menu'){
    menuT+=dt;ball.x=Math.sin(menuT*0.55)*170;ball.y=TRKH/2+Math.sin(menuT*1.3)*95;ball.rot+=dt*8;
  }
  for(let i=ob.length-1;i>=0;i--)if(ob[i].z<-140)ob.splice(i,1);
  for(let i=orbs.length-1;i>=0;i--)if(orbs[i].z<-140)orbs.splice(i,1);
  for(let i=parts.length-1;i>=0;i--){const p=parts[i];p.t+=dt;
    if(p.t>p.dur){parts.splice(i,1);continue;}p.vy+=p.g*dt;p.x+=p.vx*dt;p.y+=p.vy*dt;}
  for(let i=pops.length-1;i>=0;i--){const p=pops[i];p.t+=dt;if(p.t>p.dur)pops.splice(i,1);}
}
