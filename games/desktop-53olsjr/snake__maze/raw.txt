/* ---------- seeded rng ---------- */
function mulberry32(a){return function(){a|=0;a=a+0x6D2B79F5|0;let t=Math.imul(a^a>>>15,1|a);
  t=t+Math.imul(t^t>>>7,61|t)^t;return((t^t>>>14)>>>0)/4294967296;};}

/* ============================== AUDIO ============================== */

/* ============================== MAZE ============================== */
const idx=(x,y)=>y*MW+x;
const inB=(x,y)=>x>=0&&y>=0&&x<MW&&y<MH;

function genMaze(rw,rh,braid,rng){
  MW=2*rw+1; MH=2*rh+1; WALLS=new Uint8Array(MW*MH).fill(1);
  const vis=new Uint8Array(rw*rh), stack=[[0,0]]; vis[0]=1;
  const D=[[1,0],[-1,0],[0,1],[0,-1]];
  while(stack.length){
    const cur=stack[stack.length-1], cx=cur[0], cy=cur[1];
    let opts=[];
    for(const d of D){ const nx=cx+d[0], ny=cy+d[1];
      if(nx>=0&&ny>=0&&nx<rw&&ny<rh&&!vis[ny*rw+nx]) opts.push([nx,ny,d]); }
    if(!opts.length){ stack.pop(); continue; }
    const [nx,ny,d]=opts[(rng()*opts.length)|0];
    vis[ny*rw+nx]=1;
    WALLS[idx(2*cx+1+d[0], 2*cy+1+d[1])]=0;
    stack.push([nx,ny]);
  }
  /* braid: open some dead ends into loops */
  for(let ry=0;ry<rh;ry++)for(let rx=0;rx<rw;rx++){
    const tx=2*rx+1, ty=2*ry+1; let open=0;
    for(const d of D) if(!WALLS[idx(tx+d[0],ty+d[1])]) open++;
    if(open===1 && rng()<braid){
      const dirs=D.slice().sort(()=>rng()-.5);
      for(const d of dirs){
        const wx=tx+d[0], wy=ty+d[1], bx=tx+2*d[0], by=ty+2*d[1];
        if(inB(wx,wy)&&inB(bx,by)&&WALLS[idx(wx,wy)]&&!WALLS[idx(bx,by)]){ WALLS[idx(wx,wy)]=0; break; }
      }
    }
  }
}

function buildLevel(lv, attract){
  S.level=lv; S.attract=!!attract;
  const rw=Math.min(6+lv,13), rh=Math.min(4+lv,10);
  const braid=Math.min(.08+lv*.05,.32);
  S.seed = attract ? (Math.random()*1e9)>>>0 : ((lv*7919+421)*2654435761)>>>0;
  const rng=mulberry32(S.seed);
  genMaze(rw,rh,braid,rng);

  /* BFS from (1,1): distances + parents → farthest cell = portal */
  const N=MW*MH, dist=new Int32Array(N).fill(-1), par=new Int32Array(N).fill(-1);
  let si=idx(1,1); dist[si]=0; const q=[si];
  for(let h=0;h<q.length;h++){ const c=q[h], cx=c%MW, cy=(c/MW)|0;
    if(cx>0){const n=c-1;if(!WALLS[n]&&dist[n]<0){dist[n]=dist[c]+1;par[n]=c;q.push(n);}}
    if(cx<MW-1){const n=c+1;if(!WALLS[n]&&dist[n]<0){dist[n]=dist[c]+1;par[n]=c;q.push(n);}}
    if(cy>0){const n=c-MW;if(!WALLS[n]&&dist[n]<0){dist[n]=dist[c]+1;par[n]=c;q.push(n);}}
    if(cy<MH-1){const n=c+MW;if(!WALLS[n]&&dist[n]<0){dist[n]=dist[c]+1;par[n]=c;q.push(n);}}
  }
  let far=si,fv=-1;
  for(let i=0;i<N;i++) if(dist[i]>fv){fv=dist[i];far=i;}
  const path=[]; for(let c=far;c!==-1;c=par[c]) path.push({x:c%MW,y:(c/MW)|0});
  path.reverse();

  S.snake=path.slice(0,Math.max(2,Math.min(4,path.length)));
  while(S.snake.length<2)S.snake.push(S.snake[0]);
  S.dir={x:S.snake[1].x-S.snake[0].x, y:S.snake[1].y-S.snake[0].y};
  S.prevSnake=S.snake.map(c=>({x:c.x,y:c.y}));

  /* portal */
  S.portal={x:far%MW,y:(far/MW)|0}; S.portalActive=false;

  /* pellets */
  const startSet=new Set(S.snake.map(p=>idx(p.x,p.y)));
  let cand=[];
  for(let i=0;i<N;i++) if(!WALLS[i]&&i!==far&&!startSet.has(i)&&dist[i]>=6) cand.push({x:i%MW,y:(i/MW)|0});
  if(cand.length<12){ cand=[];
    for(let i=0;i<N;i++) if(!WALLS[i]&&i!==far&&!startSet.has(i)&&dist[i]>=3) cand.push({x:i%MW,y:(i/MW)|0}); }
  for(let i=cand.length-1;i>0;i--){const j=(rng()*(i+1))|0;const t=cand[i];cand[i]=cand[j];cand[j]=t;}

  /* per-run state */

  S.queue=[]; S.acc=0; S.shake=0; S.bumpFlash=0; S.dieT=0; S.banner=null;
  S.particles=[]; S.texts=[];

  updateHUD(true);
}

      g.fillStyle='hsl(196,26%,'+(10.5+rng()*2.5)+'%)'; g.fillRect(px,py,cell,cell);
      g.fillStyle='rgba(255,255,255,.028)';

    }
  }

}

/* ============================== STEP / RULES ============================== */
const DIRS=[{x:1,y:0},{x:-1,y:0},{x:0,y:1},{x:0,y:-1}];

function pushDir(d){
  const last=S.queue.length?S.queue[S.queue.length-1]:S.dir;
  if(d.x===last.x&&d.y===last.y)return;
  if(d.x===-last.x&&d.y===-last.y)return;
  if(S.queue.length<3)S.queue.push(d);
}

function autopilot(){
  const h=S.snake[0], opts=DIRS.filter(d=>{
    if(d.x===-S.dir.x&&d.y===-S.dir.y)return false;
    return inB(h.x+d.x,h.y+d.y)&&!WALLS[idx(h.x+d.x,h.y+d.y)];});
  if(opts.length)S.dir=opts[(Math.random()*opts.length)|0];
}

function step(){
  if(S.queue.length)S.dir=S.queue.shift(); else if(S.attract)autopilot();
  const h=S.snake[0], nx=h.x+S.dir.x, ny=h.y+S.dir.y;
  if(!inB(nx,ny)||WALLS[idx(nx,ny)]){ bumpFx(); return; }

  const pi=S.pellets.findIndex(p=>p.x===nx&&p.y===ny);
  const eating=pi!==-1;
  const willGrow=eating&&!S.attract||(eating&&S.attract&&S.snake.length<9);

  if(!S.attract){
    for(let i=0;i<S.snake.length;i++){
      if(i===S.snake.length-1&&!willGrow)continue; /* tail vacates */
      const s=S.snake[i];
      if(s.x===nx&&s.y===ny){ die('bite'); return; }
    }
  }
  const old=S.snake.map(c=>({x:c.x,y:c.y}));
  S.snake.unshift({x:nx,y:ny});
  if(!willGrow)S.snake.pop();
  S.prevSnake=old.slice(0,S.snake.length);
  while(S.prevSnake.length<S.snake.length)S.prevSnake.push(old[old.length-1]);

  S.stepParity^=1; AU.step();

  if(eating){
    S.pellets.splice(pi,1);
    const px=pOf({x:nx,y:ny});
    for(let k=0;k<12;k++){const a=Math.random()*TAU,sp=30+Math.random()*90;
      S.particles.push({x:px.x,y:px.y,vx:Math.cos(a)*sp,vy:Math.sin(a)*sp-30,
        life:.7,max:.7,size:cell*.1*(.5+Math.random()),col:'255,'+(140+(Math.random()*80|0))+',60',type:'ember'});}
    S.particles.push({x:px.x,y:px.y,vx:0,vy:0,life:.5,max:.5,size:cell*.9,col:'255,190,90',type:'ring'});
    if(S.attract){ updateHUD(); return; }
    S.motes++; const pts=25+S.level*5; S.score+=pts;
    S.fuel=Math.min(100,S.fuel+16);
    S.texts.push({x:px.x,y:px.y,txt:'+'+pts,life:1});
    AU.eat();
    if(S.motes>=S.total){ S.portalActive=true; AU.portal();
      S.banner={txt:'THE GATE OPENS',life:2.4,max:2.4}; }
  }
  updateHUD();
  if(S.portalActive&&nx===S.portal.x&&ny===S.portal.y){ clearLevel(); return; }
}

function bumpFx(){
  S.shake=Math.min(9,S.shake+5); S.bumpFlash=.35;
  const h=pOf(S.snake[0]);
  for(let k=0;k<6;k++)S.particles.push({x:h.x+S.dir.x*cell*.4,y:h.y+S.dir.y*cell*.4,
    vx:(Math.random()-.5)*120,vy:(Math.random()-.5)*120,life:.35,max:.35,size:1.6,col:'170,210,225',type:'spark'});
  if(S.time-S.lastBump>.25){ AU.bump(); S.lastBump=S.time; }
}

/* ============================== HELPERS ============================== */
function pOf(c){ return {x:ox+c.x*cell+cell/2, y:oy+c.y*cell+cell/2}; }
function lerpT(){ return Math.max(0,Math.min(1,S.acc/S.tickMs)); }
function segPos(seg,t){ const prev=S.prevSnake[S.prevSnake.indexOf? -1:-1]; /* replaced below */ return null; }
/* aligned interpolation: index i of new snake ↔ index i of old snake */
function interp(i,t){
  const cur=S.snake[i], prev=S.prevSnake[i]||cur;
  return {x:ox+(prev.x+(cur.x-prev.x)*t)*cell+cell/2, y:oy+(prev.y+(cur.y-prev.y)*t)*cell+cell/2};
}

/* ============================== UPDATE ============================== */
let last=performance.now();
function frame(ts){
  const dt=Math.min(.05,(ts-last)/1000); last=ts; S.time+=dt;

  if(S.mode==='playing'){
    S.fuel-=S.fuelRate*dt;
    if(S.fuel<=0){S.fuel=0;die('dark');}
    else if(S.fuel<28&&S.time-S.lowPulseT>1.15){AU.lowPulse();S.lowPulseT=S.time;}
    S.acc+=dt*1000;
    while(S.acc>=S.tickMs&&S.mode==='playing'){S.acc-=S.tickMs;step();}
  }else if(S.mode==='menu'){
    S.acc+=dt*1000;
    while(S.acc>=155){S.acc-=155;step();}
  }else if(S.mode==='dying'){
    S.dieT+=dt/1.1; if(S.dieT>=1)showOver();
  }

  /* particles */
  for(let i=S.particles.length-1;i>=0;i--){const p=S.particles[i];
    p.life-=dt; if(p.life<=0){S.particles.splice(i,1);continue;}
    if(p.type!=='ring'){p.x+=p.vx*dt;p.y+=p.vy*dt;p.vy+=(p.type==='ember'?-25:60)*dt;p.vx*= (1-dt*1.5);} }
  /* portal sparkle */
  if(S.portalActive&&(S.mode==='playing'||S.mode==='clear'||S.mode==='menu')&&Math.random()<dt*3){
    const pp=pOf(S.portal);
    S.particles.push({x:pp.x+(Math.random()-.5)*cell,y:pp.y+(Math.random()-.5)*cell,
      vx:(Math.random()-.5)*20,vy:-30-Math.random()*30,life:.8,max:.8,size:1.6,col:'140,230,255',type:'ember'});
  }
  /* dust */
  for(const d of S.dust){ d.x+=d.vx*dt; d.y+=d.vy*dt;
    if(d.x<ox)d.x=ox+MW*cell; if(d.x>ox+MW*cell)d.x=ox;
    if(d.y<oy)d.y=oy+MH*cell; if(d.y>oy+MH*cell)d.y=oy; }

  if(S.banner){S.banner.life-=dt;if(S.banner.life<=0)S.banner=null;}
  S.shake=Math.max(0,S.shake-dt*34);
  S.bumpFlash=Math.max(0,S.bumpFlash-dt*2.2);

  render();
  if(S.mode==='playing'||S.mode==='menu')updateHUD();
  requestAnimationFrame(frame);
}

/* ============================== RENDER ============================== */
function lightRadius(){
  const fn=S.fuel/100;
  const flick=1+.05*Math.sin(S.time*11)+.04*Math.sin(S.time*23.7)+.03*Math.sin(S.time*5.1);
  return cell*(2.3+2.8*fn)*flick;
}

}

function bodyCol(i,n){
  const f=n<=1?0:i/(n-1); /* 0 head → 1 tail */
  const r=Math.round(234-(234-24)*f), g=Math.round(255-(255-122)*f), b=Math.round(176-(176-77)*f);
  return 'rgb('+r+','+g+','+b+')';
}

const KEYMAP={ArrowLeft:{x:-1,y:0},KeyA:{x:-1,y:0},ArrowRight:{x:1,y:0},KeyD:{x:1,y:0},
              ArrowUp:{x:0,y:-1},KeyW:{x:0,y:-1},ArrowDown:{x:0,y:1},KeyS:{x:0,y:1}};

window.addEventListener('keydown',e=>{
  AU.ensure();
  const k=e.code;
  if(['ArrowLeft','ArrowRight','ArrowUp','ArrowDown','Space'].includes(k))e.preventDefault();
  if(e.repeat)return;
  if(KEYMAP[k]){ if(S.mode==='playing')pushDir(KEYMAP[k]); return; }
  switch(k){
    case 'Enter': case 'NumpadEnter': case 'Space':
      if(S.mode==='menu')startGame();
      else if(S.mode==='paused')pauseGame(false);
      else if(S.mode==='dying'){S.dieT=1;}
      else if(S.mode==='over')startGame();
      else if(S.mode==='clear')nextLevel();
      break;
    case 'KeyR':
      if(S.mode!=='menu')startGame();
      break;
    case 'KeyP': case 'Escape':
      if(S.mode==='playing')pauseGame(true);
      else if(S.mode==='paused')pauseGame(false);
      break;
    case 'KeyM': toggleMute(); break;
  }
});
