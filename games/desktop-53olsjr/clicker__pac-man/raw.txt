function makeSprite(g){ const i=GHOSTS.indexOf(g);
  return {cfg:g,sp:26+i*7+rand(0,8),x:rand(50,W-50),y:rand(50,H-50),
    tx:rand(50,W-50),ty:rand(50,H-50),rt:rand(.5,2),born:perfT,ph:rand(0,6)}; }
function syncSprites(){ const want=started?GHOSTS.filter(g=>S.ghosts[g.id]>0):GHOSTS.slice(0,3);
  sprites=sprites.filter(s=>want.includes(s.cfg));
  for(const g of want) if(!sprites.some(s=>s.cfg===g)) sprites.push(makeSprite(g)); }
function updateSprite(s,dt){ s.rt-=dt;
  let dx=s.tx-s.x, dy=s.ty-s.y, d=Math.hypot(dx,dy)||1;
  if(fright>0){ const mx=mouse.x-s.x,my=mouse.y-s.y,md=Math.hypot(mx,my);
    if(md<240&&md>1){ s.x-=mx/md*s.sp*2.6*dt; s.y-=my/md*s.sp*2.6*dt; }
    else { s.x+=dx/d*s.sp*dt; s.y+=dy/d*s.sp*dt; } }
  else { s.x+=dx/d*s.sp*dt; s.y+=dy/d*s.sp*dt; }
  if(d<10||s.rt<=0){ s.tx=rand(50,W-50); s.ty=rand(50,H-50); s.rt=rand(1.2,3); }
  s.x=clamp(s.x,34,W-34); s.y=clamp(s.y,34,H-34); }
function easeOutBack(e){ const c1=1.70158,c3=c1+1; return 1+c3*Math.pow(e-1,3)+c1*Math.pow(e-1,2); }
function drawGhost(s,t){ const e=clamp((t-s.born)/.3,0,1), sc=Math.max(.01,easeOutBack(e));
  const r=ghostR(), x=s.x, y=s.y+Math.sin(t*2+s.ph)*3;
  ctx.save(); ctx.translate(x,y); ctx.scale(sc,sc);
  const frighted=fright>0&&started;
  ctx.fillStyle=frighted?'#2b2bd6':s.cfg.color;
  ctx.beginPath(); ctx.arc(0,0,r,Math.PI,0);
  const sy=r*.82, n=4, tw=2*r/n;
  ctx.lineTo(r,sy);
  for(let i=0;i<n;i++){ ctx.lineTo(r-tw*(i+.5),sy+r*.3); ctx.lineTo(r-tw*(i+1),sy); }
  ctx.closePath(); ctx.fill();
  if(frighted){
    ctx.fillStyle='#e8f0ff';
    ctx.beginPath(); ctx.arc(-r*.34,-r*.1,r*.17,0,6.283); ctx.fill();
    ctx.beginPath(); ctx.arc(r*.34,-r*.1,r*.17,0,6.283); ctx.fill();
    ctx.strokeStyle='#e8f0ff'; ctx.lineWidth=2; ctx.beginPath();
    for(let i=0;i<=6;i++){ const px=-r*.5+i*r/3, py=r*.34+(i%2? r*.14:0); i?ctx.lineTo(px,py):ctx.moveTo(px,py); }
    ctx.stroke();
  } else {
    let lx=pac.x-x, ly=pac.y-y; const ld=Math.hypot(lx,ly)||1; lx/=ld; ly/=ld;
    ctx.fillStyle='#fff';
    ctx.beginPath(); ctx.ellipse(-r*.36,-r*.12,r*.26,r*.3,0,0,6.283); ctx.fill();
    ctx.beginPath(); ctx.ellipse(r*.36,-r*.12,r*.26,r*.3,0,0,6.283); ctx.fill();
    ctx.fillStyle='#10173f';
    ctx.beginPath(); ctx.arc(-r*.36+lx*r*.12,-r*.12+ly*r*.12,r*.13,0,6.283); ctx.fill();
    ctx.beginPath(); ctx.arc(r*.36+lx*r*.12,-r*.12+ly*r*.12,r*.13,0,6.283); ctx.fill();
  }
  ctx.restore(); }

/* ============================== PAC-MAN ============================== */
const pacR=()=>clamp(Math.min(W,H)*0.075,24,46);
function mouthOpen(t){ const idle=.16+.1*Math.sin(t*6);
  const extra=Math.sin((1-pac.chomp)*Math.PI)*.9; return clamp(Math.max(idle,extra),0,1); }
function drawPac(t){ const R=pacR(), bob=Math.sin(t*2.6)*4;
  ctx.save(); ctx.translate(pac.x,pac.y+bob); ctx.rotate(pac.ang);
  ctx.scale(1+squash*.12,1-squash*.14);
  ctx.shadowColor='rgba(255,216,0,.55)'; ctx.shadowBlur=16;
  const half=lerp(.10,1.25,mouthOpen(t));
  ctx.fillStyle='#ffd800'; ctx.beginPath(); ctx.moveTo(0,0);
  ctx.arc(0,0,R,half,Math.PI*2-half); ctx.closePath(); ctx.fill();
  ctx.restore(); }
function updatePac(dt){ const tx=clamp(mouse.x,70,W-70), ty=clamp(mouse.y,70,H-70);
  const k=1-Math.exp(-dt*2.2); pac.x+=(tx-pac.x)*k; pac.y+=(ty-pac.y)*k;
  pac.x=clamp(pac.x,60,W-60); pac.y=clamp(pac.y,60,H-60);
  if(Math.hypot(mouse.x-pac.x,mouse.y-pac.y)>18) pac.ang=Math.atan2(mouse.y-pac.y,mouse.x-pac.x);
  pac.chomp=Math.max(0,pac.chomp-dt/.22); squash=Math.max(0,squash-dt*5); }

let achT=0;
function gameUpdate(dt){ comboT-=dt; if(comboT<=0)combo=0;
  if(fright>0){ fright-=dt; if(fright<=0){ fright=0; sirenStop(); } }
  const pay=crewPay()*dt; if(pay>0)gain(pay);
  updatePellet(dt); achT-=dt; if(achT<=0){ checkAch(); achT=.3; } }
function alwaysUpdate(dt){ sprites.forEach(s=>updateSprite(s,dt)); updatePac(dt);
  for(let i=floats.length-1;i>=0;i--){ floats[i].age+=dt; if(floats[i].age>floats[i].dur)floats.splice(i,1); }
  for(let i=parts.length-1;i>=0;i--){ const p=parts[i]; p.life-=dt;
    if(p.life<=0){ parts.splice(i,1); continue; } p.vy+=420*dt; p.x+=p.vx*dt; p.y+=p.vy*dt; }
  for(let i=rips.length-1;i>=0;i--){ rips[i].age+=dt; if(rips[i].age>rips[i].dur)rips.splice(i,1); } }
function drawPellet(t){ if(!pellet.active)return;
  const blink=pellet.life<3?(Math.sin(t*14)>0?1:.25):1;
  ctx.globalAlpha=blink;
  const pr=13+2.5*Math.sin(t*7), x=pellet.x, y=pellet.y;
  const g=ctx.createRadialGradient(x,y,1,x,y,pr*2.6);
  g.addColorStop(0,'rgba(255,244,180,.9)'); g.addColorStop(.4,'rgba(255,220,80,.35)'); g.addColorStop(1,'rgba(255,220,80,0)');
  ctx.fillStyle=g; ctx.beginPath(); ctx.arc(x,y,pr*2.6,0,6.283); ctx.fill();
  ctx.fillStyle='#fff'; ctx.beginPath(); ctx.arc(x,y,pr*.55,0,6.283); ctx.fill();
  ctx.strokeStyle='rgba(255,255,255,.8)'; ctx.lineWidth=2;
  ctx.setLineDash([6,9]); ctx.lineDashOffset=-t*40;
  ctx.beginPath(); ctx.arc(x,y,pr+7,0,6.283); ctx.stroke(); ctx.setLineDash([]);
  ctx.globalAlpha=1; }
