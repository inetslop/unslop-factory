    if(pts2){scores[b.lastHit]+=pts2;pops.push({x:pc,y:pr-14*U,txt:'+'+pts2,col:'#ffffff',t:0,life:0.9});}
    pops.push({x:pc,y:pr,txt:b.fright>0?'FRIGHT! 2X PTS':'FRIGHT!',col:'#8f8fff',t:0,life:1});
    sfx.power();shake=Math.max(shake,4*U);
    power.timer=6+Math.random()*8;newPowerCell(pc,pr);
  }
}
function newPowerCell(ox,oy){
  for(var tries=0;tries<50;tries++){
    var c=2+((Math.random()*(COLS-4))|0),r=1+((Math.random()*(ROWS-2))|0);
    if(Math.abs(cellX(c)-ox)+Math.abs(cellY(r)-oy)<W*0.25)continue;
    power.c=c;power.r=r;return;
  }
}

/* ============================== GOAL / POINT ============================== */
function goal(scorer){
  lastScorer=scorer;
  var pts=b.fright>0?2:1;
  scores[scorer]+=pts;
  if(rally>bestRally)bestRally=rally;
  saveBest();
  shake=13*U;flashT=0.35;flashSide=scorer;
  var gx=scorer===0?W-40*U:40*U;
  pops.push({x:gx,y:H/2,txt:'+'+pts,col:ACC[scorer],t:0,life:1.1});
  for(var k=0;k<26;k++)parts.push({x:gx,y:b.y,vx:(scorer===0?-1:1)*(60+Math.random()*300)*U,vy:(Math.random()-0.5)*400*U,
    life:0.6,max:0.6,sz:2+Math.random()*3,col:k%2?b.color:'#ffffff'});
  if(!b.demo)sfx.score();
  pendingWinner=scores[scorer]>=WINSCORE?scorer:-1;
  state='point';stateT=0;
}

/* ============================== UPDATE ============================== */
function update(dt){
  t+=dt;
  shake=Math.max(0,shake-dt*40);
  flashT=Math.max(0,flashT-dt);
  for(var i=parts.length-1;i>=0;i--){var q=parts[i];q.life-=dt;if(q.life<=0){parts.splice(i,1);continue;}
    q.x+=q.vx*dt;q.y+=q.vy*dt;q.vx*=(1-2.5*dt);q.vy*=(1-2.5*dt);}
  for(i=pops.length-1;i>=0;i--){pops[i].t+=dt;if(pops[i].t>pops[i].life)pops.splice(i,1);}

  if(state==='pause')return;

  if(state==='menu'){
    ctrlPaddles(dt);
    if(b.demo)physics(dt);
    else serve(undefined,true);
    return;
  }
  if(state==='serve'){
    ctrlPaddles(dt);
    stateT+=dt;
    var num=Math.ceil((SERVE_T-stateT)/(SERVE_T/3));
    if(num!==update._n){update._n=num;if(!b.demo)sfx.count();}
    b.x=W/2+Math.sin(t*4)*5*U;b.y=H/2+Math.cos(t*3.1)*7*U;
    if(stateT>=SERVE_T){state='play';if(!b.demo)sfx.go();}
    return;
  }
  if(state==='play'){
    ctrlPaddles(dt);
    physics(dt);
    power.timer-=dt;
    if(power.timer<=0)newPowerCell(cellX(power.c),cellY(power.r));
    if(b.x<-26*U)goal(1);
    else if(b.x>W+26*U)goal(0);
    return;
  }
  if(state==='point'){
    ctrlPaddles(dt);stateT+=dt;
    if(stateT>1.25){
      if(pendingWinner>=0){winner=pendingWinner;pendingWinner=-1;state='over';stateT=0;confT=0;sfx.win();}
      else serve(lastScorer===0?1:-1,false);
    }
    return;
  }
  if(state==='over'){
    stateT+=dt;confT-=dt;
    if(confT<=0){confT=0.07;
      var cols=['#ffe600','#ff2b2b','#ffb8ff','#00ffff','#ffb852'];
      parts.push({x:Math.random()*W,y:-10,vx:(Math.random()-0.5)*40*U,vy:(60+Math.random()*120)*U,
        life:4,max:4,sz:2+Math.random()*3,col:cols[(Math.random()*cols.length)|0]});}
  }
}

function drawPac(x,y,r,face,mouth,col){

}
function drawGhostShape(x,y,r,col,fright,flash,jit){
