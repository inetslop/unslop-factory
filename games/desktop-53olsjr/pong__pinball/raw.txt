  b.vy+=G*h;
  const sp=Math.hypot(b.vx,b.vy);if(sp>1200){b.vx*=1200/sp;b.vy*=1200/sp;}
  b.x+=b.vx*h;b.y+=b.vy*h;
  // arch + shoulders
  if(b.y<ARCH.y){
    if(Math.abs(b.x-ARCH.x)<=ARCH.r){
      const dx=b.x-ARCH.x,dy=b.y-ARCH.y,d=Math.hypot(dx,dy),lim=ARCH.r-b.r;
      if(d>lim&&d>0){const nx=dx/d,ny=dy/d;b.x=ARCH.x+nx*lim;b.y=ARCH.y+ny*lim;
        const dot=b.vx*nx+b.vy*ny;if(dot>0){b.vx-=2*dot*nx;b.vy-=2*dot*ny;S.wall();}}
    }else if(b.y-b.r<ARCH.y){b.y=ARCH.y+b.r;if(b.vy<0){b.vy=-b.vy*.98;S.wall();}}
  }
  if(b.x<b.r){b.x=b.r;if(b.vx<0){b.vx=-b.vx*.98;S.wall();}}
  if(b.x>VW-b.r){b.x=VW-b.r;if(b.vx>0){b.vx=-b.vx*.98;S.wall();}}
  // lane divider
  if(b.y>LANE_TOP-b.r&&Math.abs(b.x-LANE_X)<b.r){
    if(b.x<LANE_X){b.x=LANE_X-b.r;if(b.vx>0)b.vx*=-.98;}else{b.x=LANE_X+b.r;if(b.vx<0)b.vx*=-.98;}}
  // plunger knob
  {const kx=572,ky=knobTopY()+14,dx=b.x-kx,dy=b.y-ky,d=Math.hypot(dx,dy),lim=b.r+13;
   if(d<lim&&d>0.001){const nx=dx/d,ny=dy/d;b.x=kx+nx*lim;b.y=ky+ny*lim;
     const dot=b.vx*nx+b.vy*ny;if(dot<0){b.vx-=2*dot*nx;b.vy-=2*dot*ny;}}}
  // bumpers
  for(const u of bumpers){const dx=b.x-u.x,dy=b.y-u.y,d=Math.hypot(dx,dy),lim=u.r+b.r;
    if(d<lim&&d>0.001){const nx=dx/d,ny=dy/d;b.x=u.x+nx*lim;b.y=u.y+ny*lim;
      const kick=Math.min(760,430+combo*8),tx=-ny,ty=nx,dotT=b.vx*tx+b.vy*ty;
      b.vx=nx*kick+tx*dotT*.35;b.vy=ny*kick+ty*dotT*.35;
      u.f=1;combo++;addScore(100*mult(),b.x,b.y-6,u.c);burst(b.x,b.y,u.c,9,240);
      rings.push({x:u.x,y:u.y,r:u.r,t:0,life:.35,c:u.c});S.bump(u.i);shake=Math.min(8,shake+1.5);}}
  // slings
  for(const s of slings){const q=segClosest(b.x,b.y,s.a,s.b),dx=b.x-q[0],dy=b.y-q[1],d=Math.hypot(dx,dy);
    if(d<b.r&&d>0.001){const nx=dx/d,ny=dy/d;b.x=q[0]+nx*b.r;b.y=q[1]+ny*b.r;
      const dot=b.vx*nx+b.vy*ny;if(dot<0){b.vx-=2*dot*nx;b.vy-=2*dot*ny;}
      b.vx+=s.kick[0]*300;b.vy+=s.kick[1]*300;s.f=1;combo++;
      addScore(50*mult(),b.x,b.y-6,'#ffd166');burst(b.x,b.y,'#ffd166',6,200);S.sling();shake=Math.min(8,shake+1);}}
  // light bank targets (drop on hit — no bounce)
  for(const t of targets){if(!t.up)continue;
    const q=segClosest(b.x,b.y,[t.x-t.tx*13,t.y-t.ty*13],[t.x+t.tx*13,t.y+t.ty*13]);
    if(Math.hypot(b.x-q[0],b.y-q[1])<b.r){t.up=false;t.dropT=.001;combo++;
      addScore(500*mult(),t.x,t.y+16,'#b8f04d');burst(t.x,t.y,'#eaffc8',12,260);S.target();shake=Math.min(8,shake+2);
      if(targets.every(v=>!v.up)){addScore(1500,300,220,'#b8f04d');showMsg('LIGHTS OUT +1500',1.5);S.bonus();puTimer=Math.min(puTimer,2.5);} }}
  // paddle
  {const p=paddle,half=p.w/2;
   if(b.vy>0&&b.y+b.r>p.y&&b.y-b.r<p.y+p.h+6&&Math.abs(b.x-p.x)<half+b.r*.5){
     b.y=p.y-b.r;const off=clamp((b.x-p.x)/half,-1,1);
     b.vy=-Math.max(Math.abs(b.vy)*.96,330);b.vx=b.vx*.35+off*280+paddle.vx*.4;
     combo=0;S.paddle(off);burst(b.x,p.y,'#4be1ec',5,160);}}
}

function phys(h){
  // paddle
  const half=paddle.w/2,px=paddle.x;
  if(keys.l)paddle.tx-=640*h;if(keys.r)paddle.tx+=640*h;
  paddle.tx=clamp(paddle.tx,half+8,LANE_X-half-8);
  paddle.x=lerp(paddle.x,paddle.tx,1-Math.exp(-h*16));
  paddle.vx=(paddle.x-px)/h;
  if(charging&&state==='serve')charge=Math.min(1,charge+h*.85);
  // effect timers + wide lerp
  fx.wide=Math.max(0,fx.wide-h);fx.x2=Math.max(0,fx.x2-h);
  paddle.w=lerp(paddle.w,fx.wide>0?150:92,h*8);
  // powerups
  puTimer-=h;
  if(puTimer<=0){puTimer=rnd(8,13);if(state==='play'&&powerups.length<2)spawnPower();}
  for(let i=powerups.length-1;i>=0;i--){const u=powerups[i];u.t+=h;u.y+=u.vy*h;u.x+=Math.sin(u.t*1.4)*22*h;
    if(Math.abs(u.x-paddle.x)<paddle.w/2+12&&u.y>paddle.y-10&&u.y<paddle.y+p.h+16){applyPower(u);powerups.splice(i,1);continue;}
    if(u.y>VH+30)powerups.splice(i,1);}
  // balls
  for(const b of balls)ballStep(b,h);
  // drains
  for(let i=balls.length-1;i>=0;i--){const b=balls[i];
    if(b.y-b.r>VH+14){balls.splice(i,1);S.drain();shake=Math.min(9,shake+5);
      rings.push({x:b.x,y:VH-8,r:6,t:0,life:.5,c:'#ff6b57'});burst(b.x,VH-24,'#ff8a70',14,260);}}
  if(!balls.length){lifeBalls--;combo=0;
    if(lifeBalls<=0)gameOver();else{showMsg('BALL '+(4-lifeBalls));serveBall();}}
}

  if(a.y<ARCH.y&&Math.abs(a.x-300)<=270){const dx=a.x-300,dy=a.y-240,d=Math.hypot(dx,dy),lim=261;

/* ============================== frame update ============================== */
function update(dtF){
  tGlobal+=dtF;
  for(const u of bumpers)u.f=Math.max(0,u.f-dtF*3.2);
  for(const s of slings)s.f=Math.max(0,s.f-dtF*3.5);
  for(const t of targets)if(!t.up&&t.dropT>0)t.dropT=Math.min(1,t.dropT+dtF*1.1);
  for(let i=parts.length-1;i>=0;i--){const p=parts[i];p.t+=dtF;p.x+=p.vx*dtF;p.y+=p.vy*dtF;p.vy+=300*dtF;if(p.t>p.life)parts.splice(i,1);}
  for(let i=rings.length-1;i>=0;i--){rings[i].t+=dtF;if(rings[i].t>rings[i].life)rings.splice(i,1);}
  for(let i=floats.length-1;i>=0;i--){floats[i].t+=dtF;if(floats[i].t>floats[i].life)floats.splice(i,1);}
  if(msg){msg.t+=dtF;if(msg.t>msg.life)msg=null;}
  shake*=Math.exp(-dtF*6);if(shake<.05)shake=0;
  for(const d of dust){d.y-=d.sp*dtF;d.x+=Math.sin(tGlobal*.4+d.ph)*8*dtF;if(d.y<-10){d.y=ch+10;d.x=rnd(cw);}}
  if(state==='menu')attStep(dtF);
  if(paused)return;
  if(state==='play'||state==='serve'){
    acc+=Math.min(dtF,.05);
    while(acc>=H){phys(H);acc-=H;if(state!=='play'&&state!=='serve')break;}
    for(const b of balls){b.trail.push({x:b.x,y:b.y});if(b.trail.length>12)b.trail.shift();}
    // anti-stuck / lane rest
    for(const b of balls){const sp=Math.hypot(b.vx,b.vy);
      if(sp<40)b.still+=dtF;else b.still=0;
      if(b.still>1.7){b.still=0;
        if(b.x>LANE_X&&balls.length===1)serveBall();
        else{const a2=rnd(-2.5,-.6);b.vx=Math.cos(a2)*430;b.vy=Math.sin(a2)*430;S.nudge();shake=Math.min(8,shake+2);}}}
  }else acc=0;
}
