  const osc=AC.createOscillator();osc.type=o.type||'square';

  const g=AC.createGain();g.gain.setValueAtTime(.0001,t0);
  g.gain.linearRampToValueAtTime(o.v||.2,t0+(o.attack||.008));
  g.gain.exponentialRampToValueAtTime(.0001,t0+o.dur);
  osc.connect(g).connect(master);osc.start(t0);osc.stop(t0+o.dur+.05);}

function typeAt(r,c){return (grid[r]&&grid[r][c])?grid[r][c].type:-1;}
function makesRun(r,c,ty){
  if(ty<0)return false;let n=1;
  for(let cc=c-1;cc>=0&&typeAt(r,cc)===ty;cc--)n++;
  for(let cc=c+1;cc<COLS&&typeAt(r,cc)===ty;cc++)n++;
  if(n>=3)return true;n=1;
  for(let rr=r-1;rr>=0&&typeAt(rr,c)===ty;rr--)n++;
  for(let rr=r+1;rr<ROWS&&typeAt(rr,c)===ty;rr++)n++;
  return n>=3;
}
function hasAnyMove(){
  for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++){
    const ty=typeAt(r,c);
    if(c+1<COLS){const nt=typeAt(r,c+1);if(makesRun(r,c,nt)||makesRun(r,c+1,ty))return true;}
    if(r+1<ROWS){const nt=typeAt(r+1,c);if(makesRun(r,c,nt)||makesRun(r+1,c,ty))return true;}}
  return false;
}
function forceMove(){

  const set=(rr,cc,v)=>{grid[rr][cc].type=v;paint(grid[rr][cc]);};
  set(r-1,c,X);set(r+1,c,X);set(r,c,Y);set(r,c+1,X);
  if(typeAt(r-2,c)===X)set(r-2,c,(X+1)%5);

}

/* ================= PROJECTILES ================= */
function fireProjectile(t,mult){
  const Ty=TYPES[t.type],rc=t.el.getBoundingClientRect();
  const sx=rc.left+rc.width/2,sy=rc.top+rc.height/2;
  let target=null,tx,ty;
  if(Ty.heal){tx=barrX+barrW*.5;ty=groundY+roadH*.35;}
  else{
    const alive=zombies.filter(z=>z.state!=='dying');
    if(alive.length){alive.sort((a,b)=>a.xf-b.xf);target=alive[fireCursor++%alive.length];}
    if(target){const lead=target.stun>0?0:target.speedFrac*.85;
      tx=(target.xf-lead)*W;ty=footY(target)-46*(H/760);}
    else{tx=W*rand(.35,.95);ty=groundY+roadH*.5;}
  }
  const d=Math.hypot(tx-sx,ty-sy),dur=clamp(d/1150,.26,.52);
  projectiles.push({sx,sy,tx,ty,t:0,dur,type:t.type,target,mult,trail:[],x:sx,y:sy});
}
function hitZombie(z,dmg,ptype,silent){
  if(!z||z.state==='dying')return;
  z.hp-=dmg;z.flash=1;
  const zx=z.xf*W;
  burst(zx,footY(z)-55*z.zs,TYPES[ptype].color,silent?5:9);
  dmgText(zx,footY(z)-96*(H/760),Math.round(dmg),TYPES[ptype].stun?'#4fd7ff':'#ffd23f');
  sfx.thump();
  if(TYPES[ptype].stun){z.stun=Math.max(z.stun||0,TYPES[ptype].stun);sfx.stun();}
  if(z.hp<=0)killZombie(z);
}
function killZombie(z){
  if(z.state==='dying')return;
  z.state='dying';z.deadT=0;
  const bt=ZT[z.ztype];addScore(bt.bonus);kills++;sfx.death();
  if(z.ztype==='brute'){shakeIt(4);burst(z.xf*W,footY(z)-40,'#c8ff5e',26);}else shakeIt(1.5);
  splats.push({x:z.xf*W,y:footY(z)+3,rx:15*z.zs,ry:5*z.zs,a:.55});
  if(splats.length>34)splats.shift();
  dmgText(z.xf*W,footY(z)-120*(H/760),'+'+bt.bonus,'#ffe27a',15);
  burst(z.xf*W,footY(z)-50*z.zs,'#8fd96a',14);
}
function impact(p){
  const Ty=TYPES[p.type];
  if(Ty.heal){barricade.hp=Math.min(BARR_MAX,barricade.hp+Ty.heal);
    burst(p.tx,p.ty,'#7dff9a',12);dmgText(p.tx,p.ty-14,'+'+Ty.heal,'#7dff9a',15);sfx.heal();return;}
  let z=(p.target&&zombies.includes(p.target)&&p.target.state!=='dying')?p.target:null;
  if(!z){let bd=1e9;
    for(const q of zombies){if(q.state==='dying')continue;
      const d=Math.hypot(q.xf*W-p.tx,footY(q)-40*q.zs-p.ty);if(d<bd){bd=d;z=q;}}
    if(bd>80){burst(p.tx,p.ty,'#5a6b52',6);noiseB({ft:'lowpass',f:500,dur:.1,v:.05});return;}}
  const dmg=Ty.dmg*p.mult;hitZombie(z,dmg,p.type,false);
  if(Ty.splash){burst(p.tx,p.ty,Ty.color,22);
    for(const q of zombies){if(q===z||q.state==='dying')continue;
      if(Math.abs(q.xf*W-z.xf*W)<115)hitZombie(q,dmg*.5,p.type,true);}}
}
function projectilesUpdate(dt){
  for(let i=projectiles.length-1;i>=0;i--){const p=projectiles[i];p.t+=dt/p.dur;
    if(p.t>=1){impact(p);projectiles.splice(i,1);continue;}
    const d=Math.hypot(p.tx-p.sx,p.ty-p.sy);
    p.x=p.sx+(p.tx-p.sx)*p.t;p.y=p.sy+(p.ty-p.sy)*p.t-Math.sin(Math.PI*p.t)*Math.min(150,d*.3);
    p.trail.push({x:p.x,y:p.y});if(p.trail.length>7)p.trail.shift();}
}

/* ================= ZOMBIES & WAVES ================= */
function footY(z){return groundY+roadH*[.3,.55,.85][z.lane];}
function spawnZombie(type){
  const d=ZT[type],wf=Math.min(1.7,1+(wave-1)*.035);
  const z={ztype:type,xf:rand(1.04,1.18),lane:ri(3),hp:d.hp+d.wHp*Math.max(1,wave),max:0,
    speedFrac:d.sp*wf*rand(.9,1.12),dps:d.dps,state:'walk',phase:rand(6),flash:0,stun:0,deadT:0,seed:Math.random()};
  z.max=z.hp;z.zs=(H/760)*[.82,1,1.14][z.lane]*d.size;
  zombies.push(z);return z;
}
function startWave(n){
  wave=n;updateWavePill(n);
  const count=Math.min(26,5+n*2);spawnQueue=[];
  for(let i=0;i<count;i++){let ty='walker';
    if(n>=2&&Math.random()<Math.min(.3,n*.05))ty='runner';
    if(n>=3&&Math.random()<Math.min(.2,n*.035))ty='brute';
    spawnQueue.push(ty);}
  spawnGap=clamp(1.3-n*.04,.5,1.3);spawnTimer=.6;
  banner('WAVE '+n,n===1?'hold the line':'the horde grows');
  tone({f:180,f2:320,dur:.25,type:'sawtooth',v:.12});
}
function waveUpdate(dt){
  if(waveInter>0){waveInter-=dt;if(waveInter<=0)startWave(wave+1);return;}
  if(spawnQueue.length){spawnTimer-=dt;
    while(spawnTimer<=0&&spawnQueue.length){spawnZombie(spawnQueue.shift());spawnTimer+=spawnGap*rand(.7,1.3);}}
  else if(!zombies.some(z=>z.state!=='dying')){
    waveInter=2.6;const bonus=120+wave*60;addScore(bonus);
    barricade.hp=Math.min(BARR_MAX,barricade.hp+8);
    banner('WAVE '+wave+' CLEARED','wall patched · +'+bonus+' pts');sfx.wave();
    dmgText(barrX+barrW/2,groundY+roadH*.35,'+8','#7dff9a',14);
    burst(barrX+barrW/2,groundY+roadH*.4,'#7dff9a',10);}
}
function zombiesUpdate(dt){
  for(let i=zombies.length-1;i>=0;i--){const z=zombies[i];
    if(z.state==='dying'){z.deadT+=dt;if(z.deadT>.95)zombies.splice(i,1);continue;}
    z.flash=Math.max(0,z.flash-dt*4);
    const stung=z.stun>0;if(stung)z.stun-=dt;
    if(!stung)z.xf-=z.speedFrac*dt*(state==='over'?.6:1);
    z.phase+=dt*((stung?2.5:4)+z.speedFrac*150);
    const gx=barrX+barrW+8;
    if(state!=='over'&&z.xf*W<=gx){
      if(z.state!=='gnaw'){z.state='gnaw';if(state==='playing'){sfx.gnawHit();shakeIt(2);}}
      else if(state==='playing'){
        barricade.hp-=z.dps*dt;barrFlash=Math.min(1,barrFlash+.08);
        if(T-lastCreak>.5){lastCreak=T;sfx.creak();}
        if(Math.random()<dt*7)parts.push({x:gx+rand(-4,6),y:footY(z)-40*z.zs,vx:rand(-30,10),vy:rand(-40,-10),g:260,t:0,life:.5,size:rand(1.5,3),c:'#ff6a4d'});
        if(barricade.hp<=0){barricade.hp=0;gameOver();}}
    }else if(z.state==='gnaw'&&z.xf*W>gx)z.state='walk';
    if(state==='menu'&&z.xf<-0.15){z.xf=rand(1.05,1.2);z.lane=ri(3);z.state='walk';}
    else if(z.xf<-0.18)zombies.splice(i,1);}
}

function startGame(){
  ensureAudio();gameId++;paused=false;
  pauseOv.classList.add('hidden');menu.classList.add('hidden');over.classList.add('hidden');
  hud.classList.remove('hide');panelEl.classList.remove('hidden');
  state='playing';score=0;shown=0;lastS=-1;kills=0;wave=0;
  barricade.hp=BARR_MAX;spawnQueue=[];waveInter=0;fireCursor=0;
  projectiles.length=0;parts.length=0;dtexts.length=0;splats.length=0;zombies.length=0;
  busy=false;selTile=null;drag=null;redFlash=0;barrFlash=0;newBestFlag=false;
  btnPause.textContent='❚❚';
  genBoard();boardEl.classList.add('live');updateWavePill(0);hudTick(true);
  banner('NIGHT 1','the horde is coming');
  const my=gameId;setTimeout(()=>{if(my===gameId)startWave(1);},950);
}
function gameOver(){
  if(state!=='playing')return;
  state='over';boardEl.classList.remove('live');clearSel();
  sfx.over();shakeIt(10);redFlash=1;
  burst(barrX+barrW/2,groundY+roadH*.4,'#8a734f',30);
  if(score>best){best=Math.round(score);saveBest();newBestFlag=true;}
  const my=gameId;
  setTimeout(()=>{if(my!==gameId)return;fillOverStats();over.classList.remove('hidden');},1200);
}

}

function drawProj(g,p){
  const Ty=TYPES[p.type];
  for(let i=0;i<p.trail.length;i++){const q=p.trail[i];
    g.globalAlpha=(i+1)/p.trail.length*.32;g.fillStyle=Ty.color;
    g.beginPath();g.arc(q.x,q.y,2+i*.45,0,TAU);g.fill();}
  g.globalAlpha=.32;g.beginPath();g.arc(p.x,p.y,9.5,0,TAU);g.fill();
  g.globalAlpha=1;
  if(Ty.heal){g.fillStyle='#e8f6ec';g.fillRect(p.x-5.5,p.y-5.5,11,11);
    g.fillStyle='#e33d3d';g.fillRect(p.x-2,p.y-4.5,4,9);g.fillRect(p.x-4.5,p.y-2,9,4);}
  else{g.fillStyle=Ty.color;g.beginPath();g.arc(p.x,p.y,4.6,0,TAU);g.fill();
    g.fillStyle='#fff';g.beginPath();g.arc(p.x-1,p.y-1,2,0,TAU);g.fill();}
  g.globalAlpha=1;
}

function initParticles(){
  stars=[];for(let i=0;i<90;i++)stars.push({fx:Math.random(),fy:Math.random()*.55,r:rand(.6,1.6),ph:rand(7),sp:rand(.5,2)});
  fogBlobs=[];for(let i=0;i<16;i++){const front=i<4;
    fogBlobs.push({x:rand(-200,W+200),fy:front?rand(.68,.9):rand(.55,.7),r:rand(70,190)*(W/1300+.5),spd:rand(4,16),a:front?rand(.05,.09):rand(.04,.08),ph:rand(7),front});}
  spores=[];for(let i=0;i<34;i++)spores.push({x:Math.random()*W,fy:rand(.1,.66),vx:-rand(3,10),ph:rand(9),r:rand(.8,1.8),tw:rand(1,3)});
}
function update(dt){
  for(const b of fogBlobs){b.x-=b.spd*dt;if(b.x<-b.r*3)b.x=W+b.r*2;}
  for(const s of spores){s.x+=s.vx*dt;if(s.x<-8){s.x=W+8;s.fy=rand(.1,.66);}}
  if(T>nextSky){skyFlash=1;nextSky=T+rand(11,26);if(AC&&!muted)noiseB({ft:'lowpass',f:95,dur:1.5,v:.12,delay:.4});}
  skyFlash=Math.max(0,skyFlash-dt*2.1);
  redFlash=Math.max(0,redFlash-dt*1.1);
  barrFlash=Math.max(0,barrFlash-dt*3);
  shootT=Math.max(0,shootT-dt);
  shakeMag=Math.max(0,shakeMag-dt*(10+shakeMag*4));
  if(AC&&!muted&&T>nextMoan){nextMoan=T+rand(9,22);moanSfx();}
  if(state==='menu')zombiesUpdate(dt);
  else if(state==='playing'){waveUpdate(dt);zombiesUpdate(dt);}
  else if(state==='over')zombiesUpdate(dt);
  projectilesUpdate(dt);
  for(let i=parts.length-1;i>=0;i--){const p=parts[i];p.t+=dt;
    if(p.t>p.life){parts.splice(i,1);continue;}p.vy+=p.g*dt;p.x+=p.vx*dt;p.y+=p.vy*dt;}
  for(let i=dtexts.length-1;i>=0;i--){const d=dtexts[i];d.t+=dt;d.y+=d.vy*dt;if(d.t>d.life)dtexts.splice(i,1);}
  for(let i=splats.length-1;i>=0;i--){splats[i].a-=dt*.055;if(splats[i].a<=0)splats.splice(i,1);}
}

}
