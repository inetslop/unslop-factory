Maze generation (recursive backtracker iterative):
```
function genMaze(cw, ch){ // cell counts
  const w = cw*2+1, h = ch*2+1;
  const g = new Uint8Array(w*h).fill(1);
  const stack=[[0,0]]; const seen=new Uint8Array(cw*ch);
  g[1]=0; seen[0]=1;
  while(stack.length){
    const [cx,cy]=stack[stack.length-1];
    const dirs=shuffle([[1,0],[-1,0],[0,1],[0,-1]]).filter(([dx,dy])=>{
      const nx=cx+dx, ny=cy+dy; return nx>=0&&ny>=0&&nx<cw&&ny<ch&&!seen[ny*cw+nx];
    });
    if(!dirs.length){stack.pop();continue;}
    const [dx,dy]=dirs[0];
    const nx=cx+dx, ny=cy+dy;
    g[(cy*2+1+dy)*w + (cx*2+1+dx)] = 0; // carve wall between
    g[(ny*2+1)*w + (nx*2+1)] = 0;
    seen[ny*cw+nx]=1; stack.push([nx,ny]);
  }
  return {grid:g,w,h};
}
```

```
function braid(g,w,h,p){
 for(let y=1;y<h-1;y++)for(let x=1;x<w-1;x++){
  if(g[y*w+x]!==1) continue;
  if((g[y*w+x-1]===0&&g[y*w+x+1]===0)||(g[(y-1)*w+x]===0&&g[(y+1)*w+x]===0))
    if(Math.random()<p) g[y*w+x]=0;
 }
}
```

```html
/* ================= maze generation ================= */
function genMaze(cw,ch){const w=cw*2+1,h=ch*2+1,g=new Uint8Array(w*h).fill(1),seen=new Uint8Array(cw*ch);
  const st=[[0,0]];seen[0]=1;g[w+1]=0;
  while(st.length){const[cx,cy]=st[st.length-1];const dirs=[[1,0],[-1,0],[0,1],[0,-1]];let i=3;
    while(i>0){const j=(Math.random()*(i+1))|0;const t=dirs[i];dirs[i]=dirs[j];dirs[j]=t;i--;}
    for(const[dx,dy]of dirs){const nx=cx+dx,ny=cy+dy;
      if(nx<0||ny<0||nx>=cw||ny>=ch||seen[ny*cw+nx])continue;
      g[(cy*2+1+dy)*w+(cx*2+1+dx)]=0;g[(ny*2+1)*w+(nx*2+1)]=0;seen[ny*cw+nx]=1;st.push([nx,ny]);break;}}
  return{g,w,h};}
function braid(g,w,h,p){for(let y=1;y<h-1;y++)for(let x=1;x<w-1;x++){if(!g[y*w+x])continue;
  const hz=g[y*w+x-1]===0&&g[y*w+x+1]===0,vt=g[(y-1)*w+x]===0&&g[(y+1)*w+x]===0;
  if((hz||vt)&&Math.random()<p)g[y*w+x]=0;}}
function bfs(g,w,h,sx,sy){const dist=new Int32Array(w*h).fill(-1);dist[sy*w+sx]=0;const q=[sy*w+sx];let head=0;
  while(head<q.length){const c=q[head++],cx=c%w,cy=(c/w)|0,d=dist[c]+1;
    if(cx>0&&!g[c-1]&&dist[c-1]<0){dist[c-1]=d;q.push(c-1);}
    if(cx<w-1&&!g[c+1]&&dist[c+1]<0){dist[c+1]=d;q.push(c+1);}
    if(cy>0&&!g[c-w]&&dist[c-w]<0){dist[c-w]=d;q.push(c-w);}
    if(cy<h-1&&!g[c+w]&&dist[c+w]<0){dist[c+w]=d;q.push(c+w);}}
  return dist;}
```
