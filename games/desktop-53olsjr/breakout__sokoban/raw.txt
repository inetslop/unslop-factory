```js
function circleRectHit(cx,cy,r,rx,ry,rw,rh){
  const nx=Math.max(rx,Math.min(cx,rx+rw));
  const ny=Math.max(ry,Math.min(cy,ry+rh));
  let dx=cx-nx, dy=cy-ny;
  const d2=dx*dx+dy*dy;
  if(d2>r*r) return null;
  let d=Math.sqrt(d2);
  if(d===0){ // center inside rect: push out along min axis
    const l=cx-rx, rr=rx+rw-cx, t=cy-ry, b=ry+rh-cy;
    const m=Math.min(l,rr,t,b);
    if(m===l) return {nx:-1,ny:0,depth:r+l};
    if(m===rr) return {nx:1,ny:0,depth:r+rr};
    if(m===t) return {nx:0,ny:-1,depth:r+t};
    return {nx:0,ny:1,depth:r+b};
  }
  return {nx:dx/d, ny:dy/d, depth:r-d};
}
```

    if(d.y>paddle.y-24&&d.y<paddle.y+26&&Math.abs(d.x-paddle.x)<paddle.w/2+18){dropCatch(d);drops.splice(i,1);}
    else if(d.y>LH+30)drops.splice(i,1);}

function draw(){

  ctx.setTransform(dpr,0,0,dpr,0,0);ctx.clearRect(0,0,W,H);
  const shx=(Math.random()*2-1)*shake,shy=(Math.random()*2-1)*shake;

  ctx.save();ctx.beginPath();ctx.rect(0,0,LW,LH);ctx.clip();
  bg();
  if(state==='menu'){menuScene();}
  else{for(const p of pads)drawPad(p);
    for(const w of wallsL)drawWall(w);
    for(const b of bricks)if(b.alive)drawBrick(b);
    for(const c of crates)drawCrate(c);
    drawDrops();drawPaddle();drawBalls();launchHint();}
  drawParts();vignette();flashDraw();
  if(state!=='menu')hud();
  bannerDraw();
  let hover=false;
  if(state==='menu')menuUI();
  else if(paused)pauseUI();
  else if(state==='clear')clearUI();
  else if(state==='over')overUI();
  ctx.restore();
  for(const h of uiHits)if(h.cb&&mouseL[0]>=h.x&&mouseL[0]<=h.x+h.w&&mouseL[1]>=h.y&&mouseL[1]<=h.y+h.h)hover=true;
