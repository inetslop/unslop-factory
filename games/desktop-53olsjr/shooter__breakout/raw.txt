/* ================= COMBAT ================= */
function fire(){
  var pulse=S.pulseT>0;
  ship.cd=pulse?.34:(S.rapidT>0?.12:.26);ship.muzzle=.06;
  var nx=ship.x,ny=ship.y-28,sp=pulse?480:900;
  function mk(a){bullets.push({x:nx,y:ny,vx:Math.sin(a)*sp,vy:-Math.cos(a)*sp,r:pulse?9:3.5,dmg:pulse?3:1,pulse:pulse});}
  if(S.spreadT>0){mk(-.26);mk(0);mk(.26);}else mk(0);
  sfx.shoot();
}
function hitCircle(px,py,pr,x,y,w,h){var cx=Math.max(x,Math.min(px,x+w)),cy=Math.max(y,Math.min(py,y+h)),dx=px-cx,dy=py-cy;return dx*dx+dy*dy<=pr*pr;}
function damageBrick(k,dmg,hx,hy,q){
  k.hp-=dmg;
  if(k.hp<=0){k.alive=false;onKill(k,q);}
  else{sfx.armor();burst(hx,hy,'#cfe3f2',4,160);S.shake=Math.max(S.shake,1.5);}
}
function onKill(k,q){
  var rc=brickRect(k),cx=rc.x+rc.w/2,cy=rc.y+rc.h/2;
  S.chain++;S.chainT=2;
  var mult=1+Math.min(9,(S.chain/3)|0);
  var pts=PTS[k.type]*mult;S.score+=pts;
  textPop(cx,cy,'+'+pts,k.type==='gem'?'#5fe8ff':'#ffd23f');
  burst(cx,cy,k.type==='armor'?'#a9c2d8':(k.type==='boom'?'#ff4f66':k.col),k.type==='armor'?16:11,210);
  sfx.pop(S.chain);S.shake=Math.max(S.shake,k.type==='armor'?4:2.5);
  if(k.type==='boom')q.push(k);
  var ch={block:.10,gem:.65,boom:.08,armor:.12}[k.type]||0;
  if(Math.random()<ch)spawnPower(cx,cy);
}
function boomAt(x,y,q){
  sfx.boom();S.shake=Math.max(S.shake,9);
  rings.push({x:x,y:y,r:6,vr:520,a:.7,c:'255,79,102'});
  burst(x,y,'#ff8a3d',18,260);
  for(var i=0;i<bricks.length;i++){var k=bricks[i];if(!k.alive)continue;
    var rc=brickRect(k),dx=rc.x+rc.w/2-x,dy=rc.y+rc.h/2-y;
    if(dx*dx+dy*dy<56*56){k.hp--;if(k.hp<=0){k.alive=false;onKill(k,q);}else burst(rc.x+rc.w/2,rc.y+rc.h/2,'#cfe3f2',4,160);}}
}
function collide(){
  var boomQ=[];
  for(var i=0;i<bullets.length;i++){var b=bullets[i];if(b.dead)continue;
    for(var j=0;j<bricks.length;j++){var k=bricks[j];if(!k.alive)continue;
      var rc=brickRect(k);
      if(hitCircle(b.x,b.y,b.r,rc.x,rc.y,rc.w,rc.h)){b.dead=true;damageBrick(k,b.dmg,b.x,b.y,boomQ);break;}}}
  bullets=bullets.filter(function(b){return !b.dead;});
  while(boomQ.length){var k2=boomQ.pop();if(!k2.alive)continue;var rc2=brickRect(k2);boomAt(rc2.x+rc2.w/2,rc2.y+rc2.h/2,boomQ);}
}
function loseLife(list){
  S.lives--;renderLives();S.graceT=2.2;S.flash=.5;S.flashCol='255,79,102';S.shake=16;sfx.life();
  for(var i=0;i<list.length;i++){var b=list[i];b.alive=false;var rc=brickRect(b);burst(rc.x+rc.w/2,rc.y+rc.h/2,b.type==='armor'?'#a9c2d8':b.col,10,200);}
  banner={text:'LINE BREACHED',sub:S.lives>0?S.lives+' SHIELD'+(S.lives>1?'S':'')+' REMAINING':'LAST SHIELD DOWN — HOLD THE LINE',t:0,life:1.6};
  if(S.lives<=0)gameOver();
}

function spawnPower(x,y){var r=Math.random(),type=r<.24?'spread':r<.48?'rapid':r<.68?'pulse':r<.86?'slow':'core';
  powerups.push({x:x,y:y,vy:150,t:Math.random()*6,type:type});}
function applyPower(p){
  var m=PWR[p.type];
  if(p.type==='spread')S.spreadT=8;else if(p.type==='rapid')S.rapidT=8;else if(p.type==='pulse')S.pulseT=7;
  else if(p.type==='slow')S.slowT=6;
  else{S.lives=Math.min(5,S.lives+1);renderLives();}
  sfx.power();textPop(p.x,ship.y-42,m.name,m.c);
  rings.push({x:ship.x,y:ship.y,r:8,vr:420,a:.6,c:m.rgb});
}
function burst(x,y,c,n,sp){for(var i=0;i<n;i++){var a=Math.random()*6.283,v=sp*(.3+Math.random()*.7);
  particles.push({kind:Math.random()<.5?1:0,x:x,y:y,vx:Math.cos(a)*v,vy:Math.sin(a)*v-40,life:0,max:.45+Math.random()*.5,size:2+Math.random()*3,c:c,rot:Math.random()*6.28,vr:(Math.random()-.5)*10});}
  if(particles.length>420)particles.splice(0,particles.length-420);}
function textPop(x,y,str,c){particles.push({kind:2,x:x,y:y,vx:0,vy:-46,life:0,max:.9,str:str,c:c,size:13});}
function updParticles(dt){
  for(var i=0;i<particles.length;i++){var p=particles[i];p.life+=dt;p.x+=p.vx*dt;p.y+=p.vy*dt;
    if(p.kind<2)p.vy+=520*dt;if(p.rot!==undefined&&p.kind===1)p.rot+=p.vr*dt;}
  particles=particles.filter(function(p){return p.life<p.max;});
  for(var j=0;j<rings.length;j++){var r=rings[j];r.r+=r.vr*dt;r.a-=dt*1.4;}
  rings=rings.filter(function(r){return r.a>0;});
}

/* ================= UPDATE ================= */
function update(dt){
  S.shake*=Math.exp(-6*dt);if(S.shake<.1)S.shake=0;
  S.flash=Math.max(0,S.flash-dt*2.2);
  if(banner){banner.t+=dt;if(banner.t>banner.life)banner=null;}

  if(S.state==='menu'){
    for(var i=0;i<attract.length;i++){var a=attract[i];a.y+=a.v*dt;if(a.y>H+30){a.y=-40;a.x=Math.random();}}
    updParticles(dt);return;
  }
  if(S.state==='over'){overT+=dt;updParticles(dt);return;}
  if(S.state==='paused')return;

  /* playing */
  S.spreadT=Math.max(0,S.spreadT-dt);S.rapidT=Math.max(0,S.rapidT-dt);
  S.pulseT=Math.max(0,S.pulseT-dt);S.slowT=Math.max(0,S.slowT-dt);S.graceT=Math.max(0,S.graceT-dt);
  if(S.chainT>0){S.chainT-=dt;if(S.chainT<=0)S.chain=0;}
  layoutWall();

  var dir=(keys.r?1:0)-(keys.l?1:0);
  if(dir){ship.x+=dir*640*dt;mouseTarget=null;}
  else if(mouseTarget!==null)ship.x+=(mouseTarget-ship.x)*Math.min(1,dt*15);
  ship.x=Math.max(34,Math.min(W-34,ship.x));
  ship.y=H-52;
  ship.cd-=dt;ship.muzzle=Math.max(0,ship.muzzle-dt);
  if(keys.fire&&ship.cd<=0)fire();

  for(var bi=0;bi<bullets.length;bi++){var b=bullets[bi];b.x+=b.vx*dt;b.y+=b.vy*dt;}
  bullets=bullets.filter(function(b){return b.y>-30&&b.x>-30&&b.x<W+30;});
  collide();

  if(S.waveDelay<=0&&bricks.some(function(b){return b.alive;}))
    S.wallOffset+=S.wallSpeed*(S.slowT>0?.45:1)*dt;

  if(S.graceT<=0&&S.waveDelay<=0){
    var bl=baseLine(),breach=null;
    for(var j=0;j<bricks.length;j++){var k=bricks[j];if(!k.alive)continue;
      var rc=brickRect(k);if(rc.y+rc.h>bl){if(!breach)breach=[];breach.push(k);}}
    if(breach)loseLife(breach);
  }

  for(var pi=0;pi<powerups.length;pi++){var p=powerups[pi];p.y+=p.vy*dt;p.t+=dt;p.x+=Math.sin(p.t*3)*20*dt;}
  powerups=powerups.filter(function(p){
    if(p.y>H-96&&Math.abs(p.x-ship.x)<46){applyPower(p);return false;}
    return p.y<H+30;});

  if(bricks.length&&bricks.every(function(b){return !b.alive;})){
    S.waveDelay=1.5;var bonus=300+S.wave*150;S.score+=bonus;
    banner={text:'WAVE '+S.wave+' CLEARED',sub:'BONUS +'+bonus,t:0,life:1.6};sfx.wave();
    rings.push({x:W/2,y:H*.4,r:10,vr:900,a:.8,c:'95,232,255'});
  }else if(S.waveDelay>0){
    S.waveDelay-=dt;
    if(S.waveDelay<=0){S.wave++;spawnWave(S.wave);layoutWall();
      banner={text:'WAVE '+S.wave,sub:waveHint(S.wave),t:0,life:1.8};}
  }

  updParticles(dt);

  el.score.textContent=fmt(S.score);el.wave.textContent=S.wave;el.best.textContent=fmt(Math.max(best,S.score));
  var mult=1+Math.min(9,(S.chain/3)|0);
  if(mult>1){el.comboWrap.classList.add('on');el.mult.textContent='x'+mult;}else el.comboWrap.classList.remove('on');
  el.cfill.style.width=(Math.max(0,S.chainT)/2*100)+'%';
  updFx();
}

  ctx.quadraticCurveTo(x,y+h,x,y+h-r);ctx.lineTo(x,y+r);ctx.quadraticCurveTo(x,y,x+r,y);ctx.closePath();}
