function drawStar(cx,cy,s){const X=cx*tile,Y=cy*tile,r=tile*0.2*s;
  ctx.save();ctx.translate(X,Y);ctx.fillStyle='#d8efff';ctx.shadowColor='#9fd8ff';ctx.shadowBlur=12;

function drawDoor(x,y,c,t){
  const wd=G.wd,col=KEY_COLORS[c].c,has=G.keysMask&(1<<c);
  const dir=wd.doorDir.get(y*wd.cols+x)||'v',X=x*tile,Y=y*tile,s=tile*0.16,bw=tile*0.15;
  ctx.save();ctx.fillStyle='#26394f';
  if(dir==='v'){
    ctx.fillRect(X+(tile-s)/2,Y+s*0.15,s,s);ctx.fillRect(X+(tile-s)/2,Y+tile-s*1.15,s,s);
    let alpha=has?0.35:0.75+0.2*Math.sin(t*3+c*2);
    ctx.globalAlpha=alpha;ctx.shadowColor=col;ctx.shadowBlur=has?4:12;ctx.fillStyle=col;
    if(has){const seg=tile*0.26;ctx.fillRect(X+(tile-bw)/2,Y+tile*0.18,bw,seg);ctx.fillRect(X+(tile-bw)/2,Y+tile*0.82-seg,bw,seg);}
    else ctx.fillRect(X+(tile-bw)/2,Y+tile*0.14,bw,tile*0.72);
  }else{
    ctx.fillRect(X+s*0.15,Y+(tile-s)/2,s,s);ctx.fillRect(X+tile-s*1.15,Y+(tile-s)/2,s,s);
    let alpha=has?0.35:0.75+0.2*Math.sin(t*3+c*2);
    ctx.globalAlpha=alpha;ctx.shadowColor=col;ctx.shadowBlur=has?4:12;ctx.fillStyle=col;
    if(has){const seg=tile*0.26;ctx.fillRect(X+tile*0.18,Y+(tile-bw)/2,seg,bw);ctx.fillRect(X+tile*0.82-seg,Y+(tile-bw)/2,seg,bw);}

function drawFog(t){
  const{cols,rows}=G.wd,p=G.player;
  fctx.globalCompositeOperation='source-over';fctx.globalAlpha=1;
  fctx.fillStyle='#050a12';fctx.fillRect(0,0,VW,VH);
  fctx.save();fctx.translate(ox+shakeX,oy+shakeY);
  const R=(3.3+G.lightBonus)*(1+0.02*Math.sin(t*6.3)+0.015*Math.sin(t*17));
  fctx.globalCompositeOperation='destination-out';
  for(let y=0;y<rows;y++)for(let x=0;x<cols;x++){
    const d=Math.hypot(x+0.5-p.fx,y+0.5-p.fy);let a=0;
    if(d<=R)a=1;else if(G.explored[y*cols+x])a=0.42;
    if(a){fctx.globalAlpha=a;fctx.fillRect(x*tile,y*tile,tile+0.5,tile+0.5);}}
  fctx.globalAlpha=1;
  const px=(p.fx+0.5)*tile,py=(p.fy+0.5)*tile,rr=R*1.2*tile;
  const gr=fctx.createRadialGradient(px,py,rr*0.2,px,py,rr);
  gr.addColorStop(0,'rgba(0,0,0,1)');gr.addColorStop(0.7,'rgba(0,0,0,0.85)');gr.addColorStop(1,'rgba(0,0,0,0)');
  fctx.fillStyle=gr;fctx.beginPath();fctx.arc(px,py,rr,0,7);fctx.fill();
  fctx.restore();fctx.globalAlpha=1;

function update(dt){
  for(const m of G.mist){m.x+=m.vx*dt;m.y+=m.vy*dt;
    if(m.x<-m.r)m.x=VW+m.r;if(m.x>VW+m.r)m.x=-m.r;if(m.y<-m.r)m.y=VH+m.r;if(m.y>VH+m.r)m.y=-m.r;}
  if(G.screen!=='play'&&G.screen!=='start')return;
  const p=G.player;
  if(p.moving){p.t+=dt/0.09;const e=Math.min(1,p.t),ee=e*e*(3-2*e);
    p.fx=lerp(p.fromX,p.toX,ee);p.fy=lerp(p.fromY,p.toY,ee);
    if(p.t>=1){p.moving=false;p.x=p.toX;p.y=p.toY;p.fx=p.x;p.fy=p.y;arrive();}}
  else{
    if(G.screen==='play'&&heldDirs.length)tryMove(heldDirs[heldDirs.length-1]);
    if(G.screen==='start'){G.aw+=dt;
      if(G.aw>0.5){G.aw=0;const opts=[];
        for(const d in DIRV){const nx=p.x+DIRV[d][0],ny=p.y+DIRV[d][1];
          if(nx>=0&&ny>=0&&nx<G.wd.cols&&ny<G.wd.rows&&!G.wd.g[ny*G.wd.cols+nx])opts.push(d);}
        if(opts.length)tryMove(opts[(Math.random()*opts.length)|0],true);}}}
  if(G.screen==='play'&&G.started)G.elapsed+=dt;
  if(Math.random()<dt*7)
    G.parts.push({x:p.fx+0.5+(Math.random()-0.5)*0.24,y:p.fy+0.3,
      vx:(Math.random()-0.5)*0.2,vy:-(0.5+Math.random()*0.6),life:0.8+Math.random()*0.7,max:1.5,col:'#ffb45e',type:'ember'});
  updateParts(dt);G.shake=Math.max(0,G.shake-dt*4);}
