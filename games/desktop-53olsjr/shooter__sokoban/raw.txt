function fire(){

  const L=Math.hypot(dx,dy)||1;dx/=L;dy/=L;

function damageEnemy(e){

  if(e.hp<=0){const i=enemies.indexOf(e);if(i>=0)enemies.splice(i,1);

    pop(e.x,e.y-.3,'+25','#ffb347');sparks(e.x,e.y,16,e.kind==='S'?'#ff9d5e':'#ff5d5d',5);ringFx(e.x,e.y,'#ff5d5d');updateHUD();
  } else sfx.hit();
}
function hurtPlayer(){
  if(state!=='play'||player.inv>0)return;
  player.hp--;player.inv=1.4;redFlash=1;trauma=Math.min(1,trauma+.4);
  sfx.hurt();pop(player.cx+.5,player.cy-.2,'-1','#ff5d5d');

function moveEntity(e,dx,dy){
  let moved=false;
  const nx=e.x+dx;if(canStand(nx,e.y,e.r)){e.x=nx;moved=true;}
  const ny=e.y+dy;if(canStand(e.x,ny,e.r)){e.y=ny;moved=true;}
  return moved;
}
function updEnemies(dt,mode){
  const pcx=player.cx+.5,pcy=player.cy+.5;
  for(const e of enemies){
    e.flash=Math.max(0,e.flash-dt);e.mouth=Math.max(0,e.mouth-dt*3);
    const dxp=pcx-e.x,dyp=pcy-e.y,d=Math.hypot(dxp,dyp)||.001;
    if(mode==='freeze'){e.spin+=dt*2;continue;}
    if(e.kind==='E'){
      let ang,sp;
      if(mode==='ai'&&d<4){ang=Math.atan2(dyp,dxp);sp=2.3;}
      else{e.wt-=dt;if(e.wt<=0){e.ang=rand(TAU);e.wt=rand(1,2.6);}ang=e.ang;sp=1.05;}
      e.face=ang;e.spin+=dt*(mode==='ai'&&d<4?7:3);
      if(!moveEntity(e,Math.cos(ang)*sp*dt,Math.sin(ang)*sp*dt))e.wt=0;
    }else{
      let mvx=0,mvy=0;
      if(d>4.4){mvx=dxp/d;mvy=dyp/d;}
      else if(d<2.6){mvx=-dxp/d;mvy=-dyp/d;}
      else{mvx=-dyp/d*e.orb*.8;mvy=dxp/d*e.orb*.8;}
      e.face=Math.atan2(dyp,dxp);e.spin+=dt*1.5;
      moveEntity(e,mvx*1.05*dt,mvy*1.05*dt);
      if(mode==='ai'){e.cd-=dt;
        if(e.cd<=0&&d<6.5&&losClear(e.x,e.y,pcx,pcy)){
          e.cd=rand(2.1,2.7);e.mouth=1;sfx.spit();
          const a=Math.atan2(dyp,dxp);
          ebullets.push({x:e.x+Math.cos(a)*.45,y:e.y+Math.sin(a)*.45,vx:Math.cos(a)*4,vy:Math.sin(a)*4,life:3});
        }}
    }
    for(const o of enemies){if(o===e)continue;const ox=e.x-o.x,oy=e.y-o.y,od=Math.hypot(ox,oy);
      if(od>0&&od<.7){e.x+=ox/od*.02;e.y+=oy/od*.02;}}
    if(mode==='ai'&&d<.62)hurtPlayer();
  }
}
function updBullets(dt){
  for(let i=bullets.length-1;i>=0;i--){const b=bullets[i];
    b.x+=b.vx*dt;b.y+=b.vy*dt;b.life-=dt;
    if(b.life<=0||blockedCell(Math.floor(b.x),Math.floor(b.y))){sparks(b.x,b.y,4,'#9fd8ff',2.5);bullets.splice(i,1);continue;}
    let hitE=false;
    for(const e of enemies){if(Math.hypot(e.x-b.x,e.y-b.y)<e.r+.16){damageEnemy(e);hits++;hitE=true;break;}}
    if(hitE){sparks(b.x,b.y,5,'#ffd9a0',3.5);bullets.splice(i,1);}
  }
}
function updEbullets(dt){
  for(let i=ebullets.length-1;i>=0;i--){const b=ebullets[i];
    b.x+=b.vx*dt;b.y+=b.vy*dt;b.life-=dt;
    if(b.life<=0||blockedCell(Math.floor(b.x),Math.floor(b.y))){sparks(b.x,b.y,4,'#ff8a5e',2);ebullets.splice(i,1);continue;}
    if(state==='play'&&Math.hypot(player.cx+.5-b.x,player.cy+.5-b.y)<.45){hurtPlayer();ebullets.splice(i,1);}
  }
}

function updDust(dt){for(const d of dust){d.x-=d.spd*dt;d.y+=Math.sin(time*.6+d.ph)*dt*.05;
  if(d.x<-4){d.x=W+4;d.y=rand(H);}if(d.y>H+4)d.y=-4;}}
function update(dt){
  time+=dt;
  trauma=Math.max(0,trauma-dt*1.7);
  redFlash=Math.max(0,redFlash-dt*2.2);
  gunKick=Math.max(0,gunKick-dt*6);crossPulse=Math.max(0,crossPulse-dt*5);
  updDust(dt);updParts(dt);updPops(dt);
  const Lp=Math.min(1,dt*26),Lc=Math.min(1,dt*20);
  player.vx+=(player.cx+.5-player.vx)*Lp;player.vy+=(player.cy+.5-player.vy)*Lp;
  for(const c of crates){c.vx+=(c.x-c.vx)*Lc;c.vy+=(c.y-c.vy)*Lc;c.squash=Math.max(0,c.squash-dt*4);}
  if(state==='play'){
    totalTime+=dt;
    player.inv-=dt;player.fireCd-=dt;player.moveCd-=dt;player.bumpT-=dt;
    const d=DIRS[moveStack[moveStack.length-1]];
    if(player.moveCd<=0&&d)tryMove(d[0],d[1]);
    if(player.moveCd>0)player.moveCd-=dt;
    const wx=(mouse.x-OX)/CELL,wy=(mouse.y-OY)/CELL;
    player.gunAng=Math.atan2(wy-(player.cy+.5),wx-(player.cx+.5));
    if(mouse.down||keys[' '])fire();
    updEnemies(dt,'ai');
  }else if(state==='clear'){
    clearT+=dt;
    if(clearT>1.9){if(levelIdx<LEVELS.length-1){loadLevel(levelIdx+1);state='play';}else winGame();}
  }else if(state==='start'||state==='over'){
    updEnemies(dt,'wander');
  }
  if(state!=='pause'){updBullets(dt);updEbullets(dt);}
  if(state==='over'&&!ovShown){overT+=dt;if(overT>.9){ovShown=true;$('overScore').textContent=String(score).padStart(6,'0');showOv('over');}}
}

    ctx.lineTo(x+Math.cos(a)*(r+8),y+Math.sin(a)*(r+8));ctx.stroke();}
  ctx.fillStyle='#ffb347';ctx.beginPath();ctx.arc(x,y,2,0,TAU);ctx.fill();
}
function drawRedFlash(){
  let a=redFlash*.45;
  if(state==='play'&&player.hp===1)a=Math.max(a,.1+.06*Math.sin(time*4));
  if(a<=0)return;
  const g=ctx.createRadialGradient(W/2,H/2,Math.min(W,H)*.3,W/2,H/2,Math.max(W,H)*.7);
  g.addColorStop(0,'rgba(255,50,50,0)');g.addColorStop(1,`rgba(255,50,50,${a})`);
  ctx.fillStyle=g;ctx.fillRect(0,0,W,H);
}

  drawBG();
  const sh=trauma*trauma*14;
  const sx=(Math.random()*2-1)*sh,sy=(Math.random()*2-1)*sh;
  ctx.save();ctx.translate(sx,sy);
  drawBoard();drawTargets();drawCrates();drawEnemies();drawBullets();drawPlayer();drawParts();drawPops();
  ctx.restore();

}
