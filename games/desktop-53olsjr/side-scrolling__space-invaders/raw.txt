```js
function buildWave(n){
  const q=[]; let t=0;
  if(n%5===0){ // boss wave
    q.push({t:t+=0.5, type:'boss'});
    const escorts=2+Math.min(4,Math.floor(n/5));
    for(let i=0;i<escorts*3;i++){ t+=rand(1.2,2.2); q.push({t, type: pick(['drone','drone','weevil'])}); }
    return q;
  }
  const count = Math.min(46, 10 + n*3);
  let ti=0;
  while(count>0){
    t += (ti===0? 0.8 : rand(0.55, 1.1) * Math.max(0.55, 1 - n*0.03));
    // choose type by wave weights
    let type='drone';
    const r=Math.random();
    if(n>=4 && r<0.14+ n*0.008) type='brute';
    else if(n>=3 && r<0.34) type='splitter';
    else if(n>=2 && r<0.62) type='weevil';
    q.push({t, type}); count--; ti++;
  }
  return q;
}
```

```html
/* ============================== WAVES / SPAWNING ============================== */
function buildWave(n){
  const q=[];let tt=0;
  if(n%5===0){
    tt=1.6;q.push({t:tt,type:'boss'});
    const esc=Math.min(12,4+Math.floor(n/5)*2);
    for(let i=0;i<esc;i++){tt+=rand(1.4,2.6);q.push({t:tt,type:pick(['drone','drone','weevil','mini'])});}
    return q;
  }
  const count=Math.min(44,9+n*3);let first=true;
  for(let i=0;i<count;i++){
    tt+=first?rand(1.1,1.7):Math.max(.45,rand(.5,1.05)-n*.02);first=false;
    const r=Math.random();let type='drone';
    if(n>=4&&r<Math.min(.3,.1+n*.01))type='brute';
    else if(n>=3&&r<.36)type='splitter';
    else if(n>=2&&r<.66)type='weevil';
    q.push({t:tt,type});
  }
  return q;
}
function startWave(n){
  wave.n=n;wave.q=buildWave(n);wave.qt=0;wave.active=true;wave.clearT=0;
  if(n%5===0){banner('DREADNOUGHT INCOMING','red');SFX.alarm();}
  else{banner('WAVE '+n);SFX.wave();}
}
function spawnEnemy(type){
  const d=ENDEF[type];
  const y=clamp(H*(0.18+irand(0,2)*0.3)+rand(-46,46),64,H-64);
  enemies.push({type,x:W+50+30,y,baseY:y,hp:Math.max(1,Math.round(d.hp*(1+(wave.n-1)*.06))),
    spd:d.spd*(1+(wave.n-1)*.05),amp:d.amp,frq:d.frq*rand(.85,1.2),ph:rand(0,TAU),
    cd:rand(d.fm,d.fx),flash:0,r:d.r,scale:type==='mini'?.62:1});
}

    amp:rand(26,44),frq:rand(2.6,3.4),ph:rand(0,TAU),cd:rand(2.5,4),flash:0,r:9,scale:.62};

/* ============================== FX / SCORE ============================== */
function addShake(m){shake=Math.min(26,shake+m);}
function sparks(x,y,c,n){for(let i=0;i<n;i++){const a=rand(0,TAU),sp=rand(40,180);
  parts.push({kind:'s',x,y,vx:Math.cos(a)*sp,vy:Math.sin(a)*sp,life:rand(.15,.4),max:.4,c});}}
function boom(x,y,c,sc){
  addShake(5*sc);
  parts.push({kind:'f',x,y,r:26*sc,life:.18,max:.18,c});
  parts.push({kind:'r',x,y,r:6,vr:260*sc,life:.4,max:.4,c});
  const n=10+Math.floor(8*sc);
  for(let i=0;i<n;i++){const a=rand(0,TAU),sp=rand(60,340)*sc;
    parts.push({kind:'s',x,y,vx:Math.cos(a)*sp,vy:Math.sin(a)*sp,life:rand(.25,.6),max:.6,c:pick([c,'#ffffff','#ffd9a0'])});}
  for(let i=0;i<4;i++)parts.push({kind:'p',x:x+rand(-8,8),y:y+rand(-8,8),vx:rand(-30,30),vy:rand(-30,10),r:rand(6,12)*sc,life:rand(.5,.9),max:.9});
  if(parts.length>480)parts.splice(0,parts.length-480);
  SFX.boom(Math.min(2,sc));
}

function addScore(base,x,y){
  combo++;comboT=2.5;mult=Math.min(8,1+Math.floor(combo/2));

/* ============================== COMBAT ============================== */
function firePlayer(){
  const rapid=t<rapidU;player.cd=rapid?.12:.24;
  const a=[0];if(t<spreadU)a.push(-.13,.13);
  for(const ang of a)bullets.push({x:player.x+26,y:player.y,vx:Math.cos(ang)*820,vy:Math.sin(ang)*820,r:4});
  parts.push({kind:'f',x:player.x+30,y:player.y,r:10,life:.1,max:.1,c:'#ffd9a0'});
  sparks(player.x+30,player.y,'#bffaff',3);
  SFX.shoot();addShake(.7);
}
function enemyFire(e,d){
  const ang=Math.atan2(player.y-e.y,player.x-e.x)+rand(-.07,.07);
  const sp=clamp(280+wave.n*8,280,430);
  if(e.type==='brute'){for(const o of[-.24,0,.24])ebullets.push({x:e.x-16,y:e.y,vx:Math.cos(ang+o)*sp*.9,vy:Math.sin(ang+o)*sp*.9,r:5,c:d.c});}
  else ebullets.push({x:e.x-14,y:e.y,vx:Math.cos(ang)*sp,vy:Math.sin(ang)*sp,r:3.5,c:d.c});
  SFX.eshoot();
}
function killEnemy(idx){
  const e=enemies[idx];enemies.splice(idx,1);
  const d=ENDEF[e.type];
  boom(e.x,e.y,d.c,d.sc);
  addScore(d.pts,e.x,e.y-14);
  if(e.type==='splitter'){enemies.push(makeMini(e.x+8,e.y-16));enemies.push(makeMini(e.x+8,e.y+16));}
  maybeDrop(e);
}
function maybeDrop(e){
  const ch={drone:.05,weevil:.1,splitter:.14,mini:.02,brute:.35}[e.type];
  if(Math.random()>ch)return;
  let type=Math.random()<.3?'shield':Math.random()<.6?'rapid':'spread';
  if(player.hull<3&&Math.random()<.18)type='life';
  pickups.push({x:e.x,y:e.y,type,t:0});
}
function applyPickup(p){
  const names={shield:'SHIELD',rapid:'RAPID FIRE',spread:'TRI-SHOT',life:'+1 HULL'};
  pop(player.x,player.y-32,names[p.type],PD[p.type].c,.9);
  SFX.pickup();addShake(2);
  if(p.type==='shield')shieldU=t+6;else if(p.type==='rapid')rapidU=t+8;
  else if(p.type==='spread')spreadU=t+10;else player.hull=Math.min(5,player.hull+1);
}
function die(){
  player.dead=true;boom(player.x,player.y,'#59f2ff',1.4);SFX.over();
  events.push({t:.35,fn:()=>boom(player.x+rand(-24,24),player.y+rand(-24,24),'#ffffff',1.6)});
  events.push({t:1.35,fn:gameOver});
}
function damagePlayer(blockable,hx,hy){
  if(player.inv>0||player.dead)return;
  if(blockable&&t<shieldU){pop(hx,hy,'BLOCKED','#59f2ff',.8);SFX.hit();addShake(4);return;}
  player.hull--;player.inv=1.8;combo=0;mult=1;
  boom(player.x,player.y,'#59f2ff',1.1);flashGo();addShake(13);SFX.playerHit();
  if(player.hull<=0)die();
}
function breach(y){
  if(player.dead||player.inv>0)return;
  player.hull--;player.inv=1.8;combo=0;mult=1;
  flashGo();addShake(9);SFX.playerHit();pop(46,y,'BREACH!','#ff5d7e',1.1);
  if(player.hull<=0)die();
}

/* ============================== UPDATE ============================== */
function updateWorld(dt,rdt){
  for(const s of stars){s.x-=s.s*dt;if(s.x<-4){s.x=W+4;s.y=rand(0,H);}}
  for(const n of nebs){n.x-=n.s*dt;if(n.x<-n.r)n.x=W+n.r;}
  if(planet){planet.x-=6*dt;if(planet.x<-planet.r*2.2)planet.x=W+planet.r*2.2;}
  shootT-=dt;
  if(shootT<=0){shootT=rand(5,13);shooters.push({x:rand(W*.4,W),y:rand(0,H*.5),vx:-rand(700,1000),vy:rand(60,160),life:.8});}
  for(let i=shooters.length-1;i>=0;i--){const s=shooters[i];s.x+=s.vx*dt;s.y+=s.vy*dt;s.life-=dt;if(s.life<=0)shooters.splice(i,1);}

  for(let i=events.length-1;i>=0;i--){const ev=events[i];ev.t-=rdt;if(ev.t<=0){events.splice(i,1);ev.fn();}}

  if(state==='menu'){
    ambT-=rdt;
    if(ambT<=0){ambT=rand(1.1,2.4);const ty=pick(['drone','weevil','splitter','brute']);
      amb.push({spr:SPR[ty],x:W+60,y:rand(70,H-70),spd:rand(28,55),amp:rand(8,30),ph:rand(0,TAU),t:0});}
    for(let i=amb.length-1;i>=0;i--){const a=amb[i];a.x-=a.spd*rdt;a.t+=rdt;if(a.x<-70)amb.splice(i,1);}
  }

  shake=Math.max(0,shake*(1-6*rdt)-4*rdt);
  zl+=((focusOn?1.05:1)-zl)*Math.min(1,rdt*6);

  if(state!=='playing')return;

  /* focus */
  const want=(keys.ShiftLeft||keys.ShiftRight||keys.KeyK)&&player.fE>0&&!focusStall&&!player.dead;
  if(want!==focusOn){focusOn=want;SFX.focus(focusOn);}
  tsTarget=player.dead?.25:(focusOn?.35:1);
  if(boss&&boss.dying&&!player.dead)tsTarget=Math.min(tsTarget,.45);
  ts+=(tsTarget-ts)*Math.min(1,rdt*7);
  if(focusOn){player.fE-=34*dt;if(player.fE<=0){player.fE=0;focusStall=true;}}
  else{player.fE=Math.min(100,player.fE+13*rdt);if(player.fE>30)focusStall=false;}

  /* player */
  if(!player.dead){
    const tv=((keys.ArrowUp||keys.KeyW)?-1:0)+((keys.ArrowDown||keys.KeyS)?1:0);
    if(player.touchY!=null)player.y+=(player.touchY-player.y)*Math.min(1,rdt*10);
    else{player.vy+=((tv*430)-player.vy)*Math.min(1,dt*14);player.y=clamp(player.y+player.vy*dt,56,H-52);}
    player.cd-=dt;player.inv=Math.max(0,player.inv-dt);
    if((keys.Space||keys.KeyJ||touchFire)&&player.cd<=0)firePlayer();
  }

  /* wave flow */
  if(wave.active&&!player.dead){
    if(wave.q.length){wave.qt+=dt;
      while(wave.q.length&&wave.q[0].t<=wave.qt){const it=wave.q.shift();it.type==='boss'?spawnBoss():spawnEnemy(it.type);}}
    else if(!enemies.length&&(!boss||boss.dying)){
      if(wave.clearT===0){const b=100+wave.n*25;score+=b;pop(W/2,H*.4,'SECTOR CLEAR  +'+b,'#ffb84d',1.4);SFX.wave();wave.clearT=2.4;}
      wave.clearT-=dt;if(wave.clearT<=0)startWave(wave.n+1);
    }
  }

  /* enemies */
  warn=false;
  for(let i=enemies.length-1;i>=0;i--){
    const e=enemies[i],d=ENDEF[e.type];
    e.x-=e.spd*dt;e.y=e.baseY+Math.sin(t*e.frq+e.ph)*e.amp;if(e.flash>0)e.flash-=dt;
    if(e.cd>0)e.cd-=dt;
    if(e.cd<=0&&e.x<W-40&&e.x>player.x+80&&!player.dead){enemyFire(e,d);e.cd=rand(d.fm,d.fx)*clamp(1-wave.n*.025,.55,1);}
    if(!player.dead&&player.inv<=0){
      const dx=e.x-player.x,dy=e.y-player.y,rr=e.r*e.scale+16;
      if(dx*dx+dy*dy<rr*rr){killEnemy(i);damagePlayer(true,e.x,e.y);continue;}
    }
    if(e.x<-50){enemies.splice(i,1);breach(e.y);continue;}
    if(e.x<player.x+260)warn=true;
  }

  /* boss */
  if(boss&&!boss.dying){
    const b=boss;if(b.flash>0)b.flash-=dt;
    if(b.x>b.tgtX)b.x-=95*dt;
    else{
      b.y=clamp(H*.5+Math.sin(t*.5)*60,120,H-120)+Math.sin(t*1.3)*8;
      if(!player.dead){b.patT-=dt;if(b.patT<=0){b.patT=rand(1.8,2.6);bossFire();}}
    }
  }

  /* player bullets */
  for(let i=bullets.length-1;i>=0;i--){
    const b=bullets[i];b.x+=b.vx*dt;b.y+=b.vy*dt;
    if(b.x>W+50||b.x<-50||b.y<-60||b.y>H+60){bullets.splice(i,1);continue;}
    let used=false;
    for(let j=enemies.length-1;j>=0;j--){const e=enemies[j];
      const dx=b.x-e.x,dy=b.y-e.y,rr=e.r*e.scale+b.r;
      if(dx*dx+dy*dy<rr*rr){used=true;e.hp--;e.flash=.09;SFX.hit();sparks(b.x,b.y,'#bffaff',4);if(e.hp<=0)killEnemy(j);break;}}
    if(!used&&boss&&!boss.dying){
      const dx=b.x-(boss.x-10),dy=b.y-boss.y;
      if(dx*dx+dy*dy<72*72){used=true;boss.hp--;boss.flash=.07;SFX.hit();sparks(b.x,b.y,'#ff8ba3',5);score+=25;
        if(boss.hp<=0)bossDeath();}
    }
    if(used)bullets.splice(i,1);
  }

  /* enemy bullets */
  for(let i=ebullets.length-1;i>=0;i--){
    const b=ebullets[i];b.x+=b.vx*dt;b.y+=b.vy*dt;
    if(b.x<-60||b.x>W+60||b.y<-70||b.y>H+70){ebullets.splice(i,1);continue;}
    if(!player.dead&&player.inv<=0){
      const dx=b.x-player.x,dy=b.y-player.y;
      if(dx*dx+dy*dy<(b.r+13)*(b.r+13)){ebullets.splice(i,1);damagePlayer(true,b.x,b.y);}
    }
  }

  /* pickups */
  for(let i=pickups.length-1;i>=0;i--){
    const p=pickups[i];p.t+=dt;p.x-=115*dt;
    const dx=player.x-p.x,dy=player.y-p.y,d2=dx*dx+dy*dy;
    if(!player.dead&&d2<120*120){p.x+=dx*Math.min(1,rdt*6);p.y+=dy*Math.min(1,rdt*6);}
    if(p.x<-40){pickups.splice(i,1);continue;}
    if(!player.dead&&d2<32*32){pickups.splice(i,1);applyPickup(p);}
  }

  /* combo / alarm */
  comboT-=dt;if(comboT<=0&&combo>0){combo=0;mult=1;}
  if(warn){alarmT-=rdt;if(alarmT<=0){SFX.alarm();alarmT=.55;}}

  /* particles & popups */
  for(let i=parts.length-1;i>=0;i--){const p=parts[i];p.life-=dt;
    if(p.life<=0){parts.splice(i,1);continue;}
    if(p.kind==='s'){p.x+=p.vx*dt;p.y+=p.vy*dt;p.vx*=(1-2.5*dt);p.vy*=(1-2.5*dt);}
    else if(p.kind==='p'){p.x+=p.vx*dt;p.y+=p.vy*dt;p.r+=30*dt;}
    else if(p.kind==='r')p.r+=p.vr*dt;}
  for(let i=pops.length-1;i>=0;i--){const p=pops[i];p.t+=dt;if(p.t>=p.max)pops.splice(i,1);}
}
```
