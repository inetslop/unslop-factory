function genWave(n){ const q=[]; const isBoss=n%5===0; let count=Math.min(28,6+n*2);
  if(isBoss){ q.push({t:'boss',gap:1.7}); count=Math.min(14,3+Math.floor(n*.8)); }
  const wG=1, wR=n>=2?Math.min(.5,.12*n):0, wM=n>=3?Math.min(.45,.06*n+.1):0, wB=n>=4?Math.min(.35,.03*n):0;
  for(let i=0;i<count;i++){ let r=Math.random()*(wG+wR+wM+wB), t='grit';
    if((r-=wG)<0)t='grit'; else if((r-=wR)<0)t='runt'; else if((r-=wM)<0)t='magma'; else t='brute';
    q.push({t,gap:Math.max(.42,1.05-n*.03)*rand(.8,1.25)}); }
  return q; }
function startWave(){ S.wave++; S.hpMul=1+(S.wave-1)*.16; S.spdMul=Math.min(1.4,1+(S.wave-1)*.015);
  S.spawnQ=genWave(S.wave); S.spawnT=.6; S.mode='wave';
  banner('WAVE '+S.wave, S.wave%5===0?'THE COLOSSUS ROLLS IN':(S.wave===1?'PREPARE YOUR TURRETS':'HOLD THE LINE'));
  sfx.horn(); }

function makeFoe(type){ const d=FOES[type]; return { type,r:d.r,hp:d.hp*S.hpMul,max:d.hp*S.hpMul,s:0,x:-70,y:300-d.r,
  rot:rand(TAU),vx:0,vy:0,state:'walk',slowT:0,slowF:1,glowT:0,seed:Math.random()*999,flingT:0,groundT:0,dead:false }; }
function spawnFoe(type){ const f=makeFoe(type); S.enemies.push(f); if(type==='boss'){ addText(800,140,'⚠ COLOSSUS INBOUND','#ff6a52',20); sfx.alarm(); shake(6); } }
function collideBox(f,b){ const cx=clamp(f.x,b.x,b.x+b.w), cy=clamp(f.y,b.y,b.y+b.h);
  let dx=f.x-cx,dy=f.y-cy,d2=dx*dx+dy*dy; if(d2>=f.r*f.r) return null;
  let d=Math.sqrt(d2),nx,ny;
  if(d<1e-4){ const l=f.x-b.x,rg=b.x+b.w-f.x,tp=f.y-b.y,bt=b.y+b.h-f.y,m=Math.min(l,rg,tp,bt);
    if(m===tp){nx=0;ny=-1;} else if(m===bt){nx=0;ny=1;} else if(m===l){nx=-1;ny=0;} else {nx=1;ny=0;} d=0; }
  else { nx=dx/d; ny=dy/d; }
  const pen=f.r-d; f.x+=nx*pen; f.y+=ny*pen;
  const vn=f.vx*nx+f.vy*ny; if(vn<0){ f.vx-=1.34*vn*nx; f.vy-=1.34*vn*ny; }
  return {nx,ny}; }
function updateFoe(f,dt){ if(f.slowT>0){ f.slowT-=dt; if(f.slowT<=0)f.slowF=1; }
  if(f.state==='walk'){ const d=FOES[f.type]; const v=d.spd*S.spdMul*f.slowF; f.s+=v*dt; f.rot+=(v/f.r)*dt;
    const p=pathPoint(f.s); f.x=p.x; f.y=p.y-f.r;
    if(f.s>=TOTAL-4) reachBase(f); return; }
  f.flingT+=dt; f.vy+=2200*dt; f.x+=f.vx*dt; f.y+=f.vy*dt;
  let grounded=false;
  for(const b of TERRAIN){ const c=collideBox(f,b); if(c&&c.ny<-.4) grounded=true; }
  if(grounded) f.vx*=Math.max(0,1-2.6*dt);
  f.rot+=(f.vx/f.r)*dt;
  if(f.x<-160){f.x=-160;f.vx=Math.abs(f.vx)*.4;} if(f.x>1730){f.x=1730;f.vx=-Math.abs(f.vx)*.4;}
  const sp=Math.hypot(f.vx,f.vy);
  if(sp<36&&grounded)f.groundT+=dt; else f.groundT=0;
  if(f.groundT>.22||f.flingT>7){ f.s=clamp(nearestS(f.x,f.y),0,TOTAL-4); f.state='walk'; f.vx=f.vy=0; } }
function foePairs(){ const E=S.enemies;
  for(let i=0;i<E.length;i++)for(let j=i+1;j<E.length;j++){ const a=E[i],b=E[j]; if(a.dead||b.dead)continue;
    const dx=b.x-a.x,dy=b.y-a.y; let d=Math.hypot(dx,dy); const min=a.r+b.r; if(d>=min)continue;
    if(d<1e-4)d=.01; const nx=dx/d,ny=dy/d,ov=min-d;
    if(a.state==='walk'&&b.state==='walk'){ if(a.s<=b.s)a.s=Math.max(0,a.s-ov*.5); else b.s=Math.max(0,b.s-ov*.5); continue; }
    const ma=a.r*a.r,mb=b.r*b.r,mt=ma+mb;
    a.x-=nx*ov*(mb/mt); a.y-=ny*ov*(mb/mt); b.x+=nx*ov*(ma/mt); b.y+=ny*ov*(ma/mt);
    const vn=(a.vx-b.vx)*nx+(a.vy-b.vy)*ny;
    if(vn>0){ const k=1.25*vn;
      if(a.state!=='flung'){a.state='flung';a.flingT=0;a.groundT=0;} a.vx-=k*(mb/mt)*nx; a.vy-=k*(mb/mt)*ny;
      if(b.state!=='flung'){b.state='flung';b.flingT=0;b.groundT=0;} b.vx+=k*(ma/mt)*nx; b.vy+=k*(ma/mt)*ny; } } }
function damageFoe(f,amt,kx,ky){ if(f.dead)return; f.hp-=amt; f.glowT=.12;
  const imp=Math.hypot(kx,ky)*f.knock;
  if(imp>40){ f.state='flung'; f.flingT=0; f.groundT=0;
    f.vx=(kx||0)*f.knock+rand(-40,40); f.vy=(ky||0)*f.knock-(imp>150?120:60); }
  spark(f.x,f.y,3,'#ffd27a'); sfx.hit();
  if(f.hp<=0) killFoe(f); }
function killFoe(f){ f.dead=true; S.kills++; const d=FOES[f.type]; S.coins+=d.bounty; S.earned+=d.bounty;
  addText(f.x,f.y-f.r-6,'+'+d.bounty,'#ffd24a',15);
  shards(f.x,f.y,7+f.r/3,d.c1,d.c2,f.type==='boss'); dust(f.x,f.y); ring(f.x,f.y,'#ffb066',f.r*2.2);
  if(f.type==='boss'){ shake(11); sfx.boom(90); addText(f.x,f.y-f.r-30,'COLOSSUS DOWN','#ffd24a',20); }
  sfx.coin(); }
function reachBase(f){ f.dead=true; S.lives=Math.max(0,S.lives-f.dmg); shake(9); sfx.alarm();
  ring(1548,258,'#ff6a52',70); spark(1548,258,10,'#ff8a5c'); addText(1548,214,'-'+f.dmg,'#ff6a52',18);
  if(S.lives<=0) gameOver(); }

function addTower(type,x,gy){ const d=TDEFS[type]; const t={type,x,gy,level:0,cool:rand(.3),ang:-Math.PI/2,recoil:0,flash:0,invested:d.cost[0]}; S.towers.push(t); return t; }
function predict(t,f,d){ if(f.state==='flung') return {x:f.x+f.vx*.18,y:f.y+f.vy*.18};
  const dir=pathDir(f.s), v=FOES[f.type].spd*S.spdMul*f.slowF, tt=clamp(Math.hypot(f.x-t.x,f.y-(t.gy-40))/d.spd,0,.45);
  return {x:f.x+dir.x*v*tt, y:f.y+dir.y*v*tt}; }
function updateTower(t,dt){ const d=TDEFS[t.type], l=t.level;
  t.cool-=dt; t.recoil=Math.max(0,t.recoil-42*dt); t.flash=Math.max(0,t.flash-dt*7);
  const cx=t.x,cy=t.gy-40,rng=d.range[l]; let best=null,bs=-1;
  for(const f of S.enemies){ if(f.dead)continue;
    if(dist2(cx,cy,f.x,f.y)<(rng+f.r)*(rng+f.r)&&f.s>bs){ bs=f.s; best=f; } }
  if(best){ const p=(t.type==='cannon')?{x:best.x,y:best.y-6}:predict(t,best,d);
    let da=Math.atan2(p.y-cy,p.x-cx)-t.ang; while(da>Math.PI)da-=TAU; while(da<-Math.PI)da+=TAU;
    t.ang+=da*Math.min(1,dt*14);
    if(t.cool<=0){ fireTower(t,best,d,cx,cy); t.cool=1/d.rof[l]; }
  } else { t.ang+=(-Math.PI/2+Math.sin(clock*.6+t.x*.05)*.55-t.ang)*Math.min(1,dt*3); } }
function fireTower(t,f,d,cx,cy){ const a=t.ang;
  const len=t.type==='cannon'?22:(t.type==='cryo'?11:26);
  const bx=cx+Math.cos(a)*len, by=cy+Math.sin(a)*len;
  if(t.type==='dart'){ S.shots.push({k:'dart',x:bx,y:by,vx:Math.cos(a)*d.spd,vy:Math.sin(a)*d.spd,dmg:d.dmg[t.level],life:1.5}); t.recoil=5; t.flash=.12; sfx.dart(); }
  else if(t.type==='cryo'){ S.shots.push({k:'bolt',x:bx,y:by,vx:Math.cos(a)*d.spd,vy:Math.sin(a)*d.spd,dmg:d.dmg[t.level],sf:d.slow[t.level],dur:d.dur[t.level],life:1.6}); t.recoil=3; t.flash=.12; sfx.cryo(); }
  else { const dx=f.x-cx, dy=(f.y-6)-cy, dist=Math.hypot(dx,dy)||1, Tf=clamp(.5+dist/900,.65,1.35);
    S.shots.push({k:'shell',x:bx,y:by,vx:dx/Tf,vy:(dy-.5*1400*Tf*Tf)/Tf,dmg:d.dmg[t.level],splash:d.splash[t.level],knock:d.knock[t.level],life:3});
    t.recoil=9; t.flash=.16; sfx.cannon(); } }
function explode(x,y,sh){ const spl=sh.splash,kn=sh.knock;
  S.parts.push({type:'ring',x,y,vx:0,vy:0,life:.45,max:.45,size:spl,col:'#ffca7a'});
  shards(x,y,10,'#6b7683','#3d4650'); dust(x,y);
  S.scorch.push({x,y,r:spl*.42,life:2.6,max:2.6});
  shake(5+spl/45); sfx.boom(spl);
  for(const f of S.enemies){ if(f.dead)continue; const dx=f.x-x,dy=f.y-y,d=Math.hypot(dx,dy);
    if(d<spl+f.r){ let ux=dx/(d||1),uy=dy/(d||1); if(d<.01){ux=0;uy=-1;}
      const fall=1-.45*Math.min(1,d/(spl+f.r)), imp=kn*(1-Math.min(1,d/(spl+f.r)));
      damageFoe(f,sh.dmg*fall,ux*imp,uy*imp); } } }
function updateShots(dt){ for(const s of S.shots){ if(s.life<=0)continue; s.life-=dt;
    if(s.k==='shell'){ s.vy+=1400*dt; if(Math.random()<.6) S.parts.push({type:'smoke',x:s.x,y:s.y,vx:rand(-10,10),vy:rand(-20,-4),life:.4,max:.4,size:3,col:'#9aa5b1'}); }
    s.x+=s.vx*dt; s.y+=s.vy*dt;
    if(s.x<-100||s.x>1760||s.y>980){ s.life=0; continue; }
    let hitT=false; for(const b of TERRAIN) if(s.x>=b.x&&s.x<=b.x+b.w&&s.y>=b.y&&s.y<=b.y+b.h){ hitT=true; break; }
    if(hitT){ if(s.k==='shell') explode(s.x,s.y,s); else dust(s.x,s.y,s.k==='bolt'); s.life=0; continue; }
    for(const f of S.enemies){ if(f.dead)continue; const rr=f.r+(s.k==='shell'?9:5);
      if(dist2(s.x,s.y,f.x,f.y)<rr*rr){
        if(s.k==='shell') explode(s.x,s.y,s);
        else { const m=Math.hypot(s.vx,s.vy)||1, ux=s.vx/m, uy=s.vy/m, imp=(s.k==='dart'?140:70);
          damageFoe(f,s.dmg,ux*imp,uy*imp);
          if(s.k==='bolt'){ f.slowT=Math.max(f.slowT,s.dur); f.slowF=Math.min(f.slowF,s.sf); spark(f.x,f.y,4,'#9fe3f2'); } }
        s.life=0; break; } } }
  S.shots=S.shots.filter(s=>s.life>0); }

/* ================= placement ================= */
function spotInfo(mx,my){ const gy=groundAt(mx); if(gy===null||mx<60||mx>1470) return {ok:false,x:clamp(mx,60,1470),gy:300};
  let ok=Math.abs(my-gy)<=95;
  if(ok){ for(const p of SAMPLES){ if(dist2(p.x,p.y-20,clamp(mx,60,1470),gy-28)<1600){ ok=false; break; } } }
  if(ok){ const x=clamp(mx,60,1470); for(const t of S.towers){ if(Math.abs(t.x-x)<54&&Math.abs(t.gy-gy)<10){ ok=false; break; } } }
  return {ok,x:clamp(mx,60,1470),gy}; }
function tryPlace(p){ const sp=spotInfo(p.x,p.y);
  if(!sp.ok){ sfx.error(); addText(p.x,p.y-30,'BLOCKED','#ff6a52',13); return; }
  const cost=TDEFS[S.tool].cost[0];
  if(S.coins<cost){ sfx.error(); addText(sp.x,sp.gy-70,'NEED '+cost+'c','#ff6a52',14); bumpCoins(); return; }
  S.coins-=cost; const t=addTower(S.tool,sp.x,sp.gy); dust(t.x,t.gy); ring(t.x,t.gy-40,'#ffe3b8',TDEFS[S.tool].range[0]); sfx.place(); }

function update(dt){ moteT-=dt; if(moteT<=0){ moteT=.14;
    S.parts.push({type:'mote',x:rand(-200,1800),y:rand(40,760),vx:rand(5,20),vy:rand(-7,-2),life:rand(4,9),max:9,size:rand(1,2.4)}); }
  riftT-=dt; if(riftT<=0){ riftT=.3; S.parts.push({type:'ember',x:30+rand(-8,8),y:274,vx:rand(-14,14),vy:-rand(30,95),life:rand(.8,1.6),max:1.6,size:rand(1.5,3)}); }
  if(S.mode==='build'){ S.buildT-=dt; if(S.buildT<=0) startWave(); }
  else if(S.mode==='wave'){
    if(S.spawnQ.length){ S.spawnT-=dt; while(S.spawnT<=0&&S.spawnQ.length){ spawnFoe(S.spawnQ.shift().t); S.spawnT+=S.spawnQ[0]?S.spawnQ[0].gap:0; if(!S.spawnQ.length)break; } }
    else if(!S.enemies.length) waveClear(); }
  for(const t of S.towers) updateTower(t,dt);
  for(const f of S.enemies){ if(f.dead)continue; f.glowT=Math.max(0,f.glowT-dt); updateFoe(f,dt); }
  foePairs();
  S.enemies=S.enemies.filter(f=>!f.dead);
  updateShots(dt);
  for(const p of S.parts){ p.life-=dt; if(p.g)p.vy+=p.g*dt; p.x+=p.vx*dt; p.y+=p.vy*dt;
    if(p.type==='smoke')p.size+=14*dt; if(p.rot!==undefined&&p.vr)p.rot+=p.vr*dt; }
  S.parts=S.parts.filter(p=>p.life>0); if(S.parts.length>430)S.parts.splice(0,S.parts.length-430);
  for(const t of S.texts){ t.life-=dt; t.y-=28*dt; } S.texts=S.texts.filter(t=>t.life>0);
  for(const s of S.scorch)s.life-=dt; S.scorch=S.scorch.filter(s=>s.life>0);
  S.shake=Math.max(0,S.shake-dt*26); }

function rangeRing(x,y,r,col,fill){ ctx.save(); ctx.strokeStyle=col; ctx.lineWidth=2; ctx.setLineDash([8,7]);
