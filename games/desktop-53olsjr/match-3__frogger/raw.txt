function adjacent(a,b){ const dr=Math.abs(a.r-b.r), dc=Math.min(Math.abs(a.c-b.c),COLS-Math.abs(a.c-b.c)); return dr+dc===1; }

function handleCell(r,c){

  const p={r,c};
  if(!sel){ sel=p; sSelect(); }
  else if(sel.r===r&&sel.c===c){ sel=null; }

}
function doSwap(a,b){
  const tmp=tiles[a.r][a.c]; tiles[a.r][a.c]=tiles[b.r][b.c]; tiles[b.r][b.c]=tmp;
  boardPhase='swap'; swapAnim={a,b,t:0,rev:false}; sSwap();
}
function scanMatches(){
  const runs=[];
  for(let c=0;c<COLS;c++){ let r=0;
    while(r<ROWS){ const t=tiles[r][c].t; if(t<0){r++;continue;}
      let e=r+1; while(e<ROWS&&tiles[e][c].t===t)e++;
      if(e-r>=3){const cells=[];for(let i=r;i<e;i++)cells.push({r:i,c});runs.push({cells});}
      r=e; } }
  for(let r=0;r<ROWS;r++){
    let s=-1;
    for(let i=0;i<COLS;i++) if(tiles[r][i].t!==tiles[r][(i+1)%COLS].t){s=i;break;}
    if(s===-1){ const cells=[];for(let c=0;c<COLS;c++)cells.push({r,c}); runs.push({cells}); continue; }
    let start=(s+1)%COLS, cnt=1;
    for(let step=1;step<=COLS;step++){
      const p=mod(start+step,COLS), q=mod(start+step-1,COLS);
      if(tiles[r][p].t===tiles[r][q].t)cnt++;
      else{ if(cnt>=3){const cells=[];for(let k=0;k<cnt;k++)cells.push({r,c:mod(q-k,COLS)});runs.push({cells});} start=p;cnt=1; }
    }}
  return runs;
}

}
function collapse(){

  }
}

function update(dt){

  for(const d of rain){ d.y+=d.v*dt; d.x-=d.v*.12*dt; if(d.y>H+20){d.y=-20;d.x=rnd(0,W);} }

  const worldActive=!(state==='play'&&paused);
  if(worldActive){
    const tMul=1+.04*(level-1);
    for(let r=0;r<ROWS;r++)laneOff[r]=mod(laneOff[r]+LANES[r].dir*LANES[r].spd*.45*tMul*dt,COLS);
    for(const h of hazards)h.x=mod(h.x+h.dir*h.v*dt,COLS);
  }

  if(state==='play'&&!paused){
    // touch hold repeat
    if((holdL||holdR)&&time-lastHold>.16){ lastHold=time; frogStep(holdL?-1:1); }

    // board machine
    if(boardPhase==='swap'&&swapAnim){
      swapAnim.t+=dt/.15;
      if(swapAnim.t>=1){ swapAnim.t=1;
        const runs=scanMatches();
        if(!swapAnim.rev&&runs.length===0){
          const a=swapAnim.a,b=swapAnim.b,tmp=tiles[a.r][a.c]; tiles[a.r][a.c]=tiles[b.r][b.c]; tiles[b.r][b.c]=tmp;
          swapAnim.rev=true; sInvalid();
        }else if(swapAnim.rev){ boardPhase='idle'; swapAnim=null; }
        else{ chain=1;maxChain=1;pendingCharge=true; beginPop(runs,1); }
      }
    }else if(boardPhase==='pop'){
      popT+=dt/.22;
      for(const cc of popCells){ const o=tiles[cc.r][cc.c]; if(o&&o.pop>=0)o.pop=Math.min(1,popT); }
      if(popT>=1){ for(const cc of popCells){const o=tiles[cc.r][cc.c]; if(o)o.t=-1;} collapse(); boardPhase='fall'; }
    }else if(boardPhase==='fall'){
      let any=false;
      for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++){ const o=tiles[r][c];
        if(o.drop<0){ any=true; o.vy+=45*cell*dt; o.drop+=o.vy*dt;
          if(o.drop>=0){ o.drop=0;o.vy=0;o.sq=.8; } } }
      if(!any){ boardPhase='idle';
        const runs=scanMatches();
        if(runs.length)beginPop(runs,chain+1);
        else if(maxChain>=3){ frog.charges=Math.min(MAX_CHARGE,frog.charges+1); chargeGlow=1; sGulp(); }
      }
    }

    // frog anims
    frog.x+=(frog.col-frog.x)*Math.min(1,dt*12); if(Math.abs(frog.col-frog.x)<.01)frog.x=frog.col;
    if(frog.hopT>=0){ frog.hopT+=dt/.26; if(frog.hopT>=1){frog.hopT=-1;frog.sq=.8;} }
    if(frog.bounceT>=0){ frog.bounceT+=dt/.3; if(frog.bounceT>=1)frog.bounceT=-1; }
    frog.sq=Math.max(0,frog.sq-dt*4);
    frog.invuln=Math.max(0,frog.invuln-dt);

    // level clear sequence
    if(frog.clearing){
      frog.clearT+=dt/.85;
      if(frog.clearT>=1){
        const bonus=250+Math.floor(levelTime*2); score+=bonus;
        banner={main:'LEVEL '+(level+1),sub:'Crossed! +'+bonus+' pts — traffic speeds up',t:0};
        level++; levelTime=LEVEL_TIME; lastTick=-1; buildHazards(); resetFrog(); confetti(); sLevel();
      }
    }else{
      // collision
      if(frog.invuln<=0){
        for(const h of hazards){ if(h.r!==frog.row)continue;
          let d=Math.abs(mod(h.x+h.len/2,COLS)-(frog.x+.5)); d=Math.min(d,COLS-d);
          if(d<h.len/2+.42){ splash(); break; } }
      }
      // timer
      levelTime-=dt;
      if(levelTime<=10){ const s=Math.ceil(levelTime); if(s!==lastTick){lastTick=s;sTick();} }
      if(levelTime<=0){ levelTime=0; endGame('time'); }
    }
  }

  // fx decay
  for(let i=particles.length-1;i>=0;i--){ const p=particles[i]; p.life-=dt;
    if(p.life<=0){particles.splice(i,1);continue;}
    p.vy+=p.g*dt; p.x+=p.vx*dt; p.y+=p.vy*dt; if(p.rot!==undefined)p.rot+=p.vr*dt; }
  for(let i=floaters.length-1;i>=0;i--){ const f=floaters[i]; f.life-=dt/1.1; f.y-=30*dt; if(f.life<=0)floaters.splice(i,1); }
  if(banner){ banner.t+=dt; if(banner.t>2.2)banner=null; }
  shake=Math.max(0,shake-dt*34);
  chargeGlow=Math.max(0,chargeGlow-dt*1.5);
}

**How it plays:** swap adjacent bugs (click/tap, lanes wrap around) — every successful match earns your frog a hop charge. Steer with ◀ ▶ to dodge the cars and logs on its lane, then press ▲/W to spend a charge and climb one lane at a time. Reach the lily-pad home row before the 60-second rain timer runs out; each level brings faster, denser traffic. You get three lily pads of health — splash three times and it's over. P pauses, M mutes, R restarts instantly.
