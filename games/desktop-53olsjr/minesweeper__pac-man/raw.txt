/* ---------- update ---------- */
function update(dt){
  tGlobal+=dt;
  for(const d of bgDots){d.x+=d.vx*dt;d.y+=d.vy*dt;
    if(d.x<-10)d.x=W+10;if(d.x>W+10)d.x=-10;if(d.y<-10)d.y=H+10;if(d.y>H+10)d.y=-10;}
  G.mouth+=dt*(G.sliding?15:6);
  for(let i=parts.length-1;i>=0;i--){const p=parts[i];p.life-=dt;
    if(p.life<=0){parts.splice(i,1);continue;}
    if(p.ring)p.r+=dt*380;else{p.x+=p.vx*dt;p.y+=p.vy*dt;p.vy+=(p.g||0)*dt;}}
  if(parts.length>400)parts.splice(0,parts.length-400);
  G.shake=Math.max(0,G.shake-dt*2.6);
  if(G.toast){G.toast.t+=dt;if(G.toast.t>G.toast.dur)G.toast=null;}
  if(G.state==='playing'){
    if(G.xray>0){G.xray-=dt;if(G.xray<=0){G.xray=0;tone('sine',700,280,0.18,0.08);}}
    if(G.sliding){const s=G.sliding;s.t+=dt/STEP;
      if(s.t>=1){G.px=s.tx;G.py=s.ty;G.sliding=null;arrive();
        if(G.buffer&&G.state==='playing'){const b=G.buffer;G.buffer=null;tryMove(b.x,b.y);}}}
  }else if(G.state==='dying'){
    G.deathT+=dt;if(G.deathT>1.5){G.state='gameover';S.lose();saveHi();}
  }
}
