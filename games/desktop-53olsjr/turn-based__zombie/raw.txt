    try{ this.ctx=new (window.AudioContext||window.webkitAudioContext)();
      this.master=this.ctx.createGain(); this.master.gain.value=.22; this.master.connect(this.ctx.destination);
      const len=this.ctx.sampleRate; this.noiseBuf=this.ctx.createBuffer(1,len,this.ctx.sampleRate);
      const d=this.noiseBuf.getChannelData(0); for(let i=0;i<len;i++)d[i]=Math.random()*2-1;
    }catch(e){} },
  env(g,t0,a,peak,dur){ g.gain.setValueAtTime(.0001,t0); g.gain.linearRampToValueAtTime(peak,t0+a); g.gain.exponentialRampToValueAtTime(.0001,t0+dur); },
  tone(f,o={}){ if(!this.ctx||!this.on)return; const {type='sine',dur=.2,peak=.3,delay=0,slide=0}=o;
    const t0=this.ctx.currentTime+delay, osc=this.ctx.createOscillator(), g=this.ctx.createGain();
    osc.type=type; osc.frequency.setValueAtTime(f,t0);
    if(slide)osc.frequency.exponentialRampToValueAtTime(Math.max(30,f+slide),t0+dur);
    this.env(g,t0,.005,peak,dur); osc.connect(g); g.connect(this.master); osc.start(t0); osc.stop(t0+dur+.06); },
  noise(o={}){ if(!this.ctx||!this.on)return; const {dur=.2,peak=.3,delay=0,type='lowpass',f=800,q=1}=o;
    const t0=this.ctx.currentTime+delay, s=this.ctx.createBufferSource(); s.buffer=this.noiseBuf; s.loop=true;
    const fl=this.ctx.createBiquadFilter(); fl.type=type; fl.frequency.value=f; fl.Q.value=q;
    const g=this.ctx.createGain(); this.env(g,t0,.005,peak,dur);
    s.connect(fl); fl.connect(g); g.connect(this.master); s.start(t0); s.stop(t0+dur+.06); }

function computePos(z){
  const e=ENT[z.lane];
  let x=e.s.x+(e.e.x-e.s.x)*z.dist, y=e.s.y+(e.e.y-e.s.y)*z.dist;
  const ang=Math.atan2(P.y-y,P.x-x), perp=ang+Math.PI/2;
  const off=z.lat+Math.sin(T*2.6+z.ph)*2.5;
  z.px=x+Math.cos(perp)*off; z.py=y+Math.sin(perp)*off;
}
function killZombie(z){
  const i=S.zombies.indexOf(z); if(i<0)return; S.zombies.splice(i,1);
  S.kills++; computePos(z);
  const t=ZT[z.type];
  bloodBurst(z.px,z.py,z.type==='brute'||z.type==='warden'?22:12);
  if(z.type!=='walker')burst(z.px,z.py,6,gore);

function gate(){
  if(!S||S.over)return false;
  if(S.phase!=='player'){err('The horde is moving — wait for your turn.');return false;}
  if(S.ap<1){err('No actions left. End the turn.');return false;}
  return true;
}
function err(msg){ SFX.err(); log(msg,'l-warn'); S.shake=Math.max(S.shake,3); }
function autoTarget(){ return S.zombies.reduce((a,b)=>b.dist>a.dist?b:a,null); }
function smartDoor(){
  let best=null,score=-1e9;
  ENT.forEach(e=>{
    if(S.barr[e.key]>=CFG.maxBarr)return;
    const cnt=S.zombies.filter(z=>ENT[z.lane].key===e.key).length;
    const s=cnt*10-S.barr[e.key];
    if(s>score){score=s;best=e.key;}
  });
  return best;
}
function actShoot(target){
  if(!gate())return;
  target=target||autoTarget();
  if(!target){err('Nothing left to shoot.');return;}
  if(S.ammo<1){err('Dry. Scavenge for ammo, or brace and pray.');return;}
  S.ap--; S.shots++; S.ammo--;
  computePos(target);
  const crit=Math.random()<.22; let d=R(13,19); if(crit)d*=2;
  target.hp-=d; S.dmgDealt+=d; target.flash=1;
  S.aim=Math.atan2(target.py-P.y,target.px-P.x); S.flashT=1;
  SFX.shoot(crit);
  floater(target.px,target.py-ZT[target.type].r-8,crit?('-'+d+' CRIT!'):('-'+d),crit?'#ffdf6b':'#fff',crit?17:14);

}
function actBarricade(key){
  if(!gate())return;
  key=key||smartDoor();
  if(S.plank<1){err('No planks left. Scavenge.');return;}
  const e=ENT.find(x=>x.key===key);
  if(!e||S.barr[key]>=CFG.maxBarr){err('That door is already fully reinforced.');return;}
  S.ap--; S.plank--; S.barr[key]=Math.min(CFG.maxBarr,S.barr[key]+CFG.plankHp);
  woodFx(e,true);
  floater(e.s.x,e.s.y+(e.in.y<0?-14:18),'+'+CFG.plankHp,'#e8a13c',14);
  log('You nail a plank across the '+e.name.toLowerCase()+' ('+S.barr[key]+'/'+CFG.maxBarr+').','l-sys');
  refresh(); updateHint();
}

async function endTurn(){
  if(!S||S.over||S.phase!=='player')return;
  const tok=S.token;
  S.phase='zombie'; refresh(); updateHint();
  await sleep(380); if(tok!==S.token||S.over)return;
  const order=[...S.zombies].filter(z=>z.hp>0).sort((a,b)=>b.dist-a.dist);
  const laneMin={}; let bites=0, doorDmg={N:0,E:0,S:0,W:0};
  for(const z of order){
    if(tok!==S.token||S.over)return;
    const before=S.hp, bd={...doorDmg};
    stepZombie(z,laneMin);
    bites+= (before-S.hp)>0?1:0;
    ENT.forEach(e=>doorDmg[e.key]+=bd[e.key]-S.barr[e.key]>0?Math.max(0,bd[e.key]-( S.barr[e.key])):0);
    await sleep(120); if(tok!==S.token||S.over)return;
  }
  const taken=S.dmgTaken; // summary handled per-bite already
  ENT.forEach(e=>{ if(doorDmg[e.key]>0)log(e.name+': the horde claws it — -'+doorDmg[e.key]+' barricade.','l-sys'); });
  refresh();
  if(S.hp<=0){ die(); return; }
  if(S.round>=CFG.nights){ dawn(); return; }
  S.round++;
  banner('NIGHT '+S.round,SUBS[S.round-1]);
  log('Night '+S.round+' — '+SUBS[S.round-1].toLowerCase()+'.','l-sys');
  await sleep(520); if(tok!==S.token||S.over)return;
  spawnWave();
  S.ap=CFG.maxAp; S.brace=false; S.phase='player';
  refresh(); updateHint();
}

/* ============================== UPDATE ============================== */
function update(dt){
  T+=dt;
  S.shake=Math.max(0,S.shake-dt*26);
  S.flashT=Math.max(0,S.flashT-dt*6);
  if(S.won)S.dawnT=Math.min(.85,S.dawnT+dt*.45);
  if(!S.hover&&S.phase==='player')S.aim+=dt*.35;
  else if(S.hover){ const p=S.hover.type==='zombie'?{x:S.hover.z.px,y:S.hover.z.py}:ENT[S.hover.lane].s; S.aim=Math.atan2(p.y-P.y,p.x-P.x); }
  for(const z of S.zombies){ z.in=Math.min(1,z.in+dt*2.5); z.flash=Math.max(0,z.flash-dt*5); computePos(z); }
  for(let i=S.particles.length-1;i>=0;i--){ const p=S.particles[i]; p.t+=dt;
    if(p.t>p.life){S.particles.splice(i,1);continue;}
    p.x+=p.vx*dt*60/4; p.y+=p.vy*dt*60/4; p.vx*=(1-2.5*dt); p.vy*=(1-2.5*dt);
    if(p.rot!==undefined)p.rot+=p.vr*dt; }
  for(let i=S.floaters.length-1;i>=0;i--){ const f=S.floaters[i]; f.t+=dt; f.y-=30*dt; if(f.t>f.life)S.floaters.splice(i,1); }
  for(let i=S.corpses.length-1;i>=0;i--){ S.corpses[i].t+=dt; if(S.corpses[i].t>S.corpses[i].max)S.corpses.splice(i,1); }
  for(let i=S.decals.length-1;i>=0;i--){ S.decals[i].t+=dt; if(S.decals[i].t>S.decals[i].max)S.decals.splice(i,1); }
  dust.forEach(d=>{ d.x+=d.vx*dt; d.y+=d.vy*dt; if(d.y<105){d.y=535;d.x=R(125,835);} if(d.x<125)d.x=835; if(d.x>835)d.x=125; });
  fog.forEach(f=>{ f.x+=f.vx*dt; f.y+=f.vy*dt; if(f.x<-100)f.x=1060; if(f.x>1060)f.x=-100; if(f.y<-100)f.y=740; if(f.y>740)f.y=-100; });
}

}
