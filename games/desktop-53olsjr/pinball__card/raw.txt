function collideSegs(){
  for(const s of walls){
    if(s.cn){
      const mx=(s.x1+s.x2)/2, my=(s.y1+s.y2)/2;
      if((ball.x-mx)*s.cn[0]+(ball.y-my)*s.cn[1] <= 0) continue;
    }
    const dx=s.x2-s.x1, dy=s.y2-s.y1, L2=dx*dx+dy*dy;
    let t=((ball.x-s.x1)*dx+(ball.y-s.y1)*dy)/L2; t=t<0?0:(t>1?1:t);
    const cx=s.x1+dx*t, cy=s.y1+dy*t;
    let nx=ball.x-cx, ny=ball.y-cy;
    const d2=nx*nx+ny*ny;
    if(d2>=BR*BR || d2===0) continue;
    const d=Math.sqrt(d2); nx/=d; ny/=d;
    ball.x=cx+nx*BR; ball.y=cy+ny*BR;
    const vn=ball.vx*nx+ball.vy*ny;
    if(vn<0){
      ball.vx-=(1+s.e)*vn*nx; ball.vy-=(1+s.e)*vn*ny;

    }
  }
}
function collideBumpers(){

function collideCards(){

function update(dt){
  if(effL()&&!prevL) SFX.flip();
  if(effR()&&!prevR) SFX.flip();
  prevL=effL(); prevR=effR();
  stepFlipper(Lf, effL(), dt);
  stepFlipper(Rf, effR(), dt);

  const inPad = ball.x>452 && ball.y>760 && phase==='run';
  if(effP() && inPad){
    charging=true;
    power+=pDir*dt*1.1;
    if(power>=1){power=1;pDir=-1;}
    if(power<=0){power=0;pDir=1;}
  } else {
    if(charging && inPad){
      ball.vy=-(950+1150*power); SFX.launch(power); shake=Math.max(shake,2);
    }
    charging=false; power=0; pDir=1;
  }

  if(phase==='run'){
    flipContact=false;
    const N=8, h=dt/N;
    for(let i=0;i<N;i++){
      ball.vy+=G*h; capV();
      ball.x+=ball.vx*h; ball.y+=ball.vy*h;
      collideSegs(); collideBumpers(); collideCards(); collideFlippers(); sensors(); capV();
    }
    ball.trail.push({x:ball.x,y:ball.y});
    if(ball.trail.length>12) ball.trail.shift();

    if(ball.y>H+40){ loseBall(); }
    else{
      const spd=Math.hypot(ball.vx,ball.vy);
      if(spd<18 && !inPad && !flipContact){
        stuckT+=dt;
        if(stuckT>1.2){ stuckT=0; ball.vx+=(Math.random()*80-40); ball.vy-=70; SFX.tick(); }
      } else stuckT=0;
    }
  }

  if(phase==='lost'){
    lostT+=dt;
    if(lostT>1.5){
      if(balls>0){ resetBall(); phase='run'; }
      else endGame();
    }
  }
  if(royalT>0){ royalT-=dt; if(royalT<=0) dealCards(Math.floor(Math.random()*4)); }
  if(laneT>0){ laneT-=dt; if(laneT<=0) lanes.forEach(l=>{l.lit=false;}); }
}

function ambient(dt){
  for(const s of bgSuits){ s.y-=s.sp*dt; if(s.y<-0.06){ s.y=1.06; s.nx=Math.random(); } }
  bumpers.forEach(b=>{ b.flash=Math.max(0,b.flash-dt*3); b.cd=Math.max(0,b.cd-dt); });
  lanes.forEach(l=>{ l.flash=Math.max(0,l.flash-dt*2.5); });
  slings.forEach(s=>{ s.t=Math.max(0,s.t-dt); });
  cards.forEach(c=>{ c.dealT=Math.min(1,c.dealT+dt/0.32); });
  for(let i=texts.length-1;i>=0;i--){ texts[i].t+=dt; if(texts[i].t>0.95) texts.splice(i,1); }
  for(let i=parts.length-1;i>=0;i--){
    const p=parts[i]; p.t+=dt;
    if(p.t>p.life){ parts.splice(i,1); continue; }
    p.vy+=1000*dt; p.x+=p.vx*dt; p.y+=p.vy*dt; p.rot+=p.vr*dt;
  }
  if(banner){ banner.t+=dt; if(banner.t>banner.dur) banner=null; }
  shake=Math.max(0,shake-dt*14);
}
