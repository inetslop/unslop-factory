function collideRect(o,r,b,rest,isP){
  const cx=clamp(o.x,b.x,b.x+b.w),cy=clamp(o.y,b.y,b.y+b.h);
  let dx=o.x-cx,dy=o.y-cy;const d2=dx*dx+dy*dy;
  if(d2>=r*r)return;
  let nx,ny,l=Math.sqrt(d2);
  if(l>0.001){nx=dx/l;ny=dy/l;o.x=cx+nx*r;o.y=cy+ny*r}
  else{
    const l1=o.x-b.x,r1=b.x+b.w-o.x,t1=o.y-b.y,b1=b.y+b.h-o.y,m=Math.min(l1,r1,t1,b1);
    if(m===l1){nx=-1;ny=0;o.x=b.x-r}
    else if(m===r1){nx=1;ny=0;o.x=b.x+b.w+r}
    else if(m===t1){nx=0;ny=-1;o.y=b.y-r}
    else{nx=0;ny=1;o.y=b.y+b.h+r}
  }
