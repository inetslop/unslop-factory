  p.camera.x = (p.world.x || 0) - cameraX;

  p.camera.z = (p.world.z || 0) - cameraZ;

}

function addRing(x,y,col){parts.push({ring:true,x,y,r:8,life:.55,max:.55,col});}
function updateParts(dt){for(let i=parts.length-1;i>=0;i--){const p=parts[i];p.life-=dt;
  if(p.life<=0){parts.splice(i,1);continue;}
  if(p.ring)p.r+=340*dt;else{p.x+=p.vx*dt;p.y+=p.vy*dt;p.vy+=p.grav*dt;}}}
function popup(txt,x,y,col,size){pops.push({txt,x,y,vy:-52,life:1.15,max:1.15,col:col||'#fff',size:size||Math.max(16,W*.02)});}
function updatePops(dt){for(let i=pops.length-1;i>=0;i--){const p=pops[i];p.life-=dt;p.y+=p.vy*dt;if(p.life<=0)pops.splice(i,1);}}

const AU={ctx:null,master:null,noise:null,o1:null,o2:null,sub:null,ef:null,eg:null,wb:null,wg:null,ng:null};
function audioInit(){
  if(AU.ctx){if(AU.ctx.state==='suspended')AU.ctx.resume();return;}
  try{
    const C=AU.ctx=new (window.AudioContext||window.webkitAudioContext)();
    AU.master=C.createGain();AU.master.gain.value=muted?0:.9;AU.master.connect(C.destination);
    // engine
    AU.ef=C.createBiquadFilter();AU.ef.type='lowpass';AU.ef.frequency.value=500;
    AU.eg=C.createGain();AU.eg.gain.value=0;
    AU.o1=C.createOscillator();AU.o1.type='sawtooth';
    AU.o2=C.createOscillator();AU.o2.type='square';
    AU.sub=C.createOscillator();AU.sub.type='sine';
    const g2=C.createGain();g2.gain.value=.5;const gs=C.createGain();gs.gain.value=.6;
    AU.o1.connect(AU.ef);AU.o2.connect(g2);g2.connect(AU.ef);AU.sub.connect(gs);gs.connect(AU.ef);
    AU.ef.connect(AU.eg);AU.eg.connect(AU.master);
    AU.o1.start();AU.o2.start();AU.sub.start();
    // noise buffer
    const len=C.sampleRate*2,buf=C.createBuffer(1,len,C.sampleRate),ch=buf.getChannelData(0);
    for(let i=0;i<len;i++)ch[i]=Math.random()*2-1;AU.noise=buf;
    // wind
    const ws=C.createBufferSource();ws.buffer=buf;ws.loop=true;
    AU.wb=C.createBiquadFilter();AU.wb.type='bandpass';AU.wb.frequency.value=600;AU.wb.Q.value=.7;
    AU.wg=C.createGain();AU.wg.gain.value=0;ws.connect(AU.wb);AU.wb.connect(AU.wg);AU.wg.connect(AU.master);ws.start();
    // nitro hiss
    const ns=C.createBufferSource();ns.buffer=buf;ns.loop=true;
    const nh=C.createBiquadFilter();nh.type='highpass';nh.frequency.value=1100;
    AU.ng=C.createGain();AU.ng.gain.value=0;ns.connect(nh);nh.connect(AU.ng);AU.ng.connect(AU.master);ns.start();
  }catch(e){}
}
function beep(f0,f1,dur,type,vol){if(!AU.ctx||muted)return;const C=AU.ctx,t=C.currentTime,o=C.createOscillator(),g=C.createGain();
  o.type=type||'square';o.frequency.setValueAtTime(f0,t);o.frequency.exponentialRampToValueAtTime(Math.max(30,f1),t+dur);
  g.gain.setValueAtTime(vol||.2,t);g.gain.exponentialRampToValueAtTime(.001,t+dur);
  o.connect(g);g.connect(AU.master);o.start(t);o.stop(t+dur+.03);}
function crashSfx(){if(!AU.ctx||muted)return;const C=AU.ctx,t=C.currentTime;
  const s=C.createBufferSource();s.buffer=AU.noise;const f=C.createBiquadFilter();f.type='lowpass';
  f.frequency.setValueAtTime(2800,t);f.frequency.exponentialRampToValueAtTime(140,t+.5);
  const g=C.createGain();g.gain.setValueAtTime(.55,t);g.gain.exponentialRampToValueAtTime(.001,t+.6);
  s.connect(f);f.connect(g);g.connect(AU.master);s.start(t);s.stop(t+.62);
  const o=C.createOscillator(),og=C.createGain();o.type='sine';
  o.frequency.setValueAtTime(150,t);o.frequency.exponentialRampToValueAtTime(34,t+.45);
  og.gain.setValueAtTime(.5,t);og.gain.exponentialRampToValueAtTime(.001,t+.5);
  o.connect(og);og.connect(AU.master);o.start(t);o.stop(t+.52);}
function audioUpdate(spct,isBoost,offroad){
  if(!AU.ctx)return;const C=AU.ctx,t=C.currentTime,on=(mode!=='over'&&!paused);
  const f=46+spct*132*(isBoost?1.18:1);
  AU.o1.frequency.setTargetAtTime(f,t,.06);
  AU.o2.frequency.setTargetAtTime(f*.5,t,.06);
  AU.sub.frequency.setTargetAtTime(f*.25,t,.06);
  AU.ef.frequency.setTargetAtTime(250+spct*900,t,.08);
  AU.eg.gain.setTargetAtTime(on?.04+spct*.11+(isBoost?.04:0):0,t,.09);
  AU.wb.frequency.setTargetAtTime(400+spct*1600,t,.1);
  AU.wg.gain.setTargetAtTime(on?spct*spct*.15+(offroad?.06:0):0,t,.1);
  AU.ng.gain.setTargetAtTime(on&&isBoost?.13:0,t,.07);
}

/* ============================== updates ============================== */
function curvePull(dt){const s=getSeg(Math.floor(playerZ/SEG));const sp=speed/MAXSP;
  if(s)playerX-=s.c*sp*sp*dt*.24;}
function updatePlay(dt){
  const spct=speed/MAXSP;
  boosting=keys.boost&&nitro>0;
  if(boosting)nitro=Math.max(0,nitro-34*dt);else nitro=Math.min(100,nitro+11*dt);
  const maxS=MAXSP*(boosting?NITRO_M:1);
  if(keys.up||touch.active)speed+=ACCEL*(boosting?1.9:1)*dt;
  else speed-=COAST*dt;
  if(keys.down)speed-=BRAKE*dt;
  if(speed>maxS)speed=Math.max(maxS,speed-MAXSP*.9*dt);
  const offroad=Math.abs(playerX)>1.15&&speed>MAXSP*.4;
  if(offroad){speed-=MAXSP*1.1*dt;shake=Math.max(shake,.12);
    const cw=Math.max(90,Math.min(W*.17,230)),cx=W/2,cy=H-cw*.4;
    addP(cx+(Math.random()<.5?-1:1)*cw*.45,cy+cw*.3,(Math.random()-.5)*60,-40-Math.random()*50,.5,4+Math.random()*4,'rgba(170,150,110,.5)',-20);}
  speed=clamp(speed,0,maxS);
  let steer=(keys.right?1:0)-(keys.left?1:0);
  if(touch.active){const tgt=clamp((touch.x/W)*2-1,-.9,.9)*.85;
    playerX+=clamp(tgt-playerX,-dt*1.9,dt*1.9);steerIn=clamp((tgt-playerX)*3,-1,1);}
  else{steerIn+=(steer-steerIn)*Math.min(1,dt*10);playerX+=steerIn*dt*2.2*(.35+.65*spct);}
  curvePull(dt);
  playerX=clamp(playerX,-1.9,1.9);
  playerZ+=speed*dt;distM+=speed*dt*MTR;score=Math.floor(distM)+overtakes*250;
  topKmh=Math.max(topKmh,speed*KMH);
  if(distM>=nextMile){popup((nextMile/1000)+' KM!',W/2,H*.34,'#ffd9a0',Math.max(22,W*.03));beep(520,860,.16,'square',.2);nextMile+=1000;}
  // traffic: overtake + collision
  for(const v of traffic){
    const rel=v.z-playerZ;
    if(!v.passed&&rel<-400){v.passed=true;overtakes++;score+=250;popup('OVERTAKE +250',W*.3,H*.4,'#9fe8d8');beep(880,1400,.12,'square',.2);}
    if(Math.abs(rel)<(PL_HALF+v.wW/2/ROADW)*.85*(ROADW/ (v.wW/2/ROADW))*0 + ((1300+v.len)/2)*.7 && Math.abs(v.off-playerX)<(PL_HALF+v.wW/2/ROADW)*.85){doCrash();return;}
  }
  updateTraffic(dt);
}
function updateAttract(dt){
  const tgt=MAXSP*.62;speed+=(tgt-speed)*Math.min(1,dt*.7);
  let want=Math.sin(tSec*.4)*.12;
  for(const v of traffic){const rel=v.z-playerZ;
    if(rel>0&&rel<30*SEG&&Math.abs(v.off-playerX)<.5){want=(playerX>v.off?1:-1)*.8;break;}}
  playerX+=clamp(want-playerX,-dt*1.2,dt*1.2);steerIn=clamp(want,-1,1);
  curvePull(dt);playerZ+=speed*dt;distM+=speed*dt*MTR;updateTraffic(dt);
}
function updateWorld(rdt,dt){
  tod=(tod+rdt/100)%1;
  const bi=Math.floor(playerZ/SEG),s=getSeg(bi);
  if(s){const pct=(playerZ-bi*SEG)/SEG;playerY=s.y1+(s.y2-s.y1)*pct;
    bgShift+=s.c*(speed/MAXSP)*rdt*40;}
  ensure(bi+DRAW+80);prune(bi);
  if(mode==='menu')updateAttract(dt);
  else if(mode==='play'&&!paused)updatePlay(dt);
  else if(mode==='count'){countT-=rdt;speed=0;
    const step=Math.ceil(countT/.55);
    if(step!==updateWorld.lastStep){updateWorld.lastStep=step;beep(step>0?440:880,step>0?440:1320,.1,'square',.2);}
    if(countT<=-.2)mode='play';}
  else if(mode==='crash'){crashT+=rdt;speed=Math.max(0,speed-MAXSP*1.4*dt);playerZ+=speed*dt;crashRot+=rdt*5;
    if(crashT>1.5)finishRun();}
  shake=Math.max(0,shake-rdt*1.3);
  updateParts(dt);updatePops(dt);
  audioUpdate(mode==='count'?(.72+Math.sin(tSec*18)*.18):clamp(speed/MAXSP/NITRO_M,0,1),boosting,mode==='play'&&Math.abs(playerX)>1.15&&speed>MAXSP*.4);
}

  ctx.lineTo(W+4,H);ctx.closePath();ctx.fill();}

/* ============================== render: road ============================== */
const SX=new Float32Array(DRAW+2),SY=new Float32Array(DRAW+2),SW=new Float32Array(DRAW+2),
  RV=new Float32Array(DRAW+2),CL=new Float32Array(DRAW+2);
let roadHor=H/2;
function renderRoad(pal){
  const bi=Math.floor(playerZ/SEG),bseg=getSeg(bi);if(!bseg)return;
  const basePct=(playerZ-bi*SEG)/SEG,camY=playerY+CAMH,camX=playerX*ROADW,halfW=W/2,halfH=H/2;
  let x=0,dx=-(bseg.c*basePct),horizon=H;roadHor=H/2;
  for(let n=1;n<DRAW;n++){
    const s=getSeg(bi+n);if(!s)break;
    const z1=(bi+n)*SEG-playerZ,z2=z1+SEG,r1=DEPTH/z1,r2=DEPTH/z2;
    const sx1=halfW+r1*(x-camX)*halfW, sy1=halfH-r1*(s.y1-camY)*halfH;
    const sx2=halfW+r2*(x+dx-camX)*halfW, sy2=halfH-r2*(s.y2-camY)*halfH;
    SX[n]=sx1;SY[n]=sy1;SW[n]=r1*ROADW*halfW;RV[n]=r1;CL[n]=horizon;
    x+=dx;dx+=s.c;
    if(!(sy2<horizon&&sy2<sy1))continue;
    horizon=sy2;roadHor=horizon;
    const band=((bi+n)/RUMBLE|0)%2;
    // ground
    ctx.fillStyle=band?rgb(pal.gd):rgb(pal.gl);
    ctx.fillRect(0,sy2,W,Math.max(1,sy1-sy2));
    const w1=r1*ROADW*halfW,w2=r2*ROADW*halfW;
    // road
    ctx.fillStyle=band?rgb(pal.rd):rgb(pal.rl);
    ctx.beginPath();ctx.moveTo(sx1-w1,sy1);ctx.lineTo(sx1+w1,sy1);ctx.lineTo(sx2+w2,sy2);ctx.lineTo(sx2-w2,sy2);ctx.closePath();ctx.fill();
    // rumble strips
    ctx.fillStyle=band?'#e8e6df':'#d94141';
    ctx.beginPath();ctx.moveTo(sx1-w1,sy1);ctx.lineTo(sx1-w1*1.17,sy1);ctx.lineTo(sx2-w2*1.17,sy2);ctx.lineTo(sx2-w2,sy2);ctx.closePath();ctx.fill();
    ctx.beginPath();ctx.moveTo(sx1+w1,sy1);ctx.lineTo(sx1+w1*1.17,sy1);ctx.lineTo(sx2+w2*1.17,sy2);ctx.lineTo(sx2+w2,sy2);ctx.closePath();ctx.fill();
    // lane dashes
    if(band===0){ctx.fillStyle='rgba(223,227,234,.85)';
      for(const f of[-.5,.5]){const lw=Math.max(1,w1*.02),lw2=Math.max(1,w2*.02);
        ctx.beginPath();ctx.moveTo(sx1+f*w1-lw,sy1);ctx.lineTo(sx1+f*w1+lw,sy1);ctx.lineTo(sx2+f*w2+lw2,sy2);ctx.lineTo(sx2+f*w2-lw2,sy2);ctx.closePath();ctx.fill();}}
    // center double yellow
    ctx.fillStyle='rgba(232,195,58,.9)';
    for(const f of[-.035,.035]){const lw=Math.max(.8,w1*.012),lw2=Math.max(.8,w2*.012);
      ctx.beginPath();ctx.moveTo(sx1+f*w1-lw,sy1);ctx.lineTo(sx1+f*w1+lw,sy1);ctx.lineTo(sx2+f*w2+lw2,sy2);ctx.lineTo(sx2+f*w2-lw2,sy2);ctx.closePath();ctx.fill();}
  }
  // fog band
  const top=Math.min(roadHor,H/2)-70,bot=roadHor+H*.3;
  const fg=ctx.createLinearGradient(0,top,0,bot);
  fg.addColorStop(0,rgba(pal.hor,0));fg.addColorStop(.14,rgba(pal.hor,.5));
  fg.addColorStop(.5,rgba(pal.hor,.26));fg.addColorStop(1,rgba(pal.hor,0));
  ctx.fillStyle=fg;ctx.fillRect(0,top,W,bot-top);
}

const P={x:0,y:0,w:0,r:0,clip:0};
function projAt(nF){
  const n0=nF|0;if(n0<1||n0+1>=DRAW)return false;
  const fr=nF-n0;
  P.x=lerp(SX[n0],SX[n0+1],fr);P.y=lerp(SY[n0],SY[n0+1],fr);
  P.w=lerp(SW[n0],SW[n0+1],fr);P.r=lerp(RV[n0],RV[n0+1],fr);P.clip=CL[n0];
  return true;}
function drawTrafficSprite(v){
  const nF=(v.z-playerZ)/SEG;if(nF<1.2||nF>DRAW-1)return;
  if(!projAt(nF))return;
  if(P.y>P.clip+6)return;
  const fa=Math.exp(-Math.pow(nF/DRAW,2)*FOGD);if(fa<.03)return;
  ctx.globalAlpha=fa;
  const w=v.wW*P.r*(W/2);if(w<2){ctx.globalAlpha=1;return;}
  const x=P.x+P.w*v.off,h=w*(v.truck?1.5:.66),y=P.y;
  ctx.fillStyle='rgba(0,0,0,.3)';ctx.beginPath();ctx.ellipse(x,y+h*.04,w*.52,h*.09,0,0,7);ctx.fill();
  ctx.fillStyle='#14161d';
  ctx.fillRect(x-w*.48,y-h*.1,w*.16,h*.13);ctx.fillRect(x+w*.32,y-h*.1,w*.16,h*.13);
  ctx.fillStyle=v.color;ctx.fillRect(x-w/2,y-h,w,h);
  ctx.fillStyle='rgba(0,0,0,.28)';ctx.fillRect(x-w/2,y-h*.42,w,h*.42);

  ctx.fillStyle='#ff4136';

  ctx.globalAlpha=1;}
function drawPropSprite(p){
  const w=(p.t===0?900:p.t===1?1300:520)*P.r*(W/2);if(w<2)return;
  const x=P.x+P.w*p.off,y=P.y,col='rgba(20,14,26,.95)';
  ctx.fillStyle=col;ctx.strokeStyle=col;
  if(p.t===0){ // palm
    const h=w*1.7,tx=x+w*.32,ty=y-h*.78;
    ctx.lineWidth=Math.max(1,w*.13);
    ctx.beginPath();ctx.moveTo(x,y);ctx.quadraticCurveTo(x-w*.12,y-h*.42,tx,ty);ctx.stroke();
    ctx.lineWidth=Math.max(1,w*.09);
    for(let i=0;i<6;i++){const a=-2.6+i*.5;
      ctx.beginPath();ctx.moveTo(tx,ty);
      ctx.quadraticCurveTo(tx+Math.cos(a)*w*.55,ty+Math.sin(a)*w*.4-w*.1,tx+Math.cos(a)*w*.95,ty+Math.sin(a)*w*.62+w*.28);ctx.stroke();}
  }else if(p.t===1){ // rock
    ctx.beginPath();
    for(let i=0;i<=7;i++){const a=Math.PI+i/7*Math.PI,rr=.7+.3*Math.sin(p.seed*3+i*2.4);
      const px=x+Math.cos(a)*w*.5*rr,py=y-Math.abs(Math.sin(a))*h? y-w*.42*rr:0;
      i===0?ctx.moveTo(px,y-w*.4*rr):ctx.lineTo(px,y-w*.4*rr);}
    ctx.closePath();ctx.fill();
  }else{ // cactus
    const h=w*1.9,tw=Math.max(1.5,w*.32);
    ctx.fillRect(x-tw/2,y-h,tw,h);
    ctx.fillRect(x-tw/2-w*.4,y-h*.78,w*.4,w*.26);ctx.fillRect(x-tw/2-w*.4,y-h*.78,w*.26,h*.4);
    ctx.fillRect(x+tw/2,y-h*.6,w*.34,w*.24);ctx.fillRect(x+tw/2+w*.08,y-h*.86,w*.26,h*.3);}
}
function renderSprites(pal){
  const bi=Math.floor(playerZ/SEG),items=[];
  for(let i=bi+DRAW-1;i>=bi+2;i--){const s=getSeg(i);if(s&&s.p)items.push({z:i*SEG,pr:s.p});}
  for(const v of traffic)items.push({z:v.z,v});
  items.sort((a,b)=>b.z-a.z);
  const pcol=mix(pal.gd,[8,5,12],.35);
  ctx.fillStyle=rgb(pcol);
  for(const it of items){
    const nF=(it.z-playerZ)/SEG;if(nF<1||nF>DRAW-1)continue;
    if(!projAt(nF))continue;
    if(P.y>P.clip+6)continue;
    const fa=Math.exp(-Math.pow(nF/DRAW,2)*FOGD);if(fa<.03)continue;
    ctx.globalAlpha=fa;
    if(it.v)drawTrafficSprite.__noop; // (handled below with alpha set)
  }
  // second pass with proper alpha per item
  for(const it of items){
    const nF=(it.z-playerZ)/SEG;if(nF<1||nF>DRAW-1)continue;
    if(!projAt(nF))continue;
    if(P.y>P.clip+6)continue;
    const fa=Math.exp(-Math.pow(nF/DRAW,2)*FOGD);if(fa<.03)continue;
    ctx.globalAlpha=fa;
    if(it.pr){ctx.fillStyle=rgb(pcol);drawPropSprite(it.pr);}
    else drawTrafficSprite(it.v);
  }
  ctx.globalAlpha=1;}
