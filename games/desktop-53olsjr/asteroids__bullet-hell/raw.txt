let AC=null, master=null, noiseBuf=null, thrustGain=null, muted=false, lastGrazeSfx=0;
function initAudio(){
  if(AC) return;
  try{
    AC=new (window.AudioContext||window.webkitAudioContext)();
    master=AC.createGain(); master.gain.value=muted?0:0.5; master.connect(AC.destination);
    const len=Math.floor(AC.sampleRate*1.2);
    noiseBuf=AC.createBuffer(1,len,AC.sampleRate);
    const d=noiseBuf.getChannelData(0);
    for(let i=0;i<len;i++) d[i]=Math.random()*2-1;
    // thrust loop
    const src=AC.createBufferSource(); src.buffer=noiseBuf; src.loop=true;
    const f=AC.createBiquadFilter(); f.type='bandpass'; f.frequency.value=340; f.Q.value=.7;
    thrustGain=AC.createGain(); thrustGain.gain.value=0;
    src.connect(f); f.connect(thrustGain); thrustGain.connect(master); src.start();
  }catch(e){ AC=null; }
}
function tone(f0,f1,dur,type,vol,delay){
  if(!AC||muted) return;
  const t=AC.currentTime+(delay||0);
  const o=AC.createOscillator(), g=AC.createGain();
  o.type=type||'square';
  o.frequency.setValueAtTime(Math.max(1,f0),t);
  if(f1&&f1!==f0) o.frequency.exponentialRampToValueAtTime(Math.max(1,f1),t+dur);
  g.gain.setValueAtTime(vol||.2,t);
  g.gain.exponentialRampToValueAtTime(.0001,t+dur);
  o.connect(g); g.connect(master);
  o.start(t); o.stop(t+dur+.03);
}
function noiseBurst(dur,f0,f1,vol){
  if(!AC||muted) return;
  const t=AC.currentTime;
  const src=AC.createBufferSource(); src.buffer=noiseBuf;
  const f=AC.createBiquadFilter(); f.type='lowpass';
  f.frequency.setValueAtTime(f0,t);
  f.frequency.exponentialRampToValueAtTime(Math.max(30,f1),t+dur);
  const g=AC.createGain();
  g.gain.setValueAtTime(vol,t);
  g.gain.exponentialRampToValueAtTime(.0001,t+dur);
  src.connect(f); f.connect(g); g.connect(master);
  src.start(t); src.stop(t+dur+.05);
}

function startWave(n){
  G.wave=n;
  const isBoss=(n%5===0);
  if(isBoss){ showBanner('THE CORE','WAVE '+pad(n,2)+' · DO NOT LET IT SETTLE'); sfx.boss(); spawnBoss(); spawnAsteroid(2,n); spawnAsteroid(2,n); }
  else{
    const name=WAVENAMES[(n-1)%8];
    showBanner('WAVE '+pad(n,2),name); sfx.wave();
    const big=Math.min(2+Math.ceil(n/2),7);
    for(let i=0;i<big;i++) spawnAsteroid(3,n);
    if(n>=2){ const wv=Math.min(Math.floor((n+1)/3),4); for(let i=0;i<wv;i++) spawnDrone('weevil'); }
    if(n>=4){ const mo=Math.min(Math.floor((n-2)/3),3); for(let i=0;i<mo;i++) spawnDrone('moth'); }
  }
}

}

function addScore(n,x,y,color){
  G.score+=n; if(G.score>G.hi)G.hi=G.score;
  if(x!==undefined) floater(x,y,'+'+n,color);
}
function killAsteroid(a,pts){
  a.dead=true;
  const c=[0,[251,191,36],[110,231,183],[74,222,188]][a.size];
  boom(a.x,a.y,a.size===3?2.1:a.size===2?1.4:.9,c);
  if(pts){ const ptsV=[0,100,50,20][a.size]; addScore(ptsV,a.x,a.y,'#8ef7dc'); }
  if(a.size>1){
    const ang=rand(0,TAU), s2=rand(36,85);
    const c1=makeAsteroid(a.x+6,a.y,a.size-1,a.vx*.4+Math.cos(ang)*s2,a.vy*.4+Math.sin(ang)*s2,G.wave);
    const c2=makeAsteroid(a.x-6,a.y,a.size-1,a.vx*.4-Math.cos(ang)*s2,a.vy*.4-Math.sin(ang)*s2,G.wave);
    asteroids.push(c1,c2);
  }
}
function killDrone(d,pts){
  d.dead=true;
  boom(d.x,d.y,1.6,d.type==='weevil'?AMBER:MAG);
  if(pts) addScore(d.type==='weevil'?150:200,d.x,d.y,d.type==='weevil'?'#ffd9a8':'#e9c8ff');
  if(!pts||Math.random()<.3) pickups.push({x:d.x,y:d.y,t:0,life:12});
}
function bossDie(){
  const b=boss; boss=null; elBossWrap.classList.remove('on');
  addScore(1500,b.x,b.y,'#ffd9a8');
  boom(b.x,b.y,3.4,CORAL); G.shake=26;
  pickups.push({x:b.x,y:b.y,t:0,life:14});
  for(let i=0;i<6;i++){
    timers.push({t:G.timeNow()+i*.16,fn:(xx,yy)=>{ boom(b.x+rand(-50,50),b.y+rand(-50,50),rand(1,2.2),[AMBER[i%2?0:0],CORAL]); }});
  }
}
function hitPlayer(){
  if(!ship.alive||ship.inv>0) return;
  G.lives--; setThrust(false);
  boom(ship.x,ship.y,2.6,CYAN);
  timers.push({t:G.timeNow()+.18,fn:(x,y)=>boom(ship.x+rand(-24,24),ship.y+rand(-24,24),1.4,[255,255,255])});
  redFlash(); sfx.hurt(); G.shake=18;
  ship.alive=false;
  if(G.lives<=0){ G.overT=1.3; } else { G.respawnT=1.15; }
}
function doRespawn(){
  ship.x=W/2; ship.y=H*.62; ship.a=-Math.PI/2; ship.vx=ship.vy=0; ship.inv=2.6; ship.alive=true;
  ebullets=ebullets.filter(b=>{
    const d=Math.hypot(b.x-ship.x,b.y-ship.y);
    if(d<190){ sparkBurst(b.x,b.y,2,CYAN); return false; } return true; });
}
function fireBomb(){
  if(G.bombs<=0||!ship.alive) return;
  G.bombs--; sfx.bomb(); G.whiteFlash=1; G.shake=Math.min(G.shake+13,26);
  addPart({type:'ring',x:ship.x,y:ship.y,r0:12,r1:Math.max(W,H)*.75,life:.55,max:.55,c:[255,255,255]});
  let cleared=0;
  for(const b of ebullets){ if(Math.random()<.3) sparkBurst(b.x,b.y,1,[255,255,255]); cleared++; }
  ebullets.length=0;
  addScore(cleared*2,ship.x,ship.y-30,'#ffffff');
  const knock=o=>{ const dx=o.x-ship.x,dy=o.y-ship.y,d=Math.hypot(dx,dy)||1;
    const f=clamp(200-d*.25,40,190); o.vx+=dx/d*f; o.vy+=dy/d*f; };
  asteroids.forEach(knock); drones.forEach(knock); if(boss){boss.vx=(boss.vx||0)+rand(-30,30);}
  for(const d of drones){ d.hp-=3; d.flash=.15; }
  if(boss){ boss.hp-=6; boss.flash=.2; }
}

function updateStars(dt){
  for(const s of stars){ s.x-=s.s*dt*.55; s.y-=s.s*dt*.28; if(s.x<-4)s.x+=W+8; if(s.y<-4)s.y+=H+8; }
}
function updateFx(dt){
  G.shake=Math.max(0,G.shake-dt*34);
  G.whiteFlash=Math.max(0,G.whiteFlash-dt*2.4);
  for(let i=particles.length-1;i>=0;i--){ const p=particles[i]; p.life-=dt;
    if(p.life<=0){particles.splice(i,1);continue;} p.x+=p.vx*dt; p.y+=p.vy*dt; }
  for(let i=floaters.length-1;i>=0;i--){ const f=floaters[i]; f.life-=dt; f.y-=26*dt; if(f.life<=0)floaters.splice(i,1); }
  for(let i=timers.length-1;i>=0;i--){ if(T>=timers[i].t){ timers[i].fn(); timers.splice(i,1); } }
  G.streakT-=dt;
  if(G.streakT<=0){ G.streakT=rand(3.5,7);
    const a=rand(Math.PI*.15,Math.PI*.4)+(Math.random()<.5?0:Math.PI);
    addPart({type:'streak',x:rand(-60,W+60),y:rand(-20,H*.5),vx:Math.cos(a)*rand(500,800),vy:Math.abs(Math.sin(a))*rand(300,500)*(a>Math.PI?1:-1),life:.9,max:.9,c:[190,235,255]});
  }
}
function updateWorld(dt){
  const fire=ship.alive&&G.state==='playing';
  /* asteroids */
  for(const a of asteroids){
    a.x+=a.vx*dt; a.y+=a.vy*dt; a.a+=a.va*dt; wrap(a,a.r);
    if(fire&&a.size===1){
      a.ft-=dt;
      if(a.ft<=0){ a.ft=rand(1.6,3.4);
        if(Math.random()<Math.min(.25+G.wave*.05,.8)){
          const an=Math.atan2(ship.y-a.y,ship.x-a.x)+rand(-.35,.35);
          eShot(a.x,a.y,an,Math.min(150+G.wave*6,240),CORAL);
        } }
    }
  }
  /* drones */
  for(const d of drones){
    d.t+=dt; d.flash=Math.max(0,d.flash-dt);
    if(d.type==='weevil'){
      if(d.entering){
        d.vx+=(d.gate.x-d.x)*4*dt; d.vy+=(d.gate.y-d.y)*4*dt;
        const dm=Math.exp(-2*dt); d.vx*=dm; d.vy*=dm;
        if(Math.hypot(d.gate.x-d.x,d.gate.y-d.y)<16||d.t>3.2) d.entering=false;
      }else{
        const ang=d.phase+d.t*.45, tx=ship.x+Math.cos(ang)*270, ty=ship.y+Math.sin(ang)*270;
        d.vx+=(tx-d.x)*2.1*dt; d.vy+=(ty-d.y)*2.1*dt;
        const dm=Math.exp(-2.1*dt); d.vx*=dm; d.vy*=dm;
        const sp=Math.hypot(d.vx,d.vy); if(sp>230){d.vx*=230/sp;d.vy*=230/sp;}
      }
      d.x+=d.vx*dt; d.y+=d.vy*dt;
      if(fire&&!d.entering&&d.ft!==undefined){
        d.ft-=dt;
        if(d.ft<=0&&d.x>-20&&d.x<W+20&&d.y>-20&&d.y<H+20&&Math.hypot(ship.x-d.x,ship.y-d.y)<640){
          d.ft=1.7; const an=Math.atan2(ship.y-d.y,ship.x-d.x);
          for(let i=-2;i<=2;i++) eShot(d.x,d.y,an+i*.13+rand(-.02,.02),Math.min(140+G.wave*6,235),AMBER);
          sfx.tick();
        }
      }
    }else{ /* moth */
      if(d.entering){
        d.vx+=(d.anchor.x-d.x)*4*dt; d.vy+=(d.anchor.y-d.y)*4*dt;
        const dm=Math.exp(-2*dt); d.vx*=dm; d.vy*=dm;
        d.x+=d.vx*dt; d.y+=d.vy*dt;
        if(Math.hypot(d.anchor.x-d.x,d.anchor.y-d.y)<16||d.t>3.2) d.entering=false;
      }else{
        d.x=d.anchor.x+Math.cos(d.t*.6)*52; d.y=d.anchor.y+Math.sin(d.t*.85)*40;
        if(fire){ d.spT-=dt; if(d.spT<=0){ d.spT=.16; d.a+=.62; eShot(d.x,d.y,d.a,Math.min(95+G.wave*4,175),MAG); } }
      }
    }
  }
  /* boss */
  if(boss){
    boss.t+=dt; boss.flash=Math.max(0,boss.flash-dt); boss.pulse=Math.max(0,boss.pulse-dt*3);
    if(boss.y<boss.ty) boss.y+=62*dt;
    else{
      boss.x+=Math.sin(boss.t*.7)*34*dt + (ship.alive?(ship.x-boss.x)*.12*dt:0);
      boss.y+=(boss.ty-boss.y)*.5*dt;
      if(fire){
        boss.ringT-=dt;
        if(boss.ringT<=0){ boss.ringT=2.3; boss.pulse=1; sfx.tick();
          const n=20+Math.min(G.wave,8), base=rand(0,TAU), sp=Math.min(115+G.wave*4,205);
          for(let i=0;i<n;i++) eShot(boss.x,boss.y,base+i/n*TAU,sp,CORAL);
        }
        boss.fanT-=dt;
        if(boss.fanT<=0){ boss.fanT=1.5;
          const an=Math.atan2(ship.y-boss.y,ship.x-boss.x), sp=Math.min(150+G.wave*6,240);
          for(let i=-1;i<=1;i++) eShot(boss.x,boss.y,an+i*.22,sp,MAG);
        }
        if(boss.hp<boss.max*.5){ boss.spT-=dt;
          if(boss.spT<=0){ boss.spT=.22; boss.a+=.78; eShot(boss.x,boss.y,boss.a,Math.min(90+G.wave*3,160),MAG); } }
      }
    }
  }
  /* bullets */
  for(let i=bolts.length-1;i>=0;i--){ const b=bolts[i]; b.x+=b.vx*dt; b.y+=b.vy*dt;
    if(b.dead||b.x<-40||b.x>W+40||b.y<-40||b.y>H+40) bolts.splice(i,1); }
  for(let i=ebullets.length-1;i>=0;i--){ const b=ebullets[i]; b.x+=b.vx*dt; b.y+=b.vy*dt;
    if(b.x<-30||b.x>W+30||b.y<-30||b.y>H+30) ebullets.splice(i,1); }
  /* pickups */
  for(let i=pickups.length-1;i>=0;i--){ const p=pickups[i]; p.t+=dt; p.life-=dt; if(p.life<=0)pickups.splice(i,1); }
}
function updateShip(dt){
  if(!ship.alive){
    if(G.lives>0){ G.respawnT-=dt; if(G.respawnT<=0) doRespawn(); }
    return;
  }
  ship.inv=Math.max(0,ship.inv-dt);
  const rot=4.6;
  if(keys['ArrowLeft']||keys['KeyA']) ship.a-=rot*dt;
  if(keys['ArrowRight']||keys['KeyD']) ship.a+=rot*dt;
  const thrust=(keys['ArrowUp']||keys['KeyW']);
  setThrust(thrust);
  if(thrust){
    ship.vx+=Math.cos(ship.a)*560*dt; ship.vy+=Math.sin(ship.a)*560*dt;
    ship.emit+=dt*95;
    while(ship.emit>1){ ship.emit--;
      const ba=ship.a+Math.PI+rand(-.4,.4), sp=rand(80,190);
      addPart({type:'spark',x:ship.x-Math.cos(ship.a)*10,y:ship.y-Math.sin(ship.a)*10,
        vx:Math.cos(ba)*sp+ship.vx*.3,vy:Math.sin(ba)*sp+ship.vy*.3,life:rand(.2,.4),max:.4,c:AMBER}); }
  }else ship.emit=0;
  const dm=Math.exp(-1.3*dt); ship.vx*=dm; ship.vy*=dm;
  const sp=Math.hypot(ship.vx,ship.vy); if(sp>430){ship.vx*=430/sp;ship.vy*=430/sp;}
  ship.x+=ship.vx*dt; ship.y+=ship.vy*dt; wrap(ship,16);
  ship.cool-=dt;
  if(keys['Space']&&ship.cool<=0){
    ship.cool=.14; sfx.shoot();
    const nx=Math.cos(ship.a),ny=Math.sin(ship.a);
    bolts.push({x:ship.x+nx*16,y:ship.y+ny*16,vx:ship.vx*.5+nx*780,vy:ship.vy*.5+ny*780,dead:false});
    addPart({type:'dot',x:ship.x+nx*18,y:ship.y+ny*18,vx:0,vy:0,size:6,life:.08,max:.08,c:[255,255,255]});
  }
}
function updateCollisions(dt){
  /* player bolts vs enemies */
  for(const b of bolts){
    if(b.dead) continue;
    for(const a of asteroids){ if(a.dead)continue;
      if(Math.hypot(b.x-a.x,b.y-a.y)<a.r*.92+4){ b.dead=true; killAsteroid(a,true); break; } }
    if(b.dead)continue;
    for(const d of drones){ if(d.dead)continue;
      if(Math.hypot(b.x-d.x,b.y-d.y)<16){ b.dead=true; d.hp--; d.flash=.08; sparkBurst(b.x,b.y,3,CYAN);
        if(d.hp<=0) killDrone(d,true); break; } }
    if(b.dead)continue;
    if(boss&&Math.hypot(b.x-boss.x,b.y-boss.y)<boss.r){ b.dead=true; boss.hp--; boss.flash=.07; sparkBurst(b.x,b.y,3,CORAL); sfx.tick();
      if(boss.hp<=0) bossDie(); }
  }
  /* enemy bullets vs ship: hit + graze */
  if(ship.alive){
    for(const b of ebullets){
      const dx=b.x-ship.x, dy=b.y-ship.y, d2=dx*dx+dy*dy;
      if(d2<49){ hitPlayer(); break; }                       /* hitbox r7 */
      else if(!b.grazed&&d2<900){ b.grazed=true;             /* graze r30 */
        G.graze++; addScore(5);
        const d=Math.sqrt(d2)||1;
        sparkBurst(ship.x+dx/d*8,ship.y+dy/d*8,2,[255,255,255]);
        sfx.graze();
        if(G.graze%50===0){ addScore(100,ship.x,ship.y-34,'#ffe28a'); floater(ship.x,ship.y-52,'GRAZE BONUS','#ffd94d'); sfx.chime(); }
      }
    }
  }
  /* ship body vs enemies */
  if(ship.alive&&ship.inv<=0){
    for(const a of asteroids){ if(a.dead)continue;
      if(Math.hypot(ship.x-a.x,ship.y-a.y)<a.r*.85+7){ killAsteroid(a,false); hitPlayer(); break; } }
    if(!ship.alive)return;
    for(const d of drones){ if(d.dead)continue;
      if(Math.hypot(ship.x-d.x,ship.y-d.y)<20){ killDrone(d,false); hitPlayer(); break; } }
    if(!ship.alive)return;
    if(boss&&Math.hypot(ship.x-boss.x,ship.y-boss.y)<boss.r+6) hitPlayer();
  }
  /* pickups */
  for(let i=pickups.length-1;i>=0;i--){ const p=pickups[i];
    if(ship.alive&&Math.hypot(p.x-ship.x,p.y-ship.y)<28){
      pickups.splice(i,1); G.bombs=Math.min(G.bombs+1,5); sfx.pickup();
      floater(p.x,p.y,'+ BOMB','#ffd9a8'); sparkBurst(p.x,p.y,8,AMBER); }
  }
}
function updatePlay(dt){
  G.timeNow=T;
  updateShip(dt);
  updateWorld(dt);
  updateCollisions(dt);
  /* cull dead */
  asteroids=asteroids.filter(a=>!a.dead);
  drones=drones.filter(d=>!d.dead);
  if(!ship.alive&&G.lives<=0){ G.overT-=dt; if(G.overT<=0){ gameOver(); return; } }
  /* wave clear */
  if(asteroids.length===0&&drones.length===0&&!boss){
    G.clearT+=dt; if(G.clearT>1.4){ G.clearT=0; startWave(G.wave+1); }
  } else G.clearT=0;
}
function ensureAttract(){
  if(asteroids.length<7&&Math.random()<dtGlobal*2) spawnAsteroid(Math.random()<.5?3:2,1);
}

function render(dt){
  ctx.fillStyle=bgGrad; ctx.fillRect(0,0,W,H);
  /* nebula drift */
  if(neb){ const ox=Math.sin(T*.05)*26- W*.075, oy=Math.cos(T*.04)*22-H*.075;
    ctx.drawImage(neb,ox,oy,W*1.15,H*1.15); }
  /* stars */

    const tw=.55+.45*Math.sin(T*s.tw+s.ph);
    ctx.globalAlpha=s.a*tw;
    ctx.fillStyle='#cfeeff';
    ctx.fillRect(s.x,s.y,s.r,s.r);
  }

  ctx.save(); ctx.translate(sx,sy);

  /* pickups */
  for(const p of pickups){

    ctx.strokeStyle='#ffb057'; ctx.lineWidth=2;
    ctx.beginPath(); ctx.moveTo(0,-11);ctx.lineTo(9,0);ctx.lineTo(0,11);ctx.lineTo(-9,0);ctx.closePath(); ctx.stroke();
    ctx.shadowBlur=0; ctx.fillStyle='#ffe3bd';
    ctx.font='700 11px Consolas,monospace'; ctx.textAlign='center'; ctx.textBaseline='middle';
    ctx.rotate(-rot); ctx.fillText('B',0,1);
    ctx.restore();
  }

  /* asteroids */
  for(const a of asteroids){
    const col=[0,'rgba(251,191,36,.9)','rgba(110,231,183,.9)','rgba(74,222,188,.9)'][a.size];
    ctx.save(); ctx.translate(a.x,a.y); ctx.rotate(a.a);
    ctx.shadowColor=col; ctx.shadowBlur=10;
    ctx.strokeStyle=col; ctx.lineWidth=a.size===3?2.5:2;

    ctx.beginPath();

    ctx.closePath(); ctx.fill(); ctx.stroke();

      const al=.45+.45*Math.sin(T*6+a.ph);
      ctx.shadowColor='rgba(255,86,105,.9)'; ctx.shadowBlur=12;
      ctx.fillStyle='rgba(255,86,105,'+al.toFixed(2)+')';
      ctx.beginPath(); ctx.arc(0,0,4,0,TAU); ctx.fill();
    }
    ctx.restore();
  }

  /* drones */
  for(const d of drones){
    ctx.save(); ctx.translate(d.x,d.y);
    if(d.type==='weevil'){
      const ang=Math.atan2(d.vy,d.vx); ctx.rotate(ang);
      ctx.shadowColor='rgba(255,176,87,.9)'; ctx.shadowBlur=10;
      ctx.strokeStyle=d.flash>0?'#ffffff':'#ffb057'; ctx.lineWidth=2;
      const flap=Math.sin(T*26+d.phase)*4;
      ctx.beginPath(); ctx.moveTo(10,0); ctx.lineTo(-8,7); ctx.lineTo(-4,0); ctx.lineTo(-8,-7); ctx.closePath(); ctx.stroke();
      ctx.beginPath(); ctx.moveTo(-3,4); ctx.lineTo(-13,9+flap); ctx.moveTo(-3,-4); ctx.lineTo(-13,-9-flap); ctx.stroke();
      ctx.shadowBlur=8; ctx.fillStyle='rgba(255,86,105,'+(.5+.5*Math.sin(T*8)).toFixed(2)+')';
      ctx.beginPath(); ctx.arc(4,0,2.6,0,TAU); ctx.fill();
    }else{ /* moth */
      ctx.shadowColor='rgba(196,132,252,.9)'; ctx.shadowBlur=12;
      ctx.strokeStyle=d.flash>0?'#ffffff':'#c484fc'; ctx.lineWidth=2;
      for(let i=0;i<3;i++){
        ctx.save(); ctx.rotate(T*.9+i*TAU/3);
        ctx.fillStyle='rgba(196,132,252,.22)';
        ctx.beginPath(); ctx.ellipse(14,0,13,6.5,0,0,TAU); ctx.fill(); ctx.stroke();
        ctx.restore();
      }
      /* emitter arm telegraph */
      ctx.strokeStyle='rgba(232,121,249,.8)'; ctx.lineWidth=2;
      ctx.beginPath(); ctx.moveTo(0,0); ctx.lineTo(Math.cos(d.a)*22,Math.sin(d.a)*22); ctx.stroke();
      ctx.fillStyle='#f6e4ff'; ctx.beginPath(); ctx.arc(Math.cos(d.a)*22,Math.sin(d.a)*22,2.4,0,TAU); ctx.fill();
      ctx.shadowBlur=14; ctx.fillStyle='#e9c8ff';
      ctx.beginPath(); ctx.arc(0,0,5.5,0,TAU); ctx.fill();
    }
    ctx.restore();
  }

  /* boss */
  if(boss){
    const sc=1+boss.pulse*.14;
    ctx.save(); ctx.translate(boss.x,boss.y); ctx.scale(sc,sc);
    ctx.shadowColor='rgba(255,90,105,.8)'; ctx.shadowBlur=22;
    ctx.strokeStyle=boss.flash>0?'#ffffff':'#ff8b6a'; ctx.lineWidth=2.5;
    ctx.fillStyle='rgba(255,90,105,.06)';
    ctx.beginPath();
    for(let i=0;i<24;i++){ const a=i/24*TAU+T*.3, r=(i%2?58:42);
      const px=Math.cos(a)*r, py=Math.sin(a)*r; if(i===0)ctx.moveTo(px,py); else ctx.lineTo(px,py); }
    ctx.closePath(); ctx.fill(); ctx.stroke();
    ctx.rotate(-T*.6);
    ctx.strokeStyle='#ffb057'; ctx.lineWidth=2;
    ctx.beginPath();
    for(let i=0;i<6;i++){ const a=i/6*TAU, px=Math.cos(a)*26, py=Math.sin(a)*26; if(i===0)ctx.moveTo(px,py); else ctx.lineTo(px,py); }
    ctx.closePath(); ctx.stroke();
    ctx.rotate(T*.6);
    /* eye tracks player */
    const ea=ship.alive?Math.atan2(ship.y-boss.y,ship.x-boss.x):T;
    ctx.shadowBlur=16; ctx.fillStyle='#1a0b12';
    ctx.beginPath(); ctx.arc(Math.cos(ea)*7,Math.sin(ea)*7,10,0,TAU); ctx.fill();
    ctx.strokeStyle='#ff5a69'; ctx.stroke();
    ctx.fillStyle='#ffd9a8';
    ctx.beginPath(); ctx.arc(Math.cos(ea)*10,Math.sin(ea)*10,3.4,0,TAU); ctx.fill();
    ctx.restore();
  }

  /* player bolts */
  ctx.shadowColor='rgba(110,243,255,.9)'; ctx.shadowBlur=8;
  ctx.strokeStyle='rgba(240,255,255,.95)'; ctx.lineWidth=3; ctx.lineCap='round';
  for(const b of bolts){
    const l=Math.hypot(b.vx,b.vy)||1;
    ctx.beginPath(); ctx.moveTo(b.x-b.vx/l*13,b.y-b.vy/l*13); ctx.lineTo(b.x+b.vx/l*4,b.y+b.vy/l*4); ctx.stroke();
  }
  ctx.shadowBlur=0;

  /* enemy bullets (sprite glow + core) */
  for(const b of ebullets){
    const g=glow(b.c[0],b.c[1],b.c[2]), sz=20;
    ctx.drawImage(g,b.x-sz/2-b.vx*.015,b.y-sz/2-b.vy*.015,sz,sz);
    ctx.fillStyle='rgba(255,255,255,.92)';
    ctx.beginPath(); ctx.arc(b.x,b.y,2.6,0,TAU); ctx.fill();
  }

  /* ship */
  if(ship.alive&&(G.state==='playing'||G.state==='paused')){
    const blink=ship.inv>0&&Math.floor(T*14)%2===0;
    ctx.save(); ctx.translate(ship.x,ship.y);
    if(blink) ctx.globalAlpha=.35;
    ctx.rotate(ship.a);
    ctx.shadowColor='rgba(110,243,255,.95)'; ctx.shadowBlur=13;
    ctx.strokeStyle='#8ff6ff'; ctx.lineWidth=2;
    ctx.fillStyle='rgba(143,246,255,.12)';
    ctx.beginPath(); ctx.moveTo(15,0); ctx.lineTo(-10,9); ctx.lineTo(-5,0); ctx.lineTo(-10,-9); ctx.closePath();
    ctx.fill(); ctx.stroke();
    ctx.restore();
    if(ship.inv>0){ /* shield ring */
      ctx.save(); ctx.translate(ship.x,ship.y); ctx.rotate(T*2);
      ctx.strokeStyle='rgba(110,243,255,.5)'; ctx.lineWidth=1.5; ctx.setLineDash([6,7]);
      ctx.beginPath(); ctx.arc(0,0,22,0,TAU); ctx.stroke(); ctx.restore();
    }
    /* hitbox dot */
    ctx.fillStyle='#ffffff'; ctx.shadowColor='#fff'; ctx.shadowBlur=6;
    ctx.beginPath(); ctx.arc(ship.x,ship.y,2.2,0,TAU); ctx.fill(); ctx.shadowBlur=0;
  }

  /* particles */
  for(const p of particles){
    const f=p.life/p.max, c='rgba('+p.c[0]+','+p.c[1]+','+p.c[2]+',';
    if(p.type==='spark'){
      ctx.strokeStyle=c+f.toFixed(3)+')'; ctx.lineWidth=2;
      ctx.beginPath(); ctx.moveTo(p.x,p.y); ctx.lineTo(p.x-p.vx*.045,p.y-p.vy*.045); ctx.stroke();
    }else if(p.type==='streak'){
      ctx.strokeStyle=c+(f*.8).toFixed(3)+')'; ctx.lineWidth=1.5;
      ctx.beginPath(); ctx.moveTo(p.x,p.y); ctx.lineTo(p.x-p.vx*.09,p.y-p.vy*.09); ctx.stroke();
    }else if(p.type==='dot'){
      ctx.fillStyle=c+f.toFixed(3)+')';
      ctx.beginPath(); ctx.arc(p.x,p.y,Math.max(.5,p.size*f),0,TAU); ctx.fill();
    }else{ /* ring */
      const r=p.r0+(p.r1-p.r0)*(1-f);
      ctx.strokeStyle=c+(f*.9).toFixed(3)+')'; ctx.lineWidth=1.5+3*f;
      ctx.beginPath(); ctx.arc(p.x,p.y,r,0,TAU); ctx.stroke();
    }
  }

  /* floaters */
  ctx.textAlign='center'; ctx.textBaseline='middle';
  for(const f of floaters){
    const a=f.life/f.max;
    ctx.font='700 15px Impact,"Arial Black",sans-serif';
    ctx.globalAlpha=a; ctx.fillStyle=f.c;
    ctx.shadowColor=f.c; ctx.shadowBlur=8;
    ctx.fillText(f.text,f.x,f.y);
    ctx.shadowBlur=0; ctx.globalAlpha=1;
  }
  ctx.restore();

  /* bomb white flash */
  if(G.whiteFlash>0){ ctx.fillStyle='rgba(255,255,255,'+(G.whiteFlash*.32).toFixed(3)+')'; ctx.fillRect(0,0,W,H); }
}

/* ============================== HUD SYNC ============================== */
function syncHUD(){
  if(ui.score!==G.score){ ui.score=G.score; elScore.textContent=pad(G.score,6); }
  if(ui.hi!==G.hi){ ui.hi=G.hi; elHi.textContent=pad(G.hi,6); $('menubi').textContent=pad(G.hi,6); }
  if(ui.wave!==G.wave){ ui.wave=G.wave; elWave.textContent=pad(G.wave,2); }
  if(ui.graze!==G.graze){ ui.graze=G.graze; elGraze.textContent=pad(G.graze,3); }
  if(ui.lives!==G.lives){ ui.lives=G.lives;
    elLives.textContent=G.lives>0?'▲'.repeat(G.lives):'—';
    elLives.classList.toggle('empty',G.lives<=0); }
  if(ui.bombs!==G.bombs){ ui.bombs=G.bombs;
    elBombs.textContent=G.bombs>0?'◆ '.repeat(G.bombs).trim():'·';
    elBombs.classList.toggle('empty',G.bombs<=0); }
  const bw=boss?Math.max(0,boss.hp/boss.max*100):-1;
  if(ui.boss!==bw){ ui.boss=bw;
    elBossWrap.classList.toggle('on',boss!==null);
    if(boss) elBossFill.style.width=bw.toFixed(1)+'%'; }
}

/* ============================== MAIN LOOP ============================== */
let last=performance.now(), dtGlobal=.016;
function frame(now){
  requestAnimationFrame(frame);
  let dt=(now-last)/1000; last=now;
  if(dt>0.05)dt=0.05; if(dt<=0)dt=.008;
  dtGlobal=dt; T+=dt;
  updateStars(dt);
  if(G.state==='playing') updatePlay(dt);
  else{ if(G.state==='menu') ensureAttract(); updateWorld(dt); }
  updateFx(dt);
  render(dt);
  syncHUD();
}

requestAnimationFrame(frame);
