Flow field:
```js
let flow=[]; // dist per cell, Infinity if unreachable
function computeFlow(){
  flow = Array.from({length:W},()=>Array(H).fill(Infinity));
  const q=[[vaultX,vaultY]]; flow[vx][vy]=0;
  BFS over cells where passable(x,y): grid!=wall && !crateAt(x,y); vault is target (dist 0) — enemies walk INTO vault cell.
}
function crateAt(x,y){return crates.find(c=>c.x===x&&c.y===y);} // 5 crates, fine; or maintain a map. Use Set of keys for speed: crateSet.

Enemy update:
```js
e = {x,y (cell coords float), type stats..., hp, cellX,cellY (current logical cell), tx,ty target center, state:'move'|'zap', ...}
```
Simplify: enemy always has a current cell (cx,cy) = last settled cell; target = center of next hop. function flowHints(n){for(let i=0;i<n;i++){

function step(dt){
 tGlobal+=dt;
 if(!waveActive){
  betweenT-=dt;
  if(betweenT<=0)startWave();
 }else{
  spawnT+=dt;
  while(spawnQ.length&&spawnQ[0].t<=spawnT)spawnEnemy(spawnQ.shift().type);
  if(spawnQ.length===0&&enemies.length===0){waveActive=false;waveCleared=true;
   const bonus=25+(waveIdx+1)*5;coins+=bonus;coinsEarned+=bonus;
   texts.push({x:COLS/2,y:ROWS/2-.5,txt:'+'+bonus,col:'#ffd166',t:0,life:1.3});
   if(waveIdx===WAVES.length-1){win();return;}
   banner={text:'WAVE CLEAR',sub:'+'+bonus+' scrap',t:0,dur:1.8};betweenT=5;}
 }
 if(keeper.bumpT>0)keeper.bumpT-=dt;
 if(!keeper.moving&&state==='play'){
  for(let i=keyOrder.length-1;i>=0;i--){const k=keyOrder[i];if(keys.has(k)){tryMove(DIRS[k][0],DIRS[k][1]);break;}}}
 if(keeper.moving){keeper.t+=dt/MOVE_DUR;if(keeper.t>=1){keeper.t=1;keeper.x=keeper.tx;keeper.y=keeper.ty;keeper.fx=keeper.tx;keeper.fy=keeper.ty;keeper.moving=false;}}
 for(const e of enemies){if(e.dead)continue;
  const s=ETYPES[e.type];e.flash=Math.max(0,e.flash-dt*6);
  if(e.type==='p'){
   if(e.state==='move'&&isFinite(flow[e.cx][e.cy])&&flow[e.cx][e.cy]<=s.zap){e.state='zap';e.zapT=1.2;}
   if(e.state==='zap'){
    e.x+=(e.cx+.5-e.x)*Math.min(1,dt*6);e.y+=(e.cy+.5-e.y)*Math.min(1,dt*6);
    e.zapT-=dt;if(e.zapT<=0){beams.push({x:e.x,y:e.y,t:0,life:.3});damageVault(s.dz);sfx('zap');e.zapT=s.zp;}
    continue;}}
  if(!e.hop){
   if(grid[e.cy][e.cx]===2){hitVault(e);continue;}
   const c=chewTarget(e.cx,e.cy);
   if(c){e.chewAcc+=dt;c.hp-=s.chew*dt;if(e.chewAcc>.15){e.chewAcc=0;
     parts.push({type:'chip',x:c.x+.5,y:c.y+.5,vx:rand(-2,2),vy:rand(-3,.5),t:0,life:.6,size:2.5,rot:rand(0,7),vr:rand(-8,8),col:'#a06024'});
     if(Math.random()<.3)AU.tone(rand(140,190),90,.05,'square',.15);}
    if(c.hp<=0){breakCrate(c);e.hop=nextHop(e.cx,e.cy);}}
   continue;}
  if(crateSet.has(ck(e.hop.x,e.hop.y)))e.hop=nextHop(e.cx,e.cy);
  if(!e.hop)continue;
  const tx=e.hop.x+.5,ty=e.hop.y+.5,dx=tx-e.x,dy=ty-e.y,d=Math.hypot(dx,dy),mv=s.spd*dt;
  if(d>0.001){e.fx=dx/d;e.fy=dy/d;}
  if(d<=mv){e.x=tx;e.y=ty;e.cx=e.hop.x;e.cy=e.hop.y;
   if(grid[e.cy][e.cx]===2){hitVault(e);continue;}
   e.hop=(crateSet.has(ck(e.cx,e.cy)))?null:nextHop(e.cx,e.cy);
   if(e.type==='p'&&e.state!=='zap'&&isFinite(flow[e.cx][e.cy])&&flow[e.cx][e.cy]<=s.zap){e.state='zap';e.zapT=1.2;}
  }else{e.x+=dx/d*mv;e.y+=dy/d*mv;}}
 enemies=enemies.filter(e=>!e.dead);
 for(const c of crates){c.cd-=dt;c.flash=Math.max(0,c.flash-dt*4);if(c.cd>0)continue;
  const st=turretStats(c);let target=null,sc=1e9;
  for(const e of enemies){if(e.dead)continue;const dx=e.x-(c.x+.5),dy=e.y-(c.y+.5),d=Math.hypot(dx,dy);
   if(d>st.rng)continue;const fd=isFinite(flow[e.cx][e.cy])?flow[e.cx][e.cy]:999;
   const score=fd*10+d;if(score<sc){sc=score;target=e;}}
  if(target){c.cd=st.cd;c.flash=1;
   const dx=target.x-(c.x+.5),dy=target.y-(c.y+.5),d=Math.hypot(dx,dy)||1;
   projs.push({x:c.x+.5,y:c.y+.5,vx:dx/d*11,vy:dy/d*11,target,dmg:st.dmg,t:1.4});sfx('shoot');}}
 for(const p of projs){p.t-=dt;if(p.t<=0){p.dead=true;continue;}
  const tg=p.target;
  if(tg&&!tg.dead&&tg.hp>0){const dx=tg.x-p.x,dy=tg.y-p.y,d=Math.hypot(dx,dy)||1;
   p.vx=lerp(p.vx,dx/d*11,.3);p.vy=lerp(p.vy,dy/d*11,.3);
   if(d<.28){damageEnemy(tg,p.dmg);p.dead=true;continue;}}
  p.x+=p.vx*dt;p.y+=p.vy*dt;}
 projs=projs.filter(p=>!p.dead);
 updateParts(dt);
 for(const b of beams)b.t+=dt;beams=beams.filter(b=>b.t<b.life);
 for(const t of texts){t.t+=dt;t.y-=dt*1.1;}texts=texts.filter(t=>t.t<t.life);
 for(const d of decals)d.a=Math.max(.25,d.a-dt*.06);
 if(banner)banner.t+=dt;
 shake=Math.max(0,shake-dt*18);vaultHitFx=Math.max(0,vaultHitFx-dt*2.5);
 tipT+=dt;if(tipT>7){tipT=0;tipI=(tipI+1)%TIPS.length;tipTxt.style.opacity=0;setTimeout(()=>{tipTxt.textContent=TIPS[tipI];tipTxt.style.opacity=1;},300);}
}

  if(c){const d=isFinite(flow[c.x][c.y])?flow[c.x][c.y]:999;if(d<bd){bd=d;best=c;}}});return best;}

WASD/arrows move you and push turret-crates — crates block the swarm's path (enemies re-route live, shown by cyan flow sparks) and auto-fire at anything in range.
