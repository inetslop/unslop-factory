/* ============================== RAYCAST (aim guide) ============================== */
function castRay(ox,oy,dx,dy,maxD){
  let bt=1e9,bnx=0,bny=0,hit=false;
  for(const s of hole.walls){
    const sx=s[2]-s[0],sy=s[3]-s[1];
    const det=sx*dy-sy*dx; if(Math.abs(det)<1e-9)continue;
    const ex=s[0]-ox,ey=s[1]-oy;
    const t=(ex*(-sy)+sx*ey)/det, u=(dx*ey-dy*ex)/det;
    if(t>0.5&&u>=0&&u<=1&&t<bt){const hx=ox+dx*t,hy=oy+dy*t;
      let nx=ox-hx,ny=oy-hy;const L=Math.hypot(nx,ny)||1;bnx=nx/L;bny=ny/L;bt=t;hit=true;}
  }
  for(const bp of hole.bumpers){
    const cx=ox-bp.x,cy=oy-bp.y,b2=2*(cx*dx+cy*dy),c2=cx*cx+cy*cy-bp.r*bp.r;
    const disc=b2*b2-4*c2; if(disc<0)continue;
    let t=(-b2-Math.sqrt(disc))/2; if(t<0)t=(-b2+Math.sqrt(disc))/2;
    if(t>0.5&&t<bt){const hx=ox+dx*t,hy=oy+dy*t;
      let nx=hx-bp.x,ny=hy-bp.y;const L=Math.hypot(nx,ny)||1;bnx=nx/L;bny=ny/L;bt=t;hit=true;}
  }
  if(hit&&bt>maxD){hit=false;bt=maxD;}
  return{hit:hit,x:ox+dx*bt,y:oy+dy*bt,nx:bnx,ny:bny};
}

let last=0;
function update(dt){
  animT+=dt;
  if(state==='pause')return;
  if(pend){pend.t-=dt;if(pend.t<=0){const f=pend.fn;pend=null;f();}}
  updateBumpers();
  if(state==='play'||state==='menu'){
    if(state==='play')holeT+=dt;
    const sub=2, h=dt/sub;
    for(let i=0;i<sub;i++){stepBall(P,h);stepBall(C,h);ballsCollide(P,C);}
    if(P.aiOn)aiStep(P,dt,1.35);
    if(C.aiOn)aiStep(C,dt,1.0);
    if(charging&&canShootP()){
      chargeT+=dt; const ph=(chargeT%1.2)/.6; Ppower=ph<1?ph:2-ph;
    }else if(Ppower)Ppower=0;
  }
  for(const b of [P,C])if(b.sunk&&b.sinkT<1)b.sinkT=Math.min(1,b.sinkT+dt/.45);
  // keyboard aim
  if(state==='play'&&!P.sunk){
    const rot=((keys['ArrowLeft']||keys['a'])?1:0)-((keys['ArrowRight']||keys['d'])?1:0);
    if(rot)aimAngle+=rot*2.2*dt;
  }
  updateParts(dt);
  shakeAmp*=Math.exp(-7*dt);
  hudUpdate();
}
function loop(ts){
  const dt=clamp((ts-last)/1000,0,.033); last=ts;
  update(dt); render();
  requestAnimationFrame(loop);
}

buildPips(); resize(); refreshBest();
setupHole(0); state='menu';
requestAnimationFrame(function(ts){last=ts;loop(ts);});
