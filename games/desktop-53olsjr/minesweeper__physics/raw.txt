let lastT=performance.now();
function frame(now){ requestAnimationFrame(frame);
  let dt=Math.min(.05,(now-lastT)/1000); lastT=now;
  if(state!=='pause')update(dt,now/1000);
  render(now/1000); updateHUD(); }

function update(dt,t){
  // ambient dust
  for(const d of dust){d.x+=d.vx*dt;d.y+=d.vy*dt;if(d.y>H+4){d.y=-4;d.x=R(0,W);}if(d.x<-4)d.x=W+4;if(d.x>W+4)d.x=-4;}
  shake*=Math.exp(-6*dt); if(shake<.05)shake=0;
  flash=Math.max(0,flash-dt*2.1); recoil=Math.max(0,recoil-dt*5);
  hintT=Math.max(0,hintT-dt);
  // pending reveals
  for(let k=pending.length-1;k>=0;k--){ if(nowMs()>=pending[k].t){const p=pending.splice(k,1)[0]; shatter(p.i,p.cause);} }
  if(boomPending&&nowMs()>=boomPending.t){const b=boomPending;boomPending=null; if(state==='play')detonate(b.i);}
  // per-cell decay
  for(const c of cells){c.jolt*=Math.exp(-5*dt);c.deny=Math.max(0,c.deny-dt*2.4);c.flash=Math.max(0,c.flash-dt*3);}
  // ball / demo
  if(state==='menu'){ if(!ball.active){demoT+=dt; if(demoT>1.3){demoT=0; launch(R(.5,1),R(-Math.PI*.86,-Math.PI*.14));} }
    stepBall(dt);
    if(safeRemaining<=0&&!ball.active)makeDemo(); }
  else if(state==='play'||state==='boom'){ stepBall(dt);
    if(charging){ const was=chargeT; chargeT=Math.min(CHARGE_T,chargeT+dt);
      if(!chargedFull&&chargeT>=CHARGE_T){chargedFull=true;AU.tick();}
      if(AU.chargeOsc)AU.chargeOsc.frequency.value=90+(chargeT/CHARGE_T)*260; } }
  if(ball.active===false&&ball.alpha>0)ball.alpha=Math.max(0,ball.alpha-dt*4);
  // particles / rings / floaters
  for(let k=particles.length-1;k>=0;k--){const p=particles[k];p.l-=dt;
    if(p.l<=0){particles.splice(k,1);continue;}
    if(p.type==='puff'){p.r+=p.vr*dt;p.x+=p.vx*dt;p.y+=p.vy*dt;}
    else{const f=Math.exp(-2.2*dt);p.vx*=f;p.vy*=f;p.x+=p.vx*dt;p.y+=p.vy*dt;p.rot+=p.vr*dt;}}
  if(particles.length>520)particles.splice(0,particles.length-520);
  for(let k=rings.length-1;k>=0;k--){const r=rings[k];r.l-=dt;if(r.l<=0){rings.splice(k,1);continue;}r.r+=r.vr*dt;r.vr*=Math.exp(-1.4*dt);}
  for(let k=floaters.length-1;k>=0;k--){const f=floaters[k];f.l-=dt*.9;if(f.l<=0)floaters.splice(k,1);else f.y-=34*dt;}
  // timer & end screens
  if(state==='play'&&started)elapsed+=dt;
  if((state==='boom'||state==='win')&&!endShown){ endTimer-=dt;
    if(state==='win'&&winBursts<9&&endTimer>0.15){ winBursts++;
      const rc=cells.filter(c=>c.state==='revealed'); if(rc.length){const c=pick(rc);
        const cx=L.gx+c.col*L.cell+L.cell/2,cy=L.gy+c.row*L.cell+L.cell/2;
        spawnShards(cx,cy,{hh:R(0,360),ss:70,ll:55},18,true); addRing(cx,cy,4,L.cell*4,.5,pick(NC),3);} }
    if(endTimer<=0){endShown=true;showEnd();} }
  // hover
  hoverIdx=(state==='play')?cellAt(mouse.x,mouse.y):-1;
}
