function collideCircleRect(b,r){
 const cx=clamp(b.x,r.x,r.x+r.w), cy=clamp(b.y,r.y,r.y+r.h);
 let dx=b.x-cx, dy=b.y-cy; let d2=dx*dx+dy*dy;
 if(d2>b.r*b.r) return;
 let nx,ny,pen;
 if(d2>1e-6){ const d=Math.sqrt(d2); nx=dx/d; ny=dy/d; pen=b.r-d }
 else { const l=b.x-r.x, rr2=r.x+r.w-b.x, t=b.y-r.y, bb=r.y+r.h-b.y, m=Math.min(l,rr2,t,bb);
  if(m===t){nx=0;ny=-1;pen=b.r+t} else if(m===bb){nx=0;ny=1;pen=b.r+bb} else if(m===l){nx=-1;ny=0;pen=b.r+l} else {nx=1;ny=0;pen=b.r+rr2} }
 b.x+=nx*pen; b.y+=ny*pen;
 const vn=b.vx*nx+b.vy*ny;
 if(vn<0){
  const e=vn<-70?b.rest:0;
  const j=-(1+e)*vn;
  b.vx+=nx*j; b.vy+=ny*j;
  // friction
  const tx=-ny,ty=nx, vt=b.vx*tx+b.vy*ty;
  const f=vt*.12;
  b.vx-=tx*f; b.vy-=ty*f;
  if(ny<-0.6){ b.vx*=.985; if(Math.hypot(b.vx,b.vy)<26){b.vx*=.94;b.vy*=.94} }
 }
}

/* ================= physics ================= */
function collideCircles(a,b){
  const dx=b.x-a.x,dy=b.y-a.y,rs=a.r+b.r;let d2=dx*dx+dy*dy;
  if(d2>=rs*rs)return;
  let d=Math.sqrt(d2);if(d<1e-4)d=1e-4;
  const nx=dx/d,ny=dy/d,pen=rs-d,ia=1/a.m,ib=1/b.m,isum=ia+ib;
  a.x-=nx*pen*(ia/isum);a.y-=ny*pen*(ia/isum);
  b.x+=nx*pen*(ib/isum);b.y+=ny*pen*(ib/isum);
  const rvx=b.vx-a.vx,rvy=b.vy-a.vy,vn=rvx*nx+rvy*ny;
  if(vn<0){
    const e=vn<-60?Math.max(a.rest,b.rest)*.9:0,j=-(1+e)*vn/isum;
    a.vx-=nx*j*ia;a.vy-=ny*j*ia;b.vx+=nx*j*ib;b.vy+=ny*j*ib;
    const tx=-ny,ty=nx,vt=rvx*tx+rvy*ty,f=vt*.08/isum;
    a.vx+=tx*f*ia;a.vy+=ty*f*ia;b.vx-=tx*f*ib;b.vy-=ty*f*ib;
    if(vn<-130)impactFX(a,b,-vn);
  }
}
function collideCircleRect(b,r){
  const cx=clamp(b.x,r.x,r.x+r.w),cy=clamp(b.y,r.y,r.y+r.h);
  let dx=b.x-cx,dy=b.y-cy;const d2=dx*dx+dy*dy;
  if(d2>b.r*b.r)return;
  let nx,ny,pen;
  if(d2>1e-6){const d=Math.sqrt(d2);nx=dx/d;ny=dy/d;pen=b.r-d}
  else{const l=b.x-r.x,r2=r.x+r.w-b.x,t=b.y-r.y,bb=r.y+r.h-b.y,m=Math.min(l,r2,t,bb);
    if(m===t){nx=0;ny=-1;pen=b.r+t}else if(m===bb){nx=0;ny=1;pen=b.r+bb}
    else if(m===l){nx=-1;ny=0;pen=b.r+l}else{nx=1;ny=0;pen=b.r+r2}}
  b.x+=nx*pen;b.y+=ny*pen;
  const vn=b.vx*nx+b.vy*ny;
  if(vn<0){
    const e=vn<-70?b.rest:0,j=-(1+e)*vn;
    b.vx+=nx*j;b.vy+=ny*j;
    const tx=-ny,ty=nx,vt=b.vx*tx+b.vy*ty,f=vt*.12;
    b.vx-=tx*f;b.vy-=ty*f;
    if(ny<-0.6){b.vx*=.985;if(Math.hypot(b.vx,b.vy)<26){b.vx*=.94;b.vy*=.94}}
  }
}
function walls(b){

  else if(b.x>Wl-WALL-b.r){b.x=Wl-WALL-b.r;if(b.vx>0)b.vx*=-b.rest}
}
function impactFX(a,b,sp){
  let other=null;
  if(a.type==='ball'&&b.type!=='ball')other=b;
  else if(b.type==='ball'&&a.type!=='ball')other=a;
  if(other&&sp>(other.type==='iron'?860:560)){destroyBarrel(other,sp);return}
  if(sp>140){
    const mx=(a.x+b.x)/2,my=(a.y+b.y)/2;
    for(let i=0;i<Math.min(5,1+(sp/300|0));i++)parts.push({kind:'chip',x:mx,y:my,vx:rnd(-160,160),vy:rnd(-220,-40),t:0,dur:rnd(.4,.7),s:rnd(3,6),rot:rnd(TAU),vr:rnd(-9,9),c:i%2?'#8a5a24':'#b57a3c'});
    sWood(sp);if(sp>700)shake=Math.min(shake+2,12);
  }
}

function stepPhysics(dt){
  for(const b of dyn){if(!b.alive)continue;
    b.vy+=G*dt;const dr=Math.exp(-b.drag*dt);b.vx*=dr;b.vy*=dr;
    b.x+=b.vx*dt;b.y+=b.vy*dt;b.age=(b.age||0)+dt;
    if(b.type!=='ball')b.ang+=(b.vx/b.r)*dt*.8;
  }
  for(let it=0;it<2;it++){
    for(let i=0;i<dyn.length;i++){const a=dyn[i];if(!a.alive)continue;
      for(let j=i+1;j<dyn.length;j++){const b=dyn[j];if(b.alive)collideCircles(a,b)}
      for(const r of statics)collideCircleRect(a,r);
      walls(a);
    }
  }
  for(const b of dyn){if(!b.alive)continue;
    if(b.y-b.r>726){if(b.type==='ball'){b.alive=false;ballCount--}else destroyBarrel(b,500+Math.hypot(b.vx,b.vy))}
  }
}

function updateFx(dt){
  time+=dt;
  shake=Math.max(0,shake-dt*30);shX=rnd(-1,1)*shake;shY=rnd(-1,1)*shake*.7;
  recoil*=Math.exp(-9*dt);flashT=Math.max(0,flashT-dt);
  for(const f of fogNear){f.x+=f.v*dt;if(f.x-f.w/2>CW+40)f.x=-f.w/2-20}
  for(const f of fogFar){f.x+=f.v*dt;if(f.x-f.w/2>CW+40)f.x=-f.w/2-20}
  for(const p of parts){p.t+=dt;
    if(p.kind==='chip'){p.vy+=950*dt;p.x+=p.vx*dt;p.y+=p.vy*dt;p.rot+=p.vr*dt}
    else{p.x+=(p.vx||0)*dt;p.y+=(p.vy||-14)*dt}}
  parts=parts.filter(p=>p.t<p.dur);if(parts.length>320)parts.splice(0,parts.length-320);
  for(const q of pops)q.t+=dt;
  pops=pops.filter(q=>q.t<q.dur);
  if(banner){banner.t+=dt;if(banner.t>banner.dur)banner=null}
  displayed=lerp(displayed,score,Math.min(1,dt*9));if(Math.abs(displayed-score)<1)displayed=score;

  for(const b of dyn){if(b.type==='ball'&&b.alive){
    const sp=Math.hypot(b.vx,b.vy);
    if(b.age>6||(b.age>1.2&&sp<40)||b.y<-360){b.alive=false;ballCount--;parts.push({kind:'smoke',x:b.x,y:b.y,vx:0,vy:-20,t:0,dur:.5,s:9})}
  }}
  if(dyn.some(b=>!b.alive))dyn=dyn.filter(b=>b.alive);

  if(state==='play'){
    comboTimer=Math.max(0,comboTimer-dt);if(comboTimer===0)combo=1;
    if(!transitioning){
      waveTime+=dt;
      if(barrelsLeft<=0)waveClear();
      else if(shotsLeft<=0&&ballCount<=0){
        let calm=true;for(const b of dyn){if(b.alive&&Math.hypot(b.vx,b.vy)>45){calm=false;break}}
        calmT=calm?calmT+dt:0;if(calmT>1.2)loseHeart();
      }
    }else{
      transT-=dt;
      if(transT<=0){const a=transAction;transitioning=false;
        if(a==='next')buildWave(wave+1);else if(a==='rebuild')buildWave(wave);else gameOver()}
    }
  }else if(state==='menu'){
    menuT-=dt;
    if(menuT<=0){
      if(barrelsLeft>0&&ballCount<9){
        const s=statics[Math.floor(rnd(statics.length))];
        aim={x:s.x+rnd(24,s.w-24),y:Math.max(120,s.y-rnd(30,150))};fireAt(aim);menuT=rnd(1.6,2.8);
      }else{attractIdx=((attractIdx-2)%7)+2;buildWave(attractIdx);menuT=1.2}
    }
  }
}

function drawAimPreview(){
  if(state!=='play'||transitioning||shotsLeft<=0)return;
  const c=cannonPos(),a=aimAngle(),dx=Math.cos(a),dy=Math.sin(a);
  let px=c.x+dx*58,py=c.y+dy*58,vx=dx*shotPower(Math.hypot(aim.x-c.x,aim.y-c.y)),vy=dy*shotPower(Math.hypot(aim.x-c.x,aim.y-c.y));
  const st=.03;
  ctx.fillStyle='#ffd97a';
  for(let i=0;i<40;i++){
    px+=vx*st;py+=vy*st;vy+=G*st;
    if(px<WALL+17||px>Wl-WALL-17||py>HW-10)break;
    let hit=false;
    for(const r of statics){const qx=clamp(px,r.x,r.x+r.w),qy=clamp(py,r.y,r.y+r.h);
      if((px-qx)*(px-qx)+(py-qy)*(py-qy)<289){hit=true;break}}
    if(i%2===0){ctx.globalAlpha=.55*(1-i/40);ctx.beginPath();ctx.arc(px,py,3.4,0,TAU);ctx.fill()}
    if(hit)break;
  }
  ctx.globalAlpha=1;
  const pu=1+Math.sin(time*6)*.12;
  ctx.strokeStyle='rgba(255,248,220,.75)';ctx.lineWidth=2;
  ctx.beginPath();ctx.arc(aim.x,aim.y,11*pu,0,TAU);ctx.stroke();
  for(let i=0;i<4;i++){const an=i*Math.PI/2;
    ctx.beginPath();ctx.moveTo(aim.x+Math.cos(an)*15*pu,aim.y+Math.sin(an)*15*pu);
    ctx.lineTo(aim.x+Math.cos(an)*20*pu,aim.y+Math.sin(an)*20*pu);ctx.stroke()}
}
