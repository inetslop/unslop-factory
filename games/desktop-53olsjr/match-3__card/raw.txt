function update(dt){
  for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++){
    const t=grid[r]&&grid[r][c]; if(!t)continue;
    if(t.mode==='wait'){t.delay-=dt;if(t.delay<=0)t.mode='fall';continue}
    if(t.mode==='tween'){const tw=t.tw;tw.t+=dt;const p=Math.min(1,tw.t/tw.dur);
      const e=1-Math.pow(1-p,3); t.x=tw.sx+(tw.ex-tw.sx)*e; t.y=tw.sy+(tw.ey-tw.sy)*e;
      if(p>=1){t.mode='idle';t.x=tw.ex;t.y=tw.ey}continue}
    if(t.mode==='fall'){t.vy=Math.min(t.vy+cell*52*dt,cell*30);t.y+=t.vy*dt;
      if(t.y>=t.ty){t.y=t.ty;if(t.vy>cell*16&&Math.random()<.5)dust(t.x,t.ty+cell*.42);t.vy=0;t.mode='idle'}continue}
    if(t.mode==='flip'){const fl=t.fl;fl.t+=dt;const p=Math.min(1,fl.t/fl.dur);t.flipP=p;
      if(!fl.mid&&p>=.5){fl.mid=true;t.kind=fl.nk}
      if(p>=1){t.mode='idle';t.flipP=null}continue}
  }
  for(let i=fxT.length-1;i>=0;i--){const f=fxT[i];f.t+=dt;
    if(f.t>=0&&!f.burst){f.burst=true;burst(f.x,f.y,f.kind)}
    if(f.t>f.dur)fxT.splice(i,1);}
  for(let i=parts.length-1;i>=0;i--){const p=parts[i];p.t+=dt;
    p.vy+=(p.g||0)*dt;p.x+=p.vx*dt;p.y+=p.vy*dt;if(p.t>p.life)parts.splice(i,1);}
  if(parts.length>350)parts.splice(0,parts.length-350);
  for(let i=beams.length-1;i>=0;i--){beams[i].t+=dt;if(beams[i].t>beams[i].dur)beams.splice(i,1)}
  for(let i=rings.length-1;i>=0;i--){rings[i].t+=dt;if(rings[i].t>rings[i].dur)rings.splice(i,1)}
  for(let i=texts.length-1;i>=0;i--){texts[i].t+=dt;if(texts[i].t>texts[i].dur)texts.splice(i,1)}
  if(banner){banner.t+=dt;if(banner.t>banner.dur)banner=null}
  shakeAmp*=Math.exp(-6*dt); if(shakeAmp<.05)shakeAmp=0;

  if(state==='deal'&&allSettled()){state='play';lastInput=performance.now()}
  else if(state==='play'&&busy&&stage==='falling'&&allSettled()){
    const info=findMatches(null,null);
    if(info){combo++;resolvePass(info,null)}else onResolveEnd();
  }
  else if(state==='play'&&reshuffling&&allSettled()){reshuffling=false;busy=false;hintDirty=true}

  const now=performance.now();
  const showHint=state==='play'&&!busy&&(now-lastInput>6500);
  if(showHint){if(hintDirty||!hintPair)hintPair=firstMovePair();} else hintPair=null;

  const total=baseScore+levelScore;
  dispScore+=(total-dispScore)*Math.min(1,dt*8);
  if(Math.abs(total-dispScore)<.6)dispScore=total;
  updateHUDFrame(total);
}

  // gloss sweep over board
  ctx.save(); rr(ox-9,oy-9,bw+18,bh+18,14); ctx.clip();
  const gx=((time*60)%(bw+320))-160;
  let gg=ctx.createLinearGradient(gx-60,0,gx+60,0);
  gg.addColorStop(0,'rgba(255,255,255,0)');gg.addColorStop(.5,'rgba(255,240,200,.05)');gg.addColorStop(1,'rgba(255,255,255,0)');
  ctx.translate(gx,oy+bh/2);ctx.rotate(-.32);ctx.fillStyle=gg;ctx.fillRect(-80,-bh,bw+160,bh*2);
  ctx.restore();

  // floating score texts
  for(const tx of texts){const p=tx.t/tx.dur,a=p<.7?1:1-(p-.7)/.3;
    const s=1+.25*(1-Math.min(1,p*4));
    ctx.save();ctx.globalAlpha=a;ctx.translate(tx.x,tx.y-p*cell*.8);ctx.scale(s,s);
    ctx.font='700 '+Math.round(tx.size)+'px Georgia, serif';ctx.textAlign='center';ctx.textBaseline='middle';
    ctx.lineWidth=4;ctx.strokeStyle='rgba(30,18,2,.8)';ctx.strokeText(tx.txt,0,0);
    ctx.fillStyle=tx.color;ctx.fillText(tx.txt,0,0);ctx.restore();}

  // big banner
  if(banner){const p=banner.t/banner.dur,a=p<.75?1:1-(p-.75)/.25;
    const c1=1.70158,c3=c1+1,sc=1+c3*Math.pow(Math.min(1,p*2.5)-1,3)+c1*Math.pow(Math.min(1,p*2.5)-1,2);
    ctx.save();ctx.globalAlpha=a;ctx.translate(W/2,oy+bh*.32);ctx.scale(sc,sc);
    ctx.font='900 '+Math.round(cell*.6)+'px Georgia, serif';ctx.textAlign='center';ctx.textBaseline='middle';
    ctx.lineWidth=6;ctx.strokeStyle='rgba(40,20,2,.85)';ctx.strokeText(banner.txt,3,3);
    ctx.fillStyle='#ffd76a';ctx.fillText(banner.txt,0,0);ctx.restore();}

  ctx.restore();
}
