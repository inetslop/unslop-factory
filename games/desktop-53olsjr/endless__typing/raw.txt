function updParts(dt){for(var i=parts.length-1;i>=0;i--){var p=parts[i];p.l-=dt;
  if(p.l<=0){parts.splice(i,1);continue;}if(!p.L)p.L=p.l;p.vy+=p.g*dt;p.x+=p.vx*dt;p.y+=p.vy*dt;}}
function drawParts(){
  for(var i=0;i<parts.length;i++){var p=parts[i],a=clamp(p.l/p.L,0,1);
    if(p.t==='m'){ctx.globalCompositeOperation='source-over';
      ctx.fillStyle='rgba(125,104,96,'+(a*.16).toFixed(3)+')';
      ctx.beginPath();ctx.arc(p.x,p.y,p.r*(1.6-a*.6),0,6.283);ctx.fill();}
    else{ctx.globalCompositeOperation='lighter';
      if(p.t==='s'){ctx.strokeStyle='hsla('+p.hue+',100%,'+(45+a*35).toFixed(0)+'%,'+a.toFixed(3)+')';
        ctx.lineWidth=p.r;ctx.beginPath();ctx.moveTo(p.x,p.y);ctx.lineTo(p.x-p.vx*.03,p.y-p.vy*.03);ctx.stroke();}
      else{ctx.fillStyle='hsla('+p.hue+',95%,'+(30+45*a).toFixed(0)+'%,'+a.toFixed(3)+')';
        ctx.beginPath();ctx.arc(p.x,p.y,p.r,0,6.283);ctx.fill();}}}
  ctx.globalCompositeOperation='source-over';
}
function floatText(x,y,txt,col){floats.push({x:x,y:y,txt:txt,col:col,l:.95,L:.95});}
function updFloats(dt){for(var i=floats.length-1;i>=0;i--){floats[i].l-=dt;if(floats[i].l<=0)floats.splice(i,1);}}

}

/* ================= words ================= */
function pickWord(h){
  var R=[[3,4],[3,5],[4,5],[4,6],[4,6],[5,7],[5,7],[5,8],[6,8],[6,9]];
  var r=R[clamp(h-1,0,9)];
  for(var t=0;t<20;t++){var len=irand(r[0],r[1]),pool=POOLS[len];if(!pool)continue;
    var w=pool[irand(0,pool.length-1)];
    if(!words.some(function(o){return o.text===w;}))return w;}
  return 'forge';
}
function spawnWord(h){
  var text=pickWord(h),len=text.length;
  var s=clamp(36-len*1.7,18,34);s=Math.min(s,(W-28)/(len*.64));
  ctx.font='700 '+s+'px Menlo, Consolas, "Courier New", monospace';
  var cw=ctx.measureText('M').width,ww=len*cw;
  var m=mouths[irand(0,mouths.length-1)];m.flash=1;
  var cx=clamp(m.x+rand(-26,26),ww/2+10,W-ww/2-10);
  var spawnY=74,trav=clamp(10-h*.6,4.5,10)*rand(.92,1.1);
  words.push({text:text,x:cx,y:spawnY,vy:(forgeY-spawnY)/trav,s:s,cw:cw,ww:ww,typed:0,errT:0,seed:Math.random()*6.283});
  sparks(m.x,70,10,30,.9);
}
function target(){var b=null;for(var i=0;i<words.length;i++)if(!b||words[i].y>b.y)b=words[i];return b;}

function resetGame(){
  if(document.activeElement&&document.activeElement.blur)document.activeElement.blur();
  words.length=0;parts.length=0;floats.length=0;
  score=0;combo=0;maxCombo=0;lives=3;wordsTyped=0;keys=hits=misses=0;
  rt=0;nextSpawn=.9;nextJet=2.5;nextGroan=7;heatNow=1;lastHeat=1;shake=0;
  updatePips();updateHUD();
  hide(startScr);hide(overScr);hide(pauseScr);
  state='playing';sfx.start();
}
function forgeHit(w){
  lives--;combo=0;shake=11;
  hitFlash.classList.remove('on');void hitFlash.offsetWidth;hitFlash.classList.add('on');
  sfx.boom();
  chunks(w.x,w.y,14);smoke(w.x,w.y-8,8);sparks(w.x,w.y,26,10,1.4);
  floatText(clamp(w.x,70,W-70),forgeY-34,'FORGE HIT','#ff6a4a');
  updatePips();updateHUD();
  if(lives<=0)gameOver();
}
function onKey(k){
  keys++;
  var w=target(),wasMult;
  if(w&&w.text[w.typed]===k){
    hits++;combo++;maxCombo=Math.max(maxCombo,combo);w.typed++;
    var lx=w.x-w.ww/2+(w.typed-1)*w.cw;
    sparks(lx+w.cw/2,w.y+6,7,38,.9);sfx.tick(combo);
    wasMult=1+Math.min(4,Math.floor((combo-1)/10));
    var mult=1+Math.min(4,Math.floor(combo/10));
    if(mult>wasMult){floatText(W/2,H*.3,'COMBO ×'+mult,'#ffd9a0');sfx.tier();}
    if(w.typed>=w.text.length){
      var gain=w.text.length*10*mult;score+=gain;wordsTyped++;
      sparks(w.x,w.y,22,45,1.3);chunks(w.x,w.y,6);smoke(w.x,w.y,4);
      floatText(clamp(w.x,60,W-60),w.y,'+'+gain,'#ffe9b8');
      sfx.smash();shake=Math.max(shake,2.5);
      words.splice(words.indexOf(w),1);
    }
  }else{
    misses++;combo=0;shake=Math.max(shake,3);sfx.err();
    if(w)w.errT=.28;
  }
  updateHUD();
}

/* ================= update ================= */
function update(dt){
  rt+=dt;heatNow=1+Math.floor(rt/16);
  if(heatNow>lastHeat){lastHeat=heatNow;floatText(W/2,H*.35,'HEAT '+heatNow,'#ff8a3c');}
  if(rt>=nextSpawn&&words.length<Math.min(3+Math.floor(heatNow/2),8)){
    spawnWord(heatNow);
    nextSpawn=rt+clamp(2.3-heatNow*.16,.85,2.3)*rand(.75,1.2);
  }
  for(var i=words.length-1;i>=0;i--){
    var w=words[i];w.y+=w.vy*dt;
    if(w.errT>0)w.errT-=dt;
    if(w.y>=forgeY-6){words.splice(i,1);forgeHit(w);if(state!=='playing')return;}
  }
  nextJet-=dt;
  if(nextJet<=0){nextJet=rand(2.2,4.5);var jx=rand(W*.1,W*.9);
    for(var j=0;j<12;j++)addPart({t:'s',x:jx+rand(-8,8),y:H-40,vx:rand(-25,25),vy:rand(-320,-160),g:-40,l:rand(.4,.9),L:0,r:rand(1,2.4),hue:rand(20,45)});}
  if(AC&&rt>=nextGroan){nextGroan=rt+rand(6,11);tone(rand(42,64),.8,'sine',.05,38);}
  updParts(dt);updFloats(dt);
  shake=Math.max(0,shake-dt*18);
}

  for(var i=0;i<embers.length;i++){var e=embers[i];if(e.front!==front)continue;

}

function drawWord(w,tw){
  var danger=clamp((w.y-(forgeY-140))/140,0,1);

  ctx.font='700 '+w.s+'px Menlo, Consolas, "Courier New", monospace';
  ctx.textBaseline='middle';ctx.textAlign='left';
  var bx=w.x-w.ww/2,by=w.y+w.s*.62;
  var ratio=clamp((forgeY-w.y)/(forgeY-74),0,1);
  ctx.fillStyle='rgba(255,255,255,.08)';ctx.fillRect(bx,by,w.ww,3);
  ctx.fillStyle=danger>0?'hsla('+lerp(42,4,danger).toFixed(0)+',95%,'+lerp(62,52,danger).toFixed(0)+'%,.85)':'hsla(42,95%,62%,.7)';
  ctx.fillRect(bx,by,w.ww*ratio,3);
  for(var i=w.typed;i<w.text.length;i++){
    var lx=bx+i*w.cw;
    if(w===tw){
      ctx.fillStyle=danger>.6?('rgb(255,'+((190-80*pulse)|0)+',120)'):'#fff3d8';
      ctx.shadowColor='rgba(255,170,70,.9)';ctx.shadowBlur=14+6*danger*pulse;
    }else{
      ctx.fillStyle=danger>.6?'#ff8f66':'#ffd9a0';
      ctx.shadowColor='rgba(255,140,50,.5)';ctx.shadowBlur=7;
    }
    ctx.fillText(w.text[i],lx,w.y);
  }
  ctx.shadowBlur=0;
  if(w===tw&&w.typed<w.text.length){
    if((T*2.6)%1<.65){var cx2=bx+w.typed*w.cw;
      ctx.fillStyle='#fff';ctx.shadowColor='rgba(255,220,150,.9)';ctx.shadowBlur=8;

function frame(now){
  var dt=Math.min(.05,(now-last)/1000);last=now;T+=dt;
  for(var i=0;i<mouths.length;i++)mouths[i].flash=Math.max(0,mouths[i].flash-dt*2.4);
  if(state==='playing')update(dt);
  else{if(state!=='paused'){updParts(dt);updFloats(dt);}shake=Math.max(0,shake-dt*18);}
  var ed;while((ed=embers.pop())){ed.y-=ed.vy*dt;ed.x+=ed.vx*dt+Math.sin(T*ed.sp+ed.ph)*.15;
    if(ed.y<-12||ed.x<-12||ed.x>W+12)embers.push(mkEmber(false));else embers.push(ed);}
  render();
  requestAnimationFrame(frame);
}
