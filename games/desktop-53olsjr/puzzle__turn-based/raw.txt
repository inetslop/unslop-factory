function computePower(){

 for(const c of C){c.powered=false;c.dist=-1;if(c.t==='G')c.active=false;}
 for(const l of S.lamps)l.lit=false;
 const q=[];
 const push=(x,y,d)=>{const c=C[y*W+x];if(!c||c.t!=='P'||c.powered)return;c.powered=true;c.dist=d;q.push([x,y]);};
 for(const c of C){
  if(c.t!=='G')continue;
  const nx=c.x+DX[c.dir],ny=c.y+DY[c.dir];
  const n=cellAt(nx,ny);
  if(n&&n.t==='P'&&(rotateMask(n.m0,n.rot)&(1<<OPP[c.dir]))){c.active=true;push(nx,ny,0);}
  else{for(const l of S.lamps)if(l.x===nx&&l.y===ny&&l.d===OPP[c.dir]){l.lit=true;c.active=true;}}
 }
 while(q.length){
  const[x,y]=q.shift();const c=C[y*W+x];const m=rotateMask(c.m0,c.rot);
  for(let d=0;d<4;d++){
   if(!(m&(1<<d)))continue;

   if(n&&n.t==='P'&&(rotateMask(n.m0,n.rot)&(1<<OPP[d]))){if(!n.powered){n.powered=true;n.dist=c.dist+1;q.push([nx,ny]);}}
   else if(!n){for(const l of S.lamps)if(l.x===nx&&l.y===ny&&l.d===OPP[d])l.lit=true;}
  }
 }
}

 for(let i=P.length-1;i>=0;i--){
