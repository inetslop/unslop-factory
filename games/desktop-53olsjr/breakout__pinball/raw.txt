function patternKeep(p,c,r,rows){
  if(p===0)return true;
  if(p===1)return (c+r)%2===0;
  if(p===2){const mid=(rows-1)/2;return Math.abs(c-3.5)/3.5+Math.abs(r-mid)/(mid||1)<=1.06;}
  return r<2||c%2===0;
}
function buildField(lv){
  bricks=[];
  const rows=Math.min(4+Math.ceil(lv/2),6), p=(lv-1)%4;
  const total=COLS*BW+(COLS-1)*BGAP, x0=252-total/2;
  for(let r=0;r<rows;r++)for(let c=0;c<COLS;c++){
    if(!patternKeep(p,c,r,rows))continue;
    let hp=1; if(lv>=2&&r<2)hp=2; if(lv>=4&&(r*3+c)%7===0)hp=2;
    bricks.push({x:x0+c*(BW+BGAP),y:BY0+r*VPITCH,row:r,hp,flash:0,pop:Math.random()*.4});
  }
}

function collideSeg(b,x1,y1,x2,y2,e){
  const dx=x2-x1,dy=y2-y1,L2=dx*dx+dy*dy;
  let t=((b.x-x1)*dx+(b.y-y1)*dy)/L2;t=t<0?0:t>1?1:t;
  const px=x1+dx*t,py=y1+dy*t;
  let nx=b.x-px,ny=b.y-py;const d=Math.hypot(nx,ny);
  if(d>=BALL_R||d===0)return 0;
  nx/=d;ny/=d;b.x=px+nx*BALL_R;b.y=py+ny*BALL_R;
  const vn=b.vx*nx+b.vy*ny;
  if(vn<0){b.vx-=(1+e)*vn*nx;b.vy-=(1+e)*vn*ny;return -vn;}
  return 0;
}
function flipDir(f){return f.side<0?{x:Math.cos(f.a),y:Math.sin(f.a)}:{x:-Math.cos(f.a),y:Math.sin(f.a)};}
function collideFlipper(b,f){
  const d=flipDir(f);
  let t=(b.x-f.ox)*d.x+(b.y-f.oy)*d.y;t=clamp(t,0,F_LEN);
  const px=f.ox+d.x*t,py=f.oy+d.y*t;
  let nx=b.x-px,ny=b.y-py;const dist=Math.hypot(nx,ny),R=BALL_R+F_R;
  if(dist>=R||dist===0)return;
  nx/=dist;ny/=dist;b.x=px+nx*R;b.y=py+ny*R;
  const dd=f.side<0?{x:-Math.sin(f.a),y:Math.cos(f.a)}:{x:Math.sin(f.a),y:Math.cos(f.a)};
  const pvx=f.omega*dd.x*t,pvy=f.omega*dd.y*t;
  const rvx=b.vx-pvx,rvy=b.vy-pvy,vn=rvx*nx+rvy*ny;
  if(vn<0){
    const e=Math.abs(vn)>140?.35:0;
    b.vx-=(1+e)*vn*nx;b.vy-=(1+e)*vn*ny;
    combo=0;
    if(Math.abs(vn)>260)sfx('fhit',()=>{noise(.04,.2,700);tone(110,.05,'triangle',.22);},.03);
  }
}
function hitBrick(br){
  br.hp--;br.flash=.18;
  if(br.hp<=0){
    const i=bricks.indexOf(br);if(i>=0)bricks.splice(i,1);
    combo++;

    spawnParts(br.x+BW/2,br.y+BH/2,BRICK_COLORS[br.row%6],10,240);
    sBrick(br.row,true);
    if(combo>=6&&frenzyT<=0){frenzyT=8;showBanner('FRENZY!','×2 SCORE — KEEP THE CHAIN GOING');sFrenzy();}
    if(bricks.length===0&&mode==='playing'){
      mode='levelup';levelT=1.6;
      const bonus=500*level;score+=bonus;
      const mb=(level%3===0);
      showBanner('FIELD CLEARED','+'+bonus+' BONUS'+(mb?' · MULTIBALL!':''));
      sClear();if(mb){sMulti();balls.push({x:150,y:70,vx:-90,vy:140,trail:[],dead:false,still:0},{x:354,y:70,vx:90,vy:140,trail:[],dead:false,still:0});}
      addShake(5,.3);
    }
  }else sBrick(br.row,false);
}
function collideBricks(b){
  for(const br of bricks){
    const cx=clamp(b.x,br.x,br.x+BW),cy=clamp(b.y,br.y,br.y+BH);
    let nx=b.x-cx,ny=b.y-cy;const d2=nx*nx+ny*ny;
    if(d2>BALL_R*BALL_R)continue;
    let d=Math.sqrt(d2);
    if(d<.001){ // center inside brick: push out along smallest axis
      const l=b.x-br.x,r=br.x+BW-b.x,t=b.y-br.y,bo=br.y+BH-b.y,m=Math.min(l,r,t,bo);
      nx=m===l?-1:m===r?1:0;ny=m===t?-1:m===bo?1:0;d=1;
    }else{nx/=d;ny/=d;}
    b.x=cx+nx*BALL_R;b.y=cy+ny*BALL_R;
    const vn=b.vx*nx+b.vy*ny;
    if(vn<0){b.vx-=1.85*vn*nx;b.vy-=1.85*vn*ny;}
    let sp=Math.hypot(b.vx,b.vy);if(sp<430&&sp>1){b.vx*=430/sp;b.vy*=430/sp;}
    hitBrick(br);return;
  }
}
function collideBumpers(b){
  for(const bu of bumpers){
    let nx=b.x-bu.x,ny=b.y-bu.y;const d=Math.hypot(nx,ny),R=BALL_R+bu.r;
    if(d>=R||d===0)continue;
    nx/=d;ny/=d;b.x=bu.x+nx*R;b.y=bu.y+ny*R;
    const vn=b.vx*nx+b.vy*ny;
    let tx=b.vx-vn*nx,ty=b.vy-vn*ny;
    b.vx=nx*640+tx*.3;b.vy=ny*640+ty*.3;
    bu.flash=1;addScore(100,bu.x,bu.y-bu.r-8,'#ffd43b');
    spawnParts(b.x-nx*BALL_R,b.y-ny*BALL_R,'#ff8ac2',7,260);
    sBumper();addShake(2.5,.12);
  }
}
function collideKickers(b){
  for(const k of kickers){
    const dx=k.bx-k.ax,dy=k.by-k.ay,L2=dx*dx+dy*dy;
    let t=((b.x-k.ax)*dx+(b.y-k.ay)*dy)/L2;t=t<0?0:t>1?1:t;
    const px=k.ax+dx*t,py=k.ay+dy*t;
    let nx=b.x-px,ny=b.y-py;const d=Math.hypot(nx,ny);
    if(d>=BALL_R||d===0)continue;
    nx/=d;ny/=d;b.x=px+nx*BALL_R;b.y=py+ny*BALL_R;
    const vn=b.vx*nx+b.vy*ny;
    if(vn<0){b.vx-=2.1*vn*nx;b.vy-=2.1*vn*ny;}
    const out=b.vx*nx+b.vy*ny;
    if(out<560){b.vx+=nx*(560-out);b.vy+=ny*(560-out);}
    k.flash=1;addScore(5,k.ax+(k.bx-k.ax)/2,k.ay-14,'#ffa94d');sSlingshot();
  }
}
function updFlipper(f,h){
  const held=f.side<0?(keyL||touchL):(keyR||touchR);
  const target=held?F_UP:F_REST,sp=held?26:15;
  let na=f.a;
  if(na<target)na=Math.min(target,na+sp*h);else na=Math.max(target,na-sp*h);
  f.omega=(na-f.a)/h;f.a=na;
}
function stepBall(b,h){
  b.vy+=G*h;b.x+=b.vx*h;b.y+=b.vy*h;
  for(const w of walls){const imp=collideSeg(b,w.x1,w.y1,w.x2,w.y2,w.e);if(imp>300&&simT-lastWallSnd>.05){lastWallSnd=simT;sWall(imp);}}
  collideKickers(b);collideBricks(b);collideBumpers(b);
  collideFlipper(b,flippers[0]);collideFlipper(b,flippers[1]);
  const sp=Math.hypot(b.vx,b.vy);
  if(sp>MAXSPD){b.vx*=MAXSPD/sp;b.vy*=MAXSPD/sp;}
}

/* ============================== UPDATE ============================== */
function step(dt){
  // timers
  if(frenzyT>0){frenzyT-=dt;if(frenzyT<=0){frenzyT=0;combo=0;}}
  for(const bu of bumpers)bu.flash=Math.max(0,bu.flash-dt*4);
  for(const k of kickers)k.flash=Math.max(0,k.flash-dt*5);
  for(const br of bricks){br.flash=Math.max(0,br.flash-dt*6);br.pop=Math.min(1,br.pop+dt*3.2);}
  if(shakeT>0)shakeT=Math.max(0,shakeT-dt);
  if(plunger.held&&canPlungerNow())plunger.charge=Math.min(1,plunger.charge+dt*1.3);
  if(mode==='levelup'){levelT-=dt;if(levelT<=0){level++;buildField(level);mode='playing';}}

  // flippers + ball substeps
  const n=4,h=dt/n;
  for(let s=0;s<n;s++){
    updFlipper(flippers[0],h);updFlipper(flippers[1],h);
    for(const b of balls)if(!b.dead)stepBall(b,h);
  }
  // drains + anti-stall + trails
  let anyDead=false;
  for(const b of balls){
    if(!b.dead&&b.y-BALL_R>H+8){b.dead=true;anyDead=true;sDrain();combo=0;addShake(4,.2);popup(252,H-130,'DRAIN','#ff6b6b',20);}
    if(b.dead)continue;
    const sp=Math.hypot(b.vx,b.vy);
    if(sp<15&&b.y<H-100){b.still+=dt;if(b.still>2.5){b.vy=-450;b.vx=Math.random()*160-80;b.still=0;}}else b.still=0;
    b.trail.push({x:b.x,y:b.y});if(b.trail.length>9)b.trail.shift();
  }
  if(anyDead){balls=balls.filter(b=>!b.dead);
    if(balls.length===0){
      if(spares>0){spares--;balls.push(readyBall());showBanner('BALL LOST','');}
      else{gameOver();return;}
    }}
  // particles & popups
  for(let i=particles.length-1;i>=0;i--){const p=particles[i];p.life-=dt;if(p.life<=0){particles.splice(i,1);continue;}
    p.vy+=620*dt;p.x+=p.vx*dt;p.y+=p.vy*dt;p.rot+=p.vr*dt;}
  for(let i=popups.length-1;i>=0;i--){popups[i].t+=dt;if(popups[i].t>.9)popups.splice(i,1);}
}

    if(k.flash>0)ctx.restore();
  }
}
function drawBumpers(){
  bumpers.forEach((bu,i)=>{
    const p=1+Math.sin(simT*2.6+i*2)*.03+bu.flash*.25,r=bu.r*p;
    const g=ctx.createRadialGradient(bu.x,bu.y,r*.2,bu.x,bu.y,r);
    g.addColorStop(0,'#6b1a44');g.addColorStop(1,'#22081a');
    ctx.fillStyle=g;ctx.beginPath();ctx.arc(bu.x,bu.y,r,0,7);ctx.fill();
    if(bu.flash>0){ctx.save();ctx.shadowColor='#ff4fa3';ctx.shadowBlur=20;}
    ctx.strokeStyle='#ff4fa3';ctx.lineWidth=3.5;ctx.beginPath();ctx.arc(bu.x,bu.y,r,0,7);ctx.stroke();
    if(bu.flash>0)ctx.restore();
    ctx.strokeStyle='rgba(255,138,194,.5)';ctx.lineWidth=2;ctx.beginPath();ctx.arc(bu.x,bu.y,r*.66,0,7);ctx.stroke();
    const g2=ctx.createRadialGradient(bu.x-4,bu.y-4,1,bu.x,bu.y,r*.45);
    g2.addColorStop(0,'#fff8d8');g2.addColorStop(1,'#ffd43b');
    ctx.fillStyle=g2;ctx.beginPath();ctx.arc(bu.x,bu.y,r*.45,0,7);ctx.fill();
    if(bu.flash>0){ctx.strokeStyle='rgba(255,255,255,'+bu.flash+')';ctx.lineWidth=3;
      ctx.beginPath();ctx.arc(bu.x,bu.y,r*(1+(1-bu.flash)*.9),0,7);ctx.stroke();}
  });
}
function drawBricks(){
  for(const br of bricks){
    const al=clamp(br.pop*2,0,1),yOff=(1-br.pop)*-8;
    ctx.globalAlpha=al;
    rr(br.x,br.y+yOff,BW,BH,4);ctx.fillStyle=BRICK_COLORS[br.row%6];ctx.fill();
    rr(br.x+1.5,br.y+yOff+1.5,BW-3,7,3);ctx.fillStyle='rgba(255,255,255,.3)';ctx.fill();
    ctx.fillStyle='rgba(0,0,0,.28)';rr(br.x+1.5,br.y+yOff+BH-4,BW-3,2.5,1);ctx.fill();
    if(br.hp>1){ctx.strokeStyle='rgba(255,255,255,.6)';ctx.lineWidth=1.5;rr(br.x+3,br.y+yOff+3,BW-6,BH-6,2.5);ctx.stroke();
      ctx.fillStyle='#fff';ctx.beginPath();const cx=br.x+BW/2,cy=br.y+yOff+BH/2,s=4;
      ctx.moveTo(cx,cy-s);ctx.lineTo(cx+s,cy);ctx.lineTo(cx,cy+s);ctx.lineTo(cx-s,cy);ctx.closePath();ctx.fill();}
    if(br.flash>0){rr(br.x,br.y+yOff,BW,BH,4);ctx.fillStyle='rgba(255,255,255,'+br.flash*.8+')';ctx.fill();}
  }
  ctx.globalAlpha=1;
}
function drawFlippers(){
  for(const f of flippers){
    const d=flipDir(f),tx=f.ox+d.x*F_LEN,ty=f.oy+d.y*F_LEN;
    ctx.lineCap='round';
    ctx.strokeStyle='#5e2405';ctx.lineWidth=F_R*2+5;
    ctx.beginPath();ctx.moveTo(f.ox,f.oy);ctx.lineTo(tx,ty);ctx.stroke();
    const g=ctx.createLinearGradient(f.ox,f.oy,tx,ty);
    g.addColorStop(0,'#ffd43b');g.addColorStop(1,'#ff7a00');
    ctx.strokeStyle=g;ctx.lineWidth=F_R*2;
    ctx.beginPath();ctx.moveTo(f.ox,f.oy);ctx.lineTo(tx,ty);ctx.stroke();
    ctx.fillStyle='#2a3560';ctx.beginPath();ctx.arc(f.ox,f.oy,F_R+2,0,7);ctx.fill();
    ctx.fillStyle='#cfe0ff';ctx.beginPath();ctx.arc(f.ox,f.oy,4,0,7);ctx.fill();
  }
}
function drawBalls(){
  for(const b of balls){if(b.dead)continue;
    for(let i=0;i<b.trail.length;i++){const p=b.trail[i],k=i/b.trail.length;
      ctx.globalAlpha=k*.3;ctx.fillStyle='#7ec8ff';
      ctx.beginPath();ctx.arc(p.x,p.y,BALL_R*(.35+.6*k),0,7);ctx.fill();}
    ctx.globalAlpha=1;
    if(frenzyT>0){ctx.save();ctx.shadowColor='#ff4fa3';ctx.shadowBlur=14;}
    const g=ctx.createRadialGradient(b.x-3,b.y-3.5,1,b.x,b.y,BALL_R);
    g.addColorStop(0,'#ffffff');g.addColorStop(.4,'#cdd9ff');g.addColorStop(1,'#4a5aa8');
    ctx.fillStyle=g;ctx.beginPath();ctx.arc(b.x,b.y,BALL_R,0,7);ctx.fill();
    if(frenzyT>0)ctx.restore();
  }
}
function drawParticles(){
  for(const p of particles){const k=p.life/p.max;ctx.globalAlpha=Math.min(1,k*1.6);
    ctx.save();ctx.translate(p.x,p.y);ctx.rotate(p.rot);ctx.fillStyle=p.color;
    ctx.fillRect(-p.size/2,-p.size/2,p.size,p.size);ctx.restore();}
  ctx.globalAlpha=1;
}
function drawPopups(){
  ctx.textAlign='center';
  for(const p of popups){const k=p.t/.9;ctx.globalAlpha=1-k;
    ctx.font='700 '+p.size+'px '+FONT_M;

    ctx.fillStyle=p.color;ctx.fillText(p.text,p.x,p.y-k*38);}
  ctx.globalAlpha=1;
}
function render(dt){
  ctx.setTransform(dpr,0,0,dpr,0,0);
  drawBG(dt);
  let sx=0,sy=0;
  if(shakeT>0){const k=shakeT/shakeDur;sx=(Math.random()*2-1)*shakeAmp*k;sy=(Math.random()*2-1)*shakeAmp*k;}
  ctx.translate(offX+sx,offY+sy);ctx.scale(scl,scl);
  drawTable();drawLane();drawWalls();drawKickers();drawBumpers();drawBricks();drawFlippers();drawBalls();drawParticles();drawPopups();
  // vignette
  ctx.setTransform(dpr,0,0,dpr,0,0);
  const v=ctx.createRadialGradient(cw/2,ch/2,Math.min(cw,ch)*.35,cw/2,ch/2,Math.max(cw,ch)*.75);
  v.addColorStop(0,'rgba(0,0,0,0)');v.addColorStop(1,'rgba(3,4,10,.55)');
  ctx.fillStyle=v;ctx.fillRect(0,0,cw,ch);
}
