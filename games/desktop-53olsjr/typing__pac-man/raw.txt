function handleLetter(c){
  if(G.sub!=='run')return;
  if(c===G.word[G.typed]){
    G.typed++;G.targetCell=G.typed;G.correct++;G.combo++;
    if(G.combo>G.bestCombo)G.bestCombo=G.combo;
    G.score+=10*mult();if(G.score>hi)hi=G.score;
    G.scorePulse=0.15;G.comboPop=0.2;
    (G.power?sfx.powerEat:sfx.eat)(G.typed);
    if(G.combo===6||G.combo===14||G.combo===25)
      popup(M.laneLeft+M.cell*G.pacCell,M.laneY-M.pacR-30,'COMBO ×'+mult(),'#ffab2e',18);
    if(G.typed>=G.L){G.sub='chomp';G.chompT=0;G.chompFrom=G.pacCell;}
  }else wrongKey(c);
}

/* ============ update ============ */
function creepCps(){return Math.min(90,16+5.5*(G.level-1))/M.cell;}

function update(dt){
  G.fxT+=dt;
  // fx always tick
  for(let i=parts.length-1;i>=0;i--){const p=parts[i];p.t+=dt;p.x+=p.vx*dt;p.y+=p.vy*dt;
    p.vx*=(1-2.4*dt);p.vy*=(1-2.4*dt);if(p.t>=p.life)parts.splice(i,1);}
  for(let i=pops.length-1;i>=0;i--){const p=pops[i];p.t+=dt;p.y-=46*dt;if(p.t>=p.life)pops.splice(i,1);}
  for(let i=rings.length-1;i>=0;i--){rings[i].t+=dt;if(rings[i].t>0.5)rings.splice(i,1);}
  G.shake.t+=dt;G.banner.t+=dt;
  G.scorePulse=Math.max(0,G.scorePulse-dt*1.4);
  G.comboPop=Math.max(0,G.comboPop-dt*1.6);
  G.pacFade=Math.max(0,G.pacFade-dt*3);

  if(G.state==='menu'){
    G.animT+=dt;G.attractT+=dt;
    G.aPac=1+(G.attractT*1.55)%9.6;
    const c=Math.floor(G.aPac),cl=G.aLast;
    for(let i=Math.floor(cl)+1;i<=c;i++)burst(M.laneLeft+M.cell*i,M.laneY,4,'#ffd23f',80,2.4,0.4);
    G.aLast=c;
  }
  else if(G.state==='play'){
    G.animT+=dt;G.playTime+=dt;
    if(G.sub==='run'){
      G.pacCell+=(G.targetCell-G.pacCell)*Math.min(1,dt*14);
      while(G.dotsEaten<G.typed&&G.pacCell>G.dotsEaten+0.72){
        G.dotsEaten++;
        burst(M.laneLeft+M.cell*G.dotsEaten,M.laneY,5,G.power?'#ffd500':'#ffd23f',90,2.6,0.45);
      }
      const ramp=1+0.55*(G.typed/G.L),fm=G.frighten>0?0.45:1;
      G.g-=creepCps()*dt*ramp*fm;
      if(G.g-G.pacCell<CONTACT)caught();
    }else if(G.sub==='chomp'){
      G.chompT+=dt;const k=Math.min(1,G.chompT/0.16);
      G.pacCell=lerp(G.chompFrom,G.g-0.55,easeOutCubic(k));
      if(k>=1)eatGhost();
    }else if(G.sub==='dying'){
      G.dieT+=dt;
      if(G.dieT>1.05){
        if(G.lives<=0)gameOver();
        else{G.typed=0;G.targetCell=0;G.pacCell=0;G.g=TRACK;G.dotsEaten=0;
          G.sub='run';G.pacFade=1;G.ghostBorn=G.animT;}
      }
    }else if(G.sub==='next'){
      G.nextT+=dt;if(G.nextT>0.5)newWord();
    }
    // trail
    const v=clamp((G.targetCell-G.pacCell)*3,0,1);
    if((G.sub==='run'||G.sub==='chomp')&&v>0.15){
      G.trail.push({x:M.laneLeft+M.cell*G.pacCell,y:M.laneY,v});
      if(G.trail.length>7)G.trail.shift();
    }else if(G.trail.length)G.trail.shift();
  }
}

/* ============ drawing ============ */
function drawPac(x,y,r,mouth,rot,alpha,color){
  ctx.save();ctx.translate(x,y);if(rot)ctx.rotate(rot);
  ctx.globalAlpha=alpha==null?1:alpha;
  ctx.shadowColor='rgba(255,213,0,.55)';ctx.shadowBlur=r*0.9;
  ctx.beginPath();const m=Math.max(0.06,mouth);
  ctx.moveTo(0,0);ctx.arc(0,0,r,m,TAU-m);ctx.closePath();
  ctx.fillStyle=color||'#ffd500';ctx.fill();ctx.restore();
}
function drawGhost(x,y,r,color,t,o){
  o=o||{};const fright=o.fright,blink=o.blink;
  ctx.save();ctx.translate(x,y);if(o.scale)ctx.scale(o.scale,o.scale);
  ctx.globalAlpha=o.alpha==null?1:o.alpha;
  if(!fright&&o.glow!=false){ctx.shadowColor=color;ctx.shadowBlur=r*0.8;}
  const by=r*0.85,n=4,seg=2*r/n;
  ctx.beginPath();ctx.arc(0,-r*0.15,r,Math.PI,TAU,false);
  ctx.lineTo(r,by);
  for(let i=0;i<=n;i++){const x=r-seg*i;
    const y=by+(i%2?-r*0.38:0)+Math.sin(t*9+i*1.7)*r*0.07;ctx.lineTo(x,y);}
  ctx.closePath();
  ctx.fillStyle=blink?'#ffffff':(fright?'#3440c8':color);ctx.fill();
  ctx.shadowBlur=0;
  if(fright){
    ctx.fillStyle='#dfe6ff';
    ctx.beginPath();ctx.arc(-r*0.3,-r*0.2,r*0.15,0,TAU);ctx.fill();
    ctx.beginPath();ctx.arc(r*0.3,-r*0.2,r*0.15,0,TAU);ctx.fill();
    ctx.strokeStyle='#dfe6ff';ctx.lineWidth=Math.max(1.5,r*0.12);ctx.beginPath();
    for(let i=0;i<=4;i++){const x=-r*0.45+i*r*0.225,y=r*0.3+((i%2)?-r*0.1:r*0.1);
      i?ctx.lineTo(x,y):ctx.moveTo(x,y);}
    ctx.stroke();
  }else{
    const look=o.look||-1;
    for(const s of[-1,1]){
      ctx.fillStyle='#fff';ctx.beginPath();
      ctx.ellipse(s*r*0.32,-r*0.25,r*0.26,r*0.34,0,0,TAU);ctx.fill();
      ctx.fillStyle='#101033';ctx.beginPath();
      ctx.arc(s*r*0.32+look*r*0.12,-r*0.2+r*0.05,r*0.13,0,TAU);ctx.fill();
    }
  }
  ctx.restore();
}

function drawMaze(){
  const mz=G.maze;if(!mz)return;
  ctx.drawImage(mz.img,mz.ox,mz.oy,mz.mw,mz.mh);
  for(const d of mz.dots){
    const a=0.4+0.38*Math.sin(G.fxT*2.4+d.ph);
    ctx.globalAlpha=a;ctx.fillStyle='#ffd23f';
    ctx.beginPath();ctx.arc(d.x,d.y,Math.max(1.5,mz.s*0.1),0,TAU);ctx.fill();
  }
  for(const p of mz.powers){
    const pr=mz.s*0.26*(1+0.22*Math.sin(G.fxT*4+p.ph));
    ctx.globalAlpha=0.9;ctx.fillStyle='#ffd500';
    ctx.shadowColor='#ffd500';ctx.shadowBlur=8;
    ctx.beginPath();ctx.arc(p.x,p.y,pr,0,TAU);ctx.fill();ctx.shadowBlur=0;
  }
  ctx.globalAlpha=1;
  // ambient patrol ghosts
  if(mz.s>7){
    const t=G.fxT;
    const ax=lerp(2,6.8,(Math.sin(t*0.45)+1)/2), ay=1.5;
    drawGhost(mz.ox+ax*mz.s,mz.oy+ay*mz.s,mz.s*0.55,'#ffa6e0',t,{alpha:0.45,glow:false});
    const bx=lerp(2,6.3,(Math.sin(t*0.38+2)+1)/2), by2=13.5;
    drawGhost(mz.ox+bx*mz.s,mz.oy+by2*mz.s,mz.s*0.5,'#00fff7',t+3,{alpha:0.4,glow:false});
  }
}

function drawLane(){
  const {laneY,laneLeft,laneRight,cell}=M,pacR=M.pacR;
  const y1=laneY-pacR*2.1,y2=laneY+pacR*2.1,x0=laneLeft-cell*0.9,x1=laneRight+cell*0.9;
  ctx.fillStyle='rgba(4,6,20,.35)';rr(x0-8,y1-6,x1-x0+16,y2-y1+12,10);ctx.fill();
  ctx.strokeStyle='#2b3df2';ctx.lineWidth=3;
  ctx.shadowColor='rgba(43,61,242,.9)';ctx.shadowBlur=10;
  ctx.beginPath();ctx.moveTo(x0,y1);ctx.lineTo(x1,y1);ctx.moveTo(x0,y2);ctx.lineTo(x1,y2);ctx.stroke();
  ctx.fillStyle='#3a52ff';
  for(const cx of[x0,x1])for(const cy of[y1,y2]){ctx.beginPath();ctx.arc(cx,cy,5,0,TAU);ctx.fill();}
  ctx.shadowBlur=0;

  const menu=G.state==='menu';
  const L=menu?11:G.L, born=menu?-9:G.animT;
  for(let i=1;i<=L;i++){
    const eaten=menu?(i<=Math.floor(G.aPac)):i<=G.dotsEaten;
    if(eaten)continue;
    let s=1;if(!menu)s=clamp((G.animT-G.dotsBorn-i*0.035)/0.18,0,1);s=easeOutBack(s);
    if(s<=0)continue;
    const x=laneLeft+cell*i;
    if(menu||!G.power){
      ctx.globalAlpha=0.95;ctx.fillStyle='#ffd23f';
      ctx.shadowColor='rgba(255,210,63,.7)';ctx.shadowBlur=6;
      ctx.beginPath();ctx.arc(x,laneY,pacR*0.2*s,0,TAU);ctx.fill();ctx.shadowBlur=0;
    }else{
      const pr=pacR*0.5*(1+0.18*Math.sin(G.fxT*6+i))*s;
      ctx.globalAlpha=0.95;ctx.fillStyle='#ffd500';
      ctx.shadowColor='#ffd500';ctx.shadowBlur=12;
      ctx.beginPath();ctx.arc(x,laneY,pr,0,TAU);ctx.fill();ctx.shadowBlur=0;
    }
  }
  ctx.globalAlpha=1;

  // pac-man (trail + body)
  if(!menu){
    const px=laneLeft+cell*G.pacCell;
    for(let i=0;i<G.trail.length;i++){const tr=G.trail[i];
      drawPac(tr.x-tr*(7-i)*2,laneY,pacR*0.9,0.5,0,0.16*(i/G.trail.length)*tr.v);}
    if(G.sub==='dying'){
      const k=clamp(G.dieT/1.0,0,1);
      drawPac(px,laneY,pacR*(1-k*k),0.3,G.dieT*20,1-k,'#ffd500');
    }else if(G.state!=='over'){
      const moving=Math.abs(G.targetCell-G.pacCell)>0.12;
      const mouth=G.sub==='chomp'?1.0:(moving?0.35+0.4*(0.5+0.5*Math.sin(G.animT*13)):0.18+0.12*Math.sin(G.animT*5));
      const pop=1+(1-G.pacFade)*0.35;
      drawPac(px,laneY,pacR*pop,mouth,0,G.pacFade,'#ffd500');
    }
  }else{
    drawPac(laneLeft+cell*G.aPac,laneY,pacR,0.3+0.4*(0.5+0.5*Math.sin(G.fxT*12)),0,1);
  }

  // ghost
  const gVis=menu||((G.sub==='run')||(G.sub==='dying'));
  if(gVis){
    const gx=laneLeft+cell*(menu?TRACK:G.g);
    const gy=laneY+Math.sin(G.animT*3)*cell*0.12;
    let sc=1;if(!menu)sc=easeOutBack(clamp((G.animT-G.ghostBorn)/0.28,0,1));
    const dist=(menu?9:G.g-G.pacCell);
    const danger=!menu&&G.sub==='run'&&dist<2;
    const blink=danger&&Math.floor(G.fxT*8)%2===0;
    drawGhost(gx,gy,M.gr,GHOST_TYPES[G.typeIdx].color,G.animT,
      {fright:G.frighten>0,blink:blink,look:-1,scale:sc});
    if(!menu){
      ctx.font='700 10px '+getComputedStyle(document.body).fontFamily;
      ctx.textAlign='center';ctx.fillStyle=G.typeIdx<0?'#fff':GHOST_TYPES[G.typeIdx].color;
      ctx.globalAlpha=0.85;ctx.fillText(GHOST_TYPES[G.typeIdx].name,gx,gy-M.gr-12);ctx.globalAlpha=1;
    }
  }

  // danger vignette
  if(!menu&&G.sub==='run'&&G.g-G.pacCell<2){
    const d=G.g-G.pacCell,a=(1-d/2)*(0.16+0.1*Math.sin(G.fxT*9));
    if(a>0.01){const gr=ctx.createRadialGradient(W/2,H*0.6,Math.min(W,H)*0.3,W/2,H*0.6,Math.max(W,H)*0.75);
      gr.addColorStop(0,'rgba(255,40,40,0)');gr.addColorStop(1,'rgba(255,40,40,'+a.toFixed(3)+')');
      ctx.fillStyle=gr;ctx.fillRect(0,0,W,H);}
  }
}

  const lt='LEVEL '+G.level,lw=ctx.measureText(lt).width;
  ctx.fillText(lt,W/2-G.frighten*10,36);ctx.shadowBlur=0;
  for(let i=0;i<G.frighten;i++)drawGhost(W/2-lw/2+8+i*20,30,7,'#3440c8',G.animT,{fright:true,glow:false});
  // lives + wpm (right, clear of DOM chips)
  const rx=W-70;
  for(let i=0;i<G.lives;i++)drawPac(rx-i*26,32,9*f,0.5,0,1);
  ctx.font='400 '+(12*f)+'px '+mono;ctx.textAlign='right';ctx.fillStyle='#8f9ad8';
  ctx.fillText(wpm()+' WPM'+(W>560?' · '+acc()+'% ACC':''),rx,58);
}

function drawFx(){
  for(const p of parts){const a=1-p.t/p.life;ctx.globalAlpha=a;ctx.fillStyle=p.color;
    ctx.beginPath();ctx.arc(p.x,p.y,p.size*a+0.4,0,TAU);ctx.fill();}
  ctx.globalAlpha=1;
  for(const r of rings){const k=r.t/0.5;
    ctx.globalAlpha=(1-k)*0.8;ctx.strokeStyle=r.color;ctx.lineWidth=3*(1-k)+1;
    ctx.beginPath();ctx.arc(r.x,r.y,lerp(10,90,easeOutCubic(k)),0,TAU);ctx.stroke();}
  ctx.globalAlpha=1;
  const mono="ui-monospace,Consolas,Menlo,monospace";
  for(const p of pops){const k=p.t/p.life,a=k<0.85?1:(1-k)/0.15;
    ctx.globalAlpha=a;ctx.font='700 '+p.size+'px '+mono;ctx.textAlign='center';
    ctx.shadowColor='rgba(0,0,0,.9)';ctx.shadowBlur=4;
    ctx.fillStyle=p.color;ctx.fillText(p.text,p.x,p.y);ctx.shadowBlur=0;}
  ctx.globalAlpha=1;
}

function drawBanner(){
  const b=G.banner;if(b.t>=b.dur)return;
  const k=b.t/b.dur,a=k<0.15?k/0.15:k>0.8?(1-k)/0.2:1;
  const y=H*0.34-(1-Math.min(1,k*6))*-24;
  ctx.save();ctx.globalAlpha=a;
  ctx.font='900 '+(clamp(W*0.05,26,48))+'px Impact,"Arial Black",sans-serif';
  ctx.textAlign='center';ctx.textBaseline='middle';
  ctx.shadowColor='#17208f';ctx.shadowOffsetY=4;ctx.shadowBlur=22;
  ctx.fillStyle='#ffd500';ctx.fillText(b.text,W/2,y);
  ctx.restore();
}

function render(){
  ctx.fillStyle=bgGrad||'#04040f';ctx.fillRect(0,0,W,H);
  // moving shimmer
  const sx=((G.fxT*36)%(W+240))-120;
  const sg=ctx.createLinearGradient(sx-70,0,sx+70,0);
  sg.addColorStop(0,'rgba(255,255,255,0)');sg.addColorStop(0.5,'rgba(160,180,255,.035)');sg.addColorStop(1,'rgba(255,255,255,0)');
  ctx.fillStyle=sg;ctx.fillRect(sx-70,0,140,H);

  let dx=0,dy=0;
  if(G.shake.t<G.shake.dur){const k=1-G.shake.t/G.shake.dur,m=G.shake.mag*k;
    dx=rand(-m,m);dy=rand(-m,m);}
  ctx.save();ctx.translate(dx,dy);
  drawMaze();
  drawLane();
  drawFx();
  if(G.state!=='menu')drawHUD();
  drawWord();
  drawBanner();
  ctx.restore();
}

/* ============ loop ============ */
let last=performance.now();
function frame(now){
  const dt=Math.min(0.05,(now-last)/1000);last=now;
  update(dt);render();
  requestAnimationFrame(frame);
}

})();
