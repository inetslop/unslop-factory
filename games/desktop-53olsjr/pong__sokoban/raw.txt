  let c=Math.min((W-sidePad*2)/cols,(H-topPad-botPad)/rowsN);
  cell=Math.max(16,c);
  bx=(W-cols*cell)/2; by=topPad+((H-topPad-botPad)-rowsN*cell)/2;
}

  cv.width=W*DPR;cv.height=H*DPR;cv.style.width=W+'px';cv.style.height=H+'px';
  ctx.setTransform(DPR,0,0,DPR,0,0);layout();}

/* ================= LEVEL / FLOW ================= */
function setBall(sp,a){ball.speed=sp;ball.vx=Math.cos(a)*sp;ball.vy=Math.sin(a)*sp;}
function loadLevel(i){
  levelIdx=i;const m=LEVELS[i].map;rowsN=m.length;cols=m[0].length;
  crates=[];pads=[];walls=[];padSet.clear();wallSet.clear();
  for(let r=0;r<rowsN;r++)for(let c=0;c<cols;c++){const ch=m[r][c];
    if(ch==='#')crates.push({cx:c,cy:r,px:c,py:r,solved:false,flash:0,squash:1});
    else if(ch==='@'){pads.push([c,r]);padSet.add(c+','+r);}
    else if(ch==='X'){walls.push([c,r]);wallSet.add(c+','+r);}}
  ball.x=cols/2;ball.y=rowsN/2;setBall(BASE_SPEED,rand(-.5,.5));
  paddle.y=rowsN/2;paddle.targetY=null;paddle.hitT=0;
  trail.length=0;parts.length=0;pops.length=0;menuResetT=0;levelTime=0;shake=0;dropFlash=0;
  layout();setTxt('lvNum','LV '+(levelIdx+1)+'/'+LEVELS.length);setTxt('lvName',LEVELS[i].name);
}
function startGame(){runStartBest=best;score=0;lives=3;pushes=0;loadLevel(0);state='play';showPanel(null);renderPips();sfx.ui();}
function retryLevel(){lives=3;loadLevel(levelIdx);state='play';showPanel(null);renderPips();sfx.ui();}
function toMenu(){score=0;lives=3;pushes=0;loadLevel(0);state='menu';setTxt('bestStart',String(best).padStart(5,'0'));showPanel('start');renderPips();}
function nextLevel(){
  if(levelIdx+1>=LEVELS.length){state='win';fillWin();showPanel('win');sfx.win();confetti();}
  else{lives=3;loadLevel(levelIdx+1);state='play';showPanel(null);renderPips();sfx.ui();}
}
function pauseGame(){if(state!=='play')return;state='paused';showPanel('pause');sfx.ui();}
function resumeGame(){if(state!=='paused')return;state='play';showPanel(null);sfx.ui();}
function saveBest(){if(score>best)best=score;try{localStorage.setItem('sokopong_best',String(best))}catch(e){}}
function fillWin(){$('wScore').textContent=String(score).padStart(5,'0');
  $('wPushes').textContent=String(pushes).padStart(2,'0');
  $('wBest').textContent=String(best).padStart(5,'0');
  $('newBest').style.display=(score>runStartBest&&score>0)?'inline-block':'none';}

/* ================= PHYSICS ================= */
function circleNormal(x,y,r,rx,ry,rw,rh){
  const cx=clamp(x,rx,rx+rw),cy=clamp(y,ry,ry+rh);
  let dx=x-cx,dy=y-cy;const d2=dx*dx+dy*dy;
  if(d2>r*r)return null;let nx,ny;
  if(d2>1e-9){const d=Math.sqrt(d2);nx=dx/d;ny=dy/d;}
  else{const l=x-rx,rr2=rx+rw-x,t=y-ry,b=ry+rh-y,m=Math.min(l,rr2,t,b);
    if(m===l){nx=-1;ny=0}else if(m===rr2){nx=1;ny=0}else if(m===t){nx=0;ny=-1}else{nx=0;ny=1}}
  if(Math.abs(nx)>=Math.abs(ny)){ny=0;nx=nx<0?-1:1}else{nx=0;ny=ny<0?-1:1}
  return{nx,ny};
}
function reflect(b,n){const d=b.vx*n.nx+b.vy*n.ny;if(d<0){b.vx-=2*d*n.nx;b.vy-=2*d*n.ny;}}
function resolveSolid(n,rx,ry,rw,rh){reflect(ball,n);
  if(n.nx<0)ball.x=rx-BALL_R;else if(n.nx>0)ball.x=rx+rw+BALL_R;
  if(n.ny<0)ball.y=ry-BALL_R;else if(n.ny>0)ball.y=ry+rh+BALL_R;}
function wallFx(x,y){sfx.wall();burst(x,y,'#6ea8ff',4,1.6,.25);}

function canPush(c,dx,dy){const tx=c.cx+dx,ty=c.cy+dy;
  if(tx<1||tx>=cols||ty<0||ty>=rowsN)return false;
  if(wallSet.has(tx+','+ty))return false;
  for(const o of crates)if(o!==c&&o.cx===tx&&o.cy===ty)return false;return true;}
function tryPush(c,dx,dy,cxp,cyp){
  if(!canPush(c,dx,dy)){c.flash=.22;sfx.blocked();shake=Math.max(shake,2.2);burst(cxp,cyp,'#8b96c9',7,2.4,.4);return false;}
  const ox=c.cx,oy=c.cy;c.cx+=dx;c.cy+=dy;c.squash=.76;
  if(state==='play'){pushes++;score+=10;}
  sfx.push();shake=Math.max(shake,1.4);
  burst(ox+(dx>0?1:dx<0?0:.5),oy+(dy>0?1:dy<0?0:.5),'#ffb347',9,2.8,.45);
  if(padSet.has(c.cx+','+c.cy)){c.solved=true;sfx.land();shake=Math.max(shake,3);
    burst(c.cx+.5,c.cy+.5,'#3ddc97',26,4.2,.8,true);
    addPop(c.cx+.5,c.cy-.1,'+150','#3ddc97');
    if(state==='play')score+=150;
    checkWin();}
  return true;
}
function checkWin(){
  if(crates.length&&crates.every(c=>c.solved)){
    if(state!=='play'){return;}
    const bonus=Math.max(50,300-Math.floor(levelTime)*6);
    score+=bonus;saveBest();state='clear';sfx.clear();
    setTimeout(()=>{if(state!=='clear')return;
      $('cBonus').textContent='+'+bonus;
      $('cNext').textContent=levelIdx+1<LEVELS.length?'NEXT: '+LEVELS[levelIdx+1].name:'FINALE';
      showPanel('clear');},700);
  }
}
function doDrop(){
  dropFlash=1;shake=Math.max(shake,5);sfx.drop();
  if(state==='menu'){respawnBall();return;}
  lives--;score=Math.max(0,score-30);renderPips(true);addPop(.7,paddle.y,'−30','#ff5c5c');
  if(lives<=0){state='over';saveBest();showPanel('over');}else respawnBall();
}
function respawnBall(){ball.x=cols/2;ball.y=rowsN/2;setBall(BASE_SPEED,rand(-.55,.55));trail.length=0;}

function physics(dt){
  const R=BALL_R,steps=Math.max(1,Math.ceil(ball.speed*dt/.15)),sdt=dt/steps;
  for(let s=0;s<steps;s++){
    ball.x+=ball.vx*sdt;ball.y+=ball.vy*sdt;
    if(ball.y<R){ball.y=R+(R-ball.y);if(ball.vy<0)ball.vy=-ball.vy;wallFx(ball.x,R);}
    else if(ball.y>rowsN-R){ball.y=rowsN-R-(ball.y-(rowsN-R));if(ball.vy>0)ball.vy=-ball.vy;wallFx(ball.x,rowsN-R);}
    if(ball.x>cols-R){ball.x=cols-R-(ball.x-(cols-R));if(ball.vx>0)ball.vx=-ball.vx;wallFx(cols-R,ball.y);}
    if(ball.vx<0){
      const n=circleNormal(ball.x,ball.y,R,FACE_X-PW,paddle.y-PH/2,PW,PH);
      if(n&&n.nx<=0){
        ball.x=FACE_X+R;
        const rel=clamp((ball.y-paddle.y)/(PH/2),-1.15,1.15);
        const ang=clamp(rel*.85+clamp(paddle.vy,-10,10)*.02,-1.05,1.05);
        ball.speed=Math.min(ball.speed+SPEED_GAIN,MAX_SPEED);
        ball.vx=Math.cos(ang)*ball.speed;ball.vy=Math.sin(ang)*ball.speed;
        paddle.hitT=.16;sfx.paddle();shake=Math.max(shake,1);
        burst(FACE_X+.05,ball.y,'#ffd23f',8,3,.35);}
    }
    if(ball.x<-R*1.6){doDrop();return;}
    for(const w of walls){const n=circleNormal(ball.x,ball.y,R,w[0],w[1],1,1);
      if(n){resolveSolid(n,w[0],w[1],1,1);wallFx(w[0]+(n.nx<0?0:1),w[1]+(n.ny<0?0:1));}}
    for(const c of crates){const n=circleNormal(ball.x,ball.y,R,c.cx,c.cy,1,1);
      if(n){resolveSolid(n,c.cx,c.cy,1,1);
        tryPush(c,-n.nx,-n.ny,c.cx+(n.nx<0?0:1),c.cy+(n.ny<0?0:1));break;}}
  }
}

/* ================= PADDLE / FX UPDATE ================= */
function updatePaddle(dt){
  const prev=paddle.y;
  if(state==='menu')paddle.y+=(clamp(ball.y,PH/2,rowsN-PH/2)-paddle.y)*Math.min(1,dt*5);
  else{const dir=(keys.up?-1:0)+(keys.down?1:0);
    if(dir)paddle.y+=dir*PADDLE_SPD*dt;
    else if(paddle.targetY!=null)paddle.y+=(paddle.targetY-paddle.y)*Math.min(1,dt*16);}
  paddle.y=clamp(paddle.y,PH/2,rowsN-PH/2);
  paddle.vy=(paddle.y-prev)/Math.max(dt,1e-4);
  paddle.hitT=Math.max(0,paddle.hitT-dt);
}
function burst(x,y,col,n,spd,life,big){
  for(let i=0;i<n;i++){const a=rand(0,Math.PI*2),v=rand(spd*.3,spd);
    parts.push({x,y,vx:Math.cos(a)*v,vy:Math.sin(a)*v-(big?1.2:0),life:rand(life*.5,life),max:life,col,s:rand(cell*.06,cell*(big?.17:.1)),g:big?2.6:.8});}
  if(parts.length>420)parts.splice(0,parts.length-420);
}
function addPop(x,y,txt,col){pops.push({x,y,txt,col,t:1});}
function confetti(){const cs=['#ffd23f','#ff5c5c','#3ddc97','#6ea8ff'];
  for(let i=0;i<6;i++)burst(rand(2,cols-2),rand(1,rowsN-1),cs[i%4],24,5,1.3,true);}
function updateFX(dt){
  shake*=Math.exp(-7*dt);if(shake<.05)shake=0;
  dropFlash=Math.max(0,dropFlash-dt*2.2);
  for(const c of crates){c.px+=(c.cx-c.px)*Math.min(1,dt*14);c.py+=(c.cy-c.py)*Math.min(1,dt*14);
    c.squash+=(1-c.squash)*Math.min(1,dt*10);c.flash=Math.max(0,c.flash-dt);}
  for(let i=parts.length-1;i>=0;i--){const p=parts[i];p.life-=dt;
    if(p.life<=0){parts.splice(i,1);continue}
    p.vy+=p.g*dt;p.x+=p.vx*dt;p.y+=p.vy*dt;}
  for(let i=pops.length-1;i>=0;i--){const p=pops[i];p.t-=dt/0.9;if(p.t<=0)pops.splice(i,1);}
  if(state==='play'||state==='menu'){trail.unshift({x:ball.x,y:ball.y});if(trail.length>16)trail.pop();}
  for(const d of dust){d.x+=d.vx*dt;d.y+=d.vy*dt;
    if(d.y>H+4)d.y=-4;if(d.x<-4)d.x=W+4;if(d.x>W+4)d.x=-4;}
  if(state==='menu'){
    if(crates.length&&crates.every(c=>c.solved)){menuResetT+=dt;if(menuResetT>1.5)loadLevel(0);}
    else menuResetT=0;
  }
}

/* ================= DRAW HELPERS ================= */
function rr(x,y,w,h,r){r=Math.min(r,w/2,h/2);ctx.beginPath();
  ctx.moveTo(x+r,y);ctx.arcTo(x+w,y,x+w,y+h,r);ctx.arcTo(x+w,y+h,x,y+h,r);
  ctx.arcTo(x,y+h,x,y,r);ctx.arcTo(x,y,x+w,y,r);ctx.closePath();}
function dia(cx,cy,s){ctx.beginPath();ctx.moveTo(cx,cy-s);ctx.lineTo(cx+s,cy);
  ctx.lineTo(cx,cy+s);ctx.lineTo(cx-s,cy);ctx.closePath();}

function render(){
  const c=cell;
  ctx.fillStyle='#0d1226';ctx.fillRect(0,0,W,H);
  let g=ctx.createRadialGradient(W*.15,H*.18,0,W*.15,H*.18,Math.max(W,H)*.45);
  g.addColorStop(0,'rgba(61,220,151,.05)');g.addColorStop(1,'rgba(0,0,0,0)');
  ctx.fillStyle=g;ctx.fillRect(0,0,W,H);
  g=ctx.createRadialGradient(W*.85,H*.8,0,W*.85,H*.8,Math.max(W,H)*.45);
  g.addColorStop(0,'rgba(255,92,92,.05)');g.addColorStop(1,'rgba(0,0,0,0)');
  ctx.fillStyle=g;ctx.fillRect(0,0,W,H);
  for(const d of dust){ctx.globalAlpha=d.a;ctx.fillStyle='#9fb4ff';ctx.fillRect(d.x,d.y,d.s,d.s);}
  ctx.globalAlpha=1;

  const shx=(Math.random()*2-1)*shake*c*.06,shy=(Math.random()*2-1)*shake*c*.06;
  ctx.save();ctx.translate(shx,shy);

  // board shadow + floor
  ctx.fillStyle='rgba(0,0,0,.4)';rr(bx-8,by+9,cols*c+16,rowsN*c+10,c*.15);ctx.fill();

  for(let j=0;j<=rowsN;j++){ctx.moveTo(bx,Y(j)+.5);ctx.lineTo(X(cols),Y(j)+.5);}
  ctx.stroke();
  // open-left lane
  ctx.fillStyle='rgba(255,92,92,.045)';ctx.fillRect(bx,by,c,rowsN*c);
  ctx.strokeStyle='rgba(255,179,71,.3)';ctx.lineWidth=2;const s=c*.16;
  for(let k=-1;k<=1;k++){const cxp=X(.5),cyp=Y(rowsN/2+k*c*1.5);
    ctx.beginPath();ctx.moveTo(cxp-s,cyp-s);ctx.lineTo(cxp+s*.4,cyp);ctx.lineTo(cxp-s,cyp+s);ctx.stroke();}

  // ball light
  g=ctx.createRadialGradient(X(ball.x),Y(ball.y),0,X(ball.x),Y(ball.y),c*3.4);
  g.addColorStop(0,'rgba(255,190,90,.13)');g.addColorStop(1,'rgba(0,0,0,0)');
  ctx.fillStyle=g;ctx.fillRect(bx,by,cols*c,rowsN*c);

  // pads
  for(const p of pads){
    const filled=crates.some(o=>o.solved&&o.cx===p[0]&&o.cy===p[1]);
    const cx=X(p[0]+.5),cy=Y(p[1]+.5),pulse=Math.sin(T*3+p[0]*1.7+p[1])*0.5+.5;
    if(filled){ctx.save();ctx.shadowColor='rgba(61,220,151,.8)';ctx.shadowBlur=c*.4;
      ctx.fillStyle='rgba(61,220,151,.85)';dia(cx,cy,c*.3);ctx.fill();
      ctx.fillStyle='#eafff5';dia(cx,cy,c*.12);ctx.fill();ctx.restore();}
    else{ctx.strokeStyle=`rgba(61,220,151,${.3+.35*pulse})`;ctx.lineWidth=2;
      ctx.setLineDash([c*.16,c*.12]);ctx.lineDashOffset=-T*c*.9;
      ctx.beginPath();ctx.arc(cx,cy,c*.34,0,7);ctx.stroke();ctx.setLineDash([]);
      ctx.fillStyle=`rgba(61,220,151,${.1+.14*pulse})`;dia(cx,cy,c*.2);ctx.fill();}
  }

  drawTrajectory(c);

  // walls
  for(const w of walls){const x=X(w[0]),y=Y(w[1]);
    ctx.fillStyle='#252f56';ctx.fillRect(x+1,y+1,c-2,c-2);
    ctx.fillStyle='rgba(140,170,255,.18)';ctx.fillRect(x+1,y+1,c-2,3);
    ctx.fillStyle='rgba(0,0,0,.35)';ctx.fillRect(x+1,y+c-4,c-2,3);
    ctx.fillStyle='#3a4a8c';ctx.beginPath();ctx.arc(x+c/2,y+c/2,c*.07,0,7);ctx.fill();}

  // crates
  for(const cr of crates){
    const px=X(cr.px),py=Y(cr.py),sq=cr.squash;
    const w=c*(1+(1-sq)*.5)*.9,h=c*sq*.9,x=px+(c-w)/2,y=py+c-h;
    ctx.fillStyle='rgba(0,0,0,.35)';ctx.fillRect(px+c*.08,py+c-c*.07,w,c*.06);
    const gg=ctx.createLinearGradient(0,y,0,y+h);
    if(cr.solved){gg.addColorStop(0,'#7ff0c0');gg.addColorStop(1,'#2fbf85');}
    else{gg.addColorStop(0,'#f2b25c');gg.addColorStop(1,'#c9832e');}
    ctx.save();
    if(cr.solved){ctx.shadowColor='rgba(61,220,151,.7)';ctx.shadowBlur=c*.35;}
    ctx.fillStyle=gg;rr(x,y,w,h,c*.1);ctx.fill();ctx.restore();
    ctx.strokeStyle='rgba(0,0,0,.3)';ctx.lineWidth=2;rr(x,y,w,h,c*.1);ctx.stroke();
    ctx.fillStyle='rgba(255,255,255,.25)';ctx.fillRect(x+3,y+3,w-6,3);
    ctx.strokeStyle=cr.solved?'rgba(0,80,50,.4)':'rgba(90,50,10,.4)';ctx.lineWidth=2;
    ctx.beginPath();ctx.moveTo(x+w*.4,y+h*.3);ctx.lineTo(x+w*.4,y+h*.7);
    ctx.moveTo(x+w*.6,y+h*.3);ctx.lineTo(x+w*.6,y+h*.7);ctx.stroke();
    if(cr.flash>0){ctx.strokeStyle=`rgba(255,92,92,${cr.flash/.22})`;ctx.lineWidth=3;rr(x-1,y-1,w+2,h+2,c*.12);ctx.stroke();}
  }

  // paddle
  {const pw=PW*c*(1-paddle.hitT*.5),phh=PH*c,px=X(FACE_X-PW),py=Y(paddle.y)-phh/2;
   if(Math.abs(paddle.vy)>4){ctx.fillStyle='rgba(255,179,71,.12)';
     ctx.fillRect(px,py-Math.sign(paddle.vy)*c*.3,pw,phh);}
   ctx.save();ctx.shadowColor='rgba(255,179,71,.9)';ctx.shadowBlur=14;
   const pg=ctx.createLinearGradient(px,0,px+pw,0);pg.addColorStop(0,'#ffd23f');pg.addColorStop(1,'#ff5c5c');
   ctx.fillStyle=pg;rr(px,py,pw,phh,c*.1);ctx.fill();ctx.restore();
   ctx.fillStyle='rgba(255,255,255,.7)';ctx.fillRect(px+pw*.42,py+3,2,phh-6);
   ctx.fillStyle='#fff2cf';ctx.fillRect(px,py,4,5);ctx.fillRect(px,py+phh-5,4,5);}

  // trail + ball
  for(let i=trail.length-1;i>=0;i--){const f=1-i/trail.length;
    ctx.fillStyle=`rgba(255,179,71,${.3*f})`;
    ctx.beginPath();ctx.arc(X(trail[i].x),Y(trail[i].y),BALL_R*c*(.4+.6*f),0,7);ctx.fill();}
  {const bxp=X(ball.x),byp=Y(ball.y),r=BALL_R*c;
   ctx.save();ctx.shadowColor='rgba(255,179,71,.9)';ctx.shadowBlur=c*.5;
   const bg=ctx.createRadialGradient(bxp-r*.35,byp-r*.35,c*.05,bxp,byp,r);
   bg.addColorStop(0,'#fff');bg.addColorStop(.4,'#ffd23f');bg.addColorStop(1,'#ff6a4d');
   ctx.fillStyle=bg;ctx.beginPath();ctx.arc(bxp,byp,r,0,7);ctx.fill();ctx.restore();}

  // particles + popups
  for(const p of parts){const a=p.life/p.max;ctx.globalAlpha=a;ctx.fillStyle=p.col;
    const sz=p.s*(.3+.7*a);ctx.fillRect(X(p.x)-sz/2,Y(p.y)-sz/2,sz,sz);}
  ctx.globalAlpha=1;
  if(pops.length){ctx.textAlign='center';ctx.font=`700 ${Math.max(13,c*.42)}px ${MONO}`;
    for(const p of pops){const yy=Y(p.y)-(1-p.t)*c*.9;ctx.globalAlpha=p.t;
      ctx.lineWidth=4;ctx.strokeStyle='rgba(0,0,0,.6)';ctx.strokeText(p.txt,X(p.x),yy);
      ctx.fillStyle=p.col;ctx.fillText(p.txt,X(p.x),yy);}ctx.globalAlpha=1;}

  // frame (left side open)
  const f=Math.max(3,c*.14);
  ctx.strokeStyle='#2c3a6e';ctx.lineWidth=f;ctx.lineCap='square';ctx.beginPath();
  ctx.moveTo(bx-f*.5,by-f*.5);ctx.lineTo(X(cols)+f*.5,by-f*.5);
  ctx.moveTo(X(cols)+f*.5,by-f*.5);ctx.lineTo(X(cols)+f*.5,Y(rowsN)+f*.5);
  ctx.moveTo(bx-f*.5,Y(rowsN)+f*.5);ctx.lineTo(X(cols)+f*.5,Y(rowsN)+f*.5);
  ctx.stroke();
  ctx.setLineDash([c*.2,c*.16]);ctx.strokeStyle='rgba(255,92,92,.3)';ctx.lineWidth=2;
  ctx.beginPath();ctx.moveTo(bx-f*.4,by);ctx.lineTo(bx-f*.4,Y(rowsN));ctx.stroke();ctx.setLineDash([]);
  ctx.strokeStyle='rgba(255,179,71,.5)';ctx.lineWidth=2;const L=c*.35;
  [[bx,by,1,1],[X(cols),by,-1,1],[bx,Y(rowsN),1,-1],[X(cols),Y(rowsN),-1,-1]].forEach(k=>{
    ctx.beginPath();ctx.moveTo(k[0]+k[2]*L,k[1]);ctx.lineTo(k[0],k[1]);ctx.lineTo(k[0],k[1]+k[3]*L);ctx.stroke();});

  ctx.restore();
  if(dropFlash>0){g=ctx.createRadialGradient(W/2,H/2,Math.min(W,H)*.3,W/2,H/2,Math.max(W,H)*.75);
    g.addColorStop(0,'rgba(0,0,0,0)');g.addColorStop(1,`rgba(255,60,60,${.4*dropFlash})`);
    ctx.fillStyle=g;ctx.fillRect(0,0,W,H);}
}

function drawTrajectory(c){
  if(state!=='play')return;
  const sp=Math.hypot(ball.vx,ball.vy)||1,dx=ball.vx/sp,dy=ball.vy/sp;
  let x=ball.x,y=ball.y;const step=.16,pts=[[x,y]];let end=null;
  for(let i=0;i<32;i++){
    x+=dx*step;y+=dy*step;pts.push([x,y]);
    if(y<BALL_R||y>rowsN-BALL_R||x>cols-BALL_R){end={t:'wall'};break;}
