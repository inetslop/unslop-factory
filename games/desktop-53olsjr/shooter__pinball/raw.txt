  for(const f of [fl,fr]){ const tgt=f.down?f.up:f.rest; const spd=f.down?20:11; let d=tgt-f.ang; const m=Math.min(Math.abs(d), spd*dt); f.ang+=Math.sign(d)*m; f.om=m/dt || 0; }

/* ================= PHYSICS ================= */
function updFlippers(h){
  for(const f of [fl,fr]){
    const tgt=f.down?f.up:f.rest,spd=f.down?20:11;
    const d=tgt-f.ang,m=Math.min(Math.abs(d),spd*h);
    f.ang+=Math.sign(d)*m;f.om=h>0?m/h:0;
  }
}
function collideSeg(b,s){
  const dx=s.x2-s.x1,dy=s.y2-s.y1,L2=dx*dx+dy*dy||1e-6;
  let t=((b.x-s.x1)*dx+(b.y-s.y1)*dy)/L2;t=t<0?0:t>1?1:t;
  const qx=s.x1+t*dx,qy=s.y1+t*dy;
  let nx=b.x-qx,ny=b.y-qy;const d2=nx*nx+ny*ny,rr=BR;
  if(d2>=rr*rr)return;
  const d=Math.sqrt(d2)||1e-6;nx/=d;ny/=d;
  b.x+=nx*(rr-d);b.y+=ny*(rr-d);
  const vn=b.vx*nx+b.vy*ny;
  if(vn<0){
    const e=vn<-120?s.e:0;
    b.vx-=(1+e)*vn*nx;b.vy-=(1+e)*vn*ny;
    if(live&&vn<-140)sfx.bounce(Math.min(1,-vn/900));
    if(s.boost&&vn<-80){
      const vn2=b.vx*nx+b.vy*ny;
      b.vx+=nx*(s.boost-vn2);b.vy+=ny*(s.boost-vn2);
      if(live){addScore(25,b.x,b.y,'#9fe8ff');sparks(b.x,b.y,'#9fe8ff',6,200);sfx.sling();}
    }
  }
}
function flipCollide(b,f){
  const tx=f.x+Math.cos(f.ang)*f.len,ty=f.y+Math.sin(f.ang)*f.len;
  const dx=tx-f.x,dy=ty-f.y,L2=dx*dx+dy*dy||1e-6;
  let t=((b.x-f.x)*dx+(b.y-f.y)*dy)/L2;t=t<0?0:t>1?1:t;
  const qx=f.x+t*dx,qy=f.y+t*dy;
  let nx=b.x-qx,ny=b.y-qy;const d=Math.hypot(nx,ny),rr=BR+FR;
  if(d>=rr||d<1e-6)return;
  nx/=d;ny/=d;b.x+=nx*(rr-d);b.y+=ny*(rr-d);
  const rx=qx-f.x,ry=qy-f.y,svx=-f.om*ry,svy=f.om*rx;
  const vn=(b.vx-svx)*nx+(b.vy-svy)*ny;
  if(vn<0){
    const e=vn<-140?.35:0;
    b.vx-=(1+e)*vn*nx;b.vy-=(1+e)*vn*ny;
    const sp=Math.hypot(b.vx,b.vy);if(sp>1450){b.vx*=1450/sp;b.vy*=1450/sp;}
    if(live&&vn<-160)sfx.bounce(Math.min(1,-vn/800));
  }
}
let live=false;
function physMove(h,b){
  b.vy+=GRAV*h;
  const sp0=Math.hypot(b.vx,b.vy);if(sp0>MAXV){b.vx*=MAXV/sp0;b.vy*=MAXV/sp0;}
  b.x+=b.vx*h;b.y+=b.vy*h;
  if(b.x<16+BR){b.x=16+BR;b.vx=Math.abs(b.vx)*.5;}
  if(b.x>464-BR){b.x=464-BR;b.vx=-Math.abs(b.vx)*.5;}
  if(b.y<BR+2){b.y=BR+2;b.vy=Math.abs(b.vy)*.5;}
  for(const s of walls)collideSeg(b,s);
  flipCollide(b,fl);flipCollide(b,fr);
  // bumpers
  for(let i=0;i<3;i++){
    const bp=bumpers[i],dx=b.x-bp.x,dy=b.y-bp.y,R=BR+bp.r,d2=dx*dx+dy*dy;
    if(d2<R*R){
      const d=Math.sqrt(d2)||1e-6,nx=dx/d,ny=dy/d;
      b.x=bp.x+nx*R;b.y=bp.y+ny*R;
      const vn=b.vx*nx+b.vy*ny;
      if(vn<560){b.vx+=nx*(560-vn);b.vy+=ny*(560-vn);}
      bp.f=live?1:.4;
      if(live&&bp.cool<=0){bp.cool=.18;addScore(75,b.x,b.y,'#ffffff');sparks(b.x,b.y,bp.c,12,320);sfx.bumper(i);shake=Math.min(1,shake+.3);}
    }
  }
  // drop targets
  if(live)for(const tg of targets){
    if(!tg.alive)continue;
    const s={x1:tg.x,y1:tg.y1,x2:tg.x,y2:tg.y2,e:.8};
    const dx=0,dy=tg.y2-tg.y1,L2=dy*dy;
    let t=((b.x-s.x1)*dx+(b.y-s.y1)*dy)/L2;t=t<0?0:t>1?1:t;
    const qx=s.x1+t*dx,qy=s.y1+t*dy;
    let nx=b.x-qx,ny=b.y-qy;const d=Math.hypot(nx,ny);
    if(d<BR&&d>1e-6){
      nx/=d;ny/=d;b.x+=nx*(BR-d);b.y+=ny*(BR-d);
      const vn=b.vx*nx+b.vy*ny;
      if(vn<0){b.vx-=1.8*vn*nx;b.vy-=1.8*vn*ny;}
      if(vn<-60){
        tg.alive=false;addScore(200,tg.x,168,'#ffd27a');sparks(tg.x,170,'#ffd27a',16,300);sfx.target();
        if(targets.every(x=>!x.alive)){addScore(1000,216,168,'#ffffff');setBanner('TARGET BANK +1000');sfx.bank();targetTimer=2.6;}
      }
    }
  }
  // rollover lanes
  for(const l of lanes){
    l.cool=Math.max(0,l.cool-h);l.flash=Math.max(0,l.flash-h*2.5);
    const dx=b.x-l.x,dy=b.y-l.y;
    if(live&&!l.lit&&l.cool<=0&&dx*dx+dy*dy<(BR+9)*(BR+9)){
      l.lit=true;l.flash=1;l.cool=.6;addScore(50,l.x,l.y-14,'#a7ffb0');sfx.lane();
      if(lanes.every(x=>x.lit)){addScore(500,216,436,'#ffffff');setBanner('LANES LIT +500');sfx.bank();laneTimer=1.1;}
    }
  }
}
function step(dt){
  shake=Math.max(0,shake-dt*2.8);drainFlash=Math.max(0,drainFlash-dt*1.6);
  bumpers.forEach(b=>{b.f=Math.max(0,b.f-dt*3);b.cool=Math.max(0,b.cool-dt);});
  if(targetTimer>0){targetTimer-=dt;
    if(targetTimer<=0)targets.forEach(tg=>{tg.alive=true;sparks(tg.x,170,'#ffd27a',8,160);});}
  if(laneTimer>0){laneTimer-=dt;if(laneTimer<=0)lanes.forEach(l=>l.lit=false);}
  for(let i=parts.length-1;i>=0;i--){const p=parts[i];p.l-=dt*1.9;p.x+=p.vx*dt;p.y+=p.vy*dt;p.vy+=620*dt;if(p.l<=0)parts.splice(i,1);}
  for(let i=pops.length-1;i>=0;i--){const p=pops[i];p.l-=dt/0.95;p.y-=46*dt;if(p.l<=0)pops.splice(i,1);}
  if(banner.t>0)banner.t-=dt;

  const b=state==='menu'?ghost:ball;
  const sp=Math.hypot(b.vx,b.vy);
  const n=(state==='play'||state==='menu')?Math.max(1,Math.min(12,Math.ceil(sp*dt/6))):1;
  live=state==='play';
  for(let i=0;i<n;i++){const h=dt/n;updFlippers(h);
    if(state==='play'||state==='menu')physMove(h,b);}

  if(charging&&state==='ready'){power=Math.min(1,power+dt/.9);ball.x=440;ball.y=706+power*13;}

  if(state==='menu'&&ghost.y>FH+30){
    ghost.x=216;ghost.y=460;ghost.vx=Math.random()*440-220;ghost.vy=-380-Math.random()*160;
  }
  if(state==='play'){
    if(ball.y>FH+26){loseBall();}
    else{
      const sp2=Math.hypot(ball.vx,ball.vy);
      if(ball.x>416&&sp2<26){
        stillT+=dt;
        if(stillT>(entered?.9:.5)){if(entered)loseBall();else{spawnBall();state='ready';}}
      }else stillT=0;
      if(ball.x<410)entered=true;
      if(sp2<16)lowT+=dt;else lowT=0;
      if(lowT>2.5){ball.vy-=320;ball.vx+=(Math.random()-.5)*160;lowT=0;}
    }
  }
}

}
