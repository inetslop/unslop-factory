/* ============================== UPDATE ======================================== */
function update(dt){ perfT+=dt;
  updMotes(dt); updParts(dt);
  shake=Math.max(0,shake-dt*22);
  if(L){ for(const b of L.bars)b.a+=b.speed*dt;
    for(const s of L.springs){s.cd-=dt;s.anim=Math.max(0,s.anim-dt*4);} }
  introT-=dt; for(let i=msgs.length-1;i>=0;i--){msgs[i].t-=dt;if(msgs[i].t<=0)msgs.splice(i,1);}

  if(mode==='play'){
    if(phase==='fly'){ flyT+=dt; const n=5,h=dt/n;
      for(let i=0;i<n;i++)stepBall(h);
      checkOrbs(); checkGoal();
      if(mode!=='play')return; // won mid-step
      if(ball.y>VH+80||ball.x<-40||ball.x>VW+40) die('OFF THE SHEET');
      else{ const sp=Math.hypot(ball.vx,ball.vy);
        if(flyT>1&&sp<26)stallT+=dt; else stallT=0;
        if(stallT>1||flyT>30)die('BALL STALLED'); }
      trail.push({x:ball.x,y:ball.y}); if(trail.length>26)trail.shift();
    }else if(phase==='wait'){ waitT-=dt;
      if(waitT<=0){ if(shots>0){placeBall();phase='aim';msg('SHOTS LEFT · '+shots);}
        else{mode='fail';snd.die();showResult(false);} } }
  }else if(mode==='win'){ winT-=dt; if(winT<=0&&!resultShown)showResult(true); }
  else if(mode==='index'||mode==='end'){ demoT+=dt; const n=4,h=dt/n;
    for(let i=0;i<n;i++)stepBall(h);
    trail.push({x:ball.x,y:ball.y}); if(trail.length>26)trail.shift();
    const sp=Math.hypot(ball.vx,ball.vy);
    if(ball.y>VH+60){placeBall();demoT=1.4;}
    if(demoT>2&&sp<90){ ball.vx=(Math.random()*.5+.35)*760; ball.vy=-(620+Math.random()*380); demoT=0; snd.launch(); } } }

/* ============================== RENDER ======================================== */
function ls(v){ try{ctx.letterSpacing=v;}catch(e){} }
function diamond(x,y,s,fill,col){ ctx.beginPath();ctx.moveTo(x,y-s);ctx.lineTo(x+s,y);ctx.lineTo(x,y+s);ctx.lineTo(x-s,y);ctx.closePath();
  if(fill){ctx.fillStyle=col;ctx.fill();}else{ctx.strokeStyle=col;ctx.lineWidth=1.4;ctx.stroke();} }

function render(){
  ctx.setTransform(DPR,0,0,DPR,0,0);
  ctx.fillStyle='#0a2440'; ctx.fillRect(0,0,W,H);
  const shx=(Math.random()-.5)*shake, shy=(Math.random()-.5)*shake;
  ctx.setTransform(DPR*sc,0,0,DPR*sc,DPR*(ox+shx),DPR*(oy+shy));

  /* --- board: paper, grid, frame --- */
  ctx.fillStyle='#123a63'; ctx.fillRect(0,0,VW,VH);
  let gg=ctx.createRadialGradient(VW*.5,VH*.42,60,VW*.5,VH*.5,VW*.72);
  gg.addColorStop(0,'rgba(130,190,245,.10)');gg.addColorStop(1,'rgba(6,20,40,.28)');
  ctx.fillStyle=gg; ctx.fillRect(0,0,VW,VH);
  ctx.lineWidth=1; ctx.beginPath();
  for(let x=20;x<VW;x+=20){ctx.moveTo(x,0);ctx.lineTo(x,VH);}
  for(let y=20;y<VH;y+=20){ctx.moveTo(0,y);ctx.lineTo(VW,y);}
  ctx.strokeStyle='rgba(170,210,250,.06)'; ctx.stroke();
  ctx.beginPath();
  for(let x=100;x<VW;x+=100){ctx.moveTo(x,0);ctx.lineTo(x,VH);}
  for(let y=100;y<VH;y+=100){ctx.moveTo(0,y);ctx.lineTo(VW,y);}
  ctx.strokeStyle='rgba(170,210,250,.13)'; ctx.stroke();
  ctx.strokeStyle='rgba(220,238,255,.4)'; ctx.strokeRect(6,6,VW-12,VH-12);
  ctx.strokeStyle='rgba(220,238,255,.18)'; ctx.strokeRect(11,11,VW-22,VH-22);

  /* --- ambient motes --- */
  for(const m of motes){ ctx.globalAlpha=.05+.05*Math.sin(perfT*1.3+m.ph);
    ctx.fillStyle='#cfe8ff'; ctx.beginPath();ctx.arc(m.x,m.y,m.r,0,TAU);ctx.fill(); }
  ctx.globalAlpha=1;

  /* --- wind streaks --- */
  if(L&&L.wind[0]){ const dir=Math.sign(L.wind[0]);
    for(let i=0;i<16;i++){ const x=((i*173+perfT*L.wind[0]*.5)%(VW+80)+VW+80)%(VW+80)-40;
      const y=24+((i*i*53)%500); ctx.globalAlpha=.13; ctx.strokeStyle='#cfe8ff'; ctx.lineWidth=1.2;
      ctx.beginPath();ctx.moveTo(x,y);ctx.lineTo(x-dir*26,y);ctx.stroke(); } ctx.globalAlpha=1; }

  if(L){
    /* goal beacon + ring */
    const g=L.goal;
    if(g){ if(allGot){ const bg=ctx.createLinearGradient(0,g.y,0,g.y-300);
        bg.addColorStop(0,'rgba(255,180,94,.16)');bg.addColorStop(1,'rgba(255,180,94,0)');
        ctx.fillStyle=bg;ctx.fillRect(g.x-34,g.y-300,68,300); }
      const col=allGot?'#ffc678':'#cfe8ff';
      ctx.save();ctx.translate(g.x,g.y);ctx.rotate(perfT*(allGot?1.5:.5));
      ctx.setLineDash([9,7]);ctx.lineWidth=2;ctx.strokeStyle=col;ctx.globalAlpha=.9;
      ctx.beginPath();ctx.arc(0,0,g.r+(allGot?Math.sin(perfT*6)*2:0),0,TAU);ctx.stroke();
      ctx.restore();ctx.setLineDash([]);ctx.globalAlpha=.5;ctx.strokeStyle=col;ctx.lineWidth=1.4;
      ctx.beginPath();ctx.moveTo(g.x-g.r*.6,g.y);ctx.lineTo(g.x+g.r*.6,g.y);
      ctx.moveTo(g.x,g.y-g.r*.6);ctx.lineTo(g.x,g.y+g.r*.6);ctx.stroke();ctx.globalAlpha=1; }

    /* solids */
    for(const s of L.solids){ if(s.x+s.w<0||s.x>VW)continue;
      ctx.fillStyle=s.ice?'rgba(150,230,255,.10)':'rgba(255,255,255,.06)';
      ctx.fillRect(s.x,s.y,s.w,s.h);
      ctx.fillStyle=hatch; ctx.fillRect(s.x,s.y,s.w,s.h);
      ctx.strokeStyle='rgba(220,238,255,.8)';ctx.lineWidth=1.6;ctx.strokeRect(s.x,s.y,s.w,s.h);
      if(s.ice){ctx.strokeStyle='rgba(180,240,255,.9)';ctx.lineWidth=2;
        ctx.beginPath();ctx.moveTo(s.x,s.y+1);ctx.lineTo(s.x+s.w,s.y+1);ctx.stroke();} }

    /* springs */
    for(const s of L.springs){ const cx=s.x+s.w/2,hot=s.cd>0;
      ctx.fillStyle=hot?'rgba(255,180,94,.3)':'rgba(255,255,255,.08)';ctx.fillRect(s.x,s.y,s.w,s.h);
      ctx.fillStyle=hatch;ctx.fillRect(s.x,s.y,s.w,s.h);
      ctx.strokeStyle=hot?'#ffc678':'rgba(220,238,255,.8)';ctx.lineWidth=1.6;ctx.strokeRect(s.x,s.y,s.w,s.h);
      const o=(perfT*42)%12, sq=s.anim>0?.4:1;
      for(let k=0;k<3;k++){ const y=s.y-7-o-k*11, a=(.55-.16*k)*sq+.15;
        ctx.globalAlpha=a;ctx.strokeStyle='#ffb45e';ctx.lineWidth=2.4;
        ctx.beginPath();ctx.moveTo(cx-8,y+6*sq);ctx.lineTo(cx,y-3*sq);ctx.lineTo(cx+8,y+6*sq);ctx.stroke(); }
      ctx.globalAlpha=1; }

    /* bars */
    for(const b of L.bars){
      ctx.save();ctx.translate(b.cx,b.cy);ctx.rotate(b.a-b.speed*.06);
      ctx.globalAlpha=.12;ctx.fillStyle='#cfe8ff';ctx.fillRect(-b.len/2,-b.th/2,b.len,b.th);
      ctx.restore();
      ctx.save();ctx.translate(b.cx,b.cy);ctx.rotate(b.a);
      ctx.fillStyle='rgba(255,255,255,.07)';ctx.fillRect(-b.len/2,-b.th/2,b.len,b.th);
      ctx.fillStyle=hatch;ctx.fillRect(-b.len/2,-b.th/2,b.len,b.th);
      ctx.strokeStyle='rgba(220,238,255,.85)';ctx.lineWidth=1.6;ctx.strokeRect(-b.len/2,-b.th/2,b.len,b.th);
      ctx.restore();
      ctx.fillStyle='#dceeff';ctx.beginPath();ctx.arc(b.cx,b.cy,4.5,0,TAU);ctx.fill();
      ctx.strokeStyle='rgba(18,42,72,.9)';ctx.lineWidth=1.4;ctx.beginPath();ctx.arc(b.cx,b.cy,1.6,0,TAU);ctx.stroke(); }

    /* orbs */
    L.orbs.forEach((o,i)=>{ if(o.got){ ctx.globalAlpha=.22; diamond(o.x,o.y,6,false,'#8fe8ff'); ctx.globalAlpha=1; return; }
      const pu=13+2*Math.sin(perfT*3+i*2);
      let og=ctx.createRadialGradient(o.x,o.y,0,o.x,o.y,17);
      og.addColorStop(0,'rgba(143,232,255,.35)');og.addColorStop(1,'rgba(143,232,255,0)');
      ctx.fillStyle=og;ctx.beginPath();ctx.arc(o.x,o.y,17,0,TAU);ctx.fill();
      diamond(o.x,o.y,6.5,true,'#bff3ff');
      ctx.setLineDash([4,5]);ctx.globalAlpha=.4;ctx.strokeStyle='#8fe8ff';ctx.lineWidth=1.2;
      ctx.beginPath();ctx.arc(o.x,o.y,pu,0,TAU);ctx.stroke();ctx.setLineDash([]);ctx.globalAlpha=1;
      const a=perfT*2.4+i*2.1; ctx.fillStyle='#fff';
      ctx.beginPath();ctx.arc(o.x+Math.cos(a)*pu,o.y+Math.sin(a)*pu,1.8,0,TAU);ctx.fill(); });

    /* notes */
    if(L.notes){ ctx.font='9px "Courier New",monospace'; ls('2px'); ctx.fillStyle='rgba(220,238,255,.45)';
      ctx.textAlign='left'; for(const n of L.notes){ ctx.fillText('+ '+n[2],n[0],n[1]); } ls('0px'); ctx.textAlign='left'; }

    /* trail + ball */
    const sp=Math.hypot(ball.vx,ball.vy);
    if(trail.length>1&&sp>60){ for(let i=1;i<trail.length;i++){ const a=(i/trail.length)*.5;
        ctx.globalAlpha=a;ctx.strokeStyle='#ffb45e';ctx.lineWidth=1+(i/trail.length)*5;ctx.lineCap='round';
        ctx.beginPath();ctx.moveTo(trail[i-1].x,trail[i-1].y);ctx.lineTo(trail[i].x,trail[i].y);ctx.stroke(); } ctx.globalAlpha=1; }
    let bg2=ctx.createRadialGradient(ball.x,ball.y,0,ball.x,ball.y,BR*3.2);
    bg2.addColorStop(0,'rgba(255,180,94,.32)');bg2.addColorStop(1,'rgba(255,180,94,0)');
    ctx.fillStyle=bg2;ctx.beginPath();ctx.arc(ball.x,ball.y,BR*3.2,0,TAU);ctx.fill();
    let bb=ctx.createRadialGradient(ball.x-3,ball.y-4,1,ball.x,ball.y,BR+1);
    bb.addColorStop(0,'#ffe9c8');bb.addColorStop(.55,'#ffb45e');bb.addColorStop(1,'#e88a2e');
    ctx.fillStyle=bb;ctx.beginPath();ctx.arc(ball.x,ball.y,BR,0,TAU);ctx.fill();
    ctx.strokeStyle='rgba(90,45,10,.55)';ctx.lineWidth=1.2;ctx.stroke();
    ctx.globalAlpha=.8;ctx.strokeStyle='#fff';ctx.lineWidth=1.6;
    ctx.beginPath();ctx.arc(ball.x-3,ball.y-3.5,BR*.5,Math.PI*1.05,Math.PI*1.7);ctx.stroke();ctx.globalAlpha=1;

    /* idle "pull" prompt */
    if(mode==='play'&&phase==='aim'&&!aim){ const pu=20+2*Math.sin(perfT*3.4);
      ctx.setLineDash([5,6]);ctx.globalAlpha=.55;ctx.strokeStyle='#ffc678';ctx.lineWidth=1.4;
      ctx.beginPath();ctx.arc(ball.x,ball.y,pu,0,TAU);ctx.stroke();ctx.setLineDash([]);ctx.globalAlpha=1;
      ctx.font='9px "Courier New",monospace';ls('3px');ctx.fillStyle='#ffc678';ctx.textAlign='center';
      ctx.fillText('PULL',ball.x,ball.y+pu+14);ls('0px');ctx.textAlign='left'; }

    /* aim UI */
    if(aim&&phase==='aim'&&mode==='play'){ let vx=(aim.sx-aim.cx)*4.2,vy=(aim.sy-aim.cy)*4.2;
      const spv=Math.hypot(vx,vy); if(spv>1){ const k=spv>VMAX?VMAX/spv:1; vx*=k;vy*=k; }
      const pwr=spv/VMAX, dxn=vx/(spv||1), dyn=vy/(spv||1), alen=26+70*pwr;

      /* trajectory preview */
      let px=ball.x,py=ball.y,pvx=vx,pvy=vy; const h=1/50,wx=L.wind[0];

      ctx.globalAlpha=1;
      if(spv>40){ ctx.font='11px "Courier New",monospace';ls('2px');
        ctx.fillStyle='#ffd9a0';ctx.textAlign='center';

    ctx.beginPath();ctx.moveTo(tbx,tby+19);ctx.lineTo(tbx+tbw,tby+19);ctx.stroke();
    ctx.font='700 10px "Courier New",monospace';ls('2px');ctx.fillStyle='#dceeff';
    ctx.fillText('SLING & SCALE',tbx+8,tby+13);
    ctx.font='8px "Courier New",monospace';ctx.fillStyle='rgba(220,238,255,.6)';

  ctx.fillStyle=vg;ctx.fillRect(0,0,W,H);

  if(mode!=='index'&&mode!=='end'){

    ctx.textAlign='left';ls('3px');
    ctx.font='11px "Courier New",monospace';ctx.fillStyle='rgba(220,238,255,.6)';
    ctx.fillText(`SHEET ${nn(cur+1)} / 07`,20,32); ls('0px');
    ctx.font='700 24px "Trebuchet MS",Verdana,sans-serif';ls('1px');ctx.fillStyle='#eaf4ff';
    ctx.fillText(L.name.toUpperCase(),20,58); ls('0px');
    ctx.font='10px "Courier New",monospace';ls('2px');ctx.fillStyle='rgba(220,238,255,.5)';
    let sub=`PAR ${L.par}`; if(L.wind[0])sub+=`  ·  WIND ${-L.wind[0]>0?'⟵':'⟶'} ${Math.abs(L.wind[0])} U/S`;

    /* top-right: shots + sparks */
    ctx.textAlign='right';ctx.font='10px "Courier New",monospace';ls('2px');ctx.fillStyle='rgba(220,238,255,.6)';
    const nS=L.shots,xS=W-24-(nS-1)*17;
    ctx.fillText('SHOTS',W-24-nS*17-8,29); ls('0px');
    for(let i=0;i<nS;i++){ const x=xS+i*17;
      if(i<shots){ctx.fillStyle='#ffb45e';ctx.beginPath();ctx.arc(x,25,5,0,TAU);ctx.fill();}
      else{ctx.strokeStyle='rgba(220,238,255,.4)';ctx.lineWidth=1.4;ctx.beginPath();ctx.arc(x,25,5,0,TAU);ctx.stroke();} }
    const nO=L.orbs.length,xO=W-24-(nO-1)*17;

    const hint=phase==='aim'?'DRAG & RELEASE TO LAUNCH — COLLECT SPARKS, REACH THE RING':
               phase==='fly'?'IN FLIGHT · READ THE ARC':'…';
    ctx.textAlign='left';ctx.fillText(hint,18,H-20); ls('0px');
    ctx.textAlign='right';ctx.fillStyle='rgba(220,238,255,.45)';
    ctx.fillText('[R] RESET · [N] NEXT · [M] INDEX',W-16,H-40);
    ctx.fillStyle=muted?'rgba(255,122,92,.8)':'rgba(255,198,120,.85)';
    ctx.fillText('[S] SOUND '+(muted?'OFF':'ON'),W-16,H-20); ls('0px'); ctx.textAlign='left';

    /* intro stamp */
    if(introT>0&&mode==='play'){ const t=1.6-introT;
      const a=Math.min(t/.15,1)*Math.min(1,introT/.4), s2=1+(1-Math.min(t/.15,1))*.07;
      ctx.save();ctx.translate(W/2,H*.34);ctx.scale(s2,s2);ctx.globalAlpha=a;
