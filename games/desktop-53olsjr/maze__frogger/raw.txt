/* ================= MAZE ================= */
function genMaze(c,r){
  const cells=[];for(let y=0;y<r;y++){const row=[];for(let x=0;x<c;x++)row.push([1,1,1,1]);cells.push(row);}
  const D=[[0,-1,0],[1,0,1],[0,1,2],[-1,0,3]];
  let cx=c>>1,cy=r-1;const seen=new Uint8Array(c*r);seen[cy*c+cx]=1;const st=[];
  for(;;){ const opts=[];
    for(const d of D){const nx=cx+d[0],ny=cy+d[1];if(nx>=0&&ny>=0&&nx<c&&ny<r&&!seen[ny*c+nx])opts.push([nx,ny,d[2]]);}
    if(!opts.length){ if(!st.length)break; const p=st.pop();cx=p[0];cy=p[1];continue;}
    const o=opts[(Math.random()*opts.length)|0];
    cells[cy][cx][o[2]]=0; cells[o[1]][o[0]][(o[2]+2)%4]=0; seen[o[1]*c+o[0]]=1; st.push([cx,cy]); cx=o[0]; cy=o[1]; }
  return {cols:c,rows:r,cells};
}
function findLanes(m){
  const res=[];
  for(let y=0;y<m.rows;y++){let x=0;while(x<m.cols){let e=1;while(x+e<m.cols&&m.cells[y][x+e-1][1]===0)e++;
    if(e>=4)res.push({axis:'h',fixed:y,x0:x,y0:y,len:e});x+=Math.max(1,e);}}
  for(let x=0;x<m.cols;x++){let y=0;while(y<m.rows){let s=1;while(y+s<m.rows&&m.cells[y+s-1][x][2]===0)s++;
    if(s>=4)res.push({axis:'v',fixed:x,x0:x,y0:y,len:s});y+=Math.max(1,s);}}
  return res;
}

function computeLayout(){ if(!maze)return;

function buildStatic(){

    if(r<.16){c.fillStyle='rgba(150,255,180,.05)';c.fillRect(x*cell+(r*9)%1*cell*.8,y*cell+((r*53)%1)*cell*.8,2,2);} }
  c.strokeStyle='rgba(140,255,170,.035)';c.lineWidth=1;c.beginPath();
  for(let x=1;x<cols;x++){c.moveTo(x*cell+.5,0);c.lineTo(x*cell+.5,h);}
  for(let y=1;y<rows;y++){c.moveTo(0,y*cell+.5);c.lineTo(w,y*cell+.5);} c.stroke();
  c.strokeStyle='rgba(170,255,190,.14)';c.lineWidth=Math.max(2,cell*.06);
  c.beginPath();c.arc((startCX+.5)*cell,(startCY+.5)*cell,cell*.34,0,TWO);c.stroke();
  const P=new Path2D(),add=(x1,y1,x2,y2)=>{P.moveTo(x1,y1);P.lineTo(x2,y2);};
  for(let y=0;y<rows;y++)for(let x=0;x<cols;x++){const wl=maze.cells[y][x];
    if(wl[0])add(x*cell,y*cell,(x+1)*cell,y*cell);
    if(wl[3])add(x*cell,y*cell,x*cell,(y+1)*cell);
    if(x===cols-1&&wl[1])add((x+1)*cell,y*cell,(x+1)*cell,(y+1)*cell);
    if(y===rows-1&&wl[2])add(x*cell,(y+1)*cell,(x+1)*cell,(y+1)*cell);}
  const wt=Math.max(5,cell*.16); c.lineCap='round';c.lineJoin='round';
  c.strokeStyle='rgba(120,255,170,.12)';c.lineWidth=wt+5;c.stroke(P);
  c.strokeStyle='#1a3a24';c.lineWidth=wt;c.stroke(P);
  c.strokeStyle='rgba(190,255,180,.22)';c.lineWidth=Math.max(1.5,wt*.32);c.stroke(P);
}

/* ================= INPUT / SIM ================= */
const DIRS=[[0,-1,0],[1,0,1],[0,1,2],[-1,0,3]];
function tryMove(dx,dy){ if(state!=='play'||paused)return; frog.face={x:dx,y:dy};
  if(frog.hop){queue=[dx,dy];return;}
  const d=DIRS.find(v=>v[0]===dx&&v[1]===dy)[2];
  if(maze.cells[frog.cy][frog.cx][d])return;
  frog.hop={fx:frog.cx,fy:frog.cy,tx:frog.cx+dx,ty:frog.cy+dy,t:0};sfx.hop(); }

function onLand(){ const f=flies.find(f=>!f.taken&&f.x===frog.cx&&f.y===frog.cy);
  if(f){f.taken=true;score+=50;updateScore();sfx.get();pops.push({x:frog.px,y:frog.py,txt:'+50',t:0,c:'#ffd166'});
    for(let i=0;i<8;i++)parts.push(spark(frog.px,frog.py,'#ffd166'));}
  if(frog.cx===exitX&&frog.cy===0)reachExit(); }

function updateFrog(dt){
  if(frog.blinkT>0){frog.blinkT-=dt;if(frog.blinkT<=0)frog.blink=.12;}
  if(frog.blink>0)frog.blink-=dt;
  frog.squash=Math.max(0,frog.squash-dt*5);
  if(!frog.hop){frog.px=(frog.cx+.5)*cell;frog.py=(frog.cy+.5)*cell;frog.air=0;return;}
  const h=frog.hop;h.t+=dt/HOP;
  if(h.t>=1){ frog.cx=h.tx;frog.cy=h.ty;frog.hop=null;frog.air=0;frog.squash=1;
    ripples.push({x:frog.px,y:frog.py,t:0});sfx.land();onLand();
    const q=queue;queue=null;if(q&&state==='play')tryMove(q[0],q[1]); }
  else{ const p=1-Math.pow(1-h.t,2);
    frog.px=(h.fx+(h.tx-h.fx)*p+.5)*cell;frog.py=(h.fy+(h.ty-h.fy)*p+.5)*cell;
    frog.air=Math.sin(Math.PI*h.t);} }

function snakeXY(sn,idx){ return sn.axis==='h'
  ? [ox+(sn.x0+idx+.5)*cell,oy+(sn.fixed+.5)*cell]
  : [ox+(sn.fixed+.5)*cell,oy+(sn.y0+idx+.5)*cell]; }
function updateSnakes(dt){ for(const sn of snakes){ sn.pos+=sn.dir*sn.sp*dt;
  if(sn.pos>sn.len-1){sn.pos=sn.len-1;sn.dir=-1;}else if(sn.pos<0){sn.pos=0;sn.dir=1;} } }
function checkCollisions(){ if(gameT<frog.invUntil)return;
  for(const sn of snakes)for(let k=0;k<sn.segs;k++){
    const idx=sn.pos-sn.dir*k*SN_SP;if(idx<-.2||idx>sn.len-1+.2)continue;
    const [sx,sy]=snakeXY(sn,idx),dx=sx-frog.px,dy=sy-frog.py,r=k===0?cell*.4:cell*.36;
    if(dx*dx+dy*dy<r*r){killFrog();return;} } }

/* particles */
function puff(x,y){const a=Math.random()*TWO,s=20+Math.random()*50;
  return{x,y,vx:Math.cos(a)*s,vy:Math.sin(a)*s-24,r:cell*.13+Math.random()*cell*.1,t:0,life:.7+Math.random()*.3,c:'puff'};}
function spark(x,y,c){const a=Math.random()*TWO,s=40+Math.random()*90;
  return{x,y,vx:Math.cos(a)*s,vy:Math.sin(a)*s-30,r:1.5+Math.random()*2,t:0,life:.5,c};}

/* fireflies */
const flyB=[],flyF=[];
function mkFly(){return{x:Math.random()*VW,y:Math.random()*VH,a:Math.random()*TWO,sp:6+Math.random()*14,ph:Math.random()*7,s:.6+Math.random()*.9};}
function initFire(){for(let i=0;i<16;i++)flyB.push(mkFly());for(let i=0;i<7;i++)flyF.push(mkFly());}
function updFlies(dt){ for(const f of flyB.concat(flyF)){
  f.a+=(Math.random()-.5)*6*dt; f.x+=Math.cos(f.a)*f.sp*dt; f.y+=Math.sin(f.a)*f.sp*.7*dt+Math.sin(gameT*.7+f.ph)*4*dt;
  if(f.x<-20)f.x=VW+20;if(f.x>VW+20)f.x=-20;if(f.y<-20)f.y=VH+20;if(f.y>VH+20)f.y=-20;} }
function drawFlyLayer(layer,front){ for(const f of layer){
  const tw=.35+.65*(.5+.5*Math.sin(gameT*2+f.ph)),r=(front?2.4:1.7)*f.s;
  ctx.fillStyle=front?'rgba(200,255,170,'+(.1*tw)+')':'rgba(190,255,160,'+(.14*tw)+')';
  ctx.beginPath();ctx.arc(f.x,f.y,r*4,0,TWO);ctx.fill();
  ctx.fillStyle='rgba(235,255,190,'+(.85*tw)+')';ctx.beginPath();ctx.arc(f.x,f.y,r,0,TWO);ctx.fill();} }

/* ================= UPDATE ================= */
function update(dt){ gameT+=dt;stateT+=dt;shake=Math.max(0,shake-dt*2.6);updFlies(dt);
  for(let i=parts.length-1;i>=0;i--){const p=parts[i];p.t+=dt;if(p.t>p.life){parts.splice(i,1);continue;}
    p.x+=p.vx*dt;p.y+=p.vy*dt;p.vy+=(p.c==='puff'?-30:260)*dt;}
  for(let i=ripples.length-1;i>=0;i--){ripples[i].t+=dt;if(ripples[i].t>.55)ripples.splice(i,1);}
  for(let i=pops.length-1;i>=0;i--){pops[i].t+=dt;if(pops[i].t>1)pops.splice(i,1);}
  if(state==='menu'||state==='over'){updateSnakes(dt);return;}
  if(state==='intro'){updateSnakes(dt);if(stateT>1.5)setState('play');return;}
  if(state==='dying'){if(stateT>.95){lives<=0?gameOver():respawn();}return;}
  if(state==='clearing'){if(stateT>2)nextLevel();return;}
  updateFrog(dt);updateSnakes(dt);checkCollisions();
  if(state!=='play')return;
  time-=dt;updateTimeUI();
  if(time<=10.5&&time>0){const s=Math.ceil(time);if(s!==lastTick){lastTick=s;sfx.tick();}}
  if(time<=0)killFrog(); }

/* ================= RENDER ================= */
function drawPond(){ const x=ox+(exitX+.5)*cell,y=oy+cell*.5,p=.5+.5*Math.sin(gameT*2.3);
  let g=ctx.createRadialGradient(x,y,0,x,y,cell*(1.25+p*.3));
  g.addColorStop(0,'rgba(96,240,205,.3)');g.addColorStop(1,'rgba(96,240,205,0)');
  ctx.fillStyle=g;ctx.beginPath();ctx.arc(x,y,cell*(1.25+p*.3),0,TWO);ctx.fill();
