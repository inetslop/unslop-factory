function collideSeg(b,s){
  if(s.gate&&b.vy<=0)return null;
  const dx=s.x2-s.x1,dy=s.y2-s.y1,L2=dx*dx+dy*dy;
  let t=((b.x-s.x1)*dx+(b.y-s.y1)*dy)/L2; t=t<0?0:t>1?1:t;
  const cx=s.x1+t*dx,cy=s.y1+t*dy;
  let nx=b.x-cx,ny=b.y-cy; const d2=nx*nx+ny*ny;
  if(d2>b.r*b.r)return null;
  let d=Math.sqrt(d2)||0.0001; nx/=d; ny/=d;
  b.x=cx+nx*b.r; b.y=cy+ny*b.r;
  const vn0=b.vx*nx+b.vy*ny; let hit=false;
  if(vn0<0){ b.vx-=(1+s.e)*vn0*nx; b.vy-=(1+s.e)*vn0*ny; hit=true; }
  if(s.boost){
    const vn1=b.vx*nx+b.vy*ny;
    if(vn1<s.boost){ b.vx+=(s.boost-vn1)*nx; b.vy+=(s.boost-vn1)*ny; s.flash=0.2; return {boost:true,impact:-vn0}; }
  }
  return hit?{impact:-vn0}:null;
}
function collideRect(b,rx,ry,rw,rh){
  const cx=Math.max(rx,Math.min(b.x,rx+rw)), cy=Math.max(ry,Math.min(b.y,ry+rh));
  let nx=b.x-cx,ny=b.y-cy,d2=nx*nx+ny*ny;
  if(d2>b.r*b.r)return null;
  if(d2===0){
    const l=b.x-rx,r=rx+rw-b.x,t=b.y-ry,bo=ry+rh-b.y,m=Math.min(l,r,t,bo);
    if(m===l){nx=-1;ny=0;b.x=rx-b.r;} else if(m===r){nx=1;ny=0;b.x=rx+rw+b.r;}
    else if(m===t){nx=0;ny=-1;b.y=ry-b.r;} else {nx=0;ny=1;b.y=ry+rh+b.r;}
  } else { const d=Math.sqrt(d2); nx/=d; ny/=d; b.x=cx+nx*b.r; b.y=cy+ny*b.r; }
  const vn0=b.vx*nx+b.vy*ny; let impact=0;
  if(vn0<0){ b.vx-=1.45*vn0*nx; b.vy-=1.45*vn0*ny; impact=-vn0; }
  return {impact};
}
function hitBumper(b,bp){
  const R=bp.r+BALL_R; let nx=b.x-bp.x,ny=b.y-bp.y; const d2=nx*nx+ny*ny;
  if(d2>R*R)return false;
  const d=Math.sqrt(d2)||0.001; nx/=d; ny/=d;
  b.x=bp.x+nx*R; b.y=bp.y+ny*R;
  const vtn=b.vx*nx+b.vy*ny, tvx=b.vx-vtn*nx, tvy=b.vy-vtn*ny;
  const kick=620+(vtn<0?-vtn*0.25:0);
  b.vx=nx*kick+tvx*0.35; b.vy=ny*kick+tvy*0.35;
  if(bp.cd<=0){ bp.cd=0.15; bp.pulse=1; G.score+=50; pop(b.x,b.y-16,'+50',bp.c); sBumper(); }
  return true;
}
function collideFlipper(b,f){
  const ca=Math.cos(f.a),sa=Math.sin(f.a),tx=f.px+f.len*ca,ty=f.py+f.len*sa;
  const dx=tx-f.px,dy=ty-f.py,L2=dx*dx+dy*dy;
  let t=((b.x-f.px)*dx+(b.y-f.py)*dy)/L2; t=t<0?0:t>1?1:t;
  const cx=f.px+t*dx,cy=f.py+t*dy;
  let nx=b.x-cx,ny=b.y-cy; const d=Math.hypot(nx,ny), R=f.r+BALL_R-1.5;
  if(d>=R)return;
  if(d<0.0001){nx=0;ny=-1;} else {nx/=d;ny/=d;}
  b.x=cx+nx*R; b.y=cy+ny*R;
  const rx=cx-f.px,ry=cy-f.py, svx=f.w*(-ry), svy=f.w*rx;
  let rvx=b.vx-svx,rvy=b.vy-svy; const vn=rvx*nx+rvy*ny;
  if(vn<0){ rvx-=1.35*vn*nx; rvy-=1.35*vn*ny; }
  b.vx=svx+rvx; b.vy=svy+rvy;
  const sp=Math.hypot(b.vx,b.vy); if(sp>1650){ b.vx*=1650/sp; b.vy*=1650/sp; }
}
function damageCell(r,c,cell){
  cell.hp--;
  const cx=GX+c*CELL+17, cy=GY+r*CELL+17;
  if(cell.hp>0){ G.score+=100; pop(cx,cy,'+100','#ffffff'); burst(cx,cy,'#ffffff',4,160); sCrack(); }
  else {
    G.grid[r][c]=null; G.score+=250; pop(cx,cy-8,'+250',cell.color,17);
    burst(cx,cy,cell.color,12,300); G.shake=5; sShatter(); saveHi();
    let rr=r; while(rr>0&&!G.grid[rr-1][c]){ G.grid[rr][c]=G.grid[rr-1][c]; G.grid[rr-1][c]=null; rr--; }
  }
}

function launch(){
  if(G.dying||G.ball.phase!=='plunger')return;
  const p=G.charge.p;
  G.ball.vy=-(980+620*p); G.ball.vx=0; G.ball.phase='live'; G.ball.trail.length=0;
  sLaunch(); G.shake=Math.max(G.shake,2+p*3);
}
function loadPlunger(){

  G.charge={holding:false,p:0,dir:1};
}
function loseBall(){

  G.balls--; sDrain(); banner('BALL LOST','#ff5d5d');
  G.ball.phase='dead'; G.reloadT=0.8;
  if(G.balls<=0){ G.dying=true; G.overReason='OUT OF BALLS'; G.dieT=1.3; sOver(); saveHi(); }

/* ============================== UPDATE ============================== */
function update(dt){
  G.time+=dt;
  for(const f of FLIP){
    const target=f.active?f.act:f.rest, prev=f.a, step=14*dt;
    const d=target-f.a; f.a+=Math.abs(d)<=step?d:Math.sign(d)*step;
    f.w=(f.a-prev)/dt;
  }
  if(G.charge.holding){
    G.charge.p+=G.charge.dir*dt/0.85;
    if(G.charge.p>=1){G.charge.p=1;G.charge.dir=-1;}
    if(G.charge.p<=0){G.charge.p=0;G.charge.dir=1;}
  }
  const b=G.ball;
  if(b.phase==='plunger'){ b.x=453; b.y=690-(G.charge.holding?G.charge.p*14:0); }
  else if(b.phase==='live'){
    const n=6,h=dt/n;
    for(let i=0;i<n;i++){
      b.vy+=GRAV*h;
      const sp=Math.hypot(b.vx,b.vy); if(sp>1650){b.vx*=1650/sp;b.vy*=1650/sp;}
      b.x+=b.vx*h; b.y+=b.vy*h;
      for(const s of WALLS){
        const r=collideSeg(b,s);
        if(r){
          if(r.boost){ G.score+=75; pop(b.x,b.y-14,'+75','#b45cff'); sSlings(); G.shake=Math.max(G.shake,2); saveHi(); }
          else if(r.impact>280&&G.time-G.tickT>0.06){ G.tickT=G.time; sTick(); }
        }
      }
      for(const bp of BUMP)hitBumper(b,bp);
      for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++){
        const cell=G.grid[r][c]; if(!cell)continue;
        const res=collideRect(b,GX+c*CELL+1,GY+r*CELL+1,CELL-2,CELL-2);
        if(res&&res.impact>170&&G.time-cell.lastHit>0.16){ cell.lastHit=G.time; damageCell(r,c,cell); }
      }
      for(const f of FLIP)collideFlipper(b,f);
    }
    b.trail.push({x:b.x,y:b.y}); if(b.trail.length>14)b.trail.shift();
    if(b.y>VH+18)loseBall();
    const sp2=Math.hypot(b.vx,b.vy);
    G.stuckT=sp2<40?G.stuckT+dt:0;
    if(G.stuckT>1.3){ G.stuckT=0; b.vx+=(Math.random()-0.5)*170; b.vy-=120+Math.random()*90; }
  }
  else if(b.phase==='dead'&&!G.dying){ G.reloadT-=dt; if(G.reloadT<=0)loadPlunger(); }

  if(!G.dying){
    if(G.clearRows){ G.clearT-=dt; if(G.clearT<=0)finishClear(); }
    if(G.piece){
      G.piece.t+=dt; let iv=dropInt(), guard=0;
      while(G.piece&&G.piece.t>=iv&&guard++<24&&!G.dying){
        G.piece.t-=iv;
        if(canPlace(G.piece.row+1,G.piece.col))G.piece.row++; else lockPiece();
      }
    } else if(!G.clearRows){ G.spawnT-=dt; if(G.spawnT<=0)spawnPiece(); }
  }
  for(const bp of BUMP){ bp.cd-=dt; bp.pulse=Math.max(0,bp.pulse-dt*3); }
  for(const s of WALLS)s.flash=Math.max(0,s.flash-dt);
  G.parts=G.parts.filter(p=>{ p.t+=dt; p.vy+=p.g*dt; p.x+=p.vx*dt; p.y+=p.vy*dt; return p.t<p.life; });
  G.pops=G.pops.filter(p=>{ p.t+=dt; p.y-=42*dt; return p.t<0.95; });
  if(G.banner){ G.banner.t+=dt; if(G.banner.t>G.banner.dur)G.banner=null; }
  G.shake=Math.max(0,G.shake-dt*14);
  if(G.dying){ G.dieT-=dt; if(G.dieT<=0)showOver(); }
}

function shade(hex,amt){ // amt -1..1 dark..light
  const n=parseInt(hex.slice(1),16), r=n>>16,g=(n>>8)&255,b=n&255, t=amt<0?0:255, k=Math.abs(amt);
  return 'rgb('+Math.round(r+(t-r)*k)+','+Math.round(g+(t-g)*k)+','+Math.round(b+(t-b)*k)+')';
}
function rr(x,y,w,h,r){ ctx.beginPath(); ctx.moveTo(x+r,y); ctx.arcTo(x+w,y,x+w,y+h,r); ctx.arcTo(x+w,y+h,x,y+h,r); ctx.arcTo(x,y+h,x,y,r); ctx.arcTo(x,y,x+w,y,r); ctx.closePath(); }
