const keys={left:false,right:false};let mouseT=null;
function update(dt){
  for(const f of fog){f.x+=f.sp*dt;if(f.x-f.r>W)f.x=-f.r;if(f.x+f.r<0)f.x=W+f.r;}
  if(state==='menu'){player.x=W/2+Math.sin(T*.5)*W*.16;zomb.x=W/2+Math.sin(T*.33+2)*W*.1;return;}
  if(state==='paused')return;
  shake=Math.max(0,shake-dt*40);flashRed=Math.max(0,flashRed-dt*1.2);zomb.flash=Math.max(0,zomb.flash-dt*5);
  updateParts(dt);updateFloats(dt);
  if(state==='point'){freeze-=dt;if(freeze<=0)serve();return;}
  if(state!=='playing')return;

  for(const k in fx)if(fx[k]>0)fx[k]=Math.max(0,fx[k]-dt);
  const pw=(fx.gun>0?player.base*1.7:player.base),zw=(fx.panic>0?zomb.base*.55:zomb.base);
  player.w+=(pw-player.w)*Math.min(1,dt*8);zomb.w+=(zw-zomb.w)*Math.min(1,dt*6);

  // player control
  const kdir=((keys.right?1:0)-(keys.left?1:0));
  if(kdir)player.x+=kdir*950*u*dt;
  else if(mouseT!=null)player.x+=(mouseT-player.x)*Math.min(1,dt*20);
  const old=player.x;player.x=clamp(player.x,player.w/2,W-player.w/2);
  player.vx=(player.x-old)/Math.max(dt,.001);

  // zombie AI
  const diff=.5+scoreP*.075+rally*.006;
  let tx=W/2+Math.sin(T*.8)*W*.09;
  if(ball.vy<0){const tt=Math.max(0,(zomb.y-ball.y)/(-ball.vy));tx=clamp(ball.x+ball.vx*tt,0,W)+zomb.err;}
  const zspd=u*(230+160*Math.min(1.4,diff)),d=tx-zomb.x;
  zomb.x+=clamp(d,-zspd*dt,zspd*dt);zomb.x=clamp(zomb.x,zomb.w/2,W-zomb.w/2);

  // ambient growl
  growlTimer-=dt;if(growlTimer<=0){SFX.growl();growlTimer=rand(7,14);}

  // ball (sub-stepped)
  let n=Math.max(1,Math.ceil((ball.speed*dt)/(ball.r*.7)));
  for(let s=0;s<n&&state==='playing';s++){const h=dt/n;
    const py=ball.y;ball.x+=ball.vx*h;ball.y+=ball.vy*h;ball.rot+=ball.vx*h*.004;
    if(ball.x<ball.r){ball.x=ball.r;ball.vx=Math.abs(ball.vx);SFX.wall();splatParticles(2,ball.y,'#cdd6d3',8);shake=Math.max(shake,3);}
    else if(ball.x>W-ball.r){ball.x=W-ball.r;ball.vx=-Math.abs(ball.vx);SFX.wall();splatParticles(W-2,ball.y,'#cdd6d3',8);shake=Math.max(shake,3);}
    if(ball.vy>0&&py<=player.y&&ball.y>=player.y&&Math.abs(ball.x-player.x)<=player.w/2+ball.r*.5)hitPaddle('player');
    else if(ball.vy<0&&py>=zomb.y+u*12&&ball.y<=zomb.y+u*12&&Math.abs(ball.x-zomb.x)<=zomb.w/2+ball.r*.5)hitPaddle('zomb');
  }
  if(state!=='playing')return;

  if(fx.fire>0)for(let i=0;i<2;i++)addPart(ball.x+rand(-5,5),ball.y+rand(-5,5),
    -ball.vx*.06+rand(-40,40),-70*u,Math.random()<.5?'#ffab4d':'#ffd76a',rand(2,4.5)*u,.3+rand(.25),-60*u,'flame');
  trail.push({x:ball.x,y:ball.y,l:1});if(trail.length>16)trail.shift();
  for(const t of trail)t.l-=dt*3;

  if(ball.y>H+u*40){loseLife();return;}
  else if(ball.y<-u*40){scorePoint();return;}

  // powerups
  dropTimer-=dt;if(dropTimer<=0&&drops.length<1){spawnDrop();dropTimer=rand(6,11);}
  for(let i=drops.length-1;i>=0;i--){const p=drops[i];
    p.y+=u*95*dt;p.x+=Math.sin(T*2+p.ph)*u*20*dt;
    if(p.y>player.y-u*30&&p.y<player.y+u*24&&Math.abs(p.x-player.x)<player.w/2+u*16){catchDrop(p);drops.splice(i,1);continue;}
    if(p.y>H+u*30)drops.splice(i,1);}
}

  [[-.3,-.2,.28],[.35,.25,.2],[.05,.45,.16],[-.4,.35,.13]].forEach(c=>{
    ctx.beginPath();ctx.arc(mx+c[0]*mr,my+c[1]*mr,c[2]*mr,0,TAU);ctx.fill();});
  ctx.globalAlpha=1;
  if(rage>0){ctx.fillStyle=`rgba(200,60,45,${rage*.28})`;ctx.beginPath();ctx.arc(mx,my,mr,0,TAU);ctx.fill();}
  // dead trees
  ctx.strokeStyle='#0b130d';ctx.lineCap='round';
  for(const segs of trees)for(const s of segs){ctx.lineWidth=s.w;
    ctx.beginPath();ctx.moveTo(s.x1,s.y1);ctx.lineTo(s.x2,s.y2);ctx.stroke();}
  // fog
  for(const f of fog){ctx.save();ctx.translate(f.x,f.y);ctx.scale(1,.45);
    const fg=ctx.createRadialGradient(0,0,0,0,0,f.r);
    fg.addColorStop(0,`rgba(140,175,150,${f.a})`);fg.addColorStop(1,'rgba(140,175,150,0)');
    ctx.fillStyle=fg;ctx.beginPath();ctx.arc(0,0,f.r,0,TAU);ctx.fill();ctx.restore();}
  const gg=ctx.createLinearGradient(0,H*.8,0,H);
  gg.addColorStop(0,'rgba(130,160,140,0)');gg.addColorStop(1,'rgba(130,160,140,.12)');
  ctx.fillStyle=gg;ctx.fillRect(0,H*.8,W,H*.2);

  if(state!=='menu'){
    // rubble center line
    ctx.fillStyle='rgba(180,200,190,.1)';
    for(let y=u*30,i=0;y<H-u*30;y+=u*34,i++){
      const jx=Math.sin(i*7.3)*u*3;ctx.fillRect(W/2-u*3+jx,y,u*6,u*16);}
    drawDrops();
    if(state==='playing'){drawTrail();}
    drawZombies();drawPlayer();
    if(state==='playing')drawBall();
    if(state==='point')drawServeMarker();
  }else{drawZombies();drawPlayer();}
  drawParts();drawFloats();
  if(state!=='menu')drawHUD();
  ctx.restore();
  if(flashRed>0){ctx.fillStyle=`rgba(210,40,30,${flashRed*.5})`;ctx.fillRect(0,0,W,H);}
  ctx.fillStyle=vignette;ctx.fillRect(0,0,W,H);
}
