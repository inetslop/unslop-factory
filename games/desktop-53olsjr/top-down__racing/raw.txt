Resample: walk raw pts accumulating, output every DS:
```js
function resample(raw){
  const out=[raw[0].slice()]; let acc=0; let prev=raw[0];
  for(let i=1;i<=raw.length;i++){
    const cur=raw[i%raw.length];
    let d=Math.hypot(cur[0]-prev[0],cur[1]-prev[1]);
    while(acc+d>=DS && out.length<raw.length*2){
      const t=(DS-acc)/d;
      const nx=lerp(prev[0],cur[0],t), ny=lerp(prev[1],cur[1],t);
      // avoid duplicating final point
      if(out.length===raw.length && Math.hypot(nx-out[0][0],ny-out[0][1])<DS*0.5) break;
      out.push([nx,ny]);
      prev=[nx,ny]; d=Math.hypot(cur[0]-prev[0],cur[1]-prev[1]); acc=0;
    }
    acc+=d; prev=cur;
  }
  return out;
}
```

```html
function stepCar(c,inp,dt){
  const fx=Math.cos(c.heading),fy=Math.sin(c.heading),rx=-fy,ry=fx;
  let vF=c.vx*fx+c.vy*fy,vL=c.vx*rx+c.vy*ry;
  const grass=c.onGrass;
  if(inp.throttle>0)vF+=inp.throttle*ACCEL*dt*(grass?.55:1);
  if(inp.brake>0){if(vF>12)vF-=BRAKE*dt*inp.brake;else vF=Math.max(vF-380*dt*inp.brake,-REVMAX);}
  c.braking=!!(inp.brake&&vF>12);
  const lin=grass?1.9:.5,q=grass?.004:.00045;
  vF-=vF*(lin+Math.abs(vF)*q)*dt;
  if(inp.hand)vF*=Math.exp(-.9*dt);
  let grip=grass?2.4:(inp.hand?2.0:6.8)*c.gripMul;
  vL*=Math.exp(-grip*dt);

function update(dt){
  const sdt=dt*timeScale;timeScale+=(1-timeScale)*(1-Math.exp(-1.3*dt));
  if(state==='count'){cd-=sdt;const num=Math.ceil(cd);
    if(num!==lastCdNum&&num>=1&&num<=3){beep(440,.14,'square',.2);lastCdNum=num;}
    for(const c of cars)stepCar(c,{steer:0,throttle:0,brake:0,hand:false},sdt);
    if(cd<=0){state='race';raceStartMs=performance.now();for(const c of cars)c.lapStart=raceStartMs;
      beep(880,.4,'square',.24);say('GO!',1,96*hs,'#ffd23e');}}
  else if(state==='race'||state==='done'){
    for(const c of cars){const inp=c.isPlayer?(c.finishTime?{steer:0,throttle:0,brake:0,hand:false}:playerInput()):aiInput(c,nowSec);
      stepCar(c,inp,sdt);}
    collide();
    for(const c of cars){
      const ld=Math.floor(c.s/L);
      if(ld>c.lapsDone){c.lapsDone=ld;const nowMs=performance.now();
        if(!c.finishTime&&ld>=TOTAL_LAPS)c.finishTime=nowMs-raceStartMs;
        if(c.isPlayer){const lap=nowMs-c.lapStart;c.lastLap=lap;c.lapStart=nowMs;
          let nb=false;if(sessionBest==null||lap<sessionBest){sessionBest=lap;newBestThisRace=true;store('apexrush.best',Math.round(sessionBest));nb=true;}
          if(c.lapsDone>=TOTAL_LAPS){state='done';doneAt=nowSec;fanfare();confetti(c.x,c.y);timeScale=.35;
            say('FINISH!',1.6,92*hs,'#ffd23e');}
          else if(nb){newBestSfx();say('NEW BEST LAP!',1.6,44*hs,'#ff6b2c',fmtTime(lap));refreshBestChip();}
          else if(c.lapsDone===TOTAL_LAPS-1)say('FINAL LAP',1.5,58*hs,'#f2efe6','LAP '+fmtTime(lap));
          else say('LAP '+(c.lapsDone+1)+'/'+TOTAL_LAPS,1.3,40*hs,'#f2efe6','LAP '+fmtTime(lap));
          if(nb)chimeLap();}}
    }
    if(state==='done'&&!showedFinish&&nowSec-doneAt>1.15){showedFinish=true;showFinish();}
  }
  else{ // menu attract
    for(const c of cars)stepCar(c,aiInput(c,nowSec),sdt);collide();
  }
  // particles
  for(let i=parts.length-1;i>=0;i--){const p=parts[i];p.life-=sdt;
    if(p.life<=0){parts.splice(i,1);continue;}
    p.x+=p.vx*sdt;p.y+=p.vy*sdt;p.vx*=Math.exp(-2.4*sdt);p.vy*=Math.exp(-2.4*sdt);
    if(p.cf)p.rot+=p.vr*sdt;}
  for(let i=skids.length-1;i>=0;i--)if(nowSec-skids[i].t>SKID_FADE)skids.splice(i,1);
  // dust / smoke emission from player
  const c=player,f=Math.cos(c.heading),s2=Math.sin(c.heading);
  if(c.onGrass&&c.speed>60&&Math.random()<.8)dust(c.x-f*14,(c.y-s2*14),(Math.random()-.5)*40-c.vx*.1,(Math.random()-.5)*40-c.vy*.1);
  if(c.slip>72&&!c.onGrass){smoke(c.x-f*11+s2*9.5,c.y-s2*11-f*9.5,-c.vx*.15+(Math.random()-.5)*30,-c.vy*.15+(Math.random()-.5)*30,6,'#d8dcd4');
    smoke(c.x-f*11-s2*9.5,c.y-s2*11+f*9.5,-c.vx*.15+(Math.random()-.5)*30,-c.vy*.15+(Math.random()-.5)*30,6,'#d8dcd4');}
  // camera
  const tx=c.x+c.vx*.28,ty=c.y+c.vy*.28;
  const k=1-Math.exp(-4.5*dt);cam.x+=(tx-cam.x)*k;cam.y+=(ty-cam.y)*k;
  cam.a+=angDiff(cam.a,c.heading)*(1-Math.exp(-3.5*dt));
  const zt=baseZ()*(1-.16*clamp(c.speed/560,0,1));cam.z+=(zt-cam.z)*(1-Math.exp(-2.5*dt));
  shake*=Math.exp(-6*dt);
  for(const m of msgs)m.t+=dt;for(let i=msgs.length-1;i>=0;i--)if(msgs[i].t>msgs[i].dur)msgs.splice(i,1);
  // audio
  if(AC){const t=AC.currentTime;let rpm;
    if(state==='count')rpm=.32+.22*Math.abs(Math.sin(nowSec*9));
    else rpm=clamp(c.speed/560,0,1)*.85+(c.isPlayer?playerInput().throttle*.15:.1)+.06;
    engO1.frequency.setTargetAtTime(50+rpm*175,t,.06);
    engO2.frequency.setTargetAtTime((50+rpm*175)*.498,t,.06);
    engF.frequency.setTargetAtTime(250+rpm*2400,t,.08);
    const g=(paused||muted)?0:.03+rpm*.07+(state==='race'&&playerInput().throttle?.015:0);
    engG.gain.setTargetAtTime(g,t,.09);
    skidG.gain.setTargetAtTime(c.onGrass?clamp((c.speed-60)/800,0,1)*.07:(!paused?clamp(c.slip/300,0,1)*.24:0),t,.07);}
}
```
