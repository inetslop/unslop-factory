function applyGrow(){ const cw=P.w,ch=P.h; if(eff.grow>0){P.w=Math.round(26*1.5);P.h=Math.round(50*1.4);} else {P.w=cw===undefined?26:26; P.h=50;}
  P.w=eff.grow>0?39:26; P.h=eff.grow>0?70:50; }

function updatePlayer(dt,canAct){
  applyGrow();
  if(P.standPlat&&P.standPlat.dx)P.x+=P.standPlat.dx;
  const ax=(keys.right?1:0)-(keys.left?1:0);
  if(canAct){
    if(ax){ P.vx+=ax*ACC*dt; P.facing=ax; }
    else{ const f=FRIC*dt; if(Math.abs(P.vx)<=f)P.vx=0; else P.vx-=Math.sign(P.vx)*f; }
    P.vx=clamp(P.vx,-RUN,RUN);
  }
  // jump / coyote / drop
  if(canAct&&jumpBuf>0){
    if(P.ground){
      if(keys.down&&P.standPlat){ P.dropT=0.24; P.y+=6; P.ground=false; P.standPlat=null; }
      else{ P.vy=-JUMP_V; P.ground=false; P.squash=-0.35; puff(P.x+P.w/2,P.y+P.h,5); sfx.jump(); }
    } else if(P.coyote>0){ P.vy=-JUMP_V; P.coyote=0; P.squash=-0.35; puff(P.x+P.w/2,P.y+P.h,4); sfx.jump(); }
  }
  jumpBuf-=dt;
  const gmul=(P.vy<0&&!keys.jump)?2.4:1;
  P.vy+=GRAV*dt*gmul; P.vy=Math.min(P.vy,950);
  const prevFeet=P.y+P.h;
  P.x=clamp(P.x+P.vx*dt,4,W-4-P.w);
  P.y+=P.vy*dt;
  // one-way platforms + floor
  const wasGround=P.ground; let landed=false;
  if(P.vy>=0){
    for(const p of plats){
      if(P.dropT>0)break;
      const feet=P.y+P.h;
      if(prevFeet<=p.y+6&&feet>=p.y&&P.x+P.w>p.x+4&&P.x<p.x+p.w-4){
        P.y=p.y-P.h; P.vy=0; P.ground=true; P.standPlat=p; landed=true; break; }
    }
  }
  const feet=P.y+P.h;
  if(feet>=FLOOR){ P.y=FLOOR-P.h; P.vy=0; P.ground=true; P.standPlat=null; landed=true; }
  if(landed&&!wasGround&&P.vy===0){ /* handled below */ }
  if(wasGround!==P.ground&&P.ground){ P.squash=0.4; puff(P.x+P.w/2,P.y+P.h,6); }
  if(!P.ground)P.coyote=0.1; else P.coyote=Math.min(0.1,P.coyote+dt*10>0.1?0.1:P.coyote),P.coyote=P.ground?0.1:P.coyote-dt;
  P.dropT-=dt;
  P.squash*=Math.pow(0.0001,dt);
  P.invuln-=dt; P.blinkT-=dt; if(P.blinkT<-0.12)P.blinkT=2+Math.random()*2.5;
  // solid vs AI paddle
  if(aabb(P,A)){ if(P.x+P.w/2<A.x+A.w/2)P.x=A.x-P.w-1; else P.x=A.x+A.w+1; P.vx=0; }
}

function damagePlayer(){
  if(P.invuln>0||eff.shield>0){ return; }
  hearts--; sfx.hurt(); addShake(0.4,12); hurtFlash=1; flash('255,90,60',0.3);
  burst(P.x+P.w/2,P.y+P.h/2,18,{col:['#ff5a3c','#fff3e0'],spd:220});
  floatTxt(P.x+P.w/2,P.y-14,'OUCH!','#ff5a3c',22);
  P.vy=-430; P.vx=(Math.random()<0.5?-1:1)*260;
  if(hearts<=0){ P.dead=true; state='dying'; stT=1.1; sfx.lose(); }
  else P.invuln=1.6;
}

function updateAI(dt){
  const total=scores.you+scores.cpu;
  const spdMax=Math.min(470,300+total*16);
  if(B.vx>0&&state==='play'&&B.x>W*0.32){
    A.errT-=dt;
    if(A.errT<=0){ A.errT=0.16+Math.random()*0.1; const amp=Math.max(12,30-total*2.5); A.err=(Math.random()*2-1)*amp; }
    const t=(A.x-B.x)/B.vx; let py=B.y+B.vy*t;
    const top=B.r+2,bot=FLOOR-B.r-2,rng=bot-top;
    py=((py-top)%(2*rng)+2*rng)%(2*rng); if(py>rng)py=2*rng-py; A.tY=top+py+A.err;
  } else A.tY=H*0.5+Math.sin(time*0.8)*70;
  const d=A.tY-(A.y+A.h/2), mv=Math.min(Math.abs(d),spdMax*dt*(B.vx>0?1:0.55));
  A.y+=Math.sign(d)*mv; A.y=clamp(A.y,B.r,FLOOR-A.h-B.r);
}

function renorm(){ const m=Math.hypot(B.vx,B.vy)||1; B.vx*=B.spd/m; B.vy*=B.spd/m; }
function enforceMinHx(){ const mh=0.32*B.spd;
  if(Math.abs(B.vx)<mh){ B.vx=(B.vx<0?-1:1)*mh; B.vy=Math.sign(B.vy||1)*Math.sqrt(Math.max(0,B.spd*B.spd-mh*mh)); } }

function paddleHit(pad,who,maxAng){
  const cx=pad.x+pad.w/2, cy=pad.y+pad.h/2, hw=pad.w/2+B.r, hh=pad.h/2+B.r;
  const dx=B.x-cx, dy=B.y-cy;
  if(Math.abs(dx)>=hw||Math.abs(dy)>=hh)return false;
  const px=hw-Math.abs(dx), py=hh-Math.abs(dy);
  let nx,ny;
  if(px<py){nx=dx>0?1:-1;ny=0;}else{nx=0;ny=dy>0?1:-1;}
  let speed=B.spd*1.05+6;
  let didSmash=false;
  if(who==='p'&&smashArm>0&&charges>0){ speed=Math.min(SMASH_CAP,speed*1.7+80); charges--; smashArm=0; B.smash=true; didSmash=true; }
  speed=Math.min(BALL_CAP,B.spd*1.055+6, speed===B.spd*1.05+6?speed:SMASH_CAP);
  speed=Math.min(didSmash?SMASH_CAP:BALL_CAP, Math.max(speed, B.spd));
  if(nx!==0){
    const off=clamp((B.y-cy)/(pad.h/2),-1,1), ang=off*maxAng;
    B.vx=nx*Math.cos(ang)*speed; B.vy=Math.sin(ang)*speed;
    B.x=cx+nx*(hw-0.5);
  } else {
    B.vy=ny*Math.abs(B.vy); B.vx=B.vx||1; renorm(); B.spd=speed; renorm();
    B.y=cy+ny*(hh-0.5);
  }
  B.spd=Math.min(didSmash?SMASH_CAP:BALL_CAP,speed);
  rally++; rallyPop=1; if(rally>maxRally)maxRally=rally;
  const hx=B.x, hy=B.y;
  if(who==='p'){
    sfx.hit(B.spd); addShake(didSmash?0.32:0.12,didSmash?11:4); freeze=Math.max(freeze,didSmash?0.07:0.035);
    burst(hx,hy,didSmash?26:9,{col:didSmash?['#ffcf3f','#fff3e0','#ff8c42']:['#fff3e0','#ffb45c'],spd:didSmash?340:190,size:didSmash?6:4});
    if(didSmash){sfx.smash(); floatTxt(hx,hy-24,'SMASH!','#ffcf3f',30); flash('255,207,63',0.2);}
  } else { sfx.hit(B.spd*0.8); burst(hx,hy,7,{col:['#9fb4c8','#fff3e0'],spd:160}); }
  return true;
}

function ballStep(h){
  const py=B.y;
  B.x+=B.vx*h; B.y+=B.vy*h;
  if(B.x<-26){point('cpu');return;}
  if(B.x>W+26){point('you');return;}
  if(B.y<B.r){B.y=B.r;B.vy=Math.abs(B.vy);sfx.wall();puff(B.x,B.y,3);}
  if(B.y+FLOOR-B.r>=FLOOR&&B.y>B.y-1){} // placeholder no-op
  if(B.y+B.r>=FLOOR){B.y=FLOOR-B.r;B.vy=-Math.abs(B.vy);sfx.floorB();puff(B.x,FLOOR,4);}
  for(const p of plats){
    if(B.vy>0&&py+B.r<=p.y+3&&B.y+B.r>=p.y&&B.x>p.x-2&&B.x<p.x+p.w+2){
      B.y=p.y-B.r; B.vy=-Math.abs(B.vy);
      if(p.dx){B.vx+=p.dx*8; renorm();}
      sfx.wall(); puff(B.x,p.y,3);
    }
  }
  if(aabbCircle(P))paddleHit(P,'p',62*Math.PI/180);
  else paddleHit(A,'a',46*Math.PI/180);
}
function aabbCircle(pad){ const cx=clamp(B.x,pad.x,pad.x+pad.w), cy=clamp(B.y,pad.y,pad.y+pad.h);
  return (B.x-cx)*(B.x-cx)+(B.y-cy)*(B.y-cy)<B.r*B.r; }

function updateBall(dt){
  const ts=eff.slow>0?0.55:1, bdt=dt*ts;
  const steps=Math.max(1,Math.ceil(B.spd*bdt/(B.r*0.9)));
  for(let i=0;i<steps;i++){ if(state!=='play')return; ballStep(bdt/steps); enforceMinHx(); }
  B.trail.push({x:B.x,y:B.y}); if(B.trail.length>15)B.trail.shift();
  if(B.smash&&Math.random()<0.5)particles.push({x:B.x,y:B.y,vx:(Math.random()-0.5)*60,vy:(Math.random()-0.5)*60,g:0,life:0.3,max:0.3,size:3,col:'#ffcf3f',shape:'spark',rot:Math.random()*6,vr:8,ph:0});
}

    if(!p.landed){ p.y+=p.vy*dt; p.x+=Math.sin(time*2+p.ph)*20*dt;

    } else p.life-=dt;

    const box={x:p.x-15,y:p.y-10,w:30,h:20};

function updateParticles(dt){
  for(let i=particles.length-1;i>=0;i--){ const p=particles[i];
    p.life-=dt; if(p.life<=0){particles.splice(i,1);continue;}
    p.vy+=p.g*dt; if(p.shape==='conf')p.vx+=Math.sin(time*4+p.ph)*30*dt;
    p.x+=p.vx*dt; p.y+=p.vy*dt; p.rot+=p.vr*dt;
  }
  for(let i=texts.length-1;i>=0;i--){const t=texts[i]; t.life-=dt; t.y-=42*dt; if(t.life<=0)texts.splice(i,1);}
}

function update(dt){
  time+=dt;
  farIsles.forEach(f=>{f.x+=f.spd*dt*0.35; if(f.x-f.w>W)f.x=-f.w;});
  clouds.forEach(c=>{c.x+=c.spd*dt; if(c.x-c.w>W)c.x=-c.w;});
  updateParticles(dt);
  shakeT-=dt; flashA=Math.max(0,flashA-dt*1.6); hurtFlash=Math.max(0,hurtFlash-dt*2);
  popYou=Math.max(0,popYou-dt*3); popCpu=Math.max(0,popCpu-dt*3); rallyPop=Math.max(0,rallyPop-dt*3);

  if(state==='menu')return;
  // effects timers
  eff.grow=Math.max(0,eff.grow-dt); eff.slow=Math.max(0,eff.slow-dt); eff.shield=Math.max(0,eff.shield-dt);
  applyGrow();

  if(state==='pause'||state==='end')return;

  // platforms
  for(const p of plats){ if(p.mov){const nx=p.baseX+Math.sin(time*p.mov.spd)*p.mov.amp; p.dx=nx-p.x; p.x=nx;} else p.dx=0; }

  const canAct=(state==='play'||state==='serve'||state==='point')&&!P.dead;
  if(P.dead){ P.vy+=GRAV*dt; P.y+=P.vy*dt; P.spin+=dt*9; }
  else updatePlayer(dt,canAct);

  // spikes
  if(!P.dead&&P.invuln<=0&&eff.shield<=0){
    for(const s of spikes){ const box={x:s.x+3,y:FLOOR-14,w:s.w-6,h:14};
      if(aabb({x:P.x+3,y:P.y+3,w:P.w-6,h:P.h-6},box)){damagePlayer();break;} }
  }

  updateAI(dt);
  updateCoins();
  updatePowerups(dt);

  // smash input
  if(smashEdge){ smashEdge=false;
    if(state==='play'&&!P.dead){ if(charges>0){smashArm=0.26;} else sfx.whiff(); } }
  smashArm=Math.max(0,smashArm-dt);

  if(state==='serve'){
    stT-=dt;
    if(stT<=0.35&&!serveGo){serveGo=true; sfx.go();}
    if(stT<=0){ state='play'; B.vx=B.pvx; B.vy=B.pvy; }
  } else if(state==='play'){
    updateBall(dt);
  } else if(state==='point'){
    B.x+=B.vx*dt; B.y+=B.vy*dt;
    B.trail.push({x:B.x,y:B.y}); if(B.trail.length>15)B.trail.shift();
    stT-=dt;
    if(stT<=0){
      if(pendingEnd){showEnd(pendingEnd==='win');}
      else{ coins.forEach(c=>c.taken=false); if(Math.random()<0.45&&powerups.length<2)spawnPowerup(); serve(); }
    }
  } else if(state==='dying'){
    stT-=dt; if(stT<=0)showEnd(false);
  }
}

function drawPlat(p){

  if(p.mov){
    ctx.fillStyle='rgba(255,180,92,0.16)';

function drawCoin(c){
  if(c.taken)return;
  const y=c.y+Math.sin(time*3+c.ph)*3, sc=0.7+0.3*Math.abs(Math.sin(time*2+c.ph));
  ctx.save(); ctx.translate(c.x,y); ctx.scale(sc,1);

}
function drawPowerup(p){
  const cols={grow:'#4fd07a',slow:'#4fc3e8',shield:'#ffcf3f'}, letters={grow:'G',slow:'S',shield:'H'};
  const a=p.landed&&p.life<1.5?Math.max(0.2,Math.sin(time*10)*0.5+0.5):1;
  ctx.save(); ctx.globalAlpha=a; ctx.translate(p.x,p.y);
  ctx.fillStyle='rgba(255,255,255,0.14)'; ctx.beginPath(); ctx.arc(0,0,17+Math.sin(time*4+p.ph)*2,0,7); ctx.fill();
  ctx.fillStyle=cols[p.type]; rr(-14,-9,28,18,9); ctx.fill();
  ctx.strokeStyle='#0a2231'; ctx.lineWidth=2; ctx.stroke();
  txt(letters[p.type],0,1,13,'#fff','center',3);
  ctx.restore();
}
function drawBall(){
  // shadow
  const hgt=FLOOR-B.y, sa=clamp(1-hgt/420,0,0.3);
  if(sa>0){ctx.fillStyle='rgba(4,12,18,'+sa+')'; ctx.beginPath(); ctx.ellipse(B.x,FLOOR+5,9+hgt*0.02,3,0,0,7); ctx.fill();}
  // trail
  for(let i=0;i<B.trail.length;i++){ const t=B.trail[i], f=i/B.trail.length;
    ctx.fillStyle=B.smash?'rgba(255,207,63,'+(f*0.35)+')':'rgba(255,243,224,'+(f*0.28)+')';
    ctx.beginPath(); ctx.arc(t.x,t.y,B.r*f*0.9,0,7); ctx.fill(); }
  const pulse=B.smash?1+0.12*Math.sin(time*24):1;
  if(B.smash){ctx.fillStyle='rgba(255,207,63,0.25)'; ctx.beginPath(); ctx.arc(B.x,B.y,B.r*2.2*pulse,0,7); ctx.fill();}
  ctx.fillStyle=B.smash?'#ffcf3f':'#fff8ef';
  ctx.beginPath(); ctx.arc(B.x,B.y,B.r*pulse,0,7); ctx.fill();
  ctx.strokeStyle=B.smash?'#a86e12':'#ff5a3c'; ctx.lineWidth=2.5; ctx.stroke();
  ctx.fillStyle='rgba(255,255,255,0.9)'; ctx.beginPath(); ctx.arc(B.x-3,B.y-3,2.6,0,7); ctx.fill();
}
function drawPlayer(){
  if(P.dead&&state==='end')return;
  const cx=P.x+P.w/2, feet=P.y+P.h;
  ctx.save(); ctx.translate(cx,feet);
  if(P.dead)ctx.rotate(P.spin);
  else ctx.rotate(clamp(P.vx/RUN,-1,1)*0.1);
  const sq=P.squash; ctx.scale(1-sq*0.5,1+sq*0.6);
  if(P.invuln>0&&!P.dead)ctx.globalAlpha=0.45+0.4*Math.sin(time*28);
  // body
  ctx.fillStyle='#fff3e0'; rr(-P.w/2,-P.h,P.w,P.h,9); ctx.fill();
  ctx.save(); rr(-P.w/2,-P.h,P.w,P.h,9); ctx.clip();
  ctx.fillStyle='#ff5a3c'; ctx.fillRect(-P.w/2,-P.h*0.62,P.w,P.h*0.3);
  ctx.restore();
  ctx.strokeStyle='#14283a'; ctx.lineWidth=3; rr(-P.w/2,-P.h,P.w,P.h,9); ctx.stroke();
  // eyes track ball
  const lx=clamp((B.x-cx)/60,-2.6,2.6), ly=clamp((B.y-(feet-P.h+14))/60,-2,2);
  const blink=P.blinkT<0&&!P.dead?0.15:1;
  [[-P.w*0.2,1],[P.w*0.22,1]].forEach(e=>{
    ctx.fillStyle='#fff'; ctx.beginPath(); ctx.ellipse(e[0]*P.w/ - P.w*0 + (e[0]),-P.h+14,4.6,5*blink,0,0,7); ctx.fill();
    ctx.strokeStyle='#14283a'; ctx.lineWidth=1.5; ctx.stroke();
    if(blink>0.5){ctx.fillStyle='#14283a'; ctx.beginPath(); ctx.arc(e[0]+lx,-P.h+14+ly,2.2,0,7); ctx.fill();}
  });
  // antenna + charge light
  ctx.strokeStyle='#14283a'; ctx.lineWidth=2;
  ctx.beginPath(); ctx.moveTo(0,-P.h); ctx.lineTo(0,-P.h-9); ctx.stroke();
  const chGlow=charges>0?0.5+0.5*Math.sin(time*(6+charges*3)):0;
  ctx.fillStyle='rgba(255,207,63,'+(0.25+0.75*chGlow)+')';
  ctx.beginPath(); ctx.arc(0,-P.h-12,3+chGlow*1.6,0,7); ctx.fill();
  if(eff.shield>0){ctx.strokeStyle='rgba(255,207,63,'+(0.5+0.3*Math.sin(time*6))+')'; ctx.lineWidth=2.5;
    ctx.beginPath(); ctx.arc(0,-P.h/2,P.h*0.75,0,7); ctx.stroke();}
  ctx.restore();
}

function drawGoalLine(side){
  const x=side?W-9:9, col=side?'159,180,200':'255,90,60';
  const active=state==='play'&&(side?B.vx>0:B.vx<0);
  for(let y=18;y<FLOOR-8;y+=26){
    ctx.fillStyle='rgba('+col+','+(active?0.35+0.3*Math.sin(time*8+y):0.14)+')';
    ctx.fillRect(x,y,4,14);
  }
  const my=H*0.42, dir=side?1:-1;
  ctx.strokeStyle='rgba('+col+','+(active?0.9:0.5)+')'; ctx.lineWidth=3; ctx.lineCap='round';
  for(let i=0;i<2;i++){ ctx.beginPath();
    ctx.moveTo(x-dir*8-i*10,my-8); ctx.lineTo(x-dir*16-i*10,my); ctx.lineTo(x-dir*8-i*10,my+8); ctx.stroke(); }
}

function render(){
  ctx.setTransform(dpr,0,0,dpr,0,0);
  ctx.clearRect(0,0,W,H);
  ctx.drawImage(bgC,0,0,W,H);
  // stars
  stars.forEach(s=>{ctx.fillStyle='rgba(255,243,224,'+(0.25+0.5*Math.abs(Math.sin(time*1.4+s.ph)))+')';
    ctx.beginPath();ctx.arc(s.x,s.y,s.r,0,7);ctx.fill();});
  // far isles + clouds
  farIsles.forEach(f=>{ctx.fillStyle='rgba(11,35,52,0.9)';
    ctx.beginPath(); ctx.ellipse(f.x,f.y+f.h*0.4,f.w/2,f.h*0.6,0,Math.PI,0); ctx.rect(f.x-f.w/2,f.y+f.h*0.4,f.w,f.h*0.6); ctx.fill();});
  clouds.forEach(c=>{ctx.fillStyle='rgba(30,90,112,'+c.a+')';
    ctx.beginPath(); ctx.ellipse(c.x,c.y,c.w/2,12,0,0,7); ctx.fill();});

  // shake
  let sx=0,sy=0;
  if(shakeT>0){const f=shakeT/shakeDur; sx=(Math.random()-0.5)*2*shakeMag*f; sy=(Math.random()-0.5)*2*shakeMag*f;}
  ctx.save(); ctx.translate(sx,sy);

  // center net (pong nod)
  ctx.strokeStyle='rgba(255,243,224,0.1)'; ctx.lineWidth=2; ctx.setLineDash([8,12]);
  ctx.beginPath(); ctx.moveTo(W/2,14); ctx.lineTo(W/2,FLOOR); ctx.stroke(); ctx.setLineDash([]);

  drawGoalLine(0); drawGoalLine(1);
  // floor
  ctx.fillStyle='#0f2f40'; ctx.fillRect(-20,FLOOR,W+40,H-FLOOR+20);
  ctx.fillStyle='#2e7d95'; ctx.fillRect(-20,FLOOR,W+40,5);
  ctx.fillStyle='rgba(10,34,49,0.6)';
  for(let x=40;x<W;x+=80)ctx.fillRect(x,FLOOR+6,2,H-FLOOR);

  plats.forEach(drawPlat);
  spikes.forEach(drawSpikes);
  coins.forEach(drawCoin);
  powerups.forEach(drawPowerup);
  if(state!=='menu')drawBall();
  drawAI();
  if(state!=='menu')drawPlayer();
  // particles
  particles.forEach(p=>{ const a=clamp(p.life/p.max,0,1);
    if(p.shape==='conf'){ctx.save();ctx.translate(p.x,p.y);ctx.rotate(p.rot);ctx.globalAlpha=a;
      ctx.fillStyle=p.col;ctx.fillRect(-p.size/2,-p.size/3,p.size,p.size*0.66);ctx.restore();}
    else if(p.shape==='spark'){ctx.save();ctx.translate(p.x,p.y);ctx.rotate(p.rot);ctx.globalAlpha=a;
      ctx.strokeStyle=p.col;ctx.lineWidth=2;ctx.beginPath();ctx.moveTo(-p.size,0);ctx.lineTo(p.size,0);ctx.stroke();ctx.restore();}
    else{ctx.globalAlpha=a*0.9;ctx.fillStyle=p.col;ctx.beginPath();ctx.arc(p.x,p.y,p.size*a+1,0,7);ctx.fill();}
    ctx.globalAlpha=1;});
  // float texts
  texts.forEach(t=>{const a=clamp(t.life/t.max,0,1), sc=t.life>t.max-0.15?1+(t.max-t.life)*4:1;
    ctx.save();ctx.translate(t.x,t.y);ctx.scale(sc,sc);ctx.globalAlpha=a;
    txt(t.t,0,0,t.size,t.col,'center',5);ctx.restore();ctx.globalAlpha=1;});
  ctx.restore();

  drawHUD();
  // serve / point banners
  if(state==='serve'){
    const a=0.6+0.4*Math.sin(time*8);
    txt(serveGo?'GO!':'READY',W/2,H*0.3,serveGo?64:40,serveGo?'#ffcf3f':'rgba(255,243,224,'+a+')','center',8);
    // pulsing ring at ball
    ctx.strokeStyle='rgba(255,243,224,'+(0.5+0.4*Math.sin(time*10))+')'; ctx.lineWidth=2;
    ctx.beginPath(); ctx.arc(B.x,B.y,B.r+8+3*Math.sin(time*10),0,7); ctx.stroke();
  }
  if(state==='point'||(state==='dying'&&bannerTxt)){
    const t=1-clamp(stT/0.25,0,1), sc=0.6+t*0.4;
    ctx.save(); ctx.translate(W/2,H*0.34); ctx.scale(sc,sc);
    txt(bannerTxt,0,0,54,bannerCol,'center',9); ctx.restore();
    txt('FIRST TO 5',W/2,H*0.34+44,15,'#bcd6e2','center');
  }
  // vignette + flashes
  ctx.drawImage(vgC,0,0,W,H);
  if(hurtFlash>0){ctx.fillStyle='rgba(255,60,40,'+(hurtFlash*0.28)+')';ctx.fillRect(0,0,W,H);}
  if(flashA>0){ctx.fillStyle='rgba('+flashCol+','+flashA+')';ctx.fillRect(0,0,W,H);}
}

let last=performance.now();
function frame(now){
  const rdt=Math.min(0.035,(now-last)/1000); last=now;
  if(freeze>0)freeze-=rdt; else update(rdt);
  render();
  requestAnimationFrame(frame);
}
