let scoreShown=-1,livesShown=99,padsShown=-1;

/* ================= world build ================= */
function buildTraffic(){
 const defs=[
  {row:5,dir:-1,speed:1.75,list:[{t:'car',len:1,c:'#b0483f'},{t:'car',len:1,c:'#3f6ea8'},{t:'car',len:1,c:'#7a8446'}]},
  {row:6,dir:1,speed:1.2,list:[{t:'truck',len:2,c:'#a86a3f'},{t:'car',len:1,c:'#5d7f9e'}]},
  {row:7,dir:1,speed:2.45,list:[{t:'car',len:1,c:'#c2a03f'},{t:'car',len:1,c:'#8a5aa8'},{t:'car',len:1,c:'#b0483f'}]},
  {row:8,dir:-1,speed:2.0,list:[{t:'car',len:1,c:'#4fae7d'},{t:'truck',len:2,c:'#8c5b3f'},{t:'car',len:1,c:'#3f6ea8'}]}
 ];
 vehRows=defs.map(d=>{const M=(3)*cell,span=bw+2*M;
  return {row:d.row,dir:d.dir,speed:d.speed,dist:rand(span),M,span,list:d.list.map((v,i)=>({...v,off:i*(span/d.list.length)+rand(0,span*.12),x:-99}))};});
}
function platRow(row,dir,speed,lens,Mcells){const M=Mcells*cell,span=bw+2*M;
 plats.push({row,dir,speed,dist:rand(span),M,span,list:lens.map((l,i)=>({len:l,off:i*(span/lens.length)+rand(0,span*.1),ph:rand(TAU),x:-99}))});}
function buildPlats(){plats=[];platRow(1,1,.55,[.9,.9,.9],2);platRow(2,-1,2.2,[4,3],5);platRow(3,1,2.95,[3,4],5);}
function buildHerons(){herons=[
 {row:9,x:bw*.3,dir:1,speed:.95,state:'patrol',tx:0,tSearch:0,honkCd:0,phase:rand(9)},
 {row:4,x:bw*.28,dir:1,speed:1.05,state:'patrol',tx:0,tSearch:0,honkCd:0,phase:rand(9)},
 {row:4,x:bw*.74,dir:-1,speed:.85,state:'patrol',tx:0,tSearch:0,honkCd:0,phase:rand(9)}];}
function buildWorld(){buildTraffic();buildPlats();buildHerons();}

/* ================= screens / hud ================= */
function showPanel(id){$('#screen').classList.add('show');document.querySelectorAll('.panel.on').forEach(p=>p.classList.remove('on'));const el=document.getElementById(id);if(el)el.classList.add('on');}
function hideScreens(){$('#screen').classList.remove('show');document.querySelectorAll('.panel.on').forEach(p=>p.classList.remove('on'));}
function loadBest(){try{return +(localStorage.getItem('hushHopBest')||0)}catch(e){return 0}}
function saveBest(){const b=loadBest();if(score>b){try{localStorage.setItem('hushHopBest',String(score))}catch(e){}}return Math.max(b,score);}
function refreshBest(){const b=loadBest();$('#menuBest').textContent=b>0?'BEST · '+b:'';}
function hud(){
 if(scoreShown!==score){scoreShown=score;$('#score').textContent=score;}
 $('#hushFill').style.width=(Math.round((1-clamp(P.hushCd,0,7)/7)*100))+'%';
 $('#hushBox').classList.toggle('ready',P.hushCd<=0&&state==='play');
 if(livesShown!==lives){livesShown=lives;[...$('#lives').children].forEach((e,i)=>e.classList.toggle('dead',i>=lives));}
 if(padsShown!==pads.length){padsShown=pads.length;[...$('#pads').children].forEach((e,i)=>e.classList.toggle('filled',pads.includes(PAD_COLS[i])));}
}

/* ================= game flow ================= */
function resetPlayer(){P.row=9;P.col=6;P.x=(COLS/2)*cell;P.y=(9.5)*cell;P.hop=null;P.squash=0;P.invuln=2.4;P.cloak=0;P.hushCd=0;P.dead=false;P.deadT=0;P.cause='';P.sinkY=0;P.bestRow=9;P.alerted=false;P.facing={x:0,y:-1};}
function startGame(){score=0;lives=3;pads=[];spdMul=1;heronMul=1;noise=0;shake=0;idleTick=0;held.length=0;
 parts.length=0;floats.length=0;rings.length=0;buildWorld();resetPlayer();state='play';hideScreens();sfxUi();hud();
 floats.push({x:P.x,y:P.y-cell*.7,txt:'REACH THE LILY PADS',t:0,life:1.6,c:'#9ff2cf',size:clamp(15,cell*.34,26)});}
function pause(on){if(state==='play'&&on){state='pause';showPanel('pause');}else if(state==='pause'&&!on){state='play';hideScreens();}}
function gameOver(cause){state='over';sfxOver();const b=saveBest();
 $('#overCause').textContent={car:'SPLATTED BY TRAFFIC',heron:'CAUGHT BY THE HERONS',water:'LOST IN THE REEDS'}[cause]||'THE NIGHT WINS';
 $('#overScore').textContent=score;$('#overBest').textContent=b;$('#overPads').textContent=pads.length+'/5';refreshBest();showPanel('over');}
function winGame(){state='win';sfxWin();const b=saveBest();
 for(let i=0;i<70;i++){const x=rand(bw),y=rand(0,cell*2);parts.push({x,y,vx:rand(-40,40),vy:rand(-90,-20),g:60,life:rand(.8,1.6),t:0,c:i%2?'#ffd98a':'#9ff2cf',r:rand(1.5,3.5)});}
 $('#winScore').textContent=score;$('#winBest').textContent=b;refreshBest();showPanel('win');}

/* ================= effects ================= */
function addScore(n,x,y,txt){score+=n;floats.push({x,y,txt:txt||('+'+n),t:0,life:.95,c:'#eaffd8',size:clamp(16,cell*.42,32)});}
function burst(x,y,n,col,spd,g){for(let i=0;i<n;i++){const a=rand(TAU),s=rand(spd*.3,spd);parts.push({x,y,vx:Math.cos(a)*s,vy:Math.sin(a)*s-30,g,life:rand(.4,.9),t:0,c:col,r:rand(1.5,3.8)});}}
function spawnRipple(x,y){rings.push({x,y,r:cell*.3,a:.5});rings.push({x,y,r:cell*.12,a:.4});}

/* ================= player actions ================= */
const DIRS={ArrowUp:[0,-1],w:[0,-1],W:[0,-1],ArrowDown:[0,1],s:[0,1],S:[0,1],ArrowLeft:[-1,0],a:[-1,0],A:[-1,0],ArrowRight:[1,0],d:[1,0],D:[1,0]};
const held=[];
function isHidden(){return P.cloak>0||BUSH.has(P.row+','+P.col);}
function finishHop(){if(!P.hop)return;P.x=P.hop.tx;P.y=P.hop.ty;P.hop=null;P.squash=1;checkPad();}
function tryHop(dx,dy){if(state!=='play'||P.dead)return;if(P.hop&&P.hop.t<.55)return;if(P.hop)finishHop();
 const nc=clamp(P.col+dx,0,COLS-1),nr=clamp(P.row+dy,0,ROWS-1);if(nc===P.col&&nr===P.row)return;
 P.facing={x:dx,y:dy};const fx=P.x,fy=P.y;P.col=nc;P.row=nr;
 P.hop={t:0,dur:.12,fx,fy,tx:(nc+.5)*cell,ty:(nr+.5)*cell};
 sfxHop(dx,dy);noise=Math.min(120,noise+34);spawnRipple(fx,fy);
 if(nr<P.bestRow){P.bestRow=nr;const z=ZONE[nr];if(z)addScore(z,(nc+.5)*cell,(nr+.5)*cell);}}
function checkPad(){if(P.row!==0)return;const c=P.col;if(!PAD_COLS.includes(c)||pads.includes(c))return;
 pads.push(c);hud();addScore(50,P.x,P.y-cell*.35,'PAD +50');sfxChime(!P.alerted);
 burst(P.x,P.y,22,'#9ff2cf',140,40);burst(P.x,P.y,10,'#ffd98a',100,40);
 if(!P.alerted)addScore(75,P.x,P.y-cell*1.0,'SILENT +75');
 kill('goal');}
function doHush(){if(state!=='play'||P.dead)return;if(P.hushCd>0){sfxDeny();const hb=$('#hushBox');hb.classList.remove('deny');void hb.offsetWidth;hb.classList.add('deny');return;}
 P.hushCd=7;P.cloak=1.6;noise=0;sfxHush();burst(P.x,P.y,14,'#8fe8ff',90,-20);
 floats.push({x:P.x,y:P.y-cell*.55,txt:'hush…',t:0,life:.8,c:'#8fe8ff',size:clamp(13,cell*.3,22)});}
function kill(cause){if(P.dead)return;P.dead=true;P.cause=cause;P.deadT=0;P.hop=null;
 if(cause==='car'){sfxSplat();burst(P.x,P.y,26,'#4fae62',180,320);shake=11;}
 else if(cause==='heron'){sfxCaught();burst(P.x,P.y,18,'#cfd8dc',120,50);shake=8;}}
function startDrown(){if(P.dead)return;P.dead=true;P.cause='water';P.deadT=0;P.sinkY=0;sfxSplash();burst(P.x,P.y,14,'#7fd4ff',120,60);shake=3;}
function respawn(){resetPlayer();noise=0;lastNoiseAlert=time;for(const h of herons){h.state='patrol';}A.blip('sine',500,700,.12,.08);}
function resolveDeath(){const c=P.cause;P.dead=false;
 if(c==='goal'){if(pads.length>=PAD_COLS.length)winGame();else respawn();return;}
 lives--;hud();if(lives<=0)gameOver(c);else respawn();}

/* ================= world stepping ================= */
function stepTraffic(dt){for(const r of vehRows){r.dist+=r.dir*r.speed*cell*spdMul*dt;
 for(const v of r.list){let p=(v.off+r.dir*r.dist)%r.span;if(p<0)p+=r.span;v.x=p-r.M;}}}
function stepPlats(dt){for(const r of plats){r.dist+=r.dir*r.speed*cell*spdMul*dt;
 for(const p of r.list){let q=(p.off+r.dir*r.dist)%r.span;if(q<0)q+=r.span;p.x=q-r.M;}}}
function vehHits(){const hr=cell*.21;for(const r of vehRows){if(r.row!==P.row)continue;
 for(const v of r.list){const x0=v.x+cell*.05,x1=x0+v.len*cell-cell*.1;
  if(P.x+hr>x0&&P.x-hr<x1){if(P.invuln<=0&&!P.dead){kill('car');return;}}}}}
function stepWater(dt){if(P.dead||P.hop)return;if(!isWater(P.row))return;let carrier=null;
 for(const r of plats){if(r.row!==P.row)continue;
  for(const p of r.list){const x0=p.x,x1=x0+p.len*cell*.94;if(P.x>x0+cell*.06&&P.x<x1-cell*.02){carrier=r;break;}}
  if(carrier)break;}
 if(carrier){P.x=clamp(P.x+carrier.dir*carrier.speed*cell*spdMul*dt,cell*.25,bw-cell*.25);}
 else startDrown();}
function honkH(h){if(h.honkCd<=0){sfxHonk();h.honkCd=1.7;}}
function heronsPatrolOnly(dt){for(const h of herons){h.honkCd=Math.max(0,h.honkCd-dt);h.phase+=dt*(2.2+h.speed*3);
 if(h.state!=='patrol')h.state='patrol';
 h.x+=h.dir*h.speed*heronMul*cell*dt;
 if(h.x<cell*.5){h.x=cell*.5;h.dir=1;}else if(h.x>bw-cell*.5){h.x=bw-cell*.5;h.dir=-1;}}}
function heronsPlay(dt){for(const h of herons){h.honkCd=Math.max(0,h.honkCd-dt);
 h.phase+=dt*(2.2+h.speed*3)*(h.state!=='patrol'?1.7:1);
 const canSee=!P.dead&&P.invuln<=0&&!isHidden();let seen=false;
 if(canSee){const dr=h.row-P.row,dx=Math.abs(h.x-P.x)/cell;
  if(Math.abs(dr)<=1&&dx<2.7){let blocked=false;if(dr===0){const hc=Math.floor(h.x/cell),lo=Math.min(hc,P.col),hi=Math.max(hc,P.col);
   for(let c=lo+1;c<hi;c++)if(BUSH.has(h.row+','+c)){blocked=true;break;}}seen=!blocked;}}
 if(seen){if(h.state==='patrol'){h.state='go';h.tx=P.x;honkH(h);P.alerted=true;}
  else if(h.state==='search'){h.state='go';h.tx=P.x;honkH(h);}
  else h.tx=P.x;}
 if(h.state==='patrol'){h.x+=h.dir*h.speed*heronMul*cell*dt;
  if(h.x<cell*.5){h.x=cell*.5;h.dir=1;}else if(h.x>bw-cell*.5){h.x=bw-cell*.5;h.dir=-1;}}
 else if(h.state==='go'){const sp=h.speed*1.8*heronMul*cell*dt,d=h.tx-h.x;h.x+=clamp(d,-sp,sp);
  if(Math.abs(d)<cell*.3){h.state='search';h.tSearch=2.5;}}
 else{h.tSearch-=dt;if(h.tSearch<=0){h.state='patrol';h.dir=h.x<bw/2?1:-1;}}
 if(!P.dead&&P.invuln<=0&&!isHidden()&&h.row===P.row&&Math.abs(h.x-P.x)<cell*.48)kill('heron');}
 if(!P.dead&&noise>62&&time-lastNoiseAlert>1.5){lastNoiseAlert=time;let best=null,bd=1e9;
  for(const h of herons){const d=Math.abs(h.x-P.x)+Math.abs(h.row-P.row)*cell*1.5;if(d<bd){bd=d;best=h;}}
  if(best&&best.state==='patrol'){best.state='go';best.tx=P.x;honkH(best);P.alerted=true;
   rings.push({x:P.x,y:P.y,r:cell*.4,a:.6,amber:true});}}}

/* ================= update ================= */
function update(dt){time+=dt;shake=Math.max(0,shake-dt*30);
 if(state==='play'){
  P.invuln=Math.max(0,P.invuln-dt);P.cloak=Math.max(0,P.cloak-dt);P.hushCd=Math.max(0,P.hushCd-dt);P.squash=Math.max(0,P.squash-dt*5);
  if(P.cloak>0&&!P.dead&&Math.random()<.6)parts.push({x:P.x+rand(-14,14),y:P.y+rand(-12,12),vx:rand(-8,8),vy:rand(-30,-10),g:-10,life:.5,t:0,c:'#8fe8ff',r:1.6});
  if(P.hop){const h=P.hop;h.t+=dt/h.dur;
   if(h.t>=1)finishHop();else{P.x=h.fx+(h.tx-h.fx)*h.t;P.y=h.fy+(h.ty-h.fy)*h.t-Math.sin(Math.PI*h.t)*cell*.32;}}
  if(!P.dead&&held.length){const d=DIRS[held[held.length-1]];if(d)tryHop(d[0],d[1]);}
  stepWater(dt);vehHits();heronsPlay(dt);
  if(!P.dead){noise=Math.max(0,noise-dt*26*(P.hop?1:1.4));}
  idleTick+=dt;if(idleTick>=1){idleTick-=1;score+=1;}
  if(P.dead){P.deadT+=dt;if(P.cause==='water')P.sinkY=Math.min(cell*.5,P.sinkY+dt*cell*.6);
   if(P.deadT>=(P.cause==='goal'?1.05:1.0))resolveDeath();}
 } else if(state!=='pause'){stepTraffic(dt);stepPlats(dt);heronsPatrolOnly(dt);}
 // ambient world always steps for menu/over/win liveness handled above; fx:
 for(let i=parts.length-1;i>=0;i--){const p=parts[i];p.t+=dt;if(p.t>p.life){parts.splice(i,1);continue;}p.vy+=p.g*dt;p.x+=p.vx*dt;p.y+=p.vy*dt;}
 if(parts.length>350)parts.splice(0,parts.length-350);
 for(let i=floats.length-1;i>=0;i--){floats[i].t+=dt;if(floats[i].t>floats[i].life)floats.splice(i,1);}
 for(let i=rings.length-1;i>=0;i--){const r=rings[i];r.r+=cell*1.8*dt;r.a-=dt*.9;if(r.a<=0)rings.splice(i,1);}
}
