function drawFlowerShape(f,scale){
  const k=FLOWERS[f.kind];
  ctx.save();ctx.translate(f.x,f.y);ctx.rotate(f.rot+(scale?f.rot:0));
  ctx.fillStyle=k.p;

  ctx.restore();
}
function draw(){
  const shx=G.shake>0?rand(-G.shake,G.shake)*.5:0,shy=G.shake>0?rand(-G.shake,G.shake)*.5:0;
  ctx.save();ctx.translate(shx,shy);
  const dc=dayColors(G.dayT);
  const bg=ctx.createLinearGradient(0,0,0,H);
  bg.addColorStop(0,`rgb(${dc.top})`);bg.addColorStop(1,`rgb(${dc.bot})`);
  ctx.fillStyle=bg;ctx.fillRect(-20,-20,W+40,H+40);
  if(grassPat){ctx.save();ctx.translate(0,G.scrollPx%160);ctx.fillStyle=grassPat;ctx.fillRect(-6,-170,W+12,H+340);ctx.restore();}

  ctx.globalAlpha=1;
  /* webs */
  for(const w of webs){ctx.save();ctx.translate(w.x,w.y);ctx.rotate(w.rot);
    ctx.lineWidth=1.4;
    for(let i=0;i<8;i++){const a=i*TAU/8;ctx.strokeStyle=`rgba(255,255,255,.4)`;
      ctx.beginPath();ctx.moveTo(0,0);ctx.lineTo(Math.cos(a)*w.r,Math.sin(a)*w.r);ctx.stroke();}
    [.35,.65,.95].forEach((rr,i)=>{ctx.strokeStyle=`rgba(255,255,255,${.4-i*.1})`;
      ctx.beginPath();ctx.arc(0,0,w.r*rr,0,TAU);ctx.stroke();});
    ctx.fillStyle='rgba(255,255,255,.6)';
    for(let i=0;i<4;i++){const a=w.rot+i*1.7;ctx.beginPath();ctx.arc(Math.cos(a)*w.r*.55,Math.sin(a)*w.r*.55,1.6,0,TAU);ctx.fill();}
    ctx.restore();}
  for(const f of flowers)drawFlowerShape(f,0);
  /* motes */
  for(const mo of motes){const p=.7+.3*Math.sin(G.t*6+mo.ph),dy=mo.y+Math.sin(G.t*2.4+mo.ph)*3;
    ctx.fillStyle=`rgba(255,214,64,${.28*p})`;ctx.beginPath();ctx.arc(mo.x,dy,12*p,0,TAU);ctx.fill();
    ctx.strokeStyle='#ffd94d';ctx.lineWidth=2;ctx.beginPath();ctx.arc(mo.x,dy,7,0,TAU);ctx.stroke();
    ctx.fillStyle='#fff8d0';ctx.beginPath();ctx.arc(mo.x,dy,3.6,0,TAU);ctx.fill();}
  /* blooms */
  for(const b of blooms){const k=FLOWERS[b.kind],p=.5+.5*Math.sin(G.t*4+b.rot);
    ctx.strokeStyle=`rgba(255,236,150,${.35+.3*p})`;ctx.lineWidth=3;ctx.beginPath();ctx.arc(b.x,b.y,b.r+8+p*4,0,TAU);ctx.stroke();
    ctx.save();ctx.translate(b.x,b.y);ctx.rotate(b.rot);ctx.fillStyle=k.p;
    for(let i=0;i<6;i++){ctx.save();ctx.rotate(i*TAU/6);ctx.beginPath();ctx.ellipse(0,-b.r*.75,b.r*.42,b.r*.8,0,0,TAU);ctx.fill();ctx.restore();}
    ctx.fillStyle=k.c;ctx.beginPath();ctx.arc(0,0,b.r*.45,0,TAU);ctx.fill();ctx.restore();}
  /* ambers */
  for(const a of ambers){const p=.5+.5*Math.sin(G.t*5);
    ctx.fillStyle=`rgba(255,210,90,${.2+.2*p})`;ctx.beginPath();ctx.arc(a.x,a.y,22+p*5,0,TAU);ctx.fill();
    ctx.save();ctx.translate(a.x,a.y);ctx.rotate(G.t*.8);
    const grd=ctx.createLinearGradient(0,-13,0,13);grd.addColorStop(0,'#ffe27a');grd.addColorStop(1,'#ff9d0a');
    ctx.fillStyle=grd;ctx.strokeStyle='#7a3d0c';ctx.lineWidth=2.5;ctx.beginPath();
    for(let i=0;i<6;i++){const an=i*TAU/6,px=Math.cos(an)*13,py=Math.sin(an)*13;i?ctx.lineTo(px,py):ctx.moveTo(px,py);}
    ctx.closePath();ctx.fill();ctx.stroke();

  for(const w of wasps)drawWasp(w);

  if(G.mode==='play'||G.mode==='over'){

    for(let i=0;i<9;i++){const lx=(i*W/8+Math.sin(i*7.3)*40+W)%W,ly=((G.t*900+i*H/8)%(H+60))-30;ctx.fillRect(lx,ly,2,34);}}
  /* popups */

    ctx.beginPath();ctx.arc(p.x,p.y,p.size,0,TAU);ctx.fill();}
  ctx.restore(); /* end shake */

  if(nf>.03){for(const f of flies){const pu=.4+.6*Math.sin(G.t*f.sp*2+f.ph);

  if(G.flash>0){ctx.fillStyle=`rgba(255,70,45,${G.flash*.32})`;ctx.fillRect(0,0,W,H);}
}
