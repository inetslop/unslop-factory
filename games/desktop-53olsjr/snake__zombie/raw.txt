  for(const s of stones)if(now()-s.riseAt>=RISE_MS&&s.x===nx&&s.y===ny)return die('stone');
  if(!ghost){const lim=growthLeft>0?snake.length:snake.length-1;
    for(let i=0;i<lim;i++)if(snake[i].x===nx&&snake[i].y===ny)return die('self');}
  snake.unshift({x:nx,y:ny});
  if(growthLeft>0)growthLeft--;else snake.pop();
  if(food&&nx===food.x&&ny===food.y)eatFood();
  if(power&&nx===power.x&&ny===power.y)eatPower();
}

  let pts=10,grow=1;

  score+=pts;brains++;growthLeft+=grow;
  texts.push({x:px,y:py-6,txt:'+'+pts,c:f.type==='brain'?'#f4a8c6':f.type==='rotten'?'#b8e070':'#ffd97a',life:950,max:950});
  splash(px,py,f.type==='brain'?['#f4a8c6','#d76f9c']:f.type==='rotten'?['#a8c46b','#6f8f3f']:['#ffe08a','#fff2c0']);
  if(f.type==='rotten'){infection=Math.min(100,infection+34);flash.c='#ff4b2f';flash.t=.3;
    toast('ROTTEN! Infection rising','t-red');sRotten();

}

function eatPower(){
  const k=power.kind,h=snake[0],px=ox+(h.x+.5)*cell,py=oy+(h.y+.5)*cell;
  if(k==='garlic'){infection=Math.max(0,infection-60);toast('GARLIC! Infection purged','t-green');sCure();
    splash(px,py,['#f2efe2','#cfe8b0'],18);}
  else if(k==='wisp'){ghostUntil=now()+3200;toast('GHOST MODE — phase through flesh & fence','t-cyan');sGhost();
    splash(px,py,['#9fe8ff','#5fc8e8'],18);}

    shake(12,450);flash.c='#ff9c3b';flash.t=.28;sBoom();}
  power=null;powerAt=now()+rand(8000,13000);
}

function frame(t){
  requestAnimationFrame(frame);
  let dt=t-lastT;lastT=t;if(dt>60)dt=60;
  // ambient
  for(const b of fog){b.x+=b.v*dt/1000;if(b.x>W+b.r)b.x=-b.r;
    b.y=(b.front?oy+ROWS*cell*.85:oy-cell*.2)+Math.sin(t/2400+b.p)*18;}
  for(let i=particles.length-1;i>=0;i--){const p=particles[i];
    p.x+=p.vx*dt/1000;p.y+=p.vy*dt/1000;p.vy+=p.g*dt/1000;p.life-=dt;
    if(p.life<=0)particles.splice(i,1);}
  if(particles.length>420)particles.splice(0,particles.length-420);
  for(let i=texts.length-1;i>=0;i--){const x=texts[i];x.y-=26*dt/1000;x.life-=dt;if(x.life<=0)texts.splice(i,1);}
  if(shakeT>0)shakeT=Math.max(0,shakeT-dt);
  if(flash.t>0)flash.t=Math.max(0,flash.t-dt/380);

  if(state==='playing'){
    infection=Math.max(0,infection-3.4*dt/1000);
    acc+=dt;
    while(acc>=stepMs){acc-=stepMs;step();if(state!=='playing')break;}
    if(state==='playing'){
      const n=now();
      for(const s of stones){
        if(n-s.riseAt>=RISE_MS){
          if(snake.some(p=>p.x===s.x&&p.y===s.y))s.riseAt+=400;
          else if(!s.clacked){s.clacked=true;sClack();}}
        else if(Math.random()<.2)particles.push({x:ox+(s.x+.5)*cell+rand(-8,8),y:oy+(s.y+.9)*cell,
          vx:rand(-40,40),vy:rand(-70,-10),g:300,life:500,max:500,size:2,c:'#4a3826'});}
      if(power&&n>power.expire){const px=ox+(power.x+.5)*cell,py=oy+(power.y+.5)*cell;
        splash(px,py,['#8a97a5','#5f6b78'],10);power=null;powerAt=n+rand(7000,12000);}
      if(power&&power.kind==='dynamite'&&Math.random()<.15)particles.push({
        x:ox+(power.x+.8)*cell,y:oy+(power.y+.1)*cell,vx:rand(-20,20),vy:rand(-40,-10),
        g:60,life:300,max:300,size:1.5,c:'#ffd23f'});
      if(n>stoneAt&&stones.length<Math.min(3+Math.floor(score/120),14))tryStone(n);
      if(!power&&n>powerAt)spawnPower(n);
      if(food&&food.type==='gold'&&n-food.tBorn>6500){toast('The golden brain crumbled…','t-dim');spawnFood(true);}
      if(n>nextMoan){sMoan();nextMoan=n+rand(7000,16000);}
      if(infection>85&&n>nextBeat){sHeart();nextBeat=n+620;}
    }
  }
  if(state==='dying'&&t-overAt>950)showOver();
  render(t);
  updateHUD(false);
}
