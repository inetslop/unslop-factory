let AC=null, master=null;
function audio(){ if(!AC){ try{ AC=new (window.AudioContext||window.webkitAudioContext)(); master=AC.createGain(); master.gain.value=.32; master.connect(AC.destination);}catch(e){} }
  if(AC&&AC.state==='suspended')AC.resume(); return AC; }
function tone(f,d,type,v,f2,when){ if(!AC||muted)return; type=type||'square'; v=v==null?.4:v; when=when||0;
  const t0=AC.currentTime+when,o=AC.createOscillator(),g=AC.createGain();
  o.type=type; o.frequency.setValueAtTime(f,t0); if(f2)o.frequency.exponentialRampToValueAtTime(Math.max(30,f2),t0+d);
  g.gain.setValueAtTime(v,t0); g.gain.exponentialRampToValueAtTime(.001,t0+d);
  o.connect(g); g.connect(master); o.start(t0); o.stop(t0+d+.03); }
function noise(d,v,when,hp){ if(!AC||muted)return; v=v==null?.4:v; when=when||0; hp=hp||700;
  const t0=AC.currentTime+when,n=Math.floor(AC.sampleRate*d),buf=AC.createBuffer(1,n,AC.sampleRate),ch=buf.getChannelData(0);
  for(let i=0;i<n;i++)ch[i]=(Math.random()*2-1)*(1-i/n);
  const s=AC.createBufferSource();s.buffer=buf;const f=AC.createBiquadFilter();f.type='highpass';f.frequency.value=hp;
  const g=AC.createGain();g.gain.value=v;s.connect(f);f.connect(g);g.connect(master);s.start(t0); }
const sfx={

function makeSnake(len){ const CX=Math.floor(COLS/2),SY=ROWS-3; const cells=[{x:CX,y:SY}]; let x=CX,y=SY,d=-1;
  while(cells.length<len&&y>5){ x+=d; if(x<1||x>COLS-2){ y--; d*=-1; x=(d===1)?1:COLS-2; } cells.push({x,y}); }
  return {cells,prev:cells.map(c=>({x:c.x,y:c.y})),dir:[1,0]}; }

function buildLevel(n){ const p=PATTERNS[(n-1)%PATTERNS.length], cycle=Math.floor((n-1)/PATTERNS.length);
  bricks=[]; let cnt=0;
  p.rows.forEach((h,r)=>{ const full=(h+[...h].reverse().join(''))+new Array(COLS).fill('.').join('');
    for(let c=0;c<COLS;c++){ const ch=full[c]; if(ch==='.')continue;
      if(ch==='#'){ bricks.push({x:c,y:r,t:'h',hp:1,max:1,seed:Math.random()*7,flash:0}); continue; }
      let t=ch; if(t==='b'&&cycle>0&&Math.random()<.25)t='p';
      bricks.push({x:c,y:r,t,hp:TYPES[t].hp,max:TYPES[t].hp,seed:Math.random()*7,flash:0}); cnt++; } });
  remaining=cnt; levelTotal=Math.max(1,cnt);
  levelName=p.name+(cycle?' ·'+(cycle+1):''); setLevel(); setBar(); }

function emptyCell(){ for(let i=0;i<90;i++){ const x=1+Math.floor(Math.random()*(COLS-2)), y=6+Math.floor(Math.random()*(ROWS-5));
  if(bricks.some(b=>b.x===x&&b.y===y))continue;
  if(snake&&snake.cells.some(c=>c.x===x&&c.y===y))continue;
  if(gold&&gold.x===x&&gold.y===y)continue; return {x,y}; } return {x:COLS>>1,y:ROWS-2}; }
function spawnFood(){ const c=emptyCell(); food={x:c.x,y:c.y}; }

  if(food&&snake.cells.some(c=>c.x===food.x&&c.y===food.y))spawnFood(); }

function die(){ lives--; setLives(); flashA=.9; shake=13; sfx.die(); combo=0; comboUI(0);

function applyPower(p){ const hx=snake.cells[0].x+.5, hy=snake.cells[0].y+.5;
  if(p.type==='shield'){ shieldT=6; float(hx,hy-.4,'SHIELD!','#ffd94a',true); sfx.power(); }
  else if(p.type==='shrink'){ const k=Math.min(4,snake.cells.length-3); snake.cells.splice(snake.cells.length-k);
    snake.prev=snake.cells.map(c=>({x:c.x,y:c.y})); float(hx,hy-.4,'SHRINK','#6ee7ff',true); sfx.power(); }
  else if(p.type==='slow'){ slowT=5; float(hx,hy-.4,'SLOW-MO','#b48cff',true); sfx.power(); }
  else { lives=Math.min(5,lives+1); setLives(); float(hx,hy-.4,'+LIFE','#ff6b81',true); sfx.life(); } }

/* ============================== STEP ============================== */
function tickMs(){ return Math.max(80,135-(level-1)*9)*(slowT>0?1.55:1); }
function applyQueue(){ while(dirQ.length){ const d=dirQ.shift();
  if(!(d[0]===-snake.dir[0]&&d[1]===-snake.dir[1])){ snake.dir=d; break; } } }
function brickAt(x,y){ for(const b of bricks)if(b.x===x&&b.y===y)return b; return null; }

function destroyBrick(b,spike){ const i=bricks.indexOf(b); if(i<0)return; bricks.splice(i,1);
  if(b.t!=='h')remaining--; brokenTotal++; setBar();
  combo++; comboT=2.5; sfx.combo(combo);
  const pts=(b.t==='h'?100:TYPES[b.t].pts)+(combo>1?(combo-1)*5:0);
  addScore(pts); comboUI(combo);
  const cx=b.x+.5, cy=b.y+.5;
  shards(cx,cy,b.t==='h'?'#ff3b3b':BCOL[b.t][0],spike?14:9); sparks(cx,cy,'#ffffff',8);
  float(cx,cy,'+'+pts,BTXT[b.t]); shake=Math.max(shake,b.t==='s'||spike?6:3.2);
  if(spike)sfx.shieldBrk(); else sfx.brk(b.t==='b'?0:b.t==='p'?1:2);
  if(Math.random()<.15){ const r=Math.random(); const type=r<.34?'shield':r<.62?'shrink':r<.92?'slow':'life';
    pus.push({x:b.x+.5,y:b.y+.4,type}); }
  if(remaining<=0&&b.t!=='h')levelClear(); }

function doStep(){ applyQueue(); const [dx,dy]=snake.dir; let hx=snake.cells[0].x+dx, hy=snake.cells[0].y+dy;
  // walls bounce
  if(hx<0||hx>=COLS||hy<0||hy>=ROWS){
    if(hx<0){snake.dir=[1,0];wallFx.L=1;} else if(hx>=COLS){snake.dir=[-1,0];wallFx.R=1;} else {snake.dir=[0,dy>0?-1:1];wallFx.T=1;}
    const cx=Math.max(.5,Math.min(COLS-.5,snake.cells[0].x+.5)), cy=snake.cells[0].y+.5;
    sparks(cx,cy,'#4fd8ff',6); sfx.wall(); shake=Math.max(shake,2.5); return; }
  // bricks
  const b=brickAt(hx,hy);
  if(b){ if(b.t==='h'){
      if(shieldT>0)destroyBrick(b,true);
      else { snake.dir=[-dx,-dy]; sparks(hx+.5,hy+.5,'#ff3b3b',7); sfx.wall(); shake=Math.max(shake,4);
        if(invulnT<=0){die();return;} }
    } else { b.hp--;
      if(b.hp<=0)destroyBrick(b,false);
      else { b.flash=1; sparks(hx+.5,hy+.5,BCOL[b.t][1],5); sfx.chew(); shake=Math.max(shake,1.6); } } }
  // self
  const willG=grow>0;
  for(let i=0;i<snake.cells.length;i++){ if(i===snake.cells.length-1&&!willG)continue;
    const c=snake.cells[i]; if(c.x===hx&&c.y===hy){ die(); return; } }
  // move
  snake.prev=snake.cells.map(c=>({x:c.x,y:c.y}));
  snake.cells.unshift({x:hx,y:hy});
  if(grow>0)grow--; else { const t=snake.cells.pop(); snake.prev.push({x:t.x,y:t.y}); }
  // pickups
  if(food&&food.x===hx&&food.y===hy){ addScore(10); sfx.food(); burst(hx+.5,food.y+.5,'#ff7a3d'); float(hx+.5,food.y,'+10','#9dffb8');
    if(snake.cells.length<MAXLEN)grow++; spawnFood(); }
  if(gold&&gold.x===hx&&gold.y===hy){ addScore(75); sfx.gold(); burst(hx+.5,gold.y+.5,'#ffe14a'); float(hx+.5,gold.y,'+75','#ffe14a',true);
    gold=null; goldNext=6+Math.random()*5; }
  for(let i=pus.length-1;i>=0;i--){ const p=pus[i];
    if(Math.round(p.x)===hx&&Math.round(p.y)===hy){ applyPower(p); pus.splice(i,1); } } }

function step(dt){
  if(dying){ deathT-=dt; if(deathT<=0){ if(lives>0)respawn(); else gameOver(); } return; }
  slowT=Math.max(0,slowT-dt); shieldT=Math.max(0,shieldT-dt); invulnT=Math.max(0,invulnT-dt);
  if(combo>0){ comboT-=dt; if(comboT<=0){combo=0;comboUI(0);} }
  if(!gold){ goldNext-=dt; if(goldNext<=0)spawnGold(); }
  else { gold.life-=dt; if(gold.life<=0){ sparks(gold.x+.5,gold.y+.5,'#ffe14a',8); gold=null; goldNext=6+Math.random()*5; } }
  const iv=tickMs(); acc+=dt*1000;
  while(acc>=iv&&mode==='play'&&!dying){ doStep(); acc-=iv; if(mode!=='play')break; }
  for(let i=pus.length-1;i>=0;i--){ pus[i].y+=dt*2.2; if(pus[i].y>ROWS-.5)pus.splice(i,1); }
  if(shieldT>0&&!dying&&snake){ const h=snake.cells[0];
    for(let i=0;i<2;i++)parts.push({type:'dot',x:h.x+.5+(Math.random()-.5)*.4,y:h.y+.5,vx:(Math.random()-.5)*1.2,vy:-1.4-Math.random()*1.6,t:0,life:.35,s:.06+Math.random()*.07,c:['#ffd94a','#ff9d2e','#ff5d2e'][i%3]}); }
}

  if(b.t==='h'){ rr(x+m,y+m,w,h,cell*.1); ctx.fillStyle='#152230'; ctx.fill();
    ctx.strokeStyle='#2b4258'; ctx.lineWidth=1.5; ctx.stroke();
    const pulse=.6+.4*Math.sin(time*5+b.seed), cx=x+cell/2, cy=y+cell/2;

    for(let a=0;a<8;a++){ const ang=a*Math.PI/4+time*.6; const bx=cx+Math.cos(ang)*cell*.2, by=cy+Math.sin(ang)*cell*.2;
      ctx.beginPath(); ctx.moveTo(bx+Math.cos(ang)*cell*.13,by+Math.sin(ang)*cell*.13);

      ctx.lineTo(bx+Math.cos(ang-.5)*cell*.06,by+Math.sin(ang-.5)*cell*.06); ctx.closePath(); ctx.fill(); }

  rr(x+m,y+m,w,h,cell*.12); ctx.fillStyle=base; ctx.fill();

function drawGold(){ if(!gold)return; const cx=ox+(gold.x+.5)*cell, cy=oy+(gold.y+.5)*cell;
  const blink=gold.life<2&&Math.floor(time*8)%2===0; if(blink)return;

  ctx.fillStyle='#ffe14a'; ctx.beginPath();
  for(let i=0;i<10;i++){ const r=i%2?cell*.16:cell*.34, a=i*Math.PI/5-Math.PI/2;
    ctx[i?'lineTo':'moveTo'](Math.cos(a)*r,Math.sin(a)*r); } ctx.closePath(); ctx.fill(); ctx.restore();
  ctx.strokeStyle='rgba(255,225,74,.6)'; ctx.lineWidth=2;
  ctx.beginPath(); ctx.arc(cx,cy,cell*.44,-Math.PI/2,-Math.PI/2+6.283*(gold.life/gold.max)); ctx.stroke(); }

function drawPower(p){ const P=PWR[p.type], cx=ox+p.x*cell, cy=oy+p.y*cell+Math.sin(time*6+p.x)*cell*.05, s=cell*.72;
  ctx.save(); ctx.shadowColor=P.c; ctx.shadowBlur=10;
  rr(cx-s/2,cy-s/2,s,s,cell*.14); ctx.fillStyle='#0c1c28'; ctx.fill();
  ctx.strokeStyle=P.c; ctx.lineWidth=2; ctx.stroke(); ctx.restore();
  ctx.fillStyle=P.c; ctx.font='700 '+Math.floor(cell*.42)+'px '+MONO; ctx.textAlign='center'; ctx.textBaseline='middle';
  ctx.fillText(P.ch,cx,cy+cell*.03); }

function strokePoly(pts,w,style){ if(pts.length<2)return; ctx.strokeStyle=style; ctx.lineWidth=w;
  ctx.beginPath(); ctx.moveTo(pts[pts.length-1].x,pts[pts.length-1].y);
  for(let i=pts.length-2;i>=0;i--)ctx.lineTo(pts[i].x,pts[i].y); ctx.stroke(); }

function drawSnake(){ if(!snake)return; const t=(mode==='play'&&!dying)?Math.min(1,acc/tickMs()):1;
  const pts=[]; for(let i=0;i<snake.cells.length;i++){ const a=snake.prev[i]||snake.cells[i], b=snake.cells[i];
    pts.push({x:ox+(a.x+(b.x-a.x)*t+.5)*cell, y:oy+(a.y+(b.y-a.y)*t+.5)*cell}); }
  ctx.save(); if(invulnT>0&&Math.floor(time*14)%2===0)ctx.globalAlpha=.35;
  ctx.lineCap='round'; ctx.lineJoin='round';
  strokePoly(pts,cell*.8,'#053823'); strokePoly(pts,cell*.62,'#2fd968'); strokePoly(pts,cell*.2,'rgba(170,255,205,.45)');
  for(let i=1;i<pts.length;i+=2){ ctx.fillStyle='rgba(6,52,31,.4)';
    ctx.beginPath(); ctx.arc(pts[i].x,pts[i].y,cell*.09,0,7); ctx.fill(); }
  const h=pts[0], [dx,dy]=snake.dir, px=-dy, py=dx;
  if(shieldT>0){ const g=ctx.createRadialGradient(h.x,h.y,1,h.x,h.y,cell*.95);
    g.addColorStop(0,'rgba(255,210,80,.5)'); g.addColorStop(1,'rgba(255,90,30,0)');
    ctx.fillStyle=g; ctx.beginPath(); ctx.arc(h.x,h.y,cell*.95,0,7); ctx.fill(); }
  ctx.beginPath(); ctx.arc(h.x,h.y,cell*.46,0,7); ctx.fillStyle=shieldT>0?'#ffb84d':'#2fd968'; ctx.fill();
  ctx.strokeStyle='#053823'; ctx.lineWidth=cell*.1; ctx.stroke();
  if((time%2.6)<.35){ const mx=h.x+dx*cell*.4, my=h.y+dy*cell*.4, tx=mx+dx*cell*.34, ty=my+dy*cell*.34;
    ctx.strokeStyle='#ff4d5e'; ctx.lineWidth=2; ctx.beginPath(); ctx.moveTo(mx,my); ctx.lineTo(tx,ty);
    ctx.moveTo(tx,ty); ctx.lineTo(tx+(dx+px)*cell*.1,ty+(dy+py)*cell*.1);
    ctx.moveTo(tx,ty); ctx.lineTo(tx+(dx-px)*cell*.1,ty+(dy-py)*cell*.1); ctx.stroke(); }
  const er=cell*.13;
  for(const s of [1,-1]){ const ex=h.x+dx*cell*.12+px*s*cell*.17, ey=h.y+dy*cell*.12+py*s*cell*.17;
    ctx.fillStyle='#fff'; ctx.beginPath(); ctx.arc(ex,ey,er,0,7); ctx.fill();
    ctx.fillStyle='#04231a'; ctx.beginPath(); ctx.arc(ex+dx*cell*.05,ey+dy*cell*.05,er*.55,0,7); ctx.fill(); }
  ctx.restore(); }

function drawParts(){ for(const p of parts){ const a=1-p.t/p.life;
  if(p.type==='shard'){ ctx.save(); ctx.translate(ox+p.x*cell,oy+p.y*cell); ctx.rotate(p.rot);
    ctx.globalAlpha=a; ctx.fillStyle=p.c; const s=p.s*cell; ctx.fillRect(-s/2,-s/2,s,s*.7); ctx.restore(); }
  else { ctx.save(); ctx.globalCompositeOperation='lighter'; ctx.globalAlpha=a;
    if(p.type==='spark'){ ctx.strokeStyle=p.c; ctx.lineWidth=2; ctx.beginPath();
      const X=ox+p.x*cell,Y=oy+p.y*cell; ctx.moveTo(X,Y); ctx.lineTo(X-p.vx*cell*.05,Y-p.vy*cell*.05); ctx.stroke(); }
    else { ctx.fillStyle=p.c; ctx.beginPath(); ctx.arc(ox+p.x*cell,oy+p.y*cell,p.s*cell,0,7); ctx.fill(); }
    ctx.restore(); } } ctx.globalAlpha=1; }

function drawFloats(){ for(const f of floats){ const a=1-f.t/f.life, Y=oy+(f.y-f.t*1.1)*cell, X=ox+f.x*cell;
  ctx.save(); ctx.globalAlpha=Math.min(1,a*1.4); ctx.textAlign='center'; ctx.textBaseline='middle';
  ctx.font=(f.big?'900 '+Math.floor(cell*.8)+'px ':'700 '+Math.floor(cell*.55)+'px ')+(f.big?DISP:MONO);
  ctx.lineWidth=3; ctx.strokeStyle='rgba(4,16,24,.8)'; ctx.strokeText(f.txt,X,Y); ctx.fillStyle=f.c; ctx.fillText(f.txt,X,Y);
  ctx.restore(); } }

function render(){
  ctx.fillStyle='#071420'; ctx.fillRect(0,0,W,H);
  const fw=cell*COLS,fh=cell*ROWS, cx=W/2;
  let g=ctx.createRadialGradient(cx,oy+cell,1,cx,oy+cell,fw*.8);
  g.addColorStop(0,'rgba(255,120,60,.10)'); g.addColorStop(1,'rgba(255,120,60,0)'); ctx.fillStyle=g; ctx.fillRect(0,0,W,H);
  g=ctx.createRadialGradient(cx,oy+fh+cell,1,cx,oy+fh+cell,fw*.7);
  g.addColorStop(0,'rgba(40,200,160,.08)'); g.addColorStop(1,'rgba(40,200,160,0)'); ctx.fillStyle=g; ctx.fillRect(0,0,W,H);
  // embers
  ctx.save(); ctx.globalCompositeOperation='lighter';
  for(const e of embers){ ctx.fillStyle=e.warm?'rgba(255,150,70,'+e.a+')':'rgba(90,200,255,'+(e.a*.8)+')';
    ctx.beginPath(); ctx.arc(e.x,e.y,e.r,0,7); ctx.fill(); } ctx.restore();
  // grid
  ctx.strokeStyle='rgba(90,180,220,.05)'; ctx.lineWidth=1; ctx.beginPath();
  for(let i=1;i<COLS;i++){ ctx.moveTo(ox+i*cell+.5,oy); ctx.lineTo(ox+i*cell+.5,oy+fh); }
  for(let j=1;j<ROWS;j++){ ctx.moveTo(ox,oy+j*cell+.5); ctx.lineTo(ox+fw,oy+j*cell+.5); }
  ctx.stroke();
  // floor line + safe glow
  const fg=ctx.createLinearGradient(0,oy+fh-cell*.6,0,oy+fh);
  fg.addColorStop(0,'rgba(125,255,176,0)'); fg.addColorStop(1,'rgba(125,255,176,.09)');
  ctx.fillStyle=fg; ctx.fillRect(ox,oy+fh-cell*.6,fw,cell*.6);
  // walls
  ctx.fillStyle='#14384a';
  ctx.fillRect(ox-7,oy-7,5,fh+14); ctx.fillRect(ox+fw+2,oy-7,5,fh+14); ctx.fillRect(ox-7,oy-9,fw+14,6);
  if(wallFx.L>0){ctx.fillStyle='rgba(90,220,255,'+wallFx.L*.5+')';ctx.fillRect(ox-7,oy-7,5,fh+14);}
  if(wallFx.R>0){ctx.fillStyle='rgba(90,220,255,'+wallFx.R*.5+')';ctx.fillRect(ox+fw+2,oy-7,5,fh+14);}
  if(wallFx.T>0){ctx.fillStyle='rgba(90,220,255,'+wallFx.T*.5+')';ctx.fillRect(ox-7,oy-9,fw+14,6);}
  ctx.save(); ctx.setLineDash([7,11]); ctx.lineDashOffset=-time*46;
  ctx.strokeStyle='rgba(90,220,255,.3)'; ctx.lineWidth=2;
  ctx.beginPath(); ctx.moveTo(ox-1,oy+fh); ctx.lineTo(ox-1,oy-1); ctx.lineTo(ox+fw+1,oy-1); ctx.lineTo(ox+fw+1,oy+fh); ctx.stroke();
  ctx.restore();
  // world
  for(const b of bricks)drawBrick(b);
  drawFood(); drawGold(); for(const p of pus)drawPower(p);
  if(!dying)drawSnake();
  drawParts(); drawFloats();
  // vignette + damage flash
  ctx.fillStyle=vig; ctx.fillRect(0,0,W,H);
  if(flashA>0){ ctx.fillStyle='rgba(255,45,45,'+(flashA*.32)+')'; ctx.fillRect(0,0,W,H); }
}

/* ============================== LOOP ============================== */
function frame(now){ requestAnimationFrame(frame);
  let dt=(now-lastT)/1000; lastT=now; if(dt>.05)dt=.05; time+=dt;
  if(mode==='play')step(dt);
  else if(mode==='clear'){ clearT-=dt;
    if(clearT<=0){ level++; buildLevel(level);
      snake=makeSnake(Math.min(snake.cells.length,40)); invulnT=1; acc=0; mode='play';
      if(food&&snake.cells.some(c=>c.x===food.x&&c.y===food.y))spawnFood();
      banner('LEVEL '+level,levelName); } }
  if(mode!=='paused'){
    for(let i=parts.length-1;i>=0;i--){ const p=parts[i]; p.t+=dt;
      if(p.t>p.life){parts.splice(i,1);continue;}
      p.x+=p.vx*dt; p.y+=p.vy*dt; if(p.type==='shard')p.vy+=6*dt; if(p.rot!==undefined)p.rot+=p.vr*dt; }
    if(parts.length>400)parts.splice(0,parts.length-400);
    for(let i=floats.length-1;i>=0;i--){ floats[i].t+=dt; if(floats[i].t>floats[i].life)floats.splice(i,1); }
    for(const b of bricks)if(b.flash>0)b.flash=Math.max(0,b.flash-dt*5);
  }
  for(const e of embers){ e.y-=e.v*dt; e.x+=Math.sin(time*e.f+e.ph)*9*dt; if(e.y<-12)Object.assign(e,newEmber(false)); }
  shake=Math.max(0,shake*(1-7*dt)); flashA=Math.max(0,flashA-dt*1.5);
  for(const s of ['L','R','T'])wallFx[s]=Math.max(0,wallFx[s]-dt*3);
  if(shake>0)ctx.translate((Math.random()-.5)*shake,(Math.random()-.5)*shake);
  render();
}
