      var d=Math.hypot(P.x-p.x,P.y-p.y);
      if(p.pr<d&&p.r>=d&&d<1000){
        det=Math.min(99,det+16);blip=3;S.ping();pings.push({x:P.x,y:P.y,r:10,a:1});shake=Math.max(shake,0.08);
      }
    }
    if(p.r>1050)pulses.splice(i,1);}
}
function updDrones(dt){
  for(var i=0;i<drones.length;i++){var d=drones[i];
    if(d.fz>0)d.fz-=dt;
    else{d.x+=d.dir*d.sp*dt;if(d.x>d.x1){d.x=d.x1;d.dir=-1;}if(d.x<d.x0){d.x=d.x0;d.dir=1;}}
    d.ft+=dt;if(d.ft>0.32){d.ft=0;d.fr^=1;}
    if(state==='play'&&d.fz<=0){
      var dy=(d.y+Math.sin(time*2+d.wob)*9)-P.y, dx=d.x-P.x, dd=Math.hypot(dx,dy);
      if(dd<95){det+=46*detScale*dt;frameSeen=true;}
      if(P.iframes<=0&&circleRect(P.x,P.y,P.r,d.x-17,d.y+Math.sin(time*2+d.wob)*9-13,34,26))hitPlayer();
    }
  }
}

function step(dt){
  var i,t;
  for(i=0;i<turrets.length;i++){t=turrets[i];if(t.stun>0)t.stun-=dt;
    t.ang=t.base+Math.sin(time*TAU/t.per+t.ph)*t.amp;}
  updParts(dt);
  for(i=pings.length-1;i>=0;i--){var pg=pings[i];pg.r+=260*dt;pg.a-=dt*1.4;if(pg.a<=0)pings.splice(i,1);}
  for(i=emps.length-1;i>=0;i--){var em=emps[i];em.r+=520*dt;if(em.r>190)emps.splice(i,1);}
  shake=Math.max(0,shake-dt*1.8);flash=Math.max(0,flash-dt*1.6);empFlash=Math.max(0,empFlash-dt);
  blip=Math.max(0,blip-dt);

  if(state==='start'||state==='clear'){ updDrones(dt);updPulses(dt);
    if(state==='clear'){clearT-=dt;if(clearT<=0)nextSector();} return; }
  if(state!=='play')return;

  sectorTime+=dt;
  /* --- movement --- */
  var g=keys['ShiftLeft']||keys['ShiftRight']||keys['KeyC'];
  if(g!==ghost){ghost=g;S.ghost(g);P.trail=[];}
  var ix=((keys['ArrowRight']||keys['KeyD'])?1:0)-((keys['ArrowLeft']||keys['KeyA'])?1:0),
      iy=((keys['ArrowDown']||keys['KeyS'])?1:0)-((keys['ArrowUp']||keys['KeyW'])?1:0);
  if(ix&&iy){ix*=0.7071;iy*=0.7071;}
  var sp=ghost?125:245, k=1-Math.exp(-14*dt);
  P.vx+=(ix*sp-P.vx)*k; P.vy+=(iy*sp-P.vy)*k;
  P.x+=P.vx*dt; P.y+=P.vy*dt;
  var m=BD+P.r; P.x=clamp(P.x,m,W-m); P.y=clamp(P.y,m,H-m);
  for(i=0;i<plats.length;i++){var q=plats[i];
    var cx=clamp(P.x,q.x,q.x+q.w),cy=clamp(P.y,q.y,q.y+q.h),dx=P.x-cx,dy=P.y-cy,d2=dx*dx+dy*dy;
    if(d2<P.r*P.r){
      if(d2<1e-4){var l=P.x-q.x,r=q.x+q.w-P.x,tp=P.y-q.y,bt=q.y+q.h-P.y,mn=Math.min(l,r,tp,bt);
        if(mn===l)P.x=q.x-P.r;else if(mn===r)P.x=q.x+q.w+P.r;else if(mn===tp)P.y=q.y-P.r;else P.y=q.y+q.h+P.r;}
      else{var dd=Math.sqrt(d2);P.x=cx+dx/dd*P.r;P.y=cy+dy/dd*P.r;}
    }}
  var spd=Math.hypot(P.vx,P.vy);
  if(spd>30)P.face=Math.atan2(P.vy,P.vx);
  P.iframes=Math.max(0,P.iframes-dt);
  P.energy=clamp(P.energy+9*dt,0,100);
  P.trailT-=dt;
  if(P.trailT<=0&&(ghost||spd>40)){P.trail.push({x:P.x,y:P.y,a:ghost?0.5:0.28,f:P.face});
    if(P.trail.length>10)P.trail.shift();P.trailT=0.04;}
  for(i=P.trail.length-1;i>=0;i--){P.trail[i].a-=dt*(ghost?1.6:2.2);if(P.trail[i].a<=0)P.trail.splice(i,1);}

  /* --- detection --- */
  frameSeen=false; detScale=1+(sector-1)*0.05;
  for(i=0;i<turrets.length;i++){t=turrets[i];t.see=false;
    if(t.stun>0)continue;
    var tdx=P.x-t.x,tdy=P.y-t.y,td=Math.hypot(tdx,tdy);
    if(td<t.len+4){var ta=Math.atan2(tdy,tdx);
      if(angD(ta,t.ang)<t.half+P.r/td && !blocked(t.x,t.y,P.x,P.y)){
        var f=1-0.45*(td/t.len); if(ghost)f*=0.4; if(spd>175)f*=1.35;
        det+=52*detScale*f*dt; frameSeen=true; t.see=true;
      }}
  }
  updDrones(dt);
  updPulses(dt);

  if(frameSeen){ if(!alarmT&&det>seenTick+8){seenTick=Math.floor(det/8)*8;S.tick();} }
  else if(alarmT<=0){ det=Math.max(0,det-9*dt); seenTick=8; }

  if(!alarmT&&det>=100){ alarmT=5;sirenT=0;sirenHi=false;S.noise(0.5,0.6,400);shake=Math.max(shake,0.3); }
  if(alarmT>0){
    alarmT-=dt; sirenT-=dt;
    if(sirenT<=0){sirenT=0.28;S.siren(sirenHi);sirenHi=!sirenHi;}
    var list=[];
    for(i=0;i<turrets.length;i++){t=turrets[i];if(t.stun>0)continue;
      list.push({t:t,d:(P.x-t.x)*(P.x-t.x)+(P.y-t.y)*(P.y-t.y)});}
    list.sort(function(a,b){return a.d-b.d;});
    for(i=0;i<list.length&&i<5;i++){var o=list[i];if(o.d>810000)continue;
      o.t.fireCd-=dt;
      if(o.t.fireCd<=0){o.t.fireCd=0.85+R(-0.1,0.2);
        var a=Math.atan2(P.y-o.t.y,P.x-o.t.x)+R(-0.13,0.13);
        bullets.push({x:o.t.x+Math.cos(a)*16,y:o.t.y+Math.sin(a)*16,vx:Math.cos(a)*340,vy:Math.sin(a)*340});
        S.shot(); o.t.fireA=a;}
    }
    if(alarmT<=0){det=42;seenTick=40;}
  }

  /* --- bullets --- */
  for(i=bullets.length-1;i>=0;i--){var b=bullets[i];b.x+=b.vx*dt;b.y+=b.vy*dt;
    var dead=b.x<-20||b.x>W+20||b.y<-20||b.y>H+20;
    if(!dead)for(var j=0;j<plats.length;j++){q=plats[j];
      if(b.x>q.x&&b.x<q.x+q.w&&b.y>q.y&&b.y<q.y+q.h){dead=true;burst(b.x,b.y,5,'#ffcf6a',120);break;}}
    if(!dead&&P.iframes<=0&&Math.hypot(b.x-P.x,b.y-P.y)<P.r+4){dead=true;hitPlayer();}
    if(dead)bullets.splice(i,1);}

  /* --- exit --- */
  if(P.x>W-58)doClear();
}
