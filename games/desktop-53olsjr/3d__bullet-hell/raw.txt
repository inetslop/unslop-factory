function addB(x,y,z,vx,vy,vz,r,col,seek){ if(bullets.length>=MAXB) return; bullets.push({x,y,z,vx,vy,vz,r,col,seek:!!seek,t:Math.random()*6,px:x,py:y,pz:z,nm:false}); }
function aimed(e,ang,S,r,col){ const dx=P.x-e.x, dy=P.y-e.y, d=Math.hypot(dx,dy)||1, T=d/S+.05;
  addB(e.x,e.y,e.z,Math.cos(ang)*S,Math.sin(ang)*S,-(e.z-ZP)/T,r,col); }
function radialAt(x,y,z,n,S,vz,r,col,ph){ for(let i=0;i<n;i++){ const a=(ph||0)+i*Math.PI*2/n; addB(x,y,z,Math.cos(a)*S,Math.sin(a)*S,vz,r,col); } }
function spawnWall(z,gapX,speed){ for(let x=-240;x<=240;x+=30){ if(Math.abs(x-gapX)<speed? 52:48) continue; addB(x,60+Math.random()*240,z,(Math.random()-.5)*30,(Math.random()-.5)*30,-(z-ZP)/3.2,8,'#ffb52e'); } }
function spawnSeekers(n){ for(let i=0;i<n;i++){ const side=i%2?1:-1; addB(side*210,70,1000,side*-40,60,-300,9,'#ffd24a',true); } }

function mkEnemy(type,x,y,z){
  const E={type,x,y,z,t:Math.random()*6,phase:Math.random()*6.28,delay:0,hit:0};
  if(type==='drone'){ E.hp=E.maxhp=3; E.r=24; E.vz=-(320+levelN*10); E.x0=x; }
  if(type==='mini'){ E.hp=E.maxhp=2; E.r=15; E.vz=-400; E.x0=x; }
  if(type==='sweeper'){ E.hp=E.maxhp=4; E.r=26; E.vz=-270; E.x0=x; }
  if(type==='orbiter'){ E.hp=E.maxhp=7; E.r=30; E.vz=-340; E.tz=760; E.life=6.5; }
  if(type==='splitter'){ E.hp=E.maxhp=5; E.r=28; E.vz=-300; }
  return E;
}
function spawnEnemy(type,x,y,z,delay){ if(enemies.length>=11) return; const e=mkEnemy(type,x,y,z); e.delay=delay||0; enemies.push(e); }

/* --- spawn director --- */
function director(d){
  if(boss||breather>0) return;
  spawnT-=d; if(spawnT>0) return;
  const L=levelN, r=Math.random();
  if(L>=3 && r<.14) spawnWall(1200,(Math.random()*400-200),260);
  else if(r<.30){ for(let i=0;i<3;i++) spawnEnemy('drone',-180+i*180+(Math.random()*.6-.3)*60,80+Math.random()*180,1500+i*260,i*.35); }
  else if(r<.48) radialAt(Math.random()*300-150,70+Math.random()*200,1500,16,150,-(1500-ZP)/4.2,7,['#ff3d7e','#46ffc8'][levelN%2],Math.random()*6);
  else if(r<.64){ spawnEnemy('sweeper',-260,90+Math.random()*160,1500); spawnEnemy('sweeper',260,90+Math.random()*160,1500,.3); }
  else if(r<.78) spawnEnemy('orbiter',Math.random()*300-150,120+Math.random()*140,1600);
  else if(r<.9){ for(let i=0;i<3;i++) spawnEnemy('splitter',-120+i*120+(Math.random()*.5-.25)*40,100+Math.random()*160,1500,i*.3); }
  else spawnSeekers(2+(L>4?1:0));
  spawnT=Math.max(.55,1.9-L*.12)*(.7+Math.random()*.6);
}

/* --- boss --- */
function spawnBoss(){
  sAlarm(); addPop(W/2,H*.3,'⚠ PULSAR CORE INBOUND ⚠','#ff3d7e',true);
  boss={x:0,y:175,z:950,t:0,hp:130+levelN*70,maxhp:130+levelN*70,r:150,fA:1.4,fB:2.8,ph:-1,flash:0};
}
function bossUpdate(d){
  const b=boss; b.t+=d; b.flash=Math.max(0,b.flash-d*3);
  b.z+=(430-b.z)*Math.min(1,d*1.3);
  b.x=Math.sin(b.t*.55)*170; b.y=175+Math.sin(b.t*.9)*45;
  const f=b.hp/b.maxhp, ph=f>.66?0:f>.33?1:2;
  if(ph!==b.ph){ b.ph=ph; b.flash=1; sAlarm(); addRingFx(b.x,b.y,b.z,20,'#ff3d7e'); addPop(W/2,H*.3,ph===2?'⚠ CORE ENRAGED ⚠':'CORE PHASE '+(ph+1),'#ffb52e',true); }
  if(b.z>520) return;
  if(ph===0){ b.fA-=d; if(b.fA<=0){ b.fA=.42; for(let i=0;i<4;i++) aimed(b,i*Math.PI/2+b.t*1.7,200,6,'#ff3d7e'); }
    b.fB-=d; if(b.fB<=0){ b.fB=2.6; const a=Math.atan2(P.y-b.y,P.x-b.x); for(let i=-2;i<=2;i++) aimed(b,a+i*.15,280,5,'#ff5c39'); } }
  else if(ph===1){ b.fA-=d; if(b.fA<=0){ b.fA=.85; const n=16; for(let i=0;i<n;i++){ const a=i/n*Math.PI*2+b.t, S=i%2?240:170; addB(b.x,b.y,b.z,Math.cos(a)*S,Math.sin(a)*S,-(b.z-ZP)/3.4,6,i%2?'#ffb52e':'#46ffc8'); } }
    b.fB-=d; if(b.fB<=0){ b.fB=3.2; spawnSeekers(2); } }
  else { b.fA-=d; if(b.fA<=0){ b.fA=.3; for(let i=0;i<6;i++) aimed(b,i*Math.PI/3+b.t*2.4,270,5,'#ff5c39'); }
    b.fB-=d; if(b.fB<=0){ b.fB=4.5; spawnWall(1150,(Math.random()*360-180),260); shake(5,.4);} }
}

/* ================= FX HELPERS ================= */
function boom(x,y,z,n,col,big){
  for(let i=0;i<n;i++){ const a=Math.random()*6.28, sp=60+Math.random()*(big?380:240);
    parts.push({x,y,z,vx:Math.cos(a)*sp,vy:Math.sin(a)*sp,vz:(Math.random()-.5)*220,life:.4+Math.random()*.5,tl:.9,r:2+Math.random()*3,col}); }
  addRingFx(x,y,z,big?18:8,big?'#ffffff':'#ffd9a8');
}
function addRingFx(x,y,z,r,col){ if(ringsFx.length>40) return; ringsFx.push({x,y,z,r,vr:340,life:.5,tl:.5,col}); }
function addPop(sx,sy,text,col,big){ pops.push({sx,sy,text,col,life:big?1.6:1,tl:big?1.6:1,big:!!big}); }
function shake(m,t){ if(m>=shakeMag*(shakeT>0?shakeT/.5:1)){ shakeMag=m; shakeT=.5; } else { shakeMag=Math.max(shakeMag,m); shakeT=Math.max(shakeT,.5);} }
function dropPower(x,y,z,force){
  const r=Math.random(), t=force||(r<.4?'P':r<.65?'S':r<.8?'B':'G');
  powers.push({x,y,z,t,rot:Math.random()*6});
}

/* ================= GAME FLOW ================= */
function resetGame(){
  bullets.length=0; shots.length=0; enemies.length=0; parts.length=0; ringsFx.length=0; pops.length=0; powers.length=0; fxQ.length=0;
  boss=null; nextBossAt=45; breather=0;
  score=0; lives=3; bombs=3; spread=1; shield=false; combo=0; comboT=0;
  elapsed=0; levelN=1; spawnT=1.4; fireT=0; bombT=0; flashT=0; slowT=0; deadT=0; overT=0; shakeT=0;
  P={x:0,y:180,vx:0,vy:0,tilt:0,dead:false,invuln:2};
}
function startGame(){
  initAudio(); sUI(); resetGame();
  state='play'; paused=false;
  $('menu').classList.add('hidden'); $('over').classList.add('hidden'); $('pause').classList.add('hidden'); $('hud').classList.remove('hidden');
  addPop(W/2,H*.32,'WAVE 01','#3ce8ff',true);
}
function gameOver(){
  state='over'; overT=0;
  if(score>best){ best=score; saveBest(); }
  $('finalScore').textContent=fmt(score);
  $('overBest').textContent='BEST '+fmt(best);
  $('newRec').classList.toggle('show', score>=best && score>0);
  $('menuBest').textContent=fmt(best); $('hudBest').textContent=fmt(best);
  $('over').classList.remove('hidden');
}
function togglePause(){ if(state!=='play') return; paused=!paused; $('pause').classList.toggle('hidden',!paused); sUI(); }
function fmt(n){ return n.toLocaleString('en-US'); }

/* ================= PLAYER ================= */
function doBomb(){
  if(bombs<=0||bombT>0) return;
  bombs--; bombT=.8; flashT=Math.max(flashT,.5); shake(9,.6); sBomb();
  let c=0;
  for(let i=bullets.length-1;i>=0;i--){ const b=bullets[i]; if(b.z>ZP&&c<260){ boom(b.x,b.y,b.z,1,'#bfeaff',false); c++; } bullets.splice(i,1); }
  for(const e of enemies.slice()) hitEnemy(e,30,null,true);
  if(boss) hitBoss(30);
}
function playerHit(){
  if(P.invuln>0||P.dead) return;
  if(shield){ shield=false; P.invuln=1.3; flashT=Math.max(flashT,.25); shake(4,.3); sBoom(); addRingFx(P.x,P.y,ZP,16,'#5dff9a'); return; }
  lives--; combo=0; P.dead=true; deadT=1.5; slowT=.7; shake(10,.8); flashT=.6; sDeath();
  boom(P.x,P.y,ZP,40,'#3ce8ff',true); boom(P.x,P.y,ZP,24,'#ffffff',true);
  for(let i=bullets.length-1;i>=0;i--){ const b=bullets[i]; if(b.z>ZP) boom(b.x,b.y,b.z,1,'#9fd8ff',false); bullets.splice(i,1); }
}
function nearMiss(ix,iy){ score+=50; comboT=Math.max(comboT,.5); addPop(proj(ix,iy,ZP)[0],proj(ix,iy,ZP)[1],'CLOSE +50','#bfeaff',false); sNear(); }

/* ================= DAMAGE / KILLS ================= */
function hitEnemy(e,dmg,s,fromBomb){
  e.hp-=dmg; e.hit=1;
  if(!fromBomb&&s) parts.push({x:s.x,y:s.y,z:s.z,vx:0,vy:0,vz:0,life:.15,tl:.15,r:3,col:'#fff'});
  sHit();
  if(e.hp<=0) killEnemy(e);
}
function hitBoss(dmg){ if(!boss||boss.z>600) return; boss.hp-=dmg; boss.flash=1; sHit();
  if(boss.hp<=0) killBoss(); }
function killBoss(){
  const b=boss, sx=W/2+(b.x-camX)*FOCAL/b.z, sy=HORIZON+(b.y-camY)*FOCAL/b.z;
  addPop(sx,sy,'+6000','#ffb52e',false); score+=6000;
  sBig(); shake(14,.9); flashT=.7;
  for(let i=0;i<3;i++) fxQ.push({t:i*.28,fn:()=>{ boom(b.x+(Math.random()-.5)*160,b.y+(Math.random()-.5)*120,b.z,34,i===2?'#ffffff':'#ff3d7e',true); sBig(); }});
  dropPower(b.x-60,b.y,b.z,'P'); dropPower(b.x+60,b.y,b.z, Math.random()<.5?'S':'B');
  boss=null; nextBossAt=elapsed+45; breather=3.2; addPop(W/2,H*.3,'CORE DESTROYED','#5dff9a',true);
}
function killEnemy(e){
  const i=enemies.indexOf(e); if(i<0) return; enemies.splice(i,1);
  combo=Math.min(8,combo+1); comboT=2;
  const base={drone:100,mini:50,sweeper:150,orbiter:250,splitter:80}[e.type]||100;
  const pts=base*combo; score+=pts;
  const p=proj(e.x,e.y,e.z); addPop(p[0],p[1],'+'+fmt(pts), combo>1?'#ff3d7e':'#bfeaff',false);
  boom(e.x,e.y,e.z,e.type==='orbiter'?26:14,['#ff3d7e','#ffb52e','#46ffc8'][Math.floor(Math.random()*3)], e.type==='orbiter');
  sBoom(); shake(e.type==='orbiter'?3:1.6,.25);
  if(e.type==='splitter'){ spawnEnemy('mini',e.x-14,e.y-10,e.z,0); spawnEnemy('mini',e.x+14,e.y+10,e.z,0); }
  if(state==='play'&&Math.random()<.13) dropPower(e.x,e.y,e.z,null);
}

/* ================= UPDATE ================= */
function level(){ return 1+Math.floor(elapsed/22); }
function update(d,dt){
  gTime+=dt;
  const spd = state==='play'?560+levelN*30:260;
  for(const s of stars){ s.z-=spd*d*(.5+s.s*.4); if(s.z<24){ s.x=(Math.random()*2-1)*900; s.y=-500+Math.random()*1000; s.z=ZFAR; } }
  for(const r of ringsT){ r.z-=spd*d; if(r.z<ZP*.6) r.z+=ZFAR; }
  for(const st of streaks){ st.z-=spd*1.4*d; if(st.z<ZP*.7) st.z+=ZFAR; }

  if(state==='menu'){
    attT-=d;
    if(attT<=0){ attT=1.4+Math.random(); const cols=['#3ce8ff','#ff3d7e','#ffb52e','#46ffc8'];
      radialAt(Math.random()*300-150,70+Math.random()*200,1500,16,150,-(1500-ZP)/4.2,7,cols[attCol++%4],Math.random()*6);
      if(Math.random()<.5) spawnEnemy('drone',Math.random()*380-190,80+Math.random()*180,1500); }
  }

  if(state==='play'){
    elapsed+=d;
    const L=levelN; levelN=level();
    if(levelN!==L){ addPop(W/2,H*.32,'WAVE '+String(levelN).padStart(2,'0'),'#3ce8ff',true); sLevel(); }
    if(!boss&&elapsed>nextBossAt) spawnBoss();
    breather=Math.max(0,breather-d);
    director(d);
    if(boss) bossUpdate(d);
    updatePlayer(d,dt);
  }

  /* enemies */
  for(let i=enemies.length-1;i>=0;i--){
    const e=enemies[i];
    if(e.delay>0){ e.delay-=d; continue; }
    e.t+=d; e.hit=Math.max(0,e.hit-d*4);
    if(e.type==='drone'||e.type==='mini'){ e.z+=e.vz*d; e.x=e.x0+Math.sin(e.t*2+e.phase)*18; }
    else if(e.type==='sweeper'){ e.z+=e.vz*d; e.x=Math.max(-245,Math.min(245,e.x0+Math.sin(e.t*1.5+e.phase)*230)); }
    else if(e.type==='orbiter'){
      if(e.z>e.tz) e.z+=e.vz*d; else { e.life-=d; e.phase+=d*1.9;
        if(Math.floor(e.t*3.8)!==Math.floor((e.t-d)*3.8)){ for(let k=0;k<3;k++) aimed(e,e.phase+k*Math.PI*2/3,185,6,'#46ffc8'); }
        if(e.life<=0) e.vz=-320; }
      if(e.z<e.tz-40) e.z+=e.vz*d;
    }
    else if(e.type==='splitter'){ e.z+=e.vz*d; }
    if(state==='play'&&!P.dead){
      if((e.type==='drone'||e.type==='mini')&&e.z<1500&&e.z>ZP+60){
        e.ft=(e.ft||2)-d; if(e.ft<=0){ e.ft=e.type==='mini'?1.4:1.7; const a=Math.atan2(P.y-e.y,P.x-e.x), S=e.type==='mini'?270:230;
          for(let k=-1;k<=1;k++) aimed(e,a+k*.22,S,6,'#ff3d7e'); } }
      if(e.type==='sweeper'){ e.ft=(e.ft||1.4)-d; if(e.ft<=0&&e.z<1500){ e.ft=1.1; const a=Math.atan2(P.y-e.y,P.x-e.x); aimed(e,a,330,5,'#ff5c39'); } }
    }
    if(e.z<ZP-24) enemies.splice(i,1);
  }

  /* bullets */
  for(let i=bullets.length-1;i>=0;i--){
    const b=bullets[i]; b.px=b.x; b.py=b.y; b.pz=b.z; b.t+=d;
    if(b.seek&&b.z>ZP+40){ const dx=P.x-b.x, dy=P.y-b.y, dist=Math.hypot(dx,dy)||1;
      b.vx+=dx/dist*700*d; b.vy+=dy/dist*700*d; const sp=Math.hypot(b.vx,b.vy);
      if(sp>300){ b.vx*=300/sp; b.vy*=300/sp; }
      if(Math.random()<.5) parts.push({x:b.x,y:b.y,z:b.z,vx:0,vy:0,vz:0,life:.2,tl:.2,r:3,col:'#ffd24a'}); }
    b.x+=b.vx*d; b.y+=b.vy*d; b.z+=b.vz*d;
    if(state==='play'&&!P.dead&&b.pz>=ZP&&b.z<ZP){
      const t=(b.pz-ZP)/((b.pz-b.z)||1e-6), ix=b.px+(b.x-b.px)*t, iy=b.py+(b.y-b.py)*t;
      const dx=ix-P.x, dy=iy-P.y, dd2=dx*dx+dy*dy;
      if(dd2<(b.r+PHIT)*(b.r+PHIT)) playerHit();
      else if(!b.nm&&dd2<(b.r+NEAR)*(b.r+NEAR)){ b.nm=true; nearMiss(ix,iy); } }
    if(b.z<ZP-16||b.z>ZFAR+400) bullets.splice(i,1);
  }

  /* shots */
  for(let i=shots.length-1;i>=0;i--){
    const s=shots[i]; s.px=s.x;s.py=s.y;s.pz=s.z; s.x+=s.vx*d; s.y+=s.vy*d; s.z+=s.vz*d;
    let dead=s.z>ZFAR+200||Math.abs(s.x)>750||s.y<-300||s.y>640;
    if(!dead){
      for(let j=enemies.length-1;j>=0;j--){ const e=enemies[j];
        if(Math.abs(s.z-e.z)<28){ const dx=s.x-e.x,dy=s.y-e.y;
          if(dx*dx+dy*dy<(e.r+8)*(e.r+8)){ hitEnemy(e,1,s); dead=true; break; } } }
      if(!dead&&boss&&boss.z<600&&Math.abs(s.z-boss.z)<34){ const dx=s.x-boss.x,dy=s.y-boss.y;
        if(dx*dx+dy*dy<(boss.r*.55)*(boss.r*.55)){ hitBoss(1); dead=true; } } }
    if(dead) shots.splice(i,1);
  }

  /* powerups */
  for(let i=powers.length-1;i>=0;i--){
    const p=powers[i]; p.z-=90*d; p.rot+=d*3;
    if(state==='play'&&!P.dead&&p.z<900){ p.x+=(P.x-p.x)*Math.min(1,d*1.1); p.y+=(P.y-p.y)*Math.min(1,d*1.1); }
    if(p.z<=ZP+6&&state==='play'&&!P.dead){ const dx=p.x-P.x,dy=p.y-P.y;
      if(dx*dx+dy*dy<40*40){ applyPower(p.t); sPow(); addPop(proj(P.x,P.y,ZP)[0],proj(P.x,P.y,ZP)[1]-30,{P:'SPREAD UP',S:'SHIELD',B:'+BOMB',G:'+750'}[p.t],'#5dff9a',false); powers.splice(i,1); continue; } }
    if(p.z<ZP-30) powers.splice(i,1);
  }

  /* fx */
  for(let i=parts.length-1;i>=0;i--){ const p=parts[i]; p.x+=p.vx*d;p.y+=p.vy*d;p.z+=p.vz*d; p.life-=d; if(p.life<=0) parts.splice(i,1); }
  for(let i=ringsFx.length-1;i>=0;i--){ const r=ringsFx[i]; r.r+=r.vr*d; r.life-=d; if(r.life<=0) ringsFx.splice(i,1); }
  for(let i=pops.length-1;i>=0;i--){ pops[i].life-=dt; if(pops[i].life<=0) pops.splice(i,1); }
  for(let i=fxQ.length-1;i>=0;i--){ fxQ[i].t-=d; if(fxQ[i].t<=0){ fxQ[i].fn(); fxQ.splice(i,1);} }
  flashT=Math.max(0,flashT-dt*1.6); shakeT=Math.max(0,shakeT-dt);

  /* combo / slowmo */
  if(comboT>0){ comboT-=d; if(comboT<=0) combo=0; }
  if(slowT>0) slowT-=dt;
  if(state==='play'&&P.dead){ deadT-=dt;
    if(deadT<=0){ if(lives>0){ P.dead=false; P.x=0; P.y=180; P.invuln=2.6; addPop(W/2,H*.4,'REDEPLOY','#3ce8ff',true); }
      else gameOver(); } }
  if(state==='over'){ overT+=dt; }

  /* camera + shake */
  const tx = state==='play'?P.x*.15:0, ty=state==='play'?(P.y-180)*.12:0;
  camX+=(tx-camX)*Math.min(1,dt*6); camY+=(ty-camY)*Math.min(1,dt*6);
  if(shakeT>0){ const m=shakeMag*(shakeT/.5); shx=(Math.random()*2-1)*m; shy=(Math.random()*2-1)*m; } else { shx=shy=0; }
}
function applyPower(t){
  if(t==='P'){ if(spread<3) spread++; else score+=400; }
  else if(t==='S') shield=true;
  else if(t==='B') bombs=Math.min(4,bombs+1);
  else score+=750;
}

/* ================= PLAYER UPDATE ================= */
function updatePlayer(d,dt){
  P.invuln=Math.max(0,P.invuln-dt); bombT=Math.max(0,bombT-d);
  if(P.dead) return;
  const focus=keys['ShiftLeft']||keys['ShiftRight'];
  let ix=(keys.ArrowRight||keys.KeyD?1:0)-(keys.ArrowLeft||keys.KeyA?1:0);
  let iy=(keys.ArrowDown||keys.KeyS?1:0)-(keys.ArrowUp||keys.KeyW?1:0);
  if(pointer.on){ const wx=(pointer.x-W/2)/K+camX, wy=(pointer.y-HORIZON)/K+camY;
    ix=Math.max(-1,Math.min(1,(wx-P.x)/30)); iy=Math.max(-1,Math.min(1,(wy-P.y)/30)); }
  const il=Math.hypot(ix,iy); if(il>1){ ix/=il; iy/=il; }
  const sp=focus?175:430;
  P.x+=ix*sp*d; P.y+=iy*sp*d;
  P.x=Math.max(-235,Math.min(235,P.x)); P.y=Math.max(40,Math.min(315,P.y));
  P.tilt+=((ix*.35)-P.tilt)*Math.min(1,d*10);
  /* thruster */
  if(Math.random()<.9) parts.push({x:P.x+(Math.random()-.5)*8,y:P.y+16,z:ZP-4,vx:(Math.random()-.5)*30,vy:60+Math.random()*60,vz:-80,life:.22,tl:.22,r:2.5,col:'#3ce8ff'});
  /* auto fire */
  fireT-=d;
  if(fireT<=0){ fireT=.085; sShoot();
    const mk=(dx,a)=>{ shots.push({x:P.x+dx,y:P.y+6,z:ZP,vx:Math.sin(a)*1500,vy:-Math.cos(a)*30,vz:1500*Math.cos(a),px:P.x+dx,py:P.y+6,pz:ZP}); };
    mk(-9,0); mk(9,0);
    if(spread>=2){ mk(-14,-.17); mk(14,.17); }
    if(spread>=3){ mk(-18,-.38); mk(18,.38); } }
}

function render(){
  ctx.fillStyle=bgGrad; ctx.fillRect(0,0,W,H);
  ctx.globalAlpha=.9;
  ctx.drawImage(nebCv, Math.sin(gTime*.05)*24-12, Math.cos(gTime*.043)*16-8, W, H);
  ctx.globalAlpha=1;
  /* stars */
  for(const s of stars){ const p=proj(s.x,s.y,s.z), r=Math.max(.6,s.s*FOCAL/s.z);
    ctx.globalAlpha=Math.min(1,1.3-s.z/ZFAR)*.9;
    ctx.fillStyle=s.c; ctx.fillRect(p[0]-r/2,p[1]-r/2,r,r); }
  ctx.globalAlpha=1;
  /* tunnel rings + rails */
  ctx.lineWidth=1;
  for(const r of ringsT){ const a=.04+.16*(1-r.z/ZFAR);
    ctx.strokeStyle='rgba(80,190,255,'+a.toFixed(3)+')';
    const c=[[ -520,-170],[520,-170],[520,560],[-520,560]].map(q=>proj(q[0],q[1],r.z));
    ctx.beginPath(); ctx.moveTo(c[0][0],c[0][1]); for(let i=1;i<4;i++) ctx.lineTo(c[i][0],c[i][1]); ctx.closePath(); ctx.stroke(); }
  ctx.strokeStyle='rgba(80,190,255,.1)';
  for(const q of [[-250,-30],[250,-30],[-250,360],[250,360]]){ const a=proj(q[0],q[1],ZFAR), b=proj(q[0],q[1],ZP);
    ctx.beginPath(); ctx.moveTo(a[0],a[1]); ctx.lineTo(b[0],b[1]); ctx.stroke(); }
  for(const st of streaks){ const a=.05+.2*(1-st.z/ZFAR), A=proj(st.x,st.y,st.z), B=proj(st.x,st.y,Math.max(ZP*.8,st.z+st.len));
    ctx.strokeStyle='rgba(190,235,255,'+a.toFixed(3)+')';
    ctx.beginPath(); ctx.moveTo(A[0],A[1]); ctx.lineTo(B[0],B[1]); ctx.stroke(); }

  /* entities far→near */
  const list=[];
  for(const e of enemies) list.push({z:e.z,fn:()=>drawEnemy(e)});
  if(boss) list.push({z:boss.z,fn:drawBoss});
  for(const p of powers) list.push({z:p.z,fn:()=>drawPower(p)});
  if(state==='play'&&!P.dead) list.push({z:ZP,fn:drawPlayer});
  list.sort((a,b)=>b.z-a.z);
  for(const it of list) it.fn();

  /* additive pass */
  ctx.globalCompositeOperation='lighter';
  for(const s of shots){ const A=proj(s.px,s.py,s.pz), B=proj(s.x,s.y,s.z);
    ctx.strokeStyle='rgba(60,232,255,.25)'; ctx.lineWidth=Math.max(4,9*B[2]); line(A,B);
    ctx.strokeStyle='#d8fbff'; ctx.lineWidth=Math.max(1.4,3.4*B[2]); line(A,B); }
  for(const b of bullets){ const p=proj(b.x,b.y,b.z), pr=b.r*p[2]*(b.r>9?1+.15*Math.sin(b.t*6):1);
    if(p[0]<-40||p[0]>W+40||p[1]<-40||p[1]>H+40) continue;
    ctx.fillStyle=hexA(b.col,.22); ctx.beginPath(); ctx.arc(p[0],p[1],Math.max(2,pr*2.1),0,7); ctx.fill();
    ctx.fillStyle=b.col; ctx.beginPath(); ctx.arc(p[0],p[1],Math.max(1.3,pr*1.25),0,7); ctx.fill();
    ctx.fillStyle='rgba(255,255,255,.85)'; ctx.beginPath(); ctx.arc(p[0],p[1],Math.max(.8,pr*.5),0,7); ctx.fill(); }
  for(const p of parts){ const q=proj(p.x,p.y,p.z); if(q[2]<=0) continue;
    const a=p.life/p.tl; ctx.globalAlpha=a; ctx.fillStyle=p.col;
    ctx.beginPath(); ctx.arc(q[0],q[1],Math.max(.8,p.r*q[2]*a*1.4),0,7); ctx.fill(); }
  ctx.globalAlpha=1;
  for(const r of ringsFx){ const q=proj(r.x,r.y,r.z), a=r.life/r.tl;
    ctx.strokeStyle=hexA(r.col,a*.8); ctx.lineWidth=Math.max(1,3*a*q[2]*4);
    ctx.beginPath(); ctx.arc(q[0],q[1],r.r*q[2],0,7); ctx.stroke(); }
  if(bombT>0){ const e=1-bombT/.8, R=e*860, q=proj(P.x,P.y,ZP), a=(1-e);
    for(const rr of [.94,1]){ ctx.strokeStyle='rgba(60,232,255,'+(a*.9)+')'; ctx.lineWidth=14*a+2;
      ctx.beginPath(); ctx.arc(q[0],q[1],R*rr*q[2],0,7); ctx.stroke(); } }
  ctx.globalCompositeOperation='source-over';

  /* popups */
  for(const p of pops){ const a=Math.min(1,p.life/p.tl*1.6), dy=(1-p.life/p.tl)*(p.big?44:26);
    ctx.globalAlpha=a; ctx.textAlign='center'; ctx.fillStyle=p.col;
    ctx.font=(p.big?'800 '+Math.max(22,W*.02)+'px ':'700 15px ')+'Consolas, Menlo, monospace';
    ctx.shadowColor=p.col; ctx.shadowBlur=14;
    ctx.fillText(p.text,p.sx,p.sy-dy); ctx.shadowBlur=0; }
  ctx.globalAlpha=1;
  if(flashT>0){ ctx.fillStyle='rgba(255,235,245,'+Math.min(.75,flashT)+')'; ctx.fillRect(0,0,W,H); }
}
function line(A,B){ ctx.beginPath(); ctx.moveTo(A[0],A[1]); ctx.lineTo(B[0],B[1]); ctx.stroke(); }
function hexA(hex,a){ const r=parseInt(hex.slice(1,3),16),g=parseInt(hex.slice(3,5),16),b=parseInt(hex.slice(5,7),16); return 'rgba('+r+','+g+','+b+','+a+')'; }

function drawPlayer(){
  const p=proj(P.x,P.y,ZP), s=p[2], focus=keys.ShiftLeft||keys.ShiftRight;
  if(P.invuln>0 && Math.floor(gTime*14)%2===0) ctx.globalAlpha=.35;
  ctx.save(); ctx.translate(p[0],p[1]); ctx.rotate(P.tilt); const u=s*(focus?.85:1);
  /* engine flame */
  const fl=(8+Math.random()*7)*u;
  ctx.fillStyle='rgba(60,232,255,.8)';
  ctx.beginPath(); ctx.moveTo(-6*u,14*u); ctx.lineTo(0,14*u+fl); ctx.lineTo(3.5*u,14*u); ctx.closePath(); ctx.fill();
  ctx.beginPath(); ctx.moveTo(6*u,14*u); ctx.lineTo(12*u? 9*u:9*u,14*u+fl*.8); ctx.lineTo(9.5*u,14*u); ctx.closePath(); ctx.fill();
  /* hull */
  ctx.fillStyle='#0e2c44'; ctx.strokeStyle='#7deaff'; ctx.lineWidth=Math.max(1,2*u);
  ctx.beginPath(); ctx.moveTo(0,-20*u); ctx.lineTo(6*u,-6*u); ctx.lineTo(17*u,11*u); ctx.lineTo(8*u,8*u);
  ctx.lineTo(0,14*u); ctx.lineTo(-8*u,8*u); ctx.lineTo(-17*u,11*u); ctx.lineTo(-6*u,-6*u); ctx.closePath();
  ctx.fill(); ctx.stroke();
  ctx.fillStyle='#3ce8ff'; ctx.beginPath(); ctx.arc(0,-4*u,3.2*u,0,7); ctx.fill();
  ctx.restore();
  if(focus){ ctx.strokeStyle='rgba(255,90,120,.9)'; ctx.lineWidth=1;
    ctx.beginPath(); ctx.arc(p[0],p[1],PHIT*s,0,7); ctx.stroke();

  if(shield){ ctx.strokeStyle='rgba(93,255,154,.8)'; ctx.lineWidth=2;
    ctx.setLineDash([10,7]); ctx.lineDashOffset=-gTime*40;
    ctx.beginPath(); ctx.arc(p[0],p[1],26*s,0,7); ctx.stroke(); ctx.setLineDash([]); }
  ctx.globalAlpha=1;
}
function drawEnemy(e){
  const p=proj(e.x,e.y,e.z), s=p[2], r=e.r*s;
  if(r<.5) return;
  ctx.save(); ctx.translate(p[0],p[1]);
  const glow=(.4+.3*Math.sin(e.t*6))*Math.min(1,r/8);
  ctx.fillStyle=hexA('#ff3d7e',glow*.25); ctx.beginPath(); ctx.arc(0,0,r*1.9,0,7); ctx.fill();
  const flash=e.hit>0;
  if(e.type==='drone'||e.type==='mini'){ const c=e.type==='mini'?'#ff8a6a':'#ff3d7e';
    ctx.fillStyle=flash?'#fff':'#3a0f22'; ctx.strokeStyle=c; ctx.lineWidth=Math.max(1,2*s);
    ctx.beginPath(); ctx.moveTo(0,-r); ctx.lineTo(r*.8,0); ctx.lineTo(0,r); ctx.lineTo(-r*.8,0); ctx.closePath(); ctx.fill(); ctx.stroke();
    ctx.fillStyle=c; ctx.beginPath(); ctx.arc(0,0,r*.3,0,7); ctx.fill(); }
  else if(e.type==='sweeper'){ const c='#ffb52e';
    ctx.fillStyle=flash?'#fff':'#3a2408'; ctx.strokeStyle=c; ctx.lineWidth=Math.max(1,2*s);
    ctx.beginPath(); ctx.moveTo(0,-r*.5); ctx.lineTo(r,r*.7); ctx.lineTo(0,r*.15); ctx.lineTo(-r,r*.7); ctx.closePath(); ctx.fill(); ctx.stroke();
    ctx.fillStyle=c; ctx.beginPath(); ctx.arc(0,r*.05,r*.22,0,7); ctx.fill(); }
  else if(e.type==='orbiter'){ const c='#46ffc8';
    ctx.strokeStyle=c; ctx.lineWidth=Math.max(1,2.4*s); ctx.fillStyle=flash?'#fff':'#06302a';
    ctx.beginPath(); ctx.arc(0,0,r*.75,0,7); ctx.fill(); ctx.stroke();
    ctx.save(); ctx.rotate(e.phase*1.4);
    for(let i=0;i<4;i++){ ctx.rotate(Math.PI/2); ctx.beginPath(); ctx.moveTo(r*.75,0); ctx.lineTo(r*1.15,0); ctx.stroke(); }
    ctx.restore();
    ctx.fillStyle=c; ctx.beginPath(); ctx.arc(0,0,r*.28*(1+.2*Math.sin(e.t*5)),0,7); ctx.fill(); }
  else if(e.type==='splitter'){ const c='#ff7a3c';
    ctx.rotate(Math.PI/4); ctx.fillStyle=flash?'#fff':'#3a1608'; ctx.strokeStyle=c; ctx.lineWidth=Math.max(1,2*s);
    const q=r*.8; ctx.beginPath(); ctx.rect(-q,-q,q*2,q*2); ctx.fill(); ctx.stroke();
    ctx.rotate(-Math.PI/4); ctx.strokeStyle=c; ctx.beginPath(); ctx.moveTo(-r*.5,-r*.2); ctx.lineTo(0,r*.1); ctx.lineTo(r*.4,-r*.3); ctx.stroke(); }
  ctx.restore();
}
function drawBoss(){
  const b=boss, p=proj(b.x,b.y,b.z), s=p[2], r=b.r*s;
  ctx.save(); ctx.translate(p[0],p[1]);
  const f=b.hp/b.maxhp, col=f>.66?'#ff3d7e':f>.33?'#ffb52e':'#ff5c39';
  ctx.fillStyle=hexA(col,.14+.1*Math.sin(b.t*4)); ctx.beginPath(); ctx.arc(0,0,r*1.5,0,7); ctx.fill();
  ctx.rotate(Math.sin(b.t*.4)*.12);
  ctx.fillStyle=b.flash>0?'#fff':'#2a0a18'; ctx.strokeStyle=col; ctx.lineWidth=Math.max(2,3.4*s);
  ctx.beginPath(); for(let i=0;i<6;i++){ const a=i*Math.PI/3; const x=Math.cos(a)*r,y=Math.sin(a)*r*.82; i?ctx.lineTo(x,y):ctx.moveTo(x,y);} ctx.closePath(); ctx.fill(); ctx.stroke();
  ctx.strokeStyle=hexA(col,.9); ctx.lineWidth=Math.max(1.5,2*s);
  for(let i=0;i<3;i++){ const a0=b.t*.9+i*Math.PI*2/3; ctx.beginPath(); ctx.arc(0,0,r*.66,a0,a0+1.1); ctx.stroke(); }
  ctx.fillStyle=col; ctx.beginPath(); ctx.arc(0,0,r*.3*(1+.15*Math.sin(b.t*7)),0,7); ctx.fill();
  ctx.fillStyle='#fff'; ctx.beginPath(); ctx.arc(0,0,r*.12,0,7); ctx.fill();
  ctx.restore();
}
function drawPower(p){
