async function clashFor(side){
  const opp=side==='p'?'e':'p';
  showBanner('CLASH','bc');await sleep(320);
  for(let l=0;l<3;l++){ if(S.over)return;
    const mine=S.lanes[l][side],theirs=S.lanes[l][opp];
    if(!mine.length)continue;
    const striker=mine.find(u=>!u.ranged&&u.hp>0);
    if(striker){
      if(theirs.length){const t=theirs[0];await lunge(striker,posOf(t).x,posOf(t).y,.55);hitUnit(side,t,striker.atk);
        if(!S.over&&t.hp>0&&t.atk>0){if(S.lanes[l][opp].includes(t)){const s2=posOf(striker);await lunge(t,s2.x,s2.y,.3);hitUnit(opp,striker,t.atk);}}}
      else{const cp=crestPos(opp),sp=posOf(striker);bolt(sp.x,sp.y,cp.x,cp.y,false);await sleep(170);coreHit(opp,striker.atk,side);}
    }
    if(S.over)return;
    for(const u of mine){ if(!u.ranged||u.hp<=0)continue;
      const t2=theirs.length?theirs[0]:null; // theirs may have shrunk; re-read
      const cur=S.lanes[l][opp];
      if(cur.length){const t=cur[0],sp=posOf(u);bolt(sp.x,sp.y,posOf(t).x,posOf(t).y,true);await sleep(160);sfx.shoot();hitUnit(side,t,u.atk);}
      else{const sp=posOf(u),cp=crestPos(opp);bolt(sp.x,sp.y,cp.x,cp.y,true);await sleep(160);sfx.shoot();coreHit(opp,u.atk,side);}
      if(S.over)return;
    }
    await sleep(140);
  }
}

async function enemyTurn(){
  S.eEnergy=Math.min(6,S.round+1);
  draw('e');await sleep(550);if(S.over)return;
  let n=0;
  while(!S.over&&n<6){
    const ch=aiChoose();if(!ch)break;
    S.eEnergy-=ch.c.cost;S.ehand.splice(S.ehand.indexOf(ch.c),1);
    const u={...ch.c,side:'e',id:S.uid++,maxHp:ch.c.hp,hp:ch.c.hp,tilt:rnd(-2.5,2.5)};
    S.lanes[ch.l].e.push(u);spawnUnitEl(u);layoutUnits();renderEnemyHand(false);sfx.play();n++;
    await sleep(700);
  }
  await sleep(320);if(S.over)return;
  setPhase('clash-e');await clashFor('e');
}
function aiChoose(){
  let best=null,bs=-1e9;
  for(const c of S.ehand){ if(c.cost>S.eEnergy)continue;
    for(let l=0;l<3;l++){ const mine=S.lanes[l].e,theirs=S.lanes[l].p;
      if(mine.length>=3)continue;
      let s=Math.random()*1.4;
      if(!theirs.length)s+=2+c.atk*.9;
      else{const f=theirs[0];
        if(c.atk>=f.hp)s+=4+(f.ranged?1:0);
        else{s+=c.atk*.35;if(!c.ranged){if(f.atk>=c.hp)s-=3.2;else s+=.8;}}
      }
      if(c.ranged)s+=.6;
      if(mine.length===2)s-=1.6;
      if(s>bs){bs=s;best={c,l};}
    }}
  return best;
}
