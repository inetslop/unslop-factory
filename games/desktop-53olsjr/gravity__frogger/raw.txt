function updateFrog(dt){

    f.t+=dt;var T=HOP_T,u=Math.min(1,f.t/T),tt=u*T;
    f.x=f.px+f.vx0*tt+.5*grav.x*tt*tt;
    f.y=f.py+f.vy0*tt+.5*grav.y*tt*tt;
    clampFrog(f);f.z=ZMAX*4*u*(1-u);
    if(u>=1){
      f.state='ground';f.z=0;f.squash=1;f.carryVx=0;sfx.land();puff(f.x,f.y);
      onLand(f);
      if(state==='playing'&&buffered&&tG-buffered.t<.28){var b=buffered;buffered=null;launch(b.dx,b.dy);}
    }
  }else{

    }else f.carryVx=0;
  }

    for(var i=0;i<traffic.length;i++){var v=traffic[i];
      if((v.kind==='car'||v.kind==='truck')&&hitCar(f,v)){squish(v);return;}}
  }
}

  requestAnimationFrame(loop);
  var dt=(now-last)/1000;last=now;if(dt>.05)dt=.05;tG+=dt;
  updFlies(dt);drawFlies();

  updateParts(dt);updateTexts(dt);updateRain(dt);
  shake=Math.max(0,shake-dt*26);
  render();drawCompass();hud();
}
