function bfs(sx,sy,tx,ty){
  if(sx===tx&&sy===ty)return[{x:sx,y:sy}];
  const prev=new Map(),q=[[sx,sy]];prev.set(key(sx,sy),null);
  const D=[[1,0],[-1,0],[0,1],[0,-1]];
  while(q.length){
    const cur=q.shift(),cx=cur[0],cy=cur[1];
    for(let d=0;d<4;d++){
      const nx=cx+D[d][0],ny=cy+D[d][1],nk=key(nx,ny);
      if(prev.has(nk)||isWall(nx,ny)||boxSet.has(nk))continue;
      prev.set(nk,key(cx,cy));
      if(nx===tx&&ny===ty){const path=[[nx,ny]];let pk=prev.get(nk);while(pk){const p=pk.split(',');path.push([+p[0],+p[1]]);pk=prev.get(pk);}return path.reverse();}
      q.push([nx,ny]);
    }
  }
  return null;
}
function planPath(tx,ty){
  if(!inB(tx,ty)||isWall(tx,ty))return null;
  if(tx===player.x&&ty===player.y)return null;
  const D=[[1,0],[-1,0],[0,1],[0,-1]],best=[];
  for(let d=0;d<4;d++){
    const nx=tx-D[d][0],ny=ty-D[d][1];
    if(isWall(nx,ny)||boxSet.has(key(nx,ny)))continue;
    const p=bfs(player.x,player.y,nx,ny);
    if(p){if(!best.length||p.length<best.length)best.push(p);}
  }
  return best.length?best[0]:null;
}

function queuePath(path){

  for(let i=1;i<path.length;i++){

}

  if(!h){toast('NOTHING TO UNDO');sfx.error();return;}
  history.pop();
  player={x:h.px,y:h.py};boxes=h.boxes.map(o=>({x:o.x,y:o.y}));
  boxSet=new Map(boxes.map(b=>[key(b.x,b.y),b]));
  moves=h.moves;pushes=h.pushes;
  queue=[];anim=null;stateVer++;hoverPath=null;bumpHover();updateHUD();sfx.undo();toast('UNDO');
}

}

  DPR=Math.min(window.devicePixelRatio||1,2);
  W=window.innerWidth;H=window.innerHeight;
  cv.width=W*DPR;cv.height=H*DPR;
  const topPad=status==='play'?78:34,botPad=status==='play'?56:30;
  board.ts=Math.min((W-28)/cols,(H-topPad-botPad)/rows);
  board.ts=clamp(board.ts,22,92);
  board.ox=(W-cols*board.ts)/2;
  board.oy=topPad+(H-topPad-botPad-rows*board.ts)/2+14;
}

function initDust(){dust=[];for(let i=0;i<46;i++)dust.push({x:Math.random(),y:Math.random(),s:0.5+Math.random()*1.8,v:2+Math.random()*7,a:0.04+Math.random()*0.1});}
function rr(x,y,w,h,r){ctx.beginPath();ctx.moveTo(x+r,y);ctx.arcTo(x+w,y,x+w,y+h,r);ctx.arcTo(x+w,y+h,x,y+h,r);ctx.arcTo(x,y+h,x,y,r);ctx.arcTo(x,y,x+w,y,r);ctx.closePath();}

function draw(t,dt){
  ctx.setTransform(DPR,0,0,DPR,0,0);
  // background
  const g=ctx.createLinearGradient(0,0,0,H);
  g.addColorStop(0,'#0c1526');g.addColorStop(0.55,'#0a1120');g.addColorStop(1,'#080d18');
  ctx.fillStyle=g;ctx.fillRect(0,0,W,H);
  const rg=ctx.createRadialGradient(W/2,H*0.42,10,W/2,H*0.42,Math.max(W,H)*0.75);
  rg.addColorStop(0,'rgba(64,110,180,0.10)');rg.addColorStop(1,'rgba(0,0,0,0)');
  ctx.fillStyle=rg;ctx.fillRect(0,0,W,H);
  // dust
  for(const d of dust){
    d.y-=d.v*dt/H;if(d.y<-0.02){d.y=1.02;d.x=Math.random();}
    ctx.globalAlpha=d.a*(0.7+0.3*Math.sin(t/900+d.x*40));
    ctx.fillStyle='#bcd7ff';ctx.fillRect(d.x*W,d.y*H,d.s,d.s);
  }
  ctx.globalAlpha=1;

  // shake
  if(shakeMag>0.1){ctx.translate((Math.random()-0.5)*shakeMag,(Math.random()-0.5)*shakeMag);shakeMag*=Math.pow(0.001,dt);}else shakeMag=0;

  const ts=board.ts,ox=board.ox,oy=board.oy,PX=x=>ox+x*ts,PY=y=>oy+y*ts;

  // floor
  for(let y=0;y<rows;y++)for(let x=0;x<cols;x++){
    if(isWall(x,y))continue;
    ctx.fillStyle=(x+y)%2?'#17253a':'#142133';
    ctx.fillRect(PX(x),PY(y),ts,ts);
    ctx.strokeStyle='rgba(255,255,255,0.025)';ctx.lineWidth=1;
    ctx.strokeRect(PX(x)+0.5,PY(y)+0.5,ts-1,ts-1);
  }
  // wall shadows on floor
  ctx.fillStyle='rgba(4,8,16,0.35)';
  for(let y=0;y<rows;y++)for(let x=0;x<cols;x++){
    if(isWall(x,y))continue;
    const s=ts*0.22;let painted=false;
    if(inB(x+1,y)&&isWall(x+1,y)){ctx.fillRect(PX(x)+ts-s,PY(y),s,ts);painted=true;}
    if(inB(x-1,y)&&isWall(x-1,y)){ctx.fillRect(PX(x),PY(y),s,ts);}
    if(inB(x,y+1)&&isWall(x,y+1)){ctx.fillRect(PX(x),PY(y)+ts-s,ts,s);painted=true;}
    if(inB(x,y-1)&&isWall(x,y-1))ctx.fillRect(PX(x),PY(y),ts,s);
  }
  // targets
  for(const k of targetSet){
    const p=k.split(','),tx=+p[0],ty=+p[1];
    if(isWall(tx,ty))continue;
    const cx=PX(tx)+ts/2,cy=PY(ty)+ts/2,filled=boxAt(tx,ty);
    const pulse=filled?1:0.5+0.3*Math.sin(t/480+(tx*3+ty*5));
    ctx.strokeStyle='rgba(79,216,255,'+(filled?0.95:0.35+0.4*pulse)+')';
    ctx.lineWidth=Math.max(2,ts*0.06);
    const rr2=ts*(filled?0.3:0.27+0.03*pulse);
    ctx.beginPath();ctx.arc(cx,cy,rr2,0,Math.PI*2);ctx.stroke();
    ctx.save();ctx.translate(cx,cy);ctx.rotate(Math.PI/4);
    const d=ts*(filled?0.16:0.1+0.03*pulse);
    ctx.fillStyle='rgba(79,216,255,'+(filled?0.95:0.4+0.45*pulse)+')';
    ctx.fillRect(-d/2,-d/2,d,d);ctx.restore();
  }
  // hover + path preview
  if(status==='play'&&hoverCell&&inB(hoverCell.x,hoverCell.y)&&!isWall(hoverCell.x,hoverCell.y)){
    if(hoverPath&&hoverPath.length>1){
      for(let i=1;i<hoverPath.length;i++){
        const c=hoverPath[i],a=0.12+0.4*(i/hoverPath.length);
        ctx.fillStyle='rgba(79,216,255,'+a+')';
        ctx.beginPath();ctx.arc(PX(c.x)+ts/2,PY(c.y)+ts/2,Math.max(2,ts*0.06),0,Math.PI*2);ctx.fill();
      }
      const last=hoverPath[hoverPath.length-1];
      if(boxAt(last.x,last.y)){ // push arrow
        const prev=hoverPath[hoverPath.length-2];
        const dx=last.x-prev.x,dy=last.y-prev.y;
        const ax=PX(last.x)+ts/2+dx*ts*0.42, ay=PY(last.y)+ts/2+dy*ts*0.42;
        ctx.strokeStyle='rgba(79,216,255,0.85)';ctx.lineWidth=Math.max(2,ts*0.06);
        ctx.beginPath();ctx.moveTo(PX(last.x)+ts/2,PY(last.y)+ts/2);ctx.lineTo(ax,ay);ctx.stroke();
        const ang=Math.atan2(dy,dx);
        ctx.beginPath();ctx.moveTo(ax,ay);
        ctx.lineTo(ax-Math.cos(ang-0.5)*ts*0.18,ay-Math.sin(ang-0.5)*ts*0.18);
        ctx.moveTo(ax,ay);
        ctx.lineTo(ax-Math.cos(ang+0.5)*ts*0.18,ay-Math.sin(ang+0.5)*ts*0.18);ctx.stroke();
      }else{
        ctx.strokeStyle='rgba(79,216,255,0.7)';ctx.lineWidth=2;
        ctx.strokeRect(PX(last.x)+3,PY(last.y)+3,ts-6,ts-6);
      }
    }else if(!(hoverCell.x===player.x&&hoverCell.y===player.y)){
      ctx.strokeStyle='rgba(255,93,115,0.4)';ctx.lineWidth=2;
      ctx.strokeRect(PX(hoverCell.x)+4,PY(hoverCell.y)+4,ts-8,ts-8);
    }
  }
  // boxes
  for(const b of boxes){
    let bx=b.x,by=b.y,scale=1,alpha=1;
    if(anim&&anim.push&&(b.x===anim.push.bx&&b.y===anim.push.by)){
      const e=easeOut(clamp((t-anim.t0)/STEP,0,1));
      bx=lerp(anim.push.bx,anim.push.bx2,e);by=lerp(anim.push.by,anim.push.by2,e);scale=1+0.06*Math.sin(e*Math.PI);
    }
    drawCrate(PX(bx),PY(by),ts,onTarget(b),scale,alpha,t);
  }
  // player
  {
    let px=player.x,py=player.y,sy=1;
    if(anim){const e=easeOut(clamp((t-anim.t0)/STEP,0,1));px=lerp(anim.px,anim.nx,e);py=lerp(anim.py,anim.ny,e);}
    const bob=Math.sin(t/320)*ts*0.02;
    if(anim&&anim.push)sy=0.9;
    drawForklift(PX(px)+ts/2,PY(py)+ts/2+bob,ts,facing,sy,t);
  }
  // particles (grid space)
  for(let i=particles.length-1;i>=0;i--){
    const p=particles[i];p.life-=dt;
    if(p.life<=0){particles.splice(i,1);continue;}
    p.x+=p.vx*dt/ts;p.y+=p.vy*dt/ts;
    if(p.type==='conf'){p.vy+=260*dt/ts;p.rot+=p.vr*dt;}
    else{p.vx*=Math.pow(0.25,dt);p.vy=p.vy*Math.pow(0.25,dt)-140*dt/ts*p.life;}
    const a=clamp(p.life/p.max,0,1);
    ctx.globalAlpha=a;ctx.fillStyle=p.col;
    if(p.type==='conf'){ctx.save();ctx.translate(PX(p.x),PY(p.y));ctx.rotate(p.rot);ctx.fillRect(-p.sz/2,-p.sz/4,p.sz,p.sz/2);ctx.restore();}
    else{ctx.beginPath();ctx.arc(PX(p.x),PY(p.y),p.sz*a,0,Math.PI*2);ctx.fill();}
  }
  ctx.globalAlpha=1;
  // ripples
  for(let i=ripples.length-1;i>=0;i--){
    const r=ripples[i];r.r+=dt*2.4;r.a-=dt*1.6;
    if(r.a<=0||r.r>r.max){ripples.splice(i,1);continue;}
    ctx.strokeStyle=r.bad?'rgba(255,93,115,'+r.a+')':'rgba(79,216,255,'+r.a+')';
    ctx.lineWidth=2;ctx.beginPath();ctx.arc(PX(r.x)+ts/2,PY(r.y)+ts/2,r.r*ts*0.5,0,Math.PI*2);ctx.stroke();
  }
  // walls (drawn last so their top sheen overlays floor shadows)
  for(let y=0;y<rows;y++)for(let x=0;x<cols;x++){
    if(!isWall(x,y))continue;
    drawWall(PX(x),PY(y),ts,x===0||!inB(x-1,y)||!isWall(x-1,y),y===0||!inB(x,y-1)||!isWall(x,y-1),x===cols-1||!inB(x+1,y)||!isWall(x+1,y),y===rows-1||!inB(x,y+1)||!isWall(x,y+1));
  }
}

function drawWall(x,y,ts,eL,eT,eR,eB){
  const h=ts*0.2;
  ctx.fillStyle='#0e1826';ctx.fillRect(x,y,ts,ts); // body/side
  if(eB)ctx.fillStyle='#0a121e';
  else ctx.fillStyle='#13202f';
  ctx.fillRect(x,y+h,ts,ts-h);
  const g=ctx.createLinearGradient(0,y,0,y+h*2.4);
  g.addColorStop(0,'#31486b');g.addColorStop(1,'#22354e');
  ctx.fillStyle=g;ctx.fillRect(x,y,ts,h*2.4>ts?ts:h*2.4);
  ctx.fillStyle='rgba(255,255,255,0.07)';ctx.fillRect(x,y,ts,1.5);
  if(eL){ctx.fillStyle='rgba(255,255,255,0.04)';ctx.fillRect(x,y,1.5,ts);}
  if(eR){ctx.fillStyle='rgba(0,0,0,0.25)';ctx.fillRect(x+ts-1.5,y,1.5,ts);}
}

function drawCrate(x,y,ts,onT,scale,alpha,t){
  const s=ts*0.78*scale,cx=x+(ts-s)/2,cy=y+(ts-s)/2;
  ctx.save();ctx.globalAlpha=alpha;
  if(onT){ctx.shadowColor='rgba(79,216,255,0.8)';ctx.shadowBlur=ts*0.3;}
  const g=ctx.createLinearGradient(cx,cy,cx,cy+s);
  if(onT){g.addColorStop(0,'#ffd489');g.addColorStop(1,'#e89a35');}
  else{g.addColorStop(0,'#f2b45c');g.addColorStop(1,'#cf7d22');}
  ctx.fillStyle=g;rr(cx,cy,s,s,Math.max(3,ts*0.09));ctx.fill();
  ctx.shadowBlur=0;
  ctx.strokeStyle=onT?'#8a5a14':'#7c4a12';ctx.lineWidth=Math.max(2,ts*0.06);
  rr(cx+1,cy+1,s-2,s-2,Math.max(2,ts*0.08));ctx.stroke();
  // cross braces
  ctx.save();rr(cx+3,cy+3,s-6,s-6,2);ctx.clip();
  ctx.strokeStyle='rgba(94,53,10,0.5)';ctx.lineWidth=Math.max(3,ts*0.1);
  ctx.beginPath();ctx.moveTo(cx,cy);ctx.lineTo(cx+s,cy+s);ctx.moveTo(cx+s,cy);ctx.lineTo(cx,cy+s);ctx.stroke();
  ctx.restore();
  // bolts
  ctx.fillStyle='rgba(60,32,5,0.8)';
  const bo=s*0.14;
  [[bo,bo],[s-bo,bo],[bo,s-bo],[s-bo,s-bo]].forEach(p=>{ctx.beginPath();ctx.arc(cx+p[0],cy+p[1],Math.max(1.4,ts*0.035),0,Math.PI*2);ctx.fill();});
  if(onT){ // check mark
    ctx.strokeStyle='#0d3a4a';ctx.lineWidth=Math.max(2.5,ts*0.07);ctx.lineCap='round';
    ctx.beginPath();ctx.moveTo(cx+s*0.68,cy+s*0.16);ctx.lineTo(cx+s*0.84,cy+s*0.3);ctx.lineTo(cx+s*1.0- s*0.02,cy+s*0.08);ctx.stroke();
    ctx.lineCap='butt';
  }
  ctx.restore();
}

function drawForklift(x,y,ts,f,sy,t){
  const s=ts*0.62;
  ctx.save();ctx.translate(x,y);ctx.scale(1,sy);
  // shadow
  ctx.fillStyle='rgba(0,0,0,0.3)';
  ctx.beginPath();ctx.ellipse(0,s*0.42,s*0.36,s*0.12,0,0,Math.PI*2);ctx.fill();
  const walk=anim?Math.sin((performance.now()-anim.t0)/STEP*Math.PI)*s*0.08:0;
  // feet
  ctx.fillStyle='#c94b3b';
  rr(-s*0.26,s*0.3-walk,s*0.18,s*0.14,2);ctx.fill();
  rr(s*0.08,s*0.3+walk,s*0.18,s*0.14,2);ctx.fill();
  // body
  const g=ctx.createLinearGradient(0,-s*0.42,0,s*0.36);
  g.addColorStop(0,'#ff8a70');g.addColorStop(1,'#f0553e');
  ctx.fillStyle=g;rr(-s*0.32,-s*0.42,s*0.64,s*0.74,Math.max(4,s*0.16));ctx.fill();
  ctx.strokeStyle='rgba(90,20,10,0.5)';ctx.lineWidth=Math.max(1.5,ts*0.03);
  rr(-s*0.32,-s*0.42,s*0.64,s*0.74,Math.max(4,s*0.16));ctx.stroke();
  // face plate
  ctx.fillStyle='rgba(255,255,255,0.14)';rr(-s*0.24,-s*0.34,s*0.48,s*0.3,Math.max(3,s*0.1));ctx.fill();
  // eyes (look toward facing)
  const ex=f.x*s*0.07,ey=f.y*s*0.05;
  ctx.fillStyle='#fff';
  ctx.beginPath();ctx.arc(-s*0.12+ex,-s*0.2+ey,s*0.085,0,Math.PI*2);ctx.fill();
  ctx.beginPath();ctx.arc(s*0.12+ex,-s*0.2+ey,s*0.085,0,Math.PI*2);ctx.fill();
  ctx.fillStyle='#231a1c';
  ctx.beginPath();ctx.arc(-s*0.12+ex*1.9,-s*0.2+ey*1.9,s*0.045,0,Math.PI*2);ctx.fill();
  ctx.beginPath();ctx.arc(s*0.12+ex*1.9,-s*0.2+ey*1.9,s*0.045,0,Math.PI*2);ctx.fill();
  // antenna
  ctx.strokeStyle='#ffd9a0';ctx.lineWidth=Math.max(1.5,ts*0.03);
  ctx.beginPath();ctx.moveTo(0,-s*0.42);ctx.lineTo(f.x*s*0.1,-s*0.6+Math.sin(t/300)*s*0.03);ctx.stroke();
  ctx.fillStyle='#ffd9a0';
  ctx.beginPath();ctx.arc(f.x*s*0.1,-s*0.62+Math.sin(t/300)*s*0.03,s*0.05,0,Math.PI*2);ctx.fill();
  // push flash
  if(pushFlash>0.05){
    ctx.globalAlpha=pushFlash*0.8;
    ctx.strokeStyle='#ffd9a0';ctx.lineWidth=Math.max(2,ts*0.05);
    const r=s*(0.5+(1-pushFlash)*0.3);
    ctx.beginPath();ctx.arc(f.x*r,f.y*r,r*0.6,0,Math.PI*2);ctx.stroke();
  }
  ctx.restore();
}

/* ============================ MAIN LOOP =========================== */
let lastT=performance.now();
function frame(t){
  const dt=Math.min(0.05,(t-lastT)/1000);lastT=t;
  if(anim&&t-anim.t0>=STEP)commit();
  pushFlash=Math.max(0,pushFlash-dt*2.4);
  draw(t,dt);
  requestAnimationFrame(frame);
}
