  const y0=Math.floor(e.y - r), y1=Math.floor(e.y + r);

function resolveAxis(e,axis,r,pd){
  const x0=Math.floor(e.x-r),x1=Math.floor(e.x+r),y0=Math.floor(e.y-r),y1=Math.floor(e.y+r);
  for(let cy=y0;cy<=y1;cy++)for(let cx=x0;cx<=x1;cx++){
    if(!solidAt(cx,cy,pd))continue;
    const ox=(e.x+r)>cx&&(e.x-r)<cx+1,oy=(e.y+r)>cy&&(e.y-r)<cy+1;
    if(!ox||!oy)continue;
    if(axis==='x'){e.x=e.vx>0?cx-r:cx+1+r;e.vx=0;e.hitX=true;}
    else{e.y=e.vy>0?cy-r:cy+1+r;e.vy=0;e.hitY=true;}
  }
}
function moveEntity(e,dt,r,pd){
  const sp=Math.hypot(e.vx,e.vy);if(sp<0.001)return;
  const n=Math.max(1,Math.ceil(sp*dt/0.35)),h=dt/n;
  for(let i=0;i<n;i++){
    e.x+=e.vx*h;resolveAxis(e,'x',r,pd);

    if(e.x<-0.5)e.x+=COLS;if(e.x>COLS+0.5)e.x-=COLS;
  }
}

function updPlayer(dt){

  const sp0=Math.hypot(p.vx,p.vy);if(sp0>VMAX){p.vx*=VMAX/sp0;p.vy*=VMAX/sp0;}

}

function playerFacing(){const p=player;if(Math.abs(p.vx)>Math.abs(p.vy))return{x:Math.sign(p.vx)||1,y:0};return{x:0,y:Math.sign(p.vy)||-1};}
function chaseTarget(g){
  const p=player;
  if(state==='START')return{x:10.5+7.5*Math.sin(time*0.45),y:10.5+6.5*Math.sin(time*0.31+1.7)};
  switch(g.id){
    case 0:return{x:p.x+p.vx*0.45,y:p.y+p.vy*0.45};
    case 1:{const f=playerFacing();return{x:p.x+f.x*3,y:p.y+f.y*3};}
    case 2:{const a=time*1.6+2;return{x:p.x+Math.cos(a)*2.4,y:p.y+Math.sin(a)*2.4};}
    default:{const d=Math.hypot(g.x-p.x,g.y-p.y);return d>6?{x:p.x,y:p.y}:{x:DEF[3].corner[0],y:DEF[3].corner[1]};}
  }
}
function roamAI(g,dt){
  let tx,ty;
  if(g.wander&&(g.wander.t-=dt)>0){tx=g.wander.x;ty=g.wander.y;}
  else if(g.fright){const p=player,dx=g.x-p.x,dy=g.y-p.y,d=Math.hypot(dx,dy)||1;tx=p.x-dx/d*9+Math.cos(time*5+g.id)*0.6;ty=p.y-dy/d*9+Math.sin(time*4+g.id)*0.6;}
  else if(phase==='scatter'){tx=DEF[g.id].corner[0];ty=DEF[g.id].corner[1];}
  else{const t=chaseTarget(g);tx=t.x;ty=t.y;}
  let dx=tx-g.x,dy=ty-g.y;const d=Math.hypot(dx,dy)||1;dx/=d;dy/=d;
  const sp=baseSpeed()*(g.fright?0.72:1);
  const bl=Math.min(1,dt*(g.fright?5:3.2));
  g.vx+=(dx*sp-g.vx)*bl;g.vy+=(dy*sp-g.vy)*bl;
  g.vx+=gVec.x*GGHOST*dt;g.vy+=gVec.y*GGHOST*dt;
  const s=Math.hypot(g.vx,g.vy);if(s>sp*1.4){g.vx*=sp*1.4/s;g.vy*=sp*1.4/s;}
  const bx=g.x,by=g.y;
  g.hitX=false;g.hitY=false;
  moveEntity(g,dt,g.r,true);
  const moved=Math.hypot(g.x-bx,g.y-by);
  if(moved<sp*dt*0.25){g.stuckT+=dt;if(g.stuckT>0.65){g.wander={x:clamp(g.x+rnd(-3,3),1,COLS-2),y:clamp(g.y+rnd(-3,3),1,ROWS-2),t:1.1};g.stuckT=0;}}
  else g.stuckT=Math.max(0,g.stuckT-dt*2);
}
function updGhost(g,dt){
  if(g.respawnT>0)g.respawnT-=dt;
  switch(g.state){
    case 'house':{
      g.x+=g.vx*dt;g.y+=g.vy*dt;
      if(g.x<HOUSE.x0+g.r||g.x>HOUSE.x1-g.r){g.vx*=-1;g.x=clamp(g.x,HOUSE.x0+g.r,HOUSE.x1-g.r);}
      if(g.y<HOUSE.y0+g.r||g.y>HOUSE.y1-g.r){g.vy*=-1;g.y=clamp(g.y,HOUSE.y0+g.r,HOUSE.y1-g.r);}
      if((g.releaseT-=dt)<=0){g.state='exit';g.fright=frightT>0;g.wp=[{x:10.5,y:g.y},{x:10.5,y:8.4}];g.wpi=0;}
      break;}
    case 'exit':{
      const t=g.wp[g.wpi],dx=t.x-g.x,dy=t.y-g.y,d=Math.hypot(dx,dy),step=baseSpeed()*1.25*dt;
      if(d<=Math.max(step,0.05)){g.x=t.x;g.y=t.y;g.wpi++;if(g.wpi>=g.wp.length){g.state='roam';g.fright=frightT>0;}}
      else{g.x+=dx/d*step;g.y+=dy/d*step;}
      break;}
    case 'orb':{
      const dx=10.5-g.x,dy=8.4-g.y,d=Math.hypot(dx,dy),step=6.5*dt;
      if(d<=step){g.state='roam';g.respawnT=1.6;}
      else{g.x+=dx/d*step;g.y+=dy/d*step;if(Math.random()<0.7)parts.push({x:g.x,y:g.y,vx:0,vy:0,life:0,max:0.3,size:0.08,col:'#9fdcff',grav:0});}
      break;}
    default:roamAI(g,dt);
  }
}
function collide(){
  const p=player;if(p.dead)return;
  for(const g of ghosts){
    if((g.state!=='roam'&&g.state!=='exit')||g.respawnT>0)continue;
    const dx=g.x-p.x,dy=g.y-p.y;
    if(dx*dx+dy*dy<0.42){
      if(g.fright){
        chain=(chain%4)+1;const pts=[200,400,800,1600][chain-1];score+=pts;
        addFloater(g.x,g.y,'+'+pts,'#7ef9ff');g.state='orb';g.fright=false;
        SFX.ghostEat(chain);burst(g.x,g.y,['#7ef9ff','#ffffff',DEF[g.id].col],14,3);shake=Math.max(shake,0.25);
        if(score>=nextLife){lives++;nextLife+=15000;banner('+1 UP');SFX.life();}
      }else if(p.invulnT<=0){startDeath();return;}
    }
  }
}

/* ================= main step ================= */
function step(dt){
  rotCD=Math.max(0,rotCD-dt);
  for(const s of stars){s.x+=gVec.x*s.z*9*dt;s.y+=gVec.y*s.z*9*dt;if(s.x<-6)s.x+=W+12;if(s.x>W+6)s.x-=W+12;if(s.y<-6)s.y+=H+12;if(s.y>H+6)s.y-=H+12;}
  switch(state){
    case 'START':{
      attractT-=dt;if(attractT<=0){attractT=4;setG((gIndex+1)%4,true);}
      for(const g of ghosts)updGhost(g,dt);
      updParts(dt);break;}
    case 'PLAYING':{
      SFX.tick(frightT>0);
      phaseT-=dt;if(phaseT<=0){phase=phase==='chase'?'scatter':'chase';phaseT=phase==='chase'?7:4.5;}
      if(frightT>0){frightT-=dt;if(frightT<=0){for(const g of ghosts)g.fright=false;chain=0;}}
      updPlayer(dt);
      if(state==='PLAYING'){for(const g of ghosts)updGhost(g,dt);collide();}
      updParts(dt);break;}
    case 'DYING':{
      deathT+=dt;
      if(deathT>0.55&&!deathBurst){deathBurst=true;burst(player.x,player.y,['#ffe14d','#ffb347','#ffffff'],26,4);}
      if(deathT>=1.7){lives--;if(lives<=0)gameOver();else respawnPlayer();}
      updParts(dt);break;}
    case 'CLEAR':{
      clearT+=dt;
      if(clearT>=1.6){wave++;buildPellets();resetGhosts();player.x=10.5;player.y=15.5;player.vx=player.vy=0;player.invulnT=2;frightT=0;phase='chase';phaseT=8;state='PLAYING';banner('WAVE '+wave);}
      updParts(dt);break;}
    case 'GAMEOVER':updParts(dt);break;
  }
  if(state!=='PLAYING')SFX.sirenOff();
  shake=Math.max(0,shake-dt*1.6);flashT=Math.max(0,flashT-dt*2.4);
  syncHud(false);frenzyUI();
}

function drawGhost(g){
  const x=g.x*T,y=g.y*T,r=g.r*1.35*T;
  ctx.save();ctx.translate(x,y);
  if(g.state==='orb'){
    const pr=r*(0.4+0.12*Math.sin(time*22));
    ctx.globalAlpha=0.9;ctx.fillStyle='#dcecff';ctx.shadowColor='#7ef9ff';ctx.shadowBlur=14;
    ctx.beginPath();ctx.arc(0,0,pr,0,TAU);ctx.fill();ctx.restore();return;
  }
  if(g.respawnT>0)ctx.globalAlpha=0.3+0.35*Math.abs(Math.sin(time*14));
  let col=DEF[g.id].col;
  const flashing=g.fright&&frightT<2&&Math.floor(time*8)%2===0;
  if(g.fright)col=flashing?'#eef3ff':'#3b5bff';
  const wv=0.5+0.5*Math.sin(time*10+g.id*1.9),bot=r*0.5,depth=r*(0.22+0.2*wv),seg=r/2;
  ctx.beginPath();ctx.arc(0,-r*0.18,r,Math.PI,0);ctx.lineTo(r,bot);
  for(let i=0;i<4;i++){ctx.lineTo(r-seg*(i+0.5),bot+depth);ctx.lineTo(r-seg*(i+1),bot);}
  ctx.closePath();
  ctx.fillStyle=col;ctx.shadowColor=col;ctx.shadowBlur=g.fright?6:10;ctx.fill();ctx.shadowBlur=0;
  if(g.fright){
    const fc=flashing?'#3b5bff':'#dfe8ff';
    ctx.fillStyle=fc;ctx.beginPath();ctx.arc(-r*0.34,-r*0.22,r*0.15,0,TAU);ctx.arc(r*0.34,-r*0.22,r*0.15,0,TAU);ctx.fill();
    ctx.strokeStyle=fc;ctx.lineWidth=r*0.13;ctx.lineJoin='round';ctx.beginPath();

    ctx.stroke();
  }else{
    const dv=Math.hypot(g.vx,g.vy)||1,ex=g.vx/dv*r*0.14,ey=g.vy/dv*r*0.14;
    for(const s of[-1,1]){
      ctx.fillStyle='#fff';ctx.beginPath();ctx.ellipse(s*r*0.38,-r*0.28,r*0.3,r*0.36,0,0,TAU);ctx.fill();
      ctx.fillStyle='#1b2a6b';ctx.beginPath();ctx.arc(s*r*0.38+ex,-r*0.28+ey,r*0.15,0,TAU);ctx.fill();
