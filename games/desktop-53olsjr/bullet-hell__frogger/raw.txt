/* ---------- update ---------- */
let lastDT=0.016;
function update(dt){
  G.t+=dt;
  const ft=(G.focusOn&&G.mode==='playing')?1:0;
  G.focus+=(ft-G.focus)*Math.min(1,dt*9);
  const ts=1-0.68*G.focus;
  const sdt=dt*ts; lastDT=sdt;

  if(G.hitstop>0){G.hitstop-=dt;upParts(dt*0.25);return;}

  if(G.mode==='start'){
    if(Math.random()<dt*2.2)G.deco.push({x:rnd(COLS),y:-1,vx:rnd(-0.3,0.3),vy:rnd(1.6,2.6),
      r:rnd(0.1,0.2),c:['#ff5fd7','#5ce9ff','#ffb648'][Math.floor(rnd(3))],type:Math.random()<0.3?'dia':'orb'});
    for(let i=G.deco.length-1;i>=0;i--){const b=G.deco[i];b.x+=b.vx*sdt;b.y+=b.vy*sdt;if(b.y>ROWS+1)G.deco.splice(i,1);}
    upParts(sdt);return;
  }
  if(G.mode==='paused'||G.mode==='gameover'){upParts(dt*0.3);return;}

  if(G.mode==='dead'){
    G.deadT-=dt;upParts(dt);
    if(G.deadT<=0){
      if(G.lives>0){resetFrog();G.mode='playing';}
      else{if(G.score>G.best){G.best=G.score;G.newBest=true;
        try{localStorage.setItem('danmakuFroggerBest',String(G.best));}catch(e){}}
        G.mode='gameover';}
    }
    return;
  }

  /* playing / levelup */
  updCars(sdt);updCarr(sdt);
  if(G.mode==='playing'){
    updDrones(sdt);updCurtain(sdt);
    updFrog(sdt);
    if(G.frog.alive){updBullets(sdt);}
    G.time-=sdt;
    const sec=Math.ceil(G.time);
    if(sec<=5&&sec!==G.lowTick){G.lowTick=sec;SFX.tick();}
    if(G.time<=0)die('time');
  }else{ /* levelup banner: world drifts, no input */
    G.lvlT-=dt;
    if(G.lvlT<=0)G.mode='playing';
  }
  upParts(sdt);
}

**How it plays:** hop with arrows/WASD (hold to keep hopping), reach all 5 lily pads before the timer runs out. Drones above fire aimed fans and rings, and cyan shard curtains rain down with a telegraphed safe gap — hold **SHIFT** for focus mode: slow-mo plus a tiny visible hitbox, and graze bullets for close-call points. Cars, turtles, logs and the void still kill, so thread both hazards at once. **P** pause · **M** mute · **R** restart; touch controls appear automatically on mobile.
