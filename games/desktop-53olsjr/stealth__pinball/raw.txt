/* ================= PHYSICS ================= */
function inLane(){return ball.x>440;}
function canCharge(){return S.state==='play'&&inLane()&&ball.y>560&&Math.hypot(ball.vx,ball.vy)<160;}
function updFlip(f,pressed,dt){const target=pressed?f.act:f.rest,d=target-f.ang,rate=pressed?24:16;
  if(Math.abs(d)<=rate*dt){f.om=(target-f.ang)/Math.max(dt,1e-4);f.ang=target;}else{f.om=Math.sign(d)*rate;f.ang+=f.om*dt;}}
function clampSpd(v){const s=Math.hypot(ball.vx,ball.vy);if(s>v){ball.vx*=v/s;ball.vy*=v/s;}}
function hitWall(s){
  const dx=s.bx-s.ax,dy=s.by-s.ay,L2=dx*dx+dy*dy;
  let t=((ball.x-s.ax)*dx+(ball.y-s.ay)*dy)/L2;t=t<0?0:t>1?1:t;
  const cx=s.ax+dx*t,cy=s.ay+dy*t;let nx=ball.x-cx,ny=ball.y-cy;const d2=nx*nx+ny*ny;
  if(d2>=BR*BR)return false;
  const d=Math.sqrt(d2)||1e-4;nx/=d;ny/=d;
  ball.x=cx+nx*BR;ball.y=cy+ny*BR;
  const vn=ball.vx*nx+ball.vy*ny;
  if(vn<0){ball.vx-=1.52*vn*nx;ball.vy-=1.52*vn*ny;}
  if(s.kick&&s.cd<=0&&vn<-60){ball.vx+=nx*s.kick;ball.vy+=ny*s.kick;clampSpd(1650);s.cd=.25;
    addScore(50,s.bx,s.by-14,'+50','#ffd23f');S.slingT=1;sfx.sling();burst((s.ax+s.bx)/2,(s.ay+s.by)/2,5,'#ffd23f',220);}
  return true;
}
function hitFlip(f){
  const tx=f.px+Math.cos(f.ang)*f.len,ty=f.py+Math.sin(f.ang)*f.len;
  const dx=tx-f.px,dy=ty-f.py,L2=dx*dx+dy*dy;
  let t=((ball.x-f.px)*dx+(ball.y-f.py)*dy)/L2;t=t<0?0:t>1?1:t;
  const cx=f.px+dx*t,cy=f.py+dy*t;let nx=ball.x-cx,ny=ball.y-cy;const d=Math.hypot(nx,ny),R=f.w+BR;
  if(d>=R)return;
  nx/=d||1e-4;ny/=d||1e-4;ball.x=cx+nx*R;ball.y=cy+ny*R;
  const rx=cx-f.px,ry=cy-f.py,vsx=-f.om*ry,vsy=f.om*rx;
  const vn=(ball.vx-vsx)*nx+(ball.vy-vsy)*ny;
  if(vn<0){ball.vx-=1.32*vn*nx;ball.vy-=1.32*vn*ny;}
}
function hitCircles(){
  for(const b of bumpers){const d=Math.hypot(ball.x-b.x,ball.y-b.y),R=b.r+BR;
    if(d<R){const nx=(ball.x-b.x)/d,ny=(ball.y-b.y)/d;ball.x=b.x+nx*R;ball.y=b.y+ny*R;
      ball.vx=nx*980;ball.vy=ny*980;b.flash=1;S.chain=Math.min(8,S.chain+1);S.chainT=1.7;
      const pts=150+(S.chain-1)*50;addScore(pts,b.x,b.y-b.r-12,'+'+pts,'#ffd23f');
      if(S.chain>=3)popup('CHAIN ×'+S.chain,240,96,'#59d9f2',13);
      burst(ball.x,ball.y,8,b.c,320);S.shake=Math.max(S.shake,3.5);sfx.bumper(S.chain);}}
  for(const p of minipins){const d=Math.hypot(ball.x-p.x,ball.y-p.y),R=p.r+BR;
    if(d<R){const nx=(ball.x-p.x)/d,ny=(ball.y-p.y)/d;ball.x=p.x+nx*R;ball.y=p.y+ny*R;
      ball.vx=nx*640;ball.vy=ny*640;p.flash=1;addScore(75,p.x,p.y-18,'+75','#cfe3f5');burst(ball.x,ball.y,5,'#cfe3f5',220);sfx.pin();}}
  for(const g of guards){const dx=ball.x-g.x,dy=ball.y-g.y,d=Math.hypot(dx,dy),R=11+BR;
    if(d<R&&d>1e-4){const nx=dx/d,ny=dy/d;ball.x=g.x+nx*R;ball.y=g.y+ny*R;
      const vn=ball.vx*nx+ball.vy*ny;if(vn<0){ball.vx-=1.5*vn*nx;ball.vy-=1.5*vn*ny;}
      if(Math.hypot(ball.vx,ball.vy)<380){ball.vx=nx*380;ball.vy=ny*380;}
      spotNow(g);}}
}
function ballStep(dt){
  updFlip(flippers[0],kbL||ptrL,dt);updFlip(flippers[1],kbR||ptrR,dt);
  if(S.kbP||ptrP){if(canCharge()){S.charge=Math.min(1,S.charge+dt*1.25);ball.x=452;ball.vx=0;ball.vy=0;ball.y=663-S.charge*14;}}
  let sp=Math.hypot(ball.vx,ball.vy);const n=Math.max(1,Math.ceil(sp*dt/(BR*0.5))),sdt=dt/n;
  for(let i=0;i<n;i++){
    ball.vy+=GRAV*sdt;sp=Math.hypot(ball.vx,ball.vy);if(sp>MAXSPD){const k=MAXSPD/sp;ball.vx*=k;ball.vy*=k;}
    ball.x+=ball.vx*sdt;ball.y+=ball.vy*sdt;
    for(const s of walls)hitWall(s);
    hitFlip(flippers[0]);hitFlip(flippers[1]);hitCircles();
  }
  if(ball.y-BR>TH+6)onDrain();
}
function releasePlunger(){
  if((S.state==='play')&&inLane()&&S.charge>0.02){ball.vy=-(1150+700*S.charge);ball.vx=0;sfx.launch(S.charge);}
  S.charge=0;
}

/* ================= UPDATE ================= */
function update(dt){
  S.t+=dt;
  S.shake=Math.max(0,S.shake-dt*26);S.redFlash=Math.max(0,S.redFlash-dt*1.4);
  S.goldFlash=Math.max(0,S.goldFlash-dt*1.1);S.alarmFlash=Math.max(0,S.alarmFlash-dt*2.2);
  for(const b of bumpers)b.flash=Math.max(0,b.flash-dt*3);
  for(const p of minipins)p.flash=Math.max(0,p.flash-dt*3);
  S.slingT=Math.max(0,S.slingT-dt*4);
  for(const s of walls)if(s.cd>0)s.cd-=dt;
  for(const g of guards){
    g.cool=Math.max(0,g.cool-dt);
    if(g.type==='patrol'){g.t+=dt*g.speed/g.len;const ph=g.t%2,u=ph<1?ph:2-ph;
      g.x=g.x0+(g.x1-g.x0)*u;g.y=g.y0+(g.y1-g.y0)*u;g.face=ph<1?g.fwd:g.fback;}
    else{g.ph+=dt*g.rate;g.face=g.base+Math.sin(g.ph)*g.amp;}
    let inC=false;
    if(S.state==='play'&&!isHidden(ball.x,ball.y)&&g.cool<=0){
      const dx=ball.x-g.x,dy=ball.y-g.y,d=Math.hypot(dx,dy);
      if(d<g.range&&d>1){let da=Math.atan2(dy,dx)-g.face;
        while(da>Math.PI)da-=6.283;while(da<-Math.PI)da+=6.283;
        if(Math.abs(da)<g.half)inC=true;}}
    if(inC){g.dwell+=dt;if(g.dwell>=DWELL)spotNow(g);}else g.dwell=Math.max(0,g.dwell-dt*2.5);
  }
  for(let i=smokes.length-1;i>=0;i--){smokes[i].life-=dt;if(smokes[i].life<=0)smokes.splice(i,1);}
  for(let i=particles.length-1;i>=0;i--){const p=particles[i];p.life-=dt;
    if(p.life<=0){particles.splice(i,1);continue;}
    if(p.type==='spark'){p.vy+=620*dt;p.vx*=1-3*dt;p.x+=p.vx*dt;p.y+=p.vy*dt;}
    else if(p.type==='smoke'){p.vy-=22*dt;p.x+=(Math.sin(S.t*2+p.size)*14)*dt;p.y+=p.vy*dt;p.size+=dt*26;}
    else{p.vy+=150*dt;p.rot+=p.vr*dt;p.x+=p.vx*dt+Math.sin(S.t*3+p.rot)*12*dt;p.y+=p.vy*dt;}}
  for(let i=popups.length-1;i>=0;i--){const p=popups[i];p.life-=dt;p.y-=34*dt;if(p.life<=0)popups.splice(i,1);}

  switch(S.state){
    case 'play':{
      S.alarm=Math.max(0,S.alarm-dt/7);S.smokeCd=Math.max(0,S.smokeCd-dt);
      if(S.chainT>0){S.chainT-=dt;if(S.chainT<=0)S.chain=0;}
      ballStep(dt);if(S.state!=='play')break;
      for(const d of diamonds)if(!d.taken&&Math.hypot(ball.x-d.x,ball.y-d.y)<30)collectDiamond(d);
      const tl=trail[0];
      if(!tl||Math.hypot(ball.x-tl.x,ball.y-tl.y)>5){trail.unshift({x:ball.x,y:ball.y});if(trail.length>16)trail.pop();}
      if(!inLane()&&Math.hypot(ball.vx,ball.vy)<12){S.stillT+=dt;
        if(S.stillT>2.6){ball.vy-=150;ball.vx+=(Math.random()-.5)*90;S.stillT=0;}}else S.stillT=0;
      if(ball.x<-40||ball.x>520||ball.y>820)onDrain();
      break;}
    case 'winwait':{ballStep(dt);S.stateT-=dt;if(S.stateT<=0)finishWin();break;}
    case 'lostwait':case 'caught':{S.stateT-=dt;if(S.stateT<=0)finalizeLoss();break;}
  }
}

**How to play:** Hold `SPACE` (or the lower-center on touch) to charge the plunger, release to launch. Use `A`/`←` and `D`/`→` for the flippers. Guards patrol with red vision cones — linger in one and you get spotted; three detections means you're caught.
