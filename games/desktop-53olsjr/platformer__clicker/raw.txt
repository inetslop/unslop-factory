function skyAt(y){
  if(y<=SKY[0][0])return SKY[0][1];
  for(let i=0;i<SKY.length-1;i++){
    const a=SKY[i],b=SKY[i+1];
    if(y<=b[0]){const k=(y-a[0])/(b[0]-a[0]);return [lerp(a[1][0],b[1][0],k),lerp(a[1][1],b[1][1],k),lerp(a[1][2],b[1][2],k)];}
  }
  return SKY[SKY.length-1][1];
}
