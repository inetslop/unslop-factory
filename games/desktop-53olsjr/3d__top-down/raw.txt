      if(e.dead||e.spawn<1) continue;
      const dx=e.x-b.x, dy=e.y-b.y;
      if(dx*dx+dy*dy<(e.r+.45)*(e+r===0?0:(e.r+.45))){
        e.hp--; e.hit=1; sfx.hit(); burst(b.x,b.y,b.z,6,[200,255,245],5);
        const d=Math.hypot(dx,dy)||1; e.x+=dx/d*.3; e.y+=dy/d*.3;
        if(e.hp<=0) killEnemy(e);
        dead=true; break;}
    }
    if(dead) bolts.splice(i,1);
  }
}
function orbUpdate(dt){
  for(let i=orbs.length-1;i>=0;i--){const o=orbs[i];
    o.x+=o.vx*dt; o.y+=o.vy*dt; o.life-=dt;
    let dead=o.life<=0||Math.abs(o.x)>A+2||Math.abs(o.y)>A+2;
    if(!dead) for(const r of rocks){
      if(o.x>r.x-r.w/2&&o.x<r.x+r.w/2&&o.y>r.y-r.d/2&&o.y<r.y+r.d/2){burst(o.x,o.y,o.z,8,[255,180,90],5); dead=true; break;}
    }
    if(!dead&&!player.dead&&player.invuln<=0){
      const dx=player.x-o.x, dy=player.y-o.y;
      if(dx*dx+dy*dy<.9*.9){damagePlayer(10,o.x,o.y); burst(o.x,o.y,o.z,8,[255,170,80],5); dead=true;}
    }
    if(dead) orbs.splice(i,1);
  }
}
function pickupUpdate(dt){
  const p=player;
  for(const pk of pickups){
    pk.life-=dt;
    if(!p.dead){
      const dx=p.x-pk.x, dy=p.y-pk.y, d=Math.hypot(dx,dy)||1;
      if(d<3.5){pk.x+=dx/d*7*dt; pk.y+=dy/d*7*dt;}
      if(d<1.25){
        pk.gone=true; ringAt(pk.x,pk.y,[255,220,140]);
        if(pk.kind==='pearl'){score+=15; textPop(pk.x,pk.y,'+15','#ffd166'); sfx.pearl();}
        else if(pk.kind==='heal'){p.hp=Math.min(100,p.hp+22); textPop(pk.x,pk.y,'+22 HULL','#7dffb0'); sfx.heal();}
        else {p.over=7; textPop(pk.x,pk.y,'OVERDRIVE!','#7dfcff'); sfx.over();}
      }
    }
  }
  pickups=pickups.filter(pk=>!pk.gone&&pk.life>0);
}

/* ============================== wave logic ============================== */
function waveLogic(dt){
  if(spawnQueue.length){spawnT-=dt; if(spawnT<=0){spawnEnemy(spawnQueue.shift()); spawnT=.85;}}
  else if(!enemies.length&&!player.dead){
    if(waveCleared===0){
      waveCleared=1; const b=40+wave*15; score+=b;
      textPop(player.x,player.y,'CLEAR +'+b,'#9ffce8'); tone('sine',700,1050,.25,.15);
    }
    waveT+=dt;
    if(waveT>1.9){wave++; buildWave(wave); banner('WAVE '+wave+' — '+wname(wave)); sfx.wave();}
  } else waveCleared=0;
}

/* ============================== cameras ============================== */
function followCam(dt){
  const p=player; let ax=p.x, ay=p.y;
  if(!p.dead){const aw=aimWorld(); if(aw){ax=p.x+(aw.x-p.x)*.45; ay=p.y+(aw.y-p.y)*.45;}}
  const zt=(mode==='over'?15:11.6)+Math.sin(T*.7)*.2;
  const k=1-Math.exp(-dt*4.5);
  cam.x+=(ax-cam.x)*k; cam.y+=(ay-cam.y)*k; cam.z+=(zt-cam.z)*k;
  makeBasis(p.x+(ax-p.x)*.6, p.y+(ay-p.y)*.6, 0);
  if(cam.shake>0){cam.x+=rand(-1,1)*cam.shake; cam.y+=rand(-1,1)*cam.shake;}
}
function attractCam(dt){
  const a=T*.12, tx=Math.cos(a)*20, ty=Math.sin(a)*20, tz=17;
  const k=1-Math.exp(-dt*3);
  cam.x+=(tx-cam.x)*k; cam.y+=(ty-cam.y)*k; cam.z+=(tz-cam.z)*k;
  makeBasis(Math.sin(T*.05)*4, Math.cos(T*.04)*4, 0);
}

/* ============================== update master ============================== */
function update(dt){
  if(mode==='pause') return;
  cam.shake*=Math.exp(-5*dt);
  updateMotes(dt); updateParts(dt);
  if(mode==='attract'){attractCam(dt); for(const e of enemies) wander(e,dt);}
  else if(mode==='play'){
    playerUpdate(dt); waveLogic(dt); enemyUpdate(dt); boltUpdate(dt); orbUpdate(dt); pickupUpdate(dt); followCam(dt);
  }
  else if(mode==='over'){
    for(const e of enemies) wander(e,dt*.5); boltUpdate(dt); orbUpdate(dt); followCam(dt);
    deathT+=dt; if(deathT>1.6&&!shownOver){shownOver=true; showOver();}
  }
  enemies=enemies.filter(e=>!e.dead);
  dmgFlash=Math.max(0,dmgFlash-dt*2.2);
}

function quadP(p1,p2,p3,p4,col){ctx.fillStyle=col; ctx.beginPath(); ctx.moveTo(p1.x,p1.y); ctx.lineTo(p2.x,p2.y); ctx.lineTo(p3.x,p3.y); ctx.lineTo(p4.x,p4.y); ctx.closePath(); ctx.fill();}
function floorShadow(px,py,s,r,al=.32){ctx.fillStyle='rgba(3,14,20,'+al+')'; ctx.beginPath(); ctx.ellipse(px+s*.16,py+s*.09,s*r,s*r*.38,0,0,TAU); ctx.fill();}
function glowC(x,y,r,col,a){const g=ctx.createRadialGradient(x,y,0,x,y,r); g.addColorStop(0,'rgba('+col[0]+','+col[1]+','+col[2]+','+a+')'); g.addColorStop(1,'rgba('+col[0]+','+col[1]+','+col[2]+',0)'); ctx.fillStyle=g; ctx.beginPath(); ctx.arc(x,y,r,0,TAU); ctx.fill();}
function rockCol(base,bri,d){
  const f=clamp((d-14)/50,0,.9);
  return 'rgb('+(((base[0]*bri)*(1-f)+8*f)|0)+','+(((base[1]*bri)*(1-f)+32*f)|0)+','+(((base[2]*bri)*(1-f)+46*f)|0)+')';
}

function drawRock(r){
  const off=[[-r.w/2,-r.d/2],[r.w/2,-r.d/2],[r.w/2,r.d/2],[-r.w/2,r.d/2]];
  const bw=[],tb=[];
  for(let i=0;i<4;i++){
    const p1=project(r.x+off[i][0],r.y+off[i][1],.02); if(!p1) return; bw.push(p1);
    const p2=project(r.x+off[i][0]*r.taper+r.j[i][0], r.y+off[i][1]*r.taper+r.j[i][1], r.h); if(!p2) return; tb.push(p2);
  }
  for(let i=0;i<4;i++){
    const j=(i+1)&3;
    const ex=off[j][0]-off[i][0], ey=off[j][1]-off[i][1];
    const nx=ey, ny=-ex;
    const mxw=r.x+( (off[i][0]+off[j][0])/2 ), myw=r.y+((off[i][1]+off[j][1])/2);
    if(nx*(cam.x-mxw)+ny*(cam.y-myw)<=0) continue;
    const d=depthOf(mxw,myw,r.h/2);
    const bri=.4+.6*Math.max(0,nx*LIGHT[0]+ny*LIGHT[1]);
    quadP(bw[i],bw[j],tb[j],tb[i],rockCol(r.base,bri,d));
  }
  const d=depthOf(r.x,r.y,r.h), top=rockCol([r.base[0]*1.28,r.base[1]*1.28,r.base[2]*1.3],.95,d);
  quadP(tb[0],tb[1],tb[2],tb[3],top);
  ctx.strokeStyle='rgba(170,240,225,.13)'; ctx.lineWidth=1;
  ctx.beginPath(); ctx.moveTo(tb[0].x,tb[0].y); for(let i=1;i<4;i++)ctx.lineTo(tb[i].x,tb[i].y); ctx.closePath(); ctx.stroke();
  for(const c of r.coral){
    const px=r.x-r.w/2+c.fx*r.w, py=r.y-r.d/2+c.fy*r.d;
    const p=project(px,py,r.h+.06); if(!p) continue;
    ctx.globalCompositeOperation='lighter';
    glowC(p.x,p.y,Math.max(2,p.s*.2),c.c,.35+.25*Math.sin(T*2+c.ph));
    ctx.globalCompositeOperation='source-over';
  }
}

  ctx.lineWidth=1;
  for(let x=x0;x<=cam.x+ext;x+=g){
    const d=depthOf(x,cam.y,0), al=.15*(1-clamp((d-6)/48,0,1)); if(al<.012) continue;
    const a=project(x,cam.y-ext,0), b=project(x,cam.y+ext,0);
    if(a&&b){ctx.strokeStyle='rgba(120,210,200,'+al+')'; ctx.beginPath(); ctx.moveTo(a.x,a.y); ctx.lineTo(b.x,b.y); ctx.stroke();}
  }
  for(let y=x0;y<=cam.y+ext;y+=g){
    const d=depthOf(cam.x,y,0), al=.15*(1-clamp((d-6)/48,0,1)); if(al<.012) continue;
    const a=project(cam.x-ext,y,0), b=project(cam.x+ext,y,0);
    if(a&&b){ctx.strokeStyle='rgba(120,210,200,'+al+')'; ctx.beginPath(); ctx.moveTo(a.x,a.y); ctx.lineTo(b.x,b.y); ctx.stroke();}
  }
  // caustic light blobs + player pool
  ctx.globalCompositeOperation='lighter';
  for(let i=0;i<4;i++){
    const wx=Math.sin(T*.06+i*2.1)*17, wy=Math.cos(T*.05+i*1.3)*17;
    const p=project(wx,wy,0); if(p) glowC(p.x,p.y,Math.max(10,p.s*9),[120,255,220],.045);
  }
  if(mode!=='attract'&&!player.dead){const p=project(player.x,player.y,0); if(p) glowC(p.x,p.y,Math.max(10,p.s*6.5),[110,240,230],.07);}
  ctx.globalCompositeOperation='source-over';
  // distance fog
  ctx.fillStyle=fogGrad; ctx.fillRect(0,0,W,H*.62);
  // sorted world
  const items=[];
  for(const r of rocks) items.push([depthOf(r.x,r.y,r.h/2),0,r]);
  for(const k of kelps) items.push([depthOf(k.x,k.y,k.h/2),1,k]);
  for(const pk of pickups) items.push([depthOf(pk.x,pk.y,.6),2,pk]);
  for(const e of enemies) items.push([depthOf(e.x,e.y,1),3,e]);
  for(const b of bolts) items.push([depthOf(b.x,b.y,b.z),4,b]);
  for(const o of orbs) items.push([depthOf(o.x,o.y,o.z),5,o]);
  if(mode!=='attract') items.push([depthOf(player.x,player.y,.8),6,player]);
  items.sort((a,b)=>b[0]-a[0]);
  for(const it of items){
    if(it[1]===0) drawRock(it[2]);
    else if(it[1]===1) drawKelp(it[2]);
    else if(it[1]===2) drawPickup(it[2]);
    else if(it[1]===3){const e=it[2]; if(e.type==='jelly')drawJelly(e); else if(e.type==='mine')drawMine(e); else drawDasher(e);}
    else if(it[1]===4) drawBolt(it[2]);
    else if(it[1]===5) drawOrb(it[2]);
    else drawPlayer();
  }
  // particles + motes
  ctx.globalCompositeOperation='lighter';
  for(const p of parts){
    const al=clamp(p.life/p.max,0,1);
    if(p.type==='bubble'){
      const pr=project(p.x,p.y,p.z); if(!pr) continue;
      ctx.fillStyle='rgba(210,255,250,'+(.3*al)+')';
      ctx.beginPath(); ctx.arc(pr.x,pr.y,Math.max(.8,pr.s*p.r),0,TAU); ctx.fill();
    } else if(p.type==='spark'){
      const a=project(p.x-p.vx*.04,p.y-p.vy*.04,p.z-.vz*.04), b=project(p.x,p.y,p.z);
      if(!a||!b) continue;
      ctx.strokeStyle='rgba('+p.col[0]+','+p.col[1]+','+p.col[2]+','+(.8*al)+')';
      ctx.lineWidth=Math.max(1,b.s*.07);
      ctx.beginPath(); ctx.moveTo(a.x,a.y); ctx.lineTo(b.x,b.y); ctx.stroke();
    } else if(p.type==='ring'){
      const pr=project(p.x,p.y,0); if(!pr) continue;
      const rr=(1-al)*pr.s*7+pr.s*.4;
      ctx.strokeStyle='rgba('+p.col[0]+','+p.col[1]+','+p.col[2]+','+(.5*al)+')';
      ctx.lineWidth=Math.max(1,pr.s*.08);
      ctx.beginPath(); ctx.ellipse(pr.x,pr.y,rr,rr*.4,0,0,TAU); ctx.stroke();
    } else if(p.type==='text'){
      const pr=project(p.x,p.y,p.z); if(!pr) continue;
      const fs=Math.max(12,pr.s*.34);
      ctx.font='bold '+fs+'px Consolas, monospace';
      ctx.textAlign='center'; ctx.fillStyle=p.col; ctx.globalAlpha=al;
      ctx.fillText(p.text,pr.x,pr.y); ctx.globalAlpha=1;
    }
  }
  for(const m of motes){
    const pr=project(m.x,m.y,m.z); if(!pr) continue;
    const d=depthOf(m.x,m.y,m.z), al=clamp(1-d/60,0,.55)*.5;
    if(al<=.02) continue;
    ctx.fillStyle='rgba(190,245,235,'+al+')';
    ctx.beginPath(); ctx.arc(pr.x,pr.y,clamp(pr.s*.05,.5,2),0,TAU); ctx.fill();
  }
  // god rays
  for(const r of RAYS){
    const cx=(r.x+Math.sin(T*r.s+r.p)*.04)*W;
    const slant=(r.x<.5?1:-1)*W*.12, w=r.w*W;
    ctx.fillStyle='rgba(150,235,220,.03)';
    ctx.beginPath();
    ctx.moveTo(cx-w,-20); ctx.lineTo(cx+w,-20);
    ctx.lineTo(cx+w*1.9+slant,H*.92); ctx.lineTo(cx-w*1.9+slant,H*.92);
    ctx.closePath(); ctx.fill();
  }
  ctx.globalCompositeOperation='source-over';
}

/* ============================== HUD tick ============================== */
let lastHp=-1,lastScore=-1;
function hudTick(){
  el.dmg.style.opacity=dmgFlash.toFixed(2);
  if(mode==='attract') return;
  const hp=Math.max(0,player.hp|0);
  if(hp!==lastHp){lastHp=hp;
    el.hpbar.style.width=hp+'%';
    el.hpbar.style.background='hsl('+(hp*1.2)+',80%,55%)';
    el.hpnum.textContent=hp;
    el.pHull.classList.toggle('low',hp<30);
  }
  if(score!==lastScore){lastScore=score; el.score.textContent=score;}
  const lb=Math.max(best,score);
  if(el.best.textContent!=lb) el.best.textContent=lb;
  el.wave.textContent=wave;
  el.depth.textContent='DEPTH '+(4096+(wave-1)*137)+'M';
  if(player.over>0){el.ovchip.style.display='block'; el.ovbar.style.width=(clamp(player.over/7,0,1)*100)+'%';}
  else el.ovchip.style.display='none';
}
