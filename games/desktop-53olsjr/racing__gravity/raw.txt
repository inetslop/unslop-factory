  addP({x:bx,y:by,vx:-car.v*.5-60-Math.random()*90,vy:(Math.random()-.5)*110,g:0,
    life:.22+Math.random()*.2,max:.4,s:3.5+Math.random()*4.5,c:["#ffd24a","#ff8a2a","#ff5a2a"][(Math.random()*3)|0],add:true});
}

/* ================= gameplay ================= */
function doFlip(){
  if(car.cd>0)return;car.dir*=-1;car.vy=car.dir*320;car.cd=FLIP_CD;sFlip();
  ring(car.x,car.y-car.dir*CAR_H*.6,"rgba(70,230,255,.9)",46);
}
function crash(cx,cy){
  if(car.inv>0||state==="count")return;
  car.v*=.38;car.vy=-car.dir*240;car.cd=Math.max(car.cd,.5);car.inv=1.2;
  shake=1;flashR=.9;sCrash();
  burst(cx,cy,18,["#ffd24a","#ff8a2a","#ff4d2a"],340,.5,true,600,7);
  burst(car.x,car.y-car.dir*CAR_H*.5,6,["#5a6b8c","#3a4a68"],120,.7,false,0,9);
  ring(cx,cy,"rgba(255,90,50,.9)",60);
}
function itemWX(p){return p+Math.round((camX-p)/L)*L}

function autoDrive(dt){
  car.autoT-=dt;const s=car.dir>0?1:-1,os=-s;let threat=1e9,opp=1e9;
  for(const o of obstacles){const dx=itemWX(o.x)-car.x;
    if(o.side===s&&dx>-30&&dx<360)threat=Math.min(threat,dx);
    else if(o.side===os&&dx>20&&dx<170)opp=Math.min(opp,dx);}
  if(car.cd<=0&&(threat<240||(car.autoT<=0&&opp>150))){doFlip();car.autoT=1.6+Math.random()*2.6;}
  return 1;
}

function finishRace(){
  state="done";totalT=raceT;
  score=Math.max(0,Math.round(10000-totalT*35+coresGot*220));
  const bT=parseFloat(loadLS(KT)||"0"),bS=parseInt(loadLS(KS)||"0",10);
  let nr=false,ns=false;
  if(!bT||totalT<bT){saveLS(KT,totalT.toFixed(2));nr=true;}
  if(score>bS){saveLS(KS,score);ns=true;}
  newRec=nr||ns;sFinish();
  burst(car.x,car.y,50,["#ffd24a","#46e6ff","#ffffff","#ff8a2a"],420,1.6,false,900,7);
  flashG=.8;banner("FINISH!","b-gold",1300);
  setTimeout(()=>{if(state!=="done")return;
    $("dTotal").textContent=fmt(totalT);$("dLap").textContent=bestLap===Infinity?"—":fmt(bestLap);
    $("dCores").textContent=coresGot+" / "+cores.length;$("dScore").textContent=score.toLocaleString();
    $("recBadge").classList.toggle("on",newRec);showOv($("ovDone"));
  },950);
}

function update(dt,t){
  tNow=t;
  if(state==="count"){
    cdT-=dt;const n=Math.max(0,Math.ceil(cdT));
    if(n!==lastCount&&n<=3){lastCount=n;banner(String(n),"b-cyan",750);sBeep(false);}
    if(cdT<=0){state="race";banner("GO!","b-gold",800);sBeep(true);
      $("hint").classList.add("show");clearTimeout(hintTO);hintTO=setTimeout(()=>$("hint").classList.remove("show"),5200);}
    engineTick(90,.6,t,true);return;
  }
  // input
  let thr=0,brk=0;
  if(state==="race"){
    thr=(keys.ArrowRight||keys.KeyD||touch.gas)?1:0;brk=(keys.ArrowLeft||keys.KeyA||touch.brake)?1:0;
    if(flipQueued){flipQueued=false;doFlip();}
  }else if(state==="menu"){thr=autoDrive(dt);}
  else if(state==="done"){car.v=Math.max(0,car.v-700*dt);}
  car.thr=thr;
  // speed
  car.v+=(thr*ACCEL-brk*BRAKE)*dt;car.v-=car.v*.5*dt;
  car.v=clamp(car.v,0,car.boost>0?960:MAXV);
  if(flipQueued&&state==="race"){flipQueued=false;doFlip();}
  // gravity
  car.cd-=dt;car.inv-=dt;car.boost=Math.max(0,car.boost-dt);
  car.vy+=GRAV*car.dir*dt;car.y+=car.vy*dt;
  const fy=floorY(car.x),cy=ceilY(car.x),wasS=car.onS;
  if(car.dir>0&&car.y+CAR_H/2>=fy){if(!wasS){sLand();burst(car.x,fy,8,["#7fd9ff","#4a6b95"],130,.4,false,-200,5);}car.y=fy-CAR_H/2;car.vy=0;car.onS=true;}
  else if(car.dir<0&&car.y-CAR_H/2<=cy){if(!wasS){sLand();burst(car.x,cy,8,["#ffd9a0","#95714a"],130,.4,false,200,5);}car.y=cy+CAR_H/2;car.vy=0;car.onS=true;}
  else car.onS=false;
  car.x+=car.v*dt;
  car.vd+=(car.dir-car.vd)*Math.min(1,dt*9);
  // exhaust
  if(thr>0&&(car.v>40||state==="menu")){exhaust();if(car.v>380)exhaust();}
  // collisions
  for(const o of obstacles){
    const dx=o.x-car.x;if(dx<-o.w-70||dx>70+o.w)continue;
    if(o.side===car.dir&&Math.abs(dx)<o.w/2+CAR_W/2+8){
      const hitH=o.type==="spike"?o.h*.7:o.h,inset=o.type==="spike"?8:0;
      const b0=o.side>0?fy-hitH:cy,b1=o.side>0?fy:cy+hitH;
      if(car.x+CAR_W/2>o.x+inset&&car.x-CAR_W/2<o.x+o.w-inset&&car.y+CAR_H/2>b0&&car.y-CAR_H/2<b1){
        crash(o.x+o.w/2,o.side>0?b0:b1);break;}
    }
  }
  for(const p of pads){p.cd-=dt;p.flash=Math.max(0,p.flash-dt*3);
    if(p.side===car.dir&&car.onS&&p.cd<=0&&car.x+CAR_W/2>p.x&&car.x-CAR_W/2<p.x+p.w){
      p.cd=1.4;p.flash=1;car.v=Math.min(car.v+340,960);car.boost=1;sBoost();
      for(let i=0;i<10;i++)addP({x:car.x-CAR_W/2,y:car.y+(Math.random()-.5)*20,vx:-700-Math.random()*400,vy:(Math.random()-.5)*60,g:0,life:.3,max:.3,s:3,c:"#7df0ff",add:true});}}
  for(const c of cores){if(c.taken)continue;const dx=c.x-car.x,dy=c.y-car.y;
    if(dx*dx+dy*dy<40*40){c.taken=true;coresGot++;sPick();flashG=Math.max(flashG,.18);
      burst(c.x,c.y,12,["#ffd24a","#fff3c0"],260,.5,true,0,5);ring(c.x,c.y,"rgba(255,210,74,.9)",38);}}
  // laps
  if(state==="race"){
    raceT+=dt;
    const ld=Math.floor((car.x-STARTX)/L);
    if(ld>lapsDone){lapsDone=ld;const lt=raceT-lapStart;lapStart=raceT;if(lt<bestLap)bestLap=lt;
      if(lapsDone>=3)finishRace();
      else{banner(lapsDone===2?"FINAL LAP":"LAP "+(lapsDone+1),"b-orange",1000);sLap();}}
  }
  // camera
  const lookX=car.x+(state==="menu"?380:260+car.v*.18);
  camX+=((lookX-(VW/S)*.34)-camX)*Math.min(1,dt*5);
  camY+=(midY(car.x)-camY)*Math.min(1,dt*3.2);
  // fx decay
  shake=Math.max(0,shake-dt*2.4);flashR=Math.max(0,flashR-dt*2.2);flashG=Math.max(0,flashG-dt*2.6);
  engineTick(car.v,thr,t,true);
}
