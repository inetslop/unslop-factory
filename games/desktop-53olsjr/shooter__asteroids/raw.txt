let AC=null,master=null,noiseBuf=null,engineGain=null,engOn=false,muted=false;
function initAudio(){
  if(AC){ if(AC.state==='suspended')AC.resume(); return; }
  const Ctx=window.AudioContext||window.webkitAudioContext; if(!Ctx)return;
  AC=new Ctx();
  master=AC.createGain(); master.gain.value=muted?0:0.55; master.connect(AC.destination);
  noiseBuf=AC.createBuffer(1,AC.sampleRate,AC.sampleRate);
  const d=noiseBuf.getChannelData(0); for(let i=0;i<d.length;i++)d[i]=Math.random()*2-1;
  const src=AC.createBufferSource(); src.buffer=noiseBuf; src.loop=true;
  const f=AC.createBiquadFilter(); f.type='bandpass'; f.frequency.value=95; f.Q.value=.7;
  engineGain=AC.createGain(); engineGain.gain.value=0;
  src.connect(f); f.connect(engineGain); engineGain.connect(master); src.start();
}
function setEngine(on){ if(on===engOn)return; engOn=on; if(engineGain)engineGain.gain.setTargetAtTime(on?.3:0,AC.currentTime,.05); }
function env(g,t,a,peak,dur){ g.gain.setValueAtTime(.0001,t); g.gain.linearRampToValueAtTime(peak,t+a); g.gain.exponentialRampToValueAtTime(.0001,t+dur); }
function tone(type,f0,f1,t0,dur,peak){ if(!AC||muted)return; const t=AC.currentTime+t0;
  const o=AC.createOscillator(); o.type=type; o.frequency.setValueAtTime(f0,t);
  if(f1)o.frequency.exponentialRampToValueAtTime(Math.max(f1,1),t+dur);
  const g=AC.createGain(); env(g,t,.005,peak,dur);
  o.connect(g); g.connect(master); o.start(t); o.stop(t+dur+.02); }
function noiseHit(f0,f1,dur,peak){ if(!AC||muted)return; const t=AC.currentTime;
  const s=AC.createBufferSource(); s.buffer=noiseBuf; s.loop=true;
  const f=AC.createBiquadFilter(); f.type='lowpass'; f.frequency.setValueAtTime(f0,t);
  f.frequency.exponentialRampToValueAtTime(Math.max(f1,20),t+dur);
  const g=AC.createGain(); env(g,t,.004,peak,dur);
  s.connect(f); f.connect(g); g.connect(master); s.start(t); s.stop(t+dur+.05); }
const sfx={
  shoot(){ tone('square',920+Math.random()*80,160,0,.09,.14); },
  boom(sz){ const d=.35+.28*sz; noiseHit(700+450*sz,50,d,.4); tone('sine',140+30*sz,38,d*.8,.35); },
  warp(){ tone('sawtooth',140,1500,0,.32,.18); noiseHit(300,4000,.35,.22); },
  pick(){ tone('triangle',620,0,0,.08,.2); tone('triangle',930,0,.09,.12,.2); },
  jingle(){ [523,659,784].forEach((f,i)=>tone('triangle',f,0,i*.09,.12,.18)); },
  over(){ [330,262,196,147].forEach((f,i)=>tone('sawtooth',f,0,i*.17,.22,.15)); },
  shieldDown(){ tone('square',300,80,0,.25,.18); noiseHit(900,100,.25,.18); }
};

function makeRock(x,y,size){
  const verts=[],n=9+(Math.random()*3|0);
  for(let i=0;i<n;i++)verts.push(.72+Math.random()*.42);
  const sp=(26+Math.random()*38)*(1+(wave-1)*.07)*(size===1?1.55:size===2?1.15:1);
  const a=Math.random()*TAU;
  return {x,y,size,verts,n,r:ROCKR[size],rot:Math.random()*TAU,vr:(Math.random()-.5)*1.8,vx:Math.cos(a)*sp,vy:Math.sin(a)*sp};
}

  }
}

function burst(x,y,col,n,sp){
  if(parts.length>380)parts.splice(0,parts.length-380);
  for(let i=0;i<n;i++){ const a=Math.random()*TAU,s=sp*(.3+Math.random()*.9);
    parts.push({x,y,vx:Math.cos(a)*s,vy:Math.sin(a)*s,life:.4+Math.random()*.6,max:1,col,sz:1+Math.random()*2.5}); }
}
function ring(x,y,col,r){ parts.push({x,y,life:.5,max:.5,col,sz:r,ring:true}); }
function popup(x,y,text,col){ pops.push({x,y,text,col,life:1}); }
function updateFx(dt){
  for(let i=parts.length-1;i>=0;i--){ const p=parts[i]; p.life-=dt;
    if(p.life<=0){ parts.splice(i,1); continue; }
    if(!p.ring){ p.x+=p.vx*dt; p.y+=p.vy*dt; const d=Math.exp(-2.5*dt); p.vx*=d; p.vy*=d; } }
  for(let i=pops.length-1;i>=0;i--){ const p=pops[i]; p.life-=dt*1.1; p.y-=28*dt; if(p.life<=0)pops.splice(i,1); }
}

function breakRock(i){
  const r=rocks[i]; rocks.splice(i,1);
  score+=ROCKPTS[r.size]; kills++;
  popup(r.x,r.y,'+'+ROCKPTS[r.size], r.size===1?'#ffd98a':'#dcecff');
  burst(r.x,r.y,ROCKCOL[r.size],10+r.size*9,120+r.size*90);
  ring(r.x,r.y,ROCKCOL[r.size],r.r*1.5);
  sfx.boom(r.size); shake=Math.max(shake,r.size===3?6:3);
  if(r.size>1){
    for(let k=0;k<2;k++){ const c=makeRock(r.x,r.y,r.size-1); c.vx+=r.vx*.4; c.vy+=r.vy*.4; rocks.push(c); }
  } else if(Math.random()<.1&&pickups.length<2){
    const p=PICKS[(Math.random()*PICKS.length)|0];
    pickups.push({x:r.x,y:r.y,k:p.k,ch:p.ch,col:p.col,name:p.name,life:12,t:Math.random()*TAU});
  }
}
function killShip(){
  if(!ship.alive)return;
  ship.alive=false; ship.lives--; syncLives(); ship.respawn=1.6;

  shake=16; sfx.boom(3); setEngine(false);
}

  ship.warpCd=1.2; warpFlash=1; sfx.warp();
  burst(ship.x,ship.y,'#9fe8ff',18,220); ring(ship.x,ship.y,'#5fd9ff',44);
  let x=W/2,y=H/2,ok=false;
  for(let t=0;t<14&&!ok;t++){ x=Math.random()*W; y=Math.random()*H; ok=true;
    if(dist(x,y,ship.x,ship.y)<200)ok=false;
    for(const r of rocks)if(dist(x,y,r.x,r.y)<r.r+30){ok=false;break;} }
  ship.x=x; ship.y=y; ship.vx*=.4; ship.vy*=.4;
  burst(x,y,'#9fe8ff',18,220); ring(x,y,'#5fd9ff',44);
  if(!ok||Math.random()<.25)killShip();
}

}

/* ============================== UPDATE ============================== */
function update(dt){
  if(state==='paused')return;
  time+=dt;
  for(const s of stars){ s.x-=(3+s.l*8)*dt; if(s.x<-3){ s.x=W+3; s.y=Math.random()*H; } }

  if(state==='menu'){
    for(const r of rocks){ r.x=wrap(r.x+r.vx*dt,-r.r,W+r.r); r.y=wrap(r.y+r.vy*dt,-r.r,H+r.r); r.rot+=r.vr*dt; }
    updateFx(dt); return;
  }
  if(state==='over'){ updateFx(dt); return; }

  /* --- playing --- */
  fireCd-=dt; ship.warpCd-=dt; warpFlash=Math.max(0,warpFlash-dt*2.5);
  if(ship.triple>0)ship.triple-=dt; if(ship.rapid>0)ship.rapid-=dt;

  if(ship.alive){
    const turn=3.8*dt;
    if(keys.ArrowLeft||keys.KeyA)ship.a-=turn;
    if(keys.ArrowRight||keys.KeyD)ship.a+=turn;
    ship.thrust=!!(keys.ArrowUp||keys.KeyW);
    if(ship.thrust){
      ship.vx+=Math.cos(ship.a)*420*dt; ship.vy+=Math.sin(ship.a)*420*dt; setEngine(true);
      for(let i=0;i<2;i++){
        const bx=ship.x-Math.cos(ship.a)*12, by=ship.y-Math.sin(ship.a)*12;
        const a=ship.a+Math.PI+(Math.random()-.5)*.7, s=80+Math.random()*80;
        parts.push({x:bx,y:by,vx:Math.cos(a)*s,vy:Math.sin(a)*s,life:.25+Math.random()*.2,max:.45,col:Math.random()<.5?'#ffb347':'#ffe066',sz:1.5});
      }
    } else setEngine(false);
    const damp=Math.exp(-.55*dt); ship.vx*=damp; ship.vy*=damp;
    const sp=Math.hypot(ship.vx,ship.vy); if(sp>520){ ship.vx*=520/sp; ship.vy*=520/sp; }
    ship.x=wrap(ship.x+ship.vx*dt,-26,W+26); ship.y=wrap(ship.y+ship.vy*dt,-26,H+26);
    if(ship.invuln>0)ship.invuln-=dt;
    if(keys.Space&&fireCd<=0&&bullets.length<(ship.triple>0?9:5)){
      fireCd=ship.rapid>0?.1:.18;
      const offs=ship.triple>0?[-.16,0,.16]:[0];
      for(const o of offs){ const a=ship.a+o;
        bullets.push({x:ship.x+Math.cos(a)*16,y:ship.y+Math.sin(a)*16,vx:Math.cos(a)*560+ship.vx*.4,vy:Math.sin(a)*560+ship.vy*.4,life:1.05}); }
      sfx.shoot();
    }
  } else {
    ship.respawn-=dt;
    if(ship.respawn<=0){
      if(ship.lives>0){ Object.assign(ship,{x:W/2,y:H/2,a:-Math.PI/2,vx:0,vy:0,alive:true,invuln:3}); popup(W/2,H/2-46,'REDEPLOYED','#5fd9ff'); }
      else { gameOver(); return; }
    }
  }

  for(let i=bullets.length-1;i>=0;i--){ const b=bullets[i];
    b.x+=b.vx*dt; b.y+=b.vy*dt; b.life-=dt;
    if(b.life<=0||b.x<-20||b.x>W+20||b.y<-20||b.y>H+20)bullets.splice(i,1); }

  for(const r of rocks){ r.x=wrap(r.x+r.vx*dt,-r.r,W+r.r); r.y=wrap(r.y+r.vy*dt,-r.r,H+r.r); r.rot+=r.vr*dt; }

  outer:for(let i=bullets.length-1;i>=0;i--){ const b=bullets[i];
    for(let j=rocks.length-1;j>=0;j--){ const r=rocks[j];
      if(dist(b.x,b.y,r.x,r.y)<r.r+3){ bullets.splice(i,1); breakRock(j); continue outer; } } }

  for(let i=pickups.length-1;i>=0;i--){ const p=pickups[i]; p.life-=dt; p.t+=dt;
    if(p.life<=0){ pickups.splice(i,1); continue; }
    if(ship.alive&&dist(p.x,p.y,ship.x,ship.y)<30){
      pickups.splice(i,1);
      if(p.k==='shield')ship.shield=true; else ship[p.k]=8;
      popup(p.x,p.y,p.name,p.col); ring(p.x,p.y,p.col,36); sfx.pick();
    } }

  if(ship.alive&&ship.invuln<=0){
    for(let j=rocks.length-1;j>=0;j--){ const r=rocks[j];
      if(dist(r.x,r.y,ship.x,ship.y)<r.r+9){
        if(ship.shield){ ship.shield=false; popup(ship.x,ship.y,'SHIELD DOWN','#5fd9ff'); sfx.shieldDown();
          burst(r.x,r.y,ROCKCOL[r.size],20,260); ring(r.x,r.y,'#5fd9ff',r.r*1.7); rocks.splice(j,1); }
        else { killShip(); break; }
      } }
  }

  if(rocks.length===0&&pickups.length===0){
    if(waveClearT<=0){ waveClearT=1.5; const bonus=100+(wave-1)*50; score+=bonus; popup(W/2,H*.4,'SECTOR CLEAR +'+bonus,'#ffc45e'); }
    else { waveClearT-=dt; if(waveClearT<=0){ spawnWave(wave+1); bannerShow('WAVE '+wave); sfx.jingle(); } }
  }

  updateFx(dt); syncHud(); syncChips();
  shake=Math.max(0,shake-40*dt);
}

/* ============================== RENDER ============================== */
function shipPath(){ ctx.beginPath(); ctx.moveTo(18,0); ctx.lineTo(-11,10); ctx.lineTo(-6,0); ctx.lineTo(-11,-10); ctx.closePath(); }
function drawRock(r){
  ctx.save(); ctx.translate(r.x,r.y); ctx.rotate(r.rot);
  ctx.lineJoin='round'; ctx.beginPath();
  for(let i=0;i<r.n;i++){ const a=i/r.n*TAU,rad=r.r*r.verts[i];
    const px=Math.cos(a)*rad,py=Math.sin(a)*rad; i?ctx.lineTo(px,py):ctx.moveTo(px,py); }
  ctx.closePath();
  const c=ROCKCOL[r.size];
  ctx.strokeStyle=c; ctx.globalAlpha=.22; ctx.lineWidth=5; ctx.stroke();
  ctx.globalAlpha=1; ctx.lineWidth=1.7; ctx.stroke();
  ctx.restore();
}
function drawPickup(p){
  const bob=Math.sin(p.t*3)*3;
  const blink=p.life<3?(Math.floor(p.t*6)%2?.35:1):1;
  ctx.save(); ctx.translate(p.x,p.y+bob); ctx.rotate(Math.sin(p.t*1.5)*.4);
  const s=13; ctx.beginPath(); ctx.rect(-s,-s,s*2,s*2);
  ctx.strokeStyle=p.col; ctx.globalAlpha=blink*.28; ctx.lineWidth=5; ctx.stroke();
  ctx.globalAlpha=blink; ctx.lineWidth=1.6; ctx.stroke();
  ctx.fillStyle=p.col; ctx.font='700 12px '+MONO; ctx.textAlign='center'; ctx.textBaseline='middle';
  ctx.fillText(p.ch,0,1);
  ctx.restore(); ctx.textBaseline='alphabetic';
}
function drawShip(){
  if(ship.invuln>0&&Math.floor(ship.invuln*9)%2===0)return;
  ctx.save(); ctx.translate(ship.x,ship.y); ctx.rotate(ship.a);
  if(ship.thrust){ const f=8+Math.random()*10;
    ctx.fillStyle='rgba(255,179,71,.85)';
    ctx.beginPath(); ctx.moveTo(-6,4.5); ctx.lineTo(-6-f,0); ctx.lineTo(-6,-4.5); ctx.closePath(); ctx.fill();
    ctx.fillStyle='#ffe066';
    ctx.beginPath(); ctx.moveTo(-6,2.4); ctx.lineTo(-6-f*.55,0); ctx.lineTo(-6,-2.4); ctx.closePath(); ctx.fill(); }
  ctx.lineJoin='round';
  ctx.strokeStyle='rgba(95,217,255,.35)'; ctx.lineWidth=6; shipPath(); ctx.stroke();
  ctx.strokeStyle='#eafcff'; ctx.lineWidth=1.8; shipPath(); ctx.stroke();
  ctx.restore();
  if(ship.shield){
    ctx.save(); ctx.translate(ship.x,ship.y); ctx.rotate(time*1.2);
    ctx.strokeStyle='rgba(95,217,255,.8)'; ctx.lineWidth=1.6; ctx.beginPath();
    for(let i=0;i<6;i++){ const a=i/6*TAU,px=Math.cos(a)*24,py=Math.sin(a)*24; i?ctx.lineTo(px,py):ctx.moveTo(px,py); }
    ctx.closePath(); ctx.stroke(); ctx.restore();
  }
  if(ship.invuln>0){
    ctx.strokeStyle='rgba(95,217,255,.35)'; ctx.lineWidth=1.4;
    ctx.beginPath(); ctx.arc(ship.x,ship.y,20+Math.sin(time*8)*2,0,TAU); ctx.stroke();
  }
}
function render(){
  ctx.clearRect(0,0,W,H);
  const nx=Math.sin(time*.05)*10, ny=Math.cos(time*.043)*10;
  if(neb)ctx.drawImage(neb,-40+nx,-40+ny);
  for(const s of stars){
    const tw=.55+.45*Math.sin(time*(1.2+s.l)+s.ph);
    ctx.globalAlpha=[.35,.6,.95][s.l]*tw;
    ctx.fillStyle=s.l===2?'#cfe8ff':'#9fc0e8';
    ctx.fillRect(s.x,s.y,s.r,s.r);
  }
  ctx.globalAlpha=1;

  ctx.save();
  if(shake>0)ctx.translate((Math.random()*2-1)*shake,(Math.random()*2-1)*shake);

  for(const p of pickups)drawPickup(p);
  for(const r of rocks)drawRock(r);

  ctx.globalCompositeOperation='lighter';
  for(const b of bullets){
    ctx.strokeStyle='rgba(255,224,102,.5)'; ctx.lineWidth=2;
    ctx.beginPath(); ctx.moveTo(b.x-b.vx*.02,b.y-b.vy*.02); ctx.lineTo(b.x,b.y); ctx.stroke();
    ctx.fillStyle='#fff3c4'; ctx.beginPath(); ctx.arc(b.x,b.y,2.4,0,TAU); ctx.fill();
  }
  for(const p of parts){
    const a=Math.max(p.life/p.max,0);
    if(p.ring){ ctx.globalAlpha=a*.8; ctx.strokeStyle=p.col; ctx.lineWidth=2;
      ctx.beginPath(); ctx.arc(p.x,p.y,(1-a)*p.sz+4,0,TAU); ctx.stroke(); }
    else { ctx.globalAlpha=a; ctx.fillStyle=p.col; ctx.fillRect(p.x-p.sz/2,p.y-p.sz/2,p.sz,p.sz); }
  }
  ctx.globalAlpha=1; ctx.globalCompositeOperation='source-over';

  if(state==='playing'&&ship.alive)drawShip();
