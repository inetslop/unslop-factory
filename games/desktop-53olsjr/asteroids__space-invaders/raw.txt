function spawnFormation(){const rows=clamp(3+((G.wave-1)>>1),3,5);
  const cols=W<620?5:(W<980?7:8);

  F.rows=rows;F.cols=cols;F.spX=spX;F.spY=spY;
  F.fx=(W-(cols-1)*spX)/2;F.fy=-rows*spY-40;F.enterY=70*S;
  F.dir=Math.random()<.5?-1:1;F.sp=(30+G.wave*6)*S;F.step=18*S;
  F.rate=Math.min(.5+G.wave*.25,4);F.shootAcc=.6;
  aliens=[];
  for(let r=0;r<rows;r++)for(let c=0;c<cols;c++){
    const type=r===0?0:(r===rows-1?2:1);
    const hw=17*S,rat=ALIEN_SPR[type][0].height/ALIEN_SPR[type][0].width;
    aliens.push({ox:c*spX,oy:r*spY,x:0,y:0,type,row:r,col:c,hw,hh:hw*rat/2,rat,alive:true});}

function dangerY(){return H-(coarse?195:120)*S;}
function updAliens(dt){if(!aliens.length)return;
  F.animT+=dt;if(F.animT>.5){F.animT=0;F.frame^=1;}
  let aliveN=0,minx=Infinity,maxx=-Infinity;
  for(const a of aliens){if(!a.alive)continue;aliveN++;if(a.ox<minx)minx=a.ox;if(a.ox>maxx)maxx=a.ox;}
  if(aliveN===0)return;
  const mv=F.sp*(1+(1-aliveN/F.total)*1.3)*dt*F.dir,m=34*S;
  if((F.fx+maxx+mv>W-m)||(F.fx+minx+mv<m)){F.dir*=-1;F.fy+=F.step;}else F.fx+=mv;
  if(F.fy<F.enterY)F.fy=Math.min(F.enterY,F.fy+(80+F.sp)*dt);
  for(const a of aliens){a.x=F.fx+a.ox;a.y=F.fy+a.oy;}

    while(F.shootAcc>=1&&ab.length<14){F.shootAcc--;
      const pool=aliens.filter(a=>a.alive);if(!pool.length)break;
      const a=pool[irnd(pool.length)];ab.push({x:a.x,y:a.y+a.hh,vx:0,vy:(230+G.wave*8)*S});SFX.ashot();}}

const RR=[0,15,27,46],RS=[0,90,58,32];
function makeRock(size,x,y,vx,vy){const n=8+irnd(5),pts=[];
  for(let i=0;i<n;i++){const r=RR[size]*S*rnd(.72,1.28),a=i/n*TAU+rnd(-.2,.2);
    pts.push([Math.cos(a)*r,Math.sin(a)*r]);}
  return{x,y,vx,vy,size,rot:rnd(TAU),vr:(Math.random()-.5)*(size===3?.7:size===2?1.3:2.4),pts};}
function spawnEdgeRock(){let size=3;if(G.wave>=4&&Math.random()<.3)size=2;
  const side=irnd(4),m=60;let x,y;
  if(side===0){x=rnd(W);y=-m;}else if(side===1){x=W+m;y=rnd(H);}
  else if(side===2){x=rnd(W);y=H+m;}else{x=-m;y=rnd(H);}
  const ang=Math.atan2(H/2-y,W/2-x)+rnd(-.5,.5),sp=RS[size]*S*rnd(.7,1.3);
  rocks.push(makeRock(size,x,y,Math.cos(ang)*sp,Math.sin(ang)*sp));}

  rings.push({x:p.x,y:p.y,r:6,vr:220*S,life:.4,max:.4,col:'#ffd166',lw:2});
  if(p.type==='twin'){ship.twin=10;grant('TWIN','#5ff3ff');}

function collide(){
  for(let i=bullets.length-1;i>=0;i--){const b=bullets[i];let hit=false;
    for(let j=rocks.length-1;j>=0;j--){const r=rocks[j],R=RR[r.size]*S*1.05;
      if((b.x-r.x)**2+(b.y-r.y)**2<R*R){hit=true;killRock(j);break;}}
    if(!hit)for(const a of aliens){if(!a.alive)continue;
      if(Math.abs(b.x-a.x)<a.hw+3&&Math.abs(b.y-a.y)<a.hh+3){a.alive=false;killAlien(a);hit=true;break;}}
    if(hit)bullets.splice(i,1);}
  if(ship.alive&&ship.inv<=0){
    for(const r of rocks){const R=RR[r.size]*S*.9+10*S;
      if((ship.x-r.x)**2+(ship.y-r.y)**2<R*R){shipHit('rock');break;}}
    if(!ship.alive||G.state!=='playing')return;
    for(let i=ab.length-1;i>=0;i--){const b=ab[i];
      if((ship.x-b.x)**2+(ship.y-b.y)**2<(14*S)*(14*S)){ab.splice(i,1);shipHit('shot');break;}}
    if(!ship.alive||G.state!=='playing')return;
    for(const a of aliens){if(!a.alive)continue;
      if(Math.abs(ship.x-a.x)<a.hw+8*S&&Math.abs(ship.y-a.y)<a.hh+8*S){
        a.alive=false;killAlien(a);shipHit('ram');break;}}}}

function update(dt){G.t+=dt;
  G.shake*=Math.exp(-4.5*dt);if(G.shake<.1)G.shake=0;
  G.flash=Math.max(0,G.flash-2.2*dt);
  updStars(dt);
  if(G.state==='playing'){
    if(G.slow>0){G.slow-=dt;dt*=.35;}
    updShip(dt);if(K.f)tryFire();
    for(let i=bullets.length-1;i>=0;i--){const b=bullets[i];b.x+=b.vx*dt;b.y+=b.vy*dt;
      if(b.x<-20||b.x>W+20||b.y<-20||b.y>H+20)bullets.splice(i,1);}
    G.rockT-=dt;const target=Math.min(3+G.wave,9);
    if(rocks.length<target&&G.rockT<=0){spawnEdgeRock();if(G.wave>2&&rocks.length<target)spawnEdgeRock();G.rockT=rnd(.7,1.6);}
    updRocks(dt);updAliens(dt);
    for(let i=ab.length-1;i>=0;i--){const b=ab[i];b.x+=b.vx*dt;b.y+=b.vy*dt;
      if(b.y>H+20||b.x<-20||b.x>W+20)ab.splice(i,1);}
    updPows(dt);collide();
    if(ship.alive&&ship.inv<=0)for(let i=pows.length-1;i>=0;i--){const p=pows[i];
      if((ship.x-p.x)**2+(ship.y-p.y)**2<(30*S)*(30*S)){applyPow(p);pows.splice(i,1);}}
    if(ship.twin>0){ship.twin-=dt;if(chips.twin)chips.twin.bar.style.transform='scaleX('+clamp(ship.twin/10,0,1)+')';if(ship.twin<=0){ship.twin=0;delChip('twin');}}
    if(ship.rapid>0){ship.rapid-=dt;if(chips.rapid)chips.rapid.bar.style.transform='scaleX('+clamp(ship.rapid/8,0,1)+')';if(ship.rapid<=0){ship.rapid=0;delChip('rapid');}}
    if(!aliens.length&&G.pendingNext===0){G.pendingNext=1.5;const bns=50*G.wave+25;
      addScore(bns,W/2,H*.34);banner('SECTOR CLEARED','+'+bns+' BONUS');SFX.clear();}
    if(G.pendingNext>0){G.pendingNext-=dt;if(G.pendingNext<=0)nextWave();}
  }else{
    updRocks(dt);
    for(let i=ab.length-1;i>=0;i--){const b=ab[i];b.y+=b.vy*dt;if(b.y>H+20)ab.splice(i,1);}
    if(G.state==='menu'&&rocks.length<6&&Math.random()<dt*.7)spawnEdgeRock();
    if(G.state==='over'){updAliens(dt);if(G.overT>0){G.overT-=dt;if(G.overT<=0)$('over').hidden=false;}}
  }
  updParts(dt);updRings(dt);hudSync();}
