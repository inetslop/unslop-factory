```js
function collideCircleWalls(c){ // c {x,y,r, vx,vy, mass}
 let resolved=0;
 for(let j=floor(y-r); j<=floor(y+r); j++) for(let i=floor(x-r); i<=floor(x+r); i++){
   if(!isWall(i,j)) continue;
   const cx = clamp(x, i, i+1), cy = clamp(y, j, j+1);
   let dx=x-cx, dy=y-cy, d2=dx*dx+dy*dy;
   if(d2 < r*r){
     let d=sqrt(d2), nx,ny,pen;
     if(d>1e-6){nx=dx/d; ny=dy/d; pen=r-d;} else { // center inside cell: push out along min axis
       const l=x-i, rr=i+1-x, t=y-j, b=j+1-y; const m=Math.min(l,rr,t,b);
       if(m===l){nx=-1;ny=0;pen=l+r;} else if(m===rr){nx=1;ny=0;pen=rr+r;} else if(m===t){nx=0;ny=-1;pen=t+r;} else {nx=0;ny=1;pen=b+r;}
     }
     c.x+=nx*pen; c.y+=ny*pen;
     const vn=c.vx*nx+c.vy*ny;
     if(vn<0){ const jimp = -(1+REST_PLAYER_WALL)*vn; c.vx-=jimp*nx; c.vy-=jimp*ny; impact(nx,ny,jimp,c); }
     resolved++;
   }
 }
}
```

```html
function physStep(dt){
  const p=G.player; if(!p)return;
  let ix=0,iy=0;
  if(keys["arrowleft"]||keys["a"])ix--; if(keys["arrowright"]||keys["d"])ix++;
  if(keys["arrowup"]||keys["w"])iy--;   if(keys["arrowdown"]||keys["s"])iy++;
  if(pointer.active){ const dx=pointer.x-p.x,dy=pointer.y-p.y,d=Math.hypot(dx,dy); if(d>.25){ix+=dx/d;iy+=dy/d;} }
  const il=Math.hypot(ix,iy); if(il>1){ix/=il;iy/=il;}
  p.vx+=ix*P_ACC*dt; p.vy+=iy*P_ACC*dt;
  const pd=Math.exp(-P_DAMP*dt); p.vx*=pd; p.vy*=pd;
  const sp=Math.hypot(p.vx,p.vy); if(sp>P_MAX){p.vx*=P_MAX/sp;p.vy*=P_MAX/sp;}
  p.x+=p.vx*dt; p.y+=p.vy*dt;
  const bd=Math.exp(-B_DAMP*dt);
  for(const b of G.boxes){ b.vx*=bd; b.vy*=bd; b.x+=b.vx*dt; b.y+=b.vy*dt; }

  for(const b of G.boxes)collideBoxWalls(b);
  for(let it=0;it<2;it++)for(let i=0;i<G.boxes.length;i++)for(let j=i+1;j<G.boxes.length;j++)collideBoxes(G.boxes[i],G.boxes[j]);
  collideCircleWalls(p);
  for(const b of G.boxes)collidePlayerBox(p,b);
}

function collideCircleWalls(c){
  const r=c.r;
  for(let j=Math.floor(c.y-r);j<=Math.floor(c.y+r);j++)for(let i=Math.floor(c.x-r);i<=Math.floor(c.x+r);i++){
    if(!isWall(i,j))continue;
    const cx=clamp(c.x,i,i+1), cy=clamp(c.y,j,j+1);
    let dx=c.x-cx, dy=c.y-cy, d2=dx*dx+dy*dy;
    if(d2>=r*r)continue;
    let nx,ny,pen,d=Math.sqrt(d2);
    if(d>1e-7){nx=dx/d;ny=dy/d;pen=r-d;}
    else{const l=c.x-i,rr=i+1-c.x,t=c.y-j,b=j+1-c.y,m=Math.min(l,rr,t,b);
      if(m===l){nx=-1;ny=0;pen=l+r;}else if(m===rr){nx=1;ny=0;pen=rr+r;}else if(m===t){nx=0;ny=-1;pen=t+r;}else{nx=0;ny=1;pen=b+r;}}
    c.x+=nx*pen; c.y+=ny*pen;
    const vn=c.vx*nx+c.vy*ny;
    if(vn<0){const jimp=-(1+E_PW)*vn; c.vx-=jimp*nx; c.vy-=jimp*ny;}
  }
}

function collideBoxWalls(b){
  let best=null;
  const x0=Math.floor(b.x-b.half),x1=Math.floor(b.x+b.half),y0=Math.floor(b.y-b.half),y1=Math.floor(b.y+b.half);
  for(let j=y0;j<=y1;j++)for(let i=x0;i<=x1;i++){
    if(!isWall(i,j))continue;
    const ox=Math.min(b.x+b.half,i+1)-Math.max(b.x-b.half,i);
    const oy=Math.min(b.y+b.half,j+1)-Math.max(b.y-b.half,j);
    if(ox<=0||oy<=0)continue;
    let nx,ny,pen;
    if(ox<oy){nx=b.x<i+.5?-1:1;pen=ox;}else{ny=b.y<j+.5?-1:1;pen=oy;}
    if(!best||pen<best.pen)best={nx,ny,pen};
  }
  if(best){
    b.x+=best.nx*best.pen; b.y+=best.ny*best.pen;
    const vn=b.vx*best.nx+b.vy*best.ny;
    if(vn<0){const jimp=-(1+E_BW)*vn; b.vx-=jimp*best.nx; b.vy-=jimp*best.ny; onImpact(b,best,jimp);}
  }
}

function collideBoxes(a,b){
  const dx=b.x-a.x,dy=b.y-a.y;
  const ox=(a.half+b.half)-Math.abs(dx),oy=(a.half+b.half)-Math.abs(dy);
  if(ox<=0||oy<=0)return;
  let nx,ny,pen;
  if(ox<oy){nx=dx<0?-1:1;pen=ox;}else{ny=dy<0?-1:1;pen=oy;}
  a.x-=nx*pen*.5; a.y-=ny*pen*.5; b.x+=nx*pen*.5; b.y+=ny*pen*.5;
  const vn=(b.vx-a.vx)*nx+(b.vy-a.vy)*ny;
  if(vn<0){
    const jimp=-(1+E_BB)*vn/(2/M_BOX);
    a.vx-=jimp*nx/M_BOX; a.vy-=jimp*ny/M_BOX; b.vx+=jimp*nx/M_BOX; b.vy+=jimp*ny/M_BOX;
    if(jimp>.45){ AU.clack(Math.min(1,jimp/3)); G.shake=Math.min(16,G.shake+Math.min(5,jimp*.8)); }
  }
}

function collidePlayerBox(p,b){
  const cx=clamp(p.x,b.x-b.half,b.x+b.half), cy=clamp(p.y,b.y-b.half,b.y+b.half);
  let dx=p.x-cx,dy=p.y-cy,d2=dx*dx+dy*dy;
  if(d2>=p.r*p.r){ b.contact=false; b.pushPrev=false; return; }
  let nx,ny,pen,d=Math.sqrt(d2);
  if(d>1e-7){nx=dx/d;ny=dy/d;pen=p.r-d;}
  else{const l=p.x-(b.x-b.half),r=(b.x+b.half)-p.x,t=p.y-(b.y-b.half),bo=(b.y+b.half)-p.y,m=Math.min(l,r,t,bo);
    if(m===l){nx=-1;ny=0;pen=l+p.r;}else if(m===r){nx=1;ny=0;pen=r+p.r;}else if(m===t){nx=0;ny=-1;pen=t+p.r;}else{nx=0;ny=1;pen=bo+p.r;}}
  const wP=M_BOX/(M_PLAYER+M_BOX),wB=M_PLAYER/(M_PLAYER+M_BOX);
  p.x+=nx*pen*wP; p.y+=ny*pen*wP; b.x-=nx*pen*wB; b.y-=ny*pen*wB;
  const vn=(p.vx-b.vx)*nx+(p.vy-b.vy)*ny;
  if(vn<0){const jimp=-(1+E_PB)*vn/(1/M_PLAYER+1/M_BOX);
    p.vx+=jimp*nx/M_PLAYER; p.vy+=jimp*ny/M_PLAYER; b.vx-=jimp*nx/M_BOX; b.vy-=jimp*ny/M_BOX;}
  if(!b.pushPrev&&(vn<0||Math.hypot(p.vx,p.vy)>.4)){G.pushes++;}
  b.pushPrev=true; b.contact=true;
}

function onImpact(b,n,j){
  if(j<.5)return;
  G.shake=Math.min(16,G.shake+Math.min(9,j*1.2));
  const now=performance.now();
  if(now-lastThud>70){lastThud=now; AU.thud(Math.min(.8,j/4));}
  const px=b.x-n.x*b.half, py=b.y-n.y*b.half;
  dustBurst(px,py,n.x,n.y,Math.min(1,j/5));
}

/* snap slow boxes to nearest legal cell center */
function boxLegal(cx,cy){
  for(let j=Math.floor(cy-B_HALF);j<=Math.floor(cy+B_HALF);j++)for(let i=Math.floor(cx-B_HALF);i<=Math.floor(cx+B_HALF);i++)
    if(isWall(i,j))return false;
  return true;
}
function snapPass(dt){
  for(const b of G.boxes){
    const sp=Math.hypot(b.vx,b.vy);
    if(sp>=.3||b.contact)continue;
    const ic=Math.round(b.x-.5), jc=Math.round(b.y-.5);
    let best=null,bd=1e9;
    for(let dj=-1;dj<=1;dj++)for(let di=-1;di<=1;di++){
      if(Math.abs(di)+Math.abs(dj)>1)continue;
      const cx=ic+di+.5, cy=jc+dj+.5;
      if(!boxLegal(cx,cy))continue;
      const d=Math.hypot(cx-b.x,cy-b.y);
      if(d<bd){bd=d;best={cx,cy};}
    }
    if(best&&bd<.38){
      const k=Math.min(1,26*dt);
      b.x+=(best.cx-b.x)*k; b.y+=(best.cy-b.y)*k; b.vx*=.5; b.vy*=.5;
      if(bd<.02&&sp<.12){b.x=best.cx;b.y=best.cy;b.vx=0;b.vy=0;}
    }
  }
}

function winCheck(dt){
  if(G.won)return;
  const all=G.boxes.length>0&&G.boxes.every(b=>{
    const sp=Math.hypot(b.vx,b.vy); if(sp>.25||b.contact)return false;
    return G.targets.some(t=>Math.abs(b.x-t.i-.5)<.45&&Math.abs(b.y-t.j-.5)<.45);
  });
  if(all){G.won=true;G.winT=.9;}
}

const parts=[];
function spawnText(str,x,y,size,colr,life){
  parts.push({type:"text",str,x,y,vx:0,vy:-.55,rot:0,vr:0,size,life,max:life,colr});
}
function dustBurst(x,y,nx,ny,pow){
  for(let i=0;i<7;i++)parts.push({type:"dust",x:x+rand(-.1,.1),y:y+rand(-.1,.1),
    vx:nx*rand(.5,2)*pow+rand(-.6,.6),vy:ny*rand(.5,2)*pow+rand(-.9,-.2),rot:0,vr:0,size:rand(.05,.13),life:rand(.3,.55),max:.55});
}
function sparkleBurst(x,y){
  for(let i=0;i<14;i++){const a=rand(0,TAU),s=rand(.8,2.6);
    parts.push({type:"spark",x,y,vx:Math.cos(a)*s,vy:Math.sin(a)*s-.6,rot:0,vr:0,size:rand(.03,.07),life:rand(.4,.8),max:.8});}
}
const CONF=["#ffb02e","#2fe6c0","#f4ead8","#ff5d4d","#dfe9f0"];
function confettiRain(){
  for(let i=0;i<90;i++)parts.push({type:"conf",x:rand(0,G.W),y:rand(-3,0),vx:rand(-1.2,1.2),vy:rand(1.5,3.5),
    rot:rand(0,TAU),vr:rand(-6,6),size:rand(.09,.18),life:rand(1.8,3),max:3,colr:CONF[i%CONF.length]});
}
function updateParticles(dt){
  for(let i=parts.length-1;i>=0;i--){const p=parts[i];
    p.life-=dt; if(p.life<=0){parts.splice(i,1);continue;}
    if(p.type==="conf"){p.vy+=5*dt;}
    else if(p.type!=="text")p.vx*=Math.exp(-3*dt);
    p.x+=p.vx*dt; p.y+=p.vy*dt; p.rot+=p.vr*dt;
  }
}
const motes=[];
for(let i=0;i<38;i++)motes.push({x:Math.random(),y:Math.random(),r:rand(.6,2),s:rand(4,14),ph:rand(0,9)});
function updateMotes(dt){for(const m of motes){m.y-=m.s*dt/H;m.x+=Math.sin(G.time*.5+m.ph)*.02*dt; if(m.y<-.02){m.y=1.02;m.x=Math.random();}}}

function rr(x,y,w,h,r){r=Math.min(r,w/2,h/2);ctx.beginPath();
  ctx.moveTo(x+r,y);ctx.arcTo(x+w,y,x+w,y+h,r);ctx.arcTo(x+w,y+h,x,y+h,r);ctx.arcTo(x,y+h,x,y,r);ctx.arcTo(x,y,x+w,y,r);ctx.closePath();}

function render(){
  ctx.setTransform(dpr,0,0,dpr,0,0);
  /* bg */
  ctx.fillStyle="#0a1520"; ctx.fillRect(0,0,W,H);
  let g=ctx.createRadialGradient(W*.18,H*.1,0,W*.18,H*.1,Math.max(W,H)*.7);
  g.addColorStop(0,"rgba(47,230,192,.06)");g.addColorStop(1,"rgba(0,0,0,0)");ctx.fillStyle=g;ctx.fillRect(0,0,W,H);
  g=ctx.createRadialGradient(W*.85,H*.95,0,W*.85,H*.95,Math.max(W,H)*.6);
  g.addColorStop(0,"rgba(255,176,46,.05)");g.addColorStop(1,"rgba(0,0,0,0)");ctx.fillStyle=g;ctx.fillRect(0,0,W,H);
  for(const m of motes){ctx.globalAlpha=.05+.09*(.5+.5*Math.sin(G.time+m.ph));ctx.fillStyle="#cfe6f2";
    ctx.beginPath();ctx.arc(m.x*W,m.y*H,m.r,0,TAU);ctx.fill();}
  ctx.globalAlpha=1;

  const shx=(Math.random()*2-1)*G.shake, shy=(Math.random()*2-1)*G.shake;
  ctx.save(); ctx.translate(view.ox+shx,view.oy+shy); ctx.scale(view.cell,view.cell);

  drawFloor(); drawTargets(); drawWalls(); drawBoxes(); drawPlayer(); drawParticlesWorld();
  ctx.restore();
  if(vign)ctx.drawImage(vign,0,0,W,H);
}

function drawFloor(){
  for(let j=0;j<G.H;j++)for(let i=0;i<G.W;i++){
    if(isWall(i,j))continue;
    const hh=h2(i,j);
    ctx.fillStyle=(i+j)%2?"#c3ced6":"#cbd5dc";
    ctx.fillRect(i,j,1,1);
    ctx.strokeStyle="rgba(20,40,55,.07)";ctx.lineWidth=.02;ctx.strokeRect(i+.01,j+.01,.98,.98);
    if(hh%5===0){ctx.fillStyle="rgba(30,50,65,.16)";
      ctx.beginPath();ctx.arc(i+((hh>>4)%7)/8+.15,j+((hh>>8)%7)/8+.15,.025,0,TAU);ctx.fill();}
    if(!isWall(i,j-1)){ctx.fillStyle="rgba(10,22,32,.14)";ctx.fillRect(i,j,1,.1);}
  }
}

function drawTargets(){
  for(const t of G.targets){
    const x=t.i+.5,y=t.j+.5, pulse=.5+.5*Math.sin(G.time*2.6+(t.i+t.j)*1.7);
    const teal=t.g<.5, r=teal?47:255, gg=teal?230:190, b=teal?192:77;
    let gr=ctx.createRadialGradient(x,y,.05,x,y,.85);
    gr.addColorStop(0,`rgba(${r},${gg},${b},${.2+.14*pulse})`);gr.addColorStop(1,"rgba(0,0,0,0)");
    ctx.fillStyle=gr;ctx.beginPath();ctx.arc(x,y,.85,0,TAU);ctx.fill();
    rr(x-.34,y-.34,.68,.68,.12);

    const s=.1+.02*pulse;ctx.fillStyle=`rgba(${r},${gg},${b},.85)`;ctx.fillRect(-s/2,-s/2,s,s);ctx.restore();
  }
}

function drawPlayer(){

  const r=p.r*(1+Math.sin(G.time*3.2)*.02);

  bg.addColorStop(0,"#ff9a78");bg.addColorStop(1,"#dd4526");

}

let last=performance.now();
function frame(now){
  requestAnimationFrame(frame);
  let dt=(now-last)/1000; last=now; if(dt>.1)dt=.1;
  G.time+=dt;

  if(G.state==="playing"){
    const SUB=6;
    for(let i=0;i<SUB;i++)physStep(dt/SUB);
    snapPass(dt); padFills(dt); winCheck(dt);
    if(!G.won&&G.started)G.tLvl+=dt;
    /* slide dust */
    for(const b of G.boxes){const sp=Math.hypot(b.vx,b.vy);
      if(sp>1.6){b.dustT-=dt;if(b.dustT<=0){b.dustT=.05;
        parts.push({type:"dust",x:b.x+rand(-.3,.3),y:b.y+.4,vx:-b.vx*.15,vy:rand(-.4,-.1),rot:0,vr:0,size:rand(.04,.09),life:.35,max:.35});}}
      else b.dustT=0; }
    if(G.won){G.winT-=dt;if(G.winT<=0)onWin();}
  }else{ padFills(dt); }

  /* slide sound */
  if(AU.slideG){let g=0;
    for(const b of G.boxes)g=Math.max(g,Math.hypot(b.vx,b.vy)/7);
    const p=G.player; if(p&&Math.hypot(p.vx,p.vy)>1)g=Math.max(g,.25);
    AU.slideG.gain.value+= (g*.3 - AU.slideG.gain.value)*Math.min(1,dt*10); }

  updateParticles(dt); updateMotes(dt);
  G.shake*=Math.exp(-5*dt); if(G.shake<.05)G.shake=0;
  render();

  if(G.state==="playing"||G.state==="won"){
    el.timeEl.textContent=fmt(G.tLvl); el.pushEl.textContent=G.pushes;
  }
}
```
