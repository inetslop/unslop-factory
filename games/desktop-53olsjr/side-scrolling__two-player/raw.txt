/* ============================== PHYSICS ============================== */
function boxHit(p,g){return p.x+HW>g.x&&p.x-HW<g.x+g.w&&p.y+HH>g.y&&p.y-HH<g.y+g.h}
function land(p,surfY,imp){
  if(!p.onGround){
    if(imp>420){puff(p.x,surfY,Math.min(10,3+(imp/160|0)),'#d9a86a');if(imp>900)shake=Math.max(shake,5);}
    p.sq=.7;
  }
  p.y=surfY-HH;p.vy=0;p.onGround=true;p.coyote=COYOTE;p.airJumps=1;
}
function moveCollide(p,dt){
  p.x+=p.vx*dt;
  if(p.x<HW+6){p.x=HW+6;if(p.vx<0)p.vx=0}
  if(p.x>WORLD_W-HW-6){p.x=WORLD_W-HW-6;if(p.vx>0)p.vx=0}
  for(const g of grounds)if(boxHit(p,g)){if(p.vx>0)p.x=g.x-HW;else if(p.vx<0)p.x=g.x+g.w+HW;p.vx=0;}
  const prevB=p.y+HH,pImp=p.vy;
  p.y+=p.vy*dt;p.onGround=false;
  for(const g of grounds)if(boxHit(p,g)){
    if(p.vy>0&&prevB<=g.y+1)land(p,g.y,pImp);
    else if(p.vy<0){p.y=g.y+g.h+HH;p.vy=0;}
  }
  if(p.vy>=0)for(const pl of plats){
    if(p.x+HW>pl.x&&p.x-HW<pl.x+pl.w&&prevB<=pl.y+2&&p.y+HH>=pl.y){land(p,pl.y,pImp);break;}
  }
}
function applyHit(a,v){
  a.hitDone=true;
  const sup=a.super;a.super=false;
  let kx=sup?980:560,ky=sup?-720:-430,st=sup?.95:.38;
  if(v.stun>0){kx*=1.35;ky*=.6;}
  v.vx=a.facing*kx;v.vy=ky;v.stun=Math.max(v.stun,st);v.onGround=false;v.sq=.72;
  a.stats.bonks++;
  hitstop=sup?.09:.055;shake=Math.max(shake,sup?18:9);
  const mx=(a.x+v.x)/2,my=(a.y+v.y)/2;
  burst(mx,my,sup?26:14,['#ffd23f','#fff',a.color]);
  floats.push({x:mx,y:my-40,txt:sup?'SUPER BONK!':'BONK!',col:a.color,t:0,max:.9,size:sup?34:26});
  if(sup)sSuper();else sBonk();
}
function updPlayer(p,o,dt){
  p.bonkCd=Math.max(0,p.bonkCd-dt);
  if(p.bonkT>0)p.bonkT-=dt;
  if(p.stun>0)p.stun-=dt;
  const canAct=state==='play'&&!p.dead&&p.stun<=0;
  let dir=0;
  if(canAct){if(keys[p.keys.L])dir--;if(keys[p.keys.R])dir++;}
  if(!p.dead)p.facing=o.x>=p.x?1:-1;
  const rate=p.onGround?13:(Math.abs(p.vx)>MAXV+60?1.7:5);
  p.vx+=(dir*MAXV-p.vx)*(1-Math.exp(-dt*rate));
  p.vy+=G*dt;if(p.vy>1500)p.vy=1500;
  if(canAct&&p.jumpBuf>0){
    if(p.onGround||p.coyote>0){p.vy=JUMP;p.airJumps=1;p.jumpBuf=0;p.coyote=0;p.sq=1.28;sJump();puff(p.x,p.y+HH,5,'#d9a86a');}
    else if(p.airJumps>0){p.vy=DJUMP;p.airJumps--;p.jumpBuf=0;p.sq=1.3;sDbl();ringFx(p.x,p.y+HH,p.color);}
  }
  p.jumpBuf=Math.max(0,p.jumpBuf-dt);
  if(!p.dead){
    moveCollide(p,dt);
    if(p.onGround)p.coyote=COYOTE;else p.coyote-=dt;
    p.dT-=dt;
    if(p.onGround&&Math.abs(p.vx)>240&&p.dT<=0){puff(p.x-p.facing*14,p.y+HH,1,'#d9a86a');p.dT=.09;}
  }else{
    p.x+=p.vx*dt;p.y+=p.vy*dt;p.spin+=dt*11;
  }
  // squash spring toward posture
  let target=1;
  if(!p.onGround&&!p.dead)target=p.vy<-100?1.16:p.vy>300?1.05:.97;
  p.sq+=(target-p.sq)*Math.min(1,dt*12);
  // super sparkles
  if(p.super){p.sparkT-=dt;if(p.sparkT<=0){p.sparkT=.08;parts.push({type:'spark',x:p.x+(Math.random()-.5)*40,y:p.y+(Math.random()-.5)*46,vx:(Math.random()-.5)*60,vy:-60-Math.random()*60,life:0,max:.4,size:3,col:'#ffd23f'});}}
  if(!p.dead&&p.y+HH>VOID_Y)p.dead=true;
}
function separate(a,b){
  if(a.dead||b.dead)return;
  const dx=b.x-a.x;
  if(Math.abs(dx)<HW*2&&Math.abs(b.y-a.y)<HH*2){
    const sgn=Math.sign(dx)||1,ox=(HW*2-Math.abs(dx))/2;
    a.x-=sgn*ox;b.x+=sgn*ox;
    a.vx-=sgn*90;b.vx+=sgn*90;
    puff((a.x+b.x)/2,a.y+HH,2,'#d9a86a');
    if(tGlobal-lastBumpT>.3){lastBumpT=tGlobal;sBump();}
  }
}
function bonkCheck(a,v){
  if(a.bonkT<=0||a.hitDone)return;
  const cx=a.x+a.facing*44,cy=a.y-2;
  if(Math.abs(cx-v.x)<26+HW&&Math.abs(cy-v.y)<34+HH)applyHit(a,v);
}

/* ============================== FX ============================== */
function puff(x,y,n,col){for(let i=0;i<n;i++)parts.push({type:'dust',x:x+(Math.random()-.5)*20,y,vx:(Math.random()-.5)*170,vy:-20-Math.random()*70,life:0,max:.35+Math.random()*.3,size:3.5+Math.random()*5,col});}
function burst(x,y,n,cols){for(let i=0;i<n;i++){const a=Math.random()*6.283,sp=140+Math.random()*300;parts.push({type:'spark',x,y,vx:Math.cos(a)*sp,vy:Math.sin(a)*sp-80,life:0,max:.3+Math.random()*.3,size:2.5+Math.random()*4,col:cols[i%cols.length]});}}
function ringFx(x,y,col){parts.push({type:'ring',x,y,r:12,vr:300,life:0,max:.35,col});}
function confettiBurst(x,y,n,cols){for(let i=0;i<n;i++)parts.push({type:'conf',x:x+(Math.random()-.5)*80,y:y+(Math.random()-.5)*40,vx:(Math.random()-.5)*520,vy:-180-Math.random()*380,rot:Math.random()*6.28,vr:(Math.random()-.5)*14,life:0,max:1.6+Math.random()*1,col:cols[i%cols.length]});}
function updParts(dt){
  for(let i=parts.length-1;i>=0;i--){const p=parts[i];p.life+=dt;
    if(p.life>=p.max){parts.splice(i,1);continue;}
    if(p.type==='dust'){p.vy+=260*dt;p.x+=p.vx*dt;p.y+=p.vy*dt;}
    else if(p.type==='spark'){p.vx*=Math.exp(-3*dt);p.vy+=900*dt;p.x+=p.vx*dt;p.y+=p.vy*dt;}
    else if(p.type==='ring'){p.r+=p.vr*dt;}
    else if(p.type==='conf'){p.vy+=700*dt;p.vx*=Math.exp(-.8*dt);p.x+=p.vx*dt;p.y+=p.vy*dt;p.rot+=p.vr*dt;}
  }
  if(parts.length>420)parts.splice(0,parts.length-420);
  for(let i=floats.length-1;i>=0;i--){const f=floats[i];f.t+=dt;f.y-=55*dt;if(f.t>f.max)floats.splice(i,1);}
}
function updTumbles(dt){
  tumbleT-=dt;
  if(tumbleT<=0){tumbleT=3+Math.random()*5;
    const left=Math.random()<.5,edge=(left?cam.x-cw/cam.s/2:cam.x+cw/cam.s/2)+90;
    if(edge>40&&edge<WORLD_W-40)tumbles.push({x:edge,y:380,vx:(left?1:-1)*(110+Math.random()*120),vy:0,r:15+Math.random()*7});
  }
  for(let i=tumbles.length-1;i>=0;i--){const t=tumbles[i];
    t.vy+=G*.8*dt;t.x+=t.vx*dt;t.y+=t.vy*dt;
    for(const g of grounds)if(t.x>g.x&&t.x<g.x+g.w&&t.y+t.r>=g.y&&t.y-t.r<g.y+40){t.y=g.y-t.r;t.vy=0;}
    if(t.y>VOID_Y||t.x<-150||t.x>WORLD_W+150)tumbles.splice(i,1);
  }
}

/* ============================== UPDATE ============================== */
function bannerSet(txt,dur,color){banner={txt,t:dur,dur,color};}
function updCam(dt){
  let ts,tx;
  if(state==='menu'){ts=s0*.96;const vw=cw/ts;tx=clamp(WORLD_W/2+Math.sin(tGlobal*.14)*780,vw/2,WORLD_W-vw/2);}
  else{
    const mid=(P1.x+P2.x)/2,need=Math.abs(P1.x-P2.x)+560;
    ts=clamp(Math.min(s0,(cw*.8)/need),s0*.38,s0);
    const vw=cw/ts;tx=vw>=WORLD_W?WORLD_W/2:clamp(mid,vw/2,WORLD_W-vw/2);
  }
  cam.s+=(ts-cam.s)*Math.min(1,dt*3.5);
  const vw2=cw/cam.s;
  tx=vw2>=WORLD_W?WORLD_W/2:clamp(tx,vw2/2,WORLD_W-vw2/2);
  cam.x+=(tx-cam.x)*Math.min(1,dt*7);
  let ty=438;
  if(state!=='menu'){const my=(P1.y+P2.y)/2;ty=436+clamp(my-560,-90,150)*.35;}
  cam.y+=(ty-cam.y)*Math.min(1,dt*4);
}
function update(dt,rdt){
  tGlobal+=dt;
  if(slowT>0)slowT-=rdt;else timeScale+=(1-timeScale)*Math.min(1,rdt*6);
  const prevCamX=cam.x;
  updCam(dt);
  camDX=(cam.x-prevCamX)*cam.s;
  for(const c of clouds){c.x-=(c.v*dt+camDX*.08);if(c.x<-260){c.x=cw+240;c.y=30+Math.random()*260;}}
  for(const b of birds){b.x-=(b.vx*dt+camDX*.16);if(b.x<-60)b.x=cw+60;}
  updTumbles(dt);updParts(dt);

  if(state==='count'){
    cdT-=dt;
    const n=cdT>1.6?3:cdT>.8?2:1;
    if(n!==lastNum){lastNum=n;bannerSet(String(n),.5,'#fff');sTick();}
    updPlayer(P1,P2,dt);updPlayer(P2,P1,dt);separate(P1,P2);
    if(cdT<=0){state='play';bannerSet('BONK!',.55,'#ffd23f');sGo();}
  }
  else if(state==='play'){
    updPlayer(P1,P2,dt);updPlayer(P2,P1,dt);separate(P1,P2);
    bonkCheck(P1,P2);bonkCheck(P2,P1);
    // bolt spawner
    boltT-=dt;
    if(!bolt&&boltT<=0){
      if(Math.random()<.6){const pl=plats[(Math.random()*plats.length)|0];bolt={x:pl.x+pl.w/2,y:pl.y-48,t:0};}
      else{const g=grounds[(Math.random()*grounds.length)|0];bolt={x:g.x+90+Math.random()*(g.w-180),y:GROUND_TOP-48,t:0};}
    }
    if(bolt){
      bolt.t+=dt;if(bolt.t>9)bolt=null;
      for(const p of [P1,P2]){
        if(!p.dead&&Math.hypot(p.x-bolt.x,p.y-(bolt.y+Math.sin(tGlobal*3)*6))<42){
          p.super=true;p.stats.bolts++;sBolt();ringFx(bolt.x,bolt.y,'#ffd23f');
          floats.push({x:bolt.x,y:bolt.y-24,txt:'SUPER CHARGED!',col:'#ffd23f',t:0,max:.9,size:22});
          bolt=null;boltT=6+Math.random()*4;break;
        }
      }
    }
    if(P1.dead||P2.dead)enterKO();
  }
  else if(state==='ko'){
    koT-=rdt;
    updPlayer(P1,P2,dt);updPlayer(P2,P1,dt);
    if(koT<=0){
      if(score[0]>=3||score[1]>=3)enterWin(score[0]>=3?P1:P2);
      else{resetRound();state='count';cdT=2.4;lastNum=99;}
    }
  }
  else if(state==='win'){
    updPlayer(P1,P2,dt);updPlayer(P2,P1,dt);
    if(winner){
      if(winner.onGround&&Math.sin(tGlobal*4.2)>.86){winner.vy=-520;puff(winner.x,winner.y+HH,4,'#ffd98a');}
      if(Math.random()<rdt*7)confettiBurst(cam.x+(Math.random()-.5)*cw/cam.s,cam.y-ch/cam.s/2-30,3,[winner.color,'#ffd23f','#fff']);
    }
    winDelay-=rdt;
    if(winDelay<=0&&!winShown)showWinOverlay();
  }
  else if(state==='menu'){
    updPlayer(P1,P2,dt);updPlayer(P2,P1,dt);
  }
}

/* ============================== RENDER ============================== */
function rr(x,y,w,h,r){r=Math.min(r,w/2,h/2);ctx.beginPath();ctx.moveTo(x+r,y);ctx.arcTo(x+w,y,x+w,y+h,r);ctx.arcTo(x+w,y+h,x,y+h,r);ctx.arcTo(x,y+h,x,y,r);ctx.arcTo(x,y,x+w,y,r);ctx.closePath();}
function star5(x,y,r,col){ctx.fillStyle=col;ctx.beginPath();for(let i=0;i<10;i++){const a=i*Math.PI/5-Math.PI/2,rr2=i%2?r*.45:r;ctx[i?'lineTo':'moveTo'](x+Math.cos(a)*rr2,y+Math.sin(a)*rr2);}ctx.closePath();ctx.fill();}
function w2sY(wy){return (wy-cam.y)*cam.s+ch/2;}

function drawRidge(arr,f,color,baseY){
  ctx.fillStyle=color;const by=w2sY(baseY);
  for(const m of arr){
    const sx=(m.x-cam.x*f)*cam.s+cw/2,sx2=sx+m.w*cam.s;
    if(sx2<-50||sx>cw+50)continue;
    ctx.beginPath();ctx.moveTo(sx,by);
    ctx.lineTo(sx+m.w*.18*cam.s,by-m.h*cam.s);
    ctx.lineTo(sx+m.w*.82*cam.s,by-m.h*cam.s);
    ctx.lineTo(sx2,by);ctx.closePath();ctx.fill();
  }
}
function drawPlayer(p,o){
  const R=27,t=tGlobal;
  // ground shadow
  let surf=Infinity;

function frame(ts){
  requestAnimationFrame(frame);
  const rdt=Math.min(.034,(ts-last)/1000||0);last=ts;
  if(state==='pause'){render();return;}
  if(hitstop>0){hitstop-=rdt;stepFX(rdt);render();return;}
  const dt=rdt*timeScale;
  stepFX(rdt);
  update(dt,rdt);
  render();
}
