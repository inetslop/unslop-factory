/* deterministic hash + value noise */
function h2(x,y,s){
  let n=(x|0)*374761393+(y|0)*668265263+(s|0)*974634;
  n=((n^(n>>13))*1274126177)|0; n^=n>>16;
  return (n>>>0)/4294967295;
}
function vnoise(x,y,s){
  const xi=Math.floor(x),yi=Math.floor(y),xf=x-xi,yf=y-yi;
  const u=xf*xf*(3-2*xf),v=yf*yf*(3-2*yf);
  const a=h2(xi,yi,s),b=h2(xi+1,yi,s),c=h2(xi,yi+1,s),d=h2(xi+1,yi+1,s);
  return lerp(lerp(a,b,u),lerp(c,d,u),v);
}

  const d=lerp(0.12,0.5,n*n)+(h2(x+91,y-57,SEED+6)-0.5)*0.15;
  return h2(x,y,SEED+5)<d;
}

function rebuild(prev,start,k){
  const path=[]; let c=k;
  while(c!==start){path.push({x:(c>>16)-OFF,y:(c&65535)-OFF});c=prev.get(c);}
  return path.reverse();
}
function bfsPath(sx,sy,tx,ty){
  if(sx===tx&&sy===ty) return [];
  if(Math.abs(sx)>OFF-4||Math.abs(sy)>OFF-4||Math.abs(tx)>OFF-4||Math.abs(ty)>OFF-4) return null;
  const key=(x,y)=>((x+OFF)<<16)|(y+OFF);
  const start=key(sx,sy), prev=new Map();
  const qx=[sx],qy=[sy]; let qh=0, fbKey=-1, fbD=1e9;
  while(qh<qx.length&&qh<900){

    for(let i=0;i<4;i++){
      const dx=ORDER[i][0],dy=ORDER[i][1],nx=x+dx,ny=y+dy;
      if(isWall(nx,ny)) continue;
      const k=key(nx,ny);
      if(prev.has(k)) continue;
      prev.set(k,key(x,y));
      if(nx===tx&&ny===ty) return rebuild(prev,start,k);
      const md=Math.abs(nx-tx)+Math.abs(ny-ty);
      if(md<fbD){fbD=md;fbKey=k;}
      qx.push(nx);qy.push(ny);
    }
  }
  return fbKey>=0?rebuild(prev,start,fbKey):null;
}

      if(!isWall(x,y)){tx=x;ty=y;break;}
    }
  }
  aiTarget={x:tx===null?pac.cx:tx, y:ty===null?pac.cy:ty};
}
function aiChoose(){
  let best=null,bs=-1e9;
  for(let i=0;i<4;i++){
    const dx=ORDER[i][0],dy=ORDER[i][1];
    if(isWall(pac.cx+dx,pac.cy+dy)) continue;
    const px=pac.cx+dx,py=pac.cy+dy;
    let danger=0;
    for(const g of ghosts){
      if(g.state==='eaten') continue;
      const gp=ePos(g);
      const d=Math.hypot(gp.x/CELL-px,gp.y/CELL-py);
      if(d<6) danger+=(6-d)*(fright>0?0.45:1);
    }
    let s=-Math.hypot(px-aiTarget.x,py-aiTarget.y)*3-danger*26;
    if(dx===-pac.dx&&dy!==-pac.dy) s-=6;
    if(s>bs){bs=s;best={x:dx,y:dy};}
  }
  if(best) desired=best;
}
function aiUpdate(dt){aiT1-=dt;aiT2-=dt; if(aiT1<=0){aiT1=0.8;aiThink();} if(aiT2<=0){aiT2=0.25;aiChoose();}}

function spawnBurst(x,y,n,color,speed,r){
  for(let i=0;i<n;i++){
    const a=rand(TAU),sp=rand(speed*0.4,speed);
    particles.push({x,y,vx:Math.cos(a)*sp,vy:Math.sin(a)*sp,t:0,dur:rand(.35,.7),color,r:rand(r*0.6,r*1.4)});
  }
  if(particles.length>320) particles.splice(0,particles.length-320);
}
function spawnPop(x,y,txt,color,size){popups.push({x,y,txt,color,t:0,dur:1.1,size}); if(popups.length>24)popups.shift();}
function updateFx(dt){
  for(let i=particles.length-1;i>=0;i--){
    const p=particles[i]; p.t+=dt;
    if(p.t>=p.dur){particles.splice(i,1);continue;}
    p.x+=p.vx*dt;p.y+=p.vy*dt;
    const d=Math.max(0,1-dt*2.5); p.vx*=d;p.vy*=d;
  }
  for(let i=popups.length-1;i>=0;i--){const p=popups[i];p.t+=dt;if(p.t>=p.dur)popups.splice(i,1);}
}

/* ============================== update ============================== */
function update(dt){
  if(screen==='play'&&paused) return;
  shake=Math.max(0,shake-dt*24);
  if(phase==='ready'){readyT-=dt;updateFx(dt);camFollow(dt);if(readyT<=0)phase='run';return;}
  if(phase==='dying'){deathT+=dt;updateFx(dt);camFollow(dt);if(deathT>=1.5)finishDeath();return;}

  updateFx(dt);
  if(screen==='over'){ // ambience: ghosts drift around the kill spot
    frightTick(dt);
    for(const g of ghosts) ghostStep(g,dt);
    camFollow(dt); return;
  }
  const inPlay=screen==='play';
  if(inPlay){
    runTime+=dt;
    const l=level();
    if(l>lastLevel){
      lastLevel=l;
      const pp=ePos(pac);
      spawnPop(pp.x,pp.y-CELL,'SPEED ×'+(ghostSpeed()/7).toFixed(2),'#ff9a3d',CELL*0.85);
      sfx.level();
    }
  }
  updateModes(dt); frightTick(dt);
  if(inPlay) spawnCheck(); else aiUpdate(dt);

  pac.spd=PAC_SPD;
  if(pac.pending){pac.pending=false;decidePac();}
  if(pac.stuck&&desired&&!isWall(pac.cx+desired.x,pac.cy+desired.y)){pac.stuck=false;beginMove(pac,desired.x,desired.y);}
  stepEntity(pac,dt,pacEnter);
  pac.chomp+=dt*(pac.stuck?2.5:9);

  for(const g of ghosts) ghostStep(g,dt);
  collide();
  maxDist=Math.max(maxDist,Math.hypot(pac.cx,pac.cy));

  if(inPlay&&level()>=2&&!pac.stuck){
    trailT-=dt;
    if(trailT<=0){trailT=0.07;const pp=ePos(pac);particles.push({x:pp.x,y:pp.y,vx:0,vy:0,t:0,dur:.35,color:'rgba(255,217,59,.4)',r:3});}
  }
  camFollow(dt); hudTick();
}

let vx0=0,vy0=0,vx1=0,vy1=0,nw=1,nh=1,vw=[],vp=new Uint8Array(0),pF,pH,pV;
function wallAt(x,y){
  const ix=x-vx0,iy=y-vy0;
  if(ix>=0&&iy>=0&&ix<nw&&iy<nh) return vw[iy*nw+ix]===1;
  return isWall(x,y);
}
function computeView(){
  vx0=Math.floor(camX/CELL)-1; vy0=Math.floor(camY/CELL)-1;
  vx1=Math.ceil((camX+W)/CELL)+2; vy1=Math.ceil((camY+H)/CELL)+2;
  nw=vx1-vx0; nh=vy1-vy0;
  vw=new Uint8Array(nw*nh); vp=new Uint8Array(nw*nh);
  pF=new Path2D(); pH=new Path2D(); pV=new Path2D();
  for(let y=vy0;y<vy1;y++)for(let x=vx0;x<vx1;x++){
    const idx=(y-vy0)*nw+(x-vx0);
    if(isWall(x,y)){
      vw[idx]=1; pF.rect(x*CELL,y*CELL,CELL,CELL);
      if(!wallAt(x,y-1)){pH.moveTo(x*CELL,y*CELL);pH.lineTo(x*CELL+CELL,y*CELL);}
      if(!wallAt(x,y+1)){pH.moveTo(x*CELL,y*CELL+CELL);pH.lineTo(x*CELL+CELL,y*CELL+CELL);}
      if(!wallAt(x-1,y)){pV.moveTo(x*CELL,y*CELL);pV.lineTo(x*CELL,y*CELL+CELL);}
      if(!wallAt(x+1,y)){pV.moveTo(x*CELL+CELL,y*CELL);pV.lineTo(x*CELL+CELL,y*CELL+CELL);}
    }else{
      const t=pelletHash(x,y);
      if(t&&!eaten.has(ekey(x,y))) vp[idx]=t;
    }
  }
}

function tone(type,f0,f1,t,dur,vol,dest){
  const o=AC.createOscillator(),g=AC.createGain();
  o.type=type;o.frequency.setValueAtTime(f0,t);
  if(f1)o.frequency.exponentialRampToValueAtTime(Math.max(20,f1),t+dur*0.9);
  g.gain.setValueAtTime(0.0001,t);
  g.gain.linearRampToValueAtTime(vol,t+0.008);
  g.gain.exponentialRampToValueAtTime(0.0001,t+dur);
  o.connect(g);g.connect(dest||sfxG);o.start(t);o.stop(t+dur+0.05);
}
function noiseHit(t,dur,vol,hp,dest){
  const s=AC.createBufferSource();s.buffer=noiseBuf;
  const f=AC.createBiquadFilter();f.type='highpass';f.frequency.value=hp;
  const g=AC.createGain();g.gain.setValueAtTime(vol,t);
  g.gain.exponentialRampToValueAtTime(0.001,t+dur);
  s.connect(f);f.connect(g);g.connect(dest||sfxG);s.start(t);s.stop(t+dur+0.03);
}

function scheduleStep(s,t){
  if(fright>0){
    if(s%4===0)tone('sawtooth',65,50,t,0.24,0.5,musicGain);
    else if(s%2===1)noiseHit(t,0.03,0.10,7000,musicGain);
    return;
  }
  if(s%4===0)tone('sine',150,45,t,0.12,0.6,musicGain);
  if(s%2===1)noiseHit(t,0.03,0.10,7000,musicGain);
  const f=BASS[s];
  if(f)tone('square',f,null,t,0.14,0.35,musicGain);
  if(s===8)tone('triangle',660,null,t,0.1,0.14,musicGain);
}
function musicTick(){
  if(!AC)return;
  const active=(screen==='menu')||(screen==='play'&&!paused&&(phase==='run'||phase==='ready'));
  if(muted||!active){nextNoteTime=AC.currentTime+0.1;return;}
  const spb=(60/(128+level()*5))/2;
  while(nextNoteTime<AC.currentTime+0.18){
    scheduleStep(mStep,nextNoteTime);
    nextNoteTime+=spb;mStep=(mStep+1)%16;
  }
}
