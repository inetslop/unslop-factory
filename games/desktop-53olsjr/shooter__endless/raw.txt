  if(!actx)return; delay=delay||0;
  const t=actx.currentTime+delay, o=actx.createOscillator(), g=actx.createGain();

  const b=boss;
  explosion(b.x,b.y,3,'#ffa94d');
  explosion(b.x-70,b.y+15,1.6,'#ff8a5c'); explosion(b.x+70,b.y-10,1.6,'#ffd166');
  const gain=3000*mult; score+=gain; if(score>hi)hi=score; kills++; combo+=6;
  mult=Math.min(9,1+((combo/6)|0)); maxMult=Math.max(maxMult,mult);
  pows.push({x:b.x-52,y:b.y,t:'W',vy:85,ph:1,tt:0});
  pows.push({x:b.x+52,y:b.y,t:player.weapon<4?'B':'S',vy:85,ph:3,tt:0});
  pows.push({x:b.x,y:b.y+20,t:'G',vy:95,ph:5,tt:0});
  pops.push({x:b.x,y:b.y-60,txt:'+'+gain,c:'#ffd166',t:0,dur:1.4,sz:24});
  showBanner('TARGET DESTROYED','SALVAGE +'+gain,false);
  boss=null;
}
function updateBoss(wdt){
  const b=boss; if(!b)return;
  b.t+=wdt; b.hitT=Math.max(0,b.hitT-wdt);
  if(b.dying>0){
    b.dying-=wdt; b.df-=wdt;
    if(b.df<=0){ b.df=.13; explosion(b.x+rand(-72,72),b.y+rand(-26,26),.8,'#ff9d7a'); }
    if(b.dying<=0) finalizeBoss();
    return;
  }
  if(b.y<b.ty){ b.y+=(b.ty-b.y)*Math.min(1,wdt*1.6); return; }
  b.x=W/2+Math.sin(b.t*.5)*Math.min(W*.3,320);
  const ph=Math.floor(b.t/4)%3; b.pt-=wdt;
  if(ph===0){ if(b.pt<=0){ b.pt=.8; fanShot(b.x,b.y+26,5,.62); } }
  else if(ph===1){ if(b.pt<=0){ b.pt=.085; b.sp+=.42;
    pushEB(b.x,b.y,Math.cos(b.sp)*190,Math.sin(b.sp)*190,'r');
    pushEB(b.x,b.y,Math.cos(b.sp+Math.PI)*190,Math.sin(b.sp+Math.PI)*190,'r'); } }
  else { if(b.pt<=0){ b.pt=1.6; ringShot(b.x,b.y,12);
    if(enemies.length<10){ spawnDrone(cx(clamp(b.x-95,50,W-50)),Math.random()*TAU);
      spawnDrone(cx(clamp(b.x+95,50,W-50)),Math.random()*TAU); } } }
  b.pt2-=wdt;
  if(b.pt2<=0 && player.alive){ b.pt2=3.4-diff(); aimShot(b.x,b.y+22,ebSpeed()*1.15,'r'); }
}

let ghostT=1;
function update(dt){
  tAmb+=dt;
  shake*=Math.exp(-7*dt); if(shake<.3)shake=0;
  flashA=Math.max(0,flashA-dt*2.4); dmgA=Math.max(0,dmgA-dt*1.5);
  document.getElementById('flash').style.opacity=flashA.toFixed(3);
  document.getElementById('dmg').style.opacity=dmgA.toFixed(3);
  updateAmbient(dt);

  if(state!=='play'){ updateParts(dt); updateFX(dt); return; }
  if(slowT>0){ slowT-=dt; timeScale=Math.max(.25,timeScale-dt*1.4); }
  else timeScale+=(1-timeScale)*Math.min(1,dt*5);
  const wdt=dt*timeScale;
  tPlay+=wdt;

  const nw=1+((tPlay/20)|0);
  if(nw!==wave){ wave=nw; onWave(); }
  if(bossPending&&!boss){ bossSpawnT-=wdt; if(bossSpawnT<=0) spawnBoss(); }
  if(!boss){ spawnT-=wdt;
    if(spawnT<=0){ spawnT=lerp(1.15,.4,Math.min(1,tPlay/280))*rand(.8,1.15); spawnPattern(); } }

  updatePlayer(wdt);
  for(let i=bullets.length-1;i>=0;i--){ const b=bullets[i];
    b.x+=b.vx*wdt; b.y+=b.vy*wdt;
    if(b.y<-30||b.x<-20||b.x>W+20) bullets.splice(i,1); }
  for(let i=ebullets.length-1;i>=0;i--){ const b=ebullets[i];
    b.x+=b.vx*wdt; b.y+=b.vy*wdt;
    if(b.y<-30||b.y>H+30||b.x<-30||b.x>W+30) ebullets.splice(i,1); }

  for(let i=enemies.length-1;i>=0;i--){ const e=enemies[i];
    if(e.hp<=0){ e.dying-=wdt; if(e.dying<=0) explodeEnemy(i); continue; }
    e.t+=wdt; e.hitT=Math.max(0,e.hitT-wdt);
    if(e.type==='drone'){ e.y+=e.vy*wdt; e.x=e.x0+Math.sin(e.t*e.fq+e.ph)*e.amp; }
    else if(e.type==='darter'){ e.x+=e.vx*wdt; e.y+=e.vy*wdt; }
    else if(e.type==='gunner'){
      if(e.y<e.ty) e.y+=95*wdt;
      else { e.y+=18*wdt; e.x=e.x0+Math.sin(e.t*.9+e.ph)*46;
        e.fireT-=wdt;
        if(e.fireT<=0 && player.alive){ e.fireT=e.fi*rand(.75,1.25); aimShot(e.x,e.y+13,ebSpeed(),'o'); } } }
    else if(e.type==='tank'){
      e.y+=34*wdt; e.x=e.x0+Math.sin(e.t*.5+e.ph)*20;
      e.fireT-=wdt;
      if(e.fireT<=0 && player.alive){ e.fireT=2.6-diff()*.8;
        for(const a of [-.4,0,.4]){ const ang=Math.PI/2+a;
          pushEB(e.x,e.y+12,Math.cos(ang)*ebSpeed()*.85,Math.sin(ang)*ebSpeed()*.85,'o'); } } }
    if(e.y>H+60||e.x<-90||e.x>W+90) enemies.splice(i,1);
  }

  updateBoss(wdt);

  for(let i=pows.length-1;i>=0;i--){ const p=pows[i];
    p.tt+=wdt; p.y+=p.vy*wdt;
    if(player.alive){ const dx=player.x-p.x, dy=player.y-p.y, d2=dx*dx+dy*dy;
      if(d2<130*130){ p.x+=dx*Math.min(1,wdt*4); p.y+=dy*Math.min(1,wdt*4); } }
    if(p.y>H+40) pows.splice(i,1);
  }

  collisions();
  updateParts(wdt); updateFX(wdt);
  hudSync();
}
function updateParts(dt){
  for(let i=parts.length-1;i>=0;i--){ const p=parts[i];
    p.l-=dt; if(p.l<=0){ parts.splice(i,1); continue; }
    const dr=Math.max(0,1-(p.k? .6:2.4)*dt);
    p.x+=p.vx*dt; p.y+=p.vy*dt; p.vx*=dr;
    if(p.k){ p.vy-=22*dt; p.sz+=34*dt; } }
  if(parts.length>520) parts.splice(0,parts.length-520);
}
function updateFX(dt){
  for(let i=rings.length-1;i>=0;i--){ rings[i].t+=dt; if(rings[i].t>rings[i].dur) rings.splice(i,1); }
  for(let i=pops.length-1;i>=0;i--){ const p=pops[i]; p.t+=dt; p.y-=36*dt; if(p.t>p.dur) pops.splice(i,1); }
}
function collisions(){
  for(let i=bullets.length-1;i>=0;i--){ const b=bullets[i]; let hit=false;
    for(let j=enemies.length-1;j>=0;j--){ const e=enemies[j]; if(e.hp<=0)continue;
      const dx=e.x-b.x, dy=e.y-b.y, rr=e.r+5;
      if(dx*dx+dy*dy<rr*rr){ hit=true; damageEnemy(j,1,b.x,b.y); break; } }
    if(!hit && boss && boss.dying<=0){
      const dx=boss.x-b.x, dy=boss.y-b.y;
      if(dx*dx+dy*dy<58*58){ hit=true; boss.hp--; boss.hitT=.06; spark(b.x,b.y,'#ff9d7a',3);
        if(boss.hp<=0) killBoss(); } }
    if(hit) bullets.splice(i,1);
  }
  const p=player;
  if(p.alive && p.invuln<=0){
    for(let i=ebullets.length-1;i>=0;i--){ const b=ebullets[i];
      const dx=p.x-b.x, dy=p.y-b.y, rr=b.r+9;
      if(dx*dx+dy*dy<rr*rr){ ebullets.splice(i,1);
        if(p.shield>0){ spark(b.x,b.y,'#5df2ff',6); rings.push({x:b.x,y:b.y,r:4,max:26,t:0,dur:.3,c:'#5df2ff'}); }
        else { hurt(); break; } } }
    if(p.alive && p.invuln<=0){
      for(let i=enemies.length-1;i>=0;i--){ const e=enemies[i]; if(e.hp<=0)continue;
        const dx=p.x-e.x, dy=p.y-e.y, rr=e.r+12;
        if(dx*dx+dy*dy<rr*rr){ enemies.splice(i,1); explosion(e.x,e.y,1,'#ffa94d');
          if(p.shield>0) spark(e.x,e.y,'#5df2ff',8); else hurt(); break; } } }
  }
  if(p.alive){
    for(let i=pows.length-1;i>=0;i--){ const q=pows[i];
      const dx=p.x-q.x, dy=p.y-q.y;
      if(dx*dx+dy*dy<900){ pows.splice(i,1); applyPow(q.t,q.x,q.y); } } }
}

}

const keys={};
window.addEventListener('keydown',e=>{
  ensureAudio();
  const c=e.code;
  if(c==='Space'||c==='ArrowUp'||c==='ArrowDown'||c==='ArrowLeft'||c==='ArrowRight') e.preventDefault();
  keys[c]=true;
  if(e.repeat)return;
  if(c==='Enter'){ if(state==='menu'||state==='over') startGame(); }
  else if(c==='KeyR'){ if(state!=='menu') startGame(); }
  else if(c==='KeyP'||c==='Escape'){ if(state==='play'||state==='pause') togglePause(); }
  else if(c==='KeyM'){ if(state==='pause'||state==='over') toMenu(); }
  else if(c==='ShiftLeft'||c==='ShiftRight'||c==='KeyX') doBomb();
});

function frame(ts){
  requestAnimationFrame(frame);
  let dt=(ts-last)/1000; last=ts;
  if(!(dt>0))dt=0; if(dt>.05)dt=.05;
  if(state!=='pause') update(dt);
  draw();
  if(state==='play' && player && !player.alive){ dieT-=dt; if(dieT<=0) showOver(); }
}
