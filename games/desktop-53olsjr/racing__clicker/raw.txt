function autopilot(){const laneRel=[1e9,1e9,1e9];
  for(const c of cars){let rel=(c.z-P.pos)%trackLen;if(rel<0)rel+=trackLen;
    if(rel<9000&&rel>200)laneRel[c.lane]=Math.min(laneRel[c.lane],rel);}
  const order=[P.lane,(P.lane+1)%3,(P.lane+2)%3];
  for(const l of order)if(laneRel[l]>4200||Math.abs(LANE_X[l]-P.x)>0.8){P.lane=l;break;}}

function update(rdt){
  const tsT=state==='crashed'?0.22:1;
  timeScale+=(tsT-timeScale)*Math.min(1,rdt*8);
  const dt=rdt*timeScale;
  shake*=Math.pow(0.0015,rdt);if(shake<0.05)shake=0;
  pedalFlash[0]=Math.max(0,pedalFlash[0]-rdt*4.2);
  pedalFlash[1]=Math.max(0,pedalFlash[1]-rdt*4.2);
  crashFlash=Math.max(0,crashFlash-rdt*1.6);

  const baseIdx=Math.floor(P.pos/SEG_LEN)%N;
  const baseSeg=segs[baseIdx];
  const pct=(P.pos%SEG_LEN)/SEG_LEN;
  P.y=lerp(baseSeg.p1y,baseSeg.p2y,pct);

  if(state==='playing'){
    runTime+=rdt;
    let dec=decayRate();if(P.offRoad)dec*=2.4;
    P.rpm=clamp(P.rpm-dec*dt,0,1);
    const wantNitro=(keys.space||nitroTouch.v)&&P.nitro>0.03&&P.rpm>0.3;
    if(wantNitro&&!nitroOn)sfx.nitro();
    nitroOn=wantNitro;
    if(nitroOn){P.nitro=Math.max(0,P.nitro-0.26*dt);shake=Math.max(shake,4);}
    else P.nitro=Math.min(1,P.nitro+recharge()*dt);
    const steerK=6.5+1.1*S.up.tires;
    P.x+=(LANE_X[P.lane]-P.x)*Math.min(1,dt*steerK);
    P.x+=baseSeg.curve*(P.speed/MAXS)*dt*0.16*driftMul();
    P.x=clamp(P.x,-1.25,1.25);
    const wasOff=P.offRoad;P.offRoad=Math.abs(P.x)>0.94;
    if(P.offRoad){shake=Math.max(shake,P.speed>3000?3:1.2);
      if(Math.random()<dt*22)particles.push({t:'dust',x:W/2+(Math.random()<0.5?-1:1)*W*0.06,
        y:H*0.92,vx:rand(-120,120),vy:rand(-160,-40),life:1,size:rand(3,7)});}
    if(P.rpm<0.08&&P.speed<500)addFloat('STALL!','#ff5a3c',W/2,H*0.62,22);
  }else if(state==='menu'){
    P.rpm=clamp(P.rpm+(0.5-P.rpm)*dt*2,0,1);autopilot();
    P.x+=(LANE_X[P.lane]-P.x)*Math.min(1,dt*3);
  }else if(state==='crashed'){
    P.spin+=rdt*9*(1-Math.min(1,P.crashT/1.1));
    P.rpm=Math.max(0,P.rpm-dt*2);nitroOn=false;P.crashT+=rdt;
    if(P.crashT>1.15&&!resultsShown)showResults();
  }

  let target=MAXS*Math.pow(P.rpm,1.06)*(nitroOn?1.5:1);
  if(state==='crashed')target=0;
  if(P.offRoad&&state!=='menu')target=Math.min(target,MAXS*0.5);
  P.speed+=(target-P.speed)*(1-Math.exp(-(target>P.speed?2.4:1.3)*dt));
  if(state==='playing'){P.dist+=P.speed*dt*0.0067;P.top=Math.max(P.top,P.speed/MAXS*290);}
  P.pos=(P.pos+P.speed*dt)%trackLen;

  /* cars */
  for(const c of cars){c.z=(c.z+c.speed*dt)%trackLen;
    let rel=(c.z-P.pos)%trackLen;if(rel>trackLen/2)rel-=trackLen;
    if(rel<-900){c.z=(P.pos+SEG_LEN*(DRAW-14)+rand(0,10*SEG_LEN))%trackLen;
      let l=irand(3);if(l===P.lane&&Math.random()<0.6)l=(l+1+irand(2))%3;
      c.lane=l;c.offset=LANE_X[l];c.speed=rand(2400,5200);}}
  /* coins */
  for(const k of coinsArr){let rel=(k.z-P.pos)%trackLen;if(rel<0)rel+=trackLen;
    if(rel>trackLen/2)rel-=trackLen;
    if(rel<-400){respawnCoin(k);continue;}
    const mr=magR();
    if(mr&&state==='playing'&&rel>0&&rel<1500&&Math.abs(P.x-k.off)<mr)
      k.off+=(P.x-k.off)*Math.min(1,dt*4.5);
    if(state==='playing'&&Math.abs(rel)<130&&Math.abs(P.x-k.off)<0.34){
      P.coins++;P.nitro=Math.min(1,P.nitro+nitroCoin());sfx.coin(P.coins);
      addFloat('+2¢','#ffd23f',W/2+rand(-60,60),H*0.74,20);
      for(let i=0;i<5;i++)particles.push({t:'spark',x:W/2+rand(-30,30),y:H*0.82,
        vx:rand(-140,140),vy:rand(-260,-40),life:1,size:rand(2,4)});
      respawnCoin(k);}}
  /* collisions */
  if(state==='playing'){for(const c of cars){let rel=(c.z-P.pos)%trackLen;
    if(rel>trackLen/2)rel-=trackLen;
    if(Math.abs(rel)<170&&Math.abs(P.x-c.offset)<0.34){doCrash();break;}}}

  /* particles & floats */
  for(let i=particles.length-1;i>=0;i--){const p=particles[i];p.life-=rdt*2.2;
    if(p.t==='debris')p.vy+=900*rdt;
    p.x+=p.vx*rdt;p.y+=p.vy*rdt;if(p.life<=0)particles.splice(i,1);}
  for(let i=floats.length-1;i>=0;i--){const f=floats[i];f.life-=rdt*1.1;f.y-=42*rdt;
    if(f.life<=0)floats.splice(i,1);}
}

function render(){

  const baseSeg=segs[baseIdx];
  const pct=(P.pos%SEG_LEN)/SEG_LEN;
  const camY=P.y+CAM_H,camX0=P.x*ROAD_W;
  const tB=performance.now()/1000;
  bgX+=((-P.x*46-baseSeg.curve*34)-bgX)*0.06;
  bgY+=(clamp(P.y,-6000,6000)*0.010-bgY)*0.06;

  ctx.save();
  if(shake>0)ctx.translate(rand(-shake,shake),rand(-shake,shake));
  drawBackground();

  /* ---- road pass (near → far) ---- */
  let x=0,dx=-(baseSeg.curve*pct),maxy=H+40;
  for(let n=0;n<DRAW;n++){const s=segs[(baseIdx+n)%N];s._clip=maxy;s.a=null;
    const z1=(n>0||pct===0)?n*SEG_LEN-P.pos:-P.pos%SEG_LEN; // behind-cam handled below

    const bX=W/2-kB*(camX0-x-dx)*W/2,bY=H/2-kB*(camY-s.p2y)*H/2,bW=kB*ROAD_W*W/2;

        drawCarSprite(xx,yy,u,0,e.pal,false);ctx.restore();
      }else{const r=ww*0.09;if(r<1)continue;
        const hide=yy-r*2,scl=clamp(s._clip,0,H+40);
        if(hide>scl)continue;
        ctx.save();if(hide<scl-2){ctx.beginPath();ctx.rect(0,-60,W,scl+60);ctx.clip();}
        drawCoin(xx,yy,r,e.ph+tB*3.2);ctx.restore();}}}

  /* ---- haze ---- */
  const far=segs[(baseIdx+DRAW-1)%N].a;
  const hy=far?far.y2:H*0.46;
  const hz=ctx.createLinearGradient(0,hy-30,0,hy+H*0.2);
  hz.addColorStop(0,'rgba(255,158,94,.8)');hz.addColorStop(1,'rgba(255,158,94,0)');
  ctx.fillStyle=hz;ctx.fillRect(0,hy-30,W,H*0.2+30);

  /* ---- player car ---- */
  const pu=W*0.072,pyb=H*0.955;
  let tilt=P.lean*0.4;if(state==='crashed')tilt+=P.spin;
  if(nitroOn){const fl=pu*(1.1+Math.random()*0.9);
    ctx.fillStyle='rgba(77,227,212,.85)';
    for(const s of[-1,1]){ctx.beginPath();
      ctx.moveTo(W/2+s*pu*0.55-pu*0.16,pyb-0.05);ctx.lineTo(W/2+s*pu*0.55+pu*0.16,pyb-0.05);
      ctx.lineTo(W/2+s*pu*0.55,pyb+fl);ctx.closePath();ctx.fill();}
    ctx.fillStyle='rgba(255,255,255,.7)';
    for(const s of[-1,1]){ctx.beginPath();
      ctx.moveTo(W/2+s*pu*0.55-pu*0.08,pyb);ctx.lineTo(W/2+s*pu*0.55+pu*0.08,pyb);
      ctx.lineTo(W/2+s*pu*0.55,pyb+fl*0.6);ctx.closePath();ctx.fill();}}
  drawCarSprite(W/2,pyb-pu*1.3,pu,tilt,['#ff8f3a','#c4571e'],true);

  /* ---- speed lines ---- */
  const sv=nitroOn?1:clamp((P.rpm-0.62)/0.38,0,1)*(state==='playing'?1:0.4);
  if(sv>0.05){const md=Math.max(W,H)*0.75;ctx.strokeStyle='#fff';
    for(let i=0;i<Math.round(22*sv);i++){const a=rand(0,6.283),r0=rand(0.3,0.6)*md,l=rand(0.1,0.4)*md*sv;
      ctx.globalAlpha=0.05+0.22*sv*Math.random();ctx.lineWidth=rand(1,2.5);
      ctx.beginPath();ctx.moveTo(W/2+Math.cos(a)*r0,H/2+Math.sin(a)*r0);
      ctx.lineTo(W/2+Math.cos(a)*(r0+l),H/2+Math.sin(a)*(r0+l));ctx.stroke();}
    ctx.globalAlpha=1;}

  /* ---- particles / floats ---- */
  for(const p of particles){const a=Math.max(0,p.life);
    if(p.t==='puff'){ctx.fillStyle='rgba(200,205,220,'+(a*0.35).toFixed(2)+')';
      ctx.beginPath();ctx.arc(p.x,p.y,p.size*(1.6-a),0,7);ctx.fill();}
    else if(p.t==='spark'){ctx.fillStyle='rgba(255,210,63,'+a.toFixed(2)+')';
      ctx.fillRect(p.x,p.y,p.size,p.size);}
    else if(p.t==='dust'){ctx.fillStyle='rgba(122,74,58,'+(a*0.6).toFixed(2)+')';
      ctx.beginPath();ctx.arc(p.x,p.y,p.size*(1.4-a),0,7);ctx.fill();}
    else{ctx.save();ctx.translate(p.x,p.y);ctx.rotate(p.rot+ (1-p.life)*8);
      ctx.fillStyle=p.col;ctx.globalAlpha=a;ctx.fillRect(-p.size/2,-p.size/2,p.size,p.size);
      ctx.restore();ctx.globalAlpha=1;}}
  for(const f of floats){ctx.font='italic 900 '+f.size+'px '+"'Arial Black',sans-serif";
    ctx.textAlign='center';ctx.fillStyle=f.col;ctx.globalAlpha=Math.min(1,f.life*1.4);
    ctx.fillText(f.t,f.x,f.y);ctx.globalAlpha=1;}

  /* ---- HUD ---- */
  if(state!=='menu')drawHUD(tB);
  ctx.restore();

  if(crashFlash>0){ctx.fillStyle='rgba(255,236,200,'+(crashFlash*0.8).toFixed(2)+')';
    ctx.fillRect(0,0,W,H);}
  engineTick(P.rpm,(state==='playing'||state==='crashed')&&!document.hidden,nitroOn);
}
