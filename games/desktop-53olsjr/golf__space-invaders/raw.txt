 if(b.state!=='roll')return;

  let hit=0;
  if(b.x<F.x+BALL_R){b.x=F.x+BALL_R;if(b.vx<0){hit=Math.abs(b.vx);b.vx=-b.vx*WALL_REST;}}

let AC=null,master=null;
function initAudio(){
  if(AC){if(AC.state==='suspended')AC.resume();return;}
  try{

    master=AC.createGain();master.gain.value=.16;master.connect(AC.destination);

}
function tone(f0,f1,dur,type,vol,delay){
  if(!AC)return;delay=delay||0;
  const t0=AC.currentTime+delay,o=AC.createOscillator(),g=AC.createGain();
  o.type=type;o.frequency.setValueAtTime(Math.max(20,f0),t0);
  o.frequency.exponentialRampToValueAtTime(Math.max(20,f1),t0+dur);
  g.gain.setValueAtTime(vol,t0);g.gain.exponentialRampToValueAtTime(.001,t0+dur);
  o.connect(g);g.connect(master);o.start(t0);o.stop(t0+dur+.03);
}
function noiseBurst(dur,vol,delay){
  if(!AC)return;delay=delay||0;
  const t0=AC.currentTime+delay,n=Math.max(8,Math.floor(AC.sampleRate*dur));
  const buf=AC.createBuffer(1,n,AC.sampleRate),d=buf.getChannelData(0);
  for(let i=0;i<n;i++)d[i]=(Math.random()*2-1)*(1-i/n);
  const s=AC.createBufferSource();s.buffer=buf;
  const g=AC.createGain();g.gain.value=vol;s.connect(g);g.connect(master);s.start(t0);
}

  b.state='roll';h.putts++;G.aim.power=0;G.charging=false;
  sfx.shoot();G.shake=Math.max(G.shake,1.5);burst(b.x,b.y,'#e8fff2',6);
}

    const sp=Math.hypot(b.vx,b.vy);if(sp>MAXS){b.vx*=MAXS/sp;b.vy*=MAXS/sp;}

  G.state='holeDone';G.doneT=0;G.shake=Math.max(G.shake,3);
}

/* ---------------- physics ---------------- */
function stepBall(dt){
  const b=G.ball,h=G.hole,c=h.cup;
  if(b.state!=='roll')return;
  let sp=Math.hypot(b.vx,b.vy);
  const steps=Math.max(1,Math.ceil(sp*dt/6)),sdt=dt/steps;
  for(let i=0;i<steps;i++){
    b.x+=b.vx*sdt;b.y+=b.vy*sdt;
    let hit=0;
    if(b.x<F.x+BALL_R){b.x=F.x+BALL_R;if(b.vx<0){hit=Math.abs(b.vx);b.vx=-b.vx*WALL_REST;}}
    else if(b.x>F.x+F.w-BALL_R){b.x=F.x+F.w-BALL_R;if(b.vx>0){hit=Math.abs(b.vx);b.vx=-b.vx*WALL_REST;}}
    if(b.y<F.y+BALL_R){b.y=F.y+BALL_R;if(b.vy<0){hit=Math.max(hit,Math.abs(b.vy));b.vy=-b.vy*WALL_REST;}}
    else if(b.y>F.y+F.h-BALL_R){b.y=F.y+F.h-BALL_R;if(b.vy>0){hit=Math.max(hit,Math.abs(b.vy));b.vy=-b.vy*WALL_REST;}}
    if(hit>60){sfx.bounce(hit);if(hit>380)G.shake=Math.max(G.shake,1.5);}
    for(const r of h.rocks){
      let dx=b.x-r.x,dy=b.y-r.y,d=Math.hypot(dx,dy),min=r.r+BALL_R;
      if(d<min){if(d===0){dx=1;dy=0;d=1;}const nx=dx/d,ny=dy/d;b.x=r.x+nx*min;b.y=r.y+ny*min;
        const vn=b.vx*nx+b.vy*ny;
        if(vn<0){b.vx-=(1+ROCK_REST)*vn*nx;b.vy-=(1+ROCK_REST)*vn*ny;if(-vn>60)sfx.bounce(-vn);if(-vn>300)G.shake=Math.max(G.shake,2);}}
    }
    for(const bar of h.bars){
      const cx=Math.max(bar.x,Math.min(b.x,bar.x+bar.w)),cy=Math.max(bar.y,Math.min(b.y,bar.y+bar.h));
      let dx=b.x-cx,dy=b.y-cy;const d2=dx*dx+dy*dy;
      if(d2<BALL_R*BALL_R){
        let nx,ny;const d=Math.sqrt(d2);
        if(d>.001){nx=dx/d;ny=dy/d;}
        else{const l=b.x-bar.x,rg=(bar.x+bar.w)-b.x,t=b.y-bar.y,bt=(bar.y+bar.h)-b.y,m=Math.min(l,rg,t,bt);
          if(m===l){nx=-1;ny=0;b.x=bar.x-BALL_R;}else if(m===rg){nx=1;ny=0;b.x=bar.x+bar.w+BALL_R;}
          else if(m===t){nx=0;ny=-1;b.y=bar.y-BALL_R;}else{nx=0;ny=1;b.y=bar.y+bar.h+BALL_R;}}
        b.x+=nx*(BALL_R-d);b.y+=ny*(BALL_R-d);
        const vn=b.vx*nx+b.vy*ny;if(vn<0){b.vx-=(1+WALL_REST)*vn*nx;b.vy-=(1+WALL_REST)*vn*ny;if(-vn>60)sfx.bounce(-vn);}
      }
    }
    const dx=c.x-b.x,dy=c.y-b.y,d=Math.hypot(dx,dy)||1;
    if(d<48){const k=1600*(1-d/48);b.vx+=dx/d*k*sdt;b.vy+=dy/d*k*sdt;}
    sp=Math.hypot(b.vx,b.vy);
    if(d<CUP_R-2&&sp<220){sinkHole();return;}
  }
  const f=Math.exp(-DECAY*dt);b.vx*=f;b.vy*=f;
  sp=Math.hypot(b.vx,b.vy);
  if(sp<9){b.vx=0;b.vy=0;b.state='aim';}
  b.trail.push({x:b.x,y:b.y});if(b.trail.length>16)b.trail.shift();
}

/* ---------------- update ---------------- */
function updatePlay(dt){
  const h=G.hole,b=G.ball;G.holeT+=dt;
  if(b.state==='aim'){
    if(keys.left)G.aim.angle-=2.6*dt*(keys.shift?.35:1);
    if(keys.right)G.aim.angle+=2.6*dt*(keys.shift?.35:1);
    if(G.charging)G.chargeP=Math.min(1,G.chargeP+dt/1.15),G.aim.power=G.chargeP;
    if(G.aiming){const dx=b.x-G.aimCur.x,dy=b.y-G.aimCur.y,len=Math.hypot(dx,dy);
      if(len>4){G.aim.angle=Math.atan2(dy,dx);G.aim.power=Math.min(1,len/230);}else G.aim.power=0;}
    if(Math.hypot(b.x-h.cup.x,b.y-h.cup.y)<16)sinkHole();
  }
  for(const a of h.aliens){
    a.y+=h.speed*dt;a.x=a.baseX+Math.sin(T*.9+a.phase)*14;
    if(a.y+12>=BOMB_Y){a.dead=true;h.penalties++;G.shake=Math.max(G.shake,5);G.redFlash=.6;
      sfx.breach();addText('PENALTY +1',a.x,BOMB_Y-18,'#ff5470');continue;}
    if(!a.dead&&b.state!=='sunk'&&Math.hypot(b.x-a.x,b.y-a.y)<ALIEN_R+BALL_R)popAlien(a);
  }
  h.aliens=h.aliens.filter(a=>!a.dead);
  stepBall(dt);
  G.laserT-=dt;
  if(G.laserT<=0&&h.aliens.length){G.laserT=1.1+Math.random()*1.2;
    const a=h.aliens[(Math.random()*h.aliens.length)|0];
    G.lasers.push({x:a.x,y:a.y,t:0,c:TIER_COLORS[a.tier]});}
  for(const L of G.lasers)L.t+=dt;
  G.lasers=G.lasers.filter(L=>L.t<.16);
}
function update(dt){
  T+=dt;
  for(const s of STARS){s.y+=s.sp*dt;if(s.y>H+2){s.y=-2;s.x=Math.random()*W;}}
  G.shake*=Math.exp(-5*dt);if(G.shake<.05)G.shake=0;
  G.redFlash=Math.max(0,G.redFlash-dt*1.6);
  for(const p of G.parts){p.x+=p.vx*dt;p.y+=p.vy*dt;p.vx*=Math.exp(-3*dt);p.vy*=Math.exp(-3*dt);p.life-=dt;}
  G.parts=G.parts.filter(p=>p.life>0);
  for(const t of G.texts){t.y-=36*dt;t.life-=dt;}
  G.texts=G.texts.filter(t=>t.life>0);
  for(const r of G.rings)r.t+=dt;
  G.rings=G.rings.filter(r=>r.t<.7);

  if(G.state==='menu')return;
  if(G.state==='play'){updatePlay(dt);return;}
  if(G.state==='holeDone'){
    G.doneT+=dt;
    for(const a of G.hole.aliens){a.y+=G.hole.speed*dt;a.x=a.baseX+Math.sin(T*.9+a.phase)*14;
      if(a.y>F.y+F.h-12)a.y=F.y+16;}
    if(G.doneT>2.8)advance();
  }
}

window.addEventListener('keydown',e=>{
  initAudio();
  const c=e.code;
  if(c==='Space'||c==='ArrowLeft'||c==='ArrowRight'||c==='ArrowUp'||c==='ArrowDown')e.preventDefault();
  if(c==='ShiftLeft'||c==='ShiftRight')keys.shift=true;
  if(G.state==='menu'){if(c==='Enter'||c==='Space'){sfx.ui();startCourse();}return;}
  if(G.state==='courseEnd'){
    if(c==='Enter'||c==='KeyR'){sfx.ui();startCourse();}else if(c==='KeyM')toMenu();return;}
  if(G.state==='holeDone'){
    if((c==='Enter'||c==='Space')&&G.doneT>.7){sfx.ui();advance();}else if(c==='KeyM')toMenu();return;}
  // play
  if(c==='KeyM'){toMenu();return;}
  if(c==='KeyR'){initHole(G.holeIdx);return;}
  if(c==='ArrowLeft'||c==='KeyA')keys.left=true;
