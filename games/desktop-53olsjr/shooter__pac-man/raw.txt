/* ================= GRID HELPERS ================= */
const cellCX=gx=>OFFX+gx*CELL+CELL/2;
const cellCY=gy=>TOP+gy*CELL+CELL/2;
const cellOf=x=>Math.floor((x-OFFX)/CELL);
const rowOf=y=>Math.floor((y-TOP)/CELL);
function openG(gx,gy){if(gx<0||gy<0||gx>=COLS||gy>=ROWS)return false;return !walls[gy][gx];}
function openP(gx,gy){if(!openG(gx,gy))return false;for(const h of HOUSE)if(h[0]===gx&&h[1]===gy)return false;return true;}
function losClear(x0,y0,x1,y1){
  const d=Math.hypot(x1-x0,y1-y0); if(d<2)return true;
  const n=Math.ceil(d/14);
  for(let i=1;i<n;i++){const px=x0+(x1-x0)*i/n,py=y0+(y1-y0)*i/n;if(!openG(cellOf(px),rowOf(py)))return false;}
  return true;
}

function rr(c,x,y,w,h,r){c.beginPath();c.moveTo(x+r,y);c.arcTo(x+w,y,x+w,y+h,r);c.arcTo(x+w,y+h,x,y+h,r);c.arcTo(x,y+h,x,y,r);c.arcTo(x,y,x+w,y,r);c.closePath();}

function buildMazeLayer(){
  mctx.setTransform(RS,0,0,RS,0,0); mctx.clearRect(0,0,W,H);
  // playfield panel
  rr(mctx,OFFX-16,TOP-16,COLS*CELL+32,ROWS*CELL+32,14);
  mctx.fillStyle='rgba(9,15,38,0.62)'; mctx.fill();
  mctx.strokeStyle='rgba(70,110,255,0.4)'; mctx.lineWidth=2; mctx.stroke();
  // faint grid
  mctx.strokeStyle='rgba(90,130,255,0.05)'; mctx.lineWidth=1;
  for(let gx=0;gx<=COLS;gx++){mctx.beginPath();mctx.moveTo(OFFX+gx*CELL,TOP);mctx.lineTo(OFFX+gx*CELL,TOP+ROWS*CELL);mctx.stroke();}
  for(let gy=0;gy<=ROWS;gy++){mctx.beginPath();mctx.moveTo(OFFX,TOP+gy*CELL);mctx.lineTo(OFFX+COLS*CELL,TOP+gy*CELL);mctx.stroke();}
  // house tint
  mctx.fillStyle='rgba(70,95,220,0.12)';
  for(const h of HOUSE)mctx.fillRect(OFFX+h[0]*CELL+2,TOP+h[1]*CELL+2,CELL-4,CELL-4);
  // wall edge segments
  const segs=[];
  const wallAt=(gx,gy)=>{if(gx<0||gy<0||gx>=COLS||gy>=ROWS)return false;return walls[gy][gx];};
  for(let gy=0;gy<ROWS;gy++)for(let gx=0;gx<COLS;gx++){
    if(!walls[gy][gx])continue;
    const x=OFFX+gx*CELL,y=TOP+gy*CELL;
    if(!wallAt(gx-1,gy))segs.push([x+2.5,y+2,x+2.5,y+CELL-2]);
    if(!wallAt(gx+1,gy))segs.push([x+CELL-2.5,y+2,x+CELL-2.5,y+CELL-2]);
    if(!wallAt(gx,gy-1))segs.push([x+2,y+2.5,x+CELL-2,y+2.5]);
    if(!wallAt(gx,gy+1))segs.push([x+2,y+CELL-2.5,x+CELL-2,y+CELL-2.5]);
  }
  const pass=(lw,col)=>{mctx.lineWidth=lw;mctx.strokeStyle=col;mctx.lineCap='round';mctx.beginPath();
    for(const s of segs){mctx.moveTo(s[0],s[1]);mctx.lineTo(s[2],s[3]);} mctx.stroke();};
  pass(7,'rgba(56,96,255,0.30)');
  pass(3,'#4d7dff');
  pass(1.2,'rgba(190,214,255,0.85)');
}

function drawDotAt(c,gx,gy){
  const x=cellCX(gx),y=cellCY(gy);
  const g=c.createRadialGradient(x,y,0,x,y,7);
  g.addColorStop(0,'rgba(255,168,110,0.35)');g.addColorStop(1,'rgba(255,168,110,0)');
  c.fillStyle=g;c.beginPath();c.arc(x,y,7,0,TAU);c.fill();
  c.fillStyle='#ffb27a';c.beginPath();c.arc(x,y,3.4,0,TAU);c.fill();
  c.fillStyle='#fff1dc';c.beginPath();c.arc(x,y,1.8,0,TAU);c.fill();
}

const SLOTS=[[10,-12],[9,6],[11,6],[10,16]];
function resetToHouse(g){
  const s=SLOTS[g.ti];
  g.gx=s[0];g.x=cellCX(s[0]);g.y=cellCY(7)+s[1];
  g.dirX=0;g.dirY=-1;g.state='house';g.hp=3;g.stun=0;g.chargeT=0;
}
function resetGhosts(){
  for(let i=0;i<4;i++){const g=ghosts[i];resetToHouse(g);g.exitT=1.2+i*1.1;g.fireCd=2.5+i*.8;g.dashT=0;g.dashCd=4+Math.random()*3;}
}
function resetPositions(){
  player.gx=10;player.gy=9;player.x=cellCX(10);player.y=cellCY(9);
  player.dirX=0;player.dirY=0;player.face=-Math.PI/2;
  invulnT=2.2;vulnT=0;combo=0;frenzyAcc=0;
  bullets.length=0;bolts.length=0;pickups.length=0;queued.length=0;
  resetGhosts();
}
function startRun(){
  initAudio();
  score=0;level=1;lives=3;energy=100;runStartHi=hi;newHi=false;
  rapidT=spreadT=shieldT=0;fireCd=0;muzzleT=0;shake=0;flashT=0;
  bullets.length=0;bolts.length=0;pickups.length=0;parts.length=0;floats.length=0;
  buildDots();resetPositions();
  state='play';paused=false;readyT=1.4;overT=0;

}

  }
  for(let i=0;i<6;i++){
    const a=Math.random()*TAU,sp=200+Math.random()*200,life=.25+Math.random()*.3;
    parts.push({kind:'p',x:x,y:y,vx:Math.cos(a)*sp,vy:Math.sin(a)*sp,life:life,max:life,size:1.6,color:'#ffffff'});

function updateParticles(dt){
  for(let i=parts.length-1;i>=0;i--){
    const p=parts[i];p.life-=dt;
    if(p.life<=0){parts.splice(i,1);continue;}
    if(p.kind==='p'){p.x+=p.vx*dt;p.y+=p.vy*dt;p.vx*=(1-3*dt);p.vy*=(1-3*dt);}
  }
  if(parts.length>420)parts.splice(0,parts.length-420);
}
function updateFloaters(dt){for(let i=floats.length-1;i>=0;i--){floats[i].t-=dt;if(floats[i].t<=0)floats.splice(i,1);}}

/* ================= MOVEMENT (grid) ================= */
function stepGrid(e,dt,open,wantFn){
  if(!e.dirX&&!e.dirY){
    const d=wantFn();
    if(d&&(d[0]||d[1])&&open(e.gx+d[0],e.gy+d[1])){e.dirX=d[0];e.dirY=d[1];}
    return;
  }
  const sp=e.speed*dt,nx=e.x+e.dirX*sp,ny=e.y+e.dirY*sp;
  const tx=e.gx+e.dirX,ty=e.gy+e.dirY;
  const tcx=cellCX(tx),tcy=cellCY(ty);
  if((e.dirX>0&&nx>=tcx)||(e.dirX<0&&nx<=tcx)||(e.dirY>0&&ny>=tcy)||(e.dirY<0&&ny<=tcy)){
    e.x=tcx;e.y=tcy;e.gx=tx;e.gy=ty;
    const d=wantFn();
    if(d&&(d[0]||d[1])&&open(tx+d[0],ty+d[1])&&(d[0]!==e.dirX||d[1]!==e.dirY)){e.dirX=d[0];e.dirY=d[1];}
    if(!open(tx+e.dirX,ty+e.dirY)){e.dirX=0;e.dirY=0;}
  }else{e.x=nx;e.y=ny;}
}

/* ================= GHOST AI ================= */
function ghostWant(g){
  const D=[[1,0],[-1,0],[0,1],[0,-1]];
  let opts=[];
  for(const d of D){if(d[0]===-g.dirX&&d[1]===-g.dirY)continue;if(openG(g.gx+d[0],g.gy+d[1]))opts.push(d);}
  if(!opts.length)return openG(g.gx-g.dirX,g.gy-g.dirY)?[-g.dirX,-g.dirY]:[0,0];
  const pgx=player.gx,pgy=player.gy;
  let tx=pgx,ty=pgy,flee=false;
  if(g.ai==='flank'){tx=pgx+player.dirX*3;ty=pgy+player.dirY*3;}
  else if(g.ai==='wander'){if(Math.random()<0.45)return opts[(Math.random()*opts.length)|0];}
  else if(g.ai==='lurker'){const dd=Math.hypot(g.gx-pgx,g.gy-pgy);if(!g.dashT&&dd<=7)flee=true;}
  let best=null,bs=flee?-1:1e9;
  for(const d of opts){
    const v=(g.gx+d[0]-tx)*(g.gx+d[0]-tx)+(g.gy+d[1]-ty)*(g.gy+d[1]-ty);
    if(flee?v>bs:v<bs){bs=v;best=d;}
  }
  return best||opts[0];
}

/* ================= COMBAT ================= */
function spawnBullet(a){
  let mx=player.x+Math.cos(a)*(player.r+8),my=player.y+Math.sin(a)*(player.r+8);
  if(!openG(cellOf(mx),rowOf(my))){mx=player.x;my=player.y;}
  bullets.push({x:mx,y:my,vx:Math.cos(a)*920,vy:Math.sin(a)*920,life:1.3,grace:.05});
}
function tryFire(){
  if(state!=='play'||paused)return;
  if(readyT>0||levelT>0||dyingT>0)return;
  if(fireCd>0)return;
  const cost=spreadT>0?14:8;
  if(energy<cost){if(dryT<=0){sfx.dry();dryT=.25;}return;}
  energy-=cost;fireCd=rapidT>0?0.065:0.13;muzzleT=0.07;
  spawnBullet(aimAngle);
  if(spreadT>0){spawnBullet(aimAngle-0.24);spawnBullet(aimAngle+0.24);}
  sfx.shoot();
}
function fireBolt(g){
  const a=Math.atan2(player.y-g.y,player.x-g.x);
  const sp=Math.min(430,300+level*12);
  bolts.push({x:g.x+Math.cos(a)*20,y:g.y+Math.sin(a)*20,vx:Math.cos(a)*sp,vy:Math.sin(a)*sp,life:6,grace:.05});
  sfx.boltFire();sparkle(g.x+Math.cos(a)*24,g.y+Math.sin(a)*24,'#ff4fd8',5);
}
function killGhost(g,viaVuln){
  if(g.state!=='out')return;
  g.state='dying';g.deathT=.35;
  const pts=viaVuln?200*(1<<Math.min(combo,3)):(shieldT>0?200:150);
  if(viaVuln)combo++;
  score+=pts;if(score>hi)hi=score;
  floats.push({x:g.x,y:g.y-16,txt:'+'+pts,color:viaVuln?'#7df5ff':g.color,t:.9,max:.9});
  explode(g.x,g.y,g.color,20);addRing(g.x,g.y,g.color);
  shake=Math.max(shake,viaVuln?8:4.5);sfx.kill();
  if(Math.random()<0.35){
    const types=['rapid','spread','shield'];
    pickups.push({x:g.x,y:g.y,type:types[(Math.random()*3)|0],life:9,bob:Math.random()*6});
  }
}
function playerHit(){
  if(invulnT>0||dyingT>0||state!=='play')return;
  lives--;
  explode(player.x,player.y,'#ffe23f',28);addRing(player.x,player.y,'#ffe23f');
  shake=15;flashT=.4;flashCol='255,70,80';
  sfx.boltHit();sfx.kill();
  dyingT=1.15;
}

/* ================= EATING / POWER ================= */
function checkEat(gx,gy){
  if(gx<0||gy<0||gx>=COLS||gy>=ROWS)return;
  if(dots[gy][gx]){
    dots[gy][gx]=0;dotsLeft--;score+=10;if(score>hi)hi=score;
    clearDotLayer(gx,gy);
    streak=(t-lastDotT<0.7)?Math.min(streak+1,12):0;lastDotT=t;
    sfx.dot(streak);sparkle(cellCX(gx),cellCY(gy),'#ffcf9a',3);
  }else if(powers[gy][gx]){
    powers[gy][gx]=0;powersLeft--;score+=50;if(score>hi)hi=score;
    clearDotLayer(gx,gy);
    energy=Math.min(100,energy+40);
    vulnT=Math.max(3.5,8-(level-1)*0.5);combo=0;frenzyAcc=0;
    for(const g of ghosts)if(g.state==='out'&&(g.dirX||g.dirY)){g.dirX*=-1;g.dirY*=-1;}
    flashT=.2;flashCol='90,150,255';
    floats.push({x:cellCX(gx),y:cellCY(gy)-18,txt:'POWER!',color:'#ffb347',t:.9,max:.9});
    sparkle(cellCX(gx),cellCY(gy),'#ffb347',16);addRing(cellCX(gx),cellCY(gy),'#ffb347');
    sfx.power();
  }
}

/* ================= UPDATE ================= */
function update(dt){
  t+=dt;
  shake=Math.max(0,shake-shake*6*dt);
  flashT=Math.max(0,flashT-dt);
  muzzleT=Math.max(0,muzzleT-dt);
  dryT=Math.max(0,dryT-dt);
  updateParticles(dt);updateFloaters(dt);

  if(state==='start'){
    for(const g of ghosts)if(g.state==='house')g.bob+=dt;
    ambAcc+=dt;
    while(ambAcc>0.14){ambAcc-=0.14;if(dotCells.length){const c=dotCells[(Math.random()*dotCells.length)|0];sparkle(cellCX(c[0]),cellCY(c[1]),'#ffcf9a',1);}}
    return;
  }
  if(state==='over'){overT+=dt;return;}
  if(paused)return;

  fireCd=Math.max(0,fireCd-dt);
  rapidT=Math.max(0,rapidT-dt);spreadT=Math.max(0,spreadT-dt);
  shieldT=Math.max(0,shieldT-dt);invulnT=Math.max(0,invulnT-dt);
  energy=Math.min(100,energy+30*dt);

  if(vulnT>0){
    vulnT-=dt;frenzyAcc+=dt;
    while(frenzyAcc>=0.22){frenzyAcc-=0.22;beepFlip=!beepFlip;beep(beepFlip?700:480,0,.05,'square',.045);}
    if(vulnT<=0)combo=0;
  }

  if(dyingT>0){
    dyingT-=dt;
    if(dyingT<=0){if(lives<=0)gameOver();else{resetPositions();readyT=1.4;}}
  }
  if(levelT>0)levelT-=dt;
  if(readyT>0)readyT-=dt;
  const frozen=dyingT>0||readyT>0||levelT>0;
  if(frozen)return;

  // aim
  const mx=(mouse.cx-offX)/scale,my=(mouse.cy-offY)/scale;
  aimAngle=Math.atan2(my-player.y,mx-player.x);

  // player
  player.moving=!!(player.dirX||player.dirY);
  stepGrid(player,dt,openP,playerWant);
  if(player.moving){chewT+=dt;player.face=Math.atan2(player.dirY,player.dirX);}
  checkEat(player.gx,player.gy);
  if(firing||spaceHeld)tryFire();

  // ghosts
  for(const g of ghosts){
    if(g.state==='dying'){g.deathT-=dt;if(g.deathT<=0){g.state='dead';g.deadT=2.4;}continue;}
    if(g.state==='dead'){g.deadT-=dt;if(g.deadT<=0){resetToHouse(g);g.exitT=1.4;}continue;}
    if(g.state==='house'){
      g.bob+=dt;
      if(g.exitT>0)g.exitT-=dt;
      else{g.gx=10;g.gy=7;g.x=cellCX(10);g.y=cellCY(7);g.dirX=0;g.dirY=-1;g.state='out';}
      continue;
    }
    if(g.stun>0){g.stun-=dt;continue;}
    if(g.ai==='lurker'){
      g.dashCd-=dt;
      if(g.dashCd<=0){g.dashT=1.1;g.dashCd=5+Math.random()*3;sfx.dash();}
      if(g.dashT>0)g.dashT-=dt;
    }
    let sp=g.base*(1+0.06*(level-1));
    if(vulnT>0)sp*=0.5;else if(g.dashT>0)sp=400;else sp=Math.min(sp,258);
    g.speed=sp;
    stepGrid(g,dt,openG,()=>ghostWant(g));
    if(g.shooter&&vulnT<=0){
      g.fireCd-=dt;
      const dist=Math.hypot(player.x-g.x,player.y-g.y);
      if(g.chargeT>0){g.chargeT-=dt;if(g.chargeT<=0)fireBolt(g);}
      else if(g.fireCd<=0&&dist<430&&losClear(g.x,g.y,player.x,player.y)){g.chargeT=0.55;sfx.charge();}
    }
    const dd=Math.hypot(player.x-g.x,player.y-g.y);
    if(dd<g.r+player.r-4){
      if(vulnT>0)killGhost(g,true);
      else if(shieldT>0)killGhost(g,false);
      else playerHit();
    }
  }

  // bullets
  for(let i=bullets.length-1;i>=0;i--){
    const b=bullets[i];b.life-=dt;b.grace-=dt;
    let dead=b.life<=0;
    for(let s2=0;s2<2&&!dead;s2++){
      b.x+=b.vx*dt/2;b.y+=b.vy*dt/2;
      if(b.grace<=0&&!openG(cellOf(b.x),rowOf(b.y))){sparkle(b.x,b.y,'#9fd4ff',5);sfx.ui();dead=true;break;}
    }
    if(!dead)for(let j=ghosts.length-1;j>=0;j--){
      const g=ghosts[j];if(g.state!=='out')continue;
      if(Math.hypot(b.x-g.x,b.y-g.y)<22){
        if(vulnT>0)killGhost(g,true);
        else{g.hp--;g.stun=.55;sfx.hitG();sparkle(b.x,b.y,'#ffffff',8);if(g.hp<=0)killGhost(g,false);}
        dead=true;break;
      }
    }
    if(!dead)for(let j=bolts.length-1;j>=0;j--){
      const bl=bolts[j];
      if(Math.hypot(b.x-bl.x,b.y-bl.y)<14){
        score+=50;if(score>hi)hi=score;
        floats.push({x:bl.x,y:bl.y-12,txt:'+50',color:'#7df5ff',t:.7,max:.7});
        sparkle(bl.x,bl.y,'#ffd6f2',8);sfx.clank();
        bolts.splice(j,1);dead=true;break;
      }
    }
    if(dead)bullets.splice(i,1);
  }

  // enemy bolts
  for(let i=bolts.length-1;i>=0;i--){
    const b=bolts[i];b.life-=dt;b.grace-=dt;
    let dead=b.life<=0;
    for(let s2=0;s2<2&&!dead;s2++){
      b.x+=b.vx*dt/2;b.y+=b.vy*dt/2;
      if(b.grace<=0&&!openG(cellOf(b.x),rowOf(b.y))){sparkle(b.x,b.y,'#ff4fd8',6);dead=true;}
    }
    if(!dead&&invulnT<=0&&dyingT<=0){
      if(Math.hypot(b.x-player.x,b.y-player.y)<21){
        if(shieldT>0){addRing(player.x,player.y,'#7dff9b');sparkle(b.x,b.y,'#7dff9b',8);sfx.clank();}
        else{playerHit();}
        dead=true;
      }
    }
    if(dead)bolts.splice(i,1);
  }

  // pickups
  for(let i=pickups.length-1;i>=0;i--){
    const p=pickups[i];p.life-=dt;p.bob+=dt;
    if(p.life<=0){pickups.splice(i,1);continue;}
    if(Math.hypot(p.x-player.x,p.y-player.y)<24){
      const names={rapid:'RAPID FIRE',spread:'SPREAD SHOT',shield:'SHIELD'};
      const cols={rapid:'#ff9f43',spread:'#7df5ff',shield:'#7dff9b'};
      if(p.type==='rapid')rapidT=8;else if(p.type==='spread')spreadT=8;else shieldT=4.5;
      floats.push({x:p.x,y:p.y-14,txt:names[p.type],color:cols[p.type],t:1,max:1});
      addRing(p.x,p.y,cols[p.type]);sfx.pickup();
      pickups.splice(i,1);
    }
  }

  // level clear
  if(dotsLeft<=0&&powersLeft<=0){
    level++;score+=500;if(score>hi)hi=score;
    buildDots();resetPositions();
    bullets.length=0;bolts.length=0;pickups.length=0;
    levelT=1.7;sfx.jingle();
  }
}

function drawGhost(g){

  if(vulnT>0&&state!=='start'&&(g.state==='out'||g.state==='house')){
    body=(vulnT<1.6&&Math.floor(t*8)%2)?'#f4f7ff':'#2440ff';
  }
  if(g.dashT>0&&Math.floor(t*10)%2)body='#ffffff';

  ctx.save();ctx.translate(x,y);
  const r=g.r;
  if(g.state==='dying'){const k=1-g.deathT/.35;ctx.globalAlpha=Math.max(0,1-k);ctx.rotate(k*7);ctx.scale(Math.max(.05,1-k),Math.max(.05,1-k));}

  ctx.beginPath();

  // eyes
  let lx,ly;
  if(g.state==='house'){const a=Math.atan2(player.y-y,player.x-x);lx=Math.cos(a)*2;ly=Math.sin(a)*2;}
  else{lx=g.dirX*2.4;ly=g.dirY*2.4;}

  }
  for(const p of parts){

      ctx.globalAlpha=al;ctx.fillStyle=p.color;

    }
  }
  ctx.globalAlpha=1;ctx.globalCompositeOperation='source-over';

  for(const f of floats){
    const k=f.t/f.max;
    ptext(f.txt,f.x,f.y-(1-k)*30,2,f.color,'center',k);

  drawHUD();

    if(readyT>0&&state==='play')ptext('READY!',W/2,TOP+10.6*CELL-8,4,'#ffe23f','center');

      ctx.fillStyle='rgba(3,5,14,0.35)';ctx.fillRect(0,0,W,H);

    }
    if(state==='over')drawOver();
    if(paused&&state==='play')drawPause();
  }

  // crosshair
  const mx=(mouse.cx-offX)/scale,my=(mouse.cy-offY)/scale;
  const cc=energy<8?'#ff5d6c':'#7df5ff';
  ctx.save();ctx.translate(mx,my);ctx.rotate(t*.8);
  ctx.strokeStyle=cc;ctx.lineWidth=1.6;ctx.globalAlpha=.9;
  ctx.beginPath();ctx.arc(0,0,9,0,TAU);ctx.stroke();
  for(let i=0;i<4;i++){const an=i*Math.PI/2;ctx.beginPath();ctx.moveTo(Math.cos(an)*11,Math.sin(an)*11);ctx.lineTo(Math.cos(an)*16,Math.sin(an)*16);ctx.stroke();}
  ctx.fillStyle=cc;ctx.beginPath();ctx.arc(0,0,1.8,0,TAU);ctx.fill();
  ctx.restore();ctx.globalAlpha=1;

  // ---- screen space: CRT finish ----
  ctx.setTransform(dpr,0,0,dpr,0,0);
  ctx.fillStyle=vigGrad;ctx.fillRect(0,0,fw,fh);
  ctx.globalAlpha=.28;ctx.fillStyle=scanPat;ctx.fillRect(0,0,fw,fh);ctx.globalAlpha=1;
  if(flashT>0){ctx.fillStyle='rgba('+flashCol+','+Math.min(.45,flashT)+')';ctx.fillRect(0,0,fw,fh);}
}
