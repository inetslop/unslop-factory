/* ---------- update ---------- */
function ballSpeed(){return Math.min(340+(S.wave-1)*16,560)*slowScale;}
function update(dt){
  T+=dt;
  // ambient always
  for(const f of fogB.concat(fogF)){f.x+=f.sp*dt;if(f.x>GW+320)f.x=-320;}
  batT-=dt;if(batT<=0){batT=rnd(4,9);const dir=Math.random()<.5?1:-1;bats.push({x:dir>0?-30:GW+30,y:rnd(60,200),vx:dir*rnd(70,120),ph:rnd(TAU)});}
  for(const b of bats)b.x+=b.vx*dt;bats=bats.filter(b=>b.x>-60&&b.x<GW+60);
  lightT-=dt;if(lightT<=0){lightT=rnd(8,17);S.flash=1;sLight();}
  S.flash=Math.max(0,S.flash-dt*2.2);
  S.shake=Math.max(0,S.shake-dt*46);
  S.redFlash=Math.max(0,S.redFlash-dt*1.6);
  for(const p of parts){p.t+=dt;p.vy+=p.g*dt;p.x+=p.vx*dt;p.y+=p.vy*dt;if(p.rot!==undefined)p.rot+=p.vr*dt;}
  parts=parts.filter(p=>p.t<p.dur);
  for(const p of pops){p.t+=dt;p.y-=42*dt;}
  pops=pops.filter(p=>p.t<p.dur);
  for(const b of bricks)if(b.flash>0)b.flash=Math.max(0,b.flash-dt*5);

  if(S.mode!=='play')return;
  if(S.banner){S.banner.t+=dt;if(S.banner.t>=S.banner.dur){const cb=S.banner.cb;S.banner=null;cb&&cb();}return;}

  // paddle
  const pw=pad.w/2;let vx=0;
  if(keys['arrowleft']||keys['a'])vx-=1;
  if(keys['arrowright']||keys['d'])vx+=1;
  if(vx!==0){pad.x+=vx*640*dt;pad.tx=null;}
  else if(pad.tx!=null){const nx=pad.x+(clamp(pad.tx,pw,GW-pw)-pad.x)*Math.min(1,dt*18);pad.vx=nx-pad.x;pad.x=nx;}
  pad.x=clamp(pad.x,pw,GW-pw);
  const twt=eff.bite>0?78:eff.wide>0?156:110;pad.tw=twt;pad.w+=(twt-pad.w)*Math.min(1,dt*8);
  eff.wide=Math.max(0,eff.wide-dt);eff.slow=Math.max(0,eff.slow-dt);eff.bite=Math.max(0,eff.bite-dt);
  slowScale+=((eff.slow>0?.62:1)-slowScale)*Math.min(1,dt*5);

  // balls
  const R=7,sp=ballSpeed()*dt;
  for(let bi=balls.length-1;bi>=0;bi--){
    const b=balls[bi];
    if(b.stuck){b.x=pad.x;b.y=pad.y-R-2;b.trail.length=0;continue;}
    const mag=Math.hypot(b.dx,b.dy)||1;b.dx/=mag;b.dy/=mag;
    if(Math.abs(b.dy)<.22){const s=b.dy<0?-1:1;b.dy=s*.24;const nx=Math.sqrt(1-b.dy*b.dy)*(b.dx<0?-1:1);b.dx=nx;}
    const steps=Math.max(1,Math.ceil(sp/9)),step=sp/steps;
    let dead=false;
    for(let s2=0;s2<steps&&!dead;s2++){
      b.x+=b.dx*step;b.y+=b.dy*step;
      if(b.x<R){b.x=R;b.dx=Math.abs(b.dx);sWall();}
      else if(b.x>GW-R){b.x=GW-R;b.dx=-Math.abs(b.dx);sWall();}
      if(b.y<R){b.y=R;b.dy=Math.abs(b.dy);sWall();}
      // bricks
      for(const br of bricks){
        if(!br.alive)continue;
        const nx=clamp(b.x,br.x,br.x+br.w),ny=clamp(b.y,br.y,br.y+br.h);
        let ddx=b.x-nx,ddy=b.y-ny,d2=ddx*ddx+ddy*ddy;
        if(d2>R*R)continue;
        let mx,my;
        if(d2<1e-6){const oL=b.x-br.x,oR=br.x+br.w-b.x,oT=b.y-br.y,oB=br.y+br.h-b.y,m=Math.min(oL,oR,oT,oB);if(m===oT){mx=0;my=-1;}else if(m===oB){mx=0;my=1;}else if(m===oL){mx=-1;my=0;}else{mx=1;my=0;}}
        else{const d=Math.sqrt(d2);mx=ddx/d;my=ddy/d;}
        b.x=nx+mx*R;b.y=ny+my*R;
        const dot=b.dx*mx+b.dy*my;if(dot<0){b.dx-=2*dot*mx;b.dy-=2*dot*my;}
        if(T-b.hitT>.06||br.type===3){b.hitT=T;b.hitBr=br;damageBrick(br);}
      }
      // paddle
      if(b.dy>0&&b.y+R>=pad.y&&b.y-R<=pad.y+pad.h&&Math.abs(b.x-pad.x)<=pad.w/2+R){
        b.y=pad.y-R;const rel=clamp((b.x-pad.x)/(pad.w/2),-1,1),th=rel*.95;
        b.dx=Math.sin(th);b.dy=-Math.cos(th);
        S.combo=0;S.mult=1;sPaddle();
      }
      if(b.y>GH+30)dead=true;
    }
    if(dead){balls.splice(bi,1);if(balls.length===0)loseLife();continue;}
    b.trail.push({x:b.x,y:b.y});if(b.trail.length>9)b.trail.shift();
  }

  // capsules
  for(let i=caps.length-1;i>=0;i--){const c=caps[i];c.y+=c.vy*dt;c.x+=Math.sin(T*3+c.ph)*22*dt;
    if(Math.abs(c.x-pad.x)<pad.w/2+20&&c.y>pad.y-14&&c.y<pad.y+pad.h+18){applyPow(c.type,c.x,c.y);caps.splice(i,1);}
    else if(c.y>GH+24)caps.splice(i,1);}

  // wave clear
  if(S.left<=0){
    S.wave++;sWave();
    const nw=S.wave;
    S.banner={t:0,dur:1.7,main:'WAVE '+nw,sub:nw%3===0?'THEY REMEMBER YOU':'THEY ONLY GET FASTER',cb:()=>{buildWave(nw);resetBall();}};
  }
}
