 const dist=new Map([[key(u.x,u.z),0]]),prev=new Map(),q=[[u.x,u.z]];
 while(q.length){const[x,z]=q.shift(),d=dist.get(key(x,z));if(d>=u.mv)continue;
  for(let dx=-1;dx<=1;dx++)for(let dz=-1;dz<=1;dz++){
   if(!dx&&!dz)continue;const nx=x+dx,nz=z+dz;if(rockAt(nx,nz))continue;
   if(dx&&dz&&(rockAt(x+dx,z)||rockAt(x,z+dz)))continue;
   const occ=unitAt(nx,nz);if(occ&&occ.team!==u.team)continue;
   const k=key(nx,nz);if(dist.has(k))continue;dist.set(k,d+1);prev.set(k,[x,z]);q.push([nx,nz]);}}
 return{dist,prev};}
function pathTo(u,tx,tz,prev){const cells=[];let k=key(tx,tz);while(k!==key(u.x,u.z)){const p=prev.get(k);if(!p)break;cells.unshift(p);k=key(p[0],p[1]);}return cells;}

function atkOptionsAt(u,x,z){return units.filter(v=>!v.dead&&v.team!==u.team&&cheb(x,z,v.x,v.z)<=u.rng&&los(x,z,v.x,v.z));}
const atkOptions=u=>atkOptionsAt(u,u.x,u.z);
function calcDmg(a,d,fromH){return Math.max(1,a.atk-d.def+(fromH>hAt(d.x,d.z)?2:0));}
function counterDmg(d,a){return Math.max(1,d.atk-a.def);}

/* ---------- selection / input flow ---------- */
function selectUnit(u){sel=u;phaseB=false;const{dist}=reachable(u);moveTiles=new Map();
 for(const[k,v]of dist){if(v===0)continue;const x=kx(k),z=kz(k);if(unitAt(x,z))continue;moveTiles.set(k,v);}
 atkTargets=atkOptions(u);Snd.play('select');updateCard();}
function deselect(){sel=null;moveTiles=null;atkTargets=null;phaseB=false;updateCard();}
function passSelected(){if(state!=='player')return;if(sel&&phaseB){finishUnitAction(sel);}else if(sel&&!phaseB){deselect();}}

function afterMove(u){const opts=atkOptions(u);if(opts.length){phaseB=true;moveTiles=null;atkTargets=opts;toast('CHOOSE A RED TARGET — SPACE TO REST');updateCard();}else finishUnitAction(u);}
function finishUnitAction(u){u.acted=true;sel=null;moveTiles=null;atkTargets=null;phaseB=false;updateCard();
 if(state!=='player')return;
 if(units.every(v=>v.team!==0||v.acted||v.dead))schedule(700,endPlayerTurn);}

function onTap(sx,sy){
 if(state!=='player')return;
 if(sel&&phaseB){const t=pickTarget(sx,sy);if(t){doAttack(sel,t,()=>finishUnitAction(sel));}else{toast('UNIT RESTS');passSelected();}return;}
 if(sel&&!phaseB){
  const t=pickTarget(sx,sy);
  if(t&&atkTargets.includes(t)){doAttack(sel,t,()=>finishUnitAction(sel));return;}
  const tile=pickTile(sx,sy);
  if(tile&&moveTiles.has(key(tile.x,tile.z))){const{prev}=reachable(sel);startMove(sel,tile.x,tile.z,pathTo(sel,tile.x,tile.z,prev),()=>afterMove(sel));return;}
  const u=pickUnit(sx,sy);
  if(u&&u.team===0&&!u.acted&&u!==sel){selectUnit(u);return;}
  deselect();return;}
 const u=pickUnit(sx,sy);
 if(u){if(u.team===0){if(u.acted)toast('UNIT ALREADY ACTED');else selectUnit(u);}
  else toast(u.name+' · '+CLASSES[u.cls].name+' — ATK'+u.atk+' DEF'+u.def+' HP'+u.hp);}
 else deselect();
}

function doAttack(att,def,onDone){

  def.hp-=dmg;def.flash=1;Snd.play('hit');shake=Math.min(.65,shake+(high?.32:.22));
  addFloater([def.px,def.gy+1.2,def.pz],'-'+dmg+(high?' ▲':''),att.team===0?'#ffd98a':'#ff7a5c');
  if(def.hp<=0)killUnit(def);
  else if(cheb(att.x,att.z,def.x,def.z)<=def.rng&&los(def.x,def.z,att.x,att.z)){
   schedule(340,()=>{if(state==='over')return;const a2=Math.atan2(att.pz-def.pz,att.px-def.px);def.fy=a2;def.tfy=a2;
    Snd.play('whoosh');def.anim={type:'lunge',dx:Math.cos(a2)*.3,dz:Math.sin(a2)*.3,t:0};
    schedule(150,()=>{if(state==='over')return;const cd=counterDmg(def,att);att.hp-=cd;att.flash=1;Snd.play('hit');shake=Math.min(.65,shake+.2);
     addFloater([att.px,att.gy+1.2,att.pz],'-'+cd,'#9fe8ff');
     if(att.hp<=0)killUnit(att);});});}});

/* ---------- enemy AI ---------- */
function aiAct(u,done){
 const foes=units.filter(v=>v.team===0&&!v.dead);if(!foes.length)return done();
 let target=foes[0],bd=1e9;for(const f of foes){const d=cheb(u.x,u.z,f.x,f.z)+f.hp/24;if(d<bd){bd=d;target=f;}}
 const nowOpts=atkOptionsAt(u,u.x,u.z);
 if(nowOpts.length){let best=nowOpts[0],bs=-1;for(const o of nowOpts){const dm=calcDmg(u,o,hAt(u.x,u.z));const sc=dm+(dm>=o.hp?60:0)-cheb(u.x,u.z,o.x,o.z)*.2;if(sc>bs){bs=sc;best=o;}}
  return doAttack(u,best,done);}
 const{dist,prev}=reachable(u);
 let best=null,bs=-1;
 for(const[k,d]of dist){if(d===0)continue;const x=kx(k),z=kz(k);if(unitAt(x,z))continue;
  for(const o of atkOptionsAt(u,x,z)){const dm=calcDmg(u,o,hAt(x,z));const sc=dm+(dm>=o.hp?60:0)-d*.4;if(sc>bs){bs=sc;best={x,z,t:o};}}}
 if(best){startMove(u,best.x,best.z,pathTo(u,best.x,best.z,prev),()=>doAttack(u,best.t,done));return;}
 let bt=null,bts=1e9;
 for(const[k,d]of dist){if(d===0)continue;const x=kx(k),z=kz(k);if(unitAt(x,z))continue;
  const s=cheb(x,z,target.x,target.z)-hAt(x,z)*.35+Math.random()*.35;if(s<bts){bts=s;bt={x,z};}}
 if(bt&&!(bt.x===u.x&&bt.z===u.z))startMove(u,bt.x,bt.z,pathTo(u,bt.x,bt.z,prev),done);else done();}

/* ================= 3D CAMERA / PROJECTION ================= */
function camBasis(){
 let tx=cam.target[0],ty=cam.target[1],tz=cam.target[2];
 if(shake>0){tx+=(Math.random()-.5)*shake*.4;ty+=(Math.random()-.5)*shake*.3;tz+=(Math.random()-.5)*shake*.4;}
 const cp=Math.cos(cam.pitch),sp=Math.sin(cam.pitch),cyw=Math.cos(cam.yaw),syw=Math.sin(cam.yaw);
 const ex=tx+cam.dist*cp*syw,ey=ty+cam.dist*sp,ez=tz+cam.dist*cp*cyw;
 let fx=tx-ex,fy=ty-ey,fz=tz-ez;const fl2=Math.hypot(fx,fy,fz);fx/=fl2;fy/=fl2;fz/=fl2;
 const rl=Math.hypot(fx,fz)||1e-6,r=[-fz/rl,0,fx/rl];
 const u=[r[1]*fz-r[2]*fy,r[2]*fx-r[0]*fz,r[0]*fy-r[1]*fx];
 return{e:[ex,ey,ez],f:[fx,fy,fz],r,u,fl:(cvH/2)/Math.tan(FOV/2)};}
function proj(p,b){const dx=p[0]-b.e[0],dy=p[1]-b.e[1],dz=p[2]-b.e[2];
 const cz=dx*b.f[0]+dy*b.f[1]+dz*b.f[2];if(cz<.35)return null;
 const cx=dx*b.r[0]+dy*b.r[1]+dz*b.r[2],cyv=dx*b.u[0]+dy*b.u[1]+dz*b.u[2];
 return[cvW/2+cx*b.fl/cz,cvH/2-cyv*b.fl/cz,cz];}

function update(dt){
 const due=[];scheduleQ=scheduleQ.filter(s=>{if(s.t<=performance.now()){due.push(s.fn);return false;}return true;});
 due.forEach(f=>f());
 if(state==='menu')cam.yaw+=dt*.12;else if(state==='over')cam.yaw+=dt*.05;
 for(const u of units){if(u.dead)continue;
  u.flash=Math.max(0,u.flash-dt*3);
  let tfy=u.team===0?Math.PI/2:-Math.PI/2;
  if(u.anim&&u.anim.type==='move'){const to=u.anim.path[u.anim.i];tfy=Math.atan2(to[0]-(u.px-u.hop*0),to[1]-(u.pz));tfy=Math.atan2((to[0])-(u.px),(to[1])-(u.pz));}
  if(u.anim){const a=u.anim;a.t+=dt/a.dur;
   if(a.type==='move'){
    const to=a.path[a.i],fx=u.px-a._ox||u.px;
    // segment interp stored in a.fx0/fz0/gy0 at seg start
    const e=clamp(a.t,0,1),ee=e*e*(3-2*e);
    u.px=lerp(a.fx0,to[0],ee);u.pz=lerp(a.fz0,to[1],ee);
    u.gy=lerp(a.gy0,hAt(to[0],to[1]),e);u.hop=Math.sin(e*Math.PI)*.16;
    if(a.t>=1){Snd.play('step');a.i++;a.t=0;
     if(a.i>=a.path.length){u.x=a.tx;u.z=a.tz;u.px=a.tx+.5;u.pz=a.tz+.5;u.gy=hAt(u.x,u.z);u.hop=0;u.anim=null;a.cb&&a.cb();}
     else{const to2=a.path[a.i];a.fx0=u.px;a.fz0=u.pz;a.gy0=u.gy;}}
   }else if(a.type==='lunge'){if(a.t>=1)u.anim=null;}}
  const da=((tfy-u.fy+Math.PI*3)%(Math.PI*2))-Math.PI;u.fy+=da*Math.min(1,dt*7);}
 for(let i=particles.length-1;i>=0;i--){const p=particles[i];p.life-=dt*1.15;
  if(p.life<=0){particles.splice(i,1);continue;}
  p.v[1]-=6.5*dt;p.p[0]+=p.v[0]*dt;p.p[1]+=p.v[1]*dt;p.p[2]+=p.v[2]*dt;p.rot+=p.vr*dt;
  const gx=clamp(Math.floor(p.p[0]),0,W-1),gz=clamp(Math.floor(p.p[2]),0,H-1);
  if(p.p[1]<hAt(gx,gz)+p.s/2&&p.v[1]<0){p.v[1]*=-.35;p.v[0]*=.7;p.v[2]*=.7;}}
 for(let i=floaters.length-1;i>=0;i--){const f=floaters[i];f.t+=dt;f.p[1]+=dt*.8;if(f.t>1.1)floaters.splice(i,1);}
 shake=Math.max(0,shake-dt*1.4);
}
