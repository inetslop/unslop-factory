function hitFX(x,y,kind,imp){
  if(imp<40)return;
  var i=Math.min(1,imp/650),n=4+Math.round(i*8);
  var col=kind==='wall'?'#bfeee2':(kind==='pillar'?'#9fdcf2':'#cfeeff');
  burst(x,y,n,col,80+i*240,2.4,.4);
  if(imp>170)shakeAdd(Math.min(11,i*13));
  if(kind==='wall')sfx.wall(i);else if(kind==='pillar')sfx.pillar(i);else sfx.bar(i);
}
function collideWalls(){
  var R=puck.r;
  if(puck.x<B.x+R){puck.x=B.x+R;if(puck.vx<0){var i1=-puck.vx;puck.vx=-puck.vx*WALL_REST;hitFX(B.x,puck.y,'wall',i1);}}
  if(puck.x>B.x+B.w-R){puck.x=B.x+B.w-R;if(puck.vx>0){var i2=puck.vx;puck.vx=-puck.vx*WALL_REST;hitFX(B.x+B.w,puck.y,'wall',i2);}}
  if(puck.y<B.y+R){puck.y=B.y+R;if(puck.vy<0){var i3=-puck.vy;puck.vy=-puck.vy*WALL_REST;hitFX(puck.x,B.y,'wall',i3);}}
  if(puck.y>B.y+B.h-R){puck.y=B.y+B.h-R;if(puck.vy>0){var i4=puck.vy;puck.vy=-puck.vy*WALL_REST;hitFX(puck.x,B.y+B.h,'wall',i4);}}
}
function collideCircles(list,rest,isBumper){
  for(var j=0;j<list.length;j++){var c=list[j];
    var dx=puck.x-c.x,dy=puck.y-c.y,d=Math.sqrt(dx*dx+dy*dy),min=c.r+puck.r;
    if(d<min&&d>0.0001){
      var nx=dx/d,ny=dy/d;
      puck.x=c.x+nx*min;puck.y=c.y+ny*min;
      var vn=puck.vx*nx+puck.vy*ny;
      if(vn<0){puck.vx-=(1+rest)*vn*nx;puck.vy-=(1+rest)*vn*ny;hitFX(c.x+nx*c.r,c.y+ny*c.r,isBumper?'pillar':'pillar',-vn);}
      if(isBumper&&c.cool<=0){
        c.cool=.3;c.flash=1;
        puck.vx+=nx*320;puck.vy+=ny*320;
        var sp=Math.hypot(puck.vx,puck.vy);if(sp>1350){puck.vx*=1350/sp;puck.vy*=1350/sp;}
        if(state==='playing'){score+=25;fText(c.x,c.y-c.r-14,'+25','#ffd23f',17);updateHUD();}
        ringFx(c.x,c.y,'#ffd23f',c.r*.6,c.r*2.4,.4);burst(c.x,c.y,12,'#ffd23f',300,2.8,.45);
        sfx.bumper();shakeAdd(8);
      }
    }
  }
}
function collideBars(){
  for(var j=0;j<bars.length;j++){var b=bars[j];
    var ca=Math.cos(b.a),sa=Math.sin(b.a);
    var ax=b.x-ca*b.hl,ay=b.y-sa*b.hl,bx=b.x+ca*b.hl,by=b.y+sa*b.hl;
    var abx=bx-ax,aby=by-ay;
    var t=((puck.x-ax)*abx+(puck.y-ay)*aby)/(abx*abx+aby*aby);
    t=Math.max(0,Math.min(1,t));
    var cx=ax+abx*t,cy=ay+aby*t;
    var dx=puck.x-cx,dy=puck.y-cy,d=Math.sqrt(dx*dx+dy*dy),min=BAR_R+puck.r;
    if(d<min){
      if(d<0.0001){dx=1;dy=0;d=1;}
      var nx=dx/d,ny=dy/d;
      puck.x=cx+nx*min;puck.y=cy+ny*min;
      var rx=cx-b.x,ry=cy-b.y,svx=-b.om*ry,svy=b.om*rx;
      var rvx=puck.vx-svx,rvy=puck.vy-svy,vn=rvx*nx+rvy*ny;
      if(vn<0){puck.vx-=1.95*vn*nx;puck.vy-=1.95*vn*ny;hitFX(cx,cy,'bar',-vn);}
    }
  }
}
function collideHoles(h){
  for(var j=0;j<holes.length;j++){var hl=holes[j];
    var dx=hl.x-puck.x,dy=hl.y-puck.y,d2=dx*dx+dy*dy,R=hl.r+puck.r+26;
    if(d2<R*R){
      var d=Math.sqrt(d2)||0.001,st=1-d/R;
      puck.vx+=dx/d*st*1500*h;puck.vy+=dy/d*st*1500*h;
      if(d<hl.r*.72){startSuck(hl);return;}
    }
  }
}
function collideOrbs(){
  for(var j=0;j<orbs.length;j++){var o=orbs[j];
    if(!o.alive)continue;
    var dx=puck.x-o.x,dy=puck.y-o.y,rr=(puck.r+ORB_R);
    if(dx*dx+dy*dy<rr*rr){
      o.alive=false;combo++;var val=100*combo;
      if(state==='playing'){score+=val;runOrbs++;updateHUD();}
      fText(o.x,o.y-20,combo>1?('+'+val+' ×'+combo):('+'+val),'#6effc9',combo>1?24:19);
      burst(o.x,o.y,16,'#6effc9',300,3,.5);ringFx(o.x,o.y,'#a8ffe0',6,72,.45);
      sfx.orb(Math.min(combo-1,6));shakeAdd(2);
    }
  }
}
function startSuck(hl){puck.sucked=true;puck.suckT=0;puck.hole=hl;sfx.hole();shakeAdd(14);burst(hl.x,hl.y,18,'#ff5470',260,3,.5);}
function respawnPuck(){puck.x=startX;puck.y=startY;puck.vx=puck.vy=0;puck.trail.length=0;}
function suckUpdate(dt){
  puck.suckT+=dt;var h=puck.hole;
  var dx=h.x-puck.x,dy=h.y-puck.y,d=Math.hypot(dx,dy)||0.001;
  puck.vx=dx/d*950;puck.vy=dy/d*950;
  puck.x+=puck.vx*dt;puck.y+=puck.vy*dt;puck.theta+=dt*22;
  for(var i=0;i<2;i++){var a=Math.random()*Math.PI*2,r=h.r+8+Math.random()*14;
    parts.push({x:h.x+Math.cos(a)*r,y:h.y+Math.sin(a)*r,vx:Math.cos(a+Math.PI)*320,vy:Math.sin(a+Math.PI)*320,col:'#ff5470',size:2.5,life:.4,max:.4});}
  if(d<10||puck.suckT>0.5){
    puck.sucked=false;
    if(state==='playing'){lives--;updateHUD();if(lives<=0)doGameOver();else respawnPuck();}
    else respawnPuck();
  }
}
function physicsStep(dt){
  var sp=Math.hypot(puck.vx,puck.vy);
  if(sp>0){var f=Math.exp(-FRICTION*dt);puck.vx*=f;puck.vy*=f;}
  var n=Math.min(12,Math.max(1,Math.ceil(sp*dt/(puck.r*.8)))),h=dt/n;
  for(var i=0;i<n;i++){
    puck.x+=puck.vx*h;puck.y+=puck.vy*h;
    puck.theta+=Math.hypot(puck.vx,puck.vy)*h/puck.r;
    collideWalls();
    collideHoles(h);if(puck.sucked)return;
    collideOrbs();collideCircles(pillars,.92,false);collideCircles(bumpers,.85,true);collideBars();
  }
}

  }
}

  return{m:m,pull:pull,vx:vx2,vy:vy2,dx:dx/m,dy:dy/m};
}

function predictPath(vx,vy){
  var pts=[],x=puck.x,y=puck.y,b=0,h=1/60,f=Math.exp(-FRICTION*h),R=puck.r;
  for(var s=0;s<84&&b<5;s++){
    vx*=f;vy*=f;if(Math.hypot(vx,vy)<70)break;
    x+=vx*h;y+=vy*h;
    if(x<B.x+R){x=B.x+R;vx=-vx*WALL_REST;b++;}
    if(x>B.x+B.w-R){x=B.x+B.w-R;vx=-vx*WALL_REST;b++;}
    if(y<B.y+R){y=B.y+R;vy=-vy*WALL_REST;b++;}
    if(y>B.y+B.h-R){y=B.y+B.h-R;vy=-vy*WALL_REST;b++;}
    var list=pillars.concat(bumpers),j;
    for(j=0;j<list.length;j++){var c=list[j],dx=x-c.x,dy=y-c.y,d=Math.sqrt(dx*dx+dy*dy),min=c.r+R;
      if(d<min&&d>0.001){var nx=dx/d,ny=dy/d;x=c.x+nx*min;y=c.y+ny*min;var vn=vx*nx+vy*ny;if(vn<0){vx-=1.9*vn*nx;vy-=1.9*vn*ny;}}}
    if(s%3===0)pts.push([x,y]);
  }
  return pts;
}

/* ================= UPDATE ================= */
function update(dt){
  time+=dt;introT-=dt;
  var j;
  for(j=0;j<bars.length;j++)bars[j].a+=bars[j].om*dt;
  for(j=0;j<bumpers.length;j++){var bp=bumpers[j];bp.cool=Math.max(0,bp.cool-dt);bp.flash=Math.max(0,bp.flash-dt*2.5);}

  if(state==='menu')demoAI(dt);
  if((state==='playing'||state==='menu')){
    if(puck.sucked)suckUpdate(dt);else physicsStep(dt);
  }

  var sp=Math.hypot(puck.vx,puck.vy);
  if((state==='playing'||state==='menu')&&!puck.sucked){
    if(sp>50){puck.trail.push({x:puck.x,y:puck.y});if(puck.trail.length>34)puck.trail.shift();
      if(sp>520&&Math.random()<.5)parts.push({x:puck.x-puck.vx/sp*14,y:puck.y-puck.vy/sp*14,vx:-puck.vx/sp*60,vy:-puck.vy/sp*60,col:'rgba(255,200,150,.5)',size:2.2,life:.3,max:.3});
    }else if(puck.trail.length)puck.trail.splice(0,2);
  }

  if(state==='playing'&&!puck.sucked){
    if(sp<6){
      puck.vx=puck.vy=0;
      ready=(throwsLeft>0&&orbs.some(function(o){return o.alive;}));
      if(throwsLeft<=0&&orbs.some(function(o){return o.alive;})){failT+=dt;if(failT>.7)doFail();}
    }else ready=false;
    if(orbs.length&&!orbs.some(function(o){return o.alive;})&&clearT<0){
      clearT=0;lastBonus=throwsLeft*150;score+=lastBonus;sfx.clear();shakeAdd(6);
      burst(puck.x,puck.y,26,'#6effc9',340,3,.7);ringFx(puck.x,puck.y,'#a8ffe0',10,120,.6);updateHUD();
    }
    if(clearT>=0){clearT+=dt;if(clearT>1.15){if(cur===LEVELS.length-1)doWin();else showClearOv();}}
  }

  for(j=parts.length-1;j>=0;j--){var p=parts[j];p.life-=dt;
    if(p.life<=0){parts.splice(j,1);continue;}
    if(!p.ring){p.x+=p.vx*dt;p.y+=p.vy*dt;var dr=Math.exp(-3.2*dt);p.vx*=dr;p.vy*=dr;}}
  for(j=floats.length-1;j>=0;j--){var fl=floats[j];fl.life-=dt*1.05;fl.y-=46*dt;if(fl.life<=0)floats.splice(j,1);}
  shake*=Math.exp(-7*dt);if(shake<.1)shake=0;

  var h='';
  if(state==='playing'){
    if(drag&&aimResult())h='RELEASE TO FLICK';
    else if(ready)h='PULL BACK & RELEASE — OR FLICK MID-ROLL TO REDIRECT';
    else if(sp>25)h='ROLLING…';
  }
  if(h!==lastHint){hintEl.textContent=h;hintEl.classList.toggle('on',!!h);lastHint=h;}
}

    var g=ctx.createRadialGradient(h.x,h.y,2,h.x,h.y,h.r);
    g.addColorStop(0,'#010a0e');g.addColorStop(.8,'#04171f');g.addColorStop(1,'#0a2531');
    circle(h.x,h.y,h.r);ctx.fillStyle=g;ctx.fill();
    ctx.save();ctx.setLineDash([9,7]);ctx.lineDashOffset=-time*40;
    circle(h.x,h.y,h.r+6);ctx.strokeStyle=near?'rgba(255,84,112,.95)':'rgba(255,84,112,.6)';ctx.lineWidth=near?4:3;ctx.stroke();ctx.restore();
    circle(h.x,h.y,h.r-4);ctx.fillStyle='rgba(255,84,112,'+(.07+.06*Math.sin(time*3+j))+')';ctx.fill();
  }
}
function drawOrbs(){
  for(var j=0;j<orbs.length;j++){var o=orbs[j];if(!o.alive)continue;
    var pulse=1+.1*Math.sin(time*3+o.ph),r=ORB_R*pulse;
    var g=ctx.createRadialGradient(o.x,o.y,1,o.x,o.y,r*2.5);
    g.addColorStop(0,'rgba(110,255,201,.4)');g.addColorStop(1,'rgba(110,255,201,0)');
    circle(o.x,o.y,r*2.5);ctx.fillStyle=g;ctx.fill();
    var bg=ctx.createRadialGradient(o.x-r*.3,o.y-r*.3,1,o.x,o.y,r);
    bg.addColorStop(0,'#eafff6');bg.addColorStop(.55,'#7df3cc');bg.addColorStop(1,'#3ecb9c');

    circle(o.x,o.y,r);ctx.strokeStyle='rgba(234,255,246,.85)';ctx.lineWidth=1.5;ctx.stroke();

  }
}

function drawBars(){
  ctx.lineCap='round';
  for(var j=0;j<bars.length;j++){var b=bars[j];
    var ca=Math.cos(b.a),sa=Math.sin(b.a);
    var ax=b.x-ca*b.hl,ay=b.y-sa*b.hl,bx=b.x+ca*b.hl,by=b.y+sa*b.hl;
    ctx.beginPath();ctx.moveTo(ax,ay);ctx.lineTo(bx,by);ctx.strokeStyle='#2b6f96';ctx.lineWidth=BAR_R*2;ctx.stroke();
    ctx.beginPath();ctx.moveTo(ax,ay);ctx.lineTo(bx,by);ctx.strokeStyle='#3f8fbf';ctx.lineWidth=BAR_R*1.3;ctx.stroke();
    var ix=ax+(bx-ax)*.16,iy=ay+(by-ay)*.16,jx=bx-(bx-ax)*.16,jy=by-(by-ay)*.16;
    ctx.beginPath();ctx.moveTo(ix,iy);ctx.lineTo(jx,jy);ctx.strokeStyle='rgba(220,250,255,.8)';ctx.lineWidth=3;ctx.stroke();
    circle(b.x,b.y,7);ctx.fillStyle='#0c2833';ctx.fill();
    circle(b.x,b.y,7);ctx.strokeStyle='#6fd2ff';ctx.lineWidth=2;ctx.stroke();
  }
}

  ctx.restore();
}

  var hg=ctx.createRadialGradient(puck.x,puck.y,1,puck.x,puck.y,r*2.8);

function drawAim(){
  var a=aimResult();if(!a)return;
  ctx.save();ctx.setLineDash([5,7]);ctx.strokeStyle='rgba(234,255,246,.2)';ctx.lineWidth=1.5;

  var col=frac<.5?lerpC('#6effc9','#ffd23f',frac*2):lerpC('#ffd23f','#ff5470',(frac-.5)*2);
  var L=44+frac*120,ax=puck.x+a.dx*(puck.r+6),ay=puck.y+a.dy*(puck.r+6),ex=ax+a.dx*L,ey=ay+a.dy*L;
  ctx.strokeStyle=col;ctx.lineWidth=5;ctx.lineCap='round';
  ctx.beginPath();ctx.moveTo(ax,ay);ctx.lineTo(ex,ey);ctx.stroke();

  ctx.beginPath();ctx.moveTo(ex+Math.cos(ang)*14,ey+Math.sin(ang)*14);

  ctx.lineTo(ex+Math.cos(ang-2.5)*10,ey+Math.sin(ang-2.5)*10);ctx.closePath();ctx.fillStyle=col;ctx.fill();

}
