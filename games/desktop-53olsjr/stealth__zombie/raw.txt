```js
function collideMove(e,dx,dy){
  e.x+=dx; resolveCircle(e); e.y+=dy; resolveCircle(e);
}
function resolveCircle(e){
  const r=e.r;
  const x0=Math.floor((e.x-r)/T),x1=Math.floor((e.x+r)/T),y0=Math.floor((e.y-r)/T),y1=Math.floor((e.y+r)/T);
  for(let ty=y0;ty<=y1;ty++)for(let tx=x0;tx<=x1;tx++){
    if(!solidTile(tx,ty))continue;
    const cx=clamp(e.x,tx*T,tx*T+T), cy=clamp(e.y,ty*T,ty*T+T);
    let dx=e.x-cx, dy=e.y-cy; const d2=dx*dx+dy*dy;
    if(d2>=r*r)continue;
    if(d2<1e-6){ // center inside: nudge out along smallest axis distance
      const l=e.x-tx*T, rr=(tx+1)*T-e.x, t=e.y-ty*T, b=(ty+1)*T-e.y;
      const m=Math.min(l,rr,t,b);
      if(m===l)e.x=tx*T-r; else if(m===rr)e.x=(tx+1)*T+r; else if(m===t)e.y=ty*T-r; else e.y=(ty+1)*T+r;
    } else { const d=Math.sqrt(d2), push=r-d; e.x+=dx/d*push; e.y+=dy/d*push; }
  }
}
```

    cur=bi;
  }
  const props=new Set();
  for(let y=1;y<MH-1;y++)for(let x=1;x<MW-1;x++) if(grid[y*MW+x]===0&&rnd()<0.05) props.add(y*MW+x);
  const spawn={x:rooms[0].cx*T+T/2,y:rooms[0].cy*T+T/2};
  let ei=1,bd=-1;
  for(let i=1;i<rooms.length;i++){ const d=(rooms[i].cx-rooms[0].cx)**2+(rooms[i].cy-rooms[0].cy)**2; if(d>bd){bd=d;ei=i;} }
  const ext={x:rooms[ei].cx*T+T/2,y:rooms[ei].cy*T+T/2};
  const solid=(px,py)=>{ const tx=Math.floor(px/T),ty=Math.floor(py/T);
    return tx<0||ty<0||tx>=MW||ty>=MH||grid[ty*MW+tx]===1||props.has(ty*MW+tx); };
  const clearArea=(cx,cy,rT)=>{ for(const k of[...props]){ const tx=k%MW,ty=(k/MW)|0;
    if(Math.abs(tx*T+T/2-cx)<rT*T&&Math.abs(ty*T+T/2-cy)<rT*T)props.delete(k); } };
  clearArea(spawn.x,spawn.y,2); clearArea(ext.x,ext.y,2);
  const zs=[], others=rooms.slice(1);
  for(let k=0;k<L.z;k++){
    let x=null,y=null,tr=0;
    while(tr<50){ tr++;
      if(k<others.length){ const r=others[k];
        x=(r.x+1+((rnd()*Math.max(1,r.w-2))|0))*T+T/2; y=(r.y+1+((rnd()*Math.max(1,r.h-2))|0))*T+T/2; }
      else{ const tx=2+((rnd()*(MW-4))|0),ty=2+((rnd()*(MH-4))|0); if(grid[ty*MW+tx]!==0)continue; x=tx*T+T/2;y=ty*T+T/2; }
      if(!solid(x,y)&&Math.hypot(x-spawn.x,y-spawn.y)>250)break; x=null;
    }
    if(x===null){ x=rooms[1].cx*T+T/2; y=rooms[1].cy*T+T/2; }
    clearArea(x,y,1); zs.push({x,y});
  }
  const lamps=[];
  for(let i=1;i<rooms.length;i++){ const r=rooms[i];
    const tx=r.x+1+((rnd()*Math.max(1,r.w-2))|0), ty=r.y+1+((rnd()*Math.max(1,r.h-2))|0);
    if(grid[ty*MW+tx]===0&&!props.has(ty*MW+tx)) lamps.push({x:tx*T+T/2,y:ty*T+T/2,ph:rnd()*TAU}); }
  return {grid,props,lamps,rooms,spawn,ext,zs};
}

function losClear(x0,y0,x1,y1){ const dx=x1-x0,dy=y1-y0,d=Math.hypot(dx,dy),n=Math.ceil(d/8);

function resolveCircle(e){ const r=e.r;
  const x0=Math.floor((e.x-r)/T),x1=Math.floor((e.x+r)/T),y0=Math.floor((e.y-r)/T),y1=Math.floor((e.y+r)/T);
  for(let ty=y0;ty<=y1;ty++)for(let tx=x0;tx<=x1;tx++){
    if(!solidTile(tx,ty))continue;
    const cx=clamp(e.x,tx*T,tx*T+T), cy=clamp(e.y,ty*T,ty*T+T);
    let dx=e.x-cx, dy=e.y-cy; const d2=dx*dx+dy*dy; if(d2>=r*r)continue;
    if(d2<1e-6){ const l=e.x-tx*T,rr=(tx+1)*T-e.x,t=e.y-ty*T,b=(ty+1)*T-e.y,m=Math.min(l,rr,t,b);
      if(m===l)e.x=tx*T-r; else if(m===rr)e.x=(tx+1)*T+r; else if(m===t)e.y=ty*T-r; else e.y=(ty+1)*T+r; }
    else{ const d=Math.sqrt(d2),push=r-d; e.x+=dx/d*push; e.y+=dy/d*push; } } }

function sightCheck(z,d){ const p=G.player;
  const range=175*(p.lightOn?1.42:1)*(1-G.L.fog*0.18);
  if(d>range)return{vis:false};
  if(p.crouch&&d<78)return{vis:false};
  const fa=Math.atan2(p.y-z.y,p.x-z.x); let da=fa-z.a;
  while(da>Math.PI)da-=TAU; while(da<-Math.PI)da+=TAU;
  if(Math.abs(da)>0.875+(d<64?0.45:0))return{vis:false};
  if(!losClear(z.x,z.y,p.x,p.y))return{vis:false};
  return{vis:true,prox:1-d/range}; }
function steerAng(z,ang){ for(const off of[0,0.55,-0.55,1.1,-1.1]){ const a=ang+off;
  if(!solidAtPx(z.x+Math.cos(a)*26,z.y+Math.sin(a)*26))return a; } return ang; }
function pickWander(z){ for(let i=0;i<12;i++){ const a=Math.random()*TAU,d=40+Math.random()*130;
  const x=z.home.x+Math.cos(a)*d,y=z.home.y+Math.sin(a)*d; if(!solidAtPx(x,y)){ z.tx=x; z.ty=y; return; } }
  z.tx=z.home.x; z.ty=z.home.y; }
function updateZombie(z,dt){
  const p=G.player, playing=(mode==='play');
  const d=Math.hypot(p.x-z.x,p.y-z.y);
  if(playing){
    const s=sightCheck(z,d);
    if(s.vis){ z.suspicion+=dt*(0.5+1.7*s.prox)*(p.crouch?0.5:1)*(p.lightOn?1.3:1); z.lsx=p.x; z.lsy=p.y; z.lostT=0; }
    else if(z.state==='chase'){ z.lostT+=dt; if(z.lostT>2.4){ z.state='investigate'; z.tx=z.lsx; z.ty=z.lsy; z.waitT=0; z.suspicion=0.3; } }
    else z.suspicion=Math.max(0,z.suspicion-dt*0.28);
    if(d<22)z.suspicion+=dt*3;
    if(z.suspicion>=1&&z.state!=='chase'){ z.state='chase'; z.lostT=0; G.chases++; G.chasesTotal=(G.chasesTotal||0)+1;
      SFX.scream(); ring(z.x,z.y,'rgba(255,80,60,$)',140);
      for(const o of G.zombies){ if(o===z||o.dead||o.state==='chase')continue;
        if(Math.hypot(o.x-p.x,o.y-p.y)<300){ o.state='investigate'; o.tx=p.x; o.ty=p.y; o.waitT=0; } } }
    if(d<19)die();
  }
  let spd,tx=z.tx,ty=z.ty,arrive=false;
  const dtg=Math.hypot(tx-z.x,ty-z.y);
  switch(z.state){
    case 'wander': spd=34*z.speedMul; if(dtg<8||z.wt-=dt<=0){ z.wt=4+Math.random()*5; pickWander(z); } break;

    case 'return': spd=70; if(dtg<8){ z.state='wander'; pickWander(z); arrive=true; } break;
  }
  if(!arrive){ let ang=Math.atan2(ty-z.y,tx-z.x);
    if(z.state==='chase'||z.state==='investigate')ang=steerAng(z,ang);
