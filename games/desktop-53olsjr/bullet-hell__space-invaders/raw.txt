function spawnShots(){
  const sp=640*S,y=P.y-13*S;
  if(P.power===1)shots.push({x:P.x,y,vx:0,vy:-sp,dead:false});
  else if(P.power===2){shots.push({x:P.x-5*S,y,vx:0,vy:-sp,dead:false},{x:P.x+5*S,y,vx:0,vy:-sp,dead:false});}
  else for(const a of[-.13,0,.13])shots.push({x:P.x,y,vx:Math.sin(a)*sp,vy:-Math.cos(a)*sp,dead:false});
  parts.push({x:P.x,y:y-6*S,vx:0,vy:-220*S,t:0,life:.12,size:2.5*S,col:[234,255,242]});
}
function updPlayer(d){
  const spd=340*S;
  let tx=0,ty=0;
  if(keys['arrowleft']||keys['a'])tx--;
  if(keys['arrowright']||keys['d'])tx++;
  if(keys['arrowup']||keys['w'])ty--;
  if(keys['arrowdown']||keys['s'])ty++;
  if(tx||ty){const l=Math.hypot(tx,ty);P.lx=tx/l;P.ly=ty/l;}
  let tvx=tx*spd,tvy=ty*spd;
  if(P.dashT>0){tvx=P.dx*950*S;tvy=P.dy*950*S;P.dashT-=d;
    P.trail.push({x:P.x,y:P.y,t:.28});if(P.trail.length>14)P.trail.shift();}
  const k=Math.min(1,d*(P.dashT>0?20:13));
  P.vx+=(tvx-P.vx)*k;P.vy+=(tvy-P.vy)*k;
  P.x+=P.vx*d;P.y+=P.vy*d;
  if(touch.active&&!tx&&!ty){
    const ty2=clamp(touch.y,H*.35,H-26);
    P.x+=(touch.x-P.x)*Math.min(1,d*10);P.y+=(ty2-P.y)*Math.min(1,d*10);
  }
  P.x=clamp(P.x,14,W-14);P.y=clamp(P.y,H*.30,H-26);
  for(const tr of P.trail)tr.t-=d;
  P.trail=P.trail.filter(tr=>tr.t>0);
  P.dashCd=Math.max(0,P.dashCd-d);P.inv-=d;P.fireCd-=d;
  if((keys[' ']||touch.fire)&&P.fireCd<=0){P.fireCd=[0,.13,.12,.11][P.power];spawnShots();sfxShoot();}
}
function updPlay(dt){
  G.t+=dt;
  const ts=G.slow>0?.35:1;G.slow=Math.max(0,G.slow-dt);
  const d=dt*ts;
  if(G.intro>0)G.intro-=dt;
  G.tWave+=d;
  G.shake*=Math.pow(.002,dt);if(G.shake<.15)G.shake=0;
  if(G.comboT>0){G.comboT-=d;if(G.comboT<=0)G.combo=0;}
  updPlayer(d);
  if(GR&&aliens.length)updGrid(d);
  if(B)updBoss(d);else if(G.bossDying)updBossDeath(d);
  if(G.curt){const c=G.curt;c.t-=d;c.acc-=d;
    while(c.acc<=0&&c.t>0){c.acc+=.075;addBullet(c.x,-12*S,0,(215+G.wave*4)*S,C_CHIT,3.4*S);}
    if(c.t<=0)G.curt=null;}
  // shots
  for(const s of shots){if(s.dead)continue;s.y+=s.vy*d;s.x+=s.vx*d;
    if(s.y<-30||s.x<-40||s.x>W+40){s.dead=true;continue;}
    if(B){const dx=s.x-B.x,dy=s.y-B.y;
      if(dx*dx+dy*dy<(58*S)*(58*S)){B.hp--;B.flash=.09;s.dead=true;spark(s.x,s.y,C_BOSS,4);
        if(B.hp<=0)bossDeath();continue;}}
    for(const a of aliens){if(a.dead||a.drop<1)continue;
      const dx=s.x-a.x,dy=s.y-a.y;
      if(dx*dx+dy*dy<(5.2*cs)*(5.2*cs)){killAlien(a);s.dead=true;break;}}
  }
  shots=shots.filter(s=>!s.dead);
  // bullets vs player
  if(P.inv<=0&&P.dashT<=0){
    for(const b of bullets){if(b.dead)continue;
      const dx=b.x-P.x,dy=b.y-P.y,rr=b.r+4*S;
      if(dx*dx+dy*dy<rr*rr){b.dead=true;hurt();break;}}
  }
  updBullets(d);
  // aliens vs player
  for(const a of aliens){if(a.dead)continue;
    const dx=a.x-P.x,dy=a.y-P.y;
    if(dx*dx+dy*dy<(7.5*cs)*(7.5*cs)){killAlien(a);hurt();}}
  // pickups
  for(const p of picks){p.t+=d;p.y+=p.vy*d;
    const dx=p.x-P.x,dy=p.y-P.y;
    if(dx*dx+dy*dy<(26*S)*(26*S)){p.dead=true;
      if(p.type==='P'){P.power=Math.min(3,P.power+1);floats.push({x:p.x,y:p.y,txt:'+POWER',t:0,life:1,col:'#7df3ff'});sfxPower();}
      else{P.shields=Math.min(5,P.shields+1);floats.push({x:p.x,y:p.y,txt:'+SHIELD',t:0,life:1,col:'#a6ff4d'});sfxPick();}}
    if(p.y>H+20)p.dead=true;}
  picks=picks.filter(p=>!p.dead);
  aliens=aliens.filter(a=>!a.dead);
  updFx(d);
  if(G.clearT>0){G.clearT-=dt;if(G.clearT<=0)spawnWave(G.wave+1);}
}
