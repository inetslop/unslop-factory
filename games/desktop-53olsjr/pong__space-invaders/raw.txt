  $('#hiscore').textContent=best>0?fmt(best):'000000';
}

  nb?show($('#newRecord')):hide($('#newRecord'));
  show($('#over'));
}

function spawnWave(n){
  const cols=W<620?8:10;
  const rows=Math.min(3+Math.ceil((n-1)/2),6);
  const cellW=Math.min(46*S,(W*.86)/cols);
  let ps=clamp(Math.floor((cellW-6)/12),2,5);
  const aw=12*ps,ah=8*ps;
  grid={x:0,y:78*S,dir:1,frame:0,cols,rows,cellW,cellH:Math.max(ah+14*S,30*S),ps,dropStep:14*S,total:cols*rows,rem:cols*rows};
  const gW=(cols-1)*cellW+aw;
  grid.x=(W-gW)/2;
  aliens=[];
  for(let r=0;r<rows;r++)for(let c=0;c<cols;c++){
    const sp=r<Math.ceil(rows*.4)?'squid':(r<Math.ceil(rows*.75)?'octo':'crab');
    aliens.push({c,r,sp,alive:true,w:aw,h:ah,dx:0,dy:0,ph:rand(0,6.28)});
  }
  bombs=[];caps=[];lasers=[];
  bombT=rand(1.4,2.2);beatT=.3;beatIdx=0;
  serveBall();setWaveHUD();
}
function layoutAliens(){
  for(const a of aliens){if(!a.alive)continue;
    a.dx=grid.x+a.c*grid.cellW+(grid.cellW-a.w)/2;
    a.dy=grid.y+a.r*grid.cellH+(grid.cellH-a.h)/2+Math.sin(tGlobal*1.6+a.ph)*1.5*S;
  }
}
function serveBall(){
  const b={x:paddle.x,y:paddleY()-10*S,vx:0,vy:0,r:6*S,sp:baseSpeed(),stick:true,st:1.0,trail:[]};
  balls=[b];
}
function launch(b){
  const ang=-Math.PI/2+clamp((paddle.x-W/2)/W,-1,1)*.5+rand(-.12,.12);
  b.vx=Math.cos(ang)*b.sp;b.vy=Math.sin(ang)*b.sp;b.stick=false;
  sPaddle(0);
}
function burst(x,y,col,n){
  for(let i=0;i<n;i++)parts.push({x,y,vx:rand(-170,170)*S,vy:rand(-170,170)*S,t:rand(.35,.7),life:.7,col,sz:rand(2,4.5)*S});
}
function floatTxt(x,y,txt,col,big){floats.push({x,y,txt,col,big:!!big,t:0,life:big?1.2:.9});}

function killAlien(a,how){
  a.alive=false;grid.rem--;streak++;
  const nm=1+Math.min(3,Math.floor(streak/5));
  if(nm!==prevMult){floatTxt(paddle.x,paddleY()-34*S,'MULTIPLIER ×'+nm,'#FFD166',true);sChime();prevMult=nm;mult=nm;updateStreak();}
  const pts=SPECIES[a.sp].val*mult;score+=pts;updateScore();
  floatTxt(a.dx+a.w/2,a.dy,'+'+pts,how==='laser'?'#9fe8ff':SPECIES[a.sp].col);
  burst(a.dx+a.w/2,a.dy+a.h/2,SPECIES[a.sp].col,14);sKill();
  if(Math.random()<.10){
    const r=Math.random(),type=r<.4?'wide':(r<.75?'multi':'shield');
    caps.push({x:a.dx+a.w/2,y:a.dy+a.h/2,type,t:0});
  }
  if(grid.rem<=0)waveClear();
}
function waveClear(){
  const bonus=100+(wave-1)*50;score+=bonus;updateScore();
  state='wavebreak';breakT=2.3;sWave();
  showBanner('WAVE '+wave+' CLEARED','BONUS +'+bonus);
  const cols=['#7CF2C4','#6EC5FF','#FFD166','#FF5A6E'];
  for(let i=0;i<46;i++)parts.push({x:rand(0,W),y:rand(-10,H*.3),vx:rand(-40,40)*S,vy:rand(40,120)*S,t:rand(.5,1.1),life:1.1,col:cols[i%4],sz:rand(2,4)*S});
}
function applyPower(type){
  if(type==='wide'){paddle.wideT=10;floatTxt(paddle.x,paddleY()-30*S,'WIDE PADDLE','#7CF2C4',true);}
  else if(type==='multi'){
    while(balls.length<4){
      const b0=balls[0];if(!b0)break;
      const nb={x:b0.x,y:b0.y,r:b0.r,sp:b0.sp,stick:b0.stick,st:b0.st?1.0:0,trail:[]};
      if(b0.stick){nb.vx=0;nb.vy=0;}
      else{const m=Math.hypot(b0.vx,b0.vy)||1,off=(balls.length%2?1:-1)*.5,ca=Math.cos(off),sa=Math.sin(off);
        nb.vx=(b0.vx/m*ca-b0.vy/m*sa)*m;nb.vy=(b0.vx/m*sa+b0.vy/m*ca)*m;}
      balls.push(nb);
    }
    floatTxt(paddle.x,paddleY()-30*S,'MULTI-BALL','#6EC5FF',true);
  }else{shieldT=8;floatTxt(paddle.x,paddleY()-30*S,'AEGIS SHIELD','#FFD166',true);}
}
function loseLife(src){

  doFlash();shake=1;sMiss();
  if(lives<=0){gameOver('lives');return;}
  if(src==='ball'){balls=[];serveBall();}
  else{balls=balls.slice(0,1);}
}

/* ============ update ============ */
function updPaddle(dt){
  const pw=120*S*(paddle.wideT>0?1.42:1);let dir=0;
  if(keys['ArrowLeft']||keys['a'])dir--;if(keys['ArrowRight']||keys['d'])dir++;
  if(dir!==0)paddle.x+=dir*560*S*dt;
  else if(useMouse&&mouseX!=null)paddle.x+=(mouseX-paddle.x)*Math.min(1,dt*16);
  paddle.x=clamp(paddle.x,pw/2+6,W-pw/2-6);
  paddle.vw+=(pw-paddle.vw)*Math.min(1,dt*10);
  paddle.kick=Math.max(0,paddle.kick-dt*5);
}
function update(dt){
  tGlobal+=dt;
  // ambient
  cometT-=dt;
  if(cometT<=0&&!comet){comet={x:rand(W*.2,W*.95),y:-20,vx:rand(-170,-80)*S,vy:rand(130,210)*S,t:0};cometT=rand(5,11);}
  if(comet){comet.t+=dt;comet.x+=comet.vx*dt;comet.y+=comet.vy*dt;if(comet.t>1.4)comet=null;}
  for(const st of nearStars){st.y+=st.vy*S*dt;if(st.y>H+4){st.y=-4;st.x=Math.random()*W;}}
  for(let i=parts.length-1;i>=0;i--){const p=parts[i];p.t-=dt;if(p.t<=0){parts.splice(i,1);continue;}p.x+=p.vx*dt;p.y+=p.vy*dt;p.vx*=Math.max(0,1-2.5*dt);p.vy*=Math.max(0,1-2.5*dt);}
  for(let i=floats.length-1;i>=0;i--){floats[i].t+=dt;if(floats[i].t>=floats[i].life)floats.splice(i,1);}
  if(shake>0)shake=Math.max(0,shake-dt*2.4);

  if(state==='menu'||state==='gameover')return;
  updPaddle(dt);
  if(state==='wavebreak'){
    breakT-=dt;
    if(breakT<=0){wave++;spawnWave(wave);showBanner('WAVE '+wave,WAVE_SUBS[(wave-1)%WAVE_SUBS.length]);state='playing';}
    return;
  }
  /* --- playing --- */
  laserCd=Math.max(0,laserCd-dt);
  paddle.wideT=Math.max(0,paddle.wideT-dt);
  shieldT=Math.max(0,shieldT-dt);

  // grid march + classic beat
  if(grid.rem>0){
    const remF=grid.rem/grid.total;
    const spd=(46+9*(wave-1))*S*(.45+1.2*(1-remF));
    grid.x+=grid.dir*spd*dt;
    if(wave>2)grid.y+=2*S*dt;
    const aw=grid.ps*12,gW=(grid.cols-1)*grid.cellW+aw;
    let dropped=false;
    if(grid.x<8*S){grid.x=8*S;grid.dir=1;dropped=true;}
    else if(grid.x+gW>W-8*S){grid.x=W-8*S-gW;grid.dir=-1;dropped=true;}
    if(dropped){grid.y+=grid.dropStep;sThud();}
    layoutAliens();
    const bottom=grid.y+(grid.rows-1)*grid.cellH+grid.ps*8;
    if(bottom>=paddleY()-2*S){gameOver('land');return;}
    beatT-=dt;
    const iv=clamp(.62-(1-remF)*.45,.16,.62);
    if(beatT<=0){beatT+=iv;if(beatT>iv*2)beatT=iv;
      tone([98,92.5,87.31,82.41][beatIdx%4],.09,'square',.09);
      if(beatIdx%2===0)grid.frame^=1;
      beatIdx++;}
  }

  // bombs spawn
  bombT-=dt;
  if(bombT<=0){
    bombT=rand(1.4,2.4)/(1+(wave-1)*.18);
    const alive=aliens.filter(a=>a.alive);
    if(alive.length&&bombs.length<3+wave){
      const a=alive[irand(0,alive.length-1)];
      bombs.push({x:a.dx+a.w/2,y:a.dy+a.h,vy:(150+wave*22)*S});
    }
  }

  // lasers
  for(let i=lasers.length-1;i>=0;i--){
    const l=lasers[i];l.y-=940*S*dt;
    let dead=l.y<-30;
    if(!dead)for(const a of aliens){
      if(!a.alive)continue;
      if(l.x>a.dx-2&&l.x<a.dx+a.w+2&&l.y<a.dy+a.h&&l.y+20*S>a.dy){killAlien(a,'laser');dead=true;break;}
    }
    if(dead)lasers.splice(i,1);
  }

  // bombs
  const py=paddleY(),pw=paddle.vw;
  for(let i=bombs.length-1;i>=0;i--){
    const b=bombs[i];b.y+=b.vy*dt;let dead=false;
    for(let j=lasers.length-1;j>=0;j--){const l=lasers[j];
      if(Math.abs(l.x-b.x)<7*S&&b.y>l.y-6*S&&b.y<l.y+20*S){
        lasers.splice(j,1);dead=true;score+=15;updateScore();floatTxt(b.x,b.y,'+15','#9fe8ff');burst(b.x,b.y,'#FFD166',8);sBlock();break;}}
    if(dead){bombs.splice(i,1);continue;}
    if(shieldT>0){const dx=b.x-paddle.x,dy=b.y-(py-6*S),rr=(pw/2+26*S);
      if(dx*dx+dy*dy<rr*rr){burst(b.x,b.y,'#6EC5FF',8);sBlock();bombs.splice(i,1);continue;}}
    if(b.y+5*S>=py&&b.y-5*S<=py+PH()&&Math.abs(b.x-paddle.x)<=pw/2+5*S){
      score+=5;updateScore();floatTxt(b.x,py-8*S,'+5','#9fd8ff');burst(b.x,py,'#FFD166',7);sBlock();bombs.splice(i,1);continue;}
    if(b.y>H+10){bombs.splice(i,1);loseLife('bomb');sBombHit();if(state!=='playing')return;continue;}
  }

  // pickups
  for(let i=caps.length-1;i>=0;i--){
    const c=caps[i];c.t+=dt;c.y+=110*S*dt;
    if(c.y+8*S>=py&&c.y-12*S<=py+PH()+6*S&&Math.abs(c.x-paddle.x)<=pw/2+14*S){applyPower(c.type);sPower();caps.splice(i,1);continue;}
    if(c.y>H+20)caps.splice(i,1);
  }

  // balls
  for(let i=balls.length-1;i>=0;i--){
    const b=balls[i];
    if(b.stick){b.x=paddle.x;b.y=py-b.r-4*S;b.st-=dt;if(b.st<=0)launch(b);continue;}
    b.x+=b.vx*dt;b.y+=b.vy*dt;
    if(b.x<b.r){b.x=b.r;b.vx=Math.abs(b.vx);sWall();}
    if(b.x>W-b.r){b.x=W-b.r;b.vx=-Math.abs(b.vx);sWall();}
    if(b.y<b.r){b.y=b.r;b.vy=Math.abs(b.vy);sWall();}
    if(b.vy>0&&b.y+b.r>=py&&b.y-b.r<=py+PH()&&Math.abs(b.x-paddle.x)<=pw/2+b.r){
      const rel=clamp((b.x-paddle.x)/(pw/2),-1,1);
      b.sp=Math.min(b.sp*1.03,MAXSP());
      const ang=-Math.PI/2+rel*1.05+rand(-.03,.03);
      b.vx=Math.cos(ang)*b.sp;b.vy=Math.sin(ang)*b.sp;b.y=py-b.r-.5;
      paddle.kick=1;sPaddle(rel);
    }
    for(const a of aliens){
      if(!a.alive)continue;
      const cx=clamp(b.x,a.dx,a.dx+a.w),cy=clamp(b.y,a.dy,a.dy+a.h);
      const ddx=b.x-cx,ddy=b.y-cy;
      if(ddx*ddx+ddy*ddy<=b.r*b.r){
        killAlien(a,'ball');
        b.sp=Math.min(b.sp*1.035,MAXSP());
        const m=Math.hypot(b.vx,b.vy);if(m>0){b.vx*=b.sp/m;b.vy*=b.sp/m;}
        break;
      }
    }
    if(state!=='playing')return;
    if(b.y-b.r>H+4){
      balls.splice(i,1);
      if(balls.length===0){loseLife('ball');if(state!=='playing')return;}
    }
    b.trail.push({x:b.x,y:b.y});if(b.trail.length>10)b.trail.shift();
  }
}

function drawParts(){
  for(const p of parts){ctx.globalAlpha=clamp(p.t/p.life,0,1);ctx.fillStyle=p.col;ctx.fillRect(p.x-p.sz/2,p.y-p.sz/2,p.sz,p.sz);}
  ctx.globalAlpha=1;
}
function drawFloats(){
  ctx.textAlign='center';
  for(const f of floats){
    const p=f.t/f.life,a=1-p,y=f.y-p*34*S;
    ctx.globalAlpha=a;
    if(f.big){ctx.fillStyle=f.col;ctx.font=`900 ${Math.round(20*Math.max(.8,S))}px Arial, sans-serif`;
      ctx.shadowColor=f.col;ctx.shadowBlur=12;}
    else{ctx.fillStyle=f.col;ctx.font=`700 ${Math.round(13*Math.max(.8,S))}px ui-monospace, monospace`;ctx.shadowBlur=0;}
    ctx.fillText(f.txt,f.x,y);ctx.shadowBlur=0;
  }
  ctx.globalAlpha=1;
}
function drawAttract(){
  const ps=Math.max(2.4,3*S),aw=12*ps,spacing=aw+26*S;
  const rows=[{sp:'squid',y:H*.2},{sp:'octo',y:H*.34},{sp:'crab',y:H*.48}];
  for(let r=0;r<rows.length;r++){
    const row=rows[r],n=Math.ceil(W/spacing)+2;
    let x=((tGlobal*(18+r*7)*S)%(spacing))-spacing-aw;
    for(let i=0;i<n;i++,x+=spacing){
      ctx.globalAlpha=.34;
      drawSprite(SPR[row.sp].f[Math.floor(tGlobal*2)%2],x,row.y+Math.sin(tGlobal*1.5+i*.7)*6,ps,SPECIES[row.sp].col);
    }
  }
  ctx.globalAlpha=1;
}
function render(){
  ctx.clearRect(0,0,W,H);
  if(bg)ctx.drawImage(bg,0,0,W,H);
  let ox=0,oy=0;if(shake>0){ox=rand(-1,1)*shake*9*S;oy=rand(-1,1)*shake*9*S;}
  ctx.save();ctx.translate(ox,oy);
  // twinkling near stars + comet
  for(const st of nearStars){
    ctx.globalAlpha=.25+.45*(.5+.5*Math.sin(tGlobal*st.tw+st.ph));
    ctx.fillStyle='#cfe4ff';ctx.fillRect(st.x,st.y,st.r,st.r);
  }
  ctx.globalAlpha=1;
  if(comet){
    ctx.save();ctx.globalCompositeOperation='lighter';
    ctx.strokeStyle=`rgba(200,235,255,${.7*(1-comet.t/1.4)})`;ctx.lineWidth=2;
    ctx.beginPath();ctx.moveTo(comet.x,comet.y);ctx.lineTo(comet.x-comet.vx*.09,comet.y-comet.vy*.09);ctx.stroke();
    ctx.restore();
  }
  if(state==='menu')drawAttract();
  else{
    drawDangerLine();
    drawCaps();drawAliens();drawBombs();drawLasers();drawShield();drawPaddle();drawBalls();
  }
  drawParts();drawFloats();
  ctx.restore();
}
