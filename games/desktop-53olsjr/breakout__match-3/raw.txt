  const L=layout(), n=cells.size;
  let pts = kind==='res' ? 24*n+16 : kind==='match' ? Math.round(20*n*(1+(step-1)*.5)) : 10*n;
  pts=Math.round(pts*(1+Math.min(combo,10)*.1));
  score+=pts; cleared+=n;
  let cx=0,cy=0;
  for(const idx of cells){const r=(idx/COLS)|0,c=idx%COLS,g=grid[r][c]; if(!g)continue;
    grid[r][c]=null;
    const px=L.gx+c*L.cell+L.cell/2, py=L.gy+r*L.cell+L.cell/2+(g.fallY||0);
    cx+=px;cy+=py; burst(px,py,g.color,kind==='res'?10:6,kind==='res');}
  cx/=n;cy/=n;
  pops.push({x:cx,y:cy,t:1.1,max:1.1,txt:'+'+pts,col:kind==='res'?'#ffffff':'#ffd27d',size:kind==='res'?20:14});
  if(kind==='res'){pops.push({x:cx,y:cy-20,t:1.3,max:1.3,txt:'RESONANCE ×'+n,col:'#8ae4ff',size:15});shake=Math.max(shake,.5);sRes(n);}
  else if(kind==='match')sPop(step,n); else sCrack();
  if(n>=6||kind==='res')shake=Math.max(shake,.35);
  refill(step);
  const m=findMatches();
  if(m.size)resolve(m,step+1,'match');
  else{if(cleared>=quota)levelUp(); ensureMoves(false);}
}
function refill(step){
  const L=layout();
  for(let c=0;c<COLS;c++){
    let write=ROWS-1;
    for(let r=ROWS-1;r>=0;r--){const g=grid[r][c];
      if(g){if(write!==r){grid[write][c]=g;grid[r][c]=null;g.fallY=-(write-r)*L.cell;}write--;}}
    for(let r=write;r>=0;r--){
      const wild=(step>=2&&Math.random()<.10);
      grid[r][c]=newGem(wild?WILD:(Math.random()*PAL.length)|0,-((r+1)*L.cell+26));}
  }
}
function shatter(r,c){
  const g=grid[r][c]; if(!g)return;
  let cells;
  if(g.color===charge&&g.color!==WILD){cells=new Set();flood(r,c,g.color,cells);}
  else cells=new Set([r*COLS+c]);
  resolve(cells,1,cells.size>1?'res':'ball');
}

/* ============ swap input ============ */
function trySwap(a,b){
  if(Math.abs(a.r-b.r)+Math.abs(a.c-b.c)!==1)return;
  const gA=grid[a.r][a.c],gB=grid[b.r][b.c]; if(!gA||!gB)return;
  swapCells(a.r,a.c,b.r,b.c);
  const L=layout();
  gA.offX=(a.c-b.c)*L.cell; gA.offY=(a.r-b.r)*L.cell;
  gB.offX=(b.c-a.c)*L.cell; gB.offY=(b.r-a.r)*L.cell;
  if(findMatches().size===0){
    swapCells(a.r,a.c,b.r,b.c);
    gA.offX=gA.offY=0;gB.offX=gB.offY=0;gA.jolt=1;gB.jolt=1;sBad();return;
  }
  sSwap(); resolve(findMatches(),1,'match');
}

/* ============ particles / fx ============ */
function burst(x,y,color,n,big){
  const col=color===WILD?('hsl('+((T*90)%360)+' 85% 70%)'):PAL[color].b;
  for(let i=0;i<n;i++){const a=Math.random()*TAU,sp=(big?190:120)*Math.random()+60;
    parts.push({t:'shard',x,y,vx:Math.cos(a)*sp,vy:Math.sin(a)*sp-80,r:Math.random()*4+2.5,
      rot:Math.random()*TAU,vr:(Math.random()-.5)*10,life:.7,max:.7,col});}
  parts.push({t:'ring',x,y,r:4,vr:big?280:160,life:.35,max:.35,col});
  for(let i=0;i<n/2;i++){const a=Math.random()*TAU,sp=260*Math.random()+80;
    parts.push({t:'spark',x,y,vx:Math.cos(a)*sp,vy:Math.sin(a)*sp,life:.3,max:.3,col});}
  if(parts.length>420)parts.splice(0,parts.length-420);
}

/* ============ update ============ */
function makeStars(){const n=Math.round(W*H/9000);stars=[];
  for(let i=0;i<n;i++)stars.push({x:Math.random()*W,y:Math.random()*H,r:Math.random()*1.6+.4,p:Math.random()*TAU,s:Math.random()*12+4});}
function updateStars(dt){for(const s of stars){s.y+=s.s*dt;s.p+=dt*2;if(s.y>H+2){s.y=-2;s.x=Math.random()*W;}}}

function updatePad(dt){
  pad.w=Math.max(84,Math.min(W*.17,190)); pad.y=H-36; pad.h=13;
  const s=Math.max(560,bsp()*1.6); let vx=0;
  if(lastInput==='key'){const d=(keys.r?1:0)-(keys.l?1:0);
    if(d){pad.x+=d*s*dt;vx=d*s;}}
  else{const px=Math.max(pad.w/2+6,Math.min(W-pad.w/2-6,pointerX));
    const dx=px-pad.x; pad.x+=dx*Math.min(1,dt*22); vx=Math.max(-300,Math.min(300,dx/Math.max(dt,.008)*.5));}
  pad.vx=vx; pad.sq=Math.max(0,pad.sq-dt*4);
}
function launch(){
  if(!ball.held||paused)return;
  ball.held=false; const a=(Math.random()-.5)*.5;
  ball.vx=Math.sin(a)*bsp()+Math.max(-200,Math.min(200,pad.vx))*.3;
  ball.vy=-Math.cos(a)*bsp(); sLaunch();
}
function updateBall(dt){
  const L=layout(), s=bsp();
  if(ball.held){ball.x=pad.x;ball.y=pad.y-pad.h/2-ball.r-3;return;}
  ball.x=Math.max(ball.r,Math.min(W-ball.r,ball.x));
  const n=Math.max(1,Math.ceil((Math.hypot(ball.vx,ball.vy)*dt)/(ball.r*.8)));
  for(let i=0;i<n;i++){const h=dt/n;
    ball.x+=ball.vx*h; ball.y+=ball.vy*h;
    if(ball.x<ball.r){ball.x=ball.r;ball.vx=Math.abs(ball.vx);sWall();}
    if(ball.x>W-ball.r){ball.x=W-ball.r;ball.vx=-Math.abs(ball.vx);sWall();}
    if(ball.y<HUDH+2+ball.r){ball.y=HUDH+2+ball.r;ball.vy=Math.abs(ball.vy);sWall();}
    // paddle
    if(ball.vy>0&&ball.y+ball.r>=pad.y-pad.h/2&&ball.y-ball.r<=pad.y+pad.h/2&&Math.abs(ball.x-pad.x)<=pad.w/2+ball.r){
      const rel=Math.max(-1,Math.min(1,(ball.x-pad.x)/(pad.w/2)));
      const ang=rel*(Math.PI*.36);
      ball.vx=s*Math.sin(ang)+Math.max(-200,Math.min(200,pad.vx))*.18;
      ball.vy=-s*Math.cos(ang);
      ball.y=pad.y-pad.h/2-ball.r-.5; pad.sq=1; sPaddle();
      burst(ball.x,pad.y-pad.h/2,PAL[charge],3,false);}
    // gems (only landed ones are solid — falling gems can be dodged)
    const gc=Math.floor((ball.x-L.gx)/L.cell), gr=Math.floor((ball.y-L.gy)/L.cell);
    if(gr>=0&&gr<ROWS&&gc>=0&&gc<COLS){const g=grid[gr][gc];
      if(g&&g.fallY===0){
        const ins=L.cell*.12, rx=L.gx+gc*L.cell+ins, ry=L.gy+gr*L.cell+ins, rw=L.cell-ins*2;
        const nx=Math.max(rx,Math.min(ball.x,rx+rw)), ny=Math.max(ry,Math.min(ball.y,ry+rw));
        const dx=ball.x-nx, dy=ball.y-ny;
        if(dx*dx+dy*dy<=ball.r*ball.r){
          const cxx=rx+rw/2, cyy=ry+rw/2;
          const ox=rw/2+ball.r-Math.abs(ball.x-cxx), oy=rw/2+ball.r-Math.abs(ball.y-cyy);
          if(ox<oy)ball.vx=ball.x<cxx?-Math.abs(ball.vx):Math.abs(ball.vx);
          else ball.vy=ball.y<cyy?-Math.abs(ball.vy):Math.abs(ball.vy);
          shatter(gr,gc);}}
  }
  let m=Math.hypot(ball.vx,ball.vy)||1; ball.vx=ball.vx/m*s; ball.vy=ball.vy/m*s;
  if(Math.abs(ball.vy)<s*.18){ball.vy=(ball.vy<0?-1:1)*s*.18;m=Math.hypot(ball.vx,ball.vy);ball.vx*=s/m;}
  ball.trail.push({x:ball.x,y:ball.y}); if(ball.trail.length>16)ball.trail.shift();
  if(ball.y-ball.r>H+16)loseLife();
}
function updateGems(dt){
  for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++){const g=grid[r][c];if(!g)continue;
    const k=Math.exp(-dt*13);
    g.offX*=k;g.offY*=k;if(Math.abs(g.offX)<.4)g.offX=0;if(Math.abs(g.offY)<.4)g.offY=0;
    g.jolt=Math.max(0,g.jolt-dt*3);
    g.sq=Math.max(0,g.sq-dt*5);
    if(g.fallY<0){g.fallV+=2600*dt;g.fallY+=g.fallV*dt;
      if(g.fallY>=0){g.fallY=0;g.fallV=0;g.sq=1;}}
  }
}
function updateFx(dt){
  for(let i=parts.length-1;i>=0;i--){const p=parts[i];p.life-=dt;
    if(p.life<=0){parts.splice(i,1);continue;}
    if(p.t==='shard'){p.vy+=900*dt;p.x+=p.vx*dt;p.y+=p.vy*dt;p.rot+=p.vr*dt;}
    else if(p.t==='spark'){const d=Math.exp(-3*dt);p.vx*=d;p.vy*=d;p.x+=p.vx*dt;p.y+=p.vy*dt;}
    else p.r+=p.vr*dt;}
  for(let i=pops.length-1;i>=0;i--){const p=pops[i];p.t-=dt;p.y-=28*dt;if(p.t<=0)pops.splice(i,1);}
  shake=Math.max(0,shake-dt*1.9); flash*=Math.exp(-dt*3.2);
  pulse=Math.max(0,pulse-dt*2.5); comboPulse=Math.max(0,comboPulse-dt*3);
  if(toast){toast.t-=dt;if(toast.t<=0)toast=null;}
}

/* ============ drawing ============ */
function rr(x,y,w,h,r){ctx.beginPath();ctx.moveTo(x+r,y);ctx.arcTo(x+w,y,x+w,y+h,r);
  ctx.arcTo(x+w,y+h,x,y+h,r);ctx.arcTo(x,y+h,x,y,r);ctx.arcTo(x,y,x+w,y,r);ctx.closePath();}
function gemPath(x,y,sx,sy){const k=.62;ctx.beginPath();
  ctx.moveTo(x,y-sy);ctx.lineTo(x+sx*k,y-sy*k);ctx.lineTo(x+sx,y);ctx.lineTo(x+sx*k,y+sy*k);
  ctx.lineTo(x,y+sy);ctx.lineTo(x-sx*k,y+sy*k);ctx.lineTo(x-sx,y);ctx.lineTo(x-sx*k,y-sy*k);ctx.closePath();}

function drawGemShape(x,y,sx,sy,color){
  const tw=.86+.14*Math.sin(T*2+(color===WILD?0:0)); // subtle global life handled by ph outside
  gemPath(x,y,sx,sy);
  if(color===WILD){
    const h=(T*90)%360;
    ctx.fillStyle='hsl('+h+' 85% 62%)';ctx.fill();
    ctx.strokeStyle='rgba(255,255,255,.8)';ctx.lineWidth=Math.max(1.2,sx*.09);ctx.stroke();
  }else{const p=PAL[color];
    ctx.fillStyle=p.b;ctx.fill();
    ctx.globalAlpha=.75;ctx.fillStyle=p.a;
    ctx.beginPath();ctx.moveTo(x,y-sy);ctx.lineTo(x+sx*.62,y-sy*.62);ctx.lineTo(x-sx*.62,y-sy*.62);ctx.closePath();ctx.fill();
    ctx.globalAlpha=.55;ctx.fillStyle=p.c;
    ctx.beginPath();ctx.moveTo(x,y+sy);ctx.lineTo(x+sx*.62,y+sy*.62);ctx.lineTo(x-sx*.62,y+sy*.62);ctx.closePath();ctx.fill();
    ctx.globalAlpha=1;
    ctx.strokeStyle=p.c;ctx.lineWidth=Math.max(1.2,sx*.09);gemPath(x,y,sx,sy);ctx.stroke();}
  ctx.globalAlpha=.18;ctx.strokeStyle='#fff';ctx.lineWidth=1;gemPath(x,y,sx*.5,sy*.5);ctx.stroke();
  ctx.globalAlpha=.55;ctx.fillStyle='#fff';
  ctx.beginPath();ctx.arc(x-sx*.32,y-sy*.38,Math.max(1,sx*.14),0,TAU);ctx.fill();
  ctx.globalAlpha=1;
}

function cellAt(x,y){const L=layout();
  if(x<L.gx||y<L.gy||x>=L.gx+L.gw||y>=L.gy+L.gh)return null;
  const c=Math.floor((x-L.gx)/L.cell), r=Math.floor((y-L.gy)/L.cell);
  return (r>=0&&r<ROWS&&c>=0&&c<COLS)?{r,c}:null;}
