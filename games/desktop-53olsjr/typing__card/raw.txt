  if(G.state==='paused'&&k===' '){e.preventDefault();resumeGame();return;}}
 if(k==='Escape'){if(G.state==='playing')pauseGame();else if(G.state==='paused')resumeGame();return;}
 if(kl==='0'){toggleMute();return;}
 if(e.repeat)return;
 if(G.state==='playing'&&!G.dying&&k.length===1&&/[a-z]/i.test(k)){e.preventDefault();handleKey(kl);}});
cv.addEventListener('pointerdown',()=>{initAudio();
 if(G.state==='start'||G.state==='over')startGame();else if(G.state==='paused')resumeGame();});
document.addEventListener('visibilitychange',()=>{if(document.hidden&&G.state==='playing')pauseGame();});
window.addEventListener('blur',()=>{if(G.state==='playing')pauseGame();});

function update(dt){
 G.at+=dt;
 for(const m of motes){m.y-=m.s*dt;m.x+=Math.sin(G.at*.5+m.p)*6*dt;if(m.y<-10){m.y=H+10;m.x=rand(0,W);}}
 for(const g of glyphs){g.y-=g.s*dt;g.rot+=g.rs*dt;if(g.y<-140){g.y=H+140;g.x=rand(0,W);}}
 if(G.state!=='paused'){
  for(let i=particles.length-1;i>=0;i--){const p=particles[i];p.t+=dt;if(p.t>=p.dur){particles.splice(i,1);continue;}
   if(p.kind===0){p.vy+=p.g*dt;p.x+=p.vx*dt;p.y+=p.vy*dt;p.rot+=p.vr*dt;}
   else if(p.kind===1){p.x+=p.vx*dt;p.y+=p.vy*dt;p.rot+=p.vr*dt;}
   else{p.x+=p.vx*dt;p.y+=p.vy*dt;}}
  for(let i=popups.length-1;i>=0;i--){const p=popups[i];p.t+=dt;p.y-=26*dt;if(p.t>=p.dur)popups.splice(i,1);}
  for(let i=ticker.length-1;i>=0;i--){ticker[i].t+=dt;if(ticker[i].t>2.8)ticker.splice(i,1);}
  if(G.banner){G.banner.t+=dt;if(G.banner.t>G.banner.dur)G.banner=null;}}
 G.shake*=Math.exp(-7*dt);if(G.shake<.05)G.shake=0;
 G.redFlash=Math.max(0,G.redFlash-1.6*dt);
 G.comboPulse=Math.max(0,G.comboPulse-3*dt);

 if(G.state==='playing'){
  G.time+=dt;
  if(!G.dying){spawnT-=dt;
   if(spawnT<=0){const maxC=Math.min(2+Math.floor((G.level-1)*.8),7,Math.max(3,W/190|0));
    const act=cards.filter(c=>!c.menu&&c.state!=='dead'&&c.state!=='flip').length;
    if(act<maxC)spawnCard();
    spawnT=Math.max(.85,2.4-(G.level-1)*.24)*rand(.75,1.25);}}
  for(let i=cards.length-1;i>=0;i--){const c=cards[i];
   if(c.menu){c.y+=c.vy*dt;c.rot+=c.vr*dt;if(c.y>H+120)cards.splice(i,1);continue;}
   if(c.state==='deal'){c.dealT+=dt/.38;if(c.dealT>=1){c.dealT=1;c.state='fall';}}
   else if(c.state==='fall'){c.elapsed+=dt;c.y+=c.vy*dt;if(c.elapsed>=c.life)expireCard(c);}
   else if(c.state==='flip'){c.flipT+=dt/.3;if(c.flipT>=1){burst(c);cards.splice(i,1);}}
   else if(c.state==='dead'){c.vy+=1500*dt;c.y+=c.vy*dt;c.rot+=c.deadSpin*dt;if(c.y>H+c.h*2)cards.splice(i,1);}
   c.flash=Math.max(0,c.flash-3.5*dt);}
  if(G.dying){G.overT+=dt;if(G.overT>1)finalizeOver();}}
 else if(G.state==='start'||G.state==='over'){
  menuT-=dt;const mc=cards.filter(c=>c.menu).length;
  if(menuT<=0&&mc<4){spawnMenu();menuT=rand(1.1,1.9);}
  for(let i=cards.length-1;i>=0;i--){const c=cards[i];if(!c.menu)continue;c.y+=c.vy*dt;c.rot+=c.vr*dt;if(c.y>H+120)cards.splice(i,1);}}
}
