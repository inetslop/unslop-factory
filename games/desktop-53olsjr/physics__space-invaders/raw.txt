/* ============================== BUNKERS ============================== */
function makeBunkers(){
  bunkers=[];
  const n=4,cw=22,chh=5,cell=8,bw=cw*cell,bh=chh*cell;
  const gap=(FIELD.w-n*bw)/(n+1);
  for(let i=0;i<n;i++){
    const cells=new Uint8Array(cw*chh);
    for(let r=0;r<chh;r++)for(let c=0;c<cw;c++){
      let on=true;
      if(r===0&&(c<3||c>=cw-3))on=false;
      else if(r===1&&(c<2||c>=cw-2))on=false;
      cells[r*cw+c]=on?3:0;
    }
    bunkers.push({x:FIELD.x+gap+i*(bw+gap),y:bunkerTop(),w:bw,h:bh,cw,chh,cells});
  }
}
function bunkerCellAt(x,y){
  for(const b of bunkers){
    if(x>=b.x&&x<b.x+b.w&&y>=b.y&&y<b.y+b.h){
      const c=(x-b.x)/8|0,r=(y-b.y)/8|0;
      if(b.cells[r*b.cw+c]>0)return{b,c,r};
    }
  }
  return null;
}
function chipCell(res,amt){res.b.cells[res.r*res.b.cw+res.c]=Math.max(0,res.b.cells[res.r*res.b.cw+res.c]-amt)}
function blastBunkers(x,y,r,power){
  for(const b of bunkers)for(let r2=0;r2<b.chh;r2++)for(let c2=0;c2<b.cw;c2++){
    const i=r2*b.cw+c2;if(!b.cells[i])continue;
    const cx=b.x+c2*8+4,cy=b.y+r2*8+4,d=Math.hypot(cx-x,cy-y);
    if(d<r){const dmg=power*(1-d/r*.55);if(dmg>=1||Math.random()<dmg*.5)b.cells[i]=Math.max(0,b.cells[i]-Math.ceil(Math.min(3,dmg)))}
  }
}

function burst(x,y,col,n,spd,opt){
  opt=opt||{};
  for(let i=0;i<n;i++){
    const a=Math.random()*TAU,v=rand(spd*.25,spd);
    parts.push({x,y,vx:Math.cos(a)*v,vy:Math.sin(a)*v-(opt.up||0),life:rand(opt.life||.5,(opt.life||.5)*1.6),
      max:(opt.life||.5)*1.6,s:rand((opt.size||8)*.5,(opt.size||8)*1.2),col,g:opt.g===undefined?300:opt.g,a:opt.al===undefined?1:opt.al});
  }
  if(parts.length>520)parts.splice(0,parts.length-520);
}
function smoke(x,y,n){for(let i=0;i<n;i++)parts.push({x:x+rand(-6,6),y,vx:rand(-30,30),vy:rand(-70,-15),life:rand(.4,.9),max:.9,s:rand(10,20),col:'#8a93a5',g:-40,a:.4})}
function debrisBurst(x,y,col){

}

    swarm.push({x:startX+c*52,y:startY+r*40,row:r,type,hp:type==='C'?2:1,alive:true,flash:0});
  }
  totalInv=swarm.length;swarmDir=1;bombT=rand(1.2,2);raiderT=n>=2?rand(5,8):99;
}

function addScore(base,x,y){
  if(G.tGame-lastKill<1.4)combo=Math.min(6,combo+1);else combo=1;
  lastKill=G.tGame;G.maxCombo=Math.max(G.maxCombo,combo);
  const pts=base*combo;G.score+=pts;
  if(G.score>G.hi){if(!G.newHi){G.newHi=true;}G.hi=G.score}
  popup(x,y,'+'+pts+(combo>1?' ×'+combo:''),combo>1?'#ffd166':'#eaffff',combo>=4);
}
function hitInvader(inv){

  debrisBurst(inv.x,inv.y,TCOL[inv.type]);

/* ============================== BOMBS ============================== */
function explodeBomb(b,floor){
  const r=b.brute?44:32;
  if(!floor)blastBunkers(b.x,b.y,r,b.brute?3:2);
  burst(b.x,Math.min(b.y,groundY()-4),'#ff9f43',b.brute?20:13,b.brute?340:260,{life:.55,size:13,g:250});
  smoke(b.x,Math.min(b.y,groundY()-8),6);
  sfxBoom(false);G.shake=Math.min(16,G.shake+(floor?3:b.brute?7:5));
  if(!floor&&player&&!player.dead){
    const d=Math.hypot(b.x-player.x,b.y-player.y);
    if(d<r+12)hitPlayer();
  }
}
function hitPlayer(){
  if(player.shieldT>0){
    player.shieldT=0;
    burst(player.x,player.y-4,'#7df9ff',20,320,{life:.5,size:10,g:150});
    sfxBounce(600);G.hitStop=.05;
  }else loseLife();
}

function stepFx(dt){
  for(let i=parts.length-1;i>=0;i--){const p=parts[i];p.vy+=p.g*dt;p.x+=p.vx*dt;p.y+=p.vy*dt;p.life-=dt;if(p.life<=0)parts.splice(i,1);}
  for(let i=debris.length-1;i>=0;i--){
    const d=debris[i];d.vy+=900*dt;d.x+=d.vx*dt;d.y+=d.vy*dt;d.rot+=d.vr*dt;d.life-=dt;
    if(d.y>groundY()-3){d.y=groundY()-3;d.vy*=-.55;d.vx*=.8;d.vr*=.7;}
    if(d.x<FIELD.x){d.x=FIELD.x;d.vx*=-.6}if(d.x>VW-FIELD.x){d.x=VW-FIELD.x;d.vx*=-.6}
    if(d.life<=0)debris.splice(i,1);
  }
  for(let i=pops.length-1;i>=0;i--){const p=pops[i];p.t+=dt;p.y-=32*dt;if(p.t>=p.max)pops.splice(i,1);}
}
function updatePlay(dt){
  /* player */
  const p=player;
  p.invT-=dt;p.shieldT-=dt;p.tripleT-=dt;p.cool-=dt;p.flashT-=dt;p.tiltT-=dt;
  p.recoil*=Math.exp(-10*dt);
  let ax=0;
  const left=keys.ArrowLeft||keys.KeyA||[...touches.values()].includes('left');
  const right=keys.ArrowRight||keys.KeyD||[...touches.values()].includes('right');
  const up=keys.ArrowUp||keys.KeyW,down=keys.ArrowDown||keys.KeyS;
  if(left){ax-=2600;p.lastDir=-1}if(right){ax+=2600;p.lastDir=1}
  p.vx+=ax*dt;
  p.vx*=Math.exp(-(ax?.7:7)*dt);
  p.vx=clamp(p.vx,-430,430);
  p.x+=p.vx*dt;
  if(p.x<FIELD.x+20){p.x=FIELD.x+20;p.vx=0}
  if(p.x>VW-FIELD.x-20){p.x=VW-FIELD.x-20;p.vx=0}
  if(up||down){p.elev=clamp(p.elev+(up?75:-75)*dt,45,90);p.tiltT=.8;}
  p.dirSign=p.vx>15?1:(p.vx<-15?-1:p.lastDir||1);
  const fireHeld=keys.Space||[...touches.values()].includes('fire');
  if(fireHeld&&p.cool<=0&&!p.dead){
    p.cool=p.tripleT>0?.46:.33;p.flashT=.07;p.recoil=1;sfxShoot();
    const a=p.elev*RAD,mx=p.x+Math.cos(a)*24*p.dirSign,my=p.y-8-Math.sin(a)*24;
    const mk=da=>{const an=a+da*RAD;bullets.push({x:mx,y:my,vx:Math.cos(an)*980*p.dirSign+p.vx*.22,vy:-Math.sin(an)*980,life:3})};
    if(p.tripleT>0){mk(-13);mk(0);mk(13)}else mk(0);
  }
  /* exhaust */
  if(Math.abs(p.vx)>70&&Math.random()<.7)
    parts.push({x:p.x-Math.sign(p.vx)*12+rand(-3,3),y:p.y+8,vx:-Math.sign(p.vx)*rand(40,90),vy:rand(20,60),life:.3,max:.3,s:rand(6,11),col:'#ffb35c',g:0,a:.7});

  /* swarm */
  let alive=0,minX=1e9,maxX=-1e9,maxY=-1e9;
  for(const inv of swarm)if(inv.alive){alive++;if(inv.x<minX)minX=inv.x;if(inv.x>maxX)maxX=inv.x;if(inv.y>maxY)maxY=inv.y;if(inv.flash>0)inv.flash-=dt;}
  if(alive){
    const speed=Math.min(175,(24+G.wave*6)*(0.55+1.9*(1-alive/totalInv)));
    for(const inv of swarm)if(inv.alive)inv.x+=swarmDir*speed*dt;
    let flip=false;
    for(const inv of swarm)if(inv.alive){if(swarmDir>0&&inv.x+24>VW-FIELD.x)flip=true;if(swarmDir<0&&inv.x-24<FIELD.x)flip=true;}
    if(flip){swarmDir*=-1;for(const inv of swarm)if(inv.alive){inv.x+=swarmDir>0?-4:4;inv.y+=20;}animF^=1;sfxStep();}
  }
  /* danger line */
  G.breachCd-=dt;
  if(alive&&maxY+16>=dangerY()&&G.breachCd<=0)breach();

  /* bomb drops */
  bombT-=dt;
  const interval=clamp(1.7-G.wave*.1-(1-alive/totalInv)*.9,.38,2);
  if(bombT<=0&&alive>0){
    bombT=interval*rand(.7,1.3);
    let pick=null;
    if(Math.random()<.6){const by=maxY;let cands=swarm.filter(i=>i.alive&&Math.abs(i.y-by)<2);if(cands.length)pick=cands[Math.random()*cands.length|0];}
    if(!pick){const cands=swarm.filter(i=>i.alive);pick=cands[Math.random()*cands.length|0];}
    const sp=Math.min(175,(24+G.wave*6));
    bombs.push({x:pick.x,y:pick.y+14,vx:swarmDir*sp*.5+rand(-18,18),vy:30,rad:pick.type==='C'?7:5,brute:pick.type==='C',t:0,bounces:0});
    sfxBombDrop();
  }

  /* raiders */
  if(G.wave>=2&&alive>0){
    raiderT-=dt;
    if(raiderT<=0&&raiders.length<2){
      raiderT=rand(7,11);
      const side=Math.random()<.5?-1:1;
      raiders.push({x:side<0?-24:VW+24,y:rand(FIELD.y+40,FIELD.y+190),vx:(side<0?1:-1)*(230+G.wave*8),vy:0,t:0});
      sfxAlarm();popup(side<0?60:VW-60,120,'!','#ff5c5c',true);
    }
  }
  for(let i=raiders.length-1;i>=0;i--){
    const r=raiders[i];r.vy=Math.min(560,r.vy+260*dt);r.x+=r.vx*dt;r.y+=r.vy*dt;r.t+=dt;
    if(Math.random()<.8)parts.push({x:r.x-Math.sign(r.vx)*10,y:r.y,vx:-r.vx*.1,vy:rand(-20,20),life:.25,max:.25,s:9,col:'#ff6b4a',g:0,a:.8});
    if(!p.dead&&Math.abs(r.x-p.x)<20&&Math.abs(r.y-p.y)<18){burst(r.x,r.y,'#ff6b4a',16,300,{life:.5,size:12,g:300});hitPlayer();raiders.splice(i,1);continue;}
    if(r.y>groundY()-6){burst(r.x,groundY()-8,'#ff6b4a',14,260,{life:.5,size:11,g:400});smoke(r.x,groundY()-8,5);sfxBoom(false);G.shake=Math.min(14,G.shake+4);raiders.splice(i,1);}
  }

  /* bullets */
  for(let i=bullets.length-1;i>=0;i--){
    const b=bullets[i];b.vy+=960*dt;b.x+=b.vx*dt;b.y+=b.vy*dt;b.life-=dt;
    if(Math.random()<.5)parts.push({x:b.x,y:b.y,vx:0,vy:0,life:.13,max:.13,s:8,col:'#6ee7ff',g:0,a:.5});
    let dead=false;
    if(b.life<=0||b.x<FIELD.x-4||b.x>VW-FIELD.x+4||b.y<-50)dead=true;
    if(!dead&&b.y>groundY()-2){smoke(b.x,groundY()-4,3);burst(b.x,groundY()-4,'#6ee7ff',4,120,{life:.3,size:7,g:200});dead=true;}
    if(!dead){const cell=bunkerCellAt(b.x,b.y);if(cell){chipCell(cell,1);sfxChip();burst(b.x,b.y,'#c9e6a8',4,140,{life:.25,size:7,g:300});dead=true;}}
    if(!dead)for(const inv of swarm){
      if(!inv.alive)continue;
      const hw=inv.type==='C'?27:23,hh=inv.type==='C'?18:16;
      if(Math.abs(b.x-inv.x)<hw&&Math.abs(b.y-inv.y)<hh){hitInvader(inv);dead=true;break;}
    }
    if(!dead)for(let j=raiders.length-1;j>=0;j--){
      const r=raiders[j];
      if(Math.hypot(r.x-b.x,r.y-b.y)<20){killRaider(r);raiders.splice(j,1);dead=true;break;}
    }
    if(dead)bullets.splice(i,1);
  }

  /* bombs */
  for(let i=bombs.length-1;i>=0;i--){
    const b=bombs[i];b.vy=Math.min(780,b.vy+430*dt);b.x+=b.vx*dt;b.y+=b.vy*dt;b.t+=dt;
    if(b.t>.05){b.t=0;parts.push({x:b.x,y:b.y,vx:0,vy:0,life:.32,max:.32,s:b.brute?16:11,col:'#ff9f43',g:0,a:.7});}
    if(b.x-b.rad<FIELD.x){b.x=FIELD.x+b.rad;b.vx*=-.5}
    if(b.x+b.rad>VW-FIELD.x){b.x=VW-FIELD.x-b.rad;b.vx*=-.5}
    let dead=false;
    const cell=bunkerCellAt(b.x,b.y);
    if(cell&&!dead){
      const bb=cell.b,cx0=bb.x+cell.c*8,cy0=bb.y+cell.r*8,sp=Math.hypot(b.vx,b.vy);
      if(sp<75||b.bounces>3)explodeBomb(b,false);
      else{
        const pl=b.x-cx0,pr=cx0+8-b.x,pt=b.y-cy0,pb=cy0+8-b.y,m=Math.min(pl,pr,pt,pb);
        if(m===pl){b.x=cx0-b.rad;b.vx=-Math.abs(b.vx)*.42;b.vy*=.72}
        else if(m===pr){b.x=cx0+8+b.rad;b.vx=Math.abs(b.vx)*.42;b.vy*=.72}
        else if(m===pt){b.y=cy0-b.rad;b.vy=-Math.abs(b.vy)*.42;b.vx*=.72}
        else{b.y=cy0+8+b.rad;b.vy=Math.abs(b.vy)*.42;b.vx*=.72}
        chipCell(cell,1);sfxBounce(sp);burst((b.x),(b.y),'#c9e6a8',4,150,{life:.25,size:7,g:300});b.bounces++;
      }
    }
    if(!dead&&b.y+b.rad>groundY()){explodeBomb(b,true);dead=true;}
    if(!dead&&!player.dead){
      if(Math.abs(b.x-player.x)<16&&b.y>player.y-14&&b.y<player.y+10){
        hitPlayer();burst(b.x,b.y,'#ff9f43',8,200,{life:.4,size:9,g:300});dead=true;
      }
    }
    if(dead)bombs.splice(i,1);
  }

  /* powerups */
  for(let i=powerups.length-1;i>=0;i--){
    const u=powerups[i];u.vy=Math.min(230,u.vy+380*dt);u.y+=u.vy*dt;u.x+=Math.sin(u.t*4+u.seed)*34*dt;u.t+=dt;
    if(!player.dead&&Math.abs(u.x-player.x)<26&&Math.abs(u.y-player.y)<24){
      if(u.type==='T'){player.tripleT=9;sfxPower();popup(player.x,player.y-30,'TRIPLE SHOT','#6ee7ff',true);}
      else if(u.type==='S'){player.shieldT=7;sfxPower();popup(player.x,player.y-30,'SHIELD','#ffd166',true);}
      else{G.lives=Math.min(5,G.lives+1);sfxLife();popup(player.x,player.y-30,'+1 LIFE','#ff8fb0',true);}
      powerups.splice(i,1);continue;
    }
    if(u.y>groundY()-6){smoke(u.x,groundY()-6,4);powerups.splice(i,1);}
  }

  stepFx(dt);
  if(combo>0&&G.tGame-lastKill>1.4)combo=0;

  /* wave clear */
  let anyAlive=false;for(const inv of swarm)if(inv.alive){anyAlive=true;break;}
  if(!anyAlive&&G.transition<=0){
    G.transition=1.6;const bonus=G.wave*100;G.score+=bonus;if(G.score>G.hi){G.newHi=true;G.hi=G.score}
    popup(VW/2,VH*.42,'+'+bonus+' BONUS','#ffd166',true);
    sfxWave();
    for(const b of bunkers)for(let j=0;j<b.cells.length;j++)b.cells[j]=Math.min(3,b.cells[j]+1);
  }
  if(G.transition>0){
    G.transition-=dt;
    if(G.transition<=0){G.wave++;spawnWave(G.wave);G.banner={txt:'WAVE '+pad(G.wave,2),sub:G.wave>=3?'THEY FALL FASTER':'',t:0,life:1.6};}
  }
  if(G.banner){G.banner.t+=dt;if(G.banner.t>G.banner.life)G.banner=null;}
  musicUpdate();

function rr(x,y,w,h,r){ctx.beginPath();ctx.moveTo(x+r,y);ctx.lineTo(x+w-r,y);ctx.arcTo(x+w,y,x+w,y+r,r);ctx.lineTo(x+w,y+h-r);ctx.arcTo(x+w,y+h,x+w-r,y+h,r);ctx.lineTo(x+r,y+h);ctx.arcTo(x,y+h,x,y+h-r,r);ctx.lineTo(x,y+r);ctx.arcTo(x,y,x+r,y,r);ctx.closePath();}
function spaced(txt,x,y,sp,align){
  if(!txt.length)return 0;
  const ws=[];let w=0;
  for(const ch of txt){const m=ctx.measureText(ch).width;ws.push(m);w+=m+sp;}
  w-=sp;
  if(align==='center')x-=w/2;else if(align==='right')x-=w;
  let cx=x;
  for(let i=0;i<txt.length;i++){ctx.fillText(txt[i],cx,y);cx+=ws[i]+sp;}
  return w;
}
