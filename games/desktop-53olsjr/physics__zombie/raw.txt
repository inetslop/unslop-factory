function physStep(dt){

      else if(z.state==='chomp'&&(i===3||i===4))ax=-90;

  }
}

/* ============ step (fixed) ============ */
function step(dt){
  time+=dt;shake*=Math.pow(.02,dt);
  for(const z of zombies)zState(z,dt);
  physStep(dt);
  projStep(dt);
  zombies=zombies.filter(z=>!z.gone);
  for(let i=parts.length-1;i>=0;i--){const p=parts[i];p.l-=dt;
    if(p.l<=0){parts.splice(i,1);continue;}
    if(p.g)p.vy+=900*dt;
    p.x+=p.vx*dt;p.y+=p.vy*dt;
    if(p.y>GROUND+p.r){p.y=GROUND+p.r;p.vy*=-.3;p.vx*=.7;
      if(Math.abs(p.vy)<60&&!p.d){p.d=true;if(decals.length<60)decals.push({x:p.x,y:GROUND+2,r:p.r*1.8,l:.8,c:p.c});}
    }}
  if(parts.length>320)parts.splice(0,parts.length-320);
  for(let i=decals.length-1;i>=0;i--){decals[i].l-=dt*.1;if(decals[i].l<=0)decals.splice(i,1);}
  for(let i=texts.length-1;i>=0;i--){const t=texts[i];t.y+=t.vy*dt;t.l-=dt;if(t.l<=0)texts.splice(i,1);}
  if(state==='play'){
    regenT+=dt;if(regenT>3&&brains<12){regenT=0;brains++;}
    if(!cleared){
      if(toSpawn>0){spawnT-=dt;if(spawnT<=0){zombies.push(makeZombie());toSpawn--;spawnT=(wave===1?2.4:1.7)*(.7+Math.random()*.6);}}
      else if(zombies.length===0){
        cleared=true;gapT=3;const bonus=50*wave;score+=bonus;
        wall.hp=Math.min(wall.max,wall.hp+8);SFX.clear();
        banner('WAVE '+wave+' CLEARED','+'+bonus+' · BARRICADE BRACED');
      }
    }else{gapT-=dt;if(gapT<=0){cleared=false;nextWave();}}
  }
}

  if(state==='play'&&wall.hp<wall.max*.25){
    ctx.fillStyle='rgba(178,58,46,'+(.07+.05*Math.sin(vt*6))+')';
    ctx.fillRect(-30,-30,GW+60,620);
  }
  ctx.restore();
  if(vig)ctx.drawImage(vig,0,0,W,H);
}

function syncHUD(){
  const f=Math.max(0,wall.hp/wall.max);
  wallFill.style.width=(f*100)+'%';
  wallFill.style.background=f>.5?'#8fae4c':f>.25?'#c99a3d':'#b23a2e';

/* ============ main loop ============ */
function frame(t){
  requestAnimationFrame(frame);
  const now=t/1000;let dt=now-last;last=now;
  if(dt>0.1)dt=0.1;if(dt<0)dt=0;
  vt+=dt;
  if((state==='play'&&!paused)||state==='over'){
    acc+=dt;let n=0;
    while(acc>=DT&&n<5){step(DT);acc-=DT;n++;}
    if(n===5)acc=0;
  }else{
    for(const f of fogs){f.x-=f.v*dt;if(f.x<-f.r)f.x=GW+f.r;}
  }
  syncHUD();draw();
}
