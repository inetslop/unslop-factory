function ensureAudio(){

function getNoise(){
  if(nbuf) return nbuf;
  const len=Math.floor(AC.sampleRate*0.5); nbuf=AC.createBuffer(1,len,AC.sampleRate);
  const d=nbuf.getChannelData(0); for(let i=0;i<len;i++) d[i]=Math.random()*2-1; return nbuf;
}
function tone(f,dur,type,vol,slideTo,delay){
  if(!AC||muted) return; type=type||'sine'; vol=vol||.3; delay=delay||0;
  const t0a=AC.currentTime+delay, o=AC.createOscillator(), g=AC.createGain();
  o.type=type; o.frequency.setValueAtTime(f,t0a);
  if(slideTo) o.frequency.exponentialRampToValueAtTime(Math.max(30,slideTo),t0a+dur);
  g.gain.setValueAtTime(0.0001,t0a);
  g.gain.exponentialRampToValueAtTime(vol,t0a+0.012);
  g.gain.exponentialRampToValueAtTime(0.0001,t0a+dur);
  o.connect(g); g.connect(master); o.start(t0a); o.stop(t0a+dur+0.06);
}
function noise(dur,vol,freq,type,delay){
  if(!AC||muted) return; type=type||'lowpass'; delay=delay||0;
  const t0a=AC.currentTime+delay, src=AC.createBufferSource();
  src.buffer=getNoise(); src.loop=true;
  const f=AC.createBiquadFilter(); f.type=type; f.frequency.value=freq; f.Q.value=1;
  const g=AC.createGain();
  g.gain.setValueAtTime(0.0001,t0a);
  g.gain.exponentialRampToValueAtTime(vol,t0a+0.012);
  g.gain.exponentialRampToValueAtTime(0.0001,t0a+dur);
  src.connect(f); f.connect(g); g.connect(master); src.start(t0a); src.stop(t0a+dur+0.06);
}

function parseLevel(map){
  const H=map.length; let W=0; map.forEach(r=>W=Math.max(W,r.length));
  const walls=new Set(), targets=new Set();
  const boxes=[]; let P=null;
  for(let y=0;y<H;y++)for(let x=0;x<W;x++){
    const ch=(map[y][x]!==undefined)?map[y][x]:' '; const key=x+','+y;
    if(ch==='#'){ walls.add(key); continue; }
    if(ch==='.'||ch==='*'||ch==='+') targets.add(key);
    if(ch==='$'||ch==='*') boxes.push({x,y,wobT:-9e9,popT:-9e9});
    if(ch==='@'||ch==='+') P={x,y};
  }
  const inside=new Set([P.x+','+P.y]); const q=[[P.x,P.y]];
  while(q.length){
    const c=q.pop();
    for(const d of DIRS4){
      const nx=c[0]+d[0], ny=c[1]+d[1];
      if(nx<0||ny<0||nx>=W||ny>=H) continue;
      const key=nx+','+ny;
      if(walls.has(key)||inside.has(key)) continue;
      inside.add(key); q.push([nx,ny]);
    }
  }
  return { W,H,walls,targets,boxes,
    wallsList:[...walls].map(k=>k.split(',').map(Number)),
    targetsList:[...targets].map(k=>k.split(',').map(Number)),
    insideList:[...inside].map(k=>k.split(',').map(Number)) };
}
function rebuild(){ boxKey=new Map(G.boxes.map((b,i)=>[b.x+','+b.y,i])); }

function queueDir(dx,dy){
  if(status!=='play') return;

    const p=anim.t/anim.dur;

}

function tryMove(dx,dy){
  if(t0===null) t0=performance.now();
  const nx=player.x+dx, ny=player.y+dy;
  if(isWall(nx,ny)){ startBump(dx,dy); return; }
  const bi=(boxKey.get(nx+','+ny))??-1;
  if(bi>=0){
    const bx=nx+dx, by=ny+dy;
    if(isWall(bx,by)||(boxKey.has(bx+','+by))){ startBump(dx,dy); return; }
    hist.push({px:player.x,py:player.y,b:bi,dx,dy});
    moves++; pushes++;
    const wasOn=G.targets.has(nx+','+ny), nowOn=G.targets.has(bx+','+by);
    anim={type:'push',fx:player.x,fy:player.y,tx:nx,ty:ny,boxI:bi,bfx:nx,bfy:ny,btx:bx,bty:by,t:0,dur:195};
    player.x=nx; player.y=ny; G.boxes[bi].x=bx; G.boxes[bi].y=by; rebuild();
    // dust behind the crate
    for(let i=0;i<6;i++){
      const a=Math.atan2(-dy,-dx)+(Math.random()-.5)*1.4, sp=.8+Math.random()*1.6;
      particles.push({kind:'dust',x:nx-dx*.4+(Math.random()-.5)*.3,y:ny-dy*.4+(Math.random()-.5)*.3,
        vx:Math.cos(a)*sp,vy:Math.sin(a)*sp,life:.45+Math.random()*.25,tl:.7});
    }
    G.boxes[bi].wobT=performance.now();
    if(nowOn){ sfx.place(); spawnPlaceFX(bx,by); G.boxes[bi].popT=performance.now(); }
    else if(wasOn) sfx.unplace();
    else sfx.push();
  }else{
    hist.push({px:player.x,py:player.y,b:-1,dx,dy});
    moves++;
    anim={type:'move',fx:player.x,fy:player.y,tx:nx,ty:ny,t:0,dur:135};
    player.x=nx; player.y=ny;
    sfx.step();
  }
  updateHUD();
  if(checkWin()) onWin();
}

    particles.push({kind:'spark',x,y,vx:Math.cos(a)*sp,vy:Math.sin(a)*sp,life:.5,tl:.5});
  }
}

  if(e.b>=0){ const b=G.boxes[e.b]; b.x-=e.dx; b.y-=e.dy; rebuild(); }
  moves--; if(e.b>=0) pushes--;
  sfx.undo(); updateHUD();
}

}

    ctx.fillStyle=filled?'rgba(255,236,180,.95)':'rgba(255,209,102,.75)'; circle(px,py,c*.045);
  }
  /* wall shadows then walls */
  for(const [x,y] of G.wallsList){
    if(G.inside.has(x+','+(y+1))){
      const px=ox+x*c, py=oy+y*c+c;
      g=ctx.createLinearGradient(0,py,0,py+c*.3);
      g.addColorStop(0,'rgba(0,0,0,.4)'); g.addColorStop(1,'rgba(0,0,0,0)');
      ctx.fillStyle=g; ctx.fillRect(px,py,c,c*.3);
    }
  }
  for(const [x,y] of G.wallsList){
    const px=ox+x*c, py=oy+y*c;
    ctx.fillStyle='#2e5470'; ctx.fillRect(px,py,c,c);
    ctx.fillStyle='#41708f'; ctx.fillRect(px,py,c,c*.16);
    ctx.fillStyle='#1d3a4e'; ctx.fillRect(px,py+c*.82,c,c*.18);
    ctx.fillStyle='rgba(255,255,255,.05)'; ctx.fillRect(px,py,c*.1,c);
  }
  /* boxes */
  for(let i=0;i<G.boxes.length;i++){
    const b=G.boxes[i]; let bx=b.x, by=b.y;
    if(anim&&anim.type==='push'&&anim.boxI===i){
      const e=ease(Math.min(1,anim.t/anim.dur));
      bx=anim.bfx+(anim.btx-anim.bfx)*e; by=anim.bfy+(anim.bty-anim.bfy)*e;
    }
    drawBox(ox+(bx+.5)*c, oy+(by+.5)*c, G.targets.has(b.x+','+b.y), b, now);
  }
  /* player */
  let ppx=player.x, ppy=player.y, phase=0, walking=false;
  if(anim){
    const p=Math.min(1,anim.t/anim.dur);
    if(anim.type==='bump'){ ppx+=anim.dx*Math.sin(p*Math.PI)*.2; ppy+=anim.dy*Math.sin(p*Math.PI)*.2; }
    else{ const e=ease(p); ppx=anim.fx+(anim.tx-anim.fx)*e; ppy=anim.fy+(anim.ty-anim.fy)*e; phase=p; walking=true; }
  }
  drawPlayer(ox+(ppx+.5)*c, oy+(ppy+.5)*c, c, phase, walking, now);
  /* particles (grid space) */
  for(const p of particles){
    const q=1-p.life/p.tl, px=ox+p.x*c, py=oy+p.y*c, a=p.life/p.tl;
    if(p.kind==='dust'){
      ctx.globalAlpha=a*.4; ctx.fillStyle='#cdb392'; circle(px,py,c*(.05+(1-a)*.06));
    }else if(p.kind==='spark'){
      ctx.globalAlpha=a; ctx.strokeStyle='#ffd166'; ctx.lineWidth=1.5;
      const L=c*.14*a;
      ctx.beginPath(); ctx.moveTo(px-L,py); ctx.lineTo(px+L,py); ctx.moveTo(px,py-L); ctx.lineTo(px,py+L); ctx.stroke();
    }else if(p.kind==='ring'){
      ctx.globalAlpha=a; ctx.strokeStyle='#ffd166'; ctx.lineWidth=2;
      ctx.beginPath(); ctx.arc(px,py,(.2+q*.55)*c,0,Math.PI*2); ctx.stroke();
    }
  }
  ctx.globalAlpha=1;
  ctx.restore();

  /* vignette + confetti */
  if(vig) ctx.drawImage(vig,0,0);
  for(const f of confetti){
    ctx.save(); ctx.globalAlpha=Math.min(1,f.life/.6);
    ctx.translate(f.x,f.y); ctx.rotate(f.rot);
    ctx.fillStyle=f.col; ctx.fillRect(-f.w/2,-f.h/2,f.w,f.h); ctx.restore();
  }
}

function drawBox(cx,cy,onT,b,now){
  const s=cellNow*.82;
  const wobAge=now-b.wobT, popAge=now-b.popT;
  const rot=(wobAge<260)?Math.sin(wobAge/260*Math.PI*2)*.07*(1-wobAge/260):0;
  const sc=(popAge<300)?1+.16*Math.sin(popAge/300*Math.PI):1;
  ctx.save(); ctx.translate(cx,cy); ctx.rotate(rot); ctx.scale(sc,sc);
  if(onT){
    const g=ctx.createRadialGradient(0,0,s*.1,0,0,s*.8);
    g.addColorStop(0,'rgba(255,209,102,.35)'); g.addColorStop(1,'rgba(255,209,102,0)');

  }
  ctx.fillStyle='rgba(0,0,0,.3)';

  ctx.fillStyle=onT?'#e6a94c':'#c78d3f';

let last=performance.now(), shownSec=-1;
function loop(ts){
  const dt=Math.min(.05,(ts-last)/1000); last=ts; cellNow=metrics().c;
  if(anim){
    anim.t+=dt*1000;
    if(anim.t>=anim.dur){ anim=null; if(pending){ const d=pending; pending=null; tryMove(d[0],d[1]); } }
  }
  shake*=Math.pow(.02,dt); if(shake<.3) shake=0;
  for(let i=particles.length-1;i>=0;i--){
    const p=particles[i]; p.life-=dt;
    if(p.kind!=='ring'){ p.x+=p.vx*dt; p.y+=p.vy*dt; p.vx*=.92; p.vy*=.92; }
    if(p.life<=0) particles.splice(i,1);
  }
  if(particles.length>320) particles.splice(0,particles.length-320);
  for(let i=confetti.length-1;i>=0;i--){
    const f=confetti[i]; f.life-=dt;
    f.vy+=520*dt; f.x+=f.vx*dt; f.y+=f.vy*dt; f.rot+=f.vr*dt;
    if(f.life<=0||f.y>H+30) confetti.splice(i,1);
  }
  draw(ts);
  const secs=t0?Math.floor(((status==='play'?ts:tEnd)-t0)/1000):0;
  if(secs!==shownSec){ shownSec=secs; $('stTime').textContent=fmt(secs); }
  requestAnimationFrame(loop);
}
