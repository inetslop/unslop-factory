function solidsNear(p){
  const R=[];
  const c0=Math.floor((p.x-34)/T),c1=Math.floor((p.x+p.w2+34)/T);
  const r0=Math.floor((p.y-34)/T),r1=Math.floor((p.y+p.h+34)/T);
  for(let r=r0;r<=r1;r++)for(let c=c0;c<=c1;c++){

  }
  for(const m of L.movers)R.push({x:m.cx-m.w/2,y:m.cy-m.h/2,w:m.w,h:m.h,m});

}
function overlap(a,b){return a.x<b.x+b.w&&a.x+a.w>b.x&&a.y<b.y+b.h&&a.y+a.h>b.y}

function physics(dt){
  const p=player;p.w2=PW;p.h=PH;
  if(p.ride){p.x+=p.ride.dx;p.y+=p.ride.dy}
  const left=keys['arrowleft']||keys['a']||tIn.L,right=keys['arrowright']||keys['d']||tIn.R;
  const ax=(right?1:0)-(left?1:0);
  if(ax)p.facing=ax;
  const acc=p.grounded?3800:2500;
  p.vx+=clamp(ax*SPEED-p.vx,-acc*dt,acc*dt);
  jumpBuf-=dt;p.coyote-=dt;
  if(jumpBuf>0&&(p.grounded||p.coyote>0))doJump();
  p.vy+=p.gd*G*dt;p.vy=clamp(p.vy,-1300,1300);
  p.landT=Math.max(0,p.landT-dt);p.jumpT=Math.max(0,p.jumpT-dt);

  const R=solidsNear(p);
  // X
  p.x+=p.vx*dt;
  for(const s of R)if(overlap(p,s)){
    if(p.vx>0)p.x=s.x-PW;else if(p.vx<0)p.x=s.x+s.w;p.vx=0;
  }
  p.x=clamp(p.x,0,L.w*T-PW);
  // Y
  const wasG=p.grounded;const fallV=Math.abs(p.vy);
  p.y+=p.vy*dt;p.grounded=false;
  for(const s of R)if(overlap(p,s)){
    if(p.vy>0){p.y=s.y-PH;if(p.gd>0)p.grounded=true}
    else if(p.vy<0){p.y=s.y+s.h;if(p.gd<0)p.grounded=true}
    p.vy=0;
  }
  if(p.grounded){
    p.coyote=.1;
    if(!wasG){p.landT=.18;if(fallV>420){sfx.land();for(let i=0;i<6;i++)dust(p.x+PW/2+(Math.random()-.5)*18,p.y+(p.gd>0?PH:0))}}
  }
  // ride detect
  p.ride=null;
  if(p.grounded)for(const m of L.movers){
    if(m.spike)continue;const mr={x:m.cx-m.w/2,y:m.cy-m.h/2,w:m.w,h:m.h};
    if(p.x+PW>mr.x+2&&p.x<mr.x+mr.w-2){
      if(p.gd>0&&Math.abs((p.y+PH)-mr.y)<9)p.ride=m;
      else if(p.gd<0&&Math.abs(p.y-(mr.y+mr.h))<9)p.ride=m;
    }
  }
  // run dust / trail
  p.dustT-=dt;
  if(Math.abs(p.vx)>230&&p.grounded&&p.dustT<=0){p.dustT=.1;dust(p.x+PW/2-p.facing*10,p.y+(p.gd>0?PH:0))}
  if(Math.abs(p.vy)>560&&Math.random()<.5)particles.push({x:p.x+PW/2,y:p.y+PH/2,vx:0,vy:0,life:.25,max:.25,sz:5,c:'rgba('+pal().accRGB.join(',')+',.5)',ty:1});

  // hazards
  const hb={x:p.x+3,y:p.y+3,w:PW-6,h:PH-6};
  for(const s of L.spikes)if(Math.abs(s.x-cam.x)<W/ZOOM&&overlap(hb,s)){die();return}
  for(const lz of L.lasers){const at=state==='menu'?wallT:simT;if(laserOn(lz,at))
    if(overlap(hb,{x:lz.c*T+12,y:lz.rT*T,w:8,h:(lz.rB-lz.rT+1)*T})){die();return}}
  for(const m of L.movers)if(m.spike&&overlap(hb,{x:m.cx-m.w/2,y:m.cy-m.h/2,w:m.w,h:m.h})){die();return}
  if(p.y>WORLD_H+70||p.y<-70){die();return}

  // cores
  const pcx=p.x+PW/2,pcy=p.y+PH/2;
  for(const c of L.cores)if(!c.taken){
    const dx=pcx-c.x,dy=pcy-c.y;
    if(dx*dx+dy*dy<26*26){c.taken=true;sfx.core();burst(c.x,c.y,10,'rgba(255,233,138,.95)',240);ring(c.x,c.y,'#ffe98a')}
  }
  // pads
  for(const pd of L.pads){
    if(pd.cd>0)continue;
    if(Math.abs(pcx-pd.x)<18&&Math.abs(pcy-pd.y)<22){pd.cd=.5;pd.anim=1;doFlip()}
  }
  // exit
  if(L.exit&&overlap(p,{x:L.exit.x-13,y:L.exit.y-20,w:26,h:44}))sectorClear();
}

function updateMovers(at){

function burst(x,y,n,c,sp){for(let i=0;i<n;i++){const a=Math.random()*6.283,v=sp*(.3+Math.random()*.7);particles.push({x,y,vx:Math.cos(a)*v,vy:Math.sin(a)*v,life:.5,max:.5,sz:2+Math.random()*3,c,ty:0})}}

function updateParticles(dt){
  for(let i=particles.length-1;i>=0;i--){
    const p=particles[i];p.life-=dt;
    if(p.life<=0){particles.splice(i,1);continue}
    if(p.ty===2)p.vy+=60*dt;
    p.x+=p.vx*dt;p.y+=p.vy*dt;
  }
  if(particles.length>380)particles.splice(0,particles.length-380);
}

let wallT=0;
function tick(dt,t){
  wallT=t;
  shake=Math.max(0,shake-dt*32);flash.a=Math.max(0,flash.a-dt*2.4);
  if(state==='menu'){
    const vw=W/ZOOM,lw=L.w*T;
    cam.x=lw>vw?((t*55)%(lw-vw)):lw/2;
    palP+=( (Math.sin(t*.45)>0?1:0)-palP )*Math.min(1,dt*1.6);
    updateMovers(t);updateParticles(dt);return;
  }
  if(paused)return;
  simT+=dt;
  for(const pd of L.pads){pd.cd=Math.max(0,pd.cd-dt);pd.anim=Math.max(0,pd.anim-dt*2.4)}
  updateMovers(simT);updateParticles(dt);

  if(state==='clear'){
    clearT-=dt;player.absorb=Math.min(1,player.absorb+dt*1.9);
    if(clearT<=0)nextSector();return;
  }
  runTime+=dt;flipCd=Math.max(0,flipCd-dt);
  if(player.dead){
    player._dt=(player._dt||0)-dt;
    if(player._dt===undefined)player._dt=.75;
    if(player._dt<=0)respawn();
  }else physics(dt);

  const vw=W/ZOOM,lw=L.w*T;
  const tc=clamp(player.x+PW/2+player.facing*64,vw/2,lw-vw/2);
  cam.x+=(tc-cam.x)*Math.min(1,dt*6);
  palP+=(palTarget-palP)*Math.min(1,dt*7);
}
