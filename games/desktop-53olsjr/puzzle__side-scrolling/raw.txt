/* ---------------- solid / collision ---------------- */
function isSolid(tx,ty){
  if(tx<0||tx>=LV.w)return true;
  if(ty<0||ty>=ROWS)return false;
  const c=LV.grid[ty][tx];
  if(c==='#')return true;
  if(c==='-'){const s=crumbles.get(tx+','+ty);return !(s&&s.st==='gone');}
  if(c==='A')return !LV.open[0];
  if(c==='B')return !LV.open[1];
  return false;
}
function collideX(){
  const y0=Math.floor(p.y/T),y1=Math.floor((p.y+PH-.01)/T);
  if(p.vx>0){const tx=Math.floor((p.x+PW)/T);
    for(let ty=y0;ty<=y1;ty++)if(isSolid(tx,ty)){p.x=tx*T-PW;p.vx=0;break;}}
  else if(p.vx<0){const tx=Math.floor(p.x/T);
    for(let ty=y0;ty<=y1;ty++)if(isSolid(tx,ty)){p.x=(tx+1)*T;p.vx=0;break;}}
}
function collideY(){
  p.grounded=false;
  const x0=Math.floor((p.x+2)/T),x1=Math.floor((p.x+PW-2)/T);
  if(p.vy>0){const ty=Math.floor((p.y+PH)/T);
    for(let tx=x0;tx<=x1;tx++)if(isSolid(tx,ty)){onLand(p.vy);p.y=ty*T-PH;p.vy=0;if(p.g===1)p.grounded=true;break;}}
  else if(p.vy<0){const ty=Math.floor(p.y/T);
    for(let tx=x0;tx<=x1;tx++)if(isSolid(tx,ty)){onLand(-p.vy);p.y=(ty+1)*T;p.vy=0;if(p.g===-1)p.grounded=true;break;}}
}
function onLand(impact){
  if(impact>360){p.squash=.32;
    for(let i=0;i<7;i++)particles.push({x:p.x+PW/2+(Math.random()-.5)*14,y:p.y+(p.g===1?PH:0),
      vx:(Math.random()-.5)*90,vy:-p.g*(20+Math.random()*50),t:0,life:.5,size:2.2,c:'dust'});
    if(impact>520)shake(.18,4);}
}
function shake(t,m){shakeT=t;shakeDur=t;shakeMag=m;}

function triggerCrumble(tx,ty){
  const k=tx+','+ty;if(crumbles.has(k))return;

}
function checkSupport(){
  if(!p.grounded)return;

  [Math.floor((p.x+4)/T),Math.floor((p.x+PW-4)/T)].forEach(tx=>{
    if(LV.grid[ty]&&LV.grid[ty][tx]==='-')triggerCrumble(tx,ty);});
}
function updateCrumbles(dt){
  crumbles.forEach((s,k)=>{if(s.st==='shake'){s.t-=dt;

}

/* ---------------- platforms (lantern lifts) ---------------- */
function updatePlats(dt){
  for(const pl of LV.plats){pl.prevY=pl.y;pl.y+=pl.spd*pl.dir*dt;
    if(pl.y<=pl.yMin){pl.y=pl.yMin;pl.dir=1;}if(pl.y>=pl.yMax){pl.y=pl.yMax;pl.dir=-1;}
    pl.dy=pl.y-pl.prevY;}
}
function checkPlats(prevTop,prevBot){
  p.ride=null;
  for(const pl of LV.plats){
    if(p.x+PW<=pl.x||p.x>=pl.x+pl.w)continue;
    if(p.g===1&&p.vy>=0&&prevBot<=pl.prevY+8&&p.y+PH>=pl.y){p.y=pl.y-PH;p.vy=0;p.grounded=true;p.ride=pl;}
    else if(p.g===-1&&p.vy<=0&&prevTop>=pl.prevY+14-8&&p.y<=pl.y+14){p.y=pl.y+14;p.vy=0;p.grounded=true;p.ride=pl;}
  }
}

/* ---------------- spikes / bounds ---------------- */
function spikeHit(){
  const bx=p.x+3,by=p.y+3,bw=PW-6,bh=PH-6;
  const x0=Math.floor(bx/T),x1=Math.floor((bx+bw)/T),y0=Math.floor(by/T),y1=Math.floor((by+bh)/T);
  for(let ty=y0;ty<=y1;ty++)for(let tx=x0;tx<=x1;tx++){
    if(ty<0||ty>=ROWS)continue;const c=LV.grid[ty][tx];
    if(c==='^'||c==='v'){const sx=tx*T+3,sy=ty*T+3;
      if(bx<sx+T-6&&bx+bw>sx&&by<sy+T-6&&by+bh>sy)return true;}}
  return false;
}

/* ---------------- player update ---------------- */
function updatePlayer(dt){
  p.flipCd=Math.max(0,p.flipCd-dt);p.squash=Math.max(0,p.squash-dt*2.2);
  p.wingT+=dt*(7+Math.abs(p.vy)/45+Math.abs(p.vx)/60);
  if(p.ride)p.y+=p.ride.dy;
  const dir=(keys.R?1:0)-(keys.L?1:0);
  if(dir){p.vx+=dir*MOVE_ACC*dt;p.face=dir;}
  else{const f=(p.grounded?FRICTION:380)*dt;if(Math.abs(p.vx)<=f)p.vx=0;else p.vx-=Math.sign(p.vx)*f;}
  p.vx=clamp(p.vx,-MOVE_MAX,MOVE_MAX);
  p.x+=p.vx*dt;p.x=clamp(p.x,4,LV.w*T-4-PW);collideX();
  const prevTop=p.y,prevBot=p.y+PH;
  p.vy=clamp(p.vy+G_ACC*p.g*dt,-MAXFALL,MAXFALL);
  p.y+=p.vy*dt;collideY();checkPlats(prevTop,prevBot);
  checkSupport();
  if(spikeHit()){die();return;}
  if(p.y>ROWS*T+70||p.y<-90){die();return;}
  checkPads();checkMotes();checkExit();
}

  }
}

}

function drawPlayer(cx,cy,scale,rot,alpha,t){
  const sy=(1-p.squash+.0)*scale*(state==='play'?clamp(1+p.vy/4000,-.1,.15):1);
  const sx=scale*(1+p.squash*.7);

  const flap=Math.sin(p.wingT*3.2)*(.5+Math.min(.6,Math.abs(p.vy)/500));
  ctx.fillStyle='rgba(255,244,214,.34)';
  for(const s of [-1,1]){ctx.save();ctx.translate(s*9,-2);ctx.rotate(s*(-.7+flap*.8));
    ctx.beginPath();ctx.ellipse(0,-6,4.5,10,0,0,6.283);ctx.fill();ctx.restore();}
  ctx.globalCompositeOperation='lighter';
  let g=ctx.createRadialGradient(0,0,2,0,0,26);
  g.addColorStop(0,'rgba(255,214,140,.5)');g.addColorStop(1,'rgba(255,180,90,0)');
  ctx.fillStyle=g;ctx.beginPath();ctx.arc(0,0,26,0,6.283);ctx.fill();
  ctx.globalCompositeOperation='source-over';

function drawParticles(){
  for(const q of particles){const a=1-q.t/q.life;if(a<=0)continue;
    if(q.c==='dust'){ctx.globalAlpha=a*.4;ctx.fillStyle='#9fb2cc';}
    else{ctx.save();ctx.globalCompositeOperation='lighter';ctx.globalAlpha=a;
      ctx.fillStyle=q.c==='spark'?'#ffd98a':q.c==='ember'?'#ff9a50':q.c==='mote'?'#ffe9a8':'#bfeaff';}
    ctx.beginPath();ctx.arc(q.x,q.y,q.size*a+.4,0,6.283);ctx.fill();
    if(q.c!=='dust')ctx.restore();}
  ctx.globalAlpha=1;
}

function render(t){
  drawSky(t);
  // fireflies (screen space)
  ctx.save();ctx.globalCompositeOperation='lighter';
  for(const f of flies){const x=mod(f.x+t*f.sp-camX*Z*.05*f.r,W+60)-30,y=H*f.y+Math.sin(t*.8+f.ph)*14;
    const a=.2+.2*Math.sin(t*2+f.ph);ctx.fillStyle='rgba(255,217,138,'+a.toFixed(3)+')';
    ctx.beginPath();ctx.arc(x,y,f.r,0,6.283);ctx.fill();}
  ctx.restore();
  // world
  let ox=0,oy=0;if(shakeT>0){const k=clamp(shakeT/shakeDur,0,1);ox=(Math.random()*2-1)*shakeMag*k;oy=(Math.random()*2-1)*shakeMag*k;}
  ctx.save();ctx.translate(-camX*Z+ox,-camY*Z+oy);ctx.scale(Z,Z);
  drawTiles(t);drawGates(t);drawPlats(t);drawMotes(t);drawExit(t);drawParticles();
  if(state==='play'||state==='pause')drawPlayer(p.x+PW/2,p.y+PH/2,1,0,1,t);
  else if(state==='menu'){const bob=Math.sin(t*2)*3;drawPlayer(LV.sx+PW/2,LV.sy+PH/2+bob*.4,1,0,1,t);}
  else if(state==='levelend'){const e=clamp(endT,0,1),s=1-e;
    drawPlayer(lerp(endFrom.x,endTo.x,e)+PW/2,lerp(endFrom.y,endTo.y,e)+PH/2,s,e*9,1-e*.6,t);}
  ctx.restore();

  ctx.fillStyle=vig;ctx.fillRect(0,0,W,H);
  if(flashT>0){ctx.fillStyle='rgba(220,245,255,'+(flashT*.7).toFixed(3)+')';ctx.fillRect(0,0,W,H);}
}

/* ---------------- main loop ---------------- */
let last=performance.now();
function frame(now){
  requestAnimationFrame(frame);
  const dt=Math.min(1/30,(now-last)/1000);last=now;const t=now/1000;
  if(shakeT>0)shakeT-=dt;if(flashT>0)flashT-=dt;
  // particles always advance (cheap, keeps menu alive too)
  for(let i=particles.length-1;i>=0;i--){const q=particles[i];q.t+=dt;
    if(q.t>=q.life){particles.splice(i,1);continue;}
    q.x+=q.vx*dt;q.y+=q.vy*dt;if(q.c==='ember')q.vy+=520*dt;if(q.c==='dust')q.vy+=140*dt;}

  if(state==='menu'){
    const full=Math.max(0,LV.w*T-W/Z);let cx=0;
    if(full>0){const c=(t*38)%(2*full);cx=c<full?c:2*full-c;}
    camX+=(cx-camX)*Math.min(1,dt*2);camY=(ROWS*T-H/Z)/2;
  }
  else if(state==='play'){
    runTime+=dt;updatePlats(dt);updatePlayer(dt);updateCrumbles(dt);
    const viewW=W/Z,viewH=H/Z;
    const tgt=p.x+PW/2+p.face*70-viewW/2;
    camX+=(tgt-camX)*Math.min(1,dt*7);
    const lo=(LV.w*T-viewW)/2,hi=LV.w*T-lo;camX=clamp(camX,lo,hi);
    if(viewH>=ROWS*T)camY=(ROWS*T-viewH)/2;
    else camY=clamp(p.y+PH/2-viewH/2,0,ROWS*T-viewH);
    // portal ambient sparks
    if(Math.random()<dt*6)particles.push({x:LV.exit.x+(Math.random()-.5)*30,y:LV.exit.y+(Math.random()-.5)*30,
      vx:(Math.random()-.5)*20,vy:-30-Math.random()*40,t:0,life:.9,size:1.8,c:'gate'});
    hudTick();
  }
  else if(state==='dead'){
    deadT+=dt;updateCrumbles(dt);
    if(deadT>.5){state='play';resetLevel();}
  }
  else if(state==='levelend'){
    endT+=dt/0.95;
    if(Math.random()<dt*20){const a=endT*14;particles.push({x:LV.exit.x+Math.cos(a)*34,y:LV.exit.y+Math.sin(a)*34,
      vx:-Math.cos(a)*60,vy:-Math.sin(a)*60,t:0,life:.5,size:2.2,c:'gate'});}
    if(endT>=1){if(pi+1<LEVELS.length){state='play';loadLevel(pi+1);}else victory();}
  }
  render(t);
}
