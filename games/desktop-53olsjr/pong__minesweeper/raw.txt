function normBall(){ const m=Math.hypot(ball.vx,ball.vy)||1; ball.vx=ball.vx/m*ball.speed; ball.vy=ball.vy/m*ball.speed; }

function hitPaddle(side,topBottom){
  const p=side==='you'?player:cpu;
  const off=clamp((ball.x-p.x)/(pw/2),-1,1);
  ball.speed=Math.min(ball.speed*RAMP,CS*BASE_SPEED*MAX_MULT);
  const a=off*(Math.PI*.34);
  ball.vx=Math.sin(a)*ball.speed;
  ball.vy=(side==='you'?-1:1)*Math.cos(a)*ball.speed;
  ball.y=topBottom-(side==='you'?R:-R);
  lastHitter=side; p.flash=1;
  spawnSparks(ball.x,topBottom,side==='you'?'#3fe0c5':'#ff6b7f',7);
  AU.paddle(off,side);
}
function stepBall(dt){
  ball.py=ball.y;
  ball.x+=ball.vx*dt; ball.y+=ball.vy*dt;
  if(ball.x<FX+R){ ball.x=FX+R; if(ball.vx<0){ ball.vx=-ball.vx; spawnSparks(FX,ball.y,'#ffd166',5); AU.wall(); } }
  if(ball.x>FX+FW-R){ ball.x=FX+FW-R; if(ball.vx>0){ ball.vx=-ball.vx; spawnSparks(FX+FW,ball.y,'#ffd166',5); AU.wall(); } }
  if(Math.abs(ball.vx)<ball.speed*.1){ const s=Math.sign(ball.vx)||(Math.random()<.5?1:-1); ball.vx=s*ball.speed*.12; }
  normBall();
  const pTop=(FY+FH-LANE*.5)-ph/2;
  if(ball.vy>0&&ball.py+R<=pTop+1&&ball.y+R>=pTop&&Math.abs(ball.x-player.x)<=pw/2+R) hitPaddle('you',pTop);
  const cBot=(FY+LANE*.5)+ph/2;
  if(ball.vy<0&&ball.py-R>=cBot-1&&ball.y-R<=cBot&&Math.abs(ball.x-cpu.x)<=pw/2+R) hitPaddle('cpu',cBot);
  const ci=cellAt(ball.x,ball.y);
  if(ci!==lastCell){ lastCell=ci;
    if(ci>=0){ if(mines.has(ci)) explode(ci); else { if(reveal(ci,true)) AU.tick(); } } }
  if(state==='play'){
    if(ball.y< FY-R*4) score('you');
    else if(ball.y> FY+FH+R*4) score('cpu');
  }
  ball.trail.push({x:ball.x,y:ball.y}); if(ball.trail.length>14) ball.trail.shift();
}

function updatePlayer(dt){
  const sp=CS*9;
  if(keys.l) player.tx-=sp*dt;
  if(keys.r) player.tx+=sp*dt;
  player.tx=clamp(player.tx,FX+pw/2,FX+FW-pw/2);
  player.x+=(player.tx-player.x)*(1-Math.exp(-dt*20));
}
function aiMove(p,skill,dt){
  const toward=(state==='play'&&ball.vy<0);
  let target;
  if(toward) target=ball.x+Math.sin(now*1.9+p.seed)*CS*(.5-.25*skill);
  else target=FX+FW/2+Math.sin(now*.8+p.seed)*CS*.35;
  const maxV=CS*(3.1+1.9*skill), d=target-p.x;
  p.x+=clamp(d,-maxV*dt,maxV*dt);
  p.x=clamp(p.x,FX+pw/2,FX+FW-pw/2);
}
function cpuSkill(){ return Math.min(.78,.6+.03*(youScore+cpuScore)); }

/* ================= UPDATE ================= */
function update(dt){
  now+=dt;
  for(const d of dust){ d.y-=d.s*dt; d.x+=Math.sin(now*.5+d.ph)*4*dt; if(d.y<-4){ d.y=H+4; d.x=Math.random()*W; } }
  if(state==='serve'){ serveT-=dt; if(serveT<=0) launch(); }
  else if(state==='play') stepBall(dt);
  else { rallyT+=dt; if(rallyT>=rallyDur&&state!=='over') endRally(); }
  if(mode==='attract') aiMove(player,.5,dt); else updatePlayer(dt);
  if(state!=='over') aiMove(cpu,mode==='attract'?.52:cpuSkill(),dt);
  player.flash=Math.max(0,player.flash-dt*4); cpu.flash=Math.max(0,cpu.flash-dt*4);
  shakeT=Math.max(0,shakeT-dt); flashA*=Math.exp(-dt*5.5);
  for(let i=parts.length-1;i>=0;i--){ const p=parts[i]; p.life-=dt;
    if(p.life<=0){ parts.splice(i,1); continue; }
    if(p.t==='deb'){ p.vy+=CS*6*dt; p.x+=p.vx*dt; p.y+=p.vy*dt; p.rot+=p.vr*dt; }
    else if(p.t==='spark'||p.t==='smoke'){ p.x+=p.vx*dt; p.y+=p.vy*dt; if(p.t==='smoke') p.r+=CS*.5*dt; }
    else if(p.t==='ring') p.r+=p.vr*dt;
  }
  for(let i=floats.length-1;i>=0;i--){ const f=floats[i];
    if(f.wait>0){ f.wait-=dt; continue; }
    f.life-=dt; f.y-=CS*.9*dt; if(f.life<=0) floats.splice(i,1); }
}

**Pong Minesweeper** — a full rally game where the court floor is booby-trapped:

- **Pong core**: you (teal, bottom) vs CPU (coral, top), mouse/touch or ←→/A D to steer. Ball speeds up with every hit; sharp paddle-edge angles bend the shot. First to 5 wins.
- **Minesweeper layer**: the ball *clears* cells as it flies — each revealed cell shows its classic colored neighbor-mine count, and zero-cells flood-fill just like the real thing. Ten hidden mines sit somewhere on the board; detonate one and the **other** side scores (so aim your shots around what you've mapped). Explosions reveal the surrounding numbers as a penalty, leaving charred craters behind.
- **Feel**: live CPU-vs-CPU attract mode runs behind the title screen, shockwave rings, debris, smoke, screen shake, warm flash, ball trail, paddle sparks, serve countdown ring, and full Web Audio SFX (blips that pitch with hit position, noise-burst booms, score jingles) — all generated after your first input.
- **Controls**: P pause · M sound · R restart · Enter/Space to start — plus persistent career wins via a guarded localStorage save.
