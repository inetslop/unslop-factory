function startWave(){
  phase='assault';const n=wave;
  const count=Math.min(46,5+Math.round(n*2.4));
  spawnQueue=[];
  for(let i=0;i<count;i++){
    const r=Math.random();let t='runner';
    if(n>=3&&r<0.16+n*0.012)t='spitter';
    else if(n>=2&&r<0.42)t='sieger';
    spawnQueue.push(t);
  }
  spawnT=0.5;
  banner('WAVE '+n,count+' HOSTILES INBOUND',2000);sfx.horn();
}
function spawnEnemy(type){
  const D=ETYPES[type],side=(Math.random()*4)|0;let x,y;
  if(side===0){x=rand(0,W);y=-24}else if(side===1){x=rand(0,W);y=H+24}
  else if(side===2){x=-24;y=rand(0,H)}else{x=W+24;y=rand(0,H)}
  const hp=Math.round(D.hp*(1+(wave-1)*0.28));
  enemies.push({type,x,y,hp,maxHp:hp,spd:D.spd*(1+Math.min(.35,(wave-1)*0.04)),r:D.r,
    reward:D.reward+Math.floor(wave*1.5),ph:rand(0,9),stun:0,cdRam:0,flash:0,fireT:rand(1,2.2),tele:0,

/* ================= combat ================= */
function damageEnemy(e,d){
  if(e.dead)return;e.hp-=d;e.flash=.12;
  if(e.hp<=0){e.dead=true;kills++;credits+=e.reward;
    explode(e.x,e.y,e.type==='sieger'?1.5:.9,e.type==='runner'?'#ff7a4d':e.type==='sieger'?'#aab4c4':'#cf8dff');
    addText(e.x,e.y-16,'+$'+e.reward,'#ffe9a8');sfx.coin();}
}
function damageCar(d){
  if(dead)return;
  car.armor-=d;car.flash=.4;shake=Math.max(shake,5);sfx.hit();
  addText(carP.x,carP.y-26,'-'+d,'#ff6a5c');
  if(car.armor<=0){car.armor=0;dead=true;overT=1.5;setBoost(false);
    explode(carP.x,carP.y,2.4,'#ffb04d');shake=18;sfx.over();}
}
function onLap(){
  laps++;const lt=T-lapStartT;lapStartT=T;
  if(bestLap===null||lt<bestLap)bestLap=lt;
  const bonus=120+wave*15;credits+=bonus;
  const wasLow=car.armor<100;car.armor=Math.min(100,car.armor+12);
  addText(carP.x,carP.y-34,'+$'+bonus,'#ffd23f');
  if(wasLow)addText(carP.x,carP.y-52,'REPAIRED +12','#7ee081');
  sfx.lap();
}

/* ================= placement ================= */
function validatePlace(x,y){
  if(x<24||x>W-24||y<24||y>H-24)return'OUT OF ZONE';
  if(dist(x,y,carP.x,carP.y)>320)return'TOO FAR FROM CAR';
  if(dist(x,y,carP.x,carP.y)<54)return'TOO CLOSE TO CAR';
  for(const t of turrets)if(dist(x,y,t.x,t.y)<48)return'SPACE BLOCKED';
  for(const e of enemies)if(!e.dead&&dist(x,y,e.x,e.y)<32)return'ENEMIES NEARBY';
  if(credits<TYPES[selected].cost)return'NOT ENOUGH $';
  return null;
}
function placeTurret(x,y){
  const r=validatePlace(x,y);
  if(r){addText(x,y-14,r,'#ff6a5c');return}
  const D=TYPES[selected];
  turrets.push({type:selected,x,y,ang:rand(0,TAU),cd:0,lvl:1,hp:D.hp,maxHp:D.hp,invested:D.cost,flash:0,hitF:0,dead:false});
  credits-=D.cost;sfx.place();
  parts.push({type:'ring',x,y,r:6,vr:180,life:.3,max:.3,color:D.color});
}
function upgradeTurret(t){
  if(t.lvl>=3){addText(t.x,t.y-24,'MAX LEVEL','#ffd23f');sfx.ui();return}
  const c=upCost(t);
  if(credits<c){addText(t.x,t.y-24,'NEED $'+c,'#ff6a5c');sfx.ui();return}
  credits-=c;t.lvl++;t.invested+=c;
  t.maxHp=TYPES[t.type].hp*(1+.7*(t.lvl-1));t.hp=t.maxHp;
  sfx.upgrade();addText(t.x,t.y-24,'LV '+t.lvl,'#7ee081');
  parts.push({type:'ring',x:t.x,y:t.y,r:6,vr:200,life:.3,max:.3,color:TYPES[t.type].color});
}

/* ================= update ================= */
function update(dt){
  if(dead){overT-=dt;updateFx(dt);shake=Math.max(0,shake-dt*14);if(overT<=0)finalizeOver();return}
  /* --- car --- */
  const dir=(held.has('arrowright')||held.has('d')?1:0)-(held.has('arrowleft')||held.has('a')?1:0);
  const brake=held.has('s')||held.has('arrowdown');
  car.latT=dir*44;
  car.lat+=(car.latT-car.lat)*Math.min(1,dt*6.5);
  car.tilt+=((clamp((car.latT-car.lat)*0.028,-0.45,0.45))-car.tilt)*Math.min(1,dt*9);
  const wantBoost=held.has(' ')&&car.nitro>2&&!brake;
  if(wantBoost!==car.boosting)setBoost(wantBoost);
  car.boosting=wantBoost;
  const spdT=brake?110:(wantBoost?435:235);
  car.speed+=(spdT-car.speed)*Math.min(1,dt*3.2);
  car.nitro=clamp(car.nitro+(wantBoost?-44:16)*dt,0,100);
  const pf=frac(car.s);car.s+=car.speed*dt;distTotal+=car.speed*dt;
  if(pf>0.9&&frac(car.s)<0.1)onLap();
  carP=trackPos(car.s,car.lat);carA=carP.a+car.tilt;
  // skids
  const skidding=Math.abs(car.latT-car.lat)>16||brake;
  if(skidding){
    const px=-Math.sin(carA),py=Math.cos(carA),dxv=Math.cos(carA),dyv=Math.sin(carA);
    const ws=[[carP.x-dxv*10+px*8,carP.y-dyv*10+py*8],[carP.x-dxv*10-px*8,carP.y-dyv*10-py*8]];
    if(!car.wPrev)car.wPrev=ws.map(w=>w.slice());
    for(let k=0;k<2;k++)skids.push({x1:car.wPrev[k][0],y1:car.wPrev[k][1],x2:ws[k][0],y2:ws[k][1],life:7});
    car.wPrev=ws.map(w=>w.slice());
    if(skids.length>180)skids.splice(0,skids.length-180);
  }else car.wPrev=null;
  if(wantBoost){shake=Math.max(shake,1.3);
    const bx=carP.x-Math.cos(carA)*20,by=carP.y-Math.sin(carA)*20;
    parts.push({type:'flame',x:bx+rand(-4,4),y:by+rand(-4,4),r:rand(3,6),life:.25,max:.25});
  }
  car.flash=Math.max(0,car.flash-dt*3);
  /* --- waves --- */
  if(phase==='prep'){prepT-=dt;if(prepT<=0){wave++;startWave()}}
  else{
    if(spawnQueue.length){spawnT-=dt;if(spawnT<=0){spawnEnemy(spawnQueue.shift());spawnT=Math.max(.4,1.35-wave*0.06)}}
    else if(enemies.length===0){
      const bonus=100+wave*25;credits+=bonus;sfx.clear();
      banner('WAVE '+wave+' CLEARED','+$'+bonus+' BONUS — REBUILD!',2200);
      phase='prep';prepT=Math.max(4.5,7-wave*.2);
    }
  }
  /* --- enemies --- */
  for(const e of enemies){
    if(e.dead)continue;
    e.flash=Math.max(0,e.flash-dt);e.cdRam=Math.max(0,e.cdRam-dt);
    let tx=carP.x,ty=carP.y,tr=null,chew=false;
    if(e.type==='sieger'){
      let bd=1e9;for(const t of turrets){if(t.dead)continue;const d=dist(e.x,e.y,t.x,t.y);if(d<bd){bd=d;tr=t}}
      if(tr){tx=tr.x;ty=tr.y;if(dist(e.x,e.y,tx,ty)<e.r+20)chew=true}
    }
    const dx=tx-e.x,dy=ty-e.y,d=Math.hypot(dx,dy)||1;
    let vx=dx/d,vy=dy/d;
    if(e.type==='runner'){const w=Math.sin(T*5+e.ph)*.35;vx+=-vy*w;vy+=vx*w;const m=Math.hypot(vx,vy);vx/=m;vy/=m}
    else if(e.type==='spitter'){
      e.fireT-=dt;
      if(d<520&&e.fireT<=0)e.tele=0.35;
      if(e.tele>0){e.tele-=dt;if(e.tele<=0){
        const a=Math.atan2(carP.y-e.y,carP.x-e.x);
        bolts.push({x:e.x+Math.cos(a)*14,y:e.y+Math.sin(a)*14,vx:Math.cos(a)*310,vy:Math.sin(a)*310,dmg:ETYPES.spitter.boltDmg,life:2.2,hostile:true,color:'#e28bff',r:5});
        tone(300,.1,'square',.1,90);e.fireT=1.9+rand(-.3,.4);
      }}
      if(d<310){const px=-vy*e.strafDir,py=vx*e.strafDir;vx=px*.85-vx*.2;vy=py*.85-vy*.2}
    }
    if(e.stun>0)e.stun-=dt;
    else if(!chew){e.x+=vx*e.spd*dt;e.y+=vy*e.spd*dt;e.dx=vx;e.dy=vy}
    // hit car
    if(dist(e.x,e.y,carP.x,carP.y)<e.r+15&&e.cdRam<=0){
      e.cdRam=1;damageCar(ETYPES[e.type].dmg);
      e.stun=.5;e.x-=vx*26;e.y-=vy*26;
      parts.push({type:'ring',x:e.x,y:e.y,r:4,vr:160,life:.22,max:.22,color:'#ff6a5c'});
    }
    // hit turrets
    for(const t of turrets){
      if(t.dead)continue;
      if(dist(e.x,e.y,t.x,t.y)<e.r+16){
        t.hp-=(e.type==='sieger'?ETYPES.sieger.dmgT:18)*dt;t.hitF=.15;
        if(Math.random()<dt*8)parts.push({type:'spark',x:t.x,y:t.y,vx:rand(-90,90),vy:rand(-90,90),life:.3,max:.3,color:'#ffd23f'});
        if(t.hp<=0){t.dead=true;explode(t.x,t.y,1.4,TYPES[t.type].color);sfx.boom(1.2);shake=Math.max(shake,7);addText(t.x,t.y-18,'TURRET LOST','#ff6a5c')}
      }
    }
  }
  // separation
  for(let i=0;i<enemies.length;i++){const a=enemies[i];if(a.dead)continue;
    for(let j=i+1;j<enemies.length;j++){const b=enemies[j];if(b.dead)continue;
      const dx=b.x-a.x,dy=b.y-a.y,d=Math.hypot(dx,dy),min=a.r+b.r;
      if(d>0&&d<min){const push=(min-d)/2,nx=dx/d,ny=dy/d;a.x-=nx*push;a.y-=ny*push;b.x+=nx*push;b.y+=ny*push}
    }}
  enemies=enemies.filter(e=>!e.dead);
  turrets=turrets.filter(t=>!t.dead);
  /* --- turrets fire --- */
  for(const t of turrets){
    t.hitF=Math.max(0,t.hitF-dt);
    const D=TYPES[t.type],st=turretStats(t);
    let tgt=null,bd=st.range;
    for(const e of enemies){if(e.dead)continue;const d=dist(t.x,t.y,e.x,e.y);if(d<bd){bd=d;tgt=e}}
    if(t.type!=='tesla'&&tgt)t.ang=angLerp(t.ang,Math.atan2(tgt.y-t.y,tgt.x-t.x),Math.min(1,dt*10));
    t.cd-=dt;
    if(tgt&&t.cd<=0){
      t.cd=1/st.rate;
      if(t.type==='blaster'){
        const a=t.ang;
        bolts.push({x:t.x+Math.cos(a)*17,y:t.y+Math.sin(a)*17,vx:Math.cos(a)*540,vy:Math.sin(a)*540,dmg:st.dmg,life:.9,hostile:false,color:'#8fe9ff',r:3});
        sfx.shoot1();
      }else if(t.type==='mortar'){
        shells.push({sx:t.x,sy:t.y,tx:tgt.x,ty:tgt.y,t:0,dur:.8,dmg:st.dmg,splash:D.splash});
        sfx.shoot2();
      }else{
        const hitList=[tgt];let last=tgt;
        for(let c=1;c<D.chain;c++){
          let best2=null,bd2=150;
          for(const e of enemies){if(e.dead||hitList.indexOf(e)>=0)continue;const d=dist(last.x,last.y,e.x,e.y);if(d<bd2){bd2=d;best2=e}}
          if(!best2)break;hitList.push(best2);last=best2;
        }
        const pts=[{x:t.x,y:t.y-8}];
        hitList.forEach((e,i)=>{pts.push({x:e.x,y:e.y});damageEnemy(e,st.dmg*Math.pow(.78,i));
          parts.push({type:'spark',x:e.x,y:e.y,vx:rand(-120,120),vy:rand(-120,120),life:.25,max:.25,color:'#d9b8ff'})});
        zaps.push({pts:jag(pts),life:.14,max:.14});sfx.zap();
      }
    }
  }
  /* --- bolts --- */
  for(let i=bolts.length-1;i>=0;i--){const b=bolts[i];
    b.x+=b.vx*dt;b.y+=b.vy*dt;b.life-=dt;
    let hit=false;
    if(b.hostile){if(dist(b.x,b.y,carP.x,carP.y)<15){damageCar(b.dmg);hit=true}}
    else{for(const e of enemies){if(e.dead)continue;if(dist(b.x,b.y,e.x,e.y)<e.r+3){damageEnemy(e,b.dmg);
      parts.push({type:'spark',x:b.x,y:b.y,vx:rand(-80,80),vy:rand(-80,80),life:.2,max:.2,color:b.color});hit=true;break}}}
    if(hit||b.life<=0||b.x<-40||b.x>W+40||b.y<-40||b.y>H+40)bolts.splice(i,1);
  }
  /* --- shells --- */
  for(let i=shells.length-1;i>=0;i--){const s=shells[i];s.t+=dt/s.dur;
    if(s.t>=1){
      for(const e of enemies){if(e.dead)continue;const d=dist(s.tx,s.ty,e.x,e.y);
        if(d<s.splash+e.r)damageEnemy(e,s.dmg*(d>s.splash*.6?.6:1))}
      explode(s.tx,s.ty,1.1,'#ffb04d');sfx.boom(1);
      if(dist(s.tx,s.ty,carP.x,carP.y)<220)shake=Math.max(shake,3.5);
      shells.splice(i,1);
    }
  }
  updateFx(dt);
  shake=Math.max(0,shake-dt*16);
}
function jag(pts){
  const out=[];
  for(let i=0;i<pts.length-1;i++){
    const a=pts[i],b=pts[i+1],dx=b.x-a.x,dy=b.y-a.y,len=Math.hypot(dx,dy)||1,nx=-dy/len,ny=dx/len,steps=5;
    for(let k=i===0?0:1;k<=steps;k++){const t=k/steps,j=(k===0||k===steps)?0:rand(-7,7);
      out.push({x:a.x+dx*t+nx*j,y:a.y+dy*t+ny*j})}
  }
  return out;
}
function updateFx(dt){
  for(let i=parts.length-1;i>=0;i--){const p=parts[i];p.life-=dt;if(p.life<=0){parts.splice(i,1);continue}
    if(p.type==='spark'||p.type==='flame'){p.x+=p.vx*dt;p.y+=p.vy*dt;p.vx*=(1-3.5*dt);p.vy*=(1-3.5*dt)}
    else if(p.type==='smoke'){p.y-=14*dt;p.r+=16*dt}
    else if(p.type==='ring')p.r+=p.vr*dt;
    else if(p.type==='flash')p.r+=130*dt;
  }
  for(let i=texts.length-1;i>=0;i--){const t=texts[i];t.life-=dt;t.y-=28*dt;if(t.life<=0)texts.splice(i,1)}
  for(let i=skids.length-1;i>=0;i--){skids[i].life-=dt;if(skids[i].life<=0)skids.splice(i,1)}
  for(let i=zaps.length-1;i>=0;i--){zaps[i].life-=dt;if(zaps[i].life<=0)zaps.splice(i,1)}
}

  }

  ctx.restore();
}
