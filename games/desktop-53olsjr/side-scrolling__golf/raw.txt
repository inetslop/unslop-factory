/* ============================== PHYSICS ============================== */
function segClosest(px,py,ax,ay,bx,by){const dx=bx-ax,dy=by-ay,L2=dx*dx+dy*dy||1;
  let t=((px-ax)*dx+(py-ay)*dy)/L2;t=t<0?0:t>1?1:t;return[ax+dx*t,ay+dy*t];}

function physics(dt){
  const sp0=Math.hypot(b.vx,b.vy);
  const n=Math.min(4,Math.max(1,Math.ceil(sp0/480))),sdt=dt/n;
  let touched=false;
  for(let s=0;s<n;s++){
    b.vy+=G*sdt;b.vx+=lvl.wind*WF*sdt*(touched?0.15:1);
    const dr=Math.max(0,1-0.045*sdt);b.vx*=dr;b.vy*=dr;
    b.x+=b.vx*sdt;b.y+=b.vy*sdt;
    // world walls
    if(b.x<R){b.x=R;if(b.vx<0)b.vx*=-.4;}
    if(b.x>lvl.W-R){b.x=lvl.W-R;if(b.vx>0)b.vx*=-.4;}
    // ground polyline
    const p=lvl.pts;
    for(let i=0;i<p.length-1;i++){
      const [ax,ay]=p[i],[bx,by]=p[i+1];
      const c=segClosest(b.x,b.y,ax,ay,bx,by);
      let px=b.x-c[0],py=b.y-c[1],d=Math.hypot(px,py);
      if(d<R){
        if(d<.001){px=0;py=-1;d=1;}
        const nx=px/d,ny=py/d;b.x=c[0]+nx*R;b.y=c[1]+ny*R;
        const vn=b.vx*nx+b.vy*ny;
        if(vn<0){
          touched=true;const sp=surfParams(c[0]);
          const tx=-ny,ty=nx;let vt=b.vx*tx+b.vy*ty;
          vt*=Math.max(0,1-sp.mu*sdt);
          const bounce=Math.abs(vn)>60?sp.e:0;
          b.vx=tx*vt+nx*(-vn*bounce);b.vy=ty*vt+ny*(-vn*bounce);
          if(Math.abs(vn)>170&&tGlobal-lastBs>.05){lastBs=tGlobal;
            if(sp.sand){sfx.sand();burst(c[0],c[1]-4,8,['#e8d29a','#d9bf85'],130,-60,.6,'p');}
            else{sfx.bounce(-vn);burst(c[0],c[1]-4,7,['#7ed957','#4fae5f','#a5e8b0'],-vn*.25,-80,.9,'p');}
            shake=Math.max(shake,Math.min(3.5,Math.abs(vn)/450));}
        } else touched=true;
      }
    }
    // circle obstacles
    for(const o of lvl.cols){
      let px=b.x-o.x,py=b.y-o.y,d=Math.hypot(px,py);
      if(d<R+o.r){
        if(d<.001){px=0;py=-1;d=1;}
        const nx=px/d,ny=py/d;b.x=o.x+nx*(R+o.r);b.y=o.y+ny*(R+o.r);
        const vn=b.vx*nx+b.vy*ny;
        if(vn<0){touched=true;const cp=COLP[o.type]||COLP.tree;
          const tx=-ny,ty=nx;let vt=b.vx*tx+b.vy*ty;vt*=Math.max(0,1-cp.mu*sdt);
          const bounce=Math.abs(vn)>60?cp.e:0;
          b.vx=tx*vt+nx*(-vn*bounce);b.vy=ty*vt+ny*(-vn*bounce);
          if(Math.abs(vn)>140&&tGlobal-lastBs>.05){lastBs=tGlobal;
            if(o.type==='rock')sfx.thud();else sfx.leaf();
            burst(b.x,b.y,6,o.type==='tree'?['#3aa257','#2f8f46']:o.type==='bush'?['#46a860','#35914c']:['#98a0a8','#7b838c'],-vn*.3,-60,.8,'p');}
        } else touched=true;
      }
    }
  }
  b.rot+=b.vx*dt/(R*(touched?1:1.4));
  const sp=Math.hypot(b.vx,b.vy);
  if(touched){if(sp<8){b.vx=0;b.vy=0;if(!b.resting){b.resting=true;lastRest={x:b.x,y:b.y};if(state==='menu')demoWait=1.4;}}else b.resting=false;}
  else b.resting=false;
}

function update(dt,t){
  tGlobal=t;
  hitLock=Math.max(0,hitLock-dt);shake*=Math.exp(-6*dt);if(shake<.05)shake=0;
  // kb aim
  if(state==='play'){
    if(keys['ArrowLeft'])kb.angle-=2.6*dt;
    if(keys['ArrowRight'])kb.angle+=2.6*dt;
    kb.angle=clamp(kb.angle,-3,-0.15);
    if(kb.charging&&b.resting&&hitLock<=0)kb.power=Math.min(1,kb.power+dt/.85);
  }
  // clouds & birds & streaks
  for(const c of clouds)c.x+=c.v*dt;
  birdTimer-=dt;
  if(birdTimer<0&&birds.length<3){birdTimer=9+Math.random()*10;
    const dir=Math.random()<.5?1:-1;
    birds.push({x:dir>0?-40:vw+40,y:vh*(.08+Math.random()*.25),vx:dir*(35+Math.random()*30),ph:Math.random()*6});}
  for(let i=birds.length-1;i>=0;i--){const bd=birds[i];bd.x+=bd.vx*dt;if(bd.x<-80||bd.x>vw+80)birds.splice(i,1);}
  const w=lvl.wind;
  for(const s of streaks){if(Math.abs(w)<=.04)continue;s.x+=w*380*dt*SC;if(s.x<-120)s.x+=vw+240;if(s.x>vw+120)s.x-=vw+240;}
  // sprinklers
  for(const s of lvl.sprk){
    const a=-Math.PI/2+Math.sin(t*s.speed+s.ph)*1.15,bx=s.x,by=gYat(bx)-8;
    const tx=bx+Math.cos(a)*s.len,ty=by+Math.sin(a)*s.len;
    if(s._tx!==undefined){s._vx=(tx-s._tx)/dt;s._vy=(ty-s._ty)/dt;}
    s._tx=tx;s._ty=ty;s._a=a;s._bx=bx;s._by=by;
    if(particles.length<380&&Math.random()<.55)
      particles.push({type:'drop',x:tx,y:ty,vx:(s._vx||0)*.4+(Math.random()-.5)*160,vy:(s._vy||0)*.4+(Math.random()-.7)*140,life:0,max:.5+Math.random()*.3,size:2+Math.random()*2,col:Math.random()<.5?'#bfe6ff':'#e8f7ff',g:1});
    if((state==='play'||state==='menu')&&!b.resting&&s._tx!==undefined){
      const c=segClosest(b.x,b.y,bx,by,tx,ty),d=Math.hypot(b.x-c[0],b.y-c[1]);
      if(d<R+6){const nx=(b.x-c[0])/(d||1),ny=(b.y-c[0]*0+b.y-c[1])/(d||1);

    }

    if(p.life>p.max){particles.splice(i,1);continue;}
    if(p.type==='ring')continue;
    p.vy+=(p.g||1)*G*.5*dt;p.x+=p.vx*dt;p.y+=p.vy*dt;
    if(p.type==='conf'){p.vx*=Math.exp(-1.2*dt);p.rot+=p.vr*dt;}
  }

  if(state==='play'||state==='menu'){
    physics(dt);
    for(const z of lvl.water)if(b.x>z.x1&&b.x<z.x2&&b.y>z.y-4){doSplash(z);break;}
    const cx=lvl.cup,cy=gYat(cx)-2,dx=b.x-cx,dy=b.y-cy,d=Math.hypot(dx,dy),sp=Math.hypot(b.vx,b.vy);
    if(d<26&&sp<190){b.vx-=dx/d*520*dt;b.vy-=dy/d*520*dt;}
    if((d<12.5&&sp<480)||d<8)startSink();
    if(sp>240){trail.push({x:b.x,y:b.y});if(trail.length>15)trail.shift();}
    else if(trail.length)trail.shift();
  }else if(state==='sinking'){
    const cx=lvl.cup,cy=gYat(cx)-2;
    sinkT+=dt;const k=Math.min(1,sinkT/.38),e=k*k;
    b.x=lerp(sinkFrom.x,cx,e);b.y=lerp(sinkFrom.y,cy-2,e);b.scale=1-.72*e;
    flagDrop=Math.min(1,flagDrop+dt*1.6);
    if(state==='sinking'){
      if(demoResetT>0&&sinkT>.9){demoResetT=0;b.x=lvl.tee;b.y=gYat(lvl.tee)-R;b.vx=b.vy=0;b.resting=true;b.scale=1;trail.length=0;state='menu';demoWait=1.6;}
      else if(sinkT>.85){if(idx>=LEVELS.length-1)gameOver();else showHoleDone();}
    }
  }
  // demo in menu
  if(state==='menu'&&b.resting&&demoResetT<=0){
    demoWait-=dt;
    if(demoWait<=0){const dx=lvl.cup-b.x,dy=gYat(lvl.cup)-b.y-40;
      let ang=Math.atan2(dy,dx);if(ang>-0.18)ang=dx>=0?-0.18:-Math.PI+0.18;
      const pw=clamp(Math.hypot(dx,dy)/1500,.3,.9);shoot(ang,pw,true);demoWait=99;}
  }
  // camera

  let tx=b.x+b.vx*.2-viewW/2,ty=b.y-viewH*.55;

  else{const k=Math.min(1,dt*5);cam.x+=(tx-cam.x)*k;cam.y+=(ty-cam.y)*k;}
}
