function explode(x,y,color,scale){
  const n=Math.round(16*scale);
  for(let i=0;i<n;i++){const a=Math.random()*6.28,sp=(40+Math.random()*130)*scale;
    parts.push({x,y,vx:Math.cos(a)*sp,vy:Math.sin(a)*sp-30,g:150,life:.5+Math.random()*.4,max:.9,color,size:2+Math.random()*3*scale});}
  for(let i=0;i<6;i++)parts.push({x,y,vx:(Math.random()-.5)*90,vy:(Math.random()-.5)*90,g:0,life:.35,max:.35,color:'#ffffff',size:2});
  anims.push({k:'ring',x,y,r0:4,r1:cell*(0.9*scale+0.6),t0:NOW,dur:340,col:color});
}
function uPos(u){const bob=Math.sin(NOW/380+(u.col*1.3+u.row))*cell*0.025;return {x:gx+(u.col+0.5)*cell,y:(u.cy==null?gy+(u.row+0.5)*cell:u.cy)+bob};}
function cellAt(x,y){const c=Math.floor((x-gx)/cell),r=Math.floor((y-gy)/cell);return (c>=0&&c<COLS&&r>=0&&r<ROWS)?{r,c}:null;}
function unitsAt(r){const a=[];for(let c=0;c<COLS;c++)if(grid[r][c])a.push(grid[r][c]);return a;}
function findUnit(t){for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++){const u=grid[r][c];if(u&&u.type===t)return u;}return null;}

function buildGrid(pattern){
  grid=[];units=[];
  for(let r=0;r<ROWS;r++){grid[r]=[];for(let c=0;c<COLS;c++){const ch=pattern[r][c];let u=null;
    if(ch!=='.'&&TYPES[ch]){u={type:ch,hp:TYPES[ch].hp,maxHp:TYPES[ch].hp,row:r,col:c,flash:0,born:NOW,cy:gy+(r+0.5)*cell-46};}
    grid[r][c]=u;if(u)units.push(u);}}
}
function drawCard(){
  if(state.deck.length===0){
    if(state.discard.length){state.deck=shuffle(state.discard.splice(0));toast('DECK RESHUFFLED');}
    else return;
  }
  const c=state.deck.pop();
  if(state.hand.length>=HAND_MAX){state.discard.push(c);toast('HAND FULL — CARD BURNED');return;}
  state.hand.push(c);SFX.draw();refreshHand();
}
function startWave(n){
  state.wave=n;state.energyCap=CAPS[n-1];state.energy=state.energyCap;state.freeze=false;
  buildGrid(WAVES[n-1]);
  while(state.hand.length<5)drawCard();
  bannerSet('WAVE '+n,WAVE_COLORS[n-1]);SFX.wave();
  updateWave();refreshEnergy();updateBasePips();refreshHand();
  state.phase='player';setEndBusy(false);
}
function startGame(){

/* ---------- damage ---------- */
function damageUnit(u,dmg){
  if(!u||u.dead)return false;
  u.hp-=dmg;u.flash=1;
  const p=uPos(u);addFloat('-'+dmg,p.x+(Math.random()-.5)*10,p.y-cell*0.42,'#ffffff',3);
  if(u.hp<=0){killUnit(u);return true;}
  return false;
}
function killUnit(u){
  u.dead=true;grid[u.row][u.col]=null;
  const i=units.indexOf(u);if(i>=0)units.splice(i,1);
  const t=TYPES[u.type];state.score+=t.pts;updateScore();state.kills++;
  const p=uPos(u);
  addFloat('+'+t.pts,p.x,p.y-cell*0.5,t.color,3.5);
  explode(p.x,p.y,t.color,u.type==='B'?2.4:1);
  SFX.explode(!!t.boss);if(t.boss)shake(9);
}
function damageBase(n){
  state.baseHP=Math.max(0,state.baseHP-n);baseFlash=1;flashRed=Math.min(1,flashRed+0.65);
  shake(Math.min(12,4+n));addFloat('-'+n,turretX+(Math.random()-.5)*20,baseY-38,'#ff4d5e',4.5);
  SFX.breach();updateBasePips();
  if(state.baseHP<=0)endGame(false);
}
function baseHeal(n){state.baseHP=Math.min(BASE_MAX,state.baseHP+n);shieldT=1;updateBasePips();
  for(let i=0;i<12;i++)parts.push({x:turretX+(Math.random()-.5)*90,y:baseY-8,vx:(Math.random()-.5)*30,vy:-40-Math.random()*50,g:20,life:.7,max:.7,color:'#5dffa8',size:2.5});
}

/* ---------- card selection & play ---------- */
function cancelTargeting(){selected=-1;targeting=null;refreshHand();}
function selectCard(i){
  if(state.phase!=='player')return;
  if(i<0||i>=state.hand.length)return;
  if(selected===i){cancelTargeting();return;}
  const id=state.hand[i],d=CARDS[id];
  if(d.c>state.energy){SFX.deny();denyCard(i);return;}
  if(!d.tgt){ // instant play
    if(id==='repair'&&state.baseHP>=BASE_MAX){toast('BASE AT FULL INTEGRITY');SFX.deny();return;}
    if(id==='emp'&&state.freeze){toast('FORMATION ALREADY FROZEN');SFX.deny();return;}
    SFX.select();playCard(i,null);return;
  }
  selected=i;targeting=d.tgt;refreshHand();SFX.select();
}
function resolveTarget(mode,cellC){
  if(mode==='unit')return grid[cellC.r][cellC.c];
  if(mode==='row')return unitsAt(cellC.r).length?{r:cellC.r}:null;
  if(mode==='col'){let any=false;for(let r=0;r<ROWS;r++)if(grid[r][cellC.c])any=true;return any?{c:cellC.c}:null;}
  return cellC;
}
function playCard(idx,target){
  const id=state.hand[idx],d=CARDS[id];
  state.energy-=d.c;state.discard.push(id);state.hand.splice(idx,1);
  selected=-1;targeting=null;refreshHand();refreshEnergy();
  let kills=0;const kill=f=>{kills+=f?1:0};
  const tx=turretX+turretAim;
  switch(id){
    case 'pulse':{const u=target;SFX.laser();anims.push({k:'laser',x1:tx,y1:baseY-18,x2:uPos(u).x,y2:uPos(u).y,t0:NOW,dur:150,col:d.col});kill(damageUnit(u,2));break;}
    case 'sniper':{const u=target;SFX.sniper();anims.push({k:'laser',x1:tx,y1:baseY-18,x2:uPos(u).x,y2:uPos(u).y,t0:NOW,dur:120,col:d.col,thick:true});kill(damageUnit(u,4));break;}
    case 'flak':{SFX.flak();const u=target;anims.push({k:'laser',x1:tx,y1:baseY-18,x2:uPos(u).x,y2:uPos(u).y,t0:NOW,dur:150,col:d.col});
      [[0,0],[1,0],[-1,0],[0,1],[0,-1]].forEach(([dr,dc])=>{const r=u.row+dr,c=u.col+dc;
        if(r>=0&&r<ROWS&&c>=0&&c<COLS){const t=grid[r][c];if(t)kill(damageUnit(t,1));}});break;}
    case 'beam':{SFX.beam();const y=gy+(target.r+0.5)*cell;
      anims.push({k:'hbeam',y,x0:gx-8,x1:gx+gridW+8,t0:NOW,dur:340,col:d.col});
      unitsAt(target.r).forEach(u=>kill(damageUnit(u,1)));break;}
    case 'rail':{SFX.rail();const x=gx+(target.c+0.5)*cell;
      anims.push({k:'vbeam',x,y0:gy-16,y1:gy+gridH+16,t0:NOW,dur:420,col:d.col});
      for(let r=0;r<ROWS;r++)if(grid[r][target.c])kill(damageUnit(grid[r][target.c],5));shake(3);break;}
    case 'nova':{SFX.bomb();const x=gx+(target.c+0.5)*cell,y=gy+(target.r+0.5)*cell;
      explode(x,y,'#ff5fa8',1.2);shake(4);
      for(let r=Math.max(0,target.r-1);r<=Math.min(ROWS-1,target.r+1);r++)
        for(let c=Math.max(0,target.c-1);c<=Math.min(COLS-1,target.c+1);c++)
          if(grid[r][c])kill(damageUnit(grid[r][c],2));break;}
    case 'emp':{SFX.zap();state.freeze=true;
      const cx=gx+gridW/2,cy=gy+gridH/2,R=Math.max(gridW,gridH)*0.75;
      anims.push({k:'ring',x:cx,y:cy,r0:10,r1:R,t0:NOW,dur:480,col:d.col});
      anims.push({k:'ring',x:cx,y:cy,r0:10,r1:R*0.7,t0:NOW+120,dur:480,col:'#ffffff'});
      units.forEach(u=>{const p=uPos(u);for(let i=0;i<3;i++)parts.push({x:p.x+(Math.random()-.5)*cell,y:p.y+(Math.random()-.5)*cell,vx:0,vy:-15,g:0,life:.8,max:.8,color:'#cfeaff',size:2});});
      break;}
    case 'repair':{SFX.repair();baseHeal(3);anims.push({k:'ring',x:turretX,y:baseY-14,r0:6,r1:70,t0:NOW,dur:420,col:d.col});addFloat('+3',turretX,baseY-52,'#5dffa8',4);break;}
    case 'salvage':{SFX.draw();drawCard();setTimeout(()=>{if(runToken===runToken)drawCard();},120);break;}
    case 'over':{SFX.energy();state.energy+=2;refreshEnergy();addFloat('+2',turretX,baseY-56,'#ffe14d',4.5);
      for(let i=0;i<10;i++)parts.push({x:turretX+(Math.random()-.5)*60,y:baseY-10,vx:(Math.random()-.5)*70,vy:-60-Math.random()*60,g:80,life:.5,max:.5,color:'#ffe14d',size:2.5});break;}
  }
  if(kills>=2){addFloat((kills===2?'DOUBLE KILL!':kills===3?'TRIPLE KILL!':'QUAD KILL!'),W/2,H*0.3,'#ffd83d',5);SFX.waveclear();}
  checkClear();
}
function checkClear(){
  if(state.phase!=='player')return;
  let any=false;for(let r=0;r<ROWS&&!any;r++)for(let c=0;c<COLS;c++)if(grid[r][c]){any=true;break;}
  if(any)return;
  state.phase='clear';setEndBusy(true);$('ebMain').textContent='CLEARING…';
  const bonus=50*state.wave;state.score+=bonus;updateScore();
  addFloat('+'+bonus,W/2,H*0.42,'#3dff8f',5);
  bannerSet('WAVE CLEAR','#3dff8f');SFX.waveclear();baseHeal(3);
  const t=runToken;
  setTimeout(()=>{if(t!==runToken)return;
    if(state.wave>=5)endGame(true);else startWave(state.wave+1);},1700);
}

/* ---------- enemy phase ---------- */
function endTurn(){
  if(state.phase!=='player')return;
  cancelTargeting();state.phase='enemy';setEndBusy(true);
  const t=runToken,guard=()=>t===runToken&&state.phase==='enemy';
  const step=(ms,fn)=>setTimeout(()=>{if(guard())fn();},ms);
  const bottom=unitsAt(ROWS-1);
  if(bottom.length)SFX.breach();
  step(260,()=>{let total=0;
    bottom.forEach((u,i)=>{const p=uPos(u);total+=u.atk||TYPES[u.type].atk;
      anims.push({k:'laser',x1:p.x,y1:p.y,x2:turretX+(Math.random()-.5)*40,y2:baseY-16,t0:NOW+i*80,dur:170,col:'#ff4d5e'});});
    damageBase(total);});
  step(920,()=>{
    if(state.freeze){state.freeze=false;bannerSet('FROZEN','#9fd7ff');SFX.zap();}
    else{for(let r=ROWS-2;r>=0;r--)for(let c=0;c<COLS;c++){const u=grid[r][c];
        if(u&&!grid[r+1][c]){grid[r+1][c]=u;grid[r][c]=null;u.row=r+1;}}
      SFX.descend();shake(1.4);}});
  step(1500,()=>{const b=findUnit('B');if(b){const p=uPos(b);
    anims.push({k:'bossbeam',x:p.x,y0:p.y+cell*0.28,y1:baseY-16,t0:NOW,dur:430});SFX.rail();damageBase(TYPES.B.atk);}});
  step(2150,()=>startPlayerTurn());
}
function startPlayerTurn(){
  state.phase='player';state.energy=state.energyCap;refreshEnergy();drawCard();setEndBusy(false);
}
function endGame(win){
  if(state.phase==='over')return;
  state.phase='over';setEndBusy(true);$('ebMain').textContent=win?'VICTORY':'OVERRUN';
  let newBest=false;
  if(win){state.score+=state.baseHP*25;updateScore();bannerSet('VICTORY','#3dff8f');SFX.win();}
  else{flashRed=1;shake(13);bannerSet('GAME OVER','#ff4d5e');SFX.lose();}
  if(state.score>best){best=state.score;saveBest(best);newBest=true;}
  $('bestVal').textContent=best;
  const t=runToken;
  setTimeout(()=>{if(t!==runToken)return;
    renderPixelWord('ovWordCv',win?'VICTORY':'GAME OVER',win?'#3dff8f':'#ff4d5e',420,9);
    $('ovFlavor').textContent=win
      ?"The armada is dust. Earth's skies are clear — for now."
      :"Overrun on wave "+state.wave+". The breach line held for "+state.kills+" kills.";
    $('stScore').textContent=state.score;
    const sb=$('stBest');sb.textContent=best;sb.classList.toggle('new',newBest);
    $('stWave').textContent=(win?5:state.wave-1)+'/5';
    $('stKills').textContent=state.kills;
    $('over').classList.remove('hidden');
  },win?1700:1300);
}

const easeOutBack=x=>{const c=1.70158;return 1+(c+1)*Math.pow(x-1,3)+c*Math.pow(x-1,2);};
function update(dt){
  for(const s of stars){s.y+=s.s*dt/1000;if(s.y>H){s.y=-2;s.x=Math.random()*W;}}
  const ease=1-Math.exp(-dt*0.006);
  for(const u of units){const ty=gy+(u.row+0.5)*cell;u.cy+=(ty-u.cy)*ease;if(Math.abs(ty-u.cy)<0.4)u.cy=ty;if(u.flash>0)u.flash=Math.max(0,u.flash-dt*0.004);}
  turretAim+=((mouseX-W/2)*0.12-turretAim)*(1-Math.exp(-dt*0.004));
  if(shieldT>0)shieldT=Math.max(0,shieldT-dt*0.0012);
  if(baseFlash>0)baseFlash=Math.max(0,baseFlash-dt*0.003);
  if(flashRed>0)flashRed=Math.max(0,flashRed-dt*0.0018);
  shakeMag*=Math.exp(-dt*0.004);if(shakeMag<0.05)shakeMag=0;
  anims=anims.filter(a=>NOW-a.t0<a.dur+250);
  for(const p of parts){p.x+=p.vx*dt/1000;p.y+=p.vy*dt/1000;p.vy+=p.g*dt/1000;p.life-=dt/1000;}
  parts=parts.filter(p=>p.life>0);
  floats=floats.filter(f=>NOW-f.t0<f.dur);
  if(banner&&NOW-banner.t0>banner.dur)banner=null;
  if(state.freeze&&Math.random()<dt*0.004){const u=units[Math.floor(Math.random()*units.length)];if(u){const p=uPos(u);parts.push({x:p.x+(Math.random()-.5)*cell,y:p.y-cell*0.3,vx:0,vy:-12,g:0,life:.9,max:.9,color:'#cfeaff',size:2});}}
}
function drawUnit(u){
  const frame=Math.floor(NOW/480)%2;
  const p=uPos(u);const type=TYPES[u.type];
  let sc=1;if(u.born)sc=Math.max(0.05,easeOutBack(Math.min(1,(NOW-u.born)/300)));
  const img=getSprite(u.type,frame);
  ctx.save();ctx.translate(p.x,p.y);ctx.scale(sc,sc);
  ctx.shadowColor=type.color;ctx.shadowBlur=u.type==='B'?16:6;
  ctx.drawImage(img,-img.width/2,-img.height/2);
  if(u.flash>0){ctx.globalCompositeOperation='lighter';ctx.globalAlpha=u.flash;
    const wi=getSprite(u.type,frame,null,true);ctx.drawImage(wi,-wi.width/2,-wi.height/2);}
  ctx.restore();
  if(u.hp<u.maxHp){const bw=u.type==='B'?cell*1.5:cell*0.62,ratio=u.hp/u.maxHp;
    ctx.fillStyle='#0a1226';ctx.fillRect(p.x-bw/2,p.y+img.height*sc/2+5,bw,4);
    ctx.fillStyle=ratio>0.5?'#3dff8f':ratio>0.25?'#ffd83d':'#ff4d5e';
    ctx.fillRect(p.x-bw/2,p.y+img.height*sc/2+5,bw*ratio,4);}
  if(state.freeze){const a=0.35+0.25*Math.sin(NOW/180);
    ctx.strokeStyle='rgba(159,215,255,'+a+')';ctx.lineWidth=1.5;
    ctx.strokeRect(p.x-cell*0.4,p.y-cell*0.42,cell*0.8,cell*0.84);}
  if(u.type==='B')drawPixelText(ctx,'DREADNOUGHT',p.x,p.y-img.height/2-14,Math.max(2,cell*0.05),'#ffc94a',0.9);
}

  if(shieldT>0){ctx.save();ctx.strokeStyle='rgba(93,255,168,'+shieldT*0.8+')';ctx.lineWidth=2;

}
function drawTargeting(){
  if(selected<0)return;
  const d=CARDS[state.hand[selected]],col=d.col,pulse=0.45+0.3*Math.sin(NOW/220);
  ctx.lineWidth=2;
  if(targeting==='unit'){
    for(const u of units){const p=uPos(u);
      ctx.strokeStyle=col.replace(')',',1)').replace('rgb','rgba');ctx.globalAlpha=pulse;
      ctx.strokeRect(p.x-cell*0.4,p.y-cell*0.42,cell*0.8,cell*0.84);ctx.globalAlpha=1;}
    if(hoverCell&&grid[hoverCell.r][hoverCell.c]){const u=grid[hoverCell.r][hoverCell.c],p=uPos(u);
      ctx.fillStyle=col;ctx.globalAlpha=0.16;ctx.fillRect(p.x-cell*0.4,p.y-cell*0.42,cell*0.8,cell*0.84);ctx.globalAlpha=1;
      ctx.strokeStyle='#fff';ctx.beginPath();ctx.moveTo(p.x-cell*0.28,p.y);ctx.lineTo(p.x+cell*0.28,p.y);ctx.moveTo(p.x,p.y-cell*0.28);ctx.lineTo(p.x,p.y+cell*0.28);ctx.stroke();}
  }else if(targeting==='row'||targeting==='col'){
    ctx.globalAlpha=0.35;ctx.strokeStyle=col;
    for(let r=0;r<ROWS;r++)if(unitsAt(r).length)ctx.strokeRect(gx+2,gy+r*cell+2,gridW-4,cell-4);
    for(let c=0;c<COLS;c++){let any=false;for(let r=0;r<ROWS;r++)if(grid[r][c])any=true;if(any)ctx.strokeRect(gx+c*cell+2,gy+2,cell-4,gridH-4);}
    ctx.globalAlpha=1;
    if(hoverCell){const valid=targeting==='row'?unitsAt(hoverCell.r).length:((()=>{for(let r=0;r<ROWS;r++)if(grid[r][hoverCell.c])return true;return false})());
      if(valid){ctx.fillStyle=col;ctx.globalAlpha=0.13+0.05*Math.sin(NOW/200);
        if(targeting==='row')ctx.fillRect(gx,gy+hoverCell.r*cell,gridW,cell);
        else ctx.fillRect(gx+hoverCell.c*cell,gy,cell,gridH);
        ctx.globalAlpha=1;}}
  }else if(targeting==='cell'){
    ctx.globalAlpha=0.14;ctx.strokeStyle=col;
    for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++)ctx.strokeRect(gx+c*cell+2,gy+r*cell+2,cell-4,cell-4);
    ctx.globalAlpha=1;
    if(hoverCell){const r0=Math.max(0,hoverCell.r-1),r1=Math.min(ROWS-1,hoverCell.r+1),c0=Math.max(0,hoverCell.c-1),c1=Math.min(COLS-1,hoverCell.c+1);

      ctx.strokeStyle=col;ctx.strokeRect(gx+c0*cell+2,gy+r0*cell+2,(c1-c0+1)*cell-4,(r1-r0+1)*cell-4);}
  }
}
function drawAnims(){
  for(const a of anims){
    const p=Math.min(1,(NOW-a.t0)/a.dur);if(p<0)continue;
    if(a.k==='laser'){const e=1-Math.pow(1-p,2),hx=a.x1+(a.x2-a.x1)*e,hy=a.y1+(a.y2-a.y1)*e;
      ctx.save();ctx.globalAlpha=p>0.9?1-p*4+1:1;ctx.strokeStyle=a.col;ctx.lineWidth=a.thick?3.5:2.2;
      ctx.shadowColor=a.col;ctx.shadowBlur=10;
      ctx.beginPath();ctx.moveTo(a.x1,a.y1);ctx.lineTo(hx,hy);ctx.stroke();
      ctx.fillStyle='#fff';ctx.beginPath();ctx.arc(hx,hy,a.thick?5:3.4,0,6.28);ctx.fill();ctx.restore();}
    else if(a.k==='hbeam'||a.k==='vbeam'){const al=1-p,w=(a.k==='hbeam'?7:9)*(1-p*0.6)+2;
      ctx.save();ctx.globalAlpha=al;ctx.strokeStyle=a.col;ctx.lineWidth=w;ctx.shadowColor=a.col;ctx.shadowBlur=18;
      ctx.beginPath();
      if(a.k==='hbeam'){ctx.moveTo(a.x0,a.y);ctx.lineTo(a.x1,a.y);}else{ctx.moveTo(a.x,a.y0);ctx.lineTo(a.x,a.y1);}
      ctx.stroke();ctx.strokeStyle='#fff';ctx.lineWidth=w*0.35;ctx.stroke();ctx.restore();}
    else if(a.k==='bossbeam'){
      ctx.save();
      if(p<0.4){const r=p/0.4*26;ctx.fillStyle='#ffc94a';ctx.shadowColor='#ffc94a';ctx.shadowBlur=20;
        ctx.beginPath();ctx.arc(a.x,a.y0,r,0,6.28);ctx.fill();}
      else{const al=1-(p-0.4)/0.6;ctx.globalAlpha=Math.min(1,al*2);ctx.strokeStyle='#ffc94a';ctx.lineWidth=7;ctx.shadowColor='#ffc94a';ctx.shadowBlur=22;
        ctx.beginPath();ctx.moveTo(a.x,a.y0);ctx.lineTo(a.x,a.y1);ctx.stroke();
        ctx.strokeStyle='#fff8e0';ctx.lineWidth=2.5;ctx.stroke();}
      ctx.restore();}
    else if(a.k==='ring'){const e=1-Math.pow(1-p,3),r=a.r0+(a.r1-a.r0)*e;
      ctx.save();ctx.globalAlpha=1-p;ctx.strokeStyle=a.col;ctx.lineWidth=3*(1-p)+1;ctx.shadowColor=a.col;ctx.shadowBlur=12;
      ctx.beginPath();ctx.arc(a.x,a.y,r,0,6.28);ctx.stroke();ctx.restore();}
  }
}
function render(){

  drawBase();

  drawTargeting();

  if(flashRed>0){ctx.fillStyle='rgba(255,40,60,'+(flashRed*0.28)+')';ctx.fillRect(0,0,W,H);}
}
