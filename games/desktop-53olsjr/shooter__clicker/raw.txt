function spawnEnemy(type,x,y){
  const d=DEFS[type], sp=1+(level-1)*.08;
  const e={type,x:(x===undefined?rand(45,W-45):x),y:(y===undefined?-30:y),r:d.r,hp:d.hp,maxHp:d.hp,pts:d.pts,color:d.color,t:Math.random()*6,seed:Math.random()*10,vx:0,vy:d.vy*sp,flash:0,dead:false};
  if(type==='speeder'){ e.vx=(e.x<W/2?1:-1)*rand(50,90); }
  enemies.push(e);
}
function pickType(){
  const L=level; let pool=[['drone',10]];
  if(L>=2)pool.push(['speeder',4+L]);
  if(L>=2)pool.push(['splitter',3+L*.8]);
  if(L>=2)pool.push(['mine',3+L*.6]);
  if(L>=3)pool.push(['brute',2+L*1.2]);
  let tot=0; for(const p of pool)tot+=p[1];
  let r=Math.random()*tot;
  for(const p of pool){ r-=p[1]; if(r<=0)return p[0]; }
  return 'drone';
}
function menuPick(){ const r=Math.random(); if(r<.55)return 'drone'; if(r<.72)return 'speeder'; if(r<.85)return 'splitter'; if(r<.94)return 'mine'; return 'brute'; }

/* ============================== COMBAT ============================== */
function fire(){
  if(fireCd>0)return;
  fireCd=time<rapidUntil?.075:.16;
  shots++; recoil=1; muzzleT=.06; shake=Math.max(shake,1.4);
  const base=turret.angle;
  const list=time<spreadUntil?[base-.13,base,base+.13]:[base];
  for(const a of list) bullets.push({x:turret.x+Math.cos(a)*42,y:TY+Math.sin(a)*42,vx:Math.cos(a)*950,vy:Math.sin(a)*950});
  S.fire();
}
function burst(x,y,c,n,pow){
  for(let i=0;i<n;i++){
    const a=Math.random()*TAU,s=60+Math.random()*(160+pow*3),l=.35+Math.random()*.4;
    parts.push({x,y,vx:Math.cos(a)*s,vy:Math.sin(a)*s-40,life:l,max:l,sz:1.5+Math.random()*2,c:Math.random()<.22?'#ffffff':c});
  }
  parts.push({ring:true,x,y,r0:4,grow:70+pow*1.6,life:.38,max:.38,c});
}
function dropPup(x,y){
  const r=Math.random(); let type=r<.4?'rapid':r<.75?'spread':'shield';
  if(type==='shield'&&shield>=6)type='rapid';
  pups.push({type,x:clamp(x,30,W-30),y,r:15,t:Math.random()*6});
}
function applyPup(p){
  score+=25; let txt,c;
  if(p.type==='rapid'){ rapidUntil=time+8; txt='RAPID FIRE!'; c='#ffb42b'; }
  else if(p.type==='spread'){ spreadUntil=time+8; txt='SPREAD SHOT!'; c='#3fe0d0'; }
  else { shield=Math.min(6,shield+1); txt='HULL +1'; c='#b8f04d'; }
  floats.push({x:p.x,y:p.y-10,txt,c,life:1.2,size:19});
  burst(p.x,p.y,c,12,10); S.power();
}
function killEnemy(e){
  e.dead=true; kills++;
  combo++; comboTimer=3; mult=Math.min(8,1+Math.floor(combo/5)); if(mult>maxMult)maxMult=mult;
  const gain=e.pts*mult; score+=gain;
  floats.push({x:e.x,y:e.y-e.r-4,txt:'+'+gain,c:'#ffd98a',life:1,size:e.type==='brute'?26:17});
  if(combo%5===0){ floats.push({x:turret.x,y:TY-46,txt:'MULTIPLIER x'+mult,c:'#ffb42b',life:1.2,size:22}); S.wave(); }
  burst(e.x,e.y,e.color,e.type==='brute'?26:14,e.r);
  if(e.type==='splitter'){
    const sp=95*(1+(level-1)*.08);
    for(const d of[-1,1]) enemies.push({type:'mini',x:e.x+d*14,y:e.y,r:9,hp:1,maxHp:1,pts:8,color:'#b8f04d',t:0,seed:Math.random()*10,vx:d*60,vy:sp,flash:0,dead:false});
  }
  if(pups.length<2 && Math.random()<(e.type==='brute'?.5:e.type==='splitter'?.3:.07)) dropPup(e.x,e.y);
  S.kill(mult); shake=Math.max(shake,e.type==='brute'?7:2.5);
}
function explodeMine(e){
  e.dead=true; combo=0; mult=1; comboTimer=0;
  floats.push({x:e.x,y:e.y-8,txt:'MINE!',c:'#ff4d5e',life:1.2,size:24});
  burst(e.x,e.y,'#ff4d5e',30,26);
  parts.push({ring:true,x:e.x,y:e.y,r0:8,grow:170,life:.5,max:.5,c:'#ff9a5e'});
  shake=16; S.boom(); damageShield();
}
function hitEnemy(e){
  hits++; e.flash=1;
  if(e.type==='mine'){ explodeMine(e); return; }
  e.hp--;
  if(e.hp<=0) killEnemy(e);
  else { S.hit(); for(let i=0;i<3;i++){ const a=Math.random()*TAU; parts.push({x:e.x,y:e.y,vx:Math.cos(a)*120,vy:Math.sin(a)*120,life:.25,max:.25,sz:2,c:'#ffffff'}); } }
}
function damageShield(){
  if(state!==ST.PLAY)return;
  shield--; dmgFlash=Math.max(dmgFlash,.85);
  if(shield<=0){ shield=0; gameOver(); }
}

/* ============================== UPDATE ============================== */
function moveEnemy(e,dt){
  switch(e.type){
    case 'drone': e.y+=e.vy*dt; e.x+=Math.sin(e.t*2+e.seed)*26*dt; break;
    case 'speeder': e.x+=e.vx*dt; e.y+=e.vy*dt; if(e.x<e.r||e.x>W-e.r)e.vx*=-1; break;
    case 'splitter': e.y+=e.vy*dt; e.x+=Math.sin(e.t*1.4+e.seed)*20*dt; break;
    case 'mini': e.y+=e.vy*dt; e.x+=Math.sin(e.t*4+e.seed)*55*dt; break;
    case 'brute': e.y+=e.vy*dt; break;
    case 'mine': e.y+=e.vy*dt; e.x+=Math.sin(e.t*.9+e.seed)*18*dt; break;
  }
  e.x=clamp(e.x,e.r,W-e.r);
}
function stepFx(dt){
  for(let i=parts.length-1;i>=0;i--){ const p=parts[i]; p.life-=dt; if(p.life<=0){parts.splice(i,1);continue;}
    if(!p.ring){ p.x+=p.vx*dt; p.y+=p.vy*dt; p.vy+=240*dt; p.vx*=Math.max(0,1-2.4*dt); } }
  for(let i=floats.length-1;i>=0;i--){ const f=floats[i]; f.life-=dt*1.1; f.y-=44*dt; if(f.life<=0)floats.splice(i,1); }
  if(parts.length>520)parts.splice(0,parts.length-520);
}
function updatePlay(dt){
  time+=dt;
  const nl=1+Math.floor(time/25);
  if(nl!==level){ level=nl; banner('WAVE '+level); S.wave(); }
  spawnT-=dt;
  const cap=Math.min(26,5+level*3), interval=Math.max(.45,1.5*Math.pow(.9,level-1));
  if(spawnT<=0&&enemies.length<cap){ spawnEnemy(pickType()); spawnT=interval*rand(.7,1.3); }

  turret.x+=(clamp(mouse.x,50,W-50)-turret.x)*Math.min(1,dt*10);
  turret.angle=Math.atan2(mouse.y-TY,mouse.x-turret.x);
  recoil=Math.max(0,recoil-dt*6); muzzleT-=dt; fireCd-=dt;
  if(firing) fire();

  if(comboTimer>0){ comboTimer-=dt; if(comboTimer<=0){combo=0;mult=1;} }

  for(let i=bullets.length-1;i>=0;i--){
    const b=bullets[i]; b.x+=b.vx*dt; b.y+=b.vy*dt;
    let used=false;
    for(let j=enemies.length-1;j>=0;j--){
      const e=enemies[j]; if(e.dead)continue;
      const dx=e.x-b.x,dy=e.y-b.y,rr=e.r+5;
      if(dx*dx+dy*dy<rr*rr){ hitEnemy(e); bullets.splice(i,1); used=true; break; }
    }
    if(!used){
      for(let j=pups.length-1;j>=0;j--){
        const p=pups[j],dx=p.x-b.x,dy=p.y-b.y;
        if(dx*dx+dy*dy<400){ applyPup(p); pups.splice(j,1); bullets.splice(i,1); used=true; break; }
      }
    }
    if(!used&&out(b)){
      if(combo>=5) floats.push({x:clamp(b.x,40,W-40),y:clamp(b.y,40,H-40),txt:'COMBO LOST',c:'#ff4d5e',life:.9,size:16});
      combo=0; mult=1; comboTimer=0; bullets.splice(i,1);
    }
  }

  for(let i=enemies.length-1;i>=0;i--){
    const e=enemies[i]; if(e.dead){enemies.splice(i,1);continue;}
    e.t+=dt; e.flash=Math.max(0,e.flash-dt*5); moveEnemy(e,dt);
    if(e.y-e.r>BASE_Y){
      enemies.splice(i,1); burst(e.x,BASE_Y,'#ff4d5e',12,e.r); S.baseHit(); damageShield();
      if(state!==ST.PLAY)break;
    }
  }
  for(let i=pups.length-1;i>=0;i--){ const p=pups[i]; p.t+=dt; p.y+=70*dt; p.x+=Math.sin(p.t*2)*22*dt; if(p.y>H-46)pups.splice(i,1); }

  stepFx(dt);
  shake=Math.max(0,shake-dt*30); dmgFlash=Math.max(0,dmgFlash-dt*1.4);
  updateHUD(false);
}
function out(b){ return b.x<-30||b.x>W+30||b.y<-40||b.y>H+30; }
function updateMenu(dt){
  menuT-=dt;
  if(menuT<=0&&enemies.length<10){ spawnEnemy(menuPick()); menuT=rand(1,1.8); }
  for(let i=enemies.length-1;i>=0;i--){ const e=enemies[i]; e.t+=dt; moveEnemy(e,dt); if(e.y>H+60)enemies.splice(i,1); }
  stepFx(dt); shake=Math.max(0,shake-dt*30);
}
function updateAftermath(dt){
  for(let i=enemies.length-1;i>=0;i--){ const e=enemies[i]; e.t+=dt; moveEnemy(e,dt); if(e.y>H+60)enemies.splice(i,1); }
  stepFx(dt); shake=Math.max(0,shake-dt*30); dmgFlash=Math.max(0,dmgFlash-dt*1.4);
}

function poly(n,r,rot){
  X.beginPath();
  for(let i=0;i<n;i++){ const a=rot+i/n*TAU; const x=Math.cos(a)*r,y=Math.sin(a)*r; i?X.lineTo(x,y):X.moveTo(x,y); }
  X.closePath();
}
function drawShape(type,r,t,flash,e){
  switch(type){
    case 'drone':{
      X.save(); X.rotate(t*.9);
      poly(6,r,0); X.fillStyle='#3a1418'; X.fill(); X.lineWidth=2; X.strokeStyle='#ff6b5e'; X.stroke();
      X.restore();
      X.save(); X.rotate(-t*1.6);
      X.beginPath(); X.arc(0,0,r*.5,0,TAU); X.strokeStyle='rgba(255,107,94,.55)'; X.lineWidth=1.5; X.stroke();
      X.restore();
      X.beginPath(); X.arc(0,0,r*.22,0,TAU); X.fillStyle='#ffb3ab'; X.fill(); break;
    }
    case 'speeder':{
      X.strokeStyle='rgba(255,210,63,.4)'; X.lineWidth=2;
      X.beginPath(); X.moveTo(0,-r*.9); X.lineTo(0,-r*2.1);
      X.moveTo(-r*.5,-r*.7); X.lineTo(-r*.8,-r*1.8);
      X.moveTo(r*.5,-r*.7); X.lineTo(r*.8,-r*1.8); X.stroke();
      X.beginPath(); X.moveTo(0,r); X.lineTo(-r*.72,-r*.75); X.lineTo(0,-r*.35); X.lineTo(r*.72,-r*.75); X.closePath();
      X.fillStyle='#4a3a10'; X.fill(); X.lineWidth=2; X.strokeStyle='#ffd23f'; X.stroke(); break;
    }
    case 'splitter':{
      for(let i=0;i<4;i++){ const a=t*.7+i*Math.PI/2;
        X.beginPath(); X.arc(Math.cos(a)*r*.8,Math.sin(a)*r*.8,r*.5,0,TAU);
        X.fillStyle='#1c3312'; X.fill(); X.lineWidth=2; X.strokeStyle='#b8f04d'; X.stroke(); }
      X.beginPath(); X.arc(0,0,r*.85,0,TAU); X.fillStyle='#1c3312'; X.fill(); X.strokeStyle='#b8f04d'; X.lineWidth=2; X.stroke();
      X.beginPath(); X.arc(0,0,r*.3*(1+.15*Math.sin(t*5)),0,TAU); X.fillStyle='#e9ffb0'; X.fill(); break;
    }
    case 'mini':{
      X.beginPath(); X.arc(0,0,r,0,TAU); X.fillStyle='#1c3312'; X.fill(); X.lineWidth=2; X.strokeStyle='#b8f04d'; X.stroke();
      X.beginPath(); X.arc(0,0,r*.35,0,TAU); X.fillStyle='#e9ffb0'; X.fill(); break;
    }
    case 'brute':{
      poly(8,r,t*.25); X.fillStyle='#2a1230'; X.fill(); X.lineWidth=3; X.strokeStyle='#e06bff'; X.stroke();
      poly(8,r*.55,-t*.4); X.strokeStyle='rgba(224,107,255,.5)'; X.lineWidth=1.5; X.stroke();
      for(let i=0;i<4;i++){ const a=i*Math.PI/2+Math.PI/4;
        X.beginPath(); X.arc(Math.cos(a)*r*.78,Math.sin(a)*r*.78,2.2,0,TAU); X.fillStyle='#e06bff'; X.fill(); }
      if(e){ const f=e.hp/e.maxHp;
        X.beginPath(); X.arc(0,0,r+8,-Math.PI/2,-Math.PI/2+f*TAU); X.strokeStyle='rgba(224,107,255,.85)'; X.lineWidth=3; X.stroke(); }
      break;
    }
    case 'mine':{
      const p=.5+.5*Math.sin(t*6);
      X.save(); X.rotate(t*.5);
      for(let i=0;i<8;i++){ const a=i*Math.PI/4;
        X.beginPath();
        X.moveTo(Math.cos(a-.16)*r,Math.sin(a-.16)*r);
        X.lineTo(Math.cos(a)*(r+7),Math.sin(a)*(r+7));
        X.lineTo(Math.cos(a+.16)*r,Math.sin(a+.16)*r); X.closePath();
        X.fillStyle='#5b6b8c'; X.fill(); }
      X.restore();
      X.beginPath(); X.arc(0,0,r,0,TAU); X.fillStyle='#141a26'; X.fill(); X.lineWidth=2; X.strokeStyle='#5b6b8c'; X.stroke();
      X.beginPath(); X.arc(0,0,r*.34*(1+.25*p),0,TAU); X.fillStyle='rgba(255,77,94,'+(.7+.3*p).toFixed(3)+')'; X.fill();
      X.globalAlpha=.12+.2*p; X.beginPath(); X.arc(0,0,r+11,0,TAU); X.strokeStyle='#ff4d5e'; X.lineWidth=1.5; X.stroke(); X.globalAlpha=1; break;
    }
  }
  if(flash>0){
    X.globalCompositeOperation='lighter';
    X.beginPath(); X.arc(0,0,r*.95,0,TAU);
    X.fillStyle='rgba(255,255,255,'+Math.min(.85,flash).toFixed(3)+')'; X.fill();
    X.globalCompositeOperation='source-over';
  }
}

function drawPup(p){
  const c=p.type==='rapid'?'#ffb42b':p.type==='spread'?'#3fe0d0':'#b8f04d';

function drawParts(){

function frame(now){
  requestAnimationFrame(frame);
  let dt=(now-last)/1000; last=now; if(dt>.05)dt=.05;
  T+=dt;
  if(state!==ST.PAUSE) stepStars(dt);
  if(state===ST.PLAY) updatePlay(dt);
  else if(state===ST.MENU) updateMenu(dt);
  else if(state===ST.OVER) updateAftermath(dt);
  render();
}

  mouse.x=x; mouse.y=y;
  initAudio(); if(AC&&AC.state==='suspended')AC.resume();

  if(state===ST.OVER){ if(performance.now()-overAt>700) startGame(); return; }
  if(state===ST.PAUSE){ resume(); firing=true; return; }
  if(state===ST.PLAY) firing=true;
}
