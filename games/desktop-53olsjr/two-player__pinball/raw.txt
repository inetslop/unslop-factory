```js
function closestOnSeg(px,py,ax,ay,bx,by){
  const abx=bx-ax, aby=by-ay;
  const t=Math.max(0,Math.min(1,((px-ax)*abx+(py-ay)*aby)/(abx*abx+aby*aby)));
  return [ax+abx*t, ay+aby*t];
}
```

```js
function collideSeg(b, ax,ay,bx,by, e=0.45){
  const [cx,cy]=closestOnSeg(b.x,b.y,ax,ay,bx,by);
  let dx=b.x-cx, dy=b.y-cy; const d2=dx*dx+dy*dy;
  if(d2 >= b.r*b.r) return false;
  let d=Math.sqrt(d2)||0.0001; let nx=dx/d, ny=dy/d;
  // if center is exactly on segment (d≈0), pick normal from segment direction
  if(d<0.001){ const len=Math.hypot(bx-ax,by-ay); nx=-(by-ay)/len; ny=(bx-ax)/len; }
  b.x=cx+nx*b.r; b.y=cy+ny*b.r;
  const vn=b.vx*nx+b.vy*ny;
  if(vn<0){
    b.vx-=(1+e)*vn*nx; b.vy-=(1+e)*vn*ny;
    // tangential friction
    const tx=-ny, ty=nx; const vt=b.vx*tx+b.vy*ty;
    b.vx-=vt*0.012*tx; b.vy-=vt*0.012*ty;  // hmm careful: subtract a small fraction of tangential velocity

  return vn<-40; // register as hit if significant
}
```

```js
function collideSeg(b,ax,ay,bx,by,e){
  const abx=bx-ax,aby=by-ay; const L2=abx*abx+aby*aby;
  let t=((b.x-ax)*abx+(b.y-ay)*aby)/L2; t=t<0?0:t>1?1:t;

  let dx=b.x-cx, dy=b.y-cy; let d2=dx*dx+dy*dy;
  if(d2>=b.r*b.r) return 0;
  let d=Math.sqrt(d2); let nx,ny;
  if(d<1e-4){ const L=Math.sqrt(L2); nx=-aby/L; ny=abx/L; d=0; } else { nx=dx/d; ny=dy/d; }
  b.x=cx+nx*(b.r+0.05); b.y=cy+ny*(b.r+0.05);
  const vn=b.vx*nx+b.vy*ny;
```

```html
/* ============================== PHYSICS ============================== */
function collideSeg(b,ax,ay,bx,by,e){
  const abx=bx-ax,aby=by-ay,L2=abx*abx+aby*aby;
  let t=((b.x-ax)*abx+(b.y-ay)*aby)/L2;t=t<0?0:t>1?1:t;
  const cx=ax+abx*t,cy=ay+aby*t;
  let dx=b.x-cx,dy=b.y-cy,d2=dx*dx+dy*dy;
  if(d2>=b.r*b.r)return 0;
  let d=Math.sqrt(d2),nx,ny;
  if(d<1e-4){const L=Math.sqrt(L2);nx=-aby/L;ny=abx/L;}else{nx=dx/d;ny=dy/d;}
  b.x=cx+nx*(b.r+0.05);b.y=cy+ny*(b.r+0.05);
  const vn=b.vx*nx+b.vy*ny;
  if(vn<0){
    b.vx-=(1+e)*vn*nx;b.vy-=(1+e)*vn*ny;
    const tx=-ny,ty=nx,vt=b.vx*tx+b.vy*ty,fr=vt*0.004;
    b.vx-=fr*tx;b.vy-=fr*ty;
    return -vn;
  }
  return 0;
}
function collideFlipper(b,f){
  const ca=Math.cos(f.angle),sa=Math.sin(f.angle);
  const tx=f.px+ca*f.len,ty=f.py+sa*f.len;
  let t=((b.x-f.px)*(tx-f.px)+(b.y-f.py)*(ty-f.py))/(f.len*f.len);t=t<0?0:t>1?1:t;
  const cx=f.px+(tx-f.px)*t,cy=f.py+(ty-f.py)*t;
  let dx=b.x-cx,dy=b.y-cy;const rr2=b.r+f.rad,d2=dx*dx+dy*dy;
  if(d2>=rr2*rr2)return;
  let d=Math.sqrt(d2),nx,ny;
  if(d<1e-4){nx=-sa;ny=ca;}else{nx=dx/d;ny=dy/d;}
  b.x=cx+nx*(rr2+0.05);b.y=cy+ny*(rr2+0.05);
  const rx=cx-f.px,ry=cy-f.py;
  let svx=-f.angVel*ry,svy=f.angVel*rx;
  const ssp=Math.hypot(svx,svy);if(ssp>2400){svx*=2400/ssp;svy*=2400/ssp;}
  const vn=(b.vx-svx)*nx+(b.vy-svy)*ny;
  if(vn<0){
    const e=(Math.abs(f.angVel)>2||-vn>300)?0.35:0;
    b.vx-=(1+e)*vn*nx;b.vy-=(1+e)*vn*ny;
  }
}
function hitCircle(b,cx,cy,r,e,kick){
  let dx=b.x-cx,dy=b.y-cy;const rr2=b.r+r,d2=dx*dx+dy*dy;
  if(d2>rr2*rr2||d2<1e-6)return 0;
  const d=Math.sqrt(d2),nx=dx/d,ny=dy/d;
  b.x=cx+nx*rr2;b.y=cy+ny*rr2;
  const vn=b.vx*nx+b.vy*ny;let imp=0;
  if(vn<0){b.vx-=(1+e)*vn*nx;b.vy-=(1+e)*vn*ny;imp=-vn;}
  if(kick){b.vx+=nx*kick;b.vy+=ny*kick;}
  return imp;
}
function bumperHit(b,bp){
  const i=b.p;bp.pop=1;
  shake=Math.max(shake,bp.core?7:3);
  spawnSparks(b.x,b.y,bp.core?'#ffc23e':'#ff3d81',bp.core?26:14,bp.core?420:300);
  addPoints(i,bp.pts,bp.x,bp.y-bp.r-16);lightUp(i);sfx.bumper(bp.core);
}
function slingHit(b,s){

  shake=Math.max(shake,2.5);
  spawnSparks((b.x+s.ax+s.bx)/3,(b.y+s.ay+s.by)/3,'#ff4d6a',8,240);
  addPoints(i,25,(s.ax+s.bx)/2+s.nx*34,(s.ay+s.by)/2+s.ny*34);
  lightUp(i);sfx.sling();
}
function collideBallsPair(a,b){
  let dx=b.x-a.x,dy=b.y-a.y;const rr2=a.r+b.r,d2=dx*dx+dy*dy;
  if(d2>=rr2*rr2)return;
  let d=Math.sqrt(d2);if(d<1e-4){d=1e-4;dx=0;dy=1;}
  const nx=dx/d,ny=dy/d,ov=(rr2-d)/2;
  a.x-=nx*ov;a.y-=ny*ov;b.x+=nx*ov;b.y+=ny*ov;
  const vn=(b.vx-a.vx)*nx+(b.vy-a.vy)*ny;
  if(vn<0){
    const j=-1.9*vn/2;
    a.vx-=j*nx;a.vy-=j*ny;b.vx+=j*nx;b.vy+=j*ny;
    if(-vn>260){spawnSparks((a.x+b.x)/2,(a.y+b.y)/2,'#ffffff',8,220);sfx.click();}
  }
}
function physStep(h){
  for(const f of flippers){
    const target=f.pressed?f.upA:f.restA,speed=f.pressed?24:10;
    const da=target-f.angle,mx=speed*h;
    const mv=da>0?Math.min(da,mx):Math.max(da,-mx);
    f.angVel=mv/h;f.angle+=mv;
  }
  for(const b of balls){
    b.vy+=GRAV*h;
    const dm=1-0.05*h;b.vx*=dm;b.vy*=dm;
    const sp=Math.hypot(b.vx,b.vy);if(sp>2400){b.vx*=2400/sp;b.vy*=2400/sp;}
    b.x+=b.vx*h;b.y+=b.vy*h;
    for(const w of WALLS)collideSeg(b,w.ax,w.ay,w.bx,w.by,0.42);
    for(const s of slings){
      const imp=collideSeg(b,s.ax,s.ay,s.bx,s.by,0.5);
      if(imp>60){b.vx+=s.nx*720;b.vy+=s.ny*720;slingHit(b,s);}
    }
    for(const f of flippers)collideFlipper(b,f);
    for(const bp of bumpers)if(hitCircle(b,bp.x,bp.y,bp.r,0.5,bp.kick)>40)bumperHit(b,bp);
    for(const ps of posts){
      const imp=hitCircle(b,ps.x,ps.y,ps.r,0.5,160);
      if(imp>220){spawnSparks(b.x,b.y,'#9fb4cc',5,180);sfx.click();}
    }
    if(b.spawnT>0)b.spawnT-=h;
  }
  if(balls.length===2)collideBallsPair(balls[0],balls[1]);
  for(let i=balls.length-1;i>=0;i--){
    const b=balls[i];
    if(b.y-b.r>LH+30||b.x<-80||b.x>LW+80){balls.splice(i,1);drainBook(b);}
  }
}

function render(){
  ctx.setTransform(1,0,0,1,0,0);
  const bg=ctx.createLinearGradient(0,0,0,cv.height);
  bg.addColorStop(0,'#05070f');bg.addColorStop(1,'#03040a');
  ctx.fillStyle=bg;ctx.fillRect(0,0,cv.width,cv.height);
  const shx=(Math.random()*2-1)*shake,shy=(Math.random()*2-1)*shake;
  ctx.setTransform(scale,0,0,scale,ox+shx*scale,oy+shy*scale);

  drawStars();drawGlows();drawFelt();
  slings.forEach(drawSling);posts.forEach(drawPost);
  drawLights(0);drawLights(1);
  bumpers.forEach(drawBumper);
  drawWalls();
  flippers.forEach(drawFlipper);
  for(const b of balls){b.trail.push({x:b.x,y:b.y});if(b.trail.length>14)b.trail.shift();}
  balls.forEach(drawBall);
  drawParticles();drawFloats();
  if(flashFx.t>0){ctx.fillStyle='rgba('+flashFx.c+','+(0.3*flashFx.t).toFixed(3)+')';ctx.fillRect(0,0,LW,LH);}

  if(state==='play'||state==='over')drawHUD();
  if(state==='play'&&!paused)drawPrompts();
  drawMsg();
  if(state==='menu')drawMenu();
  else if(state==='over')drawOver();
  else if(paused&&state==='play')drawPause();
}

let last=performance.now(),acc=0;
function frame(now){
  requestAnimationFrame(frame);
  let dt=(now-last)/1000;last=now;
  if(dt>0.05)dt=0.05;
  time+=dt;
  updateEffects(dt);
  if(state==='play'&&!paused){
    acc+=dt;let n=0;
    while(acc>=STEP&&n<12){physStep(STEP);acc-=STEP;n++;}
    if(n===12)acc=0;
    const a=players[0],b=players[1];
    if(endPending===0&&a.left===0&&b.left===0&&balls.length===0)endPending=0.9;
  }else acc=0;
  if(endPending>0){endPending-=dt;if(endPending<=0)finalizeOver();}
  render();
}
requestAnimationFrame(frame);
```
