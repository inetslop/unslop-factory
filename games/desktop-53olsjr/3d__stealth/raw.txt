function updGuard(g,dt,see){
  const p=player;
  if(state==='ALARM'||state==='FAILED'){
    const d=Math.hypot(p.x-g.x,p.z-g.z)||.001;
    g.x+=(p.x-g.x)/d*4.6*dt; g.z+=(p.z-g.z)/d*4.6*dt;
    g.ang+=angDiff(Math.atan2(p.x-g.x,p.z-g.z),g.ang)*(1-Math.exp(-8*dt));
    return;
  }
  const dx=p.x-g.x,dz=p.z-g.z,d=Math.hypot(dx,dz)||.0001;
  const range=effRange();
  const dot=(dx/d)*Math.sin(g.ang)+(dz/d)*Math.cos(g.ang);
  let vis=false;
  if(see&&losClear(g.x,g.z,p.x,p.z)){
    if(d<1.5)vis=true; else if(d<range&&dot>Math.cos(FOV/2))vis=true;
  }
  if(vis){
    if(dot<-0.25)g.susp=Math.min(1,g.susp+0.3);
    const rate=(0.5+1.4*(1-d/range))*(p.mode==='sneak'?0.7:p.mode==='sprint'?1.3:1);
    g.susp=Math.min(1,g.susp+rate*dt);
    g.lastSeen={x:p.x,z:p.z};
  }else{
    g.susp=Math.max(0,g.susp-(g.mode==='scan'?0.25:0.5)*dt);
    if(g.mode==='patrol'&&g.susp>0.3&&g.lastSeen){g.mode='toPoint';g.tx=g.lastSeen.x;g.tz=g.lastSeen.z;g.stuck=0;}
  }
  if(g.susp>=1&&state==='PLAYING'){triggerAlarm();return;}

  let spd=1.7,tx=null,tz=null;
  if(g.mode==='patrol'){
    const wp=g.route[g.wp]; tx=wp[0]; tz=wp[1];
    if(Math.hypot(tx-g.x,tz-g.z)<0.2)g.wp=(g.wp+1)%g.route.length;
  }else if(g.mode==='toPoint'){
    spd=3.2; tx=g.tx; tz=g.tz;
    if(Math.hypot(tx-g.x,tz-g.z)<0.3){g.mode='scan';g.scanT=2.6;g.faceBase=g.ang;}
  }else if(g.mode==='scan'){
    g.scanT-=dt;
    g.ang=g.faceBase+Math.sin((2.6-g.scanT)*3.0)*1.05;
    if(g.susp<0.05)g.scanT=0;
    if(g.scanT<=0){
      let bi=0,bd=1e9;
      for(let i=0;i<g.route.length;i++){const dd=(g.x-g.route[i][0])**2+(g.z-g.route[i][1])**2;if(dd<bd){bd=dd;bi=i;}}
      g.wp=bi; g.mode='patrol';
    }
  }
  if(tx!=null){
    const ox=g.x,oz=g.z;
    slide(g,(tx-g.x)*spd*dt,(tz-g.z)*spd*dt,g.r);
    if(Math.hypot(g.x-ox,g.z-oz)<spd*dt*0.25)g.stuck+=dt; else g.stuck=0;
    if(g.stuck>1.4){let bi=0,bd=1e9;for(let i=0;i<g.route.length;i++){const dd=(g.x-g.route[i][0])**2+(g.z-g.route[i][1])**2;if(dd<bd){bd=dd;bi=i;}}g.wp=bi;g.mode='patrol';g.stuck=0;}
    const want=(vis&&see)?Math.atan2(dx,dz):Math.atan2(tx-g.x,tz-g.z);
    g.ang+=angDiff(want,g.ang)*(1-Math.exp(-9*dt));
  }
}

function update(dt){
  T+=0; // (T set in tick)
  if(state==='PAUSED'||state==='FAILED')return;

  const see=state==='PLAYING';

  /* --- player --- */
  if(see){
    let vx=((k.KeyW||k.ArrowUp)?1:0)-((k.KeyS||k.ArrowDown)?1:0);
    let vz=((k.KeyD||k.ArrowRight)?1:0)-((k.KeyA||k.ArrowLeft)?1:0);
    const mvx=vx*FCX+vz*RCX, mvz=vx*FCZ+vz*RCZ;
    const len=Math.hypot(mvx,mvz);
    player.mode=k.Space?'sprint':(k.ShiftLeft||k.ShiftRight)?'sneak':'walk';
    const spd=player.mode==='sneak'?1.8:player.mode==='sprint'?5.0:3.4;
    if(len>0){
      slide(player,mvx/len*spd*dt,mvz/len*spd*dt,player.r);
      player.ang+=angDiff(Math.atan2(mvx,mvz),player.ang)*(1-Math.exp(-12*dt));
      stepT-=dt;
      if(stepT<=0){
        const m=player.mode;
        stepT=m==='sneak'?0.6:m==='walk'?0.44:0.3;
        sfx.step(m==='sneak'?0.05:m==='walk'?0.11:0.2, m==='sneak'?380:m==='walk'?620:900);
      }
      if(player.mode==='sprint'){
        pingT-=dt;
        if(pingT<=0){pingT=0.9;ripples.push({x:player.x,z:player.z,t:0});sfx.ping();hearCheck(player.x,player.z,5.6);}
      }
    } else { stepT=Math.min(stepT,0.25); }

    /* drives */
    for(const dr of drives){
      if(dr.got)continue;
      if(Math.hypot(player.x-dr.x,player.z-dr.z)<0.55){
        dr.got=true; collected++; sfx.pickup();
        toast('DATA DRIVE ACQUIRED — '+collected+'/3','cy');
        if(collected===3){exitOpen=true;sfx.unlock();toast('EXTRACTION UNLOCKED — REACH THE EXIT PAD','gr');}
      }
    }
    /* exit */
    const dE=Math.hypot(player.x-exitCell.x,player.z-exitCell.z);
    if(exitOpen&&dE<0.8){
      state='WON'; wonT=0; ripples.push({x:exitCell.x,z:exitCell.z,t:0,g:true});
      sfx.win(); shakeV=0.25;
      const bm=$('bigMsg'); bm.textContent='EXTRACTION COMPLETE'; bm.style.color='var(--gr)';
      void bm.offsetWidth; bm.classList.add('show');
    } else if(!exitOpen&&dE<2.3){
      lockHintT-=dt;
      if(lockHintT<=0){lockHintT=4;toast('EXTRACTION LOCKED — '+(3-collected)+' DRIVE'+(3-collected>1?'S':'')+' REMAIN','rd');}
    }
  }

  /* --- guards --- */
  for(const g of guards)updGuard(g,dt,see);
  maxSusp=0; for(const g of guards)if(g.susp>maxSusp)maxSusp=g.susp;
  if(see||state==='ALARM')peakSusp=Math.max(peakSusp,maxSusp);

  /* heartbeat */
  hbT-=dt;
  if(maxSusp>0.3&&hbT<=0){
    const it=(maxSusp-0.3)/0.7;
    sfx.thump(0.14+0.2*it);
    hbT=1.15-0.78*it;
  }

  if(state==='ALARM'){
    alarmT+=dt; playT+=dt;
    if(alarmT>1.7){state='FAILED';$('bigMsg').classList.remove('show');sfx.fail();showEnd(false);}
  }
  if(state==='PLAYING')playT+=dt;
  if(state==='WON'){
    wonT+=dt;
    if(wonT>0.9&&!$('scrEnd').hidden===false){state='DONE';$('bigMsg').classList.remove('strobe');showEnd(true);}
  }

  /* ripples */
  for(let i=ripples.length-1;i>=0;i--){ripples[i].t+=dt/1.1;if(ripples[i].t>1)ripples.splice(i,1);}
  heardToastT-=dt;

  /* --- camera --- */
  let tx,tz,dist,hgt;
  if(state==='START'){
    camYawT+=dt*0.22; tx=MW/2; tz=MH/2-0.5; dist=10.5; hgt=7.2;
  }else{
    tx=player.x+Math.sin(player.ang)*0.5; tz=player.z+Math.cos(player.ang)*0.5;
    dist=state==='WON'?8.4+wonT*2.2:8.4; hgt=6.3;
  }
  camYawCur+=angDiff(camYawT,camYawCur)*(1-Math.exp(-7*dt));
  FCX=Math.sin(camYawCur);FCZ=Math.cos(camYawCur);RCX=Math.cos(camYawCur);RCZ=-Math.sin(camYawCur);
  const k2=1-Math.exp(-5.5*dt);
  camX+=(tx-FCX*dist-camX)*k2;
  camZ+=(tz-FCZ*dist-camZ)*k2;
  camY+=(hgt-camY)*k2;
  shakeV*=Math.exp(-3*dt);
}

/* ---------------- render ---------------- */
function conePts(g){
  const range=effRange(), pts=[[g.x,g.z]], N=25, base=g.ang-FOV/2;
  for(let i=0;i<=N;i++){
    const a=base+FOV*i/N, dx=Math.sin(a),dz=Math.cos(a);
    let t=1;
    for(let s=0.3;s<range;s+=0.24){ if(solid(g.x+dx*s,g.z+dz*s)){t=s/range;break;} }
    pts.push([g.x+dx*range*t, g.z+dz*range*t]);
  }
  return pts;
}
function shadowQuad(x,z,ang,w,d){
  const cs=Math.cos(ang),sn=Math.sin(ang);
  const c=[[w/2,d/2],[w/2,-d/2],[-w/2,-d/2],[-w/2,d/2]]
    .map(p=>[x+p[0]*cs+p[1]*sn, z-p[0]*sn+p[1]*cs]);
  if(floorPath(c,0.03)){ctx.fillStyle='rgba(0,0,0,.34)';ctx.fill();}
}
function collectBox(b,items){
  const tc=b.c.map(tCam);
  for(const f of b.faces){
    if(f.n[0]*(camX-b.cx)+f.n[1]*(camY-b.cy)+f.n[2]*(camZ-b.cz)<=0)continue;
    let poly=[tc[f.i[0]],tc[f.i[1]],tc[f.i[2]],tc[f.i[3]]];
    if(poly.every(p=>p.z<NEAR))continue;
    poly=clipNear(poly); if(poly.length<3)continue;
    let d=0; for(const p of poly)d+=p.z; d/=poly.length;
    const q=poly.map(p=>[W/2+p.x*F/p.z, H/2-p.y*F/p.z]);
    items.push({d,q,f:shadeStr(b.col,f.n,d)});
  }
}

      let pc=pts.map(tCam);
      if(pc.every(q=>q.z<NEAR))return;
      pc=clipNear(pc); if(pc.length<3)return;
      let d=0;for(const q of pc)d+=q.z;d/=pc.length;
      items.push({d,fn(){
        ctx.globalAlpha=aA;ctx.beginPath();
        for(let i=0;i<pc.length;i++){const k=F/pc[i].z;i?ctx.lineTo(W/2+pc[i].x*k,H/2-pc[i].y*k):ctx.moveTo(W/2+pc[i].x*k,H/2-pc[i].y*k);}
        ctx.closePath();
        ctx.fillStyle=exitOpen?'#54f2a5':'#7d93ab';ctx.fill();ctx.globalAlpha=1;
      }});
    };
    beam([[cx0-w/2,0,cz0],[cx0+w/2,0,cz0],[cx0+w/2,hgt,cz0],[cx0-w/2,hgt,cz0]]);
    beam([[cx0,0,cz0-w/2],[cx0,0,cz0+w/2],[cx0,hgt,cz0+w/2],[cx0,hgt,cz0-w/2]]);
  }

  items.sort((a,b)=>b.d-a.d);
  for(const it of items){
    if(it.q){
      ctx.beginPath();
      for(let i=0;i<it.q.length;i++)i?ctx.lineTo(it.q[i][0],it.q[i][1]):ctx.moveTo(it.q[i][0],it.q[i][1]);
      ctx.closePath();
      ctx.fillStyle=it.f;ctx.fill();
      ctx.strokeStyle='rgba(5,9,16,.35)';ctx.lineWidth=1;ctx.stroke();
    } else it.fn();
  }
  ctx.restore();

  updateHUD(); drawMinimap();
}

const mmS=7, mmc=document.createElement('canvas'), mctx=mmc.getContext('2d');
mmc.width=MW*mmS; mmc.height=MH*mmS;
function buildMinimap(){
  mctx.fillStyle='#0a121d';mctx.fillRect(0,0,mmc.width,mmc.height);
  for(let r=0;r<MH;r++)for(let c=0;c<MW;c++){
    const s=grid[r][c];
    if(s===1){mctx.fillStyle='#3c4f66';mctx.fillRect(c*mmS,r*mmS,mmS,mmS);}
    else if(s===2){mctx.fillStyle='#7c6b45';mctx.fillRect(c*mmS+1,r*mmS+1,mmS-2,mmS-2);}
  }
}
function drawMinimap(){
  const mc=$('mm').getContext('2d');
  mc.clearRect(0,0,mmc.width,mmc.height);
  mc.drawImage(mmc,0,0);
  /* drives */
  for(const dr of drives){
    if(dr.got)continue;
    const x=dr.x*mmS,y=dr.z*mmS,p=1+0.25*Math.sin(T*4+dr.ph);
    mc.fillStyle='#6ef3ff';mc.beginPath();
    mc.moveTo(x,y-3*p);mc.lineTo(x+3*p,y);mc.lineTo(x,y+3*p);mc.lineTo(x-3*p,y);mc.closePath();mc.fill();
  }
  /* exit */
  const ex=exitCell.x*mmS, ez=exitCell.z*mmS;
  mc.fillStyle=exitOpen?(Math.sin(T*5)>0?'#54f2a5':'#1d6b47'):'#33475e';
  mc.fillRect(ex-3,ez-3,6,6);
  /* guards */
  for(const g of guards){
    const x=g.x*mmS,y=g.z*mmS;
    const pa=Math.atan2(Math.cos(g.ang),Math.sin(g.ang));
    mc.fillStyle='rgba(255,178,84,.16)';
    mc.beginPath();mc.moveTo(x,y);
    mc.arc(x,y,rangeMM(),pa-FOV/2,pa+FOV/2);mc.closePath();mc.fill();
    mc.fillStyle=g.susp>0.55?'#ff5566':'#ffb347';
    mc.beginPath();mc.arc(x,y,2.2,0,TAU);mc.fill();
  }
  /* player */
  const px=player.x*mmS,pz=player.z*mmS;
  const dx=Math.sin(player.ang),dz=Math.cos(player.ang);
  mc.fillStyle='#5fd9ff';mc.beginPath();
  mc.moveTo(px+dx*5,pz+dz*5);mc.lineTo(px-dx*3-dz*2.6,pz-dz*3+dx*2.6);mc.lineTo(px-dx*3+dz*2.6,pz-dz*3-dx*2.6);
  mc.closePath();mc.fill();
}
function rangeMM(){return effRange()*mmS*0.75;}
