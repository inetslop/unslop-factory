function mirrorY(y){const lo=ball.r,hi=H-ball.r,span=hi-lo;if(span<=0)return H/2;
  let m=(y-lo)%(2*span);if(m<0)m+=2*span;return m<=span?lo+m:hi-(m-span);}
function errMag(){
  if(G.alarm>0)return 3;
  const base=30-9*Math.min(1,(G.score[0]+G.score[1])/(WIN*2));
  return G.silentFlight?base*2.3:base;
}
function predict(){
  const plane=warden.x-warden.w/2-ball.r;
  if(ball.vx<=0)return ball.y;
  const t=(plane-ball.x)/ball.vx;
  if(t<0||t>6)return ball.y;
  return mirrorY(ball.y+ball.vy*t)+warden.err;
}
function aiMove(p,target,sp,dt){p.y+=clamp(target-p.y,-sp*dt,sp*dt);clampPaddle(p);}

function launch(){
  const ang=rand(-0.5,0.5),sp=B_BASE*(1+0.03*(G.score[0]+G.score[1]));
  ball.x=W/2;ball.y=H/2;
  ball.vx=Math.cos(ang)*sp*G.dir;ball.vy=Math.sin(ang)*sp;
  ball.tint=G.demo?'#eaf6ff':(Math.random()<.5?'#ffe9c4':'#eaf6ff');
  ball.trail.length=0;G.sub='rally';tone(300,520,.08,'sine',.1);
}
function triggerAlarm(){
  G.catches++;G.alarm=4.2;G.failPending=G.catches>=MAXCATCH;
  document.body.classList.add('alarm');sAlarm();G.shake=9;
  floater('!! LOCATED',W/2,H*0.3,'#ff4d5e',30);
}
function pointScored(who){
  if(G.demo){G.sub='serve';G.serveT=1.0;G.dir=Math.random()<.5?1:-1;ball.vx=ball.vy=0;return;}
  G.score[who]++;G.rally=0;G.silentFlight=false;
  const yx=clamp(ball.y,40,H-40);
  if(who===0){burst(W-26,yx,'#ffb347',26,320);floater('BREACH +1',W*0.3,H*0.3,'#ffb347',18);sScoreYou();}
  else{burst(26,yx,'#ff5d6e',26,320);floater('WARDEN +1',W*0.7,H*0.3,'#ff5d6e',18);sScoreWarden();}
  G.shake=7;
  if(G.score[who]>=WIN){endGame(who===0?'win':'points');return;}
  G.sub='serve';G.serveT=1.15;G.dir=who===0?1:-1;ball.vx=ball.vy=0;ball.trail.length=0;
}
function hitPaddle(side,planeX,ph){
  const py=side===1?player.y:warden.y;
  let off=clamp((ball.y-py)/(ph/2),-1,1),ang=off*MAXA,silent=false;
  if(side===1){
    if(isCloak()&&player.cloakF>0.5){ang+=rand(-0.26,0.26);silent=true;}
    G.det=clamp(G.det+(Math.abs(off)>0.5?3+7*Math.abs(off):0),0,100);
  }else G.silentFlight=false;
  const sp=Math.min(Math.hypot(ball.vx,ball.vy)*B_GROW,B_CAP);
  ball.vx=Math.cos(ang)*sp*(side===1?1:-1);ball.vy=Math.sin(ang)*sp;
  ball.x=side===1?planeX+ball.r+0.5:planeX-ball.r-0.5;
  G.rally++;G.longest=Math.max(G.longest,G.rally);
  if(silent){G.silent++;G.silentFlight=true;ball.tint='#7df3ff';
    floater('SILENT STRIKE',planeX,ball.y-26,'#7df3ff');sSilent();burst(planeX,ball.y,'#7df3ff',8,150);}
  else if(side===1){ball.tint='#ffe9c4';sHitP(sp);burst(planeX,ball.y,'#ffb347',9,190);G.shake=Math.max(G.shake,2.5);}
  else{ball.tint='#eaf6ff';sHitW(sp);burst(planeX,ball.y,'#ff8d9b',9,190);G.shake=Math.max(G.shake,2);}
}
function moveBall(dt){
  let rem=Math.hypot(ball.vx,ball.vy)*dt;
  while(rem>0){
    const sp=Math.hypot(ball.vx,ball.vy),s=Math.min(6,rem);rem-=s;
    ball.x+=ball.vx/sp*s;ball.y+=ball.vy/sp*s;
    if(ball.y<ball.r){ball.y=2*ball.r-ball.y;ball.vy=-ball.vy;sWall();burst(ball.x,4,'#9db8e0',4,120);}
    else if(ball.y>H-ball.r){ball.y=2*(H-ball.r)-ball.y;ball.vy=-ball.vy;sWall();burst(ball.x,H-4,'#9db8e0',4,120);}
    const plX=player.x+player.w/2;
    if(ball.vx<0&&ball.x-ball.r<=plX&&ball.x-ball.r>plX-16){
      const ph=player.h*(1-player.cloakF*0.12);
      if(Math.abs(ball.y-player.y)<=ph/2+ball.r)hitPaddle(1,plX,ph);
    }
    const wrX=warden.x-warden.w/2;
    if(ball.vx>0&&ball.x+ball.r>=wrX&&ball.x+ball.r<wrX+16){
      if(Math.abs(ball.y-warden.y)<=warden.h/2+ball.r)hitPaddle(-1,wrX,warden.h);
    }
    if(ball.x<-30){pointScored(1);return;}
    if(ball.x>W+30){pointScored(0);return;}
  }
  ball.trail.push({x:ball.x,y:ball.y});
  if(ball.trail.length>15)ball.trail.shift();
}

/* ============ update ============ */
function update(dt){
  if(G.paused)return;
  G.t+=dt;
  G.shake=Math.max(0,G.shake-dt*26);

  for(const p of parts){p.t+=dt;p.x+=p.vx*dt;p.y+=p.vy*dt;p.vx*=(1-3*dt);p.vy*=(1-3*dt);}
  parts=parts.filter(p=>p.t<p.life);
  for(const f of floats)f.t+=dt;
  floats=floats.filter(f=>f.t<f.life);

  if(G.alarm>0){
    G.alarm-=dt;
    if(G.alarm<=0){G.alarm=0;document.body.classList.remove('alarm');
      if(G.failPending&&G.mode==='play')endGame('id');else G.det=45;}
  }

  const cloaked=!G.demo&&isCloak();
  player.cloakF=lerp(player.cloakF,cloaked?1:0,Math.min(1,dt*9));
  if(!G.demo){if(cloaked!==prevCloak){cloaked?sCloakOn():sCloakOff();prevCloak=cloaked;}}

  // player control
  const maxSp=cloaked?CLOAK_SPEED:P_SPEED;
  if(G.demo){aiMove(player,ball.y+Math.sin(G.t*2.4)*50,430,dt);player.vy=0;}
  else{
    let dir=0;if(keys.up)dir--;if(keys.down)dir++;
    if(touchTarget!=null){const d=touchTarget-player.y;dir=Math.abs(d)<8?0:Math.sign(d);}
    player.vy=dir*maxSp;player.y+=player.vy*dt;clampPaddle(player);
  }
  if(player.cloakF>0.5&&Math.random()<dt*18)player.ghosts.push({y:player.y,a:.26});
  for(const g of player.ghosts)g.a-=dt*.9;
  player.ghosts=player.ghosts.filter(g=>g.a>0);

  // detection & alarm
  if(!G.demo&&G.alarm<=0){
    const sp2=Math.abs(player.vy);
    if(cloaked)G.det-=30*dt;
    else if(sp2<maxSp*0.35)G.det-=14*dt;
    else G.det+=30*Math.pow(sp2/maxSp,2)*dt;
    G.det=clamp(G.det,0,100);
    if(G.detPrev<50&&G.det>=50)sTick();
    if(G.detPrev<80&&G.det>=80){sTick();sTick();}
    G.detPrev=G.det;
    if(G.det>=100)triggerAlarm();
  }

  // warden AI
  let target,sp3;
  if(G.demo){target=predict();sp3=460;}
  else if(G.alarm>0||ball.vx>0&&G.sub==='rally'){target=predict();sp3=W_ALARM;}
  else if(G.sub==='serve'){target=H/2+Math.sin(G.t*1.4)*26;sp3=Math.min(W_BASE+(G.score[0]+G.score[1])*W_RAMP,W_MAX);}
  else{target=H/2+(ball.y-H/2)*0.18;sp3=Math.min(W_BASE+(G.score[0]+G.score[1])*W_RAMP,W_MAX);}
  if(G.alarm<=0)target+=Math.sin(G.t*6.3)*5;
  warden.y+=clamp(target-warden.y,-sp3*dt,sp3*dt);clampPaddle(warden);

  // ball
  if(G.sub==='serve'){
    G.serveT-=dt;ball.x=W/2;ball.y=H*0.5+Math.sin(G.t*3.4)*9;
    if(G.serveT<=0)launch();
  }else moveBall(dt);

  // ball direction flip → warden re-reads the shot
  const d=ball.vx>0?1:(ball.vx<0?-1:0);
  if(d!==G.ballDir){G.ballDir=d;
    if(d===1)warden.err=rand(-1,1)*(G.demo?46:errMag());}
}

/* ============ draw ============ */
function rr(x,y,w,h,r){ctx.beginPath();ctx.moveTo(x+r,y);
  ctx.arcTo(x+w,y,x+w,y+h,r);ctx.arcTo(x+w,y+h,x,y+h,r);
  ctx.arcTo(x,y+h,x,y,r);ctx.arcTo(x,y,x+w,y,r);ctx.closePath();}

function drawBG(){
  ctx.fillStyle=bgGrad;ctx.fillRect(0,0,W,H);
  ctx.lineWidth=1;ctx.strokeStyle='rgba(96,140,255,.05)';ctx.beginPath();
  for(let x=((W/2)%48)||0;x<W;x+=48){if(Math.abs(x-W/2)<3)continue;ctx.moveTo(x+.5,0);ctx.lineTo(x+.5,H);}
  for(let y=0;y<H;y+=48){ctx.moveTo(0,y+.5);ctx.lineTo(W,y+.5);}
  ctx.stroke();
  ctx.strokeStyle='rgba(96,140,255,.08)';ctx.beginPath();
  for(let x=((W/2)%240)||0;x<W;x+=240){ctx.moveTo(x+.5,0);ctx.lineTo(x+.5,H);}
  for(let y=0;y<H;y+=240){ctx.moveTo(0,y+.5);ctx.lineTo(W,y+.5);}
  ctx.stroke();
  ctx.strokeStyle='rgba(140,170,220,.16)';ctx.beginPath();
  for(let x=24;x<W-24;x+=64){ctx.moveTo(x+.5,0);ctx.lineTo(x+.5,8);ctx.moveTo(x+.5,H);ctx.lineTo(x+.5,H-8);}
  ctx.stroke();
}
function drawCenter(){
  const pulse=.10+.05*Math.sin(G.t*1.6);
  ctx.setLineDash([9,13]);ctx.lineDashOffset=-G.t*14;
  ctx.strokeStyle='rgba(150,185,235,'+pulse+')';ctx.lineWidth=2;
  ctx.beginPath();ctx.moveTo(W/2,10);ctx.lineTo(W/2,H-10);ctx.stroke();ctx.setLineDash([]);
  ctx.strokeStyle='rgba(150,185,235,.07)';ctx.lineWidth=1.5;
  ctx.beginPath();ctx.arc(W/2,H/2,Math.min(W,H)*0.16,0,7);ctx.stroke();
}
function drawSightLine(){
  if(G.demo)return;
  const lx=warden.x-warden.w/2,ly=warden.y;
  let a=(G.det/100)*0.32+(G.alarm>0?0.25:0);
  a*=.75+.25*Math.sin(G.t*19);
  if(a<=0.02)return;
  ctx.strokeStyle='rgba(255,80,95,'+a+')';ctx.setLineDash([3,9]);ctx.lineWidth=1.5;
  ctx.beginPath();ctx.moveTo(lx,ly);ctx.lineTo(player.x,player.y);ctx.stroke();ctx.setLineDash([]);
}
function drawBeams(){
  const lx=warden.x-warden.w/2,ly=warden.y;
  const base=Math.atan2(player.y-ly,player.x-lx);
  ctx.save();ctx.globalCompositeOperation='lighter';
  for(let i=-1;i<=1;i++){
    const ang=base+i*0.22+Math.sin(G.t*2.6+i)*0.06;
    const ex=lx+Math.cos(ang)*W,ey=ly+Math.sin(ang)*W;
    ctx.strokeStyle='rgba(255,60,80,.10)';ctx.lineWidth=7;
    ctx.beginPath();ctx.moveTo(lx,ly);ctx.lineTo(ex,ey);ctx.stroke();
    ctx.strokeStyle='rgba(255,90,105,.35)';ctx.lineWidth=1.5;
    ctx.beginPath();ctx.moveTo(lx,ly);ctx.lineTo(ex,ey);ctx.stroke();
  }
  ctx.restore();
}
function drawPlayer(){
  const cf=player.cloakF,ph=player.h*(1-cf*0.12),x=player.x-player.w/2,y=player.y-ph/2;
  for(const g of player.ghosts){ctx.fillStyle='rgba(255,179,71,'+g.a+')';rr(x,g.y-ph/2,player.w,ph,5);ctx.fill();}
  ctx.save();
  ctx.globalAlpha=1-cf*0.6;
  ctx.shadowColor='rgba(255,179,71,.9)';ctx.shadowBlur=18*(1-cf*.4);
  ctx.fillStyle='#ffb347';rr(x,y,player.w,ph,5);ctx.fill();
  ctx.shadowBlur=0;
  ctx.fillStyle='rgba(255,240,210,.5)';ctx.fillRect(x+player.w/2-1.5,y+6,3,ph-12);
  if(cf>0.15){ctx.globalAlpha=cf*.85;ctx.strokeStyle='#7df3ff';ctx.lineWidth=1.5;
    rr(x-2.5,y-2.5,player.w+5,ph+5,6);ctx.stroke();}
  ctx.restore();
}
function drawWarden(){
  const x=warden.x-warden.w/2,y=warden.y-warden.h/2,h=warden.h,w=warden.w;
  const alarm=G.alarm>0;
  ctx.save();

  ctx.shadowBlur=0;

  const lx=x+4,ly=warden.y;

  ctx.strokeStyle='rgba(255,120,130,.8)';ctx.lineWidth=1.2;ctx.beginPath();ctx.arc(lx,ly,5.5,0,7);ctx.stroke();
  const a=Math.atan2(ball.y-ly,ball.x-lx);
  ctx.fillStyle=alarm?'#ff2740':'#ff8d9b';
  if(alarm){ctx.shadowColor='#ff2740';ctx.shadowBlur=10;}
  ctx.beginPath();ctx.arc(lx+Math.cos(a)*2,ly+Math.sin(a)*2,2.4,0,7);ctx.fill();
  ctx.shadowBlur=0;
  if(alarm){ctx.strokeStyle='rgba(255,120,130,.6)';ctx.lineWidth=1.5;
    ctx.beginPath();ctx.arc(lx,ly,10+Math.sin(G.t*9)*1.5,G.t*3,G.t*3+4.2);ctx.stroke();}
  ctx.restore();
}
function drawBall(){
  for(let i=0;i<ball.trail.length;i++){const p=ball.trail[i],f=i/ball.trail.length;
    ctx.globalAlpha=f*0.25;ctx.fillStyle=ball.tint;
    ctx.beginPath();ctx.arc(p.x,p.y,Math.max(.5,ball.r*f*.9),0,7);ctx.fill();}
  ctx.globalAlpha=1;
  if(G.sub==='serve'){
    const k=1-clamp(G.serveT/1.3,0,1);
    ctx.strokeStyle='rgba(95,232,218,.7)';ctx.lineWidth=2;
    ctx.beginPath();ctx.arc(ball.x,ball.y,14,-Math.PI/2,-Math.PI/2+k*Math.PI*2);ctx.stroke();
  }
  ctx.save();ctx.shadowColor=ball.tint;ctx.shadowBlur=16;
  ctx.fillStyle='#fff';ctx.beginPath();ctx.arc(ball.x,ball.y,ball.r,0,7);ctx.fill();ctx.restore();
}
function drawReticle(){
  if(G.demo||G.alarm<=0)return;
  const R=player.h*0.72+18;
  ctx.save();ctx.translate(player.x,player.y);ctx.rotate(G.t*1.4);
  ctx.strokeStyle='rgba(255,77,94,.85)';ctx.lineWidth=2;
  for(let i=0;i<4;i++){ctx.beginPath();ctx.arc(0,0,R,i*Math.PI/2-0.32,i*Math.PI/2+0.32);ctx.stroke();}
  ctx.restore();
}
function drawParts(){
  ctx.save();ctx.globalCompositeOperation='lighter';
  for(const p of parts){const k=p.t/p.life;ctx.globalAlpha=(1-k)*.9;
    ctx.fillStyle=p.col;ctx.beginPath();ctx.arc(p.x,p.y,Math.max(.4,p.size*(1-k*.6)),0,7);ctx.fill();}
  ctx.restore();ctx.globalAlpha=1;
}
function drawFloats(){
  ctx.textAlign='center';
  for(const f of floats){const k=f.t/f.life;ctx.globalAlpha=1-k;
    ctx.font='700 '+f.size+'px ui-monospace,Consolas,monospace';
    ctx.fillStyle=f.col;ctx.fillText(f.txt,f.x,f.y-k*34);}
  ctx.globalAlpha=1;
}
function draw(){
  const sx=(Math.random()-.5)*G.shake,sy=(Math.random()-.5)*G.shake;
  ctx.save();ctx.translate(sx,sy);

  if(G.alarm>0)drawBeams();
