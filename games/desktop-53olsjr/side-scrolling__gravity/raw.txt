    const dx=b.x-ox,dy=b.y-oy,d2=dx*dx+dy*dy;
    if(d2<74*74){const d=Math.sqrt(d2)||1,o.spd=560*dt;
      o.x+=dx/d*o.spd;o.relY=clamp(o.relY+dy/d*o.spd/track.h,0.05,0.95);}
    if(d2<Math.pow(ballR+13,2)){o.taken=true;G.orbs.splice(i,1);
      G.chain++;G.orbsGot++;const mult=Math.min(5,1+Math.floor(G.chain/4)),pts=25*mult;
      G.orbPts+=pts;floater(ox,oy,'+'+pts+(mult>1?' ×'+mult:''),'95,242,255');
      burst(ox,oy,8,[[95,242,255],[255,255,255]],260,.5,3);sfx.orb(mult);}
  }
  // shield pickup
  for(let i=G.pickups.length-1;i>=0;i--){const p=G.pickups[i],px=p.x-G.camX,py=track.top+p.relY*track.h;
    if(px<-90){G.pickups.splice(i,1);continue;}
    const dx=b.x-px,dy=b.y-py;
    if(dx*dx+dy*dy<Math.pow(ballR+18,2)){b.shield=true;G.pickups.splice(i,1);
      burst(px,py,16,[[255,210,122],[255,255,255]],340,.7,3.5);ring(px,py,ballR,[255,210,122]);
      floater(px,py,'SHIELD','255,210,122');sfx.shield();}}

  ambient(dt,G.speed);
  G.beatAcc+=dt*(1+G.speed/700);
  if(G.beatAcc>=1){G.beatAcc-=1;kick((0.045+Math.min(0.08,G.speed/9000))*(G.state==='play'?1:0.5));}
  if(G.shake>0)G.shake=Math.max(0,G.shake-dt*60);
  if(G.flash>0)G.flash=Math.max(0,G.flash-dt*1.6);
  if(G.state==='play')hud();
}

function ambient(dt,scroll){
  for(const m of motes){m.x-=scroll*m.vf*dt*(m.streak?1.5:0.7);
    if(m.x<-(m.len||20)){m.x=W+30;m.y=track.top+Math.random()*track.h;}}
  for(let i=G.parts.length-1;i>=0;i--){const p=G.parts[i];p.life-=dt;
    if(p.life<=0){G.parts.splice(i,1);continue;}
    if(p.k==='r'){p.size+=p.vr*dt;continue;}
    p.x+=p.vx*dt;p.y+=p.vy*dt;p.vx*=(1-2.4*dt);p.vy*=(1-2.4*dt);if(p.rot!==undefined)p.rot+=p.vr*dt;}
  for(let i=G.floaters.length-1;i>=0;i--){const f=G.floaters[i];f.life-=dt;f.y-=42*dt;
    if(f.life<=0)G.floaters.splice(i,1);}
  G.planet.x-=(scroll*0.03+4)*dt;if(G.planet.x<-G.planet.r*2.4){G.planet.x=W+G.planet.r*2.4;G.planet.y=H*(0.15+Math.random()*0.2);}
}

function updateDeath(dt){
  G.deathT+=dt;const ws=dt*0.35;
  G.speed*=Math.max(0,1-2.2*dt);G.camX+=G.speed*ws;
  ambient(ws,G.speed);
  if(G.shake>0)G.shake=Math.max(0,G.shake-dt*40);
  if(G.flash>0)G.flash=Math.max(0,G.flash-dt*1.2);
  if(G.deathT>0.95){
    if(G.dieFromMenu){reset();G.state='menu';}
    else finalizeOver();
  }
}

}
