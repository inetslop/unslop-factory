/* ============================ UPDATE ============================ */
function update(dt){
  G.t+=dt;
  for(const p of PIPS){p.y-=p.sp*dt;p.x+=Math.sin(G.t*.6+p.ph)*12*dt;if(p.y<-50){p.y=H+50;p.x=rand(0,W);}}
  G.parts=G.parts.filter(p=>(p.t+=dt)<p.dur);
  for(const p of G.parts){p.x+=p.vx*dt;p.y+=p.vy*dt;p.vy+=p.g*dt;p.rot+=p.vr*dt;}
  G.texts=G.texts.filter(t=>(t.t+=dt)<t.dur);
  G.flashes=G.flashes.filter(f=>(f.t+=dt)<f.dur);
  if(G.banner){G.banner.t+=dt;if(G.banner.t>G.banner.dur)G.banner=null;}
  G.shake=Math.max(0,G.shake-dt*26);
  G.scorePulse=Math.max(0,G.scorePulse-dt*3);
  if(G.state!=='playing')return;
  const p=G.piece;if(!p)return;
  // DAS auto-repeat
  const L=!!G.keys.ArrowLeft,Rt=!!G.keys.ArrowRight;
  const dir=(Rt&&!L)?1:(L&&!Rt)?-1:0;
  if(dir!==G.dasDir){G.dasDir=dir;G.moveLock=0;G.repT=0;if(dir)doMove(dir,true);}
  else if(dir){if(G.moveLock<160)G.moveLock+=dt*1000;else{G.repT+=dt*1000;while(G.repT>=45){G.repT-=45;doMove(dir,false);}}}
  // gravity
  const soft=!!G.keys.ArrowDown;
  const iv=soft?Math.min(G.interval,42):G.interval;
  G.gravT+=dt*1000;
  while(G.gravT>=iv){G.gravT-=iv;if(!stepDown(soft))break;}
  // lock delay
  if(collideGrid(p.grid,p.x,p.y+1)){G.lockT+=dt*1000;if(G.lockT>=500)lockPiece();}
  else G.lockT=0;
}

window.addEventListener('keydown',e=>{
  const c=e.code;
  if(['ArrowLeft','ArrowRight','ArrowDown','ArrowUp','Space'].indexOf(c)>=0)e.preventDefault();
  initAudio();
  if(e.repeat)return;
  if(c==='KeyM'){toggleMute();return;}
  switch(G.state){
    case 'menu':if(c==='Enter'||c==='Space'||c==='KeyR')startGame();break;
    case 'over':if(c==='Enter'||c==='KeyR')startGame();break;
    case 'paused':
      if(c==='KeyP'||c==='Escape'||c==='Enter')G.state='playing';
      else if(c==='KeyR')startGame();break;
    case 'playing':
      if(c==='ArrowLeft'||c==='ArrowRight')G.keys[c]=true;
      else if(c==='ArrowDown')G.keys[c]=true;
      else if(c==='ArrowUp'||c==='KeyX')tryRotate(1,true);
      else if(c==='KeyZ')tryRotate(-1,true);
      else if(c==='Space')hardDrop();
      else if(c==='KeyC'||c==='KeyH')holdSwap();
      else if(c==='KeyP'||c==='Escape')G.state='paused';
      else if(c==='KeyR')startGame();break;
  }
});

function toDesign(e){const r=cv.getBoundingClientRect();return{x:(e.clientX-r.left)*W/r.width,y:(e.clientY-r.top)*H/r.height};}
function inRect(p,b){return p.x>=b.x&&p.x<=b.x+b.w&&p.y>=b.y&&p.y<=b.y+b.h;}
cv.addEventListener('pointerdown',e=>{
  initAudio();const pt=toDesign(e);
  if(G.state==='menu'){if(inRect(pt,BTN_MENU))startGame();}
  else if(G.state==='over'){if(inRect(pt,BTN_OVER))startGame();}
  else if(G.state==='paused'){G.state='playing';}
  else if(G.state==='playing'){
    if(pt.x<FX)doMove(-1,true);
    else if(pt.x>FX+FW)doMove(1,true);
    else if(pt.y<FY+FH*.45)tryRotate(1,true);
    else hardDrop();
  }
});

  const pt=toDesign(e);

  cv.style.cursor=G.hover?'pointer':'default';
});
