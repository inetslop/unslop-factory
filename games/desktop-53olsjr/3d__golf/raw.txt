function dPath(x,z,pts){
  let m=1e9;
  for(let i=0;i<pts.length-1;i++){
    const ax=pts[i][0],az=pts[i][1],bx=pts[i+1][0],bz=pts[i+1][1];
    const dx=bx-ax,dz=bz-az;
    const t=Math.max(0,Math.min(1,((x-ax)*dx+(z-az)*dz)/(dx*dx+dz*dz)));
    const px=ax+dx*t-x,pz=az+dz*t-z,d=px*px+pz*pz;
    if(d<m)m=d;
  }
  return Math.sqrt(m);
}
function baseRGB(t,stripe,sh){

  return [Math.min(255,r*m|0),Math.min(255,g*m|0),Math.min(255,b*m|0)];
}

function physStep(dt){
  const p=ball.p,v=ball.v;
  if(ball.sunk){
    p[0]+=(cup.x-p[0])*Math.min(1,dt*9);p[2]+=(cup.z-p[2])*Math.min(1,dt*9);
    p[1]=Math.max(-0.7,p[1]-dt*3.4);return;
  }
  v[1]-=G*dt;
  p[0]+=v[0]*dt;p[1]+=v[1]*dt;p[2]+=v[2]*dt;
  // trees
  for(const t of trees){
    const dx=p[0]-t.x,dz=p[2]-t.z,rr=1.35+t.s*0.4;
    if(p[1]<5.2*t.s&&dx*dx+dz*dz<rr*rr){
      const d=Math.sqrt(dx*dx+dz*dz)||1e-4,nx=dx/d,nz=dz/d;
      p[0]=t.x+nx*rr;p[2]=t.z+nz*rr;
      const dot=v[0]*nx+v[2]*nz;
      if(dot<0){v[0]-=1.6*dot*nx;v[2]-=1.6*dot*nz;v[0]*=0.55;v[2]*=0.55;sBounce(0.3);spawn(p[0],p[1],p[2],5,[47,107,58],4,0.6,12,true);}
    }
  }
  const t=surfAt(p[0],p[2]);
  if(p[1]<=R){
    if(t===4){waterEvent();return;}
    const imp=-v[1];
    v[1]=-v[1]*(t===3?0.12:0.52);
    const keep=t===3?0.42:0.8;v[0]*=keep;v[2]*=keep;p[1]=R;
    if(imp>7){sBounce(Math.min(0.6,imp*0.03));spawn(p[0],R,p[2],t===3?10:5,t===3?[233,212,161]:[110,180,95],4,0.5,12,true);
      if(marks.length>40)marks.shift();marks.push({x:p[0],z:p[2],life:6,max:6});}
    else v[1]=0;
    ball.rolling=true;
    if(t===3){const s=Math.hypot(v[0],v[2]);if(s>3){v[0]*=0.5;v[2]*=0.5;spawn(p[0],R,p[2],12,[233,212,161],5,0.7,10,true);}}
  }
  const grounded=p[1]<=R+0.02&&Math.abs(v[1])<0.05;
  if(grounded){
    const sp=Math.hypot(v[0],v[2]);
    if(sp>0){
      const fc=FRIC[t][0],fl=FRIC[t][1],dec=(fc+fl*sp)*dt;
      if(sp<=dec||sp<0.12){v[0]=0;v[2]=0;}else{const k2=1-dec/sp;v[0]*=k2;v[2]*=k2;}
    }
  }
  const hdx=p[0]-cup.x,hdz=p[2]-cup.z,hd=Math.sqrt(hdx*hdx+hdz*hdz);
  if(!ball.sunk&&p[1]<R+1.5){
    const spd=Math.hypot(v[0],v[1],v[2]);
    if(hd<0.85&&spd<8)startSink();
    else if(hd<1.6&&spd>=8){
      const nx=hdx/(hd||1e-4),nz=hdz/(hd||1e-4);
      v[0]=nx*spd*0.35+(Math.random()-.5)*2;v[2]=nz*spd*0.35+(Math.random()-.5)*2;
      v[1]=Math.max(v[1],1.5);ball.rolling=false;sBounce(0.3);
    }
  }
  if(p[0]<-Hd.W/2-8||p[0]>Hd.W/2+8||p[2]<-14||p[2]>Hd.L+16)oobEvent();
}

/* trajectory preview */
function computePreview(){
  prev.length=0;
  if(!charging)return;
  const spd=7+51*power;
  let px=ball.p[0],py=ball.p[1],pz=ball.p[2];
  let vx=Math.sin(aim)*spd*0.985,vy=spd*0.13,vz=Math.cos(aim)*spd*0.985;
  const dt=0.045;
  for(let i=0;i<260;i++){
    vy-=G*dt;px+=vx*dt;py+=vy*dt;pz+=vz*dt;
    if(i%3===0)prev.push([px,Math.max(py,0),pz]);
    if(py<=R){
      const t=surfAt(px,pz);
      if(t===4){prev.push([px,R,pz,true]);break;}
      vy=-vy*0.52;vx*=0.78;vz*=0.78;py=R;
      if(Math.abs(vy)<1.3){
        let spx=vx,spz=vz;
        for(let j=0;j<60&&Math.hypot(spx,spz)>0.4;j++){
          const s=Math.hypot(spx,spz),dec=(FRIC[t][0]+FRIC[t][1]*s)*dt;
          if(s<=dec)break;const k2=1-dec/s;spx*=k2;spz*=k2;px+=spx*dt;pz+=spz*dt;
        }
        prev.push([px,R,pz]);
      }
    }
  }
}

let acc=0;
function update(dt){
  time+=dt;
  // aim & orbit
  const L=(keys['a']||keys['ArrowLeft'])?1:0,Rt=(keys['d']||keys['ArrowRight'])?1:0;
  if(L||Rt)aim+=(L-Rt)*2.0*dt;
  if(keys['q'])cam.yaw-=2.2*dt;
  if(keys['e'])cam.yaw+=2.2*dt;
  // charge
  if(charging){
    chargeT+=dt;
    const x=(chargeT%2.3)/1.15;power=x<1?x:2-x;
    $('pfill').style.width=(power*100).toFixed(1)+'%';
    if(chargeT-lastTick>0.09){lastTick=chargeT;sTick(power);}
    computePreview();
  }else power=0;
  if(freeze>0)freeze-=dt;
  flagWob*=Math.exp(-1.5*dt);flagCoo-=dt;
  // physics
  if(state==='play'){
    acc+=dt;let n=0;
    while(acc>1/120&&n<5){physStep(1/120);acc-=1/120;n++;}
    if(ball.sunk){sinkT+=dt;if(sinkT>1.1)finishHole();}
    // flag chime
    const d=Math.hypot(ball.p[0]-cup.x,ball.p[2]-cup.z);
    if(!ball.sunk&&d<Hd.green.r+2.5&&!flagNear){flagNear=true;if(flagCoo<=0){sChime();flagCoo=2;}}
    if(d>Hd.green.r+4)flagNear=false;
  }
  if(state==='holeend'){endT+=dt;if(endT>2.6)nextHole();}
  // particles / marks / rings
  for(let i=parts.length-1;i>=0;i--){const q=parts[i];q.life-=dt;
    if(q.life<=0){parts.splice(i,1);continue;}
    q.vy-=q.g*dt;q.x+=q.vx*dt;q.y+=q.vy*dt;q.z+=q.vz*dt;if(q.y<0.05)q.y=0.05;}
  for(let i=marks.length-1;i>=0;i--){marks[i].life-=dt;if(marks[i].life<=0)marks.splice(i,1);}
  for(let i=rings.length-1;i>=0;i--){rings[i].life-=dt;if(rings[i].life<=0)rings.splice(i,1);}
  // clouds drift
  for(const c of clouds)c.a+=dt*0.006;
  // distance HUD
  const dd=Math.hypot(ball.p[0]-cup.x,ball.p[2]-cup.z);
  setTxt('distEl',(dd<10?dd.toFixed(1):Math.round(dd))+'m');
  updateCamera(dt);
}

        ctx.lineTo(T[0]-hw,y0);ctx.lineTo(T[0]+hw,y0);ctx.closePath();ctx.fill();
      }
    }
    else if(it.k===5){ /* ground mark */
      const m=it.o;if(!project(m.x,0,m.z,tmp3))continue;
      const a=m.life/m.max*0.32,r2=focal*0.4/tmp3[2];
      ctx.fillStyle='rgba(16,28,16,'+a.toFixed(3)+')';
      ctx.beginPath();ctx.ellipse(tmp3[0],tmp3[1],r2,r2*0.45,0,0,TAU);ctx.fill();
    }
    else if(it.k===11){ /* splash ring */
      const r=it.o;if(!project(r.x,0,r.z,tmp3))continue;
      const t2=1-r.life/r.max,rad=focal*(0.4+t2*2.6)/tmp3[2];
      ctx.strokeStyle=r.col+(r.life/r.max*0.7).toFixed(3)+')';ctx.lineWidth=2;
      ctx.beginPath();ctx.ellipse(tmp3[0],tmp3[1],rad,rad*0.42,0,0,TAU);ctx.stroke();
    }
    else if(it.k===4){ /* particle */
      const q=it.o;if(!project(q.x,q.y,q.z,tmp3))continue;
      ctx.globalAlpha=Math.max(0,q.life/q.max);
      ctx.fillStyle='rgb('+q.col[0]+','+q.col[1]+','+q.col[2]+')';
      ctx.beginPath();ctx.arc(tmp3[0],tmp3[1],Math.max(1,focal*q.r/tmp3[2]),0,TAU);ctx.fill();
      ctx.globalAlpha=1;
    }
    else if(it.k===8){ /* rest pulse */
      if(!project(ball.p[0],0,ball.p[2],tmp3))continue;
      const a=0.16+0.14*Math.sin(time*4),r2=focal*(0.9+0.12*Math.sin(time*4))/tmp3[2];
      ctx.strokeStyle='rgba(255,210,122,'+a.toFixed(3)+')';ctx.lineWidth=2;
      ctx.beginPath();ctx.ellipse(tmp3[0],tmp3[1],r2,r2*0.42,0,0,TAU);ctx.stroke();
    }
    else if(it.k===3){ /* ball */
      const p=ball.p;
      if(p[1]>-0.2){
        if(project(p[0],0,p[2],tmp3)){
          const h=Math.max(0,p[1]),sa=Math.max(0.06,0.42-h*0.02),sr=focal*(0.42+h*0.02)/tmp3[2];
          ctx.fillStyle='rgba(8,16,9,'+sa.toFixed(3)+')';
          ctx.beginPath();ctx.ellipse(tmp3[0],tmp3[1],sr,sr*0.42,0,0,TAU);ctx.fill();
        }
        if(project(p[0],p[1],p[2],tmp3)){
          const r2=focal*R/tmp3[2];
          ctx.drawImage(ballSpr,tmp3[0]-r2*1.15,tmp3[1]-r2*1.15,r2*2.3,r2*2.3);
        }
      }
    }
  }
  /* aim arrow + preview */
  if(state==='play'&&canShoot()){
    const len=4+power*14,dx=Math.sin(aim),dz=Math.cos(aim);
    let px=null,py=null;
    ctx.lineCap='round';
    for(let i=0;i<=8;i++){
      const t2=i/8;
      if(!project(ball.p[0]+dx*len*t2,0.25,ball.p[2]+dz*len*t2,tmp3))continue;
      if(px!==null){
        ctx.strokeStyle='rgba(255,'+((230-160*power)|0)+','+((180-140*power)|0)+',0.9)';
        ctx.lineWidth=4-3*t2;
        ctx.beginPath();ctx.moveTo(px,py);ctx.lineTo(tmp3[0],tmp3[1]);ctx.stroke();
      }
      px=tmp3[0];py=tmp3[1];
    }
    if(charging){
      for(let i=0;i<prev.length;i++){
        const pt=prev[i];if(!project(pt[0],pt[1],pt[2],tmp3))continue;
        const t2=i/prev.length;
        ctx.globalAlpha=(0.75*(1-t2)).toFixed(3);
        ctx.fillStyle='#fff3d0';
        ctx.beginPath();ctx.arc(tmp3[0],tmp3[1],2.6-1.6*t2,0,TAU);ctx.fill();
        if(pt[3]){ctx.globalAlpha=0.8;ctx.strokeStyle='#ff7a4d';ctx.lineWidth=2;
          ctx.beginPath();ctx.ellipse(tmp3[0],tmp3[1],focal*0.9/tmp3[2],focal*0.4/tmp3[2],0,0,TAU);ctx.stroke();}
      }
      ctx.globalAlpha=1;
    }
  }
}
function mixC(c,f){if(f<=0)return'rgb('+c[0]+','+c[1]+','+c[2]+')';
  return'rgb('+((c[0]+(FOGC[0]-c[0])*f)|0)+','+((c[1]+(FOGC[1]-c[1])*f)|0)+','+((c[2]+(FOGC[2]-c[2])*f)|0)+')';}

}
