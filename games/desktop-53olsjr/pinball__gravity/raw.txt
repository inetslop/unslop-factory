/* ============ physics ============ */
function hitSeg(s,e,rest){
  const dx=s.x2-s.x1, dy=s.y2-s.y1, L2=dx*dx+dy*dy||1;
  let t=((e.x-s.x1)*dx+(e.y-s.y1)*dy)/L2; t=t<0?0:t>1?1:t;
  const px=s.x1+dx*t, py=s.y1+dy*t;
  let nx=e.x-px, ny=e.y-py; const d2=nx*nx+ny*ny;
  if(d2>e.r*e.r)return 0;
  const d=Math.sqrt(d2)||1e-4; nx/=d; ny/=d;
  e.x+=nx*(e.r-d); e.y+=ny*(e.r-d);
  const vn=e.vx*nx+e.vy*ny;
  if(vn<0){ e.vx-=(1+rest)*vn*nx; e.vy-=(1+rest)*vn*ny; return -vn; }
  return 0;
}
function hitDome(){
  const b=ball; if(b.y>=dome.cy)return;
  let dx=b.x-dome.cx, dy=b.y-dome.cy; const d=Math.hypot(dx,dy)||1e-4;
  const lim=dome.r-BALL_R;
  if(d>lim){
    dx/=d; dy/=d; b.x=dome.cx+dx*lim; b.y=dome.cy+dy*lim;
    const vn=b.vx*dx+b.vy*dy;
    if(vn>0){ b.vx-=1.5*vn*dx; b.vy-=1.5*vn*dy; sWall(vn); }
  }
}
function circleHit(c,r,rest){
  const b=ball; let dx=b.x-c.x, dy=b.y-c.y; const d=Math.hypot(dx,dy)||1e-4;
  const lim=r+b.r;
  if(d<lim){
    dx/=d; dy/=d; b.x=c.x+dx*lim; b.y=c.y+dy*lim;
    const vn=b.vx*dx+b.vy*dy;
    if(vn<0){ b.vx-=(1+rest)*vn*dx; b.vy-=(1+rest)*vn*dy; sWall(-vn); }
  }
}
function hitFlip(f){
  const b=ball, dx=Math.cos(f.ang), dy=Math.sin(f.ang);
  let t=((b.x-f.px)*dx+(b.y-f.py)*dy); t=t<0?0:t>f.len?f.len:t;
  const cx=f.px+dx*t, cy=f.py+dy*t;
  let nx=b.x-cx, ny=b.y-cy; const d=Math.hypot(nx,ny)||1e-4;
  const lim=b.r+f.r;
  if(d<lim){
    nx/=d; ny/=d; b.x=cx+nx*lim; b.y=cy+ny*lim;
    const rx=cx-f.px, ry=cy-f.py, svx=f.om*(-ry), svy=f.om*rx;
    const rvx=b.vx-svx, rvy=b.vy-svy, vn=rvx*nx+rvy*ny;
    if(vn<0){ const j=-1.35*vn; b.vx+=j*nx; b.vy+=j*ny; }
  }
}
function hitBump(u){
  const b=ball; let dx=b.x-u.x, dy=b.y-u.y; const d=Math.hypot(dx,dy)||1e-4;
  const lim=u.r+b.r;
  if(d<lim){
    dx/=d; dy/=d; b.x=u.x+dx*lim; b.y=u.y+dy*lim;
    const vn=b.vx*dx+b.vy*dy;
    if(vn<0){ b.vx-=1.9*vn*dx; b.vy-=1.9*vn*dy; }
    let sp=Math.hypot(b.vx,b.vy);
    if(sp<820){ if(sp<1){b.vx=dx*820;b.vy=dy*820;} else {b.vx*=820/sp;b.vy*=820/sp;} }
    u.flash=1;
    if(u.cd<=0){ u.cd=.09; addScore(superT>0?200:100,u.x,u.y-u.r-6); sBumper(); sparks(b.x,b.y,'#ffd76a',6,300); shake=Math.max(shake,1.6); }
  }
}

let acc=0; const STEP=1/240;
function phys(dt){
  acc+=dt; let n=0;
  while(acc>=STEP&&n<32){ sub(STEP); acc-=STEP; n++; }
}
function sub(h){
  const b=ball;
  for(const f of flips){
    const tgt=(state==='playing'&&keys[f.key])?f.act:f.rest;
    const mv=clamp(tgt-f.ang,-26*h,26*h);
    f.ang+=mv; f.om=mv/h;
  }
  if(ballState==='plunger'){
    b.x=508; b.vx=0;
    b.y=946+charge*26-b.r;
    if(state==='playing'&&keys.up){ b.vy-=AG*h; if(b.vy<-40) ballState='live'; }
    else b.vy=0;
    return;
  }
  const py=b.y;
  const g=(state==='playing'&&keys.up)?-AG:G;
  b.vy+=g*h;
  for(const w of wells){
    let dx=w.x-b.x, dy=w.y-b.y; const d=Math.hypot(dx,dy);
    if(d<w.rF&&d>0.001){ const a=w.S*(1-d/w.rF)*h; b.vx+=dx/d*a; b.vy+=dy/d*a; }
    if(w.cd<=0&&d<w.core+6){
      const nx=d>0.001?dx/d:0, ny=d>0.001?dy/d:1;
      b.vx=-ny*w.spin*1250+nx*260; b.vy=nx*w.spin*1250+ny*260;
      w.cd=.45; onEject(w);
    }
  }
  b.x+=b.vx*h; b.y+=b.vy*h;
  hitDome();
  for(const s of walls){ const imp=hitSeg(s,b,0.5); if(imp>120)sWall(imp); }
  if(b.x>gate.x1&&b.x<gate.x2&&py<=gate.y&&b.y>gate.y&&b.vy>0){ b.y=gate.y; b.vy*=-0.5; sWall(300); }
  for(const p of pivots) circleHit(p,p.r,0.3);
  for(const s of slings){
    for(let i=0;i<3;i++){ const imp=hitSeg(s.segs[i],b,0.45); if(imp>120)sWall(imp); }
    if(s.cd<=0&&s.flash>0.9){ /* handled below via main-face hit */ }
  }
  for(const s of slings){
    const imp=hitSeg(s.segs[0],b,0.45);
    if(imp>140&&s.cd<=0){
      b.vx+=s.mn.x*820; b.vy+=s.mn.y*820;
      s.cd=.12; s.flash=1; addScore(superT>0?50:25,280,760,'#ff8a5c'); sSling(); sparks(b.x,b.y,'#ff8a5c',5,240);
    }
  }
  for(const u of bumps) hitBump(u);
  for(const t of targets){ if(!t.up)continue;
    const imp=hitSeg({x1:t.x,y1:t.y-19,x2:t.x,y2:t.y+19},b,0.3);
    if(imp>60) hitTarget(t);
  }
  for(const f of flips) hitFlip(f);
  if(state==='playing'){
    rollovers.forEach((r,i)=>{
      if(!r.lit&&Math.hypot(b.x-r.x,b.y-r.y)<20){
        r.lit=true; addScore(superT>0?100:50,r.x,r.y-14,'#ffd76a'); sLamp(i);
        if(rollovers.every(q=>q.lit)){ superT=15; toast.txt='SUPER ×2'; toast.t=1.4; sSuper(); }
      }
    });
  }
  if(b.x>492&&b.y>930){ ballState='plunger'; charge=0; charging=false; trail.length=0; atkT=0; }
  const sp=Math.hypot(b.vx,b.vy);
  if(sp>MAXV){ b.vx*=MAXV/sp; b.vy*=MAXV/sp; }
  if(b.y>H+40) onDrain();
}
function onDrain(){
  if(state==='start'){ resetBall(); return; }
  sDrain(); shake=8;
  rings.push({x:280,y:1000,r:6,vr:700,t:0,max:.5,col:'#ff4444'});
  balls--;
  if(balls<=0){ gameOver(); }
  else{ toast.txt=balls===1?'LAST BALL':'BALL LOST'; toast.t=1.4; resetBall(); }
}

/* ============ frame step ============ */
function step(dt){
  time+=dt; wallSndT-=dt;
  if(superT>0){ superT-=dt; if(superT<=0) rollovers.forEach(r=>r.lit=false); }
  if(tiltT>0)tiltT-=dt;
  if(toast.t>0)toast.t-=dt;
  shake=Math.max(0,shake-26*dt);
  for(const w of wells){
    w.x=w.ax+Math.cos(time*w.sp+w.ph)*w.or;
    w.y=w.ay+Math.sin(time*w.sp+w.ph)*w.or;
    w.cd=Math.max(0,w.cd-dt);
  }
  for(const u of bumps){ u.flash*=Math.exp(-8*dt); if(u.cd>0)u.cd-=dt; }
  for(const s of slings){ s.flash*=Math.exp(-9*dt); if(s.cd>0)s.cd-=dt; }
  if(targetResetT>0){ targetResetT-=dt; if(targetResetT<=0){ targets.forEach(t=>t.up=true); sSpinUp(); sparks(104,650,'#bfe9ff',8,200); } }
  if(state==='playing'&&charging){
    charge=clamp(charge+dt/0.9,0,1); tickT-=dt;
    while(tickT<=0){ sCharge(charge); tickT+=0.085; }
  }
  if(state==='start'&&ballState==='plunger'){ atkT+=dt; if(atkT>1.1) doLaunch(0.6+Math.random()*0.4); }
  if(state!=='gameover') phys(dt);
  for(const f of flips){
    const p=!!(state==='playing'&&keys[f.key]);
    if(p&&!f.pressed){ f.pressed=true; sFlip(); } else if(!p) f.pressed=false;
  }
  if(ballState==='live'&&(state==='playing'||state==='start')){
    trail.push({x:ball.x,y:ball.y}); if(trail.length>14)trail.shift();
  }
  for(let i=parts.length-1;i>=0;i--){ const p=parts[i]; p.t-=dt;
    if(p.type==='tgt'){ p.vy+=1200*dt; p.rot+=p.vr*dt; } else p.vy+=500*dt;
    p.x+=p.vx*dt; p.y+=p.vy*dt;
    if(p.t<=0||p.y>970)parts.splice(i,1);
  }
  for(let i=rings.length-1;i>=0;i--){ const r=rings[i]; r.t+=dt; r.r+=r.vr*dt; if(r.t>r.max)rings.splice(i,1); }
  for(let i=floats.length-1;i>=0;i--){ floats[i].t-=dt; if(floats[i].t<=0)floats.splice(i,1); }
}

  ctx.beginPath(); ctx.arc(w.x,w.y,w.rF,0,TAU); ctx.lineWidth=1.5; ctx.stroke();
  ctx.setLineDash([]);
  for(let i=0;i<3;i++){
    ctx.globalAlpha=.75-.2*i; ctx.lineWidth=2.6-i*.5;
    ctx.beginPath(); ctx.arc(w.x,w.y,24+i*9,rot+i*2.1,rot+i*2.1+1.7); ctx.stroke();

  const g=ctx.createRadialGradient(w.x,w.y,0,w.x,w.y,w.core+6);

  ctx.beginPath(); ctx.arc(w.x,w.y,w.core+6,0,TAU); ctx.fill();
  ctx.restore();
}

}
function draw(){
