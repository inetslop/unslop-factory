function nb(i){const n=B.n,x=i%n,y=(i/n)|0,r=[];
  for(let dy=-1;dy<=1;dy++)for(let dx=-1;dx<=1;dx++){if(!dx&&!dy)continue;

function placeMines(excl){
  const idx=[];for(let i=0;i<B.cells.length;i++)if(!excl.has(i))idx.push(i);
  shuffle(idx);
  for(let k=0;k<B.mineN;k++)B.cells[idx[k]].mine=true;
  B.cells.forEach((c,i)=>{if(!c.mine)c.adj=nb(i).reduce((s,j)=>s+(B.cells[j].mine?1:0),0);});
  const safe=B.cells.map((c,i)=>i).filter(i=>!B.cells[i].mine);

}

function reveal(i,auto){
  if(S.phase!=='play'||!B)return;
  const c=B.cells[i];if(c.rev||c.flg)return;
  if(!B.placed)placeMines(new Set([i,...nb(i)]));
  if(c.mine){hitMine(i);return;}
  // flood area (flags stay put but don't block propagation)
  const dist=new Map([[i,0]]),area=[i];let q=[i];
  while(q.length){const j=q.pop();for(const k of nb(j)){
    if(dist.has(k)||B.cells[k].rev)continue;dist.set(k,dist.get(j)+1);area.push(k);if(B.cells[k].adj===0)q.push(k);}}
  let gained=0,goldGain=0,count=0,maxD=1;
  area.forEach(j=>{const cc=B.cells[j];if(cc.rev||cc.flg)return;count++;maxD=Math.max(maxD,dist.get(j));
    const v=cellValue(cc);gained+=v;if(cc.gold)goldGain+=v;});
  if(!count){return;}
  B.revSafe+=count;S.combo+=count;S.run.cells+=count;
  S.run.bestCombo=Math.max(S.run.bestCombo,S.combo);
  S.coins+=gained;S.total+=gained;S.run.earned+=gained;S.lastReveal=performance.now();
  sfx.pop(S.combo);
  const r=cellEl(i).getBoundingClientRect(),cx=r.left+r.width/2,cy=r.top+r.height/2;
  if(goldGain>0){floater(cx,cy-14,'JACKPOT +'+fmt(goldGain),'#ffd76a',20);burst(cx,cy,'#ffb63d',14,3);}
  else floater(cx,cy-10,'+'+fmt(gained),S.combo>16?'#ff9f5c':'#eaf5ef',15);
  if(auto)sparkle(cx,cy,'#6fe3d8');

function drawFX(dt){
