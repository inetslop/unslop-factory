function updateFrog(dt){
  if(frog.invuln>0) frog.invuln-=dt;
  frog.landT+=dt;
  // blink
  if(frog.blinkT>0) frog.blinkT-=dt; else { frog.nextBlink-=dt; if(frog.nextBlink<=0){ frog.blinkT=0.13; frog.nextBlink=2+Math.random()*3; } }

  if(state==='dead'){
    deadT+=dt;
    if(deadT>1.05){ lives--; updateHUD(); if(lives<=0) gameOver(); else respawn(); }
    return;
  }
  if(state!=='playing'&&state!=='celebrate') return;
  if(state==='celebrate'){
    celebrateT-=dt;
    if(celebrateT<=0){
      goalsFilled=[false,false,false,false,false]; padPop={};
      speedMult=Math.min(2,1+totalGoals*0.07); resetFrog(); state='playing';
    }
    return;
  }
  // hop
  if(frog.hop){
    var h=frog.hop; h.t+=dt; var p=Math.min(1,h.t/h.dur);
    frog.x=h.fx+(frog.col-h.fx)*p; frog.y=h.fy+(frog.row-h.fy)*p;
    frog.z=0.24*Math.sin(Math.PI*p);
    if(p>=1){ frog.hop=null; frog.z=0; frog.landT=0; sfx.land(); puff(frog.x+0.5,frog.y+0.5,3); }
  } else if(buffer&&simT-buffer.t<0.28){ var b=buffer; buffer=null; doMove(b.dc,b.dr); }
  // jump
  if(frog.jump){
    var j=frog.jump; j.t+=dt; var q=Math.min(1,j.t/j.dur);
    frog.z=HJ*Math.sin(Math.PI*q);
    if(q>=1){ frog.jump=null; frog.z=0; frog.landT=0; sfx.land(); puff(frog.x+0.5,frog.y+0.5,4); }
  }

  var cx=frog.x+0.5, cy=frog.y+0.5;
  // road collisions (only when low enough)
  if(frog.z<HIT_Z&&frog.invuln<=0){
    for(var i=0;i<objects.length;i++){
      var o=objects[i], L=o.lane;
      if(L.type==='log'||L.type==='turtle') continue;
      if(Math.abs(cy-(L.row+0.5))>0.42) continue;
      if(cx>o.x-0.26&&cx<o.x+L.len+0.26){ die('hit'); return; }
    }
  }
  // river riding / drowning
  var r=Math.round(frog.y);
  if(r>=1&&r<=4){
    var carried=null;
    for(var k=0;k<objects.length;k++){
      var o2=objects[k], L2=o2.lane;
      if(L2.row!==r) continue;
      if(cx>o2.x-0.3&&cx<o2.x+L2.len+0.3){
        if(L2.type==='turtle'&&L2.dive&&diveState(o2)>0.15) continue;
        carried=o2; break;
      }
    }
    if(carried){ frog.x+=carried.dx; }
    else { die('drown'); return; }
  }
  if(cx<-0.05||cx>COLS+0.05){ die('drown'); return; }

  // timer
  timeLeft-=dt;
  var sec=Math.ceil(timeLeft);
  if(timeLeft<10&&sec!==lastTickSec){ lastTickSec=sec; sfx.tick(); elTimeChip.classList.add('low'); }
  if(timeLeft<=0){ timeLeft=0; die('time'); return; }

  checkGoal();
}

  var dx=t.clientX-tS.x, dy=t.clientY-tS.y, dt=performance.now()-tS.t; tS=null;
  if(Math.abs(dx)<24&&Math.abs(dy)<24&&dt<350){
    if(state==='menu'||state==='over') resetRun(); else doJump(); return;
  }
  var m=(Math.abs(dx)>Math.abs(dy))?(dx>0?[1,0]:[-1,0]):(dy>0?[0,1]:[0,-1]);
  tryMove(m[0],m[1]);
});

function rr(x,y,w,h,r){ r=Math.min(r,w/2,h/2); ctx.beginPath(); ctx.moveTo(x+r,y); ctx.arcTo(x+w,y,x+w,y+h,r); ctx.arcTo(x+w,y+h,x,y+h,r); ctx.arcTo(x,y+h,x,y,r); ctx.arcTo(x,y,x+w,y,r); ctx.closePath(); }
function el(x,y,rx,ry){ ctx.beginPath(); ctx.ellipse(x,y,rx,ry,0,0,TAU); }

/* precomputed scenery */
var stars=[],clouds=[],flies=[],speckles=[],patches=[],flowers=[];
(function(){
  var i,r=Math.random;
  for(i=0;i<26;i++) stars.push({x:r(),y:r()*0.4,s:0.6+r()*1.6,ph:r()*TAU});
  for(i=0;i<6;i++) clouds.push({x:r(),y:0.05+r()*0.3,s:0.7+r()*1.2,v:0.004+r()*0.008,a:0.05+r()*0.07});
  for(i=0;i<12;i++) flies.push({x:r(),y:0.3+r()*0.6,ph:r()*TAU,speed:0.4+r()*0.7,hue:Math.random()<0.5?52:95});
  for(i=0;i<70;i++) speckles.push({x:r()*COLS,y:6+r()*4,a:0.04+r()*0.08,s:1+r()*1.6,c:r()<0.5?'#fff':'#000'});
  var gRows=[0,5,10];
  for(i=0;i<26;i++){ var gr=gRows[i%3]; patches.push({row:gr,x:r()*COLS,y:(r()-0.5)*0.5,r:0.25+r()*0.5,d:r()<0.5}); }
  for(i=0;i<18;i++){ var g2=gRows[i%3]; flowers.push({row:g2,x:r()*COLS,y:(r()-0.5)*0.6,c:['#ffe9a0','#ffb3c7','#ffffff'][i%3]}); }
})();

function drawSky(){
  var g=ctx.createLinearGradient(0,0,0,H);
  g.addColorStop(0,'#1b1240'); g.addColorStop(0.45,'#3d1f5e'); g.addColorStop(0.78,'#8a3a56'); g.addColorStop(1,'#c96b3f');
  ctx.fillStyle=g; ctx.fillRect(0,0,W,H);
  // stars
  for(var i=0;i<stars.length;i++){ var s=stars[i];
    ctx.globalAlpha=0.3+0.5*Math.abs(Math.sin(simT*1.4+s.ph));
    ctx.fillStyle='#fff'; ctx.fillRect(s.x*W,s.y*H,s.s,s.s); }
  ctx.globalAlpha=1;
  // sun
  var sx=W*0.82, sy=H*0.68;
  var sg=ctx.createRadialGradient(sx,sy,4,sx,sy,90);
  sg.addColorStop(0,'rgba(255,214,140,.9)'); sg.addColorStop(0.35,'rgba(255,170,90,.35)'); sg.addColorStop(1,'rgba(255,150,80,0)');
  ctx.fillStyle=sg; ctx.beginPath(); ctx.arc(sx,sy,90,0,TAU); ctx.fill();
  // clouds
  for(var c2=0;c2<clouds.length;c2++){ var cl=clouds[c2];
    cl.x+=cl.v*0.016; if(cl.x>1.25) cl.x=-0.3;
    ctx.globalAlpha=cl.a; ctx.fillStyle='#f8d9c4';
    el(cl.x*W,cl.y*H,60*cl.s,16*cl.s); ctx.fill();
    el(cl.x*W+38*cl.s,cl.y*H-10*cl.s,36*cl.s,14*cl.s); ctx.fill(); }
  ctx.globalAlpha=1;
  // distant hills
  ctx.fillStyle='rgba(22,12,38,.85)';
  ctx.beginPath(); ctx.moveTo(0,H);
  for(var x=0;x<=W;x+=40) ctx.lineTo(x,H-26-Math.sin(x*0.011)*16-Math.sin(x*0.037)*8);
  ctx.lineTo(W,H); ctx.closePath(); ctx.fill();
}
function drawFlies(){
  for(var i=0;i<flies.length;i++){ var f=flies[i];
    var x=(f.x+Math.sin(simT*f.speed+f.ph)*0.035)*W, y=(f.y+Math.cos(simT*f.speed*0.8+f.ph)*0.03)*H;
    var a=0.25+0.55*Math.abs(Math.sin(simT*2.1+f.ph));
    ctx.globalAlpha=a*0.4; ctx.fillStyle='hsla('+f.hue+',90%,65%,1)'; el(x,y,7,7); ctx.fill();
    ctx.globalAlpha=a; el(x,y,2.1,2.1); ctx.fill(); }
  ctx.globalAlpha=1;
}

function drawWater(){
  var y0=OY+C, h=C*4;
  var g=ctx.createLinearGradient(0,y0,0,y0+h);
  g.addColorStop(0,'#1e5f86'); g.addColorStop(1,'#123a5c');
  ctx.fillStyle=g; ctx.fillRect(OX,y0,COLS*C,h);
  for(var r=1;r<=4;r++){
    var L=LANES[r-1], dir=L.dir, yC=OY+(r+0.5)*C, off=(simT*L.speed*dir)%2.6;
    ctx.strokeStyle='rgba(255,255,255,.13)'; ctx.lineWidth=2; ctx.lineCap='round';
    for(var k=-1;k<7;k++){
      var x=OX+((k*2.6-off)% (COLS+2)+ (COLS+2))*(C/(COLS+2));
      if(x<OX-20||x>OX+COLS*C+20) continue;
      ctx.beginPath();
      ctx.moveTo(x,yC+Math.sin(simT*3+k)*2-C*0.16);
      ctx.quadraticCurveTo(x+C*0.35,yC+Math.sin(simT*3+k+1)*2, x+C*0.7,yC+Math.sin(simT*3+k)*2+C*0.16);
      ctx.stroke();
    }
  }
}
function drawGrassRow(row){
  var y=OY+row*C;
  var g=ctx.createLinearGradient(0,y,0,y+C);
  g.addColorStop(0,'#4f9e3c'); g.addColorStop(1,'#3d7d2e');
  ctx.fillStyle=g; ctx.fillRect(OX,y,COLS*C,C);
  for(var i=0;i<patches.length;i++){ var p=patches[i]; if(p.row!==row) continue;
    ctx.globalAlpha=0.28; ctx.fillStyle=p.d?'#2c5f22':'#63b44a';
    el(OX+p.x*C,y+(0.5+p.y)*C,p.r*C,p.r*C*0.6); ctx.fill(); }
  ctx.globalAlpha=1;
  for(var f=0;f<flowers.length;f++){ var fl=flowers[f]; if(fl.row!==row) continue;
    ctx.fillStyle=fl.c; el(OX+fl.x*C,y+(0.5+fl.y)*C,2.4,2.4); ctx.fill(); }
  // grass blades at edges
  ctx.strokeStyle='rgba(30,70,25,.6)'; ctx.lineWidth=1.5;
  for(var b=0;b<COLS*3;b++){ var bx=OX+((b*0.37)%1)*COLS*C;
    var top=(row!==ROWS-1)?y:y+C, dirr=(row!==ROWS-1)?-1:1;
    ctx.beginPath(); ctx.moveTo(bx,top); ctx.lineTo(bx+2*dirr*Math.sin(simT+b),top+4.5*dirr); ctx.stroke(); }
}

function drawTurtle(o){
  var L=o.lane, cx=OX+(o.x+L.len/2)*C, cy=OY+(L.row+0.5)*C, s=C*0.3;
  var sink=L.dive?diveState(o):0;
  if(sink>0.86){ // fully under: ripple ring
    ctx.strokeStyle='rgba(210,240,255,'+(0.5*(1-sink/1))+')'; ctx.lineWidth=2;
    ctx.beginPath(); ctx.arc(cx,cy,s*(1+sink),0,TAU); ctx.stroke(); return;
  }
  var sc=1-sink*0.85, al=1-sink*0.8;
  var shell=L.tc===0?'#4f9e46':'#c8524a', patt=L.tc===0?'#3a7d33':'#96392f', skin=L.tc===0?'#cfe08a':'#e6bfa4';
  ctx.save(); ctx.translate(cx,cy); ctx.scale(sc,sc); ctx.globalAlpha=al;
  ctx.globalAlpha=al*0.3; ctx.fillStyle='#04121e'; el(2,3,s*1.25,s*1.0); ctx.fill(); ctx.globalAlpha=al;
  // flippers
  ctx.fillStyle=skin;
  for(var i=0;i<4;i++){ var ang=(L.dir>0?0:Math.PI)+ (i%2? 0.9:-0.9) + Math.sin(simT*6+i*1.7)*0.25;
    el(Math.cos(ang)*s*1.15,Math.sin(ang)*s*0.8,s*0.42,s*0.26); ctx.fill(); }
  // head
  el(L.dir*s*1.35,0,s*0.5,s*0.42); ctx.fill();
  ctx.fillStyle='#222'; el(L.dir*s*1.5,-s*0.14,s*0.09,s*0.09); ctx.fill(); el(L.dir*s*1.5,s*0.14,s*0.09,s*0.09); ctx.fill();
  // shell
  ctx.fillStyle=shell; el(0,0,s*1.25,s*1.02); ctx.fill();
  ctx.strokeStyle=patt; ctx.lineWidth=2; el(0,0,s*1.25,s*1.02); ctx.stroke();
  ctx.globalAlpha=al*0.5; ctx.fillStyle=patt;
  el(0,0,s*0.4,s*0.34); ctx.fill();
  for(var k=0;k<6;k++){ var a=k/6*TAU+simT*L.dir*0.15; el(Math.cos(a)*s*0.72,Math.sin(a)*s*0.55,s*0.28,s*0.24); ctx.fill(); }
  ctx.restore(); ctx.globalAlpha=1;
}
function drawVehicle(o){
  var L=o.lane, x=OX+o.x*C, cy=OY+(L.row+0.5)*C, w=L.len*C*0.94, h=C*0.62, d=L.dir;
  ctx.globalAlpha=0.28; ctx.fillStyle='#000'; rr(x-w/2+3,cy-h/2+4,w,h,C*0.14); ctx.fill(); ctx.globalAlpha=1;
  var col;
  if(L.type==='taxi') col='#f2d76e';
  else if(L.type==='truck') col=CARC[o.vk%3];
  else col=CARC[o.vk%CARC.length];
  ctx.fillStyle=col; rr(x-w/2,cy-h/2,w,h,C*0.14); ctx.fill();
  ctx.strokeStyle='rgba(0,0,0,.4)'; ctx.lineWidth=2; rr(x-w/2,cy-h/2,w,h,C*0.14); ctx.stroke();
  if(L.type==='truck'){
    var cabW=w*0.3, bedX=d>0?x-w/2+cabW:x+w/2-cabW-w*0.68;

    ctx.strokeStyle='rgba(0,0,0,.3)'; ctx.lineWidth=1.5;

  var lk=Math.max(0,1-frog.landT/0.16), sy2=(1-0.25*lk)*sc, sx2=(1+0.22*lk)*sc;
  if(frog.invuln>0&&state==='playing') ctx.globalAlpha=Math.sin(simT*28)>0?1:0.4;
  var rot=frog.jump? Math.sin(Math.PI*Math.min(1,frog.jump.t/frog.jump.dur))*-frog.facing*0.22 : 0;
  ctx.save(); ctx.translate(cxp,gyp-lift+((z<0.01&&!frog.hop)?Math.sin(simT*3)*C*0.02:0));
  ctx.scale(sx2,sy2); ctx.rotate(rot);
  var R=C*0.34, dead=(state==='dead');
  var body=dead?'#8fa08f':'#6cc24f', dark=dead?'#5c6b5c':'#2f7d2e', belly=dead?'#b9c4b4':'#bfe88a';
  // feet
  ctx.fillStyle=dead?'#77877a':'#4a9e3c';
  el(-R*0.78,R*0.95,R*0.34,R*0.2); ctx.fill(); el(R*0.78,R*0.95,R*0.34,R*0.2); ctx.fill();
  // body
  var bg=ctx.createLinearGradient(0,-R,0,R); bg.addColorStop(0,body); bg.addColorStop(1,dead?'#6b7a6c':'#4a9e3c');
  ctx.fillStyle=bg; el(0,R*0.12,R*1.06,R*0.98); ctx.fill();
  ctx.strokeStyle=dark; ctx.lineWidth=Math.max(2,C*0.05); el(0,R*0.12,R*1.06,R*0.98); ctx.stroke();
  ctx.fillStyle=belly; el(0,R*0.45,R*0.58,R*0.42); ctx.fill();
  // back spots
  if(!dead){ ctx.fillStyle='rgba(31,90,30,.5)'; el(-R*0.3,-R*0.15,R*0.13,R*0.1); ctx.fill(); el(R*0.32,-R*0.08,R*0.11,R*0.09); ctx.fill(); }
  // eyes
  var ex=R*0.56, ey=-R*0.72, blink=frog.blinkT>0;
  [-1,1].forEach(function(s){
    ctx.fillStyle=body; ctx.beginPath(); ctx.arc(s*ex,ey,R*0.44,0,TAU); ctx.fill();
    ctx.strokeStyle=dark; ctx.lineWidth=Math.max(1.5,C*0.035); ctx.stroke();
    if(dead){
      ctx.strokeStyle='#333'; ctx.lineWidth=2.5;
      ctx.beginPath(); ctx.moveTo(s*ex-R*0.16,ey-R*0.16); ctx.lineTo(s*ex+R*0.16,ey+R*0.16);
      ctx.moveTo(s*ex+R*0.16,ey-R*0.16); ctx.lineTo(s*ex-R*0.16,ey+R*0.16); ctx.stroke();
    } else if(blink){
      ctx.strokeStyle='#2f5d2c'; ctx.lineWidth=3;
      ctx.beginPath(); ctx.moveTo(s*ex-R*0.22,ey); ctx.lineTo(s*ex+R*0.22,ey); ctx.stroke();
    } else {
      ctx.fillStyle='#fff'; el(s*ex,ey,R*0.3,R*0.3); ctx.fill();
      var px=s*ex+frog.facing*R*0.12;
      ctx.fillStyle='#1c1c1c'; el(px,ey+R*0.04,R*0.15,R*0.15); ctx.fill();
      ctx.fillStyle='#fff'; el(px-R*0.05,ey-R*0.03,R*0.05,R*0.05); ctx.fill();
    }
  });
  // mouth
  if(dead){ ctx.fillStyle='#e88'; el(0,R*0.72,R*0.16,R*0.2); ctx.fill(); }
  else { ctx.strokeStyle='rgba(30,70,25,.8)'; ctx.lineWidth=2;
    ctx.beginPath(); ctx.arc(0,R*0.42,R*0.34,Math.PI*0.25,Math.PI*0.75); ctx.stroke(); }
  ctx.restore(); ctx.globalAlpha=1;
}

function drawParts(){
  for(var i=0;i<parts.length;i++){ var p=parts[i], k=p.t/p.life, sx=OX+p.x*C, sy=OY+p.y*C;
    if(p.type==='dust'){ ctx.fillStyle=p.c+(0.4*(1-k))+')'; el(sx,sy,p.s*C*(1+k*2),p.s*C*(1+k*2)); ctx.fill(); }
    else if(p.type==='dot'){ ctx.globalAlpha=1-k; ctx.fillStyle=p.c; el(sx,sy,p.s*C,p.s*C); ctx.fill(); ctx.globalAlpha=1; }
    else if(p.type==='ring'){ ctx.strokeStyle=p.c; ctx.globalAlpha=1-k; ctx.lineWidth=3; ctx.beginPath(); ctx.arc(sx,sy,(p.s+k*1.4)*C,0,TAU); ctx.stroke(); ctx.globalAlpha=1; }
    else if(p.type==='conf'){ ctx.save(); ctx.translate(sx,sy); ctx.rotate(p.rot); ctx.globalAlpha=1-k*k; ctx.fillStyle=p.c; ctx.fillRect(-p.s*C,-p.s*C*0.5,p.s*C*2,p.s*C); ctx.restore(); ctx.globalAlpha=1; }
  }
}

function render(){
  ctx.setTransform(DPR,0,0,DPR,0,0);
  drawSky();
  // shake
  var shx=0,shy=0;
  if(shakeT>0){ shakeT-=0.016; shx=(Math.random()-0.5)*shakeMag*(shakeT/0.35); shy=(Math.random()-0.5)*shakeMag*(shakeT/0.35); }
  ctx.save(); ctx.translate(shx,shy);
  // field shadow + frame
  ctx.fillStyle='rgba(0,0,0,.4)'; rr(OX+6,OY+8,COLS*C,ROWS*C,10); ctx.fill();
  drawWater();
  [0,5,10].forEach(drawGrassRow);
  drawRoad();
  // objects by row
  for(var r=1;r<=9;r++){
    for(var i=0;i<objects.length;i++){ var o=objects[i]; if(o.lane.row!==r) continue;
      if(o.lane.type==='log') drawLog(o); else if(o.lane.type==='turtle') drawTurtle(o); else drawVehicle(o); }
  }
  drawPads(); drawStartPad();
  if(frogVisible) drawFrog();
  drawParts();
  ctx.restore();
  // frame stroke
  ctx.strokeStyle='rgba(255,214,120,.28)'; ctx.lineWidth=2; rr(OX-3,OY-3,COLS*C+6,ROWS*C+6,12); ctx.stroke();
  drawFlies();
  if(hitFlash>0){ hitFlash=Math.max(0,hitFlash-0.04); ctx.fillStyle='rgba(255,70,50,'+(hitFlash*0.3)+')'; ctx.fillRect(0,0,W,H); }
  ctx.fillStyle=VIG; ctx.fillRect(0,0,W,H);
}

/* ============================== HUD LOOP BITS ============================== */
function hudTick(){
  var f=Math.max(0,timeLeft/TIME_MAX);
  elTimeFill.style.width=(f*100)+'%';
  var hue=f*105;
  elTimeFill.style.background='hsl('+hue+',72%,55%)';
}

/* ============================== MAIN LOOP ============================== */
var last=performance.now();
function frame(now){
  requestAnimationFrame(frame);
  var dt=Math.min(0.05,(now-last)/1000); last=now;
  if(state!=='paused'){ simT+=dt; tWorld+=dt; }
  if(state!=='paused') updateWorld(dt);
  updateFrog(dt);
  if(state!=='paused') updateParts(dt);
  // crickets
  nextChirp-=dt;
  if(nextChirp<=0&&AC){
    for(var i=0;i<3;i++) tone('sine',4200+Math.random()*300,4400,0.035,0.011,i*0.09);
    nextChirp=2.5+Math.random()*4;
  }
  if(state==='playing') hudTick();
  render();
}

layout(); buildWorld(); resetFrog(); frogVisible=false;
showMenu(); updateHUD(); elTimeFill.style.width='100%';
requestAnimationFrame(frame);
})();
