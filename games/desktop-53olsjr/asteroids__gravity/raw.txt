function fieldAt(x,y){ let ax=0,ay=0;
  for(let i=0;i<wells.length;i++){ const w=wells[i];
    const dx=wrapDelta(x,w.x,W), dy=wrapDelta(y,w.y,H);

    let a=GK*w.mass/d2; if(a>WELL_CAP)a=WELL_CAP;
    ax+=a*dx*inv; ay+=a*dy*inv; }
  return [ax,ay];
}

  while(!ok&&tries++<40){ x=rand(m,W-m); y=rand(m,H-m); ok=true;
    for(const w of wells){ if(Math.hypot(wrapDelta(x,w.x,W),wrapDelta(y,w.y,H))<260)ok=false; } }

function updWorld(dt){
  for(const w of wells){ w.x=(w.x+w.vx*dt+W)%W; w.y=(w.y+w.vy*dt+H)%H; }
  for(let i=asteroids.length-1;i>=0;i--){ const a=asteroids[i];
    const f=fieldAt(a.x,a.y), fac=[0,.9,.55,.3][a.size];
    a.vx+=f[0]*fac*dt; a.vy+=f[1]*fac*dt;
    const dmp=Math.exp(-.15*dt); a.vx*=dmp; a.vy*=dmp;
    const sp=Math.hypot(a.vx,a.vy); if(sp>300){const k=300/sp;a.vx*=k;a.vy*=k;}
    a.x=(a.x+a.vx*dt+W)%W; a.y=(a.y+a.vy*dt+H)%W; a.rot+=a.vr*dt;
    let eaten=false;
    for(const w of wells){ const dx=wrapDelta(a.x,w.x,W),dy=wrapDelta(a.y,w.y,H);
      if(dx*dx+dy*dy<(w.r+a.r*.5)*(w.r+a.r*.5)){ explode(a.x,a.y,1,['#bdf3ff','#fff']); eaten=true; break; } }
    if(eaten){ asteroids.splice(i,1);
      if(G.state==='playing')addScore(10,a.x,a.y,'CONSUMED +10','#9fb8d8'); } }
  for(let i=parts.length-1;i>=0;i--){ const p=parts[i]; p.t-=dt;
    if(p.t<=0){parts.splice(i,1);continue;}
    if(p.k==='s'){ const d=Math.exp(-2.2*dt); p.vx*=d;p.vy*=d;p.x=(p.x+p.vx*dt+W)%W;p.y=(p.y+p.vy*dt+H)%H; }
    else if(p.k==='d'){ const d=Math.exp(-1.4*dt); p.vx*=d;p.vy*=d;p.rot+=p.vr*dt;
      p.x=(p.x+p.vx*dt+W)%W;p.y=(p.y+p.vy*dt+H)%H; } }
  for(let i=popups.length-1;i>=0;i--){ popups[i].t-=dt*.9; if(popups[i].t<=0)popups.splice(i,1); }
  G.shake*=Math.exp(-5*dt); if(G.shake<.1)G.shake=0;
}
function updShip(dt){ const s=ship; if(!s)return;
  if(s.alive){
    if(keys.left)s.ang-=ROT*dt; if(keys.right)s.ang+=ROT*dt;
    s.thrusting=!!keys.up;
    if(keys.up){ const ca=Math.cos(s.ang),sa=Math.sin(s.ang); s.vx+=ca*THRUST*dt; s.vy+=sa*THRUST*dt;
      if(Math.random()<dt*46){ const rx=s.x-ca*12,ry=s.y-sa*12,sp=rand(60,150),a=Math.atan2(-sa,-ca)+rand(-.5,.5);
        parts.push({k:'s',x:rx,y:ry,vx:Math.cos(a)*sp,vy:Math.sin(a)*sp,t:rand(.15,.35),mt:.35,col:'#ff9d4d',sz:rand(1,2.2)}); } }
    const f=fieldAt(s.x,s.y); s.vx+=f[0]*dt; s.vy+=f[1]*dt;
    const dmp=Math.exp(-DAMP*dt); s.vx*=dmp; s.vy*=dmp;
    const sp=Math.hypot(s.vx,s.vy); if(sp>MAXV){const k=MAXV/sp;s.vx*=k;s.vy*=k;}
    s.x=(s.x+s.vx*dt+W)%W; s.y=(s.y+s.vy*dt+H)%W;
    s.trail.push([s.x,s.y]); if(s.trail.length>16)s.trail.shift();
    s.fireCd-=dt;
    if(keys.fire&&s.fireCd<=0){ const rapid=s.rapidT>0; s.fireCd=rapid?.08:.17;
      if(bullets.length<5){ const ca=Math.cos(s.ang),sa=Math.sin(s.ang);
        bullets.push({x:s.x+ca*16,y:s.y+sa*16,vx:ca*560+s.vx*.4,vy:sa*560+s.vy*.4,t:1.25,r:rapid});
        SFX.laser(); sparkBurst(s.x+ca*18,s.y+sa*18,2,rapid?'#ffcf6b':'#bdf3ff'); } }
    if(s.inv>0)s.inv-=dt; if(s.rapidT>0)s.rapidT=Math.max(0,s.rapidT-dt);
  }else if(G.lives>0){ s.respawnT-=dt; if(s.respawnT<=0)spawnShip(); }
}
function combat(dt){ const s=ship; if(!s)return;
  for(let i=bullets.length-1;i>=0;i--){ const b=bullets[i]; b.t-=dt;
    if(b.t<=0){bullets.splice(i,1);continue;}
    const f=fieldAt(b.x,b.y); b.vx+=f[0]*.75*dt; b.vy+=f[1]*.75*dt;
    b.x=(b.x+b.vx*dt+W)%W; b.y=(b.y+b.vy*dt+H)%W;
    let dead=false;
    for(const w of wells){ const dx=wrapDelta(b.x,w.x,W),dy=wrapDelta(b.y,w.y,H);
      if(dx*dx+dy*dy<(w.r+3)*(w.r+3)){ sparkBurst(b.x,b.y,6,w.type==='star'?'#bdf3ff':'#d0a8ff'); dead=true; break; } }
    if(!dead)for(const a of asteroids){ const dx=wrapDelta(b.x,a.x,W),dy=wrapDelta(b.y,a.y,H);
      const rr=a.r*.82+3; if(dx*dx+dy*dy<rr*rr){ hitAsteroid(a,b); dead=true; break; } }
    if(dead)bullets.splice(i,1); }
  if(s.alive){ for(const w of wells){ const dx=wrapDelta(s.x,w.x,W),dy=wrapDelta(s.y,w.y,H);
      const d=Math.hypot(dx,dy);
      if(d<w.r+8){ killShip(); break; }
      if(!w.near&&d<w.r+62){ w.near=true; addScore(15,s.x,s.y-20,'ORBIT +15','#7fe9ff'); SFX.swoosh(); }
      else if(w.near&&d>w.r+100)w.near=false; } }
  if(s.alive&&s.inv<=0){ for(const a of asteroids){ const dx=wrapDelta(s.x,a.x,W),dy=wrapDelta(s.y,a.y,H);
      const rr=a.r*.82+9; if(dx*dx+dy*dy<rr*rr){
        if(s.shield){ s.shield=false; SFX.shieldHit(); sparkBurst(s.x,s.y,16,'#7fe9ff');
          parts.push({k:'r',x:s.x,y:s.y,r0:14,r1:52,t:.4,mt:.4,col:'125,255,176'});
          const i=asteroids.indexOf(a); if(i>=0)asteroids.splice(i,1);
          explode(a.x,a.y,1,null);
          const d=Math.hypot(dx,dy)||1; s.vx-=dx/d*260; s.vy-=dy/d*260; s.inv=1.2; }
        else killShip(); break; } } }
  if(s.alive){ for(let i=pickups.length-1;i>=0;i--){ const p=pickups[i]; p.t-=dt;
      if(p.t<=0){pickups.splice(i,1);continue;}
      const f=fieldAt(p.x,p.y); p.vx+=f[0]*.35*dt; p.vy+=f[1]*.35*dt;
      const sp=Math.hypot(p.vx,p.vy); if(sp>140){const k=140/sp;p.vx*=k;p.vy*=k;}
      p.x=(p.x+p.vx*dt+W)%W; p.y=(p.y+p.vy*dt+H)%H;
      const dx=wrapDelta(s.x,p.x,W),dy=wrapDelta(s.y,p.y,H);
      if(dx*dx+dy*dy<676){ applyPickup(p); pickups.splice(i,1); } } }
  if(asteroids.length===0&&!G.between){ G.between=true; G.betweenT=1.4;
    addScore(150,W/2,H*.3,'FIELD CLEAR +150','#9dffb0'); SFX.powerup(); }
  if(G.between){ G.betweenT-=dt; if(G.betweenT<=0){G.between=false; startWave(G.wave+1);} }
}
function update(dt){ updWorld(dt); updShip(dt); combat(dt);
  if(ship&&!ship.alive&&G.lives<=0){ G.overT-=dt; if(G.overT<=0)gameOver(); }
  SFX.thrust(ship&&ship.alive&&keys.up);
  if(ship&&ship.alive){ camX+=ship.vx*dt; camY+=ship.vy*dt; }
}

/* ============================== PREDICTION ============================== */
function predict(){ const s=ship; const pts=[]; let impact=null;
  let x=s.x,y=s.y,vx=s.vx,vy=s.vy; const dt=1/60,dmp=Math.exp(-DAMP*dt);
  outer:for(let i=0;i<96;i++){
    const f=fieldAt(x,y); vx+=f[0]*dt; vy+=f[1]*dt; vx*=dmp; vy*=dmp;
    x=(x+vx*dt+W)%W; y=(y+vy*dt+H)%H;
    if(i%3===0)pts.push([x,y]);
    for(const w of wells){ const dx=wrapDelta(x,w.x,W),dy=wrapDelta(y,w.y,H);
      if(dx*dx+dy*dy<(w.r+8)*(w.r+8)){impact=[x,y];break outer;} }
    for(const a of asteroids){ const dx=wrapDelta(x,a.x,W),dy=wrapDelta(y,a.y,H);
      const rr=a.r*.82+8; if(dx*dx+dy*dy<rr*rr){impact=[x,y];break outer;} } }
  return {pts,impact};
}
