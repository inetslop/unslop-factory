function collide(ci,rot,x,y){
  const cells=ROTS[ci][rot];
  for(const c of cells){ const cx=x+c[0], cy=y+c[1];
    if(cx<0||cx>=COLS||cy>=ROWS) return true;
    if(cy>=0 && grid[cy][cx]) return true; }
  return false;
}
function spawnPiece(){
  while(G.queue.length<3){ if(!G.bag.length) G.bag=shuffle([0,1,2,3,4,5,6]); G.queue.push(G.bag.pop()); }
  const ci=G.queue.shift();
  const nv=G.forceNova || (G.pc%7===6);
  if(nv) G.forceNova=false;
  G.pc++;
  while(G.queue.length<3){ if(!G.bag.length) G.bag=shuffle([0,1,2,3,4,5,6]); G.queue.push(G.bag.pop()); }
  G.P={ci,nv,rot:0,x:3,y:ci===0?-1:-1,acc:0,lockT:0};
  if(collide(ci,0,3,-1)) skybreak();
}
function resetLock(){ if(G.P && collide(G.P.ci,G.P.rot,G.P.x,G.P.y+1)) G.P.lockT=0; }
function tryMove(dx){ const P=G.P; if(!P) return;
  if(!collide(P.ci,P.rot,P.x+dx,P.y)){ P.x+=dx; sndMove(); resetLock(); } }
function tryRot(dir){ const P=G.P; if(!P) return;
  const nr=(P.rot+dir+4)%4;
  for(const k of KICKS){ if(!collide(P.ci,nr,P.x+k[0],P.y+k[1])){ P.rot=nr; P.x+=k[0]; P.y+=k[1]; sndRot(); resetLock(); return; } } }
function hardDrop(){ const P=G.P; if(!P) return;
  let d=0; while(!collide(P.ci,P.rot,P.x,P.y+1)&&d<ROWS+4){ P.y++; d++; }
  G.score+=2*d; sndHard(); lockPiece(); }

function lockPiece(){
  const P=G.P; if(!P) return;
  let cx0=99,cy0=99;
  for(const c of ROTS[P.ci][P.rot]){ const cx=P.x+c[0], cy=P.y+c[1];
    if(cy<0) continue;
    grid[cy][cx]={ci:P.ci,hp:MAXHP,flash:0,nv:P.nv};
    cx0=Math.min(cx0,cx); cy0=Math.min(cy0,cy);
  }
  sndPlace(); burst(FX+(P.x+1.5)*CELL,FY+(P.y+1)*CELL,'#8fb4ff',5,90,2);
  /* crush: piece cells above/level with enemy body */
  const crushed=new Set(); let cp=null;
  for(const c of ROTS[P.ci][P.rot]){ const cx=P.x+c[0], cy=P.y+c[1]; if(cy<0) continue;
    for(const e of G.enemies){ if(e.dead||crushed.has(e)) continue;
      const ecx=e.vx+.5, ecy=e.r-0.42;
      if(Math.abs(ecx-(cx+.5))<.65 && (ecy-(cy+.5))>=-1.2 && (ecy-(cy+.5))<=.3){ crushed.add(e); cp=e; damageEnemy(e,2,false); } } }
  if(crushed.size){ sndCrush(); addShake(3,.2);
    if(cp) floatText(FX+(cp.vx+.5)*CELL, FY+cp.r*CELL-14, crushed.size>1?'DOUBLE CRUSH!':'CRUSH!', '#ff9f43', 16); }
  if(P.nv){ novaBlast(FX+(P.x+1.5)*CELL, FY+(P.y+1)*CELL); }
  const full=[];
  for(let r=0;r<ROWS;r++){ let f=true; for(let c=0;c<COLS;c++) if(!grid[r][c]){ f=false; break; } if(f) full.push(r); }
  if(full.length) clearRows(full);
  G.P=null; spawnPiece();
}

/* ============================== UPDATE ============================== */
function update(dt){
  G.time+=dt;
  for(const s of stars){ s.y+=s.sp*dt*10; if(s.y>1) s.y-=1; }
  /* fx decay (always) */
  if(G.shT>0){ G.shT-=dt; if(G.shT<=0) G.shM=0; }
  G.coreFlash=Math.max(0,G.coreFlash-dt);
  if(G.banner){ G.banner.t-=dt; if(G.banner.t<=0) G.banner=null; }
  for(let r=0;r<ROWS;r++) for(let c=0;c<COLS;c++){ const b=grid[r][c]; if(b&&b.flash>0) b.flash=Math.max(0,b.flash-dt); }
  for(const p of G.parts){ p.x+=p.vx*dt; p.y+=p.vy*dt; p.vy+=p.g*dt; p.life-=dt; }
  G.parts=G.parts.filter(p=>p.life>0);
  for(const b of G.beams) b.t-=dt; G.beams=G.beams.filter(b=>b.t>0);
  for(const r of G.rings) r.t-=dt; G.rings=G.rings.filter(r=>r.t>0);
  for(const t of G.texts){ t.y-=28*dt; t.t-=dt; } G.texts=G.texts.filter(t=>t.t>0);

  if(G.state==='menu'||G.state==='over'){
    for(const a of ambient){ a.y+=a.v*dt; if(a.y>720){ a.y=-90; a.x=40+Math.random()*860; a.ci=(Math.random()*7)|0; } }
  }

  if(G.state!=='playing') return;

  /* DAS movement */
  const dir=(held.l?-1:0)+(held.r?1:0);
  if(dir){ if(dasDir!==dir){ dasDir=dir; dasT=0; accM=0; } else { dasT+=dt; if(dasT>.16){ accM+=dt; while(accM>.055){ accM-=.055; tryMove(dir); } } } }
  else { dasDir=0; accM=0; }

  /* gravity + lock */
  const P=G.P;
  if(P){
    const grounded=collide(P.ci,P.rot,P.x,P.y+1);
    if(grounded){ P.lockT+=dt; if(P.lockT>LOCKDELAY) lockPiece(); }
    else{ P.lockT=0; P.acc+=dt; let iv=Math.max(.8-G.wave*.03,.25); if(held.d) iv/=12;
      if(P.acc>=iv){ P.acc=0; if(!collide(P.ci,P.rot,P.x,P.y+1)){ P.y++; if(held.d) G.score+=1; } } }
  }

  /* wave flow */
  if(G.interT>0){ G.interT-=dt; if(G.interT<=0) startWave(); }
  else{
    if(G.toSpawn>0 && G.enemies.length<9){
      G.spawnT-=dt;
      if(G.spawnT<=0){
        if(G.bossPending){ spawnEnemy('boss'); G.bossPending=false; G.toSpawn--; }
        else{ spawnEnemy(); G.toSpawn--; }
        G.spawnT=G.spawnIv*(.7+Math.random()*.6);
      }
    } else if(G.toSpawn===0 && G.enemies.length===0){
      const bonus=100*G.wave; G.score+=bonus;
      floatText(FX+FW/2,FY+FH/2-20,'WAVE CLEAR  +'+bonus,'#3dffc4',20); sndWave();
      if(G.coreHp<5 && G.wave%3===0){ G.coreHp++; floatText(FX+FW/2,FY+FH/2+14,'CORE REPAIRED','#3dffc4',16); }
      G.interT=3.5;
    }
  }

  /* enemies */
  for(const e of G.enemies){
    if(e.dead) continue;
    e.flash=Math.max(0,e.flash-dt);
    e.vx+=(e.col-e.vx)*Math.min(1,dt*7);
    if(e.state==='descend'){
      e.r+=e.spd*dt;
      if(e.r>=ROWS-.12){ coreHit(e); continue; }
      const k=Math.floor(e.r);
      if(k>=0 && grid[k][e.col]){ if(e.r>k+.001){ e.r=k; e.state='attack'; e.atkT=0; } else e.r=k; }
    } else { /* attack */
      const k=Math.floor(e.r);
      if(k<0 || !grid[k][e.col]) e.state='descend';
      else{
        e.atkT+=dt;
        if(e.atkT>=e.atkIv){ e.atkT-=e.atkIv;
          const b=grid[k][e.col];
          if(b){ b.hp-=e.dmg; b.flash=.18; spark(FX+(e.col+.5)*CELL,FY+k*CELL+4,e.col); sndHit();
            if(b.hp<=0){ grid[k][e.col]=0; burst(FX+(e.col+.5)*CELL,FY+(k+.5)*CELL,COLORS[b.nv?7:b.ci].b,10,200,3.5); sndBreak(); } }
        }
        if(e.weave){ e.weaveT+=dt; if(e.weaveT>1.1){ e.weaveT=0;
          const opts=[-1,1]; for(const d of opts){ const c2=e.col+d; if(c2>=0&&c2<COLS&&!grid[k][c2]){ e.col=c2; break; } } } }
      }
    }
  }
  G.enemies=G.enemies.filter(e=>!e.dead);

  /* low core heartbeat */
  if(G.coreHp<=2 && G.state==='playing'){ G.hbT+=dt; if(G.hbT>1.1){ G.hbT=0; sndThump(); } }
}

  const bl2=.5+.5*Math.sin(G.time*4);
  ctx.font='900 italic 18px '+F_DISP; ctx.globalAlpha=.45+.55*bl2; ctx.fillStyle='#ffd23f'; sText('PRESS ENTER TO REBUILD',cx,470,2); ctx.globalAlpha=1;

function render(){
  /* full-screen bg */
  ctx.setTransform(dpr,0,0,dpr,0,0);
  const g=ctx.createLinearGradient(0,0,0,ih);
  g.addColorStop(0,'#04060f'); g.addColorStop(.55,'#0a1128'); g.addColorStop(1,'#0e1734');
  ctx.fillStyle=g; ctx.fillRect(0,0,iw,ih);
  const rg=ctx.createRadialGradient(iw/2,ih*.42,10,iw/2,ih*.42,Math.max(iw,ih)*.55);
  rg.addColorStop(0,'rgba(60,130,255,.10)'); rg.addColorStop(1,'rgba(0,0,0,0)');
  ctx.fillStyle=rg; ctx.fillRect(0,0,iw,ih);
  for(const s of stars){ const a=.2+.55*(.5+.5*Math.sin(G.time*s.tw+s.ph));
    ctx.globalAlpha=a; ctx.fillStyle=s.c; ctx.fillRect(s.x*iw,s.y*ih,s.s,s.s); }
  ctx.globalAlpha=1;

  /* world space + shake */
  const mag=G.shM*Math.min(1,Math.max(0,G.shT)/.35);
  const shx=(Math.random()*2-1)*mag, shy=(Math.random()*2-1)*mag;
  ctx.setTransform(dpr*scaleF,0,0,dpr*scaleF,(offX+shx)*dpr,(offY+shy)*dpr);

  drawPanels();
  drawField();
  if(G.state==='menu') drawMenu();
  else if(G.state==='paused') drawPause();
  else if(G.state==='over') drawOver();
  ctx.drawImage(fxc,0,0);
}

let last=performance.now();
function loop(now){
  const dt=Math.min(.033,(now-last)/1000); last=now;
  update(dt); render();
  requestAnimationFrame(loop);
}
requestAnimationFrame(loop);

**How to play:** Build Tetris walls in the field — enemies dive from above and chew through your blocks, so keep reinforcing. Every line clear fires a **pulse** that zaps all foes, dropping a piece directly onto an enemy **crushes** it for big damage, and rare gold pieces are **Novas** that blast everything on placement. Hold the bottom core line (5 shields), survive escalating waves, and take down the BASTION BREAKER boss every 5th wave. If your tower ever reaches the sky, a Skybreak clears it — the only way to lose is losing the core.
