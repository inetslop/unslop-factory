function forcesAt(L,x,y){
  let ax=0,ay=0;
  for(const w of L.wells){const dx=w.x-x,dy=w.y-y,d=Math.hypot(dx,dy);if(d<w.r&&d>.01){const f=w.p*(1-d/w.r);ax+=f*dx/d;ay+=f*dy/d;}}
  for(const v of L.vents){const dx=v.x-x,dy=v.y-y,d=Math.hypot(dx,dy);if(d<v.r&&d>.01){const f=v.p*(1-d/v.r);ax-=f*dx/d;ay-=f*dy/d;}}
  for(const s of L.streams)if(x>=s.x&&x<=s.x+s.w&&y>=s.y&&y<=s.y+s.h){ax+=s.fx;ay+=s.fy;}
  return{ax,ay};
}
function collideSeg(b,s){
  const dx=s[2]-s[0],dy=s[3]-s[1],L2=dx*dx+dy*dy||.0001;
  let t=((b.x-s[0])*dx+(b.y-s[1])*dy)/L2;t=clamp(t,0,1);
  const cx=s[0]+t*dx,cy=s[1]+t*dy;let nx=b.x-cx,ny=b.y-cy,d=Math.hypot(nx,ny);
  if(d>=b.r)return 0;
  if(d<.0001){nx=0;ny=-1;}else{nx/=d;ny/=d;}
  b.x=cx+nx*b.r;b.y=cy+ny*b.r;
  const vn=b.vx*nx+b.vy*ny;
  if(vn<0){const j=-(1+REST)*vn;b.vx+=j*nx;b.vy+=j*ny;return -vn;}
  return 0;
}
function stepBall(b,L,dt){
  const f=forcesAt(L,b.x,b.y);
  b.vx+=f.ax*dt;b.vy+=f.ay*dt;
  const sp=Math.hypot(b.vx,b.vy);
  if(sp>0){const dec=Math.min(sp,KFRIC*sp*dt+CDRAG*dt);b.vx-=b.vx/sp*dec;b.vy-=b.vy/sp*dec;}
  b.x+=b.vx*dt;b.y+=b.vy*dt;
  let imp=0;for(const s of L._w)imp=Math.max(imp,collideSeg(b,s));
  return imp;
}

function rollStep(dt){

  // cup

  if(dc<CUPR-3&&sp<CAPTURE){startSink();return;}

  if(lastWellHit){swallow();return;}
  // rest?
  const f=forcesAt(L,ball.x,ball.y),net=Math.hypot(f.ax,f.ay);

function previewPath(dx,dy,power){
  const L=LEVELS[cur];
  const b={x:ball.x,y:ball.y,vx:dx*MAXS*power,vy:dy*MAXS*power,r:BR};
  const pts=[];let hit=false,cup=false;const dt=1/120;
  for(let i=0;i<720;i++){
    if(stepBall(b,L,dt)>0){hit=true;break;}
    if(i%3===0)pts.push({x:b.x,y:b.y});
    if(Math.hypot(b.x-L.cup[0],b.y-L.cup[1])<CUPR-2){cup=true;break;}
  }
  return{pts,hit,cup};
}
function powColor(p){return p<.45?'#46e0b8':p<.78?'#ffd76a':'#ff6b5e';}

  const dx=aim.sx-aim.cx,dy=aim.sy-aim.cy,d=Math.hypot(dx,dy);
  if(d<4)return;
  const nx=dx/d,ny=dy/d,power=clamp(d/260,0,1),col=powColor(power);
  // pull line to pointer
  ctx.setLineDash([3,7]);ctx.strokeStyle='rgba(255,255,255,.3)';ctx.lineWidth=1.4;

let perf=0,last=performance.now(),dtG=0.016;
function frame(now){
  requestAnimationFrame(frame);
  let dt=Math.min(.033,(now-last)/1000);last=now;perf+=dt;dtG=dt;
  shake*=Math.exp(-5*dt);if(shake<.1)shake=0;
  for(let i=rings.length-1;i>=0;i--){rings[i].t+=dt*2.6;if(rings[i].t>=1)rings.splice(i,1);}
  for(let i=popups.length-1;i>=0;i--){popups[i].t+=dt;if(popups[i].t>1.3)popups.splice(i,1);}
  if(screen==='game'){
    if(ball.state==='rolling')rollStep(dt);
    else if(ball.state==='sinking')sinkStep(dt);
    if(pendingDone>0&&perf>=pendingDone){pendingDone=-1;openDone();}
  }
  render();
}
function render(){
  ctx.setTransform(1,0,0,1,0,0);
  if(neb)ctx.drawImage(neb,0,0);
  ctx.setTransform(dpr,0,0,dpr,0,0);
  drawStars();
  const sx=shake>0?(Math.random()*2-1)*shake:0,sy=shake>0?(Math.random()*2-1)*shake:0;
  ctx.save();ctx.translate(ox+sx,oy+sy);ctx.scale(scale,scale);
  drawField();
  if(screen==='menu'){drawMenuScene();}
  else{
    const L=LEVELS[cur];if(!L){ctx.restore();return;}
    drawStreams(L);drawCup(L);drawWells(L);drawVents(L);drawWalls(L);drawTee(L);
    if(ball.state!=='sunk'){
      drawTrail(trail,'255,236,180');
      const sp=Math.hypot(ball.vx,ball.vy);
      if(ball.state==='rolling'||ball.state==='idle')ball.scale=1;
      drawBall(ball.x,ball.y,sp);
    }
    drawRings();drawPopups();
    if(aim.active&&ball.state==='idle'&&screen==='game')drawAim();
  }
  ctx.restore();
}
