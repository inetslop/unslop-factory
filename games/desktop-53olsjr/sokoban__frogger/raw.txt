function tone({f=440,f2,dur=.15,type='sine',vol=.3,delay=0}){
  const a=ensureAC(); const t0=a.currentTime+delay;
  const o=a.createOscillator(), g=a.createGain();
  o.type=type; o.frequency.setValueAtTime(f,t0);
  if(f2) o.frequency.exponentialRampToValueAtTime(Math.max(1,f2),t0+dur);
  g.gain.setValueAtTime(0,t0);
  g.gain.linearRampToValueAtTime(vol,t0+0.008);
  g.gain.exponentialRampToValueAtTime(.0001,t0+dur);
  o.connect(g).connect(a.destination);
  o.start(t0); o.stop(t0+dur+.02);
}
function noise({dur=.3,vol=.3,from=1200,to=200,delay=0}){
  const a=ensureAC(); const t0=a.currentTime+delay;
  const n=Math.floor(a.sampleRate*dur), buf=a.createBuffer(1,n,a.sampleRate);
  const d=buf.getChannelData(0); for(let i=0;i<n;i++)d[i]=(Math.random()*2-1)*(1-i/n);
  const s=a.createBufferSource(); s.buffer=buf;
  const f=a.createBiquadFilter(); f.type='lowpass';
  f.frequency.setValueAtTime(from,t0); f.frequency.exponentialRampToValueAtTime(Math.max(40,to),t0+dur);
  const g=a.createGain(); g.gain.value=vol;
  s.connect(f).connect(g).connect(a.destination); s.start(t0);
}

function drown(dt){
  if(tileAt(frog.x,frog.y)!=='water'){frog.drownT=0;return}
  if(logAt(frog.x,frog.y)){frog.drownT=Math.max(0,frog.drownT-dt*2);return}
  frog.drownT+=dt;
  if(Math.random()<dt*16)burst(frog.px+(Math.random()-.5)*.3,frog.py-.1,'rgba(170,225,255,.9)',1,1.4);
  if(frog.drownT>0.7)doDeath('drown');
}
function carHit(){
  for(const c of cars){if(c.row===frog.y&&Math.abs(frog.px-c.x)<c.w/2+0.3){doDeath('squash');break}}
}
function doDeath(cause){
  if(state!=='play')return;
  state='dying';deathCause=cause;deathT=0;
  if(cause==='drown'){SFX.splash();shake=.25;burst(frog.px,frog.py,'rgba(160,220,255,.9)',18,2.4);flash('rgba(40,120,200,.9)');toast('SPLASHED!')}
  else{SFX.squish();shake=.5;burst(frog.px,frog.py,'#7ed957',16,2.6);flash('rgba(220,60,40,.9)');toast('FLATTENED!')}
}

/* ============================== fx ============================== */
function burst(x,y,col,n,spd){
  for(let i=0;i<n;i++){const a=Math.random()*TAU,s=.5+Math.random()*1.6*(spd||1);
    particles.push({x,y,vx:Math.cos(a)*s,vy:Math.sin(a)*s-1,g:5,t:0,life:.4+Math.random()*.4,col,size:.035+Math.random()*.05,shape:'p'})}}
function confetti(){
  for(let i=0;i<70;i++)particles.push({x:Math.random()*W,y:-Math.random()*1.6,vx:(Math.random()-.5)*1.2,vy:1+Math.random()*2,g:.8,t:0,
    life:2+Math.random(),col:CONF[i%CONF.length],size:.09+Math.random()*.08,shape:'c',spin:2+Math.random()*5})}
function popup(x,y,txt,col){popups.push({x,y,txt,col,t:0,life:1.1})}

/* ============================== update ============================== */
function updateVehicles(dt){
  const m=1.3;
  for(const c of cars){c.x+=c.dir*c.speed*speedMul*dt;
    if(c.dir>0&&c.x>W+m)c.x-=W+2*m;else if(c.dir<0&&c.x<-m)c.x+=W+2*m}
  for(const L of logs){const nx=L.x+L.dir*L.speed*speedMul*dt;let eff=nx-L.x;
    if(L.dir>0&&nx-L.len/2>W){const w2=nx-(W+L.len);eff=w2-L.x;L.x=w2}
    else if(L.dir<0&&nx+L.len/2<0){const w2=nx+(W+L.len);eff=w2-L.x;L.x=w2}
    else L.x=nx;
    L.dx=eff}
}
function frogUpdate(dt){
  const f=frog;
  let d=f.faceT-f.faceA;while(d>Math.PI)d-=TAU;while(d<-Math.PI)d+=TAU;f.faceA+=d*Math.min(1,dt*14);
  f.bumpT=Math.max(0,f.bumpT-dt/.15);
  if(f.moving){
    f.t+=dt/.16;
    if(f.t>=1){f.moving=false;f.px=f.x+.5;f.py=f.y+.5;
      for(let i=0;i<3;i++)particles.push({x:f.px+(Math.random()-.5)*.4,y:f.py+.2,vx:(Math.random()-.5)*.8,vy:-.6-Math.random(),g:3,t:0,life:.3,col:'rgba(190,225,175,.5)',size:.04,shape:'p'})}
    else{const t=f.t;f.px=lerp(f.fx,f.x+.5,t);f.py=lerp(f.fy,f.y+.5,t)-Math.sin(Math.PI*t)*.16}
  }else{
    if(tileAt(f.x,f.y)==='water'){const L=logAt(f.x,f.y);if(L){f.px+=L.dx;f.x=Math.floor(f.px)}}
    autoRepeat();
  }
  drown(dt);carHit();
}
function update(dt){
  time+=dt;
  if(state==='pause')return;
  updateVehicles(dt);
  shake=Math.max(0,shake-dt*1.7);
  for(let i=particles.length-1;i>=0;i--){const p=particles[i];p.t+=dt;if(p.t>p.life){particles.splice(i,1);continue}
    p.x+=p.vx*dt;p.y+=p.vy*dt;p.vy+=p.g*dt}
  for(let i=popups.length-1;i>=0;i--){popups[i].t+=dt;if(popups[i].t>popups[i].life)popups.splice(i,1)}
  for(const c of crates)c.bump=Math.max(0,c.bump-dt*3.2);
  if(state==='play'||state==='clear')rideWater();
  if(state==='play')frogUpdate(dt);
  if(state==='dying'){deathT+=dt;
    if(deathT>1.05){lives--;hud();
      if(lives<=0){state='over';SFX.over();fillOver();showOv($('ovOver'))}
      else startLevel(levelIdx)}}
  if(state==='clear'){clearT+=dt;if(clearT>=2.6)advanceLevel()}
}

    if(deathCause==='squash'){spin=p*6.5;al=1-p*p;sx=sy=1+p*.7}

function fit(){
  const r=viewEl.getBoundingClientRect(),pad=18;
  const cell=Math.max(14,Math.min((r.width-pad*2)/W,(r.height-pad*2)/H));
  const cssW=cell*W,cssH=cell*H,dpr=Math.min(2,window.devicePixelRatio||1);
  cv.style.width=cssW+'px';cv.style.height=cssH+'px';
  cv.width=Math.round(cssW*dpr);cv.height=Math.round(cssH*dpr);
  K=dpr*cell;
}
