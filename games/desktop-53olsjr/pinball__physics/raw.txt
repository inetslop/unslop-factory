/* ============================== MAIN LOOP ============================== */
var last=performance.now(), acc=0, HSTEP=1/240;
function frame(now){
  requestAnimationFrame(frame);
  var dt=(now-last)/1000; last=now; if(dt>.05)dt=.05; t+=dt;

  // flipper press sounds (edge)
  var L=heldL(),R2=heldR();
  if(L&&!prevLeft)S.flip(); if(R2&&!prevRight)S.flip(); prevLeft=L; prevRight=R2;

  // plunger charge / fire edge
  var chg=(keySpace||hasPtr('P'))&&state==='play'&&inPocket();
  if(chg)power=Math.min(1,power+dt*1.05);
  else if(state==='play'&&!inPocket())power=0;
  if(prevChg&&!chg){ if(inPocket())firePlunger(); else power=0; }
  prevChg=chg;

  // physics
  if(state==='play'||state==='lost'){ acc+=dt; var n=0; while(acc>=HSTEP&&n<40){ step(HSTEP); acc-=HSTEP; n++; } }

  if(state==='play'){
    // rollover lanes
    for(var i=0;i<rolls.length;i++){ var rl=rolls[i];
      if(!rl.lit){ var dx=ball.x-rl.x,dy=ball.y-rl.y; if(dx*dx+dy*dy<(BR+13)*(BR+13)){ rl.lit=true; rl.anim=1; addScore(50*mult,rl.x,rl.y-18); S.roll(); } } }
    var allL=true; for(i=0;i<rolls.length;i++)if(!rolls[i].lit)allL=false;
    if(allL&&!rollArmed){ rollArmed=true; addScore(500*mult,240,206,'LANE BONUS +'+(500*mult),'#7ff3ff'); S.bonus(); rollResetAt=t+1.3; }
    if(rollArmed&&t>rollResetAt){ for(i=0;i<rolls.length;i++)rolls[i].lit=false; rollArmed=false; }
    // stuck safety nudge
    var spd=Math.hypot(ball.vx,ball.vy);
    if(!inPocket()&&spd<15)stuckT+=dt; else stuckT=0;
    if(stuckT>2.5){ ball.vx+=(Math.random()-.5)*140; ball.vy-=60; stuckT=0; }
    // drain
    if(ball.y-BR>806)onLost();
    // trail
    if(spd>90){ trail.push({x:ball.x,y:ball.y}); if(trail.length>15)trail.shift(); }
    else if(trail.length)trail.shift();
  }
  if(state==='lost'){ lostT+=dt; if(lostT>1.5){ newBall(); state='play'; } }

  // decays
  for(i=0;i<bumpers.length;i++){ bumpers[i].flash=Math.max(0,bumpers[i].flash-dt*3.5); bumpers[i].rot+=dt*(.6+bumpers[i].flash*8); }
  for(i=0;i<slings.length;i++)slings[i].flash=Math.max(0,slings[i].flash-dt*4);
  for(i=0;i<rolls.length;i++)rolls[i].anim=Math.max(0,rolls[i].anim-dt*3);
  for(i=0;i<targets.length;i++)if(!targets[i].up)targets[i].dropT+=dt;
  plungerAnim=Math.max(0,plungerAnim-dt*4);
  shake*=Math.exp(-6*dt); if(shake<.05)shake=0;
  flashBig=Math.max(0,flashBig-dt*1.2);
  for(i=parts.length-1;i>=0;i--){ var p=parts[i]; p.vy+=G*.35*dt; p.x+=p.vx*dt; p.y+=p.vy*dt; p.l-=dt; if(p.l<=0)parts.splice(i,1); }
  for(i=pops.length-1;i>=0;i--){ pops[i].t+=dt; if(pops[i].t>.95)pops.splice(i,1); }

  render();
}
requestAnimationFrame(frame);
})();
