function physStep(dt){
  const g=H*1.9*(1+(State.wave-1)*0.045),REST=0.72;
  for(const b of blocks){ if(b.dead)continue;
    b.vy+=g*dt; b.vx*=(1-0.06*dt);
    b.x+=b.vx*dt; b.y+=b.vy*dt; b.ang+=b.av*dt; b.av*=(1-0.25*dt); }
  for(const b of blocks){ if(b.dead)continue;
    const cs=subCircles(b); let best=null,bd=0;
    for(const c of cs){
      if(c.x-c.r<8){const d=8-(c.x-c.r); if(d>bd){bd=d;best={nx:1,ny:0,d};}}
      else if(c.x+c.r>W-8){const d=(c.x+c.r)-(W-8); if(d>bd){bd=d;best={nx:-1,ny:0,d};}}
      if(c.y-c.r<6){const d=6-(c.y-c.r); if(d>bd){bd=d;best={nx:0,ny:1,d};}}
      if(State.mode!=='play'||b.demo){ const d=(c.y+c.r)-floorY; if(d>bd){bd=d;best={nx:0,ny:-1,d};} } }
    if(best){ b.x+=best.nx*best.d; b.y+=best.ny*best.d;
      const vn=b.vx*best.nx+b.vy*best.ny;
      if(vn<0){ b.vx-=(1+REST)*vn*best.nx; b.vy-=(1+REST)*vn*best.ny; b.av=lerp(b.av,clamp(b.vx/b.bh,-6,6),0.4); }
      b.av+=rand(-0.35,0.35); }
    if(State.mode==='play'&&!b.demo){ for(const c of cs){ if(c.y+c.r>=floorY-1&&b.vy>20){killBlock(b,true);break;} } } }
  for(let i=0;i<blocks.length;i++){ const A=blocks[i]; if(A.dead)continue;
    for(let j=i+1;j<blocks.length;j++){ const B=blocks[j]; if(B.dead)continue;
      let best=null,bd=0; const ca=subCircles(A),cb=subCircles(B);
      for(const a of ca)for(const c of cb){ const dx=c.x-a.x,dy=c.y-a.y,d=Math.hypot(dx,dy),rr=a.r+c.r;
        if(d<rr&&d>0.0001){const ov=rr-d; if(ov>bd){bd=ov;best={nx:dx/d,ny:dy/d,ov};}} }
      if(best){ const imA=1/A.mass,imB=1/B.mass,ims=imA+imB;
        A.x-=best.nx*best.ov*(imA/ims)*0.85; A.y-=best.ny*best.ov*(imA/ims)*0.85;
        B.x+=best.nx*best.ov*(imB/ims)*0.85; B.y+=best.ny*best.ov*(imB/ims)*0.85;
        const rvx=B.vx-A.vx,rvy=B.vy-A.vy,vn=rvx*best.nx+rvy*best.ny;
        if(vn<0){ const jp=-(1+0.75)*vn/ims;
          A.vx-=jp*imA*best.nx; A.vy-=jp*imA*best.ny; B.vx+=jp*imB*best.nx; B.vy+=jp*imB*best.ny;
          const vt=rvx*-best.ny+rvy*best.nx,vtc=clamp(vt,-400,400)*0.004; A.av+=vtc; B.av-=vtc; } } } }
}

    ctx.globalAlpha=a;ctx.font=dispFont(p.size);ctx.textAlign='center';ctx.textBaseline='middle';
    ctx.lineWidth=4;ctx.strokeStyle='rgba(8,15,20,0.85)';ctx.strokeText(p.txt,p.x,p.y);
    ctx.fillStyle=p.col;ctx.fillText(p.txt,p.x,p.y); }
  ctx.globalAlpha=1;
}

function render(){
  const g=ctx.createLinearGradient(0,0,0,H);
  g.addColorStop(0,'#12354a');g.addColorStop(0.55,'#0d2434');g.addColorStop(1,'#0a1622');
  ctx.fillStyle=g;ctx.fillRect(0,0,W,H);
  const pulse=0.1+0.05*Math.sin(tGlobal*0.9);
  const rg=ctx.createRadialGradient(W/2,floorY,10,W/2,floorY,H*0.55);
  rg.addColorStop(0,'rgba(255,145,50,'+pulse+')');rg.addColorStop(1,'rgba(255,145,50,0)');
  ctx.fillStyle=rg;ctx.fillRect(0,0,W,H);
  for(const s of streaks){ const x=((tGlobal*s.sp+s.off)%(W+300))-150;
    ctx.strokeStyle='rgba(125,234,255,0.035)';ctx.lineWidth=s.w;
    ctx.beginPath();ctx.moveTo(x,H*1.1);ctx.lineTo(x+H*0.35,-H*0.1);ctx.stroke(); }
  for(const gr of gears)drawGear(gr.fx*W,gr.fy*H,gr.r*Math.min(W,H),gr.teeth,tGlobal*gr.spd,gr.a);
  for(const m of motes){ const yy=(((m.fy-tGlobal*m.sp)%1)+1)%1*H;
    const xx=(m.fx+Math.sin(tGlobal*0.5+m.ph)*0.02)*W;
    ctx.fillStyle='rgba(160,220,235,'+(0.05+0.05*Math.sin(tGlobal+m.ph))+')';
    ctx.fillRect(xx,yy,m.s,m.s); }
  ctx.save();
  if(State.shake>0.2)ctx.translate(rand(-State.shake,State.shake),rand(-State.shake,State.shake));
  drawFloor();
  for(const b of blocks)if(!b.dead)drawShadow(b);
  for(const b of blocks)if(!b.dead)drawBlock(b);
  drawParts();drawPops();
  ctx.restore();
}

let last=performance.now();
function frame(now){
  requestAnimationFrame(frame);
  let dt=(now-last)/1000;last=now;if(dt>0.1)dt=0.1;
  tGlobal+=dt;
  const sdt=dt*(State.dying?0.35:1);
  if(State.mode==='play'){
    if(!State.dying)updatePlay(sdt);
    else{ State.dieT+=dt; if(State.dieT>1.05)finalizeOver(); }
  }else if(State.mode==='menu')updateDemo(sdt);
  if(State.mode==='play'||State.mode==='menu'){
    acc+=sdt;let n=0;
    while(acc>=STEP&&n<8){physStep(STEP);acc-=STEP;n++;}
    blocks=blocks.filter(b=>!b.dead); }
  if(State.mode!=='pause'){updateWorld(sdt);updateParts(sdt);updatePops(sdt);}
  State.shake=Math.max(0,State.shake-dt*(18+State.shake*2));
  render();
}
