  else if(state==='paused'){state='playing';el.pauseOv.classList.add('off');}
}

/* ================= UPDATE ================= */
function update(dt){
  tGlobal+=dt;
  shake*=Math.exp(-5*dt);
  var i,j,s,a,b;
  /* particles & popups always */
  for(i=parts.length-1;i>=0;i--){
    var p=parts[i]; p.life-=dt;
    if(p.life<=0){parts.splice(i,1);continue;}
    if(p.type==='ring'){p.r+=p.vr*dt;continue;}
    p.x+=p.vx*dt; p.y+=p.vy*dt;
    p.vx*=Math.exp(-1.8*dt); p.vy*=Math.exp(-1.8*dt);
  }
  for(i=pops.length-1;i>=0;i--){var pp=pops[i];pp.life-=dt;if(pp.life<=0)pops.splice(i,1);}

  if(state==='paused')return;

  if(state!=='playing'){ /* menu / gameover ambient field */
    if(overDelay>0)overDelay-=dt;
    for(i=0;i<asteroids.length;i++){var aa=asteroids[i];moveWrap(aa,dt);aa.rot+=aa.rs*dt;}
    if(asteroids.length<7&&Math.random()<dt*0.5)
      asteroids.push(makeAsteroid(rand(0,W),rand(0,H),Math.random()<0.4?2:3,1));
    return;
  }

  /* ---- ships ---- */
  for(i=0;i<2;i++){
    s=ships[i];
    if(!s.alive){
      engineSet(s.i,0);
      if(s.lives>0){s.respawnT-=dt;if(s.respawnT<=0){if(centerClear())respawnShip(s);else s.respawnT=0.4;}}
      continue;
    }
    var L=i===0?keys.has('KeyA'):keys.has('ArrowLeft');
    var R=i===0?keys.has('KeyD'):keys.has('ArrowRight');
    var T=i===0?keys.has('KeyW'):keys.has('ArrowUp');
    var F=i===0?(keys.has('Space')||keys.has('KeyF')):(keys.has('Enter')||keys.has('NumpadEnter'));
    s.a+=((L?-1:0)+(R?1:0))*4.4*dt;
    s.thrust=T;
    if(T){
      s.vx+=Math.cos(s.a)*380*dt; s.vy+=Math.sin(s.a)*380*dt;
      if(Math.random()<0.9&&parts.length<400){
        var rx=s.x-Math.cos(s.a)*10, ry=s.y-Math.sin(s.a)*10;
        parts.push({x:rx,y:ry,vx:-Math.cos(s.a)*rand(80,150)+s.vx*0.4,vy:-Math.sin(s.a)*rand(80,150)+s.vy*0.4,
          life:0.32,max:0.32,size:1.6,color:Math.random()<0.5?'#ffb347':'#ffe08a'});
      }
    }
    var sp=Math.hypot(s.vx,s.vy), mx=340+wave*4;
    if(sp>mx){s.vx*=mx/sp;s.vy*=mx/sp;}
    s.vx*=Math.exp(-0.5*dt); s.vy*=Math.exp(-0.5*dt);
    moveWrap(s,dt);
    s.fireCd-=dt; s.warpCd-=dt; s.invuln-=dt;
    if(F&&s.fireCd<=0)fire(s);
    engineSet(i,T?0.13:0);
  }

  /* ---- asteroids ---- */
  for(i=0;i<asteroids.length;i++){a=asteroids[i];moveWrap(a,dt);a.rot+=a.rs*dt;}

  /* ---- bullets ---- */
  for(i=bullets.length-1;i>=0;i--){
    b=bullets[i]; moveWrap(b,dt); b.life-=dt;
    if(b.life<=0){bullets.splice(i,1);continue;}
    var dead=false;
    if(saucer&&b.owner>=0&&dist2(b.x,b.y,saucer.x,saucer.y)<380){destroySaucer(b.owner);dead=true;}
    if(!dead)for(j=asteroids.length-1;j>=0;j--){
      a=asteroids[j]; var rr=a.r*0.9+4;
      if(dist2(b.x,b.y,a.x,a.y)<rr*rr){hitAsteroid(a,j,b.owner);dead=true;break;}
    }
    if(!dead)for(j=0;j<2;j++){
      s=ships[j];
      if(s.i===b.owner)continue;
      if(s.alive&&s.invuln<=0&&dist2(b.x,b.y,s.x,s.y)<144){killShip(s,b.owner);dead=true;break;}
    }
    if(dead)bullets.splice(i,1);
  }

  /* ---- ship vs field ---- */
  for(i=0;i<2;i++){
    s=ships[i];
    if(!s.alive||s.invuln>0)continue;
    for(j=0;j<asteroids.length;j++){
      a=asteroids[j]; var cr=a.r*0.85+9;
      if(dist2(s.x,s.y,a.x,a.y)<cr*cr){killShip(s,-1);break;}
    }
    if(s.alive&&saucer&&dist2(s.x,s.y,saucer.x,saucer.y)<(26)*(26)){destroySaucer(-1);killShip(s,-1);}
  }

  /* ---- saucer ---- */
  if(saucer){
    var u=saucer; u.t+=dt; u.x+=u.vx*dt; u.y+=Math.sin(u.t*2.6)*40*dt;
    if(u.x<-70||u.x>W+70){saucer=null;saucerStop();saucerTimer=rand(15,26);}
    else{
      u.fireT-=dt;
      if(u.fireT<=0){
        var tg=null,bd=1e9;
        for(i=0;i<2;i++){s=ships[i];if(s.alive){var d=dist2(u.x,u.y,s.x,s.y);if(d<bd){bd=d;tg=s;}}}
        if(tg){
          var ang=Math.atan2(tg.y-u.y,tg.x-u.x)+rand(-0.14,0.14);
          bullets.push({x:u.x,y:u.y+6,vx:Math.cos(ang)*300,vy:Math.sin(ang)*300,life:1.5,owner:-1,color:'#ff4d6d'});
          beep({type:'sawtooth',f0:420,f1:900,dur:0.1,vol:0.07});
        }
        u.fireT=rand(0.9,1.5);
      }
    }
  } else {
    saucerTimer-=dt;
    if(saucerTimer<=0&&ships.some(function(x){return x.alive;}))spawnSaucer();
  }

  /* ---- wave clear ---- */
  if(asteroids.length===0){
    waveTimer+=dt;
    if(waveTimer>1.3){wave++;spawnWave(wave);banner('WAVE '+wave);beep({type:'sawtooth',f0:150,f1:700,dur:0.4,vol:0.1});}
  } else waveTimer=0;

  if(ships.every(function(x){return !x.alive&&x.lives<=0;}))gameOver();
}

var stars=[];
function genStars(){
  var n=clamp(Math.round(W*H/9000),120,320); stars=[];
  for(var i=0;i<n;i++){
    var layer=Math.random()<0.55?0:(Math.random()<0.7?1:2);
    stars.push({fx:Math.random(),fy:Math.random(),r:layer===2?rand(1.3,2):rand(0.6,1.3),ph:rand(0,TAU),sp:rand(0.5,1.5),layer:layer});
  }
}
function neb(x,y,r,c){
  var g=ctx.createRadialGradient(x,y,0,x,y,r);
  g.addColorStop(0,c); g.addColorStop(1,'rgba(0,0,0,0)');
  ctx.fillStyle=g; ctx.fillRect(x-r,y-r,r*2,r*2);
}
function drawAsteroid(a){
  ctx.save(); ctx.translate(a.x,a.y); ctx.rotate(a.rot);
  ctx.strokeStyle='#c9d6e4'; ctx.lineWidth=2; ctx.lineJoin='round';
  ctx.shadowColor='rgba(180,210,255,0.7)'; ctx.shadowBlur=6;
  ctx.beginPath();
  var n=a.shape.length,i;
  for(i=0;i<n;i++){var ang=i/n*TAU,rr=a.r*a.shape[i];var px=Math.cos(ang)*rr,py=Math.sin(ang)*rr;i?ctx.lineTo(px,py):ctx.moveTo(px,py);}
  ctx.closePath();
  ctx.fillStyle='rgba(150,175,205,0.06)'; ctx.fill(); ctx.stroke();
  ctx.restore();
}
function drawShip(s){
  var inv=s.invuln>0;
  ctx.save(); ctx.translate(s.x,s.y); ctx.rotate(s.a);
  if(inv)ctx.globalAlpha=Math.sin(tGlobal*16)>0?0.95:0.3;
  if(s.thrust&&!inv){
    var L=rand(9,17);
    ctx.shadowColor='#ff8b3d'; ctx.shadowBlur=12;
    ctx.fillStyle=Math.random()<0.5?'#ffd76a':'#ff9a3d';
    ctx.beginPath(); ctx.moveTo(-8,4.5); ctx.lineTo(-8-L,0); ctx.lineTo(-8,-4.5); ctx.closePath(); ctx.fill();
  }
  ctx.shadowColor=s.color; ctx.shadowBlur=14;
  ctx.strokeStyle=s.color; ctx.lineWidth=2; ctx.lineJoin='round';
  ctx.beginPath(); ctx.moveTo(15,0); ctx.lineTo(-9,9); ctx.lineTo(-5,0); ctx.lineTo(-9,-9); ctx.closePath(); ctx.stroke();
  ctx.restore();
  if(inv){
    ctx.save(); ctx.globalAlpha=0.22+0.14*Math.sin(tGlobal*8);
    ctx.strokeStyle=s.color; ctx.setLineDash([5,7]);
    ctx.beginPath(); ctx.arc(s.x,s.y,20,0,TAU); ctx.stroke();
    ctx.setLineDash([]); ctx.restore();
  }
}
function drawSaucer(u){
  ctx.save(); ctx.translate(u.x,u.y);
  ctx.strokeStyle='#ff4d6d'; ctx.lineWidth=2;
  ctx.shadowColor='rgba(255,77,109,0.8)'; ctx.shadowBlur=12;
  ctx.beginPath(); ctx.ellipse(0,0,16,6.5,0,0,TAU); ctx.stroke();
  ctx.beginPath(); ctx.ellipse(0,-3,7,4.5,0,Math.PI,0,true); ctx.stroke();
  var on=Math.floor(tGlobal*9)%2;
  ctx.fillStyle='#ffd6de'; ctx.shadowBlur=8;
  for(var i=-1;i<=1;i++){ctx.beginPath();ctx.arc(i*8,on?2.5:4.5,1.6,0,TAU);ctx.fill();}
  ctx.restore();
}
function render(){
  /* deep space base */
  var bg=ctx.createRadialGradient(W/2,H*0.42,0,W/2,H/2,Math.max(W,H)*0.75);
  bg.addColorStop(0,'#0a1322'); bg.addColorStop(1,'#04060d');
  ctx.fillStyle=bg; ctx.fillRect(0,0,W,H);
  var M=Math.max(W,H);
  neb(W*(0.25+0.03*Math.sin(tGlobal*0.05)),H*0.3,M*0.45,'rgba(46,140,255,0.10)');
  neb(W*0.78,H*(0.62+0.03*Math.cos(tGlobal*0.04)),M*0.4,'rgba(255,70,150,0.07)');
  neb(W*0.55,H*0.8,M*0.3,'rgba(255,150,60,0.05)');
  /* parallax starfield */
  var drift=[4,9,16],i;
  for(i=0;i<stars.length;i++){
    var st=stars[i];
    var x=((st.fx*W+tGlobal*drift[st.layer])%W+W)%W, y=st.fy*H;
    ctx.globalAlpha=(0.2+0.3*st.layer/2)*(0.6+0.4*Math.sin(tGlobal*st.sp+st.ph));
    ctx.fillStyle=st.layer===2?'#cfe9ff':'#8fb4d8';
    ctx.fillRect(x,y,st.r,st.r);
  }
  ctx.globalAlpha=1;

  ctx.save();
  if(shake>0.2)ctx.translate(rand(-shake,shake),rand(-shake,shake));

  for(i=0;i<asteroids.length;i++)drawAsteroid(asteroids[i]);
  if(saucer)drawSaucer(saucer);

  /* bullets */
  for(i=0;i<bullets.length;i++){
    var bb=bullets[i];
    ctx.shadowColor=bb.color; ctx.shadowBlur=8;
    ctx.fillStyle=bb.color;
    ctx.beginPath(); ctx.arc(bb.x,bb.y,2.6,0,TAU); ctx.fill();
  }
  ctx.shadowBlur=0;

  for(i=0;i<2;i++)if(ships[i]&&ships[i].alive)drawShip(ships[i]);

  /* particles — additive */
  ctx.globalCompositeOperation='lighter';
  for(i=0;i<parts.length;i++){
    var p=parts[i], al=p.life/p.max;
    ctx.globalAlpha=al;
    if(p.type==='ring'){
      ctx.strokeStyle=p.color; ctx.lineWidth=2;
      ctx.beginPath(); ctx.arc(p.x,p.y,p.r,0,TAU); ctx.stroke();
    } else {
      ctx.strokeStyle=p.color; ctx.lineWidth=p.size;
      ctx.beginPath(); ctx.moveTo(p.x,p.y); ctx.lineTo(p.x-p.vx*0.03,p.y-p.vy*0.03); ctx.stroke();
    }
  }
  ctx.globalCompositeOperation='source-over';
  ctx.globalAlpha=1;

  /* score popups */
  ctx.font='700 13px Consolas, Menlo, monospace'; ctx.textAlign='center';
  for(i=0;i<pops.length;i++){
    var pp=pops[i];
    ctx.globalAlpha=clamp(pp.life/pp.max,0,1);
    ctx.fillStyle=pp.color;
    ctx.fillText(pp.text,pp.x,pp.y-(pp.max-pp.life)*26);
  }
  ctx.globalAlpha=1;
  ctx.restore();
}
