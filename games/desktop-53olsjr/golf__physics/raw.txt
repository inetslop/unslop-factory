  const lv={name:def.name,par:def.par,start:{x:def.start[0],y:def.start[1]},hole:{x:def.hole[0],y:def.hole[1]},
    walls:def.walls||[],waters:def.waters||[],wind:def.wind||{x:0,y:0},
    trees:(def.trees||[]).map(t=>({x:t[0],y:t[1],r:t[2]})),
    sands:(def.sands||[]).map(s=>{const specks=[];const n=(s[2]*s[2]/240)|0;
      for(let i=0;i<n;i++){const a=Math.random()*6.283,d=Math.sqrt(Math.random())*s[2]*.88;specks.push({x:s[0]+Math.cos(a)*d,y:s[1]+Math.sin(a)*d});}

    bumpers:(def.bumpers||[]).map(b=>({x:b[0],y:b[1],r:b[2],hit:-9})),
    windmills:(def.windmills||[]).map(w=>({...w,ang:Math.random()*6.28}))};
  return lv;
}

function circleRectHit(b,r,rest){
  const cx=clamp(b.x,r.x,r.x+r.w),cy=clamp(b.y,r.y,r.y+r.h);
  let dx=b.x-cx,dy=b.y-cy,d=Math.hypot(dx,dy);
  if(d>=BALL_R)return 0;
  let nx,ny,pen;
  if(d>1e-6){nx=dx/d;ny=dy/d;pen=BALL_R-d;}
  else{const l=b.x-r.x,r2=r.x+r.w-b.x,t2=b.y-r.y,bt=r.y+r.h-b.y,m=Math.min(l,r2,t2,bt);
    if(m===l){nx=-1;ny=0;pen=l+BALL_R;}else if(m===r2){nx=1;ny=0;pen=r2+BALL_R;}
    else if(m===t2){nx=0;ny=-1;pen=t2+BALL_R;}else{nx=0;ny=1;pen=bt+BALL_R;}}
  b.x+=nx*pen;b.y+=ny*pen;
  const vn=b.vx*nx+b.vy*ny;
  if(vn<0){const j=-(1+rest)*vn;b.vx+=j*nx;b.vy+=j*ny;return -vn;}
  return 0;}
function circleCircleHit(b,cx,cy,cr,rest){
  let dx=b.x-cx,dy=b.y-cy,d=Math.hypot(dx,dy);const min=BALL_R+cr;
  if(d>=min)return 0;
  let nx,ny;if(d>1e-6){nx=dx/d;ny=dy/d;}else{nx=1;ny=0;d=0;}
  const pen=min-d;b.x+=nx*pen;b.y+=ny*pen;
  const vn=b.vx*nx+b.vy*ny;
  if(vn<0){const j=-(1+rest)*vn;b.vx+=j*nx;b.vy+=j*ny;return -vn;}
  return 0;}
function windmillHit(b,wm){
  for(let i=0;i<3;i++){const a=wm.ang+i*2.0944;
    const ex=wm.cx+Math.cos(a)*wm.arm,ey=wm.cy+Math.sin(a)*wm.arm;
    const bx=ex-wm.cx,by=ey-wm.cy,tt=clamp(((b.x-wm.cx)*bx+(b.y-wm.cy)*by)/(bx*bx+by*by),0,1);
    const px=wm.cx+bx*tt,py=wm.cy+by*tt;
    let dx=b.x-px,dy=b.y-py,d=Math.hypot(dx,dy);const minD=BALL_R+9;
    if(d<minD&&d>1e-6){const nx=dx/d,ny=dy/d;b.x+=nx*(minD-d);b.y+=ny*(minD-d);
      const rx=px-wm.cx,ry=py-wm.cy,svx=-wm.speed*ry,svy=wm.speed*rx;
      const vn=(b.vx-svx)*nx+(b.vy-svy)*ny;
      if(vn<0){const j=-(1.75)*vn;b.vx+=j*nx;b.vy+=j*ny;
        if(T-lastWallSnd>.1){lastWallSnd=T;sfx.wall(.4);}}}}}

function inSand(b){for(const s of LV.sands){const dx=b.x-s.x,dy=b.y-s.y;if(dx*dx+dy*dy<(s.r-4)*(s.r-4))return true;}return false;}
function inWater(b){return LV.waters.some(r=>b.x>r.x+3&&b.x<r.x+r.w-3&&b.y>r.y+3&&b.y<r.y+r.h-3);}

// returns 'water' | 'sink' | null ; real=true applies hazards/sound, false = preview
function physStep(b,h,real){
  const isSand=inSand(b);
  const k=isSand?3.9:.72,lin=isSand?300:52;
  b.vx*=Math.exp(-k*h);b.vy*=Math.exp(-k*h);
  let sp=Math.hypot(b.vx,b.vy),dl=lin*h;
  if(sp>0){if(sp<=dl){b.vx=b.vy=0;sp=0;}else{b.vx-=b.vx/sp*dl;b.vy-=b.vy/sp*dl;sp-=dl;}}
  b.vx+=LV.wind.x*h;b.vy+=LV.wind.y*h;
  b.x+=b.vx*h;b.y+=b.vy*h;
  let hitV=0;
  const L=P.x+BALL_R,Rt=P.x+P.w-BALL_R,Tp=P.y+BALL_R,Bt=P.y+P.h-BALL_R;
  if(b.x<L){b.x=L;if(b.vx<0){hitV=Math.max(hitV,-b.vx);b.vx=-b.vx*.72;}}
  if(b.x>Rt){b.x=Rt;if(b.vx>0){hitV=Math.max(hitV,b.vx);b.vx=-b.vx*.72;}}
  if(b.y<Tp){b.y=Tp;if(b.vy<0){hitV=Math.max(hitV,-b.vy);b.vy=-b.vy*.72;}}
  if(b.y>Bt){b.y=Bt;if(b.vy>0){hitV=Math.max(hitV,b.vy);b.vy=-b.vy*.72;}}
  for(const w of LV.walls)hitV=Math.max(hitV,circleRectHit(b,w,.72));
  for(const t of LV.trees)circleCircleHit(b,t.x,t.y,t.r,.5);
  for(const bu of LV.bumpers){const imp=circleCircleHit(b,bu.x,bu.y,bu.r,1.02);
    if(imp>0&&real){bu.hit=T;let s2=Math.hypot(b.vx,b.vy);if(s2<340&&s2>0){b.vx*=340/s2;b.vy*=340/s2;}
      sfx.bump();S.shake=Math.max(S.shake,REDUCED?0:5);burst(b.x,b.y,6,'#ffd88f',180,.35);}}
  if(hitV>150&&real){if(T-lastWallSnd>.09){lastWallSnd=T;sfx.wall(Math.min(1,hitV/1400));}
    S.shake=Math.max(S.shake,REDUCED?0:Math.min(7,hitV/220));}
  if(isSand&&real&&!S.inSandPrev&&sp>140){if(T-lastSandSnd>.25){lastSandSnd=T;sfx.sand();}burst(b.x,b.y,5,'#e2c98d',120,.3);}
  S.inSandPrev=isSand;
  if(!real){
    if(inWater(b))return 'water';
    const dx=b.x-LV.hole.x,dy=b.y-LV.hole.y,d=Math.hypot(dx,dy);
    if(d<HOLE_R&&Math.hypot(b.vx,b.vy)<340)return 'sink';
    return null;}
  for(const wm of LV.windmills)windmillHit(b,wm);
  if(inWater(b)){doWater();return 'water';}
  const dx=b.x-LV.hole.x,dy=b.y-LV.hole.y,d=Math.hypot(dx,dy)||.001;
  sp=Math.hypot(b.vx,b.vy);
  if(d<52&&sp<600){const f=(1-d/52)*950*h;b.vx-=dx/d*f;b.vy-=dy/d*f;}
  if(d<HOLE_R&&Math.hypot(b.vx,b.vy)<340){startSink();return 'sink';}
  return null;}

function burst(x,y,n,color,spd,life){if(REDUCED)n=Math.ceil(n/2);
  for(let i=0;i<n;i++){const a=Math.random()*6.283,s=spd*(.4+Math.random()*.8);
    parts.push({x,y,vx:Math.cos(a)*s,vy:Math.sin(a)*s,life:life*(.6+Math.random()*.7),max:life,color,size:1.5+Math.random()*2.5});}}
function spawnConfetti(x,y,n){if(REDUCED)n=Math.ceil(n/2);
  const cols=['#ffd23e','#d84343','#f7f1df','#7ee08a','#5ab0ff'];
  for(let i=0;i<n;i++){const a=-Math.PI/2+(Math.random()-.5)*1.8,s=260+Math.random()*520;
    confetti.push({x,y,vx:Math.cos(a)*s,vy:Math.sin(a)*s,rot:Math.random()*6.28,vr:(Math.random()-.5)*14,
      w:5+Math.random()*4,h:8+Math.random()*5,color:cols[(Math.random()*cols.length)|0],life:1.4+Math.random()*1.1});}}
function stepFx(dt){
  for(let i=parts.length-1;i>=0;i--){const p=parts[i];p.vy+=300*dt;p.x+=p.vx*dt;p.y+=p.vy*dt;p.life-=dt;if(p.life<=0)parts.splice(i,1);}
  for(let i=confetti.length-1;i>=0;i--){const c=confetti[i];c.vy+=640*dt;c.vx*=Math.exp(-.6*dt);
    c.x+=c.vx*dt;c.y+=c.vy*dt;c.rot+=c.vr*dt;c.life-=dt;if(c.life<=0)confetti.splice(i,1);}}

function simulate(x,y,vx,vy){const b={x,y,vx,vy},pts=[];let end=null;
  for(let i=0;i<110;i++){end=physStep(b,1/60,false);if(end)break;
    if(i%2===0)pts.push({x:b.x,y:b.y});
    if(Math.hypot(b.vx,b.vy)<8){end='stop';break;}}
  return{pts,end};}

function update(dt){T+=dt;
  if(LV)for(const w of LV.windmills)w.ang+=w.speed*dt;
  S.shake=Math.max(0,S.shake-dt*4);stepFx(dt);
  if(S.mode!=='play')return;
  if(S.sub==='wait'){S.waitT-=dt;if(S.waitT<=0)S.sub='idle';return;}
  if(S.sub==='sinking'){S.sinkT+=dt/.8;if(S.sinkT>=1){S.sinkT=1;holeDone();}return;}
  if(S.sub==='rolling'){const n=Math.max(1,Math.ceil(dt/(1/240))),h=dt/n;
    for(let i=0;i<n;i++){physStep(S.ball,h,true);if(S.sub!=='rolling')break;}}
  if(S.sub==='idle'){
    if(keys.l)S.kbAngle-=2.8*dt;if(keys.r)S.kbAngle+=2.8*dt;
    if(keys.q)S.kbAngle-=.9*dt;if(keys.e)S.kbAngle+=.9*dt;
    if(S.charging&&!S.pointerAim)S.chargeT+=dt;}}

    const tg=ctx.createRadialGradient(t.x-r*.3,t.y-r*.3,r*.15,t.x,t.y,r);tg.addColorStop(0,'#2f9a4e');tg.addColorStop(1,'#175c2e');
    ctx.fillStyle=tg;ctx.beginPath();ctx.arc(t.x,t.y,r,0,6.283);ctx.fill();
    ctx.strokeStyle='rgba(10,40,20,.5)';ctx.lineWidth=3;ctx.stroke();
    ctx.fillStyle='rgba(255,255,255,.12)';ctx.beginPath();ctx.arc(t.x-r*.35,t.y-r*.35,r*.32,0,6.283);ctx.fill();});
  // bumpers
  for(const bu of LV.bumpers){const p=clamp(1-(T-bu.hit)*4,0,1),sc=1+p*.22;
    ctx.fillStyle='rgba(0,0,0,.25)';ctx.beginPath();ctx.ellipse(bu.x+4,bu.y+6,bu.r*sc,bu.r*.85*sc,0,0,6.283);ctx.fill();
    const bg=ctx.createRadialGradient(bu.x-bu.r*.3,bu.y-bu.r*.3,2,bu.x,bu.y,bu.r*sc);
    if(p>.4){bg.addColorStop(0,'#fff7e0');bg.addColorStop(1,'#ffb85c');}else{bg.addColorStop(0,'#ffcf6e');bg.addColorStop(1,'#d98a2b');}
    ctx.fillStyle=bg;ctx.beginPath();ctx.arc(bu.x,bu.y,bu.r*sc,0,6.283);ctx.fill();
    ctx.strokeStyle='#8a5312';ctx.lineWidth=3;ctx.stroke();
    ctx.strokeStyle='rgba(0,0,0,.18)';ctx.lineWidth=2;ctx.beginPath();ctx.arc(bu.x,bu.y,bu.r*.55*sc,0,6.283);ctx.stroke();
    if(p>0){ctx.strokeStyle=`rgba(255,240,200,${p})`;ctx.lineWidth=4;ctx.beginPath();ctx.arc(bu.x,bu.y,bu.r*sc+5+p*10,0,6.283);ctx.stroke();}}
  // windmills
  for(const wm of LV.windmills){
    ctx.fillStyle='rgba(0,0,0,.28)';ctx.beginPath();ctx.ellipse(wm.cx+6,wm.cy+8,20,14,0,0,6.283);ctx.fill();
    ctx.fillStyle='#7d838c';ctx.beginPath();ctx.arc(wm.cx,wm.cy,16,0,6.283);ctx.fill();
    ctx.strokeStyle='#565b63';ctx.lineWidth=3;ctx.stroke();
    ctx.lineCap='round';ctx.strokeStyle='rgba(240,230,208,.1)';ctx.lineWidth=16;
    for(let i=0;i<3;i++){const a=wm.ang+i*2.0944;ctx.beginPath();ctx.arc(wm.cx,wm.cy,wm.arm*.7,a-.5,a);ctx.stroke();}
    for(let i=0;i<3;i++){const a=wm.ang+i*2.0944;
      ctx.save();ctx.translate(wm.cx,wm.cy);ctx.rotate(a);
      ctx.fillStyle=i%2?'#d84343':'#f0e6d0';rr(ctx,10,-9,wm.arm-10,18,9);ctx.fill();
      ctx.strokeStyle='rgba(0,0,0,.25)';ctx.lineWidth=2;ctx.stroke();ctx.restore();}

function drawAim(){
  if(S.mode!=='play'||S.sub!=='idle')return;
  let ang=null,power=0;
  if(S.pointerAim){const a=S.pointerAim,dx=a.sx-a.cx,dy=a.sy-a.cy,len=Math.hypot(dx,dy);
    if(len>6){ang=Math.atan2(dy,dx);power=clamp(len/300,0,1);}}
  else if(S.charging){ang=S.kbAngle;power=tri(S.chargeT/1.4);}
  if(ang===null)return;
  const b=S.ball,col=powerColor(power),L=36+power*150;
  // preview path
  const sim=simulate(b.x,b.y,Math.cos(ang)*(260+power*1240),Math.sin(ang)*(260+power*1240));
  const n=sim.pts.length;
  for(let i=0;i<n;i++){const pt=sim.pts[i],f=i/n;
    ctx.fillStyle=`rgba(255,255,255,${(1-f)*.8})`;ctx.beginPath();ctx.arc(pt.x,pt.y,4.6-2.6*f,0,6.283);ctx.fill();}
  if(sim.end==='water'){const pt=sim.pts[n-1];if(pt){ctx.fillStyle='#ff9d6b';ctx.beginPath();ctx.arc(pt.x,pt.y,7,0,6.283);ctx.fill();}}
  if(sim.end==='sink'){ctx.strokeStyle=`rgba(232,182,76,${.5+.4*Math.sin(T*6)})`;ctx.lineWidth=3;
    ctx.beginPath();ctx.arc(LV.hole.x,LV.hole.y,HOLE_R+6,0,6.283);ctx.stroke();}
  // arrow
  const sx=b.x+Math.cos(ang)*(BALL_R+6),sy=b.y+Math.sin(ang)*(BALL_R+6);
  const ex=sx+Math.cos(ang)*L,ey=sy+Math.sin(ang)*L;
  ctx.setLineDash([9,7]);ctx.strokeStyle=col;ctx.lineWidth=3.5;
  ctx.beginPath();ctx.moveTo(sx,sy);ctx.lineTo(ex,ey);ctx.stroke();ctx.setLineDash([]);
  const hs=10+power*6;ctx.fillStyle=col;ctx.save();ctx.translate(ex,ey);ctx.rotate(ang);
  ctx.beginPath();ctx.moveTo(hs,0);ctx.lineTo(-hs*.5,hs*.62);ctx.lineTo(-hs*.5,-hs*.62);ctx.closePath();ctx.fill();ctx.restore();
  // power ring
  ctx.strokeStyle=col;ctx.lineWidth=5;ctx.lineCap='round';
  ctx.beginPath();ctx.arc(b.x,b.y,24,-Math.PI/2,-Math.PI/2+power*6.283);ctx.stroke();ctx.lineCap='butt';}

function drawBall(){
  const b=S.ball;
  if(S.sub==='sinking'){const t=S.sinkT,e=t*t;
    const x=lerp(S.sinkFrom.x,LV.hole.x,e),y=lerp(S.sinkFrom.y,LV.hole.y,e),sc=Math.max(0,1-t);
    if(sc>0){ctx.fillStyle='#ffffff';ctx.beginPath();ctx.arc(x,y,BALL_R*sc,0,6.283);ctx.fill();}return;}
  ctx.fillStyle='rgba(0,0,0,.25)';ctx.beginPath();ctx.ellipse(b.x+5,b.y+7,BALL_R*.95,BALL_R*.8,0,0,6.283);ctx.fill();
  if(S.mode==='play'&&S.sub==='idle'){ctx.strokeStyle=`rgba(255,255,255,${.14+.1*Math.sin(T*3)})`;ctx.lineWidth=2;
    ctx.beginPath();ctx.arc(b.x,b.y,BALL_R+6+Math.sin(T*3)*1.5,0,6.283);ctx.stroke();}
  const g=ctx.createRadialGradient(b.x-4,b.y-5,1.5,b.x,b.y,BALL_R);
  g.addColorStop(0,'#ffffff');g.addColorStop(.75,'#eef1ea');g.addColorStop(1,'#c3cdc2');
  ctx.fillStyle=g;ctx.beginPath();ctx.arc(b.x,b.y,BALL_R,0,6.283);ctx.fill();
  ctx.strokeStyle='rgba(90,110,95,.4)';ctx.lineWidth=1.5;ctx.stroke();}

function drawFx(){
  for(const p of parts){ctx.globalAlpha=clamp(p.life/p.max,0,1);ctx.fillStyle=p.color;
    ctx.beginPath();ctx.arc(p.x,p.y,p.size,0,6.283);ctx.fill();}
  ctx.globalAlpha=1;
  for(const c of confetti){ctx.save();ctx.translate(c.x,c.y);ctx.rotate(c.rot);
    ctx.globalAlpha=clamp(c.life/.4,0,1);ctx.fillStyle=c.color;ctx.fillRect(-c.w/2,-c.h/2,c.w,c.h);ctx.restore();}
  ctx.globalAlpha=1;}

function render(){
  ctx.setTransform(DPR,0,0,DPR,0,0);
  const bg=ctx.createRadialGradient(W/2,H*.42,80,W/2,H/2,Math.max(W,H)*.75);
  bg.addColorStop(0,'#1b4630');bg.addColorStop(1,'#0a1c12');ctx.fillStyle=bg;ctx.fillRect(0,0,W,H);
  const sh=S.shake,sx=(Math.random()*2-1)*sh,sy=(Math.random()*2-1)*sh;

  v.addColorStop(0,'rgba(0,0,0,0)');v.addColorStop(1,'rgba(0,0,0,.42)');ctx.fillStyle=v;ctx.fillRect(0,0,W,H);}
