function update(dt,now){
  if(state==='countdown'){
    const el=now-cdStart;
    while(cdStep<cdPlan.length-1&&el>=cdPlan[cdStep+1].t){ cdStep++; const s=cdPlan[cdStep]; showBanner(s.txt,s.cls); (SFX[s.sfx]||SFX.tick)(); }
    if(el>cdPlan[cdPlan.length-1].t+520){ state='playing'; burst(W/2-30,H/2,C1,18,300); burst(W/2+30,H/2,C2,18,300); }
  }else if(state==='playing'){
    for(let i=0;i<2;i++){ const p=P[i];
      if(p.frenzyUntil&&now>p.frenzyUntil){ p.frenzyUntil=0; btns[i].classList.remove('frenzy'); }
      if(p.combo>0&&now-p.last>650)p.combo=0;
    }
    if(!suddenDeath){
      timeLeft-=dt;
      if(timeLeft<=5.02&&timeLeft>0){ const w=Math.ceil(timeLeft); if(w!==lastWhole){ lastWhole=w; SFX.tick(); } }
      if(timeLeft<=0){
        timeLeft=0;
        if(P[0].score===P[1].score){ suddenDeath=true; showBanner('SUDDEN DEATH','danger'); SFX.sd(); zoneFlash(0); zoneFlash(1); shakeMag=7; }
        else endRound(P[0].score>P[1].score?0:1);
      }
    }
  }
  if(state==='gameover'&&now-lastConf>70){ lastConf=now; spawnConfetti(); }
}
function render(now){
  for(let i=0;i<2;i++){ const p=P[i];
    if(p.score!==scCache[i]){ scCache[i]=p.score; scoreEls[i].textContent=p.score;
      scoreEls[i].classList.remove('bump'); void scoreEls[i].offsetWidth; scoreEls[i].classList.add('bump'); }
    const mult=Math.min(5,1+Math.floor(p.combo/10));
    const ms='×'+mult; if(multEls[i].textContent!==ms)multEls[i].textContent=ms;
    const cs='COMBO '+p.combo; if(ccEls[i].textContent!==cs)ccEls[i].textContent=cs;
    comboEls[i].classList.toggle('hot',mult>1||now<p.frenzyUntil);
    let frac,label,cls='';
    if(now<p.frenzyUntil){ frac=(p.frenzyUntil-now)/FZ*100; label='×2 ACTIVE'; cls='active'; }
    else if(p.meter>=100){ frac=100; label=i===0?'READY — PRESS S':'READY — PRESS K'; cls='ready'; }
    else{ frac=p.meter; label='FRENZY'; }
    ffills[i].style.width=frac+'%';
    const cn='fbar '+cls; if(fbars[i].className!==cn)fbars[i].className=cn;
    if(flabels[i].textContent!==label)flabels[i].textContent=label;
  }
  let ts;
  if(state==='playing'&&suddenDeath)ts='SD';
  else if(state==='countdown'||state==='menu')ts=String(roundLen);
  else ts=timeLeft>=9.95?String(Math.ceil(timeLeft)):Math.max(0,timeLeft).toFixed(1);
  if(timerEl.textContent!==ts)timerEl.textContent=ts;
  timerEl.classList.toggle('danger',state==='playing'&&!suddenDeath&&timeLeft<=5.02);
}

/* ================= FX DRAW ================= */
function drawFX(dt){
  cx.clearRect(0,0,W,H);
  for(let i=0;i<2;i++){ if(performance.now()<P[i].frenzyUntil&&Math.random()<0.5)
    parts.push({x:W/2+rand(-14,14),y:rand(0,H),vx:rand(-60,60),vy:rand(-30,30),life:.5,t:0,color:GOLD,size:rand(1.5,3.5),g:0}); }
  for(let i=parts.length-1;i>=0;i--){ const p=parts[i]; p.t+=dt;
    if(p.t>p.life||p.y>H+40){ parts.splice(i,1); continue; }
    p.vy+=(p.g||0)*dt; p.x+=p.vx*dt; p.y+=p.vy*dt; if(p.rot!=null)p.rot+=p.vr*dt;
    cx.globalAlpha=1-p.t/p.life; cx.fillStyle=p.color;
    if(p.rect){ cx.save(); cx.translate(p.x,p.y); cx.rotate(p.rot||0); cx.fillRect(-p.size/2,-p.size/4,p.size,p.size/2); cx.restore(); }
    else{ cx.beginPath(); cx.arc(p.x,p.y,p.size*(1-p.t/p.life)+.5,0,6.283); cx.fill(); }
  }
  cx.globalAlpha=1;
  cx.textAlign='center';
  for(let i=floats.length-1;i>=0;i--){ const f=floats[i]; f.t+=dt;
    if(f.t>f.life){ floats.splice(i,1); continue; }
    f.y-=55*dt;
    const a=f.t<f.life*.6?1:1-(f.t-f.life*.6)/(f.life*.4);
    cx.globalAlpha=a; cx.font='700 '+f.size+'px Menlo,Consolas,monospace';
    cx.lineWidth=4; cx.strokeStyle='rgba(0,0,0,.6)'; cx.strokeText(f.text,f.x,f.y);
    cx.fillStyle=f.color; cx.fillText(f.text,f.x,f.y);
  }
  cx.globalAlpha=1;
}

function frame(now){
  const dt=clampN((now-lastTime)/1000,0,.05); lastTime=now;
  if(shakeMag>0.3){ shakeMag*=Math.exp(-dt*7); shakeEl.style.transform='translate('+rand(-1,1)*shakeMag+'px,'+rand(-1,1)*shakeMag+'px)'; }
  else if(shakeMag!==0){ shakeMag=0; shakeEl.style.transform=''; }
  update(dt,now); render(now); drawFX(dt);
  requestAnimationFrame(frame);
}
