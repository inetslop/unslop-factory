 const nb=Array.from({length:ROWS},()=>new Array(COLS).fill(0)); let j=n;
 for(let r=0;r<ROWS;r++) if(!set.has(r)) nb[j++]=board[r];
 board=nb;
 const sh=row=>row+n-rows.filter(i=>i<row).length;
 pac.moving=false; pac.row=sh(pac.row); pac.py=pac.row; pac.px=pac.col;
 for(const g of ghosts) if(g.active) g.gr=sh(g.gr);
 for(const d of dots) d.r=sh(d.r);
 const nl=1+Math.floor(lines/10);
 if(nl>level){ level=nl; SFX.level(); banner('LEVEL '+level,'#4ade80',cell*1.2); }
}

/* ============================== PAC-MAN ============================== */
function makePac(col,row){ return {col,row,px:col,py:row,sx:col,sy:row,stepT:0,moving:false,buf:null,dir:'right',invuln:2500,phase:0,alive:true}; }
function onArrive(){
 const c=pac.col,r=pac.row;
 if(board[r][c]){ board[r][c]=0; score+=10; popup(c,r,'+10','#ffe23d',.8); SFX.block(); }
 else{
  const di=dots.findIndex(d=>d.c===c&&d.r===r);
  if(di>=0){ const d=dots[di]; dots.splice(di,1); score+=5; popup(c,r,'+5','#ffffff',.6); SFX.dot(waka); waka=!waka;
   if(d.kind==='pellet'){ frightT=6000; combo=0; SFX.power(); banner('POWER!','#4dd8ff',cell*1.3); } }
  else{ const idx=piece.cells.findIndex(cc=>piece.x+cc[0]===c&&piece.y+cc[1]===r);
   if(idx>=0){ piece.cells.splice(idx,1); score+=10; popup(c,r,'+10','#ffe23d',.8); SFX.block();
    if(!piece.cells.length){ score+=50; banner('CHOMP! +50','#ffe23d',cell*1.1); spawnNext(); } } }
 }
 collide();
}
function updatePac(dt){
 pac.invuln=Math.max(0,pac.invuln-dt);
 pac.phase+=dt*(pac.moving?0.012:0.004);
 if(!pac.moving){
  const order=[]; if(pac.buf)order.push(pac.buf); if(lastDir&&lastDir!==pac.buf)order.push(lastDir);
  for(const dir of order){ const [dc,dr]=DIRS[dir]; const nc=wrapC(pac.col+dc),nr=pac.row+dr;
   if(dr===-1&&nr<0)continue; if(dr===1&&nr>=ROWS)continue;
   pac.sx=pac.col;pac.sy=pac.row;pac.col=nc;pac.row=nr;pac.dir=dir;lastDir=dir;
   if(Math.abs(nc-pac.sx)>1){ pac.moving=false; pac.px=nc; pac.py=nr; onArrive(); }
   else { pac.moving=true; pac.stepT=0; }
   break; }
 }else{
  pac.stepT+=dt; const k=Math.min(1,pac.stepT/STEP);
  pac.px=pac.sx+(pac.col-pac.sx)*k; pac.py=pac.sy+(pac.row-pac.sy)*k;
  if(k>=1){ pac.moving=false; pac.px=pac.col; pac.py=pac.row; onArrive(); }
 }
}

/* ============================== GHOSTS ============================== */
function makeGhost(i,active){ const s=GDEF[i].s; return {i,c:GDEF[i].c,active,gc:s[0],gr:s[1],gx:s[0],gy:s[1],t:Math.random()*400,tick:Math.random()*200,vr:[1,0],shield:1500+i*250,respawnT:0}; }
const ghostTickBase=()=>Math.max(150,235-(level-1)*9);
function wrapDist(a,b,c){ let d=((b-a)%COLS+COLS)%COLS; if(d>COLS/2)d-=COLS; return d; }
function gtarget(g){
 const p={c:pac.col,r:pac.row};
 if(frightT>0){ /* flee to farthest corner */ let best=null,bd=-1; for(const cc of [0,COLS-1])for(const rr of [0,ROWS-1]){ const d=Math.abs(wrapDist(g.gc,cc,p.c))+Math.abs(rr-g.gr); if(d>bd){bd=d;best={c:cc,r:rr};} } return best; }
 switch(g.i){
  case 0: return p;
  case 1:{ const [dc,dr]=DIRS[pac.dir]; return {c:wrapC(p.col+dc*2),r:p.row+dr*2}; }
  case 2:{ g.jx=(g.jx===undefined||(g.tick>=0&&Math.random()<.5))?Math.round(Math.random()*6-3):g.jx; return {c:p.col+g.jx,r:p.row+(Math.random()<.5?1:-1)}; }
  default:{ const d=Math.abs(wrapDist(g.gc,p.c,p.c))+Math.abs(p.r-g.gr); return d>7?p:{c:0,r:ROWS-1}; }
 }
}
function stepGhost(g){
 const t=gtarget(g),opts=[];
 for(const dir in DIRS){ const [dc,dr]=DIRS[dir]; if(dc===-g.vr[0]&&dr===-g.vr[1])continue; opts.push({dir,nc:wrapC(g.gc+dc),nr:clamp(g.gr+dr,0,ROWS-1)}); }
 let best=null,bv=1e9;
 for(const o of opts){ const d=Math.abs(wrapDist(o.nc,t.c,t.c))+Math.abs(o.nr-t.r); if(d<bv){bv=d;best=o;} }
 if(Math.random()<.15) best=opts[Math.floor(Math.random()*opts.length)];
 g.gc=best.nc; g.gr=best.nr; g.vr=[DIRS[best.dir][0],DIRS[best.dir][1]];
 collide();
}
function updateGhosts(dt){
 const target=Math.min(4,1+Math.floor(lines/5));
 let activeCount=0; for(const g of ghosts)if(g.active)activeCount++;
 for(const g of ghosts){
  if(!g.active){ if(frightT<=0&&g.respawnT>0)g.respawnT-=dt;
   if(frightT<=0&&g.respawnT<=0&&activeCount<target){ g.active=true; g.gc=Math.floor(Math.random()*COLS); g.gr=0; g.gx=g.gc; g.gy=0; g.shield=1500; activeCount++; }
   continue; }
  g.t+=dt; g.shield=Math.max(0,g.shield-dt);
  g.tick+=dt*(frightT>0?1/1.6:1);
  if(g.tick>=ghostTickBase()){ g.tick=0; stepGhost(g); }
  const k=1-Math.exp(-dt/85);
  g.gx+=(g.gc-g.gx)*k; g.gy+=(g.gr-g.gy)*k;
 }
}
function collide(){
 for(const g of ghosts){ if(!g.active)continue;
  const gridHit=g.gc===pac.col&&g.gr===pac.row;
  const pxHit=Math.hypot(g.gx-pac.px,g.gy-pac.py)<.55;
  if(!gridHit&&!pxHit)continue;
  if(frightT>0){ eatGhost(g); }
  else if(g.shield<=0&&pac.invuln<=0){ hurt(); return; }
 }
}
function eatGhost(g){ g.active=false; g.respawnT=6000; const mult=[200,400,800,1600][Math.min(3,combo)]; combo++; score+=mult; popup(g.gc,g.gr,'+'+mult,'#4dd8ff',1.1); poof(g.gc,g.gr,g.c,16); SFX.ghost(); shake(3,200); }
function hurt(){
 lives--; SFX.hurt(); shake(9,500); hitStop=450; frightT=0; combo=0;
 popup(pac.col,pac.row,'CAUGHT!','#ff5c5c',1.2);
 if(lives<=0){ gameOver('CAUGHT BY THE GHOSTS'); return; }
 respawnPac(); resetGhosts(); state='ready'; readyT=1300; SFX.ready();
}
function respawnPac(){
 let c=4,r=ROWS-1; outer: for(let rr=ROWS-1;rr>=Math.max(0,ROWS-6);rr--){ for(const cc of [4,5,3,6,2,7,1,8,0,9]){ if(!board[rr][cc]){c=cc;r=rr;break outer;} } }
 pac.alive=true; pac.col=c;pac.row=r;pac.px=c;pac.py=r;pac.sx=c;pac.sy=r;pac.moving=false;pac.buf=null;pac.invuln=2500; lastDir=null;
}
function resetGhosts(){ for(const g of ghosts){ if(g.active){ const s=GDEF[g.i].s; g.gc=s[0];g.gr=s[1];g.gx=s[0];g.gy=s[1]; } else g.respawnT=Math.min(Math.max(g.respawnT,1),2500); g.shield=1600; } }

 if(state!=='play')return;
 if(hitStop>0){ hitStop-=dt; return; }

 updatePac(dt);
 /* tetris gravity */
 const soft=keys['arrowdown'];

 if(piece.t>=intv){ piece.t=0;
  if(canPlace(piece.cells,piece.x,piece.y+1)){ piece.y++; if(soft)score+=1; if(!grounded())piece.lockT=0; }
 }

 if(clearing){ clearing.t+=dt; if(clearing.t>340)commitClears(); }

}
