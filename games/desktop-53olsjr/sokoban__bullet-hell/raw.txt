function inputVec(){
  let ix=(keys.has('arrowright')||keys.has('d')?1:0)-(keys.has('arrowleft')||keys.has('a')?1:0);
  let iy=(keys.has('arrowdown')||keys.has('s')?1:0)-(keys.has('arrowup')||keys.has('w')?1:0);
  const l=Math.hypot(ix,iy)||1;return{x:ix/l,y:iy/l};
}
function updatePlayer(dt){
  const p=player,v=inputVec();p.moving=!!(v.x||v.y);
  if(!v.x&&!v.y)return;
  const sp=5.4,hh=p.r*0.86;
  p.x+=v.x*sp*dt;collideX(p,v.x>0?1:(v.x<0?-1:0),hh);
  p.y+=v.y*sp*dt;collideY(p,v.y>0?1:(v.y<0?-1:0),hh);
  if(v.x||v.y)p.face=Math.atan2(v.y,v.x);
}
function alignedY(p,cr){return Math.abs(p.y-(cr.r+0.5))<0.42;}
function alignedX(p,cr){return Math.abs(p.x-(cr.c+0.5))<0.42;}
function tryLock(cr){
  if(!cr.locked&&targets.has(cr.c+','+cr.r)){
    cr.locked=true;cr.lockT=0;state.score+=100;state.shake=Math.max(state.shake,4);
    addFloat(cr.c+0.5,cr.r,'+100 SEAL','#ffe066');
    burst(cr.c+0.5,cr.r+0.5,18,'#ffd23f',2.4,0.09);sfx.lock();
    if(lockedCount()===crates.length)startClear();
  }
}
function collideX(p,dir,hh){
  const c0=Math.floor(p.x-hh),c1=Math.floor(p.x+hh),r0=Math.floor(p.y-hh),r1=Math.floor(p.y+hh);
  for(let r=r0;r<=r1;r++)for(let c=c0;c<=c1;c++){
    if(!isWall(c,r)&&!crateAt(c,r))continue;
    const cr=crateAt(c,r);
    if(dir>0&&p.x+hh>c){
      if(cr&&!cr.locked&&alignedY(p,cr)&&freeCell(c+1,r)){cr.c+=1;p.x=cr.c-hh-0.01;sfx.push();burst(cr.c,cr.r+0.5,3,'#c9a26b',1.2,0.05);tryLock(cr);}
      else p.x=c-hh;
    }else if(dir<0&&p.x-hh<c+1){
      if(cr&&!cr.locked&&alignedY(p,cr)&&freeCell(c-1,r)){cr.c-=1;p.x=cr.c+1+hh+0.01;sfx.push();burst(cr.c+1,cr.r+0.5,3,'#c9a26b',1.2,0.05);tryLock(cr);}
      else p.x=c+1+hh;
    }
  }
}
function collideY(p,dir,hh){
  const c0=Math.floor(p.x-hh),c1=Math.floor(p.x+hh),r0=Math.floor(p.y-hh),r1=Math.floor(p.y+hh);
  for(let r=r0;r<=r1;r++)for(let c=c0;c<=c1;c++){
    if(!isWall(c,r)&&!crateAt(c,r))continue;
    const cr=crateAt(c,r);
    if(dir>0&&p.y+hh>r){
      if(cr&&!cr.locked&&alignedX(p,cr)&&freeCell(c,r+1)){cr.r+=1;p.y=cr.r-hh-0.01;sfx.push();burst(cr.c+0.5,cr.r,3,'#c9a26b',1.2,0.05);tryLock(cr);}
      else p.y=r-hh;
    }else if(dir<0&&p.y-hh<r+1){
      if(cr&&!cr.locked&&alignedX(p,cr)&&freeCell(c,r-1)){cr.r-=1;p.y=cr.r+1+hh+0.01;sfx.push();burst(cr.c+0.5,cr.r+1,3,'#c9a26b',1.2,0.05);tryLock(cr);}
      else p.y=r+1+hh;
    }
  }
}

function spawnB(x,y,a,spd,type,r){bullets.push({x:x,y:y,vx:Math.cos(a)*spd,vy:Math.sin(a)*spd,type:type,r:r||0.15,life:9,dodged:false});}
const CHARGE=0.55;
function updEmitters(dt){
  const lvl=state.level,sc=Math.max(0.5,1-lvl*0.1);
  const tx=(state.mode==='menu')?cols/2:player.x,ty=(state.mode==='menu')?rows/2:player.y;
  for(const e of emitters){
    e.t+=dt;
    if(e.phase==='idle'){if(e.t>=e.wait){e.phase='charge';e.t=0;}continue;}
    if(e.phase==='charge'){if(e.t>=CHARGE){e.phase='fire';e.t=0;e.n=0;}continue;}
    const px=e.c+0.5,py=e.r+0.5,ox=px+e.dx*0.42,oy=py+e.dy*0.42;
    if(e.pat==='aimed'){
      if(e.t>=0.13){e.n++;
        if(e.n<3){spawnB(ox,oy,Math.atan2(ty-oy,tx-ox)+(e.n-1)*0.22,4.6+lvl*0.4,'aimed');e.t=0;}
        else{e.phase='idle';e.wait=(1.5+Math.random()*1.2)*sc;e.t=0;}}
    }else if(e.pat==='ring'){
      if(e.n===0){const N=9+lvl*2;
        for(let i=0;i<N;i++)spawnB(ox,oy,(i/N)*6.283+e.ang,3.5+lvl*0.3,'ring');
        e.phase='idle';e.wait=(2.4+Math.random()*1.2)*sc;e.t=0;}
    }else if(e.pat==='spiral'){
      if(e.n===0)e.a=Math.random()*6.283;
      if(e.t>=0.085){spawnB(ox,oy,e.a,3.1+lvl*0.2,'spiral');e.a+=0.47;e.n++;e.t=0;
        if(e.n>26){e.phase='idle';e.wait=(2.2+Math.random()*1.2)*sc;}}
    }else if(e.pat==='stream'){
      if(e.t>=0.12){spawnB(ox,oy,Math.atan2(ty-oy,tx-ox)+Math.sin(e.n*0.9)*0.3,5.0+lvl*0.3,'stream',0.12);
        e.n++;e.t=0;if(e.n>12){e.phase='idle';e.wait=(1.8+Math.random()*1.2)*sc;}}
    }else if(e.pat==='seeker'){
      if(e.n===0){spawnB(ox,oy,Math.atan2(ty-oy,tx-ox),2.7,'seeker',0.2);e.n++;
        e.phase='idle';e.wait=(2.6+Math.random()*1.2)*sc;e.t=0;}
    }
  }
}
function hitPlayer(){
  if(state.mode!=='play'||player.invuln>0)return;
  state.hearts--;player.invuln=2;state.shake=16;state.flash=0.55;state.flashCol='#ff4d6d';state.hitStop=0.09;state.combo=0;
  burst(player.x,player.y,22,'#ff8ba3',3,0.09);ring(player.x,player.y,0.15,1.6,'#ff4d6d');sfx.hit();
  if(state.hearts<=0)die();
}
function updBullets(dt){
  const p=player;
  const tx=(state.mode==='menu')?cols/2:p.x,ty=(state.mode==='menu')?rows/2:p.y;
  for(let i=bullets.length-1;i>=0;i--){
    const b=bullets[i];
    if(b.type==='seeker'){
      const cur=Math.atan2(b.vy,b.vx),des=Math.atan2(ty-b.y,tx-b.x);
      let d=des-cur;while(d>Math.PI)d-=6.283;while(d<-Math.PI)d+=6.283;
      const m=2.6*dt,a=Math.abs(d)<m?des:cur+Math.sign(d)*m,spd=Math.hypot(b.vx,b.vy);
      b.vx=Math.cos(a)*spd;b.vy=Math.sin(a)*spd;
    }
    b.x+=b.vx*dt;b.y+=b.vy*dt;b.life-=dt;
    let dead=b.life<=0||isWall(Math.floor(b.x),Math.floor(b.y));
    if(!dead)for(const cr of crates){
      if(b.x>cr.c+0.06&&b.x<cr.c+0.94&&b.y>cr.r+0.06&&b.y<cr.r+0.94){dead=true;burst(b.x,b.y,3,'#ffb75d',1.6,0.05);break;}
    }
    if(!dead&&state.mode==='play'){
      const dx=b.x-p.x,dy=b.y-p.y,d=Math.hypot(dx,dy);
      if(d<p.r*0.85+b.r){hitPlayer();dead=true;burst(b.x,b.y,6,'#ff4d6d',2,0.07);}
      else if(!b.dodged&&d<p.r+b.r+0.34){
        b.dodged=true;state.combo=Math.min(state.combo+1,8);state.comboT=2.2;
        const pts=10+(state.combo-1)*5;state.score+=pts;
        addFloat(p.x,p.y-0.6,'+'+pts,'#7df9ff');sfx.graze();
      }
    }
    if(dead){if(isWall(Math.floor(b.x),Math.floor(b.y)))burst(b.x,b.y,2,'#9adfff',1.4,0.04);bullets.splice(i,1);}
  }
}

/* ============================== UPDATE ============================== */
function updFx(raw,dt){
  for(let i=parts.length-1;i>=0;i--){const p=parts[i];
    p.life-=raw;if(p.life<=0){parts.splice(i,1);continue;}
    if(!p.ring){if(p.grav)p.vy+=p.grav*dt;p.x+=p.vx*dt;p.y+=p.vy*dt;
      const dr=Math.pow(p.drag||0.9,dt*60);p.vx*=dr;p.vy*=dr;}}
  for(let i=floats.length-1;i>=0;i--){const f=floats[i];f.t+=raw;if(f.t>f.life)floats.splice(i,1);}
  state.shake+=(0-state.shake)*Math.min(1,raw*6);if(state.shake<0.08)state.shake=0;
  state.flash+=(0-state.flash)*Math.min(1,raw*5);
}
function update(raw){
  const dt=raw*state.slowmo;
  if(state.comboT>0){state.comboT-=raw;if(state.comboT<=0)state.combo=0;}
  for(const c of crates)c.lockT+=raw;
  switch(state.mode){
    case 'play':{
      state.time+=dt;state.totalTime+=dt;
      updatePlayer(dt);
      if(player.invuln>0)player.invuln-=dt;
      updEmitters(dt);updBullets(dt);updFx(raw,dt);break;}
    case 'menu':{updEmitters(dt*0.8);updBullets(dt*0.8);updFx(raw,dt);break;}
    case 'clear':{
      state.clearT+=raw;updFx(raw,dt);
      if(state.clearT>2.1){
        if(state.level+1>=LEVELS.length)winGame();
        else{loadLevel(state.level+1);state.mode='play';}
      }break;}
    case 'dying':{
      state.deathT+=raw;
      if(state.deathT>0.5)state.slowmo=Math.min(1,state.slowmo+raw*2);
      updFx(raw,dt);
      if(state.deathT>1.35){state.slowmo=1;finalizeOver();}break;}
  }
}

    const bt=state.time,a=bt<0.25?bt/0.25:(bt>1.4?(1.8-bt)/0.4:1);
    const cx=BX+cols*T/2,cy=BY+rows*T*0.42;
    ctx.save();ctx.globalAlpha=a;ctx.textAlign='center';
    ctx.strokeStyle='rgba(255,210,63,.7)';ctx.lineWidth=2;
    const w=90+14*Math.sin(bt*3);
    ctx.beginPath();ctx.moveTo(cx-160-w,cy-8);ctx.lineTo(cx-90,cy-8);ctx.moveTo(cx+90,cy-8);ctx.lineTo(cx+160+w,cy-8);ctx.stroke();
    ctx.font='42px '+DISP;ctx.save();ctx.shadowColor='rgba(255,210,63,.7)';ctx.shadowBlur=16;
    ctx.fillStyle='#ffd23f';ctx.fillText('LEVEL '+(state.level+1),cx,cy);ctx.restore();
    ctx.font='14px '+MONO;ctx.fillStyle='#7df9ff';
    ctx.fillText(L EVELS[state.level].name,cx,cy+26);
    ctx.restore();ctx.textAlign='left';
  }
}
function drawClear(t){
  const ct=state.clearT;
  ctx.fillStyle='rgba(8,12,26,'+(Math.min(0.45,ct)*0.9)+')';ctx.fillRect(BX-5,BY-5,cols*T+10,rows*T+10);
  const cx=BX+cols*T/2,cy=BY+rows*T*0.44;
  const sc=1+0.3*(1-Math.min(1,ct/0.4));
  ctx.save();ctx.translate(cx,cy);ctx.scale(sc,sc);ctx.textAlign='center';
  ctx.font='46px '+DISP;ctx.save();ctx.shadowColor='rgba(53,224,255,.8)';ctx.shadowBlur=18;
  ctx.fillStyle='#35e0ff';ctx.fillText('SECTOR CLEAR',0,0);ctx.restore();ctx.restore();
  if(state.clearInfo){
    const lines=['ALL CARGO SEALED','#SHIELD BONUS +'+state.clearInfo.shield,'#TIME BONUS +'+state.clearInfo.time];
    ctx.textAlign='center';ctx.font='15px '+MONO;
    lines.forEach(function(ln,i){
      if(ct>0.4+i*0.25){const a=Math.min(1,(ct-0.4-i*0.25)*4);
        ctx.globalAlpha=a;
        ctx.fillStyle=i===0?'#cfe3ff':(i===1?'#ff8ba3':'#ffd23f');
        ctx.fillText(ln,cx,cy+34+i*24);}
    });ctx.globalAlpha=1;ctx.textAlign='left';
  }
}
function render(t){
  ctx.clearRect(0,0,W,H);
  const g=ctx.createLinearGradient(0,0,0,H);
  g.addColorStop(0,'#0b1024');g.addColorStop(0.55,'#101736');g.addColorStop(1,'#1a1030');
  ctx.fillStyle=g;ctx.fillRect(0,0,W,H);
  const hue=(t*14)%360,bcx=BX+cols*T/2,bcy=BY+rows*T/2;
  const rg=ctx.createRadialGradient(bcx,bcy,T*0.5,bcx,bcy,Math.max(W,H)*0.7);
  rg.addColorStop(0,'hsla('+hue+',70%,60%,0.10)');rg.addColorStop(1,'rgba(0,0,0,0)');
  ctx.fillStyle=rg;ctx.fillRect(0,0,W,H);
  for(const d of dust){
    const y=((d.y-t*d.v)%H+H)%H,x=d.x+Math.sin(t*0.3+d.x)*8;
    ctx.globalAlpha=d.a;ctx.fillStyle='#8fb4ff';
    ctx.beginPath();ctx.arc(x,y,d.r,0,6.283);ctx.fill();
  }
  ctx.globalAlpha=1;
  const sx=(Math.random()*2-1)*state.shake,sy=(Math.random()*2-1)*state.shake;
  ctx.save();ctx.translate(sx,sy);
  drawBoard(t);
  for(const c of crates)drawCrate(c,t);
  if(state.mode!=='menu')drawPlayer(t);
  drawEmitters(t);drawBullets();drawParts();drawFloats();
  ctx.restore();
  drawHUD(t);
  drawBanner(t);
  if(state.mode==='clear')drawClear(t);
  if(state.flash>0.01){ctx.globalAlpha=state.flash*0.5;ctx.fillStyle=state.flashCol;ctx.fillRect(0,0,W,H);ctx.globalAlpha=1;}
}

let last=performance.now();
function frame(now){
  requestAnimationFrame(frame);
  let raw=(now-last)/1000;last=now;if(raw>0.1)raw=0.1;
  if(state.hitStop>0){state.hitStop-=raw;raw=0;}
  update(raw);render(now/1000);
}
