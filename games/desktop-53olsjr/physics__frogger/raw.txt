function burstSparkle(x,y){ for(let i=0;i<14;i++){const a=R(0,TAU),s=R(40,170);part({x,y,vx:Math.cos(a)*s,vy:Math.sin(a)*s-60,g:300,size:R(1.5,3),col:'#ffe14d',life:R(.4,.8)});} }
function confetti(){ const cols=['#ff5b5b','#ffd44f','#5aa8ff','#c07bff','#4fd6c2','#b6f34d']; for(let i=0;i<70;i++)part({x:R(80,W-80),y:R(-20,120),vx:R(-90,90),vy:R(-40,120),g:420,size:R(4,7),col:cols[i%6],type:'r',rot:R(0,TAU),vr:R(-9,9),life:R(1.2,1.9)}); }
function addRipple(x,y,r){ ripples.push({x,y,r:r||4,t:0}); }
function flash(col,a){ G.flash={col,a}; }

function respawnFrog(){
  F.x=SPAWN.x; F.y=SPAWN.y; F.vx=F.vy=0; F.z=F.pz=F.vz=0; F.riding=null; F.pad=null;
  F.ang=-PI/2; F.pop=0; F.hopping=false; F.hopT=0; F.inWater=false; F.squash=0;
  G.timeLeft=45; G.lastTick=99;
}
function startGame(){
  G.score=0; G.lives=3; G.wave=1; G.mult=1; G.homesTotal=0;
  homes=[false,false,false,false,false];
  makeCars(); regenPads();
  parts.length=0; ripples.length=0; decals.length=0; popups.length=0; ragdoll=null;
  respawnFrog(); G.mode='playing';
  banner('WAVE 1','reach the five homes!',1.6);
}
function applyWaveUp(){
  G.wave++; G.mult=Math.min(2.4,G.mult*1.18);
  homes=[false,false,false,false,false]; regenPads(); respawnFrog();
  banner('WAVE '+G.wave,'traffic is speeding up…',1.4);
}
function releaseRide(vx){ F.riding=null; F.vz=Math.max(F.vz,60); F.vx=vx+Math.sign(F.relX||1)*30; burstDust(F.x,F.y); }
function killFrog(car,msg){
  G.lives--;
  const vx=car? car.vx*1.2+(Math.sign(F.x-car.x)||1)*80 : R(-60,60);
  ragdoll={x:F.x,y:F.y,vx,vy:R(-70,-20),z:F.z+4,vz:330,rot:R(0,TAU),vr:R(-13,13),life:1.6};
  decals.push({x:F.x,y:F.y,seed:Math.random()*9,t:0});
  burstGoo(F.x,F.y); if(F.inWater){addRipple(F.x,F.y,8);burstDroplets(F.x,F.y,14);} 
  G.shake=16; flash('255,70,55',0.45); sfx.splat();
  G.deathT=1.5; G.mode='dying'; F.hopping=false;
  if(msg)banner(msg,'',0.9);
}
function reachHome(i){

  popup(HX[i],HY-34,'+100'); sfx.home(); flash('120,255,140',0.16); burstSparkle(HX[i],HY);
  F.hopping=true; F.hopT=0; G.respT=0.55;

    confetti(); sfx.wave(); banner('WAVE CLEARED!','+250 bonus',1.6);

  }
}

function inputVec(){
  let x=0,y=0;
  if(keys['ArrowLeft']||keys['KeyA'])x-=1; if(keys['ArrowRight']||keys['KeyD'])x+=1;
  if(keys['ArrowUp']||keys['KeyW'])y-=1;   if(keys['ArrowDown']||keys['KeyS'])y+=1;
  if(joy.active){x+=joy.ix;y+=joy.iy;}
  const l=Math.hypot(x,y); if(l>1){x/=l;y/=l;}
  return {x,y};
}

function updateCars(dt){
  for(const car of cars){
    const L=CAR_LANES[car.lane];
    car.vx=L.dir*L.sp*G.mult; car.x+=car.vx*dt;
    if(car.x> W/2+170){ if(F.riding===car)releaseRide(car.vx); car.x=-W/2-170; }
    else if(car.x<-W/2-170){ if(F.riding===car)releaseRide(car.vx); car.x= W/2+170; }
  }
}
function carCollide(){
  for(const car of cars){
    const L=CAR_LANES[car.lane];
    if(Math.abs(F.y-L.c)>L.h/2+16)continue;
    const ax=Math.abs(F.x-car.x);
    if(ax<car.w/2+14&&Math.abs(F.y-L.c)<L.h/2+13){
      if(F.z>=car.top-4){
        if(F.vz<0&&F.pz>car.top&&F.z<=car.top+2){
          if(ax<car.w/2-6){
            F.riding=car; F.relX=clamp(F.x-car.x,-car.w/2+9,car.w/2-9);
            F.z=car.top; F.vz=0; sfx.land(); burstDust(F.x,F.y);
            if(!tips.ride){tips.ride=true;banner('ROOF RIDE!','steer to slide off · SPACE to hop',1.5);}
          } else killFrog(car);
        }
      } else killFrog(car);
    }
    if(F.riding)break;
  }
}
function frogPhysics(dt){
  const inp=inputVec();
  let surf=surfaceAt(F.y); F.pad=null;
  if(surf==='river'){
    if(F.z<10){ const p=padUnder(F.x,F.y); if(p){surf='pad';F.pad=p;} else surf='water'; }
    else surf='water';
  }
  const P=SURF[surf];
  F.vx+=inp.x*P.A*dt; F.vy+=inp.y*P.A*dt;
  const f=Math.exp(-P.K*dt); F.vx*=f; F.vy*=f;
  if(surf==='ice'){ const f2=Math.exp(-0.15*dt); F.vx*=f2; F.vy*=f2;
    if(Math.hypot(F.vx,F.vy)>300&&Math.random()<dt*18)part({x:F.x,y:F.y+6,vx:R(-20,20),vy:R(10,40),size:R(2,4),col:'#eaf7ff',life:.5}); }
  if(surf==='water'){ F.vy+=P.cur*dt; F.swimT-=dt;
    if(F.swimT<=0){F.swimT=0.13; addRipple(F.x,F.y,6);} }
  const sp=Math.hypot(F.vx,F.vy); if(sp>900){F.vx*=900/sp;F.vy*=900/sp;}

  // jump (from ground or roof)
  if(jumpQ&&(F.z<=0.5||F.riding)){
    if(F.riding)releaseRide(CAR_LANES[F.riding.lane].dir*CAR_LANES[F.riding.lane].sp*G.mult*1.06);
    F.vz=JV; F.z=Math.max(F.z,0.5); sfx.jump(); burstDust(F.x,F.y+8);
  }

  if(F.riding){
    const lim=F.riding.w/2-9;
    F.relX+=inp.x*55*dt;
    if(Math.abs(F.relX)>lim)releaseRide(CAR_LANES[F.riding.lane].dir*CAR_LANES[F.riding.lane].sp*G.mult*1.02);
    else { F.riding.vx=CAR_LANES[F.riding.lane].dir*CAR_LANES[F.riding.lane].sp*G.mult;
           F.x=F.riding.x+F.relX; F.z=F.riding.top; F.vx=F.riding.vx; }
  } else {
    F.x+=F.vx*dt; F.y+=F.vy*dt;
    if(F.pad)F.x+=F.pad.dx;
    if(F.x<16){F.x=16;F.vx*=-0.2;} if(F.x>W-16){F.x=W-16;F.vx*=-0.2;}
    if(F.y<18){F.y=18;F.vy*=-0.2;} if(F.y>H-16){F.y=H-16;F.vy*=-0.2;}
    F.pz=F.z;
    const g=(jumpHeld()&&F.vz>0)?GHOLD:GFULL;
    F.vz-=g*dt; F.z+=F.vz*dt;
    // pad snap while descending
    if(F.z<=12&&F.vz<0&&F.y>=142&&F.y<=500){
      const p=padUnder(F.x,F.y);
      if(p){F.z=0;F.vz=0;F.pad=p;sfx.land();addRipple(p.x,p.y,7);F.squash=1;}
    }
    if(F.z<=0){
      if(F.pz>8){ // landed from air
        F.squash=1; sfx.land();
        if(surf==='water'){burstDroplets(F.x,F.y,10);} else burstDust(F.x,F.y);
        if(F.vz<-620)G.shake=Math.max(G.shake,4);
      }
      F.z=0;F.vz=0;
    }
  }
  F.squash=Math.max(0,F.squash-dt*5.5);
  const spd=Math.hypot(F.vx,F.vy);
  if(spd>30)F.ang=angLerp(F.ang,Math.atan2(F.vy,F.vx),Math.min(1,9*dt));
  F.phase+=spd*dt*0.085;
  F.blink-=dt; if(F.blink<0)F.blink=R(2.2,4.2);
  // splash on water entry
  const inW=(surf==='water');
  if(inW&&!F.inWater){ addRipple(F.x,F.y,10); burstDroplets(F.x,F.y,16); sfx.splash();
    if(!tips.current){tips.current=true;banner('THE CURRENT PULLS!','swim against it — or leap the pads',1.6);} }
  F.inWater=inW;

  if(!F.riding)carCollide();
  // homes
  if(F.z<20){
    for(let i=0;i<5;i++){
      if(!homes[i]&&Math.hypot(F.x-HX[i],F.y-HY)<26){ reachHome(i); break; }
    }
  }
}
function update(dt){
  G.t+=dt; const t=G.t;
  musicTick();
  G.shake*=Math.exp(-5*dt); if(G.shake<0.1)G.shake=0;
  if(G.flash){G.flash.a-=dt*1.1; if(G.flash.a<=0)G.flash=null;}
  if(G.banner){G.banner.t+=dt; if(G.banner.t>G.banner.dur)G.banner=null;}
  for(let i=popups.length-1;i>=0;i--){popups[i].t+=dt*1.1; if(popups[i].t>1)popups.splice(i,1);}
  for(let i=parts.length-1;i>=0;i--){const p=parts[i];p.t+=dt;p.vy+=p.g*dt;p.x+=p.vx*dt;p.y+=p.vy*dt;p.rot+=p.vr*dt; if(p.t>p.life)parts.splice(i,1);}
  for(let i=ripples.length-1;i>=0;i--){const r=ripples[i];r.t+=dt;r.r+=46*dt; if(r.t>0.9)ripples.splice(i,1);}
  for(let i=decals.length-1;i>=0;i--){decals[i].t+=dt; if(decals[i].t>2.5)decals.splice(i,1);}

  updateCars(dt); updatePads(dt);

  if(G.mode==='menu'||G.mode==='paused')return;

  if(G.mode==='dying'){
    if(ragdoll){const r=ragdoll;r.life-=dt;r.vz-=1500*dt;r.z+=r.vz*dt;
      r.x+=r.vx*dt;r.y+=r.vy*dt;r.rot+=r.vr*dt;
      const ff=Math.exp(-1.2*dt);r.vx*=ff;r.vy*=ff;}
    G.deathT-=dt;
    if(G.deathT<=0){ if(G.lives>0){respawnFrog();G.mode='playing';} else gameOver(); }
    return;
  }
  // playing
  F.pop=Math.min(1,F.pop+dt*3.2);
  if(G.postT>0){ G.postT-=dt; if(G.postT<=0)applyWaveUp(); return; }
  if(F.hopping){
    F.hopT+=dt/0.45;
    if(G.respT>0){G.respT-=dt; if(G.respT<=0&&G.postT<=0)respawnFrog();}
    return;
  }
  frogPhysics(dt);
  G.timeLeft-=dt;
  const sec=Math.ceil(G.timeLeft);
  if(G.timeLeft<6&&sec!==G.lastTick){G.lastTick=sec;sfx.tick();}
  if(G.timeLeft<=0)killFrog(null,'TOO SLOW!');
}

  ctx.font=(weight||'700')+' '+size+'px '+(font||MONO);
  ctx.textAlign=align||'center'; ctx.textBaseline='middle';
  ctx.fillStyle=col||'#fff'; ctx.fillText(s,x,y);
  ctx.globalAlpha=1;
}
function drawFrog(x,y,ang,o){
  o=o||{}; const s=o.s||1, sx=(o.sx||1)*s, sy=(o.sy||1)*s;
  const phase=o.phase||0, blink=o.blink, dead=o.dead;
  ctx.save(); ctx.globalAlpha*=(o.alpha==null?1:o.alpha);
  ctx.translate(x,y); ctx.rotate(ang+PI/2); ctx.scale(sx,sy);
  const sinP=Math.sin(phase);
  ctx.fillStyle='#2f8f3c'; ctx.strokeStyle='#1d5c22'; ctx.lineWidth=1;
  function foot(fx,fy,rx,ry,rot){ctx.save();ctx.translate(fx,fy);ctx.rotate(rot);ctx.beginPath();ctx.ellipse(0,0,rx,ry,0,0,TAU);ctx.fill();ctx.stroke();ctx.restore();}
  foot(-12,10+sinP*5,6,4,-0.7); foot(12,10-sinP*5,6,4,0.7);
  foot(-9,-6+sinP*Math.PI*0+Math.sin(phase+PI)*3,5,3.5,-0.5); foot(9,-6-Math.sin(phase+PI)*3,5,3.5,0.5);
  const g=ctx.createRadialGradient(-4,-5,3,0,0,17);
  g.addColorStop(0,'#9bea63'); g.addColorStop(1,'#3fae3f');
  ctx.fillStyle=g; ctx.beginPath(); ctx.arc(0,0,16,0,TAU); ctx.fill();
  ctx.strokeStyle='#1c5a22'; ctx.lineWidth=2; ctx.stroke();
  ctx.fillStyle='rgba(43,125,51,.3)';
  [[-5,3],[4,7],[-9,-2],[8,-4]].forEach(p=>{ctx.beginPath();ctx.arc(p[0],p[1],2,0,TAU);ctx.fill();});
  ctx.fillStyle='rgba(205,235,142,.65)'; ctx.beginPath(); ctx.ellipse(0,-6,7,8,0,0,TAU); ctx.fill();
  // eyes
  [-7.5,7.5].forEach(ex=>{
    if(blink||dead){
      if(dead){ctx.strokeStyle='#14250f';ctx.lineWidth=2.2;
        ctx.beginPath();ctx.moveTo(ex-3,-15);ctx.lineTo(ex+3,-9);ctx.moveTo(ex+3,-15);ctx.lineTo(ex-3,-9);ctx.stroke();}
      else {ctx.strokeStyle='#1c5a22';ctx.lineWidth=2;ctx.beginPath();ctx.moveTo(ex-4,-12);ctx.lineTo(ex+4,-12);ctx.stroke();}
    } else {
      ctx.fillStyle='#fff'; ctx.beginPath(); ctx.arc(ex,-12,5.5,0,TAU); ctx.fill();
      ctx.strokeStyle='#1c5a22'; ctx.lineWidth=1.5; ctx.stroke();
      ctx.fillStyle='#14250f'; ctx.beginPath(); ctx.arc(ex+((o.pd||0)*1.6),-13.8,2.6,0,TAU); ctx.fill();
      ctx.fillStyle='#fff'; ctx.beginPath(); ctx.arc(ex+((o.pd||0)*1.6)+0.9,-14.6,0.9,0,TAU); ctx.fill();
    }
  });
  ctx.restore();
}
