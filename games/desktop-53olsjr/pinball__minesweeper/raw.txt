function gravity(){ return 1400 + Math.min((level-1)*70, 280); }

function updateFlipper(f, key, isL, dt){
  const target = key ? FLIP.act : FLIP.rest;
  const base = isL ? target : (Math.PI + (key?FLIP.act:FLIP.rest) - (key?-FLIP.rest:-FLIP.rest)); // placeholder, set explicitly below
  let tA = isL ? (key?FLIP.act:FLIP.rest) : (key ? Math.PI+FLIP.act*1*-1+Math.PI*0 + (Math.PI-FLIP.rest) - FLIP.rest - FLIP.rest + FLIP.rest : 0);
  // explicit, unambiguous targets:
  tA = isL ? (key?FLIP.act:FLIP.rest) : (key ? Math.PI - FLIP.act : Math.PI - FLIP.rest);
  const rate = key ? FLIP.upRate : FLIP.dnRate;
  const prev=f.a;
  if(Math.abs(tA-f.a)<1e-4) f.a=tA;
  else {
    const step=rate*dt*(tA>f.a?1:-1);
    f.a = clamp(f.a+step, Math.min(f.a,tA), Math.max(f.a,tA));
  }
  f.omega=(f.a-prev)/dt;
  if(key && (isL?keys.l:keys.r) && !f._snd){ f._snd=true; S.flip(); }
  if(!key) f._snd=false;
}
function flipperGeom(f, isL){
  const px = isL?FLIP.pl:FLIP.pr, py=FLIP.y;
  return {px,py,qx:px+Math.cos(f.a)*FLIP.len, qy:py+Math.sin(f.a)*FLIP.len};
}
function collideFlipper(b,f,isL){
  const g=flipperGeom(f,isL);
  const dx=g.qx-g.px, dy=g.g?0:g.qy-g.py;
  const L2=dx*dx+dy*dy;
  let t=((b.x-g.px)*dx+(b.y-g.py)*dy)/L2; t=clamp(t,0,1);
  const cx=g.px+t*dx, cy=g.py+t*dy;
  let nx=b.x-cx, ny=b.y-cy; const d=Math.hypot(nx,ny), R=b.r+FLIP.r;
  if(d>=R||d===0) return;
  nx/=d; ny/=d;
  b.x+=nx*(R-d); b.y+=ny*(R-d);
  // tip velocity = len*ω*(-sin a, cos a); contact fraction t
  const tvx=FLIP.len*f.omega*(-Math.sin(f.a))*t, tvy=FLIP.len*f.omega*Math.cos(f.a)*t;
  const rvx=b.vx-tvx, rvy=b.vy-tvy, vn=rvx*nx+rvy*ny;
  if(vn<0){
    const e=Math.abs(vn)>140?0.45:0.12;
    b.vx-=(1+e)*vn*nx; b.vy-=(1+e)*vn*ny;
    if(Math.abs(vn)>260) S.bounce(Math.abs(vn));
  }
}

function physics(dt){
  const g=gravity();
  const steps=Math.max(1,Math.ceil(dt/(1/240)));
  const h=dt/steps;
  pendingHit.length=0;
  for(let s=0;s<steps;s++){
    const b=ball;
    if(b.state==='live'){
      b.vy+=g*h;
      const damp=Math.pow(0.9985,h*60); b.vx*=damp; b.vy*=damp;
      b.x+=b.vx*h; b.y+=b.vy*h;

      // walls
      let imp=segCollide(b, WALL.L,WALL.T, WALL.R,WALL.T, .5);
      imp=Math.max(imp, segCollide(b, WALL.L,WALL.T, WALL.L,600, .5));
      imp=Math.max(imp, segCollide(b, WALL.R,WALL.T, WALL.R,WALL.BOT+40, .5));
      imp=Math.max(imp, segCollide(b, 24,600, 150,700, .5));        // left inlane funnel
      imp=Math.max(imp, segCollide(b, 536,600, 410,700, .5));       // right inlane funnel
      imp=Math.max(imp, segCollide(b, 536,134, DIVX,200, .6));      // launch deflector
      if(b.x < DIVX-2) imp=Math.max(imp, segCollide(b, DIVX,DIV_TOP, DIVX,DIV_BOT, .5)); // one-way divider
      if(imp>150) S.bounce(imp);

      // bumpers (cherry bombs)
      for(const bp of BUMPERS){
        let nx=b.x-bp.x, ny=b.y-bp.y; const d=Math.hypot(nx,ny), R=b.r+bp.r;
        if(d<R&&d>0){
          nx/=d; ny/=d; b.x=bp.x+nx*R; b.y=bp.y+ny*R;
          const vn=b.vx*nx+b.vy*ny;
          if(vn<0){ b.vx-=1.9*vn*nx; b.vy-=1.9*vn*ny; }
          const cur=b.vx*nx+b.vy*ny;
          if(cur<620){ b.vx+=nx*(620-cur); b.vy+=ny*(620-cur); }
          if(bp.t<=0){ bp.t=.18; score+=150; S.bump(); pushScorePop(bp.x,bp.y-24,'+150','#ffd94d');
            for(let k=0;k<7;k++) particles.push({type:'spark',x:b.x,y:b.y,vx:rnd(-160,160),vy:rnd(-160,160),life:.3,max:.3,size:2.5,color:'#ffd94d'});
          }
        }
      }

      // hidden tiles (mines bounce + detonate; safe tiles open as holes)
      for(let i=0;i<cells.length;i++){
        const c=cells[i]; if(c.state!=='hidden') continue;
        const p=cellXY(i), R=b.r+17;
        let nx=b.x-p.x, ny=b.y-p.y; const d=Math.hypot(nx,ny);
        if(d<R&&d>0){
          if(c.mine){
            nx/=d; ny/=d; b.x=p.x+nx*R; b.y=p.y+ny*R;
            const vn=b.vx*nx+b.vy*ny;
            if(vn<0){ const e=Math.abs(vn)>140?.5:0; b.vx-=(1+e)*vn*nx; b.vy-=(1+e)*vn*ny; }
            pendingHit.push(i);
          } else {
            pendingHit.push(i); // pass through — dig a hole
          }
        }
      }

      collideFlipper(b, flipL, true);
      collideFlipper(b, flipR, false);

      const sp=Math.hypot(b.vx,b.vy);
      if(sp>1500){ b.vx*=1500/sp; b.vy*=1500/sp; }

      // stuck detection → auto nudge
      if(sp<32) stuckT+=h; else stuckT=0;
      if(stuckT>2.4){ stuckT=0; nudge(); }

      if(b.y-b.r>CABH+6) loseBall();
    }
  }
  // process board triggers after the step (safe mutation order)
  for(const i of pendingHit){ if(cells[i].state==='hidden') triggerCell(i); }
  processBoard();
  if(hiddenTotal===0 && winT<=0 && mode==='play' && ball.state!=='draining') levelClear();
}

/* ============================ UPDATE ============================ */
function update(dt){
  time+=dt;
  for(const bp of BUMPERS) bp.t=Math.max(0,bp.t-dt);
  shake*=Math.exp(-6*dt); if(shake<.05)shake=0;
  flash=Math.max(0,flash-2.2*dt);
  faceHold=Math.max(0,faceHold-dt);

  // toasts / booms / particles always animate
  for(let i=toasts.length-1;i>=0;i--){ toasts[i].t-=dt; if(toasts[i].t<=0) toasts.splice(i,1); }
  for(let i=booms.length-1;i>=0;i--){
    const bm=booms[i]; bm.t-=dt;
    if(bm.t<=0){
      S.boom(bm.chain); flash=Math.max(flash,bm.chain?.28:.45); shake=Math.max(shake,bm.chain?9:15);
      faceMood='ouch'; faceHold=.5;
      for(let k=0;k<16;k++){ const a=rnd(0,Math.PI*2), v=rnd(120,420);
        particles.push({type:'spark',x:bm.x,y:bm.y,vx:Math.cos(a)*v,vy:Math.sin(a)*v-80,life:rnd(.35,.7),max:.7,size:rnd(2,4),
          color:['#ffd94d','#ff9d3c','#ff5b4d'][(Math.random()*3)|0]}); }
      for(let k=0;k<6;k++) particles.push({type:'smoke',x:bm.x+rnd(-8,8),y:bm.y+rnd(-8,8),vx:rnd(-25,25),vy:rnd(-70,-30),life:rnd(.6,1.1),max:1.1,size:rnd(6,11),color:'#9aa0a8'});
      particles.push({type:'ring',x:bm.x,y:bm.y,life:.4,max:.4,size:10,color:bm.chain?'#ffd94d':'#ff5b4d'});
    }
  }
  for(let i=particles.length-1;i>=0;i--){
    const p=particles[i]; p.life-=dt;
    if(p.life<=0){ particles.splice(i,1); continue; }
    if(p.type==='spark'){ p.vy+=900*dt; p.x+=p.vx*dt; p.y+=p.vy*dt; }
    else if(p.type==='smoke'){ p.x+=p.vx*dt; p.y+=p.vy*dt; p.size+=14*dt; }
    else if(p.type==='confetti'){ p.vy+=260*dt; p.x+=p.vx*dt; p.y+=p.vy*dt; p.rot+=p.vr*dt; }
    else if(p.type==='txt'){ p.x+=p.vx*dt; p.y+=p.vy*dt; }
  }
  if(particles.length>420) particles.splice(0,particles.length-420);

  // tile pop anims decay
  for(const c of cells) if(c.anim>0) c.anim=Math.max(0,c.anim-dt*3.2);

  if(mode!=='play'||paused) return;

  updateFlipper(flipL, keys.l, true, dt);
  updateFlipper(flipR, keys.r, false, dt);

  if(winT>0){
    winT-=dt;
    if(winT<=0){
      level++; balls=Math.min(3,balls+1); makeBoard(); resetBall();
      addToast('LEVEL '+level+' — MORE MINES', '#ffd94d', 2.0);
    }
    return;
  }
  if(drainT>0){
    drainT-=dt;
    if(drainT<=0){
      balls--;
      if(balls>0){ resetBall(); addToast('BALL LOST — '+balls+' REMAIN', '#ff6b5e', 1.4); }
      else gameOver();
    }
    return;
  }

  if(charging) charge=Math.min(1,charge+dt*1.15);

  physics(dt);

  if(ball.state==='live'){
    ball.trail.push({x:ball.x,y:ball.y});
    if(ball.trail.length>12) ball.trail.shift();
  } else ball.trail.length=0;

  // face mood (transient overrides)
  if(faceHold<=0){
    const sp=Math.hypot(ball.vx,ball.vy);
    faceMood = (ball.state==='live' && sp>1050) ? 'wow' : 'ok';
  }
}

let last=performance.now(), acc=0;
function frame(now){
  requestAnimationFrame(frame);
  let dt=(now-last)/1000; last=now;
  if(dt>0.05) dt=0.05;
  if(!paused && mode==='play') update(dt);
  else { // idle animation only
    time+=dt;
    for(const bp of BUMPERS) bp.t=Math.max(0,bp.t-dt);
    shake*=Math.exp(-6*dt); flash=Math.max(0,flash-2.2*dt);
    faceHold=Math.max(0,faceHold-dt);
    for(let i=particles.length-1;i>=0;i--){ const p=particles[i]; p.life-=dt;
      if(p.life<=0) particles.splice(i,1);
      else { if(p.type==='spark'){p.vy+=900*dt;p.x+=p.vx*dt;p.y+=p.vy*dt;}
             if(p.type==='smoke'){p.x+=p.vx*dt;p.y+=p.vy*dt;p.size+=14*dt;} } }
    for(const c of cells) if(c.anim>0) c.anim=Math.max(0,c.anim-dt*3.2);
  }
  render();
}
