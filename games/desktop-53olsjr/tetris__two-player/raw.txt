  const t=p.queue.shift();while(p.queue.length<5)refill(p);
  const pc={type:t,x:t==='O'?4:3,y:0,rot:0};
  if(!fits(p,t,0,pc.x,pc.y)){p.over=true;endGame(p);return;}
  p.piece=pc;p.gravAcc=0;p.lockT=0;p.grounded=false;p.shifts=0;
}
function tryMove(p,dir){const pc=p.piece;if(!pc)return false;
  if(fits(p,pc.type,pc.rot,pc.x+dir,pc.y)){pc.x+=dir;
    if(p.grounded){p.lockT=0;p.shifts++;}sMove(p.pan);return true;}
  return false;}
function rotatePiece(p){const pc=p.piece;if(!pc)return;
  const from=pc.rot,to=(from+1)%4;
  if(pc.type==='O'){pc.rot=to;sRot(p.pan);if(p.grounded)p.shifts++;return;}
  const table=pc.type==='I'?KI:KJ,kicks=table[from+'>'+to];
  for(const k of kicks){const nx=pc.x+k[0],ny=pc.y+k[1];
    if(fits(p,pc.type,to,nx,ny)){pc.x=nx;pc.y=ny;pc.rot=to;sRot(p.pan);
      if(p.grounded){p.lockT=0;p.shifts++;}return;}}
}
function addParticles(p,x,y,n,col,up){for(let i=0;i<n;i++){
  p.particles.push({x,y,vx:(Math.random()-.5)*(up?120:200),vy:up?-(40+Math.random()*130):(Math.random()-.5)*80,
    life:450+Math.random()*350,max:800,c:col,r:1.5+Math.random()*2.2});}
  if(p.particles.length>160)p.particles.splice(0,p.particles.length-160);}
function addFloat(p,txt,col,x,y){p.floats.push({txt,c:col,x,y,t:0});if(p.floats.length>6)p.floats.shift();}

function hardDrop(p){const pc=p.piece;if(!pc||p.over)return;
  let d=0;while(canDown(p)){pc.y++;d++;}
  p.score+=2*d;p.shake=Math.min(10,3+d*0.35);
  const cells=ROTS[pc.type][pc.rot];let maxy=-9;
  for(const c of cells)maxy=Math.max(maxy,c[1]);
  for(const c of cells)if(c[1]===maxy)addParticles(p,(pc.x+c[0])*CELL+15,(pc.y+c[1]-2)*CELL+24,3,'#9fb2d8',false);
  if(d>=3)addFloat(p,'+'+(2*d),'#9fb2d8',(pc.x+2)*CELL,(pc.y+maxy-2)*CELL);
  sDrop(p.pan);lockPiece(p);}

function lockPiece(p){const pc=p.piece,t=pc.type;let topOut=false;
  for(const c of ROTS[t][pc.rot]){const x=pc.x+c[0],y=pc.y+c[1];p.board[y][x]=t;if(y<2)topOut=true;}
  p.piece=null;p.grounded=false;
  const full=[];for(let r=0;r<ROWS;r++)if(p.board[r].every(v=>v))full.push(r);
  if(full.length){const n=full.length,pts=[0,100,300,500,800][n];
    p.score+=pts*p.level;p.lines+=n;p.clearAnim={rows:full,t:0,dur:320};p.pendingGarb=n;
    for(const r of full)for(let x=0;x<COLS;x++)addParticles(p,x*CELL+15,(r-2)*CELL+15,2,CSET[p.board[r][x]].base,true);
    const nl=Math.min(20,1+((p.lines/10)|0));const leveled=nl>p.level;
    if(n===4){p.tetris++;p.shake=10;flashMsg('TETRIS!','big '+p.cls);}
    else if(leveled)flashMsg('LEVEL '+nl,''+p.cls);
    else if(n>=2)flashMsg(n+' LINES',''+p.cls);
    addFloat(p,'+'+pts*p.level,n===4?'#ffd23f':(p.idx===0?'#ffd9a0':'#bdf0ff'),150,(Math.min(...full)-2)*CELL);
    if(leveled){p.level=nl;sLevel(p.pan);}
    sClear(n,p.pan);
  }else{sLock(p.pan);spawnPiece(p);}
  if(topOut)endGame(p);
}
function finishClear(p){const rows=p.clearAnim.rows.slice().sort((a,b)=>b-a);
  for(const r of rows){p.board.splice(r,1);p.board.unshift(new Array(COLS).fill(0));}
  p.clearAnim=null;
  if(p.pendingGarb>0){sendGarbage(players[1-p.idx],p.pendingGarb);p.pendingGarb=0;}
  spawnPiece(p);}
function sendGarbage(opp,n){
  for(let i=0;i<n;i++){const hole=(Math.random()*COLS)|0,row=new Array(COLS).fill('G');row[hole]=0;
    opp.board.splice(ROWS-1,0,row);if(opp.board.length>ROWS)opp.board.shift();}
  opp.garbFx={t:0,n};
  const el=opp.garbEl;el.classList.remove('on');void el.offsetWidth;el.classList.add('on');
  sGarb(opp.pan);
  let dead=false;for(let r=0;r<2;r++)if(opp.board[r].some(v=>v))dead=true;
  if(dead&&!opp.over)endGame(opp);}

function updatePlayer(p,dt){fxUpdate(p,dt);
  if(p.over||!p.piece)return;
  if(p.clearAnim){p.clearAnim.t+=dt;if(p.clearAnim.t>=p.clearAnim.dur)finishClear(p);return;}
  if(p.holdL.active){p.holdL.t-=dt;while(p.holdL.t<=0){if(!tryMove(p,-1))break;p.holdL.t+=ARR;}}
  if(p.holdR.active){p.holdR.t-=dt;while(p.holdR.t<=0){if(!tryMove(p,1))break;p.holdR.t+=ARR;}}
  const iv=p.softHeld?Math.min(gravMs(p.level),45):gravMs(p.level);
  p.gravAcc+=dt;let softTick=false;
  while(p.gravAcc>=iv){p.gravAcc-=iv;
    if(canDown(p)){p.piece.y++;if(p.softHeld){p.score+=1;softTick=true;}}else break;}
  if(softTick)sSoft(p.pan);
  if(!canDown(p)){if(!p.grounded){p.grounded=true;p.lockT=0;}
    p.lockT+=dt*(p.softHeld?3:1);
    if(p.lockT>=LOCK_DELAY||p.shifts>24)lockPiece(p);}
  else{p.grounded=false;p.lockT=0;}}

function fxUpdate(p,dt){p.shake=Math.max(0,p.shake-dt*0.02);
  if(p.garbFx){p.garbFx.t+=dt;if(p.garbFx.t>900)p.garbFx=null;}
  const s=dt/1000;
  for(let i=p.particles.length-1;i>=0;i--){const q=p.particles[i];q.life-=dt;
    if(q.life<=0){p.particles.splice(i,1);continue;}
    q.x+=q.vx*s;q.y+=q.vy*s;q.vy+=520*s;}
  for(let i=p.floats.length-1;i>=0;i--){const f=p.floats[i];f.t+=dt;if(f.t>900)p.floats.splice(i,1);}}

  pendingWinner=other.over?-1:(1-loser.idx);
  state='DYING';dieT=520;}
