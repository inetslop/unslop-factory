function playUpdate(dt){
  if(TR)return;
  let dir=0;if(keys['arrowleft']||keys['a'])dir-=1;if(keys['arrowright']||keys['d'])dir+=1;
  const dash=!!keys['shift'];
  P.moving=dir!==0;
  if(dir){P.face=dir;P.x=clamp(P.x+dir*(dash?300:190)*S*dt,40,GOAL-70);P.walkT+=dt*(dash?1.5:1);}
  P.puffT-=dt;
  if(dir&&P.puffT<=0){P.puffT=dash?.15:.24;sfx.step();
    spawnP({x:P.x-P.face*10*S,y:GY-2,vx:-P.face*20,vy:-14,g:26,life:.5,max:.5,size:3*S,col:'rgb(206,193,170)',kind:'puff'});}
  const tt=clamp(P.x/(GOAL-200),0,1);
  // ambient embers / fireflies
  if(Math.random()<dt*(1.5+tt*8))spawnP({x:camX+ri(0,W),y:GY-ri(0,70*S),vx:ri(-6,6),vy:-ri(12,34),life:ri(2,4),max:4,size:ri(1.5,3)*S,col:'rgb(255,184,77)'});
  if(tt>.6&&Math.random()<dt*3){const flies=particles.filter(p=>p.kind==='fly').length;
    if(flies<14)spawnP({x:camX+ri(0,W),y:GY-ri(10,90*S),vx:0,vy:0,life:5,max:5,size:2.2*S,col:'rgb(150,240,210)',kind:'fly'});}
  // chests
  for(const c of chests)if(!c.open&&Math.abs(P.x-c.x)<26*S){c.open=true;sfx.chest();
    if(Math.random()<.5){P.potions++;floater(c.x,GY-40*S,'+1 Potion','#7ce8b0');}
    else{const g=iri(8,20);P.gold+=g;floater(c.x,GY-40*S,'+'+g+' gold','#ffd166');}
    for(let i=0;i<14;i++)spawnP({x:c.x,y:GY-14*S,vx:ri(-70,70),vy:ri(-120,-30),g:220,life:.8,max:.8,size:2.5*S,col:'rgb(255,209,102)'});}
  // merchants
  G.nearM=merchants.some(m=>Math.abs(P.x-m)<70*S);
  if(!G.msg&&P.x>GOAL-640){G.msg=true;banner('THE GATE OF EMBER KEEP');sfx.horn();}
  // enemies
  for(const e of enemies){
    e.t+=dt*(Math.abs(e.dir)?1:.5);if(e.cool>0)e.cool-=dt;
    if(!e._frozen){e.x+=e.dir*e.sp*S*dt;
      if(e.x>e.anchor+e.patrol){e.x=e.anchor+e.patrol;e.dir=-1;}
      if(e.x<e.anchor-e.patrol){e.x=e.anchor-e.patrol;e.dir=1;}}
    if(e.type==='wyrm'&&Math.abs(e.x-camX-W*.5)<W&&Math.random()<dt*4)
      spawnP({x:e.x-95*S*e.dir,y:GY-92*S,vx:ri(-10,10),vy:-ri(20,50),life:1.6,max:1.6,size:2.5*S,col:'rgb(255,150,60)'});
    if(e.cool<=0&&Math.abs(P.x-e.x)<32*S){startBattle(e);return;}
  }
  updateHUD();
}
