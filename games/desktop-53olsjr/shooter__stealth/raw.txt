/* ============================== UPDATE: GUARDS ============================== */
function updateGuards(dt){
  let maxSusp=0,anyLOS=false;
  for(const g of guards){
    if(g.dead)continue;g.flash=Math.max(0,g.flash-dt*6);g.repath-=dt;
    const playActive=(state==='play'&&!dying);
    // exposure
    let sees=false;
    if(playActive){const dx=player.x-g.x,dy=player.y-g.y,d=Math.hypot(dx,dy);
      const range=(alertOn?460:310)*(player.crouch?0.72:1);
      if(d<range){const a=Math.atan2(dy,dx),half=alertOn?0.95:0.62;
        if(Math.abs(angDiff(a,g.dir))<half&&losClear(g.x,g.y,player.x,player.y)){sees=true;
          const exposure=(1-d/range)*(0.3+0.7*lightAt(player.x,player.y))*(player.crouch?0.42:1)*(player.spd>40?1:0.7);
          g.susp+=exposure*(alertOn?1.5:1)*dt*2.1;
          if(g.susp>=1)triggerAlert(player.x,player.y);}}}
    else g.susp=Math.max(0,g.susp-dt*0.4);
    maxSusp=Math.max(maxSusp,g.susp);
    if(sees){anyLOS=true;if(alertOn){alertT=10;alertPos.x=player.x;alertPos.y=player.y;}}
    else g.susp=Math.max(0,g.susp-dt*0.5);

    // state machine
    let moveAng=null,spd=55;
    if(g.state==='patrol'){
      if(g.wps){const wp=g.wps[g.wi];const dx=wp.x-g.x,dy=wp.y-g.y,d=Math.hypot(dx,dy);
        if(d<8){g.wi^=1;g.waitT=rand(0.7,2);}
        else{moveAng=Math.atan2(dy,dx);}}
      if(g.waitT>0){g.waitT-=dt;g.dir=g.sweepBase+Math.sin(T*1.5+g.phase)*1.0;}
      else if(moveAng!==null)g.dir=angLerp(g.dir,moveAng,Math.min(1,dt*8));
    }else if(g.state==='investigate'){spd=82;
      const dx=g.target.x-g.x,dy=g.target.y-g.y,d=Math.hypot(dx,dy);
      if(d<14){g.searchT+=dt;g.dir=g.sweepBase+Math.sin(T*1.7+g.phase)*0.9;
        if(g.searchT>3){g.state='patrol';g.waitT=0;if(g.wps)g.wi=(Math.abs(g.x-g.wps[0].x)+Math.abs(g.y-g.wps[0].y)<Math.abs(g.x-g.wps[1].x)+Math.abs(g.y-g.wps[1].y))?0:1;}}
      else{moveAng=Math.atan2(dy,dx);g.sweepBase=moveAng;}}
    else if(g.state==='combat'){spd=92;
      const dx=player.x-g.x,dy=player.y-g.y,d=Math.hypot(dx,dy);
      let aimT=null;
      if(playActive&&sees){aimT={x:player.x,y:player.y};
        g.dir=angLerp(g.dir,Math.atan2(dy,dx),Math.min(1,dt*10));
        if(d>190)moveAng=Math.atan2(dy,dx);
        g.fireCd-=dt;
        if(g.fireCd<=0&&d<430){g.fireCd=rand(0.75,1.3);
          const sp=player.crouch?0.16:0.09,a=Math.atan2(dy,dx)+rand(-sp,sp);
          bullets.push({x:g.x+Math.cos(a)*14,y:g.y+Math.sin(a)*14,vx:Math.cos(a)*540,vy:Math.sin(a)*540,from:'g',t:0});
          SFX.gshot(clamp(1-d/900,0.12,1));}}
      else{aimT=alertPos;const adx=alertPos.x-g.x,ady=alertPos.y-g.y,ad=Math.hypot(adx,ady);
        if(ad<16){g.searchT+=dt;g.dir=g.sweepBase+Math.sin(T*1.6+g.phase)*0.8;}
        else{moveAng=Math.atan2(ady,adx);g.sweepBase=moveAng;}}
      if(g.repath<=0||(g.path.length&&g.path[0]!==undefined)){
        if(!g.path.length||g.repath<=0){
          const tx=aimT?aimT.x:alertPos.x,ty=aimT?aimT.y:alertPos.y;
          const p=findPath(g.x,g.y,tx,ty);if(p.length)g.path=p;g.repath=1.4;}}}
    // movement via path or direct
    if(moveAng!==null){let tx,ty;
      if(g.path.length){const n=g.path[0];tx=n.x;ty=n.y;if(Math.hypot(tx-g.x,ty-g.y)<12)g.path.shift();}
      else{tx=g.x+Math.cos(moveAng);ty=g.y+Math.sin(moveAng);}
      const da=Math.atan2(ty-g.y,tx-g.x);moveBody(g,Math.cos(da)*spd*dt,Math.sin(da)*spd*dt);
      if(state!=='menu')g.dir=angLerp(g.dir,da,Math.min(1,dt*8));}
    // bullets from guards handled in bullet pass
  }
  // alert timer
  if(alertOn){if(anyLOS)alertT=Math.max(alertT,10);else{alertT-=dt;
    if(alertT<=0){alertOn=false;sirenStop();toast('THEY LOST YOU','#7dffa8');
      for(const g of guards)if(!g.dead){g.state='patrol';g.susp=0.25;g.path=[];}}}}
  return maxSusp;}

function updateBullets(dt){
  for(let i=bullets.length-1;i>=0;i--){const b=bullets[i];b.t+=dt;let dead=b.t>1.2;
    for(let s=0;s<2&&!dead;s++){b.x+=b.vx*dt/2;b.y+=b.vy*dt/2;
      if(solidPx(b.x,b.y)){dead=true;
        for(let k=0;k<5;k++)particles.push({x:b.x,y:b.y,vx:rand(-90,90),vy:rand(-90,90),t:0,life:0.3,type:'spark',col:'#ffd27a'});break;}
      if(b.from==='p'){for(const g of guards){if(g.dead)continue;const dx=g.x-b.x,dy=g.y-b.y;
        if(dx*dx+dy*dy<(g.r+4)*(g.r+4)){dead=true;g.hp=(g.hp||100)-34;g.flash=0.2;
          for(let k=0;k<5;k++)particles.push({x:b.x,y:b.y,vx:rand(-90,90),vy:rand(-90,90),t:0,life:0.3,type:'spark',col:'#ff6a5e'});
          if(g.hp<=0)killGuard(g,false);break;}}}
      else{if(state==='play'&&!dying){const dx=player.x-b.x,dy=player.y-b.y;
        if(dx*dx+dy*dy<(player.r+4)*(player.r+4)){dead=true;player.hp-=11;dmgT=0.35;shake(5);SFX.hit();
          for(let k=0;k<6;k++)particles.push({x:b.x,y:b.y,vx:rand(-90,90),vy:rand(-90,90),t:0,life:0.3,type:'spark',col:'#ff4757'});
          if(player.hp<=0){player.hp=0;dying=true;dieT=1.0;splat(player.x,player.y,true);shake(14);SFX.lose();}}}}}
    if(dead)bullets.splice(i,1);}}
function updateFx(dt){
  for(let i=particles.length-1;i>=0;i--){const p=particles[i];p.t+=dt;
    if(p.t>p.life){particles.splice(i,1);continue;}
    p.x+=(p.vx||0)*dt;p.y+=(p.vy||0)*dt;if(p.type==='spark'){p.vx-=p.vx*4*dt;p.vy-=p.vy*4*dt;}}
  for(let i=dusts.length-1;i>=0;i--){const d=dusts[i];d.t+=dt;if(d.t>d.life){dusts.splice(i,1);continue;}
    d.x+=Math.sin(T*0.8+d.ph)*4*dt;d.y-=3*dt;}
  if(dusts.length<70&&litTiles.length&&Math.random()<dt*7){const l=litTiles[(Math.random()*litTiles.length)|0];
    dusts.push({x:l.x+rand(-90,90),y:l.y+rand(-90,90),t:0,life:rand(3,6),ph:rand(TAU)});}
  for(let i=pings.length-1;i>=0;i++){pings[i].t+=dt;if(pings[i].t>0.8)pings.splice(i,1);}
  for(let i=corpses.length-1;i>=0;i--){corpses[i].t-=dt;if(corpses[i].t<=0)corpses.splice(i,1);}
  for(let i=floats.length-1;i>=0;i--){const f=floats[i];f.t+=dt;f.y-=22*dt;if(f.t>f.life)floats.splice(i,1);}
  for(let i=toasts.length-1;i>=0;i--){toasts[i].t+=dt;if(toasts[i].t>toasts[i].life)toasts.splice(i,1);}
  dmgT=Math.max(0,dmgT-dt);shakeMag=Math.max(0,shakeMag-dt*22);hintsT=Math.max(0,hintsT-dt);}
