function flood(seeds){
  const q=[], seen=new Set(); let size=0;
  for(const s of seeds)if(!seen.has(s[0])){seen.add(s[0]);q.push(s);}
  while(q.length){
    const [i,d]=q.shift(), cell=cells[i];
    if(cell.flag||cell.revealed)continue;
    cell.revealed=true; cell.revealAt=tWorld+Math.min(d*.05,.75); size++;
    if(cell.adj===0)for(const j of NBR[i])if(!seen.has(j)&&!cells[j].flag){seen.add(j);q.push([j,d+1]);}
  }
  return size;
}
function checkWin(){
  let rv=0; for(const c of cells)if(c.revealed&&!c.mine)rv++;
  if(rv>=safeTotal)win();
}

/* ---------- actions ---------- */
function hoverIdx(){
  if(state==='menu')return -1;
  const c=Math.max(0,Math.min(cols-1,Math.floor((ship.x-gridX)/cw)));
  const r=Math.max(0,Math.min(rows-1,Math.floor((ship.y-gridY)/ch)));
  return r*cols+c;
}
function tryProbe(){
  if(state!=='playing')return;
  if(tAmb-lastFire<.24)return;
  const i=hoverIdx(); if(i<0)return;
  lastFire=tAmb; timerOn=true; sfx.probe();
  const p=cellCenter(i);
  pulses.push({sx:ship.x,sy:ship.y,tx:p.x,ty:p.y,t:0,dur:.1,i});
}
function resolveProbe(p){
  if(state!=='playing')return;
  const cell=cells[p.i];
  if(cell.flag){sfx.deny();return;}
  if(!cell.revealed){
    if(cell.mine){die(p.i);return;}
    if(!minesPlaced)placeMinesSafe(p.i);
    const n=flood([[p.i,0]]);
    sfx.shatter(n); checkWin();
  }else{
    if(cell.adj>0){
      let f=0; for(const j of NBR[p.i])if(cells[j].flag)f++;
      if(f===cell.adj){
        const t=[]; for(const j of NBR[p.i])if(!cells[j].flag&&!cells[j].revealed)t.push(j);
        if(t.length){
          for(const j of t)if(cells[j].mine){die(j);return;}
          sfx.chord(); const n=flood(t.map(j=>[j,0])); sfx.shatter(n); checkWin();
        }
      }else sfx.deny();
    }else sfx.deny();
  }
}
function tryFlag(){
  if(state!=='playing')return;
  const i=hoverIdx(); if(i<0)return;
  const c=cells[i];
  if(c.revealed){sfx.deny();return;}
  c.flag=!c.flag; flagCount+=c.flag?1:-1; c.flagPop=tWorld;
  c.flag?sfx.flagOn():sfx.flagOff();
}

function die(i){
  state='dead'; deadT=0; timeScale=.3; shake=22;
  globalFlash={a:.5,c:'255,80,90'}; setThrust(false);
  const hc=cells[i]; hc.revealed=true; hc.revealAt=tWorld; hc.mRevealAt=tWorld+.06;
  const hit=cellCenter(i), order=[];
  for(let k=0;k<cells.length;k++)if(cells[k].mine&&k!==i){
    const p=cellCenter(k); order.push({k,d:Math.hypot(p.x-hit.x,p.y-hit.y)});}
  order.sort((a,b)=>a.d-b.d);
  order.forEach((o,k)=>cells[o.k].mRevealAt=tWorld+.25+k*.03);
  for(const c of cells)if(c.flag&&!c.mine)c.wrong=true;
  explode(ship.x,ship.y);
  sfx.boom(); sfx.sting();
  endShownAt=tAmb+1.7;
}
function win(){
  state='won'; winSpin=1.3; setThrust(false);
  let k=0; for(let i=0;i<cells.length;i++)if(cells[i].mine&&!cells[i].flag){
    pendingFlags.push({i,at:tWorld+.25+k*.045}); k++;}
  const cx=gridX+gridW/2, cy=gridY+gridH/2;
  for(let i=0;i<60;i++){const a=Math.random()*TAU, sp=60+Math.random()*280;

  sfx.win(); endShownAt=tAmb+1.5;
}

function ensureAudio(){
  if(AC){if(AC.state==='suspended')AC.resume();return;}
  const Ctx=window.AudioContext||window.webkitAudioContext; if(!Ctx)return;
  AC=new Ctx(); master=AC.createGain(); master.gain.value=muted?0:.5; master.connect(AC.destination);
  noiseBuf=AC.createBuffer(1,AC.sampleRate,AC.sampleRate);
  const d=noiseBuf.getChannelData(0); for(let i=0;i<d.length;i++)d[i]=Math.random()*2-1;
  const src=AC.createBufferSource(); src.buffer=noiseBuf; src.loop=true;
  const lp=AC.createBiquadFilter(); lp.type='lowpass'; lp.frequency.value=340;
  thrustG=AC.createGain(); thrustG.gain.value=0;
  src.connect(lp); lp.connect(thrustG); thrustG.connect(master); src.start();
  const o1=AC.createOscillator(),o2=AC.createOscillator(); o1.frequency.value=54; o2.frequency.value=54.7;
  const ag=AC.createGain(); ag.gain.value=.03; o1.connect(ag); o2.connect(ag); ag.connect(master); o1.start(); o2.start();
}

function tone(f,dur,type,vol,f2,delay){
  if(!AC)return; const t0=AC.currentTime+(delay||0);
  const o=AC.createOscillator(),g=AC.createGain();
  o.type=type||'sine'; o.frequency.setValueAtTime(f,t0);
  if(f2)o.frequency.exponentialRampToValueAtTime(Math.max(1,f2),t0+dur);
  g.gain.setValueAtTime(0,t0); g.gain.linearRampToValueAtTime(vol||.3,t0+.008);
  g.gain.exponentialRampToValueAtTime(.0001,t0+dur);
  o.connect(g); g.connect(master); o.start(t0); o.stop(t0+dur+.05);
}
function noise(dur,f,q,vol,type,delay){
  if(!AC)return; const t0=AC.currentTime+(delay||0);
  const s=AC.createBufferSource(); s.buffer=noiseBuf;
  const flt=AC.createBiquadFilter(); flt.type=type||'bandpass'; flt.frequency.value=f; flt.Q.value=q;
  const g=AC.createGain(); g.gain.setValueAtTime(vol,t0); g.gain.exponentialRampToValueAtTime(.0001,t0+dur);
  s.connect(flt); flt.connect(g); g.connect(master); s.start(t0); s.stop(t0+dur+.1);
}

function update(dt){
  tAmb+=dt;
  for(const s of stars){s.y+=s.sp*dt; s.x-=s.sp*.25*dt;
    if(s.y>H+4){s.y=-4;s.x=Math.random()*W;} if(s.x<-4)s.x=W+4;}
  const wdt=dt*timeScale;
  if(state==='dead'){deadT+=dt; if(deadT>.85)timeScale=Math.min(1,timeScale+dt*2.2);}
  if(state!=='menu'&&state!=='paused')tWorld+=wdt;

  if(state==='playing'){updateShip(wdt,true); if(timerOn)playTime+=dt;}
  else if(state==='won'){ if(winSpin>0){ship.a+=10*wdt; winSpin-=dt;} updateShip(wdt,false);}

  for(let i=pulses.length-1;i>=0;i--){const p=pulses[i]; p.t+=dt;
    if(p.t>=p.dur){pulses.splice(i,1); resolveProbe(p);}}

  if(state!=='menu'){
    for(let i=0;i<cells.length;i++){const c=cells[i];
      if(c.revealed&&!c.burst&&tWorld>=c.revealAt){c.burst=true; shatterBurst(i);}}
    for(let i=pendingFlags.length-1;i>=0;i--){const f=pendingFlags[i];
      if(tWorld>=f.at){const c=cells[f.i]; if(!c.flag){c.flag=true; flagCount++; c.flagPop=tWorld; sparkle(f.i);}
        pendingFlags.splice(i,1);}}
  }

  for(let i=particles.length-1;i>=0;i--){const p=particles[i];
    p.life+=wdt; if(p.life>p.max){particles.splice(i,1);continue;}
    const dr=p.type==='shard'?1.2:2.4;
    p.vx*=Math.exp(-dr*wdt); p.vy*=Math.exp(-dr*wdt);
    p.x+=p.vx*wdt; p.y+=p.vy*wdt; if(p.vr!==undefined)p.rot+=p.vr*wdt;}

  for(let i=flashes.length-1;i>=0;i--){flashes[i].t+=dt;
    if(flashes[i].t>flashes[i].dur)flashes.splice(i,1);}

  shake*=Math.exp(-4.5*dt); if(shake<.08)shake=0;
  if(globalFlash){globalFlash.a-=dt*1.3; if(globalFlash.a<=0)globalFlash=null;}

  for(const d of deco){d.x+=d.vx*dt; d.y+=d.vy*dt; d.rot+=d.vr*dt;
    const m=140;
    if(d.x<-m)d.x=W+m; if(d.x>W+m)d.x=-m; if(d.y<-m)d.y=H+m; if(d.y>H+m)d.y=-m;}

  if((state==='dead'||state==='won')&&!endShown&&tAmb>=endShownAt){endShown=true; showEnd(state);}
}
function updateShip(dt,allowInput){
  const kL=keys.has('ArrowLeft')||keys.has('a'), kR=keys.has('ArrowRight')||keys.has('d');
  thrusting=allowInput&&(keys.has('ArrowUp')||keys.has('w'));
  if(allowInput){ if(kL)ship.a-=4.6*dt; if(kR)ship.a+=4.6*dt;}
  else setThrust(false);
  if(thrusting){
    ship.vx+=Math.cos(ship.a)*560*SF*dt; ship.vy+=Math.sin(ship.a)*560*SF*dt;
    setThrust(true);
    if(particles.length<700)for(let k=0;k<2;k++){
      const bx=ship.x-Math.cos(ship.a)*ship.R*.6, by=ship.y-Math.sin(ship.a)*ship.R*.6;
      particles.push({type:'spark',x:bx,y:by,
        vx:-Math.cos(ship.a)*(80+Math.random()*90)+(Math.random()-.5)*46,
        vy:-Math.sin(ship.a)*(80+Math.random()*90)+(Math.random()-.5)*46,
        life:0,max:.25+Math.random()*.25,size:1.5+Math.random()*2,
        color:['#ffe1b0','#ffb454','#ff8a3d'][(Math.random()*3)|0]});}
  }else if(allowInput)setThrust(false);
  const cap=430*SF, sp=Math.hypot(ship.vx,ship.vy);
  if(sp>cap){ship.vx*=cap/sp; ship.vy*=cap/sp;}
  ship.vx*=Math.exp(-.5*dt); ship.vy*=Math.exp(-.5*dt);
  ship.x+=ship.vx*dt; ship.y+=ship.vy*dt;
  const m=ship.R+2, minx=gridX+m, miny=gridY+m, maxx=gridX+gridW-m, maxy=gridY+gridH-m;
  if(ship.x<minx){ship.x=minx; ship.vx=Math.abs(ship.vx)*.75;}
  if(ship.x>maxx){ship.x=maxx; ship.vx=-Math.abs(ship.vx)*.75;}
  if(ship.y<miny){ship.y=miny; ship.vy=Math.abs(ship.vy)*.75;}
  if(ship.y>maxy){ship.y=maxy; ship.vy=-Math.abs(ship.vy)*.75;}
}

      }else{
        ctx.fillStyle='rgba(9,16,30,.5)';
        ctx.fillRect(cx-cw/2+1,cy-ch/2+1,cw-2,ch-2);
        if(c.adj>0){
          const p=tWorld-c.revealAt;
          const sc=1+Math.max(0,.4-p*2.5), al=Math.min(1,p/.15);
          ctx.save(); ctx.translate(cx,cy); ctx.scale(sc,sc); ctx.globalAlpha=al;
          ctx.font=`700 ${Math.min(cw,ch)*.46}px ${MONO}`;
          ctx.textAlign='center'; ctx.textBaseline='middle';
          ctx.shadowColor=NUMC[c.adj]; ctx.shadowBlur=10;
          ctx.fillStyle=NUMC[c.adj]; ctx.fillText(c.adj,0,2);
          ctx.restore(); ctx.globalAlpha=1;
        }
      }
    }

    // flags & mines (death)
    for(let i=0;i<cells.length;i++){
      const c=cells[i], cx=gridX+(i%cols+.5)*cw, cy=gridY+(((i/cols)|0)+.5)*ch;
      if(c.flag&&!c.revealed){
        const p=Math.max(0,tWorld-c.flagPop), sc=p<.14?.3+.7*(p/.14):1;
        const R=Math.min(cw,ch)*.26*sc, bob=Math.sin(t*3+i)*1.5;
        ctx.save(); ctx.translate(cx,cy+bob);
        ctx.strokeStyle='rgba(255,150,90,.3)'; ctx.lineWidth=1.2;
        ctx.beginPath(); ctx.arc(0,R*.55,R*(.85+.15*Math.sin(t*4+i)),0,TAU); ctx.stroke();
        ctx.strokeStyle='#ffd9b0'; ctx.lineWidth=1.6;
        ctx.beginPath(); ctx.moveTo(0,R*.55); ctx.lineTo(0,-R*.7); ctx.stroke();
        ctx.fillStyle=c.wrong?'#8a3a44':'#ff8f4d';
        ctx.strokeStyle=c.wrong?'#c96a76':'#ffc290';
        ctx.beginPath(); ctx.moveTo(0,-R*.7); ctx.lineTo(R*.62,-R*.45); ctx.lineTo(0,-R*.2); ctx.closePath();
        ctx.fill(); ctx.stroke();
        if(c.wrong){ctx.strokeStyle='#ff5566'; ctx.lineWidth=2.4;
          ctx.beginPath(); ctx.moveTo(-R*.7,-R*.1); ctx.lineTo(R*.7,R*.7);
          ctx.moveTo(R*.7,-R*.1); ctx.lineTo(-R*.7,R*.7); ctx.stroke();}
        ctx.restore();
      }
      if(state==='dead'&&c.mine&&c.mRevealAt>0&&tWorld>=c.mRevealAt){
        const p=tWorld-c.mRevealAt, sc=p<.15?.3+.7*(p/.15):1;
        const r=Math.min(cw,ch)*.2*sc;
        ctx.save(); ctx.translate(cx,cy);
        ctx.shadowColor='#ff5566'; ctx.shadowBlur=12+6*Math.sin(t*5+i);
        ctx.fillStyle='#2a0d14'; ctx.strokeStyle='#ff5566'; ctx.lineWidth=2;
        ctx.beginPath(); ctx.arc(0,0,r,0,TAU); ctx.fill(); ctx.stroke();
        for(let k=0;k<8;k++){const a=k/8*TAU+i;
          ctx.beginPath(); ctx.moveTo(Math.cos(a)*r,Math.sin(a)*r);
          ctx.lineTo(Math.cos(a)*r*1.6,Math.sin(a)*r*1.6); ctx.stroke();}
        ctx.fillStyle='#ff8d9a'; ctx.beginPath(); ctx.arc(0,0,r*.3,0,TAU); ctx.fill();
        ctx.restore();
      }
    }

    // pulses
    for(const p of pulses){
      const k=p.t/p.dur;
      const lx=p.sx+(p.tx-p.sx)*k, ly=p.sy+(p.ty-p.sy)*k;
      ctx.strokeStyle=`rgba(140,230,255,${.9*(1-k*.4)})`; ctx.lineWidth=2.5;
      ctx.beginPath(); ctx.moveTo(p.sx,p.sy); ctx.lineTo(lx,ly); ctx.stroke();
      ctx.strokeStyle=`rgba(140,230,255,${1-k})`; ctx.lineWidth=1.6;
      ctx.beginPath(); ctx.arc(p.tx,p.ty,k*Math.min(cw,ch)*.46,0,TAU); ctx.stroke();
    }

    // ship
    if(state!=='dead'){
      ctx.save(); ctx.translate(ship.x,ship.y); ctx.rotate(ship.a);
      const R=ship.R;
      if(thrusting){
        const L=R*(.9+Math.random()*.55);
        ctx.strokeStyle='rgba(255,170,80,.9)'; ctx.lineWidth=3;
        ctx.beginPath(); ctx.moveTo(-R*.55,R*.42); ctx.lineTo(-R*.55-L,0); ctx.lineTo(-R*.55,-R*.42); ctx.stroke();
        ctx.strokeStyle='rgba(255,240,210,.95)'; ctx.lineWidth=1.2;
        ctx.beginPath(); ctx.moveTo(-R*.55,R*.2); ctx.lineTo(-R*.55-L*.55,0); ctx.lineTo(-R*.55,-R*.2); ctx.stroke();
      }
      ctx.shadowColor='#6fd9ff'; ctx.shadowBlur=10;
      ctx.fillStyle='rgba(10,20,34,.9)'; ctx.strokeStyle='#eaf6ff'; ctx.lineWidth=2;
      ctx.beginPath(); ctx.moveTo(R,0); ctx.lineTo(-R*.7,R*.62); ctx.lineTo(-R*.35,0);
      ctx.lineTo(-R*.7,-R*.62); ctx.closePath(); ctx.fill(); ctx.stroke();
      ctx.restore();
    }

    // particles — solid shards
    for(const p of particles){ if(p.type!=='shard')continue;
      const a=1-p.life/p.max;
      ctx.globalAlpha=a; ctx.save(); ctx.translate(p.x,p.y); ctx.rotate(p.rot||0);
      ctx.strokeStyle=p.color; ctx.lineWidth=1.5;
      ctx.beginPath(); ctx.moveTo(p.size,0); ctx.lineTo(-p.size*.7,p.size*.6);
      ctx.lineTo(-p.size*.7,-p.size*.6); ctx.closePath(); ctx.stroke();
      ctx.restore();}
    // additive sparks / rings
    ctx.globalCompositeOperation='lighter';
    for(const p of particles){ if(p.type==='shard')continue;
      const a=1-p.life/p.max; ctx.globalAlpha=a;
      if(p.type==='ring'){
        const e=1-Math.pow(1-p.t/p.dur,3);
        ctx.strokeStyle=`rgba(255,170,140,${a*.8})`; ctx.lineWidth=2.5;
        ctx.beginPath(); ctx.arc(p.x,p.y,p.r0+(p.r1-p.r0)*e,0,TAU); ctx.stroke();
      }else{
        ctx.fillStyle=p.color;
        ctx.beginPath(); ctx.arc(p.x,p.y,Math.max(.4,p.size*(1-p.life/p.max*.6)),0,TAU); ctx.fill();
      }}
    ctx.globalCompositeOperation='source-over'; ctx.globalAlpha=1;

    // reticle
    if(hov>=0){
      const c=cells[hov];
      if(!c.revealed&&!c.flag){
        const cx=gridX+(hov%cols+.5)*cw, cy=gridY+(((hov/cols)|0)+.5)*ch;
        const half=Math.min(cw,ch)*.34+Math.sin(t*5)*2, len=Math.min(cw,ch)*.13;
        ctx.strokeStyle='rgba(140,230,255,.95)'; ctx.lineWidth=2; ctx.beginPath();
        ctx.moveTo(cx-half,cy-half+len); ctx.lineTo(cx-half,cy-half); ctx.lineTo(cx-half+len,cy-half);
        ctx.moveTo(cx+half-len,cy-half); ctx.lineTo(cx+half,cy-half); ctx.lineTo(cx+half,cy-half+len);
        ctx.moveTo(cx+half,cy+half-len); ctx.lineTo(cx+half,cy+half); ctx.lineTo(cx+half-len,cy+half);
        ctx.moveTo(cx-half+len,cy+half); ctx.lineTo(cx-half,cy+half); ctx.lineTo(cx-half,cy+half-len);
        ctx.stroke();
        ctx.fillStyle='rgba(140,230,255,.6)';
        ctx.beginPath(); ctx.arc(cx,cy,1.6,0,TAU); ctx.fill();
      }
    }

    // flashes
    for(const f of flashes){
      const a=(1-f.t/f.dur)*.5;
      const g=ctx.createRadialGradient(f.x,f.y,0,f.x,f.y,f.r);
      g.addColorStop(0,`rgba(${f.c},${a})`); g.addColorStop(1,'rgba(0,0,0,0)');
      ctx.fillStyle=g; ctx.fillRect(f.x-f.r,f.y-f.r,f.r*2,f.r*2);
    }
    ctx.restore();
  }

  if(vign){ctx.fillStyle=vign; ctx.fillRect(0,0,W,H);}
  if(state!=='menu')drawHUD(hov);
  if(globalFlash){ctx.fillStyle=`rgba(${globalFlash.c},${Math.max(0,globalFlash.a)})`; ctx.fillRect(0,0,W,H);}
}

function drawHUD(hov){
  ctx.fillStyle='rgba(5,10,20,.9)'; ctx.fillRect(0,0,W,HUD_H);
  const g=ctx.createLinearGradient(0,0,W,0);
  g.addColorStop(0,'rgba(110,200,255,0)'); g.addColorStop(.5,'rgba(110,200,255,.5)'); g.addColorStop(1,'rgba(110,200,255,0)');
  ctx.fillStyle=g; ctx.fillRect(0,HUD_H-1.5,W,1.5);
  const my=HUD_H/2;
  // ship glyph + title
  ctx.save(); ctx.translate(24,my); ctx.rotate(-TAU/4);
  ctx.strokeStyle='#7fdfff'; ctx.lineWidth=1.8;
  ctx.beginPath(); ctx.moveTo(8,0); ctx.lineTo(-5.5,5); ctx.lineTo(-2.5,0); ctx.lineTo(-5.5,-5); ctx.closePath(); ctx.stroke();
  ctx.restore();
  ctx.textAlign='left'; ctx.textBaseline='middle';
  setLS('3px'); ctx.font=`700 13px ${MONO}`; ctx.fillStyle='#d9efff';
  ctx.fillText('ASTEROID SWEEPER',42,my-8);
  setLS('2px'); ctx.font=`600 10px ${MONO}`; ctx.fillStyle='#5f7a94';
  ctx.fillText(sectorTag+' · '+DIFFS[diffIdx].name,42,my+10); setLS('0px');
  // center target readout
  if(W>760&&hov>=0){
    const r=((hov/cols)|0), c=hov%cols;
    ctx.textAlign='center'; setLS('2px'); ctx.font=`600 10px ${MONO}`; ctx.fillStyle='#4d6a85';
    ctx.fillText('TGT '+String.fromCharCode(65+r)+'-'+String(c+1).padStart(2,'0'),W/2,my); setLS('0px');
  }
  // right segments
  const remain=mineCount-flagCount;
  seg(W-16,'TIME',fmtT(playTime));
  seg(W-118,'MINES',(remain<0?'-':'')+String(Math.abs(remain)).padStart(2,'0'));
  seg(W-234,'SND',muted?'OFF':'ON');
}
function seg(xR,label,val){
  ctx.textAlign='right'; ctx.textBaseline='middle';
  setLS('2px'); ctx.font=`600 9px ${MONO}`; ctx.fillStyle='#5f7a94';
  ctx.fillText(label,xR,HUD_H/2-10);
  setLS('1px'); ctx.font=`700 16px ${MONO}`;
  ctx.fillStyle=(label==='SND'&&muted)?'#ff8d7a':'#eaf6ff';
  ctx.fillText(val,xR,HUD_H/2+9); setLS('0px');
}

/* ---------- main loop ---------- */
function frame(ts){
  if(!lastTs)lastTs=ts;
  let dt=Math.min(.05,(ts-lastTs)/1000); lastTs=ts;
  if(state==='paused'){tAmb+=dt;}
  else update(dt);
  render();
  requestAnimationFrame(frame);
}
resize();
requestAnimationFrame(frame);
