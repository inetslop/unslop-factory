/* ============ update ============ */
var keys=new Set();
function update(dt){
  if(state==='menu'){attract(dt);fx(dt);return;}
  if(paused){setEngine(0,0);return;}
  if(freeze>0){freeze-=dt;fx(dt);return;}

  movePaddle(p[0],0,dt);movePaddle(p[1],1,dt);

  if(state==='serve'){serveT+=dt;
    var idx=Math.floor(serveT/0.45);
    if(idx>lightIdx&&idx<5){lightIdx=idx;beep(480+idx*95,.09,'square',.13);}
    if(!served&&serveT>=5*0.45+0.85){served=true;goFlash=.42;
      beep(880,.28,'square',.2,1400);shake(5,.2);
      var a=(Math.random()*2-1)*0.6;
      ball.vx=Math.cos(a)*ball.sp*ball.dir;ball.vy=Math.sin(a)*ball.sp;state='play';}}

  if(state==='play'){
    var steps=Math.max(1,Math.ceil(ball.sp*dt/14)),sdt=dt/steps;
    for(var s=0;s<steps;s++){
      ball.x+=ball.vx*sdt;ball.y+=ball.vy*sdt;
      topSpeed=Math.max(topSpeed,ball.sp);
      if(ball.sp>460&&Math.random()<sdt*38)
        tire(ball.x,ball.y,Math.atan2(ball.vy,ball.vx),.16+ball.sp/MAXSP*.4,3.1);
      if(ball.y<TOP+R){ball.y=TOP+R;ball.vy=-ball.vy;tire(ball.x,TOP+4,0,.85,4.5);
        sparks(ball.x,TOP+6,7,'#ffd9d9');beep(170,.07,'triangle',.13,90);shake(3,.12);}
      if(ball.y>BOT-R){ball.y=BOT-R;ball.vy=-ball.vy;tire(ball.x,BOT-4,0,.85,4.5);
        sparks(ball.x,BOT-6,7,'#ffd9d9');beep(170,.07,'triangle',.13,90);shake(3,.12);}
      tryHit(p[0],0);tryHit(p[1],1);
      if(ball.x<-30){score(1);break;}
      if(ball.x>W+30){score(0);break;}}
    ball.ang=Math.atan2(ball.vy,ball.vx);}

  if(state==='scored'){scoreT-=dt;
    if(scoreT<=0){if(pendingWin!=null)doWin(pendingWin);else startServe();}}

  fx(dt);
  var tn=state==='play'?ball.sp/MAXSP:.12;
  setEngine(tn,paused?0:(state==='play'?.04+tn*.05:(state==='serve'?.028:0)));}

function tryHit(pp,i){var dir=i===0?1:-1;
  if(i===0?ball.vx>=0:ball.vx<=0)return;
  if(Math.abs(ball.x-pp.x)<PW/2+R+2&&Math.abs(ball.y-pp.y)<=pp.h/2+R){
    var rel=Math.max(-1,Math.min(1,(ball.y-pp.y)/(pp.h/2)));
    ball.sp=Math.min(ball.sp*1.05+6,MAXSP);
    var a=rel*1.05;
    ball.vx=Math.cos(a)*ball.sp*dir;
    ball.vy=Math.sin(a)*ball.sp+pp.vy*.22;
    if(Math.abs(ball.vy)>ball.sp*.88)ball.vy=(ball.vy>0?1:-1)*ball.sp*.88;
    ball.sp=Math.hypot(ball.vx,ball.vy);
    ball.x=pp.x+dir*(PW/2+R+1);
    freeze=.045;rally++;bestRally=Math.max(bestRally,rally);
    ball.tint=pp.color;shake(4,.15);
    sparks(ball.x,ball.y,10,pp.color);tire(ball.x,ball.y,Math.atan2(ball.vy,ball.vx),.9,5);
    beep(300+ball.sp*.35,.06,'square',.17,220+ball.sp*.2);}}

function attract(dt){
  for(var i=0;i<2;i++){var d=(ball.y-p[i].y),v=Math.max(-430,Math.min(430,d*9));p[i].y+=v*dt;
    p[i].y=Math.max(TOP+p[i].h/2+4,Math.min(BOT-p[i].h/2-4,p[i].y));}
  ball.x+=ball.vx*dt;ball.y+=ball.vy*dt;
  if(ball.y<TOP+R){ball.y=TOP+R;ball.vy=-ball.vy;}
  if(ball.y>BOT-R){ball.y=BOT-R;ball.vy=-ball.vy;}
  for(var j=0;j<2;j++){var pp=p[j],dir=j===0?1:-1;
    if(Math.abs(ball.x-pp.x)<PW/2+R&&Math.abs(ball.y-pp.y)<=pp.h/2+R){
      ball.x=pp.x+dir*(PW/2+R+1);ball.vx=dir*Math.abs(ball.vx);
      ball.vy=(Math.random()*2-1)*ball.sp*.45;}}
  if(ball.x<-40||ball.x>W+40){ball.x=W/2;ball.y=H/2;var a=Math.random()*6.28;
    ball.vx=Math.cos(a)*ball.sp;ball.vy=Math.sin(a)*ball.sp;}
  ball.ang=Math.atan2(ball.vy,ball.vx);
  if(ball.sp>500&&Math.random()<dt*30)
    tire(ball.x,ball.y,ball.ang,.14,3);
  setEngine(0,0);}

function fx(dt){
  for(var i=P.length-1;i>=0;i--){var q=P[i];q.l-=dt;
    if(q.l<=0){P.splice(i,1);continue;}
    if(q.t==='c'){q.vy+=q.g*dt;q.rot+=q.vr*dt;}
    q.x+=q.vx*dt;q.y+=q.vy*dt;if(q.t==='s')q.vx*=.94;}
  if(shakeT>0)shakeT-=dt;else shakeMag=0;
  flash.t=Math.max(0,flash.t-dt*1.1);goFlash=Math.max(0,goFlash-dt);
  if(state!=='win')dashOff+=(state==='play'?ball.sp*.35:100)*dt;
  skc.globalCompositeOperation='destination-out';
  skc.fillStyle='rgba(0,0,0,'+Math.min(.06,dt*1.4)+')';skc.fillRect(0,0,W,H);
  skc.globalCompositeOperation='source-over';}
