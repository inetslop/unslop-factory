/* ============================== update ============================== */
function update(dt){
  T+=dt;
  if(state==='play'){
    elapsed+=dt;
    if(player.moving){
      player.t+=dt/0.115;
      if(player.t>=1){
        player.moving=false;player.x=player.tx;player.y=player.ty;moves++;updateHUD();
        if(player.x===D.door.x&&player.y===D.door.y&&doorOpen)doClear();
      }
    }else{
      for(let i=dirStack.length-1;i>=0;i--){
        const e=dirStack[i];
        if(keys.has(e.k)){tryMove(DIRV[e.d][0],DIRV[e.d][1]);break;}
      }
    }
  }
  // ambient motes
  if(parts.length<240&&Math.random()<dt*7){
    const x=ox+Math.random()*tile*12,y=oy+tile*8+Math.random()*20;
    parts.push({x,y,vx:(Math.random()-0.5)*6,vy:-6-Math.random()*8,life:3+Math.random()*2,max:5,size:0.8+Math.random()*1,col:'#9fb8dd',add:false,grav:0});
  }
  for(let i=parts.length-1;i>=0;i--){
    const p=parts[i];p.life-=dt;
    if(p.life<=0){parts.splice(i,1);continue;}
    p.x+=p.vx*dt;p.y+=p.vy*dt;p.vy+=(p.grav||0)*dt;
  }
  if(shake>0)shake=Math.max(0,shake-dt*1.7);
}

function drawMirrorShape(px,py,o,fixed,alpha){
  const t=tile,pad=t*0.13;
  ctx.globalAlpha=alpha;
  const g=ctx.createLinearGradient(px,py,px,py+t);
  if(fixed){g.addColorStop(0,'#7a6238');g.addColorStop(1,'#453820');}
  else{g.addColorStop(0,'#48566f');g.addColorStop(1,'#273249');}
  ctx.fillStyle=g;ctx.fillRect(px+pad,py+pad,t-2*pad,t-2*pad);
  ctx.strokeStyle='rgba(0,0,0,.55)';ctx.lineWidth=1;
  ctx.strokeRect(px+pad+0.5,py+pad+0.5,t-2*pad-1,t-2*pad-1);
  if(fixed){
    ctx.fillStyle='rgba(255,224,168,.55)';
    const r=1.8,q=pad+3.5;
    [[q,q],[t-q,q],[q,t-q],[t-q,t-q]].forEach(p=>{ctx.beginPath();ctx.arc(px+p[0],py+p[1],r,0,7);ctx.fill();});
  }

  ctx.strokeStyle='rgba(140,190,255,.35)';ctx.lineWidth=t*0.16;
  ctx.beginPath();ctx.moveTo(a[0][0],a[0][1]);ctx.lineTo(a[1][0],a[1][1]);ctx.stroke();

        ctx.lineDashOffset=-((T*2.4)%1)*t*0.72;
        ctx.strokeStyle='rgba(255,255,255,.55)';ctx.lineWidth=t*0.06;
        ctx.beginPath();ctx.moveTo(x1,y1);ctx.lineTo(x2,y2);ctx.stroke();
        ctx.setLineDash([]);ctx.lineDashOffset=0;
      }
    }
    const pr=t*(0.09+Math.sin(T*6+x1)*0.03);
    ctx.fillStyle='rgba('+c.rgb+',.75)';
    ctx.beginPath();ctx.arc(x2,y2,pr,0,7);ctx.fill();
  }
}
function draw(){
  const bg=ctx.createLinearGradient(0,0,0,H);
  bg.addColorStop(0,'#0a1120');bg.addColorStop(1,'#05070d');
  ctx.fillStyle=bg;ctx.fillRect(0,0,W,H);
  ctx.save();
  if(shake>0){const m=shake*shake*10;ctx.translate((Math.random()*2-1)*m,(Math.random()*2-1)*m);}
  const t=tile;
  // tiles
  for(let y=0;y<8;y++)for(let x=0;x<12;x++){
    const px=ox+x*t,py=oy+y*t,k=x+','+y,hh=hash(x,y);
    if(walls.has(k)){drawWall(px,py,x,y);continue;}
    if(voids.has(k)){drawVoid(px,py,x,y);continue;}
    const base=(x+y)&1?[15,24,42]:[18,28,48];
    ctx.fillStyle='rgb('+Math.round(base[0]+hh*6)+','+Math.round(base[1]+hh*7)+','+Math.round(base[2]+hh*9)+')';
    ctx.fillRect(px,py,t,t);
    ctx.fillStyle='rgba(0,0,0,.2)';
    ctx.fillRect(px,py+t-1,t,1);ctx.fillRect(px+t-1,py,1,t);
    for(let i=0;i<2;i++){
      const sx=px+hash(x*7+i,y*13)*t,sy=py+hash(x*11,y*5+i*3)*t;
      ctx.fillStyle='rgba(200,220,255,.05)';ctx.fillRect(sx,sy,1.6,1.6);
    }
  }
  // frame
  ctx.strokeStyle='rgba(86,184,255,.1)';ctx.lineWidth=1;
  ctx.strokeRect(ox-7.5,oy-7.5,t*12+15,t*8+15);
  ctx.strokeStyle='rgba(255,200,87,.35)';ctx.lineWidth=2;
  const fo=11,L=t*0.42;
  [[ox-fo,oy-fo,1,1],[ox+t*12+fo,oy-fo,-1,1],[ox-fo,oy+t*8+fo,1,-1],[ox+t*12+fo,oy+t*8+fo,-1,-1]].forEach(c=>{
    ctx.beginPath();ctx.moveTo(c[0]+c[2]*L,c[1]);ctx.lineTo(c[0],c[1]);ctx.lineTo(c[0],c[1]+c[3]*L);ctx.stroke();
  });
  drawDoor();
  emitters.forEach((e,i)=>drawEmitter(e,i));
  prisms.forEach((p,i)=>drawPrism(p,i));
  for(const m of fixedM)drawMirrorShape(ox+m.x*t,oy+m.y*t,m.o,true,1);
  for(const m of placed)drawMirrorShape(ox+m.x*t,oy+m.y*t,m.o,false,1);
  drawPlayer();
  if(sim)drawBeams(sim.segs,1,true);
  // hover ghost + preview
  if(state==='play'&&hover){
    const hx=hover.x,hy=hover.y,k=hx+','+hy;
    ctx.setLineDash([5,4]);ctx.strokeStyle='rgba(255,200,87,.4)';ctx.lineWidth=1.5;
    ctx.strokeRect(ox+hx*t+2,oy+hy*t+2,t-4,t-4);ctx.setLineDash([]);
    const m=mirrorAt(hx,hy);let opts=null,gO=null,bad=false;
    if(m){
      if(m.fixed){gO=m.o==='/'?'\\':'/';opts={set:{x:hx,y:hy,o:gO}};}
      else opts={omit:k};
    }else if(canPlace(hx,hy)){
      gO=curO;opts={set:{x:hx,y:hy,o:gO}};
      if(placed.length>=D.budget)bad=true;
    }
    if(opts&&!bad){
      drawBeams(simulate(opts).segs,0.5,false);
      if(gO)drawMirrorShape(ox+hx*t,oy+hy*t,gO,m&&m.fixed,0.42+Math.sin(T*5)*0.08);
    }else if(bad){
      drawMirrorShape(ox+hx*t,oy+hy*t,curO,false,0.18);
    }
  }
  // particles
  for(const p of parts){
    const a=clamp(p.life/p.max,0,1);
    ctx.globalAlpha=a;
    if(p.add)ctx.globalCompositeOperation='lighter';
    ctx.fillStyle=p.col;
    ctx.beginPath();ctx.arc(p.x,p.y,p.size*(a*0.6+0.4),0,7);ctx.fill();
    ctx.globalCompositeOperation='source-over';
  }
  ctx.globalAlpha=1;
  ctx.restore();
  if(vig){ctx.fillStyle=vig;ctx.fillRect(0,0,W,H);}
}

function frame(ts){

  requestAnimationFrame(frame);
}
