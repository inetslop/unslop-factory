function typeLetter(ch){
  if(!G.asteroids.length)return;
  let target=null,ty=-1e9;
  for(const a of G.asteroids){if(a.word[a.prog]===ch&&a.y>ty){target=a;ty=a.y;}}
  if(!target){G.typedBad++;G.combo=0;updateComboUI(false);sfx.error();G.shipHitFlash=.35;return;}
  G.typedOK++; target.prog++;
  const before=multFor(G.combo); G.combo++;
  if(G.combo>G.maxCombo)G.maxCombo=G.combo;
  const after=multFor(G.combo); updateComboUI(after>before);
  sfx.key(target.prog-1);
  G.parts.push({x:target.x+Math.random()*20-10,y:target.y,vx:(Math.random()-.5)*160,vy:-80-Math.random()*80,
    life:0,max:.35,size:2,color:'#7dffd9',type:'spark'});
  if(after>before){sfx.combo(after);G.floats.push({x:target.x,y:target.y-target.r-16,txt:'COMBO ×'+after,t:0,color:'#ffc36b'});}
  if(target.prog>=target.word.length)destroyAsteroid(target);}

function destroyAsteroid(a){
  const i=G.asteroids.indexOf(a); if(i<0)return; G.asteroids.splice(i,1);
  G.destroyed++;
  const pts=Math.round(a.word.length*10*G.mult*(a.shard?1.5:1)); G.score+=pts; setScore();
  G.beams.push({x1:G.shipX,y1:lineY()-26,x2:a.x,y2:a.y,t:0});
  explodeAt(a.x,a.y,a.shard?['#ffd9a0','#e08a3c','#fff1dd']:['#9db1d6','#5d6c8e','#c9d6ef'],14);
  G.rings.push({x:a.x,y:a.y,r0:a.r*.4,max:a.r*3.2,t:0,color:a.shard?'rgba(255,180,110,':'rgba(125,255,217,'});
  G.floats.push({x:a.x,y:a.y-a.r*.5,txt:'+'+pts,t:0,color:G.mult>1?'#ffc36b':'#7dffd9'});
  sfx.boom(a.shard); G.shake=Math.max(G.shake,a.shard?4.5:2.5);}

function update(dt){
  G.t+=dt;
  G.shake=Math.max(0,G.shake-dt*26);
  G.redFlash=Math.max(0,G.redFlash-dt*1.8);
  G.shipHitFlash=Math.max(0,G.shipHitFlash-dt*3);
  for(const s of stars){s.y+=STARSPEED[s.z]*dt;if(s.y>H+2){s.y=-2;s.x=Math.random()*W;}}
  G.streakT-=dt; if(G.streakT<=0){G.streakT=3+Math.random()*5;spawnStreak();}
  if(G.state!=='paused'){
    if(G.t-G.lastPointer>2.5)G.shipTX=W/2+Math.sin(G.t*.5)*W*.15;
    G.shipX+=(G.shipTX-G.shipX)*Math.min(1,dt*7);
  }
  if(G.state==='playing')updatePlaying(dt);
  else if(G.state==='menu'||G.state==='over')updateAmbient(dt);
  if(G.state!=='paused')updateFx(dt);}

function updatePlaying(dt){
  if(G.waveDelay>0){G.waveDelay-=dt;}
  else{G.spawnT-=dt;
    if(G.toSpawn>0&&G.asteroids.length<G.concurrent&&G.spawnT<=0){
      spawnAsteroid(false);G.toSpawn--;G.spawnT=G.interval*(.7+Math.random()*.6);}}
  for(let i=G.asteroids.length-1;i>=0;i--){const a=G.asteroids[i];
    a.x+=a.vx*dt;a.y+=a.vy*dt;a.rot+=a.rs*dt;if(a.errorT>0)a.errorT-=dt;
    if(!a.shard&&a.r*2<W){if(a.x<a.r){a.x=a.r;a.vx=Math.abs(a.vx);}else if(a.x>W-a.r){a.x=W-a.r;a.vx=-Math.abs(a.vx);}}
    if(a.y+a.r*.35>=lineY())breach(a);}
  if(G.toSpawn===0&&G.asteroids.length===0&&G.waveDelay<=0){
    const bonus=50*G.wave; G.score+=bonus; setScore();
    G.floats.push({x:W/2,y:lineY()-90,txt:'WAVE CLEAR +'+bonus,t:0,color:'#7dffd9'});
    sfx.wave(); startWave(G.wave+1);}}

function updateAmbient(dt){
  if(G.state==='menu'&&G.asteroids.length<4&&Math.random()<dt*1.1)spawnAsteroid(true);
  for(let i=G.asteroids.length-1;i>=0;i--){const a=G.asteroids[i];
    a.x+=a.vx*dt;a.y+=a.vy*dt;a.rot+=a.rs*dt;
    if(a.ambient){a.autoT-=dt;
      if(a.autoT<=0){a.autoT=.26+Math.random()*.34;a.prog++;
        if(a.prog>=a.word.length){explodeAt(a.x,a.y,['#9db1d6','#5d6c8e'],10);G.rings.push({x:a.x,y:a.y,r0:a.r*.4,max:a.r*2.6,t:0,color:'rgba(125,255,217,'});G.asteroids.splice(i,1);continue;}}}
    if(a.y-a.r>H+40)G.asteroids.splice(i,1);}}

function updateFx(dt){
  for(let i=G.parts.length-1;i>=0;i--){const p=G.parts[i];p.life+=dt;
    if(p.life>=p.max){G.parts.splice(i,1);continue;}

    if(p.type==='shard'){p.rot+=p.vr*dt;p.vx*=1-dt*1.4;p.vy*=1-dt*1.4;}
    if(p.type==='streak'){} }
  for(let i=G.beams.length-1;i>=0;i--){G.beams[i].t+=dt/.15;if(G.beams[i].t>=1)G.beams.splice(i,1);}
  for(let i=G.rings.length-1;i>=0;i--){G.rings[i].t+=dt/.55;if(G.rings[i].t>=1)G.rings.splice(i,1);}
