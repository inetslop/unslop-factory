function doReshuffle(){
  let tries=0;
  do{for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++)grid[r][c].t=randType();tries++;}
  while((findMatches().set.size>0||!hasMove())&&tries<400);

function shieldHit(){
  S.shields--;drawShields();SFX.shield();S.flash=.55;S.shake=14;

function gameOver(){
  S.screen='over';S.overT=1.0;S.overShown=false;S.sel=null;SFX.over();

  S.boardDead=true;S.shake=18;S.flash=.35;
  if(S.score>S.hi){S.hi=S.score;saveHi(S.hi);S.newHi=true;}else S.newHi=false;
}

function update(dt){
  // ambience always
  for(const s of stars){s.y+=(4+s.r*9)*dt;if(s.y>H+2){s.y=-2;s.x=Math.random()*W;}}
  S.shootT-=dt;
  if(S.shootT<=0){S.shootT=rand(3,8);shoot={x:rand(W*.1,W*.9),y:rand(20,H*.35),vx:(Math.random()<.5?-1:1)*rand(260,420),vy:rand(90,170),life:.8};}
  if(shoot){shoot.x+=shoot.vx*dt;shoot.y+=shoot.vy*dt;shoot.life-=dt;if(shoot.life<=0)shoot=null;}

  // fx always
  for(const p of particles){p.vy+=p.g*dt;p.x+=p.vx*dt;p.y+=p.vy*dt;p.life-=dt;}
  particles=particles.filter(p=>p.life>0);
  for(const b of beams)b.life-=dt;
  beams=beams.filter(b=>b.life>0);
  for(const t of texts){t.life-=dt*.9;t.y+=t.vy*dt;}
  texts=texts.filter(t=>t.life>0);
  S.shake*=Math.exp(-5*dt);if(S.shake<.1)S.shake=0;
  S.flash=Math.max(0,S.flash-dt*1.4);
  S.recoil*=Math.exp(-7*dt);
  S.boardFlash=Math.max(0,S.boardFlash-dt*1.2);
  S.aimCur+=(S.aimTgt-S.aimCur)*Math.min(1,dt*14);
  if(S.banner){S.banner.t+=dt;if(S.banner.t>1.5)S.banner=null;}

  // march (attract + play)
  if(S.screen==='play'||S.screen==='start'){
    S.marchT+=dt;
    if(S.marchT>=.5){S.marchT-=.5;S.frame^=1;if(S.screen==='play')SFX.thump(S.frame);}
  }

  if(S.screen==='play'){
    S.hintT-=dt;hintEl.classList.toggle('off',S.hintT<=0);
    if(S.dropPause>0)S.dropPause-=dt;else S.drop+=S.speed*dt;
    if(S.drop>=maxDrop){shieldHit();return;}
    let falling=false;
    for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++){const o=grid[r][c];
      if(o&&o.dy>0){falling=true;o.vy=Math.min(42,o.vy+70*dt);o.dy-=o.vy*dt;
        if(o.dy<=0){o.dy=0;o.vy=0;}}}
    switch(S.state){
      case 'swap':S.swap.q+=dt/.16;if(S.swap.q>=1){S.swap.q=1;commitSwap();}break;
      case 'revert':S.swap.q-=dt/.16;if(S.swap.q<=0){S.swap=null;S.state='idle';SFX.bad();}break;
      case 'clear':S.clearT+=dt/.24;if(S.clearT>=1)doGravity();break;
      case 'fall':if(!falling)settle();break;
    }
  } else if(S.screen==='over'){
    S.overT-=dt;
    if(S.overT<=0&&!S.overShown){S.overShown=true;showOverlay('over');}
  }
}

  for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++){
    const o=grid[r][c];if(!o)continue;
    if(sw&&((sw.a.r===r&&sw.a.c===c)||(sw.b.r===r&&sw.b.c===c)))continue;
    const x=boardX+c*cell+cell/2,y=boardY+S.drop+r*cell-o.dy*cell+cell/2;
    if(o.dead){if(S.state!=='clear')continue;const k=1-S.clearT;if(k>0)drawTile(o.t,x,y,k,k);continue;}
    let sc=1,yy=y;
    if(S.sel&&S.sel.r===r&&S.sel.c===c&&!sw){yy-=3;sc=1.07;}
    drawTile(o.t,x,yy,sc,1);
  }
  if(sw){
    const q=easeInOut(clamp(sw.q,0,1)),pa=cellPos(sw.a),pb=cellPos(sw.b);
    const sc=1+.18*Math.sin(Math.PI*q);
    drawTile(grid[sw.a.r][sw.a.c].t,pa.x+(pb.x-pa.x)*q,pa.y+(pb.y-pa.y)*q,sc,1);
    drawTile(grid[sw.b.r][sw.b.c].t,pb.x+(pa.x-pb.x)*q,pb.y+(pa.y-pb.y)*q,sc,1);
  }
  if(S.screen==='play'){
    if(S.hover&&!sw){const x=boardX+S.hover.c*cell,y=boardY+S.drop+S.hover.r*cell;
      ctx.strokeStyle='rgba(160,230,255,.3)';ctx.lineWidth=1.5;ctx.strokeRect(x+3,y+3,cell-6,cell-6);}

  }
}
