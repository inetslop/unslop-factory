      let dx=b.x-cx, dy=b.y-cy;

      if(d2 > b.r*b.r) continue;
      let d=Math.sqrt(d2), nx,ny;

        const l=b.x-B.x, r=B.x+B.w-b.x, t=b.y-B.y, bt=B.y+B.h-b.y;

function floater(txt,x,y,col,size,life){floaters.push({txt,x,y,col:col||'#fff8e6',size:size||16,t:0,life:life||1.2});}
function dust(x,y,n,col){for(let i=0;i<(n||6);i++)particles.push({type:'dust',x:x+(Math.random()-.5)*14,y:y-2,vx:(Math.random()-.5)*120,vy:-Math.random()*70,t:0,life:.5+Math.random()*.3,size:3+Math.random()*4,col:col||'rgba(214,190,150,'});}
function droplets(x,y,kind){const c=kind==='lava'?['#ffd23e','#ff8c2e','#fff']:['#bff1fb','#7fdcef','#fff'];
 for(let i=0;i<12;i++)particles.push({type:'drop',x:x+(Math.random()-.5)*16,y,vx:(Math.random()-.5)*260,vy:-140-Math.random()*220,t:0,life:.6+Math.random()*.4,size:2+Math.random()*3,col:c[i%3]});}
function poof(x,y){for(let i=0;i<10;i++){const a=Math.random()*6.28;particles.push({type:'dust',x,y,vx:Math.cos(a)*160,vy:Math.sin(a)*160-40,t:0,life:.5,size:4+Math.random()*4,col:'rgba(255,255,255,'});}}
function confetti(x,y){const cols=['#ffd23e','#ef4b45','#3ecf6f','#4fb3ff','#ff9de2','#fff8e6'];
 for(let i=0;i<90;i++){const a=Math.random()*6.28,sp=120+Math.random()*380;
 particles.push({type:'conf',x,y,vx:Math.cos(a)*sp,vy:Math.sin(a)*sp-260,t:0,life:1.4+Math.random()*.9,size:4+Math.random()*5,col:cols[i%6],rot:Math.random()*6.28,vr:(Math.random()-.5)*14});}}
function updateParticles(dt){for(let i=particles.length-1;i>=0;i--){const p=particles[i];p.t+=dt;
 if(p.type==='conf'){p.vy+=900*dt;p.vx*=Math.exp(-1.1*dt);p.rot+=p.vr*dt;}
 else if(p.type==='drop'){p.vy+=1500*dt;}
 else if(p.type==='bubble'){p.y-=32*dt;p.size+=6*dt;}
 else{p.x+=p.vx*dt;p.y+=p.vy*dt;p.vx*=Math.exp(-3*dt);p.vy*=Math.exp(-3*dt);}
 if(p.t>=p.life)particles.splice(i,1);}}
function updateFloaters(dt){for(let i=floaters.length-1;i>=0;i--){const f=floaters[i];f.t+=dt;f.y-=34*dt;if(f.t>=f.life)floaters.splice(i,1);}}

/* ================= PHYSICS ================= */
function windAt(x,y){let ax=0,ay=0;for(const z of world.wind){if(x>z.x&&x<z.x+z.w&&y>z.y&&y<z.y+z.h){ax+=z.dx||0;ay+=z.dy||0;}}return[ax,ay];}
function updateMovers(){for(const m of world.movers){const a=m.mv,v=Math.sin(time*2*Math.PI/a.period+(a.phase||0));
 const nx=a.axis==='x'?a.mid-a.amp*v:m.x, ny=a.axis==='y'?a.mid-a.amp*v:m.y;
 m.dx=nx-m.x;m.dy=ny-m.y;m.x=nx;m.y=ny;}}
function collideBall(b){b.onGround=false;b.onMover=null;b.groundType='dirt';b.springCd-=1/60;
 for(let it=0;it<2;it++)for(const B of SOLIDS){
  const cx=Math.max(B.x,Math.min(b.x,B.x+B.w)),cy=Math.max(B.y,Math.min(b.y,B.y+B.h));
  let dx=b.x-cx,dy=b.y-cy,d2=dx*dx+dy*dy;
  if(d2>b.r*b.r)continue;
  let d=Math.sqrt(d2),nx,ny;
  if(d<1e-6){const l=b.x-B.x,r=B.x+B.w-b.x,t=b.y-B.y,bt=B.y+B.h-b.y,mn=Math.min(l,r,t,bt);
   if(mn===l){nx=-1;ny=0;}else if(mn===r){nx=1;ny=0;}else if(mn===t){nx=0;ny=-1;}else{nx=0;ny=1;}}
  else{nx=dx/d;ny=dy/d;}
  const pen=b.r-d;b.x+=nx*pen;b.y+=ny*pen;
  const vn=b.vx*nx+b.vy*ny;
  if(vn<0){const rest=B.type==='sand'?.08:.45;
   b.vx-=(1+rest)*vn*nx;b.vy-=(1+rest)*vn*ny;b.vx*=.985;
   if(-vn>220&&time-lastHitS>.07){lastHitS=time;SFX.bounce(-vn);if(-vn>900)shake=Math.max(shake,3);}}
  if(ny<-.6){b.onGround=true;if(B.type==='mover')b.onMover=B;b.groundType=B.type||'dirt';
   if(B.type==='spring'&&b.springCd<=0&&b.vy>=-50){b.vy=-1080;b.springCd=.4;B.springT=.3;SFX.spring();dust(b.x,B.y+10,8);}}}
}
function moveBox(e,dt){
 e.x+=e.vx*dt;
 for(const B of SOLIDS){if(e.x+e.w/2>B.x&&e.x-e.w/2<B.x+B.w&&e.y+e.h/2>B.y&&e.y-e.h/2<B.y+B.h){
  if(e.vx>0)e.x=B.x-e.w/2;else if(e.vx<0)e.x=B.x+B.w+e.w/2;e.vx=0;}}
 e.y+=e.vy*dt;e.onGround=false;e.onMover=null;e.groundBox=null;
 for(const B of SOLIDS){if(e.x+e.w/2>B.x&&e.x-e.w/2<B.x+B.w&&e.y+e.h/2>B.y&&e.y-e.h/2<B.y+B.h){
  if(e.vy>0){e.y=B.y-e.h/2;e.onGround=true;if(B.type==='mover')e.onMover=B;e.groundBox=B;}
  else if(e.vy<0){e.y=B.y+B.h+e.h/2;}e.vy=0;}}
 e.groundType=e.groundBox?(e.groundBox.type||'dirt'):'dirt';
}

function updateGame(dt){

 if(sinkT>=0&&sinkT<1){sinkStep(dt);return;}

 /* --- golfer --- */
 const g=golfer;
 if(kbAim){
  if(K['arrowleft']||K['a'])kbAngle-=2.4*dt;
  if(K['arrowright']||K['d'])kbAngle+=2.4*dt;
  if(K['arrowup']||K['w'])kbPower=Math.min(1,kbPower+1.3*dt);
  if(K['arrowdown']||K['s'])kbPower=Math.max(0,kbPower-1.3*dt);
  g.vx*=Math.exp(-8*dt);
 }else{
  const dir=((K['d']||K['arrowright'])?1:0)-((K['a']||K['arrowleft'])?1:0);
  if(dir)g.facing=dir;
  const maxS=g.groundType==='sand'?150:260, acc=g.onGround?2400:1400;
  if(dir){g.vx+=dir*acc*dt;g.vx=clamp(g.vx,-maxS,maxS);}
  else{g.vx*=Math.exp(-dt*(g.onGround?14:3));if(Math.abs(g.vx)<5)g.vx=0;}
 }
 const [wax,way]=windAt(g.x,g.y);
 g.vx+=wax*.45*dt;g.vy+=(GRAV+way)*dt;g.vy=Math.min(g.vy,1200);
 g.jumpBuf-=dt;g.coyote-=dt;

 // golfer hazards
 for(const z of world.water)if(g.x>z.x&&g.x<z.x+z.w&&g.y+13>z.sy+8){respawnGolfer('water');break;}

 if(state!=='play')return;
 for(const s of world.spikes)if(g.x+8>s.x&&g.x-8<s.x+s.w&&g.y+13>s.y-16&&g.y-13<s.y){respawnGolfer('spike');break;}
 if(state!=='play')return;
 if(g.y>VH+70)respawnGolfer('fall');

 /* --- ball --- */
 const b=ball;
 if(!b.atRest){
  const sd=dt/3;
  for(let s=0;s<3;s++){const [ax,ay]=windAt(b.x,b.y);
   b.vx+=ax*sd;b.vy+=(GRAV+ay)*sd;
   const sp=Math.hypot(b.vx,b.vy);if(sp>1700){b.vx*=1700/sp;b.vy*=1700/sp;}
   b.x+=b.vx*sd;b.y+=b.vy*sd;collideBall(b);}
  if(b.onGround){const fr=b.groundType==='sand'?9:2.4;b.vx*=Math.exp(-fr*dt);if(Math.abs(b.vx)<7)b.vx=0;}

  if(b.onGround&&Math.abs(b.vx)<8&&Math.abs(b.vy)<40){b.restT+=dt;if(b.restT>.25){b.atRest=true;b.vx=0;b.vy=0;}}
  else b.restT=0;
 }
 // ball hazards
 if(sinkT<0){
  for(const z of world.water)if(b.x>z.x&&b.x<z.x+z.w&&b.y+b.r*.5>z.sy){droplets(b.x,z.sy,'water');penalty('SPLASH','#ff5d52');break;}
  if(sinkT<0)for(const z of world.lava)if(b.x>z.x&&b.x<z.x+z.w&&b.y+b.r*.5>z.sy){droplets(b.x,z.sy,'lava');penalty('LAVA!','#ff8c2e');break;}

   if((b.x-cx)**2+(b.y-cy)**2<b.r*b.r){penalty('SPIKES!');break;}}
  if(sinkT<0&&(b.x<-30||b.x>world.W+30||b.y>VH+70))penalty('LOST BALL');
 }

  }}
 // camera
 const focus=(kbAim||charging)?(g.x+b.x)/2:g.x;

}
