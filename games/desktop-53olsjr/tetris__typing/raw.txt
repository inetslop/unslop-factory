function tick(dt){
  const dtS=dt/1000,now=performance.now();
  updateRain(dtS);
  for(let i=sparks.length-1;i>=0;i--){const s=sparks[i];s.life-=dtS;s.y+=s.vy*dtS;if(s.life<=0)sparks.splice(i,1);}
  if(shake>0){shake*=Math.pow(.0018,dtS);if(shake<.15)shake=0;}
  if(state!=='paused')updateParticles(dtS);
  if(state==='running'){
    if(buf&&now-lastKeyT>1200){buf='';renderBuf();updateChips([]);}
    if(dasDir&&keyState[dasDir]){dasT+=dt;if(dasT>160){dasAcc+=dt;while(dasAcc>=42){dasAcc-=42;doMove(dasDir==='L'?-1:1);}}}else dasAcc=0;
    if(keyState.D){softAcc+=dt;while(softAcc>=45){softAcc-=45;softDrop();}}
    if(clearing){clearing.t+=dt;if(clearing.t>=CLEAR_MS)finishClear();}
    else if(cur){
      const g=!fits(cur.x,cur.y+1,cur.type,cur.rot);
      if(g){lockTimer+=dt;if(lockTimer>=LOCK_DELAY)doLock();}
      else{gravAccum+=dt;const gm=gravityMs();let guard=0;
        while(gravAccum>=gm&&guard++<8){gravAccum-=gm;cur.y++;if(!fits(cur.x,cur.y+1,cur.type,cur.rot))break;}}
    }
    const el=(now-typeStartT)/60000,target=charsCredited/5/Math.max(el,.15);
    dispWPM+=(target-dispWPM)*Math.min(1,dt/400);
    hudT+=dt;if(hudT>180){hudT=0;wpmEl.textContent=Math.round(dispWPM);accEl.textContent=accPct()+'%';}
  }
}
