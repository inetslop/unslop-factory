function trySwap(a,b){
  if(state!=='play'||paused||seq.mode!=='idle')return;
  var g1=grid[a.r][a.c], g2=grid[b.r][b.c];
  if(!g1||!g2)return;
  sel=null; hintPair=null; lastAction=time; sfx('swap');
  seq.mode='swapping';
  tween(.16,function(k){
    g1.cx=a.c+(b.c-a.c)*k; g1.cy=a.r+(b.r-a.r)*k;
    g2.cx=b.c+(a.c-b.c)*k; g2.cy=b.r+(a.r-b.r)*k;
  },function(){
    grid[a.r][a.c]=g2; grid[b.r][b.c]=g1;
    g1.cx=b.c; g1.cy=b.r; g2.cx=a.c; g2.cy=a.r;
    var n1=g1.special&&g1.special.kind==='nova', n2=g2.special&&g2.special.kind==='nova';
    if(n1||n2){ moves--; hudMoves(); beginNova(g1,g2,a,b,n1,n2); return; }
    var m=findMatches(idx(b.r,b.c));
    if(m.cells.size===0){
      sfx('bad');
      tween(.16,function(k){
        g1.cx=b.c+(a.c-b.c)*k; g1.cy=b.r+(a.r-b.r)*k;
        g2.cx=a.c+(b.c-a.c)*k; g2.cy=a.r+(b.r-a.r)*k;
      },function(){
        grid[a.r][a.c]=g1; grid[b.r][b.c]=g2;
        g1.cx=a.c; g1.cy=a.r; g2.cx=b.c; g2.cy=b.r;
        seq.mode='idle';
      });
    } else { moves--; hudMoves(); clearStep(m.cells,m.spawns,1); }
  });
}

function beginNova(g1,g2,cellA,cellB,n1,n2){
  var set=new Set();
  function allOf(ty){ for(var r=0;r<ROWS;r++)for(var c=0;c<COLS;c++)if(grid[r][c].type===ty)set.add(idx(r,c)); }
  if(n1&&n2){ allOf(g1.type); allOf(g2.type); }
  else{
    var N=n1?g1:g2, O=n1?g2:g1, oCell=n1?cellA:cellB;
    allOf(N.type);
    if(O.special&&O.special.kind==='cross'){
      for(var i=0;i<COLS;i++)set.add(idx(oCell.r,i));
      for(i=0;i<ROWS;i++)set.add(idx(i,oCell.c)); }
  }
  var px=n1?ox+(cellB.c+.5)*cell:ox+(cellA.c+.5)*cell, py=n1?oy+(cellB.r+.5)*cell:oy+(cellA.r+.5)*cell;
  rings.push({x:px,y:py,r:cell*.4,vr:cell*11,t:0,life:.6,col:'#fff3c4'});
  sfx('nova'); shake=12;
  clearStep(set,[],1);
}

}

function startLevel(){
  tweens.length=0; particles.length=0; texts.length=0; rings.length=0; shake=0;
  sel=null; hintPair=null; dispScore=0; shownScore=-1; shownGauge=-1;
  levelScore=0; moves=20; target=900+(level-1)*550;

/* ============================== UPDATE ============================== */
var G_FALL=95, V_MAX=34;
function updateBoard(dt){
  if(state!=='play')return;
  var r,c,g;
  for(r=0;r<ROWS;r++)for(c=0;c<COLS;c++){ g=grid[r][c]; if(!g)continue;
    if(g.born<1)g.born=Math.min(1,g.born+dt/0.32);
    if(g.sq>0)g.sq=Math.max(0,g.sq-dt*5); }
  switch(seq.mode){
    case 'popping':{
      var done=true;
      for(r=0;r<ROWS;r++)for(c=0;c<COLS;c++){ g=grid[r][c];
        if(g&&g.popT>=0&&g.popT<1){ g.popT=Math.min(1,g.popT+dt/0.26); done=false; } }
      if(done){ for(r=0;r<ROWS;r++)for(c=0;c<COLS;c++){ g=grid[r][c]; if(g&&g.popT>=0)grid[r][c]=null; }
        applyGravity(); seq.mode='falling'; seq.fallT=0; }
      break; }
    case 'falling':{
      seq.fallT+=dt; var settled=true;
      for(r=0;r<ROWS;r++)for(c=0;c<COLS;c++){ g=grid[r][c]; if(!g)continue;
        if(g.cy<g.ty){ settled=false; g.vy=Math.min(V_MAX,g.vy+G_FALL*dt); g.cy+=g.vy*dt;
          if(g.cy>=g.ty){ g.cy=g.ty; if(g.vy>17)g.sq=1; g.vy=0; } }
        else g.cy=g.ty; }
      if(settled&&seq.fallT>0.06){
        var m=findMatches();
        if(m.cells.size) clearStep(m.cells,m.spawns,seq.combo+1); else endResolve(); }
      break; }
    case 'shuffling':{
      seq.t+=dt; var D=0.28,k=Math.min(1,seq.t/D);
      for(r=0;r<ROWS;r++)for(c=0;c<COLS;c++){ g=grid[r][c]; if(g)g.alpha=(seq.phase===0)?1-k:k; }
      if(k>=1){ if(seq.phase===0){ reassignTypes(); seq.phase=1; seq.t=0; }
        else { for(r=0;r<ROWS;r++)for(c=0;c<COLS;c++){ g=grid[r][c]; if(g)g.alpha=1; } seq.mode='idle'; } }
      break; }
  }
  // hint
  if(seq.mode==='idle'&&!hintPair&&time-lastAction>6){ hintPair=findValidMove(); }
}
function updateFx(dt){
  var i,p;
  for(i=particles.length-1;i>=0;i--){ p=particles[i]; p.t+=dt;
    if(p.t>p.life){ particles.splice(i,1); continue; }
    p.vy+=p.g*dt; p.x+=p.vx*dt; p.y+=p.vy*dt; p.rot+=p.vr*dt; }
  for(i=rings.length-1;i>=0;i--){ p=rings[i]; p.t+=dt; p.r+=p.vr*dt; if(p.t>p.life)rings.splice(i,1); }
  for(i=texts.length-1;i>=0;i--){ p=texts[i]; p.t+=dt; p.y-=28*dt; if(p.t>p.life)texts.splice(i,1); }
  shake=Math.max(0,shake-dt*30);
}

var motes=[];
(function(){ for(var i=0;i<46;i++)motes.push({x:Math.random(),y:Math.random(),sp:.012+Math.random()*.035,
  r:1+Math.random()*2.4,a:.08+Math.random()*.3,ws:.4+Math.random()*1.4,sw:6+Math.random()*22,ph:Math.random()*7}); })();

function drawBG(){
  var g=ctx.createLinearGradient(0,0,0,H);
  g.addColorStop(0,'#0c3a4d'); g.addColorStop(.5,'#072635'); g.addColorStop(1,'#031219');
  ctx.fillStyle=g; ctx.fillRect(-24,-24,W+48,H+48);
  // god rays
  ctx.save(); ctx.globalCompositeOperation='lighter';
  for(var i=0;i<3;i++){
    var a=(i-1)*.55+Math.sin(time*.13+i*2.1)*.13;
    ctx.save(); ctx.translate(W*.5,-30); ctx.rotate(a*.45);
    var rg=ctx.createLinearGradient(0,0,0,H*1.15);
    rg.addColorStop(0,'rgba(190,240,255,.09)'); rg.addColorStop(.75,'rgba(190,240,255,0)');
    ctx.fillStyle=rg; ctx.beginPath();
    var w0=W*.06,w1=W*.2;
    ctx.moveTo(-w0,0); ctx.lineTo(w0,0); ctx.lineTo(w1,H*1.2); ctx.lineTo(-w1,H*1.2); ctx.closePath(); ctx.fill();
    ctx.restore(); }
  ctx.restore();
  // motes
  for(i=0;i<motes.length;i++){ var m=motes[i];
    var yy=((m.y-time*m.sp)%1+1)%1, xx=(m.x*W+Math.sin(time*m.ws+m.ph)*m.sw+W)%W;
    ctx.globalAlpha=m.a*(0.6+0.4*Math.sin(time*2+m.ph));
    ctx.fillStyle='rgba(185,232,255,1)';
    ctx.beginPath(); ctx.arc(xx,yy*H,m.r,0,7); ctx.fill(); }
  ctx.globalAlpha=1;
  // vignette
  var v=ctx.createRadialGradient(W/2,H/2,Math.min(W,H)*.35,W/2,H/2,Math.max(W,H)*.75);
  v.addColorStop(0,'rgba(0,0,0,0)'); v.addColorStop(1,'rgba(0,0,0,.5)');
  ctx.fillStyle=v; ctx.fillRect(-24,-24,W+48,H+48);
}

function drawBoard(){
  var i,x,y;
  rr(ctx,ox-12,oy-12,bw+24,bh+24,16);
  ctx.fillStyle='rgba(5,24,33,.68)'; ctx.fill();
  var bg=ctx.createLinearGradient(ox,oy,ox+bw,oy+bh);
  bg.addColorStop(0,'#8a5a1c'); bg.addColorStop(.5,'#ffd873'); bg.addColorStop(1,'#8a5a1c');
  ctx.strokeStyle=bg; ctx.lineWidth=3; ctx.stroke();
  rr(ctx,ox-6,oy-6,bw+12,bh+12,11);
  ctx.strokeStyle='rgba(255,216,115,.35)'; ctx.lineWidth=1; ctx.stroke();
  // corner diamonds
  var cs=[[ox-12,oy-12],[ox+bw+12,oy-12],[ox-12,oy+bh+12],[ox+bw+12,oy+bh+12]];
  for(i=0;i<4;i++){ ctx.save(); ctx.translate(cs[i][0],cs[i][1]); ctx.rotate(Math.PI/4);
    ctx.fillStyle='#ffd873'; ctx.strokeStyle='#5e3d10'; ctx.lineWidth=1;
    ctx.fillRect(-4,-4,8,8); ctx.strokeRect(-4,-4,8,8); ctx.restore(); }
  // cells + sockets
  for(var r=0;r<ROWS;r++)for(var c=0;c<COLS;c++){
    x=ox+c*cell; y=oy+r*cell;
    ctx.fillStyle=(r+c)%2?'rgba(150,215,235,.05)':'rgba(150,215,235,.022)';
    ctx.fillRect(x,y,cell,cell);
    ctx.strokeStyle='rgba(170,225,255,.06)'; ctx.lineWidth=1;
    ctx.beginPath(); ctx.arc(x+cell/2,y+cell/2,cell*.36,0,7); ctx.stroke(); }
  // shimmer sweep
  ctx.save(); rr(ctx,ox-12,oy-12,bw+24,bh+24,16); ctx.clip();
  var bx=ox+((time*cell*3.5)%(bw+cell*8))-cell*4;
  ctx.fillStyle='rgba(200,245,255,.045)';
  ctx.beginPath(); ctx.moveTo(bx,oy); ctx.lineTo(bx+cell*1.6,oy);
  ctx.lineTo(bx-cell*.35+bw*.0,oy+bh); ctx.lineTo(bx-cell*1.95,oy+bh); ctx.closePath(); ctx.fill();
  ctx.restore();
}

function drawGem(g,r,c){
  var p=g.popT, sc=1, al=g.alpha;
  if(p>=0){ var q=Math.min(1,p);
    if(q<.25)sc=1+q*1.6; else { sc=Math.max(0,1.4*(1-(q-.25)/.75)); al*=Math.max(0,1-(q-.25)/.75); } }
  if(g.born<1)sc*=eob(g.born);
  var sx=sc*(1+.3*g.sq), sy=sc*(1-.3*g.sq);
  var idle=(g.popT<0&&g.cy===g.ty&&seq.mode==='idle'&&state==='play');
  var x=ox+(g.cx+.5)*cell, y=oy+(g.cy+.5)*cell;
  if(idle)y+=Math.sin(time*1.6+c*2.7+r*1.3)*cell*.014;
  if(sel&&state==='play'&&seq.mode==='idle'&&Math.round(g.cx)===sel.c&&Math.round(g.cy)===sel.r){ sx*=1+.05*Math.sin(time*9); sy*=1+.05*Math.sin(time*9); }
  // nova aura
  if(g.special&&g.special.kind==='nova'){
    var rot=time*.9+r+c, R1=cell*.62,R2=cell*.34;
    ctx.save(); ctx.translate(x,y); ctx.rotate(rot);
    ctx.beginPath();
    for(var i=0;i<16;i++){ var a=i*Math.PI/8, rad=(i%2===0)?R1:R2;
      if(i===0)ctx.moveTo(rad*Math.cos(a),rad*Math.sin(a)); else ctx.lineTo(rad*Math.cos(a),rad*Math.sin(a)); }
    ctx.closePath();
    ctx.fillStyle='rgba(255,230,150,.26)'; ctx.fill();
    ctx.strokeStyle='rgba(255,216,115,.55)'; ctx.lineWidth=2; ctx.stroke();
    ctx.restore(); }
  ctx.save(); ctx.translate(x,y); ctx.scale(sx,sy); ctx.globalAlpha=al;
  var S=cell*.98; ctx.drawImage(sprites[g.type],-S/2,-S/2,S,S);
  if(g.special&&g.special.kind==='cross'){
    var u=cell, pul=.55+.35*Math.sin(time*5+r+c);
    ctx.lineCap='round';
    ctx.strokeStyle='rgba(255,216,115,'+(pul*.4)+')'; ctx.lineWidth=u*.16;
    ctx.beginPath(); ctx.moveTo(-u*.3,-u*.3); ctx.lineTo(u*.3,u*.3); ctx.moveTo(u*.3,-u*.3); ctx.lineTo(-u*.3,u*.3); ctx.stroke();
    ctx.strokeStyle='rgba(255,246,216,'+pul+')'; ctx.lineWidth=u*.07;
    ctx.beginPath(); ctx.moveTo(-u*.3,-u*.3); ctx.lineTo(u*.3,u*.3); ctx.moveTo(u*.3,-u*.3); ctx.lineTo(-u*.3,u*.3); ctx.stroke();
    ctx.strokeStyle='rgba(255,216,115,'+(pul*.5)+')'; ctx.lineWidth=u*.04;
    ctx.beginPath(); ctx.arc(0,0,u*.44*(1+.05*Math.sin(time*5)),0,7); ctx.stroke(); }
  ctx.restore();
}

function drawCellMarks(){
  var i,x,y,s;
  if(state==='play'){
    if(hintPair&&seq.mode==='idle'){
      s=.5+.3*Math.sin(time*6);
      for(i=0;i<2;i++){ x=ox+hintPair[i][1]*cell+cell/2; y=oy+hintPair[i][0]*cell+cell/2;
        ctx.strokeStyle='rgba(255,216,115,'+s+')'; ctx.lineWidth=3;
        ctx.beginPath(); ctx.arc(x,y,cell*(.42+.06*Math.sin(time*6)),0,7); ctx.stroke(); } }
    if(sel&&seq.mode==='idle'){ x=ox+sel.c*cell; y=oy+sel.r*cell;
      var pu=1+.03*Math.sin(time*9);
      ctx.strokeStyle='rgba(255,216,115,.3)'; ctx.lineWidth=8;
      rr(ctx,x+4,y+4,cell-8,cell-8,7); ctx.stroke();
      ctx.strokeStyle='#ffd873'; ctx.lineWidth=3;
      rr(ctx,(x+cell/2)-(cell*pu)/2,(y+cell/2)-(cell*pu)/2,cell*pu,cell*pu,8); ctx.stroke(); }
    if(showCursor&&cur){ x=ox+cur.c*cell; y=oy+cur.r*cell; var L=cell*.3, I=5;
      ctx.strokeStyle='#ffe9b0'; ctx.lineWidth=3; ctx.lineCap='round';
      var P=[[x+I,y+I,1,1],[x+cell-I,y+I,-1,1],[x+I,y+cell-I,1,-1],[x+cell-I,y+cell-I,-1,-1]];
      for(i=0;i<4;i++){ ctx.beginPath();
        ctx.moveTo(P[i][0]+P[i][2]*L,P[i][1]); ctx.lineTo(P[i][0],P[i][1]); ctx.lineTo(P[i][0],P[i][1]+P[i][3]*L); ctx.stroke(); } }
  }
}

function drawFx(){
  var i,p,k;
  for(i=0;i<particles.length;i++){ p=particles[i]; k=p.t/p.life;
    ctx.globalAlpha=Math.max(0,1-k); ctx.fillStyle=p.col;
    ctx.save(); ctx.translate(p.x,p.y); ctx.rotate(p.rot);
    if(p.shape==='dot'){ ctx.beginPath(); ctx.arc(0,0,p.size*.6,0,7); ctx.fill(); }
    else if(p.shape==='rect'){ ctx.fillRect(-p.size/2,-p.size*.3,p.size,p.size*.6); }
    else { ctx.fillRect(-p.size/2,-p.size*.25,p.size,p.size*.5); }
    ctx.restore(); }
  ctx.globalAlpha=1;
  for(i=0;i<rings.length;i++){ p=rings[i]; k=p.t/p.life;
    ctx.globalAlpha=(1-k)*.8; ctx.strokeStyle=p.col; ctx.lineWidth=3*(1-k)+1;
    ctx.beginPath(); ctx.arc(p.x,p.y,p.r,0,7); ctx.stroke(); }
  ctx.globalAlpha=1;
  for(i=0;i<texts.length;i++){ p=texts[i]; k=p.t/p.life;
    var al=k<.15?k/.15:1-Math.max(0,(k-.5)/.5), sc2=k<.18?.6+(k/.18)*.4:1;
    ctx.globalAlpha=Math.max(0,al);
    ctx.font='bold '+Math.round(p.size*sc2)+'px "Trebuchet MS",Verdana,sans-serif';
    ctx.textAlign='center'; ctx.textBaseline='middle';
    ctx.lineWidth=4; ctx.strokeStyle='rgba(3,15,22,.7)'; ctx.strokeText(p.txt,p.x,p.y);
    ctx.fillStyle=p.col; ctx.fillText(p.txt,p.x,p.y); }
  ctx.globalAlpha=1;
}

function draw(){
  ctx.setTransform(dpr,0,0,dpr,0,0);
  if(shake>0)ctx.translate((Math.random()-.5)*shake,(Math.random()-.5)*shake);
  drawBG();
  drawBoard();
  ctx.save(); rr(ctx,ox-2,oy-2,bw+4,bh+4,8); ctx.clip();
  for(var r=0;r<ROWS;r++)for(var c=0;c<COLS;c++){ var g=grid[r]&&grid[r][c]; if(g)drawGem(g,r,c); }
  ctx.restore();
  drawCellMarks();
  drawFx();
}

  if(state!=='play'&&state!=='win'){ return; }
  var s=Math.round(dispScore+=(levelScore-dispScore)*Math.min(1,dt*10));
  if(s!==shownScore){ shownScore=s; $('scoreNum').textContent=fmt(s); }
  var gp=Math.round(Math.min(1,levelScore/target)*100);
  if(gp!==shownGauge){ shownGauge=gp; $('gaugeFill').style.width=gp+'%'; }
}

function cellAt(e){
  var rect=cv.getBoundingClientRect(), x=e.clientX-rect.left, y=e.clientY-rect.top;
  var c=Math.floor((x-ox)/cell), r=Math.floor((y-oy)/cell);
  if(c<0||c>=COLS||r<0||r>=ROWS)return null;
  return {c:c,r:r};
}
var press=null;
cv.addEventListener('pointerdown',function(e){
  audio(); if(state!=='play'||paused)return;
  var p=cellAt(e); if(!p)return;
  showCursor=false;
  if(sel&&Math.abs(sel.c-p.c)+Math.abs(sel.r-p.r)===1){ trySwap(sel,p); press=null; return; }
  sel=p; sfx('select'); hintPair=null; lastAction=time;
  press={c:p.c,r:p.r,moved:false};
});
cv.addEventListener('pointermove',function(e){
  if(!press||press.moved||state!=='play'||paused)return;
  var p=cellAt(e); if(!p)return;
  var dc=p.c-press.c, dr=p.r-press.r;
  if(Math.abs(dc)+Math.abs(dr)===1&&(Math.abs(dc)>0.5*1||true)&&Math.max(Math.abs(dc),Math.abs(dr))>=1){
    // threshold by pixel distance
  }
  var rect=cv.getBoundingClientRect();
  var start={x:ox+(press.c+.5)*cell,y:oy+(press.r+.5)*cell};
  var dx=(e.clientX-rect.left)-start.x, dy=(e.clientY-rect.top)-start.y;
  if(Math.hypot(dx,dy)>cell*.42){
    press.moved=true; sel=null; hintPair=null; lastAction=time;
    var t = Math.abs(dx)>Math.abs(dy) ? {c:press.c+(dx>0?1:-1),r:press.r} : {c:press.c,r:press.r+(dy>0?1:-1)};
    if(t.c>=0&&t.c<COLS&&t.r>=0&&t.r<ROWS){ trySwap({c:press.c,r:press.r},t); } else sfx('bad');
  }
});
window.addEventListener('pointerup',function(){ press=null; });

window.addEventListener('keydown',function(e){
  var k=e.key;
  if(['ArrowUp','ArrowDown','ArrowLeft','ArrowRight',' '].indexOf(k)>=0)e.preventDefault();
  audio();
  if(state==='start'){ if(k==='Enter'||k===' '){ startGame(); } return; }
  if(state==='win'){ if(k==='Enter'||k===' '){ nextLevel(); } return; }
  if(state==='lose'){ if(k==='Enter'||k===' '||k==='r'||k==='R'){ restartAll(); } return; }
  // playing
  if(k==='m'||k==='M'){ toggleMute(); return; }
  if(k==='p'||k==='P'){ togglePause(); return; }
  if(k==='r'||k==='R'){ restartAll(); return; }
  if(paused)return;
  var arrows={ArrowUp:[0,-1],ArrowDown:[0,1],ArrowLeft:[-1,0],ArrowRight:[1,0]};
  if(arrows[k]){
    cur=cur||{c:3,r:3}; showCursor=true;
    cur.c=Math.max(0,Math.min(COLS-1,cur.c+arrows[k][0]));
    cur.r=Math.max(0,Math.min(ROWS-1,cur.r+arrows[k][1]));
    return; }
  if(k===' '||k==='Enter'){
    var p=cur||{c:3,r:3}; showCursor=true; cur=p;
    if(!sel){ sel=p; sfx('select'); hintPair=null; lastAction=time; }
    else if(Math.abs(sel.c-p.c)+Math.abs(sel.r-p.r)===1){ trySwap(sel,p); }
    else { sel=p; sfx('select'); hintPair=null; lastAction=time; } }
  if(k==='Escape'){ sel=null; }
});
