  for(let y=0;y<f.h;y++)for(let x=0;x<f.w;x++){

function tryMove(dx,dy){

 }else if(a.type==='stair'){P.rz=0;}

/* ============================== UPDATE ============================== */
function update(dt){
 tGlobal+=dt;dtLast=dt;
 if(anim){anim.t+=dt;if(anim.t>=anim.dur)finishAnim();}
 shake.t=Math.max(0,shake.t-dt);pulse=Math.max(0,pulse-dt*2.2);
 blinkT-=dt;if(blinkT<0){blink=.13;blinkT=2+Math.random()*3;}blink=Math.max(0,blink-dt);
 for(let i=parts.length-1;i>=0;i--){const p=parts[i];p.life-=dt;p.x+=p.vx*dt;p.y+=p.vy*dt;p.z+=p.vz*dt;if(p.life<=0)parts.splice(i,1);}
 if(parts.length>350)parts.splice(0,parts.length-350);
 for(let i=confetti.length-1;i>=0;i--){const c=confetti[i];c.life-=dt;c.vy+=c.g*dt;c.x+=c.vx*dt;c.y+=c.vy*dt;c.rot+=c.vr*dt;if(c.life<=0||c.y>H+40)confetti.splice(i,1);}
 if(state!=='menu'&&Math.random()<dt*5){for(const b of L.boxes)if(L.floors[b.z].pad.has(b.x+','+b.y)){sparkle(b.x,b.y,b.z);break;}}
}

function render(){

 ctx.translate(shx,shy);
 // build sorted draw list: depth = floor * 100 + (x+y) * 2 + solidness
 const items=[];
 for(let z=0;z<L.floors.length;z++){const f=L.floors[z];
  for(const kk of f.floor){const p=kk.split(','),x=+p[0],y=+p[1];items.push({key:z*100+(x+y)*2,fn:()=>drawTile(x,y,z,f,kk)});}
  for(const kk of f.wall){const p=kk.split(','),x=+p[0],y=+p[1];items.push({key:z*100+(x+y)*2+1,fn:()=>drawWall(x,y,z,f,kk)});}}
 for(const b of L.boxes)items.push({key:b.z*100+(b.x+b.y)*2+1,fn:()=>drawBox(b)});
 items.push({key:P.z*100+(P.x+P.y)*2+1,fn:drawPlayer});
 items.sort((a,b)=>a.key-b.key);
 for(const it of items)it.fn();
 // world particles
 for(const p of parts){const a=Math.max(0,p.life/p.max),[sx,sy]=prj(p.x,p.y,p.z),s=p.s*view.scale;
  ctx.fillStyle=col(p.c,a*.8);ctx.fillRect(sx-s/2,sy-s/2,s,s);}
 ctx.restore();
 // confetti (screen space)
 for(const c of confetti){const a=Math.min(1,c.life/c.max*2);
  ctx.save();ctx.translate(c.x,c.y);ctx.rotate(c.rot);ctx.fillStyle=col(c.c,a);ctx.fillRect(-c.w/2,-c.h/2,c.w,c.h);ctx.restore();}
 if(vig)ctx.drawImage(vig,0,0,W,H);
}
