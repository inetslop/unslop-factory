  if(Math.abs(dx)>1||Math.abs(dy)>1){ x=normX(x); }
  return {x:x,y:y};
}
function stepEnt(e,dt,choose){
  if(e.stuck) return;
  e.t+=e.spd*dt;
  let guard=0;
  while(e.t>=1 && guard++<4){
    e.tx=e.toX; e.ty=e.toY; e.t-=1;
    choose(e);
    if(e.stuck){ e.t=0; break; }
  }
  const p=entPos(e); e.x=p.x; e.y=p.y;
}
function reverseEnt(e){
  if(e.stuck){ e.dir={x:-e.dir.x,y:-e.dir.y}; return; }
  const fx=e.fromX,fy=e.fromY;
  e.fromX=e.toX; e.fromY=e.toY; e.toX=fx; e.toY=fy;
  e.t=1-e.t; e.dir={x:-e.dir.x,y:-e.dir.y};
}
function choosePlayer(e){
  const opts=[e.want,e.dir];
  for(const d of opts) if(d&&open(e.tx,e.ty,d)){ startStep(e,d); return; }
  e.stuck=true;
}
function mdist(ax,ay,bx,by){ let dx=ax-bx; dx=((dx%W)+W)%W; dx=Math.min(dx,W-dx); return Math.hypot(dx,ay-by); }
function ahead(n){ let x=P.tx,y=P.ty; for(let i=0;i<n;i++){ x+=P.dir.x; y+=P.dir.y; x=normX(x);} return {x:x,y:y}; }
function ghostTarget(g){
  if(g.eye) return g.home;
  if(G.fright>0) return {x:P.tx,y:P.ty};
  if(PHASES[G.phaseIdx][0]==='scatter') return g.scatter;
  if(g.name==='ACE')   return {x:P.tx,y:P.ty};
  if(g.name==='KING')  return ahead(6);
  if(g.name==='QUEEN') return ahead(4);
  return g.goal; // JACK: erratic
}
function chooseGhost(e){
  const t=ghostTarget(e);
  let dirs=DIRS.filter(d=>open(e.tx,e.ty,d)&&!(d.x===-e.dir.x&&d.y===-e.dir.y));
  if(!dirs.length) dirs=[{x:-e.dir.x,y:-e.dir.y}];
  let best=dirs[0], bs=null;
  for(const d of dirs){
    const nx=e.tx+d.x, ny=e.ty+d.y;
    const dist=mdist(nx,ny,t.x,t.y);
    let s;
    if(e.eye||e.fright) s = (Math.random()<0.65)? -dist+Math.random()*1.5 : Math.random()*4;
    else s=dist+Math.random()*0.12;
    if(bs===null||s<bs){ bs=s; best=d; }
  }
  startStep(e,best);
}

/* --------------------------- board dealing ---------------------------- */
function dealBoard(){
  const reach=reachable();
  const ex=new Set([SPAWN.x+','+SPAWN.y]);
  JOKERS.forEach(j=>ex.add(j.x+','+j.y));
  GHOST_DEFS.forEach(g=>ex.add(g.spawn.x+','+g.spawn.y));
  const cand=[];
  reach.forEach(k=>{ if(!ex.has(k)) cand.push(k); });
  for(let i=cand.length-1;i>0;i--){ const j=Math.floor(Math.random()*(i+1)); [cand[i],cand[j]]=[cand[j],cand[i]]; }
  const n=Math.min(52,cand.length), suits=['H','D','S','C'];
  G.dots=[];
  for(let i=0;i<n;i++){
    const [x,y]=cand[i].split(',').map(Number);
    G.dots.push({x:x,y:y,suit:suits[Math.floor(Math.random()*4)],born:G.dealT+(x+y)*0.035+Math.random()*0.25});
  }
  G.jokers=JOKERS.map(j=>({x:j.x,y:j.y,alive:true}));
}

/* ------------------------------ flow ---------------------------------- */
function resetPositions(){
  P=makeEnt(SPAWN.x,SPAWN.y);
  P.spd=Math.min(8.6,7+(G.level-1)*0.25);
  G.ghosts=GHOST_DEFS.map(d=>{
    const g=makeEnt(d.spawn.x,d.spawn.y);
    g.name=d.name; g.letter=d.letter; g.color=d.color; g.scatter=d.scatter;
    g.home={x:d.spawn.x,y:d.spawn.y};
    g.fright=false; g.eye=false; g.born=0; g.goalT=Math.random()*1.5;
    g.goal={x:P.tx,y:P.ty}; g.dir={x:-1,y:0}; g.want={x:-1,y:0};
    return g;
  });
  G.fright=0; G.ghostCombo=0; G.phaseIdx=0; G.phaseT=0; G.invuln=1.4;
}
function startGame(){
  G.score=0; G.lives=3; G.level=1; G.eatCount=0; G.streak=0;
  G.parts.length=0; G.floats.length=0; G.dealT=0;
  dealBoard(); resetPositions();
  G.state='playing'; SFX.start();
}
function addFloat(x,y,txt,col){ G.floats.push({x:OX+x*TILE, y:OY+y*TILE, txt:txt, col:col||'#ffe9a8', age:0}); }
function burst(x,y,suit,col,n,pw){
  for(let i=0;i<n;i++){
    const a=Math.random()*Math.PI*2, v=(1+Math.random()*2.2)*(pw||1);
    G.parts.push({x:x*TILE,y:y*TILE,vx:Math.cos(a)*v,vy:Math.sin(a)*v-1,life:.5+Math.random()*.35,age:0,
      suit:suit,col:col,sz:4+Math.random()*4,rot:Math.random()*6,vr:(Math.random()-0.5)*8});
  }
}
function eatCard(d){
  const pts=d.suit==='H'?15:10;
  G.score+=pts; G.eatCount++; G.streak++; G.streakT=1;
  SFX.card(G.streak);
  burst(d.x+.5,d.y+.5,d.suit, d.suit==='H'||d.suit==='D' ? '#e2493f':'#2b3a6b', 5);
  addFloat(d.x+.5,d.y-.1,'+'+pts, d.suit==='H'?'#ff9aa0':null);
}
function eatJoker(j){
  G.score+=100; SFX.joker();
  G.fright=frightDur(); G.frightMax=G.fright; G.ghostCombo=0;
  burst(j.x+.5,j.y+.5,'D','#ffd76a',10,1.4);
  addFloat(j.x+.5,j.y-.2,'JOKER +100','#ffd76a');
  for(const g of G.ghosts) if(!g.eye){ g.fright=true; reverseEnt(g); }
}
function eatGhost(g){
  const pts=Math.min(1600,200*Math.pow(2,G.ghostCombo++));
  G.score+=pts; SFX.ghost(); G.shake=4;
  g.eye=true; g.fright=false;
  burst(g.x,g.y,'D','#f6f1e6',8,1.3);
  addFloat(g.x,g.y-.2,'+'+pts,'#bfe9ff');
}
function playerDeath(){
  G.state='dying'; G.deathT=0; G.shake=7; SFX.death(); G.lives--;
  burst(P.x,P.y,null,'#f6f1e6',4,1);
  ['H','D','S','C'].forEach((s,i)=>burst(P.x,P.y,s,['#e2493f','#e2493f','#2b3a6b','#2b3a6b'][i],2,1.6));
}
function levelComplete(){
  G.bonus=500+G.level*250; G.score+=G.bonus; SFX.level();
  G.state='levelup'; G.levelT=0;
}

/* ------------------------------ update -------------------------------- */
function updatePlaying(dt){
  G.dealT+=dt;
  if(G.invuln>0) G.invuln-=dt;
  if(G.streakT>0){ G.streakT-=dt; if(G.streakT<=0) G.streak=0; }

  // phases
  if(G.fright>0){
    G.fright-=dt;
    if(G.fright<=0){ G.fright=0; for(const g of G.ghosts) g.fright=false; G.ghostCombo=0; }
  }else{
    G.phaseT+=dt;
    while(G.phaseIdx<PHASES.length-1 && G.phaseT>=PHASES[G.phaseIdx][1]){
      G.phaseT-=PHASES[G.phaseIdx][1]; G.phaseIdx++;
      for(const g of G.ghosts) if(!g.eye) reverseEnt(g);
    }
  }

  // player
  P.spd=Math.min(8.6,7+(G.level-1)*0.25);
  stepEnt(P,dt,choosePlayer);
  const pp=entPos(P); P.x=pp.x; P.y=pp.y;

  // eat dots
  for(let i=G.dots.length-1;i>=0;i--){
    const d=G.dots[i];
    if(Math.abs(d.x+.5-P.x)<0.45 && Math.abs(d.y+.5-P.y)<0.45){ G.dots.splice(i,1); eatCard(d); }
  }
  for(const j of G.jokers) if(j.alive && Math.abs(j.x+.5-P.x)<0.55 && Math.abs(j.y+.5-P.y)<0.55){ j.alive=false; eatJoker(j); }

  // ghosts
  const gspd=Math.min(7.8,6.2+(G.level-1)*0.2);
  for(const g of G.ghosts){
    if(g.name==='JACK'&&!g.eye){
      g.goalT-=dt;
      if(g.goalT<=0){
        g.goalT=1.4+Math.random()*1.2;
        const opts=[{x:P.tx,y:P.ty}, ahead(3), {x:Math.random()<.5?2:W-3, y:Math.random()<.5?2:H-3}];
        g.goal=opts[Math.floor(Math.random()*opts.length)];
      }
    }
    const sp=g.eye?9.5:(g.fright?4.3:gspd);
    stepEnt(g,dt,chooseGhost);
    if(g.eye && g.stuck && g.tx===g.home.x && g.ty===g.home.y) g.eye=false;
  }

  // collisions
  for(const g of G.ghosts){
    if(g.eye) continue;
    const d=mdist(g.x,g.y,P.x,P.y);
    if(d<0.55){
      if(g.fright&&G.fright>0) eatGhost(g);
      else if(G.invuln<=0){ playerDeath(); break; }
    }
  }
  if(G.dots.length===0 && G.state==='playing') levelComplete();
}

function update(dt){
  G.t+=dt;
  if(G.shake>0) G.shake=Math.max(0,G.shake-dt*14);
  // particles & floats always live
  for(let i=G.parts.length-1;i>=0;i--){ const p=G.parts[i]; p.age+=dt;
    if(p.age>p.life){ G.parts.splice(i,1); continue; }
    p.x+=p.vx*dt; p.y+=p.vy*dt; p.vy+=6*dt; p.rot+=p.vr*dt; }
  for(let i=G.floats.length-1;i>=0;i--){ const f=G.floats[i]; f.age+=dt; if(f.age>0.95) G.floats.splice(i,1); }

  switch(G.state){
    case 'menu':
      if(Math.random()<dt*2.2){
        const s=['H','D','S','C'][Math.floor(Math.random()*4)];
        G.menuParts.push({x:Math.random()*CW,y:CH+20,vy:-(10+Math.random()*16),rot:Math.random()*6,vr:(Math.random()-.5)*1.4,suit:s,
          col:s==='H'||s==='D'?'#e2493f':'#2b3a6b',sz:7+Math.random()*8,age:0,life:9});
      }
      for(let i=G.menuParts.length-1;i>=0;i--){ const p=G.menuParts[i]; p.y+=p.vy*dt; p.rot+=p.vr*dt; if(p.y<-30) G.menuParts.splice(i,1); }
      break;
    case 'playing': updatePlaying(dt); break;
    case 'dying':
      G.deathT+=dt;
      if(G.deathT>1.5){
        if(G.lives<=0){ G.state='gameover'; G.overT=0; saveHi(); }
        else { resetPositions(); G.state='playing'; }
      }
      break;
    case 'levelup':
      G.levelT+=dt;
      if(G.levelT>1.9){ G.level++; dealBoard(); resetPositions(); G.state='playing'; }
      break;
    case 'gameover': G.overT+=dt; break;
  }
}

/* ------------------------------ drawing ------------------------------- */
function rr(c,x,y,w,h,r){
  r=Math.min(r,w/2,h/2);
  c.beginPath();
  c.moveTo(x+r,y); c.arcTo(x+w,y,x+w,y+h,r); c.arcTo(x+w,y+h,x,y+h,r);
  c.arcTo(x,y+h,x,y,r); c.arcTo(x,y,x+w,y,r); c.closePath();
}
function suitPath(c,suit,x,y,s){
  c.beginPath();
  if(suit==='D'){ c.moveTo(x,y-s); c.lineTo(x+s*.72,y); c.lineTo(x,y+s); c.lineTo(x-s*.72,y); c.closePath(); }
  else if(suit==='H'){
    c.moveTo(x,y+s*.85);
    c.bezierCurveTo(x-s*1.05,y+s*.15,x-s*.95,y-s*.75,x,y-s*.28);
    c.bezierCurveTo(x+s*.95,y-s*.75,x+s*1.05,y+s*.15,x,y+s*.85);
    c.closePath();
  } else if(suit==='S'){
    c.moveTo(x,y-s*.85);
    c.bezierCurveTo(x+s*1.05,y-s*.15,x+s*.95,y+s*.75,x,y+s*.28);
    c.bezierCurveTo(x-s*.95,y+s*.75,x-s*1.05,y-s*.15,x,y-s*.85);
    c.closePath();
    c.moveTo(x-s*.32,y+s*.95); c.lineTo(x+s*.32,y+s*.95); c.lineTo(x,y+s*.3); c.closePath();
  } else { // club
    c.arc(x,y-s*.42,s*.46,0,7);
    c.moveTo(x-s*.48+s*.46,y+s*.14); c.arc(x-s*.48,y+s*.14,s*.46,0,7);
    c.moveTo(x+s*.48+s*.46,y+s*.14); c.arc(x+s*.48,y+s*.14,s*.46,0,7);
    c.moveTo(x-s*.34,y+s*.95); c.lineTo(x+s*.34,y+s*.95); c.lineTo(x,y+s*.25); c.closePath();
  }
}
function drawSuit(c,suit,x,y,s,col){ c.fillStyle=col; suitPath(c,suit,x,y,s); c.fill('evenodd'); }

function cardBack(c,cx,cy,w,h,rot){
  c.save(); c.translate(cx,cy); c.rotate(rot||0);
  rr(c,-w/2,-h/2,w,h,3); c.fillStyle='#a32638'; c.fill();

  rr(c,-w/2+2.2,-h/2+2.2,w-4.4,h-4.4,2); c.stroke();
  drawSuit(c,'D',0,0,w*.2,'#e8c15a');
  c.restore();
}
function faceCard(c,cx,cy,w,h,suit,letter){
  c.save(); c.translate(cx,cy);
  rr(c,-w/2,-h/2,w,h,3); c.fillStyle='#f6f1e6'; c.fill();
  c.strokeStyle='#c9bfa8'; c.lineWidth=1; rr(c,-w/2+.5,-h/2+.5,w-1,h-1,3); c.stroke();

    c.fillStyle=col; c.font='700 '+(w*.42)+'px Georgia,serif';
    c.textAlign='left'; c.textBaseline='top'; c.fillText(letter,-w/2+2.5,-h/2+2);
    drawSuit(c,suit,-w/2+w*.26,-h/2+h*.3,w*.16,col);
  }
  return c;
}

/* --------------------------- dynamic drawing -------------------------- */
function easeOutBack(t){ const s=1.70158; t-=1; return 1+t*t*((s+1)*t+s); }

function drawDots(c){

    const k=(G.t-d.born)/0.3; if(k<=0) continue;
    const sc=k>=1?1:easeOutBack(Math.min(1,k));
    const bob=Math.sin(G.t*2.4+(d.x*1.7+d.y*2.3))*1.4;

    const col=d.suit==='H'||d.suit==='D'?'#d43a4a':'#26304f';
    drawSuit(c,d.suit,0,.5,5.2,col);
    drawSuit(c,d.suit,-5.5,-6.5,1.8,col);
    c.restore();
  }
}
function drawJokers(c){
  for(const j of G.jokers){
    if(!j.alive) continue;
    const x=OX+j.x*TILE+TILE/2, y=OY+j.y*TILE+TILE/2;

  }
}
function ghostPath(c,s){
  c.beginPath();
  c.moveTo(-s,-s*.28);
  c.arc(0,-s*.28,s,Math.PI,0,false);
  c.lineTo(s,s*.62);
  for(let i=0;i<3;i++){
    const x1=s-(i+0.5)*(s*2/3), x0=s-(i+1)*(s*2/3);
    c.quadraticCurveTo(x1,s*.28,x0,s*.62);
  }
  c.closePath();
}
function drawGhost(c,g){
  const x=OX+g.x*TILE, y=OY+g.y*TILE;
  const s=TILE*.43, wob=Math.sin(G.t*9+g.tx*3)*1.2;
  if(g.eye){ // just the eyes racing home
    c.save(); c.translate(x,y+wob*.5);
    for(const sx of [-1,1]){
      c.fillStyle='#fff'; c.beginPath(); c.ellipse(sx*s*.42,-s*.15,s*.36,s*.44,0,0,7); c.fill();
      c.fillStyle='#22338f'; c.beginPath(); c.arc(sx*s*.42+g.dir.x*2.5,-s*.15+g.dir.y*2.5,s*.17,0,7); c.fill();
    }
    c.restore(); return;
  }
  const blink=G.fright>0&&G.fright<1.6&&(Math.floor(G.t*8)%2===0);
  const body=g.fright?(blink?'#e2493f':'#31418f'):g.color;
  c.save(); c.translate(x,y+wob);
  c.shadowColor='rgba(0,0,0,.35)'; c.shadowBlur=4; c.shadowOffsetY=2;
  ghostPath(c,s); c.fillStyle=body; c.fill();
  c.shadowColor='transparent';
  c.strokeStyle='rgba(246,241,230,.85)'; c.lineWidth=1.6; ghostPath(c,s); c.stroke();
  if(!g.fright){ // rank mark
    c.fillStyle='rgba(255,255,255,.92)'; c.font='700 '+(s*.62)+'px Georgia,serif';
    c.textAlign='left'; c.textBaseline='top'; c.fillText(g.letter,-s*.82,-s*.62);
  } else { // zigzag mouth
    c.strokeStyle=blink?'#fff':'rgba(255,255,255,.9)'; c.lineWidth=1.6;
    c.beginPath();
    for(let i=0;i<=6;i++){ const px=-s*.55+i*(s*1.1/6); c.lineTo(px,-s*.05+(i%2?s*.14:-s*.02)); }
    c.stroke();
  }
  // eyes
  for(const sx of [-1,1]){
    const ex=sx*s*.42, ey=-s*.28;
    if(g.fright){ c.fillStyle='#fff'; c.beginPath(); c.arc(ex,ey,s*.15,0,7); c.fill(); }
    else{
      c.fillStyle='#fff'; c.beginPath(); c.ellipse(ex,ey,s*.3,s*.36,0,0,7); c.fill();
      c.fillStyle='#26304f'; c.beginPath();
      c.arc(ex+g.dir.x*s*.13, ey+g.dir.y*s*.13, s*.15, 0,7); c.fill();
    }
  }
  c.restore();
}

function draw(){
  ctx.setTransform(DPR,0,0,DPR,0,0);
  ctx.clearRect(0,0,CW,CH);
  const shx=(Math.random()-.5)*G.shake, shy=(Math.random()-.5)*G.shake;

  ctx.save(); ctx.translate(shx,shy);

  if(G.state!=='menu'&&P) drawPlayer(ctx);

  else if(G.state==='paused') drawPause(ctx);
  else if(G.state==='levelup') drawLevelUp(ctx);
  else if(G.state==='gameover') drawGameOver(ctx);
}
