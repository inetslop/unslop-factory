const LET=[['111','101','111','100','100'],['010','101','111','101','101'],['111','100','100','100','111']];

function letterAt(c,r){if(r>4)return 0;for(let i=0;i<3;i++){const b=c-LETCOLS[i];if(b>=0&&b<3)return LET[i][r][b]==='1'?1:0}return 0}

function buildLevel(n){
  grid=[];bricksLeft=0;frightMax=Math.min(9,4+n);
  const rows=Math.min(4+n,8),p=(n-1)%4;
  for(let r=0;r<rows;r++){const row=[];
    for(let c=0;c<COLS;c++){let t=0;
      if(p===0){t=1;if((r*COLS+c)%9===4)t=2}
      else if(p===1){if((r+c)%2===0)t=(c===5||c===6)?2:1;else if(r===rows-1&&c%4===2)t=2}
      else if(p===2){t=letterAt(c,r)?3:((r*COLS+c)%11===7?2:1)}
      else{const d=Math.abs(r-(rows-1)/2)+Math.abs(c-5.5);
        if(d<rows/2+.3)t=d<1.4?2:1;else if(d<rows/2+2.4&&(r+c)%2===0)t=1}
      row.push(t)}
    grid.push(row)}
  let pc=0;for(const row of grid)for(const t of row)if(t===2)pc++;
  while(pc<3){const r=(Math.random()*grid.length)|0,c=(Math.random()*COLS)|0;if(grid[r][c]===1){grid[r][c]=2;pc++}}
  for(let r=0;r<grid.length;r++)for(let c=0;c<COLS;c++){const t=grid[r][c];if(!t)continue;bricksLeft++;
    grid[r][c]={t,alive:true,x:brickX(c),y:brickY(r),w:BW,h:BH,pts:t===2?25:t===3?100:(grid.length-r)*10,color:ROWCOL[r%6]}}
}

function renorm(){const s=Math.hypot(ball.vx,ball.vy)||1;ball.vx*=ball.spd/s;ball.vy*=ball.spd/s}
function moveBall(dt){
  const dist=ball.spd*dt,steps=Math.max(1,Math.ceil(dist/7)),sx=ball.vx/ball.spd,sy=ball.vy/ball.spd;
  for(let i=0;i<steps;i++){
    ball.x+=sx*dist/steps;ball.y+=sy*dist/steps;
    if(ball.x<WALL.l+BALL_R){ball.x=WALL.l+BALL_R;ball.vx=Math.abs(ball.vx);sfx.wall();spark(WALL.l,ball.y)}
    else if(ball.x>WALL.r-BALL_R){ball.x=WALL.r-BALL_R;ball.vx=-Math.abs(ball.vx);sfx.wall();spark(WALL.r,ball.y)}
    if(ball.y<WALL.t+BALL_R){ball.y=WALL.t+BALL_R;ball.vy=Math.abs(ball.vy);sfx.wall()}
    if(ball.vy>0&&ball.y+BALL_R>=PADDLE_Y&&ball.y+BALL_R<=PADDLE_Y+PAD_H+8&&Math.abs(ball.x-paddle.x)<=paddle.w/2+BALL_R){
      const rel=clamp((ball.x-paddle.x)/(paddle.w/2),-1,1);
      ball.spd=Math.min(620,ball.spd+3);
      ball.vx=ball.spd*Math.sin(rel*1.05);ball.vy=-ball.spd*Math.cos(rel*1.05);
      ball.y=PADDLE_Y-BALL_R;streak=0;mult=1;sfx.paddle();spark(ball.x,PADDLE_Y);
    }
    hitBricks();
    if(!clearing&&ball.y-BALL_R>ROAD_Y){loseLife();return}
  }
  renorm();
}
function hitBricks(){
  for(let r=0;r<grid.length;r++)for(let c=0;c<COLS;c++){
    const b=grid[r][c];if(!b||!b.alive)continue;
    if(ball.x+BALL_R<b.x||ball.x-BALL_R>b.x+b.w||ball.y+BALL_R<b.y||ball.y-BALL_R>b.y+b.h)continue;
    const px1=ball.x+BALL_R-b.x,px2=b.x+b.w-(ball.x-BALL_R),py1=ball.y+BALL_R-b.y,py2=b.y+b.h-(ball.y-BALL_R);
    const m=Math.min(px1,px2,py1,py2);
    if(m===px1){ball.x=b.x-BALL_R;ball.vx=-Math.abs(ball.vx)}
    else if(m===px2){ball.x=b.x+b.w+BALL_R;ball.vx=Math.abs(ball.vx)}
    else if(m===py1){ball.y=b.y-BALL_R;ball.vy=-Math.abs(ball.vy)}
    else{ball.y=b.y+b.h+BALL_R;ball.vy=Math.abs(ball.vy)}
    breakBrick(b);return;
  }
}
function breakBrick(b){

  const pts=b.pts*mult;
  addScore(pts,b.x+b.w/2,b.y,'+'+pts+(mult>1?' ×'+mult:''),b.t===3?'#ff9ee0':'#ffe600');

function killGhost(i,g){
  const val=[200,400,800,1600][Math.min(3,ghostChain)];ghostChain++;
  addScore(val,g.x,g.y-12,g.name+' +'+val,'#ffe600',18);
  burst(g.x,g.y,['#3040ff','#5ef2ff','#ffffff']);
  rings.push({x:g.x,y:g.y,r:8,t:0,life:.4,col:'#5ef2ff'});
  sfx.ghostKill(ghostChain);ghosts.splice(i,1);
}
function updateGhosts(dt){
  const gs=Math.min(150,60+level*9),top=WALL.t+16,bot=ROAD_Y-46;
  for(let i=ghosts.length-1;i>=0;i--){const g=ghosts[i];
    if(g.hitFlash>0)g.hitFlash-=dt;
    if(g.spawnT>0){g.spawnT-=dt;g.x+=(g.hx-g.x)*Math.min(1,dt*4);g.y+=(g.hy-g.y)*Math.min(1,dt*4);continue}
    const fr=frightT>0&&g.fright;
    let des;
    if(fr){g.wanderT-=dt;if(g.wanderT<=0){g.wanderT=rand(.5,1.2);g.wa=Math.random()<.5?rand(-.5,.5):Math.PI-rand(-.5,.5)}des=g.wa}
    else{const tx=ball.stuck?g.hx+Math.sin(time*1.3+g.seed)*60:ball.x,ty=ball.stuck?WALL.t+90:ball.y;des=Math.atan2(ty-g.y,tx-g.x)}
    if(g.y>bot-70)des=lerpAng(des,-Math.PI/2,clamp((g.y-(bot-70))/70,0,1));
    g.a=lerpAng(g.a,des,(fr?1.4:3.2)*dt);
    const spd=gs*(fr?.52:1);
    g.x+=Math.cos(g.a)*spd*dt;g.y+=Math.sin(g.a)*spd*dt;
    if(g.x<WALL.l+GHOST_R){g.x=WALL.l+GHOST_R;g.a=Math.PI-g.a}
    if(g.x>WALL.r-GHOST_R){g.x=WALL.r-GHOST_R;g.a=Math.PI-g.a}
    if(g.y<top){g.y=top;g.a=-g.a}
    if(g.y>ROAD_Y-8){
      if(clearing){g.y=ROAD_Y-10;g.a=-Math.PI/2}else{ghosts.splice(i,1);addPopup(g.x,ROAD_Y-26,'CHOMPED!','#ff4d5e',24);loseLife()}
      continue;
    }
    if(!ball.stuck&&ball.invuln<=0){
      const dx=ball.x-g.x,dy=ball.y-g.y,d2=dx*dx+dy*dy,rr=GHOST_R+BALL_R;
      if(d2<rr*rr){const d=Math.sqrt(d2)||1,nx=dx/d,ny=dy/d;
        if(fr)killGhost(i,g);
        else{ball.x=g.x+nx*(rr+.5);ball.y=g.y+ny*(rr+.5);
          const dot=ball.vx*nx+ball.vy*ny;ball.vx-=2*dot*nx;ball.vy-=2*dot*ny;renorm();
          ball.invuln=.3;g.hitFlash=.18;sfx.boop()}}
    }
  }
}

  if(shake>0)shake=Math.max(0,shake-dt*40);

  stepFx(dt);
  if(mode!=='play'||paused)return;

  const L=keys.ArrowLeft||keys.a||keys.A,Rr=keys.ArrowRight||keys.d||keys.D;
  let moved=false;
  if(L){paddle.x-=560*dt;moved=true}

  const inst=(paddle.x-paddle.px)/Math.max(dt,1e-4);
  paddle.vx=paddle.vx*.7+inst*.3;paddle.px=paddle.x;
  paddle.x=clamp(paddle.x,WALL.l+paddle.w/2+4,WALL.r-paddle.w/2-4);

  if(ball.lockT>0)ball.lockT-=dt;
  if(ball.invuln>0)ball.invuln-=dt;
  if(ball.stuck){ball.x=paddle.x;ball.y=PADDLE_Y-BALL_R-9}
  else{moveBall(dt);if(mode==='play'&&!ball.stuck)pushTrail()}

  if(frightT>0){frightT-=dt;if(frightT<=0){frightT=0;ghostChain=0;ghosts.forEach(g=>g.fright=false)}}
  updateGhosts(dt);

  if(clearing){clearT-=dt;if(clearT<=0){
    clearing=false;level++;paddle.w=Math.max(78,PAD_W0-(level-1)*6);
    ball.stuck=true;ball.lockT=1.2;ball.spd=340;trail=[];
    buildLevel(level);spawnGhosts();

  }}
}

function drawBricks(){
  for(let r=0;r<grid.length;r++)for(let c=0;c<COLS;c++){const b=grid[r][c];if(!b||!b.alive)continue;

function drawGhost(g){
  if(g.spawnT>0&&Math.floor(time*12)%2===0)return;
  const ph=time*(frightT>0?9:7)+g.seed;
  const gBlink=frightT<1.5&&g.fright&&Math.floor(time*9)%2===0;
  let body=g.hitFlash>0?'#ffffff':(g.fright?(gBlink?'#f2f5ff':'#3040ff'):g.color);
  ctx.save();ctx.translate(g.x,g.y);
  ctx.shadowColor=body;ctx.shadowBlur=16;ctx.fillStyle=body;
  ctx.beginPath();ctx.arc(0,-3,GHOST_R,Math.PI,0,false);
  const yB=GHOST_R-7,fw=2*GHOST_R/3;let xR=GHOST_R;
  ctx.lineTo(xR,yB-2);
  for(let i=0;i<3;i++){const dy=5+Math.sin(ph+i*2.1)*2.2;ctx.quadraticCurveTo(xR-fw/2,yB-2+dy,xR-fw,yB-2);xR-=fw}
  ctx.lineTo(-GHOST_R,-3);ctx.closePath();ctx.fill();
  ctx.shadowBlur=0;
  if(g.fright){
    const fc=gBlink?'#d42a5e':'#ffffff';
    ctx.fillStyle=fc;circle(-5.5,-5,2.3);ctx.fill();circle(5.5,-5,2.3);ctx.fill();
    ctx.strokeStyle=fc;ctx.lineWidth=1.8;ctx.lineCap='round';ctx.beginPath();
    ctx.moveTo(-8,4);ctx.quadraticCurveTo(-6,7.5,-4,4);ctx.quadraticCurveTo(-2,.5,0,4);
    ctx.quadraticCurveTo(2,7.5,4,4);ctx.quadraticCurveTo(6,.5,8,4);ctx.stroke();
  }else{
    const ex=Math.cos(g.a)*2.6,ey=Math.sin(g.a)*2.2;
    ctx.fillStyle='#fff';
    ctx.beginPath();ctx.ellipse(-5.5,-5,4,4.8,0,0,TAU);ctx.fill();
    ctx.beginPath();ctx.ellipse(5.5,-5,4,4.8,0,0,TAU);ctx.fill();
    ctx.fillStyle='#1b2bd6';circle(-5.5+ex,-5+ey,2.3);ctx.fill();circle(5.5+ex,-5+ey,2.3);ctx.fill();
  }
  ctx.restore();
}

function drawBall(){
  let dir,open;
  if(ball.stuck){dir=-Math.PI/2;open=.35+.3*Math.abs(Math.sin(time*9))}
  else{dir=Math.atan2(ball.vy,ball.vx);const sp=clamp((ball.spd-340)/280,0,1);open=.28+.42*sp*Math.abs(Math.sin(time*(8+6*sp)))}
  for(const t of trail){const a=t.life/.35;ctx.fillStyle='rgba(255,230,0,'+(a*.3).toFixed(3)+')';circle(t.x,t.y,2.5*a+.6);ctx.fill()}
  ctx.save();ctx.translate(ball.x,ball.y);ctx.rotate(dir);

  ctx.beginPath();ctx.moveTo(0,0);ctx.arc(0,0,BALL_R,-half,half,true);ctx.closePath();ctx.fill();
  ctx.restore();

}
