```js
const N=1,E=2,S=4,Wb=8;
const DIRS=[{dx:0,dy:-1,b:N,o:S},{dx:1,dy:0,b:E,o:Wb},{dx:0,dy:1,b:S,o:N},{dx:-1,dy:0,b:Wb,o:E}];
function genMaze(w,h){ const cells=new Int8Array(w*h).fill(15); const vis=new Uint8Array(w*h); const st=[0]; vis[0]=1;
 while(st.length){ const cur=st[st.length-1]; const cx=cur%w, cy=(cur/w)|0; const opts=[]; for(const d of DIRS){ const nx=cx+d.dx, ny=cy+d.dy; if(nx>=0&&ny>=0&&nx<w&&ny<h&&!vis[ny*w+nx]) opts.push(d);}
  if(opts.length){ const d=opts[(Math.random()*opts.length)|0]; const ni=(cy+d.dy)*w+(cx+d.dx); cells[cur]&=~d.b; cells[ni]&=~d.o; vis[ni]=1; st.push(ni);} else st.pop(); }
```

Solution path (unique): DFS from 0 to target with parent:
```js
function solvePath(m){ const n=m.w*m.h, tgt=n-1; const par=new Int32Array(n).fill(-1); const vis=new Uint8Array(n); const st=[0]; vis[0]=1;
 while(st.length){ const cur=st[st.length-1]; if(cur===tgt) break; const cx=cur%m.w, cy=(cur/m.w)|0; for(const d of DIRS){ const nx=cx+d.dx, ny=cy+d.dy; if(nx<0||ny<0||nx>=m.w||ny>=m.h) continue; const ni=ny*m.w+nx; if(!vis[ni]){ vis[ni]=1; par[ni]=cur; st.push(ni);} } }
 const path=[]; let c=tgt; while(c!==-1){ path.push(c); c=par[c]; } return path.reverse(); }
```

bestNext(idx): BFS from idx to tgt:
```js
function bestNext(m, from){ if(from===m.w*m.h-1) return -1; const n=m.w*m.h; const par=new Int32Array(n).fill(-1); const q=[from]; let qi=0; // BFS
```

```html
genMaze=function(w,h){
  const cells=new Int8Array(w*h).fill(15), vis=new Uint8Array(w*h), st=[0]; vis[0]=1;
  const opp=b=>({1:4,2:8,4:1,8:2}[b]);
  while(st.length){
    const cur=st[st.length-1], cx=cur%w, cy=(cur/w)|0, opts=[];
    for(const d of DIRS){ const nx=cx+d.dx, ny=cy+d.dy;
      if(nx>=0&&ny>=0&&nx<w&&ny<h&&!vis[ny*w+nx]) opts.push(d); }
    if(opts.length){ const d=opts[(Math.random()*opts.length)|0];
      const ni=(cy+d.dy)*w+(cx+d.dx);
      cells[cur]&=~d.b; cells[ni]&=~opp(d.b);
      vis[ni]=1; st.push(ni);
    } else st.pop();
  }
  return {w,h,cells};
};

function solvePath(m){
  const n=m.w*m.h, tgt=n-1, par=new Int32Array(n).fill(-1), vis=new Uint8Array(n), st=[0]; vis[0]=1;
  while(st.length){
    const cur=st[st.length-1]; if(cur===tgt)break;
    const cx=cur%m.w, cy=(cur/m.w)|0;
    for(const d of DIRS){ const nx=cx+d.dx, ny=cy+d.dy;
      if(nx<0||ny<0||nx>=m.w||ny>=m.h)continue;
      const ni=ny*m.w+nx;
      if(!vis[ni]){ vis[ni]=1; par[ni]=cur; st.push(ni); } }
  }
  const p=[]; let c=tgt; while(c!==-1){p.push(c);c=par[c];} return p.reverse();
}

function bestNext(m,from){
  const n=m.w*m.h; if(from===n-1)return -1;
  const par=new Int32Array(n).fill(-1), q=[from];
  for(let head=0;head<q.length;head++){
    const cur=q[head]; if(cur===n-1)break;
    const cx=cur%m.w, cy=(cur/m.w)|0;
    for(const d of DIRS){ const nx=cx+d.dx, ny=cy+d.dy;
      if(nx<0||ny<0||nx>=m.w||ny>=m.h)continue;
      const ni=ny*m.w+nx;
      if(ni!==from&&par[ni]===-1){ par[ni]=cur; q.push(ni); } }
  }
  let c=n-1; while(par[c]!==-1&&par[c]!==from)c=par[c];
  return par[c]===from?c:-1;
}

function exitReachable(){
  const n=maze.w*maze.h, vis=new Uint8Array(n), start=player.cy*maze.w+player.cx, q=[start]; vis[start]=1;
  for(let head=0;head<q.length;head++){
    const cur=q[head]; if(cur===n-1)return true;
    const cx=cur%maze.w, cy=(cur/maze.w)|0, c=maze.cells[cur];
    for(const d of DIRS){
      const nx=cx+d.dx, ny=cy+d.dy;
      if(nx<0||ny<0||nx>=maze.w||ny>=maze.h)continue;
      if((c&d.b)&&!dugSet.has(dugKey(cx,cy,nx,ny)))continue;
      const ni=ny*maze.w+nx;
      if(!vis[ni]){ vis[ni]=1; q.push(ni); } }
  }
  return false;
}
function dugKey(x1,y1,x2,y2){
  return (x1<x2||(x1===x2&&y1<y2)) ? x1+','+y1+','+x2+','+y2 : x2+','+y2+','+x1+','+y1;
}

  for(let cy=0;cy<m.h;cy++)for(let cx=0;cx<m.w;cx++){
    const x=ox+cx*cell, y=oy+cy*cell, r1=hash(cx,cy,1), r2=hash(cx,cy,2);
    g.fillStyle=`hsl(26 ${13+r1*9}% ${7+r2*5}%)`;
    g.fillRect(x+.5,y+.5,cell-1,cell-1);
    g.fillStyle='rgba(255,220,170,.05)';
    for(let i=0;i<3;i++){ const h=hash(cx,cy,10+i);
      g.fillRect(x+2+h*(cell-6), y+2+hash(cx,cy,20+i)*(cell-6), 1.5, 1.5); }
    if(hash(cx,cy,5)<.3){ g.strokeStyle='rgba(0,0,0,.28)'; g.lineWidth=1; g.beginPath();
      let px=x+cell*.2+h1()*cell*.4, py=y+cell*.3+h2()*cell*.4; h1=()=>hash(cx,cy,30); h2=()=>hash(cx,cy,31);
      px=x+cell*(.2+hash(cx,cy,30)*.5); py=y+cell*(.25+hash(cx,cy,31)*.5);
      g.moveTo(px,py);
      g.lineTo(px+cell*.18, py+cell*.14); g.lineTo(px+cell*.3, py+cell*.1); g.stroke(); }
  }
  const line=(x1,y1,x2,y2)=>{g.beginPath();g.moveTo(x1,y1);g.lineTo(x2,y2);g.stroke();};
  g.lineCap='round';
  const seg=(x1,y1,x2,y2)=>{
    g.strokeStyle='rgba(0,0,0,.45)'; g.lineWidth=wallW+2; line(x1,y1+1.5,x2,y2+1.5);
    g.strokeStyle='#8f7352'; g.lineWidth=wallW; line(x1,y1,x2,y2);
    g.strokeStyle='rgba(48,33,19,.4)'; g.lineWidth=Math.max(1,wallW*.3); line(x1,y1,x2,y2);
  };
  for(let cy=0;cy<m.h;cy++)for(let cx=0;cx<m.w;cx++){
    const px=ox+cx*cell, py=oy+cy*cell, c=m.cells[cy*m.w+cx];
    if(c&N){ if(cy>0&&dugSet.has(dugKey(cx,cy,cx,cy-1)))continue; seg(px,py,px+cell,py); }
    if(c&Wb){ if(cx>0&&dugSet.has(dugKey(cx-1,cy,cx,cy)))continue; seg(px,py,px,py+cell); }
  }
  g.strokeStyle='#6d573c'; g.lineWidth=wallW*1.35;
  g.strokeRect(ox-wallW*.4,oy-wallW*.4,m.w*cell+wallW*.8,m.h*cell+wallW*.8);
  for(const k of dugSet){ const p=k.split(',').map(Number);
    const mx=ox+(p[0]+p[2])/2*cell, my=oy+(p[1]+p[3])/2*cell;
    g.strokeStyle='rgba(8,5,3,.8)'; g.lineWidth=wallW*.55; line(mx,my,mx,my);
    for(let i=0;i<5;i++){ const h=hash(p[0]*7+p[2],p[1]*7+p[3],40+i);
      g.fillStyle=i%2?'#5c4936':'#7b6247';
      g.beginPath(); g.arc(mx+(h-.5)*cell*.7, my+((hash(p[0]+i,p[2]-i,50+i))-.5)*cell*.7, cell*.05+h*cell*.04, 0, 7); g.fill(); }
    g.fillStyle='rgba(255,150,60,.5)';
    g.beginPath(); g.arc(mx,my,Math.max(1.2,cell*.035),0,7); g.fill();
  }
}

function newLevel(){
  const size=SIZES[Math.min(level-1,SIZES.length-1)];
  maze=genMaze(size,size); path=solvePath(maze); pathLen=path.length-1;
  par=2.5+pathLen*1.1;
  picks=Math.max(6, pathLen+Math.round(pathLen*.45));
  combo=0; cleanDigs=0; totalDigs=0; maxCombo=0; levelTime=0;
  dugSet=new Set(); noPickHint=false; actedThisLevel=false;
  player={cx:0,cy:0,ax:0,ay:0,fx0:0,fy0:0,tx:0,ty:0,moving:false,t:0,fx:-1,fy:0};
  floaters=[]; parts=[]; pendingClear=pendingOver=-1; hover=null;
  layout(); renderMazeLayer(); setBanner('LEVEL '+level); refreshHUD();
}

function startMove(nx,ny){ if(player.moving){player.ax=player.cx;player.ay=player.cy;}
  player.fx0=player.ax; player.fy0=player.ay; player.tx=nx; player.ty=ny;
  player.cx=nx; player.cy=ny; player.moving=true; player.t=0; }

function tryStep(dx,dy){
  if(state!=='playing'||pendingClear>0||pendingOver>0)return;
  const cx=player.cx, cy=player.cy, nx=cx+dx, ny=cy+dy;
  if(nx<0||ny<0||nx>=maze.w||ny>=maze.h){ bump(); return; }
  actedThisLevel=true;
  const idx=cy*maze.w+cx, nidx=ny*maze.w+nx;
  let mxp, myp;
  if((maze.cells[idx]&dirBit(dx,dy)) && !dugSet.has(dugKey(cx,cy,nx,ny))){
    if(picks<=0){ const [px2,py2]=cellPx(nx,ny); floaterAt(px2,py2,'NO PICKS!','#ff7a5c'); sfx('bump'); shake=Math.max(shake,4); return; }
    dugSet.add(dugKey(cx,cy,nx,ny)); picks--; totalDigs++; runDigs++;
    mxp=view.ox+(cx+nx)/2*view.cell; myp=view.oy+(cy+ny)/2*view.cell;
    const clean = bestNext(maze,idx)===nidx;
    if(clean){ combo=Math.min(combo+1,9); maxCombo=Math.max(maxCombo,combo); score+=25*combo;
      cleanDigs++; runClean++;
      floaterAt(mxp,myp, combo>1?'CLEAN DIG ×'+combo:'CLEAN DIG', combo>=4?'#8be08f':'#ffd479');
      sfx('dig'); sfx('clean',combo);
    } else { combo=0; floaterAt(mxp,myp,'DEAD END','#e2765a'); sfx('dig'); sfx('dead'); }
    burstDig(mxp,my); shake=Math.min(shake+3,7); renderMazeLayer();
    if(score>best){best=score;saveBest(best);}
  }
  player.fx=dx; player.fy=dy; startMove(nx,ny); sfx('move');
  if(nx===maze.w-1&&ny===maze.h-1){ pendingClear=.32; }
  else if(picks<=0){
    if(exitReachable()){ if(!noPickHint){ noPickHint=true;
      const [px2,py2]=cellPx(player.cx,player.cy); floaterAt(px2,py2-view.cell,'NO PICKS — WALK IT OUT!','#ffd479'); } }
    else pendingOver=1.05;
  }
  refreshHUD(); popCombo();
}

  const [px,py]=cellPx(player.cx+player.fx, player.cy+player.fy);
  for(let i=0;i<4;i++)parts.push({x:px,y:py,vx:rand(-25,25),vy:rand(-40,-10),g:200,life:.3,t:0,sz:2,col:'#6b5334'}); }
function hop(){ hopT=.18; sfx('move'); }

function update(dt){
  if((state==='playing')&&pendingClear<0&&pendingOver<0) levelTime+=dt;
  if(state==='playing'&&AC&&!muted){ dripT-=dt; if(dripT<=0){ sfx('drip'); dripT=rand(3,7); } }
  if(pendingClear>0){ pendingClear-=dt; if(pendingClear<=0)doClear(); }
  if(pendingOver>0){ pendingOver-=dt; if(pendingOver<=0)doGameOver(); }
  if(player.moving){ player.t+=dt/.11; const k=Math.min(1,player.t), e=1-(1-k)*(1-k);
    player.ax=lerp(player.fx0,player.tx,e); player.ay=lerp(player.fy0,player.ty,e);
    if(k>=1)player.moving=false; }
  if(hopT>0)hopT-=dt;
  shake=Math.max(0,shake-dt*24); bannerT+=dt;
  for(let i=floaters.length-1;i>=0;i--){ floaters[i].t+=dt; if(floaters[i].t>.95)floaters.splice(i,1); }
  for(let i=parts.length-1;i>=0;i--){ const p=parts[i]; p.t+=dt;
    if(p.t>=p.life){parts.splice(i,1);continue;}
    p.x+=p.vx*dt; p.y+=p.vy*dt; p.vy+=(p.g||0)*dt; }
  for(const m of motes){ m.y-=m.s*dt; m.x+=Math.sin(T*.5+m.ph)*dt*7;
    if(m.y<-6){m.y=H+6;m.x=Math.random()*W;} }
  if(state==='playing'&&maze&&!actedThisLevel&&levelTime>1.4){
    const [px,py]=cellPx(player.cx,player.cy); floaterAt(px,py-view.cell,'SWING! CLICK A WALL','#ffcf7d'); actedThisLevel=true; }
  computeHover();
  if(state==='playing'||state==='levelclear'||state==='gameover'||state==='paused')
    hudTime.textContent=fmt(levelTime);
}

function computeHover(){
  hover=null;
  if(!mouse.in||!maze||!(state==='playing'&&pendingClear<0&&pendingOver<0)){ cv.style.cursor='default'; return; }
  const cx=Math.floor((mouse.x-view.ox)/view.cell), cy=Math.floor((mouse.y-view.oy)/view.cell);
  if(cx>=0&&cy>=0&&cx<maze.w&&cy<maze.h&&(Math.abs(cx-player.cx)+Math.abs(cy-player.cy))===1){
    hover={cx,cy}; cv.style.cursor='pointer'; } else cv.style.cursor='default';
}

  ctx.drawImage(bgCv,0,0,W,H);
  ctx.save();
  if(shake>0)ctx.translate(rand(-1,1)*shake,rand(-1,1)*shake);
  if(maze){
    ctx.drawImage(mazeCv,0,0,W,H);
    drawExit();
    const [px,py]=playerPx();
    darkness(px,py);
    const flick=.9+.1*Math.sin(T*9)+.05*Math.sin(T*23+1);
    ctx.globalCompositeOperation='lighter';
    const lr=cell*2.4*flick;
    ctx.globalAlpha=.5; ctx.drawImage(sprAmber,px-lr,py-lr,lr*2,lr*2);
    ctx.globalAlpha=1; ctx.globalCompositeOperation='source-over';
    drawHover();
    drawPlayer(px,py,flick);
  } else {
    ctx.globalCompositeOperation='lighter';
    for(let i=0;i<3;i++){
      const x=W*(.3+.4*Math.sin(T*.11+i*2.1)), y=H*(.35+.3*Math.cos(T*.09+i*1.7));
      ctx.globalAlpha=.1; ctx.drawImage(i===1?sprMoss:sprAmber,x-130,y-130,260,260); }
    ctx.globalAlpha=1; ctx.globalCompositeOperation='source-over';
  }
  drawParts(); drawFloaters();
  ctx.restore();
  // motes (in front, drifting dust)
  ctx.globalCompositeOperation='lighter';
  for(const m of motes){ ctx.globalAlpha=m.a; const s=m.r*7;
    ctx.drawImage(sprMote,m.x-s/2,m.y-s/2,s,s); }
  ctx.globalAlpha=1; ctx.globalCompositeOperation='source-over';
  drawBanner();
}
function playerPx(){ return [view.ox+(player.ax+.5)*view.cell, view.oy+(player.ay+.5)*view.cell]; }

function drawExit(){
  const [ex,ey]=cellPx(maze.w-1,maze.h-1), cell=view.cell;
  const p=1+.1*Math.sin(T*3);
  ctx.globalCompositeOperation='lighter';
  ctx.drawImage(sprMoss,ex-cell*1.2*p,ey-cell*1.2*p,cell*2.4*p,cell*2.4*p);
  ctx.globalCompositeOperation='source-over';
  ctx.strokeStyle='#7ee081'; ctx.lineWidth=2;
  ctx.beginPath(); ctx.arc(ex,ey,cell*.34*p,0,7); ctx.stroke();
  ctx.fillStyle='#0c1a10'; ctx.beginPath(); ctx.arc(ex,ey,cell*.2,0,7); ctx.fill();
  for(let i=0;i<3;i++){ const a=T*1.5+i*2.09;
    ctx.fillStyle='#b6f2b8';
    ctx.beginPath(); ctx.arc(ex+Math.cos(a)*cell*.45,ey+Math.sin(a)*cell*.45,Math.max(1,cell*.03),0,7); ctx.fill(); }
}

function darkness(px,py){
  const g=dctx;
  g.globalCompositeOperation='source-over'; g.clearRect(0,0,W,H);
  g.fillStyle='rgba(7,5,3,.965)'; g.fillRect(0,0,W,H);
  g.globalCompositeOperation='destination-out';
  const flick=.9+.1*Math.sin(T*9)+.05*Math.sin(T*23+1), r=view.cell*2.7*flick;
  const gr=g.createRadialGradient(px,py,0,px,py,r);
  gr.addColorStop(0,'rgba(0,0,0,1)'); gr.addColorStop(.5,'rgba(0,0,0,.92)'); gr.addColorStop(1,'rgba(0,0,0,0)');
  g.fillStyle=gr; g.beginPath(); g.arc(px,py,r,0,7); g.fill();
  const s=view.cell*.8;
  for(const k of dugSet){ const p=k.split(',').map(Number);
    const mx=view.ox+(p[0]+p[2])/2*view.cell, my=view.oy+(p[1]+p[3])/2*view.cell;
    g.drawImage(sprDug,mx-s,my-s,s*2,s*2); }
  const d=Math.max(Math.abs(player.cx-(maze.w-1)),Math.abs(player.cy-(maze.h-1)));
  if(d<=3){ const [ex,ey]=cellPx(maze.w-1,maze.h-1), er=view.cell*(.9+(3-d)*.7);
    g.globalAlpha=.5+.28*(3-d)/3; g.drawImage(sprMoss,ex-er,ey-er,er*2,er*2); g.globalAlpha=1; }
  ctx.drawImage(darkCv,0,0,W,H);
}

function drawHover(){
  if(!hover)return;
  const cell=view.cell, x=view.ox+hover.cx*cell, y=view.oy+hover.cy*cell;
  const wall=(maze.cells[hover.cy*maze.w+hover.cx]&dirBit(hover.cx-player.cx,hover.cy-player.cy))!==0;
  ctx.strokeStyle=wall?'rgba(255,200,110,.55)':'rgba(126,224,139,.55)';
  ctx.lineWidth=2; rr(x+2,y+2,cell-4,cell-4,cell*.18); ctx.stroke();
}

function drawPlayer(px,py,flick){
  const cell=view.cell;
  const bob=Math.sin(T*3.2)*cell*.04;
  const k=player.moving?Math.sin(Math.min(1,player.t)*Math.PI):0;
  const hop=hopT>0?Math.sin((hopT/.18)*Math.PI)*.15:0;
  const by=py+bob-k*cell*.16;
  ctx.fillStyle='rgba(0,0,0,.35)';
  ctx.beginPath(); ctx.ellipse(px,py+cell*.28,cell*.26*(1-k*.2),cell*.09,0,0,7); ctx.fill();
  const lw=Math.max(1.5,cell*.045);
  // ears
  for(const s of [-1,1]){ ctx.fillStyle='#e8c795'; ctx.strokeStyle='#3a2a18'; ctx.lineWidth=lw;
    ctx.beginPath(); ctx.arc(px+s*cell*.16,by-cell*.2+player.fy*cell*.03,cell*.11,0,7); ctx.fill(); ctx.stroke(); }
  // body
  ctx.fillStyle='#f4d9a6'; ctx.strokeStyle='#3a2a18'; ctx.lineWidth=lw;
  ctx.beginPath(); ctx.ellipse(px,by,cell*.27,cell*.24*(1-k*.12-hop),0,0,7); ctx.fill(); ctx.stroke();
  // belly sheen
  ctx.fillStyle='rgba(255,255,255,.16)';
  ctx.beginPath(); ctx.ellipse(px,by+cell*.08,cell*.16,cell*.09,0,0,7); ctx.fill();
  // eyes (look toward facing)
  const eo=player.fx*cell*.07, eyo=player.fy*cell*.05;
  for(const s of [-1,1]){ ctx.fillStyle='#26190e';
    ctx.beginPath(); ctx.arc(px+eo+s*cell*.08,by-cell*.03+eyo,cell*.045,0,7); ctx.fill();
    ctx.fillStyle='#fff';
    ctx.beginPath(); ctx.arc(px+eo+s*cell*.08-1,by-cell*.045+eyo,cell*.015,0,7); ctx.fill(); }
  // blush
  ctx.fillStyle='rgba(255,140,80,.3)';
  for(const s of [-1,1]){ ctx.beginPath(); ctx.ellipse(px+s*cell*.15,by+cell*.05,cell*.035,cell*.022,0,0,7); ctx.fill(); }
  // torch flame
  const fx=px+player.fx*cell*.1, fy=by-cell*.36;
  const fl=Math.sin(T*17)*.5+Math.sin(T*29+2)*.5, fr=cell*(.15+.035*fl);
  ctx.globalCompositeOperation='lighter';
  ctx.globalAlpha=.55; ctx.drawImage(sprAmber,fx-cell*.8,fy-cell*.8,cell*1.6,cell*1.6);
  ctx.globalAlpha=1; ctx.globalCompositeOperation='source-over';
  ctx.fillStyle='#ff9a3c'; ctx.beginPath(); ctx.arc(fx,fy,fr,0,7); ctx.fill();
  ctx.fillStyle='#ffc45e'; ctx.beginPath(); ctx.arc(fx,fy-fr*.25,fr*.66,0,7); ctx.fill();
  ctx.fillStyle='#fff3c4'; ctx.beginPath(); ctx.arc(fx,fy-fr*.4,fr*.32,0,7); ctx.fill();
}

function drawParts(){
  for(const p of parts){ const a=1-p.t/p.life;
    if(p.ring){ ctx.strokeStyle=p.col; ctx.globalAlpha=a; ctx.lineWidth=2;
      ctx.beginPath(); ctx.arc(p.x,p.y,p.sz+(p.t/p.life)*view.cell*.5,0,7); ctx.stroke(); }
    else{ ctx.fillStyle=p.col; ctx.globalAlpha=a;
      ctx.fillRect(p.x-p.sz/2,p.y-p.sz/2,p.sz,p.sz); } }
  ctx.globalAlpha=1;
}
function drawFloaters(){
  if(!floaters.length)return;
  const fs=Math.max(13,view.cell*.42);
  ctx.font='700 '+Math.round(fs)+'px Impact, "Arial Black", sans-serif';
  ctx.textAlign='center'; ctx.textBaseline='middle';
  for(const f of floaters){ const a=1-f.t/.95, y=f.y-f.t*42;
    ctx.globalAlpha=a; ctx.lineWidth=3.5; ctx.strokeStyle='rgba(0,0,0,.6)';
    ctx.strokeText(f.text,f.x,y); ctx.fillStyle=f.col; ctx.fillText(f.text,f.x,y); }
  ctx.globalAlpha=1;
}
function drawBanner(){
  if(bannerT>1.8||!bannerText)return;
  const p=bannerT/1.8, a=p<.15?p/.15:(p>.75?(1-p)/.25:1), sc=.9+.1*Math.min(1,p*6);
  ctx.save(); ctx.translate(W/2,H*.3); ctx.scale(sc,sc);
  ctx.font='700 '+Math.round(clamp(view.cell*.85,22,44))+'px Impact, "Arial Black", sans-serif';
  ctx.textAlign='center'; ctx.textBaseline='middle'; ctx.globalAlpha=a;
  ctx.lineWidth=6; ctx.strokeStyle='rgba(0,0,0,.65)'; ctx.strokeText(bannerText,0,0);
  ctx.fillStyle='#ffd479'; ctx.fillText(bannerText,0,0);
  ctx.restore(); ctx.globalAlpha=1;
}
```
