let AC=null,master=null,muted=false;
function audioInit(){
  if(!AC){try{AC=new (window.AudioContext||window.webkitAudioContext)();master=AC.createGain();master.gain.value=.5;master.connect(AC.destination);}catch(e){}}
  if(AC&&AC.state==='suspended')AC.resume().catch(()=>{});
}
function tone(f0,f1,dur,type,vol,delay){
  if(!AC||muted)return;delay=delay||0;
  const t0=AC.currentTime+delay,o=AC.createOscillator(),g=AC.createGain();
  o.type=type;o.frequency.setValueAtTime(Math.max(1,f0),t0);
  o.frequency.exponentialRampToValueAtTime(Math.max(1,f1),t0+dur);
  g.gain.setValueAtTime(vol,t0);g.gain.exponentialRampToValueAtTime(.0001,t0+dur);
  o.connect(g);g.connect(master);o.start(t0);o.stop(t0+dur+.03);
}
function noiseBurst(dur,vol,freq,delay){
  if(!AC||muted)return;delay=delay||0;
  const t0=AC.currentTime+delay,len=Math.max(1,(AC.sampleRate*dur)|0);
  const buf=AC.createBuffer(1,len,AC.sampleRate),d=buf.getChannelData(0);
  for(let i=0;i<len;i++)d[i]=(Math.random()*2-1)*(1-i/len);
  const n=AC.createBufferSource();n.buffer=buf;
  const f=AC.createBiquadFilter();f.type='lowpass';f.frequency.value=freq;
  const g=AC.createGain();g.gain.value=vol;
  n.connect(f);f.connect(g);g.connect(master);n.start(t0);
}

/* ================= PIECES / BOARD LOGIC ================= */
function computeBox(p){
  p.maxC=Math.max.apply(null,p.cells.map(c=>c[0]));
  p.maxR=Math.max.apply(null,p.cells.map(c=>c[1]));
  p.w=p.maxC+1;p.hgt=p.maxR+1;
}
function rotateCells(cells){
  const h=1+Math.max.apply(null,cells.map(c=>c[1]));
  return cells.map(c=>[h-1-c[1],c[0]]);
}
function spawnPiece(decor){
  const s=SHAPES[(Math.random()*SHAPES.length)|0];
  let cells=s.cells.map(c=>c.slice());
  for(let i=0;i<4;i++)if(Math.random()<.5)cells=rotateCells(cells);
  const mc=Math.min.apply(null,cells.map(c=>c[0])),mr=Math.min.apply(null,cells.map(c=>c[1]));
  cells=cells.map(c=>[c[0]-mc,c[1]-mr]);
  const p={x:(Math.random()*(COLS-2))|0,y:-4,cells,idx:SHAPES.indexOf(s),decor:!!decor,caughtChecked:false,canCatch:false};
  computeBox(p);p.prevY=p.y;
  pieces.push(p);
}
function hitsBoard(x,y,cells){
  for(let i=0;i<cells.length;i++){
    const col=x+cells[i][0];
    if(col<0||col>=COLS)return true;
    const a=Math.floor(y+cells[i][1]),b=Math.floor(y+cells[i][1]+1-1e-6);
    for(let rr=a;rr<=b;rr++){
      if(rr>=ROWS)return true;
      if(rr>=0&&board[rr][col]>=0)return true;
    }
  }
  return false;
}
function colEmpty(col){for(let r=0;r<ROWS;r++)if(board[r][col]>=0)return false;return true;}
function catchX(p){return clamp(Math.round(ship.x-p.w/2),0,COLS-p.w);}
function components(cells){
  const seen=new Set(),comps=[];
  for(let i=0;i<cells.length;i++){
    if(seen.has(i))continue;
    const comp=[i];seen.add(i);let q=0;
    while(q<comp.length){
      const cc=cells[comp[q]][0],cr=cells[comp[q]][1];
      for(let j=0;j<cells.length;j++){
        if(seen.has(j))continue;
        if(Math.abs(cells[j][0]-cc)+Math.abs(cells[j][1]-cr)===1){seen.add(j);comp.push(j);}
      }q++;
    }
    comps.push(comp.map(j=>cells[j]));
  }
  return comps;
}
function lockPiece(p,caught){
  let topOut=false;
  for(let i=0;i<p.cells.length;i++){
    const col=p.x+p.cells[i][0],row=Math.round(p.y+p.cells[i][1]);
    if(row<0||row>=ROWS||col<0||col>=COLS){topOut=true;continue;}
    board[row][col]=p.idx;
  }
  if(topOut){gameOver();return false;}
  for(let i=0;i<p.cells.length;i++){
    const x=fx+(p.x+p.cells[i][0]+.5)*cell,y=fy+(Math.round(p.y+p.cells[i][1])+1)*cell;
    burst(x,y,caught?'#7df3ff':'rgba(160,190,255,.8)',caught?4:2,1.6);
  }
  if(caught){sfx.catch();ship.flash=1;score+=5;if(score>hi)hi=score;popAt(x=fy+fy,0,'', '',0);} // placeholder replaced below
  else sfx.land();
  checkLines();
  return true;
}
function tryCatch(p){
  const cx=catchX(p);
  let ok=Math.abs(ship.x-(cx+p.w/2))<=p.w*.55+.35;
  if(ok)for(let i=0;i<p.cells.length;i++)if(!colEmpty(cx+p.cells[i][0])){ok=false;break;}
  if(!ok)return false;
  p.x=cx;p.y=ROWS-p.hgt;p.prevY=p.y;
  const px=fx+(cx+p.w/2)*cell,py=fy+(ROWS-1)*cell;
  burst(px,py,'#7df3ff',16,3);
  popAt(px,py-8,'+5 SET','#7df3ff',13);
  if(lockPiece(p,true))return true;
  return true; // consumed (game over handled)
}
function applyHit(p,tgt){
  let j=-1;
  for(let i=0;i<p.cells.length;i++)if(p.cells[i][0]===tgt[0]&&p.cells[i][1]===tgt[1]){j=i;break;}
  if(j<0)return;
  p.cells.splice(j,1);
  score+=10;if(score>hi)hi=score;
  const col=SHAPES[p.idx].col;
  const bx=fx+(p.x+tgt[0]+.5)*cell,by=fy+(p.y+tgt[1]+.5)*cell;
  burst(bx,col?col:col,0,0); // noop guard
  burst(bx,by,col,12,3.4);
  sfx.hit();shake=Math.max(shake,2.5);
  if(p.cells.length===0){
    const pi=pieces.indexOf(p);if(pi>=0)pieces.splice(pi,1);
    score+=50;if(score>hi)hi=score;
    popAt(bx,by,'+50 SHATTER','#ffffff',15);
    burst(bx,by,col,26,5);sfx.shatter();shake=Math.max(shake,5);
    return;
  }
  const mc=Math.min.apply(null,p.cells.map(c=>c[0])),mr=Math.min.apply(null,p.cells.map(c=>c[1]));
  p.x+=mc;p.y+=mr;p.prevY=p.y;
  p.cells=p.cells.map(c=>[c[0]-mc,c[1]-mr]);
  computeBox(p);
  const comps=components(p.cells);
  if(comps.length>1){
    const pi=pieces.indexOf(p);pieces.splice(pi,1);
    for(let k=0;k<comps.length;k++){
      const comp=comps[k],c2=Math.min.apply(null,comp.map(c=>c[0])),r2=Math.min.apply(null,comp.map(c=>c[1]));
      const np={x:p.x+c2,y:p.y+r2,prevY:p.y+r2,cells:comp.map(c=>[c[0]-c2,c[1]-r2]),idx:p.idx,decor:false,caughtChecked:false,canCatch:false};
      computeBox(np);pieces.push(np);
    }
  }
}

  const full=[];
  for(let r=0;r<ROWS;r++)if(board[r].every(v=>v>=0))full.push(r);
  if(!full.length)return;
  const pts=[0,100,300,500,800][full.length]*wave;
  score+=pts;lines+=full.length;if(score>hi)hi=score;
  const names=['','SINGLE LINE','DOUBLE LINE','TRIPLE LINE','TETRA!'];

  sfx.clear(full.length);shake=Math.max(shake,4);
}

function updFX(dt){
  for(let i=parts.length-1;i>=0;i--){
    const p=parts[i];p.t+=dt;
    if(p.t>=p.dur){parts.splice(i,1);continue;}
    const f=Math.max(0,1-2.4*dt);p.vx*=f;p.vy*=f;
    p.x+=p.vx*dt;p.y+=p.vy*dt;
  }
  for(let i=pops.length-1;i>=0;i--){const p=pops[i];p.t+=dt;if(p.t>=p.dur)pops.splice(i,1);}
  shake=Math.max(0,shake-dt*26);
  flashT=Math.max(0,flashT-dt*1.4);
  ship.cd-=dt;ship.muzzle-=dt;ship.flash=Math.max(0,ship.flash-dt*4);
}
function updateAttract(dt){
  decorT-=dt;
  if(decorT<=0){spawnPiece(true);decorT=1+Math.random()*.9;}
  for(let i=pieces.length-1;i>=0;i--){pieces[i].y+=1.5*dt;if(pieces[i].y>ROWS+4)pieces.splice(i,1);}
  ship.x=COLS/2+Math.sin(at*.4)*1.6;ship.tilt=Math.sin(at*.8)*.09;
}
function updatePlay(dt){
  // spawn
  if(toSpawn>0){spawnT-=dt;if(spawnT<=0){spawnPiece(false);toSpawn--;spawnT=spawnInt*(.7+Math.random()*.6);}}
  // pieces
  for(let i=pieces.length-1;i>=0;i--){
    const p=pieces[i];p.prevY=p.y;p.y+=speed*dt;
    if(!p.caughtChecked&&p.y+p.maxR>=ROWS-1){
      p.caughtChecked=true;
      if(tryCatch(p)){pieces.splice(i,1);if(state!=='play')return;}
      continue;
    }
    if(hitsBoard(p.x,p.y,p.cells)){
      let lo=p.prevY,hi2=p.y;
      for(let k=0;k<10;k++){const mid=(lo+hi2)/2;if(hitsBoard(p.x,mid,p.cells))hi2=mid;else lo=mid;}
      p.y=lo;
      if(!lockPiece(p,false)){pieces.splice(i,1);return;}
      pieces.splice(i,1);
    }
  }
  // ship
  let dir=0;
  if(keys.left&&!keys.right)dir=-1;else if(keys.right&&!keys.left)dir=1;
  ship.x+=dir*SHIP_SPEED*dt;
  if(pointer.active)ship.x+=(pointer.tx-ship.x)*Math.min(1,dt*14);
  ship.x=clamp(ship.x,.7,COLS-.7);
  ship.tilt+=(dir-ship.tilt)*Math.min(1,dt*9);
  if((keys.space||pointer.active)&&ship.cd<=0){
    bullets.push({c:ship.x,r:ROWS-.5,dead:false});
    ship.cd=FIRE_CD;ship.muzzle=.07;sfx.laser();
  }
  // bullets
  const hits=[];
  for(let i=bullets.length-1;i>=0;i--){
    const b=bullets[i];b.r-=BULLET_SPEED*dt;
    if(b.r<-.6){bullets.splice(i,1);continue;}
    let gone=false;
    // vs pieces (above any blocks) - rising bullet order: check piece then board
    outer:
    for(let pi=0;pi<pieces.length;pi++){
      const p=pieces[pi];
      for(let cj=0;cj<p.cells.length;cj++){
        const cx2=p.x+p.cells[cj][0],cy2=p.y+p.cells[cj][1];
        if(b.c-.1<cx2+1&&b.c+.1>cx2&&b.r-.2<cy2+1&&b.r+.2>cy2){
          hits.push({p:p,cell:p.cells[cj].slice()});b.dead=true;gone=true;break outer;
        }
      }
    }
    if(gone)continue;
    // vs board
    const c0=Math.floor(b.c-.1),c1=Math.ceil(b.c+.1);
    const r0=Math.floor(b.r-.2),r1=Math.ceil(b.r+.2);
    for(let cc=c0;cc<=c1&&!gone;cc++)for(let rr=r0;rr<=r1;rr++){
      if(cc>=0&&cc<COLS&&rr>=0&&rr<ROWS&&board[rr][cc]>=0){
        burst(fx+(cc+.5)*cell,fy+rr*cell,'#9fb4e8',5,2);sfx.block();b.dead=true;gone=true;break;
      }
    }
  }
  for(let k=0;k<hits.length;k++)if(pieces.indexOf(hits[k].p)>=0)applyHit(hits[k].p,hits[k].cell);
  bullets=bullets.filter(b=>!b.dead);
  if(state!=='play')return;
  // clear anim
  if(clearAnim){
    clearAnim.t+=dt;
    if(clearAnim.t>=.32){
      const rows=clearAnim.rows.slice().sort((a,b)=>b-a);
      for(let k=0;k<rows.length;k++){board.splice(rows[k],1);board.unshift(new Array(COLS).fill(-1));}
      for(let k=0;k<clearAnim.rows.length;k++){
        const y=fy+clearAnim.rows[k]*cell+cell/2;
        for(let c=0;c<COLS;c+=2)burst(fx+c*cell+cell/2,y,'#ffffff',3,4);
      }
      for(let k=0;k<clearAnim.texts.length;k++){const t=clearAnim.texts[k];pops.push({x:t.x,y:t.y,txt:t.txt,col:t.col,t:0,dur:1.25,size:t.size});}
      clearAnim=null;
    }
  }
  // wave progression
  if(toSpawn===0&&pieces.length===0&&!clearAnim){
    if(!waveDone){
      waveDone=true;betweenT=1.4;
      const b=25*wave;score+=b;if(score>hi)hi=score;
      popAt(fx+fw/2,fy+fh*.4,'WAVE CLEAR +'+b,'#ffd93b',17);sfx.wave();
    }else{betweenT-=dt;if(betweenT<=0)startWave(wave+1);}
  }
}

  ctx.strokeStyle='rgba(110,140,255,.06)';ctx.lineWidth=1;ctx.beginPath();
  for(let c=1;c<COLS;c++){ctx.moveTo(fx+c*cell,fy);ctx.lineTo(fx+c*cell,deckTop);}
  for(let r=1;r<ROWS;r++){ctx.moveTo(fx,fy+r*cell);ctx.lineTo(fx+fw,fy+r*cell);}
  ctx.stroke();
  // deck
  ctx.fillStyle='rgba(13,20,52,.9)';ctx.fillRect(fx,deckTop,fw,cell*DECK);
  ctx.strokeStyle='rgba(62,230,255,'+(anyGhost()?.8:.4)+')';ctx.lineWidth=2;
  ctx.beginPath();ctx.moveTo(fx,deckTop+1);ctx.lineTo(fx+fw,deckTop+1);ctx.stroke();
  ctx.strokeStyle='rgba(62,230,255,.3)';ctx.lineWidth=1;ctx.beginPath();
  for(let c=0;c<=COLS;c++){ctx.moveTo(fx+c*cell,deckTop+4);ctx.lineTo(fx+c*cell,deckTop+8);}
  ctx.stroke();
  // aim guide
  if(state==='play'){
    ctx.setLineDash([2,7]);ctx.strokeStyle='rgba(120,160,255,.13)';ctx.lineWidth=1;
    ctx.beginPath();ctx.moveTo(fx+ship.x*cell,deckTop);ctx.lineTo(fx+ship.x*cell,fy);ctx.stroke();ctx.setLineDash([]);
  }
  // danger band
  if(state==='play'){
    let top=ROWS;for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++)if(board[r][c]>=0){top=r;break;}
    if(top<=4){
      const a=.25+.4*(.5+.5*Math.sin(at*6));
      ctx.fillStyle='rgba(255,77,94,'+a+')';ctx.fillRect(fx,fy,fw,3);
      ctx.font='bold 9px monospace';ctx.textAlign='left';ctx.fillStyle='rgba(255,120,130,'+(a+.2)+')';
      ctx.fillText('TOP-OUT RISK',fx+6,fy+14);
    }
  }
}
function anyGhost(){
  for(let i=0;i<pieces.length;i++){const p=pieces[i];if(!p.decor&&p.canCatch&&p.y+p.maxR>ROWS-7)return true;}
  return false;
}
function drawBoard(){
  const ca=clearAnim;
  for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++){
    const idx=board[r][c];if(idx<0)continue;
    const x=fx+c*cell,y=fy+r*cell;
    if(ca&&ca.rows.indexOf(r)>=0){
      const t=Math.min(1,ca.t/.32),inset=cell*t*.5;
      ctx.fillStyle='rgba(255,255,255,'+((Math.sin(t*40)>0?.9:.55)*(1-t*.3))+')';
      ctx.fillRect(x+inset,y+inset,Math.max(1,cell-inset*2),Math.max(1,cell-inset*2));
    }else ctx.drawImage(sprites[idx],x+1,y+1,cell-2,cell-2);
  }
}
function drawGhosts(){
  if(state!=='play')return;
  ctx.setLineDash([5,4]);ctx.lineWidth=1.5;
  for(let i=0;i<pieces.length;i++){
    const p=pieces[i];
    if(p.decor||!p.canCatch||p.y+p.maxR<=ROWS-7)continue;
    const cx=catchX(p),col=SHAPES[p.idx].col;
    ctx.fillStyle='rgba(255,255,255,.05)';ctx.strokeStyle=col;ctx.globalAlpha=.55;
    for(let j=0;j<p.cells.length;j++){
      const x=fx+(cx+p.cells[j][0])*cell,y=fy+(ROWS-p.hgt+p.cells[j][1])*cell;
      ctx.fillRect(x+2,y+2,cell-4,cell-4);ctx.strokeRect(x+2,y+2,cell-4,cell-4);
    }
    ctx.globalAlpha=1;
  }
  ctx.setLineDash([]);
}
function drawPieces(){
  if(!pieces.length)return;
  ctx.save();ctx.beginPath();ctx.rect(fx,fy,fw,cell*ROWS);ctx.clip();
  for(let i=0;i<pieces.length;i++){
    const p=pieces[i],col=SHAPES[p.idx].col;
    ctx.globalAlpha=p.decor?.5:1;ctx.shadowColor=col;ctx.shadowBlur=13;
    for(let j=0;j<p.cells.length;j++)
      ctx.drawImage(sprites[p.idx],fx+(p.x+p.cells[j][0])*cell+1,fy+(p.y+p.cells[j][1])*cell+1,cell-2,cell-2);
  }
  ctx.restore();ctx.globalAlpha=1;ctx.shadowBlur=0;
}
function drawBullets(){
  if(!bullets.length)return;
  ctx.save();ctx.beginPath();ctx.rect(fx,fy,fw,cell*ROWS);ctx.clip();
  for(let i=0;i<bullets.length;i++){
    const b=bullets[i],x=fx+b.c*cell,y=fy+b.r*cell;
    ctx.strokeStyle='rgba(120,240,255,.5)';ctx.lineWidth=3;
    ctx.beginPath();ctx.moveTo(x,y+6);ctx.lineTo(x,y+17);ctx.stroke();
    ctx.shadowColor='#7df3ff';ctx.shadowBlur=8;ctx.fillStyle='#eaffff';
    ctx.fillRect(x-2,y-7,4,12);ctx.shadowBlur=0;
  }
  ctx.restore();
}
function drawShip(){
  const x=fx+ship.x*cell,y=deckTop+cell*DECK*.52;
  ctx.save();ctx.translate(x,y);ctx.rotate(ship.tilt||0);
  const w=cell*.46,h=cell*.5;
  const fl=h*(.5+.3*Math.sin(at*41)+Math.random()*.25);
  for(let k=-1;k<=1;k+=2){
    const ex=k*w*.42;
    ctx.beginPath();ctx.moveTo(ex-w*.16,h*.42);ctx.lineTo(ex+w*.16,h*.42);ctx.lineTo(ex,h*.42+fl);ctx.closePath();
    ctx.fillStyle='rgba(255,176,64,.85)';ctx.fill();
    ctx.beginPath();ctx.moveTo(ex-w*.08,h*.42);ctx.lineTo(ex+w*.08,h*.42);ctx.lineTo(ex,h*.42+fl*.55);ctx.closePath();
    ctx.fillStyle='rgba(255,240,170,.9)';ctx.fill();
  }
  ctx.fillStyle='#8fa3d8';ctx.fillRect(-w*.09,-h*1.02,w*.18,h*.5);
  ctx.beginPath();
  ctx.moveTo(0,-h);ctx.lineTo(w*.42,-h*.18);ctx.lineTo(w,h*.5);ctx.lineTo(w*.34,h*.28);
  ctx.lineTo(0,h*.52);ctx.lineTo(-w*.34,h*.28);ctx.lineTo(-w,h*.5);ctx.lineTo(-w*.42,-h*.18);ctx.closePath();
  const hg=ctx.createLinearGradient(0,-h,0,h);hg.addColorStop(0,'#f2f6ff');hg.addColorStop(1,'#9db1e8');
  ctx.fillStyle=hg;ctx.fill();ctx.strokeStyle='#5a70b8';ctx.lineWidth=1.5;ctx.stroke();
  ctx.shadowColor='#3ee6ff';ctx.shadowBlur=7;
  ctx.beginPath();ctx.ellipse(0,-h*.3,w*.15,h*.22,0,0,7);ctx.fillStyle='#3ee6ff';ctx.fill();ctx.shadowBlur=0;
  if(ship.muzzle>0){
    ctx.beginPath();ctx.arc(0,-h*1.1,w*(.3+Math.random()*.2),0,7);
    ctx.fillStyle='rgba(255,240,180,.9)';ctx.fill();
  }
  if(ship.flash>0){ctx.globalAlpha=ship.flash*.7;ctx.fillStyle='#fff';ctx.fill();ctx.globalAlpha=1;}
  ctx.restore();
}
function drawParts(){

    const p=parts[i],a=1-p.t/p.dur;

  }
  ctx.globalAlpha=1;
}

  for(let i=0;i<pops.length;i++){

  ctx.globalAlpha=1;
}
function draw(){
  ctx.setTransform(DPR,0,0,DPR,0,0);
  ctx.fillStyle='#04061a';ctx.fillRect(0,0,W,H);

  ctx.restore();

}
