function update(dt){
  if(phase==='build'){buildT-=dt;if(buildT<=0)startWave();}
  if(phase==='wave'&&queue.length){spawnT-=dt;
    while(spawnT<=0&&queue.length){const it=queue.shift();spawn(it.kind,it.side);spawnT+=(it.kind==='boss'?1.5:spawnGap);}}

  // enemies
  for(const e of enemies){
    if(e.dead)continue;
    e.slowT=Math.max(0,e.slowT-dt);
    let rem=e.sp*(e.slowT>0?1-e.slowAmt:1)*dt;
    const wps=PATHS[e.side];
    while(rem>0&&e.wp<wps.length){
      const t=wps[e.wp],dx=t.x-e.x,dy=t.y-e.y,d=Math.hypot(dx,dy);
      if(d<=rem){e.x=t.x;e.y=t.y;e.prog+=d;e.wp++;rem-=d;}
      else{e.x+=dx/d*rem;e.y+=dy/d*rem;e.prog+=rem;e.ang=Math.atan2(dy,dx);rem=0;}
    }
    if(e.wp>=wps.length){e.dead=true;leaked++;coreHit(e.dmg);}
  }
  enemies=enemies.filter(e=>!e.dead);

  // towers
  for(const t of towers){
    const d=TOWERS[t.type],lvl=t.lvl,rng=d.range*(1+.12*(lvl-1));
    t.cool-=dt;t.recoil=Math.max(0,t.recoil-dt*5);
    if(d.id==='frost'){
      if(t.cool<=0){
        let hits=0;const amt=Math.min(.75,d.slow+.08*(lvl-1)),pd=d.pdmg*Math.pow(1.6,lvl-1);
        for(const e of enemies){if(e.dead)continue;const dx=e.x-t.x,dy=e.y-t.y;
          if(dx*dx+dy*dy<=rng*rng){hitEnemy(e,pd,t.owner);e.slowT=Math.max(e.slowT,d.slowDur);e.slowAmt=Math.max(e.slowAmt,amt);hits++;}}
        if(hits>0){ring(t.x,t.y,rng,'rgba(160,230,255,.8)',2.5);SFX.frost();t.recoil=1;}
        t.cool=d.cd*Math.pow(.9,lvl-1);
      }
    }else{
      let tgt=null,sc=-1;
      for(const e of enemies){if(e.dead)continue;const dx=e.x-t.x,dy=e.y-t.y;
        if(dx*dx+dy*dy>rng*rng)continue;
        const s=d.id==='rail'?e.hp:e.prog;if(s>sc){sc=s;tgt=e;}}
      if(tgt)t.ang=Math.atan2(tgt.y-t.y,tgt.x-t.x);
      if(t.cool<=0&&tgt){
        t.recoil=1;const dm=d.dmg*Math.pow(1.5,lvl-1);
        if(d.id==='cannon'){
          projs.push({x:t.x+Math.cos(t.ang)*16,y:t.y+Math.sin(t.ang)*16,px:0,py:0,tgt,lx:tgt.x,ly:tgt.y,sp:560,dmg:dm,owner:t.owner,col:PCOL[t.owner],life:1.7});
          SFX.shoot();
        }else{
          hitEnemy(tgt,dm,t.owner);
          beams.push({x1:t.x+Math.cos(t.ang)*14,y1:t.y+Math.sin(t.ang)*14,x2:tgt.x,y2:tgt.y,t:0,ttl:.15,col:PCOL[t.owner]});
          burst(tgt.x,tgt.y,PCOL[t.owner],6,160);SFX.rail();
        }
        t.cool=d.cd*Math.pow(.9,lvl-1);
      }
    }
  }

  // projectiles
  for(const p of projs){
    p.life-=dt;p.px=p.x;p.py=p.y;
    if(p.tgt&&!p.tgt.dead){p.lx=p.tgt.x;p.ly=p.tgt.y;}
    const dx=p.lx-p.x,dy=p.ly-p.y,d=Math.hypot(dx,dy),step=p.sp*dt;
    if(d<=step+(p.tgt&&!p.tgt.dead?p.tgt.r:2)){
      if(p.tgt&&!p.tgt.dead)hitEnemy(p.tgt,p.dmg,p.owner);
      burst(p.lx,p.ly,p.col,4,120);p.life=0;
    }else{p.x+=dx/d*step;p.y+=dy/d*step;}
  }
  projs=projs.filter(p=>p.life>0);

  // fx
  for(const b of beams)b.t+=dt;beams=beams.filter(b=>b.t<b.ttl);
  for(const r of rings)r.t+=dt;rings=rings.filter(r=>r.t<r.ttl);
  for(const p of parts){p.t+=dt;p.x+=p.vx*dt;p.y+=p.vy*dt;p.vx*=.92;p.vy*=.92;}
  parts=parts.filter(p=>p.t<p.ttl);
  for(const f of floats)f.t+=dt;floats=floats.filter(f=>f.t<f.ttl);
  if(banner){banner.t-=dt;if(banner.t<=0)banner=null;}
  shake=Math.max(0,shake-dt*2.4);coreFlash=Math.max(0,coreFlash-dt*3);

  // wave cleared?
  if(phase==='wave'&&queue.length===0&&enemies.length===0){
    wavesHeld=waveNum;const bonus=40+waveNum*6;gold[0]+=bonus;gold[1]+=bonus;SFX.clear();
    float(320,200,'+'+bonus,PCOL[0],18);float(960,520,'+'+bonus,PCOL[1],18);
    if(waveNum>=15&&!endless){winGame();return;}
    startBuild(12);banner={t:2,txt:'WAVE CLEARED',sub:'BONUS ◈'+bonus+' TO EACH COMMAND'};
  }
}

/* ================= DRAW HELPERS ================= */
function poly(x,y,r,n,rot){ctx.beginPath();for(let i=0;i<n;i++){const a=rot+i/n*Math.PI*2;i?ctx.lineTo(x+Math.cos(a)*r,y+Math.sin(a)*r):ctx.moveTo(x+Math.cos(a)*r,y+Math.sin(a)*r);}ctx.closePath();}
function glow(x,y,r,col,a){const g=ctx.createRadialGradient(x,y,0,x,y,r);g.addColorStop(0,col.replace(')',',1)').replace('rgb','rgba')||col);return g;}
function softGlow(x,y,r,cr,cg,cb,a){const g=ctx.createRadialGradient(x,y,0,x,y,r);g.addColorStop(0,'rgba('+cr+','+cg+','+cb+','+a+')');g.addColorStop(1,'rgba('+cr+','+cg+','+cb+',0)');ctx.fillStyle=g;ctx.beginPath();ctx.arc(x,y,r,0,7);ctx.fill();}

function drawTowerBody(t,col,ghost){
  const lvl=t.lvl||1;
  ctx.save();ctx.translate(t.x,t.y);if(ghost)ctx.globalAlpha=.55;
  // base plate
  poly(0,0,19,8,-Math.PI/8);ctx.fillStyle='#0e1830';ctx.fill();ctx.strokeStyle=col;ctx.lineWidth=2;ctx.stroke();
  const d=TOWERS[t.type];
  if(d.id==='cannon'){
    ctx.rotate(t.ang||-Math.PI/2);const rec=(t.recoil||0)*4;
    ctx.fillStyle='#182742';ctx.fillRect(-rec-2,-3.5,18,7);
    ctx.strokeStyle=col;ctx.lineWidth=1.5;ctx.strokeRect(-rec-2,-3.5,18,7);
    ctx.beginPath();ctx.arc(-rec*.4,0,9,0,7);ctx.fillStyle=col;ctx.fill();
    ctx.beginPath();ctx.arc(-rec*.4,0,4,0,7);ctx.fillStyle='#eafcff';ctx.fill();
    if((t.recoil||0)>.55){ctx.beginPath();ctx.arc(16-rec,0,5,0,7);ctx.fillStyle='rgba(255,255,255,.85)';ctx.fill();}
  }else if(d.id==='frost'){
    const pl=1+(t.recoil||0)*.15;
    poly(0,0,12*pl,6,tNow*.9);ctx.fillStyle='rgba(150,225,255,.28)';ctx.fill();ctx.strokeStyle='#bfeaff';ctx.lineWidth=2;ctx.stroke();
    ctx.save();ctx.rotate(-tNow*1.4);ctx.strokeStyle='#eafcff';ctx.lineWidth=2;
    for(let i=0;i<3;i++){const a=i*Math.PI/3;ctx.beginPath();ctx.moveTo(Math.cos(a)*9,Math.sin(a)*9);ctx.lineTo(-Math.cos(a)*9,-Math.sin(a)*9);ctx.stroke();}
    ctx.restore();
  }else{
    ctx.rotate(t.ang||0);
    ctx.beginPath();ctx.moveTo(18,0);ctx.lineTo(0,-5);ctx.lineTo(-12,0);ctx.lineTo(0,5);ctx.closePath();
    ctx.fillStyle=col;ctx.fill();ctx.strokeStyle='rgba(255,255,255,.7)';ctx.lineWidth=1.2;ctx.stroke();
    ctx.beginPath();ctx.arc(0,0,4.5,0,7);ctx.fillStyle='#fff';ctx.fill();
  }
  ctx.restore();
  // level pips
  if(!ghost){for(let i=0;i<lvl;i++){const px=t.x-(lvl-1)*6+i*12;
    ctx.save();ctx.translate(px,t.y-27);ctx.rotate(Math.PI/4);ctx.fillStyle=col;ctx.fillRect(-2.6,-2.6,5.2,5.2);ctx.restore();}}
}

  // floaters
  for(const f of floats){const a=1-f.t/f.ttl;

  // banner
  if(banner){
    const a=Math.min(1,(2.1-banner.t)/.25,banner.t/.4);

    try{ctx.letterSpacing='6px';}catch(e){}
    ctx.font='64px Impact,"Arial Narrow",sans-serif';
    ctx.strokeStyle='rgba(4,8,16,.9)';ctx.lineWidth=6;ctx.strokeText(banner.txt,640,236);
    ctx.fillStyle='#eaf4ff';ctx.fillText(banner.txt,640,236);
    try{ctx.letterSpacing='3px';}catch(e){}
    ctx.font='15px Consolas,Menlo,monospace';
    ctx.strokeStyle='rgba(4,8,16,.9)';ctx.lineWidth=4;ctx.strokeText(banner.sub,640,266);
    ctx.fillStyle='#ffd76a';ctx.fillText(banner.sub,640,266);
    try{ctx.letterSpacing='0px';}catch(e){}
    ctx.globalAlpha=1;
  }
  ctx.restore();

  // core-critical vignette (outside shake)
  if(state!=='menu'&&CORE.hp/CORE.max<=.3&&CORE.hp>0){
    const a=.12+.1*Math.sin(tNow*5);
    const g=ctx.createRadialGradient(640,360,300,640,360,760);

    ctx.fillStyle=g;ctx.fillRect(0,0,W,H);
  }
}

    el.pm.textContent=(waveNum===0?'DEPLOY PHASE · FIRST WAVE IN ':'NEXT WAVE IN ')+s+'s — SPACE TO LAUNCH';

/* ================= LOOP ================= */
let last=performance.now();
function loop(now){
  const dt=Math.min(.05,(now-last)/1000);last=now;tNow+=dt;
  for(const m of motes){m.y-=m.v*dt;m.x+=Math.sin(tNow*m.f+m.ph)*8*dt;if(m.y<-8){m.y=H+8;m.x=Math.random()*W;}}
  if(state==='play'&&!paused)update(dt);
  render();hud();
