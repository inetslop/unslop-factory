function doTick(){

  for(let i=0;i<g.snake.length;i++){
    if(i===g.snake.length-1&&!grow)continue;
    const s=g.snake[i];if(s.x===nx&&s.y===ny)return crash('tail');
  }
  const box=g.boxMap[nk];
  if(box){
    const ox=nx+g.dir.x,oy=ny+g.dir.y,ok=K(ox,oy);
    if(g.walls.has(ok)||g.boxMap[ok])return crash('jam');
    for(let i=0;i<g.snake.length;i++){
      if(i===g.snake.length-1&&!grow)continue;
      const s=g.snake[i];if(s.x===ox&&s.y===oy)return crash('jam');
    }
    box.px=box.x;box.py=box.y;box.x=nx;box.y=ny;
    g.boxMap[nk]=box;delete g.boxMap[ok];
    const was=box.onGoal;box.onGoal=g.goalSet.has(nk);

    if(box.onGoal&&!was){run.score+=100;syncScore();floatText(nx,ny,'+100');ringFx(nx,ny);sfx.goal();}
    updatePips();
  }
  g.prevSnake=g.snake.map(p=>({x:p.x,y:p.y}));
  g.snake.unshift({x:nx,y:ny});
  if(grow){
    g.foods.delete(nk);run.score+=25;syncScore();g.stats.food++;

}

  if(!g)return;
  const r=boardWrap.getBoundingClientRect();
  const aw=Math.max(120,r.width-26),ah=Math.max(120,r.height-26);
  cell=clamp(Math.floor(Math.min(aw/g.cols,ah/g.rows)),14,68);
  W=cell*g.cols;H=cell*g.rows;
  dpr=Math.min(2,window.devicePixelRatio||1);
  cv.style.width=W+'px';cv.style.height=H+'px';
  cv.width=Math.round(W*dpr);cv.height=Math.round(H*dpr);
  vigGrad=ctx.createRadialGradient(W/2,H/2,Math.min(W,H)*0.3,W/2,H/2,Math.max(W,H)*0.72);
  vigGrad.addColorStop(0,'rgba(0,0,0,0)');vigGrad.addColorStop(1,'rgba(0,0,0,.3)');
}

const CC=['#ffd45e','#f0a63c','#ff6b57','#8fd14f','#f4e9c8'];
function burst(x,y,color,n){for(let i=0;i<n;i++){const a=Math.random()*Math.PI*2,sp=1.2+Math.random()*3;
  particles.push({shape:'dot',x:x*cell+cell/2,y:y*cell+cell/2,vx:Math.cos(a)*sp,vy:Math.sin(a)*sp-0.6,g:0.07,life:1,dec:0.02+Math.random()*0.03,size:cell*(0.05+Math.random()*0.06),col:color});}}
function dustFx(x,y){for(let i=0;i<4;i++)particles.push({shape:'dot',x:x*cell+cell/2+(Math.random()*10-5),y:y*cell+cell*0.7,vx:(Math.random()-0.5)*0.6,vy:-0.3-Math.random()*0.4,g:0,life:0.8,dec:0.05,size:cell*(0.06+Math.random()*0.05),col:'rgba(215,230,205,.55)'});}
function ringFx(x,y){particles.push({shape:'ring',x:x*cell+cell/2,y:y*cell+cell/2,r:cell*0.2,grow:cell*0.03,life:1,dec:0.035,col:'#ffd45e'});}
function confettiBurst(){for(let i=0;i<64;i++)particles.push({shape:'rect',x:Math.random()*W,y:-8-Math.random()*60,vx:(Math.random()-0.5)*1.2,vy:1+Math.random()*2,g:0.035,life:1,dec:0.004,size:cell*(0.12+Math.random()*0.1),col:CC[i%5],rot:Math.random()*6.3,rv:(Math.random()-0.5)*0.25});}
function floatText(x,y,txt){floats.push({x:x*cell+cell/2,y:y*cell+cell*0.3,txt,t:0});}
function stepParticles(dt){const k=dt/16.667;
  for(let i=particles.length-1;i>=0;i--){const p=particles[i];
    p.x+=p.vx*k;p.y+=p.vy*k;if(p.g)p.vy+=p.g*k;
    if(p.rot!==undefined)p.rot+=p.rv*k;if(p.grow)p.r+=p.grow*k;
    p.life-=p.dec*k;
    if(p.life<=0||p.y>H+60)particles.splice(i,1);
  }
  for(let i=floats.length-1;i>=0;i--){floats[i].t+=dt;if(floats[i].t>950)floats.splice(i,1);}
}

/* ================= drawing ================= */
function rr(x,y,w,h,r){ctx.beginPath();ctx.moveTo(x+r,y);ctx.arcTo(x+w,y,x+w,y+h,r);ctx.arcTo(x+w,y+h,x,y+h,r);ctx.arcTo(x,y+h,x,y,r);ctx.arcTo(x,y,x+w,y,r);ctx.closePath();}
function animProg(now){return (state==='play'||state==='intro')?clamp((now-lastTick)/g.step,0,1):1;}

function drawBoard(){
  for(let y=0;y<g.rows;y++)for(let x=0;x<g.cols;x++){
    const px=x*cell,py=y*cell;
    if(g.walls.has(K(x,y))){
      const h=hash(x,y,g.idx+3);
      ctx.fillStyle='#1c3423';ctx.fillRect(px,py,cell,cell);
      ctx.fillStyle='rgba(0,0,0,'+(0.15+h*0.18).toFixed(2)+')';ctx.fillRect(px,py,cell,cell);
      ctx.fillStyle='rgba(255,255,255,.07)';ctx.fillRect(px,py,cell,3);
      ctx.fillStyle='rgba(0,0,0,.28)';ctx.fillRect(px,py+cell-3,cell,3);
      if(h<0.35){ctx.fillStyle='rgba(110,170,80,.35)';ctx.beginPath();
        ctx.arc(px+cell*(0.2+h*0.6),py+cell*(0.25+(h*5%1)*0.5),cell*0.13,0,7);ctx.fill();}
    }else{
      ctx.fillStyle=((x+y)&1)?'#2f6a3b':'#2b6136';ctx.fillRect(px,py,cell,cell);
      const h=hash(x,y,g.idx+7),h2=hash(y,x,g.idx+13);
      if(h<0.13){
        ctx.strokeStyle='rgba(120,205,130,.35)';ctx.lineWidth=1.4;
        const bx=px+cell*(0.25+h2*0.5),by=py+cell*(0.3+(h*7%1)*0.4);
        ctx.beginPath();ctx.moveTo(bx,by+3);ctx.lineTo(bx-2,by);ctx.moveTo(bx+3,by+3);ctx.lineTo(bx+2,by-1);ctx.stroke();
      }else if(h<0.2){
        ctx.fillStyle='rgba(0,0,0,.14)';ctx.beginPath();
        ctx.ellipse(px+cell*0.55,py+cell*0.6,cell*0.09,cell*0.06,0.5,0,7);ctx.fill();
      }else if(h<0.26){
        ctx.fillStyle='rgba(255,225,150,.5)';ctx.beginPath();
        ctx.arc(px+cell*(0.3+h2),py+cell*0.4,cell*0.045,0,7);ctx.fill();
      }
    }
  }
}
function drawGoals(now){
  for(const gl of g.goals){

    ctx.strokeStyle='rgba(244,233,200,.25)';ctx.lineWidth=1.5;ctx.setLineDash([4,5]);
    ctx.beginPath();ctx.arc(cx,cy,cell*0.36,now*0.001,now*0.001+Math.PI*1.7);ctx.stroke();

  }
}

function computeWarn(now){
  if(state!=='play'){lastWarn=null;return null;}
  const h=g.snake[0],fx=h.x+g.dir.x,fy=h.y+g.dir.y;
  const b=g.boxMap[K(fx,fy)];let w=null;
  if(b){const ox=fx+g.dir.x,oy=fy+g.dir.y;
    if(g.walls.has(K(ox,oy))||g.boxMap[K(ox,oy)])w=b;}
  if(w!==lastWarn&&w)sfx.warn();
  lastWarn=w;return w;
}
function drawBoxes(now){
  const warn=computeWarn(now),prog=animProg(now);
  for(const b of g.boxList){
    const x=lerp(b.px,b.x,prog)*cell+cell/2,y=lerp(b.py,b.y,prog)*cell+cell/2;
    let s=cell*0.76;if(b.onGoal)s*=1+0.04*Math.sin(now*0.006);
    const jx=(warn===b)?(Math.random()*2-1):0,jy=(warn===b)?(Math.random()*2-1):0;
    ctx.save();ctx.translate(x+jx,y+jy);

    if(b.onGoal){ctx.shadowColor='rgba(255,205,90,.85)';ctx.shadowBlur=cell*0.4;}

    ctx.beginPath();ctx.moveTo(-h+3,-h+2.5);ctx.lineTo(-h+2.5,h-3);ctx.moveTo(-h+2.5,-h+3);ctx.lineTo(h-3,-h+2.5);ctx.stroke();

    if(b.onGoal){ctx.strokeStyle='#8fd14f';ctx.lineWidth=2;ctx.beginPath();
      ctx.moveTo(0,-h+2);ctx.quadraticCurveTo(2,-h-cell*0.16,cell*0.12,-h-cell*0.14);ctx.stroke();}

      ctx.strokeStyle='rgba(255,80,60,'+a.toFixed(2)+')';ctx.lineWidth=3;rr(-h-3,-h-3,s+6,s+6,cell*0.12);ctx.stroke();}

  }
}
function drawSnake(now){
  const n=g.snake.length,prog=animProg(now),pts=[];
  for(let i=0;i<n;i++){
    const p=g.prevSnake[Math.min(i,g.prevSnake.length-1)];
    let x=lerp(p.x,g.snake[i].x,prog)*cell+cell/2,y=lerp(p.y,g.snake[i].y,prog)*cell+cell/2;
    if(i>0){const a=pts[i-1],dx=a.x-x,dy=a.y-y,L=Math.hypot(dx,dy)||1;
      const off=Math.sin(now*0.004+i*0.9)*cell*0.018;x+=(-dy/L)*off;y+=(dx/L)*off;}
    pts.push({x,y});
  }
  const wH=cell*0.62,wT=cell*0.3,deadFace=(state==='dying'||state==='dead');
  function mixc(t){return 'rgb('+Math.round(lerp(255,201,t))+','+Math.round(lerp(212,110,t))+','+Math.round(lerp(94,32,t))+')';}
  function segPass(extra,dark){
    for(let i=0;i<pts.length-1;i++){
      const t=i/(pts.length-1);
      ctx.strokeStyle=dark?'rgba(12,24,14,.5)':mixc(t);
      ctx.lineWidth=lerp(wH,wT,t)+extra;
      ctx.beginPath();ctx.moveTo(pts[i].x,pts[i].y);ctx.lineTo(pts[i+1].x,pts[i+1].y);ctx.stroke();
    }
  }
  ctx.lineCap='round';ctx.lineJoin='round';
  segPass(5,true);segPass(0,false);
  for(let i=1;i<pts.length;i+=2){const t=i/(pts.length-1);
    ctx.fillStyle='rgba(122,60,16,.4)';ctx.beginPath();ctx.arc(pts[i].x,pts[i].y,lerp(wH,wT,t)*0.3,0,7);ctx.fill();}
  const hd=pts[0],f=g.dir,hr=wH*0.58;
  ctx.fillStyle=mixc(0);ctx.beginPath();ctx.arc(hd.x,hd.y,hr,0,7);ctx.fill();
  ctx.strokeStyle='rgba(12,24,14,.5)';ctx.lineWidth=3;ctx.stroke();
  const exo=f.x*hr*0.3,eyo=f.y*hr*0.3,ax=-f.y,ay=f.x;
  const blink=deadFace?1:(now%3400<120?0.15:1);
  for(const sg of [-1,1]){
    const cx=hd.x+exo+ax*hr*0.42*sg,cy=hd.y+eyo+ay*hr*0.42*sg;
    if(deadFace){
      ctx.strokeStyle='#2b1a05';ctx.lineWidth=2.5;
      ctx.beginPath();
      ctx.moveTo(cx-hr*0.22,cy-hr*0.22);ctx.lineTo(cx+hr*0.22,cy+hr*0.22);
      ctx.moveTo(cx+hr*0.22,cy-hr*0.22);ctx.lineTo(cx-hr*0.22,cy+hr*0.22);ctx.stroke();
    }else{
      ctx.fillStyle='#fff';ctx.beginPath();ctx.arc(cx,cy,hr*0.33,0,7);ctx.fill();
      ctx.save();ctx.translate(cx,cy);ctx.scale(1,blink);
      ctx.fillStyle='#241608';ctx.beginPath();ctx.arc(f.x*hr*0.1,f.y*hr*0.1,hr*0.16,0,7);ctx.fill();
      ctx.restore();
    }
  }
  const cyc=now%2800;
  if(cyc<320&&!deadFace&&(state!=='clear'&&state!=='win')){
    const k=cyc/320,len=hr*(1+(k<0.5?k*2:(1-k)*2))*1.1;
    const tx=hd.x+f.x*hr,ty=hd.y+f.y*hr,ex2=hd.x+f.x*(hr+len),ey2=hd.y+f.y*(hr+len);
    ctx.strokeStyle='#ff5a5a';ctx.lineWidth=2.5;
    ctx.beginPath();ctx.moveTo(tx,ty);ctx.lineTo(ex2,ey2);
    ctx.moveTo(ex2,ey2);ctx.lineTo(ex2-ay*cell*0.08+f.x*cell*0.07,ey2+ax*cell*0.08+f.y*cell*0.07);
    ctx.moveTo(ex2,ey2);ctx.lineTo(ex2+ay*cell*0.08+f.x*cell*0.07,ey2-ax*cell*0.08+f.y*cell*0.07);
    ctx.stroke();
  }
}
function drawParticles(){
  for(const p of particles){
    if(p.shape==='ring'){ctx.globalAlpha=Math.max(0,p.life);ctx.strokeStyle=p.col;ctx.lineWidth=2.5;
      ctx.beginPath();ctx.arc(p.x,p.y,p.r,0,7);ctx.stroke();continue;}
    ctx.globalAlpha=Math.max(0,p.life);
    if(p.shape==='rect'){ctx.save();ctx.translate(p.x,p.y);ctx.rotate(p.rot);ctx.fillStyle=p.col;
      ctx.fillRect(-p.size/2,-p.size/3,p.size,p.size*0.66);ctx.restore();}
    else{ctx.fillStyle=p.col;ctx.beginPath();ctx.arc(p.x,p.y,Math.max(0.5,p.size*p.life),0,7);ctx.fill();}
  }
  ctx.globalAlpha=1;
}
function drawFloats(){
  for(const f of floats){

    ctx.globalAlpha=a;
    ctx.font='800 '+Math.round(cell*0.42)+'px ui-monospace,Menlo,Consolas,monospace';

    ctx.strokeText(f.txt,f.x,y);ctx.fillStyle='#ffd45e';ctx.fillText(f.txt,f.x,y);
  }
  ctx.globalAlpha=1;
}
function render(now){
  ctx.setTransform(dpr,0,0,dpr,0,0);
  if(shakeT>0){const k=shakeT/420;ctx.translate((Math.random()*2-1)*9*k,(Math.random()*2-1)*9*k);}
  drawBoard();drawGoals(now);drawFood(now);drawBoxes(now);drawSnake(now);
  drawParticles();drawFloats();
  if(vigGrad){ctx.fillStyle=vigGrad;ctx.fillRect(0,0,W,H);}
  if(flashA>0){ctx.fillStyle='rgba('+flashCol+','+flashA.toFixed(3)+')';ctx.fillRect(-20,-20,W+40,H+40);}
}

/* ================= main loop ================= */
function frame(now){
  requestAnimationFrame(frame);
  const dt=Math.min(50,now-lastFrame);lastFrame=now;
  if(state==='intro'&&now-introAt>=INTRO_MS){state='play';playStart=now;}
  if(state==='play'){acc+=dt;let n=0;while(acc>=g.step&&state==='play'&&n<4){doTick();acc-=g.step;n++;}}
  if(state==='dying'&&now-deathAt>700){state='dead';fillDead();showPanel('pDead');}
  if(shakeT>0)shakeT=Math.max(0,shakeT-dt);
  if(flashA>0)flashA=Math.max(0,flashA-dt*0.0015);
  stepParticles(dt);
  render(now);
}
