const CELL=80;

function genLevel(n){
  const nx=clamp(n+4,5,9),ny=clamp(n+3,4,7);
  const cols=nx*2-1,rows=ny*2-1;
  const g=new Uint8Array(cols*rows).fill(1);
  const id=(x,y)=>y*cols+x;
  const seen=new Set([id(1,1)]),stack=[[1,1]];
  while(stack.length){
    const top=stack[stack.length-1],cx=top[0],cy=top[1];
    const opts=shuffle([[2,0],[-2,0],[0,2],[0,-2]]).filter(p=>{
      const a=cx+p[0],b=cy+p[1];return a>=1&&a<cols&&b>=1&&b<rows&&!seen.has(id(a,b));});
    if(!opts.length){stack.pop();continue;}
    const p=opts[0];g[id(cx+p[0]/2,cy+p[1]/2)]=0;seen.add(id(cx+p[0],cy+p[1]));stack.push([cx+p[0],cy+p[1]]);
  }
  // braid: open a few extra passages so the maze has loops
  for(let y=1;y<rows-1;y++)for(let x=1;x<cols-1;x++){
    if(g[id(x,y)]!==1)continue;
    const h=g[id(x-1,y)]===0&&g[id(x+1,y)]===0, v=g[id(x,y-1)]===0&&g[id(x,y+1)]===0;
    if((h||v)&&Math.random()<.15)g[id(x,y)]=0;
  }
  // BFS distances from start (1,1)
  const dist=new Int32Array(cols*rows).fill(-1);dist[id(1,1)]=0;
  const q=[[1,1]];let qi=0;
  while(qi<q.length){const x=q[qi][0],y=q[qi][1];qi++;
    for(let d=0;d<4;d++){const a=x+[[1,0],[-1,0],[0,1],[0,-1]][d][0],b=y+[[1,0],[-1,0],[0,1],[0,-1]][d][1];
      if(a<0||b<0||a>=cols||b>=rows)continue;const i2=id(a,b);
      if(g[i2]===0&&dist[i2]<0){dist[i2]=dist[id(x,y)]+1;q.push([a,b]);}}}
  let portal=[1,1],mx=-1;
  for(let y=0;y<rows;y++)for(let x=0;x<cols;x++){const i2=id(x,y);if(g[i2]===0&&dist[i2]>mx){mx=dist[i2];portal=[x,y];}}
  // cores: spread out in the far half of the maze
  const K=Math.min(2+n,7),cand=[];
  for(let y=0;y<rows;y++)for(let x=0;x<cols;x++){const i2=id(x,y);
    if(g[i2]===0&&dist[i2]>mx*.3&&!(x===1&&y===1)&&!(x===portal[0]&&y===portal[1]))cand.push([x,y]);}
  shuffle(cand);const picked=[];
  if(cand.length)picked.push(cand.pop());
  while(picked.length<K&&cand.length){let bi=0,bd=-1;
    for(let i2=0;i2<cand.length;i2++){const x=cand[i2][0],y=cand[i2][1];let m=1e9;
      for(const p of picked)m=Math.min(m,(x-p[0])*(x-p[0])+(y-p[1])*(y-p[1]));
      if(m>bd){bd=m;bi=i2;}}
    picked.push(cand.splice(bi,1)[0]);}
  // loose (shootable) boulders in interior walls
  const D=Math.min(1+n,6),dC=[];

  shuffle(dC);const dList=dC.slice(0,D).map(p=>({x:p[0],y:p[1],ph:rand(0,TAU)}));
  return {cols,rows,g,portal,cores:picked,destructibles:dList,worldW:cols*CELL,worldH:rows*CELL,shapes:{}};
}

function updateFx(dt){
  for(let i=parts.length-1;i>=0;i--){const p=parts[i];p.life-=dt;
    if(p.life<=0){parts.splice(i,1);continue;}
    p.x+=p.vx*dt;p.y+=p.vy*dt;const dr=Math.exp(-(p.drag||2)*dt);p.vx*=dr;p.vy*=dr;
    if(p.rot!==undefined)p.rot+=p.vr*dt;if(p.type==='smoke')p.size+=14*dt;}
  for(let i=pops.length-1;i>=0;i--){pops[i].t+=dt;if(pops[i].t>=pops[i].dur)pops.splice(i,1);}
  shake*=Math.exp(-5*dt);if(shake<.05)shake=0;
}

/* ================= collisions ================= */
function circleRect(cx,cy,r,rx,ry){
  const x0=rx,y0=ry,x1=rx+CELL,y1=ry+CELL;
  const nx=clamp(cx,x0,x1),ny=clamp(cy,y0,y1);
  let dx=cx-nx,dy=cy-ny;const d2=dx*dx+dy*dy;
  if(d2>=r*r)return null;
  let d=Math.sqrt(d2);
  if(d<1e-6){const l=cx-x0,rr=x1-cx,t=cy-y0,b=y1-cy,m=Math.min(l,rr,t,b);
    if(m===l)return{nx:-1,ny:0,pen:r+l};if(m===rr)return{nx:1,ny:0,pen:r+rr};
    if(m===t)return{nx:0,ny:-1,pen:r+t};return{nx:0,ny:1,pen:r+b};}
  return{nx:dx/d,ny:dy/d,pen:r-d};
}
function collideShipWalls(){
  const x0=Math.floor((ship.x-ship.r)/CELL),x1=Math.floor((ship.x+ship.r)/CELL),
        y0=Math.floor((ship.y-ship.r)/CELL),y1=Math.floor((ship.y+ship.r)/CELL);
  for(let cy=y0;cy<=y1;cy++)for(let cx=x0;cx<=x1;cx++){
    if(cx<0||cy<0||cx>=L.cols||cy>=L.rows)continue;
    if(L.g[cy*L.cols+cx]!==1)continue;
    const h=circleRect(ship.x,ship.y,ship.r,cx*CELL,cy*CELL);if(!h)continue;
    ship.x+=h.nx*h.pen;ship.y+=h.ny*h.pen;
    const d=ship.vx*h.nx+ship.vy*h.ny;
    if(d<0){ship.vx-=d*h.nx;ship.vy-=d*h.ny;
      if(-d>140&&Math.random()<.3)sparkBurst(ship.x-h.nx*ship.r,ship.y-h.ny*ship.r,2,'#cfe2f4','#8fa8c8');}
  }
}
function collideRockWalls(r){
  const x0=Math.floor((r.x-r.r)/CELL),x1=Math.floor((r.x+r.r)/CELL),
        y0=Math.floor((r.y-r.r)/CELL),y1=Math.floor((r.y+r.r)/CELL);
  for(let cy=y0;cy<=y1;cy++)for(let cx=x0;cx<=x1;cx++){
    if(cx<0||cy<0||cx>=L.cols||cy>=L.rows)continue;
    if(L.g[cy*L.cols+cx]!==1)continue;
    const h=circleRect(r.x,r.y,r.r,cx*CELL,cy*CELL);if(!h)continue;
    r.x+=h.nx*h.pen;r.y+=h.ny*h.pen;
    const d=r.vx*h.nx+r.vy*h.ny;
    if(d<0){r.vx-=1.85*d*h.nx;r.vy-=1.85*d*h.ny;
      if(-d>120&&r.cool<=0){r.cool=.35;AudioSys.clink();dust(r.x-h.nx*r.r,r.y-h.ny*r.r,4);}}
  }
}

function spawnRock(){
  let x=0,y=0,ok=false;
  for(let t=0;t<50&&!ok;t++){
    const cx=irand(1,L.cols-2),cy=irand(1,L.rows-2);
    if(L.g[cy*L.cols+cx]!==0)continue;
    const px=(cx+.5)*CELL,py=(cy+.5)*CELL;
    if((px-ship.x)**2+(py-ship.y)**2<(CELL*3.2)**2)continue;
    x=px;y=py;ok=true;}
  if(!ok)return;
  const s=Math.random()<.55?2:(Math.random()<.6?1:0);
  const rr=[10,17,26][s],sp=(36+level*5)*rand(.8,1.2)*(1+(2-s)*.35),a=rand(0,TAU);
  rocks.push({x,y,vx:Math.cos(a)*sp,vy:Math.sin(a)*sp,size:s,r:rr,rot:rand(0,TAU),vr:rand(-1.4,1.4),birth:.5,cool:0,sh:makeRockShape(rr,false,true)});
}

/* ================= updates ================= */
const keys=new Set();
function updatePlaying(dt){
  levelTime+=dt;
  const left=keys.has('a')||keys.has('arrowleft'),right=keys.has('d')||keys.has('arrowright');
  const thr=keys.has('w')||keys.has('arrowup');thrusting=thr;
  ship.ang+=((right?1:0)-(left?1:0))*4.7*dt;
  if(thr){ship.vx+=Math.cos(ship.ang)*ACC*dt;ship.vy+=Math.sin(ship.ang)*ACC*dt;}
  const dr=Math.exp(-1.15*dt);ship.vx*=dr;ship.vy*=dr;
  const sp=Math.hypot(ship.vx,ship.vy);if(sp>MAXSP){ship.vx*=MAXSP/sp;ship.vy*=MAXSP/sp;}
  ship.x+=ship.vx*dt;ship.y+=ship.vy*dt;
  collideShipWalls();
  ship.cool-=dt;
  if(keys.has(' ')&&ship.cool<=0){ship.cool=.16;fireBullet();}
  AudioSys.thrust(thr);

  // bullets
  for(let i=bullets.length-1;i>=0;i--){
    const b=bullets[i];let dead=false;
    b.x+=b.vx*dt;b.y+=b.vy*dt;b.life-=dt;
    if(b.life<=0)dead=true;
    if(!dead){
      const cx=Math.floor(b.x/CELL),cy=Math.floor(b.y/CELL);
      if(cx<0||cy<0||cx>=L.cols||cy>=L.rows)dead=true;
      else{const id=cy*L.cols+cx;
        if(dset.has(id)){breakDestructible(id);dead=true;}
        else if(L.g[id]===1){sparkBurst(b.x,b.y,3,'#cfe2f4','#ffd166');dead=true;}}
    }
    if(!dead){
      for(let j=rocks.length-1;j>=0;j--){const r=rocks[j];if(r.birth>0)continue;
        const dx=r.x-b.x,dy=r.y-b.y;
        if(dx*dx+dy*dy<(r.r+3)*(r.r+3)){
          rocks.splice(j,1);dead=true;
          const pts=[100,50,20][r.size];score+=pts;popup(r.x,r.y,'+'+pts,'#cfe2f4');
          explosion(r.x,r.y,r.r);AudioSys.boom(.3+r.size*.3);shake=Math.max(shake,2+r.size*1.5);
          if(r.size>0){const cr=[10,17,26][r.size-1];
            for(let c2=0;c2<2;c2++){const a=rand(0,TAU),cs=(36+level*4)*rand(.9,1.4)*(1+r.size*.25);
              rocks.push({x:r.x,y:r.y,vx:r.vx*.4+Math.cos(a)*cs,vy:r.vy*.4+Math.sin(a)*cs,size:r.size-1,r:cr,
                rot:rand(0,TAU),vr:rand(-1.6,1.6),birth:0,cool:0,sh:makeRockShape(cr,false,true)});}}
          break;}}
    }
    if(dead)bullets.splice(i,1);
  }

  // rocks
  spawnTimer-=dt;
  if(spawnTimer<=0&&rocks.length<Math.min(1+level,9)){spawnRock();spawnTimer=clamp(7.5-level*.6,3,7.5)*rand(.8,1.2);}
  for(let i=rocks.length-1;i>=0;i--){const r=rocks[i];
    r.birth-=dt;r.cool-=dt;r.x+=r.vx*dt;r.y+=r.vy*dt;r.rot+=r.vr*dt;
    collideRockWalls(r);
    if(r.birth<=0&&ship.alive){
      const dx=ship.x-r.x,dy=ship.y-r.y;
      if(dx*dx+dy*dy<(r.r+ship.r)*(r.r+ship.r)){hitShip();break;}}}

  // cores
  for(const c of L.cores){if(c.taken)continue;
    const cx=(c.cx+.5)*CELL,cy=(c.cy+.5)*CELL;
    if((ship.x-cx)**2+(ship.y-cy)**2<(ship.r+14)**2){
      c.taken=true;score+=300;popup(cx,cy,'+300','#ffd166');burstCore(cx,cy);AudioSys.coreSfx();
      if(L.cores.every(q=>q.taken))activateBeacon();}}

  // beacon
  if(beacon.active){
    const bx=(beacon.cx+.5)*CELL,by=(beacon.cy+.5)*CELL;
    if((ship.x-bx)**2+(ship.y-by)**2<(ship.r+20)**2){levelClear();return;}}

  if(ship.invuln>0)ship.invuln-=dt;
  updateFx(dt);hudSync();
}

function drawRock(r){
  const born=clamp(r.birth/.5,0,1);
  ctx.save();ctx.translate(r.x,r.y);ctx.rotate(r.rot);
  if(r.birth>0){const s2=.4+.6*born;ctx.scale(s2,s2);ctx.globalAlpha=.3+.7*born;}
  paintRock(ctx,0,0,r.sh);ctx.restore();
  if(r.birth>0&&born<.6){ctx.save();ctx.translate(r.x,r.y);
    ctx.setLineDash([5,6]);ctx.strokeStyle='rgba(255,120,80,'+(.7*(1-born/.6))+')';ctx.lineWidth=2;
    ctx.beginPath();ctx.arc(0,0,r.r+8,perfT*3,perfT*3+TAU);ctx.stroke();ctx.setLineDash([]);ctx.restore();}
  ctx.globalAlpha=1;
}
function drawShip(){
  if(!ship||!ship.alive)return;
  let sc2=1,al=1;
  if(state==='clear'){const k=clamp(clearT/.45,0,1);sc2=1-k;al=1-k;if(k>=1)return;}
  if(ship.invuln>0&&state==='playing')al*=.45+.55*Math.abs(Math.sin(perfT*9));
  ctx.save();ctx.translate(ship.x,ship.y);ctx.rotate(ship.ang);ctx.scale(sc2,sc2);ctx.globalAlpha=al;
  if(thrusting){const fl=ship.r*(1.4+Math.random()*.9+Math.sin(perfT*60)*.25);
    ctx.beginPath();ctx.moveTo(-7,-4.2);ctx.lineTo(-7-fl,0);ctx.lineTo(-7,4.2);ctx.closePath();
    ctx.fillStyle='rgba(255,157,69,.85)';ctx.fill();
    ctx.beginPath();ctx.moveTo(-6.5,-2.2);ctx.lineTo(-6.5-fl*.55,0);ctx.lineTo(-6.5,2.2);ctx.closePath();
    ctx.fillStyle='rgba(255,240,200,.95)';ctx.fill();}
  ctx.beginPath();ctx.moveTo(ship.r*1.15,0);ctx.lineTo(-9,8);ctx.lineTo(-5.5,0);ctx.lineTo(-9,-8);ctx.closePath();
  ctx.fillStyle='rgba(180,235,255,.14)';ctx.fill();
  ctx.lineWidth=2;ctx.strokeStyle='#dff4ff';ctx.shadowColor='rgba(95,232,255,.9)';ctx.shadowBlur=10;ctx.stroke();ctx.shadowBlur=0;
  ctx.restore();
  if(ship.invuln>0&&state==='playing'){ctx.save();ctx.translate(ship.x,ship.y);
    ctx.strokeStyle='rgba(125,238,255,'+(.3+.3*Math.sin(perfT*7))+')';ctx.lineWidth=2;
    ctx.beginPath();ctx.arc(0,0,ship.r+9,0,TAU);ctx.stroke();
    ctx.setLineDash([4,10]);ctx.lineDashOffset=-perfT*40;ctx.strokeStyle='rgba(125,238,255,.7)';
    ctx.beginPath();ctx.arc(0,0,ship.r+13,0,TAU);ctx.stroke();ctx.setLineDash([]);ctx.restore();}
  ctx.globalAlpha=1;
}

function render(){
  ctx.drawImage(bgCv,0,0,W,H);
  for(const s of stars){const tw=.35+.65*Math.pow(Math.sin(perfT*s.tw+s.ph),2);
    ctx.globalAlpha=tw*(.35+.65*s.z);ctx.fillStyle='#d8e9ff';ctx.fillRect(s.x,s.y,s.r,s.r);}
  ctx.globalAlpha=1;
  if(meteor){const mx2=meteor.x-meteor.vx*.07,my2=meteor.y-meteor.vy*.07;
    const g2=ctx.createLinearGradient(meteor.x,meteor.y,mx2,my2);
    g2.addColorStop(0,'rgba(255,255,255,.9)');g2.addColorStop(1,'rgba(255,255,255,0)');
    ctx.strokeStyle=g2;ctx.lineWidth=1.6;ctx.beginPath();ctx.moveTo(meteor.x,meteor.y);ctx.lineTo(mx2,my2);ctx.stroke();}
  if((state==='start'||state==='gameover')&&amb.length){
    for(const a of amb){ctx.save();ctx.translate(a.x,a.y);ctx.rotate(a.rot);ctx.globalAlpha=.35;paintRock(ctx,0,0,a.sh);ctx.restore();}
    ctx.globalAlpha=1;}

  if(L){
    const hudH=56,availW=Math.max(80,W-24),availH=Math.max(80,H-hudH-20);
    const sc=Math.min(availW/L.worldW,availH/L.worldH);
    const ox=(W-L.worldW*sc)/2+(shake?rand(-1,1)*shake:0);
    const oy=hudH+(H-hudH-12-L.worldH*sc)/2+(shake?rand(-1,1)*shake:0);
    ctx.save();ctx.translate(ox,oy);ctx.scale(sc,sc);
    // arena frame + corner ticks
    ctx.strokeStyle='rgba(95,232,255,.14)';ctx.lineWidth=2;ctx.strokeRect(-8,-8,L.worldW+16,L.worldH+16);
    ctx.strokeStyle='rgba(95,232,255,.5)';ctx.lineWidth=3;const T=26;
    ctx.beginPath();
    ctx.moveTo(-8,-8);ctx.lineTo(-8+T,-8);ctx.moveTo(-8,-8);ctx.lineTo(-8,-8+T);
    ctx.moveTo(L.worldW+8,-8);ctx.lineTo(L.worldW+8-T,-8);ctx.moveTo(L.worldW+8,-8);ctx.lineTo(L.worldW+8,-8+T);
    ctx.moveTo(-8,L.worldH+8);ctx.lineTo(-8+T,L.worldH+8);ctx.moveTo(-8,L.worldH+8);ctx.lineTo(-8,L.worldH+8-T);
    ctx.moveTo(L.worldW+8,L.worldH+8);ctx.lineTo(L.worldW+8-T,L.worldH+8);ctx.moveTo(L.worldW+8,L.worldH+8);ctx.lineTo(L.worldW+8,L.worldH+8-T);
    ctx.stroke();
    ctx.drawImage(wallCv,0,0,L.worldW,L.worldH);
    drawDestructibleGlow();
    drawBeacon();
    for(const c of L.cores)drawCore(c);
    for(const b of bullets){
      const tx=b.x-b.vx*.02,ty=b.y-b.vy*.02;
      ctx.strokeStyle='rgba(255,224,130,.9)';ctx.lineWidth=3.4;ctx.beginPath();ctx.moveTo(tx,ty);ctx.lineTo(b.x,b.y);ctx.stroke();
      ctx.strokeStyle='#ffffff';ctx.lineWidth=1.4;ctx.beginPath();ctx.moveTo(tx,ty);ctx.lineTo(b.x,b.y);ctx.stroke();}
    for(const r of rocks)drawRock(r);
    drawShip();
    // particles
    for(const p of parts){const a=clamp(p.life/p.max,0,1);
      if(p.type==='shard'){ctx.save();ctx.translate(p.x,p.y);ctx.rotate(p.rot);ctx.globalAlpha=a;ctx.fillStyle=p.col;
        ctx.beginPath();ctx.moveTo(p.size,0);ctx.lineTo(-p.size*.7,p.size*.6);ctx.lineTo(-p.size*.5,-p.size*.7);ctx.closePath();ctx.fill();ctx.restore();}
      else if(p.type==='smoke'){ctx.globalAlpha=a*.2;ctx.fillStyle='#aab4c8';ctx.beginPath();ctx.arc(p.x,p.y,p.size,0,TAU);ctx.fill();}
      else{ctx.globalAlpha=a;ctx.strokeStyle=p.col;ctx.lineWidth=p.size;
        ctx.beginPath();ctx.moveTo(p.x,p.y);ctx.lineTo(p.x-p.vx*.035,p.y-p.vy*.035);ctx.stroke();}}
    ctx.globalAlpha=1;
    // popups
    for(const p of pops){const k=p.t/p.dur;ctx.globalAlpha=1-k;ctx.fillStyle=p.col;
      ctx.font='900 16px "Arial Black",Impact,sans-serif';ctx.textAlign='center';ctx.fillText(p.txt,p.x,p.y-26*k);}
    ctx.globalAlpha=1;
    if(state==='clear'){const k=clamp(clearT/1.6,0,1);
      const bx=(beacon.cx+.5)*CELL,by=(beacon.cy+.5)*CELL;
      for(let i=0;i<3;i++){const kk=clamp(k*1.3-i*.25,0,1);

          ctx.strokeStyle='rgba(160,240,255,'+((1-kk)*.8)+')';ctx.lineWidth=3+8*(1-kk);ctx.stroke();}}}

  }

    ctx.fillStyle='rgba(235,248,255,'+k+')';ctx.fillRect(0,0,W,H);}
  if(levelFade>0){ctx.fillStyle='rgba(2,5,10,'+clamp(levelFade,0,1)+')';ctx.fillRect(0,0,W,H);}
}

let last=performance.now();
function frame(now){
  requestAnimationFrame(frame);
  let dt=(now-last)/1000;last=now;if(dt>0.05)dt=0.05;
  perfT+=dt;
  if(needsBg){buildBg();buildStars();buildAmb();needsBg=false;}
  updateStars(dt);updateMeteor(dt);
  levelFade=Math.max(0,levelFade-dt*1.6);
  if(state==='playing')updatePlaying(dt);
  else if(state==='dying')updateDying(dt);
  else if(state==='clear')updateClear(dt);
  else updateFx(dt);
  if(state==='start'||state==='gameover'){
    for(const a of amb){a.x+=a.vx*dt;a.y+=a.vy*dt;a.rot+=a.vr*dt;
      if(a.x<-80)a.x=W+80;if(a.x>W+80)a.x=-80;if(a.y<-80)a.y=H+80;if(a.y>H+80)a.y=-80;}}
  render();
}

      else if(k==='m')toggleMute();
      else if(k==='r')startGame();
      break;

      if(k==='p'||k==='escape'||k==='enter')resumeGame();
      else if(k==='r')startGame();
      else if(k==='m')toggleMute();
      break;
    case 'gameover':if(k==='r'||k==='enter')startGame();break;
  }
});
