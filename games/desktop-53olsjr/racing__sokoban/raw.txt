function resolveC(px,py,r,rx,ry,rw,rh){
 const cx=Math.max(rx,Math.min(px,rx+rw)), cy=Math.max(ry,Math.min(py,ry+rh));
 let dx=px-cx, dy=py-cy; const d2=dx*dx+dy*dy;
 if(d2>r*r) return null;
 if(d2>1e-8){ const d=Math.sqrt(d2); return {nx:dx/d, ny:dy/d, pen:r-d}; }
 // center inside rect
 const l=px-rx, rgt=rx+rw-px, tp=py-ry, bt=ry+rh-py;
 const m=Math.min(l,rgt,tp,bt);
 if(m===l) return {nx:-1,ny:0,pen:r+l};
 if(m===rgt) return {nx:1,ny:0,pen:r+rgt};
 if(m===tp) return {nx:0,ny:-1,pen:r+tp};
 return {nx:0,ny:1,pen:r+bt};
}

function resolveC(px,py,r,rx,ry,rw,rh){
  const cx=Math.max(rx,Math.min(px,rx+rw)),cy=Math.max(ry,Math.min(py,ry+rh));
  let dx=px-cx,dy=py-cy;const d2=dx*dx+dy*dy;
  if(d2>r*r)return null;
  if(d2>1e-8){const d=Math.sqrt(d2);return{nx:dx/d,ny:dy/d,pen:r-d};}
  const l=px-rx,rgt=rx+rw-px,tp=py-ry,bt=ry+rh-py,m=Math.min(l,rgt,tp,bt);
  if(m===l)return{nx:-1,ny:0,pen:r+l};
  if(m===rgt)return{nx:1,ny:0,pen:r+rgt};
  if(m===tp)return{nx:0,ny:-1,pen:r+tp};
  return{nx:0,ny:1,pen:r+bt};
}
function canPush(b,dx,dy){
  const cx=Math.round(b.x),cy=Math.round(b.y);
  if(Math.abs(b.x-cx)>0.02||Math.abs(b.y-cy)>0.02)return false;
  const tx=cx+dx,ty=cy+dy;
  if(!floorSet.has(key(tx,ty)))return false;
  for(const o of boxes){if(o===b)continue;if(Math.round(o.x)===tx&&Math.round(o.y)===ty)return false;}
  return true;
}

  for(let cy=y0;cy<=y1;cy++)for(let cx=x0;cx<=x1;cx++){
    if(!walls.has(key(cx,cy)))continue;
    const ov=resolveC(p.x,p.y,R,cx,cy,1,1);if(!ov)continue;
    p.x+=ov.nx*ov.pen;p.y+=ov.ny*ov.pen;
    const vn=p.vx*ov.nx+p.vy*ov.ny;
    if(vn<0){p.vx-=ov.nx*vn;p.vy-=ov.ny*vn;
      if(-vn>1.5)wallHit(-vn,p.x-ov.nx*R,p.y-ov.ny*R,ov.nx,ov.ny);}
  }
}
function collideBoxes(){
  const p=player;
  for(const b of boxes){
    if(b.state==='returning'){b.x=b.ox;b.y=b.oy;b.state='rest';}
    const ov=resolveC(p.x,p.y,R,b.x,b.y,1,1);
    if(!ov){if(b.state==='pushing')b.lost=true;continue;}
    b.lost=false;
    let nx=0,ny=0;
    if(Math.abs(ov.nx)>=Math.abs(ov.ny))nx=Math.sign(ov.nx)||1;else ny=Math.sign(ov.ny)||1;
    const bdx=-nx,bdy=-ny,into=p.vx*bdx+p.vy*bdy;
    if(b.state==='pushing'){
      if(b.pdx===bdx&&b.pdy===bdy){
        anyPush=true;
        const off=(b.x-b.ox)*bdx+(b.y-b.oy)*bdy;
        let mv=Math.min(ov.pen,Math.max(0,1-off));
        b.x+=bdx*mv;b.y+=bdy*mv;
        if(ov.pen>mv){const rm=ov.pen-mv;p.x+=nx*rm;p.y+=ny*rm;
          const vn=p.vx*nx+p.vy*ny;if(vn>0){p.vx-=nx*vn;p.vy-=ny*vn;}}
      }else{pushOut(p,ov);sfx.blocked();}
    }else{
      if(into>0.15&&canPush(b,bdx,bdy)){
        b.state='pushing';b.ox=b.x;b.oy=b.y;b.pdx=bdx;b.pdy=bdy;anyPush=true;
        const mv=Math.min(ov.pen,1);
        b.x+=bdx*mv;b.y+=bdy*mv;
        if(ov.pen>mv){const rm=ov.pen-mv;p.x+=nx*rm;p.y+=ny*rm;
          const vn=p.vx*nx+p.vy*ny;if(vn>0){p.vx-=nx*vn;p.vy-=ny*vn;}}
      }else{pushOut(p,ov);if(into>1.3)sfx.blocked();}
    }
  }
}
function settleBoxes(dt){
  for(const b of boxes){
    if(b.state==='pushing'){
      const off=(b.x-b.ox)*b.pdx+(b.y-b.oy)*b.pdy;
      if(off>=0.999){b.x=b.ox+b.pdx;b.y=b.oy+b.pdy;b.state='rest';onBoxMoved(b);}
      else if(b.lost)b.state='returning';
    }else if(b.state==='returning'){
      const dx=b.ox-b.x,dy=b.oy-b.y,d=Math.hypot(dx,dy),mv=14*dt;
      if(d<=mv+0.001){b.x=b.ox;b.y=b.oy;b.state='rest';}
      else{b.x+=dx/d*mv;b.y+=dy/d*mv;}
    }
  }
}
function stepPhysics(dt){
  const p=player;
  let ix=(keys.right?1:0)-(keys.left?1:0),iy=(keys.down?1:0)-(keys.up?1:0);
  if(ix&&iy){ix*=0.7071;iy*=0.7071;}
  p.vx+=ix*ACC*dt;p.vy+=iy*ACC*dt;
  const dr=Math.exp(-((ix||iy)?DRAG_IN:DRAG_COAST)*dt);
  p.vx*=dr;p.vy*=dr;
  const sp=Math.hypot(p.vx,p.vy);
  if(sp>MAXV){p.vx*=MAXV/sp;p.vy*=MAXV/sp;}
  p.x+=p.vx*dt;p.y+=p.vy*dt;
  anyPush=false;
  for(let it=0;it<2;it++){collideWalls();collideBoxes();}
  settleBoxes(dt);
}

}

  moves++;b.squash=1;

  if(goals.has(key(gx,gy))){

  }else{
    b.onGoal=false;if(gate('clunk',80)){sfx.clunk();shake(0.12);puff(b.x+0.5,b.y+0.5,4);}
  }
  if(goalsFilled()===goalList.length)winSequence();
}

let physAcc=0;
function update(dt,t){
  shakeAmt*=Math.exp(-5*dt);if(shakeAmt<0.002)shakeAmt=0;
  flashA=Math.max(0,flashA-dt*1.4);
  redFlash=Math.max(0,redFlash-dt*1.1);
  for(const b of boxes)b.squash*=Math.exp(-8*dt);

  if(state==='menu'){
    if(Math.random()<dt*2)particles.push({type:'dust',x:Math.random()*cols,y:Math.random()*rows,
      vx:(Math.random()-.5)*.3,vy:-.15-Math.random()*.2,life:2,max:2,size:.05,alpha:.14});
  }else if(state!=='paused'){
    for(let i=particles.length-1;i>=0;i--){const pt=particles[i];pt.life-=dt;
      if(pt.life<=0){particles.splice(i,1);continue;}
      pt.x+=pt.vx*dt;pt.y+=pt.vy*dt;
      if(pt.type==='dust'){pt.vx*=Math.exp(-2.5*dt);pt.vy*=Math.exp(-2.5*dt);pt.size+=dt*0.22;}
      else if(pt.type==='conf'){pt.vy+=4.6*dt;pt.rot+=pt.vr*dt;}
      else if(pt.type==='ring')pt.r+=pt.vr*dt;
    }
  }

  if(state==='countdown'){
    cd-=dt;
    const val=cd>2.6?3:cd>1.6?2:cd>0.6?1:0;
    if(val!==lastCdVal){lastCdVal=val;
      countNum.textContent=val?''+val:'GO!';
      countNum.classList.toggle('go',val===0);
      countNum.classList.remove('pop');void countNum.offsetWidth;countNum.classList.add('pop');
      if(val)sfx.count();else sfx.go();}
    if(cd<=0.45){state='playing';showScreen(null);}
  }
  else if(state==='playing'){
    physAcc+=dt;const h=1/120;let n=0;
    while(physAcc>=h&&n<12&&state==='playing'){stepPhysics(h);physAcc-=h;n++;}
    if(n===12)physAcc=0;
    const p=player,sp=Math.hypot(p.vx,p.vy);
    if(sp>0.35){const ta=Math.atan2(p.vy,p.vx);let da=ta-p.ang;
      while(da>Math.PI)da-=6.283;while(da<-Math.PI)da+=6.283;
      p.ang+=da*Math.min(1,dt*14);}
    if(sp>1.5){const bx=p.x-p.vx/sp*0.3,by=p.y-p.vy/sp*0.3,px=-p.vy/sp,py=p.vx/sp,
      a=Math.min(0.4,(sp-1.5)/2.5);
      marks.push({x:bx+px*0.17,y:by+py*0.17,a:a},{x:bx-px*0.17,y:by-py*0.17,a:a});
      if(marks.length>600)marks.splice(0,marks.length-600);}
    if(anyPush&&sp>0.8){const b=boxes.find(o=>o.state==='pushing');
      if(b)puff(b.x+0.5-b.pdx*0.4,b.y+0.5-b.pdy*0.4,1);}
    const prev=Math.ceil(timeLeft);timeLeft-=dt;const cur=Math.ceil(timeLeft);
    if(cur<prev&&cur<=10&&cur>0)sfx.tick();
    if(timeLeft<=0){timeLeft=0;failSequence();}
  }
  else if(state==='winning'){winT+=dt;if(winT>1.25)showComplete();}
  else if(state==='failing'){failT+=dt;if(failT>0.6){state='fail';showScreen('sFail');}}

  updateEngine();
  syncHud();
}
