function closestOnSeg(px,py,x1,y1,x2,y2){ const dx=x2-x1, dy=y2-y1; const L2=dx*dx+dy*dy||1e-9; let t=((px-x1)*dx+(py-y1)*dy)/L2; t=Math.max(0,Math.min(1,t)); return {x:x1+t*dx, y:y1+t*dy, t}; }
function collideSeg(b, x1,y1,x2,y2, rest){ const c=closestOnSeg(b.x,b.y,x1,y1,x2,y2); let dx=b.x-c.x, dy=b.y-c.y; const d=Math.hypot(dx,dy); if(d < b.r){ if(d<1e-6){ // degenerate: push along seg normal
      const nx=-(y2-y1), ny=(x2-x1); const nl=Math.hypot(nx,ny); dx=nx/nl; dy=ny/nl; } else { dx/=d; dy/=d; }
   b.x = c.x + dx*b.r; b.y = c.y + dy*b.r;
   const vn = b.vx*dx + b.vy*dy; if(vn<0){ b.vx -= (1+rest)*vn*dx; b.vy -= (1+rest)*vn*dy; }
   return true; } return false; }

let AC=null,master=null,muted=false;
function initAudio(){
  if(AC){ if(AC.state==='suspended') AC.resume(); return; }
  try{ AC=new (window.AudioContext||window.webkitAudioContext)();
    master=AC.createGain(); master.gain.value=muted?0:0.16; master.connect(AC.destination);
  }catch(e){ AC=null; }
}
function tone(f0,f1,dur,type,vol,when){
  if(!AC)return; type=type||'sine'; vol=vol==null?1:vol; when=when||0;
  const t=AC.currentTime+when,o=AC.createOscillator(),g=AC.createGain();
  o.type=type; o.frequency.setValueAtTime(Math.max(20,f0),t);
  o.frequency.exponentialRampToValueAtTime(Math.max(20,f1),t+dur);
  g.gain.setValueAtTime(vol,t); g.gain.exponentialRampToValueAtTime(0.001,t+dur);
  o.connect(g); g.connect(master); o.start(t); o.stop(t+dur+0.03);
}
function noiseS(dur,vol,fc,when){
  if(!AC)return; vol=vol==null?1:vol; fc=fc||800; when=when||0;
  const t=AC.currentTime+when,n=Math.max(1,Math.floor(AC.sampleRate*dur)),
        buf=AC.createBuffer(1,n,AC.sampleRate),d=buf.getChannelData(0);
  for(let i=0;i<n;i++) d[i]=(Math.random()*2-1)*(1-i/n);
  const s=AC.createBufferSource(); s.buffer=buf;
  const f=AC.createBiquadFilter(); f.type='lowpass'; f.frequency.value=fc;
  const g=AC.createGain(); g.gain.value=vol;
  s.connect(f); f.connect(g); g.connect(master); s.start(t);
}

/* ================= PHYSICS ================= */
function closestOnSeg(px,py,s){
  const dx=s.x2-s.x1, dy=s.y2-s.y1, L2=dx*dx+dy*dy||1e-9;
  let t=((px-s.x1)*dx+(py-s.y1)*dy)/L2; t=t<0?0:t>1?1:t;
  return [s.x1+t*dx, s.y1+t*dy];
}
function hitSeg(b,s,e){
  const c=closestOnSeg(b.x,b.y,s); let dx=b.x-c[0], dy=b.y-c[1]; let d=Math.hypot(dx,dy);
  if(d>=BALL_R)return false;
  if(d<1e-6){dx=0;dy=-1;} else {dx/=d;dy/=d;}
  b.x=c[0]+dx*BALL_R; b.y=c[1]+dy*BALL_R;
  const vn=b.vx*dx+b.vy*dy; if(vn<0){ b.vx-=(1+e)*vn*dx; b.vy-=(1+e)*vn*dy; }
  return true;
}
function hitSlingFace(b,sg,i1,i2,kick){
  const p1=sg.pts[i1],p2=sg.pts[i2], s={x1:p1[0],y1:p1[1],x2:p2[0],y2:p2[1]};
  const c=closestOnSeg(b.x,b.y,s); let dx=b.x-c[0], dy=b.y-c[1]; let d=Math.hypot(dx,dy);
  if(d>=BALL_R)return false;
  if(d<1e-6){dx=0;dy=-1;} else {dx/=d;dy/=d;}
  b.x=c[0]+dx*BALL_R; b.y=c[1]+dy*BALL_R;
  if(kick){ b.vx=sg.nx*640; b.vy=sg.ny*640; sg.f=0.15; b.splashes=0;
    score+=25; ftext(b.x,b.y-12,'+25','#ff8fc0',12); sfx.sling(); shake(2,0.1); }
  else { const vn=b.vx*dx+b.vy*dy; if(vn<0){ b.vx-=1.15*vn*dx; b.vy-=1.15*vn*dy; } }
  return true;
}
function hitPad(b,p,turtle){
  const s={x1:p.x-p.hl,y1:p.y,x2:p.x+p.hl,y2:p.y};
  const c=closestOnSeg(b.x,b.y,s); let dx=b.x-c[0], dy=b.y-c[1]; let d=Math.hypot(dx,dy);
  const Rr=BALL_R+p.r; if(d>=Rr)return;
  if(d<1e-6){dx=0;dy=-1;d=1;} const nx=dx/d, ny=dy/d;
  b.x=c[0]+nx*Rr; b.y=c[1]+ny*Rr;
  const rel=(b.vx-p.vx)*nx+b.vy*ny; if(rel<0){ b.vx-=1.3*rel*nx; b.vy-=1.3*rel*ny; }
  if(ny<-0.5 && Math.abs(b.vy)<420){
    b.vy=turtle?-380:-170; b.vx=p.vx*1.1+(b.vx-p.vx)*0.15; b.splashes=0;
    if(uiT-lastPadSnd>0.09){ lastPadSnd=uiT; if(turtle)sfx.boing(); else noiseS(0.05,0.18,700); }
  } else { b.vx+=(p.vx-b.vx)*0.1; }
}
function step(dt){
  for(const f of flips){
    f.prev=f.ang; const tgt=f.pressed?f.active:f.rest;
    const d=tgt-f.ang, m=30*dt; f.ang+=d<-m?-m:d>m?m:d; f.w=(f.ang-f.prev)/dt;
  }
  for(const b of balls){
    if(b.state!=='live'||b.dead)continue;
    b.vy+=G*dt; const dm=1-0.05*dt; b.vx*=dm; b.vy*=dm;
    b.x+=b.vx*dt; b.y+=b.vy*dt;
    for(const s of walls) hitSeg(b,s,0.42);
    for(const sg of slings){ hitSlingFace(b,sg,2,0,true); hitSlingFace(b,sg,0,1,false); hitSlingFace(b,sg,1,2,false); }
    for(const bp of bumpers){
      let dx=b.x-bp.x, dy=b.y-bp.y; const d=Math.hypot(dx,dy);
      if(d<BALL_R+bp.r){
        if(d<1e-6){dx=1;dy=0;d=1;} const nx=dx/d, ny=dy/d;
        b.x=bp.x+nx*(BALL_R+bp.r); b.y=bp.y+ny*(BALL_R+bp.r);
        b.vx=nx*560; b.vy=ny*560; bp.f=0.18; b.splashes=0;
        score+=15; sfx.bumper(); shake(2,0.1);
      }
    }
    for(const p of turtles) hitPad(b,p,true);
    for(const p of logs) hitPad(b,p,false);
    for(const f of flips){
      const ex=f.px+Math.cos(f.ang)*f.len, ey=f.py+Math.sin(f.ang)*f.len;
      const c=closestOnSeg(b.x,b.y,{x1:f.px,y1:f.py,x2:ex,y2:ey});
      let dx=b.x-c[0], dy=b.y-c[1]; let d=Math.hypot(dx,dy); const Rr=BALL_R+f.r;
      if(d<Rr){
        if(d<1e-6){dx=0;dy=-1;d=1;} const nx=dx/d, ny=dy/d;
        b.x=c[0]+nx*Rr; b.y=c[1]+ny*Rr;
        const svx=-f.w*(c[1]-f.py), svy=f.w*(c[0]-f.px);
        const rel=(b.vx-svx)*nx+(b.vy-svy)*ny; if(rel<0){ b.vx-=1.35*rel*nx; b.vy-=1.35*rel*ny; }
      }
    }
    const sp=Math.hypot(b.vx,b.vy); if(sp>MAXV){ const k=MAXV/sp; b.vx*=k; b.vy*=k; }
    if(b.splCd>0)b.splCd-=dt;
    let dead=false;
    for(const c of cars){
      const rx=c.x-c.w/2, ry=c.y-13, px=clamp(b.x,rx,rx+c.w), py=clamp(b.y,ry,ry+26);
      const ddx=b.x-px, ddy=b.y-py;
      if(ddx*ddx+ddy*ddy < BALL_R*BALL_R){ loseBall(b,'squash'); dead=true; break; }
    }
    if(dead)continue;
    if(b.y>506 && b.x<428 && b.splCd<=0){ splash(b); if(b.state!=='live')continue; }
    if(b.y>736){ loseBall(b,'drain'); continue; }
    b.x=clamp(b.x,14,466); b.y=Math.max(b.y,14);
    for(const h of homes){ const dx=b.x-h.x, dy=b.y-h.y; if(dx*dx+dy*dy<144){ captureHome(b,h); break; } }
    if(b.dead)continue;
    if(b.x>434 && b.y>672){
      const v=Math.hypot(b.vx,b.vy), other=balls.find(o=>o!==b&&o.state==='wait');
      if(!other && v<80){ b.state='wait'; b.x=449; b.y=689; b.vx=b.vy=0; }
      else if(other && b.kickCd<=0){ b.vx=-260; b.vy=-180; b.kickCd=1; }
    } else if(b.kickCd>0) b.kickCd-=dt;
  }
  balls=balls.filter(b=>!b.dead);
}

function drawFrog(x,y,a,r,bl){
  ctx.fillStyle='#3f8a2e';
  for(const s of[-1,1]){
    const lx=x+Math.cos(a+Math.PI+s*0.95)*r*0.85, ly=y+Math.sin(a+Math.PI+s*0.95)*r*0.85;
    ctx.beginPath(); ctx.arc(lx,ly,r*0.3,0,7); ctx.fill();
    const fx2=x+Math.cos(a+s*1.5)*r*0.9, fy2=y+Math.sin(a+s*1.5)*r*0.9;
    ctx.beginPath(); ctx.arc(fx2,fy2,r*0.24,0,7); ctx.fill();
  }
  const g=ctx.createRadialGradient(x-r*0.3,y-r*0.35,r*0.2,x,y,r);
  g.addColorStop(0,'#b8f07e'); g.addColorStop(1,'#4c9e35');
  ctx.fillStyle=g; ctx.beginPath(); ctx.arc(x,y,r,0,7); ctx.fill();
  ctx.strokeStyle='#2f6b1f'; ctx.lineWidth=1.5; ctx.stroke();
  ctx.strokeStyle='rgba(18,54,12,0.75)'; ctx.lineWidth=Math.max(1,r*0.16);
  ctx.beginPath(); ctx.arc(x+Math.cos(a)*r*0.45,y+Math.sin(a)*r*0.45,r*0.3,a-0.8,a+0.8); ctx.stroke();
  const blinking = bl && (uiT%3.2)<0.16;
  for(const s of[-1,1]){
    const ex=x+Math.cos(a+s*0.8)*r*0.58, ey=y+Math.sin(a+s*0.8)*r*0.58;
    if(blinking){ ctx.strokeStyle='#2f6b1f'; ctx.lineWidth=Math.max(1,r*0.14);
      ctx.beginPath(); ctx.moveTo(ex-r*0.3,ey); ctx.lineTo(ex+r*0.3,ey); ctx.stroke(); }
    else { ctx.fillStyle='#fff'; ctx.beginPath(); ctx.arc(ex,ey,r*0.36,0,7); ctx.fill();
      ctx.fillStyle='#1c2a14'; ctx.beginPath();
      ctx.arc(ex+Math.cos(a)*r*0.14, ey+Math.sin(a)*r*0.14, r*0.19,0,7); ctx.fill(); }
  }
}
function drawFrogIcon(x,y,r,alive){
  ctx.save(); ctx.globalAlpha=alive?1:0.18;
  ctx.fillStyle='#6fbf4a'; ctx.beginPath(); ctx.arc(x,y+2,r,0,7); ctx.fill();
  ctx.fillStyle='#fff';
  ctx.beginPath(); ctx.arc(x-r*0.5,y-r*0.6,r*0.42,0,7); ctx.fill();
  ctx.beginPath(); ctx.arc(x+r*0.5,y-r*0.6,r*0.42,0,7); ctx.fill();
  ctx.fillStyle='#1c2a14';
  ctx.beginPath(); ctx.arc(x-r*0.5,y-r*0.6,r*0.2,0,7); ctx.fill();
  ctx.beginPath(); ctx.arc(x+r*0.5,y-r*0.6,r*0.2,0,7); ctx.fill();
  ctx.restore();
}
function drawCar(c){
  const w=c.w; ctx.save(); ctx.translate(c.x,c.y);
  ctx.fillStyle='rgba(0,0,0,0.3)'; ctx.beginPath(); ctx.ellipse(0,4,w/2+1,15,0,0,7); ctx.fill();
  rr(-w/2,-13,w,26,9); ctx.fillStyle=c.c; ctx.fill();
  ctx.strokeStyle='rgba(0,0,0,0.35)'; ctx.lineWidth=2; ctx.stroke();
  rr(-w/2+w*0.2,-9,w*0.6,18,6); ctx.fillStyle='rgba(15,20,28,0.55)'; ctx.fill();
  const fx=c.dir>0? w/2-4 : -w/2+4, tx=-fx;
  ctx.fillStyle='#fff8c4';
  ctx.beginPath(); ctx.arc(fx,-7,2.3,0,7); ctx.fill();
  ctx.beginPath(); ctx.arc(fx,7,2.3,0,7); ctx.fill();
  ctx.fillStyle='#ff3b30';
  ctx.beginPath(); ctx.arc(tx,-7,2.3,0,7); ctx.fill();
  ctx.beginPath(); ctx.arc(tx,7,2.3,0,7); ctx.fill();
  ctx.restore();
}
function drawTurtle(t){
  const s=t.vx>0?1:-1; ctx.save(); ctx.translate(t.x,t.y);
  ctx.fillStyle='rgba(0,0,0,0.25)'; ctx.beginPath(); ctx.ellipse(0,6,22,9,0,0,7); ctx.fill();
  ctx.fillStyle='#5a9e57';
  for(const p of[[s*14,8],[s*14,-8],[-s*10,9],[-s*10,-9]]){ ctx.beginPath(); ctx.arc(p[0],p[1],3,0,7); ctx.fill(); }
  ctx.beginPath(); ctx.arc(-s*20,0,2.5,0,7); ctx.fill();
  ctx.fillStyle='#3e8f4c'; ctx.strokeStyle='#275f31'; ctx.lineWidth=2;
  ctx.beginPath(); ctx.ellipse(0,0,19,11.5,0,0,7); ctx.fill(); ctx.stroke();
  ctx.strokeStyle='rgba(20,50,25,0.55)'; ctx.lineWidth=1.2;
  ctx.beginPath(); ctx.ellipse(0,0,12,7,0,0,7); ctx.stroke();
  ctx.beginPath(); ctx.moveTo(-14,0); ctx.lineTo(14,0); ctx.stroke();
  ctx.fillStyle='#6fbf6a'; ctx.beginPath(); ctx.arc(s*23,0,4.5,0,7); ctx.fill();
  ctx.fillStyle='#1c2a14'; ctx.beginPath(); ctx.arc(s*25,-1.5,1.2,0,7); ctx.fill();
  ctx.restore();
}

  rr(-p.hl,-p.r,p.hl*2,p.r*2,p.r); ctx.fillStyle='#7a5230'; ctx.fill();
  ctx.strokeStyle='#54371d'; ctx.lineWidth=2; ctx.stroke();
  for(const sx of[-1,1]){
    ctx.strokeStyle='rgba(60,38,18,0.7)'; ctx.lineWidth=1.5;

  for(const bx of[-30,-8,14,34]){ ctx.beginPath(); ctx.moveTo(bx,-p.r+3); ctx.lineTo(bx,p.r-3); ctx.stroke(); }
  ctx.restore();
}

/* ================= DRAW: TABLE ================= */
function drawTable(){
  // cabinet
  rr(-10,-10,TW+20,TH+20,28); ctx.fillStyle='#0d2417'; ctx.fill();
  ctx.strokeStyle='#1e5c33'; ctx.lineWidth=3; ctx.stroke();

  ctx.fillStyle='#0d2b31'; ctx.fillRect(12,112,456,308);

    const rg=ctx.createRadialGradient(bp.x,bp.y,4,bp.x,bp.y,bp.r*3.2);
    rg.addColorStop(0,'rgba(90,209,200,0.10)'); rg.addColorStop(1,'rgba(90,209,200,0)');
    ctx.fillStyle=rg; ctx.beginPath(); ctx.arc(bp.x,bp.y,bp.r*3.2,0,7); ctx.fill();
  }

  // river / pond
  g=ctx.createLinearGradient(0,420,0,522); g.addColorStop(0,'#16557f'); g.addColorStop(1,'#0c3557');
  ctx.fillStyle=g; ctx.fillRect(12,420,418,102);
  ctx.strokeStyle='rgba(190,230,255,0.08)'; ctx.lineWidth=1.5;
  for(let i=0;i<3;i++){ const y=436+i*32; ctx.beginPath();
    for(let x=16;x<=426;x+=12){ const yy=y+Math.sin(x*0.05+uiT*1.6+i*2)*2.5; x===16?ctx.moveTo(x,yy):ctx.lineTo(x,yy); }
    ctx.stroke(); }
  ctx.strokeStyle='rgba(8,40,66,0.9)'; ctx.lineWidth=2;
  ctx.beginPath(); ctx.moveTo(12,421); ctx.lineTo(430,421); ctx.stroke();
  ctx.beginPath(); ctx.moveTo(12,521); ctx.lineTo(430,521); ctx.stroke();
  // reeds
  for(const rx of[26,418]){
    ctx.strokeStyle='#1d5426'; ctx.lineWidth=2.5;
    for(let i=-1;i<=1;i++){ ctx.beginPath(); ctx.moveTo(rx+i*7,424);
      ctx.quadraticCurveTo(rx+i*7+4,404,rx+i*7+i*3,390); ctx.stroke(); }
    ctx.fillStyle='#6b4a2c'; ctx.beginPath(); ctx.ellipse(rx+3,388,3,7,0.2,0,7); ctx.fill();
  }

  // road
  ctx.fillStyle='#242730'; ctx.fillRect(12,522,418,156);
  for(const s of speckles){ ctx.fillStyle=s.lw?'rgba(255,255,255,0.05)':'rgba(0,0,0,0.25)';
    ctx.beginPath(); ctx.arc(s.x,s.y,1.4,0,7); ctx.fill(); }
  ctx.strokeStyle='rgba(255,255,255,0.16)'; ctx.lineWidth=3; ctx.setLineDash([16,18]);
  for(const y of[567,605]){ ctx.beginPath(); ctx.moveTo(16,y); ctx.lineTo(426,y); ctx.stroke(); }
  ctx.setLineDash([]);
  ctx.strokeStyle='rgba(217,169,60,0.5)'; ctx.lineWidth=2;
  ctx.beginPath(); ctx.moveTo(12,525); ctx.lineTo(430,525); ctx.stroke();
  ctx.beginPath(); ctx.moveTo(12,675); ctx.lineTo(430,675); ctx.stroke();

  // deck (flipper zone)
  g=ctx.createLinearGradient(0,678,0,748); g.addColorStop(0,'#3a2b1c'); g.addColorStop(1,'#2b2013');
  ctx.fillStyle=g; ctx.fillRect(12,678,418,70);
  ctx.strokeStyle='rgba(0,0,0,0.3)'; ctx.lineWidth=2;
  for(let y=692;y<748;y+=14){ ctx.beginPath(); ctx.moveTo(12,y); ctx.lineTo(430,y); ctx.stroke(); }
  // drain slot
  rr(215,742,50,7,3); ctx.fillStyle='#000'; ctx.fill();
  ctx.strokeStyle='rgba(255,180,84,'+(0.25+0.15*Math.sin(uiT*4))+')'; ctx.lineWidth=1.5; ctx.stroke();

  // shooter lane strip
  ctx.fillStyle='#0a1611'; ctx.fillRect(430,210,38,538);
  ctx.strokeStyle='rgba(141,255,87,'+(0.12+0.08*Math.sin(uiT*3))+')'; ctx.lineWidth=2;
  for(const ay of[600,540]){ ctx.beginPath(); ctx.moveTo(441,ay+8); ctx.lineTo(449,ay-4); ctx.lineTo(457,ay+8); ctx.stroke(); }

  // homes (frog holes)
  for(const h of homes){
    ctx.fillStyle='#3c4a44'; ctx.beginPath(); ctx.arc(h.x,h.y,17,0,7); ctx.fill();
    const hg=ctx.createRadialGradient(h.x-3,h.y-4,2,h.x,h.y,15);
    hg.addColorStop(0,'#2b1d12'); hg.addColorStop(1,'#0e0906');
    ctx.fillStyle=hg; ctx.beginPath(); ctx.arc(h.x,h.y,14,0,7); ctx.fill();
    ctx.strokeStyle='#3f8f4a'; ctx.lineWidth=3.5;
    ctx.beginPath(); ctx.arc(h.x,h.y,14.5,0.6,5.7); ctx.stroke();
    if(h.jackpot){ ctx.strokeStyle='#ffd93b'; ctx.lineWidth=2;
      ctx.beginPath(); ctx.arc(h.x,h.y,18,0,7); ctx.stroke();
      ctx.fillStyle='#ff8fc0'; for(let i=0;i<5;i++){ const a=i/5*6.283+uiT*0.4;
        ctx.beginPath(); ctx.arc(h.x+Math.cos(a)*5,h.y+Math.sin(a)*5,1.8,0,7); ctx.fill(); } }
    if(homesSaved.has(h.idx)) drawFrog(h.x,h.y,-Math.PI/2,6,false);
  }

  // slings (flower kickers)
  for(const sg of slings){
    const fl=sg.f>0?sg.f/0.15:0;
    ctx.beginPath(); ctx.moveTo(sg.pts[0][0],sg.pts[0][1]);
    ctx.lineTo(sg.pts[1][0],sg.pts[1][1]); ctx.lineTo(sg.pts[2][0],sg.pts[2][1]); ctx.closePath();
    const sg2=ctx.createLinearGradient(sg.pts[0][0],sg.pts[0][1],sg.pts[2][0],sg.pts[2][1]);
    sg2.addColorStop(0,'#ff5fa2'); sg2.addColorStop(1,'#a92e63');
    ctx.fillStyle=sg2; ctx.fill();

    if(fl>0){ ctx.fillStyle='rgba(255,255,255,'+fl*0.5+')'; ctx.fill(); }
  }

  // lily-pad pop bumpers
  for(const bp of bumpers){
    const fl=bp.f>0?bp.f/0.18:0, r=bp.r*(1+fl*0.14);
    const bg2=ctx.createRadialGradient(bp.x-r*0.3,bp.y-r*0.3,r*0.2,bp.x,bp.y,r);
    bg2.addColorStop(0,'#59c25f'); bg2.addColorStop(1,'#2e7d3a');
    ctx.fillStyle=bg2; ctx.beginPath(); ctx.arc(bp.x,bp.y,r,0,7); ctx.fill();
    ctx.strokeStyle='#1d5426'; ctx.lineWidth=3; ctx.stroke();

    for(const a of[1.2,2.9,4.6]){ ctx.beginPath(); ctx.moveTo(bp.x,bp.y);
      ctx.lineTo(bp.x+Math.cos(a)*(r-3),bp.y+Math.sin(a)*(r-3)); ctx.stroke(); }
    if(fl>0){ ctx.fillStyle='rgba(255,255,255,'+fl*0.6+')'; ctx.beginPath(); ctx.arc(bp.x,bp.y,r,0,7); ctx.fill(); }
  }

  // pads & cars
  for(const t of turtles) drawTurtle(t);
  for(const p of logs) drawLog(p);
  for(const c of cars) drawCar(c);

  // plunger
  const chg=charging?charge:0;
  rr(438,736,22,10,3); ctx.fillStyle='#5a2d2d'; ctx.fill();
  ctx.strokeStyle='#c94f4f'; ctx.lineWidth=3; ctx.beginPath();
  let py0=714+chg*10;
  for(let i=0;i<=6;i++){ const yy=py0+(736-py0)*i/6; ctx.lineTo(442+(i%2?8:0),yy); }
  ctx.stroke();
  rr(443,700,12,15,3); ctx.fillStyle='#d95c5c'; ctx.fill();
  if(chg>0){ ctx.strokeStyle='rgba(141,255,87,'+(0.4+chg*0.6)+')'; ctx.lineWidth=3;
    ctx.beginPath(); ctx.arc(449,662,15,-Math.PI/2,-Math.PI/2+chg*5.5); ctx.stroke(); }

  // walls (glow + core)
  ctx.lineCap='round'; ctx.lineJoin='round';
  const drawWalls=()=>{ ctx.beginPath(); for(const s of walls){ ctx.moveTo(s.x1,s.y1); ctx.lineTo(s.x2,s.y2); } ctx.stroke(); };
  ctx.strokeStyle='rgba(141,255,87,0.25)'; ctx.lineWidth=7; drawWalls();
  ctx.strokeStyle='#9dff6a'; ctx.lineWidth=3; drawWalls();

  // flippers
  for(const f of flips){
    const ex=f.px+Math.cos(f.ang)*f.len, ey=f.py+Math.sin(f.ang)*f.len;
    ctx.lineCap='round';
    ctx.strokeStyle='#7a3d14'; ctx.lineWidth=23;
    ctx.beginPath(); ctx.moveTo(f.px,f.py); ctx.lineTo(ex,ey); ctx.stroke();
    ctx.strokeStyle=f.pressed?'#ffe07a':'#ffcf4d'; ctx.lineWidth=18;
    ctx.beginPath(); ctx.moveTo(f.px,f.py); ctx.lineTo(ex,ey); ctx.stroke();
    ctx.fillStyle='#fff3c4'; ctx.beginPath(); ctx.arc(ex,ey,5.5,0,7); ctx.fill();
    ctx.fillStyle='#2b2b2b'; ctx.beginPath(); ctx.arc(f.px,f.py,7,0,7); ctx.fill();
    ctx.fillStyle='#666'; ctx.beginPath(); ctx.arc(f.px,f.py,3,0,7); ctx.fill();
  }

  // balls (frogs)
  for(const b of balls){
    const sp=Math.hypot(b.vx,b.vy); if(sp>40&&b.state==='live') b.ha=Math.atan2(b.vy,b.vx);
    drawFrog(b.x,b.y,b.ha,9,b.state==='wait');
    if(b.state==='wait' && state==='play'){
      ctx.strokeStyle='rgba(141,255,87,'+(0.3+0.25*Math.sin(uiT*5))+')';
      ctx.lineWidth=2; ctx.setLineDash([4,4]);
      ctx.beginPath(); ctx.arc(b.x,b.y,15+Math.sin(uiT*5)*2,0,7); ctx.stroke(); ctx.setLineDash([]);
      if(!charging){ ctx.font='bold 9px '+F_B; ctx.fillStyle='rgba(217,255,228,0.6)';
        ctx.textAlign='center'; ctx.fillText('HOLD SPACE',449,652); }
    }
  }

  // particles
  for(const p of particles){
    const a=clamp(p.life/p.max,0,1);
    if(p.ring){ ctx.strokeStyle=p.c; ctx.globalAlpha=a; ctx.lineWidth=2.5;
      ctx.beginPath(); ctx.arc(p.x,p.y,p.r,0,7); ctx.stroke(); }
    else { ctx.globalAlpha=a; ctx.fillStyle=p.c;
      ctx.beginPath(); ctx.arc(p.x,p.y,p.size*a+0.4,0,7); ctx.fill(); }
  }
  ctx.globalAlpha=1;

  // floating texts
  for(const t of texts){
    const a=clamp(t.life/t.max,0,1);
    ctx.globalAlpha=a; ctx.font='bold '+t.size+'px '+F_M; ctx.textAlign='center'; ctx.textBaseline='middle';
    ctx.lineWidth=4; ctx.strokeStyle='#04120c'; ctx.strokeText(t.t,t.x,t.y);
    ctx.fillStyle=t.c; ctx.fillText(t.t,t.x,t.y);
  }
  ctx.globalAlpha=1;

  // vignette
  const vg=ctx.createRadialGradient(240,380,300,240,380,640);
  vg.addColorStop(0,'rgba(0,0,0,0)'); vg.addColorStop(1,'rgba(0,0,0,0.35)');
  ctx.fillStyle=vg; ctx.fillRect(-10,-10,TW+20,TH+20);

  // banners
  banners.forEach((bn,i)=>{
    const p=bn.p/bn.dur, sc=p<0.15?eob(p/0.15):1, al=p>0.7?1-(p-0.7)/0.3:1;
    ctx.save(); ctx.translate(221,236+i*64); ctx.scale(sc,sc); ctx.globalAlpha=clamp(al,0,1);
    ctx.textAlign='center'; ctx.textBaseline='middle';
    ctx.font="900 40px "+F_D; ctx.lineWidth=8; ctx.strokeStyle='#04120c';
    ctx.strokeText(bn.t,0,0); ctx.fillStyle=bn.c; ctx.fillText(bn.t,0,0);
    if(bn.sub){ ctx.font='bold 15px '+F_B; ctx.lineWidth=5; ctx.strokeText(bn.sub,0,28);
      ctx.fillStyle='#eafff0'; ctx.fillText(bn.sub,0,28); }
    ctx.restore();
  });
}

function frame(now){
  requestAnimationFrame(frame);
  let dt=(now-lastT)/1000; lastT=now; if(dt>0.05)dt=0.05; uiT+=dt;
  if(shT>0){ shT-=dt; if(shT<=0)shMag=0; }
  const Lp=keys.ArrowLeft||keys.KeyA||pf==='L', Rp=keys.ArrowRight||keys.KeyD||pf==='R';
  for(const f of flips){
    f.pressed=(f===flips[0]?Lp:Rp);
    if(f.pressed&&!f.wasPressed) sfx.flip();
    f.wasPressed=f.pressed;
  }
  for(const bp of bumpers) if(bp.f>0)bp.f-=dt;
  for(const sg of slings) if(sg.f>0)sg.f-=dt;

  if(state==='play'){
    gameT+=dt;
    if(charging) charge=Math.min(1,charge+dt/0.9);
    updHazards(dt);
    acc+=dt; const hstep=1/240; let n=0;
    while(acc>=hstep&&n<10){ step(hstep); acc-=hstep; n++; }
    if(n===10)acc=0;
    if(respawnT>0){ respawnT-=dt;
      if(respawnT<=0 && !balls.some(b=>b.state==='wait')) balls.push(newBall()); }
    if(overT>0){ overT-=dt;
      if(overT<=0){ state='over'; newRecord=score>startHi&&score>0; saveHi(); } }
    nudgeCd=Math.max(0,nudgeCd-dt);
  } else if(state==='start'||state==='over'){ updHazards(dt); }

  for(const p of particles){ p.life-=dt;
    if(p.ring) p.r+=p.vr*dt;
    else { p.vy+=(p.grav||0)*dt; p.x+=p.vx*dt; p.y+=p.vy*dt; } }
  particles=particles.filter(p=>p.life>0);
  for(const t of texts){ t.life-=dt; t.y-=28*dt; }
  texts=texts.filter(t=>t.life>0);
  for(const b of banners) b.p+=dt;
  banners=banners.filter(b=>b.p<b.dur);

  draw();
}
