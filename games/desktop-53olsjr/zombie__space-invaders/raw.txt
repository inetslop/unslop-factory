let score=0, hi=loadHi(), lives=3, wave=1, spreadLvl=1, shield=false, slowT=0, invuln=0;
let shake=0, muzzleT=0, fogAmt=0, banner=null, breakT=0, spitT=4, hbT=1, moanT=2, killsSinceDrop=0, danger=0;
let gridX=250, gridDir=1, gridY=96;
let zombies=[],bullets=[],globs=[],drops=[],parts=[],pops=[],bombs=[],bunkers=[];
const keys={}; const touch={x:null,fire:false};

function loadHi(){try{return parseInt(localStorage.getItem('deadwave_hi')||'0',10)||0;}catch(e){return 0;}}
function saveHi(v){try{localStorage.setItem('deadwave_hi',String(v));}catch(e){}}
const FLAVOR=['they multiply','the fog thickens','bones in the wind','no one is coming','moonrise again','they smell the living','hold… hold'];

function rowTypes(w){
  const T=[
    ['runner','shambler','shambler','shambler','shambler'],
    ['runner','spitter','shambler','shambler','shambler'],
    ['brute','runner','spitter','shambler','shambler'],
    ['brute','spitter','runner','shambler','spitter'],
    ['brute','brute','spitter','runner','spitter']
  ]; return T[Math.min(w,5)-1];
}
function makeGrid(w){
  const types=rowTypes(w),arr=[];
  for(let r=0;r<5;r++)for(let c=0;c<8;c++){const ty=types[r],T=ZTYPES[ty];
    arr.push({type:ty,c,r,hp:T.hp,ph:Math.random()*6.28,pop:0,alive:true});}
  return arr;
}
function zpos(z){ const T=ZTYPES[z.type];
  return {x:gridX+z.c*64+(z.type==='runner'?Math.sin(t*9+z.ph)*3:0), y:gridY+z.r*52+Math.sin(t*3+z.ph)*3}; }

/* ============ BUNKERS (tombstone shields) ============ */
const BMASK=['.XXX.','XXXXX','XXXXX','XXXXX','XXXXX','XXXXX'], CS=12;
function buildBunkers(){ bunkers=[]; const xs=[180,395,610,825];
  for(const cx of xs){const b={x:cx-30,y:VH-252,cells:{},shade:{}};
    for(let r=0;r<6;r++)for(let c=0;c<5;c++)if(BMASK[r][c]==='X'){b.cells[r*5+c]=3;b.shade[r*5+c]=(Math.random()*4)|0;}
    bunkers.push(b);} }
const STONE=['#98a1a6','#8b949a','#a5adb0','#7f888e'];
function bunkerHit(x,y,acid){
  for(const b of bunkers){ const c=Math.floor((x-b.x)/CS), r=Math.floor((y-b.y)/CS);
    if(c<0||c>=5||r<0||r>=6)continue; const k=r*5+c; if(b.cells[k]===undefined)continue;
    b.cells[k]--;
    const px2=b.x+c*CS+6, py2=b.y+r*CS+6;
    for(let i=0;i<4;i++)parts.push({x:px2,y:py2,vx:rand(-70,70),vy:rand(-90,10),g:260,life:.5,max:.5,s:rand(2,4),col:acid?'#8b46ff':STONE[(Math.random()*4)|0]});
    noise(.05, acid?2600:1400, .07, 0, 'highpass');
    return true; }
  return false;
}

/* ============ FX helpers ============ */
function burst(x,y,col,n,spd){ for(let i=0;i<n;i++){const a=rand(0,6.28),v=rand(30,spd);
  parts.push({x,y,vx:Math.cos(a)*v,vy:Math.sin(a)*v-40,g:300,life:rand(.35,.7),max:.7,s:rand(2,5),col});} }
function ring(x,y,col,r){ parts.push({x,y,ring:1,r,life:.45,max:.45,col}); }
function addPop(x,y,txt,col){ pops.push({x,y,t:0,life:.9,txt,col}); }

  zombies=makeGrid(1); gridX=250; gridDir=1; gridY=96; px=VW/2;
  bullets=[];globs=[];drops=[];bombs=[];parts=[];pops=[]; buildBunkers();
  banner={text:'WAVE 1',sub:'hold the line, soldier',t:0,dur:2.4}; sfx.wave();
}
function nextWave(){
  wave++; zombies=makeGrid(wave); gridX=250; gridDir=Math.random()<.5?1:-1; gridY=96;
  globs=[]; drops=[]; spitT=rand(3,5); killsSinceDrop=0;
  banner={text:'WAVE '+wave,sub:FLAVOR[(wave-2)%FLAVOR.length],t:0,dur:2.4}; sfx.wave();
}
function startBreak(){ breakT=2.4; score+=100*wave; addPop(VW/2,VH*.3,'BONUS +'+(100*wave),'#ffd76a');
  banner={text:'WAVE CLEARED',sub:'the fog exhales…',t:0,dur:2.4}; sfx.wave(); }

  $('ovWave').textContent=wave;

/* ============ COMBAT ============ */
let px=VW/2, fireCd=0;
function shoot(){
  sfx.shoot(); muzzleT=.06;
  const offs=[[0],[-7,7],[-8,0,8]][spreadLvl-1];
  for(const o of offs)bullets.push({x:px+o,y:PY-32});
  fireCd=spreadLvl===3?.19:.16;
}
function hitZombie(z,b){
  z.hp--; z.pop=1; burst(b.x,b.y,ZTYPES[z.type].col,4,120);
  if(z.hp<=0)killZombie(z,false); else sfx.zhit();
}
function killZombie(z,quiet){
  if(!z.alive)return; z.alive=false;
  const T=ZTYPES[z.type], p=zpos(z);
  score+=T.score; addPop(p.x,p.y,'+'+T.score,T.col);
  burst(p.x,p.y,T.col,10,220); burst(p.x,p.y,'#1a3a22',5,120); ring(p.x,p.y,T.col,34);
  shake=Math.max(shake,quiet?3:5);
  killsSinceDrop++;
  if(killsSinceDrop>=9||Math.random()<.13){ spawnDrop(p.x,p.y); killsSinceDrop=0; }
  quiet?sfx.pop():sfx.zdie();
}
function spawnDrop(x,y){
  const r=Math.random(); const type=r<.35?'double':r<.55?'shield':r<.78?'fog':'bomb';
  drops.push({x:clamp(x,60,VW-60),y,type,t:0,rest:false,fade:1.4});
}
function applyDrop(type){
  sfx.pickup();
  if(type==='double'){ if(spreadLvl<3){spreadLvl++; addPop(px,PY-44,'SPREAD ×'+spreadLvl,'#6fe08c');}
    else {score+=250; addPop(px,PY-44,'OVERCLOCK +250','#ffd76a');} }
  else if(type==='shield'){ shield=true; sfx.shieldUp(); addPop(px,PY-44,'AEGIS UP','#7fd7ff'); }
  else if(type==='fog'){ slowT=8; addPop(px,PY-44,'FOG RISES','#b9a7ff'); }
  else { shake=14; sfx.bomb(); ring(px,PY,'#ffb347',260); addPop(px,PY-44,'MIDNIGHT BOMB','#ffb347');
    let i=0; for(const z of zombies)if(z.alive)bombs.push({z,d:(i++%5)*.06+rand(0,.18)}); }
}
function damagePlayer(){ // returns true if glob consumed
  if(invuln>0&&!shield)return false;
  if(shield){ shield=false; sfx.shieldBreak(); addPop(px,PY-44,'SHIELD DOWN','#7fd7ff'); shake=Math.max(shake,6); invuln=1; return true; }
  if(invuln>0)return false;
  lives--; sfx.boom(); shake=13; burst(px,PY,'#ffb347',22,320); burst(px,PY,'#59d96e',14,240); ring(px,PY,'#ffb347',80);
  if(lives<=0)gameOver('dead'); else invuln=2.2;
  return true;
}

/* ============ UPDATE ============ */
function update(dt){
  if(shake>0)shake=Math.max(0,shake-dt*26);
  muzzleT-=dt; fireCd-=dt;
  fogAmt+=((slowT>0?1:0)-fogAmt)*Math.min(1,dt*3);
  if(banner){banner.t+=dt; if(banner.t>banner.dur)banner=null;}
  if(state==='menu'){ moanT-=dt; if(moanT<=0){moanT=rand(3,7); sfx.moan();} }

  if(state==='play'){
    slowT=Math.max(0,slowT-dt); invuln=Math.max(0,invuln-dt);
    // player move
    const sp=430; let mv=0;
    if(keys.ArrowLeft||keys.KeyA)mv--; if(keys.ArrowRight||keys.KeyD)mv++;
    if(touch.x!==null){ px+=clamp(touch.x-px,-sp*1.6*dt,sp*1.6*dt); } else px+=mv*sp*dt;
    px=clamp(px,60,VW-60);
    if((keys.Space||touch.fire)&&fireCd<=0)shoot();

    // bullets
    for(let i=bullets.length-1;i>=0;i--){ const b=bullets[i]; b.y-=640*dt; let dead=b.y<26;
      if(!dead&&bunkerHit(b.x,b.y-8,false))dead=true;
      if(!dead){ for(const z of zombies){ if(!z.alive)continue; const T=ZTYPES[z.type], p=zpos(z);
        if(Math.abs(b.x-p.x)<T.hw+3 && b.y>p.y-T.hh-8 && b.y<p.y+T.hh){ hitZombie(z,b); dead=true; break; } } }
      if(dead)bullets.splice(i,1); }

    // grid
    if(zombies.length){
      let n=0; for(const z of zombies)if(z.alive)n++;
      const spd=(26+wave*7)*(1+(1-n/zombies.length)*2.4)*(slowT>0?.35:1);
      gridX+=gridDir*spd*dt;
      if(gridX<96){gridX=96;gridDir=1;gridY+=30;} else if(gridX>480){gridX=480;gridDir=-1;gridY+=30;}
      const by=gridY+4*52+16;
      danger=clamp((by-380)/(KILLY-380),0,1);
      if(by>KILLY){ gameOver('breach'); }
      // spitters
      const spis=zombies.filter(z=>z.alive&&z.type==='spitter');
      if(spis.length){ spitT-=dt; if(spitT<=0){ const s=spis[(Math.random()*spis.length)|0], p=zpos(s);
        globs.push({x:p.x,y:p.y+14,vy:150+wave*9,ph:rand(0,6)}); sfx.glob();
        spitT=rand(Math.max(1.4,3.2-wave*.18),Math.max(2.2,5-wave*.2)); } }
      if(danger>0){ hbT-=dt*(.4+danger*.9); if(hbT<=0){hbT=rand(1.1,1.5); sfx.thump();} }
    } else if(!breakT&&bombs.length===0){ danger=0; startBreak(); }
    if(breakT>0){ breakT-=dt; if(breakT<=0)nextWave(); }

    // bombs
    for(let i=bombs.length-1;i>=0;i--){ const bk=bombs[i]; bk.d-=dt;
      if(bk.d<=0){ if(bk.z.alive)killZombie(bk.z,true); bombs.splice(i,1);} }

    // globs
    for(let i=globs.length-1;i>=0;i--){ const g=globs[i]; g.y+=g.vy*dt; g.x+=Math.sin(t*3+g.ph)*26*dt; let dead=false;
      if(bunkerHit(g.x,g.y,true))dead=true;
      else if(Math.abs(g.x-px)<24&&Math.abs(g.y-PY)<22){ if(damagePlayer())dead=true; }
      if(!dead&&g.y>GROUND-4){ burst(g.x,GROUND-6,'#8b46ff',5,90); dead=true; }
      if(dead)globs.splice(i,1); }

    // drops
    for(let i=drops.length-1;i>=0;i--){ const d=drops[i]; d.t+=dt;
      if(!d.rest){ d.y+=95*dt; if(d.y>=GROUND-12){d.y=GROUND-12;d.rest=true;} }
      else d.fade-=dt;
      if(Math.abs(d.x-px)<32&&d.y>PY-48&&d.y<PY+18){ applyDrop(d.type); drops.splice(i,1); continue; }
      if(d.fade<=0)drops.splice(i,1); }

    if(score>hi)hi=score;
  } else {
    // ambient sway behind menu / game-over
    const spd=state==='menu'?26:10;
    gridX+=gridDir*spd*dt;
    if(gridX<96){gridX=96;gridDir=1;} if(gridX>480){gridX=480;gridDir=-1;}
  }

  // particles & popups (all states)
  for(let i=parts.length-1;i>=0;i--){ const p=parts[i]; p.life-=dt;
    if(p.ring){ if(p.life<=0)parts.splice(i,1); continue; }
    p.x+=p.vx*dt; p.y+=p.vy*dt; p.vy+=p.g*dt;
    if(p.life<=0||p.y>GROUND+6)parts.splice(i,1); }
  if(parts.length>420)parts.splice(0,parts.length-420);
  for(let i=pops.length-1;i>=0;i--){ const p=pops[i]; p.t+=dt; if(p.t>p.life)pops.splice(i,1); }
}

/* ============ SCENE EXTRAS ============ */
let stars=[],deco=[],fogList=[],vig=null,redVig=null;
function buildScene(){
  stars=[]; const n=Math.min(220,(cw*ch/9000)|0);
  for(let i=0;i<n;i++)stars.push({x:Math.random()*cw,y:Math.random()*ch*.75,r:rand(.6,1.8),ph:rand(0,6),spd:rand(.4,1.6),a:Math.random()<.14});
  deco=[]; for(let i=0;i<46;i++)deco.push({x:Math.random()*cw,h:rand(4,11),lean:rand(-3,3),rock:Math.random()<.15});
  fogList=[]; for(let i=0;i<9;i++)fogList.push({nx:Math.random(),ny:rand(.35,.92),s:rand(2.2,4.6),a:rand(.04,.09),spd:rand(6,20),far:i<3});
  vig=document.createElement('canvas'); vig.width=cw; vig.height=ch;
  let g=vig.getContext('2d').createRadialGradient(cw/2,ch/2,Math.min(cw,ch)*.42,cw/2,ch/2,Math.max(cw,ch)*.75);
  g.addColorStop(0,'rgba(0,0,0,0)'); g.addColorStop(1,'rgba(1,3,2,.55)');
  vig.getContext('2d').fillStyle=g; vig.getContext('2d').fillRect(0,0,cw,ch);
  redVig=document.createElement('canvas'); redVig.width=cw; redVig.height=ch;
  g=redVig.getContext('2d').createRadialGradient(cw/2,ch/2,Math.min(cw,ch)*.3,cw/2,ch/2,Math.max(cw,ch)*.7);
  g.addColorStop(0,'rgba(255,40,30,0)'); g.addColorStop(1,'rgba(255,40,30,.8)');
  redVig.getContext('2d').fillStyle=g; redVig.getContext('2d').fillRect(0,0,cw,ch);
}
function resize(){
  dpr=Math.min(2,window.devicePixelRatio||1); cw=window.innerWidth; ch=window.innerHeight;
  canvas.width=cw*dpr; canvas.height=ch*dpr;
  scl=Math.min(cw/VW,ch/VH); ox=(cw-VW*scl)/2; oy=(ch-VH*scl)/2;
  buildScene();
}
window.addEventListener('resize',resize);

/* ============ RENDER ============ */
function drawTree(x,y,s){
  ctx.strokeStyle='#0a130b'; ctx.lineCap='round';
  ctx.lineWidth=7*s; ctx.beginPath(); ctx.moveTo(x,y); ctx.quadraticCurveTo(x-3*s,y-45*s,x-6*s,y-80*s); ctx.stroke();
  ctx.lineWidth=4*s;
  ctx.beginPath(); ctx.moveTo(x-2*s,y-40*s); ctx.lineTo(x-30*s,y-70*s); ctx.stroke();
  ctx.beginPath(); ctx.moveTo(x-4*s,y-58*s); ctx.lineTo(x+22*s,y-92*s); ctx.stroke();
  ctx.lineWidth=2.4*s;
  ctx.beginPath(); ctx.moveTo(x-16*s,y-58*s); ctx.lineTo(x-34*s,y-62*s); ctx.stroke();
  ctx.beginPath(); ctx.moveTo(x+10*s,y-76*s); ctx.lineTo(x+26*s,y-70*s); ctx.stroke();
}
function drawPlayer(){
  if(state==='menu')return; if(state==='over'&&lives<=0)return;
  const blink=invuln>0&&(Math.floor(t*14)%2===0);
  ctx.save(); ctx.translate(px,PY); if(blink)ctx.globalAlpha=.35;
  if(shield){ const r=36+Math.sin(t*5)*2.5;
    ctx.strokeStyle='rgba(127,215,255,.8)'; ctx.lineWidth=2;
    ctx.beginPath(); ctx.arc(0,-6,r,0,7); ctx.stroke();
    ctx.fillStyle='rgba(127,215,255,.07)'; ctx.fill(); }
  ctx.fillStyle='#2c343d'; ctx.fillRect(-22,2,44,12);
  ctx.fillStyle='#39434e'; ctx.fillRect(-16,-6,32,10);
  ctx.fillStyle='#4a5763'; ctx.beginPath(); ctx.arc(0,-8,12,Math.PI,0); ctx.fill();
  const rec=muzzleT>0?3:0;
  ctx.fillStyle='#59666f'; ctx.fillRect(-4,-30+rec,8,22);
  ctx.fillStyle='#313b44'; ctx.fillRect(-4,-30+rec,8,4);
  ctx.fillStyle=(Math.floor(t*2)%2)?'#ffb347':'#7a4d16';
  ctx.beginPath(); ctx.arc(0,-21,2.5,0,7); ctx.fill();
  if(muzzleT>0){ ctx.fillStyle='rgba(190,255,170,.85)';
    ctx.beginPath(); ctx.arc(0,-34,5,0,7); ctx.fill();
    ctx.fillRect(-1.5,-42,3,10); ctx.fillRect(-6,-35.5,12,3); }
  ctx.restore();
}
function chipDraw(txt,col,x){
  ctx.font='700 11px "Courier New",monospace';
  const w=ctx.measureText(txt).width+14;
  ctx.fillStyle='rgba(8,12,9,.75)'; ctx.fillRect(x-w,VH-86,w,18);
  ctx.strokeStyle=col; ctx.lineWidth=1; ctx.strokeRect(x-w+.5,VH-86.5,w-1,17);
  ctx.fillStyle=col; ctx.textAlign='left'; ctx.fillText(txt,x-w+7,VH-73);
}
function render(){
  ctx.setTransform(dpr,0,0,dpr,0,0);
  // sky
  let g=ctx.createLinearGradient(0,0,0,ch);
  g.addColorStop(0,'#04090f'); g.addColorStop(.55,'#081511'); g.addColorStop(.85,'#0c2016'); g.addColorStop(1,'#0e241a');
  ctx.fillStyle=g; ctx.fillRect(0,0,cw,ch);
  // stars
  for(const s of stars){ const a=.22+.45*(.5+.5*Math.sin(t*s.spd+s.ph));
    ctx.globalAlpha=a; ctx.fillStyle=s.a?'#ffe9b8':'#cfe8d2'; ctx.fillRect(s.x,s.y,s.r,s.r); }
  ctx.globalAlpha=1;
  // moon
  const mx=cw*.82,my=ch*.17,mr=Math.min(cw,ch)*.09;
  g=ctx.createRadialGradient(mx,my,mr*.4,mx,my,mr*3);
  g.addColorStop(0,'rgba(225,240,215,.14)'); g.addColorStop(1,'rgba(225,240,215,0)');
  ctx.fillStyle=g; ctx.fillRect(mx-mr*3,my-mr*3,mr*6,mr*6);
  ctx.fillStyle='#e6eeda'; ctx.beginPath(); ctx.arc(mx,my,mr,0,7); ctx.fill();
  ctx.fillStyle='rgba(140,165,135,.35)';
  ctx.beginPath(); ctx.arc(mx-mr*.28,my-mr*.12,mr*.22,0,7); ctx.fill();
  ctx.beginPath(); ctx.arc(mx+mr*.3,my+mr*.3,mr*.14,0,7); ctx.fill();
  ctx.beginPath(); ctx.arc(mx+mr*.1,my-mr*.42,mr*.1,0,7); ctx.fill();
  // far fog
  for(const f of fogList)if(f.far){ const x=((f.nx*cw+t*f.spd)%(cw+560))-280;
    ctx.globalAlpha=f.a*.7; ctx.drawImage(fogSpr,x-160*f.s,ch*f.ny-80*f.s,320*f.s,160*f.s); }
  ctx.globalAlpha=1;
  // ground (full bleed)
  const gy=oy+GROUND*scl;
  g=ctx.createLinearGradient(0,gy,0,ch);
  g.addColorStop(0,'#131f12'); g.addColorStop(.25,'#0d160c'); g.addColorStop(1,'#070b06');
  ctx.fillStyle=g; ctx.fillRect(0,gy,cw,ch-gy);
  ctx.fillStyle='rgba(120,190,120,.13)'; ctx.fillRect(0,gy,cw,2);
  for(const d of deco){ const y=gy+rand? gy : gy; // deterministic per frame is fine
    if(d.rock){ ctx.fillStyle='#101a10'; ctx.beginPath(); ctx.ellipse(d.x,gy+6,7,4,0,0,7); ctx.fill(); }
    else { ctx.strokeStyle='#1d2b18'; ctx.lineWidth=1.5;
      for(let k=-1;k<=1;k++){ ctx.beginPath(); ctx.moveTo(d.x+k*3,gy+4); ctx.lineTo(d.x+k*3+d.lean,gy+4-d.h); ctx.stroke(); } } }
  drawTree(cw*.06,gy+8,Math.min(1.4,cw/900)); drawTree(cw*.94,gy+8,Math.min(1.4,cw/900));

  // ---- world pass ----
  ctx.save();
  ctx.translate(ox+(Math.random()-.5)*shake*scl, oy+(Math.random()-.5)*shake*scl);
  ctx.scale(scl,scl);
  // field boundary floodlights
  for(const bx of [34,VW-34]){
    const dir=bx<VW/2?1:-1;
    ctx.fillStyle='rgba(120,255,150,.05)'; ctx.fillRect(bx-1,60,2,VH-120);
    ctx.strokeStyle='#22301f'; ctx.lineWidth=4;
    ctx.beginPath(); ctx.moveTo(bx,GROUND); ctx.lineTo(bx,GROUND-150); ctx.stroke();
    const lg=ctx.createLinearGradient(bx,GROUND-150,bx+dir*260,GROUND-40);
    lg.addColorStop(0,'rgba(140,255,160,.10)'); lg.addColorStop(1,'rgba(140,255,160,0)');
    ctx.fillStyle=lg; ctx.beginPath(); ctx.moveTo(bx,GROUND-150);
    ctx.lineTo(bx+dir*300,GROUND-30); ctx.lineTo(bx+dir*40,GROUND-150); ctx.closePath(); ctx.fill();
    ctx.fillStyle='#8dffa0'; ctx.beginPath(); ctx.arc(bx,GROUND-152,4,0,7); ctx.fill();
  }
  // kill line
  ctx.setLineDash([8,10]); ctx.strokeStyle='rgba(255,70,60,'+(.16+danger*.35)+')'; ctx.lineWidth=1.5;
  ctx.beginPath(); ctx.moveTo(60,KILLY); ctx.lineTo(VW-60,KILLY); ctx.stroke(); ctx.setLineDash([]);
  // bunkers
  for(const b of bunkers)for(let r=0;r<6;r++)for(let c=0;c<5;c++){ const k=r*5+c,hp=b.cells[k];
    if(hp===undefined||hp<=0)continue;
    const x=b.x+c*CS,y=b.y+r*CS;
    ctx.fillStyle=STONE[b.shade[k]]; ctx.fillRect(x+1,y+1,CS-2,CS-2);
    ctx.fillStyle='rgba(255,255,255,.14)'; ctx.fillRect(x+1,y+1,CS-2,2);
    if(hp<3){ ctx.strokeStyle='rgba(20,26,22,.7)'; ctx.lineWidth=1; ctx.beginPath();
      ctx.moveTo(x+3,y+9); ctx.lineTo(x+7,y+5); ctx.lineTo(x+10,y+8); ctx.stroke(); }
    if(hp<2){ ctx.strokeStyle='rgba(15,20,16,.8)'; ctx.beginPath();
      ctx.moveTo(x+9,y+3); ctx.lineTo(x+5,y+7); ctx.lineTo(x+8,y+11); ctx.stroke(); } }
  // drops
  const DCOL={double:'#6fe08c',shield:'#7fd7ff',fog:'#b9a7ff',bomb:'#ffb347'};
  for(const d of drops){ const al=d.rest?clamp(d.fade/1.4,0,1):1;
    ctx.save(); ctx.globalAlpha=al; ctx.translate(d.x,d.y+Math.sin(t*3+d.t)*2);
    ctx.shadowColor=DCOL[d.type]; ctx.shadowBlur=12;
    ctx.fillStyle='#101812'; ctx.fillRect(-9,-11,18,22);
    ctx.strokeStyle=DCOL[d.type]; ctx.lineWidth=2; ctx.strokeRect(-9,-11,18,22);
    ctx.shadowBlur=0; ctx.fillStyle=DCOL[d.type];
    ctx.font='700 13px "Courier New",monospace'; ctx.textAlign='center';
    ctx.fillText(d.type[0].toUpperCase(),0,4.5); ctx.restore(); }
  // zombies
  for(const z of zombies){ if(!z.alive)continue; const T=ZTYPES[z.type], p=zpos(z);
    z.pop=Math.max(0,z.pop-.04);
    ctx.save(); ctx.translate(p.x,p.y);
    const s=1+z.pop*.28; ctx.scale(s,s);
    if(z.type==='runner')ctx.rotate(Math.sin(t*8+z.ph)*.09);
    if(z.type==='brute')ctx.rotate(Math.sin(t*2+z.ph)*.03);
    const spr=T.spr[Math.floor(t*6+z.ph)%2];
    ctx.drawImage(spr,-T.w/2,-T.h/2); ctx.restore(); }
  // globs
  for(const gl of globs){ ctx.save(); ctx.translate(gl.x,gl.y);
    ctx.fillStyle='rgba(139,70,255,.9)'; ctx.beginPath(); ctx.arc(0,0,6,0,7); ctx.fill();
    ctx.fillStyle='rgba(210,160,255,.8)'; ctx.beginPath(); ctx.arc(-1.5,-1.5,2,0,7); ctx.fill();
    ctx.strokeStyle='rgba(139,70,255,.5)'; ctx.lineWidth=2;
    ctx.beginPath(); ctx.moveTo(0,4); ctx.lineTo(Math.sin(t*9+gl.ph)*3,12); ctx.stroke(); ctx.restore(); }
  drawPlayer();
  // bullets
  for(const b of bullets){
    ctx.strokeStyle='rgba(150,255,170,.35)'; ctx.lineWidth=6;
    ctx.beginPath(); ctx.moveTo(b.x,b.y+12); ctx.lineTo(b.x,b.y-4); ctx.stroke();
    ctx.strokeStyle='#eaffdc'; ctx.lineWidth=2.5;
    ctx.beginPath(); ctx.moveTo(b.x,b.y+10); ctx.lineTo(b.x,b.y-6); ctx.stroke(); }
  // particles
  for(const p of parts){ const a=p.life/p.max;
    if(p.ring){ ctx.globalAlpha=a*.8; ctx.strokeStyle=p.col; ctx.lineWidth=2.5;
      ctx.beginPath(); ctx.arc(p.x,p.y,(1-a)*p.r+4,0,7); ctx.stroke(); }
    else { ctx.globalAlpha=a; ctx.fillStyle=p.col; ctx.fillRect(p.x-p.s/2,p.y-p.s/2,p.s,p.s); } }
  ctx.globalAlpha=1;
  // popups
  for(const p of pops){ const a=1-p.t/p.life;
    ctx.globalAlpha=a; ctx.font='700 13px "Courier New",monospace'; ctx.textAlign='center';
    ctx.fillStyle=p.col; ctx.fillText(p.txt,p.x,p.y-p.t*34); }
  ctx.globalAlpha=1;
  // banner
  if(banner){ const b=banner, a=clamp(Math.min(b.t/.25,(b.dur-b.t)/.4),0,1);
    ctx.save(); ctx.globalAlpha=a; ctx.textAlign='center';
    try{ctx.letterSpacing='4px';}catch(e){}
    ctx.font='900 62px Impact,"Arial Black",sans-serif';
    ctx.shadowColor='rgba(110,255,140,.7)'; ctx.shadowBlur=24; ctx.fillStyle='#d8ffd2';
    ctx.fillText(b.text,VW/2,VH*.36-(1-Math.min(1,b.t/.5))*26);
    ctx.shadowBlur=0; try{ctx.letterSpacing='8px';}catch(e){}
    ctx.font='700 14px "Courier New",monospace'; ctx.fillStyle='rgba(190,230,190,.85)';
    ctx.fillText(b.sub.toUpperCase(),VW/2,VH*.36+30);
    try{ctx.letterSpacing='0px';}catch(e){}
    ctx.restore(); }
  // HUD
  if(state!=='menu'){
    ctx.textAlign='left';
    ctx.font='700 11px "Courier New",monospace'; ctx.fillStyle='#6fae79';
    ctx.fillText('SCORE',24,28);
    ctx.font='400 27px Impact,"Arial Black",sans-serif'; ctx.fillStyle='#e8f0d8';
    ctx.fillText(String(score).padStart(6,'0'),24,54);
    ctx.textAlign='right'; ctx.font='700 11px "Courier New",monospace'; ctx.fillStyle='#b98a3e';
    ctx.fillText('BEST',VW-24,28);
    ctx.font='400 27px Impact,"Arial Black",sans-serif'; ctx.fillStyle='#ffd76a';
    ctx.fillText(String(hi).padStart(6,'0'),VW-24,54);
    ctx.textAlign='center'; ctx.fillStyle='#8fb591';
    ctx.font='400 20px Impact,"Arial Black",sans-serif';
    ctx.fillText('WAVE '+wave,VW/2,42);
    // lives
    for(let i=0;i<Math.max(0,lives);i++){ const x=30+i*26,y=VH-78;
      ctx.fillStyle='#39434e'; ctx.fillRect(x-10,y+2,20,7);
      ctx.fillStyle='#4a5763'; ctx.beginPath(); ctx.arc(x,y+2,7,Math.PI,0); ctx.fill();
      ctx.fillStyle='#59666f'; ctx.fillRect(x-2,y-11,4,9); }
    // buffs
    let bx=VW-24;
    if(slowT>0){ const s='FOG '+Math.ceil(slowT); chipDraw(s,'#b9a7ff',bx); bx-=ctx.measureText?s.length*8+30:90; }
    if(shield){ chipDraw('AEGIS','#7fd7ff',bx-((slowT>0)?s2w(slowT):0)); }
    function s2w(v){return ('FOG '+Math.ceil(v)).length*8+30;}
    if(spreadLvl>1)chipDraw('SPREAD ×'+spreadLvl,'#6fe08c',bx-((shield)?78:0)-((slowT>0&&shield)?s2w(slowT):0));
    ctx.textAlign='center'; ctx.font='700 10px "Courier New",monospace';
    ctx.fillStyle='rgba(150,190,155,.4)';
    ctx.fillText('P·PAUSE   R·RESTART   M·SOUND',VW/2,VH-70);
  }
  ctx.restore();
  // ---- near fog / vignette / danger ----
  for(const f of fogList)if(!f.far){ const x=((f.nx*cw+t*f.spd)%(cw+560))-280;
    ctx.globalAlpha=f.a*(1+fogAmt*1.6); ctx.drawImage(fogSpr,x-160*f.s,ch*f.ny-80*f.s,320*f.s,160*f.s); }
  ctx.globalAlpha=1;
  if(redVig)ctx.drawImage(vig,0,0);
  if(danger>0&&state==='play'){ ctx.globalAlpha=danger*(.14+.1*Math.sin(t*6));
    ctx.drawImage(redVig,0,0); ctx.globalAlpha=1; }
}
