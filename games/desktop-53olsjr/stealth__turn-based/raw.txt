const DIRS=[[1,0],[-1,0],[0,1],[0,-1]];
function inB(x,y){ return x>=0&&y>=0&&x<W&&y<H; }
function stepToward(g,tx,ty){
 if(!inB(tx,ty))return null;
 const start=g.x+','+g.y;
 if(start===tx+','+ty)return null;
 const prev=new Map([[start,null]]); let q=[[g.x,g.y]];
 while(q.length){
  const cur=q.shift(), x=cur[0], y=cur[1];
  if(x===tx&&y===ty){ let k=x+','+y; for(;;){ const pr=prev.get(k); if(pr===null)return null; if(pr===start)return k.split(',').map(Number); k=pr; } }
  for(const d of DIRS){ const nx=x+d[0],ny=y+d[1],nk=nx+','+ny;
   if(prev.has(nk)||!inB(nx,ny))continue; const c=grid[ny][nx];
   if(c==='#'||c==='C')continue; prev.set(nk,x+','+y); q.push([nx,ny]); }
 }
 return null;
}
function losClear(x0,y0,x1,y1){ let x=x0,y=y0; const dx=Math.abs(x1-x),dy=-Math.abs(y1-y);
 const sx=x<x1?1:-1,sy=y<y1?1:-1; let err=dx+dy;
 while(!(x===x1&&y===y1)){ const e2=2*err; if(e2>=dy){err+=dy;x+=sx;} if(e2<=dx){err+=dx;y+=sy;}
  if(x===x0&&y===y0)continue; const c=grid[y][x]; if(c==='#'||c==='C')return false; }
 return true; }
function canSee(g){ if(p.hidden)return false;
