function tone(type,f0,f1,dur,vol,delay){
  if(!AC||muted)return; vol=vol||0.4; delay=delay||0;
  const t=AC.currentTime+delay, o=AC.createOscillator(), g=AC.createGain();

  o.connect(g); g.connect(master); o.start(t); o.stop(t+dur+0.03);
}
function noiseBurst(dur,vol,delay){
  if(!AC||muted)return; vol=vol||0.3; delay=delay||0;
  const t=AC.currentTime+delay, n=Math.floor(AC.sampleRate*dur), buf=AC.createBuffer(1,n,AC.sampleRate);

  const s=AC.createBufferSource(); s.buffer=buf;
  const f=AC.createBiquadFilter(); f.type='lowpass'; f.frequency.value=900;
  const g=AC.createGain(); g.gain.value=vol;
  s.connect(f); f.connect(g); g.connect(master); s.start(t);
}

      adv-=st;
    }
  }

/* ============================== PLAYER ============================== */
function solidCell(c,r){ if(r>=ROWS)return true; if(c<0||c>=COLS)return true; if(r<0)return false; return !!grid[r][c]; }
function computePlayerCells(){
  const P=player, s=new Set();
  const cs=[Math.floor(P.x-0.34),Math.floor(P.x+0.34)], rs=[Math.floor(P.y-0.85),Math.floor(P.y-0.2)];
  for(const c of new Set(cs)) for(const r of new Set(rs)) if(r>=0&&r<ROWS) s.add(c+','+r);
  P.cells=s;
}
function collideX(P){
  const x0=P.x-HW,x1=P.x+HW,y0=P.y-HH,y1=P.y-0.02;
  const c0=Math.floor(x0+1e-6),c1=Math.floor(x1-1e-6),r0=Math.floor(y0+1e-6),r1=Math.floor(y1-1e-6);
  for(let r=r0;r<=r1;r++)for(let c=c0;c<=c1;c++){ if(!solidCell(c,r))continue;
    if(P.vx>0)P.x=Math.min(P.x,c-HW-1e-4); else if(P.vx<0)P.x=Math.max(P.x,c+1+HW+1e-4); P.vx=0; }
}
function collideY(P,vBefore){
  const x0=P.x-HW+1e-4,x1=P.x+HW-1e-4,y0=P.y-HH,y1=P.y;
  const c0=Math.floor(x0),c1=Math.floor(x1);
  if(P.vy>=0){
    const ra=Math.floor(y0+1e-6),rb=Math.floor(y1-1e-6);
    outer: for(let r=ra;r<=rb;r++)for(let c=c0;c<=c1;c++){ if(solidCell(c,r)){
      P.y=r; P.vy=0;
      if(vBefore>9){ SFX.land(vBefore); P.squash=-Math.min(.8,vBefore*.045); dust('land'); shake.p=Math.max(shake.p,2.2); }
      break outer; } }
  }else{
    const ra=Math.floor(y0+1e-6),rb=Math.floor(y1-1e-6);
    for(let r=ra;r<=rb;r++)for(let c=c0;c<=c1;c++){ if(solidCell(c,r)){ P.y=r+1+HH; P.vy=0; break; } }
  }
}
function probeGround(P){
  const r=Math.floor(P.y+0.02);
  return solidCell(Math.floor(P.x-0.3),r)||solidCell(Math.floor(P.x+0.3),r);
}
function dust(kind){
  const u=L.u, x=L.fx+player.x*u, y=L.fy+player.y*u;
  for(let i=0;i<6;i++) parts.push({x:x+(Math.random()*2-1)*u*.3,y:y-2,vx:(Math.random()*2-1)*u*1.6,
    vy:-Math.random()*u*(kind==='jump'?1.4:.5),g:u*9,life:.4,max:.4,size:u*.1,col:'rgba(220,230,255,.8)'});
}
function updPlayer(dt){
  const P=player; const dir=(keys.r?1:0)-(keys.l?1:0); if(dir)P.face=dir;
  const acc=P.onG?55:32;
  if(dir) P.vx+=dir*acc*dt;
  else{ const f=(P.onG?60:10)*dt; if(P.vx>0)P.vx=Math.max(0,P.vx-f); else P.vx=Math.min(0,P.vx+f); }
  P.vx=Math.max(-RUNMAX,Math.min(RUNMAX,P.vx));
  P.coyote=P.onG?COYOTE:P.coyote-dt;
  if(jbuf>0&&(P.onG||P.coyote>0)){ P.vy=-JUMPV; jbuf=0; P.onG=false; P.coyote=0; P.squash=1; SFX.jump(); dust('jump'); }
  P.vy=Math.min(MAXFALL,P.vy+G*dt);
  P.x+=P.vx*dt; collideX(P);
  const vBefore=P.vy; P.y+=P.vy*dt; collideY(P,vBefore);
  P.onG=probeGround(P);
  P.squash+=(0-P.squash)*Math.min(1,dt*9);
  P.blink-=dt; if(P.blink<0)P.blink=2+Math.random()*3;
}

function addFloat(text,x,y,col,size){ floaters.push({text,x,y,t:0,dur:.95,col:col||'#fff',size:size||16}); }
function updParts(dt){

    p.vy+=p.g*dt; p.x+=p.vx*dt; p.y+=p.vy*dt; p.life-=dt; if(p.life<=0)parts.splice(i,1); }
  if(parts.length>420)parts.splice(0,parts.length-420);
}
function upFloaters(dt){ for(let i=floaters.length-1;i>=0;i--){ const f=floaters[i]; f.t+=dt; f.y-=38*dt; if(f.t>f.dur)floaters.splice(i,1); } }

/* ============================== UPDATE ============================== */
function update(dt){
  tGlobal+=dt;
  for(const s of bgStars) s.y=(s.y+dt*0.01*s.z)%1;
  shake.p*=Math.exp(-7*dt); if(shake.p<.05)shake.p=0;
  updParts(dt); upFloaters(dt);

  if(state==='play'){
    jbuf=Math.max(0,jbuf-dt);
    computePlayerCells();
    if(clearAnim){ clearAnim.t-=dt; if(clearAnim.t<=0)applyClears(); }
    else{
      spawnT-=dt;
      if(spawnT<=0&&pieces.length<cap()){ spawn(); spawnT=spawnInterval()*(0.85+Math.random()*0.4); }
      updatePieces(dt);
    }
    if(state==='play'){ updPlayer(dt);
      if(!goalDone){ const dx=player.x-COLS/2, dy=(player.y-0.47)-1.55; if(Math.hypot(dx,dy)<1.05) doWin(); } }
  }
  else if(state==='menu'){
    spawnT-=dt;
    if(spawnT<=0&&pieces.length<2){ spawn(); spawnT=1.3+Math.random()*0.5; }
    updatePieces(dt);
  }
  else if(state==='dead'){ deadT-=dt; if(deadT<=0&&!panelShown){ panelShown=true; showPanel(deadHTML()); } }
  else if(state==='win'){ winT-=dt; if(winT<=0&&!panelShown){ panelShown=true; showPanel(winHTML()); } }
}

function rrect(x,y,w,h,r){ r=Math.min(r,w/2,h/2); ctx.beginPath();
  ctx.moveTo(x+r,y); ctx.arcTo(x+w,y,x+w,y+h,r); ctx.arcTo(x+w,y+h,x,y+h,r);
  ctx.arcTo(x,y+h,x,y,r); ctx.arcTo(x,y,x+w,y,r); ctx.closePath(); }
function starPath(x,y,R,r,rot){ ctx.beginPath();
  for(let i=0;i<10;i++){ const a=rot+i*Math.PI/5-Math.PI/2, rad=i%2?r:R;
    const px=x+Math.cos(a)*rad, py=y+Math.sin(a)*rad; i?ctx.lineTo(px,py):ctx.moveTo(px,py); }
  ctx.closePath(); }
function blob(x,y,r,col){ const g=ctx.createRadialGradient(x,y,0,x,y,r);
  g.addColorStop(0,col); g.addColorStop(1,'rgba(0,0,0,0)'); ctx.fillStyle=g; ctx.fillRect(x-r,y-r,r*2,r*2); }
function block(x,y,s,col){
  ctx.fillStyle=col; ctx.fillRect(x+1,y+1,s-2,s-2);
  ctx.fillStyle='rgba(255,255,255,.3)'; ctx.fillRect(x+1,y+1,s-2,Math.max(2,(s-2)*0.28));
  ctx.fillStyle='rgba(0,0,0,.22)'; const sh=(s-2)*0.28; ctx.fillRect(x+1,y+s-1-sh,s-2,sh);
  ctx.strokeStyle='rgba(0,0,0,.35)'; ctx.lineWidth=1; ctx.strokeRect(x+1.5,y+1.5,s-3,s-3);
}

function drawStar(){
  if(goalDone)return;
  const u=L.u, sx=L.fx+COLS/2*u, sy=L.fy+1.55*u, pu=.5+.5*Math.sin(tGlobal*3);
  let near=false;
  if(state==='play'&&!goalDone){ const dx=player.x-COLS/2, dy=(player.y-.47)-1.55; near=Math.hypot(dx,dy)<3.5; }
  const hr=u*(1.5+.45*pu)*(near?1.3:1);
  const g=ctx.createRadialGradient(sx,sy,0,sx,sy,hr);
  g.addColorStop(0,'rgba(255,217,60,'+(near?.34:.2)+')'); g.addColorStop(1,'rgba(255,217,60,0)');
  ctx.fillStyle=g; ctx.beginPath(); ctx.arc(sx,sy,hr,0,7); ctx.fill();
  ctx.save(); ctx.shadowColor='#ffd94f'; ctx.shadowBlur=16;
  starPath(sx,sy,u*.55*(1+.08*pu),u*.24,tGlobal*.9);
  ctx.fillStyle='#ffd94f'; ctx.fill(); ctx.lineWidth=2; ctx.strokeStyle='#fff3c8'; ctx.stroke(); ctx.restore();

}
