function bestLine(b){return b?('BEST ROUND · '+b.s+' STROKES ('+(b.d===0?'EVEN':(b.d>0?'+':'−')+Math.abs(b.d))+')'):'NO ROUNDS YET — THE COURSE AWAITS';}

/* ================= SNAKE ================= */
function pushQueue(d){
  const ref=snake.queue.length?snake.queue[snake.queue.length-1]:snake.dir;
  if(d.x===ref.x&&d.y===ref.y)return;
  if(d.x===-ref.x&&d.y===-ref.y)return;
  if(snake.queue.length<3)snake.queue.push(d);
}
function doSnakeStep(){
  while(snake.queue.length){const d=snake.queue.shift();if(!(d.x===snake.dir.x&&d.y===snake.dir.y)){snake.dir=d;break;}}
  const nh={x:snake.segs[0].x+snake.dir.x,y:snake.segs[0].y+snake.dir.y};
  if(solidAt(nh.x,nh.y)){snake.bump=1;return;}
  const body=snake.grow>0?snake.segs:snake.segs.slice(0,-1);
  if(body.some(s=>s.x===nh.x&&s.y===nh.y)){tangle();return;}
  snake.segs.unshift(nh);
  if(snake.grow>0)snake.grow--;else snake.segs.pop();
  const wi=worms.findIndex(w=>w.x===nh.x&&w.y===nh.y);
  if(wi>=0){
    worms.splice(wi,1);snake.grow+=3;
    snake.powerBonus=Math.min(.18,snake.powerBonus+.06);
    sMunch();updateHUD();
    addText(nh.x,nh.y,'POWER+','#e8c56a');
    burst(nh.x,nh.y,10,{colors:['#ffd98a','#ff8fab','#7ee08a'],speed:3,g:2,life:.5,size:.09});
  }
}
function tangle(){
  const remove=Math.min(4,snake.segs.length-3);
  snake.segs.splice(snake.segs.length-remove,remove);
  snake.stun=.3;sTangle();
  const h=snake.segs[0];
  burst(h.x,h.y,8,{colors:['#a97142','#e8d29a'],speed:2.5,g:3,life:.45,size:.08});
  addText(h.x,h.y,'TANGLE!','#ff9d84');
}

/* ================= BALL ================= */
function swing(){
  if(!charge.active)return;charge.active=false;els.power.classList.remove('show');
  if(state!=='play'||ball.state!=='held')return;
  const hd=snake.segs[0];
  if(solidAt(hd.x+snake.dir.x,hd.y+snake.dir.y)){sBlock();snake.bump=1.4;return;}
  strokes++;updateHUD();strokePulse();
  const p=chargeVal(),v=(MINV+(maxV()-MINV)*p);
  ball.state='rolling';
  ball.x=hd.x+snake.dir.x*.42;ball.y=hd.y+snake.dir.y*.42;
  ball.vx=snake.dir.x*v;ball.vy=snake.dir.y*v;
  ball.lastSafe={x:hd.x,y:hd.y};ball.rollT=0;
  snake.recoil=1;sSwing(p);
}
function moveAxis(axis,h){
  const v=axis==='x'?ball.vx:ball.vy;if(!v)return;
  const cur=(axis==='x')?ball.x:ball.y;
  const np=cur+v*h,r=.19;
  const lo=Math.floor((axis==='x')?ball.y-r:np-r),hi=Math.floor((axis==='x')?ball.y+r:np+r);
  let a0,a1;if(axis==='x'){a0=Math.floor(np-r);a1=Math.floor(np+r);}else{a0=Math.floor(np-r);a1=Math.floor(np+r);}
  for(let i=lo;i<=hi;i++)for(let j=a0;j<=a1;j++){
    if(axis==='x'){if(solidAt(j,i)){hitWall(v,'x',j);return;}}
    else{if(solidAt(i,j)){hitWall(v,'y',j);return;}}
  }
  if(axis==='x')ball.x=np;else ball.y=np;
}
function hitWall(v,axis,face){
  const eps=.002;let cx,cy;
  if(v>0){if(axis==='x'){ball.x=face-.19-eps;cx=face-.19;}else{ball.y=face-.19-eps;cy=face-.19;}}
  else{if(axis==='x'){ball.x=face+.19+eps;cx=face+.19;}else{ball.y=face+.19+eps;cy=face+.19;}}
  if(cx===undefined){cx=(axis==='x')?ball.x:ball.y;if(axis==='y'){} }
  const px=(axis==='x')?ball.x:ball.y, py=(axis==='x')?ball.y:ball.x;
  if(axis==='x'){ball.vx=-v*REST;}else{ball.vy=-v*REST;}
  if(Math.abs(v)>1.8)sBounce(Math.abs(v));
  burst(px,py,4,{colors:['#e8d29a','#ffffff'],speed:1.6,g:2,life:.35,size:.07});
}
function ballStep(dt){
  if(ball.state!=='rolling')return;
  let sp=Math.hypot(ball.vx,ball.vy);if(sp<.01){stopBall();return;}
  const cdx=grid.cup.x-ball.x,cdy=grid.cup.y-ball.y,cd=Math.hypot(cdx,cdy);
  if(cd>.001&&cd<.62&&sp<9){ball.vx+=cdx/cd*13*dt;ball.vy+=cdy/cd*13*dt;}
  const steps=Math.max(1,Math.ceil(sp*dt/.28)),h=dt/steps;
  for(let i=0;i<steps;i++){moveAxis('x',h);if(ball.state!=='rolling')return;moveAxis('y',h);if(ball.state!=='rolling')return;}
  const t=tileAt(Math.round(ball.x),Math.round(ball.y));
  if(t===T.WATER){splash();return;}
  sp=Math.hypot(ball.vx,ball.vy);
  const ns=Math.max(0,sp-DECELV[t]*dt);if(sp>0){ball.vx*=ns/sp;ball.vy*=ns/sp;}
  ball.rot+=ns*dt/.19;
  if(t!==T.WATER)ball.lastSafe={x:ball.x,y:ball.y};
  // rolling tick sound
  if(ns>2.5){ball.rollT+=dt;if(ball.rollT>Math.max(.09,6/ns)){ball.rollT=0;tone(340,300,.04,'triangle',.045);}}
  const d2=Math.hypot(ball.x-grid.cup.x,ball.y-grid.cup.y),s2=Math.hypot(ball.vx,ball.vy);
  if(d2<.3){
    if(s2<7.5)doSink();
    else{let nx,ny;if(d2>.001){nx=(ball.x-grid.cup.x)/d2;ny=(ball.y-grid.cup.y)/d2;}else{nx=1;ny=0;}
      ball.vx=nx*s2*.45;ball.vy=ny*s2*.45;sLip();}
    return;
  }
  if(s2<.7)stopBall();
}
function stopBall(){ball.vx=0;ball.vy=0;ball.state='loose';ball.x=Math.round(ball.x);ball.y=Math.round(ball.y);}
function splash(){
  strokes++;updateHUD();strokePulse();sSplash();
  burst(ball.x,ball.y,14,{colors:['#7cc7ee','#ffffff','#2f8fc4'],speed:4,g:16,life:.7,size:.09});
  banner('SPLASH!','+1 penalty stroke','bad');
  ball.x=ball.lastSafe.x;ball.y=ball.lastSafe.y;ball.vx=0;ball.vy=0;ball.state='loose';
}
function doSink(){
  scores[holeIdx]=strokes;
  ball.state='sunk';sinkAnim=0;flagT=0;
  state='holeDone';holeTimer=2.2;
  sSink();updateHUD();
  const cx=grid.cup.x,cy=grid.cup.y;
  burst(cx,cy,56,{colors:['#e8c56a','#f2ead6','#e6483d','#7ee08a','#ffffff'],speed:6,g:14,life:1.4,size:.12,rect:true});
  const d=strokes-grid.par;let word,cls;
  if(strokes===1){word='ACE!!';cls='good';}
  else if(d<=-3){word='EAGLE!';cls='good';}
  else if(d===-2){word='EAGLE!';cls='good';}
  else if(d===-1){word='BIRDIE!';cls='good';}
  else if(d===0){word='PAR';cls='';}
  else if(d===1){word='BOGEY';cls='bad';}
  else if(d===2){word='DOUBLE BOGEY';cls='bad';}
  else{word='+'+d;cls='bad';}
  setTimeout(()=>banner(word,strokes+' stroke'+(strokes===1?'':'s')+' · par '+grid.par,cls),450);
}
function nextHole(){
  if(holeIdx>=HOLES.length-1){endRound();return;}
  loadHole(holeIdx+1);state='play';introToast();
}

/* ================= END ================= */
function endRound(){
  state='end';
  const total=scores.reduce((a,b)=>a+(b||0),0),diff=total-34;
  let ph;
  if(diff<=-5)ph='Legendary. The caddie is weeping.';
  else if(diff<=-3)ph='Blistering work out there — a round for the club records.';
  else if(diff===-2)ph='Sharp as a freshly honed wedge.';
  else if(diff===-1)ph='One under par — the flag knows your name now.';
  else if(diff===0)ph='Even par. A respectable stroll around nine.';
  else if(diff<=3)ph='The water had its day.';
  else if(diff<=6)ph='Perhaps fewer worms next time.';
  else ph='The snake has opinions about you now.';
  let html='';
  HOLES.forEach((h,i)=>{
    const s=scores[i]??'—',d=(scores[i]!=null)?scores[i]-h.par:0;
    const cls=scores[i]==null?'':(d<0?'dn':d>0?'dp':'dz');
    const dtxt=scores[i]==null?'':(d===0?'E':(d>0?'+':'−')+Math.abs(d));
    html+='<tr><td class="hn">'+(i+1)+' · '+h.name+'</td><td>'+h.par+'</td><td class="st">'+s+'</td><td class="'+cls+'">'+dtxt+'</td></tr>';
  });
  els.endRows.innerHTML=html;
  els.endTotal.textContent=total;
  const dCls=diff<0?'dn':diff>0?'dp':'dz';
  els.endDiff.innerHTML='<span class="'+dCls+'" style="font-size:19px;font-family:Impact,sans-serif">'+(diff===0?'EVEN':(diff>0?'+':'−')+Math.abs(diff))+'</span>';
  els.endPhrase.textContent=ph;
  const prev=loadBest();let isBest=false;
  if(!prev||total<prev.s){isBest=true;saveBest(total,diff);}
  els.newBest.style.display=isBest?'inline-block':'none';
  const b2=loadBest();els.endBest.textContent='BEST ROUND · '+b2.s+' STROKES ('+(b2.d===0?'EVEN':(b2.d>0?'+':'−')+Math.abs(b2.d))+')';
  els.menu.classList.add('hidden');els.end.classList.remove('hidden');
  for(let i=0;i<3;i++)burst(Math.random()*grid.w,Math.random()*grid.h*.6,40,{colors:['#e8c56a','#f2ead6','#7ee08a'],speed:6,g:12,life:1.6,size:.12,rect:true});
}

/* ================= PARTICLES ================= */
function burst(x,y,n,o){
  for(let i=0;i<n;i++){
    const a=Math.random()*Math.PI*2,sp=(o.speed||4)*(.4+Math.random()*.8);
    particles.push({x,y,vx:Math.cos(a)*sp,vy:Math.sin(a)*sp-(o.rect?3:0),g:o.g||10,
      life:(o.life||.6)*(.7+Math.random()*.6),maxLife:o.life||.6,
      color:(o.colors||['#fff'])[i%(o.colors||['#fff']).length],size:(o.size||.09)*cell,
      rot:Math.random()*6,vr:(Math.random()-.5)*10,rect:!!o.rect});
  }
}
function addText(x,y,txt,color){particles.push({x,y,vx:0,vy:-1.2,g:0,life:.9,maxLife:.9,text:txt,color,size:cell*.4,rot:0,vr:0});}
function updateParticles(dt){
  for(let i=particles.length-1;i>=0;i--){const p=particles[i];
    p.life-=dt;if(p.life<=0){particles.splice(i,1);continue;}
    p.vy+=p.g*dt;p.x+=p.vx*dt;p.y+=p.vy*dt;p.rot+=p.vr*dt;
  }
}

/* ================= UPDATE ================= */
function update(dt){
  for(const c of clouds){c.x+=c.v*dt;if(c.x>grid.w+c.r)c.x=-c.r;}
  updateParticles(dt);
  if(snake){if(snake.bump>0)snake.bump=Math.max(0,snake.bump-dt*4);if(snake.recoil>0)snake.recoil=Math.max(0,snake.recoil-dt*5);}
  if(state==='play'){
    if(snake.stun>0)snake.stun-=dt;
    else if(!charge.active){snake.tickT+=dt;while(snake.tickT>=TICK){snake.tickT-=TICK;doSnakeStep();}}
    if(charge.active){
      charge.t+=dt*1.25;
      const p=chargeVal();els.pCover.style.width=((1-p)*100)+'%';
    }
    ballStep(dt);
    if(ball.state==='loose'){
      const hd=snake.segs[0];
      if(hd.x===ball.x&&hd.y===ball.y){ball.state='held';sPickup();burst(ball.x,ball.y,6,{colors:['#ffffff','#e8c56a'],speed:2,g:0,life:.4,size:.07});}
    }
  }else if(state==='holeDone'){
    sinkAnim+=dt;if(flagT>=0)flagT+=dt;
    holeTimer-=dt;if(holeTimer<=0)nextHole();
  }
}

function resize(){
  W=window.innerWidth;H=window.innerHeight;dpr=Math.min(2,window.devicePixelRatio||1);
  cv.width=W*dpr;cv.height=H*dpr;cv.style.width=W+'px';cv.style.height=H+'px';
  ctx.setTransform(dpr,0,0,dpr,0,0);layout();
}
function layout(){
  if(!grid)return;
  cell=Math.max(14,Math.floor(Math.min((W-24)/grid.w,(H-175)/grid.h)));
  ox=Math.floor((W-cell*grid.w)/2);
  oy=92+Math.max(0,Math.floor((H-92-72-cell*grid.h)/2));
}
function circle(x,y,r){ctx.beginPath();ctx.arc(x,y,r,0,7);ctx.fill();}

  // backdrop
  ctx.fillStyle='#0b241a';ctx.fillRect(0,0,W,H);
  const vg=ctx.createRadialGradient(W/2,H*.42,60,W/2,H*.42,Math.max(W,H)*.75);

  ctx.save();ctx.shadowColor='rgba(0,0,0,.55)';ctx.shadowBlur=26;ctx.shadowOffsetY=12;
  ctx.fillStyle='#0e2f1d';ctx.fillRect(ox-6,oy-6,grid.w*cell+12,grid.h*cell+12);ctx.restore();

  // tiles
  for(let y=0;y<grid.h;y++)for(let x=0;x<grid.w;x++){
    const t=grid.tiles[y][x],px=ox+x*cell,py=oy+y*cell;
    if(t===T.HEDGE){
      ctx.fillStyle='#1a5230';ctx.fillRect(px,py,cell+1,cell+1);
      ctx.fillStyle='#256b41';

      ctx.strokeStyle='rgba(7,26,15,.5)';ctx.lineWidth=1;ctx.strokeRect(px+.5,py+.5,cell-1,cell-1);
    }else if(t===T.GRASS){
      ctx.fillStyle=(x%2)?'#4ea63f':'#58b24a';ctx.fillRect(px,py,cell+1,cell+1);
      if(hash(x*3,y*7)>.8){ctx.fillStyle='rgba(30,90,40,.25)';ctx.fillRect(px+hash(y,x)*cell*.8,py+hash(x,y)*cell*.8,2.5,2.5);}
    }else if(t===T.ROUGH){
      ctx.fillStyle='#3c7d3a';ctx.fillRect(px,py,cell+1,cell+1);

      for(let k=0;k<4;k++){const hx=px+hash(x*5+k,y*13+k)*cell,hy=py+hash(y*7+k,x*11+k)*cell;

    }else if(t===T.SAND){

      for(let k=0;k<5;k++)circle(px+hash(x*3+k,y*5+k)*cell,py+hash(y*9+k,x*3+k)*cell,1.2);
    }else if(t===T.WATER){
      ctx.fillStyle='#2f8fc4';ctx.fillRect(px,py,cell+1,cell+1);
      const wy=py+cell*(.5+.22*Math.sin(time*2+(x*1.3+y*1.7)));
      ctx.strokeStyle='rgba(255,255,255,.3)';ctx.lineWidth=2;
      ctx.beginPath();ctx.moveTo(px+4,wy);ctx.lineTo(px+cell-4,wy);ctx.stroke();
      const wy2=py+cell*(.5-.2*Math.sin(time*1.6+(x*.9+y*2.1)+Math.PI));
      ctx.strokeStyle='rgba(255,255,255,.14)';
      ctx.beginPath();ctx.moveTo(px+5,wy2);ctx.lineTo(px+cell-5,wy2);ctx.stroke();
      if(hash(x,y)>.72){const a=.3+.3*Math.sin(time*3+hash(y,x)*9);if(a>.3){ctx.fillStyle='rgba(255,255,255,'+(a-.3)+')';circle(px+hash(x+1,y)*cell,py+hash(x,y+1)*cell,1.6);}}

      if(tileAt(x,y-1)!==T.WATER){ctx.beginPath();ctx.moveTo(px,py+1.5);ctx.lineTo(px+cell,py+1.5);ctx.stroke();}

  }
}

function drawSnake(){
  const sway=(state==='play'||state==='menu')?1:0;
  const pts=snake.segs.map((s,i)=>{
    const sw=Math.sin(time*2.4+i*.7)*cell*.03*sway;
    return {x:ox+(s.x)*cell+cell/2,y:oy+s.y*cell+cell/2+sw};
  });
  const hd=pts[0];
  const rx=hd.x-snake.dir.x*snake.recoil*cell*.2, ry=hd.y-snake.dir.y*snake.recoil*cell*.2;
  pts[0]={x:rx,y:ry};
  ctx.lineJoin='round';ctx.lineCap='round';
  const path=()=>{ctx.beginPath();ctx.moveTo(pts[0].x,pts[0].y);for(let i=1;i<pts.length;i++)ctx.lineTo(pts[i].x,pts[i].y);};
  path();ctx.strokeStyle='#7c3f07';ctx.lineWidth=cell*.66;ctx.stroke();
  path();ctx.strokeStyle='#f59e0b';ctx.lineWidth=cell*.52;ctx.stroke();

  // head

  ctx.strokeStyle='#7c3f07';ctx.lineWidth=3;ctx.beginPath();ctx.arc(rx,ry,hr,0,7);ctx.stroke();
  // tongue
  if(snake.stun<=0&&(time%2.6)<.32){
    const tp=(time%2.6)/.32,len=cell*.55*Math.sin(tp*Math.PI);
    const fx=rx+snake.dir.x*hr,fy=ry+snake.dir.y*hr;
    const ex=fx+snake.dir.x*len,ey=fy+snake.dir.y*len;
    ctx.strokeStyle='#e6483d';ctx.lineWidth=2;
    ctx.beginPath();ctx.moveTo(fx,fy);ctx.lineTo(ex,ey);
    ctx.moveTo(ex,ey);ctx.lineTo(ex+Math.cos(Math.atan2(snake.dir.y,snake.dir.x)+.5)*cell*.12,ey+Math.sin(Math.atan2(snake.dir.y,snake.dir.x)+.5)*cell*.12);
    ctx.moveTo(ex,ey);ctx.lineTo(ex+Math.cos(Math.atan2(snake.dir.y,snake.dir.x)-.5)*cell*.12,ey+Math.sin(Math.atan2(snake.dir.y,snake.dir.x)-.5)*cell*.12);
    ctx.stroke();
  }
  // eyes
  const px=-snake.dir.y,py=snake.dir.x;
  let lx,ly,lxv=0,lyv=0;
  if(ball.state==='loose'){lx=ball.x*cell+ox-rx;ly=ball.y*cell+oy-ry;const L=Math.hypot(lx,ly)||1;lxv=lx/L*.035*cell;lyv=ly/L*.035*cell;}
  else{lxv=snake.dir.x*.02*cell;lyv=snake.dir.y*.02*cell;}
  for(const s of[-1,1]){
    const ex=rx+px*s*cell*.16+snake.dir.x*cell*.1,ey=ry+py*s*cell*.16+snake.dir.y*cell*.1;
    ctx.fillStyle='#fff';circle(ex,ey,cell*.085);
    ctx.fillStyle='#1a1208';circle(ex+lxv,ey+lyv,cell*.042);
  }
}
function drawBall(){
  let bx,by,scale=1;
  if(ball.state==='held'){const hd=snake.segs[0];bx=(hd.x+snake.dir.x*.42)*cell+ox;by=(hd.y+snake.dir.y*.42)*cell+oy;}
  else{bx=ball.x*cell+ox;by=ball.y*cell+oy;}

  const r=cell*.17*scale;

}
function drawAimGuide(){
  const hd=snake.segs[0];
  if(solidAt(hd.x+snake.dir.x,hd.y+snake.dir.y))return;
  const p=chargeVal(),v=MINV+(maxV()-MINV)*p;
  let L=v*v/(2*DECELV[T.GRASS]);
  let k=1;while(k<30&&!solidAt(hd.x+snake.dir.x*k,hd.y+snake.dir.y*k))k++;
  L=Math.min(L,(k-1)*.9);if(L<.4)return;
  const sx=(hd.x+snake.dir.x*.42)*cell+ox,sy=(hd.y+snake.dir.y*.42)*cell+oy;
  const ex=sx+snake.dir.x*L*cell,ey=sy+snake.dir.y*L*cell;
  ctx.save();ctx.strokeStyle='rgba(255,255,240,'+(.3+.15*Math.sin(time*6))+')';ctx.lineWidth=2;
  ctx.setLineDash([7,9]);ctx.beginPath();ctx.moveTo(sx,sy);ctx.lineTo(ex,ey);ctx.stroke();ctx.setLineDash([]);
  const a=Math.atan2(snake.dir.y,snake.dir.x);
  ctx.fillStyle='rgba(255,255,240,.5)';
  ctx.beginPath();ctx.moveTo(ex+Math.cos(a)*cell*.16,ey+Math.sin(a)*cell*.16);

  ctx.lineTo(ex+Math.cos(a-2.5)*cell*.14,ey+Math.sin(a-2.5)*cell*.14);ctx.closePath();ctx.fill();
  ctx.restore();
}

function drawParticles(){
  for(const p of particles){
    const a=Math.max(0,p.life/p.maxLife);

  }
}
