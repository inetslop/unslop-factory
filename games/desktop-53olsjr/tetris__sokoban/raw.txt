function nearestEmpty(sx,sy){
  if(!solid(sx,sy)) return [sx,sy];
  const q=[[sx,sy]], seen=new Set([idx(sx,sy)]);
  while(q.length){
    const [x,y]=q.shift();
    for(const [dx,dy] of [[1,0],[-1,0],[0,1],[0,-1]]){
      const nx=x+dx,ny=y+dy;
      if(nx<0||nx>=COLS||ny<0||ny>=ROWS) continue;
      const i=idx(nx,ny);
      if(seen.has(i)||solid(nx,ny)) continue;
      return [nx,ny];
    }
    seen.add(i); q.push([x,y]);
  }
  return null;
}
function moveWorker(dx,dy){

function attUpdate(dt){
  const p=att.p; if(!p) return;
  att.a+=dt;
  if(att.a>430){
    att.a=0; const r=Math.random();
    if(r<.28){ if(canPlace(pCells(p).map(([x,y])=>[x-1,y]))) p.x--; }
    else if(r<.56){ if(canPlace(pCells(p).map(([x,y])=>[x+1,y]))) p.x++; }
    else if(r<.78){ const nr=(p.r+1)&3; const c=ROTS[p.t][nr].map(([x,y])=>[p.x+x,p.y+y]); if(canPlace(c)) p.r=nr; }
    else { while(canPlace(pCells(p).map(([x,y])=>[x,y+1]))) p.y++; attLock(); return; }
  }
  att.t+=dt;
  while(att.t>520){ att.t-=520;
    if(canPlace(pCells(p).map(([x,y])=>[x,y+1]))) p.y++; else { attLock(); break; } }
}

window.addEventListener('keydown',e=>{
  const k=e.key;
  if([' ','ArrowUp','ArrowDown','ArrowLeft','ArrowRight'].includes(k)) e.preventDefault();
  initAudio();
  if(state==='start'||state==='over'){ if(k==='Enter'||k===' ') startGame(); return; }
  if(k==='p'||k==='P'){ togglePause(); return; }
  if(k==='m'||k==='M'){ muted=!muted; sndEl.textContent=muted?'OFF':'ON'; sndEl.style.color=muted?'var(--red)':'var(--mint)'; return; }
  if(state==='pause'){ if(k==='Enter'||k===' ') togglePause(); return; }
  if(state!=='play') return;
  if(phase==='drop'){
    switch(k){
      case 'ArrowLeft':  held.l=true; lastMove.l=now; movePiece(-1); break;
      case 'ArrowRight': held.r=true; lastMove.r=now; movePiece(1); break;
      case 'ArrowUp': case 'x': case 'X': if(!e.repeat) rotatePiece(1); break;
      case 'q': case 'Q': if(!e.repeat) rotatePiece(-1); break;
      case 'ArrowDown':  held.d=true; gravT=Math.max(gravT,20); break;
      case ' ': if(!e.repeat) hardDrop(); break;
    }
  } else {
    switch(k){
      case 'ArrowLeft':  held.l=true; lastMove.ul=now; moveWorker(-1,0); break;
      case 'ArrowRight': held.r=true; lastMove.ur=now; moveWorker(1,0); break;
      case 'ArrowUp':    held.u=true; lastMove.uu=now; moveWorker(0,-1); break;
      case 'ArrowDown':  held.d=true; lastMove.ud=now; moveWorker(0,1); break;
      case 'z': case 'Z': if(!e.repeat) undoPush(); break;
      case 'r': case 'R': if(!e.repeat) resetLevel(); break;
      case ' ': case 'Enter': if(!e.repeat) callNext(); break;
    }
  }
});
window.addEventListener('keyup',e=>{
  if(e.key==='ArrowLeft') held.l=false;
  if(e.key==='ArrowRight') held.r=false;
  if(e.key==='ArrowDown') held.d=false;
  if(e.key==='ArrowUp') held.u=false;
});

/* ================= UPDATE ================= */
function update(dt){
  for(const m of motes){ m.y-=m.v*dt; if(m.y<-4){ m.y=H+4; m.x=Math.random()*W; } }
  shake=Math.max(0,shake-dt*26);
  worker.squash=Math.max(0,worker.squash-dt*6);
  worker.duck=Math.max(0,worker.duck-dt*3.5);
  for(let i=parts.length-1;i>=0;i--){
    const p=parts[i]; p.life+=dt;
    if(p.life>p.max){ parts.splice(i,1); continue; }
    p.vy+=p.g*dt; p.x+=p.vx*dt; p.y+=p.vy*dt;
  }
  for(let i=floats.length-1;i>=0;i--){ floats[i].t+=dt; if(floats[i].t>floats[i].dur) floats.splice(i,1); }

  if(state==='start'){ attUpdate(dt); return; }
  if(state!=='play'&&state!=='clear') return;
  if(state==='clear'){
    if(now-clearStart>2000){ setupLevel(level+1); state='play'; }
    return;
  }
  if(phase==='drop'){
    if(held.l&&now-lastMove.l>140){ lastMove.l=now; movePiece(-1); }
    if(held.r&&now-lastMove.r>140){ lastMove.r=now; movePiece(1); }
    if(piece){
      for(const [x,y] of pCells(piece)) if(x===worker.x&&y===worker.y) worker.duck=1;
      gravT+=dt; const iv=held.d?52:gravMs(); let g=0;
      while(gravT>=iv&&g++<8){ gravT-=iv; if(!stepDown()) break; }
    }
  } else {
    const rep=165;
    if(held.l&&now-lastMove.ul>rep){ lastMove.ul=now; moveWorker(-1,0); }
    if(held.r&&now-lastMove.ur>rep){ lastMove.ur=now; moveWorker(1,0); }
    if(held.u&&now-lastMove.uu>rep){ lastMove.uu=now; moveWorker(0,-1); }
    if(held.d&&now-lastMove.ud>rep){ lastMove.ud=now; moveWorker(0,1); }
  }
}

function rr(x,y,w,h,r){
  r=Math.min(r,w/2,h/2);
  ctx.beginPath();
  ctx.moveTo(x+r,y); ctx.arcTo(x+w,y,x+w,y+h,r); ctx.arcTo(x+w,y+h,x,y+h,r);
  ctx.arcTo(x,y+h,x,y,r); ctx.arcTo(x,y,x+w,y,r); ctx.closePath();
}
function drawBG(){
  const g=ctx.createLinearGradient(0,0,0,H);
  g.addColorStop(0,'#0b1626'); g.addColorStop(.55,'#0d1c2e'); g.addColorStop(1,'#0a1420');
  ctx.fillStyle=g; ctx.fillRect(0,0,W,H);
  const cx=L.bx+L.cell*COLS/2, cy=L.by+L.cell*ROWS/2;
  const rad=ctx.createRadialGradient(cx,cy,20,cx,cy,Math.max(W,H)*.65);
  const glow = state==='clear' ? 'rgba(87,227,154,.12)' : 'rgba(90,215,255,.06)';
  rad.addColorStop(0,glow); rad.addColorStop(1,'rgba(0,0,0,0)');
  ctx.fillStyle=rad; ctx.fillRect(0,0,W,H);
  for(const m of motes){
    ctx.globalAlpha=.05+m.z*.12;
    ctx.fillStyle='#a8dcff';
    ctx.fillRect(m.x,m.y,m.s,m.s);
  }
  ctx.globalAlpha=1;
}
function drawPanel(){

  // chute
  ctx.fillStyle='#0a1624';

  ctx.fillStyle='#101d2e'; rr(x,y,bw,bh,8); ctx.fill();
  ctx.restore();

  tg.addColorStop(0,'rgba(90,215,255,.07)'); tg.addColorStop(.15,'rgba(0,0,0,0)');
  ctx.fillStyle=tg; rr(x,y,bw,bh,8); ctx.fill();

  ctx.strokeStyle='rgba(255,255,255,.03)'; ctx.lineWidth=1;
  for(let i=1;i<COLS;i++){ ctx.beginPath(); ctx.moveTo(x+i*L.cell,y+1); ctx.lineTo(x+i*L.cell,y+bh-1); ctx.stroke(); }

  ctx.strokeStyle='#3f6a95'; ctx.lineWidth=2.5; const b=12;

    ctx.beginPath(); ctx.moveTo(bx+sx*b,by); ctx.lineTo(bx,by); ctx.lineTo(bx,by+sy*b); ctx.stroke();

function drawPads(){
  for(const [x,y] of targetList){
    const px=L.bx+x*L.cell, py=L.by+y*L.cell;
    const cx0=px+L.cell/2, cy0=py+L.cell/2, s=L.cell*.52;
    const cov=covered.has(idx(x,y));
    const pl=.5+.35*Math.sin(now/300+(x*7+y*13));
    ctx.save();
    if(cov){ ctx.shadowColor='rgba(87,227,154,.9)'; ctx.shadowBlur=14; }
    ctx.strokeStyle=cov?'#aef7cf':`rgba(87,227,154,${.3+.4*pl})`;
    ctx.lineWidth=Math.max(1.5,L.cell*.07);
    rr(cx0-s/2,cy0-s/2,s,s,s*.28); ctx.stroke();
    ctx.fillStyle=cov?'rgba(87,227,154,.9)':`rgba(87,227,154,${.18+.3*pl})`;
    const d=s*.2; rr(cx0-d/2,cy0-d/2,d,d,d*.3); ctx.fill();
    ctx.restore();
  }
}

    if(!v) continue;
    const px=L.bx+x*L.cell, py=L.by+y*L.cell, s=L.cell;
    if(v===1){ // rock
      ctx.fillStyle='#3a5068'; rr(px+1,py+1,s-2,s-2,4); ctx.fill();
      ctx.fillStyle='rgba(255,255,255,.09)'; rr(px+2,py+2,s-4,Math.max(3,s*.2),3); ctx.fill();
      ctx.strokeStyle='#22344a'; ctx.lineWidth=2; rr(px+1,py+1,s-2,s-2,4); ctx.stroke();
      ctx.fillStyle='rgba(0,0,0,.28)';
      const r=Math.max(1.5,s*.05);
      [[.2,.2],[.8,.2],[.2,.8],[.8,.8]].forEach(([fx,fy])=>{ ctx.beginPath(); ctx.arc(px+s*fx,py+s*fy,r,0,7); ctx.fill(); });
    } else { // crate
      const age=now-(flash[i]||0);
      ctx.save();
      if(covered.has(i)){ ctx.shadowColor='rgba(87,227,154,.85)'; ctx.shadowBlur=16; }
      ctx.fillStyle='#e2a54a'; rr(px+1.5,py+1.5,s-3,s-3,5); ctx.fill();

function drawCell(p,x,y,color,alpha,ghost){
  const px=L.bx+x*L.cell, py=L.by+y*L.cell, s=L.cell;
  if(ghost){
    ctx.strokeStyle=color; ctx.globalAlpha=.35*alpha; ctx.lineWidth=2;
    rr(px+2,py+2,s-4,s-4,4); ctx.stroke();
    ctx.fillStyle=color; ctx.globalAlpha=.08*alpha; rr(px+2,py+2,s-4,s-4,4); ctx.fill();
    ctx.globalAlpha=1; return;
  }
  ctx.save();
  ctx.globalAlpha=alpha;
  ctx.fillStyle=color; rr(px+1.5,py+1.5,s-3,s-3,5); ctx.fill();
  ctx.fillStyle='rgba(255,255,255,.3)'; rr(px+3,py+3,s-6,Math.max(2,s*.24),3); ctx.fill();
  ctx.fillStyle='rgba(0,0,0,.28)'; rr(px+3,py+s-Math.max(4,s*.3)-1,s-6,Math.max(2,s*.22),3); ctx.fill();
  ctx.strokeStyle='rgba(10,20,30,.6)'; ctx.lineWidth=1.5; rr(px+1.5,py+1.5,s-3,s-3,5); ctx.stroke();
  ctx.restore();
}
function drawGhost(){
  if(!piece) return;
  let gy=piece.y;
  while(canPlace(pCells(piece).map(([x,y])=>[x,y+1]))) gy++;
  for(const [ox,oy] of ROTS[piece.t][piece.r]){
    const y=gy+oy; if(y<0) continue;
    drawCell(piece,piece.x+ox,y,PCOL[piece.t],1,true);
  }
}
function drawActivePiece(){
  const p = state==='start' ? att.p : piece;
  if(!p) return;
  for(const [x,y] of pCells(p)){ if(y<-2) continue; drawCell(p,x,y,PCOL[p.t],1,false); }
}
function drawWorker(){
  const w=worker;
  const pop=easeOutBack(Math.min(1,(now-w.born)/260));
  const cxp=L.bx+(w.x+.5)*L.cell, cyp=L.by+(w.y+.5)*L.cell;
  const s=L.cell*.74*pop;
  if(s<=0) return;
  const sy=(1-.22*w.squash)*(1-.62*w.duck), sx=(1+.18*w.squash)*(1+.3*w.duck);
  ctx.fillStyle='rgba(0,0,0,.35)';
  ctx.beginPath(); ctx.ellipse(cxp,cyp+L.cell*.42,s*.5*sx,s*.13,0,0,7); ctx.fill();
  ctx.save();
  ctx.translate(cxp,cyp); ctx.scale(sx,sy);
  ctx.fillStyle='#ff8a70'; rr(-s/2,-s/2,s,s,s*.3); ctx.fill();
  ctx.strokeStyle='#b23f31'; ctx.lineWidth=Math.max(2,s*.09); rr(-s/2,-s/2,s,s,s*.3); ctx.stroke();
  const ex=w.face[0]*s*.05, ey=w.face[1]*s*.05;
  ctx.fillStyle='#fff';
  ctx.beginPath(); ctx.arc(-s*.16-ex*0,-s*.08+ey*.5,s*.14,0,7); ctx.fill();
  ctx.beginPath(); ctx.arc(s*.16-ex*0,-s*.08+ey*.5,s*.14,0,7); ctx.fill();
  ctx.fillStyle='#1c2836';
  ctx.beginPath(); ctx.arc(-s*.16+ex,-s*.08+ey,s*.065,0,7); ctx.fill();
  ctx.beginPath(); ctx.arc(s*.16+ex,-s*.08+ey,s*.065,0,7); ctx.fill();
  // antenna
  const blink=w.duck>.4 ? '#ff5d5d' : (Math.sin(now/300)>0?'#ffd23f':'#f5a83c');
  ctx.strokeStyle='#b23f31'; ctx.lineWidth=Math.max(1.5,s*.07);
  ctx.beginPath(); ctx.moveTo(0,-s/2); ctx.lineTo(0,-s/2-s*.18); ctx.stroke();
  ctx.fillStyle=blink; ctx.beginPath(); ctx.arc(0,-s/2-s*.24,Math.max(2,s*.09),0,7); ctx.fill();
  ctx.restore();
}

/* ================= HUD SYNC ================= */
const hudCache={};
function setTxt(el,v){ if(hudCache[el.id]!==v){ hudCache[el.id]=v; el.textContent=v; } }
function syncHUD(){
  setTxt(scoreEl,fmt(score)); setTxt(bestEl,fmt(best));
  setTxt(lvlNumEl,''+(level+1)); setTxt(lvlNameEl,LEVELS[level%LEVELS.length].name);
  setTxt(padsTxtEl,covered.size+'/'+targetList.length);
  const w=targetList.length?Math.round(covered.size/targetList.length*100)+'%':'0%';
  if(hudCache.pbar!==w){ hudCache.pbar=w; padsBarEl.style.width=w; }
  let chipTxt,chipCls;
  if(state==='play'){ chipTxt=phase==='drop'?'DROP':'SHOVE'; chipCls=phase==='drop'?'c-drop':'c-shove'; }
  else if(state==='clear'){ chipTxt='CLEAR!'; chipCls='c-clear'; }
  else if(state==='over'){ chipTxt='OVER'; chipCls='c-idle'; }
  else { chipTxt='—'; chipCls='c-idle'; }
  const key=chipTxt+chipCls;
  if(hudCache.chip!==key){ hudCache.chip=key; chipEl.textContent=chipTxt; chipEl.className='chip '+chipCls; }
  hintDrop.classList.toggle('hidden',!(state==='play'&&phase==='drop'));
  hintShove.classList.toggle('hidden',!(state==='play'&&phase==='shove'));
}

function render(){
  ctx.setTransform(dpr,0,0,dpr,0,0);
  drawBG();
  const sx=(Math.random()-.5)*shake, sy=(Math.random()-.5)*shake;
  ctx.save(); ctx.translate(sx,sy);
  drawPanel(); drawPads(); drawBlocks();
  if(state!=='start') drawGhost();
  drawActivePiece();
  if(state!=='start') drawWorker();
  drawDanger(); drawShovePrompt();
  ctx.restore();
  drawPreview();
  drawParticles(); drawFloats(); drawBanner();
  const v=ctx.createRadialGradient(W/2,H/2,Math.min(W,H)*.35,W/2,H/2,Math.max(W,H)*.75);
  v.addColorStop(0,'rgba(0,0,0,0)'); v.addColorStop(1,'rgba(4,8,14,.45)');
  ctx.fillStyle=v; ctx.fillRect(0,0,W,H);
}
function frame(ts){
  requestAnimationFrame(frame);
  let dt=(ts-lastT)/1000; lastT=ts; if(dt>0.05) dt=0.05;
  now=ts;
  computeLayout();
  if(state!=='pause') update(dt);
  render();
  syncHUD();
}
