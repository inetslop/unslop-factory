const solidAt=(x,y)=>{const ix=x|0,iy=y|0;if(ix<0||iy<0||ix>=N||iy>=N)return true;return MAP[iy*N+ix]>0};
function tryMove(o,nx,ny,r){
  let s=nx>o.x?1:nx<o.x?-1:1;
  if(!solidAt(nx+s*r,o.y-r)&&!solidAt(nx+s*r,o.y+r))o.x=nx;
  s=ny>o.y?1:ny<o.y?-1:1;
  if(!solidAt(o.x-r,ny+s*r)&&!solidAt(o.x+r,ny+s*r))o.y=ny;
}
function castWall(px,py,dx,dy,maxD){
  let mx=px|0,my=py|0;
  const ddx=Math.abs(1/dx),ddy=Math.abs(1/dy);
  let stx,sdx,sty,sdy;
  if(dx<0){stx=-1;sdx=(px-mx)*ddx}else{stx=1;sdx=(mx+1-px)*ddx}
  if(dy<0){sty=-1;sdy=(py-my)*ddy}else{sty=1;sdy=(my+1-py)*ddy}
  for(let i=0;i<96;i++){
    let side;
    if(sdx<sdy){mx+=stx;sdx+=ddx;side=0}else{my+=sty;sdy+=ddy;side=1}
    if(MAP[my*N+mx]>0){const d=side===0?sdx-ddx:sdy-ddy;return{d,side}}
    if(Math.min(sdx,sdy)>maxD)break;
  }
  return null;
}

function update(dt){

  S.fireT=Math.max(0,S.fireT-dt);
  S.beamT=Math.max(0,S.beamT-dt);

  }else P.moving=Math.max(0,P.moving-4*dt);

  if(S.charging){S.charge=Math.min(1,S.charge+dts/.85);setCharge(S.charge)}

  /* enemies */
  for(const e of enemies){
    e.spawnT+=dt;e.flash=Math.max(0,e.flash-dt);e.lunge=Math.max(0,e.lunge-dt);
    const dx=P.x-e.x,dy=P.y-e.y,d=Math.hypot(dx,dy)||1e-4;

      let ang=Math.atan2(dy,dx)+Math.sin(S.t*3+e.seed)*.35*(los?1:0);
      tryMove(e,e.x+Math.cos(ang)*sp*dts,e.y+Math.sin(ang)*sp*dts,.3);
      if(d<1&&e.cd<=0&&e.spawnT>.4){e.cd=.7;e.lunge=.25;damagePlayer(8+(S.wave>>1))}
      else if(e.cd>0)e.cd-=dt;
    }else{
      let vx=0,vy=0;
      if(los){
        if(d<4.2){vx=-dx/d;vy=-dy/d}
        else if(d>8.5){vx=dx/d;vy=dy/d}
        else{const q=Math.sin(S.t*1.3+e.seed)>0?1:-1;vx=-dy/d*q;vy=dx/d*q}
      }else{vx=dx/d*.4;vy=dy/d*.4}
      tryMove(e,e.x+vx*e.spd*dts,e.y+vy*e.spd*dts,.32);
      if(los&&d<13&&e.cd<=0&&e.windup<=0)e.windup=.4;
      if(e.windup>0){e.windup-=dt;
        if(e.windup<=0){e.cd=e.cdr*(.8+Math.random()*.5);
          const bs=Math.min(9,6.5+S.wave*.12);
          bolts.push({x:e.x,y:e.y,vx:dx/d*bs,vy:dy/d*bs,dmg:e.dmg,life:3});sfx.bolt();}}
      else if(e.cd>0)e.cd-=dt;
    }
  }
  for(let i=0;i<enemies.length;i++)for(let j=i+1;j<enemies.length;j++){
    const a=enemies[i],b=enemies[j],dx=b.x-a.x,dy=b.y-a.y,d2=dx*dx+dy*dy;
    if(d2<.56&&d2>1e-4){const d=Math.sqrt(d2),push=(.74-d)*.5/d;
      tryMove(a,a.x-dx*push,a.y-dy*push,.3);tryMove(b,b.x+dx*push,b.y+dy*push,.3)}
  }

  /* bolts */
  for(let i=bolts.length-1;i>=0;i--){const b=bolts[i];
    b.x+=b.vx*dts;b.y+=b.vy*dts;b.life-=dts;
    let dead=b.life<=0||solidAt(b.x,b.y);
    const pdx=P.x-b.x,pdy=P.y-b.y;
    if(pdx*pdx+pdy*pdy<.12){damagePlayer(b.dmg);dead=true}
    if(dead){const s=toScreen(b.x,b.y);if(s)burst(s.x,H/2,4,'255,176,46',40);bolts.splice(i,1)}
  }

  /* pickups */
  for(let i=picks.length-1;i>=0;i--){const p=picks[i];p.t+=dts;
    const dx=P.x-p.x,dy=P.y-p.y,d=Math.hypot(dx,dy)||1e-4;
    if(d<1.6){p.x+=dx/d*2.5*dts;p.y+=dy/d*2.5*dts}
    if(d<.45){
      const s=toScreen(p.x,p.y);
      if(p.type==='med'){P.hp=Math.min(100,P.hp+35);if(s)pop(s.x,H/2,'+35 HULL','#49e6ff')}
      else{S.score+=15;if(s)pop(s.x,H/2,'+15','#ffd75e')}
      sfx.pickup();burst(W/2,H*.7,8,p.type==='med'?'53,224,255':'255,215,94',60);
      picks.splice(i,1);continue}
  }

  /* wave flow */
  if(S.deathT<0){
    if(S.pending>0){S.spawnT-=dt;if(S.spawnT<=0){S.spawnT=.55;spawnEnemy();S.pending--}}
    else if(enemies.length===0){
      if(!S.clearing){S.clearing=true;S.breakT=2.6;const bonus=50*S.wave;S.score+=bonus;
        banner('SECTOR CLEARED','BONUS +'+bonus);sfx.clear()}
      else{S.breakT-=dt;if(S.breakT<=0){S.clearing=false;startWave(S.wave+1)}}
    }
  }
  updParts(dt);
}

function render(){
  const w=cv.width,h=cv.height;
  const dirX=Math.cos(P.a),dirY=Math.sin(P.a);
  const planeX=-Math.sin(P.a)*.66,planeY=Math.cos(P.a)*.66;
  hor=clamp(H/2+S.kick*4+Math.sin(P.bob)*1.4*Math.min(1,P.moving),H*.2,H*.8);

  /* ceiling */
  const cg=bctx.createLinearGradient(0,0,0,hor);
  cg.addColorStop(0,'#04070c');cg.addColorStop(.75,'#0a1420');cg.addColorStop(1,'#16283e');
  bctx.fillStyle=cg;bctx.fillRect(0,0,W,hor+1);

  /* floor (per-pixel) */
  const y0=Math.max(0,Math.ceil(hor));
  const d=imgFloor.data,F=.94+.06*Math.sin(S.t*2.1);
  for(let y=y0;y<H;y++){
    const rd=(H*.5)/(y-hor);
    const A=Math.exp(-rd*.34),kA=A*Math.min(1,1.7/(1+rd*.85))*F,f=1-A;
    const fr=9*f,fg=16*f,fb=26*f;
    let wx=P.x+rd*(dirX-planeX),wz=P.y+rd*(dirY-planeY);
    const ax=2*rd*planeX/W,ay=2*rd*planeY/W;
    let o=y*W*4;
    for(let x=0;x<W;x++,o+=4){
      const ti=(Math.floor(wz)*64&63)*64+(Math.floor(wx)*64&63);
      d[o]=FLR.r[ti]*kA+fr;d[o+1]=FLR.g[ti]*kA+fg;d[o+2]=FLR.b[ti]*kA+fb;d[o+3]=255;
      wx+=ax;wz+=ay;
    }
  }
  bctx.putImageData(imgFloor,0,y0,0,0,W,H-y0);

  /* walls */
  const TEXC=[null,texA,texH,texO];
  for(let x=0;x<W;x++){
    const u=2*x/W-1,rayX=dirX+planeX*u,rayY=dirY+planeY*u;
    let mx=P.x|0,my=P.y|0;
    const ddx=Math.abs(1/rayX),ddy=Math.abs(1/rayY);
    let stx,sdx,sty,sdy;
    if(rayX<0){stx=-1;sdx=(P.x-mx)*ddx}else{stx=1;sdx=(mx+1-P.x)*ddx}
    if(rayY<0){sty=-1;sdy=(P.y-my)*ddy}else{sty=1;sdy=(my+1-P.y)*ddy}
    let side=0,cell=1;
    for(let i=0;i<96;i++){
      if(sdx<sdy){mx+=stx;sdx+=ddx;side=0}else{my+=sty;sdy+=ddy;side=1}
      cell=MAP[my*N+mx];if(cell)break;
    }
    const perp=side===0?sdx-ddx:sdy-ddy;zbuf[x]=perp;
    let hh=H/perp;if(hh>H*3)hh=H*3;
    const yS=hor-hh/2;
    let wtx=side===0?P.y+perp*rayY:P.x+perp*rayX;
    wtx-=Math.floor(wtx);
    let texX=wtx*64|0;
    if((side===0&&rayX>0)||(side===1&&rayY<0))texX=63-texX;
    const T=TEXC[cell]||texA;
    bctx.drawImage(side===1?T.d:T.c,texX,0,1,64,x,yS,1,hh);
    const fi=Math.min(47,(perp*47/FOG)|0);
    if(fi>0){bctx.fillStyle=fogSt[fi];bctx.fillRect(x,yS,1,hh)}
  }

  /* sprites */
  const invDet=1/(planeX*dirY-dirX*planeY),SL=[];
  function addSpr(x,y,img,hU,vc,alpha,lighter){
    const rx=x-P.x,ry=y-P.y;
    const tx=invDet*(dirY*rx-dirX*ry),ty=invDet*(-planeY*rx+planeX*ry);
    if(ty<=.08||ty>40)return null;
    const sxp=(W/2)*(1+tx/ty),sh=(H/ty)*hU;
    if(sxp<-sh||sxp>W+sh)return null;
    SL.push({ty,sxp,sh,yT:hor+vc*(H/ty)-sh/2,img,alpha,lighter});
    return{x:sxp,ty};
  }
  for(const e of enemies){
    const si=Math.min(1,e.spawnT/.35);
    let hU=e.hU*si,vc=e.vc;
    if(e.type===0&&e.lunge>0){hU*=1+e.lunge*.5;vc-=e.lunge*.22}
    if(e.type===1&&e.windup>0)vc-=e.windup*.04;
    const img=e.flash>0||(e.type===1&&e.windup>0&&Math.sin(S.t*40)>0)?e.wimg:e.img;
    const r=addSpr(e.x,e.y,img,hU,vc,si,false);
    if(r){e.sxp=r.x;e.ty=r.ty}
  }
  for(const b of bolts)addSpr(b.x,b.y,sprBolt,.34,.05,1,true);
  for(const p of picks)addSpr(p.x,p.y,p.type==='med'?sprMed:sprShard,.42,.17+Math.sin(p.t*3.2)*.05,1,false);
  SL.sort((a,b)=>b.ty-a.ty);
  for(const s of SL){
    bctx.globalAlpha=s.alpha;
    bctx.globalCompositeOperation=s.lighter?'lighter':'source-over';
    const x0s=s.sxp-s.sh/2;
    let L=Math.max(0,Math.ceil(x0s)),Rr=Math.min(W-1,Math.floor(x0s+s.sh)),run=-1;
    for(let x=L;x<=Rr;x++){
      if(zbuf[x]>s.ty){if(run<0)run=x}
      else{if(run>=0){const sw=(x-run)/s.sh*64;bctx.drawImage(s.img,((run-x0s)/s.sh)*64,0,sw,64,run,s.yT,x-run,s.sh);run=-1}}
    }
    if(run>=0){const len=Rr+1-run,sw=len/s.sh*64;bctx.drawImage(s.img,((run-x0s)/s.sh)*64,0,sw,64,run,s.yT,len,s.sh)}
  }
  bctx.globalAlpha=1;bctx.globalCompositeOperation='source-over';

  /* world particles */
  if(parts.length){
    bctx.globalCompositeOperation='lighter';
    for(const p of parts){const a=1-p.t/p.life;
      bctx.fillStyle='rgba('+p.col+','+(a*.9).toFixed(2)+')';
      const sz=p.s*(a+.3);bctx.fillRect(p.x-sz/2,p.y-sz/2,sz,sz)}
    bctx.globalCompositeOperation='source-over';
  }

  /* blit */
  ctx.setTransform(1,0,0,1,0,0);
  ctx.fillStyle='#000';ctx.fillRect(0,0,w,h);
  const shx=(Math.random()*2-1)*S.shake*6,shy=(Math.random()*2-1)*S.shake*5;
  ctx.imageSmoothingEnabled=false;
  ctx.drawImage(buf,shx,shy,w,h);

  if(S.mode==='playing'||S.mode==='paused'){drawGun(w,h);drawHUD(w,h)}
  ctx.drawImage(vigC,0,0);

  /* damage / low-hp overlays */
  if(S.dmg>.01){
    const g=ctx.createRadialGradient(w/2,h/2,Math.min(w,h)*.3,w/2,h/2,Math.max(w,h)*.75);
    g.addColorStop(0,'rgba(255,40,60,0)');g.addColorStop(1,'rgba(255,40,60,'+(S.dmg*.5).toFixed(3)+')');
    ctx.fillStyle=g;ctx.fillRect(0,0,w,h);
  }
  if(S.mode==='playing'&&P.hp<35&&P.hp>0){
    const pl=(Math.sin(S.t*5)+1)/2;
    const g=ctx.createRadialGradient(w/2,h/2,Math.min(w,h)*.4,w/2,h/2,Math.max(w,h)*.75);
    g.addColorStop(0,'rgba(255,30,60,0)');g.addColorStop(1,'rgba(255,30,60,'+(.1+.14*pl).toFixed(3)+')');
    ctx.fillStyle=g;ctx.fillRect(0,0,w,h);
  }
  if(S.mode==='playing'&&S.charge>=.85){
    const pl=(Math.sin(S.t*9)+1)/2;
    ctx.fillStyle='rgba(255,176,46,'+(pl*.05).toFixed(3)+')';ctx.fillRect(0,0,w,h);
  }
}
