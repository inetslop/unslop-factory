const COLS=15, ROWS=15;

let actx=null;
function initAudio(){ if(!actx){ try{ actx=new (window.AudioContext||window.webkitAudioContext)(); }catch(e){} } if(actx&&actx.state==='suspended') actx.resume(); }
function tone(f,d,type,vol,delay){
  if(!actx) return;
  const t0=actx.currentTime+(delay||0);
  const o=actx.createOscillator(), g=actx.createGain();
  o.type=type||'square'; o.frequency.setValueAtTime(f,t0);
  g.gain.setValueAtTime(vol||0.12,t0);
  g.gain.exponentialRampToValueAtTime(0.001,t0+d);
  o.connect(g); g.connect(actx.destination);
  o.start(t0); o.stop(t0+d+0.02);
}
function sfxReveal(n){ tone(300+n*90,0.07,'triangle',0.08); }
function sfxDie(){ tone(160,0.5,'sawtooth',0.2); tone(90,0.7,'square',0.18,0.08); }
function sfxWin(){ [523,659,784,1047].forEach((f,i)=>tone(f,0.18,'triangle',0.14,i*0.12)); }

/* ---------- setup ---------- */
function initBoard(){
  mines=new Set();
  revealed=new Array(COLS*ROWS).fill(false);
  const cx=Math.floor(COLS/2), cy=Math.floor(ROWS/2);
  for(let y=cy-1;y<=cy+1;y++)for(let x=cx-1;x<=cx+3;x++){
    if(x>=0&&x<COLS&&y>=0&&y<ROWS) mines.delete(y*COLS+x);
  }
  const want=Math.floor(COLS*ROWS*0.14);
  let guard=0;
  while(mines.size<want && guard++<5000){
    mines.add(Math.floor(Math.random()*COLS*ROWS));
  }
  safeTotal=COLS*ROWS-mines.size;
}
function countMines(x,y){
  let n=0;
  for(let dy=-1;dy<=1;dy++)for(let dx=-1;dx<=1;dx++){
    if(!dx&&!dy)continue;
    const nx=x+dx,ny=y+dy;
    if(nx>=0&&nx<COLS&&ny>=0&&ny<ROWS&&mines.has(ny*COLS+nx))n++;
  }
  return n;
}
function startGame(){
  initBoard();
  const cx=Math.floor(COLS/2), cy=Math.floor(ROWS/2);
  snake=[{x:cx,y:cy},{x:cx-1,y:cy},{x:cx-2,y:cy}];
  dir={x:1,y:0}; queue=[];
  for(const s of snake) revealed[s.y*COLS+s.x]=true;
  clearedCount=3; newCount=3;
  showMines=false; deadReason=''; paused=false;
  state='playing';
  $('startScr').classList.add('hidden');
  $('endScr').classList.add('hidden');
  $('pauseScr').classList.add('hidden');
  $('hud').classList.remove('hidden');
  $('hTotal').textContent=safeTotal;
  updateHud();
}
function endGame(win,reason){
  state=win?'win':'dead';
  if(!win) showMines=true;
  const got=clearedCount-3;
  if(got>best){ best=got; try{ localStorage.setItem('snakeMinesweeperBest',String(best)); }catch(e){} }
  $('hBest').textContent=best;
  setTimeout(()=>{
    if(state!=='playing'){
      $('endTitle').textContent = win ? 'FIELD CLEARED!' : (reason==='mine'?'💥 BOOM!':'CRASHED');
      $('endTitle').style.color = win ? '#4ade80' : '#f87171';
      $('endMsg').innerHTML = win
        ? 'You swept the whole minefield. Impressive.'
        : (reason==='mine'?'The snake stepped on a live mine.':
           reason==='self'?'The snake bit its own tail.':'The snake hit the wall.');
      $('endScore').textContent='Cleared '+got+' cells  ·  Best '+best;
      $('endScr').classList.remove('hidden');
    }
  }, win?400:700);
}
function updateHud(){ $('hScore').textContent=clearedCount; $('hBest').textContent=best; }

/* ---------- movement ---------- */
function queueDir(d){
  const last=queue.length?queue[queue.length-1]:dir;
  if((d.x===-last.x&&d.y===-last.y)||(d.x===last.x&&d.y===last.y))return;
  if(queue.length<3)queue.push(d);
}
function step(){
  if(queue.length)dir=queue.shift();
  const hx=snake[0].x+dir.x, hy=snake[0].y+dir.y;
  if(hx<0||hy<0||hx>=COLS||hy>=ROWS){ endGame(false,'wall'); return; }
  const idx=hy*COLS+hx;
  snake.unshift({x:hx,y:hy});
  if(mines.has(idx)){ endGame(false,'mine'); return; }
  for(let i=1;i<snake.length;i++){ if(snake[i].x===hx&&snake[i].y===hy){ endGame(false,'self'); return; } }
  const isNew=!revealed[idx];
  revealed[idx]=true;
  if(isNew){
    clearedCount++; newCount++;
    sfxReveal(countMines(hx,hy));
    updateHud();
    if(clearedCount>=safeTotal){ endGame(true); return; }
  }
  if(!(isNew && (newCount-3)%8===0)) snake.pop();
}
function tickMs(){ return Math.max(70, 150 - Math.floor((clearedCount-3)/12)*4); }

function resize(){
  const dpr=window.devicePixelRatio||1;
  cv.width=window.innerWidth*dpr; cv.height=window.innerHeight*dpr;

  ox=(window.innerWidth-cell*COLS)/2;
  oy=(window.innerHeight-cell*ROWS)/2+14;
}

function draw(){

  ctx.fillRect(0,0,window.innerWidth,window.innerHeight);
  if(state==='start')return;
  for(let y=0;y<ROWS;y++)for(let x=0;x<COLS;x++){
    const i=y*COLS+x, px=ox+x*cell, py=oy+y*cell;
    if(mines.has(i)&&showMines){
      ctx.fillStyle='#7f1d1d'; ctx.fillRect(px+1,py+1,cell-2,cell-2);
      drawMine(px+cell/2,py+cell/2,cell*0.3);
      continue;
    }
    if(revealed[i]){
      const n=countMines(x,y);
      ctx.fillStyle = n===0 ? '#10161f' : '#1a2332';
      ctx.fillRect(px+1,py+1,cell-2,cell-2);
      if(n>0){
        ctx.fillStyle=NUMC[n]||'#e5e7eb';
        ctx.font='bold '+Math.floor(cell*0.5)+'px Courier New';
        ctx.textAlign='center'; ctx.textBaseline='middle';
        ctx.fillText(n,px+cell/2,py+cell/2+1);
      }
    }else{
      ctx.fillStyle='#232d3f';
      ctx.fillRect(px+1,py+1,cell-2,cell-2);
      ctx.strokeStyle='rgba(255,255,255,0.06)';
      ctx.strokeRect(px+2.5,py+2.5,cell-5,cell-5);
    }
  }
  // snake
  for(let i=snake.length-1;i>=0;i--){
    const s=snake[i], px=ox+s.x*cell, py=oy+s.y*cell;
    if(i===0){
      ctx.fillStyle='#a3e635';
      roundRect(px+2,py+2,cell-4,cell-4,5); ctx.fill();
      // eyes

      eye(px+cell/2+ex*cell*0.18 - ey*cell*0.22, py+cell/2+ey*cell*0.18 + ex*cell*0.22, cell*0.09);
      eye(px+cell/2+ex*cell*0.18 + ey*cell*0.22, py+cell/2+ey*cell*0.18 - ex*cell*0.22, cell*0.09);
    }else{
      const t=i/snake.length;
      ctx.fillStyle='rgb('+Math.round(74-40*t)+','+Math.round(222-150*t)+','+Math.round(128-80*t)+')';

    }
  }
}
function eye(x,y,r){ ctx.beginPath(); ctx.arc(x,y,r,0,7); ctx.fill(); }
function drawMine(x,y,r){
  ctx.fillStyle='#111';
  ctx.beginPath(); ctx.arc(x,y,r,0,7); ctx.fill();
  ctx.strokeStyle='#ddd'; ctx.lineWidth=Math.max(2,r*0.22);
  for(let a=0;a<8;a++){
    const ang=a*Math.PI/4;
    ctx.beginPath();
    ctx.moveTo(x+Math.cos(ang)*r*0.5, y+Math.sin(ang)*r*0.5);
    ctx.lineTo(x+Math.cos(ang)*r*1.15, y+Math.sin(ang)*r*1.15);
    ctx.stroke();
  }
  ctx.fillStyle='#fff';
  ctx.beginPath(); ctx.arc(x-r*0.3,y-r*0.3,r*0.22,0,7); ctx.fill();
}
function roundRect(x,y,w,h,r){
  ctx.beginPath();
  ctx.moveTo(x+r,y);
  ctx.arcTo(x+w,y,x+w,y+h,r);
  ctx.arcTo(x+w,y+h,x,y+h,r);
  ctx.arcTo(x,y+h,x,y,r);
  ctx.arcTo(x,y,x+w,y,r);
  ctx.closePath();
}

    acc+=dt;
    while(acc>=tickMs()){ acc-=tickMs(); step(); if(state!=='playing')break; }
  }
  draw();
}
