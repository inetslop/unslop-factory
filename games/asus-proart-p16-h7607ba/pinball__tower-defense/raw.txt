          gems+=5;score+=10*(fast?2:1);
          boom(e.x,e.y,'#8aff6a',16);
          beep(1000,.2,'sawtooth',.15,200);
        }
      }
    }

    /* lost */
    if(b.y>H+40){
      balls.splice(i,1);
      boom(b.x,H-40,'#5ef2ff',10);
      beep(150,.3,'sawtooth',.15,60);
      setTimeout(()=>{if(state==='play')newBall();},900);
    }
  }

  /* towers fire */
  for(const t of towers){
    if(t.pulse>0)t.pulse-=dt;
    if(t.type!==1)continue;
    t.cool-=dt;
    let best=null,bd=380*380;
    for(const e of enemies){
      const d2=(e.x-t.x)**2+(e.y-t.y)**2;
      if(d2<bd){bd=d2;best=e;}
    }
    if(best){
      t.ang=Math.atan2(best.y-t.y,best.x-t.x);
      if(t.cool<=0){
        t.cool=.75;
        bullets.push({x:t.x+Math.cos(t.ang)*18,y:t.y+Math.sin(t.ang)*18,
          vx:Math.cos(t.ang)*620,vy:Math.sin(t.ang)*620,r:4});
        beep(1200,.05,'square',.07);
      }
    }else t.ang+=dt*1.5;
  }

  /* bullets */
  for(let i=bullets.length-1;i>=0;i--){
    const bl=bullets[i];
    bl.x+=bl.vx*dt;bl.y+=bl.vy*dt;
    let dead=(bl.x<-20||bl.x>W+20||bl.y<-20||bl.y>H+20);
    if(!dead)for(let j=enemies.length-1;j>=0;j--){
      const e=enemies[j];
      if(Math.hypot(bl.x-e.x,bl.y-e.y)<e.r+bl.r){
        e.hp--;dead=true;beep(950,.05,'square',.08);
        boom(bl.x,bl.y,'#ffd75e',4);
        if(e.hp<=0){enemies.splice(j,1);gems+=5;score+=20;boom(e.x,e.y,'#8aff6a',16);beep(1000,.18,'sawtooth',.13,200);}
        break;
      }
    }
    if(dead)bullets.splice(i,1);
  }

  /* particles */
  for(let i=particles.length-1;i>=0;i--){
    const p=particles[i];
    p.life-=dt;p.x+=p.vx*dt;p.y+=p.vy*dt;p.vy+=300*dt;
    if(p.life<=0)particles.splice(i,1);
  }
}

function gameOver(){
  state='over';
  document.getElementById('finalScore').innerHTML=
    'SCORE: '+score+' &nbsp;•&nbsp; WAVE: '+wave;
  document.getElementById('overScreen').classList.remove('hidden');
  beep(200,.8,'sawtooth',.3,40);
}

/* ---------- render ---------- */
function draw(){
  ctx.setTransform(dpr,0,0,dpr,0,0);
  ctx.fillStyle='#05060f';ctx.fillRect(0,0,cv.width/dpr,cv.height/dpr);
  ctx.setTransform(scale*dpr,0,0,scale*dpr,ox*dpr,oy*dpr);
  if(shake>0)ctx.translate((Math.random()-.5)*shake,(Math.random()-.5)*shake);

  /* playfield */
  ctx.fillStyle='#0a1230';ctx.fillRect(20,20,W-40,H-40);
  ctx.strokeStyle='#1e3a6e';ctx.lineWidth=4;ctx.strokeRect(20,20,W-40,H-40);

  /* faint grid */
  ctx.strokeStyle='rgba(60,110,200,.08)';ctx.lineWidth=1;
  for(let x=20;x<W;x+=50){ctx.beginPath();ctx.moveTo(x,20);ctx.lineTo(x,H-40);ctx.stroke();}
  for(let y=20;y<H;y+=50){ctx.beginPath();ctx.moveTo(20,y);ctx.lineTo(W-20,y);ctx.stroke();}

  /* base */
  ctx.fillStyle=(baseFlash>0)?'#7a1830':'#14263f';
  ctx.fillRect(20,H-40,W-40,40);
  const hpw=(W-40)*Math.max(0,hp)/100;
  ctx.fillStyle=hp>50?'#5ef28a':hp>25?'#ffd75e':'#ff5e7a';
  ctx.fillRect(20+(W-40-hpw)/2,H-30,hpw,20);
  ctx.strokeStyle='#5ef2ff';ctx.lineWidth=2;ctx.strokeRect(20+(W-40-hpw)/2,H-30,hpw,20);
  ctx.fillStyle='#cfe9ff';ctx.font='bold 16px Courier New';ctx.textAlign='center';
  ctx.fillText('B A S E   '+Math.max(0,Math.round(hp))+'%',W/2,H-14);

  /* towers */
  for(const t of towers){
    const T=TOWERS[t.type];
    let r=T.r+(t.pulse>0?t.pulse*25:0);
    ctx.beginPath();ctx.arc(t.x,t.y,r,0,7);
    ctx.fillStyle=t.type===0?'rgba(94,242,255,.15)':'rgba(255,215,94,.15)';
    ctx.fill();
    ctx.strokeStyle=T.color;ctx.lineWidth=3;ctx.stroke();
    if(t.type===0){
      ctx.beginPath();ctx.arc(t.x,t.y,r*.5,0,7);ctx.fillStyle=T.color;ctx.fill();
    }else{
      ctx.save();ctx.translate(t.x,t.y);ctx.rotate(t.ang);
      ctx.fillStyle=T.color;ctx.fillRect(0,-4,T.r+8,8);
      ctx.beginPath();ctx.arc(0,0,r*.55,0,7);ctx.fill();
      ctx.restore();
    }
  }

  /* enemies */
  for(const e of enemies){
    const pulse=1+Math.sin(e.ph*3)*.08;
    ctx.save();ctx.translate(e.x,e.y);ctx.scale(pulse,pulse);
    ctx.beginPath();
    for(let i=0;i<6;i++){
      const a=i/6*6.283+e.ph*.5;
      const rr=e.r*(i%2? .8:1);
      i===0?ctx.moveTo(Math.cos(a)*rr,Math.sin(a)*rr):ctx.lineTo(Math.cos(a)*rr,Math.sin(a)*rr);
    }
    ctx.closePath();
    ctx.fillStyle='#3a1450';ctx.fill();
    ctx.strokeStyle='#c86aff';ctx.lineWidth=3;ctx.stroke();
    ctx.beginPath();ctx.arc(0,0,e.r*.35,0,7);ctx.fillStyle='#ff5ef2';ctx.fill();
    ctx.restore();
    /* hp bar */
    if(e.hp<e.maxhp){
      const w=e.r*2;
      ctx.fillStyle='#311';ctx.fillRect(e.x-w/2,e.y-e.r-9,w,4);
      ctx.fillStyle='#8aff6a';ctx.fillRect(e.x-w/2,e.y-e.r-9,w*e.hp/e.maxhp,4);
    }
  }

  /* bullets */
  for(const bl of bullets){
    ctx.beginPath();ctx.arc(bl.x,bl.y,bl.r,0,7);
    ctx.fillStyle='#ffd75e';ctx.fill();
    ctx.strokeStyle='rgba(255,215,94,.4)';ctx.lineWidth=3;
    ctx.beginPath();ctx.moveTo(bl.x,bl.y);ctx.lineTo(bl.x-bl.vx*.02,bl.y-bl.vy*.02);ctx.stroke();
  }

  /* particles */
  for(const p of particles){
    ctx.globalAlpha=Math.max(0,p.life/p.max);
    ctx.beginPath();ctx.arc(p.x,p.y,p.r,0,7);ctx.fillStyle=p.color;ctx.fill();
  }
  ctx.globalAlpha=1;

  /* flippers */
  for(const f of FL){
    const tx=f.px+f.L*Math.cos(f.a),ty=f.py+f.L*Math.sin(f.a);
    ctx.lineCap='round';
    ctx.strokeStyle='#0a2a3f';ctx.lineWidth=36;
    ctx.beginPath();ctx.moveTo(f.px,f.py);ctx.lineTo(tx,ty);ctx.stroke();
    ctx.strokeStyle='#5ef2ff';ctx.lineWidth=28;
    ctx.beginPath();ctx.moveTo(f.px,f.py);ctx.lineTo(tx,ty);ctx.stroke();
    ctx.strokeStyle='#bffaff';ctx.lineWidth=10;
    ctx.beginPath();ctx.moveTo(f.px,f.py);ctx.lineTo(tx,ty);ctx.stroke();
  }

  /* balls */
  for(const b of balls){
    if(b.waiting){
      ctx.fillStyle='rgba(94,242,255,'+(0.6+Math.sin(time*6)*.3)+')';
      ctx.font='bold 18px Courier New';ctx.textAlign='center';
      ctx.fillText('SPACE &#11015;'.replace('&#11015;','▲'),b.x,b.y-24);
    }
    const g=ctx.createRadialGradient(b.x-4,b.y-4,2,b.x,b.y,b.r);
    g.addColorStop(0,'#ffffff');g.addColorStop(1,'#5ef2ff');
    ctx.beginPath();ctx.arc(b.x,b.y,b.r,0,7);
    ctx.fillStyle=g;ctx.fill();
    ctx.strokeStyle='#bffaff';ctx.lineWidth=2;ctx.stroke();
  }

  /* HUD */
  ctx.textAlign='left';ctx.font='bold 20px Courier New';
  ctx.fillStyle='#5ef2ff';ctx.fillText('♥ '+Math.max(0,Math.round(hp)),34,H/2-160+ (H>1200?0:0)+ (H===1200?-580:0));
  /* simpler fixed HUD at top */
  ctx.fillStyle='rgba(5,8,22,.8)';ctx.fillRect(20,20,W-40,54);
  ctx.font='bold 22px Courier New';
  ctx.textAlign='left';
  ctx.fillStyle=hp>50?'#5ef28a':hp>25?'#ffd75e':'#ff5e7a';
  ctx.fillText('♥'+Math.max(0,Math.round(hp)),36,54);
  ctx.fillStyle='#ffd75e';ctx.fillText('◆ '+gems,140,54);
  ctx.textAlign='center';ctx.fillStyle='#cfe9ff';
  ctx.fillText('WAVE '+wave,W/2,54);
  ctx.textAlign='right';ctx.fillStyle='#8aff6a';
  ctx.fillText('SCORE '+score,W-36,54);

  /* tower select */
  for(let i=0;i<2;i++){
    const T=TOWERS[i],x=W-190+i*95;
    const sel=(i===selTower),afford=gems>=T.cost;
    ctx.fillStyle=sel?(afford?'rgba(94,242,255,.2)':'rgba(255,94,122,.15)'):'rgba(30,58,110,.4)';
    ctx.fillRect(x,H-110,86,44);
    ctx.strokeStyle=sel?T.color:'#1e3a6e';ctx.lineWidth=2;ctx.strokeRect(x,H-110,86,44);
    ctx.textAlign='center';
    ctx.fillStyle=afford?T.color:'#5a6a8a';ctx.font='bold 15px Courier New';
    ctx.fillText((i+1)+' '+T.name,x+43,H-92);
    ctx.fillStyle=afford?'#fff':'#ff5e7a';ctx.font='14px Courier New';
    ctx.fillText(T.cost+' gems',x+43,H-74);
  }

  if(state==='paused'){
    ctx.fillStyle='rgba(4,6,18,.7)';ctx.fillRect(20,20,W-40,H-40);
    ctx.fillStyle='#5ef2ff';ctx.font='bold 44px Courier New';ctx.textAlign='center';
    ctx.fillText('PAUSED',W/2,H/2);
    ctx.font='20px Courier New';ctx.fillStyle='#9fd4ff';
    ctx.fillText('press P to resume',W/2,H/2+40);
  }
}

let last=performance.now();
function loop(now){
  let dt=Math.min(.033,(now-last)/1000);last=now;
  if(state==='play')update(dt);
  else if(state!=='start'){/* keep drawing frozen scene */}
  draw();
  requestAnimationFrame(loop);
}
