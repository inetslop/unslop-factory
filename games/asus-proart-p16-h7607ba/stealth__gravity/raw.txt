      const lv = LEVELS[S.levelIdx];
      if (px < 0 || px >= lv.cols || py < 0 || py >= lv.rows.length) return true;
      if (lv.rows[py][px] === '#') return true;
    }
  return false;
}

function moveEntity(e, dt){
  // X axis
  let nx = e.x + e.vx * dt;
  if (!rectHitsSolid(nx, e.y, e.w, e.h)) e.x = nx;
  else {
    const step = Math.sign(e.vx);
    while (!rectHitsSolid(e.x + step, e.y, e.w, e.h) && Math.abs(step) < T) e.x += step;
    e.vx = 0;
  }
  // Y axis
  let ny = e.y + e.vy * dt;
  if (!rectHitsSolid(e.x, ny, e.w, e.h)) e.y = ny;
  else {
    const step = Math.sign(e.vy);
    while (!rectHitsSolid(e.x, e.y + step, e.w, e.h) && Math.abs(step) < T) e.y += step;
    e.vy = 0;
  }
}

function aabb(a, b){
  return a.x < b.x + b.w && a.x + a.w > b.x && a.y < b.y + b.h && a.y + a.h > b.y;
}

function guardBody(g){
  if (g.up) return { x: g.x - 13, y: g.y, w: 26, h: 30 };
  return { x: g.x - 13, y: g.y - 30, w: 26, h: 30 };
}
function guardEye(g){
  const ex = g.x + g.dir * 5;
  return g.up ? { x: ex, y: g.y + 12 } : { x: ex, y: g.y - 22 };
}

function losClear(x0, y0, x1, y1){
  const dx = x1 - x0, dy = y1 - y0;
  const dist = Math.hypot(dx, dy);
  const steps = Math.ceil(dist / 8);
  for (let i = 1; i < steps; i++){
    const t = i / steps;
    if (solidAt(x0 + dx * t, y0 + dy * t)) return false;
  }
  return true;
}

function laserOn(l, time){
  const ph = (time + l.offset) % l.period;
  return ph < l.onTime;
}
function laserRect(l){
  const x = (l.col + .5) * T;
  return { x: x - 4, y: l.r1 * T, w: 8, h: (l.r2 - l.r1 + 1) * T };
}

/* ---------------- Update ---------------- */
function update(dt){
  S.time += dt;
  S.shake = Math.max(0, S.shake - dt * 3);

  if (S.mode === 'caught'){
    S.stateT -= dt;
    if (S.stateT <= 0){
      S.lives--;
      if (S.lives <= 0){ S.mode = 'gameover'; }
      else { loadLevel(S.levelIdx); S.mode = 'play'; S.stateT = .8; }
    }
    return;
  }
  if (S.mode === 'clear'){
    S.stateT -= dt;
    if (S.stateT <= 0){
      if (S.levelIdx + 1 < LEVELS.length){ loadLevel(S.levelIdx + 1); S.mode = 'play'; S.stateT = 1.4; }
      else { S.mode = 'win'; sfx.win(); }
    }
    return;
  }
  if (S.mode !== 'play') return;

  if (S.stateT > 0) S.stateT -= dt;

  const p = S.player;
  // movement
  const left = keys['a'] || keys['arrowleft'];
  const right = keys['d'] || keys['arrowright'];
  const target = (right ? 1 : 0) - (left ? 1 : 0);
  const acc = 2200;
  if (target !== 0){
    p.vx += target * acc * dt;
    p.vx = Math.max(-250, Math.min(250, p.vx));
  } else {
    p.vx -= Math.sign(p.vx) * Math.min(Math.abs(p.vx), 2600 * dt);
  }
  p.vy += p.g * 1500 * dt;
  p.vy = Math.max(-620, Math.min(620, p.vy));
  moveEntity(p, dt);

  // guards
  let seenAny = false;
  for (const g of S.guards){
    g.x += g.dir * g.speed * dt;
    if (g.x < g.minX){ g.x = g.minX; g.dir = 1; }
    if (g.x > g.maxX){ g.x = g.maxX; g.dir = -1; }

    const eye = guardEye(g);
    const pcx = p.x + p.w / 2, pcy = p.y + p.h / 2;
    const dx = pcx - eye.x, dy = pcy - eye.y;
    const dist = Math.hypot(dx, dy);
    const range = 250;
    let seen = false;
    if (dist < range){
      let ang = Math.atan2(dy, dx);
      const base = g.dir > 0 ? 0 : Math.PI;
      let diff = ang - base;
      while (diff > Math.PI) diff -= 2 * Math.PI;
      while (diff < -Math.PI) diff += 2 * Math.PI;
      if (Math.abs(diff) < 0.95 && losClear(eye.x, eye.y, pcx, pcy)) seen = true;
    }
    g.seen = seen;
    if (seen) seenAny = true;

    // contact
    if (aabb(p, guardBody(g))) { catchPlayer(); return; }
  }

  // lasers
  for (const l of S.lasers){
    if (laserOn(l, S.time) && aabb(p, laserRect(l))){ catchPlayer(); return; }
  }

  // detection meter
  if (seenAny){
    S.meter = Math.min(1, S.meter + 1.35 * dt);
    if (S.meter >= 0.45 && !S.alertFired){ S.alertFired = true; sfx.alert(); }
    if (S.meter >= 1){ catchPlayer(); return; }
  } else {
    S.meter = Math.max(0, S.meter - 1.9 * dt);
    if (S.meter < 0.45) S.alertFired = false;
  }

  // exit
  const er = { x: S.exit.x - 20, y: S.exit.y - 20, w: 40, h: 40 };
  if (aabb(p, er)){
    S.mode = 'clear'; S.stateT = 1.3; sfx.clear();
    for (let i=0;i<30;i++){
      S.particles.push({x:S.exit.x, y:S.exit.y, vx:(Math.random()-.5)*380, vy:(Math.random()-.5)*380, life:.9, max:.9, c:'120,255,180'});
    }
  }

  // particles
  for (let i = S.particles.length - 1; i >= 0; i--){
    const pt = S.particles[i];
    pt.life -= dt; pt.x += pt.vx * dt; pt.y += pt.vy * dt;
    if (pt.life <= 0) S.particles.splice(i, 1);
  }

  // camera
  let cx = p.x + p.w/2 - W/2, cy = p.y + p.h/2 - H/2;
  S.camX = S.levelW < W ? (S.levelW - W)/2 : Math.max(0, Math.min(S.levelW - W, cx));
  S.camY = S.levelH < H ? (S.levelH - H)/2 : Math.max(0, Math.min(S.levelH - H, cy));
}

function catchPlayer(){
  sfx.caught();
  S.shake = 1;
  const p = S.player;
  for (let i=0;i<40;i++){
    S.particles.push({x:p.x+p.w/2, y:p.y+p.h/2, vx:(Math.random()-.5)*420, vy:(Math.random()-.5)*420, life:.8, max:.8, c:'255,90,90'});
  }
  S.mode = 'caught'; S.stateT = 1.1;
}

/* ---------------- Drawing ---------------- */
function drawBackground(){
  const grd = ctx.createLinearGradient(0, 0, 0, H);
  grd.addColorStop(0, '#070b18');
  grd.addColorStop(1, '#0b1226');
  ctx.fillStyle = grd;
  ctx.fillRect(0, 0, W, H);
}

function drawLevel(){
  const lv = LEVELS[S.levelIdx];
  const c0 = Math.max(0, Math.floor(S.camX / T) - 1);
  const c1 = Math.min(lv.cols - 1, Math.ceil((S.camX + W) / T) + 1);
  const r0 = Math.max(0, Math.floor(S.camY / T) - 1);
  const r1 = Math.min(lv.rows.length - 1, Math.ceil((S.camY + H) / T) + 1);

  // grid
  ctx.strokeStyle = 'rgba(80,120,220,0.06)';
  ctx.lineWidth = 1;
  ctx.beginPath();
  for (let c = c0; c <= c1; c++){ ctx.moveTo(c*T - S.camX, r0*T - S.camY); ctx.lineTo(c*T - S.camX, r1*T - S.camY); }
  for (let r = r0; r <= r1; r++){ ctx.moveTo(c0*T - S.camX, r*T - S.camY); ctx.lineTo(c1*T - S.camX, r*T - S.camY); }
  ctx.stroke();

  // blocks
  for (let r = r0; r <= r1; r++){
    for (let c = c0; c <= c1; c++){
      if (lv.rows[r][c] !== '#') continue;
      const x = c*T - S.camX, y = r*T - S.camY;
      ctx.fillStyle = '#141b33';
      ctx.fillRect(x, y, T, T);
      ctx.fillStyle = '#1d2748';
      ctx.fillRect(x+1, y+1, T-2, T-2);
      // edge highlights facing open space
      ctx.fillStyle = 'rgba(90,160,255,0.35)';
      if (r === 0 || lv.rows[r-1][c] !== '#') ctx.fillRect(x, y, T, 3);
      if (r === lv.rows.length-1 || lv.rows[r+1][c] !== '#') ctx.fillRect(x, y+T-3, T, 3);
    }
  }
}

function drawExit(t){
  const x = S.exit.x - S.camX, y = S.exit.y - S.camY;
  const pulse = 1 + Math.sin(t * 4) * 0.12;
  ctx.save();
  ctx.translate(x, y);
  ctx.shadowColor = 'rgba(120,255,180,0.9)';
  ctx.shadowBlur = 22;
  ctx.strokeStyle = '#7dffc4';
  ctx.lineWidth = 3;
  ctx.beginPath();
  ctx.arc(0, 0, 16 * pulse, 0, Math.PI * 2);
  ctx.stroke();
  ctx.strokeStyle = 'rgba(125,255,196,0.4)';
  ctx.lineWidth = 1.5;
  ctx.beginPath();
  ctx.arc(0, 0, 22 * pulse + Math.sin(t*7)*3, 0, Math.PI*2);
  ctx.stroke();
  ctx.fillStyle = 'rgba(125,255,196,0.25)';
  ctx.beginPath(); ctx.arc(0,0,14*pulse,0,Math.PI*2); ctx.fill();
  ctx.restore();
}

function drawLasers(t){
  for (const l of S.lasers){
    const x = (l.col + .5) * T - S.camX;
    const y1 = l.r1 * T - S.camY, y2 = (l.r2 + 1) * T - S.camY;
    // emitters
    ctx.fillStyle = '#3a2030';
    ctx.fillRect(x - 7, y1 - 8, 14, 8);
    ctx.fillRect(x - 7, y2, 14, 8);
    ctx.fillStyle = laserOn(l, t) ? '#ff5566' : '#663344';
    ctx.fillRect(x - 4, y1 - 6, 8, 4);
    ctx.fillRect(x - 4, y2 + 2, 8, 4);

    if (laserOn(l, t)){
      ctx.save();
      ctx.shadowColor = 'rgba(255,60,80,0.9)';
      ctx.shadowBlur = 16;
      const g = ctx.createLinearGradient(x-4, 0, x+4, 0);
      g.addColorStop(0, 'rgba(255,60,80,0.15)');
      g.addColorStop(.5, 'rgba(255,120,140,0.95)');
      g.addColorStop(1, 'rgba(255,60,80,0.15)');
      ctx.fillStyle = g;
      ctx.fillRect(x - 4, y1, 8, y2 - y1);
      ctx.fillStyle = 'rgba(255,230,235,0.9)';
      ctx.fillRect(x - 1, y1, 2, y2 - y1);
      ctx.restore();
    } else {
      // faint warning line
      ctx.strokeStyle = 'rgba(255,80,100,0.12)';
      ctx.setLineDash([4, 8]);
      ctx.beginPath(); ctx.moveTo(x, y1); ctx.lineTo(x, y2); ctx.stroke();
      ctx.setLineDash([]);
    }
  }
}

function drawGuards(){
  for (const g of S.guards){
    const eye = guardEye(g);
    const ex = eye.x - S.camX, ey = eye.y - S.camY;
    const base = g.dir > 0 ? 0 : Math.PI;
    const range = 250;

    // vision cone
    ctx.save();
    ctx.globalAlpha = g.seen ? 0.30 : 0.14;
    const col = S.meter > 0.01 ? '255,' + Math.floor(180 - S.meter*120) + ',' + Math.floor(160 - S.meter*140) : '255,210,90';
    const grd = ctx.createRadialGradient(ex, ey, 10, ex, ey, range);
    grd.addColorStop(0, 'rgba(' + col + ',0.8)');
    grd.addColorStop(1, 'rgba(' + col + ',0)');
    ctx.fillStyle = grd;
    ctx.beginPath();
    ctx.moveTo(ex, ey);
    ctx.arc(ex, ey, range, base - 0.95, base + 0.95);
    ctx.closePath();
    ctx.fill();
    ctx.restore();

    // body
    const b = guardBody(g);
    const bx = b.x - S.camX, by = b.y - S.camY;
    ctx.fillStyle = '#2a1f3d';
    ctx.beginPath();
    ctx.roundRect(bx, by, b.w, b.h, 6);
    ctx.fill();
    ctx.strokeStyle = 'rgba(180,140,255,0.5)';
    ctx.lineWidth = 1.5;
    ctx.stroke();

    // visor / eye
    ctx.save();
    ctx.shadowColor = g.seen ? '#ff4455' : '#ffd75e';
    ctx.shadowBlur = 10;
    ctx.fillStyle = g.seen ? '#ff4455' : '#ffd75e';
    ctx.beginPath();
    ctx.arc(ex, ey, 4.5, 0, Math.PI * 2);
    ctx.fill();
    ctx.restore();

    // alert marker
    if (g.seen){
      ctx.fillStyle = '#ff5566';
      ctx.font = 'bold 18px monospace';
      ctx.textAlign = 'center';
      const ay = g.up ? by + b.h + 18 : by - 8;
      ctx.fillText('!', g.x - S.camX, ay);
    }
  }
}

function drawPlayer(){
  const p = S.player;
  const x = p.x - S.camX, y = p.y - S.camY;
  ctx.save();
  ctx.shadowColor = 'rgba(90,220,255,0.9)';
  ctx.shadowBlur = 16;
  // cloak body
  const grd = ctx.createLinearGradient(x, y, x + p.w, y + p.h);
  grd.addColorStop(0, '#bff4ff');
  grd.addColorStop(1, '#3aa8e0');
  ctx.fillStyle = grd;
  ctx.beginPath();
  ctx.roundRect(x, y, p.w, p.h, 9);
  ctx.fill();
  ctx.restore();

  // gravity indicator arrow
  ctx.strokeStyle = 'rgba(6,30,50,0.8)';
  ctx.lineWidth = 2;
  const ax = x + p.w/2, ay = y + p.h/2;
  const d = p.g * 7;
  ctx.beginPath();
  ctx.moveTo(ax - 4, ay - d);
  ctx.lineTo(ax, ay + d*0.6);
  ctx.lineTo(ax + 4, ay - d);
  ctx.stroke();

  // cloak tip flicker (opposite of gravity)
  const tip = y + p.h/2 - p.g * p.h * 0.5;
  ctx.fillStyle = 'rgba(180,240,255,0.7)';
  for (let i = -1; i <= 1; i++){
    const fl = Math.sin(S.time * 18 + i * 2) * 3;
    ctx.beginPath();
    ctx.moveTo(ax + i*6 - 2, tip);
    ctx.lineTo(ax + i*6 + 2, tip);
    ctx.lineTo(ax + i*6 + fl, tip - p.g * 7);
    ctx.closePath();
    ctx.fill();
  }
}

  for (const pt of S.particles){
    const a = pt.life / pt.max;
    ctx.fillStyle = 'rgba(' + pt.c + ',' + a.toFixed(2) + ')';
    ctx.fillRect(pt.x - S.camX - 2, pt.y - S.camY - 2, 4, 4);
  }
}

/* ---------------- Loop ---------------- */
let last = performance.now();
function frame(now){
  let dt = (now - last) / 1000;
  last = now;
  if (dt > 0.05) dt = 0.05;
  update(dt);
  render();
  requestAnimationFrame(frame);
}

  CanvasRenderingContext2D.prototype.roundRect = function(x, y, w, h, r){
    if (typeof r === 'number') r = [r,r,r,r];

    this.arcTo(x + w, y, x + w, y + h, r[1]);
    this.arcTo(x + w, y + h, x, y + h, r[2]);
    this.arcTo(x, y + h, x, y, r[3]);
    this.arcTo(x, y, x + w, y, r[0]);
    this.closePath();
  };
