function update(dt){
  G.time+=dt;laserCd=Math.max(0,laserCd-dt);
  // particles always animate
  for(let i=particles.length-1;i>=0;i--){const p=particles[i];p.life-=dt;
    if(p.life<=0){particles.splice(i,1);continue;}
    p.x+=p.vx*dt;p.y+=p.vy*dt;p.vx*=0.985;p.vy*=0.985;}
  if(G.state!=='playing')return;

  if(charging)power=Math.min(1,power+dt/1.1);

  // ball
  let sp=Math.hypot(ball.vx,ball.vy);
  if(sp>0){
    ball.x+=ball.vx*dt;ball.y+=ball.vy*dt;
    const f=Math.pow(0.45,dt);ball.vx*=f;ball.vy*=f;
    if(ball.x<ball.r){ball.x=ball.r;ball.vx=-ball.vx*0.75;tick();}
    if(ball.x>W-ball.r){ball.x=W-ball.r;ball.vx=-ball.vx*0.75;tick();}
    if(ball.y<ball.r){ball.y=ball.r;ball.vy=-ball.vy*0.75;tick();}
    if(ball.y>H-ball.r){ball.y=H-ball.r;ball.vy=-ball.vy*0.75;tick();}
    // hole suction + capture
    const hd=Math.hypot(ball.x-hole.x,ball.y-hole.y);sp=Math.hypot(ball.vx,ball.vy);
    if(hd<hole.r*1.6&&sp<330){ball.vx+=(hole.x-ball.x)*dt*4;ball.vy+=(hole.y-ball.y)*dt*4;}
    if(hd<hole.r-4&&Math.hypot(ball.vx,ball.vy)<300){sinkHole();return;}
    // asteroid collisions
    for(const a of asteroids){
      const dx=ball.x-a.x,dy=ball.y-a.y,d=Math.hypot(dx,dy),minD=ball.r+a.r*0.82;
      if(d<minD&&d>0.001){
        const nx=dx/d,ny=dy/d;
        ball.x=a.x+nx*minD;ball.y=a.y+ny*minD;
        const dot=ball.vx*nx+ball.vy*ny;
        if(dot<0){ball.vx-=2*dot*nx;ball.vy-=2*dot*ny;ball.vx*=0.85;ball.vy*=0.85;}
        ball.vx+=a.vx*0.15;ball.vy+=a.vy*0.15;tick();
      }
    }
    if(Math.hypot(ball.vx,ball.vy)<4){
      ball.vx=0;ball.vy=0;
      if(G.strokes>=G.maxStrokes)gameOver();
    }
  }else{
    if(G.strokes>=G.maxStrokes)gameOver();
  }

  // asteroids
  for(const a of asteroids){
    a.x+=a.vx*dt;a.y+=a.vy*dt;a.rot+=a.rs*dt;
    if(a.x<a.r){a.x=a.r;a.vx=Math.abs(a.vx);}
    if(a.x>W-a.r){a.x=W-a.r;a.vx=-Math.abs(a.vx);}
    if(a.y<a.r){a.y=a.r;a.vy=Math.abs(a.vy);}
    if(a.y>H-a.r){a.y=H-a.r;a.vy=-Math.abs(a.vy);}
  }

  // lasers
  for(let i=lasers.length-1;i>=0;i--){
    const l=lasers[i];l.x+=l.vx*dt;l.y+=l.vy*dt;l.life-=dt;
    let dead=l.life<=0||l.x<-20||l.x>W+20||l.y<-20||l.y>H+20;
    if(!dead){
      for(let j=asteroids.length-1;j>=0;j--){
        const a=asteroids[j];
        if(Math.hypot(l.x-a.x,l.y-a.y)<a.r*0.85){destroyAst(j);dead=true;break;}
      }
    }
    if(dead)lasers.splice(i,1);
  }
}
