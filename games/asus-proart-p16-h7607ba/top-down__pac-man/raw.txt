}
function homing(e,t,dist){
  const dx=t.x-e.x,dy=t.y-e.y,L=Math.hypot(dx,dy);
  if(L<=dist){e.x=t.x;e.y=t.y;return true;}
  e.x+=dx/L*dist;e.y+=dy/L*dist;return false;
}

/* ---------- game flow ---------- */
function newGame(){
  loadMaze();buildWalls();
  score=0;lives=3;paused=false;
  resetActors();
  state='ready';readyT=1.6;
  hideOverlay();
}
function endGame(win){
  state=win?'win':'over';
  if(score>hiScore)hiScore=score;
  try{localStorage.setItem('pacmaze_hi',String(hiScore));}catch(e){}
  showOverlay(
    '<h1>'+(win?'YOU WIN!':'GAME OVER')+'</h1>'+
    '<p>Final score: <b style="color:#ffe135">'+score+'</b><br>Best: '+hiScore+'</p>'+
    '<button id="goBtn" class="btn">PLAY AGAIN &nbsp;(R)</button>');
  document.getElementById('goBtn').onclick=()=>{initAudio();newGame();};
}

/* ---------- update ---------- */
function eatAt(c,r){
  const ch=grid[r][c];
  if(ch==='.'){grid[r][c]=' ';score+=10;pellets--;sfxWaka();checkWin();}
  else if(ch==='o'){grid[r][c]=' ';score+=50;pellets--;powerUp();checkWin();}
}
function checkWin(){if(pellets<=0&&state==='playing')endGame(true);}
function powerUp(){
  frightTimer=6;combo=0;sfxPower();
  for(const g of ghosts){
    if(g.mode!=='eyes'&&g.mode!=='entering'){
      g.frightened=true;
      if(g.mode==='normal'&&g.dir)g.dir={x:-g.dir.x,y:-g.dir.y};
    }
  }
}
function updateGhost(g,dt){
  switch(g.mode){
    case 'inhouse':
      if(homing(g,{x:9.5,y:8.5},HOUSE_SPEED*dt))g.mode='exiting';break;
    case 'exiting':
      if(homing(g,{x:9.5,y:6.5},HOUSE_SPEED*dt)){g.mode='normal';g.dir={x:-1,y:0};g.canHouse=false;}break;
    case 'eyes':
      if(homing(g,{x:9.5,y:7.5},EYES_SPEED*dt))g.mode='entering';break;
    case 'entering':
      if(homing(g,{x:9.5,y:8.5},EYES_SPEED*dt)){g.mode='inhouse';g.frightened=false;}break;
    default:
      stepEntity(g,(g.frightened?FRIGHT_SPEED:GHOST_SPEED)*dt);
  }
}
function update(dt){
  time+=dt;
  if(paused||state==='menu')return;
  if(state==='ready'){readyT-=dt;if(readyT<=0)state='playing';return;}
  if(state==='dying'){
    deathT-=dt;
    if(deathT<=0){
      lives--;
      if(lives<=0)endGame(false);
      else{resetActors();state='ready';readyT=1.6;}
    }
    return;
  }
  if(state!=='playing')return;

  modeT-=dt;
  if(modeT<=0){
    mi=(mi+1)%MODES.length;modeT=MODES[mi][1];
    for(const g of ghosts)if(g.mode==='normal'&&g.dir&&!g.frightened)g.dir={x:-g.dir.x,y:-g.dir.y};
  }
  if(frightTimer>0){frightTimer-=dt;if(frightTimer<=0)ghosts.forEach(g=>g.frightened=false);}

  if(pac.dir===null&&pac.wantDir&&pacOpen(Math.floor(pac.x)+pac.wantDir.x,Math.floor(pac.y)+pac.wantDir.y))
    pac.dir={x:pac.wantDir.x,y:pac.wantDir.y};
  if(pac.dir){pac.face=pac.dir;stepEntity(pac,PAC_SPEED*dt);}

  const c=Math.floor(((pac.x%COLS)+COLS)%COLS),r=Math.floor(pac.y);
  if(c!==lastCell.c||r!==lastCell.r){lastCell={c,r};eatAt(c,r);}

  for(const g of ghosts)updateGhost(g,dt);

  for(const g of ghosts){
    if(g.mode!=='normal')continue;
    const d=Math.hypot(g.x-pac.x,g.y-pac.y);
    if(d<0.55){
      if(g.frightened){
        g.frightened=false;g.mode='eyes';g.canHouse=true;
        score+=100<<combo;combo++;sfxGhost();
      }else{
        state='dying';deathT=1.4;sfxDeath();break;
      }
    }
  }
}

/* ---------- drawing ---------- */
function drawPacShape(cx,cy,face,open,scale){
  const x=ox+cx*tile,y=oy+cy*tile,r=tile*0.42*scale;
  if(r<=1)return;
  const a=Math.atan2(face.y,face.x);
  ctx.fillStyle='#ffe135';
  ctx.beginPath();ctx.moveTo(x,y);
  ctx.arc(x,y,r,a+open,a-open,false);
  ctx.closePath();ctx.fill();
}
function drawEyesOnly(x,y,r,g){
  for(const s of[-1,1]){
    ctx.fillStyle='#fff';
    ctx.beginPath();ctx.ellipse(x+s*r*0.32,y-r*0.12,r*0.3,r*0.36,0,0,7);ctx.fill();
    ctx.fillStyle='#1b2cff';
    ctx.beginPath();ctx.arc(x+s*r*0.32+g.dir.x*r*0.14,y-r*0.12+g.dir.y*r*0.14,r*0.15,0,7);ctx.fill();
  }
}
function drawGhost(g){
  const x=ox+g.x*tile,y=oy+g.y*tile,r=tile*0.42;
  if(g.mode==='eyes'){drawEyesOnly(x,y,r,g);return;}
  let body=g.color;
  if(g.frightened)body=(frightTimer<2&&Math.floor(time*6)%2===0)?'#e8ecff':'#2135d6';
  ctx.fillStyle=body;
  ctx.beginPath();
  ctx.arc(x,y,r,Math.PI,0);
  const feet=6;
  for(let i=1;i<=feet;i++){
    const fx=x+r-2*r*i/feet,fy=y+(i%2===0?r*0.7:r);
    ctx.lineTo(fx,fy);
  }
  ctx.closePath();ctx.fill();
  if(g.frightened){
    ctx.fillStyle='#ffc3b8';
    for(const s of[-1,1]){ctx.beginPath();ctx.arc(x+s*r*0.3,y-r*0.12,r*0.14,0,7);ctx.fill();}
    ctx.strokeStyle='#ffc3b8';ctx.lineWidth=Math.max(1.5,r*0.1);
    ctx.beginPath();
    for(let i=0;i<=6;i++){const mx=x-r*0.5+i*r*(1/6),my=y+r*0.3+((i%2)?-r*0.1:r*0.1);
      i?ctx.lineTo(mx,my):ctx.moveTo(mx,my);}
    ctx.stroke();
  }else{
    drawEyesOnly(x,y,r,g);
  }
}
function draw(){
  ctx.fillStyle='#04061a';ctx.fillRect(0,0,canvas.width,canvas.height);
  if(!tile||!grid)return;

  ctx.strokeStyle='#2f5cff';ctx.lineWidth=Math.max(2,tile*0.09);ctx.lineCap='round';
  ctx.beginPath();
  for(const s of wallSegs){ctx.moveTo(ox+s[0]*tile,oy+s[1]*tile);ctx.lineTo(ox+s[2]*tile,oy+s[3]*tile);}
  ctx.stroke();
  ctx.strokeStyle='#ffb8de';ctx.lineWidth=Math.max(2,tile*0.09);
  ctx.beginPath();ctx.moveTo(ox+9*tile,oy+7.5*tile);ctx.lineTo(ox+10*tile,oy+7.5*tile);ctx.stroke();

  for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++){
    const ch=grid[r][c];
    if(ch==='.'||ch==='o'){
      const x=ox+(c+.5)*tile,y=oy+(r+.5)*tile;
      if(ch==='.'){ctx.fillStyle='#ffd28f';ctx.beginPath();ctx.arc(x,y,tile*0.07,0,7);ctx.fill();}
      else if(Math.sin(time*10)>-0.3){ctx.fillStyle='#ffe9c9';ctx.beginPath();ctx.arc(x,y,tile*0.18,0,7);ctx.fill();}
    }
  }

  if(state!=='menu'){
    for(const g of ghosts)drawGhost(g);
    if(state==='dying'){
      const p=1-deathT/1.4;
      drawPacShape(pac.x,pac.y,pac.face,Math.min(p*Math.PI*1.2,Math.PI),1);
    }else{
      const open=pac.dir?0.1+0.35*Math.abs(Math.sin(time*14)):0.08;
      drawPacShape(pac.x,pac.y,pac.face,open,1);
    }
  }

  const fs=Math.max(13,tile*0.42);
  ctx.font='bold '+fs+'px "Courier New",monospace';
  ctx.textBaseline='bottom';ctx.textAlign='left';
  ctx.fillStyle='#fff';
  ctx.fillText('SCORE '+String(score).padStart(6,'0'),ox,oy-8);
  ctx.textAlign='center';ctx.fillStyle='#9fb0ff';
  ctx.fillText('HI '+String(Math.max(hiScore,score)).padStart(6,'0'),ox+COLS*tile/2,oy-8);
  for(let i=0;i<lives-1;i++)drawPacShape(COLS-0.7-i*0.95,-0.5,{x:-1,y:0},0.3,0.6);

  if(state==='ready'){
    ctx.fillStyle='#ffe135';ctx.textAlign='center';ctx.textBaseline='middle';
    ctx.font='bold '+(tile*0.7)+'px "Courier New",monospace';
    ctx.fillText('READY!',ox+COLS*tile/2,oy+11.6*tile);
  }
  if(paused&&(state==='playing'||state==='ready')){
    ctx.fillStyle='#fff';ctx.textAlign='center';ctx.textBaseline='middle';
    ctx.font='bold '+(tile*0.8)+'px "Courier New",monospace';
    ctx.fillText('PAUSED',ox+COLS*tile/2,oy+ROWS*tile/2);
  }
}

const KEYMAP={ArrowUp:[0,-1],w:[0,-1],W:[0,-1],ArrowDown:[0,1],s:[0,1],S:[0,1],
              ArrowLeft:[-1,0],a:[-1,0],A:[-1,0],ArrowRight:[1,0],d:[1,0],D:[1,0]};
function setDir(d){
  if(!pac)return;
  pac.wantDir={x:d[0],y:d[1]};
  if(pac.dir&&pac.dir.x===-d[0]&&pac.dir.y===-d[1])pac.dir={x:d[0],y:d[1]};
}
window.addEventListener('keydown',e=>{
  initAudio();
  const k=e.key;
  if(KEYMAP[k]){e.preventDefault();setDir(KEYMAP[k]);return;}
  if(k===' '||k==='Enter'){
    e.preventDefault();
    if(state==='menu'||state==='over'||state==='win')newGame();
    return;
  }
  if(k==='r'||k==='R'){newGame();return;}
  if(k==='p'||k==='P'){if(state==='playing'||state==='ready')paused=!paused;}
});
window.addEventListener('pointerdown',initAudio);

});
