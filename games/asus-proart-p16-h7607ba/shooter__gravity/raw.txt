var AC=null,masterGain=null;
function initAudio(){
  if(AC)return;
  try{
    AC=new (window.AudioContext||window.webkitAudioContext)();
    masterGain=AC.createGain();masterGain.gain.value=0.25;
    masterGain.connect(AC.destination);
  }catch(e){AC=null;}
}
function beep(freq,dur,type,vol,slide){
  if(!AC)return;
  try{
    var o=AC.createOscillator(),g=AC.createGain(),t=AC.currentTime;
    o.type=type||'square';o.frequency.setValueAtTime(freq,t);
    if(slide)o.frequency.exponentialRampToValueAtTime(Math.max(30,freq+slide),t+dur);
    g.gain.setValueAtTime(vol||0.5,t);
    g.gain.exponentialRampToValueAtTime(0.001,t+dur);
    o.connect(g);g.connect(masterGain);
    o.start(t);o.stop(t+dur+0.02);
  }catch(e){}
}
function noiseBurst(dur,vol){
  if(!AC)return;
  try{
    var n=Math.floor(AC.sampleRate*dur),buf=AC.createBuffer(1,n,AC.sampleRate),d=buf.getChannelData(0);
    for(var i=0;i<n;i++)d[i]=(Math.random()*2-1)*(1-i/n);
    var s=AC.createBufferSource(),g=AC.createGain(),f=AC.createBiquadFilter();
    f.type='lowpass';f.frequency.value=900;
    g.gain.value=vol||0.6;s.buffer=buf;
    s.connect(f);f.connect(g);g.connect(masterGain);s.start();
  }catch(e){}
}

function nextWave(){
  G.wave++;

  spawnTimer=0.5;

function spawnEnemy(){
  var side=Math.random()<0.5?-1:1;
  var r=Math.random(),type='drifter';
  if(G.wave>=2&&r<0.28)type='diver';
  if(G.wave>=3&&r>0.82)type='tank';
  var e={
    type:type,x:side<0?-40:W+40,
    y:60+Math.random()*(groundY-180),
    vx:(side<0?1:-1)*(50+Math.random()*40+G.wave*6),
    r:16,hp:1,maxHp:1,t:Math.random()*6,flash:0,seed:Math.random()*10
  };
  if(type==='diver'){e.hp=1;e.r=13;e.vx*=1.5;}
  if(type==='tank'){e.hp=2+Math.floor(G.wave/4);e.maxHp=e.hp;e.r=26;e.vx*=0.45;}

}

// ---------- effects ----------
function addText(x,y,txt,life,col){texts.push({x:x,y:y,txt:txt,life:life,max:life,col:col||'#fff'});}
function explode(x,y,col,big){
  var n=big?26:14;
  for(var i=0;i<n;i++){
    var a=Math.random()*Math.PI*2,s=60+Math.random()*(big?280:170);
    particles.push({x:x,y:y,vx:Math.cos(a)*s,vy:Math.sin(a)*s-40,life:0.5+Math.random()*0.5,max:1,col:col,r:2+Math.random()*3,g:big?500:700});
  }
  G.shake=Math.min(G.shake+(big?10:4),16);
}

// ---------- update ----------
function shoot(){
  if(player.cd>0)return;
  player.cd=0.22;
  var px=player.x,py=groundY-player.h-8;
  var dx=mouse.x-px,dy=mouse.y-py,d=Math.hypot(dx,dy)||1;
  var sp=560;
  bullets.push({x:px,y:py,vx:dx/d*sp,vy:dy/d*sp,r:4});
  beep(880,0.08,'square',0.25,-300);
}
function damagePlayer(){
  if(player.inv>0)return;
  G.lives--;player.inv=1.6;G.shake=14;
  explode(player.x,groundY-player.h/2,'#ff7f7f',true);
  noiseBurst(0.35,0.8);beep(120,0.4,'sawtooth',0.6,-60);
  if(G.lives<=0){
    G.state='gameover';
    G.best=Math.max(G.best,G.score);saveBest();
    document.getElementById('finalScore').textContent='SCORE '+G.score;
    document.getElementById('finalWave').textContent='You reached wave '+G.wave;
    document.getElementById('bestScore').textContent='Best: '+G.best;
    document.getElementById('over').classList.remove('hidden');
  }
}

function update(dt){
  G.time+=dt;
  if(G.shake>0)G.shake=Math.max(0,G.shake-dt*30);

  // player move
  var sp=340,mv=0;
  if(keys['KeyA']||keys['ArrowLeft'])mv-=1;
  if(keys['KeyD']||keys['ArrowRight'])mv+=1;
  player.x+=mv*sp*dt;
  player.x=Math.max(20,Math.min(W-20,player.x));
  player.cd-=dt;player.inv-=dt;
  if(mouse.down)shoot();

  // wave logic
  if(toSpawn<=0&&enemies.length===0){
    waveTimer+=dt;
    if(waveTimer>intermission){waveTimer=0;nextWave();}
  }else{
    if(toSpawn>0){
      spawnTimer-=dt;
      if(spawnTimer<=0){spawnEnemy();toSpawn--;spawnTimer=Math.max(0.35,1.4-G.wave*0.08);}
    }
  }

  // bullets (gravity!)
  var grav=620;
  for(var i=bullets.length-1;i>=0;i--){
    var b=bullets[i];
    b.vy+=grav*dt;b.x+=b.vx*dt;b.y+=b.vy*dt;
    if(b.y>groundY-2){ // hit ground: sparks
      for(var k=0;k<5;k++)particles.push({x:b.x,y:groundY,vx:(Math.random()-0.5)*160,vy:-Math.random()*160,life:0.3,max:0.3,col:'#8fa8cc',r:2,g:600});
      bullets.splice(i,1);continue;
    }
    if(b.x<-40||b.x>W+40||b.y<-H-100)bullets.splice(i,1);
  }

  // enemies
  for(var j=enemies.length-1;j>=0;j--){
    var e=enemies[j];e.t+=dt;e.flash-=dt;
    if(e.type==='diver'){
      var dxp=player.x-e.x,dyp=(groundY-player.h*0.6)-e.y,dd=Math.hypot(dxp,dyp)||1;
      e.vx+=dxp/dd*220*dt;e.vy=(e.vy||0)+dyp/dd*220*dt;
      var v=Math.hypot(e.vx,e.vy),max=170+G.wave*8;
      if(v>max){e.vx*=max/v;e.vy*=max/v;}
      e.y+=e.vy*dt;
    }else{
      e.y+=Math.sin(e.t*2+e.seed)*36*dt;
      if(e.type==='tank')e.y=Math.max(50,Math.min(groundY-90,e.y));
      else e.y=Math.max(40,Math.min(groundY-50,e.y));
    }
    e.x+=e.vx*dt;
    if((e.vx>0&&e.x>W+80)||(e.vx<0&&e.x<-80)){enemies.splice(j,1);continue;}

    // hit ground
    if(e.y+e.r>=groundY){
      explode(e.x,groundY-6,'#ff9a4d',true);noiseBurst(0.25,0.6);
      damagePlayer();enemies.splice(j,1);continue;
    }
    // hit player
    if(Math.hypot(e.x-player.x,e.y-(groundY-player.h/2))<e.r+22){
      explode(e.x,e.y,'#ff9a4d',true);noiseBurst(0.3,0.7);
      damagePlayer();enemies.splice(j,1);continue;
    }
    // bullet hits
    for(var k=bullets.length-1;k>=0;k--){
      var bl=bullets[k];
      if(Math.hypot(bl.x-e.x,bl.y-e.y)<e.r+bl.r){
        bullets.splice(k,1);
        e.hp--;e.flash=0.12;
        if(e.hp<=0){
          var col=e.type==='tank'?'#c58aff':(e.type==='diver'?'#ff6f9a':'#ffd27f');
          explode(e.x,e.y,col,e.type==='tank');
          noiseBurst(0.18,0.5);beep(e.type==='tank'?160:300,0.18,'square',0.4,-150);
          var pts=e.type==='tank'?250:(e.type==='diver'?150:100);
          G.score+=pts;addText(e.x,e.y-20,'+'+pts,0.9,col);
          enemies.splice(j,1);
        }else{
          beep(500,0.06,'square',0.3,-100);
        }
        break;
      }
    }
  }

  // particles & texts
  for(var p=particles.length-1;p>=0;p--){
    var pa=particles[p];pa.life-=dt;
    if(pa.life<=0){particles.splice(p,1);continue;}
    pa.vy+=pa.g*dt;pa.x+=pa.vx*dt;pa.y+=pa.vy*dt;
  }
  for(var t2=texts.length-1;t2>=0;t2--){
    texts[t2].life-=dt;texts[t2].y-=30*dt;
    if(texts[t2].life<=0)texts.splice(t2,1);
  }
}

    var ppx=player.x,ppy=groundY-player.h-8;
    var dxm=mouse.x-ppx,dym=mouse.y-ppy,dm=Math.hypot(dxm,dym)||1;
    var sx2=dxm/dm*560,sy2=dym/dm*560,tx=ppx,ty=ppy;
    ctx.fillStyle='rgba(255,243,176,0.35)';
    for(var t3=0;t3<0.9;t3+=0.06){
      var pxp=ppx+sx2*t3,pyp=ppy+sy2*t3+0.5*620*t3*t3;
      if(pyp>groundY)break;
      ctx.beginPath();ctx.arc(pxp,pyp,1.6,0,Math.PI*2);ctx.fill();
    }
  }

  // texts
  for(var t3=texts.length-1;t3>=0;t3--){
    var tx2=texts[t3];
    ctx.globalAlpha=Math.max(0,tx2.life/tx2.max);
    ctx.fillStyle=tx2.col;ctx.font='bold 20px Courier New';ctx.textAlign='center';
    ctx.fillText(tx2.txt,tx2.x,tx2.y);
  }
  ctx.globalAlpha=1;

  // HUD
  if(G.state==='playing'||G.state==='paused'){
    ctx.textAlign='left';ctx.font='bold 22px Courier New';
    ctx.fillStyle='#e8f0ff';ctx.fillText('SCORE '+G.score,16,34);
    ctx.fillStyle='#9fb4d8';ctx.font='16px Courier New';
    ctx.fillText('BEST '+Math.max(G.best,G.score),16,56);
    ctx.textAlign='center';ctx.font='bold 20px Courier New';ctx.fillStyle='#7fd8ff';
    ctx.fillText('WAVE '+G.wave,W/2,34);
    // lives
    for(var l=0;l<3;l++){
      ctx.fillStyle=l<G.lives?'#ff6f9a':'rgba(255,111,154,0.2)';
      var hx=W-40-l*30;
      ctx.beginPath();
      ctx.moveTo(hx,hx? 0:0); // heart via simple shape below
      ctx.arc(W-46-l*30+7,26,7,Math.PI,0);ctx.arc(W-46-l*30-7,26,7,Math.PI,0);
      ctx.lineTo(W-46-l*30-14,26);ctx.lineTo(W-46-l*30,42);ctx.lineTo(W-46-l*30+14,26);
      ctx.closePath();ctx.fill();
    }
    if(G.state==='paused'){
      ctx.fillStyle='rgba(5,8,20,0.7)';ctx.fillRect(0,0,W,H);
      ctx.fillStyle='#7fd8ff';ctx.font='bold 48px Courier New';ctx.textAlign='center';
      ctx.fillText('PAUSED',W/2,H/2);
      ctx.font='18px Courier New';ctx.fillStyle='#9fb4d8';
      ctx.fillText('Press P to resume',W/2,H/2+40);
    }
  }
  ctx.restore();
}

  if(G.state==='playing')update(dt);

}
