function castDist(px,py,dx,dy){ // DDA wall distance along dir
  let mx=px|0,my=py|0;
  const ddx=Math.abs(1/dx),ddy=Math.abs(1/dy);
  let sx,sy,sdx,sdy;
  if(dx<0){sx=-1;sdx=(px-mx)*ddx;}else{sx=1;sdx=(mx+1-px)*ddx;}
  if(dy<0){sy=-1;sdy=(py-my)*ddy;}else{sy=1;sdy=(my+1-py)*ddy;}
  for(let i=0;i<64;i++){
    if(sdx<sdy){mx+=sx;sdx+=ddx;}else{my+=sy;sdy+=ddy;}
    if(cell(mx,my))return sdx<sdy?sdx-ddx:sdy-ddy;
  }
  return 64;
}

    if(rdy<0){sy=-1;sdy=(P.y-my)*ddy;}else{sy=1;sdy=(my+1-P.y)*ddy;}
    let hit=0;
    for(let i=0;i<64&&!hit;i++){
      if(sdx<sdy){mx+=sx;sdx+=ddx;side=0;}else{my+=sy;sdy+=ddy;side=1;}
      hit=cell(mx,my);
    }
    const dist=Math.max(.02,(side? sdy-ddy : sdx-ddx));
    zbuf[x]=dist;
    const lh=BH/dist;
    const y0=(BH-lh)/2;
    const c=WALLCOL[hit]||WALLCOL[1];
    let f=Math.max(.12,1-dist/13)*(side?.7:1);
    g.fillStyle=shade(c,f);
    g.fillRect(x,y0,1,lh);
  }

  // sprites far -> near
  const list=zombies.map(z=>({z,d:(z.x-P.x)**2+(z.y-P.y)**2})).sort((a,b)=>b.d-a.d);
  for(const it of list){
    const z=it.z;
    const rx=z.x-P.x,ry=z.y-P.y;
    const inv=1/(plX*dirY-dirX*plY);
    const tx=inv*(dirY*rx-dirX*ry), ty=inv*(-plY*rx+plX*ry);
    if(ty<=.05)continue;
    const sx=(BW/2)*(1+tx/ty);
    const h=BH/ty*.86, w=h*(zCan.width/zCan.height);
    let bob=0;
    if(z.dead>0){bob=h*z.dead*1.4;}
    else bob=Math.sin(z.ph)*h*.02;
    const x0=sx-w/2,y0=(BH-h)/2+h*.07+bob;
    const alpha=z.dead>0?Math.max(0,1-z.dead/.8):1;
    if(alpha<=0)continue;
    g.globalAlpha=alpha;
    const strip=3,sw=zCan.width/strip;
    for(let s=0;s<strip;s++){
      const dx0=Math.floor(x0+w*s/strip),dx1=Math.floor(x0+w*(s+1)/strip);
      let drawn=false;
      for(let x=Math.max(0,dx0);x<Math.min(BW,dx1);x+=2){ if(zbuf[x]>ty){drawn=true;break;} }
      if(drawn)g.drawImage(zCan,sw*s,0,sw,zCan.height,dx0,y0,dx1-dx0+1,h);
    }
    g.globalAlpha=1;
  }

  // gun
  const gx=BW/2+Math.sin(P.bob)*8, gy=BH-70+Math.abs(Math.cos(P.bob))*6+gunKick*22;
  drawGun(gx,gy);
  if(muzzle>0){
    g.fillStyle='rgba(255,230,120,.9)';
    const fr=14+Math.random()*10;
    g.beginPath();g.arc(gx,gy-8,fr,0,7);g.fill();
    g.fillStyle='rgba(255,255,255,.9)';
    g.beginPath();g.arc(gx,gy-8,fr*.45,0,7);g.fill();
  }

  // crosshair
  g.strokeStyle=reloading?'#f80':'#fff';g.lineWidth=2;
  const cx=BW/2,cy=BH/2,cl=6+gunKick*10;
  g.beginPath();g.moveTo(cx-cl,cy);g.lineTo(cx-2,cy);g.moveTo(cx+2,cy);g.lineTo(cx+cl,cy);
  g.moveTo(cx,cy-cl);g.lineTo(cx,cy-2);g.moveTo(cx,cy+2);g.lineTo(cx,cy+cl);g.stroke();

  // HUD
  g.font='bold 16px Courier New';g.textBaseline='top';
  // health bar
  g.fillStyle='rgba(0,0,0,.5)';g.fillRect(14,BH-34,170,22);
  g.fillStyle=P.hp>30?'#4c4':'#f44';
  g.fillRect(16,BH-32,166*Math.max(0,P.hp)/100,18);
  g.strokeStyle='#8a8';g.strokeRect(14.5,BH-33.5,169,21);
  g.fillStyle='#fff';g.fillText('HP '+Math.ceil(P.hp),20,BH-30);
  // ammo
  g.font='bold 20px Courier New';g.textAlign='right';
  if(reloading){g.fillStyle='#fc6';g.fillText('RELOADING...',BW-14,BH-58);}
  else{g.fillStyle=P.ammo>2?'#fff':'#f55';g.fillText(P.ammo+' / 8',BW-14,BH-58);g.font='12px Courier New';g.fillStyle='#9a9';g.fillText('[R] RELOAD',BW-14,BH-36);}
  g.textAlign='left';g.font='bold 16px Courier New';
  g.fillStyle='#7cfc00';g.fillText('WAVE '+G.wave,14,12);
  g.fillStyle='#ff6';g.fillText('SCORE '+G.score,14,32);
  g.fillStyle='#f88';g.fillText('KILLS '+G.kills,150,32);

  if(G.waveMsgT>0){
    g.globalAlpha=Math.min(1,G.waveMsgT);
    g.font='bold 34px Courier New';g.textAlign='center';
    g.fillStyle='#f44';g.fillText('WAVE '+G.wave,BW/2,BH*.28);
    g.font='14px Courier New';g.fillStyle='#ccc';g.fillText('The dead are coming...',BW/2,BH*.28+36);
    g.textAlign='left';g.globalAlpha=1;
  }
  if(dmgFlash>0){g.fillStyle='rgba(200,0,0,'+(dmgFlash*.35)+')';g.fillRect(0,0,BW,BH);}
}

function drawGun(x,y){
  g.save();g.translate(x,y);
  g.fillStyle='#2b2f33';

}
