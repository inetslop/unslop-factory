  if(player.x<player.r){player.x=player.r;player.vx=0;}
  if(player.x>W-player.r){player.x=W-player.r;player.vx=0;}

  // platform collision (landing on the face gravity hits)
  var grounded=false;
  for(var i=0;i<platforms.length;i++){
    var p=platforms[i];
    if(player.x<p.x-8||player.x>p.x+p.w+8)continue;
    if(player.grav===1&&player.vy>0){
      if(prevY+player.r<=p.y+6&&player.y+player.r>=p.y){player.y=p.y-player.r;player.vy=0;grounded=true;}
    }else if(player.grav===-1&&player.vy<0){
      if(prevY-player.r>=p.y+p.h-6&&player.y-player.r<=p.y+p.h){player.y=p.y+p.h+player.r;player.vy=0;grounded=true;}
    }
  }

  // mouth animation
  var speed=Math.abs(player.vx)+Math.abs(player.vy);
  player.mouth=grounded?0.35:0.28+0.6*Math.abs(Math.sin(t*11));

  // pellets
  for(var j=0;j<pellets.length;j++){
    var el=pellets[j];
    var dx=player.x-el.x,dy=player.y-el.y;
    if(dx*dx+dy*dy<(player.r+el.r)*(player.r+el.r)){
      el.taken=true;
      score+=el.power?50:10;
      if(score>hi){hi=score;}
      if(el.power){powered=7;sfxPower();}else sfxEat();
    }
  }
  powered=Math.max(0,powered-dt);
  player.inv=Math.max(0,player.inv-dt);

  // depth score
  if(player.y>maxDepth){
    depthScoreAcc+=player.y-maxDepth;maxDepth=player.y;
    while(depthScoreAcc>=16){depthScoreAcc-=16;score++;if(score>hi)hi=score;}
  }else if(player.y<maxDepth-400){ // going back up? keep max, fine

  }

  // ghosts
  ghostTimer-=dt;
  var cap=Math.min(5,1+Math.floor(maxDepth/1800));
  if(ghostTimer<=0&&ghosts.length<cap){spawnGhost();ghostTimer=Math.max(1.6,4-maxDepth/2500);}
  for(var g=ghosts.length-1;g>=0;g--){
    var gh=ghosts[g];
    var ddx=player.x-gh.x,ddy=player.y-gh.y;
    var dist=Math.sqrt(ddx*ddx+ddy*ddy)||1;
    var wob=Math.sin(t*2.4+gh.seed)*40;
    var tvx=(ddx/dist)*gh.sp+wob*(ddy/dist);
    var tvy=(ddy/dist)*gh.sp-wob*(ddx/dist);
    gh.vx+=(tvx-gh.vx)*Math.min(1,3*dt);
    gh.vy+=(tvy-gh.vy)*Math.min(1,3*dt);
    gh.x+=gh.vx*dt;gh.y+=gh.vy*dt;
    if(gh.x<gh.r){gh.x=gh.r;gh.vx=Math.abs(gh.vx);}
    if(gh.x>W-gh.r){gh.x=W-gh.r;gh.vx=-Math.abs(gh.vx);}
    // cull far ghosts
    if(Math.abs(gh.y-player.y)>H*2.5){ghosts.splice(g,1);continue;}
    // collision
    var d2x=player.x-gh.x,d2y=player.y-gh.y;
    if(d2x*d2x+d2y*d2y<(player.r+gh.r-3)*(player.r+gh.r-3)){
      if(powered>0){score+=200;if(score>hi)hi=score;sfxGhost();ghosts.splice(g,1);}
      else if(player.inv<=0){die();}
    }
  }

  // camera
  var target=player.y-H*0.5;
  camY+=(target-camY)*Math.min(1,dt*6);
  ensureGen();
}

/* ---------- drawing ---------- */
function drawPac(x,y,r,face,mouth,bg){
  ctx.save();ctx.translate(x,y);ctx.rotate(face);
  ctx.fillStyle=bg;ctx.beginPath();ctx.arc(0,0,r,0,Math.PI*2);ctx.fill();
  ctx.fillStyle='#ffe23a';
  ctx.beginPath();
  var a=mouth*Math.PI/2;
  ctx.moveTo(0,0);
  ctx.arc(0,0,r,a,-a,false);
  ctx.closePath();ctx.fill();
  // eye
  ctx.fillStyle='#111';
  ctx.beginPath();ctx.arc(r*0.15,-r*0.45,r*0.13,0,Math.PI*2);ctx.fill();
  ctx.restore();
}

function drawGhost(g){
  var w=g.r;
  var c=(powered>0)?(Math.floor(t*9)%2?'#3a7bff':'#eef4ff'):g.col;
  ctx.fillStyle=c;
  ctx.beginPath();
  ctx.arc(g.x,g.y-3,w,Math.PI,0);
  ctx.lineTo(g.x+w,g.y+12);
  var bw=(2*w)/3;
  for(var i=0;i<3;i++){
    ctx.quadraticCurveTo(g.x+w-bw*(i+0.5),g.y+4,g.x+w-bw*(i+1),g.y+12);
  }
  ctx.closePath();ctx.fill();
  // eyes
  var lx=player.x-g.x,ly=player.y-g.y,ld=Math.sqrt(lx*lx+ly*ly)||1;
  lx/=ld;ly/=ld;
  for(var e=-1;e<=1;e+=2){
    ctx.fillStyle='#fff';
    ctx.beginPath();ctx.arc(g.x+e*w*0.4,g.y-5,w*0.32,0,Math.PI*2);ctx.fill();
    ctx.fillStyle=(powered>0)?'#f33':'#123';
    ctx.beginPath();ctx.arc(g.x+e*w*0.4+lx*w*0.14,g.y-5+ly*w*0.14,w*0.16,0,Math.PI*2);ctx.fill();
  }
}

function draw(){
  ctx.fillStyle='#05030f';ctx.fillRect(0,0,W,H);

  ctx.save();
  ctx.translate(0,-camY);

  // faint maze grid
  ctx.strokeStyle='rgba(40,60,190,0.28)';ctx.lineWidth=2;
  var gs=52;
  var y0=Math.floor(camY/gs)*gs;
  ctx.beginPath();
  for(var gy=y0;gy<camY+H+gs;gy+=gs){ctx.moveTo(0,gy);ctx.lineTo(W,gy);}
  for(var gx=0;gx<W;gx+=gs){ctx.moveTo(gx,camY);ctx.lineTo(gx,camY+H);}
  ctx.stroke();

  // side walls
  ctx.fillStyle='#1d2f9e';
  ctx.fillRect(0,camY,6,H);ctx.fillRect(W-6,camY,6,H);
  ctx.fillStyle='rgba(120,150,255,.5)';
  for(var wy=Math.floor(camY/24)*24;wy<camY+H;wy+=24){
    ctx.fillRect(1,wy,4,10);ctx.fillRect(W-5,wy,4,10);
  }

  // platforms
  for(var i=0;i<platforms.length;i++){
    var p=platforms[i];
    if(p.y<camY-30||p.y>camY+H+30)continue;
    ctx.fillStyle='#2a3fd6';
    ctx.fillRect(p.x,p.y,p.w,p.h);
    ctx.fillStyle='#7d95ff';
    ctx.fillRect(p.x,p.y,p.w,3);
    ctx.fillRect(p.x,p.y+p.h-3,p.w,3);
  }

  // pellets
  for(var j=0;j<pellets.length;j++){
    var el=pellets[j];
    if(el.taken||el.y<camY-20||el.y>camY+H+20)continue;
    if(el.power){
      ctx.globalAlpha=0.55+0.45*Math.sin(t*8);
      ctx.fillStyle='#ffd23a';
      ctx.shadowColor='#ffb300';ctx.shadowBlur=14;
      ctx.beginPath();ctx.arc(el.x,el.y,el.r,0,Math.PI*2);ctx.fill();
      ctx.shadowBlur=0;ctx.globalAlpha=1;
    }else{
      ctx.fillStyle='#ffe9a8';
      ctx.beginPath();ctx.arc(el.x,el.y,el.r,0,Math.PI*2);ctx.fill();
    }
  }

  // ghosts
  for(var g=0;g<ghosts.length;g++)drawGhost(ghosts[g]);

  // player (blink when invulnerable)
  if(player.inv<=0||Math.floor(t*10)%2===0){
    drawPac(player.x,player.y,player.r,player.face,player.mouth,'#0a0716');
  }

  ctx.restore();

  // powered aura vignette
  if(powered>0){
    var a=powered<1.5?Math.floor(t*8)%2:0;
    if(a){ctx.fillStyle='rgba(40,90,255,0.10)';ctx.fillRect(0,0,W,H);}
  }

  // HUD
  ctx.font='bold '+Math.min(26,Math.max(16,W*0.03))+"px 'Courier New',monospace";
  ctx.textBaseline='top';
  ctx.fillStyle='#ffe23a';ctx.textAlign='left';
  ctx.fillText('SCORE '+score,16,14);
  ctx.fillStyle='#8fa3ff';ctx.textAlign='center';
  ctx.fillText('BEST '+hi,W/2,14);

  // lives as mini pacs
  for(var l=0;l<lives-1;l++)drawPac(W-36-l*32,30,11,Math.PI,0.5,'#05030f');

  // gravity indicator
  ctx.fillStyle=(player.grav===1)?'#7df9ff':'#ff8ad9';
  ctx.textAlign='right';ctx.font='bold '+Math.min(20,Math.max(14,W*0.024))+"px 'Courier New',monospace";
  ctx.fillText('GRAVITY \u25BC',W-16,H-34);
  if(powered>0){
    ctx.fillStyle='#ffd23a';ctx.textAlign='left';
    ctx.fillText('POWER '+powered.toFixed(1),16,H-34);
  }

  if(paused){
    ctx.fillStyle='rgba(0,0,10,0.6)';ctx.fillRect(0,0,W,H);
    ctx.fillStyle='#7df9ff';ctx.textAlign='center';
    ctx.font='bold '+Math.min(52,Math.max(24,W*0.07))+"px 'Courier New',monospace";
    ctx.fillText('PAUSED  (P to resume)',W/2,H/2);
  }
}

  else t+=dt*0.5; // keep menu anim alive
  draw();

}
