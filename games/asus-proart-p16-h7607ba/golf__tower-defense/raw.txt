  for(let a=0;a<8;a++){ const an=a/8*6.283+tGlobal*(a%2? .5:-.5); cx.beginPath(); cx.arc(Math.cos(an)*r*.55,Math.sin(an)*r*.55,r*.13,0,7); cx.fill(); }
  if(maxhp>1){ // hp pips
    for(let k=0;k<maxhp;k++){ cx.fillStyle=k<hp?'#ff6b6b':'rgba(255,255,255,.4)'; cx.beginPath(); cx.arc((k-(maxhp-1)/2)*7,-r-8,3,0,7); cx.fill(); }
  }
  cx.restore();
}

function draw(){
  cx.save();
  if(shake>0) cx.translate((Math.random()-.5)*shake*2,(Math.random()-.5)*shake*2);

  // sky
  const sky=cx.createLinearGradient(0,0,0,groundY);
  sky.addColorStop(0,'#63b4e8'); sky.addColorStop(.7,'#aee1f5'); sky.addColorStop(1,'#d8f3ff');
  cx.fillStyle=sky; cx.fillRect(-20,-20,W+40,groundY+20);
  // sun
  cx.fillStyle='rgba(255,240,180,.9)'; cx.beginPath(); cx.arc(W*.14,H*.16,38,0,7); cx.fill();
  cx.fillStyle='rgba(255,240,180,.25)'; cx.beginPath(); cx.arc(W*.14,H*.16,58,0,7); cx.fill();
  // clouds
  cx.fillStyle='rgba(255,255,255,.85)';
  clouds.forEach(c=>{ cx.beginPath();
    cx.ellipse(c.x,c.y,42*c.s,16*c.s,0,0,7); cx.ellipse(c.x+30*c.s,c.y-10*c.s,30*c.s,14*c.s,0,0,7); cx.ellipse(c.x-30*c.s,c.y-6*c.s,26*c.s,12*c.s,0,0,7); cx.fill(); });
  // hills
  cx.fillStyle='#8fd47a'; cx.beginPath(); cx.moveTo(0,groundY);
  for(let x=0;x<=W;x+=40) cx.lineTo(x, groundY-40-Math.sin(x*.008)*35-Math.sin(x*.021+2)*18);
  cx.lineTo(W,groundY); cx.fill();

  // grass
  const gr=cx.createLinearGradient(0,groundY,0,H);
  gr.addColorStop(0,'#5cb54a'); gr.addColorStop(1,'#3c8a2e');
  cx.fillStyle=gr; cx.fillRect(-20,groundY,W+40,H-groundY+20);
  cx.fillStyle='rgba(0,0,0,.06)';
  for(let x=(tGlobal*0)%80;x<W;x+=160) cx.fillRect(x,groundY,80,H-groundY);

  // floating green slab with cup
  const sx=greenX-SLAB_HALF;
  cx.fillStyle='rgba(0,0,0,.2)'; cx.beginPath(); cx.ellipse(greenX,groundY-4,SLAB_HALF*.9,7,0,0,7); cx.fill();
  cx.fillStyle='#8d5a2b'; cx.fillRect(sx,greenTop+12,SLAB_HALF*2,16);
  const gg=cx.createLinearGradient(0,greenTop,0,greenTop+14);
  gg.addColorStop(0,'#7ed957'); gg.addColorStop(1,'#4aa338');
  cx.fillStyle=gg; cx.fillRect(sx,greenTop,SLAB_HALF*2,12);
  // cup notch
  cx.fillStyle='#1a2b12';
  cx.beginPath();
  cx.moveTo(cupX-16,greenTop); cx.lineTo(cupX+16,greenTop); cx.lineTo(cupX+10,greenTop+14); cx.lineTo(cupX-10,greenTop+14); cx.closePath(); cx.fill();
  // cup flag
  cx.strokeStyle='#eee'; cx.lineWidth=2.5; cx.beginPath(); cx.moveTo(cupX,greenTop); cx.lineTo(cupX,greenTop-34); cx.stroke();
  cx.fillStyle='#ff4b4b'; const fw=Math.sin(tGlobal*4)*3;
  cx.beginPath(); cx.moveTo(cupX,greenTop-34); cx.lineTo(cupX+22+fw,greenTop-28); cx.lineTo(cupX,greenTop-21); cx.closePath(); cx.fill();

  // player flag (the thing to defend)
  cx.fillStyle='#e9e4d0'; cx.beginPath(); cx.ellipse(flagX,groundY,34,10,0,Math.PI,0); cx.fill();
  cx.strokeStyle='#fff'; cx.lineWidth=4; cx.beginPath(); cx.moveTo(flagX,groundY-4); cx.lineTo(flagX,groundY-72); cx.stroke();
  cx.fillStyle='#ffd93d'; const fw2=Math.sin(tGlobal*5)*4;
  cx.beginPath(); cx.moveTo(flagX,groundY-72); cx.lineTo(flagX+30+fw2,groundY-62); cx.lineTo(flagX,groundY-51); cx.closePath(); cx.fill();

  // enemies & bullets
  enemies.forEach(e=>drawBall(e.x,e.y,e.r,e.c,e.hp,e.maxhp));
  bullets.forEach(b=>{
    cx.strokeStyle='rgba(255,217,61,.4)'; cx.lineWidth=4; cx.beginPath(); cx.moveTo(b.x-b.vx*.03,b.y-b.vy*.03); cx.lineTo(b.x,b.y); cx.stroke();
    drawBall(b.x,b.y,b.r,'#ffcf4d',1,1);
  });

  // turret
  const ty=groundY-14;
  cx.fillStyle='rgba(0,0,0,.25)'; cx.beginPath(); cx.ellipse(turretX,groundY-2,30,8,0,0,7); cx.fill();
  cx.save(); cx.translate(turretX,ty);
  cx.rotate(aim);
  const bg=cx.createLinearGradient(0,-9,0,9); bg.addColorStop(0,'#5d6b7a'); bg.addColorStop(.5,'#38424e'); bg.addColorStop(1,'#20262e');
  cx.fillStyle=bg; cx.beginPath(); cx.roundRect ? cx.roundRect(0,-9,40,18,5) : cx.rect(0,-9,40,18); cx.fill();
  cx.fillStyle='#1c2128'; cx.fillRect(34,-6,7,12);
  cx.restore();
  const tg=cx.createRadialGradient(turretX-6,ty-8,4,turretX,ty,20);
  tg.addColorStop(0,'#7a8a9c'); tg.addColorStop(1,'#2b333d');
  cx.fillStyle=tg; cx.beginPath(); cx.arc(turretX,ty,17,0,7); cx.fill();
  if(rapidT>0){ cx.strokeStyle='rgba(255,157,92,.8)'; cx.lineWidth=3; cx.beginPath(); cx.arc(turretX,ty,22,-Math.PI/2,-Math.PI/2+ (rapidT/3.5)*6.283); cx.stroke(); }
  // aim guide dots
  if(state==='play'){
    let gx=turretX, gy=ty, gvx=Math.cos(aim)*760, gvy=Math.sin(aim)*760;
    cx.fillStyle='rgba(255,255,255,.35)';
    for(let i=1;i<=9;i++){ const s=i*.09; gx=turretX+gvx*s; gy=ty+gvy*s+.5*GRAV*s*s; if(gy>groundY||gx>W)break; cx.beginPath(); cx.arc(gx,gy,2.5,0,7); cx.fill(); }
  }

  // particles
  parts.forEach(p=>{ cx.globalAlpha=Math.max(0,p.life*1.8); cx.fillStyle=p.c; cx.beginPath(); cx.arc(p.x,p.y,p.r,0,7); cx.fill(); });
  cx.globalAlpha=1;

  // floating text
  floats.forEach(f=>{
    cx.globalAlpha=Math.min(1,f.t); cx.font='bold '+Math.min(26,W*.03)+'px Trebuchet MS'; cx.textAlign='center';
    cx.strokeStyle='rgba(0,0,0,.6)'; cx.lineWidth=4; cx.strokeText(f.txt,f.x,f.y);
    cx.fillStyle=f.c; cx.fillText(f.txt,f.x,f.y);
  });
  cx.globalAlpha=1;

  // HUD
  if(state!=='start'){
    cx.textAlign='left'; cx.font='bold '+Math.min(24,W*.035)+'px Trebuchet MS';
    cx.strokeStyle='rgba(0,0,0,.5)'; cx.lineWidth=4;
    cx.strokeText('SCORE  '+score,16,34); cx.fillStyle='#fff'; cx.fillText('SCORE  '+score,16,34);
    cx.strokeText('WAVE  '+wave,16,62); cx.fillStyle='#b7f39a'; cx.fillText('WAVE  '+wave,16,62);
    // lives
    cx.textAlign='right';
    for(let i=0;i<5;i++){
      const hx=W-24-i*30;
      cx.font=Math.min(26,W*.035)+'px serif';
      cx.fillStyle=i<lives?'#ff6b6b':'rgba(255,255,255,.25)';
      cx.fillText('\u2665',hx,34);
    }
  }
  cx.restore();
}

/* ---------- main loop ---------- */
let last=performance.now();
function loop(now){
  let dt=(now-last)/1000; last=now; if(dt>0.05)dt=0.05;
  if(state==='play') update(dt);
  else { tGlobal+=dt; clouds.forEach(c=>{ c.x+=c.v*dt*.4; if(c.x>W+120)c.x=-120; }); }
  draw();
  requestAnimationFrame(loop);
}
reset(); // pre-populate clouds for start-screen backdrop
requestAnimationFrame(loop);
})();
