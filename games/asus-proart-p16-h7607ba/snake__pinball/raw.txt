function segCollide(b,x1,y1,x2,y2,rest){
  var dx=x2-x1, dy=y2-y1;
  var L2=dx*dx+dy*dy;
  var t=((b.x-x1)*dx+(b.y-y1)*dy)/L2;
  t=Math.max(0,Math.min(1,t));
  var qx=x1+t*dx, qy=y1+t*dy;
  var nx=b.x-qx, ny=b.y-qy;
  var d=Math.sqrt(nx*nx+ny*ny);
  if(d>=b.r||d===0)return false;
  nx/=d; ny/=d;
  b.x=qx+nx*b.r; b.y=qy+ny*b.r;
  var vn=b.vx*nx+b.vy*ny;
  if(vn<0){ b.vx-=(1+rest)*vn*nx; b.vy-=(1+rest)*vn*ny; }
  return true;
}

      for(var g=0;g<guides.length;g++){var gd=guides[g]; segCollide(b,gd.x1,gd.y1,gd.x2,gd.y2,0.55);}
    }
    // slingshots (both states)
    for(var sl=0;sl<slings.length;sl++){
      var S=slings[sl];
      if(segCollide(b,S.x1,S.y1,S.x2,S.y2,0.9)){
        var sp3=Math.sqrt(b.vx*b.vx+b.vy*b.vy);
        if(sp3<5){ b.vx*=5/sp3; b.vy*=5/sp3; }
        S.flash=12; addText((S.x1+S.x2)/2,(S.y1+S.y2)/2,'25'); score+=25; sSling();
      }
    }
    // bumpers
    for(var i=0;i<bumpers.length;i++){
      var bm=bumpers[i];
      var nx=b.x-bm.x, ny=b.y-bm.y;
      var d=Math.sqrt(nx*nx+ny*ny);
      if(d<bm.r+b.r&&d>0){
        nx/=d; ny/=d;
        b.x=bm.x+nx*(bm.r+b.r); b.y=bm.y+ny*(bm.r+b.r);
        var vn=b.vx*nx+b.vy*ny;
        if(vn<0){ b.vx-=2*vn*nx; b.vy-=2*vn*ny; }
        var sp4=Math.sqrt(b.vx*b.vx+b.vy*b.vy);
        if(sp4<4.5){ b.vx*=4.5/sp4; b.vy*=4.5/sp4; }
        bm.flash=14; score+=100; addText(bm.x,bm.y-bm.r-6,'100'); sBumper();
      }
    }
    // flippers
    flipperCollide(b,fL);
    fliperFix(fL);
    flipperCollide(b,fR);

    // clamp speed
    var sp5=Math.sqrt(b.vx*b.vx+b.vy*b.vy);
    if(sp5>13){ b.vx*=13/sp5; b.vy*=13/sp5; }
  }
  function fliperFix(f){}
}

function updateSnake(){
  var speed=1.5+Math.min(1.2,score/6000);
  var h=snake[0];
  var dx=ball.x-h.x, dy=ball.y-h.y;
  var d=Math.sqrt(dx*dx+dy*dy)||1;
  var tx=dx/d, ty=dy/d;
  // steer: keep a velocity dir stored on head
  if(!h.vx){h.vx=tx;h.vy=ty;}
  h.vx+=(tx-h.vx)*0.10; h.vy+=(ty-h.vy)*0.10;
  var dl=Math.sqrt(h.vx*h.vx+h.vy*h.vy)||1;
  h.vx/=dl; h.vy/=dl;
  h.x+=h.vx*speed; h.y+=h.vy*speed;
  // keep in bounds
  h.x=Math.max(16,Math.min(W-16,h.x));
  h.y=Math.max(30,Math.min(H-16,h.y));
  for(var i=1;i<SNK_N;i++){
    var p=snake[i], q=snake[i-1];
    var ex=q.x-p.x, ey=q.y-p.y;
    var el=Math.sqrt(ex*ex+ey*ey)||1;
    p.x=q.x-ex/el*sp; p.y=q.y-ey/el*sp;
  }
  // collision with ball
  if(invuln>0){invuln--;return;}
  for(var i2=0;i2<SNK_N;i2++){
    var s3=snake[i2];
    var rr=(i2===0?8:5)+ball.r-3;
    var ddx=s3.x-ball.x, ddy=s3.y-ball.y;
    if(ddx*ddx+ddy*ddy<rr*rr){
      lives--; invuln=130; sDrain();
      addText(ball.x,ball.y-20,'OUCH!');
      if(lives<=0){
        state='gameover'; sOver();
        if(score>hiScore){hiScore=score;saveHi();}
      } else {
        resetBall(); resetSnake();
      }
      return;
    }
  }
}

function drainCheck(){
  if(!ball.inPlunger && ball.y>H+25){
    lives--; sDrain();
    addText(200,400,'DRAIN!');
    if(lives<=0){
      state='gameover'; sOver();
      if(score>hiScore){hiScore=score;saveHi();}
    } else {
      resetBall(); resetSnake(); invuln=90;
    }
  }
}

  if(k==='r'||k==='R'){ initAudio(); newGame(); }

});

function rr(x,y,w,h,r){ctx.beginPath();ctx.moveTo(x+r,y);ctx.arcTo(x+w,y,x+w,y+h,r);ctx.arcTo(x+w,y+h,x,y+h,r);ctx.arcTo(x,y+h,x,y,r);ctx.arcTo(x,y,x+w,y,r);ctx.closePath();}

function draw(){

  ctx.strokeStyle='#3a4a8a'; ctx.lineWidth=4;

    ctx.lineWidth=9;

  }

function tick(){
  if(state==='playing'&&!paused){
    updateFlippers();
    stepBall();
    drainCheck();
    updateSnake();
    if(charging)charge=Math.min(1,charge+0.02);
  } else {

    fL.w=0;fR.w=0;

  draw();

}
