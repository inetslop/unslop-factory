    const sp=Math.hypot(ball.vx,ball.vy);

  }
}

  roundRect(x+5,y+5,CELL-10,(CELL-10)/2.4,4); ctx.fill();
  ctx.globalAlpha=1;
}
function draw(){
  ctx.setTransform(2,0,0,2,0,0);
  // bg
  const g=ctx.createLinearGradient(0,0,0,H);
  g.addColorStop(0,'#0c1329'); g.addColorStop(1,'#070b14');
  ctx.fillStyle=g; ctx.fillRect(0,0,W,H);

  // field frame
  ctx.strokeStyle='#2a3f6e'; ctx.lineWidth=4;
  ctx.strokeRect(FX-2,FY-2,FW+4,FH+4);
  // subtle grid
  ctx.strokeStyle='rgba(120,150,220,.07)'; ctx.lineWidth=1;
  for(let x=1;x<COLS;x++){ ctx.beginPath();ctx.moveTo(FX+x*CELL,FY);ctx.lineTo(FX+x*CELL,FY+FH);ctx.stroke(); }
  for(let y=1;y<ROWS;y++){ ctx.beginPath();ctx.moveTo(FX,FY+y*CELL);ctx.lineTo(FX+FW,FY+y*CELL);ctx.stroke(); }

  if(state==='start'){ drawStart(); return; }

  // locked cells
  for(let y=0;y<ROWS;y++) for(let x=0;x<COLS;x++)
    if(grid[y][x]) drawCell(FX+x*CELL,FY+y*CELL,grid[y][x]);

  // ghost + active piece
  if(state==='play' && piece){
    const gy=ghostY();
    ctx.globalAlpha=0.25;
    for(const [x,y] of cellsOf(piece,0,gy-piece.y))
      if(y>=0) drawCell(FX+x*CELL,FY+y*CELL,piece.c);
    ctx.globalAlpha=1;
    for(const [x,y] of cellsOf(piece))
      if(y>=0) drawCell(FX+x*CELL,FY+y*CELL,piece.c);
  }

  // flash on line clear
  if(flashT>0){
    ctx.fillStyle='rgba(255,255,255,'+(flashT*1.4)+')';
    ctx.fillRect(FX,FY,FW,FH);
  }

  // funnels
  ctx.strokeStyle='#3b5aa0'; ctx.lineWidth=6; ctx.lineCap='round';
  for(const f of funnels){ ctx.beginPath();ctx.moveTo(f.ax,f.ay);ctx.lineTo(f.bx,f.by);ctx.stroke(); }

  // flippers
  for(const f of flippers){
    const tx=f.p.x+Math.cos(f.a)*f.len, ty=f.p.y+Math.sin(f.a)*f.len;
    ctx.strokeStyle='#f5c542'; ctx.lineWidth=14; ctx.lineCap='round';
    ctx.beginPath();ctx.moveTo(f.p.x,f.p.y);ctx.lineTo(tx,ty);ctx.stroke();
    ctx.strokeStyle='#ffe9a0'; ctx.lineWidth=5;
    ctx.beginPath();ctx.moveTo(f.p.x,f.p.y);ctx.lineTo(tx,ty);ctx.stroke();
  }

  // ball
  if(ball.active){
    const bg=ctx.createRadialGradient(ball.x-3,ball.y-3,1,ball.x,ball.y,BALL_R);
    bg.addColorStop(0,'#ffffff'); bg.addColorStop(0.6,'#c8d4e8'); bg.addColorStop(1,'#7a8aa5');
    ctx.fillStyle=bg;
    ctx.beginPath();ctx.arc(ball.x,ball.y,BALL_R,0,Math.PI*2);ctx.fill();
  } else if(state==='play'){
    ctx.fillStyle='rgba(255,255,255,.6)'; ctx.font='bold 14px monospace'; ctx.textAlign='center';
    ctx.fillText('BALL INCOMING...',FX+FW/2,FY+FH-10);
  }

  // particles
  for(const p of particles){
    ctx.globalAlpha=Math.max(0,p.life*1.6);
    ctx.fillStyle=p.col;
    ctx.fillRect(p.x-p.size/2,p.y-p.size/2,p.size,p.size);
  }
  ctx.globalAlpha=1;

  // floating texts
  for(const t of texts){
    ctx.globalAlpha=Math.max(0,Math.min(1,t.life*1.5));
    ctx.fillStyle=t.big?'#ffe9a0':'#fff';
    ctx.font=(t.big?'bold 26px':'bold 14px')+' monospace';
    ctx.textAlign='center';
    ctx.fillText(t.txt,t.x,t.y);
  }
  ctx.globalAlpha=1;

  drawHUD();

  if(paused){
    ctx.fillStyle='rgba(0,0,0,.6)'; ctx.fillRect(FX,FY,FW,FH);
    ctx.fillStyle='#fff'; ctx.font='bold 34px monospace'; ctx.textAlign='center';
    ctx.fillText('PAUSED',FX+FW/2,FY+FH/2);
    ctx.font='14px monospace'; ctx.fillText('Press P to resume',FX+FW/2,FY+FH/2+30);
  }
  if(state==='over') drawOver();
}

const keys={};
window.addEventListener('keydown',function(e){
  initAudio();
  const k=e.key.toLowerCase();
  if(['arrowleft','arrowright','arrowup','arrowdown',' '].indexOf(k)>=0) e.preventDefault();
  keys[k]=true;

  if(k==='enter'){
    if(state!=='play') begin();
    return;
  }
  if(state==='over' && k==='r'){ begin(); return; }
  if(state!=='play') return;

  if(k==='p'){ paused=!paused; return; }
  if(paused) return;
  if(k==='r'){ begin(); return; }

  if(!e.repeat){
    if(k==='arrowup'||k==='x') tryRotate();
    if(k===' ') hardDrop();
  }
  if(k==='arrowleft') tryMove(-1,0);
  if(k==='arrowright') tryMove(1,0);
});
window.addEventListener('keyup',function(e){ keys[e.key.toLowerCase()]=false; });
cv.addEventListener('pointerdown',function(){ initAudio(); if(state!=='play') begin(); });

let last=performance.now();
function frame(t){
  requestAnimationFrame(frame);
  let dt=Math.min(0.033,(t-last)/1000); last=t;
  if(state==='play' && !paused) update(dt);
  draw();
}
requestAnimationFrame(frame);
})();
