function resize(){
  const dpr = window.devicePixelRatio || 1;
  canvas.width = window.innerWidth * dpr;
  canvas.height = window.innerHeight * dpr;
  canvas.style.width = window.innerWidth + 'px';
  canvas.style.height = window.innerHeight + 'px';
  ctx.setTransform(dpr,0,0,dpr,0,0);
  const hudH = 56;

const WORDS = ["SNAKE","APPLE","TYPING","PYTHON","CIPHER","KEYBOARD","PIXEL","GARDEN","FROST","BLAZE","COMET","DRAGON","EMERALD","FALCON","GLACIER","HARBOR","JUNGLE","KITTEN","LIGHTNING","MAPLE","NEBULA","ORCHID","PUZZLE","QUARTZ","RIVER","SILVER","THUNDER","UNICORN","VELVET","WIZARD","YARROW","ZEPHYR","COCOA","MANGO","PANDA","ROCKET","STORM","TRUFFLE","BUBBLE","CHERRY","DIAMOND","EAGLE","FIREWORK","GALAXY","HONEY","JELLY","KNIGHT","LEMON","MOONBEAM","NUTMEG","OCEAN","PEARL","RUBY","SPARKLE","TURTLE","VINE","WILLOW","ZEBRA"];

let state = 'start';
let snake, dir, dirQueue, food, word, typed, charged, score, lives, highScore=0, tickInterval, lastTick, grow, flashTimer, pulse;

try { highScore = parseInt(localStorage.getItem('snakeTyperHigh')||'0',10)||0; } catch(e){}

function saveHigh(){ try{ localStorage.setItem('snakeTyperHigh', String(highScore)); }catch(e){} }

/* ---------- Audio (created on first input) ---------- */
let actx = null;
function audio(){
  if(!actx){
    try{ actx = new (window.AudioContext||window.webkitAudioContext)(); }catch(e){}
  }
  if(actx && actx.state === 'suspended') actx.resume();
}
function beep(freq, dur, type, vol, delay){
  if(!actx) return;
  const t = actx.currentTime + (delay||0);
  const o = actx.createOscillator(), g = actx.createGain();
  o.type = type||'square'; o.frequency.value = freq;
  g.gain.setValueAtTime(vol||0.15, t);
  g.gain.exponentialRampToValueAtTime(0.001, t+dur);
  o.connect(g); g.connect(actx.destination);
  o.start(t); o.stop(t+dur+0.02);
}
const sfx = {
  eat(){ beep(520,.09,'square',.12); beep(780,.1,'square',.1,.06); },
  charged(){ [440,660,880].forEach((f,i)=>beep(f,.12,'triangle',.14,i*.07)); },
  chargeEat(){ [523,659,784,1046].forEach((f,i)=>beep(f,.12,'square',.13,i*.06)); },
  wrong(){ beep(140,.22,'sawtooth',.16); },
  over(){ [400,300,220,140].forEach((f,i)=>beep(f,.22,'sawtooth',.15,i*.16)); }
};

/* ---------- Game setup ---------- */
function pickWord(){
  let w;
  do { w = WORDS[Math.floor(Math.random()*WORDS.length)]; } while(w === word);
  return w;
}
function freeCell(){
  let c, tries=0;
  do{
    c = { x: Math.floor(Math.random()*COLS), y: Math.floor(Math.random()*ROWS) };
    tries++;
  } while(tries < 500 && (snake.some(s=>s.x===c.x&&s.y===c.y) || (food && food.x===c.x&&food.y===c.y)));
  return c;
}
function startGame(){
  snake = [{x:6,y:8},{x:5,y:8},{x:4,y:8}];
  dir = {x:1,y:0}; dirQueue = [];
  food = freeCell();
  word = pickWord(); typed = 0; charged = false;
  score = 0; lives = 3; grow = 0; flashTimer = 0; pulse = 0;
  tickInterval = 150; lastTick = performance.now();
  state = 'playing';
  startScreen.classList.add('hidden');
  overScreen.classList.add('hidden');
}

function gameOver(){
  state = 'over';
  sfx.over();
  if(score > highScore){ highScore = score; saveHigh(); }
  finalScoreEl.textContent = score;
  finalHighEl.textContent = highScore;

const DIRS = {
  ArrowUp:{x:0,y:-1}, ArrowDown:{x:0,y:1}, ArrowLeft:{x:-1,y:0}, ArrowRight:{x:1,y:0},

window.addEventListener('keydown', function(e){
  audio();
  const k = e.key;
  if(k === ' ' || k === 'Enter'){
    e.preventDefault();
    if(state !== 'playing') startGame();
    return;
  }
  if(state !== 'playing') return;

  if(DIRS[k]){
    e.preventDefault();
    const d = DIRS[k];
    const last = dirQueue.length ? dirQueue[dirQueue.length-1] : dir;
    if((d.x !== -last.x || d.y !== -last.y) && (d.x !== last.x || d.y !== last.y)){
      if(dirQueue.length < 3) dirQueue.push(d);
    }
    return;
  }

  if(/^[a-zA-Z]$/.test(k)){
    const ch = k.toLowerCase();
    if(ch === word[typed].toLowerCase()){
      typed++;
      beep(300 + typed*60, .05, 'sine', .08);

        pulse = 1;
      }
    } else {
      lives--;
      flashTimer = 1;
      sfx.wrong();
      typed = 0;
      if(lives <= 0) gameOver();
    }
  }
});

/* ---------- Update ---------- */
function tick(now){
  if(dirQueue.length) dir = dirQueue.shift();
  const head = { x: snake[0].x + dir.x, y: snake[0].y + dir.y };

  if(head.x<0 || head.y<0 || head.x>=COLS || head.y>=ROWS ||
     snake.some(s=>s.x===head.x && s.y===head.y)){
    gameOver(); return;
  }
  snake.unshift(head);

  if(head.x === food.x && head.y === food.y){
    if(charged){ score += 10; grow += 2; sfx.chargeEat(); }
    else { score += 1; grow += 1; sfx.eat(); }
    tickInterval = Math.max(62, 150 - score*3);
    food = freeCell();
    word = pickWord(); typed = 0; charged = false;
  }

  if(grow > 0) grow--;
  else snake.pop();
}

/* ---------- Render ---------- */
function rr(x,y,w,h,r){
  ctx.beginPath();
  ctx.moveTo(x+r,y);
  ctx.arcTo(x+w,y,x+w,y+h,r);
  ctx.arcTo(x+w,y+h,x,y+h,r);
  ctx.arcTo(x,y+h,x,y,r);
  ctx.arcTo(x,y,x+w,y,r);
  ctx.closePath();
}

function draw(now){
  const W = window.innerWidth, H = window.innerHeight;
  ctx.clearRect(0,0,W,H);

  // board bg
  ctx.fillStyle = '#101828';
  rr(offX-4, offY-4, cell*COLS+8, cell*ROWS+8, 8); ctx.fill();

  if(charged){

    g.addColorStop(0,'rgba(255,209,102,.85)');

    ctx.fillStyle = g;
    ctx.beginPath(); ctx.arc(fx,fy,cell*1.4,0,Math.PI*2); ctx.fill();
  }
  const bob = Math.sin(now/300)*2;
  ctx.fillStyle = charged ? '#ffd166' : '#ff5d5d';
  ctx.beginPath(); ctx.arc(fx, fy+bob, cell*0.34, 0, Math.PI*2); ctx.fill();
  ctx.strokeStyle = '#7a4a1e'; ctx.lineWidth = 2;
  ctx.beginPath(); ctx.moveTo(fx, fy+bob-cell*0.3); ctx.lineTo(fx+cell*0.12, fy+bob-cell*0.5); ctx.stroke();
  ctx.fillStyle = '#6ef2a0';
  ctx.beginPath(); ctx.ellipse(fx+cell*0.18, fy+bob-cell*0.48, cell*0.13, cell*0.07, -0.5, 0, Math.PI*2); ctx.fill();

  // snake

    const t = i/snake.length;

    rr(offX+s.x*cell+1.5, offY+s.y*cell+1.5, cell-3, cell-3, i===0 ? 7 : 5);
    ctx.fill();
  }
  // eyes

  const ex = offX+h.x*cell + cell/2, ey = offY+h.y*cell + cell/2;

    ctx.beginPath(); ctx.arc(ox, oy, cell*0.09, 0, Math.PI*2); ctx.fill();

  ctx.font = 'bold 16px "Courier New", monospace';
  ctx.fillStyle = '#9fb8d8';
  ctx.textAlign = 'left';
  ctx.fillText('SCORE ' + score, 16, hudH/2);

  ctx.textAlign = 'right';
  let hearts = '';

  const cx = W/2;

    } else ctx.fillStyle = 'rgba(159,184,216,.4)';
    ctx.fillText(word[i], xStart + i*16, hudH/2);
  }
  if(charged){
    ctx.font = 'bold 13px "Courier New", monospace';
    ctx.fillStyle = Math.floor(now/200)%2===0 ? '#ffd166' : 'rgba(255,209,102,.4)';

    ctx.fillStyle = `rgba(255,60,60,${flashTimer*0.22})`;
    ctx.fillRect(0,0,W,H);
  }
}

/* ---------- Main loop ---------- */
let lastFrame = performance.now();
function loop(now){
  const dt = now - lastFrame;
  lastFrame = now;
  if(flashTimer > 0) flashTimer = Math.max(0, flashTimer - dt/300);
  if(pulse > 0) pulse = Math.max(0, pulse - dt/600);

  if(state === 'playing'){
    if(now - lastTick >= tickInterval){
      lastTick = now;
      tick(now);
    }
  }
  draw(now);
  requestAnimationFrame(loop);
}
