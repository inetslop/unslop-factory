<script>
(function(){
"use strict";
const canvas=document.getElementById('game');
const ctx=canvas.getContext('2d');

// logical court size
const W=960,H=640;
let scale=1,offX=0,offY=0;

function resize(){
  canvas.width=window.innerWidth;
  canvas.height=window.innerHeight;
  scale=Math.min(canvas.width/W,canvas.height/H);
  offX=(canvas.width-W*scale)/2;
  offY=(canvas.height-H*scale)/2;
}
window.addEventListener('resize',resize);
resize();

// ---------- audio ----------
let actx=null;
function initAudio(){
  if(!actx){
    try{actx=new (window.AudioContext||window.webkitAudioContext)();}catch(e){}
  }
  if(actx&&actx.state==='suspended')actx.resume();
}
function beep(freq,dur,type,vol,slide){
  if(!actx)return;
  const o=actx.createOscillator(),g=actx.createGain();
  o.type=type||'square';
  o.frequency.setValueAtTime(freq,actx.currentTime);
  if(slide)o.frequency.exponentialRampToValueAtTime(Math.max(30,freq+slide),actx.currentTime+dur);
  g.gain.setValueAtTime(vol||0.15,actx.currentTime);
  g.gain.exponentialRampToValueAtTime(0.0001,actx.currentTime+dur);
  o.connect(g);g.connect(actx.destination);
  o.start();o.stop(actx.currentTime+dur);
}

// ---------- game state ----------
const STATE={START:0,PLAY:1,OVER:2};
let state=STATE.START;
let paused=false;

const WIN_SCORE=7;
let playerScore=0,aiScore=0;
let winnerText='';

const paddleW=14,paddleH=96;
const ball={x:W/2,y:H/2,vx:0,vy:0,r:8,speed:340};
const player={x:26,y:H/2-paddleH/2,speed:460};
const ai={x:W-26-paddleW,y:H/2-paddleH/2,speed:395,err:0,errT:0};

let asteroids=[];
let particles=[];
let spawnTimer=1.2;
let serveTimer=0;
let time=0;

function resetBall(dir){
  ball.x=W/2;ball.y=H/2+ (Math.random()*160-80);
  const ang=(Math.random()*0.7-0.35);
  ball.speed=340;
  ball.vx=Math.cos(ang)*ball.speed*dir;
  ball.vy=Math.sin(ang)*ball.speed;
  serveTimer=1.0;
}

function resetGame(){
  playerScore=0;aiScore=0;
  player.y=H/2-paddleH/2;
  ai.y=H/2-paddleH/2;
  asteroids=[];particles=[];
  spawnTimer=1.5;
  paused=false;
  resetBall(Math.random()<0.5?1:-1);
}

function makeAsteroid(x,y,size){
  const r=size===3?34:size===2?22:13;
  const verts=[];
  const n=9+Math.floor(Math.random()*3);
  for(let i=0;i<n;i++){
    verts.push(r*(0.72+Math.random()*0.38));
  }
  const sp=(size===3?55:size===2?75:100);
  const a=Math.random()*Math.PI*2;
  return {x,y,vx:Math.cos(a)*sp,vy:Math.sin(a)*sp,r,size,
    rot:Math.random()*Math.PI*2,vrot:(Math.random()-0.5)*2.4,hp:size,verts};
}

function spawnAsteroid(){
  let tries=30;
  while(tries-->0){
    const size=Math.random()<0.6?3:2;
    const x=W*0.28+Math.random()*W*0.44;
    const y=70+Math.random()*(H-140);
    if(Math.hypot(x-ball.x,y-ball.y)<90)continue;
    asteroids.push(makeAsteroid(x,y,size));
    return;
  }
}

function burst(x,y,color,count){
  for(let i=0;i<count;i++){
    const a=Math.random()*Math.PI*2,s=60+Math.random()*220;
    particles.push({x,y,vx:Math.cos(a)*s,vy:Math.sin(a)*s,life:0.5+Math.random()*0.4,max:0.9,color});
  }
}

function destroyAsteroid(a){
  burst(a.x,a.y,'#c9b28a',14+a.size*6);
  beep(120+a.size*60,0.18,'sawtooth',0.12,-80);
  if(a.size>1){
    for(let i=0;i<2;i++){
      const na=makeAsteroid(a.x+(Math.random()-0.5)*20,a.y+(Math.random()-0.5)*20,a.size-1);
      na.vx=a.vx*0.6+(Math.random()-0.5)*80;
      na.vy=a.vy*0.6+(Math.random()-0.5)*80;
      asteroids.push(na);
    }
  }
}

// ---------- input ----------
const keys={};
window.addEventListener('keydown',function(e){
  initAudio();
  keys[e.code]=true;
  if(['ArrowUp','ArrowDown','Space'].indexOf(e.code)>=0)e.preventDefault();

  if(state===STATE.START&&(e.code==='Space'||e.code==='Enter')){
    resetGame();state=STATE.PLAY;beep(440,0.15,'square',0.12);
  }else if(state===STATE.OVER&&(e.code==='Space'||e.code==='Enter'||e.code==='KeyR')){
    resetGame();state=STATE.PLAY;beep(440,0.15,'square',0.12);
  }else if(state===STATE.PLAY){
    if(e.code==='KeyP')paused=!paused;
    if(e.code==='KeyR'){resetGame();}
  }
});
window.addEventListener('keyup',function(e){keys[e.code]=false;});
canvas.addEventListener('mousedown',function(){
  initAudio();
  if(state===STATE.START){resetGame();state=STATE.PLAY;}
  else if(state===STATE.OVER){resetGame();state=STATE.START?0:STATE.PLAY;}
});

// ---------- update ----------
function movePaddle(p,dt,up,down){
  if(up)p.y-=p.speed*dt;
  if(down)p.y+=p.speed*dt;
  p.y=Math.max(4,Math.min(H-paddleH-4,p.y));
}

function paddleHit(p,isPlayer){
  const hitOff=(ball.y-(p.y+paddleH/2))/(paddleH/2);
  const ang=hitOff*0.95; // max ~54deg
  ball.speed=Math.min(ball.speed+16,560);
  const dir=isPlayer?1:-1;
  ball.vx=Math.cos(ang)*ball.speed*dir;
  ball.vy=Math.sin(ang)*ball.speed;
  if(isPlayer){ball.x=p.x+paddleW+ball.r;}else{ball.x=p.x-ball.r;}
  beep(300,0.07,'square',0.14);
}

function update(dt){
  time+=dt;
  if(state!==STATE.PLAY||paused)return;

  // player
  movePaddle(player,dt,keys['KeyW']||keys['ArrowUp'],keys['KeyS']||keys['ArrowDown']);

  // serve delay (ball frozen briefly)
  if(serveTimer>0){
    serveTimer-=dt;
  }else{
    // AI
    ai.errT-=dt;
    if(ai.errT<=0){ai.err=(Math.random()-0.5)*46;ai.errT=0.35;}
    let targetY=ball.y+ai.err;
    if(ball.vx<0)targetY=H/2; // drift to center when ball moving away
    const c=ai.y+paddleH/2;
    const d=targetY-c;
    const step=Math.min(Math.abs(d),ai.speed*dt)*Math.sign(d);
    ai.y+=step;
    ai.y=Math.max(4,Math.min(H-paddleH-4,ai.y));

    // ball move (substeps for stability)
    const steps=3;
    for(let s=0;s<steps;s++){
      ball.x+=ball.vx*dt/steps;
      ball.y+=ball.vy*dt/steps;

      // walls
      if(ball.y-ball.r<0){ball.y=ball.r;ball.vy=Math.abs(ball.vy);beep(220,0.05,'triangle',0.1);}
      if(ball.y+ball.r>H){ball.y=H-ball.r;ball.vy=-Math.abs(ball.vy);beep(220,0.05,'triangle',0.1);}

      // player paddle
      if(ball.vx<0&&ball.x-ball.r<player.x+paddleW&&ball.x+ball.r>player.x-4&&
         ball.y>player.y-ball.r&&ball.y<player.y+paddleH+ball.r){
        paddleHit(player,true);
      }
      // AI paddle
      if(ball.vx>0&&ball.x+ball.r>ai.x&&ball.x-ball.r<ai.x+paddleW+4&&
         ball.y>ai.y-ball.r&&ball.y<ai.y+paddleH+ball.r){
        paddleHit(ai,false);
      }

      // asteroids
      for(let i=asteroids.length-1;i>=0;i--){
        const a=asteroids[i];
        const dx=ball.x-a.x,dy=ball.y-a.y;
        const dist=Math.hypot(dx,dy);
        const minD=ball.r+a.r*0.85;
        if(dist<minD&&dist>0.001){
          const nx=dx/dist,ny=dy/dist;
          ball.x=a.x+nx*minD;
          ball.y=a.y+ny*minD;
          const dot=ball.vx*nx+ball.vy*ny;
          if(dot<0){
            ball.vx-=2*dot*nx;
            ball.vy-=2*dot*ny;
            // keep speed
            const sp=Math.hypot(ball.vx,ball.vy);
            const t=ball.speed/Math.max(sp,1);
            ball.vx*=t;ball.vy*=t;
          }
          a.vx+=nx*55;a.vy+=ny*55;
          a.hp--;
          beep(480,0.06,'square',0.1,-200);
          if(a.hp<=0){destroyAsteroid(a);asteroids.splice(i,1);}
        }
      }

      // scoring
      if(ball.x<-30){aiScore++;scoreFx(-1);}
      else if(ball.x>W+30){playerScore++;scoreFx(1);}
    }
  }

  // asteroid spawn & motion
  spawnTimer-=dt;
  if(spawnTimer<=0&&asteroids.length<4){spawnAsteroid();spawnTimer=2.0+Math.random()*1.5;}

  for(let i=asteroids.length-1;i>=0;i--){
    const a=asteroids[i];
    a.x+=a.vx*dt;a.y+=a.vy*dt;
    a.rot+=a.vrot*dt;
    if(a.x-a.r<0||a.x+a.r>W)a.vx*=-1;
    if(a.y-a.r<0||a.y+a.r>H)a.vy*=-1;
    a.x=Math.max(a.r,Math.min(W-a.r,a.x));
    a.y=Math.max(a.r,Math.min(H-a.r,a.y));
    // slow asteroids over time (damping)
    a.vx*=Math.pow(0.5,dt/2.5);a.vy*=Math.pow(0.5,dt/2.5);
  }

  // particles
  for(let i=particles.length-1;i>=0;i--){
    const p=particles[i];
    p.x+=p.vx*dt;p.y+=p.vy*dt;
    p.vx*=Math.pow(0.05,dt);p.vy*=Math.pow(0.05,dt);
    p.life-=dt;
    if(p.life<=0)particles.splice(i,1);
  }

  // win check
  if(playerScore>=WIN_SCORE){endGame('YOU WIN!');}
  else if(aiScore>=WIN_SCORE){endGame('CPU WINS!');}
}

function scoreFx(dir){
  burst(ball.x<0?30:W-30,ball.y,'#7ecbff',26);
  beep(140,0.35,'sawtooth',0.16,-90);
  resetBall(dir>0?-1:1); // serve toward the player who just lost point? classic: loser serves; keep random-ish
}

function endGame(txt){
  winnerText=txt;
  state=STATE.OVER;
  beep(200,0.5,'sawtooth',0.18,-140);
}

// ---------- drawing ----------
function drawAsteroid(a){
  ctx.save();
  ctx.translate(a.x,a.y);
  ctx.rotate(a.rot);
  ctx.beginPath();
  const n=a.verts.length;
  for(let i=0;i<n;i++){
    const ang=i/n*Math.PI*2;
    const r=a.verts[i];
    if(i===0)ctx.moveTo(Math.cos(ang)*r,Math.sin(ang)*r);
    else ctx.lineTo(Math.cos(ang)*r,Math.sin(ang)*r);
  }
  ctx.closePath();
  ctx.strokeStyle='#c9b28a';
  ctx.lineWidth=2;
  ctx.stroke();
  // crater detail
  ctx.beginPath();
  ctx.arc(a.r*0.3,-a.r*0.15,a.r*0.16,0,Math.PI*2);
  ctx.stroke();
  ctx.restore();
}

function draw(){
  ctx.setTransform(1,0,0,1,0,0);
  ctx.fillStyle='#04060f';
  ctx.fillRect(0,0,canvas.width,canvas.height);
  ctx.setTransform(scale,0,0,scale,offX,offY);

  // court frame + center line
  ctx.strokeStyle='#1d2a4a';
  ctx.lineWidth=3;
  ctx.strokeRect(6,6,W-12,H-12);
  ctx.setLineDash([10,14]);
  ctx.beginPath();ctx.moveTo(W/2,10);ctx.lineTo(W/2,H-10);ctx.stroke();
  ctx.setLineDash([]);
  ctx.beginPath();ctx.arc(W/2,H/2,52,0,Math.PI*2);ctx.stroke();

  // particles
  for(const p of particles){
    ctx.globalAlpha=Math.max(0,p.life/p.max);
    ctx.fillStyle=p.color;
    ctx.fillRect(p.x-2,p.y-2,4,4);
  }
  ctx.globalAlpha=1;

  // asteroids
  for(const a of asteroids)drawAsteroid(a);

  // ball (hidden during serve delay, blinking)
  if(state===STATE.PLAY){
    if(!(serveTimer>0&&Math.floor(serveTimer*8)%2===0)){
      ctx.beginPath();ctx.arc(ball.x,ball.y,ball.r,0,Math.PI*2);
      ctx.fillStyle='#ffffff';ctx.fill();
      ctx.strokeStyle='rgba(126,203,255,0.8)';ctx.lineWidth=2;ctx.stroke();
    }
  }

  // paddles
  ctx.fillStyle='#7ecbff';
  ctx.fillRect(player.x,player.y,paddleW,paddleH);
  ctx.fillStyle='#ff7e9d';
  ctx.fillRect(ai.x,ai.y,paddleW,paddleH);

  // scores
  ctx.textAlign='center';
  ctx.font='bold 64px monospace';
  ctx.fillStyle='rgba(126,203,255,0.85)';
  ctx.fillText(playerScore,W/2-90,70);
  ctx.fillStyle='rgba(255,126,157,0.85)';
  ctx.fillText(aiScore,W/2+90,70);

  // overlays
  if(state===STATE.START){
    overlay('PONG ASTEROIDS',
      ['First to '+WIN_SCORE+' points wins.',
       'W / S  or  Arrow Keys — move paddle',
       'P — pause     R — restart',
       'Blast asteroids! They split when hit.'],
      'Press SPACE to start');
  }else if(state===STATE.PLAY&&paused){
    overlay('PAUSED',['','Press P to resume'],'');
  }else if(state===STATE.OVER){
    overlay(winnerText,
      ['Final score: '+playerScore+' — '+aiScore,''],
      'Press SPACE or R to play again');
  }
}

function overlay(title,lines,hint){
  ctx.fillStyle='rgba(4,6,15,0.78)';
  ctx.fillRect(0,0,W,H);
  ctx.textAlign='center';
  ctx.font='bold 62px monospace';
  ctx.fillStyle='#ffffff';
  ctx.fillText(title,W/2,H/2-90);
  ctx.font='20px monospace';
  ctx.fillStyle='#9fb4d8';
  lines.forEach((l,i)=>ctx.fillText(l,W/2,H/2-30+i*32));
  if(hint){
    ctx.font='bold 24px monospace';
    ctx.fillStyle=Math.floor(time*2)%2===0?'#7ecbff':'#ffffff';
    ctx.fillText(hint,W/2,H/2+110);
  }
}

// ---------- loop ----------
let last=performance.now();
function frame(now){
  let dt=(now-last)/1000;
  last=now;
  if(dt>0.05)dt=0.05;
  update(dt);
  draw();
  requestAnimationFrame(frame);
}
requestAnimationFrame(frame);
})();
</script>
