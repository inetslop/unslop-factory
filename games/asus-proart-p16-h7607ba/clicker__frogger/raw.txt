function beep(freq, dur, type, vol, slideTo){
  if(!actx) return;
  const t = actx.currentTime;
  const o = actx.createOscillator(), g = actx.createGain();
  o.type = type || 'square';

}

function update(dt){
  waterT += dt;
  if(msgTimer>0) msgTimer -= dt;
  if(state==='play') timeInLevel += dt;

    for(const it of L.items){

      const half = L.w/2;
      if(L.dir>0 && it.x > W + half) it.x = -half;
      if(L.dir<0 && it.x < -half) it.x = W + half;
    }
  }

  // frog hop
  if(frog.hop){
    const hp = frog.hop;
    hp.t += dt/hp.dur;
    if(hp.t>=1){ frog.x=hp.tx; frog.y=hp.ty; frog.hop=null; }
    else {
      const t = hp.t, e = 1 - Math.pow(1-t,2);
      frog.x = hp.sx + (hp.tx-hp.sx)*e;
      frog.y = hp.sy + (hp.ty-hp.sy)*e;
    }
  }

  if(state==='dead'){
    frog.deadT += dt;
    if(frog.deadT > 1.1){
      if(lives<=0) showGameOver();
      else { resetFrog(); state='play'; }
    }
    return;
  }
  if(state!=='play') return;

  const row = Math.floor(frog.y / CELL);

  // road collision
  if(row>=1 && row<=4){
    const L = lanes[row];
    for(const it of L.items){
      if(Math.abs(frog.x - it.x) < (L.w/2 + 13)){ die(); return; }
    }
  }

  // water
  if(row>=6 && row<=9){
    const L = lanes[row];
    let onLog = false, carried = 0;
    for(const it of L.items){
      if(frog.x > it.x - L.w/2 + 6 && frog.x < it.x + L.w/2 - 6){
        onLog = true;
        carried = L.dir * L.speed * dt;
        break;
      }
    }
    if(onLog){
      frog.drown = 0;
      frog.x += carried;
      if(frog.x < 14 || frog.x > W-14){ sfx.splash(); die(); return; }
    } else {
      frog.drown += dt;
      if(frog.drown > 0.65){ sfx.splash(); die(); return; }
    }
  } else frog.drown = 0;

  // goal check (only when settled)
  if(row===0 && !frog.hop){
    for(const c of GOAL_COLS){
      const hx = (c+0.5)*CELL;
      if(Math.abs(frog.x - hx) < CELL*0.38){ levelClear(); return; }
    }
  }
}

// ---------- draw ----------
function roundRect(x,y,w,h,r){
  ctx.beginPath();
  ctx.moveTo(x+r,y);
  ctx.arcTo(x+w,y,x+w,y+h,r);
  ctx.arcTo(x+w,y+h,x,y+h,r);
  ctx.arcTo(x,y+h,x,y,r);
  ctx.arcTo(x,y,x+w,y,r);
  ctx.closePath();
}

function drawBackground(){
  // grass rows
  ctx.fillStyle = '#2f7d3a';
  ctx.fillRect(0,0,W,H);
  // subtle grass texture stripes
  ctx.fillStyle = 'rgba(255,255,255,0.04)';
  for(let r=0;r<ROWS;r++) if(r%2===0) ctx.fillRect(0,r*CELL,W,CELL);

  // road rows 1..4
  for(let r=1;r<=4;r++){
    ctx.fillStyle = '#3a3f47';
    ctx.fillRect(0,r*CELL,W,CELL);
  }
  // lane dashes
  ctx.strokeStyle = 'rgba(255,220,90,0.5)';
  ctx.lineWidth = 2;
  ctx.setLineDash([14,12]);
  for(let r=2;r<=4;r++){
    ctx.beginPath(); ctx.moveTo(0,r*CELL); ctx.lineTo(W,r*CELL); ctx.stroke();
  }
  ctx.setLineDash([]);

  // water rows 6..9
  for(let r=6;r<=9;r++){
    const g = ctx.createLinearGradient(0,r*CELL,0,(r+1)*CELL);
    g.addColorStop(0,'#1b5e8a'); g.addColorStop(1,'#144a70');
    ctx.fillStyle = g;
    ctx.fillRect(0,r*CELL,W,CELL);
  }
  // animated ripples
  ctx.strokeStyle = 'rgba(255,255,255,0.13)';
  ctx.lineWidth = 2;
  for(let r=6;r<=9;r++){
    const off = (waterT*22 + r*17) % 48;
    for(let x=-48+off; x<W; x+=48){
      const y = r*CELL + CELL/2 + Math.sin(x*0.08 + waterT*3)*4;
      ctx.beginPath();
      ctx.moveTo(x, y-3);
      ctx.quadraticCurveTo(x+12, y-7, x+24, y-3);
      ctx.stroke();
    }
  }

  // borders top/bottom
  ctx.fillStyle = '#1c5a28';
  ctx.fillRect(0,0,W,4);
  ctx.fillRect(0,H-4,W,4);

  // goal holes
  for(const c of GOAL_COLS){
    const cx=(c+0.5)*CELL, cy=CELL/2;
    ctx.fillStyle = '#173a1e';
    ctx.beginPath(); ctx.arc(cx,cy,15,0,Math.PI*2); ctx.fill();
    ctx.strokeStyle='#0d2410'; ctx.lineWidth=3; ctx.stroke();
    // lily pad
    ctx.fillStyle = '#4caf50';
    ctx.beginPath(); ctx.moveTo(cx,cy); ctx.arc(cx,cy,11,0.5,Math.PI*2-0.1); ctx.closePath(); ctx.fill();
    ctx.strokeStyle='#2e7d32'; ctx.lineWidth=1.5; ctx.stroke();
  }
}

function drawLanes(){
  for(let r=0;r<ROWS;r++){
    const L = lanes[r];
    if(!L) continue;
    const y = (r+0.5)*CELL;
    for(const it of L.items){
      const x = it.x;
      if(L.type==='car'){
        // shadow
        ctx.fillStyle='rgba(0,0,0,0.3)';
        roundRect(x-L.w/2-2, y-L.h/2+3, L.w, L.h, 6); ctx.fill();
        // body
        ctx.fillStyle = L.color;
        roundRect(x-L.w/2, y-L.h/2, L.w, L.h, 6); ctx.fill();
        // roof / windshield
        ctx.fillStyle='rgba(0,0,0,0.35)';
        const dir = L.dir;
        roundRect(x - L.w*0.18 + dir*L.w*0.08, y-L.h/2+4, L.w*0.36, L.h-8, 4); ctx.fill();
        // wheels
        ctx.fillStyle='#111';
        ctx.fillRect(x-L.w/2+5, y-L.h/2-2, 9, 4);
        ctx.fillRect(x+L.w/2-14, y-L.h/2-2, 9, 4);
        ctx.fillRect(x-L.w/2+5, y+L.h/2-2, 9, 4);
        ctx.fillRect(x+L.w/2-14, y+L.h/2-2, 9, 4);
      } else {
        // log
        ctx.fillStyle='rgba(0,0,0,0.25)';
        roundRect(x-L.w/2-2, y-L.h/2+3, L.w, L.h, 8); ctx.fill();
        const g = ctx.createLinearGradient(0,y-L.h/2,0,y+L.h/2);
        g.addColorStop(0,'#a5713b'); g.addColorStop(0.5,'#8a5a2c'); g.addColorStop(1,'#6e4520');
        ctx.fillStyle=g;
        roundRect(x-L.w/2, y-L.h/2, L.w, L.h, 8); ctx.fill();
        // grain
        ctx.strokeStyle='rgba(60,35,12,0.5)';
        ctx.lineWidth=1.5;
        for(let i=1;i<4;i++){
          const gx = x - L.w/2 + (L.w/4)*i;
          ctx.beginPath(); ctx.moveTo(gx, y-L.h/2+4); ctx.lineTo(gx, y+L.h/2-4); ctx.stroke();
        }
      }
    }
  }
}

function drawFrog(){
  const x = frog.x, y = frog.y;
  let scale = 1;
  if(frog.hop){
    const t = frog.hop.t;
    scale = 1 + Math.sin(t*Math.PI)*0.35;
  }
  ctx.save();
  ctx.translate(x,y);
  if(frog.dead){
    const p = Math.min(1, frog.deadT/0.9);
    ctx.rotate(p*Math.PI*1.6);
    ctx.globalAlpha = 1-p;
    ctx.scale(1-p*0.5, 1-p*0.5);
  } else {
    ctx.scale(scale, 2-scale);
  }
  // shadow
  if(!frog.dead){
    ctx.fillStyle='rgba(0,0,0,0.3)';
    ctx.beginPath(); ctx.ellipse(0,6,14,8,0,0,Math.PI*2); ctx.fill();
  }
  // body
  ctx.fillStyle = frog.drown>0 ? '#c9e265' : '#6fd44b';
  ctx.beginPath(); ctx.ellipse(0,0,13,11,0,0,Math.PI*2); ctx.fill();
  ctx.strokeStyle='#2e7d32'; ctx.lineWidth=2; ctx.stroke();
  // belly
  ctx.fillStyle='rgba(255,255,255,0.35)';
  ctx.beginPath(); ctx.ellipse(0,3,8,6,0,0,Math.PI*2); ctx.fill();
  // eyes
  for(const s of [-1,1]){
    ctx.fillStyle='#fff';
    ctx.beginPath(); ctx.arc(s*7,-9,5,0,Math.PI*2); ctx.fill();
    ctx.strokeStyle='#2e7d32'; ctx.lineWidth=1.5; ctx.stroke();
    ctx.fillStyle='#111';
    ctx.beginPath(); ctx.arc(s*7+ (s<0?-1:1),-9,2.3,0,Math.PI*2); ctx.fill();
  }
  // feet
  ctx.fillStyle = frog.drown>0 ? '#c9e265' : '#6fd44b';
  for(const s of [-1,1]){
    ctx.beginPath(); ctx.ellipse(s*11,7,5,3.5,s*0.5,0,Math.PI*2); ctx.fill();
    ctx.strokeStyle='#2e7d32'; ctx.lineWidth=1.5; ctx.stroke();
  }
  ctx.restore();

  // drowning bubbles
  if(frog.drown>0 && !frog.dead){
    ctx.fillStyle='rgba(255,255,255,0.8)';
    for(let i=0;i<3;i++){
      const bt = (waterT*2 + i*0.7)%1;
      ctx.beginPath();
      ctx.arc(x + Math.sin(i*2+waterT*4)*8, y - 8 - bt*16, 2+i%2, 0, Math.PI*2);
      ctx.fill();
    }
  }
}

function drawHUD(){
  // top strip
  ctx.fillStyle='rgba(0,0,0,0.55)';
  ctx.fillRect(0,0,W,18);
  ctx.font='bold 12px "Courier New", monospace';
  ctx.textBaseline='middle';
  ctx.fillStyle='#ffe066';
  ctx.textAlign='left';
  ctx.fillText('SCORE '+score, 8, 10);
  ctx.textAlign='center';
  ctx.fillStyle='#7dff8a';
  ctx.fillText('LEVEL '+level, W/2, 10);
  ctx.textAlign='right';
  let s='';
  for(let i=0;i<Math.max(0,lives);i++) s+='🐸';
  ctx.fillStyle='#fff';
  ctx.fillText(s || '—', W-8, 10);

  if(msgTimer>0 && (state==='play'||state==='levelup')){
    const a = Math.min(1, msgTimer/0.4);
    ctx.globalAlpha = a;
    ctx.fillStyle='rgba(0,0,0,0.6)';
    ctx.fillRect(W/2-130,H/2-30,260,58);
    ctx.strokeStyle='#7dff8a'; ctx.lineWidth=2;
    ctx.strokeRect(W/2-130,H/2-30,260,58);
    ctx.fillStyle='#7dff8a';
    ctx.font='bold 24px "Courier New", monospace';
    ctx.textAlign='center';
    ctx.fillText(msgText, W/2, H/2+1);
    ctx.globalAlpha=1;
  }
}

function draw(){
  drawBackground();
  drawLanes();
  if(!(state==='over')) drawFrog();
  drawHUD();
}

// ---------- loop ----------
let last = performance.now();
function loop(now){
  let dt = (now - last)/1000;
  last = now;
  if(dt > 0.05) dt = 0.05;
  if(state==='play' || state==='dead' || state==='levelup'){
    update(dt);
    if(state==='levelup' && msgTimer<=0) nextLevel();
  } else {
    waterT += dt; // keep water animating behind menus
  }
  draw();
  requestAnimationFrame(loop);
}

// init idle scene behind start screen
buildLanes(); resetFrog();
requestAnimationFrame(loop);

})();
