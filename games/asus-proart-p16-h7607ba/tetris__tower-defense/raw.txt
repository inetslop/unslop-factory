// ---------- Audio ----------
let actx = null;
function initAudio(){ if(!actx){ try{ actx = new (window.AudioContext||window.webkitAudioContext)(); }catch(e){} } if(actx && actx.state==='suspended') actx.resume(); }
function beep(freq, dur, type, vol, slideTo){
  if(!actx) return;
  try{
    const o = actx.createOscillator(), g = actx.createGain();
    o.type = type||'square'; o.frequency.value = freq;
    if(slideTo) o.frequency.exponentialRampToValueAtTime(Math.max(30,slideTo), actx.currentTime+dur);
    g.gain.value = vol||0.08;
    g.gain.exponentialRampToValueAtTime(0.0001, actx.currentTime+dur);
    o.connect(g); g.connect(actx.destination);
    o.start(); o.stop(actx.currentTime+dur);
  }catch(e){}
}
const sfx = {
  shoot: ()=>beep(880,0.07,'square',0.03,1400),
  hit:   ()=>beep(240,0.09,'sawtooth',0.05,120),
  boom:  ()=>beep(160,0.3,'sawtooth',0.1,40),

};

  enemies = []; bullets = []; particles = [];

  newPiece();
}

function rotCW(m){
  const n=m.length, out=[];
  for(let r=0;r<n;r++){ out.push([]); for(let c=0;c<n;c++) out[r].push(m[n-1-c][r]); }
  return out;
}
function newPiece(){
  const def = SHAPES[Math.floor(Math.random()*SHAPES.length)];
  const m = def.m.map(r=>r.slice());
  piece = { m, c:def.c, x:Math.floor((COLS-1-m[0].length)/2), y:-m.length+ (m.length>1?1:2) };
  // start just above top
  piece.y = -m.length + 2;
  if(!valid(piece.m, piece.x, piece.y)){
    baseHP -= 3; sfx.hurt(); shake=8;
    piece = null;
    if(baseHP<=0) gameOver();
  }
}
function valid(m, px, py){
  for(let r=0;r<m.length;r++)for(let c=0;c<m[r].length;c++){
    if(!m[r][c]) continue;
    const gx=px+c, gy=py+r;
    if(gx<0||gx>=COLS||gy>=ROWS) return false;
    if(gy>=0 && grid[gy][gx]) return false;
  }
  return true;
}
function lockPiece(){
  let below = false;
  for(let r=0;r<piece.m.length;r++)for(let c=0;c<piece.m[r].length;c++){
    if(!piece.m[r][c]) continue;
    const gx=piece.x+c, gy=piece.y+r;
    if(gy>=0){ grid[gy][gx] = {hp:6, max:6, c:piece.c, cd:Math.random()*1.2}; below=true; }
  }
  sfx.lock();
  for(let i=0;i<6;i++) spawnParticles(ox+(piece.x+1)*cell, oy+(piece.y+1)*cell, piece.c, 3);
  piece = null;
  if(below) newPiece();
}
function canDown(){ return valid(piece.m, piece.x, piece.y+1); }

// ---------- Enemies / bullets ----------
function spawnEnemy(){
  const r = Math.floor(Math.random()*ROWS);

}
function spawnParticles(x,y,c,n){
  for(let i=0;i<n;i++){
    const a=Math.random()*Math.PI*2, s=(1+Math.random()*3);
    particles.push({x,y,vx:Math.cos(a)*s,vy:Math.sin(a)*s,life:0.5+Math.random()*0.4,c});
  }
}
function damageBase(n){ baseHP-=n; shake=6; sfx.hurt(); }

function update(dt){
  // waves
  waveTimer += dt;
  if(waveTimer > 18){ waveTimer = 0; wave++; sfx.wave(); }

  // spawn enemies
  spawnTimer -= dt;
  const interval = Math.max(0.9, 2.6 - wave*0.15);
  if(spawnTimer<=0){ spawnTimer = interval*(0.7+Math.random()*0.6); spawnEnemy(); }

  // piece gravity
  if(piece){
    dropTimer += dt;
    const grav = Math.max(0.18, 0.55 - wave*0.02);
    if(dropTimer>grav){ dropTimer=0;
      if(canDown()) piece.y++; else lockPiece();
    }
  }

  // blocks fire
  for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++){
    const b = grid[r][c]; if(!b) continue;
    b.cd -= dt;
    if(b.cd<=0){
      let target=null;
      for(const e of enemies) if(e.r===r && e.x>c+0.3 && (!target||e.x<target.x)) target=e;
      if(target){ bullets.push({x:c+0.5, r}); b.cd = 1.0 + Math.random()*0.25; sfx.shoot(); }
      else b.cd = 0.1;
    }
  }

  // bullets
  for(let i=bullets.length-1;i>=0;i--){
    const bl = bullets[i];
    bl.x += 9*dt;
    let hit=false;
    for(const e of enemies){
      if(e.r===bl.r && Math.abs(e.x-bl.x)<0.45 && e.x>bl.x-0.3){
        e.hp--; e.flash=0.12; sfx.hit(); hit=true;
        spawnParticles(ox+e.x*cell, oy+(e.r+0.5)*cell, '#ffd76a', 4);
        if(e.hp<=0){ enemies.splice(enemies.indexOf(e),1); score+=10; sfx.boom(); shake=3;
          spawnParticles(ox+e.x*cell, oy+(e.r+0.5)*cell, '#ff7b4d', 12); }
        break;
      }
    }
    if(hit || bl.x>COLS+1) bullets.splice(i,1);
  }

  // enemies
  const speed = 0.7 + wave*0.05;
  for(const e of enemies){
    e.flash = Math.max(0, e.flash - dt);
    e.atkT -= dt;
    let blocked=false;
    if(e.x-0.4 <= 0.25){ e.x=0.65; blocked=true; }
    else{
      const frontCol = Math.floor(e.x-0.45);

      if(!blocked) e.x -= speed*dt;

      if(frontCol>=0 && grid[e.r][frontCol]){
        const b = grid[e.r][frontCol];
        b.hp--; sfx.hit();
        spawnParticles(ox+(frontCol+0.5)*cell, oy+(e.r+0.5)*cell, b.c, 4);
        if(b.hp<=0){
          grid[e.r][frontCol]=null; sfx.boom(); shake=3;
          spawnParticles(ox+(frontCol+0.5)*cell, oy+(e.r+0.5)*cell, b.c, 10);
        }
      } else damageBase(2);
    }
  }

  // particles
  for(let i=particles.length-1;i>=0;i--){
    const p=particles[i];
    p.life-=dt; p.x+=p.vx*cell*dt; p.y+=p.vy*cell*dt; p.vy+=6*dt;
    if(p.life<=0) particles.splice(i,1);

  if(baseHP<=0) gameOver();
}

    const x=ox+e.x*cell, y=oy+(e.r+0.5)*cell, rad=cell*0.34;
    ctx.save();
    ctx.fillStyle = e.flash>0 ? '#ffffff' : '#ff4d6d';
    ctx.shadowColor='#ff2255'; ctx.shadowBlur=12;
    ctx.beginPath(); ctx.arc(x,y,rad,0,Math.PI*2); ctx.fill();
    ctx.restore();
    // eyes
    ctx.fillStyle='#fff';
    ctx.beginPath(); ctx.arc(x-rad*0.35,y-rad*0.15,rad*0.22,0,Math.PI*2); ctx.fill();
    ctx.beginPath(); ctx.arc(x+rad*0.1,y-rad*0.15,rad*0.22,0,Math.PI*2); ctx.fill();
    ctx.fillStyle='#222';
    ctx.beginPath(); ctx.arc(x-rad*0.42,y-rad*0.15,rad*0.1,0,Math.PI*2); ctx.fill();
    ctx.beginPath(); ctx.arc(x+rad*0.03,y-rad*0.15,rad*0.1,0,Math.PI*2); ctx.fill();
    // hp bar
    if(e.hp<e.maxHp){
      ctx.fillStyle='rgba(0,0,0,0.5)'; ctx.fillRect(x-rad,y-rad-8,rad*2,4);
      ctx.fillStyle='#7dff9b'; ctx.fillRect(x-rad,y-rad-8,rad*2*(e.hp/e.maxHp),4);
    }
  }

  // particles
  for(const p of particles){
    ctx.globalAlpha=Math.max(0,p.life);
    ctx.fillStyle=p.c;
    ctx.fillRect(p.x-2,p.y-2,4,4);
  }
  ctx.globalAlpha=1;
  ctx.restore();

  // HUD
  ctx.fillStyle='rgba(15,20,40,0.9)'; ctx.fillRect(0,0,W,hudH);
  ctx.fillStyle='#eaf2ff'; ctx.font='bold '+Math.floor(hudH*0.38)+'px Segoe UI, Arial';
  ctx.textBaseline='middle';
  ctx.textAlign='left';
  ctx.fillText('SCORE '+score, 16, hudH*0.5);
  ctx.textAlign='center';
  ctx.fillStyle='#b45cff';
  ctx.fillText('WAVE '+wave, W/2, hudH*0.5);
  // base hp bar
  const bw=Math.min(220,W*0.3), bx2=W-bw-16;
  ctx.textAlign='right'; ctx.fillStyle='#8fa3c8';
  ctx.font='bold '+Math.floor(hudH*0.28)+'px Segoe UI, Arial';
  ctx.fillText('BASE', W-16-hudH*0.5, hudH*0.5);
  ctx.fillStyle='rgba(0,0,0,0.5)'; ctx.fillRect(bx2,hudH*0.32,bw,hudH*0.36);
  const pct=Math.max(0,baseHP/12);
  ctx.fillStyle = pct>0.4 ? '#37c8ff' : '#ff5d5d';
  ctx.fillRect(bx2,hudH*0.32,bw*pct,hudH*0.36);
}

// ---------- Input ----------
function move(dx){ if(!piece) return; if(valid(piece.m,piece.x+dx,piece.y)) piece.x+=dx; }
function rotate(){
  if(!piece) return;
  const m = rotCW(piece.m);
  const kicks=[0,-1,1,-2,2];
  for(const k of kicks){ if(valid(m,piece.x+k,piece.y)){ piece.m=m; piece.x+=k; break; } }
}
function hardDrop(){
  if(!piece) return;
  let n=0;
  while(valid(piece.m,piece.x,piece.y+1)){ piece.y++; n++; }
  score += n; sfx.drop(); lockPiece();
}
window.addEventListener('keydown', function(e){
  initAudio();
  const k=e.key;
  if(['ArrowLeft','ArrowRight','ArrowDown','ArrowUp',' '].includes(k)) e.preventDefault();
  if(state==='start' && (k==='Enter'||k===' ')){ startGame(); return; }
  if(state==='over' && (k==='r'||k==='R'||k==='Enter')){ startGame(); return; }
  if(state!=='playing') return;
  if(k==='ArrowLeft') move(-1);
  else if(k==='ArrowRight') move(1);
  else if(k==='ArrowDown'){ if(piece&&canDown()){ piece.y++; dropTimer=0; score+=1; } }
  else if(k==='ArrowUp'||k==='x'||k==='X') rotate();
  else if(k===' ') hardDrop();
  else if(k==='r'||k==='R') startGame();
});

  initAudio();
  document.getElementById('startScreen').classList.add('hidden');
  document.getElementById('overScreen').classList.add('hidden');
  reset();
  state='playing';
}
document.getElementById('startBtn').addEventListener('click', startGame);
document.getElementById('restartBtn').addEventListener('click', startGame);

  let dt=(t-last)/1000; last=t;
  if(dt>0.1) dt=0.1;
  if(state==='playing') update(dt);
  draw();
  requestAnimationFrame(loop);
}
