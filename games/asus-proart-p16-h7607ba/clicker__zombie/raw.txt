function beep(freq, dur, type='square', vol=0.12, slide=0) {
  if (!AC) return;
  try {
    const t = AC.currentTime;
    const o = AC.createOscillator(), g = AC.createGain();
    o.type = type; o.frequency.setValueAtTime(freq, t);
    if (slide) o.frequency.exponentialRampToValueAtTime(Math.max(30, freq + slide), t + dur);
    g.gain.setValueAtTime(vol, t);
    g.gain.exponentialRampToValueAtTime(0.001, t + dur);
    o.connect(g).connect(AC.destination);
    o.start(t); o.stop(t + dur + 0.02);
  } catch(e){}
}

function startGame() {
  G.state = 'play';
  G.brains = 0; G.totalBrains = 0;
  G.hp = 100; G.maxHp = 100;
  G.wave = 0; G.waveTimer = 1.2; G.spawnTimer = 0; G.toSpawn = 0;
  G.zombies.length = 0; G.particles.length = 0; G.texts.length = 0; G.bullets.length = 0;

function killZombie(z) {
  const idx = G.zombies.indexOf(z);
  if (idx >= 0) G.zombies.splice(idx, 1);
  G.kills++;
  const reward = Math.round((z.big ? 14 : 5) * (1 + G.wave * 0.08));
  G.brains += reward; G.totalBrains += reward;
  blood(z.x, z.y - z.r, z.big ? 26 : 14);
  addText(z.x, z.y - z.r*2.2, '+' + reward + ' 🧠', '#ff9ecb', 18);
  sfx.kill();
}

/* ---------- update ---------- */
let last = performance.now();
function update(dt) {
  if (G.state !== 'play') return;
  G.time += dt;
  G.shake = Math.max(0, G.shake - dt*30);

  // waves
  if (G.toSpawn === undefined || G.toSpawn <= 0) {
    if (G.zombies.length === 0) {
      G.waveTimer -= dt;
      if (G.waveTimer <= 0) startWave();
    }
  } else {
    G.spawnTimer -= dt;
    if (G.spawnTimer <= 0) {
      spawnZombie();
      G.toSpawn--;
      const interval = Math.max(0.35, 1.6 - G.wave*0.08);
      G.spawnTimer = interval * rand(0.6, 1.3);
    }
  }

  // zombies
  for (let i = G.zombies.length-1; i >= 0; i--) {
    const z = G.zombies[i];
    z.x -= z.speed * dt;
    z.wob += dt * 8;
    z.hitFlash = Math.max(0, z.hitFlash - dt);
    if (z.x <= baseX() + 34) {
      // hits the house
      const dmg = z.big ? 25 : 12;
      G.hp -= dmg;
      G.shake = 12;
      blood(z.x, z.y - z.r, 16, '#ff3333');
      addText(baseX()+50, H*0.4, '-' + dmg + ' HP', '#ff6b6b', 24);
      sfx.hurt();
      G.zombies.splice(i,1);
      if (G.hp <= 0) { G.hp = 0; gameOver(); return; }
    }
  }

  // turret
  const tv = G.up[1].lvl;
  if (tv > 0 && G.zombies.length) {
    G.turretCooldown = (G.turretCooldown || 0) - dt;
    let target = null, tx = -1e9;
    for (const z of G.zombies) if (z.x < tx) { tx = z.x; target = z; } // closest to base
    const bx = baseX() + 30, by = groundY() - 70;
    G.turretAngle = Math.atan2((target.y - target.r) - by, target.x - bx);
    if (G.turretCooldown <= 0 && target) {
      G.turretCooldown = Math.max(0.18, 0.6 - tv*0.04);
      const rateDmg = turretDPS() * (0.6 - Math.min(0.42, tv*0.04));
      target.hp -= rateDmg;
      target.hitFlash = 0.1;
      G.bullets.push({x:bx, y:by, tx:target.x, ty:target.y-target.r, life:.09});
      blood(target.x, target.y - target.r, 4);
      sfx.shot();
      if (target.hp <= 0) killZombie(target);
    }
  }

  // particles / texts / bullets
  for (let i=G.particles.length-1;i>=0;i--) {
    const p = G.particles[i];
    p.life -= dt; p.x += p.vx*dt; p.y += p.vy*dt; p.vy += 500*dt;
    if (p.life <= 0) G.particles.splice(i,1);
  }
  for (let i=G.texts.length-1;i>=0;i--) {
    const t = G.texts[i];
    t.life -= dt*0.9; t.y -= 40*dt;
    if (t.life <= 0) G.texts.splice(i,1);
  }
  for (let i=G.bullets.length-1;i>=0;i--) {
    const b = G.bullets[i];
    b.life -= dt;
    if (b.life <= 0) G.bullets.splice(i,1);
  }
}

function drawZombie(z) {
  const bob = Math.sin(z.wob) * 3;
  const x = z.x, y = z.y + bob, r = z.r;
  ctx.save();
  // shadow
  ctx.fillStyle = 'rgba(0,0,0,.4)';
  ctx.beginPath(); ctx.ellipse(x, z.y+8, r*0.9, r*0.28, 0, 0, Math.PI*2); ctx.fill();
  const flash = z.hitFlash > 0;
  // legs
  ctx.strokeStyle = '#2a3d24'; ctx.lineWidth = r*0.28; ctx.lineCap='round';
  ctx.beginPath();
  ctx.moveTo(x - r*0.25, y); ctx.lineTo(x - r*0.4 + Math.sin(z.wob)*r*0.25, z.y+8);
  ctx.moveTo(x + r*0.25, y); ctx.lineTo(x + r*0.4 - Math.sin(z.wob)*r*0.25, z.y+8);
  ctx.stroke();
  // body
  ctx.fillStyle = flash ? '#ffffff' : z.color;
  ctx.beginPath(); ctx.ellipse(x, y - r*0.9, r*0.75, r*0.95, 0, 0, Math.PI*2); ctx.fill();
  // arms reaching forward (left)
  const armWob = Math.sin(z.wob*1.3)*r*0.15;
  ctx.strokeStyle = flash ? '#ffffff' : z.color; ctx.lineWidth = r*0.3;
  ctx.beginPath();
  ctx.moveTo(x - r*0.4, y - r*1.2); ctx.lineTo(x - r*1.4, y - r*1.1 + armWob);
  ctx.moveTo(x - r*0.4, y - r*0.9); ctx.lineTo(x - r*1.4, y - r*0.8 - armWob);
  ctx.stroke();
  // head
  ctx.fillStyle = flash ? '#ffffff' : z.color;
  ctx.beginPath(); ctx.arc(x - r*0.25, y - r*1.9, r*0.55, 0, Math.PI*2); ctx.fill();
  // eyes
  ctx.fillStyle = '#ff3b3b';
  ctx.beginPath();
  ctx.arc(x - r*0.45, y - r*1.95, r*0.09, 0, Math.PI*2);
  ctx.arc(x - r*0.05, y - r*1.95, r*0.09, 0, Math.PI*2);
  ctx.fill();
  // mouth
  ctx.strokeStyle = '#3a1414'; ctx.lineWidth = 2;
  ctx.beginPath(); ctx.moveTo(x - r*0.5, y - r*1.7); ctx.lineTo(x - r*0.15, y - r*1.68); ctx.stroke();
  // hp bar
  if (z.hp < z.maxHp) {
    const w = r*1.6;
    ctx.fillStyle = 'rgba(0,0,0,.6)';
    ctx.fillRect(x - w/2, y - r*2.75, w, 5);
    ctx.fillStyle = '#ff5b5b';
    ctx.fillRect(x - w/2, y - r*2.75, w * Math.max(0, z.hp/z.maxHp), 5);
  }
  if (z.big) {
    ctx.fillStyle = 'rgba(255,220,80,.9)';
    ctx.font = `bold ${r*0.4}px Courier New`;
    ctx.textAlign='center';
    ctx.fillText('BRUTE', x, y - r*3);
  }
  ctx.restore();
}

    ctx.font = `${fs*0.85}px Courier New`;
    ctx.fillStyle = afford ? '#ffe66b' : '#8a6a4a';
    ctx.fillText(`🧠 ${c}`, x+8, y + bh - 10);
  }
}

function roundRect(x,y,w,h,r) {
  ctx.beginPath();
  ctx.moveTo(x+r,y);
  ctx.arcTo(x+w,y,x+w,y+h,r);
  ctx.arcTo(x+w,y+h,x,y+h,r);
  ctx.arcTo(x,y+h,x,y,r);
  ctx.arcTo(x,y,x+w,y,r);
  ctx.closePath();
}

function render() {
  ctx.save();
  if (G.shake > 0) ctx.translate(rand(-G.shake,G.shake), rand(-G.shake,G.shake));

  // sky
  const grad = ctx.createLinearGradient(0,0,0,H);
  grad.addColorStop(0,'#0b1226');
  grad.addColorStop(0.55,'#1a2438');
  grad.addColorStop(0.78,'#2a2f3a');
  grad.addColorStop(1,'#1c2418');
  ctx.fillStyle = grad;
  ctx.fillRect(-20,-20,W+40,H+40);

  // stars
  ctx.fillStyle = 'rgba(255,255,255,.5)';
  for (let i=0;i<40;i++) {
    const sx = (i*137.5 % W), sy = (i*89.7 % (H*0.5));
    ctx.fillRect(sx, sy, 2, 2);
  }
  // moon
  ctx.fillStyle = '#e8e6c9';
  ctx.beginPath(); ctx.arc(W*0.82, H*0.15, Math.min(38, W*0.04), 0, Math.PI*2); ctx.fill();
  ctx.fillStyle = 'rgba(232,230,201,.15)';
  ctx.beginPath(); ctx.arc(W*0.82, H*0.15, Math.min(60, W*0.06), 0, Math.PI*2); ctx.fill();

  // ground
  const gy = groundY() + 8;
  ctx.fillStyle = '#243020';
  ctx.fillRect(-20, gy, W+40, H - gy + 20);
  ctx.strokeStyle = 'rgba(120,160,90,.25)';
  ctx.lineWidth = 1;
  for (let x=0; x<W; x+=38) {
    ctx.beginPath(); ctx.moveTo(x, gy+4); ctx.lineTo(x-6, gy+12); ctx.stroke();
  }

  if (G.state !== 'menu') {
    drawHouse();
    // bullets
    for (const b of G.bullets) {
      ctx.strokeStyle = 'rgba(255,230,120,.9)';
      ctx.lineWidth = 2;
      ctx.beginPath(); ctx.moveTo(b.x, b.y); ctx.lineTo(b.tx, b.ty); ctx.stroke();
    }
    const zs = [...G.zombies].sort((a,b)=>a.y-b.y);
    for (const z of zs) drawZombie(z);

    // particles
    for (const p of G.particles) {
      ctx.globalAlpha = Math.max(0, p.life / p.max);
      ctx.fillStyle = p.color;
      ctx.beginPath(); ctx.arc(p.x, p.y, p.r, 0, Math.PI*2); ctx.fill();
    }
    ctx.globalAlpha = 1;

    // floating texts
    for (const t of G.texts) {
      ctx.globalAlpha = Math.max(0, t.life);
      ctx.fillStyle = t.color;
      ctx.font = `bold ${t.size}px Courier New`;
      ctx.textAlign = 'center';
      ctx.fillText(t.str, t.x, t.y);
    }
    ctx.globalAlpha = 1;

    drawShop();

    // HUD
    const fs = Math.min(20, W*0.025);
    ctx.font = `bold ${fs}px Courier New`;
    ctx.textAlign='left';
    ctx.fillStyle = '#ff9ecb';
    ctx.fillText('🧠 ' + Math.floor(G.brains), 14, fs + 8);
    ctx.fillStyle = '#cfe8cf';
    ctx.fillText('WAVE ' + G.wave, 14, fs*2 + 10);
    const m = Math.floor(G.time/60), s = Math.floor(G.time%60);
    ctx.fillText(`⏱ ${m}:${s.toString().padStart(2,'0')}   ☠ ${G.kills}`, 14, fs*3 + 12);
    ctx.textAlign='right';
    ctx.fillStyle = '#9fbf9f';
    ctx.font = `${fs*0.8}px Courier New`;
    ctx.fillText(`click dmg: ${clickDamage()}   crit: ${Math.round(critChance()*100)}%`, W-14, fs + 6);
    ctx.fillText('[R] restart', W-14, fs*2);
  } else {
    // ambient menu zombies
    ctx.fillStyle = 'rgba(0,0,0,.3)';
    ctx.fillRect(-20, gy, W+40, H-gy+20);
  }
  ctx.restore();
}

/* ---------- loop ---------- */
function loop(now) {
  const dt = Math.min(0.05, (now - last)/1000);
  last = now;
  update(dt);
  render();
  requestAnimationFrame(loop);
}
requestAnimationFrame(loop);

})();
