var G;
function reset(){
  G={ state:'playing', wave:0, lives:15, energy:8, score:0,
      towers:[], enemies:[], projs:[], flashes:[], parts:[], texts:[],
      hand:TKEYS.map(randomType), sel:-1,
      phase:'break', breakT:4, spawnQ:[], spawnT:0, energyT:0 };
  renderHand(); updateTop();
}
function randomType(){ return TKEYS[Math.floor(Math.random()*TKEYS.length)]; }

// ---------- waves ----------
function startWave(){
  G.wave++; G.phase='spawning'; G.spawnQ=[]; G.spawnT=0.5;
  var n=G.wave, count=6+n*2;
  for(var i=0;i<count;i++){
    var t='grunt';
    if(n>=2 && i%4===3) t='fast';
    if(n>=3 && i%5===4) t='tank';
    G.spawnQ.push(t);
  }
  if(n%5===0){ G.spawnQ.pop(); G.spawnQ.push('boss'); }
  SFX.wave(); addText(COLS/2,ROWS/2,'WAVE '+n,'#ffd76a',1.4);
}
function hpScale(){ return 1+(G.wave-1)*0.22; }

// ---------- enemies ----------
function spawnEnemy(type){
  var d=ENEMY_TYPES[type], p=pathPts()[0];
  G.enemies.push({type:type,x:p.x,y:p.y,wp:1,hp:d.hp*hpScale(),maxHp:d.hp*hpScale(),
    speed:d.speed*(1+G.wave*0.01),radius:d.radius,color:d.color,reward:d.reward,dmg:d.dmg,
    slowT:0,progress:0,dead:false});
}
function moveEnemy(e,dt){
  var sp=e.speed*(e.slowT>0?0.5:1)*dt;
  var pts=pathPts();
  while(sp>0 && e.wp<pts.length){
    var t=pts[e.wp], dx=t.x-e.x, dy=t.y-e.y, d=Math.hypot(dx,dy);
    if(d<=sp){ e.x=t.x; e.y=t.y; e.wp++; sp-=d; }
    else { e.x+=dx/d*sp; e.y+=dy/d*sp; sp=0; }
  }
  e.progress+=sp;
  if(e.slowT>0) e.slowT-=dt;
  if(e.wp>=pts.length){ // leaked
    e.dead=true; G.lives-=e.dmg; SFX.leak();
    addText(pts[pts.length-1].x-1,pts[pts.length-1].y,'-'+e.dmg+' ❤','#ff6a6a',1);
    if(G.lives<=0){ G.lives=0; gameOver(); }
  }
}
function hurt(e,dmg){
  if(e.dead) return;
  e.hp-=dmg;
  if(e.hp<=0){
    e.dead=true;
    var gain=e.reward;
    G.energy=Math.min(40,G.energy+gain);
    G.score+=e.reward*10;
    SFX.kill();
    addText(e.x,e.y-0.3,'+'+gain+'⚡','#ffd76a',0.8);
    for(var i=0;i<10;i++){
      var a=Math.random()*Math.PI*2,s=1+Math.random()*3;
      G.parts.push({x:e.x,y:e.y,vx:Math.cos(a)*s,vy:Math.sin(a)*s,life:0.4,maxLife:0.4,color:e.color,size:0.08});
    }
  }
}

// ---------- towers ----------
function towerAt(gx,gy){ for(var i=0;i<G.towers.length;i++) if(G.towers[i].gx===gx&&G.towers[i].gy===gy) return G.towers[i]; return null; }
function placeTower(gx,gy,type){
  var d=TOWERS[type];
  if(G.energy<d.cost) { SFX.error(); addText(gx+0.5,gy+0.5,'Need ⚡','#ff6a6a',0.8); return; }
  if(pathCells[gx+','+gy]) { SFX.error(); return; }
  if(towerAt(gx,gy)) { SFX.error(); return; }
  G.energy-=d.cost;
  G.towers.push({type:type,gx:gx,gy:gy,x:gx+0.5,y:gy+0.5,cd:0,angle:-Math.PI/2});
  SFX.place();
}
function sellTower(t){
  var refund=Math.floor(TOWERS[t.type].cost*0.6);
  G.energy=Math.min(40,G.energy+refund);
  G.towers.splice(G.towers.indexOf(t),1);
  SFX.sell();
  addText(t.x,t.y,'Sold +'+refund+'⚡','#8fd2ff',0.9);
}
function fireTower(t,dt){
  t.cd-=dt; if(t.cd>0) return;
  var d=TOWERS[t.type];
  // target: most-progressed enemy in range
  var best=null,bp=-1;
  for(var i=0;i<G.enemies.length;i++){
    var e=G.enemies[i]; if(e.dead) continue;
    var dist=Math.hypot(e.x-t.x,e.y-t.y);
    if(dist<=d.range && e.progress>bp){ bp=e.progress; best=e; }
  }
  if(!best) return;
  t.cd=d.rate; t.angle=Math.atan2(best.y-t.y,best.x-t.x);
  if(t.type==='tesla'){
    var hitList=[],cur=best,used={};
    for(var c=0;c<d.chain;c++){
      if(!cur||used[cur]) break;
      used[cur]=true; hitList.push(cur);
      hurt(cur,d.dmg);
      var nx=null,nd=1e9;
      for(var i=0;i<G.enemies.length;i++){
        var e2=G.enemies[i]; if(e2.dead||used[e2])continue;
        var dd=Math.hypot(e2.x-cur.x,e2.y-cur.y);
        if(dd<2.2&&dd<nd){nd=dd;nx=e2;}
      }
      cur=nx;
    }
    var pts=[{x:t.x,y:t.y}].concat(hitList.map(function(h){return {x:h.x,y:h.y};}));
    G.flashes.push({pts:pts,t:0.15});
    SFX.tesla();
  } else if(t.type==='sniper'){
    G.flashes.push({pts:[{x:t.x,y:t.y},{x:best.x,y:best.y}],t:0.08,beam:true});
    hurt(best,d.dmg); SFX.sniper();
  } else {
    G.projs.push({x:t.x,y:t.y,target:best,lastX:best.x,lastY:best.y,speed:12,dmg:d.dmg,type:t.type});
    if(t.type==='cannon')SFX.cannon(); else if(t.type==='frost')SFX.frost(); else SFX.shoot();
  }
}

// ---------- projectiles / fx ----------
function updateProjs(dt){
  for(var i=G.projs.length-1;i>=0;i--){
    var p=G.projs[i];
    if(p.target&&!p.target.dead){ p.lastX=p.target.x; p.lastY=p.target.y; }
    var dx=p.lastX-p.x, dy=p.lastY-p.y, d=Math.hypot(dx,dy), step=p.speed*dt;
    if(d<=step||d<0.12){
      // impact
      if(p.type==='cannon'){
        var r=TOWERS.cannon.splash;
        for(var j=0;j<G.enemies.length;j++){
          var e=G.enemies[j]; if(e.dead)continue;
          if(Math.hypot(e.x-p.lastX,e.y-p.lastY)<=r+e.radius) hurt(e,p.dmg);
        }
        for(var k=0;k<14;k++){ var a=Math.random()*Math.PI*2,s=1+Math.random()*3;
          G.parts.push({x:p.lastX,y:p.lastY,vx:Math.cos(a)*s,vy:Math.sin(a)*s,life:0.35,maxLife:0.35,color:'#ff9c4a',size:0.1}); }
        tone(80,0.2,'triangle',0.12,30);
      } else {
        if(p.target&&!p.target.dead){
          hurt(p.target,p.dmg);
          if(p.type==='frost'&&!p.target.dead) p.target.slowT=TOWERS.frost.slow;
        }
      }
      G.projs.splice(i,1);
    } else { p.x+=dx/d*step; p.y+=dy/d*step; }
  }
}
function addText(x,y,txt,color,scale){ G.texts.push({x:x,y:y,txt:txt,color:color,t:1.2,scale:scale||1}); }

// ---------- main update ----------
var last=0;
function loop(ts){
  requestAnimationFrame(loop);
  var dt=Math.min(0.05,(ts-last)/1000||0.016); last=ts;
  if(G&&G.state==='playing') update(dt);
  draw();
}
function update(dt){
  // energy regen
  G.energyT+=dt; while(G.energyT>=1.1){ G.energyT-=1.1; G.energy=Math.min(40,G.energy+1); }

  if(G.phase==='break'){
    G.breakT-=dt;
    if(G.breakT<=0) startWave();
  } else if(G.phase==='spawning'){
    G.spawnT-=dt;
    if(G.spawnT<=0 && G.spawnQ.length){
      spawnEnemy(G.spawnQ.shift());
      G.spawnT=Math.max(0.35,0.9-G.wave*0.02);
    }
    if(!G.spawnQ.length) G.phase='fighting';
  } else if(G.phase==='fighting'){
    if(!G.enemies.length){
      G.phase='break'; G.breakT=6;
      var bonus=4+Math.floor(G.wave/2);
      G.energy=Math.min(40,G.energy+bonus);
      SFX.clear(); addText(COLS/2,ROWS/2,'Wave cleared! +'+bonus+'⚡','#8fff9c',1.2);
    }
  }

  for(var i=G.enemies.length-1;i>=0;i--){ var e=G.enemies[i]; if(!e.dead) moveEnemy(e,dt); else G.enemies.splice(i,1); }
  for(var i=0;i<G.towers.length;i++) fireTower(G.towers[i],dt);
  updateProjs(dt);

  for(var i=G.flashes.length-1;i>=0;i--){ G.flashes[i].t-=dt; if(G.flashes[i].t<=0)G.flashes.splice(i,1); }
  for(var i=G.parts.length-1;i>=0;i--){ var p=G.parts[i]; p.x+=p.vx*dt;p.y+=p.vy*dt;p.life-=dt; if(p.life<=0)G.parts.splice(i,1); }
  for(var i=G.texts.length-1;i>=0;i--){ var t=G.texts[i]; t.y-=dt*0.6; t.t-=dt; if(t.t<=0)G.texts.splice(i,1); }
  updateTop();
}

// ---------- drawing ----------
function P(x,y){ return {x:ox+x*cell, y:oy+y*cell}; }
function draw(){
  var w=wrap.clientWidth,h=wrap.clientHeight;
  ctx.clearRect(0,0,w,h);
  if(!G) return;
  // grass checker
  for(var gy=0;gy<ROWS;gy++)for(var gx=0;gx<COLS;gx++){
    ctx.fillStyle=((gx+gy)%2===0)?'#2e5c38':'#2a5534';
    ctx.fillRect(ox+gx*cell,oy+gy*cell,cell,cell);
  }
  // path
  for(var key in pathCells){
    var p=key.split(','),gx=+p[0],gy=+p[1];
    ctx.fillStyle='#8a7346'; ctx.fillRect(ox+gx*cell,oy+gy*cell,cell,cell);
    ctx.fillStyle='rgba(0,0,0,.12)';
    ctx.fillRect(ox+gx*cell+2,oy+gy*cell+2,cell-4,3);
  }
  // start/end markers
  var s=P(-0.5,1), eP=P(COLS+0.2,6);
  ctx.font=(cell*0.7)+'px serif'; ctx.textAlign='center'; ctx.textBaseline='middle';
  ctx.fillText('🚪',ox+ (COLS-0.4)*cell, oy+(6.5)*cell);

  // hover preview
  if(hover.gx>=0&&hover.gx<COLS&&hover.gy>=0&&hover.gy<ROWS){
    var gx=hover.gx,gy=hover.gy;
    var valid = !pathCells[gx+','+gy] && !towerAt(gx,gy);
    if(G.sel>=0){
      var d=TOWERS[G.hand[G.sel]];
      ctx.fillStyle=valid?'rgba(120,255,140,.28)':'rgba(255,90,90,.3)';
      ctx.fillRect(ox+gx*cell,oy+gy*cell,cell,cell);
      var c=P(gx+0.5,gy+0.5);
      ctx.beginPath(); ctx.arc(c.x,c.y,d.range*cell,0,Math.PI*2);
      ctx.fillStyle='rgba(255,255,255,.08)'; ctx.fill();
      ctx.strokeStyle='rgba(255,255,255,.35)'; ctx.setLineDash([6,5]); ctx.stroke(); ctx.setLineDash([]);
    } else if(towerAt(gx,gy)){
      var t=towerAt(gx,gy), td=TOWERS[t.type], c2=P(t.x,t.y);
      ctx.beginPath(); ctx.arc(c2.x,c2.y,td.range*cell,0,Math.PI*2);
      ctx.fillStyle='rgba(255,215,106,.08)'; ctx.fill();
      ctx.strokeStyle='rgba(255,215,106,.4)'; ctx.setLineDash([6,5]); ctx.stroke(); ctx.setLineDash([]);
      ctx.fillStyle='rgba(255,120,120,.35)';
      ctx.fillRect(ox+gx*cell,oy+gy*cell,cell,cell);
    }
  }

  // towers
  for(var i=0;i<G.towers.length;i++){
    var t=G.towers[i], d=TOWERS[t.type], c=P(t.x,t.y), r=cell*0.36;
    ctx.fillStyle='rgba(0,0,0,.3)'; ctx.beginPath(); ctx.arc(c.x,c.y+2,r,0,Math.PI*2); ctx.fill();
    ctx.fillStyle=d.color; roundRect(c.x-r,c.y-r,r*2,r*2,cell*0.14); ctx.fill();
    ctx.strokeStyle='rgba(0,0,0,.35)'; ctx.lineWidth=2; roundRect(c.x-r,c.y-r,r*2,r*2,cell*0.14); ctx.stroke();
    // glyph
    ctx.save(); ctx.translate(c.x,c.y); ctx.rotate(t.angle);
    ctx.fillStyle='#fff'; ctx.strokeStyle='#fff'; ctx.lineWidth=Math.max(2,cell*0.06);
    if(t.type==='archer'){ ctx.beginPath(); ctx.moveTo(r*0.8,0); ctx.lineTo(-r*0.4,-r*0.5); ctx.lineTo(-r*0.4,r*0.5); ctx.closePath(); ctx.fill(); }
    else if(t.type==='cannon'){ ctx.beginPath(); ctx.arc(0,0,r*0.42,0,Math.PI*2); ctx.fill(); ctx.fillStyle='#331'; ctx.fillRect(r*0.1,-r*0.18,r*0.75,r*0.36); }
    else if(t.type==='frost'){ for(var k=0;k<3;k++){ ctx.save(); ctx.rotate(k*Math.PI/3); ctx.beginPath(); ctx.moveTo(-r*0.6,0); ctx.lineTo(r*0.6,0); ctx.stroke(); ctx.restore(); } }
    else if(t.type==='sniper'){ ctx.beginPath(); ctx.moveTo(0,0); ctx.lineTo(r*0.85,0); ctx.stroke(); ctx.beginPath(); ctx.arc(r*0.15,0,r*0.3,0,Math.PI*2); ctx.stroke(); }
    else { ctx.beginPath(); ctx.moveTo(-r*0.4,-r*0.6); ctx.lineTo(r*0.15,-r*0.05); ctx.lineTo(-r*0.1,r*0.05); ctx.lineTo(r*0.4,r*0.6); ctx.stroke(); }
    ctx.restore();
  }

  // enemies
  for(var i=0;i<G.enemies.length;i++){
    var e=G.enemies[i], c=P(e.x,e.y), r=e.radius*cell;
    ctx.fillStyle='rgba(0,0,0,.3)'; ctx.beginPath(); ctx.ellipse(c.x,c.y+r*0.7,r,r*0.4,0,0,Math.PI*2); ctx.fill();
    ctx.fillStyle=e.color; ctx.beginPath(); ctx.arc(c.x,c.y,r,0,Math.PI*2); ctx.fill();
    ctx.strokeStyle='rgba(0,0,0,.4)'; ctx.lineWidth=2; ctx.stroke();
    if(e.slowT>0){ ctx.strokeStyle='#8fd8ff'; ctx.lineWidth=3; ctx.beginPath(); ctx.arc(c.x,c.y,r+3,0,Math.PI*2); ctx.stroke(); }
    // eyes
    ctx.fillStyle='#fff';
    var ex=r*0.35;
    ctx.beginPath(); ctx.arc(c.x-ex*0.4,c.y-r*0.15,r*0.18,0,Math.PI*2); ctx.arc(c.x+ex*0.6,c.y-r*0.15,r*0.18,0,Math.PI*2); ctx.fill();
    // hp bar
    var bw=r*2,bh=Math.max(4,cell*0.09), bx=c.x-r, by=c.y-r-cell*0.16;
    ctx.fillStyle='rgba(0,0,0,.55)'; ctx.fillRect(bx-1,by-1,bw+2,bh+2);
    var f=Math.max(0,e.hp/e.maxHp);
    ctx.fillStyle=f>0.5?'#7ee06a':(f>0.25?'#ffd76a':'#ff6a5a');
    ctx.fillRect(bx,by,bw*f,bh);
  }

  // projectiles
  for(var i=0;i<G.projs.length;i++){
    var p=G.projs[i], c=P(p.x,p.y);
    if(p.type==='cannon'){ ctx.fillStyle='#3a2a1a'; ctx.beginPath(); ctx.arc(c.x,c.y,cell*0.14,0,Math.PI*2); ctx.fill(); }
    else if(p.type==='frost'){ ctx.fillStyle='#bfeaff'; ctx.beginPath(); ctx.arc(c.x,c.y,cell*0.1,0,Math.PI*2); ctx.fill(); }
    else { ctx.strokeStyle='#d8ffb0'; ctx.lineWidth=3; var a=Math.atan2(p.lastY-p.y,p.lastX-p.x);
      ctx.beginPath(); ctx.moveTo(c.x-Math.cos(a)*cell*0.16,c.y-Math.sin(a)*cell*0.16);
      ctx.lineTo(c.x+Math.cos(a)*cell*0.16,c.y+Math.sin(a)*cell*0.16); ctx.stroke(); }
  }

  // flashes (lightning / beams)
  for(var i=0;i<G.flashes.length;i++){
    var f=G.flashes[i];
    ctx.strokeStyle=f.beam?'rgba(230,180,255,'+(f.t/0.08)+')':'rgba(240,220,90,'+(f.t/0.15)+')';
    ctx.lineWidth=f.beam?2:Math.max(2,cell*0.07);
    ctx.beginPath();
    for(var j=0;j<f.pts.length;j++){ var c=P(f.pts[j].x,f.pts[j].y); if(j===0)ctx.moveTo(c.x,c.y); else ctx.lineTo(c.x,c.y); }
    ctx.stroke();
  }

  // particles
  for(var i=0;i<G.parts.length;i++){
    var p=G.parts[i], c=P(p.x,p.y), a=Math.max(0,p.life/p.maxLife);
    ctx.globalAlpha=a; ctx.fillStyle=p.color;
    ctx.beginPath(); ctx.arc(c.x,c.y,p.size*cell,0,Math.PI*2); ctx.fill();
    ctx.globalAlpha=1;
  }

  // floating text
  ctx.textAlign='center'; ctx.textBaseline='middle';
  for(var i=0;i<G.texts.length;i++){
    var t=G.texts[i], c=P(t.x,t.y);
    ctx.globalAlpha=Math.min(1,t.t);
    ctx.font='bold '+(cell*0.35*(t.scale||1))+'px Segoe UI, Arial';
    ctx.strokeStyle='rgba(0,0,0,.7)'; ctx.lineWidth=3; ctx.strokeText(t.txt,c.x,c.y);
    ctx.fillStyle=t.color; ctx.fillText(t.txt,c.x,c.y);
    ctx.globalAlpha=1;
  }

  // break banner
  if(G.state==='playing'&&G.phase==='break'){
    var m=Math.ceil(G.breakT);
    ctx.font='bold '+(cell*0.5)+'px Segoe UI, Arial';
    ctx.strokeStyle='rgba(0,0,0,.7)'; ctx.lineWidth=4;
    ctx.strokeText('Next wave in '+m+' — press SPACE',ox+COLS*cell/2,oy-cell*0.4);
    ctx.fillStyle='#ffd76a';
    ctx.fillText('Next wave in '+m+' — press SPACE',ox+COLS*cell/2,oy-cell*0.4);
  }
}
function roundRect(x,y,w,h,r){ ctx.beginPath(); ctx.moveTo(x+r,y); ctx.arcTo(x+w,y,x+w,y+h,r); ctx.arcTo(x+w,y+h,x,y+h,r); ctx.arcTo(x,y+h,x,y,r); ctx.arcTo(x,y,x+w,y,r); ctx.closePath(); }

// ---------- top bar / hand ----------
function updateTop(){
  sLives.textContent=G.lives; sWave.textContent=G.wave;
  sEnergy.textContent=Math.floor(G.energy); sScore.textContent=G.score;
  waveBtn.disabled = !(G.state==='playing'&&G.phase==='break');
  for(var i=0;i<cardEls.length;i++){
    cardEls[i].classList.toggle('noafford', G.energy<TOWERS[G.hand[i]].cost);
    cardEls[i].classList.toggle('sel', i===G.sel);
  }
}
var cardEls=[];
function renderHand(){
  handEl.innerHTML=''; cardEls=[];
  for(var i=0;i<G.hand.length;i++){
    (function(idx){
      var d=TOWERS[G.hand[idx]];
      var el=document.createElement('div'); el.className='card';
      el.innerHTML='<div class="cost">'+d.cost+'⚡</div><div class="cicon" style="color:'+d.color+'">'+d.icon+'</div><div class="cname" style="color:'+d.color+'">'+d.name+'</div><div class="cdesc">'+d.desc+'</div>';
      el.addEventListener('click',function(){ initAudio(); G.sel=(G.sel===idx?-1:idx); updateTop(); });
      handEl.appendChild(el); cardEls.push(el);
    })(i);
  }
}

// ---------- input ----------
var hover={gx:-1,gy:-1};
cv.addEventListener('mousemove',function(ev){
  var r=cv.getBoundingClientRect(), mx=ev.clientX-r.left, my=ev.clientY-r.top;
  hover.gx=Math.floor((mx-ox)/cell); hover.gy=Math.floor((my-oy)/cell);
});
cv.addEventListener('mouseleave',function(){ hover.gx=-1; hover.gy=-1; });
cv.addEventListener('click',function(ev){
  initAudio();
  if(!G||G.state!=='playing') return;
  var r=cv.getBoundingClientRect(), gx=Math.floor((ev.clientX-r.left-ox)/cell), gy=Math.floor((ev.clientY-r.top-oy)/cell);
  if(gx<0||gx>=COLS||gy<0||gy>=ROWS) return;
  if(G.sel>=0){
    placeTower(gx,gy,G.hand[G.sel]);
    G.hand[G.sel]=randomType(); renderHand();
    G.sel=-1; updateTop();
  } else {
    var t=towerAt(gx,gy);
    if(t) sellTower(t), updateTop();
  }
});
window.addEventListener('keydown',function(ev){
  initAudio();
  if(!G) return;
  var k=ev.key;
  if(k>='1'&&k<='5'){ var i=+k-1; G.sel=(G.sel===i?-1:i); updateTop(); }
  else if(k==='Escape'||k==='0'){ G.sel=-1; updateTop(); }
  else if(k===' '){ ev.preventDefault(); sendWaveEarly(); }
  else if((k==='r'||k==='R')&&G.state!=='playing'){ startGame(); }
});
function sendWaveEarly(){
  if(G&&G.state==='playing'&&G.phase==='break'){
    G.energy=Math.min(40,G.energy+3); addText(COLS/2,ROWS/2-0.5,'+3⚡ early!','#8fff9c',0.9); startWave();
  }
}
waveBtn.addEventListener('click',function(){ initAudio(); sendWaveEarly(); });

// ---------- screens ----------
function showOverlay(html){ overlay.innerHTML=html; overlay.classList.add('show'); }
function gameOver(){
  G.state='over'; SFX.over(); saveBest();
  showOverlay('<h1>GAME OVER</h1><p class="big">You survived '+G.wave+(G.wave===1?' wave':' waves')+' — Score: '+G.score+'</p><p>Best score: '+(bestScore()||0)+'</p><button id="againBtn">Play Again (R)</button>');
