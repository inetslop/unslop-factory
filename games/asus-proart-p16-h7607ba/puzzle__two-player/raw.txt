    an.cells.forEach(k=>{const p=k.split(',');board[p[0]][p[1]]=-1;});
    const falls=applyGravity();
    anim={type:'fall',falls:falls,t:0,dur:0.28};
  }else if(an.type==='fall'){
    const m=findMatches();
    if(m.size>0){
      cascade++;
      anim={type:'match',cells:m,t:0,dur:0.32};
      sMatch(cascade);
    }else{
      endTurn();
    }
  }
}

function confirm(){
  if(state!=='playing'||anim)return;
  if(!selected){selected={r:cursor.r,c:cursor.c};tone(600,0.05,'sine',0.05);}
  else if(selected.r===cursor.r&&selected.c===cursor.c){selected=null;tone(400,0.05,'sine',0.05);}
  else if(Math.abs(selected.r-cursor.r)+Math.abs(selected.c-cursor.c)===1){
    trySwap({r:selected.r,c:selected.c},{r:cursor.r,c:cursor.c});selected=null;
  }else{selected={r:cursor.r,c:cursor.c};tone(600,0.05,'sine',0.05);}
}

/* ---------- update ---------- */
function update(dt){
  if(msg){msg.t-=dt;if(msg.t<=0)msg=null;}
  if(anim){
    anim.t+=dt;
    if(anim.t>=anim.dur){
      if(anim.type==='swap'&&anim.invalid){
        // reverse swap finished: revert board to original and pass turn
        const a=anim.a,b=anim.b;
        const tmp=board[a.r][a.c];board[a.r][a.c]=board[b.r][b.c];board[b.r][b.c]=tmp;
        anim={type:'swapInvalidDone',t:0,dur:0.01};
      }else completeAnim();
    }
  }
}

/* ---------- drawing helpers ---------- */
function ease(t){t=Math.max(0,Math.min(1,t));return t*t*(3-2*t);}
function rr(x,y,w,h,r){
  ctx.beginPath();
  ctx.moveTo(x+r,y);ctx.lineTo(x+w-r,y);ctx.quadraticCurveTo(x+w,y,x+w,y+r);
  ctx.lineTo(x+w,y+h-r);ctx.quadraticCurveTo(x+w,y+h,x+w-r,y+h);
  ctx.lineTo(x+r,y+h);ctx.quadraticCurveTo(x,y+h,x,y+h-r);
  ctx.lineTo(x,y+r);ctx.quadraticCurveTo(x,y,x+r,y);ctx.closePath();
}
function gemPath(x,y,s,k){
  const r=s*0.36;
  ctx.beginPath();
  if(k===0){ctx.arc(x,y,r,0,Math.PI*2);}
  else if(k===1){ctx.moveTo(x,y-r*1.25);ctx.lineTo(x+r,y);ctx.lineTo(x,y+r*1.25);ctx.lineTo(x-r,y);ctx.closePath();}
  else if(k===2){const q=r*0.88;ctx.rect(x-q,y-q,2*q,2*q);}
  else if(k===3){ctx.moveTo(x,y-r*1.15);ctx.lineTo(x+r*1.1,y+r*0.85);ctx.lineTo(x-r*1.1,y+r*0.85);ctx.closePath();}
  else{for(let i=0;i<6;i++){const a=Math.PI/3*i-Math.PI/2;const px=x+Math.cos(a)*r,py=y+Math.sin(a)*r;if(i)ctx.lineTo(px,py);else ctx.moveTo(px,py);}ctx.closePath();}
}
function drawGem(x,y,s,k,alpha){
  if(k<0)return;
  ctx.save();
  ctx.globalAlpha=alpha==null?1:alpha;
  gemPath(x,y,s,k);
  const g=ctx.createLinearGradient(x-s/2,y-s/2,x+s/2,y+s/2);
  g.addColorStop(0,'#ffffff');g.addColorStop(0.25,GEMCOL[k]);g.addColorStop(1,GEMDARK[k]);
  ctx.fillStyle=g;ctx.fill();
  ctx.lineWidth=Math.max(1.5,s*0.05);
  ctx.strokeStyle='rgba(0,0,0,0.45)';ctx.stroke();
  ctx.beginPath();ctx.arc(x-s*0.12,y-s*0.16,s*0.10,0,Math.PI*2);
  ctx.fillStyle='rgba(255,255,255,0.55)';ctx.fill();
  ctx.restore();
}
