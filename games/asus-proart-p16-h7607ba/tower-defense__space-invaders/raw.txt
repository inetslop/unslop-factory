function beep(f0,f1,dur,type,vol){
  var a=audio();if(!a)return;
  try{
    var o=a.createOscillator(),g=a.createGain(),t=a.currentTime;
    o.type=type||'square';
    o.frequency.setValueAtTime(f0,t);
    o.frequency.exponentialRampToValueAtTime(Math.max(20,f1),t+dur);

    o.connect(g);g.connect(a.destination);
    o.start(t);o.stop(t+dur+0.02);
  }catch(e){}
}

    else{tx2=sh.x+sh.lx;ty2=sh.y+sh.ly;}

    if(dd<=mv||dd<6){
      // impact
      if(sh.tgt&&sh.tgt.hp>0){
        sh.tgt.hp-=sh.dmg;sfx.hit();

  document.getElementById('btn').textContent='RETRY';

/* ---------- drawing ---------- */
var PIXA=["..1..1..","...11...",".111111.","1.111.1.","11111111","1.1..1.1"];
var PIXB=["..1..1..","...11...",".111111.","1.111.1.","11111111",".11..11."];
function drawInvader(e){
  var pat=Math.floor(e.t*4)%2?PIXB:PIXA;
  var px=(e.r*2)/8, x0=e.x-e.r, y0=e.y-e.r*0.75;
  ctx.fillStyle=e.col;
  for(var r=0;r<6;r++)for(var c=0;c<8;c++)
    if(pat[r][c]==='1')ctx.fillRect(x0+c*px,y0+r*px,px+0.5,px+0.5);
  // hp bar
  if(e.hp<e.maxHp){
    ctx.fillStyle='#304';ctx.fillRect(e.x-e.r,e.y-e.r-8,e.r*2,3);
    ctx.fillStyle=e.col;ctx.fillRect(e.x-e.r,e.y-e.r-8,e.r*2*Math.max(0,e.hp/e.maxHp),3);
  }
}
function drawTower(tw){
  var x=(tw.c+0.5)*CELL,y=(tw.r+0.5)*CELL,T=TYPES[tw.type];
  ctx.save();ctx.translate(x,y);
  // base
  ctx.fillStyle='#12304a';ctx.strokeStyle=T.color;ctx.lineWidth=2;
  ctx.beginPath();
  for(var i=0;i<6;i++){var a=i/6*6.283+0.52;var px=Math.cos(a)*18,py=Math.sin(a)*18;i?ctx.lineTo(px,py):ctx.moveTo(px,py);}
  ctx.closePath();ctx.fill();ctx.stroke();
  // turret
  ctx.rotate(tw.has?tw.angle:-Math.PI/2);
  ctx.fillStyle=T.color;
  ctx.fillRect(0,-4,22,8);
  ctx.beginPath();ctx.arc(0,0,9,0,6.283);ctx.fill();
  ctx.restore();
}
function draw(){
  ctx.clearRect(0,0,W,H);
  // bg
  var g=ctx.createLinearGradient(0,0,0,H);
  g.addColorStop(0,'#050914');g.addColorStop(1,'#0a1230');
  ctx.fillStyle=g;ctx.fillRect(0,0,W,H);
  // stars
  for(var i=0;i<stars.length;i++){
    var s=stars[i];
    ctx.globalAlpha=0.4+0.6*Math.abs(Math.sin(time*0.8+s.p));
    ctx.fillStyle='#cfe';ctx.fillRect(s.x,s.y,s.s,s.s);
  }
  ctx.globalAlpha=1;

  if(state==='menu')return;

  // grid
  ctx.strokeStyle='rgba(70,140,255,0.10)';ctx.lineWidth=1;
  ctx.beginPath();
  for(var c=0;c<=COLS;c++){ctx.moveTo(c*CELL,0);ctx.lineTo(c*CELL,H);}
  for(var r=0;r<=ROWS;r++){ctx.moveTo(0,r*CELL);ctx.lineTo(W,r*CELL);}
  ctx.stroke();

  // defense line
  var pulse=0.5+0.5*Math.sin(time*3);
  ctx.strokeStyle='rgba(80,255,160,'+(0.5+pulse*0.4)+')';ctx.lineWidth=3;
  ctx.beginPath();ctx.moveTo(0,H-6);ctx.lineTo(W,H-6);ctx.stroke();

  // hover cell preview
  if(state==='playing'&&!paused&&hover){
    var T=TYPES[sel];
    var hx=(hover.c+0.5)*CELL,hy=(hover.r+0.5)*CELL;
    var ok=!towers[hover.c+','+hover.r]&&credits>=T.cost;
    ctx.fillStyle=ok?'rgba(90,255,160,0.12)':'rgba(255,80,80,0.14)';
    ctx.fillRect(hover.c*CELL,hover.r*CELL,CELL,CELL);
    if(ok){
      ctx.strokeStyle='rgba(90,255,160,0.35)';ctx.setLineDash([6,6]);
      ctx.beginPath();ctx.arc(hx,hy,T.range,0,6.283);ctx.stroke();ctx.setLineDash([]);
    }
  }

  for(var key in towers)drawTower(towers[key]);
  for(var e=0;e<enemies.length;e++)drawInvader(enemies[e]);

  // shots
  for(var s2=0;s2<shots.length;s2++){
    var sh=shots[s2];
    ctx.fillStyle=sh.col;ctx.shadowColor=sh.col;ctx.shadowBlur=8;
    ctx.beginPath();ctx.arc(sh.x,sh.y,sh.sz,0,6.283);ctx.fill();
  }
  ctx.shadowBlur=0;

  // particles
  for(var p=0;p<particles.length;p++){
    var pa=particles[p];
    ctx.globalAlpha=Math.max(0,pa.life/pa.max);
    ctx.fillStyle=pa.col;ctx.fillRect(pa.x-pa.s/2,pa.y-pa.s/2,pa.s,pa.s);
  }
  ctx.globalAlpha=1;

  // wave banner
  if(state==='playing'){
    if(waveState==='build'&&wave>0){
      ctx.fillStyle='#7ffbff';ctx.font='bold 30px Courier New';ctx.textAlign='center';
      ctx.fillText('WAVE '+(wave+1)+' INCOMING — '+Math.ceil(buildTimer),W/2,H*0.22);
      ctx.font='15px Courier New';ctx.fillStyle='#8ab';
      ctx.fillText('BUILD YOUR DEFENSES',W/2,H*0.22+26);
    }else if(waveState==='build'){
      ctx.fillStyle='#7ffbff';ctx.font='bold 34px Courier New';ctx.textAlign='center';
      ctx.fillText('DEFEND THE PLANET',W/2,H*0.25);
      ctx.font='16px Courier New';ctx.fillStyle='#8ab';
      ctx.fillText('First wave in '+Math.ceil(buildTimer),W/2,H*0.25+30);
    }
    if(paused){
      ctx.fillStyle='rgba(4,6,15,0.6)';ctx.fillRect(0,0,W,H);
      ctx.fillStyle='#7ffbff';ctx.font='bold 40px Courier New';ctx.textAlign='center';
      ctx.fillText('PAUSED',W/2,H/2);
      ctx.font='15px Courier New';ctx.fillStyle='#8ab';
      ctx.fillText('[P] resume   [R] restart',W/2,H/2+34);
    }
  }
}

var cardEls=[];
function buildCards(){
  var bar=document.getElementById('cards');bar.innerHTML='';cardEls=[];
  TYPES.forEach(function(T,i){

  });
}

  updateCards();
}
