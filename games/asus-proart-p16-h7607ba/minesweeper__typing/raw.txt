  for(var p=pool.length-1;p>0;p--){ var q=Math.floor(Math.random()*(p+1)); var t=pool[p];pool[p]=pool[q];pool[q]=t; }
  pool.slice(0,d.mines).forEach(function(idx){ cells[idx].mine=true; });
  for(var i2=0;i2<total;i2++) if(!cells[i2].mine) cells[i2].adj=neighbors(i2).filter(function(n){return cells[n].mine;}).length;
  placed=true;
}

function flood(idx){
  var stack=[idx];
  while(stack.length){
    var i=stack.pop(), c=cells[i];
    if(c.rev||c.flag) continue;
    c.rev=true;
    if(c.adj===0) neighbors(i).forEach(function(n){ if(!cells[n].rev) stack.push(n); });
  }
}

function openCell(idx){
  var c=cells[idx];
  if(state!=='play'||c.rev||c.flag) return;
  if(!placed){ placeMines(idx); startTime=performance.now(); started=true; }
  if(c.mine){ lose(idx); return; }
  flood(idx);
  sOpen(c.adj);
  checkWin();
}

function toggleFlag(idx){
  var c=cells[idx];
  if(state!=='play'||c.rev) return;
  c.flag=!c.flag; sFlag(); updateHUD();
}

function flagsCount(){ var n=0; cells.forEach(function(c){ if(c.flag)n++; }); return n; }
function revCount(){ var n=0; cells.forEach(function(c){ if(c.rev)n++; }); return n; }

function checkWin(){
  if(revCount()===total-DIFFS[diffKey].mines) win();
}

function lose(idx){
  state='over'; hitIndex=idx; elapsed=started?performance.now()-startTime:0;
  cells.forEach(function(c){ if(c.mine)c.rev=true; });
  sBoom(); shakeUntil=performance.now()+400;
  showEnd(false);
}

function win(){
  state='win'; elapsed=started?performance.now()-startTime:0;
  cells.forEach(function(c){ if(c.mine){c.flag=true;c.rev=false;} });
  sWin();
  var b=getBest(), prev=b[diffKey];
  var nb=(prev===undefined||elapsed<prev);
  if(nb) setBest(diffKey,elapsed);
  showEnd(true,nb);
}

  var b=getBest(), cur=b[diffKey];
  endBest.innerHTML = newBest ? '<span class="newbest">★ NEW BEST TIME! ★</span>' : (cur!==undefined ? 'Best: '+fmt(cur) : '');
  endOv.classList.remove('hidden');
}

function tryAction(code){
  var idx=-1;
  for(var i=0;i<total;i++) if(!cells[i].rev && cells[i].code===code){ idx=i; break; }
  if(idx<0){ sErr(); shakeUntil=performance.now()+250; return; }
  var c=cells[idx];
  if(c.flag){ if(flagMode) toggleFlag(idx); else tone(300,0.06,'sine',0.08); }
  else if(flagMode) toggleFlag(idx);
  else openCell(idx);
}

function onChar(ch){
  buffer=(buffer+ch).slice(-2);
  showInput();
  clearTimeout(idleT); idleT=setTimeout(function(){ buffer=''; showInput(); },1500);
  if(buffer.length===2){ tryAction(buffer); buffer=''; showInput(); }
}

window.addEventListener('keydown', function(e){
  ensureAudio();
  var k=e.key;
  if(state==='menu'){
    if(k==='Enter'||k===' '){ startGame(); e.preventDefault(); }
    else if(k==='1'){ diffKey='easy'; syncDiffBtns(); startGame(); }
    else if(k==='2'){ diffKey='medium'; syncDiffBtns(); startGame(); }
    else if(k==='3'){ diffKey='hard'; syncDiffBtns(); startGame(); }
    return;
  }
  if(state==='over'||state==='win'){
    if(k==='r'||k==='R') startGame();
    else if(k==='m'||k==='M') goMenu();
    return;
  }
  // playing
  if(k==='F'||k==='f'){ flagMode=!flagMode; setModeUI(); sFlag(); e.preventDefault(); return; }
  if(k==='Escape'){ if(flagMode){ flagMode=false; setModeUI(); } buffer=''; showInput(); e.preventDefault(); return; }
  if(k==='Backspace'){ buffer=''; showInput(); clearTimeout(idleT); e.preventDefault(); return; }
  if(k==='r'||k==='R'){ startGame(); return; }
  if(k==='m'||k==='M'){ goMenu(); return; }
  var up=k.toUpperCase();
  if(/^[A-Z0-9]$/.test(up)){ onChar(up); e.preventDefault(); }
});

function resize(){
  var dpr=window.devicePixelRatio||1;
  W=wrap.clientWidth; H=wrap.clientHeight;
  canvas.width=W*dpr; canvas.height=H*dpr;
  ctx.setTransform(dpr,0,0,dpr,0,0);
  var d=DIFFS[diffKey], pad=12;
  gap=Math.max(2,Math.floor(Math.min(W/d.cols,H/d.rows)*0.07));
  tSize=Math.floor(Math.min((W-pad*2)/d.cols - gap*(d.cols-1)/d.cols, (H-pad*2)/d.rows - gap*(d.rows-1)/d.rows));
  tSize=Math.max(14,tSize);
  var bw=d.cols*tSize+(d.cols-1)*gap, bh=d.rows*tSize+(d.rows-1)*gap;
  ox=Math.floor((W-bw)/2); oy=Math.floor((H-bh)/2);
}

function rr(x,y,w,h,r){ ctx.beginPath(); ctx.moveTo(x+r,y); ctx.arcTo(x+w,y,x+w,y+h,r); ctx.arcTo(x+w,y+h,x,y+h,r); ctx.arcTo(x,y+h,x,y,r); ctx.arcTo(x,y,x+w,y,r); ctx.closePath(); }

function drawMineIcon(x,y,s,col){
  ctx.fillStyle=col; ctx.strokeStyle=col;
  ctx.lineWidth=Math.max(1.5,s*0.09);
  for(var i=0;i<8;i++){ var a=i*Math.PI/4; ctx.beginPath(); ctx.moveTo(x+Math.cos(a)*s*0.2,y+Math.sin(a)*s*0.2); ctx.lineTo(x+Math.cos(a)*s*0.55,y+Math.sin(a)*s*0.55); ctx.stroke(); }
  ctx.beginPath(); ctx.arc(x,y,s*0.3,0,Math.PI*2); ctx.fill();
}

}

function draw(){
  var now=performance.now();
  ctx.clearRect(0,0,W,H);
  if(state==='menu') return;
  var d=DIFFS[diffKey];
  ctx.save();
  if(now<shakeUntil){ ctx.translate((Math.random()-0.5)*8,(Math.random()-0.5)*8); }
  for(var i=0;i<total;i++){
    var r=Math.floor(i/d.cols), c=i%d.cols;
    var x=ox+c*(tSize+gap), y=oy+r*(tSize+gap), cell=cells[i];
    var rad=Math.max(3,tSize*0.12);
    if(!cell.rev){
      ctx.fillStyle = cell.flag ? '#4a3550' : (i%2? '#33415f':'#384869');
      rr(x,y,tSize,tSize,rad); ctx.fill();
      ctx.strokeStyle='#4d6191'; ctx.lineWidth=1.5; ctx.stroke();
      ctx.fillStyle='#e8efff';
      ctx.font='bold '+Math.floor(tSize*0.34)+'px "Courier New",monospace';
      ctx.textAlign='center'; ctx.textBaseline='middle';
      ctx.fillText(cell.code, x+tSize/2, y+tSize/2+1);
      if(cell.flag) drawFlagIcon(x+tSize*0.78, y+tSize*0.3, tSize*0.5);
    } else {
      var bg = (i===hitIndex) ? '#c0392b' : (i%2? '#161d2f':'#192134');
      ctx.fillStyle=bg; rr(x,y,tSize,tSize,rad); ctx.fill();
      ctx.strokeStyle='#232e4a'; ctx.lineWidth=1; ctx.stroke();
      if(cell.mine){ drawMineIcon(x+tSize/2,y+tSize/2,tSize*0.8, i===hitIndex?'#ffe14d':'#11151f'); }
      else if(cell.adj>0){
        ctx.fillStyle=NUMCOL[cell.adj]||'#fff';
        ctx.font='bold '+Math.floor(tSize*0.42)+'px "Courier New",monospace';
        ctx.textAlign='center'; ctx.textBaseline='middle';
        ctx.fillText(cell.adj, x+tSize/2, y+tSize/2+1);
      }
    }
  }
  ctx.restore();
}
