function neighbors(x,y){ return [[x+1,y],[x-1,y],[x,y+1],[x,y-1]]; }
function genBoard(){
  for(var tries=0;tries<80;tries++){
    grid=[];
    for(var y=0;y<ROWS;y++){
      grid[y]=[];
      for(var x=0;x<COLS;x++){
        var mine=(y>0)&&Math.random()<MINE_PROB;
        grid[y][x]={mine:mine,count:0};
      }
    }
    // counts

      if(grid[y][x].mine)continue;
      var c=0;
      for(var dy=-1;dy<=1;dy++)for(var dx=-1;dx<=1;dx++){
        if(!dx&&!dy)continue;
        var nx=x+dx,ny=y+dy;
        if(nx>=0&&nx<COLS&&ny>=1&&ny<ROWS&&grid[ny][nx].mine)c++;
      }
      grid[y][x].count=c;
    }
    // find a safe start in bottom row with a path to some pad
    var startX=-1, found=false;
    for(x=0;x<COLS&&!found;x++){
      if(!grid[ROWS-1][x].mine){ startX=x; found=bfsPath(x,ROWS-1); }
    }
    if(found){ player.x=startX; return; }
  }
  // fallback: clear a column (shouldn't happen)
  for(y=0;y<ROWS;y++)for(x=0;x<COLS;x++)grid[y][x].mine=false;
  for(y=1;y<ROWS;y++)for(x=0;x<COLS;x++){
    var c=0;
    for(var dy=-1;dy<=1;dy++)for(var dx=-1;dx<=1;dx++){
      if(!dx&&!dy)continue;
      var nx=x+dx,ny=y+dy;
      if(nx>=0&&nx<COLS&&ny>=1&&ny<ROWS&&grid[ny][nx].mine)c++;
    }
    grid[y][x].count=c;
  }
  player.x=6;
}
function bfsPath(sx,sy){
  var seen=[],q=[[sx,sy]];
  for(var y=0;y<ROWS;y++)seen[y]=[];
  while(q.length){
    var c=q.shift(), x=c[0], y=c[1];
    if(seen[y][x])continue; seen[y][x]=true;
    if(y===0&&PAD_COLS.indexOf(x)>=0)return true;
    for(var i=0;i<4;i++){
      var nx=x+neighbors(x,y)[i][0], ny=y+neighbors(x,y)[i][1];
      if(nx<0||nx>=COLS||ny<0||ny>=ROWS)continue;
      if(ny===0){ if(PAD_COLS.indexOf(nx)>=0&&PAD_COLS.indexOf(nx)!==-1){/*ok*/} else continue; }
      else if(grid[ny][nx].mine)continue;
      q.push([nx,ny]);
    }
  }
  return false;
}

function tryMove(dx,dy){
  if(state!=='play')return;
  var nx=player.x+dx, ny=player.y+dy;
  if(nx<0||nx>=COLS||ny<0||ny>=ROWS)return;
  hopFrom={x:player.x,y:player.y}; hopT=0;
  player.x=nx; player.y=ny;
  if(ny===0){
    var pi=PAD_COLS.indexOf(nx);
    if(pi>=0&&!pads[pi]){ pads[pi]=true; sfxPad(); if(capturedCount()===5){ setTimeout(function(){endGame(true);},350); state='winning'; } }
  }else{
    if(grid[ny][nx].mine){ explode(); return; }
    sfxMove();
  }
}
