// ---------- game state ----------
const GRAV=2600, MAXL=6;

let blocks=[], platforms=[], particles=[];

function groundY(){return H-26;}
function jumpV(){return Math.sqrt(2*GRAV*(Math.min(H*0.55,430)+40));}

const pDefs=[[0.08,0.46,0.24],[0.68,0.46,0.24],[0.38,0.30,0.24],[0.06,0.63,0.20],[0.74,0.63,0.20]];

let grid={cols:12,bw:50,gap:5,bx0:0,bh:28,top:60};
function computeGrid(){
  const cols=12, gap=5;
  const bw=(Math.min(W*0.86,980)-gap*(cols-1))/cols;

}

function makeBlocks(){
  computeGrid(); blocks=[];

  const pat=level%3, cols=grid.cols;
  for(let r=0;r<rows;r++)for(let c=0;c<cols;c++){
    let on=true;
    if(pat===1&&(r+c)%2===0)on=false;
    if(pat===2){const d=Math.abs(c-(cols-1)/2); on=d<=(rows-1-r)*1.6+0.6;}
    if(!on)continue;
    let hp=1;
    if(level>=2&&r===0&&Math.abs(c-(cols-1)/2)<2.5)hp=2;
    if(level>=4&&(r===rows-1)&&c%3===0)hp=2;
    blocks.push({r,c,hp,maxHp:hp,alive:true,flash:0,hue:(r*46+level*30)%360});
  }
}

function resetPlayer(){P.x=W/2-P.w/2;P.y=groundY()-P.h;P.vx=0;P.vy=0;P.onGround=true;}
function resetBall(){ball.attached=true;ball.speed=Math.min(360+level*25,640);ball.vx=0;ball.vy=0;}
function startGame(){score=0;lives=3;level=1;particles=[];makeBlocks();resetPlayer();resetBall();mode='play';sfxLevel();}

// ---------- audio ----------

  if(AC&&AC.state==='suspended')AC.resume();

function tone(f,dur,type,vol,f2){
  if(!AC)return;
  try{
    const t=AC.currentTime,o=AC.createOscillator(),g=AC.createGain();
    o.type=type||'square';o.frequency.setValueAtTime(f,t);
    if(f2)o.frequency.exponentialRampToValueAtTime(Math.max(f2,1),t+dur);
    g.gain.setValueAtTime(vol||0.12,t);g.gain.exponentialRampToValueAtTime(0.001,t+dur);
    o.connect(g);g.connect(AC.destination);o.start(t);o.stop(t+dur+0.03);
  }catch(e){}
}

function spawnParticles(x,y,color){
  for(let i=0;i<12;i++)particles.push({x,y,vx:(Math.random()-0.5)*340,vy:(Math.random()-0.5)*340-80,life:0.5,max:0.5,color});
}

function hitRect(rect,isPlayer,block){
  const cx=Math.max(rect.x,Math.min(ball.x,rect.x+rect.w));
  const cy=Math.max(rect.y,Math.min(ball.y,rect.y+rect.h));
  let dx=ball.x-cx,dy=ball.y-cy;const d2=dx*dx+dy*dy;
  if(d2>ball.r*ball.r)return false;
  let nx,ny;
  if(d2>1e-6){const d=Math.sqrt(d2);nx=dx/d;ny=dy/d;}
  else{
    const l=ball.x-rect.x,r=rect.x+rect.w-ball.x,t=ball.y-rect.y,b=rect.y+rect.h-ball.y;
    const m=Math.min(l,r,t,b);
    if(m===l){nx=-1;ny=0;}else if(m===r){nx=1;ny=0;}else if(m===t){nx=0;ny=-1;}else{nx=0;ny=1;}
  }
  ball.x=cx+nx*(ball.r+0.5);ball.y=cy+ny*(ball.r+0.5);
  const dot=ball.vx*nx+ball.vy*ny;
  if(dot<0){ball.vx-=2*dot*nx;ball.vy-=2*dot*ny;}
  if(isPlayer&&ny<-0.5){
    const sp=Math.hypot(ball.vx,ball.vy);
    const off=(ball.x-(rect.x+rect.w/2))/(rect.w/2);
    ball.vx=off*sp*0.8;
    ball.vy=-Math.sqrt(Math.max(sp*sp-ball.vx*ball.vx,(sp*0.35)*(sp*0.35)));
  }
  if(isPlayer)sfxPlayer();else if(block){
    block.hp--;block.flash=0.15;
    if(block.hp<=0){block.alive=false;score+=block.maxHp>1?25:10;spawnParticles(ball.x,ball.y,'hsl('+block.hue+',80%,60%)');sfxBreak(block.r);}
    else sfxThud();
    ball.speed=Math.min(ball.speed+3,720);
  }else sfxBounce();
  return true;
}

function moveBall(dt){
  const sp=Math.hypot(ball.vx,ball.vy)||1;
  const steps=Math.max(1,Math.ceil(sp*dt/5)),sdt=dt/steps;
  for(let i=0;i<steps;i++){
    ball.x+=ball.vx*sdt;ball.y+=ball.vy*sdt;
    if(ball.x<ball.r){ball.x=ball.r;ball.vx=Math.abs(ball.vx);sfxBounce();}
    else if(ball.x>W-ball.r){ball.x=W-ball.r;ball.vx=-Math.abs(ball.vx);sfxBounce();}
    if(ball.y<ball.r){ball.y=ball.r;ball.vy=Math.abs(ball.vy);sfxBounce();}
    hitRect(P,true,null);
    for(const p of platforms)hitRect(pRect(p),false,null);
    for(const b of blocks){if(b.alive&&hitRect(bRect(b),true,b))break;}
    const n=Math.hypot(ball.vx,ball.vy);
    if(n>0){ball.vx*=ball.speed/n;ball.vy*=ball.speed/n;}
    if(ball.y>H+40){onBallLost();return;}
  }
}

function onBallLost(){
  lives--;sfxDeath();
  if(lives<=0)mode='gameover';else resetBall();
}

function playerUpdate(dt){
  const dir=(keys.has('ArrowRight')||keys.has('d')?1:0)-(keys.has('ArrowLeft')||keys.has('a')?1:0);
  if(dir){P.vx+=dir*2600*dt;P.facing=dir;}
  else{const s=Math.sign(P.vx);P.vx-=s*Math.min(Math.abs(P.vx),2400*dt);}
  P.vx=Math.max(-430,Math.min(430,P.vx));
  const prevBottom=P.y+P.h;
  P.vy=Math.min(P.vy+GRAV*dt,1500);
  P.x+=P.vx*dt;
  if(P.x<0){P.x=0;P.vx=0;}if(P.x>W-P.w){P.x=W-P.w;P.vx=0;}
  P.y+=P.vy*dt;P.onGround=false;
  if(P.y+P.h>=groundY()){P.y=groundY()-P.h;P.vy=0;P.onGround=true;}
  else if(P.vy>=0){
    for(const p of platforms){
      const r=pRect(p);
      if(P.x+P.w>r.x+3&&P.x<r.x+r.w-3&&prevBottom<=r.y+2&&P.y+P.h>=r.y){
        P.y=r.y-P.h;P.vy=0;P.onGround=true;break;
      }
    }
  }
}

function update(dt){
  for(let i=particles.length-1;i>=0;i--){
    const p=particles[i];p.life-=dt;p.x+=p.vx*dt;p.y+=p.vy*dt;p.vy+=900*dt;
    if(p.life<=0)particles.splice(i,1);
  }
  for(const b of blocks)if(b.flash>0)b.flash-=dt;

  if(mode==='play'){
    playerUpdate(dt);
    if(ball.attached){ball.x=P.x+P.w/2;ball.y=P.y-ball.r-4;}
    else moveBall();
    if(mode==='play'&&blocks.length&&blocks.every(b=>!b.alive)){
      mode='levelup';timer=2;bonusMsg=100*level;score+=bonusMsg;sfxLevel();
    }
  }else if(mode==='levelup'){
    timer-=dt;
    if(timer<=0){
      level++;
      if(level>MAXL){mode='win';}
      else{makeBlocks();resetBall();mode='play';}
    }
  }
}
