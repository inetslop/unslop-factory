    var spd = 30*dt;
    var prev = f.a;
    if(f.a < target) f.a = Math.min(target, f.a + spd);
    else if(f.a > target) f.a = Math.max(target, f.a - spd);
    f.omega = (f.a - prev)/dt;
  }

  var steps = 5, sdt = dt/steps;
  for(var s=0;s<steps;s++){
    ball.x += ball.vx*sdt;
    ball.y += ball.vy*sdt;
    var fr = Math.exp(-0.4*sdt);
    ball.vx *= fr; ball.vy *= fr;

    // pockets
    if(invuln <= 0){
      for(var p=0;p<pockets.length;p++){
        var pk = pockets[p];
        var pdx = ball.x-pk.x, pdy = ball.y-pk.y;
        if(pdx*pdx+pdy*pdy < (pk.r-6)*(pk.r-6)){ loseBall(); return; }
      }
    }

    // walls
    if(ball.x < ball.r){ ball.x = ball.r; if(ball.vx<0){ball.vx=-ball.vx*0.9; beep(120,0.05,'sine',0.07);} }
    if(ball.x > W-ball.r){ ball.x = W-ball.r; if(ball.vx>0){ball.vx=-ball.vx*0.9; beep(120,0.05,'sine',0.07);} }
    if(ball.y < ball.r){ ball.y = ball.r; if(ball.vy<0){ball.vy=-ball.vy*0.9; beep(120,0.05,'sine',0.07);} }
    if(ball.y > H-ball.r){ ball.y = H-ball.r; if(ball.vy>0){ball.vy=-ball.vy*0.9; beep(120,0.05,'sine',0.07);} }

    // bumpers
    for(var b=0;b<bumpers.length;b++){
      var bu = bumpers[b];
      var bx = ball.x-bu.x, by = ball.y-bu.y;
      var bd = Math.sqrt(bx*bx+by*by);
      if(bd < bu.r + ball.r && bd > 0.001){
        bx/=bd; by/=bd;
        ball.x = bu.x + bx*(bu.r+ball.r);
        ball.y = bu.y + by*(bu.r+ball.r);
        var vn2 = ball.vx*bx + ball.vy*by;
        if(vn2 < 0){
          ball.vx -= (1+0.5)*vn2*bx;
          ball.vy -= (1+0.5)*vn2*by;
          ball.vx += bx*380;
          ball.vy += by*380;
          bu.hitT = 0.28;
          addScore(bu.pts, bu.x, bu.y - bu.r - 14);
          beep(560 + Math.random()*220, 0.09, 'square', 0.13);
        }
      }
    }

    // flippers
    for(var fl=0;fl<2;fl++) collideFlipper(flippers[fl]);

    // speed cap
    var sp = Math.sqrt(ball.vx*ball.vx+ball.vy*ball.vy);
    if(sp > 950){ ball.vx *= 950/sp; ball.vy *= 950/sp; }
  }

  invuln -= dt;
  bumpers.forEach(function(b){ if(b.hitT>0) b.hitT -= dt; });

  // stuck detection / auto nudge
  var spd2 = Math.sqrt(ball.vx*ball.vx+ball.vy*ball.vy);
  if(spd2 < 9) stuckT += dt; else stuckT = 0;
  if(stuckT > 6){
    var na = Math.random()*Math.PI*2;
    ball.vx += Math.cos(na)*340; ball.vy += Math.sin(na)*340;
    stuckT = 0;
    addPopup(ball.x, ball.y-24, 'NUDGED!', '#ffd76a');
  }

  // trail
  trail.push({x:ball.x, y:ball.y});
  if(trail.length > 16) trail.shift();
}

function loseBall(){
  balls--;
  beep(500, 0.5, 'sawtooth', 0.16, 70);
  if(balls <= 0){ gameOver(); }
  else {
    addPopup(ball.x, ball.y, 'BALL LOST!', '#ff7b6a');
    resetBall();
  }
}

function addScore(pts, x, y){ score += pts; addPopup(x, y, '+'+pts, '#7fe8ff'); }

/* ---------- rendering ---------- */
function draw(){
  ctx.clearRect(0,0,W,H);

  // table background
  var g = ctx.createRadialGradient(W/2,H/2,60, W/2,H/2,W*0.7);
  g.addColorStop(0,'#101c34'); g.addColorStop(1,'#080d1a');
  ctx.fillStyle = g; ctx.fillRect(0,0,W,H);

  // decorative rings
  ctx.strokeStyle = 'rgba(60,140,220,0.10)'; ctx.lineWidth = 2;
  for(var rr=70; rr<W/2; rr+=95){ ctx.beginPath(); ctx.arc(W/2,H/2,rr,0,Math.PI*2); ctx.stroke(); }
  ctx.setLineDash([6,10]);
  ctx.strokeStyle = 'rgba(80,180,255,0.08)';
  ctx.beginPath(); ctx.moveTo(W/2,30); ctx.lineTo(W/2,H-30); ctx.stroke();
  ctx.setLineDash([]);

  // border
  ctx.strokeStyle = '#2f7fb8'; ctx.lineWidth = 5;
  ctx.shadowColor = 'rgba(60,190,255,0.6)'; ctx.shadowBlur = 14;
  roundRect(3,3,W-6,H-6,12); ctx.stroke();
  ctx.shadowBlur = 0;

  // pockets
  for(var p=0;p<pockets.length;p++){
    var pk = pockets[p];
    var pg = ctx.createRadialGradient(pk.x,pk.y,2, pk.x,pk.y,pk.r);
    pg.addColorStop(0,'#000'); pg.addColorStop(0.75,'#050608'); pg.addColorStop(1,'#1a0d12');
    ctx.fillStyle = pg;
    ctx.beginPath(); ctx.arc(pk.x,pk.y,pk.r,0,Math.PI*2); ctx.fill();
    ctx.strokeStyle = 'rgba(255,90,80,0.7)'; ctx.lineWidth = 2.5;
    ctx.beginPath(); ctx.arc(pk.x,pk.y,pk.r-1,0,Math.PI*2); ctx.stroke();
  }

  // bumpers
  for(var b=0;b<bumpers.length;b++){
    var bu = bumpers[b];
    var flash = bu.hitT > 0 ? Math.min(1, bu.hitT/0.28) : 0;
    ctx.save();
    if(flash>0){ ctx.shadowColor = 'hsla('+bu.hue+',100%,70%,0.9)'; ctx.shadowBlur = 30*flash; }
    ctx.fillStyle = 'hsl('+bu.hue+','+(60+30*flash)+'%,'+(38+28*flash)+'%)';
    ctx.beginPath(); ctx.arc(bu.x,bu.y,bu.r*(1+0.12*flash),0,Math.PI*2); ctx.fill();
    ctx.fillStyle = 'hsl('+bu.hue+',90%,'+(70+20*flash)+'%)';
    ctx.beginPath(); ctx.arc(bu.x,bu.y,bu.r*0.55,0,Math.PI*2); ctx.fill();
    ctx.strokeStyle = 'rgba(255,255,255,'+(0.35+0.5*flash)+')'; ctx.lineWidth = 2;
    ctx.beginPath(); ctx.arc(bu.x,bu.y,bu.r*(1+0.12*flash),0,Math.PI*2); ctx.stroke();
    ctx.fillStyle = 'rgba(10,20,35,0.9)';
    ctx.font = 'bold '+(bu.r*0.62)+'px Courier New'; ctx.textAlign='center'; ctx.textBaseline='middle';
    ctx.fillText(bu.pts, bu.x, bu.y+1);
    ctx.restore();
  }

  // flippers
  for(var f=0;f<2;f++){
    var fl = flippers[f], t = flipperTip(fl);
    ctx.strokeStyle = '#ffb347'; ctx.lineCap='round'; ctx.lineWidth = fl.fr*2;
    ctx.shadowColor = 'rgba(255,170,60,0.55)'; ctx.shadowBlur = 12;
    ctx.beginPath(); ctx.moveTo(fl.px,fl.py); ctx.lineTo(t.x,t.y); ctx.stroke();
    ctx.shadowBlur = 0;
    ctx.fillStyle = '#8a4b12';
    ctx.beginPath(); ctx.arc(fl.px,fl.py,fl.fr*0.55,0,Math.PI*2); ctx.fill();
  }

  // trail
  for(var tr=0;tr<trail.length;tr++){
    var a = (tr/trail.length)*0.28;
    ctx.fillStyle = 'rgba(160,230,255,'+a+')';
    ctx.beginPath(); ctx.arc(trail[tr].x, trail[tr].y, ball.r*(tr/trail.length)*0.9, 0, Math.PI*2); ctx.fill();
  }

  // ball
  if(state === 'playing'){
    var bg = ctx.createRadialGradient(ball.x-4,ball.y-5,2, ball.x,ball.y,ball.r);
    bg.addColorStop(0,'#ffffff'); bg.addColorStop(0.5,'#cfe8ff'); bg.addColorStop(1,'#5f9fd8');
    ctx.fillStyle = bg;
    if(invuln > 0 && Math.floor(invuln*12)%2===0) ctx.globalAlpha = 0.45;
    ctx.beginPath(); ctx.arc(ball.x,ball.y,ball.r,0,Math.PI*2); ctx.fill();
    ctx.globalAlpha = 1;
  }

  // popups
  for(var pp=popups.length-1; pp>=0; pp--){
    var pu = popups[pp];
    ctx.globalAlpha = Math.max(0, Math.min(1, pu.t));
    ctx.fillStyle = pu.col;
    ctx.font = 'bold 20px Courier New'; ctx.textAlign='center';
    ctx.fillText(pu.txt, pu.x, pu.y - (1.2-pu.t)*40);
    ctx.globalAlpha = 1;
  }

  // HUD
  if(state !== 'start'){
    ctx.fillStyle = 'rgba(200,240,255,0.85)';
    ctx.font = 'bold 24px Courier New'; ctx.textAlign='center';
    ctx.fillText('SCORE ' + score, W/2, 34);
    ctx.font = '15px Courier New'; ctx.fillStyle = 'rgba(160,200,235,0.7)';
    var dots = ''; for(var d=0; d<balls; d++) dots += '\u25CF ';
    ctx.fillText('BALLS ' + (dots||'—'), W/2, 56);
    ctx.textAlign='left'; ctx.font='13px Courier New'; ctx.fillStyle='rgba(140,180,220,0.4)';
    ctx.fillText('BEST '+best, 14, H-14);
  }
}

function roundRect(x,y,w,h,r){
  ctx.beginPath();
  ctx.moveTo(x+r,y);
  ctx.arcTo(x+w,y,x+w,y+h,r);
  ctx.arcTo(x+w,y+h,x,y+h,r);
  ctx.arcTo(x,y+h,x,y,r);
  ctx.arcTo(x,y,x+w,y,r);
  ctx.closePath();
}

var last = performance.now();
function frame(now){
  var dt = Math.min(0.033, (now-last)/1000);
  last = now;
  if(state === 'playing') step(dt);
  for(var i=popups.length-1;i>=0;i--){ popups[i].t -= dt*0.9; if(popups[i].t<=0) popups.splice(i,1); }
  draw();
  requestAnimationFrame(frame);
}
requestAnimationFrame(frame);

function handleKey(code, down){
  initAudio();
  if(down && (code==='Enter' || code==='Space')){
    if(state==='start'){ startGame(); return; }
  }
  if(down && code==='KeyR'){ startGame(); return; }
  if(code==='ArrowLeft'||code==='KeyA') keys.left = down;
  if(code==='ArrowRight'||code==='KeyD') keys.right = down;
  if(down && code==='Space' && state==='playing'){
    var a = Math.random()*Math.PI*2;
    ball.vx += Math.cos(a)*300; ball.vy += Math.sin(a)*300;
    stuckT = 0;
    beep(300, 0.06, 'triangle', 0.1);
    addPopup(ball.x, ball.y-24, 'NUDGE', '#9fd8ff');
  }
}
