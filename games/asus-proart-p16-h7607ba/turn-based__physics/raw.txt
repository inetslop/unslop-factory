    if(keys['ArrowLeft'])p.angle-=1.6*dt*3;
    if(keys['ArrowRight'])p.angle+=1.6*dt*3;
    p.angle=Math.max(p.minA,Math.min(p.maxA,p.angle));
    if(keys['ArrowUp'])p.power=Math.min(1,p.power+dt*1.2);
    if(keys['ArrowDown'])p.power=Math.max(0,p.power-dt*1.2);
    if(charging){chargeT+=dt*1.4;let t=chargeT%2;p.power=t<1?t:2-t;}
  }
  if(state==='flying'&&ball)stepBall(dt);
  for(let i=particles.length-1;i>=0;i--){const p=particles[i];p.life-=dt;p.vy+=GRAV*.5*dt;p.x+=p.vx*dt;p.y+=p.vy*dt;if(p.life<=0)particles.splice(i,1);}
  for(let i=flashes.length-1;i>=0;i--){flashes[i].t-=dt;if(flashes[i].t<=0)flashes.splice(i,1);}
}

// ---------- render ----------
function draw(){
  ctx.setTransform(dpr*scale,0,0,dpr*scale,dpr*ox,dpr*oy);
  // bg
  const g=ctx.createLinearGradient(0,0,0,H);g.addColorStop(0,'#101830');g.addColorStop(1,'#0a0e1a');
  ctx.fillStyle=g;ctx.fillRect(-ox/scale,-oy/scale,innerWidth/scale,innerHeight/scale);
  ctx.save();ctx.beginPath();ctx.rect(0,0,W,H);ctx.clip();
  // grid
  ctx.strokeStyle='rgba(255,255,255,.04)';ctx.lineWidth=1;
  for(let x=0;x<=W;x+=60){ctx.beginPath();ctx.moveTo(x,0);ctx.lineTo(x,H);ctx.stroke();}
  for(let y=0;y<=H;y+=60){ctx.beginPath();ctx.moveTo(0,y);ctx.lineTo(W,y);ctx.stroke();}
  // obstacles
  for(const r of obstacles){
    const rg=ctx.createLinearGradient(r.x,r.y,r.x,r.y+r.h);
    rg.addColorStop(0,'#3a4568');rg.addColorStop(1,'#242c47');
    ctx.fillStyle=rg;ctx.fillRect(r.x,r.y,r.w,r.h);
    ctx.strokeStyle='rgba(255,255,255,.2)';ctx.strokeRect(r.x+.5,r.y+.5,r.w-1,r.h-1);
  }
  // launchers
  for(let i=0;i<players.length;i++){
    const p=players[i];
    ctx.fillStyle='#1c2340';ctx.beginPath();ctx.arc(p.lx,p.ly,22,Math.PI,0);ctx.fill();
    if(i===cur&&state==='aim'){
      // barrel + preview
      ctx.strokeStyle=p.col;ctx.lineWidth=7;ctx.lineCap='round';
      const bl=40+p.power*36;
      ctx.beginPath();ctx.moveTo(p.lx,p.ly);ctx.lineTo(p.lx+Math.cos(p.angle)*bl,p.ly+Math.sin(p.angle)*bl);ctx.stroke();
      // trajectory preview
      let px=p.lx+Math.cos(p.angle)*34,py=p.ly+Math.sin(p.angle)*34;
      const sp=500+p.power*980;let vx=Math.cos(p.angle)*sp,vy=Math.sin(p.angle)*sp;
      ctx.fillStyle=p.col;
      for(let s=0;s<110;s++){
        vy+=GRAV/60;px+=vx/60;py+=vy/60;
        if(px<BALL_R||px>W-BALL_R)break;if(py>H-BALL_R)break;
        let hit=false;for(const r of obstacles){if(px>r.x-BALL_R&&px<r.x+r.w+BALL_R&&py>r.y-BALL_R&&py<r.y+r.h+BALL_R)hit=true;}
        if(hit)break;
        if(s%4===0){ctx.globalAlpha=Math.max(.05,.6-s*.005);ctx.beginPath();ctx.arc(px,py,3.2,0,7);ctx.fill();}
      }
      ctx.globalAlpha=1;
      // power bar
      ctx.fillStyle='rgba(255,255,255,.15)';ctx.fillRect(p.lx-40,p.ly+34,80,8);
      ctx.fillStyle=p.col;ctx.fillRect(p.lx-40,p.ly+34,80*p.power,8);
    } else {
      ctx.strokeStyle='rgba(255,255,255,.15)';ctx.lineWidth=7;ctx.lineCap='round';
      ctx.beginPath();ctx.moveTo(p.lx,p.ly);ctx.lineTo(p.lx+Math.cos(i===0?-45*Math.PI/180:-135*Math.PI/180)*40,p.ly+Math.sin(i===0?-45*Math.PI/180:-135*Math.PI/180)*40);ctx.stroke();
    }
  }
  // cores
  for(let i=0;i<players.length;i++){
    const p=players[i];if(p.hp<=0)continue;
    ctx.save();
    ctx.shadowColor=p.col;ctx.shadowBlur=30;
    const cg=ctx.createRadialGradient(p.cx-8,p.cy-8,4,p.cx,p.cy,CORE_R);
    cg.addColorStop(0,'#ffffff');cg.addColorStop(.35,p.col);cg.addColorStop(1,p.glow+'0.6)');
    ctx.fillStyle=cg;ctx.beginPath();ctx.arc(p.cx,p.cy,CORE_R,0,7);ctx.fill();
    ctx.restore();
    ctx.strokeStyle='rgba(255,255,255,.5)';ctx.lineWidth=2;
    ctx.beginPath();ctx.arc(p.cx,p.cy,CORE_R,0,7);ctx.stroke();
    // hp pips
    for(let k=0;k<p.hp;k++){ctx.fillStyle='#fff';ctx.beginPath();ctx.arc(p.cx-14+k*14,p.cy+6,3.5,0,7);ctx.fill();}
  }
  // flashes
  for(const f of flashes){
    ctx.strokeStyle='rgba(255,255,255,'+(f.t/.4)+')';ctx.lineWidth=4;
    ctx.beginPath();ctx.arc(f.x,f.y,CORE_R+(0.4-f.t)*90,0,7);ctx.stroke();
  }
  // ball
  if(ball){
    ctx.save();ctx.shadowColor='#ffd24d';ctx.shadowBlur=18;
    const bg=ctx.createRadialGradient(ball.x-4,ball.y-4,2,ball.x,ball.y,BALL_R);
    bg.addColorStop(0,'#fff6d8');bg.addColorStop(1,'#ffb347');
    ctx.fillStyle=bg;ctx.beginPath();ctx.arc(ball.x,ball.y,BALL_R,0,7);ctx.fill();ctx.restore();
  }
  // particles
  for(const p of particles){ctx.globalAlpha=Math.max(0,p.life/p.max);ctx.fillStyle=p.col;ctx.beginPath();ctx.arc(p.x,p.y,p.r,0,7);ctx.fill();}
  ctx.globalAlpha=1;
  ctx.restore();
}

let last=performance.now();
function loop(now){
  const dt=Math.min(.033,(now-last)/1000);last=now;
  if(state==='aim'||state==='flying')update(dt);
  else{for(let i=particles.length-1;i>=0;i--){const p=particles[i];p.life-=dt;p.vy+=GRAV*.5*dt;p.x+=p.vx*dt;p.y+=p.vy*dt;if(p.life<=0)particles.splice(i,1);}
       for(let i=flashes.length-1;i>=0;i--){flashes[i].t-=dt;if(flashes[i].t<=0)flashes.splice(i,1);}}
  draw();
  requestAnimationFrame(loop);
}
requestAnimationFrame(loop);
