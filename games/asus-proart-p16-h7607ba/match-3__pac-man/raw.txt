function wouldMatch(x,y,t){

function newBoard(){
  for(var y=0;y<N;y++)for(var x=0;x<N;x++){
    var t,tries=0;
    do{ t=randType(); tries++; }while(tries<50 && wouldMatch(x,y,t));

  }
}

function findMatches(){
  var set={};
  function add(x,y){ set[x+','+y]=true; }
  for(var y=0;y<N;y++){
    var run=1;
    for(var x=1;x<=N;x++){
      if(x<N && type[y][x]===type[y][x-1] && type[y][x]>=0) run++;
      else{ if(run>=3){ for(var k=x-run;k<x;k++) add(k,y); } run=1; }
    }
  }
  for(var x=0;x<N;x++){
    var r2=1;
    for(var y2=1;y2<=N;y2++){

    }
  }

}

function hasValidMove(){
  function test(x1,y1,x2,y2){
    var a=type[y1][x1]; type[y1][x1]=type[y2][x2]; type[y2][x2]=a;

    type[y2][x2]=type[y1][x1]; type[y1][x1]=a; // swap back (careful)

function doSwap(a,b){

function swapCells(a,b){
  var t=type[a.y][a.x]; type[a.y][a.x]=type[b.y][b.x]; type[b.y][b.x]=t;
  ox[a.y][a.x]=(a.x-b.x); oy[a.y][a.x]=(a.y-b.y);
  ox[b.y][b.x]=(b.x-a.x); oy[b.y][b.x]=(b.y-a.y);

function attemptSwap(a,b){
  if(phase!=='idle'||screen!=='play') return;
  if(Math.abs(a.x-b.x)+Math.abs(a.y-b.y)!==1) return;
  swapCells(a,b);
  moves--; combo=1;
  sfx.swap();
  phase='swap';
}

function startPop(cells){
  var cx=0,cy=0;
  for(var i=0;i<cells.length;i++){
    var c=cells[i];
    popT[c.y][c.x]=1;
    cx+=c.x; cy+=c.y;

  addFloat(cx/cells.length, cy/cells.length, (cells.length*10*combo)+(combo>1?'  x'+combo:''), combo>1);
  sfx.pop(combo);
  phase='pop';
}

function applyGravity(){
  for(var x=0;x<N;x++){
    var write=N-1;
    for(var y=N-1;y>=0;y--){
      if(type[y][x]>=0){
        if(write!==y){
          type[write][x]=type[y][x]; type[y][x]=-1;
          ox[write][x]=0; oy[write][x]=-(write-y); popT[write][x]=1;
        }
        write--;
      }
    }
    var empty=write+1; // cells 0..write are empty
    for(var y2=write;y2>=0;y2--){
      type[y2][x]=randType();
      ox[y2][x]=0; oy[y2][x]=-(y2+1); popT[y2][x]=1;
    }
  }
}

    ctx.lineWidth=2;
    roundRect(L.bx+cur.x*cell+2,L.by+cur.y*cell+2,cell-4,cell-4,8); ctx.stroke();
  }
  // tiles
  for(var y=0;y<N;y++)for(var x=0;x<N;x++){
    var t=type[y][x];
    if(t<0) continue;
    var px=L.bx+x*cell+cell/2+ox[y][x]*cell;
    var py=L.by+y*cell+cell/2+oy[y][x]*cell;
    var sc=(popT[y][x]!==undefined&&popT[y][x]<1)?Math.max(popT[y][x],0.01):1;
    ctx.globalAlpha=sc<1?sc:1;
    if(t===4) drawPower(ctx,px,py,cell*sc);
    else drawGhost(ctx,px,py,cell,COLORS[t],sc);
    ctx.globalAlpha=1;
  }
  // floats
  for(var i=floats.length-1;i>=0;i--){
    var f=floats[i];
    ctx.globalAlpha=Math.min(1,f.t);
    ctx.fillStyle=f.big?'#ffe14a':'#fff';
    ctx.font='bold '+(f.big?Math.floor(cell*0.55):Math.floor(cell*0.38))+'px monospace';
    ctx.textAlign='center';
    ctx.fillText(f.txt,L.bx+f.x*cell+cell/2,L.by+f.y*cell+cell/2-(1.4-f.t)*40);
    ctx.globalAlpha=1;
  }
}

function roundRect(x,y,w,h,r){
  ctx.beginPath();
  ctx.moveTo(x+r,y);
  ctx.arcTo(x+w,y,x+w,y+h,r);
  ctx.arcTo(x+w,y+h,x,y+h,r);
  ctx.arcTo(x,y+h,x,y,r);
  ctx.arcTo(x,y,x+w,y,r);
  ctx.closePath();
}

/* ---------- update ---------- */
var last=performance.now();
function update(dt){
  time+=dt;
  for(var i=floats.length-1;i>=0;i--){ floats[i].t-=dt; if(floats[i].t<=0) floats.splice(i,1); }
  if(screen!=='play') return;

  var settled=true, i,x,y;
  for(y=0;y<N;y++)for(x=0;x<N;x++){
    if(Math.abs(ox[y][x])>0.02||Math.abs(oy[y][x])>0.02) settled=false;
  }

  if(phase==='swap'||phase==='revert'){
    var k=Math.min(1,dt*14);
    for(y=0;y<N;y++)for(x=0;x<N;x++){ ox[y][x]+=(0-ox[y][x])*k; oy[y][x]+=(0-oy[y][x])*k; }
    if(settled){
      for(y=0;y<N;y++)for(x=0;x<N;x++){ ox[y][x]=0; oy[y][x]=0; }
      var m=findMatches();
      if(phase==='swap'){
        if(m.length) startPop(m);
        else{
          // revert
          for(var i2=0;i2<lastSwap.length;i2+=4){}
          swapCells(lastSwapA,lastSwapB);
          sfx.bad();
          phase='revert';
        }
      }else{
        afterSettle();
      }
    }
  }
  else if(phase==='pop'){
    var done=true;
    for(y=0;y<N;y++)for(x=0;x<N;x++){
      if(popT[y][x]!==undefined && popT[y][x]<1){
        popT[y][x]-=dt/0.26;
        if(popT[y][x]>0) done=false;
      }
    }
    if(done){
      for(y=0;y<N;y++)for(x=0;x<N;x++){
        if(popT[y][x]!==undefined && popT[y][x]<=0){ type[y][x]=-1; popT[y][x]=1; }
      }
      applyGravity();
      phase='fall';
    }
  }
  else if(phase==='fall'){
    var sp=32*dt, done2=true;
    for(y=0;y<N;y++)for(x=0;x<N;x++){
      if(oy[y][x]<0){
        oy[y][x]=Math.min(0,oy[y][x]+sp);
        if(oy[y][x]<0) done2=false;
      }
    }
    if(done2){
      for(y=0;y<N;y++)for(x=0;x<N;x++) oy[y][x]=0;
      var m2=findMatches();
      if(m2.length){ combo++; startPop(m2); }
      else afterSettle();
    }
  }
}

var lastSwapA={x:0,y:0}, lastSwapB={x:0,y:0};
// hook: attemptSwap stores swap endpoints
var _attempt=attemptSwap;
attemptSwap=function(a,b){
  if(phase!=='idle'||screen!=='play') return;
  if(Math.abs(a.x-b.x)+Math.abs(a.y-b.y)!==1) return;
  lastSwapA={x:a.x,y:a.y}; lastSwapB={x:b.x,y:b.y};
  swapCells(lastSwapA,lastSwapB);
  moves--; combo=1;
  sfx.swap();
  phase='swap';
};

/* ---------- render loop ---------- */
function frame(now){
  var dt=Math.min(0.05,(now-last)/1000); last=now;
  update(dt);
  ctx.fillStyle='#000'; ctx.fillRect(0,0,W,H);
  if(screen==='start'){ drawStart(); }
  else{
    // subtle maze lines background
    ctx.strokeStyle='rgba(43,59,255,0.12)'; ctx.lineWidth=1;
    var step=48;
    for(var x=(W/2)%step;x<W;x+=step){ ctx.beginPath(); ctx.moveTo(x,0); ctx.lineTo(x,H); ctx.stroke(); }
    for(var y=y0=(H/2)%step;y<H;y+=step){ ctx.beginPath(); ctx.moveTo(0,y); ctx.lineTo(W,y); ctx.stroke(); }
    drawBoard();
    drawHUD();
    if(screen==='over') drawOver();
  }
  requestAnimationFrame(frame);
}

function cellAt(px,py){
  var L=layout();
  var x=Math.floor((px-L.bx)/L.cell), y=Math.floor((py-L.by)/L.cell);
  if(x<0||y<0||x>=N||y>=N) return null;
  return {x:x,y:y};
}

  var r=canvas.getBoundingClientRect();
  return {x:e.clientX-r.left,y:e.clientY-r.top};
}

  dragDone=true;
  var t={x:downCell.x,y:downCell.y};
  if(Math.abs(dx)>Math.abs(dy)) t.x+=dx>0?1:-1; else t.y+=dy>0?1:-1;

  downCell=null;
});

});
