function roundRect(x,y,w,h,r){

  ctx.moveTo(x+r,y); ctx.arcTo(x+w,y,x+w,y+h,r); ctx.arcTo(x+w,y+h,x,y+h,r);
  ctx.arcTo(x,y+h,x,y,r); ctx.arcTo(x,y,x+w,y,r); ctx.closePath();
}

function draw(t,dt){
  const g=DIRS[gravIdx], L=layout();
  const cs=L.cell, ox=L.ox, oy=L.oy;

  /* background */
  const bg=ctx.createLinearGradient(0,0,0,H);
  bg.addColorStop(0,'#181b34'); bg.addColorStop(1,'#101226');
  ctx.fillStyle=bg; ctx.fillRect(0,0,W,H);

  /* HUD */
  ctx.textBaseline='middle';
  ctx.font='bold '+Math.max(13,Math.min(18,W*0.018))+'px "Courier New",monospace';
  ctx.fillStyle='#dfe3ff'; ctx.textAlign='left';
  ctx.fillText('LEVEL '+(levelIdx+1)+'/'+LEVELS.length+' \u2014 '+LEVELS[levelIdx].name,16,26);
  ctx.textAlign='center';
  ctx.fillStyle='#8f97c9';
  const best=getBest(levelIdx);
  ctx.fillText('MOVES: '+moves+(best!=null?'   BEST: '+best:''),W/2,48);

  /* gravity badge */
  const bx=W-52, by=34;
  ctx.beginPath(); ctx.arc(bx,by,26,0,Math.PI*2);
  ctx.fillStyle='rgba(255,215,106,0.1)'; ctx.fill();
  ctx.strokeStyle='#ffd76a'; ctx.lineWidth=2; ctx.stroke();
  drawArrow(bx,by,30,g,0.9);
  ctx.textAlign='right'; ctx.fillStyle='#8f97c9';
  ctx.fillText('GRAVITY',bx-34,by);

  /* board shadow */
  ctx.fillStyle='rgba(0,0,0,0.35)';
  roundRect(ox-6,oy-6,cs*cols+12,cs*rows+12,10); ctx.fill();

  /* tiles */
  for(let y=0;y<rows;y++) for(let x=0;x<cols;x++){
    const px=ox+x*cs, py=oy+y*cs;
    if(walls.has(K(x,y))){
      ctx.fillStyle=((x+y)%2)?'#454b74':'#3f4569';
      ctx.fillRect(px,py,cs,cs);
      ctx.strokeStyle='rgba(0,0,0,0.3)'; ctx.lineWidth=1;
      ctx.strokeRect(px+0.5,py+0.5,cs-1,cs-1);
    } else {
      ctx.fillStyle=((x*7+y*13)%2)?'#262a48':'#232743';
      ctx.fillRect(px,py,cs,cs);
    }
  }

  /* targets */
  const pulse=0.5+0.5*Math.sin(t*4);
  for(const tk of targets){
    const p=tk.split(','), x=+p[0], y=+p[1];
    const px=ox+x*cs, py=oy+y*cs;
    ctx.strokeStyle='rgba(255,184,77,'+(0.55+0.4*pulse)+')';
    ctx.lineWidth=3;
    roundRect(px+cs*0.18,py+cs*0.18,cs*0.64,cs*0.64,6); ctx.stroke();
    ctx.beginPath(); ctx.arc(px+cs/2,py+cs/2,cs*(0.1+0.05*pulse),0,Math.PI*2);
    ctx.fillStyle='rgba(255,184,77,'+(0.6+0.4*pulse)+')'; ctx.fill();
  }

  /* particles */
  for(let i=particles.length-1;i>=0;i--){
    const p=particles[i];
    p.life-=dt; if(p.life<=0){ particles.splice(i,1); continue; }
    p.x+=p.vx*dt; p.y+=p.vy*dt; p.vy+=2.5*dt;
    ctx.globalAlpha=Math.max(0,p.life/p.max)*0.7;
    ctx.fillStyle='#c9b48a';
    const s=cs*0.08;
    ctx.fillRect(ox+p.x*cs,oy+p.y*cs,s,s);
  }
  ctx.globalAlpha=1;

  /* boxes */
  for(const b of boxes){
    const px=ox+b.x*cs, py=oy+b.y*cs;
    const onT=targets.has(K(b.x,b.y));
    const m=cs*0.12;
    ctx.fillStyle=onT?'#7a5c2e':'#8a6a35';
    roundRect(px+m,py+m,cs-2*m,cs-2*m,cs*0.1); ctx.fill();
    ctx.strokeStyle=onT?'#7dff9e':'#5c441f'; ctx.lineWidth=Math.max(2,cs*0.06);
    roundRect(px+m,py+m,cs-2*m,cs-2*m,cs*0.1); ctx.stroke();
    /* X brace */
    ctx.strokeStyle='rgba(60,40,15,0.8)'; ctx.lineWidth=Math.max(1.5,cs*0.05);
    ctx.beginPath();
    ctx.moveTo(px+m*1.6,py+m*1.6); ctx.lineTo(px+cs-m*1.6,py+cs-m*1.6);
    ctx.moveTo(px+cs-m*1.6,py+m*1.6); ctx.lineTo(px+m*1.6,py+cs-m*1.6);
    ctx.stroke();
    if(onT){
      ctx.strokeStyle='rgba(125,255,158,'+(0.4+0.4*pulse)+')';
      ctx.lineWidth=3;
      roundRect(px+m-3,py+m-3,cs-2*m+6,cs-2*m+6,cs*0.12); ctx.stroke();
    }
  }

  /* player (anti-grav astronaut) */
  {
    const px=ox+player.x*cs, py=oy+player.y*cs;
    const bob=Math.sin(t*3)*cs*0.04;
    const m=cs*0.18, s=cs-2*m;
    /* boot glow */
    ctx.beginPath(); ctx.arc(px+cs/2,py+cs-m*0.6,s*0.42,0,Math.PI*2);
    ctx.fillStyle='rgba(110,231,255,'+(0.12+0.08*pulse)+')'; ctx.fill();
    /* body */
    ctx.fillStyle='#7ad0ff';
    roundRect(px+m,py+m+bob,s,s,cs*0.2); ctx.fill();
    ctx.strokeStyle='#3a9fd4'; ctx.lineWidth=Math.max(2,cs*0.06);
    roundRect(px+m,py+m+bob,s,s,cs*0.2); ctx.stroke();
    /* visor */
    ctx.fillStyle='#1b2547';
    const vx=px+cs/2+player.fx*s*0.18, vy=py+cs/2+bob+player.fy*s*0.1;
    ctx.beginPath(); ctx.arc(vx,vy,s*0.3,0,Math.PI*2); ctx.fill();
    ctx.fillStyle='#bfeaff';
    ctx.beginPath(); ctx.arc(vx+s*0.09,vy-s*0.1,s*0.08,0,Math.PI*2); ctx.fill();
  }

  /* gravity rotation flash: big translucent arrow across the board */
  if(rotFlash>0){
    rotFlash=Math.max(0,rotFlash-dt*2.2);
    const cx=ox+cs*cols/2, cy=oy+cs*rows/2;
    drawArrow(cx,cy,Math.min(cs*cols,cs*rows)*0.5,g,rotFlash*0.3);
  }
}

parse(0);
let prevT=performance.now();
function frame(now){
  const dt=Math.min(0.05,(now-prevT)/1000); prevT=now;
  if(state==='play' && lastDirCode && held.has(lastDirCode) && now-lastMoveT>135){
    const d=DIRKEYS[lastDirCode];
    movePlayer(d[0],d[1]);
    lastMoveT=now;
  }
  draw(now/1000,dt);
  requestAnimationFrame(frame);
}
