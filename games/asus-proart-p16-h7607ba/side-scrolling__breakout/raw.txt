    ball.y = H - 48 - ball.r - 2;
    return;

  ball.x += ball.vx*dt;
  ball.y += ball.vy*dt;

  // top wall
  if(ball.y - ball.r < 0){ ball.y = ball.r; ball.vy = Math.abs(ball.vy); beep(180,0.06,'square',0.08); }
  // left wall (moves with camera)
  if(ball.x - ball.r < camX){ ball.x = camX + ball.r; ball.vx = Math.abs(ball.vx); beep(180,0.06,'square',0.08); }
  // keep from escaping far right
  if(ball.x - (camX + W) > 320 && ball.vx > 0){ ball.vx = -Math.abs(ball.vx); }

  // speed clamp & anti-horizontal
  let sp = Math.hypot(ball.vx, ball.vy);

  if(sp<minS||sp>maxS){ const k=(sp<minS?minS:maxS)/sp; ball.vx*=k; ball.vy*=k; sp=k*sp; }

    ball.vy = (ball.vy<0?-1:1)*130;
    const k2 = Math.sqrt(Math.max(0, sp*sp - 130*130));
    ball.vx = (ball.vx<0?-1:1)*k2;
  }

  // paddle collision
  const py = H - 48;
  const bx = ball.x - camX;

     ball.y + ball.r >= py && ball.y - ball.r <= py + paddle.h){

    ball.vy = -Math.cos(ang)*s;

    combo = 0;
    beep(230,0.08,'square',0.12);

    const b=bricks[i];
    const cx=Math.min(Math.max(ball.x,b.x),b.x+b.w);
    const cy=Math.min(Math.max(ball.y,b.y),b.y+b.h);
    const dx=ball.x-cx, dy=ball.y-cy;
    if(dx*dx+dy*dy < ball.r*ball.r){
      if(Math.abs(dx)>Math.abs(dy)){
        ball.vx = (dx<0?-1:1)*Math.abs(ball.vx);
        ball.x = dx<0 ? b.x-ball.r-0.5 : b.x+b.w+ball.r+0.5;
      } else {
        ball.vy = (dy<0?-1:1)*Math.abs(ball.vy);
        ball.y = dy<0 ? b.y-ball.r-0.5 : b.y+b.h+ball.r+0.5;
      }
      combo++; maxCombo=Math.max(maxCombo,combo);
      score += 5 + combo*5;
      burst(b.x+b.w/2, b.y+b.h/2, b.c);
      bricks.splice(i,1);
      beep(440 + Math.min(combo,10)*40, 0.09,'square',0.13);
      break;

      ball.stuck=true;
      beep(200,0.2,'sawtooth',0.15);
    }

    const p=particles[i];

    if(p.life<=0){ particles.splice(i,1); continue; }
    p.vy+=900*dt;
    p.x+=p.vx*dt; p.y+=p.vy*dt;
  }
}

function drawBackground(){
  const g = ctx.createLinearGradient(0,0,0,H);
  g.addColorStop(0,'#141a33');
  g.addColorStop(0.6,'#0e1226');
  g.addColorStop(1,'#080a16');
  ctx.fillStyle=g;
  ctx.fillRect(0,0,W,H);

  // far parallax towers (silhouette bricks)
  const fOff = camX*0.25;
  const step=140;
  const i0=Math.floor(fOff/step)-1;
  for(let i=i0;i<i0+Math.ceil(W/step)+3;i++){
    const h1 = 60 + hash(i)*180;
    const x = i*step - fOff;
    ctx.fillStyle='rgba(70,90,160,0.15)';
    ctx.fillRect(x, H-h1-40, 46, h1);
    ctx.fillRect(x+70, H-(h1*0.6)-40, 30, h1*0.6);
  }
  // mid parallax dots (stars)
  const mOff = camX*0.5;
  const mstep=90;
  const j0=Math.floor(mOff/mstep)-1;
  for(let j=j0;j<j0+Math.ceil(W/mstep)+3;j++){
    const x=j*mstep - mOff + hash(j+50)*40;
    const y=hash(j+99)*(H*0.6);
    ctx.fillStyle='rgba(150,180,255,'+(0.15+hash(j)*0.3)+')';
    ctx.fillRect(x,y,3,3);
  }
  // left bounce wall glow
  const lg=ctx.createLinearGradient(0,0,26,0);
  lg.addColorStop(0,'rgba(90,200,255,0.35)');
  lg.addColorStop(1,'rgba(90,200,255,0)');
  ctx.fillStyle=lg;
  ctx.fillRect(0,0,26,H);
}

function drawWorld(){
  // bricks
  for(const b of bricks){
    const sx = b.x - camX;
    if(sx > W+10 || sx+b.w < -10) continue;
    ctx.fillStyle=b.c;
    rr(sx,b.y,b.w,b.h,5); ctx.fill();
    ctx.fillStyle='rgba(255,255,255,0.35)';
    rr(sx+3,b.y+3,b.w-6,6,3); ctx.fill();
    ctx.fillStyle='rgba(0,0,0,0.2)';
    rr(sx+3,b.y+b.h-8,b.w-6,5,3); ctx.fill();
  }

  // particles
  for(const p of particles){
    ctx.globalAlpha = Math.max(0,p.life*2);
    ctx.fillStyle=p.c;
    ctx.fillRect(p.x-camX-3, p.y-3, 6, 6);
  }
  ctx.globalAlpha=1;

  // paddle
  const py = H-48;
  const pg = ctx.createLinearGradient(0,py,0,py+paddle.h);
  pg.addColorStop(0,'#9fe8ff');
  pg.addColorStop(1,'#2f7fd6');
  ctx.shadowColor='#5cc8ff'; ctx.shadowBlur=14;
  ctx.fillStyle=pg;
  rr(paddle.sx,py,paddle.w,paddle.h,7); ctx.fill();
  ctx.shadowBlur=0;

  // ball
  const bx = ball.x - camX;
  ctx.shadowColor='#ffffff'; ctx.shadowBlur=12;
  ctx.fillStyle='#ffffff';
  ctx.beginPath(); ctx.arc(bx,ball.y,ball.r,0,Math.PI*2); ctx.fill();
  ctx.shadowBlur=0;
}

function drawHUD(){
  ctx.textAlign='left';
  ctx.font='bold 20px monospace';
  ctx.fillStyle='#e8f1ff';
  ctx.fillText('SCORE '+score, 16, 30);
  ctx.font='14px monospace';
  ctx.fillStyle='rgba(200,220,255,0.7)';
  ctx.fillText('BEST '+best, 16, 50);

  // lives
  for(let i=0;i<3;i++){
    ctx.beginPath();
    ctx.arc(W-24-i*26, 26, 8, 0, Math.PI*2);
    if(i<lives){ ctx.fillStyle='#ffffff'; ctx.fill(); }
    else { ctx.strokeStyle='rgba(255,255,255,0.3)'; ctx.lineWidth=2; ctx.stroke(); }
  }

  // speed meter
  const spd = Math.min(1,(130+Math.min(150,t*3)-130)/150);
  ctx.fillStyle='rgba(255,255,255,0.15)';
  ctx.fillRect(W-140,48,124,6);
  ctx.fillStyle='#ff9d5c';
  ctx.fillRect(W-140,48,124*spd,6);

  // combo
  if(combo>1){
    ctx.textAlign='center';
    ctx.font='bold 26px monospace';
    ctx.fillStyle='#ffe066';
    ctx.fillText('COMBO x'+combo, W/2, 40);
  }
  // launch hint
  if(ball.stuck){
    ctx.textAlign='center';
    ctx.font='16px monospace';
    ctx.fillStyle='rgba(255,255,255,'+(0.5+0.4*Math.sin(t*6))+')';
    ctx.fillText('PRESS SPACE / CLICK TO LAUNCH', W/2, H-90);
  }
}

function drawCenter(title, lines){
  ctx.fillStyle='rgba(5,8,18,0.78)';
  ctx.fillRect(0,0,W,H);
  ctx.textAlign='center';
  ctx.font='bold '+Math.min(56, W/9)+'px monospace';
  ctx.shadowColor='#5cc8ff'; ctx.shadowBlur=24;
  ctx.fillStyle='#eaf6ff';
  ctx.fillText(title, W/2, H*0.3);
  ctx.shadowBlur=0;
  ctx.font='17px monospace';
  ctx.fillStyle='rgba(210,230,255,0.9)';
  let y = H*0.44;
  for(const l of lines){ ctx.fillText(l, W/2, y); y+=28; }
}

function draw(){
  drawBackground();
  if(state==='start'){
    drawCenter('NEON RUSH BREAKOUT',[
      'A side-scrolling breakout — the world keeps moving!',
      '',
      'MOVE   :  arrow keys / A D  (or mouse / touch)',
      'LAUNCH :  SPACE or click',
      'RESTART:  R',
      '',
      'Destroy bricks for points. Chain hits without touching',
      'the paddle to build a combo multiplier.',
      'Don\'t let the ball fall — you have 3 lives.'
    ]);
    ctx.font='bold 22px monospace';
    ctx.fillStyle='#ffe066';
    ctx.fillText('PRESS SPACE OR CLICK TO START', W/2, H*0.82);
    return;
  }

  drawWorld();
  drawHUD();

  if(state==='over'){
    drawCenter('GAME OVER',[
      'SCORE   '+score,
      'BEST    '+best,
      'MAX COMBO  x'+maxCombo,
      '',
      'PRESS R / ENTER OR CLICK TO PLAY AGAIN'
    ]);
  }
}

/* ---------- main loop ---------- */
let last = performance.now();
function frame(now){
  const dt = Math.min(0.033, (now-last)/1000);
  last = now;
  if(state==='play') t += 0; // time handled in update
  update(dt);
  draw();
  requestAnimationFrame(frame);
}
requestAnimationFrame(frame);
})();
