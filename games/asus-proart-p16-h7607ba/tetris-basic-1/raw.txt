let AC=null,master=null,muted=false;
function initAudio(){
  if(AC){if(AC.state==='suspended')AC.resume();return;}
  try{AC=new (window.AudioContext||window.webkitAudioContext)();
    master=AC.createGain();master.gain.value=.16;master.connect(AC.destination);}catch(e){}
}
function tone(f,dur,type,vol,slide,delay){
  if(!AC||muted)return;type=type||'square';vol=vol==null?1:vol;delay=delay||0;
  const t=AC.currentTime+delay,o=AC.createOscillator(),g=AC.createGain();
  o.type=type;o.frequency.setValueAtTime(f,t);
  if(slide)o.frequency.exponentialRampToValueAtTime(Math.max(30,f+slide),t+dur);
  g.gain.setValueAtTime(vol*.5,t);g.gain.exponentialRampToValueAtTime(.001,t+dur);
  o.connect(g);g.connect(master);o.start(t);o.stop(t+dur+.03);
}
function noiseHit(dur,vol,delay){
  if(!AC||muted)return;delay=delay||0;
  const t=AC.currentTime+delay,len=Math.max(1,(AC.sampleRate*dur)|0),buf=AC.createBuffer(1,len,AC.sampleRate),d=buf.getChannelData(0);
  for(let i=0;i<len;i++)d[i]=(Math.random()*2-1)*(1-i/len);
  const s=AC.createBufferSource(),g=AC.createGain(),f=AC.createBiquadFilter();
  f.type='lowpass';f.frequency.value=900;s.buffer=buf;g.gain.value=vol*.6;
  s.connect(f);f.connect(g);g.connect(master);s.start(t);
}
const sfx={
  move:()=>tone(340,.045,'square',.5),
  rotate:()=>tone(480,.06,'square',.6,120),
  soft:()=>tone(210,.035,'square',.3),
  hard:()=>{noiseHit(.14,.9);tone(150,.12,'sine',1,-110);},
  lock:()=>tone(180,.07,'triangle',.9,-60),
  hold:()=>tone(600,.06,'square',.5,200),
  clear:n=>{const N=[523,659,784,1046,1318].slice(0,n+1);N.forEach((f,i)=>tone(f,.11,'square',.7,0,i*.055));if(n>=4)noiseHit(.3,.7,.1);},
  tspin:()=>{[784,988,1175].forEach((f,i)=>tone(f,.1,'sawtooth',.4,0,i*.05));},
  level:()=>tone(300,.32,'sawtooth',.5,600),
  over:()=>{[392,311,233,155].forEach((f,i)=>tone(f,.24,'triangle',.8,0,i*.17));}
};

let grid,cur,holdType=null,holdUsed=false,bag=[],queue=[];
let score=0,lines=0,level=1,combo=-1,maxCombo=0,lastDiff=false,playTime=0;
let state='menu',clearing=null,overAt=0;
let particles=[],popups=[],shake=0,borderPulse=0;
let gravAcc=0,lockTimer=0,lockResets=0,lastRot=false;

const soft={on:false,acc:0,lastTick:0};
let best={score:0,lines:0,level:0};
try{const s=localStorage.getItem('neonGridBest');if(s)best=Object.assign(best,JSON.parse(s));}catch(e){}
function saveBest(){try{localStorage.setItem('neonGridBest',JSON.stringify(best));}catch(e){}}

/* ================= pieces ================= */
function refillBag(){const b='IJLOSTZ'.split('');for(let i=b.length-1;i>0;i--){const j=(Math.random()*(i+1))|0;[b[i],b[j]]=[b[j],b[i]];}bag.push(...b);}
function nextType(){refillBag();return bag.shift();}
function ensureQueue(){while(queue.length<5)queue.push(nextType());}
function rotMat(m,dir){const n=m.length,res=Array.from({length:n},()=>Array(n).fill(0));
  for(let r=0;r<n;r++)for(let c=0;c<n;c++){if(!m[r][c])continue;
    if(dir>0)res[c][n-1-r]=1;else res[n-1-c][r]=1;}return res;}
function collides(mat,x,y){
  for(let r=0;r<mat.length;r++)for(let c=0;c<mat[r].length;c++){if(!mat[r][c])continue;
    const nx=x+c,ny=y+r;
    if(nx<0||nx>=COLS||ny>=ROWS)return true;
    if(ny>=0&&grid[ny][nx])return true;}
  return false;}
function cellsOf(mat,x,y){const out=[];for(let r=0;r<mat.length;r++)for(let c=0;c<mat[r].length;c++)if(mat[r][c])out.push([x+c,y+r]);return out;}
function spawn(type){
  cur={type,mat:SHAPES[type].map(r=>r.slice()),x:SPAWN_X[type],y:-1,rot:0};
  if(collides(cur.mat,cur.x,cur.y)){gameOver();return;}
  gravAcc=0;lockTimer=0;lockResets=0;lastRot=false;
}
function move(dx,dy){if(!collides(cur.mat,cur.x+dx,cur.y+dy)){cur.x+=dx;cur.y+=dy;return true;}return false;}
function onSuccessfulMove(){if(collides(cur.mat,cur.x,cur.y+1)&&lockResets<15){lockTimer=0;lockResets++;}}
function tryMove(d){if(move(d,0)){sfx.move();onSuccessfulMove();}}
function rotate(dir){
  if(state!=='playing'||!cur)return;
  const m=rotMat(cur.mat,dir),from=cur.rot,to=(cur.rot+dir+4)%4;
  const kicks=(cur.type==='I'?KICKS_I:KICKS)[from+''+to]||[[0,0]];
  for(const k of kicks){const nx=cur.x+k[0],ny=cur.y-k[1];
    if(!collides(m,nx,ny)){cur.mat=m;cur.x=nx;cur.y=ny;cur.rot=to;lastRot=true;sfx.rotate();onSuccessfulMove();return;}}
}
function tSpinCorners(){let n=0;const cs=[[0,0],[2,0],[0,2],[2,2]];
  for(const c of cs){const nx=cur.x+c[0],ny=cur.y+c[1];
    if(nx<0||nx>=COLS||ny>=ROWS||(ny>=0&&grid[ny][nx]))n++;}return n;}

/* ================= scoring / locking ================= */
function addPopup(text,color,size,y){popups.push({text,color,size:size||Math.min(30,cell*.9),x:COLS*cell/2,y:(y!=null?y:ROWS*cell*.4),life:1150,max:1150});}
function burst(x,y,color,n,pow){for(let i=0;i<n;i++)particles.push({x,y,vx:rnd(-pow,pow),vy:rnd(-pow*1.3,pow*.3),life:rnd(400,780),max:780,color,size:rnd(2,5)});}
function lockPiece(){
  let topOut=false;
  for(const [cx,cy] of cellsOf(cur.mat,cur.x,cur.y)){
    if(cy<0){topOut=true;continue;}grid[cy][cx]=cur.type;}
  sfx.lock();shake=Math.max(shake,2.5);
  if(topOut){gameOver();return;}
  const ts=cur.type==='T'&&lastRot&&tSpinCorners()>=3;
  const rows=[];for(let r=0;r<ROWS;r++){let f=true;for(let c=0;c<COLS;c++)if(!grid[r][c]){f=false;break;}if(f)rows.push(r);}
  combo=rows.length?combo+1:-1;maxCombo=Math.max(maxCombo,combo);
  if(rows.length){startClearing(rows,ts);}
  else{
    if(ts){const p=400*level;score+=p;addPopup('T-SPIN +'+p,'#c16bff');sfx.tspin();}
    afterLock();
  }
  updateHUD();
}
function startClearing(rows,ts){
  const n=rows.length;let pts,label,color;
  if(ts){pts=[0,800,1200,1600][n]*level;label='T-SPIN '+['','SINGLE','DOUBLE','TRIPLE'][n];color='#c16bff';}
  else{pts=[0,100,300,500,800][n]*level;label=['','SINGLE','DOUBLE','TRIPLE','TETRIS!'][n];
    color=n===4?'#ffb84d':'#dfeaff';}
  const diff=n===4||(ts&&n>0);
  if(diff&&lastDiff){pts=Math.floor(pts*1.5);addPopup('B2B ×1.5','#45d8ff',Math.min(17,cell*.6),ROWS*cell*.4+30);}
  lastDiff=diff;
  if(combo>=1){const cb=50*combo*level;pts+=cb;addPopup('COMBO ×'+combo+' +'+cb,'#ffd23f',Math.min(16,cell*.58),ROWS*cell*.4+(diff?54:30));}
  score+=pts;addPopup(label,color,n===4?Math.min(40,cell*1.3):null);
  sfx.clear(n);if(ts)sfx.tspin();
  shake=Math.max(shake,n===4?11:3+n*1.5);borderPulse=n>=3?1:borderPulse;
  for(const r of rows)for(let c=0;c<COLS;c++){const col=COLORS[grid[r][c]]||'#fff';burst(c*cell+cell/2,r*cell+cell/2,col,3,150);}
  lines+=n;
  const nl=Math.floor(lines/10)+1;
  if(nl>level){level=nl;borderPulse=1;addPopup('LEVEL '+level,'#45d8ff',Math.min(26,cell*.9),ROWS*cell*.5);sfx.level();}
  state='clearing';clearing={rows,t:0};updateHUD();
}
function collapse(){
  const rs=clearing.rows.slice().sort((a,b)=>b-a);
  for(const r of rs){grid.splice(r,1);grid.unshift(Array(COLS).fill(0));}
}
function afterLock(){
  holdUsed=false;drawHold();
  const t=queue.shift();ensureQueue();drawNext();spawn(t);
}
function doHold(){
  if(state!=='playing'||holdUsed||!cur)return;
  sfx.hold();const t=cur.type;
  if(holdType==null){holdType=t;const nt=queue.shift();ensureQueue();drawNext();spawn(nt);}
  else{const h=holdType;holdType=t;spawn(h);}
  holdUsed=true;drawHold();updateHUD();
}
function hardDrop(){
  if(state!=='playing'||!cur)return;
  let d=0;while(!collides(cur.mat,cur.x,cur.y+1)){cur.y++;d++;}
  if(d>0){score+=2*d;}
  const lows={};for(const [cx,cy] of cellsOf(cur.mat,cur.x,cur.y))if(lows[cx]==null||cy>lows[cx])lows[cx]=cy;
  for(const cx in lows)burst((cx+.5)*cell,(lows[cx]+1)*cell,'#9fd8ff',2,70);
  shake=Math.max(shake,6);sfx.hard();updateHUD();lockPiece();
}

/* ================= flow ================= */
function resetGame(){
  grid=Array.from({length:ROWS},()=>Array(COLS).fill(0));
  bag=[];queue=[];ensureQueue();
  holdType=null;holdUsed=false;score=0;lines=0;level=1;combo=-1;maxCombo=0;lastDiff=false;playTime=0;
  particles=[];popups=[];shake=0;borderPulse=0;clearing=null;cur=null;
  gravAcc=0;lockTimer=0;soft.acc=0;
  const t=queue.shift();ensureQueue();spawn(t);
  drawNext();drawHold();updateHUD();
}
function startGame(){resetGame();state='playing';$('#startOv').classList.add('hide');$('#overOv').classList.add('hide');$('#pauseOv').classList.add('hide');}
function restart(){if(state==='menu'||(state==='over'&&performance.now()-overAt<450))return;startGame();}
function togglePause(force){
  if(state==='playing'&&(force!==false)){state='paused';$('#pauseOv').classList.remove('hide');}
  else if(state==='paused'){state='playing';$('#pauseOv').classList.add('hide');}
}
function gameOver(){
  state='over';overAt=performance.now();sfx.over();shake=12;
  let rec=false;
  if(score>best.score){best.score=score;rec=true;}
  if(lines>best.lines)best.lines=lines;if(level>best.level)best.level=level;
  saveBest();
  $('#newBest').hidden=!rec;
  $('#stScore').textContent=fmt(score);$('#stLines').textContent=fmt(lines);
  $('#stLevel').textContent=level;$('#stCombo').textContent='×'+maxCombo;
  playTime=Math.round(playTime/1000);
  $('#stTime').textContent=(playTime/60|0)+':'+String(playTime%60).padStart(2,'0');
  updateHUD();refreshBestUI();
  setTimeout(()=>{if(state==='over')$('#overOv').classList.remove('hide');},520);
}

/* ================= HUD ================= */
function refreshBestUI(){$('#best').textContent=fmt(best.score);}
function updateHUD(){
  $('#score').textContent=fmt(score);$('#hsc').textContent=fmt(score);
  $('#level').textContent=level;$('#lines').textContent=fmt(lines);
  const hi=Math.max(best.score,score);$('#hi').textContent=fmt(hi);
  $('#hiWrap').classList.toggle('hot',best.score>0&&score>best.score);
  const c=$('#combo');
  if(combo>=1){c.textContent='COMBO ×'+combo;c.classList.add('on');}
  else{c.textContent='COMBO ×0';c.classList.remove('on');}
}

/* ================= drawing ================= */
function drawBlock(g,x,y,s,c,glow){
  const p=Math.max(1,s*.05);
  if(glow){g.shadowColor=c;g.shadowBlur=s*.45;}
  g.fillStyle=c;g.fillRect(x+p,y+p,s-2*p,s-2*p);
  g.shadowBlur=0;
  const e=Math.max(2,s*.15);
  g.fillStyle='rgba(255,255,255,.32)';g.fillRect(x+p,y+p,s-2*p,e);
  g.fillStyle='rgba(255,255,255,.10)';g.fillRect(x+p,y+p,e,s-2*p);
  g.fillStyle='rgba(0,0,0,.30)';g.fillRect(x+p,y+s-p-e,s-2*p,e);
}
function drawGhostCell(g,x,y,s,c){
  g.globalAlpha=.5;g.strokeStyle=c;g.lineWidth=1.6;g.strokeRect(x+2.5,y+2.5,s-5,s-5);
  g.globalAlpha=.09;g.fillStyle=c;g.fillRect(x+2,y+2,s-4,s-4);g.globalAlpha=1;
}
function drawPreview(g,type,cx,cy,s,dim){
  const m=SHAPES[type];let x0=9,x1=-1,y0=9,y1=-1;
  for(let r=0;r<m.length;r++)for(let c=0;c<m[r].length;c++)if(m[r][c]){x0=Math.min(x0,c);x1=Math.max(x1,c);y0=Math.min(y0,r);y1=Math.max(y1,r);}
  const w=(x1-x0+1)*s,h=(y1-y0+1)*s,ox=cx-w/2,oy=cy-h/2;
  if(dim)g.globalAlpha=.35;
  for(let r=y0;r<=y1;r++)for(let c=x0;c<=x1;c++)if(m[r][c])drawBlock(g,ox+(c-x0)*s,oy+(r-y0)*s,s,COLORS[type],false);
  g.globalAlpha=1;
}
function drawHold(){
  const w=holdCv.width/dpr,h=holdCv.height/dpr;
  hctx.setTransform(dpr,0,0,dpr,0,0);hctx.clearRect(0,0,w,h);
  if(holdType)drawPreview(hctx,holdType,w/2,h/2,Math.min(cell,(w-14)/4),holdUsed);
}
function drawNext(){
  const w=nextCv.width/dpr;
  nctx.setTransform(dpr,0,0,dpr,0,0);nctx.clearRect(0,0,w,nextCv.height/dpr);
  for(let i=0;i<3&&i<queue.length;i++){
    drawPreview(nctx,queue[i],w/2,nextSlotH*i+nextSlotH/2,Math.min(cell,(w-14)/4)*(i===0?1:.86),false);}
}
function renderBoard(){
  const W=COLS*cell,H=ROWS*cell;
  bctx.setTransform(dpr,0,0,dpr,0,0);
  if(shake>0)bctx.translate((Math.random()*2-1)*shake,(Math.random()*2-1)*shake);
  bctx.clearRect(-20,-20,W+40,H+40);
  bctx.fillStyle='rgba(5,11,20,.6)';bctx.fillRect(0,0,W,H);
  for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++){
    if((r+c)%2){bctx.fillStyle='rgba(148,197,255,.022)';bctx.fillRect(c*cell,r*cell,cell,cell);}}
  bctx.strokeStyle='rgba(148,197,255,.07)';bctx.lineWidth=1;bctx.beginPath();
  for(let c=1;c<COLS;c++){bctx.moveTo(c*cell+.5,0);bctx.lineTo(c*cell+.5,H);}
  for(let r=1;r<ROWS;r++){bctx.moveTo(0,r*cell+.5);bctx.lineTo(W,r*cell+.5);}
  bctx.stroke();
  // locked
  for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++)if(grid[r][c])drawBlock(bctx,c*cell,r*cell,cell,COLORS[grid[r][c]],false);
  // clearing flash
  if(clearing){const p=Math.min(1,clearing.t/CLEAR_MS),a=Math.sin(p*Math.PI);
    for(const r of clearing.rows){
      bctx.fillStyle='rgba(255,255,255,'+(.85*a)+')';bctx.fillRect(0,r*cell,W,cell);
      bctx.fillStyle='rgba(69,216,255,'+(.35*a*(1-p))+')';bctx.fillRect(0,r*cell,W*p,cell);}}
  // ghost + active
  if(cur&&state==='playing'){
    let gy=cur.y;while(!collides(cur.mat,cur.x,gy+1))gy++;
    if(gy>cur.y)for(const [cx,cy] of cellsOf(cur.mat,cur.x,gy))drawGhostCell(bctx,cx*cell,cy*cell,cell,COLORS[cur.type]);
    for(const [cx,cy] of cellsOf(cur.mat,cur.x,cur.y))if(cy>=0)drawBlock(bctx,cx*cell,cy*cell,cell,COLORS[cur.type],true);
  }
  // particles (additive)
  bctx.globalCompositeOperation='lighter';
  for(const p of particles){const a=clamp(p.life/p.max,0,1);bctx.globalAlpha=a;bctx.fillStyle=p.color;
    bctx.fillRect(p.x-p.size/2,p.y-p.size/2,p.size,p.size);}
  bctx.globalAlpha=1;bctx.globalCompositeOperation='source-over';
  // popups
  for(const p of popups){const a=clamp(p.life/p.max,0,1);bctx.globalAlpha=a;
    bctx.font='900 '+p.size+'px Impact,"Arial Black",sans-serif';bctx.textAlign='center';bctx.textBaseline='middle';
    bctx.lineWidth=5;bctx.strokeStyle='rgba(4,9,16,.85)';bctx.strokeText(p.text,p.x,p.y);
    bctx.shadowColor=p.color;bctx.shadowBlur=16;bctx.fillStyle=p.color;bctx.fillText(p.text,p.x,p.y);
    bctx.shadowBlur=0;bctx.globalAlpha=1;}
  // danger glow
  let top=-1;outer:for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++)if(grid[r][c]){top=r;break outer;}
  if(top>=0&&top<=5){const a=(.10+.09*Math.sin(performance.now()*.012))*(1-top/6);
    const gr=bctx.createLinearGradient(0,0,0,cell*3.2);gr.addColorStop(0,'rgba(255,90,95,'+a+')');gr.addColorStop(1,'rgba(255,90,95,0)');
    bctx.fillStyle=gr;bctx.fillRect(0,0,W,cell*3.2);}
  // border pulse
  if(borderPulse>0){bctx.strokeStyle='rgba(69,216,255,'+(borderPulse*.85)+')';bctx.lineWidth=4;
    bctx.strokeRect(2,2,W-4,H-4);}
}

/* ================= ambient background ================= */
let flakes=[],stars=[];
function layoutBg(){
  const W=innerWidth,H=innerHeight;bgCv.width=W*dpr;bgCv.height=H*dpr;
  bgCv.style.width=W+'px';bgCv.style.height=H+'px';gctx.setTransform(dpr,0,0,dpr,0,0);
  if(!flakes.length){const T='IJLOSTZ'.split('');
    for(let i=0;i<13;i++)flakes.push({x:rnd(0,W),y:rnd(0,H),s:rnd(9,26),rot:rnd(0,6.3),vr:rnd(-.15,.15),vy:rnd(7,22),type:T[(Math.random()*7)|0],a:rnd(.04,.1)});
    for(let i=0;i<85;i++)stars.push({x:rnd(0,W),y:rnd(0,H),r:rnd(.6,1.9),tw:rnd(.4,2.2),o:rnd(0,6.3)});}
}
function renderBg(t,dt){
  const W=innerWidth,H=innerHeight;
  gctx.clearRect(0,0,W,H);
  for(const s of stars){const a=.12+.3*Math.abs(Math.sin(t*.001*s.tw+s.o));
    gctx.globalAlpha=a;gctx.fillStyle=s.r>1.4?'#9fd8ff':'#dfeaff';gctx.fillRect(s.x,s.y,s.r,s.r);}
  gctx.globalAlpha=1;
  for(const f of flakes){f.y+=f.vy*dt/1000;f.rot+=f.vr*dt/1000;
    if(f.y>H+f.s*4){f.y=-f.s*4;f.x=rnd(0,W);}
    gctx.save();gctx.translate(f.x,f.y);gctx.rotate(f.rot);
    const m=SHAPES[f.type],off=(m.length-1)/2*f.s;
    gctx.globalAlpha=f.a;gctx.strokeStyle=COLORS[f.type];gctx.lineWidth=1.4;
    for(let r=0;r<m.length;r++)for(let c=0;c<m[r].length;c++)if(m[r][c])gctx.strokeRect((c-off)*f.s+2,(r-off)*f.s+2,f.s-4,f.s-4);
    gctx.restore();}
  gctx.globalAlpha=1;
}

/* ================= update loop ================= */
function gravMs(l){return Math.max(50,1000*Math.pow(.8-(l-1)*.007,l-1));}
function step(dt,t){
  if(state==='playing'){
    playTime+=dt;
    // DAS
    if(lastDir!==0){
      while(t>=dasNext){tryMove(lastDir);dasNext+=ARR;}
    }
    // soft drop
    if(soft.on&&cur){
      soft.acc+=dt;
      while(soft.acc>=32){soft.acc-=32;
        if(move(0,1)){score++;if(t-soft.lastTick>80){sfx.soft();soft.lastTick=t;}updateHUD();}else break;}
    }
    // gravity
    if(cur&&!soft.on){gravAcc+=dt;const iv=gravMs(level);
      while(gravAcc>=iv){gravAcc-=iv;if(!move(0,1))break;}}
    // lock delay
    if(cur&&collides(cur.mat,cur.x,cur.y+1)){lockTimer+=dt;if(lockTimer>=LOCK_MS)lockPiece();}
    else if(cur)lockTimer=0;
  }else if(state==='clearing'&&clearing){
    playTime+=dt;clearing.t+=dt;
    if(clearing.t>=CLEAR_MS){collapse();clearing=null;state='playing';afterLock();}
  }
  // fx
  for(let i=particles.length-1;i>=0;i--){const p=particles[i];p.life-=dt;
    p.vy+=380*dt/1000;p.x+=p.vx*dt/1000;p.y+=p.vy*dt/1000;if(p.life<=0)particles.splice(i,1);}
  for(let i=popups.length-1;i>=0;i--){const p=popups[i];p.life-=dt;p.y-=26*dt/1000;if(p.life<=0)popups.splice(i,1);}
  shake=Math.max(0,shake-dt*.02);borderPulse=Math.max(0,borderPulse-dt/700);
}
