    roundRect(ox-8,oy-8,cell*N+16,cell*N+16,14);ctx.fill();
    // checker cells
    for(r=0;r<N;r++)for(c=0;c<N;c++){
      ctx.fillStyle=(r+c)%2?'rgba(255,255,255,.035)':'rgba(255,255,255,.07)';
      roundRect(ox+c*cell+1,oy+r*cell+1,cell-2,cell-2,8);ctx.fill();
    }
    // selection
    if(sel){
      ctx.strokeStyle='rgba(255,255,255,.9)';ctx.lineWidth=3;
      roundRect(ox+sel.c*cell+2,oy+sel.r*cell+2,cell-4,cell-4,10);ctx.stroke();
    }
    for(r=0;r<N;r++)for(c=0;c<N;c++){
      cl=board[r][c];if(!cl)continue;
      var isSel=sel&&sel.r===r&&sel.c===c;
      drawGem(cl.x,cl.y,cl.scale*(isSel?1.08:1),cl.color,isSel);
    }
    ctx.restore();
  }
  requestAnimationFrame(frame);
}
frame();
