/* ---------- Canvas / scaling ---------- */
const canvas = document.getElementById('game');
const ctx = canvas.getContext('2d');
const W = 960, H = 540; // logical arena
let scale = 1, ox = 0, oy = 0;
function resize(){
  const dpr = window.devicePixelRatio || 1;
  canvas.width  = Math.floor(window.innerWidth * dpr);
  canvas.height = Math.floor(window.innerHeight * dpr);
  canvas.style.width = window.innerWidth + 'px';
  canvas.style.height = window.innerHeight + 'px';
  scale = Math.min(canvas.width / W, canvas.height / H);
  ox = (canvas.width - W * scale) / 2;
  oy = (canvas.height - H * scale) / 2;
}
window.addEventListener('resize', resize);

/* ---------- Audio (Web Audio, starts on first input) ---------- */
let AC = null;
function audio(){
  if (!AC){ try { AC = new (window.AudioContext || window.webkitAudioContext)(); } catch(e){} }
  if (AC && AC.state === 'suspended') AC.resume();
  return AC;
}
function beep(freq, dur, type, vol, slide){
  const ac = audio(); if (!ac) return;
  try{
    const o = ac.createOscillator(), g = ac.createGain();
    o.type = type || 'square';
    o.frequency.setValueAtTime(freq, ac.currentTime);
    if (slide) o.frequency.exponentialRampToValueAtTime(Math.max(30, freq + slide), ac.currentTime + dur);
    g.gain.setValueAtTime(vol || .15, ac.currentTime);
    g.gain.exponentialRampToValueAtTime(.0001, ac.currentTime + dur);
    o.connect(g); g.connect(ac.destination);
    o.start(); o.stop(ac.currentTime + dur + .02);
  }catch(e){}
}
const sfx = {
  shoot: () => beep(880, .09, 'square', .07, -500),
  hit:   () => { beep(160, .18, 'sawtooth', .18, -100); },
  wall:  () => beep(220, .06, 'triangle', .06, -80),
  orb:   () => { beep(660, .1, 'sine', .14, 300); setTimeout(()=>beep(990,.12,'sine',.12,200),70); },
  count: () => beep(440, .12, 'square', .12),
  go:    () => beep(880, .3, 'square', .15),
  win:   () => { [523,659,784,1047].forEach((f,i)=>setTimeout(()=>beep(f,.22,'square',.13), i*140)); }
};

/* ---------- Input ---------- */

window.addEventListener('keydown', e => {
  audio();
  if (['ArrowUp','ArrowDown','ArrowLeft','ArrowRight',' ','Spacebar'].includes(e.key) || e.code === 'Space') e.preventDefault();
  const k = e.code;

  return { x, y: H/2, r: 16, hp: 100, score: 0, faceX: x < W/2 ? 1 : -1, faceY: 0,
           cool: 0, flash: 0, color, glow };

  state = STATE.COUNT; countdown = 3.4; lastCountBeep = -1;
}

function onKeyPress(k){
  if (state === STATE.START && (k === 'Enter' || k === 'NumpadEnter')){ startGame(); return; }
  if ((state === STATE.OVER || state === STATE.PLAY) && k === 'KeyR'){ resetRound(); return; }
  if (state === STATE.OVER && (k === 'Enter' || k === 'NumpadEnter')){ resetRound(); return; }
}

/* ---------- Helpers ---------- */
function circleRect(cx, cy, cr, r){
  const nx = Math.max(r.x, Math.min(cx, r.x + r.w));
  const ny = Math.max(r.y, Math.min(cy, r.y + r.h));
  const dx = cx - nx, dy = cy - ny;
  return dx*dx + dy*dy < cr*cr;
}
function resolveCircleRect(p, r){
  const nx = Math.max(r.x, Math.min(p.x, r.x + r.w));
  const ny = Math.max(r.y, Math.min(p.y, r.y + r.h));
  let dx = p.x - nx, dy = p.y - ny;
  let d2 = dx*dx + dy*dy;
  if (d2 < p.r * p.r){
    if (d2 === 0){ // center inside rect: push out along smallest axis
      const l = p.x - r.x, rt = r.x + r.w - p.x, t = p.y - r.y, b = r.y + r.h - p.y;
      const m = Math.min(l, rt, t, b);
      if (m === l) p.x = r.x - p.r; else if (m === rt) p.x = r.x + r.w + p.r;
      else if (m === t) p.y = r.y - p.r; else p.y = r.y + r.h + p.r;
    } else {
      const d = Math.sqrt(d2);
      p.x = nx + dx / d * p.r;
      p.y = ny + dy / d * p.r;
    }
  }
}
function spawnParticles(x, y, color, n, spd){
  for (let i = 0; i < n; i++){
    const a = Math.random() * Math.PI * 2, s = (0.3 + Math.random()) * spd;
    particles.push({ x, y, vx: Math.cos(a)*s, vy: Math.sin(a)*s, life: .5 + Math.random()*.4, max:.9, color });
  }
}
function freeSpot(){
  for (let i = 0; i < 40; i++){
    const x = 50 + Math.random() * (W - 100);
    const y = 60 + Math.random() * (H - 120);
    let ok = true;
    for (const w of walls){ if (x > w.x-30 && x < w.x+w.w+30 && y > w.y-30 && y < w.y+w.h+30) ok = false; }
    if (ok && Math.hypot(x-p1.x, y-p1.y) > 70 && Math.hypot(x-p2.x, y-p2.y) > 70) return {x,y};
  }
  return null;
}

/* ---------- Update ---------- */
function movePlayer(p, up, down, left, right, dt){
  const spd = 260 * dt;
  let mx = (right?1:0) - (left?1:0);
  let my = (down?1:0) - (up?1:0);
  if (mx || my){
    const l = Math.hypot(mx,my);
    p.x += mx/l * spd; p.y += my/l * spd;
    p.faceX = mx/l; p.faceY = my/l;
  }
  p.x = Math.max(p.r + 4, Math.min(W - p.r - 4, p.x));
  p.y = Math.max(p.r + 4, Math.min(H - p.r - 4, p.y));
  for (const w of walls) resolveCircleRect(p, w);
}

function shoot(p, dt){
  if (p.cool > 0) return;
  p.cool = .38;
  const a = Math.atan2(p.faceY, p.faceX);
  bullets.push({ x: p.x + Math.cos(a)*(p.r+6), y: p.y + Math.sin(a)*(p.r+6),
                 vx: Math.cos(a)*520, vy: Math.sin(a)*520, r: 5, life: 1.4, owner: p, color: p.color });
  spawnParticles(p.x + Math.cos(a)*p.r, p.y + Math.sin(a)*p.r, p.color, 4, 60);
  sfx.shoot();
}

function endRound(w){
  winner = w;
  state = STATE.OVER;
  sfx.win();
  const t = document.getElementById('winTitle');
  t.textContent = (w === p1 ? 'PLAYER 1' : 'PLAYER 2') + ' WINS!';
  t.style.color = w.color;
  t.style.textShadow = '0 0 20px ' + w.color;
  document.getElementById('winSub').textContent =
    'Final — P1 HP ' + Math.max(0,Math.round(p1.hp)) + ' / P2 HP ' + Math.max(0,Math.round(p2.hp)) +
    '   •   Orbs: ' + p1.score + ' vs ' + p2.score;
  overScreen.classList.remove('hidden');
}

function update(dt){
  if (state === STATE.COUNT){
    countdown -= dt;
    const c = Math.ceil(countdown);
    if (c !== lastCountBeep && c > 0 && c <= 3){ lastCountBeep = c; sfx.count(); }
    if (countdown <= 0){ state = STATE.PLAY; sfx.go(); }
    return;
  }
  if (state !== STATE.PLAY) return;

  movePlayer(p1, keys['KeyW'], keys['KeyS'], keys['KeyA'], keys['KeyD'], dt);
  movePlayer(p2, keys['ArrowUp'], keys['ArrowDown'], keys['ArrowLeft'], keys['ArrowRight'], dt);

  p1.cool -= dt; p2.cool -= dt; p1.flash = Math.max(0, p1.flash - dt); p2.flash = Math.max(0, p2.flash - dt);
  if (keys['KeyF'] || keys['Space']) shoot(p1, dt);
  if (keys['KeyL'] || keys['Enter'] || keys['NumpadEnter']) shoot(p2, dt);

  // bullets
  for (let i = bullets.length - 1; i >= 0; i--){
    const b = bullets[i];
    b.x += b.vx * dt; b.y += b.vy * dt; b.life -= dt;
    let dead = b.life <= 0 || b.x < -20 || b.x > W+20 || b.y < -20 || b.y > H+20;
    if (!dead) for (const w of walls){ if (circleRect(b.x, b.y, b.r, w)){ dead = true; sfx.wall(); spawnParticles(b.x,b.y,'#9fb4ff',5,80); break; } }
    if (!dead){
      const target = b.owner === p1 ? p2 : p1;
      if (Math.hypot(b.x - target.x, b.y - target.y) < b.r + target.r){
        dead = true;
        target.hp -= 12;
        target.flash = .35;
        const a = Math.atan2(b.vy, b.vx);
        target.x += Math.cos(a)*10; target.y += Math.sin(a)*10;
        spawnParticles(b.x, b.y, target.color, 14, 160);
        sfx.hit();
        if (target.hp <= 0){ target.hp = 0; endRound(b.owner); }
      }
    }
    if (dead) bullets.splice(i,1);
  }

  // orbs
  orbTimer -= dt;
  if (orbTimer <= 0 && orbs.length < 3){
    const s = freeSpot();
    if (s) orbs.push({ x:s.x, y:s.y, r:10, t:Math.random()*6 });
    orbTimer = 3 + Math.random()*2;
  }
  for (let i = orbs.length - 1; i >= 0; i--){
    const o = orbs[i]; o.t += dt*4;
    for (const p of [p1,p2]){
      if (Math.hypot(o.x-p.x, o.y-p.y) < o.r + p.r){
        p.hp = Math.min(100, p.hp + 25);
        p.score++;
        spawnParticles(o.x,o.y,'#7dff9b',12,140);
        sfx.orb();
        orbs.splice(i,1);
        break;
      }
    }
  }

  // particles
  for (let i = particles.length - 1; i >= 0; i--){
    const pt = particles[i];
    pt.x += pt.vx*dt; pt.y += pt.vy*dt;
    pt.vx *= .94; pt.vy *= .94;
    pt.life -= dt;
    if (pt.life <= 0) particles.splice(i,1);
  }
}

/* ---------- Render ---------- */
function drawArena(){
  ctx.fillStyle = '#070a18';
  ctx.fillRect(0,0,W,H);
  // grid
  ctx.strokeStyle = 'rgba(63,242,255,.06)';

  for (let y = 0; y <= H; y += 48){ ctx.moveTo(0,y); ctx.lineTo(W,y); }
  ctx.stroke();
  // border
  ctx.strokeStyle = 'rgba(120,160,255,.5)';
  ctx.lineWidth = 3;
  ctx.strokeRect(2,2,W-4,H-4);
  // walls
  for (const w of walls){
    ctx.fillStyle = '#182040';
    ctx.fillRect(w.x, w.y, w.w, w.h);
    ctx.strokeStyle = 'rgba(130,170,255,.7)';
    ctx.lineWidth = 2;
    ctx.strokeRect(w.x, w.y, w.w, w.h);
  }
}

function drawPlayer(p){
  const f = p.flash > 0 ? p.flash / .35 : 0;
  ctx.save();
  // glow
  const g = ctx.createRadialGradient(p.x,p.y,4,p.x,p.y,p.r*2.6);
  g.addColorStop(0, p.glow + (.35 + f*.4) + ')');
  g.addColorStop(1, p.glow + '0)');
  ctx.fillStyle = g;
  ctx.beginPath(); ctx.arc(p.x,p.y,p.r*2.6,0,Math.PI*2); ctx.fill();
  // body
  ctx.fillStyle = f > 0 ? '#ffffff' : p.color;
  ctx.beginPath(); ctx.arc(p.x,p.y,p.r,0,Math.PI*2); ctx.fill();
  ctx.fillStyle = 'rgba(5,8,20,.75)';
  ctx.beginPath(); ctx.arc(p.x,p.y,p.r*.62,0,Math.PI*2); ctx.fill();
  // facing nub
  const a = Math.atan2(p.faceY, p.faceX);
  ctx.fillStyle = f > 0 ? '#fff' : p.color;
  ctx.beginPath();
  ctx.arc(p.x + Math.cos(a)*p.r*.78, p.y + Math.sin(a)*p.r*.78, 5.5, 0, Math.PI*2);
  ctx.fill();
  ctx.restore();
}

function bar(x, y, w, h, pct, color, alignRight){
  ctx.fillStyle = 'rgba(255,255,255,.1)';
  ctx.fillRect(x, y, w, h);
  const fw = Math.max(0, w * pct);
  ctx.fillStyle = color;
  if (alignRight) ctx.fillRect(x + w - fw, y, fw, h); else ctx.fillRect(x, y, fw, h);
  ctx.strokeStyle = 'rgba(255,255,255,.4)';
  ctx.lineWidth = 1.5;
  ctx.strokeRect(x, y, w, h);
}

}

function render(){
  ctx.setTransform(1,0,0,1,0,0);
  ctx.fillStyle = '#03040b';
  ctx.fillRect(0,0,canvas.width,canvas.height);
  ctx.setTransform(scale,0,0,scale,ox,oy);

  drawArena();

  // orbs
  for (const o of orbs){
    const pulse = 1 + Math.sin(o.t)*.2;
    const g = ctx.createRadialGradient(o.x,o.y,2,o.x,o.y,o.r*3*pulse);
    g.addColorStop(0,'rgba(125,255,155,.9)');
    g.addColorStop(1,'rgba(125,255,155,0)');
    ctx.fillStyle = g;
    ctx.beginPath(); ctx.arc(o.x,o.y,o.r*3*pulse,0,Math.PI*2); ctx.fill();
    ctx.fillStyle = '#7dff9b';
    ctx.beginPath(); ctx.arc(o.x,o.y,o.r*.85*pulse,0,Math.PI*2); ctx.fill();
  }

  // particles
  for (const pt of particles){
    ctx.globalAlpha = Math.max(0, pt.life / .9);
    ctx.fillStyle = pt.color;
    ctx.beginPath(); ctx.arc(pt.x, pt.y, 3, 0, Math.PI*2); ctx.fill();
  }
  ctx.globalAlpha = 1;

  // bullets
  for (const b of bullets){
    const g = ctx.createRadialGradient(b.x,b.y,1,b.x,b.y,b.r*3);
    g.addColorStop(0,'rgba(255,255,255,.9)');
    g.addColorStop(.4, b.color + 'aa');
    g.addColorStop(1,'rgba(0,0,0,0)');
    ctx.fillStyle = g;
    ctx.beginPath(); ctx.arc(b.x,b.y,b.r*3,0,Math.PI*2); ctx.fill();
    ctx.fillStyle = '#fff';
    ctx.beginPath(); ctx.arc(b.x,b.y,b.r*.7,0,Math.PI*2); ctx.fill();
  }

  if (state !== STATE.START){
    drawPlayer(p1); drawPlayer(p2);
    drawHUD();

    if (state === STATE.COUNT){
      const c = Math.ceil(countdown);
      ctx.textAlign = 'center'; ctx.textBaseline = 'middle';
      ctx.font = 'bold 90px Segoe UI, Arial';
      ctx.fillStyle = c <= 1 ? '#7dff9b' : '#eafcff';
      ctx.shadowColor = ctx.fillStyle; ctx.shadowBlur = 30;
      ctx.fillText(c > 0 ? c : 'GO!', W/2, H/2);
      ctx.shadowBlur = 0;
    }
  } else {
    // idle demo backdrop behind start overlay
  }
}

/* ---------- Loop ---------- */
let last = performance.now();
function loop(now){
  let dt = (now - last) / 1000;
  last = now;
  if (dt > .05) dt = .05;
  update(dt);
  render();
  requestAnimationFrame(loop);
}

resetRound();

/* localStorage: remember best orb count (wrapped in try/catch) */

  try {
    const best = JSON.parse(localStorage.getItem('neonDuelBest') || '0');
    if (p1.score + p2.score > best) localStorage.setItem('neonDuelBest', String(p1.score + p2.score));
  } catch(e){}
}
const _end = endRound;
endRound = function(w){ saveBest(); _end(w); };
