  ctx.beginPath();ctx.arc(hx+d[0]*cell*0.16-px*cell*0.18,hy+d[1]*cell*0.16-py*cell*0.18,cell*0.11,0,7);ctx.fill();

  /* HUD */
  ctx.textBaseline='middle';

}
