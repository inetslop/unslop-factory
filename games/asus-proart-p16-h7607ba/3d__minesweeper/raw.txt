const stage=document.getElementById('stage'), boardEl=document.getElementById('board'),
      overlay=document.getElementById('overlay'), startCard=document.getElementById('startCard'),
      endCard=document.getElementById('endCard'), ovMsg=document.getElementById('ovMsg'),
      ovText=document.getElementById('ovText');

let mines, opened, flagged, activeLayer=0, running=false, over=false;
let firstMove=true, flagsPlaced=0, openedCount=0;
let timerInt=null, elapsed=0, best=null;
try { const b=localStorage.getItem('m3d_best'); if(b) best=parseInt(b,10)||null; } catch(e){}

let rx=56, ry=-28;
function applyView(){ boardEl.style.transform='rotateX('+rx+'deg) rotateY('+ry+'deg)'; }

/* ---------- audio ---------- */
let AC=null;
function audio(){ if(!AC){ try{ AC=new (window.AudioContext||window.webkitAudioContext)(); }catch(e){} } if(AC&&AC.state==='suspended') AC.resume(); return AC; }
function tone(f,dur,type,vol,when){
  const ac=audio(); if(!ac) return;
  const t=ac.currentTime+(when||0), o=ac.createOscillator(), g=ac.createGain();
  o.type=type||'triangle'; o.frequency.value=f;
  g.gain.setValueAtTime(0,t); g.gain.linearRampToValueAtTime(vol||.15,t+.012);
  g.gain.exponentialRampToValueAtTime(.0001,t+dur);
  o.connect(g); g.connect(ac.destination); o.start(t); o.stop(t+dur+.05);
}
const sfxOpen=()=>tone(340+Math.random()*60,.12,'triangle',.12);
const sfxFlag=()=>tone(560,.1,'square',.08);
const sfxUnflag=()=>tone(380,.1,'square',.07);
const sfxBoom=()=>{ tone(90,.5,'sawtooth',.28); tone(55,.6,'square',.22); };
const sfxWin=()=>{ [523,659,784,1047].forEach((f,i)=>tone(f,.22,'triangle',.14,i*.12)); };

/* ---------- board construction (DOM once) ---------- */
const cells=[];
for(let z=0;z<L;z++){
  const layer=document.createElement('div');
  layer.className='layer';
  layer.style.transform='translate(-50%,-50%) translateZ('+((L-1-z)*LZ)+'px)';
  const plate=document.createElement('div'); plate.className='plate';
  const tag=document.createElement('div'); tag.className='tag'; tag.textContent='F'+(z+1);
  layer.appendChild(plate); layer.appendChild(tag);
  for(let y=0;y<H;y++)for(let x=0;x<W;x++){
    const c=document.createElement('div');
    c.className='cell'; c.dataset.idx=(x+W*(y+H*z));
    layer.appendChild(c); cells.push(c);
  }
  boardEl.appendChild(layer);
}
const layers=[...boardEl.querySelectorAll('.layer')];

/* ---------- logic ---------- */
function idx(x,y,z){ return x + W*(y + H*z); }
function neighbors(i){
  const x=i%W, y=Math.floor(i/W)%H, z=Math.floor(i/(W*H));
  const out=[];
  for(let dz=-1;dz<=1;dz++)for(let dy=-1;dy<=1;dy++)for(let dx=-1;dx<=1;dx++){
    if(!dx&&!dy&&!dz) continue;
    const nx=x+dx, ny=y+dy, nz=z+dz;
    if(nx<0||ny<0||nz<0||nx>=W||ny>=H||nz>=L) continue;
    out.push(idx(nx,ny,nz));
  }
  return out;
}
function countMines(i){ return neighbors(i).reduce((s,j)=>s+(mines[j]?1:0),0); }

function newGame(){
  mines=new Array(W*H*L).fill(false);
  opened=new Array(W*H*L).fill(false);
  flagged=new Array(W*H*L).fill(false);
  const pool=[...Array(W*H*L).keys()];
  for(let i=pool.length-1;i>0;i--){ const j=Math.floor(Math.random()*(i+1)); [pool[i],pool[j]]=[pool[j],pool[i]]; }
  for(let k=0;k<MINES;k++) mines[pool[k]]=true;
  flagsPlaced=0; openedCount=0; firstMove=true; running=true; over=false; elapsed=0; activeLayer=0;
  cells.forEach(c=>{ c.className='cell'; c.textContent=''; });
  updateFloor(); updateHUD();
  clearInterval(timerInt); timerInt=null;
  document.getElementById('timer').textContent='0s';
  overlay.classList.add('hidden');
  startCard.style.display='none'; endCard.style.display='none';
}

function startTimer(){
  if(timerInt) return;
  timerInt=setInterval(()=>{ elapsed++; document.getElementById('timer').textContent=elapsed+'s'; },1000);
}
function stopTimer(){ clearInterval(timerInt); timerInt=null; }

function paintCell(i){
  const c=cells[i];
  if(opened[i]){
    c.classList.add('open');
    if(mines[i]){ c.textContent='💣'; c.classList.add('mine'); }
    else { const n=countMines(i); c.textContent=n||''; if(n) c.classList.add('n'+n); }
  } else if(flagged[i]) c.textContent='🚩';
}

function doOpen(i){
  if(!running||over||flagged[i]||opened[i]) return;
  if(firstMove){
    firstMove=false; startTimer();
    if(mines[i]){ // never let the first click be a mine
      const others=[...Array(W*H*L).keys()].filter(j=>j!==i&&!mines[j]);
      mines[i]=false; mines[others[Math.floor(Math.random()*others.length)]]=true;
    }
  }
  const stack=[i];
  while(stack.length){
    const j=stack.pop();
    if(opened[j]||flagged[j]||mines[j]) continue;
    opened[j]=true; openedCount++; paintCell(j); sfxOpen();
    if(countMines(j)===0){
      for(const n of neighbors(j)) if(!opened[n]&&!flagged[n]&&!mines[n]) stack.push(n);
    } else {
      // auto-chord: all adjacent mines already flagged -> open the rest
      const nb=neighbors(j), f=nb.reduce((s,n)=>s+(flagged[n]?1:0),0);
      if(f===countMines(j)) for(const n of nb) if(!opened[n]&&!flagged[n]) stack.push(n);
    }
  }
  updateHUD();
  if(openedCount===W*H*L-MINES) win();
}

function toggleFlag(i){
  if(!running||over||opened[i]) return;
  flagged[i]=!flagged[i]; flagsPlaced+=flagged[i]?1:-1;
  paintCell(i); (flagged[i]?sfxFlag:sfxUnflag)(); updateHUD();
}

function lose(hit){
  over=true; stopTimer(); sfxBoom();
  cells.forEach((c,j)=>{ if(mines[j]&&!flagged[j]){ c.classList.add('open','mine'); c.textContent='💣'; }
    if(!mines[j]&&flagged[j]){ c.classList.add('wrong'); c.textContent='❌'; } });
  cells[hit].classList.remove('mine'); cells[hit].classList.add('boom');
  endCard.style.display='block'; startCard.style.display='none';
  ovMsg.textContent='💥 BOOM!';
  ovText.innerHTML='The floor gave way under you.<br>Survived <b>'+elapsed+'s</b>. Press <b>R</b> to try again.';
  overlay.classList.remove('hidden');
}

function win(){
  over=true; stopTimer(); sfxWin();
  let newBest=false;
  if(best===null||elapsed<best){ best=elapsed; newBest=true; try{ localStorage.setItem('m3d_best',String(best)); }catch(e){} }
  updateHUD();
  endCard.style.display='block'; startCard.style.display='none';
  ovMsg.textContent='🎉 CLEARED!';
  ovText.innerHTML='All 266 cells swept in <b>'+elapsed+'s</b>.'+(newBest?'<br>🏆 <b>New best time!</b>':'');
  overlay.classList.remove('hidden');
}

/* ---------- HUD / floor ---------- */
function updateHUD(){
  document.getElementById('mineCount').textContent=Math.max(0,MINES-flagsPlaced);
  document.getElementById('bestTime').textContent=best===null?'—':best+'s';
}
function updateFloor(){
  layers.forEach((l,z)=>{ l.classList.toggle('active',z===activeLayer); l.classList.toggle('dim',z!==activeLayer); });
  document.getElementById('floorNum').textContent=activeLayer+1;
}

/* ---------- input: rotate + click ---------- */
let dragging=false, moved=false, sx=0, sy=0, srx=0, sry=0;
stage.addEventListener('mousedown',e=>{
  if(overlay.classList.contains('hidden')===false && e.target.closest('.card')) return;
  if(e.button!==0) return;
  dragging=true; moved=false; sx=e.clientX; sy=e.clientY; srx=rx; sry=ry;
});
window.addEventListener('mousemove',e=>{
  if(!dragging) return;
  const dx=e.clientX-sx, dy=e.clientY-sy;
  if(Math.abs(dx)+Math.abs(dy)>5) moved=true;
  ry=sry+dx*0.35;
  rx=Math.max(12,Math.min(86,srx-dy*0.35));
  applyView();
});
window.addEventListener('mouseup',e=>{
  if(!dragging) return; dragging=false;
  if(moved||!running||over) return;
  const t=e.target.closest('.cell');
  if(t && +t.dataset.idx!==undefined){
    const i=+t.dataset.idx, z=Math.floor(i/(W*H));
    if(z===activeLayer) doOpen(i);
  }
});
stage.addEventListener('contextmenu',e=>{
  e.preventDefault();
  const t=e.target.closest('.cell');
  if(t && running && !over){
    const i=+t.dataset.idx, z=Math.floor(i/(W*H));
    if(z===activeLayer) toggleFlag(i);
  }
});
stage.addEventListener('wheel',e=>{
  e.preventDefault();
  activeLayer=Math.max(0,Math.min(L-1,activeLayer+(e.deltaY>0?1:-1)));
  updateFloor();
},{passive:false});

/* touch: drag = rotate, tap = open */
let tMoved=false;
stage.addEventListener('touchstart',e=>{ if(e.touches.length===1){ tMoved=false; sx=e.touches[0].clientX; sy=e.touches[0].clientY; srx=rx; sry=ry; } },{passive:true});
stage.addEventListener('touchmove',e=>{
  if(e.touches.length!==1) return;
  const dx=e.touches[0].clientX-sx, dy=e.touches[0].clientY-sy;
  if(Math.abs(dx)+Math.abs(dy)>8) tMoved=true;
  ry=sry+dx*0.35; rx=Math.max(12,Math.min(86,srx-dy*0.35)); applyView();
},{passive:true});
stage.addEventListener('touchend',e=>{
  if(tMoved||!running||over) return;
  const t=document.elementFromPoint(e.changedTouches[0].clientX,e.changedTouches[0].clientY);
  const c=t&&t.closest?t.closest('.cell'):null;
  if(c){ const i=+c.dataset.idx, z=Math.floor(i/(W*H)); if(z===activeLayer) doOpen(i); }
});

/* ---------- keyboard ---------- */
window.addEventListener('keydown',e=>{
  audio();
  const k=e.key;
  if(k===' '||k==='Enter'){ e.preventDefault(); if(!running&&!over&&startCard.style.display!=='none') newGame(); else if(over) newGame(); return; }
  if(k==='r'||k==='R'){ if(running||over) newGame(); return; }
  if(!running&&startCard.style.display!=='none') return;
  if(k==='q'||k==='Q'||k==='PageUp'){ e.preventDefault(); activeLayer=Math.max(0,activeLayer-1); updateFloor(); }
  else if(k==='e'||k==='E'||k==='PageDown'){ e.preventDefault(); activeLayer=Math.min(L-1,activeLayer+1); updateFloor(); }
  else if(k>='1'&&k<='5'){ activeLayer=+k-1; updateFloor(); }
  else if(k==='ArrowLeft'){ ry-=6; applyView(); }
  else if(k==='ArrowRight'){ ry+=6; applyView(); }
  else if(k==='ArrowUp'){ rx=Math.min(86,rx+4); applyView(); }
  else if(k==='ArrowDown'){ rx=Math.max(12,rx-4); applyView(); }
});

document.getElementById('startBtn').addEventListener('click',()=>{ audio(); newGame(); });
document.getElementById('againBtn').addEventListener('click',()=>{ audio(); newGame(); });

/* ---------- fit to window ---------- */
function resize(){
  const s=Math.min(window.innerWidth/720, window.innerHeight/840);
  stage.style.transform='scale('+s+')';
}
window.addEventListener('resize',resize);

updateHUD(); updateFloor(); applyView(); resize();
})();
</script>
</body>
