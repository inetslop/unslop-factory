function circleRect(b,o){
  const cx=Math.max(o.x,Math.min(b.x,o.x+o.w)),cy=Math.max(o.y,Math.min(b.y,o.y+o.h));
  let dx=b.x-cx,dy=b.y-cy,d=Math.hypot(dx,dy);
  if(d<b.r){
    if(d===0){dx=0;dy=-1;d=1;}
    const nx=dx/d,ny=dy/d;
    b.x=cx+nx*b.r;b.y=cy+ny*b.r;
    const dot=b.vx*nx+b.vy*ny;
    if(dot<0){b.vx-=2*dot*nx;b.vy-=2*dot*ny;b.vx*=.85;b.vy*=.85;}
    return true;
  }
  return false;
}

function update(dt){

  if(G.state!=='play')return;

  const b=G.ball,g=G.golfer;

  // ball physics
  if(!G.sinking){

    for(const s of G.sands)if(dist(b.x,b.y,s.x,s.y)<s.r){fr=4.2;break;}
    const k=Math.exp(-fr*dt);
    b.vx*=k;b.vy*=k;
    if(Math.hypot(b.vx,b.vy)<5){b.vx=0;b.vy=0;}
    b.x+=b.vx*dt;b.y+=b.vy*dt;
    let hitWall=false;
    if(b.x<b.r){b.x=b.r;b.vx=Math.abs(b.vx)*.85;hitWall=true;}
    if(b.x>W-b.r){b.x=W-b.r;b.vx=-Math.abs(b.vx)*.85;hitWall=true;}
    if(b.y<b.r){b.y=b.r;b.vy=Math.abs(b.vy)*.85;hitWall=true;}
    if(b.y>H-b.r){b.y=H-b.r;b.vy=-Math.abs(b.vy)*.85;hitWall=true;}
    for(const o of G.obstacles)if(circleRect(b,o))hitWall=true;
    if(hitWall&&Math.hypot(b.vx,b.vy)>60)SFX.wall();

    // hole
    const dh=dist(b.x,b.y,G.hole.x,G.hole.y);
    if(dh<G.hole.r&&Math.hypot(b.vx,b.vy)<300){
      G.sinking=true;G.sinkT=0;SFX.sink();

      const bonus=Math.max(50,200-G.strokes*20)+G.level*20;

    }else if(dh<G.hole.r+10){ // rim pull
      const pull=90*dt;
      b.vx+=(G.hole.x-b.x)/dh*pull;b.vy+=(G.hole.y-b.y)/dh*pull;
    }

        }

  const zsp=(42+G.level*9)*Math.min(2,1+G.time*.001);

    if(f.t>1.4)G.floaters.splice(i,1);}
  G.holeAnim=Math.max(0,G.holeAnim-dt);
}

  ctx.beginPath();ctx.ellipse(0,2,8,9,0,0,7);ctx.fill();
  // head
  ctx.fillStyle='#f2c99a';
  ctx.beginPath();ctx.arc(0,-9,6.5,0,7);ctx.fill();
  // cap
  ctx.fillStyle='#d94b3a';
  ctx.beginPath();ctx.arc(0,-11,6.5,Math.PI,0);ctx.fill();
  ctx.fillRect(-8,-12,16,3);
  // face
  ctx.fillStyle='#332';
  ctx.beginPath();ctx.arc(-2,-9,1,0,7);ctx.arc(2.5,-9,1,0,7);ctx.fill();
  ctx.restore();
}

function draw(){
  // fairway
  const grd=ctx.createLinearGradient(0,0,0,H);
  grd.addColorStop(0,'#3f8f3a');grd.addColorStop(1,'#2f6e2c');
  ctx.fillStyle=grd;ctx.fillRect(0,0,W,H);
  // mow stripes
  ctx.fillStyle='rgba(255,255,255,.04)';
  for(let x=0;x<W;x+=120)if((x/120)%2===0)ctx.fillRect(x,0,60,H);

  // sand
  for(const s of G.sands){
    ctx.fillStyle='#d8c07a';
    ctx.beginPath();ctx.ellipse(s.x,s.y,s.r,s.r*.75,0,0,7);ctx.fill();
    ctx.fillStyle='rgba(0,0,0,.12)';
    for(let i=0;i<6;i++){
      const a=i*1.1+s.x;
      ctx.beginPath();ctx.arc(s.x+Math.cos(a)*s.r*.5,s.y+Math.sin(a)*s.r*.35,3,0,7);ctx.fill();
    }
  }

  // hole
  const h=G.hole,pulse=1+Math.sin(G.time*4)*.05;
  ctx.fillStyle='#111';
  ctx.beginPath();ctx.ellipse(h.x,h.y,h.r*pulse,h.r*.8*pulse,0,0,7);ctx.fill();
  // flag
  const fy=h.y-38+Math.sin(G.time*2)*1.5;
  ctx.strokeStyle='#eee';ctx.lineWidth=2.5;
  ctx.beginPath();ctx.moveTo(h.x,h.y-4);ctx.lineTo(h.x,fy);ctx.stroke();
  ctx.fillStyle='#ff4b4b';
  ctx.beginPath();ctx.moveTo(h.x,fy);ctx.lineTo(h.x+20,fy+6);ctx.lineTo(h.x,fy+13);ctx.closePath();ctx.fill();

  // obstacles (wooden planks)
  for(const o of G.obstacles){
    ctx.fillStyle='#8a5a2b';
    ctx.fillRect(o.x,o.y,o.w,o.h);
    ctx.strokeStyle='#5c3a17';ctx.lineWidth=3;
    ctx.strokeRect(o.x+1.5,o.y+1.5,o.w-3,o.h-3);
    ctx.strokeStyle='rgba(0,0,0,.2)';ctx.lineWidth=1;
    for(let i=1;i<o.w/22;i++){
      ctx.beginPath();ctx.moveTo(o.x+i*22,o.y+3);ctx.lineTo(o.x+i*22,o.y+o.h-3);ctx.stroke();
    }
  }

  // zombies
  for(const z of G.zombies)drawZombie(z);

  // golfer
  drawGolfer();

  // ball
  if(!G.sinking||G.sinkT<.5){
    const b=G.ball;
    let scale=1;
    if(G.sinking)scale=Math.max(0,1-G.sinkT*2);
    ctx.save();ctx.translate(b.x,b.y);ctx.scale(scale,scale);
    // shadow
    ctx.fillStyle='rgba(0,0,0,.3)';
    ctx.beginPath();ctx.ellipse(2,4,b.r,b.r*.6,0,0,7);ctx.fill();
    ctx.fillStyle='#fff';
    ctx.beginPath();ctx.arc(0,0,b.r,0,7);ctx.fill();
    ctx.strokeStyle='rgba(0,0,0,.25)';ctx.lineWidth=1;ctx.stroke();
    ctx.fillStyle='rgba(0,0,0,.15)';
    for(const [dx,dy] of [[-3,-2],[2,-4],[4,2],[-1,3],[0,-1]]){
      ctx.beginPath();ctx.arc(dx,dy,1.2,0,7);ctx.fill();
    }
    ctx.restore();
  }

  // particles
  for(const p of G.particles){
    ctx.globalAlpha=1-p.t/p.life;
    ctx.fillStyle=p.col;
    ctx.beginPath();ctx.arc(p.x,p.y,p.r,0,7);ctx.fill();
  }
  ctx.globalAlpha=1;

  // aim indicator
  if(aiming&&G.state==='play'){
    const dx=aimSx-aimX,dy=aimSy-aimY,len=Math.hypot(dx,dy);
    if(len>8){
      const power=Math.min(len,240)/240;
      const ang=Math.atan2(dy,dx);
      ctx.save();
      ctx.strokeStyle='rgba(255,255,255,.9)';ctx.lineWidth=3;ctx.setLineDash([8,7]);
      ctx.beginPath();ctx.moveTo(G.ball.x,G.ball.y);
      const L=40+power*160;
      ctx.lineTo(G.ball.x+Math.cos(ang)*L,G.ball.y+Math.sin(ang)*L);
      ctx.stroke();ctx.setLineDash([]);
      // arrowhead
      const ax=G.ball.x+Math.cos(ang)*L,ay=G.ball.y+Math.sin(ang)*L;
      ctx.fillStyle='rgba(255,255,255,.9)';
      ctx.beginPath();
      ctx.moveTo(ax,ay);
      ctx.lineTo(ax-Math.cos(ang-.4)*14,ay-Math.sin(ang-.4)*14);
      ctx.lineTo(ax-Math.cos(ang+.4)*14,ay-Math.sin(ang+.4)*14);
      ctx.closePath();ctx.fill();
      // power bar
      ctx.fillStyle='rgba(0,0,0,.5)';ctx.fillRect(G.ball.x-30,G.ball.y-32,60,8);
      ctx.fillStyle=power>.8?'#ff5b5b':power>.45?'#ffd75b':'#9fe86f';
      ctx.fillRect(G.ball.x-28,G.ball.y-30,56*power,4);
      ctx.restore();
    }
  }

  // floaters
  ctx.textAlign='center';ctx.font='bold 17px Courier New';
  for(const f of G.floaters){
    ctx.globalAlpha=Math.max(0,1-f.t/1.4);
    ctx.fillStyle=f.col;ctx.fillText(f.txt,f.x,f.y);
  }
  ctx.globalAlpha=1;

  // HUD
  if(G.state==='play'){
    ctx.fillStyle='rgba(0,0,0,.45)';
    ctx.fillRect(0,0,W,42);
    ctx.textAlign='left';ctx.font='bold 18px Courier New';ctx.fillStyle='#e8f5e0';
    ctx.fillText('HOLE '+G.level,16,27);
    ctx.fillText('STROKES '+G.strokes,130,27);
    ctx.fillText('SCORE '+G.score,290,27);
    // hearts
    for(let i=0;i<3;i++){
      const hx=W-150+i*44;
      ctx.fillStyle=i<G.lives?'#ff5b6e':'rgba(255,255,255,.18)';
      drawHeart(hx,21,9);
    }
  }
}

function drawHeart(x,y,s){
  ctx.beginPath();
  ctx.moveTo(x,y+s*.9);
  ctx.bezierCurveTo(x-s*1.4,y,x-s*.9,y-s*.9,x,y-s*.25);
  ctx.bezierCurveTo(x+s*.9,y-s*.9,x+s*1.4,y,x,y+s*.9);
  ctx.fill();
}

/* ---------- main loop ---------- */
let last=performance.now();
function loop(now){
  const dt=Math.min(.033,(now-last)/1000);last=now;
  update(dt);draw();
  requestAnimationFrame(loop);
}
newLevel(); // pre-populate for backdrop behind start screen
requestAnimationFrame(loop);
})();
