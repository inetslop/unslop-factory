  let first=true;
  for(let y=y0;y<=y1;y+=step){
    const c=cxAt(y),h=hwAt(y);
    if(first){p.moveTo(c-h,y);first=false;}else p.lineTo(c-h,y);
  }
  p.lineTo(cxAt(y1)+hwAt(y1),y1);
  for(let y=y1;y>=y0;y-=step)p.lineTo(cxAt(y)+hwAt(y),y);
  p.closePath();
  return p;
}
function render(){
  ctx.setTransform(1,0,0,1,0,0);
  ctx.fillStyle='#2c6329';ctx.fillRect(0,0,W,H);

  if(state==='menu'){drawMenu();return;}

  /* camera */
  const a=balls[0],b=balls[1];
  const sep=Math.abs(a.y-b.y);
  const scale=Math.max(0.35,Math.min(1.15,Math.min(W/720,H/(sep+560))));
  const camX=(a.x+b.x)/2,camY=(a.y+b.y)/2;
  ctx.setTransform(scale,0,0,scale,W/2-camX*scale,H*0.42-camY*scale);

  const y0=camY-H*0.5/scale-60,y1=camY+H*0.6/scale+60;
  const path=drawTrackPath(y0,y1);

  /* fairway */
  ctx.fillStyle='#5fae4e';ctx.fill(path);
  ctx.save();ctx.clip(path);
  for(let y=Math.floor(y0/200)*200;y<y1;y+=400){
    ctx.fillStyle='rgba(255,255,255,0.07)';ctx.fillRect(cxAt(y)-300,y,600,200);
  }
  /* hazards */
  for(let i=0;i<hazards.length;i++){
    const hz=hazards[i];if(hz.y<y0||hz.y>y1)continue;
    ctx.beginPath();ctx.arc(hz.x,hz.y,hz.r,0,7);
    if(hz.type==='sand'){ctx.fillStyle='#e6d29a';ctx.fill();ctx.strokeStyle='#c9b273';ctx.lineWidth=4;ctx.stroke();}
    else if(hz.type==='ice'){ctx.fillStyle='rgba(180,225,250,0.85)';ctx.fill();ctx.strokeStyle='#ffffff';ctx.lineWidth=3;ctx.stroke();}
    else if(hz.type==='water'){ctx.fillStyle='#3f7fd0';ctx.fill();ctx.beginPath();ctx.arc(hz.x,hz.y,hz.r*0.6,0,7);ctx.fillStyle='#5d9be0';ctx.fill();}
    else{ /* boost */
      const pul=1+0.12*Math.sin(stateTime*8+i);
      ctx.fillStyle='rgba(255,150,40,0.35)';ctx.beginPath();ctx.arc(hz.x,hz.y,hz.r*pul,0,7);ctx.fill();
      ctx.fillStyle='#ff9d2e';
      for(let k=-1;k<=1;k+=2){
        ctx.beginPath();
        ctx.moveTo(hz.x-14,hz.y+k*8+6);ctx.lineTo(hz.x+14,hz.y+k*8+6);ctx.lineTo(hz.x,hz.y+k*8-10);
        ctx.closePath();ctx.fill();
      }
    }
  }
  /* finish line */
  if(FINISH_Y>y0&&FINISH_Y<y1){
    const c=cxAt(FINISH_Y),h=hwAt(FINISH_Y),sq=22;
    for(let r=0;r<2;r++)for(let x=-c+h-sq;x<c-h;x+=sq){
      ctx.fillStyle=((Math.round((x+c)/sq)+r)%2===0)?'#111':'#fff';
      ctx.fillRect(x-r*0.01,FINISH_Y+r*sq-44,sq,sq);
    }
  }
  /* hole */
  {
    const hx=cxAt(HOLE_Y),hy=HOLE_Y;
    if(hy>y0&&hy<y1){
      ctx.fillStyle='rgba(255,255,255,0.12)';ctx.beginPath();ctx.arc(hx,hy,60,0,7);ctx.fill();
      ctx.fillStyle='#0c0c0c';ctx.beginPath();ctx.arc(hx,hy,17,0,7);ctx.fill();
      ctx.strokeStyle='#fff';ctx.lineWidth=3;ctx.stroke();
      ctx.strokeStyle='#ddd';ctx.lineWidth=4;
      ctx.beginPath();ctx.moveTo(hx,hy);ctx.lineTo(hx,hy-70);ctx.stroke();
      ctx.fillStyle='#e33';ctx.beginPath();ctx.moveTo(hx,hy-70);ctx.lineTo(hx+34,hy-58);ctx.lineTo(hx,hy-46);ctx.closePath();ctx.fill();
      ctx.fillStyle='rgba(255,255,255,0.9)';ctx.font='bold 20px sans-serif';ctx.textAlign='center';
      ctx.fillText('SLOW DOWN TO SINK!',hx,hy+95);
    }
  }
  /* walls */
  ctx.lineWidth=10;ctx.strokeStyle='#7a5b34';
  const step=30;
  ctx.beginPath();for(let y=y0;y<=y1;y+=step){const c=cxAt(y),h=hwAt(y);if(y===y0)ctx.moveTo(c-h,y);else ctx.lineTo(c-h,y);}ctx.stroke();
  ctx.beginPath();for(let y=y0;y<=y1;y+=step){const c=cxAt(y),h=hwAt(y);if(y===y0)ctx.moveTo(c+h,y);else ctx.lineTo(c+h,y);}ctx.stroke();
  /* checkpoints */
  for(let cp=900;cp<FINISH_Y;cp+=900){
    if(cp<y0||cp>y1)continue;
    const c=cxAt(cp),h=hwAt(cp);
    ctx.strokeStyle='rgba(255,255,255,0.35)';ctx.lineWidth=3;ctx.setLineDash([14,12]);
    ctx.beginPath();ctx.moveTo(c-h,cp);ctx.lineTo(c+h,cp);ctx.stroke();ctx.setLineDash([]);
  }

  /* balls */
  for(let bi=0;bi<2;bi++){
    const bl=balls[bi];if(bl.gone)continue;
    if(bl.splash>0){ /* splash ring */
      ctx.strokeStyle='rgba(150,210,255,'+(bl.splash/1.1)+')';ctx.lineWidth=6;
      ctx.beginPath();ctx.arc(bl.x,bl.y,(1.1-bl.splash)*70+14,0,7);ctx.stroke();
      continue;
    }
    for(let i=0;i<bl.trail.length;i++){
      const t=bl.trail[i],al=i/bl.trail.length*0.35;
      ctx.fillStyle='rgba(255,255,255,'+al+')';
      ctx.beginPath();ctx.arc(t.x,t.y,BR*(i/bl.trail.length),0,7);ctx.fill();
    }
    ctx.fillStyle='rgba(0,0,0,0.3)';ctx.beginPath();ctx.ellipse(bl.x+3,bl.y+5,BR,BR*0.6,0,0,7);ctx.fill();
    ctx.fillStyle='#fff';ctx.beginPath();ctx.arc(bl.x,bl.y,BR,0,7);ctx.fill();
    ctx.strokeStyle=bl.color;ctx.lineWidth=3.5;ctx.stroke();
    if(bl.charging){
      ctx.strokeStyle='rgba(255,233,138,'+(0.4+0.6*bl.charge)+')';ctx.lineWidth=4;
      ctx.beginPath();ctx.arc(bl.x,bl.y,BR+7,-Math.PI/2,-Math.PI/2+bl.charge*Math.PI*2);ctx.stroke();
    }
    if(bl.cd>0&&!bl.charging){
      ctx.strokeStyle='rgba(255,255,255,0.3)';ctx.lineWidth=2;
      ctx.beginPath();ctx.arc(bl.x,bl.y,BR+7,-Math.PI/2,-Math.PI/2+(1-bl.cd/2.2)*Math.PI*2);ctx.stroke();
    }
  }

  /* floating msgs */
  ctx.textAlign='center';
  for(let i=0;i<msgs.length;i++){
    const m=msgs[i];
    ctx.globalAlpha=Math.min(1,m.t);
    ctx.fillStyle=m.c;ctx.font='bold 26px sans-serif';
    ctx.fillText(m.txt,m.x,m.y-(1.2-m.t)*40);
    ctx.globalAlpha=1;
  }

  /* ---------- HUD (screen space) ---------- */
  ctx.setTransform(1,0,0,1,0,0);
  drawHUD();

  if(state==='count'){
    ctx.fillStyle='rgba(0,0,0,0.35)';ctx.fillRect(0,0,W,H);
    ctx.textAlign='center';
    const n=Math.ceil(countT);
    ctx.fillStyle='#fff';ctx.font='bold '+Math.min(160,H*0.25)+'px sans-serif';
    ctx.fillText(n>3?'':n,W/2,H/2+20);
  }
  if(state==='over'){drawOver();}
}
function drawHUD(){
  const pct=bl=>Math.min(1,Math.max(0,(bl.y-140)/(HOLE_Y-140)));
  ctx.font='bold 16px sans-serif';ctx.textAlign='left';
  for(let i=0;i<2;i++){
    const bl=balls[i],x=i===0?16:W-256;
    ctx.fillStyle=bl.color;ctx.beginPath();ctx.arc(x+8,24,8,0,7);ctx.fill();
    ctx.fillStyle='#fff';ctx.fillText('P'+(i+1),x+24,30);
    ctx.strokeStyle='#fff';ctx.lineWidth=2;ctx.strokeRect(x,40,220,14);
    ctx.fillStyle=bl.color;ctx.fillRect(x+2,42,216*pct(bl),10);
  }
  /* timer */
  const m=Math.floor(gameT/60),s=(gameT%60).toFixed(1).padStart(4,'0');
  ctx.textAlign='center';ctx.fillStyle='#fff';ctx.font='bold 22px sans-serif';
  ctx.fillText(m+':'+s,W/2,34);
}
function drawMenu(){
  const g=ctx.createLinearGradient(0,0,0,H);
  g.addColorStop(0,'#1d4a1c');g.addColorStop(1,'#2f6b2c');
  ctx.fillStyle=g;ctx.fillRect(0,0,W,H);
  /* decorative hole */
  ctx.fillStyle='#3f8f45';ctx.beginPath();ctx.ellipse(W/2,H*0.78,Math.min(W*0.35,300),60,0,0,7);ctx.fill();
  ctx.fillStyle='#111';ctx.beginPath();ctx.ellipse(W/2,H*0.78,34,14,0,0,7);ctx.fill();
  ctx.strokeStyle='#ddd';ctx.lineWidth=5;ctx.beginPath();ctx.moveTo(W/2,H*0.78);ctx.lineTo(W/2,H*0.6);ctx.stroke();
  ctx.fillStyle='#e33';ctx.beginPath();ctx.moveTo(W/2,H*0.6);ctx.lineTo(W/2+50,H*0.6+16);ctx.lineTo(W/2,H*0.6+32);ctx.closePath();ctx.fill();

  ctx.textAlign='center';
  ctx.fillStyle='#fff';
  ctx.font='bold '+Math.min(84,W*0.1)+'px sans-serif';
  ctx.shadowColor='#000';ctx.shadowBlur=14;
  ctx.fillText('RACING GOLF',W/2,H*0.2);
  ctx.shadowBlur=0;
  ctx.font=Math.min(22,W*0.035)+'px sans-serif';
  ctx.fillStyle='#ffe98a';
  ctx.fillText('Two balls. One hole. First to sink wins!',W/2,H*0.29);
  ctx.fillStyle='#eaffea';ctx.font='bold '+Math.min(19,W*0.03)+'px sans-serif';
  const l=Math.min(26,W*0.045);
  ctx.fillText('PLAYER 1 (RED):   A / D steer      SPACE hold = charge putt, release to fire',W/2,H*0.4+l);
  ctx.fillText('PLAYER 2 (BLUE):   \u2190 / \u2192 steer      ENTER hold = charge putt, release to fire',W/2,H*0.4+l*3);
  ctx.fillStyle='#cfe8cf';ctx.font=Math.min(16,W*0.028)+'px sans-serif';
  const notes=['\u25CF Sand slows you   \u25CF Ice is slippery   \u25CF Ponds reset you to a checkpoint','\u25BA Orange pads boost   \u25CF Cross the finish line, then slow down to sink the hole!'];
  for(let i=0;i<notes.length;i++)ctx.fillText(notes[i],W/2,H*0.4+l*6+i*l);
  ctx.fillStyle='#fff';ctx.font='bold '+Math.min(30,W*0.05)+'px sans-serif';
  if(Math.floor(stateTime*2)%2===0)ctx.fillText('PRESS ENTER OR CLICK TO START',W/2,H*0.92);
}
function drawOver(){
  ctx.fillStyle='rgba(0,0,0,0.55)';ctx.fillRect(0,0,W,H);
  ctx.textAlign='center';
  ctx.fillStyle=winner===1?P1C:P2C;
  ctx.font='bold '+Math.min(76,W*0.09)+'px sans-serif';
  ctx.shadowColor='#000';ctx.shadowBlur=12;
  ctx.fillText('PLAYER '+winner+' WINS!',W/2,H*0.38);
  ctx.shadowBlur=0;
  const m=Math.floor(gameT/60),s=(gameT%60).toFixed(1).padStart(4,'0');
  ctx.fillStyle='#fff';ctx.font='bold '+Math.min(30,W*0.05)+'px sans-serif';
  ctx.fillText('Race time: '+m+':'+s,W/2,H*0.5);
  if(Math.floor(stateTime*2)%2===0){
    ctx.fillStyle='#ffe98a';ctx.font='bold '+Math.min(24,W*0.04)+'px sans-serif';
    ctx.fillText('PRESS R / ENTER OR CLICK TO RACE AGAIN',W/2,H*0.62);
  }
}

/* ---------- loop ---------- */
let last=performance.now();
function frame(now){
  let dt=(now-last)/1000;last=now;
  if(dt>0.05)dt=0.05;
  update(dt);
  render();
  requestAnimationFrame(frame);
}
requestAnimationFrame(frame);
})();
