    ctx.strokeStyle='rgba(124,82,24,.55)';ctx.lineWidth=3;ctx.stroke();
    if(on){
      ctx.fillStyle='#eafff0';ctx.font='bold 20px monospace';ctx.textAlign='center';
      ctx.fillText('✓',b.x+T/2,b.y+T/2+7);
    }
  }

  // player
  if(player){
    const px=player.x,py=player.y;
    ctx.fillStyle='rgba(0,0,0,.25)';
    ctx.beginPath();ctx.ellipse(px+17,(Math.floor((py+40)/T))*T-3,16,5,0,0,7);ctx.fill();
    // body
    ctx.fillStyle='#ff8c42';roundRect(px,py+10,34,30,8);ctx.fill();
    ctx.strokeStyle='#b34d0e';ctx.lineWidth=3;roundRect(px,py+10,34,30,8);ctx.stroke();
    // head
    ctx.fillStyle='#ffd9b3';ctx.beginPath();ctx.arc(px+17,py+10,12,0,7);ctx.fill();
    ctx.strokeStyle='#b34d0e';ctx.lineWidth=2.5;ctx.stroke();
    // cap
    ctx.fillStyle='#e63946';ctx.beginPath();ctx.arc(px+17,py+8,12,Math.PI,0);ctx.fill();
    ctx.fillRect(px+17+(player.facing>0?0:-16),py+5,16,4);
    // eyes
    const eo=player.facing*3;
    ctx.fillStyle='#fff';
    ctx.beginPath();ctx.arc(px+12+eo,py+9,3.4,0,7);ctx.arc(px+22+eo,py+9,3.4,0,7);ctx.fill();
    ctx.fillStyle='#222';
    ctx.beginPath();ctx.arc(px+12.6+eo,py+9,1.6,0,7);ctx.arc(px+22.6+eo,py+9,1.6,0,7);ctx.fill();
  }
  ctx.restore();
}

function roundRect(x,y,w,h,r){
  ctx.beginPath();
  ctx.moveTo(x+r,y);ctx.arcTo(x+w,y,x+w,y+h,r);ctx.arcTo(x+w,y+h,x,y+h,r);
  ctx.arcTo(x,y+h,x,y,r);ctx.arcTo(x,y,x+w,y,r);ctx.closePath();
}
