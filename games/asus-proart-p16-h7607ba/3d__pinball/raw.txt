function ac(){if(!AC){try{AC=new (window.AudioContext||window.webkitAudioContext)();}catch(e){}}

function tone(f0,f1,dur,type,vol){const a=ac();if(!a)return;
  const o=a.createOscillator(),g=a.createGain(),t=a.currentTime;
  o.type=type;o.frequency.setValueAtTime(Math.max(1,f0),t);
  o.frequency.exponentialRampToValueAtTime(Math.max(1,f1),t+dur);
  g.gain.setValueAtTime(vol,t);g.gain.exponentialRampToValueAtTime(0.001,t+dur);
  o.connect(g);g.connect(a.destination);o.start(t);o.stop(t+dur+0.02);}

/* ---------- physics ---------- */
const G=420;
function collideSeg(w){
  const a=w.a,b=w.b,dx=b[0]-a[0],dz=b[1]-a[1];
  const len2=dx*dx+dz*dz;if(len2<1e-9)return;
  let t=((ball.x-a[0])*dx+(ball.z-a[1])*dz)/len2;t=t<0?0:(t>1?1:t);
  const cx=a[0]+dx*t,cz=a[1]+dz*t;
  let nx=ball.x-cx,nz=ball.z-cz,d=Math.hypot(nx,nz);
  if(d>=BR)return;
  if(d<1e-6){nx=0;nz=-1;d=1e-6;}else{nx/=d;nz/=d;}
  ball.x=cx+nx*BR;ball.z=cz+nz*BR;
  const vn=ball.vx*nx+ball.vz*nz;
  if(vn<0){
    ball.vx-=(1+w.e)*vn*nx;ball.vz-=(1+w.e)*vn*nz;
    if(w.boost&&vn<-60){ball.vx+=nx*w.boost;ball.vz+=nz*w.boost;
      score+=w.pts;addPopup(cx,cz,'+'+w.pts,'#ff9db3');sfx.sling();setHUD();}
  }
}
function hitCircle(c){
  let nx=ball.x-c.x,nz=ball.z-c.z,d=Math.hypot(nx,nz);
  const R=c.r+BR;
  if(d>=R||d<1e-6)return;
  nx/=d;nz/=d;
  ball.x=c.x+nx*R;ball.z=c.z+nz*R;
  const vn=ball.vx*nx+ball.vz*nz;
  if(vn<0){ball.vx-=1.8*vn*nx;ball.vz-=1.8*vn*nz;}
  const sp=Math.hypot(ball.vx,ball.vz);
  if(sp<c.min&&sp>1e-6){const k=c.min/sp;ball.vx*=k;ball.vz*=k;}
  else if(sp<=1e-6){ball.vx=nx*c.min*0.5;ball.vz=nz*c.min*0.5;}
  score+=c.pts;c.flash=1;addPopup(c.x,c.z+8,'+'+c.pts,c.color);
  (c.pts>200?sfx.targ:sfx.bump)();setHUD();
}
function flipCollide(f){
  const p=f.pivot,tx=p.x+f.L*Math.cos(f.angle),tz=p.z+f.L*Math.sin(f.angle);
  const dx=tx-p.x,dz=tz-p.z;
  let t=((ball.x-p.x)*dx+(ball.z-p.z)*dz)/(f.L*f.L);t=t<0?0:(t>1?1:t);
  const cx=p.x+dx*t,cz=p.z+dz*t;
  let nx=ball.x-cx,nz=ball.z-cz,d=Math.hypot(nx,nz);
  const R=BR+f.r;if(d>=R)return;
  if(d<1e-6){nx=0;nz=-1;d=1e-6;}else{nx/=d;nz/=d;}
  ball.x=cx+nx*R;ball.z=cz+nz*R;
  const qx=cx-p.x,qz=cz-p.z;
  const fx=f.omega*(-qz),fz=f.omega*qx;          // surface velocity at contact
  const rvx=ball.vx-fx,rvz=ball.vz-fz,vn=rvx*nx+rvz*nz;
  if(vn<0){ball.vx-=1.35*vn*nx;ball.vz-=1.35*vn*nz;}
}
function substep(dt){
  ball.vz-=G*dt;
  const drag=1-0.12*dt;ball.vx*=drag;ball.vz*=drag;
  ball.x+=ball.vx*dt;ball.z+=ball.vz*dt;
  for(let i=0;i<walls.length;i++)collideSeg(walls[i]);
  for(let i=0;i<bumpers.length;i++)hitCircle(bumpers[i]);
  for(let i=0;i<targets.length;i++)hitCircle(targets[i]);
  for(let i=0;i<flippers.length;i++)flipCollide(flippers[i]);
  const sp=Math.hypot(ball.vx,ball.vz);
  if(sp>950){const k=950/sp;ball.vx*=k;ball.vz*=k;}
}
function step(dt){
  for(const c of bumpers)c.flash=Math.max(0,c.flash-dt*3);
  for(const c of targets)c.flash=Math.max(0,c.flash-dt*3);
  for(let i=popups.length-1;i>=0;i--){popups[i].t+=dt*1.4;if(popups[i].t>1)popups.splice(i,1);}
  if(state!=='playing')return;

  // flippers
  for(const f of flippers){
    const target=f.active?f.A1:f.A0,diff=target-f.angle,maxd=24*dt;
    if(Math.abs(diff)<=maxd){f.omega=(target-f.angle)/dt;f.angle=target;}
    else{const s=Math.sign(diff)*maxd;f.omega=s/dt;f.angle+=s;}
  }
  if(charging)charge=Math.min(1,charge+dt*1.15);

  if(ball.gone){drainT-=dt;if(drainT<=0)spawnBall();return;}

  const N=6;
  for(let i=0;i<N;i++)substep(dt/N);

  // drain
  if(ball.z<-26||ball.x<-170||ball.x>170){
    ball.gone=true;drainT=0.9;ballsLeft--;sfx.drain();setHUD();
    if(ballsLeft<=0)gameOver();
  }
}

function drawWall(w){
  const A=prj(w.a[0],w.a[1],w.h),B=prj(w.b[0],w.b[1],w.h),
        C=prj(w.b[0],w.b[1],0),D=prj(w.a[0],w.a[1],0);
  ctx.beginPath();ctx.moveTo(A.x,A.y);ctx.lineTo(B.x,B.y);ctx.lineTo(C.x,C.y);ctx.lineTo(D.x,D.y);

  ctx.beginPath();ctx.moveTo(A.x,A.y);ctx.lineTo(B.x,B.y);

function drawCyl(c,hgt,faceColor){
  const base=prj(c.x,c.z,0),rx=c.r*base.s,ry=rx*TILT,hpx=hgt*base.s;
  const tx=base.x,ty=base.y-hpx,N=26;
  ctx.beginPath();
  for(let i=0;i<=N;i++){const a=i/N*Math.PI*2;
    const x=tx+rx*Math.cos(a),y=ty+ry*Math.sin(a);i?ctx.lineTo(x,y):ctx.moveTo(x,y);}
  for(let i=N;i>=0;i--){const a=i/N*Math.PI*2;ctx.lineTo(base.x+rx*Math.cos(a),base.y+ry*Math.sin(a));}
  ctx.closePath();
  const side=shade(faceColor,-45);ctx.fillStyle=side;ctx.fill();
  ctx.beginPath();ctx.ellipse(tx,ty,rx,ry,0,0,Math.PI*2);
  ctx.fillStyle=faceColor;ctx.fill();
  if(c.flash>0){ctx.beginPath();ctx.ellipse(tx,ty,rx,ry,0,0,Math.PI*2);
    ctx.fillStyle='rgba(255,255,255,'+(c.flash*0.85)+')';ctx.fill();}
  ctx.beginPath();ctx.ellipse(base.x,base.y,rx,ry,0,0,Math.PI*2);
  ctx.strokeStyle='rgba(0,0,0,.4)';ctx.lineWidth=1;ctx.stroke();
}
function shade(hex,amt){
  const n=parseInt(hex.slice(1),16);
  let r=(n>>16)+amt,g=((n>>8)&255)+amt,b=(n&255)+amt;
  r=r<0?0:(r>255?255:r);g=g<0?0:(g>255?255:g);b=b<0?0:(b>255?255:b);
  return 'rgb('+r+','+g+','+b+')';
}
