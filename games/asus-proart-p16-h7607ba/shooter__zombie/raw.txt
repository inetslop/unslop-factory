"use strict";
const canvas = document.getElementById('game');
const ctx = canvas.getContext('2d');
let W = 0, H = 0;
function resize(){ W = canvas.width = window.innerWidth; H = canvas.height = window.innerHeight; }
window.addEventListener('resize', resize); resize();

// ---------- Audio ----------
let actx = null;
function initAudio(){ if(!actx){ try{ actx = new (window.AudioContext||window.webkitAudioContext)(); }catch(e){} } if(actx && actx.state==='suspended') actx.resume(); }
function sfx(type){
  if(!actx) return;
  const t = actx.currentTime;
  try{
    if(type==='shot'){
      const o=actx.createOscillator(), g=actx.createGain(), n=actx.createBufferSource();
      const buf=actx.createBuffer(1,2205,44100), d=buf.getChannelData(0);
      for(let i=0;i<2205;i++) d[i]=(Math.random()*2-1)*Math.pow(1-i/2205,2.5);
      n.buffer=buf;
      const f=actx.createBiquadFilter(); f.type='lowpass'; f.frequency.value=1800;
      o.type='square'; o.frequency.setValueAtTime(160,t); o.frequency.exponentialRampToValueAtTime(50,t+0.09);
      g.gain.setValueAtTime(0.22,t); g.gain.exponentialRampToValueAtTime(0.001,t+0.1);
      n.connect(f).connect(g).connect(actx.destination);
      o.connect(g);
      o.start(t); o.stop(t+0.1); n.start(t);
    } else if(type==='reload'){
      const o=actx.createOscillator(), g=actx.createGain();
      o.type='square'; o.frequency.setValueAtTime(700,t); o.frequency.setValueAtTime(500,t+0.12);
      g.gain.setValueAtTime(0.08,t); g.gain.exponentialRampToValueAtTime(0.001,t+0.2);
      o.connect(g).connect(actx.destination); o.start(t); o.stop(t+0.2);
    } else if(type==='zhit'){
      const n=actx.createBufferSource(), buf=actx.createBuffer(1,3000,44100), d=buf.getChannelData(0);
      for(let i=0;i<3000;i++) d[i]=(Math.random()*2-1)*Math.pow(1-i/3000,2);
      n.buffer=buf; const f=actx.createBiquadFilter(); f.type='bandpass'; f.frequency.value=400;
      const g=actx.createGain(); g.gain.setValueAtTime(0.18,t);
      n.connect(f).connect(g).connect(actx.destination); n.start(t);
    } else if(type==='zdie'){
      const o=actx.createOscillator(), g=actx.createGain();
      o.type='sawtooth'; o.frequency.setValueAtTime(180,t); o.frequency.exponentialRampToValueAtTime(45,t+0.35);
      g.gain.setValueAtTime(0.15,t); g.gain.exponentialRampToValueAtTime(0.001,t+0.35);
      o.connect(g).connect(actx.destination); o.start(t); o.stop(t+0.36);
    } else if(type==='hurt'){
      const o=actx.createOscillator(), g=actx.createGain();
      o.type='sawtooth'; o.frequency.setValueAtTime(220,t); o.frequency.exponentialRampToValueAtTime(80,t+0.25);
      g.gain.setValueAtTime(0.22,t); g.gain.exponentialRampToValueAtTime(0.001,t+0.25);
      o.connect(g).connect(actx.destination); o.start(t); o.stop(t+0.26);
    } else if(type==='dash'){
      const o=actx.createOscillator(), g=actx.createGain();
      o.type='sine'; o.frequency.setValueAtTime(300,t); o.frequency.exponentialRampToValueAtTime(900,t+0.12);
      g.gain.setValueAtTime(0.1,t); g.gain.exponentialRampToValueAtTime(0.001,t+0.15);
      o.connect(g).connect(actx.destination); o.start(t); o.stop(t+0.16);
    } else if(type==='wave'){
      [220,330,440].forEach((fr,i)=>{
        const o=actx.createOscillator(), g=actx.createGain();
        o.type='triangle'; o.frequency.value=fr;
        g.gain.setValueAtTime(0.1,t+i*0.12); g.gain.exponentialRampToValueAtTime(0.001,t+i*0.12+0.3);
        o.connect(g).connect(actx.destination); o.start(t+i*0.12); o.stop(t+i*0.12+0.31);
      });
    }
  }catch(e){}
}

// ---------- Input ----------
const keys = {};
window.addEventListener('keydown', e=>{
  keys[e.code]=true;
  if(['Space','ArrowUp','ArrowDown','ArrowLeft','ArrowRight'].includes(e.code)) e.preventDefault();
  if(gameState==='playing' && e.code==='KeyR') startReload();
  if(gameState==='gameover' && (e.code==='KeyR'||e.code==='Enter')) startGame();
  if(gameState==='start' && (e.code==='Enter'||e.code==='Space')) startGame();
});
window.addEventListener('keyup', e=>{ keys[e.code]=false; });
let mouse={x:W/2,y:H/2,down:false};
canvas.addEventListener('mousemove', e=>{ const r=canvas.getBoundingClientRect(); mouse.x=e.clientX-r.left; mouse.y=e.clientY-r.top; });
canvas.addEventListener('mousedown', e=>{ initAudio(); if(e.button===0) mouse.down=true; });
window.addEventListener('mouseup', ()=> mouse.down=false);
canvas.addEventListener('contextmenu', e=>e.preventDefault());

// ---------- Game state ----------
let gameState='start'; // start | playing | gameover
let player, bullets, zombies, particles, floaters, wave, score, spawnQueue, spawnTimer, waveBreak, shakeT;

function resetGame(){
  player = { x:W/2, y:H/2, r:14, hp:100, maxHp:100, speed:230, mag:12, magSize:12, reloading:0, fireCd:0, dashCd:0, inv:0, angle:0 };
  bullets=[]; zombies=[]; particles=[]; floaters=[];
  wave=0; score=0; spawnQueue=0; spawnTimer=0; waveBreak=1.5; shakeT=0;
  nextWave();
}
function nextWave(){
  wave++;
  spawnQueue = 5 + wave*3;
  spawnTimer = 0;
  sfx('wave');
  floaters.push({x:W/2, y:H/2-80, text:'WAVE '+wave, life:2, max:2, size:42, color:'#7fbf5f'});
}

function spawnZombie(){
  const side = Math.floor(Math.random()*4);
  let x,y;
  if(side===0){ x=Math.random()*W; y=-30; }
  else if(side===1){ x=W+30; y=Math.random()*H; }
  else if(side===2){ x=Math.random()*W; y=H+30; }
  else { x=-30; y=Math.random()*H; }
  const runner = wave>=3 && Math.random()<Math.min(0.15+wave*0.02, 0.4);
  const tank  = wave>=5 && !runner && Math.random()<Math.min(0.08+wave*0.01, 0.25);
  let z;
  if(tank){
    z={x,y,r:22,hp:6+wave*1.5,maxHp:6+wave*1.5,speed:55+Math.random()*15,dmg:22,pts:30,color:'#4a7a3a',type:'tank',wob:Math.random()*6.28};
  } else if(runner){
    z={x,y,r:11,hp:2+wave*0.4,maxHp:2+wave*0.4,speed:150+Math.random()*40,dmg:8,pts:15,color:'#c9a83c',type:'runner',wob:Math.random()*6.28};
  } else {
    z={x,y,r:15,hp:3+wave*0.7,maxHp:3+wave*0.7,speed:70+Math.random()*35+dawnSpeed(),dmg:12,pts:10,color:'#5e9c4a',type:'walker',wob:Math.random()*6.28};
  }
  function dawnSpeed(){ return Math.min(wave*3,40); }
  z.hitFlash=0; z.deadT=0;
  zombies.push(z);
}

function startReload(){
  if(player.reloading>0 || player.mag===player.magSize) return;
  player.reloading = 1.2;
  sfx('reload');
}

function shoot(){
  const p=player;
  if(p.fireCd>0 || p.reloading>0) return;
  if(p.mag<=0){ startReload(); return; }
  p.mag--; p.fireCd = 0.14;
  const a = Math.atan2(mouse.y-p.y, mouse.x-p.x) + (Math.random()-0.5)*0.06;
  bullets.push({x:p.x+Math.cos(a)*20, y:p.y+Math.sin(a)*20, vx:Math.cos(a)*780, vy:Math.sin(a)*780, life:1.1});
  // muzzle flash particles
  for(let i=0;i<4;i++){
    const sp = 60+Math.random()*120;
    const aa=a+(Math.random()-0.5)*0.9;
    particles.push({x:p.x+Math.cos(a)*22, y:p.y+Math.sin(a)*22, vx:Math.cos(aa)*sp, vy:Math.sin(aa)*sp, life:0.12, max:0.12, size:3+Math.random()*3, color:'#ffcf5a'});
  }
  shakeT = Math.max(shakeT, 0.06);
  sfx('shot');
  if(p.mag===0) startReload();
}

function bloodBurst(x,y,color,n,speed){
  for(let i=0;i<n;i++){
    const a=Math.random()*Math.PI*2, sp=speed*(0.3+Math.random());
    particles.push({x,y,vx:Math.cos(a)*sp,vy:Math.sin(a)*sp,life:0.4+Math.random()*0.5,max:0.9,size:2+Math.random()*4,color});
  }
}

function startGame(){
  initAudio();
  resetGame();
  gameState='playing';
  document.getElementById('startScreen').classList.add('hidden');
  document.getElementById('gameOverScreen').classList.add('hidden');
  document.getElementById('hud').classList.remove('hidden');
}
function gameOver(){
  gameState='gameover';
  document.getElementById('finalStats').textContent = 'WAVE '+wave+'  •  SCORE '+score;
  document.getElementById('gameOverScreen').classList.remove('hidden');
  // best score
  try{
    const b = parseInt(localStorage.getItem('deadzone_best')||'0',10);
    if(score>b) localStorage.setItem('deadzone_best', String(score));
  }catch(e){}
}

// ---------- Update ----------
function update(dt){
  const p=player;
  // movement
  let mx=0,my=0;
  if(keys['KeyW']||keys['ArrowUp']) my--;
  if(keys['KeyS']||keys['ArrowDown']) my++;
  if(keys['KeyA']||keys['ArrowLeft']) mx--;
  if(keys['KeyD']||keys['ArrowRight']) mx++;
  const len=Math.hypot(mx,my);
  if(len>0){ mx/=len; my/=len; p.x+=mx*p.speed*dt; p.y+=my*p.speed*dt; }
  p.x=Math.max(p.r,Math.min(W-p.r,p.x));
  p.y=Math.max(p.r,Math.min(H-p.r,p.y));

  // dash
  if(p.dashCd>0) p.dashCd-=dt;
  if(keys['Space'] && len>0 && p.dashCd<=0){
    p.x+=mx*90; p.y+=my*90;
    p.x=Math.max(p.r,Math.min(W-p.r,p.x)); p.y=Math.max(p.r,Math.min(H-p.r,p.y));
    p.dashCd=1.2; p.inv=Math.max(p.inv,0.25);
    for(let i=0;i<8;i++) particles.push({x:p.x-mx*14,y:p.y-my*14,vx:-mx*(80+Math.random()*80)+(Math.random()-0.5)*60,vy:-my*(80+Math.random()*80)+(Math.random()-0.5)*60,life:0.3,max:0.3,size:3,color:'#8fd3ff'});
    sfx('dash');
  }

  p.angle=Math.atan2(mouse.y-p.y,mouse.x-p.x);
  if(p.fireCd>0) p.fireCd-=dt;
  if(p.inv>0) p.inv-=dt;
  if(p.reloading>0){ p.reloading-=dt; if(p.reloading<=0) p.mag=p.magSize; }
  if(mouse.down) shoot();

  // wave spawning
  if(waveBreak>0){
    waveBreak-=dt;
    if(waveBreak<=0 && spawnQueue===0) nextWave();
  } else if(spawnQueue>0){
    spawnTimer-=dt;
    if(spawnTimer<=0){
      spawnZombie();
      spawnQueue--;
      spawnTimer = Math.max(0.25, 1.4 - wave*0.07) * (0.6+Math.random()*0.8);
    }
  } else {
    if(zombies.length===0){ waveBreak=2; score+=wave*25; floaters.push({x:W/2,y:H/2-40,text:'+'+(wave*25)+' WAVE BONUS',life:1.5,max:1.5,size:26,color:'#ffd94a'}); }
  }

  // bullets
  for(let i=bullets.length-1;i>=0;i--){
    const b=bullets[i];
    b.x+=b.vx*dt; b.y+=b.vy*dt; b.life-=dt;
    let hit=false;
    for(const z of zombies){
      if(z.deadT>0) continue;
      const dx=z.x-b.x, dy=z.y-b.y;
      if(dx*dx+dy*dy < (z.r+4)*(z.r+4)){
        z.hp -= 1;
        z.hitFlash=0.12;
        bloodBurst(b.x,b.y,'#a01820',6,90);
        // knockback
        const kb = z.type==='tank'?18:38;
        const al=Math.hypot(b.vx,b.vy)||1;
        z.x+=b.vx/al*kb*dt*4; z.y+=b.vy/al*kb*dt*4;
        sfx('zhit');
        hit=true;
        if(z.hp<=0){
          z.deadT=0.6;
          score+=z.pts;
          floaters.push({x:z.x,y:z.z||z.y-14,text:'+'+z.pts,life:0.8,max:0.8,size:15,color:'#ffd94a'});
          bloodBurst(z.x,z.y,'#7a1218',16,160);
          sfx('zdie');
        }
        break;
      }
    }
    if(hit || b.life<=0 || b.x<-20||b.x>W+20||b.y<-20||b.y>H+20) bullets.splice(i,1);
  }

  // zombies
  for(let i=zombies.length-1;i>=0;i--){
    const z=zombies[i];
    if(z.deadT>0){
      z.deadT-=dt;
      if(z.deadT<=0) zombies.splice(i,1);
      continue;
    }
    if(z.hitFlash>0) z.hitFlash-=dt;
    z.wob+=dt*8;
    const dx=p.x-z.x, dy=p.y-z.y, d=Math.hypot(dx,dy)||1;
    // separation
    let sx=0,sy=0;
    for(const o of zombies){
      if(o===z||o.deadT>0) continue;
      const ox=z.x-o.x, oy=z.y-o.y, od=Math.hypot(ox,oy);
      if(od<z.r+o.r && od>0){ sx+=ox/od*(1-od/(z.r+o.r)); sy+=oy/od*(1-od/(z.r+o.r)); }
    }
    const sp=z.speed*(1+Math.sin(z.wob)*0.12);
    z.x+=(dx/d*sp + sx*40)*dt;
    z.y+=(dy/d*sp + sy*40)*dt;
    if(d < z.r+p.r && p.inv<=0){
      p.hp-=z.dmg*dt*3;
      p.inv=0.25; shakeT=Math.max(shakeT,0.15);
      bloodBurst(p.x,p.y,'#c8202a',8,120);
      sfx('hurt');
      if(p.hp<=0){ p.hp=0; gameOver(); return; }
    }
  }

  // particles
  for(let i=particles.length-1;i>=0;i--){
    const q=particles[i];
    q.x+=q.vx*dt; q.y+=q.vy*dt; q.vx*=0.92; q.vy*=0.92; q.life-=dt;
    if(q.life<=0) particles.splice(i,1);
  }
  for(let i=floaters.length-1;i>=0;i--){
    const f=floaters[i]; f.y-=30*dt; f.life-=dt;
    if(f.life<=0) floaters.splice(i,1);
  }
  if(shakeT>0) shakeT-=dt;

  // HUD
  document.getElementById('hpText').textContent = Math.ceil(p.hp);
  document.getElementById('ammoText').textContent = p.reloading>0 ? '...' : (p.mag+'/'+p.magSize);
  document.getElementById('waveVal').textContent = wave;
  document.getElementById('scoreVal').textContent = score;
}

// ---------- Draw ----------
function draw(){
  ctx.save();
  if(shakeT>0 && gameState==='playing'){
    ctx.translate((Math.random()-0.5)*shakeT*60,(Math.random()-0.5)*shakeT*60);
  }
  // background
  ctx.fillStyle='#141419';
  ctx.fillRect(-20,-20,W+40,H+40);
  // grid
  ctx.strokeStyle='rgba(255,255,255,0.035)';
  ctx.lineWidth=1;
  ctx.beginPath();
  for(let x=0;x<W;x+=60){ ctx.moveTo(x,0); ctx.lineTo(x,H); }
  for(let y=0;y<H;y+=60){ ctx.moveTo(0,y); ctx.lineTo(W,y); }
  ctx.stroke();
  // vignette-ish stains
  ctx.fillStyle='rgba(90,20,25,0.06)';
  for(let i=0;i<6;i++){
    const sx=(i*419)%W, sy=(i*733)%H;
    ctx.beginPath(); ctx.ellipse(sx,sy,80+i*22,50+i*15,i,0,Math.PI*2); ctx.fill();
  }

  if(gameState==='playing'||gameState==='gameover'){
    // particles under
    for(const q of particles){
      ctx.globalAlpha=Math.max(0,q.life/q.max);
      ctx.fillStyle=q.color;
      ctx.beginPath(); ctx.arc(q.x,q.y,q.size,0,Math.PI*2); ctx.fill();
    }
    ctx.globalAlpha=1;

    // zombies
    for(const z of zombies){
      const dead = z.deadT>0;
      const alpha = dead ? Math.max(0,z.deadT/0.6) : 1;
      ctx.globalAlpha=alpha;
      const bob=Math.sin(z.wob)*2;
      // body
      ctx.fillStyle = z.hitFlash>0 ? '#ffffff' : (dead ? '#2a3a24' : z.color);
      ctx.beginPath(); ctx.arc(z.x,z.y+bob,z.r,0,Math.PI*2); ctx.fill();
      if(!dead){
        // arms reaching toward player
        const a=Math.atan2(player.y-z.y,player.x-z.x);
        ctx.strokeStyle=z.color; ctx.lineWidth=5; ctx.lineCap='round';
        for(const off of [-0.5,0.5]){
          ctx.beginPath();
          ctx.moveTo(z.x+Math.cos(a+off)*z.r*0.5, z.y+bob+Math.sin(a+off)*z.r*0.5);
          ctx.lineTo(z.x+Math.cos(a+off)*(z.r+10)+Math.sin(z.wob+off*3)*3, z.y+bob+Math.sin(a+off)*(z.r+10));
          ctx.stroke();
        }
        // eyes
        const ea=a;
        ctx.fillStyle='#ffcc22';
        ctx.beginPath();
        ctx.arc(z.x+Math.cos(ea-0.4)*z.r*0.5, z.y+bob+Math.sin(ea-0.4)*z.r*0.5, 2.4,0,Math.PI*2);
        ctx.arc(z.x+Math.cos(ea+0.4)*z.r*0.5, z.y+bob+Math.sin(ea+0.4)*z.r*0.5, 2.4,0,Math.PI*2);
        ctx.fill();
        // hp bar if damaged
        if(z.hp<z.maxHp){
          ctx.fillStyle='rgba(0,0,0,0.6)';
          ctx.fillRect(z.x-z.r, z.y-z.r-10+bob, z.r*2, 4);
          ctx.fillStyle='#c8202a';
          ctx.fillRect(z.x-z.r, z.y-z.r-10+bob, z.r*2*(z.hp/z.maxHp), 4);
        }
      }
    }
    ctx.globalAlpha=1;

    // bullets
    ctx.strokeStyle='#ffe28a'; ctx.lineWidth=3; ctx.lineCap='round';
    for(const b of bullets){
      ctx.beginPath();
      ctx.moveTo(b.x-b.vx*0.016, b.y-b.vy*0.016);
      ctx.lineTo(b.x,b.y);
      ctx.stroke();
    }

    // player
    const p=player;
    if(p.inv>0 && Math.floor(p.inv*30)%2===0) ctx.globalAlpha=0.4;
    // gun
    ctx.strokeStyle='#c8c8d0'; ctx.lineWidth=6; ctx.lineCap='round';
    ctx.beginPath();
    ctx.moveTo(p.x+Math.cos(p.angle)*8, p.y+Math.sin(p.angle)*8);
    ctx.lineTo(p.x+Math.cos(p.angle)*24, p.y+Math.sin(p.angle)*24);
    ctx.stroke();
    // body
    ctx.fillStyle='#3a6ea5';
    ctx.beginPath(); ctx.arc(p.x,p.y,p.r,0,Math.PI*2); ctx.fill();
    ctx.fillStyle='#8ec8f0';
    ctx.beginPath(); ctx.arc(p.x+Math.cos(p.angle)*4, p.y+Math.sin(p.angle)*4, p.r*0.55,0,Math.PI*2); ctx.fill();
    // reload indicator
    if(p.reloading>0){
      ctx.strokeStyle='#ffd94a'; ctx.lineWidth=3;
      ctx.beginPath();
      ctx.arc(p.x,p.y,p.r+8,-Math.PI/2,-Math.PI/2+(1-p.reloading/1.2)*Math.PI*2);
      ctx.stroke();
    }
    // dash cooldown ring (subtle)
    if(p.dashCd>0){
      ctx.strokeStyle='rgba(143,211,255,0.4)'; ctx.lineWidth=2;
      ctx.beginPath();
      ctx.arc(p.x,p.y,p.r+13,-Math.PI/2,-Math.PI/2+(1-p.dashCd/1.2)*Math.PI*2);
      ctx.stroke();
    }
    ctx.globalAlpha=1;

    // floating text
    for(const f of floaters){
      ctx.globalAlpha=Math.max(0,f.life/f.max);
      ctx.fillStyle=f.color;
      ctx.font='bold '+f.size+'px Courier New';
      ctx.textAlign='center';
      ctx.fillText(f.text, f.x, f.y);
    }
    ctx.globalAlpha=1;

    // low hp overlay
    if(p.hp<35 && gameState==='playing'){
      ctx.fillStyle='rgba(160,20,30,'+(0.18*(1-p.hp/35)+Math.sin(Date.now()/150)*0.04)+')';
      ctx.fillRect(-20,-20,W+40,H+40);
    }

    // crosshair
    ctx.strokeStyle='rgba(255,80,90,0.9)'; ctx.lineWidth=2;
    const cs=10;
    ctx.beginPath();
    ctx.moveTo(mouse.x-cs,mouse.y); ctx.lineTo(mouse.x-4,mouse.y);
    ctx.moveTo(mouse.x+4,mouse.y); ctx.lineTo(mouse.x+cs,mouse.y);
    ctx.moveTo(mouse.x,mouse.y-cs); ctx.lineTo(mouse.x,mouse.y-4);
    ctx.moveTo(mouse.x,mouse.y+4); ctx.lineTo(mouse.x,mouse.y+cs);
    ctx.stroke();

    // wave break banner
    if(waveBreak>0 && spawnQueue===0 && zombies.length===0){
      ctx.fillStyle='rgba(127,191,95,'+(Math.min(1,waveBreak))+')';
      ctx.font='bold 34px Courier New';
      ctx.textAlign='center';
      ctx.fillText('GET READY...', W/2, H/2-60);
    }
  }
  ctx.restore();
}

// ---------- Loop ----------
let last=performance.now();
function loop(now){
  const dt=Math.min(0.05,(now-last)/1000);
  last=now;
  if(gameState==='playing') update(dt);
  draw();
  requestAnimationFrame(loop);
}
