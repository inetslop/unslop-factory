(function(){
"use strict";
const cv = document.getElementById('cv');
const ctx = cv.getContext('2d');
let W=0,H=0,DPR=1;

function resize(){
  DPR = Math.min(window.devicePixelRatio||1,2);
  W = window.innerWidth; H = window.innerHeight;
  cv.width = W*DPR; cv.height = H*DPR;
  ctx.setTransform(DPR,0,0,DPR,0,0);
}
window.addEventListener('resize', resize);
resize();

// ---------- audio ----------
let AC=null;
function ensureAudio(){
  if(!AC){ try{ AC=new (window.AudioContext||window.webkitAudioContext)(); }catch(e){} }
  if(AC && AC.state==='suspended') AC.resume();
}
function beep(freq,dur,type,vol,slide){
  if(!AC) return;
  const o=AC.createOscillator(), g=AC.createGain();
  o.type=type||'square'; o.frequency.value=freq;
  if(slide) o.frequency.exponentialRampToValueAtTime(Math.max(30,freq+slide), AC.currentTime+dur);
  g.gain.setValueAtTime(vol||.15, AC.currentTime);
  g.gain.exponentialRampToValueAtTime(.0001, AC.currentTime+dur);
  o.connect(g).connect(AC.destination);
  o.start(); o.stop(AC.currentTime+dur);
}
function noiseBurst(dur,vol){
  if(!AC) return;
  const len=Math.floor(AC.sampleRate*dur), buf=AC.createBuffer(1,len,AC.sampleRate), d=buf.getChannelData(0);
  for(let i=0;i<len;i++) d[i]=(Math.random()*2-1)*(1-i/len);
  const s=AC.createBufferSource(), g=AC.createGain();
  s.buffer=buf; g.gain.value=vol||.3;
  s.connect(g).connect(AC.destination); s.start();
}
function sfxCoin(){ beep(880,.09,'square',.12); setTimeout(()=>beep(1320,.12,'square',.1),60); }
function sfxCrash(){ noiseBurst(.5,.4); beep(90,.5,'sawtooth',.25,-60); }
function sfxScrape(){ beep(180,.08,'sawtooth',.05,-40); }

// ---------- game state ----------
const STATE={MENU:0,PLAY:1,DEAD:2};
let state=STATE.MENU;
let car, entities, particles, spawnX, dist, coins, speed, baseSpeed, shake, deadTimer, best=0;
try{ best = parseInt(localStorage.getItem('ttdd_best')||'0',10)||0; }catch(e){}

const keys={};
window.addEventListener('keydown',e=>{
  ensureAudio();
  const k=e.key;
  if(['ArrowUp','ArrowDown','ArrowLeft','ArrowRight',' '].includes(k)) e.preventDefault();
  keys[k.toLowerCase()]=true; keys[k]=true;
  if(k===' '){
    if(state===STATE.MENU) startGame();
    else if(state===STATE.DEAD && deadTimer<=0) startGame();
  }
  if((k==='r'||k==='R') && state!==STATE.PLAY) startGame();
});
window.addEventListener('keyup',e=>{ keys[e.key]=false; keys[e.key.toLowerCase()]=false; });

function roadMetrics(){
  const margin=Math.max(40,H*0.12);
  return {top:margin, bottom:H-margin, h:H-margin*2};
}

function reset(){
  const rm=roadMetrics();
  car={ x:0, y:(rm.top+rm.bottom)/2, w:34, h:62, vy:0, tilt:0 };
  entities=[]; particles=[];
  dist=0; coins=0; baseSpeed=220; speed=baseSpeed;
  spawnX=W*1.1; shake=0; deadTimer=0;
}

function startGame(){
  ensureAudio();
  reset();
  state=STATE.PLAY;
  document.getElementById('startScreen').classList.add('hidden');
  document.getElementById('overScreen').classList.add('hidden');
}
document.getElementById('startBtn').addEventListener('click',startGame);
document.getElementById('restartBtn').addEventListener('click',startGame);

// ---------- spawning ----------
const CAR_COLORS=['#e74c3c','#3498db','#9b59b6','#f39c12','#1abc9c','#ecf0f1'];
function spawn(){
  const rm=roadMetrics();
  const laneCount=3;
  const laneH=rm.h/laneCount;
  while(spawnX < car.x + W*1.4){
    const r=Math.random();
    if(r<0.42){ // enemy car in a random lane
      const lane=Math.floor(Math.random()*laneCount);
      entities.push({type:'car', x:spawnX, y:rm.top+laneH*(lane+0.5), w:34,h:62,
        color:CAR_COLORS[Math.floor(Math.random()*CAR_COLORS.length)],
        vx:60+Math.random()*90});
    } else if(r<0.62){ // cone pair or single
      const lane=Math.floor(Math.random()*laneCount);
      const y=rm.top+laneH*(lane+0.5);
      entities.push({type:'cone', x:spawnX, y:y, w:18,h:18});
      if(Math.random()<0.5) entities.push({type:'cone', x:spawnX+46, y:y, w:18,h:18});
    } else if(r<0.75){ // oil slick — slows you
      const lane=Math.floor(Math.random()*laneCount);
      entities.push({type:'oil', x:spawnX, y:rm.top+laneH*(lane+0.5), w:64,h:36});
    } else if(r<0.95){ // coin row
      const lane=Math.floor(Math.random()*laneCount);
      const y=rm.top+laneH*(lane+0.5);
      for(let i=0;i<4;i++) entities.push({type:'coin', x:spawnX+i*38, y:y, w:20,h:20});
    } else { // nothing — breather
    }
    const gap = Math.max(160, 340 - speed*0.35) + Math.random()*220;
    spawnX += gap;
  }
}

// ---------- particles ----------
function explode(x,y){
  for(let i=0;i<40;i++){
    const a=Math.random()*Math.PI*2, s=60+Math.random()*260;
    particles.push({x:x,y:y,vx:Math.cos(a)*s,vy:Math.sin(a)*s,life:.5+Math.random()*.7,max:1.2,
      size:3+Math.random()*6,color:['#ff9f1c','#ffbf69','#e74c3c','#8d99ae'][i%4]});
  }
}
function spark(x,y,color){
  for(let i=0;i<3;i++){
    const a=Math.random()*Math.PI*2,s=40+Math.random()*120;
    particles.push({x:x,y:y,vx:Math.cos(a)*s,vy:Math.sin(a)*s,life:.25,max:.25,size:2,color:color});
  }
}

// ---------- update ----------
function update(dt){
  if(state!==STATE.PLAY && state!==STATE.DEAD) return;

  const rm=roadMetrics();

  if(state===STATE.PLAY){
    // throttle
    const accel = (keys['arrowright']||keys['d']) ? 240 : -60;
    speed += accel*dt;
    speed = Math.max(120, Math.min(baseSpeed+speed*0, baseSpeed + dist*0.05 + (keys['arrowright']||keys['d'])?260:0), 900);
    // natural ramp with distance
    const targetMax = baseSpeed + dist*0.04;
    if(speed>targetMax) speed=Math.max(targetMax,speed-180*dt);

    // steer
    const up=keys['arrowup']||keys['w'], down=keys['arrowdown']||keys['s'];
    car.vy += ((up?-380:0)+(down?380:0))*dt*3;
    car.vy *= Math.pow(0.0018, dt);
    car.y += car.vy*dt;
    car.tilt = Math.max(-0.28,Math.min(0.28,car.vy/900));

    if(car.y - car.h/2 < rm.top+6){ car.y=rm.top+car.h/2+6; car.vy=Math.abs(car.vy)*0.3; sfxScrape(); spark(car.x+17,car.y-car.h/2,'#ffd23f'); }
    if(car.y + car.h/2 > rm.bottom-6){ car.y=rm.bottom-car.h/2-6; car.vy=-Math.abs(car.vy)*0.3; sfxScrape(); spark(car.x+17,car.y+car.h/2,'#ffd23f'); }

    car.x += speed*dt;
    dist += speed*dt;
  } else {
    deadTimer -= dt;
    if(deadTimer<=0 && state===STATE.DEAD) showGameOver();
  }

  // world entities
  for(const e of entities){
    if(e.type==='car') e.x += e.vx*dt;
  }
  // cull behind
  entities = entities.filter(e => e.x > car.x - W*0.35);

  spawn();

  // collisions (only while playing)
  if(state===STATE.PLAY){
    for(const e of entities){
      const hitX = Math.abs(e.x-(car.x+17)) < (e.w+34)/2 - 6;
      const hitY = Math.abs(e.y-car.y) < (e.h+62)/2 - 8;
      if(hitX && hitY){
        if(e.type==='coin'){
          e.dead=true; coins++; sfxCoin();
          spark(e.x,e.y,'#ffd23f');
        } else if(e.type==='oil'){
          e.dead=true; speed=Math.max(120,speed-200); shake=Math.max(shake,6);
          for(let i=0;i<8;i++) spark(car.x+17,car.y,'#555');
          sfxScrape();
        } else {
          // crash
          state=STATE.DEAD; deadTimer=1.4; shake=22;
          explode(car.x+17, car.y);
          sfxCrash();
          break;
        }
      }
    }
    entities = entities.filter(e=>!e.dead);
  }

  // particles
  for(const p of particles){
    p.life-=dt; p.x+=p.vx*dt; p.y+=p.vy*dt; p.vx*=Math.pow(.2,dt); p.vy*=Math.pow(.2,dt);
  }
  particles = particles.filter(p=>p.life>0);

  if(shake>0) shake=Math.max(0,shake-40*dt);
}

function showGameOver(){
  const score = Math.floor(dist/10)+coins*50;
  let newBest=false;
  if(score>best){ best=score; newBest=true; try{ localStorage.setItem('ttdd_best',String(best)); }catch(e){} }
  document.getElementById('finalScore').textContent='SCORE '+score+'   ·   COINS '+coins;
  document.getElementById('bestScore').innerHTML=(newBest?'★ NEW BEST! ★<br>':'')+'BEST: '+best;
  document.getElementById('overScreen').classList.remove('hidden');
}

// ---------- drawing ----------
function drawCar(x,y,w,h,color,tilt,isPlayer){
  ctx.save();
  ctx.translate(x+w/2,y+h/2);
  ctx.rotate(tilt||0);
  // shadow
  ctx.fillStyle='rgba(0,0,0,.3)';
  roundRect(-w/2+3,-h/2+4,w,h,8); ctx.fill();
  // body
  ctx.fillStyle=color;
  roundRect(-w/2,-h/2,w,h,8); ctx.fill();
  ctx.strokeStyle='rgba(0,0,0,.35)'; ctx.lineWidth=2;
  roundRect(-w/2,-h/2,w,h,8); ctx.stroke();
  // windshield & rear window
  ctx.fillStyle='rgba(180,225,255,.9)';
  const dir = isPlayer? -1 : 1; // player faces right (moving +x)
  roundRect(-w/2+5, dir<0? -h/2+8 : h/2-26, w-10, 14, 3); ctx.fill();
  roundRect(-w/2+5, dir<0? h/2-22 : -h/2+8, w-10, 12, 3); ctx.fill();
  // wheels
  ctx.fillStyle='#111';
  ctx.fillRect(-w/2-3,-h/2+6,5,14); ctx.fillRect(w/2-2,-h/2+6,5,14);
  ctx.fillRect(-w/2-3,h/2-20,5,14); ctx.fillRect(w/2-2,h/2-20,5,14);
  // headlights (player faces +x)
  if(isPlayer){
    ctx.fillStyle='#fff7c0';
    ctx.fillRect(w/2-3,-h/2+6,4,8); ctx.fillRect(w/2-3,h/2-14,4,8);
  } else {
    ctx.fillStyle='#ff5b5b';
    ctx.fillRect(-w/2-1,-h/2+6,4,8); ctx.fillRect(-w/2-1,h/2-14,4,8);
  }
  ctx.restore();
}

function roundRect(x,y,w,h,r){
  ctx.beginPath();
  ctx.moveTo(x+r,y);
  ctx.arcTo(x+w,y,x+w,y+h,r);
  ctx.arcTo(x+w,y+h,x,y+h,r);
  ctx.arcTo(x,y+h,x,y,r);
  ctx.arcTo(x,y,x+w,y,r);
  ctx.closePath();
}

function draw(){
  const rm=roadMetrics();
  const camX = car ? car.x : 0;
  const carScreenX = W*0.28 - 17; // player drawn fixed-ish

  ctx.save();
  if(shake>0) ctx.translate((Math.random()-.5)*shake,(Math.random()-.5)*shake);

  // grass
  const g=ctx.createLinearGradient(0,rm.top,0,rm.bottom);
  g.addColorStop(0,'#2e7d32'); g.addColorStop(.5,'#388e3c'); g.addColorStop(1,'#2e7d32');
  ctx.fillStyle='#245c28';
  ctx.fillRect(-20,-20,W+40,H+40);

  // grass texture stripes (scrolling)
  ctx.fillStyle='rgba(0,0,0,.10)';
  const gs=90;
  let off = -(camX%gs);
  for(let x=off;x<W+gs;x+=gs){
    ctx.fillRect(x,-20,gs*0.5,H+40);
  }

  // road
  ctx.fillStyle='#3a3f47';
  ctx.fillRect(-20,rm.top,W+40,rm.h);
  // edge lines
  ctx.fillStyle='#f5f5f5';
  ctx.fillRect(-20,rm.top+4,W+40,4);
  ctx.fillRect(-20,rm.bottom-8,W+40,4);

  // lane dashes
  const lanes=3, laneH=rm.h/lanes;
  ctx.fillStyle='#e8d44d';
  const dash=36, gap=30, period=dash+gap;
  let doff = -((camX)%period);
  for(let l=1;l<lanes;l++){
    const ly=rm.top+laneH*l-4;
    for(let x=doff;x<W+period;x+=period) ctx.fillRect(x,ly,dash,8);
  }

  if(car){
    // entities -> screen coords
    for(const e of entities){
      const sx=e.x-camX+carScreenX;
      if(sx<-120||sx>W+120) continue;
      if(e.type==='car'){
        drawCar(sx,e.y,e.w,e.h,e.color,0,false);
      } else if(e.type==='cone'){
        ctx.fillStyle='rgba(0,0,0,.3)';
        ctx.beginPath(); ctx.ellipse(sx+9,e.y+11,10,5,0,0,Math.PI*2); ctx.fill();
        ctx.fillStyle='#ff7f11';
        ctx.beginPath(); ctx.moveTo(sx+9,e.y-2); ctx.lineTo(sx+17,e.y+12); ctx.lineTo(sx+1,e.y+12); ctx.closePath(); ctx.fill();
        ctx.fillStyle='#fff'; ctx.fillRect(sx+4,e.y+5,10,3);
      } else if(e.type==='oil'){
        ctx.fillStyle='rgba(20,20,25,.75)';
        ctx.beginPath(); ctx.ellipse(sx+e.w/2,e.y,e.w/2,e.h/2,0,0,Math.PI*2); ctx.fill();
        ctx.strokeStyle='rgba(120,80,200,.4)'; ctx.lineWidth=3;
        ctx.beginPath(); ctx.ellipse(sx+e.w/2,e.y,e.w/2-6,e.h/2-5,0,0,Math.PI*2); ctx.stroke();
      } else if(e.type==='coin'){
        const t=performance.now()/180 + e.x;
        const sq=Math.abs(Math.sin(t));
        ctx.fillStyle='rgba(0,0,0,.25)';
        ctx.beginPath(); ctx.ellipse(sx+10,e.y+13,9*sq+2,4,0,0,Math.PI*2); ctx.fill();
        ctx.fillStyle='#ffd23f';
        ctx.beginPath(); ctx.ellipse(sx+10,e.y,9*sq+2,10,0,0,Math.PI*2); ctx.fill();
        ctx.fillStyle='#b8860b';
        ctx.beginPath(); ctx.ellipse(sx+10,e.y,(9*sq+2)*0.55,6,0,0,Math.PI*2); ctx.fill();
      }
    }

    // player
    if(state!==STATE.DEAD || deadTimer>0.9){
      // exhaust puffs when fast
      drawCar(carScreenX,car.y,car.w,car.h,'#ff3b30',car.tilt,true);
    } else {
      // wreck
      ctx.save();
      ctx.translate(carScreenX+17,car.y); ctx.rotate(0.6);
      ctx.fillStyle='#555'; roundRect(-17,-31,34,62,8); ctx.fill();
      ctx.restore();
    }
  }

  // particles (world space)
  for(const p of particles){
    const sx = car ? p.x-camX+carScreenX : p.x;
    ctx.globalAlpha=Math.max(0,p.life/p.max);
    ctx.fillStyle=p.color;
    ctx.fillRect(sx-p.size/2,p.y-p.size/2,p.size,p.size);
  }
  ctx.globalAlpha=1;

  // speed lines at high speed
  if(state===STATE.PLAY && speed>480){
    ctx.strokeStyle='rgba(255,255,255,.12)'; ctx.lineWidth=2;
    for(let i=0;i<6;i++){
      const y=(i*97+ (dist*2)%H )%H;
      const x=W-((dist*3+i*331)%(W*0.5));
      if(y>rm.top&&y<rm.bottom){ ctx.beginPath(); ctx.moveTo(x,y); ctx.lineTo(x+60,y); ctx.stroke(); }
    }
  }

  ctx.restore();

  // HUD
  if(state===STATE.PLAY||state===STATE.DEAD){
    const score = Math.floor(dist/10)+coins*50;
    ctx.font='bold 22px "Courier New",monospace';
    ctx.textAlign='left';
    ctx.fillStyle='rgba(0,0,0,.5)';
    roundRect(14,14,230,86,10); ctx.fill();
    ctx.fillStyle='#fff'; ctx.fillText('SCORE  '+score, 26, 42);
    ctx.fillStyle='#ffd23f'; ctx.fillText('COINS  '+coins, 26, 68);
    ctx.fillStyle='#9fd6ff'; ctx.fillText(Math.round(speed*0.6)+' km/h', 26, 94);

    // speed bar
    ctx.fillStyle='rgba(0,0,0,.5)'; roundRect(W-214,H-38,200,22,8); ctx.fill();
    const pct=Math.min(1,(speed-baseSpeed)/700);
    ctx.fillStyle= pct>0.6?'#e74c3c':pct>0.3?'#ffd23f':'#2ecc71';
    roundRect(W-210,H-34,192*pct,14,6); ctx.fill();
  }
}

// ---------- loop ----------
let last=performance.now();
function frame(now){
  const dt=Math.min(0.05,(now-last)/1000); last=now;
  if(state===STATE.MENU){
    // idle demo: gentle auto-scroll background
    if(!car) reset();
    car.x += 120*dt; dist+=0;
    spawn();
    entities = entities.filter(e=>e.x>car.x-W*0.35);
  } else {
    update(dt);
  }
  draw();
  requestAnimationFrame(frame);
}
reset();
requestAnimationFrame(frame);
})();
