function inRect(px,py,r){ return px>=r.x&&px<=r.x+r.w&&py>=r.y&&py<=r.y+r.h; }
function ballInZone(zones){ for(var i=0;i<zones.length;i++) if(inRect(ball.x,ball.y,zones[i]))return zones[i]; return null; }

function collideWalls(L){
  for(var i=0;i<L.walls.length;i++){
    var w=L.walls[i];
    var nx=Math.max(w.x,Math.min(ball.x,w.x+w.w));
    var ny=Math.max(w.y,Math.min(ball.y,w.y+w.h));
    var dx=ball.x-nx, dy=ball.y-ny;
    var d2=dx*dx+dy*dy;
    if(d2<BALL_R*BALL_R){
      var d=Math.sqrt(d2)||0.001;
      var ux=dx/d, uy=dy/d;
      // push out
      ball.x=nx+ux*(BALL_R+0.1); ball.y=ny+uy*(BALL_R+0.1);
      // reflect
      var dot=ball.vx*ux+ball.vy*uy;
      if(dot<0){
        ball.vx-=(1+0.85)*dot*ux;
        ball.vy-=(1+0.85)*dot*uy;
        if(Math.abs(dot)>60)sWall();
      }
    }
  }
}

function step(dt){
  if(state==='sinking'){
    sinkT+=dt/0.55;
    ball.x+=(cupAt.x-ball.x)*Math.min(1,dt*14);
    ball.y+=(cupAt.y-ball.y)*Math.min(1,dt*14);
    if(sinkT>=1)levelComplete();
    return;
  }
  if(state!=='play')return;

  var L=LEVELS[lvl];
  var sp=Math.hypot(ball.vx,ball.vy);
  if(sp<1)return;

  // sand?
  var inSand=!!ballInZone(L.sand);
  var fr = inSand?5.2:0.85; // exponential decay per second
  ball.vx*=Math.exp(-fr*dt);
  ball.vy*=Math.exp(-fr*dt);

  // substeps for stability
  var steps=Math.max(1,Math.ceil(sp*dt/6));
  for(var s=0;s<steps;s++){
    ball.x+=ball.vx*dt/steps;
    ball.y+=ball.vy*dt/steps;
    collideWalls(L);
    // bounds
    if(ball.x<BALL_R){ball.x=BALL_R;ball.vx=Math.abs(ball.vx)*.85;}
    if(ball.x>W-BALL_R){ball.x=W-BALL_R;ball.vx=-Math.abs(ball.vx)*.85;}
    if(ball.y<BALL_R){ball.y=BALL_R;ball.vy=Math.abs(ball.vy)*.85;}
    if(ball.y>H-BALL_R){ball.y=H-BALL_R;ball.vy=-Math.abs(ball.vy)*.85;}
  }

  // water
  if(ballInZone(L.water)){
    sWater();
    ripples.push({x:ball.x,y:ball.y,t:0});
    strokes[lvl]++; updateHUD();
    ball.x=lastPos.x; ball.y=lastPos.y; ball.vx=0; ball.vy=0;
    return;
  }

  // cup
  var c=L.cup, d=Math.hypot(ball.x-c[0],ball.y-c[1]);
  if(d<CUP_R+6&&d>0.01){
    var pull=(1-d/(CUP_R+6))*900*dt;
    ball.vx+=(c[0]-ball.x)/d*pull;
    ball.vy+=(c[1]-ball.y)/d*pull;
  }
  if(d<CUP_R-2&&Math.hypot(ball.vx,ball.vy)<420){
    state='sinking'; sinkT=0; cupAt={x:c[0],y:c[1]}; ball.vx=0;ball.vy=0;
    return;
  }

  if(Math.hypot(ball.vx,ball.vy)<4){ball.vx=0;ball.vy=0;}
}

    ctx.strokeStyle='#2a75ad'; ctx.lineWidth=3; ctx.stroke();
    ctx.strokeStyle='rgba(255,255,255,.4)'; ctx.lineWidth=2;
    var t=performance.now()/600;
    for(var i=0;i<w.w/40;i++){
      var wx=w.x+15+i*40+Math.sin(t+i)*4;
      ctx.beginPath(); ctx.moveTo(wx,w.y+w.h*.3); ctx.lineTo(wx+18,w.y+w.h*.3);
      ctx.moveTo(wx-8,w.y+w.h*.7); ctx.lineTo(wx+12,w.y+w.h*.7); ctx.stroke();
    }
  });
  // ripples (water splash)
  for(var ri=ripples.length-1;ri>=0;ri--){
    var rp=ripples[ri]; rp.t+=0.03;
    if(rp.t>1){ripples.splice(ri,1);continue;}
    ctx.strokeStyle='rgba(255,255,255,'+(1-rp.t)+')'; ctx.lineWidth=3;
    ctx.beginPath(); ctx.arc(rp.x,rp.y,6+rp.t*26,0,7); ctx.stroke();
  }
  // walls
  L.walls.forEach(function(w){
    roundRect(w.x,w.y,w.w,w.h,4);
    ctx.fillStyle='#5d4037'; ctx.fill();
    ctx.strokeStyle='#3e2723'; ctx.lineWidth=3; ctx.stroke();
    ctx.fillStyle='rgba(255,255,255,.12)'; ctx.fillRect(w.x+3,w.y+3,w.w-6,4);
  });

  // cup
  var c=L.cup;
  ctx.beginPath(); ctx.arc(c[0],c[1],CUP_R,0,7);
  ctx.fillStyle='#111'; ctx.fill();
  ctx.strokeStyle='#000'; ctx.lineWidth=2; ctx.stroke();
  ctx.beginPath(); ctx.arc(c[0],c[1],CUP_R-4,0,7);
  ctx.strokeStyle='rgba(255,255,255,.25)'; ctx.lineWidth=1.5; ctx.stroke();
  // flag
  var sway=Math.sin(performance.now()/400)*3;
  ctx.strokeStyle='#eee'; ctx.lineWidth=3;
  ctx.beginPath(); ctx.moveTo(c[0],c[1]); ctx.lineTo(c[0],c[1]-46); ctx.stroke();
  ctx.fillStyle='#e53935';
  ctx.beginPath();
  ctx.moveTo(c[0],c[1]-46); ctx.lineTo(c[0]+26+sway,c[1]-38); ctx.lineTo(c[0],c[1]-30);
  ctx.closePath(); ctx.fill();

  // tee marker
  var t=L.tee;
  ctx.strokeStyle='rgba(255,255,255,.5)'; ctx.lineWidth=1.5;
  ctx.beginPath(); ctx.arc(t[0],t[1],BALL_R+5,0,7); ctx.stroke();

  // aim arrow (slingshot: opposite of drag)
  if(aiming&&state==='play'){
    var dx=aimX-ball.x, dy=aimY-ball.y, len=Math.hypot(dx,dy);
    if(len>4){
      var pw=Math.min(len,MAX_DRAG)/MAX_DRAG;
      var ax=-dx/len, ay=-dy/len;
      var alen=pw*110+20;
      var ex=ball.x+ax*alen, ey=ball.y+ay*alen;
      ctx.setLineDash([7,6]);
      ctx.strokeStyle='rgba(255,255,255,.85)'; ctx.lineWidth=3;
      ctx.beginPath(); ctx.moveTo(ball.x,ball.y); ctx.lineTo(ex,ey); ctx.stroke();
      ctx.setLineDash([]);
      // arrowhead
      var ang=Math.atan2(ay,ax);
      ctx.fillStyle='rgba(255,255,255,.9)';
      ctx.beginPath();
      ctx.moveTo(ex+Math.cos(ang)*14,ey+Math.sin(ang)*14);
      ctx.lineTo(ex+Math.cos(ang+2.5)*10,ey+Math.sin(ang+2.5)*10);
      ctx.lineTo(ex+Math.cos(ang-2.5)*10,ey+Math.sin(ang-2.5)*10);
      ctx.closePath(); ctx.fill();
      // power bar
      var ph=pw<0.4?'#66bb6a':pw<0.75?'#ffb300':'#e53935';
      ctx.fillStyle='rgba(0,0,0,.4)'; roundRect(ball.x-32,ball.y+18,64,8,4); ctx.fill();
      ctx.fillStyle=ph; roundRect(ball.x-30,ball.y+20,60*pw,4,2); ctx.fill();
    }
  }

  // ball (with sink shrink)
  var br=BALL_R*(state==='sinking'?Math.max(0,1-sinkT):1);
  if(br>0.5){
    ctx.fillStyle='rgba(0,0,0,.28)';
    ctx.beginPath(); ctx.ellipse(ball.x+2,ball.y+3,br,br*.6,0,0,7); ctx.fill();
    var g=ctx.createRadialGradient(ball.x-3,ball.y-3,1,ball.x,ball.y,br);
    g.addColorStop(0,'#ffffff'); g.addColorStop(1,'#c9c9c9');
    ctx.fillStyle=g;
    ctx.beginPath(); ctx.arc(ball.x,ball.y,br,0,7); ctx.fill();
  }

  // in-water penalty flash text
  if(state==='play'&&strokes[lvl]>0){}
}

function roundRect(x,y,w,h,r){
  r=Math.min(r,w/2,h/2);
  ctx.beginPath();
  ctx.moveTo(x+r,y);
  ctx.arcTo(x+w,y,x+w,y+h,r);
  ctx.arcTo(x+w,y+h,x,y+h,r);
  ctx.arcTo(x,y+h,x,y,r);
  ctx.arcTo(x,y,x+w,y,r);
  ctx.closePath();
}
