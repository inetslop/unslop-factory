function beep(freq,dur,type,vol,slide){
  if(!AC)return;
  var o=AC.createOscillator(),g=AC.createGain();
  o.type=type||'square'; o.frequency.value=freq;
  if(slide)o.frequency.exponentialRampToValueAtTime(Math.max(40,freq+slide),AC.currentTime+dur);
  g.gain.setValueAtTime(vol||0.12,AC.currentTime);
  g.gain.exponentialRampToValueAtTime(0.001,AC.currentTime+dur);
  o.connect(g); g.connect(AC.destination);
  o.start(); o.stop(AC.currentTime+dur+0.02);
}

/* ---------- grid logic ---------- */
function randT(){ return Math.floor(Math.random()*TYPES); }

function makeGrid(){
  grid=[];
  for(var r=0;r<SIZE;r++){ grid[r]=[];
    for(var c=0;c<SIZE;c++){
      var t;
      do{ t=randT(); }while(
        (c>=2 && grid[r][c-1]&&grid[r][c-2]&&grid[r][c-1].t===t&&grid[r][c-2].t===t)||
        (r>=2 && grid[r-1][c]&&grid[r-2][c]&&grid[r-1][c].t===t&&grid[r-2][c].t===t));
      grid[r][c]={t:t,dy:-(Math.random()*2+0.5)};
    }
  }
}

function findMatches(){
  var marks={};
  function mark(r,c){ marks[r+','+c]=true; }
  // rows
  for(var r=0;r<SIZE;r++){
    var run=1;
    for(var c=1;c<=SIZE;c++){
      var same=c<SIZE && grid[r][c]&&grid[r][c-1]&&grid[r][c].t===grid[r][c-1].t;
      if(same){run++;}
      else{ if(run>=3)for(var k=c-run;k<c;k++)mark(r,k); run=1; }
    }
  }
  // cols
  for(var c2=0;c2<SIZE;c2++){
    var run2=1;
    for(var r2=1;r2<=SIZE;r2++){
      var same2=r2<SIZE && grid[r2][c2]&&grid[r2-1][c2]&&grid[r2][c2].t===grid[r2-1][c2].t;
      if(same2){run2++;}
      else{ if(run2>=3)for(var k2=r2-run2;k2<r2;k2++)mark(k2,c2); run2=1; }
    }
  }
  return marks;
}

function countRuns(marks){
  // count groups of >=4 for bonus: simple — total gems in marked set, and check big runs
  var r=0,big=0;
  while(r<SIZE){
    var run=1;
    for(var c=1;c<=SIZE;c++){
      var same=c<SIZE && marks[r+','+c]&&marks[r+','+(c-1)];
      if(same)run++; else { if(run>=4)big++; run=1; }
    }
    r++;
  }
  r=0;
  while(r<SIZE){
    var run2=1;
    for(var c2=0;c2<SIZE;c2++){
      var same2=(r+1)<SIZE && marks[(r+1)+','+c2]&&marks[r+','+c2];
      if(same2)run2++; else { if(run2>=4)big++; run2=1; }
    }
    r++;
  }
  return big;
}

function hasAnyMove(){
  function test(r1,c1,r2,c2){
    var a=grid[r1][c1],b=grid[r2][c2]; if(!a||!b)return false;
    grid[r1][c1]=b; grid[r2][c2]=a;
    var m=findMatches();
    grid[r1][c1]=a; grid[r2][c2]=b;
    for(var k in m)return true;
    return false;
  }
  for(var r=0;r<SIZE;r++)for(var c=0;c<SIZE;c++){
    if(c<SIZE-1&&test(r,c,r,c+1))return true;
    if(r<SIZE-1&&test(r,c,r+1,c))return true;
  }
  return false;
}

function removeAndDrop(marks){
  var total=0;
  for(var r=0;r<SIZE;r++)for(var c=0;c<SIZE;c++){
    if(marks[r+','+c]){
      total++; gemsMatched++;
      spawnBurst(r,c,grid[r][c].t);
      grid[r][c]=null;
    }
  }
  // gravity per column
  for(var c2=0;c2<SIZE;c2++){
    var stack=[];
    for(var r2=SIZE-1;r2>=0;r2--) if(grid[r2][c2])stack.push({g:grid[r2][c2],src:r2});
    for(var d=0;d<SIZE;d++){
      var row=SIZE-1-d;
      if(d<stack.length){
        var item=stack[d];
        grid[row][c2]=item.g;
        item.g.dy=item.src-row; // negative offset (in cells) to animate fall
      }else{
        grid[row][c2]={t:randT(),dy:-(row+1)};
      }
    }
  }
  return total;
}

function trySwap(r1,c1,r2,c2){
  if(phase!=='idle'||state!=='play')return;
  if(Math.abs(r1-r2)+Math.abs(c1-c2)!==1)return;
  var a=grid[r1][c1],b=grid[r2][c2]; if(!a||!b)return;
  grid[r1][c1]=b; grid[r2][c2]=a;
  var m=findMatches();
  if(m&&Object.keys(m).length>0){
    sfxSwap();
    combo=1;
    doResolve(m);
  }else{
    grid[r1][c1]=a; grid[r2][c2]=b;
    shakeCells.push({r:r1,c:c1,t:0.3},{r:r2,c:c2,t:0.3});
    sfxBad();
  }
}

function doResolve(marks){

  var big=countRuns(marks);
  var distGain=total*0.014*(1+0.25*(combo-1));
  carP=Math.min(1,carP+distGain);
  score+=total*10*combo;
  if(big>0){ timeLeft=Math.min(totalTime+15,timeLeft+big*2.5); addFloat('+'+(big*2.5).toFixed(1)+'s BONUS!','#ffd54a'); }
  if(combo>1)addFloat('COMBO x'+combo,'#7ee8fa');
  sfxMatch(combo);
  phase='settling';
}

  }
}

  if(phase==='settling'){

    for(var r=0;r<SIZE;r++)for(var c=0;c<SIZE;c++){
      var g=grid[r][c];
      if(g&&g.dy<0){
        g.dy=Math.min(0,g.dy+dt*9);
        if(g.dy<0)settled=false;
      }
    }
    if(settled){
      var m=findMatches();
      if(m&&Object.keys(m).length>0){ combo++; doResolve(m); }
      else{ phase='idle'; combo=0; if(!hasAnyMove()){ makeGrid(); addFloat('RESHUFFLE!','#b45eff'); phase='settling'; } }
    }
  }

  // timer
  timeLeft-=dt;
  if(timeLeft<=0){ timeLeft=0; endGame(false); return; }
  if(carP>=1){ carP=1; endGame(true); return; }

  // trucks
  truckTimer-=dt;
  if(truckTimer<=0&&trucks.length<3){
    trucks.push({p:carP+0.25+Math.random()*0.6,v:0.028+Math.random()*0.02});
    truckTimer=7+Math.random()*4;
  }
  for(var i=trucks.length-1;i>=0;i--){
    var t=trucks[i];
    t.p-=t.v*dt;
    if(t.p<=carP+0.012){
      // collision
      trucks.splice(i,1);
      carP=Math.max(0,carP-0.045);
      timeLeft=Math.max(0,timeLeft-0.8);
      hitFlash=0.5; sfxHit(); addFloat('HIT! -time','#ff5e7e');
    }else if(t.p<carP-0.1){ trucks.splice(i,1); }
  }

  // fx
  for(var j=particles.length-1;j>=0;j--){
    var p=particles[j];
    p.life-=dt; p.x+=p.vx*dt; p.y+=p.vy*dt; p.vy+=300*dt;
    if(p.life<=0)particles.splice(j,1);
  }
  for(var f=floats.length-1;f>=0;f--){ floats[f].life-=dt; floats[f].y-=40*dt; if(floats[f].life<=0)floats.splice(f,1); }
  for(var s=shakeCells.length-1;s>=0;s--){ shakeCells[s].t-=dt; if(shakeCells[s].t<=0)shakeCells.splice(s,1); }
  hitFlash=Math.max(0,hitFlash-dt);
  carBob+=dt*8;
}

/* ---------- draw ---------- */
function drawGem(t,x,y,size,scale){
  var col=GEM_COLORS[t], shape=GEM_SHAPES[t];
  ctx.save();
  ctx.translate(x,y);
  ctx.scale(scale||1,scale||1);
  var s=size*0.36;
  // shadow
  ctx.fillStyle='rgba(0,0,0,0.28)';
  ctx.beginPath();ctx.arc(0,s*0.45,s*1.05,0,Math.PI*2);ctx.fill();
  ctx.fillStyle=col;
  ctx.strokeStyle='rgba(255,255,255,0.65)';
  ctx.lineWidth=Math.max(1.5,size*0.04);
  ctx.beginPath();
  if(shape==='circle'){ ctx.arc(0,0,s,0,Math.PI*2); }
  else if(shape==='square'){ var q=s*0.85; ctx.rect(-q,-q,q*2,q*2); }
  else if(shape==='diamond'){ ctx.moveTo(0,-s*1.15);ctx.lineTo(s*1.05,0);ctx.lineTo(0,s*1.15);ctx.lineTo(-s*1.05,0);ctx.closePath(); }
  else if(shape==='triangle'){ ctx.moveTo(0,-s*1.1);ctx.lineTo(s*1.05,s*0.85);ctx.lineTo(-s*1.05,s*0.85);ctx.closePath(); }
  else { for(var i=0;i<6;i++){var a=Math.PI/6+i*Math.PI/3;var px=Math.cos(a)*s*1.05,py=Math.sin(a)*s*1.05;i?ctx.lineTo(px,py):ctx.moveTo(px,py);} ctx.closePath(); }
  ctx.fill(); ctx.stroke();
  // shine
  ctx.fillStyle='rgba(255,255,255,0.4)';
  ctx.beginPath();ctx.arc(-s*0.35,-s*0.4,s*0.28,0,Math.PI*2);ctx.fill();
  ctx.restore();
}

function draw(){
  var L=layout();
  ctx.setTransform(DPR,0,0,DPR,0,0);
  // bg
  var bg=ctx.createLinearGradient(0,0,0,H);
  bg.addColorStop(0,'#141b3a'); bg.addColorStop(1,'#0b1020');
  ctx.fillStyle=bg; ctx.fillRect(0,0,W,H);

  if(state==='start'){ return; }

  /* road */
  var roadH=L.roadH, skyEnd=roadH*0.55;
  var sky=ctx.createLinearGradient(0,0,0,skyEnd);
  sky.addColorStop(0,'#1c2a6b'); sky.addColorStop(1,'#3a4fb0');
  ctx.fillStyle=sky; ctx.fillRect(0,0,W,skyEnd);
  // stars
  ctx.fillStyle='rgba(255,255,255,0.7)';
  for(var i=0;i<40;i++){
    var sx=(i*137.5)%W, sy=(i*61.3)%(skyEnd*0.9);
    ctx.globalAlpha=0.3+((i*31)%10)/14; ctx.fillRect(sx,sy,2,2);
  }
  ctx.globalAlpha=1;
  // road surface
  var roadTop=skyEnd;
  ctx.fillStyle='#2b2f3a'; ctx.fillRect(0,roadTop,W,roadH-roadTop);
  // edges
  ctx.fillStyle='#e8e8e8'; ctx.fillRect(0,roadTop,W,4);
  ctx.fillRect(0,roadH-6,W,4);
  // lane dashes (scroll with car for speed feel)
  var scroll=(carP*W)%60;
  ctx.fillStyle='rgba(255,255,255,0.5)';
  var laneY=roadTop+(roadH-roadTop)*0.5;
  for(var dx=-60;dx<W+60;dx+=60){ ctx.fillRect(dx-scroll,laneY-3,34,6); }
  // finish line
  var fx=W*0.965;
  for(var fr=0;fr<Math.floor((roadH-roadTop)/12);fr++)for(var fc=0;fc<2;fc++){
    ctx.fillStyle=((fr+fc)%2)?'#fff':'#111';
    ctx.fillRect(fx+fc*7,roadTop+6+fr*12,7,12);
  }

  /* trucks */
  for(var ti=0;ti<trucks.length;ti++){
    var tp=L.gx-40+trucks[ti].p*(fx-(L.gx-40));
    drawTruck(tp,laneY-16,Math.max(28,(roadH-roadTop)*0.3));
  }

  /* car */
  var cx=L.gx-40+carP*(fx-(L.gx-40));
  var cy=laneY+(roadH-roadTop)*0.22+Math.sin(carBob)*1.5;
  drawCar(cx,cy,Math.max(34,(roadH-roadTop)*0.42),hitFlash);

  /* HUD on road */
  ctx.textAlign='left'; ctx.textBaseline='top';
  ctx.fillStyle='#fff'; ctx.font='bold '+Math.max(16,roadH*0.19)+'px Segoe UI,Arial';
  ctx.fillText('TIME '+timeLeft.toFixed(1),14,8);
  ctx.textAlign='right';
  ctx.fillText('SCORE '+score,W-14,8);
  ctx.textAlign='center';
  ctx.fillStyle=timeLeft<10?'#ff5e7e':'#ffd54a';
  ctx.fillText((carP*100).toFixed(0)+'% TO FINISH',W/2,8);

  /* distance bar */
  var barY=roadH-2;
  ctx.fillStyle='rgba(255,255,255,0.15)';ctx.fillRect(0,barY,W,4);
  ctx.fillStyle='#58ff9e';ctx.fillRect(0,barY,W*carP,4);

  /* grid */
  // board bg
  ctx.fillStyle='rgba(255,255,255,0.06)';
  roundRect(L.gx-8,L.gy-8,L.cell*SIZE+16,L.cell*SIZE+16,14);ctx.fill();

  var shakeMap={};
  for(var si=0;si<shakeCells.length;si++)shakeMap[shakeCells[si].r+','+shakeCells[si].c]=shakeCells[si].t;

  for(var r=0;r<SIZE;r++)for(var c=0;c<SIZE;c++){
    var cellX=L.gx+c*L.cell, cellY=L.gy+r*L.cell;
    // checker
    ctx.fillStyle=(r+c)%2?'rgba(255,255,255,0.05)':'rgba(255,255,255,0.09)';
    roundRect(cellX+2,cellY+2,L.cell-4,L.cell-4,8);ctx.fill();

    var g=grid[r]&&grid[r][c];
    if(g){
      var off=g.dy*L.cell;
      var sh=shakeMap[r+','+c];
      var sx=sh?Math.sin(sh*50)*4:0;
      drawGem(g.t,cellX+L.cell/2+sx,cellY+L.cell/2+off,L.cell,1);
    }
  }

  // selection & cursor
  if(sel){
    ctx.strokeStyle='#ffd54a'; ctx.lineWidth=4;
    roundRect(L.gx+sel.c*L.cell+3,L.gy+sel.r*L.cell+3,L.cell-6,L.cell-6,9);ctx.stroke();
  }
  ctx.strokeStyle='rgba(126,232,250,'+(0.6+0.4*Math.sin(Date.now()/200))+')';
  ctx.lineWidth=3;
  roundRect(L.gx+cursor.c*L.cell+2,L.gy+cursor.r*L.cell+2,L.cell-4,L.cell-4,9);ctx.stroke();

  // particles
  for(var pi=0;pi<particles.length;pi++){
    var p=particles[pi];
    ctx.globalAlpha=Math.max(0,p.life/p.max);
    ctx.fillStyle=p.color;
    ctx.fillRect(p.x-3,p.y-3,6,6);
  }
  ctx.globalAlpha=1;

  // floating texts
  ctx.textAlign='center';
  for(var fi=0;fi<floats.length;fi++){
    var fl=floats[fi];
    ctx.globalAlpha=Math.min(1,fl.life);
    ctx.fillStyle=fl.color;
    ctx.font='bold '+Math.max(20,L.cell*0.4)+'px Segoe UI,Arial';
    ctx.fillText(fl.text,W/2,L.gy+L.cell*0.8+fl.y);
  }
  ctx.globalAlpha=1;

  // hit flash overlay
  if(hitFlash>0){
    ctx.fillStyle='rgba(255,60,90,'+(hitFlash*0.35)+')';
    ctx.fillRect(0,0,W,H);
  }

  /* win confetti */
  if(state==='win'){
    for(var ci=0;ci<60;ci++){
      var ctpx=(ci*97+Date.now()*0.1*(1+(ci%5)*0.3))%W;
      var ctpy=(ci*211+Date.now()*0.18)%H;
      ctx.fillStyle=GEM_COLORS[ci%TYPES];
      ctx.globalAlpha=0.8;
      ctx.fillRect(ctpx,ctpy,7,7);
    }
    ctx.globalAlpha=1;
  }
}

function roundRect(x,y,w,h,r){
  ctx.beginPath();
  ctx.moveTo(x+r,y);
  ctx.arcTo(x+w,y,x+w,y+h,r);
  ctx.arcTo(x+w,y+h,x,y+h,r);
  ctx.arcTo(x,y+h,x,y,r);
  ctx.arcTo(x,y,x+w,y,r);
  ctx.closePath();
}

function drawCar(x,y,w,hit){
  var h=w*0.5;
  ctx.save();ctx.translate(x,y);
  if(hit)ctx.rotate(Math.sin(Date.now()/30)*0.1);
  // wheels
  ctx.fillStyle='#111';
  ctx.beginPath();ctx.arc(-w*0.28,h*0.42,w*0.16,0,Math.PI*2);ctx.arc(w*0.28,h*0.42,w*0.16,0,Math.PI*2);ctx.fill();
  // body
  ctx.fillStyle='#ff5e7e';
  roundRect(-w/2,-h*0.25,w,h*0.9,6);ctx.fill();
  // cabin
  ctx.fillStyle='#ffd54a';
  roundRect(-w*0.18,-h*0.75,w*0.55,h*0.6,5);ctx.fill();
  // window
  ctx.fillStyle='#bfe9ff';
  roundRect(-w*0.12,-h*0.65,w*0.4,h*0.4,4);ctx.fill();
  // stripe
  ctx.fillStyle='#fff';ctx.fillRect(-w/2,-h*0.05,w,h*0.12);
  ctx.restore();
}

function drawTruck(x,y,h){

  ctx.save();ctx.translate(x,y);
  ctx.fillStyle='#111';
  ctx.beginPath();ctx.arc(-w*0.3,h*0.42,w*0.12,0,Math.PI*2);ctx.arc(w*0.3,h*0.42,w*0.12,0,Math.PI*2);ctx.fill();
