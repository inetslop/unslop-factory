  ctx.moveTo(0,6*CELL+30);ctx.lineTo(W,6*CELL+30);ctx.stroke();ctx.setLineDash([]);

  /* river waves */
  ctx.strokeStyle='rgba(120,190,255,.35)';ctx.lineWidth=2;
  for(i=0;i<8;i++){var wy=7*CELL+14+(i%3)*16,wx=((i*97+time*22*(i%2?1:-1))%(W+40))-20;
    ctx.beginPath();ctx.moveTo(wx,wy);ctx.quadraticCurveTo(wx+15,wy-5,wx+30,wy);ctx.stroke();}

  /* cars */
  for(i=0;i<cars.length;i++){var c=cars[i];
    ctx.fillStyle='#111';ctx.fillRect(c.x+6,c.y-3,c.w-12,c.h+6);          // tires
    ctx.fillStyle=c.c;ctx.fillRect(c.x,c.y,c.w,c.h);
    ctx.fillStyle='rgba(20,40,70,.85)';
    if(c.v<0)ctx.fillRect(c.x+8,c.y+5,16,c.h-10);else ctx.fillRect(c.x+c.w-24,c.y+5,16,c.h-10);
    ctx.fillStyle='rgba(255,255,255,.25)';ctx.fillRect(c.x,c.y,c.w,5);}

  /* lizards */
  for(i=0;i<lizards.length;i++){var l=lizards[i];
    ctx.fillStyle='#3fae4c';ctx.fillRect(l.x,l.y+4,l.w-10,10);
    ctx.beginPath();ctx.moveTo(l.x+l.w-10,l.y+9);
    ctx.lineTo(l.x+(l.v>0?l.w:-8),l.y+2);ctx.lineTo(l.x+(l.v>0?l.w:-8),l.y+16);ctx.closePath();ctx.fill();
    ctx.fillStyle='#1f7a30';
    for(var k=0;k<4;k++)ctx.fillRect(l.x+4+k*9,l.y+(k%2?0:14),5,4);}

  /* boats */
  for(i=0;i<boats.length;i++){var b=boats[i];
    ctx.fillStyle='#8b6f4e';ctx.fillRect(b.x,b.y,b.w,b.h);
    ctx.fillStyle='#5c4630';ctx.beginPath();
    if(b.v<0){ctx.moveTo(b.x,b.y+b.h/2);ctx.lineTo(b.x-12,b.y-3);ctx.lineTo(b.x-12,b.y+b.h+3);}
    else{ctx.moveTo(b.x+b.w,b.y+b.h/2);ctx.lineTo(b.x+b.w+12,b.y-3);ctx.lineTo(b.x+b.w+12,b.y+b.h+3);}
    ctx.closePath();ctx.fill();
    ctx.fillStyle='#3a2c1c';ctx.fillRect(b.x+b.w/2-10,b.y+4,20,b.h-8);}

  /* invaders */
  var icol='hsl('+((wave*67)%360)+' 90% 65%)',dark='hsl('+((wave*67)%360)+' 90% 30%)';
  if(state==='play'||state==='dying'||state==='wave'){
    for(i=0;i<inv.list.length;i++){var o=inv.list[i];
      var ix=120+o.c*CELL+inv.gx+(CELL-44)/2, iy=(1+o.r)*CELL+inv.gy+(CELL-32)/2;
      iy+=Math.sin(time*5+o.c*0.9+o.r)*2;
      drawMap(INV,ix,iy,4,{X:icol});}
  }

  /* bolts */
  ctx.fillStyle='#7dfcff';
  for(i=0;i<bolts.length;i++){var bl=bolts[i];
    ctx.globalAlpha=0.35;ctx.fillRect(bl.x-5,bl.y-12,10,24);ctx.globalAlpha=1;
    ctx.fillRect(bl.x-2,bl.y-10,4,20);}

  /* particles */
  for(i=0;i<parts.length;i++){var p=parts[i];
    ctx.globalAlpha=Math.max(0,p.life*1.6);ctx.fillStyle=p.col;
    ctx.fillRect(p.x-2,p.y-2,4,4);}
  ctx.globalAlpha=1;

  /* frog */
  if(state==='play'||state==='wave'){
    var blink=frog.invuln>0&&Math.floor(time*12)%2===0;
    if(!blink)drawMap(FROG,frog.cx-22,frog.cy-16,4,{G:'#43e06b',E:'#0a2f16',D:'#1fa84c'});
  }

  /* pause */
  if(paused){ctx.fillStyle='rgba(3,5,14,.7)';ctx.fillRect(0,0,W,H);
    ctx.fillStyle='#fff';ctx.font='bold 40px Courier New';ctx.textAlign='center';
    ctx.fillText('PAUSED',W/2,H/2);}

  if(state==='start'){/* dim handled by overlay */}
}

/* ---------- loop ---------- */
function loop(ts){
  requestAnimationFrame(loop);
  var dt=Math.min(0.033,(ts-last)/1000||0.016);last=ts;
  if(!paused)update(dt);
  render();
}
requestAnimationFrame(loop);
