function buildWave(n){
  const count=8+n*3, hpMul=1+0.24*(n-1);
  const interval=Math.max(0.35, 0.9-n*0.03);
  spawnQueue=[];

    if(n%5===0 && i===count-1) type="boss";

function spawnEnemy(type,hpMul){
  const d=ENEMY_DEFS[type];
  const x=WP[0][0]*CELL+CELL/2, y=WP[0][1]*CELL+CELL/2;
  enemies.push({
    type,x,y,wp:1,hp:d.hp*hpMul,maxhp:d.hp*hpMul,speed:d.speed,slowT:0,
    r:d.r,color:d.color,reward:d.reward,dmg:d.dmg,travel:0,dead:false,wob:Math.random()*6.28
  });
}
function updateEnemies(dt){
  for(const e of enemies){
    if(e.dead) continue;
    const sp=e.speed*(e.slowT>0?0.45:1);
    if(e.slowT>0) e.slowT-=dt;
    let move=sp*dt;
    while(move>0 && e.wp<WP.length){
      const tx=WP[e.wp][0]*CELL+CELL/2, ty=WP[e.wp][1]*CELL+CELL/2;
      const dx=tx-e.x, dy=ty-e.y, dist=Math.hypot(dx,dy);
      if(dist<=move){ e.x=tx; e.y=ty; e.travel+=dist; move-=dist; e.wp++; }
      else { e.x+=dx/dist*move; e.y+=dy/dist*move; e.travel+=move; move=0; }
    }
    e.wob+=dt*8;
    if(e.wp>=WP.length){ // reached exit
      e.dead=true; lives-=e.dmg; sfx.lose();
      for(let i=0;i<8;i++) particles.push(particle(e.x,e.y,e.color,120));
      if(lives<=0){ lives=0; updateHUD(); gameOver(); }
    }
  }
  enemies=enemies.filter(e=>!e.dead);
}

function sellVal(t){ return Math.round(t.invested*0.7); }
function tStat(t){
  const d=TOWER_DEFS[t.idx], m=Math.pow(1.6,t.level-1), r=Math.pow(1.15,t.level-1);
  return {dmg:Math.round(d.dmg*m), range:Math.round(d.range*r), rate:+(d.rate*Math.pow(1.12,t.level-1)).toFixed(2)};
}
function updateTowers(dt){
  for(const t of towers){
    t.cd-=dt;
    const s=tStat(t);
    // find target: in range, most progressed
    let best=null,bestTr=-1;
    for(const e of enemies){
      if(e.dead) continue;
      const d=Math.hypot(e.x-t.x,e.y-t.y);
      if(d<=s.range && e.travel>bestTr){ best=e; bestTr=e.travel; }
    }
    if(best){
      t.angle=Math.atan2(best.y-t.y,best.x-t.x);
      if(t.cd<=0){
        t.cd=1/s.rate;
        bullets.push({x:t.x,y:t.y,tgt:best,sp:t.idx===1?520:380,dmg:s.dmg,type:t.idx});
        sfx.shot(t.idx);
      }
    }
  }
}
function updateBullets(dt){
  for(const b of bullets){
    if(b.tgt && !b.tgt.dead){ b.lx=b.tgt.x; b.ly=b.tgt.y; }
    else { b.done=true; continue; }
    const dx=b.lx-b.x, dy=b.ly-b.y, d=Math.hypot(dx,dy), step=b.sp*dt;
    if(d<=step+ (b.tgt.r||8)){
      b.done=true; hitEnemy(b.tgt,b);
    } else { b.x+=dx/d*step; b.y+=dy/d*step; }
  }
  bullets=bullets.filter(b=>!b.done);
}
function hitEnemy(e,b){
  if(e.dead) return;
  e.hp-=b.dmg;
  if(b.type===2){ e.slowT=1.6; for(let i=0;i<4;i++) particles.push(particle(e.x,e.y,"#9ff5e8",90)); }
  else { for(let i=0;i<3;i++) particles.push(particle(e.x,e.y,b.type===1?"#ffd27f":"#cfe6ff",110)); }
  if(e.hp<=0){
    e.dead=true; gold+=e.reward; sfx.coin();
    if(e.type==="boss") sfx.boom();
    for(let i=0;i<(e.type==="boss"?26:10);i++) particles.push(particle(e.x,e.y,e.color,150));
    updateHUD();
  }
}
function particle(x,y,c,sp){
  const a=Math.random()*Math.PI*2, s=sp*(0.3+Math.random()*0.7);
  return {x,y,vx:Math.cos(a)*s,vy:Math.sin(a)*s,life:0.4+Math.random()*0.3,max:0.7,c};
}

/* ---------- loop ---------- */
let lastT=0;
function frame(ts){
  requestAnimationFrame(frame);
  const dt=Math.min(0.05,(ts-lastT)/1000||0); lastT=ts;
  if(state==="playing"){
    // spawning
    if(waveActive && spawnQueue.length){
      spawnTimer-=dt;
      if(spawnTimer<=0){
        const s=spawnQueue.shift();
        spawnEnemy(s.type,s.hpMul);
        spawnTimer=spawnQueue.interval||0.8;
      }
    }
    updateEnemies(dt);
    updateTowers(dt);
    updateBullets(dt);
    for(const p of particles){ p.x+=p.vx*dt; p.y+=p.vy*dt; p.vx*=0.92; p.vy*=0.92; p.life-=dt; }
    particles=particles.filter(p=>p.life>0);
    if(waveActive && !spawnQueue.length && !enemies.length){
      waveActive=false;
      const bonus=30+wave*5; gold+=bonus; sfx.coin();
      updateHUD(); refreshPanel();
    }
  }
  draw();
}

/* ---------- drawing ---------- */
function draw(){
  const w=cv.width/dpr, h=cv.height/dpr;
  ctx.setTransform(dpr,0,0,dpr,0,0);
  ctx.fillStyle="#12141c"; ctx.fillRect(0,0,w,h);
  ctx.setTransform(dpr*scale,0,0,dpr*scale,dpr*ox,dpr*oy);
  ctx.drawImage(bgCanvas,0,0);

  // ghost / hover
  if(placeIdx>=0 && hoverCell && state==="playing"){
    const [cx,cy]=hoverCell;
    const ok=!pathCells.has(cx+","+cy) && !towerAt(cx,cy) && gold>=TOWER_DEFS[placeIdx].cost;
    const x=cx*CELL+CELL/2, y=cy*CELL+CELL/2;
    ctx.globalAlpha=0.35;
    ctx.fillStyle=ok?TOWER_DEFS[placeIdx].color:"#ff4444";
    ctx.beginPath(); ctx.arc(x,y,TOWER_DEFS[placeIdx].range,0,7); ctx.fill();
    ctx.globalAlpha=0.9;
    ctx.fillStyle=ok?TOWER_DEFS[placeIdx].color:"#ff4444";
    ctx.beginPath(); ctx.arc(x,y,16,0,7); ctx.fill();
    ctx.globalAlpha=1;
  }

  // selected tower range
  if(selTower){
    const s=tStat(selTower);
    ctx.globalAlpha=0.25; ctx.fillStyle="#ffffff";
    ctx.beginPath(); ctx.arc(selTower.x,selTower.y,s.range,0,7); ctx.fill();
    ctx.globalAlpha=1;
    ctx.strokeStyle="#ffd44d"; ctx.lineWidth=2;
    ctx.strokeRect(selTower.cx*CELL+3,selTower.cy*CELL+3,CELL-6,CELL-6);
  }

  // towers
  for(const t of towers){
    const d=TOWER_DEFS[t.idx];
    ctx.fillStyle="#1c2030"; ctx.beginPath(); ctx.arc(t.x,t.y,17,0,7); ctx.fill();
    ctx.save(); ctx.translate(t.x,t.y); ctx.rotate(t.angle);
    ctx.fillStyle=d.barrel; ctx.fillRect(4,-4,18+t.level*2,8); // barrel
    ctx.restore();
    ctx.fillStyle=d.color; ctx.beginPath(); ctx.arc(t.x,t.y,13,0,7); ctx.fill();
    ctx.strokeStyle="rgba(0,0,0,.35)"; ctx.lineWidth=2; ctx.stroke();
    if(t.type===undefined) {} // noop
    // level pips
    for(let i=0;i<t.level;i++){
      ctx.fillStyle="#ffd44d";
      ctx.beginPath(); ctx.arc(t.x-8+i*8, t.y+22, 3, 0, 7); ctx.fill();
    }
  }

  // bullets
  for(const b of bullets){
    if(b.type===1){ ctx.fillStyle="#ffd27f"; ctx.beginPath(); ctx.arc(b.x,b.y,5,0,7); ctx.fill(); }
    else if(b.type===2){ ctx.fillStyle="#9ff5e8"; ctx.beginPath(); ctx.arc(b.x,b.y,4,0,7); ctx.fill(); }
    else { ctx.fillStyle="#cfe6ff"; ctx.beginPath(); ctx.arc(b.x,b.y,3.5,0,7); ctx.fill(); }
  }

  // enemies
  for(const e of enemies){
    const wob=Math.sin(e.wob)*1.5;
    ctx.fillStyle="rgba(0,0,0,.25)";
    ctx.beginPath(); ctx.ellipse(e.x,e.y+e.r*0.75,e.r*0.9,e.r*0.4,0,0,7); ctx.fill();
    ctx.fillStyle=e.slowT>0?"#8fd8ff":e.color;
    ctx.beginPath(); ctx.arc(e.x,e.y+wob,e.r,0,7); ctx.fill();
    ctx.strokeStyle="rgba(0,0,0,.4)"; ctx.lineWidth=2; ctx.stroke();
    // eyes toward path
    const tx=e.wp<WP.length?WP[e.wp][0]*CELL+CELL/2:e.x, ty=e.wp<WP.length?WP[e.wp][1]*CELL+CELL/2:e.y;
    const a=Math.atan2(ty-e.y,tx-e.x);
    ctx.fillStyle="#fff";
    ctx.beginPath(); ctx.arc(e.x+Math.cos(a-0.5)*e.r*0.45,e.y+wob+Math.sin(a-0.5)*e.r*0.45,e.r*0.22,0,7); ctx.fill();
    ctx.beginPath(); ctx.arc(e.x+Math.cos(a+0.5)*e.r*0.45,e.y+wob+Math.sin(a+0.5)*e.r*0.45,e.r*0.22,0,7); ctx.fill();
    // hp bar
    const w=e.r*2, pct=Math.max(0,e.hp/e.maxhp);
    ctx.fillStyle="rgba(0,0,0,.6)"; ctx.fillRect(e.x-w/2,e.y-e.r-9,w,5);
    ctx.fillStyle=pct>0.5?"#6ee06e":pct>0.25?"#ffd44d":"#ff5b5b";
    ctx.fillRect(e.x-w/2,e.y-e.r-9,w*pct,5);
  }

  // particles
  for(const p of particles){
    ctx.globalAlpha=Math.max(0,p.life/p.max);
    ctx.fillStyle=p.c;
    ctx.fillRect(p.x-2.5,p.y-2.5,5,5);
  }
  ctx.globalAlpha=1;

  // wave banner
  if(waveActive && spawnQueue.length>4){
    ctx.font="bold 20px Arial"; ctx.textAlign="left";
    ctx.fillStyle="rgba(255,255,255,.7)";
    ctx.fillText("Wave "+wave+" — "+(spawnQueue.length+enemies.length)+" left",14,WORLD_H-14);
  }
}

const tpanel=document.getElementById("tpanel");

function refreshPanel(){
  if(!selTower){ hidePanel(); return; }
  const d=TOWER_DEFS[selTower.idx], s=tStat(selTower);
  document.getElementById("tpName").textContent=d.name+" — Lv "+selTower.level;
  document.getElementById("tpStats").innerHTML=
    "Damage: <b>"+s.dmg+"</b> · Range: <b>"+s.range+"</b><br>Fire rate: <b>"+s.rate+"/s</b>";
  const up=document.getElementById("upBtn");
  if(selTower.level>=3){ up.textContent="MAX LEVEL"; up.disabled=true; }
  else {
    const c=upCost(selTower);
    up.textContent="⬆ Upgrade $"+c;
    up.disabled = gold<c;
  }
  document.getElementById("sellBtn").textContent="💰 Sell $"+sellVal(selTower);
  tpanel.style.display="block";
}

  if(!selTower) return;
  const c=upCost(selTower);
  if(selTower.level>=3 || gold<c){ sfx.error(); return; }
  gold-=c; selTower.invested+=c; selTower.level++;
  sfx.place(); updateHUD(); refreshPanel();
};
document.getElementById("sellBtn").onclick=()=>{
  if(!selTower) return;
  gold+=sellVal(selTower);
  towers=towers.filter(t=>t!==selTower);
  selTower=null; sfx.coin(); updateHUD(); hidePanel();
};

  const r=cv.getBoundingClientRect();
