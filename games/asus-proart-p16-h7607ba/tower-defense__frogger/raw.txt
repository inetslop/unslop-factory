function resize(){
  W=cv.width=window.innerWidth; H=cv.height=window.innerHeight;

/* ---------- audio ---------- */
let AC=null;
function audio(){ if(!AC){ try{AC=new (window.AudioContext||window.webkitAudioContext)();}catch(e){} } return AC; }
function sfx(kind){
  const ac=audio(); if(!ac) return;
  const t=ac.currentTime, o=ac.createOscillator(), g=ac.createGain();
  o.connect(g); g.connect(ac.destination);
  let f=440,d=0.08,type='square',v=0.15,f2=null;
  if(kind==='hop'){f=300;f2=500;d=0.09;}
  else if(kind==='build'){f=200;f2=400;d=0.15;type='triangle';}
  else if(kind==='shoot'){f=900;f2=300;d=0.06;v=0.08;}
  else if(kind==='boom'){f=120;f2=40;d=0.25;type='sawtooth';v=0.2;}
  else if(kind==='die'){f=300;f2=60;d=0.5;type='sawtooth';v=0.2;}
  else if(kind==='win'){f=400;f2=800;d=0.35;type='triangle';}
  o.type=type; o.frequency.setValueAtTime(f,t);
  if(f2) o.frequency.exponentialRampToValueAtTime(Math.max(30,f2),t+d);
  g.gain.setValueAtTime(v,t); g.gain.exponentialRampToValueAtTime(0.001,t+d);
  o.start(t); o.stop(t+d+0.02);
}

/* ---------- state ---------- */
let S=null, running=false, mouse={cx:-1,cy:-1};

const LANE_DEFS=[
 {row:1,type:'boat',dir: 1,speed:2.0,count:3},
 {row:2,type:'croc',dir:-1,speed:1.1,count:4},
 {row:3,type:'boat',dir:-1,speed:2.5,count:3},
 {row:5,type:'car', dir: 1,speed:2.8,count:3,color:'#e74c3c'},
 {row:6,type:'car', dir:-1,speed:1.9,count:4,color:'#f1c40f'},
 {row:7,type:'car', dir: 1,speed:3.5,count:2,color:'#a55eea'},
 {row:8,type:'car', dir:-1,speed:1.5,count:5,color:'#489fe8'}
];
const HP={car:2,boat:3,croc:4};
const REWARD={car:15,boat:20,croc:25};

function makeEnemies(level){
  const mult=1+(level-1)*0.12;
  const list=[];
  for(const L of LANE_DEFS){
    for(let i=0;i<L.count;i++){
      const span=(COLS+4)/L.count;
      let x=i*span + Math.random()*span*0.5 - 2;
      list.push({x, row:L.row, dir:L.dir, speed:L.speed*mult, type:L.type,
        hp:HP[L.type], maxHp:HP[L.type], color:L.color||null, wobble:Math.random()*6.28});
    }
  }
  return list;
}

function reset(){
  S={ lives:3, coins:START_COINS, score:0, level:1, time:0,
      enemies:makeEnemies(1), towers:[], bullets:[], parts:[], tracers:[],
      frog:{col:6,row:10,hopT:1,invuln:2} };
}

function startGame(){
  audio(); reset(); running=true;
  startScreen.classList.add('hidden'); overScreen.classList.add('hidden');
}
document.getElementById('startBtn').addEventListener('click',startGame);
document.getElementById('restartBtn').addEventListener('click',startGame);

/* ---------- input ---------- */
window.addEventListener('keydown',e=>{
  const k=e.key;
  if(['ArrowUp','ArrowDown','ArrowLeft','ArrowRight',' '].includes(k)) e.preventDefault();
  if(!running){ if(k==='Enter'||k===' ') startGame(); return; }
  let dc=0,dr=0;
  if(k==='ArrowUp'||k==='w'||k==='W') dr=-1;
  else if(k==='ArrowDown'||k==='s'||k==='S') dr=1;
  else if(k==='ArrowLeft'||k==='a'||k==='A') dc=-1;
  else if(k==='ArrowRight'||k==='d'||k==='D') dc=1;
  else return;
  const f=S.frog;
  const nc=f.col+dc, nr=f.row+dr;
  if(nc<0||nc>=COLS||nr<0||nr>=ROWS) return;
  f.col=nc; f.row=nr; f.hopT=0; sfx('hop');
});

  mouse.cx=Math.floor((e.clientX-r.left-ox)/CELL);

  if(!running) return;

    sfx('build');
  }
});

let flashMsg='', flashT=0;

/* ---------- update ---------- */
function killEnemy(en,idx){
  S.enemies.splice(idx,1);
  const r=REWARD[en.type];
  S.coins+=r; S.score+=r;
  for(let i=0;i<10;i++){
    const a=Math.random()*6.28, sp=1+Math.random()*3;
    S.parts.push({x:en.x,y:en.row,vx:Math.cos(a)*sp,vy:Math.sin(a)*sp,life:0.5,t:0,color:en.type==='croc'?'#7dff8a':'#ffae42'});
  }
  sfx('boom');
}

function frogDied(){
  const f=S.frog;
  for(let i=0;i<14;i++){
    const a=Math.random()*6.28, sp=1+Math.random()*3;
    S.parts.push({x:f.col,y:f.row,vx:Math.cos(a)*sp,vy:Math.sin(a)*sp,life:0.6,t:0,color:'#5eff7a'});
  }
  S.lives--; sfx('die');
  if(S.lives<=0){ gameOver(); return; }
  f.col=6; f.row=10; f.invuln=2;
}

function frogWon(){
  S.level++;
  S.lives=Math.min(5,S.lives+1);
  S.coins+=40; S.score+=100;
  sfx('win');
  flashMsg='LEVEL '+S.level+' — traffic faster!'; flashT=1.6;
  const f=S.frog; f.col=6; f.row=10; f.invuln=2;
  S.enemies=makeEnemies(S.level);
}

function gameOver(){
  running=false;
  finalScore.textContent='Score: '+S.score+'   •   Level reached: '+S.level;
  overScreen.classList.remove('hidden');
}

function update(dt){
  S.time+=dt;
  if(flashT>0) flashT-=dt;
  const f=S.frog;
  f.hopT=Math.min(1,f.hopT+dt*5);
  if(f.invuln>0) f.invuln-=dt;

  // frog goal check
  if(f.row===0 && PADS.includes(f.col)){ frogWon(); return; }

  // enemies
  for(const en of S.enemies){
    en.x+=en.dir*en.speed*dt;
    en.wobble+=dt*6;
    if(en.dir>0 && en.x>COLS+1.5) en.x=-1.5;
    if(en.dir<0 && en.x<-1.5) en.x=COLS+1.5;
  }

  // collision with frog
  if(f.invuln<=0){
    for(const en of S.enemies){
      if(en.row===f.row && Math.abs(en.x-f.col)<0.62){ frogDied(); return; }
    }
  }

  // towers
  for(const t of S.towers){
    t.cd-=dt;
    let best=null,bd=1e9;
    const range=3.0;
    for(const en of S.enemies){
      const dx=en.x-t.col, dy=en.row-t.row, d=Math.hypot(dx,dy);
      if(d<range && d<bd){bd=d;best=en;}
    }
    if(best) t.angle=Math.atan2(best.row-t.row,best.x-t.col);
    if(best && t.cd<=0){
      t.cd=0.55;
      S.bullets.push({x:t.col,y:t.row,target:best,px:best.x,py:best.row});
      sfx('shoot');
    }
  }

  // bullets
  for(let i=S.bullets.length-1;i>=0;i--){
    const b=S.bullets[i];
    if(S.enemies.includes(b.target)){ b.px=b.target.x; b.py=b.target.row; }
    const dx=b.px-b.x, dy=b.py-b.y, d=Math.hypot(dx,dy);
    if(d<0.3){
      S.bullets.splice(i,1);
      if(S.enemies.includes(b.target)){
        b.target.hp--;
        if(b.target.hp<=0) killEnemy(b.target,S.enemies.indexOf(b.target));
        else S.tracers.push({x:b.px,y:b.py,t:0.15});
      }
      continue;
    }
    const sp=9*dt;
    b.x+=dx/d*sp; b.y+=dy/d*sp;
  }

  // particles & tracers
  for(let i=S.parts.length-1;i>=0;i--){
    const p=S.parts[i]; p.t+=dt;
    if(p.t>p.life){S.parts.splice(i,1);continue;}
    p.x+=p.vx*dt; p.y+=p.vy*dt; p.vx*=0.94; p.vy*=0.94;
  }
  for(let i=S.tracers.length-1;i>=0;i--){ S.tracers[i].t-=dt; if(S.tracers[i].t<=0) S.tracers.splice(i,1); }
}

    else col=(r%2?'#2b2f36':'#31363e');

    if(r>=1&&r<=3){ // water shimmer
      ctx.strokeStyle='rgba(120,190,255,0.18)'; ctx.lineWidth=2;
      for(let i=0;i<6;i++){
        const wx=ox+((i*CELL*2+S.time*40*(r%2?1:-1))%(COLS*CELL+CELL))-CELL/2;
        ctx.beginPath(); ctx.moveTo(Math.max(ox,wx),y+CELL*0.5);
        ctx.lineTo(Math.min(ox+COLS*CELL,wx+CELL*0.6),y+CELL*0.5); ctx.stroke();
      }
    } else if(r>=5&&r<=8){ // road dashes
      ctx.strokeStyle='#c9a227'; ctx.lineWidth=3; ctx.setLineDash([14,12]);
      ctx.beginPath(); ctx.moveTo(ox,y+CELL/2); ctx.lineTo(ox+COLS*CELL,y+CELL/2); ctx.stroke();
      ctx.setLineDash([]);
    } else if(GRASS_ROWS.includes(r)){ // buildable marker
      ctx.fillStyle='rgba(255,255,255,0.07)';
      for(let c=0;c<COLS;c++) ctx.fillRect(ox+c*CELL+1,y+1,CELL-2,CELL-2);
    }
  }

  // lily pads
  for(const c of PADS){
    const [px,py]=cellPx(c,0);
    ctx.fillStyle='#3fae54';
    ctx.beginPath(); ctx.arc(px+CELL/2,py+CELL/2,CELL*0.36,0,6.29); ctx.fill();
    ctx.fillStyle='#2c8a3e';
    ctx.beginPath(); ctx.moveTo(px+CELL/2,py+CELL/2);
    ctx.arc(px+CELL/2,py+CELL/2,CELL*0.36,-0.4,0.4); ctx.fill();
  }

  // enemies
  for(const en of S.enemies){
    const [px,py]=cellPx(en.x,en.row);
    ctx.save(); ctx.translate(px+CELL/2,py+CELL/2);
    if(en.type==='car'){
      ctx.fillStyle=en.color;
      rr(-CELL*0.4,-CELL*0.3,CELL*0.8,CELL*0.6,5); ctx.fill();
      ctx.fillStyle='rgba(255,255,255,0.85)';
      rr(en.dir>0? -CELL*0.15:-CELL*0.3, -CELL*0.18, CELL*0.45, CELL*0.36, 3); ctx.fill();
      ctx.fillStyle='#1b1e24';
      ctx.beginPath(); ctx.arc(en.dir>0? -CELL*0.28:CELL*0.28, CELL*0.32, CELL*0.12,0,6.29);
      ctx.arc(en.dir>0? CELL*0.28:-CELL*0.28, CELL*0.32, CELL*0.12,0,6.29); ctx.fill();
    } else if(en.type==='boat'){
      ctx.scale(en.dir,1);
      ctx.fillStyle='#d8b48a';
      ctx.beginPath();
      ctx.moveTo(-CELL*0.55,0); ctx.quadraticCurveTo(0,-CELL*0.32,CELL*0.62,0);
      ctx.quadraticCurveTo(0,CELL*0.3,-CELL*0.55,0); ctx.fill();
      ctx.fillStyle='#8a6b47';
      ctx.beginPath(); ctx.arc(0,0,CELL*0.16,0,6.29); ctx.fill();
    } else { // croc
      const w=Math.sin(en.wobble)*0.15;
      ctx.rotate(w);
      ctx.fillStyle='#4f9e3f';
      rr(-CELL*0.42,-CELL*0.22,CELL*0.84,CELL*0.44,6); ctx.fill();
      ctx.fillStyle='#3a7d30';
      for(let i=-1;i<=1;i++){ ctx.beginPath(); ctx.arc(i*CELL*0.22,0,CELL*0.1,0,6.29); ctx.fill(); }
      ctx.fillStyle='#ffdd55';
      ctx.beginPath(); ctx.arc(en.dir>0?CELL*0.36:-CELL*0.36,-CELL*0.08,CELL*0.05,0,6.29);
      ctx.arc(en.dir>0?CELL*0.36:-CELL*0.36, CELL*0.08,CELL*0.05,0,6.29); ctx.fill();
    }
    // hp bar
    if(en.hp<en.maxHp){
      ctx.rotate(0); ctx.fillStyle='#111'; ctx.fillRect(-CELL*0.4,-CELL*0.48,CELL*0.8,5);
      ctx.fillStyle='#ff5d5d'; ctx.fillRect(-CELL*0.4,-CELL*0.48,CELL*0.8*(en.hp/en.maxHp),5);
    }
    ctx.restore();
  }

  // towers
  for(const t of S.towers){
    const [px,py]=cellPx(t.col,t.row);
    ctx.fillStyle='#5a6472'; rr(px+CELL*0.18,py+CELL*0.18,CELL*0.64,CELL*0.64,6); ctx.fill();
    ctx.fillStyle='#39414d'; ctx.beginPath(); ctx.arc(px+CELL/2,py+CELL/2,CELL*0.22,0,6.29); ctx.fill();
    ctx.save(); ctx.translate(px+CELL/2,py+CELL/2); ctx.rotate(t.angle);
    ctx.fillStyle='#8b95a5'; ctx.fillRect(0,-CELL*0.07,CELL*0.42,CELL*0.14);
    ctx.restore();
  }

  // bullets & tracers
  for(const b of S.bullets){
    const [px,py]=cellPx(b.x,b.y);
    ctx.fillStyle='#ffe14d';
    ctx.beginPath(); ctx.arc(px+CELL/2,py+CELL/2,4,0,6.29); ctx.fill();
  }
  for(const tr of S.tracers){
    const [px,py]=cellPx(tr.x,tr.y);
    ctx.strokeStyle='rgba(255,180,80,'+(tr.t*4)+')'; ctx.lineWidth=3;
    ctx.beginPath(); ctx.arc(px+CELL/2,py+CELL/2,8*(1-tr.t*2),0,6.29); ctx.stroke();
  }

  // particles
  for(const p of S.parts){
    const [px,py]=cellPx(p.x,p.y);
    ctx.globalAlpha=1-p.t/p.life;
    ctx.fillStyle=p.color;
    ctx.fillRect(px+CELL/2-3,py+CELL/2-3,6,6);
  }
  ctx.globalAlpha=1;

  // frog
  const f=S.frog;
  if(!(f.invuln>0 && Math.floor(S.time*10)%2===0)){
    const [px,py]=cellPx(f.col,f.row);
    const hop=Math.sin(Math.min(1,f.hopT)*Math.PI)*CELL*0.18;
    ctx.save(); ctx.translate(px+CELL/2,py+CELL/2-hop);
    ctx.fillStyle='#39d353';
    ctx.beginPath(); ctx.ellipse(0,0,CELL*0.34,CELL*0.3,0,0,6.29); ctx.fill();
    ctx.fillStyle='#2aa841';
    ctx.beginPath(); ctx.arc(-CELL*0.2,CELL*0.28,CELL*0.12,0,6.29);
    ctx.arc(CELL*0.2,CELL*0.28,CELL*0.12,0,6.29); ctx.fill();
    ctx.fillStyle='#fff';
    ctx.beginPath(); ctx.arc(-CELL*0.14,-CELL*0.18,CELL*0.11,0,6.29);
    ctx.arc(CELL*0.14,-CELL*0.18,CELL*0.11,0,6.29); ctx.fill();
    ctx.fillStyle='#111';
    ctx.beginPath(); ctx.arc(-CELL*0.14,-CELL*0.18,CELL*0.05,0,6.29);
    ctx.arc(CELL*0.14,-CELL*0.18,CELL*0.05,0,6.29); ctx.fill();
    ctx.restore();
  }

  // hover placement preview
  if(running && GRASS_ROWS.includes(mouse.cy) && mouse.cx>=0 && mouse.cx<COLS){
    const occupied=S.towers.some(t=>t.col===mouse.cx&&t.row===mouse.cy);
    const ok=!occupied && S.coins>=TOWER_COST;
    const [px,py]=cellPx(mouse.cx,mouse.cy);
    ctx.strokeStyle=ok?'rgba(120,255,140,0.9)':'rgba(255,90,90,0.9)';
    ctx.lineWidth=3; ctx.strokeRect(px+3,py+3,CELL-6,CELL-6);
    if(ok){
      ctx.strokeStyle='rgba(120,255,140,0.3)';
      ctx.beginPath(); ctx.arc(px+CELL/2,py+CELL/2,3*CELL,0,6.29); ctx.stroke();
    }
  }

  // HUD
  const hudY=Math.max(18,oy-14);
  ctx.font='bold '+Math.min(20,CELL*0.4)+'px Segoe UI, Arial';
  ctx.fillStyle='#eaf7ff'; ctx.textAlign='left';
  ctx.fillText('♥'.repeat(Math.max(0,S.lives)).padEnd(5,'♡')+'  💰'+S.coins, ox+4, hudY);
  ctx.textAlign='center';
  ctx.fillText('SCORE '+S.score+'   •   LEVEL '+S.level, W/2, hudY);
  ctx.textAlign='right';
  ctx.fillStyle='#9fd8ff';
  ctx.fillText('Turret: '+TOWER_COST+' coins', ox+COLS*CELL-4, hudY);

  if(flashT>0){
    ctx.globalAlpha=Math.min(1,flashT);
    ctx.textAlign='center'; ctx.fillStyle='#ffd76a';
    ctx.font='bold '+Math.min(28,CELL*0.55)+'px Segoe UI, Arial';
    ctx.fillText(flashMsg, W/2, oy+ROWS*CELL+30> H-10 ? oy+CELL*1.4 : oy+ROWS*CELL+34);
    ctx.globalAlpha=1;
  }
}

function rr(x,y,w,h,r){
  ctx.beginPath();
  ctx.moveTo(x+r,y); ctx.arcTo(x+w,y,x+w,y+h,r); ctx.arcTo(x+w,y+h,x,y+h,r);
  ctx.arcTo(x,y+h,x,y,r); ctx.arcTo(x,y,x+w,y,r); ctx.closePath();
}

/* ---------- loop ---------- */
let last=performance.now();
function frame(now){
  const dt=Math.min(0.05,(now-last)/1000); last=now;
  if(running) update(dt);
  draw();
  requestAnimationFrame(frame);
}
reset(); // so background board shows behind start screen
requestAnimationFrame(frame);

  try{ if(S&&S.score>0) localStorage.setItem('froggerDefenseBest', String(Math.max(+localStorage.getItem('froggerDefenseBest')||0,S.score))); }catch(e){}
});
})();
