/* ---------- setup ---------- */
const canvas = document.getElementById('game');

let actx = null;
function initAudio(){
  if(!actx){
    try{ actx = new (window.AudioContext||window.webkitAudioContext)(); }catch(e){}
    if(actx && actx.state === 'suspended') actx.resume();
  }
}
function tone(freq, endFreq, dur, type, vol){
  if(!actx) return;
  try{
    const t = actx.currentTime;
    const o = actx.createOscillator(), g = actx.createGain();
    o.type = type; o.frequency.setValueAtTime(freq, t);
    o.frequency.exponentialRampToValueAtTime(Math.max(30,endFreq), t+dur);
    g.gain.setValueAtTime(vol, t);
    g.gain.exponentialRampToValueAtTime(0.0001, t+dur);
    o.connect(g); g.connect(actx.destination);
    o.start(t); o.stop(t+dur+0.02);
  }catch(e){}
}
const sfx = {
  shoot: ()=>tone(950,420,0.06,'square',0.035),
  hit:   ()=>{ tone(180,40,0.35,'sawtooth',0.14); tone(90,30,0.4,'triangle',0.12); },
  boom:  ()=>tone(320,50,0.25,'square',0.08),
  shield:()=>{ tone(400,900,0.15,'sine',0.1); tone(600,1200,0.2,'sine',0.07); },
  power: ()=>tone(300,1400,0.25,'triangle',0.1),
  over:  ()=>{ tone(400,60,0.9,'sawtooth',0.12); }
};

/* ---------- input ---------- */
const keys = {};
window.addEventListener('keydown', e=>{

  keys[e.key.toLowerCase()] = true;
  if((e.key==='Enter'||e.key===' ') && state==='start') startGame();
  if((e.key==='r'||e.key==='R'||e.key==='Enter') && state==='over') startGame();
});
window.addEventListener('keyup', e=>{ keys[e.key.toLowerCase()] = false; });
canvas.addEventListener('mousedown', initAudio);

let score=0, lives=3, timeAlive=0, shake=0;

try{ hiScore = parseInt(localStorage.getItem('bulletRushHi')||'0',10)||0; }catch(e){}

const player = { x:0, y:0, r:9, speed:340, fireCd:0, shield:0, inv:0 };

}
makeStars();

/* ---------- helpers ---------- */
function rnd(a,b){ return a+Math.random()*(b-a); }
function dist2(ax,ay,bx,by){ const dx=ax-bx, dy=ay-by; return dx*dx+dy*dy; }

function addParticles(x,y,color,n,sp){
  for(let i=0;i<n;i++){
    const a=Math.random()*Math.PI*2, v=rnd(sp*0.3,sp);
    particles.push({x,y,vx:Math.cos(a)*v,vy:Math.sin(a)*v,life:rnd(0.3,0.7),max:0.7,color,r:rnd(1.5,3.5)});
  }
}

function fireEnemyBullet(x,y,ang,sp){
  if(eBullets.length>700) return;
  eBullets.push({x,y,vx:Math.cos(ang)*sp,vy:Math.sin(ang)*sp,r:4.5,hue:rnd(0,1)});
}

/* ---------- enemy spawning ---------- */
function spawnEnemy(){
  const t = timeAlive;
  const roll = Math.random();
  const x = rnd(60, W-60);
  if(t>25 && roll<0.28){
    // spiral turret: hovers, fires rotating bursts
    enemies.push({type:'turret',x,y:-40,targetY:rnd(90,H*0.35),hp:8+Math.floor(t/20),r:16,ang:Math.random()*6,timer:0,burst:0});
  } else if(t>12 && roll<0.5){
    // splitter
    enemies.push({type:'split',x,y:-30,hp:2,r:14,vy:rnd(70,110),timer:rnd(0.8,1.5)});
  } else if(t>6 && roll<0.7){
    // dasher: aims and strafes
    enemies.push({type:'dasher',x,y:-30,hp:4,r:13,vy:rnd(90,140),dir:Math.random()<0.5?-1:1,timer:rnd(0.6,1.2)});
  } else {
    // drone: aimed shots
    enemies.push({type:'drone',x,y:-30,hp:3,r:12,vy:rnd(80,130),timer:rnd(0.7,1.4)});
  }
}

function spawnSplitKids(x,y){
  for(let s=-1;s<=1;s+=2)
    enemies.push({type:'shard',x:x+s*15,y,hp:1,r:9,vx:s*rnd(60,100),vy:rnd(140,190),timer:rnd(0.5,1)});
}

/* ---------- update ---------- */
function update(dt){
  timeAlive += dt;
  score += Math.floor(dt*10); // survival points (integrated below)

  const diff = 1 + timeAlive/30;

  /* --- player --- */
  const slow = keyDown('shift') ? 0.42 : 1;
  let mx=0,my=0;
  if(keyDown('arrowleft','a')) mx-=1;
  if(keyDown('arrowright','d')) mx+=1;
  if(keyDown('arrowup','w')) my-=1;
  if(keyDown('arrowdown','s')) my+=1;
  const len = Math.hypot(mx,my)||1;
  player.x += (mx/len)*player.speed*slow*dt;
  player.y += (my/len)*player.speed*slow*dt;
  player.x = Math.max(16, Math.min(W-16, player.x));
  player.y = Math.max(16, Math.min(H-16, player.y));

  if(player.inv>0) player.inv-=dt;
  if(player.shield>0) player.shield-=dt;

  // fire (auto while space held, or always light auto? -> hold space)
  player.fireCd -= dt;
  if(keyDown(' ') && player.fireCd<=0){
    player.fireCd = 0.14;
    pBullets.push({x:player.x-7,y:player.y-8,vx:0,vy:-620});
    pBullets.push({x:player.x+7,y:player.y-8,vx:0,vy:-620});
    if(player.shield>0){ // shield = extra side cannons
      pBullets.push({x:player.x-14,y:player.y,vx:-90,vy:-560});
      pBullets.push({x:player.x+14,y:player.y,vx:90,vy:-560});
    }
    sfx.shoot();
  }

  /* --- spawn enemies --- */
  spawnT -= dt;
  if(spawnT<=0){
    spawnEnemy();
    spawnT = Math.max(0.35, rnd(0.9,1.4)/Math.sqrt(diff));
  }

  /* --- player bullets --- */
  for(let i=pBullets.length-1;i>=0;i--){
    const b=pBullets[i];
    b.x+=b.vx*dt; b.y+=b.vy*dt;
    if(b.y<-20||b.x<-20||b.x>W+20) pBullets.splice(i,1);
  }

  /* --- enemies --- */
  for(let i=enemies.length-1;i>=0;i--){
    const e=enemies[i];
    switch(e.type){
      case 'drone':
        e.y += e.vy*dt;
        e.timer-=dt;
        if(e.timer<=0 && e.y>0){
          e.timer = Math.max(0.5, 1.3/diff);
          const a=Math.atan2(player.y-e.y, player.x-e.x);
          fireEnemyBullet(e.x,e.y+8,a,rnd(170,230)*Math.min(1.4,diff*0.9));
        }
        break;
      case 'dasher':
        e.y += e.vy*dt;
        e.x += e.dir*160*dt;
        if(e.x<30||e.x>W-30) e.dir*=-1;
        e.timer-=dt;
        if(e.timer<=0 && e.y>0){
          e.timer = Math.max(0.55, 1.2/diff);
          const a=Math.atan2(player.y-e.y, player.x-e.x);
          for(let k=-1;k<=1;k++) fireEnemyBullet(e.x,e.y+6,a+k*0.18,rnd(190,240));
        }
        break;
      case 'turret':
        if(e.y<e.targetY) e.y += 90*dt;
        e.ang += dt*2.2;
        e.timer-=dt;
        if(e.timer<=0 && e.y>=e.targetY-1){
          e.timer = Math.max(0.11, 0.16/diff);
          for(let k=0;k<3;k++) fireEnemyBullet(e.x,e.y, e.ang + k*(Math.PI*2/3), rnd(130,170));
        }
        if(timeAlive%40 < 0 && false){}
        // leave after a while
        e.life=(e.life||0)+dt;
        if(e.life>9) e.dead=true;
        break;
      case 'split':
        e.y += e.vy*dt;
        e.timer-=dt;
        if(e.timer<=0 && e.y>0){
          e.timer=1.1;
          const a=Math.atan2(player.y-e.y, player.x-e.x);
          fireEnemyBullet(e.x,e.y+6,a,200);
        }
        break;
      case 'shard':
        e.x+=e.vx*dt; e.y+=e.vy*dt;
        if(e.x<15||e.x>W-15) e.vx*=-1;
        e.timer-=dt;
        if(e.timer<=0){ e.timer=0.9; fireEnemyBullet(e.x,e.y,Math.PI/2+ (Math.random()-0.5),160); }
        break;
    }

    // collision with player bullets
    for(let j=pBullets.length-1;j>=0;j--){
      const b=pBullets[j];
      if(dist2(b.x,b.y,e.x,e.y) < (e.r+4)*(e.r+4)){
        pBullets.splice(j,1);
        e.hp--;
        addParticles(b.x,b.y,'#9fdcff',3,120);
        if(e.hp<=0){ e.dead=true; break; }
      }
    }

    if(e.dead){
      score += (e.type==='turret'?60: e.type==='dasher'?40: e.type==='split'?25: 30);
      sfx.boom();
      addParticles(e.x,e.y, e.type==='turret'?'#ff9f5a':'#ff5a8a', 14, 220);
      if(e.type==='split') spawnSplitKids(e.x,e.y);
      // pickup drop chance
      if(Math.random()<0.14 && e.y>0) pickups.push({x:e.x,y:e.y,t:0});
      enemies.splice(i,1);
      continue;
    }
    if(e.y>H+50||e.x<-60||e.x>W+60) { enemies.splice(i,1); }
  }

  /* --- enemy bullets --- */
  for(let i=eBullets.length-1;i>=0;i--){
    const b=eBullets[i];
    b.x+=b.vx*dt; b.y+=b.vy*dt;
    if(b.y<-30||b.y>H+30||b.x<-30||b.x>W+30){ eBullets.splice(i,1); continue; }
    // hit player (small hitbox)
    if(player.inv<=0 && dist2(b.x,b.y,player.x,player.y)<(5+b.r)*(5+b.r)){
      eBullets.splice(i,1);
      damagePlayer();
    }
  }

  /* --- enemy body collision --- */
  if(player.inv<=0){
    for(let i=enemies.length-1;i>=0;i--){
      const e=enemies[i];
      if(dist2(e.x,e.y,player.x,player.y)<(e.r+6)*(e.r+6)){
        enemies.splice(i,1);
        addParticles(e.x,e.y,'#ff5a8a',14,220);
        sfx.boom();
        damagePlayer();
        break;
      }
    }
  }

  /* --- pickups --- */
  for(let i=pickups.length-1;i>=0;i--){
    const p=pickups[i];
    p.t+=dt; p.y+=45*dt;
    if(p.y>H+20){ pickups.splice(i,1); continue; }
    if(dist2(p.x,p.y,player.x,player.y)<(18)*(18)){
      pickups.splice(i,1);
      if(player.shield>3) score+=150;
      else { player.shield=Math.min(6,player.shield+4); sfx.power(); }
      addParticles(p.x,p.y,'#7fff9e',12,180);
    }
  }

  /* --- particles --- */
  for(let i=particles.length-1;i>=0;i--){
    const p=particles[i];
    p.life-=dt;
    if(p.life<=0){ particles.splice(i,1); continue; }
    p.x+=p.vx*dt; p.y+=p.vy*dt;
    p.vx*=0.96; p.vy*=0.96;
  }

  /* --- stars --- */
  const sp = keyDown('shift')?0.5:1;
  for(const s of starField){
    s.y += s.v*sp*dt*(1+timeAlive/60);
    if(s.y>H){ s.y=-4; s.x=Math.random()*W; }
  }

  score = Math.floor(score + dt*8*diff); // handled above approximately; keep simple int accumulation
  if(shake>0) shake-=dt*3;
}

function damagePlayer(){
  if(player.shield>0){
    player.shield=0;
    player.inv=1.2;
    sfx.shield();
    addParticles(player.x,player.y,'#7fff9e',20,260);
    // clear bullets near player
    for(let i=eBullets.length-1;i>=0;i--)
      if(dist2(eBullets[i].x,eBullets[i].y,player.x,player.y)<140*140) eBullets.splice(i,1);
    return;
  }
  lives--;
  player.inv=2.2;
  shake=1;
  sfx.hit();
  addParticles(player.x,player.y,'#ff9f5a',30,320);
  for(let i=eBullets.length-1;i>=0--)
    if(dist2(eBullets[i].x,eBullets[i].y,player.x,player.y)<170*170) eBullets.splice(i,1);
  if(lives<=0) endGame();
}

/* ---------- render ---------- */
function drawShip(x,y){
  ctx.save();
  ctx.translate(x,y);
  // engine glow
  const fl = 8+Math.random()*6;
  ctx.fillStyle='rgba(90,190,255,0.7)';
  ctx.beginPath();
  ctx.moveTo(-5,14); ctx.lineTo(0,14+fl); ctx.lineTo(5,14); ctx.closePath(); ctx.fill();
  // body
  const grad = ctx.createLinearGradient(0,-16,0,16);
  grad.addColorStop(0,'#eaffff'); grad.addColorStop(0.5,'#6fd2ff'); grad.addColorStop(1,'#1a5fd0');
  ctx.fillStyle=grad;
  ctx.beginPath();
  ctx.moveTo(0,-17); ctx.lineTo(-13,10); ctx.lineTo(-6,6); ctx.lineTo(0,14);
  ctx.lineTo(6,6); ctx.lineTo(13,10); ctx.closePath(); ctx.fill();
  // cockpit
  ctx.fillStyle='#0a2a5a';
  ctx.beginPath(); ctx.ellipse(0,-4,3,6,0,0,Math.PI*2); ctx.fill();
  if(player.inv>0 && Math.floor(timeAlive*12)%2===0) ctx.globalAlpha=0.35;
  // shield ring
  if(player.shield>0){
    ctx.strokeStyle='rgba(127,255,158,'+(0.4+0.3*Math.sin(timeAlive*6))+')';
    ctx.lineWidth=2.5;
    ctx.beginPath(); ctx.arc(0,0,24,0,Math.PI*2); ctx.stroke();
  }
  ctx.restore();
}

function drawEnemy(e){
  ctx.save();
  ctx.translate(e.x,e.y);
  switch(e.type){
    case 'drone':
      ctx.fillStyle='#ff5a8a';
      ctx.beginPath(); ctx.moveTo(0,12); ctx.lineTo(-11,-8); ctx.lineTo(0,-3); ctx.lineTo(11,-8); ctx.closePath(); ctx.fill();
      ctx.fillStyle='#4a0f26'; ctx.beginPath(); ctx.arc(0,0,4,0,Math.PI*2); ctx.fill();
      break;
    case 'dasher':
      ctx.rotate(Math.PI/2*e.dir*(e.x<W/2?1:-1));
      ctx.fillStyle='#ffb85a';
      ctx.beginPath(); ctx.moveTo(0,-14); ctx.lineTo(13,8); ctx.lineTo(0,14); ctx.lineTo(-13,8); ctx.closePath(); ctx.fill();
      ctx.fillStyle='#5a2f00'; ctx.beginPath(); ctx.arc(0,3,4,0,Math.PI*2); ctx.fill();
      break;
    case 'turret':
      ctx.rotate(e.ang);
      ctx.fillStyle='#c98aff';
      for(let k=0;k<3;k++){
        ctx.rotate(Math.PI*2/3);
        ctx.beginPath(); ctx.moveTo(0,-18); ctx.lineTo(5,-4); ctx.lineTo(-5,-4); ctx.closePath(); ctx.fill();
      }
      ctx.fillStyle='#4a1f6e'; ctx.beginPath(); ctx.arc(0,0,9,0,Math.PI*2); ctx.fill();
      ctx.fillStyle='#e8c8ff'; ctx.beginPath(); ctx.arc(0,0,4,0,Math.PI*2); ctx.fill();
      break;
    case 'split':
      ctx.rotate(timeAlive*3);
      ctx.fillStyle='#5affd8';
      ctx.beginPath();
      for(let k=0;k<6;k++){ const a=k*Math.PI/3; ctx.lineTo(Math.cos(a)*14,Math.sin(a)*14); }
      ctx.closePath(); ctx.fill();
      ctx.fillStyle='#0a4a3a'; ctx.beginPath(); ctx.arc(0,0,5,0,Math.PI*2); ctx.fill();
      break;
    case 'shard':
      ctx.rotate(timeAlive*6);
      ctx.fillStyle='#8affea';
      ctx.beginPath(); ctx.moveTo(0,-10); ctx.lineTo(8,7); ctx.lineTo(-8,7); ctx.closePath(); ctx.fill();
      break;
  }
  ctx.restore();
}

function render(){
  // background
  const g=ctx.createLinearGradient(0,0,0,H);
  g.addColorStop(0,'#050914'); g.addColorStop(1,'#0a1230');
  ctx.fillStyle=g; ctx.fillRect(0,0,W,H);

  ctx.save();
  if(shake>0) ctx.translate(rnd(-1,1)*shake*8, rnd(-1,1)*shake*8);

  // stars
  ctx.fillStyle='#9fc8ff';
  for(const s of starField){
    ctx.globalAlpha=0.3+s.s*0.3;
    ctx.fillRect(s.x,s.y,s.s,s.s*2.5);
  }
  ctx.globalAlpha=1;

  // pickups
  for(const p of pickups){
    const pu=0.6+0.4*Math.sin(p.t*8);
    ctx.fillStyle='rgba(127,255,158,'+pu+')';
    ctx.beginPath(); ctx.arc(p.x,p.y,9,0,Math.PI*2); ctx.fill();
    ctx.fillStyle='#0a3a1f';
    ctx.beginPath(); ctx.moveTo(p.x,p.y-5); ctx.lineTo(p.x+5,p.y+4); ctx.lineTo(p.x-5,p.y+4); ctx.closePath(); ctx.fill();
  }

  // enemies
  for(const e of enemies) drawEnemy(e);

  // player bullets
  ctx.fillStyle='#aef1ff';
  for(const b of pBullets){
    ctx.fillRect(b.x-2,b.y-8,4,14);
  }

  // enemy bullets (additive glow)
  ctx.globalCompositeOperation='lighter';
  for(const b of eBullets){
    const hue = b.hue<0.5 ? '#ff4a6a' : '#ffd24a';
    ctx.fillStyle=hue;
    ctx.beginPath(); ctx.arc(b.x,b.y,b.r,0,Math.PI*2); ctx.fill();
    ctx.globalAlpha=0.35;
    ctx.beginPath(); ctx.arc(b.x,b.y,b.r*2,0,Math.PI*2); ctx.fill();
    ctx.globalAlpha=1;
  }
  ctx.globalCompositeOperation='source-over';

  // particles
  for(const p of particles){
    ctx.globalAlpha=Math.max(0,p.life/p.max);
    ctx.fillStyle=p.color;
    ctx.beginPath(); ctx.arc(p.x,p.y,p.r,0,Math.PI*2); ctx.fill();
  }
  ctx.globalAlpha=1;

  // player
  if(state==='playing') drawShip(player.x,player.y);

  ctx.restore();

  /* HUD */
  if(state==='playing'){
    ctx.font='bold 18px Courier New';
    ctx.textAlign='left';
    ctx.fillStyle='#cfe9ff';
    ctx.fillText('SCORE '+score, 16, 28);
    ctx.fillStyle='#8fd0ff';
    ctx.fillText('TIME '+timeAlive.toFixed(1)+'s', 16, 50);
    // lives
    for(let i=0;i<3;i++){
      ctx.fillStyle = i<lives ? '#7fe3ff' : 'rgba(80,120,180,.3)';
      const lx = W-90+i*26;
      ctx.beginPath();
      ctx.moveTo(lx+9,18); ctx.lineTo(lx,24); ctx.lineTo(lx+4,27); ctx.lineTo(lx+9,32);
      ctx.lineTo(lx+14,27); ctx.lineTo(lx+18,24); ctx.closePath(); ctx.fill();
    }
    if(player.shield>0){
      ctx.fillStyle='#7fff9e';
      ctx.fillText('SHIELD '+player.shield.toFixed(1)+'s', W-150, 50);
    }
    if(keyDown('shift')){
      ctx.fillStyle='rgba(127,227,255,.8)';
      ctx.font='13px Courier New';
      ctx.textAlign='center';
      ctx.fillText('— FOCUS MODE —', W/2, H-14);
    }
  }
}

/* ---------- main loop ---------- */
let last = performance.now();
function loop(now){
  let dt = (now-last)/1000;
  last = now;
  if(dt>0.05) dt=0.05;
  if(state==='playing') update(dt);
  else { // idle star drift on menus
    for(const s of starField){ s.y+=s.v*dt*0.4; if(s.y>H){s.y=-4;s.x=Math.random()*W;} }
  }
  render();
  requestAnimationFrame(loop);
}
requestAnimationFrame(loop);

})();
