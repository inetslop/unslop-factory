<script>
(function(){
  'use strict';

  // ---------- Audio ----------
  var AC = null;
  function audio(){ if(!AC){ try{ AC = new (window.AudioContext||window.webkitAudioContext)(); }catch(e){} } if(AC && AC.state==='suspended') AC.resume(); return AC; }
  function tone(freq, dur, type, vol, when){
    var a = audio(); if(!a) return;
    var t = a.currentTime + (when||0);
    var o = a.createOscillator(), g = a.createGain();
    o.type = type||'sine'; o.frequency.value = freq;
    g.gain.setValueAtTime(vol||0.15, t);
    g.gain.exponentialRampToValueAtTime(0.001, t + dur);
    o.connect(g); g.connect(a.destination);
    o.start(t); o.stop(t + dur + 0.05);
  }
  function sFlip(){ tone(320,.12,'triangle',.18); tone(520,.1,'sine',.08,.04); }
  function sMatch(){ tone(660,.15,'sine',.2); tone(880,.25,'sine',.2,.1); }
  function sMiss(){ tone(220,.2,'sawtooth',.08); tone(180,.25,'sawtooth',.08,.1); }
  function sWin(){ [523,659,784,1047,1319].forEach(function(f,i){ tone(f,.35,'triangle',.2,i*.13); }); }

  // ---------- Data ----------
  var DIFFS = [
    { key:'easy',   name:'Easy · 4×3',  cols:4, rows:3 },
    { key:'medium', name:'Medium · 6×4',cols:6, rows:4 },
    { key:'hard',   name:'Hard · 8×5',  cols:8, rows:5 }
  ];
  var SUITS = ['♠','♥','♦','♣'];
  var RANKS = ['A','K','Q','J','10','9','8','7','6','5'];

  var state = { diff:DIFFS[0], cards:[], flipped:[], lock:false, moves:0, seconds:0, timer:null, running:false, sel:0 };
  var best = {};
  try{ best = JSON.parse(localStorage.getItem('cardflip-best')||'{}'); }catch(e){ best={}; }
  function saveBest(){ try{ localStorage.setItem('cardflip-best', JSON.stringify(best)); }catch(e){} }

  // ---------- DOM ----------
  var $ = function(id){ return document.getElementById(id); };
  var screens = { start:$('screen-start'), game:$('screen-game'), win:$('screen-win') };
  var board = $('board');
  function show(name){ for(var k in screens) screens[k].classList.toggle('active', k===name); }

  // difficulty buttons
  DIFFS.forEach(function(d,i){
    var b = document.createElement('button');
    b.className='btn'; b.textContent = (i+1)+' · '+d.name;
    b.onclick = function(){ audio(); startGame(d); };
    $('diff-buttons').appendChild(b);
  });

  function fmt(s){ var m=Math.floor(s/60), r=s%60; return (m<10?'0':'')+m+':'+(r<10?'0':'')+r; }
  function bestText(k){
    if(!best[k]) return '—';
    return best[k].moves+' moves · '+fmt(best[k].time);
  }
  function refreshStartBest(){
    var parts = DIFFS.map(function(d){ return d.name.split('·')[0].trim()+': <span>'+bestText(d.key)+'</span>'; });
    $('start-best').innerHTML = '🏅 Best — '+parts.join(' &nbsp;|&nbsp; ');
  }
  refreshStartBest();

  // ---------- Game ----------
  function buildDeck(diff){
    var pairs = diff.cols*diff.rows/2, faces=[];
    for(var i=0;i<pairs;i++){
      var f = { rank:RANKS[i%10], suit:SUITS[Math.floor(i/10)%4], id:'p'+i };
      faces.push(f,f);
    }
    for(var j=faces.length-1;j>0;j--){ var k=Math.floor(Math.random()*(j+1)); var t=faces[j]; faces[j]=faces[k]; faces[k]=t; }
    return faces;
  }

  function startGame(diff){
    state.diff = diff||state.diff;
    state.cards = buildDeck(state.diff);
    state.flipped=[]; state.lock=false; state.moves=0; state.seconds=0;
    state.running=false; state.sel=0;
    if(state.timer){ clearInterval(state.timer); state.timer=null; }
    $('hud-moves').textContent='0'; $('hud-time').textContent='00:00';
    $('hud-pairs').textContent='0/'+(state.cards.length/2);

    board.innerHTML='';
    board.style.gridTemplateColumns = 'repeat('+state.diff.cols+', var(--cell))';
    state.cards.forEach(function(f,i){
      var c=document.createElement('div');
      c.className='card'; c.dataset.i=i;
      var red = (f.suit==='♥'||f.suit==='♦')?'red':'black';
      c.innerHTML =
        '<div class="inner">'+
          '<div class="face back"></div>'+
          '<div class="face front '+red+'">'+
            '<div class="corner">'+f.rank+'<br>'+f.suit+'</div>'+
            '<div class="pip">'+f.suit+'</div>'+
            '<div class="corner bot">'+f.rank+'<br>'+f.suit+'</div>'+
          '</div>'+
        '</div>';
      c.addEventListener('click', function(){ audio(); flip(i); });
      board.appendChild(c);
    });
    layout();
    markSel();
    show('game');
  }

  function layout(){
    var cols=state.diff.cols, rows=state.diff.rows;
    if(!screens.game.classList.contains('active')) return;
    var hudH = $('hud').offsetHeight + 12;
    var availW = window.innerWidth - 40;
    var availH = window.innerHeight - hudH - 30;
    var gap = Math.max(4, Math.min(10, availW/(cols*2)));
    var cell = Math.min((availW - gap*(cols-1))/cols, (availH - gap*(rows-1))/(rows*1.45));
    cell = Math.max(30, Math.floor(cell));
    board.style.setProperty('--cell', cell+'px');
    board.style.setProperty('--gap', gap+'px');
  }
  window.addEventListener('resize', layout);

  function markSel(){
    var kids=board.children;
    for(var i=0;i<kids.length;i++) kids[i].classList.toggle('sel', i===state.sel && !screens.win.classList.contains('active'));
  }

  function startTimer(){
    if(state.running) return;
    state.running=true;
    state.timer=setInterval(function(){
      state.seconds++; $('hud-time').textContent=fmt(state.seconds);
    },1000);
  }

  function flip(i){
    if(state.lock) return;
    var card = state.cards[i], el = board.children[i];
    if(!card || el.classList.contains('flipped')||el.classList.contains('matched')) return;
    markSel();
    el.classList.add('flipped');
    sFlip();
    startTimer();
    state.flipped.push(i);

    if(state.flipped.length===2){
      state.moves++; $('hud-moves').textContent=state.moves;
      var a=state.cards[state.flipped[0]], b=state.cards[state.flipped[1]];
      if(a.id===b.id){
        board.children[state.flipped[0]].classList.add('matched');
        board.children[state.flipped[1]].classList.add('matched');
        sMatch();
        state.flipped=[];
        var done = board.querySelectorAll('.matched').length/2;
        $('hud-pairs').textContent=done+'/'+(state.cards.length/2);
        if(done===state.cards.length/2) setTimeout(win, 500);
      } else {
        state.lock=true; sMiss();
        var f=state.flipped.slice();
        setTimeout(function(){
          board.children[f[0]].classList.remove('flipped');
          board.children[f[1]].classList.remove('flipped');
          state.flipped=[]; state.lock=false;
        }, 750);
      }
    }
  }

  function win(){
    if(state.timer){ clearInterval(state.timer); state.timer=null; }
    sWin();
    var d=state.diff, isBest=false;
    var old=best[d.key];
    if(!old || state.moves<old.moves || (state.moves===old.moves && state.seconds<old.time)){
      best[d.key]={ moves:state.moves, time:state.seconds }; saveBest(); isBest=true;
    }
    $('win-stats').innerHTML =
      'Difficulty: <b>'+d.name+'</b><br>'+
      'Moves: <b>'+state.moves+'</b> &nbsp;·&nbsp; Time: <b>'+fmt(state.seconds)+'</b><br>'+
      (isBest ? '<span class="newbest">★ NEW BEST! ★</span>' : 'Best: '+bestText(d.key));
    refreshStartBest();
    show('win');
  }

  // ---------- Keyboard ----------
  window.addEventListener('keydown', function(e){
    audio();
    var k=e.key;
    if(['ArrowUp','ArrowDown','ArrowLeft','ArrowRight',' '].indexOf(k)>=0) e.preventDefault();

    if(screens.start.classList.contains('active')){
      if(k==='1') startGame(DIFFS[0]);
      else if(k==='2') startGame(DIFFS[1]);
      else if(k==='3') startGame(DIFFS[2]);
      return;
    }
    if(screens.win.classList.contains('active')){
      if(k==='r'||k==='R'||k==='Enter'){ startGame(state.diff); }
      else if(k==='m'||k==='M'||k==='Escape'){ show('start'); refreshStartBest(); }
      return;
    }
    // game screen
    var n=state.cards.length, cols=state.diff.cols;
    if(k==='ArrowRight') state.sel=(state.sel+1)%n;
    else if(k==='ArrowLeft') state.sel=(state.sel-1+n)%n;
    else if(k==='ArrowDown'){ state.sel=(state.sel+cols)%n; }
    else if(k==='ArrowUp'){ state.sel=(state.sel-cols+n)%n; }
    else if(k==='Enter'||k===' ') flip(state.sel);
    else if(k==='r'||k==='R') startGame(state.diff);
    else if(k==='m'||k==='M'||k==='Escape'){ show('start'); refreshStartBest(); return; }
    markSel();
  });

  // ---------- Buttons ----------
  $('btn-restart').onclick=function(){ audio(); startGame(state.diff); };
  $('btn-menu').onclick=function(){ audio(); show('start'); refreshStartBest(); };
  $('btn-again').onclick=function(){ audio(); startGame(state.diff); };
  $('btn-winmenu').onclick=function(){ audio(); show('start'); refreshStartBest(); };

  // unlock audio on first pointer input too
  window.addEventListener('pointerdown', function once(){ audio(); }, {once:false});
})();
</script>
</body>
</html>
