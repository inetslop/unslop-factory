  const g=ctx.createLinearGradient(0,-h/2,0,h/2);
  if(isMineRevealed){ g.addColorStop(0,'#7a3a48'); g.addColorStop(1,'#54262f'); }
  else{ g.addColorStop(0,'#3d4d72'); g.addColorStop(1,'#2b3752'); }
  rr(-w/2,-h/2,w,h,r); ctx.fillStyle=g; ctx.fill();
  ctx.lineWidth=1.5; ctx.strokeStyle=isMineRevealed?'#a05568':'#55689c'; ctx.stroke();
  rr(-w/2+3,-h/2+3,w-6,4,2); ctx.fillStyle='rgba(255,255,255,.10)'; ctx.fill();

  if(isMineRevealed){
    const R=w*.26;
    ctx.strokeStyle='#151118'; ctx.lineWidth=Math.max(2,w*.07);
    for(let k=0;k<4;k++){ const a=k*Math.PI/4;
      ctx.beginPath(); ctx.moveTo(Math.cos(a)*R,Math.sin(a)*R);
      ctx.lineTo(-Math.cos(a)*R,-Math.sin(a)*R); ctx.stroke(); }
    ctx.beginPath(); ctx.arc(0,0,R,0,7); ctx.fillStyle='#151118'; ctx.fill();
    ctx.beginPath(); ctx.arc(-R*.3,-R*.3,R*.24,0,7); ctx.fillStyle='rgba(255,255,255,.5)'; ctx.fill();
  } else if(t.flagged){
    const s=w;
    ctx.strokeStyle='#e8ecf5'; ctx.lineWidth=Math.max(2,s*.06);
    ctx.beginPath(); ctx.moveTo(s*.04,s*.30); ctx.lineTo(s*.04,-s*.18); ctx.stroke();
    ctx.fillStyle='#ff5d5d'; ctx.beginPath();
    ctx.moveTo(s*.04,-s*.18); ctx.lineTo(s*.34,-s*.06); ctx.lineTo(s*.04,s*.05); ctx.closePath(); ctx.fill();
    ctx.strokeStyle='#e8ecf5'; ctx.beginPath();
    ctx.moveTo(-s*.12,s*.30); ctx.lineTo(s*.2,s*.30); ctx.stroke();
  } else if(t.num>0 && adj[t.i].some(n=>tiles[n].removed)){
    ctx.fillStyle=NUMCOL[t.num]||'#fff';
    ctx.font='700 '+Math.floor(w*.48)+'px "Segoe UI",Arial,sans-serif';
    ctx.textAlign='center'; ctx.textBaseline='middle';
    ctx.fillText(t.num,0,w*.03);
  }
  ctx.restore();
}

function draw(){
  ctx.clearRect(0,0,W,H);
  ctx.save();
  if(shake>0) ctx.translate((Math.random()-.5)*shake,(Math.random()-.5)*shake);
  // playfield
  rr(ox-12,oy-12,COLS*cell+24,ROWS*cell+24,14);
  ctx.fillStyle='#1c2336'; ctx.fill();
  ctx.lineWidth=2; ctx.strokeStyle='#2c3852'; ctx.stroke();
  // dug-out holes at original spots
  for(const t of tiles){ if(!t.removed)continue;
    ctx.fillStyle='rgba(0,0,0,.30)';
    rr(ox+t.c*cell+PAD,oy+t.r*cell+PAD,t.w,t.h,4); ctx.fill(); }
  // floor line
  ctx.strokeStyle='#3a4a6e'; ctx.lineWidth=2;
  ctx.beginPath(); ctx.moveTo(ox-12,floorY+8); ctx.lineTo(ox+COLS*cell+12,floorY+8); ctx.stroke();
  for(const t of tiles) if(!t.removed) drawTile(t);
  for(const p of particles){ ctx.globalAlpha=Math.max(0,p.life/p.max);
    ctx.fillStyle=p.col; ctx.fillRect(p.x-p.size/2,p.y-p.size/2,p.size,p.size); }
  ctx.globalAlpha=1;
  ctx.restore();
}

function frame(now){
  const dt=Math.min(.033,(now-last)/1000); last=now;
  if(state==='playing'){ time+=dt; hTime.textContent=fmt(time); }
  physics(dt); updateParticles(dt);
  shake=Math.max(0,shake-45*dt);
  draw();
  requestAnimationFrame(frame);
}
