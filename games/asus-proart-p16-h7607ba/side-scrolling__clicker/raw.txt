<script>
(() => {
"use strict";
const canvas = document.getElementById('game');
const ctx = canvas.getContext('2d');
let W = 0, H = 0, DPR = 1;

function resize(){
  DPR = Math.min(window.devicePixelRatio || 1, 2);
  W = window.innerWidth; H = window.innerHeight;
  canvas.width = W * DPR; canvas.height = H * DPR;
  ctx.setTransform(DPR,0,0,DPR,0,0);
}
window.addEventListener('resize', resize);
resize();

// ---------- Audio (starts after first input) ----------
let actx = null;
function ensureAudio(){
  if(!actx){ try { actx = new (window.AudioContext||window.webkitAudioContext)(); } catch(e){} }
  if(actx && actx.state === 'suspended') actx.resume();
}
function beep(freq, dur, type, vol, slide){
  if(!actx) return;
  const t = actx.currentTime;
  const o = actx.createOscillator(), g = actx.createGain();
  o.type = type || 'sine';
  o.frequency.setValueAtTime(freq, t);
  if(slide) o.frequency.exponentialRampToValueAtTime(Math.max(40, slide), t+dur);
  g.gain.setValueAtTime(vol||0.15, t);
  g.gain.exponentialRampToValueAtTime(0.0001, t+dur);
  o.connect(g); g.connect(actx.destination);
  o.start(t); o.stop(t+dur+0.02);
}
function noiseBurst(dur, vol){
  if(!actx) return;
  const t = actx.currentTime;
  const len = Math.floor(actx.sampleRate * dur);
  const buf = actx.createBuffer(1, len, actx.sampleRate);
  const d = buf.getChannelData(0);
  for(let i=0;i<len;i++) d[i] = (Math.random()*2-1) * (1 - i/len);
  const src = actx.createBufferSource(); src.buffer = buf;
  const g = actx.createGain(); g.gain.setValueAtTime(vol, t);
  g.gain.exponentialRampToValueAtTime(0.0001, t+dur);
  const f = actx.createBiquadFilter(); f.type='lowpass'; f.frequency.value=900;
  src.connect(f); f.connect(g); g.connect(actx.destination);
  src.start(t);
}

// ---------- State ----------
const state = { running:false, score:0, lives:3, combo:0, maxCombo:0, time:0, spawnTimer:0, best:0 };
let entities = [], particles = [], floaters = [], stars = [];

try { state.best = parseInt(localStorage.getItem('cometClickerBest')||'0',10) || 0; } catch(e){}

function initStars(){
  stars = [];
  const n = Math.floor(W*H/6000);
  for(let i=0;i<n;i++) stars.push({
    x:Math.random()*W, y:Math.random()*H,
    z:Math.random(), tw:Math.random()*Math.PI*2
  });
}
initStars();

function difficulty(){ return Math.min(1, state.time/90); } // ramps over 90s
function baseSpeed(){ return (W*0.18 + W*0.55*difficulty()); } // px/sec

function spawnEntity(){
  const r = Math.random();
  let type;
  if(r < 0.08) type = 'gem';
  else if(r < 0.26 + 0.14*difficulty()) type = 'bomb';
  else type = 'comet';
  const size = type==='gem' ? 16+Math.random()*8 : type==='bomb' ? 20+Math.random()*12 : 22+Math.random()*30;
  entities.push({
    type, x: W + size*2, y: size + Math.random()*(H - size*2),
    r:size/2, baseSize:size,
    speed: baseSpeed() * (type==='gem' ? 1.4 : 0.85 + Math.random()*0.5),
    wobble: Math.random()*Math.PI*2,
    wobbleAmp: type==='bomb' ? 0 : 8+Math.random()*20,
    rot: Math.random()*Math.PI*2,
    rotSpd:(Math.random()-0.5)*3,
    dead:false
  });
}

function burst(x,y,color,n,speed){
  for(let i=0;i<n;i++){
    const a = Math.random()*Math.PI*2, s = speed*(0.3+Math.random());
    particles.push({x,y,vx:Math.cos(a)*s,vy:Math.sin(a)*s,life:1,color,r:2+Math.random()*4});
  }
}
function floater(x,y,text,color){
  floaters.push({x,y,text,color,life:1});
}

// ---------- Input ----------
function pointerPos(e){
  const rect = canvas.getBoundingClientRect();
  const src = e.touches ? e.touches[0] : e;
  return { x: src.clientX - rect.left, y: src.clientY - rect.top };
}
function handleClick(p){
  ensureAudio();
  if(!state.running) return;
  // topmost entity first
  for(let i=entities.length-1;i>=0;i--){
    const e = entities[i];
    if(e.dead) continue;
    const dx=p.x-e.x, dy=p.y-e.y;
    if(dx*dx+dy*dy <= (e.r+14)*(e.r+14)){
      hitEntity(e);
      return;
    }
  }
  // whiff: small feedback
  burst(p.x,p.y,'#3b4a6b',5,60);
}
function hitEntity(e){
  e.dead = true;
  if(e.type === 'bomb'){
    state.lives--;
    state.combo = 0;
    beep(90,0.35,'sawtooth',0.25,40);
    noiseBurst(0.3,0.3);
    burst(e.x,e.y,'#f87171',26,260);
    burst(e.x,e.y,'#fbbf24',18,180);
    floater(e.x,e.y,'BOOM! -1 ♥','#f87171');
    shake = 14;
    if(state.lives <= 0) endGame();
  } else {
    state.combo++;
    state.maxCombo = Math.max(state.maxCombo, state.combo);
    const mult = 1 + Math.floor(state.combo/5);
    const pts = (e.type==='gem' ? 50 : Math.round(e.baseSize/2)) * mult;
    state.score += pts;
    beep(e.type==='gem' ? 880 : 500+Math.random()*200, 0.12, 'triangle', 0.18, e.type==='gem'?1600:900);
    if(state.combo % 5 === 0){ beep(1320,0.2,'sine',0.15,1760); floater(e.x,e.y-30,'COMBO ×'+mult+'!','#c084fc'); }
    burst(e.x,e.y, e.type==='gem' ? '#a5f3fc' : '#fbbf24', 16, 190);
    floater(e.x,e.y,'+'+pts, e.type==='gem' ? '#67e8f9' : '#fde68a');
  }
}
canvas.addEventListener('mousedown', e => handleClick(pointerPos(e)));
canvas.addEventListener('touchstart', e => { e.preventDefault(); handleClick(pointerPos(e)); }, {passive:false});

function startGame(){
  ensureAudio();
  state.running = true;
  state.score = 0; state.lives = 3; state.combo = 0; state.maxCombo = 0;
  state.time = 0; state.spawnTimer = 0;
  entities = []; particles = []; floaters = [];
  document.getElementById('startScreen').classList.add('hidden');
  document.getElementById('overScreen').classList.add('hidden');
  beep(440,0.15,'sine',0.12,880);
}
function endGame(){
  state.running = false;
  let savedBest = state.best;
  if(state.score > state.best){
    state.best = state.score;
    try { localStorage.setItem('cometClickerBest', String(state.best)); } catch(e){}
    savedBest = 'NEW BEST!';
  }
  document.getElementById('finalScore').textContent = state.score + ' pts' + (state.maxCombo>0 ? '  ·  max combo ×'+(1+Math.floor(state.maxCombo/5)) : '');
  document.getElementById('bestLine').textContent = savedBest === 'NEW BEST!' ? '🏆 New record! 🏆' : 'Best: ' + state.best;
  setTimeout(()=> document.getElementById('overScreen').classList.remove('hidden'), 500);
  beep(220,0.6,'sawtooth',0.2,60);
}
document.getElementById('startBtn').addEventListener('click', startGame);
document.getElementById('restartBtn').addEventListener('click', startGame);
window.addEventListener('keydown', e => {
  if(e.code === 'Space' || e.code === 'Enter'){
    e.preventDefault();
    ensureAudio();
    if(!state.running) startGame();
  }
});

// ---------- Update & Draw ----------
let last = performance.now(), shake = 0;

function update(dt){
  state.time += dt;
  // stars drift
  for(const s of stars){
    s.x -= (20 + s.z*90) * (1+difficulty()) * dt;
    s.tw += dt*3;
    if(s.x < -4){ s.x = W+4; s.y = Math.random()*H; }
  }
  if(state.running){
    state.spawnTimer -= dt;
    const interval = Math.max(0.28, 1.1 - difficulty()*0.75);
    if(state.spawnTimer <= 0){
      spawnEntity();
      if(difficulty() > 0.4 && Math.random() < 0.3) spawnEntity();
      state.spawnTimer = interval * (0.6 + Math.random()*0.8);
    }
  }
  // entities
  for(const e of entities){
    e.x -= e.speed*dt;
    e.wobble += dt*2;
    e.rot += e.rotSpd*dt;
    if(!e.dead && e.x < -e.r*2){
      e.dead = true;
      if(state.running && e.type !== 'bomb'){
        state.lives--;
        state.combo = 0;
        beep(160,0.25,'square',0.15,80);
        floater(30, e.y, 'MISSED! -1 ♥', '#f87171');
        shake = Math.max(shake, 8);
        if(state.lives <= 0) endGame();
      }
    }
  }
  entities = entities.filter(e => !e.dead && e.x > -e.r*3);

  for(const p of particles){
    p.x += p.vx*dt; p.y += p.vy*dt;
    p.vx *= (1-2*dt); p.vy *= (1-2*dt);
    p.life -= dt*1.8;
  }
  particles = particles.filter(p => p.life > 0);
  for(const f of floaters){ f.y -= 45*dt; f.life -= dt*1.1; }
  floaters = floaters.filter(f => f.life > 0);
  shake = Math.max(0, shake - dt*40);
}

function drawComet(e){
  ctx.save();
  // trail
  const grad = ctx.createLinearGradient(e.x - e.r*6, e.y, e.x, e.y);
  grad.addColorStop(0,'rgba(251,191,36,0)');
  grad.addColorStop(1,'rgba(251,191,36,0.35)');
  ctx.fillStyle = grad;
  ctx.beginPath();
  ctx.moveTo(e.x - e.r*6, e.y);
  ctx.lineTo(e.x, e.y - e.r*0.8);
  ctx.lineTo(e.x, e.y + e.r*0.8);
  ctx.closePath(); ctx.fill();
  // body
  ctx.translate(e.x,e.y); ctx.rotate(e.rot);
  const g = ctx.createRadialGradient(-e.r*0.3,-e.r*0.3,1,0,0,e.r);
  g.addColorStop(0,'#fef3c7'); g.addColorStop(0.5,'#fbbf24'); g.addColorStop(1,'#b45309');
  ctx.fillStyle = g;
  ctx.beginPath(); ctx.arc(0,0,e.r,0,Math.PI*2); ctx.fill();
  // craters
  ctx.fillStyle='rgba(120,53,15,0.5)';
  ctx.beginPath(); ctx.arc(e.r*0.3,-e.r*0.2,e.r*0.22,0,Math.PI*2); ctx.fill();
  ctx.beginPath(); ctx.arc(-e.r*0.25,e.r*0.3,e.r*0.16,0,Math.PI*2); ctx.fill();
  ctx.restore();
}
function drawGem(e){
  const pulse = 1 + Math.sin(e.wobble*3)*0.12;
  ctx.save();
  ctx.translate(e.x,e.y); ctx.rotate(Math.sin(e.wobble)*0.4);
  const r = e.r*pulse;
  ctx.shadowColor='#67e8f9'; ctx.shadowBlur=25;
  const g = ctx.createLinearGradient(-r,-r,r,r);
  g.addColorStop(0,'#a5f3fc'); g.addColorStop(1,'#0891b2');
  ctx.fillStyle = g;
  ctx.beginPath();
  ctx.moveTo(0,-r); ctx.lineTo(r*0.7,0); ctx.lineTo(0,r); ctx.lineTo(-r*0.7,0);
  ctx.closePath(); ctx.fill();
  ctx.strokeStyle='rgba(255,255,255,0.8)'; ctx.lineWidth=2; ctx.stroke();
  ctx.restore();
}
function drawBomb(e){
  const pulse = 1 + Math.sin(e.wobble*6)*0.08;
  ctx.save();
  ctx.translate(e.x,e.y);
  // fuse
  ctx.strokeStyle='#a8a29e'; ctx.lineWidth=3;
  ctx.beginPath(); ctx.moveTo(0,-e.r*pulse);
  ctx.quadraticCurveTo(e.r*0.6,-e.r*pulse-14, e.r*0.9,-e.r*pulse-10); ctx.stroke();
  const spark = Math.random();
  ctx.fillStyle = spark>0.5 ? '#fde047' : '#fb923c';
  ctx.beginPath(); ctx.arc(e.r*0.9,-e.r*pulse-10, 3+spark*4, 0, Math.PI*2); ctx.fill();
  // body
  const g = ctx.createRadialGradient(-e.r*0.3,-e.r*0.3,1,0,0,e.r*pulse);
  g.addColorStop(0,'#6b7280'); g.addColorStop(1,'#111827');
  ctx.fillStyle = g;
  ctx.beginPath(); ctx.arc(0,0,e.r*pulse,0,Math.PI*2); ctx.fill();
  // skull-ish X mark
  ctx.strokeStyle='#f87171'; ctx.lineWidth=3;
  const m = e.r*0.45;
  ctx.beginPath();
  ctx.moveTo(-m,-m); ctx.lineTo(m,m);
  ctx.moveTo(m,-m); ctx.lineTo(-m,m);
  ctx.stroke();
  ctx.restore();
}

function draw(){
  ctx.save();
  if(shake > 0) ctx.translate((Math.random()-0.5)*shake, (Math.random()-0.5)*shake);
  // bg
  const bg = ctx.createLinearGradient(0,0,0,H);
  bg.addColorStop(0,'#0a0920'); bg.addColorStop(0.6,'#0d1030'); bg.addColorStop(1,'#171040');
  ctx.fillStyle = bg;
  ctx.fillRect(-20,-20,W+40,H+40);

  for(const s of stars){
    const a = 0.3 + 0.7*Math.abs(Math.sin(s.tw));
    ctx.globalAlpha = a * (0.4 + s.z*0.6);
    ctx.fillStyle = '#fff';
    ctx.fillRect(s.x, s.y, s.z*2.5, s.z*2.5);
  }
  ctx.globalAlpha = 1;

  for(const e of entities){
    if(e.type==='comet') drawComet(e);
    else if(e.type==='gem') drawGem(e);
    else drawBomb(e);
  }
  for(const p of particles){
    ctx.globalAlpha = Math.max(0,p.life);
    ctx.fillStyle = p.color;
    ctx.beginPath(); ctx.arc(p.x,p.y,p.r*p.life,0,Math.PI*2); ctx.fill();
  }
  ctx.globalAlpha = 1;

  // floaters
  for(const f of floaters){
    ctx.globalAlpha = Math.max(0,f.life);
    ctx.font = 'bold 20px Segoe UI, sans-serif';
    ctx.textAlign='center';
    ctx.fillStyle = f.color;
    ctx.shadowColor='rgba(0,0,0,0.7)'; ctx.shadowBlur=6;
    ctx.fillText(f.text, f.x, f.y);
    ctx.shadowBlur=0;
  }
  ctx.globalAlpha = 1;

  // HUD
  if(state.running || state.score > 0){
    ctx.textAlign='left';
    ctx.font='bold ' + Math.max(20, W*0.03) + 'px Segoe UI, sans-serif';
    ctx.fillStyle='#fde68a';
    ctx.shadowColor='rgba(0,0,0,0.6)'; ctx.shadowBlur=8;
    ctx.fillText('SCORE  ' + state.score, 20, 42);
    if(state.combo >= 3){
      ctx.fillStyle='#c084fc';
      ctx.font='bold 16px Segoe UI, sans-serif';
      ctx.fillText('COMBO ×' + (1+Math.floor(state.combo/5)) + ' (' + state.combo + ')', 20, 70);
    }
    // lives
    ctx.textAlign='right';
    ctx.font = Math.max(20, W*0.03) + 'px Segoe UI, sans-serif';
    let hearts = '';
    for(let i=0;i<3;i++) hearts += i < state.lives ? '♥ ' : '♡ ';
    ctx.fillStyle = state.lives <= 1 ? '#f87171' : '#f472b6';
    ctx.fillText(hearts.trim(), W-20, 42);
    // speed bar
    const bw = Math.min(200, W*0.3), bx = W/2 - bw/2;
    ctx.shadowBlur=0;
    ctx.fillStyle='rgba(255,255,255,0.15)';
    ctx.fillRect(bx, 16, bw, 8);
    const sg = ctx.createLinearGradient(bx,0,bx+bw,0);
    sg.addColorStop(0,'#38bdf8'); sg.addColorStop(1,'#f43f5e');
    ctx.fillStyle = sg;
    ctx.fillRect(bx, 16, bw*difficulty(), 8);
    ctx.font='11px Segoe UI, sans-serif';
    ctx.textAlign='center'; ctx.fillStyle='#7c8db5';
    ctx.fillText('SPEED', W/2, 38);
    ctx.shadowBlur=0;
  }
  ctx.restore();
}

function loop(now){
  const dt = Math.min(0.05, (now-last)/1000);
  last = now;
  update(dt);
  draw();
  requestAnimationFrame(loop);
}
requestAnimationFrame(loop);
})();
</script>
