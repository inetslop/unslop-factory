function update(dt){
  G.time+=dt;
  if(G.state!=="play")return;
  if(G.moveTimer>0)G.moveTimer-=dt;
  if(G.msgFlashT>0)G.msgFlashT-=dt;

    if(d<sightRadius(g)&&G.moveTimer>0)seen=true;

  else G.seenTime=Math.max(0,G.seenTime-dt*1.4);

    var pt=G.particles[p];pt.y+=pt.vy*dt;pt.life-=dt;
