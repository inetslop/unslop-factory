  document.getElementById('overOv').classList.remove('hidden');
}

/* ---------- PROJECTION ---------- */
function camBasis(){
  const cy=Math.cos(yaw), sy=Math.sin(yaw), cp=Math.cos(pitch), sp=Math.sin(pitch);
  const f={x:sy*cp,y:sp,z:cy*cp};
  const r={x:cy,y:0,z:-sy};
  return {f,r,u:{x:f.y*r.z, y:f.z*r.x-f.x*r.z, z:-f.y*r.x}};
}
function toCam(b,x,y,z){
  const dx=x-player.x, dy=y-1.6, dz=z-player.z;
  return { x:dx*b.r.x+dy*b.r.y+dz*b.r.z,
           y:dx*b.u.x+dy*b.u.y+dz*b.u.z,
           z:dx*b.f.x+dy*b.f.y+dz*b.f.z };
}
function proj(c){
  if(c.z<0.15) return null;
  return {x:CX+F*c.x/c.z, y:CY-F*c.y/c.z, s:F/c.z};
}
function fogA(dist){ return Math.max(0,Math.min(1,1-dist/70)); }

/* ---------- RENDER ---------- */
function render(){
  const b=camBasis();
  // sky
  const g=ctx.createLinearGradient(0,0,0,H);
  g.addColorStop(0,'#050018'); g.addColorStop(.55,'#12043a'); g.addColorStop(1,'#2a0a4a');
  ctx.fillStyle=g; ctx.fillRect(0,0,W,H);
  // stars
  ctx.save();
  for(let i=0;i<70;i++){
    const sx=((i*997)%1000)/1000, sy=((i*613)%620)/1000;
    let px=(sx+.5)*W - CX, py=-sy*H*.8 + CY*0.3;
    // simple rotate by yaw for parallax feel
    const ang=yaw*.7;
    const rx=px*Math.cos(ang)-py*Math.sin(ang), ry=px*Math.sin(ang)+py*Math.cos(ang);
    px=rx+CX; py=ry+CY-H*.15;
    if(py>0&&py<H*.62&&px>-10&&px<W+10){
      ctx.globalAlpha=.3+.5*((i*37)%10)/10;
      ctx.fillStyle='#fff';
      ctx.fillRect(px,py,i%5===0?2:1,i%5===0?2:1);
    }
  }
  ctx.restore();

  // floor grid
  const step=4;
  ctx.lineWidth=1;
  for(let i=-ARENA;i<=ARENA;i+=step){
    gridLine(i,0,i,ARENA*3,false);
    gridLine(-ARENA*3,i,ARENA*3,i,true);
  }
  function gridLine(x1,z1,x2,z2,main){
    const a=toCam(b,x1,0,z1), c=toCam(b,x2,0,z2);
    let p1=a,q1=c;
    // near clip
    if(p1.z<0.15&&q1.z<0.15) return;
    const t=(p1.z<0.15)?(0.15-p1.z)/(q1.z-p1.z):((q1.z<0.15)?(0.15-q1.z)/(p1.z-q1.z):null);
    if(t!==null){
      const nx=p1.x+(q1.x-p1.x)*t, ny=p1.y+(q1.y-p1.y)*t, nz=.15;
      if(p1.z<0.15) p1={x:nx,y:ny,z:nz}; else q1={x:nx,y:ny,z:nz};
    }
    const P=proj(p1),Q=proj(q1);
    if(!P||!Q) return;
    const dist=Math.hypot((x1+x2)/2-player.x,(z1+z2)/2-player.z);
    ctx.strokeStyle='rgba(0,255,255,'+(fogA(dist)*(main?.28:.16)).toFixed(3)+')';
    ctx.beginPath(); ctx.moveTo(P.x,P.y); ctx.lineTo(Q.x,Q.y); ctx.stroke();
  }

  // walls (triangulated, clipped)
  const hgt=7;
  const corners=[[-ARENA,-ARENA],[ARENA,-ARENA],[ARENA,ARENA],[-ARENA,ARENA]];
  for(let i=0;i<4;i++){
    const A=corners[i],B=corners[(i+1)%4];
    const pts=[toCam(b,A[0],0,A[1]),toCam(b,B[0],0,B[1]),toCam(b,B[0],hgt,B[1]),toCam(b,A[0],hgt,A[1])];
    for(const tri of [[pts[0],pts[1],pts[2]],[pts[0],pts[2],pts[3]]]){
      if(tri.some(p=>p.z<0.15)) continue;
      const P=tri.map(proj); if(P.some(p=>!p)) continue;
      const dist=Math.hypot((A[0]+B[0])/2-player.x,(A[1]+B[1])/2-player.z);
      ctx.fillStyle='rgba(40,0,80,'+(fogA(dist)*.75).toFixed(3)+')';
      ctx.strokeStyle='rgba(255,0,255,'+(fogA(dist)*.8).toFixed(3)+')';
      ctx.lineWidth=1.5;
      ctx.beginPath(); ctx.moveTo(P[0].x,P[0].y); ctx.lineTo(P[1].x,P[1].y); ctx.lineTo(P[2].x,P[2].y);
      ctx.closePath(); ctx.fill(); ctx.stroke();
    }
  }

  // tracers
  for(const t of tracers){
    const P=proj(toCam(b,t.x1,t.y1,t.z1)), Q=proj(toCam(b,t.x2,t.y2,t.z2));
    if(P&&Q){
      ctx.strokeStyle='rgba(180,255,255,'+(t.life/.08).toFixed(2)+')';
      ctx.lineWidth=2;
      ctx.beginPath(); ctx.moveTo(P.x,P.y); ctx.lineTo(Q.x,Q.y); ctx.stroke();
    }
  }

  // enemies (depth sorted)
  const list=enemies.map(e=>({e,d:(e.x-player.x)**2+(e.z-player.z)**2})).sort((a,b)=>b.d-a.d);
  for(const {e} of list){
    const c=toCam(b,e.x,e.y||1.2,e.z), P=proj(c);
    if(!P) continue;
    const r=Math.max(2,P.s*e.r), fa=fogA(Math.sqrt((e.x-player.x)**2+(e.z-player.z)**2));
    ctx.save(); ctx.globalAlpha=fa;
    // glow
    const rg=ctx.createRadialGradient(P.x,P.y,r*.2,P.x,P.y,r*2.1);
    rg.addColorStop(0,'rgba(255,40,120,.5)'); rg.addColorStop(1,'rgba(255,40,120,0)');
    ctx.fillStyle=rg; ctx.beginPath(); ctx.arc(P.x,P.y,r*2.1,0,7); ctx.fill();
    // body
    const bg=ctx.createRadialGradient(P.x-r*.3,P.y-r*.3,r*.1,P.x,P.y,r);
    if(e.flash>0){ bg.addColorStop(0,'#fff'); bg.addColorStop(1,'#ff88bb'); }
    else { bg.addColorStop(0,'#ff77aa'); bg.addColorStop(.6,'#c81050'); bg.addColorStop(1,'#4a0020'); }
    ctx.fillStyle=bg; ctx.beginPath(); ctx.arc(P.x,P.y,r,0,7); ctx.fill();
    // eyes
    const ew=r*.32, eh=r*.55*(1+Math.sin(performance.now()/200+e.phase)*.15);
    ctx.fillStyle='#fff';
    ctx.beginPath(); ctx.ellipse(P.x-r*.32,P.y-r*.15,ew*.6,eh,0,0,7); ctx.fill();
    ctx.beginPath(); ctx.ellipse(P.x+r*.32,P.y-r*.15,ew*.6,eh,0,0,7); ctx.fill();
    ctx.fillStyle='#f0f';
    ctx.beginPath(); ctx.arc(P.x-r*.32,P.y-r*.15,ew*.28,0,7); ctx.fill();
    ctx.beginPath(); ctx.arc(P.x+r*.32,P.y-r*.15,ew*.28,0,7); ctx.fill();
    // hp bar
    if(e.hp<e.maxHp){
      const bw=r*2;
      ctx.fillStyle='rgba(0,0,0,.6)'; ctx.fillRect(P.x-bw/2,P.y-r-10,bw,5);
      ctx.fillStyle='#0f8'; ctx.fillRect(P.x-bw/2,P.y-r-10,bw*Math.max(0,e.hp/e.maxHp),5);
    }
    ctx.restore();
  }

  // particles
  for(const p of particles){
    const P=proj(toCam(b,p.x,p.y,p.z));
    if(!P) continue;
    const a=p.life/p.max, s=Math.max(1,P.s*.08);
    ctx.fillStyle='hsla('+p.hue+',100%,65%,'+a.toFixed(2)+')';
    ctx.beginPath(); ctx.arc(P.x,P.y,s,0,7); ctx.fill();
  }

  drawGun(b);
  drawHUD();

  // damage flash
  if(dmgFlash>0){
    ctx.fillStyle='rgba(255,0,40,'+(dmgFlash*.35).toFixed(3)+')';
    ctx.fillRect(0,0,W,H);
  }
  if(state==='play'&&player.hp<35){
    const v=ctx.createRadialGradient(CX,CY,H*.3,CX,CY,H*.75);
    v.addColorStop(0,'rgba(255,0,0,0)'); v.addColorStop(1,'rgba(180,0,20,'+(.4*(1-player.hp/35)).toFixed(3)+')');
    ctx.fillStyle=v; ctx.fillRect(0,0,W,H);
  }
}

function drawGun(b){
  const bobX=Math.sin(bobPhase)*6, bobY=Math.abs(Math.cos(bobPhase))*8;
  const rec=recoil*16;
  const gx=CX+W*.2+bobX, gy=H- H*.24 + bobY + rec;
  ctx.save();
  // muzzle flash
  if(muzzleT>0){
    const fx=gx-W*.055, fy=gy-H*.16;
    const mg=ctx.createRadialGradient(fx,fy,2,fx,fy,40+Math.random()*25);
    mg.addColorStop(0,'rgba(255,255,220,.95)'); mg.addColorStop(.4,'rgba(120,220,255,.7)'); mg.addColorStop(1,'rgba(0,150,255,0)');
    ctx.fillStyle=mg; ctx.beginPath(); ctx.arc(fx,fy,65,0,7); ctx.fill();
  }
  // barrel
  ctx.translate(gx,gy); ctx.rotate(-.28+recoil*.05);
  ctx.fillStyle='#0a1430'; ctx.strokeStyle='#0ff'; ctx.lineWidth=2;
  ctx.fillRect(-W*.09,-H*.16,W*.075,H*.05); ctx.strokeRect(-W*.09,-H*.16,W*.075,H*.05);
  // body
  ctx.fillStyle='#101d45';
  ctx.beginPath();
  ctx.moveTo(-W*.03,0); ctx.lineTo(W*.02,0); ctx.lineTo(-W*.005,-H*.13); ctx.lineTo(-W*.085,-H*.13);
  ctx.closePath(); ctx.fill(); ctx.stroke();
  // glow strip
  ctx.fillStyle='rgba(0,255,255,.7)';
  ctx.fillRect(-W*.075,-H*.09,W*.06,H*.008);
  ctx.restore();
}

function drawHUD(){
  if(state!=='play'&&state!=='pause') return;
  // crosshair
  const spread=6+recoil*14+(keys['KeyW']?3:0);
  ctx.strokeStyle='rgba(0,255,255,.9)'; ctx.lineWidth=2;
  ctx.beginPath();
  ctx.moveTo(CX-spread-8,CY); ctx.lineTo(CX-spread,CY);
  ctx.moveTo(CX+spread,CY); ctx.lineTo(CX+spread+8,CY);
  ctx.moveTo(CX,CY-spread-8); ctx.lineTo(CX,CY-spread);
  ctx.moveTo(CX,CY+spread); ctx.lineTo(CX,CY+spread+8);
  ctx.stroke();
  ctx.fillStyle='rgba(0,255,255,.9)'; ctx.fillRect(CX-1,CY-1,2,2);

  // health
  const hw=Math.min(260,W*.3), hx=24, hy=24;
  ctx.fillStyle='rgba(0,0,0,.5)'; ctx.fillRect(hx,hy,hw,18);
  const hpFrac=player.hp/100;
  ctx.fillStyle=hpFrac>.5?'#0f8':(hpFrac>.25?'#ff0':'#f33');
  ctx.fillRect(hx+2,hy+2,(hw-4)*Math.max(0,hpFrac),14);
  ctx.strokeStyle='#0ff'; ctx.lineWidth=1; ctx.strokeRect(hx,hy,hw,18);
  ctx.fillStyle='#fff'; ctx.font='bold 13px Courier New'; ctx.textAlign='left';
  ctx.fillText('HULL '+Math.ceil(player.hp)+'%',hx+6,hy-6);

  // score / wave
  ctx.textAlign='right';
  ctx.fillStyle='#0ff'; ctx.font='bold 22px Courier New';
  ctx.fillText('SCORE '+score,W-24,40);
  ctx.fillStyle='#f0f'; ctx.font='bold 15px Courier New';
  ctx.fillText('WAVE '+wave+'   HOSTILES: '+enemies.length,W-24,62);

  // ammo
  ctx.textAlign='right';
  if(reloadT>0){
    ctx.fillStyle='#ff0'; ctx.font='bold 18px Courier New';
    ctx.fillText('RELOADING...',W-24,H-40);
  } else {
    ctx.fillStyle=player.ammo===0?'#f33':'#fff'; ctx.font='bold 30px Courier New';
    ctx.fillText(player.ammo+' / '+player.maxAmmo,W-24,H-36);
  }

  // wave banner
  if(waveBannerT>0){
    const a=Math.min(1,waveBannerT/0.5);
    ctx.globalAlpha=a;
    ctx.textAlign='center';
    ctx.fillStyle='#0ff'; ctx.font='bold '+Math.min(64,W*.08)+'px Courier New';
    ctx.shadowColor='#0ff'; ctx.shadowBlur=25;
    ctx.fillText('WAVE '+wave,CX,H*.3);
    ctx.shadowBlur=0; ctx.globalAlpha=1;
  } else if(enemies.length===0){
    ctx.textAlign='center'; ctx.fillStyle='#f0f'; ctx.font='bold 24px Courier New';
    ctx.fillText('NEXT WAVE INBOUND...',CX,H*.3);
  }
}

/* ---------- LOOP ---------- */
let last=performance.now();
function loop(now){
  const dt=Math.min(.05,(now-last)/1000); last=now;
  if(state==='play') update(dt);
  render();
  requestAnimationFrame(loop);
}
requestAnimationFrame(loop);
