<script>
(function(){
"use strict";

/* ---------- setup / scaling ---------- */
const W=960, H=540, FLOOR=512;
const canvas=document.getElementById('game');
const ctx=canvas.getContext('2d');
let scale=1, offX=0, offY=0;
function resize(){
  const w=window.innerWidth, h=window.innerHeight;
  canvas.width=w; canvas.height=h;
  scale=Math.min(w/W, h/H);
  offX=(w-W*scale)/2; offY=(h-H*scale)/2;
}
window.addEventListener('resize',resize); resize();

/* ---------- audio ---------- */
let actx=null;
function initAudio(){ if(!actx){ try{ actx=new (window.AudioContext||window.webkitAudioContext)(); }catch(e){} } if(actx&&actx.state==='suspended') actx.resume(); }
function beep(freq,dur,type,vol,slide){
  if(!actx) return;
  const o=actx.createOscillator(), g=actx.createGain();
  o.type=type||'square'; o.frequency.value=freq;
  if(slide) o.frequency.exponentialRampToValueAtTime(Math.max(40,freq+slide), actx.currentTime+dur);
  g.gain.value=vol||0.12;
  g.gain.exponentialRampToValueAtTime(0.0001, actx.currentTime+dur);
  o.connect(g); g.connect(actx.destination);
  o.start(); o.stop(actx.currentTime+dur+0.02);
}

/* ---------- input ---------- */
const keys={};
window.addEventListener('keydown',e=>{
  if(['ArrowLeft','ArrowRight','ArrowUp','Space','KeyA','KeyD','KeyW'].includes(e.code)) e.preventDefault();
  keys[e.code]=true;
  initAudio();
  if(state==='start' && (e.code==='Enter'||e.code==='Space')) startGame();
  else if((state==='over'||state==='win') && (e.code==='KeyR'||e.code==='Enter')) startGame();
});
window.addEventListener('keyup',e=>{ keys[e.code]=false; });
canvas.addEventListener('mousedown',()=>{
  initAudio();
  if(state==='start') startGame();
  else if(state==='over'||state==='win') startGame();
});

/* ---------- world ---------- */
const platforms=[
  {x:0,   y:FLOOR, w:W},
  {x:170, y:392,   w:180},
  {x:610, y:392,   w:180},
  {x:395, y:288,   w:170},
  {x:120, y:196,   w:140},
  {x:700, y:196,   w:140}
];

const player={x:110,y:FLOOR-46,w:18,h:46,vx:0,vy:0,onGround:true,color:'#3df2ff'};
const ai={x:W-38,y:200,w:18,h:46,color:'#ff5d7a'};
const ball={x:W/2,y:H/2,vx:0,vy:0,r:9,speed:6};
let particles=[];

const WIN_SCORE=7;
let state='start';
let pScore=0, aScore=0, lives=3, serveTimer=0, serveDir=1, shake=0, flash=0;
let best=0;
try{ best=parseInt(localStorage.getItem('pongplat_best')||'0',10)||0; }catch(e){}

function startGame(){
  pScore=0; aScore=0; lives=3; particles=[];
  player.x=110; player.y=FLOOR-player.h; player.vx=0; player.vy=0;
  ai.y=200;
  state='playing';
  serve(1);
  beep(660,0.12,'square',0.1,300);
}

function serve(dir){
  ball.x=W/2; ball.y=H/2-40; ball.speed=6;
  const ang=(Math.random()*0.5-0.25)*Math.PI; // mostly horizontal
  ball.vx=Math.cos(ang)*ball.speed*dir;
  ball.vy=Math.sin(ang)*ball.speed;
  serveTimer=45;
}

function burst(x,y,color,n){
  for(let i=0;i<n;i++){
    const a=Math.random()*Math.PI*2, s=1+Math.random()*4;
    particles.push({x,y,vx:Math.cos(a)*s,vy:Math.sin(a)*s,life:20+Math.random()*15,color});
  }
}

/* ---------- physics helpers ---------- */
function aabb(ax,ay,aw,ah,bx,by,bw,bh){
  return ax<bx+bw && ax+aw>bx && ay<by+bh && ay+ah>by;
}

function stepPlayer(){
  const spd=6.2;
  let mv=0;
  if(keys['ArrowLeft']||keys['KeyA']) mv-=1;
  if(keys['ArrowRight']||keys['KeyD']) mv+=1;
  player.vx=mv*spd;
  if((keys['Space']||keys['ArrowUp']||keys['KeyW']) && player.onGround){
    player.vy=-12.4; player.onGround=false;
    beep(300,0.1,'triangle',0.1,250);
    burst(player.x+player.w/2, player.y+player.h, '#3df2ff', 6);
  }
  const prevBottom=player.y+player.h;
  player.vy+=0.55; if(player.vy>15) player.vy=15;
  player.x+=player.vx;
  if(player.x<4){player.x=4;} if(player.x>W-4-player.w){player.x=W-4-player.w;}
  player.y+=player.vy;
  player.onGround=false;
  for(const p of platforms){
    if(player.vy>=0 && prevBottom<=p.y+1 && player.y+player.h>=p.y &&
       player.x+player.w>p.x+2 && player.x<p.x+p.w-2){
      player.y=p.y-player.h; player.vy=0; player.onGround=true;
    }
  }
}

function stepAI(){
  const diff=5.4 + (pScore+aScore)*0.28;
  let target=ball.y+ball.r - ai.h/2;
  if(ball.vx<0) target+= (Math.random()-0.5)*30; // wobble when ball coming away
  const d=target-ai.y;
  const maxS = (ball.vx>0)? diff : diff*0.7;
  if(Math.abs(d)>4) ai.y += Math.max(-maxS, Math.min(maxS, d));
  ai.y=Math.max(12, Math.min(FLOOR-ai.h, ai.y));
}

function hitPaddle(pad,isPlayer){
  const cy=pad.y+pad.h/2;
  let off=(ball.y-cy)/(pad.h/2);
  if(off<-1)off=-1; if(off>1)off=1;
  ball.speed=Math.min(ball.speed+0.45, 15);
  const dir=isPlayer?1:-1;
  let vy=off*3.6;
  let vx=Math.sqrt(Math.max(ball.speed*ball.speed - vy*vy, (ball.speed*0.48)*(ball.speed*0.48)));
  ball.vx=vx*dir; ball.vy=vy;
  // push out of paddle
  if(isPlayer) ball.x=pad.x+pad.w+ball.r+1; else ball.x=pad.x-ball.r-1;
  burst(ball.x,ball.y,pad.color,10);
  shake=Math.min(shake+3,8);
  beep(isPlayer?440:220,0.07,'square',0.12, isPlayer?160:-120);
}

function stepBall(){
  if(serveTimer>0){ serveTimer--; return; }
  ball.x+=ball.vx; ball.y+=ball.vy;
  // walls
  if(ball.y-ball.r<4){ ball.y=4+ball.r; ball.vy=Math.abs(ball.vy); beep(200,0.05,'square',0.07); }
  if(ball.y+ball.r>FLOOR){ ball.y=FLOOR-ball.r; ball.vy=-Math.abs(ball.vy); beep(200,0.05,'square',0.07); }
  // platforms (not floor handled above)
  for(let i=1;i<platforms.length;i++){
    const p=platforms[i];
    if(aabb(ball.x-ball.r,ball.y-ball.r,ball.r*2,ball.r*2,p.x,p.y,p.w,p.h)){
      const pl=ball.x-(p.x+ball.r), pr=(p.x+p.w+ball.r)-ball.x;
      const pt=ball.y-(p.y+ball.r), pb=(p.y+p.h+ball.r)-ball.y;
      const m=Math.min(pl,pr,pt,pb);
      if(m===pt){ ball.y=p.y+ball.r; ball.vy=-Math.abs(ball.vy);} 
      else if(m===pb){ ball.y=p.y+p.h-ball.r; ball.vy=Math.abs(ball.vy);}
      else if(m===pl){ ball.x=p.x+ball.r; ball.vx=-Math.abs(ball.vx);}
      else { ball.x=p.x+p.w-ball.r; ball.vx=Math.abs(ball.vx);}
      beep(260,0.05,'triangle',0.08);
    }
  }
  // paddles
  if(aabb(ball.x-ball.r,ball.y-ball.r,ball.r*2,ball.r*2,player.x,player.y,player.w,player.h)) hitPaddle(player,true);
  else if(aabb(ball.x-ball.r,ball.y-ball.r,ball.r*2,ball.r*2,ai.x,ai.y,ai.w,ai.h)) hitPaddle(ai,false);
  // scoring
  if(ball.x<-30){
    aScore++; lives--; flash=1; shake=10;
    burst(30,ball.y,'#ff5d7a',24);
    beep(120,0.35,'sawtooth',0.16,-80);
    if(lives<=0){ endGame(false); } else serve(-1); // toward the player who missed
  } else if(ball.x>W+30){
    pScore++; flash=1; shake=8;
    burst(W-30,ball.y,'#3df2ff',24);
    beep(520,0.3,'square',0.16,300);
    if(pScore>=WIN_SCORE) endGame(true); else serve(1);
  }
}

function endGame(won){
  state=won?'win':'over';
  best=Math.max(best,pScore);
  try{ localStorage.setItem('pongplat_best',String(best)); }catch(e){}
  if(won) beep(880,0.5,'triangle',0.14,220); else beep(90,0.6,'sawtooth',0.16,-60);
}

/* ---------- update ---------- */
function update(){
  if(state!=='playing') return;
  stepPlayer(); stepAI(); stepBall();
  for(let i=particles.length-1;i>=0;i--){
    const p=particles[i];
    p.x+=p.vx; p.y+=p.vy; p.vy+=0.12; p.life--;
    if(p.life<=0) particles.splice(i,1);
  }
  if(shake>0) shake*=0.85;
  if(flash>0) flash-=0.03;
}

/* ---------- drawing ---------- */
function roundRect(x,y,w,h,r){
  ctx.beginPath();
  ctx.moveTo(x+r,y); ctx.arcTo(x+w,y,x+w,y+h,r); ctx.arcTo(x+w,y+h,x,y+h,r);
  ctx.arcTo(x,y+h,x,y,r); ctx.arcTo(x,y,x+w,y,r); ctx.closePath();
}

function draw(){
  ctx.setTransform(1,0,0,1,0,0);
  ctx.fillStyle='#05060f';
  ctx.fillRect(0,0,canvas.width,canvas.height);
  const sx=(Math.random()-0.5)*shake, sy=(Math.random()-0.5)*shake;
  ctx.setTransform(scale,0,0,scale,offX+sx*scale,offY+sy*scale);

  // bg grid
  ctx.strokeStyle='rgba(60,80,160,0.12)'; ctx.lineWidth=1;
  for(let x=0;x<=W;x+=48){ ctx.beginPath();ctx.moveTo(x,0);ctx.lineTo(x,FLOOR);ctx.stroke(); }
  for(let y=0;y<=FLOOR;y+=48){ ctx.beginPath();ctx.moveTo(0,y);ctx.lineTo(W,y);ctx.stroke(); }

  // net line (visual only)
  ctx.strokeStyle='rgba(120,140,255,0.25)'; ctx.setLineDash([8,10]); ctx.lineWidth=2;
  ctx.beginPath(); ctx.moveTo(W/2,10); ctx.lineTo(W/2,FLOOR); ctx.stroke();
  ctx.setLineDash([]);

  // platforms
  for(let i=1;i<platforms.length;i++){
    const p=platforms[i];
    ctx.shadowColor='#5a6cff'; ctx.shadowBlur=14;
    ctx.fillStyle='#20295e';
    roundRect(p.x,p.y,p.w,10,4); ctx.fill();
    ctx.shadowBlur=0;
    ctx.fillStyle='#7c8cff';
    ctx.fillRect(p.x+3,p.y+2,p.w-6,2);
  }
  // floor
  ctx.shadowColor='#5a6cff'; ctx.shadowBlur=14;
  ctx.fillStyle='#20295e'; ctx.fillRect(0,FLOOR,W,H-FLOOR);
  ctx.shadowBlur=0;
  ctx.fillStyle='#7c8cff'; ctx.fillRect(0,FLOOR,W,3);

  // serve countdown
  if(state==='playing'&&serveTimer>0){
    ctx.globalAlpha=Math.min(1,serveTimer/20);
    ctx.fillStyle='#fff'; ctx.font='bold 40px monospace'; ctx.textAlign='center';
    ctx.fillText('READY',W/2,H/2+60);
    ctx.globalAlpha=1;
  }

  // ball
  if(state==='playing'||state==='win'){
    ctx.shadowColor='#ffd93d'; ctx.shadowBlur=18;
    ctx.fillStyle='#ffd93d';
    ctx.beginPath(); ctx.arc(ball.x,ball.y,ball.r,0,Math.PI*2); ctx.fill();
    ctx.shadowBlur=0;
  }

  // player paddle
  if(state!=='start'){
    ctx.shadowColor=player.color; ctx.shadowBlur=16;
    ctx.fillStyle=player.color;
    roundRect(player.x,player.y,player.w,player.h,5); ctx.fill();
    ctx.shadowBlur=0;
    // ai paddle
    ctx.shadowColor=ai.color; ctx.shadowBlur=16;
    ctx.fillStyle=ai.color;
    roundRect(ai.x,ai.y,ai.w,ai.h,5); ctx.fill();
    ctx.shadowBlur=0;
  }

  // particles
  for(const p of particles){
    ctx.globalAlpha=Math.max(0,p.life/35);
    ctx.fillStyle=p.color;
    ctx.fillRect(p.x-2,p.y-2,4,4);
  }
  ctx.globalAlpha=1;

  // flash
  if(flash>0){ ctx.fillStyle='rgba(255,255,255,'+(flash*0.18)+')'; ctx.fillRect(0,0,W,FLOOR); }

  drawHUD();
  if(state==='start') drawStart();
  if(state==='over') drawEnd(false);
  if(state==='win') drawEnd(true);
}

function drawHUD(){
  if(state==='start') return;
  ctx.textAlign='center'; ctx.font='bold 42px monospace';
  ctx.fillStyle='#3df2ff'; ctx.fillText(pScore, W/2-60, 52);
  ctx.fillStyle='#ff5d7a'; ctx.fillText(aScore, W/2+60, 52);
  ctx.fillStyle='rgba(255,255,255,0.3)'; ctx.font='bold 24px monospace';
  ctx.fillText(':', W/2, 48);

  // lives
  for(let i=0;i<3;i++){
    const alive=i<lives;
    ctx.shadowColor=alive?player.color:'transparent'; ctx.shadowBlur=alive?10:0;
    ctx.fillStyle=alive?player.color:'rgba(255,255,255,0.15)';
    roundRect(24+i*30, 20, 12, 34, 4); ctx.fill();
    ctx.shadowBlur=0;
  }
  ctx.textAlign='right'; ctx.font='14px monospace'; ctx.fillStyle='rgba(255,255,255,0.4)';
  ctx.fillText('BEST '+best, W-16, 30);
}

function drawStart(){
  overlay();
  ctx.textAlign='center';
  ctx.shadowColor='#3df2ff'; ctx.shadowBlur=25;
  ctx.fillStyle='#3df2ff'; ctx.font='bold 58px monospace';
  ctx.fillText('PONG', W/2, 170);
  ctx.fillStyle='#ff5d7a'; ctx.shadowColor='#ff5d7a';
  ctx.fillText('PLATFORMER', W/2, 235);
  ctx.shadowBlur=0;
  ctx.fillStyle='rgba(255,255,255,0.85)'; ctx.font='18px monospace';
  ctx.fillText('← → / A D  move      SPACE / W / ↑  jump', W/2, 310);
  ctx.fillText('Return the ball with your paddle — use platforms to position!', W/2, 345);
  ctx.fillText('First to '+WIN_SCORE+' wins. You have 3 lives.', W/2, 375);
  pulseText('PRESS ENTER OR CLICK TO START', W/2, 440, 22);
}

function drawEnd(won){
  overlay();
  ctx.textAlign='center';
  ctx.font='bold 56px monospace';
  if(won){
    ctx.shadowColor='#3df2ff'; ctx.shadowBlur=25; ctx.fillStyle='#3df2ff';
    ctx.fillText('YOU WIN!', W/2, 210);
  }else{
    ctx.shadowColor='#ff5d7a'; ctx.shadowBlur=25; ctx.fillStyle='#ff5d7a';
    ctx.fillText('GAME OVER', W/2, 210);
  }
  ctx.shadowBlur=0;
  ctx.fillStyle='rgba(255,255,255,0.85)'; ctx.font='22px monospace';
  ctx.fillText(pScore+'  —  '+aScore, W/2, 270);
  ctx.fillStyle='rgba(255,255,255,0.5)'; ctx.font='16px monospace';
  ctx.fillText('BEST: '+best, W/2, 305);
  pulseText('PRESS R OR CLICK TO RESTART', W/2, 380, 22);
}

function overlay(){
  ctx.fillStyle='rgba(5,6,15,0.78)';
  ctx.fillRect(0,0,W,H);
}

let t=0;
function pulseText(txt,x,y,size){
  const a=0.5+0.5*Math.sin(t*0.08);
  ctx.globalAlpha=0.4+a*0.6;
  ctx.fillStyle='#fff'; ctx.font='bold '+size+'px monospace';
  ctx.fillText(txt,x,y);
  ctx.globalAlpha=1;
}

/* ---------- loop ---------- */
function loop(){
  t++;
  update();
  draw();
  requestAnimationFrame(loop);
}
loop();

})();
</script>
