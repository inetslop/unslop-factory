function drawTurtle(t){
  const y=laneY(t.row)+CELL/2;
  ctx.save();ctx.translate(t.x,y);
  ctx.fillStyle='#3f7d3a';
  // head + legs
  const d=t.sp>=0?1:-1;
  ctx.beginPath();ctx.arc(d*26,0,7,0,7);ctx.fill();
  [[-14,-20],[14,-20],[-14,20],[14,20]].forEach(p=>{ctx.beginPath();ctx.arc(p[0],p[1],5,0,7);ctx.fill();});
  // shell
  ctx.fillStyle='#2c5e28';ctx.beginPath();ctx.ellipse(0,0,22,19,0,0,7);ctx.fill();
  ctx.strokeStyle='#4f8f45';ctx.lineWidth=2;
  ctx.beginPath();ctx.ellipse(0,0,22,19,0,0,7);ctx.stroke();
  ctx.beginPath();ctx.moveTo(-22,0);ctx.lineTo(22,0);ctx.moveTo(0,-19);ctx.lineTo(0,19);ctx.stroke();
  ctx.restore();
}

function drawCar(c){
  const y=laneY(c.row)+CELL/2;
  ctx.save();ctx.translate(c.x,y);
  if(c.kind==='truck'){
    ctx.fillStyle='#c8542f';rr(-46,-19,70,38,6);ctx.fill();
    ctx.fillStyle='#e8d9b0';rr(24,-19,22,38,5);ctx.fill();
    ctx.fillStyle='#8a3a1e';ctx.fillRect(-46,-4,70,8);
  }else{
    const cols=['#e0b422','#c0392b','#4a7fb5','#8e6bbf'];
    ctx.fillStyle=cols[Math.floor(c.x/40+cols.length)%cols.length];
    rr(-34,-18,68,36,9);ctx.fill();
    ctx.fillStyle='#bfe6ff';rr(-16,-12,34,24,5);ctx.fill();
  }
  ctx.fillStyle='#1a1a1a';
  [[-28,16],[28,16],[-28,-16],[28,-16]].forEach(p=>{ctx.beginPath();ctx.arc(p[0],p[1],5,0,7);ctx.fill();});
  ctx.restore();
}

function drawBug(b){
  ctx.save();ctx.translate(b.x,b.y);
  ctx.fillStyle='#3a2418';
  for(let i=-1;i<=1;i++){
    ctx.beginPath();ctx.ellipse(-10,i*7,6,2.5,-0.5,0,7);ctx.fill();
    ctx.beginPath();ctx.ellipse(10,i*7,6,2.5,0.5,0,7);ctx.fill();
  }
  ctx.fillStyle='#e8433a';ctx.beginPath();ctx.ellipse(0,0,13,10,0,0,7);ctx.fill();
  ctx.fillStyle='#ff7d6e';ctx.beginPath();ctx.ellipse(-2,-3,7,5,0,0,7);ctx.fill();
  ctx.fillStyle='#111';
  ctx.beginPath();ctx.arc(9,-4,2.4,0,7);ctx.fill();
  ctx.beginPath();ctx.arc(9,4,2.4,0,7);ctx.fill();
  ctx.restore();
}

function draw(){
  ctx.clearRect(0,0,W,H);
  // background per row
  for(let r=0;r<ROWS;r++){
    const y=laneY(r);
    if(r===0){ctx.fillStyle='#123c6b';}
    else if(r>=1&&r<=4){ctx.fillStyle=(r%2)?'#1a5fa8':'#1e6ab8';}
    else if(r===5||r===10){ctx.fillStyle='#2f7d33';}
    else{ctx.fillStyle=(r%2)?'#3a3f47':'#343941';}
    ctx.fillRect(0,y,W,CELL);
  }
  // road dashes
  ctx.strokeStyle='rgba(255,255,255,.35)';ctx.lineWidth=3;ctx.setLineDash([18,16]);
  for(let r=6;r<10;r++){
    ctx.beginPath();ctx.moveTo(0,laneY(r)+CELL);ctx.lineTo(W,laneY(r)+CELL);ctx.stroke();
  }
  ctx.setLineDash([]);

  // pads
  for(const pc of pads){
    const x=pc*CELL+CELL/2,y=laneY(0)+CELL/2;
    ctx.fillStyle='#3fae4c';ctx.beginPath();ctx.ellipse(x,y,24,18,0,0,7);ctx.fill();
    ctx.fillStyle='#5fd96f';ctx.beginPath();ctx.ellipse(x-4,y-4,14,9,0,0,7);ctx.fill();
  }

  for(const t of turtles)drawTurtle(t);
  for(const c of cars)drawCar(c);
  for(const b of bugs)if(!b.dead)drawBug(b);

  // bullets
  ctx.fillStyle='#ffe45e';
  for(const bl of bullets){
    ctx.beginPath();ctx.moveTo(bl.x,bl.y-12);ctx.lineTo(bl.x-5,bl.y+6);ctx.lineTo(bl.x+5,bl.y+6);ctx.closePath();ctx.fill();
    ctx.fillStyle='rgba(255,200,60,.4)';ctx.fillRect(bl.x-1.5,bl.y+6,3,8);ctx.fillStyle='#ffe45e';
  }

  // frog
  if(frog.alive&&(state==='play')){
    const bob=Math.sin(time*10)*1.5;
    drawFrog(frog.x,frog.y+bob,1);
  }else if(state==='dead'){
    ctx.fillStyle='#fff';ctx.font='bold 42px monospace';ctx.textAlign='center';
    ctx.fillText('!',frog.x,frog.y+14);
  }

  // particles
  for(const p of particles){
    ctx.globalAlpha=Math.max(0,p.life/p.max);
    ctx.fillStyle=p.color;
    ctx.fillRect(p.x-p.size/2,p.y-p.size/2,p.size,p.size);
  }
  ctx.globalAlpha=1;

  /* HUD */
  ctx.fillStyle='#0d1524';ctx.fillRect(0,0,W,HUD);
  ctx.strokeStyle='rgba(80,200,255,.3)';ctx.lineWidth=2;
  ctx.beginPath();ctx.moveTo(0,HUD);ctx.lineTo(W,HUD);ctx.stroke();
  ctx.textAlign='left';ctx.fillStyle='#fff';ctx.font='bold 22px monospace';
  ctx.fillText('SCORE '+score,14,38);
  ctx.textAlign='center';ctx.fillStyle='#7ec8ff';ctx.font='16px monospace';
  ctx.fillText('BEST '+best+'   LV '+level,W/2+30,38);
  ctx.textAlign='right';
  for(let i=0;i<lives;i++)drawFrog(W-24-i*30,30,0.55);

  /* overlays */
  if(state==='start'){
    overlay();
    ctx.fillStyle='#3ecf62';ctx.font='bold 52px monospace';ctx.textAlign='center';
    ctx.fillText('FROG SHOOTER',W/2,240);
    ctx.fillStyle='#fff';ctx.font='18px monospace';
    ctx.fillText('Cross the road and river... but this time,',W/2,310);
    ctx.fillText('you can shoot the mutant bugs!',W/2,336);
    ctx.fillStyle='#ffe45e';ctx.font='17px monospace';
    ctx.fillText('ARROWS / WASD  -  move',W/2,410);
    ctx.fillText('SPACE  -  shoot bug (+50)',W/2,438);
    ctx.fillText('REACH A LILY PAD  -  next level (+200)',W/2,466);
    ctx.fillStyle='#9fd';ctx.font='16px monospace';
    ctx.fillText('Cars and bugs kill you. Water kills too. Ride turtles!',W/2,510);
    blink(W/2,580,'PRESS ENTER TO START');
  }
  if(state==='gameover'){
    overlay();
    ctx.fillStyle='#ff5a4d';ctx.font='bold 52px monospace';ctx.textAlign='center';
    ctx.fillText('GAME OVER',W/2,300);
    ctx.fillStyle='#fff';ctx.font='24px monospace';
    ctx.fillText('SCORE '+score,W/2,360);
    ctx.fillStyle='#ffe45e';ctx.font='18px monospace';
    ctx.fillText(score>=best&&score>0?'NEW BEST!':'BEST '+best,W/2,395);
    blink(W/2,470,'PRESS ENTER TO RESTART');
  }
}

function overlay(){
  ctx.fillStyle='rgba(5,10,20,.82)';ctx.fillRect(0,HUD,W,H-HUD);
}
function blink(x,y,txt){
  if(Math.floor(time*2)%2)return;
  ctx.fillStyle='#fff';ctx.font='bold 20px monospace';ctx.textAlign='center';
  ctx.fillText(txt,x,y);
}

/* ---------- loop ---------- */
let last=performance.now();
function loop(now){
  let dt=(now-last)/1000;last=now;
  if(dt>0.05)dt=0.05;
  update(dt);
  draw();
  requestAnimationFrame(loop);
}
requestAnimationFrame(loop);
})();
