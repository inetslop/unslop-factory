function stepBall(b,dt){

        if(d<0.0001){nx=0;ny=-1;b.y=w.y-R-0.5;}else{nx=dx/d;ny=dy/d;b.x=cx+nx*(R+0.2);b.y=cy+ny*(R+0.2);}
        var dot=b.vx*nx+b.vy*ny;
        if(dot<0){b.vx-=2*dot*nx;b.vy-=2*dot*ny;}
        b.n++;
      }
    }
  }
}

/* ---------- actions ---------- */
function fire(){
  if(state!=='playing'||ball||ballsLeft<=0)return;
  ball={x:CX+Math.cos(aim)*34,y:CY+Math.sin(aim)*34,vx:Math.cos(aim)*SPEED,vy:Math.sin(aim)*SPEED,n:0,trail:[]};
  ballsLeft--; recoil=1; sFire();
}

function nextLevel(){
  if(lvl+1<LEVELS.length){setLevel(lvl+1);state='playing';hideOverlay();}
  else{state='win';sWin();showWin();}
}

/* ---------- update ---------- */
function update(dt){
  timeNow+=dt; recoil=Math.max(0,recoil-dt*4);
  for(var i=particles.length-1;i>=0;i--){
    var p=particles[i]; p.life-=dt;
    if(p.life<=0){particles.splice(i,1);continue;}
    p.x+=p.vx*dt;p.y+=p.vy*dt;p.vy+=300*dt;p.vx*=0.99;
  }
  if(state==='playing'&&ball){
    stepBall(ball,dt);
    if(ball.n>MAXB){burst(ball.x,ball.y,'#5ad0ff',12);ball=null;}
    if(targets.length===0&&state==='playing'){
      state='complete'; completeAt=timeNow+0.7; sWin();
      for(var c=0;c<90;c++){
        var cols=['#5ad0ff','#a07bff','#ffd76b','#7dffa8'];
        particles.push({x:Math.random()*W,y:-10-Math.random()*120,vx:(Math.random()-0.5)*160,vy:80+Math.random()*160,
          life:1.5+Math.random(),max:2,col:cols[c%4],sz:3+Math.random()*4});
      }
    }
  }
  if(state==='playing'&&!ball&&ballsLeft<=0&&targets.length>0){showGameOver();}
  if(state==='complete'&&timeNow>=completeAt){
    try{if(lvl+1>best-0){best=Math.max(best,lvl+2);localStorage.setItem('ricochet_best',String(Math.min(best,LEVELS.length)));}}catch(e){}
    showComplete();
  }
}

/* ---------- render ---------- */
function rr(x,y,w,h,r){ctx.beginPath();ctx.moveTo(x+r,y);ctx.arcTo(x+w,y,x+w,y+h,r);ctx.arcTo(x+w,y+h,x,y+h,r);ctx.arcTo(x,y+h,x,y,r);ctx.arcTo(x,y,x+w,y,r);ctx.closePath();}

function render(){
  ctx.setTransform(1,0,0,1,0,0);
  ctx.fillStyle='#0b0e17'; ctx.fillRect(0,0,cv.width,cv.height);
  ctx.setTransform(scale,0,0,scale,ox,oy);

  // field
  ctx.fillStyle='#141a2e'; ctx.fillRect(0,0,W,H);
  ctx.strokeStyle='#2b3560'; ctx.lineWidth=4; ctx.strokeRect(2,2,W-4,H-4);
  ctx.strokeStyle='rgba(90,140,255,0.05)'; ctx.lineWidth=1;
  for(var g=90;g<W;g+=90){ctx.beginPath();ctx.moveTo(g,0);ctx.lineTo(g,H);ctx.stroke();}
  for(g=90;g<H;g+=90){ctx.beginPath();ctx.moveTo(0,g);ctx.lineTo(W,g);ctx.stroke();}

  // walls
  for(var i=0;i<walls.length;i++){
    var w=walls[i];
    ctx.fillStyle='#3a4675'; rr(w.x,w.y,w.w,w.h,6); ctx.fill();
    ctx.fillStyle='#4d5b95'; rr(w.x+3,w.y+3,Math.max(2,w.w-6),Math.max(2,w.h-6),4); ctx.fill();
  }

  // targets
  for(i=0;i<targets.length;i++){
    var t=targets[i],pu=1+Math.sin(timeNow*4+i)*0.08;
    ctx.beginPath();ctx.arc(t.x,t.y,TR*pu+5,0,7);ctx.fillStyle='rgba(255,215,107,0.15)';ctx.fill();
    ctx.beginPath();ctx.arc(t.x,t.y,TR*pu,0,7);ctx.fillStyle='#ffd76b';ctx.fill();
    ctx.beginPath();ctx.arc(t.x,t.y,TR*pu*0.55,0,7);ctx.fillStyle='#ff9d3c';ctx.fill();
    ctx.beginPath();ctx.arc(t.x-3,t.y-4,TR*0.28,0,7);ctx.fillStyle='rgba(255,255,255,0.8)';ctx.fill();
  }

  // preview
  if(state==='playing'&&!ball&&ballsLeft>0){
    var sim=simulate();
    if(sim){
      for(i=0;i<sim.pts.length;i++){
        var a=1-i/sim.pts.length;
        ctx.beginPath();ctx.arc(sim.pts[i].x,sim.pts[i].y,3,0,7);
        ctx.fillStyle='rgba(255,255,255,'+(0.15+a*0.55)+')';ctx.fill();
      }
      if(sim.hit){
        ctx.beginPath();ctx.arc(sim.hit.x,sim.hit.y,TR+6+Math.sin(timeNow*8)*3,0,7);
        ctx.strokeStyle='#ffd76b';ctx.lineWidth=2;ctx.stroke();
      }
    }
  }

  // cannon
  var ang=aim, rec=recoil*10;
  ctx.save(); ctx.translate(CX,CY);
  ctx.rotate(ang);
  ctx.fillStyle='#5ad0ff'; rr(-6-rec,-9,44+rec,18,7); ctx.fill();
  ctx.fillStyle='#a07bff'; rr(-6-rec,-9,12+rec,18,6); ctx.fill();
  ctx.restore();
  ctx.beginPath();ctx.arc(CX,CY,20,0,7);ctx.fillStyle='#1e2745';ctx.fill();
  ctx.lineWidth=3;ctx.strokeStyle='#5ad0ff';ctx.stroke();
  ctx.beginPath();ctx.arc(CX,CY,8,0,7);ctx.fillStyle='#a07bff';ctx.fill();

  // ball + trail
  if(ball){
    for(i=0;i<ball.trail.length;i++){
      var tr=ball.trail[i],ta=i/ball.trail.length;
      ctx.beginPath();ctx.arc(tr.x,tr.y,R*ta,0,7);ctx.fillStyle='rgba(90,208,255,'+(ta*0.3)+')';ctx.fill();
    }
    ctx.beginPath();ctx.arc(ball.x,ball.y,R+4,0,7);ctx.fillStyle='rgba(90,208,255,0.25)';ctx.fill();
    ctx.beginPath();ctx.arc(ball.x,ball.y,R,0,7);ctx.fillStyle='#eafcff';ctx.fill();
    ctx.beginPath();ctx.arc(ball.x-2,ball.y-2,3,0,7);ctx.fillStyle='#fff';ctx.fill();
  }

  // particles
  for(i=0;i<particles.length;i++){
    var p=particles[i];
    ctx.globalAlpha=Math.max(0,p.life/p.max);
    ctx.fillStyle=p.col; ctx.fillRect(p.x-p.sz/2,p.y-p.sz/2,p.sz,p.sz);
  }
  ctx.globalAlpha=1;

  // HUD
  if(state!=='menu'){
    ctx.textAlign='left';ctx.font='700 20px Segoe UI, Arial';
    ctx.fillStyle='#e8ecff'; ctx.fillText('LEVEL '+(lvl+1)+' / '+LEVELS.length,20,34);
    ctx.textAlign='right';
    ctx.fillStyle='#9aa5c8'; ctx.font='600 14px Segoe UI, Arial';
    ctx.fillText('TARGETS LEFT',W-20,24);
    for(i=0;i<ballsLeft;i++){
      ctx.beginPath();ctx.arc(W-20-i*24,42,8,0,7);ctx.fillStyle='#5ad0ff';ctx.fill();
    }
    if(ballsLeft===0){ctx.fillStyle='#6b7499';ctx.font='600 13px Segoe UI,Arial';ctx.fillText('— no balls —',W-20,48);}
    ctx.textAlign='left';ctx.fillStyle='#5a648c';ctx.font='600 12px Segoe UI,Arial';
    ctx.fillText('click / space to fire · R restart · M menu',20,H-14);
  }
}

/* ---------- loop ---------- */
var lastT=performance.now();
function frame(t){
  var dt=Math.min(0.033,(t-lastT)/1000); lastT=t;
  update(dt); render();
  requestAnimationFrame(frame);
}
showMenu();
requestAnimationFrame(frame);
})();
