    let dead=b.life<=0||b.x<0||b.x>W||b.y<0||b.y>H;
    if(!dead)for(const a of asteroids)if(Math.hypot(b.x-a.x,b.y-a.y)<a.r){dead=true;break;}
    if(!dead&&p.inv<=0&&Math.hypot(b.x-p.x,b.y-p.y)<15){dead=true;damagePlayer();}
    if(dead)bullets.splice(i,1);
  }

  // beacons
  for(const b of beacons){
    b.t+=dt*3;
    if(!b.got&&Math.hypot(p.x-b.x,p.y-b.y)<28){
      b.got=true;collected++;sCollect();shotFlash=1;
      for(let i=0;i<24;i++){const a=rnd(0,6.28),v=rnd(30,180);
        particles.push({x:b.x,y:b.y,vx:Math.cos(a)*v,vy:Math.sin(a)*v,life:.8,max:.8,c:'120,255,200'});}
      if(collected>=beacons.length){endGame(true);return;}
    }
  }

  // particles
  for(let i=particles.length-1;i>=0;i--){
    const q=particles[i];q.x+=q.vx*dt;q.y+=q.vy*dt;q.life-=dt;
    if(q.life<=0)particles.splice(i,1);
  }

  // camera
  camX=Math.max(0,Math.min(W-VW,p.x-VW/2));camY=Math.max(0,Math.min(H-VH,p.y-VH/2));
  if(W>=W)camX=(W<VW)?(W-VW)/2:camX;
  if(VW>W)camX=(W-VW)/2;if(VH>H)camY=(H-VH)/2;
}

// ---------- render ----------
function drawCone(e){
  const sf=0.5+1.1*(Math.min(Math.hypot(player.vx,player.vy),300)/300);
  const range=300*sf,col=spotted?'255,80,70':(exposure>60?'255,140,80':'255,205,110');
  const g=ctx.createRadialGradient(e.x,e.y,10,e.x,e.y,range);
  g.addColorStop(0,'rgba('+col+',.28)');g.addColorStop(1,'rgba('+col+',0)');
  ctx.fillStyle=g;ctx.beginPath();ctx.moveTo(e.x,e.y);
  ctx.arc(e.x,e.y,range,e.a-0.6,e.a+0.6);ctx.closePath();ctx.fill();
}
function render(){
  ctx.fillStyle='#04060d';ctx.fillRect(0,0,VW,VH);
  const sx=(Math.random()-.5)*shake,sy=(Math.random()-.5)*shake;
  ctx.save();ctx.translate(-camX+sx,-camY+sy);

  // stars + grid
  for(const s of stars){
    if(s.x<camX-10||s.x>camX+VW+10||s.y<camY-10||s.y>camY+VH+10)continue;
    ctx.fillStyle='rgba(200,225,255,'+(s.b*.6)+')';ctx.fillRect(s.x,s.y,s.s,s.s);
  }
  ctx.strokeStyle='rgba(70,110,180,.07)';ctx.lineWidth=1;
  for(let x=0;x<=W;x+=200){ctx.beginPath();ctx.moveTo(x,0);ctx.lineTo(x,H);ctx.stroke();}
  for(let y=0;y<=H;y+=200){ctx.beginPath();ctx.moveTo(0,y);ctx.lineTo(W,y);ctx.stroke();}
  ctx.strokeStyle='rgba(120,180,255,.25)';ctx.lineWidth=3;ctx.strokeRect(0,0,W,H);

  // vision cones (under entities)
  for(const e of enemies)drawCone(e);

  // beacons
  for(const b of beacons){
    if(b.got)continue;
    const pu=1+Math.sin(b.t)*.25;
    ctx.save();ctx.translate(b.x,b.y);ctx.rotate(b.t*.4);
    ctx.shadowColor='#5fffd0';ctx.shadowBlur=18;
    ctx.strokeStyle='#5fffd0';ctx.lineWidth=2.5;
    ctx.strokeRect(-9*pu,-9*pu,18*pu,18*pu);
    ctx.fillStyle='rgba(95,255,208,.35)';ctx.fillRect(-4,-4,8,8);
    ctx.restore();
  }

  // asteroids
  for(const a of asteroids){
    if(a.x+ a.r<camX-20||a.x-a.r>camX+VW+20||a.y+a.r<camY-20||a.y-a.r>camY+VH+20)continue;
    ctx.save();ctx.translate(a.x,a.y);ctx.rotate(a.rot);
    ctx.beginPath();
    for(let i=0;i<a.pts.length;i++){
      const an=i/a.pts.length*6.283,r=a.r*a.pts[i];
      i?ctx.lineTo(Math.cos(an)*r,Math.sin(an)*r):ctx.moveTo(Math.cos(an)*r,Math.sin(an)*r);
    }
    ctx.closePath();
    ctx.fillStyle='#2a3140';ctx.fill();ctx.strokeStyle='#4c5872';ctx.lineWidth=2;ctx.stroke();
    ctx.fillStyle='rgba(10,14,22,.5)';
    ctx.beginPath();ctx.arc(a.r*.3,-a.r*.15,a.r*.16,0,6.28);ctx.fill();
    ctx.beginPath();ctx.arc(-a.r*.25,a.r*.3,a.r*.12,0,6.28);ctx.fill();
    ctx.restore();
  }

  // bullets
  for(const b of bullets){
    ctx.strokeStyle='rgba(255,90,70,.9)';ctx.lineWidth=3;ctx.shadowColor='#ff5a46';ctx.shadowBlur=10;
    ctx.beginPath();ctx.moveTo(b.x,b.y);ctx.lineTo(b.x-b.vx*.03,b.y-b.vy*.03);ctx.stroke();ctx.shadowBlur=0;
  }

  // enemies
  for(const e of enemies){
    ctx.save();ctx.translate(e.x,e.y);ctx.rotate(e.a);
    ctx.shadowColor=spotted?'#ff5a46':'#ffb040';ctx.shadowBlur=12;
    ctx.fillStyle=spotted?'#e0483a':'#c98a3a';
    ctx.beginPath();ctx.moveTo(18,0);ctx.lineTo(-12,-11);ctx.lineTo(-6,0);ctx.lineTo(-12,11);ctx.closePath();ctx.fill();
    ctx.fillStyle='#0a0f18';ctx.beginPath();ctx.arc(4,0,4,0,6.28);ctx.fill();
    ctx.restore();ctx.shadowBlur=0;
  }

  // pulse ring
  if(pulseT>0){
    const pr=(3-pulseT)/3;
    ctx.strokeStyle='rgba(90,220,255,'+(0.7*(1-pr))+')';ctx.lineWidth=3;
    ctx.beginPath();ctx.arc(player.x,player.y,pr*420,0,6.28);ctx.stroke();
  }

  // player
  const p=player;
  if(!(p.inv>0&&Math.floor(time*14)%2===0)){
    ctx.save();ctx.translate(p.x,p.y);ctx.rotate(p.face);
    ctx.shadowColor='#7fe7ff';ctx.shadowBlur=16;
    ctx.fillStyle='#dff6ff';
    ctx.beginPath();ctx.moveTo(15,0);ctx.lineTo(-10,-9);ctx.lineTo(-5,0);ctx.lineTo(-10,9);ctx.closePath();ctx.fill();
    ctx.fillStyle='#3aa8d8';ctx.fillRect(-7,-2.5,6,5);
    if(keys['shift']&&Math.hypot(p.vx,p.vy)>40){
      ctx.fillStyle='rgba(150,220,255,.7)';
      ctx.beginPath();ctx.moveTo(-10,0);ctx.lineTo(-22-rnd(0,8),0);ctx.lineTo(-10,4);ctx.closePath();ctx.fill();
    }
    ctx.restore();ctx.shadowBlur=0;
  }

  // particles
  for(const q of particles){
    ctx.fillStyle='rgba('+q.c+','+(q.life/q.max)+')';
    ctx.fillRect(q.x-2,q.y-2,4,4);
  }
  ctx.restore();

  // ---- HUD ----
  // exposure bar
  const bw=Math.min(300,VW*.4),bx=(VW-bw)/2;
  ctx.fillStyle='rgba(10,18,32,.7)';ctx.fillRect(bx-2,14,bw+4,18);
  const ecol=exposure<50?'#6fe08a':exposure<80?'#ffd27f':'#ff5a46';
  ctx.fillStyle=ecol;ctx.fillRect(bx,16,bw*exposure/100,14);
  ctx.strokeStyle='rgba(140,200,255,.4)';ctx.strokeRect(bx-2,14,bw+4,18);
  ctx.fillStyle=spotted?'#ff7f7f':'#cfe8ff';ctx.font='11px Courier New';ctx.textAlign='center';
  ctx.fillText(spotted?'⚠ SPOTTED — RUN ⚠':'EXPOSURE',VW/2,44);

  // shields + beacons + time
  ctx.textAlign='left';ctx.font='16px Courier New';
  let shTxt='';for(let i=0;i<3;i++)shTxt+=i<shields?'▲ ':'▵ ';
  ctx.fillStyle='#7fe7ff';ctx.fillText(shTxt,16,30);
  ctx.fillStyle='#5fffd0';ctx.fillText('BEACONS '+collected+'/'+beacons.length,16,52);
  ctx.fillStyle='#8aa4c8';ctx.fillText('TIME '+time.toFixed(1)+'s',16,74);
  ctx.fillStyle=pulseCd<=0?'#7fe7ff':'#4a5f7d';
  ctx.fillText(pulseCd<=0?'PULSE READY [SPACE]':'PULSE '+pulseCd.toFixed(1)+'s',16,96);

  // minimap
  const mw=170,mh=mw*H/W,mx=VW-mw-14,my=14;
  ctx.fillStyle='rgba(8,14,26,.8)';ctx.fillRect(mx,my,mw,mh);
  ctx.strokeStyle='rgba(120,180,255,.4)';ctx.strokeRect(mx,my,mw,mh);
  const k=mw/W;
  ctx.fillStyle='#3a4560';for(const a of asteroids)ctx.fillRect(mx+a.x*k-1,my+a.y*k-1,3,3);
  ctx.fillStyle='#5fffd0';for(const b of beacons)if(!b.got)ctx.fillRect(mx+b.x*k-1.5,my+b.y*k-1.5,4,4);
  for(const e of enemies){
    const d=Math.hypot(e.x-p.x,e.y-p.y);
    if(d<520||pulseT>0||spotted){ctx.fillStyle=spotted?'#ff5a46':'#ffb040';ctx.fillRect(mx+e.x*k-2,my+e.y*k-2,5,5);}
  }
  ctx.fillStyle='#fff';ctx.fillRect(mx+p.x*k-2,my+p.y*k-2,5,5);

  // flashes
  if(flash>0){ctx.fillStyle='rgba(255,60,40,'+(flash*.35)+')';ctx.fillRect(0,0,VW,VH);}
  if(shotFlash>0){ctx.fillStyle='rgba(95,255,208,'+(shotFlash*.15)+')';ctx.fillRect(0,0,VW,VH);}
  if(spotted){
    ctx.strokeStyle='rgba(255,80,60,'+(.3+.25*Math.sin(time*8))+')';ctx.lineWidth=6;
    ctx.strokeRect(3,3,VW-6,VH-6);
  }
}

// ---------- loop ----------
let last=performance.now();
function loop(t){
  const dt=Math.min(.033,(t-last)/1000);last=t;
  if(state==='play')update(dt);
  if(state!=='start')render();
  else{ // idle backdrop on start screen
    ctx.fillStyle='#04060d';ctx.fillRect(0,0,VW,VH);
  }
  requestAnimationFrame(loop);
}
requestAnimationFrame(loop);
})();
