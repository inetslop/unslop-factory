      if(v.kind==='turtle'&&+r===ROCK.row){
        const ov=v.x<ROCK.b&&v.x+v.w>ROCK.a;
        if(ov&&!v.passing){if(v.pauseT==null)v.pauseT=1.0;v.pauseT-=dt;
          if(v.pauseT<=0){v.passing=true;v.pauseT=null;}moving=false;}
        else if(v.passing){sp*=3;if(!ov)v.passing=false;}
      }
      v.dx=v.dir*sp*dt*(moving?1:0);v.x+=v.dx;
      if(v.x<-5)v.x=COLS+2;if(v.x>COLS+5)v.x=-2;
    }
  }

  // carried by log / turtle
  const L=S.lanes[S.py];
  if(L)for(const v of L){if(overlap(S.px,v)){S.px+=v.dx;break;}}
  S.px=Math.min(COLS-0.5,Math.max(0.5,S.px));

  // eat pellets
  for(let i=S.pellets.length-1;i>=0;i--){const p=S.pellets[i];
    if(p.y===S.py&&Math.abs(p.x-S.px)<0.62){S.pellets.splice(i,1);S.score+=10;sEat();}}
  if(S.pellets.length===0){S.score+=500;S.powerT=6;S.msg='ALL CLEAR! +500 — POWER MODE!';S.msgT=2.2;
    sPower();S.pellets=makePellets();}

  // reach a golden pad
  if(S.py===0)for(const p of PADS){
    if(S.px>=p&&S.px<=p+2){const b=500+S.homes*250;S.score+=b;S.homes++;sWin();
      S.msg='HOME! +'+b;S.msgT=1.8;S.px=COLS/2;S.py=STARTROW;S.cd=0.4;break;}}
  if(S.score>HI)HI=S.score;

  // hazards (power mode = invincible, cars turn blue)
  if(S.powerT<=0){
    for(const r in S.lanes){if(+r!==S.py)continue;
      for(const v of S.lanes[r])if(v.kind==='car'&&overlap(S.px,v)){die();return;}}
    if(S.py>=4&&S.py<=7&&!riding()&&!(S.py===ROCK.row&&S.px>=ROCK.a&&S.px<ROCK.b))die();
  }
}

/* ---------- drawing ---------- */
function drawPac(x,y,r,face,mouth,color,alpha){
  ctx.save();ctx.globalAlpha=alpha==null?1:alpha;
  ctx.translate(x,y);if(face<0)ctx.scale(-1,1);
  ctx.fillStyle=color;ctx.beginPath();ctx.moveTo(0,0);
  ctx.arc(0,0,r,mouth,Math.PI*2-mouth);ctx.closePath();ctx.fill();
  ctx.fillStyle='#111';ctx.beginPath();ctx.arc(r*0.25,-r*0.45,r*0.14,0,7);ctx.fill();
  ctx.restore();
}

function drawBoard(){
  const c=cell;
  for(let r=0;r<ROWS;r++){
    let col;
    if(r===0)col='#12351a';
    else if(r>=4&&r<=7)col='#0b2a4a';
    else if(r===MEDIAN||r===STARTROW)col='#3d2f16';
    else col='#1c1c22';
    ctx.fillStyle=col;ctx.fillRect(ox,oy+r*c,COLS*c,c);
  }
  // road dashes
  ctx.strokeStyle='rgba(255,255,255,.15)';ctx.lineWidth=Math.max(1,c*0.05);
  ctx.setLineDash([c*0.4,c*0.5]);
  for(let r=9;r<=13;r++){ctx.beginPath();ctx.moveTo(ox,oy+r*c);ctx.lineTo(ox+COLS*c,oy+r*c);ctx.stroke();}
  ctx.setLineDash([]);

  // golden pads
  for(const p of PADS){const x=ox+p*c;
    ctx.fillStyle='rgba(255,215,0,.18)';ctx.fillRect(x+3,oy+3,2*c-6,c-6);
    ctx.strokeStyle='#ffd700';ctx.lineWidth=2;ctx.strokeRect(x+3,oy+3,2*c-6,c-6);}

  // boulders
  for(let i=0;i<3;i++){const cx=ox+(ROCK.a+i+0.5)*c,cy=oy+(ROCK.row+0.5)*c;
    ctx.fillStyle='#8d9099';ctx.beginPath();ctx.arc(cx,cy,c*0.38,0,7);ctx.fill();
    ctx.fillStyle='#b9bcc6';ctx.beginPath();ctx.arc(cx-c*0.1,cy-c*0.12,c*0.16,0,7);ctx.fill();}

  // pellets
  ctx.fillStyle=S.powerT>0?'#8ffcff':'#ffe9b8';
  for(const p of S.pellets){ctx.beginPath();ctx.arc(ox+p.x*c,oy+(p.y+0.5)*c,c*0.07,0,7);ctx.fill();}

  // vehicles
  for(const r in S.lanes){const y=oy+r*c;
    for(const v of S.lanes[r]){const x=ox+v.x*c,w=v.w*c;
      if(v.kind==='car'){
        const blue=S.powerT>0;
        ctx.fillStyle=blue?'#3b62ff':v.color;rr(x+1,y+c*0.15,w-2,c*0.7,4);ctx.fill();
        ctx.fillStyle=blue?'#a8c0ff':'rgba(18,18,30,.85)';
        ctx.fillRect(x+w*0.14,y+c*0.28,w*0.26,c*0.3);ctx.fillRect(x+w*0.58,y+c*0.28,w*0.26,c*0.3);
        ctx.fillStyle='#fff7ae';
        ctx.fillRect(v.dir>0?x+w-3:x+1,y+c*0.2,2,c*0.15);
        ctx.fillRect(v.dir>0?x+w-3:x+1,y+c*0.65,2,c*0.15);
      }else if(v.kind==='log'){
        ctx.fillStyle='#7a4a1f';rr(x+1,y+c*0.12,w-2,c*0.76,6);ctx.fill();
        ctx.strokeStyle='#5d3714';ctx.lineWidth=Math.max(1,c*0.05);
        for(let k=1;k<3;k++){ctx.beginPath();ctx.moveTo(x+4,y+c*0.12+k*c*0.25);
          ctx.lineTo(x+w-4,y+c*0.12+k*c*0.25);ctx.stroke();}
      }else{ // turtle
        const cx=x+w/2,cy=y+c*0.5;
        ctx.fillStyle='#3fae4c';
        ctx.beginPath();ctx.arc(cx+v.dir*w*0.42,cy,c*0.16,0,7);ctx.fill(); // head
        ctx.fillStyle='#2e8b3d';ctx.beginPath();ctx.ellipse(cx,cy,w*0.42,c*0.32,0,0,7);ctx.fill();
        ctx.strokeStyle='#1f6b2c';ctx.lineWidth=Math.max(1,c*0.05);
        for(let k=-1;k<=1;k++){ctx.beginPath();ctx.arc(cx,k*w*0.18,w*0.3,0,7);ctx.stroke();}
      }
    }}

  // player
  if(S.state!=='gameover'){
    const x=ox+S.px*c,y=oy+(S.py+0.5)*c;
    let color='#ffe135';
    if(S.powerT>0)color=Math.sin(T*20)>0?'#ffffff':'#7ef9ff';
    if(S.state==='dying'){
      drawPac(x,y,c*0.42,S.face,1.2,color,Math.max(0,S.dieT/1.1));
      ctx.strokeStyle='rgba(255,80,80,'+Math.max(0,S.dieT/1.1)+')';ctx.lineWidth=3;
      ctx.beginPath();ctx.arc(x,y,c*(1.6-S.dieT*0.9),0,7);ctx.stroke();
    }else{
      drawPac(x,y,c*0.42,S.face,0.18+0.3*Math.abs(Math.sin(T*9)),color,1);
    }
  }

  // HUD
  const fs=Math.round(Math.max(13,c*0.55)),hy=oy>36?oy-10:c*0.7;
  ctx.font='bold '+fs+'px "Courier New",monospace';ctx.fillStyle='#fff';
  ctx.textAlign='left';ctx.fillText('SCORE '+S.score,ox+8,hy);
  ctx.textAlign='center';ctx.fillText('HI '+HI,W/2,hy);
  for(let i=0;i<S.lives;i++)drawPac(ox+COLS*c-30-i*c*0.8,hy-5,c*0.26,1,0.5,'#ffe135',1);

  // message
  if(S.msgT>0){ctx.globalAlpha=Math.min(1,S.msgT);
    ctx.font='bold '+Math.round(c*0.9)+'px "Courier New",monospace';
    ctx.fillStyle='#ffd700';ctx.textAlign='center';
    ctx.fillText(S.msg,W/2,oy+ROWS*c*0.45);ctx.globalAlpha=1;}

  if(S.state==='gameover'){
    ctx.fillStyle='rgba(0,0,0,.7)';ctx.fillRect(ox,oy,COLS*c,ROWS*c);
    ctx.textAlign='center';ctx.fillStyle='#ff5252';
    ctx.font='bold '+Math.round(c*1.6)+'px "Courier New",monospace';
    ctx.fillText('GAME OVER',W/2,oy+ROWS*c*0.38);
    ctx.fillStyle='#fff';ctx.font='bold '+Math.round(c*0.9)+'px "Courier New",monospace';
    ctx.fillText('SCORE  '+S.score+'     HI  '+HI,W/2,oy+ROWS*c*0.52);
    if(Math.sin(T*5)>-0.3){ctx.fillStyle='#ffe135';
      ctx.fillText('PRESS ENTER OR R TO PLAY AGAIN',W/2,oy+ROWS*c*0.66);}
  }
}
