function tone(freq, dur, type, vol, delay){
  if(!actx) return;
  var t = actx.currentTime + (delay||0);
  var o = actx.createOscillator(), g = actx.createGain();

}

}

function drawCard(idx){
  if(G.state !== 'playing') return;

}

function passTurn(idx){
  if(G.state !== 'playing') return;

}

  maybeBotAct();
}

  if(G.state !== 'playing') return;

  if(p.human || !p.alive) return;

    botTimer = null;

    // Bot: estimate chance the next card is safe (bots peek at deck — they are "clever")
    var lt = lastTotal(), safe = 0, n = G.deck.length;

      var t = lt + cardValue(G.deck[i].rank);
      if(t <= 21 && G.totals.indexOf(t) === -1) safe++;

    else passTurn(G.turn);
  }, 900 + Math.random()*700);
}

function roundRect(x,y,w,h,r){
  ctx.beginPath();
  ctx.moveTo(x+r,y);
  ctx.arcTo(x+w,y,x+w,y+h,r);
  ctx.arcTo(x+w,y+h,x,y+h,r);
  ctx.arcTo(x,y+h,x,y,r);
  ctx.arcTo(x,y,x+w,y,r);
  ctx.closePath();
}

  ctx.restore();
  // center suit
  ctx.font = (w*0.62)+'px "Segoe UI Symbol", Arial';
  ctx.textAlign='center'; ctx.textBaseline='middle';
  ctx.fillText(card.suit.s, x+w/2, y+h*0.58);
  ctx.restore();
}

function drawCardBack(x,y,w,h){
  ctx.save();
  roundRect(x,y,w,h,Math.max(4,w*0.12));
  ctx.fillStyle = '#1c6b4d'; ctx.fill();
  ctx.lineWidth = Math.max(1.5, w*0.03);
  ctx.strokeStyle = '#ffd97a'; ctx.stroke();
  roundRect(x+w*0.12,y+h*0.1,w-w*0.24,h-h*0.2,Math.max(3,w*0.08));
  ctx.fillStyle = 'rgba(255,217,122,.16)'; ctx.fill();
  ctx.strokeStyle = 'rgba(255,217,122,.55)'; ctx.lineWidth = 1; ctx.stroke();
  ctx.fillStyle = '#ffd97a';
  ctx.font = (w*0.4)+'px Arial';
  ctx.textAlign='center'; ctx.textBaseline='middle';
  ctx.fillText('\u2660', x+w/2, y+h*0.5);
  ctx.restore();
}

var btnRects = [];

function drawButton(x,y,w,h,label,enabled,id){
  var b = {x:x,y:y,w:w,h:h,id:id};
  btnRects.push(b);
  ctx.save();
  roundRect(x,y,w,h,8);
  if(enabled){
    var g = ctx.createLinearGradient(0,y,0,y+h);
    g.addColorStop(0,'#ffd97a'); g.addColorStop(1,'#eaa925');
    ctx.fillStyle = g;
  } else {
    ctx.fillStyle = 'rgba(255,255,255,.14)';
  }
  ctx.fill();
  ctx.lineWidth = 2;
  ctx.strokeStyle = enabled ? '#8a6508' : 'rgba(255,255,255,.3)';
  ctx.stroke();
  ctx.fillStyle = enabled ? '#4a3200' : 'rgba(255,255,255,.5)';
  ctx.font = 'bold '+Math.max(13,h*0.42)+'px "Segoe UI", Arial';
  ctx.textAlign='center'; ctx.textBaseline='middle';
  ctx.fillText(label, x+w/2, y+h/2+1);
  ctx.restore();
}

function playerSlot(x,y,w,h,p,isTurn){
  // glow for turn
  if(isTurn && p.alive){
    ctx.save();
    ctx.shadowColor = '#ffd97a'; ctx.shadowBlur = 26;
    roundRect(x-w/2-8, y-8, w+16, h+16, 14);
    ctx.fillStyle = 'rgba(255,217,122,.10)'; ctx.fill();
    ctx.restore();
  }
  // name + wins
  ctx.save();
  ctx.font = 'bold '+Math.max(13,h*0.16)+'px "Segoe UI", Arial';
  ctx.textAlign='center'; ctx.textBaseline='alphabetic';
  ctx.fillStyle = p.alive ? (isTurn ? '#ffd97a' : '#eafaf1') : 'rgba(255,255,255,.4)';
  ctx.fillText(p.name + (p.wins>0 ? '  ★'+p.wins : ''), x, y - h*0.16);
  if(!p.alive){
    // OUT badge
    roundRect(x-w*0.32, y+h*0.28, w*0.64, h*0.34, 6);
    ctx.fillStyle = 'rgba(160,30,30,.85)'; ctx.fill();
    ctx.fillStyle = '#fff';
    ctx.font = 'bold '+Math.max(12,h*0.22)+'px "Segoe UI", Arial';
    ctx.textBaseline='middle';
    ctx.fillText('OUT', x, y+h*0.46);
  }
  ctx.restore();
}

/* ---------- Main render ---------- */
function render(now){
  requestAnimationFrame(render);
  if(!G) return;
  var dt = Math.min(0.1,(now-lastFrame)/1000); lastFrame = now;
  if(G.msgT > 0) G.msgT -= dt;

  btnRects = [];
  ctx.setTransform(DPR,0,0,DPR,0,0);

  // background felt
  var bg = ctx.createRadialGradient(W/2,H/2,50, W/2,H/2, Math.max(W,H)*0.75);
  bg.addColorStop(0,'#1a6b4c'); bg.addColorStop(1,'#082b1e');
  ctx.fillStyle = bg; ctx.fillRect(0,0,W,H);
  // subtle felt texture lines
  ctx.strokeStyle = 'rgba(255,255,255,.02)';
  for(var i=0;i<W;i+=26){ ctx.beginPath(); ctx.moveTo(i,0); ctx.lineTo(i,H); ctx.stroke(); }

  if(G.state === 'start'){ return; }

  var cw = Math.max(40, Math.min(W/15, H/8.5));
  var ch = cw*1.48;

  /* --- Bots (top) --- */
  var botY = Math.max(34, H*0.045);
  for(var i=1;i<G.players.length;i++){
    var p = G.players[i];
    var bx = W*(0.2 + 0.3*i);
    var isTurn = (G.turn===i && G.state==='playing');
    playerSlot(bx, botY+ch/2, cw*1.5, ch, p, isTurn);
    drawCardBack(bx-cw/2, botY, cw, ch);
  }

  /* --- Pile (center) --- */
  var px = W/2 - cw/2, py = H*0.26;
  ctx.save();
  ctx.font = 'bold '+Math.max(13,cw*0.2)+'px "Segoe UI", Arial';
  ctx.fillStyle = 'rgba(255,255,255,.7)';
  ctx.textAlign='center'; ctx.textBaseline='alphabetic';
  ctx.fillText('THE PILE', W/2, py-10);
  ctx.restore();

  var perRow = Math.max(3, Math.floor((W*0.62)/(cw+10)));
  for(var ci=0; ci<G.pile.length; ci++){
    var row = Math.floor(ci/perRow), col = ci%perRow;
    var rowCount = (row < G.pile.length/perRow-1) ? perRow : Math.ceil(G.pile.length/perRow);
    var rowW = rowCount*(cw+10)-10;
    var cx = W/2 - rowW/2 + col*(cw+10);
    var cy = py + row*(ch*0.62+24);
    drawCardFace(cx, cy, cw, ch, G.pile[ci]);
    // running total under card
    ctx.save();
    ctx.font = 'bold '+Math.max(13,cw*0.24)+'px "Segoe UI", Arial';
    ctx.textAlign='center'; ctx.textBaseline='top';
    var bust = G.pile.length===ci+1 && !G.players[G.turn] === false; // (no-op safety)
    ctx.fillStyle = '#ffd97a';
    ctx.fillText(''+lastTotalAtIndex(ci), cx+cw/2, cy+ch+4);
    ctx.restore();
  }

  /* --- Deck (right) --- */
  var dx = W - cw - Math.max(18,W*0.03), dy = H*0.26;
  drawCardBack(dx, dy, cw, ch);
  if(G.deck.length > 1){ drawCardBack(dx-5, dy+5, cw, ch); }
  ctx.save();
  ctx.font = 'bold '+Math.max(13,cw*0.24)+'px "Segoe UI", Arial';
  ctx.fillStyle='#eafaf1'; ctx.textAlign='center'; ctx.textBaseline='top';
  ctx.fillText(G.deck.length+' left', dx+cw/2, dy+ch+8);
  ctx.restore();

  /* --- Player (bottom) --- */
  var pp = G.players[0];
  var pyy = H - ch - Math.max(46,H*0.09);
  var isTurn = (G.turn===0 && G.state==='playing');
  playerSlot(W/2, pyy+ch/2, cw*1.5, ch, pp, isTurn);
  drawCardFace(W/2-cw/2, pyy, cw, ch, pp.card);

  var hint = lastTotal();
  ctx.save();
  ctx.font = 'bold '+Math.max(14,cw*0.26)+'px "Segoe UI", Arial';
  ctx.textAlign='center'; ctx.textBaseline='middle';
  ctx.fillStyle = '#ffd97a';
  ctx.fillText('Current total: '+hint, W/2, pyy + ch + Math.max(16,H*0.028));
  ctx.restore();

  // Buttons for human
  if(G.state === 'playing'){
    var bw = Math.min(150, W*0.2), bh = Math.max(38, H*0.055);
    var byy = pyy + ch + Math.max(34,H*0.055);
    var forced = forcedDraws();
    drawButton(W/2 - bw - 14, byy, bw, bh, 'DRAW [D]', isTurn, 'draw');
    drawButton(W/2 + 14, byy, bw, bh, forced ? 'FORCED!' : 'PASS [P]', isTurn && !forced, 'pass');
    if(forced && isTurn){
      ctx.save();
      ctx.font = 'bold '+Math.max(13,H*0.02)+'px "Segoe UI", Arial';
      ctx.fillStyle = '#ff9c7a'; ctx.textAlign='center';
      ctx.fillText('Everyone passed — you must draw!', W/2, byy - 8);
      ctx.restore();
    }
  }

  /* --- History strip --- */
  if(G.totals.length){
    var hist = 'Totals: ';
    for(var hi=0;hi<G.totals.length;hi++) hist += G.totals[hi] + (hi < G.totals.length-1 ? ' → ' : '');
    ctx.save();
    ctx.font = Math.max(12, cw*0.2)+'px "Segoe UI", Arial';
    ctx.fillStyle = 'rgba(255,255,255,.65)';
    ctx.textAlign='left'; ctx.textBaseline='top';
    var hx = Math.max(14, W*0.02);
    // wrap if too long
    while(ctx.measureText(hist).width > W*0.8 && hist.length>3){
      ctx.fillText(hist.slice(0, 60), hx, H-20);
      hist = '…' + hist.slice(58);
    }
    ctx.fillText(hist, hx, H-20);
    ctx.restore();
  }

  /* --- Message banner --- */
  if(G.msgT > 0 && G.message){
    var a = Math.min(1, G.msgT/0.4);
    ctx.save();
    ctx.globalAlpha = a;
    ctx.font = 'bold '+Math.max(18, H*0.035)+'px "Segoe UI", Arial';
    var tw = ctx.measureText(G.message).width;
    roundRect(W/2 - tw/2 - 24, H*0.115, tw+48, Math.max(36,H*0.06), 10);
    ctx.fillStyle = 'rgba(0,0,0,.55)'; ctx.fill();
    ctx.strokeStyle = '#ffd97a'; ctx.lineWidth=1.5; ctx.stroke();
    ctx.fillStyle = '#fff';
    ctx.textAlign='center'; ctx.textBaseline='middle';
    ctx.fillText(G.message, W/2, H*0.115 + Math.max(36,H*0.06)/2);
    ctx.restore();
  }

  /* --- Round over dim --- */
  if(G.state === 'roundover'){
    ctx.fillStyle = 'rgba(4,26,18,.35)';
    ctx.fillRect(0,0,W,H);
  }
}

function lastTotalAtIndex(ci){
  var t = 0;
  for(var i=0;i<=ci;i++) t += cardValue(G.pile[i].rank);
  return t;
}

newGame();
requestAnimationFrame(function(t){ lastFrame = t; render(t); });
})();
