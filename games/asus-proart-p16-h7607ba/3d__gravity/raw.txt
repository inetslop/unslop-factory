  if(od>520){ outOfZone=true; const pull=(od-520)*0.6; const d=vnorm(vscale(p.pos,-1)); p.vel.x+=d.x*pull*dt; p.vel.y+=d.y*pull*dt; p.vel.z+=d.z*pull*dt; }
  document.getElementById('warn').style.display = outOfZone ? 'block':'none';

  p.pos.x+=p.vel.x*dt; p.pos.y+=p.vel.y*dt; p.pos.z+=p.vel.z*dt;

  // orb collect
  for(const o of orbs){
    if(vlen(vsub(o.pos,p.pos))<11){ score++; sfxCollect(); o.pos=randOrbPos(); }
  }

  // collision
  for(const pl of planets){
    if(vlen(vsub(pl.pos,p.pos)) < pl.r+3){ gameOver(); return; }
  }

  // trail
  trail.push({x:p.pos.x,y:p.pos.y,z:p.pos.z});
  if(trail.length>46) trail.shift();

  // camera
  if(spd>4) camYaw=lerpAngle(camYaw, Math.atan2(p.vel.x,p.vel.z), 1-Math.exp(-3*dt));
  const f={x:Math.sin(camYaw),y:0,z:Math.cos(camYaw)};
  const desired=vadd(vsub(p.pos,vscale(f,62)),{x:0,y:30,z:0});
  const k=1-Math.exp(-5*dt);
  camPos.x+=(desired.x-camPos.x)*k; camPos.y+=(desired.y-camPos.y)*k; camPos.z+=(desired.z-camPos.z)*k;
  lookTarget=p.pos;

  document.getElementById('scoreEl').textContent='SCORE '+score;
  document.getElementById('spdEl').textContent='SPD '+vlen(p.vel).toFixed(1);
}

// ---------- rendering ----------
function drawPlanet(pl, pr){
  const R=pl.r*pr.s;
  if(R<0.4) return;
  // glow
  const g0=ctx.createRadialGradient(pr.x,pr.y,R*0.6,pr.x,pr.y,R*2.1);
  g0.addColorStop(0, pl.color+'55'); g0.addColorStop(1,'transparent');
  ctx.fillStyle=g0; ctx.beginPath(); ctx.arc(pr.x,pr.y,R*2.1,0,7); ctx.fill();
  // body (light from camera-ish upper-left)
  const lx=pr.x-R*0.4, ly=pr.y-R*0.45;
  const g=ctx.createRadialGradient(lx,ly,R*0.1,pr.x,pr.y,R);
  g.addColorStop(0,'#ffffff'); g.addColorStop(0.25,pl.color); g.addColorStop(1,'#0a0d1c');
  ctx.fillStyle=g; ctx.beginPath(); ctx.arc(pr.x,pr.y,R,0,7); ctx.fill();
  // ring
  if(pl.ring){
    ctx.save();
    ctx.translate(pr.x,pr.y); ctx.rotate(-0.5);
    ctx.strokeStyle='rgba(255,220,160,.5)'; ctx.lineWidth=Math.max(1,R*0.09);
    ctx.beginPath(); ctx.ellipse(0,0,R*1.7,R*0.55,0,0,7); ctx.stroke();
    ctx.strokeStyle='rgba(255,220,160,.22)'; ctx.lineWidth=Math.max(1,R*0.16);
    ctx.beginPath(); ctx.ellipse(0,0,R*2.05,R*0.68,0,0,7); ctx.stroke();
    ctx.restore();
  }
}

function drawOrb(o){
  const pr=project(o.pos); if(!pr) return;
  const R=Math.max(2.5, 4.5*pr.s)*(1+0.25*Math.sin(time*5+o.pulse));
  const g=ctx.createRadialGradient(pr.x,pr.y,0,pr.x,pr.y,R*3);
  g.addColorStop(0,'#ffffff'); g.addColorStop(0.3,'#aef6ff'); g.addColorStop(1,'transparent');
  ctx.fillStyle=g; ctx.beginPath(); ctx.arc(pr.x,pr.y,R*3,0,7); ctx.fill();
}

function drawPlayer(){
  // trail
  for(let i=1;i<trail.length;i++){
    const a=project(trail[i-1]), b=project(trail[i]);
    if(!a||!b) continue;
    const t=i/trail.length;
    ctx.strokeStyle='rgba(140,220,255,'+(t*0.5).toFixed(3)+')';
    ctx.lineWidth=Math.max(0.5, 2.4*t*Math.min(b.s/600,2));
    ctx.beginPath(); ctx.moveTo(a.x,a.y); ctx.lineTo(b.x,b.y); ctx.stroke();
  }
  const pr=project(player.pos); if(!pr) return;
  const R=Math.max(3, 3*pr.s);
  // engine flame opposite velocity (screen approx via right vector)
  const v=player.vel, sl=vlen(v);
  if(sl>2){
    const fx=-v.x/sl, fz=-v.z/sl;
    const sx=fx*cam.r.x+fz*cam.r.z, sy=(0)*cam.u.y-0; // horizontal only
    const fl=R*(1.6+Math.random()*1.4);
    ctx.strokeStyle='rgba(255,180,90,.8)'; ctx.lineWidth=Math.max(1,R*0.5);
    ctx.beginPath(); ctx.moveTo(pr.x,pr.y); ctx.lineTo(pr.x+sx*fl, pr.y-sy*fl); ctx.stroke();
  }
  const g=ctx.createRadialGradient(pr.x,pr.y,0,pr.x,pr.y,R*4);
  g.addColorStop(0,'#ffffff'); g.addColorStop(0.35,'#9fdcff'); g.addColorStop(1,'transparent');
  ctx.fillStyle=g; ctx.beginPath(); ctx.arc(pr.x,pr.y,R*4,0,7); ctx.fill();
}

function render(){
  // background
  const bg=ctx.createRadialGradient(W/2,H*0.45,0,W/2,H/2,Math.max(W,H)*0.8);
  bg.addColorStop(0,'#0c1330'); bg.addColorStop(0.6,'#060a1c'); bg.addColorStop(1,'#02030a');
  ctx.fillStyle=bg; ctx.fillRect(0,0,W,H);

  updateCamera();

  // stars
  for(const s of stars){
    const pr=project(s); if(!pr) continue;
    const tw=0.5+0.5*Math.sin(time*2+s.x);
    ctx.fillStyle='rgba(200,215,255,'+(0.25+0.5*tw).toFixed(2)+')';
    ctx.fillRect(pr.x,pr.y,s.s,s.s);
  }

  // depth-sorted objects
  const list=[];
  for(const pl of planets){ const pr=project(pl.pos); if(pr) list.push({z:pr.z, fn:()=>drawPlanet(pl,pr)}); }
  for(const o of orbs){ const pr=project(o.pos); if(pr) list.push({z:pr.z, fn:()=>drawOrb(o)}); }
  if(state!=='start') list.push({z:(project(player.pos)||{z:1e9}).z, fn:drawPlayer});
  list.sort((a,b)=>b.z-a.z);
  for(const it of list) it.fn();

  // start screen slow orbit camera
  if(state==='start'){
    const a=time*0.12;
    camPos={x:Math.cos(a)*300, y:90+Math.sin(time*0.3)*20, z:Math.sin(a)*300};
    lookTarget={x:0,y:0,z:0};
  }
}

// ---------- main loop ----------
let last=performance.now();
function loop(now){
  const dt=Math.min(0.05,(now-last)/1000); last=now;
  if(state==='playing') update(dt);
  else time+=dt;
  render();
  requestAnimationFrame(loop);
}
reset(); state='start';
document.getElementById('hud').classList.add('hidden');
requestAnimationFrame(loop);
})();
