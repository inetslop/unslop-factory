function step(dt){
  if(state!=='play') return;
  // pending auto-fire
  if(pendingFire>0){ pendingFire-=dt; if(pendingFire<=0){ fireShot(2+word.length*1.5); return; } }

  if(!ballStopped()){
    var steps=3, s=dt/steps;
    for(var i=0;i<steps && !ballStopped();i++){
      ball.x+=ball.vx*s*60/1; // v is px/frame@60
      ball.y+=ball.vy*s*60/1;

      // simpler: apply friction per substep
      var f = inSand(ball.x,ball.y) ? 0.942 : 0.9885;
      ball.vx*=f; ball.vy*=f;

    }
    if(Math.hypot(ball.vx,ball.vy)<0.06){ ball.vx=0; ball.vy=0; renderWord(false); }

  for(var i=0;i<wl.length;i++){
    var w=wl[i], cx=Math.max(w[0],Math.min(ball.x,w[0]+w[2])), cy=Math.max(w[1],Math.min(ball.y,w[1]+w[3]));
    var dx=ball.x-cx, dy=ball.y-cy, d2=dx*dx+dy*dy;
    if(d2<r*r){
      var d=Math.sqrt(d2)||0.0001, nx=dx/d, ny=dy/d;
      ball.x=cx+nx*r; ball.y=cy+ny*r;
      var dot=ball.vx*nx+ball.vy*ny;
      if(dot<0){ ball.vx-=2*dot*nx*0.85; ball.vy-=2*dot*ny*0.85; bounced=true; }
    }
  }
  if(bounced && Math.hypot(ball.vx,ball.vy)>1) sBounce();
}

function sinkHole(){
  state='holeDone'; holeDoneTimer=1.7;
  ball.vx=0; ball.vy=0;
  var L=LEVELS[level]; ball.x=L.hole.x; ball.y=L.hole.y;
  sSink();
  var diff=strokes-L.par, msg;
  if(diff<=-2) msg='ALBATROSS!!'; else if(diff===-1) msg='BIRDIE!';
  else if(diff===0) msg='PAR'; else if(diff===1) msg='BOGEY'; else msg='+'+diff;
  showBanner(msg, 1500);
}

function endGame(){
  state='end';
  var totalPar=LEVELS.reduce(function(a,l){return a+l.par;},0);
  var d=totalStrokes-totalPar;
  var rating = d<=-3 ? 'GRAND SLAM ACE! You are a legend.' :
               d<=-1 ? 'TOUR PRO — dazzling type-and-putt!' :
               d===0  ? 'CLUB CHAMPION — solid round on the greens.' :
               d<=2   ? 'CLUBHOUSE REGULAR — keep practicing those fingers.' :
                        'GOLFING ROOKIE — the word processor awaits!';
  document.getElementById('endTitle').textContent='&#9873; ROUND COMPLETE!'.replace('&#9873;','\u26F3');
  document.getElementById('endStats').innerHTML =
    'Total strokes: <b>'+totalStrokes+'</b> &nbsp;vs&nbsp; Par <b>'+totalPar+'</b><br>' +
    '<span style="font-size:min(5vw,26px);color:#ffe14d;font-weight:bold;">'+rating+'</span>';
  hud.classList.add('hidden'); wordBox.classList.add('hidden');
  document.getElementById('end').classList.remove('hidden');
  sDone();
}

/* ---------- DRAW ---------- */
function draw(){
  ctx.setTransform(1,0,0,1,0,0);
  ctx.fillStyle='#0b3d1f'; ctx.fillRect(0,0,canvas.width,canvas.height);
  ctx.setTransform(scale,0,0,scale,ox,oy);

  if(state==='start') return;

  var L=LEVELS[level];
  // green
  var g=ctx.createLinearGradient(0,0,0,H);
  g.addColorStop(0,'#3fae5a'); g.addColorStop(1,'#2f9448');
  ctx.fillStyle=g; ctx.fillRect(0,0,W,H);
  // mow stripes
  ctx.fillStyle='rgba(255,255,255,0.05)';
  for(var i=0;i<W;i+=120) if((i/120)%2===0) ctx.fillRect(i,0,60,H);

  // sand
  L.sand.forEach(function(s){
    ctx.beginPath(); ctx.arc(s.x,s.y,s.r,0,7);
    ctx.fillStyle='#e8d493'; ctx.fill();
    ctx.strokeStyle='#c9b06a'; ctx.lineWidth=4; ctx.stroke();
  });

  // hole + flag
  ctx.beginPath(); ctx.arc(L.hole.x,L.hole.y,16,0,7);
  ctx.fillStyle='#111'; ctx.fill();
  ctx.strokeStyle='#000'; ctx.lineWidth=2; ctx.stroke();
  ctx.strokeStyle='#fff'; ctx.lineWidth=3;
  ctx.beginPath(); ctx.moveTo(L.hole.x,L.hole.y-4); ctx.lineTo(L.hole.x,L.hole.y-58); ctx.stroke();
  ctx.fillStyle='#e33';
  ctx.beginPath(); ctx.moveTo(L.hole.x,L.hole.y-58); ctx.lineTo(L.hole.x+30,L.hole.y-48); ctx.lineTo(L.hole.x,L.hole.y-38); ctx.closePath(); ctx.fill();

  // walls
  L.walls.forEach(function(w){
    ctx.fillStyle='#20415e'; ctx.fillRect(w[0],w[1],w[2],w[3]);
    ctx.strokeStyle='#3d6a94'; ctx.lineWidth=3; ctx.strokeRect(w[0]+1.5,w[1]+1.5,w[2]-3,w[3]-3);
  });

  // aim line (only when can type)
  if(state==='play' && ballStopped()){
    var len=40+(pendingFire>0?word.length:(typed<word.length?word.length:word.length))*8;
    if(pendingFire<=0) len=46+typed*9;
    var ex=ball.x+Math.cos(aim)*len, ey=ball.y+Math.sin(aim)*len;
    ctx.save();
    ctx.strokeStyle=pendingFire>0?'#ffe14d':'rgba(255,255,255,0.85)';
    ctx.lineWidth=3; ctx.setLineDash([8,7]);
    ctx.beginPath(); ctx.moveTo(ball.x+Math.cos(aim)*16, ball.y+Math.sin(aim)*16); ctx.lineTo(ex,ey); ctx.stroke();
    ctx.setLineDash([]);
    // arrowhead
    ctx.fillStyle=ctx.strokeStyle;
    ctx.translate(ex,ey); ctx.rotate(aim);
    ctx.beginPath(); ctx.moveTo(0,0); ctx.lineTo(-12,-7); ctx.lineTo(-12,7); ctx.closePath(); ctx.fill();
    ctx.restore();
  }

  // ball
  if(state!=='end'){
    ctx.save();
    ctx.shadowColor='rgba(0,0,0,0.4)'; ctx.shadowBlur=6; ctx.shadowOffsetY=3;
    var bg=ctx.createRadialGradient(ball.x-3,ball.y-4,2,ball.x,ball.y,ball.r);
    bg.addColorStop(0,'#ffffff'); bg.addColorStop(1,'#cfcfcf');
    ctx.fillStyle=bg;
    ctx.beginPath(); ctx.arc(ball.x,ball.y,ball.r,0,7); ctx.fill();
    ctx.restore();
  }

  if(errFlash>0){ errFlash-=1/60; }
}

/* ---------- LOOP ---------- */
var last=performance.now();
function loop(t){
  var dt=Math.min(0.05,(t-last)/1000); last=t;
  step(dt);
  if(state==='holeDone'){
    holeDoneTimer-=dt;
    if(holeDoneTimer<=0){
      level++;
      if(level>=LEVELS.length) endGame();
      else beginHole();
    }
  }
  draw();
  requestAnimationFrame(loop);
}
requestAnimationFrame(loop);
