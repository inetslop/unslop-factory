/* ---------- physics ---------- */
function segCollide(o,r,seg,rest){
  var dx=seg.x2-seg.x1,dy=seg.y2-seg.y1,L2=dx*dx+dy*dy||1;
  var t=((o.x-seg.x1)*dx+(o.y-seg.y1)*dy)/L2;t=t<0?0:(t>1?1:t);
  var qx=seg.x1+t*dx,qy=seg.y1+t*dy;
  var nx=o.x-qx,ny=o.y-qy,d2=nx*nx+ny*ny;
  if(d2>=r*r)return false;
  var d=Math.sqrt(d2)||0.0001;nx/=d;ny/=d;
  o.x=qx+nx*r;o.y=qy+ny*r;
  var vn=o.vx*nx+o.vy*ny;
  if(vn<0){o.vx-=(1+rest)*vn*nx;o.vy-=(1+rest)*vn*ny;}
  return true;
}
function flipCollide(f,o){
  var dx=Math.cos(f.angle),dy=Math.sin(f.angle);
  var tx=f.px+dx*f.len,ty=f.py+dy*f.len;
  var sx=tx-f.px,sy=ty-f.py,L2=sx*sx+sy*sy;
  var t=((o.x-f.px)*sx+(o.y-f.py)*sy)/L2;t=t<0?0:(t>1?1:t);
  var qx=f.px+t*sx,qy=f.py+t*sy;
  var nx=o.x-qx,ny=o.y-qy,d=Math.sqrt(nx*nx+ny*ny),R=f.r+o.r;
  if(d>=R||d===0)return;
  nx/=d;ny/=d;
  o.x=qx+nx*R;o.y=qy+ny*R;
  var rx=qx-f.px,ry=qy-f.py;
  var svx=-f.omega*ry,svy=f.omega*rx;
  var rvx=o.vx-svx,rvy=o.vy-svy;
  var vn=rvx*nx+rvy*ny;
  if(vn<0){rvx-=1.5*vn*nx;rvy-=1.5*vn*ny;}
  o.vx=svx+rvx;o.vy=svy+rvy;
}

}

function update(dt){
  if(state!=='playing')return;
  time+=dt;

  // flippers
  for(var i=0;i<flippers.length;i++){
    var f=flippers[i];
    var pressed=(f.side==='L'?input.left:input.right);
    if(pressed&&!f.prev)sFlip();
    f.prev=pressed;
    var target=pressed?f.active:f.rest;
    var prevA=f.angle;
    var delta=target-f.angle;
    var step=Math.max(-f.speed*dt,Math.min(f.speed*dt,delta));
    f.angle+=step;
    f.omega=(f.angle-prevA)/dt;
  }

  // plunger
  if(input.plunger&&locked){power=Math.min(1,power+dt*1.3);}
  else if(!input.plunger&&locked&&power>0){
    locked=false;
    b.vx=0;b.vy=-(700+950*power);
    sLaunch();
    power=0;
  }

  if(locked)return; // ball waits on plunger

  var sub=8,h=dt/sub;
  for(var s=0;s<sub;s++){
    b.vy+=G*h;
    b.x+=b.vx*h;b.y+=b.vy*h;
    var sp=Math.sqrt(b.vx*b.vx+b.vy*b.vy);
    if(sp>1700){b.vx*=1700/sp;b.vy*=1700/sp;}
    for(var w=0;w<walls.length;w++){
      var seg=walls[w];
      if(segCollide(b,10,seg,seg.rest)&&seg.sling){
        var n2x=b.x-(seg.x1+(seg.x2-seg.x1)*0.5),n2y=b.y-(seg.y1+(seg.y2-seg.y1)*0.5);
        var nn=Math.sqrt(n2x*n2x+n2y*n2y)||1;
        var bs=Math.sqrt(b.vx*b.vx+b.vy*b.vy);
        if(bs<830){b.vx=n2x/nn*830;b.vy=n2y/nn*830;}
        if(time-slingCd[seg.sling]>0.35){slingCd[seg.sling]=time;score+=20;sSling();}
      }
    }
    flipCollide(flippers[0],b);
    flipCollide(flippers[1],b);
  }

  // bumpers
  for(var bi=0;bi<bumpers.length;bi++){
    var bp=bumpers[bi];
    if(bp.flash>0)bp.flash-=dt;
    var dx=b.x-bp.x,dy=b.y-bp.y,d=Math.sqrt(dx*dx+dy*dy),R=bp.r+10;
    if(d<R&&d>0.001){
      var nx=dx/d,ny=dy/d;
      b.x=bp.x+nx*R;b.y=bp.y+ny*R;
      var vn=b.vx*nx+b.vy*ny;
      if(vn<0){b.vx-=2*vn*nx;b.vy-=2*vn*ny;}
      var sp2=Math.sqrt(b.vx*b.vx+b.vy*b.vy);
      if(sp2<950){b.vx=nx*950;b.vy=ny*950;}
      if(bp.flash<=0){score+=50;sBump();}
      bp.flash=0.18;
    }
  }

  // dots
  var left=0;
  for(var di=0;di<dots.length;di++){
    var dt2=dots[di];
    if(!dt2.e){
      var ddx=b.x-dt2.x,ddy=b.y-dt2.y;
      if(ddx*ddx+ddy*ddy<169){dt2.e=true;score+=10;sDot();}
      else left++;
    }
  }
  if(left===0)waveComplete();

  // ghosts
  for(var gi=0;gi<ghosts.length;gi++){
    var g=ghosts[gi];
    var tx2=240,ty2=170; // home (ball is always un-locked here)
    var gdx=b.x-g.x,gdy=b.y-g.y,gd=Math.sqrt(gdx*gdx+gdy*gdy)||1;
    g.x+=gdx/gd*g.speed*dt;g.y+=gdy/gd*g.speed*dt;
    for(var gw=0;gw<walls.length;gw++)segCollide(g,9,walls[gw],0.2);
    for(var gb=0;gb<bumpers.length;gb++){
      var bb=bumpers[gb];
      var bdx=g.x-bb.x,bdy=g.y-bb.y,bd=Math.sqrt(bdx*bdx+bdy*bdy),BR=bb.r+9;
      if(bd<BR&&bd>0.001){g.x=bb.x+bdx/bd*BR;g.y=bb.y+bdy/bd*BR;}
    }
    if(gd<19){loseLife();return;}
  }

  // drain
  if(b.y>H+40)loseLife();
}
