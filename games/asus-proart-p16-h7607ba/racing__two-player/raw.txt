function control(car, up, dn, lf, rt){
  var th = car.a;
  var fx = Math.cos(th), fy = Math.sin(th);
  // forward speed
  var fSpd = car.vx*fx + car.vy*fy;
  // throttle / brake
  if(up)      { car.vx += fx*0.42*dt; car.vy += fy*0.42*dt; }
  if(dn)      {
    if(fSpd > 0.02){ car.vx -= fx*0.55*dt; car.vy -= fy*0.55*dt; }        // brake
    else           { car.vx -= fx*0.28*dt; car.vy -= fy*0.28*dt; }        // reverse
  }
  // drag
  var dr = car.offtrack ? 3.4 : 1.5;
  car.vx -= car.vx*dr*dt; car.vy -= car.vy*dr*dt;
  // lateral grip
  var latX = car.vx - fSpd*fx, latY = car.vy - fSpd*fy;
  var grip = car.offtrack ? 0.86 : 0.94;
  car.vx = fSpd*fx + latX*grip;
  car.vy = fSpd*fy + latY*grip;
  // speed clamp
  var sp = Math.hypot(car.vx,car.vy);
  var maxF = car.offtrack ? 0.17 : 0.46;
  var maxR = 0.16;
  if(fSpd > maxF){ var s=maxF/sp; car.vx*=s; car.vy*=s; }
  else if(fSpd < -maxR){ var s2=(-maxR)/sp; car.vx*=s2; car.vy*=s2; }
  // steering
  var steer = (rt?1:0) - (lf?1:0);
  if(steer){
    var fac = Math.min(1, Math.abs(fSpd)/0.13);
    car.a += steer * 2.7 * dt * fac * (fSpd<0?-1:1);
  }
}

    updateEngine(i, Math.hypot(c.vx,c.vy));
  }

  // car-car collision
  var dx=c2.x-c1.x, dy=c2.y-c1.y, d=Math.hypot(dx,dy), R=0.030;
  if(d>0 && d< R*2){
    var nx=dx/d, ny=dy/d, push=(R*2-d)/2;
    c1.x-=nx*push; c1.y-=ny*push; c2.x+=nx*push; c2.y+=ny*push;
    c1.vx*=0.82; c1.vy*=0.82; c2.vx*=0.82; c2.vy*=0.82;
    if(Math.random()<0.5) beep(180,0.06,'sawtooth',0.1);
  }

  // win
  for(var j=0;j<2;j++){
    if(cars[j].laps >= LAPS_TO_WIN){
      winner = j; state='finished'; stopEngines();
      beep(523,0.15,'square',0.2); 
      setTimeout(function(){beep(659,0.15,'square',0.2);},140);
      setTimeout(function(){beep(784,0.15,'square',0.2);},280);
      setTimeout(function(){beep(1047,0.4,'square',0.22);},430);
      break;
    }
  }
}

/* ---------- Rendering ---------- */
function worldTransform(){
  var M = Math.min(W,H), s = M*dpr;
  var ox = (W - M)*dpr/1, oy = (H - M)*dpr/1;
  // center the square
  ox = (W*dpr - s)/2; oy = (H*dpr - s)/2;
  ctx.setTransform(s,0,0,s,ox,oy);
}

function drawTrack(){
  // grass
  ctx.fillStyle = '#1e5c2a';
  ctx.fillRect(-0.3,-0.3,1.6,1.6);
  // mow stripes
  ctx.fillStyle = 'rgba(255,255,255,0.04)';
  for(var i=0;i<10;i+=2) ctx.fillRect(-0.3, i*0.1-0.3, 1.6, 0.1);

  function strokePath(w,color,dash){
    ctx.beginPath();
    ctx.moveTo(PTS[0][0],PTS[0][1]);
    for(var i=1;i<N;i++) ctx.lineTo(PTS[i][0],PTS[i][1]);
    ctx.closePath();
    ctx.lineJoin='round'; ctx.lineCap='round';
    ctx.setLineDash(dash||[]);
    ctx.lineWidth=w; ctx.strokeStyle=color; ctx.stroke();
    ctx.setLineDash([]);
  }
  strokePath(TRACK_HALF*2+0.030, '#d8d8d8');   // edge lines
  strokePath(TRACK_HALF*2, '#4a4a52');          // asphalt
  strokePath(0.006, 'rgba(255,215,0,0.7)', [0.03,0.035]); // center dashes

  // start / finish checkered line
  var d = dirAt(0), pp=[-d[1],d[0]], c=PTS[0];
  var sq=0.014, cols=Math.floor(TRACK_HALF/sq);
  ctx.save();
  for(var r=0;r<2;r++){
    for(var k=-cols;k<cols;k++){
      var bx = c[0] + pp[0]*(k*sq + sq/2) + d[0]*(r*sq - sq/2);
      var by = c[1] + pp[1]*(k*sq + sq/2) + d[1]*(r*sq - sq/2);
      ctx.save();
      ctx.translate(bx,by);
      ctx.rotate(Math.atan2(d[1],d[0]));
      ctx.fillStyle = ((k+r)%2===0)?'#f4f4f4':'#111';
      ctx.fillRect(-sq/2,-sq/2,sq,sq);
      ctx.restore();
    }
  }
  ctx.restore();
}

function drawCar(car, ghost){
  ctx.save();
  ctx.translate(car.x, car.y);
  ctx.rotate(car.a);
  var L=0.048, Wd=0.026;
  // shadow
  ctx.fillStyle='rgba(0,0,0,0.3)';
  ctx.fillRect(-L/2+0.004, -Wd/2+0.005, L, Wd);
  // wheels
  ctx.fillStyle='#111';
  ctx.fillRect(-L*0.42,-Wd/2-0.004,L*0.2,0.008);
  ctx.fillRect(-L*0.42, Wd/2-0.004,L*0.2,0.008);
  ctx.fillRect( L*0.18,-Wd/2-0.004,L*0.2,0.008);
  ctx.fillRect( L*0.18, Wd/2-0.004,L*0.2,0.008);
  // body
  ctx.fillStyle = ghost? 'rgba(255,255,255,0.35)' : car.color;
  roundRect(-L/2,-Wd/2,L,Wd,0.006); ctx.fill();
  // nose highlight
  ctx.fillStyle='rgba(255,255,255,0.35)';
  roundRect(L*0.08,-Wd/2+0.004,L*0.36,Wd-0.008,0.004); ctx.fill();
  // cockpit
  ctx.fillStyle='#1b2735';
  roundRect(-L*0.22,-Wd/2+0.005,L*0.34,Wd-0.010,0.004); ctx.fill();
  ctx.restore();
}
function roundRect(x,y,w,h,r){
  ctx.beginPath();
  ctx.moveTo(x+r,y);
  ctx.arcTo(x+w,y,x+w,y+h,r);
  ctx.arcTo(x+w,y+h,x,y+h,r);
  ctx.arcTo(x,y+h,x,y,r);
  ctx.arcTo(x,y,x+w,y,r);
  ctx.closePath();
}

function screenTransform(){ ctx.setTransform(dpr,0,0,dpr,0,0); }

function fmt(t){

}
