function idx(x,y){return y*COLS+x;}

function adjacent(x,y){
  return [[1,0],[-1,0],[0,1],[0,-1]].some(d=>{
    const nx=x+d[0],ny=y+d[1];
    return inB(nx,ny)&&!blocked(nx,ny,[player]);

function computeReachable(){
  reachable=new Set();

  }

const sleep=ms=>new Promise(r=>setTimeout(r,ms));
function bfsPath(sx,sy,tx,ty,obstacles){
  const prev={};prev[idx(sx,sy)]=null;
  const q=[[sx,sy]];
  while(q.length){
    const [x,y]=q.shift();
    if(x===tx&&y===ty){
      const path=[];let i=idx(x,y);
      while(prev[i]!==null){path.unshift([i%COLS,Math.floor(i/COLS)]);i=prev[i];}
      return path.slice(1);
    }
    for(const d of [[1,0],[-1,0],[0,1],[0,-1]]){
      const nx=x+d[0],ny=y+d[1],ni=idx(nx,ny);
      if(!inB(nx,ny)||grid[ni]===1)continue;
      if(obstacles.some(o=>o.x===nx&&o.y===ny))continue;
      if(prev[ni]!==undefined)continue;
      prev[ni]=idx(x,y);q.push([nx,ny]);
    }
  }
  return null;
}

function enemyAdjacentToPlayer(e){
  return Math.abs(e.x-player.x)+Math.abs(e.y-player.y)===1;
}

async function enemyTurns(){
  const token=animToken;
  player.acting=false;
  computeReachable();

    if(token!==animToken)return;

      doAttack(e,player,e.dmg);
      await sleep(600);
      continue;
    }
    const others=enemies.filter(o=>o!==e&&o.hp>0);
    const path=bfsPath(e.x,e.y,player.x,player.y,others);
    let steps=3;
    if(path){
      for(const [px,py] of path){
        if(steps<=0)break;
        if(enemyAdjacentToPlayer(e))break;

        beep(180+Math.random()*60,0.04,'sine',0.03);
        await sleep(170);
        if(token!==animToken)return;
      }
    }
    if(enemyAdjacentToPlayer(e)){
      doAttack(e,player,e.dmg);
      await sleep(600);
    }
    if(player.hp<=0){state='lost';return;}
  }
  if(token!==animToken)return;
  turnNum++;
  player.mp=3;
  player.acting=true;
  computeReachable();
}

function endPlayerTurn(){
