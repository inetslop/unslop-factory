function tryMove(p,dx,dy,crateArr){
  const targets=cellsOf(p).map(([c,r])=>[c+dx,r+dy]);
  // bounds
  for(const [c,r] of targets) if(c<0||c>=W||r>H) return false;

  // crates hit
  const pushed=[];
  for(const [c,r] of targets){
    if(r<0) continue;
    const idx=crateArr.findIndex(k=>k.c===c&&k.r===r);
    if(idx>=0) pushed.push({idx, oc:c, or:r, dc:c+dx, dr:r+dy});
  }

  if(pushed.length){
    // verify each push destination
    for(const q of pushed){
      if(!inBounds(q.dc,q.dr)) return false;
      if(board[q.dr][q.dc]) return false;
      if(crateArr.some((k,i)=>i!==q.idx && k.c===q.dc && k.r===q.dr)) return false;
    }
    // no two pushes into same cell
    for(let i=0;i<pushed.length;i++)for(let j=i+1;j<pushed.length;j++)
      if(pushed[i].dc===pushed[j].dc && pushed[i].dr===pushed[j].dr) return false;

    // remaining target cells must be free (old crate cells become free)
    for(const [c,r] of targets){
      if(r<0) continue;
      const isOldCrate=pushed.some(q=>q.oc===c&&q.or===r);
      if(isOldCrate) continue;
      if(board[r][c]) return false;
      if(crateArr.some((k,i)=>i!==-1 && k.c===c && k.r===r)) return false;
    }

    for(const q of pushed){ const k=crateArr[q.idx]; k.c=q.dc; k.r=q.dr; }
    p.c+=dx; p.r+=dy;
    sfx('push'); score+=10;
    checkWin();
    return true;
  }

  // plain move: no crate involved
  for(const [c,r] of targets){
    if(r<0) continue;
    if(board[r][c]) return false;
    if(crateArr.some(k=>k.c===c&&k.r===r)) return false;
  }
  p.c+=dx; p.r+=dy;
  return true;
}

function rotate(dir){
  const m=cur.m, n=m.length;
  const nm=[];
  for(let r=0;r<n;r++){ nm.push([]); for(let c=0;c<n;c++)
    nm[r].push(dir>0 ? m[n-1-c][r] : m[c][n-1-r]); }
  // trim empty rows/cols to keep footprint tight (only matters for asymmetric shapes)
  const old=m; cur.m=nm;
  for(const kick of [0,-1,1,-2,2]){
    if(cellsOf(cur).every(([c,r])=> r<0 ? true : (!boardAt(c,r) && !crates.some(k=>k.c===c&&k.r===r)))){
      cur.c+=kick; sfx('rotate'); return;
    }
  }
  cur.m=old;
}

function hardDrop(){
  let d=0;
  while(tryMove(cur,0,1)) d++;
  if(d>0) score+=2*d;
  sfx('drop');
  lock();
}

function lock(){
  let anyAbove=false;
  for(const [c,r] of cellsOf(cur)){
    if(r<0){ anyAbove=true; continue; }
    board[r][c]=cur.col;
  }
  if(anyAbove){ gameOver(); return; }
  sfx('lock');

  // line clears (rows without crates)
  let cleared=0;
  for(let r=H-1;r>=0;r--){
    const hasCrate=crates.some(k=>k.r===r);
    if(hasCrate) continue;
    if(board[r].every(v=>v)){
      board.splice(r,1); board.unshift(Array(W).fill(null));
      crates.forEach(k=>{ if(k.r>r) k.r--; });
      cleared++; r++;
    }
  }
  if(cleared){ score+=[0,40,100,300,1200][cleared]*Math.max(1,level); sfx('clear'); flash('CLEAR! x'+cleared); }

  checkWin();
  if(state==='play') spawn();
}

function checkWin(){
  if(state!=='play') return;
  if(crates.length && crates.every(k=>goals.has(key(k.c,k.r)))){
    state='levelup'; flashTime=1.4; flashText='LEVEL '+level+' COMPLETE! +500';
    score+=500; sfx('win');
  }
}

function flash(t){ flashText=t; flashTime=1.0; }

function startGame(){
  level=1; score=0; paused=false;
  loadLevel(0); spawn();
  state='play'; gravTimer=0;
  document.getElementById('startScreen').classList.add('hidden');
  document.getElementById('overScreen').classList.add('hidden');
}

function gameOver(){
  state='over'; sfx('lose');
  if(score>best){ best=score; try{localStorage.setItem('sokotet_best',''+best);}catch(e){} }
  document.getElementById('finalScore').innerHTML='Score: <b style="color:#ffd75e">'+score+'</b><br>Level reached: '+level;
  document.getElementById('bestScore').textContent='Best: '+best;
  document.getElementById('overScreen').classList.remove('hidden');
}

window.addEventListener('keydown',function(e){
  initAudio();
  const k=e.key;
  if(['ArrowLeft','ArrowRight','ArrowDown','ArrowUp',' '].includes(k)) e.preventDefault();

  if(state==='start' && (k==='Enter'||k===' ')){ startGame(); return; }
  if(state==='over' && (k==='Enter'||k.toLowerCase()==='r')){ startGame(); return; }
  if(state!=='play') return;
  if(k.toLowerCase()==='p'){ paused=!paused; return; }
  if(paused) return;
  if(k.toLowerCase()==='r'){ startGame(); return; }

  switch(k){
    case 'ArrowLeft':  if(tryMove(cur,-1,0)) sfx('move'); break;
    case 'ArrowRight': if(tryMove(cur, 1,0)) sfx('move'); break;
    case 'ArrowDown':  softDrop=true; if(tryMove(cur,0,1)){ score+=1; } break;
    case 'ArrowUp': case 'x': case 'X': rotate(1); break;
    case 'z': case 'Z': rotate(-1); break;
    case ' ': hardDrop(); break;
  }
});
window.addEventListener('keyup',function(e){ if(e.key==='ArrowDown') softDrop=false; });
document.getElementById('startBtn').addEventListener('click',function(){ initAudio(); startGame(); });
document.getElementById('restartBtn').addEventListener('click',function(){ initAudio(); startGame(); });

/* ---------- drawing ---------- */
function drawCell(c,r,color,alpha,ghost){
  const x=ox+c*cell, y=oy+r*cell;
  ctx.save(); ctx.globalAlpha=alpha==null?1:alpha;
  if(ghost){
    ctx.strokeStyle=color; ctx.lineWidth=2;
    ctx.strokeRect(x+3,y+3,cell-6,cell-6);
  }else{
    ctx.fillStyle=color;
    roundRect(x+1.5,y+1.5,cell-3,cell-3,4); ctx.fill();
    ctx.fillStyle='rgba(255,255,255,.25)';
    roundRect(x+1.5,y+1.5,cell-3,(cell-3)*0.35,4); ctx.fill();
  }
  ctx.restore();
}
function roundRect(x,y,w,h,r){
  ctx.beginPath();
  ctx.moveTo(x+r,y); ctx.arcTo(x+w,y,x+w,y+h,r); ctx.arcTo(x+w,y+h,x,y+h,r);
  ctx.arcTo(x,y+h,x,y,r); ctx.arcTo(x,y,x+w,y,r); ctx.closePath();
}

function drawCrate(k){
  const x=ox+k.c*cell, y=oy+k.r*cell;
  const onGoal=goals.has(key(k.c,k.r));
  ctx.save();
  if(onGoal){ ctx.shadowColor='#5dff7c'; ctx.shadowBlur=14; }
  ctx.fillStyle=onGoal?'#3f9e52':'#8a5a2b';
  roundRect(x+2,y+2,cell-4,cell-4,4); ctx.fill();
  ctx.shadowBlur=0;
  ctx.strokeStyle=onGoal?'#a7ffc4':'#c98f4d'; ctx.lineWidth=2;
  ctx.strokeRect(x+3.5,y+3.5,cell-7,cell-7);
  ctx.beginPath();
  ctx.moveTo(x+3.5,y+3.5); ctx.lineTo(x+cell-3.5,y+cell-3.5);
  ctx.moveTo(x+cell-3.5,y+3.5); ctx.lineTo(x+3.5,y+cell-3.5);
  ctx.stroke();
  if(onGoal){
    ctx.fillStyle='#eaffea'; ctx.font='bold '+(cell*0.4)+'px monospace';
    ctx.textAlign='center'; ctx.textBaseline='middle';
    ctx.fillText('\u2713',x+cell/2,y+cell/2+1);
  }
  ctx.restore();
}

function draw(){
  const w=cv.width, h=cv.height;
  ctx.fillStyle='#0b0e1a'; ctx.fillRect(0,0,w,h);
  if(!board) return;

  // board bg + grid
  ctx.fillStyle='#141a30'; ctx.fillRect(ox-2,oy-2,cell*W+4,cell*H+4);
  ctx.strokeStyle='#232c4d'; ctx.lineWidth=1;
  for(let c=0;c<=W;c++){ ctx.beginPath(); ctx.moveTo(ox+c*cell+.5,oy); ctx.lineTo(ox+c*cell+.5,oy+cell*H); ctx.stroke(); }
  for(let r=0;r<=H;r++){ ctx.beginPath(); ctx.moveTo(ox,oy+r*cell+.5); ctx.lineTo(ox+cell*W,oy+r*cell+.5); ctx.stroke(); }

  // goals
  for(const g of goals){
    const [c,r]=g.split(',').map(Number);
    const x=ox+c*cell, y=oy+r*cell;
    ctx.save();
    ctx.translate(x+cell/2,y+cell/2); ctx.rotate(Math.PI/4);
    ctx.strokeStyle='#3fd8ff'; ctx.lineWidth=2.5;
    ctx.shadowColor='#3fd8ff'; ctx.shadowBlur=10;
    const s=cell*0.26;
    ctx.strokeRect(-s,-s,s*2,s*2);
    ctx.restore();
  }

  // board cells
  for(let r=0;r<H;r++)for(let c=0;c<W;c++) if(board[r][c]) drawCell(c,r,board[r][c]);

  // crates
  crates.forEach(drawCrate);

  // ghost + current piece
  if(cur && state==='play'){
    const ghost={m:cur.m,c:cur.c,r:cur.r};
    const gCrates=crates.map(k=>({c:k.c,r:k.r}));
    while(tryMove(ghost,0,1,gCrates)){}
    for(const [c,r] of cellsOf(ghost)) if(r>=0 && r!==cur.r) drawCell(c,r,cur.col,0.35,true);
    for(const [c,r] of cellsOf(cur)) if(r>=0) drawCell(c,r,cur.col);
  }

  // HUD
  ctx.fillStyle='#dfe6ff'; ctx.font='bold '+Math.max(12,cell*0.42)+'px monospace';
  ctx.textAlign='left'; ctx.textBaseline='top';
  ctx.fillText('SCORE '+score, ox, oy-cell*0.95);
  ctx.textAlign='right';
  ctx.fillStyle='#ffd75e';
  ctx.fillText('LEVEL '+level, ox+cell*W, oy-cell*0.95);

  // flash text
  if(flashTime>0){
    ctx.save();
    ctx.globalAlpha=Math.min(1,flashTime*2);
    ctx.fillStyle='#ffd75e';
    ctx.font='bold '+Math.max(16,cell*0.7)+'px monospace';
    ctx.textAlign='center'; ctx.textBaseline='middle';
    ctx.shadowColor='#ffd75e'; ctx.shadowBlur=18;
    ctx.fillText(flashText, ox+cell*W/2, oy+cell*H/2);
    ctx.restore();
  }

  if(paused){
    ctx.fillStyle='rgba(10,13,28,.6)'; ctx.fillRect(ox,oy,cell*W,cell*H);
    ctx.fillStyle='#fff'; ctx.font='bold '+Math.max(18,cell*0.8)+'px monospace';
    ctx.textAlign='center'; ctx.textBaseline='middle';
    ctx.fillText('PAUSED', ox+cell*W/2, oy+cell*H/2);
  }
}

/* ---------- main loop ---------- */
let last=performance.now();
function loop(now){
  const dt=Math.min(100, now-last); last=now;
  if(state==='play' && !paused){
    gravTimer+=dt;
    const interval=softDrop?55:Math.max(200,520-(level-1)*45);
    while(gravTimer>=interval && state==='play'){
      gravTimer-=interval;
      if(!tryMove(cur,0,1)) lock();
    }
  }
  if(state==='levelup'){
    flashTime-=dt/1000;
    if(flashTime<=0){ level++; loadLevel(level-1); spawn(); state='play'; gravTimer=0; }
  } else if(flashTime>0 && state==='play'){
    flashTime-=dt/1000;
  }
  draw();
  requestAnimationFrame(loop);
}

// initial empty board so background renders before start
board=emptyBoard(); crates=[]; goals=new Set(); cur=null; level=1; score=0;
requestAnimationFrame(loop);
})();
