function tone(f0,f1,dur,type,vol,delay){
  const ac=audio(); if(!ac) return;
  const t=ac.currentTime+(delay||0);
  const o=ac.createOscillator(), g=ac.createGain();

function aabb(px,py,pw,ph,r){ return px<r.x+r.w && px+pw>r.x && py<r.y+r.h && py+ph>r.y; }

function collideAxis(axis){
  for(const s of SOLIDS){
    if(aabb(player.x,player.y,player.w,player.h,s)){
      if(axis==='x'){
        if(player.vx>0) player.x=s.x-player.w; else if(player.vx<0) player.x=s.x+s.w;
        player.vx=0;

        if(player.vy>0){ player.y=s.y-player.h; if(!player.onGround){player.squash=1; sfxLand();} player.vy=0; player.onGround=true; }
        else if(player.vy<0){ player.y=s.y+s.h; player.vy=0; }
      }
    }
  }
}

// sight blocked by solids? sample segment, ignore near endpoints (feet-grazing)
function lineBlocked(x1,y1,x2,y2){
  const N=9;
  for(let i=1;i<N;i++){
    const t=i/N; if(t<0.18||t>0.85) continue;
    const sx=x1+(x2-x1)*t, sy=y1+(y2-y1)*t;
    for(const s of SOLIDS){
      if(sx>s.x+2 && sx<s.x+s.w-2 && sy>s.y+2 && sy<s.y+s.h-2) return true;
    }
  }
  return false;
}

let last=performance.now();
function update(dt){
  if(state!=='play') return;
  timer+=dt;

  // input
  const L=keys['ArrowLeft']||keys['KeyA'], R=keys['ArrowRight']||keys['KeyD'];
  const J=keys['ArrowUp']||keys['KeyW']||keys['Space'];
  if(L){ player.vx-=MOVE*dt; player.face=-1; }
  if(R){ player.vx+=MOVE*dt; player.face=1; }
  if(!L&&!R){
    if(player.vx>0) player.vx=Math.max(0,player.vx-FRICTION*dt);
    else if(player.vx<0) player.vx=Math.min(0,player.vx+FRICTION*dt);
  }
  player.vx=Math.max(-MAXVX,Math.min(MAXVX,player.vx));
  if(J&&player.onGround){ player.vy=-JUMPV; player.onGround=false; sfxJump(); }

  player.vy=Math.min(950,player.vy+GRAV*dt);
  player.x+=player.vx*dt; collideAxis('x');
  player.y+=player.vy*dt;
  player.onGround=false; collideAxis('y');
  if(player.x<0){player.x=0;player.vx=0;} if(player.x>W-player.w){player.x=W-player.w;player.vx=0;}
  player.squash=Math.max(0,player.squash-dt*5);

  // guards patrol
  for(const g of guards){
    g.x+=g.dir*g.speed*dt;
    if(g.x<g.minX){g.x=g.minX;g.dir=1;}
    if(g.x>g.maxX){g.x=g.maxX;g.dir=-1;}
  }

  // detection
  const pcx=player.x+player.w/2, pcy=player.y+player.h/2;
  let detected=false;
  for(let i=0;i<guards.length;i++){
    const g=guards[i];
    const ex=g.x, ey=g.y-40;
    const dx=pcx-ex, dy=pcy-ey;
    const dist=Math.hypot(dx,dy);
    if(dist>g.range) continue;
    let sees=false;
    const nx=dx/dist, ny=dy/dist;
    const facingDot=nx*g.dir + 0; // guard faces horizontally in g.dir
    if(facingDot>Math.cos(g.half)) sees=true;                     // inside cone
    else if(dist<70) sees=true;                                   // too close even from behind
    if(sees && lineBlocked(ex,ey,pcx,pcy)) sees=false;
    if(sees){
      detected=true; detectedGuard=i;
      const rate = dist<70 ? 0.55 : (1.2 + 2.6*(1-dist/g.range));
      meter=Math.min(1,meter+rate*dt);
    }
  }
  if(!detected){ meter=Math.max(0,meter-1.4*dt); detectedGuard=-1; }
  // milestone warning beeps
  const m=meter*100;
  if(m>=25&&m<25.6) sfxWarn();
  if(m>=50&&m<50.6) sfxWarn();
  if(m>=75&&m<75.6) sfxWarn();
  if(meter>=1){ state='caught'; sfxCaught(); }

  // portal
  const pd=Math.hypot(pcx-PORTAL.x,pcy-PORTAL.y);
  if(pd<58){ winTime=timer; state='won'; sfxWin(); }
}

  ctx.beginPath(); ctx.arc(0,-32-sq*2,8.5,0,7); ctx.fill();
  // visor glow
  ctx.fillStyle = meter>0.6 ? '#ff4433' : '#5df2ff';
  ctx.shadowColor=ctx.fillStyle; ctx.shadowBlur=10;
  ctx.fillRect(player.face>0?0:-8, -34-sq*2, 8, 3);
  ctx.shadowBlur=0;
  // scarf flutter
  ctx.strokeStyle='#5df2ff'; ctx.lineWidth=2.5; ctx.beginPath();
  ctx.moveTo(-player.face*6,-28-sq*2);
  ctx.quadraticCurveTo(-player.face*14,-30+Math.sin(now/100)*3, -player.face*20,-26+Math.sin(now/90)*4);
  ctx.stroke();

  // darkness vignette for stealth feel
  const vg=ctx.createRadialGradient(pcx(),pcy(),120,pcx(),pcy(),700);
  vg.addColorStop(0,'rgba(0,0,10,0)'); vg.addColorStop(1,'rgba(0,0,10,0.55)');
  ctx.fillStyle=vg; ctx.fillRect(0,0,W,H);

  // ---------- HUD ----------
  if(state!=='menu'){
    ctx.textAlign='left'; ctx.font='20px monospace';
    ctx.fillStyle='#cfd8ff';
    ctx.fillText('TIME  '+timer.toFixed(1)+'s', 24, 36);
    // detection meter
    const mw=300, mx=W/2-mw/2, my=22;
    ctx.textAlign='center';
    ctx.fillStyle='#8fa0d8'; ctx.font='14px monospace';
    ctx.fillText('DETECTION', W/2, my-2);
    ctx.fillStyle='#1a2040'; ctx.fillRect(mx,my+6,mw,16);
    const mcol = meter<0.5 ? '#ffd84d' : meter<0.8 ? '#ff9040' : '#ff3b2f';
    ctx.fillStyle=mcol; ctx.shadowColor=mcol; ctx.shadowBlur=meter>0?14:0;
    ctx.fillRect(mx+2,my+8,(mw-4)*meter,12);
    ctx.shadowBlur=0;
    ctx.strokeStyle='#59639b'; ctx.strokeRect(mx+0.5,my+6.5,mw-1,15);
  }

  // ---------- overlays ----------
  function overlay(alpha){ ctx.fillStyle='rgba(4,6,16,'+alpha+')'; ctx.fillRect(0,0,W,H); }
  ctx.textAlign='center';
  if(state==='menu'){
    overlay(0.72);
    ctx.fillStyle='#5df2ff'; ctx.shadowColor='#5df2ff'; ctx.shadowBlur=24;
    ctx.font='bold 84px monospace'; ctx.fillText('SHADOW SHIFT', W/2, 300);
    ctx.shadowBlur=0;
    ctx.fillStyle='#cfd8ff'; ctx.font='24px monospace';
    ctx.fillText('A physics stealth game — reach the portal unseen.', W/2, 375);
    ctx.font='20px monospace'; ctx.fillStyle='#9fb0e8';
    ctx.fillText('← → / A D   move      ↑ / W / SPACE   jump', W/2, 460);
    ctx.fillText('R   restart', W/2, 495);
    ctx.fillText('Stay out of the guards\u2019 light cones. Walls block their sight.', W/2, 545);
    const blink=Math.sin(now/350)>0;
    if(blink){ ctx.fillStyle='#8fffe8'; ctx.font='bold 26px monospace';
      ctx.fillText('PRESS  ENTER  TO  START', W/2, 640); }
  }
  if(state==='caught'){
    overlay(0.7);
    ctx.fillStyle='#ff4b35'; ctx.shadowColor='#ff4b35'; ctx.shadowBlur=26;
    ctx.font='bold 80px monospace'; ctx.fillText('DETECTED!', W/2, 400);
    ctx.shadowBlur=0;
    ctx.fillStyle='#cfd8ff'; ctx.font='22px monospace';
    ctx.fillText('You were spotted at '+timer.toFixed(1)+'s.', W/2, 470);
    if(Math.sin(now/350)>0){ ctx.fillStyle='#8fffe8'; ctx.font='bold 26px monospace';
      ctx.fillText('PRESS  R  TO  RETRY', W/2, 560); }
  }
  if(state==='won'){
    overlay(0.7);
    ctx.fillStyle='#8fffe8'; ctx.shadowColor='#8fffe8'; ctx.shadowBlur=26;
    ctx.font='bold 80px monospace'; ctx.fillText('ESCAPED!', W/2, 400);
    ctx.shadowBlur=0;
    ctx.fillStyle='#cfd8ff'; ctx.font='24px monospace';
    ctx.fillText('You vanished into the night in '+winTime.toFixed(1)+' seconds.', W/2, 470);
    if(Math.sin(now/350)>0){ ctx.fillStyle='#5df2ff'; ctx.font='bold 26px monospace';
      ctx.fillText('PRESS  R  TO  PLAY  AGAIN', W/2, 560); }
  }
}

function pcx(){ return player.x+player.w/2; }
function pcy(){ return player.y+player.h/2; }

  const dt=Math.min(0.033,(now-last)/1000); last=now;

  draw(now);
  requestAnimationFrame(loop);
}
