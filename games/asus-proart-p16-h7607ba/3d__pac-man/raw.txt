  if(Math.hypot(pac.x-(c+0.5), pac.y-(r+0.5))<0.55){
    delete dots[k];
    if(d===1){ score+=50; fright=6; combo=0; sfx.pellet(); }
    else { score+=10; sfx.dot(); }
  }
}

function update(dt){
  tGlobal+=dt;
  if(state==='dying'){
    dieT+=dt;
    if(dieT>1.6){
      lives--;
      if(lives>0){ resetActors(); fright=0; state='playing'; }
      else { saveHigh(); endOverlay('GAME OVER','PRESS R TO RESTART'); }
    }
    return;
  }
  if(state!=='playing') return;

  readInput();
  // direction logic (sliding handled by collisions)
  if(pac.wx||pac.wy){
    if(pac.wx===-pac.dirx && pac.wy===-pac.diry || (pac.dirx===0&&pac.diry===0))
      { pac.dirx=pac.wx; pac.diry=pac.wy; }
    else if(!(pac.wx===pac.dirx && pac.wy===pac.diry) &&
            ((pac.wy!==0&&pac.diry!==0)||(pac.wx!==0&&pac.dirx!==0)))
      { pac.dirx=pac.wx; pac.diry=pac.wy; }
  }
  var psp=4.1*mul();
  if(pac.dirx||pac.diry) moveEntity(pac, pac.dirx*psp*dt, pac.diry*psp*dt);
  pac.mouth = (pac.dirx||pac.diry) ? 0.35+0.28*Math.sin(tGlobal*12) : 0.12;

  eatAtPac();
  if(fright>0) fright-=dt;

  updateGhosts(dt);

  // collisions
  for(var i=0;i<ghosts.length;i++){
    var g=ghosts[i];
    if(g.invuln>0) continue;
    if(Math.hypot(g.x-pac.x,g.y-pac.y)<0.62){
      if(fright>0){
        score+=200*Math.pow(2,combo); combo++;
        g.x=g.homeX; g.y=g.homeY; g.invuln=1.5; sfx.ghost();
      } else {
        state='dying'; dieT=0; sfx.death(); return;
      }
    }
  }

  // win?
  if(Object.keys(dots).length===0){
    saveHigh(); sfx.levelup();
    $('overTitle').textContent='LEVEL '+level+' CLEAR!';
    $('overScore').textContent='SCORE '+score+'   •   HIGH '+high;
    $('overKey').textContent='PRESS ENTER FOR LEVEL '+(level+1);
    state='win';
    $('over').classList.remove('hidden');
  }
}

/* ============================== 3D RENDERER ============================== */
var cam={x:7.5,y:13.5,z:4.8}, fwd=[0,0,0], right=[0,0,0], upv=[0,0,0];
function nrm(v){ var l=Math.hypot(v[0],v[1],v[2])||1; return [v[0]/l,v[1]/l,v[2]/l]; }
function cross(a,b){ return [a[1]*b[2]-a[2]*b[1], a[2]*b[0]-a[0]*b[2], a[0]*b[1]-a[1]*b[0]]; }
function dot3(a,b){ return a[0]*b[0]+a[1]*b[1]+a[2]*b[2]; }

function updateCam(dt){
  var tx=pac.x, ty=pac.y+7.2, tz=4.8;
  var k=Math.min(1,dt*5);
  cam.x+=(tx-cam.x)*k; cam.y+=(ty-cam.y)*k; cam.z+=(tz-cam.z)*k;
  fwd=nrm([pac.x-cam.x, pac.y-1.3-cam.y, -cam.z]);
  right=nrm(cross([0,0,1],fwd));
  upv=cross(fwd,right);
}
var FOCAL=1;
function proj(x,y,z){
  var vx=x-cam.x, vy=y-cam.y, vz=z-cam.z;
  var zc=dot3([vx,vy,vz],fwd);
  if(zc<0.2) return null;
  var s=FOCAL/zc;
  return { x: W/2 + dot3([vx,vy,vz],right)*s,
           y: H/2 - dot3([vx,vy,vz],upv)*s,
           z: zc, s: s };
}

var LIGHT=nrm([-0.35,0.3,0.9]);
function shade(col,b,fog){
  var r=col[0]*b, g=col[1]*b, bl=col[2]*b;
  if(fog>0){ r+=(5-r)*fog; g+=(6-g)*fog; bl+=(15-bl)*fog; }
  return 'rgb('+(r|0)+','+(g|0)+','+(bl|0)+')';
}

function render(){
  FOCAL = (H/2)/Math.tan(27.5*Math.PI/180);
  ctx.fillStyle='#05060f'; ctx.fillRect(0,0,W,H);
  // vignette glow center
  var grad=ctx.createRadialGradient(W/2,H*0.45,0,W/2,H*0.45,Math.max(W,H)*0.7);
  grad.addColorStop(0,'#0a0f26'); grad.addColorStop(1,'#04050c');
  ctx.fillStyle=grad; ctx.fillRect(0,0,W,H);

  var items=[];
  function pushFace(pts,col){ // pts: [[x,y,z]x4], col base rgb
    var P=[proj(pts[0][0],pts[0][1],pts[0][2]),proj(pts[1][0],pts[1][1],pts[1][2]),
           proj(pts[2][0],pts[2][1],pts[2][2]),proj(pts[3][0],pts[3][1],pts[3][2])];
    if(P.some(function(p){return !p;})) return;
    var e1=[pts[1][0]-pts[0][0],pts[1][1]-pts[0][1],pts[1][2]-pts[0][2]];
    var e2=[pts[3][0]-pts[0][0],pts[3][1]-pts[0][1],pts[3][2]-pts[0][2]];
    var n=nrm(cross(e1,e2));
    var b=0.35+0.65*Math.max(0,dot3(n,LIGHT));
    var zc=(P[0].z+P[1].z+P[2].z+P[3].z)/4;
    var fog=Math.max(0,Math.min(1,(zc-7)/15));
    var fill=shade(col,b,fog);
    items.push({z:zc, fn:function(){
      ctx.beginPath();
      ctx.moveTo(P[0].x,P[0].y);
      for(var i=1;i<4;i++) ctx.lineTo(P[i].x,P[i].y);
      ctx.closePath(); ctx.fillStyle=fill; ctx.strokeStyle=fill; ctx.lineWidth=1;
      ctx.fill(); ctx.stroke();
    }});
  }

  var pc=Math.floor(pac.x), pr=Math.floor(pac.y);
  // floor tiles
  for(var r=pr-9;r<=pr+9;r++)for(var c=pc-9;c<=pc+9;c++){
    if(isWall(c,r)) continue;
    pushFace([[c,r,0],[c+1,r,0],[c+1,r+1,0],[c,r+1,0]], (c+r)%2 ? [26,32,74] : [19,23,58]);
  }
  // walls
  var WC=[38,64,205];
  for(var r2=pr-9;r2<=pr+9;r2++)for(var c2=pc-9;c2<=pc+9;c2++){
    if(!isWall(c2,r2)) continue;
    var nearFloor = !isWall(c2+1,r2)||!isWall(c2-1,r2)||!isWall(c2,r2+1)||!isWall(c2,r2-1);
    if(!nearFloor) continue;
    pushFace([[c2,r2,WH],[c2+1,r2,WH],[c2+1,r2+1,WH],[c2,r2+1,WH]], WC); // top
    if(!isWall(c2,r2+1)) pushFace([[c2,r2+1,0],[c2+1,r2+1,0],[c2+1,r2+1,WH],[c2,r2+1,WH]], [30,52,168]);
    if(!isWall(c2,r2-1)) pushFace([[c2,r2,0],[c2+1,r2,0],[c2+1,r2,WH],[c2,r2,WH]], [46,78,232]);
    if(!isWall(c2+1,r2)) pushFace([[c2+1,r2,0],[c2+1,r2+1,0],[c2+1,r2+1,WH],[c2+1,r2,WH]], [42,70,215]);
    if(!isWall(c2-1,r2)) pushFace([[c2,r2,0],[c2,r2+1,0],[c2,r2+1,WH],[c2,r2,WH]], [34,58,186]);
  }

  // ---- sprites ----
  function pushSprite(z,fn){ items.push({z:z,fn:fn}); }

  // dots & pearls
  for(var key in dots){
    var idx=key.indexOf(','), dc=+key.slice(0,idx), dr=+key.slice(idx+1);
    if(Math.abs(dc-pc)>9||Math.abs(dr-pr)>9) continue;
    (function(dc,dr,type){
      var p=proj(dc+0.5,dr+0.5, type===1?0.24:0.12);
      if(!p) return;
      pushSprite(p.z,function(){
        var col=[255,184,255];
        var rad = (type===1 ? 0.2+0.05*Math.sin(tGlobal*7) : 0.09)*p.s;
        ctx.globalAlpha=0.35; ctx.fillStyle='rgb('+col[0]+','+col[1]+','+col[2]+')';
        ctx.beginPath(); ctx.arc(p.x,p.y,rad*1.9,0,7); ctx.fill();
        ctx.globalAlpha=1;
        ctx.fillStyle='#ffd6ff';
        ctx.beginPath(); ctx.arc(p.x,p.y,rad,0,7); ctx.fill();
      });
    })(dc,dr,dots[key]);
  }

  // pac-man
  (function(){
    var zc=0.38;
    var p=proj(pac.x,pac.y,zc);
    if(!p) return;
    pushSprite(p.z,function(){
      var r=(state==='dying'? Math.max(2,0.4*p.s*(1-dieT/1.6)) : 0.4*p.s);
      var fp=proj(pac.x+pac.dirx*0.8,pac.y+pac.diry*0.8,zc);
      var ang = fp? Math.atan2(fp.y-p.y,fp.x-p.x) : -Math.PI/2;
      var mouth = state==='dying' ? Math.min(Math.PI,dieT*4) : pac.mouth;
      var g=ctx.createRadialGradient(p.x-r*0.35,p.y-r*0.4,r*0.1,p.x,p.y,r);
      g.addColorStop(0,'#fff7ae'); g.addColorStop(0.55,'#ffe14d'); g.addColorStop(1,'#e6a800');
      ctx.fillStyle=g;
      ctx.beginPath(); ctx.moveTo(p.x,p.y);
      ctx.arc(p.x,p.y,r, ang+mouth/2, ang-mouth/2+Math.PI*2);
      ctx.closePath(); ctx.fill();
      if(state!=='dying'){
        // eyes
        var ex=Math.cos(ang)*r*0.15, ey=Math.sin(ang)*r*0.15;
        ctx.fillStyle='#fff';
        ctx.beginPath(); ctx.arc(p.x-r*0.32+ex,p.y-r*0.34+ey,r*0.13,0,7); ctx.fill();
        ctx.beginPath(); ctx.arc(p.x+r*0.32+ex,p.y-r*0.34+ey,r*0.13,0,7); ctx.fill();
        ctx.fillStyle='#111';
        ctx.beginPath(); ctx.arc(p.x-r*0.32+ex*1.8,p.y-r*0.34+ey*1.8,r*0.06,0,7); ctx.fill();
        ctx.beginPath(); ctx.arc(p.x+r*0.32+ex*1.8,p.y-r*0.34+ey*1.8,r*0.06,0,7); ctx.fill();
      }
    });
  })();

  // ghosts
  for(var i=0;i<ghosts.length;i++)(function(g){
    var p=proj(g.x,g.y,0.42);
    if(!p||g.invuln>0&&state==='playing') { if(p) pushSprite(p.z,function(){
        // respawn blink
      }); }
    pushSprite(p? p.z:999,function(){ drawGhost(g,p); });
  })(ghosts[i]);

  items.sort(function(a,b){return b.z-a.z;});
  for(var n=0;n<items.length;n++) items[n].fn();
}

function drawGhost(g,p){
  if(!p) return;
  var r=0.42*p.s, cx=p.x, cy=p.y;
  var frighten = fright>0 && state==='playing';
  var flash = frighten && (fright<2 && Math.floor(tGlobal*8)%2===0);
  var col = frighten ? (flash? [255,255,255] : [46,72,230]) : g.col;
  var grd=ctx.createRadialGradient(cx-r*0.3,cy-r*0.5,r*0.1,cx,cy,r*1.2);
  grd.addColorStop(0,'rgba(255,255,255,.85)');
  grd.addColorStop(0.35,shade(col,1,0));
  grd.addColorStop(1,shade(col,0.55,0));
  ctx.fillStyle=grd;
  ctx.beginPath();
  ctx.arc(cx,cy-r*0.12,r,Math.PI,0);
  ctx.lineTo(cx+r,cy+r*0.9);
  var n=3,w=2*r/n;
  for(var i=0;i<n;i++){
    ctx.lineTo(cx+r-(i+0.5)*w, cy+r*0.38);
    ctx.lineTo(cx+r-(i+1)*w,   cy+r*0.9);
  }
  ctx.closePath(); ctx.fill();

  if(frighten){
    ctx.fillStyle= flash? '#c22' : '#ffd6d6';
    ctx.beginPath(); ctx.arc(cx-r*0.35,cy-r*0.2,r*0.14,0,7); ctx.fill();
    ctx.beginPath(); ctx.arc(cx+r*0.35,cy-r*0.2,r*0.14,0,7); ctx.fill();
    ctx.strokeStyle= flash? '#c22' : '#ffd6d6'; ctx.lineWidth=Math.max(1,r*0.09);
    ctx.beginPath();
    for(var m=0;m<=6;m++){ var mx=cx-r*0.5+m*r*0.166;
      ctx.lineTo(mx, cy+r*0.28 + (m%2? -r*0.1 : r*0.1)); }
    ctx.stroke();
  } else {
    var ex=g.dirx*r*0.14, ey=g.diry*r*0.14;
    ctx.fillStyle='#fff';
    ctx.beginPath(); ctx.ellipse(cx-r*0.38+ex,cy-r*0.28+ey,r*0.26,r*0.32,0,0,7); ctx.fill();
    ctx.beginPath(); ctx.ellipse(cx+r*0.38+ex,cy-r*0.28+ey,r*0.26,r*0.32,0,0,7); ctx.fill();
    ctx.fillStyle='#1b2f9e';
    ctx.beginPath(); ctx.arc(cx-r*0.38+ex*1.9,cy-r*0.24+ey*1.9,r*0.13,0,7); ctx.fill();
    ctx.beginPath(); ctx.arc(cx+r*0.38+ex*1.9,cy-r*0.24+ey*1.9,r*0.13,0,7); ctx.fill();
  }
}

  scEl.textContent=score;
  lvEl.textContent=level;
  if(lives!==lastHudLives){ lastHudLives=lives;
    liEl.textContent='●'.repeat(Math.max(0,lives)); }
}

var last=performance.now();
function loop(now){
  var dt=Math.min(0.05,(now-last)/1000); last=now;
  update(dt);
  updateCam(dt);
  render();
  hud();
  requestAnimationFrame(loop);
}
