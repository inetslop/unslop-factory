// ---------- physics ----------
function collideSeg(s, fl){
  var dx=s.x2-s.x1, dy=s.y2-s.y1;
  var len2 = dx*dx+dy*dy;
  var t = ((ball.x-s.x1)*dx + (ball.y-s.y1)*dy)/len2;
  if(t<0) t=0; else if(t>1) t=1;
  var cx = s.x1+t*dx, cy = s.y1+t*dy;
  var rx = ball.x-cx, ry = ball.y-cy;
  var d2 = rx*rx+ry*ry;
  var minD = BALL_R + (fl?FLIP_RAD:0);
  if(d2 >= minD*minD || d2===0) return false;
  var d = Math.sqrt(d2), nx=rx/d, ny=ry/d;
  // push out
  ball.x = cx+nx*minD; ball.y = cy+ny*minD;
  // flipper surface velocity
  var fvX=0, fvY=0;
  if(fl){
    var rxx=cx-fl.px, ryy=cy-fl.py;
    fvX = -fl.om*ryy; fvY = fl.om*rxx;
  }
  var rvx = ball.vx-fvX, rvy = ball.vy-fvY;
  var vn = rvx*nx+rvy*ny;
  if(vn < 0){
    var e = s.e!==undefined ? s.e : (fl?0.35:0.5);
    rvx -= (1+e)*vn*nx; rvy -= (1+e)*vn*ny;
    // tangential friction
    var tx=-ny, ty=nx, vt=rvx*tx+rvy*ty;
    rvx -= 0.06*vt*tx; rvy -= 0.06*vt*ty;
    ball.vx = rvx+fvX; ball.vy = rvy+fvY;
    if(s.imp){ ball.vx += nx*s.imp; ball.vy += ny*s.imp; }
    return true;
  }
  return false;
}

function step(dt){
  // flippers
  var lt = keys.left ? leftFlip.act : leftFlip.rest;
  var rt = keys.right ? rightFlip.act : rightFlip.rest;
  var na = moveToward(leftFlip.angle, lt, FLIP_SPEED*dt);
  leftFlip.om = (na-leftFlip.angle)/dt; leftFlip.angle=na;
  na = moveToward(rightFlip.angle, rt, FLIP_SPEED*dt);
  rightFlip.om = (na-rightFlip.angle)/dt; rightFlip.angle=na;

  ball.vy += GRAV*dt;
  ball.x += ball.vx*dt;
  ball.y += ball.vy*dt;

  // speed clamp
  var sp = Math.hypot(ball.vx,ball.vy);
  if(sp>1300){ ball.vx*=1300/sp; ball.vy*=1300/sp; }

  for(var i=0;i<walls.length;i++) collideSeg(walls[i], null);

  // flippers as capsules
  var lx = leftFlip.px + Math.cos(leftFlip.angle)*FLIP_LEN;
  var ly = leftFlip.py + Math.sin(leftFlip.angle)*FLIP_LEN;
  if(collideSeg({x1:leftFlip.px,y1:leftFlip.py,x2:lx,y2:ly,e:0.35}, leftFlip)) sfx(100,0.06,'square',0.08);
  var rx = rightFlip.px + Math.cos(rightFlip.angle)*FLIP_LEN;
  var ry = rightFlip.py + Math.sin(rightFlip.angle)*FLIP_LEN;
  if(collideSeg({x1:rightFlip.px,y1:rightFlip.py,x2:rx,y2:ry,e:0.35}, rightFlip)) sfx(100,0.06,'square',0.08);

  // bumpers
  for(var j=0;j<bumpers.length;j++){
    var b=bumpers[j];
    var ddx=ball.x-b.x, ddy=ball.y-b.y;
    var dd=Math.hypot(ddx,ddy), minD=b.r+BALL_R;
    if(dd<minD && dd>0){
      var nx2=ddx/dd, ny2=ddy/dd;

      ball.vy = ny2*560 + (Math.random()-0.5)*60;

    sfx(160,0.35,'sawtooth',0.15,60);
  }
}

  ctx.fillStyle=done?'#2f9e63':color;
  ctx.font='bold 18px monospace';
  ctx.textAlign='center'; ctx.textBaseline='middle';
  ctx.fillText(done?'✓':label, x, y);
  ctx.restore();
}

function render(){
  var W=canvas.width, H=canvas.height;
  var s = Math.min(W/WORLD.w, H/WORLD.h);
  var ox=(W-WORLD.w*s)/2 - WORLD.x*s;
  var oy=(H-WORLD.h*s)/2 - WORLD.y*s;

  ctx.setTransform(1,0,0,1,0,0);
  ctx.fillStyle='#05060f';
  ctx.fillRect(0,0,W,H);
  ctx.setTransform(s,0,0,s,ox,oy);

  // table background
  ctx.fillStyle='#0b1024';
  ctx.beginPath();
  ctx.moveTo(10,470); ctx.lineTo(95,530); ctx.lineTo(95,585);
  ctx.lineTo(265,585); ctx.lineTo(265,530); ctx.lineTo(350,470);
  ctx.lineTo(350,40); 
  // arc back
  for(var i=1;i<=12;i++){
    var t=i/12, a=(1-t)*(1-t), b2=2*(1-t)*t, c=t*t;
    ctx.lineTo(a*10+b2*180+c*350, a*40+b2*(-90)+c*40);
  }
  ctx.closePath(); ctx.fill();

  // decorative lanes
  ctx.strokeStyle='rgba(80,120,255,0.08)';
  ctx.lineWidth=2;
  for(var k=-3;k<=3;k++){
    ctx.beginPath(); ctx.moveTo(180+k*45,-20); ctx.lineTo(180+k*60,560); ctx.stroke();
  }

  // walls
  ctx.lineCap='round';
  ctx.strokeStyle='#3b4a8f';
  ctx.lineWidth=5;
  ctx.beginPath();
  for(var w2=0;w2<walls.length;w2++){
    var ww=walls[w2];
    ctx.moveTo(ww.x1,ww.y1); ctx.lineTo(ww.x2,ww.y2);
  }
  ctx.stroke();

  // slingshot highlight
  ctx.strokeStyle='#ff5c8a';
  ctx.lineWidth=6;
  ctx.beginPath(); ctx.moveTo(28,456); ctx.lineTo(74,498); ctx.stroke();
  ctx.beginPath(); ctx.moveTo(332,456); ctx.lineTo(286,498); ctx.stroke();

  // bumpers
  for(var j=0;j<bumpers.length;j++){
    var bp=bumpers[j];
    if(bp.flash>0) bp.flash=Math.max(0,bp.flash-0.08);
    ctx.save();
    if(bp.flash>0){ ctx.shadowColor='#ffd75e'; ctx.shadowBlur=25*bp.flash; }
    ctx.fillStyle = bp.flash>0 ? '#ffd75e' : '#1b2a55';
    ctx.beginPath(); ctx.arc(bp.x,bp.y,bp.r,0,7); ctx.fill();
    ctx.strokeStyle='#ff9d3c'; ctx.lineWidth=3;
    ctx.beginPath(); ctx.arc(bp.x,bp.y,bp.r,0,7); ctx.stroke();
    ctx.fillStyle='#ffb85c';
    ctx.beginPath(); ctx.arc(bp.x,bp.y,bp.r*0.4,0,7); ctx.fill();
    ctx.restore();
  }

  // checkpoints
  var now=performance.now();
  for(var i=0;i<checkpoints.length;i++){
    var cp=checkpoints[i];
    if(i<nextCP) drawRing(cp.x,cp.y,cp.r,'','','#', true);
    else if(i===nextCP && state==='playing') drawRing(cp.x,cp.y,cp.r,cp.label,'#ffd75e',true,false);
    else if(i===nextCP) drawRing(cp.x,cp.y,cp.r,cp.label,'#ffd75e',false,false);
    else drawRing(cp.x,cp.y,cp.r,cp.label,'rgba(140,150,200,0.6)',false,false);
  }
  // finish
  if(nextCP>=checkpoints.length && state==='playing'){
    ctx.save();
    ctx.shadowColor='#39ff8a'; ctx.shadowBlur=20+10*Math.sin(now/140);
    ctx.strokeStyle='#39ff8a'; ctx.lineWidth=5;
    ctx.beginPath(); ctx.arc(finish.x,finish.y,finish.r,0,7); ctx.stroke();
    ctx.fillStyle='#39ff8a'; ctx.font='bold 12px monospace';
    ctx.textAlign='center'; ctx.textBaseline='middle';
    ctx.fillText('FINISH', finish.x, finish.y);
    ctx.restore();
  } else {
    ctx.strokeStyle='rgba(60,255,140,0.3)'; ctx.lineWidth=4;
    ctx.beginPath(); ctx.arc(finish.x,finish.y,finish.r,0,7); ctx.stroke();
  }

  // flippers
  drawFlipper(leftFlip, keys.left?'#ffd75e':'#ff9d3c');
  drawFlipper(rightFlip, keys.right?'#ffd75e':'#ff9d3c');

  // trail
  for(var t2=0;t2<trail.length;t2++){
    var p=trail[t2], a=(t2+1)/trail.length;
    ctx.fillStyle='rgba(90,200,255,'+(a*0.35)+')';
    ctx.beginPath(); ctx.arc(p.x,p.y,BALL_R*a,0,7); ctx.fill();
  }

  // ball
  if(state!=='menu'){
    ctx.save();
    ctx.shadowColor='#5ad0ff'; ctx.shadowBlur=14;
    var grad=ctx.createRadialGradient(ball.x-3,ball.y-3,1,ball.x,ball.y,BALL_R);
    grad.addColorStop(0,'#eaffff'); grad.addColorStop(1,'#2f9fd8');
    ctx.fillStyle=grad;
    ctx.beginPath(); ctx.arc(ball.x,ball.y,BALL_R,0,7); ctx.fill();
    ctx.restore();
  }

  // HUD
  ctx.textAlign='left'; ctx.textBaseline='alphabetic';
  ctx.font='bold 13px monospace';
  ctx.fillStyle='#8fa4ff';
  ctx.fillText('TIME '+fmt(state==='finished'?time:time), 20, -30);
  if(best!==null){ ctx.fillStyle='#ffd75e'; ctx.fillText('BEST '+(state==='finished'&&best===time?'★':'')+fmt(best), 120, -30); }
  ctx.fillStyle='#8fa4ff';
  ctx.fillText('CP '+Math.min(nextCP+ (state==='finished'?1:0), checkpoints.length)+'/'+checkpoints.length, 240, -30);
  ctx.fillStyle='#ff5c8a';
  ctx.fillText('DRAIN '+drains, 305, -30);

  // message
  if(msgT>0){
    msgT-=1/60;
    ctx.save();
    ctx.globalAlpha=Math.min(1,msgT*2);
    ctx.font='bold 22px monospace'; ctx.textAlign='center';
    ctx.fillStyle='#ffd75e';
    ctx.shadowColor='#ffd75e'; ctx.shadowBlur=12;
    ctx.fillText(lastMsg, 180, 300);
    ctx.restore();
  }

  // overlays
  if(state==='menu'){
    overlay(0.78);
    ctx.textAlign='center';
    ctx.fillStyle='#5ad0ff';
    ctx.font='bold 44px monospace';
    ctx.fillText('PINBALL RACER', 180, 210);
    ctx.font='16px monospace'; ctx.fillStyle='#cfd8ff';
    ctx.fillText('Collect checkpoints 1→5 in order,', 180, 270);
    ctx.fillText('then hit the FINISH ring. Beat the clock!', 180, 294);
    ctx.font='15px monospace'; ctx.fillStyle='#ffd75e';
    ctx.fillText('◄ / A ... left flipper', 180, 350);
    ctx.fillText('► / D ... right flipper', 180, 374);
    ctx.fillText('R ... restart round', 180, 398);
    if(best!==null){ ctx.fillStyle='#39ff8a'; ctx.fillText('BEST TIME: '+fmt(best), 180, 440); }
    ctx.fillStyle='rgba(255,255,255,'+(0.6+0.4*Math.sin(now/300))+')';
    ctx.font='bold 18px monospace';
    ctx.fillText('PRESS ENTER TO START', 180, 495);
  }

  if(state==='finished'){
    overlay(0.72);
    ctx.textAlign='center';
    ctx.fillStyle='#39ff8a';
    ctx.font='bold 40px monospace';
    ctx.fillText('FINISH!', 180, 220);
    ctx.fillStyle='#eaffff'; ctx.font='bold 26px monospace';
    ctx.fillText(fmt(time), 180, 275);
    if(best!==null){
      ctx.fillStyle = (best===time)?'#ffd75e':'#cfd8ff';
      ctx.font='18px monospace';
      ctx.fillText((best===time?'★ NEW BEST: ':'BEST: ')+fmt(best), 180, 315);
    }
    ctx.fillStyle='#ff9d3c'; ctx.font='15px monospace';
    ctx.fillText('Drains: '+drains, 180, 350);
    ctx.fillStyle='rgba(255,255,255,'+(0.6+0.4*Math.sin(now/300))+')';
    ctx.font='bold 18px monospace';
    ctx.fillText('PRESS ENTER TO RACE AGAIN', 180, 420);
  }
}

function overlay(a){
  ctx.fillStyle='rgba(4,6,15,'+a+')';
  ctx.fillRect(WORLD.x,WORLD.y,WORLD.w,WORLD.h);
}

  else if(k===' '||k==='Enter'){
    e.preventDefault();
    if(state==='menu' || state==='finished') startGame();
  }
  else if(k==='r'||k==='R'){
    if(state!=='menu') startGame();
  }
});

// ---------- main loop ----------
var last = performance.now(), acc=0;
var FIXED = 1/60, SUB=4;
function loop(now){
  var dt = Math.min(0.05,(now-last)/1000);
  last = now;
  if(state==='playing'){
    time += dt;
    acc += dt;
    while(acc >= FIXED){
      for(var i=0;i<SUB;i++) step(FIXED/SUB);
      checkCP();
      trail.push({x:ball.x,y:ball.y});
      if(trail.length>22) trail.shift();
      acc -= FIXED;
    }
  } else {
    acc=0;
  }
  render();
  requestAnimationFrame(loop);
}
requestAnimationFrame(loop);
})();
