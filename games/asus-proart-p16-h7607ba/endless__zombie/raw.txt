(function(){
"use strict";
const canvas=document.getElementById('game');
const ctx=canvas.getContext('2d');
let W=0,H=0;
function resize(){W=canvas.width=window.innerWidth;H=canvas.height=window.innerHeight;}

// ---------- Audio (created after first input) ----------
let AC=null;
function audioInit(){if(!AC){try{AC=new (window.AudioContext||window.webkitAudioContext)();}catch(e){}}if(AC&&AC.state==='suspended')AC.resume();}
function beep(type,f0,f1,dur,vol){
  if(!AC)return;
  const t=AC.currentTime,o=AC.createOscillator(),g=AC.createGain();
  o.type=type;o.frequency.setValueAtTime(f0,t);
  o.frequency.exponentialRampToValueAtTime(Math.max(1,f1),t+dur);
  g.gain.setValueAtTime(vol,t);g.gain.exponentialRampToValueAtTime(0.0001,t+dur);
  o.connect(g);g.connect(AC.destination);o.start(t);o.stop(t+dur);
}
function noiseBurst(dur,vol,freq){
  if(!AC)return;
  const t=AC.currentTime,len=Math.floor(AC.sampleRate*dur),buf=AC.createBuffer(1,len,AC.sampleRate),d=buf.getChannelData(0);
  for(let i=0;i<len;i++)d[i]=(Math.random()*2-1)*(1-i/len);
  const s=AC.createBufferSource();s.buffer=buf;
  const f=AC.createBiquadFilter();f.type='lowpass';f.frequency.value=freq;
  const g=AC.createGain();g.gain.value=vol;
  s.connect(f);f.connect(g);g.connect(AC.destination);s.start(t);
}
const sShoot =()=>{beep('square',880,180,0.09,0.06);noiseBurst(0.05,0.05,3000);};
const sZDie  =()=>{beep('sawtooth',140,40,0.25,0.12);noiseBurst(0.18,0.1,600);};
const sHurt  =()=>{beep('triangle',220,70,0.2,0.2);};
const sHit   =()=>{beep('square',300,150,0.05,0.04);};

// ---------- State ----------
const STATE={MENU:0,PLAY:1,OVER:2};
let state=STATE.MENU;
let player,bullets,zombies,particles,floaters,time,score,kills,spawnTimer,shake,shootCd,muzzle;
const keys={};
const mouse={x:0,y:0,down:false};

function reset(){
  player={x:W/2,y:H/2,r:14,hp:100,maxHp:100,speed:270};
  bullets=[];zombies=[];particles=[];floaters=[];
  time=0;score=0;kills=0;spawnTimer=0.5;shake=0;shootCd=0;muzzle=0;
}

function startGame(){
  audioInit();
  reset();
  state=STATE.PLAY;
  document.getElementById('startScreen').classList.add('hidden');
  document.getElementById('overScreen').classList.add('hidden');
}

function gameOver(){
  state=STATE.OVER;
  beep('sawtooth',200,30,0.8,0.25);noiseBurst(0.6,0.15,400);
  let hi=0;
  try{hi=parseInt(localStorage.getItem('zombieEndlessHigh')||'0',10)||0;}catch(e){}
  if(score>hi){hi=score;try{localStorage.setItem('zombieEndlessHigh',String(hi));}catch(e){}}
  document.getElementById('finalScore').textContent='Score: '+score;
  document.getElementById('finalStats').innerHTML=
    'Survived <b>'+time.toFixed(1)+'s</b> &nbsp;·&nbsp; Zombies killed: <b>'+kills+'</b>';
  document.getElementById('highScore').textContent='★ Best score: '+hi;
  document.getElementById('overScreen').classList.remove('hidden');
}

// ---------- Input ----------
window.addEventListener('keydown',e=>{
  audioInit();
  keys[e.code]=true;
  if(e.code==='Space')e.preventDefault();
  if((e.code==='KeyR'||e.code==='Enter')&&state!==STATE.MENU)startGame();
  if(state===STATE.MENU&&(e.code==='Enter'||e.code==='Space'))startGame();
});
window.addEventListener('keyup',e=>{keys[e.code]=false;});
canvas.addEventListener('mousemove',e=>{const r=canvas.getBoundingClientRect();mouse.x=e.clientX-r.left;mouse.y=e.clientY-r.top;});
canvas.addEventListener('mousedown',e=>{audioInit();mouse.down=true;if(state===STATE.MENU)startGame();});
window.addEventListener('mouseup',()=>{mouse.down=false;});
canvas.addEventListener('contextmenu',e=>e.preventDefault());
document.getElementById('startBtn').addEventListener('click',startGame);
document.getElementById('restartBtn').addEventListener('click',startGame);

// ---------- Spawning ----------
function spawnZombie(){
  const side=Math.floor(Math.random()*4);
  let x,y;
  if(side===0){x=Math.random()*W;y=-30;}
  else if(side===1){x=W+30;y=Math.random()*H;}
  else if(side===2){x=Math.random()*W;y=H+30;}
  else{x=-30;y=Math.random()*H;}
  const t=time;
  let type='walker';
  const roll=Math.random();
  if(t>15&&roll<0.22)type='runner';
  if(t>35&&roll>0.85)type='brute';
  const spdScale=1+Math.min(1,t*0.012);
  let z;
  if(type==='walker')z={x,y,r:14,hp:30,maxHp:30,spd:(60+Math.random()*35)*spdScale,dmg:12,col:'#5a8f3c',score:10};
  else if(type==='runner')z={x,y,r:11,hp:16,maxHp:16,spd:(135+Math.random()*45)*spdScale,dmg:8,col:'#9dbb4a',score:15};
  else z={x,y,r:22,hp:130,maxHp:130,spd:(42+Math.random()*15)*spdScale,dmg:25,col:'#3f6e2e',score:40};
  z.hitCd=0;z.wob=Math.random()*Math.PI*2;
  zombies.push(z);
}

function blood(x,y,n,col){
  for(let i=0;i<n;i++){
    const a=Math.random()*Math.PI*2,s=40+Math.random()*180;
    particles.push({x,y,vx:Math.cos(a)*s,vy:Math.sin(a)*s,life:0.3+Math.random()*0.4,maxLife:0.7,col,r:1.5+Math.random()*2.5});
  }
}
function floater(x,y,txt,col){floaters.push({x,y,txt,col,life:0.9});}

// ---------- Update ----------
function update(dt){
  time+=dt;score+=Math.floor(dt*10); // survival points tick handled below instead
  score-=Math.floor(dt*10);          // (kept simple: score from kills + time bonus on death)
  const t=time;

  // player movement
  let mx=0,my=0;
  if(keys['KeyA']||keys['ArrowLeft'])mx--;
  if(keys['KeyD']||keys['ArrowRight'])mx++;
  if(keys['KeyW']||keys['ArrowUp'])my--;
  if(keys['KeyS']||keys['ArrowDown'])my++;
  if(mx||my){const l=Math.hypot(mx,my);player.x+=mx/l*player.speed*dt;player.y+=my/l*player.speed*dt;}
  player.x=Math.max(player.r,Math.min(W-player.r,player.x));
  player.y=Math.max(player.r,Math.min(H-player.r,player.y));

  // shooting
  shootCd-=dt;muzzle-=dt;
  if((mouse.down||keys['Space'])&&shootCd<=0){
    shootCd=0.13;
    const a=Math.atan2(mouse.y-player.y,mouse.x-player.x);
    bullets.push({x:player.x+Math.cos(a)*18,y:player.y+Math.sin(a)*18,vx:Math.cos(a)*750,vy:Math.sin(a)*750,life:1.4});
    muzzle=0.06;sShoot();
  }

  // bullets
  for(let i=bullets.length-1;i>=0;i--){
    const b=bullets[i];
    b.x+=b.vx*dt;b.y+=b.vy*dt;b.life-=dt;
    if(b.life<=0||b.x<-20||b.x>W+20||b.y<-20||b.y>H+20){bullets.splice(i,1);continue;}
    for(let j=zombies.length-1;j>=0;j--){
      const z=zombies[j];
      if(Math.hypot(b.x-z.x,b.y-z.y)<z.r+4){
        z.hp-=25;blood(b.x,b.y,6,'#8b1a1a');sHit();
        bullets.splice(i,1);
        if(z.hp<=0){
          kills++;score+=z.score;
          blood(z.x,z.y,14,'#7a1515');
          floater(z.x,z.y-10,'+'+z.score,'#ffd75e');
          zombies.splice(j,1);sZDie();shake=Math.max(shake,3);
        }
        break;
      }
    }
  }

  // spawning
  spawnTimer-=dt;
  const interval=Math.max(0.22,1.15-t*0.016);
  if(spawnTimer<=0){spawnTimer=interval*(0.7+Math.random()*0.6);spawnZombie();
    if(t>40&&Math.random()<0.35)spawnZombie();}

  // zombies
  for(let i=0;i<zombies.length;i++){
    const z=zombies[i];
    z.wob+=dt*8;z.hitCd-=dt;
    let dx=player.x-z.x,dy=player.y-z.y,d=Math.hypot(dx,dy)||1;
    z.x+=dx/d*z.spd*dt;z.y+=dy/d*z.spd*dt;
    // separation
    for(let j=i+1;j<zombies.length;j++){
      const o=zombies[j],sx=o.x-z.x,sy=o.y-z.y,sd=Math.hypot(sx,sy);
      const min=z.r+o.r;
      if(sd>0&&sd<min){const push=(min-sd)*0.5,px=sx/sd*push,py=sy/sd*push;z.x-=px;z.y-=py;o.x+=px;o.y+=py;}
    }
    // attack
    if(d<z.r+player.r&&z.hitCd<=0){
      z.hitCd=0.7;
      player.hp-=z.dmg;shake=Math.max(shake,6);sHurt();
      blood(player.x,player.y,8,'#c22');
      floater(player.x,player.y-18,'-'+z.dmg,'#ff5544');
      if(player.hp<=0){player.hp=0;blood(player.x,player.y,30,'#a01818');shake=14;gameOver();}
    }
  }

  // particles & floaters
  for(let i=particles.length-1;i>=0;i--){const p=particles[i];p.life-=dt;if(p.life<=0){particles.splice(i,1);continue;}p.x+=p.vx*dt;p.y+=p.vy*dt;p.vx*=0.92;p.vy*=0.92;}
  for(let i=floaters.length-1;i>=0;i--){const f=floaters[i];f.life-=dt;f.y-=30*dt;if(f.life<=0)floaters.splice(i,1);}

  shake=Math.max(0,shake-dt*20);
}

// ---------- Draw ----------
function draw(){
  ctx.save();
  if(shake>0)ctx.translate((Math.random()-0.5)*shake,(Math.random()-0.5)*shake);

  // ground
  ctx.fillStyle='#141a12';ctx.fillRect(-20,-20,W+40,H+40);
  ctx.strokeStyle='rgba(60,80,50,0.15)';ctx.lineWidth=1;
  const gs=64;
  for(let x=(W/2)%gs;x<W;x+=gs){ctx.beginPath();ctx.moveTo(x,0);ctx.lineTo(x,H);ctx.stroke();}
  for(let y=(H/2)%gs;y<H;y+=gs){ctx.beginPath();ctx.moveTo(0,y);ctx.lineTo(W,y);ctx.stroke();}

  // particles under
  for(const p of particles){ctx.globalAlpha=Math.max(0,p.life/p.maxLife);ctx.fillStyle=p.col;ctx.beginPath();ctx.arc(p.x,p.y,p.r,0,7);ctx.fill();}
  ctx.globalAlpha=1;

  // zombies
  for(const z of zombies){
    const w=Math.sin(z.wob)*2;
    ctx.save();ctx.translate(z.x,z.y+w*0.3);
    ctx.fillStyle='rgba(0,0,0,0.3)';ctx.beginPath();ctx.ellipse(0,z.r*0.85,z.r*0.9,z.r*0.4,0,0,7);ctx.fill();
    ctx.fillStyle=z.col;ctx.beginPath();ctx.arc(0,0,z.r,0,7);ctx.fill();
    ctx.strokeStyle='rgba(0,0,0,0.4)';ctx.lineWidth=2;ctx.stroke();
    // arms toward player
    const a=Math.atan2(player.y-z.y,player.x-z.x);
    ctx.strokeStyle=z.col;ctx.lineWidth=z.r*0.35;ctx.lineCap='round';
    ctx.beginPath();ctx.moveTo(Math.cos(a+0.6)*z.r*0.6,Math.sin(a+0.6)*z.r*0.6);
    ctx.lineTo(Math.cos(a+0.4)*(z.r+8+w),Math.sin(a+0.4)*(z.r+8+w));ctx.stroke();
    ctx.beginPath();ctx.moveTo(Math.cos(a-0.6)*z.r*0.6,Math.sin(a-0.6)*z.r*0.6);
    ctx.lineTo(Math.cos(a-0.4)*(z.r+8-w),Math.sin(a-0.4)*(z.r+8-w));ctx.stroke();
    // eyes
    ctx.fillStyle='#ff3b2f';
    ctx.beginPath();ctx.arc(Math.cos(a)*z.r*0.45-Math.sin(a)*z.r*0.3,Math.sin(a)*z.r*0.45+Math.cos(a)*z.r*0.3,z.r*0.16,0,7);ctx.fill();

    // hp bar if hurt
    if(z.hp<z.maxHp){
      const bw=z.r*2;
      ctx.fillStyle='rgba(0,0,0,0.6)';ctx.fillRect(-bw/2,-z.r-9,bw,4);
      ctx.fillStyle='#7CFC00';ctx.fillRect(-bw/2,-z.r-9,bw*Math.max(0,z.hp/z.maxHp),4);
    }
    ctx.restore();
  }

  // bullets
  ctx.fillStyle='#ffe98a';
  for(const b of bullets){
    ctx.beginPath();ctx.arc(b.x,b.y,3.5,0,7);ctx.fill();
    ctx.globalAlpha=0.4;ctx.beginPath();ctx.moveTo(b.x,b.y);ctx.lineTo(b.x-b.vx*0.02,b.y-b.vy*0.02);ctx.strokeStyle='#ffe98a';ctx.lineWidth=3;ctx.stroke();ctx.globalAlpha=1;
  }

  // player
  if(state!==STATE.OVER){
    const a=Math.atan2(mouse.y-player.y,mouse.x-player.x);
    ctx.save();ctx.translate(player.x,player.y);
    ctx.fillStyle='rgba(0,0,0,0.35)';ctx.beginPath();ctx.ellipse(0,player.r*0.85,player.r*0.9,player.r*0.4,0,0,7);ctx.fill();
    // gun
    ctx.strokeStyle='#2b2b2b';ctx.lineWidth=6;ctx.lineCap='round';
    ctx.beginPath();ctx.moveTo(Math.cos(a)*6,Math.sin(a)*6);ctx.lineTo(Math.cos(a)*(player.r+12),Math.sin(a)*(player.r+12));ctx.stroke();
    // body
    ctx.fillStyle=muzzle>0?'#d8ffe0':'#4a7fd4';ctx.beginPath();ctx.arc(0,0,player.r,0,7);ctx.fill();
    ctx.strokeStyle='#cfe8ff';ctx.lineWidth=2;ctx.stroke();
    ctx.fillStyle='#1b2a45';ctx.beginPath();ctx.arc(Math.cos(a)*4,Math.sin(a)*4,player.r*0.45,0,7);ctx.fill();
    // muzzle flash
    if(muzzle>0){
      ctx.fillStyle='rgba(255,230,120,0.9)';
      ctx.beginPath();ctx.arc(Math.cos(a)*(player.r+16),Math.sin(a)*(player.r+16),7,0,7);ctx.fill();
    }
    ctx.restore();
  }

  // floaters
  ctx.font='bold 14px Courier New';ctx.textAlign='center';
  for(const f of floaters){ctx.globalAlpha=Math.min(1,f.life*2);ctx.fillStyle=f.col;ctx.fillText(f.txt,f.x,f.y);}
  ctx.globalAlpha=1;

  // vignette
  const vg=ctx.createRadialGradient(W/2,H/2,Math.min(W,H)*0.35,W/2,H/2,Math.max(W,H)*0.75);
  vg.addColorStop(0,'rgba(0,0,0,0)');vg.addColorStop(1,'rgba(0,0,0,0.55)');
  ctx.fillStyle=vg;ctx.fillRect(-20,-20,W+40,H+40);

  // low hp pulse
  if(state===STATE.PLAY&&player.hp<30){
    ctx.fillStyle='rgba(160,20,20,'+(0.12+0.1*Math.sin(time*8))+')';
    ctx.fillRect(-20,-20,W+40,H+40);
  }

  // HUD
  if(state===STATE.PLAY||state===STATE.OVER){
    ctx.textAlign='left';
    ctx.fillStyle='rgba(0,0,0,0.5)';ctx.fillRect(14,14,210,46);
    ctx.fillStyle='#333';ctx.fillRect(22,22,194,14);
    ctx.fillStyle=player.hp>30?'#7CFC00':'#ff5544';
    ctx.fillRect(22,22,194*Math.max(0,player.hp/player.maxHp),14);
    ctx.strokeStyle='#888';ctx.strokeRect(22,22,194,14);
    ctx.fillStyle='#fff';ctx.font='bold 13px Courier New';
    ctx.fillText('HP '+Math.ceil(player.hp)+'/100',26,34);
    ctx.fillStyle='#cfe8cf';ctx.font='12px Courier New';
    ctx.fillText('ZOMBIES: '+zombies.length,26,54);

    ctx.textAlign='right';ctx.fillStyle='#fff';ctx.font='bold 20px Courier New';
    ctx.fillText('SCORE '+(score+kills*0+Math.floor(time)*5),W-16,34);
    ctx.fillStyle='#9abf9a';ctx.font='14px Courier New';
    ctx.fillText('TIME '+time.toFixed(1)+'s   KILLS '+kills,W-16,54);
  }
  ctx.restore();
}

// ---------- Loop ----------
let last=performance.now();
function loop(now){
  let dt=(now-last)/1000;last=now;
  if(dt>0.05)dt=0.05;
  if(state===STATE.PLAY)update(dt);
  draw();
  requestAnimationFrame(loop);
}
reset();
requestAnimationFrame(loop);
