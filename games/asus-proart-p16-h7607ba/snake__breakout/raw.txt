function sampleHistory(age){
  // find position `age` seconds ago
  var t=gameTime-age;
  for(var i=history.length-1;i>=0;i--){
    if(history[i].t<=t){
      if(i===0)return history[0];
      var a=history[i],b=history[i+1];
      var f=(t-a.t)/Math.max(0.0001,(b.t-a.t));
      return {x:a.x+(b.x-a.x)*f,y:a.y};
    }
  }
  return history[0]||{x:headX,y:headY};
}

function getSegments(){
  var s=[{x:headX,y:headY,r:HEAD_R}];
  for(var i=1;i<SEGS;i++){
    var p=sampleHistory(i*SEG_DELAY);
    s.push({x:p.x,y:headY,r:SEG_R});
  }
  return s;
}

    var st=dt/steps;

        var nx=dx/d,ny=dy/d;
        var dot=ball.vx*nx+ball.vy*ny;
        if(dot<0){
          ball.vx-=2*dot*nx;
          ball.vy-=2*dot*ny;
          // re-normalize speed
          var m=Math.sqrt(ball.vx*ball.vx+ball.vy*ball.vy)||1;
          ball.vx=ball.vx/m*ball.speed;
          ball.vy=ball.vy/m*ball.speed;
        }
        ball.x=sg.x+nx*(minD+0.5);
        ball.y=sg.y+ny*(minD+0.5);
        beep(330,0.06,'triangle',0.12,80);
        break;
      }
    }

    // bricks

      if(!br.alive)continue;

  }

function roundRect(x,y,w,h,r){
  ctx.beginPath();

  ctx.arcTo(x,y+h,x,y,r);
  ctx.arcTo(x,y,x+w,y,r);
  ctx.closePath();
}
