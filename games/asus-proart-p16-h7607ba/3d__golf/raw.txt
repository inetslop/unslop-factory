function lerpAng(a,b,t){var d=(b-a+Math.PI*3)%(Math.PI*2)-Math.PI;return a+d*t;}
function project(wx,wy,wz){
  var dx=wx-camX,dy=wy-camY,dz=wz-camZ;
  var cz=dx*fxv+dz*fzv;
  if(cz<0.3)return null;
  var cx=dx*rxv+dz*rzv;
  return {x:W/2+cx*FOV/cz, y:H/2-dy*FOV/cz, z:cz};
}

function seg(x1,y1,z1,x2,y2,z2,color,w){
  var a=project(x1,y1,z1), b=project(x2,y2,z2);
  if(!a||!b)return;
  ctx.strokeStyle=color;ctx.lineWidth=w||1;
  ctx.beginPath();ctx.moveTo(a.x,a.y);ctx.lineTo(b.x,b.y);ctx.stroke();
}
function groundCircle(x,z,r,color){
  var pts=[];
  for(var i=0;i<=24;i++){var a=i/24*Math.PI*2,p=project(x+Math.cos(a)*r,0,z+Math.sin(a)*r);if(!p)return;pts.push(p);}
  ctx.fillStyle=color;ctx.beginPath();ctx.moveTo(pts[0].x,pts[0].y);
  for(i=1;i<pts.length;i++)ctx.lineTo(pts[i].x,pts[i].y);
  ctx.closePath();ctx.fill();
}

function draw(){
  // sky
  var far=project(ball.x+fxv*600,0,ball.z+fzv*600);
  var hy=far?far.y:H*0.4;
  var sky=ctx.createLinearGradient(0,0,0,hy);
  sky.addColorStop(0,'#7ec8f5');sky.addColorStop(1,'#d8effc');
  ctx.fillStyle=sky;ctx.fillRect(0,0,W,Math.max(0,hy));
  var gnd=ctx.createLinearGradient(0,hy,0,H);
  gnd.addColorStop(0,'#4f9e3f');gnd.addColorStop(1,'#2f7a2c');
  ctx.fillStyle=gnd;ctx.fillRect(0,Math.max(0,hy),W,H-Math.max(0,hy));
  // sun
  ctx.fillStyle='rgba(255,244,200,.9)';ctx.beginPath();ctx.arc(W*0.78,Math.max(30,hy-60),26,0,7);ctx.fill();

  // grid
  var step=10;
  for(var i=-8;i<=8;i++){
    var ox=ball.x+rxv*i*step, oz=ball.z+rzv*i*step;
    for(var t=-250;t<320;t+=step){
      seg(ox+fxv*t,0,oz+fzv*t, ox+fxv*(t+step),0,oz+fzv*(t+step),'rgba(255,255,255,.10)',1);
    }
  }
  for(var j=-30;j<=32;j++){
    var wx=ball.x+fxv*j*step, wz=ball.z+fzv*j*step;
    for(i=-8;i<8;i++){
      seg(wx+rxv*i*step,0,wz+rzv*i*step, wx+rxv*(i+1)*step,0,wz+rzv*(i+1)*step,'rgba(255,255,255,.14)',1);
    }
    if(j>0&&j%2===0){
      var mp=project(wx,0.05,wz);
      if(mp&&mp.z<90){ctx.fillStyle='rgba(255,255,255,.5)';ctx.font=Math.max(10,mp.z*0.4)+'px sans-serif';ctx.textAlign='center';
        ctx.fillText((j*step- Math.round(Math.hypot(wx-holes[0].x,wz-holes[0].z))===0?'':'')+'', mp.x,mp.y); // placeholder removed below
      }
    }
  }

  // traps
  for(var ti=0;ti<traps.length;ti++){
    var tr=traps[ti];
    groundCircle(tr.x,tr.z,tr.r, tr.type==='sand'?'#e6d28a':'#4aa3e0');
    groundCircle(tr.x,tr.z,tr.r*0.7, tr.type==='sand'?'#ddc574':'#3b8fc9');
  }

  // hole + flag
  var h=holes[0];
  groundCircle(h.x,h.z,1.1,'#1a1a1a');
  var base=project(h.x,0,h.z), top=project(h.x,5.2,h.z);
  if(base&&top){
    ctx.strokeStyle='#f5f5f5';ctx.lineWidth=Math.max(1.5,top.z*0.35);
    ctx.beginPath();ctx.moveTo(base.x,base.y);ctx.lineTo(top.x,top.y);ctx.stroke();
    var fw=project(h.x+rxv*1.4,0,h.z+rzv*1.4);
    if(fw){var fs=top.z*0.5;
      ctx.fillStyle='#e33';ctx.beginPath();
      ctx.moveTo(top.x,top.y);ctx.lineTo(top.x,top.y+fs*0.8);ctx.lineTo(top.x+Math.sign(rxv||1)*fs*1.6, top.y+fs*0.4);
      ctx.closePath();ctx.fill();}
  }

  // aim line when at rest
  if(atRest&&state==='play'){
    for(var a=0;a<7;a++){
      seg(ball.x+fxv*(a*1.2),0.05,ball.z+fzv*(a*1.2), ball.x+fxv*(a*1.2+0.9),0.05,ball.z+fzv*(a*1.2+0.9),'rgba(255,255,80,'+(0.7-a*0.09)+')',3);
    }
  }

  // ball shadow + ball
  var sh=project(ball.x,0.02,ball.z);
  if(sh){ctx.fillStyle='rgba(0,0,0,.3)';ctx.beginPath();
    ctx.ellipse(sh.x,sh.y,Math.max(1,0.4*FOV/sh.z),Math.max(0.5,0.16*FOV/sh.z),0,0,7);ctx.fill();}
  var bp=project(ball.x,ball.y+0.32,ball.z);
  if(bp){var br=Math.max(2,0.34*FOV/bp.z);
    var bg=ctx.createRadialGradient(bp.x-br*0.35,bp.y-br*0.35,br*0.15,bp.x,bp.y,br);
    bg.addColorStop(0,'#ffffff');bg.addColorStop(1,'#c9c9c9');
    ctx.fillStyle=bg;ctx.beginPath();ctx.arc(bp.x,bp.y,br,0,7);ctx.fill();}

  // ---------- HUD ----------
  ctx.textAlign='left';
  ctx.font='bold '+Math.max(16,H*0.03)+'px Segoe UI, Arial';
  ctx.fillStyle='rgba(0,0,0,.45)';
  roundRect(12,12,Math.max(230,W*0.3),H*0.085,10);ctx.fill();
  ctx.fillStyle='#fff';
  ctx.fillText('Hole '+(holeIndex+1)+'/'+HOLE_COUNT+'   Par '+par,26,H*0.045);
  ctx.fillText('Strokes: '+strokes+'     Total: '+totalStrokes,26,H*0.045+H*0.038);

  // power bar
  if(charging){
    var bw=Math.min(420,W*0.6), bx=(W-bw)/2, by=H-52;
    ctx.fillStyle='rgba(0,0,0,.5)';roundRect(bx-6,by-6,bw+12,34,8);ctx.fill();
    var pg=ctx.createLinearGradient(bx,0,bx+bw,0);
    pg.addColorStop(0,'#5be07a');pg.addColorStop(0.6,'#ffd94a');pg.addColorStop(1,'#ff4d4d');
    ctx.fillStyle=pg;roundRect(bx,by,bw*power,22,6);ctx.fill();
    ctx.strokeStyle='#fff';ctx.lineWidth=2;roundRect(bx,by,bw,22,6);ctx.stroke();
    ctx.fillStyle='#fff';ctx.textAlign='center';ctx.font='bold 14px Arial';
    ctx.fillText('POWER',W/2,by-12);
  }

  // message
  if(msgT>0){
    ctx.textAlign='center';
    var fs=Math.min(52,W*0.08);
    ctx.font='bold '+fs+'px Segoe UI, Arial';
    ctx.fillStyle='rgba(0,0,0,.55)';
    roundRect(W/2-ctx.measureText(message).width/2-24,H*0.3-40,ctx.measureText(message).width+48,fs+34,14);ctx.fill();
    ctx.fillStyle='#fff';ctx.fillText(message,W/2,H*0.3);
  }
  if(state==='holeDone'){
    ctx.textAlign='center';ctx.fillStyle='rgba(255,255,255,.9)';
    ctx.font='bold '+Math.max(16,H*0.028)+'px Arial';
    var txt=nextIsEnd?'Press ENTER to see your scorecard':'Press ENTER for next hole';
    ctx.fillText(txt,W/2,H*0.3+fs+34);
  }
}
function roundRect(x,y,w,h,r){ctx.beginPath();ctx.moveTo(x+r,y);ctx.arcTo(x+w,y,x+w,y+h,r);ctx.arcTo(x+w,y+h,x,y+h,r);ctx.arcTo(x,y+h,x,y,r);ctx.arcTo(x,y,x+w,y,r);ctx.closePath();}

// ---------- main loop ----------
var last=performance.now();
function frame(now){
  var dt=Math.min(0.05,(now-last)/1000); last=now;
  if(state==='play'||state==='holeDone'){
    if(state==='play'&&atRest){
      if(keys.left)aimYaw+=2.4*dt;
      if(keys.right)aimYaw-=2.4*dt;
      if(charging){chargeT+=dt;power=1-Math.abs(2*(chargeT%1)-1);} // oscillate 0..1
    }
    physStep(dt);
    updateCamera(dt);
    if(msgT>0)msgT-=dt;
    if(state==='holeDone'){holeDoneT+=dt;if(holeDoneT>4){ if(nextIsEnd)endRound(); else startHole(holeIndex+1);} }
  }
  draw();
  requestAnimationFrame(frame);
}
requestAnimationFrame(frame);

})();
