function collideSolid(p,r){

  if(Math.abs(relN)>180||(a.onGround&&b.onGround)){
    shake=Math.max(shake,6);sHit();winGlow=0.25;
    spawn((a.x+b.x)/2,(a.y+b.y)/2,14,['#ffffff','#ffe066',a.color,b.color],300,0.4);
  }
  return true;
}

/* ---------- Update ---------- */
function update(dt){
  if(state==='play'){
    stepPlayer(p1,CTRL1,dt,p2);
    if(state==='play')stepPlayer(p2,CTRL2,dt,p1);
    playerPush(p1,p2);playerPush(p1,p2);
  }else if(state==='ko'){
    koTimer-=dt;
    if(koTimer<=0){
      if(p1.score>=WIN_SCORE||p2.score>=WIN_SCORE){
        const w=p1.score>p2.score?1:2;
        document.getElementById('winTitle').innerHTML=w===1?'<span style="color:#4db8ff">PLAYER 1</span> WINS!':'<span style="color:#ff5d5d">PLAYER 2</span> WINS!';
        document.getElementById('winScore').textContent='Final score  '+p1.score+' — '+p2.score;
        winScreen.classList.remove('hidden');state='win';sWin();
      }else{resetPositions();state='play';}
    }
  }
  // particles
  for(let i=particles.length-1;i>=0;i--){
    const q=particles[i];q.life-=dt;
    if(q.life<=0){particles.splice(i,1);continue;}
    q.vy+=700*dt;q.x+=q.vx*dt;q.y+=q.vy*dt;
  }
  shake=Math.max(0,shake-dt*30);
  winGlow=Math.max(0,winGlow-dt);
}

/* ---------- Render ---------- */
function drawPlatform(pl){
  if(pl.solid){
    const g=ctx.createLinearGradient(0,pl.y,0,pl.y+pl.h);
    g.addColorStop(0,'#3a4470');g.addColorStop(1,'#20263f');
    ctx.fillStyle=g;
    roundRect(pl.x,pl.y,pl.w,pl.h,8);ctx.fill();
    ctx.fillStyle='#59d98c';roundRect(pl.x,pl.y,pl.w,6,4);ctx.fill();
  }else{
    ctx.fillStyle='#39406b';roundRect(pl.x,pl.y,pl.w,pl.h,6);ctx.fill();
    ctx.fillStyle='#7f8ac2';roundRect(pl.x,pl.y,pl.w,4,3);ctx.fill();
  }
}
function roundRect(x,y,w,h,r){
  r=Math.min(r,w/2,h/2);
  ctx.beginPath();
  ctx.moveTo(x+r,y);ctx.arcTo(x+w,y,x+w,y+h,r);ctx.arcTo(x+w,y+h,x,y+h,r);
  ctx.arcTo(x,y+h,x,y,r);ctx.arcTo(x,y,x+w,y,r);ctx.closePath();
}
function drawPlayer(p){
  const sq=p.landSquash*0.25;
  ctx.save();
  ctx.translate(p.x,p.y);
  // dash cooldown ring
  if(p.dashCd>0){
    ctx.strokeStyle='rgba(255,255,255,.35)';ctx.lineWidth=3;
    ctx.beginPath();ctx.arc(0,0,p.r+6,-Math.PI/2,-Math.PI/2+(1-p.dashCd/0.9)*Math.PI*2);ctx.stroke();
  }else{
    ctx.strokeStyle='rgba(255,255,255,.8)';ctx.lineWidth=2;
    ctx.beginPath();ctx.arc(0,0,p.r+6,0,Math.PI*2);ctx.stroke();
  }
  ctx.scale(1+sq*0.5,1-sq);
  const g=ctx.createRadialGradient(-p.r*0.3,-p.r*0.4,4,0,0,p.r*1.2);
  g.addColorStop(0,'#ffffff');g.addColorStop(0.25,p.color);g.addColorStop(1,p.dark);
  ctx.fillStyle=g;
  ctx.beginPath();ctx.arc(0,0,p.r,0,Math.PI*2);ctx.fill();
  // eyes
  const ex=p.facing*6;
  ctx.fillStyle='#fff';
  ctx.beginPath();ctx.arc(ex-6,-4,5.5,0,Math.PI*2);ctx.arc(ex+7,-4,5.5,0,Math.PI*2);ctx.fill();
  ctx.fillStyle='#141830';
  ctx.beginPath();ctx.arc(ex-6+p.facing*2,-4,2.6,0,Math.PI*2);ctx.arc(ex+7+p.facing*2,-4,2.6,0,Math.PI*2);ctx.fill();
  // mouth
  ctx.strokeStyle='#141830';ctx.lineWidth=2;
  ctx.beginPath();ctx.arc(ex+0.5,4,4,0.15*Math.PI,0.85*Math.PI);ctx.stroke();
  ctx.restore();
}
function render(){
  ctx.setTransform(dpr,0,0,dpr,0,0);
  // full-window backdrop
  const bg=ctx.createLinearGradient(0,0,0,innerHeight);
  bg.addColorStop(0,'#101433');bg.addColorStop(1,'#080a18');
  ctx.fillStyle=bg;ctx.fillRect(0,0,innerWidth,innerHeight);

  ctx.save();
  ctx.translate(offX+(Math.random()-0.5)*shake*scale,offY+(Math.random()-0.5)*shake*scale);
  ctx.scale(scale,scale);

  // arena backdrop
  const sky=ctx.createLinearGradient(0,0,0,H);
  sky.addColorStop(0,'#1a2150');sky.addColorStop(0.6,'#2b2a5e');sky.addColorStop(1,'#4a2340');
  ctx.fillStyle=sky;ctx.fillRect(0,0,W,H);
  // stars
  ctx.fillStyle='rgba(255,255,255,.5)';
  for(let i=0;i<40;i++){
    const sx=(i*197)%W,sy=(i*89)%260;
    ctx.globalAlpha=0.2+((i*37)%10)/14;
    ctx.fillRect(sx,sy,2,2);
  }
  ctx.globalAlpha=1;
  // moon
  ctx.fillStyle='#f4e9c8';ctx.beginPath();ctx.arc(820,90,34,0,Math.PI*2);ctx.fill();
  ctx.fillStyle='rgba(255,255,255,.08)';ctx.beginPath();ctx.arc(820,90,50,0,Math.PI*2);ctx.fill();
  // glowing pit below
  const pit=ctx.createLinearGradient(0,H-70,0,H+40);
  pit.addColorStop(0,'rgba(255,60,60,0)');pit.addColorStop(1,'rgba(255,80,60,.45)');
  ctx.fillStyle=pit;ctx.fillRect(0,H-70,W,120);

  platforms.forEach(drawPlatform);
  if(state!=='start'){drawPlayer(p1);drawPlayer(p2);}

  // particles
  for(const q of particles){
    ctx.globalAlpha=Math.max(0,q.life/q.max);
    ctx.fillStyle=q.c;ctx.fillRect(q.x-q.sz/2,q.y-q.sz/2,q.sz,q.sz);
  }
  ctx.globalAlpha=1;

  // HUD
  if(state!=='start'){
    ctx.textAlign='center';
    ctx.font='bold 34px "Segoe UI",Arial,sans-serif';
    ctx.fillStyle='#4db8ff';ctx.fillText(String(p1.score),W/2-60,52);
    ctx.fillStyle='#ff5d5d';ctx.fillText(String(p2.score),W/2+60,52);
    ctx.fillStyle='rgba(255,255,255,.7)';ctx.font='bold 20px Arial';
    ctx.fillText('—',W/2,48);
    ctx.font='13px Arial';ctx.fillStyle='rgba(255,255,255,.5)';
    ctx.fillText('FIRST TO '+WIN_SCORE,W/2,70);
    ctx.textAlign='left';ctx.font='bold 14px Arial';
    ctx.fillStyle='#4db8ff';ctx.fillText('P1',W/2-96,30);
    ctx.textAlign='right';ctx.fillStyle='#ff5d5d';ctx.fillText('P2',W/2+96,30);

    if(winGlow>0){
      ctx.globalAlpha=winGlow;ctx.fillStyle='#fff';ctx.fillRect(0,0,W,H);ctx.globalAlpha=1;
    }
  }
  if(state==='ko'){
    const t=1.4-koTimer;
    const s=Math.min(1,t*5)*(1+Math.sin(t*20)*0.03*(t<0.6?1:0));
    ctx.save();ctx.translate(W/2,H/2-20);ctx.scale(s,s);
    ctx.textAlign='center';
    ctx.font='bold 72px "Segoe UI",Arial,sans-serif';
    ctx.lineWidth=8;ctx.strokeStyle='#141830';ctx.strokeText('K.O.!',0,0);
    ctx.fillStyle='#ffe066';ctx.fillText('K.O.!',0,0);
    ctx.font='bold 26px Arial';
    ctx.strokeStyle='#141830';ctx.lineWidth=5;ctx.strokeText(koText,0,48);
    ctx.fillStyle='#fff';ctx.fillText(koText,0,48);
    ctx.restore();
  }
  ctx.restore();
}

/* ---------- Loop ---------- */
let last=performance.now();
function loop(now){
  let dt=Math.min(0.033,(now-last)/1000);last=now;
  if(state==='play'||state==='ko')update(dt);
  render();
  jp.clear();
  requestAnimationFrame(loop);
}
