    (lives>0?'<br>Bonus for lives: +'+(lives*250):''),
    'Press ENTER or click to play again');
  sWin();
}

}

}

/* ---------- physics ---------- */
function moveEntityX(e,dx){
  e.x+=dx;
  var c1=Math.floor(e.x/TS),c2=Math.floor((e.x+e.w-0.01)/TS),
      r1=Math.floor(e.y/TS),r2=Math.floor((e.y+e.h-0.01)/TS);
  for(var r=r1;r<=r2;r++)for(var c=c1;c<=c2;c++){
    if(solid(c,r)){
      if(dx>0)e.x=c*TS-e.w; else e.x=(c+1)*TS;
      return (dx>0?-1:1);
    }
  }
  return 0;
}
function moveEntityY(e,dy){
  e.y+=dy;
  var c1=Math.floor(e.x/TS),c2=Math.floor((e.x+e.w-0.01)/TS),
      r1=Math.floor(e.y/TS),r2=Math.floor((e.y+e.h-0.01)/TS);
  for(var r=r1;r<=r2;r++)for(var c=c1;c<=c2;c++){
    if(solid(c,r)){
      if(dy>0){e.y=r*TS-e.h;return 1;} else {e.y=(r+1)*TS;return -1;}
    }
  }
  return 0;
}
function aabb(a,b){return a.x<b.x+b.w&&a.x+a.w>b.x&&a.y<b.y+b.h&&a.y+a.h>b.y;}

function killPlayer(){
  if(state!=='playing')return;
  state='dying';stateTimer=0;shake=10;
  player.vy=-560;player.vx=(Math.random()<0.5?-1:1)*120;
  lives--;sDie();
}
function respawnOrEnd(){
  if(lives<=0){gameOver();return;}
  loadLevel(levelIdx);state='playing';
}

/* ---------- update ---------- */
function update(dt){
  time+=dt;
  if(shake>0)shake=Math.max(0,shake-dt*30);
  if(state==='paused'||state==='start')return;

  if(state==='dying'){
    stateTimer+=dt;
    player.vy=Math.min(player.vy+GRAV*dt,MAXFALL);
    player.x+=player.vx*dt;player.y+=player.vy*dt;
    if(stateTimer>1)respawnOrEnd();
    updateCam(dt);return;
  }
  if(state!=='playing')return;

  var p=player;
  var left=(keys.ArrowLeft||keys.KeyA),right=(keys.ArrowRight||keys.KeyD);
  var target=(right?MOVE:0)-(left?MOVE:0);
  p.vx+=(target-p.vx)*Math.min(1,dt*14);
  if(Math.abs(p.vx)<4)p.vx=0;
  if(target>0)p.facing=1;else if(target<0)p.facing=-1;

  p.vy=Math.min(p.vy+GRAV*dt,MAXFALL);
  p.coyote-=dt;p.jbuf-=dt;
  if(p.jbuf>0&&(p.onGround||p.coyote>0)){
    p.vy=JUMPV;p.jbuf=0;p.coyote=0;sJump();
  }

  var wasGround=p.onGround;
  moveEntityX(p,p.vx*dt);
  var vyHit=moveEntityY(p,p.vy*dt);
  p.onGround=(vyHit===1);
  if(!p.onGround&&wasGround)p.coyote=0.09;

  // fell into pit
  if(p.y>grid.length*TS+80){killPlayer();return;}

  // coins
  for(var i=0;i<coins.length;i++){
    var c=coins[i];
    if(!c.taken&&p.x<c.x+11&&p.x+p.w>c.x-11&&p.y<c.y+11&&p.y+p.h>c.y-11){
      c.taken=true;score+=100;sCoin();
    }
  }
  // spikes
  for(i=0;i<spikes.length;i++){
    var s=spikes[i];
    if(aabb(p,{x:s.c*TS+5,y:s.r*TS+14,w:22,h:18})){killPlayer();return;}
  }
  // enemies
  for(i=0;i<enemies.length;i++){
    var e=enemies[i];if(!e.alive)continue;
    var nx=e.x+e.dir*e.speed*dt;
    var fx=(e.dir>0)?nx+e.w:nx,fc=Math.floor(fx/TS);
    if(solid(fc,e.row)||!solid(fc,e.row+1))e.dir*=-1;else e.x=nx;
    if(aabb(p,e)){
      if(p.vy>140&&(p.y+p.h-e.y)<15){
        e.alive=false;score+=100;p.vy=-440;sStomp();shake=5;
      }else{killPlayer();return;}
    }
  }
  // exit
  if(exitTile&&aabb(p,{x:exitTile.c*TS+2,y:exitTile.r*TS+2,w:TS-4,h:TS-4})){
    score+=500;saveBest();
    state='clear';sClear();
    var last=(levelIdx+1>=LEVELS.length);
    showOverlay(last?'FINAL DOOR!':'LEVEL '+(levelIdx+1)+' CLEAR','+500 bonus',
      'Score: <b style="color:#ffd760">'+score+'</b> &nbsp;|&nbsp; Lives: '+lives,
      last?'Press ENTER to see results':'Press ENTER for level '+(levelIdx+2));
    if(last)state='clear-final';
  }

  updateCam(dt);
}
function updateCam(dt){
  var mw=grid[0].length*TS,mh=grid.length*TS;
  var tx=player.x+player.w/2-W/2,ty=player.y+player.h/2-H/2;
  if(mw<W)tx=(mw-W)/2;else tx=Math.max(0,Math.min(tx,mw-W));
  if(mh<H)ty=(mh-H)/2;else ty=Math.max(-40,Math.min(ty,mh-H));
  cam.x+=(tx-cam.x)*Math.min(1,dt*8);
  cam.y+=(ty-cam.y)*Math.min(1,dt*8);
}

/* ---------- render ---------- */
function rr(x,y,w,h,r){
  ctx.beginPath();
  ctx.moveTo(x+r,y);ctx.arcTo(x+w,y,x+w,y+h,r);ctx.arcTo(x+w,y+h,x,y+h,r);
  ctx.arcTo(x,y+h,x,y,r);ctx.arcTo(x,y,x+w,y,r);ctx.closePath();
}
function render(){
  // background
  var g=ctx.createLinearGradient(0,0,0,H);
  g.addColorStop(0,'#181b2e');g.addColorStop(0.6,'#232a47');g.addColorStop(1,'#101223');
  ctx.fillStyle=g;ctx.fillRect(0,0,W,H);
  // stars (parallax)
  ctx.fillStyle='rgba(255,255,255,0.35)';
  for(var i=0;i<70;i++){
    var sx=((i*173.3)-cam.x*0.25)%W;if(sx<0)sx+=W;
    var sy=((i*97.7)-cam.y*0.25)%H;if(sy<0)sy+=H;
    var tw=0.4+0.6*Math.abs(Math.sin(time*1.5+i));
    ctx.globalAlpha=0.25*tw;ctx.fillRect(sx,sy,2,2);
  }
  ctx.globalAlpha=1;

  if(state==='start'){return;} // overlay covers

  var ox=Math.round(-cam.x+(shake?(Math.random()-0.5)*shake:0)),
      oy=Math.round(-cam.y+(shake?(Math.random()-0.5)*shake:0));
  ctx.save();ctx.translate(ox,oy);

  var mw=grid[0].length*TS,mh=grid.length*TS;
  var c0=Math.max(0,Math.floor(-cam.x/TS)-1),c1=Math.min(grid[0].length-1,Math.ceil((W-cam.x)/TS)+1);
  var r0=Math.max(0,Math.floor(-cam.y/TS)-1),r1=Math.min(grid.length-1,Math.ceil((H-cam.y)/TS)+1);

  // tiles
  for(var r=r0;r<=r1;r++)for(var c=c0;c<=c1;c++){
    if(!solid(c,r))continue;
    var x=c*TS,y=r*TS;
    ctx.fillStyle='#46548f';ctx.fillRect(x,y,TS,TS);
    ctx.fillStyle='#39456f';ctx.fillRect(x,y+TS-6,TS,6);
    if(!solid(c,r-1)){ctx.fillStyle='#7286c9';ctx.fillRect(x,y,TS,5);}
    ctx.strokeStyle='rgba(0,0,0,0.25)';ctx.lineWidth=1;
    ctx.strokeRect(x+0.5,y+0.5,TS-1,TS-1);
  }

  // exit door
  if(exitTile){
    var ex=exitTile.c*TS,ey=exitTile.r*TS,pul=0.6+0.4*Math.sin(time*4);
    ctx.save();
    ctx.shadowColor='rgba(125,255,176,'+(0.8*pul)+')';ctx.shadowBlur=22;
    ctx.fillStyle='#1e3a2c';rr(ex+4,ey-14,TS-8,TS+14,6);ctx.fill();
    ctx.shadowBlur=0;
    ctx.fillStyle='hsl('+(140+25*Math.sin(time*3))+',90%,'+(55+15*pul)+'%)';
    rr(ex+7,ey-9,TS-14,TS+4,4);ctx.fill();
    ctx.fillStyle='#fffbe8';ctx.beginPath();ctx.arc(ex+TS-12,ey+3,2.5,0,7);ctx.fill();
    ctx.restore();
  }

  // spikes
  for(var i=0;i<spikes.length;i++){
    var s=spikes[i],sx=s.c*TS,sy=s.r*TS;
    if(sx<c0*TS-TS||sx>(c1+1)*TS)continue;
    ctx.fillStyle='#9aa7cc';
    for(var k=0;k<3;k++){
      ctx.beginPath();
      ctx.moveTo(sx+2+k*10,sy+TS);ctx.lineTo(sx+7+k*10,sy+13);ctx.lineTo(sx+12+k*10,sy+TS);
      ctx.closePath();ctx.fill();
    }
  }

  // coins
  for(i=0;i<coins.length;i++){
    var cn=coins[i];if(cn.taken)continue;
    var sc=Math.abs(Math.sin(time*4+i*1.3)),bob=Math.sin(time*3+i)*2;
    ctx.save();ctx.translate(cn.x,cn.y+bob);
    ctx.shadowColor='rgba(255,215,96,0.7)';ctx.shadowBlur=10;
    ctx.fillStyle='#ffd760';
    ctx.beginPath();ctx.ellipse(0,0,Math.max(2,8*sc),9,0,0,7);ctx.fill();
    ctx.shadowBlur=0;
    ctx.strokeStyle='#b3541e';ctx.lineWidth=2;
    ctx.beginPath();ctx.ellipse(0,0,Math.max(1.5,5*sc),6,0,0,7);ctx.stroke();
    ctx.restore();
  }

  // enemies
  for(i=0;i<enemies.length;i++){
    var e=enemies[i];if(!e.alive)continue;
    var wob=Math.sin(time*10+e.ph)*2;
    ctx.save();ctx.translate(e.x,e.y);
    ctx.fillStyle='#d94f6b';
    rr(1,4+wob*0.3,e.w,e.h-4,8);ctx.fill();
    ctx.fillStyle='#a83250';
    ctx.beginPath();ctx.ellipse(7,e.h-2+Math.abs(wob)*0.6,5,3,0,0,7);ctx.fill();
    ctx.beginPath();ctx.ellipse(e.w-7,e.h-2-Math.abs(wob)*0.6,5,3,0,0,7);ctx.fill();
    // eyes
    var look=e.dir*3;
    ctx.fillStyle='#fff';
    ctx.beginPath();ctx.arc(8+look,12,4.5,0,7);ctx.fill();
    ctx.beginPath();ctx.arc(e.w-8+look,12,4.5,0,7);ctx.fill();
    ctx.fillStyle='#222';
    ctx.beginPath();ctx.arc(8+look*1.6,12,2,0,7);ctx.fill();
    ctx.beginPath();ctx.arc(e.w-8+look*1.6,12,2,0,7);ctx.fill();
    ctx.strokeStyle='#5c1b30';ctx.lineWidth=2;
    ctx.beginPath();ctx.moveTo(4,7);ctx.lineTo(12,9);ctx.stroke();
    ctx.beginPath();ctx.moveTo(e.w-4,7);ctx.lineTo(e.w-12,9);ctx.stroke();
    ctx.restore();
  }

  // player
  var p=player;
  if(p){
    ctx.save();ctx.translate(p.x+p.w/2,p.y+p.h/2);
    if(state==='dying')ctx.rotate(stateTimer*9);
    ctx.translate(-p.w/2,-p.h/2);
    ctx.shadowColor='rgba(110,231,255,0.5)';ctx.shadowBlur=8;
    ctx.fillStyle='#6ee7ff';rr(0,0,p.w,p.h,7);ctx.fill();
    ctx.shadowBlur=0;
    ctx.fillStyle='#39b8d6';rr(0,p.h-8,p.w,8,5);ctx.fill();
    var lx=p.facing*4;
    ctx.fillStyle='#fff';
    ctx.beginPath();ctx.arc(p.w/2-5+lx,10,4.2,0,7);ctx.fill();
    ctx.beginPath();ctx.arc(p.w/2+5+lx,10,4.2,0,7);ctx.fill();
    ctx.fillStyle='#123';
    ctx.beginPath();ctx.arc(p.w/2-5+lx*1.8,10,2,0,7);ctx.fill();
    ctx.beginPath();ctx.arc(p.w/2+5+lx*1.8,10,2,0,7);ctx.fill();
    if(state==='dying'){ctx.strokeStyle='#123';ctx.lineWidth=2;
      ctx.beginPath();ctx.moveTo(6,17);ctx.lineTo(10,21);ctx.moveTo(10,17);ctx.lineTo(6,21);ctx.stroke();}
    else{ctx.strokeStyle='#123';ctx.lineWidth=2;
      ctx.beginPath();ctx.arc(p.w/2+lx*0.8,18,3,0.2,Math.PI-0.2);ctx.stroke();}
    ctx.restore();
  }
  ctx.restore();

  // HUD
  ctx.save();
  ctx.font='bold 20px "Segoe UI",Arial,sans-serif';
  ctx.textBaseline='top';
  ctx.fillStyle='rgba(10,12,24,0.55)';rr(10,10,238,74,10);ctx.fill();
  ctx.fillStyle='#ffd760';ctx.fillText('SCORE '+score,24,20);
  ctx.fillStyle='#cfd9ff';ctx.font='15px "Segoe UI",Arial,sans-serif';
  ctx.fillText('LEVEL '+(levelIdx+1)+' / '+LEVELS.length+'     BEST '+best,24,46);
  ctx.font='20px Arial';ctx.fillStyle='#ff5d7a';
  var hearts='';for(var h=0;h<Math.max(0,lives);h++)hearts+='\u2665 ';
  ctx.fillText(hearts||'-',24,62);
  ctx.restore();
}
