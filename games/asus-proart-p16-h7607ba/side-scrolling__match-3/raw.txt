  if(selected&&selected.c===p.c&&selected.r===p.r){ selected=null; tone(240,0.05,'sine',0.06); return; }
  if(selected&&(Math.abs(selected.c-p.c)+Math.abs(selected.r-p.r))===1){
    // swap
    const a=columns[selected.c][selected.r], b=columns[p.c][p.r];
    columns[selected.c][selected.r]=b; columns[p.c][p.r]=a;
    if(a&&b){ a.oy=0; b.oy=0; }
    if(findMatches().size>0){
      sSwap(); selected=null; resolving=true; combo=0; tryResolve();
    }else{
      columns[selected.c][selected.r]=a; columns[p.c][p.r]=b;
      sBad();
      invalids.push({c:selected.c,r:selected.r,t:0.3},{c:p.c,r:p.r,t:0.3});
      selected=null;
    }
  }else{
    selected=p; tone(520,0.06,'sine',0.08);
  }
}

/* ---------- update ---------- */
function update(dt){
  // falling animation always
  for(let c=0;c<columns.length;c++) for(let r=0;r<ROWS;r++){
    const g=columns[c][r];
    if(g&&g.oy!==0){
      g.oy += -g.oy*Math.min(1,dt*14);
      if(Math.abs(g.oy)<0.5) g.oy=0;
    }
  }
  for(let i=particles.length-1;i>=0;i--){ const p=particles[i]; p.t-=dt; p.x+=p.vx*dt; p.y+=p.vy*dt; p.vy+=380*dt; if(p.t<=0) particles.splice(i,1); }
  for(let i=popups.length-1;i>=0;i--){ const p=popups[i]; p.t-=dt; p.y-=30*dt; if(p.t<=0) popups.splice(i,1); }
  for(let i=invalids.length-1;i>=0;i--){ invalids[i].t-=dt; if(invalids[i].t<=0) invalids.splice(i,1); }

  if(state!=='play') return;

  // cascade timing
  if(resolving&&resolveTimer>0){ resolveTimer-=dt; if(resolveTimer<=0) tryResolve(); }

  // scroll
  speed=Math.min(95, 26+(level-1)*7);
  xOffset -= speed*dt;
  while(xOffset<=-cell){
    const leaving=columns.shift();
    let darks=0; for(const g of leaving) if(g&&g.dark) darks++;
    xOffset+=cell;
    columns.push(makeColumn(true));
    if(darks>0){
      lives-=darks; sLife();
      popups.push({x:cell*1.2,y:H*0.5,text:'-'+darks+' ♥',t:1,max:1,size:cell*0.6,col:'#ff5e7a'});
      for(let i=0;i<14;i++) particles.push({x:0,y:Math.random()*H,vx:80+Math.random()*120,vy:(Math.random()-0.5)*200,t:0.6,max:0.6,col:'#ff5e7a'});
      if(lives<=0){ lives=0; gameOver(); return; }
    }
  }

  // level up
  const nl=Math.floor(score/400)+1;
  if(nl>level){
    level=nl; lives=Math.min(5,lives+1); sLevel();
    popups.push({x:W/2,y:H*0.35,text:'LEVEL '+level+'  ·  +1 ♥',t:1.4,max:1.4,size:cell*0.7,col:'#7cf29c'});
  }
}

/* ---------- render ---------- */
function drawGem(g,x,y){
  const rad=cell*0.38;
  if(g.dark){
    const pulse=0.5+0.5*Math.sin(performance.now()/220);
    // glow
    const gl=ctx.createRadialGradient(x,y,rad*0.4,x,y,rad*1.5);
    gl.addColorStop(0,'rgba(255,60,90,'+(0.35+0.3*pulse)+')'); gl.addColorStop(1,'rgba(255,60,90,0)');
    ctx.fillStyle=gl; ctx.beginPath(); ctx.arc(x,y,rad*1.5,0,7); ctx.fill();
    // spikes
    ctx.save(); ctx.translate(x,y); ctx.rotate(performance.now()/1400);
    ctx.fillStyle='#1a1030';
    ctx.beginPath();
    const n=8;
    for(let i=0;i<n*2;i++){
      const rr=(i%2===0)?rad:rad*0.62, a=i/(n*2)*Math.PI*2;
      if(i===0) ctx.moveTo(Math.cos(a)*rr,Math.sin(a)*rr); else ctx.lineTo(Math.cos(a)*rr,Math.sin(a)*rr);
    }
    ctx.closePath(); ctx.fill();
    ctx.strokeStyle=COLORS[g.color][0]; ctx.lineWidth=Math.max(2,cell*0.05); ctx.stroke();
    // core
    const cg=ctx.createRadialGradient(x-rad*0.3,y-rad*0.3,rad*0.1,x,y,rad*0.62);
    cg.addColorStop(0,COLORS[g.color][1]); cg.addColorStop(1,COLORS[g.color][0]);
    ctx.beginPath(); ctx.arc(x,y,rad*0.58,0,7); ctx.fillStyle=cg; ctx.fill();
    ctx.restore();
  }else{
    const gr=ctx.createRadialGradient(x-rad*0.35,y-rad*0.4,rad*0.15,x,y,rad);
    gr.addColorStop(0,COLORS[g.color][1]); gr.addColorStop(1,COLORS[g.color][0]);
    ctx.beginPath(); ctx.arc(x,y,rad,0,7); ctx.fillStyle=gr; ctx.fill();
    // shine
    ctx.beginPath(); ctx.ellipse(x-rad*0.32,y-rad*0.4,rad*0.28,rad*0.16,-0.6,0,7);
    ctx.fillStyle='rgba(255,255,255,0.5)'; ctx.fill();
  }
}

function render(){
  // background
  const bg=ctx.createLinearGradient(0,0,0,H);
  bg.addColorStop(0,'#141a35'); bg.addColorStop(0.6,'#0b0e1f'); bg.addColorStop(1,'#12081f');
  ctx.fillStyle=bg; ctx.fillRect(0,0,W,H);
  // stars
  for(const s of stars){
    const sx=((s.x - (xOffset*0.25*s.z))%W+W)%W;
    ctx.globalAlpha=0.3+0.5*s.z;
    ctx.fillStyle='#9fb4ff';
    ctx.fillRect(sx,s.y,s.z*2,s.z*2);
  }
  ctx.globalAlpha=1;

  // playfield border
  const top=Math.max(8,Math.floor((H-ROWS*cell)/2));
  const fh=ROWS*cell;
  if(state!=='start'){
    ctx.fillStyle='rgba(255,255,255,0.04)';
    ctx.fillRect(-xOffset%cell, top-fh/ (1), 0,0); // noop keep structure
    ctx.fillRect(xOffset>0?0:0, 0,0,0);
  }

  if(state!=='start'){
    // field backdrop
    ctx.fillStyle='rgba(255,255,255,0.03)';
    ctx.fillRect(0, top, W, fh);
    // left danger zone
    const dz=ctx.createLinearGradient(0,0,cell*1.4,0);
    dz.addColorStop(0,'rgba(255,60,90,0.30)'); dz.addColorStop(1,'rgba(255,60,90,0)');
    ctx.fillStyle=dz; ctx.fillRect(0,top,cell*1.4,fh);
    // grid dots
    ctx.fillStyle='rgba(255,255,255,0.05)';
    for(let c=0;c<columns.length;c++){
      const x=c*cell+xOffset; if(x<-cell||x>W+cell) continue;
      for(let r=0;r<ROWS;r++) ctx.fillRect(x+cell/2-1,top+r*cell+cell/2-1,2,2);
    }

    // gems (clip to field)
    ctx.save();
    ctx.beginPath(); ctx.rect(0,top,W,fh); ctx.clip();
    for(let c=0;c<columns.length;c++){
      const x=c*cell+xOffset; if(x<-cell||x>W+cell) continue;
      for(let r=0;r<ROWS;r++){
        const g=columns[c][r]; if(!g) continue;
        drawGem(g, x+cell/2, top+r*cell+g.oy+cell/2);
      }
    }
    // selection ring
    if(selected){
      const sx=selected.c*cell+xOffset+cell/2, sy=top+selected.r*cell+cell/2;
      ctx.strokeStyle='#ffffff'; ctx.lineWidth=Math.max(3,cell*0.07);
      ctx.shadowColor='#fff'; ctx.shadowBlur=14;
      ctx.beginPath(); ctx.arc(sx,sy,cell*0.46,0,7); ctx.stroke();
      ctx.shadowBlur=0;
    }
    // invalid flashes
    for(const iv of invalids){
      const ix=iv.c*cell+xOffset+cell/2, iy=top+iv.r*cell+cell/2;
      ctx.strokeStyle='rgba(255,80,100,'+(iv.t/0.3)+')'; ctx.lineWidth=4;
      ctx.beginPath(); ctx.arc(ix,iy,cell*0.46,0,7); ctx.stroke();
    }
    // particles
    for(const p of particles){
      ctx.globalAlpha=Math.max(0,p.t/p.max);
      ctx.fillStyle=p.col;
      ctx.fillRect(p.x-3,p.y-3,6,6);
    }
    ctx.globalAlpha=1;
    ctx.restore();

    // HUD
    ctx.textAlign='left'; ctx.textBaseline='top';
    ctx.font='bold '+Math.max(20,cell*0.45)+'px Segoe UI, Arial';
    ctx.fillStyle='#ffd166'; ctx.shadowColor='#000'; ctx.shadowBlur=6;
    ctx.fillText('SCORE  '+score, 16, 14);
    ctx.font='bold '+Math.max(14,cell*0.3)+'px Segoe UI, Arial';
    ctx.fillStyle='#9fb4ff';
    ctx.fillText('LEVEL '+level+'   SPEED x'+(speed/26).toFixed(1), 16, 14+Math.max(24,cell*0.5));
    ctx.textAlign='right';
    ctx.font=Math.max(20,cell*0.45)+'px Segoe UI, Arial';
    let hearts=''; for(let i=0;i<5;i++) hearts += i<lives?'♥':'♡';
    ctx.fillStyle='#ff5e7a';
    ctx.fillText(hearts, W-16, 14);
    ctx.shadowBlur=0;

    // popups
    for(const p of popups){
      ctx.globalAlpha=Math.min(1,p.t/p.max*1.6);
      ctx.textAlign='center';
      ctx.font='bold '+p.size+'px Segoe UI, Arial';
      ctx.fillStyle=p.col; ctx.shadowColor='#000'; ctx.shadowBlur=8;
      ctx.fillText(p.text,p.x,p.y);
    }
    ctx.globalAlpha=1; ctx.shadowBlur=0;
  }
}
