let walls,goals,boxes,emitters,player,bullets,time=0;
let moveCd=0,invuln=0,flash=0,shake=0,clearT=0,faceDir={x:1,y:0};

function key(x,y){return x+','+y;}
function loadLevel(i){
  levelIdx=i;
  const L=LEVELS[i];
  walls=new Set();goals=new Set();boxes=new Set();emitters=[];bullets=[];
  player={x:1,y:1,px:1,py:1};
  invuln=0;flash=0;shake=0;moveCd=0.3;
  L.map.forEach((row,y)=>{
    for(let x=0;x<row.length;x++){
      const ch=row[x];
      if(ch==='#')walls.add(key(x,y));
      if(ch==='G')goals.add(key(x,y));
      if(ch==='B')boxes.add(key(x,y));
      if(ch==='P'){player.x=x;player.y=y;player.px=x;player.py=y;}
      if(ch==='E')emitters.push({x:x,y:y,type:((emitters.length+i)%3),timer:1.2+Math.random(),angle:Math.random()*6.28});
    }
  });
}
function startGame(){lives=3;loadLevel(0);state='play';paused=false;}
function retryLevel(){lives=3;loadLevel(levelIdx);state='play';paused=false;}

/* ---------- Input ---------- */
const keys={};
window.addEventListener('keydown',e=>{
  initAudio();
  const k=e.key.toLowerCase();
  if(['arrowup','arrowdown','arrowleft','arrowright',' '].indexOf(k)>=0)e.preventDefault();
  if(state==='start'&&(k==='enter'||k===' ')){startGame();return;}
  if((state==='over')&&(k==='r'||k==='enter')){retryLevel();return;}
  if((state==='win')&&(k==='r'||k==='enter')){startGame();return;}
  if(state==='play'&&k==='p'){paused=!paused;return;}
  if(k==='r'&&(state==='play'||state==='clear')){startGame();return;}
  keys[k]=true;
});
window.addEventListener('keyup',e=>{keys[e.key.toLowerCase()]=false;});
cvs.addEventListener('pointerdown',()=>{initAudio();if(state==='start')startGame();else if(state==='over')retryLevel();else if(state==='win')startGame();});

/* ---------- Gameplay ---------- */
function isWall(x,y){return walls.has(key(x,y));}
function isBox(x,y){return boxes.has(key(x,y));}
function onGoal(x,y){return goals.has(key(x,y));}

function tryMove(dx,dy){
  faceDir={x:dx,y:dy};
  const nx=player.x+dx,ny=player.y+dy;
  if(isWall(nx,ny)){sMove();return;}
  if(isBox(nx,ny)){
    const bx=nx+dx,by=ny+dy;
    if(isWall(bx,by)||isBox(bx,by)){sMove();shake=Math.max(shake,0.12);return;}
    boxes.delete(key(nx,ny));boxes.add(key(bx,by));
    sPush();
    if(onGoal(bx,by))sGoal();
  }else{
    sMove();
  }
  player.x=nx;player.y=ny;
  moveCd=0.16;
  // win check
  let all=true;
  boxes.forEach((bk)=>{const p=bk.split(',');if(!onGoal(+p[0],+p[1]))all=false;});
  if(all){
    state='clear';clearT=1.6;sWin();
  }
}

function fire(e){
  const L=LEVELS[levelIdx];
  const cx=e.x,cy=e.y,px=player.px,py=player.py;
  const s=L.speed;
  function add(a,sp,r){
    if(bullets.length>320)return;
    bullets.push({x:cx+0.5,y:cy+0.5,vx:Math.cos(a)*sp,vy:Math.sin(a)*sp,r:r||0.16,col:e.type});
  }
  if(e.type===0){ // aimed spread
    const base=Math.atan2(py-cy-0.5+0.5-(cy+0.5)-(py-cy-0.5),px-cx); // placeholder
    const a=Math.atan2((player.py+0.5)-(cy+0.5),(player.px+0.5)-(cx+0.5));
    const n=levelIdx===0?3:5;
    for(let i=0;i<n;i++)add(a+(i-(n-1)/2)*0.22,s);
  }else if(e.type===1){ // ring
    const n=8,off=Math.random()*6.28;
    for(let i=0;i<n;i++)add(off+i/n*6.283,s*0.85);
  }else{ // twin spiral
    e.angle+=0.55+levelIdx*0.1;
    add(e.angle,s*1.05,0.18);add(e.angle+Math.PI,s*1.05,0.18);
  }
  sShoot();
}

function hitPlayer(){
  lives--;invuln=1.8;flash=0.5;shake=0.45;sHit();
  if(lives<=0){state='over';sOver();}
}

/* ---------- Update ---------- */
let last=performance.now();
function update(dt){
  time+=dt;
  if(flash>0)flash-=dt;
  if(shake>0)shake-=dt*1.5;
  if(state==='clear'){
    clearT-=dt;
    if(clearT<=0){
      if(levelIdx+1<LEVELS.length)loadLevel(levelIdx+1),state='play';
      else state='win';
    }
    return;
  }
  if(state!=='play'||paused)return;

  // player animation
  const k=Math.min(1,dt*22);
  player.px+=(player.x-player.px)*k;
  player.py+=(player.y-player.py)*k;
  if(Math.abs(player.x-player.px)<0.02)player.px=player.x;
  if(Math.abs(player.y-player.py)<0.02)player.py=player.y;

  // movement input
  moveCd-=dt;invuln-=dt;
  const busy=(Math.abs(player.x-player.px)>0.35||Math.abs(player.y-player.py)>0.35);
  if(!busy&&moveCd<=0){
    let d=null;
    if(keys['arrowup']||keys['w'])d={x:0,y:-1};
    else if(keys['arrowdown']||keys['s'])d={x:0,y:1};
    else if(keys['arrowleft']||keys['a'])d={x:-1,y:0};
    else if(keys['arrowright']||keys['d'])d={x:1,y:0};
    if(d)tryMove(d.x,d.y);
  }

  // emitters
  const L=LEVELS[levelIdx];
  emitters.forEach(e=>{
    e.timer-=dt;
    if(e.timer<=0){fire(e);e.timer=L.rate*(0.85+Math.random()*0.3);}
  });

  // bullets
  for(let i=bullets.length-1;i>=0;i--){
    const b=bullets[i];
    b.x+=b.vx*dt;b.y+=b.vy*dt;
    const cx=Math.floor(b.x),cy=Math.floor(b.y);
    if(isWall(cx,cy)||isBox(cx,cy)){bullets.splice(i,1);continue;}
    // player hit
    if(invuln<=0){
      const dx=b.x-(player.px+0.5),dy=b.y-(player.py+0.5);
      if(dx*dx+dy*dy<Math.pow(0.36+b.r,2)){bullets.splice(i,1);hitPlayer();if(state!=='play')return;}
    }
  }
}

/* ---------- Rendering ---------- */
function view(){
  const L=LEVELS[levelIdx];
  const rows=L.map.length,cols=L.map[0].length;
  let T=Math.min((W-30)/cols,(H-95)/rows);
  T=Math.max(18,Math.floor(T));
  const ox=(W-cols*T)/2,oy=56+(H-56-rows*T)/2;
  return {T,ox,oy,cols,rows};
}

function drawGame(){
  const v=view(),{T,ox,oy}=v;
  const sx=shake>0?(Math.random()-0.5)*shake*18:0;
  const sy=shake>0?(Math.random()-0.5)*shake*18:0;
  ctx.save();ctx.translate(sx,sy);

  for(let y=0;y<v.rows;y++)for(let x=0;x<v.cols;x++){
    const px=ox+x*T,py=oy+y*T;
    if(walls.has(key(x,y))){
      ctx.fillStyle='#2b2f4a';ctx.fillRect(px,py,T,T);
      ctx.fillStyle='#3b4066';ctx.fillRect(px+2,py+2,T-4,Math.max(3,T*0.18));
    }else{
      ctx.fillStyle=((x+y)%2===0)?'#151726':'#171a2c';
      ctx.fillRect(px,py,T,T);
    }
  }
  // goals
  goals.forEach(gk=>{
    const p=gk.split(','),gx=ox+(+p[0])*T,gy=oy+(+p[1])*T;
    const c=T/2,pu=0.75+0.25*Math.sin(time*4);
    ctx.strokeStyle='rgba(80,255,160,'+(0.55+0.3*pu)+')';
    ctx.lineWidth=2;
    ctx.beginPath();
    ctx.moveTo(c+gx,c+gy-T*0.26*pu);ctx.lineTo(c+gx+T*0.26*pu,c+gy);
    ctx.lineTo(c+gx,c+gy+T*0.26*pu);ctx.lineTo(c+gx-T*0.26*pu,c+gy);
    ctx.closePath();ctx.stroke();
  });
  // boxes
  boxes.forEach(bk=>{
    const p=bk.split(','),bx=ox+(+p[0])*T,by=oy+(+p[1])*T;
    const ok=onGoal(+p[0],+p[1]);
    ctx.fillStyle=ok?'#1f6b45':'#7a4d1e';
    roundRect(bx+T*0.1,by+T*0.1,T*0.8,T*0.8,T*0.12);ctx.fill();
    ctx.strokeStyle=ok?'#5cff9d':'#c98b3f';ctx.lineWidth=2;
    roundRect(bx+T*0.1,by+T*0.1,T*0.8,T*0.8,T*0.12);ctx.stroke();
    ctx.strokeStyle=ok?'rgba(140,255,190,0.7)':'rgba(230,180,110,0.6)';
    ctx.beginPath();
    ctx.moveTo(bx+T*0.1,by+T*0.1);ctx.lineTo(bx+T*0.9,by+T*0.9);
    ctx.moveTo(bx+T*0.9,by+T*0.1);ctx.lineTo(bx+T*0.1,by+T*0.9);
    ctx.stroke();
  });
  // emitters
  emitters.forEach(e=>{
    const ex=ox+(e.x+0.5)*T,ey=oy+(e.y+0.5)*T;
    const cols=['#ff4d6a','#c26bff','#39d7ff'];
    const col=cols[e.type];
    const pu=0.8+0.2*Math.sin(time*5+e.x);
    ctx.strokeStyle=col;ctx.lineWidth=2;
    for(let i=0;i<4;i++){
      const a=time*1.6+i*Math.PI/2;
      ctx.beginPath();ctx.moveTo(ex,ey);
      ctx.lineTo(ex+Math.cos(a)*T*0.42*pu,ey+Math.sin(a)*T*0.42*pu);ctx.stroke();
    }
    ctx.fillStyle=col;ctx.beginPath();ctx.arc(ex,ey,T*0.2*pu,0,7);ctx.fill();
    ctx.fillStyle='#fff';ctx.beginPath();ctx.arc(ex,ey,T*0.08*pu,0,7);ctx.fill();
  });
  // player
  {
    const px=ox+(player.px+0.5)*T,py=oy+(player.py+0.5)*T;
    let a=1;
    if(invuln>0)a=(Math.sin(time*30)>0)?1:0.25;
    ctx.globalAlpha=a;
    ctx.fillStyle='#ffd94a';
    ctx.beginPath();ctx.arc(px,py,T*0.32,0,7);ctx.fill();
    ctx.strokeStyle='#fff2b0';ctx.lineWidth=2;
    ctx.beginPath();ctx.arc(px,py,T*0.32,0,7);ctx.stroke();
    const ex=faceDir.x*T*0.1,ey=faceDir.y*T*0.1;
    ctx.fillStyle='#222';
    ctx.beginPath();ctx.arc(px-T*0.1+ex,py-T*0.06+ey,T*0.05,0,7);ctx.fill();
    ctx.beginPath();ctx.arc(px+T*0.1+ex,py-T*0.06+ey,T*0.05,0,7);ctx.fill();
    ctx.globalAlpha=1;
  }
  // bullets
  const bcols=['#ff4d6a','#c26bff','#39d7ff'];
  bullets.forEach(b=>{
    const bx=ox+b.x*T,by=oy+b.y*T,r=b.r*T;
    ctx.strokeStyle='rgba(255,255,255,0.15)';ctx.lineWidth=r*1.4;
    ctx.beginPath();ctx.moveTo(bx-b.vx*0.06*T/by&&0+(-b.vx*0.05),by-(-b.vy*0.05));
    ctx.moveTo(bx-b.vx*0.06,by-b.vy*0.06);ctx.lineTo(bx,by);ctx.stroke();
    ctx.fillStyle=bcols[b.col];
    ctx.globalAlpha=0.35;ctx.beginPath();ctx.arc(bx,by,r*2.1,0,7);ctx.fill();
    ctx.globalAlpha=1;ctx.beginPath();ctx.arc(bx,by,r,0,7);ctx.fill();
    ctx.fillStyle='#fff';ctx.beginPath();ctx.arc(bx,by,r*0.45,0,7);ctx.fill();
  });
  ctx.restore();

  // flash
  if(flash>0){ctx.fillStyle='rgba(255,60,60,'+Math.min(0.35,flash)+')';ctx.fillRect(0,0,W,H);}

  /* HUD */
  ctx.textAlign='left';ctx.font='bold 18px monospace';
  for(let i=0;i<3;i++){
    ctx.fillStyle=i<lives?'#ff4d6a':'rgba(255,77,106,0.2)';
    ctx.fillText('♥',16+i*24,34);
  }
  ctx.fillStyle='#9aa3d0';ctx.textAlign='center';
  ctx.fillText('LEVEL '+(levelIdx+1)+' / '+LEVELS.length,W/2,34);
  let placed=0;boxes.forEach(bk=>{const p=bk.split(',');if(onGoal(+p[0],+p[1]))placed++;});
  ctx.textAlign='right';
  ctx.fillStyle='#5cff9d';ctx.fillText('CRATES '+placed+'/'+boxes.size,W-16,34);
}

function roundRect(x,y,w,h,r){
  ctx.beginPath();
  ctx.moveTo(x+r,y);ctx.lineTo(x+w-r,y);ctx.quadraticCurveTo(x+w,y,x+w,y+r);
  ctx.lineTo(x+w,y+h-r);ctx.quadraticCurveTo(x+w,y+h,x+w-r,y+h);
  ctx.lineTo(x+r,y+h);ctx.quadraticCurveTo(x,y+h,x,y+h-r);
  ctx.lineTo(x,y+r);ctx.quadraticCurveTo(x,y,x+r,y);ctx.closePath();
}

function overlay(title,sub,color){
  ctx.fillStyle='rgba(6,7,14,0.72)';ctx.fillRect(0,0,W,H);
  ctx.textAlign='center';
  ctx.fillStyle=color;ctx.font='bold '+Math.min(52,W*0.08)+'px monospace';
  ctx.fillText(title,W/2,H*0.34);
  ctx.fillStyle='#c9d1f5';ctx.font=Math.min(20,W*0.035)+'px monospace';
  sub.forEach((s,i)=>ctx.fillText(s,W/2,H*0.46+i*30));
}

function draw(){
  ctx.fillStyle='#0a0a14';ctx.fillRect(0,0,W,H);
  if(state==='start'){
    // decorative drifting bullets
    for(let i=0;i<24;i++){
      const x=(i*173.3+time*(20+i%5*8))%(W+40)-20;
      const y=((i*97.7)%(H-60))+30;
      ctx.fillStyle='rgba(90,120,255,0.25)';
      ctx.beginPath();ctx.arc(x,y,3+(i%4),0,7);ctx.fill();
    }
    overlay('DANMAKU SOKOBAN',[
      'Push every crate onto a green pad.',
      'Bullets fly from the orbs — and crates block shots!',
      '',
      'ARROWS / WASD  move & push',
      'P pause     R restart',
      '',
      '— press ENTER to start —'
    ],'#ffd94a');
    return;
  }
  drawGame();
  if(state==='clear'){
    ctx.fillStyle='rgba(6,7,14,0.5)';ctx.fillRect(0,0,W,H);
    ctx.textAlign='center';ctx.fillStyle='#5cff9d';
    ctx.font='bold '+Math.min(46,W*0.07)+'px monospace';
    ctx.fillText('LEVEL CLEAR!',W/2,H/2);
  }
  if(state==='over'){
    overlay('GAME OVER',['You were shredded by the danmaku.','','— press R or ENTER to retry —'],'#ff4d6a');
  }
  if(state==='win'){
    overlay('YOU WIN!',['All levels cleared. The bullet storm bows to you.','','','— press R or ENTER to play again —'],'#5cff9d');
  }
  if(paused&&state==='play'){
    overlay('PAUSED',['press P to resume'],'#39d7ff');
  }
}

function loop(now){
  const dt=Math.min(0.033,(now-last)/1000);last=now;
  update(dt);draw();
  requestAnimationFrame(loop);
}
requestAnimationFrame(loop);

})();
