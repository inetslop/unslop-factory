function overlapsStatic(p){

/* ---------- physics step ---------- */
function stepPhysics(dt){
  var p = piece;
  p.vy += GRAVITY * dt;

  // horizontal control: force-based for a physical feel
  var ax = 0;
  if(input.left)  ax -= 2600;
  if(input.right) ax += 2600;
  p.vx += ax*dt;
  p.vx *= Math.exp(-AIR_DRAG*dt);
  if(ax===0) p.vx *= Math.exp(-3.2*dt); // extra damping when not pushing
  p.vx = Math.max(-520, Math.min(520, p.vx));

  p.cx += p.vx*dt;
  p.cy += p.vy*dt;

  var contact=false;
  var sqs = squaresOf(p);

  for(var iter=0; iter<3; iter++){
    sqs = squaresOf(p);
    for(var i=0;i<sqs.length;i++){
      var s=sqs[i];
      // floor
      if(s.y+CELL > WORLD_H){
        p.cy -= (s.y+CELL-WORLD_H);
        if(Math.abs(p.vy) > 90){ bounceFx(p, s.x+CELL/2, WORLD_H, Math.abs(p.vy)); p.vy = -p.vy*REST; }
        else p.vy = 0;
        p.vx *= Math.exp(-FRICTION*dt*(iter===0?1:0.3));
        contact=true;
      }
      // walls
      if(s.x < 0){
        p.cx -= s.x;
        if(Math.abs(p.vx)>90){ bounceFx(p,0,s.y+CELL/2,Math.abs(p.vx)); p.vx=-p.vx*REST; } else p.vx=0;
        contact=true;
      } else if(s.x+CELL > WORLD_W){
        p.cx -= (s.x+CELL-WORLD_W);
        if(Math.abs(p.vx)>90){ bounceFx(p,WORLD_W,s.y+CELL/2,Math.abs(p.vx)); p.vx=-p.vx*REST; } else p.vx=0;
        contact=true;
      }
      // grid blocks
      var c = cellOf(s);
      for(var gy=c.gy-1; gy<=c.gy+1; gy++){
        if(gy<0||gy>=ROWS) continue;
        for(var gx=c.gx-1; gx<=c.gx+1; gx++){
          if(gx<0||gx>=COLS||!grid[gy][gx]) continue;
          var bx=gx*CELL, by=HEADER+gy*CELL;
          var ox = Math.min(s.x+CELL-bx, bx+CELL-s.x);
          var oy = Math.min(s.y+CELL-by, by+CELL-s.y);
          if(ox<=0||oy<=0) continue;
          if(ox < oy){
            if(s.x+CELL/2 < bx+CELL/2){ p.cx -= ox; } else { p.cx += ox; }
            if(Math.abs(p.vx)>90){ bounceFx(p, s.x+CELL/2, s.y+CELL/2, Math.abs(p.vx)); p.vx=-p.vx*REST; } else p.vx=0;
          } else {
            if(s.y+CELL/2 < by+CELL/2){ p.cy -= oy; } else { p.cy += oy; }
            if(Math.abs(p.vy)>90){ bounceFx(p, s.x+CELL/2, s.y+CELL/2, Math.abs(p.vy)); p.vy=-p.vy*REST; } else p.vy=0;
            p.vx *= (iter===0)?Math.exp(-FRICTION*dt):1;
          }
          contact=true;
        }
      }
    }
  }

  // settle / lock
  var speed2 = p.vx*p.vx + p.vy*p.vy;
  if(contact && speed2 < 4500){
    p.lockT += dt;
    p.vx *= Math.exp(-FRICTION*dt*2);
  } else {
    p.lockT = Math.max(0, p.lockT - dt*3);
  }
  p.flash = Math.max(0, p.flash - dt*4);

  if(p.lockT > 0.45) lockPiece();
}

function bounceFx(p, x, y, v){
  p.flash = 1;
  shakeT = Math.min(0.25, v/3000);
  sBounce(v);
  var n = Math.min(8, 2 + (v/200)|0);
  for(var i=0;i<n;i++){
    particles.push({
      x:x, y:y, vx:(Math.random()-0.5)*260, vy:-Math.random()*220-40,
      life:0.5+Math.random()*0.3, size:3+Math.random()*4, color:p.color
    });
  }
}

/* ---------- locking & line clears ---------- */
function lockPiece(){
  var sqs = squaresOf(piece);
  // find a fitting grid placement (shifts in cell units)
  var found=false;
  var shifts=[ [0,0],[-1,0],[1,0],[-2,0],[2,0],[0,-1],[-1,-1],[1,-1],[0,-2] ];
  for(var i=0;i<shifts.length && !found;i++){
    var dx=shifts[i][0], dy=shifts[i][1];
    var cells=[];
    var ok=true;
    for(var j=0;j<sqs.length;j++){
      var gx = Math.round(sqs[j].x/CELL)+dx;
      var gy = Math.round((sqs[j].y-HEADER)/CELL)+dy;
      if(gx<0||gx>=COLS||gy<0||gy>=ROWS){ ok=false; break; }
      cells.push([gx,gy]);
    }
    if(!ok) continue;
    for(var k=0;k<cells.length;k++){
      if(grid[cells[k][1]][cells[k][0]]){ ok=false; break; }
    }
    if(ok){
      for(var m=0;m<cells.length;m++) grid[cells[m][1]][cells[m][0]] = piece.color;
      found=true;
    }
  }

  // particles where piece was
  var sqs2 = squaresOf(piece);
  for(var q=0;q<sqs2.length;q++){
    for(var pp=0;pp<3;pp++){
      particles.push({
        x:sqs2[q].x+CELL/2, y:sqs2[q].y+CELL/2,
        vx:(Math.random()-0.5)*180, vy:-Math.random()*160-30,
        life:0.4+Math.random()*0.3, size:3+Math.random()*3, color:piece.color
      });
    }
  }
  sLock();

  if(!found){ gameOver(); return; }

  score += 10;

  // detect full rows
  var full=[];
  for(var y=0;y<ROWS;y++){
    var f=true;
    for(var x=0;x<COLS;x++) if(!grid[y][x]) { f=false; break; }
    if(f) full.push(y);
  }

  if(full.length){
    clearRows = full; clearTimer = 0.35; state='clearing';
    sClear(full.length);
    shakeT = Math.min(0.4, full.length*0.12);
  } else {
    spawnPiece();
  }
}

function finishClear(){
  var n = clearRows.length;
  // remove rows (top to bottom so indices stay valid)
  for(var i=clearRows.length-1;i>=0;i--){
    grid.splice(clearRows[i],1);
    grid.unshift(new Array(COLS).fill(0));
  }
  lines += n;
  level = Math.floor(lines/5)+1;
  score += [0,100,300,500,800][Math.min(n,4)] * level;
  clearRows=null;
  state='play';
  spawnPiece();
}

function gameOver(){
  state='over';
  sOver();
  if(score > hiScore){
    hiScore = score;
    try{ localStorage.setItem('tetrisPhysHi', String(hiScore)); }catch(e){}
  }
  document.getElementById('finalScore').textContent = 'SCORE: '+score;
  document.getElementById('hiScoreLine').textContent = 'BEST: '+hiScore;
  setTimeout(function(){ document.getElementById('overScreen').classList.remove('hidden'); }, 500);
}

/* ---------- input ---------- */
var input = { left:false, right:false };
window.addEventListener('keydown', function(e){
  ensureAudio();
  var k = e.key;
  if(['ArrowLeft','ArrowRight','ArrowUp','ArrowDown',' '].indexOf(k)>=0) e.preventDefault();

  if(k==='r'||k==='R'){ startGame(); return; }
  if(state==='start' && (k==='Enter'||k===' ')){ startGame(); return; }
  if(state==='over' && (k==='Enter')){ startGame(); return; }
  if(state!=='play') return;

  if(k==='ArrowLeft') input.left=true;
  else if(k==='ArrowRight') input.right=true;
  else if(k==='ArrowUp' || k==='x' || k==='X'){ tryRotate(1); }
  else if(k==='z' || k==='Z' || k==='ArrowDown'){ tryRotate(-1); }
  else if(k===' '){ piece.vy += 1700; sBounce(800); }
});
window.addEventListener('keyup', function(e){
  if(e.key==='ArrowLeft') input.left=false;
  if(e.key==='ArrowRight') input.right=false;
});

function tryRotate(dir){
  var p = piece;
  var oldRot = p.rot;
  p.rot = (p.rot + dir + 4) % 4;
  if(!overlapsStatic(p)){ sRotate(); return; }
  // wall kicks
  var kicks=[[0,-1],[ -CELL,0],[CELL,0],[-2*CELL,0],[2*CELL,0],[0,-2*CELL]];
  for(var i=0;i<kicks.length;i++){
    var tx=p.cx+kicks[i][0], ty=p.cy+kicks[i][1];
    p.cx=tx; p.cy=ty;
    if(!overlapsStatic(p)){ sRotate(); return; }
  }
  p.rot = oldRot; // rotate blocked
}

/* ---------- particles ---------- */
function stepParticles(dt){
  for(var i=particles.length-1;i>=0;i--){
    var p=particles[i];
    p.life -= dt;
    if(p.life<=0){ particles.splice(i,1); continue; }
    p.vy += 900*dt;
    p.x += p.vx*dt;
    p.y += p.vy*dt;
  }
}

/* ---------- render ---------- */
function roundRect(x,y,w,h,r){
  ctx.beginPath();
  ctx.moveTo(x+r,y);
  ctx.arcTo(x+w,y,x+w,y+h,r);
  ctx.arcTo(x+w,y+h,x,y+h,r);
  ctx.arcTo(x,y+h,x,y,r);
  ctx.arcTo(x,y,x+w,y,r);
  ctx.closePath();
}

function drawBlock(x,y,color,flash){
  ctx.fillStyle = color;
  roundRect(x+1.5,y+1.5,CELL-3,CELL-3,4);
  ctx.fill();
  // glossy top
  ctx.fillStyle='rgba(255,255,255,.28)';
  roundRect(x+3,y+3,CELL-6,(CELL-6)*0.35,3);
  ctx.fill();
  ctx.strokeStyle='rgba(0,0,0,.35)';
  ctx.lineWidth=1;
  roundRect(x+1.5,y+1.5,CELL-3,CELL-3,4);
  ctx.stroke();
  if(flash>0){
    ctx.fillStyle='rgba(255,255,255,'+(flash*0.7)+')';
    roundRect(x+1.5,y+1.5,CELL-3,CELL-3,4);
    ctx.fill();
  }
}

function render(){
  ctx.setTransform(1,0,0,1,0,0);
  ctx.fillStyle='#0a0a14';
  ctx.fillRect(0,0,canvas.width,canvas.height);

  // subtle starfield backdrop
  ctx.fillStyle='rgba(120,150,255,.12)';
  for(var i=0;i<40;i++){
    var sx=(i*97.3)%canvas.width, sy=(i*53.7)%canvas.height;
    ctx.fillRect(sx,sy,2,2);
  }

  var shx=0, shy=0;
  if(shakeT>0){
    shx=(Math.random()-0.5)*shakeT*40;
    shy=(Math.random()-0.5)*shakeT*40;
  }
  ctx.setTransform(scale,0,0,scale, offX+shx*scale, offY+shy*scale);

  // world frame
  ctx.fillStyle='#11162b';
  roundRect(-8,-8,WORLD_W+16,WORLD_H+16,14);
  ctx.fill();
  ctx.strokeStyle='#2a3a5c';
  ctx.lineWidth=3;
  roundRect(-8,-8,WORLD_W+16,WORLD_H+16,14);
  ctx.stroke();

  // play area
  ctx.fillStyle='#0d1224';
  ctx.fillRect(0,HEADER,WORLD_W,ROWS*CELL);

  // grid lines
  ctx.strokeStyle='rgba(90,120,200,.1)';
  ctx.lineWidth=1;
  ctx.beginPath();
  for(var gx=1;gx<COLS;gx++){ ctx.moveTo(gx*CELL+.5,HEADER); ctx.lineTo(gx*CELL+.5,WORLD_H); }
  for(var gy=0;gy<=ROWS;gy++){ ctx.moveTo(0,HEADER+gy*CELL+.5); ctx.lineTo(WORLD_W,HEADER+gy*CELL+.5); }
  ctx.stroke();

  // danger line
  ctx.strokeStyle='rgba(255,90,120,.3)';
  ctx.setLineDash([6,6]);
  ctx.beginPath();
  ctx.moveTo(0,HEADER+2.5*CELL+.5); ctx.lineTo(WORLD_W,HEADER+2.5*CELL+.5);
  ctx.stroke();
  ctx.setLineDash([]);

  // settled blocks
  if(grid){
    for(var y=0;y<ROWS;y++){
      var flashing = clearRows && clearRows.indexOf(y)>=0;
      for(var x=0;x<COLS;x++){
        var col = grid[y][x];
        if(!col) continue;
        if(flashing && Math.floor(clearTimer*14)%2===0) continue; // flicker out
        drawBlock(x*CELL, HEADER+y*CELL, col, 0);
      }
    }
  }

  // current piece (with slight squash while grounded)
  if(piece && (state==='play'||state==='clearing')){
    var sqs = squaresOf(piece);
    var f = piece.flash;
    for(var i=0;i<sqs.length;i++){
      drawBlock(sqs[i].x, sqs[i].y, piece.color, f);
    }
  }

  // particles
  for(var pi=0;pi<particles.length;pi++){
    var p=particles[pi];
    ctx.globalAlpha = Math.max(0, p.life*2);
    ctx.fillStyle=p.color;
    ctx.fillRect(p.x-p.size/2, p.y-p.size/2, p.size, p.size);
  }
  ctx.globalAlpha=1;

  // header HUD
  if(grid){
    ctx.fillStyle='#cfe0ff';
    ctx.font='bold 26px Courier New';
    ctx.textAlign='left';
    ctx.fillText('SCORE', 14, 30);
    ctx.fillStyle='#fff';
    ctx.font='bold 24px Courier New';
    ctx.fillText(String(score), 14, 55);

    ctx.fillStyle='#cfe0ff';
    ctx.font='bold 26px Courier New';
    ctx.textAlign='center';
    ctx.fillText('LV '+level, WORLD_W/2, 30);
    ctx.fillStyle='#8fa3c7';
    ctx.font='16px Courier New';
    ctx.fillText(lines+' LINES', WORLD_W/2, 54);

    // next piece preview
    ctx.textAlign='right';
    ctx.fillStyle='#cfe0ff';
    ctx.font='bold 26px Courier New';
    ctx.fillText('NEXT', WORLD_W-14, 30);
    if(nextShape){
      var nsh = SHAPES[nextShape];
      var offs = nsh.rots[0];
      // center of preview
      var minx=9,maxx=-9,miny=9,maxy=-9;
      offs.forEach(function(o){ minx=Math.min(minx,o[0]); maxx=Math.max(maxx,o[0]); miny=Math.min(miny,o[1]); maxy=Math.max(maxy,o[1]); });
      var pcx = WORLD_W - 34, pcy = 46;
      var s = 8; // mini cell
      offs.forEach(function(o){
        ctx.fillStyle = nsh.color;
        ctx.fillRect(pcx + (o[0]-(minx+maxx)/2)*s - s/2 + s/2, pcy + (o[1]-(miny+maxy)/2)*s - s/2 + s/2, s-1.5, s-1.5);
      });
    }
  }

  // "getting close" warning glow at top
  if(state==='play' && piece){
    var sqs2=squaresOf(piece);
    var minY=9999;
    sqs2.forEach(function(s){ minY=Math.min(minY,s.y); });
    if(minY < HEADER+3*CELL){
      ctx.fillStyle='rgba(255,60,90,'+(0.12+0.1*Math.sin(Date.now()/120))+')';
      ctx.fillRect(0,HEADER,WORLD_W,40);
    }
  }

  // start/over dim handled by DOM overlays
}

/* ---------- main loop ---------- */
var lastT = performance.now();
function frame(now){
  var dt = Math.min(0.033, (now-lastT)/1000);
  lastT = now;

  if(state==='play'){
    stepPhysics(dt);
  } else if(state==='clearing'){
    clearTimer -= dt;
    if(clearTimer<=0) finishClear();
  }
  shakeT = Math.max(0, shakeT - dt*2.5);
  stepParticles(dt);
  render();
  requestAnimationFrame(frame);
}

// initialize a preview grid so the background looks right before start
grid=[]; for(var y=0;y<ROWS;y++) grid.push(new Array(COLS).fill(0));
piece=null;
requestAnimationFrame(frame);

})();
