function sfx(freq, dur, type, vol, slide){
  if(!AC) return;
  try{
    const t=AC.currentTime, o=AC.createOscillator(), g=AC.createGain();
    o.type=type||'square'; o.frequency.setValueAtTime(freq,t);
    if(slide) o.frequency.exponentialRampToValueAtTime(Math.max(30,freq+slide), t+dur);
    g.gain.setValueAtTime(vol||0.04, t);
    g.gain.exponentialRampToValueAtTime(0.0001, t+dur);
    o.connect(g); g.connect(AC.destination); o.start(t); o.stop(t+dur+0.02);
  }catch(e){}
}

/* ---------- update ---------- */
function update(dt){
  if(state!=='play'||paused) return;

  // wave management
  if(!inWave){
    betweenTimer-=dt;
    if(betweenTimer<=0) startWave();
  } else {
    if(spawnLeft>0){
      spawnTimer-=dt;
      if(spawnTimer<=0){
        spawnEnemy();
        spawnLeft--;
        spawnTimer = Math.max(0.3, 1.05 - wave*0.04);
      }
    } else if(enemies.length===0 && ebullets.length===0){
      inWave=false;
      betweenTimer=4;
      const bonus = 20 + wave*5;
      gold += bonus; score += 100*wave;
      banner='WAVE CLEAR  +'+bonus+'g'; bannerT=2.2;
      sfx(660,0.25,'triangle',0.06,300);
    }
  }

  const core = corePos();

  // enemies
  for(let i=enemies.length-1;i>=0;i--){
    const e=enemies[i];
    e.x -= e.spd*dt;
    e.wob += e.wobSpd*dt;
    e.y = e.baseY + Math.sin(e.wob)*(e.type==='speeder'?26:12);
    if(e.shootT>0){
      e.shootT-=dt;
      if(e.shootT<=0){
        e.shootT = e.type.shoot || ENEMY_TYPES[e.type].shoot;
        const a=Math.atan2(core.y-e.y, core.x-e.x);
        ebullets.push({x:e.x-14,y:e.y,vx:Math.cos(a)*150,vy:Math.sin(a)*150,r:5});
        sfx(220,0.07,'square',0.03,-60);
      }
    }
    if(e.x < core.x+24){
      enemies.splice(i,1);
      coreHP -= e.dmg;
      burst(e.x,e.y,'#ff6b8a',14,160);
      sfx(120,0.2,'sawtooth',0.07,-60);
      if(coreHP<=0){ gameOver(); return; }
    }
  }

  // enemy bullets
  for(let i=ebullets.length-1;i>=0;i--){
    const b=ebullets[i];
    b.x+=b.vx*dt; b.y+=b.vy*dt;
    if(Math.hypot(b.x-core.x,b.y-core.y)<24){
      ebullets.splice(i,1);
      coreHP--;
      burst(core.x,core.y,'#c77dff',10,140);
      sfx(150,0.15,'sawtooth',0.06,-80);
      if(coreHP<=0){ gameOver(); return; }
    } else if(b.x<-20||b.y<-20||b.y>H+20) ebullets.splice(i,1);
  }

  // towers fire
  for(const t of towers){
    const tt=TOWER_TYPES[t.type];
    t.cd-=dt;
    t.angle+=dt*0.5;
    if(t.cd<=0){
      let best=null,bd=tt.range;
      for(const e of enemies){
        const d=Math.hypot(e.x-t.x,e.y-t.y);
        if(d<bd){ bd=d; best=e; }
      }
      if(best){
        t.cd=tt.rate;
        const baseA=Math.atan2(best.y-t.y,best.x-t.x);
        for(let s=0;s<tt.spread;s++){
          const off = tt.spread>1 ? (s-(tt.spread-1)/2)*0.24 : 0;
          pbullets.push({x:t.x,y:t.y,vx:Math.cos(baseA+off)*tt.speed,vy:Math.sin(baseA+off)*tt.speed,
                         dmg:tt.dmg,pierce:tt.pierce,hit:new Set(),color:tt.color});
        }
        if(performance.now()-lastShootSfx>60){ lastShootSfx=performance.now(); sfx(t.type===2?1400:880,0.05,'square',0.02,-300); }
      }
    }
  }

  // player bullets
  for(let i=pbullets.length-1;i>=0;i--){
    const b=pbullets[i];
    b.x+=b.vx*dt; b.y+=b.vy*dt;
    if(b.x>W+20||b.x<-20||b.y<-20||b.y>H+20){ pbullets.splice(i,1); continue; }
    for(const e of enemies){
      if(b.hit.has(e)) continue;
      if(Math.hypot(e.x-b.x,e.y-b.y) < e.r+4){
        b.hit.add(e);
        e.hp -= b.dmg;
        burst(b.x,b.y,b.color,3,80);
        if(e.hp<=0){
          const idx=enemies.indexOf(e);
          if(idx>=0) enemies.splice(idx,1);
          gold += e.reward; score += e.reward*10;
          burst(e.x,e.y,ENEMY_TYPES[e.type].color,14,170);
          sfx(300+Math.random()*200,0.1,'triangle',0.04,-150);
        }
        if(b.pierce>0){ b.pierce--; } else { pbullets.splice(i,1); break; }
      }
    }
  }

  // particles
  for(let i=particles.length-1;i>=0;i--){
    const p=particles[i];
    p.life-=dt; p.x+=p.vx*dt; p.y+=p.vy*dt; p.vx*=0.96; p.vy*=0.96;
    if(p.life<=0) particles.splice(i,1);
  }

  if(bannerT>0) bannerT-=dt;
  updateHud();
}

/* ---------- draw ---------- */
function poly(x,y,r,n,rot){
  ctx.beginPath();
  for(let i=0;i<n;i++){
    const a=rot+i*Math.PI*2/n;
    i?ctx.lineTo(x+Math.cos(a)*r,y+Math.sin(a)*r):ctx.moveTo(x+Math.cos(a)*r,y+Math.sin(a)*r);
  }
  ctx.closePath();
}

function draw(){
  ctx.clearRect(0,0,W,H);
  const core=corePos();

  // grid
  ctx.strokeStyle='rgba(90,110,200,0.10)'; ctx.lineWidth=1;
  ctx.beginPath();
  for(let c=0;c<=cols;c++){ ctx.moveTo(ox+c*CELL,oy); ctx.lineTo(ox+c*CELL,oy+rows*CELL); }
  for(let r=0;r<=rows;r++){ ctx.moveTo(ox,oy+r*CELL); ctx.lineTo(ox+cols*CELL,oy+r*CELL); }
  ctx.stroke();

  // core zone shading
  const zg=ctx.createLinearGradient(0,0,140,0);
  zg.addColorStop(0,'rgba(255,80,120,0.10)'); zg.addColorStop(1,'rgba(255,80,120,0)');
  ctx.fillStyle=zg; ctx.fillRect(0,0,140,H);

  // core
  const pulse=1+Math.sin(performance.now()/300)*0.06;
  ctx.save();
  ctx.shadowColor='#ff5d73'; ctx.shadowBlur=25;
  ctx.fillStyle='#ff5d73'; ctx.beginPath(); ctx.arc(core.x,core.y,20*pulse,0,7); ctx.fill();
  ctx.shadowBlur=0;
  ctx.fillStyle='#ffd9e2'; ctx.beginPath(); ctx.arc(core.x,core.y,10*pulse,0,7); ctx.fill();
  // hp arc
  ctx.strokeStyle='rgba(255,255,255,0.85)'; ctx.lineWidth=3;
  ctx.beginPath(); ctx.arc(core.x,core.y,28,-Math.PI/2,-Math.PI/2+ (Math.max(0,coreHP)/20)*Math.PI*2); ctx.stroke();
  ctx.restore();

  // towers
  for(const t of towers){
    const tt=TOWER_TYPES[t.type];
    ctx.save();
    ctx.translate(t.x,t.y);
    ctx.shadowColor=tt.color; ctx.shadowBlur=10;
    ctx.fillStyle='#1a2145'; ctx.strokeStyle=tt.color; ctx.lineWidth=2;
    if(t.type===0){ ctx.fillRect(-16,-16,32,32); ctx.strokeRect(-16,-16,32,32); }
    else if(t.type===1){ poly(0,0,19,3,t.angle*0.4); ctx.fill(); ctx.stroke(); }
    else { poly(0,0,17,4,t.angle*0.4); ctx.fill(); ctx.stroke(); }
    // barrel toward last angle
    const tt2=TOWER_TYPES[t.type];
    ctx.shadowBlur=0;
    ctx.strokeStyle=tt2.color; ctx.lineWidth=5;
    ctx.beginPath(); ctx.moveTo(0,0); ctx.lineTo(Math.cos(t.aim||0)*18,Math.sin(t.aim||0)*18); ctx.stroke();
    // level pips (cost shown by glow)
    ctx.fillStyle=tt.color; ctx.font='9px monospace'; ctx.textAlign='center';
    ctx.fillText('$'+tt2.cost, 0, 26);
    ctx.restore();
  }

  // enemies
  for(const e of enemies){
    const et=ENEMY_TYPES[e.type];
    ctx.save();
    ctx.translate(e.x,e.y);
    ctx.shadowColor=et.color; ctx.shadowBlur=8;
    ctx.fillStyle=et.color;
    poly(0,0,e.r,et.sides, performance.now()/400 + e.wob);
    ctx.fill();
    ctx.shadowBlur=0;
    ctx.fillStyle='rgba(10,12,25,0.8)';
    ctx.beginPath(); ctx.arc(0,0,e.r*0.4,0,7); ctx.fill();
    // hp bar
    const w=e.r*2;
    ctx.fillStyle='rgba(0,0,0,0.6)'; ctx.fillRect(-e.r,-e.r-8,w,3);
    ctx.fillStyle='#7dff9a'; ctx.fillRect(-e.r,-e.r-8,w*Math.max(0,e.hp/e.maxhp),3);
    ctx.restore();
  }

  // enemy bullets
  for(const b of ebullets){
    ctx.save();
    ctx.shadowColor='#c77dff'; ctx.shadowBlur=12;
    ctx.fillStyle='#e0b3ff';
    ctx.beginPath(); ctx.arc(b.x,b.y,b.r,0,7); ctx.fill();
    ctx.restore();
  }

  // player bullets
  for(const b of pbullets){
    ctx.save();
    ctx.strokeStyle=b.color; ctx.lineWidth=3; ctx.shadowColor=b.color; ctx.shadowBlur=8;
    const l=Math.hypot(b.vx,b.vy)*0.025;
    ctx.beginPath(); ctx.moveTo(b.x,b.y); ctx.lineTo(b.x-b.vx/l*0.14,b.y-b.vy/l*0.14); ctx.stroke();
    ctx.restore();
  }

  // particles
  for(const p of particles){
    ctx.globalAlpha=Math.max(0,p.life/p.max);
    ctx.fillStyle=p.color;
    ctx.beginPath(); ctx.arc(p.x,p.y,3,0,7); ctx.fill();
  }
  ctx.globalAlpha=1;

  // ghost preview
  if(state==='play' && mouse.in){
    const cell=cellOf(mouse.x,mouse.y);
    if(cell){
      const tt=TOWER_TYPES[selType];
      const occupied=towerAt(cell.r,cell.c);
      const ok=!occupied && gold>=tt.cost;
      const gx=ox+cell.c*CELL+CELL/2, gy=oy+cell.r*CELL+CELL/2;
      ctx.save();
      ctx.globalAlpha=0.35;
      ctx.strokeStyle=ok?tt.color:'#ff4d6a';
      ctx.setLineDash([6,6]);
      ctx.beginPath(); ctx.arc(gx,gy,tt.range,0,7); ctx.stroke();
      ctx.setLineDash([]);
      ctx.fillStyle=ok?tt.color:'#ff4d6a';
      ctx.fillRect(ox+cell.c*CELL+8,oy+cell.r*CELL+8,CELL-16,CELL-16);
      ctx.restore();
    }
  }

  // banner
  if(bannerT>0 && state==='play'){
    ctx.save();
    ctx.globalAlpha=Math.min(1,bannerT);
    ctx.fillStyle='#fff'; ctx.font='bold 34px "Segoe UI",Arial'; ctx.textAlign='center';
    ctx.shadowColor='#5a7dff'; ctx.shadowBlur=20;
    ctx.fillText(banner, W/2, H/2-40);
    if(!inWave && banner!=='WAVE CLEAR  +'){
      ctx.font='15px "Segoe UI",Arial'; ctx.fillStyle='#9fb0ff';
      ctx.fillText('Next wave in '+Math.max(0,betweenTimer).toFixed(1)+'s  —  press N to start now', W/2, H/2-8);
    }
    ctx.restore();
  }

  if(paused && state==='play'){
    ctx.fillStyle='rgba(5,6,14,0.7)'; ctx.fillRect(0,0,W,H);
    ctx.fillStyle='#fff'; ctx.font='bold 40px "Segoe UI",Arial'; ctx.textAlign='center';
    ctx.fillText('PAUSED', W/2, H/2);
    ctx.font='15px "Segoe UI",Arial'; ctx.fillStyle='#9fb0ff';
    ctx.fillText('Press P to resume', W/2, H/2+30);
  }
}

cvs.addEventListener('mousedown',e=>{
  initAudio();
  if(state!=='play'||paused) return;
  const r=cvs.getBoundingClientRect();
  const px=e.clientX-r.left, py=e.clientY-r.top;
  const cell=cellOf(px,py);
  if(!cell) return;

  if(e.button===2){ // sell
    const t=towerAt(cell.r,cell.c);
    if(t){
      towers.splice(towers.indexOf(t),1);
      gold += Math.floor(TOWER_TYPES[t.type].cost*0.6);
      burst(t.x,t.y,'#ffd968',12,140);
      sfx(500,0.1,'triangle',0.05,-200);
    }
  } else if(e.button===0){
    const tt=TOWER_TYPES[selType];
    if(towerAt(cell.r,cell.c)) return;
    if(gold<tt.cost){ sfx(140,0.12,'square',0.05,-60); return; }
    gold-=tt.cost;
    towers.push({r:cell.r,c:cell.c,type:selType,x:ox+cell.c*CELL+CELL/2,y:oy+cell.r*CELL+CELL/2,cd:0,angle:0,aim:0});
    burst(ox+cell.c*CELL+CELL/2, oy+cell.r*CELL+CELL/2, tt.color, 14, 150);
    sfx(700,0.1,'triangle',0.06,300);
  }
});
cvs.addEventListener('contextmenu',e=>e.preventDefault());

// tower aiming visual: recompute aim each frame in update? do it lightly here via draw hook
const _update=update;
function updateWrapped(dt){
  _update(dt);
  for(const t of towers){
    const tt=TOWER_TYPES[t.type];
    let best=null,bd=tt.range;
    for(const e of enemies){ const d=Math.hypot(e.x-t.x,e.y-t.y); if(d<bd){bd=d;best=e;} }
    if(best) t.aim=Math.atan2(best.y-t.y,best.x-t.x);
  }
}

  updateWrapped(dt);
  draw();
  requestAnimationFrame(frame);
}
