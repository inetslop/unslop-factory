<script>
(function(){
"use strict";

// ---------- Canvas / scaling ----------
const canvas = document.getElementById('game');
const ctx = canvas.getContext('2d');
let W = 0, H = 0;
function resize(){
  W = window.innerWidth; H = window.innerHeight;
  canvas.width = W; canvas.height = H;
}
window.addEventListener('resize', resize);
resize();

// ---------- DOM ----------
const hud = document.getElementById('hud');
const scoreEl = document.getElementById('score');
const levelEl = document.getElementById('level');
const livesEl = document.getElementById('lives');
const startScreen = document.getElementById('startScreen');
const gameOverScreen = document.getElementById('gameOverScreen');
const pauseScreen = document.getElementById('pauseScreen');
const startHighEl = document.getElementById('startHigh');
const finalScoreEl = document.getElementById('finalScore');
const finalHighEl = document.getElementById('finalHigh');

// ---------- localStorage (guarded) ----------
function loadHigh(){ try { return parseInt(localStorage.getItem('astSideHigh')||'0',10)||0; } catch(e){ return 0; } }
function saveHigh(v){ try { localStorage.setItem('astSideHigh', String(v)); } catch(e){} }
let highScore = loadHigh();
startHighEl.textContent = 'HIGH SCORE: ' + highScore;

// ---------- Audio (Web Audio, starts on first input) ----------
let actx = null;
function initAudio(){
  if(!actx){ try { actx = new (window.AudioContext||window.webkitAudioContext)(); } catch(e){} }
  if(actx && actx.state === 'suspended') actx.resume();
}
function beep(freq, dur, type, vol, slideTo){
  if(!actx) return;
  const t = actx.currentTime;
  const o = actx.createOscillator();
  const g = actx.createGain();
  o.type = type || 'square';
  o.frequency.setValueAtTime(freq, t);
  if(slideTo) o.frequency.exponentialRampToValueAtTime(Math.max(20, slideTo), t+dur);
  g.gain.setValueAtTime(vol||0.15, t);
  g.gain.exponentialRampToValueAtTime(0.0001, t+dur);
  o.connect(g); g.connect(actx.destination);
  o.start(t); o.stop(t+dur+0.02);
}
function noiseBurst(dur, vol){
  if(!actx) return;
  const t = actx.currentTime;
  const len = Math.floor(actx.sampleRate * dur);
  const buf = actx.createBuffer(1, len, actx.sampleRate);
  const d = buf.getChannelData(0);
  for(let i=0;i<len;i++) d[i] = (Math.random()*2-1) * (1 - i/len);
  const src = actx.createBufferSource();
  src.buffer = buf;
  const g = actx.createGain();
  g.gain.setValueAtTime(vol||0.25, t);
  g.gain.exponentialRampToValueAtTime(0.0001, t+dur);
  const f = actx.createBiquadFilter();
  f.type = 'lowpass';
  f.frequency.setValueAtTime(1400, t);
  f.frequency.exponentialRampToValueAtTime(120, t+dur);
  src.connect(f); f.connect(g); g.connect(actx.destination);
  src.start(t);
}
const sfx = {
  shoot(){ beep(880, .09, 'square', .07, 220); },
  explode(big){ noiseBurst(big?0.5:0.3, big?0.35:0.22); if(big) beep(90,.4,'sawtooth',.15,40); },
  hit(){ noiseBurst(.7,.4); beep(60,.6,'sawtooth',.25,30); },
  wave(){ beep(440,.12,'triangle',.12,880); setTimeout(()=>beep(660,.15,'triangle',.12,1320),120); }
};

// ---------- Input ----------
const keys = {};
window.addEventListener('keydown', function(e){
  initAudio();
  if(['ArrowUp','ArrowDown','ArrowLeft','ArrowRight',' '].includes(e.key)) e.preventDefault();
  keys[e.key.toLowerCase()] = true;
  if(e.key.toLowerCase() === 'p' && state==='playing'){ pauseGame(true); }
  else if(e.key.toLowerCase() === 'p' && state==='paused'){ pauseGame(false); }
  if(e.key === 'Enter'){
    if(state==='start') startGame();
    else if(state==='gameover') startGame();
  }
});
window.addEventListener('keyup', function(e){ keys[e.key.toLowerCase()] = false; });
document.getElementById('startBtn').addEventListener('click', ()=>{ initAudio(); startGame(); });
document.getElementById('restartBtn').addEventListener('click', ()=>{ initAudio(); startGame(); });

// ---------- Game state ----------
let state = 'start'; // start | playing | paused | gameover
let score = 0, lives = 3, wave = 1;
let player, bullets, asteroids, particles, stars, spawnTimer, shootCooldown, invuln, scrollSpeed;
let lastTime = 0;

// Starfield (side-scrolling parallax)
function makeStars(){
  const s = [];
  for(let i=0;i<130;i++){
    s.push({
      x: Math.random()*W, y: Math.random()*H,
      depth: Math.random()<.5 ? 0 : (Math.random()<.6 ? 1 : 2),
      size: Math.random()<.7?1:2
    });
  }
  return s;
}

function resetGame(){
  score = 0; lives = 3; wave = 1;
  player = { x: W*0.14, y: H/2, vy: 0, r: 16, thrusting:false };
  bullets = []; asteroids = []; particles = [];
  stars = makeStars();
  spawnTimer = 0.8; shootCooldown = 0; invuln = 2; scrollSpeed = 90;
  updateHud();
}

function startGame(){
  resetGame();
  state = 'playing';
  startScreen.classList.add('hidden');
  gameOverScreen.classList.add('hidden');
  pauseScreen.classList.add('hidden');
  hud.classList.remove('hidden');
  sfx.wave();
  lastTime = performance.now();
}

function pauseGame(on){
  if(on){ state='paused'; pauseScreen.classList.remove('hidden'); }
  else { state='playing'; pauseScreen.classList.add('hidden'); lastTime = performance.now(); }
}

function endGame(){
  state = 'gameover';
  if(score > highScore){ highScore = score; saveHigh(highScore); }
  finalScoreEl.textContent = 'SCORE: ' + score;
  finalHighEl.textContent = 'HIGH SCORE: ' + highScore + (score>=highScore && score>0 ? '  ★ NEW RECORD' : '');
  hud.classList.add('hidden');
  gameOverScreen.classList.remove('hidden');
}

function updateHud(){
  scoreEl.textContent = 'SCORE ' + score;
  levelEl.textContent = 'WAVE ' + wave;
  livesEl.innerHTML = 'LIVES ' + (lives>0 ? '&#9654;'.repeat(Math.min(lives,5)) : '—');
}

// ---------- Asteroids ----------
function makeAsteroid(size, x, y){
  const radii = [1,2,3]; // 1 small ... 3 large
  const r = size===3 ? 46 : size===2 ? 28 : 15;
  const verts = [];
  const n = 9 + Math.floor(Math.random()*4);
  for(let i=0;i<n;i++){
    verts.push(0.7 + Math.random()*0.4);
  }
  const speedMul = (wave-1)*0.06;
  return {
    x: x!==undefined ? x : W + r + 20,
    y: y!==undefined ? y : r + Math.random()*(H - 2*r),
    r: r, size: size, verts: verts,
    vx: -(60 + Math.random()*70) * (1 + speedMul) * (size===1?1.5:size===2?1.15:1),
    vy: (Math.random()-0.5) * 60,
    rot: Math.random()*Math.PI*2,
    vrot: (Math.random()-0.5)*2
  };
}

function spawnWaveAsteroids(){
  const count = 1 + Math.floor(Math.random()*2) + (wave>3?1:0);
  for(let i=0;i<count;i++){
    const size = Math.min(3, wave<2 ? 2 : (Math.random()<.4?3:2));
    asteroids.push(makeAsteroid(size, W + 60 + Math.random()*W*0.5));
  }
}

function burst(x, y, n, color, speed){
  for(let i=0;i<n;i++){
    const a = Math.random()*Math.PI*2;
    const s = (0.3+Math.random()) * speed;
    particles.push({
      x:x, y:y, vx:Math.cos(a)*s, vy:Math.sin(a)*s,
      life: 0.4 + Math.random()*0.5, maxLife: 0.9,
      color: color, size: 1+Math.random()*2.5
    });
  }
}

function destroyAsteroid(idx){
  const a = asteroids[idx];
  asteroids.splice(idx,1);
  const pts = a.size===3 ? 40 : a.size===2 ? 20 : 10;
  score += pts;
  if(score > highScore){ highScore = score; }
  burst(a.x, a.y, a.size*8, '#c9b391', 160);
  sfx.explode(a.size===3);
  if(a.size>1){
    for(let i=0;i<2;i++){
      const child = makeAsteroid(a.size-1, a.x, a.y + (i? a.r*0.7 : -a.r*0.7));
      child.vx = a.vx * 0.8 + (Math.random()-0.5)*90;
      child.vy = a.vy + (Math.random()-0.5)*120;
      asteroids.push(child);
    }
  }
  updateHud();
}

function hitPlayer(){
  lives--;
  invuln = 2.5;
  burst(player.x, player.y, 40, '#7fd4ff', 240);
  sfx.hit();
  updateHud();
  if(lives<=0){
    setTimeout(endGame, 600);
  }
}

// ---------- Update ----------
function update(dt){
  scrollSpeed = 90 + (wave-1)*15;

  // stars
  for(const s of stars){
    const sp = scrollSpeed * (s.depth===2?2.4: s.depth===1?1.3 : 0.6);
    s.x -= sp*dt;
    if(s.x < -2){ s.x = W+2; s.y = Math.random()*H; }
  }

  // player
  const up = keys['arrowup']||keys['w'];
  const down = keys['arrowdown']||keys['s'];
  player.thrusting = !!(up||down);
  const accel = 900;
  if(up) player.vy -= accel*dt;
  if(down) player.vy += accel*dt;
  if(!up && !down) player.vy *= Math.pow(0.02, dt); // damping
  player.vy = Math.max(-420, Math.min(420, player.vy));
  player.y += player.vy*dt;
  if(player.y < player.r){ player.y = player.r; player.vy = 0; }
  if(player.y > H-player.r){ player.y = H-player.r; player.vy = 0; }

  // exhaust particles
  if(player.thrusting && Math.random()<0.7){
    particles.push({
      x: player.x - 16, y: player.y + (Math.random()-0.5)*8,
      vx: -120-Math.random()*80, vy:(Math.random()-0.5)*40,
      life: .3, maxLife:.3, color:'#ffb35a', size: 2
    });
  }

  // shooting
  shootCooldown -= dt;
  if((keys[' ']||keys['shift']) && shootCooldown<=0){
    bullets.push({ x: player.x + 20, y: player.y, vx: 780 });
    shootCooldown = 0.22;
    sfx.shoot();
  }

  // bullets
  for(let i=bullets.length-1;i>=0;i--){
    bullets[i].x += bullets[i].vx*dt;
    if(bullets[i].x > W+10) bullets.splice(i,1);
  }

  // asteroid spawning
  spawnTimer -= dt;
  const maxA = Math.min(4 + wave*2, 22);
  if(spawnTimer<=0 && asteroids.length < maxA){
    spawnWaveAsteroids();
    spawnTimer = Math.max(0.5, 2.6 - wave*0.18) * (0.7+Math.random()*0.6);
  }

  // asteroids move
  for(let i=asteroids.length-1;i>=0;i--){
    const a = asteroids[i];
    a.x += a.vx*dt; a.y += a.vy*dt; a.rot += a.vrot*dt;
    if(a.y < a.r) { a.y = a.r; a.vy = Math.abs(a.vy); }
    if(a.y > H-a.r){ a.y = H-a.r; a.vy = -Math.abs(a.vy); }
    if(a.x < -a.r-30) asteroids.splice(i,1);
  }

  // bullet vs asteroid
  for(let i=bullets.length-1;i>=0;i--){
    const b = bullets[i];
    for(let j=asteroids.length-1;j>=0;j--){
      const a = asteroids[j];
      const dx=b.x-a.x, dy=b.y-a.y;
      if(dx*dx+dy*dy < a.r*a.r){
        bullets.splice(i,1);
        destroyAsteroid(j);
        break;
      }
    }
  }

  // player vs asteroid
  invuln -= dt;
  if(invuln<=0){
    for(let j=asteroids.length-1;j>=0;j--){
      const a = asteroids[j];
      const dx=player.x-a.x, dy=player.y-a.y;
      const rr = a.r + player.r*0.75;
      if(dx*dx+dy*dy < rr*rr){
        destroyAsteroid(j);
        hitPlayer();
        break;
      }
    }
  }

  // wave progression
  const newWave = 1 + Math.floor(score/250);
  if(newWave > wave){ wave = newWave; updateHud(); sfx.wave(); burst(W*0.5, H*0.5, 30, '#8fd7ff', 200); }

  // particles
  for(let i=particles.length-1;i>=0;i--){
    const p = particles[i];
    p.x += p.vx*dt; p.y += p.vy*dt;
    p.life -= dt;
    if(p.life<=0) particles.splice(i,1);
  }
}

// ---------- Draw ----------
function drawShip(){
  if(invuln>0 && Math.floor(invuln*12)%2===0) return; // blink
  ctx.save();
  ctx.translate(player.x, player.y);
  const tilt = Math.max(-0.35, Math.min(0.35, player.vy/900));
  ctx.rotate(tilt);
  // flame
  if(player.thrusting){
    const fl = 8 + Math.random()*10;
    ctx.beginPath();
    ctx.moveTo(-16, -5); ctx.lineTo(-16-fl, 0); ctx.lineTo(-16, 5);
    ctx.closePath();
    ctx.fillStyle = 'rgba(255,160,70,.9)';
    ctx.fill();
  }
  // hull
  ctx.beginPath();
  ctx.moveTo(22, 0); ctx.lineTo(-14, -13); ctx.lineTo(-8, 0); ctx.lineTo(-14, 13);
  ctx.closePath();
  ctx.fillStyle = '#dff4ff';
  ctx.shadowColor = '#57c0ff'; ctx.shadowBlur = 16;
  ctx.fill();
  ctx.strokeStyle = '#57c0ff'; ctx.lineWidth = 2; ctx.stroke();
  ctx.restore();
}

function drawAsteroid(a){
  ctx.save();
  ctx.translate(a.x, a.y);
  ctx.rotate(a.rot);
  ctx.beginPath();
  const n = a.verts.length;
  for(let i=0;i<n;i++){
    const ang = i/n * Math.PI*2;
    const rr = a.r * a.verts[i];
    const px = Math.cos(ang)*rr, py = Math.sin(ang)*rr;
    if(i===0) ctx.moveTo(px,py); else ctx.lineTo(px,py);
  }
  ctx.closePath();
  ctx.fillStyle = 'rgba(70,62,54,.9)';
  ctx.fill();
  ctx.strokeStyle = '#c9b391';
  ctx.lineWidth = 2;
  ctx.shadowColor = 'rgba(200,180,150,.5)'; ctx.shadowBlur = 8;
  ctx.stroke();
  // craters
  ctx.shadowBlur = 0;
  ctx.strokeStyle = 'rgba(255,235,200,.25)';
  ctx.beginPath();
  ctx.arc(-a.r*0.25, -a.r*0.15, a.r*0.28, 0, Math.PI*1.4);
  ctx.stroke();
  ctx.restore();
}

function draw(){
  // background gradient
  const g = ctx.createLinearGradient(0,0,0,H);
  g.addColorStop(0,'#070a1c');
  g.addColorStop(.5,'#04060f');
  g.addColorStop(1,'#0a0818');
  ctx.fillStyle = g;
  ctx.fillRect(0,0,W,H);

  // stars
  for(const s of stars){
    const c = s.depth===2 ? '#eaf6ff' : s.depth===1 ? '#9fc4e8' : '#4a6a90';
    ctx.fillStyle = c;
    ctx.fillRect(s.x, s.y, s.size, s.size);
  }

  if(state==='start'){ return; } // start screen over starfield

  // particles
  for(const p of particles){
    const a = Math.max(0, p.life/p.maxLife);
    ctx.globalAlpha = a;
    ctx.fillStyle = p.color;
    ctx.fillRect(p.x - p.size/2, p.y - p.size/2, p.size, p.size);
  }
  ctx.globalAlpha = 1;

  // bullets
  for(const b of bullets){
    ctx.strokeStyle = '#aef';
    ctx.lineWidth = 3;
    ctx.shadowColor = '#6cf'; ctx.shadowBlur = 10;
    ctx.beginPath();
    ctx.moveTo(b.x-8, b.y); ctx.lineTo(b.x+4, b.y);
    ctx.stroke();
    ctx.shadowBlur = 0;
  }

  for(const a of asteroids) drawAsteroid(a);
  if(lives>0 || state==='gameover'===false) drawShip();
}

// ---------- Main loop ----------
function loop(now){
  requestAnimationFrame(loop);
  const dt = Math.min(0.033, (now - lastTime)/1000 || 0.016);
  lastTime = now;

  if(state==='playing'){
    update(dt);
    draw();
  } else if(state==='paused'){
    draw(); // frozen frame behind pause overlay
  } else if(state==='gameover'){
    // keep drifting background + particles for effect
    scrollSpeed *= 0.98;
    for(const s of stars){ s.x -= scrollSpeed*dt; if(s.x<-2){s.x=W+2;s.y=Math.random()*H;} }
    for(let i=particles.length-1;i>=0;i--){
      const p = particles[i]; p.x+=p.vx*dt; p.y+=p.vy*dt; p.life-=dt;
      if(p.life<=0) particles.splice(i,1);
    }
    draw();
  } else { // start: drift stars
    for(const s of (stars||makeStars())){
      stars = stars || makeStars();
      s.x -= 60*(s.depth===2?2:s.depth===1?1.3:0.6)*dt;
      if(s.x<-2){ s.x=W+2; s.y=Math.random()*H; }
    }
    draw();
  }
}
stars = makeStars();
requestAnimationFrame(loop);

})();
</script>
</body>
</html>
