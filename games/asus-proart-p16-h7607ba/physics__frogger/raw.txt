function circleRect(cx,cy,r,rx,ry,rw,rh){

  if(frog.y<26){frog.y=26;if(frog.vy<0)frog.vy*=-0.3;}

  }else frog.drownT=0;

  for(const L of riverLanes)for(const log of L.logs)log.x+=L.dir*L.speed*dt;

  if(state==='play'||state==='paused'){
    const drowny=frog.drownT>0;
    drawFrog(frog.x,frog.y,frog.angle,drowny?Math.max(0.35,1-frog.drownT):1);
    if(drowny){
      ctx.strokeStyle='rgba(80,160,255,'+(0.4+0.3*Math.sin(t*20))+')';
      ctx.lineWidth=3;
      ctx.beginPath();ctx.arc(frog.x,frog.y,20+frog.drownT*14,0,7);ctx.stroke();
    }
  }else if(state==='dying'&&deathKind!=='crash'){
    // frog sinks
    drawFrog(frog.x,frog.y+ (0.9-deathTimer)*18,frog.angle,deathTimer/0.9);
  }

  // particles
  for(const p of particles){
    ctx.globalAlpha=Math.max(0,p.life/p.max);
    ctx.fillStyle=p.color;
    ctx.beginPath();ctx.arc(p.x,p.y,p.r,0,7);ctx.fill();
  }
  ctx.globalAlpha=1;

  // HUD
  if(state!=='start'){
    ctx.fillStyle='rgba(0,0,0,0.5)';rr(8,6,W-16,34,8);ctx.fill();
    ctx.fillStyle='#fff';ctx.font='bold 16px monospace';ctx.textAlign='left';
    ctx.fillText('SCORE '+score,18,29);
    ctx.textAlign='center';
    if(combo>1)ctx.fillText('COMBO x'+Math.min(combo,5),W/2,29);
    ctx.textAlign='right';
    ctx.fillText('LVL '+level+'   BEST '+best,W-18,29);
    // lives
    for(let i=0;i<lives-1;i++)drawFrog(28+i*30,62,-Math.PI/2,1);
  }

  // overlays
  ctx.textAlign='center';
  if(state==='start'){
    panel();
    ctx.fillStyle='#7CFC00';ctx.font='bold 44px monospace';
    ctx.fillText('FROGGER',W/2,220);
    ctx.fillStyle='#fff';ctx.font='bold 30px monospace';
    ctx.fillText('P H Y S I C S',W/2,262);
    drawFrog(W/2,330,-Math.PI/2+Math.sin(t*3)*0.4,1);
    ctx.fillStyle='#cfe8cf';ctx.font='15px monospace';
    ctx.fillText('Your frog has momentum — it slides!',W/2,400);
    ctx.fillText('Cross the road. Ride the logs.',W/2,424);
    ctx.fillText('Fill all 5 holes to level up.',W/2,448);
    ctx.fillStyle='#fff';ctx.font='bold 16px monospace';
    ctx.fillText('ARROWS / WASD — move (with inertia)',W/2,495);
    ctx.fillText('P — pause      R — restart',W/2,520);
    if(Math.sin(t*4)>-0.3){ctx.fillStyle='#7CFC00';ctx.font='bold 20px monospace';ctx.fillText('PRESS ENTER TO START',W/2,575);}
    ctx.fillStyle='#9ab';ctx.font='13px monospace';
    ctx.fillText('BEST: '+best,W/2,605);
  }else if(state==='over'){
    panel();
    ctx.fillStyle='#ff6b4a';ctx.font='bold 42px monospace';
    ctx.fillText('GAME OVER',W/2,270);
    ctx.fillStyle='#fff';ctx.font='bold 22px monospace';
    ctx.fillText('SCORE: '+score,W/2,330);
    if(score>=best&&score>0){ctx.fillStyle='#ffd93b';ctx.font='bold 18px monospace';ctx.fillText('NEW BEST!',W/2,362);}
    else{ctx.fillStyle='#9ab';ctx.font='15px monospace';ctx.fillText('BEST: '+best,W/2,362);}
    if(Math.sin(t*4)>-0.3){ctx.fillStyle='#7CFC00';ctx.font='bold 20px monospace';ctx.fillText('PRESS ENTER TO RESTART',W/2,430);}
  }else if(state==='paused'){
    ctx.fillStyle='rgba(0,0,0,0.55)';ctx.fillRect(0,0,W,H);
    ctx.fillStyle='#fff';ctx.font='bold 36px monospace';
    ctx.fillText('PAUSED',W/2,H/2-10);
    ctx.font='16px monospace';ctx.fillText('Press P to resume',W/2,H/2+28);
  }else if(state==='levelup'){
    ctx.fillStyle='#ffd93b';ctx.font='bold 34px monospace';
    ctx.fillText('LEVEL '+level+' CLEAR!',W/2,H/2-10);
    ctx.fillStyle='#fff';ctx.font='16px monospace';
    ctx.fillText('Speeding up... level '+level+1,W/2,H/2+26);
  }
}
function panel(){
  ctx.fillStyle='rgba(5,15,30,0.92)';ctx.fillRect(0,0,W,H);
  ctx.strokeStyle='#7CFC00';ctx.lineWidth=4;
  rr(W/2-190,180,380,440,16);ctx.stroke();
}

/* ---------- loop ---------- */
let last=performance.now();
function frame(now){
  let dt=(now-last)/1000;last=now;
  if(dt>0.05)dt=0.05;
  t+=dt;
  update(dt);
  draw();
  requestAnimationFrame(frame);
}
requestAnimationFrame(frame);
})();
