function circleRect(cx,cy,r,rc){
  const nx=Math.max(rc.x,Math.min(cx,rc.x+rc.w));
  const ny=Math.max(rc.y,Math.min(cy,rc.y+rc.h));
  const dx=cx-nx, dy=cy-ny;
  return dx*dx+dy*dy < r*r;
}
function collides(x,y){
  const len=cs*0.5, wid=cs*0.34;
  const cos=Math.cos(car.a), sin=Math.sin(car.a);
  const pts=[ [len*0.28,-wid/2],[len*0.28,wid/2],[-len*0.28,-wid/2],[-len*0.28,wid/2],[0,0] ];
  for(let i=0;i<walls.length;i++){
    const rc=walls[i];
    for(let p=0;p<pts.length;p++){
      const px=x+cos*pts[p][0]-sin*pts[p][1];
      const py=y+sin*pts[p][0]+cos*pts[p][1];
      if(circleRect(px,py,wid*0.42,rc)) return true;
    }
  }
  return false;
}

function update(dt){
  if(state!=='play') { if(engineGain) engineGain.gain.value=0; return; }
  raceTime+=dt;
  const maxF=cs*6.5, maxR=cs*2.8, acc=cs*16, fric=cs*9;

  if(keys['arrowup']||keys['w']) car.speed+=acc*dt;
  else if(keys['arrowdown']||keys['s']) car.speed-=acc*dt;
  else {
    if(car.speed>0) car.speed=Math.max(0,car.speed-fric*dt);
    else car.speed=Math.min(0,car.speed+fric*dt);
  }
  car.speed=Math.max(-maxR,Math.min(maxF,car.speed));

  const steer=2.9*Math.max(-1,Math.min(1,car.speed/(cs*3)));
  if(keys['arrowleft']||keys['a']) car.a+=steer*dt;
  if(keys['arrowright']||keys['d']) car.a-=steer*dt;

  const nx=car.x+Math.cos(car.a)*car.speed*dt;
  const ny=car.y+Math.sin(car.a)*car.speed*dt;
  let hit=false;
  if(!collides(nx,car.y)) car.x=nx; else hit=true;
  if(!collides(car.x,ny)) car.y=ny; else hit=true;
  if(hit){
    car.speed*=-0.25;
    beep(120,0.08,'sawtooth',0.08);
  }

  // finish check
  const gi=Math.floor((car.x-ox)/cs), gj=Math.floor((car.y-oy)/cs);
  if(gi===COLS-1 && gj===ROWS-1 && !finished){
    finished=true; state='finish';
    beep(523,0.12,'square',0.14); setTimeout(()=>beep(659,0.12,'square',0.14),130);
    setTimeout(()=>beep(784,0.25,'square',0.14),260);
    let isBest=false;
    if(best===null||raceTime<best){ best=raceTime; isBest=true;
      try{ localStorage.setItem('mazeRacerBest',String(best)); }catch(e){} }
    showFinish(isBest);
  }

  if(engineGain&&engineOsc){
    const s=Math.abs(car.speed)/cs;
    engineGain.gain.value=Math.min(0.06,0.015+s*0.012);
    engineLp.frequency.value=300+s*380;
    engineOsc.frequency.value=70+s*42;
  }
}

  // HUD
  if(state!=='start'){

    ctx.fillText('BEST  '+(best!==null?fmt(best):'--:--'),16,12+Math.max(20,cs*0.5));
    // speed bar
    const sp=Math.abs(car.speed)/(cs*6.5);
    ctx.fillStyle='rgba(255,255,255,0.12)';
    ctx.fillRect(W-180,18,164,14);
    ctx.fillStyle=sp>0.75?'#ff5a5a':'#ffd54a';

  }
}
