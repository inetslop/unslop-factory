function proj(v){ // v = [depth, up, right]
  if(v[0]<0.4)return null;
  return [CX+v[2]*FOCAL/v[0], CY-v[1]*FOCAL/v[0], v[0]];
}

/* ---------- update ---------- */
function rodrigues(v,k,t){
  var c=Math.cos(t),s=Math.sin(t);
  var kx=k[0],ky=k[1],kz=k[2];
  var dot=v[0]*kx+v[1]*ky+v[2]*kz;
  var cxp=ky*v[2]-kz*v[1],cyp=kz*v[0]-kx*v[2],czp=kx*v[1]-ky*v[0];
  return [v[0]*c+cxp*s+kx*dot*(1-c), v[1]*c+cyp*s+ky*dot*(1-c), v[2]*c+czp*s+kz*dot*(1-c)];
}

function update(dt){
  time+=dt;
  var k=keys;
  // ship control
  var turn=(k['KeyA']||k['ArrowLeft']?-1:0)+(k['KeyD']||k['ArrowRight']?1:0);
  var pit=(k['KeyQ']?1:0)+(k['KeyE']?-1:0);
  ship.yaw+=turn*2.4*dt;
  ship.pitch=Math.max(-0.9,Math.min(0.9,ship.pitch+pit*1.6*dt));
  var f=fwd(ship.yaw,ship.pitch);
  var thr=(k['KeyW']||k['ArrowUp'])?1:0, rev=(k['KeyS']||k['ArrowDown'])?-0.5:0;
  var acc=(thr+rev)*46;
  ship.vx+=f[0]*acc*dt;ship.vy+=f[1]*acc*dt;ship.vz+=f[2]*acc*dt;
  var damp=Math.pow(0.35,dt);
  ship.vx*=damp;ship.vy*=damp;ship.vz*=damp;
  ship.x+=ship.vx*dt;ship.y+=ship.vy*dt;ship.z+=ship.vz*dt;
  if(AC&&engGain)try{engGain.gain.setTargetAtTime(thr?0.06:0,AC.currentTime,0.08);}catch(e){}

  // camera follow
  var ty=ship.yaw, tp=Math.max(-0.4,Math.min(0.5,ship.pitch*0.6+0.16));
  cam.yaw+=(ty-cam.yaw)*Math.min(1,dt*6);
  cam.pitch+=(tp-cam.pitch)*Math.min(1,dt*5);
  updateCamBasis();
  var cf=fwd(cam.yaw,cam.pitch);
  cam.x=ship.x-cf[0]*cam.dist;
  cam.y=ship.y-cf[1]*cam.dist+cam.up;
  cam.z=ship.z-cf[2]*cam.dist;

  // shooting
  shootCd-=dt;
  if((k['Space'])&&shootCd<=0){
    shootCd=0.18;
    var rf=fwd(ship.yaw,ship.pitch);
    bullets.push({x:ship.x+rf[0]*3,y:ship.y+rf[1]*3,z:ship.z+rf[2]*3,
                  px:ship.x,py:ship.y,pz:ship.z,
                  vx:rf[0]*190+ship.vx,vy:rf[1]*190+ship.vy,vz:rf[2]*190+ship.vz,life:0.8});
    blip(950,0.1,'square',0.08);
  }
  for(var i=bullets.length-1;i>=0;i--){
    var b=bullets[i];b.px=b.x;b.py=b.y;b.pz=b.z;
    b.x+=b.vx*dt;b.y+=b.vy*dt;b.z+=b.vz*dt;b.life-=dt;
    if(b.life<=0)bullets.splice(i,1);
  }

  // asteroids
  for(i=asteroids.length-1;i>=0;i--){
    var a=asteroids[i];
    a.x+=a.vx*dt;a.y+=a.vy*dt;a.z+=a.vz*dt;
    a.ang+=a.spd*dt;
    var dx=a.x-ship.x,dy=a.y-ship.y,dz=a.z-ship.z;
    if(dx*dx+dy*dy+dz*dz>600*600){ // recycle far rocks toward player
      var ang2=rnd(0,Math.PI*2),el2=rnd(-1,1),dist2=rnd(300,450);
      a.x=ship.x+Math.sin(ang2)*Math.cos(el2)*dist2;
      a.y=ship.y+Math.sin(el2)*dist2;
      a.z=ship.z+Math.cos(ang2)*Math.cos(el2)*dist2;
    }
    // bullet hits
    for(var j=bullets.length-1;j>=0;j--){
      var bb=bullets[j];
      dx=a.x-bb.x;dy=a.y-bb.y;dz=a.z-bb.z;
      if(dx*dx+dy*dy+dz*dz<(a.r+2)*(a.r+2)){
        bullets.splice(j,1);
        score+=[0,100,50,20][a.size];
        splitRock(a);explode(a.x,a.y,a.z,a.size);boom(a.size);
        asteroids.splice(i,1);updateHud();break;
      }
    }
    // ship hit
    if(asteroids.indexOf(a)>=0&&ship.inv<=0){
      dx=a.x-ship.x;dy=a.y-ship.y;dz=a.z-ship.z;
      var rr=a.r+2.4;
      if(dx*dx+dy*dy+dz*dz<rr*rr){
        explode(ship.x,ship.y,ship.z,3);boom(3);
        lives--;updateHud();
        if(lives<=0){gameOver();return;}
        ship.vx=ship.vy=ship.vz=0;ship.inv=2.5;
      }
    }
  }
  ship.inv-=dt;

  // wave clear
  if(asteroids.length===0){
    waveDelay+=dt;
    if(waveDelay>1.4){wave++;spawnWave();waveDelay=0;updateHud();}
  }

  // particles
  for(i=parts.length-1;i>=0;i--){
    var p=parts[i];p.t+=dt;
    if(p.t>=p.life){parts.splice(i,1);continue;}
    p.x+=p.vx*dt;p.y+=p.vy*dt;p.z+=p.vz*dt;
    p.vx*=Math.pow(0.3,dt);p.vy*=Math.pow(0.3,dt);p.vz*=Math.pow(0.3,dt);
  }
}

/* ---------- render ---------- */
function seg(x1,y1,z1,x2,y2,z2,color,width){
  var v1=viewPt(x1,y1,z1),v2=viewPt(x2,y2,z2);
  if(v1[0]<0.4&&v2[0]<0.4)return;
  if(v1[0]<0.4||v2[0]<0.4){
    var t=(0.4-v1[0])/(v2[0]-v1[0]);
    if(t<0||t>1)return;
    if(v1[0]<0.4){v1=[0.4,v1[1]+(v2[1]-v1[1])*t,v1[2]+(v2[2]-v1[2])*t];}
    else{v2=[0.4,v2[1]+(v1[1]-v2[1])*t,v2[2]+(v1[2]-v2[2])*t];}
  }
  var s1=proj(v1),s2=proj(v2);
  if(!s1||!s2)return;
  ctx.strokeStyle=color;ctx.lineWidth=width;
  ctx.beginPath();ctx.moveTo(s1[0],s1[1]);ctx.lineTo(s2[0],s2[1]);ctx.stroke();
}

function render(){
  updateCamBasis();
  ctx.fillStyle='#02030a';ctx.fillRect(0,0,W,H);
  // nebula glow
  var g=ctx.createRadialGradient(CX,CY,0,CX,CY,Math.max(W,H)*0.7);
  g.addColorStop(0,'rgba(20,10,50,0.35)');g.addColorStop(1,'rgba(0,0,0,0)');
  ctx.fillStyle=g;ctx.fillRect(0,0,W,H);

  // stars
  for(var i=0;i<stars.length;i++){
    var s=stars[i];
    var v=viewPt(s.x+ship.x,s.y+ship.y,s.z+ship.z);
    if(v[0]<1)continue;
    var p=proj(v);if(!p)continue;
    var tw=0.5+0.5*Math.sin(time*2+s.b*40);
    ctx.fillStyle='rgba(180,210,255,'+(0.25+0.6*s.b*tw)+')';
    var sz=Math.max(1,Math.min(3,90/v[0]));
    ctx.fillRect(p[0],p[1],sz,sz);
  }

  // floor grid (wraps around ship)
  var SP=40,fy=ship.y-26;
  var base=Math.round(ship.x/SP)*SP,baseZ=Math.round(ship.z/SP)*SP;
  for(i=-5;i<=5;i++){
    var gx=base+i*SP,gz=baseZ+i*SP;
    seg(gx,fy,ship.z-180,gx,fy,ship.z+180,'rgba(30,90,140,0.35)',1);
    seg(ship.x-180,fy,gz,ship.x+180,fy,gz,'rgba(30,90,140,0.35)',1);
  }

  // collect drawables with depth
  var items=[];
  for(i=0;i<asteroids.length;i++){
    var a=asteroids[i];
    var v=viewPt(a.x,a.y,a.z);
    if(v[0]>1)items.push({d:v[0],type:'a',a:a});
  }
  for(i=0;i<bullets.length;i++){
    var b=bullets[i];
    var vb=viewPt(b.x,b.y,b.z);
    if(vb[0]>0.4)items.push({d:vb[0],type:'b',b:b});
  }
  for(i=0;i<parts.length;i++){
    var pa=parts[i];
    var vp=viewPt(pa.x,pa.y,pa.z);
    if(vp[0]>0.4)items.push({d:vp[0],type:'p',pa:pa});
  }
  // ship
  if(state==='play'&&!(ship.inv>0&&Math.floor(time*10)%2===0))items.push({d:cam.dist,type:'s'});
  items.sort(function(m,n){return n.d-m.d;});

  for(i=0;i<items.length;i++){
    var it=items[i];
    if(it.type==='a'){
      drawRock(it.a,it.d);
    }else if(it.type==='b'){
      var b2=it.b,v1=viewPt(b2.px,b2.py,b2.pz),v2=viewPt(b2.x,b2.y,b2.z);
      var p1=proj(v1),p2=proj(v2);
      if(p1&&p2){
        ctx.strokeStyle='#fff';ctx.lineWidth=Math.max(1,3*FOCAL/it.d/60);
        ctx.beginPath();ctx.moveTo(p1[0],p1[1]);ctx.lineTo(p2[0],p2[1]);ctx.stroke();
        ctx.fillStyle='#8ef';ctx.beginPath();ctx.arc(p2[0],p2[1],Math.max(1.5,4*FOCAL/it.d/60),0,7);ctx.fill();
      }
    }else if(it.type==='p'){
      var pa2=it.pa,vp2=viewPt(pa2.x,pa2.y,pa2.z),pp=proj(vp2);
      if(pp){
        ctx.globalAlpha=1-pa2.t/pa2.life;
        ctx.fillStyle=pa2.c;
        var sz=Math.max(1.5,Math.min(8,300/vp2[0]));
        ctx.fillRect(pp[0]-sz/2,pp[1]-sz/2,sz,sz);
        ctx.globalAlpha=1;
      }
    }else{
      drawShip();
    }
  }

  // speed lines at high velocity
  var sp=Math.sqrt(ship.vx*ship.vx+ship.vy*ship.vy+ship.vz*ship.vz);
  if(sp>38){
    ctx.strokeStyle='rgba(150,220,255,0.25)';ctx.lineWidth=1;
    for(i=0;i<10;i++){
      var a=rnd(0,Math.PI*2),r0=rnd(0.3,0.9)*Math.min(W,H);
      ctx.beginPath();
      ctx.moveTo(CX+Math.cos(a)*r0,CY+Math.sin(a)*r0);
      ctx.lineTo(CX+Math.cos(a)*(r0+rnd(15,50)),CY+Math.sin(a)*(r0+rnd(15,50)));
      ctx.stroke();
    }
  }
}

function drawRock(a,d){
  var c=Math.cos(a.ang),s=Math.sin(a.ang);
  // cheap rotation around axis: rotate verts with rodrigues each frame (small n)
  var col='rgba(150,160,190,'+Math.max(0.25,Math.min(1,140/d))+')';
  ctx.strokeStyle=col;ctx.lineWidth=Math.max(1,Math.min(3,260/d));
  var pv=[];
  for(var i=0;i<a.verts.length;i++){
    var rv=rodrigues(a.verts[i],a.axis,a.ang);
    a.rot[i]=rv;
    var v=viewPt(rv[0]+a.x,rv[1]+a.y,rv[2]+a.z);
    pv.push(v[0]>0.4?proj(v):null);
  }
  ctx.beginPath();
  for(i=0;i<a.edges.length;i++){
    var e=a.edges[i],p1=pv[e[0]],p2=pv[e[1]];
    if(p1&&p2){ctx.moveTo(p1[0],p1[1]);ctx.lineTo(p2[0],p2[1]);}
  }
  ctx.stroke();
  // core dot for depth cue
  var vc=viewPt(a.x,a.y,a.z),pc=proj(vc);
  if(pc){ctx.fillStyle='rgba(90,100,140,'+Math.max(0.15,Math.min(0.6,80/d))+')';
    ctx.beginPath();ctx.arc(pc[0],pc[1],Math.max(1,200/d),0,7);ctx.fill();}
}

function drawShip(){
  // local ship verts -> pitch (X) then yaw (Y)
  var local=[[0,-0.4,3],[-2.1,-0.5,-1.6],[0,0.5,-1],[2.1,-0.5,-1.6]];
  var cp=Math.cos(ship.pitch),sp2=Math.sin(ship.pitch);
  var cyw=Math.cos(ship.yaw),syw=Math.sin(ship.yaw);
  function T(v){
    var y=v[1]*cp+v[2]*sp2, z=-v[1]*sp2+v[2]*cyw*0+(-v[1]*sp2+v[2]*cp); // pitch around X
    // simpler: after pitch: (x, y', z')
    var yy=v[1]*cp+v[2]*sp2, zz=-v[1]*sp2+v[2]*cp;
    // yaw around Y: x' = x cos + z sin ; z' = -x sin + z cos  (matches fwd)
    return [v[0]*cyw+zz*syw, yy, -v[0]*syw+zz*cyw];
  }
  var pv=[];
  for(var i=0;i<local.length;i++){
    var w=T(local[i]);
    var v=viewPt(w[0]+ship.x,w[1]+ship.y,w[2]+ship.z);
    pv.push(v[0]>0.4?proj(v):null);
  }
  if(pv[0]&&pv[1]&&pv[2]&&pv[3]){
    ctx.fillStyle='rgba(70,255,220,0.9)';
    ctx.beginPath();ctx.moveTo(pv[0][0],pv[0][1]);ctx.lineTo(pv[1][0],pv[1][1]);ctx.lineTo(pv[2][0],pv[2][1]);ctx.closePath();ctx.fill();
    ctx.beginPath();ctx.moveTo(pv[0][0],pv[0][1]);ctx.lineTo(pv[3][0],pv[3][1]);ctx.lineTo(pv[2][0],pv[2][1]);ctx.closePath();ctx.fill();
    ctx.strokeStyle='#cffff4';ctx.lineWidth=1.5;
    ctx.beginPath();ctx.moveTo(pv[0][0],pv[0][1]);ctx.lineTo(pv[3][0],pv[3][1]);ctx.stroke();
    // engine flame
    if(keys['KeyW']||keys['ArrowUp']){
      var f=fwd(ship.yaw,ship.pitch);
      var b1=[ship.x-f[0]*1.6+rnd(-0.5,0.5),ship.y-f[1]*1.6,ship.z-f[2]*1.6];
      var b2=[ship.x-f[0]*(3+Math.random()*2),ship.y-f[1]*(3+Math.random()*2)-0.4,ship.z-f[2]*(3+Math.random()*2)];
      seg(b1[0],b1[1],b1[2],b2[0],b2[1],b2[2],'rgba(255,170,60,0.9)',3);
    }
  }
}

/* ---------- main loop ---------- */
var last=performance.now();
function loop(now){
  var dt=Math.min(0.05,(now-last)/1000);last=now;
  if(state==='play')update(dt);
  else{time+=dt;cam.yaw+=dt*0.12;cam.pitch=0.16+Math.sin(time*0.3)*0.05;}
  render();
  requestAnimationFrame(loop);
}
requestAnimationFrame(loop);
})();
