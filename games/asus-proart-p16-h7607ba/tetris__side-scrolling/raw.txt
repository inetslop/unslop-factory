function collides(cells,col,row){
  for(var i=0;i<cells.length;i++){
    var c=col+cells[i][0],r=row+cells[i][1];
    if(c<0||c>=W||r>=H)return true;
    if(r>=0&&grid[r][c])return true;

  return false;

function tryMove(dc,dr){
  var nc=piece.col+dc,nr=piece.row+dr;

  if(!collides(piece.cells,nc,nr)){piece.col=nc;piece.row=nr;return true;}
  return false;

function rotate(){
  var t=piece.cells.map(function(c){return[-c[1],c[0]];});

  t.forEach(function(p){if(p[0]<mx)mx=p[0];if(p[1]<my)my=p[1];});

  }
}
