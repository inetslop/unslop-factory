function tone(freq, dur, type, gain, slideTo){

  try {

  } else if (state === 'PLAY' || state === 'SCORE'){
    // faint shimmer hint: a tiny flicker dot so the field isn't fully silent
    const sh = (Math.sin(time*13) > 0.96) ? 0.12 : 0;
    if (sh > 0){
      ctx.fillStyle = 'rgba(255,240,190,' + sh + ')';
      ctx.beginPath(); ctx.arc(ball.x, ball.y, 3, 0, Math.PI*2); ctx.fill();
    }
  }

  // Paddles
  function drawPaddle(x, y, color, glow){
    ctx.shadowColor = glow; ctx.shadowBlur = 18;
    ctx.fillStyle = color;
    roundRect(x, y - PADDLE_H/2, PADDLE_W, PADDLE_H, 5);
    ctx.fill();
    ctx.shadowBlur = 0;
  }
  drawPaddle(PLAYER_X, player.y, '#3dffc4', 'rgba(61,255,196,0.8)');
  drawPaddle(AI_X, ai.y, '#ff4d88', 'rgba(255,77,136,0.8)');

  // HUD
  ctx.textAlign = 'center';
  ctx.font = 'bold 44px monospace';
  ctx.fillStyle = 'rgba(61,255,196,0.9)';
  ctx.fillText(scoreP, W/2 - 70, 58);
  ctx.fillStyle = 'rgba(255,77,136,0.9)';
  ctx.fillText(scoreA, W/2 + 70, 58);
  ctx.fillStyle = 'rgba(140,170,200,0.5)';
  ctx.font = '12px monospace';
  ctx.fillText('YOU', W/2 - 70, 76);
  ctx.fillText('SHADOW AI', W/2 + 70, 76);

  // Sonar energy bar
  const bw = 180;
  ctx.fillStyle = 'rgba(140,170,200,0.5)';
  ctx.font = '11px monospace';
  ctx.textAlign = 'left';
  ctx.fillText('SONAR [SPACE]', 24, H - 34);
  ctx.strokeStyle = 'rgba(120,255,230,0.5)';
  ctx.strokeRect(24, H - 26, bw, 10);
  const ef = energy / 100;
  ctx.fillStyle = ef >= PING_COST/100 ? 'rgba(120,255,230,0.85)' : 'rgba(120,255,230,0.4)';
  ctx.fillRect(26, H - 24, (bw-4)*ef, 6);

  // Controls hint
  ctx.textAlign = 'center';
  ctx.fillStyle = 'rgba(140,170,200,0.45)';
  ctx.font = '12px monospace';
  ctx.fillText('W/S or ↑/↓ MOVE   ·   SPACE SONAR PING   ·   R RESTART', W/2, H - 14);

  // Flash message
  if (flashT > 0 && state === 'SCORE'){
    ctx.globalAlpha = clamp(flashT / 0.4, 0, 1);
    ctx.fillStyle = flashText.indexOf('YOU') !== -1 ? '#3dffc4' : '#ff4d88';
    ctx.font = 'bold 34px monospace';
    ctx.fillText(flashText, W/2, H/2 - 90);
    ctx.globalAlpha = 1;
  }

  // Scanlines
  ctx.fillStyle = 'rgba(0,0,0,0.10)';
  for (let y = 0; y < H; y += 4) ctx.fillRect(0, y, W, 1);

  ctx.restore();

  // Overlays
  if (state === 'START') drawStart();
  if (state === 'OVER') drawOver();
}

function roundRect(x,y,w,h,r){
  ctx.beginPath();
  ctx.moveTo(x+r,y);
  ctx.arcTo(x+w,y,x+w,y+h,r);
  ctx.arcTo(x+w,y+h,x,y+h,r);
  ctx.arcTo(x,y+h,x,y,r);
  ctx.arcTo(x,y,x+w,y,r);
  ctx.closePath();
}

function panel(){
  ctx.fillStyle = 'rgba(4,10,16,0.82)';
  roundRect(W/2 - 340, H/2 - 170, 680, 340, 14);
  ctx.fill();
  ctx.strokeStyle = 'rgba(120,255,230,0.4)';
  ctx.lineWidth = 2;
  roundRect(W/2 - 340, H/2 - 170, 680, 340, 14);
  ctx.stroke();
}
