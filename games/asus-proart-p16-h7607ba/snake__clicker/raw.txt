function audio(){
  if(!AC){ try{ AC = new (window.AudioContext||window.webkitAudioContext)(); }catch(e){} }

function beep(freq, dur, type, vol, slideTo){
  if(!AC) return;
  const t = AC.currentTime;
  const o = AC.createOscillator(), g = AC.createGain();
  o.type = type || 'square';
  o.frequency.setValueAtTime(freq, t);
  if(slideTo) o.frequency.exponentialRampToValueAtTime(Math.max(30,slideTo), t+dur);
  g.gain.setValueAtTime(vol||0.12, t);
  g.gain.exponentialRampToValueAtTime(0.0001, t+dur);
  o.connect(g); g.connect(AC.destination);
  o.start(t); o.stop(t+dur+0.02);
}

  return false;
}

// ---------- pathfinding ----------
function nextStep(tx, ty){
  // BFS from head to target avoiding body (tail will move unless growing — keep it simple: avoid whole body except tail cell)
  const key = (x,y)=> y*COLS+x;
  const startK = key(snake[0].x, snake[0].y);
  const goalK = key(tx,ty);
  const prev = new Map();
  const q = [startK];
  const seen = new Set([startK]);
  const head = snake[0];
  while(q.length){
    const cur = q.shift();
    if(cur === goalK) break;
    const cx = cur % COLS, cy = (cur/COLS)|0;
    for(const d of [{x:1,y:0},{x:-1,y:0},{x:0,y:1},{x:0,y:-1}]){
      const nx=cx+d.x, ny=cy+d.y;
      if(nx<0||ny<0||nx>=COLS||ny>=ROWS) continue;
      const nk = key(nx,ny);
      if(seen.has(nk)) continue;
      // block body except tail (tail vacates when not eating into it — conservative: block all but last segment)
      const isBody = snake.some((s,i)=> i < snake.length-1 && s.x===nx && s.y===ny);
      if(isBody) continue;
      seen.add(nk); prev.set(nk, cur);
      q.push(nk);
    }
  }
  if(!prev.has(goalK)) return null;
  // walk back to first step
  let k = goalK;
  while(prev.get(k) !== startK) k = prev.get(k);
  return { x:(k%COLS)-head.x, y:((k/COLS)|0)-head.y };
}

function safeMoves(){
  const head = snake[0];
  const body = new Set(snake.slice(0,-1).map(s=>s.y*COLS+s.x));
  const res = [];
  for(const d of [{x:1,y:0},{x:-1,y:0},{x:0,y:1},{x:0,y:-1}]){
    const nx=head.x+d.x, ny=head.y+d.y;
    if(nx<0||ny<0||nx>=COLS||ny>=ROWS) continue;
    if(body.has(ny*COLS+nx)) continue;
    res.push(d);
  }
  return res;
}

function tick(){
  const head = snake[0];
  let step = null;
  // target: nearest food, else last click target
  let goal = null;
  if(food.length){
    let bd = Infinity;
    for(const f of food){
      const d = Math.abs(f.x-head.x)+Math.abs(f.y-head.y);
      if(d<bd){ bd=d; goal=f; }
    }
  } else if(targetCell && !cellOccupied(targetCell.x,targetCell.y)){
    goal = targetCell;
  }
  if(goal) step = nextStep(goal.x, goal.y);

  if(!step){
    const moves = safeMoves();
    if(!moves.length){ gameOver(); return; }
    // prefer continuing direction
    let pick = moves.find(m=>m.x===dir.x&&m.y===dir.y) || moves[(Math.random()*moves.length)|0];
    step = pick;
  }
  dir = step;
  const nx = head.x+step.x, ny = head.y+step.y;

  // walls
  if(nx<0||ny<0||nx>=COLS||ny>=ROWS){ gameOver(); return; }

  // eating?
  let ate = null;
  for(let i=0;i<food.length;i++){
    if(food[i].x===nx && food[i].y===ny){ ate = food[i]; food.splice(i,1); break; }
  }

  // self collision (tail moves if not eating)
  const limit = ate ? snake.length : snake.length-1;
  for(let i=0;i<limit;i++){
    if(snake[i].x===nx && snake[i].y===ny){ gameOver(); return; }
  }

  snake.unshift({x:nx,y:ny});
  if(ate){
    const base = ate.type==='gold' ? 25 : 10;
    score += base;
    spawnQueue++;
    addBurst(nx, ny, ate.type);
    beep(ate.type==='gold'?880:640, 0.09, 'square', 0.12, ate.type==='gold'?1320:960);
  } else {
    snake.pop();
  }

  // respawn green pellets (keep at least 3)
  if(spawnQueue>0 && spawnTimer<=0){
    const c = randFreeCell();
    if(c){ food.push({x:c.x,y:c.y,type:'green'}); spawnQueue--; }
    spawnTimer = 0.9;
  }
  updateHUD();
}

function addBurst(cx, cy, type){
  const px = ox + (cx+0.5)*cell, py = oy + (cy+0.5)*cell;
  const col = type==='gold' ? '#ffd75e' : '#7dff9e';
  for(let i=0;i<14;i++){
    const a = Math.random()*Math.PI*2, sp = 60+Math.random()*160;
    particles.push({x:px,y:py,vx:Math.cos(a)*sp,vy:Math.sin(a)*sp,life:0.55,max:0.55,col:col,r:2+Math.random()*3});
  }
  particles.push({text:'+'+(type==='gold'?25:10), x:px, y:py-8, vy:-46, life:0.9, max:0.9, col:col});
}

    particles.push({x:px,y:py,vx:Math.cos(a)*sp,vy:Math.sin(a)*sp,life:0.35,max:0.35,col:'#8fb8ff',r:1.5+Math.random()*2});
  }
  particles.push({text:'+'+pts, x:px, y:py-6, vy:-40, life:0.7, max:0.7, col:'#ffd75e'});
}

function cellFromEvent(e){
  const r = cv.getBoundingClientRect();
  const mx = e.clientX - r.left, my = e.clientY - r.top;
  const cx = Math.floor((mx-ox)/cell), cy = Math.floor((my-oy)/cell);
  return {cx:cx, cy:cy, px:mx, py:my, inside: cx>=0&&cy>=0&&cx<COLS&&cy<ROWS};
}

  lastClickT = now;
  const pts = 2 + Math.min(combo,10);
  score += pts;

  // steer toward click
  if(!cellOccupied(p.cx,p.cy)) targetCell = {x:p.cx, y:p.cy};

  // drop a golden pellet on empty cells (clicker mechanic)
  if(!cellOccupied(p.cx,p.cy) && food.length < 7){
    food.push({x:p.cx, y:p.cy, type:'gold'});
  }
  addClickFX(p.px, p.py, pts);
  beep(300+Math.min(combo,10)*40, 0.05, 'triangle', 0.08);
  updateHUD();
});

  for(const f of food){
    const fx = ox+(f.x+0.5)*cell, fy = oy+(f.y+0.5)*cell;
    const wob = 0.85 + 0.15*Math.sin(performance.now()/200 + f.x*3 + f.y);
    ctx.save();
    if(f.type==='gold'){
      ctx.shadowColor='#ffd75e'; ctx.shadowBlur=14;
      ctx.fillStyle='#ffd75e';
      ctx.beginPath();
      for(let i=0;i<8;i++){
        const a=i/8*Math.PI*2, rr=(i%2? cell*0.18: cell*0.36)*wob;
        const pxp=fx+Math.cos(a)*rr, pyp=fy+Math.sin(a)*rr;
        i?ctx.lineTo(pxp,pyp):ctx.moveTo(pxp,pyp);
      }
      ctx.closePath(); ctx.fill();
    } else {
      ctx.shadowColor='#7dff9e'; ctx.shadowBlur=10;
      ctx.fillStyle='#3fe678';
      ctx.beginPath(); ctx.arc(fx,fy,cell*0.3*wob,0,Math.PI*2); ctx.fill();
      ctx.shadowBlur=0; ctx.fillStyle='rgba(255,255,255,0.5)';
      ctx.beginPath(); ctx.arc(fx-cell*0.08,fy-cell*0.1,cell*0.07,0,Math.PI*2); ctx.fill();
    }
    ctx.restore();
  }

  // snake
  for(let i=snake.length-1;i>=0;i--){
    const s = snake[i];
    const px = ox+s.x*cell, py = oy+s.y*cell;
    const t = i/snake.length;
    const g = Math.round(217 - t*120);
    ctx.fillStyle = i===0 ? '#a5ffb8' : 'rgb('+(47-t*30|0)+','+g+','+(107-t*60|0)+')';
    const pad = i===0?1:2.5;
    roundRect(px+pad, py+pad, cell-pad*2, cell-pad*2, 6);
    ctx.fill();
  }
  // eyes on head
  const h = snake[0];
  const hx = ox+(h.x+0.5)*cell, hy = oy+(h.y+0.5)*cell;
  ctx.fillStyle='#0a1428';
  const ex = dir.x*cell*0.16, ey = dir.y*cell*0.16;
  const pxp = -dir.y*cell*0.16, pyp = dir.x*cell*0.16;
  ctx.beginPath(); ctx.arc(hx+ex+pxp, hy+ey+pyp, cell*0.09, 0, Math.PI*2); ctx.fill();
  ctx.beginPath(); ctx.arc(hx+ex-pxp, hy+ey-pyp, cell*0.09, 0, Math.PI*2); ctx.fill();

  // particles
  for(const p of particles){
    const a = Math.max(0, p.life/p.max);
    if(p.text){
      ctx.font = 'bold '+Math.round(cell*0.55)+'px Segoe UI, Arial';
      ctx.fillStyle = p.col;
      ctx.globalAlpha = a;
      ctx.textAlign='center';
      ctx.fillText(p.text, p.x, p.y);
      ctx.globalAlpha = 1;
    } else {
      ctx.globalAlpha = a;
      ctx.fillStyle = p.col;
      ctx.beginPath(); ctx.arc(p.x,p.y,p.r*a+0.5,0,Math.PI*2); ctx.fill();
      ctx.globalAlpha = 1;
    }
  }
}

function roundRect(x,y,w,h,r){
  r = Math.min(r, w/2, h/2);
  ctx.beginPath();
  ctx.moveTo(x+r,y);
  ctx.arcTo(x+w,y,x+w,y+h,r);
  ctx.arcTo(x+w,y+h,x,y+h,r);
  ctx.arcTo(x,y+h,x,y,r);
  ctx.arcTo(x,y,x+w,y,r);
  ctx.closePath();
}

// ---------- main loop ----------
function loop(t){
  requestAnimationFrame(loop);
  if(!lastT) lastT = t;
  let dt = (t-lastT)/1000; lastT = t;
  if(dt>0.25) dt=0.25;

  if(state==='play'){
    if(spawnTimer>0) spawnTimer -= dt;
    const interval = Math.max(0.085, 0.16 - snake.length*0.003);
    acc += dt;
    while(acc >= interval && state==='play'){ acc -= interval; tick(); }
  }

  // particles update
  for(let i=particles.length-1;i>=0;i--){
    const p = particles[i];
    p.life -= dt;
    if(p.life<=0){ particles.splice(i,1); continue; }
    if(p.vx!==undefined){ p.x+=p.vx*dt; p.y+=p.vy*dt; p.vx*=0.92; p.vy*=0.92; }
    else { p.y += (p.vy||-40)*dt; }
  }

  draw();
}
requestAnimationFrame(loop);
})();
