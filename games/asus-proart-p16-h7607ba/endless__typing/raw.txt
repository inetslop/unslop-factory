```html
<!DOCTYPE html>
<html lang="en">
<head>
<meta charset="UTF-8">
<title>Endless Typing</title>
<style>
  html, body {
    margin: 0; padding: 0; width: 100%; height: 100%;
    overflow: hidden; background: #0b0e1a;
    font-family: 'Courier New', monospace;
  }
  #game { display: block; width: 100vw; height: 100vh; }
</style>
</head>
<body>
<canvas id="game"></canvas>
<script>
(function(){
  "use strict";

  var canvas = document.getElementById('game');
  var ctx = canvas.getContext('2d');
  var W = 0, H = 0;

  function resize(){
    W = window.innerWidth;
    H = window.innerHeight;
    canvas.width = W;
    canvas.height = H;
  }
  window.addEventListener('resize', resize);
  resize();

  // ---------- Audio (Web Audio API, starts after first input) ----------
  var actx = null;
  function audio(){
    if(!actx){
      try { actx = new (window.AudioContext || window.webkitAudioContext)(); } catch(e){ actx = null; }
    }
    if(actx && actx.state === 'suspended'){ actx.resume(); }
    return actx;
  }
  function beep(freq, dur, type, vol, slideTo){
    var a = audio();
    if(!a) return;
    try {
      var o = a.createOscillator(), g = a.createGain();
      o.type = type || 'square';
      o.frequency.setValueAtTime(freq, a.currentTime);
      if(slideTo) o.frequency.exponentialRampToValueAtTime(slideTo, a.currentTime + dur);
      g.gain.setValueAtTime(vol || 0.12, a.currentTime);
      g.gain.exponentialRampToValueAtTime(0.0001, a.currentTime + dur);
      o.connect(g); g.connect(a.destination);
      o.start(); o.stop(a.currentTime + dur + 0.02);
    } catch(e){}
  }

  // ---------- Word pool ----------
  var WORDS = ("the and for are but not you all can had her was one our out day get has him his how man new now old see two way who boy did its let put say she too use ten any like long first little then them three time very small great come down even over here last find give made back some take only them year know grow still most other much need must when such well also away each else ever full head high just keep left life look make many more near open once play side soon stop take told turn want word work upon with into from that this they will what your were does done both half have been come went made give know find take came went said seen gave took held kept".split(" "));
  // dedupe
  var pool = [];
  (function(){ var s = {}; WORDS.forEach(function(w){ if(!s[w]) { s[w]=1; pool.push(w); } }); })();

  function randWord(prev){
    var w;
    do { w = pool[Math.floor(Math.random()*pool.length)]; } while(w === prev && pool.length > 1);
    return w;
  }

  // ---------- Game state ----------
  var state = 'start'; // start | playing | gameover
  var word, score, lives, level, speed, elapsed, best, prevWord, spawnTimer, shake, particles, stars;

  try { best = parseInt(localStorage.getItem('endlessTypingBest') || '0', 10) || 0; } catch(e){ best = 0; }

  function initStars(){
    stars = [];
    for(var i=0;i<70;i++){
      stars.push({ x: Math.random()*W, y: Math.random()*H, s: Math.random()*1.6+0.4, v: Math.random()*30+15 });
    }
  }
  initStars();

  function spawnWord(){
    var lane = Math.floor(Math.random()*3);
    word = {
      text: randWord(prevWord),
      progress: 0,
      lane: lane,
      x: W + 40,
      flashUntil: 0,
      popScale: 1.25
    };
    prevWord = word.text;
  }

  function reset(){
    score = 0; lives = 3; level = 1; speed = 70; elapsed = 0; shake = 0;
    particles = [];
    spawnWord();
  }

  // ---------- Input ----------
  window.addEventListener('keydown', function(e){
    audio(); // first input unlocks audio

    if(state === 'start'){
      if(e.key === ' ' || e.key === 'Enter' || e.key === 'r' || e.key === 'R'){
        reset(); state = 'playing';
        beep(440, 0.1, 'square', 0.1);
      }
      return;
    }
    if(state === 'gameover'){
      if(e.key === 'r' || e.key === 'R' || e.key === ' ' || e.key === 'Enter'){
        reset(); state = 'playing';
        beep(440, 0.1, 'square', 0.1);
      }
      return;
    }

    // playing
    if(e.key.length === 1 && /[a-zA-Z]/.test(e.key)){
      var ch = e.key.toLowerCase();
      if(ch === word.text[word.progress]){
        word.progress++;
        beep(420 + word.progress*55, 0.06, 'square', 0.08);
        if(word.progress >= word.text.length){
          // word complete!
          var gained = 10 * level;
          score += gained;
          levelUpCheck();
          burstParticles(word.x + ctx.measureText(word.text).width*0.5, laneY(word.lane), '#4dff88');
          beep(660, 0.09, 'triangle', 0.12);
          setTimeout(function(){ beep(880, 0.12, 'triangle', 0.12); }, 70);
          spawnWord();
        }
      } else {
        // wrong key: reset progress + flash
        word.progress = 0;
        word.flashUntil = performance.now() + 250;
        shake = 8;
        beep(130, 0.18, 'sawtooth', 0.14, 70);
      }
    }
    if(e.key === 'r' || e.key === 'R'){
      reset(); state = 'playing';
    }
  });

  canvas.addEventListener('mousedown', function(){
    audio();
    if(state === 'start' || state === 'gameover'){
      reset(); state = 'playing';
      beep(440, 0.1, 'square', 0.1);
    }
  });

  function laneY(lane){
    return H * (0.32 + lane * 0.23);
  }

  function levelUpCheck(){
    var nl = Math.floor(score / 100) + 1;
    if(nl > level){
      level = nl;
      speed = 70 + (level-1)*18;
      beep(523, 0.1, 'triangle', 0.12);
      setTimeout(function(){ beep(659, 0.1, 'triangle', 0.12); }, 90);
      setTimeout(function(){ beep(784, 0.16, 'triangle', 0.12); }, 180);
    }
  }

  function loseLife(){
    lives--;
    shake = 14;
    burstParticles(word.x + 30, laneY(word.lane), '#ff5566');
    beep(220, 0.25, 'sawtooth', 0.16, 90);
    if(lives <= 0){
      state = 'gameover';
      if(score > best){
        best = score;
        try { localStorage.setItem('endlessTypingBest', String(best)); } catch(e){}
      }
      beep(160, 0.5, 'sawtooth', 0.18, 50);
    } else {
      spawnWord();
    }
  }

  function burstParticles(x, y, color){
    for(var i=0;i<24;i++){
      var a = Math.random()*Math.PI*2, v = Math.random()*180+60;
      particles.push({ x:x, y:y, vx:Math.cos(a)*v, vy:Math.sin(a)*v, life:0.7, max:0.7, color:color });
    }
  }

  // ---------- Update & draw ----------
  var last = performance.now();

  function update(dt){
    // stars always drift
    for(var i=0;i<stars.length;i++){
      stars[i].x -= stars[i].v * dt * (state==='playing' ? 1 + speed/200 : 0.5);
      if(stars[i].x < -3){ stars[i].x = W+3; stars[i].y = Math.random()*H; }
    }
    // particles
    for(var j=particles.length-1;j>=0;j--){
      var p = particles[j];
      p.life -= dt;
      if(p.life <= 0){ particles.splice(j,1); continue; }
      p.x += p.vx*dt; p.y += p.vy*dt; p.vy += 300*dt;
    }
    shake = Math.max(0, shake - dt*40);

    if(state !== 'playing') return;
    elapsed += dt;
    speed += dt * 1.2; // gentle ramp even within a level
    word.x -= speed * dt;
    word.popScale += (1 - word.popScale) * Math.min(1, dt*8);

    var dangerX = W * 0.14;
    if(word.x < dangerX){
      loseLife();
    }
  }

  function drawBackground(){
    var g = ctx.createLinearGradient(0,0,0,H);
    g.addColorStop(0,'#0b0e1a');
    g.addColorStop(1,'#141a35');
    ctx.fillStyle = g;
    ctx.fillRect(0,0,W,H);

    ctx.fillStyle = 'rgba(255,255,255,0.6)';
    for(var i=0;i<stars.length;i++){
      var s = stars[i];
      ctx.globalAlpha = 0.2 + s.s*0.3;
      ctx.fillRect(s.x, s.y, s.s, s.s);
    }
    ctx.globalAlpha = 1;

    // danger zone
    var dangerX = W * 0.14;
    var dg = ctx.createLinearGradient(0,0,dangerX,0);
    dg.addColorStop(0,'rgba(255,60,80,0.22)');
    dg.addColorStop(1,'rgba(255,60,80,0)');
    ctx.fillStyle = dg;
    ctx.fillRect(0,0,dangerX,H);
    ctx.strokeStyle = 'rgba(255,80,100,0.8)';
    ctx.lineWidth = 3;
    ctx.setLineDash([12,10]);
    ctx.beginPath();
    ctx.moveTo(dangerX, 0); ctx.lineTo(dangerX, H);
    ctx.stroke();
    ctx.setLineDash([]);
  }

  function drawHUD(){
    ctx.textAlign = 'left';
    ctx.textBaseline = 'top';
    ctx.font = 'bold 22px "Courier New", monospace';
    ctx.fillStyle = '#8fd3ff';
    ctx.fillText('SCORE  ' + score, 18, 16);
    ctx.fillText('LEVEL  ' + level, 18, 46);
    ctx.fillStyle = '#ffd27a';
    ctx.fillText('BEST   ' + Math.max(best, score), 18, 76);

    // lives as hearts
    for(var i=0;i<3;i++){
      drawHeart(W-34-i*36, 24, i < lives ? '#ff5577' : 'rgba(255,255,255,0.15)');
    }

    // speed bar
    var maxSpeed = 70 + (level)*18;
    ctx.fillStyle = 'rgba(255,255,255,0.15)';
    ctx.fillRect(W-160, 64, 142, 8);
    ctx.fillStyle = '#4dff88';
    ctx.fillRect(W-160, 64, Math.min(142, (speed/ maxSpeed)*142), 8);
    ctx.font = '12px "Courier New", monospace';
    ctx.fillStyle = 'rgba(255,255,255,0.5)';
    ctx.textAlign = 'right';
    ctx.fillText('SPEED', W-18, 78);
  }

  function drawHeart(x,y,color){
    ctx.save();
    ctx.translate(x,y);
    ctx.fillStyle = color;
    ctx.beginPath();
    ctx.moveTo(0,6);
    ctx.bezierCurveTo(-12,-6,-4,-14,0,-8);
    ctx.bezierCurveTo(4,-14,12,-6,0,6);
    ctx.fill();
    ctx.restore();
  }

  function drawWord(){
    if(!word) return;
    var y = laneY(word.lane);
    var now = performance.now();
    var flashing = now < word.flashUntil;
    var fs = Math.min(56, W*0.07);
    ctx.font = 'bold ' + fs + 'px "Courier New", monospace';
    ctx.textBaseline = 'middle';
    ctx.textAlign = 'left';

    ctx.save();
    ctx.translate(word.x, y);
    ctx.scale(word.popScale, word.popScale);

    var cx = 0;
    for(var i=0;i<word.text.length;i++){
      var ch = word.text[i];
      if(i < word.progress){
        ctx.fillStyle = flashing ? '#ff9aa8' : '#4dff88';
        ctx.shadowColor = 'rgba(77,255,136,0.7)';
        ctx.shadowBlur = 14;
      } else if(i === word.progress){
        // current char highlight
        ctx.fillStyle = flashing ? '#ff5566' : '#ffffff';
        ctx.shadowColor = 'rgba(255,255,255,0.9)';
        ctx.shadowBlur = 18;
      } else {
        ctx.fillStyle = flashing ? '#ff9aa8' : 'rgba(230,240,255,0.75)';
        ctx.shadowBlur = 0;
      }
      ctx.fillText(ch, cx, 0);
      cx += ctx.measureText(ch).width;
    }
    ctx.restore();

    // caret under current char
    if(state === 'playing'){
      var off = 0;
      for(var k=0;k<word.progress;k++) off += ctx.measureText(word.text[k]).width;
      var blink = (Math.floor(now/350) % 2 === 0);
      if(blink){
        ctx.fillStyle = '#ffffff';
        ctx.fillRect(word.x + off*word.popScale, y + fs*0.62, Math.min(56,W*0.07)*0.18, fs*0.34);
      }
    }
  }

  function drawParticles(){
    for(var i=0;i<particles.length;i++){
      var p = particles[i];
      ctx.globalAlpha = Math.max(0, p.life/p.max);
      ctx.fillStyle = p.color;
      ctx.fillRect(p.x-3, p.y-3, 6, 6);
    }
    ctx.globalAlpha = 1;
  }

  function drawCenterPanel(title, lines, titleColor){
    ctx.textAlign = 'center';
    ctx.textBaseline = 'middle';
    var fs = Math.min(72, W*0.09);
    ctx.font = 'bold ' + fs + 'px "Courier New", monospace';
    ctx.fillStyle = titleColor || '#8fd3ff';
    ctx.shadowColor = titleColor || '#8fd3ff';
    ctx.shadowBlur = 24;
    ctx.fillText(title, W/2, H*0.26);
    ctx.shadowBlur = 0;

    var lf = Math.min(24, W*0.035);
    ctx.font = lf + 'px "Courier New", monospace';
    var y0 = H*0.42;
    for(var i=0;i<lines.length;i++){
      ctx.fillStyle = lines[i].c || 'rgba(230,240,255,0.85)';
      ctx.fillText(lines[i].t, W/2, y0 + i*lf*1.6);
    }

    var pulse = 0.6 + 0.4*Math.sin(performance.now()/300);
    ctx.globalAlpha = pulse;
    ctx.font = 'bold ' + (lf+4) + 'px "Courier New", monospace';
    ctx.fillStyle = '#ffd27a';
    ctx.fillText(lines[lines.length-1].t || '', W/2, H*0.78);
    ctx.globalAlpha = 1;
  }

  function draw(){
    ctx.save();
    if(shake > 0){
      ctx.translate((Math.random()-0.5)*shake, (Math.random()-0.5)*shake);
    }
    drawBackground();
    drawParticles();

    if(state === 'playing'){
      drawWord();
      drawHUD();
    } else if(state === 'start'){
      drawCenterPanel('ENDLESS TYPING', [
        {t:'Words fly in from the right.'},
        {t:'Type each word BEFORE it hits the red line!'},
        {t:''},
        {t:'Wrong key  → progress resets'},
        {t:'Word lost  → you lose a heart (3 total)'},
        {t:'Speed increases as your score grows'},
        {t:'R — restart anytime', c:'#ffd27a'}
      ]);
      if(best > 0){
        ctx.textAlign='center';
        ctx.font = 'bold 24px "Courier New", monospace';
        ctx.fillStyle = '#ffd27a';
        ctx.fillText('BEST SCORE: ' + best, W/2, H*0.9);
      }
    } else if(state === 'gameover'){
      drawCenterPanel('GAME OVER', [
        {t:'Score: ' + score, c:'#4dff88'},
        {t:'Level reached: ' + level},
        {t:''},
        {t:'Best: ' + Math.max(best, score), c:'#ffd27a'}
      ]);
    }
    ctx.restore();
  }

  function frame(now){
    var dt = Math.min(0.05, (now - last)/1000);
    last = now;
    update(dt);
    draw();
    requestAnimationFrame(frame);
  }
  requestAnimationFrame(frame);
})();
</script>
</body>
</html>
```
