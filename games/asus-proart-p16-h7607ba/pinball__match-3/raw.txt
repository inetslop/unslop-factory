  if(state==='aim'){
    if(keys.ArrowLeft)ang-=2.3*dt;
    if(keys.ArrowRight)ang+=2.3*dt;
    ang=clamp(ang,-1,1);
  }

  if(state==='fly'&&ball){
    flightTime+=dt;
    trail.push({x:ball.x,y:ball.y});if(trail.length>14)trail.shift();
    const steps=3,sdt=dt/steps;
    for(let s=0;s<steps;s++){
      ball.vy+=GRAV*sdt;
      let sp=Math.hypot(ball.vx,ball.vy);
      if(sp>MAXSP){ball.vx*=MAXSP/sp;ball.vy*=MAXSP/sp;sp=MAXSP;}
      ball.x+=ball.vx*sdt;ball.y+=ball.vy*sdt;

      // walls (left/right/top)
      if(ball.x<BALL_R){ball.x=BALL_R;if(ball.vx<0){ball.vx=-ball.vx;sBounce();}}
      if(ball.x>COLS-BALL_R){ball.x=COLS-BALL_R;if(ball.vx>0){ball.vx=-ball.vx;sBounce();}}
      if(ball.y<BALL_R&&ball.vy<0){ball.y=BALL_R;ball.vy=-ball.vy;sBounce();}

      // pegs
      for(const p of PEGS){
        const dx=ball.x-p.x,dy=ball.y-p.y,d2=dx*dx+dy*dy,rrad=BALL_R+p.r;
        if(d2<rrad*rrad&&d2>1e-6){
          const d=Math.sqrt(d2),nx=dx/d,ny=dy/d;
          ball.x=p.x+nx*rrad;ball.y=p.y+ny*rrad;
          const dot=ball.vx*nx+ball.vy*ny;
          if(dot<0){ball.vx-=2*dot*nx;ball.vy-=2*dot*ny;}
          const s2=Math.hypot(ball.vx,ball.vy)||1,ns=Math.min(MAXSP,Math.max(s2*1.06,7));
          ball.vx*=ns/s2;ball.vy*=ns/s2;
          p.bump=1;
          if(p.cool<=0){p.cool=0.3;score+=5;flightScore+=5;if(score>best)best=score;sPeg();
            floaters.push({x:p.x,y:p.y-0.35,txt:'+5',life:0.8,color:'#9fd0ff',big:false});}
        }
      }

      // orbs
      for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++){
        const o=grid[r][c];if(!o)continue;
        const ox=c+0.5,oy=o.y+0.5,dx=ball.x-ox,dy=ball.y-oy,d2=dx*dx+dy*dy,rrad=BALL_R+ORB_R;
        if(d2<rrad*rrad&&d2>1e-6){
          const d=Math.sqrt(d2),nx=dx/d,ny=dy/d;
          ball.x=ox+nx*rrad;ball.y=oy+ny*rrad;
          const dot=ball.vx*nx+ball.vy*ny;
          if(dot<0){
            ball.vx-=2*dot*nx;ball.vy-=2*dot*ny;
            ball.vx*=0.99;ball.vy*=0.99;
            sBounce();
          }
          if(o.color===ballColor)tryPop(r,c);
        }
      }
    }
    if(ball.y>ROWS+0.6||flightTime>15)endFlight();
  }
}

function drawOrb(px,py,r,color){
  const g=ctx.createRadialGradient(px-r*0.35,py-r*0.42,r*0.12,px,py,r);
  g.addColorStop(0,shade(color,0.75));
  g.addColorStop(0.55,color);
  g.addColorStop(1,shade(color,-0.4));
  ctx.fillStyle=g;
  ctx.beginPath();ctx.arc(px,py,r,0,Math.PI*2);ctx.fill();
  ctx.fillStyle='rgba(255,255,255,0.85)';
  ctx.beginPath();ctx.ellipse(px-r*0.32,py-r*0.45,r*0.22,r*0.13,-0.6,0,Math.PI*2);ctx.fill();
}
function drawBall(px,py,r,color){
  const g=ctx.createRadialGradient(px-r*0.35,py-r*0.42,r*0.1,px,py,r);
  g.addColorStop(0,'#ffffff');g.addColorStop(0.4,'#d9e2f2');g.addColorStop(1,'#7a869c');
  ctx.fillStyle=g;
  ctx.beginPath();ctx.arc(px,py,r,0,Math.PI*2);ctx.fill();
  ctx.lineWidth=Math.max(2,r*0.35);ctx.strokeStyle=color;
  ctx.beginPath();ctx.arc(px,py,r*0.68,0,Math.PI*2);ctx.stroke();
  drawOrb(px,py,r*0.4,color);
}
function draw(){

  }

    ctx.fillStyle='#ffd23f';

}

**How to play:** Fire a pinball from the top of the field with **SPACE/click**. Aim with **← →** (or your mouse), switch your ball's color with **↑ ↓**, and restart anytime with **R**. When the silver pinball strikes orbs that match its color, any connected cluster of 3 or more pops for points — chain multiple pops in one flight for combo multipliers. Bouncing off the silver pegs scores bonus points. If a flight ends without popping anything, you lose a life; three misses and it's game over.
