Segment collision function:

function collideSeg(b, s){ // b has x,y,vx,vy,r
 const dx=s.x2-s.x1, dy=s.y2-s.y1;
 const L2=dx*dx+dy*dy;
 let t=((b.x-s.x1)*dx+(b.y-s.y1)*dy)/L2;
 t=Math.max(0,Math.min(1,t));
 const cx=s.x1+t*dx, cy=s.y1+t*dy;
 let nx=b.x-cx, ny=b.y-cy;
 const d2=nx*nx+ny*ny;
 if(d2 < b.r*b.r && d2>0.0001){
   const d=Math.sqrt(d2); nx/=d; ny/=d;
   b.x=cx+nx*b.r; b.y=cy+ny*b.r;
   const vn=b.vx*nx+b.vy*ny;
   if(vn<0){ b.vx-=(1+e)*vn*nx; b.vy-=(1+e)*vn*ny; }
   return true;
 }
 return false;
}

function segCollide(b,e){
  for(var i=0;i<segs.length;i++){
    var s=segs[i],dx=s.x2-s.x1,dy=s.y2-s.y1,L2=dx*dx+dy*dy;
    var tt=((b.x-s.x1)*dx+(b.y-s.y1)*dy)/L2;tt=Math.max(0,Math.min(1,tt));
    var cx=s.x1+tt*dx,cy=s.y1+tt*dy;
    var nx=b.x-cx,ny=b.y-cy,d2=nx*nx+ny*ny;
    if(d2<b.r*b.r&&d2>1e-6){
      var d=Math.sqrt(d2);nx/=d;ny/=d;
      b.x=cx+nx*b.r;b.y=cy+ny*b.r;
      var vn=b.vx*nx+b.vy*ny;
      if(vn<0){b.vx-=(1+e)*vn*nx;b.vy-=(1+e)*vn*ny;}
      return true;
    }
  }
  return false;
}

}

  }else if(r<0.7){ // ring
    var n2=8+Math.min(4,wave),off=Math.random()*6.28;
    for(var k2=0;k2<n2;k2++)spawnB(e.x,e.y,k2/n2*6.283+off,105+wave*5,320);
  }else{ // downward fan
    for(var k3=-2;k3<=2;k3++)spawnB(e.x,e.y,Math.PI/2+k3*0.28,140+wave*4,190);
  }
  if(wave>=2&&Math.random()<0.12){e.spiral=2.5;e.st=0;}
}

/* ---------- update ---------- */
function update(dt){
  t+=dt;
  if(bannerT>0)bannerT-=dt;
  if(flashT>0)flashT-=dt;
  shake*=Math.max(0,1-dt*7);if(shake<0.2)shake=0;
  for(var bI=0;bI<bumpers.length;bI++)if(bumpers[bI].f>0)bumpers[bI].f=Math.max(0,bumpers[bI].f-dt*4);

  // flippers
  for(var fI=0;fI<flips.length;fI++){
    var f=flips[i=fI],on=f.key==='L'?keys.L:keys.R;
    if(on&&!f.wasOn)sFlip();
    f.wasOn=on;
    var target=on?f.actA:f.restA,sp=on?26:13,prev=f.a;
    if(f.a<target)f.a=Math.min(target,f.a+sp*dt);else f.a=Math.max(target,f.a-sp*dt);
    f.w=(f.a-prev)/Math.max(dt,0.0001);
  }

  if(state!=='play')return;

  if(charging&&ball.ready&&!ball.dead)charge=Math.min(1,charge+dt/0.9);

  if(ball.dead){respawnT-=dt;if(respawnT<=0)resetBall();}

  // ball physics
  if(!ball.ready&&!ball.dead){
    var steps=4,sdt=dt/steps;
    for(var sI=0;sI<steps;sI++){
      ball.vy+=G*sdt;
      ball.x+=ball.vx*sdt;ball.y+=ball.vy*sdt;
      segCollide(ball,0.85);
      // bumpers
      for(var bp=0;bp<bumpers.length;bp++){
        var u=bumpers[bp],dx=ball.x-u.x,dy=ball.y-u.y,R=ball.r+u.r,d2=dx*dx+dy*dy;
        if(d2<R*R&&d2>1e-6){
          var d=Math.sqrt(d2),nx=dx/d,ny=dy/d;
          ball.x=u.x+nx*R;ball.y=u.y+ny*R;
          var vn=ball.vx*nx+ball.vy*ny;
          if(vn<0){ball.vx-=1.9*vn*nx;ball.vy-=1.9*vn*ny;}
          ball.vx+=nx*140;ball.vy+=ny*140;
          u.f=1;score+=100;sBumper();shake=Math.max(shake,2);
        }
      }
      // flippers
      var fc=flipCapsule(ball);
      if(fc){
        var Rr=ball.r+fc.f.fr;
        ball.x=fc.cx+fc.nx*Rr;ball.y=fc.cy+fc.ny*Rr;
        var rx=fc.cx-fc.f.px,ry=fc.cy-fc.f.py;
        var svx=fc.f.w*(-ry),svy=fc.f.w*rx;
        var rvx=ball.vx-svx,rvy=ball.vy-svy,vn2=rvx*fc.nx+rvy*fc.ny;
        if(vn2<0){ball.vx-=1.6*vn2*fc.nx;ball.vy-=1.6*vn2*fc.ny;}
        ball.vx*=0.995;ball.vy*=0.995;
      }
      // enemies (bounce)
      for(var eI=0;eI<enemies.length;eI++){
        var en=enemies[eI];if(en.dead)continue;
        var dx2=ball.x-en.x,dy2=ball.y-en.y,Re=ball.r+13,d2e=dx2*dx2+dy2*dy2;
        if(d2e<Re*Re&&d2e>1e-6){
          var de=Math.sqrt(d2e),nx2=dx2/de,ny2=dy2/de;
          ball.x=en.x+nx2*Re;ball.y=en.y+ny2*Re;
          var vn3=ball.vx*nx2+ball.vy*ny2;
          if(vn3<0){ball.vx-=1.9*vn3*nx2;ball.vy-=1.9*vn3*ny2;}
          ball.vx+=nx2*70;ball.vy+=ny2*70;score+=25;
        }
      }
    }
    var spd=Math.hypot(ball.vx,ball.vy);
    if(spd>1250){ball.vx*=1250/spd;ball.vy*=1250/spd;}
    if(ball.inv>0)ball.inv-=dt;
    // drain
    if(ball.y>H+20){
      lives--;sDrain();shake=8;flashT=0.4;
      if(lives>0){ball.dead=true;respawnT=1.3;}else{gameOver();}
    }
  }

  // enemies move + spiral fire
  var allDead=enemies.length>0;
  for(var e2=0;e2<enemies.length;e2++){
    var en2=enemies[e2];
    if(en2.dead){allDead=false;continue;}
    en2.y=en2.by+Math.sin(t*2.2+en2.ph)*16;
    en2.x=en2.bx+Math.sin(t*1.3+en2.ph)*10;
    if(en2.spiral>0){
      en2.spiral-=dt;en2.sa+=7*dt;en2.st-=dt;
      if(en2.st<=0){en2.st=0.08;spawnB(en2.x,en2.y,en2.sa,150,140);}
    }
  }
  if(allDead){wave++;spawnWave(wave);}

  // firing
  if(!ball.ready&&!ball.dead){
    fireT-=dt;
    if(fireT<=0){fireT=Math.max(0.45,1.5-wave*0.09);firePattern();}
  }

  // bullets
  for(var i=bullets.length-1;i>=0;i--){
    var u=bullets[i];
    u.x+=u.vx*dt;u.y+=u.vy*dt;u.life-=dt;
    var dead=u.life<=0;
    if(!dead){
      if(segCollide(u,0.9)){u.b++;if(u.b>4)dead=true;}
    }
    if(!dead){
      for(var bb=0;bb<bumpers.length;bb++){
        var bu=bumpers[bb],dx3=u.x-bu.x,dy3=u.y-bu.y,Rb=u.r+bu.r;
        if(dx3*dx3+dy3*dy3<Rb*Rb){dead=true;bu.f=1;score+=50;sKillB();break;}
      }
    }
    if(!dead){
      var fc2=flipCapsule(u);
      if(fc2){dead=true;score+=25;sKillB();}
    }
    if(!dead){
      for(var ee=0;ee<enemies.length;ee++){
        var en3=enemies[ee];if(en3.dead)continue;
        var dx4=u.x-en3.x,dy4=u.y-en3.y;
        if(dx4*dx4+dy4*dy4<(u.r+13)*(u.r+13)){
          dead=true;en3.dead=true;score+=500;sEnemy();shake=10;break;
        }
      }
    }
    if(!dead&&!ball.ready&&!ball.dead&&ball.inv<=0){
      var dx5=u.x-ball.x,dy5=u.y-ball.y;
      if(dx5*dx5+dy5*dy5<(u.r+ball.r)*(u.r+ball.r)){
        dead=true;ball.inv=1.4;
        ball.vx+=u.vx*0.45;ball.vy+=u.vy*0.45;
        score=Math.max(0,score-50);flashT=0.3;sHit();shake=Math.max(shake,4);
      }
    }
    if(dead)bullets.splice(i,1);
  }
}

function draw(){
  ctx.save();
  // background
  var g=ctx.createLinearGradient(0,0,0,H);
  g.addColorStop(0,'#0b1226');g.addColorStop(0.55,'#080d1c');g.addColorStop(1,'#05070f');
  ctx.fillStyle=g;ctx.fillRect(0,0,W,H);
  ctx.fillStyle='#9fd8ff';
  for(var i=0;i<stars.length;i++){var st=stars[i];ctx.globalAlpha=st.a*(0.6+0.4*Math.sin(t*2+i));ctx.fillRect(st.x,st.y,st.r,st.r);}
  ctx.globalAlpha=1;

  if(shake>0)ctx.translate((Math.random()-0.5)*shake,(Math.random()-0.5)*shake);

  // table walls
  ctx.strokeStyle='#19e6ff';ctx.lineWidth=4;ctx.lineCap='round';
  ctx.shadowColor='#19e6ff';ctx.shadowBlur=12;

  for(var sI=0;sI<segs.length;sI++){var s=segs[sI];ctx.moveTo(s.x1,s.y1);ctx.lineTo(s.x2,s.y2);}
  ctx.stroke();
  ctx.shadowBlur=0;

  // plunger
  ctx.fillStyle='#1c2a45';ctx.fillRect(216,738,48,16);
  ctx.strokeStyle='#3a5a80';ctx.lineWidth=2;ctx.strokeRect(216,738,48,16);
  if(charge>0){
    ctx.fillStyle='hsl('+(120-charge*120)+',90%,55%)';
    ctx.fillRect(272,748-charge*40,10,charge*40);
    ctx.strokeStyle='#fff';ctx.lineWidth=1;ctx.strokeRect(272,708,10,40);
  }

  // bumpers
  for(var bp=0;bp<bumpers.length;bp++){
    var u=bumpers[bp],fl=u.f;
    ctx.shadowColor='#ffd23e';ctx.shadowBlur=10+fl*15;

    ctx.strokeStyle='#ff2d78';ctx.lineWidth=f.fr*2;ctx.lineCap='round';
    ctx.beginPath();ctx.moveTo(f.px,f.py);ctx.lineTo(tx,ty);ctx.stroke();
    ctx.shadowBlur=0;
    ctx.fillStyle='#ffd7e8';ctx.beginPath();ctx.arc(f.px,f.py,4,0,6.283);ctx.fill();
  }

  // enemies
  for(var eI=0;eI<enemies.length;eI++){
    var en=enemies[eI];if(en.dead)continue;
    ctx.save();ctx.translate(en.x,en.y);ctx.rotate(t*2+en.ph);
    ctx.shadowColor='#b26bff';ctx.shadowBlur=16;
    var p=13+Math.sin(t*6+en.ph)*1.5;
    ctx.fillStyle='#2a1548';ctx.strokeStyle='#c88fff';ctx.lineWidth=2.5;
    ctx.beginPath();ctx.moveTo(0,-p);ctx.lineTo(p,0);ctx.lineTo(0,p);ctx.lineTo(-p,0);ctx.closePath();
    ctx.fill();ctx.stroke();ctx.shadowBlur=0;
    ctx.fillStyle=en.spiral>0?'#7dffa8':'#fff';
    ctx.beginPath();ctx.arc(0,0,4,0,6.283);ctx.fill();
    ctx.restore();
  }

  // bullets
  for(var bI=0;bI<bullets.length;bI++){
    var u=bullets[bI];
    ctx.fillStyle='hsla('+u.hue+',100%,65%,0.25)';
    ctx.beginPath();ctx.arc(u.x,u.y,u.r*2.4,0,6.283);ctx.fill();
    ctx.shadowColor='hsl('+u.hue+',100%,65%)';ctx.shadowBlur=8;
    ctx.fillStyle='hsl('+u.hue+',100%,72%)';
    ctx.beginPath();ctx.arc(u.x,u.y,u.r,0,6.283);ctx.fill();
    ctx.shadowBlur=0;
  }

  // ball
  if(!ball.dead){
    ctx.globalAlpha=(ball.inv>0&&Math.sin(t*40)<0)?0.35:1;
    var bg=ctx.createRadialGradient(ball.x-3,ball.y-3,1,ball.x,ball.y,ball.r);
    bg.addColorStop(0,'#ffffff');bg.addColorStop(0.7,'#c9d6e8');bg.addColorStop(1,'#7a8aa5');
    ctx.shadowColor='#bfe9ff';ctx.shadowBlur=12;
    ctx.fillStyle=bg;ctx.beginPath();ctx.arc(ball.x,ball.y,ball.r,0,6.283);ctx.fill();
    ctx.shadowBlur=0;ctx.globalAlpha=1;
  }

  // HUD strip
  ctx.restore();
  ctx.fillStyle='#0a0f1e';ctx.fillRect(0,700,W,60);
  ctx.strokeStyle='#1c2f4d';ctx.lineWidth=2;
  ctx.beginPath();ctx.moveTo(0,700);ctx.lineTo(W,700);ctx.stroke();

  ctx.fillStyle='#e8f4ff';ctx.textAlign='left';
  ctx.fillText('SCORE '+String(score).padStart(6,'0'),12,720);

  ctx.fillText('WAVE '+wave,12,746);
  ctx.textAlign='right';
  ctx.fillStyle='#ffd23e';ctx.fillText('HI '+String(Math.max(hi,score)).padStart(6,'0'),W-12,720);
  for(var li=0;li<3;li++){

    ctx.fillText('HOLD SPACE / TAP TO LAUNCH',240,756);
  }

  // banner
  if(bannerT>0){
    ctx.globalAlpha=Math.min(1,bannerT);
    ctx.textAlign='center';ctx.font='bold 44px "Courier New",monospace';
    ctx.shadowColor='#00e5ff';ctx.shadowBlur=20;
    ctx.fillStyle='#dffcff';ctx.fillText(banner,240,360);
    ctx.shadowBlur=0;ctx.globalAlpha=1;
  }

  // hit flash
  if(flashT>0){ctx.fillStyle='rgba(255,40,60,'+(flashT*0.4)+')';ctx.fillRect(0,0,W,H);}
}
