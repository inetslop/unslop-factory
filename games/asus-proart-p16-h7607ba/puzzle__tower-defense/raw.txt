function beep(f, d, type, vol, slide){
  if(!AC) return;
  try{
    var o = AC.createOscillator(), g = AC.createGain();
    o.type = type||'square'; o.frequency.value = f;
    if(slide) o.frequency.exponentialRampToValueAtTime(Math.max(30,f+slide), AC.currentTime+d);
    g.gain.value = vol||0.08;
    g.gain.exponentialRampToValueAtTime(0.0001, AC.currentTime+d);
    o.connect(g); g.connect(AC.destination);
    o.start(); o.stop(AC.currentTime+d);
  }catch(e){}
}

// ---------- update ----------
function update(dt){
  if(G.state!=='playing') return;

  // spawning
  if(G.waveActive && G.spawnQueue.length){
    G.spawnT-=dt;
    if(G.spawnT<=0){ spawnEnemy(G.spawnQueue.shift()); G.spawnT = G.wave===G.maxWave?0.7:0.85; }
  }

  // enemies
  for(var i=G.enemies.length-1;i>=0;i--){
    var en=G.enemies[i];
    var spd=en.spd*(en.slowT>0?0.45:1);
    if(en.slowT>0) en.slowT-=dt;
    var tgt=wpPx(en.wp), dx=tgt.x-en.x, dy=tgt.y-en.y, dist=Math.hypot(dx,dy);
    var step=spd*dt;
    if(dist<=step){
      en.x=tgt.x; en.y=tgt.y; en.wp++;
      if(en.wp>=WPS.length){
        G.enemies.splice(i,1);
        G.lives-=en.dmg; beep(160,0.25,'sawtooth',0.08,-60);
        if(G.lives<=0){ G.lives=0; updateHUD(); endGame(false); return; }
        updateHUD(); continue;
      }
    } else { en.x+=dx/dist*step; en.y+=dy/dist*step; }
  }

  // towers
  for(var ti=0;ti<G.towers.length;ti++){
    var tw=G.towers[ti];
    tw.cd-=dt;
    // find most-advanced enemy in range
    var best=null, bestProg=-1;
    for(var ei=0;ei<G.enemies.length;ei++){
      var e2=G.enemies[ei], ddx=e2.x-tw.x, ddy=e2.y-tw.y;
      if(ddx*ddx+ddy*ddy<=tw.range*tw.range){
        var prog=e2.wp - Math.hypot(wpPx(e2.wp).x-e2.x, wpPx(e2.wp).y-e2.y)/1000;
        if(prog>bestProg){ bestProg=prog; best=e2; }
      }
    }
    if(best){
      tw.angle=Math.atan2(best.y-tw.y,best.x-tw.x);
      if(tw.cd<=0){
        tw.cd=1/tw.rate;
        G.projs.push({ x:tw.x, y:tw.y, target:best, spd:TDEFS[tw.type].pspeed, dmg:tw.dmg, type:tw.type,
          color:TDEFS[tw.type].proj });
        if(tw.type==='gun') beep(880,0.03,'square',0.025);
        else if(tw.type==='cannon') beep(180,0.09,'sawtooth',0.04,-60);
        else beep(1200,0.04,'sine',0.03);
      }
    }
  }

  // projectiles
  for(var pi=G.projs.length-1;pi>=0;pi--){
    var pr=G.projs[pi];
    var tx=pr.target, alive=G.enemies.indexOf(pr.target)>=0;
    if(!alive){ G.projs.splice(pi,1); continue; }
    var px=tx.x-pr.x, py=tx.y-pr.y, pd=Math.hypot(px,py), stp=pr.spd*dt;
    if(pd<=stp+tx.r){
      // hit
      if(pr.type==='cannon'){
        G.effects.push({x:tx.x,y:tx.y,r:62,t:0.3,max:0.3,color:'rgba(255,170,80,'});
        beep(90,0.15,'sawtooth',0.09,-40);
        for(var si=G.enemies.length-1;si>=0;si--){
          var se=G.enemies[si];
          if(Math.hypot(se.x-tx.x,se.y-tx.y)<=62) hitEnemy(se,pr.dmg);
        }
      } else {
        if(pr.type==='frost'){ tx.slowT=1.6; G.effects.push({x:tx.x,y:tx.y,r:20,t:0.2,max:0.2,color:'rgba(140,255,230,'}); }
        else beep(600,0.02,'square',0.015);
        hitEnemy(tx,pr.dmg);
      }
      G.projs.splice(pi,1);
    } else { pr.x+=px/pd*stp; pr.y+=py/pd*stp; }
  }

  // effects
  for(var fi=G.effects.length-1;fi>=0;fi--){
    G.effects[fi].t-=dt;
    if(G.effects[fi].t<=0) G.effects.splice(fi,1);
  }

  // wave end
  if(G.waveActive && !G.spawnQueue.length && !G.enemies.length){
    G.waveActive=false;
    var bonus=20+G.wave*5;
    G.money+=bonus;
    beep(700,0.1,'square',0.06); setTimeout(function(){beep(900,0.15,'square',0.06);},100);
    if(G.wave>=G.maxWave){ endGame(true); return; }
  }
  updateHUD();
}
function hitEnemy(en,dmg){
  en.hp-=dmg;
  if(en.hp<=0 && G.enemies.indexOf(en)>=0){
    G.enemies.splice(G.enemies.indexOf(en),1);
    G.money+=en.gold;
    G.effects.push({x:en.x,y:en.y,r:en.r*1.6,t:0.25,max:0.25,color:'rgba(255,235,150,'});
    beep(en.type==='boss'?140:1000, en.type==='boss'?0.3:0.05, 'triangle', 0.05, -300);
  }
}

var grassCache=document.createElement('canvas'); grassCache.width=W; grassCache.height=H;
(function(){
  var g=grassCache.getContext('2d');
  for(var c=0;c<COLS;c++) for(var r=0;r<ROWS;r++){
    if(pathCells[c+','+r]){ g.fillStyle=(c+r)%2?'#b79a63':'#ac8f58'; }
    else { g.fillStyle=(c+r)%2?'#1e4d2b':'#1a4527'; }
    g.fillRect(c*T,r*T,T,T);
  }
  g.strokeStyle='rgba(0,0,0,0.08)';
  for(var c2=0;c2<=COLS;c2++){ g.beginPath(); g.moveTo(c2*T,0); g.lineTo(c2*T,H); g.stroke(); }
  for(var r2=0;r2<=ROWS;r2++){ g.beginPath(); g.moveTo(0,r2*T); g.lineTo(W,r2*T); g.stroke(); }
})();

function render(){
  ctx.clearRect(0,0,W,H);
  ctx.drawImage(grassCache,0,0);

  // entry / exit markers
  ctx.fillStyle='rgba(80,255,140,0.25)'; ctx.fillRect(0,2*T,T,T);
  ctx.fillStyle='rgba(255,80,80,0.3)'; ctx.fillRect((COLS-1)*T,7*T,T,T);
  ctx.font='bold 26px Arial'; ctx.textAlign='center'; ctx.textBaseline='middle';
  ctx.fillStyle='#cfe'; ctx.fillText('▶', T/2+4, 2*T+T/2);
  ctx.fillStyle='#fd8'; ctx.fillText('⌂', (COLS-0.5)*T, 7*T+T/2);

  // hover build preview
  if(G.state==='playing' && G.hover && G.selType){
    var hc=G.hover.c, hr=G.hover.r;
    if(hc>=0&&hc<COLS&&hr>=0&&hr<ROWS){
      var ok=!pathCells[hc+','+hr]&&!towerAt(hc,hr)&&G.money>=TDEFS[G.selType].cost;
      ctx.fillStyle=ok?'rgba(120,255,140,0.25)':'rgba(255,90,90,0.3)';
      ctx.fillRect(hc*T,hr*T,T,T);
      if(ok){
        ctx.beginPath(); ctx.arc((hc+0.5)*T,(hr+0.5)*T,TDEFS[G.selType].range,0,7);
        ctx.fillStyle='rgba(255,255,255,0.06)'; ctx.fill();
        ctx.strokeStyle='rgba(255,255,255,0.25)'; ctx.stroke();
      }
    }
  }

  // selected tower range
  if(G.selTower){
    ctx.beginPath(); ctx.arc(G.selTower.x,G.selTower.y,G.selTower.range,0,7);
    ctx.fillStyle='rgba(255,215,94,0.08)'; ctx.fill();
    ctx.strokeStyle='rgba(255,215,94,0.5)'; ctx.stroke();
  }

  // towers
  for(var i=0;i<G.towers.length;i++){
    var tw=G.towers[i], d=TDEFS[tw.type];
    ctx.fillStyle='#0d1826'; ctx.beginPath(); ctx.arc(tw.x,tw.y,24,0,7); ctx.fill();
    ctx.strokeStyle=d.color; ctx.lineWidth=3; ctx.stroke(); ctx.lineWidth=1;
    // barrel
    ctx.save(); ctx.translate(tw.x,tw.y); ctx.rotate(tw.angle);
    ctx.fillStyle=d.color;
    if(tw.type==='cannon') ctx.fillRect(0,-7,26,14);
    else ctx.fillRect(0,-4,24,8);
    ctx.restore();
    ctx.beginPath(); ctx.arc(tw.x,tw.y,tw.type==='cannon'?13:10,0,7);
    ctx.fillStyle=d.color; ctx.fill();
    // level pips
    for(var l=0;l<tw.lvl;l++){
      ctx.beginPath(); ctx.arc(tw.x-12+l*12, tw.y+18, 3.5,0,7);
      ctx.fillStyle='#ffd75e'; ctx.fill();
    }
  }

  // enemies
  for(var ei=0;ei<G.enemies.length;ei++){
    var en=G.enemies[i===0?0:ei]||G.enemies[ei]; en=G.enemies[ei];
    var col = en.type==='fast'?'#c8ff6e':en.type==='tank'?'#c97aff':en.type==='boss'?'#ff4f4f':'#ffb3b3';
    if(en.slowT>0) col='#a8e8ff';
    var wob=Math.sin(performance.now()/200+en.wob)*1.5;
    ctx.beginPath(); ctx.arc(en.x,en.y+wob,en.r,0,7);
    ctx.fillStyle=col; ctx.fill();
    ctx.strokeStyle='rgba(0,0,0,0.4)'; ctx.stroke();
    if(en.type==='boss'){ ctx.font='bold 22px Arial'; ctx.fillStyle='#3a0000'; ctx.fillText('☠',en.x,en.y+wob+1); }
    // hp bar
    var bw=en.r*2, bx=en.x-en.r, by=en.y-en.r-9;
    ctx.fillStyle='rgba(0,0,0,0.55)'; ctx.fillRect(bx,by,bw,5);
    ctx.fillStyle=en.hp/en.maxhp>0.4?'#5eff8a':'#ff6a5e';
    ctx.fillRect(bx,by,bw*Math.max(0,en.hp/en.maxhp),5);
  }

  // projectiles
  for(var pi=0;pi<G.projs.length;pi++){
    var pr=G.projs[pi];
    ctx.beginPath(); ctx.arc(pr.x,pr.y, pr.type==='cannon'?7:4.5, 0,7);
    ctx.fillStyle=pr.color; ctx.fill();
  }

  // effects
  for(var fi=0;fi<G.effects.length;fi++){
    var ef=G.effects[fi], k=1-ef.t/ef.max;
    ctx.beginPath(); ctx.arc(ef.x,ef.y,ef.r*(0.4+0.6*k),0,7);
    ctx.strokeStyle=ef.color+(0.8*(1-k))+')'; ctx.lineWidth=3; ctx.stroke(); ctx.lineWidth=1;
  }

  // wave banner
  if(G.waveActive && G.spawnQueue.length===G.spawnQueue.length && G.enemies.length<2){
    // (kept simple; no banner spam)
  }
}

// ---------- loop ----------
var last=performance.now();
function frame(now){
  var dt=Math.min(0.05,(now-last)/1000); last=now;
  update(dt); render();
  requestAnimationFrame(frame);
}
requestAnimationFrame(frame);
})();
