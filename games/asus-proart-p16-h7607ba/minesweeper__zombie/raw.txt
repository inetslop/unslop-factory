function tone(freq, dur, type, vol, slideTo){
  if(!AC) return;
  try{
    const o=AC.createOscillator(), g=AC.createGain();
    o.type=type||'square'; o.frequency.setValueAtTime(freq, AC.currentTime);
    if(slideTo) o.frequency.exponentialRampToValueAtTime(Math.max(20,slideTo), AC.currentTime+dur);
    g.gain.setValueAtTime(vol||0.12, AC.currentTime);
    g.gain.exponentialRampToValueAtTime(0.0001, AC.currentTime+dur);
    o.connect(g); g.connect(AC.destination);
    o.start(); o.stop(AC.currentTime+dur+0.02);
  }catch(e){}
}
function noiseBurst(dur, vol){
  if(!AC) return;
  try{
    const n = Math.floor(AC.sampleRate*dur);
    const buf = AC.createBuffer(1,n,AC.sampleRate);
    const d = buf.getChannelData(0);
    for(let i=0;i<n;i++) d[i]=(Math.random()*2-1)*(1-i/n);
    const src=AC.createBufferSource(); src.buffer=buf;
    const g=AC.createGain(); g.gain.setValueAtTime(vol||0.3, AC.currentTime);
    g.gain.exponentialRampToValueAtTime(0.0001, AC.currentTime+dur);
    const f=AC.createBiquadFilter(); f.type='lowpass'; f.frequency.value=900;
    src.connect(f); f.connect(g); g.connect(AC.destination); src.start();
  }catch(e){}
}

/* ---------- game logic ---------- */
function inb(x,y){ return x>=0 && y>=0 && x<gw && y<gh; }
function neighbors(x,y){
  const out=[];
  for(let dy=-1;dy<=1;dy++)for(let dx=-1;dx<=1;dx++){
    if(!dx&&!dy) continue;
    if(inb(x+dx,y+dy)) out.push([x+dx,y+dy]);
  }
  return out;
}
function newBoard(lvl){
  clearTimeout(loseTimer);
  level=lvl;
  const L=LEVELS[lvl];
  gw=L.w; gh=L.h;
  grid=[];
  for(let y=0;y<gh;y++){
    const row=[];
    for(let x=0;x<gw;x++) row.push({z:false,adj:0,rev:false,flag:false,phase:Math.random()*Math.PI*2});
    grid.push(row);
  }
  flagsPlaced=0; revealedCount=0; firstClick=true;
  particles=[]; elapsed=0; startTime=performance.now();
  state='playing';
}
function placeZombies(sx,sy){
  const L=LEVELS[level];
  const safe=new Set([sy*gw+sx]);
  neighbors(sx,sy).forEach(n=>safe.add(n[1]*gw+n[0]));
  let placed=0;
  while(placed<L.z){
    const x=(Math.random()*gw)|0, y=(Math.random()*gh)|0;
    if(grid[y][x].z || safe.has(y*gw+x)) continue;
    grid[y][x].z=true; placed++;
  }
  for(let y=0;y<gh;y++)for(let x=0;x<gw;x++){
    let c=0;
    neighbors(x,y).forEach(n=>{ if(grid[n[1]][n[0]].z) c++; });
    grid[y][x].adj=c;
  }
}
function spawnBurst(px,py,color,n,pow){
  for(let i=0;i<n;i++){
    const a=Math.random()*Math.PI*2, sp=(0.5+Math.random())*pow;
    particles.push({x:px,y:py,vx:Math.cos(a)*sp,vy:Math.sin(a)*sp-1,life:1,decay:0.012+Math.random()*0.02,r:2+Math.random()*4,c:color});
  }
}
function revealCell(x,y){
  const st=grid[y][x];
  if(st.rev||st.flag) return;
  st.rev=true; revealedCount++;
  if(st.z){ spawnBurst(boardOx+x*cell+cell/2, boardOy+y*cell+cell/2, '#b7ff5e', 18, 4); }
  else if(st.adj===0){ neighbors(x,y).forEach(n=>revealCell(n[0],n[1])); }
}
function tryReveal(x,y){
  if(state!=='playing') return;
  const st=grid[y][x];
  if(st.flag) return;
  if(st.rev){ chord(x,y); return; }
  if(firstClick){ firstClick=false; placeZombies(x,y); }
  sfx.dig();
  revealCell(x,y);
  checkEnd();
}
function toggleFlag(x,y){
  if(state!=='playing') return;
  const st=grid[y][x];
  if(st.rev) return;
  st.flag=!st.flag;
  flagsPlaced+=st.flag?1:-1;
  st.flag?sfx.flag():sfx.unflag();
}
function chord(x,y){
  const st=grid[y][x];
  if(!st.adj) return;
  let f=0; neighbors(x,y).forEach(n=>{ if(grid[n[1]][n[0]].flag) f++; });
  if(f!==st.adj) return;
  for(const [nx,ny] of neighbors(x,y)){
    const c=grid[ny][nx];
    if(c.flag||c.rev) continue;
    c.rev=true; revealedCount++;
    if(c.z){ spawnBurst(boardOx+nx*cell+cell/2, boardOy+ny*cell+cell/2,'#b7ff5e',18,4); }
    else if(c.adj===0){ const stack=[[nx,ny]]; while(stack.length){ const [cx,cy]=stack.pop(); const s=grid[cy][cx]; if(s.rev||s.flag) continue; s.rev=true; revealedCount++; if(!s.z&&s.adj===0) neighbors(cx,cy).forEach(n=>{ if(!grid[n[1]][n[0]].rev&&!grid[n[1]][n[0]].flag) stack.push(n); }); } }
  }
}
function checkEnd(){
  const total=gw*gh, zombies=LEVELS[level].z;
  if(revealedCount===total-zombies){ winGame(); return; }
  // lose handled by revealCell via z flag below
}
function loseAt(x,y){
  state='lost'; sfx.boom();

    const c=grid[yy][xx];
    if(c.z && !c.flag) c.rev=true;
    else if(!c.z && c.flag) c.wrong=true;

}

  for(let y=0;y<gh;y++)for(let x=0;x<gw;x++){ if(grid[y][x].z){ grid[y][x].flag=true; f++; } }

  spawnBurst(cssW/2, cssH/2, '#a8e063', 60, 9);
  loseTimer=setTimeout(()=>{ if(state==='won') showOverlay('😎 You survived the outbreak! All ' + LEVELS[level].z + ' zombies flagged.'); },700);
}

function drawZombieFace(cx,cy,r,alpha){

  ctx.fillStyle='#243318';

  ctx.beginPath(); ctx.arc(cx+r*0.38,cy-r*0.15,r*0.16,0,Math.PI*2); ctx.fill();
  // pupils glint
  ctx.fillStyle='#c9ff7a';
  ctx.beginPath(); ctx.arc(cx-r*0.34,cy-r*0.19,r*0.05,0,Math.PI*2); ctx.fill();
  ctx.beginPath(); ctx.arc(cx+r*0.42,cy-r*0.19,r*0.05,0,Math.PI*2); ctx.fill();
  // mouth
  ctx.strokeStyle='#243318'; ctx.lineWidth=Math.max(1,r*0.12);
  ctx.beginPath(); ctx.moveTo(cx-r*0.3,cy+r*0.42); ctx.lineTo(cx+r*0.3,cy+r*0.42); ctx.stroke();
  // bite mark
  ctx.fillStyle='#5d8539';
  ctx.beginPath(); ctx.arc(cx+r*0.7,cy-r*0.6,r*0.22,0,Math.PI*2); ctx.fill();
  ctx.restore();
}
function draw(t){
  const L=LEVELS[level];
  // background
  const g=ctx.createLinearGradient(0,0,0,cssH);
  g.addColorStop(0,'#233118'); g.addColorStop(1,'#141d0e');
  ctx.fillStyle=g; ctx.fillRect(0,0,cssW,cssH);

  // subtle fog blobs
  ctx.save(); ctx.globalAlpha=0.05; ctx.fillStyle='#a8e063';
  for(let i=0;i<5;i++){
    const fx=(cssW*(i+0.5)) + Math.sin(t*0.0004+i*2)*40;
    const fy=cssH*0.5 + Math.cos(t*0.0003+i)*cssH*0.3;
    ctx.beginPath(); ctx.arc(fx,fy,80+30*Math.sin(i*3),0,Math.PI*2); ctx.fill();
  }
  ctx.restore();

  // layout
  const headerH=54, pad=12;
  cell=Math.floor(Math.min((cssW-2*pad)/L.w,(cssH-headerH-2*pad)/L.h));
  cell=Math.max(10,cell);
  boardOx=Math.round((cssW-cell*L.w)/2);
  boardOy=Math.round(headerH+(cssH-headerH-cell*L.h)/2);

  // header bar
  ctx.fillStyle='rgba(10,16,7,.85)'; ctx.fillRect(0,0,cssW,headerH-6);
  ctx.strokeStyle='#3d5426'; ctx.lineWidth=2;
  ctx.beginPath(); ctx.moveTo(0,headerH-6); ctx.lineTo(cssW,headerH-6); ctx.stroke();

  faceRect={x:12,y:8,w:42,h:42};
  roundRect(faceRect.x,faceRect.y,faceRect.w,faceRect.h,9);
  ctx.fillStyle='#33481f'; ctx.fill(); ctx.strokeStyle='#5c7a3a'; ctx.stroke();
  ctx.font='26px serif'; ctx.textAlign='center'; ctx.textBaseline='middle';
  ctx.fillText(state==='won'?'😎':state==='lost'?'😵':'🙂', faceRect.x+21, faceRect.y+23);

  ctx.font='bold 20px "Segoe UI",Arial';
  const flagsLeft = LEVELS[level].z - flagsPlaced;
  ctx.textAlign='left';
  ctx.fillStyle='#e8f2d4';
  ctx.fillText('🚩 ' + String(Math.max(0,flagsLeft)).padStart(3,'0'), faceRect.x+60, headerH/2-1);
  const mins = Math.floor(elapsed/60000), secs=Math.floor((elapsed%60000)/1000);
  ctx.textAlign='right';
  ctx.fillText('⏱ ' + String(mins).padStart(2,'0')+':'+String(secs).padStart(2,'0'), cssW-18, headerH/2-1);

  // board shadow
  ctx.fillStyle='rgba(0,0,0,.45)';
  roundRect(boardOx-6,boardOy-6,cell*L.w+12,cell*L.h+12,8); ctx.fill();

  if(grid.length){
    for(let y=0;y<gh;y++)for(let x=0;x<gw;x++){
      const c=grid[y][x];
      const px=boardOx+x*cell, py=boardOy+y*cell;
      const s=cell-2;
      if(c.rev){
        ctx.fillStyle = c.z ? '#6e2b23' : '#39452a';
        roundRect(px+1,py+1,s,s,4); ctx.fill();
        ctx.strokeStyle='rgba(0,0,0,.35)'; ctx.lineWidth=1; ctx.stroke();
        if(c.z){
          ctx.font=Math.floor(s*0.72)+'px serif';
          ctx.textAlign='center'; ctx.textBaseline='middle';
          ctx.fillText('🧟', px+1+s/2, py+1+s/2+1);
        } else if(c.adj>0){
          ctx.fillStyle=NUM_COLORS[c.adj];
          ctx.font='bold '+Math.floor(s*0.58)+'px "Segoe UI",Arial';
          ctx.textAlign='center'; ctx.textBaseline='middle';
          ctx.fillText(c.adj, px+1+s/2, py+1+s/2+1);
        }
      } else {
        // raised tile
        ctx.fillStyle='#8fa663'; roundRect(px+1,py+1,s,s,4); ctx.fill();
        // bevel
        ctx.strokeStyle='rgba(255,255,255,.35)'; ctx.lineWidth=2;
        ctx.beginPath(); ctx.moveTo(px+3,py+s-1); ctx.lineTo(px+3,py+3); ctx.lineTo(px+s-1,py+3); ctx.stroke();
        ctx.strokeStyle='rgba(0,0,0,.4)';
        ctx.beginPath(); ctx.moveTo(px+s-2,py+3); ctx.lineTo(px+s-2,py+s-2); ctx.lineTo(px+3,py+s-2); ctx.stroke();

        if(c.flag){
          if(c.wrong){
            ctx.font=Math.floor(s*0.6)+'px serif'; ctx.textAlign='center'; ctx.textBaseline='middle';
            ctx.fillText('🚩', px+1+s/2, py+1+s/2);
            ctx.strokeStyle='#ff4d3a'; ctx.lineWidth=Math.max(2,s*0.08);
            ctx.beginPath();
            ctx.moveTo(px+s*0.22,py+s*0.22); ctx.lineTo(px+s*0.78,py+s*0.78);
            ctx.moveTo(px+s*0.78,py+s*0.22); ctx.lineTo(px+s*0.22,py+s*0.78);
            ctx.stroke();
          } else {
            ctx.font=Math.floor(s*0.68)+'px serif'; ctx.textAlign='center'; ctx.textBaseline='middle';
            ctx.fillText('🚩', px+1+s/2, py+1+s/2+1);
          }
        } else {
          // zombie peeking & bobbing
          const bob=Math.sin(t*0.0035+c.phase)*s*0.06;
          const peek=0.55+0.45*Math.max(0,Math.sin(t*0.0012+c.phase*1.7));
          drawZombieFace(px+1+s/2, py+1+s/2+bob, s*0.30, 0.35+0.5*peek);
        }
      }
    }
    // hover highlight
    if(hover && state==='playing'){
      const c=grid[hover.y][hover.x];
      ctx.strokeStyle='rgba(200,255,120,.8)'; ctx.lineWidth=2;
      roundRect(boardOx+hover.x*cell+1, boardOy+hover.y*cell+1, cell-2, cell-2, 4);
      ctx.stroke();
    }
  }

  // particles
  for(let i=particles.length-1;i>=0;i--){
    const p=particles[i];
    p.x+=p.vx; p.y+=p.vy; p.vy+=0.12; p.life-=p.decay;
    if(p.life<=0){ particles.splice(i,1); continue; }
    ctx.globalAlpha=p.life;
    ctx.fillStyle=p.c;
    ctx.beginPath(); ctx.arc(p.x,p.y,p.r*p.life,0,Math.PI*2); ctx.fill();
  }
  ctx.globalAlpha=1;

  requestAnimationFrame(draw);
}

/* ---------- timer (in loop) ---------- */
setInterval(()=>{ if(state==='playing') elapsed=performance.now()-startTime; },250);

requestAnimationFrame(draw);
})();
