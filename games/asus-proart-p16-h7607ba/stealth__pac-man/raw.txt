function beep(f,dur,type,vol,slideTo){
  if(!AC)return;
  try{
    var o=AC.createOscillator(),g=AC.createGain(),t=AC.currentTime;
    o.type=type||'square';o.frequency.setValueAtTime(f,t);
    if(slideTo)o.frequency.exponentialRampToValueAtTime(Math.max(30,slideTo),t+dur);
    g.gain.setValueAtTime(vol||0.12,t);
    g.gain.exponentialRampToValueAtTime(0.001,t+dur);
    o.connect(g);g.connect(AC.destination);o.start(t);o.stop(t+dur+0.02);
  }catch(e){}
}

/* ---------- movement ---------- */
function moveEnt(e,dt,decide){
  var dist=e.speed*dt;
  var guard=0;
  while(dist>0&&guard++<64&&(e.dir.x!==0||e.dir.y!==0)){
    var tx=(e.cx+e.dir.x+0.5)*cell,ty=(e.cy+e.dir.y+0.5)*cell;
    var dx=tx-e.x,dy=ty-e.y,d=Math.hypot(dx,dy);
    if(d<=dist){e.x=tx;e.y=ty;e.cx+=e.dir.x;e.cy+=e.dir.y;dist-=d;decide(e);}
    else{e.x+=dx/d*dist;e.y+=dy/d*dist;dist=0;}
  }
}
function playerDecide(){
  for(var i=buffer.length-1;i>=0;i--){
    var d=buffer[i];
    if((d.x!==-p.dir.x||d.y!==-p.dir.y)&&!isWall(p.cx+d.x,p.cy+d.y)){p.dir=d;buffer.length=0;return;}
  }
  if(isWall(p.cx+p.dir.x,p.cy+p.dir.y)){p.dir={x:0,y:0};buffer.length=0;}
}
function ghostDecide(g){
  var opts=[],i,d;
  for(i=0;i<4;i++){d=DIRS[i];if(d.x!==-g.dir.x||d.y!==-g.dir.y){if(!isWall(g.cx+d.x,g.cy+d.y))opts.push(d);}}
  var pool=opts.length?opts:[];
  if(!pool.length)for(i=0;i<4;i++){d=DIRS[i];if(!isWall(g.cx+d.x,g.cy+d.y))pool.push(d);}
  if(!pool.length)return;
  if(g.state==='chase'){
    var best=null,bd=1e9;
    for(i=0;i<pool.length;i++){
      d=pool[i];
      var dd=Math.pow(g.cx+d.x-p.cx,2)+Math.pow(g.cy+d.y-p.cy,2);
      if(dd<bd-0.001||(Math.abs(dd-bd)<0.001&&Math.random()<0.5)){bd=dd;best=d;}
    }
    g.dir=best;
  }else{
    var keep=false;
    for(i=0;i<opts.length;i++)if(opts[i]===g.dir)keep=true;
    if(Math.random()<0.6&&keep){/* go straight */}
    else g.dir=pool[(Math.random()*pool.length)|0];
  }
}

/* ---------- vision / stealth ---------- */
function los(ax,ay,bx,by){
  var x0=ax,y0=ay,dx=Math.abs(bx-ax),dy=-Math.abs(by-ay),sx=ax<bx?1:-1,sy=ay<by?1:-1,err=dx+dy;
  while(true){
    if(!(x0===ax&&y0===ay)&&!(x0===bx&&y0===by)){if(isWall(x0,y0))return false;}
    if(x0===bx&&y0===by)break;
    var e2=2*err;
    if(e2>=dy){err+=dy;x0+=sx;}
    if(e2<=dx){err+=dx;y0+=sy;}
  }
  return true;
}
function inSight(g,rangePx){
  if(Math.hypot(g.x-p.x,g.y-p.y)>rangePx)return false;
  return los(g.cx,g.cy,p.cx,p.cy);
}

function eatDot(){
  dots[p.cy][p.cx]=false;dotsLeft--;score+=10;
  if(score>hi){hi=score;}
  sfxEat();
  effects.push({x:p.x,y:p.y,r:3,maxR:cell*1.7,life:0.45,t:0.45,c:'255,214,107'});
  for(var i=0;i<ghosts.length;i++){
    var g=ghosts[i];
    if(Math.hypot(g.x-p.x,g.y-p.y)<cell*4)g.alert=Math.max(g.alert,1.3);
  }
}
function onCaught(){
  lives--;hitFlash=0.6;sfxCaught();
  effects.push({x:p.x,y:p.y,r:4,maxR:cell*2.4,life:0.5,t:0.5,c:'255,84,104'});
  if(lives<=0){state='gameover';sfxOver();saveHi();showGameOver();}
  else{resetPlayer();p.invuln=2.5;}
}

/* ---------- update ---------- */
function update(dt){
  tGlobal+=dt;
  hitFlash=Math.max(0,hitFlash-dt);
  for(var i=effects.length-1;i>=0;i--){var f=effects[i];f.t-=dt;if(f.t<=0)effects.splice(i,1);}

  if(state==='levelup'){
    bannerT-=dt;
    if(bannerT<=0){
      level++;regenDots();resetGhosts();resetPlayer();
      p.invuln=1.2;state='playing';
    }
    return;
  }
  if(state!=='playing')return;

  var sp=speeds();
  p.speed=sp.player;
  p.invuln=Math.max(0,p.invuln-dt);
  if(p.dir.x||p.dir.y)p.lastAng=Math.atan2(p.dir.y,p.dir.x);

  moveEnt(p,dt,playerDecide);

  if(dots[p.cy]&&dots[p.cy][p.cx])eatDot();

  for(var gi=0;gi<ghosts.length;gi++){
    var g=ghosts[gi];
    var seen=inSight(g,sp.range*cell);
    if(seen)g.sight+=dt;else g.sight=Math.max(0,g.sight-dt*2.5);
    g.alert=Math.max(0,g.alert-dt);
    var chasing=g.state==='chase';
    if(!chasing&&(g.sight>0.18||g.alert>0)){g.state='chase';sfxSpot();}
    else if(chasing&&g.sight<=0&&g.alert<=0){g.state='patrol';}
    g.speed=g.state==='chase'?sp.chase:sp.patrol;
    moveEnt(g,dt,ghostDecide);

    if(p.invuln<=0&&Math.hypot(g.x-p.x,g.y-p.y)<cell*0.55){onCaught();break;}
  }

  if(dotsLeft<=0&&state==='playing'){
    score+=100*level;if(score>hi){hi=score;saveHi();}
    sfxLevel();state='levelup';bannerT=1.7;
  }
}
