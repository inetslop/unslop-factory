function beep(f,d,type,vol,slide){
  var a=audio(); if(!a)return;
  type=type||'square'; vol=vol||0.08; d=d||0.08; slide=slide||0;
  try{
    var o=a.createOscillator(), g=a.createGain(), t=a.currentTime;
    o.type=type; o.frequency.setValueAtTime(f,t);
    if(slide)o.frequency.exponentialRampToValueAtTime(Math.max(30,f+slide),t+d);
    g.gain.setValueAtTime(vol,t);
    g.gain.exponentialRampToValueAtTime(0.001,t+d);
    o.connect(g); g.connect(a.destination);
    o.start(t); o.stop(t+d+0.02);

function collide(p,ox,oy){
  for(var i=0;i<p.cells.length;i++){
    var x=p.x+p.cells[i][0]+(ox||0), y=p.y+p.cells[i][1]+(oy||0);
    if(x<0||x>=COLS||y>=ROWS) return true;
    if(y>=0&&board[y][x]) return true;
  }
  return false;
}

function spawnPiece(){
  refill();
  piece=makePiece(nextQ.shift());
  refill();
  canHold=true; lockResets=0; grounded=false; gravTimer=0; lockTimer=0;
  if(collide(piece,0,0)){ gameOver(); }
}

function gameOver(){ state='over'; saveHigh(); sOver(); }

function tryMove(dx,dy){
  if(state!=='play'||!piece)return false;
  if(!collide(piece,dx,dy)){
    piece.x+=dx; piece.y+=dy;
    if(dy>0)grounded=false;
    if(lockResets<15&&grounded)lockTimer=0;
    sMove(); return true;
  }
  return false;
}

function tryRotate(){
  if(state!=='play'||!piece)return;
  var p=piece, s=p.s;
  var nc=p.cells.map(function(c){return [s-1-c[1],c[0]];});
  var kicks=[[0,0],[-1,0],[1,0],[0,-1],[-1,-1],[1,-1],[-2,0],[2,0]];
  for(var i=0;i<kicks.length;i++){
    var test={cells:nc,x:p.x+kicks[i][0],y:p.y+kicks[i][1]};
    if(!collide(test,0,0)){ p.cells=nc; p.x+=kicks[i][0]; p.y+=kicks[i][1]; sRotate(); if(lockResets<15)lockTimer=0; return; }
  }
}

function ghostY(){
  var gy=0;
  while(!collide(piece,0,gy+1))gy++;
  return piece.y+gy;
}

function doHold(){
  if(state!=='play'||!canHold||!piece)return;
  canHold=false; sHold();
  var cur=piece.type;
  if(holdType){
    var t=holdType; holdType=cur; piece=makePiece(t);
    if(collide(piece,0,0)){gameOver();return;}
  } else {
    holdType=cur; spawnPiece();
  }
  grounded=false; gravTimer=0; lockTimer=0;
}

function lockPiece(){
  var topOut=false;
  for(var i=0;i<piece.cells.length;i++){
    var x=piece.x+piece.cells[i][0], y=piece.y+piece.cells[i][1];
    if(y<HIDDEN)topOut=true;
    if(y>=0)board[y][x]=piece.type;
  }
  // particles from locked cells
  for(var i=0;i<piece.cells.length;i++){
    var x=piece.x+piece.cells[i][0], y=piece.y+piece.cells[i][1];
    spawnParticles(BOARD_X+x*CELL+CELL/2, BOARD_Y+(y-HIDDEN)*CELL+CELL/2, COLORS[piece.type], 3);
  }
  sDrop();
  if(topOut){ piece=null; gameOver(); return; }
  clearLines();
  spawnPiece();
}

function clearLines(){
  var full=[];
  for(var r=0;r<ROWS;r++){
    var ok=true;
    for(var c=0;c<COLS;c++)if(!board[r][c]){ok=false;break;}
    if(ok)full.push(r);
  }
  if(!full.length)return;
  sClear(full.length);
  flash=Math.min(1,full.length/4+0.3);
  for(var f=0;f<full.length;f++){
    var r=full[f];
    for(var c=0;c<COLS;c++){
      spawnParticles(BOARD_X+c*CELL+CELL/2, BOARD_Y+(r-HIDDEN)*CELL+CELL/2, COLORS[board[r][c]]||'#fff', 4);
    }
  }
  full.sort(function(a,b){return a-b;});
  for(var f=0;f<full.length;f++){
    board.splice(full[f],1);
    board.unshift(new Array(COLS).fill(null));
  }
  var pts=[0,100,300,500,800][full.length]*level;
  score+=pts; lines+=full.length;
  level=Math.floor(lines/10)+1;
}

    particles.push({x:x,y:y,vx:Math.cos(a)*sp,vy:Math.sin(a)*sp-40,life:0.5+Math.random()*0.4,color:color});

}

function update(dt){
  if(flash>0)flash=Math.max(0,flash-dt*2.5);
  for(var i=particles.length-1;i>=0;i--){
    var p=particles[i];

  // DAS repeat
  var dir=null;
  if(keys.left)dir=-1; else if(keys.right)dir=1;
  if(dir!==null){
    repeatT+=dt*1000;
    if(repeatT>160 && Math.floor((repeatT-160)/45)!==Math.floor(((repeatT-dt*1000)-160)/45)){ tryMove(dir,0); }
  }

  var g=gravityMs();
  if(keys.down)g=Math.min(g,45);
  gravTimer+=dt*1000;
  while(gravTimer>=g){
    gravTimer-=g;

    else { grounded=true; break; }
  }

  }

function drawMini(type,cx,cy,s){
  var sh=SHAPES[type], col=COLORS[type];
  var cells=sh.c;
  // center within bounding box
  var minX=9,maxX=-9,minY=9,maxY=-9;
  for(var i=0;i<cells.length;i++){
    minX=Math.min(minX,cells[i][0]);maxX=Math.max(maxX,cells[i][0]+1);
    minY=Math.min(minY,cells[i][1]);maxY=Math.max(maxY,cells[i][1]+1);
  }
  var w=(maxX-minX)*s, h=(maxY-minY)*s;
  var ox=cx-w/2-minX*s, oy=cy-h/2-minY*s;
  for(var i=0;i<cells.length;i++){
    ctx.fillStyle=col;
    ctx.fillRect(ox+cells[i][0]*s+1,oy+cells[i][1]*s+1,s-2,s-2);
    ctx.fillStyle='rgba(255,255,255,0.3)';
    ctx.fillRect(ox+cells[i][0]*s+1,oy+cells[i][1]*s+1,s-2,3);
  }
}

function roundRect(x,y,w,h,r){
  ctx.beginPath();
  ctx.moveTo(x+r,y);
  ctx.arcTo(x+w,y,x+w,y+h,r);
  ctx.arcTo(x+w,y+h,x,y+h,r);
  ctx.arcTo(x,y+h,x,y,r);
  ctx.arcTo(x,y,x+w,y,r);
  ctx.closePath();
}

    ctx.fillRect(p.x-3,p.y-3,6,6);
  }
  ctx.globalAlpha=1;

  // left panel: hold
  panelBox(15,BOARD_Y,125,90,'HOLD');
  if(holdType)drawMini(holdType,15+62.5,BOARD_Y+55,18);

  // right panel: next + stats
  panelBox(BOARD_X+COLS*CELL+25,BOARD_Y,130,240,'NEXT');
  for(var n=0;n<3&&n<nextQ.length;n++)drawMini(nextQ[n],BOARD_X+COLS*CELL+90,BOARD_Y+55+n*70,16);

  panelBox(BOARD_X+COLS*CELL+25,BOARD_Y+260,130,190,'STATS');
  ctx.fillStyle='#e8ecff'; ctx.font='bold 14px monospace'; ctx.textAlign='left';
  var sx=BOARD_X+COLS*CELL+40;
  ctx.fillText('SCORE',sx,BOARD_Y+295);
  ctx.fillStyle='#7ef3ff'; ctx.font='bold 18px monospace';
  ctx.fillText(''+score,sx,BOARD_Y+316);
  ctx.fillStyle='#e8ecff'; ctx.font='bold 14px monospace';
  ctx.fillText('LEVEL',sx,BOARD_Y+345);
  ctx.fillStyle='#ffd52e';
  ctx.fillText(''+level,sx,BOARD_Y+366);
  ctx.fillStyle='#e8ecff'; ctx.font='bold 14px monospace';
  ctx.fillText('LINES',sx,BOARD_Y+395);
  ctx.fillStyle='#4dff88';
  ctx.fillText(''+lines,sx,BOARD_Y+416);
  ctx.fillStyle='#c76bff'; ctx.font='bold 12px monospace';
  ctx.fillText('BEST '+highScore,sx,BOARD_Y+438);

  // flash on line clear
  if(flash>0){
    ctx.globalAlpha=flash*0.5;
    ctx.fillStyle='#ffffff';
    ctx.fillRect(BOARD_X-8,BOARD_Y-8,COLS*CELL+16,(ROWS-HIDDEN)*CELL+16);
    ctx.globalAlpha=1;
  }

  // title
  ctx.textAlign='center';
  drawTitle(GW/2,25,'PUZZLE TETRIS');

  /* overlays */
  if(state==='start'){
    overlay();
    ctx.fillStyle='#e8ecff'; ctx.font='bold 16px monospace';
    var cy=BOARD_Y+140;
    var linesTxt=['← →   Move','↑ / X  Rotate','↓      Soft drop','SPACE  Hard drop','C      Hold piece','P      Pause'];
    for(var i=0;i<linesTxt.length;i++){
      ctx.fillStyle=i%2?'#9fb0ff':'#e8ecff';
      ctx.fillText(linesTxt[i],GW/2,cy+i*30);
    }
    blinkText(GW/2,BOARD_Y+400,'PRESS ENTER TO START');
    if(highScore>0){ctx.fillStyle='#c76bff';ctx.font='bold 14px monospace';ctx.fillText('HIGH SCORE: '+highScore,GW/2,BOARD_Y+440);}
  }
  else if(state==='pause'){
    overlay();
    ctx.fillStyle='#ffd52e'; ctx.font='bold 34px monospace';
    ctx.fillText('PAUSED',GW/2,BOARD_Y+300);
    blinkText(GW/2,BOARD_Y+345,'PRESS P TO RESUME');
  }
  else if(state==='over'){
    overlay();
    ctx.fillStyle='#ff4d5e'; ctx.font='bold 38px monospace';
    ctx.fillText('GAME OVER',GW/2,BOARD_Y+280);
    ctx.fillStyle='#e8ecff'; ctx.font='bold 18px monospace';
    ctx.fillText('SCORE: '+score,GW/2,BOARD_Y+325);
    if(score>=highScore&&score>0){ctx.fillStyle='#ffd52e';ctx.fillText('★ NEW HIGH SCORE ★',GW/2,BOARD_Y+355);}
    else{ctx.fillStyle='#c76bff';ctx.fillText('BEST: '+highScore,GW/2,BOARD_Y+355);}
    blinkText(GW/2,BOARD_Y+405,'PRESS ENTER / R TO RESTART');
  }

  function overlay(){
    ctx.fillStyle='rgba(5,6,15,0.82)';
    ctx.fillRect(BOARD_X-8,BOARD_Y-8,COLS*CELL+16,(ROWS-HIDDEN)*CELL+16);
  }
}

function drawTitle(x,y,text){
  ctx.font='bold 26px monospace';
  var colors=['#29e6ff','#ffd52e','#c76bff','#4dff88','#ff4d5e','#4d7bff','#ff9a3d'];
  var total=0; for(var i=0;i<text.length;i++)total+=ctx.measureText(text[i]).width;
  var cx=x-total/2; ctx.textAlign='left';
  for(var i=0;i<text.length;i++){
    if(text[i]===' '){cx+=ctx.measureText(' ').width;continue;}
    ctx.fillStyle=colors[(i+Math.floor(performance.now()/300))%colors.length];
    ctx.fillText(text[i],cx,y);
    cx+=ctx.measureText(text[i]).width;
  }
  ctx.textAlign='center';
}
function blinkText(x,y,text){
  if(Math.floor(performance.now()/500)%2)return;
  ctx.fillStyle='#7ef3ff'; ctx.font='bold 18px monospace';
  ctx.fillText(text,x,y);
}

/* ---------- main loop ---------- */
function loop(now){
  var dt=Math.min(0.05,(now-lastTime)/1000);
  lastTime=now;
  update(dt);
  draw();
  requestAnimationFrame(loop);
}
requestAnimationFrame(loop);
})();
