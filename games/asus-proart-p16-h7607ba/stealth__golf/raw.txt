function inRect(px,py,r){ return px>=r.x&&px<=r.x+r.w&&py>=r.y&&py<=r.y+r.h; }
function losBlocked(x1,y1,x2,y2){
  const n=32;

function collideWalls(){
  for(const r of walls){
    const cx=Math.max(r.x,Math.min(ball.x,r.x+r.w));
    const cy=Math.max(r.y,Math.min(ball.y,r.y+r.h));

  }
}

function update(dt){
  time+=dt;

    while(angDiff>Math.PI)angDiff-=2*Math.PI;

    ball.vx*=f; ball.vy*=f;

  }
}

    roundRect(r.x,r.y,r.w,r.h,5);ctx.fill();

    roundRect(r.x+2,r.y+2,Math.max(1,r.w-4),Math.min(5,r.h/2),3);ctx.fill();
  }

  // guards
  for(const g of guards){
    ctx.save();ctx.translate(g.x,g.y);
    ctx.fillStyle=g.seen?'#ff5040':'#f1c40f';
    ctx.beginPath();ctx.arc(0,0,13,0,7);ctx.fill();
    ctx.strokeStyle='#111827';ctx.lineWidth=2.5;ctx.stroke();
    ctx.rotate(g.ang);
    ctx.fillStyle='#111827';
    ctx.beginPath();ctx.moveTo(4,-6);ctx.lineTo(15,0);ctx.lineTo(4,6);ctx.closePath();ctx.fill();
    ctx.restore();
    if(time-lastSeen<0.6||g.seen){
      ctx.fillStyle='#ff5040';ctx.font='bold 26px monospace';ctx.textAlign='center';
      ctx.fillText('!',g.x,g.y-22);
    }
  }

  // aim indicator
  if(state==='play'&&atRest()){
    let ax=0,ay=0,power=0;
    if(mouse.down){
      const dx=ball.x-mouse.x, dy=ball.y-mouse.y, d=Math.hypot(dx,dy);
      if(d>4){ax=dx/d;ay=dy/d;power=Math.min(1,d/260);}
    } else if(kbDir){
      ax=kbDir.x;ay=kbDir.y;
      power=charging?Math.min(1,chargeT*1.3):0.25;
    }
    if(power>0.02){
      const len=40+power*160;
      ctx.strokeStyle=`rgba(255,255,255,${0.5+power*0.5})`;
      ctx.lineWidth=3;ctx.setLineDash([8,7]);
      ctx.beginPath();ctx.moveTo(ball.x,ball.y);ctx.lineTo(ball.x+ax*len,ball.y+ay*len);ctx.stroke();
      ctx.setLineDash([]);
      // arrow head
      const hx=ball.x+ax*len,hy=ball.y+ay*len;
      ctx.fillStyle='rgba(255,255,255,0.9)';
      ctx.beginPath();
      ctx.moveTo(hx+ax*12,hy+ay*12);
      ctx.lineTo(hx-ay*7,hy+ax*7);ctx.lineTo(hx+ay*7,hy-ax*7);
      ctx.closePath();ctx.fill();
      // power bar near ball
      ctx.fillStyle='rgba(0,0,0,0.5)';roundRect(ball.x-30,ball.y+22,60,8,4);ctx.fill();
      ctx.fillStyle=power>0.8?'#ff5040':power>0.5?'#f1c40f':'#2ecc71';
      roundRect(ball.x-28,ball.y+24,56*power,4,2);ctx.fill();
    }
  }

  // ball
  if(!(sinkT>0.5)){
    const bs=sinkT>0?Math.max(0.1,1-sinkT/0.7):1;
    ctx.fillStyle='rgba(0,0,0,0.3)';
    ctx.beginPath();ctx.ellipse(ball.x+2,ball.y+4,ball.r*bs,ball.r*0.6*bs,0,0,7);ctx.fill();
    const bg=ctx.createRadialGradient(ball.x-3,ball.y-4,1,ball.x,ball.y,ball.r*bs);
    bg.addColorStop(0,'#ffffff');bg.addColorStop(1,'#c8d2e0');
    ctx.fillStyle=bg;
    ctx.beginPath();ctx.arc(ball.x,ball.y,ball.r*bs,0,7);ctx.fill();
  }

  drawHUD();
}

function panel(x,y,w,h){
  ctx.fillStyle='rgba(10,16,30,0.82)';
  roundRect(x,y,w,h,10);ctx.fill();
  ctx.strokeStyle='rgba(255,255,255,0.15)';ctx.lineWidth=1.5;ctx.stroke();
}
function centerText(txt,y,size,color,font){
  ctx.font=(font||'bold ')+size+'px monospace';
  ctx.textAlign='center';ctx.fillStyle=color||'#fff';
  ctx.fillText(txt,W/2,y);
}
