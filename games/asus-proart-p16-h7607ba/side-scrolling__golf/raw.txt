let AC=null, noiseBuf=null;
function initAudio(){
  if(AC) return;

function tone(freq,dur,type,vol,slideTo){
  if(!AC) return;
  const t=AC.currentTime, o=AC.createOscillator(), g=AC.createGain();
  o.type=type||'sine'; o.frequency.setValueAtTime(freq,t);

}

function groundY(level,x){
  const g=level.ground;
  if(x<=g[0][0]) return g[0][1];
  for(let i=0;i<g.length-1;i++){
    const a=g[i], b=g[i+1];
    if(x>=a[0] && x<=b[0]){
      const t=(x-a[0])/(b[0]-a[0]);
      return a[1]+(b[1]-a[1])*t;
    }
  }
  return g[g.length-1][1];
}

function onSand(L,x){ return L.sand.some(s=>x>=s.x1&&x<=s.x2); }
function inWater(L,x,y){ return L.water.some(w=>x>=w.x1&&x<=w.x2 && y>w.y-8); }

function physics(dt){
  const L=LEVELS[levelIdx];
  if(state==='charging'){
    chargeT+=dt;
    power=(Math.sin(chargeT*4.5 - Math.PI/2)+1)/2 * 100; // ping-pong-ish ramp
  }
  if(state==='flying'){
    ball.vy += 900*dt;
    ball.x += ball.vx*dt;
    ball.y += ball.vy*dt;

    // walls
    if(ball.x<ball.r){ ball.x=ball.r; ball.vx=-ball.vx*0.5; }
    if(ball.x>L.width-ball.r){ ball.x=L.width-ball.r; ball.vx=-ball.vx*0.5; }

    const sand=onSand(L,ball.x);
    // water
    if(inWater(L,ball.x,ball.y)){
      sfx.splash();
      for(let i=0;i<14;i++) particles.push({x:ball.x,y:L.water.find(w=>xRange(w,ball.x)).y,vx:(Math.random()-0.5)*200,vy:-Math.random()*260-60,life:0.6,c:'#bfe8ff'});
      ball.x=lastRest.x; ball.y=lastRest.y; ball.vx=0; ball.vy=0;
      strokes++; // penalty
      state='aim'; power=0;
      bannerT=1.4; holeBanner="Splash! +1";
      return;
    }
    // trees (foliage circle)
    for(const t of L.trees){
      const cy=groundY(L,t.x)-t.r*0.95;
      const dx=ball.x-t.x, dy=ball.y-cy, d=Math.hypot(dx,dy), rr=t.r+ball.r;
      if(d<rr && d>0.001){
        const nx=dx/d, ny=dy/d;
        ball.x=t.x+nx*rr; ball.y=cy+ny*rr;
        const dot=ball.vx*nx+ball.vy*ny;
        ball.vx=(ball.vx-2*dot*nx)*0.45;
        ball.vy=(ball.vy-2*dot*ny)*0.45;
        sfx.bounce(Math.abs(dot));
      }
    }
    // ground
    const gy=groundY(L,ball.x);
    if(ball.y+ball.r>gy){
      ball.y=gy-ball.r;
      if(ball.vy>60) sfx.bounce(ball.vy);
      ball.vy = -ball.vy*(sand?0.25:0.45);
      ball.vx *= (sand?0.7:0.85);
      if(Math.abs(ball.vy)<40) ball.vy=0;
    }
    // rolling friction
    if(ball.y+ball.r>=gy-1 && Math.abs(ball.vy)<1){
      const f = sand?4.2:1.3;
      ball.vx -= ball.vx*f*dt;
    }
    // hole capture
    const hx=L.hole.x, hgy=groundY(L,hx);
    if(Math.abs(ball.x-hx)<13 && Math.abs(ball.y+ball.r-hgy)<8){
      const sp=Math.hypot(ball.vx,ball.vy);
      if(sp<280){
        state='sinking'; sinkT=0;
        ball.vx=0; ball.vy=0;
        sfx.sink();
        for(let i=0;i<16;i++) particles.push({x:hx,y:hgy,vx:(Math.random()-0.5)*260,vy:-Math.random()*300-80,life:0.7,c:['#ffd700','#fff','#ff6b6b'][i%3]});
        return;
      }
    }
    // rest
    const sp=Math.hypot(ball.vx,ball.vy);
    if(sp<7 && ball.y+ball.r>=gy-1.5){
      ball.vx=0; ball.vy=0; ball.y=gy-ball.r;
      lastRest={x:ball.x,y:ball.y};
      state='aim'; power=0;
    }
  }
  if(state==='sinking'){
    sinkT+=dt;
    const L2=LEVELS[levelIdx], hx=L2.hole.x, hgy=groundY(L2,hx);
    ball.x += (hx-ball.x)*Math.min(1,dt*10);
    ball.y += ((hgy-2)-ball.y)*Math.min(1,dt*10);
    if(sinkT>0.5 && state==='sinking'){
      const par=L2.par;
      let word;
      const d=strokes-par;
      if(d<=-2)word="EAGLE!"; else if(d===-1)word="BIRDIE!"; else if(d===0)word="PAR";
      else if(d===1)word="BOGEY"; else if(d===2)word="DOUBLE BOGEY"; else word="+"+d;
      holeResults.push({name:L2.name,par:par,strokes:strokes});
      holeBanner=word+"  ("+strokes+" strokes)";
      bannerT=99; state='holed';
    }
  }
  // particles
  for(let i=particles.length-1;i>=0;i--){
    const p=particles[i];
    p.life-=dt; p.x+=p.vx*dt; p.y+=p.vy*dt; p.vy+=800*dt;
    if(p.life<=0) particles.splice(i,1);
  }
  if(bannerT<98 && bannerT>0) bannerT-=dt;

  // aim keys
  if(state==='aim'||state==='charging'){
    const rot=1.6*dt;
    if(keys['ArrowLeft']||keys['KeyA']) aimAngle-=rot;
    if(keys['ArrowRight']||keys['KeyD']) aimAngle+=rot;
    if(keys['ArrowUp']||keys['KeyW']) aimAngle=Math.max(aimAngle-3.2*dt, -Math.PI+0.15);
    if(keys['ArrowDown']||keys['KeyS']) aimAngle=Math.min(aimAngle+3.2*dt, -0.15);
  }

  // camera
  const viewW=W/SCALE;
  let target=ball.x;
  camX += (target-camX)*Math.min(1,dt*4);
  if(viewW<L.width) camX=Math.max(0,Math.min(L.width-viewW,camX));
  else camX=(L.width-viewW)/2;
}

  for(let sx=0;sx<=W;sx+=20){
    const wxp=sx/SCALE+camX*0.3;
    ctx.lineTo(sx, wy(300+Math.sin(wxp*0.004)*60+Math.sin(wxp*0.011)*30));
  }
  ctx.lineTo(W,H); ctx.closePath(); ctx.fill();

  // ground
  const g=L.ground;
  ctx.beginPath();
  let started=false;
  for(let wxp=Math.floor(camX-50/SCALE); wxp<=camX+W/SCALE+50; wxp+=16){
    const sx=wx(wxp), sy=wy(groundY(L,wxp));
    if(!started){ctx.moveTo(sx,sy);started=true;} else ctx.lineTo(sx,sy);
  }
  ctx.lineTo(W+5,H+5); ctx.lineTo(-5,H+5); ctx.closePath();
  const gg=ctx.createLinearGradient(0,wy(300),0,H);
  gg.addColorStop(0,'#4caf50'); gg.addColorStop(1,'#2e7d32');
  ctx.fillStyle=gg; ctx.fill();
  // top edge line + mow stripes
  ctx.save();
  ctx.clip();
  ctx.strokeStyle='#8bd98b'; ctx.lineWidth=3*SCALE;
  ctx.beginPath();
  started=false;
  for(let wxp=Math.floor(camX-50/SCALE); wxp<=camX+W/SCALE+50; wxp+=16){
    const sx=wx(wxp), sy=wy(groundY(L,wxp));
    if(!started){ctx.moveTo(sx,sy);started=true;} else ctx.lineTo(sx,sy);
  }
  ctx.stroke();
  ctx.fillStyle='rgba(255,255,255,0.06)';
  for(let wxp=Math.floor(camX/80)*80; wxp<camX+W/SCALE+80; wxp+=160){
    const sx1=wx(wxp), sx2=wx(wxp+80);
    ctx.fillRect(sx1,wy(250),sx2-sx1,H);
  }
  ctx.restore();

  // sand traps
  for(const s of L.sand){
    ctx.fillStyle='#e8d48b';
    ctx.beginPath();
    let st=false;
    for(let x=s.x1;x<=s.x2;x+=12){
      const sx=wx(x), sy=wy(groundY(L,x));
      if(!st){ctx.moveTo(sx,sy);st=true;} else ctx.lineTo(sx,sy);
    }
    ctx.lineTo(wx(s.x2),wy(groundY(L,s.x2)+40));
    ctx.lineTo(wx(s.x1),wy(groundY(L,s.x1)+40));
    ctx.closePath(); ctx.fill();
    ctx.fillStyle='rgba(160,130,60,0.5)';
    for(let i=0;i<20;i++){
      const x=s.x1+((i*73)%(s.x2-s.x1));
      ctx.fillRect(wx(x),wy(groundY(L,x)+4+((i*37)%20)),2,2);
    }
  }
  // water
  for(const w of L.water){
    const wyTop=wy(w.y);
    const wg=ctx.createLinearGradient(0,wyTop,0,H);
    wg.addColorStop(0,'#4fc3f7'); wg.addColorStop(1,'#0277bd');
    ctx.fillStyle=wg;
    ctx.fillRect(wx(w.x1),wyTop,wx(w.x2)-wx(w.x1),H-wyTop+5);
    ctx.strokeStyle='rgba(255,255,255,0.6)'; ctx.lineWidth=2;
    for(let i=0;i<4;i++){
      const yy=wyTop+8+i*10+Math.sin(time*3+i)*2;
      ctx.beginPath();
      ctx.moveTo(wx(w.x1)+6,yy);
      ctx.lineTo(wx(w.x2)-6,yy);
      ctx.stroke();
    }
  }
  // trees
  for(const t of L.trees){
    const gx=wx(t.x), gy=wy(groundY(L,t.x));
    const cy=gy - t.r*0.95*SCALE;
    ctx.fillStyle='#6d4c2b';
    ctx.fillRect(gx-4*SCALE, gy-t.r*1.3*SCALE, 8*SCALE, t.r*1.3*SCALE);
    ctx.fillStyle='#2e7d32';
    ctx.beginPath(); ctx.arc(gx,cy,t.r*SCALE,0,7); ctx.fill();
    ctx.fillStyle='#43a047';
    ctx.beginPath(); ctx.arc(gx-t.r*0.3*SCALE,cy-t.r*0.25*SCALE,t.r*0.6*SCALE,0,7); ctx.fill();
  }
  // hole + flag
  const hx=wx(L.hole.x), hgy=wy(groundY(L,L.hole.x));
  ctx.fillStyle='#1b1b1b';
  ctx.beginPath(); ctx.ellipse(hx,hgy,13*SCALE,4.5*SCALE,0,0,7); ctx.fill();
  ctx.strokeStyle='#fff'; ctx.lineWidth=2;
  ctx.beginPath(); ctx.moveTo(hx,hgy-90*SCALE); ctx.lineTo(hx,hgy); ctx.stroke();
  const wave=Math.sin(time*4)*6*SCALE;
  ctx.fillStyle='#e53935';
  ctx.beginPath();
  ctx.moveTo(hx,hgy-90*SCALE);
  ctx.lineTo(hx+38*SCALE+wave,hgy-78*SCALE);
  ctx.lineTo(hx,hgy-64*SCALE);
  ctx.closePath(); ctx.fill();

  // particles
  for(const p of particles){
    ctx.globalAlpha=Math.max(0,p.life*1.5);
    ctx.fillStyle=p.c;
    ctx.fillRect(wx(p.x)-2,wy(p.y)-2,4,4);
  }
  ctx.globalAlpha=1;

  // ball shadow
  const bgy=groundY(L,ball.x);
  if(ball.y < bgy-ball.r){
    const h=(bgy-ball.y)/200;
    ctx.fillStyle='rgba(0,0,0,'+Math.max(0.05,0.3-h*0.15)+')';
    ctx.beginPath();
    ctx.ellipse(wx(ball.x),wy(bgy)-2, (ball.r*SCALE)*(1+h*0.5),(ball.r*0.4)*SCALE*(1+h*0.5),0,0,7);
    ctx.fill();
  }
  // ball
  const scale = state==='sinking' ? Math.max(0.05,1-sinkT*2) : 1;
  if(!(state==='holed')){
    const bx=wx(ball.x), by=wy(ball.y);
    ctx.fillStyle='#fff';
    ctx.beginPath(); ctx.arc(bx,by,ball.r*SCALE*scale,0,7); ctx.fill();
    ctx.strokeStyle='rgba(0,0,0,0.25)'; ctx.lineWidth=1;
    ctx.beginPath(); ctx.arc(bx,by,ball.r*SCALE*scale,0,7); ctx.stroke();
    ctx.fillStyle='rgba(0,0,0,0.3)';
    for(let i=0;i<4;i++){
      const a=i*1.6+time;
      ctx.fillRect(bx+Math.cos(a)*3*scale-1,by+Math.sin(a)*3*scale-1,2,2);
    }
  }

  // aim arrow
  if(state==='aim'||state==='charging'){
    let ang=aimAngle, pw=power;
    if(mouseAim){
      const dx=mouseAim.sx-mouseAim.cx, dy=mouseAim.sy-mouseAim.cy;
      const len=Math.hypot(dx,dy);
      if(len>10){ ang=Math.atan2(dy,dx); pw=Math.min(100,len/1.6); }
    }
    const bx=wx(ball.x), by=wy(ball.y);
    const len=(30+pw*1.1)*SCALE;
    const ex=bx+Math.cos(ang)*len, ey=by+Math.sin(ang)*len;
    ctx.strokeStyle='rgba(255,255,255,0.9)'; ctx.lineWidth=3;
    ctx.setLineDash([8,6]);
    ctx.beginPath(); ctx.moveTo(bx,by); ctx.lineTo(ex,ey); ctx.stroke();
    ctx.setLineDash([]);
    // arrowhead
    ctx.fillStyle='rgba(255,255,255,0.9)';
    ctx.save(); ctx.translate(ex,ey); ctx.rotate(ang);
    ctx.beginPath(); ctx.moveTo(10,0); ctx.lineTo(-4,-6); ctx.lineTo(-4,6); ctx.closePath(); ctx.fill();
    ctx.restore();
  }

  /* ---- HUD ---- */
  const par=L.par;
  hudText("Hole "+(levelIdx+1)+"/"+LEVELS.length+": "+L.name, 16, 30, 'bold 20px sans-serif', '#fff');
  hudText("Par "+par+"   Strokes: "+strokes, 16, 56, '16px sans-serif', '#fff');
  if(holeResults.length){
    const tot=holeResults.reduce((a,h)=>a+h.strokes,0);
    const tp=holeResults.reduce((a,h)=>a+h.par,0);
    hudText("So far: "+tot+" ("+fmtDiff(tot-tp)+")", W-16, 30, 'bold 18px sans-serif', '#fff', 'right');
  }
  // power bar
  if(state==='charging'){
    const bx=24, by=H-52, bw=220, bh=18;
    ctx.fillStyle='rgba(0,0,0,0.35)';
    roundRect(bx-4,by-4,bw+8,bh+8,6); ctx.fill();
    const pg=ctx.createLinearGradient(bx,0,bx+bw,0);
    pg.addColorStop(0,'#4caf50'); pg.addColorStop(0.6,'#ffc107'); pg.addColorStop(1,'#e53935');
    ctx.fillStyle=pg;
    roundRect(bx,by,bw*power/100,bh,4); ctx.fill();
    ctx.strokeStyle='#fff'; ctx.lineWidth=2;
    roundRect(bx,by,bw,bh,4); ctx.stroke();
    hudText("POWER "+Math.round(power)+"%", bx, by-10, 'bold 14px sans-serif', '#fff');
  }
  if(state==='aim'){
    hudText("← → aim   ↑ ↓ fine-tune   hold SPACE to charge & release to hit  (or drag with mouse)", W/2, H-20, '14px sans-serif','rgba(255,255,255,0.95)','center');
  }
  // banner
  if(bannerT>0 && state!=='start' && state!=='gameover'){
    ctx.save();
    ctx.globalAlpha=Math.min(1,bannerT);
    ctx.fillStyle='rgba(0,0,0,0.55)';
    roundRect(W/2-170,H*0.3-36,340,72,14); ctx.fill();
    hudText(holeBanner, W/2, H*0.3+8, 'bold 30px sans-serif', '#ffd700','center');
    ctx.restore();
  }

  if(state==='holed'){
    panel(W/2,H/2+40, 360, 120);
    hudText(holeBanner, W/2, H/2+10, 'bold 28px sans-serif','#ffd700','center');
    hudText("Press ENTER / click for "+(levelIdx+1<LEVELS.length?"next hole":"results"), W/2, H/2+45,'15px sans-serif','#fff','center');
  }

  if(state==='start') drawStart();
  if(state==='gameover') drawGameOver();
}
function fmtDiff(d){ return d===0?'E':(d>0?'+':'-')+Math.abs(d); }
function hudText(t,x,y,font,color,align){
  ctx.font=font; ctx.textAlign=align||'left'; ctx.textBaseline='middle';
  ctx.fillStyle='rgba(0,0,0,0.45)'; ctx.fillText(t,x+2,y+2);
  ctx.fillStyle=color; ctx.fillText(t,x,y);
}
function roundRect(x,y,w,h,r){
  if(w<1)w=1;
  ctx.beginPath();
  ctx.moveTo(x+r,y); ctx.arcTo(x+w,y,x+w,y+h,r); ctx.arcTo(x+w,y+h,x,y+h,r);
  ctx.arcTo(x,y+h,x,y,r); ctx.arcTo(x,y,x+w,y,r); ctx.closePath();
}
function panel(cx,cy,w,h){
  ctx.fillStyle='rgba(0,0,0,0.55)';
  roundRect(cx-w/2,cy-h/2,w,h,14); ctx.fill();
  ctx.strokeStyle='rgba(255,255,255,0.3)'; ctx.lineWidth=2;
  roundRect(cx-w/2,cy-h/2,w,h,14); ctx.stroke();
}
function drawClouds(par,speed,y,alpha){
  ctx.fillStyle='rgba(255,255,255,'+alpha+')';
  const off=(camX*par*SCALE + speed)% (W+300);
  for(let i=0;i<4;i++){
    let cx=((i*(W+300)/4) - off)%(W+300); if(cx<-150)cx+=W+300;
    const cy=y+i*22;
    ctx.beginPath();
    ctx.arc(cx,cy,26,0,7); ctx.arc(cx+28,cy-8,20,0,7); ctx.arc(cx+52,cy,24,0,7);
    ctx.fill();
  }
}
function drawStart(){
  ctx.fillStyle='rgba(10,30,60,0.55)'; ctx.fillRect(0,0,W,H);
  hudText("⛳ FAIRWAY FLING", W/2, H*0.24, 'bold 54px sans-serif','#fff','center');
  hudText("A Side-Scrolling Golf Adventure", W/2, H*0.34,'italic 20px sans-serif','#bfe6ff','center');
  const lines=[
    "← / →  — aim (flip sides & tilt)",
    "↑ / ↓  — fine-tune angle",
    "HOLD SPACE — charge power, RELEASE to hit",
    "MOUSE / TOUCH — drag from the ball like a slingshot",
    "R — restart game anytime",
    "",
    "3 holes. Watch out for water (+1), sand and trees!"
  ];
  lines.forEach((l,i)=>hudText(l, W/2, H*0.46+i*30,'18px sans-serif','#e8f7ff','center'));
  const blink=Math.sin(time*4)>0;
  if(blink) hudText("— Click or press ENTER to play —", W/2, H*0.85,'bold 22px sans-serif','#ffd700','center');
}
function drawGameOver(){
  ctx.fillStyle='rgba(10,30,60,0.7)'; ctx.fillRect(0,0,W,H);
  hudText("FINAL SCORECARD", W/2, H*0.14,'bold 40px sans-serif','#ffd700','center');
  const rows=[["Hole","Par","You"]].concat(holeResults.map((h,i)=>[""+(i+1),""+h.par,""+h.strokes]));
  const tw=320, rh=38, topY=H*0.24;
  ctx.fillStyle='rgba(255,255,255,0.1)';
  roundRect(W/2-tw/2,topY,tw,rows.length*rh+rh,10); ctx.fill();
  rows.forEach((r,i)=>{
    const y=topY+r*i*38;
    if(i===0){ hudText(r[0],W/2-110,y,'bold 16px sans-serif','#fff','center'); hudText(r[1],W/2,y,'bold 16px sans-serif','#fff','center'); hudText(r[2],W/2+110,y,'bold 16px sans-serif','#fff','center'); }
    else{
      const d=r[2]-r[1];
      ctx.fillStyle = d<0?'#8bd98b':(d===0?'#e8f7ff':'#ffb3b3');
      hudText(r[0],W/2-110,y,'16px sans-serif',ctx.fillStyle,'center');
      hudText(r[1],W/2,y,'16px sans-serif',ctx.fillStyle,'center');
      hudText(r[2]+(d===0?'':fmtDiff(d)),W/2+110,y,'16px sans-serif',ctx.fillStyle,'center');
    }
  });
  const tot=holeResults.reduce((a,h)=>a+h.strokes,0);
  const tp=holeResults.reduce((a,h)=>a+h.par,0);
  let verdict = tot<tp ? "You beat par! Champion!" : (tot===tp?"Even par — solid round!":"+"+(tot-tp)+" over par. Keep practicing!");
  hudText("Total: "+tot+"  ("+fmtDiff(tot-tp)+")", W/2, topY+rows.length*38+40,'bold 26px sans-serif','#fff','center');
  hudText(verdict, W/2, topY+rows.length*38+75,'19px sans-serif','#bfe6ff','center');
  if(Math.sin(time*4)>0) hudText("Click or press ENTER to play again   (R restarts)", W/2, H*0.9,'bold 18px sans-serif','#ffd700','center');
}

/* ---------------- LOOP ---------------- */
function loop(t){
  const dt=Math.min(0.033,(t-lastT)/1000||0.016);
  lastT=t; time+=dt;
  physics(dt);
  draw();
  requestAnimationFrame(loop);
}
loadLevel(0); state='start';
requestAnimationFrame(loop);
})();
