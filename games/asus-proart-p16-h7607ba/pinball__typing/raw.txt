function beep(freq, dur, type, vol, slide){
  if(!AC) return;
  try{
    var t = AC.currentTime;
    var o = AC.createOscillator(), g = AC.createGain();
    o.type = type || 'sine';
    o.frequency.setValueAtTime(freq, t);
    if(slide) o.frequency.exponentialRampToValueAtTime(Math.max(30,freq+slide), t+dur);
    g.gain.setValueAtTime(vol||0.15, t);
    g.gain.exponentialRampToValueAtTime(0.001, t+dur);
    o.connect(g); g.connect(AC.destination);
    o.start(t); o.stop(t+dur+0.02);
  }catch(e){}
}

// ---------- Physics helpers ----------
function segHit(b, x1,y1,x2,y2, rest, vfx, vfy){
  vfx=vfx||0; vfy=vfy||0;
  var dx=x2-x1, dy=y2-y1;
  var l2 = dx*dx+dy*dy || 1e-9;
  var t = ((b.x-x1)*dx + (b.y-y1)*dy)/l2;
  t = Math.max(0, Math.min(1, t));
  var qx = x1+t*dx, qy = y1+t*dy;
  var nx=b.x-qx, ny=b.y-qy;
  var d = Math.sqrt(nx*nx+ny*ny);
  if(d < b.r){
    if(d<1e-6){ nx=0; ny=-1; d=1; } else { nx/=d; ny/=d; }
    b.x = qx + nx*b.r; b.y = qy + ny*b.r;
    var rvx=b.vx-vfx, rvy=b.vy-vfy;
    var dot = rvx*nx + rvy*ny;
    if(dot < 0){
      b.vx -= (1+rest)*dot*nx;
      b.vy -= (1+rest)*dot*ny;
      return true;
    }
  }
  return false;
}

}

function stepBall(dt){

  // clamp speed

  if(sp>1500){ ball.vx*=1500/sp; ball.vy*=1500/sp; }
  ball.x += ball.vx*dt;
  ball.y += ball.vy*dt;

  if(ball.x < 16+ball.r){ ball.x=16+ball.r; if(ball.vx<0){ ball.vx=-ball.vx*0.75; beep(200,0.05,'triangle',0.06);} }

  if(ball.y < 16+ball.r){ ball.y=16+ball.r; if(ball.vy<0){ ball.vy=-ball.vy*0.75; beep(200,0.05,'triangle',0.06);} }

function hitFlipper(b, x1,y1,x2,y2, f){
  var dx=x2-x1, dy=y2-y1;
  var l2=dx*dx+dy*dy||1e-9;
  var t=((b.x-x1)*dx+(b.y-y1)*dy)/l2;
  t=Math.max(0,Math.min(1,t));
  var qx=x1+t*dx, qy=y1+t*dy;
  var nx=b.x-qx, ny=b.y-qy;
  var d=Math.sqrt(nx*nx+ny*ny);
  var R = b.r + f.r;
  if(d < R){
    if(d<1e-6){nx=0;ny=-1;d=1;} else {nx/=d;ny/=d;}
    b.x = qx+nx*R; b.y = qy+ny*R;
    // surface velocity from angular motion: v = w * (-ry, rx)
    var rx=qx-x1, ry=qy-y1;
    var vfx = -f.omega*ry, vfy = f.omega*rx;
    var rvx=b.vx-vfx, rvy=b.vy-vfy;
    var dot=rvx*nx+rvy*ny;
    if(dot<0){
      b.vx -= 1.35*dot*nx;
      b.vy -= 1.35*dot*ny;
      beep(700,0.04,'triangle',0.08);
    }
  }
}

// ---------- Update ----------
function update(dt){
  if(state!=='play') return;

  // flippers
  var targets = [flipL, flipR];
  var pressed = [keys.left, keys.right];
  for(var i=0;i<2;i++){
    var f=targets[i];
    var prev=f.angle;
    var tgt = pressed[i] ? -0.55 : 0.45;
    var spd = 18*dt;
    if(f.angle > tgt) f.angle = Math.max(tgt, f.angle-spd);
    else f.angle = Math.min(tgt, f.angle+spd);
    f.omega = (f.angle-prev)/Math.max(dt,1e-6);
  }

  // word timer
  if(wordIdx < word.length){
    wordTime -= dt;
    if(wordTime<=0){
      lives--; combo=0; flashWrong=0.5; shake=8;
      beep(90,0.4,'sawtooth',0.18,-50);
      if(lives<=0){ saveBest(); state='over'; return; }
      pickWord();
    }
  }

  // respawn
  if(!ball.active && lives>0){
    respawnTimer -= dt;
    if(respawnTimer<=0){
      ball.x=W-42; ball.y=130; ball.vx=-70; ball.vy=0; ball.active=true;
    }
  }

  // substeps
  var n=4, sdt=dt/n;
  for(var k=0;k<n;k++) stepBall(sdt);

  // effects decay
  for(var bI=0;bI<bumpers.length;bI++) bumpers[bI].f=Math.max(0,bumpers[bI].f-dt*4);
  for(var sI=0;sI<slings.length;sI++) slings[sI].f=Math.max(0,slings[sI].f-dt*4);
  flashWrong=Math.max(0,flashWrong-dt);
  flashWordDone=Math.max(0,flashWordDone-dt*2);
  shake=Math.max(0,shake-dt*30);

  for(var p=particles.length-1;p>=0;p--){
    var pt=particles[p];
    pt.life-=dt;
    if(pt.life<=0){ particles.splice(p,1); continue; }
    pt.x+=pt.vx*dt; pt.y+=pt.vy*dt; pt.vy+=300*dt;
  }
}

      // word complete!
      var bonus = 100 + word.length*5;
      score += bonus;
      flashWordDone=1;
      burst(240, 70, '#6dff9c', 22);
      beep(660,0.1,'sine',0.16); setTimeout(function(){beep(880,0.12,'sine',0.16);},80); setTimeout(function(){beep(1320,0.15,'sine',0.14);},160);
      // boost the ball
      if(ball.active){ ball.vy = -860; ball.vx += (Math.random()*2-1)*120; }
      pickWord();
    }
  } else {
    combo=0;
    flashWrong=0.3;
    beep(120,0.12,'square',0.14,-40);
    if(ball.active) ball.vx += (Math.random()*2-1)*160;
  }
}

function roundRect(x,y,w,h,r){
  ctx.beginPath();
  ctx.moveTo(x+r,y);
  ctx.arcTo(x+w,y,x+w,y+h,r);
  ctx.arcTo(x+w,y+h,x,y+h,r);
  ctx.arcTo(x,y+h,x,y,r);
  ctx.arcTo(x,y,x+w,y,r);
  ctx.closePath();
}

function draw(){
  var vw=window.innerWidth, vh=window.innerHeight;
  ctx.fillStyle='#070a14';
  ctx.fillRect(0,0,vw,vh);

  ctx.save();
  var shx=(Math.random()*2-1)*shake*0.6, shy=(Math.random()*2-1)*shake*0.6;
  ctx.translate(ox+shx*scale, oy+shy*scale);
  ctx.scale(scale,scale);

  // table
  var g=ctx.createLinearGradient(0,0,0,H);
  g.addColorStop(0,'#141b38'); g.addColorStop(1,'#0d1226');
  roundRect(8,8,W-16,H-16,22); ctx.fillStyle=g; ctx.fill();
  ctx.lineWidth=8; ctx.strokeStyle='#2a3560'; ctx.stroke();

  // funnel guides
  ctx.lineWidth=7; ctx.lineCap='round'; ctx.strokeStyle='#3b4a86';
  ctx.beginPath(); ctx.moveTo(16,H-230); ctx.lineTo(flipL.px-8,H-70); ctx.stroke();
  ctx.beginPath(); ctx.moveTo(W-16,H-230); ctx.lineTo(flipR.px+8,H-70); ctx.stroke();

  // bumpers
  for(var i=0;i<bumpers.length;i++){
    var b=bumpers[i];
    var rad=b.r+b.f*5;
    var bg=ctx.createRadialGradient(b.x,b.y,4,b.x,b.y,rad);
    if(b.f>0.4){ bg.addColorStop(0,'#fff3c0'); bg.addColorStop(1,'#ff9d3a'); }
    else { bg.addColorStop(0,'#5a6fd8'); bg.addColorStop(1,'#28336b'); }
    ctx.beginPath(); ctx.arc(b.x,b.y,rad,0,7); ctx.fillStyle=bg; ctx.fill();
    ctx.lineWidth=4; ctx.strokeStyle=b.f>0.4?'#ffd35a':'#7f92e8'; ctx.stroke();
    ctx.beginPath(); ctx.arc(b.x,b.y,rad*0.45,0,7); ctx.strokeStyle='rgba(255,255,255,0.35)'; ctx.lineWidth=2; ctx.stroke();
  }

  // slingshots
  for(var s=0;s<slings.length;s++){
    var k=slings[s], r=k.r+k.f*4;
    ctx.beginPath(); ctx.arc(k.x,k.y,r,0,7);
    ctx.fillStyle = k.f>0.3 ? '#bdf3ff' : '#1f8ba6';
    ctx.fill();
    ctx.lineWidth=3; ctx.strokeStyle='#5fdcff'; ctx.stroke();
  }

  // flippers
  function drawFlip(f){
    var t=flipTip(f);
    ctx.beginPath();
    ctx.moveTo(f.px,f.py); ctx.lineTo(t.x,t.y);
    ctx.lineWidth=f.r*2; ctx.lineCap='round';
    ctx.strokeStyle='#e8b13c'; ctx.stroke();
    ctx.beginPath();
    ctx.moveTo(f.px,f.py); ctx.lineTo(t.x,t.y);
    ctx.lineWidth=f.r*0.9; ctx.strokeStyle='#ffe08a'; ctx.stroke();
  }
  drawFlip(flipL); drawFlip(flipR);

  // ball
  if(ball.active){
    var bg2=ctx.createRadialGradient(ball.x-4,ball.y-4,2,ball.x,ball.y,ball.r);
    bg2.addColorStop(0,'#ffffff'); bg2.addColorStop(1,'#9fb4ff');
    ctx.beginPath(); ctx.arc(ball.x,ball.y,ball.r,0,7);
    ctx.fillStyle=bg2; ctx.fill();
    ctx.lineWidth=2; ctx.strokeStyle='#5a6fd8'; ctx.stroke();
  } else if(state==='play' && lives>0){
    ctx.fillStyle='rgba(255,255,255,0.5)';
    ctx.font='14px monospace'; ctx.textAlign='center';
    ctx.fillText('ball incoming...', W/2, H-90);
  }

  // particles
  for(var p=0;p<particles.length;p++){
    var pt=particles[p];
    ctx.globalAlpha = Math.max(0, pt.life/pt.max);
    ctx.fillStyle=pt.c;
    ctx.fillRect(pt.x-2,pt.y-2,4,4);
  }
  ctx.globalAlpha=1;

  // ---- HUD ----
  ctx.textAlign='left';
  ctx.font='bold 18px monospace';
  ctx.fillStyle='#cfe0ff';
  ctx.fillText('SCORE '+score, 26, 40);
  if(combo>1){ ctx.fillStyle='#6dff9c'; ctx.font='bold 14px monospace'; ctx.fillText('COMBO x'+combo, 26, 58); }

  // lives
  ctx.textAlign='right'; ctx.font='18px monospace'; ctx.fillStyle='#ff7a9c';
  var hearts=''; for(var h=0;h<lives;h++) hearts+='\u2665 ';
  ctx.fillText(hearts.trim(), W-26, 40);

  // word display
  if(state==='play'){
    var fs = Math.min(40, (W-80)/word.length);
    ctx.textAlign='center';
    ctx.font='bold '+fs+'px monospace';
    var totalW=0;
    // measure approximate widths
    var widths=[]; for(var cI=0;cI<word.length;cI++){ widths.push(ctx.measureText(word[cI]).width); }
    var sumW=0; for(var wI=0;wI<widths.length;wI++) sumW+=widths[wI];
    var cx2 = W/2 - sumW/2;
    for(var cI=0;cI<word.length;cI++){
      if(cI < wordIdx) ctx.fillStyle='#6dff9c';
      else if(cI === wordIdx){
        ctx.fillStyle = flashWrong>0 ? '#ff4d5e' : (Math.floor(performance.now()/250)%2 ? '#ffe14d':'#fff');
      } else ctx.fillStyle='rgba(230,240,255,0.85)';
      ctx.textAlign='left';
      ctx.fillText(word[cI], cx2, 96);
      cx2 += widths[cI];
    }
    // timer bar
    var tw = Math.max(0, wordTime/WORD_TIME)*(W-120);
    ctx.fillStyle='rgba(255,255,255,0.12)';
    roundRect(W/2-(W-120)/2, 108, W-120, 8, 4); ctx.fill();
    if(tw>0){
      ctx.fillStyle = wordTime<2 ? '#ff4d5e' : '#6dff9c';
      roundRect(W/2-(W-120)/2, 108, tw, 8, 4); ctx.fill();
    }
    if(flashWordDone>0){
      ctx.globalAlpha=flashWordDone;
      ctx.textAlign='center';
      ctx.font='bold 26px monospace';
      ctx.fillStyle='#6dff9c';
      ctx.fillText('WORD BONUS! +boost', W/2, 150);
      ctx.globalAlpha=1;
    }
  }

  // ---- overlays ----
  if(state==='start' || state==='over'){
    ctx.fillStyle='rgba(5,8,20,0.78)';
    roundRect(40, H/2-190, W-80, 380, 18); ctx.fill();
    ctx.lineWidth=3; ctx.strokeStyle='#3b4a86'; ctx.stroke();

    ctx.textAlign='center';
    if(state==='start'){
      ctx.font='bold 40px monospace';
      ctx.fillStyle='#ffe14d';
      ctx.fillText('TYPING', W/2, H/2-110);
      ctx.fillStyle='#7ae0ff';
      ctx.fillText('PINBALL', W/2, H/2-68);

      ctx.font='15px monospace';
      ctx.fillStyle='#cfe0ff';
      var lines=[
        '\u2190 / \u2192  flip the flippers',
        'Type the word to score',
        'Complete a word: bonus + ball boost!',
        'Wrong key or timeout costs a life.',
        '3 lives. Don\u2019t drain the ball!'
      ];
      for(var l=0;l<lines.length;l++) ctx.fillText(lines[l], W/2, H/2-14+l*26);

      if(Math.floor(performance.now()/500)%2){
        ctx.font='bold 18px monospace';
        ctx.fillStyle='#6dff9c';
        ctx.fillText('PRESS ENTER TO PLAY', W/2, H/2+130);
      }
    } else {
      ctx.font='bold 34px monospace';
      ctx.fillStyle='#ff7a9c';
      ctx.fillText('GAME OVER', W/2, H/2-110);
      ctx.font='bold 24px monospace';
      ctx.fillStyle='#ffe14d';
      ctx.fillText('SCORE  '+score, W/2, H/2-58);
      ctx.font='16px monospace';
      ctx.fillStyle='#cfe0ff';
      ctx.fillText('BEST  '+best, W/2, H/2-30);

      if(Math.floor(performance.now()/500)%2){
        ctx.font='bold 18px monospace';
        ctx.fillStyle='#6dff9c';
        ctx.fillText('PRESS ENTER OR R TO RESTART', W/2, H/2+40);
      }
    }
  }

  // footer hint
  if(state==='play'){
    ctx.textAlign='center';
    ctx.font='12px monospace';
    ctx.fillStyle='rgba(207,224,255,0.35)';
    ctx.fillText('\u2190 \u2192 flippers   \u00b7   type the word   \u00b7   R restart', W/2, H-16);
  }

  ctx.restore();
}
