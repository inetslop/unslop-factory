function sfx(freq,dur,type,vol){
  if(!actx)return;
  try{
    const o=actx.createOscillator(),g=actx.createGain();
    o.type=type||'square';o.frequency.value=freq;
    g.gain.setValueAtTime(vol||0.12,actx.currentTime);
    g.gain.exponentialRampToValueAtTime(0.0001,actx.currentTime+dur);
    o.connect(g);g.connect(actx.destination);
    o.start();o.stop(actx.currentTime+dur+0.02);
  }catch(e){}
}
function boom(){
  if(!actx)return;
  try{
    const len=Math.floor(actx.sampleRate*0.7);
    const buf=actx.createBuffer(1,len,actx.sampleRate);
    const d=buf.getChannelData(0);

    o.type='sine';o.frequency.setValueAtTime(120,actx.currentTime);

    og.gain.exponentialRampToValueAtTime(0.001,actx.currentTime+0.6);
    o.connect(og);og.connect(actx.destination);o.start();o.stop(actx.currentTime+0.65);
  }catch(e){}
}

// ---------- game data ----------
const COLS=15,ROWS=10,MINES=22,SCAN_R=2,SEE_R=2;
const EXIT={x:COLS-1,y:ROWS-1},START={x:0,y:0};
let state='start'; // start | playing | over

const SENT_PATHS=[
  {pts:[[3,2],[4,2],[5,2],[6,2],[7,2],[8,2],[9,2],[10,2],[11,2]],idx:0,dir:1},
  {pts:[[10,4],[10,5],[10,6],[10,7],[10,8]],idx:0,dir:1},
  {pts:[[5,7],[6,7],[7,7],[8,7],[9,7],[10,7],[11,7],[12,7],[13,7]],idx:4,dir:-1}
];

function newGame(){
  mines=Array.from({length:ROWS},()=>new Array(COLS).fill(false));
  const cells=[];
  for(let y=0;y<ROWS;y++)for(let x=0;x<COLS;x++){
    if((x===START.x&&y===START.y)||(x===EXIT.x&&y===EXIT.y))continue;
    let onPath=false;
    for(const p of SENT_PATHS)if(p.pts.some(pt=>pt[0]===x&&pt[1]===y))onPath=true;
    if(onPath)continue;
    cells.push([x,y]);
  }
  for(let i=cells.length-1;i>0;i--){const j=Math.floor(Math.random()*(i+1));[cells[i],cells[j]]=[cells[j],cells[i]];}
  for(let i=0;i<MINES;i++){mines[cells[i][1]][cells[i][0]]=true;}
  numGrid=Array.from({length:ROWS},()=>new Array(COLS).fill(0));
  for(let y=0;y<ROWS;y++)for(let x=0;x<COLS;x++){
    let n=0;
    for(let dy=-1;dy<=1;dy++)for(let dx=-1;dx<=1;dx++){
      if(!dx&&!dy)continue;
      const nx=x+dx,ny=y+dy;
      if(nx>=0&&nx<COLS&&ny>=0&&ny<ROWS&&mines[ny][nx])n++;
    }
    numGrid[y][x]=n;
  }
  revealed=Array.from({length:ROWS},()=>new Array(COLS).fill(0));
  player={x:START.x,y:START.y,dx:1,dy:0};
  sentinels=SENT_PATHS.map(p=>({pts:p.pts.map(q=>q.slice()),idx:p.idx,dir:p.dir}));
  steps=0;startTime=performance.now();endTime=0;overReason='';winFlag=false;deathPos=null;
  particles=[];sentTimer=0;scanFlash=0;
  revealAround(player.x,player.y,true);
  state='playing';
}

function revealAround(cx,cy,silent){
  let added=0;const now=performance.now();
  for(let dy=-SCAN_R;dy<=SCAN_R;dy++)for(let dx=-SCAN_R;dx<=SCAN_R;dx++){
    const x=cx+dx,y=cy+dy;
    if(x<0||x>=COLS||y<0||y>=ROWS)continue;
    if(!revealed[y][x]){revealed[y][x]=now;added++;}
  }
  if(added>0&&!silent){scanFlash=1;sfx(760+Math.random()*240,0.05,'triangle',0.06);}
  else if(!silent)sfx(200,0.03,'square',0.05);
}

function cheb(ax,ay,bx,by){return Math.max(Math.abs(ax-bx),Math.abs(ay-by));}

function sentinelPos(s){const p=s.pts[s.idx];return{x:p[0],y:p[1]};}

function checkSpotted(){
  for(const s of sentinels){
    const p=sentinelPos(s);
    if(cheb(p.x,p.y,player.x,player.y)<=SEE_R){die('SPOTTED BY SENTINEL');return true;}
  }
  return false;
}

function die(reason){
  state='over';overReason=reason;winFlag=false;
  deathPos={x:player.x,y:player.y};endTime=performance.now();
  boom();
  const c=cellCenter(player.x,player.y);
  for(let i=0;i<60;i++){
    const a=Math.random()*Math.PI*2,sp=40+Math.random()*260;
    particles.push({x:c.x,y:c.y,vx:Math.cos(a)*sp,vy:Math.sin(a)*sp,life:1,decay:0.7+Math.random()*0.8,
      color:['#ff5533','#ffaa33','#ffee66','#ff2222'][Math.floor(Math.random()*4)],size:2+Math.random()*4});
  }
}

function winGame(){
  state='over';winFlag=true;endTime=performance.now();
  jingle();
  const c=cellCenter(EXIT.x,EXIT.y);
  for(let i=0;i<50;i++){
    const a=Math.random()*Math.PI*2,sp=30+Math.random()*180;
    particles.push({x:c.x,y:c.y,vx:Math.cos(a)*sp,vy:Math.sin(a)*sp-60,life:1,decay:0.5+Math.random()*0.6,
      color:['#44ffcc','#44ddff','#88ffee','#ffffff'][Math.floor(Math.random()*4)],size:2+Math.random()*3});
  }
}

function movePlayer(dx,dy){
  if(state!=='playing')return;
  player.dx=dx;player.dy=dy;
  const nx=player.x+dx,ny=player.y+dy;
  if(nx<0||nx>=COLS||ny<0||ny>=ROWS)return;
  sfx(240,0.03,'square',0.05);
  player.x=nx;player.y=ny;steps++;
  if(mines[ny][nx]){die('STEPPED ON A MINE');return;}
  revealAround(nx,ny,false);
  if(checkSpotted())return;
  if(nx===EXIT.x&&ny===EXIT.y)winGame();
}

function updateSentinels(dt){
  sentTimer+=dt;
  while(sentTimer>=SENT_INT){
    sentTimer-=SENT_INT;
    for(const s of sentinels){
      let ni=s.idx+s.dir;
      if(ni<0||ni>=s.pts.length){s.dir*=-1;ni=s.idx+s.dir;}
      s.idx=ni;
    }
    if(state==='playing')checkSpotted();
  }
}

    ctx.fillStyle='#ff3311';
    ctx.beginPath();ctx.arc(c.x,c.y,cs*0.42,0,Math.PI*2);ctx.fill();
    txt('✖',c.x,c.y+1,cs*0.6,'#fff','center',true);
  }

  // particles
  for(const p of particles){
    ctx.globalAlpha=Math.max(0,p.life);
    ctx.fillStyle=p.color;
    ctx.fillRect(p.x-p.size/2,p.y-p.size/2,p.size,p.size);
  }
  ctx.globalAlpha=1;

  // HUD
  if(state!=='start'){
    const t=(state==='playing'?now:endTime)-startTime;
    const sec=Math.max(0,Math.floor(t/1000));
    txt('MINESWEEPER STEALTH',bx,26,15,'#3a5566','left',true);
    txt('STEPS '+steps,bx+COLS*cs-140,26,13,'#8fd0ff','right');
    txt('TIME '+sec+'s',bx+COLS*cs-14,26,13,'#8fd0ff','right');
    txt('REACH THE EXIT · AVOID MINES & SENTINEL VISION',bx+COLS*cs/2,26,11,'#5d7a8a','center');
  }

  // overlays
  if(state==='start')drawStart(now);
  else if(state==='over')drawOver(now);
}

function drawStart(now){
  ctx.fillStyle='rgba(5,8,11,0.82)';ctx.fillRect(0,0,W,H);
  const cx=W/2,cy=H/2;
  txt('MINESWEEPER',cx,cy-90,Math.min(46,W*0.07),'#5dff9d','center',true);

function drawOver(now){
  const a=Math.min(1,(now-endTime)/400);
  ctx.fillStyle='rgba(5,8,11,'+(0.7*a)+')';ctx.fillRect(0,0,W,H);
  if(a<0.3)return;
  const cx=W/2,cy=H/2;
  const sec=Math.max(0,Math.floor((endTime-startTime)/1000));
  if(winFlag){
    txt('EXTRACTED!',cx,cy-70,Math.min(52,W*0.08),'#44ffcc','center',true);
    txt('You cleared the minefield.',cx,cy-20,18,'#9fc3d4','center');
  }else{
    txt('MISSION FAILED',cx,cy-70,Math.min(52,W*0.08),'#ff5d5d','center',true);
    txt(overReason,cx,cy-20,18,'#ffb3a0','center');
  }
  txt('STEPS '+steps+'   ·   TIME '+sec+'s',cx,cy+16,16,'#8fd0ff','center');
  if(Math.floor(now/500)%2===0)
    txt('— PRESS ENTER OR R TO RETRY —',cx,cy+70,18,'#ffffff','center',true);
}

// ---------- loop ----------
let lastT=performance.now();
function loop(now){
  const dt=Math.min(100,now-lastT);lastT=now;
  if(state==='playing')updateSentinels(dt);
  scanFlash=Math.max(0,scanFlash-dt/300);
  for(let i=particles.length-1;i>=0;i--){
    const p=particles[i];
    p.x+=p.vx*dt/1000;p.y+=p.vy*dt/1000;
    p.vy+=160*dt/1000;p.life-=p.decay*dt/1000;
    if(p.life<=0)particles.splice(i,1);
  }
  draw(now);
  requestAnimationFrame(loop);
}
requestAnimationFrame(loop);
