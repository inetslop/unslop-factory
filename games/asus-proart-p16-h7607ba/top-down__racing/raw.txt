function step(dt){
  if (state === 'countdown'){
    countT -= dt;
    var n = Math.ceil(countT);
    if (n <= 3 && n >= 1 && n !== step._lastC){ step._lastC = n; beep(440,.12); }
    if (countT <= 0){ state='racing'; raceStart=performance.now(); lastLapStart=raceStart; step._lastC=0; beep(880,.3,.2); }
    return;
  }
  if (state !== 'racing') return;

  var thr = (keys['w']||keys['arrowup']) ? 1 : 0;
  var brk = (keys['s']||keys['arrowdown']) ? 1 : 0;
  var st  = ((keys['a']||keys['arrowleft']) ? -1 : 0) + ((keys['d']||keys['arrowright']) ? 1 : 0);

  var off = !onTrack(car.x, car.y);
  if (thr) car.speed += 560*dt;
  if (brk){
    if (car.speed > 10) car.speed -= 700*dt;
    else car.speed -= 320*dt; // reverse

  car.speed *= Math.pow(off ? 0.12 : 0.55, dt);
  var maxSp = off ? 240 : 500, minRev = -160;

  if (car.speed < minRev) car.speed = minRev;

  var grip = Math.min(Math.abs(car.speed)/180, 1);
  var dir = car.speed >= 0 ? 1 : -1;
  car.angle += st * 2.6 * grip * dir * dt;

  car.x += Math.cos(car.angle) * car.speed * dt;
  car.y += Math.sin(car.angle) * car.speed * dt;

  // keep in world
  car.x = Math.max(30, Math.min(W-30, car.x));
  car.y = Math.max(30, Math.min(H-30, car.y));

  }

  updateEngine(car.speed);
}

      ctx.fillStyle = (row + c) % 2 ? '#fff' : '#111';
      ctx.fillRect(xA + c*sq, yTop + row*sq, Math.min(sq, xB - xA - c*sq), sq);
    }
  }
}

function roundRect(x,y,w,h,r){
  ctx.beginPath();

  ctx.arcTo(x,y+h,x,y,r);
  ctx.arcTo(x,y,x+w,y,r);
  ctx.closePath();
}

function drawMinimap(){
