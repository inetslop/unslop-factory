function initAudio(){try{if(!AC)AC=new (window.AudioContext||window.webkitAudioContext)();if(AC&&AC.state==='suspended')AC.resume();}catch(e){}}
function tone(f,dur,type,vol,slide){if(!AC)return;try{const t=AC.currentTime,o=AC.createOscillator(),g=AC.createGain();o.type=type||'square';o.frequency.setValueAtTime(f,t);if(slide)o.frequency.exponentialRampToValueAtTime(Math.max(30,f+slide),t+dur);g.gain.setValueAtTime(vol||.12,t);g.gain.exponentialRampToValueAtTime(.001,t+dur);o.connect(g);g.connect(AC.destination);o.start(t);o.stop(t+dur);}catch(e){}}
const sfx={
  shoot:()=>tone(650,.1,'square',.06,-350),
  hit:()=>tone(180,.12,'sawtooth',.1,-90),
  place:()=>{tone(440,.08,'triangle',.14);setTimeout(()=>tone(660,.1,'triangle',.12),70)},
  death:()=>tone(260,.2,'sawtooth',.11,-180),
  hurt:()=>{tone(120,.35,'square',.14,-70);setTimeout(()=>tone(90,.3,'square',.12,-40),120)},
  turn:()=>tone(330,.07,'triangle',.08,60),
  up:()=>{tone(520,.08,'triangle',.12);setTimeout(()=>tone(780,.1,'triangle',.12),80)},
  win:()=>[523,659,784,1046].forEach((f,i)=>setTimeout(()=>tone(f,.25,'triangle',.14),i*140)),
  lose:()=>[400,320,240,150].forEach((f,i)=>setTimeout(()=>tone(f,.3,'sawtooth',.12),i*160))
};

/* ---------- grid & path ---------- */
const COLS=12,ROWS=8;
const WP=[[-1,1],[3,1],[3,5],[6,5],[6,2],[9,2],[9,6],[11,6]];
const segs=[];let totalLen=0;
for(let i=0;i<WP.length-1;i++){const a=WP[i],b=WP[i+1],len=Math.abs(b[0]-a[0])+Math.abs(b[1]-a[1]);segs.push({x0:a[0],y0:a[1],x1:b[0],y1:b[1],len});totalLen+=len;}
const pathTiles=new Set();
for(const s of segs){let c=Math.min(s.x0,s.x1),r=s.y0;const dc=Math.sign(s.x1-s.x0);if(dc===0){c=s.x0;for(let y=Math.min(s.y0,s.y1);y<=Math.max(s.y0,s.y1);y++)pathTiles.add(c+','+y);}else{for(let x=c;x!==s.x1+dc;x+=dc,pathTiles.add(x+','+r));}}
function posAt(d){if(d<=0)return{x:WP[0][0],y:WP[0][1]};let rem=d;for(const s of segs){if(rem<s.len||s===segs[segs.length-1]){const f=Math.min(1,rem/s.len);return{x:s.x0+(s.x1-s.x0)*f,y:s.y0+(s.y1-s.y0)*f};}rem-=s.len;}const p=WP[WP.length-1];return{x:p[0],y:p[1]};}

/* ---------- tower types ---------- */
const TT={
  gun:{name:'Gun',cost:50,range:2.3,dmg:12,color:'#4da3ff'},
  cannon:{name:'Cannon',cost:90,range:2.1,dmg:22,splash:1.3,color:'#ff8c42'},
  frost:{name:'Frost',cost:60,range:1.9,dmg:6,slow:true,color:'#7ef0e0'},
  sniper:{name:'Sniper',cost:140,range:4.6,dmg:50,color:'#c48cff'}
};
const upCost=t=>Math.round(TT[t.type].cost*0.6)*t.lv;

/* ---------- game state ---------- */
let G=null;
function makeEnemy(type,dist,w){
  const base={grunt:{hp:30+10*w,sp:1,rw:8,r:.28,c:'#e74c3c'},
              fast:{hp:20+6*w,sp:2,rw:10,r:.22,c:'#f1c40f'},
              tank:{hp:90+25*w,sp:.5,rw:18,r:.34,c:'#8fa3b8'},
              boss:{hp:300+70*w,sp:.75,rw:80,r:.45,c:'#ff5edb'}}[type];
  return{type,dist,hp:base.hp,maxhp:base.hp,speed:base.sp,reward:base.rw,r:base.r,color:base.c,slow:0};
}
function nextWave(){
  G.wave++;
  if(G.wave>15){return win();}
  const w=G.wave,arr=[];
  for(let i=0;i<4+w*2;i++)arr.push('grunt');
  if(w>=3)for(let i=0;i<Math.floor((w-1)/2);i++)arr.push('fast');
  if(w>=4)for(let i=0;i<Math.floor((w-2)/2);i++)arr.push('tank');
  if(w%5===0)arr.push('boss');
  for(let i=arr.length-1;i>0;i--){const j=Math.random()*(i+1)|0;[arr[i],arr[j]]=[arr[j],arr[i]];}
  G.enemies=arr.map((t,i)=>makeEnemy(t,-i*0.6,w));
}
function newGame(){
  G={mode:'playing',money:140,lives:20,wave:0,turn:0,enemies:[],towers:[],shots:[],parts:[],placing:null,sel:null,phase:'',moveAt:0,spawnAt:0};
  nextWave();updatePanel();hideOverlays();refreshHUD();
}

/* ---------- turn logic ---------- */
function pickTarget(t){let best=null,bd=-1;for(const e of G.enemies){if(e.hp<=0)continue;const p=posAt(e.dist),d=Math.hypot(p.x-t.col,p.y-t.row);if(d<=t.range&&e.dist>bd){bd=e.dist;best=e;}}return best;}
function endTurn(){
  if(!G||G.mode!=='playing')return;
  sfx.turn();G.turn++;G.money+=2;G.mode='anim';G.phase='shots';G.moveAt=0;G.spawnAt=0;
  const shots=[];let i=0;
  for(const t of G.towers){const e=pickTarget(t);if(e){const p=posAt(e.dist);t.angle=Math.atan2(p.y-t.row,p.x-t.col);shots.push({tw:t,sx:t.col,sy:t.row,tx:p.x,ty:p.y,target:e,start:performance.now()+i*150,dur:300,done:false,fired:false});i++;}}
  G.shots=shots;
}
function applyShot(s){
  const t=s.tw;
  if(t.splash){for(const e of G.enemies){if(e.hp<=0)continue;const p=posAt(e.dist);if(Math.hypot(p.x-s.tx,p.y-s.ty)<=t.splash)e.hp-=t.dmg;}sfx.hit();}
  else{const e=s.target;if(e.hp>0){e.hp-=t.dmg;if(t.slow)e.slow=1;sfx.hit();}}
  burst(s.tx,s.ty,t.color,8);
  checkDeaths();
}
function checkDeaths(){
  const dead=G.enemies.filter(e=>e.hp<=0);
  if(!dead.length)return;
  for(const e of dead){G.money+=e.reward;const p=posAt(e.dist);burst(p.x,p.y,e.color,14);sfx.death();}
  G.enemies=G.enemies.filter(e=>e.hp>0);
}
function doMove(){
  for(let i=G.enemies.length-1;i>=0;i--){const e=G.enemies[i];e.dist+=e.speed*(e.slow?0.5:1);e.slow=0;
    if(e.dist>=totalLen){G.enemies.splice(i,1);G.lives--;burst(WP[WP.length-1][0],WP[WP.length-1][1],'#ff4444',16);sfx.hurt();}}
  refreshHUD();
  if(G.lives<=0)gameOver();
}
function gameOver(){G.mode='gameover';sfx.lose();showEnd(false);}
function win(){G.mode='victory';sfx.win();showEnd(true);}

/* ---------- particles ---------- */
function burst(tx,ty,color,n){for(let i=0;i<n;i++){const a=Math.random()*Math.PI*2,v=.04+Math.random()*.1;G.parts.push({x:tx,y:ty,vx:Math.cos(a)*v,vy:Math.sin(a)*v,life:1,color,r:.05+Math.random()*.07});}}

/* ---------- canvas ---------- */
const cv=document.getElementById('cv'),ctx=cv.getContext('2d');
let cw=0,ch=0,dpr=1;
function resize(){dpr=window.devicePixelRatio||1;cw=cv.clientWidth;ch=cv.clientHeight;cv.width=Math.max(1,cw*dpr);cv.height=Math.max(1,ch*dpr);}
window.addEventListener('resize',resize);
function geom(){const cell=Math.min(cw/COLS,ch/ROWS),ox=(cw-cell*COLS)/2,oy=(ch-cell*ROWS)/2;return{cell,ox,oy};}

let hover={c:-1,r:-1,inside:false};
cv.addEventListener('mousemove',e=>{const r=cv.getBoundingClientRect(),{cell,ox,oy}=geom();const c=Math.floor((e.clientX-r.left-ox)/cell),rr=Math.floor((e.clientY-r.top-oy)/cell);hover.inside=c>=0&&c<COLS&&rr>=0&&rr<ROWS;hover.c=c;hover.r=rr;});
cv.addEventListener('mouseleave',()=>hover.inside=false);
cv.addEventListener('contextmenu',e=>{e.preventDefault();cancelPlacing();G&&(G.sel=null,updatePanel());});

function tileFree(c,r){if(c<0||c>=COLS||r<0||r>=ROWS)return false;if(pathTiles.has(c+','+r))return false;return !G.towers.some(t=>t.col===c&&t.row===r);}
function towerAt(c,r){return G.towers.find(t=>t.col===c&&t.row===r)||null;}

cv.addEventListener('click',()=>{
  initAudio();if(!G||G.mode!=='playing'||!hover.inside)return;
  const c=hover.c,r=hover.r;
  if(G.placing){const def=TT[G.placing];
    if(tileFree(c,r)&&G.money>=def.cost){G.money-=def.cost;
      G.towers.push({type:G.placing,col:c,row:r,lv:1,dmg:def.dmg,range:def.range,splash:def.splash||0,slow:!!def.slow,color:def.color,angle:-Math.PI/2,invested:def.cost});
      sfx.place();refreshHUD();}
    return;}
  const t=towerAt(c,r);G.sel=t;updatePanel();
});

const $=id=>document.getElementById(id);
function refreshHUD(){$('money').textContent=G.money;$('lives').textContent=G.lives;$('wave').textContent=G.wave;$('turn').textContent=G.turn;
  document.querySelectorAll('.tbtn').forEach(b=>{const t=b.dataset.type;b.classList.toggle('poor',G.money<TT[t].cost);b.classList.toggle('sel',G.placing===t);});}

  const p=$('panel');if(!G||!G.sel){p.classList.add('hidden');return;}
  const t=G.sel;p.classList.remove('hidden');
  $('pInfo').innerHTML=`<span class="dot" style="background:${t.color};display:inline-block;width:12px;height:12px;border-radius:50%;margin-right:6px"></span><b>${TT[t.type].name}</b> Lv${t.lv} · dmg ${t.dmg}${t.slow?' · slows':''}${t.splash?' · splash':''}`;
  const ub=$('upBtn');

}
function doUpgrade(){if(!G||G.mode!=='playing'||!G.sel)return;const t=G.sel;if(t.lv>=3)return;const c=upCost(t);if(G.money<c)return;G.money-=c;t.invested+=c;t.lv++;t.dmg=Math.round(t.dmg*1.5);t.range+=0.25;sfx.up();refreshHUD();updatePanel();}
function doSell(){if(!G||G.mode!=='playing'||!G.sel)return;const t=G.sel;G.money+=Math.floor(t.invested*0.7);G.towers.splice(G.towers.indexOf(t),1);G.sel=null;sfx.hit();refreshHUD();updatePanel();}

/* ---------- draw ---------- */
function draw(now){
  requestAnimationFrame(draw);
  ctx.setTransform(dpr,0,0,dpr,0,0);
  ctx.clearRect(0,0,cw,ch);
  if(!G)return;
  const {cell,ox,oy}=geom(),P=(x,y)=>[ox+x*cell+cell/2,oy+y*cell+cell/2];

  /* tiles */
  for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++){
    const isPath=pathTiles.has(c+','+r);
    ctx.fillStyle=isPath?(c+r)%2?'#d9b26f':'#cfab64':(c+r)%2?'#2e4a35':'#31503a';
    ctx.fillRect(ox+c*cell,oy+r*cell,cell+1,cell+1);
  }
  /* entry / exit markers */
  {const [ex,ey]=P(-0.35,1);ctx.fillStyle='#3ecf6e';ctx.beginPath();ctx.arc(ex,ey,cell*.28,0,7);ctx.fill();}
  {const [sx,sy]=P(11,6);ctx.fillStyle='#ff4444';ctx.font=`${cell*.5}px system-ui`;ctx.textAlign='center';ctx.textBaseline='middle';ctx.fillText('☠',sx,sy);}

  /* placement ghost */
  if(G.mode==='playing'&&G.placing&&hover.inside){
    const ok=tileFree(hover.c,hover.r)&&G.money>=TT[G.placing].cost;
    ctx.fillStyle=ok?'#3ecf6e55':'#ff444455';ctx.fillRect(ox+hover.c*cell,oy+hover.r*cell,cell,cell);
    const def=TT[G.placing];
    ctx.beginPath();const [gx,gy]=P(hover.c,hover.r);ctx.arc(gx,gy,def.range*cell,0,7);ctx.fillStyle='#ffffff14';ctx.fill();ctx.strokeStyle='#ffffff55';ctx.setLineDash([6,5]);ctx.stroke();ctx.setLineDash([]);
  }

  /* selected tower range */
  if(G.sel){const t=G.sel,[tx,ty]=P(t.col,t.row);
    ctx.beginPath();ctx.arc(tx,ty,t.range*cell,0,7);ctx.fillStyle='#ffd76a18';ctx.fill();ctx.strokeStyle='#ffd76a99';ctx.setLineDash([6,5]);ctx.stroke();ctx.setLineDash([]);}

  /* towers */
  for(const t of G.towers){const [x,y]=P(t.col,t.row);
    ctx.save();ctx.translate(x,y);
    ctx.rotate(t.angle);ctx.fillStyle='#20293c';ctx.fillRect(0,-cell*.08,cell*.52,cell*.16);ctx.restore();
    ctx.beginPath();ctx.arc(x,y,cell*.32,0,7);ctx.fillStyle=t.color;ctx.fill();ctx.lineWidth=2;ctx.strokeStyle='#141a26';ctx.stroke();
    if(G.sel===t){ctx.beginPath();ctx.arc(x,y,cell*.38,0,7);ctx.strokeStyle='#ffd76a';ctx.lineWidth=3;ctx.stroke();}
    for(let i=0;i<t.lv-1;i++){ctx.beginPath();ctx.arc(x-cell*.14+i*cell*.14,y+cell*.42,cell*.05,0,7);ctx.fillStyle='#fff';ctx.fill();}
  }

  /* enemies */
  for(const e of G.enemies){const p=posAt(e.dist),[x,y]=P(p.x,p.y);if(x<ox-cell)continue;
    const r=e.r*cell;
    if(e.slow){ctx.beginPath();ctx.arc(x,y,r+5,0,7);ctx.strokeStyle='#7ef0e0aa';ctx.lineWidth=2;ctx.stroke();}
    ctx.fillStyle=e.color;ctx.strokeStyle='#141a26';ctx.lineWidth=2;
    ctx.beginPath();
    if(e.type==='fast'){ctx.moveTo(x,y-r);ctx.lineTo(x+r,y+r*.8);ctx.lineTo(x-r,y+r*.8);ctx.closePath();}
    else if(e.type==='tank'){ctx.rect(x-r,y-r,r*2,r*2);}
    else{ctx.arc(x,y,r,0,7);}
    ctx.fill();ctx.stroke();
    const bw=r*2.2,bh=Math.max(3,cell*.08),by=y-r-cell*.16;
    ctx.fillStyle='#000a';ctx.fillRect(x-bw/2-1,by-1,bw+2,bh+2);
    ctx.fillStyle=e.hp/e.maxhp>.4?'#3ecf6e':'#ff5e5e';
    ctx.fillRect(x-bw/2,by,bw*Math.max(0,e.hp/e.maxhp),bh);
  }

  /* shots */
  for(const s of G.shots){
    if(s.done)continue;
    const t=(now-s.start)/s.dur;
    if(t<0)continue;
    const [ax,ay]=P(s.sx,s.sy),[bx,by]=P(s.tx,s.ty);
    if(t>=1)continue;
    const x=ax+(bx-ax)*t,y=ay+(by-ay)*t;
    ctx.strokeStyle=s.tw.color+'cc';ctx.lineWidth=2;ctx.beginPath();ctx.moveTo(ax,ay);ctx.lineTo(x,y);ctx.stroke();
    ctx.beginPath();ctx.arc(x,y,cell*.09,0,7);ctx.fillStyle=s.tw.color;ctx.fill();
  }

  /* particles */
  for(let i=G.parts.length-1;i>=0;i--){const p=G.parts[i];p.x+=p.vx;p.y+=p.vy;p.life-=.04;
    if(p.life<=0){G.parts.splice(i,1);continue;}
    const [x,y]=P(p.x,p.y);ctx.globalAlpha=p.life;ctx.beginPath();ctx.arc(x,y,p.r*cell,0,7);ctx.fillStyle=p.color;ctx.fill();ctx.globalAlpha=1;}

  /* animation phases */
  if(G.mode==='anim'){
    if(G.phase==='shots'){
      for(const s of G.shots)if(!s.done){
        if(now>=s.start&&!s.fired){s.fired=true;sfx.shoot();}
        if(now>=s.start+s.dur){s.done=true;applyShot(s);}
      }
      if(G.shots.length&&G.shots.every(s=>s.done)){G.phase='move';G.moveAt=now+350;}
      else if(!G.shots.length){G.phase='move';G.moveAt=now+200;}
    }else if(G.phase==='move'&&now>=G.moveAt){doMove();if(G.mode!=='gameover'){G.phase='spawn';G.spawnAt=now+300;}}
    else if(G.phase==='spawn'&&now>=G.spawnAt){G.mode='playing';refreshHUD();if(G.enemies.length===0)nextWave();}
  }
  refreshHUD();
}
